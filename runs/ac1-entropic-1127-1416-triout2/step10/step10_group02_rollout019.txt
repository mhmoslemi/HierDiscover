    a = a0.copy()
    n = len(a)
    best_a = a.copy()
    best_val = compute_score(a)
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    t = 0
    while time.time() < deadline:
        t += 1
        # Compute convex combination coefficient
        alpha = min(alpha_end, alpha_start + (alpha_end - alpha_start) * (t / 20000.0))
        conv = full_convolution(a)
        max_c = conv.max()
        # stable softmax
        exp_vals = np.exp(alpha * (conv - max_c))

        p_rev = p[::-1]
        corr = linear_convolution(a, p_rev)  # length 3n-2
        grad = 2.0 * corr[(len(p)-1) - np.arange(n)]
        # Adam updates

"""
Non‑negative step‑function optimiser for ``evaluate_sequence``.

The optimiser works in three phases:
1. **Exploratory sampling** – generate a diverse pool of candidates using
   analytic families (power‑law, symmetric product‑law, exponentials,
   beta‑shaped densities) and random Dirichlet draws.  The best few
   candidates (by the exact objective 2·n·max(conv)) are retained.
2. **Smooth‑max gradient descent** – starting from each retained candidate,
   minimise a smooth approximation of the max‑convolution
   `L(a) = (1/α)·log Σ_k exp(α·(a*a)[k])`.  The gradient of `L` is
   analytically obtained as a correlation of `a` with the soft‑max
   weights, which is computed efficiently by FFT.  An Adam‑style update,
   Euclidean projection onto the probability simplex and a forced symmetry
   (`a[i]=a[n‑1‑i]`) keep the iterates feasible.
3. **Final greedy hill‑climb** – a cheap stochastic mass‑transfer
   neighbourhood is explored with greedy acceptance to polish the result.

All phases respect the remaining wall‑clock budget; the best sequence found
is returned as a plain Python list of floats.
"""

import time
import numpy as np

# ------------------------------------------------------------
# Numerical utilities
# ------------------------------------------------------------

def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / v.shape[0])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, v.shape[0] + 1)
    mask = u - cssv / ind > 0
    rho = np.where(mask)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def full_convolution(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a using a real FFT (length 2·n‑1)."""
    n = a.shape[0]
    m = 2 * n - 1
    n_fft = 1 << ((m - 1).bit_length())          # next power of two ≥ m
    A = np.fft.rfft(a, n=n_fft)
    C = A * A
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


def linear_convolution(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Linear convolution a * b (full length)."""
    size = a.shape[0] + b.shape[0] - 1
    n_fft = 1 << ((size - 1).bit_length())
    A = np.fft.rfft(a, n=n_fft)
    B = np.fft.rfft(b, n=n_fft)
    C = A * B
    conv = np.fft.irfft(C, n=n_fft)[:size]
    return conv


def compute_score(a: np.ndarray) -> float:
    """Exact objective for a normalised vector (sum = 1):
       2·n·max(conv(a,a))
    """
    n = a.shape[0]
    conv = full_convolution(a)
    return float(2.0 * n * conv.max())


# ------------------------------------------------------------
# Families for random initialisation
# ------------------------------------------------------------

def power_law_vector(n: int, p: float) -> np.ndarray:
    a = np.arange(1, n + 1, dtype=float) ** (-p)
    a /= a.sum()
    return a


def symmetric_product_vector(n: int, p: float) -> np.ndarray:
    i = np.arange(1, n + 1, dtype=float)
    prod = (i * (n - i + 1)) ** (-p)
    prod /= prod.sum()
    return prod


def exponential_vector(n: int, lam: float) -> np.ndarray:
    a = np.exp(-lam * np.arange(n, dtype=float))
    a /= a.sum()
    return a


def symmetric_exponential_vector(n: int, lam: float) -> np.ndarray:
    idx = np.minimum(np.arange(n), np.arange(n)[::-1])
    a = np.exp(-lam * idx.astype(float))
    a /= a.sum()
    return a


def beta_vector(n: int, a: float, b: float) -> np.ndarray:
    """Discrete Beta( a , b ) density on {1,…,n}."""
    i = np.arange(1, n + 1, dtype=float)
    vals = (i ** (a - 1.0)) * ((n - i + 1) ** (b - 1.0))
    # protect against NaNs when a or b < 1 (i.e. negative powers)
    vals = np.maximum(vals, 0.0)
    total = vals.sum()
    if total == 0.0:
        return np.full(n, 1.0 / n)
    return vals / total


def random_dirichlet_vector(n: int, scale: float = 1.0) -> np.ndarray:
    """Dirichlet draw with random concentration (controls sparsity)."""
    alpha = np.random.rand(n) * scale + 0.05   # avoid exact zeros
    sample = np.random.gamma(shape=alpha, scale=1.0)
    sample /= sample.sum()
    return sample


# ------------------------------------------------------------
# Optimisation primitives
# ------------------------------------------------------------

def smooth_descent_optimize(a0: np.ndarray, deadline: float) -> np.ndarray:
    """
    Adam optimiser on the smooth approximation of max‑convolution.
    The loss is L(a)= (1/α)·log Σ_k exp(α·(a*a)[k]).
    """
    a = a0.copy()
    n = a.shape[0]
    best_a, best_val = a.copy(), compute_score(a)

    # Adam parameters
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8
    lr = 0.2
    t = 0

    while time.time() < deadline:
        t += 1
        # slowly increase smoothing parameter α
        alpha = min(200.0, 0.5 + 0.01 * t)

        conv = full_convolution(a)
        max_c = conv.max()
        exp_vals = np.exp(alpha * (conv - max_c))   # reduce overflow
        p = exp_vals / exp_vals.sum()               # soft‑max weights (length 2n‑1)

        # gradient = 2 * correlation(a, p)
        p_rev = p[::-1]
        corr = linear_convolution(a, p_rev)                     # length 3n‑2
        grad = 2.0 * corr[(len(p) - 1) - np.arange(n)]

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)

        a = a - lr * m_hat / (np.sqrt(v_hat) + eps)

        # enforce symmetry and simplex feasibility
        a = (a + a[::-1]) / 2.0
        a = project_to_simplex(a)

        val = compute_score(a)
        if val < best_val:
            best_val, best_a = val, a.copy()

        # occasional tiny jitter to escape plateaux
        if t % 1000 == 0:
            jitter = np.random.randn(n) * (0.001 / np.sqrt(t))
            a += jitter
            a = project_to_simplex(a)

    return best_a


def subgradient_polish(a0: np.ndarray, deadline: float, step0: float = 0.12) -> np.ndarray:
    """
    Simple projected sub‑gradient descent on the true max‑convolution.
    Uses the active index of the current maximum.
    """
    a = a0.copy()
    n = a.shape[0]
    best_a, best_val = a.copy(), compute_score(a)
    t = 0
    while time.time() < deadline:
        t += 1
        conv = full_convolution(a)
        k = int(np.argmax(conv))
        idx = np.arange(n)
        j = k - idx
        mask = (j >= 0) & (j < n)
        g = np.zeros_like(a)
        g[mask] = 2.0 * a[j[mask]]
        step = step0 / np.sqrt(t)
        a = a - step * g
        a = (a + a[::-1]) / 2.0          # keep symmetry
        a = project_to_simplex(a)
        val = compute_score(a)
        if val < best_val:
            best_val, best_a = val, a.copy()
        if t % 1000 == 0:                # jitter to avoid stagnation
            jitter = np.random.randn(n) * (step0 * 0.01 / np.sqrt(t))
            a += jitter
            a = project_to_simplex(a)
    return best_a


def hill_climb(a_start: np.ndarray, deadline: float,
               max_no_improve: int = 300_000) -> np.ndarray:
    """
    Greedy stochastic hill‑climb: transfer a fraction of mass between
    random coordinates, always keeping the best point.
    """
    a = a_start.copy()
    n = a.shape[0]
    best_a = a.copy()
    best_val = compute_score(a)
    step_scale = 0.05
    it = 0
    no_improve = 0

    while time.time() < deadline and it < max_no_improve:
        i, j = np.random.randint(n, size=2)
        if i == j:
            continue
        delta = step_scale * a[i]
        cand = a.copy()
        cand[i] -= delta
        cand[j] += delta
        cand = np.maximum(cand, 0.0)
        cand = (cand + cand[::-1]) / 2.0      # enforce symmetry
        cand = project_to_simplex(cand)
        val = compute_score(cand)
        if val < best_val:
            best_val = val
            best_a = cand.copy()
            a = cand
            step_scale = min(step_scale * 1.07, 0.5)
            no_improve = 0
        else:
            step_scale *= 0.985
            no_improve += 1
        it += 1
    return best_a

def random_search_pool(n: int, deadline: float, top_k: int = 10):
    """
    Generate a heterogeneous pool of candidates, keeping the top_k according
    to the exact objective.  The function returns a list of (score, vector)
    sorted increasingly.
    """
    top = []   # list of (score, vector)

    def consider(vec):
        nonlocal top
        # enforce feasibility before scoring
        vec = project_to_simplex(vec)
        vec = (vec + vec[::-1]) / 2.0
        val = compute_score(vec)
        if len(top) < top_k:
            top.append((val, vec.copy()))
            top.sort(key=lambda x: x[0])
        elif val < top[-1][0]:
            top[-1] = (val, vec.copy())
            top.sort(key=lambda x: x[0])

    # deterministic families -------------------------------------------------
    consider(np.full(n, 1.0 / n))                         # uniform

    for p in np.linspace(0.4, 1.2, 13):                  # monotone power‑law
        consider(power_law_vector(n, p))
        consider(symmetric_product_vector(n, p))

    for lam in np.linspace(5e-4, 0.015, 12):            # exponential decays
        consider(exponential_vector(n, lam))
        consider(symmetric_exponential_vector(n, lam))

    for _ in range(30):                                 # random Beta shapes
        a = np.random.rand() * 0.8 + 0.1
        b = np.random.rand() * 0.8 + 0.1
        consider(beta_vector(n, a, b))

    # random Dirichlet draws -------------------------------------------------
    while time.time() < deadline:
        # occasionally favour sparse draws
        scale = np.random.rand() * 0.5 if np.random.rand() < 0.4 else np.random.rand() * 5 + 0.5
        consider(random_dirichlet_vector(n, scale=scale))

    return top
