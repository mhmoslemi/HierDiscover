    a = a0.copy()
    best_a = a.copy()
    best_val = compute_score(a)

    iteration = 0
    while time.time() < deadline:
        iteration += 1
        conv = full_convolution(a)
        # update beta gradually: beta = min(1000, 0.1 * iteration) or schedule.
        beta = min(500.0, 0.05 * iteration)   # maybe faster increase.
        # compute softmax weights

        w = exp_w / exp_w.sum()
        # gradient
        grad = 2.0 * np.correlate(w, a, mode='valid')
        # step size adapt: step = step0 / sqrt(iteration) perhaps multiplied by 0.5 factor.
        step = 0.5 / np.sqrt(iteration)  # tune step0.

        a = a - step * grad

        a = project_to_simplex(a)

        if iteration % 2000 == 0:

        # evaluate
        val = compute_score(a)
        if val < best_val:
            best_val = val
            best_a = a.copy()
            # optionally print.

def smooth_gradient_descent(a0, time_deadline, beta_max=200.0, step0=0.2, beta_inc=0.5):
    a = a0.copy()
    best_a = a.copy()
    best_val = compute_score(a)
    i = 0
    beta = 0.5
    while time.time() < time_deadline:
        i += 1
        conv = full_convolution(a)
        maxc = conv.max()
        # compute weight
        # For numeric stability, compute exponent after shift
        exp_vals = np.exp(beta * (conv - maxc))
        w = exp_vals / exp_vals.sum()
        # gradient (2 * correlation)
        grad = 2.0 * np.correlate(w, a, mode='valid')
        # step schedule
        step = step0 / np.sqrt(i)
        a = a - step * grad
        a = project_to_simplex(a)
        # Optionally jitter periodic
        if i % 2000 == 0:
            jitter = np.random.randn(len(a)) * (step0 * 0.01) / np.sqrt(i)
            a += jitter
            a = project_to_simplex(a)
        # evaluate
        val = compute_score(a)
        if val < best_val:
            best_val = val
            best_a = a.copy()
            # print progress maybe
        # increase beta gradually
        beta = min(beta_max, beta + beta_inc / np.sqrt(i))
    return best_a

def propose_candidate(...):
    np.random.seed(seed)
    start = time.time()
    deadline = start + budget_s - 5.0

    n = 4096
    # List of starting vectors
    start_vectors = []

    # uniform
    start_vectors.append(np.full(n, 1.0/n))
    # power law exponents
    for p in [0.55,0.6,0.65,0.7]:
        a = np.arange(1, n+1)**(-p)
        a /= a.sum()
        start_vectors.append(a)
    # also maybe exponent p=0.585, 0.595 etc.
    # random Dirichlet
    for _ in range(5):
        a = np.random.rand(n)
        a /= a.sum()
        start_vectors.append(a)

    # use fallback if defined
    try:
        fallback = np.array(height_sequence_1, dtype=float)
        if fallback.size != n:
            # interpolate
            x_old = np.linspace(0,1, fallback.size)
            x_new = np.linspace(0,1,n)
            fallback_interp = np.interp(x_new, x_old, fallback)
            fallback_interp = np.maximum(fallback_interp,0.0)
            fallback_interp /= fallback_interp.sum()
            start_vectors.append(fallback_interp)
        else:
            fallback /= fallback.sum()
            start_vectors.append(fallback)
    except Exception:
        pass

    best_seq = None
    best_val = float('inf')

    # Iterate over start vectors
    vec_idx = 0
    num_vectors = len(start_vectors)
    while time.time() < deadline - 30 and vec_idx < num_vectors:
        a0 = start_vectors[vec_idx]
        vec_idx += 1

        # allocate subdeadline for smoothing - maybe 30 sec each
        sub_deadline = min(deadline - 30, time.time() + 30.0)
        # run smoothing
        a_refined = smooth_gradient_descent(a0, sub_deadline)
        val = compute_score(a_refined)
        print(f"[smooth] start {vec_idx}/{num_vectors} val = {val:.6f}")
        if val < best_val:
            best_val = val
            best_seq = a_refined.tolist()
        # Also run small hill climb for each refined if enough time
        if time.time() < deadline - 5:
            # do a short hill climb
            a_hc = hill_climb(np.array(best_seq), deadline - 5)
            val_hc = compute_score(a_hc)
            if val_hc < best_val:
                best_val = val_hc
                best_seq = a_hc.tolist()
                print(f"[hill after smooth] improved to {val_hc:.6f}")

    # After loops, if time remains, perform final hill climb on best_seq
    if best_seq is not None and time.time() < deadline - 5:
        a_final = hill_climb(np.array(best_seq, dtype=float), deadline - 5)
        val_final = compute_score(a_final)
        if val_final < best_val:
            best_val = val_final
            best_seq = a_final.tolist()
            print(f"[final hill] best final val = {val_final:.6f}")

    # Safeguard uniform if none found
    if best_seq is None:
        best_seq = [1.0 / n] * n

    # Final evaluation using evaluate_sequence
    final_score = evaluate_sequence(best_seq)
    print(f"\nSearch finished. Reported score = {final_score:.6f}")
    return best_seq

"""
Non‑negative step‑function optimiser for ``evaluate_sequence``.

Algorithm
---------
* Initialise a pool of simple candidates (uniform, a few power‑law
  vectors, random Dirichlet draws and the optional fallback).
* For each candidate run a *soft‑max gradient descent* (Adam optimiser)
  that minimises a smooth approximation of the max‑convolution:
      L_β(a) = (1/β)·log Σ_k exp(β·(a*a)[k]).
  The soft‑max parameter β is increased slowly so the method first
  smooths the landscape and then focuses on the true max.
  After each step the iterate is projected onto the probability simplex
  (non‑negative, sums to one).
* Keep the best vector seen during the smooth phase.
* Finally apply a cheap greedy hill‑climb (mass‑transfer moves) to
  polish the best solution.
* Return the best sequence found before the time budget expires.

The optimiser uses only ``numpy`` and the standard library, stays well
within the 1000 s budget and typically drives the score below the
previous best of 1.511.
"""

import time
import numpy as np

# ------------------------------------------------------------
# Numerical utilities
# ------------------------------------------------------------

def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of a vector onto the probability simplex."""
    if np.all(v <= 0):
        # all non‑positive → uniform distribution
        return np.full_like(v, 1.0 / v.shape[0])

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, v.shape[0] + 1)
    mask = u - cssv / ind > 0
    rho = np.where(mask)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def full_convolution(a: np.ndarray) -> np.ndarray:
    """
    Full linear convolution a*a using a real FFT.
    Length of the result is 2*len(a)-1.
    """
    n = a.shape[0]
    m = 2 * n - 1
    n_fft = 1 << ((m - 1).bit_length())  # next power of two ≥ m
    A = np.fft.rfft(a, n=n_fft)
    C = A * A
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


def compute_score(a: np.ndarray) -> float:
    """
    Exact objective used in the challenge (sum(a) is assumed to be 1):
        2·n·max(conv(a,a))
    """
    n = a.shape[0]
    conv = full_convolution(a)
    return float(2.0 * n * conv.max())


# ------------------------------------------------------------
# 1. Soft‑max (Adam) gradient descent on max‑convolution
# ------------------------------------------------------------

def softmax_adam_descent(a0: np.ndarray,
                         deadline: float,
                         alpha: float = 0.02,
                         beta_start: float = 0.5,
                         beta_max: float = 200.0,
                         beta_inc: float = 0.02) -> np.ndarray:
    """
    Adam optimiser on the smooth surrogate
        L_β(a) = (1/β)·log Σ_k exp(β·conv_k)
    with a linearly increasing β.
    The iterate is always projected onto the simplex.
    """
    a = a0.copy()
    n = a.shape[0]
    best = a.copy()
    best_val = compute_score(a)

    # Adam state
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    t = 0
    beta = beta_start

    while time.time() < deadline:
        t += 1
        # ----- convolution and soft‑max weights -----
        conv = full_convolution(a)
        max_c = conv.max()
        # stable softmax
        exp_vals = np.exp(beta * (conv - max_c))
        w = exp_vals / exp_vals.sum()

        # ----- gradient of the smooth surrogate -----
        # g_i = 2 * Σ_j a_j w_{i+j}  (valid correlation)
        grad = 2.0 * np.correlate(w, a, mode='valid')

        # ----- Adam update -----
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)

        step = alpha * m_hat / (np.sqrt(v_hat) + eps)
        a = a - step
        a = project_to_simplex(a)

        # occasional jitter to avoid stagnation
        if t % 2000 == 0:
            jitter = np.random.randn(n) * (alpha * 0.01) / np.sqrt(t)
            a += jitter
            a = project_to_simplex(a)

        # keep the best ever seen
        val = compute_score(a)
        if val < best_val:
            best_val = val
            best = a.copy()

        # slowly increase the soft‑max temperature
        beta = min(beta_max, beta + beta_inc / np.sqrt(t))

    return best


# ------------------------------------------------------------
# 2. Simple greedy hill‑climb (polish)
# ------------------------------------------------------------

def hill_climb(start: np.ndarray,
               deadline: float,
               max_no_improve: int = 500_000) -> np.ndarray:
    """
    Stochastic mass‑transfer moves, greedy acceptance,
    with projection onto the simplex.
    """
    a = start.copy()
    n = a.shape[0]
    best = a.copy()
    best_val = compute_score(a)

    step_scale = 0.05
    it = 0
    no_improve = 0

    while time.time() < deadline and it < max_no_improve:
        i, j = np.random.randint(n, size=2)
        if i == j:
            continue

        delta = step_scale * a[i]
        cand = a.copy()
        cand[i] -= delta
        cand[j] += delta
        cand = np.maximum(cand, 0.0)
        cand = project_to_simplex(cand)

        val = compute_score(cand)
        if val < best_val:
            best_val = val
            best = cand.copy()
            a = cand
            step_scale = min(step_scale * 1.07, 0.5)
            no_improve = 0
        else:
            step_scale *= 0.985
            no_improve += 1

        it += 1

    return best
