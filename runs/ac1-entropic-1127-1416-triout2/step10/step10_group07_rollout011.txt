"""
Propose a non‑negative step‑function (list of floats) that minimises
``evaluate_sequence``.  The search runs for up to the given time budget.
It combines a broad sweep of analytic families, a fast mirror‑subgradient
descent on the exact max‑convolution objective, an optional high‑β
Adam refinement of a smooth log‑sum‑exp surrogate, optional symmetrisation
and a tiny simulated‑annealing hill‑climb.  Longer vectors give a lower
theoretical lower bound (≈ 2·n·max/∑² → 1 as n→∞), so we use the largest
n fitting in the time budget.
"""

import time
import numpy as np


# ----------------------------------------------------------------------
# Basic utilities
# ----------------------------------------------------------------------
def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full aperiodic convolution using real FFT (fast, no Python loops)."""
    m = a.size + b.size - 1
    nfft = next_pow2(m)
    A = np.fft.rfft(a, n=nfft)
    B = np.fft.rfft(b, n=nfft)
    C = A * B
    conv = np.fft.irfft(C, n=nfft)[:m]
    return conv


def compute_score(a: np.ndarray) -> float:
    """
    Exact objective for a probability vector (sum(a)=1):
        2·n·max(a * a)
    """
    n = a.shape[0]
    conv = convolution_fft(a, a)
    return 2.0 * n * float(conv.max())


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection onto the probability simplex (sum = 1, v ≥ 0)."""
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / v.shape[0])

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, v.shape[0] + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.where(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def softmax(z: np.ndarray) -> np.ndarray:
    """Numerically stable soft‑max."""
    zmax = np.max(z)
    e = np.exp(z - zmax)
    return e / e.sum()


# ----------------------------------------------------------------------
# Analytic families (all normalised to sum = 1)
# ----------------------------------------------------------------------
def symmetric_product_family(n: int, alpha: float) -> np.ndarray:
    """a_i ∝ [(i+½)(n‑½‑i)]^(‑α), symmetric, normalized."""
    i = np.arange(n, dtype=np.float64)
    logprod = np.log(i + 0.5) + np.log(n - 0.5 - i)
    w = np.exp(-alpha * logprod)
    w[np.isinf(w)] = np.finfo(np.float64).max
    a = w / w.sum()
    return a


def left_heavy_power_family(n: int, beta: float) -> np.ndarray:
    """a_i ∝ (i+½)^(‑β), monotone decreasing, normalised."""
    i = np.arange(n, dtype=np.float64)
    w = np.exp(-beta * np.log(i + 0.5))
    a = w / w.sum()
    return a


def exponential_decay_family(n: int, gamma: float) -> np.ndarray:
    """a_i ∝ exp(-γ·i), monotone decreasing, normalised."""
    i = np.arange(n, dtype=np.float64)
    w = np.exp(-gamma * i)
    a = w / w.sum()
    return a


def triangular_family(n: int) -> np.ndarray:
    """Symmetric triangular shape, peak in the centre, normalised."""
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    w = 1.0 - np.abs(2.0 * x - 1.0)          # linear rise to centre
    a = w / w.sum()
    return a


def cosine_family(n: int) -> np.ndarray:
    """Cosine bump: a_i ∝ 1 + cos(π·(2x−1)), normalised."""
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    w = 1.0 + np.cos(np.pi * (2.0 * x - 1.0))
    a = w / w.sum()
    return a


def symmetric_beta_family(n: int, a: float) -> np.ndarray:
    """
    Symmetric Beta( a , a ) density sampled at mid‑points,
    i.e. a_i ∝ (x·(1−x))^{a−1},  x∈(0,1).
    """
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    w = (x * (1.0 - x)) ** (a - 1.0)
    a_vec = w / w.sum()
    return a_vec


# ----------------------------------------------------------------------
# Random initialisers
# ----------------------------------------------------------------------
def random_dirichlet_start(n: int, conc: float) -> np.ndarray:
    """Draw from a symmetric Dirichlet(α) distribution."""
    vec = np.random.gamma(conc, 1.0, size=n)
    return vec / vec.sum()


def random_sparse_start(n: int) -> np.ndarray:
    """Very sparse uniform probability vector (~5 % non‑zeros)."""
    sparsity = np.random.uniform(0.03, 0.07)
    k = max(2, int(round(sparsity * n)))
    idx = np.random.choice(n, size=k, replace=False)
    a = np.zeros(n, dtype=np.float64)
    a[idx] = 1.0 / k
    return project_to_simplex(a)


# ----------------------------------------------------------------------
# Smoothed optimisation (log‑sum‑exp) via Adam on the soft‑max layer
# ----------------------------------------------------------------------
def adam_smooth_optimize(
    start_a: np.ndarray,
    deadline: float,
    max_iters: int = 150_000,
    beta_start: float = 5.0,
    beta_end: float = 20000.0,
    lr: float = 0.018,
) -> tuple[np.ndarray, float]:
    """
    Minimise the smooth surrogate
          f_β(a) = (1/β)·log Σ_k exp(β·(a*a)_k)
    using Adam on the soft‑max parametrisation a = softmax(z).
    β is increased exponentially from beta_start to beta_end.
    """
    eps = 1e-12
    z = np.log(start_a + eps)               # initialise from log‑softmax
    m = np.zeros_like(z)
    v = np.zeros_like(z)

    beta1, beta2 = 0.9, 0.999
    eps_adam = 1e-12

    best_a = softmax(z)
    best_score = compute_score(best_a)

    for it in range(1, max_iters + 1):
        if time.time() >= deadline:
            break

        # schedule for β
        t = it / max_iters
        beta = beta_start * (beta_end / beta_start) ** t

        a = softmax(z)

        # smoothed max‑convolution
        conv = convolution_fft(a, a)                    # length 2n‑1
        maxc = conv.max()
        exp_w = np.exp(beta * (conv - maxc))            # stabilise
        w = exp_w / exp_w.sum()                         # soft‑max over shifts

        # gradient d f_β / d a = 2 * (a_rev * w)  (full conv, then slice)
        a_rev = a[::-1]
        conv_rev = convolution_fft(a_rev, w)            # length 3n‑2
        n = a.shape[0]
        grad_a = 2.0 * conv_rev[n - 1 : n - 1 + n]

        # back‑prop through soft‑max
        dot_ag = np.dot(a, grad_a)
        grad_z = a * grad_a - dot_ag * a

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad_z
        v = beta2 * v + (1.0 - beta2) * (grad_z ** 2)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        z -= lr * m_hat / (np.sqrt(v_hat) + eps_adam)

        # occasional exact evaluation
        if it % 400 == 0:
            a_tmp = softmax(z)
            sc = compute_score(a_tmp)
            if sc < best_score:
                best_score = sc
                best_a = a_tmp.copy()
                print(f"[{time.time():.1f}]   Adam iter {it:7d} – new best = {best_score:.6f}")

    # final check
    a_fin = softmax(z)
    sc_fin = compute_score(a_fin)
    if sc_fin < best_score:
        best_score = sc_fin
        best_a = a_fin.copy()
    return best_a, best_score


# ----------------------------------------------------------------------
# Mirror (exponentiated) sub‑gradient descent on the true max‑objective
# ----------------------------------------------------------------------
def mirror_subgrad_optimize(
    start_a: np.ndarray,
    deadline: float,
    step0: float = 0.9,
    check_interval: int = 2000,
) -> tuple[np.ndarray, float]:
    """
    Entropy‑mirror (multiplicative) sub‑gradient descent for
    2·n·max(a*a).  Step size η_t = step0 / sqrt(t).
    """
    n = start_a.shape[0]
    a = start_a.copy()
    best_a = a.copy()
    best_score = compute_score(a)

    t = 0
    while time.time() < deadline:
        t += 1
        conv = convolution_fft(a, a)
        k = int(np.argmax(conv))                  # active shift index

        # sub‑gradient for active shift
        idx = np.arange(n)
        j = k - idx
        mask = (j >= 0) & (j < n)
        g = np.zeros_like(a)
        g[mask] = 2.0 * a[j[mask]]

        eta = step0 / np.sqrt(t)
        a = a * np.exp(-eta * g)                  # multiplicative update
        s = a.sum()
        if s == 0.0:
            a = np.full_like(a, 1.0 / n)
        else:
            a = a / s                               # renormalise

        if t % check_interval == 0:
            sc = compute_score(a)
            if sc < best_score:
                best_score = sc
                best_a = a.copy()
                print(f"[{time.time():.1f}]   Mirror step {t:7d} – new best = {best_score:.6f}")

    sc = compute_score(a)
    if sc < best_score:
        best_score = sc
        best_a = a.copy()
    return best_a, best_score


# ----------------------------------------------------------------------
# Stochastic hill‑climb (tiny simulated‑annealing)
# ----------------------------------------------------------------------
def hill_climb_sa(
    start_a: np.ndarray,
    deadline: float,
    max_steps: int = 1_200_000,
    init_temp_factor: float = 1e-4,
) -> tuple[np.ndarray, float]:
    """
    Very low‑temperature simulated‑annealing walk on the simplex.
    - with prob 0.5 swap two entries
    - otherwise transfer a fraction of mass from one index to another
    """
    n = start_a.size
    cur_a = start_a.copy()
    cur_score = compute_score(cur_a)

    best_a = cur_a.copy()
    best_score = cur_score

    T0 = best_score * init_temp_factor
    T = T0
    step_scale = 0.12

    for step in range(1, max_steps + 1):
        if time.time() >= deadline:
            break

        cand = cur_a.copy()
        if np.random.rand() < 0.5:
            i, j = np.random.randint(n, size=2)
            cand[i], cand[j] = cand[j], cand[i]
        else:
            i, j = np.random.randint(n, size=2)
            if i != j:
                delta = step_scale * cur_a[i]
                cand[i] -= delta
                cand[j] += delta
                if cand[i] < 0.0:
                    cand[i] = 0.0

        cand = cand / cand.sum()
        cand_score = compute_score(cand)
        delta = cand_score - cur_score

        # Metropolis acceptance
        if delta < 0.0 or np.random.rand() < np.exp(-delta / (T + 1e-18)):
            cur_a = cand
            cur_score = cand_score
            if cur_score < best_score:
                best_score = cur_score
                best_a = cur_a.copy()
                step_scale = min(step_scale * 1.07, 0.5)
        else:
            step_scale *= 0.97

        # linear cooling
        T = T0 * (1.0 - step / max_steps)

    return best_a, best_score


# ----------------------------------------------------------------------
# High‑level search driver
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 12345, budget_s: int = 1000, **_):
    """
    Search for a non‑negative step‑function (list of floats) that minimises the
    ``evaluate_sequence`` objective.  The routine respects the time budget,
    explores several analytic families, refines the best candidate with
    mirror‑subgradient descent, optionally performs a high‑β Adam sweep and
    finishes with a tiny simulated‑annealing hill‑climb.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 2.0  # small safety margin

    # -------------------------------------------------------------
    # Choose a length.  Bigger n gives a lower theoretical lower bound.
    # -------------------------------------------------------------
    if budget_s >= 800:
        n = 32768
    elif budget_s >= 500:
        n = 16384
    else:
        n = 8192
    print(f"[{time.time():.1f}] Using length n = {n}")

    # -------------------------------------------------------------
    # Global best bookkeeping
    # -------------------------------------------------------------
    best_seq = None
    best_score = float("inf")

    def maybe_update(vec: np.ndarray, tag: str = ""):
        """Update global best if the vector improves the score."""
        nonlocal best_seq, best_score
        sc = compute_score(vec)
        if sc < best_score:
            best_score = sc
            best_seq = vec.tolist()
            print(f"[{time.time():.1f}] new best {tag:<20} = {best_score:.6f}")

    # -------------------------------------------------------------
    # 0. Fallback – use any previously known construction
    # -------------------------------------------------------------
    try:
        fb_vec = [float(x) for x in height_sequence_1]   # type: ignore
        fb_sc = evaluate_sequence(fb_vec)
        print(f"[{time.time():.1f}] fallback score = {fb_sc:.6f}")
        if fb_sc < best_score:
            best_score = fb_sc
            best_seq = fb_vec
    except Exception:
        pass

    # -------------------------------------------------------------
    # 1. Analytic families + random draws (cheap scan)
    # -------------------------------------------------------------
    uniform = np.full(n, 1.0 / n)
    maybe_update(uniform, "uniform")

    # triangular and cosine bumps
    maybe_update(triangular_family(n), "triangular")
    maybe_update(cosine_family(n), "cosine")

    # symmetric beta family (covers U‑shaped to peaked)
    for a_par in [0.3, 0.5, 0.8, 1.2, 1.5, 2.0, 3.0]:
        if time.time() >= deadline - 30.0:
            break
        maybe_update(symmetric_beta_family(n, a_par), f"beta_a={a_par:.2f}")

    # symmetric product family
    for alpha in np.linspace(0.0, 6.0, 21):
        if time.time() >= deadline - 30.0:
            break
        maybe_update(symmetric_product_family(n, alpha), f"sym‑α={alpha:.2f}")

    # left‑heavy power law
    for beta in np.linspace(0.0, 6.0, 21):
        if time.time() >= deadline - 30.0:
            break
        maybe_update(left_heavy_power_family(n, beta), f"pow‑β={beta:.2f}")

    # exponential decay
    for gamma in np.linspace(0.0, 8.0, 25):
        if time.time() >= deadline - 30.0:
            break
        maybe_update(exponential_decay_family(n, gamma), f"exp‑γ={gamma:.2f}")

    # Dirichlet draws (wide range of concentrations)
    conc_vals = np.logspace(-2, 1, 15)          # 0.01 … 10
    for conc in conc_vals:
        for _ in range(2):
            if time.time() >= deadline - 30.0:
                break
            maybe_update(random_dirichlet_start(n, conc), f"dir‑α={conc:.3f}")

    # -------------------------------------------------------------
    # 2. Mirror‑subgradient refinement (exact max objective)
    # -------------------------------------------------------------
    if best_seq is not None and time.time() < deadline - 60.0:
        start_vec = np.array(best_seq, dtype=np.float64)
        print(f"[{time.time():.1f}] starting mirror sub‑gradient descent")
        a_opt, sc_opt = mirror_subgrad_optimize(start_vec, deadline, step0=0.9)
        if sc_opt < best_score:
            best_score = sc_opt
            best_seq = a_opt.tolist()
            print(f"[{time.time():.1f}] after mirror sub‑grad = {best_score:.6f}")

    # -------------------------------------------------------------
    # 3. High‑β Adam fine‑tuning (smooth surrogate → true max)
    # -------------------------------------------------------------
    if best_seq is not None and time.time() < deadline - 30.0:
        remaining = deadline - time.time()
        if remaining > 5.0:   # ensure a tiny window
            print(f"[{time.time():.1f}] high‑β Adam refinement")
            a_ref, sc_ref = adam_smooth_optimize(
                np.array(best_seq, dtype=np.float64),
                deadline - 5.0,
                max_iters=120_000,
                beta_start=5.0,
                beta_end=50000.0,
                lr=0.015,
            )
            if sc_ref < best_score:
                best_score = sc_ref
                best_seq = a_ref.tolist()
                print(f"[{time.time():.1f}] after high‑β Adam = {best_score:.6f}")

    # -------------------------------------------------------------
    # 4. Symmetrisation (many optimal solutions are symmetric)
    # -------------------------------------------------------------
    if best_seq is not None:
        a_arr = np.array(best_seq, dtype=np.float64)
        a_sym = (a_arr + a_arr[::-1]) * 0.5
        a_sym = project_to_simplex(a_sym)
        maybe_update(a_sym, "symmetrised")

    # -------------------------------------------------------------
    # 5. Tiny simulated‑annealing hill‑climb (escape shallow minima)
    # -------------------------------------------------------------
    if best_seq is not None and time.time() < deadline - 5.0:
        print(f"[{time.time():.1f}] final hill‑climb (SA)")
        a_final, sc_final = hill_climb_sa(
            np.array(best_seq, dtype=np.float64),
            deadline,
            max_steps=800_000,
        )
        if sc_final < best_score:
            best_score = sc_final
            best_seq = a_final.tolist()
            print(f"[{time.time():.1f}] hill‑climb improved best = {best_score:.6f}")

    # -------------------------------------------------------------
    # Safeguard – fall back to uniform if everything failed
    # -------------------------------------------------------------
    if best_seq is None:
        n_small = 256
        best_seq = [1.0 / n_small] * n_small
        best_score = evaluate_sequence(best_seq)

    # -------------------------------------------------------------
    # Final official evaluation (the harness will also call it)
    # -------------------------------------------------------------
    final_score = evaluate_sequence(best_seq)
    print(f"\n[{time.time():.1f}] finished search – final reported score = {final_score:.6f}")

    # Return a plain Python list as required by the harness.
    return best_seq
