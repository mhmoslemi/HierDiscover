"""
step‑function optimizer (≈ 1 000 s budget)

The routine builds a non‑negative vector `a` (sum = 1) that minimises

    2·n·max_k (a∗a)[k]

which is exactly the value returned by ``evaluate_sequence`` (the
objective is homogeneous, so the sum can be fixed to 1).

Strategy
--------
1. **Choose a size** – larger `n` gives more degrees of freedom.
2. **One‑shot scans** of a few 1‑parameter families
   (exponential, power‑law, Beta, Gaussian, half‑sine).  The best
   candidate is kept as a warm start.
3. **Smooth‑max exponentiated‑gradient** – minimise a
   log‑sum‑exp surrogate of the maximum convolution using a KL‑mirror
   update.  The smoothing parameter β is annealed to a large value
   (≈ 2·10⁴) so the surrogate becomes very close to the true max.
4. **Exact max‑descent** – sub‑gradient descent on the true max
   (again with a KL‑mirror step) for further refinement.
5. **Fast targeted hill‑climb** – repeatedly move a small packet of
   mass from the index that contributes most to the current maximal
   convolution entry to a low‑mass index.  The convolution is updated
   in‑place via slice operations (O(n) per move), allowing
   hundreds of thousands of cheap steps.
6. Return the best sequence observed.

All operations keep the vector on the probability simplex, which makes
the evaluation invariant under scaling and matches the behaviour of
`evaluate_sequence`.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Convolution utilities (FFT based, aperiodic)
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full aperiodic convolution using a real FFT (zero‑padded)."""
    m = a.size + b.size - 1
    nfft = next_pow2(m)
    A = np.fft.rfft(a, n=nfft)
    B = np.fft.rfft(b, n=nfft)
    C = A * B
    return np.fft.irfft(C, n=nfft)[:m]


def compute_score(a: np.ndarray) -> float:
    """
    Exact objective used by ``evaluate_sequence`` for a vector that already
    satisfies the simplex constraint (sum = 1).
    Returns 2·n·max(a∗a).
    """
    n = a.shape[0]
    conv = convolution_fft(a, a)
    return 2.0 * n * float(conv.max())

# Simple families of candidate vectors
# ----------------------------------------------------------------------


def exponential_shape(n: int, lam: float) -> np.ndarray:
    """a_i ∝ exp(-lam·i)."""
    i = np.arange(n, dtype=float)
    if lam <= 0.0:
        w = np.ones_like(i)
    else:
        w = np.exp(-lam * i)
    return w / w.sum()


def power_shape(n: int, p: float) -> np.ndarray:
    """a_i ∝ (i+1)^(-p)."""
    i = np.arange(1, n + 1, dtype=float)
    if p <= 0.0:
        w = np.ones_like(i)
    else:
        w = i ** (-p)
    return w / w.sum()


def beta_shape(n: int, a: float, b: float) -> np.ndarray:
    """Discrete Beta distribution on {0,…,n‑1}."""
    x = (np.arange(n, dtype=float) + 0.5) / n
    w = np.power(x, a - 1.0) * np.power(1.0 - x, b - 1.0)
    w[np.isnan(w)] = 0.0
    w = np.maximum(w, 0.0)
    if w.sum() == 0.0:
        w = np.ones_like(w)
    return w / w.sum()


def gaussian_shape(n: int, sigma: float) -> np.ndarray:
    """Discrete Gaussian centred in the middle."""
    i = np.arange(n, dtype=float)
    mu = (n - 1) / 2.0
    if sigma <= 0.0:
        w = np.ones_like(i)
    else:
        w = np.exp(-0.5 * ((i - mu) / sigma) ** 2)
    return w / w.sum()


def sine_shape(n: int) -> np.ndarray:
    """Half‑sine window a_i ∝ sin(π·(i+0.5)/n)."""
    i = np.arange(n, dtype=float) + 0.5
    w = np.sin(np.pi * i / n)
    w = np.maximum(w, 0.0)
    return w / w.sum()

def exp_smooth_optimize(start_a: np.ndarray,
                        deadline: float,
                        max_iters: int = 600_000,
                        beta_start: float = 0.5,
                        beta_end: float = 20_000.0,
                        eta0: float = 0.6) -> tuple[np.ndarray, float]:
    """
    KL‑mirror descent on a log‑sum‑exp surrogate of the max‑convolution.
    β grows exponentially from ``beta_start`` to ``beta_end``.
    """
    a = np.maximum(start_a.astype(float), 1e-15)
    a /= a.sum()
    n = a.shape[0]

    best_a = a.copy()
    best_sc = compute_score(a)

    for it in range(1, max_iters + 1):
        if time.time() >= deadline:
            break

        t = it / max_iters
        beta = beta_start * (beta_end / beta_start) ** t

        conv = convolution_fft(a, a)
        maxc = conv.max()
        # stable soft‑max weights
        expw = np.exp(beta * (conv - maxc))
        w = expw / expw.sum()

        # gradient of smooth max (see derivation in the description)
        a_rev = a[::-1]
        grad = 2.0 * convolution_fft(a_rev, w)[n - 1:n - 1 + n]

        eta = eta0 / np.sqrt(it)

        a = np.maximum(a * np.exp(-eta * grad), 1e-15)
        a /= a.sum()

        if it % 400 == 0:
            sc = compute_score(a)
            if sc < best_sc:
                best_sc, best_a = sc, a.copy()

    # final exact check
    sc = compute_score(a)
    if sc < best_sc:
        best_sc, best_a = sc, a.copy()
    return best_a, best_sc


def max_descent_optimize(start_a: np.ndarray,
                         deadline: float,
                         max_iters: int = 500_000,
                         eta0: float = 0.5) -> tuple[np.ndarray, float]:
    """
    Sub‑gradient descent on the true max‑convolution using a KL‑mirror update.
    """
    a = np.maximum(start_a.astype(float), 1e-15)
    a /= a.sum()
    n = a.shape[0]

    best_a = a.copy()
    best_sc = compute_score(a)

    for it in range(1, max_iters + 1):
        if time.time() >= deadline:
            break

        conv = convolution_fft(a, a)
        k_max = int(conv.argmax())

        # build sub‑gradient of the active convolution entry
        g = np.zeros_like(a)
        i_start = max(0, k_max - (n - 1))
        i_end = min(n - 1, k_max)
        idx = np.arange(i_start, i_end + 1)
        g[idx] = 2.0 * a[k_max - idx]

        eta = eta0 / np.sqrt(it)

        a = np.maximum(a * np.exp(-eta * g), 1e-15)
        a /= a.sum()

        if it % 500 == 0:
            sc = compute_score(a)
            if sc < best_sc:
                best_sc, best_a = sc, a.copy()

    sc = compute_score(a)
    if sc < best_sc:
        best_sc, best_a = sc, a.copy()
    return best_a, best_sc


def fast_targeted_hill_climb(start_a: np.ndarray,
                             deadline: float,
                             max_steps: int = 400_000) -> tuple[np.ndarray, float]:
    """
    Greedy walk that moves mass from the index with the largest contribution
    to the current maximal convolution entry towards a low‑mass index.
    The convolution is updated in‑place using O(n) slice operations,
    allowing many cheap iterations.
    """
    a = start_a.copy()
    n = a.size
    conv = convolution_fft(a, a)                     # length 2·n‑1
    best_a = a.copy()
    best_sc = compute_score(best_a)

    # start with a relatively aggressive packet size
    step_scale = 0.25
    steps = 0

    # pre‑allocate a view for the sorted indices (updated lazily)
    sorted_idx = np.argsort(a)

    while steps < max_steps and time.time() < deadline:
        # current maximal vertex
        k_max = int(conv.argmax())
        i_start = max(0, k_max - (n - 1))
        i_end = min(n - 1, k_max)
        idx = np.arange(i_start, i_end + 1)
        contrib = a[idx] * a[k_max - idx]
        i_max = idx[np.argmax(contrib)]

        # choose a destination with small mass (deterministic, cheap)
        # refresh the sorted list only every thousand steps
        if steps % 1000 == 0:
            sorted_idx = np.argsort(a)
        # pick the smallest‑mass index that is not i_max
        for cand in sorted_idx[:10]:
            if cand != i_max:
                j = cand
                break
        else:
            # fallback to a random different index
            j = np.random.randint(n)
            while j == i_max:
                j = np.random.randint(n)

        delta = step_scale * a[i_max]
        if delta < 1e-12:
            step_scale *= 0.5
            continue

        # --- tentative move -------------------------------------------------
        a_new = a.copy()
        conv_new = conv.copy()

        # linear part of the convolution update (uses the *old* a)
        conv_new[i_max:i_max + n] -= 2.0 * delta * a
        conv_new[j:j + n] += 2.0 * delta * a

        # quadratic (δ²) correction
        conv_new[2 * j] += delta * delta
        conv_new[2 * i_max] += delta * delta
        conv_new[i_max + j] -= 2.0 * delta * delta

        # update the vector itself
        a_new[i_max] -= delta
        a_new[j] += delta

        # evaluate
        new_sc = 2.0 * n * float(conv_new.max())

        if new_sc < best_sc - 1e-13:
            # accept
            a = a_new
            conv = conv_new
            best_a = a_new.copy()
            best_sc = new_sc
            # slowly enlarge the packet size (capped)
            step_scale = min(step_scale * 1.03, 0.5)
        else:
            # reject – shrink the packet
            step_scale *= 0.96

        steps += 1

    return best_a, best_sc

                print(f"[{time.time():.1f}] gauss σ={sigma:.2f} → {sc:.6f}")

        # half‑sine
        a = sine_shape(n)
        sc = compute_score(a)
        if sc < best_score:
            best_score, best_seq = sc, a.tolist()
            print(f"[{time.time():.1f}] sine shape → {sc:.6f}")

    scan_families()

    # -------------------------------------------------
    # 2. Random Dirichlet starts (diversify)
    # -------------------------------------------------
    for _ in range(5):
        if time.time() >= deadline - 180:
            break
        a_rand = np.random.rand(n)
        a_rand /= a_rand.sum()
        sc = compute_score(a_rand)
        if sc < best_score:
            best_score, best_seq = sc, a_rand.tolist()
            print(f"[{time.time():.1f}] random start improved to {sc:.6f}")

    # -------------------------------------------------
    # 3. Smooth‑max exponentiated‑gradient phase
    # -------------------------------------------------
    if best_seq is not None and time.time() < deadline - 150:
        print(f"[{time.time():.1f}] starting smooth exponentiated‑gradient")
        a_opt, sc_opt = exp_smooth_optimize(
            np.array(best_seq, dtype=float),
            deadline - 140,
            max_iters=600_000,
            beta_end=20_000.0,
            eta0=0.6,
        )
        if sc_opt < best_score:
            best_score, best_seq = sc_opt, a_opt.tolist()
            print(f"[{time.time():.1f}] after smooth phase best = {best_score:.6f}")

    # -------------------------------------------------
    # 4. Exact max‑descent (sub‑gradient)
    # -------------------------------------------------
    if best_seq is not None and time.time() < deadline - 100:
        print(f"[{time.time():.1f}] starting exact max‑descent")
        a_opt2, sc_opt2 = max_descent_optimize(
            np.array(best_seq, dtype=float),
            deadline - 90,
            max_iters=500_000,
            eta0=0.5,
        )
        if sc_opt2 < best_score:
            best_score, best_seq = sc_opt2, a_opt2.tolist()
            print(f"[{time.time():.1f}] after max‑descent best = {best_score:.6f}")

    # -------------------------------------------------
    # 5. Fast targeted hill‑climb polishing (many cheap moves)
    # -------------------------------------------------
    if best_seq is not None and time.time() < deadline - 5:
        print(f"[{time.time():.1f}] final fast hill‑climb")
        a_final, sc_final = fast_targeted_hill_climb(
            np.array(best_seq, dtype=float),
            deadline,
            max_steps=400_000,
        )
        if sc_final < best_score:
            best_score, best_seq = sc_final, a_final.tolist()
            print(f"[{time.time():.1f}] hill‑climb improved best = {best_score:.6f}")

    # -------------------------------------------------
    # Safeguard: tiny uniform vector if everything failed
    # -------------------------------------------------
    if best_seq is None:
        n_small = 256
        best_seq = [1.0 / n_small] * n_small
        best_score = evaluate_sequence(best_seq)

    # -------------------------------------------------
    # Final official evaluation (the harness will also call it)
    # -------------------------------------------------
    final_score = evaluate_sequence(best_seq)
    print(f"\n[{time.time():.1f}] finished – final reported score = {final_score:.6f}")

    # The harness expects a plain Python list of floats.
    return best_seq
