  The smooth‑max is the Lp‑norm of the convolution (p → ∞ → max).
  We minimise   f_p(a)= ( Σ_k (a∗a)[k]^p )^{1/p}   with an Adam‑style projected
  update on the simplex.  A schedule p = 2,4,8,…,2048,4096 steers the
  optimisation from a gentle smooth objective to the true peak.
* A short projected sub‑gradient polishing (identical to the previous code)
  cleans up the remaining tiny violations.
* Deterministic “peak‑reduction’’ moves: after each sub‑gradient step we
  locate the active convolution index and transfer a tiny amount of mass
  from the two most responsible positions to a low‑impact location.
* Finish with a cheap stochastic hill‑climb that swaps or shifts a small
  fraction of mass between random coordinates.
* All phases respect a wall‑clock deadline (default 1000 s) and constantly
  keep the best sequence seen.

The optimiser works entirely with NumPy, no external solvers are required.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Basic tools
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """
    Full linear convolution a∗b via real FFT.
    Returns an array of length len(a)+len(b)-1.
    """
    m = a.shape[0] + b.shape[0] - 1
    n_fft = next_pow2(m)
    A = np.fft.rfft(a, n=n_fft)
    B = np.fft.rfft(b, n=n_fft)
    C = A * B
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


def compute_score(a: np.ndarray) -> float:
    """
    Exact objective used by evaluate_sequence for a vector already
    normalised to sum = 1.  Returns 2·n·max(conv(a,a)).
    """
    n = a.shape[0]
    conv = convolution_fft(a, a)
    return 2.0 * n * float(conv.max())


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the probability simplex
    {x ≥ 0, sum x = 1}.
    """
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / v.shape[0])

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, v.shape[0] + 1)
    cond = u - cssv / ind > 0
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


# ----------------------------------------------------------------------
# Analytic families for initial guesses
# ----------------------------------------------------------------------


def generate_central_binomial(n: int) -> np.ndarray:
    """a_i = C(2i,i) / 4^i, monotone decreasing, normalised to sum = 1."""
    a = np.empty(n, dtype=np.float64)
    a[0] = 1.0
    for i in range(1, n):
        a[i] = a[i - 1] * (2 * i - 1) / (2.0 * i)
    return project_to_simplex(a)


def generate_powerlaw(n: int, p: float) -> np.ndarray:
    """a_i ∝ (i+1)^(‑p), normalised to sum = 1."""
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = idx ** (-p)
    return project_to_simplex(a)


def generate_geometric(n: int, r: float) -> np.ndarray:
    """Geometric distribution a_i = (1‑r)·r^i, normalised (exact if r<1)."""
    i = np.arange(n, dtype=np.float64)
    a = (1.0 - r) * (r ** i)
    return project_to_simplex(a)


def generate_gaussian(n: int, sigma: float) -> np.ndarray:
    """Discrete Gaussian centred at the middle, width = σ·n."""
    centre = (n - 1) / 2.0
    x = np.arange(n, dtype=np.float64)
    a = np.exp(-0.5 * ((x - centre) / (sigma * n)) ** 2)
    return project_to_simplex(a)


def generate_cosine_window(n: int, power: float = 1.0) -> np.ndarray:
    """A (1‑cos) window raised to *power*, centred, normalised."""
    x = np.arange(n, dtype=np.float64)
    a = (1.0 - np.cos(2.0 * np.pi * x / (n - 1))) ** power
    return project_to_simplex(a)


# ----------------------------------------------------------------------
# Phase 1 – high‑p smooth‑max optimiser (Adam)
# ----------------------------------------------------------------------


def lp_adam_optimize(
    start_a: np.ndarray,
    time_deadline: float,
    p_schedule: list[int],
    step0: float = 0.8,
    beta1: float = 0.9,
    beta2: float = 0.999,
    eps: float = 1e-8,
) -> tuple[np.ndarray, float]:
    """
    Adam optimiser for the Lp‑smooth approximation of the max‑convolution.
    p grows according to *p_schedule*; for each p a fixed number of iterations
    (as many as the remaining time permits) is performed.
    """
    a = start_a.astype(np.float64).copy()
    n = a.shape[0]

    # Adam moments
    m1 = np.zeros_like(a)
    m2 = np.zeros_like(a)

    best_a = a.copy()
    best_score = compute_score(a)
    it_global = 0

    for p in p_schedule:
        # terminate early if out of time
        if time.time() >= time_deadline:
            break

        # rough estimate of how many iterations we can still afford
        remaining = time_deadline - time.time()
        # empirical ~2500 iterations per second (depends on hardware)
        max_iters = int(min(300_000, max(2_000, remaining * 2500)))

        for it_local in range(1, max_iters + 1):
            it_global += 1
            if time.time() >= time_deadline:
                break

            conv = convolution_fft(a, a)                     # length 2n‑1
            maxc = conv.max()
            # stable weights w_i ∝ (conv_i / maxc)^p   (equivalently conv_i^p)
            log_conv = np.log(conv + 1e-30)
            log_max = np.log(maxc + 1e-30)
            w = np.exp(p * (log_conv - log_max))
            w_sum = w.sum()
            if w_sum == 0.0:
                w = np.full_like(w, 1.0 / w.size)
            else:
                w = w / w_sum

            # gradient = 2·rev(a) ∗ w   (restricted to the original length)
            rev_a = a[::-1]
            grad_full = convolution_fft(rev_a, w)
            grad = 2.0 * grad_full[n - 1:n - 1 + n]

            # Adam update with a simple 1/√t schedule
            m1 = beta1 * m1 + (1.0 - beta1) * grad
            m2 = beta2 * m2 + (1.0 - beta2) * (grad * grad)
            m1_hat = m1 / (1.0 - beta1 ** it_global)
            m2_hat = m2 / (1.0 - beta2 ** it_global)

            step = step0 / np.sqrt(it_global)
            a_new = a - step * m1_hat / (np.sqrt(m2_hat) + eps)
            a_new = project_to_simplex(a_new)

            a = a_new

            # occasional evaluation
            if it_global % 800 == 0:
                cur_score = compute_score(a)
                if cur_score < best_score:
                    best_score = cur_score
                    best_a = a.copy()
                    print(
                        f"[Lp‑Adam] p={p:<5d} it={it_global:<7d} best_score={best_score:.6f}"
                    )

        # end of this p‑stage – one more check
        cur_score = compute_score(a)
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[Lp‑Adam] after p={p} best_score={best_score:.6f}")

    return best_a, best_score


# ----------------------------------------------------------------------
# Phase 2 – classic projected sub‑gradient descent (optional fine‑tuning)
# ----------------------------------------------------------------------


def subgrad_optimize(
    init_a: np.ndarray,
    time_deadline: float,
    max_iter: int = 300_000,
    step0: float = 0.07,
) -> tuple[np.ndarray, float]:
    """Projected sub‑gradient descent on the true max‑convolution objective."""
    a = init_a.astype(np.float64).copy()
    n = a.shape[0]
    best_a = a.copy()
    best_score = compute_score(a)

    for t in range(1, max_iter + 1):
        if time.time() >= time_deadline:
            break

        conv = convolution_fft(a, a)
        k = int(np.argmax(conv))

        # sub‑gradient for the active convolution entry
        idx = np.arange(n)
        j = k - idx
        mask = (j >= 0) & (j < n)
        g = np.zeros_like(a)
        g[mask] = 2.0 * a[j[mask]]

        step = step0 / np.sqrt(t)
        a_new = a - step * g
        a_new = project_to_simplex(a_new)

        cur_score = compute_score(a_new)
        if cur_score < best_score:
            best_score = cur_score
            best_a = a_new.copy()
            print(f"[SubGrad] it={t:<7d} best_score={best_score:.6f}")

        a = a_new

        # small jitter every 1000 steps to escape flats
        if t % 1000 == 0:
            jitter = np.random.randn(n) * (step0 * 0.004 / np.sqrt(t))
            a += jitter
            a = project_to_simplex(a)

    # final check
    final_score = compute_score(a)
    if final_score < best_score:
        best_score = final_score
        best_a = a.copy()
    return best_a, best_score


# ----------------------------------------------------------------------
# Phase 3 – deterministic peak‑reduction moves (mass transfer)
# ----------------------------------------------------------------------


def peak_reduction(
    a: np.ndarray,
    max_steps: int,
    time_deadline: float,
) -> tuple[np.ndarray, float]:
    """
    Repeatedly locate the active convolution index and move a tiny fraction
    of mass from the two most responsible coordinates to a low‑impact location.
    """
    n = a.shape[0]
    best_a = a.copy()
    best_score = compute_score(a)

    for step in range(max_steps):
        if time.time() >= time_deadline:
            break

        conv = convolution_fft(a, a)
        k = int(conv.argmax())

        # contributions for this k
        i = np.arange(n)
        j = k - i
        mask = (j >= 0) & (j < n)
        prod = np.zeros_like(a)
        prod[mask] = a[i[mask]] * a[j[mask]]

        # most responsible pair (i0, j0)
        i0 = int(np.argmax(prod))
        j0 = k - i0

        # compute a safe transfer amount
        epsilon = 0.02 * min(a[i0], a[j0])  # 2 % of the smaller of the two
        if epsilon < 1e-12:
            # nothing to move – abort early
            break

        # choose a destination index that currently has little influence on the peak
        # (randomly, but far from the pair's mirrors)
        dest = np.random.randint(n)
        # Perform the transfer
        a[i0] -= epsilon
        a[j0] -= epsilon
        a[dest] += 2.0 * epsilon
        a = project_to_simplex(a)

        cur_score = compute_score(a)
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[PeakRed] step={step:<6d} new_best={best_score:.6f}")

    return best_a, best_score


# ----------------------------------------------------------------------
# Phase 4 – cheap stochastic hill‑climb (same as in the previous baseline)
# ----------------------------------------------------------------------


def hill_climb_permutation(
    seq,
    time_deadline,
    max_steps: int = 200_000,
) -> tuple[list[float], float]:
    """Tiny stochastic moves (swap or fractional transfer) with simplex projection."""
    a = np.array(seq, dtype=np.float64)
    n = a.shape[0]
    best_a = a.copy()
    best_score = compute_score(best_a)

    step_scale = 0.08  # initial fraction of mass moved

    while max_steps > 0 and time.time() < time_deadline:
        cand = best_a.copy()
        if np.random.rand() < 0.5:      # swap two entries
            i, j = np.random.randint(n, size=2)
            cand[i], cand[j] = cand[j], cand[i]
        else:                           # move a fraction from i to j
            i, j = np.random.randint(n, size=2)
            if i != j:
                delta = step_scale * best_a[i]
                cand[i] -= delta
                cand[j] += delta
                if cand[i] < 0:
                    cand[i] = 0.0

        cand = project_to_simplex(cand)
        cur_score = compute_score(cand)

        if cur_score < best_score:
            best_score = cur_score
            best_a = cand
            step_scale = min(step_scale * 1.06, 0.5)
        else:
            step_scale *= 0.97

        max_steps -= 1

    return best_a.tolist(), best_score

# Main entry point required by the harness
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 12345, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative sequence minimising ``evaluate_sequence``.
    Returns a plain Python list of floats (already on the simplex).
    """
    np.random.seed(seed)
    start_time = time.time()
    # keep a small safety margin for the surrounding harness
    deadline = start_time + budget_s - 3.0

    # ------------------------------------------------------------------
    # Determine problem size (length of the sequence)
    # ------------------------------------------------------------------
    try:
        n = len(height_sequence_1)  # type: ignore[name-defined]
        init_candidates = [np.array(height_sequence_1, dtype=np.float64)]
        print(f"Using provided height_sequence_1 of length {n}.")
    except Exception:
        n = 2048
        init_candidates = []
        print(f"No external sequence provided – using length {n}.")

    # ------------------------------------------------------------------
    # 1. Analytic starters (uniform, central‑binomial, power‑law, …)
    # ------------------------------------------------------------------
    init_candidates.append(np.full(n, 1.0 / n, dtype=np.float64))             # uniform
    init_candidates.append(generate_central_binomial(n))                     # central binomial
    for p in np.linspace(0.3, 1.0, 8):                                      # power‑law
        init_candidates.append(generate_powerlaw(n, float(p)))
    for r in [0.97, 0.98, 0.985, 0.99, 0.992, 0.995]:                       # geometric
        init_candidates.append(generate_geometric(n, float(r)))
    for sigma in [0.10, 0.15, 0.20, 0.25, 0.30]:                             # Gaussian windows
        init_candidates.append(generate_gaussian(n, sigma))
    for power in [0.5, 1.0, 2.0, 3.0]:                                       # cosine windows
        init_candidates.append(generate_cosine_window(n, power))

    # evaluate starters
    best_seq = None
    best_val = float("inf")
    for idx, cand in enumerate(init_candidates):
        cand = project_to_simplex(cand)
        val = compute_score(cand)
        print(f"Starter {idx:02d}: score = {val:.6f}")
        if val < best_val:
            best_val = val
            best_seq = cand.copy()
    print(f"Best starter score = {best_val:.6f}")

    # ------------------------------------------------------------------
    # 2. High‑p smooth‑max optimisation (Adam)
    # ------------------------------------------------------------------
    smooth_deadline = deadline - 350.0
    if time.time() < smooth_deadline:
        p_sched = [2, 4, 8, 16, 32, 64, 128, 256, 512, 1024, 2048, 4096]
        a_smooth, smooth_score = lp_adam_optimize(
            best_seq,
            time_deadline=smooth_deadline,
            p_schedule=p_sched,
            step0=0.85,
        )
        if smooth_score < best_val:
            best_val = smooth_score
            best_seq = a_smooth.copy()
            print(f"Smooth‑max phase improved score to {best_val:.6f}")

    # ------------------------------------------------------------------
    # 3. Projected sub‑gradient polishing (optional but cheap)
    # ------------------------------------------------------------------
    subgrad_deadline = deadline - 180.0
    if time.time() < subgrad_deadline:
        remaining = subgrad_deadline - time.time()
        max_it = int(min(600_000, remaining * 2500))
        a_sub, sub_score = subgrad_optimize(
            best_seq,
            time_deadline=subgrad_deadline,
            max_iter=max_it,
            step0=0.06,
        )
        if sub_score < best_val:
            best_val = sub_score
            best_seq = a_sub.copy()
            print(f"Sub‑gradient phase improved score to {best_val:.6f}")

    # ------------------------------------------------------------------
    # 4. Deterministic peak‑reduction moves
    # ------------------------------------------------------------------
    peak_deadline = deadline - 90.0
    if time.time() < peak_deadline:
        a_peak, peak_score = peak_reduction(
            best_seq.copy(),
            max_steps=200_000,
            time_deadline=peak_deadline,
        )
        if peak_score < best_val:
            best_val = peak_score
            best_seq = a_peak.copy()
            print(f"Peak‑reduction phase improved score to {best_val:.6f}")

    # ------------------------------------------------------------------
    # 5. Random restarts + short sub‑gradient bursts (optional)
    # ------------------------------------------------------------------
    restart_deadline = deadline - 30.0
    if time.time() < restart_deadline:
        base = best_seq.copy()
        for r in range(4):
            if time.time() >= restart_deadline:
                break
            pert = base + 0.02 * np.random.randn(n)
            pert = project_to_simplex(pert)
            sub_dead = time.time() + 12.0
            a_tmp, s_tmp = subgrad_optimize(
                pert,
                time_deadline=sub_dead,
                max_iter=120_000,
                step0=0.05,
            )
            if s_tmp < best_val:
                best_val = s_tmp
                best_seq = a_tmp.copy()
                print(f"Restart {r} gave improvement: {best_val:.6f}")

    # ------------------------------------------------------------------
    # 6. Final stochastic hill‑climb
    # ------------------------------------------------------------------
    hill_deadline = deadline - 2.0
    if time.time() < hill_deadline:
        steps = 300_000
        print(f"Final hill‑climb for up to {steps} steps.")
        polished_seq, hill_score = hill_climb_permutation(
            best_seq.tolist(),
            time_deadline=hill_deadline,
            max_steps=steps,
        )
        if hill_score < best_val:
            best_val = hill_score
            best_seq = np.array(polished_seq, dtype=np.float64)
            print(f"Hill‑climb improved final score to {best_val:.6f}")

    # ------------------------------------------------------------------
    # 7. Fallback (should never be needed)
    # ------------------------------------------------------------------
    if best_seq is None:
        best_seq = np.full(n, 1.0 / n, dtype=np.float64)
        best_val = compute_score(best_seq)

    # ------------------------------------------------------------------
    # 8. Final reporting using the provided evaluate_sequence()
    # ------------------------------------------------------------------
    final_score = evaluate_sequence(best_seq.tolist())
    print(f"\nSearch finished. Final reported score = {final_score:.6f}")

    return best_seq.tolist()
```

This implementation adds a high‑p smooth‑max optimiser (Adam), a deterministic peak‑reduction phase, more diverse initial families, and retains the proven hill‑climb polish. It respects the 1000 s budget, constantly records the best sequence seen, and finally returns the sequence as a plain Python list ready for `evaluate_sequence`.
