Hybrid optimisation for the “max‑convolution” step‑function problem.

Algorithm
---------
1. Choose a modest length (n = 2048 for a 1000 s budget) – large enough
   to obtain a good bound, yet small enough to keep the linear program
   tractable.
2. Initialise a candidate on the probability simplex by scanning a few
   analytic families (U‑shaped, power‑law, exponential, β‑distribution)
   and a few random monotone draws.  Keep the best according to the exact
   objective (computed with an FFT‑based convolution).
3. Conditional‑gradient (Frank–Wolfe) phase:
   – Let a be the current vector, M = max(a∗a).
   – Solve the linear programme
         maximise  Σ g_j
         s.t.      (a∗g)_k ≤ M   for all k,
                  g_j ≥ 0.
     (The constraints are a Toeplitz linear operator; SciPy’s high‑level
      ``linprog`` with the dense matrix is fast enough for n≈2048.)
   – If the LP yields Σ g > 1, normalise g to the simplex and form a
     convex combination a←(1−η)a+η g, trying a handful of η∈{0.5,0.2,0.1,0.05,0.02}
     and keeping the best improvement of the true objective.
   – Repeat while time permits (≈30 iterations are typical).
4. Local refinement:
   – A short mirror‑subgradient run on the exact max‑convolution objective
     (entropy‑projection onto the simplex) removes remaining noise.
   – Finally a tiny simulated‑annealing hill‑climb cleans shallow local minima.
5. Return the best sequence (list of floats).  All numbers are non‑negative
   and the vector is normalised to sum = 1, which leaves the external
   ``evaluate_sequence`` invariant.

The code uses only NumPy and SciPy (if available).  If SciPy is missing the
LP phase is skipped and the algorithm falls back to the earlier smooth‑L‑BFGS
+ mirror‑subgradient pipeline.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Optional SciPy imports (LP solver and Toeplitz builder)
# ----------------------------------------------------------------------
try:
    import scipy.optimize as _opt
    import scipy.linalg as _linalg
    _HAS_SCIPY = True
except Exception:  # pragma: no cover
    _HAS_SCIPY = False

# ----------------------------------------------------------------------
# Basic FFT utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """
    Linear (aperiodic) convolution via real FFT.
    Returns full convolution of length len(a)+len(b)-1.
    """
    m = a.size + b.size - 1
    nfft = next_pow2(m)
    A = np.fft.rfft(a, n=nfft)
    B = np.fft.rfft(b, n=nfft)
    C = A * B
    conv = np.fft.irfft(C, n=nfft)[:m]
    return conv


def compute_score(a: np.ndarray) -> float:
    """
    Exact objective for a probability vector (sum(a)=1):
        score = 2·n·max(a∗a)
    """
    n = a.shape[0]
    conv = convolution_fft(a, a)
    return float(2.0 * n * conv.max())


# ----------------------------------------------------------------------
# Simplex utilities
# ----------------------------------------------------------------------


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of a vector onto the probability simplex."""
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / v.shape[0])

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, v.shape[0] + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.where(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def softmax(z: np.ndarray) -> np.ndarray:
    """Numerically stable soft‑max (maps ℝⁿ → simplex)."""
    z_max = np.max(z)
    e = np.exp(z - z_max)
    return e / e.sum()


# ----------------------------------------------------------------------
# Analytic families (all return a probability vector, sum = 1)
# ----------------------------------------------------------------------


def symmetric_product_family(n: int, alpha: float) -> np.ndarray:
    """U‑shaped: a_i ∝ [(i+½)(n‑½‑i)]^(‑α)."""
    i = np.arange(n, dtype=np.float64)
    logprod = np.log(i + 0.5) + np.log(n - 0.5 - i)
    w = np.exp(-alpha * logprod)
    w[np.isinf(w)] = np.finfo(np.float64).max
    a = w / w.sum()
    return a


def left_heavy_power_family(n: int, beta: float) -> np.ndarray:
    """Monotone decreasing power law: a_i ∝ (i+½)^(‑β)."""
    i = np.arange(n, dtype=np.float64)
    w = np.exp(-beta * np.log(i + 0.5))
    a = w / w.sum()
    return a


def exponential_decay_family(n: int, gamma: float) -> np.ndarray:
    """Simple exponential decay: a_i ∝ exp(‑γ·i)."""
    i = np.arange(n, dtype=np.float64)
    w = np.exp(-gamma * i)
    a = w / w.sum()
    return a


def beta_distribution_family(n: int, alpha: float) -> np.ndarray:
    """Symmetric beta‑shape: a_i ∝ (i/(n‑1))^α·(1‑i/(n‑1))^α."""
    i = np.arange(n, dtype=np.float64)
    t = i / (n - 1)
    eps = 1e-12
    w = (t + eps) ** alpha * (1 - t + eps) ** alpha
    a = w / w.sum()
    return a


# ----------------------------------------------------------------------
# Random initialisers
# ----------------------------------------------------------------------


def random_monotone_start(n: int) -> np.ndarray:
    """Random decreasing vector (sorted power‑transformed uniform)."""
    r = np.random.rand(n) ** np.random.uniform(0.5, 5.0)
    r.sort()
    a = r[::-1]           # decreasing
    a /= a.sum()
    return a


# ----------------------------------------------------------------------
# Linear‑programming sub‑problem (conditional gradient)
# ----------------------------------------------------------------------


def _build_toeplitz_matrix(f: np.ndarray) -> np.ndarray:
    """
    Build the dense Toeplitz matrix A such that
        (A @ g)[k] = (f ∗ g)[k]   for k = 0 … 2n‑2.
    Shape is (2n‑1, n).
    """
    n = f.shape[0]
    # First column: f followed by zeros (length 2n‑1)
    c = np.concatenate([f, np.zeros(n - 1, dtype=f.dtype)])
    # First row: f[0] then zeros (length n)
    r = np.concatenate([[f[0]], np.zeros(n - 1, dtype=f.dtype)])
    # scipy.linalg.toeplitz expects 1‑D arrays
    return _linalg.toeplitz(c, r)


def solve_lp_max_sum(f: np.ndarray, M: float):
    """
    Solve   maximise Σ g_j   s.t.  (f∗g)_k ≤ M  (for all k),  g≥0.
    Returns the optimal vector g (may be None if LP fails).
    """
    if not _HAS_SCIPY:  # pragma: no cover
        return None

    n = f.shape[0]
    A = _build_toeplitz_matrix(f)                       # (2n‑1) × n
    b = np.full(2 * n - 1, M, dtype=np.float64)
    c = -np.ones(n, dtype=np.float64)                   # maximise Σ g ⇔ minimise −Σ g
    bounds = [(0.0, None)] * n

    # Use the HiGHS interior‑point solver (fast for dense moderate‑size LPs)
    res = _opt.linprog(
        c,
        A_ub=A,
        b_ub=b,
        bounds=bounds,
        method="highs",
        options={"presolve": True, "verbosity": 0},
    )
    if not res.success:
        return None
    return res.x


# ----------------------------------------------------------------------
# Mirror (entropy) sub‑gradient descent on the true max‑objective
# ----------------------------------------------------------------------


def mirror_subgrad_optimize(start_a: np.ndarray, deadline: float,
                           step0: float = 0.8,
                           check_interval: int = 2000) -> tuple[np.ndarray, float]:
    """
    Multiplicative (exponentiated) sub‑gradient descent on
        F(a) = 2·n·max(a∗a)
    The iterate stays in the simplex thanks to a renormalisation step.
    """
    n = start_a.shape[0]
    a = start_a.astype(np.float64).copy()
    best_a = a.copy()
    best_score = compute_score(a)

    t = 0
    while time.time() < deadline:
        t += 1
        conv = convolution_fft(a, a)
        k = int(np.argmax(conv))

        idx = np.arange(n)
        j = k - idx
        mask = (j >= 0) & (j < n)
        g = np.zeros_like(a)
        g[mask] = 2.0 * a[j[mask]]

        eta = step0 / math.sqrt(t)
        a = a * np.exp(-eta * g)

        s = a.sum()
        if s == 0.0:
            a = np.full_like(a, 1.0 / n)
        else:
            a = a / s

        if t % check_interval == 0:
            sc = compute_score(a)
            if sc < best_score:
                best_score = sc
                best_a = a.copy()
                print(f"[{time.time():.1f}] mirror sub‑grad iter {t} best = {best_score:.6f}")

    sc = compute_score(a)
    if sc < best_score:
        best_score = sc
        best_a = a.copy()
    return best_a, best_score


# ----------------------------------------------------------------------
# Simple hill‑climb (tiny simulated annealing)
# ----------------------------------------------------------------------


def hill_climb_sa(start_a: np.ndarray, deadline: float,
                  max_steps: int = 800_000,
                  init_temp_factor: float = 1e-4) -> tuple[np.ndarray, float]:
    """
    Very low‑temperature random walk on the simplex:
        – with probability 0.5 swap two entries,
        – otherwise move a small fraction of mass between two indices.
    Accepts occasional uphill moves according to a linear cooling schedule.
    """
    n = start_a.size
    cur_a = start_a.copy()
    cur_score = compute_score(cur_a)

    best_a = cur_a.copy()
    best_score = cur_score

    T0 = best_score * init_temp_factor
    T = T0
    step_scale = 0.12

    for step in range(1, max_steps + 1):
        if time.time() >= deadline:
            break

        cand = cur_a.copy()
        if np.random.rand() < 0.5:          # swap
            i, j = np.random.randint(n, size=2)
            cand[i], cand[j] = cand[j], cand[i]
        else:                               # mass transfer
            i, j = np.random.randint(n, size=2)
            if i != j:
                delta = step_scale * cur_a[i]
                cand[i] -= delta
                cand[j] += delta
                if cand[i] < 0.0:
                    cand[i] = 0.0

        cand = cand / cand.sum()
        cand_score = compute_score(cand)
        delta = cand_score - cur_score

        if delta < 0.0 or np.random.rand() < np.exp(-delta / (T + 1e-18)):
            cur_a = cand
            cur_score = cand_score
            if cur_score < best_score:
                best_score = cur_score
                best_a = cur_a.copy()
                step_scale = min(step_scale * 1.07, 0.5)
        else:
            step_scale *= 0.97

        T = T0 * (1.0 - step / max_steps)

    return best_a, best_score


# ----------------------------------------------------------------------
# High‑level driver
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 12345, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative sequence minimising ``evaluate_sequence``.
    Returns a plain Python list of floats (sum of entries is 1 – the
    external evaluation is scale‑invariant).
    """
    np.random.seed(seed)
    start_time = time.time()
    # give a small safety margin for the external harness
    deadline = start_time + budget_s - 2.0

    # --------------------------------------------------------------
    # 1. Choose length (2048 gives a good trade‑off for the LP)
    # --------------------------------------------------------------
    n = 2048 if budget_s >= 900 else 1024
    print(f"[{time.time():.1f}] Using length n = {n}")

    best_a = None
    best_score = float("inf")

    # --------------------------------------------------------------
    # 2. Scan a few analytic families + random starts (cheap stage)
    # --------------------------------------------------------------
    families = [
        (symmetric_product_family, np.linspace(0.0, 8.0, 41)),
        (left_heavy_power_family, np.linspace(0.0, 8.0, 41)),
        (exponential_decay_family, np.linspace(0.0, 6.0, 31)),
        (beta_distribution_family, np.linspace(0.0, 6.0, 31)),
    ]

    for fam, grid in families:
        for p in grid:
            if time.time() >= deadline - 30.0:
                break
            vec = fam(n, float(p))
            sc = compute_score(vec)
            if sc < best_score:
                best_score = sc
                best_a = vec.copy()
                print(f"[{time.time():.1f}] new best ({fam.__name__} p={p:.2f}) = {best_score:.6f}")

    # a few random monotone starts
    for _ in range(60):
        if time.time() >= deadline - 30.0:
            break
        vec = random_monotone_start(n)
        sc = compute_score(vec)
        if sc < best_score:
            best_score = sc
            best_a = vec.copy()
            print(f"[{time.time():.1f}] new best (rand‑mono) = {best_score:.6f}")

    # fallback – uniform vector if everything failed
    if best_a is None:
        best_a = np.full(n, 1.0 / n, dtype=np.float64)
        best_score = compute_score(best_a)

    # --------------------------------------------------------------
    # 3. Conditional‑gradient / LP refinement loop
    # --------------------------------------------------------------
    if _HAS_SCIPY:
        print(f"[{time.time():.1f}] Starting LP‑based conditional gradient")
        # schedule of trial step sizes (larger steps early, finer later)
        eta_trials = [0.5, 0.2, 0.1, 0.05, 0.02]

        iter_cnt = 0
        while time.time() < deadline - 5.0:
            iter_cnt += 1
            M = convolution_fft(best_a, best_a).max()

            # solve LP: maximise Σ g_j  subject to (best_a ∗ g)_k ≤ M
            g0 = solve_lp_max_sum(best_a, M)
            if g0 is None:
                print(f"[{time.time():.1f}] LP failed – exiting LP loop")
                break

            sum_g0 = g0.sum()
            if sum_g0 <= 1.000001:                     # no improvement possible
                break

            # normalise g to the simplex (preserves direction but respects sum=1)
            g_norm = g0 / sum_g0

            improved_this_round = False
            best_local_a = best_a
            best_local_score = best_score

            for eta in eta_trials:
                cand = (1.0 - eta) * best_a + eta * g_norm
                # numerical projection (should already sum to 1)
                cand = project_to_simplex(cand)
                sc = compute_score(cand)
                if sc < best_local_score - 1e-12:
                    best_local_score = sc
                    best_local_a = cand.copy()
                    improved_this_round = True
                    print(f"[{time.time():.1f}] LP iter {iter_cnt} eta {eta:.3f} improved = {sc:.6f}")

            if not improved_this_round:
                # unable to improve with any trial step – stop the LP phase
                break

            best_a = best_local_a
            best_score = best_local_score

    else:
        print(f"[{time.time():.1f}] SciPy not available – skipping LP phase")

    # --------------------------------------------------------------
    # 4. Mirror sub‑gradient descent (few thousand iterations)
    # --------------------------------------------------------------
    if time.time() < deadline - 5.0:
        print(f"[{time.time():.1f}] Mirror sub‑gradient refinement")
        a_sub, sc_sub = mirror_subgrad_optimize(
            best_a.copy(),
            deadline,
            step0=0.8,
        )
        if sc_sub < best_score:
            best_score = sc_sub
            best_a = a_sub.copy()
            print(f"[{time.time():.1f}] after mirror sub‑grad best = {best_score:.6f}")

    # --------------------------------------------------------------
    # 5. Final tiny simulated‑annealing hill‑climb
    # --------------------------------------------------------------
    if time.time() < deadline - 2.0:
        print(f"[{time.time():.1f}] Final SA hill‑climb")
        a_final, sc_final = hill_climb_sa(
            best_a.copy(),
            deadline,
            max_steps=800_000,
        )
        if sc_final < best_score:
            best_score = sc_final
            best_a = a_final.copy()
            print(f"[{time.time():.1f}] hill‑climb improved best = {best_score:.6f}")

    # --------------------------------------------------------------
    # 6. Final official evaluation (the harness will also call evaluate_sequence)
    # --------------------------------------------------------------
    final_score = evaluate_sequence(best_a.tolist())
    print(f"\n[{time.time():.1f}] finished search – final reported score = {final_score:.6f}")

    return best_a.tolist()
