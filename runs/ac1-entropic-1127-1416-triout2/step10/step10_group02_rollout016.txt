def smooth_max_descent(a0, deadline):
    a = a0.copy()
    best_a = a.copy()
    best_score = compute_score(a)
    t = 0
    eps = 1.0
    eps_target = 200.0
    # choose step scale
    step_scale = 0.5
    while time.time() < deadline:
        # compute conv
        conv = full_convolution(a)
        # Softmax weights
        conv_max = conv.max()
        w = np.exp(eps * (conv - conv_max))
        w_sum = w.sum()
        w = w / w_sum
        # gradient via convolution
        w_rev = w[::-1]
        conv_rev = np.convolve(a, w_rev)[:3*len(a)-2] # although np.convolve returns exactly that length
        idx_offset = 2*len(a) - 2
        g = 2.0 * conv_rev[idx_offset - np.arange(len(a))]
        # step size
        t += 1
        step = step_scale / np.sqrt(t) / (np.linalg.norm(g) + 1e-12)
        a_new = a - step * g
        a_new = project_to_simplex(a_new)
        a = a_new
        # evaluate
        val = compute_score(a)
        if val < best_score:
            best_score, best_a = val, a.copy()
        # increase epsilon slowly
        if eps < eps_target:
            eps = min(eps * 1.002, eps_target)
    return best_a

"""
Step‑function optimizer using smooth max‑approximation (softmax) gradient descent,
random Dirichlet restarts and a final greedy hill‑climb.
The core idea is to minimise 2·n·max(conv(a,a)) / (∑a)^2 by repeatedly
smooth‑approximating the worst convolution entry, computing the exact gradient
( g_i = 2·∑_j a_j·w_{i+j} ), projecting onto the probability simplex and
annealing the softness parameter ε.  Multiple starts and a short stochastic
post‑processing are used within the 1000 s budget.
"""

import time
import numpy as np

def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / v.shape[0])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, v.shape[0] + 1)
    mask = u - cssv / ind > 0
    rho = np.where(mask)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w

def full_convolution(a: np.ndarray) -> np.ndarray:
    """Linear convolution a*a using a real FFT (length 2*len(a)-1)."""
    n = a.shape[0]
    m = 2 * n - 1
    n_fft = 1 << ((m - 1).bit_length())
    A = np.fft.rfft(a, n=n_fft)
    C = A * A
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv

def compute_score(a: np.ndarray) -> float:
    """Exact objective for a normalised vector (∑a = 1)."""
    n = a.shape[0]
    conv = full_convolution(a)
    return float(2.0 * n * conv.max())

def best_power_law_vector(n: int, deadline: float) -> np.ndarray:
    """Search over a_i ∝ (i+1)^{-p} in a wide interval."""
    best_a = None
    best_val = float('inf')
    best_p = 0.5
    # coarse sweep
    coarse_ps = np.linspace(0.4, 1.0, 61)
    for p in coarse_ps:
        if time.time() >= deadline:
            break
        a = np.arange(1, n + 1, dtype=float) ** (-p)
        a /= a.sum()
        val = compute_score(a)
        if val < best_val:
            best_val, best_a, best_p = val, a, p
    # fine sweep around best
    if best_a is not None:
        lo = max(0.3, best_p - 0.05)
        hi = min(1.2, best_p + 0.05)
        fine_ps = np.linspace(lo, hi, 41)
        for p in fine_ps:
            if time.time() >= deadline:
                break
            a = np.arange(1, n + 1, dtype=float) ** (-p)
            a /= a.sum()
            val = compute_score(a)
            if val < best_val:
                best_val, best_a, best_p = val, a, p
    return best_a

def smooth_max_descent(a0: np.ndarray, deadline: float,
                       init_step: float = 0.5,
                       eps_start: float = 0.5,
                       eps_end: float = 200.0,
                       eps_factor: float = 1.001) -> np.ndarray:
    """
    Annealed gradient descent on a smooth approximation of max(conv(a,a)).
    The weight vector w is a soft‑max of the convolution values, controlled by ε.
    """
    a = a0.copy()
    n = a.shape[0]
    best_a = a.copy()
    best_val = compute_score(a)
    t = 0
    step_scale = init_step
    eps = eps_start

    while time.time() < deadline:
        t += 1
        # convolution of the current iterate
        conv = full_convolution(a)                # length 2n-1
        max_conv = conv.max()
        # soft‑max weights (stable)
        exps = np.exp(eps * (conv - max_conv))
        w = exps / exps.sum()                     # length 2n-1

        # gradient g_i = 2 * Σ_j a_j * w_{i+j}
        w_rev = w[::-1]                           # reversed for easy convolution
        conv_rev = np.convolve(a, w_rev)          # length 3n-2
        idx = (2 * n - 2) - np.arange(n)          # indices n‑1 … 2n‑2
        g = 2.0 * conv_rev[idx]

        # normalise gradient direction (L2)
        norm = np.linalg.norm(g) + 1e-12
        direction = g / norm

        # step size decays like 1/√t
        step = step_scale / np.sqrt(t)
        a = a - step * direction
        a = project_to_simplex(a)

        # keep track of the best true objective
        val = compute_score(a)
        if val < best_val:
            best_val, best_a = val, a.copy()

        # slowly increase ε (making the approximation tighter)
        if eps < eps_end:
            eps = min(eps * eps_factor, eps_end)

    return best_a

def hill_climb(a_start: np.ndarray, deadline: float,
               max_iter: int = 1_000_000) -> np.ndarray:
    """
    Very cheap stochastic mass‑transfer moves.  Greedy acceptance,
    with a slight temperature‑like decay of the move magnitude.
    """
    a = a_start.copy()
    n = a.shape[0]
    best_a = a.copy()
    best_val = compute_score(a)

    step_scale = 0.05
    it = 0
    while time.time() < deadline and it < max_iter:
        i, j = np.random.randint(n, size=2)
        if i == j:
            continue
        # move a fraction of the mass from i to j
        delta = step_scale * a[i]
        cand = a.copy()
        cand[i] -= delta
        cand[j] += delta
        cand = np.maximum(cand, 0.0)
        cand = project_to_simplex(cand)

        val = compute_score(cand)
        if val < best_val:
            best_val, best_a, a = val, cand.copy(), cand
            step_scale = min(step_scale * 1.07, 0.5)   # enlarge when we improve
        else:
            # shrink step size slowly
            step_scale *= 0.985
        it += 1
    return best_a

def propose_candidate(seed: int = 12345, budget_s: int = 1000, **kwargs):
    """
    Entry‑point used by the harness.
    The routine combines a power‑law seed, a few random Dirichlet seeds,
    annealed soft‑max gradient descent and a final greedy hill‑climb.
    """
    np.random.seed(seed)
    start_time = time.time()
    # keep a small safety margin for the harness bookkeeping
    deadline = start_time + budget_s - 5.0

    n = 2048
    best_seq = None
    best_val = float('inf')

    # ------------------------------------------------------------------
    # Optional fallback that may already be good
    # ------------------------------------------------------------------
    try:
        fallback = [float(x) for x in height_sequence_1]   # type: ignore
        val = evaluate_sequence(fallback)
        print(f"[fallback] score = {val:.6f}")
        if val < best_val:
            best_val, best_seq = val, fallback
    except Exception:
        pass

    # ------------------------------------------------------------------
    # 0. Power‑law seed (cheap global scan)
    # ------------------------------------------------------------------
    pll_deadline = deadline - 500.0
    if time.time() < pll_deadline:
        a_pow = best_power_law_vector(n, pll_deadline)
        if a_pow is not None:
            val = compute_score(a_pow)
            print(f"[power sweep] initial score = {val:.6f}")
            if val < best_val:
                best_val, best_seq = val, a_pow.tolist()

    # ------------------------------------------------------------------
    # 1. Gradient descent on each available seed (power‑law + random)
    # ------------------------------------------------------------------
    descent_deadline = deadline - 200.0
    seeds = []

    if best_seq is not None:
        seeds.append(np.array(best_seq, dtype=float))
    else:
        # ensure we have at least one seed
        seeds.append(np.full(n, 1.0 / n))

    # add a couple of random Dirichlet seeds
    while time.time() < descent_deadline and len(seeds) < 5:
        # concentration parameter controls sparsity – try a few values
        alpha_val = np.random.choice([0.3, 0.8, 1.5])
        a_rand = np.random.dirichlet(alpha=np.full(n, alpha_val))
        seeds.append(a_rand)

    for idx, a0 in enumerate(seeds):
        if time.time() >= descent_deadline:
            break
        sub_deadline = descent_deadline
        a_refined = smooth_max_descent(a0, sub_deadline)
        val = compute_score(a_refined)
        print(f"[descent {idx}] refined score = {val:.6f}")
        if val < best_val:
            best_val, best_seq = val, a_refined.tolist()

    # ------------------------------------------------------------------
    # 2. Final greedy hill‑climb (fine‑grained local optimisation)
    # ------------------------------------------------------------------
    hill_deadline = deadline
    if best_seq is not None and time.time() < hill_deadline:
        a_start = np.array(best_seq, dtype=float)
        a_final = hill_climb(a_start, hill_deadline)
        val = compute_score(a_final)
        print(f"[hill climb] final score = {val:.6f}")
        if val < best_val:
            best_val, best_seq = val, a_final.tolist()

    # ------------------------------------------------------------------
    # 3. Safety fallback (uniform) – should never be reached for a real run
    # ------------------------------------------------------------------
    if best_seq is None:
        best_seq = [1.0 / n] * n
        best_val = evaluate_sequence(best_seq)

    # ------------------------------------------------------------------
    # Report using the official evaluator
    # ------------------------------------------------------------------
    final_score = evaluate_sequence(best_seq)
    print(f"\nSearch finished. Reported score = {final_score:.6f}")
    return best_seq

"""
Step‑function optimiser.

The objective is 2·n·max(conv(a,a)) / (∑a)².  The algorithm works in three
phases:

1.  A cheap global scan over power‑law shapes to obtain a reasonable seed.
2.  An annealed gradient descent that smooth‑approximates the max‑operator
    with a soft‑max (weights w ∝ exp(ε·conv)).  The exact gradient
    g_i = 2·∑_j a_j·w_{i+j} is computed by an FFT‑based convolution.
    ε is increased gradually so the optimisation moves from smoothing the
    whole convolution towards the true max (sub‑gradient) regime.
3.  A fast greedy hill‑climb that performs stochastic mass‑transfer moves
    with projection onto the probability simplex.

Multiple random Dirichlet seeds are tried in the second phase; a final
hill‑climb refines the best result.  All work fits into the 1000 s budget.
"""

import time
import numpy as np

# ------------------------------------------------------------
# Utility functions
# ------------------------------------------------------------

def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / v.shape[0])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, v.shape[0] + 1)
    mask = u - cssv / ind > 0
    rho = np.where(mask)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def full_convolution(a: np.ndarray) -> np.ndarray:
    """Linear convolution a*a using an FFT (length 2·len(a)−1)."""
    n = a.shape[0]
    m = 2 * n - 1
    n_fft = 1 << ((m - 1).bit_length())
    A = np.fft.rfft(a, n=n_fft)
    C = A * A
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


def compute_score(a: np.ndarray) -> float:
    """Exact objective for a normalised vector (∑a = 1)."""
    n = a.shape[0]
    conv = full_convolution(a)
    return float(2.0 * n * conv.max())

# ------------------------------------------------------------
# 1. Power‑law sweep (cheap global seed)
# ------------------------------------------------------------

def best_power_law_vector(n: int, deadline: float) -> np.ndarray:
    """Search over a_i ∝ (i+1)^{‑p} for p in a wide interval."""
    best_a = None
    best_val = float('inf')
    best_p = 0.5

    # coarse sweep
    coarse_ps = np.linspace(0.4, 1.0, 61)
    for p in coarse_ps:
        if time.time() >= deadline:
            break
        a = np.arange(1, n + 1, dtype=float) ** (-p)
        a /= a.sum()
        val = compute_score(a)
        if val < best_val:
            best_val, best_a, best_p = val, a, p

    # fine sweep around the best p
    if best_a is not None:
        lo = max(0.3, best_p - 0.05)
        hi = min(1.2, best_p + 0.05)
        fine_ps = np.linspace(lo, hi, 41)
        for p in fine_ps:
            if time.time() >= deadline:
                break
            a = np.arange(1, n + 1, dtype=float) ** (-p)
            a /= a.sum()
            val = compute_score(a)
            if val < best_val:
                best_val, best_a, best_p = val, a, p

    return best_a

# ------------------------------------------------------------
# 2. Annealed smooth‑max gradient descent
# ------------------------------------------------------------

def smooth_max_descent(a0: np.ndarray, deadline: float,
                       init_step: float = 0.5,
                       eps_start: float = 0.5,
                       eps_end: float = 200.0,
                       eps_factor: float = 1.001) -> np.ndarray:
    """
    Gradient descent on a smooth approximation of max(conv).
    The weights w are a soft‑max of the convolution values,
    controlled by ε which is increased (annealed) during the run.
    """
    a = a0.copy()
    n = a.shape[0]
    best_a = a.copy()
    best_val = compute_score(a)

    t = 0
    step_scale = init_step
    eps = eps_start

    # length needed for the convolution a * w_rev
    len_w = 2 * n - 1
    len_rev = n + len_w - 1                 # = 3·n‑2
    n_fft = 1 << ((len_rev - 1).bit_length())   # next power of two ≥ 3·n‑2

    while time.time() < deadline:
        t += 1

        # --- convolution of a with itself (FFT) ---
        A_fft = np.fft.rfft(a, n=n_fft)
        conv_full = np.fft.irfft(A_fft * A_fft, n=n_fft)[:len_w]
        max_conv = conv_full.max()

        # --- soft‑max weights ---
        exps = np.exp(eps * (conv_full - max_conv))  # stable because max is subtracted
        w = exps / exps.sum()                         # shape (2·n‑1,)

        # --- gradient g_i = 2·∑_j a_j·w_{i+j} ---
        w_rev = w[::-1]
        W_fft = np.fft.rfft(w_rev, n=n_fft)
        conv_rev = np.fft.irfft(A_fft * W_fft, n=n_fft)[:len_rev]
        idx = (2 * n - 2) - np.arange(n)               # pick indices n‑1 … 2n‑2
        g = 2.0 * conv_rev[idx]

        # --- normalised descent direction ---
        norm = np.linalg.norm(g) + 1e-12
        direction = g / norm

        # --- step size (1/√t decay) ---
        step = step_scale / np.sqrt(t)
        a = a - step * direction
        a = project_to_simplex(a)

        # --- keep the best true objective ---
        val = compute_score(a)
        if val < best_val:
            best_val, best_a = val, a.copy()

        # --- slowly increase ε to focus on the true maximum ---
        if eps < eps_end:
            eps = min(eps * eps_factor, eps_end)

    return best_a

# ------------------------------------------------------------
# 3. Final greedy hill‑climb (local refinement)
# ------------------------------------------------------------

def hill_climb(a_start: np.ndarray, deadline: float,
               max_iter: int = 1_000_000) -> np.ndarray:
    """
    Very cheap stochastic mass‑transfer moves with greedy acceptance.
    The move size adapts to the recent progress.
    """
    a = a_start.copy()
    n = a.shape[0]
    best_a = a.copy()
    best_val = compute_score(a)

    step_scale = 0.05
    it = 0
    while time.time() < deadline and it < max_iter:
        i, j = np.random.randint(n, size=2)
        if i == j:
            continue
        delta = step_scale * a[i]
        cand = a.copy()
        cand[i] -= delta
        cand[j] += delta
        cand = np.maximum(cand, 0.0)
        cand = project_to_simplex(cand)

        val = compute_score(cand)
        if val < best_val:
            best_val, best_a, a = val, cand.copy(), cand
            step_scale = min(step_scale * 1.07, 0.5)   # enlarge step after improvement
        else:
            step_scale *= 0.985                         # shrink otherwise
        it += 1
    return best_a
