def solve_lp_for_g(f, M):
    n = f.shape[0]
    # Build constraint matrix.
    # rows = all possible convolution indices.
    # using broadcasting:
    idx = np.arange(n)
    k = np.arange(2*n-1)[:, None] - idx[None, :]
    mask = (k >= 0) & (k < n)
    A = np.zeros((2*n-1, n), dtype=np.float64)
    A[mask] = f[k[mask]]
    # inequality A @ g <= M
    bounds = [(0, None)] * n
    # objective: maximize sum(g) -> minimize -sum(g)
    c = -np.ones(n)
    res = _opt.linprog(c, A_ub=A, b_ub=np.full(2*n-1, M), bounds=bounds, method='highs')
    if res.success:
        return res.x
    else:
        # fallback: return uniform or random
        return np.full(n, 1.0/n)

def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Linear (aperiodic) convolution via real FFT."""
    m = a.size + b.size - 1
    nfft = next_pow2(m)
    A = np.fft.rfft(a, n=nfft)
    B = np.fft.rfft(b, n=nfft)
    C = A * B
    conv = np.fft.irfft(C, n=nfft)[:m]
    return conv


def compute_score(a: np.ndarray) -> float:
    """
    Exact objective for a normalised probability vector (∑a = 1):
        score = 2·n·max(a∗a)
    This matches `evaluate_sequence` up to the built‑in normalisation.
    """
    n = a.shape[0]
    conv = convolution_fft(a, a)
    return 2.0 * n * float(conv.max())

def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection onto the probability simplex."""
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / v.shape[0])

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, v.shape[0] + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.where(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def softmax(z: np.ndarray) -> np.ndarray:
    """Numerically stable soft‑max (maps ℝⁿ → simplex)."""
    z_max = np.max(z)
    e = np.exp(z - z_max)
    return e / e.sum()

def symmetric_product_family(n: int, alpha: float) -> np.ndarray:
    """
    U‑shaped: a_i ∝ [(i+½)(n‑½‑i)]^(‑α).  Alpha ≥ 0.
    """
    i = np.arange(n, dtype=np.float64)
    logprod = np.log(i + 0.5) + np.log(n - 0.5 - i)
    logw = -alpha * logprod
    # stabilise by subtracting max
    logw -= np.max(logw)
    w = np.exp(logw)
    a = w / w.sum()
    return a

    sparsity = np.random.uniform(0.03, 0.07)
    k = max(2, int(round(sparsity * n)))
    idx = np.random.choice(n, size=k, replace=False)
    a = np.zeros(n, dtype=np.float64)
    a[idx] = 1.0 / k
    return project_to_simplex(a)


def random_monotone_start(n: int) -> np.ndarray:
    """Random decreasing vector (sorted power‑transformed uniform)."""
    r = np.random.rand(n) ** np.random.uniform(0.5, 5.0)
    r.sort()
    a = r[::-1]
    a /= a.sum()
    return a


# ----------------------------------------------------------------------
# Smooth surrogate optimisation (L‑BFGS‑B)
# ----------------------------------------------------------------------


def _smooth_obj_and_grad(z: np.ndarray, beta: float, n: int):
    """
    Returns (f, grad) where
        f   = (1/β)·log∑exp[β·(a*a)_k]   (smooth max)
        grad = ∂f/∂z   (z are the log‑soft‑max parameters)
    """
    a = softmax(z)                     # a in simplex
    conv = convolution_fft(a, a)       # length 2n‑1
    maxc = conv.max()
    exp_terms = np.exp(beta * (conv - maxc))
    sum_exp = exp_terms.sum()
    f = (np.log(sum_exp) + beta * maxc) / beta

    # gradient w.r.t. a
    w = exp_terms / sum_exp            # same shape as conv
    a_rev = a[::-1]
    conv_rev = convolution_fft(a_rev, w)   # length 3n‑2
    grad_a = 2.0 * conv_rev[n - 1 : 2 * n - 1]

    # chain rule through softmax
    dot_ag = np.dot(a, grad_a)
    grad_z = a * (grad_a - dot_ag)
    return float(f), grad_z.astype(np.float64)


def l_bfgs_refine(start_a: np.ndarray, deadline: float,
                  beta_list=(200.0, 800.0, 2000.0, 8000.0, 20000.0),
                  max_iter_per_beta=1200) -> tuple[np.ndarray, float]:
    """
    Refine `start_a` using L‑BFGS‑B on the smoothed surrogate.
    Returns the best vector and its exact score.
    """
    if not _HAS_SCIPY:          # pragma: no cover
        return start_a, compute_score(start_a)

    best_a = start_a.copy()
    best_score = compute_score(best_a)
    eps = 1e-12
    z = np.log(best_a + eps)        # initialise logits

    for beta in beta_list:
        if time.time() >= deadline - 5.0:
            break
        remaining = deadline - time.time()
        max_iter = min(max_iter_per_beta, int(remaining * 8))
        if max_iter <= 0:
            break

        def func(z_vec):
            f, g = _smooth_obj_and_grad(z_vec, beta, best_a.shape[0])
            return f, g

        try:
            opt_res = _opt.fmin_l_bfgs_b(
                func,
                z,
                maxiter=max_iter,
                disp=False,
                factr=1e7,
                pgtol=1e-12,
            )
        except Exception as exc:  # pragma: no cover
            print(f"[{time.time():.1f}] L‑BFGS failed at β={beta}: {exc}")
            continue

        z = opt_res[0]
        a_opt = softmax(z)
        sc_opt = compute_score(a_opt)
        if sc_opt < best_score:
            best_score = sc_opt
            best_a = a_opt.copy()
            print(f"[{time.time():.1f}] L‑BFGS β={beta:.0f} improved score = {best_score:.6f}")

    return best_a, best_score


# ----------------------------------------------------------------------
# Mirror (exponentiated) sub‑gradient descent on the true max objective
# ----------------------------------------------------------------------


def mirror_subgrad_optimize(start_a: np.ndarray, deadline: float,
                           step0: float = 0.8,
                           check_interval: int = 3000) -> tuple[np.ndarray, float]:
    """
    Multiplicative (exponentiated) sub‑gradient descent on
        F(a) = 2·n·max(a∗a)
    The iterate stays in the simplex thanks to a renormalisation step.
    """
    n = start_a.shape[0]
    a = start_a.astype(np.float64).copy()
    best_a = a.copy()
    best_score = compute_score(a)

    t = 0
    while time.time() < deadline:
        t += 1
        conv = convolution_fft(a, a)
        k = int(np.argmax(conv))

        # sub‑gradient of the active convolution entry
        idx = np.arange(n)
        j = k - idx
        mask = (j >= 0) & (j < n)
        g = np.zeros_like(a)
        g[mask] = 2.0 * a[j[mask]]

        eta = step0 / math.sqrt(t)
        a = a * np.exp(-eta * g)

        # entropy‑style projection onto the simplex
        s = a.sum()
        if s == 0.0:
            a = np.full_like(a, 1.0 / n)
        else:
            a = a / s

        if t % check_interval == 0:
            sc = compute_score(a)
            if sc < best_score:
                best_score = sc
                best_a = a.copy()
                print(f"[{time.time():.1f}] mirror sub‑grad iter {t} best = {best_score:.6f}")

    # final check
    sc = compute_score(a)
    if sc < best_score:
        best_score = sc
        best_a = a.copy()
    return best_a, best_score


# ----------------------------------------------------------------------
# Linear‑programming based improvement step
# ----------------------------------------------------------------------


def solve_lp_for_g(f: np.ndarray, M: float):
    """
    Solve  maximise Σ g_i   subject to  f∗g ≤ M (componentwise), g ≥ 0.
    Returns the raw (unnormalised) optimal g; if the LP fails a uniform
    vector is returned as a safe fallback.
    """
    if not _HAS_SCIPY:
        return np.full_like(f, 1.0 / f.shape[0])

    n = f.shape[0]
    # Build Toeplitz matrix representing convolution f∗g.
    col = np.concatenate([f, np.zeros(n - 1, dtype=f.dtype)])
    row = np.concatenate([[f[0]], np.zeros(n - 1, dtype=f.dtype)])
    A = _linalg.toeplitz(col, row)          # shape (2n‑1, n)

    # Linear programme: minimise −Σ g_i  (i.e. maximise Σ g_i)
    c = -np.ones(n, dtype=np.float64)
    bounds = [(0.0, None)] * n
    b_ub = np.full(2 * n - 1, M, dtype=np.float64)

    # Use HiGHS (default in recent SciPy)
    res = _opt.linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds,
                       method='highs', options={'presolve': True})

    if res.success:
        return np.maximum(res.x, 0.0)
    else:
        # fallback – uniform distribution (guaranteed feasible)
        return np.full_like(f, 1.0 / n)


def lp_improvement(f_start: np.ndarray, deadline: float,
                   max_iters: int = 15, tau0: float = 0.20) -> tuple[np.ndarray, float]:
    """
    Perform a few LP‑based refinement steps.
    Each step mixes the current f with the LP‑solution g (normalised)
    using a decreasing mixing factor τ.
    """
    f = f_start.copy()
    best_f = f.copy()
    best_score = compute_score(f)

    tau = tau0
    for it in range(1, max_iters + 1):
        if time.time() >= deadline - 5.0:
            break

        conv = convolution_fft(f, f)
        M = conv.max()
        g_raw = solve_lp_for_g(f, M)
        sum_g = g_raw.sum()
        if sum_g == 0.0:
            break
        g = g_raw / sum_g                       # normalise to a probability vector

        # try a few candidate mixing coefficients to ensure descent
        candidates = [(1 - tau) * f + tau * g]
        # also try a smaller step if the above increases the score
        if tau > 1e-3:
            candidates.append((1 - tau * 0.5) * f + (tau * 0.5) * g)

        improved = False
        for cand in candidates:
            cand /= cand.sum()
            sc = compute_score(cand)
            if sc < best_score:
                best_score = sc
                best_f = cand.copy()
                f = cand.copy()
                improved = True
                print(f"[{time.time():.1f}] LP iter {it} improved score = {best_score:.6f}")
                break

        if not improved:
            # even if not improving, still move a little to escape plateaus
            f = (1 - tau) * f + tau * g
            f /= f.sum()

        # reduce mixing factor slowly
        tau *= 0.85

    return best_f, best_score


# ----------------------------------------------------------------------
# Simple hill‑climb (tiny simulated annealing)
# ----------------------------------------------------------------------


def hill_climb_sa(start_a: np.ndarray, deadline: float,
                  max_steps: int = 1_500_000,
                  init_temp_factor: float = 1e-4) -> tuple[np.ndarray, float]:
    """
    Random walk on the simplex with a very low temperature.
    """
    n = start_a.size
    cur_a = start_a.copy()
    cur_score = compute_score(cur_a)

    best_a = cur_a.copy()
    best_score = cur_score

    T0 = best_score * init_temp_factor
    T = T0
    step_scale = 0.12

    for step in range(1, max_steps + 1):
        if time.time() >= deadline:
            break

        cand = cur_a.copy()
        if np.random.rand() < 0.5:                     # swap
            i, j = np.random.randint(n, size=2)
            cand[i], cand[j] = cand[j], cand[i]
        else:                                             # mass transfer
            i, j = np.random.randint(n, size=2)
            if i != j:
                delta = step_scale * cur_a[i]
                cand[i] -= delta
                cand[j] += delta
                if cand[i] < 0.0:
                    cand[i] = 0.0

        cand /= cand.sum()
        cand_score = compute_score(cand)
        delta = cand_score - cur_score

        if delta < 0.0 or np.random.rand() < np.exp(-delta / (T + 1e-18)):
            cur_a = cand
            cur_score = cand_score
            if cur_score < best_score:
                best_score = cur_score
                best_a = cur_a.copy()
                step_scale = min(step_scale * 1.07, 0.5)
        else:
            step_scale *= 0.97

        T = T0 * (1.0 - step / max_steps)

    return best_a, best_score

def propose_candidate(seed: int = 12345, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step‑function that minimises
    `evaluate_sequence`.  Returns a plain Python list of floats.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 2.0  # small safety margin

    # ------------------------------------------------------------------
    # Choose a convenient length (trade‑off quality vs speed)
    # ------------------------------------------------------------------
    n = 16384 if budget_s >= 900 else 8192
    print(f"[{time.time():.1f}] Using length n = {n}")

    best_seq = None
    best_score = float("inf")

    # ------------------------------------------------------------------
    # 0.  Optional fallback (if a known good construction is available)
    # ------------------------------------------------------------------
    try:
        fb_vec = [float(x) for x in height_sequence_1]   # type: ignore
        fb = evaluate_sequence(fb_vec)
        print(f"[{time.time():.1f}] fallback score = {fb:.6f}")
        if fb < best_score:
            best_score = fb
            best_seq = fb_vec
    except Exception:
        pass

    # ------------------------------------------------------------------
    # 1. Scan analytic families (expanded ranges & new families)
    # ------------------------------------------------------------------
    # --- symmetric product (U‑shaped) ---------------------------------
    for alpha in np.linspace(0.0, 30.0, 61):
        if time.time() >= deadline - 30.0:
            break
        vec = symmetric_product_family(n, alpha)
        sc = compute_score(vec)
        if sc < best_score:
            best_score = sc
            best_seq = vec.tolist()
            print(f"[{time.time():.1f}] new best (sym‑α={alpha:.2f}) = {best_score:.6f}")

    # --- left‑heavy power law ----------------------------------------
    for beta in np.linspace(0.0, 10.0, 101):
        if time.time() >= deadline - 30.0:
            break
        vec = left_heavy_power_family(n, beta)
        sc = compute_score(vec)
        if sc < best_score:
            best_score = sc
            best_seq = vec.tolist()
            print(f"[{time.time():.1f}] new best (left‑β={beta:.2f}) = {best_score:.6f}")

    # --- exponential decay --------------------------------------------
    for gamma in np.linspace(0.0, 10.0, 81):
        if time.time() >= deadline - 30.0:
            break
        vec = exponential_decay_family(n, gamma)
        sc = compute_score(vec)
        if sc < best_score:
            best_score = sc
            best_seq = vec.tolist()
            print(f"[{time.time():.1f}] new best (exp‑γ={gamma:.2f}) = {best_score:.6f}")

    # --- symmetric beta (α ≥ 0) ---------------------------------------
    for alpha in np.linspace(0.0, 8.0, 81):
        if time.time() >= deadline - 30.0:
            break
        vec = beta_distribution_family(n, alpha)
        sc = compute_score(vec)
        if sc < best_score:
            best_score = sc
            best_seq = vec.tolist()
            print(f"[{time.time():.1f}] new best (beta‑α={alpha:.2f}) = {best_score:.6f}")

    # --- new negative‑beta beta family (U‑shaped) ----------------------
    for alpha in np.linspace(-0.9, -0.05, 86):
        if time.time() >= deadline - 30.0:
            break
        vec = beta_negative_family(n, alpha)
        sc = compute_score(vec)
        if sc < best_score:
            best_score = sc
            best_seq = vec.tolist()
            print(f"[{time.time():.1f}] new best (beta‑neg‑α={alpha:.3f}) = {best_score:.6f}")

    # --- inverse‑sqrt family ------------------------------------------
    if time.time() < deadline - 15.0:
        vec = inverse_sqrt_family(n)
        sc = compute_score(vec)
        if sc < best_score:
            best_score = sc
            best_seq = vec.tolist()
            print(f"[{time.time():.1f}] new best (inv‑sqrt) = {best_score:.6f}")

    # --- random starts ------------------------------------------------
    for _ in range(250):
        if time.time() >= deadline - 30.0:
            break
        vec = random_monotone_start(n) if np.random.rand() < 0.5 else random_sparse_start(n)
        sc = compute_score(vec)
        if sc < best_score:
            best_score = sc
            best_seq = vec.tolist()
            print(f"[{time.time():.1f}] new best (rand) = {best_score:.6f}")

    # ------------------------------------------------------------------
    # 2. Refine the current best with L‑BFGS on the smoothed surrogate
    # ------------------------------------------------------------------
    if best_seq is not None and time.time() < deadline - 300.0:
        a_start = np.array(best_seq, dtype=np.float64)
        print(f"[{time.time():.1f}] Starting L‑BFGS refinement")
        a_refined, sc_refined = l_bfgs_refine(
            a_start,
            deadline - 200.0,
            beta_list=(200.0, 800.0, 2000.0, 8000.0, 20000.0),
            max_iter_per_beta=1000,
        )
        if sc_refined < best_score:
            best_score = sc_refined
            best_seq = a_refined.tolist()
            print(f"[{time.time():.1f}] after L‑BFGS best = {best_score:.6f}")

    # ------------------------------------------------------------------
    # 3. Mirror sub‑gradient descent on the exact max objective
    # ------------------------------------------------------------------
    if best_seq is not None and time.time() < deadline - 100.0:
        print(f"[{time.time():.1f}] Starting mirror sub‑gradient descent")
        a_sub, sc_sub = mirror_subgrad_optimize(
            np.array(best_seq, dtype=np.float64),
            deadline - 50.0,
            step0=0.8,
            check_interval=2000,
        )
        if sc_sub < best_score:
            best_score = sc_sub
            best_seq = a_sub.tolist()
            print(f"[{time.time():.1f}] after mirror sub‑grad best = {best_score:.6f}")

    # ------------------------------------------------------------------
    # 4. LP‑based improvement steps (the "alternating" scheme)
    # ------------------------------------------------------------------
    if best_seq is not None and time.time() < deadline - 120.0:
        print(f"[{time.time():.1f}] Starting LP‑based refinement")
        a_lp, sc_lp = lp_improvement(
            np.array(best_seq, dtype=np.float64),
            deadline - 20.0,
            max_iters=12,
            tau0=0.25,
        )
        if sc_lp < best_score:
            best_score = sc_lp
            best_seq = a_lp.tolist()
            print(f"[{time.time():.1f}] after LP refinement best = {best_score:.6f}")

    # ------------------------------------------------------------------
    # 5. Optional symmetrisation – many optimal solutions are symmetric
    # ------------------------------------------------------------------
    if best_seq is not None:
        a_arr = np.array(best_seq, dtype=np.float64)
        a_sym = (a_arr + a_arr[::-1]) * 0.5
        a_sym = project_to_simplex(a_sym)
        sc_sym = compute_score(a_sym)
        if sc_sym < best_score:
            best_score = sc_sym
            best_seq = a_sym.tolist()
            print(f"[{time.time():.1f}] symmetrisation improved best = {best_score:.6f}")

    # ------------------------------------------------------------------
    # 6. Final cheap hill‑climb (simulated annealing)
    # ------------------------------------------------------------------
    if best_seq is not None and time.time() < deadline - 5.0:
        print(f"[{time.time():.1f}] Final hill‑climb (SA)")
        a_final, sc_final = hill_climb_sa(
            np.array(best_seq, dtype=np.float64),
            deadline,
            max_steps=1_200_000,
        )
        if sc_final < best_score:
            best_score = sc_final
            best_seq = a_final.tolist()
            print(f"[{time.time():.1f}] hill‑climb improved best = {best_score:.6f}")

    # ------------------------------------------------------------------
    # Safeguard – if nothing succeeded fall back to a modest uniform vector
    # ------------------------------------------------------------------
    if best_seq is None:
        n_small = 256
        best_seq = [1.0 / n_small] * n_small
        best_score = evaluate_sequence(best_seq)

    # ------------------------------------------------------------------
    # Final official evaluation (the harness will also call evaluate_sequence)
    # ------------------------------------------------------------------
    final_score = evaluate_sequence(best_seq)
    print(f"\n[{time.time():.1f}] finished search – final reported score = {final_score:.6f}")

    return best_seq
