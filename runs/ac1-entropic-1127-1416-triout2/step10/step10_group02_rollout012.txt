def subgradient_polish(a0, time_deadline, step0=0.1):
    a = a0.copy()
    best = a.copy()
    best_M = full_convolution(a).max()
    t=0
    while time.time() < time_deadline:
        t+=1
        conv = full_convolution(a)
        M = conv.max()
        # active indices
        active = np.where(conv >= M - 1e-12)[0]
        # gradient = sum over active of 2* a shifted
        g = np.zeros_like(a)
        for k in active:
            idx = np.arange(len(a))
            j = k - idx
            mask = (j >= 0) & (j < len(a))
            g[mask] += 2.0 * a[j[mask]]
        step = step0 / np.sqrt(t)
        a = a - step * g
        # project onto simplex
        a = project_to_simplex(a)
        # evaluate
        M_new = full_convolution(a).max()
        if M_new < best_M:
            best_M = M_new
            best = a.copy()
    return best
```

def solve_lp_for_g(f, M, max_iters=2000):
    n = f.shape[0]
    # Build matrix A
    rows = 2*n-1
    A = np.empty((rows, n), dtype=np.float64)

"""
Step‑function optimiser for `evaluate_sequence`.

Algorithm
---------
1. **Global parametric sweep** – generate candidate vectors from two
   simple families (power‑law and geometric decay) and keep the best
   (lowest ∥a*a∥∞) one.

2. **Conditional‑gradient (Frank‑Wolfe) refinement** – treat the
   current vector `f` as a feasible point for the problem
        minimise  max_k (f*f)[k]   subject to  f ≥ 0, Σf = 1 .
   At each iteration we solve a *linear* program

        maximise Σ g_i   s.t.   (f * g) ≤ M ,  g ≥ 0

   where `M = max(f*f)`.  The solution `g` points towards a direction
   that can reduce the maximum convolution while keeping the bound on
   the mixed convolution `f*g`.  A short line‑search picks a mixing
   factor `t` and we replace `f ← (1‑t)f + t ĝ` (with `ĝ` rescaled to
   Σĝ = 1).  Repeating this step monotonically lowers `M` and
   typically yields a significant improvement over the naive sweep.

3. **Projected sub‑gradient polishing** – with the shape from (2) we
   run a classic sub‑gradient descent on the non‑smooth objective
   `M = max(conv(a,a))`.  The gradient is the sum of the active
   convolution rows; projection onto the probability simplex keeps the
   iterate feasible.

4. **Stochastic hill‑climb** – a cheap greedy mass‑transfer neighbourhood
   explores the final local basin.  Small random jitter is added
   periodically to escape tiny plateaus.

All phases respect the remaining wall‑clock budget (default 1000 s).
The returned Python list contains non‑negative floats whose sum is 1,
so `evaluate_sequence` sees the same objective as `compute_score`.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Optional CVXPY – used only for the Frank‑Wolfe LP step
# ----------------------------------------------------------------------

    _HAVE_CVX = True
except Exception:   # pragma: no cover
    _HAVE_CVX = False

# Basic linear‑algebra utilities
# ----------------------------------------------------------------------
def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    if np.all(v <= 0):
        # uniform distribution is the projection of the zero vector
        return np.full_like(v, 1.0 / v.shape[0])

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, v.shape[0] + 1)
    mask = u - cssv / ind > 0
    rho = np.where(mask)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def full_convolution(a: np.ndarray) -> np.ndarray:
    """
    Full linear convolution a*a using a real FFT.
    Length of the result is 2*len(a)-1.
    """
    n = a.shape[0]
    m = 2 * n - 1
    n_fft = 1 << ((m - 1).bit_length())          # next power of two ≥ m
    A = np.fft.rfft(a, n=n_fft)
    C = A * A
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


def compute_score(a: np.ndarray) -> float:
    """
    Exact objective for a normalised vector (Σa = 1):
        2·n·max(conv(a,a))
    The function is scale‑invariant, therefore we always keep Σa = 1.
    """
    n = a.shape[0]
    conv = full_convolution(a)
    return float(2.0 * n * conv.max())

# Phase 1 – global parametric sweep
# ----------------------------------------------------------------------
def best_initial_vector(n: int, time_deadline: float) -> np.ndarray:
    """Search simple families (power‑law, geometric) and return the best."""
    best_a = None
    best_val = float('inf')

    # ----- power‑law sweep -----
    for p in np.linspace(0.30, 1.00, 71):   # step ≈0.01
        if time.time() >= time_deadline:
            break
        a = (np.arange(1, n + 1, dtype=np.float64) ** (-p))
        a /= a.sum()
        val = compute_score(a)
        if val < best_val:
            best_val, best_a = val, a

    # ----- geometric (exponential) decay -----
    for r in np.linspace(0.50, 0.99, 81):   # step ≈0.006
        if time.time() >= time_deadline:
            break
        a = (r ** np.arange(n, dtype=np.float64))
        s = a.sum()
        if s == 0:
            continue
        a /= s
        val = compute_score(a)
        if val < best_val:
            best_val, best_a = val, a

    # ----- mixed (power‑law + uniform plateau) -----
    # small extra exploration: a flat head of length L followed by power‑law tail
    for L_frac in np.linspace(0.05, 0.30, 6):
        if time.time() >= time_deadline:
            break
        L = int(L_frac * n)
        if L == 0 or L >= n:
            continue
        for p in np.linspace(0.4, 0.8, 9):
            tail = (np.arange(1, n - L + 1, dtype=np.float64) ** (-p))
            head = np.full(L, tail.max())
            a = np.concatenate([head, tail])
            a /= a.sum()
            val = compute_score(a)
            if val < best_val:
                best_val, best_a = val, a

    return best_a


# ----------------------------------------------------------------------
# Phase 2 – Frank‑Wolfe (LP) refinement
# ----------------------------------------------------------------------
def _build_conv_matrix(f: np.ndarray) -> np.ndarray:
    """
    Build the matrix A such that A @ g = conv(f, g) (full linear convolution).
    Shape of A is (2*n-1, n).
    """
    n = f.shape[0]
    rows = 2 * n - 1
    A = np.empty((rows, n), dtype=np.float64)
    # each column i contributes f[i] to rows i … i+n-1
    for i in range(n):
        A[i:i + n, i] = f[i]
    return A


def _solve_lp_for_g(f: np.ndarray, M: float) -> np.ndarray | None:
    """
    Solve the linear program:
        maximise Σ g_i
        s.t.   conv(f, g) ≤ M   (elementwise)
               g ≥ 0
    Returns the optimal g as a NumPy array, or None on failure.
    """
    if not _HAVE_CVX:                     # pragma: no cover
        return None
    n = f.shape[0]
    A = _build_conv_matrix(f)

    g = cp.Variable(n, nonneg=True)
    constraints = [A @ g <= M]
    prob = cp.Problem(cp.Maximize(cp.sum(g)), constraints)

    try:
        prob.solve(solver=cp.ECOS, warm_start=True,
                   max_iters=2000, abstol=1e-9, reltol=1e-9)
    except Exception:                    # pragma: no cover
        return None

    if prob.status not in ("optimal", "optimal_inaccurate"):
        return None
    return np.asarray(g.value).flatten()


def frank_wolfe_refine(f0: np.ndarray, time_deadline: float,
                       max_iter: int = 200) -> np.ndarray:
    """
    Conditional‑gradient (Frank‑Wolfe) improvement of a probability vector.
    Returns the best vector found (still Σ = 1, non‑negative).
    """
    f = f0.copy()
    best_f = f.copy()
    best_M = full_convolution(f).max()
    it = 0

    # a modest set of candidate mixing factors
    t_candidates = np.array([0.02, 0.05, 0.10, 0.15, 0.20, 0.25, 0.30])

    while time.time() < time_deadline and it < max_iter:
        it += 1
        cur_M = full_convolution(f).max()

        g = _solve_lp_for_g(f, cur_M)
        if g is None:
            break
        sum_g = g.sum()
        if sum_g <= 0:
            break
        # normalise g to the simplex
        g_hat = g / sum_g

        # line‑search: pick the best t that actually reduces the max‑conv
        best_t = 0.0
        best_new_M = cur_M
        for t in t_candidates:
            f_candidate = (1.0 - t) * f + t * g_hat
            M_candidate = full_convolution(f_candidate).max()
            if M_candidate < best_new_M:
                best_new_M = M_candidate
                best_t = t

        if best_t == 0.0:
            # no improvement with the tested step sizes → stop early
            break

        # update current iterate
        f = (1.0 - best_t) * f + best_t * g_hat
        cur_M = best_new_M

        if cur_M < best_M:
            best_M = cur_M
            best_f = f.copy()

    return best_f

def subgradient_polish(a0: np.ndarray, time_deadline: float,
                       step0: float = 0.12) -> np.ndarray:
    """Projected sub‑gradient descent on max(conv(a,a))."""
    a = a0.copy()
    best_a = a.copy()
    best_M = full_convolution(a).max()
    t = 0
    n = a.shape[0]

    while time.time() < time_deadline:
        t += 1
        conv = full_convolution(a)
        M = conv.max()
        # collect all indices that are (almost) active
        active = np.where(conv >= M - 1e-12)[0]

        g = np.zeros_like(a)
        # the sub‑gradient of each active entry is 2 * a shifted
        for k in active:
            idx = np.arange(n)
            j = k - idx
            mask = (j >= 0) & (j < n)
            g[mask] += 2.0 * a[j[mask]]

        step = step0 / np.sqrt(t)
        a = a - step * g
        a = project_to_simplex(a)

        # occasional tiny random jitter to avoid tiny plateaus
        if t % 1000 == 0:
            jitter = np.random.randn(n) * (step0 * 0.01 / np.sqrt(t))
            a += jitter
            a = project_to_simplex(a)

        M_new = full_convolution(a).max()
        if M_new < best_M:
            best_M = M_new
            best_a = a.copy()

    return best_a

        cand = project_to_simplex(cand)

        M_cand = full_convolution(cand).max()
        if M_cand < best_M:
            best_M = M_cand
            best_a = cand.copy()
            a = cand
            step_scale = min(step_scale * 1.07, 0.5)
            no_improve = 0
        else:
            step_scale *= 0.985
            no_improve += 1

        it += 1
        # occasional random re‑jitter to keep the neighbourhood lively
        if no_improve > 100_000:
            jitter = np.random.randn(n) * 0.001
            a = best_a + jitter
            a = project_to_simplex(a)
            no_improve = 0

    return best_a


# ----------------------------------------------------------------------
# Main entry point required by the harness
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 12345, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative coefficient sequence that minimises
    ``evaluate_sequence``.  Returns a plain Python list of floats.
    """
    np.random.seed(seed)
    start_time = time.time()
    # leave a small safety margin for the harness bookkeeping
    deadline = start_time + budget_s - 5.0

    # -----------------------------------------------------------------
    # 0. Optional fallback from a previously discovered construction
    # -----------------------------------------------------------------
    best_seq = None
    best_val = float("inf")
    try:
        fallback = [float(x) for x in height_sequence_1]   # type: ignore
        val = evaluate_sequence(fallback)
        print(f"[fallback] score = {val:.6f}")
        if val < best_val:
            best_val, best_seq = val, fallback
    except Exception:
        pass

    n = 2048                                          # length of the step function

    # -----------------------------------------------------------------
    # 1. Global parametric sweep
    # -----------------------------------------------------------------
    pll_deadline = deadline - 400.0                    # reserve time for later phases
    if time.time() < pll_deadline:
        a_initial = best_initial_vector(n, pll_deadline)
        if a_initial is not None:
            val = compute_score(a_initial)
            print(f"[global sweep] best score = {val:.6f}")
            if val < best_val:
                best_val, best_seq = val, a_initial.tolist()

    # -----------------------------------------------------------------
    # 2. Frank‑Wolfe LP refinement (if cvxpy is available)
    # -----------------------------------------------------------------
    fw_deadline = deadline - 200.0
    if best_seq is not None and _HAVE_CVX and time.time() < fw_deadline:
        f0 = np.array(best_seq, dtype=float)
        f_refined = frank_wolfe_refine(f0, fw_deadline, max_iter=200)
        val = compute_score(f_refined)
        print(f"[Frank‑Wolfe] refined score = {val:.6f}")
        if val < best_val:
            best_val, best_seq = val, f_refined.tolist()

    # -----------------------------------------------------------------
    # 3. Sub‑gradient polishing
    # -----------------------------------------------------------------
    sub_deadline = deadline - 80.0
    if best_seq is not None and time.time() < sub_deadline:
        a0 = np.array(best_seq, dtype=float)
        a_polished = subgradient_polish(a0, sub_deadline, step0=0.10)
        val = compute_score(a_polished)
        print(f"[sub‑grad] polished score = {val:.6f}")
        if val < best_val:
            best_val, best_seq = val, a_polished.tolist()

    # -----------------------------------------------------------------
    # 4. Final stochastic hill‑climb
    # -----------------------------------------------------------------
    hill_deadline = deadline - 5.0
    if best_seq is not None and time.time() < hill_deadline:
        a_start = np.array(best_seq, dtype=float)
        a_final = hill_climb(a_start, hill_deadline)
        val = compute_score(a_final)
        print(f"[hill‑climb] final score = {val:.6f}")
        if val < best_val:
            best_val, best_seq = val, a_final.tolist()

    # -----------------------------------------------------------------
    # Safeguard – fall back to a uniform vector if nothing was found
    # -----------------------------------------------------------------
    if best_seq is None:
        best_seq = [1.0 / n] * n
        best_val = evaluate_sequence(best_seq)

    # -----------------------------------------------------------------
    # Final report (using the official evaluate_sequence)
    # -----------------------------------------------------------------
    final_score = evaluate_sequence(best_seq)
    print(f"\nSearch finished. Reported score = {final_score:.6f}")
    return best_seq
