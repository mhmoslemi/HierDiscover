def run_adam(start_a, deadline, max_iters):
    t0 = time.time()
    best_seq = start_a
    best_sc = compute_score(start_a)
    y = np.log(start_a + 1e-30)  # add epsilon
    m = np.zeros_like(y)
    v = np.zeros_like(y)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    for it in range(1, max_iters + 1):
        if time.time() >= deadline:
            break
        # compute a via softmax
        # To avoid overflow: y_shift = y - max(y)
        y_shift = y - np.max(y)
        exps = np.exp(y_shift)
        a = exps / np.sum(exps)
        conv = convolution_fft(a, a)
        max_c = conv.max()
        k = int(conv.argmax())
        # gradient for max conv (subgradient)
        g_a = np.zeros_like(a)
        i_start = max(0, k - (len(a) - 1))
        i_end = min(len(a)-1, k)
        idx = np.arange(i_start, i_end+1)
        # gradient: 2 * a[k-idx] maybe.
        g_a[idx] = 2.0 * a[k - idx]
        # optionally multiply by (2 * n) to match objective scaling:
        g_a = g_a * (2.0 * len(a))
        # chain rule to y
        dot = np.dot(a, g_a)
        g_y = a * (g_a - dot)
        # Adam update
        m = beta1 * m + (1 - beta1) * g_y
        v = beta2 * v + (1 - beta2) * (g_y ** 2)
        m_hat = m / (1 - beta1 ** it)
        v_hat = v / (1 - beta2 ** it)
        # step size schedule
        alpha = 0.05 / np.sqrt(it)
        y = y - alpha * m_hat / (np.sqrt(v_hat) + eps)
        # occasional evaluate
        if it % 500 == 0:
            # compute a and score
            y_shift = y - np.max(y)
            exps = np.exp(y_shift)
            a = exps / np.sum(exps)
            sc = compute_score(a)
            if sc < best_sc:
                best_sc, best_seq = sc, a.copy()
                print(f"[{time.time():.1f}] Adam improved best to {best_sc:.6f}")
    # final check
    y_shift = y - np.max(y)
    exps = np.exp(y_shift)
    a = exps / np.sum(exps)
    sc = compute_score(a)
    if sc < best_sc:
        best_sc, best_seq = sc, a.copy()
    return best_seq, best_sc

"""
Advanced search for a low‑ℓ∞ auto‑convolution step‑function.

The algorithm proceeds in several timed phases:

1. Choose a sequence length `n` (larger when time permits).
2. Fast grid search over three simple parametric families
   (exponential, power‑law, discrete Beta) and a few random Dirichlet vectors.
   The best vector so far is kept.
3. **Adam‑mirror descent** on the exact max‑convolution:
   - The vector is represented in log‑softmax space so that positivity
     and the sum‑to‑1 simplex are automatic.
   - In each iteration the true active convolution entry is located,
     its sub‑gradient is formed and an Adam update (with KL‑mirror geometry)
     is applied to the log‑probabilities.
4. Several short “perturbed‑restart’’ runs: small Gaussian noise is added
   to the current best log‑vector and Adam is run for a few thousand steps.
   This helps escape local minima.
5. A final targeted hill‑climb that repeatedly steals mass from the index
   contributing most to the current maximal convolution entry and redistributes
   it elsewhere, accepting only improving moves.
6. Return the best sequence discovered (as a plain Python list of floats).

All numeric work uses FFT‑based full convolutions for speed and respects the
1000 s wall‑clock budget.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Basic linear‑algebra utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full aperiodic convolution using a real FFT (zero‑padded)."""
    m = a.size + b.size - 1
    nfft = next_pow2(m)
    A = np.fft.rfft(a, n=nfft)
    B = np.fft.rfft(b, n=nfft)
    C = A * B
    return np.fft.irfft(C, n=nfft)[:m]


def compute_score(a: np.ndarray) -> float:
    """
    Exact objective used by `evaluate_sequence` for a vector already
    normalised to sum = 1. Returns 2·n·max(conv) where conv = a∗a.
    """
    n = a.shape[0]
    conv = convolution_fft(a, a)
    return 2.0 * n * float(conv.max())


def softmax(y: np.ndarray) -> np.ndarray:
    """Numerically stable soft‑max (sum to 1, all entries >0)."""
    y_shift = y - np.max(y)
    exps = np.exp(y_shift)
    return exps / exps.sum()


# ----------------------------------------------------------------------
# Parameterised families of candidate vectors
# ----------------------------------------------------------------------


def exponential_shape(n: int, lam: float) -> np.ndarray:
    """a_i ∝ exp(-lam·i), normalised to sum = 1."""
    i = np.arange(n, dtype=float)
    if lam == 0.0:
        w = np.ones_like(i)
    else:
        w = np.exp(-lam * i)
    return w / w.sum()


def power_shape(n: int, p: float) -> np.ndarray:
    """a_i ∝ (i+1)^(-p), normalised to sum = 1."""
    i = np.arange(1, n + 1, dtype=float)
    if p == 0.0:
        w = np.ones_like(i)
    else:
        w = i ** (-p)
    return w / w.sum()


def beta_shape(n: int, a: float, b: float) -> np.ndarray:
    """
    Discrete Beta distribution on {0,…,n‑1}:
        a_i ∝ ((i+0.5)/n)^(a‑1) * (1‑((i+0.5)/n))^(b‑1)
    """
    x = (np.arange(n, dtype=float) + 0.5) / n
    w = np.power(x, a - 1.0) * np.power(1.0 - x, b - 1.0)
    w[np.isnan(w)] = 0.0
    w = np.maximum(w, 0.0)
    if w.sum() == 0.0:
        w = np.ones_like(w)
    return w / w.sum()


# ----------------------------------------------------------------------
# Fast grid scans for the simple families
# ----------------------------------------------------------------------


def scan_exponential_power(n: int, deadline: float):
    """Exhaustive scan of exponential and power‑law families (coarse grids)."""
    best_a, best_sc = None, float("inf")

    # exponential λ ∈ [0, 5e‑3] (log‑space) + the uniform case λ=0
    lam_grid = np.concatenate((np.array([0.0]), np.logspace(-7, -2, 50)))
    for lam in lam_grid:
        if time.time() >= deadline:
            break
        a = exponential_shape(n, lam)
        sc = compute_score(a)
        if sc < best_sc:
            best_sc, best_a = sc, a.copy()

    # power‑law p ∈ [0, 6] (linear grid)
    p_grid = np.linspace(0.0, 6.0, 80)
    for p in p_grid:
        if time.time() >= deadline:
            break
        a = power_shape(n, p)
        sc = compute_score(a)
        if sc < best_sc:
            best_sc, best_a = sc, a.copy()

    return best_a, best_sc


def scan_beta_family(n: int, deadline: float):
    """Modest 2‑parameter Beta family scan."""
    best_a, best_sc = None, float("inf")
    # grids concentrate on 0.2–3.0 where the shape changes most rapidly
    grid = np.concatenate((
        np.logspace(-0.5, -0.1, 8),
        np.linspace(0.5, 3.0, 8)
    ))
    for a in grid:
        for b in grid:
            if time.time() >= deadline:
                return best_a, best_sc
            a_vec = beta_shape(n, a, b)
            sc = compute_score(a_vec)
            if sc < best_sc:
                best_sc, best_a = sc, a_vec.copy()
    return best_a, best_sc


# ----------------------------------------------------------------------
# Adam‑mirror descent on the exact max‑convolution
# ----------------------------------------------------------------------


def adam_optimize(start_a: np.ndarray,
                  deadline: float,
                  max_iters: int = 300_000,
                  alpha0: float = 0.08) -> tuple[np.ndarray, float]:
    """
    Mirror‑Adam on the log‑softmax representation.
    Returns the best sequence found and its exact score.
    """
    eps = 1e-30
    y = np.log(start_a + eps)               # log‑probability vector
    n = start_a.shape[0]
    beta1, beta2 = 0.9, 0.999
    m = np.zeros_like(y)
    v = np.zeros_like(y)
    best_a = softmax(y)
    best_sc = compute_score(best_a)

    for it in range(1, max_iters + 1):
        if time.time() >= deadline:
            break

        a = softmax(y)                       # current simplex point

        # full auto‑convolution, locate the active index
        conv = convolution_fft(a, a)
        k_max = int(conv.argmax())

        # sub‑gradient of the active entry (2·a_{k‑i} as derived)
        g_a = np.zeros_like(a)
        i_start = max(0, k_max - (n - 1))
        i_end = min(n - 1, k_max)
        idx = np.arange(i_start, i_end + 1)
        g_a[idx] = 2.0 * a[k_max - idx]      # factor 2 for the quadratic term

        # (optional) scale by 2·n to match the true objective; absorbed in step‑size
        # g_a *= (2.0 * n)

        # chain rule to log‑softmax variables
        dot = np.dot(a, g_a)
        g_y = a * (g_a - dot)                # ∂L/∂y

        # Adam moments
        m = beta1 * m + (1.0 - beta1) * g_y
        v = beta2 * v + (1.0 - beta2) * (g_y ** 2)

        # bias‑corrected estimates
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        # learning‑rate schedule (α ∝ 1/√t)
        alpha = alpha0 / np.sqrt(it)

        # KL‑mirror (exponentiated) Adam update
        y -= alpha * m_hat / (np.sqrt(v_hat) + 1e-12)

        # occasional exact evaluation
        if it % 500 == 0:
            a = softmax(y)
            sc = compute_score(a)
            if sc < best_sc:
                best_sc, best_a = sc, a.copy()
                print(f"[{time.time():.1f}] Adam improved best to {best_sc:.6f}")

    # final check
    a = softmax(y)
    sc = compute_score(a)
    if sc < best_sc:
        best_sc, best_a = sc, a.copy()
    return best_a, best_sc


# ----------------------------------------------------------------------
# Perturb‑restart wrapper around Adam (adds stochastic jumps)
# ----------------------------------------------------------------------


def perturbed_restart_optimize(base_a: np.ndarray,
                               deadline: float,
                               n_rest: int = 3,
                               iters_per: int = 40_000,
                               noise_scale: float = 0.15) -> tuple[np.ndarray, float]:
    """
    Starting from `base_a`, perform `n_rest` short Adam runs each
    after adding Gaussian noise (in log‑space) of size `noise_scale`.
    Returns the best vector seen.
    """
    best_a = base_a.copy()
    best_sc = compute_score(best_a)
    for r in range(n_rest):
        if time.time() >= deadline:
            break
        # perturb log‑probabilities
        y = np.log(best_a + 1e-30)
        y += np.random.normal(scale=noise_scale, size=y.shape)
        # run a short Adam optimisation
        a_candidate, sc_candidate = adam_optimize(
            softmax(y), deadline, max_iters=iters_per, alpha0=0.06
        )
        if sc_candidate < best_sc:
            best_sc, best_a = sc_candidate, a_candidate.copy()
            print(f"[{time.time():.1f}] Perturb‑restart {r+1} improved best to {best_sc:.6f}")
    return best_a, best_sc


# ----------------------------------------------------------------------
# Targeted hill‑climb polishing (same idea as the previous version)
# ----------------------------------------------------------------------


def targeted_hill_climb(start_seq: np.ndarray,
                        deadline: float,
                        max_steps: int = 300_000) -> tuple[np.ndarray, float]:
    """
    Greedy walk that always extracts mass from the index that contributes most
    to the current maximal convolution entry and redistributes it elsewhere.
    Accepts only improving moves.
    """
    a = start_seq.copy()
    n = a.size
    best_a = a.copy()
    best_sc = compute_score(best_a)

    step_scale = 0.10
    steps = 0

    while steps < max_steps and time.time() < deadline:
        conv = convolution_fft(best_a, best_a)
        k_max = int(conv.argmax())

        i_start = max(0, k_max - (n - 1))
        i_end = min(n - 1, k_max)
        idx = np.arange(i_start, i_end + 1)

        # contribution of each participating index to the active sum
        contrib = best_a[idx] * best_a[k_max - idx]
        i_max = idx[np.argmax(contrib)]

        # pick a different destination
        j = np.random.randint(n)
        if j == i_max:
            continue

        delta = step_scale * best_a[i_max]
        if delta <= 0.0:
            break

        cand = best_a.copy()
        cand[i_max] -= delta
        cand[j] += delta
        # re‑project onto the simplex (numerical safety)
        cand = np.maximum(cand, 0.0)
        cand /= cand.sum()

        sc = compute_score(cand)
        if sc < best_sc:
            best_sc, best_a = sc, cand.copy()
            step_scale = min(step_scale * 1.07, 0.5)
        else:
            step_scale *= 0.96

        steps += 1

    return best_a, best_sc
