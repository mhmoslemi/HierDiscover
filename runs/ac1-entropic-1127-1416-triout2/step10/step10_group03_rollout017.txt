def eg_optimize(a, deadline):
    start = time.time()
    t = 0
    best_a = a.copy()
    best_score = compute_score(a)
    while time.time() < deadline:
        conv = convolution_fft(a, a)
        max_conv = conv.max()
        # evaluate true objective:
        cur_score = 2.0 * len(a) * max_conv
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
        # temperature schedule
        elapsed = time.time() - start
        total = deadline - start
        fraction = elapsed / total
        beta = 5.0 + fraction * (5000.0 - 5.0)  # up to high
        shifted = conv - max_conv
        # stable softmax
        exp_vals = np.exp(beta * shifted)  # max = 1
        w = exp_vals / exp_vals.sum()
        grad = 2.0 * beta * np.correlate(w, a, mode="valid")
        # step size
        t += 1
        eta = 0.5 / (np.sqrt(t) * (1 + beta/1000))  # some scaling
        # EG update
        a = a * np.exp(-eta * grad)
        # re-normalize
        a_sum = a.sum()
        if a_sum == 0 or np.isnan(a_sum):
            a = np.full_like(a, 1.0 / len(a))
        else:
            a = a / a_sum
        # occasional log
        if t % 5000 == 0:
            print(f"[EG] it {t}, time {time.time()-start:.1f}s, best {best_score:.6f}, beta {beta:.1f}")
    return best_a, best_score
```

def eg_descent(a, start_score, deadline):
    n = a.shape[0]
    cur_a = a.copy()
    best_a = cur_a.copy()
    best_score = start_score
    it = 0
    start = time.time()
    while time.time() < deadline:
        conv = convolution_fft(cur_a, cur_a)
        max_conv = conv.max()
        cur_score = 2.0 * n * float(max_conv)
        if cur_score < best_score:
            best_score = cur_score
            best_a = cur_a.copy()
        # temperature schedule
        elapsed = time.time() - start
        total = deadline - start
        fraction = elapsed / total if total>0 else 1.0
        beta = 5.0 + fraction * (5000.0 - 5.0)
        shifted = conv - max_conv
        exp_vals = np.exp(beta * shifted)  # stable: max=0 =>exp<=1
        w = exp_vals / exp_vals.sum()
        grad = 2.0 * beta * np.correlate(w, cur_a, mode="valid")
        it += 1
        eta = 0.02 / (np.sqrt(it) * (1.0 + beta*0.001))  # adjust
        # exponentiated gradient update
        cur_a = cur_a * np.exp(-eta * grad)
        # Prevent underflow / enforce positivity
        cur_a = np.maximum(cur_a, 1e-15)
        s = cur_a.sum()
        cur_a /= s
        if it % 5000 == 0:
            print(f"[EG] it {it:6d} time {time.time()-start:6.1f}s best {best_score:.6f} beta {beta:.1f}")
    return best_a, best_score
```

"""
Exponentiated‑gradient search for a non‑negative step function
that minimises evaluate_sequence.  The algorithm mixes several
seed families, an EG descent on a smoothed max‑convolution
objective (soft‑max with a temperature that grows to ≈5e3) and
a final annealed hill‑climb.  Time is split between a primary
search (n≈2048) and, if budget permits, a secondary search at
n≈4096.  The best sequence found is returned.
"""

import time
import numpy as np

def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()

def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full linear convolution using a real FFT."""
    m = a.shape[0] + b.shape[0] - 1
    n_fft = next_pow2(m)
    A = np.fft.rfft(a, n=n_fft)
    B = np.fft.rfft(b, n=n_fft)
    C = A * B
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv

def compute_score(a: np.ndarray) -> float:
    """Exact objective for a probability vector (sum = 1):
    2·n·max(conv(a,a)).
    """
    n = a.shape[0]
    conv = convolution_fft(a, a)
    return 2.0 * n * float(conv.max())

def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection onto the probability simplex."""
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / v.shape[0])

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, v.shape[0] + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.where(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

# ----------------------------------------------------------------------
# Seed generators
# ----------------------------------------------------------------------
def generate_deterministic(n: int) -> np.ndarray:
    i = np.arange(1, n + 1, dtype=float)
    a = np.sin(np.pi * i / (n + 1.0)) ** 2
    a /= a.sum()
    return a

def generate_beta_family(n: int, p: float) -> np.ndarray:
    i = np.arange(n, dtype=float)
    w = ((i + 1.0) * (n - i)) ** p
    w[w < 0] = 0.0
    a = w / w.sum()
    return a

def random_dirichlet(n: int, conc: float = 1.0) -> np.ndarray:
    if conc == 1.0:
        a = np.random.rand(n)
    else:
        a = np.random.gamma(conc, 1.0, size=n)
    a /= a.sum()
    return a

def random_spike(n: int, k: int = 70) -> np.ndarray:
    idx = np.random.choice(n, size=min(k, n), replace=False)
    w = np.random.rand(len(idx))
    a = np.zeros(n, dtype=float)
    a[idx] = w
    a /= a.sum()
    return a

# ----------------------------------------------------------------------
# Exponentiated‑gradient descent on a smoothed max‑convolution
# ----------------------------------------------------------------------
def eg_descent(a: np.ndarray, start_score: float, deadline: float):
    """
    Multiplicative (exponentiated) gradient descent on the surrogate
    Lβ(a)=log Σ exp(β·conv(a,a)).
    The temperature β is increased linearly up to ≈5 000.
    """
    n = a.shape[0]
    cur_a = a.copy()
    best_a = cur_a.copy()
    best_score = start_score
    it = 0
    start = time.time()
    while time.time() < deadline:
        conv = convolution_fft(cur_a, cur_a)
        max_conv = conv.max()
        cur_score = 2.0 * n * float(max_conv)
        if cur_score < best_score:
            best_score = cur_score
            best_a = cur_a.copy()

        elapsed = time.time() - start
        total = deadline - start
        fraction = elapsed / total if total > 0 else 1.0
        beta = 5.0 + fraction * (5000.0 - 5.0)

        # soft‑max weights
        shifted = conv - max_conv                 # max becomes 0
        exp_vals = np.exp(beta * shifted)        # ≤ 1, stable
        w = exp_vals / exp_vals.sum()            # distribution over 2n‑1 points

        # gradient of the smooth surrogate
        grad = 2.0 * beta * np.correlate(w, cur_a, mode="valid")

        it += 1
        eta = 0.12 / (np.sqrt(it) * (1.0 + beta * 0.001))  # step size

        # exponentiated‑gradient update (multiplicative)
        cur_a = cur_a * np.exp(-eta * grad)

        # keep numbers away from exact zero – helps numerical stability
        cur_a = np.maximum(cur_a, 1e-15)

        cur_a /= cur_a.sum()

        if it % 5000 == 0:
            print(f"[EG] it {it:6d} | time {time.time()-start:6.1f}s | best {best_score:.6f} | β {beta:.1f}")

    return best_a, best_score

# ----------------------------------------------------------------------
# Simulated‑annealing hill‑climb (pairwise mass transfer)
# ----------------------------------------------------------------------
def annealing_hill_climb(a: np.ndarray, deadline: float):
    """
    Stochastic moves (transfer a fraction of mass from one index to another)
    are accepted if they improve the exact objective or with a probability
    that decreases with a linear temperature schedule.
    """
    n = a.shape[0]
    best_a = a.copy()
    best_score = compute_score(best_a)
    start = time.time()
    it = 0
    while time.time() < deadline:
        it += 1
        # linear temperature decay
        elapsed = time.time() - start
        total = deadline - start
        t_frac = elapsed / total if total > 0 else 1.0
        temp = 0.12 * (1.0 - t_frac) + 1e-5 * t_frac  # from ~0.12 to ~0

        cand = best_a.copy()
        i, j = np.random.randint(n, size=2)
        if cand[i] > 0.0:
            delta = np.random.rand() * 0.25 * cand[i]   # move up to 25 % of source mass
            cand[i] -= delta
            cand[j] += delta

        # re‑normalise (numerical drift)
        cand /= cand.sum()
        cand_score = compute_score(cand)

        if cand_score < best_score:
            best_a, best_score = cand, cand_score
        else:
            # Metropolis acceptance
            if np.random.rand() < np.exp(-(cand_score - best_score) / max(temp, 1e-12)):
                best_a, best_score = cand, cand_score

        if it % 5000 == 0:
            print(f"[HC] it {it:6d} | time {time.time()-start:6.1f}s | best {best_score:.6f}")

    return best_a, best_score

# ----------------------------------------------------------------------
# Full search for a given length
# ----------------------------------------------------------------------
def run_search(n: int, deadline: float):
    """
    Run the whole optimisation pipeline for a vector of length `n`
    up to the supplied absolute `deadline`.
    Returns the best vector (as a NumPy array) and its exact score.
    """
    # ---- 0. initialise a diverse set of seeds -----------------------
    seeds = []
    seeds.append(generate_deterministic(n))
    for p in np.linspace(-2.5, 2.5, 9):
        seeds.append(generate_beta_family(n, p))
    for conc in (0.2, 0.5, 1.0, 2.0, 5.0):
        for _ in range(2):
            seeds.append(random_dirichlet(n, conc))
    for k in (30, 50, 80):
        seeds.append(random_spike(n, k))
    # uniform baseline
    seeds.append(np.full(n, 1.0 / n, dtype=float))

    # fallback from previous state‑of‑the‑art if available
    try:
        fb = np.array(height_sequence_1, dtype=float)
        if fb.shape[0] == n:
            seeds.append(fb / fb.sum())
    except Exception:
        pass

    # ---- 1. pick the best seed (exact objective) -------------------
    best_a = None
    best_val = float("inf")
    for s in seeds:
        s = project_to_simplex(np.asarray(s, dtype=float))
        val = compute_score(s)
        if val < best_val:
            best_val = val
            best_a = s.copy()
    if best_a is None:
        best_a = np.full(n, 1.0 / n, dtype=float)
        best_val = compute_score(best_a)

    print(f"Chosen initial seed: score = {best_val:.6f}")

    # ---- 2. exponentiated‑gradient descent (≈70 % of remaining time) ---
    now = time.time()
    rem = deadline - now
    phase1_deadline = now + rem * 0.70

    # ---- 3. final hill‑climb (remaining time) ------------------------
    if time.time() < deadline - 0.5:

    return cur_a, cur_score
