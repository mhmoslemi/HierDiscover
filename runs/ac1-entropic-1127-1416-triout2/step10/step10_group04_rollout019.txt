def simulated_annealing(seq, time_deadline, init_temp=0.1, final_temp=1e-6, cooling_rate=0.9995):

    a = seq.copy()
    best = a.copy()
    best_score = compute_score(a)
    temp = init_temp
    step_scale = 0.05
    while time.time() < time_deadline:
        # propose move: pick i,j distinct
        i, j = np.random.randint(n, size=2)
        if i == j: continue
        # fraction of a_i to move
        delta = step_scale * a[i]
        cand = a.copy()
        cand[i] -= delta
        cand[j] += delta
        # projection to simplex (ensures nonnegativity and sum)
        cand = project_to_simplex(cand)
        cur = compute_score(cand)
        diff = cur - best_score
        if diff < 0 or np.random.rand() < np.exp(-diff / temp):
            a = cand
            if cur < best_score:
                best_score = cur
                best = cand.copy()
                # adapt step scale?
                step_scale = min(step_scale * 1.05, 0.5)
        else:
            step_scale *= 0.99

        # cool temperature slowly
        temp = max(temp * cooling_rate, final_temp)

    return best, best_score

"""
Non‑negative step‑function optimiser for ``evaluate_sequence``.

Main ideas
----------
* Several analytical families (uniform, central‑binomial, power‑law,
  geometric, random Dirichlet) give a diverse set of starters.
* A long smooth‑max Adam phase minimises a log‑sum‑exp surrogate of the
  true L∞ objective.  The temperature is annealed to a large value
  (β≈1500) so the surrogate becomes very close to the true max.
* A projected sub‑gradient stage refines the result using the exact
  max‑convolution (averaging the sub‑gradient over all active indices).
* A simulated‑annealing / hill‑climbing stage moves small amounts of
  mass between positions while staying on the simplex.  A large number of
  cheap moves gives a fine‑grained polishing.
* All phases respect the global wall‑clock deadline (default 1000 s) and
  keep the best vector seen so far.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Basic utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """
    Full linear convolution via real FFT.
    Returns an array of length len(a)+len(b)-1.
    """
    m = a.shape[0] + b.shape[0] - 1
    n_fft = next_pow2(m)
    A = np.fft.rfft(a, n=n_fft)
    B = np.fft.rfft(b, n=n_fft)
    C = A * B
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


def compute_score(a: np.ndarray) -> float:
    """
    Exact objective for a vector already on the probability simplex.
    Returns 2·n·max(conv(a,a)).
    """
    n = a.shape[0]
    conv = convolution_fft(a, a)
    return 2.0 * n * float(conv.max())


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of v onto the probability simplex:
        sum(v) = 1, v >= 0.
    """
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / v.shape[0])

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, v.shape[0] + 1)
    cond = u - cssv / ind > 0
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


# ----------------------------------------------------------------------
# Analytic families for initial guesses
# ----------------------------------------------------------------------


def generate_uniform(n: int) -> np.ndarray:
    """Uniform distribution (already on simplex)."""
    return np.full(n, 1.0 / n, dtype=np.float64)


def generate_central_binomial(n: int) -> np.ndarray:
    """a_i = C(2i,i) / 4^i, monotone decreasing, then simplex‑projected."""
    a = np.empty(n, dtype=np.float64)
    a[0] = 1.0
    for i in range(1, n):
        a[i] = a[i - 1] * (2 * i - 1) / (2.0 * i)
    return project_to_simplex(a)


def generate_powerlaw(n: int, p: float) -> np.ndarray:
    """a_i ∝ (i+1)^(-p), normalised to sum 1."""
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = idx ** (-p)
    return project_to_simplex(a)


def generate_geometric(n: int, r: float) -> np.ndarray:
    """Geometric law a_i = (1‑r)·r^i, normalised."""
    i = np.arange(n, dtype=np.float64)
    a = (1.0 - r) * (r ** i)
    return project_to_simplex(a)


def generate_dirichlet(n: int, alpha: float = 0.5) -> np.ndarray:
    """Random Dirichlet draw – useful for cheap random restarts."""
    x = np.random.gamma(alpha, 1.0, size=n)
    return project_to_simplex(x)


# ----------------------------------------------------------------------
# Smooth (log‑sum‑exp) optimisation – Adam
# ----------------------------------------------------------------------


def smooth_gradient_optimize(
    start_a: np.ndarray,
    time_deadline: float,
    max_iters: int = 500_000,
    beta_start: float = 5.0,
    beta_end: float = 1500.0,
    step0: float = 1.0,
    adam_beta1: float = 0.9,
    adam_beta2: float = 0.999,
    adam_eps: float = 1e-8,
) -> tuple[np.ndarray, float]:
    """
    Adam optimiser for the smoothed max‑convolution objective.
    Uses a log‑sum‑exp surrogate with an annealed temperature β.
    """
    a = start_a.astype(np.float64).copy()
    n = a.shape[0]
    m1 = np.zeros_like(a)
    m2 = np.zeros_like(a)

    best_a = a.copy()
    best_score = compute_score(a)

    for it in range(1, max_iters + 1):
        if time.time() >= time_deadline:
            break

        # anneal temperature (smoothness) β
        t_frac = it / max_iters
        beta = beta_start * (beta_end / beta_start) ** t_frac

        conv = convolution_fft(a, a)
        maxc = conv.max()

        # soft‑max weights (log‑sum‑exp trick)
        exp_w = np.exp(beta * (conv - maxc))
        w = exp_w / exp_w.sum()

        # gradient of the surrogate:
        rev_a = a[::-1]
        grad_full = convolution_fft(rev_a, w)      # length 3n‑2
        grad = 2.0 * grad_full[n - 1:n - 1 + n]

        # Adam update
        m1 = adam_beta1 * m1 + (1.0 - adam_beta1) * grad
        m2 = adam_beta2 * m2 + (1.0 - adam_beta2) * (grad * grad)
        m1_hat = m1 / (1.0 - adam_beta1 ** it)
        m2_hat = m2 / (1.0 - adam_beta2 ** it)

        step = step0 / np.sqrt(it)
        a_new = a - step * m1_hat / (np.sqrt(m2_hat) + adam_eps)

        a_new = project_to_simplex(a_new)

        a = a_new

        # occasional exact evaluation
        if it % 500 == 0:
            cur_score = compute_score(a)
            if cur_score < best_score:
                best_score = cur_score
                best_a = a.copy()

    final_score = compute_score(a)
    if final_score < best_score:
        best_score, best_a = final_score, a.copy()
    return best_a, best_score


# ----------------------------------------------------------------------
# Exact sub‑gradient optimiser on the true max‑objective
# ----------------------------------------------------------------------


def subgrad_optimize(
    init_a: np.ndarray,
    time_deadline: float,
    max_iter: int = 600_000,
    step0: float = 0.06,
    tol: float = 1e-12,
) -> tuple[np.ndarray, float]:
    """
    Projected sub‑gradient descent on 2·n·max(conv(a,a)).
    The sub‑gradient is averaged over all (near) active convolution indices.
    """
    a = init_a.astype(np.float64).copy()
    n = a.shape[0]

    best_a = a.copy()
    best_score = compute_score(a)

    for t in range(1, max_iter + 1):
        if time.time() >= time_deadline:
            break

        conv = convolution_fft(a, a)
        max_val = conv.max()
        # indices that are (almost) maximal
        active = np.where(conv >= max_val - tol)[0]

        # build a simple weight vector that distributes mass uniformly over active set
        w = np.zeros_like(conv)
        w[active] = 1.0 / max(len(active), 1)

        rev_a = a[::-1]
        grad_full = convolution_fft(rev_a, w)
        grad = 2.0 * grad_full[n - 1:n - 1 + n]

        step = step0 / np.sqrt(t)
        a_new = a - step * grad
        a_new = project_to_simplex(a_new)

        cur_score = compute_score(a_new)
        if cur_score < best_score:
            best_score = cur_score
            best_a = a_new.copy()

        a = a_new

        # occasional jitter to leave flat plateaus
        if t % 1000 == 0:
            jitter = np.random.randn(n) * (step0 * 0.003 / np.sqrt(t))
            a += jitter
            a = project_to_simplex(a)

    final_score = compute_score(a)
    if final_score < best_score:
        best_score, best_a = final_score, a.copy()
    return best_a, best_score


# ----------------------------------------------------------------------
# Simulated‑annealing / hill‑climb fine polishing
# ----------------------------------------------------------------------


def simulated_annealing(
    start_seq,
    time_deadline,
    init_temp: float = 5e-4,
    final_temp: float = 1e-9,
    cooling: float = 0.9998,
) -> tuple[np.ndarray, float]:
    """
    Tiny stochastic mass‑transfer walk with a simulated‑annealing acceptance rule.
    Moves are always projected back onto the simplex.
    """
    a = np.array(start_seq, dtype=np.float64)
    n = a.shape[0]

    best_a = a.copy()
    best_score = compute_score(best_a)

    temp = init_temp
    # the amount of mass moved each step (fraction of source's current mass)
    step_scale = 0.12

    while time.time() < time_deadline and temp > final_temp:
        i, j = np.random.randint(n, size=2)
        if i == j:
            continue

        # transfer a fraction of the source to the destination
        delta = step_scale * a[i]
        cand = a.copy()
        cand[i] -= delta
        cand[j] += delta

        cand = project_to_simplex(cand)
        cur_score = compute_score(cand)
        diff = cur_score - best_score

        # accept if better, otherwise according to temperature
        if diff < 0 or np.random.rand() < np.exp(-diff / max(temp, 1e-12)):
            a = cand
            if cur_score < best_score:
                best_score = cur_score
                best_a = cand.copy()
                # when we improve we can be a little more daring
                step_scale = min(step_scale * 1.05, 0.5)

        else:
            # slightly shrink the step when we reject
            step_scale *= 0.98

        # Geometric cooling
        temp = max(temp * cooling, final_temp)

    return best_a, best_score


def final_hill_climb(
    start_seq,
    time_deadline,
    max_steps: int = 400_000,
) -> tuple[np.ndarray, float]:
    """
    Very cheap hill‑climb (only accepts strict improvements).
    Moves are swaps or fractional transfers; after each move we project onto the simplex.
    """
    a = np.array(start_seq, dtype=np.float64)
    n = a.shape[0]
    best_a = a.copy()
    best_score = compute_score(best_a)

    step_scale = 0.07

    steps = 0
    while steps < max_steps and time.time() < time_deadline:
        cand = best_a.copy()
        if np.random.rand() < 0.5:          # swap two entries
            i, j = np.random.randint(n, size=2)
            cand[i], cand[j] = cand[j], cand[i]
        else:                               # move a fraction from i → j
            i, j = np.random.randint(n, size=2)
            if i != j:
                delta = step_scale * cand[i]
                cand[i] -= delta
                cand[j] += delta
                if cand[i] < 0:
                    cand[i] = 0.0

        cand = project_to_simplex(cand)
        cur_score = compute_score(cand)

        if cur_score < best_score:
            best_score = cur_score
            best_a = cand
            step_scale = min(step_scale * 1.06, 0.5)
        else:
            step_scale *= 0.97

        steps += 1

    return best_a, best_score

# Main entry point required by the harness
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 12345, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative sequence that minimises ``evaluate_sequence``.
    Returns a plain Python list of floats (already on the simplex).
    """
    np.random.seed(seed)
    start_time = time.time()
    # keep a few seconds as safety margin for the harness
    deadline = start_time + budget_s - 4.0

    # ------------------------------------------------------------------
    # 0. Problem size (length of the sequence)
    # ------------------------------------------------------------------
    try:
        n = len(height_sequence_1)           # type: ignore[name-defined]
        init_candidates = [np.array(height_sequence_1, dtype=np.float64)]
        print(f"Using supplied height_sequence_1 of length {n}.")
    except Exception:
        n = 2048
        init_candidates = []
        print(f"No external sequence – using default length {n}.")

    # ------------------------------------------------------------------
    # 1. Build a rich set of analytic starters and a random Dirichlet
    # ------------------------------------------------------------------
    init_candidates.append(generate_uniform(n))
    init_candidates.append(generate_central_binomial(n))

    for p in np.linspace(0.3, 1.2, 9):
        init_candidates.append(generate_powerlaw(n, float(p)))

    for r in [0.95, 0.97, 0.98, 0.985, 0.99, 0.992, 0.995]:
        init_candidates.append(generate_geometric(n, float(r)))

    # a few random Dirichlet draws – cheap way to diversify the search
    for _ in range(4):
        init_candidates.append(generate_dirichlet(n, alpha=0.5))

    # keep the best starter according to the exact score
    best_seq = None
    best_val = float("inf")
    for idx, cand in enumerate(init_candidates):
        cand = project_to_simplex(cand)
        val = compute_score(cand)
        print(f"Starter {idx:02d}: score = {val:.6f}")
        if val < best_val:
            best_val = val
            best_seq = cand.copy()

    print(f"Best starter score = {best_val:.6f}")

    # ------------------------------------------------------------------
    # 2. Smooth (log‑sum‑exp) Adam optimisation – longest phase.
    # ------------------------------------------------------------------
    smooth_deadline = deadline - 300.0   # reserve time for later phases
    if time.time() < smooth_deadline:
        remaining = smooth_deadline - time.time()
        # estimate feasible number of iterations (≈ 2500 it/s)
        max_iters = int(min(1_200_000, remaining * 2500))
        print(f"Smooth Adam optimisation → up to {max_iters} iters.")
        a_smooth, smooth_score = smooth_gradient_optimize(
            best_seq,
            time_deadline=smooth_deadline,
            max_iters=max_iters,
            beta_start=5.0,
            beta_end=1500.0,
            step0=1.0,
        )
        if smooth_score < best_val:
            best_val = smooth_score
            best_seq = a_smooth.copy()
            print(f"Smooth phase improved score to {best_val:.6f}")

    # ------------------------------------------------------------------
    # 3. Projected sub‑gradient polishing
    # ------------------------------------------------------------------
    subgrad_deadline = deadline - 130.0
    if time.time() < subgrad_deadline:
        remaining = subgrad_deadline - time.time()
        max_iter = int(min(1_000_000, remaining * 2100))
        print(f"Sub‑gradient optimisation → up to {max_iter} iters.")
        a_sub, sub_score = subgrad_optimize(
            best_seq,
            time_deadline=subgrad_deadline,
            max_iter=max_iter,
            step0=0.05,
        )
        if sub_score < best_val:
            best_val = sub_score
            best_seq = a_sub.copy()
            print(f"Sub‑gradient phase improved score to {best_val:.6f}")

    # ------------------------------------------------------------------
    # 4. Simulated‑annealing fine polishing (mass transfers)
    # ------------------------------------------------------------------
    anneal_deadline = deadline - 40.0
    if time.time() < anneal_deadline:
        print("Running simulated‑annealing fine polishing.")
        a_anneal, anneal_score = simulated_annealing(
            best_seq.tolist(),
            time_deadline=anneal_deadline,
            init_temp=8e-4,
            final_temp=1e-12,
            cooling=0.9997,
        )
        if anneal_score < best_val:
            best_val = anneal_score
            best_seq = a_anneal.copy()
            print(f"Anneal phase improved score to {best_val:.6f}")

    # ------------------------------------------------------------------
    # 5. Final cheap hill‑climb
    # ------------------------------------------------------------------
    hill_deadline = deadline - 2.0
    if time.time() < hill_deadline:
        steps = 500_000
        print(f"Final hill‑climb for up to {steps} steps.")
        polished_seq, hill_score = final_hill_climb(
            best_seq.tolist(),
            time_deadline=hill_deadline,
            max_steps=steps,
        )
        if hill_score < best_val:
            best_val = hill_score
            best_seq = np.array(polished_seq, dtype=np.float64)
            print(f"Hill‑climb improved final score to {best_val:.6f}")

    # ------------------------------------------------------------------
    # 6. Fall‑back safety
    # ------------------------------------------------------------------
    if best_seq is None:
        best_seq = generate_uniform(n)
        best_val = compute_score(best_seq)
        print("No candidate produced – falling back to uniform.")

    # ------------------------------------------------------------------
    # 7. Final reporting with the provided evaluate_sequence()
    # ------------------------------------------------------------------
    final_score = evaluate_sequence(best_seq.tolist())
    print(f"\nSearch finished. Final reported score = {final_score:.6f}")

    # Return a plain list (already on the simplex)
    return best_seq.tolist()
