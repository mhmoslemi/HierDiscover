"""
propose_candidate – search for a non‑negative step‑function that minimises
evaluate_sequence.

Algorithm (≈ 1000 s total)
-------------------------------------------------

   we use 32768 when the budget allows, otherwise 16384 or 8192).

   The surrogate
        f_β(a) = (1/β)·log Σ_k exp(β·conv_k)
   is minimised by Adam on the soft‑max parametrisation a = softmax(z).
   β is increased exponentially from a modest start (≈ 5) to a very large
   final value (≥ 2·10⁴) so that the surrogate becomes almost the true

# Utility functions

def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full aperiodic convolution using real FFT (fast, no loops)."""
    m = a.size + b.size - 1
    nfft = next_pow2(m)
    A = np.fft.rfft(a, n=nfft)
    B = np.fft.rfft(b, n=nfft)
    C = A * B
    conv = np.fft.irfft(C, n=nfft)[:m]
    return conv


def compute_score(a: np.ndarray) -> float:
    """
    Exact objective for a probability vector (sum = 1):
        2·n·max(a * a)
    """
    n = a.shape[0]
    conv = convolution_fft(a, a)
    return 2.0 * n * float(conv.max())


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection onto the probability simplex (sum = 1, v ≥ 0).
    """
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / v.shape[0])

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, v.shape[0] + 1)
    cond = u - cssv / ind > 0
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def softmax(z: np.ndarray) -> np.ndarray:
    """Numerically stable soft‑max."""
    z_max = np.max(z)
    e = np.exp(z - z_max)
    return e / e.sum()


# ----------------------------------------------------------------------
# Analytic families (all return a probability vector, i.e. sum = 1)
# ----------------------------------------------------------------------


def symmetric_product_family(n: int, alpha: float) -> np.ndarray:
    """a_i ∝ [(i+½)(n‑½‑i)]^(‑α) , symmetric, normalised to sum = 1."""
    i = np.arange(n, dtype=np.float64)
    logprod = np.log(i + 0.5) + np.log(n - 0.5 - i)
    w = np.exp(-alpha * logprod)
    w[np.isinf(w)] = np.finfo(np.float64).max
    a = w / w.sum()
    return a


def left_heavy_power_family(n: int, beta: float) -> np.ndarray:
    """a_i ∝ (i+½)^(‑β) , monotone decreasing, sum = 1."""
    i = np.arange(n, dtype=np.float64)
    w = np.exp(-beta * np.log(i + 0.5))
    a = w / w.sum()
    return a


def exponential_decay_family(n: int, gamma: float) -> np.ndarray:
    """a_i ∝ exp(-γ·i) , monotone decreasing, sum = 1."""
    i = np.arange(n, dtype=np.float64)
    w = np.exp(-gamma * i)
    a = w / w.sum()
    return a


def beta_distribution_family(n: int, a: float, b: float) -> np.ndarray:
    """
    Discretised Beta(a,b) density on the grid i/(n‑1),
    normalised to sum = 1.  a,b>0 (values <1 give U‑shaped densities).
    """
    t = np.linspace(0.0, 1.0, n, dtype=np.float64)
    # Beta pdf (ignoring normalisation constant)
    w = np.power(t, a - 1.0) * np.power(1.0 - t, b - 1.0)
    # Handle endpoints when a or b < 1 (they become inf)
    w[np.isinf(w)] = np.finfo(np.float64).max
    a_vec = w / w.sum()
    return a_vec


# ----------------------------------------------------------------------
# Random initialisers
# ----------------------------------------------------------------------


def random_dirichlet_start(n: int, conc: float) -> np.ndarray:
    """Symmetric Dirichlet draw with concentration parameter conc."""
    vec = np.random.gamma(conc, 1.0, size=n)
    return vec / vec.sum()


def random_sparse_start(n: int, k: int) -> np.ndarray:
    """
    Uniform mass on exactly k randomly chosen indices (k ≥ 2).
    The resulting vector is already on the simplex.
    """
    idx = np.random.choice(n, size=k, replace=False)
    a = np.zeros(n, dtype=np.float64)
    a[idx] = 1.0 / k
    return a


# ----------------------------------------------------------------------
# Smoothed optimisation (log‑sum‑exp) via Adam on the soft‑max layer
# ----------------------------------------------------------------------


def adam_smooth_optimize(
    start_a: np.ndarray,
    deadline: float,
    max_iters: int = 250_000,
    beta_start: float = 5.0,
    beta_end: float = 20_000.0,
    lr: float = 0.02,
) -> tuple[np.ndarray, float]:
    """
    Minimise the smoothed surrogate
          f_β(a) = (1/β)·log Σ_k exp(β·(a*a)_k)
    using Adam on the soft‑max parametrisation a = softmax(z).
    β is increased exponentially from beta_start to beta_end.
    """
    eps = 1e-12
    z = np.log(start_a + eps)                     # initialise from log‑softmax
    m = np.zeros_like(z)
    v = np.zeros_like(z)

    beta1 = 0.9
    beta2 = 0.999
    eps_adam = 1e-12

    best_a = softmax(z)
    best_score = compute_score(best_a)

    for it in range(1, max_iters + 1):
        if time.time() >= deadline:
            break

        # ---- β schedule – exponential annealing ----
        t = it / max_iters
        beta = beta_start * (beta_end / beta_start) ** t

        # ---- current a ----
        a = softmax(z)

        # ---- smoothed weighting over shifts ----
        conv = convolution_fft(a, a)               # length 2n‑1
        maxc = conv.max()
        # stabilised exponentials
        exp_w = np.exp(beta * (conv - maxc))
        w = exp_w / exp_w.sum()                    # soft‑max over shifts

        # ---- gradient of surrogate w.r.t. a ----
        a_rev = a[::-1]
        # conv of reversed a with w gives the required gradient vector
        conv_rev = convolution_fft(a_rev, w)       # length 3n‑2
        grad_a = 2.0 * conv_rev[a.shape[0] - 1 : a.shape[0] - 1 + a.shape[0]]

        # ---- back‑prop through soft‑max ----
        dot_ag = np.dot(a, grad_a)
        grad_z = a * grad_a - dot_ag * a

        # ---- Adam update ----
        m = beta1 * m + (1.0 - beta1) * grad_z
        v = beta2 * v + (1.0 - beta2) * (grad_z ** 2)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        z -= lr * m_hat / (np.sqrt(v_hat) + eps_adam)

        # ---- occasional exact evaluation ----
        if it % 400 == 0:
            a_tmp = softmax(z)
            sc = compute_score(a_tmp)
            if sc < best_score:
                best_score = sc
                best_a = a_tmp.copy()
                print(f"[{time.time():.1f}]   Adam iter {it:7d} – new best = {best_score:.6f}")

    # final check
    a_fin = softmax(z)
    sc_fin = compute_score(a_fin)
    if sc_fin < best_score:
        best_score = sc_fin
        best_a = a_fin.copy()
    return best_a, best_score


# ----------------------------------------------------------------------
# Mirror (entropy) sub‑gradient descent on the true max‑objective
# ----------------------------------------------------------------------


def mirror_subgrad_optimize(
    start_a: np.ndarray,
    deadline: float,
    step0: float = 0.9,
    check_interval: int = 2000,
) -> tuple[np.ndarray, float]:
    """
    Entropy‑mirror (multiplicative) sub‑gradient descent for
    2·n·max(a*a).  Step size η_t = step0 / sqrt(t).
    """
    n = start_a.shape[0]
    a = start_a.astype(np.float64).copy()
    best_a = a.copy()
    best_score = compute_score(a)

    t = 0
    while time.time() < deadline:
        t += 1
        conv = convolution_fft(a, a)
        k = int(np.argmax(conv))                     # active shift index

        # sub‑gradient for the active shift (ordered pairs)
        idx = np.arange(n)
        j = k - idx
        mask = (j >= 0) & (j < n)
        g = np.zeros_like(a)
        g[mask] = 2.0 * a[j[mask]]

        eta = step0 / np.sqrt(t)
        a = a * np.exp(-eta * g)                     # multiplicative update
        s = a.sum()
        if s == 0.0:
            a = np.full_like(a, 1.0 / n)
        else:
            a = a / s

        if t % check_interval == 0:
            sc = compute_score(a)
            if sc < best_score:
                best_score = sc
                best_a = a.copy()
                print(f"[{time.time():.1f}]   Mirror step {t:7d} – new best = {best_score:.6f}")

    # final evaluation
    sc = compute_score(a)
    if sc < best_score:
        best_score = sc
        best_a = a.copy()
    return best_a, best_score


# ----------------------------------------------------------------------
# Simulated‑annealing hill‑climb (tiny temperature)
# ----------------------------------------------------------------------


def hill_climb_sa(
    start_a: np.ndarray,
    deadline: float,
    max_steps: int = 1_200_000,
    init_temp_factor: float = 1e-4,
) -> tuple[np.ndarray, float]:
    """
    Very low‑temperature SA on the simplex.
    - ½ probability: swap two entries
    - ½ probability: transfer a fraction of mass from i → j
    """
    n = start_a.size
    cur_a = start_a.copy()
    cur_score = compute_score(cur_a)

    best_a = cur_a.copy()
    best_score = cur_score

    T0 = best_score * init_temp_factor
    T = T0
    step_scale = 0.12

    for step in range(1, max_steps + 1):
        if time.time() >= deadline:
            break

        cand = cur_a.copy()
        if np.random.rand() < 0.5:                       # swap two entries
            i, j = np.random.randint(n, size=2)
            cand[i], cand[j] = cand[j], cand[i]
        else:                                            # mass transfer
            i, j = np.random.randint(n, size=2)
            if i != j:
                delta = step_scale * cur_a[i]
                cand[i] -= delta
                cand[j] += delta
                if cand[i] < 0.0:
                    cand[i] = 0.0

        # re‑normalise (simple projection onto simplex)
        cand = cand / cand.sum()
        cand_score = compute_score(cand)
        delta = cand_score - cur_score

        # Metropolis acceptance
        if delta < 0.0 or np.random.rand() < np.exp(-delta / (T + 1e-18)):
            cur_a = cand
            cur_score = cand_score
            if cur_score < best_score:
                best_score = cur_score
                best_a = cur_a.copy()
                step_scale = min(step_scale * 1.07, 0.5)
        else:
            step_scale *= 0.97

        # linear cooling
        T = T0 * (1.0 - step / max_steps)

    return best_a, best_score


# ----------------------------------------------------------------------
# High‑level search driver
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 12345, budget_s: int = 1000, **_):
    """
    Search for a non‑negative step‑function (list of floats) that minimises
    evaluate_sequence.  The algorithm respects the given time budget and
    returns a plain Python list.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 2.0    # small safety margin

    # -------------------------------------------------------------
    # Problem size – larger n gives a richer design space but costs more time.
    # -------------------------------------------------------------
    if budget_s >= 900:
        n = 32768
    elif budget_s >= 600:
        n = 16384
    else:
        n = 8192
    print(f"[{time.time():.1f}] Using length n = {n}")

    # -------------------------------------------------------------
    # Global best bookkeeping
    # -------------------------------------------------------------
    best_seq = None
    best_score = float("inf")

    def maybe_update(vec: np.ndarray, tag: str = ""):
        """Update global best if ``vec`` improves the score."""
        nonlocal best_seq, best_score
        sc = compute_score(vec)
        if sc < best_score:
            best_score = sc
            best_seq = vec.tolist()
            print(f"[{time.time():.1f}] new best {tag:<20} = {best_score:.6f}")

    # -------------------------------------------------------------
    # 0. ‑‑ Use fallback if a known construction is available
    # -------------------------------------------------------------
    try:
        fb_vec = [float(x) for x in height_sequence_1]   # type: ignore
        fb_sc = evaluate_sequence(fb_vec)
        print(f"[{time.time():.1f}] fallback score = {fb_sc:.6f}")
        if fb_sc < best_score:
            best_score = fb_sc
            best_seq = fb_vec
    except Exception:
        pass

    # -------------------------------------------------------------
    # 1. Fast exhaustive scan of analytic families + simple random starts
    # -------------------------------------------------------------
    uniform = np.full(n, 1.0 / n)
    maybe_update(uniform, "uniform")

    # -- symmetric product family (α extended up to 12) --
    for alpha in np.linspace(0.0, 12.0, 61):
        if time.time() >= deadline - 30.0:
            break
        vec = symmetric_product_family(n, alpha)
        maybe_update(vec, f"sym‑α={alpha:.2f}")

    # -- left‑heavy power family (β up to 12) --
    for beta in np.linspace(0.0, 12.0, 61):
        if time.time() >= deadline - 30.0:
            break
        vec = left_heavy_power_family(n, beta)
        maybe_update(vec, f"pow‑β={beta:.2f}")

    # -- exponential decay family (γ up to 12) --
    for gamma in np.linspace(0.0, 12.0, 61):
        if time.time() >= deadline - 30.0:
            break
        vec = exponential_decay_family(n, gamma)
        maybe_update(vec, f"exp‑γ={gamma:.2f}")

    # -- beta‑distribution family (α,β in a modest grid) --
    a_vals = np.array([0.2, 0.5, 1.0, 2.0, 4.0])
    b_vals = np.array([0.2, 0.5, 1.0, 2.0, 4.0])
    for a_par in a_vals:
        for b_par in b_vals:
            if time.time() >= deadline - 30.0:
                break
            vec = beta_distribution_family(n, a_par, b_par)
            maybe_update(vec, f"beta‑a={a_par:.2f},b={b_par:.2f}")

    # -- random Dirichlet draws (diverse concentration) --
    conc_vals = np.concatenate((
        np.logspace(-2, -0.5, 12),   # very sparse
        np.logspace(-0.5, 0.5, 12),  # medium
        np.logspace(0.5, 1.0, 12)    # near‑uniform
    ))
    for conc in conc_vals:
        for _ in range(2):
            if time.time() >= deadline - 30.0:
                break
            vec = random_dirichlet_start(n, conc)
            maybe_update(vec, f"dir‑α={conc:.3f}")

    # -- random sparse masks with different sparsities --
    for k in (int(n * 0.02), int(n * 0.04), int(n * 0.08), int(n * 0.12)):
        if k < 2:
            continue
        for _ in range(2):
            if time.time() >= deadline - 30.0:
                break
            vec = random_sparse_start(n, k)
            maybe_update(vec, f"sparse‑k={k}")

    # -------------------------------------------------------------
    # 2.  High‑β Adam refinement – run as many restarts as time permits.
    # -------------------------------------------------------------
    while time.time() < deadline - 250.0:
        start_vec = random_sparse_start(n, int(n * 0.03))  # modest sparsity
        a_opt, sc_opt = adam_smooth_optimize(
            start_vec,
            deadline - 200.0,
            max_iters=300_000,
            beta_start=5.0,
            beta_end=30_000.0,
            lr=0.018,
        )
        if sc_opt < best_score:
            best_score = sc_opt
            best_seq = a_opt.tolist()
            print(f"[{time.time():.1f}]   new best after Adam‑smooth = {best_score:.6f}")

    # -------------------------------------------------------------
    # 3.  Intensive high‑β Adam on the current best candidate.
    # -------------------------------------------------------------
    if best_seq is not None and time.time() < deadline - 150.0:
        print(f"[{time.time():.1f}]   intensive high‑β Adam on best candidate")
        a_ref, sc_ref = adam_smooth_optimize(
            np.array(best_seq, dtype=np.float64),
            deadline - 130.0,
            max_iters=200_000,
            beta_start=30_000.0,
            beta_end=100_000.0,
            lr=0.010,
        )
        if sc_ref < best_score:
            best_score = sc_ref
            best_seq = a_ref.tolist()
            print(f"[{time.time():.1f}]   after intensive Adam = {best_score:.6f}")

    # -------------------------------------------------------------
    # 4.  Mirror‑subgradient fine‑tuning
    # -------------------------------------------------------------
    if best_seq is not None and time.time() < deadline - 70.0:
        print(f"[{time.time():.1f}]   mirror sub‑gradient finetune")
        a_sub, sc_sub = mirror_subgrad_optimize(
            np.array(best_seq, dtype=np.float64),
            deadline - 20.0,
            step0=0.9,
        )
        if sc_sub < best_score:
            best_score = sc_sub
            best_seq = a_sub.tolist()
            print(f"[{time.time():.1f}]   after mirror sub‑grad best = {best_score:.6f}")

    # -------------------------------------------------------------
    # 5.  Symmetrisation (many optimal solutions are symmetric)
    # -------------------------------------------------------------
    if best_seq is not None:
        a_arr = np.array(best_seq, dtype=np.float64)
        a_sym = (a_arr + a_arr[::-1]) * 0.5
        a_sym = project_to_simplex(a_sym)
        maybe_update(a_sym, "symmetrised")

    # -------------------------------------------------------------
    # 6.  Final tiny simulated‑annealing hill‑climb
    # -------------------------------------------------------------
    if best_seq is not None and time.time() < deadline - 5.0:
        print(f"[{time.time():.1f}]   final hill‑climb (SA)")
        a_final, sc_final = hill_climb_sa(
            np.array(best_seq, dtype=np.float64),
            deadline,
            max_steps=1_200_000,
        )
        if sc_final < best_score:
            best_score = sc_final
            best_seq = a_final.tolist()
            print(f"[{time.time():.1f}]   hill‑climb improved best = {best_score:.6f}")

    # -------------------------------------------------------------
    # 7.  Safeguard – fall back to a modest uniform vector if everything failed
    # -------------------------------------------------------------
    if best_seq is None:
        n_small = 256
        best_seq = [1.0 / n_small] * n_small
        best_score = evaluate_sequence(best_seq)

    # -------------------------------------------------------------
    # 8.  Final official evaluation (the external harness will also call it)
    # -------------------------------------------------------------
    final_score = evaluate_sequence(best_seq)
    print(f"\n[{time.time():.1f}] finished search – final reported score = {final_score:.6f}")

    return best_seq
