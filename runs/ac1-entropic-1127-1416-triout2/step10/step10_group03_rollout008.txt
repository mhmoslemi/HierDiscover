def softmax_optimize(a_init, start_score, deadline, beta_start=5, beta_end=5000, lr=0.05, eps=1e-8):
    # a_init: array sum=1
    # We'll maintain logits x (log a_i) (or logit = log(a_i+1e-12) perhaps)
    # Actually proper simple: set x = np.log(a_init + 1e-12) initial.
    # Use Adam optimizer.
    m = np.zeros_like(x); v = np.zeros_like(x); t = 0
    best_a = a_init.copy()
    best_score = start_score
    while time.time() < deadline:
        # compute conv
        conv = convolution_fft(a, a)   # a = softmax(x)
        max_conv = conv.max()
        # Evaluation for tracking
        cur_score = 2.0 * n * max_conv
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
        # compute beta schedule
        elapsed = time.time() - start_time_global
        total = deadline - start_time_global
        fraction = elapsed / total
        beta = beta_start + fraction * (beta_end - beta_start)
        # softmax weights
        shifted = conv - max_conv
        exp_beta = np.exp(beta * shifted)
        w = exp_beta / exp_beta.sum()
        # gradient wrt a
        grad_a = 2.0 * beta * np.correlate(w, a, mode='valid')
        # gradient wrt logits x
        # compute dot = sum_j a_j * grad_a_j
        dot = np.dot(a, grad_a)
        grad_x = a * (grad_a - dot)
        # Adam update
        t += 1
        m = 0.9 * m + 0.1 * grad_x
        v = 0.999 * v + 0.001 * (grad_x**2)
        m_hat = m / (1 - 0.9**t)
        v_hat = v / (1 - 0.999**t)
        # step size annealed maybe as lr / (np.sqrt(t))
        step = lr / (np.sqrt(t))
        x = x - step * m_hat / (np.sqrt(v_hat) + eps)
        a = np.exp(x)
        a = a / a.sum()
        # possibly also occasional checks and prints
    return best_a, best_score

def adam_optimize(a_init, start_score, start_time, deadline, beta_start=5, beta_end=20000, base_lr=0.05):
    a = a_init.copy()
    # Use logits x = np.log(a + 1e-16)
    x = np.log(a + 1e-16)
    m = np.zeros_like(x)
    v = np.zeros_like(x)
    best_a = a.copy()
    best_score = start_score
    t = 0
    while time.time() < deadline:
        # compute conv of a
        conv = convolution_fft(a, a)
        max_conv = conv.max()
        cur_score = 2.0 * n * float(max_conv)
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
        # schedule temp
        elapsed = time.time() - start_time
        fraction = elapsed / (deadline - start_time)
        beta = beta_start + fraction * (beta_end - beta_start)
        # compute softmax weights
        shifted = conv - max_conv
        # For numeric stability, clip maybe:
        exp_vals = np.exp(beta * shifted)
        w = exp_vals / exp_vals.sum()
        # compute gradient wrt a using FFT correlation
        w_rev = w[::-1]
        conv_grad_full = convolution_fft(a, w_rev)
        start_idx = len(w) - 1  # =2n-2
        grad_a = 2.0 * beta * conv_grad_full[start_idx:start_idx + n]
        # compute gradient wrt logits
        dot = np.dot(a, grad_a)
        grad_x = a * (grad_a - dot)
        # Adam updates
        t += 1
        m = 0.9 * m + 0.1 * grad_x
        v = 0.999 * v + 0.001 * (grad_x**2)
        m_hat = m / (1 - 0.9**t)
        v_hat = v / (1 - 0.999**t)
        # step size decays maybe with sqrt(t) but also base lr.
        step_lr = base_lr / np.sqrt(t)
        x = x - step_lr * m_hat / (np.sqrt(v_hat) + 1e-8)
        # recompute a
        a = np.exp(x)
        a = a / a.sum()
        # optional print progress occasionally
        if t % 5000 == 0:
            print(f"[Adam] it {t} time {elapsed:.1f}s best {best_score:.6f} beta {beta:.1f}")
    return best_a, best_score

   random Dirichlet draws.  The best seed (according to the exact
   objective) is kept.

2. **Adam‑softmax optimisation** – minimise a smooth approximation of the
   max‑norm

        Lβ(a) = log Σ exp(β·conv(a,a))

   using the Adam optimiser in the *log‑probability* (soft‑max) space.
   The temperature β is annealed from a small value to a large one
   (up to 2·10⁴).  Gradients of Lβ are computed via FFT‑based
   correlation, which makes each iteration O(n log n).

3. **Projected sub‑gradient refinement** – a classic sub‑gradient
   descent on the true max‑objective, starting from the best iterate of
   phase 2.

4. **Stochastic hill‑climb polishing** – cheap random moves that are
   kept only when they strictly improve the exact score.

The whole search respects the 1000 s wall‑clock budget; a small safety
margin is kept for the external harness.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Low‑level utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Return the smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full linear convolution using a real FFT (real inputs)."""
    m = a.shape[0] + b.shape[0] - 1
    n_fft = next_pow2(m)
    A = np.fft.rfft(a, n=n_fft)
    B = np.fft.rfft(b, n=n_fft)
    C = A * B
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


def compute_score(a: np.ndarray) -> float:
    """
    Exact objective for a probability vector (sum = 1):
        2·n·max(conv(a,a))
    """
    n = a.shape[0]
    conv = convolution_fft(a, a)
    return 2.0 * n * float(conv.max())


# ----------------------------------------------------------------------
# Seed generators
# ----------------------------------------------------------------------


def generate_sin_power(n: int, power: float = 2.0) -> np.ndarray:
    """a_i ∝ sin^power(π·i/(n+1))."""
    i = np.arange(1, n + 1, dtype=float)
    a = np.sin(np.pi * i / (n + 1.0)) ** power
    a /= a.sum()
    return a


def generate_beta_family(n: int, p: float) -> np.ndarray:
    """
    Symmetric polynomial shape a_i ∝ ((i+1)*(n-i))**p.
    Positive p concentrates mass in the centre, negative p pushes it to the edges.
    """
    i = np.arange(n, dtype=float)
    w = ((i + 1.0) * (n - i)) ** p
    w[w < 0] = 0.0
    a = w / w.sum()
    return a


def random_dirichlet(n: int, conc: float = 1.0) -> np.ndarray:
    """Draw a random probability vector from Dirichlet(conc)."""
    if conc == 1.0:
        a = np.random.rand(n)
    else:
        a = np.random.gamma(conc, 1.0, size=n)
    a /= a.sum()
    return a


def random_spike(n: int, k: int = 80) -> np.ndarray:
    """
    Very sparse vector: choose `k` distinct positions, assign random weights,
    then normalise.
    """
    idx = np.random.choice(n, size=min(k, n), replace=False)
    w = np.random.rand(len(idx))
    a = np.zeros(n, dtype=float)
    a[idx] = w
    a /= a.sum()
    return a


# ----------------------------------------------------------------------
# Phase 2 – Adam optimisation of the smooth max surrogate
# ----------------------------------------------------------------------


def adam_softmax_optimize(
    a0: np.ndarray,
    start_score: float,
    start_time: float,
    deadline: float,
    beta_start: float = 5.0,
    beta_end: float = 20000.0,
    base_lr: float = 0.12,
) -> tuple[np.ndarray, float]:
    """
    Adam optimisation in the log‑softmax (probability simplex) space.
    Minimises Lβ(a)=log Σ exp(β·conv(a,a)).  The temperature β is
    linearly annealed from `beta_start` to `beta_end`.
    """
    n = a0.shape[0]
    # initialise logits (log‑probabilities); add a tiny epsilon to avoid log(0)
    x = np.log(a0 + 1e-16)
    m = np.zeros_like(x)
    v = np.zeros_like(x)

    best_a = np.exp(x)
    best_a /= best_a.sum()
    best_score = start_score

    it = 0
    while time.time() < deadline:
        # ----- reconstruction of probability vector -----
        a = np.exp(x)
        a /= a.sum()

        # ----- exact objective for bookkeeping -----
        conv = convolution_fft(a, a)
        max_conv = conv.max()
        cur_score = 2.0 * n * float(max_conv)

        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()

        # ----- schedule temperature β -----
        elapsed = time.time() - start_time
        total = deadline - start_time
        frac = elapsed / total
        beta = beta_start + frac * (beta_end - beta_start)

        # ----- soft‑max weights -----
        shifted = conv - max_conv               # max becomes 0
        exp_vals = np.exp(beta * shifted)       # safe: values ≤ 1
        w = exp_vals / exp_vals.sum()           # shape (2n‑1,)

        # ----- gradient w.r.t. a (via FFT‑based correlation) -----
        # grad_a = 2·β·correlate(w, a)   (valid mode → length n)
        w_rev = w[::-1]
        grad_full = convolution_fft(a, w_rev)          # length 3n‑2
        start_idx = len(w) - 1                         # = 2n‑2
        grad_a = 2.0 * beta * grad_full[start_idx:start_idx + n]

        # ----- gradient w.r.t. logits (soft‑max Jacobian) -----
        dot = np.dot(a, grad_a)
        grad_x = a * (grad_a - dot)

        # ----- Adam update (step size decays as 1/√t) -----
        it += 1
        m = 0.9 * m + 0.1 * grad_x
        v = 0.999 * v + 0.001 * (grad_x ** 2)
        m_hat = m / (1.0 - 0.9 ** it)
        v_hat = v / (1.0 - 0.999 ** it)

        step = base_lr / np.sqrt(it)
        x = x - step * m_hat / (np.sqrt(v_hat) + 1e-8)

        # occasional progress report
        if it % 6000 == 0:
            print(
                f"[Adam] it {it:6d} | time {elapsed:6.1f}s | best {best_score:.6f} | β {beta:.1f}"
            )

    # final best vector (already normalised)
    return best_a, best_score


# ----------------------------------------------------------------------
# Phase 3 – Projected sub‑gradient refinement (unchanged)
# ----------------------------------------------------------------------


def subgradient_refine(
    a: np.ndarray,
    start_score: float,
    deadline: float,
    max_iter: int = 500_000,
    step0: float = 0.07,
) -> tuple[np.ndarray, float]:
    """Projected sub‑gradient descent on the true max‑objective."""
    n = a.shape[0]
    cur_a = a.copy()
    cur_score = start_score
    best_a = cur_a.copy()
    best_score = cur_score

    for t in range(1, max_iter + 1):
        if time.time() >= deadline:
            break

        conv = convolution_fft(cur_a, cur_a)
        k = int(np.argmax(conv))

        idx = np.arange(n)
        j = k - idx
        mask = (j >= 0) & (j < n)
        g = np.zeros_like(cur_a)
        g[mask] = 2.0 * cur_a[j[mask]]

        step = step0 / np.sqrt(t)
        cur_a -= step * g
        # Euclidean projection onto the simplex
        cur_a = project_to_simplex(cur_a)

        cur_score = compute_score(cur_a)
        if cur_score < best_score:
            best_score = cur_score
            best_a = cur_a.copy()

        if t % 1000 == 0:
            # small jitter to escape flat regions
            jitter = np.random.randn(n) * (step0 * 0.004 / np.sqrt(t))
            cur_a += jitter
            cur_a = project_to_simplex(cur_a)

    return best_a, best_score


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of `v` onto the probability simplex
    {x | x_i ≥ 0, Σ x_i = 1}.
    """
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / v.shape[0])

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, v.shape[0] + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.where(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


# ----------------------------------------------------------------------
# Phase 4 – Greedy hill‑climb polishing (unchanged)
# ----------------------------------------------------------------------


def hill_climb(
    a: np.ndarray,
    deadline: float,
    max_steps: int = 1_200_000,
) -> tuple[np.ndarray, float]:
    """Stochastic moves that are kept only when they improve the exact score."""
    n = a.shape[0]
    best_a = a.copy()
    best_score = compute_score(best_a)
    step_scale = 0.08

    steps = 0
    while steps < max_steps and time.time() < deadline:
        cand = best_a.copy()
        if np.random.rand() < 0.5:
            i, j = np.random.randint(n, size=2)
            cand[i], cand[j] = cand[j], cand[i]
        else:
            i, j = np.random.randint(n, size=2)
            if cand[i] > 0.0 and i != j:
                delta = step_scale * cand[i]
                cand[i] -= delta
                cand[j] += delta
                if cand[i] < 0.0:
                    cand[i] = 0.0

        cand /= cand.sum()
        cur_score = compute_score(cand)

        if cur_score < best_score:
            best_score = cur_score
            best_a = cand
            step_scale = min(step_scale * 1.07, 0.5)
        else:
            step_scale *= 0.96

        steps += 1

    return best_a, best_score


# ----------------------------------------------------------------------
# Main entry point required by the harness
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 314159, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step‑function that minimises
    `evaluate_sequence`.  Returns a plain Python list of floats.
    """
    np.random.seed(seed)
    start_time = time.time()
    # leave a small safety margin for the surrounding harness
    deadline = start_time + budget_s - 1.0
    total_time = deadline - start_time

    # ------------------------------------------------------------------
    # 0. Optional fallback from a previously discovered sequence
    # ------------------------------------------------------------------
    best_seq = None
    best_val = float("inf")
    try:
        fallback = [float(x) for x in height_sequence_1]  # type: ignore
        fb = evaluate_sequence(fallback)
        print(f"Fallback score = {fb:.6f}")
        if fb < best_val:
            best_val = fb
            best_seq = fallback
    except Exception:
        pass

    # ------------------------------------------------------------------
    # 1. Choose a length (2048 is a good compromise; use 4096 if time permits)
    # ------------------------------------------------------------------
    n = 2048
    if total_time > 850.0:
        n = 4096
    print(f"Using sequence length = {n}")

    # ------------------------------------------------------------------
    # 2. Generate a diverse set of seeds and keep the best one (exact score)
    # ------------------------------------------------------------------
    seeds = []

    # deterministic families
    seeds.append(generate_sin_power(n, power=2.0))
    for p in (-3.0, -2.0, -1.0, -0.5, 0.0, 0.5, 1.0, 2.0, 3.0):
        seeds.append(generate_beta_family(n, p))

    # random Dirichlet draws (various concentrations)
    for conc in (0.2, 0.5, 1.0, 2.0, 5.0):
        for _ in range(2):
            seeds.append(random_dirichlet(n, conc=conc))

    # sparse spikes – good for flattening the auto‑correlation
    for k in (30, 60, 120, 200):
        seeds.append(random_spike(n, k=k))

    # include fallback if it matches the current length
    if best_seq is not None and len(best_seq) == n:
        seeds.append(np.array(best_seq, dtype=float) / np.sum(best_seq))

    # evaluate seeds (exact objective) and keep the best
    for idx, s in enumerate(seeds):
        s = np.asarray(s, dtype=float)
        s = project_to_simplex(s)  # guard against tiny drift
        score = compute_score(s)
        if score < best_val:
            best_val = score
            best_seq = s.tolist()
            print(f"[Seed {idx}] improvement: score = {best_val:.6f}")

    cur_a = np.asarray(best_seq, dtype=float)
    cur_a /= cur_a.sum()
    cur_score = compute_score(cur_a)

    # ------------------------------------------------------------------
    # 3. Adam‑softmax optimisation (large portion of the budget)
    # ------------------------------------------------------------------
    phase1_deadline = start_time + total_time * 0.63
    print("\n[Phase 1] Adam soft‑max optimisation")
    cur_a, cur_score = adam_softmax_optimize(
        cur_a,
        cur_score,
        start_time,
        phase1_deadline,
        beta_start=5.0,
        beta_end=20000.0,
        base_lr=0.12,
    )
    if cur_score < best_val:
        best_val = cur_score
    print(f"After Adam: best score = {best_val:.6f}")

    # ------------------------------------------------------------------
    # 4. Sub‑gradient refinement (as before)
    # ------------------------------------------------------------------
    if time.time() < deadline - 30.0:
        print("\n[Phase 2] Sub‑gradient refinement")
        phase2_deadline = deadline - 30.0
        cur_a, cur_score = subgradient_refine(
            cur_a, cur_score, phase2_deadline, max_iter=600_000, step0=0.07
        )
        if cur_score < best_val:
            best_val = cur_score
            print(f"After sub‑grad: best score = {best_val:.6f}")

    # ------------------------------------------------------------------
    # 5. Final hill‑climb polishing
    # ------------------------------------------------------------------
    if time.time() < deadline - 1.0:
        print("\n[Phase 3] Hill‑climb polishing")
        cur_a, cur_score = hill_climb(cur_a, deadline, max_steps=1_300_000)
        if cur_score < best_val:
            best_val = cur_score
            print(f"After polishing: best score = {best_val:.6f}")

    # ------------------------------------------------------------------
    # 6. Final conversion and official evaluation
    # ------------------------------------------------------------------
    final_seq = cur_a.tolist()
    final_val = evaluate_sequence(final_seq)
    print(f"\nFinished search. Final reported score = {final_val:.6f}")

    return final_seq
