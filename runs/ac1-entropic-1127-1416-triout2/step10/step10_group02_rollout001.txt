def smooth_max_and_grad(a, alpha):
    # a: np.ndarray shape (n,)
    conv = full_convolution(a)  # length m=2n-1
    # compute log-sum-exp and gradient.
    # For stability: subtract max.
    maxc = conv.max()
    # compute exp(alpha * (conv - maxc)) to avoid overflow
    exp_vals = np.exp(alpha * (conv - maxc))
    sum_exp = exp_vals.sum()
    f = maxc + np.log(sum_exp) / alpha  # since (1/alpha) * log(sum exp(alpha*(conv-maxc))) + maxc
    # weight w_k = exp(alpha*(conv[k] - maxc)) / sum_exp
    w = exp_vals / sum_exp
    # g = 2 * correlation(a, w)
    g = correlation_grad(a, w)  # function defined
    return f, g
```

Now implement correlation_grad:

```
def correlation_grad(a, w):
    n = a.shape[0]
    L = w.shape[0] # 2n-1
    # compute correlation of a with w (full) using FFT.
    n_fft = 1 << ((n + L - 1 - 1).bit_length())
    A = np.fft.rfft(a, n=n_fft)
    W = np.fft.rfft(w, n=n_fft)
    # correlation: irfft(A * conj(W))
    corr_full = np.fft.irfft(A * np.conj(W), n=n_fft)[:n+L-1]
    # take indices L-1 to L-1 + n -1 inclusive for g_i.
    start = L-1
    end = start + n
    g = 2 * corr_full[start:end]
    return g
```

Check: A shape: length n_fft. W shape: length n_fft. conj(W) needed for correlation (since convolution uses multiplication, correlation uses conj of second). Indeed correlation via FFT: corr = ifft( FFT(a) * conj(FFT(b)) ). This is correct.

def smooth_gradient_descent(a0, deadline, alpha_schedule, steps_per_alpha):
    a = a0.copy()
    best_a = a.copy()
    best_val = compute_score(a)
    t_total = 0
    for alpha, steps in zip(alpha_schedule, steps_per_alpha):
        for s in range(steps):
            if time.time() >= deadline:
                return best_a
            f, g = smooth_max_and_grad(a, alpha)
            # step size schedule: maybe decreasing with t_total.
            # We could use step = step0 / np.sqrt(t_total + 1)
            step = 0.05 / np.sqrt(t_total + 1)  # chosen step0 0.05
            a = a - step * g
            a = project_to_simplex(a)
            # evaluate exact objective
            val = compute_score(a)
            if val < best_val:
                best_val = val
                best_a = a.copy()
            t_total += 1
    return best_a
```

"""
Non‑negative step‑function optimiser for ``evaluate_sequence``.

Algorithm
---------
1. **Fallback** – use a previously discovered good sequence if available.
2. **Power‑law sweep** – cheap global scan of a_i ∝ (i+1)^(-p) to obtain a
   decent initial candidate.
3. **Multi‑start smooth‑max gradient descent** – for each random start we
   minimise a smooth approximation of the ‑max‑convolution objective:
        f_α(a) = (1/α)·log∑_k exp(α·(a∗a)_k)
   Its gradient is a correlation of *a* with the soft‑max weights.
   The schedule of α grows from 1 to a few hundred, allowing the optimiser
   to first smooth the landscape and then focus on the true max.
4. **Projected sub‑gradient polish** – a short run of the classic
   sub‑gradient method (active‑index) to fine‑tune the best candidate.
5. **Greedy hill‑climb** – cheap stochastic mass‑transfer moves on the
   simplex, always keeping the best sequence found.

All phases respect the overall wall‑clock budget (default 1000 s) and the
final result is projected onto the probability simplex (∑a_i = 1, a_i≥0),
which makes ``evaluate_sequence`` equivalent to the exact objective
2·n·max(conv(a,a)).
"""

import time
import numpy as np

# ------------------------------------------------------------
# Numerical utilities
# ------------------------------------------------------------

def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    # Based on algorithm from
    # https://arxiv.org/pdf/1101.6081.pdf (section 3)
    n = v.shape[0]
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / n)

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    mask = u - cssv / ind > 0
    rho = np.where(mask)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def full_convolution(a: np.ndarray) -> np.ndarray:
    """
    Compute the full linear convolution a*a using a real FFT.
    Length of the result is 2*len(a)-1.
    """
    n = a.shape[0]
    m = 2 * n - 1
    n_fft = 1 << ((m - 1).bit_length())          # next power of two ≥ m
    A = np.fft.rfft(a, n=n_fft)
    C = A * A
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


def compute_score(a: np.ndarray) -> float:
    """
    Exact objective for a normalised vector (sum = 1):
        2·n·max(conv(a,a))
    """
    n = a.shape[0]
    conv = full_convolution(a)
    return float(2.0 * n * conv.max())


def correlation_grad(a: np.ndarray, w: np.ndarray) -> np.ndarray:
    """
    Compute the gradient of the smooth max (soft‑max) objective.
    g = 2 * correlation(a, w) where w are the soft‑max weights.
    """
    n = a.shape[0]
    L = w.shape[0]                     # should be 2n‑1
    n_fft = 1 << ((n + L - 1 - 1).bit_length())
    A_fft = np.fft.rfft(a, n=n_fft)
    W_fft = np.fft.rfft(w, n=n_fft)
    # correlation via FFT: irfft(A * conj(W))
    corr_full = np.fft.irfft(A_fft * np.conj(W_fft), n=n_fft)[:n + L - 1]
    start = L - 1
    g = 2.0 * corr_full[start:start + n]
    return g


def smooth_max_and_grad(a: np.ndarray, alpha: float):
    """
    Returns a smooth approximation of max(conv(a,a)) and its gradient.
    f_α(a) = (1/α)·log∑_k exp(α·conv_k)
    Gradient = 2·correlation(a, softmax_weights)
    """
    conv = full_convolution(a)                     # length 2n‑1
    maxc = conv.max()
    # stability: shift by max before exponentiation
    shifted = conv - maxc
    exp_vals = np.exp(alpha * shifted)
    sum_exp = exp_vals.sum()
    # smooth max value
    f = maxc + np.log(sum_exp) / alpha
    # soft‑max weights (sum to 1)
    w = exp_vals / sum_exp
    # gradient
    g = correlation_grad(a, w)
    return f, g

def best_power_law_vector(n: int, time_deadline: float) -> np.ndarray:
    """Scan p∈[0.5,0.7] (coarse then fine) and return the best normalised vector."""
    best_a = None
    best_val = float("inf")
    best_p = 0.5

    # coarse grid
    for p in np.linspace(0.5, 0.7, 41):   # step 0.005
        if time.time() >= time_deadline:
            break
        a = np.arange(1, n + 1, dtype=float) ** (-p)
        a /= a.sum()
        val = compute_score(a)
        if val < best_val:
            best_val, best_a, best_p = val, a, p

    # fine grid around the best p
    if best_a is not None:
        low = max(0.5, best_p - 0.03)
        high = min(0.8, best_p + 0.03)
        for p in np.linspace(low, high, 31):
            if time.time() >= time_deadline:
                break
            a = np.arange(1, n + 1, dtype=float) ** (-p)
            a /= a.sum()
            val = compute_score(a)
            if val < best_val:
                best_val, best_a, best_p = val, a, p

    return best_a

def smooth_gradient_descent(a0: np.ndarray,
                            deadline: float,
                            alphas,
                            steps_per_alpha,
                            step0: float = 0.3) -> np.ndarray:
    """
    Projected gradient descent on the smooth‑max objective.
    alphas – list of temperature parameters (α).
    steps_per_alpha – list of iteration counts for each α.
    """
    a = a0.copy()
    best_a = a.copy()
    best_val = compute_score(a)
    t = 0

    for alpha, steps in zip(alphas, steps_per_alpha):
        for _ in range(steps):
            if time.time() >= deadline:
                return best_a
            t += 1
            f, g = smooth_max_and_grad(a, alpha)
            # simple diminishing step size
            step = step0 / np.sqrt(t)
            a -= step * g
            a = project_to_simplex(a)

            # evaluate true (non‑smooth) objective occasionally
            if t % 20 == 0:
                val = compute_score(a)
                if val < best_val:
                    best_val = val
                    best_a = a.copy()

    # one final exact evaluation
    final_val = compute_score(a)
    if final_val < best_val:
        best_a = a.copy()
    return best_a


def random_simplex_vector(n: int, concentration: float = 0.5) -> np.ndarray:
    """Draw a Dirichlet‑like random point on the simplex."""
    v = np.random.gamma(concentration, 1.0, size=n)
    v /= v.sum()
    return v


def multi_start_optimisation(n: int,
                             deadline: float,
                             num_restarts: int = 20) -> np.ndarray:
    """
    Perform several random restarts of smooth‑max gradient descent.
    Returns the best vector discovered.
    """
    best_a = None
    best_val = float("inf")
    # schedule for α (start smooth → focus on max)
    alphas = [1.0, 2.0, 5.0, 10.0, 20.0, 40.0, 80.0, 160.0, 320.0]
    # allocate more steps to early (smoother) stages
    steps_per_alpha = [3000, 2500, 2000, 1500, 1200, 800, 600, 400, 300]

    for rs in range(num_restarts):
        if time.time() >= deadline:
            break
        # randomise the seed of the start – either power‑law or Dirichlet
        if np.random.rand() < 0.5:
            p = np.random.uniform(0.45, 0.85)
            a0 = np.arange(1, n + 1, dtype=float) ** (-p)
        else:
            a0 = random_simplex_vector(n, concentration=0.5)
        a0 /= a0.sum()
        # give each restart a slice of the remaining budget
        restart_deadline = min(deadline,
                               time.time() + 30.0)   # ≤30 s per restart
        a_opt = smooth_gradient_descent(a0,
                                         restart_deadline,
                                         alphas,
                                         steps_per_alpha,
                                         step0=0.4)
        val = compute_score(a_opt)
        print(f"[restart {rs+1}/{num_restarts}] best value = {val:.6f}")
        if val < best_val:
            best_val, best_a = val, a_opt.copy()

    return best_a

def subgradient_polish(a0: np.ndarray,
                       deadline: float,
                       step0: float = 0.12) -> np.ndarray:
    """Projected sub‑gradient descent on the true max‑convolution objective."""
    a = a0.copy()
    n = a.shape[0]
    best_a = a.copy()
    best_val = compute_score(a)

    t = 0
    while time.time() < deadline:
        t += 1
        conv = full_convolution(a)
        k = int(np.argmax(conv))

        # sub‑gradient of the active entry (active index k)
        idx = np.arange(n)
        j = k - idx
        mask = (j >= 0) & (j < n)
        g = np.zeros_like(a)
        g[mask] = 2.0 * a[j[mask]]

        step = step0 / np.sqrt(t)
        a -= step * g
        a = project_to_simplex(a)

        val = compute_score(a)
        if val < best_val:
            best_val, best_a = val, a.copy()

        # occasional jitter to escape plateaus
        if t % 1000 == 0:
            jitter = np.random.randn(n) * (step0 * 0.01 / np.sqrt(t))
            a += jitter
            a = project_to_simplex(a)

    return best_a

def hill_climb(a_start: np.ndarray,
               deadline: float,
               max_no_improve: int = 500_000) -> np.ndarray:
    """Small stochastic mass‑transfer moves with greedy acceptance."""
    a = a_start.copy()
    n = a.shape[0]
    best_a = a.copy()
    best_val = compute_score(a)

    step_scale = 0.05
    it = 0
    no_improve = 0

    while time.time() < deadline and it < max_no_improve:
        i, j = np.random.randint(n, size=2)
        if i == j:
            continue
        delta = step_scale * a[i]
        cand = a.copy()
        cand[i] -= delta
        cand[j] += delta
        cand = np.maximum(cand, 0.0)
        cand = project_to_simplex(cand)

        val = compute_score(cand)
        if val < best_val:
            best_val = val
            best_a = cand.copy()
            a = cand
            step_scale = min(step_scale * 1.07, 0.5)
            no_improve = 0
        else:
            step_scale *= 0.985
            no_improve += 1

        it += 1

    return best_a

        if a_pow is not None:
            val = compute_score(a_pow)
            print(f"[power sweep] best p‑score = {val:.6f}")
            if val < best_val:
                best_val, best_seq = val, a_pow.tolist()

    # -----------------------------------------------------------------
    # 2. Multi‑start smooth‑max gradient descent
    # -----------------------------------------------------------------
    ms_deadline = deadline - 150.0
    if time.time() < ms_deadline:
        # number of restarts is chosen adaptively based on remaining time
        remaining = ms_deadline - time.time()
        # roughly up to 20 restarts, but stop earlier if time is scarce
        max_restarts = min(20, int(remaining // 30))
        if max_restarts < 1:
            max_restarts = 1
        a_multi = multi_start_optimisation(n, ms_deadline, num_restarts=max_restarts)
        if a_multi is not None:
            val = compute_score(a_multi)
            print(f"[multi‑start] best value = {val:.6f}")
            if val < best_val:
                best_val, best_seq = val, a_multi.tolist()

    # -----------------------------------------------------------------
    # 3. Sub‑gradient polishing (short run)
    # -----------------------------------------------------------------
    sub_deadline = deadline - 30.0
    if best_seq is not None and time.time() < sub_deadline:
        a0 = np.array(best_seq, dtype=float)
        a_refined = subgradient_polish(a0, sub_deadline, step0=0.12)
        val = compute_score(a_refined)
        print(f"[sub‑grad] refined score = {val:.6f}")
        if val < best_val:
            best_val, best_seq = val, a_refined.tolist()

    # -----------------------------------------------------------------
    # 4. Final greedy hill‑climb
    # -----------------------------------------------------------------
    hill_deadline = deadline - 5.0
    if best_seq is not None and time.time() < hill_deadline:
        a_start = np.array(best_seq, dtype=float)
        a_final = hill_climb(a_start, hill_deadline)
        val = compute_score(a_final)
        print(f"[hill‑climb] final score = {val:.6f}")
        if val < best_val:
            best_val, best_seq = val, a_final.tolist()

    # -----------------------------------------------------------------
    # Safeguard – fall back to a uniform vector if nothing was found
    # -----------------------------------------------------------------
    if best_seq is None:
        best_seq = [1.0 / n] * n
        best_val = evaluate_sequence(best_seq)

    # -----------------------------------------------------------------
    # Final report (using the official evaluate_sequence)
    # -----------------------------------------------------------------
    final_score = evaluate_sequence(best_seq)
    print(f"\nSearch finished. Reported score = {final_score:.6f}")
    return best_seq
