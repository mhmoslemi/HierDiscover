def solve_lp(f, M):
    n = len(f)
    # construct A_ub as sparse.
    # We'll use scipy.sparse.diags, but need to use shape (2*n-1, n). Offsets from 0 to n-1.
    # Using diags with each column constant f_i.
    from scipy.sparse import diags
    data = [np.full(n, f[i]) for i in range(n)]
    offsets = list(range(n))
    A = diags(data, offsets, shape=(2*n-1, n), format='csc')
    # but each column row positions: row = col + offset. This matches convolution.
    # Actually each column j has values f[i] at row i + j for i from 0..n-1, which is exactly the convolution.
    # So this A works.

    # Now linear program.
    c = -np.ones(n)  # minimize -sum(g)
    b_ub = np.full(2*n - 1, float(M))
    bounds = [(0, None)] * n
    res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
    if not res.success:
        print(f"LP failed: {res.message}")
        return None
    return res.x

"""
Search for a non‑negative step function that minimizes
``evaluate_sequence``.  The algorithm proceeds in three stages:

1. **Coarse seeding** – scan a few analytic families (symmetric product,
   left‑heavy power, exponential‑U) and a few random starts to obtain a
   decent initial vector on the simplex.

2. **LP‑based alternating improvement** – given a vector `f` with
   max‑auto‑convolution `M = max(f*f)`, solve the linear program
       maximise   Σ g_i
       subject to f * g ≤ M   (convolution, element‑wise)
                  g ≥ 0
   to obtain a direction `g`.  After normalising `g` to sum 1 we replace
   `f ← (1‑t)·f + t·g` (with a moderate blending factor `t`) and project
   back onto the simplex.  Repeating this step strictly decreases the
   objective and quickly refines the shape.

3. **Fine‑grained polishing** – a smoothed‑max Adam optimiser (as in the
   reference code) followed by a short simulated‑annealing hill‑climb.

All vectors are kept symmetric (`a = (a+a[::-1])/2`) and projected onto
the probability simplex so that `∑a_i = 1`.  The final list of floats is
returned; its score equals ``evaluate_sequence`` because the sum is 1.
"""

import time
import numpy as np
from typing import List, Tuple

# ----------------------------------------------------------------------
# Core utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full aperiodic convolution using real FFT."""
    m = a.size + b.size - 1
    nfft = next_pow2(m)
    A = np.fft.rfft(a, nfft)
    B = np.fft.rfft(b, nfft)
    C = A * B
    conv = np.fft.irfft(C, nfft)[:m]
    return conv


def compute_score(a: np.ndarray) -> float:
    """
    Exact objective for a probability vector (∑a = 1):
    2·n·max(a*a).  This equals ``evaluate_sequence`` once the sum is 1.
    """
    n = a.shape[0]
    conv = convolution_fft(a, a)
    return 2.0 * n * float(conv.max())


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex (sum=1, v≥0)."""
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / v.shape[0])

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, v.shape[0] + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.where(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Analytic families for seeding
# ----------------------------------------------------------------------


def symmetric_product_family(n: int, alpha: float) -> np.ndarray:
    """a_i ∝ [(i+½)(n‑½‑i)]^{‑α}, symmetric, on the simplex."""
    i = np.arange(n, dtype=np.float64)
    logprod = np.log(i + 0.5) + np.log(n - 0.5 - i)
    w = np.exp(-alpha * logprod)
    # handle possible infinities at the ends (when i≈0 or i≈n‑1)
    w[np.isinf(w)] = np.finfo(np.float64).max
    w /= w.sum()
    return w


def left_heavy_power_family(n: int, beta: float) -> np.ndarray:
    """a_i ∝ (i+½)^{‑β}, monotone decreasing, on the simplex."""
    i = np.arange(n, dtype=np.float64)
    w = np.exp(-beta * np.log(i + 0.5))
    w /= w.sum()
    return w


def exp_u_family(n: int, gamma: float) -> np.ndarray:
    """U‑shape: a_i ∝ exp(-γ·min(i, n‑1‑i))."""
    i = np.arange(n, dtype=np.float64)
    dist = np.minimum(i, n - 1 - i)
    w = np.exp(-gamma * dist)
    w /= w.sum()
    return w


def random_monotone_start(n: int) -> np.ndarray:
    """Random decreasing vector obtained by sorting a powered uniform."""
    r = np.random.rand(n) ** np.random.uniform(0.5, 5.0)
    r.sort()
    a = r[::-1]
    a /= a.sum()
    return a


def random_u_start(n: int) -> np.ndarray:
    """Random exponential‑U shape."""
    gamma = np.random.uniform(0.0, 6.0)
    return exp_u_family(n, gamma)

# Linear‑programming refinement
# ----------------------------------------------------------------------


def solve_lp(f: np.ndarray, M: float) -> np.ndarray | None:
    """
    Solve the LP:
        maximise   Σ g_i
        subject to f * g ≤ M   (full convolution)
                   g_i ≥ 0
    Returns the optimal vector g (may contain very small entries) or None.
    """
    from scipy.optimize import linprog
    from scipy.sparse import diags

    n = f.shape[0]

    # Build the (2n‑1)×n constraint matrix A where (A·g)_k = (f*g)_k.
    # The i‑th diagonal (offset i) is filled with f[i].
    data = [np.full(n, f[i], dtype=np.float64) for i in range(n)]
    offsets = np.arange(n, dtype=int)
    A = diags(data, offsets, shape=(2 * n - 1, n), format='csc')

    c = -np.ones(n, dtype=np.float64)                # maximise sum(g)  → minimise -sum(g)
    b_ub = np.full(2 * n - 1, M, dtype=np.float64)  # all constraints have RHS = M
    bounds = [(0, None)] * n

    res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds,
                  method='highs', options={'presolve': True})
    if not res.success:
        # LP failed – fall back to None
        return None
    return res.x

def softmax(z: np.ndarray) -> np.ndarray:
    """Numerically stable soft‑max."""
    z_max = np.max(z)
    e = np.exp(z - z_max)
    return e / e.sum()


def adam_smooth_optimize(start_a: np.ndarray,
                         deadline: float,
                         max_iters: int = 200_000,
                         beta_start: float = 5.0,
                         beta_end: float = 1500.0,
                         lr: float = 0.018,
                         sym_interval: int = 4000) -> Tuple[np.ndarray, float]:
    """
    Minimise the smoothed max of the auto‑convolution using Adam on the
    soft‑max parametrisation.  Periodically symmetrises the iterate.
    """
    n = start_a.shape[0]
    eps = 1e-12
    z = np.log(start_a + eps)

    m = np.zeros_like(z)
    v = np.zeros_like(z)

    beta1 = 0.9
    beta2 = 0.999
    eps_adam = 1e-12

    best_a = softmax(z)
    best_score = compute_score(best_a)

    for it in range(1, max_iters + 1):
        if time.time() >= deadline:
            break

        # exponential schedule for the smoothing parameter β
        t = it / max_iters
        beta = beta_start * (beta_end / beta_start) ** t

        a = softmax(z)

        if sym_interval and (it % sym_interval == 0):
            a = (a + a[::-1]) * 0.5
            a = project_to_simplex(a)
            z = np.log(a + eps)

        conv = convolution_fft(a, a)
        maxc = conv.max()
        # smooth‑max weights
        exp_w = np.exp(beta * (conv - maxc))
        w = exp_w / exp_w.sum()

        # gradient w.r.t. a (derived from smooth max)
        a_rev = a[::-1]
        conv_rev = convolution_fft(a_rev, w)
        grad_a = 2.0 * conv_rev[n - 1:n - 1 + n]

        # back‑prop through soft‑max
        dot_ag = np.dot(a, grad_a)
        grad_z = a * grad_a - dot_ag * a

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad_z
        v = beta2 * v + (1.0 - beta2) * (grad_z ** 2)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        z -= lr * m_hat / (np.sqrt(v_hat) + eps_adam)

        if it % 1000 == 0:
            a_tmp = softmax(z)
            a_tmp = (a_tmp + a_tmp[::-1]) * 0.5
            a_tmp = project_to_simplex(a_tmp)
            sc = compute_score(a_tmp)
            if sc < best_score:
                best_score = sc
                best_a = a_tmp.copy()
                print(f"[{time.time():.1f}] Adam improved to {best_score:.6f}")

    # final projection
    a_fin = softmax(z)
    a_fin = (a_fin + a_fin[::-1]) * 0.5
    a_fin = project_to_simplex(a_fin)
    sc_fin = compute_score(a_fin)
    if sc_fin < best_score:
        best_score = sc_fin
        best_a = a_fin.copy()

    return best_a, best_score

def hill_climb_sa(start_a: np.ndarray,
                  deadline: float,
                  max_steps: int = 500_000) -> Tuple[np.ndarray, float]:
    """
    Random walk on the simplex with occasional uphill moves.
    50 % swap two entries, 50 % transfer a fraction of mass.
    """
    n = start_a.size
    cur_a = start_a.copy()
    cur_score = compute_score(cur_a)

    best_a = cur_a.copy()
    best_score = cur_score

    # initial temperature based on current score
    T0 = best_score * 1e-4
    T = T0
    step_scale = 0.12

    for step in range(1, max_steps + 1):
        if time.time() >= deadline:
            break

        cand = cur_a.copy()
        if np.random.rand() < 0.5:      # swap two entries
            i, j = np.random.randint(n, size=2)
            cand[i], cand[j] = cand[j], cand[i]
        else:                           # transfer a fraction of mass
            i, j = np.random.randint(n, size=2)
            if i != j:
                delta = step_scale * cur_a[i]
                cand[i] -= delta
                cand[j] += delta
                if cand[i] < 0.0:
                    cand[i] = 0.0

        cand = project_to_simplex(cand)
        cand_score = compute_score(cand)

        delta = cand_score - cur_score
        if delta < 0.0 or np.random.rand() < np.exp(-delta / (T + 1e-18)):
            cur_a = cand
            cur_score = cand_score
            if cur_score < best_score:
                best_score = cur_score
                best_a = cur_a.copy()
                step_scale = min(step_scale * 1.07, 0.5)
        else:
            step_scale *= 0.97

        # linear cooling
        T = T0 * (1.0 - step / max_steps)

    return best_a, best_score

    candidates = []

    # symmetric product family (α)
    for alpha in np.linspace(0.0, 12.0, 7):
        candidates.append(symmetric_product_family(n, alpha))

    # left‑heavy power family (β)
    for beta in np.linspace(0.0, 12.0, 7):
        candidates.append(left_heavy_power_family(n, beta))

    # exponential U‑shape (γ)
    for gamma in np.linspace(0.0, 6.0, 7):
        candidates.append(exp_u_family(n, gamma))

    # a few random starts
    for _ in range(10):
        if np.random.rand() < 0.5:
            candidates.append(random_monotone_start(n))
        else:
            candidates.append(random_u_start(n))

    # evaluate all candidates (quick FFT‑based score)
    for vec in candidates:
        vec = (vec + vec[::-1]) * 0.5           # enforce symmetry
        vec = project_to_simplex(vec)
        sc = compute_score(vec)
        if sc < best_score:
            best_score = sc
            best_seq = vec.tolist()
            print(f"[{time.time():.1f}] new seed best = {best_score:.6f}")

    # -----------------------------------------------------------
    # 2. Linear‑programming alternating refinement
    # -----------------------------------------------------------
    if best_seq is not None and time.time() < deadline - 30.0:
        print(f"[{time.time():.1f}] entering LP refinement (n={n})")
        f = np.array(best_seq, dtype=np.float64)

        max_lp_iters = 12
        for lp_it in range(max_lp_iters):
            if time.time() >= deadline - 10.0:
                break
            # current bound
            conv = convolution_fft(f, f)
            M = conv.max()
            # solve LP
            g0 = solve_lp(f, M)
            if g0 is None:
                print(f"[{time.time():.1f}] LP failed at iteration {lp_it}")
                break
            sum_g = g0.sum()
            if sum_g <= 0:
                break
            g = g0 / sum_g

            # blend (moderate mixing)
            t = 0.4 if lp_it < 4 else 0.2
            f_new = (1.0 - t) * f + t * g
            f_new = project_to_simplex(f_new)
            # enforce symmetry again
            f_new = (f_new + f_new[::-1]) * 0.5
            f_new = project_to_simplex(f_new)

            new_score = compute_score(f_new)
            if new_score + 1e-12 < best_score:   # strict improvement
                best_score = new_score
                best_seq = f_new.tolist()
                print(f"[{time.time():.1f}] LP step {lp_it} improved to {best_score:.6f}")
                f = f_new
            else:
                # no improvement – stop early
                print(f"[{time.time():.1f}] LP step {lp_it} gave no improvement")
                break

    # -----------------------------------------------------------
    # 3. Adam smoothing (first pass)
    # -----------------------------------------------------------
    if best_seq is not None and time.time() < deadline - 20.0:
        print(f"[{time.time():.1f}] Adam smoothing (pass 1)")
        a_start = np.array(best_seq, dtype=np.float64)
        a_opt, sc_opt = adam_smooth_optimize(
            a_start,
            deadline - 5.0,
            max_iters=180_000,
            beta_start=5.0,
            beta_end=1500.0,
            lr=0.018,
            sym_interval=4000,
        )
        if sc_opt < best_score:
            best_score = sc_opt
            best_seq = a_opt.tolist()
            print(f"[{time.time():.1f}] after Adam pass 1 best = {best_score:.6f}")

    # -----------------------------------------------------------
    # 4. Simulated‑annealing hill‑climb (short)
    # -----------------------------------------------------------
    if best_seq is not None and time.time() < deadline - 5.0:
        print(f"[{time.time():.1f}] final hill‑climb (SA)")
        a_final, sc_final = hill_climb_sa(
            np.array(best_seq, dtype=np.float64),
            deadline,
            max_steps=400_000,
        )
        if sc_final < best_score:
            best_score = sc_final
            best_seq = a_final.tolist()
            print(f"[{time.time():.1f}] SA improved best to {best_score:.6f}")

    # -----------------------------------------------------------
    # 5. Safeguard fallback (very small uniform vector)
    # -----------------------------------------------------------
    if best_seq is None:
        n_small = 256
        best_seq = [1.0 / n_small] * n_small
        best_score = evaluate_sequence(best_seq)

    # Final sanity – enforce sum=1 (evaluate_sequence rescales anyway)
    best_arr = np.array(best_seq, dtype=np.float64)
    best_arr = project_to_simplex(best_arr)
    final_score = evaluate_sequence(best_arr.tolist())
    print(f"\n[{time.time():.1f}] finished search – final reported score = {final_score:.6f}")

    return best_arr.tolist()
