def solve_lp_for_direction(a, M, eps=1e-12):
    n = a.shape[0]
    # ensure a positive
    a_eps = np.maximum(a, eps)
    a_eps = a_eps / a_eps.sum()
    # Build matrix A (size (2n-1) x n)
    rows = 2*n - 1
    A = np.zeros((rows, n), dtype=np.float64)
    for j in range(n):
        A[j:j+n, j] = a_eps
    # Linear program to maximize sum(b) i.e. minimize -sum(b)
    c = -np.ones(n, dtype=np.float64)
    b_ub = np.full(rows, M, dtype=np.float64)
    bounds = [(eps, None) for _ in range(n)]
    try:
        res = opt.linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
    except Exception as e:
        print(f"linprog exception: {e}")
        return None
    if not res.success:
        print(f"linprog failed: {res.message}")
        return None
    b = res.x
    if b.sum() == 0:
        return None
    g = b / b.sum()
    return g
```

"""
Non‑negative step‑function optimiser (LP‑guided).

Algorithm
---------
1. Initialise a probability vector a (sum = 1).  If a global
   ``height_sequence_1`` is present we start from it, otherwise from a
   central‑binomial distribution (known to be a strong baseline).

2. Repeatedly apply an *LP refinement*:
   * Let M = max(conv(a,a)).
   * Solve the linear programme
         maximise   ∑ b_i
         subject to conv(a,b) ≤ M   (component‑wise)
                    b_i ≥ ε
     (ε≈1e‑9 prevents zero columns and guarantees boundedness.)
   * Normalise the optimiser b to a probability vector g = b/∑b.
   * Mix a with g (several mixing coefficients) and keep the mixture
     with the smallest objective 2·n·max(conv).

   The LP step pushes mass towards indices that are “quiet’’ for the
   current a, thereby flattening the autocorrelation.  A few iterations
   (≈10‑15) already improve the score noticeably.

3. A short projected sub‑gradient polish (as in the previous code) is
   run to clean up any remaining irregularities.

4. Finally a cheap stochastic hill‑climb makes tiny local adjustments.

All phases respect the wall‑clock deadline (default 1000 s) and always
keep the best sequence seen.  The returned list already lies on the
simplex, contains only non‑negative numbers and can be passed directly
to ``evaluate_sequence``.
"""

import time
import numpy as np
import scipy.optimize as opt

# ----------------------------------------------------------------------
# Basic utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """
    Full linear convolution via real FFT.
    Returns length len(a)+len(b)-1 array.
    """
    m = a.shape[0] + b.shape[0] - 1
    n_fft = next_pow2(m)
    A = np.fft.rfft(a, n=n_fft)
    B = np.fft.rfft(b, n=n_fft)
    C = A * B
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


def compute_score(a: np.ndarray) -> float:
    """
    Exact objective for a probability vector ``a`` (∑a = 1).
    Returns   2·n·max(conv(a,a)).
    """
    n = a.shape[0]
    conv = convolution_fft(a, a)
    return 2.0 * n * float(conv.max())


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of v onto the probability simplex:
        sum(v) = 1, v ≥ 0.
    """
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / v.shape[0])

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, v.shape[0] + 1)
    cond = u - cssv / ind > 0
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


# ----------------------------------------------------------------------
# Analytic starter families
# ----------------------------------------------------------------------


def generate_central_binomial(n: int) -> np.ndarray:
    """a_i = C(2i,i) / 4^i, normalised to sum = 1 (monotone decreasing)."""
    a = np.empty(n, dtype=np.float64)
    a[0] = 1.0
    for i in range(1, n):
        a[i] = a[i - 1] * (2 * i - 1) / (2.0 * i)
    return project_to_simplex(a)


def generate_powerlaw(n: int, p: float) -> np.ndarray:
    """a_i ∝ (i+1)^{-p}, normalised to sum = 1."""
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = idx ** (-p)
    return project_to_simplex(a)


def generate_geometric(n: int, r: float) -> np.ndarray:
    """Geometric distribution a_i = (1‑r)·r^i, normalised (exactly sum = 1 if r<1)."""
    i = np.arange(n, dtype=np.float64)
    a = (1.0 - r) * (r ** i)
    return project_to_simplex(a)


# ----------------------------------------------------------------------
# Linear‑programming refinement (LP step)
# ----------------------------------------------------------------------


def lp_refine_direction(a: np.ndarray, M: float, eps: float = 1e-9):
    """
    Solve   maximise   sum(b)
    s.t.    conv(a,b) ≤ M   (component‑wise),   b ≥ eps.

    Returns the normalised direction g = b / sum(b) or None on failure.
    """
    n = a.shape[0]

    # enforce a_i ≥ eps to keep the LP bounded
    a_pos = np.maximum(a, eps)
    a_pos = a_pos / a_pos.sum()

    # Build the (2n‑1) × n matrix A with A[k,j] = a_pos[k-j] (0 when out of range)
    rows = 2 * n - 1
    A = np.zeros((rows, n), dtype=np.float64)
    for j in range(n):
        A[j:j + n, j] = a_pos

    # Linear programme: maximise sum(b)  →  minimise -sum(b)
    c = -np.ones(n, dtype=np.float64)
    b_ub = np.full(rows, M, dtype=np.float64)
    bounds = [(eps, None) for _ in range(n)]

    try:
        res = opt.linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method="highs")
    except Exception as exc:  # pragma: no cover
        print(f"linprog raised exception: {exc}")
        return None

    if not res.success:
        # LP can be infeasible for an excessively tight M;
        # just abort the refinement step.
        return None

    b = res.x
    s = b.sum()
    if s <= 0:
        return None
    g = b / s
    return g


# ----------------------------------------------------------------------
# Sub‑gradient optimiser (used for a short polish)
# ----------------------------------------------------------------------


def subgrad_optimize(
    init_a: np.ndarray,
    time_deadline: float,
    max_iter: int = 300_000,
    step0: float = 0.07,
) -> tuple[np.ndarray, float]:
    """Projected sub‑gradient descent on 2·n·max(conv(a,a))."""
    a = init_a.astype(np.float64).copy()
    best_a = a.copy()
    best_score = compute_score(a)
    n = a.shape[0]

    for t in range(1, max_iter + 1):
        if time.time() >= time_deadline:
            break

        conv = convolution_fft(a, a)
        k = int(np.argmax(conv))

        # gradient of the active entry
        idx = np.arange(n)
        j = k - idx
        mask = (j >= 0) & (j < n)
        g = np.zeros_like(a)
        g[mask] = 2.0 * a[j[mask]]

        step = step0 / np.sqrt(t)
        a_new = a - step * g
        a_new = project_to_simplex(a_new)

        cur_score = compute_score(a_new)
        if cur_score < best_score:
            best_score, best_a = cur_score, a_new.copy()

        a = a_new

        # occasional jitter to escape flats
        if t % 1000 == 0:
            jitter = np.random.randn(n) * (step0 * 0.004 / np.sqrt(t))
            a += jitter
            a = project_to_simplex(a)

    final_score = compute_score(a)
    if final_score < best_score:
        best_score, best_a = final_score, a.copy()
    return best_a, best_score


# ----------------------------------------------------------------------
# Stochastic hill‑climb (final polishing)
# ----------------------------------------------------------------------


def hill_climb_permutation(
    seq,
    time_deadline,
    max_steps: int = 200_000,
) -> tuple[list[float], float]:
    """Tiny stochastic moves (swap or fractional transfer) with simplex projection."""
    a = np.array(seq, dtype=np.float64)
    n = a.shape[0]
    best_a = a.copy()
    best_score = compute_score(best_a)

    step_scale = 0.08  # initial fraction of mass moved

    while max_steps > 0 and time.time() < time_deadline:
        cand = best_a.copy()
        if np.random.rand() < 0.5:      # swap two entries
            i, j = np.random.randint(n, size=2)
            cand[i], cand[j] = cand[j], cand[i]
        else:                           # move a fraction from i to j
            i, j = np.random.randint(n, size=2)
            if i != j:
                delta = step_scale * best_a[i]
                cand[i] -= delta
                cand[j] += delta
                if cand[i] < 0:
                    cand[i] = 0.0

        cand = project_to_simplex(cand)
        cur_score = compute_score(cand)

        if cur_score < best_score:
            best_score = cur_score
            best_a = cand
            step_scale = min(step_scale * 1.06, 0.5)
        else:
            step_scale *= 0.97

        max_steps -= 1

    return best_a.tolist(), best_score
