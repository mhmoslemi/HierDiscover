    n = len(a)
    conv = convolution_fft(a, a)
    k = np.argmax(conv)
    # compute subgradient vector
    # i indices where 0 <= k-i < n -> i in [k-n+1, k]

Now produce answer.assistantfinal**Solution Overview**

The objective is  

\[
F(a)=\frac{2\,n\;\max_k (a*a)_k}{\bigl(\sum_i a_i\bigr)^2},
\qquad a_i\ge 0 .
\]

Because the ratio is homogeneous, we may restrict the search to the
probability simplex (`∑a_i = 1`).  
The maximum of the full (aperiodic) self‑convolution is the only
non‑smooth part of the objective.  We minimise a smooth *log‑sum‑exp*
approximation of the maximum and tighten the approximation during the
run.  The optimisation is performed on three hierarchical levels:

1. **Wide exploration** – a large family of analytically‑parameterised
   vectors (uniform, exponential, left‑heavy power, symmetric product,
   *asymmetric* product) together with many Dirichlet draws.
2. **Smooth optimisation** – Adam on the soft‑max parametrisation of a
   vector, with an annealed temperature `β`.  Two phases are used:
   a long low‑temperature phase (`β≈5…5 000`) and a short high‑temperature
   phase (`β≈5 000…50 000`).  The number of iterations is adapted to the
   remaining wall‑clock time.
3. **Exact sub‑gradient descent** – a mirror‑descent (multiplicative)
   update using the true sub‑gradient of the `max`‐operator, followed by
   a longer simulated‑annealing (SA) hill‑climb that is allowed to modify
   both the values and the permutation of the entries.

All stages keep a global best vector (plain Python list) and stop early
when the time budget is exhausted – the harness will then evaluate the
last saved vector.

The implementation is completely self‑contained, uses only NumPy, and
fits easily into the 1000 s budget (the code checks the remaining time
after each major loop).

```python
"""
Propose a step‑function sequence that minimises the evaluation function.

Algorithm (≈ 1000 s)
--------------------
1. Choose a length `n`.  The larger the better; we use 32768 when the
   budget permits, otherwise 16384 or 8192.
2. Scan a rich collection of analytic families (uniform, exponential,
   left‑heavy power, symmetric product, *asymmetric* product) and a
   large number of Dirichlet draws.  The best candidate from this cheap
   stage is stored.
3. Smooth optimisation:
   – Adam on the soft‑max parametrisation `a = softmax(z)`.
   – Two annealing phases: (i) low temperature (β 5 → 5 000) for many
     iterations; (ii) high temperature (β 5 000 → 50 000) to tighten the
     surrogate.  Restarts are performed with different random starts
     while time permits.
4. Exact optimisation:
   – Mirror (entropy) sub‑gradient descent on the true objective.
   – Afterwards a simulated‑annealing hill‑climb that is allowed to
     swap entries or transfer mass.  This stage refines the permutation
     as well as the values.
5. Return the best plain Python list found before the deadline.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Basic utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Aperiodic full convolution using a real FFT (fast, no Python loops)."""
    m = a.size + b.size - 1
    nfft = next_pow2(m)
    A = np.fft.rfft(a, n=nfft)
    B = np.fft.rfft(b, n=nfft)
    C = A * B
    conv = np.fft.irfft(C, n=nfft)[:m]
    return conv


def compute_score(a: np.ndarray) -> float:
    """
    Exact objective for a probability vector (∑a = 1):
        2·n·max(a * a)
    """
    n = a.shape[0]
    conv = convolution_fft(a, a)
    return 2.0 * n * float(conv.max())


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the probability simplex
    (non‑negative, sum = 1).
    """
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / v.shape[0])

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, v.shape[0] + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.where(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def softmax(z: np.ndarray) -> np.ndarray:
    """Numerically stable soft‑max."""
    z_max = np.max(z)
    e = np.exp(z - z_max)
    return e / e.sum()


# ----------------------------------------------------------------------
# Analytic families
# ----------------------------------------------------------------------


def uniform_family(n: int) -> np.ndarray:
    """Uniform probability vector."""
    return np.full(n, 1.0 / n)


def exponential_decay_family(n: int, gamma: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    w = np.exp(-gamma * i)
    return w / w.sum()


def left_heavy_power_family(n: int, beta: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    w = np.exp(-beta * np.log(i + 0.5))
    return w / w.sum()


def symmetric_product_family(n: int, alpha: float) -> np.ndarray:
    """a_i ∝ [(i+½)(n‑½‑i)]^(‑α), symmetric."""
    i = np.arange(n, dtype=np.float64)
    logprod = np.log(i + 0.5) + np.log(n - 0.5 - i)
    w = np.exp(-alpha * logprod)
    w[np.isinf(w)] = np.finfo(np.float64).max
    return w / w.sum()


def asymmetric_product_family(n: int, alpha: float, beta: float) -> np.ndarray:
    """
    a_i ∝ (i+½)^(‑α)·(n‑½‑i)^(‑β)  (no symmetry forced).
    """
    i = np.arange(n, dtype=np.float64)
    w = np.exp(-alpha * np.log(i + 0.5) - beta * np.log(n - 0.5 - i))
    w[np.isinf(w)] = np.finfo(np.float64).max
    return w / w.sum()


# ----------------------------------------------------------------------
# Random initialisers
# ----------------------------------------------------------------------


def random_dirichlet_start(n: int, conc: float) -> np.ndarray:
    """Draw a symmetric Dirichlet(α) vector (α = conc)."""
    vec = np.random.gamma(conc, 1.0, size=n)
    vec = vec / vec.sum()
    return vec


def random_sparse_start(n: int) -> np.ndarray:
    """Very sparse vector (≈ 5 % non‑zeros) projected to the simplex."""
    sparsity = np.random.uniform(0.03, 0.07)
    k = max(2, int(round(sparsity * n)))
    idx = np.random.choice(n, size=k, replace=False)
    a = np.zeros(n, dtype=np.float64)
    a[idx] = 1.0 / k
    return project_to_simplex(a)


# ----------------------------------------------------------------------
# Smoothed optimisation (log‑sum‑exp) via Adam
# ----------------------------------------------------------------------


def adam_smooth_optimize(
    start_a: np.ndarray,
    deadline: float,
    max_iters: int,
    beta_start: float,
    beta_end: float,
    lr: float,
) -> tuple[np.ndarray, float]:
    """
    Minimise the smooth surrogate
          f_β(a) = (1/β)·log Σ_k exp(β·(a*a)_k)
    using Adam on the soft‑max parametrisation a = softmax(z).
    β is increased exponentially from beta_start to beta_end.
    """
    eps = 1e-12
    z = np.log(start_a + eps)               # initialise from log‑softmax
    m = np.zeros_like(z)
    v = np.zeros_like(z)

    beta1 = 0.9
    beta2 = 0.999
    eps_adam = 1e-12

    best_a = softmax(z)
    best_score = compute_score(best_a)

    for it in range(1, max_iters + 1):
        if time.time() >= deadline:
            break

        # ---- β schedule (exponential) ----
        t = it / max_iters
        beta = beta_start * (beta_end / beta_start) ** t

        a = softmax(z)

        # smoothing weight over shifts
        conv = convolution_fft(a, a)
        maxc = conv.max()
        exp_w = np.exp(beta * (conv - maxc))
        w = exp_w / exp_w.sum()                         # soft‑max over shifts

        # gradient of f_β w.r.t. a
        a_rev = a[::-1]
        conv_rev = convolution_fft(a_rev, w)
        grad_a = 2.0 * conv_rev[a.shape[0] - 1 : a.shape[0] - 1 + a.shape[0]]

        # back‑prop through soft‑max
        dot_ag = np.dot(a, grad_a)
        grad_z = a * grad_a - dot_ag * a

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad_z
        v = beta2 * v + (1.0 - beta2) * (grad_z ** 2)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        z -= lr * m_hat / (np.sqrt(v_hat) + eps_adam)

        # occasional exact check
        if it % 500 == 0:
            a_tmp = softmax(z)
            sc = compute_score(a_tmp)
            if sc < best_score:
                best_score = sc
                best_a = a_tmp.copy()
                print(f"[{time.time():.1f}]   Adam iter {it:7d} – new best = {best_score:.6f}")

    # final check
    a_fin = softmax(z)
    sc_fin = compute_score(a_fin)
    if sc_fin < best_score:
        best_score = sc_fin
        best_a = a_fin.copy()
    return best_a, best_score


# ----------------------------------------------------------------------
# Mirror (entropy) sub‑gradient descent on the true max‑objective
# ----------------------------------------------------------------------


def mirror_subgrad_optimize(
    start_a: np.ndarray,
    deadline: float,
    step0: float = 0.9,
    check_interval: int = 2000,
) -> tuple[np.ndarray, float]:
    """
    Entropy‑mirror (multiplicative) sub‑gradient descent for
    2·n·max(a*a).  Step size η_t = step0 / sqrt(t).
    """
    n = start_a.shape[0]
    a = start_a.astype(np.float64).copy()
    best_a = a.copy()
    best_score = compute_score(a)

    t = 0
    while time.time() < deadline:
        t += 1
        conv = convolution_fft(a, a)
        k = int(np.argmax(conv))                     # active shift index

        # sub‑gradient for the active shift
        idx = np.arange(n)
        j = k - idx
        mask = (j >= 0) & (j < n)
        g = np.zeros_like(a)
        g[mask] = 2.0 * a[j[mask]]

        eta = step0 / np.sqrt(t)
        a = a * np.exp(-eta * g)                     # multiplicative update
        s = a.sum()
        if s == 0.0:
            a = np.full_like(a, 1.0 / n)
        else:
            a = a / s                                 # renormalise to simplex

        if t % check_interval == 0:
            sc = compute_score(a)
            if sc < best_score:
                best_score = sc
                best_a = a.copy()
                print(f"[{time.time():.1f}]   Mirror step {t:7d} – new best = {best_score:.6f}")

    sc = compute_score(a)
    if sc < best_score:
        best_score = sc
        best_a = a.copy()
    return best_a, best_score


# ----------------------------------------------------------------------
# Stochastic hill‑climb (simulated annealing)
# ----------------------------------------------------------------------


def hill_climb_sa(
    start_a: np.ndarray,
    deadline: float,
    max_steps: int = 2_000_000,
    init_temp_factor: float = 5e-4,
) -> tuple[np.ndarray, float]:
    """
    Simulated‑annealing walk on the simplex.
    - swap two entries (probability 0.5)
    - otherwise transfer a small fraction of mass from i to j
    """
    n = start_a.size
    cur_a = start_a.copy()
    cur_score = compute_score(cur_a)

    best_a = cur_a.copy()
    best_score = cur_score

    T0 = best_score * init_temp_factor
    T = T0
    step_scale = 0.12

    for step in range(1, max_steps + 1):
        if time.time() >= deadline:
            break

        cand = cur_a.copy()
        if np.random.rand() < 0.5:                     # swap
            i, j = np.random.randint(n, size=2)
            cand[i], cand[j] = cand[j], cand[i]
        else:                                          # mass transfer
            i, j = np.random.randint(n, size=2)
            if i != j:
                delta = step_scale * cur_a[i]
                cand[i] -= delta
                cand[j] += delta
                if cand[i] < 0.0:
                    cand[i] = 0.0

        cand = project_to_simplex(cand)
        cand_score = compute_score(cand)
        delta = cand_score - cur_score

        # Metropolis acceptance
        if delta < 0.0 or np.random.rand() < np.exp(-delta / (T + 1e-18)):
            cur_a = cand
            cur_score = cand_score
            if cur_score < best_score:
                best_score = cur_score
                best_a = cur_a.copy()
                step_scale = min(step_scale * 1.07, 0.5)
        else:
            step_scale *= 0.97

        # linear cooling schedule
        T = T0 * (1.0 - step / max_steps)

    return best_a, best_score


# ----------------------------------------------------------------------
# High‑level driver
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 12345, budget_s: int = 1000, **_) -> list[float]:
    """
    Search for a non‑negative step‑function (list of floats) that minimises
    ``evaluate_sequence`` under the given time budget.
    Returns a plain Python list.
    """
    np.random.seed(seed)
    start_time = time.time()
    # safety margin for the surrounding harness
    deadline = start_time + budget_s - 2.0

    # -------------------------------------------------------------
    # Choose an appropriate length
    # -------------------------------------------------------------
    if budget_s >= 900:
        n = 32768
    elif budget_s >= 600:
        n = 16384
    else:
        n = 8192
    print(f"[{time.time():.1f}] Using length n = {n}")

    # -------------------------------------------------------------
    # Global best bookkeeping
    # -------------------------------------------------------------
    best_seq: list[float] | None = None
    best_score = float("inf")

    def maybe_update(vec: np.ndarray, tag: str = "") -> None:
        nonlocal best_seq, best_score
        sc = compute_score(vec)
        if sc < best_score:
            best_score = sc
            best_seq = vec.tolist()
            print(f"[{time.time():.1f}] new best {tag:<20} = {best_score:.6f}")

    # -------------------------------------------------------------
    # 0. Fallback – use any previously known construction if present
    # -------------------------------------------------------------
    try:
        fb_vec = [float(x) for x in height_sequence_1]  # type: ignore
        fb_sc = evaluate_sequence(fb_vec)
        print(f"[{time.time():.1f}] fallback score = {fb_sc:.6f}")
        if fb_sc < best_score:
            best_score = fb_sc
            best_seq = fb_vec
    except Exception:
        pass

    # -------------------------------------------------------------
    # 1. Analytic families + random Dirichlet draws (cheap sweep)
    # -------------------------------------------------------------
    maybe_update(uniform_family(n), "uniform")

    # symmetric product (α‑grid)
    for alpha in np.linspace(0.0, 8.0, 41):
        if time.time() >= deadline - 30.0:
            break
        vec = symmetric_product_family(n, alpha)
        maybe_update(vec, f"sym‑α={alpha:.2f}")

    # left‑heavy power law (β‑grid)
    for beta in np.linspace(0.0, 6.0, 41):
        if time.time() >= deadline - 30.0:
            break
        vec = left_heavy_power_family(n, beta)
        maybe_update(vec, f"pow‑β={beta:.2f}")

    # exponential decay (γ‑grid)
    for gamma in np.linspace(0.0, 8.0, 51):
        if time.time() >= deadline - 30.0:
            break
        vec = exponential_decay_family(n, gamma)
        maybe_update(vec, f"exp‑γ={gamma:.2f}")

    # asymmetric product (α,β) – a coarse 2‑D sweep
    alphas = np.linspace(0.0, 6.0, 7)
    betas = np.linspace(0.0, 6.0, 7)
    for alpha in alphas:
        for beta in betas:
            if time.time() >= deadline - 30.0:
                break
            vec = asymmetric_product_family(n, alpha, beta)
            maybe_update(vec, f"asym‑α={alpha:.1f}_β={beta:.1f}")

    # Dirichlet draws with many concentration values
    conc_vals = np.logspace(-2, 1, 25)  # 0.01 … 10
    for conc in conc_vals:
        for _ in range(2):
            if time.time() >= deadline - 30.0:
                break
            vec = random_dirichlet_start(n, conc)
            maybe_update(vec, f"dir‑α={conc:.3f}")

    # -------------------------------------------------------------
    # 2. Smoothed optimisation (Adam) – many restarts while time permits
    # -------------------------------------------------------------
    # Allocate roughly 60 % of the remaining budget to this phase.
    remaining = deadline - time.time()
    if remaining > 300.0:
        phase_deadline = time.time() + remaining * 0.6
        while time.time() < phase_deadline - 30.0:
            # a mixture of dense and sparse starts improves diversity
            if np.random.rand() < 0.5:
                start_vec = random_dirichlet_start(n, np.random.uniform(0.05, 5.0))
            else:
                start_vec = random_sparse_start(n)

            a_opt, sc_opt = adam_smooth_optimize(
                start_vec,
                deadline=phase_deadline,
                max_iters=300_000,
                beta_start=5.0,
                beta_end=5_000.0,
                lr=0.018,
            )
            if sc_opt < best_score:
                best_score = sc_opt
                best_seq = a_opt.tolist()
                print(f"[{time.time():.1f}]   new best after low‑β Adam = {best_score:.6f}")

    # -------------------------------------------------------------
    # 3. High‑β Adam (tight surrogate)
    # -------------------------------------------------------------
    if best_seq is not None and time.time() < deadline - 120.0:
        high_beta_deadline = deadline - 100.0
        print(f"[{time.time():.1f}] high‑β Adam refinement")
        a_ref, sc_ref = adam_smooth_optimize(
            np.array(best_seq, dtype=np.float64),
            deadline=high_beta_deadline,
            max_iters=150_000,
            beta_start=5_000.0,
            beta_end=50_000.0,
            lr=0.012,
        )
        if sc_ref < best_score:
            best_score = sc_ref
            best_seq = a_ref.tolist()
            print(f"[{time.time():.1f}] after high‑β Adam = {best_score:.6f}")

    # -------------------------------------------------------------
    # 4. Exact mirror sub‑gradient descent
    # -------------------------------------------------------------
    if best_seq is not None and time.time() < deadline - 70.0:
        subgrad_deadline = deadline - 50.0
        print(f"[{time.time():.1f}] mirror sub‑gradient descent")
        a_sub, sc_sub = mirror_subgrad_optimize(
            np.array(best_seq, dtype=np.float64),
            deadline=subgrad_deadline,
            step0=0.95,
        )
        if sc_sub < best_score:
            best_score = sc_sub
            best_seq = a_sub.tolist()
            print(f"[{time.time():.1f}] after mirror descent = {best_score:.6f}")

    # -------------------------------------------------------------
    # 5. Final long simulated‑annealing hill‑climb
    # -------------------------------------------------------------
    if best_seq is not None and time.time() < deadline - 5.0:
        print(f"[{time.time():.1f}] final hill‑climb (SA)")
        sa_deadline = deadline
        a_final, sc_final = hill_climb_sa(
            np.array(best_seq, dtype=np.float64),
            deadline=sa_deadline,
            max_steps=2_000_000,
        )
        if sc_final < best_score:
            best_score = sc_final
            best_seq = a_final.tolist()
            print(f"[{time.time():.1f}] SA improved best = {best_score:.6f}")

    # -------------------------------------------------------------
    # 6. Safety fallback (should never be needed)
    # -------------------------------------------------------------
    if best_seq is None:
        n_small = 256
        best_seq = [1.0 / n_small] * n_small
        best_score = evaluate_sequence(best_seq)

    # -------------------------------------------------------------
    # Final evaluation (the external harness will also call it)
    # -------------------------------------------------------------
    final_score = evaluate_sequence(best_seq)
    print(f"\n[{time.time():.1f}] finished – final reported score = {final_score:.6f}")

    return best_seq
```
