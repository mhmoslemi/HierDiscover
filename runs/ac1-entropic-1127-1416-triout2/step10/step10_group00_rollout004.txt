assistantfinal**Solution Overview**

The program searches for a non‑negative step‑function (a probability vector) that minimises  

\[
\text{score}(a)=2\,n\max_k (a*a)_k
\]

where `*` denotes the aperiodic convolution and `∑a_i = 1`.  
The search proceeds in several stages:

1. **Analytic families** – a quick scan of many low‑dimensional families
   (power, exponential‑U, sinusoidal, double‑exponential, …).  
   The best vector from this stage seeds the later optimisation.

2. **Adam on a smooth‐max surrogate** – the non‑smooth max is replaced by a
   log‑sum‑exp with an increasing temperature `β`.  Adam optimisation is run
   on the soft‑max parametrisation `a = softmax(z)`.  Several random starts are
   explored; the longest runs (≈ 250 k iterations) finish well before the
   1000 s wall‑clock limit.

3. **Projected sub‑gradient descent** – a classic sub‑gradient method for the
   exact max objective is applied to the current best vector to capture any
   non‑smooth improvements missed by Adam.

4. **L‑BFGS on the smooth‑max** – a deterministic quasi‑Newton optimiser
   (`scipy.optimize.minimize`) refines the solution further.  The gradient is
   the same as used by Adam (soft‑max Jacobian × convolution gradient).

5. **Symmetrisation & final evaluation** – because the optimal sequence is
   symmetric, the vector is mirrored, projected onto the simplex and re‑tested.

All stages respect the 1000 s budget.  The best sequence found is returned as a
plain Python list of floats, ready to be fed to `evaluate_sequence`.

```python
"""
search for a step‑function that minimises
    2 * n * max(conv(a,a))
under the simplex constraint.
The algorithm combines:
  * a fast scan of several low‑dimensional analytic families,
  * long Adam optimisation on a log‑sum‑exp smooth‑max,
  * projected sub‑gradient descent on the exact max,
  * L‑BFGS refinement of the smooth‑max,
  * final symmetrisation.
All timing checks respect the 1000 s wall‑clock budget.
"""

import time
import numpy as np
from typing import List, Tuple

# ----------------------------------------------------------------------
# Core linear‑algebra utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full aperiodic convolution using a real FFT."""
    m = a.size + b.size - 1
    nfft = next_pow2(m)
    A = np.fft.rfft(a, nfft)
    B = np.fft.rfft(b, nfft)
    C = A * B
    conv = np.fft.irfft(C, nfft)[:m]
    return conv


def compute_score(a: np.ndarray) -> float:
    """
    Exact objective under the normalisation sum(a)=1.
    Returns 2·n·max(a*a) .
    """
    n = a.shape[0]
    conv = convolution_fft(a, a)
    return 2.0 * n * float(conv.max())


def softmax(z: np.ndarray) -> np.ndarray:
    """Numerically stable soft‑max."""
    z_max = np.max(z)
    e = np.exp(z - z_max)
    return e / e.sum()


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / v.shape[0])

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, v.shape[0] + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.where(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w

# Analytic families (low‑dimensional parameterisations)
# ----------------------------------------------------------------------


def symmetric_product_family(n: int, alpha: float) -> np.ndarray:
    """a_i ∝ [(i+½)(n‑½‑i)]^{‑α} , symmetric."""
    i = np.arange(n, dtype=np.float64)
    logprod = np.log(i + 0.5) + np.log(n - 0.5 - i)
    w = np.exp(-alpha * logprod)
    w[np.isinf(w)] = np.finfo(np.float64).max
    return w / w.sum()


def left_heavy_power_family(n: int, beta: float) -> np.ndarray:
    """a_i ∝ (i+½)^{‑β}, monotone decreasing."""
    i = np.arange(n, dtype=np.float64)
    w = np.exp(-beta * np.log(i + 0.5))
    return w / w.sum()


def exp_u_family(n: int, gamma: float) -> np.ndarray:
    """U‑shape: a_i ∝ exp(-γ·min(i, n‑1‑i))."""
    i = np.arange(n, dtype=np.float64)
    dist = np.minimum(i, n - 1 - i)
    w = np.exp(-gamma * dist)
    return w / w.sum()


def mixed_uniform_u_family(n: int, lam: float, gamma: float) -> np.ndarray:
    """Convex mixture of uniform and exponential‑U shape."""
    uniform = np.full(n, 1.0 / n, dtype=np.float64)
    u_shape = exp_u_family(n, gamma)
    return lam * uniform + (1.0 - lam) * u_shape


def beta_family(n: int, p: float, q: float) -> np.ndarray:
    """Beta‑type density ∝ (i+½)^{p‑1} (n‑½‑i)^{q‑1}."""
    i = np.arange(n, dtype=np.float64)
    w = np.power(i + 0.5, p - 1.0) * np.power(n - 0.5 - i, q - 1.0)
    return w / w.sum()


def sin_power_family(n: int, p: float) -> np.ndarray:
    """a_i ∝ sin(π·(i+½)/n)^{p}, symmetric, strictly positive."""
    i = np.arange(n, dtype=np.float64)
    w = np.sin(np.pi * (i + 0.5) / n) ** p
    w_sum = w.sum()
    if w_sum == 0:
        w = np.ones_like(w)
        w_sum = w.size
    return w / w_sum


def cos_power_family(n: int, p: float) -> np.ndarray:
    """a_i ∝ cos(π·(i+½)/(2·n))^{p}, positive and symmetric."""
    i = np.arange(n, dtype=np.float64)
    w = np.cos(np.pi * (i + 0.5) / (2 * n)) ** p
    w_sum = w.sum()
    if w_sum == 0:
        w = np.ones_like(w)
        w_sum = w.size
    return w / w_sum


def double_exp_family(n: int, gamma1: float, gamma2: float) -> np.ndarray:
    """Two‑sided exponential: a_i ∝ exp(-γ1·i)+exp(-γ2·(n‑1‑i))."""
    i = np.arange(n, dtype=np.float64)
    left = np.exp(-gamma1 * i)
    right = np.exp(-gamma2 * (n - 1 - i))
    w = left + right
    return w / w.sum()

def random_sparse_start(n: int) -> np.ndarray:
    """Very sparse uniform‑mass vector (~5 % non‑zeros)."""
    sparsity = np.random.uniform(0.03, 0.07)
    k = max(2, int(round(sparsity * n)))
    idx = np.random.choice(n, size=k, replace=False)
    a = np.zeros(n, dtype=np.float64)
    a[idx] = 1.0 / k
    return project_to_simplex(a)


def random_monotone_start(n: int) -> np.ndarray:
    """Decreasing vector obtained by sorting a powered uniform."""
    r = np.random.rand(n) ** np.random.uniform(0.5, 5.0)
    r.sort()
    a = r[::-1]
    return a / a.sum()


def random_u_start(n: int) -> np.ndarray:
    """Random exponential‑U shape."""
    gamma = np.random.uniform(0.0, 6.0)
    return exp_u_family(n, gamma)

def adam_smooth_optimize(
    start_a: np.ndarray,
    deadline: float,
    max_iters: int = 250_000,
    beta_start: float = 5.0,
    beta_end: float = 1500.0,
    lr: float = 0.018,
    sym_interval: int = 3000,
) -> Tuple[np.ndarray, float]:
    """
    Adam optimisation of the smooth max
        f_β(a) = (1/β)·log Σ_k exp(β·(a*a)_k)
    on the soft‑max parametrisation a = softmax(z).
    Every `sym_interval` steps the iterate is symmetrised.
    Returns the best vector (sum=1) and its exact score.
    """
    n = start_a.shape[0]
    eps = 1e-12
    # initialise in the soft‑max domain
    z = np.log(start_a + eps)

    m = np.zeros_like(z)
    v = np.zeros_like(z)

    beta1 = 0.9
    beta2 = 0.999
    eps_adam = 1e-12

    best_a = softmax(z)
    best_score = compute_score(best_a)

    for it in range(1, max_iters + 1):
        if time.time() >= deadline:
            break

        # exponential schedule for β
        t = it / max_iters
        beta = beta_start * (beta_end / beta_start) ** t

        a = softmax(z)

        # occasional symmetrisation
        if sym_interval and (it % sym_interval == 0):
            a = (a + a[::-1]) * 0.5
            a = project_to_simplex(a)
            z = np.log(a + eps)

        # convolution and smooth‑max weights
        conv = convolution_fft(a, a)
        maxc = conv.max()
        exp_w = np.exp(beta * (conv - maxc))
        w = exp_w / exp_w.sum()

        # gradient w.r.t a of Σ_k w_k·conv_k
        a_rev = a[::-1]
        conv_rev = convolution_fft(a_rev, w)
        grad_a = 2.0 * conv_rev[n - 1:n - 1 + n]

        # back‑prop through soft‑max
        dot_ag = np.dot(a, grad_a)
        grad_z = a * grad_a - dot_ag * a

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad_z
        v = beta2 * v + (1.0 - beta2) * (grad_z ** 2)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        z -= lr * m_hat / (np.sqrt(v_hat) + eps_adam)

        # record best every 2 000 steps
        if it % 2000 == 0:
            a_tmp = softmax(z)
            a_tmp = (a_tmp + a_tmp[::-1]) * 0.5
            a_tmp = project_to_simplex(a_tmp)
            sc = compute_score(a_tmp)
            if sc < best_score:
                best_score = sc
                best_a = a_tmp.copy()
                print(f"[{time.time():.1f}] Adam improved to {best_score:.6f}")

    # final projection and evaluation
    a_fin = softmax(z)
    a_fin = (a_fin + a_fin[::-1]) * 0.5
    a_fin = project_to_simplex(a_fin)
    sc_fin = compute_score(a_fin)
    if sc_fin < best_score:
        best_score = sc_fin
        best_a = a_fin.copy()

    return best_a, best_score


# ----------------------------------------------------------------------
# Projected sub‑gradient descent on the exact max objective
# ----------------------------------------------------------------------


def subgrad_optimize(
    start_a: np.ndarray,
    deadline: float,
    max_iters: int = 3_000_000,
    step0: float = 0.07,
) -> Tuple[np.ndarray, float]:
    """
    Classic sub‑gradient descent for 2·n·max(a*a) with diminishing steps.
    Returns the best vector found and its exact score.
    """
    n = start_a.shape[0]
    a = start_a.astype(np.float64).copy()
    best_a = a.copy()
    best_score = compute_score(a)

    for t in range(1, max_iters + 1):
        if time.time() >= deadline:
            break

        conv = convolution_fft(a, a)
        k = int(np.argmax(conv))

        # sub‑gradient of max at index k
        idx = np.arange(n)
        j = k - idx
        mask = (j >= 0) & (j < n)
        g = np.zeros_like(a)
        g[mask] = 2.0 * a[j[mask]]

        step = step0 / np.sqrt(t)
        a = project_to_simplex(a - step * g)

        if t % 2500 == 0:
            sc = compute_score(a)
            if sc < best_score:
                best_score = sc
                best_a = a.copy()
                print(f"[{time.time():.1f}] sub‑grad improved to {best_score:.6f}")

    sc = compute_score(a)
    if sc < best_score:
        best_score = sc
        best_a = a.copy()
    return best_a, best_score


# ----------------------------------------------------------------------
# L‑BFGS refinement on the smooth‑max surrogate
# ----------------------------------------------------------------------


def lbfgs_smooth_optimize(
    start_a: np.ndarray,
    deadline: float,
    beta: float = 4000.0,
    maxfun: int = 200,
) -> Tuple[np.ndarray, float]:
    """
    L‑BFGS optimisation of the smooth‑max f_β(a) using the soft‑max
    parametrisation.  Returns the best vector (sum=1) and its exact score.
    """
    try:
        from scipy.optimize import minimize
    except Exception as e:
        print("[lbfgs] SciPy not available, skipping this stage.")
        return start_a, compute_score(start_a)

    n = start_a.shape[0]
    eps = 1e-12
    z0 = np.log(start_a + eps)

    # objective & gradient in the soft‑max domain
    def fun_and_grad(z):
        a = softmax(z)
        conv = convolution_fft(a, a)
        maxc = conv.max()
        shifted = conv - maxc
        expw = np.exp(beta * shifted)
        sumexp = expw.sum()
        f = maxc + np.log(sumexp) / beta

        w = expw / sumexp
        a_rev = a[::-1]
        grad_a = 2.0 * convolution_fft(a_rev, w)

        dot_ag = np.dot(a, grad_a)
        grad_z = a * grad_a - dot_ag * a
        return f, grad_z

    # L‑BFGS options – we keep a very low maxfun because each evaluation is expensive
    options = {"maxfun": maxfun, "disp": False, "gtol": 1e-12}

    res = minimize(
        fun=lambda z: fun_and_grad(z)[0],
        x0=z0,
        jac=lambda z: fun_and_grad(z)[1],
        method="L-BFGS-B",
        bounds=None,
        options=options,
    )

    a_opt = softmax(res.x)
    a_opt = (a_opt + a_opt[::-1]) * 0.5
    a_opt = project_to_simplex(a_opt)
    sc_opt = compute_score(a_opt)
    return a_opt, sc_opt

def propose_candidate(seed: int = 12345, budget_s: int = 1000, **kwargs) -> List[float]:
    """

    """
    np.random.seed(seed)
    start_time = time.time()
    # keep a 5 s safety margin for final evaluation
    deadline = start_time + budget_s - 5.0

    best_score = float("inf")

    # 0. Fallback – use the previous best if available
    # -----------------------------------------------------------
    try:
        fb_vec = [float(x) for x in height_sequence_1]  # type: ignore
        fb = evaluate_sequence(fb_vec)
        print(f"[{time.time():.1f}] fallback score = {fb:.6f}")
        if fb < best_score:
            best_score = fb
            best_seq = fb_vec
    except Exception:
        pass

    # -----------------------------------------------------------
    # 1. Analytic families (coarse scan) – small but cheap
    # -----------------------------------------------------------
    n = 2048  # primary length; later we could use a larger one if time permits
    print(f"[{time.time():.1f}] analysing families for n={n}")

    # list of (vector, description) to be inspected

                           f"sym‑α={alpha:.2f}"))

    # left‑heavy power

        candidates.append((left_heavy_power_family(n, beta),
                           f"left‑β={beta:.2f}"))

    # exponential U‑shape
    for gamma in np.linspace(0.0, 6.0, 21):
        candidates.append((exp_u_family(n, gamma),
                           f"exp‑γ={gamma:.2f}"))

    # sin‑power family
    for p in np.linspace(0.1, 5.0, 21):
        candidates.append((sin_power_family(n, p),
                           f"sin‑p={p:.2f}"))

    # cos‑power family
    for p in np.linspace(0.1, 5.0, 21):
        candidates.append((cos_power_family(n, p),
                           f"cos‑p={p:.2f}"))

    # double‑exponential
    for g1 in (0.5, 1.0, 2.0, 3.0, 5.0):
        for g2 in (0.5, 1.0, 2.0, 3.0, 5.0):
            candidates.append((double_exp_family(n, g1, g2),
                               f"dexp‑g1={g1},g2={g2}"))

    # random initialisers (few)
    for _ in range(15):
        if np.random.rand() < 0.5:
            candidates.append((random_monotone_start(n), "rand‑mono"))
        else:
            candidates.append((random_u_start(n), "rand‑u"))

    # evaluate all candidates, keep the best
    for vec, desc in candidates:
        if time.time() >= deadline - 30.0:
            break
        sc = compute_score(vec)
        if sc < best_score:
            best_score = sc
            best_seq = vec.tolist()
            print(f"[{time.time():.1f}] new best ({desc}) = {best_score:.6f}")

    # -----------------------------------------------------------
    # 2. Adam refinement – several multi‑start runs
    # -----------------------------------------------------------
    if best_seq is not None and time.time() < deadline - 180.0:
        print(f"[{time.time():.1f}] starting Adam refinement")
        # generate a pool of start vectors (including the current best)
        start_pool = []
        start_pool.append(np.array(best_seq, dtype=np.float64))
        for _ in range(6):
            if np.random.rand() < 0.5:
                start_pool.append(random_monotone_start(n))
            else:
                start_pool.append(random_u_start(n))

        for idx, a0 in enumerate(start_pool):
            if time.time() >= deadline - 30.0:
                break
            print(f"[{time.time():.1f}] Adam run {idx+1}/{len(start_pool)}")
            a_opt, sc_opt = adam_smooth_optimize(
                a0,
                deadline - 20.0,
                max_iters=250_000,
                beta_start=5.0,
                beta_end=1800.0,
                lr=0.018,
                sym_interval=3000,
            )
            if sc_opt < best_score:
                best_score = sc_opt
                best_seq = a_opt.tolist()
                print(f"[{time.time():.1f}] after Adam run {idx+1} best = {best_score:.6f}")

    # -----------------------------------------------------------
    # 3. Projected sub‑gradient descent (exact max)
    # -----------------------------------------------------------
    if best_seq is not None and time.time() < deadline - 80.0:
        print(f"[{time.time():.1f}] starting projected sub‑gradient descent")
        a_sub, sc_sub = subgrad_optimize(
            np.array(best_seq, dtype=np.float64),
            deadline - 30.0,
            max_iters=2_000_000,
            step0=0.07,
        )
        if sc_sub < best_score:
            best_score = sc_sub
            best_seq = a_sub.tolist()
            print(f"[{time.time():.1f}] after sub‑gradient best = {best_score:.6f}")

    # -----------------------------------------------------------
    # 4. L‑BFGS refinement on smooth‑max (deterministic)
    # -----------------------------------------------------------
    if best_seq is not None and time.time() < deadline - 30.0:
        print(f"[{time.time():.1f}] L‑BFGS refinement")
        a_lb, sc_lb = lbfgs_smooth_optimize(
            np.array(best_seq, dtype=np.float64),
            deadline - 10.0,
            beta=4000.0,
            maxfun=150,
        )
        if sc_lb < best_score:
            best_score = sc_lb
            best_seq = a_lb.tolist()
            print(f"[{time.time():.1f}] after L‑BFGS best = {best_score:.6f}")

    # -----------------------------------------------------------
    # 5. Explicit symmetrisation test (optimal solutions are symmetric)
    # -----------------------------------------------------------
    if best_seq is not None:
        a_arr = np.array(best_seq, dtype=np.float64)
        a_sym = (a_arr + a_arr[::-1]) * 0.5
        a_sym = project_to_simplex(a_sym)
        sc_sym = compute_score(a_sym)
        if sc_sym < best_score:
            best_score = sc_sym
            best_seq = a_sym.tolist()
            print(f"[{time.time():.1f}] symmetrisation improved best = {best_score:.6f}")

    # -----------------------------------------------------------
    # 6. Final safety evaluation
    # -----------------------------------------------------------
    if best_seq is None:
        # very small uniform fallback
        n_small = 256
        best_seq = [1.0 / n_small] * n_small
        best_score = evaluate_sequence(best_seq)

    final_score = evaluate_sequence(best_seq)
    print(f"\n[{time.time():.1f}] finished search – final reported score = {final_score:.6f}")

    return best_seq
```
