def simulated_annealing(a, start_temp, end_temp, deadline):
    n = len(a)
    cur = a.copy()
    cur_score = compute_score(cur)
    best = cur.copy()
    best_score = cur_score
    start_time = time.time()
    while time.time() < deadline:
        # progress fraction
        elapsed = time.time() - start_time
        fraction = elapsed / (deadline - start_time)
        # temperature linearly decays
        T = start_temp * (1 - fraction) + end_temp * fraction
        # propose move
        i, j = np.random.randint(n, size=2)
        if i == j:
            continue
        # delta proportion: sample from uniform [0,1] * T * cur[i]
        delta = cur[i] * T * np.random.rand()
        if delta > cur[i]:
            delta = cur[i]
        if delta < 1e-12:
            continue
        cand = cur.copy()
        cand[i] -= delta
        cand[j] += delta
        # normalize just in case (drift)
        cand /= cand.sum()
        cand_score = compute_score(cand)
        if cand_score < cur_score or np.random.rand() < np.exp(-(cand_score - cur_score) / max(T, 1e-12)):
            cur = cand
            cur_score = cand_score
            if cur_score < best_score:
                best = cur.copy()
                best_score = cur_score
                # maybe increase step size? not needed
        # optionally sometimes random restart to keep diversity
    return best, best_score

"""
Advanced optimizer for the self‑convolution L∞ problem.

The algorithm has three stages within the 1000 s budget:

1. **Seeding** – a diverse collection of hand‑crafted shapes (sin²,
   beta‑family, sparse spikes) together with a few Dirichlet draws.
   The best seed (exact objective) is kept.

2. **Soft‑max gradient descent** – minimises the smooth surrogate
   Lβ(a)=log∑exp(β·conv(a,a)) using projected gradient descent with
   momentum.  The temperature β is annealed from 5 to 5000, the step
   size decays with √iteration and β, and after every update the
   vector is projected onto the probability simplex and forced to be
   symmetric (a = a[::-1]).

3. **Simulated‑annealing polishing** – a long stochastic search that
   repeatedly transfers a random amount of mass from one index to
   another (or swaps two entries).  Moves are accepted if they improve
   the true objective or with a Metropolis probability that decays
   linearly from a high temperature (≈30 % of the mass may be moved)
   to a tiny temperature.  The best vector found during this phase
   is kept.

All phases respect the wall‑clock limit; the final vector is returned
as a plain list of floats (non‑negative, normalised – the scaling is
irrelevant for `evaluate_sequence`).  The code is fully self‑contained
and uses only NumPy and the standard library.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Low‑level utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full linear convolution using a real FFT (real‑valued inputs)."""
    m = a.shape[0] + b.shape[0] - 1
    n_fft = next_pow2(m)
    A = np.fft.rfft(a, n=n_fft)
    B = np.fft.rfft(b, n=n_fft)
    C = A * B
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


def compute_score(a: np.ndarray) -> float:
    """
    Exact objective for a probability vector (sum = 1):
        2·n·max(conv(a,a)).
    """
    n = a.shape[0]
    conv = convolution_fft(a, a)
    return 2.0 * n * float(conv.max())


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of `v` onto the probability simplex
    {x | x_i ≥ 0, Σ x_i = 1}.
    """
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / v.shape[0])

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, v.shape[0] + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.where(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def symmetrize(v: np.ndarray) -> np.ndarray:
    """Force symmetry: v = (v + v[::-1]) / 2."""
    return (v + v[::-1]) * 0.5


# ----------------------------------------------------------------------
# Seed generators
# ----------------------------------------------------------------------


def generate_deterministic(n: int) -> np.ndarray:
    """Sin² shape a_i = sin²(pi·i/(n+1)), normalised to sum 1."""
    i = np.arange(1, n + 1, dtype=float)
    a = np.sin(np.pi * i / (n + 1.0)) ** 2
    a /= a.sum()
    return a


def generate_beta_family(n: int, p: float) -> np.ndarray:
    """
    Symmetric polynomial shape a_i ∝ ((i+1)*(n-i))**p.
    p>0 clusters mass in the centre, p<0 pushes it to the edges.
    """
    i = np.arange(n, dtype=float)
    w = ((i + 1.0) * (n - i)) ** p
    w[w < 0] = 0.0
    a = w / w.sum()
    return a


def random_dirichlet(n: int, conc: float = 1.0) -> np.ndarray:
    """Draw a random probability vector from Dirichlet(conc)."""
    if conc == 1.0:
        a = np.random.rand(n)
    else:
        a = np.random.gamma(conc, 1.0, size=n)
    a /= a.sum()
    return a


def random_spike(n: int, k: int = 70) -> np.ndarray:
    """Very sparse vector: choose `k` distinct positions, assign random weights."""
    idx = np.random.choice(n, size=min(k, n), replace=False)
    w = np.random.rand(len(idx))
    a = np.zeros(n, dtype=float)
    a[idx] = w
    a /= a.sum()
    return a


# ----------------------------------------------------------------------
# Phase 2 – soft‑max gradient descent with momentum
# ----------------------------------------------------------------------


def softmax_gradient_descent(
    a: np.ndarray,
    start_score: float,
    deadline: float,
    beta_start: float = 5.0,
    beta_end: float = 5000.0,
    step_coef: float = 0.0015,
    momentum: float = 0.9,
) -> tuple[np.ndarray, float]:
    """
    Projected gradient descent on the log‑sum‑exp surrogate.
    Uses a simple momentum term and enforces symmetry after each step.
    """
    n = a.shape[0]
    cur_a = a.copy()
    cur_score = start_score
    best_a = cur_a.copy()
    best_score = cur_score

    velocity = np.zeros_like(cur_a)

    start = time.time()
    it = 0
    while time.time() < deadline:
        # exact objective (for bookkeeping)
        conv = convolution_fft(cur_a, cur_a)
        max_conv = conv.max()
        cur_score = 2.0 * n * float(max_conv)
        if cur_score < best_score:
            best_score = cur_score
            best_a = cur_a.copy()

        # temperature schedule (linear in wall‑clock fraction)
        elapsed = time.time() - start
        fraction = elapsed / (deadline - start)
        beta = beta_start + fraction * (beta_end - beta_start)

        # soft‑max weights
        shifted = conv - max_conv          # max(conv) becomes 0
        exp_vals = np.exp(beta * shifted)  # safe: all ≤ 1
        w = exp_vals / exp_vals.sum()

        # gradient of surrogate
        grad = 2.0 * beta * np.correlate(w, cur_a, mode="valid")

        # step size (decays with √iteration and with β)
        it += 1
        step = step_coef / (np.sqrt(it) * np.sqrt(beta))

        # momentum update
        velocity = momentum * velocity - step * grad
        nxt = cur_a + velocity

        # simplex projection and symmetry enforcement
        nxt = project_to_simplex(nxt)
        nxt = symmetrize(nxt)

        cur_a = nxt

        # occasional progress report
        if it % 4000 == 0:
            print(
                f"[GD] it {it:6d} | time {elapsed:6.1f}s | best {best_score:.6f} | β {beta:.1f}"
            )

    return best_a, best_score


# ----------------------------------------------------------------------
# Phase 3 – simulated‑annealing polishing
# ----------------------------------------------------------------------


def simulated_annealing(
    a: np.ndarray,
    start_temp: float,
    end_temp: float,
    deadline: float,
) -> tuple[np.ndarray, float]:
    """
    Stochastic mass‑transfer / swap moves with a linearly decaying temperature.
    The Metropolis acceptance rule allows occasional uphill moves early on.
    """
    n = a.shape[0]
    cur = a.copy()
    cur_score = compute_score(cur)
    best = cur.copy()
    best_score = cur_score

    start = time.time()
    steps = 0
    while time.time() < deadline:
        # linearly decaying temperature
        elapsed = time.time() - start
        fraction = elapsed / (deadline - start)
        T = start_temp * (1.0 - fraction) + end_temp * fraction

        # propose a move
        i, j = np.random.randint(n, size=2)
        cand = cur.copy()

        if np.random.rand() < 0.5:
            # simple swap – useful for disrupting pathological patterns
            cand[i], cand[j] = cand[j], cand[i]
        else:
            # transfer a random fraction of the mass at i to j
            if cand[i] > 0.0:
                # delta is at most T·cand[i]; T∈[0,1] scales the aggressiveness
                delta = cand[i] * T * np.random.rand()
                cand[i] -= delta
                cand[j] += delta

        # re‑normalise (numerical drift) and enforce symmetry
        cand = project_to_simplex(cand)
        cand = symmetrize(cand)

        cand_score = compute_score(cand)

        # Metropolis acceptance
        if cand_score < cur_score or np.random.rand() < np.exp(-(cand_score - cur_score) / max(T, 1e-12)):
            cur = cand
            cur_score = cand_score
            if cur_score < best_score:
                best = cur.copy()
                best_score = cur_score

        steps += 1
        # occasional status line
        if steps % 200_000 == 0:
            print(
                f"[SA] step {steps:,d} | time {elapsed:6.1f}s | best {best_score:.6f} | T {T:.5f}"
            )

    return best, best_score


# ----------------------------------------------------------------------
# Optional tiny hill‑climb (fallback for the last few seconds)
# ----------------------------------------------------------------------


def hill_climb(
    a: np.ndarray,
    deadline: float,
    max_steps: int = 300_000,
) -> tuple[np.ndarray, float]:
    """Very simple stochastic hill‑climb kept for the final seconds."""
    n = a.shape[0]
    best = a.copy()
    best_score = compute_score(best)
    step_scale = 0.08
    steps = 0
    while steps < max_steps and time.time() < deadline:
        cand = best.copy()
        if np.random.rand() < 0.5:
            i, j = np.random.randint(n, size=2)
            cand[i], cand[j] = cand[j], cand[i]
        else:
            i, j = np.random.randint(n, size=2)
            if cand[i] > 0.0 and i != j:
                delta = step_scale * cand[i]
                cand[i] -= delta
                cand[j] += delta
                cand[i] = max(cand[i], 0.0)

        cand = project_to_simplex(cand)
        cand = symmetrize(cand)
        cur_score = compute_score(cand)

        if cur_score < best_score:
            best = cand
            best_score = cur_score
            step_scale = min(step_scale * 1.07, 0.5)
        else:
            step_scale *= 0.96

        steps += 1
    return best, best_score
