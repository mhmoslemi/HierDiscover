def solve_lp_max_sum(f, M):
    n = f.shape[0]
    L = 2*n - 1
    # build A
    A = np.zeros((L, n), dtype=np.float64)
    for k in range(L):

        i_end = min(k, n-1)
        # f[i] corresponds to g[j] with j = k-i
        # So for i from i_start to i_end:
        #   j = k - i
        #   A[k, j] = f[i]
        # vectorized: indices = np.arange(i_start,i_end+1); j = k - indices
        #   assign
        idx = np.arange(i_start, i_end+1)
        A[k, k - idx] = f[idx]
    # objective: minimize -sum(g) i.e., maximize sum(g)
    c = -np.ones(n)
    b = np.full(L, M)
    bounds = [(0, None)] * n
    res = linprog(c, A_ub=A, b_ub=b, bounds=bounds, method='highs')
    if not res.success:
        print("LP failed:", res.message)
        return None
    return res.x

"""
Propose a step‑function (non‑negative list of floats) that minimises the
objective used by ``evaluate_sequence``.

The objective is   2·n·max(conv(a,a)) / (sum(a)^2) .
Because the expression is invariant under scaling of the vector, we can
restrict ourselves to probability vectors (sum = 1).  The problem then
becomes:  minimise the ℓ∞‑norm of the convolution (i.e. the maximal
probability of the sum of two independent draws).

The code follows a three‑stage recipe:

1. **Exploration of analytic families** – Beta‑type densities, symmetric
   power families (including the “interior‑peaked’’ case) and a few
   random Dirichlet draws are evaluated very cheaply.

2. **Linear‑programming “active‑set’’ step** – Starting from the best
   candidate of stage 1 we repeatedly solve the LP
       maximise   1ᵀg
       subject to  f * g ≤ M   and   g ≥ 0                      (1)
   where ``f`` is the current distribution and ``M`` the current
   maximum convolution value.  The LP pushes mass away from positions
   that would increase the convolution peak.  The new candidate ``g``
   is mixed with ``f`` (convex combination with a decreasing step size)
   and the process is iterated until the time budget is exhausted.

3. **Fine‑tuning** – A short entropy‑mirror sub‑gradient descent on the
   true max‑convolution (the same routine used in the reference code)
   and a final tiny simulated‑annealing hill‑climb are performed.

All intermediate improvements are printed, the best vector is always
kept (as a plain Python list) and finally returned.  The routine obeys
the 1000 s wall‑clock limit imposed by the harness.
"""

import time
import numpy as np
from scipy.optimize import linprog

# ----------------------------------------------------------------------
# Basic utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full aperiodic convolution using real FFT (fast, no Python loops)."""
    m = a.size + b.size - 1
    nfft = next_pow2(m)
    A = np.fft.rfft(a, n=nfft)
    B = np.fft.rfft(b, n=nfft)
    C = A * B
    conv = np.fft.irfft(C, n=nfft)[:m]
    return conv


def compute_score(a: np.ndarray) -> float:
    """
    Exact objective for a probability vector (sum(a) = 1):
        2·n·max(a * a)
    This is the quantity used internally for quick comparison.
    """
    n = a.shape[0]
    conv = convolution_fft(a, a)
    return 2.0 * n * float(conv.max())


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection onto the probability simplex (sum = 1, v ≥ 0)."""
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / v.shape[0])

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, v.shape[0] + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.where(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


# ----------------------------------------------------------------------
# Analytic families (more diverse than the reference code)
# ----------------------------------------------------------------------


def symmetric_product_family(n: int, alpha: float) -> np.ndarray:
    """a_i ∝ [(i+½)(n‑½‑i)]^(‑α)  (U‑shaped for α>0, flat for α=0)."""
    i = np.arange(n, dtype=np.float64)
    logprod = np.log(i + 0.5) + np.log(n - 0.5 - i)
    w = np.exp(-alpha * logprod)
    w[np.isinf(w)] = np.finfo(np.float64).max
    a = w / w.sum()
    return a


def interior_power_family(n: int, gamma: float) -> np.ndarray:
    """a_i ∝ [(i+½)(n‑½‑i)]^(+γ)  – interior‑peaked shapes (γ>0)."""
    i = np.arange(n, dtype=np.float64)
    logprod = np.log(i + 0.5) + np.log(n - 0.5 - i)
    w = np.exp(gamma * logprod)        # same as product**gamma
    w[np.isinf(w)] = np.finfo(np.float64).max
    a = w / w.sum()
    return a


def beta_family(n: int, alpha: float, beta: float) -> np.ndarray:
    """
    Discrete Beta(α,β) density on {0,…,n‑1}:
        a_i ∝ ((i+½)/n)^(α‑1) * (1-(i+½)/n)^(β‑1)
    """
    i = np.arange(n, dtype=np.float64) + 0.5
    x = i / float(n)
    # avoid 0⁰ or 1⁰ problems by clamping
    eps = 1e-12
    x = np.clip(x, eps, 1 - eps)
    w = np.power(x, alpha - 1.0) * np.power(1.0 - x, beta - 1.0)
    w = np.nan_to_num(w, nan=0.0, posinf=0.0, neginf=0.0)
    a = w / w.sum()
    return a


def random_dirichlet_start(n: int, conc: float) -> np.ndarray:
    """Draw from a symmetric Dirichlet(α) distribution (α = conc)."""
    vec = np.random.gamma(conc, 1.0, size=n)
    vec = vec / vec.sum()
    return vec


def random_sparse_start(n: int) -> np.ndarray:
    """Very sparse uniform vector (≈ 5 % non‑zeros)."""
    sparsity = np.random.uniform(0.03, 0.07)
    k = max(2, int(round(sparsity * n)))
    idx = np.random.choice(n, size=k, replace=False)
    a = np.zeros(n, dtype=np.float64)
    a[idx] = 1.0 / k
    return project_to_simplex(a)


# ----------------------------------------------------------------------
# Linear‑programming optimisation step (solve (1))
# ----------------------------------------------------------------------


def build_conv_matrix(f: np.ndarray) -> np.ndarray:
    """Return the matrix A such that A @ g = conv(f,g) (full convolution)."""
    n = f.shape[0]
    L = 2 * n - 1
    A = np.zeros((L, n), dtype=np.float64)
    # each row k contains the coefficients f[i] that multiply g[k-i]
    for k in range(L):
        i_start = max(0, k - (n - 1))
        i_end = min(k, n - 1)
        idx = np.arange(i_start, i_end + 1, dtype=int)
        A[k, k - idx] = f[idx]
    return A


def lp_maximize_sum_under_conv_bound(f: np.ndarray, M: float):
    """
    Solve the LP:
        maximise   Σ g_i
        subject to  f * g ≤ M   (element‑wise)   and   g ≥ 0 .
    Returns the optimal g (as a numpy array) or ``None`` on failure.
    """
    n = f.shape[0]
    A = build_conv_matrix(f)                 # shape (2n‑1, n)
    b = np.full(2 * n - 1, M, dtype=np.float64)

    # linprog minimises; maximise Σ g_i  ⇒ minimise -Σ g_i
    c = -np.ones(n, dtype=np.float64)

    bounds = [(0, None)] * n
    res = linprog(c, A_ub=A, b_ub=b, bounds=bounds, method="highs", options={"presolve": True})
    if not res.success:
        print(f"   [LP] failed – {res.message}")
        return None
    return res.x


# ----------------------------------------------------------------------
# Mirror (entropy) sub‑gradient descent on the exact max‑objective
# ----------------------------------------------------------------------


def mirror_subgrad_optimize(
    start_a: np.ndarray,
    deadline: float,
    step0: float = 0.9,
    check_interval: int = 2000,
) -> tuple[np.ndarray, float]:
    """Entropy‑mirror sub‑gradient descent for 2·n·max(a*a)."""
    n = start_a.shape[0]
    a = start_a.astype(np.float64).copy()
    best_a = a.copy()
    best_score = compute_score(a)

    t = 0
    while time.time() < deadline:
        t += 1
        conv = convolution_fft(a, a)
        k = int(np.argmax(conv))                     # active shift index

        # sub‑gradient for the active shift (ordered pairs)
        idx = np.arange(n)
        j = k - idx
        mask = (j >= 0) & (j < n)
        g = np.zeros_like(a)
        g[mask] = 2.0 * a[j[mask]]

        eta = step0 / np.sqrt(t)
        a = a * np.exp(-eta * g)                     # multiplicative update
        s = a.sum()
        if s == 0.0:
            a = np.full_like(a, 1.0 / n)
        else:
            a = a / s

        if t % check_interval == 0:
            sc = compute_score(a)
            if sc < best_score:
                best_score = sc
                best_a = a.copy()
                print(f"[{time.time():.1f}]   Mirror step {t:7d} – new best = {best_score:.6f}")

    sc = compute_score(a)
    if sc < best_score:
        best_score = sc
        best_a = a.copy()
    return best_a, best_score


# ----------------------------------------------------------------------
# Tiny simulated‑annealing hill‑climb (final polishing)
# ----------------------------------------------------------------------


def hill_climb_sa(
    start_a: np.ndarray,
    deadline: float,
    max_steps: int = 1_500_000,
    init_temp_factor: float = 1e-4,
) -> tuple[np.ndarray, float]:
    """Very low‑temperature simulated‑annealing walk on the simplex."""
    n = start_a.size
    cur_a = start_a.copy()
    cur_score = compute_score(cur_a)

    best_a = cur_a.copy()
    best_score = cur_score

    T0 = best_score * init_temp_factor
    T = T0
    step_scale = 0.12

    for step in range(1, max_steps + 1):
        if time.time() >= deadline:
            break

        cand = cur_a.copy()
        if np.random.rand() < 0.5:                       # swap two entries
            i, j = np.random.randint(n, size=2)
            cand[i], cand[j] = cand[j], cand[i]
        else:                                            # transfer a fraction
            i, j = np.random.randint(n, size=2)
            if i != j:
                delta = step_scale * cur_a[i]
                cand[i] -= delta
                cand[j] += delta
                if cand[i] < 0.0:
                    cand[i] = 0.0

        cand = cand / cand.sum()
        cand_score = compute_score(cand)
        delta = cand_score - cur_score

        if delta < 0.0 or np.random.rand() < np.exp(-delta / (T + 1e-18)):
            cur_a = cand
            cur_score = cand_score
            if cur_score < best_score:
                best_score = cur_score
                best_a = cur_a.copy()
                step_scale = min(step_scale * 1.07, 0.5)
        else:
            step_scale *= 0.97

        T = T0 * (1.0 - step / max_steps)

    return best_a, best_score

        n = 512
    print(f"[{time.time():.1f}] Using length n = {n}")

    best_seq = None
    best_score = float("inf")

    def maybe_update(vec: np.ndarray, tag: str = ""):
        """Update the globally best vector if it improves the true score."""
        nonlocal best_seq, best_score
        sc = evaluate_sequence(vec.tolist())
        if sc < best_score:
            best_score = sc
            best_seq = vec.tolist()
            print(f"[{time.time():.1f}] new best {tag:<20} = {best_score:.6f}")

    # ------------------------------------------------------------------
    # 0. Optional fallback to the previously known construction
    # ------------------------------------------------------------------
    try:
        fb_vec = [float(x) for x in height_sequence_1]   # type: ignore
        maybe_update(np.array(fb_vec), "fallback")
    except Exception:
        pass

    # ------------------------------------------------------------------
    # 1. Broad scan of analytic families + random Dirichlet starts
    # ------------------------------------------------------------------
    uniform = np.full(n, 1.0 / n)
    maybe_update(uniform, "uniform")

    # symmetric product (U‑shaped) – already present in the reference code
    for alpha in np.linspace(0.0, 6.0, 31):
        if time.time() >= deadline - 30.0:
            break
        vec = symmetric_product_family(n, alpha)
        maybe_update(vec, f"sym‑α={alpha:.2f}")

    # interior‑peaked power family (γ>0)
    for gamma in np.linspace(0.2, 3.0, 15):
        if time.time() >= deadline - 30.0:
            break
        vec = interior_power_family(n, gamma)
        maybe_update(vec, f"int‑γ={gamma:.2f}")

    # two‑parameter Beta family – this adds a lot of diversity
    alphas = [0.3, 0.5, 0.7, 1.0, 1.5, 2.0, 3.0]
    betas  = [0.3, 0.5, 0.7, 1.0, 1.5, 2.0, 3.0]
    for a in alphas:
        for b in betas:
            if time.time() >= deadline - 30.0:
                break
            vec = beta_family(n, a, b)
            maybe_update(vec, f"beta‑a={a:.1f}_b={b:.1f}")

    # random Dirichlet draws (wide concentration range)
    conc_vals = np.logspace(-2, 1, 25)   # 0.01 … 10
    for conc in conc_vals:
        for _ in range(2):
            if time.time() >= deadline - 30.0:
                break
            vec = random_dirichlet_start(n, conc)
            maybe_update(vec, f"dir‑α={conc:.3f}")

    # ------------------------------------------------------------------
    # 2. Linear‑programming refinement loop
    # ------------------------------------------------------------------
    # start from the best vector discovered so far (fallback to uniform)
    f = np.array(best_seq if best_seq is not None else uniform, dtype=np.float64)
    # ensure it lives on the simplex
    f = project_to_simplex(f)

    t_mix = 0.2          # initial mixing weight
    lp_iter = 0
    while time.time() < deadline - 120.0:
        lp_iter += 1
        # current max‑convolution bound
        M = convolution_fft(f, f).max()
        # solve LP (1)
        g = lp_maximize_sum_under_conv_bound(f, M)
        if g is None:
            # LP failed – give up the LP phase
            break
        # normalise g (probability vector)
        if g.sum() == 0.0:
            break
        g = project_to_simplex(g)

        # convex combination and re‑normalise
        f = (1.0 - t_mix) * f + t_mix * g
        f = project_to_simplex(f)

        maybe_update(f, f"LP‑iter{lp_iter}")

        # gradually decay the mixing weight so that we take finer steps later
        t_mix = max(0.02, t_mix * 0.93)

    # ------------------------------------------------------------------
    # 3. Mirror sub‑gradient descent (exact max‑objective)
    # ------------------------------------------------------------------
    if time.time() < deadline - 70.0:
        print(f"[{time.time():.1f}] Starting mirror sub‑gradient descent")
        f_sub, sc_sub = mirror_subgrad_optimize(
            f,
            deadline - 20.0,
            step0=0.9,
        )
        maybe_update(f_sub, "mirror")

    # ------------------------------------------------------------------
    # 4. Symmetrisation (many optimal solutions are symmetric)
    # ------------------------------------------------------------------
    if best_seq is not None:
        a_arr = np.array(best_seq, dtype=np.float64)
        a_sym = (a_arr + a_arr[::-1]) * 0.5
        a_sym = project_to_simplex(a_sym)
        maybe_update(a_sym, "symmetrised")

    # ------------------------------------------------------------------
    # 5. Final tiny hill‑climb to squeeze out any remaining slack
    # ------------------------------------------------------------------
    if best_seq is not None and time.time() < deadline - 5.0:
        print(f"[{time.time():.1f}] Final hill‑climb (SA)")
        a_final, sc_final = hill_climb_sa(
            np.array(best_seq, dtype=np.float64),
            deadline,
            max_steps=1_200_000,
        )
        maybe_update(a_final, "SA‑final")

    # ------------------------------------------------------------------
    # Safeguard – if everything failed, fall back to a modest uniform vector
    # ------------------------------------------------------------------
    if best_seq is None:
        n_small = 256
        best_seq = [1.0 / n_small] * n_small
        best_score = evaluate_sequence(best_seq)

    final_score = evaluate_sequence(best_seq)
    print(f"\n[{time.time():.1f}] finished search – final reported score = {final_score:.6f}")

    return best_seq
