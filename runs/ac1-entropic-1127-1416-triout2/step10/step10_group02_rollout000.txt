"""
Non‑negative step‑function optimiser for ``evaluate_sequence``.

Algorithm
---------
1. Initialise from the best known construction (``height_sequence_1``) or
   from a uniform distribution.
2. Perform a *continuation* gradient descent that minimises a smooth
   approximation of the objective `max(conv(a,a))`.  The smooth
   approximation is `log‑sum‑exp` applied to the convolution values,
   which yields a soft‑max weight vector `w`.  The gradient of the
   weighted convolution is obtained by a second convolution, and the
   update is a projected (additive) step onto the probability simplex.
   The temperature `beta` of the soft‑max is increased from a small
   value to a larger one, turning the smoothed problem into the actual
   max‑problem.
3. Refine the result with a fast stochastic hill‑climb that repeatedly
   moves a fraction of mass between two coordinates and keeps the
   best sequence found.
4. Return the best sequence (as a plain list of floats) before the
   allocated time expires.

The method combines a global, gradient‑based phase with a local,
discrete search and stays well within the 1000 s budget.
"""

import time
import numpy as np

# ------------------------------------------------------------
# Numerical utilities
# ------------------------------------------------------------

def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    if np.all(v <= 0):
        # Uniform distribution is the projection of the zero vector.
        return np.full_like(v, 1.0 / v.shape[0])

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, v.shape[0] + 1)
    mask = u - cssv / ind > 0
    rho = np.where(mask)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def full_convolution(a: np.ndarray) -> np.ndarray:
    """
    Full linear convolution a*a using a real FFT.
    Length of the result is 2*len(a)-1.
    """
    n = a.shape[0]
    m = 2 * n - 1
    n_fft = 1 << ((m - 1).bit_length())          # next power of two ≥ m
    A = np.fft.rfft(a, n=n_fft)
    C = A * A
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


def compute_score(a: np.ndarray) -> float:
    """
    Exact objective for a normalised vector (sum = 1):
        2·n·max(conv(a,a))
    """
    n = a.shape[0]
    conv = full_convolution(a)
    return float(2.0 * n * conv.max())


# ------------------------------------------------------------
# 1. Smoothed gradient descent (soft‑max of the convolution)
# ------------------------------------------------------------

def smoothed_gradient_descent(a0: np.ndarray,
                              deadline: float,
                              beta_start: float = 2.0,
                              beta_end: float = 40.0,
                              step0: float = 0.5) -> np.ndarray:
    """
    Projected gradient descent on a smooth approximation of the max‑convolution.
    The smoothness is controlled by a soft‑max temperature beta that increases
    linearly from beta_start to beta_end as the optimisation proceeds.
    Returns the best vector encountered.
    """
    a = a0.copy()
    n = a.shape[0]

    best_a = a.copy()
    best_val = compute_score(a)

    it = 0
    while time.time() < deadline:
        it += 1
        # ----------------------------------------------------
        # 1️⃣  Compute convolution and the soft‑max weights
        # ----------------------------------------------------
        conv = full_convolution(a)               # length 2n‑1
        conv_max = conv.max()
        if conv_max == 0.0:                      # degenerate – should not happen
            break

        # soft‑max weights that emphasise large entries
        # w_i = exp( beta * (conv_i/conv_max - 1) )   (max weight = 1)
        progress = (time.time() - (deadline - (deadline - start_time))) / (deadline - start_time)
        beta = beta_start + (beta_end - beta_start) * progress
        w = np.exp(beta * (conv / conv_max - 1.0))
        w_sum = w.sum()
        if w_sum == 0.0:
            # if beta is huge and underflows to zero, fall back to uniform weights
            w = np.ones_like(conv)
            w_sum = w.size

        # ----------------------------------------------------
        # 2️⃣  Gradient of the weighted convolution
        # ----------------------------------------------------
        rev_a = a[::-1]
        g_full = np.convolve(w, rev_a)          # length 3n‑2
        # slice to obtain gradient for each coordinate
        g = 2.0 * g_full[n - 1:n - 1 + n] / w_sum

        # ----------------------------------------------------
        # 3️⃣  Additive step + projection onto simplex
        # ----------------------------------------------------
        step = step0 / np.sqrt(it)
        a = a - step * g
        a = project_to_simplex(a)

        # ----------------------------------------------------
        # 4️⃣  Keep the best solution seen so far
        # ----------------------------------------------------
        val = compute_score(a)
        if val < best_val:
            best_val = val
            best_a = a.copy()

        # occasional jitter to escape plateaus
        if it % 500 == 0:
            jitter = np.random.randn(n) * (step0 * 0.01 / np.sqrt(it))
            a = a + jitter
            a = project_to_simplex(a)

    return best_a


# ------------------------------------------------------------
# 2. Simple stochastic hill‑climb (greedy)
# ------------------------------------------------------------

def hill_climb(a_start: np.ndarray, deadline: float) -> np.ndarray:
    """
    Greedy stochastic mass‑transfer moves.
    A fraction of the mass at a random source index is moved to a random
    destination index; the move is kept if it improves the objective.
    The loop runs until the deadline is reached.
    """
    a = a_start.copy()
    n = a.shape[0]

    best_a = a.copy()
    best_val = compute_score(a)

    # initial displacement fraction – will be adapted
    step_scale = 0.05

    while time.time() < deadline:
        i, j = np.random.randint(n, size=2)
        if i == j:
            continue

        # move a small fraction of the mass from i to j
        delta = step_scale * a[i]
        if delta <= 0.0:
            continue

        cand = a.copy()
        cand[i] -= delta
        cand[j] += delta

        # protect against rounding errors
        cand = np.maximum(cand, 0.0)
        cand = project_to_simplex(cand)

        val = compute_score(cand)
        if val < best_val:
            best_val = val
            best_a = cand.copy()
            a = cand
            # successful move → increase aggressiveness a bit
            step_scale = min(step_scale * 1.07, 0.5)
        else:
            # unsuccessful → become more conservative
            step_scale *= 0.985

    return best_a

        if official_val < best_val:
            best_val, best_seq = official_val, a_refined.tolist()

    # -----------------------------------------------------------------
    # 3️⃣  Stochastic hill‑climb for fine‑grained improvement
    # -----------------------------------------------------------------
    hill_deadline = deadline
    if time.time() < hill_deadline:
        a_start = np.array(best_seq, dtype=float)
        a_start = a_start / max(a_start.sum(), 1e-12)
        print("[hill‑climb] start")
        a_final = hill_climb(a_start, hill_deadline)
        val = compute_score(a_final)
        print(f"[hill‑climb] final internal score = {val:.6f}")
        official_val = evaluate_sequence(a_final.tolist())
        print(f"[hill‑climb] official score = {official_val:.6f}")
        if official_val < best_val:
            best_val, best_seq = official_val, a_final.tolist()

    # -----------------------------------------------------------------
    # 4️⃣  Safeguard – ensure a sequence is always returned
    # -----------------------------------------------------------------
    if best_seq is None:
        # very unlikely – return a safe uniform vector
        best_seq = [1.0 / n] * n
        best_val = evaluate_sequence(best_seq)

    # -----------------------------------------------------------------
    # 5️⃣  Final report
    # -----------------------------------------------------------------
    final_score = evaluate_sequence(best_seq)
    print(f"\nSearch finished. Reported score = {final_score:.6f}")
    return best_seq
