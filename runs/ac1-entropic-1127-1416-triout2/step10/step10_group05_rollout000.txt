"""Step‑function optimisation.

This version builds on the previous exponentiated‑gradient / max‑descent
pipeline but adds several new ideas to push the score below 1.511:

* A much larger search space (up to 65536 coefficients) when time permits.
* A far‑richer Beta‑distribution scan that includes very sharp
  α,β∈[0.01,0.5] – the arcsine‑like shapes are known to flatten
  self‑convolution.
* Many cheap random Dirichlet starts, each followed by a short
  smooth‑max descent, to diversify the search.
* Adaptive schedules for the smooth‑max temperature β and for the
  KL‑mirror learning rate.
* A final hill‑climb that always moves mass from the index that
  contributes most to the current maximal convolution entry towards the
  globally smallest coefficient (instead of a random destination).

The routine respects the 1000 s wall‑clock budget, prints progress
messages and finally returns a plain Python list of non‑negative
floats that can be fed to ``evaluate_sequence``.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Basic linear‑algebra utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full aperiodic convolution using a real FFT (zero‑padded)."""
    m = a.size + b.size - 1
    nfft = next_pow2(m)
    A = np.fft.rfft(a, n=nfft)
    B = np.fft.rfft(b, n=nfft)
    C = A * B
    return np.fft.irfft(C, n=nfft)[:m]


def compute_score(a: np.ndarray) -> float:
    """
    Exact objective used by `evaluate_sequence` for a vector already
    normalised to sum = 1.  Returns 2·n·max(a*a) where the convolution
    is full (length 2·n‑1).
    """
    n = a.shape[0]
    conv = convolution_fft(a, a)
    return 2.0 * n * float(conv.max())


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection onto the probability simplex (sum = 1, v≥0)."""
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / v.shape[0])

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, v.shape[0] + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.where(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


# ----------------------------------------------------------------------
# Parameterised families of candidate vectors
# ----------------------------------------------------------------------


def exponential_shape(n: int, lam: float) -> np.ndarray:
    """a_i ∝ exp(-lam·i), normalised to sum = 1."""
    i = np.arange(n, dtype=float)
    if lam <= 0.0:
        w = np.ones_like(i)
    else:
        w = np.exp(-lam * i)
    return w / w.sum()


def power_shape(n: int, p: float) -> np.ndarray:
    """a_i ∝ (i+1)^(-p), normalised to sum = 1."""
    i = np.arange(1, n + 1, dtype=float)
    if p <= 0.0:
        w = np.ones_like(i)
    else:
        w = i ** (-p)
    return w / w.sum()


def beta_shape(n: int, a: float, b: float) -> np.ndarray:
    """
    Discrete Beta distribution on {0,…,n‑1}:
        a_i ∝ ((i+0.5)/n)^(a‑1) * (1‑(i+0.5)/n)^(b‑1)
    """
    x = (np.arange(n, dtype=float) + 0.5) / n
    # avoid NaNs when a or b < 1
    w = np.power(x, a - 1.0) * np.power(1.0 - x, b - 1.0)
    w[np.isnan(w)] = 0.0
    w = np.maximum(w, 0.0)
    total = w.sum()
    if total == 0.0:
        w = np.ones_like(w)
        total = w.sum()
    return w / total


# ----------------------------------------------------------------------
# Search over simple families
# ----------------------------------------------------------------------


def scan_exponential_and_power(n: int, deadline: float):
    """Exhaustive scan of exponential and power‑law families."""
    best_a, best_sc = None, float("inf")

    # exponential lambda grid (including λ=0 → uniform)
    lam_grid = np.concatenate(
        (np.array([0.0]), np.logspace(-5, 0.5, 250))
    )
    for lam in lam_grid:
        if time.time() >= deadline:
            break
        a = exponential_shape(n, lam)
        sc = compute_score(a)
        if sc < best_sc:
            best_sc, best_a = sc, a.copy()

    # power‑law grid
    p_grid = np.linspace(0.0, 8.0, 250)
    for p in p_grid:
        if time.time() >= deadline:
            break
        a = power_shape(n, p)
        sc = compute_score(a)
        if sc < best_sc:
            best_sc, best_a = sc, a.copy()

    return best_a, best_sc


def scan_beta_family(n: int, deadline: float):
    """Rich scan of the Beta family, especially in the (0,0.5) region."""
    best_a, best_sc = None, float("inf")

    # grids for α and β – more points near 0 (sharp edges) and near 1
    small_grid = np.logspace(-4, -0.1, 16)       # 0.0001 … 0.79
    medium_grid = np.logspace(-0.1, 0.5, 12)    # 0.79 … 3.16
    large_grid = np.linspace(0.5, 5.0, 12)      # 0.5 … 5
    a_grid = np.unique(np.concatenate((small_grid, medium_grid, large_grid)))
    b_grid = a_grid.copy()

    for a in a_grid:
        for b in b_grid:
            if time.time() >= deadline:
                return best_a, best_sc
            vec = beta_shape(n, a, b)
            sc = compute_score(vec)
            if sc < best_sc:
                best_sc, best_a = sc, vec.copy()
    return best_a, best_sc


# ----------------------------------------------------------------------
# Optimisation phases
# ----------------------------------------------------------------------


def smooth_max_optimize(start_a: np.ndarray,
                        deadline: float,
                        max_iters: int = 300_000,
                        beta_start: float = 0.5,
                        beta_end: float = 5000.0,
                        eta0: float = 0.6) -> tuple[np.ndarray, float]:
    """
    KL‑mirror descent on a log‑sum‑exp approximation of the max‑convolution.
    β grows exponentially from `beta_start` to `beta_end`.
    """
    eps = 1e-15
    a = np.maximum(start_a.astype(float), eps)
    a /= a.sum()
    n = a.shape[0]

    best_a = a.copy()
    best_sc = compute_score(a)

    for it in range(1, max_iters + 1):
        if time.time() >= deadline:
            break

        # temperature schedule
        t = it / max_iters
        beta = beta_start * (beta_end / beta_start) ** t

        conv = convolution_fft(a, a)
        maxc = conv.max()
        # stable soft‑max weights
        exponents = beta * (conv - maxc)
        # prevent overflow
        max_exp = np.max(exponents)
        expw = np.exp(exponents - max_exp)
        w = expw / expw.sum()

        # gradient of smooth max (see derivation in comments)
        a_rev = a[::-1]
        grad = 2.0 * convolution_fft(a_rev, w)[n - 1:n - 1 + n]

        # KL‑mirror step size
        eta = eta0 / np.sqrt(it)

        a = np.maximum(a * np.exp(-eta * grad), eps)
        a /= a.sum()

        if it % 400 == 0:
            sc = compute_score(a)
            if sc < best_sc:
                best_sc, best_a = sc, a.copy()
    # final exact check
    sc = compute_score(a)
    if sc < best_sc:
        best_sc, best_a = sc, a.copy()
    return best_a, best_sc


def max_descent_optimize(start_a: np.ndarray,
                         deadline: float,
                         max_iters: int = 250_000,
                         eta0: float = 0.5) -> tuple[np.ndarray, float]:
    """
    Sub‑gradient descent on the true max‑convolution using a KL‑mirror
    update (multiplicative).
    """
    eps = 1e-15
    a = np.maximum(start_a.astype(float), eps)
    a /= a.sum()
    n = a.shape[0]

    best_a = a.copy()
    best_sc = compute_score(a)

    for it in range(1, max_iters + 1):
        if time.time() >= deadline:
            break

        conv = convolution_fft(a, a)
        k_max = int(conv.argmax())

        # sub‑gradient for the active convolution entry
        g = np.zeros_like(a)
        i_start = max(0, k_max - (n - 1))
        i_end = min(n - 1, k_max)
        idx = np.arange(i_start, i_end + 1)
        g[idx] = 2.0 * a[k_max - idx]

        eta = eta0 / np.sqrt(it)

        a = np.maximum(a * np.exp(-eta * g), eps)
        a /= a.sum()

        if it % 300 == 0:
            sc = compute_score(a)
            if sc < best_sc:
                best_sc, best_a = sc, a.copy()
    # final exact check
    sc = compute_score(a)
    if sc < best_sc:
        best_sc, best_a = sc, a.copy()
    return best_a, best_sc


def targeted_hill_climb(start_seq: np.ndarray,
                        deadline: float,
                        max_steps: int = 500_000) -> tuple[np.ndarray, float]:
    """
    Greedy hill‑climb: each step removes a fraction of the coefficient that
    contributes most to the current maximal convolution entry and adds it to
    the smallest coefficient in the vector.  The move is kept only if it
    improves the exact objective.
    """
    a = start_seq.copy()
    n = a.size
    best_a = a.copy()
    best_sc = compute_score(best_a)

    # adaptive step size (relative to the source coefficient)
    step_scale = 0.08
    steps = 0

    while steps < max_steps and time.time() < deadline:
        conv = convolution_fft(best_a, best_a)
        k_max = int(conv.argmax())

        # indices that participate in the maximal entry
        i_start = max(0, k_max - (n - 1))
        i_end = min(n - 1, k_max)
        idx = np.arange(i_start, i_end + 1)
        # contribution of each participant
        contrib = best_a[idx] * best_a[k_max - idx]
        i_max = idx[np.argmax(contrib)]

        # destination = index with minimal current mass
        j = int(np.argmin(best_a))

        if j == i_max:
            # if the smallest is also the source pick a random other index
            j = (j + 1) % n
            if j == i_max:
                break

        delta = step_scale * best_a[i_max]
        if delta <= 0.0:
            break

        cand = best_a.copy()
        cand[i_max] -= delta
        cand[j] += delta
        cand = project_to_simplex(cand)

        sc = compute_score(cand)
        if sc < best_sc:
            best_sc, best_a = sc, cand.copy()
            step_scale = min(step_scale * 1.06, 0.5)   # become bolder after success
        else:
            step_scale *= 0.94                         # become more cautious

        steps += 1

    return best_a, best_sc
