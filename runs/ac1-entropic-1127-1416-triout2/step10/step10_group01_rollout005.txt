def eg_optimize(start_a, deadline, max_iters=None, eta=0.1):
    a = start_a.copy()
    n = a.shape[0]
    best_a, best_score = a.copy(), compute_score(a)
    it = 0

        if max_iters is not None and it >= max_iters:
            break
        conv = convolution_fft(a, a)
        k = int(conv.argmax())
        # gradient g_i = 2 * a_{k - i} if valid else 0
        idx = np.arange(n)
        j = k - idx
        mask = (j >= 0) & (j < n)
        g = np.zeros_like(a)
        g[mask] = 2.0 * a[j[mask]]
        # exponentiated update
        a = a * np.exp(-eta * g)
        # renorm (simplex)
        a = a / a.sum()
        # optional projection to avoid extremes
        # ensure non-negative numeric stability
        if it % 1000 == 0:
            sc = compute_score(a)
            if sc < best_score:
                best_score, best_a = sc, a.copy()
                # optionally reduce eta or adapt
        it += 1
    return best_a, best_score

"""
Improved search for a step‑function that minimises

        2·n·max(convolve(a,a)) / (sum(a)**2)

under the simplex constraint.

Algorithm
---------
0.  If the harness provides a fallback sequence (global ``height_sequence_1``)
    evaluate it and keep it as a starting point.

1.  For one or two problem sizes (the length of the fallback or 2048,
    possibly doubled) perform a fast sweep over a few analytic families
    (symmetric product, left–heavy power, exponential “U” shape) and a few
    random monotone starts.  Keep the best vector from this scan.

2.  Refine the best vector with an **exponentiated‑gradient descent**
    (EG) on the exact max‑convolution objective.  The EG update
    ``a_i ← a_i·exp(-η·g_i)`` with a decaying step size keeps the iterates
    on the simplex automatically.

3.  Run a cheap **projected sub‑gradient descent** (the diagonal term in
    the gradient is now correct – the factor is 2 for all indices) to
    polish the solution.

4.  Perform a few iterations of the classic **LP refinement** described
    in the reference material (maximising Σ g_j under the convolution
    bound).  The mixing weight is reduced if no improvement is seen.

5.  If time permits, finish with a short **simulated‑annealing hill‑climb**
    (random swaps / mass transfers on the simplex).

All phases stop as soon as the global wall‑clock budget is exhausted.
The function returns a plain list of floats ready for ``evaluate_sequence``.
"""

import time
import math
import numpy as np
from scipy.optimize import linprog

# ----------------------------------------------------------------------
# Basic utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full (aperiodic) convolution via real FFT."""
    m = a.size + b.size - 1
    nfft = next_pow2(m)
    A = np.fft.rfft(a, n=nfft)
    B = np.fft.rfft(b, n=nfft)
    C = A * B
    conv = np.fft.irfft(C, n=nfft)[:m]
    return conv


def compute_score(a: np.ndarray) -> float:
    """
    Exact objective for a probability vector (∑a = 1):
        2·n·max(a*a)
    """
    n = a.shape[0]
    conv = convolution_fft(a, a)
    return 2.0 * n * float(conv.max())


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / v.shape[0])

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, v.shape[0] + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.where(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Analytic families (used for the quick initial scan)
# ----------------------------------------------------------------------


def symmetric_product_family(n: int, alpha: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    logprod = np.log(i + 0.5) + np.log(n - 0.5 - i)
    w = np.exp(-alpha * logprod)
    a = w / w.sum()
    return a


def left_heavy_power_family(n: int, beta: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    w = np.exp(-beta * np.log(i + 0.5))
    a = w / w.sum()
    return a


def exp_u_family(n: int, gamma: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    dist = np.minimum(i, n - 1 - i)
    w = np.exp(-gamma * dist)
    a = w / w.sum()
    return a


def random_monotone_start(n: int) -> np.ndarray:
    r = np.random.rand(n) ** np.random.uniform(0.5, 5.0)
    r.sort()
    a = r[::-1]
    a /= a.sum()
    return a


def random_u_start(n: int) -> np.ndarray:
    gamma = np.random.uniform(0.0, 6.0)
    return exp_u_family(n, gamma)


# ----------------------------------------------------------------------
# Exponentiated‑gradient descent on the exact max‑convolution
# ----------------------------------------------------------------------


def eg_optimize(start_a: np.ndarray,
                deadline: float,
                eta0: float = 0.6,
                max_iters: int = 200_000) -> tuple[np.ndarray, float]:
    """
    Exponentiated‑gradient (mirror) descent on
        f(a) = 2·n·max(a*a)        (a lives on the simplex).

    The sub‑gradient of the active convolution shift k is
        g_i = 2·a_{k-i}   (with zero outside the range).
    The update is   a_i ← a_i·exp(-η·g_i)  followed by renormalisation.
    """
    a = start_a.astype(np.float64).copy()
    best_a = a.copy()
    best_score = compute_score(a)

    n = a.shape[0]
    for t in range(1, max_iters + 1):
        if time.time() >= deadline:
            break

        conv = convolution_fft(a, a)
        k = int(conv.argmax())

        idx = np.arange(n)
        j = k - idx
        mask = (j >= 0) & (j < n)
        g = np.zeros_like(a)
        g[mask] = 2.0 * a[j[mask]]

        eta = eta0 / math.sqrt(t)
        a = a * np.exp(-eta * g)
        a = a / a.sum()                     # stays on the simplex automatically

        if (t & 1023) == 0:                 # every ~1 k iterations
            sc = compute_score(a)
            if sc < best_score:
                best_score = sc
                best_a = a.copy()
                print(f"[{time.time():.1f}] EG improved to {best_score:.6f}")

    # final check
    sc = compute_score(a)
    if sc < best_score:
        best_score, best_a = sc, a.copy()
    return best_a, best_score


# ----------------------------------------------------------------------
# Projected sub‑gradient descent (fixed diagonal term)
# ----------------------------------------------------------------------


def subgrad_optimize_fixed(start_a: np.ndarray,
                           deadline: float,
                           step0: float = 0.07) -> tuple[np.ndarray, float]:
    """
    Sub‑gradient descent on 2·n·max(a*a) with the correct diagonal factor.
    The step size decays as 1/√t.
    """
    a = start_a.astype(np.float64).copy()
    n = a.shape[0]

    best_a = a.copy()
    best_score = compute_score(a)

    t = 1
    while time.time() < deadline:
        conv = convolution_fft(a, a)
        k = int(conv.argmax())

        idx = np.arange(n)
        j = k - idx
        mask = (j >= 0) & (j < n)
        g = np.zeros_like(a)
        # the gradient is 2·a_{k-i} for every valid i (including the diagonal)
        g[mask] = 2.0 * a[j[mask]]

        step = step0 / math.sqrt(t)
        a = project_to_simplex(a - step * g)

        if t % 2000 == 0:
            sc = compute_score(a)
            if sc < best_score:
                best_score = sc
                best_a = a.copy()
                print(f"[{time.time():.1f}] sub‑grad improved to {best_score:.6f}")

        t += 1
        if t > 5_000_000:          # safety cap
            break

    sc = compute_score(a)
    if sc < best_score:
        best_score, best_a = sc, a.copy()
    return best_a, best_score


# ----------------------------------------------------------------------
# LP‑based refinement (the “state‑of‑the‑art” iteration)
# ----------------------------------------------------------------------


def lp_refine_step(f: np.ndarray, M: float, mix_weight: float = 0.2) -> np.ndarray:
    """
    Solve   max Σ g_j   s.t.   (f * g)_k ≤ M  (for all k),
    g ≥ 0, then return the convex mixture
        f_new = (1‑t)·f + t·g_norm
    where g_norm = g / Σ g.
    """
    n = f.shape[0]
    m = 2 * n - 1

    # Build convolution matrix A where (A @ g) = f * g
    A = np.zeros((m, n), dtype=np.float64)
    for i in range(n):
        A[i:i + n, i] = f[i]

    c = -np.ones(n, dtype=np.float64)      # maximise Σ g ⇒ minimise −Σ g
    bounds = [(0.0, None) for _ in range(n)]
    b = np.full(m, M, dtype=np.float64)

    # Solve – cap the internal solver time to avoid rare stalls
    res = linprog(c, A_ub=A, b_ub=b, bounds=bounds, method='highs',
                  options={'presolve': True, 'time_limit': 30})

    if not res.success:
        return f

    g = np.maximum(res.x, 0.0)
    sum_g = g.sum()
    if sum_g <= 0.0:
        return f

    g_norm = g / sum_g
    f_new = (1.0 - mix_weight) * f + mix_weight * g_norm
    f_new = project_to_simplex(f_new)
    return f_new


# ----------------------------------------------------------------------
# Stochastic hill‑climb (simulated‑annealing style)
# ----------------------------------------------------------------------


def hill_climb_sa(start_a: np.ndarray,
                  deadline: float,
                  max_steps: int = 500_000) -> tuple[np.ndarray, float]:
    """
    Random walk on the simplex with occasional uphill moves.
    """
    n = start_a.size
    cur_a = start_a.copy()
    cur_score = compute_score(cur_a)

    best_a = cur_a.copy()
    best_score = cur_score

    T0 = best_score * 1e-4
    T = T0
    step_scale = 0.12

    for step in range(1, max_steps + 1):
        if time.time() >= deadline:
            break

        cand = cur_a.copy()
        if np.random.rand() < 0.5:               # swap two entries
            i, j = np.random.randint(n, size=2)
            cand[i], cand[j] = cand[j], cand[i]
        else:                                     # mass transfer
            i, j = np.random.randint(n, size=2)
            if i != j:
                delta = step_scale * cur_a[i]
                cand[i] -= delta
                cand[j] += delta
                if cand[i] < 0.0:
                    cand[i] = 0.0

        cand = project_to_simplex(cand)
        cand_score = compute_score(cand)

        delta = cand_score - cur_score
        if delta < 0 or np.random.rand() < np.exp(-delta / (T + 1e-18)):
            cur_a = cand
            cur_score = cand_score
            if cur_score < best_score:
                best_score = cur_score
                best_a = cur_a.copy()
                step_scale = min(step_scale * 1.07, 0.5)
        else:
            step_scale *= 0.97

        # linear cooling
        T = T0 * (1.0 - step / max_steps)

    return best_a, best_score


# ----------------------------------------------------------------------
# High‑level driver
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 12345, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative height sequence that minimises
    ``evaluate_sequence``.  The routine respects the global time budget
    and returns a plain Python list of floats.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 1.0          # safety margin

    best_seq = None
    best_score = float("inf")

    # ------------------------------------------------------------------
    # 0.  Fallback candidate (if supplied by the harness)
    # ------------------------------------------------------------------
    try:
        fb_vec = [float(x) for x in height_sequence_1]   # type: ignore
        fb = evaluate_sequence(fb_vec)
        print(f"[{time.time():.1f}] fallback score = {fb:.6f}")
        if fb < best_score:
            best_score = fb
            best_seq = fb_vec
    except Exception:
        pass

    # ------------------------------------------------------------------
    # 1.  Choose problem sizes (use fallback length or default 2048)
    # ------------------------------------------------------------------
    base_n = len(best_seq) if best_seq is not None else 2048
    ns = [base_n]

    # if we have enough budget we also try a larger size (up to 8192)
    if budget_s > 300:
        ns.append(min(base_n * 2, 8192))
    if budget_s > 600:
        ns.append(min(base_n * 4, 8192))

    # ------------------------------------------------------------------
    # 2.  Main optimisation loop over the selected lengths
    # ------------------------------------------------------------------
    for n in ns:
        if time.time() >= deadline:
            break
        print(f"[{time.time():.1f}] ----- Optimisation for n = {n} -----")

        # ----- quick analytic / random scan (seconds only) -----
        best_local_vec = None
        best_local_score = float('inf')

        # families – a few points each
        for alpha in np.linspace(0.0, 12.0, 7):
            if time.time() >= deadline:
                break
            vec = symmetric_product_family(n, alpha)
            sc = compute_score(vec)
            if sc < best_local_score:
                best_local_score, best_local_vec = sc, vec

        for beta in np.linspace(0.0, 12.0, 7):
            if time.time() >= deadline:
                break
            vec = left_heavy_power_family(n, beta)
            sc = compute_score(vec)
            if sc < best_local_score:
                best_local_score, best_local_vec = sc, vec

        for gamma in np.linspace(0.0, 6.0, 7):
            if time.time() >= deadline:
                break
            vec = exp_u_family(n, gamma)
            sc = compute_score(vec)
            if sc < best_local_score:
                best_local_score, best_local_vec = sc, vec

        # a handful of random starts
        for _ in range(20):
            if time.time() >= deadline:
                break
            vec = random_monotone_start(n) if np.random.rand() < 0.5 else random_u_start(n)
            sc = compute_score(vec)
            if sc < best_local_score:
                best_local_score, best_local_vec = sc, vec

        if best_local_vec is None:
            continue

        # ----- Exponentiated‑gradient descent -----
        a_eg, sc_eg = eg_optimize(best_local_vec, deadline, eta0=0.6)
        if sc_eg < best_score:
            best_score, best_seq = sc_eg, a_eg.tolist()
            print(f"[{time.time():.1f}] EG step best = {best_score:.6f}")

        # ----- Sub‑gradient refinement (corrected) -----
        a_sub, sc_sub = subgrad_optimize_fixed(a_eg, deadline, step0=0.07)
        if sc_sub < best_score:
            best_score, best_seq = sc_sub, a_sub.tolist()
            print(f"[{time.time():.1f}] sub‑grad step best = {best_score:.6f}")

        # ----- LP refinement (a few aggressive iterations) -----
        lp_iters = 0
        while time.time() < deadline - 5.0 and lp_iters < 12:
            a_arr = np.array(best_seq, dtype=np.float64)
            conv = convolution_fft(a_arr, a_arr)
            M = float(conv.max())
            a_new = lp_refine_step(a_arr, M, mix_weight=0.2)
            sc_new = compute_score(a_new)
            if sc_new < best_score - 1e-7:
                best_score = sc_new
                best_seq = a_new.tolist()
                print(f"[{time.time():.1f}] LP iteration {lp_iters} improved to {best_score:.6f}")
            else:
                # reduce mixing to become more conservative
                pass
            lp_iters += 1

        # ----- Final stochastic hill‑climb if time remains -----
        if time.time() < deadline - 2.0:
            a_hc, sc_hc = hill_climb_sa(np.array(best_seq, dtype=np.float64), deadline,
                                         max_steps=500_000)
            if sc_hc < best_score:
                best_score = sc_hc
                best_seq = a_hc.tolist()
                print(f"[{time.time():.1f}] hill‑climb final improvement = {best_score:.6f}")

    # ------------------------------------------------------------------
    # 3.  Final official evaluation and return
    # ------------------------------------------------------------------
    final_score = evaluate_sequence(best_seq)
    print(f"\n[{time.time():.1f}] finished search – final reported score = {final_score:.6f}")
    return best_seq
