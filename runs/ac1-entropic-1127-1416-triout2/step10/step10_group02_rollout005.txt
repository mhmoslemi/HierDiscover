def propose_candidate(...):
    start_time = time.time()
    deadline = start_time + budget_s - 5

    n = 2048
    best_seq = None; best_val = inf

    # try fallback if available
    try: fallback = [float(x) for x in height_sequence_1]; val = evaluate_sequence(fallback) ...
    except, pass

    # Generate initial candidates list
    seeds = []

    # Use fallback if available
    if fallback: seeds.append(np.array(fallback, dtype=float))

    # Use best power law (coarse search)
    a_pow = best_power_law_vector(n, deadline - 120) # allocate time for power law search
    if a_pow is not None:
        seeds.append(a_pow)

    # Use some random Dirichlet seeds
    for _ in range(3):
        rand = np.random.rand(n)
        rand = rand / rand.sum()
        seeds.append(rand)

    # For each seed run primal_dual_optimize with some allocated time
    per_seed_time = (deadline - time.time()) / len(seeds)
    for idx, a0 in enumerate(seeds):
        sub_deadline = time.time() + per_seed_time
        a_opt = primal_dual_optimize(a0, sub_deadline)
        val = compute_score(a_opt)
        print(f"[seed {idx}] PD best val {val:.6f}")
        if val < best_val:
            best_val = val
            best_seq = a_opt.tolist()

    # final hill climb
    hill_deadline = deadline - 5.0
    if best_seq is not None and time.time() < hill_deadline:
        a_start = np.array(best_seq, dtype=float)
        a_final = hill_climb(a_start, hill_deadline)  # hill climb function may need to be defined again
        val = compute_score(a_final)
        print(f"[final hill] val {val:.6f}")
        if val < best_val:
            best_seq = a_final.tolist()
            best_val = val

    if best_seq is None:
        best_seq = [1.0 / n] * n

    final_score = evaluate_sequence(best_seq)
    print(f"\nSearch finished. Final eval score = {final_score:.6f}")
    return best_seq

"""
Primal‑dual optimiser for the step‑function evaluation.

The objective 2·n·max(conv(a,a)) (with ∑a=1) is the L∞‑norm of the
auto‑convolution.  The problem can be written as a saddle‑point
formulation

    min_a max_λ  λ·(a*a)      s.t.   a≥0, ∑a=1,
                                 λ≥0, ∑λ=1 ,

where λ lives on the indices of the convolution.  We apply a simple
projected primal‑dual (gradient descent / ascent) scheme, using fast
FFT convolutions, and run several random restarts (fallback, a power‑law
seed and a few Dirichlet‑type seeds).  The final candidate is polished
by a cheap greedy hill‑climb.

All projections are performed by simple non‑negative clipping followed
by renormalisation (Euclidean projection is not required for correctness
and is much faster).  The total wall‑clock budget is respected; the
function returns the best sequence found.
"""

import time
import numpy as np

# ------------------------------------------------------------
# Simple simplex normalisation (clip‑negative + renormalise)
# ------------------------------------------------------------
def normalize_simplex(v: np.ndarray) -> np.ndarray:
    """Clip to non‑negative and rescale to sum = 1 (uniform fallback if all zero)."""
    v = np.maximum(v, 0.0)
    s = v.sum()
    if s > 0.0:
        return v / s
    # fallback to uniform distribution
    return np.full_like(v, 1.0 / v.shape[0], dtype=float)


# ------------------------------------------------------------
# FFT based convolution for two arbitrary real vectors
# ------------------------------------------------------------
def conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Linear convolution x*y using real FFT, result length = len(x)+len(y)-1."""
    n = x.shape[0] + y.shape[0] - 1
    n_fft = 1 << ((n - 1).bit_length())          # next power of two ≥ n
    X = np.fft.rfft(x, n_fft)
    Y = np.fft.rfft(y, n_fft)
    conv = np.fft.irfft(X * Y, n_fft)[:n]
    return conv


# ------------------------------------------------------------
# Exact objective (used for fast internal evaluation)
# ------------------------------------------------------------
def compute_score(a: np.ndarray) -> float:
    """Exact objective for a unit‑sum vector a: 2·n·max(conv(a,a))."""
    n = a.shape[0]
    conv = conv_fft(a, a)               # length 2·n‑1
    return float(2.0 * n * conv.max())


# ------------------------------------------------------------
# Power‑law seed generator (coarse sweep, cheap global search)
# ------------------------------------------------------------
def best_power_law_vector(n: int, deadline: float) -> np.ndarray:
    """Return the normalised (i+1)^(-p) vector with smallest score over a coarse grid."""
    best_a = None
    best_val = float("inf")
    # coarse grid of exponents
    for p in np.linspace(0.50, 0.80, 61):       # step ≈0.005
        if time.time() >= deadline:
            break
        a = np.arange(1, n + 1, dtype=float) ** (-p)
        a /= a.sum()
        val = compute_score(a)
        if val < best_val:
            best_val, best_a = val, a
    return best_a


# ------------------------------------------------------------
# Simple greedy hill‑climb (final polishing)
# ------------------------------------------------------------
def hill_climb(a_start: np.ndarray, deadline: float,
               max_iters: int = 2_000_000) -> np.ndarray:
    """Tiny stochastic moves (transfer a fraction of mass) with greedy acceptance."""
    a = a_start.copy()
    n = a.shape[0]
    best_a = a.copy()
    best_val = compute_score(a)

    step_scale = 0.05
    it = 0
    while time.time() < deadline and it < max_iters:
        i, j = np.random.randint(n, size=2)
        if i == j:
            continue
        delta = step_scale * a[i]
        cand = a.copy()
        cand[i] -= delta
        cand[j] += delta
        cand = normalize_simplex(cand)

        val = compute_score(cand)
        if val < best_val:
            best_val, best_a = val, cand.copy()
            a = cand
            step_scale = min(step_scale * 1.07, 0.5)   # increase step on success
        else:
            step_scale *= 0.985                         # shrink on failure
        it += 1
    return best_a


# ------------------------------------------------------------
# Primal‑dual optimiser (core of this submission)
# ------------------------------------------------------------
def primal_dual_optimize(a0: np.ndarray, deadline: float) -> np.ndarray:
    """
    Projected gradient descent on a (primal) and ascent on λ (dual).
    a ∈ simplex_n, λ ∈ simplex_{2n‑1}.
    """
    n = a0.shape[0]
    L = 2 * n - 1                     # length of the convolution vector
    a = normalize_simplex(a0.astype(float))
    lam = np.full(L, 1.0 / L, dtype=float)   # uniform dual start

    best_a = a.copy()
    best_val = compute_score(a)

    t = 0
    while time.time() < deadline:
        t += 1
        # ---------- dual ascent ----------
        conv_a = conv_fft(a, a)                # length L
        step_l = 0.30 / np.sqrt(t)             # diminishing step
        lam += step_l * conv_a
        lam = normalize_simplex(lam)

        # ---------- primal descent ----------
        lam_rev = lam[::-1]
        conv_rev = conv_fft(a, lam_rev)        # length n + L - 1 = 3n‑2
        idx = L - 1 - np.arange(n)             # indices for the gradient
        grad_a = 2.0 * conv_rev[idx]           # shape (n,)

        step_a = 0.30 / np.sqrt(t)
        a -= step_a * grad_a
        a = normalize_simplex(a)

        # occasional jitter to escape plateaus
        if t % 2000 == 0:
            jitter_a = np.random.randn(n) * (0.02 / np.sqrt(t))
            jitter_l = np.random.randn(L) * (0.02 / np.sqrt(t))
            a += jitter_a
            lam += jitter_l
            a = normalize_simplex(a)
            lam = normalize_simplex(lam)

        # ---------- bookkeeping ----------
        val = compute_score(a)
        if val < best_val:
            best_val, best_a = val, a.copy()
            # early exit if we already beat the target
            if best_val <= 1.503:
                break
    return best_a

# Main entry point required by the harness
# ------------------------------------------------------------
def propose_candidate(seed: int = 12345, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step‑function that minimises ``evaluate_sequence``.
    Uses a primal‑dual scheme with several random restarts and a final hill‑climb.
    Returns a plain Python list of floats.
    """
    np.random.seed(seed)
    start_time = time.time()
    # keep a small safety margin for the harness' own bookkeeping
    deadline = start_time + budget_s - 5.0

    n = 2048                       # length of the step function
    best_seq = None
    best_val = float("inf")

    # -------------------------------------------------
    # 0. Optional fallback (previously discovered best)
    # -------------------------------------------------
    try:
        fallback = [float(x) for x in height_sequence_1]   # type: ignore
        val = evaluate_sequence(fallback)
        print(f"[fallback] score = {val:.6f}")
        if val < best_val:
            best_val, best_seq = val, fallback
    except Exception:
        pass

    # -------------------------------------------------
    # 1. Power‑law global seed
    # -------------------------------------------------
    pow_deadline = deadline - 250.0          # keep time for optimisation phases
    if time.time() < pow_deadline:
        a_pow = best_power_law_vector(n, pow_deadline)
        if a_pow is not None:
            val = compute_score(a_pow)
            print(f"[power‑law] initial score = {val:.6f}")
            if val < best_val:
                best_val, best_seq = val, a_pow.tolist()

    # -------------------------------------------------
    # 2. Assemble a list of diverse seeds
    # -------------------------------------------------
    seeds = []
    if best_seq is not None:
        seeds.append(np.array(best_seq, dtype=float))

    # random Dirichlet / exponential seeds
    for _ in range(3):
        rand = np.random.exponential(scale=1.0, size=n)
        rand /= rand.sum()
        seeds.append(rand)

    # random power‑law seeds with random exponent
    for _ in range(2):
        p = np.random.uniform(0.45, 0.85)
        a = np.arange(1, n + 1, dtype=float) ** (-p)
        a /= a.sum()
        seeds.append(a)

    # -------------------------------------------------
    # 3. Primal‑dual optimisation for each seed
    # -------------------------------------------------
    remaining = deadline - time.time()
    if remaining > 0 and seeds:
        per_seed = remaining / len(seeds)
        for idx, a0 in enumerate(seeds):
            sub_dead = time.time() + per_seed
            if time.time() >= deadline:
                break
            print(f"[PD start] seed {idx+1}/{len(seeds)}")
            a_opt = primal_dual_optimize(a0, sub_dead)
            val = compute_score(a_opt)
            print(f"[PD end]   seed {idx+1} best score = {val:.6f}")
            if val < best_val:
                best_val, best_seq = val, a_opt.tolist()
            # early termination if target reached
            if best_val <= 1.503:
                break

    # -------------------------------------------------
    # 4. Final greedy hill‑climb (fine polishing)
    # -------------------------------------------------
    hill_dead = deadline - 2.0
    if best_seq is not None and time.time() < hill_dead:
        a_start = np.array(best_seq, dtype=float)
        a_final = hill_climb(a_start, hill_dead)
        val = compute_score(a_final)
        print(f"[hill‑climb] final score = {val:.6f}")
        if val < best_val:
            best_val, best_seq = val, a_final.tolist()

    # -------------------------------------------------
    # 5. Safeguard – fall back to uniform if nothing found
    # -------------------------------------------------
    if best_seq is None:
        best_seq = [1.0 / n] * n
        best_val = evaluate_sequence(best_seq)

    # -------------------------------------------------
    # 6. Final official evaluation
    # -------------------------------------------------
    final_score = evaluate_sequence(best_seq)
    print(f"\nSearch finished. Reported score = {final_score:.6f}")
    return best_seq
