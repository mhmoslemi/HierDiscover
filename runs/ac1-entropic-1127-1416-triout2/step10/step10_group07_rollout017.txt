def convolution_fft(a,b):
    m = a.size+b.size-1
    nfft = 1 << ((m - 1).bit_length())
    A = np.fft.rfft(a, nfft)
    B = np.fft.rfft(b, nfft)
    C = A*B
    conv = np.fft.irfft(C, nfft)[:m]
    return conv
```

```
def compute_score(a):
    n = a.shape[0]
    conv = convolution_fft(a,a)
    return 2.0 * n * float(conv.max())

"""
Propose a step‑function (non‑negative list of floats) that minimises the
``evaluate_sequence`` objective.

The algorithm
----------------
1. Choose a large length `n` (the longer the vector the lower the theoretical
   bound).  For a 1000 s budget we use `n = 32768` (or a smaller value if
   the budget is tighter).
2. A *block representation* with `B` blocks (here `B = 512`) reduces the
   optimisation dimension: a full vector is obtained by repeating the block
   probabilities `p_j` uniformly over `block_len = n // B` positions and
   scaling by `1 / block_len` so that the whole vector sums to one.
3. Initialise the best candidate with the uniform distribution (objective = 2).
4. Scan a few analytic families (uniform, exponential, power‑law,
   symmetric product) – this is cheap and may already improve over the uniform
   vector.
5. Perform a large number of cheap random draws from a symmetric Dirichlet
   distribution in the low‑dimensional block space and keep the best.
6. Run a simple simulated‑annealing / stochastic hill‑climb directly in the
   block space.  A move transfers a random fraction of probability mass from
   one block to another; the temperature is linearly cooled.  The full vector
   is evaluated with an FFT‑based convolution (`compute_score`).
7. Take the best block vector found and **polish** it with the
   smoothed‑max Adam optimiser (`adam_smooth_optimize`).  The surrogate
   `f_β(a) = (1/β)·log∑exp(β·(a*a)_k)` approaches the true max as β → ∞.
   We start with a modest β and increase it exponentially up to 5·10⁴.
8. Return the final full‑length list.  The harness will call
   ``evaluate_sequence`` on the returned list; we also print the final
   official score for monitoring.

The whole search respects the provided time budget (a small safety margin
is kept).  All helper functions are defined at top‑level and contain no
closed‑over variables.
"""

import time
import numpy as np
from scipy.optimize import differential_evolution  # optional, not used in final code


# ----------------------------------------------------------------------
# Basic numeric utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full aperiodic convolution using real FFT (fast, no Python loops)."""
    m = a.size + b.size - 1
    nfft = next_pow2(m)
    A = np.fft.rfft(a, n=nfft)
    B = np.fft.rfft(b, n=nfft)
    C = A * B
    conv = np.fft.irfft(C, n=nfft)[:m]
    return conv


def compute_score(a: np.ndarray) -> float:
    """
    Exact objective for a probability vector (sum(a) = 1):
        2·n·max(a * a)
    """
    n = a.shape[0]
    conv = convolution_fft(a, a)
    return 2.0 * n * float(conv.max())


def softmax(z: np.ndarray) -> np.ndarray:
    """Numerically stable soft‑max."""
    z_max = np.max(z)
    e = np.exp(z - z_max)
    return e / e.sum()


# ----------------------------------------------------------------------
# Analytic families (used for cheap initial scans)
# ----------------------------------------------------------------------


def uniform_family(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n)


def symmetric_product_family(n: int, alpha: float) -> np.ndarray:
    """a_i ∝ [(i+½)(n‑½‑i)]^(‑α) , symmetric, normalised to sum = 1."""
    i = np.arange(n, dtype=np.float64)
    logprod = np.log(i + 0.5) + np.log(n - 0.5 - i)
    w = np.exp(-alpha * logprod)
    # avoid NaNs for very large α
    w[np.isinf(w)] = np.finfo(np.float64).max
    return w / w.sum()


def left_heavy_power_family(n: int, beta: float) -> np.ndarray:
    """a_i ∝ (i+½)^(‑β) , monotone decreasing, sum = 1."""
    i = np.arange(n, dtype=np.float64)
    w = np.exp(-beta * np.log(i + 0.5))
    return w / w.sum()


def exponential_decay_family(n: int, gamma: float) -> np.ndarray:
    """a_i ∝ exp(-γ·i) , monotone decreasing, sum = 1."""
    i = np.arange(n, dtype=np.float64)
    w = np.exp(-gamma * i)
    return w / w.sum()


def interior_heavy_family(n: int, e: float) -> np.ndarray:
    """
    Symmetric interior‑heavy shape:
        a_i ∝ [(i+½)(n‑½‑i)]^e , e ≥ 0.
    """
    i = np.arange(n, dtype=np.float64)
    w = np.power((i + 0.5) * (n - 0.5 - i), e)
    return w / w.sum()


# ----------------------------------------------------------------------
# Random initialisers for the low‑dimensional block representation
# ----------------------------------------------------------------------


def random_dirichlet_block(B: int, conc: float) -> np.ndarray:
    """Draw a B‑dimensional probability vector from Dirichlet(α)."""
    vec = np.random.gamma(conc, 1.0, size=B)
    vec /= vec.sum()
    return vec


# ----------------------------------------------------------------------
# Adam optimiser on the smoothed max‑convolution surrogate
# ----------------------------------------------------------------------


def adam_smooth_optimize(
    start_a: np.ndarray,
    deadline: float,
    max_iters: int = 200_000,
    beta_start: float = 5.0,
    beta_end: float = 20_000.0,
    lr: float = 0.018,
) -> tuple[np.ndarray, float]:
    """
    Minimise the smoothed surrogate
          f_β(a) = (1/β)·log Σ_k exp(β·(a*a)_k)
    using Adam on the soft‑max parametrisation a = softmax(z).
    β is increased exponentially from beta_start to beta_end.
    """
    eps = 1e-12
    z = np.log(start_a + eps)           # initialise from log‑softmax
    m = np.zeros_like(z)
    v = np.zeros_like(z)

    beta1 = 0.9
    beta2 = 0.999
    eps_adam = 1e-12

    best_a = softmax(z)
    best_score = compute_score(best_a)

    for it in range(1, max_iters + 1):
        if time.time() >= deadline:
            break

        t = it / max_iters
        beta = beta_start * (beta_end / beta_start) ** t

        a = softmax(z)

        conv = convolution_fft(a, a)                # length 2n‑1
        maxc = conv.max()
        exp_w = np.exp(beta * (conv - maxc))        # stabilised
        w = exp_w / exp_w.sum()                     # soft‑max over shifts

        a_rev = a[::-1]
        conv_rev = convolution_fft(a_rev, w)        # length 3n‑2
        grad_a = 2.0 * conv_rev[a.shape[0] - 1 : a.shape[0] - 1 + a.shape[0]]

        dot_ag = np.dot(a, grad_a)
        grad_z = a * grad_a - dot_ag * a

        m = beta1 * m + (1.0 - beta1) * grad_z
        v = beta2 * v + (1.0 - beta2) * (grad_z ** 2)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        z -= lr * m_hat / (np.sqrt(v_hat) + eps_adam)

        # occasional exact check
        if it % 1000 == 0:
            a_tmp = softmax(z)
            sc = compute_score(a_tmp)
            if sc < best_score:
                best_score = sc
                best_a = a_tmp.copy()
                print(f"[{time.time():.1f}]   Adam iter {it:7d} – new best = {best_score:.6f}")

    a_fin = softmax(z)
    sc_fin = compute_score(a_fin)
    if sc_fin < best_score:
        best_score = sc_fin
        best_a = a_fin.copy()
    return best_a, best_score


# ----------------------------------------------------------------------
# Block‑level helper
# ----------------------------------------------------------------------


def expand_blocks(p: np.ndarray, block_len: int) -> np.ndarray:
    """
    Given a block probability vector `p` (sum(p)=1) and a block length,
    return the full length‑`n` probability vector:
        a_i = p_{block(i)} / block_len
    so that sum(a)=1.
    """
    a = np.repeat(p, block_len) / block_len
    return a


# ----------------------------------------------------------------------
# Main search driver
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 12345, budget_s: int = 1000, **_) -> list[float]:
    """
    Search for a non‑negative step‑function (list of floats) that
    minimises ``evaluate_sequence`` within the allocated time budget.
    """
    np.random.seed(seed)
    start_time = time.time()
    # small safety margin for the outer harness
    deadline = start_time + budget_s - 2.0

    # -------------------------------------------------------------
    # Choose length (larger n => lower theoretical bound)
    # -------------------------------------------------------------
    if budget_s >= 800:
        n = 32768
    elif budget_s >= 500:
        n = 16384
    else:
        n = 8192
    print(f"[{time.time():.1f}] Using length n = {n}")

    # block representation
    B = 512 if n >= 16384 else 256
    block_len = n // B
    print(f"[{time.time():.1f}] Block count B = {B}, block_len = {block_len}")

    # -------------------------------------------------------------
    # Global best bookkeeping
    # -------------------------------------------------------------
    best_seq: list[float] | None = None
    best_score = float("inf")

    def maybe_update(vec: np.ndarray, tag: str = "") -> None:
        nonlocal best_seq, best_score
        sc = compute_score(vec)
        if sc < best_score:
            best_score = sc
            best_seq = vec.tolist()
            print(f"[{time.time():.1f}] new best {tag:<20} = {best_score:.6f}")

    # -------------------------------------------------------------
    # Baseline: uniform
    # -------------------------------------------------------------
    a_uni = uniform_family(n)
    maybe_update(a_uni, "uniform")

    # -------------------------------------------------------------
    # 1. cheap scan of analytic families (full‑vector, not block)
    # -------------------------------------------------------------
    # symmetric product family (α from 0 to 6)
    for alpha in np.linspace(0.0, 6.0, 31):
        if time.time() >= deadline - 30.0:
            break
        vec = symmetric_product_family(n, alpha)
        maybe_update(vec, f"sym‑α={alpha:.2f}")

    # left‑heavy power law
    for beta in np.linspace(0.0, 6.0, 31):
        if time.time() >= deadline - 30.0:
            break
        vec = left_heavy_power_family(n, beta)
        maybe_update(vec, f"pow‑β={beta:.2f}")

    # exponential decay
    for gamma in np.linspace(0.0, 8.0, 41):
        if time.time() >= deadline - 30.0:
            break
        vec = exponential_decay_family(n, gamma)
        maybe_update(vec, f"exp‑γ={gamma:.2f}")

    # interior‑heavy family (e ≥ 0)
    for e in np.linspace(0.0, 3.0, 31):
        if time.time() >= deadline - 30.0:
            break
        vec = interior_heavy_family(n, e)
        maybe_update(vec, f"int‑e={e:.2f}")

    # -------------------------------------------------------------
    # 2. Random Dirichlet draws in block space (very cheap)
    # -------------------------------------------------------------
    conc_vals = np.logspace(-2, 1, 15)   # 0.01 … 10
    for conc in conc_vals:
        for _ in range(5):
            if time.time() >= deadline - 30.0:
                break
            p = random_dirichlet_block(B, conc)
            a = expand_blocks(p, block_len)
            maybe_update(a, f"block‑α={conc:.3f}")

    # -------------------------------------------------------------
    # 3. Stochastic hill‑climb (simulated annealing) in block space
    # -------------------------------------------------------------
    # initialise from the current best (project onto blocks)
    if best_seq is not None:
        # average each block to obtain a block probability vector
        cur_blocks = np.array(
            [sum(best_seq[i * block_len:(i + 1) * block_len]) for i in range(B)],
            dtype=np.float64,
        )
        cur_blocks /= cur_blocks.sum()
    else:
        cur_blocks = uniform_family(B)  # fallback

    cur_score = compute_score(expand_blocks(cur_blocks, block_len))
    best_block = cur_blocks.copy()
    best_block_score = cur_score

    # SA parameters
    T0 = best_score * 1e-3  # initial temperature proportional to current score
    max_sa_steps = 300_000
    step_scale = 0.12

    for step in range(1, max_sa_steps + 1):
        if time.time() >= deadline - 20.0:
            break

        # propose a move: transfer some mass from i → j
        i, j = np.random.randint(B, size=2)
        if i == j:
            continue
        delta = step_scale * cur_blocks[i] * np.random.rand()
        cand = cur_blocks.copy()
        cand[i] -= delta
        cand[j] += delta
        if cand[i] < 0.0:
            continue

        # evaluate
        a_cand = expand_blocks(cand, block_len)
        cand_score = compute_score(a_cand)
        delta_sc = cand_score - cur_score

        # linear cooling
        T = max(1e-16, T0 * (1.0 - step / max_sa_steps))

        # Metropolis acceptance
        if delta_sc < 0.0 or np.random.rand() < np.exp(-delta_sc / T):
            cur_blocks = cand
            cur_score = cand_score
            if cand_score < best_block_score:
                best_block_score = cand_score
                best_block = cand.copy()
                print(f"[{time.time():.1f}]   SA block step {step:7d} – new best = {best_block_score:.6f}")
                # also propagate to global best
                maybe_update(a_cand, "SA‑block")
        else:
            # adapt step scale slowly
            step_scale *= 0.9995

        # occasional scale‑adjustment to keep moves sized sensibly
        if step % 5000 == 0:
            step_scale = min(step_scale * 1.02, 0.5)

    # -------------------------------------------------------------
    # 4. Final polishing with high‑β Adam on the full vector
    # -------------------------------------------------------------
    if best_seq is not None and time.time() < deadline - 5.0:
        print(f"[{time.time():.1f}] polishing with high‑β Adam")
        a_start = np.array(best_seq, dtype=np.float64)
        a_refined, sc_refined = adam_smooth_optimize(
            a_start,
            deadline - 2.0,
            max_iters=200_000,
            beta_start=2000.0,
            beta_end=50_000.0,
            lr=0.010,
        )
        maybe_update(a_refined, "Adam‑final")
        if sc_refined < best_score:
            best_score = sc_refined
            best_seq = a_refined.tolist()
            print(f"[{time.time():.1f}] after final Adam = {best_score:.6f}")

    # -------------------------------------------------------------
    # Safeguard – fall back to a modest uniform vector if nothing succeeded
    # -------------------------------------------------------------
    if best_seq is None:
        n_small = 256
        best_seq = [1.0 / n_small] * n_small

    # -------------------------------------------------------------
    # Final official evaluation (the harness will also call evaluate_sequence)
    # -------------------------------------------------------------
    final_score = evaluate_sequence(best_seq)
    print(f"\n[{time.time():.1f}] finished search – final reported score = {final_score:.6f}")

    return best_seq
