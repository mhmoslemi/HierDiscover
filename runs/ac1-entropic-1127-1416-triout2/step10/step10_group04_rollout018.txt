def propose_candidate(seed=12345, budget_s=1000):
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 5     # safety margin

    # Determine n:
    try:
        n = len(height_sequence_1)
        base_seq = np.array(height_sequence_1, dtype=np.float64)
        print(f"Using provided height_sequence_1 of length {n}")
    except Exception:
        n = 2048
        base_seq = None
        print(f"Using default length {n}")

    # Precompute vector of indices
    idx = np.arange(n, dtype=np.float64)

    # start candidate list
    candidates = []

    # uniform
    candidates.append(np.full(n, 1.0 / n, dtype=np.float64))

    # central binomial
    candidates.append(generate_central_binomial(n))

    # power law grid
    for p in np.linspace(0.01, 4.0, 200):
        candidates.append(generate_powerlaw(n, p))

    # symmetric power law (edge heavy)
    for p in np.linspace(0.01, 2.0, 200):
        a = ((idx + 1) * (n - idx)) ** (-p)   # (i+1)*(n-i)
        candidates.append(project_to_simplex(a))

    # beta family grid (α,β)
    alphas = np.linspace(0.1, 2.0, 20)
    betas = np.linspace(0.1, 2.0, 20)
    for a_ in alphas:
        for b_ in betas:
            a = (idx + 1) ** (a_ - 1) * (n - idx) ** (b_ - 1)
            candidates.append(project_to_simplex(a))

    # Evaluate all candidates and keep k best
    best_score = float('inf')
    best_seq = None
    cand_scores = []
    for i, cand in enumerate(candidates):
        # quick score using compute_score (exact)
        score = compute_score(cand)
        cand_scores.append((score, cand))
        if score < best_score:
            best_score = score
            best_seq = cand.copy()
    print(f"Found initial best score {best_score:.6f}")

    # Keep top M candidates for refinement
    cand_scores.sort(key=lambda x: x[0])
    top_candidates = [c for _, c in cand_scores[:5]]

    # Smooth Adam phase on each top candidate
    smooth_deadline = deadline - 300.0   # reserve for later phases
    for i, start_vec in enumerate(top_candidates):
        if time.time() >= smooth_deadline:
            break
        # allocate time proportionally
        remain = smooth_deadline - time.time()
        max_iters = int(min(500_000, remain * 2000))   # approx 0.5k per iteration? 2000 iter per sec
        print(f"Smooth Adam on candidate {i}, max_iters {max_iters}")
        a_opt, sc = smooth_gradient_optimize(start_vec,
                        time_deadline=smooth_deadline,
                        max_iters=max_iters,
                        beta_start=5.0,
                        beta_end=2000.0,
                        step0=0.9)
        if sc < best_score:
            best_score = sc
            best_seq = a_opt.copy()
            print(f"New best after smooth Adam: {best_score:.6f}")

    # Subgradient polishing
    subgrad_deadline = deadline - 60.0
    if time.time() < subgrad_deadline:
        remain = subgrad_deadline - time.time()
        max_iter = int(min(600_000, remain * 2500))
        print(f"Subgradient polishing with up to {max_iter} iterations")
        a_opt, sc = subgrad_optimize(best_seq,
                    time_deadline=subgrad_deadline,
                    max_iter=max_iter,
                    step0=0.05)
        if sc < best_score:
            best_score = sc
            best_seq = a_opt.copy()
            print(f"Best after subgrad: {best_score:.6f}")

    # Random restart bursts (optional)
    restart_deadline = deadline - 30.0
    if time.time() < restart_deadline:
        base = best_seq.copy()
        for r in range(8):
            if time.time() >= restart_deadline:
                break
            pert = base + 0.01 * np.random.randn(n)
            pert = project_to_simplex(pert)
            # short subgrad
            sub_dead = time.time() + 10.0
            a_tmp, s_tmp = subgrad_optimize(
                pert,
                time_deadline=sub_dead,
                max_iter=80_000,
                step0=0.04,
            )
            if s_tmp < best_score:
                best_score = s_tmp
                best_seq = a_tmp.copy()
                print(f"Restart {r} improvement: {best_score:.6f}")

    # Final hill climb (long)
    hill_deadline = deadline - 5.0
    if time.time() < hill_deadline:
        steps = 500_000
        print(f"Final hill climb for up to {steps} steps")
        polished_seq, hill_score = hill_climb_permutation(best_seq.tolist(),
                                    time_deadline=hill_deadline,
                                    max_steps=steps)
        if hill_score < best_score:
            best_score = hill_score
            best_seq = np.array(polished_seq, dtype=np.float64)
            print(f"Hill climb improvement: {best_score:.6f}")

    # final safety fallback
    if best_seq is None:
        best_seq = np.full(n, 1.0/n, dtype=np.float64)
        best_score = compute_score(best_seq)

    # final evaluation using provided evaluate_sequence
    final_score = evaluate_sequence(best_seq.tolist())
    print(f"Search finished, final reported score = {final_score:.6f}")

    return best_seq.tolist()

"""
Non‑negative step‑function optimiser for ``evaluate_sequence``.

Main ideas
-----------
* Exhaustively scan cheap parametric families (power‑law, geometric,
  beta‑like and truncated power‑law).  The score is inexpensive (a single
  FFT), so thousands of candidates can be examined in a few seconds.
* Keep the top few candidates and refine the best one with a long
  smooth‑max (log‑sum‑exp) Adam optimisation.  The temperature β is
  annealed from a modest value up to several thousand, so the surrogate
  more and more focuses on the true peak of the convolution.
* Polish the result with projected sub‑gradient descent on the exact
  max‑convolution objective.
* Finish with a simple stochastic hill‑climb that moves a tiny amount of
  mass between random positions while staying on the probability simplex.
* All phases respect the wall‑clock budget (default 1000 s) and keep the
  best sequence seen.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# FFT based convolution utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Linear convolution via real FFT (full length)."""
    m = a.shape[0] + b.shape[0] - 1
    n_fft = next_pow2(m)
    A = np.fft.rfft(a, n=n_fft)
    B = np.fft.rfft(b, n=n_fft)
    C = A * B
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


def compute_score(a: np.ndarray) -> float:
    """
    Exact objective for a probability vector (sum = 1).
    Returns 2·n·max(conv(a,a)).
    """
    n = a.shape[0]
    conv = convolution_fft(a, a)
    return 2.0 * n * float(conv.max())


# ----------------------------------------------------------------------
# Simplex projection (Euclidean)
# ----------------------------------------------------------------------


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection onto the probability simplex
    {x | sum(x)=1, x>=0}.
    """
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / v.shape[0])

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, v.shape[0] + 1)
    cond = u - cssv / ind > 0
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


# ----------------------------------------------------------------------
# Analytic families (parameterised)
# ----------------------------------------------------------------------


def generate_central_binomial(n: int) -> np.ndarray:
    """C(2i,i)/4^i, normalised to sum 1 (monotonically decreasing)."""
    a = np.empty(n, dtype=np.float64)
    a[0] = 1.0
    for i in range(1, n):
        a[i] = a[i - 1] * (2 * i - 1) / (2.0 * i)
    return project_to_simplex(a)


def generate_powerlaw(n: int, p: float) -> np.ndarray:
    """a_i ∝ (i+1)^(-p), normalised to sum 1."""
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = idx ** (-p)
    return project_to_simplex(a)


def generate_powerlaw_truncated(n: int, p: float, L: int) -> np.ndarray:
    """Power‑law on the first L entries, zeros elsewhere, normalised."""
    a = np.zeros(n, dtype=np.float64)
    a[:L] = np.arange(1, L + 1, dtype=np.float64) ** (-p)
    return project_to_simplex(a)


def generate_geometric(n: int, r: float) -> np.ndarray:
    """Geometric distribution (1‑r) r^i, normalised."""
    i = np.arange(n, dtype=np.float64)
    a = (1.0 - r) * (r ** i)
    return project_to_simplex(a)


def generate_beta_like(n: int, alpha: float, beta: float) -> np.ndarray:
    """
    a_i ∝ (i+1)^{alpha‑1}·(n‑i)^{beta‑1},
    a beta‑type density on {0,…,n‑1}, then projected.
    """
    idx = np.arange(n, dtype=np.float64)
    a = (idx + 1) ** (alpha - 1) * (n - idx) ** (beta - 1)
    return project_to_simplex(a)


def generate_symmetric_powerlaw(n: int, p: float) -> np.ndarray:
    """
    a_i ∝ ((i+1)*(n‑i))^{-p}, a symmetric U‑shaped family.
    """
    idx = np.arange(n, dtype=np.float64)
    a = ((idx + 1) * (n - idx)) ** (-p)
    return project_to_simplex(a)


# ----------------------------------------------------------------------
# Smooth (log‑sum‑exp) optimiser – Adam
# ----------------------------------------------------------------------


def smooth_gradient_optimize(
    start_a: np.ndarray,
    time_deadline: float,
    max_iters: int = 300_000,
    beta_start: float = 5.0,
    beta_end: float = 200.0,
    step0: float = 0.8,
    adam_beta1: float = 0.9,
    adam_beta2: float = 0.999,
    adam_eps: float = 1e-8,
) -> tuple[np.ndarray, float]:
    """
    Adam optimiser for the smoothed max‑convolution objective.
    The surrogate is  (1/β)·log Σ exp(β·conv).
    """
    a = start_a.astype(np.float64).copy()
    n = a.shape[0]
    m1 = np.zeros_like(a)
    m2 = np.zeros_like(a)

    best_a = a.copy()
    best_score = compute_score(a)

    for it in range(1, max_iters + 1):
        if time.time() >= time_deadline:
            break

        # anneal temperature
        t_frac = it / max_iters
        beta = beta_start * (beta_end / beta_start) ** t_frac

        conv = convolution_fft(a, a)
        maxc = conv.max()

        # soft‑max weights (log‑sum‑exp)
        exp_w = np.exp(beta * (conv - maxc))
        w = exp_w / exp_w.sum()

        # gradient of the smooth objective:
        #   ∂/∂a = 2 * (rev(a) * w)   (restricted to length n)
        rev_a = a[::-1]
        grad_full = convolution_fft(rev_a, w)          # length 3n‑2
        grad = 2.0 * grad_full[n - 1 : n - 1 + n]

        # Adam update
        m1 = adam_beta1 * m1 + (1.0 - adam_beta1) * grad
        m2 = adam_beta2 * m2 + (1.0 - adam_beta2) * (grad * grad)
        m1_hat = m1 / (1.0 - adam_beta1 ** it)
        m2_hat = m2 / (1.0 - adam_beta2 ** it)

        step = step0 / np.sqrt(it)
        a_new = a - step * m1_hat / (np.sqrt(m2_hat) + adam_eps)
        a_new = project_to_simplex(a_new)

        a = a_new

        # occasional exact evaluation
        if it % 500 == 0:
            cur_score = compute_score(a)
            if cur_score < best_score:
                best_score = cur_score
                best_a = a.copy()

    final_score = compute_score(a)
    if final_score < best_score:
        best_score, best_a = final_score, a.copy()
    return best_a, best_score


# ----------------------------------------------------------------------
# Exact sub‑gradient optimiser on the true max‑objective
# ----------------------------------------------------------------------


def subgrad_optimize(
    init_a: np.ndarray,
    time_deadline: float,
    max_iter: int = 300_000,
    step0: float = 0.07,
) -> tuple[np.ndarray, float]:
    """Projected sub‑gradient descent on 2·n·max(conv(a,a))."""
    a = init_a.astype(np.float64).copy()
    best_a = a.copy()
    best_score = compute_score(a)
    n = a.shape[0]

    for t in range(1, max_iter + 1):
        if time.time() >= time_deadline:
            break

        conv = convolution_fft(a, a)
        k = int(np.argmax(conv))

        # sub‑gradient for the active convolution entry
        idx = np.arange(n)
        j = k - idx
        mask = (j >= 0) & (j < n)
        g = np.zeros_like(a)
        g[mask] = 2.0 * a[j[mask]]

        step = step0 / np.sqrt(t)
        a_new = a - step * g
        a_new = project_to_simplex(a_new)

        cur_score = compute_score(a_new)
        if cur_score < best_score:
            best_score, best_a = cur_score, a_new.copy()

        a = a_new

        # small jitter each 1000 iterations to escape flats
        if t % 1000 == 0:
            jitter = np.random.randn(n) * (step0 * 0.004 / np.sqrt(t))
            a += jitter
            a = project_to_simplex(a)

    final_score = compute_score(a)
    if final_score < best_score:
        best_score, best_a = final_score, a.copy()
    return best_a, best_score


# ----------------------------------------------------------------------
# Tiny stochastic hill‑climb (only improvements are kept)
# ----------------------------------------------------------------------


def hill_climb_permutation(
    seq,
    time_deadline,
    max_steps: int = 200_000,
) -> tuple[list[float], float]:
    """Random moves (swap or fractional transfer) with simplex projection."""
    a = np.array(seq, dtype=np.float64)
    n = a.shape[0]
    best_a = a.copy()
    best_score = compute_score(best_a)

    step_scale = 0.08  # initial fraction of mass that may be moved

    steps_done = 0
    while steps_done < max_steps and time.time() < time_deadline:
        cand = best_a.copy()
        if np.random.rand() < 0.5:               # swap two entries
            i, j = np.random.randint(n, size=2)
            cand[i], cand[j] = cand[j], cand[i]
        else:                                    # move a fraction from i to j
            i, j = np.random.randint(n, size=2)
            if i != j:
                delta = step_scale * best_a[i]
                cand[i] -= delta
                cand[j] += delta
                if cand[i] < 0:
                    cand[i] = 0.0

        cand = project_to_simplex(cand)
        cur_score = compute_score(cand)

        if cur_score < best_score:
            best_score = cur_score
            best_a = cand
            step_scale = min(step_scale * 1.06, 0.5)
        else:
            step_scale *= 0.97

        steps_done += 1

    return best_a.tolist(), best_score

                start_vec,
                time_deadline=smooth_deadline,
                max_iters=budget_iters,
                beta_start=5.0,
                beta_end=2500.0,      # higher final temperature → tighter max‑focus
                step0=0.9,
            )
            if sc < best_score:
                best_score = sc
                best_seq = a_opt.copy()
                print(f"  New best after smooth Adam (candidate {i}) = {best_score:.6f}")

    # ------------------------------------------------------------------
    # 3. Projected sub‑gradient polishing
    # ------------------------------------------------------------------
    subgrad_deadline = deadline - 70.0
    if time.time() < subgrad_deadline:
        remaining = subgrad_deadline - time.time()
        max_iter = int(min(600_000, remaining * 2500))
        print(f"Sub‑gradient polishing – up to {max_iter} iterations.")
        a_opt, sc = subgrad_optimize(
            best_seq,
            time_deadline=subgrad_deadline,
            max_iter=max_iter,
            step0=0.05,
        )
        if sc < best_score:
            best_score = sc
            best_seq = a_opt.copy()
            print(f"  Improvement after sub‑gradient = {best_score:.6f}")

    # ------------------------------------------------------------------
    # 4. Random‑restart bursts (short sub‑gradient runs)
    # ------------------------------------------------------------------
    restart_deadline = deadline - 30.0
    if time.time() < restart_deadline:
        base = best_seq.copy()
        for r in range(7):
            if time.time() >= restart_deadline:
                break
            pert = base + 0.015 * np.random.randn(n)
            pert = project_to_simplex(pert)
            short_dead = time.time() + 13.0
            a_tmp, sc_tmp = subgrad_optimize(
                pert,
                time_deadline=short_dead,
                max_iter=80_000,
                step0=0.04,
            )
            if sc_tmp < best_score:
                best_score = sc_tmp
                best_seq = a_tmp.copy()
                print(f"  Restart {r} yielded improvement: {best_score:.6f}")

    # ------------------------------------------------------------------
    # 5. Final stochastic hill‑climb
    # ------------------------------------------------------------------
    hill_deadline = deadline - 3.0
    if time.time() < hill_deadline:
        steps = 400_000
        print(f"Final hill‑climb – up to {steps} steps.")
        polished_seq, hill_score = hill_climb_permutation(
            best_seq.tolist(),
            time_deadline=hill_deadline,
            max_steps=steps,
        )
        if hill_score < best_score:
            best_score = hill_score
            best_seq = np.array(polished_seq, dtype=np.float64)
            print(f"  Hill‑climb improvement: {best_score:.6f}")

    # ------------------------------------------------------------------
    # 6. Fallback (should never be needed)
    # ------------------------------------------------------------------
    if best_seq is None:
        best_seq = np.full(n, 1.0 / n, dtype=np.float64)
        best_score = compute_score(best_seq)

    # ------------------------------------------------------------------
    # 7. Final report using the provided ``evaluate_sequence``
    # ------------------------------------------------------------------
    final_score = evaluate_sequence(best_seq.tolist())
    print(f"\nSearch finished – final reported score = {final_score:.6f}")

    # Return a plain Python list (the harness expects a list of floats)
    return best_seq.tolist()
