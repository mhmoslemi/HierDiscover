def propose_candidate(seed=12345, budget_s=1000):
    import numpy as np, time
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 2.0

    # length selection: try to adapt based on time, we choose 2048 by default
    n = 2048

    # initial vector: from global if exists else uniform
    try:
        init_vec = np.array(height_sequence_1, dtype=np.float64)
        if init_vec.shape[0] != n:
            # adjust length by resizing / cropping
            if init_vec.shape[0] > n:
                init_vec = init_vec[:n]
            else:
                # pad with zeros and renormalize
                init_vec = np.pad(init_vec, (0, n - init_vec.shape[0]), mode='constant')
        init_vec = init_vec / init_vec.sum()
    except Exception:
        init_vec = np.full(n, 1.0 / n, dtype=np.float64)

    # function to compute score using evaluate_sequence
    def get_score(seq):
        return evaluate_sequence(seq.tolist())   # convert to list

    best_seq = init_vec.copy()
    best_score = get_score(best_seq)
    print(f"[{time.time():.1f}] start score = {best_score:.6f}")

    # define smoothing target uniform
    target_val = 1.0 / (2 * n - 1)

    # J-optim with Adam using softmax
    def adam_flatten(start_a, max_iters=300_000, lr=0.02, sym_interval=4000):
        n = start_a.size
        # logit initialization
        eps = 1e-12
        z = np.log(start_a + eps)  # shape (n,)
        m = np.zeros_like(z)
        v = np.zeros_like(z)
        beta1, beta2 = 0.9, 0.999
        eps_adam = 1e-12

        best_a = softmax(z)
        best_obj = np.inf
        best_score_local = float('inf')

        for it in range(1, max_iters + 1):
            if time.time() >= deadline:
                break
            a = softmax(z)

            if sym_interval and (it % sym_interval == 0):
                a = (a + a[::-1]) * 0.5
                # re-normalize (softmax ensures sum=1 but after symmetrisation sum may deviate)
                a = a / a.sum()
                z = np.log(a + eps)

            # convolution
            conv = convolution_fft(a, a)   # length 2n-1
            diff = conv - target_val

            # gradient wrt a
            a_rev = a[::-1]
            # compute conv of diff with reversed a
            grad_a = 2.0 * convolution_fft(diff, a_rev)[n-1:n-1 + n]

            # backprop via softmax
            dot_ag = np.dot(a, grad_a)
            grad_z = a * grad_a - dot_ag * a

            # Adam update
            m = beta1 * m + (1.0 - beta1) * grad_z
            v = beta2 * v + (1.0 - beta2) * (grad_z ** 2)
            m_hat = m / (1.0 - beta1 ** it)
            v_hat = v / (1.0 - beta2 ** it)
            z -= lr * m_hat / (np.sqrt(v_hat) + eps_adam)

            if it % 2000 == 0:
                a_tmp = softmax(z)
                a_tmp = (a_tmp + a_tmp[::-1]) * 0.5
                a_tmp = a_tmp / a_tmp.sum()
                # compute objective J and evaluate score
                conv_tmp = convolution_fft(a_tmp, a_tmp)
                obj_val = np.sum((conv_tmp - target_val) ** 2)
                score_tmp = evaluate_sequence(a_tmp.tolist())
                if obj_val < best_obj:
                    best_obj = obj_val
                    best_a = a_tmp.copy()
                    print(f"[{time.time():.1f}] iter {it}, obj {obj_val:.6e}, score {score_tmp:.6f}")
                # also keep best by score
                if score_tmp < best_score_local:
                    best_score_local = score_tmp
                    best_a_score = a_tmp.copy()
        # final evaluate both
        a_final = softmax(z)
        a_final = (a_final + a_final[::-1]) * 0.5
        a_final = a_final / a_final.sum()
        return a_final

    # run flattening

    if sc_flat < best_score:
        best_score = sc_flat
        best_seq = a_flat

    # define function to run smoothed_max_Adam with configurable β schedule etc.
    def adam_smooth_optimize(start_a, max_iters=200_000, beta_start=5.0, beta_end=3000.0, lr=0.015, sym_interval=2000):
        n = start_a.size
        eps = 1e-12
        z = np.log(start_a + eps)
        m = np.zeros_like(z)
        v = np.zeros_like(z)
        beta1, beta2 = 0.9, 0.999
        eps_adam = 1e-12

        best_a = start_a.copy()
        for it in range(1, max_iters + 1):
            if time.time() >= deadline:
                break
            # beta schedule exponential
            t = it / max_iters
            beta = beta_start * (beta_end / beta_start) ** t

            a = softmax(z)
            if sym_interval and (it % sym_interval == 0):
                a = (a + a[::-1]) * 0.5
                a = a / a.sum()
                z = np.log(a + eps)

            conv = convolution_fft(a, a)
            maxc = conv.max()
            # compute softmax weights
            exp_w = np.exp(beta * (conv - maxc))
            w = exp_w / exp_w.sum()
            # gradient w.r.t a: same as earlier but using w as weighting
            a_rev = a[::-1]
            grad_a = 2.0 * convolution_fft(w, a_rev)[n-1:n-1 + n]
            # backprop through softmax
            dot_ag = np.dot(a, grad_a)
            grad_z = a * grad_a - dot_ag * a
            # Adam
            m = beta1 * m + (1.0 - beta1) * grad_z
            v = beta2 * v + (1.0 - beta2) * (grad_z ** 2)

                a_tmp = (a_tmp + a_tmp[::-1]) * 0.5
                a_tmp = a_tmp / a_tmp.sum()
                sc = evaluate_sequence(a_tmp.tolist())
                if sc < best_score_local:
                    best_score_local = sc
                    best_a = a_tmp.copy()
                    print(f"[{time.time():.1f}] smooth iter {it}, score {sc:.6f}")

        # final
        a_fin = softmax(z)
        a_fin = (a_fin + a_fin[::-1]) * 0.5
        a_fin = a_fin / a_fin.sum()
        sc_fin = evaluate_sequence(a_fin.tolist())
        if sc_fin < best_score_local:
            best_score_local = sc_fin
            best_a = a_fin.copy()
        return best_a, best_score_local

    # run final smooth optimization with remaining time
    remaining_iters = 200_000
    a_final, sc_final = adam_smooth_optimize(best_seq, max_iters=200_000, beta_start=10.0, beta_end=5000.0, lr=0.015, sym_interval=2000)
    if sc_final < best_score:
        best_score = sc_final
        best_seq = a_final

    # maybe final explicit symmetrization
    a_sym = (best_seq + best_seq[::-1]) * 0.5

    if sc_sym < best_score:

        best_seq = a_sym
        print(f"[{time.time():.1f}] symmetrisation improvement to {best_score:.6f}")

    # ensure final list format
    final_seq = best_seq.tolist()
    print(f"[{time.time():.1f}] final score = {evaluate_sequence(final_seq):.6f}")
    return final_seq

"""
Propose a step‑function (non‑negative vector on the simplex) that minimises
evaluate_sequence(seq).  The algorithm consists of three phases:

1.  Initialise from the best known construction (global height_sequence_1)
    or from the uniform distribution.
2.  “Flattening” phase – minimise the L2 distance between the auto‑convolution
    and the uniform target 1/(2 n‑1) using Adam on the soft‑max parametrisation.
    This pushes the convolution towards a constant shape and usually reduces
    the peak dramatically.
3.  Final refinement – minimise a smooth approximation of the max‑norm
    (log‑sum‑exp) with a second Adam run (increasing β to a large value) and
    occasional symmetrisation.

Both phases respect the wall‑clock budget (default 1000 s) and keep the
vector on the probability simplex via the soft‑max trick.
The returned list is ready for evaluate_sequence().
"""

import time
import numpy as np
from typing import List

# ----------------------------------------------------------------------
# Core utilities (FFT based convolution, softmax, etc.)
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Return the smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full aperiodic convolution via real FFT (zero‑padded)."""
    m = a.size + b.size - 1
    nfft = next_pow2(m)
    A = np.fft.rfft(a, n=nfft)
    B = np.fft.rfft(b, n=nfft)
    C = A * B
    conv = np.fft.irfft(C, nfft)[:m]
    return conv


def softmax(z: np.ndarray) -> np.ndarray:
    """Numerically stable soft‑max (returns a probability vector)."""
    z_max = np.max(z)
    e = np.exp(z - z_max)
    return e / e.sum()


def compute_score(a: np.ndarray) -> float:
    """Fast evaluation of the objective (sum(a)=1 is assumed)."""
    n = a.shape[0]
    conv = convolution_fft(a, a)
    return 2.0 * n * float(conv.max())


# ----------------------------------------------------------------------
# Adam optimiser for the L2‑flattening objective
# ----------------------------------------------------------------------


def adam_flatten(start_a: np.ndarray,
                 deadline: float,
                 max_iters: int = 300_000,
                 lr: float = 0.020,
                 sym_interval: int = 4000) -> np.ndarray:
    """
    Minimises   Σ_k ( (a∗a)_k – μ )²   with μ = 1/(2·n‑1)
    using Adam on the soft‑max parametrisation.
    Periodically symmetrises the iterate (a ← (a+a[::-1])/2).
    """
    n = start_a.size
    eps = 1e-12
    # initialise logits so that softmax(z)=start_a
    z = np.log(start_a + eps)

    m = np.zeros_like(z)
    v = np.zeros_like(z)
    beta1, beta2 = 0.9, 0.999
    eps_adam = 1e-12

    target = 1.0 / (2 * n - 1)          # uniform target for the convolution

    best_a = softmax(z)
    best_obj = float("inf")
    best_score = compute_score(best_a)

    for it in range(1, max_iters + 1):
        if time.time() >= deadline:
            break

        a = softmax(z)

        # occasional exact symmetrisation
        if sym_interval and (it % sym_interval == 0):
            a = (a + a[::-1]) * 0.5
            a = a / a.sum()
            z = np.log(a + eps)

        # ----- gradient of the L2 objective -----
        conv = convolution_fft(a, a)                     # (a∗a)
        diff = conv - target                              # deviation from uniform

        a_rev = a[::-1]
        # ∇_a Σ (conv‑target)²  = 2· (diff ▷ a_rev)
        grad_a = 2.0 * convolution_fft(diff, a_rev)[n - 1:n - 1 + n]

        # back‑prop through soft‑max
        dot_ag = np.dot(a, grad_a)
        grad_z = a * grad_a - dot_ag * a

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad_z
        v = beta2 * v + (1.0 - beta2) * (grad_z ** 2)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        z -= lr * m_hat / (np.sqrt(v_hat) + eps_adam)

        # logging every few thousand steps
        if it % 2000 == 0:
            a_tmp = softmax(z)
            a_tmp = (a_tmp + a_tmp[::-1]) * 0.5
            a_tmp = a_tmp / a_tmp.sum()
            obj_val = np.sum((convolution_fft(a_tmp, a_tmp) - target) ** 2)
            scr = compute_score(a_tmp)
            if obj_val < best_obj:
                best_obj = obj_val
                best_a = a_tmp.copy()
                print(f"[{time.time():.1f}] flatten iter {it:6d}  obj={obj_val:.3e}  score={scr:.6f}")

            if scr < best_score:
                best_score = scr
                best_a = a_tmp.copy()

    # final symmetrised version
    a_fin = softmax(z)
    a_fin = (a_fin + a_fin[::-1]) * 0.5
    a_fin = a_fin / a_fin.sum()
    return a_fin


# ----------------------------------------------------------------------
# Adam optimiser for the smoothed‑max (log‑sum‑exp) objective
# ----------------------------------------------------------------------


def adam_smooth(start_a: np.ndarray,
                deadline: float,
                max_iters: int = 200_000,
                beta_start: float = 10.0,
                beta_end: float = 5000.0,
                lr: float = 0.015,
                sym_interval: int = 2000) -> np.ndarray:
    """
    Minimises the smooth approximation of the max‑norm:
        f_β(a) = (1/β)·log Σ_k exp(β·(a∗a)_k)
    using Adam.  β follows an exponential schedule from beta_start
    to beta_end.  Symmetrisation every `sym_interval` steps.
    """
    n = start_a.size
    eps = 1e-12
    z = np.log(start_a + eps)

    m = np.zeros_like(z)
    v = np.zeros_like(z)
    beta1, beta2 = 0.9, 0.999
    eps_adam = 1e-12

    best_a = softmax(z)
    best_score = compute_score(best_a)

    for it in range(1, max_iters + 1):
        if time.time() >= deadline:
            break

        # exponential β schedule
        t = it / max_iters
        beta = beta_start * (beta_end / beta_start) ** t

        a = softmax(z)

        if sym_interval and (it % sym_interval == 0):
            a = (a + a[::-1]) * 0.5
            a = a / a.sum()
            z = np.log(a + eps)

        conv = convolution_fft(a, a)
        maxc = conv.max()
        # soft‑max weights over the convolution entries
        exp_w = np.exp(beta * (conv - maxc))
        w = exp_w / exp_w.sum()

        a_rev = a[::-1]
        # gradient of the smoothed max w.r.t. a
        grad_a = 2.0 * convolution_fft(w, a_rev)[n - 1:n - 1 + n]

        # back‑prop through soft‑max
        dot_ag = np.dot(a, grad_a)
        grad_z = a * grad_a - dot_ag * a

        # Adam step
        m = beta1 * m + (1.0 - beta1) * grad_z
        v = beta2 * v + (1.0 - beta2) * (grad_z ** 2)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        z -= lr * m_hat / (np.sqrt(v_hat) + eps_adam)

        if it % 2000 == 0:
            a_tmp = softmax(z)
            a_tmp = (a_tmp + a_tmp[::-1]) * 0.5
            a_tmp = a_tmp / a_tmp.sum()
            sc = compute_score(a_tmp)
            if sc < best_score:
                best_score = sc
                best_a = a_tmp.copy()
                print(f"[{time.time():.1f}] smooth iter {it:6d}  score={sc:.6f}")

    a_fin = softmax(z)
    a_fin = (a_fin + a_fin[::-1]) * 0.5
    a_fin = a_fin / a_fin.sum()
    fin_sc = compute_score(a_fin)
    if fin_sc < best_score:
        best_a = a_fin.copy()
        best_score = fin_sc
    return best_a

def propose_candidate(seed: int = 12345, budget_s: int = 1000, **kwargs) -> List[float]:
    """
    Search for a non‑negative step‑function sequence that minimises
    ``evaluate_sequence``.  The routine respects the wall‑clock budget
    and returns a plain Python list of floats.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 2.0          # safety margin

    # ------------------------------------------------------------------
    # 0. Choose problem size
    # ------------------------------------------------------------------
    n = 2048                # length that gave the previous best

    # ------------------------------------------------------------------
    # 1. Initialise (use previously discovered sequence if available)
    # ------------------------------------------------------------------
    try:
        init = np.array(height_sequence_1, dtype=np.float64)   # type: ignore
        if init.shape[0] != n:
            if init.shape[0] > n:
                init = init[:n]
            else:
                init = np.pad(init, (0, n - init.shape[0]), mode='constant')
        init = init / init.sum()
        print(f"[{time.time():.1f}] loaded height_sequence_1 (len={init.shape[0]})")
    except Exception:
        init = np.full(n, 1.0 / n, dtype=np.float64)            # uniform fallback
        print(f"[{time.time():.1f}] using uniform start (len={n})")

    best_seq = init.copy()
    best_score = compute_score(best_seq)
    print(f"[{time.time():.1f}] initial score = {best_score:.6f}")

    # ------------------------------------------------------------------
    # 2. Flattening phase (L2 minimisation of a∗a vs uniform)
    # ------------------------------------------------------------------
    if time.time() < deadline - 100.0:          # leave room for the final phase
        print(f"[{time.time():.1f}] starting L2‑flattening")
        flat = adam_flatten(best_seq,
                            deadline - 80.0,
                            max_iters=300_000,
                            lr=0.020,
                            sym_interval=4000)
        sc_flat = compute_score(flat)
        print(f"[{time.time():.1f}] after flattening score = {sc_flat:.6f}")
        if sc_flat < best_score:
            best_score = sc_flat
            best_seq = flat.copy()

    # ------------------------------------------------------------------
    # 3. Final smooth‑max refinement
    # ------------------------------------------------------------------
    if time.time() < deadline - 30.0:
        print(f"[{time.time():.1f}] final smooth‑max optimisation")
        refined = adam_smooth(best_seq,
                              deadline - 10.0,
                              max_iters=200_000,
                              beta_start=10.0,
                              beta_end=5000.0,
                              lr=0.015,
                              sym_interval=2000)
        sc_ref = compute_score(refined)
        print(f"[{time.time():.1f}] after smooth‑max score = {sc_ref:.6f}")
        if sc_ref < best_score:
            best_score = sc_ref
            best_seq = refined.copy()

    # ------------------------------------------------------------------
    # 4. One more exact symmetrisation (often helpful)
    # ------------------------------------------------------------------
    a_sym = (best_seq + best_seq[::-1]) * 0.5
    a_sym = a_sym / a_sym.sum()
    sc_sym = compute_score(a_sym)
    if sc_sym < best_score:
        best_score = sc_sym
        best_seq = a_sym
        print(f"[{time.time():.1f}] symmetrisation improved score to {sc_sym:.6f}")

    # ------------------------------------------------------------------
    # 5. Return the result as a plain list (evaluate_sequence will see
    #    the same value because the vector sums to 1 and contains
    #    only non‑negative numbers).
    # ------------------------------------------------------------------
    final_seq = best_seq.tolist()
    final_score = compute_score(best_seq)
    print(f"\n[{time.time():.1f}] finished – final reported score = {final_score:.6f}")
    return final_seq
