def softmax_adam_optimize(start_a, time_deadline, max_iters=500000,
                          beta_start=5.0, beta_end=500.0,
                          step0=0.5,
                          adam_beta1=0.9, adam_beta2=0.999, adam_eps=1e-8):
    n = start_a.shape[0]
    eps = 1e-16
    z = np.log(start_a + eps)
    # shift to zero mean
    z = z - np.mean(z)
    m = np.zeros_like(z)
    v = np.zeros_like(z)
    best_a = softmax(z)
    best_score = compute_score(best_a)
    for it in range(1, max_iters+1):
        if time.time() >= time_deadline:
            break
        # compute a from logits
        a = softmax(z)
        # conv
        conv = convolution_fft(a, a)   # length 2n-1
        # temperature
        t_frac = it / max_iters
        beta = beta_start * (beta_end / beta_start) ** t_frac
        # compute softmax weights w
        maxc = conv.max()
        # stable exp
        exp_w = np.exp(beta * (conv - maxc))
        w = exp_w / exp_w.sum()
        # gradient wrt a
        rev_a = a[::-1]
        # compute conv of rev_a and w
        grad_full = convolution_fft(rev_a, w)
        grad = 2.0 * grad_full[n-1:n-1+n]  # shape n
        # gradient wrt logits
        # softmax Jacobian applied: grad_z = a * (grad - dot(a,grad))
        dot = np.dot(a, grad)
        grad_z = a * (grad - dot)
        # Adam update
        m = adam_beta1 * m + (1 - adam_beta1) * grad_z
        v = adam_beta2 * v + (1 - adam_beta2) * (grad_z * grad_z)
        m_hat = m / (1 - adam_beta1 ** it)
        v_hat = v / (1 - adam_beta2 ** it)
        step = step0 / np.sqrt(it)  # similar schedule
        z = z - step * m_hat / (np.sqrt(v_hat) + adam_eps)
        # record best every 200 iters optionally
        if it % 200 == 0:
            cur_score = compute_score(a)
            if cur_score < best_score:
                best_score = cur_score
                best_a = a.copy()
    # final evaluation
    final_a = softmax(z)
    final_score = compute_score(final_a)
    if final_score < best_score:
        best_score = final_score
        best_a = final_a.copy()
    return best_a, best_score

"""
Non‑negative step‑function optimiser.

The search proceeds in several stages:
  1️⃣  Build a collection of analytic seed vectors (uniform, central‑binomial,
      power‑law, geometric, arcsine‑like) and keep the best according to the
      exact objective 2·n·max(conv(a,a)).
  2️⃣  Refine the best seed with a *soft‑max* Adam optimiser.  The distribution
      a is represented by logits z (a = softmax(z)), which guarantees positivity
      and the simplex constraint without explicit projection.  A log‑sum‑exp
      surrogate of the max‑convolution is annealed (β from 5 to ≈500) and an
      Adam update is applied to the logits.
  3️⃣  Polish the result with projected sub‑gradient descent on the true
      max‑convolution objective.
  4️⃣  Final stochastic hill‑climbing (simulated‑annealing style) moves a
      small amount of mass between random positions, accepting uphill moves
      with a temperature schedule.
  5️⃣  Enforce symmetry (a_i = a_{n‑1‑i}) – optimal solutions are believed to be
      symmetric – and project back onto the simplex.
The best vector found is returned as a plain list of floats.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Basic utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """
    Full linear convolution via real FFT.
    Returns array of length len(a)+len(b)-1.
    """
    m = a.shape[0] + b.shape[0] - 1
    n_fft = next_pow2(m)
    A = np.fft.rfft(a, n=n_fft)
    B = np.fft.rfft(b, n=n_fft)
    C = A * B
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


def compute_score(a: np.ndarray) -> float:
    """
    Exact objective for a vector already on the simplex (sum = 1).
    Returns 2·n·max(conv(a,a)).
    """
    n = a.shape[0]
    conv = convolution_fft(a, a)
    return float(2.0 * n * conv.max())


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of v onto the probability simplex:
        sum(v) = 1, v >= 0.
    """
    if np.all(v <= 0):
        # uniform fallback
        return np.full_like(v, 1.0 / v.shape[0])

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, v.shape[0] + 1)
    cond = u - cssv / ind > 0
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def softmax(z: np.ndarray) -> np.ndarray:
    """Numerically stable softmax."""
    z_max = np.max(z)
    exp_z = np.exp(z - z_max)
    return exp_z / exp_z.sum()


# ----------------------------------------------------------------------
# Analytic families for initial guesses
# ----------------------------------------------------------------------


def generate_central_binomial(n: int) -> np.ndarray:
    """a_i = C(2i,i) / 4^i, normalised to the simplex (monotone decreasing)."""
    a = np.empty(n, dtype=np.float64)
    a[0] = 1.0
    for i in range(1, n):
        a[i] = a[i - 1] * (2.0 * i - 1) / (2.0 * i)
    return project_to_simplex(a)


def generate_powerlaw(n: int, p: float) -> np.ndarray:
    """a_i ∝ (i+1)^(-p), normalised to the simplex."""
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = idx ** (-p)
    return project_to_simplex(a)


def generate_geometric(n: int, r: float) -> np.ndarray:
    """Geometric distribution a_i = (1‑r)·r^i, normalised (exactly sum = 1 if r<1)."""
    i = np.arange(n, dtype=np.float64)
    a = (1.0 - r) * (r ** i)
    return project_to_simplex(a)


# ----------------------------------------------------------------------
# Softmax‑Adam optimiser on a smoothed max‑convolution objective
# ----------------------------------------------------------------------


def softmax_adam_optimize(
    start_a: np.ndarray,
    time_deadline: float,
    max_iters: int = 500_000,
    beta_start: float = 5.0,
    beta_end: float = 500.0,
    step0: float = 0.8,
    adam_beta1: float = 0.9,
    adam_beta2: float = 0.999,
    adam_eps: float = 1e-8,
) -> tuple[np.ndarray, float]:
    """
    Adam optimiser on the log‑sum‑exp (smooth‑max) surrogate.
    The distribution a is parameterised as a = softmax(z) to keep positivity
    and the simplex constraint automatically.
    """
    n = start_a.shape[0]
    eps = 1e-16

    # initialise logits from the start vector
    z = np.log(start_a + eps)
    z = z - np.mean(z)  # zero‑mean for numerical stability

    m1 = np.zeros_like(z)
    m2 = np.zeros_like(z)

    best_a = softmax(z)
    best_score = compute_score(best_a)

    for it in range(1, max_iters + 1):
        if time.time() >= time_deadline:
            break

        # current probability vector
        a = softmax(z)

        # full auto‑convolution
        conv = convolution_fft(a, a)                 # length 2n‑1

        # annealed temperature β
        t_frac = it / max_iters
        beta = beta_start * (beta_end / beta_start) ** t_frac

        # smooth max via log‑sum‑exp (stable)
        maxc = conv.max()
        exp_w = np.exp(beta * (conv - maxc))
        w = exp_w / exp_w.sum()                     # softmax weights, sum = 1

        # gradient w.r.t. a of the smoothed objective
        rev_a = a[::-1]
        grad_full = convolution_fft(rev_a, w)       # length 3n‑2
        grad_a = 2.0 * grad_full[n - 1:n - 1 + n]   # slice to length n

        # gradient w.r.t. logits (softmax Jacobian)
        dot = np.dot(a, grad_a)
        grad_z = a * (grad_a - dot)

        # Adam update on logits
        m1 = adam_beta1 * m1 + (1.0 - adam_beta1) * grad_z
        m2 = adam_beta2 * m2 + (1.0 - adam_beta2) * (grad_z * grad_z)
        m1_hat = m1 / (1.0 - adam_beta1 ** it)
        m2_hat = m2 / (1.0 - adam_beta2 ** it)
        step = step0 / np.sqrt(it)
        z -= step * m1_hat / (np.sqrt(m2_hat) + adam_eps)

        # occasional bookkeeping
        if it % 400 == 0:
            cur_score = compute_score(a)
            if cur_score < best_score:
                best_score = cur_score
                best_a = a.copy()

    # final evaluation
    final_a = softmax(z)
    final_score = compute_score(final_a)
    if final_score < best_score:
        best_score, best_a = final_score, final_a.copy()

    return best_a, best_score


# ----------------------------------------------------------------------
# Exact sub‑gradient optimiser on the true max‑convolution objective
# ----------------------------------------------------------------------


def subgrad_optimize(
    init_a: np.ndarray,
    time_deadline: float,
    max_iter: int = 300_000,
    step0: float = 0.07,
) -> tuple[np.ndarray, float]:
    """
    Projected sub‑gradient descent on 2·n·max(conv(a,a)).
    The sub‑gradient is taken at the index attaining the current maximum.
    """
    a = init_a.astype(np.float64).copy()
    n = a.shape[0]

    best_a = a.copy()
    best_score = compute_score(a)

    for t in range(1, max_iter + 1):
        if time.time() >= time_deadline:
            break

        conv = convolution_fft(a, a)
        k = int(np.argmax(conv))

        # sub‑gradient for the active entry
        idx = np.arange(n)
        j = k - idx
        mask = (j >= 0) & (j < n)
        g = np.zeros_like(a)
        g[mask] = 2.0 * a[j[mask]]

        step = step0 / np.sqrt(t)
        a_new = a - step * g
        a_new = project_to_simplex(a_new)

        cur_score = compute_score(a_new)
        if cur_score < best_score:
            best_score, best_a = cur_score, a_new.copy()

        a = a_new

        # occasional jitter to escape flat regions
        if t % 1000 == 0:
            jitter = np.random.randn(n) * (step0 * 0.004 / np.sqrt(t))
            a += jitter
            a = project_to_simplex(a)

    final_score = compute_score(a)
    if final_score < best_score:
        best_score, best_a = final_score, a.copy()

    return best_a, best_score


# ----------------------------------------------------------------------
# Simulated‑annealing hill‑climb (mass‑transfer moves)
# ----------------------------------------------------------------------


def simulated_annealing_hill_climb(
    start_a: np.ndarray,
    time_deadline: float,
    max_steps: int = 400_000,
    init_temp: float = 0.02,
    final_temp: float = 2e-4,
) -> tuple[np.ndarray, float]:
    """
    Tiny stochastic hill‑climb that occasionally accepts uphill moves.
    The temperature drops exponentially from ``init_temp`` to ``final_temp``.
    """
    a = start_a.copy()
    n = a.shape[0]

    cur_score = compute_score(a)
    best_a = a.copy()
    best_score = cur_score

    step = 0
    while step < max_steps and time.time() < time_deadline:
        # linearised progress for temperature schedule
        t_frac = step / max_steps
        temp = init_temp * (final_temp / init_temp) ** t_frac

        # propose a small mass transfer
        i, j = np.random.randint(n, size=2)
        if i != j:
            delta = temp * a[i]                     # amount proportional to current temperature
            if delta > 0:
                cand = a.copy()
                cand[i] -= delta
                cand[j] += delta
                # clip negatives that may arise from numeric issues
                cand[i] = max(cand[i], 0.0)
                cand = project_to_simplex(cand)

                cand_score = compute_score(cand)

                # Metropolis acceptance
                if cand_score < cur_score or np.random.rand() < np.exp(-(cand_score - cur_score) / max(temp, 1e-12)):
                    a = cand
                    cur_score = cand_score
                    if cur_score < best_score:
                        best_score, best_a = cur_score, a.copy()

        # every few thousand steps try a random swap of two entries (helps exploration)
        if step % 5000 == 0:
            i, j = np.random.randint(n, size=2)
            cand = a.copy()
            cand[i], cand[j] = cand[j], cand[i]
            cand = project_to_simplex(cand)
            cand_score = compute_score(cand)
            if cand_score < cur_score or np.random.rand() < np.exp(-(cand_score - cur_score) / max(temp, 1e-12)):
                a = cand
                cur_score = cand_score
                if cur_score < best_score:
                    best_score, best_a = cur_score, a.copy()

        step += 1

    return best_a, best_score

# Main entry point required by the harness
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 12345, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative sequence that minimises ``evaluate_sequence``.
    Returns a plain Python list of floats (already on the simplex).
    """
    np.random.seed(seed)
    start_time = time.time()
    # Leave a small safety margin for the surrounding harness
    deadline = start_time + budget_s - 3.0

    # ------------------------------------------------------------------
    # Determine problem size (length of the sequence)
    # ------------------------------------------------------------------
    try:
        n = len(height_sequence_1)                # type: ignore[name-defined]
        init_candidates = [np.array(height_sequence_1, dtype=np.float64)]
        print(f"[Info] Using provided height_sequence_1 of length {n}.")
    except Exception:
        n = 2048
        init_candidates = []
        print(f"[Info] No external sequence provided – using length {n}.")

    # ------------------------------------------------------------------
    # 1️⃣ Analytic starters
    # ------------------------------------------------------------------
    init_candidates.append(np.full(n, 1.0 / n, dtype=np.float64))          # uniform
    init_candidates.append(generate_central_binomial(n))                  # central‑binomial
    for p in np.linspace(0.3, 1.0, 9):                                    # power‑law family
        init_candidates.append(generate_powerlaw(n, float(p)))
    for r in [0.97, 0.98, 0.985, 0.99, 0.992, 0.995]:                    # geometric family
        init_candidates.append(generate_geometric(n, float(r)))
    # arcsine‑like shape a_i ∝ 1/√((i+1)*(n-i))
    idx = np.arange(n, dtype=np.float64)
    a_arc = 1.0 / np.sqrt((idx + 1) * (n - idx))
    init_candidates.append(project_to_simplex(a_arc))

    # evaluate starters
    best_seq = None
    best_val = float("inf")
    for i, cand in enumerate(init_candidates):
        cand = project_to_simplex(cand)
        val = compute_score(cand)
        print(f"[Starter {i:02d}] score = {val:.6f}")
        if val < best_val:
            best_val = val
            best_seq = cand.copy()
    print(f"[Info] Best starter score = {best_val:.6f}")

    # ------------------------------------------------------------------
    # 2️⃣ Softmax‑Adam refinement (largest time chunk)
    # ------------------------------------------------------------------
    soft_deadline = deadline - 250.0
    if time.time() < soft_deadline:
        remaining = soft_deadline - time.time()
        # estimate ~3000 iterations per second for the 2048‑dim problem
        max_iters = int(min(1_200_000, remaining * 3000))
        print(f"[Phase] Softmax‑Adam up to {max_iters} iterations.")
        a_soft, val_soft = softmax_adam_optimize(
            best_seq,
            time_deadline=soft_deadline,
            max_iters=max_iters,
            beta_start=5.0,
            beta_end=500.0,
            step0=0.8,
        )
        if val_soft < best_val:
            best_val = val_soft
            best_seq = a_soft.copy()
            print(f"[Update] Softmax‑Adam improved to {best_val:.6f}")

    # ------------------------------------------------------------------
    # 3️⃣ Projected sub‑gradient polishing
    # ------------------------------------------------------------------
    sub_deadline = deadline - 70.0
    if time.time() < sub_deadline:
        remaining = sub_deadline - time.time()
        max_iter = int(min(800_000, remaining * 3000))
        print(f"[Phase] Sub‑gradient polishing up to {max_iter} iterations.")
        a_sub, val_sub = subgrad_optimize(
            best_seq,
            time_deadline=sub_deadline,
            max_iter=max_iter,
            step0=0.07,
        )
        if val_sub < best_val:
            best_val = val_sub
            best_seq = a_sub.copy()
            print(f"[Update] Sub‑gradient improved to {best_val:.6f}")

    # ------------------------------------------------------------------
    # 4️⃣ Simulated‑annealing hill climbing (final fine‑tuning)
    # ------------------------------------------------------------------
    hill_deadline = deadline - 2.0
    if time.time() < hill_deadline:
        print("[Phase] Simulated‑annealing hill climb.")
        a_hill, val_hill = simulated_annealing_hill_climb(
            best_seq,
            time_deadline=hill_deadline,
        )
        if val_hill < best_val:
            best_val = val_hill
            best_seq = a_hill.copy()
            print(f"[Update] Hill climb improved to {best_val:.6f}")

    # ------------------------------------------------------------------
    # 5️⃣ Enforce symmetry (a_i = a_{n‑1‑i}) – the optimum is believed to be symmetric.
    # ------------------------------------------------------------------
    if best_seq is not None:
        sym = (best_seq + best_seq[::-1]) / 2.0
        best_seq = project_to_simplex(sym)

    # ------------------------------------------------------------------
    # 6️⃣ Final evaluation via the provided ``evaluate_sequence`` (security wrapper).
    # ------------------------------------------------------------------
    final_score = evaluate_sequence(best_seq.tolist())
    print(f"\n[Result] Search finished. Final reported score = {final_score:.6f}")

    return best_seq.tolist()
