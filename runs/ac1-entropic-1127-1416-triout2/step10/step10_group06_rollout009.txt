def lp_refine(f, step_t, iters, deadline):
    # f normalized sum=1
    best_f = f.copy()
    best_score = compute_score(best_f)
    for i in range(iters):
        if time.time() >= deadline - 1:
            break
        M = convolution_fft(f,f).max()
        # Build constraint matrix
        n = f.shape[0]
        c = np.concatenate([f, np.zeros(n-1)], dtype=np.float64)
        r = np.concatenate(([f[0]], np.zeros(n-1)), dtype=np.float64)
        A = scipy.linalg.toeplitz(c, r)
        b_ub = np.full(2*n-1, M, dtype=np.float64)
        # Solve LP
        res = scipy.optimize.linprog(c=-np.ones(n), A_ub=A, b_ub=b_ub, bounds=[(0,None)]*n, method='highs')
        if not res.success:
            print(f"LP failed: {res.message}")
            break
        b = res.x
        sum_b = b.sum()
        if sum_b == 0:
            break
        # Normalize b to sum1
        g = b / sum_b
        # Mix
        f = (1 - step_t) * f + step_t * g
        # Enforce simplex (should remain sum1) but due to numeric may deviate; project.
        f = project_to_simplex(f)
        cur_score = compute_score(f)
        if cur_score < best_score:
            best_score = cur_score
            best_f = f.copy()
            print(f"[{time.time():.1f}] LP iteration {i+1} improved score = {best_score:.6f}")
        # maybe shrink step_t
        step_t = max(step_t * 0.9, 0.01)

    return best_f, best_score

def upsample_sequence(a_small, n_big):
    """Interpolate a_small to length n_big using linear interpolation preserving sum=1."""
    n_small = a_small.shape[0]
    x_small = np.linspace(0, 1, n_small)
    x_big = np.linspace(0, 1, n_big)
    a_big = np.interp(x_big, x_small, a_small)
    # ensure non-negative (interp may produce negative if small noise? but likely non-negative)
    a_big = np.maximum(a_big, 0.0)
    # renormalize to sum=1
    a_big /= a_big.sum()
    return a_big

"""
Propose a step‑function sequence that minimises `evaluate_sequence`.

Algorithm
---------
* Work with two resolutions – a small one (≈2048) where a cheap
  linear‑programming refinement is cheap, and a larger one (≈8192)
  where we run smooth‑surrogate L‑BFGS and a long mirror‑sub‑gradient
  descent.
* A random start (families, sparse or monotone) is refined on the
  small scale:
    – a few hundred‑thousand mirror‑sub‑gradient steps,
    – a short L‑BFGS on the log‑sum‑exp surrogate,
    – 3‑5 iterations of an LP that maximises the total mass of a
      partner vector `b` under the convolution bound `f*b ≤ M`.
      The partner is mixed into `f` (large step‐size) which typically
      lowers the peak of the self‑convolution.
* The improved small‑scale vector is linearly interpolated to a larger
  size, re‑normalised and then refined again (L‑BFGS → mirror sub‑grad →
  optional hill‑climb).
* Symmetrisation and a final call to `evaluate_sequence` give the
  returned list of floats.
* All stages respect the global wall‑clock deadline (`budget_s` seconds).

The code uses only NumPy and SciPy (if available) and runs with a
single CPU.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Optional SciPy imports
# ----------------------------------------------------------------------
try:
    import scipy.optimize as _opt
    import scipy.linalg as _linalg
    _HAS_SCIPY = True
except Exception:  # pragma: no cover
    _HAS_SCIPY = False

# ----------------------------------------------------------------------
# Basic FFT utilities
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """
    Linear (aperiodic) convolution via real FFT.
    Returns the full convolution of length len(a)+len(b)-1.
    """
    m = a.size + b.size - 1
    nfft = _next_pow2(m)
    A = np.fft.rfft(a, n=nfft)
    B = np.fft.rfft(b, n=nfft)
    C = A * B
    conv = np.fft.irfft(C, n=nfft)[:m]
    return conv


def compute_score(a: np.ndarray) -> float:
    """
    Exact objective for a probability vector (sum(a)=1):
        score = 2·n·max(a * a)
    This matches `evaluate_sequence` up to the normalisation.
    """
    n = a.shape[0]
    conv = convolution_fft(a, a)
    return 2.0 * n * float(conv.max())


# ----------------------------------------------------------------------
# Simplex utilities
# ----------------------------------------------------------------------


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of a vector onto the probability simplex."""
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / v.shape[0])

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, v.shape[0] + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.where(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def softmax(z: np.ndarray) -> np.ndarray:
    """Numerically stable soft‑max (maps ℝⁿ → simplex)."""
    z_max = np.max(z)
    e = np.exp(z - z_max)
    return e / e.sum()


# ----------------------------------------------------------------------
# Analytic families (all return a probability vector)
# ----------------------------------------------------------------------


def symmetric_product_family(n: int, alpha: float) -> np.ndarray:
    """U‑shaped: a_i ∝ [(i+½)(n‑½‑i)]^(‑α)."""
    i = np.arange(n, dtype=np.float64)
    logprod = np.log(i + 0.5) + np.log(n - 0.5 - i)
    w = np.exp(-alpha * logprod)
    # Guard against overflow at the very ends
    w[np.isinf(w)] = np.finfo(np.float64).max
    a = w / w.sum()
    return a


def left_heavy_power_family(n: int, beta: float) -> np.ndarray:
    """Monotone decreasing power law: a_i ∝ (i+½)^(‑β)."""
    i = np.arange(n, dtype=np.float64)
    w = np.exp(-beta * np.log(i + 0.5))
    a = w / w.sum()
    return a


def exponential_decay_family(n: int, gamma: float) -> np.ndarray:
    """Simple exponential decay: a_i ∝ exp(‑γ·i)."""
    i = np.arange(n, dtype=np.float64)
    w = np.exp(-gamma * i)
    a = w / w.sum()
    return a


def beta_distribution_family(n: int, alpha: float) -> np.ndarray:
    """Symmetric beta‑shape: a_i ∝ (i/(n‑1))^α·(1‑i/(n‑1))^α."""
    i = np.arange(n, dtype=np.float64)
    t = i / (n - 1)
    eps = 1e-12
    w = (t + eps) ** alpha * (1 - t + eps) ** alpha
    a = w / w.sum()
    return a

# Random initialisers

def random_sparse_start(n: int) -> np.ndarray:
    """Sparse probability vector (≈5 % non‑zeros)."""
    sparsity = np.random.uniform(0.03, 0.07)
    k = max(2, int(round(sparsity * n)))
    idx = np.random.choice(n, size=k, replace=False)
    a = np.zeros(n, dtype=np.float64)
    a[idx] = 1.0 / k
    return project_to_simplex(a)


def random_monotone_start(n: int) -> np.ndarray:
    """Generate a random decreasing vector (sorted power‑transformed uniform)."""
    r = np.random.rand(n) ** np.random.uniform(0.5, 5.0)
    r.sort()
    a = r[::-1]                # decreasing
    a = a / a.sum()
    return a


# ----------------------------------------------------------------------
# Smooth surrogate optimisation (L‑BFGS‑B)
# ----------------------------------------------------------------------


def _smooth_obj_and_grad(z: np.ndarray, beta: float, n: int):
    """
    Returns (f, grad) where
        f   = (1/β)·log∑_k exp[β·(a*a)_k]      (log‑sum‑exp surrogate)
        grad = ∂f/∂z   (z are the log‑soft‑max parameters)
    """
    a = softmax(z)                     # a ∈ simplex
    conv = convolution_fft(a, a)       # length 2n‑1
    maxc = conv.max()
    exp_terms = np.exp(beta * (conv - maxc))
    sum_exp = exp_terms.sum()
    f = (np.log(sum_exp) + beta * maxc) / beta

    # gradient wrt a
    w = exp_terms / sum_exp            # same shape as conv
    a_rev = a[::-1]
    # Convolution of reversed a with w gives the vector of partial derivatives
    conv_rev = convolution_fft(a_rev, w)   # length 3n‑2, but we need middle n entries
    grad_a = 2.0 * conv_rev[n - 1:2 * n - 1]

    # chain rule through soft‑max
    dot_ag = np.dot(a, grad_a)
    grad_z = a * (grad_a - dot_ag)
    return float(f), grad_z.astype(np.float64)


def l_bfgs_refine(start_a: np.ndarray, deadline: float,
                  beta_list=(200.0, 800.0, 2000.0),
                  max_iter_per_beta=600) -> tuple[np.ndarray, float]:
    """
    Refine `start_a` using L‑BFGS‑B on the smooth log‑sum‑exp surrogate.
    Returns the best vector and its exact score.
    """
    if not _HAS_SCIPY:          # pragma: no cover
        return start_a, compute_score(start_a)

    best_a = start_a.copy()
    best_score = compute_score(best_a)
    eps = 1e-12
    z = np.log(best_a + eps)          # initialise log‑soft‑max parameters

    for beta in beta_list:
        if time.time() >= deadline - 5.0:
            break

        remaining = deadline - time.time()
        max_iter = min(max_iter_per_beta, int(remaining * 8))  # roughly 8 iters / sec

        def func(z_vec):
            f, g = _smooth_obj_and_grad(z_vec, beta, best_a.shape[0])
            return f, g

        try:
            opt_res = _opt.fmin_l_bfgs_b(
                func,
                z,
                maxiter=max_iter,
                disp=False,
                factr=1e7,
                pgtol=1e-12,
            )
        except Exception as exc:  # pragma: no cover
            print(f"[{time.time():.1f}] L‑BFGS failed at β={beta}: {exc}")
            continue

        z = opt_res[0]
        a_opt = softmax(z)
        sc_opt = compute_score(a_opt)
        if sc_opt < best_score:
            best_score = sc_opt
            best_a = a_opt.copy()
            print(f"[{time.time():.1f}] L‑BFGS β={beta:.0f} improved score = {best_score:.6f}")

    return best_a, best_score


# ----------------------------------------------------------------------
# Mirror (entropy) sub‑gradient descent on the true max‑objective
# ----------------------------------------------------------------------


def mirror_subgrad_optimize(start_a: np.ndarray, deadline: float,
                           step0: float = 0.8,
                           check_interval: int = 2000) -> tuple[np.ndarray, float]:
    """
    Multiplicative (exponentiated) sub‑gradient descent on
        F(a) = 2·n·max(a*a)
    The iterate stays in the simplex thanks to a renormalisation step.
    """
    n = start_a.shape[0]
    a = start_a.astype(np.float64).copy()
    best_a = a.copy()
    best_score = compute_score(a)

    t = 0
    while time.time() < deadline:
        t += 1
        conv = convolution_fft(a, a)
        k = int(np.argmax(conv))

        # sub‑gradient of the active entry of the convolution
        idx = np.arange(n)
        j = k - idx
        mask = (j >= 0) & (j < n)
        g = np.zeros_like(a)
        g[mask] = 2.0 * a[j[mask]]

        eta = step0 / math.sqrt(t)
        a = a * np.exp(-eta * g)            # multiplicative update

        # re‑project onto the simplex (entropy projection)
        s = a.sum()
        if s == 0.0:
            a = np.full_like(a, 1.0 / n)
        else:
            a = a / s

        if t % check_interval == 0:
            sc = compute_score(a)
            if sc < best_score:
                best_score = sc
                best_a = a.copy()
                print(f"[{time.time():.1f}] mirror sub‑grad iter {t} best = {best_score:.6f}")

    sc = compute_score(a)
    if sc < best_score:
        best_score = sc
        best_a = a.copy()
    return best_a, best_score


# ----------------------------------------------------------------------
# Linear‑programming refinement (the “LP‑step” of the textbook method)
# ----------------------------------------------------------------------


def _build_conv_matrix(f: np.ndarray) -> np.ndarray:
    """
    Build the (2n‑1)×n matrix A such that A @ b = f * b (linear convolution).
    """
    n = f.shape[0]
    c = np.concatenate([f, np.zeros(n - 1, dtype=np.float64)])
    r = np.concatenate((np.array([f[0]], dtype=np.float64), np.zeros(n - 1, dtype=np.float64)))
    return _linalg.toeplitz(c, r)


def lp_refine(start_f: np.ndarray,
              t_mix: float = 0.5,
              max_iters: int = 5,
              deadline: float = 1e20) -> tuple[np.ndarray, float]:
    """
    Perform a few LP‑based refinements of `start_f`.

    For the current f compute M = max(f*f).  Solve
        maximise   ∑ b_j
        subject to  f * b ≤ M,   b ≥ 0.
    The optimal b is normalised to a probability vector g and mixed:
        f ← (1‑t_mix)·f + t_mix·g.
    The process is repeated a few times (or until the deadline is reached).
    Returns the best vector seen and its exact score.
    """
    if not _HAS_SCIPY:                      # pragma: no cover
        return start_f, compute_score(start_f)

    f = start_f.copy()
    best_f = f.copy()
    best_score = compute_score(f)

    for it in range(max_iters):
        if time.time() >= deadline - 2.0:
            break

        # 1) current bound M
        M = convolution_fft(f, f).max()

        # 2) build linear constraints A @ b ≤ M
        A = _build_conv_matrix(f)
        b_ub = np.full(A.shape[0], M, dtype=np.float64)

        # 3) solve LP: maximise 1ᵀb  <=> minimise -1ᵀb
        c_obj = -np.ones(f.shape[0], dtype=np.float64)
        bounds = [(0.0, None)] * f.shape[0]

        res = _opt.linprog(c=c_obj,
                           A_ub=A,
                           b_ub=b_ub,
                           bounds=bounds,
                           method='highs',
                           options={'presolve': True})
        if not res.success:
            print(f"[{time.time():.1f}] LP iteration {it+1} failed: {res.message}")
            break
        b = res.x
        if b.sum() == 0.0:
            break

        # 4) normalise and mix
        g = b / b.sum()
        f = (1.0 - t_mix) * f + t_mix * g
        f = project_to_simplex(f)          # safeguard against tiny numerical drift

        sc = compute_score(f)
        if sc < best_score:
            best_score = sc
            best_f = f.copy()
            print(f"[{time.time():.1f}] LP iteration {it+1} improved score = {best_score:.6f}")

        # slowly reduce the mixing factor to stabilise
        t_mix = max(t_mix * 0.85, 0.05)

    return best_f, best_score

def upsample_sequence(a_small: np.ndarray, n_big: int) -> np.ndarray:
    """
    Linear interpolation of a probability vector to a larger length.
    The result is re‑normalised to sum to one.
    """
    n_small = a_small.shape[0]
    x_small = np.linspace(0.0, 1.0, n_small)
    x_big = np.linspace(0.0, 1.0, n_big)
    a_big = np.interp(x_big, x_small, a_small)
    a_big = np.maximum(a_big, 0.0)
    a_big /= a_big.sum()
    return a_big


# ----------------------------------------------------------------------
# Tiny simulated‑annealing hill‑climb (optional final polish)
# ----------------------------------------------------------------------


def hill_climb_sa(start_a: np.ndarray, deadline: float,
                  max_steps: int = 1_200_000,
                  init_temp_factor: float = 1e-4) -> tuple[np.ndarray, float]:
    """
    Random walk on the simplex with a tiny temperature.
    """
    n = start_a.size
    cur_a = start_a.copy()
    cur_score = compute_score(cur_a)

    best_a = cur_a.copy()
    best_score = cur_score

    T0 = best_score * init_temp_factor
    T = T0
    step_scale = 0.12

    for step in range(1, max_steps + 1):
        if time.time() >= deadline:
            break

        cand = cur_a.copy()
        if np.random.rand() < 0.5:                     # swap two entries
            i, j = np.random.randint(n, size=2)
            cand[i], cand[j] = cand[j], cand[i]
        else:                                          # transfer a fraction
            i, j = np.random.randint(n, size=2)
            if i != j:
                delta = step_scale * cand[i]
                cand[i] -= delta
                cand[j] += delta
                if cand[i] < 0.0:
                    cand[i] = 0.0

        cand = cand / cand.sum()
        cand_score = compute_score(cand)
        delta = cand_score - cur_score

        if delta < 0.0 or np.random.rand() < np.exp(-delta / (T + 1e-18)):
            cur_a = cand
            cur_score = cand_score
            if cur_score < best_score:
                best_score = cur_score
                best_a = cur_a.copy()
                step_scale = min(step_scale * 1.07, 0.5)
        else:
            step_scale *= 0.97

        T = T0 * (1.0 - step / max_steps)

    return best_a, best_score

    # ------------------------------------------------------------------
    final_score = evaluate_sequence(best_seq)
    print(f"\n[{time.time():.1f}] finished search – final reported score = {final_score:.6f}")

    return best_seq
