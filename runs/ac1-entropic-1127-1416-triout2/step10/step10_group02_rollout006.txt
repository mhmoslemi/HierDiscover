def propose_move(a, k, step_scale):
    n = len(a)
    # contributions
    idx = np.arange(n)
    j = k - idx
    mask = (j >= 0) & (j < n)
    contrib = np.zeros_like(a)
    contrib[mask] = a[idx[mask]] * a[j[mask]]
    # If sum of contrib is zero -> uniform random i
    total = contrib.sum()
    if total > 0:
        # pick i weighted by contrib
        # Use np.random.rand() * total, then find first index where cumulative >= r
        r = np.random.rand() * total
        # cumulative sum on the fly
        cum = 0.0
        # Faster: use np.searchsorted after cumsum
        # Avoid building large arrays? okay.
        cum_contrib = np.cumsum(contrib)
        i = np.searchsorted(cum_contrib, r)
    else:
        i = np.random.randint(n)
    # j index for target. Choose uniformly random j != i
    j_target = np.random.randint(n-1)
    if j_target >= i:
        j_target += 1
    # delta
    delta = step_scale * a[i] * np.random.rand()
    cand = a.copy()
    cand[i] = max(cand[i] - delta, 0.0)
    cand[j_target] += delta
    # renormalize to sum 1:
    cand = project_to_simplex(cand)

    return cand

        t += 1

        conv = full_convolution(a)
        max_idx = int(np.argmax(conv))
        # propose candidate

        cand_val = compute_score(cand)
        # acceptance

            assess = cand_val
        # update best
        if cand_val < best_val:
            best_val, best = cand_val, cand.copy()
        # temperature decay
        T *= decay
        # step scale decay
        step_scale *= step_decay
        # occasionally reset step_scale if too small
        if step_scale < 1e-4:
            step_scale = 0.01

    return best

def anneal_optim(a0, deadline):
    a = a0.copy()
    n = a.shape[0]
    conv = full_convolution(a)
    max_idx = int(np.argmax(conv))
    score_cur = 2.0 * n * conv.max()
    best = a.copy()
    best_score = score_cur
    step_scale = 0.15
    temp = max(1e-6, 0.1 * score_cur)
    decay = 0.9997
    step_decay = 0.9999
    no_improve = 0
    while time.time() < deadline:
        # propose move based on max_idx
        cand = propose_move(a, max_idx, step_scale)
        # compute conv for cand
        conv_cand = full_convolution(cand)
        max_idx_cand = int(np.argmax(conv_cand))
        score_cand = 2.0 * n * conv_cand.max()
        # acceptance
        if score_cand < score_cur or np.random.rand() < np.exp(-(score_cand - score_cur) / max(temp, 1e-12)):
            a = cand
            conv = conv_cand
            max_idx = max_idx_cand
            score_cur = score_cand
        # update best
        if score_cand < best_score:
            best = cand.copy()
            best_score = score_cand
            no_improve = 0
        else:
            no_improve += 1
        # schedule
        temp *= decay
        step_scale *= step_decay
        # occasional resetting step_scale if too small
        if step_scale < 1e-4:
            step_scale = 0.01
        if no_improve > 5000:
            # Jump: random perturbation
            # add small jitter to a
            jitter = np.random.randn(n) * (step_scale * 0.1)
            cand = a + jitter
            cand = project_to_simplex(cand)
            # evaluate
            conv_cand = full_convolution(cand)
            max_idx_cand = int(np.argmax(conv_cand))
            score_cand = 2.0 * n * conv_cand.max()
            if score_cand < score_cur:
                a, conv, max_idx, score_cur = cand, conv_cand, max_idx_cand, score_cand
            no_improve = 0
    return best

"""
Non‑negative step‑function optimizer.

The objective to minimise is

    score = 2·n·max(conv(a,a)) / (sum(a)**2)

where `conv(a,a)` is the linear convolution of the coefficient vector
with itself.  Because the score is homogeneous, we may always keep
`sum(a)=1` and only search over the probability simplex.

Algorithm
---------

1. **Initial pool** – generate a handful of promising shapes:
   * a power‑law vector (a_i ∝ (i+1)^{-p}) with a fine sweep over p,
   * several Dirichlet random vectors with different concentration
     parameters (to explore both sparse and dense regimes),
   * a simple “mass at the ends” construction,
   * optionally the historic fallback sequence.
   Keep the best few (default 5) after normalising.

2. **Simulated‑annealing refinement** – for each seed vector run a
   temperature‑controlled stochastic search.  At every iteration we
   * compute the current auto‑convolution and locate the index k where
     it attains its maximum,
   * propose a move that transfers a random fraction of the mass from an
     index that contributes heavily to the active term (chosen with
     probability proportional to a_i·a_{k‑i}) to a uniformly chosen
     destination index,
   * accept the move if it improves the score, otherwise with the
     Metropolis probability `exp(-(Δscore)/T)`.
   The temperature T and the step‑size decay exponentially.
   The best vector seen during the run is returned.

3. **Optional sub‑gradient polish** – a few projected sub‑gradient steps
   (as in the previous baseline) are applied to the best candidate
   obtained from the annealing stage.

4. **Greedy hill‑climb** – a final cheap greedy local search with
   mass‑transfer moves is performed if time permits.

All phases respect the overall wall‑clock budget (≈1000 s).  The
returned list of floats is a valid coefficient sequence for
`evaluate_sequence`.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities
# ----------------------------------------------------------------------


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / v.shape[0])

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, v.shape[0] + 1)
    mask = u - cssv / ind > 0
    rho = np.where(mask)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def full_convolution(a: np.ndarray) -> np.ndarray:
    """
    Full linear convolution a*a using a real FFT.
    Length of the result is 2*len(a)-1.
    """
    n = a.shape[0]
    m = 2 * n - 1
    n_fft = 1 << ((m - 1).bit_length())   # next power of two ≥ m
    A = np.fft.rfft(a, n=n_fft)
    C = A * A
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


def compute_score(a: np.ndarray) -> float:
    """Exact objective for a normalised vector (sum = 1):
       2·n·max(conv(a,a))
    """
    n = a.shape[0]
    conv = full_convolution(a)
    return float(2.0 * n * conv.max())


# ----------------------------------------------------------------------
# 1. Power‑law sweep (global search)
# ----------------------------------------------------------------------


def best_power_law_vector(n: int, deadline: float) -> np.ndarray | None:
    """
    Scan p∈[0.5,0.8] (coarse then fine) and return the best
    normalised vector a_i ∝ (i+1)^{-p}.
    """
    best_a = None
    best_val = float("inf")
    best_p = 0.5

    # coarse grid
    coarse_ps = np.linspace(0.5, 0.8, 61)   # step ≈0.005
    for p in coarse_ps:
        if time.time() >= deadline:
            break
        a = np.arange(1, n + 1, dtype=float) ** (-p)
        a /= a.sum()
        val = compute_score(a)
        if val < best_val:
            best_val, best_a, best_p = val, a, p

    # fine grid around best_p
    if best_a is not None:
        low = max(0.5, best_p - 0.03)
        high = min(0.9, best_p + 0.03)
        fine_ps = np.linspace(low, high, 41)   # step ≈0.0015
        for p in fine_ps:
            if time.time() >= deadline:
                break
            a = np.arange(1, n + 1, dtype=float) ** (-p)
            a /= a.sum()
            val = compute_score(a)
            if val < best_val:
                best_val, best_a, best_p = val, a, p

    return best_a


# ----------------------------------------------------------------------
# 2. Dirichlet random vectors (exploratory)
# ----------------------------------------------------------------------


def random_dirichlet_vector(n: int, alpha: float) -> np.ndarray:
    """Draw a random point on the simplex with concentration alpha."""
    a = np.random.gamma(alpha, 1.0, size=n)
    if a.sum() == 0:
        a = np.ones(n)
    a /= a.sum()
    return a


# ----------------------------------------------------------------------
# 3. Targeted move for annealing
# ----------------------------------------------------------------------


def propose_move(a: np.ndarray, max_idx: int, step_scale: float) -> np.ndarray:
    """
    Produce a neighbour of `a` by moving a fraction of mass from an
    index that contributes strongly to the active convolution term
    (max_idx) to a uniformly chosen destination.
    """
    n = a.shape[0]
    idx = np.arange(n)
    j = max_idx - idx
    mask = (j >= 0) & (j < n)

    # contribution of each i to the active term
    contrib = np.zeros_like(a)
    contrib[mask] = a[idx[mask]] * a[j[mask]]
    total = contrib.sum()

    if total > 0:
        # weighted sampling without explicit loop
        r = np.random.rand() * total
        i = np.searchsorted(np.cumsum(contrib), r)
    else:
        i = np.random.randint(n)

    # pick a destination different from i
    j_dest = np.random.randint(n - 1)
    if j_dest >= i:
        j_dest += 1

    # amount to move (random fraction of the source mass)
    delta = step_scale * a[i] * np.random.rand()
    cand = a.copy()
    cand[i] = max(cand[i] - delta, 0.0)
    cand[j_dest] += delta

    # renormalise onto the simplex (tiny numerical drift)
    cand = project_to_simplex(cand)
    return cand


# ----------------------------------------------------------------------
# 4. Simulated‑annealing optimiser
# ----------------------------------------------------------------------


def anneal_optim(a0: np.ndarray, deadline: float) -> np.ndarray:
    """
    Stochastic annealing starting from `a0`.  Returns the best vector seen.
    """
    a = a0.copy()
    n = a.shape[0]

    # initialise state
    conv = full_convolution(a)
    max_idx = int(np.argmax(conv))
    cur_score = 2.0 * n * conv.max()

    best = a.copy()
    best_score = cur_score

    # hyper‑parameters (tuned empirically)
    step_scale = 0.15          # initial fraction of mass that may be moved
    temp = max(1e-6, 0.1 * cur_score)   # initial temperature
    decay = 0.9997             # temperature decay per iteration
    step_decay = 0.9999        # step‑size decay
    no_improve = 0

    while time.time() < deadline:
        # propose a neighbour using the active index max_idx
        cand = propose_move(a, max_idx, step_scale)

        # evaluate the candidate (single convolution)
        conv_cand = full_convolution(cand)
        max_idx_cand = int(np.argmax(conv_cand))
        cand_score = 2.0 * n * conv_cand.max()

        # Metropolis acceptance
        if cand_score < cur_score or np.random.rand() < np.exp(
            -(cand_score - cur_score) / max(temp, 1e-12)
        ):
            a = cand
            conv = conv_cand
            max_idx = max_idx_cand
            cur_score = cand_score

        # keep global best
        if cand_score < best_score:
            best = cand.copy()
            best_score = cand_score
            no_improve = 0
        else:
            no_improve += 1

        # schedule updates
        temp *= decay
        step_scale *= step_decay
        if step_scale < 1e-4:
            step_scale = 0.01

        # occasional random jitter if stuck
        if no_improve > 8000:
            jitter = np.random.randn(n) * (step_scale * 0.2)
            cand = a + jitter
            cand = project_to_simplex(cand)
            conv_cand = full_convolution(cand)
            cand_score = 2.0 * n * conv_cand.max()
            if cand_score < cur_score:
                a = cand
                conv = conv_cand
                max_idx = int(np.argmax(conv_cand))
                cur_score = cand_score
            no_improve = 0

    return best


# ----------------------------------------------------------------------
# 5. Sub‑gradient polish (as in the baseline)
# ----------------------------------------------------------------------


def subgradient_polish(a0: np.ndarray, deadline: float,
                       step0: float = 0.12) -> np.ndarray:
    """Projected sub‑gradient descent on the true max‑convolution objective."""
    a = a0.copy()
    n = a.shape[0]
    best = a.copy()
    best_val = compute_score(a)

    t = 0
    while time.time() < deadline:
        t += 1
        conv = full_convolution(a)
        k = int(np.argmax(conv))

        # sub‑gradient of the active entry
        idx = np.arange(n)
        j = k - idx
        mask = (j >= 0) & (j < n)
        g = np.zeros_like(a)
        g[mask] = 2.0 * a[j[mask]]

        step = step0 / np.sqrt(t)
        a = a - step * g
        a = project_to_simplex(a)

        val = compute_score(a)
        if val < best_val:
            best_val, best = val, a.copy()

        # occasional jitter to avoid stagnation
        if t % 1000 == 0:
            jitter = np.random.randn(n) * (step0 * 0.01 / np.sqrt(t))
            a += jitter
            a = project_to_simplex(a)

    return best


# ----------------------------------------------------------------------
# 6. Simple greedy hill‑climb (final polishing)
# ----------------------------------------------------------------------


def hill_climb(a_start: np.ndarray, deadline: float,
               max_no_improve: int = 600_000) -> np.ndarray:
    """
    Very cheap stochastic moves (mass transfer) with greedy acceptance.
    """
    a = a_start.copy()
    n = a.shape[0]
    best = a.copy()
    best_val = compute_score(a)

    step_scale = 0.05          # initial fraction of mass moved
    it = 0
    no_improve = 0

    while time.time() < deadline and it < max_no_improve:
        i, j = np.random.randint(n, size=2)
        if i == j:
            continue

        delta = step_scale * a[i]
        cand = a.copy()
        cand[i] -= delta
        cand[j] += delta
        cand = np.maximum(cand, 0.0)
        cand = project_to_simplex(cand)

        val = compute_score(cand)
        if val < best_val:
            best_val, best, a = val, cand.copy(), cand
            step_scale = min(step_scale * 1.07, 0.5)
            no_improve = 0
        else:
            step_scale *= 0.985
            no_improve += 1

        it += 1

    return best
