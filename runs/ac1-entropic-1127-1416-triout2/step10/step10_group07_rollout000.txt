"""
search_v2.py – Multi‑scale Adam + mirror descent optimizer

Goal
----
Find a non‑negative step‑function (a list of floats) that minimizes
``evaluate_sequence``.  The objective for a vector `a` with sum = 1 is
`2·n·max(conv(a,a))`.  We therefore try to minimise the maximum of the
auto‑convolution.

Main ideas
----------
* Use a **block representation**: the vector of length `n` is divided
  into `B` blocks (default 256).  All entries inside a block share the
  same value, so the optimisation lives in a low‑dimensional simplex.
* Optimise the block weights with **Adam** on the smooth log‑sum‑exp
  surrogate of the max‑convolution.  The smoothing parameter β is
  increased exponentially from a modest value to a very large one,
  turning the surrogate into the true max.
* After the block phase we **refine** the full‑resolution vector with
  entropy‑mirror sub‑gradient descent (the same “mirror” method used in
  the previous state‑of‑the‑art code).  This takes a few thousand steps
  and often yields a few‑percent improvement.
* The whole pipeline respects the supplied time budget (`budget_s`),
  falling back to a uniform vector if nothing is found.

The code is self‑contained and deliberately avoids heavy external
dependencies – only ``numpy`` and the standard library are used.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Basic utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full aperiodic convolution using real FFT (fast, no Python loops)."""
    m = a.size + b.size - 1
    nfft = next_pow2(m)
    A = np.fft.rfft(a, n=nfft)
    B = np.fft.rfft(b, n=nfft)
    C = A * B
    conv = np.fft.irfft(C, n=nfft)[:m]
    return conv


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex (sum=1, v≥0)."""
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / v.shape[0])

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, v.shape[0] + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.where(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def softmax(z: np.ndarray) -> np.ndarray:
    """Numerically stable soft‑max."""
    z_max = np.max(z)
    e = np.exp(z - z_max)
    return e / e.sum()


def compute_score(a: np.ndarray) -> float:
    """
    Exact objective for a probability vector (sum = 1):
        2·n·max(a * a)
    """
    n = a.shape[0]
    conv = convolution_fft(a, a)
    return float(2.0 * n * conv.max())


# ----------------------------------------------------------------------
# Block representation helpers
# ----------------------------------------------------------------------


def make_block_structure(n: int, B: int = 256):
    """
    Divide length n into B blocks (as equal as possible).  Returns a list
    `sizes` where `sizes[i]` is the size of block i, and an array
    `indices` where `indices[i]` is the start index of block i.
    """
    if B > n:
        B = n
    base = n // B
    rem = n % B
    sizes = [base + 1 if i < rem else base for i in range(B)]
    indices = np.cumsum([0] + sizes[:-1])
    return np.array(sizes, dtype=int), np.array(indices, dtype=int)


def expand_blocks(b: np.ndarray, sizes: np.ndarray, indices: np.ndarray) -> np.ndarray:
    """
    Convert a block vector `b` (length B) into a full‑resolution vector `a`
    of length sum(sizes).  Within each block the mass is spread uniformly.
    """
    n = int(sizes.sum())
    a = np.empty(n, dtype=np.float64)
    for j, (sz, start) in enumerate(zip(sizes, indices)):
        a[start:start + sz] = b[j] / sz
    return a


def grad_block_from_a(grad_a: np.ndarray, sizes: np.ndarray, indices: np.ndarray) -> np.ndarray:
    """
    Aggregate a gradient defined on the full vector `a` back to block
    gradient: each block receives the average of the entries that belong
    to it.
    """
    B = sizes.shape[0]
    g = np.empty(B, dtype=np.float64)
    for j in range(B):
        start = indices[j]
        sz = sizes[j]
        g[j] = grad_a[start:start + sz].sum() / sz
    return g


# ----------------------------------------------------------------------
# Adam on the smoothed max‑convolution (block version)
# ----------------------------------------------------------------------


def block_adam_optimize(
    start_b: np.ndarray,
    sizes: np.ndarray,
    indices: np.ndarray,
    deadline: float,
    max_iters: int = 120_000,
    beta_start: float = 5.0,
    beta_end: float = 20_000.0,
    lr: float = 0.12,
    tag: str = "",
) -> tuple[np.ndarray, float]:
    """
    Adam optimisation in the low‑dimensional block space.
    Returns the best block vector (still on the simplex) and its
    corresponding full‑resolution score.
    """
    B = start_b.shape[0]
    b = start_b.copy()
    m = np.zeros_like(b)
    v = np.zeros_like(b)

    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-12

    # keep best seen
    a_best = expand_blocks(b, sizes, indices)
    best_score = compute_score(a_best)

    for it in range(1, max_iters + 1):
        if time.time() >= deadline:
            break

        # ----- schedule for β (exponential) -----
        t = it / max_iters
        beta = beta_start * (beta_end / beta_start) ** t

        # ----- build full vector -----
        a = expand_blocks(b, sizes, indices)

        # ----- smoothed surrogate -----
        conv = convolution_fft(a, a)                # length 2n‑1
        maxc = conv.max()
        w = softmax(beta * (conv - maxc))           # weights sum to 1

        # ----- gradient w.r.t. a (as in the original code) -----
        a_rev = a[::-1]
        conv_rev = convolution_fft(a_rev, w)       # length 3n‑2
        grad_a = 2.0 * conv_rev[len(a) - 1 : len(a) - 1 + len(a)]

        # ----- aggregate to block gradient -----
        grad_b = grad_block_from_a(grad_a, sizes, indices)

        # enforce symmetry of the block vector (most good solutions are symmetric)
        grad_b = (grad_b + grad_b[::-1]) * 0.5

        # ----- Adam update -----
        m = beta1 * m + (1.0 - beta1) * grad_b
        v = beta2 * v + (1.0 - beta2) * (grad_b ** 2)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        b -= lr * m_hat / (np.sqrt(v_hat) + eps)

        # ----- keep on simplex and enforce symmetry -----
        b = project_to_simplex(b)
        b = (b + b[::-1]) * 0.5

        # ----- occasional exact evaluation -----
        if it % 2000 == 0:
            a_tmp = expand_blocks(b, sizes, indices)
            sc = compute_score(a_tmp)
            if sc < best_score:
                best_score = sc
                a_best = a_tmp.copy()
                print(f"[{time.time():.1f}]   {tag} Adam iter {it:7d} – new best = {best_score:.6f}")

    # final check
    a_final = expand_blocks(b, sizes, indices)
    sc_final = compute_score(a_final)
    if sc_final < best_score:
        best_score = sc_final
        a_best = a_final.copy()

    # return the block vector that produced the best full‑resolution a
    # (reconstruct block vector from a_best for consistency)
    # Here we simply project a_best back to block form by averaging per block.
    b_best = np.empty_like(b)
    for j in range(B):
        start = indices[j]
        sz = sizes[j]
        b_best[j] = a_best[start:start + sz].sum()
    b_best = project_to_simplex(b_best)
    return b_best, best_score


# ----------------------------------------------------------------------
# Mirror sub‑gradient descent on the true max‑convolution (full resolution)
# ----------------------------------------------------------------------


def mirror_subgrad_optimize(
    start_a: np.ndarray,
    deadline: float,
    step0: float = 0.9,
    check_interval: int = 1500,
) -> tuple[np.ndarray, float]:
    """
    Entropy‑mirror (multiplicative) sub‑gradient descent for the exact
    objective 2·n·max(a*a).  Step size η_t = step0 / sqrt(t).
    """
    n = start_a.shape[0]
    a = start_a.astype(np.float64).copy()
    best_a = a.copy()
    best_score = compute_score(a)

    t = 0
    while time.time() < deadline:
        t += 1
        conv = convolution_fft(a, a)
        k = int(np.argmax(conv))

        # gradient for the active shift
        idx = np.arange(n)
        j = k - idx
        mask = (j >= 0) & (j < n)
        g = np.zeros_like(a)
        g[mask] = 2.0 * a[j[mask]]

        eta = step0 / np.sqrt(t)
        a = a * np.exp(-eta * g)          # multiplicative update
        s = a.sum()
        if s == 0.0:
            a = np.full_like(a, 1.0 / n)
        else:
            a = a / s

        if t % check_interval == 0:
            sc = compute_score(a)
            if sc < best_score:
                best_score = sc
                best_a = a.copy()
                print(f"[{time.time():.1f}]   Mirror step {t:7d} – new best = {best_score:.6f}")

    # final check
    sc = compute_score(a)
    if sc < best_score:
        best_score = sc
        best_a = a.copy()

    return best_a, best_score

def propose_candidate(seed: int = 12345, budget_s: int = 1000, **_):
    """
    Search for a non‑negative step‑function (list of floats) that
    minimises ``evaluate_sequence``.  The routine respects a wall‑clock
    budget (≈ budget_s seconds) and returns a plain Python list.
    """
    np.random.seed(seed)
    t0 = time.time()
    deadline = t0 + budget_s - 2.0          # small safety margin

    # -------------------------------------------------------------
    # Choose a length – larger n gives more room for improvement.
    # -------------------------------------------------------------
    if budget_s >= 900:
        n = 16384
    elif budget_s >= 600:
        n = 8192
    else:
        n = 4096
    print(f"[{time.time():.1f}] Using length n = {n}")

    # -------------------------------------------------------------
    # Block structure
    # -------------------------------------------------------------
    B = 256
    sizes, indices = make_block_structure(n, B)
    print(f"[{time.time():.1f}] Block count = {B}, typical block size ≈ {int(np.mean(sizes))}")

    best_seq = None
    best_score = float("inf")

    def maybe_update(vec: np.ndarray, tag: str = ""):
        """Update the global best if the new vector improves the score."""
        nonlocal best_seq, best_score
        sc = compute_score(vec)
        if sc < best_score:
            best_score = sc
            best_seq = vec.tolist()
            print(f"[{time.time():.1f}] new best {tag:<20} = {best_score:.6f}")

    # -------------------------------------------------------------
    # 0. Simple baseline – a few analytic families (coarse scan)
    # -------------------------------------------------------------
    uniform = np.full(n, 1.0 / n)
    maybe_update(uniform, "uniform")

    # symmetric product family (coarse sweep)
    for alpha in np.linspace(0.0, 6.0, 31):
        if time.time() >= deadline - 30:
            break
        a = symmetric_product_family(n, alpha)
        maybe_update(a, f"sym‑α={alpha:.2f}")

    # left‑heavy power law
    for beta in np.linspace(0.0, 6.0, 31):
        if time.time() >= deadline - 30:
            break
        a = left_heavy_power_family(n, beta)
        maybe_update(a, f"pow‑β={beta:.2f}")

    # exponential decay
    for gamma in np.linspace(0.0, 8.0, 41):
        if time.time() >= deadline - 30:
            break
        a = exponential_decay_family(n, gamma)
        maybe_update(a, f"exp‑γ={gamma:.2f}")

    # -------------------------------------------------------------
    # 1. Block‑level Adam optimisation (many random restarts)
    # -------------------------------------------------------------
    # allocate roughly half of the remaining time to this phase
    block_phase_deadline = deadline - 250.0
    restart = 0
    while time.time() < block_phase_deadline:
        restart += 1
        # random Dirichlet start for blocks (concentration varies)
        conc = 10 ** np.random.uniform(-2, 0.5)   # between 0.01 and ~3.2
        b0 = np.random.gamma(conc, 1.0, size=B)
        b0 = project_to_simplex(b0)

        # small symmetry enforcement
        b0 = (b0 + b0[::-1]) * 0.5

        b_opt, sc_opt = block_adam_optimize(
            start_b=b0,
            sizes=sizes,
            indices=indices,
            deadline=block_phase_deadline,
            max_iters=180_000,
            beta_start=5.0,
            beta_end=2000.0,
            lr=0.15,
            tag=f"restart{restart}",
        )
        if sc_opt < best_score:
            best_score = sc_opt
            best_seq = expand_blocks(b_opt, sizes, indices).tolist()
            print(f"[{time.time():.1f}] new best after block‑Adam (restart {restart}) = {best_score:.6f}")

    # -------------------------------------------------------------
    # 2. Fine‑grained Adam with huge β on the best full‑resolution vector
    # -------------------------------------------------------------
    if best_seq is not None and time.time() < deadline - 130.0:
        print(f"[{time.time():.1f}] high‑β Adam refinement")
        a_ref, sc_ref = adam_smooth_optimize(
            start_a=np.array(best_seq, dtype=np.float64),
            deadline=deadline - 100.0,
            max_iters=120_000,
            beta_start=2000.0,
            beta_end=20000.0,
            lr=0.010,
        )
        if sc_ref < best_score:
            best_score = sc_ref
            best_seq = a_ref.tolist()
            print(f"[{time.time():.1f}] after high‑β Adam = {best_score:.6f}")

    # -------------------------------------------------------------
    # 3. Mirror sub‑gradient descent on the exact objective
    # -------------------------------------------------------------
    if best_seq is not None and time.time() < deadline - 70.0:

        a_sub, sc_sub = mirror_subgrad_optimize(

        if sc_sub < best_score:

            best_seq = a_sub.tolist()
            print(f"[{time.time():.1f}] after mirror descent = {best_score:.6f}")

    # -------------------------------------------------------------
    # 4. Symmetrisation (many solutions turn out symmetric)
    # -------------------------------------------------------------
    if best_seq is not None:
        a_arr = np.array(best_seq, dtype=np.float64)
        a_sym = (a_arr + a_arr[::-1]) * 0.5
        a_sym = project_to_simplex(a_sym)

def symmetric_product_family(n: int, alpha: float) -> np.ndarray:
    """a_i ∝ [(i+½)(n‑½‑i)]^(‑α) , symmetric, normalised to sum = 1."""
    i = np.arange(n, dtype=np.float64)
    logprod = np.log(i + 0.5) + np.log(n - 0.5 - i)
    w = np.exp(-alpha * logprod)
    w[np.isinf(w)] = np.finfo(np.float64).max
    a = w / w.sum()
    return a


def left_heavy_power_family(n: int, beta: float) -> np.ndarray:
    """a_i ∝ (i+½)^(‑β) , monotone decreasing, sum = 1."""
    i = np.arange(n, dtype=np.float64)
    w = np.exp(-beta * np.log(i + 0.5))
    a = w / w.sum()
    return a


def exponential_decay_family(n: int, gamma: float) -> np.ndarray:
    """a_i ∝ exp(-γ·i) , monotone decreasing, sum = 1."""
    i = np.arange(n, dtype=np.float64)
    w = np.exp(-gamma * i)
    a = w / w.sum()
    return a


# ----------------------------------------------------------------------
# Smoothed optimisation (log‑sum‑exp) via Adam on the soft‑max layer
# (same implementation as before, kept for the high‑β refinement stage)
# ----------------------------------------------------------------------


def adam_smooth_optimize(
    start_a: np.ndarray,
    deadline: float,
    max_iters: int = 200_000,
    beta_start: float = 5.0,
    beta_end: float = 2000.0,
    lr: float = 0.018,
) -> tuple[np.ndarray, float]:
    """
    Minimise the smoothed surrogate
          f_β(a) = (1/β)·log Σ_k exp(β·(a*a)_k)
    using Adam on the soft‑max parametrisation a = softmax(z).
    β is increased exponentially from beta_start to beta_end.
    """
    eps = 1e-12
    z = np.log(start_a + eps)               # initialise from log‑softmax
    m = np.zeros_like(z)
    v = np.zeros_like(z)

    beta1 = 0.9
    beta2 = 0.999
    eps_adam = 1e-12

    best_a = softmax(z)
    best_score = compute_score(best_a)

    for it in range(1, max_iters + 1):
        if time.time() >= deadline:
            break

        t = it / max_iters
        beta = beta_start * (beta_end / beta_start) ** t

        a = softmax(z)

        conv = convolution_fft(a, a)
        maxc = conv.max()
        exp_w = np.exp(beta * (conv - maxc))
        w = exp_w / exp_w.sum()

        a_rev = a[::-1]
        conv_rev = convolution_fft(a_rev, w)
        grad_a = 2.0 * conv_rev[len(a) - 1 : len(a) - 1 + len(a)]

        dot_ag = np.dot(a, grad_a)
        grad_z = a * grad_a - dot_ag * a

        m = beta1 * m + (1.0 - beta1) * grad_z
        v = beta2 * v + (1.0 - beta2) * (grad_z ** 2)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        z -= lr * m_hat / (np.sqrt(v_hat) + eps_adam)

        if it % 400 == 0:
            a_tmp = softmax(z)
            sc = compute_score(a_tmp)
            if sc < best_score:
                best_score = sc
                best_a = a_tmp.copy()
                print(f"[{time.time():.1f}]   Adam iter {it:7d} – new best = {best_score:.6f}")

    a_fin = softmax(z)
    sc_fin = compute_score(a_fin)
    if sc_fin < best_score:
        best_score = sc_fin
        best_a = a_fin.copy()
    return best_a, best_score


# ----------------------------------------------------------------------
# Stochastic hill‑climb (tiny simulated‑annealing) – unchanged from earlier
# ----------------------------------------------------------------------


def hill_climb_sa(
    start_a: np.ndarray,
    deadline: float,
    max_steps: int = 1_500_000,
    init_temp_factor: float = 1e-4,
) -> tuple[np.ndarray, float]:
    """
    Very low‑temperature simulated‑annealing walk on the simplex.
    - with prob 0.5 swap two entries
    - otherwise transfer a fraction of mass from one index to another
    """
    n = start_a.size
    cur_a = start_a.copy()
    cur_score = compute_score(cur_a)

    best_a = cur_a.copy()
    best_score = cur_score

    T0 = best_score * init_temp_factor
    T = T0
    step_scale = 0.12

    for step in range(1, max_steps + 1):
        if time.time() >= deadline:
            break

        cand = cur_a.copy()
        if np.random.rand() < 0.5:                       # swap
            i, j = np.random.randint(n, size=2)
            cand[i], cand[j] = cand[j], cand[i]
        else:                                            # mass transfer
            i, j = np.random.randint(n, size=2)
            if i != j:
                delta = step_scale * cur_a[i]
                cand[i] -= delta
                cand[j] += delta
                if cand[i] < 0.0:
                    cand[i] = 0.0

        cand = cand / cand.sum()                         # simplex projection
        cand_score = compute_score(cand)
        delta = cand_score - cur_score

        # Metropolis acceptance
        if delta < 0.0 or np.random.rand() < np.exp(-delta / (T + 1e-18)):
            cur_a = cand
            cur_score = cand_score
            if cur_score < best_score:
                best_score = cur_score
                best_a = cur_a.copy()
                step_scale = min(step_scale * 1.07, 0.5)
        else:
            step_scale *= 0.97

        # linear cooling
        T = T0 * (1.0 - step / max_steps)

    return best_a, best_score
