propose_candidate
-----------------
Search for a non‑negative step‑function (a probability vector) that minimizes
the objective used by ``evaluate_sequence``.  The search proceeds in several
stages, each respecting a global wall‑clock deadline (default 1000 s).

* A moderately large length (≈16–32 k) is chosen – this gives enough degrees
  of freedom while keeping FFTs cheap.
* Several analytic families are examined.  In addition to the families used
  previously, a *product‑offset* family a_i ∝ ( (i+½)(n‑½‑i) / p_max + δ )^{-α}

def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Linear (full) convolution using real FFT."""
    m = a.size + b.size - 1
    nfft = next_pow2(m)
    A = np.fft.rfft(a, n=nfft)
    B = np.fft.rfft(b, n=nfft)
    C = A * B
    conv = np.fft.irfft(C, n=nfft)[:m]
    return conv


def compute_score(a: np.ndarray) -> float:
    """
    Exact objective for a probability vector (sum(a)=1):
        score = 2·n·max(a*a)
    This matches ``evaluate_sequence`` up to the normalisation.
    """
    n = a.shape[0]
    conv = convolution_fft(a, a)
    return 2.0 * n * float(conv.max())

def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection onto the probability simplex."""
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / v.shape[0])

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, v.shape[0] + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.where(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def softmax(z: np.ndarray) -> np.ndarray:
    """Numerically stable soft‑max (maps ℝⁿ → simplex)."""
    z_max = np.max(z)
    e = np.exp(z - z_max)
    return e / e.sum()

def symmetric_product_family(n: int, alpha: float) -> np.ndarray:
    """U‑shaped: a_i ∝ [(i+½)(n‑½‑i)]^{‑α}."""
    i = np.arange(n, dtype=np.float64)
    logprod = np.log(i + 0.5) + np.log(n - 0.5 - i)
    w = np.exp(-alpha * logprod)
    a = w / w.sum()
    return a


def product_offset_family(n: int, alpha: float, delta: float) -> np.ndarray:
    """
    Same as ``symmetric_product_family`` but with an additive offset δ
    inside the power to smooth the edges:
        a_i ∝ ( ((i+½)(n‑½‑i))/p_max + δ )^{‑α}
    where p_max = ((n‑½)/2)² normalises the product into [0,1].
    """
    i = np.arange(n, dtype=np.float64)
    p = (i + 0.5) * (n - 0.5 - i)
    p_max = ((n - 0.5) / 2.0) ** 2
    norm = p / p_max  # lies in (≈2/n , 1]
    eps = 1e-15
    w = (norm + delta + eps) ** (-alpha)
    a = w / w.sum()
    return a


def left_heavy_power_family(n: int, beta: float) -> np.ndarray:
    """Monotone decreasing power law: a_i ∝ (i+½)^{‑β}."""
    i = np.arange(n, dtype=np.float64)
    w = (i + 0.5) ** (-beta)
    a = w / w.sum()
    return a


def exponential_decay_family(n: int, gamma: float) -> np.ndarray:
    """Simple exponential decay: a_i ∝ exp(‑γ·i)."""
    i = np.arange(n, dtype=np.float64)
    w = np.exp(-gamma * i)
    a = w / w.sum()
    return a


def beta_distribution_family(n: int, alpha: float) -> np.ndarray:
    """Symmetric beta–shape."""
    i = np.arange(n, dtype=np.float64)
    t = i / (n - 1)
    eps = 1e-12
    w = (t + eps) ** alpha * (1 - t + eps) ** alpha
    a = w / w.sum()
    return a


def random_dirichlet_start(n: int) -> np.ndarray:
    """Draw a random point from the simplex using a Dirichlet(1,…,1)."""
    a = np.random.gamma(1.0, 1.0, size=n)
    return a / a.sum()

def _smooth_obj_and_grad(z: np.ndarray, beta: float, n: int):
    """
    Returns (f, grad) for the surrogate:
        f   = (1/β)·log Σ_k exp[β·(a*a)_k]
        grad = ∂f/∂z   (z are the log‑softmax parameters)
    """
    a = softmax(z)
    conv = convolution_fft(a, a)
    maxc = conv.max()

    # stable log‑sum‑exp
    exp_terms = np.exp(beta * (conv - maxc))
    sum_exp = exp_terms.sum()
    f = (np.log(sum_exp) + beta * maxc) / beta

    # gradient w.r.t. a
    w = exp_terms / sum_exp                     # shape (2n‑1,)
    a_rev = a[::-1]
    conv_rev = convolution_fft(a_rev, w)       # length 3n‑2
    grad_a = 2.0 * conv_rev[n - 1 : 2 * n - 1]

    # chain rule through soft‑max
    dot_ag = np.dot(a, grad_a)
    grad_z = a * (grad_a - dot_ag)
    return float(f), grad_z.astype(np.float64)


def l_bfgs_refine(start_a: np.ndarray, deadline: float,
                  beta_list=(200.0, 800.0, 2000.0),
                  max_iter_per_beta=400) -> tuple[np.ndarray, float]:
    """
    Refine ``start_a`` with L‑BFGS‑B on the smooth surrogate.
    The optimisation is performed sequentially for each β in ``beta_list``.
    Returns the best vector and its exact (non‑surrogate) score.
    """
    if not _HAS_SCIPY:          # pragma: no cover
        return start_a, compute_score(start_a)

    best_a = start_a.copy()
    best_score = compute_score(best_a)
    eps = 1e-12
    z = np.log(best_a + eps)                     # initialise log‑softmax params

    for beta in beta_list:
        if time.time() >= deadline - 5.0:
            break

        remaining = deadline - time.time()
        max_iter = min(max_iter_per_beta, int(remaining * 12))  # ~12 iters / sec

        def func(z_vec):
            f, g = _smooth_obj_and_grad(z_vec, beta, best_a.shape[0])
            return f, g

        try:
            opt_res = _opt.fmin_l_bfgs_b(
                func,
                z,
                maxiter=max_iter,
                disp=False,
                factr=1e7,
                pgtol=1e-12,
            )
        except Exception as exc:  # pragma: no cover
            print(f"[{time.time():.1f}] L‑BFGS failed at β={beta}: {exc}")
            continue

        z = opt_res[0]
        a_opt = softmax(z)
        sc_opt = compute_score(a_opt)

        if sc_opt < best_score:
            best_score = sc_opt
            best_a = a_opt.copy()
            print(f"[{time.time():.1f}] L‑BFGS β={beta:.0f} improved score = {best_score:.6f}")

    return best_a, best_score

def mirror_subgrad_optimize(start_a: np.ndarray, deadline: float,
                           step0: float = 0.8,
                           check_interval: int = 3000) -> tuple[np.ndarray, float]:
    """
    Multiplicative sub‑gradient descent on
        F(a) = 2·n·max(a*a)
    The iterate stays on the simplex thanks to a normalisation step.
    """
    n = start_a.shape[0]
    a = start_a.astype(np.float64).copy()
    best_a = a.copy()
    best_score = compute_score(a)

    t = 0
    while time.time() < deadline:
        t += 1
        conv = convolution_fft(a, a)
        k = int(np.argmax(conv))

        # sub‑gradient for the active shift k
        idx = np.arange(n)
        j = k - idx
        mask = (j >= 0) & (j < n)
        g = np.zeros_like(a)
        g[mask] = 2.0 * a[j[mask]]

        eta = step0 / math.sqrt(t)
        a = a * np.exp(-eta * g)            # multiplicative update
        a_sum = a.sum()
        if a_sum == 0.0:
            a = np.full_like(a, 1.0 / n)
        else:
            a = a / a_sum

        if t % check_interval == 0:
            sc = compute_score(a)
            if sc < best_score:
                best_score = sc
                best_a = a.copy()
                print(f"[{time.time():.1f}] mirror sub‑grad iter {t} best = {best_score:.6f}")

    # final check
    sc = compute_score(a)
    if sc < best_score:
        best_score = sc
        best_a = a.copy()
    return best_a, best_score


# ----------------------------------------------------------------------
# Tiny simulated‑annealing hill‑climb
# ----------------------------------------------------------------------


def hill_climb_sa(start_a: np.ndarray, deadline: float,
                  max_steps: int = 800_000,
                  init_temp_factor: float = 1e-4) -> tuple[np.ndarray, float]:
    """
    Random walk on the simplex with a tiny temperature.
    Moves are either a swap of two entries or a transfer of a fraction of mass.
    """
    n = start_a.size
    cur_a = start_a.copy()
    cur_score = compute_score(cur_a)

    best_a = cur_a.copy()
    best_score = cur_score

    T0 = best_score * init_temp_factor
    T = T0
    step_scale = 0.12

    for step in range(1, max_steps + 1):
        if time.time() >= deadline:
            break

        cand = cur_a.copy()
        if np.random.rand() < 0.5:                     # swap
            i, j = np.random.randint(n, size=2)
            cand[i], cand[j] = cand[j], cand[i]
        else:                                          # mass transfer
            i, j = np.random.randint(n, size=2)
            if i != j:
                delta = step_scale * cur_a[i]
                cand[i] -= delta
                cand[j] += delta
                if cand[i] < 0.0:
                    cand[i] = 0.0

        cand = cand / cand.sum()
        cand_score = compute_score(cand)
        delta = cand_score - cur_score

        if delta < 0.0 or np.random.rand() < np.exp(-delta / (T + 1e-18)):
            cur_a = cand
            cur_score = cand_score
            if cur_score < best_score:
                best_score = cur_score
                best_a = cur_a.copy()
                step_scale = min(step_scale * 1.07, 0.5)
        else:
            step_scale *= 0.97

        # linear cooling
        T = T0 * (1.0 - step / max_steps)

    return best_a, best_score

# High‑level driver
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 12345, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative sequence minimising ``evaluate_sequence``.
    Returns a plain Python list of floats.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 2.0   # safety margin

    # --------------------------------------------------------------
    # Choose length (larger n gives more flexibility)
    # --------------------------------------------------------------
    if budget_s >= 900:
        n = 32768          # approx 30 k – a good trade‑off
    elif budget_s >= 600:
        n = 16384
    else:
        n = 8192
    print(f"[{time.time():.1f}] Using length n = {n}")

    best_seq = None
    best_score = float("inf")

    # --------------------------------------------------------------
    # 0.  Optional fallback to a known construction
    # --------------------------------------------------------------
    try:
        fb_vec = [float(x) for x in height_sequence_1]   # type: ignore
        fb = evaluate_sequence(fb_vec)
        print(f"[{time.time():.1f}] fallback score = {fb:.6f}")
        if fb < best_score:
            best_score = fb
            best_seq = fb_vec
    except Exception:
        pass  # no fallback available

    # --------------------------------------------------------------
    # 1. Analytic families (quick grid scans)
    # --------------------------------------------------------------

    # Uniform baseline (useful for sanity)
    uniform = np.full(n, 1.0 / n, dtype=np.float64)
    sc_uni = compute_score(uniform)
    if sc_uni < best_score:
        best_score = sc_uni
        best_seq = uniform.tolist()
        print(f"[{time.time():.1f}] uniform baseline = {best_score:.6f}")

    # ------------------- product family ----------------------------
    alphas = np.linspace(0.2, 4.0, 81)               # coarse sweep
    for alpha in alphas:
        if time.time() >= deadline - 30.0:
            break
        vec = symmetric_product_family(n, alpha)
        sc = compute_score(vec)
        if sc < best_score:
            best_score = sc
            best_seq = vec.tolist()
            print(f"[{time.time():.1f}] new best (prod‑α={alpha:.3f}) = {best_score:.6f}")

    # ------------------- product‑offset family ----------------------
    # Two‑parameter search (α, δ).  δ is tiny – it smooths the edge blow‑up.
    deltas = np.linspace(0.0, 0.08, 9)                # 0, 0.01, …, 0.08
    for delta in deltas:
        for alpha in np.linspace(0.4, 3.2, 61):
            if time.time() >= deadline - 30.0:
                break
            vec = product_offset_family(n, alpha, delta)
            sc = compute_score(vec)
            if sc < best_score:
                best_score = sc
                best_seq = vec.tolist()
                print(f"[{time.time():.1f}] new best (p‑off α={alpha:.3f}, δ={delta:.3f}) = {best_score:.6f}")

    # ------------------- left‑heavy power family -------------------
    for beta in np.linspace(0.2, 5.0, 81):
        if time.time() >= deadline - 30.0:
            break
        vec = left_heavy_power_family(n, beta)
        sc = compute_score(vec)
        if sc < best_score:
            best_score = sc
            best_seq = vec.tolist()
            print(f"[{time.time():.1f}] new best (pow‑β={beta:.3f}) = {best_score:.6f}")

    # ------------------- exponential decay family -------------------
    for gamma in np.linspace(0.0, 5.0, 61):
        if time.time() >= deadline - 30.0:
            break
        vec = exponential_decay_family(n, gamma)
        sc = compute_score(vec)
        if sc < best_score:
            best_score = sc
            best_seq = vec.tolist()
            print(f"[{time.time():.1f}] new best (exp‑γ={gamma:.3f}) = {best_score:.6f}")

    # ------------------- beta‑distribution family -------------------
    for alpha in np.linspace(0.0, 6.0, 61):
        if time.time() >= deadline - 30.0:
            break
        vec = beta_distribution_family(n, alpha)
        sc = compute_score(vec)
        if sc < best_score:
            best_score = sc
            best_seq = vec.tolist()
            print(f"[{time.time():.1f}] new best (beta‑α={alpha:.3f}) = {best_score:.6f}")

    # ------------------- random Dirichlet starts --------------------
    for _ in range(30):
        if time.time() >= deadline - 30.0:
            break
        vec = random_dirichlet_start(n)
        sc = compute_score(vec)
        if sc < best_score:
            best_score = sc
            best_seq = vec.tolist()
            print(f"[{time.time():.1f}] new best (rand‑dir) = {best_score:.6f}")

    # --------------------------------------------------------------
    # 2. Refine a few top candidates with L‑BFGS on the smooth surrogate.
    #    We take the current best and a handful of the next‑best
    #    candidates (if they exist) to diversify the optimisation.
    # --------------------------------------------------------------
    if best_seq is not None and time.time() < deadline - 300.0:
        # collect a few promising starts
        starts = [np.array(best_seq, dtype=np.float64)]

        # add a few random Dirichlet starts (already evaluated) for diversity
        for _ in range(4):
            vec = random_dirichlet_start(n)
            starts.append(vec)

        best_local_seq = None
        best_local_score = float("inf")
        for idx, a_start in enumerate(starts):
            if time.time() >= deadline - 200.0:
                break
            print(f"[{time.time():.1f}] L‑BFGS refinement #{idx+1}")
            a_ref, sc_ref = l_bfgs_refine(
                a_start,
                deadline - 150.0,
                beta_list=(200.0, 800.0, 2000.0),
                max_iter_per_beta=300,
            )
            if sc_ref < best_local_score:
                best_local_score = sc_ref
                best_local_seq = a_ref
                print(f"[{time.time():.1f}]   improvement to {best_local_score:.6f}")

        if best_local_seq is not None and best_local_score < best_score:
            best_score = best_local_score
            best_seq = best_local_seq.tolist()
            print(f"[{time.time():.1f}] after L‑BFGS best = {best_score:.6f}")

    # --------------------------------------------------------------
    # 3. Mirror sub‑gradient descent on the exact max‑objective.
    # --------------------------------------------------------------
    if best_seq is not None and time.time() < deadline - 100.0:
        print(f"[{time.time():.1f}] Starting mirror sub‑gradient descent")
        a_sub, sc_sub = mirror_subgrad_optimize(
            np.array(best_seq, dtype=np.float64),
            deadline - 30.0,
            step0=0.8,
        )
        if sc_sub < best_score:
            best_score = sc_sub
            best_seq = a_sub.tolist()
            print(f"[{time.time():.1f}] after mirror sub‑grad best = {best_score:.6f}")

    # --------------------------------------------------------------
    # 4. Symmetrisation (the optimum is often symmetric)
    # --------------------------------------------------------------
    if best_seq is not None:
        a_arr = np.array(best_seq, dtype=np.float64)
        a_sym = (a_arr + a_arr[::-1]) * 0.5
        a_sym = project_to_simplex(a_sym)
        sc_sym = compute_score(a_sym)
        if sc_sym < best_score:
            best_score = sc_sym
            best_seq = a_sym.tolist()
            print(f"[{time.time():.1f}] symmetrisation improved best = {best_score:.6f}")

    # --------------------------------------------------------------
    # 5. Final tiny simulated‑annealing hill‑climb
    # --------------------------------------------------------------
    if best_seq is not None and time.time() < deadline - 5.0:
        print(f"[{time.time():.1f}] Final hill‑climb (SA)")
        a_final, sc_final = hill_climb_sa(
            np.array(best_seq, dtype=np.float64),
            deadline,
            max_steps=600_000,
        )
        if sc_final < best_score:
            best_score = sc_final
            best_seq = a_final.tolist()
            print(f"[{time.time():.1f}] hill‑climb improved best = {best_score:.6f}")

    # --------------------------------------------------------------
    # 6. Safeguard – if nothing succeeded fall back to a modest uniform vector
    # --------------------------------------------------------------
    if best_seq is None:
        n_small = 256
        best_seq = [1.0 / n_small] * n_small
        best_score = evaluate_sequence(best_seq)

    # --------------------------------------------------------------
    # Final official evaluation (the harness will also call evaluate_sequence)
    # --------------------------------------------------------------
    final_score = evaluate_sequence(best_seq)
    print(f"\n[{time.time():.1f}] finished search – final reported score = {final_score:.6f}")

    return best_seq
