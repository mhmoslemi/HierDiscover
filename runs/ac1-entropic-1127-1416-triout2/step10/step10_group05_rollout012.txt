def adam_smooth_optimize(a0, deadline, beta_initial=5.0, beta_target=5000.0, beta_factor=2.0, steps_per_beta=2000, lr=0.5):
    n = a0.shape[0]
    # initialize t = log(a0) - logsumexp to keep scale? Actually we don't need subtract constant.
    t = np.log(np.maximum(a0, 1e-15))
    m = np.zeros_like(t)
    v = np.zeros_like(t)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    best_a = a0.copy()
    best_sc = compute_score(best_a)
    step = 1
    beta = beta_initial
    while beta <= beta_target and time.time() < deadline:
        for i in range(steps_per_beta):
            if time.time() >= deadline:
                break
            # compute a via softmax
            max_t = np.max(t)
            exp_t = np.exp(t - max_t)
            a = exp_t / exp_t.sum()
            # compute smooth objective and gradient wr.t
            conv = convolution_fft(a, a)
            maxc = conv.max()
            # compute w
            exp_shift = np.exp(beta * (conv - maxc))
            Z = exp_shift.sum()
            w = exp_shift / Z
            smooth_max = maxc + np.log(Z) / beta
            f = 2.0 * n * smooth_max
            # grad w.r.t a
            grad_a = 2.0 * convolution_fft(a[::-1], w)[n-1:n-1+n]
            # grad w.r.t t
            dot = np.dot(a, grad_a)
            grad_t = a * (grad_a - dot)
            # Adam updates
            m = beta1 * m + (1 - beta1) * grad_t
            v = beta2 * v + (1 - beta2) * (grad_t ** 2)
            m_hat = m / (1 - beta1 ** step)
            v_hat = v / (1 - beta2 ** step)
            step += 1
            t = t - lr * m_hat / (np.sqrt(v_hat) + eps)
            # optional periodic evaluation
            if i % 200 == 0:
                # compute exact score
                # recompute a to ensure no drift? use current a computed above.
                sc = compute_score(a)
                if sc < best_sc:
                    best_sc, best_a = sc, a.copy()
        # after steps per beta, increase beta
        beta = beta * beta_factor
    # final evaluation
    # compute final a from optimal t
    max_t = np.max(t)
    exp_t = np.exp(t - max_t)
    a = exp_t / exp_t.sum()
    sc = compute_score(a)
    if sc < best_sc:
        best_sc, best_a = sc, a.copy()

def upsample(seq, n):
    old = np.array(seq, dtype=float)
    old_n = old.size
    if old_n == n:
        return old / old.sum()
    # Interpolate linearly over indices
    x_old = np.arange(old_n)
    x_new = np.linspace(0, old_n - 1, n)
    up = np.interp(x_new, x_old, old)
    # Ensure non-negative
    up[up < 0] = 0.
    up = up / up.sum()
    return up
```

"""
Propose a step‑function (non‑negative sequence) that minimises
    2·n·max(a*a) / (sum(a)^2)   (the same as `evaluate_sequence`).

Algorithm
---------
1. Choose a length `n` depending on the remaining time (larger gives more freedom).
2. Build an initial candidate:
   * Upsample the historic solution `height_sequence_1` if available.
   * Else use a simple exponential / power‐law family or a random Dirichlet vector.
3. Refine the candidate by *Adam* optimisation of a *smooth max* surrogate:
   * The objective is `f(a)=2·n*smooth_max(conv(a,a))` with smooth_max approximated
     by a log‑sum‑exp with a continuation schedule for the temperature β.
   * Variables are the unconstrained logits `t` such that `a = softmax(t)`.
   * Adam’s adaptive steps and the β‑schedule give faster convergence than plain
     KL‑mirror descent.
4. Perform a short exact max‑descent (sub‑gradient mirror update) to fine‑tune
   the true non‑smoothed objective.
5. Apply a greedy “targeted hill‑climb” that repeatedly moves a small packet of
   mass away from the index that contributes most to the current maximal
   convolution entry.
6. Return the best sequence seen; the final check uses `evaluate_sequence`.

All helper functions are top‑level; the implementation respects the 1000 s
wall‑clock limit and prints progress information.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Basic linear‑algebra utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Aperiodic convolution via real FFT (zero‑padded)."""
    m = a.size + b.size - 1
    nfft = next_pow2(m)
    A = np.fft.rfft(a, n=nfft)
    B = np.fft.rfft(b, n=nfft)
    C = A * B
    return np.fft.irfft(C, n=nfft)[:m]


def compute_score(a: np.ndarray) -> float:
    """
    Exact objective for a vector already on the probability simplex.
    Returns 2·n·max(a*a) where the convolution is full (length 2·n‑1).
    """
    n = a.shape[0]
    conv = convolution_fft(a, a)
    return 2.0 * n * float(conv.max())


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / v.shape[0])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, v.shape[0] + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.where(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


# ----------------------------------------------------------------------
# Parameterised families of candidate vectors
# ----------------------------------------------------------------------


def exponential_shape(n: int, lam: float) -> np.ndarray:
    """a_i ∝ exp(-lam·i), normalised to sum = 1."""
    i = np.arange(n, dtype=float)
    w = np.exp(-lam * i) if lam > 0.0 else np.ones_like(i)
    return w / w.sum()


def power_shape(n: int, p: float) -> np.ndarray:
    """a_i ∝ (i+1)^(-p), normalised to sum = 1."""
    i = np.arange(1, n + 1, dtype=float)
    w = i ** (-p) if p > 0.0 else np.ones_like(i)
    return w / w.sum()


def upsample_sequence(seq, n: int) -> np.ndarray:
    """Linear interpolation of a short sequence to length `n`, then normalise."""
    old = np.asarray(seq, dtype=float)
    old_n = old.size
    if old_n == n:
        out = old.copy()
    else:
        x_old = np.arange(old_n, dtype=float)
        x_new = np.linspace(0, old_n - 1, n)
        out = np.interp(x_new, x_old, old)
    out[out < 0] = 0.0
    s = out.sum()
    if s == 0.0:
        out = np.full(n, 1.0 / n)
    else:
        out = out / s
    return out


# ----------------------------------------------------------------------
# Adam optimisation of the smooth‑max surrogate
# ----------------------------------------------------------------------


def adam_smooth_optimize(start_a: np.ndarray,
                         deadline: float,
                         beta_initial: float = 5.0,
                         beta_target: float = 5000.0,
                         beta_factor: float = 2.0,
                         steps_per_beta: int = 2000,
                         lr: float = 0.5) -> tuple[np.ndarray, float]:
    """
    minimise `f(a)=2·n·smooth_max(conv(a,a))` using Adam on the logits.
    The temperature β is increased geometrically (continuation) from
    `beta_initial` to `beta_target`.  The method keeps the best exact score
    seen during the run.
    """
    n = start_a.shape[0]
    # initialise logits t = log(a) (softmax will enforce simplex)
    eps = 1e-15
    t = np.log(np.maximum(start_a, eps))

    m = np.zeros_like(t)
    v = np.zeros_like(t)
    beta1, beta2 = 0.9, 0.999
    eps_adam = 1e-8
    step = 1

    best_a = np.exp(t - np.max(t))
    best_a = best_a / best_a.sum()
    best_sc = compute_score(best_a)

    beta = beta_initial
    while beta <= beta_target and time.time() < deadline:
        for _ in range(steps_per_beta):
            if time.time() >= deadline:
                break

            # softmax to get a on the simplex
            max_t = np.max(t)
            exp_t = np.exp(t - max_t)
            a = exp_t / exp_t.sum()

            # smooth max of the convolution
            conv = convolution_fft(a, a)
            maxc = conv.max()
            # numerically stable softmax over convolution entries
            exp_shift = np.exp(beta * (conv - maxc))
            Z = exp_shift.sum()
            w = exp_shift / Z               # behaviour of log‑sum‑exp derivative
            smooth_val = maxc + np.log(Z) / beta
            f = 2.0 * n * smooth_val        # smooth surrogate objective

            # gradient w.r.t. a (from the smooth max)
            grad_a = 2.0 * convolution_fft(a[::-1], w)[n - 1:n - 1 + n]

            # chain rule through the softmax: ∂f/∂t = a ∘ (grad_a - <a,grad_a>)
            dot = np.dot(a, grad_a)
            grad_t = a * (grad_a - dot)

            # Adam updates on the logits
            m = beta1 * m + (1 - beta1) * grad_t
            v = beta2 * v + (1 - beta2) * (grad_t ** 2)
            m_hat = m / (1 - beta1 ** step)
            v_hat = v / (1 - beta2 ** step)

            t = t - lr * m_hat / (np.sqrt(v_hat) + eps_adam)
            step += 1

            # occasional exact evaluation (every 200 steps)
            if step % 200 == 0:
                sc = compute_score(a)
                if sc < best_sc:
                    best_sc, best_a = sc, a.copy()
        # increase temperature
        beta = beta * beta_factor

    # final exact evaluation
    max_t = np.max(t)
    exp_t = np.exp(t - max_t)
    a = exp_t / exp_t.sum()
    sc = compute_score(a)
    if sc < best_sc:
        best_sc, best_a = sc, a.copy()
    return best_a, best_sc


# ----------------------------------------------------------------------
# Exact max‑descent (KL‑mirror) optimisation
# ----------------------------------------------------------------------


def max_descent_optimize(start_a: np.ndarray,
                         deadline: float,
                         max_iters: int = 250_000,
                         eta0: float = 0.4) -> tuple[np.ndarray, float]:
    """
    Sub‑gradient descent on the genuine max‑convolution (no smoothing) using a
    KL‑mirror update, i.e. `a ← a·exp(-η·g)` followed by L1‑normalisation.
    """
    a = np.maximum(start_a.astype(float), 1e-15)
    a /= a.sum()
    n = a.shape[0]

    best_a = a.copy()
    best_sc = compute_score(a)

    for it in range(1, max_iters + 1):
        if time.time() >= deadline:
            break

        conv = convolution_fft(a, a)
        k_max = int(conv.argmax())

        # sub‑gradient of the active convolution entry
        g = np.zeros_like(a)
        i_start = max(0, k_max - (n - 1))
        i_end = min(n - 1, k_max)
        idx = np.arange(i_start, i_end + 1)
        g[idx] = 2.0 * a[k_max - idx]

        eta = eta0 / np.sqrt(it)

        a = np.maximum(a * np.exp(-eta * g), 1e-15)
        a /= a.sum()

        if it % 250 == 0:
            sc = compute_score(a)
            if sc < best_sc:
                best_sc, best_a = sc, a.copy()
    # final check
    sc = compute_score(a)
    if sc < best_sc:
        best_sc, best_a = sc, a.copy()
    return best_a, best_sc

# Targeted hill‑climb polishing
# ----------------------------------------------------------------------


def targeted_hill_climb(start_seq: np.ndarray,
                        deadline: float,
                        max_steps: int = 300_000) -> tuple[np.ndarray, float]:
    """
    Greedy walk that repeatedly extracts a small packet of mass from the index
    that contributes most to the current maximal convolution entry and redistributes
    it elsewhere.  Only moves that strictly improve the exact objective are kept.
    """
    a = start_seq.copy()
    n = a.size
    best_a = a.copy()
    best_sc = compute_score(best_a)

    step_scale = 0.10
    steps = 0

    while steps < max_steps and time.time() < deadline:
        conv = convolution_fft(best_a, best_a)
        k_max = int(conv.argmax())

        i_start = max(0, k_max - (n - 1))
        i_end = min(n - 1, k_max)
        idx = np.arange(i_start, i_end + 1)
        contrib = best_a[idx] * best_a[k_max - idx]

        i_max = idx[np.argmax(contrib)]

        # pick a random destination different from i_max
        j = np.random.randint(n)
        if j == i_max:
            continue

        delta = step_scale * best_a[i_max]
        if delta <= 0.0:
            break

        cand = best_a.copy()
        cand[i_max] -= delta
        cand[j] += delta
        cand = project_to_simplex(cand)

        sc = compute_score(cand)
        if sc < best_sc:
            best_sc, best_a = sc, cand.copy()
            step_scale = min(step_scale * 1.07, 0.5)
        else:
            step_scale *= 0.96

        steps += 1

    return best_a, best_sc

    # Choose a length (larger is better if time permits)
    # -------------------------------------------------
    remaining = deadline - time.time()
    if remaining > 800:
        n = 32768
    elif remaining > 600:
        n = 16384
    elif remaining > 400:
        n = 8192
    else:
        n = 4096
    print(f"[{time.time():.1f}] using sequence length = {n}")

    best_seq = None
    best_score = float("inf")

    # -------------------------------------------------
    # Seed with the historic construction if possible
    # -------------------------------------------------
    try:
        fallback = [float(x) for x in height_sequence_1]   # type: ignore
        fb_vec = upsample_sequence(fallback, n)
        fb_val = evaluate_sequence(fb_vec.tolist())
        print(f"[{time.time():.1f}] fallback score = {fb_val:.6f}")
        if fb_val < best_score:
            best_score = fb_val
            best_seq = fb_vec.tolist()
    except Exception:
        pass

    # -------------------------------------------------
    # Quick scans of simple parametric families
    # -------------------------------------------------
    # Exponential family
    lam_vals = np.logspace(-4, -1, 30)
    for lam in lam_vals:
        if time.time() >= deadline - 300:
            break
        a = exponential_shape(n, lam)
        sc = compute_score(a)
        if sc < best_score:
            best_score, best_seq = sc, a.tolist()
            print(f"[{time.time():.1f}] exponential λ={lam:.4f} → {best_score:.6f}")

    # Power‑law family
    p_vals = np.linspace(0.0, 4.0, 30)
    for p in p_vals:
        if time.time() >= deadline - 300:
            break
        a = power_shape(n, p)
        sc = compute_score(a)
        if sc < best_score:
            best_score, best_seq = sc, a.tolist()
            print(f"[{time.time():.1f}] power p={p:.2f} → {best_score:.6f}")

    # A few random Dirichlet starts for diversification
    for _ in range(3):
        if time.time() >= deadline - 300:
            break
        a_rand = np.random.rand(n)
        a_rand /= a_rand.sum()
        sc = compute_score(a_rand)
        if sc < best_score:
            best_score, best_seq = sc, a_rand.tolist()
            print(f"[{time.time():.1f}] random start improved → {best_score:.6f}")

    # -------------------------------------------------
    # 1. Adam optimisation on a smooth max surrogate
    # -------------------------------------------------
    if best_seq is not None and time.time() < deadline - 150:
        print(f"[{time.time():.1f}] starting Adam smooth optimisation")
        a_opt, sc_opt = adam_smooth_optimize(
            np.array(best_seq, dtype=float),
            deadline - 100,
            beta_initial=5.0,
            beta_target=5000.0,
            beta_factor=2.0,
            steps_per_beta=2000,
            lr=0.5,
        )
        if sc_opt < best_score:
            best_score, best_seq = sc_opt, a_opt.tolist()
            print(f"[{time.time():.1f}] after Adam best = {best_score:.6f}")

    # -------------------------------------------------
    # 2. Exact max‑descent refinement
    # -------------------------------------------------
    if best_seq is not None and time.time() < deadline - 80:
        print(f"[{time.time():.1f}] starting exact max‑descent")
        a_opt2, sc_opt2 = max_descent_optimize(
            np.array(best_seq, dtype=float),
            deadline - 50,
            max_iters=250_000,
            eta0=0.4,
        )
        if sc_opt2 < best_score:
            best_score, best_seq = sc_opt2, a_opt2.tolist()
            print(f"[{time.time():.1f}] after max‑descent best = {best_score:.6f}")

    # -------------------------------------------------
    # 3. Final targeted hill‑climb polishing
    # -------------------------------------------------
    if best_seq is not None and time.time() < deadline - 5:
        print(f"[{time.time():.1f}] final hill‑climb polishing")
        a_final, sc_final = targeted_hill_climb(
            np.array(best_seq, dtype=float),
            deadline,
            max_steps=500_000,
        )
        if sc_final < best_score:
            best_score, best_seq = sc_final, a_final.tolist()
            print(f"[{time.time():.1f}] hill‑climb improved best = {best_score:.6f}")

    # -------------------------------------------------
    # Safety net: uniform tiny vector if nothing succeeded
    # -------------------------------------------------
    if best_seq is None:
        n_small = 256
        best_seq = [1.0 / n_small] * n_small
        best_score = evaluate_sequence(best_seq)

    # -------------------------------------------------
    # Final official evaluation (the harness will also call it)
    # -------------------------------------------------
    final_score = evaluate_sequence(best_seq)
    print(f"\n[{time.time():.1f}] finished – final reported score = {final_score:.6f}")

    # Return a plain Python list as required by the harness
    return best_seq
