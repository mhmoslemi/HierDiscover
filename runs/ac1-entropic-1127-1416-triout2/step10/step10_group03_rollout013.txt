def build_convolution_constraints(f):
    n = len(f)
    rows = []
    cols = []
    data = []
    for k in range(2*n -1):
        # i from max(0, k-(n-1)) to min(k, n-1)
        i_min = max(0, k-(n-1))
        i_max = min(k, n-1)
        for i in range(i_min, i_max+1):
            j = k-i
            # column index j, coefficient f[i]
            rows.append(k)
            cols.append(j)
            data.append(f[i])
    A = sp.coo_matrix((data, (rows, cols)), shape=(2*n-1, n))
    return A

"""
Propose a non‑negative step‑function (list of heights) that minimises

    evaluate_sequence(a) = 2·n·max(conv(a,a)) / (sum(a)^2)

The algorithm improves on the previous state‑of‑the‑art by adding a
Frank–Wolfe‑style linear‑programming refinement step.  The pipeline is:

0.  Initialise a diverse seed set (deterministic shapes, Dirichlet draws,
    random sparse spikes and the optional fallback `height_sequence_1`).

1.  Soft‑max gradient descent on the smooth surrogate
        L_β(a)=log Σ exp(β·conv(a,a))
    with an annealed temperature β (0→2000) and a projected gradient step
    onto the probability simplex (∑a_i=1, a_i≥0).

2.  LP‑based refinement:
    * Let M = max(conv(a,a)).
    * Solve the linear program
          maximise   1ᵀb
          subject to  conv(a,b) ≤ M   (2n‑1 linear inequalities)
                     b ≥ 0
      The solution b typically has sum(b)>1.
    * Scale b to a probability vector and perform a line‑search on the
      mixing parameter t∈[0,1] for  a(t) = (1‑t)·a + t·b, keeping the best
      t that reduces the true objective.
    * Repeat a few times (while time permits).

3.  Projected sub‑gradient descent on the true max‑objective (the same
    routine used previously).

4.  Stochastic hill‑climbing polishing.

All phases respect the global wall‑clock budget (≈1000 s).  The final
vector is returned as a plain Python list; `evaluate_sequence` will be
called once for the official score.
"""

import time
import numpy as np
import scipy.sparse as sp
import scipy.optimize as opt

# ----------------------------------------------------------------------
# Low‑level utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full linear convolution using a real FFT (fast for many calls)."""
    m = a.shape[0] + b.shape[0] - 1
    n_fft = next_pow2(m)
    A = np.fft.rfft(a, n=n_fft)
    B = np.fft.rfft(b, n=n_fft)
    C = A * B
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


def compute_score(a: np.ndarray) -> float:
    """
    Exact objective for a probability vector (∑a_i = 1):
        2·n·max(conv(a,a))
    """
    n = a.shape[0]
    conv = convolution_fft(a, a)
    return 2.0 * n * float(conv.max())


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection onto the probability simplex {x≥0, Σx=1}.
    """
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / v.shape[0])

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, v.shape[0] + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.where(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def generate_deterministic(n: int) -> np.ndarray:
    """Sin² shape a_i = sin²(π·i/(n+1)), normalised to sum = 1."""
    i = np.arange(1, n + 1, dtype=float)
    a = np.sin(np.pi * i / (n + 1.0)) ** 2
    a /= a.sum()
    return a


def generate_beta_family(n: int, p: float) -> np.ndarray:
    """
    Symmetric polynomial shape a_i ∝ ((i+1)*(n-i))**p.
    Positive p clusters mass in the centre, negative p pushes it to the edges.
    """
    i = np.arange(n, dtype=float)
    w = ((i + 1.0) * (n - i)) ** p
    w[w < 0] = 0.0
    a = w / w.sum()
    return a


def random_dirichlet(n: int, conc: float = 1.0) -> np.ndarray:
    """Draw a random probability vector from Dirichlet(conc)."""
    if conc == 1.0:
        a = np.random.rand(n)
    else:
        a = np.random.gamma(conc, 1.0, size=n)
    a /= a.sum()
    return a


def random_spike(n: int, k: int = 70) -> np.ndarray:
    """Very sparse vector: choose `k` distinct positions, assign random weights,
    then normalise."""
    idx = np.random.choice(n, size=min(k, n), replace=False)
    w = np.random.rand(len(idx))
    a = np.zeros(n, dtype=float)
    a[idx] = w
    a /= a.sum()
    return a


def generate_triangular(n: int) -> np.ndarray:
    """Deterministic triangular distribution a_i ∝ (i+1)*(n-i)."""
    i = np.arange(n, dtype=float)
    a = (i + 1.0) * (n - i)
    a /= a.sum()
    return a


def generate_cosine(n: int) -> np.ndarray:
    """Cosine shape a_i = (1 - cos(2π·i/(n-1))) / 2, normalised."""
    i = np.arange(n, dtype=float)
    a = (1.0 - np.cos(2.0 * np.pi * i / (n - 1))) / 2.0
    a /= a.sum()
    return a


# ----------------------------------------------------------------------
# Phase 1 – Soft‑max gradient descent on a smooth max surrogate
# ----------------------------------------------------------------------


def softmax_gradient_descent(
    a: np.ndarray,
    start_score: float,
    deadline: float,
    beta_start: float = 5.0,
    beta_end: float = 2000.0,
    step_coef: float = 0.001,
) -> tuple[np.ndarray, float]:
    """
    Projected gradient descent on
        L_β(a)=log Σ exp(β·conv(a,a))

    β is annealed from `beta_start` to `beta_end`.  The routine stops
    when `deadline` is reached and returns the best iterate (according to
    the exact objective).
    """
    n = a.shape[0]
    cur_a = a.copy()
    best_a = cur_a.copy()
    best_score = start_score

    it = 0
    start = time.time()
    while time.time() < deadline:
        conv = convolution_fft(cur_a, cur_a)
        max_conv = conv.max()
        cur_score = 2.0 * n * float(max_conv)
        if cur_score < best_score:
            best_score = cur_score
            best_a = cur_a.copy()

        # temperature schedule
        elapsed = time.time() - start
        fraction = elapsed / (deadline - start)
        beta = beta_start + fraction * (beta_end - beta_start)

        # soft‑max weights
        shifted = conv - max_conv            # max becomes 0 → stable exp
        exp_vals = np.exp(beta * shifted)
        w = exp_vals / exp_vals.sum()        # soft‑max distribution

        # gradient of L_β
        grad = 2.0 * beta * np.correlate(w, cur_a, mode="valid")

        # step size (decays with √iteration and with β)
        it += 1
        step = step_coef / (np.sqrt(it) * beta)

        nxt = cur_a - step * grad
        nxt = project_to_simplex(nxt)

        cur_a = nxt

        if it % 5000 == 0:
            print(f"[GD] it {it:6d} | time {elapsed:6.1f}s | best {best_score:.6f} | β {beta:.1f}")

    return best_a, best_score


# ----------------------------------------------------------------------
# Phase 2 – Linear‑programming (Frank–Wolfe) refinement
# ----------------------------------------------------------------------


def build_convolution_matrix(f: np.ndarray) -> sp.csr_matrix:
    """
    Build a sparse matrix A such that (A @ b) = conv(f, b) (full linear convolution).
    Shape of A: (2·n‑1, n)
    """
    n = f.shape[0]
    # Each diagonal offset i (0 ≤ i < n) contains the constant value f[i]
    offsets = np.arange(n, dtype=int)
    data = [np.full(n, f[i], dtype=float) for i in range(n)]
    A = sp.diags(data, offsets, shape=(2 * n - 1, n), format="csr")
    return A


def optimal_mix_t(a: np.ndarray, b: np.ndarray, n_iters: int = 20) -> float:
    """
    Ternary search for t∈[0,1] minimising compute_score((1‑t)·a + t·b).
    The objective is convex in t, so ternary search converges quickly.
    """
    lo, hi = 0.0, 1.0
    for _ in range(n_iters):
        t1 = lo + (hi - lo) / 3.0
        t2 = hi - (hi - lo) / 3.0
        s1 = compute_score((1.0 - t1) * a + t1 * b)
        s2 = compute_score((1.0 - t2) * a + t2 * b)
        if s1 < s2:
            hi = t2
        else:
            lo = t1
    return (lo + hi) / 2.0


def lp_refine(
    a: np.ndarray,
    start_score: float,
    deadline: float,
    max_outer_iters: int = 8,
) -> tuple[np.ndarray, float]:
    """
    Frank–Wolfe style improvement:
        * compute current M = max(conv(a,a))
        * solve LP maximise 1ᵀb subject to conv(a,b) ≤ M, b≥0
        * scale b to a probability vector
        * line‑search for optimal mixing t and update a ← (1‑t)a + t b
    The loop repeats while there is time and improvement.
    """
    cur_a = a.copy()
    cur_score = start_score
    best_a = cur_a.copy()
    best_score = cur_score

    n = cur_a.shape[0]

    for outer in range(max_outer_iters):
        if time.time() >= deadline:
            break

        conv = convolution_fft(cur_a, cur_a)
        M = conv.max()

        # Build convolution matrix of cur_a (sparse)
        A = build_convolution_matrix(cur_a)

        # Linear program: maximise sum(b)  ⇔ minimise -sum(b)
        c = -np.ones(n, dtype=float)
        bounds = [(0, None)] * n
        rhs = np.full(2 * n - 1, M, dtype=float)

        # Solve LP with HiGHS (fast, accepts sparse A)
        res = opt.linprog(
            c,
            A_ub=A,
            b_ub=rhs,
            bounds=bounds,
            method="highs",
            options={"presolve": True},
        )

        if not res.success:
            print(f"[LP] iteration {outer}: linprog failed ({res.message})")
            break

        b = res.x
        if b.sum() <= 0.0:
            print(f"[LP] iteration {outer}: trivial solution")
            break

        # Scale to a probability vector
        b = b / b.sum()

        # Find the best mixing parameter t
        t_opt = optimal_mix_t(cur_a, b, n_iters=15)

        if t_opt <= 0.0:
            # No improvement possible
            break

        # Update
        cur_a = (1.0 - t_opt) * cur_a + t_opt * b
        cur_a = project_to_simplex(cur_a)  # safeguard against numerical drift
        cur_score = compute_score(cur_a)

        print(
            f"[LP] iter {outer:2d} | t={t_opt:.4f} | score={cur_score:.6f} | M={M:.6e}"
        )

        if cur_score < best_score - 1e-9:
            best_score = cur_score
            best_a = cur_a.copy()

        # If the LP did not increase the sum(b) enough, we are probably near a fixed point
        if res.fun >= -1.0 - 1e-6:  # sum(b)≈1 → no room for improvement
            print("[LP] sum(b) close to 1, terminating LP refinement")
            break

    return best_a, best_score

def subgradient_refine(
    a: np.ndarray,
    start_score: float,
    deadline: float,
    max_iter: int = 400_000,
    step0: float = 0.07,
) -> tuple[np.ndarray, float]:
    """Projected sub‑gradient descent on the exact L∞‑objective."""
    n = a.shape[0]
    cur_a = a.copy()
    best_a = cur_a.copy()
    best_score = start_score

    for t in range(1, max_iter + 1):
        if time.time() >= deadline:
            break

        conv = convolution_fft(cur_a, cur_a)
        k = int(np.argmax(conv))

        # sub‑gradient of max(conv) at shift k
        idx = np.arange(n)
        j = k - idx
        mask = (j >= 0) & (j < n)
        g = np.zeros_like(cur_a)
        g[mask] = 2.0 * cur_a[j[mask]]

        step = step0 / np.sqrt(t)
        cur_a -= step * g
        cur_a = project_to_simplex(cur_a)

        cur_score = compute_score(cur_a)
        if cur_score < best_score:
            best_score = cur_score
            best_a = cur_a.copy()

        if t % 1000 == 0:
            # small random jitter to avoid stagnation
            jitter = np.random.randn(n) * (step0 * 0.004 / np.sqrt(t))
            cur_a += jitter
            cur_a = project_to_simplex(cur_a)

    return best_a, best_score

def hill_climb(
    a: np.ndarray,
    deadline: float,
    max_steps: int = 800_000,
) -> tuple[np.ndarray, float]:
    """Stochastic moves that are kept only when they improve the exact score."""
    n = a.shape[0]
    best_a = a.copy()
    best_score = compute_score(best_a)
    step_scale = 0.08
    steps = 0

    while steps < max_steps and time.time() < deadline:
        cand = best_a.copy()
        if np.random.rand() < 0.5:  # occasional swap
            i, j = np.random.randint(n, size=2)
            cand[i], cand[j] = cand[j], cand[i]
        else:  # small mass transfer
            i, j = np.random.randint(n, size=2)
            if cand[i] > 0.0 and i != j:
                delta = step_scale * cand[i]
                cand[i] -= delta
                cand[j] += delta
                if cand[i] < 0.0:
                    cand[i] = 0.0

        cand /= cand.sum()                     # renormalise
        cur_score = compute_score(cand)

        if cur_score < best_score:
            best_score = cur_score
            best_a = cand
            step_scale = min(step_scale * 1.07, 0.5)
        else:
            step_scale *= 0.96

        steps += 1

    return best_a, best_score

        print(f"Fallback score = {fb:.6f}")

    except Exception:

    # 1. Choose the sequence length (larger n gives a better asymptotic bound)
    # --------------------------------------------------------------
    n = 2048
    if total_time > 800.0:

    # --------------------------------------------------------------
    # 2. Generate a diverse set of seeds and keep the best one
    # --------------------------------------------------------------
    seeds = []

    # deterministic families
    seeds.append(generate_deterministic(n))
    seeds.append(generate_triangular(n))
    seeds.append(generate_cosine(n))

        seeds.append(generate_beta_family(n, p))

    # Evaluate all seeds (exact score) and keep the best
    for s in seeds:
        s = np.asarray(s, dtype=float)
        s = project_to_simplex(s)

    )

            print(f"After LP refinement: best score = {best_val:.6f}")

    # --------------------------------------------------------------
    # 6. Phase 3 – Sub‑gradient refinement
    # --------------------------------------------------------------
    if time.time() < deadline - 15.0:
        print("\n[Phase 3] Sub‑gradient refinement")
        phase3_deadline = deadline - 15.0
        cur_a, cur_score = subgradient_refine(cur_a, cur_score, phase3_deadline, max_iter=800_000)
        if cur_score < best_val:
            best_val = cur_score
            print(f"After sub‑grad: best score = {best_val:.6f}")

    # --------------------------------------------------------------
    # 7. Phase 4 – Hill‑climb polishing
    # --------------------------------------------------------------
    if time.time() < deadline - 1.0:
        print("\n[Phase 4] Hill‑climb polishing")
        cur_a, cur_score = hill_climb(cur_a, deadline, max_steps=1_200_000)
        if cur_score < best_val:
            best_val = cur_score
            print(f"After polishing: best score = {best_val:.6f}")

    # --------------------------------------------------------------
    # 8. Final conversion and evaluation
    # --------------------------------------------------------------
    final_seq = cur_a.tolist()
    final_val = evaluate_sequence(final_seq)
    print(f"\nFinished search. Final reported score = {final_val:.6f}")

    return final_seq
