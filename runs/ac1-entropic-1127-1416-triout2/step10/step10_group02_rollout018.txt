    a = a0.copy()
    best_a = a.copy()
    best_val = compute_score(a)
    beta = 0.5
    iter = 0

        e = np.exp(beta * (conv - maxc))

def adam_optimize(start_vec, time_deadline, beta1=0.9, beta2=0.999, eps=1e-8, lr0=0.05):
    a = start_vec.copy()
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    t = 0
    best_a = a.copy()
    best_score = compute_score(a)
    beta = 0.5
    # schedule for beta (increase gradually)
    while time.time() < time_deadline:
        t += 1
        conv = full_convolution_fft(a)
        maxc = conv.max()
        # compute softmax weights
        e = np.exp(beta * (conv - maxc))
        w = e / e.sum()
        # gradient
        g = 2.0 * cross_corr_fft(w, a)
        # Adam update
        m = beta1 * m + (1 - beta1) * g
        v = beta2 * v + (1 - beta2) * (g * g)
        m_hat = m / (1 - beta1 ** t)
        v_hat = v / (1 - beta2 ** t)
        step = lr0 / (np.sqrt(v_hat) + eps)
        a = a - step * m_hat
        a = project_to_simplex(a)

        # evaluate real objective using conv (already have)
        score = compute_score(a)  # 2*n*conv.max()
        if score < best_score:
            best_score = score
            best_a = a.copy()
        # update beta schedule
        if t % 200 == 0:
            beta = min(beta * 1.1, 10.0)  # increasing
    return best_a
```

def cross_corr_fft(w, a):
    n = a.shape[0]
    conv_len = len(w) + n - 1
    n_fft = 1 << ((conv_len - 1).bit_length())
    W = np.fft.rfft(w, n=n_fft)
    A_rev = np.fft.rfft(a[::-1], n=n_fft)
    C = W * A_rev
    conv = np.fft.irfft(C, n=n_fft)[:conv_len]
    return conv[n-1:n-1+n]
```

def smooth_optimize_adam(a0, deadline,
                         beta_start=0.5, beta_factor=1.2,
                         beta_max=10.0,
                         lr=0.05, eps=1e-8):

    n = a.shape[0]
    best_a = a.copy()
    best_val = compute_score_fast(a)
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    t = 0
    beta = beta_start
    while time.time() < deadline:
        t += 1
        conv = full_convolution_fft(a)
        maxc = conv.max()
        # compute softmax weights safely
        # scale to avoid overflow
        # note: exp(beta*(conv-maxc)) in [0,1]
        w = np.exp(beta * (conv - maxc))
        w_sum = w.sum()
        if w_sum == 0:
            # all underflow (beta large). fallback to subgradient of max entry
            # take k = argmax conv
            k = int(np.argmax(conv))
            # gradient of conv[k] is 2 * a shifted (as earlier)
            idx = np.arange(n)

            mask = (j >= 0) & (j < n)

            best_val = val
            best_a = a.copy()
        # schedule beta increase
        if t % 200 == 0:
            beta = min(beta * beta_factor, beta_max)
    return best_a

def hill_climb(a_start, deadline, max_no_improve=200000):

"""
non‑negative step‑function optimiser for ``evaluate_sequence``.

The new approach replaces the simple sub‑gradient step with a
continuation method: minimise a smoothed version of the
max‑convolution using a soft‑max (log‑sum‑exp) surrogate.
The surrogate gradient is the cross‑correlation of the weights
with the current vector, which can be obtained by a single FFT‑based
convolution.  An Adam optimiser with a decaying learning‑rate and a
gradually increasing soft‑max temperature (β) is used.  After the
continuation phase a cheap greedy hill‑climb refines the best
candidate.  Multiple random restarts are performed while respecting
the 1000 s wall‑clock budget.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Simplex projection (Euclidean projection onto the probability simplex)
# ----------------------------------------------------------------------
def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Project a vector onto the simplex {x >= 0, sum(x)=1}."""
    if np.all(v <= 0):
        # uniform if the whole vector is non‑positive
        return np.full_like(v, 1.0 / v.shape[0])

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, v.shape[0] + 1)
    mask = u - cssv / ind > 0
    rho = np.where(mask)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# FFT‑based convolutions (full auto‑convolution and cross‑correlation)
# ----------------------------------------------------------------------
def full_convolution_fft(a: np.ndarray) -> np.ndarray:
    """Linear convolution a*a (length 2·len(a)−1) via real FFT."""
    n = a.shape[0]
    m = 2 * n - 1
    n_fft = 1 << ((m - 1).bit_length())          # next power of two
    A = np.fft.rfft(a, n=n_fft)
    C = A * A
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


def cross_corr_fft(w: np.ndarray, a: np.ndarray) -> np.ndarray:
    """
    Cross‑correlation g_i = Σ_j w_{i+j} a_j for i=0…n−1.
    Obtained as a slice of the convolution of w with the reversed a.
    """
    n = a.shape[0]
    conv_len = len(w) + n - 1
    n_fft = 1 << ((conv_len - 1).bit_length())
    W = np.fft.rfft(w, n=n_fft)
    A_rev = np.fft.rfft(a[::-1], n=n_fft)
    C = W * A_rev
    conv = np.fft.irfft(C, n=n_fft)[:conv_len]
    # slice that corresponds to indices n‑1 … n‑1+n‑1
    return conv[n - 1:n - 1 + n]


# ----------------------------------------------------------------------
# Objective helpers
# ----------------------------------------------------------------------
def compute_score_fast(a: np.ndarray) -> float:
    """Exact objective for a vector normalised to sum = 1:
    2·n·max(conv(a,a)).
    """
    n = a.shape[0]
    conv = full_convolution_fft(a)
    return float(2.0 * n * conv.max())


def power_law_vector(n: int, p: float) -> np.ndarray:
    """a_i ∝ (i+1)^{−p}, normalised to sum = 1."""
    a = np.arange(1, n + 1, dtype=float) ** (-p)
    a /= a.sum()
    return a


# ----------------------------------------------------------------------
# Smoothing optimiser (Adam on a soft‑max surrogate of the max‑conv)
# ----------------------------------------------------------------------
def smooth_optimize_adam(a0: np.ndarray,
                         time_deadline: float,
                         beta_start: float = 0.5,
                         beta_factor: float = 1.2,
                         beta_max: float = 10.0,
                         lr0: float = 0.08,
                         eps: float = 1e-8) -> np.ndarray:
    """
    Adam‑style projected optimisation of the smoothed objective
    F_β(a) = (1/β)·log( Σ_k exp(β·conv_k) ).
    β is increased periodically (continuation) so that the algorithm
    gradually focuses on the true max‑convolution.
    """
    a = a0.copy()
    n = a.shape[0]

    # Adam state
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1, beta2 = 0.9, 0.999
    t = 0

    best_a = a.copy()
    best_val = compute_score_fast(a)

    beta = beta_start

    while time.time() < time_deadline:
        t += 1

        # ----- compute current convolution -----
        conv = full_convolution_fft(a)               # length 2n‑1
        maxc = conv.max()

        # ----- soft‑max weights (stable) -----
        # exp(beta*(conv-maxc)) ∈ (0,1]  → at least one entry = 1
        w = np.exp(beta * (conv - maxc))
        w /= w.sum()                                 # normalised weights

        # ----- gradient of the smoothed objective -----
        # ∇F = Σ_k w_k ∇conv_k  = 2·cross_corr(w, a)
        g = 2.0 * cross_corr_fft(w, a)

        # ----- Adam update -----
        m = beta1 * m + (1.0 - beta1) * g
        v = beta2 * v + (1.0 - beta2) * (g * g)

        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)

        step = lr0 / (np.sqrt(v_hat) + eps)
        a = a - step * m_hat

        # ----- projection onto the simplex -----
        a = project_to_simplex(a)

        # ----- evaluate the true objective -----
        cur_val = compute_score_fast(a)
        if cur_val < best_val:
            best_val, best_a = cur_val, a.copy()

        # ----- continuation: increase β periodically -----
        if t % 250 == 0:
            beta = min(beta * beta_factor, beta_max)

        # ----- occasional jitter to escape tiny plateaus -----
        if t % 1000 == 0:
            a += np.random.randn(n) * (lr0 * 0.02 / np.sqrt(t))
            a = project_to_simplex(a)

    return best_a


# ----------------------------------------------------------------------
# Simple greedy hill‑climb (same idea as the reference implementation)
# ----------------------------------------------------------------------
def hill_climb(a_start: np.ndarray,
               time_deadline: float,
               max_no_improve: int = 400_000) -> np.ndarray:
    """
    Stochastic mass‑transfer moves with greedy acceptance.
    Each move transfers a fraction of the mass from i to j and the
    result is projected back onto the simplex.
    """
    a = a_start.copy()
    n = a.shape[0]
    best_a = a.copy()
    best_val = compute_score_fast(a)

    step_scale = 0.05
    it = 0
    no_improve = 0

    while time.time() < time_deadline and it < max_no_improve:
        i, j = np.random.randint(n, size=2)
        if i == j:
            continue

        delta = step_scale * a[i]
        cand = a.copy()
        cand[i] -= delta
        cand[j] += delta

        # numerical safety
        cand = np.maximum(cand, 0.0)
        cand = project_to_simplex(cand)

        val = compute_score_fast(cand)
        if val < best_val:
            best_val = val
            best_a = cand.copy()
            a = cand
            step_scale = min(step_scale * 1.07, 0.5)
            no_improve = 0
        else:
            step_scale *= 0.985
            no_improve += 1

        it += 1
        if no_improve > 5_000:
            # slowly shrink the step if we are stuck
            step_scale = max(step_scale * 0.90, 1e-4)

    return best_a
