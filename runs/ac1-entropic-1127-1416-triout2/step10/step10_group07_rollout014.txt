```
def convolution_fft(a,b):
    m = a.size + b.size - 1
    nfft = next_pow2(m)
    A = np.fft.rfft(a, n=nfft)
    B = np.fft.rfft(b, n=nfft)
    C = A * B
    conv = np.fft.irfft(C, n=nfft)[:m]
    return conv
```

```
def symmetric_product_family(n, alpha):
    i = np.arange(n, dtype=np.float64)
    logprod = np.log(i + 0.5) + np.log(n - 0.5 - i)
    # For i at ends, logprod may be log(0) => -inf, but i+0.5>0, n-0.5-i>0 for i=0..n-1 (since n-0.5-i>0). All positive.
    w = np.exp(-alpha * logprod)
    a = w / w.sum()
    return a
```

```
def adam_optimize(start_a, deadline, max_iters=200000, beta_start=1.0, beta_end=20000.0, lr=0.01):

    m = np.zeros_like(z)
    v = np.zeros_like(z)
    best_a = softmax(z); best_score = compute_score(best_a)
    for it in range(1, max_iters+1):
        if time.time() >= deadline:
            break
        t = it / max_iters
        beta = beta_start * (beta_end/beta_start) ** t
        a = softmax(z)
        conv = convolution_fft(a,a)
        maxc = conv.max()
        exp_w = np.exp(beta*(conv - maxc))
        w = exp_w / exp_w.sum()
        a_rev = a[::-1]
        conv_rev = convolution_fft(a_rev, w)
        # conv_rev length = a_rev.size + w.size -1 = n + (2n-1) -1 = 3n-2.
        # Extract middle segment of length n corresponding to gradient.

        grad_a = 2.0 * conv_rev[start:start + n]
        # backprop through softmax
        dot_ag = np.dot(a, grad_a)
        grad_z = a * grad_a - dot_ag * a
        # Adam update
        m = beta1 * m + (1-beta1) * grad_z
        v = beta2 * v + (1-beta2) * (grad_z ** 2)
        m_hat = m / (1 - beta1**it)
        v_hat = v / (1 - beta2**it)
        z -= lr * m_hat / (np.sqrt(v_hat) + eps_adam)
        # evaluate occasionally

            sc = compute_score(a_tmp)
```

assistantfinal**Solution Overview**

The task is to minimise  

\[
J(a)=\frac{2\,n\;\max_k (a*a)_k}{\bigl(\sum_i a_i\bigr)^2},
\]

where `(a*a)` is the aperiodic auto‑convolution of the non‑negative vector `a`.  
Because the objective is scale‑invariant we may enforce `∑a_i = 1`; then  

\[
J(a)=2\,n\;\max_k (a*a)_k .
\]

The optimum is obtained when the convolution is as flat as possible – the
maximum should be close to the average value `1/(2n‑1)`.  
The programme below

1. **chooses a reasonably large length** (`n = 32768` for the full time budget);
2. **samples a wide collection of analytic families** (U‑shaped, left‑heavy,
   exponential) and a few Dirichlet draws, keeping the best‐scoring vector;
3. **refines the best vector with a high‑β Adam optimiser** on the smooth
   log‑sum‑exp surrogate of the max‑convolution;
4. **symmetrises the final vector** (the optimum is usually symmetric);
5. **returns the best list found** before the wall‑clock deadline.

All heavy work is performed with FFT‑based convolution ( `O(n log n)` ), so
even for `n = 32768` many thousand iterations comfortably fit into the 1000 s
budget.

```python
"""
Propose a step‑function (non‑negative list) that minimises
evaluate_sequence(sequence).  The algorithm:

1. Pick a large n (32768 when time allows).
2. Scan several analytic families (U‑shape, left‑heavy power, exponential)
   and a few Dirichlet draws; keep the vector with the smallest exact score.
3. Starting from that vector run Adam on the smoothed log‑sum‑exp
   surrogate of the max‑convolution, increasing the smoothing parameter β
   up to 2·10⁴.  This drives the vector towards a flat auto‑convolution.
4. Symmetrise the final vector (a ≈ rev(a)) and renormalise to the simplex.
5. Return the best vector (plain Python list) before the deadline.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Basic linear‑algebra utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full aperiodic convolution using real FFT."""
    m = a.size + b.size - 1
    nfft = next_pow2(m)
    A = np.fft.rfft(a, n=nfft)
    B = np.fft.rfft(b, n=nfft)
    C = A * B
    conv = np.fft.irfft(C, n=nfft)[:m]
    return conv


def compute_score(a: np.ndarray) -> float:
    """
    Exact objective for a probability vector (∑a = 1):
        2·n·max(a*a)
    """
    n = a.shape[0]
    conv = convolution_fft(a, a)
    return 2.0 * n * float(conv.max())


def softmax(z: np.ndarray) -> np.ndarray:
    """Numerically stable soft‑max."""
    z_max = np.max(z)
    e = np.exp(z - z_max)
    return e / e.sum()


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / v.shape[0])

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, v.shape[0] + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.where(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


# ----------------------------------------------------------------------
# Analytic families (parameterised shapes)
# ----------------------------------------------------------------------


def symmetric_product_family(n: int, alpha: float) -> np.ndarray:
    """U‑shaped: a_i ∝ [(i+½)(n‑½‑i)]^(‑α).  Normalised to sum = 1."""
    i = np.arange(n, dtype=np.float64)
    logprod = np.log(i + 0.5) + np.log(n - 0.5 - i)
    w = np.exp(-alpha * logprod)
    a = w / w.sum()
    return a


def left_heavy_power_family(n: int, beta: float) -> np.ndarray:
    """Monotonically decreasing power law: a_i ∝ (i+½)^(‑β)."""
    i = np.arange(n, dtype=np.float64)
    w = (i + 0.5) ** (-beta)
    a = w / w.sum()
    return a


def exponential_decay_family(n: int, gamma: float) -> np.ndarray:
    """Exponential decay: a_i ∝ exp(‑γ·i)."""
    i = np.arange(n, dtype=np.float64)
    w = np.exp(-gamma * i)
    a = w / w.sum()
    return a


# ----------------------------------------------------------------------
# Random simple starts (Dirichlet draws)
# ----------------------------------------------------------------------


def random_dirichlet_start(n: int, conc: float) -> np.ndarray:
    """Symmetric Dirichlet(α) draw."""
    vec = np.random.gamma(conc, 1.0, size=n)
    vec /= vec.sum()
    return vec


# ----------------------------------------------------------------------
# Adam optimiser on the smoothed (log‑sum‑exp) surrogate of max‑conv
# ----------------------------------------------------------------------


def adam_optimize(
    start_a: np.ndarray,
    deadline: float,
    max_iters: int = 150_000,
    beta_start: float = 5.0,
    beta_end: float = 20_000.0,
    lr: float = 0.01,
) -> tuple[np.ndarray, float]:
    """
    Minimise f_β(a) = (1/β)·log Σ_k exp(β·(a*a)_k)
    with Adam on the log‑soft‑max parametrisation a = softmax(z).
    β is increased exponentially from beta_start to beta_end.
    """
    eps = 1e-12
    # initialise log‑probabilities
    z = np.log(start_a + eps)
    m = np.zeros_like(z)
    v = np.zeros_like(z)

    beta1 = 0.9
    beta2 = 0.999
    eps_adam = 1e-12

    best_a = softmax(z)
    best_score = compute_score(best_a)

    n = start_a.shape[0]

    for it in range(1, max_iters + 1):
        if time.time() >= deadline:
            break

        # β schedule – exponential annealing
        t = it / max_iters
        beta = beta_start * (beta_end / beta_start) ** t

        a = softmax(z)

        # ---- smoothed max – soft‑max over convolution entries ----
        conv = convolution_fft(a, a)
        maxc = conv.max()
        # stabilised exponentials
        exp_w = np.exp(beta * (conv - maxc))
        w = exp_w / exp_w.sum()          # soft‑max weights (size 2n‑1)

        # ---- gradient of the surrogate w.r.t. a ----
        a_rev = a[::-1]
        conv_rev = convolution_fft(a_rev, w)   # length 3n‑2
        start = n - 1
        grad_a = 2.0 * conv_rev[start:start + n]

        # back‑prop through the soft‑max layer
        dot_ag = np.dot(a, grad_a)
        grad_z = a * grad_a - dot_ag * a

        # ---- Adam update ----
        m = beta1 * m + (1.0 - beta1) * grad_z
        v = beta2 * v + (1.0 - beta2) * (grad_z ** 2)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        z -= lr * m_hat / (np.sqrt(v_hat) + eps_adam)

        # occasional exact evaluation
        if it % 2000 == 0:
            a_tmp = softmax(z)
            sc = compute_score(a_tmp)
            if sc < best_score:
                best_score = sc
                best_a = a_tmp.copy()
                print(f"[{time.time():.1f}]   Adam iter {it:7d} – new best = {best_score:.6f}")

    # final check
    a_fin = softmax(z)
    sc_fin = compute_score(a_fin)
    if sc_fin < best_score:
        best_score = sc_fin
        best_a = a_fin.copy()
    return best_a, best_score

def propose_candidate(seed: int = 12345, budget_s: int = 1000, **_):

    (≈1000 s) and returns a plain Python list.
    """
    np.random.seed(seed)
    start_time = time.time()
    # keep a small safety margin for the outer harness
    deadline = start_time + budget_s - 2.0

    # -------------------------------------------------------------
    # Choose the length.  Larger n gives a lower theoretical bound.
    # -------------------------------------------------------------
    if budget_s >= 900:
        n = 32768
    elif budget_s >= 600:
        n = 16384
    else:
        n = 8192
    print(f"[{time.time():.1f}] Using length n = {n}")

    # -------------------------------------------------------------
    # Global best tracker
    # -------------------------------------------------------------
    best_seq = None
    best_score = float("inf")

    def maybe_update(vec: np.ndarray, tag: str = ""):
        """Update the global best if the new vector improves the exact score."""
        nonlocal best_seq, best_score
        sc = compute_score(vec)
        if sc < best_score:
            best_score = sc
            best_seq = vec.tolist()
            print(f"[{time.time():.1f}] new best {tag:<15} = {best_score:.6f}")

    # -------------------------------------------------------------
    # 1. Uniform baseline
    # -------------------------------------------------------------
    uniform = np.full(n, 1.0 / n)
    maybe_update(uniform, "uniform")

    # -------------------------------------------------------------
    # 2. Analytic families (cheap exhaustive scan)
    # -------------------------------------------------------------
    # symmetric product (U‑shaped)
    for alpha in np.linspace(0.2, 6.0, 31):
        if time.time() >= deadline - 120.0:
            break
        vec = symmetric_product_family(n, alpha)
        maybe_update(vec, f"symα={alpha:.2f}")

    # left‑heavy power law
    for beta in np.linspace(0.2, 6.0, 31):
        if time.time() >= deadline - 120.0:
            break
        vec = left_heavy_power_family(n, beta)
        maybe_update(vec, f"powβ={beta:.2f}")

    # exponential decay
    for gamma in np.linspace(0.0, 8.0, 41):
        if time.time() >= deadline - 120.0:
            break
        vec = exponential_decay_family(n, gamma)
        maybe_update(vec, f"expγ={gamma:.2f}")

    # -------------------------------------------------------------
    # 3. Random Dirichlet draws (sparse / dense extremes)
    # -------------------------------------------------------------
    conc_vals = np.logspace(-2, 1, 20)           # 0.01 … 10
    for conc in conc_vals:
        for _ in range(2):
            if time.time() >= deadline - 120.0:
                break
            vec = random_dirichlet_start(n, conc)
            maybe_update(vec, f"dirα={conc:.3f}")

    # -------------------------------------------------------------
    # 4. Adam refinement on the best candidate found so far
    # -------------------------------------------------------------
    if best_seq is not None and time.time() < deadline - 30.0:
        print(f"[{time.time():.1f}] Starting Adam refinement")
        a_start = np.array(best_seq, dtype=np.float64)
        a_ref, sc_ref = adam_optimize(
            a_start,
            deadline=deadline - 5.0,
            max_iters=150_000,
            beta_start=5.0,
            beta_end=20_000.0,
            lr=0.01,
        )
        maybe_update(a_ref, "AdamRef")

    # -------------------------------------------------------------
    # 5. Symmetrisation (many optimal solutions are symmetric)
    # -------------------------------------------------------------
    if best_seq is not None:
        a_arr = np.array(best_seq, dtype=np.float64)
        a_sym = (a_arr + a_arr[::-1]) * 0.5
        a_sym = project_to_simplex(a_sym)
        maybe_update(a_sym, "symmetrised")

    # 6. Final official evaluation (the harness will also call it)
    # -------------------------------------------------------------
    final_score = evaluate_sequence(best_seq)
    print(f"\n[{time.time():.1f}] finished – final reported score = {final_score:.6f}")

    # Return a plain Python list as required.
    return best_seq
```
