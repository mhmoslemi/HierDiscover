assistantfinal**Solution Overview**

The program searches for a non‑negative step‑function (a probability vector) that minimizes  

\[
\operatorname{score}(a)=\frac{2\,n\;\max (a*a)}{(\sum a)^2}
\]

which, after normalising `∑a = 1`, is simply  

\[
\operatorname{score}(a)=2\,n\;\max(a*a) .
\]

The algorithm proceeds in three phases:

1. **Analytic seeding** – a dense scan of a few one‑parameter families
   (symmetric product, left‑heavy power, exponential‑U) for `n = 2048`
   and a fine golden‑section search for the best parameter of the
   symmetric product family. The best vector found becomes the start
   point.

2. **Smooth‑max Adam optimisation** – the vector is forced to be
   symmetric (`a[i]=a[n‑1‑i]`) by optimising only the first half, using a
   soft‑max parametrisation (`p = softmax(z)`).  The smoothed max
   \(\frac{1}{\beta}\log\sum\exp(\beta\,(a*a)_k)\) is minimised with Adam.
   The smoothing parameter `β` is increased exponentially from `5` to
   `2·10⁴`, giving an increasingly accurate approximation of the true
   max.  The best iterate is stored.

3. **Projected sub‑gradient descent on the true max** – starting from the
   Adam result a classic sub‑gradient method is run for a large number of
   iterations (limited only by the remaining wall‑clock time).  Each step
   moves a small amount of probability mass away from the indices that
   realise the current maximal convolution value.  After every
   `report_interval` steps the exact score is evaluated and the best
   vector kept.

A final cheap simulated‑annealing hill‑climb is performed if time permits.
The whole routine respects the 1000 s budget, prints progress messages,
and finally returns the best sequence as a plain `list[float]`.

```python
"""
propose_candidate – search for a step‑function with minimal
max‑auto‑convolution score.

Phase 0 : analytic families (fine α‑search for symmetric product)
Phase 1 : symmetric Adam optimiser on a smoothed max (β ↑ 2·10⁴)
Phase 2 : projected sub‑gradient descent on the true max
Phase 3 : tiny simulated‑annealing hill‑climb (if time left)

All vectors are kept on the probability simplex (∑a = 1) and are
forced to be symmetric, which greatly reduces the dimensionality
and improves convergence.
"""
import time
import numpy as np
from typing import List, Tuple

# ----------------------------------------------------------------------
# Core utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full aperiodic convolution using real FFT (length = len(a)+len(b)-1)."""
    m = a.size + b.size - 1
    nfft = next_pow2(m)
    A = np.fft.rfft(a, nfft)
    B = np.fft.rfft(b, nfft)
    C = A * B
    return np.fft.irfft(C, nfft)[:m]


def compute_score(a: np.ndarray) -> float:
    """
    Exact objective for a probability vector (∑a = 1):
    score = 2·n·max(conv(a,a))
    """
    n = a.shape[0]
    conv = convolution_fft(a, a)
    return 2.0 * n * float(conv.max())


def simple_simplex_proj(v: np.ndarray) -> np.ndarray:
    """Clamp negative entries, renormalise to unit sum (fast, not Euclidean)."""
    v = np.maximum(v, 0.0)
    s = v.sum()
    if s <= 0.0:
        return np.full_like(v, 1.0 / v.shape[0])
    return v / s


def softmax(z: np.ndarray) -> np.ndarray:
    """Numerically stable soft‑max."""
    m = np.max(z)
    e = np.exp(z - m)
    return e / e.sum()


# ----------------------------------------------------------------------
# Analytic families (used for seeding)
# ----------------------------------------------------------------------


def symmetric_product_family(n: int, alpha: float) -> np.ndarray:
    """
    a_i ∝ [(i+½)(n‑½‑i)]^{‑α}, symmetric, sum‑normalised.
    """
    i = np.arange(n, dtype=np.float64)
    logprod = np.log(i + 0.5) + np.log(n - 0.5 - i)
    w = np.exp(-alpha * logprod)
    # protect against overflow for α=0 (logprod may be -inf)
    w[np.isinf(w)] = np.finfo(np.float64).max
    w /= w.sum()
    return w


def left_heavy_power_family(n: int, beta: float) -> np.ndarray:
    """a_i ∝ (i+½)^{‑β}, monotone decreasing."""
    i = np.arange(n, dtype=np.float64)
    w = np.exp(-beta * np.log(i + 0.5))
    w /= w.sum()
    return w


def exp_u_family(n: int, gamma: float) -> np.ndarray:
    """U‑shape: a_i ∝ exp(-γ·min(i, n‑1‑i))."""
    i = np.arange(n, dtype=np.float64)
    d = np.minimum(i, n - 1 - i)
    w = np.exp(-gamma * d)
    w /= w.sum()
    return w


# ----------------------------------------------------------------------
# Golden‑section search for the best α (symmetric product)
# ----------------------------------------------------------------------


def best_alpha_symmetric_product(n: int,
                                 lo: float = 0.0,
                                 hi: float = 15.0,
                                 max_iter: int = 30) -> Tuple[float, np.ndarray]:
    """
    Fine search of α ∈ [lo,hi] for the symmetric product family.
    Returns (α_best, vector_best) with the smallest exact score.
    """
    phi = (np.sqrt(5) - 1) / 2  # 0.618...
    a = lo
    b = hi
    # initial points
    c = b - phi * (b - a)
    d = a + phi * (b - a)
    vec_c = symmetric_product_family(n, c)
    vec_d = symmetric_product_family(n, d)
    sc_c = compute_score(vec_c)
    sc_d = compute_score(vec_d)

    for _ in range(max_iter):
        if sc_c < sc_d:
            b, vec_d, sc_d = d, vec_c, sc_c
            d = a + phi * (b - a)
            vec_d = symmetric_product_family(n, d)
            sc_d = compute_score(vec_d)
        else:
            a, vec_c, sc_c = c, vec_d, sc_d
            c = b - phi * (b - a)
            vec_c = symmetric_product_family(n, c)
            sc_c = compute_score(vec_c)
        if b - a < 1e-4:
            break

    if sc_c < sc_d:
        return c, vec_c
    else:
        return d, vec_d


# ----------------------------------------------------------------------
# Smoothed‑max Adam optimiser (symmetric parametrisation)
# ----------------------------------------------------------------------


def adam_symmetric_optimize(start_half: np.ndarray,
                            deadline: float,
                            max_iters: int = 300_000,
                            beta_start: float = 5.0,
                            beta_end: float = 20000.0,
                            lr: float = 0.018,
                            report_interval: int = 4000) -> Tuple[np.ndarray, float]:
    """
    Adam optimiser on the smoothed max of the autocorrelation.
    The full vector is built as
        a = 0.5 * concatenate([p, p[::-1]])   with p = softmax(z).
    Returns the best full‑length vector and its exact score.
    """
    n = start_half.size * 2
    half_len = start_half.size
    eps = 1e-12

    # initialise logits from the start distribution (add eps for safety)
    z = np.log(start_half + eps)

    m_adam = np.zeros_like(z)
    v_adam = np.zeros_like(z)
    beta1, beta2 = 0.9, 0.999
    eps_adam = 1e-12

    # best bookkeeping
    p = softmax(z)
    a_full = np.concatenate([p, p[::-1]]) * 0.5
    best_a = a_full.copy()
    best_sc = compute_score(best_a)

    for it in range(1, max_iters + 1):
        if time.time() >= deadline:
            break

        # exponential schedule for β
        t_frac = it / max_iters
        beta = beta_start * (beta_end / beta_start) ** t_frac

        # current probability half‑vector
        p = softmax(z)               # sum(p)=1
        a_full = np.concatenate([p, p[::-1]]) * 0.5   # sum(a_full)=1

        # convolution and smooth‑max weights
        conv = convolution_fft(a_full, a_full)
        maxc = conv.max()
        # stable log‑sum‑exp
        w_exp = np.exp(beta * (conv - maxc))
        w = w_exp / w_exp.sum()

        # gradient w.r.t the full vector a_full
        rev_a = a_full[::-1]
        conv_rev = convolution_fft(rev_a, w)          # length = 2n‑1
        grad_full = 2.0 * conv_rev[n - 1:n - 1 + n]   # size n

        # back‑prop to the half‑vector (factor 0.5 from the symmetric construction)
        grad_half = 0.5 * (grad_full[:half_len] + grad_full[-half_len:][::-1])

        # back‑prop through the soft‑max
        dot = np.dot(p, grad_half)
        grad_z = p * grad_half - dot * p

        # Adam update
        m_adam = beta1 * m_adam + (1.0 - beta1) * grad_z
        v_adam = beta2 * v_adam + (1.0 - beta2) * (grad_z ** 2)
        m_hat = m_adam / (1.0 - beta1 ** it)
        v_hat = v_adam / (1.0 - beta2 ** it)
        z -= lr * m_hat / (np.sqrt(v_hat) + eps_adam)

        # occasional exact evaluation
        if it % report_interval == 0:
            p = softmax(z)
            a_full = np.concatenate([p, p[::-1]]) * 0.5
            sc = compute_score(a_full)
            if sc < best_sc:
                best_sc = sc
                best_a = a_full.copy()
                print(f"[{time.time():.1f}] Adam improved to {best_sc:.6f}")

    # final evaluation
    p = softmax(z)
    a_full = np.concatenate([p, p[::-1]]) * 0.5
    sc = compute_score(a_full)
    if sc < best_sc:
        best_sc = sc
        best_a = a_full.copy()
    return best_a, best_sc


# ----------------------------------------------------------------------
# Projected sub‑gradient descent on the true max‑objective
# ----------------------------------------------------------------------


def subgrad_descent(start_full: np.ndarray,
                    deadline: float,
                    step0: float = 0.07,
                    report_interval: int = 5000) -> Tuple[np.ndarray, float]:
    """
    Classical sub‑gradient method for 2·n·max(conv).
    The iterate is always projected onto the simplex by simple
    clamp‑and‑renorm, which is much faster than the exact Euclidean
    projection.
    """
    n = start_full.size
    a = start_full.copy()
    # ensure feasibility
    a = simple_simplex_proj(a)

    best_a = a.copy()
    best_sc = compute_score(a)

    t = 1
    while time.time() < deadline:
        conv = convolution_fft(a, a)
        k = int(np.argmax(conv))

        # sub‑gradient of max‑conv at index k
        idx = np.arange(n)
        j = k - idx
        mask = (j >= 0) & (j < n)
        g = np.zeros_like(a)
        g[mask] = 2.0 * a[j[mask]]

        step = step0 / np.sqrt(t)
        a = a - step * g
        a = simple_simplex_proj(a)

        if t % report_interval == 0:
            sc = compute_score(a)
            if sc < best_sc:
                best_sc = sc
                best_a = a.copy()
                print(f"[{time.time():.1f}] Sub‑grad improved to {best_sc:.6f}")
        t += 1

    # final check
    sc = compute_score(a)
    if sc < best_sc:
        best_sc = sc
        best_a = a.copy()
    return best_a, best_sc


# ----------------------------------------------------------------------
# Tiny simulated‑annealing hill‑climb (optional)
# ----------------------------------------------------------------------


def hill_climb_sa(start_full: np.ndarray,
                  deadline: float,
                  max_steps: int = 800_000) -> Tuple[np.ndarray, float]:
    """
    Very light SA that swaps or transfers a fraction of mass.
    Used only if time remains after the two main phases.
    """
    n = start_full.size
    cur = start_full.copy()
    cur_sc = compute_score(cur)
    best = cur.copy()
    best_sc = cur_sc

    # annealing schedule
    T0 = cur_sc * 5e-4
    T = T0
    step_scale = 0.12

    for step in range(1, max_steps + 1):
        if time.time() >= deadline:
            break
        cand = cur.copy()
        if np.random.rand() < 0.5:      # swap two entries
            i, j = np.random.randint(n, size=2)
            cand[i], cand[j] = cand[j], cand[i]
        else:                           # transfer a fraction of mass
            i, j = np.random.randint(n, size=2)
            if i != j:
                delta = step_scale * cand[i]
                cand[i] -= delta
                cand[j] += delta
                if cand[i] < 0.0:
                    cand[i] = 0.0

        cand = simple_simplex_proj(cand)
        cand_sc = compute_score(cand)

        if cand_sc < cur_sc or np.random.rand() < np.exp(-(cand_sc - cur_sc) / (T + 1e-18)):
            cur, cur_sc = cand, cand_sc
            if cur_sc < best_sc:
                best_sc = cur_sc
                best = cur.copy()
                step_scale = min(step_scale * 1.07, 0.5)
        else:
            step_scale *= 0.97

        # linear cooling
        T = T0 * (1.0 - step / max_steps)

    return best, best_sc


# ----------------------------------------------------------------------
# Main driver
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 12345,
                     budget_s: int = 1000,
                     **kwargs) -> List[float]:
    """
    Search for a non‑negative step‑function with minimal
    ``evaluate_sequence`` score.  Returns a plain Python list.
    """
    np.random.seed(seed)
    start_time = time.time()
    # safety margin (a few seconds for the final call to evaluate_sequence)
    deadline = start_time + budget_s - 5.0

    best_seq: List[float] | None = None
    best_score = float("inf")

    # -----------------------------------------------------------
    # 0. optional fallback from previous run
    # -----------------------------------------------------------
    try:
        fb = [float(x) for x in height_sequence_1]    # type: ignore
        fb_sc = evaluate_sequence(fb)
        print(f"[{time.time():.1f}] fallback score = {fb_sc:.6f}")
        if fb_sc < best_score:
            best_score = fb_sc
            best_seq = fb
    except Exception:
        pass

    # -----------------------------------------------------------
    # 1. Analytic families – coarse scan + fine α optimisation
    # -----------------------------------------------------------
    n = 2048
    lengths = [n]

    for nlen in lengths:
        if time.time() >= deadline - 200:
            break
        print(f"[{time.time():.1f}] Analytic scan for n = {nlen}")

        # --- coarse grids -------------------------------------------------
        alphas = np.linspace(0.0, 12.0, 61)
        for alpha in alphas:
            vec = symmetric_product_family(nlen, alpha)
            sc = compute_score(vec)
            if sc < best_score:
                best_score = sc
                best_seq = vec.tolist()
                print(f"[{time.time():.1f}]  new best (sym‑α={alpha:.2f}) = {best_score:.6f}")

        betas = np.linspace(0.0, 12.0, 61)
        for beta in betas:
            vec = left_heavy_power_family(nlen, beta)
            sc = compute_score(vec)
            if sc < best_score:
                best_score = sc
                best_seq = vec.tolist()
                print(f"[{time.time():.1f}]  new best (left‑β={beta:.2f}) = {best_score:.6f}")

        gammas = np.linspace(0.0, 6.0, 31)
        for gamma in gammas:
            vec = exp_u_family(nlen, gamma)
            sc = compute_score(vec)
            if sc < best_score:
                best_score = sc
                best_seq = vec.tolist()
                print(f"[{time.time():.1f}]  new best (exp‑γ={gamma:.2f}) = {best_score:.6f}")

        # --- refined α search for the symmetric product family ----------
        if time.time() < deadline - 50:
            print(f"[{time.time():.1f}] Refining α with golden‑section")
            alpha_opt, vec_opt = best_alpha_symmetric_product(
                nlen, lo=0.0, hi=12.0, max_iter=25)
            sc = compute_score(vec_opt)
            if sc < best_score:
                best_score = sc
                best_seq = vec_opt.tolist()
                print(f"[{time.time():.1f}]  refined α={alpha_opt:.6f} gives {best_score:.6f}")

    # -----------------------------------------------------------
    # 2. Symmetric Adam refinement
    # -----------------------------------------------------------
    if best_seq is not None and time.time() < deadline - 250:
        print(f"[{time.time():.1f}] Starting Adam refinement (symmetric)")
        a_start = np.array(best_seq, dtype=np.float64)
        # enforce exact symmetry and unit sum
        a_start = (a_start + a_start[::-1]) * 0.5
        a_start = simple_simplex_proj(a_start)
        half_len = n // 2
        start_half = a_start[:half_len]          # first half (sum = 0.5)
        # because the full vector will be 0.5·[p,p[::-1]], we need half‑sum = 1
        start_half = start_half * 2.0            # now sum(start_half) = 1
        a_opt, sc_opt = adam_symmetric_optimize(
            start_half,
            deadline - 150,
            max_iters=300_000,
            beta_start=5.0,
            beta_end=20000.0,
            lr=0.018,
            report_interval=4000,
        )
        if sc_opt < best_score:
            best_score = sc_opt
            best_seq = a_opt.tolist()
            print(f"[{time.time():.1f}] After Adam best = {best_score:.6f}")

    # -----------------------------------------------------------
    # 3. Projected sub‑gradient descent on true max
    # -----------------------------------------------------------
    if best_seq is not None and time.time() < deadline - 70:
        print(f"[{time.time():.1f}] Starting long sub‑gradient descent")
        a_sub, sc_sub = subgrad_descent(
            np.array(best_seq, dtype=np.float64),
            deadline - 30,
            step0=0.07,
            report_interval=5000,
        )
        if sc_sub < best_score:
            best_score = sc_sub
            best_seq = a_sub.tolist()
            print(f"[{time.time():.1f}] After sub‑grad best = {best_score:.6f}")

    # -----------------------------------------------------------
    # 4. Final small SA hill‑climb (optional)
    # -----------------------------------------------------------
    if best_seq is not None and time.time() < deadline - 5:
        print(f"[{time.time():.1f}] Final simulated‑annealing hill‑climb")
        a_hc, sc_hc = hill_climb_sa(
            np.array(best_seq, dtype=np.float64),
            deadline,
            max_steps=800_000,
        )
        if sc_hc < best_score:
            best_score = sc_hc
            best_seq = a_hc.tolist()
            print(f"[{time.time():.1f}] Hill‑climb improved best = {best_score:.6f}")

    # -----------------------------------------------------------
    # 5. Safeguard fallback (very small uniform vector)
    # -----------------------------------------------------------
    if best_seq is None:
        n_small = 256
        best_seq = [1.0 / n_small] * n_small
        best_score = evaluate_sequence(best_seq)

    # final validation with the provided evaluator
    final_score = evaluate_sequence(best_seq)
    print(f"\n[{time.time():.1f}] finished search – final reported score = {final_score:.6f}")

    return best_seq
```
