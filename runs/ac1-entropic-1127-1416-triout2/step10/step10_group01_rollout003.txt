"""
Propose a non‑negative step‑function (sequence) that minimises

        2·n·max(convolve(a,a)) / (sum(a)**2)

under the simplex constraint.

Algorithm
---------
0.  If a fallback ``height_sequence_1`` is provided, keep it as an initial
    candidate.

1.  Quick analytic / random scan (few seconds).  All candidates are
    normalised to sum = 1 and the best few (up to 5) are kept.

2.  Adam optimisation on a smooth‐max surrogate
        fβ(a) = (1/β)·log Σ exp(β·(a*a)_k)
    starting from the best seed from step 1.  The smoothing parameter β
    is increased exponentially up to a large value, making the surrogate
    approach the true max.  This gives a good “local optimum”.

3.  Projected sub‑gradient descent (very cheap per iteration) for a
    longer period – it works directly on the true max‑objective.

4.  Repeated LP‑based refinement (the classic state‑of‑the‑art step):
    given the current vector f, solve
        max Σ g_j   s.t.   (f * g)_k ≤ M   for all k,
    where M = max(f*f).  The LP solution g is normalised to a probability
    vector ĝ.  If ĝ already improves the score it is taken; otherwise a
    convex mixture (1‑t)·f + t·ĝ (t≈0.2) is projected back onto the simplex.
    The loop runs as long as time permits (up to a few hundred iterations).

5.  Symmetrisation: the sequence is replaced by (a+a[::-1])/2 whenever
    this reduces the score.

6.  Final stochastic hill‑climb (simulated‑annealing style) for the
    remaining budget.

All phases respect the global wall‑clock budget (default 1000 s).  The
function returns a plain Python list of floats ready for ``evaluate_sequence``.
"""

import time
import math
import numpy as np
from scipy.optimize import linprog

# ----------------------------------------------------------------------
# Basic utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full (aperiodic) convolution via real FFT."""
    m = a.size + b.size - 1
    nfft = next_pow2(m)
    A = np.fft.rfft(a, n=nfft)
    B = np.fft.rfft(b, n=nfft)
    C = A * B
    conv = np.fft.irfft(C, n=nfft)[:m]
    return conv


def compute_score(a: np.ndarray) -> float:
    """Exact objective assuming ``a`` is normalised (sum = 1)."""
    n = a.shape[0]
    conv = convolution_fft(a, a)
    return 2.0 * n * float(conv.max())


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of ``v`` onto the probability simplex
    (non‑negative entries that sum to 1).
    """
    n = v.shape[0]
    # if all entries are non‑positive, return the uniform vector
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / n)

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.where(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def softmax(z: np.ndarray) -> np.ndarray:
    """Numerically stable soft‑max."""
    z_max = np.max(z)
    e = np.exp(z - z_max)
    return e / e.sum()


# ----------------------------------------------------------------------
# Analytic families (used for the quick initial scan)
# ----------------------------------------------------------------------


def symmetric_product_family(n: int, alpha: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    logprod = np.log(i + 0.5) + np.log(n - 0.5 - i)
    w = np.exp(-alpha * logprod)
    a = w / w.sum()
    return a


def left_heavy_power_family(n: int, beta: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    w = np.exp(-beta * np.log(i + 0.5))
    a = w / w.sum()
    return a


def exp_u_family(n: int, gamma: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    dist = np.minimum(i, n - 1 - i)
    w = np.exp(-gamma * dist)
    a = w / w.sum()
    return a


def mixed_uniform_u_family(n: int, lam: float, gamma: float) -> np.ndarray:
    uniform = np.full(n, 1.0 / n, dtype=np.float64)
    u_shape = exp_u_family(n, gamma)
    a = lam * uniform + (1.0 - lam) * u_shape
    a /= a.sum()
    return a


def random_monotone_start(n: int) -> np.ndarray:
    r = np.random.rand(n) ** np.random.uniform(0.5, 5.0)
    r.sort()
    a = r[::-1]
    a /= a.sum()
    return a


def random_u_start(n: int) -> np.ndarray:
    gamma = np.random.uniform(0.0, 6.0)
    return exp_u_family(n, gamma)


def gaussian_family(n: int, sigma: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    mu = (n - 1) / 2.0
    w = np.exp(-0.5 * ((i - mu) / sigma) ** 2)
    a = w / w.sum()
    return a


# ----------------------------------------------------------------------
# Smoothed‑max optimisation (Adam on the soft‑max parametrisation)
# ----------------------------------------------------------------------


def adam_smooth_optimize(start_a: np.ndarray,
                         deadline: float,
                         max_iters: int = 200_000,
                         beta_start: float = 4.0,
                         beta_end: float = 800.0,
                         lr: float = 0.018) -> tuple[np.ndarray, float]:
    """
    Minimise the smooth surrogate
          f_β(a) = (1/β)·log Σ_k exp(β·(a*a)_k)
    with Adam on the soft‑max parametrisation a = softmax(z).
    """
    eps = 1e-12
    z = np.log(start_a + eps)

    m = np.zeros_like(z)
    v = np.zeros_like(z)
    beta1 = 0.9
    beta2 = 0.999
    eps_adam = 1e-12

    best_a = softmax(z)
    best_score = compute_score(best_a)

    for it in range(1, max_iters + 1):
        if time.time() >= deadline:
            break

        # schedule for β
        t = it / max_iters
        beta = beta_start * (beta_end / beta_start) ** t

        a = softmax(z)
        conv = convolution_fft(a, a)
        maxc = conv.max()
        exp_w = np.exp(beta * (conv - maxc))
        w = exp_w / exp_w.sum()

        # gradient wrt a (derived in the original script)
        a_rev = a[::-1]
        conv_rev = convolution_fft(a_rev, w)
        grad_a = 2.0 * conv_rev[len(a) - 1:len(a) - 1 + len(a)]

        # back‑prop through soft‑max
        dot_ag = np.dot(a, grad_a)
        grad_z = a * grad_a - dot_ag * a

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad_z
        v = beta2 * v + (1.0 - beta2) * (grad_z ** 2)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        z -= lr * m_hat / (np.sqrt(v_hat) + eps_adam)

        if it % 400 == 0:
            a_tmp = softmax(z)
            sc = compute_score(a_tmp)
            if sc < best_score:
                best_score = sc
                best_a = a_tmp.copy()
                print(f"[{time.time():.1f}] Adam improved to {best_score:.6f}")

    # final check
    a_fin = softmax(z)
    sc_fin = compute_score(a_fin)
    if sc_fin < best_score:
        best_score = sc_fin
        best_a = a_fin.copy()
    return best_a, best_score

# Projected sub‑gradient descent (cheapest iteration)
# ----------------------------------------------------------------------


def subgrad_optimize(start_a: np.ndarray,
                     deadline: float,
                     max_iters: int = 3_000_000,
                     step0: float = 0.07) -> tuple[np.ndarray, float]:
    """
    Sub‑gradient descent on the exact max‑objective.
    Stepsize  = step0 / sqrt(t) .
    """
    n = start_a.shape[0]
    a = start_a.astype(np.float64).copy()

    best_a = a.copy()
    best_score = compute_score(a)

    for t in range(1, max_iters + 1):
        if time.time() >= deadline:
            break

        conv = convolution_fft(a, a)
        k = int(np.argmax(conv))

        idx = np.arange(n)
        j = k - idx
        mask = (j >= 0) & (j < n)
        g = np.zeros_like(a)
        g[mask] = 2.0 * a[j[mask]]
        diag = (j == idx) & mask
        if np.any(diag):
            g[diag] = a[diag]

        step = step0 / np.sqrt(t)
        a = project_to_simplex(a - step * g)

        if t % 2000 == 0:
            sc = compute_score(a)
            if sc < best_score:
                best_score = sc
                best_a = a.copy()
                print(f"[{time.time():.1f}] sub‑grad improved to {best_score:.6f}")

    sc = compute_score(a)
    if sc < best_score:
        best_score = sc
        best_a = a.copy()
    return best_a, best_score

def hill_climb_sa(start_a: np.ndarray,
                  deadline: float,
                  max_steps: int = 1_500_000) -> tuple[np.ndarray, float]:
    """
    Simple SA on the simplex.
    """
    n = start_a.size
    cur_a = start_a.copy()
    cur_score = compute_score(cur_a)

    best_a = cur_a.copy()
    best_score = cur_score

    T0 = best_score * 1e-4
    T = T0
    step_scale = 0.12

    for step in range(1, max_steps + 1):
        if time.time() >= deadline:
            break

        cand = cur_a.copy()
        if np.random.rand() < 0.5:               # swap two entries
            i, j = np.random.randint(n, size=2)
            cand[i], cand[j] = cand[j], cand[i]
        else:                                     # mass transfer
            i, j = np.random.randint(n, size=2)
            if i != j:
                delta = step_scale * cur_a[i]
                cand[i] -= delta
                cand[j] += delta
                if cand[i] < 0.0:
                    cand[i] = 0.0

        cand = project_to_simplex(cand)
        cand_score = compute_score(cand)

        delta = cand_score - cur_score
        if delta < 0 or np.random.rand() < np.exp(-delta / (T + 1e-18)):
            cur_a = cand
            cur_score = cand_score
            if cur_score < best_score:
                best_score = cur_score
                best_a = cur_a.copy()
                step_scale = min(step_scale * 1.07, 0.5)
        else:
            step_scale *= 0.97

        T = T0 * (1.0 - step / max_steps)

    return best_a, best_score


# ----------------------------------------------------------------------
# LP‑based refinement – core solver
# ----------------------------------------------------------------------


def lp_solve_g(f: np.ndarray, M: float) -> np.ndarray | None:
    """
    Solve   max Σ g_j   s.t.   (f * g)_k ≤ M  (all k),
    g ≥ 0.
    Returns the raw optimiser vector g (not normalised), or None on failure.
    """
    n = f.shape[0]
    m = 2 * n - 1

    # Build convolution matrix A:   (f * g) = A @ g
    A = np.zeros((m, n), dtype=np.float64)
    for i in range(n):
        A[i:i + n, i] = f[i]

    c = -np.ones(n, dtype=np.float64)          # maximise Σ g  → minimise −Σ g
    bounds = [(0.0, None) for _ in range(n)]
    b = np.full(m, M, dtype=np.float64)

    res = linprog(c, A_ub=A, b_ub=b, bounds=bounds,
                  method='highs',
                  options={'presolve': True, 'time_limit': 15})

    if not res.success:
        return None

    g = np.maximum(res.x, 0.0)
    if g.sum() <= 0.0:
        return None
    return g


def lp_refine_loop(a: np.ndarray, deadline: float) -> tuple[np.ndarray, float]:
    """
    Repeated LP refinement (step 5 of the description).
    Returns the best vector found and its score.
    """
    best_a = a.copy()
    best_score = compute_score(best_a)
    mix_weight = 0.2
    it = 0

    while time.time() < deadline - 5.0:
        conv = convolution_fft(best_a, best_a)
        M = float(conv.max())
        g = lp_solve_g(best_a, M)
        if g is None:
            break

        g_norm = g / g.sum()
        sc_g = compute_score(g_norm)

        # direct replacement if it already improves
        if sc_g < best_score - 1e-9:
            best_score = sc_g
            best_a = g_norm.copy()
            print(f"[{time.time():.1f}] LP direct replace iter {it} score {best_score:.6f}")
            it += 1
            continue

        # otherwise try a convex mixture
        a_mix = (1.0 - mix_weight) * best_a + mix_weight * g_norm
        a_mix = project_to_simplex(a_mix)
        sc_mix = compute_score(a_mix)

        if sc_mix < best_score - 1e-9:
            best_score = sc_mix
            best_a = a_mix.copy()
            print(f"[{time.time():.1f}] LP mix iter {it} weight {mix_weight:.3f} score {best_score:.6f}")
        else:
            # no improvement – become more conservative
            mix_weight = max(0.05, mix_weight * 0.9)

        it += 1
        if it >= 250:          # safety cap
            break

    return best_a, best_score

def propose_candidate(seed: int = 12345, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step‑function that minimises the evaluation
    function.  The function respects a global wall‑clock budget (default 1000 s)
    and returns a plain Python list of floats.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 2.0   # safety margin

    best_seq = None
    best_score = float("inf")

    # ------------------------------------------------------------------
    # 0.  Fallback candidate (if supplied by the harness)
    # ------------------------------------------------------------------
    try:
        fb_vec = [float(x) for x in height_sequence_1]   # type: ignore
        fb = evaluate_sequence(fb_vec)
        print(f"[{time.time():.1f}] fallback score = {fb:.6f}")
        if fb < best_score:
            best_score = fb
            best_seq = fb_vec
    except Exception:
        pass

    # ------------------------------------------------------------------
    # 1. Determine problem size
    # ------------------------------------------------------------------
    n = len(best_seq) if best_seq is not None else 2048

    # ------------------------------------------------------------------
    # 2. Quick analytic / random scan (few seconds)
    # ------------------------------------------------------------------
    candidates = []   # stores tuples (score, vector)

    def consider(vec):
        vec = np.maximum(vec, 0.0)
        vec = vec / vec.sum()
        sc = compute_score(vec)
        candidates.append((sc, vec))
        return sc

    if time.time() < deadline - 250.0:
        print(f"[{time.time():.1f}] quick scan on n={n}")

        alphas = np.linspace(0.0, 12.0, 13)
        betas = np.linspace(0.0, 12.0, 13)
        gammas = np.linspace(0.0, 6.0, 13)
        sigmas = np.linspace(0.5, n / 4.0, 10)

        for alpha in alphas:
            if time.time() >= deadline - 30.0:
                break
            vec = symmetric_product_family(n, alpha)
            sc = consider(vec)
            if sc < best_score:
                best_score = sc
                best_seq = vec.tolist()
                print(f"[{time.time():.1f}] new best (sym‑α={alpha:.2f}) = {best_score:.6f}")

        for beta in betas:
            if time.time() >= deadline - 30.0:
                break
            vec = left_heavy_power_family(n, beta)
            sc = consider(vec)
            if sc < best_score:
                best_score = sc
                best_seq = vec.tolist()
                print(f"[{time.time():.1f}] new best (left‑β={beta:.2f}) = {best_score:.6f}")

        for gamma in gammas:
            if time.time() >= deadline - 30.0:
                break
            vec = exp_u_family(n, gamma)
            sc = consider(vec)
            if sc < best_score:
                best_score = sc
                best_seq = vec.tolist()
                print(f"[{time.time():.1f}] new best (exp‑γ={gamma:.2f}) = {best_score:.6f}")

        for sigma in sigmas:
            if time.time() >= deadline - 30.0:
                break
            vec = gaussian_family(n, sigma)
            sc = consider(vec)
            if sc < best_score:
                best_score = sc
                best_seq = vec.tolist()
                print(f"[{time.time():.1f}] new best (gauss‑σ={sigma:.2f}) = {best_score:.6f}")

        # random draws
        for _ in range(200):
            if time.time() >= deadline - 30.0:
                break
            if np.random.rand() < 0.5:
                vec = random_monotone_start(n)
            else:
                vec = random_u_start(n)
            sc = consider(vec)
            if sc < best_score:
                best_score = sc
                best_seq = vec.tolist()
                print(f"[{time.time():.1f}] new best (rand) = {best_score:.6f}")

    # keep only the few best candidates for the next phase
    candidates.sort(key=lambda x: x[0])
    top_candidates = [vec for _, vec in candidates[:5]]
    if not top_candidates:
        # fallback to uniform if nothing was generated
        top_candidates = [np.full(n, 1.0 / n, dtype=np.float64)]

    # ------------------------------------------------------------------
    # 3. Adam‑based smooth‑max refinement (multiple restarts if time permits)
    # ------------------------------------------------------------------
    a = top_candidates[0].copy()
    best_score = compute_score(a)
    if time.time() < deadline - 180.0:
        print(f"[{time.time():.1f}] starting Adam refinement")
        a_opt, sc_opt = adam_smooth_optimize(
            a,
            deadline - 60.0,
            max_iters=200_000,
            beta_start=4.0,
            beta_end=800.0,
            lr=0.018,
        )
        if sc_opt < best_score:
            best_score = sc_opt
            a = a_opt
            print(f"[{time.time():.1f}] after Adam best = {best_score:.6f}")

        # Optional second restart on the second‑best seed
        if len(top_candidates) > 1 and time.time() < deadline - 120.0:
            a2 = top_candidates[1].copy()
            a2_opt, sc2_opt = adam_smooth_optimize(
                a2,
                deadline - 30.0,
                max_iters=120_000,
                beta_start=4.0,
                beta_end=800.0,
                lr=0.018,
            )
            if sc2_opt < best_score:
                best_score = sc2_opt
                a = a2_opt
                print(f"[{time.time():.1f}] after second Adam best = {best_score:.6f}")

    # ------------------------------------------------------------------
    # 4. Long projected sub‑gradient descent
    # ------------------------------------------------------------------
    if time.time() < deadline - 120.0:
        print(f"[{time.time():.1f}] starting sub‑gradient refinement")
        a_opt, sc_opt = subgrad_optimize(
            a,
            deadline - 30.0,
            max_iters=4_000_000,
            step0=0.07,
        )
        if sc_opt < best_score:
            best_score = sc_opt
            a = a_opt
            print(f"[{time.time():.1f}] after sub‑grad best = {best_score:.6f}")

    # ------------------------------------------------------------------
    # 5. LP‑based refinement (as many iterations as time allows)
    # ------------------------------------------------------------------
    if time.time() < deadline - 35.0:
        print(f"[{time.time():.1f}] starting LP refinement loop")
        a, best_score = lp_refine_loop(a, deadline - 5.0)
        print(f"[{time.time():.1f}] LP loop finished – best now {best_score:.6f}")

    # ------------------------------------------------------------------
    # 6. Symmetrisation test
    # ------------------------------------------------------------------
    a_sym = (a + a[::-1]) * 0.5
    a_sym = project_to_simplex(a_sym)
    sc_sym = compute_score(a_sym)
    if sc_sym < best_score:
        best_score = sc_sym
        a = a_sym
        print(f"[{time.time():.1f}] symmetrisation improved best = {best_score:.6f}")

    # ------------------------------------------------------------------
    # 7. Final stochastic hill‑climb (SA)
    # ------------------------------------------------------------------
    if time.time() < deadline - 5.0:
        print(f"[{time.time():.1f}] final hill‑climb (SA)")
        a_opt, sc_opt = hill_climb_sa(a, deadline, max_steps=1_200_000)
        if sc_opt < best_score:
            best_score = sc_opt
            a = a_opt
            print(f"[{time.time():.1f}] hill‑climb improved best = {best_score:.6f}")

    # ------------------------------------------------------------------
    # 8. Final official evaluation and return
    # ------------------------------------------------------------------
    final_seq = a.tolist()
    final_score = evaluate_sequence(final_seq)
    print(f"\n[{time.time():.1f}] finished search – final reported score = {final_score:.6f}")
    return final_seq
