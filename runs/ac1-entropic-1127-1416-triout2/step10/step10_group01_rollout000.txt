def mirror_descent(start_a, deadline, max_iters=2_000_000, eta0=0.1):
    a = start_a.copy()
    best_score = compute_score(a)
    best_a = a.copy()
    for t in range(1, max_iters+1):
        if time.time() >= deadline: break
        conv = convolution_fft(a, a)
        k = int(np.argmax(conv))
        # compute subgradient g as before
        n = a.shape[0]
        idx = np.arange(n)
        j = k - idx
        mask = (j >= 0) & (j < n)
        g = np.zeros_like(a)
        g[mask] = 2.0 * a[j[mask]]
        diag = (j == idx) & mask
        if np.any(diag):
            g[diag] = a[diag]
        # step
        eta = eta0 / np.sqrt(t)
        a = a * np.exp(-eta * g)
        a_sum = a.sum()
        if a_sum == 0:
            a = np.full_like(a, 1.0 / n)
        else:
            a = a / a_sum
        if t % 2000 == 0:
            sc = compute_score(a)
            if sc < best_score:
                best_score = sc
                best_a = a.copy()
                print(f"[{time.time():.1f}] Mirror improved to {best_score:.6f}")
    return best_a, best_score

"""
Propose a step‑function (non‑negative sequence) that minimises

        2·n·max(convolve(a,a)) / (sum(a)**2)

under the simplex constraint.

Algorithm (≈ 1000 s budget)
---------------------------------
0.  Load a fallback candidate if supplied.
1.  Choose a problem size (default 2048, also try 1024 and 4096).
2.  For each size perform a fast scan of several analytic families
    (including new cosine‑type families) to obtain a good starter.
3.  Refine the starter with three complementary local‑optimisers:
    * Adam on a smooth‑max surrogate (log‑sum‑exp) with a long β‑schedule.
    * Mirror‑descent (exponentiated‑gradient) on the true max‑objective.
    * Projected sub‑gradient descent (as in the reference code).
4.  Run a longer LP‑based refinement loop (state‑of‑the‑art iteration);
    the loop continues as long as time permits.
5.  Symmetrise the vector (average with its reverse) if it improves the score.
6.  Finish with a stochastic hill‑climb (simulated annealing) for the
    remaining budget.
7.  Return the best sequence found (as a plain Python list of floats).

All steps respect the global wall‑clock budget, and intermediate progress
is printed for diagnostics.
"""

def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full aperiodic convolution via real FFT."""
    m = a.size + b.size - 1
    nfft = next_pow2(m)
    A = np.fft.rfft(a, n=nfft)
    B = np.fft.rfft(b, n=nfft)
    C = A * B
    conv = np.fft.irfft(C, n=nfft)[:m]
    return conv


def compute_score(a: np.ndarray) -> float:
    """
    Exact objective for a simplex‑normalised vector (∑a = 1):
        2·n·max(conv(a,a)).
    """
    n = a.shape[0]
    conv = convolution_fft(a, a)
    return 2.0 * n * float(conv.max())


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / v.shape[0])

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, v.shape[0] + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.where(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def softmax(z: np.ndarray) -> np.ndarray:
    """Numerically stable soft‑max."""
    z_max = np.max(z)
    e = np.exp(z - z_max)
    return e / e.sum()

def symmetric_product_family(n: int, alpha: float) -> np.ndarray:
    """U‑shape: w_i ∝ ((i+0.5)*(n-0.5-i))^{-α}."""
    i = np.arange(n, dtype=np.float64)
    logprod = np.log(i + 0.5) + np.log(n - 0.5 - i)
    w = np.exp(-alpha * logprod)
    a = w / w.sum()
    return a


def left_heavy_power_family(n: int, beta: float) -> np.ndarray:
    """Monotonically decreasing: w_i ∝ (i+0.5)^(-β)."""
    i = np.arange(n, dtype=np.float64)
    w = np.exp(-beta * np.log(i + 0.5))
    a = w / w.sum()
    return a


def exp_u_family(n: int, gamma: float) -> np.ndarray:
    """Symmetric exponential decay from the centre."""
    i = np.arange(n, dtype=np.float64)
    dist = np.minimum(i, n - 1 - i)
    w = np.exp(-gamma * dist)
    a = w / w.sum()
    return a


def mixed_uniform_u_family(n: int, lam: float, gamma: float) -> np.ndarray:
    """Mixture of uniform and exponential‑U."""
    uniform = np.full(n, 1.0 / n, dtype=np.float64)
    u_shape = exp_u_family(n, gamma)
    a = lam * uniform + (1.0 - lam) * u_shape
    return a / a.sum()


def cosine_family(n: int, p: float) -> np.ndarray:
    """
    Cosine‑U shape: w_i ∝ (1 + cos(π·i/(n‑1)))^p,
    giving high mass at the ends and low at the centre.
    """
    i = np.arange(n, dtype=np.float64)
    w = (1.0 + np.cos(np.pi * i / (n - 1))) ** p
    a = w / w.sum()
    return a


def double_exp_family(n: int, gamma1: float, gamma2: float, w: float) -> np.ndarray:
    """
    Mixture of two exponentials with different decay rates.
    w∈[0,1] controls the mixture proportion.
    """
    i = np.arange(n, dtype=np.float64)
    d = np.minimum(i, n - 1 - i)
    w1 = np.exp(-gamma1 * d)
    w2 = np.exp(-gamma2 * d)
    a = w * w1 + (1.0 - w) * w2
    return a / a.sum()


def random_monotone_start(n: int) -> np.ndarray:
    """Random decreasing sequence (normalised)."""
    r = np.random.rand(n) ** np.random.uniform(0.5, 5.0)
    r.sort()
    a = r[::-1]
    a /= a.sum()
    return a


def random_u_start(n: int) -> np.ndarray:
    """Random symmetric exponential‑U."""
    gamma = np.random.uniform(0.0, 6.0)
    return exp_u_family(n, gamma)

def adam_smooth_optimize(start_a: np.ndarray,
                         deadline: float,
                         max_iters: int = 500_000,
                         beta_start: float = 4.0,
                         beta_end: float = 2000.0,
                         lr: float = 0.018) -> tuple[np.ndarray, float]:
    """
    Minimise the smooth surrogate
          f_β(a) = (1/β)·log Σ exp(β·(a*a)_k)
    using Adam on the soft‑max parametrisation a = softmax(z).
    """
    eps = 1e-12
    z = np.log(start_a + eps)

    m = np.zeros_like(z)
    v = np.zeros_like(z)
    beta1 = 0.9
    beta2 = 0.999
    eps_adam = 1e-12

    best_a = softmax(z)
    best_score = compute_score(best_a)

    for it in range(1, max_iters + 1):
        if time.time() >= deadline:
            break

        # Exponential schedule for β
        t = it / max_iters
        beta = beta_start * (beta_end / beta_start) ** t

        a = softmax(z)
        conv = convolution_fft(a, a)
        maxc = conv.max()
        exp_w = np.exp(beta * (conv - maxc))
        w = exp_w / exp_w.sum()

        # Gradient wrt a (derived in the reference implementation)
        a_rev = a[::-1]
        conv_rev = convolution_fft(a_rev, w)
        grad_a = 2.0 * conv_rev[len(a) - 1:len(a) - 1 + len(a)]

        # Back‑prop through soft‑max
        dot_ag = np.dot(a, grad_a)
        grad_z = a * grad_a - dot_ag * a

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad_z
        v = beta2 * v + (1.0 - beta2) * (grad_z ** 2)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        z -= lr * m_hat / (np.sqrt(v_hat) + eps_adam)

        if it % 500 == 0:
            a_tmp = softmax(z)
            sc = compute_score(a_tmp)
            if sc < best_score:
                best_score = sc
                best_a = a_tmp.copy()
                print(f"[{time.time():.1f}] Adam improved to {best_score:.6f}")

    # Final check
    a_fin = softmax(z)
    sc_fin = compute_score(a_fin)
    if sc_fin < best_score:
        best_score = sc_fin
        best_a = a_fin.copy()
    return best_a, best_score

def mirror_descent_optimize(start_a: np.ndarray,
                            deadline: float,
                            max_iters: int = 2_500_000,
                            eta0: float = 0.15) -> tuple[np.ndarray, float]:
    """
    Exponentiated‑gradient (mirror descent) on the max‑convolution objective.
    Updates:
        a_i ← a_i * exp(-η_t * g_i)   (g is a sub‑gradient of max(conv))
    followed by renormalisation onto the simplex.
    """
    a = np.clip(start_a, 1e-18, None)  # avoid zeros causing NaNs
    a = a / a.sum()

    best_a = a.copy()
    best_score = compute_score(a)
    n = a.shape[0]

    for t in range(1, max_iters + 1):
        if time.time() >= deadline:
            break

        conv = convolution_fft(a, a)
        k = int(np.argmax(conv))

        # Sub‑gradient of the max‑convolution entry w.r.t. a
        idx = np.arange(n)
        j = k - idx
        mask = (j >= 0) & (j < n)
        g = np.zeros_like(a)
        g[mask] = 2.0 * a[j[mask]]
        diag = (j == idx) & mask
        if np.any(diag):
            g[diag] = a[diag]

        eta = eta0 / np.sqrt(t)
        a = a * np.exp(-eta * g)
        s = a.sum()
        if s == 0.0:
            a = np.full_like(a, 1.0 / n)
        else:
            a = a / s

        if t % 2500 == 0:
            sc = compute_score(a)
            if sc < best_score:
                best_score = sc
                best_a = a.copy()
                print(f"[{time.time():.1f}] Mirror‑descent improved to {best_score:.6f}")

    return best_a, best_score

def subgrad_optimize(start_a: np.ndarray,
                     deadline: float,
                     max_iters: int = 2_500_000,
                     step0: float = 0.07) -> tuple[np.ndarray, float]:
    """
    Sub‑gradient method for the exact max‑objective with a 1/√t stepsize.
    """
    n = start_a.shape[0]
    a = start_a.astype(np.float64).copy()

    best_a = a.copy()
    best_score = compute_score(a)

    for t in range(1, max_iters + 1):
        if time.time() >= deadline:

        conv = convolution_fft(a, a)
        k = int(np.argmax(conv))

        idx = np.arange(n)
        j = k - idx
        mask = (j >= 0) & (j < n)
        g = np.zeros_like(a)
        g[mask] = 2.0 * a[j[mask]]

def hill_climb_sa(start_a: np.ndarray,
                  deadline: float,
                  max_steps: int = 1_800_000) -> tuple[np.ndarray, float]:
    """
    Random walk on the simplex with occasional uphill moves.
    """
    n = start_a.size
    cur_a = start_a.copy()
    cur_score = compute_score(cur_a)

    best_a = cur_a.copy()
    best_score = cur_score

    T0 = best_score * 1e-4
    T = T0
    step_scale = 0.12

    for step in range(1, max_steps + 1):
        if time.time() >= deadline:
            break

        cand = cur_a.copy()
        if np.random.rand() < 0.5:               # swap two entries
            i, j = np.random.randint(n, size=2)
            cand[i], cand[j] = cand[j], cand[i]
        else:                                     # transfer a fraction of mass
            i, j = np.random.randint(n, size=2)
            if i != j:
                delta = step_scale * cur_a[i]
                cand[i] -= delta
                cand[j] += delta
                if cand[i] < 0.0:
                    cand[i] = 0.0

        cand = project_to_simplex(cand)
        cand_score = compute_score(cand)

        delta = cand_score - cur_score
        if delta < 0 or np.random.rand() < np.exp(-delta / (T + 1e-18)):
            cur_a = cand
            cur_score = cand_score
            if cur_score < best_score:
                best_score = cur_score
                best_a = cur_a.copy()
                step_scale = min(step_scale * 1.07, 0.5)
        else:
            step_scale *= 0.97

        # linear cooling
        T = T0 * (1.0 - step / max_steps)

    return best_a, best_score

# Linear‑programming refinement (state‑of‑the‑art iteration)
# ----------------------------------------------------------------------


def lp_refine_step(f: np.ndarray, M: float, mix_weight: float = 0.2) -> np.ndarray:
    """
    Solve   max Σ g_j   s.t.   (f * g)_k ≤ M   for all k,
    g ≥ 0, then return the convex mixture
        f_new = (1‑t)·f + t·g_norm,
    where g_norm = g / Σ g.
    """
    n = f.shape[0]
    m = 2 * n - 1

    # Build convolution matrix A with shape (m, n)
    A = np.zeros((m, n), dtype=np.float64)
    for i in range(n):
        A[i:i + n, i] = f[i]

    c = -np.ones(n, dtype=np.float64)          # maximise Σ g  ⇔ minimise −Σ g
    bounds = [(0.0, None) for _ in range(n)]
    b = np.full(m, M, dtype=np.float64)

    # Solve with HiGHS (default).  Time limit is a safety net.
    res = linprog(c, A_ub=A, b_ub=b, bounds=bounds, method='highs',
                  options={'presolve': True, 'time_limit': 30})

    if not res.success:
        return f

    g = np.maximum(res.x, 0.0)
    sum_g = g.sum()
    if sum_g <= 0.0:
        return f

    g_norm = g / sum_g
    f_new = (1.0 - mix_weight) * f + mix_weight * g_norm
    f_new = project_to_simplex(f_new)
    return f_new

                M = float(conv.max())
                a_new = lp_refine_step(a, M, mix_weight=mix_weight)
                sc = compute_score(a_new)
                if sc < best_local_score - 1e-8:
                    best_local_score = sc
                    a = a_new
                    print(f"[{time.time():.1f}] LP step {lp_iters} improved to {best_local_score:.6f}")
                else:
                    mix_weight = max(0.05, mix_weight * 0.9)
                lp_iters += 1
                if lp_iters >= 25:      # safety cap
                    break

        # ------------------------------------------------------------------
        # 2g. Symmetrisation test
        # ------------------------------------------------------------------
        a_sym = (a + a[::-1]) * 0.5
        a_sym = project_to_simplex(a_sym)
        sc_sym = compute_score(a_sym)
        if sc_sym < best_local_score:
            best_local_score = sc_sym
            a = a_sym
            print(f"[{time.time():.1f}] Symmetrisation improved best = {best_local_score:.6f}")

        # ------------------------------------------------------------------
        # 2h. Final hill‑climb (simulated annealing)
        # ------------------------------------------------------------------
        if time.time() < deadline - 5.0:
            print(f"[{time.time():.1f}] Final hill‑climb (SA) (n={n})")
            a_opt, sc_opt = hill_climb_sa(a, deadline, max_steps=1_500_000)
            if sc_opt < best_local_score:
                best_local_score = sc_opt
                a = a_opt
                print(f"[{time.time():.1f}] Hill‑climb improved best = {best_local_score:.6f}")

        # ------------------------------------------------------------------
        # 2i. Record the best candidate of this size
        # ------------------------------------------------------------------
        final_seq = a.tolist()
        final_score = evaluate_sequence(final_seq)
        print(f"[{time.time():.1f}] Size {n} final reported score = {final_score:.6f}")

        if final_score < best_score:
            best_score = final_score
            best_seq = final_seq
            print(f"[{time.time():.1f}] *** New global best score = {best_score:.6f} ***")

    # ------------------------------------------------------------------
    # 3. Return the overall best sequence (fallback if nothing better)
    # ------------------------------------------------------------------
    if best_seq is None:
        # As a last resort return a simple U‑shape of length 2048
        n = 2048
        best_seq = exp_u_family(n, gamma=2.0).tolist()
        print(f"[{time.time():.1f}] No candidate found – using default U‑shape.")
    else:
        print(f"[{time.time():.1f}] Search finished – best score = {best_score:.6f}")

    return best_seq
