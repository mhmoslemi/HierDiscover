"""
Propose a step‑function sequence minimising the evaluation function.

Search strategy (all stages respect a global wall‑clock deadline, default 1000 s):
1. Choose a moderate length `n` (≈8 k) so that many FFTs still fit in time.
2. Scan several analytic families (U‑shaped, power‑law, exponential,
   symmetric & asymmetric β‑distributions, logistic, sine‑power) on a coarse
   grid and keep the best.
3. Add many random monotone / sparse starts.
4. Refine the current best with a smoothed max‑convolution surrogate
   `(1/β)·log∑exp(β·(a*a)_k)`.  The surrogate is minimised by L‑BFGS‑B on the
   log‑soft‑max parameters (`z`).  A schedule of β={200,800,2000,5000} drives
   the surrogate to the true max.
5. Perform a long mirror (exponentiated) sub‑gradient descent on the true
   objective `2·n·max(a*a)`.  The iterate stays on the simplex via an
   entropy projection.  After the main run several short restarts from a
   small random perturbation are executed to escape possible local minima.
6. Finish with a cheap simulated‑annealing hill‑climb that swaps or moves a
   fraction of mass.
7. Return the best sequence found (as a plain Python list of floats).

All vectors are projected onto the probability simplex, therefore
`evaluate_sequence` returns the same value as the internal `compute_score`
used throughout the search.
"""

import time
import math
import numpy as np

# optional scipy for L‑BFGS‑B
try:
    import scipy.optimize as _opt
    _HAS_SCIPY = True
except Exception:
    _HAS_SCIPY = False

# ----------------------------------------------------------------------
# FFT utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Linear (aperiodic) convolution via real FFT."""
    m = a.size + b.size - 1
    nfft = next_pow2(m)
    A = np.fft.rfft(a, n=nfft)
    B = np.fft.rfft(b, n=nfft)
    C = A * B
    conv = np.fft.irfft(C, n=nfft)[:m]
    return conv


def compute_score(a: np.ndarray) -> float:
    """
    Exact objective for a probability vector (∑a = 1):
        score = 2·n·max(a * a)   (a*a = convolution)
    This matches `evaluate_sequence` when the vector is normalised.
    """
    n = a.shape[0]
    conv = convolution_fft(a, a)
    return 2.0 * n * float(conv.max())


# ----------------------------------------------------------------------
# Simplex utilities
# ----------------------------------------------------------------------


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection onto the probability simplex."""
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / v.shape[0])

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, v.shape[0] + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.where(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def softmax(z: np.ndarray) -> np.ndarray:
    """Stable soft‑max maps ℝⁿ → simplex."""
    z_max = np.max(z)
    e = np.exp(z - z_max)
    return e / e.sum()


# ----------------------------------------------------------------------
# Analytic families (all return a probability vector)
# ----------------------------------------------------------------------


def symmetric_product_family(n: int, alpha: float) -> np.ndarray:
    """U‑shaped: a_i ∝ [(i+½)(n‑½‑i)]^(‑α)."""
    i = np.arange(n, dtype=np.float64)
    logprod = np.log(i + 0.5) + np.log(n - 0.5 - i)
    w = np.exp(-alpha * logprod)
    w[np.isinf(w)] = np.finfo(np.float64).max
    a = w / w.sum()
    return a


def left_heavy_power_family(n: int, beta: float) -> np.ndarray:
    """Monotone decreasing power law: a_i ∝ (i+½)^(‑β)."""
    i = np.arange(n, dtype=np.float64)
    w = np.exp(-beta * np.log(i + 0.5))
    a = w / w.sum()
    return a


def exponential_decay_family(n: int, gamma: float) -> np.ndarray:
    """Simple exponential decay: a_i ∝ exp(‑γ·i)."""
    i = np.arange(n, dtype=np.float64)
    w = np.exp(-gamma * i)
    a = w / w.sum()
    return a


def beta_distribution_family(n: int, alpha: float) -> np.ndarray:
    """Symmetric beta shape: a_i ∝ (t·(1‑t))^α with t = i/(n‑1)."""
    i = np.arange(n, dtype=np.float64)
    t = i / (n - 1)
    eps = 1e-12
    w = (t + eps) ** alpha * (1 - t + eps) ** alpha
    a = w / w.sum()
    return a


def beta_asymmetric_family(n: int, alpha: float, beta: float) -> np.ndarray:
    """Asymmetric beta: a_i ∝ t^{α‑1}(1‑t)^{β‑1}."""
    i = np.arange(n, dtype=np.float64)
    t = i / (n - 1)
    eps = 1e-12
    w = (t + eps) ** (alpha - 1) * (1 - t + eps) ** (beta - 1)
    a = w / w.sum()
    return a


def logistic_family(n: int, kappa: float, center: float = None) -> np.ndarray:
    """Decreasing logistic: a_i ∝ 1/(1+exp(kappa·(i‑center)))."""
    if center is None:
        center = (n - 1) / 2.0
    i = np.arange(n, dtype=np.float64)
    w = 1.0 / (1.0 + np.exp(kappa * (i - center)))
    a = w / w.sum()
    return a


def sine_power_family(n: int, p: float) -> np.ndarray:
    """Shape built from sin(π·(i+½)/n) raised to power p."""
    i = np.arange(n, dtype=np.float64)
    w = np.sin(np.pi * (i + 0.5) / n) ** p
    a = w / w.sum()
    return a

def random_sparse_start(n: int) -> np.ndarray:
    """Sparse probability vector (~5 % non‑zeros)."""
    sparsity = np.random.uniform(0.03, 0.07)
    k = max(2, int(round(sparsity * n)))
    idx = np.random.choice(n, size=k, replace=False)
    a = np.zeros(n, dtype=np.float64)
    a[idx] = 1.0 / k
    return project_to_simplex(a)


def random_monotone_start(n: int) -> np.ndarray:
    """Random decreasing vector obtained by sorting a power‑transformed uniform."""
    r = np.random.rand(n) ** np.random.uniform(0.5, 5.0)
    r.sort()
    a = r[::-1]               # decreasing
    a = a / a.sum()
    return a


# ----------------------------------------------------------------------
# Smooth surrogate optimisation (L‑BFGS‑B)
# ----------------------------------------------------------------------


def _smooth_obj_and_grad(z: np.ndarray, beta: float, n: int):
    """
    Returns (f, grad) for the log‑sum‑exp surrogate:
        f   = (1/β)·log∑_k exp[β·(a*a)_k]
        grad = ∂f/∂z   (z are the log‑soft‑max parameters)
    """
    a = softmax(z)                     # a ∈ simplex
    conv = convolution_fft(a, a)       # length 2n‑1
    maxc = conv.max()
    exp_terms = np.exp(beta * (conv - maxc))
    sum_exp = exp_terms.sum()
    f = (np.log(sum_exp) + beta * maxc) / beta

    # gradient w.r.t. a
    w = exp_terms / sum_exp
    a_rev = a[::-1]
    conv_rev = convolution_fft(a_rev, w)   # length 3n‑2
    grad_a = 2.0 * conv_rev[n - 1:2 * n - 1]

    # chain rule through soft‑max
    dot_ag = np.dot(a, grad_a)
    grad_z = a * (grad_a - dot_ag)
    return float(f), grad_z.astype(np.float64)


def l_bfgs_refine(start_a: np.ndarray, deadline: float,
                  beta_list=(200.0, 800.0, 2000.0, 5000.0),
                  max_iter_per_beta=1200) -> tuple[np.ndarray, float]:
    """
    Refine `start_a` using L‑BFGS‑B on the smoothed surrogate.
    Returns the best vector and its exact score.
    """
    if not _HAS_SCIPY:          # pragma: no cover
        return start_a, compute_score(start_a)

    best_a = start_a.copy()
    best_score = compute_score(best_a)
    eps = 1e-12
    z = np.log(best_a + eps)          # initialise log‑soft‑max parameters

    for beta in beta_list:
        if time.time() >= deadline - 5.0:
            break
        remaining = deadline - time.time()
        max_iter = min(max_iter_per_beta, int(remaining * 5))  # crude budget

        def func(z_vec):
            f, g = _smooth_obj_and_grad(z_vec, beta, best_a.shape[0])
            return f, g

        try:
            opt_res = _opt.fmin_l_bfgs_b(
                func,
                z,
                maxiter=max_iter,
                disp=False,
                factr=1e7,
                pgtol=1e-12,
            )
        except Exception as exc:  # pragma: no cover
            print(f"[{time.time():.1f}] L‑BFGS failed at β={beta}: {exc}")
            continue

        z = opt_res[0]
        a_opt = softmax(z)
        sc_opt = compute_score(a_opt)
        if sc_opt < best_score:
            best_score = sc_opt
            best_a = a_opt.copy()
            print(f"[{time.time():.1f}] L‑BFGS β={beta:.0f} improved to {best_score:.6f}")

    return best_a, best_score


# ----------------------------------------------------------------------
# Mirror (entropy) sub‑gradient descent on the true max‑objective
# ----------------------------------------------------------------------


def mirror_subgrad_optimize(start_a: np.ndarray, deadline: float,
                           step0: float = 0.8,
                           check_interval: int = 2000) -> tuple[np.ndarray, float]:
    """
    Exponentiated sub‑gradient descent on
        F(a) = 2·n·max(a*a)
    The iterate stays on the simplex via an entropy projection.
    """
    n = start_a.shape[0]
    a = start_a.astype(np.float64).copy()
    best_a = a.copy()
    best_score = compute_score(a)

    t = 0
    while time.time() < deadline:
        t += 1
        conv = convolution_fft(a, a)
        k = int(np.argmax(conv))

        # sub‑gradient of the active entry
        idx = np.arange(n)
        j = k - idx
        mask = (j >= 0) & (j < n)
        g = np.zeros_like(a)
        g[mask] = 2.0 * a[j[mask]]

        eta = step0 / math.sqrt(t)
        a = a * np.exp(-eta * g)            # multiplicative update
        s = a.sum()
        if s == 0.0:
            a = np.full_like(a, 1.0 / n)
        else:
            a = a / s

        if t % check_interval == 0:
            sc = compute_score(a)
            if sc < best_score:
                best_score = sc
                best_a = a.copy()
                print(f"[{time.time():.1f}] mirror sub‑grad iter {t} best = {best_score:.6f}")

    # final check
    sc = compute_score(a)
    if sc < best_score:
        best_score = sc
        best_a = a.copy()
    return best_a, best_score


# ----------------------------------------------------------------------
# Simple hill‑climb (tiny simulated annealing)
# ----------------------------------------------------------------------


def hill_climb_sa(start_a: np.ndarray, deadline: float,
                  max_steps: int = 800_000,
                  init_temp_factor: float = 5e-5) -> tuple[np.ndarray, float]:
    """
    Random walk on the simplex with a tiny temperature.
    - 50 % swaps two entries.
    - otherwise transfer a fraction of mass from one index to another.
    """
    n = start_a.size
    cur_a = start_a.copy()
    cur_score = compute_score(cur_a)

    best_a = cur_a.copy()
    best_score = cur_score

    T0 = best_score * init_temp_factor
    T = T0
    step_scale = 0.12

    for step in range(1, max_steps + 1):
        if time.time() >= deadline:
            break

        cand = cur_a.copy()
        if np.random.rand() < 0.5:
            i, j = np.random.randint(n, size=2)
            cand[i], cand[j] = cand[j], cand[i]
        else:
            i, j = np.random.randint(n, size=2)
            if i != j:
                delta = step_scale * cur_a[i]
                cand[i] -= delta
                cand[j] += delta
                if cand[i] < 0.0:
                    cand[i] = 0.0

        cand = cand / cand.sum()
        cand_score = compute_score(cand)
        delta = cand_score - cur_score

        if delta < 0.0 or np.random.rand() < np.exp(-delta / (T + 1e-18)):
            cur_a = cand
            cur_score = cand_score
            if cur_score < best_score:
                best_score = cur_score
                best_a = cur_a.copy()
                step_scale = min(step_scale * 1.07, 0.5)
        else:
            step_scale *= 0.97

        T = T0 * (1.0 - step / max_steps)

    return best_a, best_score


# ----------------------------------------------------------------------
# High‑level driver
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 12345, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative sequence minimising `evaluate_sequence`.
    Returns a plain Python list of floats.
    """
    np.random.seed(seed)
    start_time = time.time()
    # safety margin for the outer harness
    deadline = start_time + budget_s - 1.0

    # ------------------------------------------------------------------
    # 0. Choose length
    # ------------------------------------------------------------------
    n = 8192 if budget_s >= 850 else 4096
    print(f"[{time.time():.1f}] Using length n = {n}")

    best_seq = None
    best_score = float("inf")

    # ------------------------------------------------------------------
    # 1. optional fallback to a known construction
    # ------------------------------------------------------------------
    try:
        fb_vec = [float(x) for x in height_sequence_1]   # type: ignore
        fb = evaluate_sequence(fb_vec)
        print(f"[{time.time():.1f}] fallback score = {fb:.6f}")
        if fb < best_score:
            best_score = fb
            best_seq = fb_vec
    except Exception:
        pass

    # helper to test a candidate and keep the best
    def try_update(vec_np):
        nonlocal best_seq, best_score
        sc = compute_score(vec_np)
        if sc < best_score:
            best_score = sc
            best_seq = vec_np.tolist()
            print(f"[{time.time():.1f}] new best = {best_score:.6f}")

    # ------------------------------------------------------------------
    # 2. Scan analytic families (coarse grid)
    # ------------------------------------------------------------------
    # U‑shaped (symmetric product)
    for alpha in np.linspace(0.0, 12.0, 121):
        if time.time() >= deadline - 30.0:
            break
        try_update(symmetric_product_family(n, alpha))

    # left‑heavy power‑law
    for beta in np.linspace(0.0, 12.0, 121):
        if time.time() >= deadline - 30.0:
            break
        try_update(left_heavy_power_family(n, beta))

    # exponential decay
    for gamma in np.linspace(0.0, 10.0, 101):
        if time.time() >= deadline - 30.0:
            break
        try_update(exponential_decay_family(n, gamma))

    # symmetric beta distribution
    for alpha in np.linspace(0.0, 8.0, 81):
        if time.time() >= deadline - 30.0:
            break
        try_update(beta_distribution_family(n, alpha))

    # asymmetric beta
    alphas = np.linspace(0.1, 5.0, 21)
    betas = np.linspace(0.1, 5.0, 21)
    for a in alphas:
        for b in betas:
            if time.time() >= deadline - 30.0:
                break
            try_update(beta_asymmetric_family(n, a, b))

    # logistic (decreasing)
    for kappa in np.linspace(0.01, 5.0, 61):
        if time.time() >= deadline - 30.0:
            break
        try_update(logistic_family(n, kappa))

    # sine‑power family
    for p in np.linspace(0.2, 10.0, 81):
        if time.time() >= deadline - 30.0:
            break
        try_update(sine_power_family(n, p))

    # ------------------------------------------------------------------
    # 3. Random starts
    # ------------------------------------------------------------------
    for _ in range(200):
        if time.time() >= deadline - 30.0:
            break
        if np.random.rand() < 0.5:
            vec = random_monotone_start(n)
        else:
            vec = random_sparse_start(n)
        try_update(vec)

    # ensure we have a numeric numpy array for the subsequent phases
    if best_seq is None:
        best_arr = np.full(n, 1.0 / n, dtype=np.float64)
        best_score = compute_score(best_arr)
        best_seq = best_arr.tolist()
    else:
        best_arr = np.array(best_seq, dtype=np.float64)

    # ------------------------------------------------------------------
    # 4. L‑BFGS refinement on the smoothed surrogate
    # ------------------------------------------------------------------
    if time.time() < deadline - 250.0:
        print(f"[{time.time():.1f}] Starting L‑BFGS refinement")
        a_ref, sc_ref = l_bfgs_refine(
            best_arr,
            deadline - 200.0,
            beta_list=(200.0, 800.0, 2000.0, 5000.0),
            max_iter_per_beta=1200,
        )
        if sc_ref < best_score:
            best_score = sc_ref
            best_arr = a_ref
            best_seq = a_ref.tolist()
            print(f"[{time.time():.1f}] after L‑BFGS best = {best_score:.6f}")

    # ------------------------------------------------------------------
    # 5. Mirror sub‑gradient descent (main run + short restarts)
    # ------------------------------------------------------------------
    if time.time() < deadline - 60.0:
        print(f"[{time.time():.1f}] Starting mirror sub‑gradient descent")
        a_sub, sc_sub = mirror_subgrad_optimize(
            best_arr,
            deadline - 30.0,
            step0=1.0,
        )
        if sc_sub < best_score:
            best_score = sc_sub
            best_arr = a_sub
            best_seq = a_sub.tolist()
            print(f"[{time.time():.1f}] after mirror sub‑grad best = {best_score:.6f}")

        # a few short restarts from a slightly perturbed version
        while time.time() < deadline - 15.0:
            noise = np.random.randn(n) * 0.02
            pert = project_to_simplex(best_arr + noise)
            a_sub2, sc_sub2 = mirror_subgrad_optimize(
                pert,
                deadline - 10.0,
                step0=0.7,
                check_interval=3000,
            )
            if sc_sub2 < best_score:
                best_score = sc_sub2
                best_arr = a_sub2
                best_seq = a_sub2.tolist()
                print(f"[{time.time():.1f}] restart sub‑grad improved to {best_score:.6f}")

    # ------------------------------------------------------------------
    # 6. Final tiny simulated‑annealing hill‑climb
    # ------------------------------------------------------------------
    if best_seq is not None and time.time() < deadline - 5.0:
        print(f"[{time.time():.1f}] Final hill‑climb (SA)")
        a_final, sc_final = hill_climb_sa(
            np.array(best_seq, dtype=np.float64),
            deadline,
            max_steps=800_000,
            init_temp_factor=5e-5,
        )
        if sc_final < best_score:
            best_score = sc_final
            best_seq = a_final.tolist()
            print(f"[{time.time():.1f}] hill‑climb final best = {best_score:.6f}")

    # ------------------------------------------------------------------
    # 7. Safeguard – fall back to a modest uniform vector if necessary
    # ------------------------------------------------------------------
    if best_seq is None:
        n_small = 256
        best_seq = [1.0 / n_small] * n_small
        best_score = evaluate_sequence(best_seq)

    # ------------------------------------------------------------------
    # 8. Final official evaluation (the harness will also call it)
    # ------------------------------------------------------------------
    final_score = evaluate_sequence(best_seq)
    print(f"\n[{time.time():.1f}] finished search – final reported score = {final_score:.6f}")

    return best_seq
