```
def adam_optimize(start_a, deadline, max_iters, beta_start, beta_end, lr, sym_interval):
    n = start_a.shape[0]
    eps = 1e-12
    z = np.log(start_a + eps)
    m = np.zeros_like(z)
    v = np.zeros_like(z)
    beta1=0.9; beta2=0.999; eps_adam=1e-12

    best_a = softmax(z)
    best_score = compute_score(best_a)

    for it in range(1, max_iters+1):
        if time.time() >= deadline:
            break
        t = it / max_iters
        beta = beta_start * (beta_end / beta_start) ** t

        a = softmax(z)
        # symmetrize each iteration
        if sym_interval and (it % sym_interval == 0):
            a = (a + a[::-1]) * 0.5
            a = project_to_simplex(a)
            z = np.log(a + eps)

        conv = convolution_fft(a, a)
        maxc = conv.max()
        exp_w = np.exp(beta * (conv - maxc))
        w = exp_w / exp_w.sum()
        a_rev = a[::-1]
        conv_rev = convolution_fft(a_rev, w)
        grad_a = 2.0 * conv_rev[n-1:n-1+n]

        dot_ag = np.dot(a, grad_a)
        grad_z = a * grad_a - dot_ag * a

        m = beta1 * m + (1.0 - beta1) * grad_z
        v = beta2 * v + (1.0 - beta2) * (grad_z ** 2)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        z -= lr * m_hat / (np.sqrt(v_hat) + eps_adam)

        if it % 2000 == 0:
            a_tmp = softmax(z)
            a_tmp = (a_tmp + a_tmp[::-1]) * 0.5
            a_tmp = project_to_simplex(a_tmp)
            sc = compute_score(a_tmp)
            if sc < best_score:
                best_score = sc
                best_a = a_tmp.copy()
                print(f"[{time.time():.1f}] Adam iter {it} improved to {best_score:.6f}")

    # final evaluation
    a_fin = softmax(z)
    a_fin = (a_fin + a_fin[::-1]) * 0.5
    a_fin = project_to_simplex(a_fin)
    sc_fin = compute_score(a_fin)
    if sc_fin < best_score:
        best_score = sc_fin
        best_a = a_fin.copy()
    return best_a, best_score
```

The optimiser builds a non‑negative step‑function `a = (a₀,…,aₙ₋1)` that minimises  

```
score(a) = 2·n·maxₖ (a*a)ₖ / (∑ a)² .
```

Because the score is *scale invariant* we keep the vector on the
probability simplex (`∑a = 1`).  The search consists of three stages

1. **Coarse analytic families** – a fast scan of several parametrised
   families (product‑shape, power‑law, U‑shape, β‑distribution,
   uniform+U‑mix) on a single length `n = 2048`.  
   The best candidate becomes the seed for the next stage.

2. **Stochastic gradient descent on a smooth approximation of the max**
   – we minimise  

   ```
   Fβ(a) = (1/β)·log Σₖ exp(β·(a*a)ₖ)
   ```

   using Adam on the soft‑max parametrisation `a = softmax(z)`.  
   After each Adam step the vector is symmetrised, projected onto the
   simplex and the best iterate is memorised.  Several short random‑restart
   Adam runs are performed; the best result is taken forward.

3. **Fine‑tuning** – a long high‑β Adam run (β up to 4000) followed by a
   short projected sub‑gradient descent and a lightweight simulated‑annealing
   hill‑climb.  All moves respect the non‑negativity and simplex constraints.

The whole procedure respects the 1000 s wall‑clock limit; if no candidate
has been found a uniform fallback is returned.

```python
"""
Optimise a non‑negative step‑function (probability vector) to minimise
  score = 2·n·max(a*a) / (sum(a)²)
using a fast analytic scan, several stochastic‑gradient Adam runs on a
smooth approximation of the max‑operator, a final high‑β Adam refinement,
and a brief simulated‑annealing hill‑climb.  All vectors are kept on the
simplex (∑a = 1) and forced to be symmetric to accelerate convergence.
"""
import time
from typing import List, Tuple

import numpy as np

# ----------------------------------------------------------------------
# Core utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full aperiodic convolution using real FFT."""
    m = a.size + b.size - 1
    nfft = next_pow2(m)
    A = np.fft.rfft(a, nfft)
    B = np.fft.rfft(b, nfft)
    C = A * B
    conv = np.fft.irfft(C, nfft)[:m]
    return conv


def compute_score(a: np.ndarray) -> float:
    """
    Exact objective for a probability vector (∑a = 1).
    Returns 2·n·max(conv) where conv = a*a.
    """
    n = a.shape[0]
    conv = convolution_fft(a, a)
    return 2.0 * n * float(conv.max())


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / v.shape[0])

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, v.shape[0] + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.where(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def softmax(z: np.ndarray) -> np.ndarray:
    """Numerically stable soft‑max producing a probability vector."""
    z_max = np.max(z)
    e = np.exp(z - z_max)
    return e / e.sum()


# ----------------------------------------------------------------------
# Analytic families
# ----------------------------------------------------------------------


def symmetric_product_family(n: int, alpha: float) -> np.ndarray:
    """a_i ∝ [(i+½)(n‑½‑i)]^{‑α}, symmetric, on the simplex."""
    i = np.arange(n, dtype=np.float64)
    logprod = np.log(i + 0.5) + np.log(n - 0.5 - i)
    w = np.exp(-alpha * logprod)
    w[np.isinf(w)] = np.finfo(np.float64).max
    return w / w.sum()


def left_heavy_power_family(n: int, beta: float) -> np.ndarray:
    """a_i ∝ (i+½)^{‑β}, monotone decreasing."""
    i = np.arange(n, dtype=np.float64)
    w = np.exp(-beta * np.log(i + 0.5))
    return w / w.sum()


def exp_u_family(n: int, gamma: float) -> np.ndarray:
    """U‑shape: a_i ∝ exp(-γ·min(i, n‑1‑i))."""
    i = np.arange(n, dtype=np.float64)
    dist = np.minimum(i, n - 1 - i)
    w = np.exp(-gamma * dist)
    return w / w.sum()


def mixed_uniform_u_family(n: int, lam: float, gamma: float) -> np.ndarray:
    """Convex mixture of uniform and exponential‑U shape."""
    uniform = np.full(n, 1.0 / n, dtype=np.float64)
    u_shape = exp_u_family(n, gamma)
    return lam * uniform + (1.0 - lam) * u_shape


def beta_family(n: int, p: float, q: float) -> np.ndarray:
    """Beta‑type density ∝ (i+½)^{p‑1} (n‑½‑i)^{q‑1}."""
    i = np.arange(n, dtype=np.float64)
    w = np.power(i + 0.5, p - 1.0) * np.power(n - 0.5 - i, q - 1.0)
    return w / w.sum()


# ----------------------------------------------------------------------
# Random initialisers
# ----------------------------------------------------------------------


def random_monotone_start(n: int) -> np.ndarray:
    """Decreasing vector obtained by sorting a powered uniform."""
    r = np.random.rand(n) ** np.random.uniform(0.5, 5.0)
    r.sort()
    a = r[::-1]
    return a / a.sum()


def random_u_start(n: int) -> np.ndarray:
    """Random exponential‑U shape."""
    gamma = np.random.uniform(0.0, 6.0)
    return exp_u_family(n, gamma)


# ----------------------------------------------------------------------
# Smoothed‑max Adam optimiser
# ----------------------------------------------------------------------


def adam_smooth_optimize(start_a: np.ndarray,
                         deadline: float,
                         max_iters: int = 200_000,
                         beta_start: float = 5.0,
                         beta_end: float = 1500.0,
                         lr: float = 0.018,
                         sym_interval: int = 2000) -> Tuple[np.ndarray, float]:
    """
    Adam optimiser on the smooth approximation of max‑convolution.
    Parameters
    ----------
    start_a : ndarray
        Initial probability vector (sum = 1, non‑negative).
    deadline : float
        Wall‑clock time at which the routine must stop.
    max_iters : int
        Upper bound on the number of Adam steps.
    beta_start, beta_end : float
        Exponential schedule for the smoothing parameter β.
    lr : float
        Base learning rate.
    sym_interval : int
        How often the iterate is forced to be symmetric.
    Returns
    -------
    best_a, best_score : (ndarray, float)
        Best vector seen and its exact score (computed with compute_score).
    """
    n = start_a.shape[0]
    eps = 1e-12
    z = np.log(start_a + eps)          # soft‑max parameters
    m = np.zeros_like(z)
    v = np.zeros_like(z)

    beta1 = 0.9
    beta2 = 0.999
    eps_adam = 1e-12

    best_a = softmax(z)
    best_score = compute_score(best_a)

    for it in range(1, max_iters + 1):
        if time.time() >= deadline:
            break

        # exponential schedule for β
        t = it / max_iters
        beta = beta_start * (beta_end / beta_start) ** t

        a = softmax(z)

        # enforce symmetry regularly
        if sym_interval and (it % sym_interval == 0):
            a = (a + a[::-1]) * 0.5
            a = project_to_simplex(a)
            z = np.log(a + eps)

        conv = convolution_fft(a, a)
        maxc = conv.max()
        exp_w = np.exp(beta * (conv - maxc))
        w = exp_w / exp_w.sum()

        # gradient of the smooth max w.r.t. a
        a_rev = a[::-1]
        conv_rev = convolution_fft(a_rev, w)
        grad_a = 2.0 * conv_rev[n - 1:n - 1 + n]

        # back‑prop through soft‑max
        dot_ag = np.dot(a, grad_a)
        grad_z = a * grad_a - dot_ag * a

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad_z
        v = beta2 * v + (1.0 - beta2) * (grad_z ** 2)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        z -= lr * m_hat / (np.sqrt(v_hat) + eps_adam)

        # occasional exact evaluation
        if it % 2000 == 0:
            a_tmp = softmax(z)
            a_tmp = (a_tmp + a_tmp[::-1]) * 0.5
            a_tmp = project_to_simplex(a_tmp)
            sc = compute_score(a_tmp)
            if sc < best_score:
                best_score = sc
                best_a = a_tmp.copy()
                print(f"[{time.time():.1f}] Adam iter {it} improved to {best_score:.6f}")

    # final evaluation
    a_fin = softmax(z)
    a_fin = (a_fin + a_fin[::-1]) * 0.5
    a_fin = project_to_simplex(a_fin)
    sc_fin = compute_score(a_fin)
    if sc_fin < best_score:
        best_score = sc_fin
        best_a = a_fin.copy()
    return best_a, best_score


# ----------------------------------------------------------------------
# Projected sub‑gradient descent on the true max‑objective
# ----------------------------------------------------------------------


def subgrad_optimize(start_a: np.ndarray,
                     deadline: float,
                     max_iters: int = 3_000_000,
                     step0: float = 0.07) -> Tuple[np.ndarray, float]:
    """
    Classic sub‑gradient descent for the exact max‑objective.
    Uses a diminishing step size ∝1/√t and Euclidean projection onto the simplex.
    """
    n = start_a.shape[0]
    a = start_a.astype(np.float64).copy()
    best_a = a.copy()
    best_score = compute_score(a)

    for t in range(1, max_iters + 1):
        if time.time() >= deadline:
            break

        conv = convolution_fft(a, a)
        k = int(np.argmax(conv))

        idx = np.arange(n)
        j = k - idx
        mask = (j >= 0) & (j < n)
        g = np.zeros_like(a)
        g[mask] = 2.0 * a[j[mask]]

        step = step0 / np.sqrt(t)
        a = project_to_simplex(a - step * g)

        if t % 2000 == 0:
            sc = compute_score(a)
            if sc < best_score:
                best_score = sc
                best_a = a.copy()
                print(f"[{time.time():.1f}] sub‑grad iter {t} improved to {best_score:.6f}")

    sc = compute_score(a)
    if sc < best_score:
        best_score = sc
        best_a = a.copy()
    return best_a, best_score


# ----------------------------------------------------------------------
# Simulated‑annealing hill‑climb
# ----------------------------------------------------------------------


def hill_climb_sa(start_a: np.ndarray,
                  deadline: float,
                  max_steps: int = 500_000) -> Tuple[np.ndarray, float]:
    """
    Lightweight SA on the simplex:
      – 50 % swap two entries,
      – 50 % transfer a fraction of mass.
    A very low temperature schedule allows occasional uphill moves.
    """
    n = start_a.size
    cur_a = start_a.copy()
    cur_score = compute_score(cur_a)

    best_a = cur_a.copy()
    best_score = cur_score

    T0 = best_score * 1e-4
    T = T0
    step_scale = 0.12

    for step in range(1, max_steps + 1):
        if time.time() >= deadline:
            break

        cand = cur_a.copy()
        if np.random.rand() < 0.5:            # swap
            i, j = np.random.randint(n, size=2)
            cand[i], cand[j] = cand[j], cand[i]
        else:                                 # mass transfer
            i, j = np.random.randint(n, size=2)
            if i != j:
                delta = step_scale * cur_a[i]
                cand[i] -= delta
                cand[j] += delta
                if cand[i] < 0.0:
                    cand[i] = 0.0

        cand = project_to_simplex(cand)
        cand_score = compute_score(cand)

        delta = cand_score - cur_score
        if delta < 0 or np.random.rand() < np.exp(-delta / (T + 1e-18)):
            cur_a = cand
            cur_score = cand_score
            if cur_score < best_score:
                best_score = cur_score
                best_a = cur_a.copy()
                step_scale = min(step_scale * 1.07, 0.5)
        else:
            step_scale *= 0.97

        T = T0 * (1.0 - step / max_steps)

    return best_a, best_score

def propose_candidate(seed: int = 12345,
                     budget_s: int = 1000,
                     **kwargs) -> List[float]:
    """
    Search for a non‑negative step‑function sequence minimising
    ``evaluate_sequence`` under the simplex constraint.
    The routine respects the wall‑clock budget and returns a plain Python list.
    """
    np.random.seed(seed)
    start_time = time.time()
    # keep a small safety margin
    deadline = start_time + budget_s - 1.0

    best_seq = None
    best_score = float("inf")

    # -----------------------------------------------------------
    # 0. fallback candidate (if available)
    # -----------------------------------------------------------
    try:
        fb_seq = [float(x) for x in height_sequence_1]   # type: ignore
        fb = evaluate_sequence(fb_seq)
        print(f"[{time.time():.1f}] fallback score = {fb:.6f}")
        if fb < best_score:
            best_score = fb
            best_seq = fb_seq
    except Exception:
        pass

    # -----------------------------------------------------------
    # 1. Analytic families – quick coarse scan (n = 2048)
    # -----------------------------------------------------------
    n = 2048
    if time.time() < deadline - 30.0:
        print(f"[{time.time():.1f}] scanning analytic families (n={n})")
        # symmetric product family
        for alpha in np.linspace(0.0, 12.0, 31):
            if time.time() >= deadline - 10.0:
                break
            vec = symmetric_product_family(n, alpha)
            sc = compute_score(vec)
            if sc < best_score:
                best_score = sc
                best_seq = vec.tolist()
                print(f"[{time.time():.1f}] new best (sym‑α={alpha:.2f}) = {best_score:.6f}")

        # left‑heavy power family
        for beta in np.linspace(0.0, 12.0, 31):
            if time.time() >= deadline - 10.0:
                break
            vec = left_heavy_power_family(n, beta)
            sc = compute_score(vec)
            if sc < best_score:
                best_score = sc
                best_seq = vec.tolist()
                print(f"[{time.time():.1f}] new best (left‑β={beta:.2f}) = {best_score:.6f}")

        # exponential U‑shape family
        for gamma in np.linspace(0.0, 6.0, 21):
            if time.time() >= deadline - 10.0:
                break
            vec = exp_u_family(n, gamma)
            sc = compute_score(vec)
            if sc < best_score:
                best_score = sc
                best_seq = vec.tolist()
                print(f"[{time.time():.1f}] new best (exp‑γ={gamma:.2f}) = {best_score:.6f}")

        # beta‑type family
        for p in np.linspace(0.2, 3.0, 10):
            if time.time() >= deadline - 10.0:
                break
            for q in np.linspace(0.2, 3.0, 10):
                vec = beta_family(n, p, q)
                sc = compute_score(vec)
                if sc < best_score:
                    best_score = sc
                    best_seq = vec.tolist()
                    print(f"[{time.time():.1f}] new best (beta p={p:.2f}, q={q:.2f}) = {best_score:.6f}")

        # mixture of uniform and U‑shape
        for lam in (0.0, 0.25, 0.5, 0.75, 1.0):
            for gamma in np.linspace(0.0, 6.0, 13):
                if time.time() >= deadline - 10.0:
                    break
                vec = mixed_uniform_u_family(n, lam, gamma)
                sc = compute_score(vec)
                if sc < best_score:
                    best_score = sc
                    best_seq = vec.tolist()
                    print(f"[{time.time():.1f}] new best (mix λ={lam:.2f}, γ={gamma:.2f}) = {best_score:.6f}")

        # a few random starts for extra coverage
        for _ in range(30):
            if time.time() >= deadline - 5.0:
                break
            vec = random_monotone_start(n) if np.random.rand() < 0.5 else random_u_start(n)
            vec = (vec + vec[::-1]) * 0.5
            vec = project_to_simplex(vec)
            sc = compute_score(vec)
            if sc < best_score:
                best_score = sc
                best_seq = vec.tolist()
                print(f"[{time.time():.1f}] new best (random) = {best_score:.6f}")

    # -----------------------------------------------------------
    # 2. Stochastic Adam runs with random restarts
    # -----------------------------------------------------------
    while best_seq is not None and time.time() < deadline - 250.0:
        # allocate a short chunk (≈20 s) for each random‑restart Adam run
        chunk_deadline = min(deadline - 180.0, time.time() + 20.0)
        n = len(best_seq)
        start_vec = random_monotone_start(n)
        start_vec = (start_vec + start_vec[::-1]) * 0.5
        start_vec = project_to_simplex(start_vec)

        a_opt, sc_opt = adam_smooth_optimize(
            start_vec,
            chunk_deadline,
            max_iters=120_000,
            beta_start=5.0,
            beta_end=2000.0,
            lr=0.015,
            sym_interval=2000,
        )
        if sc_opt < best_score:
            best_score = sc_opt
            best_seq = a_opt.tolist()
            print(f"[{time.time():.1f}] random‑restart Adam improved to {best_score:.6f}")

    # -----------------------------------------------------------
    # 3. Final high‑β Adam refinement
    # -----------------------------------------------------------
    if best_seq is not None and time.time() < deadline - 120.0:
        print(f"[{time.time():.1f}] final high‑β Adam refinement")
        a_start = np.array(best_seq, dtype=np.float64)
        a_opt, sc_opt = adam_smooth_optimize(
            a_start,
            deadline - 80.0,
            max_iters=300_000,
            beta_start=20.0,
            beta_end=4000.0,
            lr=0.01,
            sym_interval=1000,
        )
        if sc_opt < best_score:
            best_score = sc_opt
            best_seq = a_opt.tolist()
            print(f"[{time.time():.1f}] after high‑β Adam best = {best_score:.6f}")

    # -----------------------------------------------------------
    # 4. Projected sub‑gradient descent (optional)
    # -----------------------------------------------------------
    if best_seq is not None and time.time() < deadline - 60.0:
        print(f"[{time.time():.1f}] projected sub‑gradient descent")
        a_sub, sc_sub = subgrad_optimize(
            np.array(best_seq, dtype=np.float64),
            deadline - 30.0,
            max_iters=2_000_000,
            step0=0.07,
        )
        if sc_sub < best_score:
            best_score = sc_sub
            best_seq = a_sub.tolist()
            print(f"[{time.time():.1f}] after sub‑gradient best = {best_score:.6f}")

    # -----------------------------------------------------------
    # 5. Light simulated‑annealing hill‑climb
    # -----------------------------------------------------------
    if best_seq is not None and time.time() < deadline - 10.0:
        print(f"[{time.time():.1f}] final hill‑climb (SA)")
        a_final, sc_final = hill_climb_sa(
            np.array(best_seq, dtype=np.float64),
            deadline,
            max_steps=300_000,
        )
        if sc_final < best_score:
            best_score = sc_final
            best_seq = a_final.tolist()
            print(f"[{time.time():.1f}] hill‑climb improved best = {best_score:.6f}")

    # -----------------------------------------------------------
    # 6. Safeguard fallback (should never be needed)
    # -----------------------------------------------------------
    if best_seq is None:
        n_small = 256
        best_seq = [1.0 / n_small] * n_small

    final_score = evaluate_sequence(best_seq)
    print(f"\n[{time.time():.1f}] finished search – final reported score = {final_score:.6f}")
    return best_seq
```
