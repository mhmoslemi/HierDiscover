"""
Propose a non‑negative step‑function (list of heights) that minimises

    2·n·max(convolve(a,a)) / (sum(a)**2)

using a multi‑phase local‑search:
1.  Load a fallback if available.
2.  Quick scan of several analytic families (including a new sinusoid family)
    and random Dirichlet starts.
3.  Long Adam optimisation on a smoothed max (β increased up to 3000).
4.  Very cheap projected sub‑gradient descent (more iterations than before).
5.  Repeated LP‑based refinement (the classic “max‑sum” mixing).
6.  Symmetrisation.
7.  An aggressive simulated‑annealing hill‑climb with a higher initial
    temperature and many steps.
8.  Final sparsification (zeroing tiny entries) and return the best sequence.

All phases respect a wall‑clock budget (default 1000 s).  The code is
self‑contained and only relies on ``numpy`` and ``scipy``.
"""

import time
import math
import numpy as np
from scipy.optimize import linprog

# ----------------------------------------------------------------------
# Helper utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full (aperiodic) convolution via real FFT."""
    m = a.size + b.size - 1
    nfft = next_pow2(m)
    A = np.fft.rfft(a, n=nfft)
    B = np.fft.rfft(b, n=nfft)
    C = A * B
    conv = np.fft.irfft(C, n=nfft)[:m]
    return conv


def compute_score(a: np.ndarray) -> float:
    """Exact objective when sum(a)=1: 2·n·max(a*a)."""
    n = a.shape[0]
    conv = convolution_fft(a, a)
    return 2.0 * n * float(conv.max())


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of a vector onto the probability simplex."""
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / v.shape[0])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, v.shape[0] + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.where(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def softmax(z: np.ndarray) -> np.ndarray:
    """Numerically stable soft‑max."""
    z_max = np.max(z)
    e = np.exp(z - z_max)
    return e / e.sum()


# ----------------------------------------------------------------------
# Analytic families (used in the quick scan)
# ----------------------------------------------------------------------


def symmetric_product_family(n: int, alpha: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    logprod = np.log(i + 0.5) + np.log(n - 0.5 - i)
    w = np.exp(-alpha * logprod)
    a = w / w.sum()
    return a


def left_heavy_power_family(n: int, beta: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    w = np.exp(-beta * np.log(i + 0.5))
    a = w / w.sum()
    return a


def exp_u_family(n: int, gamma: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    dist = np.minimum(i, n - 1 - i)
    w = np.exp(-gamma * dist)
    a = w / w.sum()
    return a


def mixed_uniform_u_family(n: int, lam: float, gamma: float) -> np.ndarray:
    uniform = np.full(n, 1.0 / n, dtype=np.float64)
    u_shape = exp_u_family(n, gamma)
    a = lam * uniform + (1.0 - lam) * u_shape
    return a


def sinusoid_family(n: int, power: float) -> np.ndarray:
    """a_i ∝ sin(π·(i+0.5)/n)^power , symmetric and zero at the ends."""
    i = np.arange(n, dtype=np.float64)
    w = np.sin(np.pi * (i + 0.5) / n) ** power
    a = w / w.sum()
    return a


def random_dirichlet_start(n: int, concentration: float = 1.0) -> np.ndarray:
    """Draw a point from Dirichlet(concentration,…,concentration)."""
    x = np.random.gamma(concentration, 1.0, size=n)
    a = x / x.sum()
    return a


# ----------------------------------------------------------------------
# Smoothed‑max Adam optimisation (more aggressive than the previous code)
# ----------------------------------------------------------------------


def adam_smooth_optimize(start_a: np.ndarray,
                         deadline: float,
                         max_iters: int = 500_000,
                         beta_start: float = 4.0,
                         beta_end: float = 3000.0,
                         lr: float = 0.012) -> tuple[np.ndarray, float]:
    """
    Adam optimiser for the smooth surrogate
        f_β(a) = (1/β)·log Σ exp(β·(a*a)_k)
    using the soft‑max parametrisation a = softmax(z).
    """
    eps = 1e-12
    z = np.log(start_a + eps)               # initialise logits

    m = np.zeros_like(z)
    v = np.zeros_like(z)
    beta1 = 0.9
    beta2 = 0.999
    eps_adam = 1e-12

    best_a = softmax(z)
    best_score = compute_score(best_a)

    for it in range(1, max_iters + 1):
        if time.time() >= deadline:
            break

        # schedule β → β_end
        t = it / max_iters
        beta = beta_start * (beta_end / beta_start) ** t

        a = softmax(z)
        conv = convolution_fft(a, a)
        maxc = conv.max()
        # smooth‑max weights
        exp_w = np.exp(beta * (conv - maxc))
        w = exp_w / exp_w.sum()

        # gradient ∂f_β/∂a = 2 * (a_rev * w)  (central part only)
        a_rev = a[::-1]
        conv_rev = convolution_fft(a_rev, w)
        grad_a = 2.0 * conv_rev[len(a) - 1:len(a) - 1 + len(a)]

        # back‑prop through soft‑max
        dot_ag = np.dot(a, grad_a)
        grad_z = a * grad_a - dot_ag * a

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad_z
        v = beta2 * v + (1.0 - beta2) * (grad_z ** 2)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        z -= lr * m_hat / (np.sqrt(v_hat) + eps_adam)

        if it % 1000 == 0:
            a_tmp = softmax(z)
            sc = compute_score(a_tmp)
            if sc < best_score:
                best_score = sc
                best_a = a_tmp.copy()
                print(f"[{time.time():.1f}] Adam improved to {best_score:.6f}")

    # final check
    a_fin = softmax(z)
    sc_fin = compute_score(a_fin)
    if sc_fin < best_score:
        best_score = sc_fin
        best_a = a_fin.copy()
    return best_a, best_score


# ----------------------------------------------------------------------
# Projected sub‑gradient descent (long cheap phase)
# ----------------------------------------------------------------------


def subgrad_optimize(start_a: np.ndarray,
                     deadline: float,
                     max_iters: int = 6_000_000,
                     step0: float = 0.09) -> tuple[np.ndarray, float]:
    """
    Sub‑gradient method on the true max‑objective.
    Uses a diminishing stepsize step0 / sqrt(t) and projects onto the simplex.
    """
    n = start_a.shape[0]
    a = start_a.astype(np.float64).copy()

    best_a = a.copy()
    best_score = compute_score(a)

    for t in range(1, max_iters + 1):
        if time.time() >= deadline:
            break

        conv = convolution_fft(a, a)
        k = int(np.argmax(conv))

        # build sub‑gradient for the max position k
        idx = np.arange(n)
        j = k - idx
        mask = (j >= 0) & (j < n)
        g = np.zeros_like(a)
        g[mask] = 2.0 * a[j[mask]]
        # diagonal terms (j == idx) already have factor 2, which is correct
        # because d(a_i*a_i)/da_i = 2 a_i

        step = step0 / np.sqrt(t)
        a = project_to_simplex(a - step * g)

        if t % 4000 == 0:
            sc = compute_score(a)
            if sc < best_score:
                best_score = sc
                best_a = a.copy()
                print(f"[{time.time():.1f}] sub‑grad improved to {best_score:.6f}")

    sc = compute_score(a)
    if sc < best_score:
        best_score = sc
        best_a = a.copy()
    return best_a, best_score


# ----------------------------------------------------------------------
# LP‑based refinement (state‑of‑the‑art iteration)
# ----------------------------------------------------------------------


def lp_refine_step(f: np.ndarray, M: float, mix_weight: float = 0.2) -> np.ndarray:
    """
    Solve   max Σ g_j   s.t.   (f * g)_k ≤ M   for all k,
    g ≥ 0, then mix f←(1−t)f + t·g_norm and re‑project onto the simplex.
    """
    n = f.shape[0]
    m = 2 * n - 1

    # Build convolution matrix A (A @ g = f * g)
    A = np.zeros((m, n), dtype=np.float64)
    for i in range(n):
        A[i:i + n, i] = f[i]

    c = -np.ones(n, dtype=np.float64)                 # maximise Σ g ⇒ minimise −Σ g
    bounds = [(0.0, None) for _ in range(n)]
    b = np.full(m, M, dtype=np.float64)

    res = linprog(c, A_ub=A, b_ub=b, bounds=bounds,
                  method='highs',
                  options={'presolve': True, 'time_limit': 30})

    if not res.success:
        return f

    g = np.maximum(res.x, 0.0)
    sum_g = g.sum()
    if sum_g <= 0.0:
        return f

    g_norm = g / sum_g
    f_new = (1.0 - mix_weight) * f + mix_weight * g_norm
    f_new = project_to_simplex(f_new)
    return f_new

# Aggressive simulated‑annealing hill‑climb
# ----------------------------------------------------------------------


def hill_climb_sa(start_a: np.ndarray,
                  deadline: float,
                  max_steps: int = 2_500_000) -> tuple[np.ndarray, float]:
    """
    Simulated‑annealing style walk on the simplex.
    Uses a higher initial temperature to escape deeper local minima.
    """
    n = start_a.size
    cur_a = start_a.copy()
    cur_score = compute_score(cur_a)

    best_a = cur_a.copy()
    best_score = cur_score

    # start temperature = a few percent of current score (instead of 1e‑4)
    T0 = max(best_score * 0.02, 1e-4)
    T = T0
    step_scale = 0.15

    for step in range(1, max_steps + 1):
        if time.time() >= deadline:
            break

        cand = cur_a.copy()
        # random elementary operation
        if np.random.rand() < 0.5:
            i, j = np.random.randint(n, size=2)
            cand[i], cand[j] = cand[j], cand[i]          # swap
        else:
            i, j = np.random.randint(n, size=2)
            if i != j:
                delta = step_scale * cur_a[i]
                cand[i] -= delta
                cand[j] += delta
                if cand[i] < 0.0:
                    cand[i] = 0.0

        cand = project_to_simplex(cand)
        cand_score = compute_score(cand)

        delta = cand_score - cur_score
        if delta < 0 or np.random.rand() < np.exp(-delta / (T + 1e-18)):
            cur_a = cand
            cur_score = cand_score
            if cur_score < best_score:
                best_score = cur_score
                best_a = cur_a.copy()
                step_scale = min(step_scale * 1.08, 0.6)
        else:
            step_scale *= 0.96

        # linear cooling
        T = T0 * (1.0 - step / max_steps)

    return best_a, best_score

def sparsify(a: np.ndarray, eps: float = 1e-8) -> np.ndarray:
    a = np.where(a < eps, 0.0, a)
    if a.sum() == 0.0:
        # fallback to uniform if everything vanished
        a = np.full_like(a, 1.0 / a.size)
    else:
        a = a / a.sum()
    return a

        while time.time() < deadline - 15.0:
            conv = convolution_fft(a, a)
            M = float(conv.max())
            a_new = lp_refine_step(a, M, mix_weight=mix_weight)
            sc = compute_score(a_new)
            if sc < best_score - 1e-7:
                best_score = sc
                a = a_new
                print(f"[{time.time():.1f}] LP step {lp_iters} improved to {best_score:.6f}")
            else:
                mix_weight = max(0.05, mix_weight * 0.9)
            lp_iters += 1
            if lp_iters >= 20:
                break

    # --------------------------------------------------------------
    # 6. Symmetrisation test
    # --------------------------------------------------------------
    a_sym = (a + a[::-1]) * 0.5
    a_sym = project_to_simplex(a_sym)
    sc_sym = compute_score(a_sym)
    if sc_sym < best_score:
        best_score = sc_sym
        a = a_sym
        print(f"[{time.time():.1f}] symmetrisation improved best = {best_score:.6f}")

    # --------------------------------------------------------------
    # 7. Final aggressive stochastic hill‑climb (SA)
    # --------------------------------------------------------------
    if time.time() < deadline - 10.0:
        print(f"[{time.time():.1f}] final hill‑climb (SA)")
        a_opt, sc_opt = hill_climb_sa(a, deadline, max_steps=2_500_000)
        if sc_opt < best_score:
            best_score = sc_opt
            a = a_opt
            print(f"[{time.time():.1f}] hill‑climb improved best = {best_score:.6f}")

    # --------------------------------------------------------------
    # 8. Sparsification (remove tiny entries) + final projection
    # --------------------------------------------------------------
    a = sparsify(a, eps=1e-8)

    # --------------------------------------------------------------
    # 9. Final official evaluation and return
    # --------------------------------------------------------------
    final_seq = a.tolist()
    final_score = evaluate_sequence(final_seq)
    print(f"\n[{time.time():.1f}] finished search – final reported score = {final_score:.6f}")
    return final_seq
