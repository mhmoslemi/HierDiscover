def lp_refine_step(f: np.ndarray, M: float, mix_weight_max: float = 0.5) -> np.ndarray:
    """
    Solve max Σ g_j s.t. (f*g)_k ≤ M, then mix the current f with g.
    The mixing weight t ∈ [0, mix_weight_max] is chosen (by a small
    line‑search) to minimise the resulting max‑convolution.
    """
    n = f.shape[0]
    m = 2 * n - 1
    # Build convolution matrix A (size m×n)
    A = np.zeros((m, n), dtype=np.float64)
    for i in range(n):
        A[i:i + n, i] = f[i]

    # Solve LP: maximise Σ g_j  ⇔ minimise - Σ g_j
    c = -np.ones(n, dtype=np.float64)
    bounds = [(0.0, None) for _ in range(n)]
    b = np.full(m, M, dtype=np.float64)

    res = linprog(c, A_ub=A, b_ub=b, bounds=bounds, method='highs')
    if not res.success:
        return f

    g = np.maximum(res.x, 0.0)
    sum_g = g.sum()
    if sum_g <= 0.0:
        return f

    g_norm = g / sum_g

    # Pre‑compute the three convolutions needed for the quadratic mixture.
    conv_f = convolution_fft(f, f)
    conv_g = convolution_fft(g_norm, g_norm)
    conv_fg = convolution_fft(f, g_norm)

    # Search for the best mixing weight t ∈ [0, mix_weight_max].
    # Coarse grid first, then a local refinement.
    best_t = 0.0
    best_M = float('inf')
    for t in np.linspace(0.0, mix_weight_max, 11):
        mixed = ((1 - t) ** 2) * conv_f + 2 * t * (1 - t) * conv_fg + (t ** 2) * conv_g
        cur_M = mixed.max()
        if cur_M < best_M:
            best_M = cur_M
            best_t = t

    # Refine around the best point
    if best_t > 0.0 and best_t < mix_weight_max:
        left = max(0.0, best_t - mix_weight_max / 10)
        right = min(mix_weight_max, best_t + mix_weight_max / 10)
        for t in np.linspace(left, right, 9):
            mixed = ((1 - t) ** 2) * conv_f + 2 * t * (1 - t) * conv_fg + (t ** 2) * conv_g
            cur_M = mixed.max()
            if cur_M < best_M:
                best_M = cur_M
                best_t = t

    # Form the new mixture
    f_new = (1.0 - best_t) * f + best_t * g_norm
    f_new = project_to_simplex((f_new + f_new[::-1]) * 0.5)   # keep symmetry
    return f_new

"""
Multi‑restart optimisation for the step‑function problem
-------------------------------------------------------

Goal: minimise  

    2·n·max(convolve(a, a)) / (sum(a)**2)

over non‑negative sequences a (the simplex constraint).  The objective is
homogeneous, therefore we work with probability vectors (∑a = 1) and
the score simplifies to 2·n·max(convolution).

The search combines several ideas:

1. **Analytic seeding** – a quick scan of several parametric families
   (U‑shape, exponential, power‑law, mixtures) provides a decent starter.

2. **Smooth‑max Adam** – minimise a differentiable surrogate of the max
   (log‑sum‑exp) with Adam on the soft‑max parametrisation of a.  The
   smoothing parameter β is increased from 4 up to 2000, giving a tight
   approximation to the true max.

3. **Projected sub‑gradient descent** – a cheap true‑max optimiser that
   follows the sub‑gradient of the active convolution entry.

4. **LP refinement** – solve the classic linear programme  

        maximise Σ g_j   s.t. (f * g)_k ≤ M  (∀k) , g ≥ 0,

   where M is the current maximum.  After obtaining g, the mixing weight
   t is optimally chosen (by a tiny line‑search) to minimise the new
   max‑convolution.

5. **Symmetrisation** – the problem is symmetric, so after every major
   step we replace a by (a + a[::-1]) / 2 and project onto the simplex.

6. **Simulated‑annealing hill‑climb** – a final stochastic local search
   with occasional uphill moves.

7. **Multi‑restart** – the whole pipeline is launched from many different
   random starts (or a perturbed version of the current best) until the
   wall‑clock budget (default 1000 s) is exhausted.  The best sequence
   encountered is returned.

All helper functions are defined at top‑level; no closures or file I/O
are used.  The function respects the global time budget and prints
useful progress information.
"""

import time
import math
import numpy as np
from scipy.optimize import linprog

# ----------------------------------------------------------------------
# Basic utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full (aperiodic) convolution via real FFT."""
    m = a.size + b.size - 1
    nfft = next_pow2(m)
    A = np.fft.rfft(a, n=nfft)
    B = np.fft.rfft(b, n=nfft)
    C = A * B
    conv = np.fft.irfft(C, n=nfft)[:m]
    return conv


def compute_score(a: np.ndarray) -> float:
    """
    Exact objective for a probability vector (∑a = 1):
        2·n·max(a*a)
    """
    n = a.shape[0]
    conv = convolution_fft(a, a)
    return 2.0 * n * float(conv.max())


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of v onto the probability simplex.
    """
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / v.shape[0])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, v.shape[0] + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.where(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def softmax(z: np.ndarray) -> np.ndarray:
    """Numerically stable soft‑max."""
    z_max = np.max(z)
    e = np.exp(z - z_max)
    return e / e.sum()


# ----------------------------------------------------------------------
# Analytic families (for quick seeding)
# ----------------------------------------------------------------------


def symmetric_product_family(n: int, alpha: float) -> np.ndarray:
    """U‑shaped family a_i ∝ ((i+0.5)*(n‑0.5‑i))^{‑α}."""
    i = np.arange(n, dtype=np.float64)
    logprod = np.log(i + 0.5) + np.log(n - 0.5 - i)
    w = np.exp(-alpha * logprod)
    a = w / w.sum()
    return a


def left_heavy_power_family(n: int, beta: float) -> np.ndarray:
    """Monotonically decreasing power‑law a_i ∝ (i+0.5)^{‑β}."""
    i = np.arange(n, dtype=np.float64)
    w = np.exp(-beta * np.log(i + 0.5))
    a = w / w.sum()
    return a


def exp_u_family(n: int, gamma: float) -> np.ndarray:
    """Symmetric exponential “U” shape a_i ∝ exp(-γ·dist(i, centre))."""
    i = np.arange(n, dtype=np.float64)
    dist = np.minimum(i, n - 1 - i)
    w = np.exp(-gamma * dist)
    a = w / w.sum()
    return a


def mixed_uniform_u_family(n: int, lam: float, gamma: float) -> np.ndarray:
    """
    Mixture of the uniform distribution and the exponential U‑shape.
    lam ∈ [0,1] is the weight of the uniform part.
    """
    uniform = np.full(n, 1.0 / n, dtype=np.float64)
    u_shape = exp_u_family(n, gamma)
    a = lam * uniform + (1.0 - lam) * u_shape
    return a


def random_monotone_start(n: int) -> np.ndarray:
    """Random monotone decreasing vector (normalized)."""
    r = np.random.rand(n) ** np.random.uniform(0.5, 5.0)
    r.sort()
    a = r[::-1]
    a /= a.sum()
    return a


def random_u_start(n: int) -> np.ndarray:
    """Random exponential‑U shape."""
    gamma = np.random.uniform(0.0, 8.0)
    return exp_u_family(n, gamma)

# Optimisation primitives

def adam_smooth_optimize(start_a: np.ndarray,
                         deadline: float,
                         max_iters: int = 600_000,
                         beta_start: float = 4.0,
                         beta_end: float = 2000.0,
                         lr: float = 0.02) -> tuple[np.ndarray, float]:
    """
    Adam optimiser on the smooth surrogate

        f_β(a) = (1/β)·log Σ exp(β·(a*a)_k).

    a is represented as a soft‑max of a free vector z.
    """
    eps = 1e-12
    z = np.log(start_a + eps)

    m = np.zeros_like(z)
    v = np.zeros_like(z)
    beta1 = 0.9
    beta2 = 0.999
    eps_adam = 1e-12

    best_a = softmax(z)
    best_score = compute_score(best_a)

    for it in range(1, max_iters + 1):
        if time.time() >= deadline:
            break

        t = it / max_iters
        beta = beta_start * (beta_end / beta_start) ** t

        a = softmax(z)
        conv = convolution_fft(a, a)
        maxc = conv.max()
        # soft‑max weights
        exp_w = np.exp(beta * (conv - maxc))
        w = exp_w / exp_w.sum()

        # gradient w.r.t. a
        a_rev = a[::-1]
        conv_rev = convolution_fft(a_rev, w)
        grad_a = 2.0 * conv_rev[len(a) - 1:len(a) - 1 + len(a)]

        # back‑prop through soft‑max
        dot_ag = np.dot(a, grad_a)
        grad_z = a * grad_a - dot_ag * a

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad_z
        v = beta2 * v + (1.0 - beta2) * (grad_z ** 2)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        z -= lr * m_hat / (np.sqrt(v_hat) + eps_adam)

        if it % 500 == 0:
            a_tmp = softmax(z)
            sc = compute_score(a_tmp)
            if sc < best_score:
                best_score = sc
                best_a = a_tmp.copy()
                print(f"[{time.time():.1f}] Adam improved to {best_score:.6f}")

    # final check
    a_fin = softmax(z)
    sc_fin = compute_score(a_fin)
    if sc_fin < best_score:
        best_score = sc_fin
        best_a = a_fin.copy()
    return best_a, best_score


def subgrad_optimize(start_a: np.ndarray,
                     deadline: float,
                     max_iters: int = 2_500_000,
                     step0: float = 0.07) -> tuple[np.ndarray, float]:
    """
    Sub‑gradient descent for the true max‑objective.
    The gradient for the active convolution index k is:

        g_i = 2·a_{k‑i}   (i ≠ k‑i)
        g_i = 2·a_i       (i = k‑i)

    The step size follows 1/√t and each iterate is projected onto the simplex.
    """
    n = start_a.shape[0]
    a = start_a.astype(np.float64).copy()

    best_a = a.copy()
    best_score = compute_score(a)

    for t in range(1, max_iters + 1):
        if time.time() >= deadline:
            break

        conv = convolution_fft(a, a)
        k = int(np.argmax(conv))

        idx = np.arange(n)
        j = k - idx
        mask = (j >= 0) & (j < n)

        g = np.zeros_like(a)
        # off‑diagonal contributions
        g[mask] = 2.0 * a[j[mask]]
        # diagonal (i=j) contributions – use factor 2 instead of 1
        diag = (j == idx) & mask
        if np.any(diag):
            g[diag] = 2.0 * a[diag]

        step = step0 / np.sqrt(t)
        a = project_to_simplex(a - step * g)

        # optional symmetrisation each 2000 iterations (helps stability)
        if t % 2000 == 0:
            a = (a + a[::-1]) * 0.5
            a = project_to_simplex(a)

        if t % 2000 == 0:
            sc = compute_score(a)
            if sc < best_score:
                best_score = sc
                best_a = a.copy()
                print(f"[{time.time():.1f}] sub‑grad improved to {best_score:.6f}")

    sc = compute_score(a)
    if sc < best_score:
        best_score = sc
        best_a = a.copy()
    return best_a, best_score


def lp_refine_step(f: np.ndarray, M: float, mix_weight_max: float = 0.4) -> np.ndarray:
    """
    Linear‑programming refinement.
    Solve   max Σ g_j   s.t.   (f * g)_k ≤ M,   g ≥ 0.
    Afterwards mix f and the normalised LP solution g_norm with an
    optimally chosen weight t ∈ [0, mix_weight_max] that minimises the
    resulting max‑convolution.
    """
    n = f.shape[0]
    m = 2 * n - 1

    # Build full convolution matrix A (A @ g = f * g)
    A = np.zeros((m, n), dtype=np.float64)
    for i in range(n):
        A[i:i + n, i] = f[i]

    # LP: maximise Σ g_j   ⇔ minimise –Σ g_j
    c = -np.ones(n, dtype=np.float64)
    bounds = [(0.0, None) for _ in range(n)]
    b = np.full(m, M, dtype=np.float64)

    res = linprog(c, A_ub=A, b_ub=b, bounds=bounds, method='highs')
    if not res.success:
        return f

    g = np.maximum(res.x, 0.0)
    sum_g = g.sum()
    if sum_g <= 0.0:
        return f

    g_norm = g / sum_g

    # Pre‑compute convolutions needed for the quadratic mixture
    conv_f = convolution_fft(f, f)
    conv_g = convolution_fft(g_norm, g_norm)
    conv_fg = convolution_fft(f, g_norm)

    # Search for the best mixing weight t ∈ [0, mix_weight_max]
    best_t = 0.0
    best_M = float('inf')
    # coarse grid
    for t in np.linspace(0.0, mix_weight_max, 11):
        mixed = ((1 - t) ** 2) * conv_f + 2 * t * (1 - t) * conv_fg + (t ** 2) * conv_g
        cur_M = mixed.max()
        if cur_M < best_M:
            best_M = cur_M
            best_t = t
    # local refinement around the best coarse point
    if 0.0 < best_t < mix_weight_max:
        left = max(0.0, best_t - mix_weight_max / 10)
        right = min(mix_weight_max, best_t + mix_weight_max / 10)
        for t in np.linspace(left, right, 9):
            mixed = ((1 - t) ** 2) * conv_f + 2 * t * (1 - t) * conv_fg + (t ** 2) * conv_g
            cur_M = mixed.max()
            if cur_M < best_M:
                best_M = cur_M
                best_t = t

    # Form the new mixture and re‑project onto the simplex (and symmetrise)
    f_new = (1.0 - best_t) * f + best_t * g_norm
    f_new = (f_new + f_new[::-1]) * 0.5
    f_new = project_to_simplex(f_new)
    return f_new


def hill_climb_sa(start_a: np.ndarray,
                  deadline: float,
                  max_steps: int = 1_500_000) -> tuple[np.ndarray, float]:
    """
    Simple simulated‑annealing hill‑climb on the exact max‑objective.
    """
    n = start_a.size
    cur_a = start_a.copy()
    cur_score = compute_score(cur_a)

    best_a = cur_a.copy()
    best_score = cur_score

    T0 = best_score * 1e-4
    T = T0
    step_scale = 0.12

    for step in range(1, max_steps + 1):
        if time.time() >= deadline:
            break

        cand = cur_a.copy()
        if np.random.rand() < 0.5:               # swap two entries
            i, j = np.random.randint(n, size=2)
            cand[i], cand[j] = cand[j], cand[i]
        else:                                     # mass transfer
            i, j = np.random.randint(n, size=2)
            if i != j:
                delta = step_scale * cur_a[i]
                cand[i] -= delta
                cand[j] += delta
                if cand[i] < 0.0:
                    cand[i] = 0.0

        cand = project_to_simplex(cand)
        cand_score = compute_score(cand)

        delta = cand_score - cur_score
        if delta < 0 or np.random.rand() < np.exp(-delta / (T + 1e-18)):
            cur_a = cand
            cur_score = cand_score
            if cur_score < best_score:
                best_score = cur_score
                best_a = cur_a.copy()
                step_scale = min(step_scale * 1.07, 0.5)
        else:
            step_scale *= 0.97

        # linear cooling
        T = T0 * (1.0 - step / max_steps)

    return best_a, best_score


def symmetrize(a: np.ndarray) -> np.ndarray:
    """Enforce a = (a + a[::-1]) / 2 and re‑project onto the simplex."""
    a = (a + a[::-1]) * 0.5
    return project_to_simplex(a)

def propose_candidate(seed: int = 12345, budget_s: int = 1000, **kwargs):
    """
    Search for a sequence of non‑negative heights that minimises
    `evaluate_sequence`.  The function respects the global wall‑clock
    budget (default 1000 s) and returns a plain Python list of floats.
    """
    np.random.seed(seed)
    start_time = time.time()
    # safety margin for the harness
    deadline = start_time + budget_s - 2.0

    best_seq = None
    best_score = float("inf")

    # ------------------------------------------------------------------
    # 0. Fallback candidate (if supplied by the harness)
    # ------------------------------------------------------------------
    try:
        fb_vec = [float(x) for x in height_sequence_1]  # type: ignore
        fb = evaluate_sequence(fb_vec)
        print(f"[{time.time():.1f}] fallback score = {fb:.6f}")
        if fb < best_score:
            best_score = fb
            best_seq = fb_vec
    except Exception:
        pass

    # ------------------------------------------------------------------
    # 1. Problem size
    # ------------------------------------------------------------------
    n = len(best_seq) if best_seq is not None else 2048

    # ------------------------------------------------------------------
    # 2. Quick analytic scan (single run, cheap)
    # ------------------------------------------------------------------
    if time.time() < deadline - 200.0:
        print(f"[{time.time():.1f}] quick analytic scan (n={n})")
        # explore a modest grid
        for alpha in np.linspace(0.0, 12.0, 13):
            vec = symmetric_product_family(n, alpha)
            sc = compute_score(vec)
            if sc < best_score:
                best_score = sc
                best_seq = vec.tolist()
                print(f"[{time.time():.1f}] new best (sym‑α={alpha:.2f}) = {best_score:.6f}")
        for beta in np.linspace(0.0, 12.0, 13):
            vec = left_heavy_power_family(n, beta)
            sc = compute_score(vec)
            if sc < best_score:
                best_score = sc
                best_seq = vec.tolist()
                print(f"[{time.time():.1f}] new best (left‑β={beta:.2f}) = {best_score:.6f}")
        for gamma in np.linspace(0.0, 8.0, 13):
            vec = exp_u_family(n, gamma)
            sc = compute_score(vec)
            if sc < best_score:
                best_score = sc
                best_seq = vec.tolist()
                print(f"[{time.time():.1f}] new best (exp‑γ={gamma:.2f}) = {best_score:.6f}")
        # a few random starts
        for _ in range(200):
            if np.random.rand() < 0.5:
                vec = random_monotone_start(n)
            else:
                vec = random_u_start(n)
            sc = compute_score(vec)
            if sc < best_score:
                best_score = sc
                best_seq = vec.tolist()
                print(f"[{time.time():.1f}] new best (rand) = {best_score:.6f}")

    # ------------------------------------------------------------------
    # Helper: one full optimisation pipeline
    # ------------------------------------------------------------------
    def pipeline(start_vec: np.ndarray) -> tuple[np.ndarray, float]:
        a = start_vec.copy()
        a = np.maximum(a, 0.0)
        a /= a.sum()
        a = symmetrize(a)

        # --- Adam smooth‑max -------------------------------------------------
        if time.time() < deadline - 60.0:
            a, sc = adam_smooth_optimize(
                a,
                deadline - 30.0,
                max_iters=600_000,
                beta_start=4.0,
                beta_end=2000.0,
                lr=0.02,
            )
            best_local = sc
        else:
            best_local = compute_score(a)

        # --- Projected sub‑gradient descent ---------------------------------
        if time.time() < deadline - 30.0:
            a, sc2 = subgrad_optimize(
                a,
                deadline - 15.0,
                max_iters=2_500_000,
                step0=0.07,
            )
            best_local = min(best_local, sc2)

        # --- LP refinement ----------------------------------------------------
        if time.time() < deadline - 15.0:
            mix_weight_max = 0.4
            while time.time() < deadline - 5.0:
                conv = convolution_fft(a, a)
                M = float(conv.max())
                a_new = lp_refine_step(a, M, mix_weight_max=mix_weight_max)
                sc_new = compute_score(a_new)
                if sc_new < best_local - 1e-9:
                    a = a_new
                    best_local = sc_new
                    # after an improvement we allow a slightly larger step size
                    mix_weight_max = min(0.5, mix_weight_max * 1.05)
                else:
                    mix_weight_max = max(0.05, mix_weight_max * 0.9)
                if mix_weight_max < 0.05:
                    break

        # --- Final symmetrisation --------------------------------------------
        a = symmetrize(a)
        best_local = min(best_local, compute_score(a))

        # --- Simulated‑annealing hill‑climb ----------------------------------
        if time.time() < deadline - 5.0:
            a, sc_sa = hill_climb_sa(a, deadline, max_steps=800_000)
            best_local = min(best_local, sc_sa)

        final_seq = a.tolist()
        final_score = evaluate_sequence(final_seq)
        return final_seq, final_score

    # ------------------------------------------------------------------
    # 3. Multi‑restart loop (as many as time allows)
    # ------------------------------------------------------------------
    while time.time() < deadline - 10.0:
        # Random-start choice
        if best_seq is not None and np.random.rand() < 0.3:
            # perturb the current best
            base = np.array(best_seq, dtype=np.float64)
            noise = np.random.normal(scale=0.015, size=n)
            start = base + noise
            start = np.maximum(start, 0.0)
            if start.sum() == 0.0:
                start = np.full(n, 1.0 / n)
        else:
            r = np.random.rand()
            if r < 0.2:
                start = symmetric_product_family(n, np.random.uniform(0.0, 15.0))
            elif r < 0.4:
                start = left_heavy_power_family(n, np.random.uniform(0.0, 15.0))
            elif r < 0.6:
                start = exp_u_family(n, np.random.uniform(0.0, 8.0))
            elif r < 0.8:
                start = mixed_uniform_u_family(
                    n,
                    np.random.uniform(0.0, 1.0),
                    np.random.uniform(0.0, 8.0),
                )
            else:
                start = random_monotone_start(n)

        seq, scr = pipeline(start)
        if scr < best_score:
            best_score = scr
            best_seq = seq
            print(f"[{time.time():.1f}] GLOBAL new best = {best_score:.6f}")

        # early exit if we have already reached a very good bound
        if best_score <= 1.503:
            print(f"[{time.time():.1f}] target bound reached, terminating early")
            break

    # ------------------------------------------------------------------
    # 4. Final sanity check
    # ------------------------------------------------------------------
    if best_seq is None:
        best_seq = np.full(n, 1.0 / n).tolist()
        best_score = evaluate_sequence(best_seq)
        print(f"[{time.time():.1f}] fallback uniform final score = {best_score:.6f}")

    print(f"\n[{time.time():.1f}] search finished – best reported score = {best_score:.6f}")
    return best_seq
