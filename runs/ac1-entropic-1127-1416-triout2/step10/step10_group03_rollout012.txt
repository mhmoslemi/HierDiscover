from scipy.optimize import linprog
from scipy.linalg import toeplitz

def max_sum_g(f, M):
    # f already normalized sum to 1.
    n = len(f)
    # Build convolution matrix C shape (2n-1, n)
    # first column: f[0] to f[n-1], then zeros (n-1)
    col = np.concatenate([f, np.zeros(n - 1)])
    row = np.concatenate([f[:1], np.zeros(n - 1)])
    C = toeplitz(col, row)  # shape (2n-1, n)
    # constraints: C @ g <= M
    A_ub = C
    b_ub = np.full(2 * n - 1, M + 1e-12)  # small epsilon for numerical tolerance
    # maximize sum(g) = -c @ g minimized for linprog, cos we provide c vector for minimizing.
    c = -np.ones(n)  # minimize -sum(g) i.e., maximize sum(g)
    bounds = [(0, None)] * n
    # Use method='highs' which is default.
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
    if not res.success:
        # fallback: return original f maybe.
        return f.copy()  # no improvement
    g = res.x
    return g

def upsample(vec, new_n):
    old_n = len(vec)
    if new_n == old_n: return vec.copy()
    # Linear interpolation
    x_old = np.arange(old_n)
    x_new = np.linspace(0, old_n-1, new_n)
    new_vec = np.interp(x_new, x_old, vec)
    # ensure non-neg and sum=1
    new_vec = np.maximum(new_vec, 0)
    new_vec /= new_vec.sum()
    return new_vec

"""
Search for a non‑negative step function minimizing
    evaluate_sequence(a) = 2·n·max(conv(a,a))/(sum(a)^2).

Algorithm:
1. Generate a rich set of seed vectors (deterministic shapes,
   Beta‑family, random Dirichlet, sparse spikes) and keep the best.
2. Run a few iterations of a linear‑programming “upper‑bound”
   improvement step on a reduced dimension (n/2).  For fixed f we solve
   a LP that maximises the total mass of a vector g under the constraint
   ‖f∗g‖∞ ≤ ‖f∗f‖∞.  Mixing f and the normalised optimum g strictly
   reduces the peak of the autocorrelation.
3. Upsample the improved small‑size vector to the final length.
4. Refine the result with the smooth soft‑max surrogate gradient descent
   (as in the previous state‑of‑the‑art code) and a final sub‑gradient /
   hill‑climb polishing.
The whole pipeline respects the 1000 s wall‑clock budget.
"""

import time
import numpy as np
from scipy.optimize import linprog
from scipy.linalg import toeplitz

# ----------------------------------------------------------------------
# Low‑level utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full linear convolution using a real FFT."""
    m = a.shape[0] + b.shape[0] - 1
    n_fft = next_pow2(m)
    A = np.fft.rfft(a, n=n_fft)
    B = np.fft.rfft(b, n=n_fft)
    C = A * B
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


def compute_score(a: np.ndarray) -> float:
    """
    Exact objective for a probability vector (sum = 1):
        2·n·max(conv(a,a)).
    """
    n = a.shape[0]
    conv = convolution_fft(a, a)
    return 2.0 * n * float(conv.max())


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of `v` onto the probability simplex
    {x | x_i ≥ 0, Σ x_i = 1}.
    """
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / v.shape[0])

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, v.shape[0] + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.where(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def upsample(vec: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation from len(vec) to new_n and renormalise."""
    old_n = len(vec)
    if new_n == old_n:
        return vec.copy()
    x_old = np.arange(old_n)
    x_new = np.linspace(0, old_n - 1, new_n)
    new_vec = np.interp(x_new, x_old, vec)
    new_vec = np.maximum(new_vec, 0.0)
    s = new_vec.sum()
    if s == 0:
        # avoid division by zero – fallback to uniform
        new_vec = np.full(new_n, 1.0 / new_n)
    else:
        new_vec /= s
    return new_vec


# ----------------------------------------------------------------------
# Seed generators (same as in the previous implementation)
# ----------------------------------------------------------------------


def generate_deterministic(n: int) -> np.ndarray:
    """Sin^2 shape a_i = sin^2(pi·i/(n+1)), normalised to sum 1."""
    i = np.arange(1, n + 1, dtype=float)
    a = np.sin(np.pi * i / (n + 1.0)) ** 2
    a /= a.sum()
    return a


def generate_beta_family(n: int, p: float) -> np.ndarray:
    """
    Symmetric polynomial shape a_i ∝ ((i+1)*(n-i))**p.
    p>0 clusters mass in the centre, p<0 pushes it to the edges.
    """
    i = np.arange(n, dtype=float)
    w = ((i + 1.0) * (n - i)) ** p
    w[w < 0] = 0.0
    a = w / w.sum()
    return a


def random_dirichlet(n: int, conc: float = 1.0) -> np.ndarray:
    """Draw a random probability vector from Dirichlet(conc)."""
    if conc == 1.0:
        a = np.random.rand(n)
    else:
        a = np.random.gamma(conc, 1.0, size=n)
    a /= a.sum()
    return a


def random_spike(n: int, k: int = 70) -> np.ndarray:
    """
    Very sparse vector: choose `k` distinct positions and assign
    random positive weights, then normalise.
    """
    idx = np.random.choice(n, size=min(k, n), replace=False)
    w = np.random.rand(len(idx))
    a = np.zeros(n, dtype=float)
    a[idx] = w
    a /= a.sum()
    return a


# ----------------------------------------------------------------------
# Linear‑programming improvement step (the “M‑LP” from the paper)
# ----------------------------------------------------------------------


def solve_max_sum_lp(f: np.ndarray, M: float, eps: float = 1e-12) -> np.ndarray:
    """
    For a fixed non‑negative vector `f` (sum = 1) and a bound `M`,
    solve the linear program

        max   Σ g_j
        s.t.  (f * g)[k] ≤ M   for all k
              g_j ≥ 0

    and return the optimal `g`.  The LP is built explicitly via the
    convolution matrix.
    """
    n = f.shape[0]
    # Build the linear convolution matrix C such that C @ g = conv(f, g)
    col = np.concatenate([f, np.zeros(n - 1)])
    row = np.concatenate([f[:1], np.zeros(n - 1)])
    C = toeplitz(col, row)               # shape (2n‑1, n)

    A_ub = C
    b_ub = np.full(2 * n - 1, M + eps)   # allow a tiny tolerance
    c = -np.ones(n)                      # linprog minimises, so -sum → maximise sum
    bounds = [(0, None)] * n

    # Use the HiGHS interior‑point solver (fast for dense problems)
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method="highs")
    if not res.success:
        # If the solver fails (should be rare) fall back to the original vector.
        return f.copy()
    g = res.x
    # Numerical cleanup
    g[g < 0] = 0.0
    return g


def lp_improvement_loop(f0: np.ndarray, time_deadline: float,
                        max_iters: int = 30, mix_factor: float = 0.1) -> np.ndarray:
    """
    Perform a short sequence of LP‑based improvements on `f0`.
    The loop stops when the deadline is reached or when `max_iters`
    iterations have been performed.
    """
    f = f0.copy()
    best_f = f.copy()
    best_score = compute_score(f)
    it = 0
    while time.time() < time_deadline and it < max_iters:
        conv_f = convolution_fft(f, f)
        M = conv_f.max()
        g = solve_max_sum_lp(f, M)
        # Normalise g to sum = 1 (the LP does not enforce a total mass)
        if g.sum() == 0:
            break
        g = g / g.sum()
        # Linear interpolation towards g
        f = (1.0 - mix_factor) * f + mix_factor * g
        f = f / f.sum()
        cur_score = compute_score(f)
        if cur_score < best_score:
            best_score = cur_score
            best_f = f.copy()
        it += 1
        if it % 5 == 0:
            print(f"[LP] iteration {it:02d}, best small score {best_score:.6f}")
    return best_f

def softmax_gradient_descent(
    a: np.ndarray,
    start_score: float,
    deadline: float,
    beta_start: float = 5.0,
    beta_end: float = 2000.0,
    step_coef: float = 0.0015,
) -> tuple[np.ndarray, float]:
    """
    Minimise a smooth surrogate of the L∞‑norm via projected gradient
    descent.  The surrogate is Lβ(a)=log Σ exp(β·conv(a,a)).
    """
    n = a.shape[0]
    cur_a = a.copy()
    best_a = cur_a.copy()
    best_score = start_score

    start = time.time()
    it = 0
    while time.time() < deadline:
        # exact objective for bookkeeping
        conv = convolution_fft(cur_a, cur_a)
        max_conv = conv.max()
        cur_score = 2.0 * n * float(max_conv)
        if cur_score < best_score:
            best_score = cur_score
            best_a = cur_a.copy()

        elapsed = time.time() - start
        fraction = elapsed / (deadline - start)
        beta = beta_start + fraction * (beta_end - beta_start)

        shifted = conv - max_conv
        # stable exponentiation: values ≤ 0, exp ≤ 1
        exp_vals = np.exp(beta * shifted)
        w = exp_vals / exp_vals.sum()

        # gradient of the surrogate
        grad = 2.0 * beta * np.correlate(w, cur_a, mode="valid")

        it += 1
        step = step_coef / (np.sqrt(it) * beta)

        nxt = cur_a - step * grad
        nxt = project_to_simplex(nxt)

        cur_a = nxt

        if it % 5000 == 0:
            print(
                f"[GD] it {it:6d} | elapsed {elapsed:6.1f}s | best {best_score:.6f} | β {beta:.1f}"
            )

    return best_a, best_score

# Phase‑3: sub‑gradient refinement (identical to previous implementation)

def subgradient_refine(
    a: np.ndarray,
    start_score: float,
    deadline: float,
    max_iter: int = 500_000,
    step0: float = 0.07,
) -> tuple[np.ndarray, float]:
    """Projected sub‑gradient descent on the true max‑objective."""
    n = a.shape[0]
    cur_a = a.copy()
    best_a = cur_a.copy()
    best_score = start_score

    for t in range(1, max_iter + 1):
        if time.time() >= deadline:
            break

        conv = convolution_fft(cur_a, cur_a)
        k = int(np.argmax(conv))

        idx = np.arange(n)
        j = k - idx
        mask = (j >= 0) & (j < n)
        g = np.zeros_like(cur_a)
        g[mask] = 2.0 * cur_a[j[mask]]

        step = step0 / np.sqrt(t)
        cur_a -= step * g
        cur_a = project_to_simplex(cur_a)

        cur_score = compute_score(cur_a)
        if cur_score < best_score:
            best_score = cur_score
            best_a = cur_a.copy()

        if t % 1000 == 0:
            # occasional jitter to escape shallow plateaus
            jitter = np.random.randn(n) * (step0 * 0.004 / np.sqrt(t))
            cur_a += jitter
            cur_a = project_to_simplex(cur_a)

    return best_a, best_score


# ----------------------------------------------------------------------
# Phase‑4: greedy hill‑climb polishing
# ----------------------------------------------------------------------


def hill_climb(
    a: np.ndarray,
    deadline: float,
    max_steps: int = 800_000,
) -> tuple[np.ndarray, float]:
    """Stochastic moves that are kept only when they improve the exact score."""
    n = a.shape[0]
    best_a = a.copy()
    best_score = compute_score(best_a)
    step_scale = 0.08

    steps = 0
    while steps < max_steps and time.time() < deadline:
        cand = best_a.copy()
        if np.random.rand() < 0.5:
            # swap two entries
            i, j = np.random.randint(n, size=2)
            cand[i], cand[j] = cand[j], cand[i]
        else:
            # transfer a fraction from i to j
            i, j = np.random.randint(n, size=2)
            if cand[i] > 0.0 and i != j:

"""
Search for a non‑negative step function that minimizes

    evaluate_sequence(a) = 2·n·max(conv(a,a)) / (sum(a)^2)

The algorithm combines three ideas:

1. **Diverse seeding** – deterministic shapes, a Beta family, random
   Dirichlet draws and very sparse spikes.  The best seed (exact objective)
   is kept.

2. **LP‑based upper‑bound reduction** – on a reduced dimension (≈ n/2)
   we solve, for the current vector `f`, the linear programme

        max   Σ g_j
        s.t.  (f * g)[k] ≤ M   ∀k   (M = max(f∗f))
              g_j ≥ 0

   The optimal `g` carries as much mass as possible while respecting the
   same autocorrelation bound `M`.  Mixing `f` and the normalised `g`
   (`f ← (1‑t)·f + t·g`) strictly lowers the peak of the autocorrelation.
   A handful of such iterations already yields a noticeably better shape.

3. **Continuous refinement** – the improved vector is up‑sampled to the
   final length and polished with the smooth soft‑max surrogate gradient
   descent, a classic sub‑gradient step and a stochastic hill‑climb.
   All phases respect the 1000 s wall‑clock budget.

The final sequence (a plain ``list[float]``) is returned.
"""

import time
import numpy as np
from scipy.optimize import linprog
from scipy.linalg import toeplitz

# ----------------------------------------------------------------------
# Low‑level utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full linear convolution using a real FFT."""
    m = a.shape[0] + b.shape[0] - 1
    n_fft = next_pow2(m)
    A = np.fft.rfft(a, n=n_fft)
    B = np.fft.rfft(b, n=n_fft)
    C = A * B
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


def compute_score(a: np.ndarray) -> float:
    """
    Exact objective for a probability vector (sum = 1):
        2·n·max(conv(a,a)).
    """
    n = a.shape[0]
    conv = convolution_fft(a, a)
    return 2.0 * n * float(conv.max())


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of `v` onto the probability simplex
    {x | x_i ≥ 0, Σ x_i = 1}.
    """
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / v.shape[0])

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, v.shape[0] + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.where(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def upsample(vec: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation from len(vec) to new_n and renormalise."""
    old_n = len(vec)
    if new_n == old_n:
        return vec.copy()
    x_old = np.arange(old_n)
    x_new = np.linspace(0, old_n - 1, new_n)
    new_vec = np.interp(x_new, x_old, vec)
    new_vec = np.maximum(new_vec, 0.0)
    s = new_vec.sum()
    if s == 0:
        new_vec = np.full(new_n, 1.0 / new_n)
    else:
        new_vec /= s
    return new_vec


# ----------------------------------------------------------------------
# Seed generators
# ----------------------------------------------------------------------


def generate_deterministic(n: int) -> np.ndarray:
    """Sin² shape a_i = sin²(pi·i/(n+1)), normalised to sum 1."""
    i = np.arange(1, n + 1, dtype=float)
    a = np.sin(np.pi * i / (n + 1.0)) ** 2
    a /= a.sum()
    return a


def generate_beta_family(n: int, p: float) -> np.ndarray:
    """
    Symmetric polynomial shape a_i ∝ ((i+1)*(n-i))**p.
    p>0 clusters mass in the centre, p<0 pushes it to the edges.
    """
    i = np.arange(n, dtype=float)
    w = ((i + 1.0) * (n - i)) ** p
    w[w < 0] = 0.0
    a = w / w.sum()
    return a


def random_dirichlet(n: int, conc: float = 1.0) -> np.ndarray:
    """Random probability vector from Dirichlet(conc)."""
    if conc == 1.0:
        a = np.random.rand(n)
    else:
        a = np.random.gamma(conc, 1.0, size=n)
    a /= a.sum()
    return a


def random_spike(n: int, k: int = 70) -> np.ndarray:
    """Very sparse vector: k distinct positions with random positive weights."""
    idx = np.random.choice(n, size=min(k, n), replace=False)
    w = np.random.rand(len(idx))
    a = np.zeros(n, dtype=float)
    a[idx] = w
    a /= a.sum()
    return a


# ----------------------------------------------------------------------
# LP‑based improvement (the “M‑LP” step)
# ----------------------------------------------------------------------


def solve_max_sum_lp(f: np.ndarray, M: float, eps: float = 1e-12) -> np.ndarray:
    """
    For a fixed non‑negative vector `f` (∑f=1) and a bound `M`,
    solve

        max   Σ g_j
        s.t.  (f * g)[k] ≤ M  (∀k)
              g_j ≥ 0

    and return the optimal `g`.  The convolution matrix
    C with C·g = conv(f,g) is built explicitly via a Toeplitz matrix.
    """
    n = f.shape[0]
    col = np.concatenate([f, np.zeros(n - 1)])
    row = np.concatenate([f[:1], np.zeros(n - 1)])
    C = toeplitz(col, row)                # (2n‑1) × n

    A_ub = C
    b_ub = np.full(2 * n - 1, M + eps)    # tiny tolerance for numerical safety
    c = -np.ones(n)                       # linprog minimises ⇒ –sum → maximise sum
    bounds = [(0, None)] * n

    res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method="highs")
    if not res.success:
        # Very unlikely, but fall back to the current vector.
        return f.copy()
    g = res.x
    g[g < 0] = 0.0
    return g


def lp_improvement_loop(f0: np.ndarray, time_deadline: float,
                        max_iters: int = 25, mix_factor: float = 0.1) -> np.ndarray:
    """
    Perform a short sequence of LP‑based improvements on `f0`.
    Stops when the deadline is reached or `max_iters` is hit.
    """
    f = f0.copy()
    best_f = f.copy()
    best_score = compute_score(f)
    it = 0
    while time.time() < time_deadline and it < max_iters:
        conv_f = convolution_fft(f, f)
        M = conv_f.max()
        g = solve_max_sum_lp(f, M)
        if g.sum() == 0:
            break
        g = g / g.sum()
        f = (1.0 - mix_factor) * f + mix_factor * g
        f = f / f.sum()
        cur_score = compute_score(f)
        if cur_score < best_score:
            best_score = cur_score
            best_f = f.copy()
        it += 1
        if it % 5 == 0:
            print(f"[LP] iteration {it:02d}, best small score {best_score:.6f}")
    return best_f


# ----------------------------------------------------------------------
# Soft‑max gradient descent (smooth surrogate of the L∞ norm)
# ----------------------------------------------------------------------


def softmax_gradient_descent(
    a: np.ndarray,
    start_score: float,
    deadline: float,
    beta_start: float = 5.0,
    beta_end: float = 3000.0,
    step_coef: float = 0.0015,
) -> tuple[np.ndarray, float]:
    """
    Minimise Lβ(a)=log Σ exp(β·conv(a,a)) via projected gradient descent.
    The temperature β is annealed linearly; after each step the iterate is
    projected onto the simplex.
    """
    n = a.shape[0]
    cur_a = a.copy()
    best_a = cur_a.copy()
    best_score = start_score
    start = time.time()
    it = 0
    while time.time() < deadline:
        conv = convolution_fft(cur_a, cur_a)
        max_conv = conv.max()
        cur_score = 2.0 * n * float(max_conv)
        if cur_score < best_score:
            best_score = cur_score
            best_a = cur_a.copy()

        elapsed = time.time() - start
        fraction = elapsed / (deadline - start)
        beta = beta_start + fraction * (beta_end - beta_start)

        shifted = conv - max_conv
        exp_vals = np.exp(beta * shifted)          # ≤ 1, stable
        w = exp_vals / exp_vals.sum()

        grad = 2.0 * beta * np.correlate(w, cur_a, mode="valid")

        it += 1
        step = step_coef / (np.sqrt(it) * beta)

        nxt = cur_a - step * grad
        nxt = project_to_simplex(nxt)
        cur_a = nxt

        if it % 5000 == 0:
            print(
                f"[GD] it {it:6d} | elapsed {elapsed:6.1f}s | best {best_score:.6f} | β {beta:.1f}"
            )
    return best_a, best_score


# ----------------------------------------------------------------------
# Sub‑gradient refinement (identical to the previous implementation)
# ----------------------------------------------------------------------


def subgradient_refine(
    a: np.ndarray,
    start_score: float,
    deadline: float,
    max_iter: int = 500_000,
    step0: float = 0.07,
) -> tuple[np.ndarray, float]:
    """Projected sub‑gradient descent on the true max‑objective."""
    n = a.shape[0]
    cur_a = a.copy()
    best_a = cur_a.copy()
    best_score = start_score

    for t in range(1, max_iter + 1):
        if time.time() >= deadline:
            break

        conv = convolution_fft(cur_a, cur_a)
        k = int(np.argmax(conv))

        idx = np.arange(n)
        j = k - idx
        mask = (j >= 0) & (j < n)
        g = np.zeros_like(cur_a)
        g[mask] = 2.0 * cur_a[j[mask]]

        step = step0 / np.sqrt(t)
        cur_a -= step * g
        cur_a = project_to_simplex(cur_a)

        cur_score = compute_score(cur_a)
        if cur_score < best_score:
            best_score = cur_score
            best_a = cur_a.copy()

        if t % 1000 == 0:
            jitter = np.random.randn(n) * (step0 * 0.004 / np.sqrt(t))
            cur_a += jitter
            cur_a = project_to_simplex(cur_a)

    return best_a, best_score


# ----------------------------------------------------------------------
# Greedy hill‑climb polishing
# ----------------------------------------------------------------------


def hill_climb(
    a: np.ndarray,
    deadline: float,
    max_steps: int = 1_200_000,
) -> tuple[np.ndarray, float]:
    """Stochastic moves that are kept only when they improve the exact score."""
    n = a.shape[0]
    best_a = a.copy()
    best_score = compute_score(best_a)
    step_scale = 0.08
    steps = 0
    while steps < max_steps and time.time() < deadline:
        cand = best_a.copy()
        if np.random.rand() < 0.5:
            i, j = np.random.randint(n, size=2)
            cand[i], cand[j] = cand[j], cand[i]
        else:
            i, j = np.random.randint(n, size=2)
            if cand[i] > 0.0 and i != j:
                delta = step_scale * cand[i]
                cand[i] -= delta
                cand[j] += delta
                if cand[i] < 0.0:
                    cand[i] = 0.0
        cand /= cand.sum()
        cur_score = compute_score(cand)

        if cur_score < best_score:
            best_score = cur_score
            best_a = cand
            step_scale = min(step_scale * 1.07, 0.5)
        else:
            step_scale *= 0.96

        steps += 1

    return best_a, best_score
