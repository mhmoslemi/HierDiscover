The required sequence is the minimiser of  

\[
J(a)=\frac{2\,n\;\max_k (a*a)_k}{\bigl(\sum_i a_i\bigr)^2},
\qquad a_i\ge 0 ,
\]

where “\(*\)” is the ordinary (zero‑padded) convolution.  
The objective is scale‑invariant, therefore we can (and do) keep the
vector on the probability simplex \(\sum_i a_i=1\).  
The algorithm below improves the previous state‑of‑the‑art by

* fixing a subtle bug in the sub‑gradient implementation (the diagonal
  terms were halved);
* exploring a richer family of left‑heavy power‑law shapes and a
  two‑sided mixture;
* performing a much broader parameter sweep in the fast analytic
  **quick‑scan**;
* running the smooth‑max Adam optimiser for a larger \(\beta\) and
  more iterations;
* applying a longer projected sub‑gradient descent with the corrected
  gradient;
* iterating the LP‑refinement more often and with an adaptive mixing
  weight;
* doing a longer simulated‑annealing hill‑climb at the end.

All phases respect the global wall‑clock budget (default 1000 s).  
The final sequence is returned as a plain Python list ready for
`evaluate_sequence`.

```python
"""
propose_candidate – search for a non‑negative step function that minimises

        2·n·max(convolve(a,a)) / (sum(a)**2)

under the simplex constraint.

The algorithm consists of

1.   a quick analytic / random scan over many parametrised families;
2.   Adam optimisation of a smooth‑max surrogate (β up to 2000);
3.   a long projected sub‑gradient descent (fixed diagonal gradient);
4.   several LP‑refinement steps with an adaptive mixing weight;
5.   optional symmetrisation;
6.   a final simulated‑annealing hill‑climb.

The total wall‑clock time is limited to the supplied budget (≈ 1000 s).
The returned list can be fed directly to ``evaluate_sequence``.
"""

import time
import numpy as np
from scipy.optimize import linprog

# ----------------------------------------------------------------------
# Numerical utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full (aperiodic) convolution using a real FFT."""
    m = a.size + b.size - 1
    nfft = next_pow2(m)
    A = np.fft.rfft(a, n=nfft)
    B = np.fft.rfft(b, n=nfft)
    C = A * B
    conv = np.fft.irfft(C, n=nfft)[:m]
    return conv


def compute_score(a: np.ndarray) -> float:
    """
    Exact objective when the vector is normalised to sum‑one.
    Returns 2·n·max(conv(a,a)).
    """
    n = a.shape[0]
    conv = convolution_fft(a, a)
    return 2.0 * n * float(conv.max())


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / v.shape[0])

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, v.shape[0] + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.where(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def softmax(z: np.ndarray) -> np.ndarray:
    """Stable soft‑max."""
    z_max = np.max(z)
    e = np.exp(z - z_max)
    return e / e.sum()


# ----------------------------------------------------------------------
# Parametric families for the quick scan
# ----------------------------------------------------------------------


def left_heavy_power_family(n: int, beta: float) -> np.ndarray:
    """a_i ∝ (i+0.5)^(-beta)  (mass concentrated on the left)."""
    i = np.arange(n, dtype=np.float64)
    w = np.exp(-beta * np.log(i + 0.5))
    a = w / w.sum()
    return a


def two_sided_power_mix(
    n: int, beta_left: float, beta_right: float, lam: float = 0.5
) -> np.ndarray:
    """
    Convex combination of a left‑heavy and a right‑heavy power law.
    lam controls the weight of the left part.
    """
    left = left_heavy_power_family(n, beta_left)
    right = left_heavy_power_family(n, beta_right)[::-1]
    a = lam * left + (1.0 - lam) * right
    a /= a.sum()
    return a


def random_monotone_start(n: int) -> np.ndarray:
    """Random decreasing sequence (used as a heuristic start)."""
    r = np.random.rand(n) ** np.random.uniform(0.5, 5.0)
    r.sort()
    a = r[::-1]
    a /= a.sum()
    return a


def random_u_start(n: int) -> np.ndarray:
    """Symmetric exponential “U‑shape”. """
    gamma = np.random.uniform(0.0, 6.0)
    i = np.arange(n, dtype=np.float64)
    dist = np.minimum(i, n - 1 - i)
    w = np.exp(-gamma * dist)
    a = w / w.sum()
    return a


# ----------------------------------------------------------------------
# Adam optimisation of the smooth‑max surrogate
# ----------------------------------------------------------------------


def adam_smooth_optimize(
    start_a: np.ndarray,
    deadline: float,
    max_iters: int = 500_000,
    beta_start: float = 4.0,
    beta_end: float = 2000.0,
    lr: float = 0.018,
) -> tuple[np.ndarray, float]:
    """
    Minimise the smooth surrogate

        f_β(a) = (1/β)·log Σ_k exp(β·(a*a)_k)

    using Adam on the soft‑max parametrisation a = softmax(z).
    The smoothing parameter β is increased exponentially from
    ``beta_start`` to ``beta_end``.
    """
    eps = 1e-12
    z = np.log(start_a + eps)

    m = np.zeros_like(z)
    v = np.zeros_like(z)
    beta1 = 0.9
    beta2 = 0.999
    eps_adam = 1e-12

    best_a = softmax(z)
    best_score = compute_score(best_a)

    for it in range(1, max_iters + 1):
        if time.time() >= deadline:
            break

        # exponential schedule for β
        t = it / max_iters
        beta = beta_start * (beta_end / beta_start) ** t

        a = softmax(z)
        conv = convolution_fft(a, a)
        maxc = conv.max()
        # smooth‑max weights
        exp_w = np.exp(beta * (conv - maxc))
        w = exp_w / exp_w.sum()

        # gradient wrt a  (see derivation in the original script)
        a_rev = a[::-1]
        conv_rev = convolution_fft(a_rev, w)
        grad_a = 2.0 * conv_rev[len(a) - 1 : len(a) - 1 + len(a)]

        # back‑prop through soft‑max
        dot_ag = np.dot(a, grad_a)
        grad_z = a * grad_a - dot_ag * a

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad_z
        v = beta2 * v + (1.0 - beta2) * (grad_z ** 2)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        z -= lr * m_hat / (np.sqrt(v_hat) + eps_adam)

        if it % 500 == 0:
            a_tmp = softmax(z)
            sc = compute_score(a_tmp)
            if sc < best_score:
                best_score, best_a = sc, a_tmp.copy()
                print(f"[{time.time():.1f}] Adam improved to {best_score:.6f}")

    # final check
    a_fin = softmax(z)
    sc_fin = compute_score(a_fin)
    if sc_fin < best_score:
        best_score, best_a = sc_fin, a_fin.copy()
    return best_a, best_score


# ----------------------------------------------------------------------
# Projected sub‑gradient descent (fixed diagonal term)
# ----------------------------------------------------------------------


def subgrad_optimize(
    start_a: np.ndarray,
    deadline: float,
    max_iters: int = 3_000_000,
    step0: float = 0.07,
) -> tuple[np.ndarray, float]:
    """
    Sub‑gradient method for the exact objective.
    The sub‑gradient of   max_k (a*a)_k  at the active shift k* is

        g_j = 2·a_{k*‑j}  (zero if the index is out of range).

    The diagonal term is *not* halved – this was a bug in the
    previous implementation.
    """
    n = start_a.shape[0]
    a = start_a.astype(np.float64).copy()

    best_a = a.copy()
    best_score = compute_score(a)

    for t in range(1, max_iters + 1):
        if time.time() >= deadline:
            break

        conv = convolution_fft(a, a)
        k = int(np.argmax(conv))

        idx = np.arange(n)
        j = k - idx
        mask = (j >= 0) & (j < n)
        g = np.zeros_like(a)
        g[mask] = 2.0 * a[j[mask]]          # correct diagonal (2·a_j)

        step = step0 / np.sqrt(t)
        a = project_to_simplex(a - step * g)

        if t % 2000 == 0:
            sc = compute_score(a)
            if sc < best_score:
                best_score, best_a = sc, a.copy()
                print(f"[{time.time():.1f}] sub‑grad improved to {best_score:.6f}")

    sc = compute_score(a)
    if sc < best_score:
        best_score, best_a = sc, a.copy()
    return best_a, best_score


# ----------------------------------------------------------------------
# Linear‑programming refinement (state‑of‑the‑art step)
# ----------------------------------------------------------------------


def lp_refine_step(f: np.ndarray, M: float, mix_weight: float = 0.2) -> np.ndarray:
    """
    Solve   max Σ g_j   s.t.   (f * g)_k ≤ M   (for all k),
    g ≥ 0.  Return the convex mixture
        f_new = (1‑t)·f + t·g_norm
    with a given mixing weight t.
    """
    n = f.shape[0]
    m = 2 * n - 1

    # build the Toeplitz‑like convolution matrix
    A = np.zeros((m, n), dtype=np.float64)
    for i in range(n):
        A[i : i + n, i] = f[i]

    c = -np.ones(n, dtype=np.float64)          # maximise Σ g  → minimise −Σ g
    bounds = [(0.0, None) for _ in range(n)]
    b = np.full(m, M, dtype=np.float64)

    res = linprog(
        c,
        A_ub=A,
        b_ub=b,
        bounds=bounds,
        method="highs",
        options={"presolve": True, "time_limit": 30},
    )

    if not res.success:
        return f

    g = np.maximum(res.x, 0.0)
    sum_g = g.sum()
    if sum_g <= 0.0:
        return f
    g_norm = g / sum_g

    f_new = (1.0 - mix_weight) * f + mix_weight * g_norm
    f_new = project_to_simplex(f_new)
    return f_new


# ----------------------------------------------------------------------
# Simulated‑annealing hill‑climb
# ----------------------------------------------------------------------


def hill_climb_sa(
    start_a: np.ndarray,
    deadline: float,
    max_steps: int = 2_500_000,
) -> tuple[np.ndarray, float]:
    """Random walk on the simplex with a slow cooling schedule."""
    n = start_a.size
    cur_a = start_a.copy()
    cur_score = compute_score(cur_a)

    best_a = cur_a.copy()
    best_score = cur_score

    T0 = best_score * 1e-4
    T = T0
    step_scale = 0.12

    for step in range(1, max_steps + 1):
        if time.time() >= deadline:
            break

        cand = cur_a.copy()
        if np.random.rand() < 0.5:               # swap two coordinates
            i, j = np.random.randint(n, size=2)
            cand[i], cand[j] = cand[j], cand[i]
        else:                                     # move a small amount of mass
            i, j = np.random.randint(n, size=2)
            if i != j:
                delta = step_scale * cur_a[i]
                cand[i] -= delta
                cand[j] += delta
                if cand[i] < 0.0:
                    cand[i] = 0.0

        cand = project_to_simplex(cand)
        cand_score = compute_score(cand)

        delta = cand_score - cur_score
        if delta < 0 or np.random.rand() < np.exp(-delta / (T + 1e-18)):
            cur_a, cur_score = cand, cand_score
            if cur_score < best_score:
                best_score, best_a = cur_score, cur_a.copy()
                step_scale = min(step_scale * 1.07, 0.5)
        else:
            step_scale *= 0.97

        # linear cooling
        T = T0 * (1.0 - step / max_steps)

    return best_a, best_score

# Main driver
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 12345, budget_s: int = 1000, **kwargs):
    """
    Search for a sequence of non‑negative heights that minimises the
    given evaluation function.  The routine respects the supplied time
    budget (≈ 1000 s) and returns a plain list of floats.
    """
    np.random.seed(seed)
    start_time = time.time()
    # small safety margin for the outer harness
    deadline = start_time + budget_s - 2.0

    best_seq = None
    best_score = float("inf")

    # ------------------------------------------------------------------
    # 0.  Fallback candidate (if supplied by the harness)
    # ------------------------------------------------------------------
    try:
        fb_vec = [float(x) for x in height_sequence_1]  # type: ignore
        fb = evaluate_sequence(fb_vec)
        print(f"[{time.time():.1f}] fallback score = {fb:.6f}")
        if fb < best_score:
            best_score, best_seq = fb, fb_vec
    except Exception:
        pass

    # ------------------------------------------------------------------
    # 1.  Problem size (keep the fallback length if available)
    # ------------------------------------------------------------------
    n = len(best_seq) if best_seq is not None else 2048

    # ------------------------------------------------------------------
    # 2.  Quick analytic / random scan (≈ 200 s)
    # ------------------------------------------------------------------
    if time.time() < deadline - 200.0:
        print(f"[{time.time():.1f}] quick scan on n={n}")

        # --- explore a wider range of exponents for the left‑heavy family ---
        for beta in np.concatenate(
            (np.linspace(0.0, 12.0, 13), np.linspace(13.0, 30.0, 10))
        ):
            if time.time() >= deadline - 30.0:
                break
            vec = left_heavy_power_family(n, beta)
            sc = compute_score(vec)
            if sc < best_score:
                best_score, best_seq = sc, vec.tolist()
                print(f"[{time.time():.1f}] new best (β={beta:.2f}) = {best_score:.6f}")

        # --- two‑sided mixtures (adds a right‑hand tail) ---
        for betaL in [4.0, 8.0, 12.0, 20.0]:
            for betaR in [4.0, 8.0, 12.0, 20.0]:
                for lam in [0.3, 0.5, 0.7]:
                    if time.time() >= deadline - 30.0:
                        break
                    vec = two_sided_power_mix(n, betaL, betaR, lam)
                    sc = compute_score(vec)
                    if sc < best_score:
                        best_score, best_seq = sc, vec.tolist()
                        print(
                            f"[{time.time():.1f}] new best (mix βL={betaL:.1f},βR={betaR:.1f},λ={lam:.2f}) = {best_score:.6f}"
                        )

        # --- random monotone starts (more samples) ---
        for _ in range(400):
            if time.time() >= deadline - 30.0:
                break
            vec = random_monotone_start(n) if np.random.rand() < 0.5 else random_u_start(n)
            sc = compute_score(vec)
            if sc < best_score:
                best_score, best_seq = sc, vec.tolist()
                print(f"[{time.time():.1f}] new best (random) = {best_score:.6f}")

    # ------------------------------------------------------------------
    # Normalise the current best candidate (guarantee sum = 1)
    # ------------------------------------------------------------------
    if best_seq is None:
        a = np.full(n, 1.0 / n, dtype=np.float64)
    else:
        a = np.array(best_seq, dtype=np.float64)
    a = np.maximum(a, 0.0)
    a /= a.sum()
    best_seq = a.tolist()
    best_score = compute_score(a)

    # ------------------------------------------------------------------
    # 3. Adam‑based smooth‑max refinement (≈ 180 s)
    # ------------------------------------------------------------------
    if time.time() < deadline - 180.0:
        print(f"[{time.time():.1f}] starting Adam refinement")
        a_opt, sc_opt = adam_smooth_optimize(
            a,
            deadline - 60.0,
            max_iters=500_000,
            beta_start=4.0,
            beta_end=2000.0,
            lr=0.018,
        )
        if sc_opt < best_score:
            best_score, a = sc_opt, a_opt
            print(f"[{time.time():.1f}] after Adam best = {best_score:.6f}")

    # ------------------------------------------------------------------
    # 4. Long projected sub‑gradient descent (≈ 120 s)
    # ------------------------------------------------------------------
    if time.time() < deadline - 120.0:
        print(f"[{time.time():.1f}] starting sub‑gradient descent")
        a_opt, sc_opt = subgrad_optimize(
            a,
            deadline - 30.0,
            max_iters=4_000_000,
            step0=0.07,
        )
        if sc_opt < best_score:
            best_score, a = sc_opt, a_opt
            print(f"[{time.time():.1f}] after sub‑grad best = {best_score:.6f}")

    # ------------------------------------------------------------------
    # 5. LP‑based refinement loop (≈ 30 s)
    # ------------------------------------------------------------------
    if time.time() < deadline - 30.0:
        print(f"[{time.time():.1f}] starting LP refinement")
        mix_weight = 0.2
        lp_it = 0
        while time.time() < deadline - 5.0 and lp_it < 20:
            conv = convolution_fft(a, a)
            M = float(conv.max())
            a_new = lp_refine_step(a, M, mix_weight=mix_weight)
            sc = compute_score(a_new)
            if sc + 1e-8 < best_score:
                best_score, a = sc, a_new
                print(f"[{time.time():.1f}] LP step {lp_it} improved to {best_score:.6f}")
                mix_weight = 0.2  # reset after an improvement
            else:
                mix_weight = max(0.05, mix_weight * 0.9)  # become more conservative
            lp_it += 1

    # ------------------------------------------------------------------
    # 6. Optional symmetrisation
    # ------------------------------------------------------------------
    if best_seq is not None:
        a_sym = (a + a[::-1]) * 0.5
        a_sym = project_to_simplex(a_sym)
        sc_sym = compute_score(a_sym)
        if sc_sym < best_score:
            best_score, a = sc_sym, a_sym
            print(f"[{time.time():.1f}] symmetrisation improved best = {best_score:.6f}")

    # ------------------------------------------------------------------
    # 7. Final simulated‑annealing hill‑climb (≈ 30 s)
    # ------------------------------------------------------------------
    if time.time() < deadline - 5.0:
        print(f"[{time.time():.1f}] final hill‑climb (SA)")
        a_opt, sc_opt = hill_climb_sa(a, deadline, max_steps=2_500_000)
        if sc_opt < best_score:
            best_score, a = sc_opt, a_opt
            print(f"[{time.time():.1f}] hill‑climb improved best = {best_score:.6f}")

    # ------------------------------------------------------------------
    # 8. Final official evaluation and return
    # ------------------------------------------------------------------
    final_seq = a.tolist()
    final_score = evaluate_sequence(final_seq)
    print(
        f"\n[{time.time():.1f}] finished search – final reported score = {final_score:.6f}"
    )
    return final_seq
```
