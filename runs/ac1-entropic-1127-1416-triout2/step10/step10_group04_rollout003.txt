"""
Non‑negative step‑function optimiser for ``evaluate_sequence`` (improved).

Algorithm
---------
1. Choose the problem size n (use the provided ``height_sequence_1`` if
   available, otherwise 2048).  All candidates are projected onto the
   probability simplex (∑a_i = 1, a_i ≥ 0).

2. **Analytic & parametric families** – generate a rich collection of
   candidates:
   * uniform, central‑binomial, an “arcsine” shape, geometric,
   * power‑law, a damped central‑binomial (central‑binomial·r^i),
   * mixtures of a power‑law and a geometric law.
   A small grid over the parameters (p, r, α) is explored and the best
   candidate is kept.

3. **Smooth Adam optimisation** – minimise a log‑sum‑exp smooth
   approximation of the max‑convolution objective.  The temperature β is
   annealed from a modest start (≈5) to a very large end (≈2000) so that
   the surrogate tightly follows the true max.  Multiple random restarts
   (jittered copies of the current best) are performed; each run runs as
   long as the remaining wall‑clock time permits.

4. **Projected sub‑gradient polishing** – a classic sub‑gradient descent
   on the exact max‑convolution objective to remove any remaining slack.

5. **Stochastic hill‑climb** – tiny random mass transfers (or swaps)
   between positions, always projected back onto the simplex.  The step
   size is dynamically adapted; the procedure runs until the time budget
   is exhausted.

6. The best sequence found is finally evaluated with the supplied
   ``evaluate_sequence`` and returned as a plain Python list.

All phases respect a 1000 s wall‑clock budget (a 3 s safety margin is
kept).  The implementation uses only ``numpy`` and the standard library,
so it runs on a single CPU core.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Core utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """
    Full linear convolution of real vectors via FFT.
    Returns an array of length len(a)+len(b)-1.
    """
    m = a.shape[0] + b.shape[0] - 1
    n_fft = next_pow2(m)
    A = np.fft.rfft(a, n=n_fft)
    B = np.fft.rfft(b, n=n_fft)
    C = A * B
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


def compute_score(a: np.ndarray) -> float:
    """
    Exact objective for a vector already on the simplex (∑a = 1, a ≥ 0):
        score = 2·n·max(conv(a,a))
    """
    n = a.shape[0]
    conv = convolution_fft(a, a)
    return 2.0 * n * float(conv.max())


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of v onto the probability simplex:
        sum(v) = 1, v >= 0.
    """
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / v.shape[0])

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, v.shape[0] + 1)
    cond = u - cssv / ind > 0
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


# ----------------------------------------------------------------------
# Analytic / parametric families
# ----------------------------------------------------------------------


def generate_uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def generate_central_binomial(n: int) -> np.ndarray:
    """a_i = C(2i,i) / 4^i  (monotone decreasing)."""
    a = np.empty(n, dtype=np.float64)
    a[0] = 1.0
    for i in range(1, n):
        a[i] = a[i - 1] * (2.0 * i - 1) / (2.0 * i)
    return project_to_simplex(a)


def generate_central_binomial_mod(n: int, r: float) -> np.ndarray:
    """
    Central‑binomial shape with exponential damping:
        a_i ∝ C(2i,i) / 4^i * r**i
    """
    a = generate_central_binomial(n)
    powers = r ** np.arange(n, dtype=np.float64)
    a = a * powers
    return project_to_simplex(a)


def generate_powerlaw(n: int, p: float) -> np.ndarray:
    """a_i ∝ (i+1)^(-p)."""
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = idx ** (-p)
    return project_to_simplex(a)


def generate_geometric(n: int, r: float) -> np.ndarray:
    """Geometric distribution (1‑r)·r^i, already normalised for r<1."""
    i = np.arange(n, dtype=np.float64)
    a = (1.0 - r) * (r ** i)
    return project_to_simplex(a)


def generate_arcsine(n: int) -> np.ndarray:
    """
    Discrete arcsine (Beta(½,½)) shape:
        a_i ∝ 1 / sqrt( (i+½)·(n‑½‑i) )
    """
    i = np.arange(n, dtype=np.float64)
    a = 1.0 / np.sqrt((i + 0.5) * (n - 0.5 - i))
    return project_to_simplex(a)


def generate_mixture(n: int, p: float, r: float, alpha: float) -> np.ndarray:
    """
    Convex mixture α·powerlaw(p) + (1‑α)·geometric(r).
    """
    a_pw = generate_powerlaw(n, p)
    a_geo = generate_geometric(n, r)
    a = alpha * a_pw + (1.0 - alpha) * a_geo
    return project_to_simplex(a)


# ----------------------------------------------------------------------
# Smooth (log‑sum‑exp) Adam optimiser
# ----------------------------------------------------------------------


def smooth_adam_optimize(
    start_a: np.ndarray,
    time_deadline: float,
    max_iters: int = 400_000,
    beta_start: float = 5.0,
    beta_end: float = 2000.0,
    step0: float = 0.9,
    adam_beta1: float = 0.9,
    adam_beta2: float = 0.999,
    adam_eps: float = 1e-10,
) -> tuple[np.ndarray, float]:
    """
    Adam optimiser for the smoothed max‑convolution objective.
    Uses a log‑sum‑exp surrogate with temperature β linearly
    (in log‑scale) annealed from ``beta_start`` to ``beta_end``.
    """
    a = start_a.astype(np.float64).copy()
    n = a.shape[0]
    m1 = np.zeros_like(a)
    m2 = np.zeros_like(a)

    best_a = a.copy()
    best_score = compute_score(a)

    for it in range(1, max_iters + 1):
        if time.time() >= time_deadline:
            break

        # anneal temperature (log‑linear schedule)
        t_frac = it / max_iters
        beta = beta_start * (beta_end / beta_start) ** t_frac

        # convolution a * a
        conv = convolution_fft(a, a)
        maxc = conv.max()

        # soft‑max weights (stable)
        exp_w = np.exp(beta * (conv - maxc))
        w = exp_w / exp_w.sum()

        # gradient of the smoothed objective:
        rev_a = a[::-1]
        grad_full = convolution_fft(rev_a, w)          # length 3n‑2
        grad = 2.0 * grad_full[n - 1:n - 1 + n]        # slice back to length n

        # Adam update
        m1 = adam_beta1 * m1 + (1.0 - adam_beta1) * grad
        m2 = adam_beta2 * m2 + (1.0 - adam_beta2) * (grad * grad)
        m1_hat = m1 / (1.0 - adam_beta1 ** it)
        m2_hat = m2 / (1.0 - adam_beta2 ** it)

        step = step0 / np.sqrt(it)                     # simple decay
        a_new = a - step * m1_hat / (np.sqrt(m2_hat) + adam_eps)
        a_new = project_to_simplex(a_new)

        a = a_new

        # occasional exact evaluation
        if it % 800 == 0:
            cur_score = compute_score(a)
            if cur_score < best_score:
                best_score = cur_score
                best_a = a.copy()

    # final check
    final_score = compute_score(a)
    if final_score < best_score:
        best_score, best_a = final_score, a.copy()
    return best_a, best_score


# ----------------------------------------------------------------------
# Sub‑gradient optimiser (exact max)
# ----------------------------------------------------------------------


def subgradient_optimize(
    init_a: np.ndarray,
    time_deadline: float,
    max_iter: int = 300_000,
    step0: float = 0.07,
) -> tuple[np.ndarray, float]:
    """Projected sub‑gradient descent on the true max‑convolution objective."""
    a = init_a.astype(np.float64).copy()
    best_a = a.copy()
    best_score = compute_score(a)
    n = a.shape[0]

    for t in range(1, max_iter + 1):
        if time.time() >= time_deadline:
            break
        conv = convolution_fft(a, a)
        k = int(np.argmax(conv))

        # sub‑gradient for the active convolution entry
        idx = np.arange(n)
        j = k - idx
        mask = (j >= 0) & (j < n)
        g = np.zeros_like(a)
        g[mask] = 2.0 * a[j[mask]]

        step = step0 / np.sqrt(t)
        a_new = a - step * g
        a_new = project_to_simplex(a_new)

        cur_score = compute_score(a_new)
        if cur_score < best_score:
            best_score, best_a = cur_score, a_new.copy()

        a = a_new

        # occasional jitter to avoid stagnation
        if t % 1000 == 0:
            jitter = np.random.randn(n) * (step0 * 0.004 / np.sqrt(t))
            a = project_to_simplex(a + jitter)

    final_score = compute_score(a)
    if final_score < best_score:
        best_score, best_a = final_score, a.copy()
    return best_a, best_score

def hill_climb_optimize(
    start_seq,
    time_deadline,
    max_steps: int = 300_000,
) -> tuple[np.ndarray, float]:
    """
    Very cheap stochastic hill‑climb:
    - either swap two entries or move a small fraction of mass from i to j,
    - always re‑project onto the simplex,
    - adapt step scale based on acceptance.
    """
    a = np.array(start_seq, dtype=np.float64)
    n = a.shape[0]
    best_a = a.copy()
    best_score = compute_score(best_a)

    step_scale = 0.07  # fraction of current mass moved when doing a transfer

    while max_steps > 0 and time.time() < time_deadline:
        cand = best_a.copy()
        if np.random.rand() < 0.5:          # swap
            i, j = np.random.randint(n, size=2)
            cand[i], cand[j] = cand[j], cand[i]
        else:                               # transfer mass
            i, j = np.random.randint(n, size=2)
            if i != j:
                delta = step_scale * cand[i]
                cand[i] -= delta
                cand[j] += delta
                if cand[i] < 0.0:
                    cand[i] = 0.0

        cand = project_to_simplex(cand)
        cur_score = compute_score(cand)

        if cur_score < best_score:
            best_score = cur_score
            best_a = cand
            step_scale = min(step_scale * 1.07, 0.5)
        else:
            step_scale *= 0.96

        max_steps -= 1

    return best_a, best_score

def propose_candidate(seed: int = 12345, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative sequence that minimises ``evaluate_sequence``.
    Returns a plain Python list of floats (already on the simplex).
    """
    np.random.seed(seed)
    start_time = time.time()
    # keep a few seconds as a safety buffer for the harness
    deadline = start_time + budget_s - 3.0

    # ------------------------------------------------------------------
    # 0. Problem size
    # ------------------------------------------------------------------
    try:
        n = len(height_sequence_1)            # type: ignore[name-defined]
        print(f"Using supplied height_sequence_1 of length {n}.")
        init_candidates = [np.array(height_sequence_1, dtype=np.float64)]
    except Exception:
        n = 2048
        print(f"No external sequence – using default length {n}.")
        init_candidates = []

    # ------------------------------------------------------------------
    # 1. Analytic starters (including richer families)
    # ------------------------------------------------------------------
    init_candidates.append(generate_uniform(n))
    init_candidates.append(generate_central_binomial(n))
    init_candidates.append(generate_arcsine(n))

    # power‑law sweep
    for p in np.linspace(0.35, 0.9, 12):
        init_candidates.append(generate_powerlaw(n, float(p)))

    # geometric sweep
    for r in [0.97, 0.98, 0.985, 0.99, 0.992, 0.995, 0.997, 0.999]:
        init_candidates.append(generate_geometric(n, float(r)))

    # damped central‑binomial
    for r in [0.99, 0.995, 0.998, 0.999, 0.9995]:
        init_candidates.append(generate_central_binomial_mod(n, float(r)))

    # ------------------------------------------------------------------
    # 2. Grid‑search over mixtures (power‑law + geometric)
    # ------------------------------------------------------------------
    # coarse grid – cheap to evaluate
    for p in np.linspace(0.40, 0.70, 7):
        for r in np.linspace(0.985, 0.998, 7):
            for alpha in np.linspace(0.0, 1.0, 5):
                init_candidates.append(
                    generate_mixture(n, float(p), float(r), float(alpha))
                )

    # ------------------------------------------------------------------
    # Evaluate all starters and keep the best so far
    # ------------------------------------------------------------------
    best_seq = None
    best_val = float("inf")
    for idx, cand in enumerate(init_candidates):
        cand = project_to_simplex(cand)
        val = compute_score(cand)
        print(f"Starter {idx:03d}: score = {val:.6f}")
        if val < best_val:
            best_val = val
            best_seq = cand.copy()

    print(f"Best starter after analytical sweep: {best_val:.6f}")

    # ------------------------------------------------------------------
    # 3. Smooth Adam optimisation (multiple random restarts)
    # ------------------------------------------------------------------
    adam_deadline = deadline - 260.0   # reserve time for later phases
    if time.time() < adam_deadline:
        remaining = adam_deadline - time.time()
        # roughly 2 k iterations per second on typical hardware
        max_iters = int(min(800_000, remaining * 2000))
        print(f"Running smooth Adam optimisation – up to {max_iters} iters.")
        # Perform a few restarts (best + jittered copies)
        for restart in range(3):
            if time.time() >= adam_deadline:
                break
            # jitter the start a bit (except for the first restart)
            if restart == 0:
                start_a = best_seq.copy()
            else:
                start_a = best_seq + 0.02 * np.random.randn(n)
                start_a = project_to_simplex(start_a)

            a_adam, score_adam = smooth_adam_optimize(
                start_a,
                time_deadline=adam_deadline,
                max_iters=max_iters,
                beta_start=5.0,
                beta_end=2000.0,
                step0=0.85,
            )
            print(f"  Adam restart {restart}: score = {score_adam:.6f}")
            if score_adam < best_val:
                best_val = score_adam
                best_seq = a_adam.copy()
                print(f"  -> new best after Adam: {best_val:.6f}")

    # ------------------------------------------------------------------
    # 4. Projected sub‑gradient polishing
    # ------------------------------------------------------------------
    subgrad_deadline = deadline - 70.0
    if time.time() < subgrad_deadline:
        remaining = subgrad_deadline - time.time()
        max_iter = int(min(600_000, remaining * 2500))
        print(f"Sub‑gradient polishing – up to {max_iter} iters.")
        a_sub, score_sub = subgradient_optimize(
            best_seq,
            time_deadline=subgrad_deadline,
            max_iter=max_iter,
            step0=0.06,
        )
        print(f"  Sub‑gradient result: score = {score_sub:.6f}")
        if score_sub < best_val:
            best_val = score_sub
            best_seq = a_sub.copy()
            print(f"  -> new best after sub‑grad: {best_val:.6f}")

    # ------------------------------------------------------------------
    # 5. Random‑restart short sub‑gradient bursts (optional)
    # ------------------------------------------------------------------
    restart_deadline = deadline - 35.0
    if time.time() < restart_deadline:
        base = best_seq.copy()
        for r in range(4):
            if time.time() >= restart_deadline:
                break
            pert = base + 0.015 * np.random.randn(n)
            pert = project_to_simplex(pert)
            sub_dead = time.time() + 10.0
            a_tmp, s_tmp = subgradient_optimize(
                pert,
                time_deadline=sub_dead,
                max_iter=80_000,
                step0=0.05,
            )
            if s_tmp < best_val:
                best_val = s_tmp
                best_seq = a_tmp.copy()
                print(f"  Restart {r} improved to {best_val:.6f}")

    # ------------------------------------------------------------------
    # 6. Final stochastic hill‑climb
    # ------------------------------------------------------------------
    hill_deadline = deadline - 2.0
    if time.time() < hill_deadline:
        steps = 300_000
        print(f"Final hill‑climb – up to {steps} steps.")
        polished_seq, hill_score = hill_climb_optimize(
            best_seq.tolist(),
            time_deadline=hill_deadline,
            max_steps=steps,
        )
        print(f"  Hill‑climb score = {hill_score:.6f}")
        if hill_score < best_val:
            best_val = hill_score
            best_seq = polished_seq.copy()
            print(f"  -> new best after hill‑climb: {best_val:.6f}")

    # ------------------------------------------------------------------
    # 7. Safeguard fallback
    # ------------------------------------------------------------------
    if best_seq is None:
        best_seq = generate_uniform(n)
        best_val = compute_score(best_seq)

    # ------------------------------------------------------------------
    # 8. Final reporting (uses the provided evaluate_sequence)
    # ------------------------------------------------------------------
    final_score = evaluate_sequence(best_seq.tolist())
    print(f"\nSearch finished. Final reported score = {final_score:.6f}")

    # Return the plain Python list (already simplex‑normalised)
    return best_seq.tolist()
