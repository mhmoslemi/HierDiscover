def simulated_annealing(a0, deadline):
    a_cur = a0.copy()
    cur_val = compute_score(a_cur)
    best = a_cur.copy()
    best_val = cur_val
    start = time.time()
    total = deadline - start
    # parameters
    max_temp = best_val * 0.02  # guess 2% of score
    min_temp = 1e-6
    max_move_frac_initial = 0.5
    # precomputations
    n = a_cur.shape[0]
    while time.time() < deadline:
        elapsed = time.time() - start
        progress = elapsed / total
        temp = max(min_temp, max_temp * (0.1 ** progress))  # exponential down to min_temp near end
        # move fraction scale
        move_frac = max_move_frac_initial * (temp / max_temp) + 0.001  # ensure minimal moves
        i = np.random.randint(n)
        j = np.random.randint(n)
        while j == i:
            j = np.random.randint(n)
        # decide delta
        max_delta = a_cur[i]  # can't exceed.
        if max_delta <= 0:
            continue
        # sample fraction uniform [0, move_frac] or maybe random*move_frac
        frac = np.random.rand() * move_frac
        delta = max_delta * frac
        if delta <= 0:
            continue
        cand = a_cur.copy()
        cand[i] -= delta
        cand[j] += delta
        # numeric errors: ensure nonneg
        # Normalization: we preserve sum, but rounding may cause sum not exactly 1.
        # So renormalize:
        sum_cand = cand.sum()
        if sum_cand <= 0:
            continue
        cand /= sum_cand
        cand_score = compute_score(cand)
        if cand_score < cur_val:
            accept = True
        else:
            # acceptance probability
            if temp <= 0:
                accept = False
            else:
                prob = np.exp((cur_val - cand_score) / temp)
                accept = np.random.rand() < prob
        if accept:
            a_cur = cand
            cur_val = cand_score
            if cand_score < best_val:
                best_val = cand_score
                best = cand.copy()
    return best

"""
Non‑negative step‑function optimiser for ``evaluate_sequence``.

Approach
--------
1. **Exploratory sweep** – generate a few families of candidate shapes
   (power‑law a_i ∝ (i+1)^‑p  and   exponential a_i ∝ r^i) and keep the
   best one (normalised to sum = 1).

2. **Multi‑active sub‑gradient descent** – minimise
   2·n·max(conv(a,a)) while staying on the simplex.
   At each iteration the gradient is the average of the sub‑gradients of
   *all* convolution entries that are within a tiny tolerance of the
   current maximum.  The step size decays as 1/√t and the iterate is
   projected onto the simplex.

3. **Simulated annealing fine‑tuning** – a long stochastic hill‑climb that
   occasionally accepts worsening moves.  The temperature and the amount
   of mass that is moved shrink exponentially with the elapsed time, which
   lets the search explore widely at the start and focus on tiny refinements
   at the end.

All three phases respect a global 1000 s wall‑clock budget.  The final
candidate is returned as a plain Python list of floats.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Numerical utilities
# ----------------------------------------------------------------------


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    if np.all(v <= 0):
        # uniform distribution is the projection of the zero vector
        return np.full_like(v, 1.0 / v.shape[0])

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, v.shape[0] + 1)
    mask = u - cssv / ind > 0
    rho = np.where(mask)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def full_convolution(a: np.ndarray) -> np.ndarray:
    """
    Full linear convolution a*a using a real FFT.
    Length of the result is 2*len(a)-1.
    """
    n = a.shape[0]
    m = 2 * n - 1
    n_fft = 1 << ((m - 1).bit_length())          # next power of two ≥ m
    A = np.fft.rfft(a, n=n_fft)
    C = A * A
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


def compute_score(a: np.ndarray) -> float:
    """
    Exact objective for a normalised vector (sum = 1):
        2·n·max(conv(a,a))
    """
    n = a.shape[0]
    conv = full_convolution(a)
    return float(2.0 * n * conv.max())


# ----------------------------------------------------------------------
# 1. Exploratory families (power‑law & exponential)
# ----------------------------------------------------------------------


def sweep_initial_candidates(n: int, deadline: float) -> np.ndarray:
    """
    Scan two simple families (power‑law and exponential) and return the
    best normalised vector found before the deadline.
    """
    best_a = None
    best_val = float("inf")

    # ---- power‑law -----------------------------------------------------
    # p ∈ [0.3, 0.9] – a fairly wide window, 0.005 steps
    for p in np.linspace(0.3, 0.9, 121):
        if time.time() >= deadline:
            break
        a = np.arange(1, n + 1, dtype=float) ** (-p)
        a /= a.sum()
        val = compute_score(a)
        if val < best_val:
            best_val, best_a = val, a.copy()

    # ---- exponential ----------------------------------------------------
    # r ∈ [0.90, 0.999] – geometric decay, 0.001 steps
    for r in np.linspace(0.90, 0.999, 120):
        if time.time() >= deadline:
            break
        a = r ** np.arange(n, dtype=float)
        a /= a.sum()
        val = compute_score(a)
        if val < best_val:
            best_val, best_a = val, a.copy()

    # fallback – uniform (should never be chosen as best)
    if best_a is None:
        best_a = np.full(n, 1.0 / n)

    return best_a


# ----------------------------------------------------------------------
# 2. Multi‑active sub‑gradient refinement
# ----------------------------------------------------------------------


def subgradient_refine_multi(a0: np.ndarray, deadline: float,
                             reserve: float = 240.0,
                             step0: float = 0.5) -> np.ndarray:
    """
    Projected sub‑gradient descent on 2·n·max(conv(a,a)).
    Uses *all* active constraints (conv entries within a tiny tolerance of the
    maximum) to build a smoother descent direction.
    The loop stops ``reserve`` seconds before the global deadline to leave
    time for the subsequent simulated‑annealing phase.
    """
    a = a0.copy()
    n = a.shape[0]
    best_a = a.copy()
    best_val = compute_score(a)

    t = 0
    while time.time() < deadline - reserve:
        t += 1
        conv = full_convolution(a)
        max_c = conv.max()
        # indices whose value is within 1e-9 relative tolerance of the max
        eps = max_c * 1e-9
        active = np.where(conv >= max_c - eps)[0]
        if active.size == 0:               # safety fallback
            active = np.array([np.argmax(conv)])

        # average sub‑gradient over the active set
        g = np.zeros_like(a)
        for k in active:
            idx = np.arange(n)
            j = k - idx
            mask = (j >= 0) & (j < n)
            g[mask] += 2.0 * a[j[mask]]
        g /= float(active.size)

        # diminishing step size
        step = step0 / np.sqrt(t)
        a = a - step * g
        a = project_to_simplex(a)

        val = compute_score(a)
        if val < best_val:
            best_val, best_a = val, a.copy()

    return best_a


# ----------------------------------------------------------------------
# 3. Simulated‑annealing fine‑tuning
# ----------------------------------------------------------------------


def simulated_annealing(a_start: np.ndarray, deadline: float,
                        min_temp: float = 1e-6,
                        max_move_frac: float = 0.5) -> np.ndarray:
    """
    Stochastic hill‑climb with a classic temperature schedule.
    Moves consist of transferring a random fraction of the mass of a
    randomly selected source index to a different target index.
    The temperature (and consequently the move size) decays exponentially,
    allowing large jumps early on and precise adjustments later.
    """
    a_cur = a_start.copy()
    cur_val = compute_score(a_cur)

    best_a = a_cur.copy()
    best_val = cur_val

    n = a_cur.shape[0]
    start = time.time()
    total = deadline - start

    # initial temperature is set to a small fraction of the current score
    max_temp = max(0.02 * cur_val, 1e-3)

    while time.time() < deadline:
        elapsed = time.time() - start
        progress = elapsed / total
        # exponential cooling: temperature drops by a factor 10 over the run
        temp = max(min_temp, max_temp * (0.1 ** progress))

        # gradually shrink the amount of mass we are allowed to move
        move_frac = max_move_frac * (temp / max_temp) + 1e-4

        i = np.random.randint(n)
        j = np.random.randint(n)
        while j == i:
            j = np.random.randint(n)

        # fraction of the source mass we actually move
        if a_cur[i] <= 0.0:
            continue
        delta = a_cur[i] * np.random.rand() * move_frac
        if delta <= 0.0:
            continue

        cand = a_cur.copy()
        cand[i] -= delta
        cand[j] += delta

        # re‑normalise to avoid floating‑point drift
        cand_sum = cand.sum()
        if cand_sum <= 0.0:
            continue
        cand /= cand_sum

        cand_val = compute_score(cand)

        # Metropolis acceptance
        if cand_val < cur_val:
            accept = True
        else:
            if temp <= 0.0:
                accept = False
            else:
                prob = np.exp((cur_val - cand_val) / temp)
                accept = np.random.rand() < prob

        if accept:
            a_cur = cand
            cur_val = cand_val
            if cand_val < best_val:
                best_val, best_a = cand_val, cand.copy()

    return best_a


# ----------------------------------------------------------------------
# Main entry point required by the harness
# ----------------------------------------------------------------------

    np.random.seed(seed)
    start_time = time.time()
    # keep a small safety margin for the harness
    deadline = start_time + budget_s - 5.0

    n = 2048                     # length of the step function

    # -----------------------------------------------------------------

    try:
        fallback = [float(x) for x in height_sequence_1]   # type: ignore
