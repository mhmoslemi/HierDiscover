def adam_softmax_opt(a_init, start_score, deadline):
    a = a_init.copy()
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1=0.9, beta2=0.999, eps=1e-8
    t=0
    best_a = a.copy()
    best_score = start_score
    # schedule beta (temperature) as before; also learning rate schedule maybe.
    while time.time() < deadline:
        # compute conv
        conv = convolution_fft(a,a)
        max_conv = conv.max()
        cur_score = 2.0 * n * float(max_conv)
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
        # schedule temperature beta
        elapsed = time.time() - start
        fraction = elapsed / (deadline - start)
        beta = beta_start + fraction*(beta_end-beta_start)
        # compute softmax weights
        shifted = conv - max_conv
        exp_vals = np.exp(beta * shifted)
        w = exp_vals / exp_vals.sum()
        # gradient
        # compute gradient via FFT
        # a_rev = a[::-1]
        conv_full = convolution_fft(w, a[::-1])
        grad = 2.0 * beta * conv_full[n-1:2*n-1]
        # Adam update
        t += 1
        m = beta1*m + (1-beta1)*grad
        v = beta2*v + (1-beta2)*(grad**2)
        m_hat = m / (1-beta1**t)
        v_hat = v / (1-beta2**t)
        step = step_size / (np.sqrt(v_hat) + eps)  # step_size baseline
        a = a - step * m_hat
        # enforce simplex and non-negative
        a = project_to_simplex(a)
        # optionally enforce symmetry
        a = (a + a[::-1]) / 2
        a = project_to_simplex(a)   # reproject after sym.

    return best_a, best_score

"""
Propose a step‑function that minimizes

    evaluate_sequence(a) = 2·n·max(conv(a,a)) / (sum(a)²)

The algorithm consists of three phases

1️⃣ Seeding – a collection of hand‑crafted shapes (sine‑squared,
   beta‑family, sparse spikes) and random Dirichlet draws.  The best
   seed (exact objective) is kept.

2️⃣ Adam‑based soft‑max gradient descent – we minimise a smooth surrogate
   of the max‑convolution:

        Lβ(a) = log Σ exp(β·conv(a,a))

   and use the Adam optimiser (projected onto the probability simplex)
   while slowly annealing the temperature β from a small value to a
   large one.  Symmetry (aᵢ = a_{n‑1‑i}) is enforced after each step,
   which empirically reduces the peak of the convolution.

3️⃣ Refinement – classic projected sub‑gradient descent on the true
   max‑objective followed by a short stochastic hill‑climb.

All work respects the 1000 s wall‑clock budget.  The function returns a
plain ``list[float]`` of non‑negative heights.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Low‑level utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full linear convolution using a real FFT."""
    m = a.shape[0] + b.shape[0] - 1
    n_fft = next_pow2(m)
    A = np.fft.rfft(a, n=n_fft)
    B = np.fft.rfft(b, n=n_fft)
    C = A * B
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


def compute_score(a: np.ndarray) -> float:
    """
    Exact objective for a probability vector (sum = 1):
        2·n·max(conv(a,a)).
    """
    n = a.shape[0]
    conv = convolution_fft(a, a)
    return 2.0 * n * float(conv.max())


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of `v` onto the probability simplex
    {x | x_i ≥ 0, Σ x_i = 1}.
    """
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / v.shape[0])

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, v.shape[0] + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.where(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


# ----------------------------------------------------------------------
# Seed generators
# ----------------------------------------------------------------------


def generate_deterministic(n: int) -> np.ndarray:
    """Sin² shape a_i = sin²(π·i/(n+1)), normalised to sum 1."""
    i = np.arange(1, n + 1, dtype=float)
    a = np.sin(np.pi * i / (n + 1.0)) ** 2
    a /= a.sum()
    return a


def generate_beta_family(n: int, p: float) -> np.ndarray:
    """
    Symmetric polynomial shape a_i ∝ ((i+1)*(n-i))**p.
    p>0 clusters mass in the centre, p<0 pushes it to the edges.
    """
    i = np.arange(n, dtype=float)
    w = ((i + 1.0) * (n - i)) ** p
    w = np.where(w > 0, w, 0.0)
    a = w / w.sum()
    return a


def random_dirichlet(n: int, conc: float = 1.0) -> np.ndarray:
    """Draw a random probability vector from Dirichlet(conc)."""
    if conc == 1.0:
        a = np.random.rand(n)
    else:
        a = np.random.gamma(conc, 1.0, size=n)
    a /= a.sum()
    return a


def random_spike(n: int, k: int = 70) -> np.ndarray:
    """
    Very sparse vector: choose `k` distinct positions and assign
    random positive weights, then normalise.
    """
    idx = np.random.choice(n, size=min(k, n), replace=False)
    w = np.random.rand(len(idx))
    a = np.zeros(n, dtype=float)
    a[idx] = w
    a /= a.sum()
    return a


# ----------------------------------------------------------------------
# Phase 2 – Adam soft‑max optimisation
# ----------------------------------------------------------------------


def adam_softmax_opt(
    a_init: np.ndarray,
    start_score: float,
    deadline: float,
    beta_start: float = 5.0,
    beta_end: float = 4000.0,
    step_size: float = 0.01,
    enforce_symmetry: bool = True,
) -> tuple[np.ndarray, float]:
    """
    Minimise the log‑sum‑exp surrogate of the max‑convolution using
    Adam (projected onto the probability simplex).  The temperature β
    is linearly annealed from ``beta_start`` to ``beta_end``.
    """
    n = a_init.shape[0]
    a = a_init.copy()
    best_a = a.copy()
    best_score = start_score

    # Adam state
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    t = 0

    start = time.time()
    last_improve = 0
    cur_best_iter = 0

    while time.time() < deadline:
        # ---- exact objective (only for bookkeeping) ----
        conv_aa = convolution_fft(a, a)
        max_conv = conv_aa.max()
        cur_score = 2.0 * n * float(max_conv)
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            last_improve = t
            cur_best_iter = t

        # ---- temperature schedule ----
        elapsed = time.time() - start
        fraction = elapsed / max(1e-12, (deadline - start))
        beta = beta_start + fraction * (beta_end - beta_start)

        # ---- soft‑max weights ----
        shifted = conv_aa - max_conv  # max becomes 0
        exp_vals = np.exp(beta * shifted)  # stable, all ≤ 1
        w = exp_vals / exp_vals.sum()  # distribution over 2n‑1 points

        # ---- gradient of the surrogate ----
        # gradient = 2·β·correlate(w, a)  (valid mode)
        # compute via FFT: correlate(w, a) = conv(w, a[::-1]) valid slice
        conv_full = convolution_fft(w, a[::-1])
        grad = 2.0 * beta * conv_full[n - 1 : 2 * n - 1]

        # ---- Adam update ----
        t += 1
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)

        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)

        step = step_size * m_hat / (np.sqrt(v_hat) + eps)

        a = a - step

        # ---- projection & optional symmetry ----
        a = project_to_simplex(a)
        if enforce_symmetry:
            a = (a + a[::-1]) / 2.0
            a = project_to_simplex(a)

        # ---- occasional random perturbation if stagnating ----
        if t - last_improve > 5000:
            # perturb the best known point a little and restart Adam state
            perturb = 0.02 * np.random.randn(n)
            a = best_a + perturb
            a = np.maximum(a, 0.0)
            a = project_to_simplex(a)
            if enforce_symmetry:
                a = (a + a[::-1]) / 2.0
                a = project_to_simplex(a)
            m.fill(0.0)
            v.fill(0.0)
            last_improve = t  # reset stagnation counter

    return best_a, best_score


# ----------------------------------------------------------------------
# Phase 3 – Sub‑gradient refinement (as before)
# ----------------------------------------------------------------------


def subgradient_refine(
    a: np.ndarray,
    start_score: float,
    deadline: float,
    max_iter: int = 400_000,
    step0: float = 0.07,
) -> tuple[np.ndarray, float]:
    """Projected sub‑gradient descent on the true max‑objective."""
    n = a.shape[0]
    cur_a = a.copy()
    best_a = cur_a.copy()
    best_score = start_score

    for t in range(1, max_iter + 1):
        if time.time() >= deadline:
            break

        conv = convolution_fft(cur_a, cur_a)
        k = int(np.argmax(conv))

        idx = np.arange(n)
        j = k - idx
        mask = (j >= 0) & (j < n)
        g = np.zeros_like(cur_a)
        g[mask] = 2.0 * cur_a[j[mask]]

        step = step0 / np.sqrt(t)
        cur_a -= step * g
        cur_a = project_to_simplex(cur_a)

        cur_score = compute_score(cur_a)
        if cur_score < best_score:
            best_score = cur_score
            best_a = cur_a.copy()

        if t % 1000 == 0:
            # tiny jitter to shake out of flat regions
            jitter = np.random.randn(n) * (step0 * 0.004 / np.sqrt(t))
            cur_a += jitter
            cur_a = project_to_simplex(cur_a)

    return best_a, best_score


# ----------------------------------------------------------------------
# Phase 4 – Hill‑climb polishing
# ----------------------------------------------------------------------


def hill_climb(
    a: np.ndarray,
    deadline: float,
    max_steps: int = 500_000,
) -> tuple[np.ndarray, float]:
    """Stochastic hill‑climb that accepts only improving moves."""
    n = a.shape[0]
    best_a = a.copy()
    best_score = compute_score(best_a)
    step_scale = 0.08

    steps = 0
    while steps < max_steps and time.time() < deadline:
        cand = best_a.copy()
        if np.random.rand() < 0.5:
            # swap two entries
            i, j = np.random.randint(n, size=2)
            cand[i], cand[j] = cand[j], cand[i]
        else:
            # transfer a fraction from i to j
            i, j = np.random.randint(n, size=2)
            if cand[i] > 0.0 and i != j:
                delta = step_scale * cand[i]
                cand[i] -= delta
                cand[j] += delta

        cand /= cand.sum()  # re‑normalise any numerical drift
        cur_score = compute_score(cand)

        if cur_score < best_score:
            best_score = cur_score
            best_a = cand
            step_scale = min(step_scale * 1.07, 0.5)
        else:
            step_scale *= 0.96

        steps += 1

    return best_a, best_score
