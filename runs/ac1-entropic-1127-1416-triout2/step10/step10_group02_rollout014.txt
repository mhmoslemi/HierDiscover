When we modify a[i] by -δ and a[j] by +δ (i != j), updated a' = a + δ*(e_j - e_i). Then convolution conv' = conv + 2δ * (a * (e_j - e_i)) + δ^2 * (e_j*e_j - e_i*e_i) convolution? Here (a * Δ) = δ * (a * e_j - a * e_i) => δ * (conv_shifted by j - conv_shifted by i). The convolution of a and e_j is just a shifted by j (i.e., a shifted right by j): (a * e_j)_k = a_{k-j} if in bounds else 0. Similarly for e_i. def anneal(a, deadline):
    score = compute_score(a)
    best = a.copy()
    best_score = score
    n = len(a)
    iteration = 0
    T = score * 0.1  # initial temperature
    while time.time() < deadline:
        iteration += 1
        # gradually cool
        if iteration % 1000 == 0:
            T *= 0.99
        i = np.random.randint(n)
        j = np.random.randint(n-1)
        if j >= i:
            j += 1
        # Choose delta
        max_delta = a[i]  # cannot exceed entire mass.
        if max_delta == 0:
            continue
        # step scale decays gradually
        step_frac = 0.2 * (0.99 ** (iteration // 1000))
        delta = step_frac * max_delta * np.random.rand()
        # Apply move
        a_new = a.copy()
        a_new[i] -= delta
        a_new[j] += delta
        # Ensure nonneg, sum unchanged
        # Compute new score
        val = compute_score(a_new)
        # Accept condition
        if val < score or np.random.rand() < np.exp((score - val) / max(T, 1e-12)):
            a = a_new
            score = val
            if val < best_score:
                best_score = val
                best = a.copy()
        # else keep a unchanged
        # optionally periodic restart randomization if stuck
    return best, best_score

def random_candidate(n):

def subgradient_polish(a0, deadline, step0=0.1):
    a = a0.copy()
    n = a.shape[0]
    best = a.copy()
    best_score = compute_score(a)
    t = 0
    while time.time() < deadline:
        t += 1
        conv = full_convolution(a)
        max_val = conv.max()
        # indices where conv is within 1e-12 of the max
        active = np.where(conv >= max_val - 1e-12)[0]
        # gradient = sum_{k in active} 2 * shifted a
        grad = np.zeros_like(a)
        # vectorised accumulation using slices
        for k in active:
            # shift a by k: contribution for i where j=k-i in [0,n)
            # Equivalent to: grad[i] += 2*a[k-i] for i where 0<=k-i<n
            start = max(0, k - (n-1))
            end = min(n-1, k)
            i_idx = np.arange(start, end+1, dtype=int)
            j_idx = k - i_idx
            grad[i_idx] += 2.0 * a[j_idx]
        step = step0 / np.sqrt(t)
        a = a - step * grad
        a = project_to_simplex(a)
        val = compute_score(a)
        if val < best_score:
            best_score = val
            best = a.copy()
    return best

    ``evaluate_sequence``.  Returns a plain Python list of floats.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 5.0   # safety margin

    n = 2048
    best_seq = None
    best_score = float("inf")

    # 0. Optional fallback sequence

        val = compute_score(a)   # internal objective (sum = 1)
        if val < best_score:
            best_score = val
            best_seq = a.tolist()
            print(f"[sample] new best {best_score:.6f}")

    # 2. Simulated‑annealing / hill‑climb refinement on the best so far
    # -----------------------------------------------------------------

    else:

    # Final report using the official evaluator
    # -----------------------------------------------------------------
    final_score = evaluate_sequence(best_seq)

"""
Simulated‑annealing / random‑search optimiser for the objective
defined by ``evaluate_sequence``.  The algorithm works in three
stages:

1. **Broad random sampling** – many candidate probability vectors
   are generated from a mixture of families (power‑law, exponential,
   Beta‑shaped, Dirichlet, uniform).  The best candidate seen so far is
   kept.

2. **Temperature‑controlled mass‑transfer hill‑climb** – starting from
   the best vector, a simulated‑annealing loop repeatedly moves a
   small amount of mass from one coordinate to another, accepting moves
   that improve the objective or, with a Metropolis probability,
   occasionally worsening moves.  This refines the shape of the
   distribution.

3. **Optional sub‑gradient polish & uniform blending** – a short
   projected sub‑gradient descent on the true ‑max‑convolution functional
   is run, followed by a few convex blends with the uniform vector to
   smooth any remaining spikes.

All work respects the time budget supplied by the harness.  The
returned list contains non‑negative floats (the sum is forced to 1),
which is exactly the format expected by ``evaluate_sequence``.
"""

import time
import math
import numpy as np

# ------------------------------------------------------------
# Basic numerical utilities
# ------------------------------------------------------------

def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    if np.all(v <= 0):
        # uniform distribution is the projection of the zero vector
        return np.full_like(v, 1.0 / v.shape[0])

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, v.shape[0] + 1)
    mask = u - cssv / ind > 0
    rho = np.where(mask)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def full_convolution(a: np.ndarray) -> np.ndarray:
    """
    Full linear convolution a*a using a real FFT.
    Length of the result is 2*len(a)-1.
    """
    n = a.shape[0]
    m = 2 * n - 1
    # next power of two for efficient FFT
    n_fft = 1 << ((m - 1).bit_length())
    A = np.fft.rfft(a, n=n_fft)
    C = A * A
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


def compute_score(a: np.ndarray) -> float:
    """
    Internal objective for a normalised vector (sum == 1):
        2·n·max(conv(a,a))
    This equals the value returned by ``evaluate_sequence`` when the
    sum of a is 1 (the scaling invariance of the original formula).
    """
    n = a.shape[0]
    conv = full_convolution(a)
    return float(2.0 * n * conv.max())


# ------------------------------------------------------------
# Random candidate generators (different families)
# ------------------------------------------------------------

def random_candidate(n: int) -> np.ndarray:
    """Return a random non‑negative vector of length n, normalised to sum = 1."""
    kind = np.random.choice(['power', 'exp', 'beta', 'dirichlet', 'uniform'])
    if kind == 'power':
        p = np.random.uniform(0.3, 0.9)
        a = np.arange(1, n + 1, dtype=float) ** (-p)
    elif kind == 'exp':
        lam = np.random.uniform(0.001, 0.02)
        a = np.exp(-lam * np.arange(n, dtype=float))
    elif kind == 'beta':
        # Beta(α,β) on the discrete grid, α,β > 0
        α = np.random.uniform(0.2, 2.5)
        β = np.random.uniform(0.2, 2.5)
        x = (np.arange(n, dtype=float) + 0.5) / n          # in (0,1)
        a = np.power(x, α - 1.0) * np.power(1.0 - x, β - 1.0)
    elif kind == 'dirichlet':
        conc = np.random.uniform(0.2, 4.0)
        a = np.random.gamma(conc, 1.0, size=n)
    else:   # uniform
        a = np.ones(n, dtype=float)

    # tiny jitter to avoid exact zeros when α or β < 1
    a += np.random.rand(n) * 1e-6
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s == 0.0:
        a = np.full(n, 1.0 / n)
    else:
        a /= s
    return a


# ------------------------------------------------------------
# Simulated‑annealing / hill‑climb refinement
# ------------------------------------------------------------

def anneal(a0: np.ndarray, deadline: float) -> tuple[np.ndarray, float]:
    """
    Mass‑transfer simulated‑annealing starting from a0.
    Returns (best_vector, best_score) found before the deadline.
    """
    n = a0.shape[0]
    cur = a0.copy()
    cur_score = compute_score(cur)
    best = cur.copy()
    best_score = cur_score
    iteration = 0
    # initialise temperature to a fraction of the current objective
    T = max(cur_score * 0.5, 1.0)
    last_improve = 0

    while time.time() < deadline:
        iteration += 1

        # simple exponential cooling
        if iteration % 1000 == 0:
            T *= 0.95

        # pick two distinct indices
        i = np.random.randint(n)
        j = np.random.randint(n - 1)
        if j >= i:
            j += 1

        # amount of mass we are allowed to move from i
        max_delta = cur[i]
        if max_delta <= 0.0:
            continue

        # step size slowly shrinks with elapsed iterations
        step_frac = 0.4 * (0.98 ** (iteration // 5000))
        delta = step_frac * max_delta * np.random.rand()
        if delta <= 0.0:
            continue

        # propose the move
        nxt = cur.copy()
        nxt[i] -= delta
        nxt[j] += delta

        val = compute_score(nxt)

        # Metropolis acceptance
        accept = False
        if val < cur_score:
            accept = True
        else:
            prob = math.exp((cur_score - val) / max(T, 1e-12))
            if np.random.rand() < prob:
                accept = True

        if accept:
            cur = nxt
            cur_score = val
            if val < best_score:
                best_score = val
                best = cur.copy()
                last_improve = iteration
                print(f"[anneal] iter {iteration}, new best {best_score:.6f}")

        # occasional random perturbation if we have been stuck for a long time
        if iteration - last_improve > 20000:
            noise = np.random.randn(n) * 1e-3
            cur = cur + noise
            cur = np.maximum(cur, 0.0)
            cur = project_to_simplex(cur)
            cur_score = compute_score(cur)
            last_improve = iteration

    return best, best_score


# ------------------------------------------------------------
# Sub‑gradient polishing (optional final touch)
# ------------------------------------------------------------

def subgradient_polish(a0: np.ndarray, deadline: float,
                       step0: float = 0.1) -> np.ndarray:
    """
    Projected sub‑gradient descent on 2·n·max(conv(a,a)).
    Uses all active indices where the convolution is maximal.
    """
    a = a0.copy()
    n = a.shape[0]
    best_a = a.copy()
    best_val = compute_score(a)
    t = 0

    while time.time() < deadline:
        t += 1
        conv = full_convolution(a)
        mx = conv.max()
        active = np.where(conv >= mx - 1e-12)[0]

        grad = np.zeros_like(a)
        # accumulate sub‑gradient contributions from every active index
        for k in active:
            # i such that j = k - i lies in [0, n)
            i_low = max(0, k - (n - 1))
            i_high = min(n - 1, k)
            i_idx = np.arange(i_low, i_high + 1, dtype=int)
            j_idx = k - i_idx
            grad[i_idx] += 2.0 * a[j_idx]

        step = step0 / np.sqrt(t)
        a -= step * grad
        a = project_to_simplex(a)

        val = compute_score(a)
        if val < best_val:
            best_val = val
            best_a = a.copy()

    return best_a


# ------------------------------------------------------------
# Small uniform blend – sometimes reduces a remaining spike
# ------------------------------------------------------------

def blend_with_uniform(a: np.ndarray, alphas: list[float]) -> tuple[np.ndarray, float]:
    """Mix a with the uniform distribution for a few α values and keep the best."""
    n = a.shape[0]
    uniform = np.full(n, 1.0 / n)
    best = a
    best_score = compute_score(a)

    for α in alphas:
        cand = (1.0 - α) * a + α * uniform
        cand = cand / cand.sum()
        val = compute_score(cand)
        if val < best_score:
            best_score = val
            best = cand
    return best, best_score
