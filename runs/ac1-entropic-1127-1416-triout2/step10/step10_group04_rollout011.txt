"""
Step‑function optimiser for the ``evaluate_sequence`` challenge.

Algorithmic outline
-------------------
1. **Problem size** – Use the supplied ``height_sequence_1`` if present,
   otherwise start with a power‑of‑two length (4096).  The length is capped
   at 8192 to keep the FFT cheap.

2. **Rich initial pool** – Several analytic families are sampled
   (uniform, central‑binomial, power‑law, geometric, truncated Gaussian,
   Beta‑distribution).  The best‑scoring vector (exact max‑convolution
   score) seeds the optimisation.

3. **Smooth max‑approximation** – minimise a log‑sum‑exp surrogate of the
   auto‑convolution using the Adam optimiser.  The temperature β is
   annealed from a modest value (≈5) to a very large one (≥5000) so that
   the surrogate approaches the true max.  The Adam parameters and the
   number of iterations are chosen dynamically from the remaining wall‑clock.

4. **Projected sub‑gradient polishing** – After the smooth phase we
   switch to the true max objective and perform a sub‑gradient descent
   with a line‑search style step size (the maximal feasible step in the
   descent direction, scaled by a slowly decreasing factor).

5. **Targeted stochastic hill‑climb** – Tiny mass‑transfer moves guided
   by the gradient of the active convolution entry are applied, with a
   simple acceptance rule (keep the move only if it improves the score).

6. **Random restarts** – A few short restarts (perturb‑‑and‑‑polish)
   explore different basins of attraction.

All phases respect the global time budget (default 1000 s) and the best
vector found at any time is kept.  The final vector is returned as a plain
list of floats (already on the simplex, i.e. sum = 1, non‑negative).
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Core utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Return the smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """
    Compute the full linear convolution a * b using real FFT.
    Output length = len(a) + len(b) - 1.
    """
    m = a.shape[0] + b.shape[0] - 1
    n_fft = next_pow2(m)
    A = np.fft.rfft(a, n=n_fft)
    B = np.fft.rfft(b, n=n_fft)
    C = A * B
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


def compute_score(a: np.ndarray) -> float:
    """
    Exact objective for a vector already on the simplex (sum = 1).
    Score = 2·n·max(conv(a, a)).
    """
    n = a.shape[0]
    conv = convolution_fft(a, a)
    return 2.0 * n * float(conv.max())


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the probability simplex:
    {x | x >= 0, sum x = 1}.
    """
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / v.shape[0])

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, v.shape[0] + 1)
    cond = u - cssv / ind > 0
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Analytic families for initial guesses
# ----------------------------------------------------------------------


def generate_uniform(n: int) -> np.ndarray:
    """Uniform distribution on the simplex."""
    return np.full(n, 1.0 / n, dtype=np.float64)


def generate_central_binomial(n: int) -> np.ndarray:
    """a_i ∝ C(2i,i)/4^i (decreases ~i^{-1/2}), normalised."""
    a = np.empty(n, dtype=np.float64)
    a[0] = 1.0
    for i in range(1, n):
        a[i] = a[i - 1] * (2 * i - 1) / (2.0 * i)
    return project_to_simplex(a)


def generate_powerlaw(n: int, p: float) -> np.ndarray:
    """a_i ∝ (i+1)^{-p}."""
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = idx ** (-p)
    return project_to_simplex(a)


def generate_geometric(n: int, r: float) -> np.ndarray:
    """Geometric distribution a_i = (1-r) r^i."""
    i = np.arange(n, dtype=np.float64)
    a = (1.0 - r) * (r ** i)
    return project_to_simplex(a)


def generate_gaussian(n: int, sigma: float) -> np.ndarray:
    """Discrete Gaussian centred in the middle of the vector."""
    centre = (n - 1) / 2.0
    x = np.arange(n, dtype=np.float64)
    a = np.exp(-0.5 * ((x - centre) / sigma) ** 2)
    return project_to_simplex(a)


def generate_beta(n: int, alpha: float, beta: float) -> np.ndarray:
    """Beta‑distribution density on the discrete grid {0,…,n‑1}."""
    # map indices to [0,1]
    t = np.linspace(0.0, 1.0, n, dtype=np.float64)
    # avoid exact 0/1 when alpha<1 or beta<1 (division by zero)
    eps = 1e-12
    t = np.clip(t, eps, 1.0 - eps)
    a = (t ** (alpha - 1.0)) * ((1.0 - t) ** (beta - 1.0))
    return project_to_simplex(a)


# ----------------------------------------------------------------------
# Smooth (log‑sum‑exp) optimisation – Adam
# ----------------------------------------------------------------------


def smooth_adam_optimize(start_a: np.ndarray,
                         time_deadline: float,
                         max_iters: int,
                         beta_start: float = 5.0,
                         beta_end: float = 5000.0,
                         step0: float = 0.9,
                         adam_beta1: float = 0.9,
                         adam_beta2: float = 0.999,
                         adam_eps: float = 1e-8) -> tuple[np.ndarray, float]:
    """
    Adam optimiser for the smoothed max‑convolution objective.
    Uses a log‑sum‑exp surrogate with temperature β annealed
    from ``beta_start`` to ``beta_end``.
    """
    a = start_a.astype(np.float64).copy()
    n = a.shape[0]
    m1 = np.zeros_like(a)
    m2 = np.zeros_like(a)

    best_a = a.copy()
    best_score = compute_score(a)

    for it in range(1, max_iters + 1):
        if time.time() >= time_deadline:
            break

        # Linear anneal of log‑temperature on a log‑scale
        t_frac = it / max_iters
        beta = beta_start * (beta_end / beta_start) ** t_frac

        # Convolution a * a
        conv = convolution_fft(a, a)
        maxc = conv.max()

        # soft‑max weights
        exp_w = np.exp(beta * (conv - maxc))
        w = exp_w / exp_w.sum()

        # Gradient of the smoothed objective: 2·rev(a) * w
        rev_a = a[::-1]
        grad_full = convolution_fft(rev_a, w)          # length = 3n-2
        grad = 2.0 * grad_full[n - 1:n - 1 + n]

        # Adam update
        m1 = adam_beta1 * m1 + (1.0 - adam_beta1) * grad
        m2 = adam_beta2 * m2 + (1.0 - adam_beta2) * (grad * grad)
        m1_hat = m1 / (1.0 - adam_beta1 ** it)
        m2_hat = m2 / (1.0 - adam_beta2 ** it)

        step = step0 / math.sqrt(it)          # classic 1/√t decay
        a_new = a - step * m1_hat / (np.sqrt(m2_hat) + adam_eps)
        a_new = project_to_simplex(a_new)

        a = a_new

        if it % 2000 == 0:
            cur = compute_score(a)
            if cur < best_score:
                best_score, best_a = cur, a.copy()

    final = compute_score(a)
    if final < best_score:
        best_score, best_a = final, a.copy()
    return best_a, best_score


# ----------------------------------------------------------------------
# Exact sub‑gradient optimiser on the true max‑objective
# ----------------------------------------------------------------------


def subgradient_optimize(start_a: np.ndarray,
                         time_deadline: float,
                         max_iter: int = 1_000_000,
                         step0: float = 0.12) -> tuple[np.ndarray, float]:
    """
    Projected sub‑gradient descent on 2·n·max(conv(a,a)).
    Uses an adaptive step size based on the maximal feasible move
    (all components stay non‑negative) and a slowly decaying scale.
    """
    a = start_a.astype(np.float64).copy()
    n = a.shape[0]
    best_a = a.copy()
    best_score = compute_score(a)

    for t in range(1, max_iter + 1):
        if time.time() >= time_deadline:
            break

        conv = convolution_fft(a, a)
        k = int(np.argmax(conv))               # index of the active peak

        # Sub‑gradient of the active convolution entry:
        # g_i = 2 * a_{k-i} whenever the index is valid, else 0.
        idx = np.arange(n)
        j = k - idx
        mask = (j >= 0) & (j < n)
        g = np.zeros_like(a)
        g[mask] = 2.0 * a[j[mask]]

        # Determine the largest step we can take without leaving the simplex.
        # For each i with g_i>0, the component a_i will be decreased.
        # The maximum feasible α is min_i a_i / g_i.
        positive = g > 1e-16
        if np.any(positive):
            alpha_max = np.min(a[positive] / g[positive])
        else:
            alpha_max = 0.0

        # Diminishing step size; keep it a fraction of the feasible maximum.
        decay = step0 / math.sqrt(t)
        alpha = decay * alpha_max

        a_new = a - alpha * g
        a_new = project_to_simplex(a_new)

        cur = compute_score(a_new)
        if cur < best_score:
            best_score, best_a = cur, a_new.copy()

        a = a_new

        if t % 5000 == 0:
            print(f" subgrad iter {t:7d}, best score = {best_score:.6f}")

    final = compute_score(a)
    if final < best_score:
        best_score, best_a = final, a.copy()
    return best_a, best_score


# ----------------------------------------------------------------------
# Stochastic hill‑climb fine‑tuning
# ----------------------------------------------------------------------


def hill_climb(start_seq,
               time_deadline,
               max_steps: int = 300_000) -> tuple[list[float], float]:
    """
    Tiny stochastic moves (swap or fractional mass transfer) that are
    accepted only when they improve the exact max‑convolution score.
    """
    a = np.array(start_seq, dtype=np.float64)
    n = a.shape[0]
    best_a = a.copy()
    best_score = compute_score(best_a)

    step_scale = 0.07  # initial fraction of mass moved

    while max_steps > 0 and time.time() < time_deadline:
        cand = best_a.copy()

        if np.random.rand() < 0.5:          # swap two entries
            i, j = np.random.randint(n, size=2)
            cand[i], cand[j] = cand[j], cand[i]
        else:                               # move a fraction from i to j
            i, j = np.random.randint(n, size=2)
            if i != j:
                delta = step_scale * best_a[i]
                cand[i] -= delta
                cand[j] += delta
                if cand[i] < 0.0:
                    cand[i] = 0.0

        cand = project_to_simplex(cand)
        cur = compute_score(cand)

        if cur < best_score:
            best_score = cur
            best_a = cand
            step_scale = min(step_scale * 1.05, 0.5)
        else:
            step_scale *= 0.97

        max_steps -= 1

    return best_a.tolist(), best_score

def propose_candidate(seed: int = 12345, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative sequence that minimises ``evaluate_sequence``.
    Returns a plain Python list of floats (already on the simplex).
    """
    np.random.seed(seed)
    start_time = time.time()
    # keep a small safety margin for the harness
    deadline = start_time + budget_s - 3.0

    # --------------------------------------------------------------
    # 0. Determine problem size (length of the sequence)
    # --------------------------------------------------------------
    try:
        # If an external construction is available, reuse its length.
        external = list(height_sequence_1)          # type: ignore
        n0 = len(external)
        n = next_pow2(max(2048, n0))
        if n > 8192:
            n = 8192
        print(f"Using supplied height_sequence_1, padded to length {n}.")
        base_external = np.zeros(n, dtype=np.float64)
        base_external[:n0] = np.array(external, dtype=np.float64)
        base_external = project_to_simplex(base_external)
        init_candidates = [base_external]
    except Exception:
        n = 4096
        init_candidates = []
        print(f"No external sequence – using length {n}.")

    # --------------------------------------------------------------
    # 1. Build a diverse set of analytic starters.
    # --------------------------------------------------------------
    init_candidates.append(generate_uniform(n))
    init_candidates.append(generate_central_binomial(n))

    # power‑law family (both deterministic grid and random draws)
    for p in np.linspace(0.3, 1.0, 7):
        init_candidates.append(generate_powerlaw(n, float(p)))
    for _ in range(5):
        p = np.random.uniform(0.2, 1.4)
        init_candidates.append(generate_powerlaw(n, p))

    # geometric family
    for r in [0.97, 0.98, 0.985, 0.99, 0.992, 0.995]:
        init_candidates.append(generate_geometric(n, float(r)))
    for _ in range(3):
        r = np.random.uniform(0.96, 0.999)
        init_candidates.append(generate_geometric(n, r))

    # truncated Gaussian family
    centre = (n - 1) / 2.0
    for sigma_factor in [0.10, 0.15, 0.20, 0.30]:
        sigma = sigma_factor * n
        init_candidates.append(generate_gaussian(n, sigma))
    for _ in range(4):
        sigma = np.random.uniform(0.07, 0.35) * n
        init_candidates.append(generate_gaussian(n, sigma))

    # Beta distributions (including heavy‑tailed cases)
    for (a, b) in [(0.5, 0.5), (0.8, 0.8), (1.0, 1.0), (2.0, 2.0), (3.0, 1.0)]:
        init_candidates.append(generate_beta(n, a, b))
    for _ in range(8):
        a = np.random.uniform(0.3, 3.0)
        b = np.random.uniform(0.3, 3.0)
        init_candidates.append(generate_beta(n, a, b))

    # --------------------------------------------------------------
    # 2. Evaluate starters and keep the best.
    # --------------------------------------------------------------
    best_seq = None
    best_val = float("inf")
    for idx, cand in enumerate(init_candidates):
        cand = project_to_simplex(cand)
        val = compute_score(cand)
        print(f"Starter {idx:02d}: score = {val:.6f}")
        if val < best_val:
            best_val = val
            best_seq = cand.copy()
    print(f"Best starter score = {best_val:.6f}")

    # --------------------------------------------------------------
    # 3. Smooth Adam optimisation (majority of the remaining budget).
    # --------------------------------------------------------------
    smooth_deadline = deadline - 200.0  # reserve time for polishing phases
    if time.time() < smooth_deadline:
        remaining = smooth_deadline - time.time()
        # Rough estimate: ~0.0003 s per iteration ⇒ ~3000 it/s
        max_iters = int(min(2_000_000, remaining * 3000))
        print(f"Smooth Adam phase: up to {max_iters} iterations.")
        a_smooth, smooth_score = smooth_adam_optimize(
            best_seq,
            time_deadline=smooth_deadline,
            max_iters=max_iters,
            beta_start=5.0,
            beta_end=5000.0,
            step0=0.9,
        )
        if smooth_score < best_val:
            best_val = smooth_score
            best_seq = a_smooth.copy()
            print(f" Smooth phase improved score to {best_val:.6f}")

    # --------------------------------------------------------------
    # 4. Exact sub‑gradient descent polishing.
    # --------------------------------------------------------------
    subgrad_deadline = deadline - 80.0
    if time.time() < subgrad_deadline:
        remaining = subgrad_deadline - time.time()

        a_sub, sub_score = subgradient_optimize(
            best_seq,

    # --------------------------------------------------------------

    restart_deadline = deadline - 30.0

        base = best_seq.copy()

            best_seq.tolist(),

        )

    # --------------------------------------------------------------
    if best_seq is None:

    # --------------------------------------------------------------
    final_score = evaluate_sequence(best_seq.tolist())
    print(f"\nSearch finished. Final reported score = {final_score:.6f}")

    # Return a plain list of floats (already on the simplex)
    return best_seq.tolist()
