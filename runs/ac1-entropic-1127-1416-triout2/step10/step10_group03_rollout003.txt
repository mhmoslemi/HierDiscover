def propose_candidate(seed=12345, budget_s=1000):
    np.random.seed(seed)
    start=time.time()
    deadline=start+budget_s-1

    n = 4096 if (deadline-start)>800 else 2048
    # generate seeds
    seeds = []
    # uniform
    seeds.append(np.ones(n)/n)
    # deterministic shape sin^2:
    seeds.append(generate_deterministic(n))
    # beta variants p in [-3,3, step 0.75]
    for p in np.linspace(-3,3,13):
        seeds.append(generate_beta_family(n,p))
    # random Dirichlet concentrations
    for conc in [0.2,0.5,0.8,1.0,2.0,5.0]:
        for _ in range(2):
            seeds.append(random_dirichlet(n, conc))
    # random spikes
    for k in [20,40,80,120]:
        seeds.append(random_spike(n,k))
    # optional fallback from height_sequence_1 if present
    try:
        fallback = [float(x) for x in height_sequence_1]  # type: ignore
        if len(fallback) == n:
            seeds.append(np.array(fallback,dtype=float)/np.sum(fallback))
    except Exception:
        pass

    best_val = float('inf')
    best_seq = None
    for s in seeds:
        s = np.asarray(s,float)
        s = project_to_simplex(s)
        val = compute_score(s)
        if val < best_val:
            best_val = val
            best_seq = s.copy()
            print(f"Seed found better: {val:.6f}")

    cur_a = best_seq
    cur_val = best_val

    # Phase 1: Adam gradient descent with softmax surrogate
    phase1_deadline = start + (deadline-start)*0.6
    cur_a, cur_val = adam_softmax_descent(cur_a, cur_val, phase1_deadline)
    if cur_val < best_val:
        best_val = cur_val; best_seq = cur_a.copy()
        print(f"After Adam GD best {best_val:.6f}")

    # Phase2: subgradient refine
    if time.time() < deadline-30:
        phase2_deadline = deadline - 30
        cur_a, cur_val = subgradient_refine(cur_a, cur_val, phase2_deadline)
        if cur_val < best_val:
            best_val = cur_val; best_seq = cur_a.copy()
            print(f"After subgrad best {best_val:.6f}")

    # Phase3: hill climb
    if time.time() < deadline - 1:
        cur_a, cur_val = hill_climb(cur_a, deadline)
        if cur_val < best_val:
            best_val = cur_val; best_seq = cur_a.copy()
            print(f"After hill climb best {best_val:.6f}")

    final_seq = best_seq.tolist()
    final_score = evaluate_sequence(final_seq)
    print(f"Final reported score {final_score:.6f}")
    return final_seq

"""
Advanced optimisation for the step‑function problem.

Goal:
    minimise 2·n·max(conv(a,a)) / (sum(a)**2)
    (the evaluator clamps to non‑negative numbers and rescales to
    [0,1000] and rejects near‑zero sums).

Strategy (≈1000 s budget)

0️⃣  Seed generation – a diverse collection of hand‑crafted and random
    probability vectors (uniform, sine‑squared, Beta‑family,
    Dirichlet draws, sparse spikes).  The best seed is kept.

1️⃣  Adam‑based soft‑max gradient descent.
    We minimise the smooth surrogate
        Lβ(a)=log Σ exp(β·conv(a,a))
    using the Adam optimiser with a linear β‑schedule
    (β∈[5,10000]).  After each step we project onto the probability
    simplex.  The best iterate (according to the true objective) is
    retained.

2️⃣  Projected sub‑gradient descent on the true max‑objective.
    The sub‑gradient is obtained from the index attaining the current
    maximum of the convolution.  A decreasing step‑size (∝1/√t) is
    used.  Random jitter is added every 1000 steps to escape flat
    plateaus.

3️⃣  Stochastic hill‑climb polishing.
    Small random swaps / mass transfers are kept only when they improve
    the exact score.  The step‑size adapts depending on recent success.

The whole pipeline respects the 1000 s wall‑clock budget, prints
progress information and finally returns a plain ``list[float]`` that
can be passed to ``evaluate_sequence``.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Low‑level utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """
    Linear convolution using a real FFT (full length = len(a)+len(b)-1).
    """
    m = a.shape[0] + b.shape[0] - 1
    n_fft = next_pow2(m)
    A = np.fft.rfft(a, n=n_fft)
    B = np.fft.rfft(b, n=n_fft)
    C = A * B
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


def compute_score(a: np.ndarray) -> float:
    """
    Exact objective for a probability vector (sum = 1):
        2·n·max(conv(a,a)).
    """
    n = a.shape[0]
    conv = convolution_fft(a, a)
    return 2.0 * n * float(conv.max())


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of `v` onto the probability simplex
    {x | x_i ≥ 0, Σ x_i = 1}.
    """
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / v.shape[0])

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, v.shape[0] + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.where(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


# ----------------------------------------------------------------------
# Seed generators
# ----------------------------------------------------------------------


def generate_deterministic(n: int) -> np.ndarray:
    """Sin^2 shape a_i = sin^2(pi·i/(n+1)), normalised to sum 1."""
    i = np.arange(1, n + 1, dtype=float)
    a = np.sin(np.pi * i / (n + 1.0)) ** 2
    a /= a.sum()
    return a


def generate_beta_family(n: int, p: float) -> np.ndarray:
    """
    Symmetric polynomial shape a_i ∝ ((i+1)*(n-i))**p.
    Positive p concentrates in the centre, negative p pushes mass to the edges.
    """
    i = np.arange(n, dtype=float)
    w = ((i + 1.0) * (n - i)) ** p
    w[w < 0] = 0.0
    a = w / w.sum()
    return a


def random_dirichlet(n: int, conc: float = 1.0) -> np.ndarray:
    """Draw a random probability vector from Dirichlet(conc)."""
    if conc == 1.0:
        a = np.random.rand(n)
    else:
        a = np.random.gamma(conc, 1.0, size=n)
    a /= a.sum()
    return a


def random_spike(n: int, k: int = 70) -> np.ndarray:
    """
    Very sparse vector: choose `k` distinct positions and assign
    random positive weights, then normalise.
    """
    idx = np.random.choice(n, size=min(k, n), replace=False)
    w = np.random.rand(len(idx))
    a = np.zeros(n, dtype=float)
    a[idx] = w
    a /= a.sum()
    return a

# Phase 1 – Adam + soft‑max surrogate
# ----------------------------------------------------------------------


def adam_softmax_descent(
    a: np.ndarray,
    start_score: float,
    deadline: float,
    beta_start: float = 5.0,
    beta_end: float = 10000.0,
    lr: float = 0.02,
    beta1: float = 0.9,
    beta2: float = 0.999,
    eps: float = 1e-8,
) -> tuple[np.ndarray, float]:
    """
    Projected Adam optimiser on the smooth surrogate
        Lβ(a) = log Σ exp(β·conv(a,a)).
    β is increased linearly from ``beta_start`` to ``beta_end``.
    """
    n = a.shape[0]
    cur = a.copy()
    best = cur.copy()
    best_score = start_score

    m = np.zeros_like(cur)
    v = np.zeros_like(cur)
    it = 0
    start = time.time()

    while time.time() < deadline:
        it += 1

        # exact objective (for bookkeeping)
        conv = convolution_fft(cur, cur)
        max_conv = conv.max()
        cur_score = 2.0 * n * float(max_conv)
        if cur_score < best_score:
            best_score = cur_score
            best = cur.copy()

        # schedule β
        elapsed = time.time() - start
        frac = min(elapsed / (deadline - start), 1.0)
        beta = beta_start + frac * (beta_end - beta_start)

        # soft‑max weights (stable)
        shifted = conv - max_conv                # max(conv) → 0
        exp_vals = np.exp(beta * shifted)       # ≤ 1
        w = exp_vals / exp_vals.sum()           # length 2n‑1, sums to 1

        # gradient of the surrogate
        # grad_i = 2·β· Σ_k w_k·a_{k‑i}  → correlation of w with a
        grad = 2.0 * beta * np.correlate(w, cur, mode="valid")

        # Adam update (note: grad may be large, Adam normalises it)
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        step = lr * m_hat / (np.sqrt(v_hat) + eps)
        nxt = cur - step
        cur = project_to_simplex(nxt)

        # occasional tiny jitter to avoid stagnation
        if it % 5000 == 0:
            jitter = np.random.randn(n) * (lr * 0.005 / np.sqrt(it))
            cur += jitter
            cur = project_to_simplex(cur)

    return best, best_score

# Phase 2 – projected sub‑gradient refinement
# ----------------------------------------------------------------------


def subgradient_refine(
    a: np.ndarray,
    start_score: float,
    deadline: float,
    max_iter: int = 400_000,
    step0: float = 0.07,
) -> tuple[np.ndarray, float]:
    """Projected sub‑gradient descent on the true max‑objective."""
    n = a.shape[0]
    cur = a.copy()
    best = cur.copy()
    best_score = start_score

    for t in range(1, max_iter + 1):
        if time.time() >= deadline:
            break

        conv = convolution_fft(cur, cur)
        k = int(np.argmax(conv))

        # sub‑gradient g_i = 2·a_{k-i} for indices where 0≤k-i<n, else 0
        idx = np.arange(n)
        j = k - idx
        mask = (j >= 0) & (j < n)
        g = np.zeros_like(cur)
        g[mask] = 2.0 * cur[j[mask]]

        step = step0 / np.sqrt(t)
        cur -= step * g
        cur = project_to_simplex(cur)

        cur_score = compute_score(cur)
        if cur_score < best_score:
            best_score = cur_score
            best = cur.copy()

        # occasional jitter to shake out of plateaus
        if t % 1000 == 0:
            jitter = np.random.randn(n) * (step0 * 0.004 / np.sqrt(t))
            cur += jitter
            cur = project_to_simplex(cur)

    return best, best_score

# Phase 3 – greedy hill‑climb polishing
# ----------------------------------------------------------------------


def hill_climb(
    a: np.ndarray,
    deadline: float,
    max_steps: int = 1_200_000,
) -> tuple[np.ndarray, float]:
    """Stochastic moves kept only when they improve the exact score."""
    n = a.shape[0]
    best = a.copy()
    best_score = compute_score(best)
    step_scale = 0.08

    steps = 0
    while steps < max_steps and time.time() < deadline:
        cand = best.copy()
        if np.random.rand() < 0.5:
            # simple swap of two entries
            i, j = np.random.randint(n, size=2)
            cand[i], cand[j] = cand[j], cand[i]
        else:
            # transfer a small fraction from i to j
            i, j = np.random.randint(n, size=2)
            if cand[i] > 0.0 and i != j:
                delta = step_scale * cand[i]
                cand[i] -= delta
                cand[j] += delta
                if cand[i] < 0.0:
                    cand[i] = 0.0

        # Re‑normalise (numeric drift)
        cand /= cand.sum()
        cur_score = compute_score(cand)

        if cur_score < best_score:
            best_score = cur_score
            best = cand
            step_scale = min(step_scale * 1.07, 0.5)
        else:
            step_scale *= 0.96

        steps += 1

    return best, best_score

    print(f"Using sequence length = {n}")

    # --------------------------------------------------------------
    # 2. Generate a diverse set of seeds and keep the best one
    # --------------------------------------------------------------
    seeds = []

    # hand‑crafted shapes
    seeds.append(generate_deterministic(n))
    for p in np.linspace(-3.0, 3.0, 13):
        seeds.append(generate_beta_family(n, p))

    # random Dirichlet draws with various concentrations
    for conc in (0.2, 0.5, 0.8, 1.0, 2.0, 5.0):
        for _ in range(2):
            seeds.append(random_dirichlet(n, conc))

    # sparse spikes – often a good starting point
    for k in (20, 40, 80, 120):
        seeds.append(random_spike(n, k))

    # uniform – the theoretical baseline
    seeds.append(np.full(n, 1.0 / n, dtype=float))

    # include fallback if it matches the current length
    if best_seq is not None and len(best_seq) == n:
        seeds.append(np.array(best_seq, dtype=float) / np.sum(best_seq))

    # evaluate all seeds (exact objective) and keep the best one
    for s in seeds:
        s = np.asarray(s, dtype=float)
        s = project_to_simplex(s)
        val = compute_score(s)
        if val < best_val:
            best_val = val
            best_seq = s.tolist()
            print(f"Initial seed improvement: score = {best_val:.6f}")

    # --------------------------------------------------------------
    # 3. Main optimisation loop
    # --------------------------------------------------------------
    cur_a = np.array(best_seq, dtype=float)
    cur_a = cur_a / cur_a.sum()
    cur_score = compute_score(cur_a)

    # ------------------- Phase 1 – Adam soft‑max GD -----------------
    phase1_deadline = start_time + total_time * 0.58
    print("\n[Phase 1] Adam soft‑max gradient descent")
    cur_a, cur_score = adam_softmax_descent(
        cur_a,
        cur_score,
        phase1_deadline,
        beta_start=5.0,
        beta_end=10000.0,
        lr=0.02,
    )
    if cur_score < best_val:
        best_val = cur_score
    print(f"After Phase 1: best score = {best_val:.6f}")

    # ------------------- Phase 2 – sub‑gradient refinement ----------
    if time.time() < deadline - 40.0:
        print("\n[Phase 2] Sub‑gradient refinement")
        phase2_deadline = deadline - 40.0
        cur_a, cur_score = subgradient_refine(
            cur_a, cur_score, phase2_deadline, max_iter=800_000, step0=0.07
        )
        if cur_score < best_val:
            best_val = cur_score
            print(f"After Phase 2: best score = {best_val:.6f}")

    # ------------------- Phase 3 – hill‑climb polishing ------------
    if time.time() < deadline - 2.0:
        print("\n[Phase 3] Hill‑climb polishing")
        cur_a, cur_score = hill_climb(cur_a, deadline, max_steps=1_500_000)
        if cur_score < best_val:
            best_val = cur_score
            print(f"After Phase 3: best score = {best_val:.6f}")

    # --------------------------------------------------------------
    # 4. Final conversion and official evaluation
    # --------------------------------------------------------------
    final_seq = cur_a.tolist()
    final_val = evaluate_sequence(final_seq)
    print(f"\nFinished search. Final reported score = {final_val:.6f}")

    return final_seq
