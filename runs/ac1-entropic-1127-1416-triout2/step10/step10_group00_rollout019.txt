"""
Search for a step‑function (non‑negative vector) that minimizes
evaluate_sequence().  The new strategy focuses on constructing a *sparse*
support set with a very small maximum number of ordered pairs
(i + j) that share the same sum.  For a set S of indices, the convolution
value at offset k is exactly the number of ordered pairs (i,j)∈S×S with
i+j=k, multiplied by the square of the uniform weight (1/|S|)².
If the maximum ordered‑pair count “maxc” is low, the objective

    2·n·maxc / |S|²   (with sum = 1)

can be far below the uniform bound 2.  We therefore

1. greedily build many random index sets respecting a bound on
   unordered pair collisions (maxc_target//2).  This yields
   low‐collision “Sidon‑like’’ sets with thousands of candidates.
2. keep the best candidate seen (smallest objective).
3. optionally polish the weights of the best set with a short
   projected sub‑gradient descent.
4. return the final sequence as a plain Python list.

All steps respect the global 1000 s wall‑clock budget.
"""

import time
import numpy as np
from typing import List, Tuple

# ----------------------------------------------------------------------
# Core convolution utilities (fast via FFT)
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full aperiodic convolution using a real FFT."""
    m = a.size + b.size - 1
    nfft = next_pow2(m)
    A = np.fft.rfft(a, nfft)
    B = np.fft.rfft(b, nfft)
    C = A * B
    conv = np.fft.irfft(C, nfft)[:m]
    return conv


def compute_score(a: np.ndarray) -> float:
    """Exact objective for a vector with sum = 1."""
    n = a.shape[0]
    conv = convolution_fft(a, a)
    return 2.0 * n * float(conv.max())


# ----------------------------------------------------------------------
# Simplex projection (used by the sub‑gradient routine)
# ----------------------------------------------------------------------


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / v.shape[0])

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, v.shape[0] + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.where(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Projected sub‑gradient descent (fine‑tuning of weights)
# ----------------------------------------------------------------------


def subgrad_optimize(start_a: np.ndarray,
                     deadline: float,
                     max_iters: int = 500_000,
                     step0: float = 0.07) -> Tuple[np.ndarray, float]:
    """
    Sub‑gradient descent for the exact max‑convolution objective.
    Keeps the iterate on the simplex by Euclidean projection.
    """
    n = start_a.shape[0]
    a = start_a.astype(np.float64).copy()
    best_a = a.copy()
    best_score = compute_score(a)

    for t in range(1, max_iters + 1):
        if time.time() >= deadline:
            break

        conv = convolution_fft(a, a)
        k = int(np.argmax(conv))

        # sub‑gradient of the max‑convolution entry (ordered pairs)
        idx = np.arange(n)
        j = k - idx
        mask = (j >= 0) & (j < n)
        g = np.zeros_like(a)
        g[mask] = 2.0 * a[j[mask]]

        step = step0 / np.sqrt(t)
        a = project_to_simplex(a - step * g)

        if t % 2000 == 0:
            sc = compute_score(a)
            if sc < best_score:
                best_score = sc
                best_a = a.copy()
                print(f"[{time.time():.1f}] sub‑grad improved to {best_score:.6f}")

    sc = compute_score(a)
    if sc < best_score:
        best_score = sc
        best_a = a.copy()
    return best_a, best_score


# ----------------------------------------------------------------------
# Greedy construction of low‑collision index sets
# ----------------------------------------------------------------------


def greedy_set_maxc(n: int, maxc_target: int) -> Tuple[np.ndarray, int]:
    """
    Build a set S ⊂ {0,…,n‑1} such that the *ordered* pair count per sum
    does not exceed maxc_target.  The construction is greedy:
    a random permutation of indices is scanned and an index is kept iff
    adding it would keep every unordered sum ≤ maxc_target//2.
    Returns the index array and the achieved ordered max multiplicity.
    """
    limit_unordered = maxc_target // 2          # max unordered pairs per sum
    perm = np.random.permutation(n)
    counts = np.zeros(2 * n - 1, dtype=np.int16)  # unordered pair histogram
    S = []

    for idx in perm:
        if not S:
            # first element always OK (self‑sum will be 1 ≤ limit_unordered)
            if counts[2 * idx] + 1 > limit_unordered:
                continue
            S.append(idx)
            counts[2 * idx] += 1
            continue

        existing = np.array(S, dtype=np.int32)
        sums = idx + existing                     # unordered sums with earlier elems

        # check the unordered limit for all new sums
        if (counts[sums] + 1 > limit_unordered).any():
            continue
        # self‑sum also counted as unordered
        if counts[2 * idx] + 1 > limit_unordered:
            continue

        # accept the element
        S.append(idx)
        counts[sums] += 1
        counts[2 * idx] += 1

    # compute the *ordered* maximum multiplicity for verification
    ind = np.array(S, dtype=np.int32)
    sums_full = ind[:, None] + ind[None, :]
    ordered_counts = np.bincount(sums_full.ravel(), minlength=2 * n - 1)
    maxc_ordered = int(ordered_counts.max())
    return ind, maxc_ordered
