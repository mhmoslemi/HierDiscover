"""
step‑function optimiser for ``evaluate_sequence``

Algorithm
---------
1. Determine a problem size `n` (use the provided ``height_sequence_1`` if
   available, otherwise default to 2048).
2. Build a modest pool of initial candidates:
   * uniform, central‑binomial, a few power‑law and geometric families,
   * several random Dirichlet draws (different concentration parameters).
   Keep the best of these as the starting point.
3. **Smooth LSE‑Adam** – minimise a log‑sum‑exp surrogate of
   `max(conv(a,a))` with an annealed temperature `β` that grows from ≈1
   to ≈1000.  This phase gets roughly half of the time budget.
4. **Projected sub‑gradient descent** on the true `max` objective
   (the same approach as before) for ~20 % of the time.
5. **Simulated annealing** – random mass transfers between two positions,
   Metropolis acceptance with a temperature schedule.  This helps escape
   shallow local minima and consumes another ~20 % of the budget.
6. **Stochastic hill‑climb** (the cheap final polishing from the
   previous version) for the remaining time.
7. Return the best sequence (already normalised on the simplex) as a
   plain Python list.  The final reported score is obtained via the
   supplied ``evaluate_sequence``.
"""

def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """
    Full linear convolution using real FFT.
    Returns an array of length len(a)+len(b)-1.
    """
    m = a.shape[0] + b.shape[0] - 1
    n_fft = next_pow2(m)
    A = np.fft.rfft(a, n=n_fft)
    B = np.fft.rfft(b, n=n_fft)
    C = A * B
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


def compute_score(a: np.ndarray) -> float:
    """
    Exact objective for a vector already on the probability simplex (sum = 1):
        2·n·max(conv(a,a))
    """
    n = a.shape[0]
    conv = convolution_fft(a, a)
    return 2.0 * n * float(conv.max())


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of v onto the probability simplex:
        sum(v) = 1, v >= 0.
    """
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / v.shape[0])

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, v.shape[0] + 1)
    cond = u - cssv / ind > 0
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

# Analytic families for initial guesses
# ----------------------------------------------------------------------


def generate_central_binomial(n: int) -> np.ndarray:
    """a_i = C(2i,i) / 4^i, normalised to sum 1 (monotone decreasing)."""
    a = np.empty(n, dtype=np.float64)
    a[0] = 1.0
    for i in range(1, n):
        a[i] = a[i - 1] * (2 * i - 1) / (2.0 * i)
    return project_to_simplex(a)


def generate_powerlaw(n: int, p: float) -> np.ndarray:
    """a_i ∝ (i+1)^(-p), normalised to sum 1."""
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = idx ** (-p)
    return project_to_simplex(a)


def generate_geometric(n: int, r: float) -> np.ndarray:
    """Geometric distribution a_i = (1‑r)·r^i, normalised (exactly sum 1 if r<1)."""
    i = np.arange(n, dtype=np.float64)
    a = (1.0 - r) * (r ** i)
    return project_to_simplex(a)


# ----------------------------------------------------------------------
# Smooth (log‑sum‑exp) optimisation – Adam
# ----------------------------------------------------------------------


def smooth_gradient_optimize(
    start_a: np.ndarray,
    time_deadline: float,
    beta_start: float = 0.8,
    beta_end: float = 1000.0,
    step0: float = 1.0,
    adam_beta1: float = 0.9,
    adam_beta2: float = 0.999,
    adam_eps: float = 1e-8,
) -> tuple[np.ndarray, float]:
    """
    Adam optimiser for the smoothed max‑convolution objective.
    Uses a log‑sum‑exp surrogate with an exponentially annealed temperature.
    """
    a = start_a.astype(np.float64).copy()
    n = a.shape[0]
    m1 = np.zeros_like(a)
    m2 = np.zeros_like(a)

    best_a = a.copy()
    best_score = compute_score(a)

    it = 0
    while True:
        now = time.time()
        if now >= time_deadline:
            break
        it += 1

        # Estimate remaining time to decide whether to continue
        # (no hard stop here, the outer loop already checks the deadline).

        # temperature schedule (geometric progression)
        t_frac = it / 500_000  # we anticipate up to ~5e5 iters
        if t_frac > 1.0:
            t_frac = 1.0
        beta = beta_start * (beta_end / beta_start) ** t_frac

        # full auto‑convolution
        conv = convolution_fft(a, a)
        maxc = conv.max()

        # soft‑max weights (log‑sum‑exp)
        # avoid overflow/underflow
        exp_w = np.exp(beta * (conv - maxc))
        w = exp_w / exp_w.sum()

        # gradient of the smoothed objective:
        #   ∂/∂a  = 2 * (rev(a) * w)   (restricted to length n)
        rev_a = a[::-1]
        grad_full = convolution_fft(rev_a, w)  # length 3n‑2
        grad = 2.0 * grad_full[n - 1:n - 1 + n]

        # Adam update
        m1 = adam_beta1 * m1 + (1.0 - adam_beta1) * grad
        m2 = adam_beta2 * m2 + (1.0 - adam_beta2) * (grad * grad)
        m1_hat = m1 / (1.0 - adam_beta1 ** it)
        m2_hat = m2 / (1.0 - adam_beta2 ** it)

        # step size decays slowly with sqrt(it)
        step = step0 / np.sqrt(it)
        a_new = a - step * m1_hat / (np.sqrt(m2_hat) + adam_eps)
        a_new = project_to_simplex(a_new)

        a = a_new

        if it % 500 == 0:
            cur_score = compute_score(a)
            if cur_score < best_score:
                best_score = cur_score
                best_a = a.copy()

    final_score = compute_score(a)
    if final_score < best_score:
        best_score, best_a = final_score, a.copy()
    return best_a, best_score

# Exact sub‑gradient optimiser on the true max‑objective
# ----------------------------------------------------------------------


def subgrad_optimize(
    init_a: np.ndarray,
    time_deadline: float,
    step0: float = 0.07,
    max_iter: int = 600_000,
) -> tuple[np.ndarray, float]:
    """Projected sub‑gradient descent on 2·n·max(conv(a,a))."""
    a = init_a.astype(np.float64).copy()
    best_a = a.copy()
    best_score = compute_score(a)
    n = a.shape[0]

    for t in range(1, max_iter + 1):
        if time.time() >= time_deadline:
            break

        conv = convolution_fft(a, a)
        k = int(np.argmax(conv))

        # sub‑gradient for the active convolution entry
        idx = np.arange(n)
        j = k - idx
        mask = (j >= 0) & (j < n)
        g = np.zeros_like(a)
        g[mask] = 2.0 * a[j[mask]]

        step = step0 / np.sqrt(t)
        a_new = a - step * g
        a_new = project_to_simplex(a_new)

        cur_score = compute_score(a_new)
        if cur_score < best_score:
            best_score, best_a = cur_score, a_new.copy()

        a = a_new

        # occasional jitter to escape flat spots
        if t % 1000 == 0:
            jitter = np.random.randn(n) * (step0 * 0.004 / np.sqrt(t))
            a += jitter
            a = project_to_simplex(a)

    final_score = compute_score(a)
    if final_score < best_score:
        best_score, best_a = final_score, a.copy()
    return best_a, best_score

def simulated_annealing_optimize(
    start_a: np.ndarray,
    time_deadline: float,
    init_temp: float = 0.1,
    final_temp: float = 1e-5,
    init_step_scale: float = 0.2,
) -> tuple[np.ndarray, float]:
    """
    Metropolis annealing on the simplex.
    Moves a random fraction of mass from i → j.
    Acceptance probability: exp(-(Δ)/T) for uphill moves.
    Temperature and step scale decay exponentially.
    """
    a = start_a.astype(np.float64).copy()
    cur_score = compute_score(a)
    best_a = a.copy()
    best_score = cur_score

    n = a.shape[0]
    start_time = time.time()
    total_time = time_deadline - start_time
    if total_time <= 0:
        return best_a, best_score

    it = 0
    # exponential decay rates
    decay_rate = 0.9995  # per iteration
    T = init_temp
    step_scale = init_step_scale

    while time.time() < time_deadline:
        it += 1
        # propose move
        i, j = np.random.randint(n, size=2)
        if i == j:
            continue
        # delta is a fraction of the current mass at i
        delta = step_scale * a[i] * np.random.rand()
        if delta <= 0:
            continue
        cand = a.copy()
        cand[i] -= delta
        cand[j] += delta
        # numerical safety: ensure non‑negative (it should be by construction)
        if cand[i] < 0:
            cand[i] = 0.0
        # projection is not needed because sum is unchanged and we stay non‑negative,
        # but tiny negative may appear due to rounding, so we project anyway.
        cand = project_to_simplex(cand)

        cand_score = compute_score(cand)
        if cand_score < cur_score:
            accept = True
        else:
            # Metropolis acceptance
            accept = np.random.rand() < np.exp(-(cand_score - cur_score) / max(T, 1e-12))

        if accept:
            a = cand
            cur_score = cand_score
            if cur_score < best_score:
                best_score = cur_score
                best_a = a.copy()

        # update temperature and step scale
        T = max(final_temp, T * decay_rate)
        step_scale = max(0.001, step_scale * decay_rate)

        # occasional occasional jitter to keep diversity (every ~5000 iters)
        if it % 5000 == 0:
            # small Gaussian noise on all entries, then re‑project
            noise = np.random.randn(n) * (init_step_scale * 0.01)
            a = project_to_simplex(a + noise)
            cur_score = compute_score(a)

    return best_a, best_score

def hill_climb_permutation(
    seq,
    time_deadline,
    max_steps: int = 300_000,
) -> tuple[list[float], float]:
    """Tiny stochastic moves (swap or fractional transfer) with simplex projection."""
    a = np.array(seq, dtype=np.float64)
    n = a.shape[0]
    best_a = a.copy()
    best_score = compute_score(best_a)

    step_scale = 0.08  # initial fraction of mass moved

    steps_done = 0
    while steps_done < max_steps and time.time() < time_deadline:
        cand = best_a.copy()
        if np.random.rand() < 0.5:      # swap two entries
            i, j = np.random.randint(n, size=2)
            cand[i], cand[j] = cand[j], cand[i]
        else:                           # move a fraction from i to j
            i, j = np.random.randint(n, size=2)
            if i != j:
                delta = step_scale * best_a[i]
                cand[i] -= delta
                cand[j] += delta
                if cand[i] < 0:
                    cand[i] = 0.0
        cand = project_to_simplex(cand)
        cur_score = compute_score(cand)

        if cur_score < best_score:
            best_score = cur_score
            best_a = cand
            step_scale = min(step_scale * 1.06, 0.5)
        else:
            step_scale *= 0.97

        steps_done += 1

    return best_a.tolist(), best_score

    return best_seq.tolist()
