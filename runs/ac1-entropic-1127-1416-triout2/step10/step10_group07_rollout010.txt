"""
Propose a step‑function sequence that (approximately) minimises
`evaluate_sequence`.  The routine builds several analytic families,
draws many Dirichlet samples, keeps the best few and then refines them
using a multi‑stage Adam optimisation on a soft‑max (log‑sum‑exp)
surrogate of the max‑convolution.  The surrogate gets progressively
sharper (larger β) and finally a long entropy‑mirror sub‑gradient
descent plus a short simulated‑annealing hill‑climb are applied.
The whole search respects the supplied time budget (≈1000 s) and
returns a plain Python list of non‑negative numbers.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Low‑level utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """
    Full aperiodic convolution using FFT (real transforms).
    Returns a float64 array of length len(a)+len(b)-1.
    """
    m = a.size + b.size - 1
    nfft = next_pow2(m)
    A = np.fft.rfft(a, n=nfft)
    B = np.fft.rfft(b, n=nfft)
    C = A * B
    conv = np.fft.irfft(C, n=nfft)[:m]
    return conv


def compute_score(a: np.ndarray) -> float:
    """Exact objective for a probability vector (sum = 1)."""
    n = a.shape[0]
    conv = convolution_fft(a, a)
    return 2.0 * n * float(conv.max())


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / v.shape[0])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, v.shape[0] + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.where(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def softmax(z: np.ndarray) -> np.ndarray:
    """Numerically stable soft‑max."""
    m = np.max(z)
    e = np.exp(z - m)
    return e / e.sum()


# ----------------------------------------------------------------------
# Analytic families (shape generators)
# ----------------------------------------------------------------------


def symmetric_product_family(n: int, alpha: float) -> np.ndarray:
    """a_i ∝ [(i+½)(n‑½‑i)]^(‑α), symmetric, normalised to sum = 1."""
    i = np.arange(n, dtype=np.float64)
    logprod = np.log(i + 0.5) + np.log(n - 0.5 - i)
    w = np.exp(-alpha * logprod)
    w[np.isinf(w)] = np.finfo(np.float64).max
    a = w / w.sum()
    return a


def left_heavy_power_family(n: int, beta: float) -> np.ndarray:
    """a_i ∝ (i+½)^(‑β), monotone decreasing, sum = 1."""
    i = np.arange(n, dtype=np.float64)
    w = np.exp(-beta * np.log(i + 0.5))
    a = w / w.sum()
    return a


def exponential_decay_family(n: int, gamma: float) -> np.ndarray:
    """a_i ∝ exp(-γ·i), monotone decreasing, sum = 1."""
    i = np.arange(n, dtype=np.float64)
    w = np.exp(-gamma * i)
    a = w / w.sum()
    return a


def beta_family(n: int, p: float, q: float) -> np.ndarray:
    """
    a_i ∝ (i+1)^p·(n-i)^q, both ends can be tuned, sum = 1.
    (p,q ≥ 0 give centre‑heavy, p,q < 0 give edge‑heavy.)
    """
    i = np.arange(n, dtype=np.float64)
    w = np.power(i + 1.0, p) * np.power(n - i, q)
    a = w / w.sum()
    return a


# ----------------------------------------------------------------------
# Random initialisers
# ----------------------------------------------------------------------


def random_dirichlet_start(n: int, conc: float) -> np.ndarray:
    """Draw a symmetric Dirichlet(α=conc) vector and normalise to sum = 1."""
    vec = np.random.gamma(conc, 1.0, size=n)
    vec = vec / vec.sum()
    return vec


def random_sparse_start(n: int) -> np.ndarray:
    """Very sparse (≈ 5 % non‑zeros) simplex‑projected vector."""
    sparsity = np.random.uniform(0.03, 0.07)
    k = max(2, int(round(sparsity * n)))
    idx = np.random.choice(n, size=k, replace=False)
    a = np.zeros(n, dtype=np.float64)
    a[idx] = 1.0 / k
    return project_to_simplex(a)


# ----------------------------------------------------------------------
# Smoothed optimisation – Adam on the log‑sum‑exp surrogate
# ----------------------------------------------------------------------


def adam_smooth_optimize(
    start_a: np.ndarray,
    deadline: float,
    max_iters: int = 200_000,
    beta_start: float = 5.0,
    beta_end: float = 2_000.0,
    lr: float = 0.018,
) -> tuple[np.ndarray, float]:
    """
    Minimise f_β(a) = (1/β)·log Σ_k exp(β·(a*a)_k) by Adam on the soft‑max
    parametrisation a = softmax(z).  β is increased exponentially from
    beta_start to beta_end.
    Returns the best vector seen and its exact score.
    """
    eps = 1e-12
    z = np.log(start_a + eps)          # initialise from log‑softmax
    m = np.zeros_like(z)
    v = np.zeros_like(z)

    beta1 = 0.9
    beta2 = 0.999
    eps_adam = 1e-12

    best_a = softmax(z)
    best_score = compute_score(best_a)

    for it in range(1, max_iters + 1):
        if time.time() >= deadline:
            break

        # β schedule (exponential)
        t = it / max_iters
        beta = beta_start * (beta_end / beta_start) ** t

        a = softmax(z)

        conv = convolution_fft(a, a)               # full auto‑convolution
        maxc = conv.max()
        # stabilized soft‑max over the 2n‑1 entries
        exp_w = np.exp(beta * (conv - maxc))
        w = exp_w / exp_w.sum()                     # weights that approximate arg‑max

        # gradient of the smoothed max w.r.t. a  (see derivation in comment block)
        a_rev = a[::-1]
        conv_rev = convolution_fft(a_rev, w)       # length 3n‑2
        grad_a = 2.0 * conv_rev[a.shape[0] - 1 : a.shape[0] - 1 + a.shape[0]]

        # back‑prop through soft‑max
        dot_ag = np.dot(a, grad_a)
        grad_z = a * grad_a - dot_ag * a

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad_z
        v = beta2 * v + (1.0 - beta2) * (grad_z ** 2)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        z -= lr * m_hat / (np.sqrt(v_hat) + eps_adam)

        # occasional exact score check
        if it % 500 == 0:
            a_tmp = softmax(z)
            sc = compute_score(a_tmp)
            if sc < best_score:
                best_score = sc
                best_a = a_tmp.copy()
                print(f"[{time.time():.1f}]   Adam iter {it:7d} – new best = {best_score:.6f}")

    # final check
    a_fin = softmax(z)
    sc_fin = compute_score(a_fin)
    if sc_fin < best_score:
        best_score = sc_fin
        best_a = a_fin.copy()
    return best_a, best_score


# ----------------------------------------------------------------------
# Mirror (entropy) sub‑gradient descent on the exact max‑objective
# ----------------------------------------------------------------------


def mirror_subgrad_optimize(
    start_a: np.ndarray,
    deadline: float,
    step0: float = 0.9,
    check_interval: int = 2000,
) -> tuple[np.ndarray, float]:
    """
    Entropy‑mirror (multiplicative) sub‑gradient descent for
    f(a) = 2·n·max(a*a).  Step size η_t = step0 / sqrt(t).
    """
    n = start_a.shape[0]
    a = start_a.astype(np.float64).copy()
    best_a = a.copy()
    best_score = compute_score(a)

    t = 0
    while time.time() < deadline:
        t += 1
        conv = convolution_fft(a, a)
        k = int(np.argmax(conv))                     # active shift

        # sub‑gradient (ordered pairs) for the active shift
        idx = np.arange(n)
        j = k - idx
        mask = (j >= 0) & (j < n)
        g = np.zeros_like(a)
        g[mask] = 2.0 * a[j[mask]]

        eta = step0 / np.sqrt(t)
        a = a * np.exp(-eta * g)                     # multiplicative update
        s = a.sum()
        if s == 0.0:
            a = np.full_like(a, 1.0 / n)
        else:
            a = a / s

        if t % check_interval == 0:
            sc = compute_score(a)
            if sc < best_score:
                best_score = sc
                best_a = a.copy()
                print(f"[{time.time():.1f}]   Mirror step {t:7d} – new best = {best_score:.6f}")

    sc = compute_score(a)
    if sc < best_score:
        best_score = sc
        best_a = a.copy()
    return best_a, best_score


# ----------------------------------------------------------------------
# Tiny simulated‑annealing hill‑climb (SA)
# ----------------------------------------------------------------------


def hill_climb_sa(
    start_a: np.ndarray,
    deadline: float,
    max_steps: int = 1_200_000,
    init_temp_factor: float = 1e-4,
) -> tuple[np.ndarray, float]:
    """
    Low‑temperature simulated‑annealing walk on the simplex.
    Moves: swap two entries or transfer a fraction of mass.
    """
    n = start_a.size
    cur_a = start_a.copy()
    cur_score = compute_score(cur_a)

    best_a = cur_a.copy()
    best_score = cur_score

    T0 = best_score * init_temp_factor
    T = T0
    step_scale = 0.12

    for step in range(1, max_steps + 1):
        if time.time() >= deadline:
            break

        cand = cur_a.copy()
        if np.random.rand() < 0.5:        # swap two entries
            i, j = np.random.randint(n, size=2)
            cand[i], cand[j] = cand[j], cand[i]
        else:                             # transfer a fraction of mass
            i, j = np.random.randint(n, size=2)
            if i != j:
                delta = step_scale * cur_a[i]
                cand[i] -= delta
                cand[j] += delta
                if cand[i] < 0.0:
                    cand[i] = 0.0

        cand = cand / cand.sum()          # (re‑)normalise
        cand_score = compute_score(cand)
        delta = cand_score - cur_score

        # Metropolis acceptance
        if delta < 0.0 or np.random.rand() < np.exp(-delta / (T + 1e-18)):
            cur_a = cand
            cur_score = cand_score
            if cur_score < best_score:
                best_score = cur_score
                best_a = cur_a.copy()
                step_scale = min(step_scale * 1.07, 0.5)
        else:
            step_scale *= 0.97

        # linear cooling
        T = T0 * (1.0 - step / max_steps)

    return best_a, best_score


# ----------------------------------------------------------------------
# Main driver – propose_candidate
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 12345, budget_s: int = 1000, **_):
    """
    Search for a non‑negative step‑function (list of floats) that
    minimises ``evaluate_sequence``.  The algorithm respects the given
    time budget (≈ 1000 s) and returns a Python list.
    """
    np.random.seed(seed)
    start_time = time.time()
    # leave a couple of seconds as safety margin for the harness
    deadline = start_time + budget_s - 2.0

    # -------------------------------------------------------------
    # Choose a length – larger n gives a lower theoretical lower bound.
    # -------------------------------------------------------------
    if budget_s >= 900:
        n = 32768
    elif budget_s >= 600:
        n = 16384
    elif budget_s >= 400:
        n = 8192
    else:
        n = 4096
    print(f"[{time.time():.1f}] Using length n = {n}")

    # -------------------------------------------------------------
    # Helper for global best tracking
    # -------------------------------------------------------------
    best_seq = None
    best_score = float("inf")

    def maybe_update(vec: np.ndarray, tag: str = ""):
        nonlocal best_seq, best_score
        sc = compute_score(vec)
        if sc < best_score:
            best_score = sc
            best_seq = vec.tolist()
            print(f"[{time.time():.1f}] new best {tag:<20} = {best_score:.6f}")

    # -------------------------------------------------------------
    # 0. Optional fallback from a previously known construction
    # -------------------------------------------------------------
    try:
        fb_vec = [float(x) for x in height_sequence_1]   # type: ignore
        fb_sc = evaluate_sequence(fb_vec)
        print(f"[{time.time():.1f}] fallback score = {fb_sc:.6f}")
        if fb_sc < best_score:
            best_score = fb_sc
            best_seq = fb_vec
    except Exception:
        pass

    # -------------------------------------------------------------
    # 1. Exhaustive scanning of analytic families (extended ranges)
    # -------------------------------------------------------------
    uniform = np.full(n, 1.0 / n)
    maybe_update(uniform, "uniform")

    # symmetric product (edge‑heavy) – α from 0 to 12
    for alpha in np.linspace(0.0, 12.0, 49):
        if time.time() >= deadline - 30.0:
            break
        vec = symmetric_product_family(n, alpha)
        maybe_update(vec, f"sym‑α={alpha:.2f}")

    # left‑heavy power – β from 0 to 8
    for beta in np.linspace(0.0, 8.0, 41):
        if time.time() >= deadline - 30.0:
            break
        vec = left_heavy_power_family(n, beta)
        maybe_update(vec, f"pow‑β={beta:.2f}")

    # exponential decay – γ from 0 to 12
    for gamma in np.linspace(0.0, 12.0, 49):
        if time.time() >= deadline - 30.0:
            break
        vec = exponential_decay_family(n, gamma)
        maybe_update(vec, f"exp‑γ={gamma:.2f}")

    # beta‑family (two‑parameter) – coarse grid
    for p in np.linspace(-2.0, 2.0, 13):
        for q in np.linspace(-2.0, 2.0, 13):
            if time.time() >= deadline - 30.0:
                break
            vec = beta_family(n, p, q)
            maybe_update(vec, f"beta‑p={p:.1f} q={q:.1f}")

    # -------------------------------------------------------------
    # 2. Random Dirichlet draws (broad concentration sweep)
    # -------------------------------------------------------------
    conc_vals = np.logspace(-3, 2, 30)  # 0.001 … 100
    for conc in conc_vals:
        for _ in range(4):
            if time.time() >= deadline - 30.0:
                break
            vec = random_dirichlet_start(n, conc)
            maybe_update(vec, f"dir‑α={conc:.3f}")

    # -------------------------------------------------------------
    # 3. Populate a small elite pool (top‑K) for later recombination
    # -------------------------------------------------------------
    K = 15
    elite = []  # list of (score, vector) sorted descending (worst first)

    def push_elite(vec: np.ndarray, sc: float):
        """Insert a candidate into the elite pool, keep size ≤ K."""
        if len(elite) < K:
            elite.append((sc, vec))
            elite.sort(key=lambda x: x[0], reverse=True)   # worst first
        else:
            # replace worst if better
            if sc < elite[0][0]:
                elite[0] = (sc, vec)
                elite.sort(key=lambda x: x[0], reverse=True)

    # Fill elite from all vectors examined so far
    # (We have already called maybe_update for each, now duplicate into pool)
    # Re‑evaluate a few promising families quickly
    for alpha in np.linspace(0.0, 12.0, 13):
        vec = symmetric_product_family(n, alpha)
        sc = compute_score(vec)
        push_elite(vec, sc)
    for beta in np.linspace(0.0, 8.0, 9):
        vec = left_heavy_power_family(n, beta)
        sc = compute_score(vec)
        push_elite(vec, sc)
    for gamma in np.linspace(0.0, 12.0, 9):
        vec = exponential_decay_family(n, gamma)
        sc = compute_score(vec)
        push_elite(vec, sc)

    # -------------------------------------------------------------
    # 4. Genetic‑style recombination + occasional mutation
    # -------------------------------------------------------------
    recomb_deadline = deadline - 200.0
    it = 0
    while time.time() < recomb_deadline:
        it += 1
        # pick two parents (biased towards better scores)
        scores = np.array([s for s, _ in elite])
        probs = np.exp(-scores)                      # lower score → higher prob
        probs /= probs.sum()
        parents_idx = np.random.choice(len(elite), size=2, replace=False, p=probs)
        p1 = elite[parents_idx[0]][1]
        p2 = elite[parents_idx[1]][1]

        lam = np.random.rand()
        child = lam * p1 + (1.0 - lam) * p2

        # occasional mutation in log‑space
        if np.random.rand() < 0.3:
            log_child = np.log(child + 1e-12)
            sigma = 0.12
            log_child += sigma * np.random.randn(n)
            child = softmax(log_child)

        # exact evaluation
        sc = compute_score(child)
        maybe_update(child, f"recomb‑{it}")

        push_elite(child, sc)

        # every 5000 recombinations try a quick Adam refinement on current best
        if it % 5000 == 0 and time.time() < recomb_deadline - 30.0:
            a_ref, sc_ref = adam_smooth_optimize(
                np.array(best_seq, dtype=np.float64),
                deadline - 30.0,
                max_iters=80_000,
                beta_start=5.0,
                beta_end=2000.0,
                lr=0.015,
            )
            maybe_update(a_ref, "adam‑mid")
            push_elite(a_ref, sc_ref)

    # -------------------------------------------------------------
    # 5. Multi‑stage high‑β Adam refinement (sharp surrogate)
    # -------------------------------------------------------------
    if best_seq is not None and time.time() < deadline - 120.0:
        stages = [
            (5.0, 2_000.0, 0.02, 120_000),
            (2_000.0, 20_000.0, 0.012, 100_000),
            (20_000.0, 80_000.0, 0.008, 80_000),
        ]
        a_cur = np.array(best_seq, dtype=np.float64)
        for beta_s, beta_e, lr, iters in stages:
            if time.time() >= deadline - 30.0:
                break
            a_cur, sc_cur = adam_smooth_optimize(
                a_cur,
                deadline - 30.0,
                max_iters=iters,
                beta_start=beta_s,
                beta_end=beta_e,
                lr=lr,
            )
            maybe_update(a_cur, f"high‑β‑{beta_e:.0e}")

    # -------------------------------------------------------------
    # 6. Entropy‑mirror sub‑gradient descent on the exact max objective
    # -------------------------------------------------------------
    if best_seq is not None and time.time() < deadline - 70.0:
        print(f"[{time.time():.1f}] Starting mirror sub‑gradient descent")
        a_sub, sc_sub = mirror_subgrad_optimize(
            np.array(best_seq, dtype=np.float64),
            deadline - 20.0,
            step0=0.9,
        )
        maybe_update(a_sub, "mirror")
        best_seq = a_sub.tolist()
        best_score = sc_sub

    # -------------------------------------------------------------
    # 7. Symmetrisation (many optimal solutions are symmetric)
    # -------------------------------------------------------------
    if best_seq is not None:
        a_arr = np.array(best_seq, dtype=np.float64)
        a_sym = (a_arr + a_arr[::-1]) * 0.5
        a_sym = project_to_simplex(a_sym)
        maybe_update(a_sym, "symmetrised")

    # -------------------------------------------------------------
    # 8. Final tiny simulated‑annealing hill‑climb
    # -------------------------------------------------------------
    if best_seq is not None and time.time() < deadline - 5.0:
        print(f"[{time.time():.1f}] Final hill‑climb (SA)")
        a_final, sc_final = hill_climb_sa(
            np.array(best_seq, dtype=np.float64),
            deadline,
            max_steps=1_200_000,
        )
        maybe_update(a_final, "final‑SA")
        best_seq = a_final.tolist()
        best_score = sc_final

    # -------------------------------------------------------------
    # 9. Fallback in the extremely unlikely case that nothing was found
    # -------------------------------------------------------------
    if best_seq is None:
        n_small = 256
        best_seq = [1.0 / n_small] * n_small
        best_score = evaluate_sequence(best_seq)

    # -------------------------------------------------------------
    # Final official evaluation (the harness will also call evaluate_sequence)
    # -------------------------------------------------------------
    final_score = evaluate_sequence(best_seq)
    print(
        f"\n[{time.time():.1f}] finished search – final reported score = {final_score:.6f}"
    )
    return best_seq
