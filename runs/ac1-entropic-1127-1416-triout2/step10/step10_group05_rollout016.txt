    a = np.maximum(start_a, 1e-15); a /= a.sum()
    m = np.zeros_like(a); v = np.zeros_like(a)
    beta1, beta2 = 0.9, 0.999

    best_a = a.copy()

    for t in range(1, max_iters+1):
        if time.time() >= deadline: break
        conv = convolution_fft(a, a)
        k_max = int(conv.argmax())
        # build subgradient g
        g = np.zeros_like(a)
        i_start = max(0, k_max - (len(a)-1))
        i_end = min(len(a)-1, k_max)
        idx = np.arange(i_start, i_end+1)
        g[idx] = 2.0 * a[k_max - idx]
        # Adam update
        m = beta1 * m + (1-beta1) * g
        v = beta2 * v + (1-beta2) * (g**2)
        m_hat = m / (1 - beta1**t)
        v_hat = v / (1 - beta2**t)

        a = np.maximum(a - update, 1e-15)
        a /= a.sum()
        if t % 200 == 0:
            sc = compute_score(a)
            if sc < best_sc:
                best_sc, best_a = sc, a.copy()
    # final check
    sc = compute_score(a)
    if sc < best_sc:
        best_sc, best_a = sc, a.copy()

def zero_region_hill_climb(a0, deadline, max_steps=1_000_000):
    a = a0.copy()
    n = a.size
    best_a = a.copy()
    best_sc = compute_score(a)
    step_scale = 0.05
    steps = 0
    while steps < max_steps and time.time() < deadline:
        conv = convolution_fft(best_a, best_a)
        k_max = int(conv.argmax())
        i_start = max(0, k_max - (n-1))
        i_end = min(n-1, k_max)
        idx = np.arange(i_start, i_end+1)
        contrib = best_a[idx] * best_a[k_max - idx]
        i_max = idx[np.argmax(contrib)]
        # Choose j outside the active range:
        # candidate set: [0, i_start-1] U [i_end+1, n-1]
        if i_start > 0:
            # choose j randomly from 0..i_start-1
            j = np.random.randint(i_start)
        elif i_end < n-1:
            j = i_end + 1 + np.random.randint(n-1 - i_end)
        else:
            # all indices are in active range (unlikely). pick random far if possible
            j = np.random.randint(n)
            if j == i_max:
                continue
        delta = step_scale * best_a[i_max]
        if delta <= 1e-12:
            break
        cand = best_a.copy()
        cand[i_max] -= delta
        cand[j] += delta
        # ensure non-negative: after subtract might go negative due to floating errors
        cand = np.maximum(cand, 0.0)
        cand /= cand.sum()
        sc = compute_score(cand)
        if sc < best_sc:
            best_sc, best_a = sc, cand.copy()
            step_scale = min(step_scale * 1.05, 0.5)
        else:
            step_scale *= 0.95
        steps += 1
    return best_a, best_sc

    a = np.maximum(start_a, 1e-15)
    a /= a.sum()
    n = a.shape[0]
    m = np.zeros_like(a)
    v = np.zeros_like(a)

    beta2 = 0.999

"""
Algorithm for minimising `evaluate_sequence`.

1. **Problem size** – pick a length `n` that fits the remaining wall‑time.
2. **Simple family scans** – exhaustive scans of exponential, power‑law,
   a discrete Beta family and a linear ramp shape.  The best vector is kept.
   Several random Dirichlet draws with a low concentration parameter are
   also tried for diversification.
3. **Momentum sub‑gradient descent (Adam)** – optimise the true
   `max(a*a)` (the quantity appearing in the objective) using an
   Adam‑style update that works on the simplex.  A few short restarts are
   performed.
4. **Zero‑region hill‑climb** – repeatedly take a small fraction of mass
   from the index that contributes most to the current maximal convolution
   entry and move it to an index that does *not* participate in that entry
   (i.e. outside the active index range).  The step size is adapted
   automatically; the procedure stops when the time budget expires.
5. The best sequence found is returned.  All vectors are kept on the
   probability simplex, which makes the objective scale‑invariant.
"""

import numpy as np

# Basic linear‑algebra utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full (linear) convolution using a real FFT."""
    m = a.size + b.size - 1
    nfft = next_pow2(m)
    A = np.fft.rfft(a, n=nfft)
    B = np.fft.rfft(b, n=nfft)
    C = A * B
    return np.fft.irfft(C, n=nfft)[:m]


def compute_score(a: np.ndarray) -> float:
    """
    Exact objective used by `evaluate_sequence` for a vector that already
    satisfies sum = 1 (probability simplex).  Returns
        2·n·max(a*a)   where the convolution is full (length 2·n‑1).
    """
    n = a.shape[0]
    conv = convolution_fft(a, a)
    return 2.0 * n * float(conv.max())


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / v.shape[0])

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, v.shape[0] + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.where(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


# ----------------------------------------------------------------------
# Parametric families
# ----------------------------------------------------------------------


def exponential_shape(n: int, lam: float) -> np.ndarray:
    """a_i ∝ exp(-lam·i), normalised to sum = 1."""
    i = np.arange(n, dtype=float)
    w = np.exp(-lam * i) if lam > 0.0 else np.ones_like(i)
    return w / w.sum()


def power_shape(n: int, p: float) -> np.ndarray:
    """a_i ∝ (i+1)^(-p), normalised to sum = 1."""
    i = np.arange(1, n + 1, dtype=float)
    w = i ** (-p) if p > 0.0 else np.ones_like(i)
    return w / w.sum()


def beta_shape(n: int, a: float, b: float) -> np.ndarray:
    """
    Discrete Beta distribution on {0,…,n‑1}:
        a_i ∝ ((i+0.5)/n)^(a‑1)·(1‑(i+0.5)/n)^(b‑1)
    The 0.5 offset avoids division‑by‑zero for a,b<1.
    """
    x = (np.arange(n, dtype=float) + 0.5) / n
    w = np.power(x, a - 1.0) * np.power(1.0 - x, b - 1.0)
    w[np.isnan(w)] = 0.0
    w = np.maximum(w, 0.0)
    if w.sum() == 0.0:
        w = np.ones_like(w)
    return w / w.sum()


def linear_shape(n: int) -> np.ndarray:
    """Strictly decreasing linear ramp, normalised to sum = 1."""
    w = np.arange(n, 0, -1, dtype=float)
    return w / w.sum()


# ----------------------------------------------------------------------
# Simple scans
# ----------------------------------------------------------------------


def scan_exponential_and_power(n: int, deadline: float):
    """Exhaustive scan of exponential and power‑law families."""
    best_a, best_sc = None, float("inf")

    lam_grid = np.concatenate((np.array([0.0]), np.logspace(-5, -2, 200)))
    for lam in lam_grid:
        if time.time() >= deadline:
            break
        a = exponential_shape(n, lam)
        sc = compute_score(a)
        if sc < best_sc:
            best_sc, best_a = sc, a.copy()

    p_grid = np.linspace(0.0, 6.0, 200)
    for p in p_grid:
        if time.time() >= deadline:
            break
        a = power_shape(n, p)
        sc = compute_score(a)
        if sc < best_sc:
            best_sc, best_a = sc, a.copy()

    return best_a, best_sc


def scan_beta_family(n: int, deadline: float):
    """Modest scan of a two‑parameter Beta family."""
    best_a, best_sc = None, float("inf")
    a_grid = np.concatenate((
        np.logspace(-0.7, -0.1, 12),
        np.logspace(-0.1, 0.3, 12),
        np.linspace(0.5, 3.0, 12),
    ))
    b_grid = a_grid.copy()
    for a in a_grid:
        for b in b_grid:
            if time.time() >= deadline:
                return best_a, best_sc
            a_vec = beta_shape(n, a, b)
            sc = compute_score(a_vec)
            if sc < best_sc:
                best_sc, best_a = sc, a_vec.copy()
    return best_a, best_sc


# ----------------------------------------------------------------------
# Adam – momentum sub‑gradient descent on the exact max convolution
# ----------------------------------------------------------------------


def adam_max_descent(start_a: np.ndarray,
                     deadline: float,
                     max_iters: int = 250_000,
                     eta: float = 0.6) -> tuple[np.ndarray, float]:
    """
    Adam‑style optimiser for  f(a)=max(a*a)  (the inner max‑convolution).
    The vector is kept on the simplex by normalising after each step.
    """
    a = np.maximum(start_a.astype(float), 1e-15)
    a /= a.sum()
    n = a.shape[0]

    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8

    best_a = a.copy()
    best_sc = compute_score(a)

    for it in range(1, max_iters + 1):
        if time.time() >= deadline:
            break

        conv = convolution_fft(a, a)
        k_max = int(conv.argmax())

        # sub‑gradient of the active convolution entry
        grad = np.zeros_like(a)
        i_start = max(0, k_max - (n - 1))
        i_end = min(n - 1, k_max)
        idx = np.arange(i_start, i_end + 1)
        grad[idx] = 2.0 * a[k_max - idx]

        # Adam moments
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)

        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        step = eta * m_hat / (np.sqrt(v_hat) + eps)

        a = a - step
        a = np.maximum(a, 1e-15)
        a /= a.sum()

        if it % 500 == 0:
            sc = compute_score(a)
            if sc < best_sc:
                best_sc, best_a = sc, a.copy()

    # final check
    sc = compute_score(a)
    if sc < best_sc:
        best_sc, best_a = sc, a.copy()
    return best_a, best_sc


# ----------------------------------------------------------------------
# Aggressive hill‑climb that moves mass into the “zero‑region”
# ----------------------------------------------------------------------


def zero_region_hill_climb(start_seq: np.ndarray,
                           deadline: float,
                           max_steps: int = 2_000_000) -> tuple[np.ndarray, float]:
    """
    Greedy descent that always extracts a fraction of mass from the index
    contributing most to the current maximal convolution entry and
    relocates it to an index that does **not** belong to that entry
    (i.e. outside the active range).  The move is kept only if it
    strictly improves the exact objective.
    """
    a = start_seq.copy()
    n = a.size
    best_a = a.copy()
    best_sc = compute_score(best_a)

    step_scale = 0.20  # start by moving up to 20 % of the selected mass
    steps = 0

    while steps < max_steps and time.time() < deadline:
        conv = convolution_fft(best_a, best_a)
        k_max = int(conv.argmax())

        i_start = max(0, k_max - (n - 1))
        i_end = min(n - 1, k_max)

        # contributions to the active entry
        idx = np.arange(i_start, i_end + 1)
        contrib = best_a[idx] * best_a[k_max - idx]
        i_max = idx[np.argmax(contrib)]

        # choose a destination outside the active range (zero contribution)
        if i_start > 0:
            # any index left of the active block
            j = np.random.randint(i_start)
        elif i_end < n - 1:
            # any index right of the active block
            j = i_end + 1 + np.random.randint(n - 1 - i_end)
        else:
            # extremely rare: the active block covers all indices
            j = np.random.randint(n)
            if j == i_max:
                continue

        delta = step_scale * best_a[i_max]
        if delta <= 1e-15:
            break

        cand = best_a.copy()
        cand[i_max] -= delta
        cand[j] += delta

        # renormalise to stay on the simplex and guard against negativity
        cand = np.maximum(cand, 0.0)
        cand /= cand.sum()

        sc = compute_score(cand)
        if sc < best_sc:
            best_sc, best_a = sc, cand.copy()
            step_scale = min(step_scale * 1.05, 0.5)
        else:
            step_scale *= 0.94

        steps += 1

    return best_a, best_sc

def propose_candidate(seed: int = 12345, budget_s: int = 1000, **kwargs):
    """Search for a step‑function that minimises `evaluate_sequence`."""
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 2.0   # safety margin

    # -------------------------------------------------
    # Choose a length based on the remaining time
    # -------------------------------------------------
    remaining = deadline - start_time
    if remaining > 800:
        n = 32768
    elif remaining > 500:
        n = 16384
    elif remaining > 250:
        n = 8192
    else:
        n = 4096
    print(f"[{time.time():.1f}] problem length = {n}")

    best_seq = None
    best_score = float("inf")

    # -------------------------------------------------
    # Optional fallback from a previously discovered vector
    # -------------------------------------------------
    try:
        fallback = [float(x) for x in height_sequence_1]   # type: ignore
        fb = evaluate_sequence(fallback)
        print(f"[{time.time():.1f}] fallback score = {fb:.6f}")
        if fb < best_score:
            best_score, best_seq = fb, fallback
    except Exception:
        pass

    # -------------------------------------------------
    # 1. Scan simple families
    # -------------------------------------------------
    a_exp, sc_exp = scan_exponential_and_power(n, deadline - 200.0)
    if a_exp is not None and sc_exp < best_score:
        best_score, best_seq = sc_exp, a_exp.tolist()
        print(f"[{time.time():.1f}] after expo/power scan = {best_score:.6f}")

    a_beta, sc_beta = scan_beta_family(n, deadline - 180.0)
    if a_beta is not None and sc_beta < best_score:
        best_score, best_seq = sc_beta, a_beta.tolist()
        print(f"[{time.time():.1f}] after beta scan = {best_score:.6f}")

    a_lin = linear_shape(n)
    sc_lin = compute_score(a_lin)
    if sc_lin < best_score:
        best_score, best_seq = sc_lin, a_lin.tolist()
        print(f"[{time.time():.1f}] linear shape = {best_score:.6f}")

    # -------------------------------------------------
    # 2. Random Dirichlet starts (low concentration)
    # -------------------------------------------------
    for _ in range(6):
        if time.time() >= deadline - 150.0:
            break
        conc = 10 ** np.random.uniform(-2, 0)   # between 0.01 and 1
        draw = np.random.gamma(conc, 1.0, size=n)
        draw /= draw.sum()
        sc = compute_score(draw)
        if sc < best_score:
            best_score, best_seq = sc, draw.tolist()
            print(f"[{time.time():.1f}] random Dirichlet improved to {best_score:.6f}")

    # -------------------------------------------------
    # 3. If nothing succeeded, start from uniform
    # -------------------------------------------------
    if best_seq is None:
        a_uni = np.full(n, 1.0 / n)
        best_score = compute_score(a_uni)
        best_seq = a_uni.tolist()
        print(f"[{time.time():.1f}] uniform fallback score = {best_score:.6f}")

    # -------------------------------------------------
    # 4. Adam‑style momentum sub‑gradient descent (several restarts)
    # -------------------------------------------------
    if time.time() < deadline - 100.0:
        restarts = 3
        for r in range(restarts):
            if time.time() >= deadline - 100.0:
                break
            init = np.array(best_seq, dtype=float)
            # tiny perturbation before each restart
            init += 1e-4 * np.random.rand(n)
            init = np.maximum(init, 0.0)
            init /= init.sum()
            a_opt, sc_opt = adam_max_descent(
                init,
                deadline - 30.0,
                max_iters=250_000,
                eta=0.6,
            )
            if sc_opt < best_score:
                best_score, best_seq = sc_opt, a_opt.tolist()
                print(f"[{time.time():.1f}] Adam restart {r} improved to {best_score:.6f}")

    # -------------------------------------------------
    # 5. Aggressive zero‑region hill‑climb
    # -------------------------------------------------
    if time.time() < deadline - 5.0:
        print(f"[{time.time():.1f}] starting zero‑region hill‑climb")
        a_hc, sc_hc = zero_region_hill_climb(
            np.array(best_seq, dtype=float),
            deadline,
            max_steps=2_000_000,
        )
        if sc_hc < best_score:
            best_score, best_seq = sc_hc, a_hc.tolist()
            print(f"[{time.time():.1f}] hill‑climb best = {best_score:.6f}")

    # -------------------------------------------------
    # Final official evaluation (the harness will also call it)
    # -------------------------------------------------
    final_score = evaluate_sequence(best_seq)
    print(f"\n[{time.time():.1f}] finished – final reported score = {final_score:.6f}")

    # return a plain Python list of floats
    return best_seq
