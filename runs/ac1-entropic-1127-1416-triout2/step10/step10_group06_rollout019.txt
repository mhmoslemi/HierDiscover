"""
Step‑function optimizer for `evaluate_sequence`.

The search consists of three phases:

1. **Broad analytic sweep** – several one‑parameter families (product,
   left‑heavy power, exponential, additive power, additive power‑γ,
   two‑Gaussian tails) are scanned on a moderately large length
   (≈8 k).  The best vector (always projected onto the probability simplex)
   is kept.

2. **Fine surrogate optimisation** – the best candidate is refined by
   L‑BFGS‑B on a smooth log‑sum‑exp surrogate of the max‑convolution.
   The temperature schedule uses five β‑values up to 2·10⁴ and a generous
   iteration budget, driving the surrogate very close to the true max.

3. **Exact mirror sub‑gradient & stochastic polishing** – an exponentiated
   mirror‑descent on the exact objective (2·n·max(a∗a)) reduces the
   remaining gap, followed by a tiny simulated‑annealing hill‑climb and
   a final symmetrisation step.

All phases respect the global 1000 s wall clock and print progress.
The returned object is a plain ``list[float]`` ready for the outer harness.
"""

import math

# FFT utilities

def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Aperiodic linear convolution via real FFT."""
    m = a.size + b.size - 1
    nfft = next_pow2(m)
    A = np.fft.rfft(a, n=nfft)
    B = np.fft.rfft(b, n=nfft)
    C = A * B
    conv = np.fft.irfft(C, n=nfft)[:m]
    return conv


def compute_score(a: np.ndarray) -> float:
    """Exact objective for a probability vector (∑a=1)."""
    n = a.shape[0]
    conv = convolution_fft(a, a)
    return float(2.0 * n * conv.max())


# ----------------------------------------------------------------------
# Simplex utilities
# ----------------------------------------------------------------------
def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection onto the probability simplex."""
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / v.shape[0])

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, v.shape[0] + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.where(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def softmax(z: np.ndarray) -> np.ndarray:
    """Stable soft‑max (maps ℝⁿ → simplex)."""
    z_max = np.max(z)
    e = np.exp(z - z_max)
    return e / e.sum()

# Analytic families (all return a probability vector)
# ----------------------------------------------------------------------
def symmetric_product_family(n: int, alpha: float) -> np.ndarray:
    """U‑shaped: a_i ∝ [(i+½)(n‑½‑i)]^(‑α)."""
    i = np.arange(n, dtype=np.float64)
    logprod = np.log(i + 0.5) + np.log(n - 0.5 - i)
    w = np.exp(-alpha * logprod)
    w[np.isinf(w)] = np.finfo(np.float64).max
    a = w / w.sum()
    return a


def left_heavy_power_family(n: int, beta: float) -> np.ndarray:
    """Monotone decreasing power law: a_i ∝ (i+½)^(‑β)."""
    i = np.arange(n, dtype=np.float64)
    w = np.exp(-beta * np.log(i + 0.5))
    a = w / w.sum()
    return a


def exponential_decay_family(n: int, gamma: float) -> np.ndarray:
    """Simple exponential decay: a_i ∝ exp(‑γ i)."""
    i = np.arange(n, dtype=np.float64)
    w = np.exp(-gamma * i)
    a = w / w.sum()
    return a


def additive_power_family(n: int, alpha: float) -> np.ndarray:
    """Additive tails: a_i ∝ (i+½)^(‑α) + (n‑½‑i)^(‑α)."""
    i = np.arange(n, dtype=np.float64)
    w = (i + 0.5) ** (-alpha) + (n - 0.5 - i) ** (-alpha)
    a = w / w.sum()
    return a


def additive_power_gamma_family(n: int, alpha: float, gamma: float) -> np.ndarray:
    """Additive tails with extra exponent γ: a_i ∝ [(i+½)^(‑α)+(…)]^γ."""
    i = np.arange(n, dtype=np.float64)
    w = ((i + 0.5) ** (-alpha) + (n - 0.5 - i) ** (-alpha)) ** gamma
    a = w / w.sum()
    return a


def two_gaussian_tail_family(n: int, sigma: float) -> np.ndarray:
    """Two symmetric Gaussian tails centred at the ends."""
    i = np.arange(n, dtype=np.float64)
    w = np.exp(-(i ** 2) / (2.0 * sigma ** 2)) + np.exp(-(((n - 1 - i) ** 2) / (2.0 * sigma ** 2)))
    a = w / w.sum()
    return a

def random_sparse_start(n: int) -> np.ndarray:
    """Sparse probability vector (~5 % non‑zeros)."""
    sparsity = np.random.uniform(0.03, 0.07)
    k = max(2, int(round(sparsity * n)))
    idx = np.random.choice(n, size=k, replace=False)
    a = np.zeros(n, dtype=np.float64)
    a[idx] = 1.0 / k
    return project_to_simplex(a)


def random_monotone_start(n: int) -> np.ndarray:
    """Random decreasing vector (sorted power‑transformed uniform)."""
    r = np.random.rand(n) ** np.random.uniform(0.5, 5.0)
    r.sort()
    a = r[::-1]                     # decreasing
    a = a / a.sum()
    return a

# Smooth surrogate optimisation (L‑BFGS‑B)
# ----------------------------------------------------------------------
def _smooth_obj_and_grad(z: np.ndarray, beta: float, n: int):
    """
    Returns (f, grad) for the log‑sum‑exp surrogate
        f = (1/β)·log∑_k exp[β·(a*a)_k],
    where a = softmax(z).  The gradient is w.r.t. the log‑soft‑max
    parameters z.
    """
    a = softmax(z)                       # a on simplex
    conv = convolution_fft(a, a)         # length 2n‑1
    maxc = conv.max()
    # stable log‑sum‑exp
    exp_terms = np.exp(beta * (conv - maxc))
    sum_exp = exp_terms.sum()
    f = (np.log(sum_exp) + beta * maxc) / beta

    # ∂f/∂a   (vector of same length as a)
    w = exp_terms / sum_exp              # weighting of each convolution entry
    a_rev = a[::-1]
    # convolution of a_rev with w gives gradient components for a
    conv_rev = convolution_fft(a_rev, w)  # length 3n‑2
    grad_a = 2.0 * conv_rev[n - 1:2 * n - 1]

    # chain rule through soft‑max
    dot_ag = np.dot(a, grad_a)
    grad_z = a * (grad_a - dot_ag)
    return float(f), grad_z.astype(np.float64)


def l_bfgs_refine(start_a: np.ndarray, deadline: float,
                  beta_list=(200.0, 800.0, 2000.0, 8000.0, 20000.0),
                  max_iter_per_beta=1500) -> tuple[np.ndarray, float]:
    """
    Refine `start_a` using L‑BFGS‑B on the smoothed surrogate.
    Returns the best vector and its exact score.
    """
    if not _HAS_SCIPY:                     # pragma: no cover
        return start_a, compute_score(start_a)

    best_a = start_a.copy()
    best_score = compute_score(best_a)
    eps = 1e-12
    z = np.log(best_a + eps)               # initialise log‑soft‑max

    for beta in beta_list:
        if time.time() >= deadline - 3.0:
            break
        remaining = deadline - time.time()
        max_iter = min(max_iter_per_beta, int(remaining * 5))   # crude budget

        def func(z_vec):
            return _smooth_obj_and_grad(z_vec, beta, best_a.shape[0])

        try:
            opt_res = _opt.fmin_l_bfgs_b(
                func,
                z,
                maxiter=max_iter,
                disp=False,
                factr=1e7,
                pgtol=1e-12,
            )
        except Exception as exc:   # pragma: no cover
            print(f"[{time.time():.1f}] L‑BFGS failed (β={beta}): {exc}")
            continue

        z = opt_res[0]
        a_opt = softmax(z)
        sc_opt = compute_score(a_opt)
        if sc_opt < best_score:
            best_score = sc_opt
            best_a = a_opt.copy()
            print(f"[{time.time():.1f}] L‑BFGS β={beta:.0f} improved → {best_score:.6f}")

    return best_a, best_score


# ----------------------------------------------------------------------
# Mirror (entropy) sub‑gradient descent on the true max‑objective
# ----------------------------------------------------------------------
def mirror_subgrad_optimize(start_a: np.ndarray, deadline: float,
                           step0: float = 1.0,
                           check_interval: int = 2000) -> tuple[np.ndarray, float]:
    """
    Exponentiated sub‑gradient descent on F(a)=2·n·max(a*a)
    with projection onto the simplex after each multiplicative update.
    """
    n = start_a.shape[0]
    a = start_a.astype(np.float64).copy()
    best_a = a.copy()
    best_score = compute_score(a)

    t = 0
    while time.time() < deadline:
        t += 1
        conv = convolution_fft(a, a)
        k = int(np.argmax(conv))

        # sub‑gradient for the active convolution entry
        idx = np.arange(n)
        j = k - idx
        mask = (j >= 0) & (j < n)
        g = np.zeros_like(a)
        g[mask] = 2.0 * a[j[mask]]

        eta = step0 / math.sqrt(t)
        a = a * np.exp(-eta * g)          # multiplicative update
        s = a.sum()
        a = a / s if s > 0 else np.full_like(a, 1.0 / n)

        if t % check_interval == 0:
            sc = compute_score(a)
            if sc < best_score:
                best_score = sc
                best_a = a.copy()
                print(f"[{time.time():.1f}] mirror sub‑grad iter {t} best = {best_score:.6f}")

    # final check
    sc = compute_score(a)
    if sc < best_score:
        best_score = sc
        best_a = a.copy()
    return best_a, best_score

# Tiny simulated‑annealing hill‑climb
# ----------------------------------------------------------------------
def hill_climb_sa(start_a: np.ndarray, deadline: float,
                  max_steps: int = 800_000,
                  init_temp_factor: float = 3e-4) -> tuple[np.ndarray, float]:
    """
    Random walk on the simplex with a very low temperature.
    """
    n = start_a.size
    cur_a = start_a.copy()
    cur_score = compute_score(cur_a)

    best_a = cur_a.copy()
    best_score = cur_score

    T0 = best_score * init_temp_factor
    T = T0
    step_scale = 0.12

    for step in range(1, max_steps + 1):
        if time.time() >= deadline:
            break

        cand = cur_a.copy()
        if np.random.rand() < 0.5:          # swap two entries
            i, j = np.random.randint(n, size=2)
            cand[i], cand[j] = cand[j], cand[i]
        else:                               # transfer a fraction of mass
            i, j = np.random.randint(n, size=2)
            if i != j:
                delta = step_scale * cur_a[i]
                cand[i] -= delta
                cand[j] += delta
                if cand[i] < 0.0:
                    cand[i] = 0.0

        cand = cand / cand.sum()
        cand_score = compute_score(cand)
        delta = cand_score - cur_score

        if delta < 0.0 or np.random.rand() < np.exp(-delta / (T + 1e-18)):
            cur_a = cand
            cur_score = cand_score
            if cur_score < best_score:
                best_score = cur_score
                best_a = cur_a.copy()
                step_scale = min(step_scale * 1.07, 0.5)
        else:
            step_scale *= 0.97

        T = T0 * (1.0 - step / max_steps)

    return best_a, best_score

    # 0. Fallback to any previously‑known construction
    # --------------------------------------------------------------
    try:
        fb_vec = [float(x) for x in height_sequence_1]   # type: ignore
        fb = evaluate_sequence(fb_vec)
        print(f"[{time.time():.1f}] fallback score = {fb:.6f}")
        if fb < best_score:
            best_score = fb
            best_seq = fb_vec
    except Exception:
        pass

    # --------------------------------------------------------------
    # 1. Scan analytic families – dense grids for the most promising ones
    # --------------------------------------------------------------
    # symmetric product (U‑shaped)
    for alpha in np.linspace(0.0, 6.0, 121):
        if time.time() >= deadline - 100.0:
            break
        vec = symmetric_product_family(n, alpha)
        sc = compute_score(vec)
        if sc < best_score:
            best_score = sc
            best_seq = vec.tolist()
            print(f"[{time.time():.1f}] new best (sym‑α={alpha:.3f}) = {best_score:.6f}")

    # left‑heavy power‑law
    for beta in np.linspace(0.0, 6.0, 121):
        if time.time() >= deadline - 100.0:
            break
        vec = left_heavy_power_family(n, beta)
        sc = compute_score(vec)
        if sc < best_score:
            best_score = sc
            best_seq = vec.tolist()
            print(f"[{time.time():.1f}] new best (left‑β={beta:.3f}) = {best_score:.6f}")

    # exponential decay
    for gamma in np.linspace(0.0, 5.0, 101):
        if time.time() >= deadline - 100.0:
            break
        vec = exponential_decay_family(n, gamma)
        sc = compute_score(vec)
        if sc < best_score:
            best_score = sc
            best_seq = vec.tolist()
            print(f"[{time.time():.1f}] new best (exp‑γ={gamma:.3f}) = {best_score:.6f}")

    # additive power tails
    for alpha in np.linspace(0.1, 5.0, 81):
        if time.time() >= deadline - 100.0:
            break
        vec = additive_power_family(n, alpha)
        sc = compute_score(vec)
        if sc < best_score:
            best_score = sc
            best_seq = vec.tolist()
            print(f"[{time.time():.1f}] new best (add‑α={alpha:.3f}) = {best_score:.6f}")

    # additive power with extra exponent γ
    for alpha in [0.5, 1.0, 1.5, 2.0]:
        for gamma in [0.5, 1.0, 1.5, 2.0, 3.0]:
            if time.time() >= deadline - 100.0:
                break
            vec = additive_power_gamma_family(n, alpha, gamma)
            sc = compute_score(vec)
            if sc < best_score:
                best_score = sc
                best_seq = vec.tolist()
                print(f"[{time.time():.1f}] new best (add‑α={alpha:.1f},γ={gamma:.1f}) = {best_score:.6f}")

    # two symmetric Gaussian tails
    for sigma in np.linspace(5.0, 300.0, 60):
        if time.time() >= deadline - 100.0:
            break
        vec = two_gaussian_tail_family(n, sigma)
        sc = compute_score(vec)
        if sc < best_score:
            best_score = sc
            best_seq = vec.tolist()
            print(f"[{time.time():.1f}] new best (σ={sigma:.1f}) = {best_score:.6f}")

    # random monotone starts
    for _ in range(120):
        if time.time() >= deadline - 100.0:
            break
        vec = random_monotone_start(n)
        sc = compute_score(vec)
        if sc < best_score:
            best_score = sc
            best_seq = vec.tolist()
            print(f"[{time.time():.1f}] new best (rand‑mono) = {best_score:.6f}")

    # random sparse starts
    for _ in range(80):
        if time.time() >= deadline - 100.0:
            break
        vec = random_sparse_start(n)
        sc = compute_score(vec)
        if sc < best_score:
            best_score = sc
            best_seq = vec.tolist()
            print(f"[{time.time():.1f}] new best (rand‑sparse) = {best_score:.6f}")

    # --------------------------------------------------------------
    # 2. Refine the current best with an aggressive L‑BFGS schedule
    # --------------------------------------------------------------
    if best_seq is not None and time.time() < deadline - 300.0:
        print(f"[{time.time():.1f}] Starting aggressive L‑BFGS refinement")
        a_start = np.array(best_seq, dtype=np.float64)
        a_refined, sc_refined = l_bfgs_refine(
            a_start,
            deadline - 200.0,
            beta_list=(200.0, 800.0, 2000.0, 8000.0, 20000.0),
            max_iter_per_beta=1500,
        )
        if sc_refined < best_score:
            best_score = sc_refined
            best_seq = a_refined.tolist()
            print(f"[{time.time():.1f}] after L‑BFGS best = {best_score:.6f}")

    # --------------------------------------------------------------
    # 3. Mirror sub‑gradient descent on the exact max‑objective
    # --------------------------------------------------------------
    if best_seq is not None and time.time() < deadline - 120.0:
        print(f"[{time.time():.1f}] Starting mirror sub‑gradient descent")
        a_sub, sc_sub = mirror_subgrad_optimize(
            np.array(best_seq, dtype=np.float64),
            deadline - 60.0,
            step0=1.0,
        )
        if sc_sub < best_score:
            best_score = sc_sub
            best_seq = a_sub.tolist()
            print(f"[{time.time():.1f}] after mirror sub‑grad best = {best_score:.6f}")

    # --------------------------------------------------------------
    # 4. Symmetrisation – many optimal solutions are symmetric
    # --------------------------------------------------------------
    if best_seq is not None:
        a_arr = np.array(best_seq, dtype=np.float64)
        a_sym = (a_arr + a_arr[::-1]) * 0.5
        a_sym = project_to_simplex(a_sym)
        sc_sym = compute_score(a_sym)
        if sc_sym < best_score:
            best_score = sc_sym
            best_seq = a_sym.tolist()
            print(f"[{time.time():.1f}] symmetrisation improved best = {best_score:.6f}")

    # --------------------------------------------------------------
    # 5. Final tiny simulated‑annealing hill‑climb
    # --------------------------------------------------------------
    if best_seq is not None and time.time() < deadline - 5.0:
        print(f"[{time.time():.1f}] Final hill‑climb (SA)")
        a_final, sc_final = hill_climb_sa(
            np.array(best_seq, dtype=np.float64),
            deadline,
            max_steps=800_000,
        )
        if sc_final < best_score:
            best_score = sc_final
            best_seq = a_final.tolist()
            print(f"[{time.time():.1f}] hill‑climb improved best = {best_score:.6f}")

    # --------------------------------------------------------------
    # Safeguard – fall back to a modest uniform vector if everything failed
    # --------------------------------------------------------------
    if best_seq is None:
        n_small = 256
        best_seq = [1.0 / n_small] * n_small
        best_score = evaluate_sequence(best_seq)

    # Final official evaluation (the harness will also call evaluate_sequence)
    final_score = evaluate_sequence(best_seq)
    print(f"\n[{time.time():.1f}] finished search – final reported score = {final_score:.6f}")
    return best_seq
