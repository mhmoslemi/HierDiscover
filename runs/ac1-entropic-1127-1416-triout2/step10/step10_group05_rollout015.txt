"""
Improved step‑function optimizer
--------------------------------

The algorithm searches for a non‑negative vector `a` (sum ≈ 1) that minimizes

    E(a) = 2·n·max(a * a)                     (exact objective)
    # evaluate_sequence returns the same value after the required scaling

The search proceeds in the following stages, respecting the 1000 s wall‑clock
budget:

1. **Exploratory grid search** over a rich collection of 1‑parameter families
   (exponential, power‑law, symmetric exponential, linear, √‑shape,
   inverse‑√‑shape, a two‑parameter product‑power shape and a Beta family) as
   well as a handful of random Dirichlet draws.  The best candidate is kept.

2. **Smooth‑max optimisation** – minimise a log‑sum‑exp surrogate of the
   convolution maximum by exponentiated‑gradient (KL‑mirror) updates.  The
   smoothness parameter `β` is raised from 0.5 to a very large value
   (`β_end≈5·10⁴`), turning the surrogate gradually into the true max.

3. **Exact sub‑gradient descent** on the true max‑convolution.  This phase
   runs many more mirror‑updates than before (up to the remaining time).

4. **Targeted hill‑climb (deterministic move)** – each step removes a small
   packet of mass from the index that contributes most to the current maximal
   convolution entry and deposits it at the index that yields the smallest
   increase at that entry.  The move is accepted only if it improves the exact
   objective; otherwise the step size is shrunk.  This deterministic rule
   is more effective than the purely random walk used earlier.

All vectors are kept on the probability simplex (non‑negative, sum = 1),
which makes the objective invariant under scaling and matches the behaviour
of the supplied `evaluate_sequence`.

The routine prints progress messages and finally returns a plain Python list
of floats – the format required by the evaluation harness.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Basic linear‑algebra utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Aperiodic full convolution via real FFT (zero‑padded)."""
    m = a.size + b.size - 1
    nfft = next_pow2(m)
    A = np.fft.rfft(a, n=nfft)
    B = np.fft.rfft(b, n=nfft)
    C = A * B
    return np.fft.irfft(C, n=nfft)[:m]


def compute_score(a: np.ndarray) -> float:
    """
    Exact objective used by `evaluate_sequence` for a simplex‑normalised vector.
    Returns 2·n·max(a*a) where the convolution is full (length 2·n‑1).
    """
    n = a.shape[0]
    conv = convolution_fft(a, a)
    return 2.0 * n * float(conv.max())


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / v.shape[0])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, v.shape[0] + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.where(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

# ----------------------------------------------------------------------


def exponential_shape(n: int, lam: float) -> np.ndarray:
    """a_i ∝ exp(-lam·i) (lam≥0)."""
    i = np.arange(n, dtype=float)
    w = np.exp(-lam * i) if lam > 0.0 else np.ones_like(i)
    return w / w.sum()


def power_shape(n: int, p: float) -> np.ndarray:
    """a_i ∝ (i+1)^(-p) (p≥0)."""
    i = np.arange(1, n + 1, dtype=float)
    w = i ** (-p) if p > 0.0 else np.ones_like(i)
    return w / w.sum()


def beta_shape(n: int, a: float, b: float) -> np.ndarray:
    """Discrete Beta on {0,…,n‑1}."""
    x = (np.arange(n, dtype=float) + 0.5) / n
    w = np.power(x, a - 1.0) * np.power(1.0 - x, b - 1.0)
    w[np.isnan(w)] = 0.0
    w = np.maximum(w, 0.0)
    if w.sum() == 0.0:
        w = np.ones_like(w)
    return w / w.sum()


def symmetric_exponential_shape(n: int, lam: float) -> np.ndarray:
    """a_i ∝ exp(-lam·|i-(n‑1)/2|)."""
    i = np.arange(n, dtype=float) - (n - 1) / 2.0
    w = np.exp(-lam * np.abs(i)) if lam > 0.0 else np.ones_like(i)
    return w / w.sum()


def linear_decrease_shape(n: int) -> np.ndarray:
    """a_i ∝ (n‑i)."""
    w = np.arange(n, 0, -1, dtype=float)
    return w / w.sum()


def sqrt_shape(n: int) -> np.ndarray:
    """a_i ∝ sqrt(i+1)."""
    w = np.sqrt(np.arange(1, n + 1, dtype=float))
    return w / w.sum()


def inv_sqrt_shape(n: int) -> np.ndarray:
    """a_i ∝ 1/√(i+1)."""
    w = 1.0 / np.sqrt(np.arange(1, n + 1, dtype=float))
    return w / w.sum()


def power_product_shape(n: int, p: float, q: float) -> np.ndarray:
    """
    a_i ∝ (i+1)^p * (n-i)^q
    (both exponents can be negative or positive).
    """
    i = np.arange(1, n + 1, dtype=float)
    left = i ** p
    right = (n + 1 - i) ** q
    w = left * right
    w[np.isnan(w)] = 0.0
    w = np.maximum(w, 0.0)
    if w.sum() == 0.0:
        w = np.ones_like(w)
    return w / w.sum()

def scan_families(n: int, deadline: float) -> tuple[np.ndarray, float]:
    """Exhaustively scan a variety of simple families (time‑limited)."""
    best_a, best_sc = None, float("inf")

    def maybe_update(a_vec: np.ndarray):
        nonlocal best_a, best_sc
        sc = compute_score(a_vec)
        if sc < best_sc:
            best_sc, best_a = sc, a_vec.copy()

    # 1‑parameter families -------------------------------------------------
    lam_grid = np.concatenate((
        np.logspace(-5, -1, 120),                     # very gentle decays
        np.linspace(0.0, 2.0, 120)                    # up to fairly steep decays
    ))
    for lam in lam_grid:
        if time.time() >= deadline:
            return best_a, best_sc
        maybe_update(exponential_shape(n, lam))

    p_grid = np.linspace(0.0, 12.0, 150)
    for p in p_grid:
        if time.time() >= deadline:
            return best_a, best_sc
        maybe_update(power_shape(n, p))

    # symmetric exponential -------------------------------------------------
    lam_grid_sym = np.concatenate((np.logspace(-5, -1, 100), np.linspace(0.0, 2.0, 100)))
    for lam in lam_grid_sym:
        if time.time() >= deadline:
            return best_a, best_sc
        maybe_update(symmetric_exponential_shape(n, lam))

    # deterministic simple shapes -----------------------------------------
    maybe_update(linear_decrease_shape(n))
    maybe_update(sqrt_shape(n))
    maybe_update(inv_sqrt_shape(n))

    # 2‑parameter product‑power shape -------------------------------------
    p_vals = np.concatenate((np.linspace(-2.0, 2.0, 40), np.logspace(-2, 2, 40)))
    q_vals = p_vals.copy()
    for p in p_vals:
        for q in q_vals:
            if time.time() >= deadline:
                return best_a, best_sc
            maybe_update(power_product_shape(n, p, q))

    # Beta family -----------------------------------------------------------
    a_grid = np.concatenate((
        np.logspace(-0.7, -0.1, 12),
        np.logspace(-0.1, 0.3, 12),
        np.linspace(0.5, 3.0, 12)
    ))
    b_grid = a_grid.copy()
    for a in a_grid:
        for b in b_grid:
            if time.time() >= deadline:
                return best_a, best_sc
            maybe_update(beta_shape(n, a, b))

    return best_a, best_sc


def random_dirichlet_samples(n: int, deadline: float, count: int = 10) -> tuple[np.ndarray, float]:
    """Generate a few Dirichlet‑distributed vectors with varying concentration."""
    best_a, best_sc = None, float("inf")
    for _ in range(count):
        if time.time() >= deadline:
            break
        # Random concentration in [0.1, 5.0]
        alpha = np.random.uniform(0.1, 5.0)
        x = np.random.gamma(alpha, 1.0, size=n)
        a = x / x.sum()
        sc = compute_score(a)
        if sc < best_sc:
            best_sc, best_a = sc, a.copy()
    return best_a, best_sc

def exp_smooth_optimize(start_a: np.ndarray,
                        deadline: float,
                        max_iters: int = 300_000,
                        beta_start: float = 0.5,
                        beta_end: float = 5e4,
                        eta0: float = 0.6) -> tuple[np.ndarray, float]:
    """
    Minimise a log‑sum‑exp surrogate of the max‑convolution by KL‑mirror steps.
    The smoothness parameter β grows exponentially from `beta_start` to `beta_end`.
    """
    a = np.maximum(start_a.astype(float), 1e-15)
    a /= a.sum()
    n = a.shape[0]

    best_a = a.copy()
    best_sc = compute_score(a)

    for it in range(1, max_iters + 1):
        if time.time() >= deadline:
            break

        # schedule for β (smoothness)
        t = it / max_iters
        beta = beta_start * (beta_end / beta_start) ** t

        conv = convolution_fft(a, a)
        maxc = conv.max()
        # stable soft‑max weights
        expw = np.exp(beta * (conv - maxc))
        w = expw / expw.sum()

        # Gradient of the smooth max (KL‑mirror)
        a_rev = a[::-1]
        grad = 2.0 * convolution_fft(a_rev, w)[n - 1:n - 1 + n]

        eta = eta0 / np.sqrt(it)

        a = np.maximum(a * np.exp(-eta * grad), 1e-15)
        a /= a.sum()

        if it % 500 == 0:
            sc = compute_score(a)
            if sc < best_sc:
                best_sc, best_a = sc, a.copy()
    # final exact check
    sc = compute_score(a)
    if sc < best_sc:
        best_sc, best_a = sc, a.copy()
    return best_a, best_sc


def max_descent_optimize(start_a: np.ndarray,
                         deadline: float,
                         max_iters: int = 600_000,
                         eta0: float = 0.8) -> tuple[np.ndarray, float]:
    """
    Direct KL‑mirror sub‑gradient descent on the true max‑convolution.
    """
    a = np.maximum(start_a.astype(float), 1e-15)
    a /= a.sum()
    n = a.shape[0]

    best_a = a.copy()
    best_sc = compute_score(a)

    for it in range(1, max_iters + 1):
        if time.time() >= deadline:
            break

        conv = convolution_fft(a, a)
        k_max = int(conv.argmax())

        # sub‑gradient for the active entry
        g = np.zeros_like(a)
        i_start = max(0, k_max - (n - 1))
        i_end = min(n - 1, k_max)
        idx = np.arange(i_start, i_end + 1)
        g[idx] = 2.0 * a[k_max - idx]

        eta = eta0 / np.sqrt(it)

        a = np.maximum(a * np.exp(-eta * g), 1e-15)
        a /= a.sum()

        if it % 500 == 0:
            sc = compute_score(a)
            if sc < best_sc:
                best_sc, best_a = sc, a.copy()
    # final exact check
    sc = compute_score(a)
    if sc < best_sc:
        best_sc, best_a = sc, a.copy()
    return best_a, best_sc


def deterministic_hill_climb(start_seq: np.ndarray,
                             deadline: float,
                             max_steps: int = 300_000,
                             init_step_scale: float = 0.10) -> tuple[np.ndarray, float]:
    """
    Greedy walk: each step removes a fraction of mass from the index that
    contributes most to the current maximal convolution entry and deposits it
    at the index that causes the smallest increase at that entry.
    Only improving moves are kept; the step scale is tuned adaptively.
    """
    a = start_seq.copy()
    n = a.size
    best_a = a.copy()
    best_sc = compute_score(best_a)

    step_scale = init_step_scale
    steps = 0

    while steps < max_steps and time.time() < deadline:
        conv = convolution_fft(best_a, best_a)
        k_max = int(conv.argmax())

        # indices that actually appear in the active convolution term
        i_start = max(0, k_max - (n - 1))
        i_end = min(n - 1, k_max)
        idx = np.arange(i_start, i_end + 1)
        contrib = best_a[idx] * best_a[k_max - idx]

        i_max = idx[np.argmax(contrib)]

        # partner values for every possible destination j
        # partner = a_{k_max - j} (0 if out of range)
        j_candidates = np.arange(n)
        partner_idx = k_max - j_candidates
        partner_vals = np.where(
            (partner_idx >= 0) & (partner_idx < n),
            best_a[partner_idx],
            0.0
        )
        # choose j that minimises the partner value (greedy)
        j = int(j_candidates[np.argmin(partner_vals)])

        # avoid a no‑op
        if j == i_max:
            # pick a random different index as a fallback
            j = np.random.randint(n)
            if j == i_max:
                steps += 1
                continue

        delta = step_scale * best_a[i_max]
        if delta <= 0.0:
            break

        cand = best_a.copy()
        cand[i_max] -= delta
        cand[j] += delta
        # tiny numerical drifts are fixed by a simplex projection
        cand = project_to_simplex(cand)

        sc = compute_score(cand)
        if sc < best_sc:
            best_sc, best_a = sc, cand.copy()
            step_scale = min(step_scale * 1.07, 0.5)
        else:
            step_scale *= 0.96

        steps += 1

    return best_a, best_sc

# Main entry point
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 12345, budget_s: int = 1000, **kwargs):
    """Search for a step‑function that minimises `evaluate_sequence`."""
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 2.0  # a small safety buffer

    # -------------------------------------------------
    # Choose a sequence length (larger ⇒ more freedom)
    # -------------------------------------------------
    remaining = deadline - start_time
    if remaining > 800:
        n = 32768
    elif remaining > 500:
        n = 16384
    else:
        n = 8192
    print(f"[{time.time():.1f}] chosen length = {n}")

    best_seq = None
    best_score = float("inf")

    # -------------------------------------------------
    # Optional fallback from a previously discovered vector
    # -------------------------------------------------
    try:
        fallback = [float(x) for x in height_sequence_1]    # type: ignore
        fb = evaluate_sequence(fallback)
        print(f"[{time.time():.1f}] fallback score = {fb:.6f}")
        if fb < best_score:
            best_score, best_seq = fb, fallback
    except Exception:
        pass

    # -------------------------------------------------
    # 1. Broad family scans (time‑limited)
    # -------------------------------------------------
    fam_deadline = deadline - 250.0
    a_fam, sc_fam = scan_families(n, fam_deadline)
    if a_fam is not None and sc_fam < best_score:
        best_score, best_seq = sc_fam, a_fam.tolist()
        print(f"[{time.time():.1f}] after family scan = {best_score:.6f}")

    # -------------------------------------------------
    # 2. Random Dirichlet samples
    # -------------------------------------------------
    rand_deadline = deadline - 200.0
    a_rand, sc_rand = random_dirichlet_samples(n, rand_deadline, count=15)
    if a_rand is not None and sc_rand < best_score:
        best_score, best_seq = sc_rand, a_rand.tolist()
        print(f"[{time.time():.1f}] after random init = {best_score:.6f}")

    # -------------------------------------------------
    # 3. Smooth max optimisation
    # -------------------------------------------------
    if best_seq is not None and time.time() < deadline - 140.0:
        print(f"[{time.time():.1f}] starting smooth‑max optimisation")
        a_smooth, sc_smooth = exp_smooth_optimize(
            np.array(best_seq, dtype=float),
            deadline - 130.0,
            max_iters=300_000,
            beta_end=5e4,
            eta0=0.6,
        )
        if sc_smooth < best_score:
            best_score, best_seq = sc_smooth, a_smooth.tolist()
            print(f"[{time.time():.1f}] smooth phase best = {best_score:.6f}")

    # -------------------------------------------------
    # 4. Exact max‑descent (longer than before)
    # -------------------------------------------------
    if best_seq is not None and time.time() < deadline - 80.0:
        print(f"[{time.time():.1f}] starting exact max‑descent")
        a_max, sc_max = max_descent_optimize(
            np.array(best_seq, dtype=float),
            deadline - 70.0,
            max_iters=600_000,
            eta0=0.8,
        )
        if sc_max < best_score:
            best_score, best_seq = sc_max, a_max.tolist()
            print(f"[{time.time():.1f}] after max‑descent best = {best_score:.6f}")

    # -------------------------------------------------
    # 5. Deterministic hill‑climb polishing
    # -------------------------------------------------
    if best_seq is not None and time.time() < deadline - 5.0:
        print(f"[{time.time():.1f}] final deterministic hill‑climb")
        a_final, sc_final = deterministic_hill_climb(
            np.array(best_seq, dtype=float),
            deadline,
            max_steps=400_000,
        )
        if sc_final < best_score:
            best_score, best_seq = sc_final, a_final.tolist()
            print(f"[{time.time():.1f}] hill‑climb improved best = {best_score:.6f}")

    # -------------------------------------------------
    # Fallback to a tiny uniform vector (should never happen)
    # -------------------------------------------------
    if best_seq is None:
        n_small = 256
        best_seq = [1.0 / n_small] * n_small
        best_score = evaluate_sequence(best_seq)

    # -------------------------------------------------
    # Final official evaluation
    # -------------------------------------------------
    final_score = evaluate_sequence(best_seq)
    print(f"\n[{time.time():.1f}] finished – final reported score = {final_score:.6f}")

    # Return a plain Python list as required by the harness
    return best_seq
