"""
Non‑negative step‑function optimiser for `evaluate_sequence`.

Strategy
--------
1. Determine the problem size from the optional global variable
   ``height_sequence_1`` (otherwise use n = 2048).
2. Generate a diverse set of analytic starter families:
   - uniform, central‑binomial, geometric, power‑law
   - Beta‑shaped (U‑ or bell‑shaped) densities
   - random Dirichlet draws
   All candidates are projected onto the probability simplex.
3. Keep the best candidate according to the exact surrogate objective
   `2·n·max(conv(a,a))` (implemented as ``compute_score``).
4. Re‑search the neighbourhood extensively:
   * Run several *Adam* optimisations on a log‑sum‑exp smoothing of the
     max‑convolution, each starting from a perturbed copy of the current best.
   * Each run is limited only by the remaining wall‑clock time.
5. Polish the result with a cheap stochastic hill‑climb that moves a small
   fraction of mass between random positions while staying on the simplex.
6. Return the best sequence found (as a plain Python list).  All timings are
   guarded so the function always finishes well before the 1000 s budget.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Basic utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Linear convolution via real FFT (full length)."""
    m = a.shape[0] + b.shape[0] - 1
    n_fft = next_pow2(m)
    A = np.fft.rfft(a, n=n_fft)
    B = np.fft.rfft(b, n=n_fft)
    C = A * B
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


def compute_score(a: np.ndarray) -> float:
    """
    Exact objective for a simplex‑normalised vector (sum = 1).
    Returns 2·n·max(conv(a,a)).
    """
    n = a.shape[0]
    conv = convolution_fft(a, a)
    return 2.0 * n * float(conv.max())


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex (sum=1, v≥0)."""
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / v.shape[0])

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, v.shape[0] + 1)
    cond = u - cssv / ind > 0
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


# ----------------------------------------------------------------------
# Analytic families for initial guesses
# ----------------------------------------------------------------------


def generate_central_binomial(n: int) -> np.ndarray:
    """a_i = C(2i,i) / 4^i, normalised to sum 1 (monotone decreasing)."""
    a = np.empty(n, dtype=np.float64)
    a[0] = 1.0
    for i in range(1, n):
        a[i] = a[i - 1] * (2.0 * i - 1) / (2.0 * i)
    return project_to_simplex(a)


def generate_powerlaw(n: int, p: float) -> np.ndarray:
    """a_i ∝ (i+1)^(-p), normalised to sum 1."""
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = idx ** (-p)
    return project_to_simplex(a)


def generate_geometric(n: int, r: float) -> np.ndarray:
    """Geometric distribution a_i = (1‑r)·r^i, normalised (exactly sum 1 if r<1)."""
    i = np.arange(n, dtype=np.float64)
    a = (1.0 - r) * (r ** i)
    return project_to_simplex(a)


def generate_beta(n: int, alpha: float, beta_: float) -> np.ndarray:
    """
    Beta‑shaped density on {0,…,n‑1}:
        a_i ∝ (i+1)^{alpha‑1}·(n‑i)^{beta‑1}
    """
    i = np.arange(1, n + 1, dtype=np.float64)          # 1 … n
    a = (i ** (alpha - 1.0)) * ((n - i + 1) ** (beta_ - 1.0))
    return project_to_simplex(a)


# ----------------------------------------------------------------------
# Smooth (log‑sum‑exp) optimiser – Adam
# ----------------------------------------------------------------------


def smooth_optimize(
    start_a: np.ndarray,
    time_deadline: float,
    max_iters: int = 500_000,
    beta_start: float = 5.0,
    beta_end: float = 250.0,
    step0: float = 0.9,
    adam_beta1: float = 0.9,
    adam_beta2: float = 0.999,
    adam_eps: float = 1e-8,
) -> tuple[np.ndarray, float]:
    """
    Adam optimiser for the smoothed max‑convolution objective.
    The surrogate is the log‑sum‑exp of the convolution entries with an
    annealed temperature β.  All iterates stay on the simplex.
    """
    a = start_a.astype(np.float64).copy()
    n = a.shape[0]
    m1 = np.zeros_like(a)
    m2 = np.zeros_like(a)

    best_a = a.copy()
    best_score = compute_score(a)

    for it in range(1, max_iters + 1):
        if time.time() >= time_deadline:
            break

        # anneal temperature
        t_frac = it / max_iters
        beta = beta_start * (beta_end / beta_start) ** t_frac

        # full auto‑convolution
        conv = convolution_fft(a, a)
        maxc = conv.max()

        # soft‑max weights
        exp_w = np.exp(beta * (conv - maxc))
        w = exp_w / exp_w.sum()

        # gradient of the smooth surrogate:
        #   ∂/∂a  = 2 * rev(a) * w   (restricted to length n)
        rev_a = a[::-1]
        grad_full = convolution_fft(rev_a, w)          # length = 3n‑2
        grad = 2.0 * grad_full[n - 1:n - 1 + n]        # central n entries

        # Adam update (projected onto simplex)
        m1 = adam_beta1 * m1 + (1.0 - adam_beta1) * grad
        m2 = adam_beta2 * m2 + (1.0 - adam_beta2) * (grad * grad)
        m1_hat = m1 / (1.0 - adam_beta1 ** it)
        m2_hat = m2 / (1.0 - adam_beta2 ** it)

        step = step0 / np.sqrt(it)
        a_new = a - step * m1_hat / (np.sqrt(m2_hat) + adam_eps)
        a_new = project_to_simplex(a_new)

        a = a_new

        # occasional exact evaluation
        if it % 400 == 0:
            cur_score = compute_score(a)
            if cur_score < best_score:
                best_score = cur_score
                best_a = a.copy()

    # final evaluation
    final_score = compute_score(a)
    if final_score < best_score:
        best_score, best_a = final_score, a.copy()
    return best_a, best_score


# ----------------------------------------------------------------------
# Simple stochastic hill‑climb (mass transfer)
# ----------------------------------------------------------------------


def hill_climb(
    seq,
    time_deadline,
    steps: int = 300_000,
) -> tuple[list[float], float]:
    """
    Tiny stochastic moves (fractional transfer) with simplex projection.
    The step size adapts depending on success/failure.
    """
    a = np.array(seq, dtype=np.float64)
    n = a.shape[0]
    best_a = a.copy()
    best_score = compute_score(best_a)

    step_frac = 0.07  # initial fraction of mass moved each step

    while steps > 0 and time.time() < time_deadline:
        cand = best_a.copy()
        i, j = np.random.randint(n, size=2)
        # move a small fraction from i to j
        delta = step_frac * best_a[i]
        cand[i] -= delta
        cand[j] += delta
        if cand[i] < 0.0:
            cand[i] = 0.0

        cand = project_to_simplex(cand)
        cur_score = compute_score(cand)

        if cur_score < best_score:
            best_score = cur_score
            best_a = cand
            step_frac = min(step_frac * 1.06, 0.5)
        else:
            step_frac *= 0.97

        steps -= 1

    return best_a.tolist(), best_score


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 12345, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative sequence that minimises ``evaluate_sequence``.
    Returns a plain Python list of floats (already on the simplex).
    """
    np.random.seed(seed)
    start_time = time.time()
    # keep a small safety margin for the harness
    deadline = start_time + budget_s - 3.0

    # ------------------------------------------------------------------
    # Determine sequence length
    # ------------------------------------------------------------------
    try:
        n = len(height_sequence_1)  # type: ignore
        print(f"Using provided height_sequence_1 of length {n}.")
    except Exception:
        n = 2048
        print(f"No external sequence provided – using default length {n}.")

    # ------------------------------------------------------------------
    # 0. Build a diverse pool of analytic starters
    # ------------------------------------------------------------------
    starters = []

    # uniform
    starters.append(np.full(n, 1.0 / n, dtype=np.float64))

    # central‑binomial
    starters.append(generate_central_binomial(n))

    # geometric family (r close to 1)
    for r in np.linspace(0.90, 0.9999, 12):
        starters.append(generate_geometric(n, float(r)))

    # power‑law family (0.05 … 2.5)
    for p in np.linspace(0.05, 2.5, 15):
        starters.append(generate_powerlaw(n, float(p)))

    # Beta‑shaped families (random parameters)
    for _ in range(30):
        alpha = np.random.uniform(0.2, 4.5)
        beta_ = np.random.uniform(0.2, 4.5)
        starters.append(generate_beta(n, alpha, beta_))

    # random Dirichlet draws (various concentrations)
    for _ in range(30):
        conc = np.random.uniform(0.05, 5.0)
        a = np.random.gamma(conc, 1.0, size=n)
        a = a / a.sum()
        starters.append(a)

    # Occasionally add a symmetric copy of each candidate (helps flatness)
    symmetric_copy = []
    for s in starters:
        sym = (s + s[::-1]) / 2.0
        symmetric_copy.append(project_to_simplex(sym))
    starters.extend(symmetric_copy)

    # ------------------------------------------------------------------
    # 1. Evaluate all starters and keep the best
    # ------------------------------------------------------------------
    best_seq = None
    best_val = float("inf")
    for idx, cand in enumerate(starters):
        cand = project_to_simplex(cand)
        val = compute_score(cand)
        if val < best_val:
            best_val = val
            best_seq = cand.copy()
        if idx % 20 == 0:
            print(f"Starter {idx:03d}: score = {val:.6f}")

    print(f"Best starter after exhaustive pool: score = {best_val:.6f}")

    # ------------------------------------------------------------------
    # 2. Repeated Adam smoothing runs (time‑limited)
    # ------------------------------------------------------------------
    smooth_deadline = deadline - 20.0         # reserve ~20 s for final hill‑climb
    while time.time() < smooth_deadline - 2.0:
        remaining = smooth_deadline - time.time()
        # allocate a modest chunk (max 25 s) for this run
        run_time = min(25.0, remaining)
        run_deadline = time.time() + run_time

        # start from a slightly perturbed version of the current best
        pert = best_seq + 0.02 * np.random.randn(n)
        pert = project_to_simplex(pert)

        # estimate a safe number of iterations (≈1500 it/s)
        max_iters = int(min(500_000, run_time * 1500))

        a_opt, score_opt = smooth_optimize(
            pert,
            time_deadline=run_deadline,
            max_iters=max_iters,
            beta_start=5.0,
            beta_end=300.0,
            step0=0.95,
        )
        if score_opt < best_val:
            best_val = score_opt
            best_seq = a_opt.copy()
            print(f"Smooth run improved score to {best_val:.6f}")

    # ------------------------------------------------------------------
    # 3. Final stochastic hill‑climb (very cheap, pure exploration)
    # ------------------------------------------------------------------
    hill_deadline = deadline
    if time.time() < hill_deadline:
        hill_steps = 400_000
        print(f"Starting final hill‑climb with up to {hill_steps} steps.")
        polished_seq, hill_score = hill_climb(
            best_seq.tolist(),
            time_deadline=hill_deadline,
            steps=hill_steps,
        )
        if hill_score < best_val:
            best_val = hill_score
            best_seq = np.array(polished_seq, dtype=np.float64)
            print(f"Hill‑climb improved final score to {best_val:.6f}")

    # ------------------------------------------------------------------
    # 4. Fallback safeguard
    # ------------------------------------------------------------------
    if best_seq is None:
        best_seq = np.full(n, 1.0 / n, dtype=np.float64)
        best_val = compute_score(best_seq)

    # ------------------------------------------------------------------
    # 5. Final reporting using the provided evaluate_sequence()
    # ------------------------------------------------------------------
    final_score = evaluate_sequence(best_seq.tolist())
    print(f"\nSearch finished. Final reported score = {final_score:.6f}")
    return best_seq.tolist()
