"""
search for a step‑function (non‑negative sequence) that minimises

        2·n·max(convolve(a,a)) / (sum(a)**2)

under the simplex constraint.

The algorithm is a multi‑phase optimisation:

0.  Load a fallback solution if provided.
1.  Fast analytic / random scan to obtain a good starter.
2.  Long Adam optimisation on a smooth‑max surrogate
    (β is increased up to a very large value, making the surrogate
    close to the true max).
3.  Projected sub‑gradient descent with a larger initial step.
4.  A few iterations of the classic LP‑refinement (mix‑in the LP
    optimum with a relatively high mixing weight).
5.  Symmetrisation (optional).
6.  Final simulated‑annealing style hill‑climb with many steps.

All phases respect the global wall‑clock budget (default 1000 s) and
the best sequence found so far is always kept.  The returned list is
ready for ``evaluate_sequence``.
"""

import time
import math
import numpy as np
from scipy.optimize import linprog

# ----------------------------------------------------------------------
# Basic utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """
    Full (aperiodic) convolution via real FFT.  Both inputs are 1‑D.
    """
    m = a.size + b.size - 1
    nfft = next_pow2(m)
    A = np.fft.rfft(a, n=nfft)
    B = np.fft.rfft(b, n=nfft)
    C = A * B
    conv = np.fft.irfft(C, n=nfft)[:m]
    return conv


def compute_score(a: np.ndarray) -> float:
    """Exact objective for a probability vector (∑a = 1):
       2·n·max(a*a)
    """
    n = a.shape[0]
    conv = convolution_fft(a, a)
    return 2.0 * n * float(conv.max())


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    # implementation of the algorithm from
    # https://arxiv.org/pdf/1101.6081.pdf
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / v.shape[0])

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, v.shape[0] + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.where(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def softmax(z: np.ndarray) -> np.ndarray:
    """Numerically stable soft‑max."""
    m = np.max(z)
    e = np.exp(z - m)
    return e / e.sum()


# ----------------------------------------------------------------------
# Analytic families (used for the quick initial scan)
# ----------------------------------------------------------------------


def symmetric_product_family(n: int, alpha: float) -> np.ndarray:
    """U‑shape: w_i = ((i+0.5)*(n-0.5-i))^{-alpha}."""
    i = np.arange(n, dtype=np.float64)
    w = ((i + 0.5) * (n - 0.5 - i)) ** (-alpha)
    w[np.isinf(w)] = 0.0
    a = w / w.sum()
    return a


def left_heavy_power_family(n: int, beta: float) -> np.ndarray:
    """Monotone decreasing: w_i = (i+0.5)^{-beta}."""
    i = np.arange(n, dtype=np.float64)
    w = (i + 0.5) ** (-beta)
    a = w / w.sum()
    return a


def exp_u_family(n: int, gamma: float) -> np.ndarray:
    """Exponential decay from the ends: w_i = exp(-gamma * dist)."""
    i = np.arange(n, dtype=np.float64)
    dist = np.minimum(i, n - 1 - i)
    w = np.exp(-gamma * dist)
    a = w / w.sum()
    return a


def beta_shape_family(n: int, a_param: float, b_param: float) -> np.ndarray:
    """
    Discretised Beta distribution on {0,…,n‑1}:
        w_i ∝ (i+1)^{a-1} * (n-i)^{b-1}
    """
    i = np.arange(1, n + 1, dtype=np.float64)
    w = (i ** (a_param - 1.0)) * ((n - i + 1) ** (b_param - 1.0))
    w[np.isnan(w)] = 0.0
    a = w / w.sum()
    return a


def asymmetric_power_family(n: int, p_left: float, p_right: float) -> np.ndarray:
    """
    Generalised U‑shape with two exponents:
        w_i = (i+0.5)^{-p_left} * (n-0.5-i)^{-p_right}
    """
    i = np.arange(n, dtype=np.float64)
    w = (i + 0.5) ** (-p_left) * (n - 0.5 - i) ** (-p_right)
    w[np.isinf(w)] = 0.0
    a = w / w.sum()
    return a


def random_monotone_start(n: int) -> np.ndarray:
    """Random decreasing shape (useful for diversification)."""
    r = np.random.rand(n) ** np.random.uniform(0.5, 5.0)
    r.sort()
    a = r[::-1]
    a /= a.sum()
    return a


def random_u_start(n: int) -> np.ndarray:
    """U‑shape with random exponential decay."""
    gamma = np.random.uniform(0.0, 6.0)
    return exp_u_family(n, gamma)


# ----------------------------------------------------------------------
# Adam optimisation of a smooth‑max surrogate
# ----------------------------------------------------------------------


def adam_smooth_optimize(start_a: np.ndarray,
                         deadline: float,
                         max_iters: int = 800_000,
                         beta_start: float = 4.0,
                         beta_end: float = 5000.0,
                         lr: float = 0.02) -> tuple[np.ndarray, float]:
    """
    Minimise the smooth surrogate
        f_β(a) = (1/β)·log Σ_k exp(β·(a*a)_k)
    using Adam on the soft‑max parametrisation a = softmax(z).
    The smoothing parameter β is increased exponentially.
    """
    eps = 1e-12
    z = np.log(start_a + eps)

    m = np.zeros_like(z)
    v = np.zeros_like(z)
    beta1 = 0.9
    beta2 = 0.999
    eps_adam = 1e-12

    best_a = softmax(z)
    best_score = compute_score(best_a)

    for it in range(1, max_iters + 1):
        if time.time() >= deadline:
            break

        # exponential schedule for β
        t = it / max_iters
        beta = beta_start * (beta_end / beta_start) ** t

        a = softmax(z)                     # current point on simplex
        conv = convolution_fft(a, a)
        maxc = conv.max()

        # soft‑max weights for the smooth max
        exp_w = np.exp(beta * (conv - maxc))
        w = exp_w / exp_w.sum()

        # gradient wrt a : ∂f/∂a = 2 * (a_rev * w)   (see derivation)
        a_rev = a[::-1]
        conv_rev = convolution_fft(a_rev, w)
        grad_a = 2.0 * conv_rev[len(a) - 1:len(a) - 1 + len(a)]

        # back‑prop through soft‑max (see standard derivation)
        dot_ag = np.dot(a, grad_a)
        grad_z = a * grad_a - dot_ag * a

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad_z
        v = beta2 * v + (1.0 - beta2) * (grad_z ** 2)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        z -= lr * m_hat / (np.sqrt(v_hat) + eps_adam)

        # occasional reporting
        if it % 1000 == 0:
            sc = compute_score(softmax(z))
            if sc < best_score:
                best_score = sc
                best_a = softmax(z).copy()
                print(f"[{time.time():.1f}] Adam improved to {best_score:.6f}")

    # final evaluation
    a_final = softmax(z)
    sc_final = compute_score(a_final)
    if sc_final < best_score:
        best_score = sc_final
        best_a = a_final.copy()
    return best_a, best_score


# ----------------------------------------------------------------------
# Projected sub‑gradient descent (cheapest iteration)
# ----------------------------------------------------------------------


def subgrad_optimize(start_a: np.ndarray,
                     deadline: float,
                     max_iters: int = 3_000_000,
                     step0: float = 0.3) -> tuple[np.ndarray, float]:
    """
    Sub‑gradient descent for   f(a) = 2·n·max(a*a) .
    Uses a diminishing stepsize step0/√t and Euclidean projection onto the simplex.
    """
    n = start_a.shape[0]
    a = start_a.astype(np.float64).copy()

    best_a = a.copy()
    best_score = compute_score(a)

    for t in range(1, max_iters + 1):
        if time.time() >= deadline:
            break

        conv = convolution_fft(a, a)
        k = int(np.argmax(conv))

        # sub‑gradient of the max‑convolution entry
        idx = np.arange(n)
        j = k - idx
        mask = (j >= 0) & (j < n)
        g = np.zeros_like(a)
        g[mask] = 2.0 * a[j[mask]]
        # diagonal terms (i=j) have coefficient 1 instead of 2
        diag = (j == idx) & mask
        if np.any(diag):
            g[diag] = a[diag]

        step = step0 / np.sqrt(t)
        a = project_to_simplex(a - step * g)

        if t % 5000 == 0:
            sc = compute_score(a)
            if sc < best_score:
                best_score = sc
                best_a = a.copy()
                print(f"[{time.time():.1f}] sub‑grad improved to {best_score:.6f}")

    final_sc = compute_score(a)
    if final_sc < best_score:
        best_score = final_sc
        best_a = a.copy()
    return best_a, best_score


# ----------------------------------------------------------------------
# Linear‑programming refinement (the “state‑of‑the‑art” iteration)
# ----------------------------------------------------------------------


def lp_refine_step(f: np.ndarray, M: float, mix_weight: float = 0.2) -> np.ndarray:
    """
    Solve   max Σ g_j   s.t.   (f * g)_k ≤ M  (for all k),
    g ≥ 0, then return the convex mixture
        f_new = (1‑t)·f + t·g_norm
    where g_norm = g / Σ g.
    """
    n = f.shape[0]
    m = 2 * n - 1

    # build convolution matrix A so that A @ g = f * g
    A = np.zeros((m, n), dtype=np.float64)
    for i in range(n):
        A[i:i + n, i] = f[i]

    c = -np.ones(n, dtype=np.float64)          # maximise Σ g ⇒ minimise -Σ g
    bounds = [(0.0, None) for _ in range(n)]
    b = np.full(m, M, dtype=np.float64)

    # solve – the solver is fast for these modest dimensions
    res = linprog(c, A_ub=A, b_ub=b, bounds=bounds,
                  method='highs', options={'presolve': True, 'time_limit': 15})

    if not res.success:
        return f

    g = np.maximum(res.x, 0.0)
    sum_g = g.sum()
    if sum_g <= 0.0:
        return f

    g_norm = g / sum_g
    f_new = (1.0 - mix_weight) * f + mix_weight * g_norm
    f_new = project_to_simplex(f_new)
    return f_new


# ----------------------------------------------------------------------
# Stochastic hill‑climb (simulated‑annealing style)
# ----------------------------------------------------------------------


def hill_climb_sa(start_a: np.ndarray,
                  deadline: float,
                  max_steps: int = 3_000_000) -> tuple[np.ndarray, float]:
    """
    Random walk on the simplex with an exponential‐cooling schedule.
    Occasionally accepts uphill moves.
    """
    n = start_a.size
    cur_a = start_a.copy()
    cur_score = compute_score(cur_a)

    best_a = cur_a.copy()
    best_score = cur_score

    # initial temperature is a tiny fraction of the current score
    T0 = best_score * 1e-4
    T = T0
    step_scale = 0.15

    for step in range(1, max_steps + 1):
        if time.time() >= deadline:
            break

        cand = cur_a.copy()

        # two simple perturbations: swap or small mass transfer
        if np.random.rand() < 0.5:                     # swap
            i, j = np.random.randint(n, size=2)
            cand[i], cand[j] = cand[j], cand[i]
        else:                                          # mass transfer
            i, j = np.random.randint(n, size=2)
            if i != j:
                delta = step_scale * cand[i]         # move a fraction of i's mass
                cand[i] -= delta
                cand[j] += delta
                if cand[i] < 0.0:
                    cand[i] = 0.0

        cand = project_to_simplex(cand)
        cand_score = compute_score(cand)

        delta = cand_score - cur_score
        # Metropolis acceptance
        if delta < 0.0 or np.random.rand() < math.exp(-delta / (T + 1e-18)):
            cur_a = cand
            cur_score = cand_score
            if cur_score < best_score:
                best_score = cur_score
                best_a = cur_a.copy()
                # slowly increase step size when we improve
                step_scale = min(step_scale * 1.07, 0.6)
        else:
            # shrink step size on rejection
            step_scale *= 0.97

        # linear cooling (total budget → T → 0)
        T = T0 * (1.0 - step / max_steps)

    return best_a, best_score
