def deterministic_hill_climb(a_start, deadline, max_steps=300000, step_factor=0.2):
    a = a_start.copy()
    n = a.size
    best_a = a.copy()
    best_sc = compute_score(best_a)
    step_scale = step_factor
    while time.time() < deadline and max_steps>0:
        conv = convolution_fft(a, a)
        k_max = int(conv.argmax())
        # contributions ranges i_start..i_end
        i_start = max(0, k_max - (n - 1))
        i_end = min(n - 1, k_max)
        idx = np.arange(i_start, i_end+1)
        # compute contributions a_i * a_kmax-i
        contrib = a[idx] * a[k_max - idx]
        i_max = idx[np.argmax(contrib)]
        # Determine best destination j: minimize partner weight at k_max
        # partner for j is a[k_max - j] if index in range else 0.
        # So we want j where partner is minimal.
        # Let's compute partner array:
        partner = np.zeros(n)
        low = max(0, k_max - (n - 1))
        high = min(n - 1, k_max)
        # For j in [low, high], partner[j] = a[k_max - j] else 0.
        partner[low:high+1] = a[k_max - np.arange(low, high+1)]
        # find j with minimal partner (tie broken random)
        # Exclude i_max.
        mask = np.arange(n) != i_max
        # maybe also avoid index with high a (like to not increase endpoint squares)
        # but rely on partner.
        candidate_js = np.where(mask)[0]
        # compute only within mask.
        # Choose j with minimal partner value.
        # To reduce computational overhead, we can compute min in O(n).
        j = candidate_js[np.argmin(partner[candidate_js])]
        # move delta
        delta = step_scale * a[i_max]
        if delta <= 0: break
        a[i_max] -= delta
        a[j] += delta
        a = project_to_simplex(a) # maintain sum=1 and nonnegativity
        sc = compute_score(a)
        if sc < best_sc:
            best_sc, best_a = sc, a.copy()
            step_scale = min(step_scale * 1.05, 0.5)
        else:
            step_scale *= 0.97
        max_steps -= 1
    return best_a, best_sc

def deterministic_hill_climb(start_seq, deadline, max_steps=200000):

    a = start_seq.copy()
    n = a.size
    best_a = a.copy()
    best_sc = compute_score(a)
    step_scale = 0.1
    steps = 0
    while time.time() < deadline and steps < max_steps:
        conv = convolution_fft(a, a)
        k_max = int(conv.argmax())
        i_start = max(0, k_max - (n - 1))
        i_end = min(n - 1, k_max)
        idx = np.arange(i_start, i_end+1)
        # contributions
        contrib = a[idx] * a[k_max - idx]
        i_max = idx[np.argmax(contrib)]
        # partner weights for destination j
        # compute partner array for all j:
        partner = np.zeros(n)
        low = max(0, k_max - (n - 1))
        high = min(n - 1, k_max)
        # For j in [low, high], partner[j] = a[k_max - j]
        partner[low:high+1] = a[k_max - np.arange(low, high+1)]
        # choose destination j minimizing partner, exclude i_max
        mask = np.arange(n) != i_max
        # optional: also avoid j where a[j] zero but partner zero too maybe okay
        j_candidates = np.where(mask)[0]
        # For speed we can pre-select limited set: random subset of maybe 100 candidates.
        # But direct argmin over all n is fine.
        j = j_candidates[np.argmin(partner[j_candidates])]
        # move mass
        delta = step_scale * a[i_max]
        if delta <= 0: break
        a[i_max] -= delta
        a[j] += delta
        # Project onto simplex to fix any rounding errors (and ensure nonneg).
        a = project_to_simplex(a)
        sc = compute_score(a)
        if sc < best_sc:
            best_sc, best_a = sc, a.copy()
            step_scale = min(step_scale * 1.05, 0.5)
        else:
            step_scale *= 0.96
        steps += 1
    return best_a, best_sc

def adam_smooth_optimize(start_a: np.ndarray, deadline: float, max_iters: int = 200_000, beta_start: float = 1.0, beta_end: float = 20000.0, lr0: float = 0.5, beta1=0.9, beta2=0.999, epsilon=1e-8):
    w = np.log(start_a + 1e-15)  # initial logits
    n = start_a.shape[0]
    # Adam state
    m = np.zeros_like(w)
    v = np.zeros_like(w)
    best_a = start_a.copy()
    best_sc = compute_score(best_a)
    for t in range(1, max_iters+1):
        if time.time() >= deadline:
            break
        # compute a via softmax
        w_shift = w - np.max(w)
        a = np.exp(w_shift)
        a_sum = a.sum()
        a = a / a_sum

        # compute conv
        conv = convolution_fft(a, a)
        maxc = conv.max()
        # schedule beta
        beta = beta_start * (beta_end / beta_start) ** (t / max_iters)

        # compute softmax weights over conv
        # to avoid overflow when beta large, we already subtract maxc
        exp_weights = np.exp(beta * (conv - maxc))
        w_conv = exp_weights / exp_weights.sum()

        # gradient wrt a
        a_rev = a[::-1]
        grad_a = 2.0 * convolution_fft(a_rev, w_conv)[n-1:n-1+n]

        # gradient wrt w (logits) using softmax Jacobian
        dot = np.dot(a, grad_a)
        grad_w = a * (grad_a - dot)

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad_w
        v = beta2 * v + (1.0 - beta2) * (grad_w ** 2)

        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)

        lr = lr0 / np.sqrt(t)

        w = w - lr * m_hat / (np.sqrt(v_hat) + epsilon)

        # (optional) maintain w shift to avoid overflow
        # w = w - np.max(w)

        # occasional evaluation
        if t % 500 == 0:
            sc = compute_score(a)
            if sc < best_sc:
                best_sc, best_a = sc, a.copy()
                print(f"[{time.time():.1f}] Adam iter {t}, new best score {best_sc:.6f}")

    # final check
    w_shift = w - np.max(w)
    a = np.exp(w_shift)
    a = a / a.sum()
    sc = compute_score(a)
    if sc < best_sc:
        best_sc, best_a = sc, a.copy()
    return best_a, best_sc

"""
Combined multi‑phase optimizer for the “step‑function” bound.

The routine follows four stages that respect a 1000 s wall‑clock budget:

1. **Simple family scan** – try exponential, power‑law and a 2‑parameter
   discrete Beta family (plus a fallback from a previously discovered
   sequence).  The best candidate becomes the starting point.
2. **Adam‑based smooth‑max descent** – minimise a log‑sum‑exp (smooth‑max)
   surrogate of the convolution using an Adam update on the logits of a
   probability vector (soft‑max parametrisation).  The smoothing parameter
   β grows exponentially from `beta_start` to `beta_end`, turning the
   surrogate into the true max.
3. **Exact max‑descent (KL‑mirror)** – sub‑gradient descent on the true
   maximal convolution entry with a multiplicative (exponentiated‑gradient)
   update; this refines the solution obtained by the smooth phase.
4. **Deterministic hill‑climb** – repeatedly moves a small packet of mass
   from the index that contributes most to the current maximal convolution
   entry towards the index whose partner weight in that entry is smallest.
   Moves are accepted only if they improve the exact objective.

All phases keep the vector on the probability simplex (non‑negative,
sum = 1), which makes the final `evaluate_sequence` invariant under scaling.
The best sequence found is returned as a plain list of floats.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Basic linear‑algebra utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full aperiodic convolution using a real FFT (zero‑padded)."""
    m = a.size + b.size - 1
    nfft = next_pow2(m)
    A = np.fft.rfft(a, n=nfft)
    B = np.fft.rfft(b, n=nfft)
    C = A * B
    return np.fft.irfft(C, n=nfft)[:m]


def compute_score(a: np.ndarray) -> float:
    """
    Exact objective used by `evaluate_sequence` for a vector that already
    satisfies sum = 1.  Returns 2·n·max(a*a) where the convolution is full.
    """
    n = a.shape[0]
    conv = convolution_fft(a, a)
    return 2.0 * n * float(conv.max())


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / v.shape[0])

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, v.shape[0] + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.where(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


# ----------------------------------------------------------------------
# Parameterised families of candidate vectors
# ----------------------------------------------------------------------


def exponential_shape(n: int, lam: float) -> np.ndarray:
    """a_i ∝ exp(-lam·i), normalised to sum = 1."""
    i = np.arange(n, dtype=float)
    w = np.exp(-lam * i) if lam > 0.0 else np.ones_like(i)
    return w / w.sum()


def power_shape(n: int, p: float) -> np.ndarray:
    """a_i ∝ (i+1)^(-p), normalised to sum = 1."""
    i = np.arange(1, n + 1, dtype=float)
    w = i ** (-p) if p > 0.0 else np.ones_like(i)
    return w / w.sum()


def beta_shape(n: int, a: float, b: float) -> np.ndarray:
    """
    Discrete Beta distribution on {0,…,n‑1}:
        a_i ∝ ((i+0.5)/n)^(a‑1) * (1‑(i+0.5)/n)^(b‑1)
    """
    x = (np.arange(n, dtype=float) + 0.5) / n
    w = np.power(x, a - 1.0) * np.power(1.0 - x, b - 1.0)
    w[np.isnan(w)] = 0.0
    w = np.maximum(w, 0.0)
    if w.sum() == 0.0:
        w = np.ones_like(w)
    return w / w.sum()

def scan_exponential_and_power(n: int, deadline: float):
    """Exhaustive scan of exponential and power‑law families."""
    best_a, best_sc = None, float("inf")

    lam_grid = np.concatenate((np.array([0.0]), np.logspace(-5, -2, 200)))
    for lam in lam_grid:
        if time.time() >= deadline:

        a = exponential_shape(n, lam)
        sc = compute_score(a)
        if sc < best_sc:
            best_sc, best_a = sc, a.copy()

    p_grid = np.linspace(0.0, 6.0, 200)
    for p in p_grid:
        if time.time() >= deadline:
            break
        a = power_shape(n, p)
        sc = compute_score(a)
        if sc < best_sc:
            best_sc, best_a = sc, a.copy()

    return best_a, best_sc


def scan_beta_family(n: int, deadline: float):
    """Scan a modest 2‑parameter Beta family."""
    best_a, best_sc = None, float("inf")
    a_grid = np.concatenate((np.logspace(-0.7, -0.1, 12), np.logspace(-0.1, 0.3, 12),

            sc = compute_score(a_vec)
            if sc < best_sc:

    return best_a, best_sc

def adam_smooth_optimize(start_a: np.ndarray,
                         deadline: float,
                         max_iters: int = 200_000,
                         beta_start: float = 1.0,
                         beta_end: float = 20_000.0,
                         lr0: float = 0.5) -> tuple[np.ndarray, float]:
    """
    Adam optimiser on the log‑sum‑exp approximation of the max convolution.
    The distribution is kept on the simplex via a soft‑max parametrisation.
    """
    n = start_a.shape[0]
    # initialise logits (soft‑max parameters)
    w = np.log(start_a + 1e-15)
    m = np.zeros_like(w)
    v = np.zeros_like(w)
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8

    best_a = start_a.copy()
    best_sc = compute_score(best_a)

    for t in range(1, max_iters + 1):
        if time.time() >= deadline:
            break

        # soft‑max to obtain a probability vector
        w_shift = w - np.max(w)          # for numerical stability
        a = np.exp(w_shift)
        a /= a.sum()

        # convolution and smooth‑max weights
        conv = convolution_fft(a, a)
        maxc = conv.max()
        beta = beta_start * (beta_end / beta_start) ** (t / max_iters)

        # safe log‑sum‑exp weighting
        exp_w = np.exp(beta * (conv - maxc))
        w_conv = exp_w / exp_w.sum()

        # gradient w.r.t. a (smooth max)
        grad_a = 2.0 * convolution_fft(a[::-1], w_conv)[n - 1:n - 1 + n]

        # convert to gradient w.r.t. logits (soft‑max Jacobian)
        dot = np.dot(a, grad_a)
        grad_w = a * (grad_a - dot)

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad_w
        v = beta2 * v + (1.0 - beta2) * (grad_w ** 2)

        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)

        lr = lr0 / np.sqrt(t)
        w -= lr * m_hat / (np.sqrt(v_hat) + eps)

        # occasional exact evaluation
        if t % 500 == 0:
            sc = compute_score(a)
            if sc < best_sc:
                best_sc, best_a = sc, a.copy()
                print(f"[{time.time():.1f}] Adam iter {t}, new best {best_sc:.6f}")

    # final evaluation
    w_shift = w - np.max(w)
    a = np.exp(w_shift)
    a /= a.sum()
    sc = compute_score(a)
    if sc < best_sc:
        best_sc, best_a = sc, a.copy()
    return best_a, best_sc

def max_descent_optimize(start_a: np.ndarray,
                         deadline: float,
                         max_iters: int = 300_000,
                         eta0: float = 0.5) -> tuple[np.ndarray, float]:
    """
    Sub‑gradient descent on the true maximum of the convolution.
    Uses a KL‑mirror (multiplicative) update and keeps the vector on the simplex.
    """
    a = np.maximum(start_a.astype(float), 1e-15)
    a /= a.sum()
    n = a.shape[0]

    best_a = a.copy()
    best_sc = compute_score(a)

    for it in range(1, max_iters + 1):
        if time.time() >= deadline:
            break

        conv = convolution_fft(a, a)
        k_max = int(conv.argmax())

        # sub‑gradient of the active entry
        g = np.zeros_like(a)
        i_start = max(0, k_max - (n - 1))
        i_end = min(n - 1, k_max)
        idx = np.arange(i_start, i_end + 1)
        g[idx] = 2.0 * a[k_max - idx]

        eta = eta0 / np.sqrt(it)

        a = np.maximum(a * np.exp(-eta * g), 1e-15)
        a /= a.sum()

        if it % 250 == 0:
            sc = compute_score(a)
            if sc < best_sc:
                best_sc, best_a = sc, a.copy()
                print(f"[{time.time():.1f}] max‑descent it {it}, best {best_sc:.6f}")

    # final check
    sc = compute_score(a)
    if sc < best_sc:
        best_sc, best_a = sc, a.copy()
    return best_a, best_sc


# ----------------------------------------------------------------------
# Deterministic hill‑climb that moves mass away from the dominant
# convolution contribution towards the index with the smallest partner.
# ----------------------------------------------------------------------


def deterministic_hill_climb(start_seq: np.ndarray,
                             deadline: float,
                             max_steps: int = 300_000) -> tuple[np.ndarray, float]:
    """Greedy polishing that always extracts mass from the worst contributor."""
    a = start_seq.copy()
    n = a.size
    best_a = a.copy()
    best_sc = compute_score(best_a)

    step_scale = 0.10
    steps = 0

    while steps < max_steps and time.time() < deadline:
        conv = convolution_fft(a, a)
        k_max = int(conv.argmax())

        i_start = max(0, k_max - (n - 1))
        i_end = min(n - 1, k_max)
        idx = np.arange(i_start, i_end + 1)

        contrib = a[idx] * a[k_max - idx]
        i_max = idx[np.argmax(contrib)]

        # partner weight for every possible destination j
        partner = np.zeros(n)
        low = max(0, k_max - (n - 1))
        high = min(n - 1, k_max)
        partner[low:high + 1] = a[k_max - np.arange(low, high + 1)]

        mask = np.arange(n) != i_max
        j_candidates = np.where(mask)[0]
        j = j_candidates[np.argmin(partner[j_candidates])]

        delta = step_scale * a[i_max]
        if delta <= 0.0:
            break

        cand = a.copy()
        cand[i_max] -= delta
        cand[j] += delta
        cand = project_to_simplex(cand)

        sc = compute_score(cand)
        if sc < best_sc:
            best_sc, best_a = sc, cand.copy()
            step_scale = min(step_scale * 1.07, 0.5)
        else:
            step_scale *= 0.96

        a = cand
        steps += 1

    return best_a, best_sc

# Main entry point expected by the judging harness
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 12345, budget_s: int = 1000, **kwargs):
    """Search for a step‑function that minimises `evaluate_sequence`."""
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 2.0  # tiny safety margin

    # -------------------------------------------------
    # Choose a sequence length (larger gives more freedom)
    # -------------------------------------------------
    remaining = deadline - start_time
    if remaining > 800:
        n = 16384
    elif remaining > 500:
        n = 8192
    else:
        n = 4096
    print(f"[{time.time():.1f}] chosen length = {n}")

    best_seq = None
    best_score = float("inf")

    # -------------------------------------------------
    # 0.  Fallback on any previously discovered vector
    # -------------------------------------------------
    try:
        # global variable may contain a list of floats
        fallback = [float(x) for x in globals()["height_sequence_1"]]
        # up‑sample / truncate to the chosen length
        if len(fallback) != n:
            arr = np.array(fallback, dtype=float)
            arr = np.maximum(arr, 0.0)
            if arr.sum() == 0:
                raise ValueError
            arr /= arr.sum()
            x_old = np.arange(arr.size, dtype=float)
            x_new = np.linspace(0, arr.size - 1, n)
            arr = np.interp(x_new, x_old, arr)
            arr = np.maximum(arr, 0.0)
            arr /= arr.sum()
            fallback = arr.tolist()
        fb = evaluate_sequence(fallback)
        print(f"[{time.time():.1f}] fallback score = {fb:.6f}")
        if fb < best_score:
            best_score, best_seq = fb, fallback
    except Exception:
        pass

    # -------------------------------------------------
    # 1. Quick scans of simple parametric families
    # -------------------------------------------------
    a_exp, sc_exp = scan_exponential_and_power(n, deadline - 180.0)
    if a_exp is not None and sc_exp < best_score:
        best_score, best_seq = sc_exp, a_exp.tolist()
        print(f"[{time.time():.1f}] after exp/power scan = {best_score:.6f}")

    a_beta, sc_beta = scan_beta_family(n, deadline - 150.0)
    if a_beta is not None and sc_beta < best_score:
        best_score, best_seq = sc_beta, a_beta.tolist()
        print(f"[{time.time():.1f}] after beta scan = {best_score:.6f}")

    # -------------------------------------------------
    # 2. A few random Dirichlet starts for diversification
    # -------------------------------------------------
    for _ in range(3):
        if time.time() >= deadline - 130.0:
            break
        rnd = np.random.rand(n)
        rnd /= rnd.sum()
        sc = compute_score(rnd)
        if sc < best_score:
            best_score, best_seq = sc, rnd.tolist()
            print(f"[{time.time():.1f}] random start improved to {best_score:.6f}")

    # -------------------------------------------------
    # 3. Adam‑based smooth‑max optimisation
    # -------------------------------------------------
    if best_seq is not None and time.time() < deadline - 80.0:
        print(f"[{time.time():.1f}] starting Adam smooth‑max phase")
        a_opt, sc_opt = adam_smooth_optimize(
            np.array(best_seq, dtype=float),
            deadline - 60.0,
            max_iters=200_000,
            beta_start=1.0,
            beta_end=20_000.0,
            lr0=0.6,
        )
        if sc_opt < best_score:
            best_score, best_seq = sc_opt, a_opt.tolist()
            print(f"[{time.time():.1f}] after Adam phase best = {best_score:.6f}")

    # -------------------------------------------------
    # 4. Exact max‑descent (KL‑mirror) for fine‑tuning
    # -------------------------------------------------
    if best_seq is not None and time.time() < deadline - 40.0:
        print(f"[{time.time():.1f}] starting exact max‑descent")
        a_opt2, sc_opt2 = max_descent_optimize(
            np.array(best_seq, dtype=float),
            deadline - 30.0,
            max_iters=300_000,
            eta0=0.5,
        )
        if sc_opt2 < best_score:
            best_score, best_seq = sc_opt2, a_opt2.tolist()
            print(f"[{time.time():.1f}] after max‑descent best = {best_score:.6f}")

    # -------------------------------------------------
    # 5. Deterministic hill‑climb polishing
    # -------------------------------------------------
    if best_seq is not None and time.time() < deadline - 5.0:
        print(f"[{time.time():.1f}] final deterministic hill‑climb")
        a_final, sc_final = deterministic_hill_climb(
            np.array(best_seq, dtype=float),
            deadline,
            max_steps=400_000,
        )
        if sc_final < best_score:
            best_score, best_seq = sc_final, a_final.tolist()
            print(f"[{time.time():.1f}] hill‑climb improved best = {best_score:.6f}")

    # -------------------------------------------------
    # Safeguard: tiny uniform vector if everything failed
    # -------------------------------------------------
    if best_seq is None:
        n_small = 256
        best_seq = [1.0 / n_small] * n_small
        best_score = evaluate_sequence(best_seq)

    # -------------------------------------------------
    # Final official evaluation (the harness will also call it)
    # -------------------------------------------------
    final_score = evaluate_sequence(best_seq)
    print(f"\n[{time.time():.1f}] finished – final reported score = {final_score:.6f}")

    # The harness expects a plain Python list of floats.
    return best_seq
