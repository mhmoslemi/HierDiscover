def adam_opt(z0, loss_grad_func, steps, lr=0.01, beta1=0.9, beta2=0.999, eps=1e-8):
   m=0, v=0
   z=z0.copy()
   for t in range(1, steps+1):
       f,g = loss_grad_func(z)
       m = beta1*m + (1-beta1)*g
       v = beta2*v + (1-beta2)*(g*g)
       m_hat = m/(1-beta1**t)
       v_hat = v/(1-beta2**t)
       z = z - lr * m_hat / (np.sqrt(v_hat)+eps)
   return z

"""
Propose a step‑function (non‑negative sequence) that minimises
`evaluate_sequence`.

Approach
--------
* Fix a moderate length (≈8 k) – large enough to give a good
  asymptotic factor but still cheap to convolve many times.
* Generate a diverse pool of initial candidates:
    – uniform, several analytic families (U‑shaped, power‑law,
      exponential, β‑distribution),
    – random Dirichlet draws,
    – random monotone draws.
* Keep the best candidate seen so far (exact score via FFT).
* Refine the best candidate many times with a *smooth surrogate* of the
  max‑convolution:
        fβ(z) = (1/β)·log∑exp(β·(softmax(z)*softmax(z))_k)
  Optimise the log‑softmax parameters `z` with Adam (several β‑values,
  warm‑starting each stage).  The surrogate drives the solution towards
  a true maximiser of the max‑convolution while remaining smooth.
* After the surrogate phase run a long **exponentiated sub‑gradient**
  (mirror) descent on the true objective `2·n·max(a*a)`.
  The step‑size decays as ηₜ≈c/√t; a simple momentum variant is used.
* Optional symmetrisation (averaging with the reversed vector) – only
  kept if it improves the score.
* Finish with a cheap simulated‑annealing hill‑climb that swaps mass
  between entries.
* The whole routine respects a wall‑clock deadline (default 1000 s) and
  returns the best list of floats found.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# FFT‑based convolution utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Linear convolution via real FFT (full length a+b‑1)."""
    m = a.size + b.size - 1
    nfft = next_pow2(m)
    A = np.fft.rfft(a, n=nfft)
    B = np.fft.rfft(b, n=nfft)
    C = A * B
    conv = np.fft.irfft(C, n=nfft)[:m]
    return conv


def compute_score(a: np.ndarray) -> float:
    """
    Exact objective for a probability vector (∑a = 1):
        score = 2·n·max(a*a)
    """
    n = a.shape[0]
    conv = convolution_fft(a, a)
    return 2.0 * n * float(conv.max())


# ----------------------------------------------------------------------
# Simplex helpers
# ----------------------------------------------------------------------


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection onto the probability simplex."""
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / v.shape[0])

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, v.shape[0] + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.where(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def softmax(z: np.ndarray) -> np.ndarray:
    """Stable soft‑max (maps ℝⁿ → simplex)."""
    z_max = np.max(z)
    e = np.exp(z - z_max)
    return e / e.sum()


# ----------------------------------------------------------------------
# Analytic families (return probability vectors)
# ----------------------------------------------------------------------


def symmetric_product_family(n: int, alpha: float) -> np.ndarray:
    """U‑shaped: a_i ∝ [(i+½)(n‑½‑i)]^(‑α)."""
    i = np.arange(n, dtype=np.float64)
    logprod = np.log(i + 0.5) + np.log(n - 0.5 - i)
    w = np.exp(-alpha * logprod)
    w[np.isinf(w)] = np.finfo(np.float64).max
    return w / w.sum()


def left_heavy_power_family(n: int, beta: float) -> np.ndarray:
    """Monotone decreasing power law: a_i ∝ (i+½)^(‑β)."""
    i = np.arange(n, dtype=np.float64)
    w = np.exp(-beta * np.log(i + 0.5))
    return w / w.sum()


def exponential_decay_family(n: int, gamma: float) -> np.ndarray:
    """Exponential decay: a_i ∝ exp(‑γ·i)."""
    i = np.arange(n, dtype=np.float64)
    w = np.exp(-gamma * i)
    return w / w.sum()


def beta_distribution_family(n: int, alpha: float) -> np.ndarray:
    """Symmetric beta‑shaped: a_i ∝ (i/(n‑1))^α·(1‑i/(n‑1))^α."""
    i = np.arange(n, dtype=np.float64)
    t = i / (n - 1)
    eps = 1e-12
    w = (t + eps) ** alpha * (1 - t + eps) ** alpha
    return w / w.sum()


# ----------------------------------------------------------------------
# Random starts
# ----------------------------------------------------------------------


def random_sparse_start(n: int) -> np.ndarray:
    """Very sparse probability vector (≈5 % non‑zeros)."""
    sparsity = np.random.uniform(0.03, 0.07)
    k = max(2, int(round(sparsity * n)))
    idx = np.random.choice(n, size=k, replace=False)
    a = np.zeros(n, dtype=np.float64)
    a[idx] = 1.0 / k
    return project_to_simplex(a)


def random_monotone_start(n: int) -> np.ndarray:
    """Random decreasing vector (sorted power‑transformed uniform)."""
    r = np.random.rand(n) ** np.random.uniform(0.5, 5.0)
    r.sort()
    a = r[::-1]
    return a / a.sum()


# ----------------------------------------------------------------------
# Smooth surrogate (log‑sum‑exp) utilities
# ----------------------------------------------------------------------


def _smooth_obj_and_grad(z: np.ndarray, beta: float, n: int):
    """
    For a log‑soft‑max parameter vector `z` return
        f(z) = (1/β)·log∑exp(β·(a*a)_k)
    and its gradient ∂f/∂z.
    """
    a = softmax(z)                     # a ∈ simplex
    conv = convolution_fft(a, a)       # length 2n‑1
    maxc = conv.max()
    # log‑sum‑exp in a stable fashion
    exp_terms = np.exp(beta * (conv - maxc))
    sum_exp = exp_terms.sum()
    f = (np.log(sum_exp) + beta * maxc) / beta

    # gradient w.r.t. a
    w = exp_terms / sum_exp            # same shape as conv
    a_rev = a[::-1]
    # convolution of reversed a with w gives ∂f/∂a (length 3n‑2)
    conv_rev = convolution_fft(a_rev, w)
    grad_a = 2.0 * conv_rev[n - 1 : 2 * n - 1]

    # chain rule through soft‑max
    dot_ag = np.dot(a, grad_a)
    grad_z = a * (grad_a - dot_ag)
    return float(f), grad_z.astype(np.float64)


def adam_optimize(z0: np.ndarray, beta: float, steps: int, lr: float = 0.1):
    """Adam optimisation of the smooth surrogate for a fixed β."""
    z = z0.copy()
    m = np.zeros_like(z)
    v = np.zeros_like(z)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    for t in range(1, steps + 1):
        _, g = _smooth_obj_and_grad(z, beta, z.shape[0])
        m = beta1 * m + (1.0 - beta1) * g
        v = beta2 * v + (1.0 - beta2) * (g * g)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        z -= lr * m_hat / (np.sqrt(v_hat) + eps)
        # keep logits bounded (prevent overflow in later softmax)
        if np.max(np.abs(z)) > 50:
            z = np.clip(z, -50, 50)
    return z


# ----------------------------------------------------------------------
# Mirror (exponentiated) sub‑gradient descent on true max‑objective
# ----------------------------------------------------------------------


def mirror_subgrad_optimize(start_a: np.ndarray, deadline: float,
                           step0: float = 0.8,
                           check_interval: int = 2000) -> tuple[np.ndarray, float]:
    """
    Multiplicative sub‑gradient descent on F(a)=2·n·max(a*a)
    with entropy‑like projection onto the simplex.
    """
    n = start_a.shape[0]
    a = start_a.astype(np.float64).copy()
    best_a = a.copy()
    best_score = compute_score(a)

    t = 0
    while time.time() < deadline:
        t += 1
        conv = convolution_fft(a, a)
        k = int(np.argmax(conv))

        # sub‑gradient for the active shift
        idx = np.arange(n)
        j = k - idx
        mask = (j >= 0) & (j < n)
        g = np.zeros_like(a)
        g[mask] = 2.0 * a[j[mask]]

        eta = step0 / math.sqrt(t)
        a = a * np.exp(-eta * g)

        # renormalise onto the simplex
        s = a.sum()
        if s == 0.0:
            a = np.full_like(a, 1.0 / n)
        else:
            a = a / s

        if t % check_interval == 0:
            sc = compute_score(a)
            if sc < best_score:
                best_score = sc
                best_a = a.copy()
                print(f"[{time.time():.1f}] mirror sub‑grad iter {t} best = {best_score:.6f}")

    # final check
    sc = compute_score(a)
    if sc < best_score:
        best_score = sc
        best_a = a.copy()
    return best_a, best_score

def hill_climb_sa(start_a: np.ndarray, deadline: float,
                  max_steps: int = 800_000,
                  init_temp_factor: float = 5e-5) -> tuple[np.ndarray, float]:
    """
    Very low‑temperature SA that swaps or moves a small fraction of mass.
    """
    n = start_a.size
    cur_a = start_a.copy()
    cur_score = compute_score(cur_a)

    best_a = cur_a.copy()
    best_score = cur_score

    T0 = best_score * init_temp_factor
    T = T0
    step_scale = 0.12

    for step in range(1, max_steps + 1):
        if time.time() >= deadline:
            break

        cand = cur_a.copy()
        if np.random.rand() < 0.5:               # swap two entries
            i, j = np.random.randint(n, size=2)
            cand[i], cand[j] = cand[j], cand[i]
        else:                                     # move a fraction of mass
            i, j = np.random.randint(n, size=2)
            if i != j:
                delta = step_scale * cur_a[i]
                cand[i] -= delta
                cand[j] += delta
                if cand[i] < 0.0:
                    cand[i] = 0.0

        cand = cand / cand.sum()
        cand_score = compute_score(cand)
        delta = cand_score - cur_score

        if delta < 0.0 or np.random.rand() < np.exp(-delta / (T + 1e-18)):
            cur_a = cand
            cur_score = cand_score
            if cur_score < best_score:
                best_score = cur_score
                best_a = cur_a.copy()
                step_scale = min(step_scale * 1.07, 0.5)
        else:
            step_scale *= 0.97

        # linear cooling
        T = T0 * (1.0 - step / max_steps)

    return best_a, best_score

def l_bfgs_refine(start_a: np.ndarray, deadline: float,
                  beta_list=(200.0, 800.0, 2000.0, 5000.0),
                  max_iter_per_beta=800) -> tuple[np.ndarray, float]:

    Refine `start_a` with L‑BFGS‑B on the smoothed surrogate.
    Returns the best vector and its *exact* score.
    """
    if not _HAS_SCIPY:
        return start_a, compute_score(start_a)

    best_a = start_a.copy()
    best_score = compute_score(best_a)
    eps = 1e-12
    z = np.log(best_a + eps)

    for beta in beta_list:
        if time.time() >= deadline - 5.0:
            break
        remaining = deadline - time.time()
        max_iter = min(max_iter_per_beta, int(remaining * 12))

        def func(z_vec):
            f, g = _smooth_obj_and_grad(z_vec, beta, best_a.shape[0])
            return f, g

        try:
            opt_res = _opt.fmin_l_bfgs_b(func, z, maxiter=max_iter,
                                         disp=False, factr=1e7, pgtol=1e-12)
        except Exception as exc:  # pragma: no cover
            print(f"[{time.time():.1f}] L‑BFGS failed at β={beta}: {exc}")
            continue

        z = opt_res[0]
        a_opt = softmax(z)
        sc_opt = compute_score(a_opt)
        if sc_opt < best_score:
            best_score = sc_opt
            best_a = a_opt.copy()
            print(f"[{time.time():.1f}] L‑BFGS β={beta:.0f} improved score = {best_score:.6f}")

    return best_a, best_score

            update_best(a_ref)

    # ------------------------------------------------------------------
    # 3. Long mirror sub‑gradient descent on the true max‑objective
    # ------------------------------------------------------------------
    if best_seq is not None and time.time() < deadline - 120.0:
        a_sub, sc_sub = mirror_subgrad_optimize(
            np.array(best_seq, dtype=np.float64),
            deadline - 30.0,
            step0=0.6,
        )
        update_best(a_sub)

    # ------------------------------------------------------------------
    # 4. Symmetrisation – many optimal solutions are symmetric,
    #    but we keep the version that actually improves the score.
    # ------------------------------------------------------------------
    if best_seq is not None:
        a_arr = np.array(best_seq, dtype=np.float64)
        a_sym = (a_arr + a_arr[::-1]) * 0.5
        a_sym = project_to_simplex(a_sym)
        update_best(a_sym)

    # ------------------------------------------------------------------
    # 5. Final tiny simulated‑annealing hill‑climb
    # ------------------------------------------------------------------
    if best_seq is not None and time.time() < deadline - 5.0:
        a_final, sc_final = hill_climb_sa(
            np.array(best_seq, dtype=np.float64),
            deadline,
            max_steps=800_000,
            init_temp_factor=5e-5,
        )
        update_best(a_final)

    # ------------------------------------------------------------------
    # Fallback – should never happen, but keep a simple uniform vector
    # ------------------------------------------------------------------
    if best_seq is None:
        n_small = 256
        best_seq = [1.0 / n_small] * n_small
        best_score = evaluate_sequence(best_seq)

    # ------------------------------------------------------------------
    # Final official evaluation (the outer harness will also call it)
    # ------------------------------------------------------------------
    final_score = evaluate_sequence(best_seq)
    print(f"\n[{time.time():.1f}] finished search – final reported score = {final_score:.6f}")

    return best_seq
