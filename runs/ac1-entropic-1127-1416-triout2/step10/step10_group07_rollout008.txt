"""
Propose a non‑negative step‑function (list of floats) that minimises
``evaluate_sequence``.  The routine uses a quick sweep of analytic families,
random Dirichlet draws, a large‑β smoothed Adam optimisation on the
log‑sum‑exp surrogate of the max‑convolution, a mirror‑subgradient phase,
symmetrisation and a very light simulated‑annealing hill‑climb.
All work respects the supplied wall‑clock budget (≈ 1000 s).
"""

from __future__ import annotations
import time
import numpy as np

# ----------------------------------------------------------------------
# Basic utilities (FFT‑based convolution, simplex projection, etc.)
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """
    Full aperiodic convolution using real FFT.
    Returns length len(a)+len(b)-1 array.
    """
    m = a.size + b.size - 1
    nfft = next_pow2(m)
    A = np.fft.rfft(a, n=nfft)
    B = np.fft.rfft(b, n=nfft)
    C = A * B
    conv = np.fft.irfft(C, n=nfft)[:m]
    return conv


def compute_score(a: np.ndarray) -> float:
    """
    Exact objective used inside the search (sum(a) == 1):
        2 * n * max(a * a)
    """
    n = a.shape[0]
    conv = convolution_fft(a, a)
    return 2.0 * n * float(conv.max())


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the probability simplex
    (v_i ≥ 0, sum v_i = 1).  Works for arbitrary shape (returns a flat vector).
    """
    n = v.shape[0]
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / n)

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.where(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)

    w = np.maximum(v - theta, 0.0)
    return w


def softmax(z: np.ndarray) -> np.ndarray:
    """Numerically stable soft‑max."""
    z_max = np.max(z)
    e = np.exp(z - z_max)
    return e / e.sum()


# ----------------------------------------------------------------------
# Analytic families (hand‑crafted shapes)
# ----------------------------------------------------------------------


def uniform_family(n: int) -> np.ndarray:
    """Plain uniform distribution."""
    return np.full(n, 1.0 / n)


def symmetric_product_family(n: int, alpha: float) -> np.ndarray:
    """a_i ∝ [(i+½)(n‑½‑i)]^(‑α)  (mass concentrated near the ends)."""
    i = np.arange(n, dtype=np.float64)
    logprod = np.log(i + 0.5) + np.log(n - 0.5 - i)
    w = np.exp(-alpha * logprod)
    w[np.isinf(w)] = np.finfo(np.float64).max
    a = w / w.sum()
    return a


def left_heavy_power_family(n: int, beta: float) -> np.ndarray:
    """a_i ∝ (i+½)^(‑β)  (monotonically decreasing)."""
    i = np.arange(n, dtype=np.float64)
    w = np.exp(-beta * np.log(i + 0.5))
    a = w / w.sum()
    return a


def exponential_decay_family(n: int, gamma: float) -> np.ndarray:
    """a_i ∝ exp(‑γ·i)  (exponential tail)."""
    i = np.arange(n, dtype=np.float64)
    w = np.exp(-gamma * i)
    a = w / w.sum()
    return a


def sine_square_family(n: int) -> np.ndarray:
    """a_i ∝ sin²(π·(i+½)/n) – a smooth symmetric bump."""
    i = np.arange(n, dtype=np.float64)
    w = np.sin(np.pi * (i + 0.5) / n) ** 2
    a = w / w.sum()
    return a


def beta_family(n: int, p: float, q: float) -> np.ndarray:
    """
    Generalised Beta‑type family:
    a_i ∝ (i+½)ᵖ·(n‑½‑i)ᵠ,
    with p,q possibly negative.
    """
    i = np.arange(n, dtype=np.float64)
    # work in log‑space for stability
    logw = p * np.log(i + 0.5) + q * np.log(n - 0.5 - i)
    logw -= np.max(logw)       # shift to avoid overflow
    w = np.exp(logw)
    a = w / w.sum()
    return a


def gaussian_family(n: int, sigma: float) -> np.ndarray:
    """a_i ∝ exp(-((i‑c)²)/(2σ²))  with centre c = (n‑1)/2."""
    i = np.arange(n, dtype=np.float64)
    c = (n - 1) / 2.0
    w = np.exp(-((i - c) ** 2) / (2.0 * sigma ** 2))
    a = w / w.sum()
    return a


# ----------------------------------------------------------------------
# Random initialisers
# ----------------------------------------------------------------------


def random_dirichlet_start(n: int, conc: float) -> np.ndarray:
    """Draw from a symmetric Dirichlet(α) – α = conc."""
    vec = np.random.gamma(conc, 1.0, size=n)
    vec = vec / vec.sum()
    return vec


def random_sparse_start(n: int) -> np.ndarray:
    """Very sparse uniform‑mass vector (~5 % non‑zeros)."""
    sparsity = np.random.uniform(0.03, 0.07)
    k = max(2, int(round(sparsity * n)))
    idx = np.random.choice(n, size=k, replace=False)
    a = np.zeros(n, dtype=np.float64)
    a[idx] = 1.0 / k
    return project_to_simplex(a)


# ----------------------------------------------------------------------
# Smoothed optimisation (log‑sum‑exp surrogate) via Adam
# ----------------------------------------------------------------------


def adam_smooth_optimize(
    start_a: np.ndarray,
    deadline: float,
    max_iters: int = 150_000,
    beta_start: float = 5.0,
    beta_end: float = 20_000.0,
    lr: float = 0.015,
) -> tuple[np.ndarray, float]:
    """
    Minimise the smoothed surrogate
        f_β(a) = (1/β)·log Σ_k exp(β·(a*a)_k)
    using Adam on the soft‑max parametrisation a = softmax(z).
    β is increased exponentially from beta_start to beta_end.
    """
    eps = 1e-12
    # initialise logits from the starting simplex vector
    z = np.log(start_a + eps)

    m = np.zeros_like(z)
    v = np.zeros_like(z)

    beta1 = 0.9
    beta2 = 0.999
    eps_adam = 1e-12

    best_a = softmax(z)
    best_score = compute_score(best_a)

    it = 0
    while time.time() < deadline and it < max_iters:
        it += 1

        # ----- β schedule (exponential) -----
        t = it / max_iters
        beta = beta_start * (beta_end / beta_start) ** t

        # ----- current a -----
        a = softmax(z)

        # ----- smoothed weighting over convolution entries -----
        conv = convolution_fft(a, a)                     # length 2n‑1
        maxc = conv.max()
        # stabilised exponentiation
        exp_w = np.exp(beta * (conv - maxc))
        w = exp_w / exp_w.sum()                         # shape (2n‑1,)

        # ----- gradient of f_β w.r.t. a -----
        a_rev = a[::-1]
        conv_rev = convolution_fft(a_rev, w)            # length 3n‑2
        grad_a = 2.0 * conv_rev[a.shape[0] - 1 : a.shape[0] - 1 + a.shape[0]]

        # ----- back‑prop through soft‑max -----
        dot_ag = np.dot(a, grad_a)
        grad_z = a * grad_a - dot_ag * a

        # ----- Adam update -----
        m = beta1 * m + (1.0 - beta1) * grad_z
        v = beta2 * v + (1.0 - beta2) * (grad_z ** 2)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        z -= lr * m_hat / (np.sqrt(v_hat) + eps_adam)

        # ----- occasional exact evaluation -----
        if it % 500 == 0:
            a_tmp = softmax(z)
            sc = compute_score(a_tmp)
            if sc < best_score:
                best_score = sc
                best_a = a_tmp.copy()
                print(f"[{time.time():.1f}]   Adam iter {it:7d} – new best = {best_score:.6f}")

    # final check
    a_fin = softmax(z)
    sc_fin = compute_score(a_fin)
    if sc_fin < best_score:
        best_score = sc_fin
        best_a = a_fin.copy()
    return best_a, best_score

# Mirror (entropy) sub‑gradient descent on the true max‑objective
# ----------------------------------------------------------------------


def mirror_subgrad_optimize(
    start_a: np.ndarray,
    deadline: float,
    step0: float = 0.9,
    check_interval: int = 2500,
) -> tuple[np.ndarray, float]:
    """
    Entropy‑mirror (multiplicative) sub‑gradient descent for 2·n·max(a*a).
    Step size η_t = step0 / sqrt(t).
    """
    n = start_a.shape[0]
    a = start_a.astype(np.float64).copy()
    best_a = a.copy()
    best_score = compute_score(a)

    t = 0
    while time.time() < deadline:
        t += 1
        conv = convolution_fft(a, a)
        k = int(np.argmax(conv))                     # active shift index

        # sub‑gradient for the active shift
        idx = np.arange(n)
        j = k - idx
        mask = (j >= 0) & (j < n)
        g = np.zeros_like(a)
        g[mask] = 2.0 * a[j[mask]]

        eta = step0 / np.sqrt(t)
        a = a * np.exp(-eta * g)                     # multiplicative update
        s = a.sum()
        if s == 0.0:
            a = np.full_like(a, 1.0 / n)
        else:
            a = a / s

        if t % check_interval == 0:
            sc = compute_score(a)
            if sc < best_score:
                best_score = sc
                best_a = a.copy()
                print(f"[{time.time():.1f}]   Mirror step {t:7d} – new best = {best_score:.6f}")

    # final check
    sc = compute_score(a)
    if sc < best_score:
        best_score = sc
        best_a = a.copy()
    return best_a, best_score


# ----------------------------------------------------------------------
# Tiny simulated‑annealing hill‑climb
# ----------------------------------------------------------------------


def hill_climb_sa(
    start_a: np.ndarray,
    deadline: float,
    max_steps: int = 1_200_000,
    init_temp_factor: float = 1e-4,
) -> tuple[np.ndarray, float]:
    """
    Very low‑temperature simulated‑annealing walk on the simplex.
    - 50 % swap two entries
    - otherwise move a fraction of mass from one index to another
    """
    n = start_a.size
    cur_a = start_a.copy()
    cur_score = compute_score(cur_a)

    best_a = cur_a.copy()
    best_score = cur_score

    T0 = best_score * init_temp_factor
    T = T0
    step_scale = 0.12

    for step in range(1, max_steps + 1):
        if time.time() >= deadline:
            break

        cand = cur_a.copy()
        if np.random.rand() < 0.5:               # swap two coordinates
            i, j = np.random.randint(n, size=2)
            cand[i], cand[j] = cand[j], cand[i]
        else:                                    # transfer a small chunk of mass
            i, j = np.random.randint(n, size=2)
            if i != j:
                delta = step_scale * cur_a[i]
                cand[i] -= delta
                cand[j] += delta
                if cand[i] < 0.0:
                    cand[i] = 0.0

        cand = project_to_simplex(cand)
        cand_score = compute_score(cand)
        delta = cand_score - cur_score

        # Metropolis acceptance
        if delta < 0.0 or np.random.rand() < np.exp(-delta / (T + 1e-18)):
            cur_a = cand
            cur_score = cand_score
            if cur_score < best_score:
                best_score = cur_score
                best_a = cur_a.copy()
                step_scale = min(step_scale * 1.07, 0.5)
        else:
            step_scale *= 0.97

        # linear cooling schedule
        T = T0 * (1.0 - step / max_steps)

    return best_a, best_score

def propose_candidate(seed: int = 12345, budget_s: int = 1000, **_):
    """
    Search for a non‑negative step‑function that minimises
    ``evaluate_sequence``.  The routine respects the given wall‑clock budget
    (≈ 1000 s) and returns a plain Python list of floats.
    """
    np.random.seed(seed)
    start_time = time.time()
    # keep a small safety margin for the outer harness
    deadline = start_time + budget_s - 2.0

    # ------------------------------------------------------------------
    # Choose a length: larger n gives a lower theoretical lower bound.
    # ------------------------------------------------------------------
    if budget_s >= 850:
        n = 16384
    elif budget_s >= 600:
        n = 8192
    else:
        n = 4096
    print(f"[{time.time():.1f}] Using length n = {n}")

    # ------------------------------------------------------------------
    # Globals for the best found vector
    # ------------------------------------------------------------------
    best_seq: list[float] | None = None
    best_score = float("inf")
    topk: list[tuple[float, np.ndarray]] = []   # keep a few elite vectors

    def maybe_update(vec: np.ndarray, tag: str = "") -> None:
        """Update global best if the new vector improves the score."""
        nonlocal best_seq, best_score, topk
        sc = compute_score(vec)
        if sc < best_score - 1e-12:
            best_score = sc
            best_seq = vec.tolist()
            print(f"[{time.time():.1f}] new best {tag:<20} = {best_score:.6f}")

        # keep a short sorted list of elite candidates (max size 10)
        if len(topk) < 10:
            topk.append((sc, vec.copy()))
            topk.sort(key=lambda x: x[0])
        else:
            if sc < topk[-1][0]:
                topk[-1] = (sc, vec.copy())
                topk.sort(key=lambda x: x[0])

    # ------------------------------------------------------------------
    # 0. Fallback – previously known construction (if available)
    # ------------------------------------------------------------------
    try:
        fb_vec = [float(x) for x in height_sequence_1]   # type: ignore[name-defined]
        fb_sc = evaluate_sequence(fb_vec)
        print(f"[{time.time():.1f}] fallback score = {fb_sc:.6f}")
        maybe_update(np.array(fb_vec) / np.sum(fb_vec), "fallback")
    except Exception:
        pass

    # ------------------------------------------------------------------
    # 1. Analytic families – cheap exhaustive scan (time‑limited)
    # ------------------------------------------------------------------
    if time.time() < deadline - 30.0:
        maybe_update(uniform_family(n), "uniform")

    # symmetric product family (ends‑heavy)
    if time.time() < deadline - 30.0:
        for alpha in np.linspace(0.0, 12.0, 61):
            if time.time() >= deadline - 30.0:
                break
            maybe_update(symmetric_product_family(n, alpha), f"sym‑α={alpha:.1f}")

    # left‑heavy power law
    if time.time() < deadline - 30.0:
        for beta in np.linspace(0.0, 12.0, 61):
            if time.time() >= deadline - 30.0:
                break
            maybe_update(left_heavy_power_family(n, beta), f"pow‑β={beta:.1f}")

    # exponential decay
    if time.time() < deadline - 30.0:
        for gamma in np.linspace(0.0, 12.0, 61):
            if time.time() >= deadline - 30.0:
                break
            maybe_update(exponential_decay_family(n, gamma), f"exp‑γ={gamma:.1f}")

    # sine‑square bump (symmetric)
    if time.time() < deadline - 30.0:
        maybe_update(sine_square_family(n), "sine‑sq")

    # Gaussian‑type bump – try several bandwidths
    if time.time() < deadline - 30.0:
        for sigma_factor in [0.12, 0.18, 0.25, 0.35]:
            sigma = sigma_factor * n

    # Beta‑type families (p,q grid)
    if time.time() < deadline - 30.0:
        p_vals = np.linspace(-1.5, 4.5, 13)   # includes negative values

    if best_seq is not None and time.time() < deadline - 60.0:
        mirror_deadline = time.time() + (deadline - time.time()) * 0.5
        print(f"[{time.time():.1f}] starting mirror sub‑gradient descent")
        a_sub, sc_sub = mirror_subgrad_optimize(
            np.array(best_seq, dtype=np.float64),
            mirror_deadline,
            step0=0.9,
        )
        if sc_sub < best_score:
            best_score = sc_sub
            best_seq = a_sub.tolist()
            print(f"[{time.time():.1f}] after mirror sub‑grad best = {best_score:.6f}")

    # ------------------------------------------------------------------
    # 6. Symmetrisation (most optimal solutions appear symmetric)
    # ------------------------------------------------------------------
    if best_seq is not None:
        a_arr = np.array(best_seq, dtype=np.float64)
        a_sym = (a_arr + a_arr[::-1]) * 0.5
        a_sym = project_to_simplex(a_sym)
        maybe_update(a_sym, "symmetrised")

    # ------------------------------------------------------------------
    # 7. Final lightweight hill‑climb (simulated annealing)
    # ------------------------------------------------------------------
    if best_seq is not None and time.time() < deadline - 5.0:
        print(f"[{time.time():.1f}] final hill‑climb (SA)")
        a_final, sc_final = hill_climb_sa(
            np.array(best_seq, dtype=np.float64),
            deadline,
            max_steps=1_200_000,
        )
        if sc_final < best_score:
            best_score = sc_final
            best_seq = a_final.tolist()
            print(f"[{time.time():.1f}] hill‑climb improved best = {best_score:.6f}")

    # ------------------------------------------------------------------
    # 8. Safeguard – fall back to a modest uniform vector if everything failed
    # ------------------------------------------------------------------
    if best_seq is None:
        n_small = 256
        best_seq = [1.0 / n_small] * n_small
        best_score = evaluate_sequence(best_seq)

    # ------------------------------------------------------------------
    # 9. Final official evaluation (the harness will also call evaluate_sequence)
    # ------------------------------------------------------------------
    final_score = evaluate_sequence(best_seq)
    print(f"\n[{time.time():.1f}] finished search – final reported score = {final_score:.6f}")

    # Return a plain Python list (required by the harness).
    return best_seq
