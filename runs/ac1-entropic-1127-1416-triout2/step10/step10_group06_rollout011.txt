def smooth_adam_opt(start_a, deadline, beta_schedule=(10,50,200,800,2000,5000), steps_per_beta=2000):
    eps = 1e-12
    z = np.log(start_a + eps)
    m = np.zeros_like(z)
    v = np.zeros_like(z)
    t = 0
    step_size = 0.1
    beta1 = 0.9
    beta2 = 0.999
    best_a = start_a.copy()
    best_score = compute_score(best_a)
    for beta in beta_schedule:
        if time.time() >= deadline - 2: break
        for _ in range(steps_per_beta):
            if time.time() >= deadline - 2: break
            t += 1
            f, g = _smooth_obj_and_grad(z, beta, len(start_a))
            # Adam update
            m = beta1 * m + (1 - beta1) * g
            v = beta2 * v + (1 - beta2) * (g ** 2)
            m_hat = m / (1 - beta1**t)
            v_hat = v / (1 - beta2**t)
            z = z - step_size * m_hat / (np.sqrt(v_hat) + 1e-8)
            if t % 500 == 0:
                a_cur = softmax(z)
                sc = compute_score(a_cur)
                if sc < best_score:
                    best_score = sc
                    best_a = a_cur.copy()
    return best_a, best_score

"""
Propose a step‑function sequence that minimises `evaluate_sequence`.

Algorithm
---------
1. Pick a length `n` (larger ‑‑► more expressive, but slower FFT).
2. Scan a variety of analytic families (U‑shaped, power‑law, exponential,
   beta, sine, quadratic) on a coarse grid and keep the best.
   Add many random starts (Dirichlet, sparse, two‑ended blocks).
3. Refine the current best with a smooth surrogate
   f_β(a)= (1/β)·log∑exp(β·(a*a)_k) using a hand‑rolled Adam optimiser.
   The schedule β={10,50,200,800,2000,5000} drives the surrogate to the
   true max‑convolution.
4. Perform a short mirror‑subgradient descent on the exact objective to
   polish the result.
5. Symmetrise the sequence (most optima are symmetric) and finish with a
   tiny simulated‑annealing hill‑climb.
6. Return the best sequence found (a plain list of floats).

All stages respect a global wall‑clock deadline (default 1000 s) and
use only NumPy (no external solvers).  The code is fully self‑contained
and may be imported safely.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# FFT utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """
    Linear convolution via real FFT.
    Returns an array of length len(a)+len(b)-1.
    """
    m = a.size + b.size - 1
    nfft = next_pow2(m)
    A = np.fft.rfft(a, n=nfft)
    B = np.fft.rfft(b, n=nfft)
    C = A * B
    conv = np.fft.irfft(C, n=nfft)[:m]
    return conv


def compute_score(a: np.ndarray) -> float:
    """
    Exact objective for a probability vector (∑a = 1):
        score = 2·n·max(a * a)
    This matches `evaluate_sequence` up to the normalisation.
    """
    n = a.shape[0]
    conv = convolution_fft(a, a)
    return 2.0 * n * float(conv.max())


# ----------------------------------------------------------------------
# Simplex utilities
# ----------------------------------------------------------------------


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection onto the probability simplex."""
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / v.shape[0])

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, v.shape[0] + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.where(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def softmax(z: np.ndarray) -> np.ndarray:
    """Stable softmax that maps ℝⁿ → simplex."""
    z_max = np.max(z)
    e = np.exp(z - z_max)
    return e / e.sum()


# ----------------------------------------------------------------------
# Analytic families (all return a probability vector)
# ----------------------------------------------------------------------


def symmetric_product_family(n: int, alpha: float) -> np.ndarray:
    """U‑shaped: a_i ∝ [(i+½)(n‑½‑i)]^(‑α)."""
    i = np.arange(n, dtype=np.float64)
    logprod = np.log(i + 0.5) + np.log(n - 0.5 - i)
    w = np.exp(-alpha * logprod)
    w[np.isinf(w)] = np.finfo(np.float64).max
    a = w / w.sum()
    return a


def left_heavy_power_family(n: int, beta: float) -> np.ndarray:
    """Monotone decreasing power law: a_i ∝ (i+½)^(‑β)."""
    i = np.arange(n, dtype=np.float64)
    w = np.exp(-beta * np.log(i + 0.5))
    a = w / w.sum()
    return a


def exponential_decay_family(n: int, gamma: float) -> np.ndarray:
    """Simple exponential decay: a_i ∝ exp(‑γ·i)."""
    i = np.arange(n, dtype=np.float64)
    w = np.exp(-gamma * i)
    a = w / w.sum()
    return a


def beta_distribution_family(n: int, alpha: float) -> np.ndarray:
    """Symmetric beta shape."""
    i = np.arange(n, dtype=np.float64)
    t = i / (n - 1)
    eps = 1e-12
    w = (t + eps) ** alpha * (1 - t + eps) ** alpha
    a = w / w.sum()
    return a


def sine_family(n: int, power: float) -> np.ndarray:
    """Positive sine shape: a_i ∝ sin(π·(i+½)/n) ** power."""
    i = np.arange(n, dtype=np.float64) + 0.5
    w = np.sin(np.pi * i / n) ** power
    a = w / w.sum()
    return a


def quadratic_power_family(n: int, exponent: float) -> np.ndarray:
    """Central‑peak (exponent>0) or U‑shape (exponent<0):
       a_i ∝ [(i+½)(n‑½‑i)] ** exponent."""
    i = np.arange(n, dtype=np.float64)
    prod = (i + 0.5) * (n - 0.5 - i)
    # protect against negative exponents of zero
    w = np.where(prod > 0, prod ** exponent, 0.0)
    a = w / w.sum() if w.sum() > 0 else np.full_like(w, 1.0 / n)
    return a


# ----------------------------------------------------------------------
# Random initialisers
# ----------------------------------------------------------------------


def random_sparse_start(n: int) -> np.ndarray:
    """Sparse probability vector (≈5 % non‑zeros)."""
    sparsity = np.random.uniform(0.03, 0.07)
    k = max(2, int(round(sparsity * n)))
    idx = np.random.choice(n, size=k, replace=False)
    a = np.zeros(n, dtype=np.float64)
    a[idx] = 1.0 / k
    return project_to_simplex(a)


def random_monotone_start(n: int) -> np.ndarray:
    """Random decreasing vector (sorted power‑transformed uniform)."""
    r = np.random.rand(n) ** np.random.uniform(0.5, 5.0)
    r.sort()
    a = r[::-1]
    a = a / a.sum()
    return a


def random_two_end_block_start(n: int) -> np.ndarray:
    """Mass concentrated in two symmetric edge blocks."""
    block_frac = np.random.uniform(0.01, 0.05)
    k = max(1, int(block_frac * n))
    a = np.zeros(n, dtype=np.float64)
    a[:k] = 0.5 / k          # left block gets half the mass
    a[-k:] = 0.5 / k         # right block gets the other half
    return project_to_simplex(a)


# ----------------------------------------------------------------------
# Smooth‑surrogate optimisation (Adam)
# ----------------------------------------------------------------------


def _smooth_obj_and_grad(z: np.ndarray, beta: float, n: int):
    """
    Return (f, grad) where
        f   = (1/β)·log∑_k exp[β·(a*a)_k]
        grad = ∂f/∂z   (z are the log‑softmax parameters)
    """
    a = softmax(z)                     # a ∈ simplex
    conv = convolution_fft(a, a)       # length 2n‑1
    cmax = conv.max()
    # log‑sum‑exp in a stable way
    exp_terms = np.exp(beta * (conv - cmax))
    sum_exp = exp_terms.sum()
    f = (np.log(sum_exp) + beta * cmax) / beta

    # gradient wrt a
    w = exp_terms / sum_exp            # shape (2n‑1,)
    a_rev = a[::-1]
    # convolution of the reversed a with the weight vector w
    conv_rev = convolution_fft(a_rev, w)   # length 3n‑2
    grad_a = 2.0 * conv_rev[n - 1 : 2 * n - 1]

    # chain rule through softmax (standard soft‑max gradient)
    dot_ag = np.dot(a, grad_a)
    grad_z = a * (grad_a - dot_ag)
    return float(f), grad_z.astype(np.float64)


def smooth_adam_opt(start_a: np.ndarray,
                    deadline: float,
                    beta_schedule=(10.0, 50.0, 200.0, 800.0, 2000.0, 5000.0),
                    steps_per_beta: int = 2000) -> tuple[np.ndarray, float]:
    """
    Adam optimiser on the log‑sum‑exp surrogate.
    Returns the best vector found and its exact score.
    """
    eps = 1e-12
    z = np.log(start_a + eps)          # logits
    m = np.zeros_like(z)
    v = np.zeros_like(z)
    t = 0
    beta1, beta2 = 0.9, 0.999
    best_a = start_a.copy()
    best_score = compute_score(best_a)

    for beta in beta_schedule:
        if time.time() >= deadline - 5.0:
            break
        # modest step size; we shrink it a little for larger β
        base_lr = 0.15 / (beta / 200.0 + 1.0)
        for _ in range(steps_per_beta):
            if time.time() >= deadline - 5.0:
                break
            t += 1
            f, g = _smooth_obj_and_grad(z, beta, start_a.shape[0])
            # Adam update on logits
            m = beta1 * m + (1 - beta1) * g
            v = beta2 * v + (1 - beta2) * (g ** 2)
            m_hat = m / (1 - beta1 ** t)
            v_hat = v / (1 - beta2 ** t)
            z = z - base_lr * m_hat / (np.sqrt(v_hat) + 1e-8)

            # periodic exact evaluation
            if t % 500 == 0:
                a_cur = softmax(z)
                sc = compute_score(a_cur)
                if sc < best_score:
                    best_score = sc
                    best_a = a_cur.copy()
                    print(f"[{time.time():.1f}] Adam β={beta:.0f} iter {t} best={best_score:.6f}")

    return best_a, best_score


# ----------------------------------------------------------------------
# Mirror (entropy) sub‑gradient descent on the true max‑objective
# ----------------------------------------------------------------------


def mirror_subgrad_optimize(start_a: np.ndarray,
                            deadline: float,
                            step0: float = 0.8,
                            check_interval: int = 5000) -> tuple[np.ndarray, float]:
    """
    Multiplicative (exponentiated) sub‑gradient descent on
        F(a) = 2·n·max(a*a)
    The iterate stays in the simplex thanks to a renormalisation step.
    """
    n = start_a.shape[0]
    a = start_a.astype(np.float64).copy()
    best_a = a.copy()
    best_score = compute_score(a)

    t = 0
    while time.time() < deadline:
        t += 1
        conv = convolution_fft(a, a)
        k = int(np.argmax(conv))

        # sub‑gradient of the active convolution entry
        idx = np.arange(n)
        j = k - idx
        mask = (j >= 0) & (j < n)
        g = np.zeros_like(a)
        g[mask] = 2.0 * a[j[mask]]

        eta = step0 / math.sqrt(t)
        a = a * np.exp(-eta * g)        # multiplicative update
        s = a.sum()
        a = a / s if s > 0 else np.full_like(a, 1.0 / n)

        if t % check_interval == 0:
            sc = compute_score(a)
            if sc < best_score:
                best_score = sc
                best_a = a.copy()
                print(f"[{time.time():.1f}] mirror sub‑grad iter {t} best={best_score:.6f}")

    # final check
    sc = compute_score(a)
    if sc < best_score:
        best_score = sc
        best_a = a.copy()
    return best_a, best_score


# ----------------------------------------------------------------------
# Tiny simulated‑annealing hill‑climb (final polishing)
# ----------------------------------------------------------------------


def hill_climb_sa(start_a: np.ndarray,
                  deadline: float,
                  max_steps: int = 1_200_000,
                  init_temp_factor: float = 1e-4) -> tuple[np.ndarray, float]:
    """
    Simple SA on the simplex:
      - with prob 0.5 swap two entries,
      - otherwise shift a small fraction of mass.
    """
    n = start_a.size
    cur_a = start_a.copy()
    cur_score = compute_score(cur_a)

    best_a = cur_a.copy()
    best_score = cur_score

    T0 = best_score * init_temp_factor
    T = T0
    step_scale = 0.12

    for step in range(1, max_steps + 1):
        if time.time() >= deadline:
            break

        cand = cur_a.copy()
        if np.random.rand() < 0.5:  # swap
            i, j = np.random.randint(n, size=2)
            cand[i], cand[j] = cand[j], cand[i]
        else:                       # mass transfer
            i, j = np.random.randint(n, size=2)
            if i != j:
                delta = step_scale * cur_a[i]
                cand[i] -= delta
                cand[j] += delta
                if cand[i] < 0.0:
                    cand[i] = 0.0

        cand = cand / cand.sum()
        cand_score = compute_score(cand)
        delta = cand_score - cur_score

        if delta < 0.0 or np.random.rand() < np.exp(-delta / (T + 1e-18)):
            cur_a = cand
            cur_score = cand_score
            if cur_score < best_score:
                best_score = cur_score
                best_a = cur_a.copy()
                step_scale = min(step_scale * 1.07, 0.5)
        else:
            step_scale *= 0.97

        T = T0 * (1.0 - step / max_steps)

    return best_a, best_score

                                                step0=0.8,
                                                check_interval=5000)
        if sc_sub < best_score:
            best_score = sc_sub
            best_vec = a_sub.copy()
            print(f"[{time.time():.1f}] after mirror sub‑grad best = {best_score:.6f}")

    # ------------------------------------------------------------------
    # 6. Symmetrisation (most optima are symmetric)
    # ------------------------------------------------------------------
    if best_vec is not None:
        a_sym = (best_vec + best_vec[::-1]) * 0.5
        a_sym = project_to_simplex(a_sym)
        sc_sym = compute_score(a_sym)
        if sc_sym < best_score:
            best_score = sc_sym
            best_vec = a_sym.copy()
            print(f"[{time.time():.1f}] symmetrisation improved = {best_score:.6f}")

    # ------------------------------------------------------------------
    # 7. Tiny simulated‑annealing hill‑climb for final polishing
    # ------------------------------------------------------------------
    if best_vec is not None and time.time() < deadline - 5.0:
        print(f"[{time.time():.1f}] Final hill‑climb (SA)")
        a_fin, sc_fin = hill_climb_sa(best_vec, deadline)
        if sc_fin < best_score:
            best_score = sc_fin
            best_vec = a_fin.copy()
            print(f"[{time.time():.1f}] hill‑climb improved = {best_score:.6f}")

    # ------------------------------------------------------------------
    # Fallback in the unlikely event that nothing was produced
    # ------------------------------------------------------------------
    if best_vec is None:
        n_small = 256
        best_vec = np.full(n_small, 1.0 / n_small, dtype=np.float64)
        best_score = compute_score(best_vec)

    # ------------------------------------------------------------------
    # Final official evaluation (the harness will also call it)
    # ------------------------------------------------------------------
    final_score = evaluate_sequence(best_vec.tolist())
    print(f"\n[{time.time():.1f}] finished search – final reported score = {final_score:.6f}")

    return best_vec.tolist()
