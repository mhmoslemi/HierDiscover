"""
Propose a step‑function (non‑negative sequence) that minimises

        2·n·max(convolve(a,a)) / (sum(a)**2)

under the simplex constraint.

Algorithm
---------
0.  If a fallback sequence is supplied via the global variable
    ``height_sequence_1`` use it as an initial reference.

1.  Quick scan of a rich family of analytic shapes (beta‑type,
    Gaussian, exponential‑U, sinusoidal, random Dirichlet) to obtain a
    decent starter.

2.  Perform several independent *restarts*.  For each restart the
    starter is slightly perturbed and the following pipeline is run
    (all steps respect the current time budget):

       • Adam optimisation of a smooth‑max surrogate
         (β is annealed from 4 to 800).

       • Projected sub‑gradient descent on the true max‑objective.
         The gradient has been corrected (the diagonal term receives
         factor 2, not 1).

       • A handful of LP refinement iterations (the classic “f * g ≤ M”
         step) with an adaptive mixing weight.

       • Symmetrisation (averaging with the reversed vector) and a
         final stochastic hill‑climb (simulated‑annealing style).

3.  Keep the best sequence seen over all restarts and return it as a
    plain Python list.  All heavy work uses FFT‑based convolutions which
    are O(n log n), allowing sequences up to a few thousand entries to be
    handled comfortably in the 1000 s budget.

The code is self‑contained, uses only NumPy / SciPy and respects the
required function signature.

def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full aperiodic convolution via real FFT (fast for large n)."""
    m = a.size + b.size - 1
    nfft = next_pow2(m)
    A = np.fft.rfft(a, n=nfft)
    B = np.fft.rfft(b, n=nfft)
    C = A * B
    conv = np.fft.irfft(C, n=nfft)[:m]
    return conv


def compute_score(a: np.ndarray) -> float:
    """
    Exact objective for a probability vector (∑a = 1):
        2·n·max(a*a)
    """
    n = a.shape[0]
    conv = convolution_fft(a, a)
    return 2.0 * n * float(conv.max())


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection onto the probability simplex."""
    if np.all(v <= 0):
        # all non‑positive → uniform distribution
        return np.full_like(v, 1.0 / v.shape[0])

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, v.shape[0] + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.where(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def softmax(z: np.ndarray) -> np.ndarray:
    """Numerically stable soft‑max."""
    z_max = np.max(z)
    e = np.exp(z - z_max)
    return e / e.sum()

def beta_distribution_family(n: int, alpha: float, beta: float) -> np.ndarray:
    """
    Discrete Beta‑type shape:
        w_i ∝ (i+1)^α * (n-i)^β   for i = 0,…,n‑1
    """
    i = np.arange(n, dtype=np.float64)
    w = (i + 1) ** alpha * (n - i) ** beta
    a = w / w.sum()
    return a


def gaussian_family(n: int, sigma: float) -> np.ndarray:
    """Symmetric Gaussian centred in the middle of the interval."""
    i = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    w = np.exp(-0.5 * ((i - centre) / sigma) ** 2)
    a = w / w.sum()
    return a


def sin_window_family(n: int, power: float) -> np.ndarray:
    """Shape a_i ∝ sin(π·(i+1)/(n+1))^power."""
    i = np.arange(1, n + 1, dtype=np.float64)
    w = np.sin(np.pi * i / (n + 1)) ** power
    a = w / w.sum()
    return a


def exp_u_family(n: int, gamma: float) -> np.ndarray:
    """Classic “U‑shape”: exponential decay from the centre."""
    i = np.arange(n, dtype=np.float64)
    dist = np.minimum(i, n - 1 - i)
    w = np.exp(-gamma * dist)
    a = w / w.sum()
    return a


def random_dirichlet_start(n: int, conc: float = 0.5) -> np.ndarray:
    """Draw a random point from a Dirichlet(α) with α_i = conc."""
    alpha = np.full(n, conc, dtype=np.float64)
    a = np.random.gamma(alpha, 1.0)
    a = a / a.sum()
    return a

# Adam optimisation of the smooth‑max surrogate

def adam_smooth_optimize(start_a: np.ndarray,
                         deadline: float,
                         max_iters: int = 250_000,
                         beta_start: float = 4.0,
                         beta_end: float = 800.0,
                         lr: float = 0.018) -> tuple[np.ndarray, float]:
    """
    Minimise the smooth surrogate
          f_β(a) = (1/β)·log Σ_k exp(β·(a*a)_k)
    using Adam on the soft‑max parametrisation a = softmax(z).
    The schedule for β is exponential.
    """
    eps = 1e-12
    # initialise logits from the current a (avoid log(0))
    z = np.log(start_a + eps)

    m = np.zeros_like(z)
    v = np.zeros_like(z)
    beta1 = 0.9
    beta2 = 0.999
    eps_adam = 1e-12

    best_a = softmax(z)
    best_score = compute_score(best_a)

    for it in range(1, max_iters + 1):
        if time.time() >= deadline:
            break

        # ----- anneal the smoothing parameter β -----
        t = it / max_iters
        beta = beta_start * (beta_end / beta_start) ** t

        # ----- forward pass -----
        a = softmax(z)
        conv = convolution_fft(a, a)
        maxc = conv.max()
        # soft‑max weights (stable formulation)
        exp_w = np.exp(beta * (conv - maxc))
        w = exp_w / exp_w.sum()

        # ----- gradient wrt a -----
        # The derivation yields grad_a = 2 * (a_rev * w) restricted to length n
        a_rev = a[::-1]
        conv_rev = convolution_fft(a_rev, w)
        # extract the relevant slice (length n)
        grad_a = 2.0 * conv_rev[len(a) - 1:len(a) - 1 + len(a)]

        # ----- back‑prop through the soft‑max -----
        dot_ag = np.dot(a, grad_a)
        grad_z = a * grad_a - dot_ag * a

        # ----- Adam update -----
        m = beta1 * m + (1.0 - beta1) * grad_z
        v = beta2 * v + (1.0 - beta2) * (grad_z ** 2)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        z -= lr * m_hat / (np.sqrt(v_hat) + eps_adam)

        # ----- bookkeeping -----
        if it % 1000 == 0 or it == max_iters:
            a_tmp = softmax(z)
            sc = compute_score(a_tmp)
            if sc < best_score:
                best_score = sc
                best_a = a_tmp.copy()
                print(f"[{time.time():.1f}] Adam improved to {best_score:.6f}")

    # final evaluation
    a_fin = softmax(z)
    sc_fin = compute_score(a_fin)
    if sc_fin < best_score:
        best_score = sc_fin
        best_a = a_fin.copy()
    return best_a, best_score

def subgrad_optimize(start_a: np.ndarray,
                     deadline: float,
                     max_iters: int = 1_500_000,
                     step0: float = 0.07) -> tuple[np.ndarray, float]:
    """
    Sub‑gradient method for the true objective.
    The gradient is the exact sub‑gradient of max_k (a*a)_k.
    """
    n = start_a.shape[0]
    a = start_a.astype(np.float64).copy()

    best_a = a.copy()
    best_score = compute_score(a)

    for t in range(1, max_iters + 1):
        if time.time() >= deadline:
            break

        conv = convolution_fft(a, a)
        k = int(np.argmax(conv))

        # ----- compute correct sub‑gradient for the active shift k -----
        idx = np.arange(n)
        j = k - idx
        # j is valid where 0 <= j < n
        mask = (j >= 0) & (j < n)

        g = a.copy()                     # the a_i term
        g[mask] += a[j[mask]]            # add a_{k-i} where defined

        # step size 1/√t schedule
        step = step0 / math.sqrt(t)
        a = project_to_simplex(a - step * g)

        if t % 2000 == 0:
            sc = compute_score(a)
            if sc < best_score:
                best_score = sc
                best_a = a.copy()
                print(f"[{time.time():.1f}] sub‑grad improved to {best_score:.6f}")

    # final check
    sc = compute_score(a)
    if sc < best_score:
        best_score = sc
        best_a = a.copy()
    return best_a, best_score

def lp_refine_step(f: np.ndarray, M: float, mix_weight: float = 0.2) -> np.ndarray:
    """
    Solve   max Σ g_j   s.t.   (f * g)_k ≤ M   (for all k),
    g ≥ 0, then return the convex mixture
        f_new = (1−t)·f + t·g_norm
    where g_norm = g / Σ g.
    """
    n = f.shape[0]
    m = 2 * n - 1

    # Build convolution matrix A such that A @ g = f * g (full aperiodic conv)
    A = np.zeros((m, n), dtype=np.float64)
    for i in range(n):
        A[i:i + n, i] = f[i]

    c = -np.ones(n, dtype=np.float64)               # maximise Σ g ⇔ minimise −Σ g
    bounds = [(0.0, None) for _ in range(n)]
    b = np.full(m, M, dtype=np.float64)

    # Solve – apply a modest time limit to keep the overall budget safe
    res = linprog(c, A_ub=A, b_ub=b, bounds=bounds, method='highs',
                  options={'presolve': True, 'time_limit': 30})

    if not res.success:
        return f

    g = np.maximum(res.x, 0.0)
    sum_g = g.sum()
    if sum_g <= 0.0:
        return f

    g_norm = g / sum_g
    f_new = (1.0 - mix_weight) * f + mix_weight * g_norm
    f_new = project_to_simplex(f_new)
    return f_new

def hill_climb_sa(start_a: np.ndarray,
                  deadline: float,
                  max_steps: int = 1_500_000) -> tuple[np.ndarray, float]:
    """
    Random walk on the simplex with occasional uphill moves.
    Very cheap per step; useful for fine‑tuning the final solution.
    """
    n = start_a.size
    cur_a = start_a.copy()
    cur_score = compute_score(cur_a)

    best_a = cur_a.copy()
    best_score = cur_score

    T0 = best_score * 1e-4   # initial temperature proportional to current score
    T = T0
    step_scale = 0.12

    for step in range(1, max_steps + 1):
        if time.time() >= deadline:
            break

        cand = cur_a.copy()
        if np.random.rand() < 0.5:               # swap two entries
            i, j = np.random.randint(n, size=2)
            cand[i], cand[j] = cand[j], cand[i]
        else:                                     # transfer a small amount of mass
            i, j = np.random.randint(n, size=2)
            if i != j:
                delta = step_scale * cur_a[i]
                cand[i] -= delta
                cand[j] += delta
                if cand[i] < 0.0:
                    cand[i] = 0.0

        cand = project_to_simplex(cand)
        cand_score = compute_score(cand)

        delta = cand_score - cur_score
        if delta < 0 or np.random.rand() < np.exp(-delta / (T + 1e-18)):
            cur_a = cand
            cur_score = cand_score
            if cur_score < best_score:
                best_score = cur_score
                best_a = cur_a.copy()
                step_scale = min(step_scale * 1.07, 0.5)
        else:
            step_scale *= 0.97

        # linear cooling schedule
        T = T0 * (1.0 - step / max_steps)

    return best_a, best_score

                max_iters=2_000_000,
                step0=0.07,
            )
            if sc_opt < best_score:
                best_score = sc_opt
                a = a_opt
                print(f"[{time.time():.1f}] After sub‑grad restart {r+1} best = {best_score:.6f}")

        # ---- 4.3 LP refinement loop ----
        if time.time() < restart_deadline - 30.0:
            mix_weight = 0.2
            lp_iter = 0
            while time.time() < restart_deadline - 10.0 and lp_iter < 12:
                conv = convolution_fft(a, a)
                M = float(conv.max())
                a_new = lp_refine_step(a, M, mix_weight=mix_weight)
                sc = compute_score(a_new)
                if sc < best_score - 1e-8:
                    best_score = sc
                    a = a_new
                    print(f"[{time.time():.1f}] LP step {lp_iter} improved to {best_score:.6f}")
                else:
                    # gently reduce the mixing weight if we hit a plateau
                    mix_weight = max(0.05, mix_weight * 0.85)
                lp_iter += 1

        # ---- 4.4 Symmetrisation (optional) ----
        if time.time() < restart_deadline - 5.0:
            a_sym = (a + a[::-1]) * 0.5
            a_sym = project_to_simplex(a_sym)
            sc_sym = compute_score(a_sym)
            if sc_sym < best_score:
                best_score = sc_sym
                a = a_sym
                print(f"[{time.time():.1f}] Symmetrisation after restart {r+1} improved best = {best_score:.6f}")

        # ---- 4.5 Final hill‑climb (SA) ----
        if time.time() < restart_deadline:
            a_opt, sc_opt = hill_climb_sa(a, deadline=restart_deadline, max_steps=1_200_000)
            if sc_opt < best_score:
                best_score = sc_opt
                a = a_opt
                print(f"[{time.time():.1f}] Hill‑climb after restart {r+1} improved best = {best_score:.6f}")

    # ------------------------------------------------------------------
    # 5. Final symmetrisation test (once more, just in case)
    # ------------------------------------------------------------------
    a_sym = (a + a[::-1]) * 0.5
    a_sym = project_to_simplex(a_sym)
    sc_sym = compute_score(a_sym)
    if sc_sym < best_score:
        best_score = sc_sym
        a = a_sym
        print(f"[{time.time():.1f}] Final symmetrisation improved best = {best_score:.6f}")

    # ------------------------------------------------------------------
    # 6. Final official evaluation and return
    # ------------------------------------------------------------------
    final_seq = a.tolist()
    final_score = evaluate_sequence(final_seq)
    print(f"\n[{time.time():.1f}] finished search – final reported score = {final_score:.6f}")
    return final_seq
