"""
Propose a step‑function sequence that minimises ``evaluate_sequence``.

Algorithm – mixture optimisation
--------------------------------
1. Choose a moderate length `n` (2048 – 4096) – larger than the historic
   2048 item construction but still cheap enough for many FFT iterations.
2. Build a *basis* of candidate probability vectors:
   * uniform,
   * analytic families (symmetric product, left‑heavy power, exponential)
     for a dense grid of parameters,
   * random Dirichlet draws with many concentration levels,
   * sparse random vectors (≈ 5 % support).
   All vectors are normalised to sum = 1.
3. Optimise a convex combination `a = Σ λ_i·basis_i` where λ lives on the
   simplex.  The optimisation is performed with Adam on a *smoothed*
   surrogate of the max‑convolution:
        f_β(a) = (1/β)·log Σ_k exp(β·(a*a)_k)
   The smoothing parameter β is increased exponentially from a small
   value (≈ 5) to a very large one (≈ 20000) so that the surrogate
   becomes practically the true objective.
4. The gradient of `f_β` w.r.t. λ is obtained analytically:
   * compute `a`,
   * compute convolution `c = a*a`,
   * obtain the soft‑max weights `w_k ∝ exp(β·c_k)`,
   * back‑propagate through the convolution to get `∂f/∂a`,
   * multiply by the basis matrix to get `∂f/∂λ`,
   * finally back‑propagate through the soft‑max parametrisation of λ.
5. Adam maintains running moments (m, v) and a learning‑rate of 0.018.
   Every few thousand iterations the *exact* score `2·n·max(a*a)` is
   evaluated; the best vector is remembered.
6. After the smooth phase a short high‑β refinement is run, then a
   standard entropy‑mirror sub‑gradient descent on the *true* max‑objective
   (the same update used in the reference code) for a final polish.
7. Optional symmetrisation (averaging with the reversed vector) and a
   tiny simulated‑annealing hill‑climb are applied.
8. The best vector found before the deadline is returned as a plain
   Python list of floats.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# FFT‑based convolution utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full aperiodic convolution using real FFT (no Python loops)."""
    m = a.size + b.size - 1
    nfft = next_pow2(m)
    A = np.fft.rfft(a, n=nfft)
    B = np.fft.rfft(b, n=nfft)
    C = A * B
    conv = np.fft.irfft(C, n=nfft)[:m]
    return conv


def compute_score(a: np.ndarray) -> float:
    """
    Exact objective for a probability vector (∑a = 1):
        2·n·max(a * a)
    """
    n = a.shape[0]
    conv = convolution_fft(a, a)
    return 2.0 * n * float(conv.max())


def softmax(z: np.ndarray) -> np.ndarray:
    """Numerically stable soft‑max."""
    z_max = np.max(z)
    e = np.exp(z - z_max)
    return e / e.sum()


# ----------------------------------------------------------------------
# Analytic families (all vectors are already normalised)
# ----------------------------------------------------------------------


def symmetric_product_family(n: int, alpha: float) -> np.ndarray:
    """a_i ∝ [(i+½)(n‑½‑i)]^(‑α) , symmetric, sum = 1."""
    i = np.arange(n, dtype=np.float64)
    logprod = np.log(i + 0.5) + np.log(n - 0.5 - i)
    w = np.exp(-alpha * logprod)
    # protect against overflow at the ends
    w[np.isinf(w)] = np.finfo(np.float64).max
    a = w / w.sum()
    return a


def left_heavy_power_family(n: int, beta: float) -> np.ndarray:
    """a_i ∝ (i+½)^(‑β) , monotone decreasing, sum = 1."""
    i = np.arange(n, dtype=np.float64)
    w = np.exp(-beta * np.log(i + 0.5))
    a = w / w.sum()
    return a


def exponential_decay_family(n: int, gamma: float) -> np.ndarray:
    """a_i ∝ exp(-γ·i) , monotone decreasing, sum = 1."""
    i = np.arange(n, dtype=np.float64)
    w = np.exp(-gamma * i)
    a = w / w.sum()
    return a


# ----------------------------------------------------------------------
# Random initialisers (also already normalised)
# ----------------------------------------------------------------------


def random_dirichlet_start(n: int, conc: float) -> np.ndarray:
    """Draw a symmetric Dirichlet(α) vector and normalise to sum = 1."""
    vec = np.random.gamma(conc, 1.0, size=n)
    if vec.sum() == 0.0:
        vec = np.ones_like(vec)
    return vec / vec.sum()


def random_sparse_start(n: int) -> np.ndarray:
    """Sparse vector with ≈ 5 % non‑zeros, then projected onto the simplex."""
    sparsity = np.random.uniform(0.03, 0.07)  # proportion of non‑zeros
    k = max(2, int(round(sparsity * n)))
    idx = np.random.choice(n, size=k, replace=False)
    a = np.zeros(n, dtype=np.float64)
    a[idx] = 1.0 / k
    return a / a.sum()


# ----------------------------------------------------------------------
# Adam optimiser for the *smoothed* max‑convolution (mixture λ)
# ----------------------------------------------------------------------


def adam_mix_optimize(
    bases: np.ndarray,
    deadline: float,
    max_iters: int = 200_000,
    beta_start: float = 5.0,
    beta_end: float = 20_000.0,
    lr: float = 0.018,
) -> tuple[np.ndarray, np.ndarray, float]:
    """
    Optimise a convex combination of `bases` (shape K×n) using Adam on
    the log‑sum‑exp surrogate of the max‑convolution.
    Returns (best_lambda, best_a, best_score).
    """
    eps = 1e-12
    K, n = bases.shape

    # initialise λ uniformly (soft‑max parametrisation)
    z = np.log(np.full(K, 1.0 / K) + eps)
    m = np.zeros_like(z)
    v = np.zeros_like(z)

    beta1 = 0.9
    beta2 = 0.999
    eps_adam = 1e-12

    best_lambda = None
    best_a = None
    best_score = float("inf")

    for it in range(1, max_iters + 1):
        if time.time() >= deadline:
            break

        # ---- β schedule (exponential) ----
        t = it / max_iters
        beta = beta_start * (beta_end / beta_start) ** t

        # ----- current mixture -----
        lam = softmax(z)                     # shape (K,)
        a = lam @ bases                       # shape (n,)

        # ----- smoothed weights w_k ∝ exp(β·conv_k) -----
        conv = convolution_fft(a, a)          # length 2n‑1
        maxc = conv.max()
        exp_w = np.exp(beta * (conv - maxc))  # stabilised
        w = exp_w / exp_w.sum()               # soft‑max over shifts

        # ----- gradient of f_β wrt a -----
        a_rev = a[::-1]
        conv_rev = convolution_fft(a_rev, w)  # length 3n‑2
        grad_a = 2.0 * conv_rev[n - 1 : n - 1 + n]

        # ----- gradient wrt λ (chain rule) -----
        grad_lam = bases @ grad_a             # shape (K,)

        # ----- back‑prop through λ‑softmax -----
        dot = np.dot(lam, grad_lam)
        grad_z = lam * grad_lam - dot * lam   # shape (K,)

        # ----- Adam update -----
        m = beta1 * m + (1.0 - beta1) * grad_z
        v = beta2 * v + (1.0 - beta2) * (grad_z ** 2)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        z -= lr * m_hat / (np.sqrt(v_hat) + eps_adam)

        # ----- occasional exact evaluation -----
        if it % 1000 == 0:
            lam_cur = softmax(z)
            a_cur = lam_cur @ bases
            sc = compute_score(a_cur)
            if sc < best_score:
                best_score = sc
                best_lambda = lam_cur.copy()
                best_a = a_cur.copy()
                print(f"[{time.time():.1f}]   AdamMix iter {it:7d} – new best = {best_score:.6f}")

    # final check
    lam_final = softmax(z)
    a_final = lam_final @ bases
    sc_final = compute_score(a_final)
    if sc_final < best_score:
        best_score = sc_final
        best_lambda = lam_final.copy()
        best_a = a_final.copy()

    return best_lambda, best_a, best_score


# ----------------------------------------------------------------------
# Entropy‑mirror sub‑gradient descent for the *exact* max objective
# ----------------------------------------------------------------------


def mirror_subgrad_optimize(
    start_a: np.ndarray,
    deadline: float,
    step0: float = 0.8,
    check_interval: int = 2000,
) -> tuple[np.ndarray, float]:
    """
    Multiplicative mirror sub‑gradient descent on the exact objective:
        2·n·max(a*a)
    The step size is η_t = step0 / sqrt(t).
    Returns the best vector encountered and its score.
    """
    n = start_a.shape[0]
    a = start_a.astype(np.float64).copy()
    best_a = a.copy()
    best_score = compute_score(a)

    t = 0
    while time.time() < deadline:
        t += 1
        conv = convolution_fft(a, a)
        k = int(np.argmax(conv))

        # Sub‑gradient for the active shift
        idx = np.arange(n)
        j = k - idx
        mask = (j >= 0) & (j < n)
        g = np.zeros_like(a)
        g[mask] = 2.0 * a[j[mask]]

        eta = step0 / np.sqrt(t)
        a = a * np.exp(-eta * g)               # multiplicative update
        s = a.sum()
        if s == 0.0:
            a = np.full_like(a, 1.0 / n)
        else:
            a = a / s

        if t % check_interval == 0:
            sc = compute_score(a)
            if sc < best_score:
                best_score = sc
                best_a = a.copy()
                print(f"[{time.time():.1f}]   Mirror step {t:7d} – new best = {best_score:.6f}")

    # final check
    sc = compute_score(a)
    if sc < best_score:
        best_score = sc
        best_a = a.copy()
    return best_a, best_score


# ----------------------------------------------------------------------
# Tiny simulated‑annealing hill‑climb (final polish)
# ----------------------------------------------------------------------


def hill_climb_sa(
    start_a: np.ndarray,
    deadline: float,
    max_steps: int = 1_200_000,
    init_temp_factor: float = 1e-4,
) -> tuple[np.ndarray, float]:
    """
    Low‑temperature simulated‑annealing walk on the simplex.
    """
    n = start_a.size
    cur_a = start_a.copy()
    cur_score = compute_score(cur_a)

    best_a = cur_a.copy()
    best_score = cur_score

    T0 = best_score * init_temp_factor
    T = T0
    step_scale = 0.12

    for step in range(1, max_steps + 1):
        if time.time() >= deadline:
            break

        cand = cur_a.copy()
        if np.random.rand() < 0.5:                       # swap two entries
            i, j = np.random.randint(n, size=2)
            cand[i], cand[j] = cand[j], cand[i]
        else:                                            # transfer a fraction
            i, j = np.random.randint(n, size=2)
            if i != j:
                delta = step_scale * cur_a[i]
                cand[i] -= delta
                cand[j] += delta
                if cand[i] < 0.0:
                    cand[i] = 0.0

        cand = cand / cand.sum()
        cand_score = compute_score(cand)
        delta = cand_score - cur_score

        if delta < 0.0 or np.random.rand() < np.exp(-delta / (T + 1e-18)):
            cur_a = cand
            cur_score = cand_score
            if cur_score < best_score:
                best_score = cur_score
                best_a = cur_a.copy()
                step_scale = min(step_scale * 1.07, 0.5)
        else:
            step_scale *= 0.97

        T = T0 * (1.0 - step / max_steps)

    return best_a, best_score

    # still allowing the theoretical bound to dip below 1.52.
    # -------------------------------------------------------------

        n = 4096
    elif budget_s >= 600:
        n = 3072
    else:
        n = 2048
    print(f"[{time.time():.1f}] Using length n = {n}")

    # -------------------------------------------------------------
    # Build a rich basis of candidate vectors (already normalised)
    # -------------------------------------------------------------
    bases = []

    # uniform
    bases.append(np.full(n, 1.0 / n))

    for alpha in np.linspace(0.0, 8.0, 21):
        bases.append(symmetric_product_family(n, alpha))
    for beta in np.linspace(0.0, 8.0, 21):
        bases.append(left_heavy_power_family(n, beta))
    for gamma in np.linspace(0.0, 8.0, 21):
        bases.append(exponential_decay_family(n, gamma))

    # random Dirichlet draws with many concentration values
    conc_vals = np.logspace(-2, 2, 15)   # 0.01 … 100
    for conc in conc_vals:

            bases.append(random_dirichlet_start(n, conc))

    # random sparse vectors
    for _ in range(12):
        bases.append(random_sparse_start(n))

    # stack and enforce exact simplex normalisation
    bases = np.stack(bases, axis=0)
    bases = bases / bases.sum(axis=1, keepdims=True)

    K = bases.shape[0]
    print(f"[{time.time():.1f}] Basis size K = {K}")

    # -------------------------------------------------------------

    # -------------------------------------------------------------
    best_seq = None
    best_score = float("inf")

        if sc < best_score:
            best_score = sc
            best_seq = vec.tolist()
            best_idx = idx
    print(f"[{time.time():.1f}] Best single‑basis score = {best_score:.6f}")

    # -------------------------------------------------------------
    # Mixture optimisation (Adam on the smoothed surrogate)
    # -------------------------------------------------------------
    remaining = deadline - time.time()
    if remaining > 30.0:

        mix_deadline = time.time() + max(30.0, 0.7 * remaining)

        print(f"[{time.time():.1f}] Starting mixture optimisation")
        lam_opt, a_opt, sc_opt = adam_mix_optimize(
            bases,
            mix_deadline,
            max_iters=250_000,
            beta_start=5.0,
            beta_end=20_000.0,
            lr=0.018,
        )
        if sc_opt < best_score:
            best_score = sc_opt
            best_seq = a_opt.tolist()
            print(f"[{time.time():.1f}] New best after mixture = {best_score:.6f}")

    # -------------------------------------------------------------
    # Optional high‑β refinement (starting from the best mixture)
    # -------------------------------------------------------------
    if best_seq is not None and time.time() < deadline - 40.0:
        print(f"[{time.time():.1f}] High‑β refinement")
        a_start = np.array(best_seq, dtype=np.float64)
        lam_start = np.full(K, 1.0 / K)  # we'll let Adam re‑initialise
        # Use a short Adam run with a gigantic β range
        lam_ref, a_ref, sc_ref = adam_mix_optimize(
            bases,
            deadline - 25.0,
            max_iters=120_000,
            beta_start=2_000.0,
            beta_end=200_000.0,
            lr=0.010,
        )
        if sc_ref < best_score:
            best_score = sc_ref
            best_seq = a_ref.tolist()
            print(f"[{time.time():.1f}] After high‑β refinement = {best_score:.6f}")

    # -------------------------------------------------------------
    # Mirror sub‑gradient descent on the exact max‑objective (fine‑tune)
    # -------------------------------------------------------------
    if best_seq is not None and time.time() < deadline - 15.0:
        print(f"[{time.time():.1f}] Mirror sub‑gradient descent")
        a_sub, sc_sub = mirror_subgrad_optimize(
            np.array(best_seq, dtype=np.float64),
            deadline - 10.0,
            step0=0.9,
        )
        if sc_sub < best_score:
            best_score = sc_sub
            best_seq = a_sub.tolist()
            print(f"[{time.time():.1f}] After mirror sub‑grad = {best_score:.6f}")

    # -------------------------------------------------------------
    # Symmetrisation – many optimal solutions turn out symmetric
    # -------------------------------------------------------------
    if best_seq is not None:
        a_arr = np.array(best_seq, dtype=np.float64)
        a_sym = (a_arr + a_arr[::-1]) * 0.5
        a_sym = a_sym / a_sym.sum()  # re‑normalise (tiny numerical drift)
        sc_sym = compute_score(a_sym)
        if sc_sym < best_score:
            best_score = sc_sym
            best_seq = a_sym.tolist()
            print(f"[{time.time():.1f}] Symmetrised improvement = {best_score:.6f}")

    # -------------------------------------------------------------
    # Final tiny hill‑climb (simulated annealing) to escape shallow pits
    # -------------------------------------------------------------
    if best_seq is not None and time.time() < deadline - 5.0:
        print(f"[{time.time():.1f}] Final hill‑climb (SA)")
        a_final, sc_final = hill_climb_sa(
            np.array(best_seq, dtype=np.float64),
            deadline,
            max_steps=1_200_000,
        )
        if sc_final < best_score:
            best_score = sc_final
            best_seq = a_final.tolist()
            print(f"[{time.time():.1f}] Hill‑climb improved best = {best_score:.6f}")

    # -------------------------------------------------------------
    # Safety fallback – uniform vector if everything else failed
    # -------------------------------------------------------------
    if best_seq is None:
        n_small = 256
        best_seq = [1.0 / n_small] * n_small
        best_score = evaluate_sequence(best_seq)

    # -------------------------------------------------------------
    # Final official evaluation (the harness will also call it)
    # -------------------------------------------------------------
    final_score = evaluate_sequence(best_seq)
    print(f"\n[{time.time():.1f}] finished search – final reported score = {final_score:.6f}")

    # Return a plain Python list as required by the harness.
    return best_seq
