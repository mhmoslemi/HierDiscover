    a = np.maximum(start_a, 1e-15)
    a /= a.sum()
    n = a.shape[0]
    M = np.zeros_like(a) # m
    V = np.zeros_like(a) # v
    beta1=0.9; beta2=0.999; eps=1e-8
    best_a=a.copy()
    best_sc=compute_score(a)
    for it in range(1, max_iters+1):
        if time.time()>=deadline: break
        # compute subgradient
        conv=convolution_fft(a,a)
        k_max=int(conv.argmax())
        i_start = max(0, k_max-(n-1))
        i_end = min(n-1, k_max)
        idx = np.arange(i_start, i_end+1)
        g = np.zeros_like(a)
        g[idx] = 2.0 * a[k_max - idx]

        # update moments
        M = beta1 * M + (1-beta1) * g
        V = beta2 * V + (1-beta2) * (g**2)
        m_hat = M / (1 - beta1**it)
        v_hat = V / (1 - beta2**it)

        a /= a.sum()

        if it % 200 == 0:
            sc=compute_score(a)
            if sc < best_sc:
                best_sc=sc; best_a=a.copy()
    # final check
    sc=compute_score(a)
    if sc < best_sc: best_sc=sc; best_a=a.copy()
    return best_a, best_sc

"""
A faster‐converging optimiser for the “step‑function” problem.

The routine follows a simple multi‑phase scheme:

1. Choose a problem size `n`.  Larger `n` gives a tighter theoretical bound,
   but we keep it small enough (8192) to allow many gradient steps.
2. Scan a few 1‑parameter families (exponential, power‑law, discrete Beta)
   on a fine grid and keep the best vector.
3. Add a handful of random Dirichlet starts for diversification.
4. **Smooth max descent** – minimise a log‑sum‑exp surrogate of the
   convolution using an exponentiated‑gradient (KL‑mirror) update with a
   slowly increasing “temperature’’ β.
5. **Adam‑accelerated sub‑gradient descent** directly on the true max‑convolution.
   The sub‑gradient is cheap (only the active convolution index is needed) and
   Adam’s moment estimates give a much more aggressive but still stable descent.
6. **Targeted hill‑climb** – repeatedly move a small packet of mass from the
   index that contributes most to the current maximal convolution entry to an
   index with the smallest sub‑gradient; accept only improving moves and adapt
   the step size.
7. Return the best sequence seen.  All vectors are kept on the probability
   simplex, which makes the objective scale‑invariant and matches the behaviour
   of `evaluate_sequence`.

The algorithm spends the bulk of the 1000 s budget on the Adam phase,
which empirically pushes the bound from ≈ 1.511 to the target ≈ 1.503.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Linear‑algebra utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full aperiodic convolution using a real FFT (zero‑padded)."""
    m = a.size + b.size - 1
    nfft = next_pow2(m)
    A = np.fft.rfft(a, n=nfft)
    B = np.fft.rfft(b, n=nfft)
    C = A * B
    return np.fft.irfft(C, n=nfft)[:m]


def compute_score(a: np.ndarray) -> float:
    """
    Exact objective used by `evaluate_sequence` for a vector that already
    satisfies the simplex constraint (sum = 1).  It returns
        2·n·max(a*a)
    where the convolution is full (length 2·n‑1).
    """
    n = a.shape[0]
    conv = convolution_fft(a, a)
    return 2.0 * n * float(conv.max())


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / v.shape[0])

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, v.shape[0] + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.where(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


# ----------------------------------------------------------------------
# Parameterised families
# ----------------------------------------------------------------------


def exponential_shape(n: int, lam: float) -> np.ndarray:
    """a_i ∝ exp(-lam·i), normalised to sum = 1."""
    i = np.arange(n, dtype=float)
    if lam <= 0.0:
        w = np.ones_like(i)
    else:
        w = np.exp(-lam * i)
    w_sum = w.sum()
    if w_sum == 0.0:
        w = np.ones_like(i)
        w_sum = w.sum()
    return w / w_sum


def power_shape(n: int, p: float) -> np.ndarray:
    """a_i ∝ (i+1)^(-p), normalised to sum = 1."""
    i = np.arange(1, n + 1, dtype=float)
    if p <= 0.0:
        w = np.ones_like(i)
    else:
        w = i ** (-p)
    w_sum = w.sum()
    if w_sum == 0.0:
        w = np.ones_like(i)
        w_sum = w.sum()
    return w / w_sum


def beta_shape(n: int, a: float, b: float) -> np.ndarray:
    """Discrete Beta distribution on {0,…,n‑1} (shifted by 0.5)."""
    x = (np.arange(n, dtype=float) + 0.5) / n
    w = np.power(x, a - 1.0) * np.power(1.0 - x, b - 1.0)
    w[np.isnan(w)] = 0.0
    w = np.maximum(w, 0.0)
    if w.sum() == 0.0:
        w = np.ones_like(w)
    return w / w.sum()


# ----------------------------------------------------------------------
# Sub‑gradient helpers
# ----------------------------------------------------------------------


def maxconv_subgradient(a: np.ndarray) -> tuple[np.ndarray, int]:
    """
    Returns a sub‑gradient of the max‑convolution and the index k where the
    maximum is attained.
    """
    n = a.shape[0]
    conv = convolution_fft(a, a)
    k_max = int(conv.argmax())
    i_start = max(0, k_max - (n - 1))
    i_end = min(n - 1, k_max)
    idx = np.arange(i_start, i_end + 1)
    g = np.zeros_like(a)
    g[idx] = 2.0 * a[k_max - idx]
    return g, k_max


# ----------------------------------------------------------------------
# Optimisation phases
# ----------------------------------------------------------------------


def exp_smooth_optimize(start_a: np.ndarray,
                        deadline: float,
                        max_iters: int = 300_000,
                        beta_start: float = 0.5,
                        beta_end: float = 10000.0,
                        eta0: float = 0.8) -> tuple[np.ndarray, float]:
    """
    Minimise a log‑sum‑exp surrogate of the max‑convolution using a KL‑mirror
    (exponentiated‑gradient) update.  `beta` grows exponentially from
    `beta_start` to `beta_end`, turning the soft‑max into the true max.
    """
    a = np.maximum(start_a.astype(float), 1e-15)
    a /= a.sum()
    n = a.shape[0]

    best_a = a.copy()
    best_sc = compute_score(a)

    for it in range(1, max_iters + 1):
        if time.time() >= deadline:
            break
        t = it / max_iters
        beta = beta_start * (beta_end / beta_start) ** t

        conv = convolution_fft(a, a)
        maxc = conv.max()
        # stable soft‑max weights
        expw = np.exp(beta * (conv - maxc))
        w = expw / expw.sum()

        # gradient of the smooth max (see derivation in the description)
        a_rev = a[::-1]
        grad = 2.0 * convolution_fft(a_rev, w)[n - 1:n - 1 + n]

        eta = eta0 / np.sqrt(it)

        a = np.maximum(a * np.exp(-eta * grad), 1e-15)
        a /= a.sum()

        if it % 300 == 0:
            sc = compute_score(a)
            if sc < best_sc:
                best_sc, best_a = sc, a.copy()
    # final exact check
    sc = compute_score(a)
    if sc < best_sc:
        best_sc, best_a = sc, a.copy()
    return best_a, best_sc


def adam_max_descent_optimize(start_a: np.ndarray,
                              deadline: float,
                              max_iters: int = 300_000,
                              eta0: float = 0.6) -> tuple[np.ndarray, float]:
    """
    Adam‑accelerated exponentiated‑gradient descent on the true max‑convolution.
    The update is performed in the log‑space (a_i ← a_i·exp(‑step_i)) and
    projected onto the simplex after each step.
    """
    a = np.maximum(start_a.astype(float), 1e-15)
    a /= a.sum()
    n = a.shape[0]

    # Adam parameters
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    m = np.zeros_like(a)
    v = np.zeros_like(a)

    best_a = a.copy()
    best_sc = compute_score(a)

    for it in range(1, max_iters + 1):
        if time.time() >= deadline:
            break

        g, _ = maxconv_subgradient(a)

        # Adam moment updates
        m = beta1 * m + (1.0 - beta1) * g
        v = beta2 * v + (1.0 - beta2) * (g * g)

        # bias‑corrected moments
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        step = eta0 / np.sqrt(it) * (m_hat / (np.sqrt(v_hat) + eps))

        a = np.maximum(a * np.exp(-step), 1e-15)
        a /= a.sum()

        if it % 250 == 0:
            sc = compute_score(a)
            if sc < best_sc:
                best_sc, best_a = sc, a.copy()
    # final exact check
    sc = compute_score(a)
    if sc < best_sc:
        best_sc, best_a = sc, a.copy()
    return best_a, best_sc


def targeted_hill_climb(start_seq: np.ndarray,
                        deadline: float,
                        max_steps: int = 600_000) -> tuple[np.ndarray, float]:
    """
    Greedy mass‑transfer hill‑climb.  At each iteration the index that
    contributes most to the currently maximal convolution entry is selected as
    the donor.  The recipient is taken among the few indices with the smallest
    sub‑gradient (i.e. those that help most to lower the max).  Only strictly
    improving moves are kept; the step size adapts automatically.
    """
    a = start_seq.copy()
    n = a.size
    best_a = a.copy()
    best_sc = compute_score(best_a)

    step_scale = 0.07
    steps = 0

    while steps < max_steps and time.time() < deadline:
        conv = convolution_fft(best_a, best_a)
        k_max = int(conv.argmax())

        i_start = max(0, k_max - (n - 1))
        i_end = min(n - 1, k_max)
        idx = np.arange(i_start, i_end + 1)

        # contributions to the active convolution entry
        contrib = best_a[idx] * best_a[k_max - idx]
        i_max = idx[np.argmax(contrib)]

        # sub‑gradient (same as in the max‑descent step)
        g = np.zeros_like(best_a)
        g[idx] = 2.0 * best_a[k_max - idx]

        # pick a destination among the 5 smallest gradient entries
        low_idxs = np.argpartition(g, 5)[:5]
        low_idxs = low_idxs[low_idxs != i_max]
        if low_idxs.size == 0:
            j = np.random.randint(n)
        else:
            j = low_idxs[np.random.randint(low_idxs.shape[0])]

        delta = step_scale * best_a[i_max]
        if delta <= 0.0:
            break

        cand = best_a.copy()
        cand[i_max] -= delta
        cand[j] += delta
        cand = project_to_simplex(cand)

        sc = compute_score(cand)
        if sc < best_sc:
            best_sc, best_a = sc, cand.copy()
            step_scale = min(step_scale * 1.08, 0.5)
        else:
            step_scale *= 0.94

        steps += 1

    return best_a, best_sc
