def adam_optimize(a0, time_deadline, max_iters=50000, step_lr=0.1):
    a = a0.copy()
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1=0.9; beta2=0.999; eps=1e-8
    best_a=a; best_score=compute_score(a)
    t=0
    while time.time()<time_deadline and t<max_iters:
        t+=1
        conv=full_convolution(a)
        k=np.argmax(conv)
        # gradient
        idx=np.arange(a.shape[0])
        j=k-idx
        mask=(j>=0)&(j<a.shape[0])
        g=np.zeros_like(a)
        g[mask]=2*a[j[mask]]
        # Adam updates
        m = beta1*m + (1-beta1)*g
        v = beta2*v + (1-beta2)*(g**2)
        m_hat = m/(1 - beta1**t)
        v_hat = v/(1 - beta2**t)
        step = step_lr * m_hat / (np.sqrt(v_hat) + eps)
        a = a - step
        # Project onto simplex
        a = project_to_simplex(a)
        # Optional jitter at intervals
        if t%5000==0:
            a += np.random.randn(len(a))*0.001* (1.0/np.sqrt(t))
            a = project_to_simplex(a)
        score=compute_score(a)
        if score < best_score:
            best_score=score
            best_a=a.copy()
    return best_a

def adam_optimize(a0, deadline, lr=0.2, beta1=0.9, beta2=0.999, eps=1e-8):
    a = a0.copy()
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    best_a = a.copy()
    best_val = compute_score(a)
    t = 0
    while time.time() < deadline:
        t += 1
        conv = full_convolution(a)
        k = int(np.argmax(conv))
        idx = np.arange(a.shape[0])
        j = k - idx
        mask = (j >= 0) & (j < a.shape[0])
        g = np.zeros_like(a)
        g[mask] = 2.0 * a[j[mask]]
        # Adam update
        m = beta1 * m + (1 - beta1) * g
        v = beta2 * v + (1 - beta2) * (g * g)
        m_hat = m / (1 - beta1 ** t)
        v_hat = v / (1 - beta2 ** t)
        step = lr * m_hat / (np.sqrt(v_hat) + eps)
        a = a - step
        # Projection onto simplex
        a = project_to_simplex(a)
        # Occasionally jitter to avoid stagnation
        if t % 2000 == 0:
            a += np.random.randn(a.shape[0]) * 0.001/(np.sqrt(t))
            a = project_to_simplex(a)
        val = compute_score(a)
        if val < best_val:
            best_val = val
            best_a = a.copy()
    return best_a

assistantfinal**Solution Overview**

The optimisation task is  

\[
\min_{a\ge 0,\;\sum a_i = 1}\;2n\;\max_k (a*a)_k ,
\qquad n=2048 .
\]

The objective is the *maximum* entry of the linear‑convolution of a with itself.
A simple sub‑gradient descent (used in the previous baseline) already finds a good
local minimum (≈ 1.511).  
The new optimiser improves on this by

* **exploring many different starting points** – uniform, power‑law,
  Dirichlet samples, and several classical window functions (Hamming,
  Hann, Blackman, Kaiser).  
* **using a projected Adam optimiser** – Adam gives a much more adaptive
  step direction than plain sub‑gradient descent, while the Euclidean
  projection onto the probability simplex guarantees feasibility.
* **periodically adding a tiny random jitter** to escape plateaus.
* **finishing with a fast greedy hill‑climb** that moves a small amount of
  mass between two indices and keeps the best solution found.

All phases respect the 1000 s wall‑clock limit; the algorithm keeps the
best sequence seen so far and finally returns it as a plain Python list.

```python
"""Non‑negative step‑function optimiser for ``evaluate_sequence``.

The algorithm performs many short optimisation runs from diverse
initialisations (uniform, power‑law, Dirichlet, and classic window
functions).  Each run uses a projected Adam optimiser (sub‑gradient of the
active convolution entry) with a simplex projection.  Small random jitters
are added periodically.  The best vector found is polished by a cheap
greedy hill‑climb.  The whole search terminates well before the 1000 s
budget, returning the best sequence discovered.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper functions – projection, convolution, score
# ----------------------------------------------------------------------


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the probability simplex
    {x | x_i >= 0, sum x_i = 1}.
    """
    if np.all(v <= 0):
        # all non‑positive – the projection is the uniform distribution
        return np.full_like(v, 1.0 / v.shape[0])

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, v.shape[0] + 1)
    mask = u - cssv / ind > 0
    rho = np.where(mask)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def full_convolution(a: np.ndarray) -> np.ndarray:
    """
    Linear convolution a*a of length 2*len(a)-1 using an FFT of the next
    power‑of‑two size.  Returns a real array.
    """
    n = a.shape[0]
    m = 2 * n - 1
    n_fft = 1 << ((m - 1).bit_length())          # next power of two ≥ m
    A = np.fft.rfft(a, n=n_fft)
    C = A * A
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


def compute_score(a: np.ndarray) -> float:
    """
    Exact objective value for a normalised vector (sum = 1):
        2 * n * max(conv(a, a))
    """
    n = a.shape[0]
    conv = full_convolution(a)
    return float(2.0 * n * conv.max())


# ----------------------------------------------------------------------
# Generation of diverse starting points
# ----------------------------------------------------------------------


def random_power_law(n: int) -> np.ndarray:
    """a_i ∝ (i+1)^{-p} with a random exponent p ∈ [0.55, 0.75]."""
    p = np.random.uniform(0.55, 0.75)
    a = np.arange(1, n + 1, dtype=float) ** (-p)
    a /= a.sum()
    return a


def random_power_law_shuffled(n: int) -> np.ndarray:
    """Same as random_power_law but with a random permutation."""
    a = random_power_law(n)
    np.random.shuffle(a)
    return a


def random_dirichlet(n: int) -> np.ndarray:
    """Draw a Dirichlet sample with a random concentration α ∈ [0.3, 5]."""
    alpha = np.random.uniform(0.3, 5.0)
    a = np.random.gamma(alpha, 1.0, size=n)
    a /= a.sum()
    return a


def window_candidate(n: int) -> np.ndarray:
    """Choose a classical window (Hamming, Hann, Blackman, Kaiser) at random."""
    choice = np.random.choice(["hamming", "hann", "blackman", "kaiser"])
    if choice == "hamming":
        w = np.hamming(n)
    elif choice == "hann":
        w = np.hanning(n)
    elif choice == "blackman":
        w = np.blackman(n)
    else:   # kaiser
        beta = np.random.uniform(2.0, 12.0)
        w = np.kaiser(n, beta)
    w = np.maximum(w, 0.0)      # safety – all windows are non‑negative
    w /= w.sum()
    return w


def uniform_candidate(n: int) -> np.ndarray:
    """Uniform distribution (the simple baseline)."""
    return np.full(n, 1.0 / n)


def generate_start(n: int) -> np.ndarray:
    """Pick one of the above generators uniformly at random."""
    generators = [
        uniform_candidate,
        random_power_law,
        random_power_law_shuffled,
        random_dirichlet,
        window_candidate,
    ]
    gen = np.random.choice(generators)
    return gen(n)


# ----------------------------------------------------------------------
# Projected Adam optimiser (sub‑gradient of the active convolution entry)
# ----------------------------------------------------------------------


def adam_optimize(a0: np.ndarray,
                  deadline: float,
                  lr: float = 0.15,
                  beta1: float = 0.9,
                  beta2: float = 0.999,
                  eps: float = 1e-8) -> np.ndarray:
    """
    Projected Adam optimiser for 2·n·max(conv(a,a)).
    The sub‑gradient is taken at the index where the convolution attains
    its maximum.
    """
    a = a0.copy()
    n = a.shape[0]
    m = np.zeros_like(a)
    v = np.zeros_like(a)

    best_a = a.copy()
    best_val = compute_score(a)

    t = 0
    while time.time() < deadline:
        t += 1
        conv = full_convolution(a)
        k = int(np.argmax(conv))

        # sub‑gradient of conv_k w.r.t. a
        idx = np.arange(n)
        j = k - idx
        mask = (j >= 0) & (j < n)
        g = np.zeros_like(a)
        g[mask] = 2.0 * a[j[mask]]

        # Adam update
        m = beta1 * m + (1.0 - beta1) * g
        v = beta2 * v + (1.0 - beta2) * (g * g)

        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)

        step = lr * m_hat / (np.sqrt(v_hat) + eps)
        a = a - step
        a = project_to_simplex(a)

        # occasional tiny random jitter to avoid stagnation
        if t % 2000 == 0:
            jitter = np.random.randn(n) * (lr * 0.005 / np.sqrt(t))
            a += jitter
            a = project_to_simplex(a)

        val = compute_score(a)
        if val < best_val:
            best_val = val
            best_a = a.copy()

    return best_a


# ----------------------------------------------------------------------
# Simple greedy hill‑climb (same spirit as the baseline)
# ----------------------------------------------------------------------


def hill_climb(a_start: np.ndarray,
               deadline: float,
               max_no_improve: int = 500_000) -> np.ndarray:
    """
    Stochastic mass‑transfer moves: move a small fraction of probability
    from one index to another, keep the best solution.
    """
    a = a_start.copy()
    n = a.shape[0]

    best_a = a.copy()
    best_val = compute_score(a)

    step_scale = 0.05      # initial fraction of mass moved
    it = 0
    no_improve = 0

    while time.time() < deadline and it < max_no_improve:
        i, j = np.random.randint(n, size=2)
        if i == j:
            continue

        delta = step_scale * a[i]
        cand = a.copy()
        cand[i] -= delta
        cand[j] += delta
        cand = np.maximum(cand, 0.0)
        cand = project_to_simplex(cand)

        val = compute_score(cand)
        if val < best_val:
            best_val = val
            best_a = cand.copy()
            a = cand
            step_scale = min(step_scale * 1.07, 0.5)
            no_improve = 0
        else:
            step_scale *= 0.985
            no_improve += 1

        it += 1

    return best_a

        best_seq = [1.0 / n] * n
        best_val = compute_score(np.array(best_seq, dtype=float))

    # ------------------------------------------------------------------
    # 4. official evaluation report
    # ------------------------------------------------------------------
    official_score = evaluate_sequence(best_seq)
    print(f"\nSearch finished. Official score = {official_score:.6f}")
    return best_seq
```
