"""
Hybrid optimiser for the “max‑auto‑convolution” problem.

Main ideas
----------
1. Initialise from a collection of analytic families (uniform, power‑law,
   geometric, central‑binomial) and keep the best starter.
2. Apply a *linear‑programming bound reduction* step: for the current
   vector f we maximise the total mass of a new vector g under the
   constraint   (f * g)[k] ≤ M := max(f*f)   for every convolution index k.
   The LP solution can be mixed with f (convex combination) which strictly
   reduces the bound M.  Repeating this cheap LP step a few dozen times
   drives the objective down considerably.
3. Refine the result with a smooth log‑sum‑exp surrogate minimised by Adam.
4. Polish with true sub‑gradient descent on the max‑convolution objective.
5. Finish with a short stochastic hill‑climb.

All phases respect a global wall‑clock deadline (default 1000 s) and
store the best sequence encountered.  The returned sequence is a plain
Python list of non‑negative floats whose sum is 1, ready to be fed to
``evaluate_sequence``.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Basic linear‑algebra utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """
    Full linear convolution using a real FFT.
    Returns an array of length len(a)+len(b)-1.
    """
    m = a.shape[0] + b.shape[0] - 1
    n_fft = next_pow2(m)
    A = np.fft.rfft(a, n=n_fft)
    B = np.fft.rfft(b, n=n_fft)
    C = A * B
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


def compute_score(a: np.ndarray) -> float:
    """
    Exact objective for a vector already on the probability simplex
    (sum = 1).  Returns 2·n·max(conv(a,a)).
    """
    n = a.shape[0]
    conv = convolution_fft(a, a)
    return 2.0 * n * float(conv.max())


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of v onto the probability simplex:
        sum(v) = 1, v >= 0.
    """
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / v.shape[0])

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, v.shape[0] + 1)
    cond = u - cssv / ind > 0
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


# ----------------------------------------------------------------------
# Analytic starter families
# ----------------------------------------------------------------------


def generate_central_binomial(n: int) -> np.ndarray:
    """a_i = C(2i,i) / 4^i, then simplex‑projected."""
    a = np.empty(n, dtype=np.float64)
    a[0] = 1.0
    for i in range(1, n):
        a[i] = a[i - 1] * (2 * i - 1) / (2.0 * i)
    return project_to_simplex(a)


def generate_powerlaw(n: int, p: float) -> np.ndarray:
    """a_i ∝ (i+1)^{-p}."""
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = idx ** (-p)
    return project_to_simplex(a)


def generate_geometric(n: int, r: float) -> np.ndarray:
    """a_i = (1‑r)·r^i, exact sum = 1 for r<1."""
    i = np.arange(n, dtype=np.float64)
    a = (1.0 - r) * (r ** i)
    return project_to_simplex(a)


# ----------------------------------------------------------------------
# Smooth (log‑sum‑exp) optimisation – Adam
# ----------------------------------------------------------------------


def smooth_gradient_optimize(
    start_a: np.ndarray,
    time_deadline: float,
    max_iters: int = 300_000,
    beta_start: float = 5.0,
    beta_end: float = 200.0,
    step0: float = 0.8,
    adam_beta1: float = 0.9,
    adam_beta2: float = 0.999,
    adam_eps: float = 1e-8,
) -> tuple[np.ndarray, float]:
    """
    Adam optimiser for the smoothed max‑convolution objective.
    Uses a log‑sum‑exp surrogate with an annealed temperature β.
    """
    a = start_a.astype(np.float64).copy()
    n = a.shape[0]
    m1 = np.zeros_like(a)
    m2 = np.zeros_like(a)

    best_a = a.copy()
    best_score = compute_score(a)

    for it in range(1, max_iters + 1):
        if time.time() >= time_deadline:
            break

        # anneal temperature
        t_frac = it / max_iters
        beta = beta_start * (beta_end / beta_start) ** t_frac

        # full auto‑convolution
        conv = convolution_fft(a, a)
        maxc = conv.max()

        # soft‑max weights
        exp_w = np.exp(beta * (conv - maxc))
        w = exp_w / exp_w.sum()

        # gradient of the smoothed objective:
        #   ∂/∂a = 2 * (rev(a) * w)   (restricted to length n)
        rev_a = a[::-1]
        grad_full = convolution_fft(rev_a, w)            # length 3n‑2
        grad = 2.0 * grad_full[n - 1:n - 1 + n]          # central n entries

        # Adam update
        m1 = adam_beta1 * m1 + (1.0 - adam_beta1) * grad
        m2 = adam_beta2 * m2 + (1.0 - adam_beta2) * (grad * grad)
        m1_hat = m1 / (1.0 - adam_beta1 ** it)
        m2_hat = m2 / (1.0 - adam_beta2 ** it)

        step = step0 / np.sqrt(it)
        a_new = a - step * m1_hat / (np.sqrt(m2_hat) + adam_eps)
        a_new = project_to_simplex(a_new)

        a = a_new

        if it % 500 == 0:
            cur_score = compute_score(a)
            if cur_score < best_score:
                best_score = cur_score
                best_a = a.copy()

    final_score = compute_score(a)
    if final_score < best_score:
        best_score, best_a = final_score, a.copy()
    return best_a, best_score


# ----------------------------------------------------------------------
# Exact sub‑gradient optimiser on the true max‑objective
# ----------------------------------------------------------------------


def subgrad_optimize(
    init_a: np.ndarray,
    time_deadline: float,
    max_iter: int = 300_000,
    step0: float = 0.07,
) -> tuple[np.ndarray, float]:
    """Projected sub‑gradient descent on 2·n·max(conv(a,a))."""
    a = init_a.astype(np.float64).copy()
    best_a = a.copy()
    best_score = compute_score(a)
    n = a.shape[0]

    for t in range(1, max_iter + 1):
        if time.time() >= time_deadline:
            break
        conv = convolution_fft(a, a)
        k = int(np.argmax(conv))

        # sub‑gradient for the active convolution entry
        idx = np.arange(n)
        j = k - idx
        mask = (j >= 0) & (j < n)
        g = np.zeros_like(a)
        g[mask] = 2.0 * a[j[mask]]

        step = step0 / np.sqrt(t)
        a_new = a - step * g
        a_new = project_to_simplex(a_new)

        cur_score = compute_score(a_new)
        if cur_score < best_score:
            best_score, best_a = cur_score, a_new.copy()

        a = a_new

        if t % 1000 == 0:
            jitter = np.random.randn(n) * (step0 * 0.004 / np.sqrt(t))
            a += jitter
            a = project_to_simplex(a)

    final_score = compute_score(a)
    if final_score < best_score:
        best_score, best_a = final_score, a.copy()
    return best_a, best_score


# ----------------------------------------------------------------------
# Simple stochastic hill‑climb (final polishing)
# ----------------------------------------------------------------------


def hill_climb_permutation(
    seq,
    time_deadline,
    max_steps: int = 200_000,
) -> tuple[list[float], float]:
    """Tiny stochastic moves (swap or fractional transfer) with simplex projection."""
    a = np.array(seq, dtype=np.float64)
    n = a.shape[0]
    best_a = a.copy()
    best_score = compute_score(best_a)

    step_scale = 0.08  # initial fraction of mass moved

    while max_steps > 0 and time.time() < time_deadline:
        cand = best_a.copy()
        if np.random.rand() < 0.5:      # swap two entries
            i, j = np.random.randint(n, size=2)
            cand[i], cand[j] = cand[j], cand[i]
        else:                           # move a fraction from i to j
            i, j = np.random.randint(n, size=2)
            if i != j:
                delta = step_scale * best_a[i]
                cand[i] -= delta
                cand[j] += delta
                if cand[i] < 0:
                    cand[i] = 0.0

        cand = project_to_simplex(cand)
        cur_score = compute_score(cand)

        if cur_score < best_score:
            best_score = cur_score
            best_a = cand
            step_scale = min(step_scale * 1.06, 0.5)
        else:
            step_scale *= 0.97

        max_steps -= 1

    return best_a.tolist(), best_score

def _build_constraint_matrix(f: np.ndarray) -> _sp.csr_matrix:
    """
    Build the (2n‑1) × n matrix A for constraints (f * g)[k] ≤ M.
    Row k contains f shifted by k positions (zero‑padded).
    """
    n = f.shape[0]
    rows = 2 * n - 1
    # dense construction then conversion – still modest for n≈2048
    A_dense = np.zeros((rows, n), dtype=np.float64)
    for j in range(n):
        A_dense[j:j + n, j] = f
    return _sp.csr_matrix(A_dense)


def _lp_step(f: np.ndarray, M: float) -> np.ndarray | None:
    """
    Solve  max 1ᵀg  s.t.  A g ≤ M·1, g ≥ 0,
    where A is the convolution matrix built from f.
    Returns the optimal g or None on failure.
    """
    if not SCIPY_AVAILABLE:
        return None
    n = f.shape[0]
    A = _build_constraint_matrix(f)
    b = np.full(2 * n - 1, M, dtype=np.float64)
    c = -np.ones(n, dtype=np.float64)          # minimise –sum(g)
    # Use HiGHS – works well for medium‑size LPs
    res = _opt.linprog(c, A_ub=A, b_ub=b, bounds=(0, None),
                       method="highs", options={"presolve": True})
    if not res.success:
        return None
    return res.x


def lp_bound_reduction(
    start_f: np.ndarray,
    time_deadline: float,
    max_iter: int = 30,
    mix_param: float = 0.35,
) -> tuple[np.ndarray, float]:
    """
    Repeatedly solve the LP described above and mix the solution with the
    current vector.  This provably never raises the max‑convolution bound
    and in practice drives the objective down quickly.
    """
    f = start_f.copy()
    best_f = f.copy()
    best_score = compute_score(f)

    for it in range(max_iter):
        if time.time() >= time_deadline:
            break
        conv = convolution_fft(f, f)
        M = conv.max()
        g0 = _lp_step(f, M)
        if g0 is None:
            break

        sum_g0 = float(np.sum(g0))
        if sum_g0 <= 1e-12:
            break

        # Scaling down to keep the normalisation (sum=1).  sum_g0 ≥ 1
        g_scaled = g0 * (1.0 / sum_g0)

        # Convex combination – mixing parameter can be reduced if progress stalls
        f_new = (1.0 - mix_param) * f + mix_param * g_scaled
        f_new = project_to_simplex(f_new)

        new_score = compute_score(f_new)
        if new_score < best_score:
            best_score = new_score
            best_f = f_new.copy()
            f = f_new
            print(f"LP step {it}: improved score = {best_score:.6f}")
        else:
            # If we did not improve, shrink the mixing weight
            mix_param *= 0.7
            # still keep the current f for the next iteration
            if mix_param < 1e-3:
                break

    return best_f, best_score

def propose_candidate(seed: int = 12345, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative sequence that minimises ``evaluate_sequence``.
    Returns a plain Python list of floats (already on the simplex).
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 3.0         # tiny safety margin

    # --------------------------------------------------------------
    # 0. Problem size
    # --------------------------------------------------------------
    try:
        n = len(height_sequence_1)                 # type: ignore[name-defined]
        init_candidates = [np.array(height_sequence_1, dtype=np.float64)]
        print(f"Using externally supplied height_sequence_1 (n={n}).")
    except Exception:
        n = 2048
        init_candidates = []
        print(f"No external sequence – using default length n={n}.")

    # --------------------------------------------------------------
    # 1. Analytic starters (uniform, central‑binomial, power‑law, geometric)
    # --------------------------------------------------------------
    init_candidates.append(np.full(n, 1.0 / n, dtype=np.float64))   # uniform
    init_candidates.append(generate_central_binomial(n))

    for p in np.linspace(0.3, 1.0, 9):
        init_candidates.append(generate_powerlaw(n, float(p)))

    for r in [0.97, 0.98, 0.985, 0.99, 0.992, 0.995]:
        init_candidates.append(generate_geometric(n, float(r)))

    # Fine‑grained search for a geometric distribution (often very competitive)
    if time.time() < deadline - 30.0:
        for r in np.linspace(0.95, 0.9999, 40):
            cand = generate_geometric(n, float(r))
            init_candidates.append(cand)

    # Evaluate starters and keep the best
    best_seq = None
    best_val = float("inf")
    for idx, cand in enumerate(init_candidates):
        cand = project_to_simplex(cand)
        val = compute_score(cand)
        print(f"Starter {idx:02d}: score = {val:.6f}")
        if val < best_val:
            best_val = val
            best_seq = cand.copy()

    print(f"Best starter score = {best_val:.6f}")

    # --------------------------------------------------------------
    # 2. LP bound‑reduction phase (if SciPy is available)
    # --------------------------------------------------------------
    lp_deadline = deadline - 500.0   # keep ~500 s for later phases
    if SCIPY_AVAILABLE and time.time() < lp_deadline:
        print("Starting LP bound‑reduction phase.")
        lp_seq, lp_score = lp_bound_reduction(
            best_seq,
            time_deadline=lp_deadline,
            max_iter=40,
            mix_param=0.35,
        )
        if lp_score < best_val:
            best_val = lp_score
            best_seq = lp_seq.copy()
            print(f"LP phase improved score to {best_val:.6f}")

    # --------------------------------------------------------------
    # 3. Smooth (log‑sum‑exp) Adam optimisation – longest phase
    # --------------------------------------------------------------
    smooth_deadline = deadline - 200.0
    if time.time() < smooth_deadline:
        remaining = smooth_deadline - time.time()
        # Rough estimate: ~2500 it/s for n=2048
        max_iters = int(min(1_200_000, remaining * 2500))
        print(f"Smooth Adam optimisation – up to {max_iters} iterations.")
        a_smooth, smooth_score = smooth_gradient_optimize(
            best_seq,
            time_deadline=smooth_deadline,
            max_iters=max_iters,
            beta_start=4.0,
            beta_end=300.0,
            step0=0.9,
        )
        if smooth_score < best_val:
            best_val = smooth_score
            best_seq = a_smooth.copy()
            print(f"Smooth phase improved score to {best_val:.6f}")

    # --------------------------------------------------------------
    # 4. Projected sub‑gradient polishing
    # --------------------------------------------------------------
    subgrad_deadline = deadline - 70.0
    if time.time() < subgrad_deadline:
        remaining = subgrad_deadline - time.time()
        max_iter = int(min(800_000, remaining * 2000))
        print(f"Sub‑gradient optimisation – up to {max_iter} iterations.")
        a_sub, sub_score = subgrad_optimize(
            best_seq,
            time_deadline=subgrad_deadline,
            max_iter=max_iter,
            step0=0.06,
        )
        if sub_score < best_val:
            best_val = sub_score
            best_seq = a_sub.copy()
            print(f"Sub‑gradient phase improved score to {best_val:.6f}")

    # --------------------------------------------------------------
    # 5. Random perturbation + short sub‑gradient burst (optional)
    # --------------------------------------------------------------
    restart_deadline = deadline - 30.0
    if time.time() < restart_deadline:
        base = best_seq.copy()
        for r in range(5):
            if time.time() >= restart_deadline:
                break
            pert = base + 0.02 * np.random.randn(n)
            pert = project_to_simplex(pert)
            sub_dead = time.time() + 12.0
            a_tmp, s_tmp = subgrad_optimize(
                pert,
                time_deadline=sub_dead,
                max_iter=120_000,
                step0=0.05,
            )
            if s_tmp < best_val:
                best_val = s_tmp
                best_seq = a_tmp.copy()
                print(f"Restart {r} gave improvement: {best_val:.6f}")

    # --------------------------------------------------------------
    # 6. Final stochastic hill‑climb
    # --------------------------------------------------------------
    hill_deadline = deadline - 2.0
    if time.time() < hill_deadline:
        steps = 300_000
        print(f"Final hill‑climb for up to {steps} steps.")
        polished_seq, hill_score = hill_climb_permutation(
            best_seq.tolist(),
            time_deadline=hill_deadline,
            max_steps=steps,
        )
        if hill_score < best_val:
            best_val = hill_score
            best_seq = np.array(polished_seq, dtype=np.float64)
            print(f"Hill‑climb improved final score to {best_val:.6f}")

    # --------------------------------------------------------------
    # 7. Safeguard – fall back to uniform if nothing was found
    # --------------------------------------------------------------
    if best_seq is None:
        best_seq = np.full(n, 1.0 / n, dtype=np.float64)
        best_val = compute_score(best_seq)

    # --------------------------------------------------------------
    # 8. Final reporting via the provided evaluate_sequence()
    # --------------------------------------------------------------
    final_score = evaluate_sequence(best_seq.tolist())
    print(f"\nSearch finished. Final reported score = {final_score:.6f}")

    return best_seq.tolist()
