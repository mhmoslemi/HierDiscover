def lp_refine(f, time_deadline, max_iters=20, t=0.2):
    n = len(f)
    best_f = f.copy()
    best_score = compute_score(f)
    while max_iters > 0 and time.time() < time_deadline:
        # compute current M
        conv = convolution_fft(f, f)
        M = float(conv.max())
        # build constraints matrix
        len_c = 2*n - 1
        # Build full matrix
        A = np.zeros((len_c, n), dtype=np.float64)
        for j in range(n):
            A[j:j+n, j] = f
        # Solve LP: maximize sum(g) => minimize -sum(g)
        c = -np.ones(n)
        b_ub = np.full(len_c, M)
        bounds = [(0, None)] * n
        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs', options={'presolve': True})
        if not res.success:
            print("LP failed or infeasible")
            break
        g = res.x
        # mix
        f = (1 - t) * f + t * g
        f = project_to_simplex(f)
        # evaluate
        cur_score = compute_score(f)
        if cur_score < best_score:
            best_score = cur_score
            best_f = f.copy()
            print(f"LP iteration improved score to {best_score:.6f}")
        max_iters -= 1
    return best_f, best_score

"""
Non‑negative step‑function optimiser for ``evaluate_sequence``.

Main ideas
----------
* Build a rich pool of analytic starting vectors (uniform, central binomial,
  power‑law, geometric, Beta‑shaped, mixtures and a few random Dirichlet
  draws).  Keep the best.
* Refine the best vector with a long Adam optimisation of a smooth
  log‑sum‑exp surrogate for the max‑convolution.
* Perform a sequence of linear‑programming refinement steps.  For the current
  vector f we solve

      maximise   Σ g_i
      subject to  (f * g)[k] ≤ M   ∀k   (M = max(f*f) ),
                 g_i ≥ 0

  The optimum pushes mass towards indices that are “quiet’’ for f.  Mixing a
  small fraction of the LP solution back into f provably never increases the
  max‑convolution and usually lowers it.
* A short projected sub‑gradient descent on the true objective follows.
* Finish with a cheap stochastic hill‑climb that swaps or moves tiny amounts
  of mass between positions.
* All phases respect the wall‑clock deadline (default 1000 s) and keep the
  best sequence seen.  The final sequence is returned as a plain list of
  floats (already on the simplex, i.e. non‑negative and summing to 1).
"""

import time
import numpy as np
from scipy.optimize import linprog

# ----------------------------------------------------------------------
# Basic utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full linear convolution via real FFT."""
    m = a.shape[0] + b.shape[0] - 1
    n_fft = next_pow2(m)
    A = np.fft.rfft(a, n=n_fft)
    B = np.fft.rfft(b, n=n_fft)
    C = A * B
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


def compute_score(a: np.ndarray) -> float:
    """
    Exact objective for a vector already on the simplex (∑a = 1).
    Returns 2·n·max(conv(a,a)).
    """
    n = a.shape[0]
    conv = convolution_fft(a, a)
    return 2.0 * n * float(conv.max())


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / v.shape[0])

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, v.shape[0] + 1)
    cond = u - cssv / ind > 0
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


# ----------------------------------------------------------------------
# Analytic families for initial guesses
# ----------------------------------------------------------------------


def generate_central_binomial(n: int) -> np.ndarray:
    """a_i = C(2i,i) / 4^i, normalised to sum 1 (monotone decreasing)."""
    a = np.empty(n, dtype=np.float64)
    a[0] = 1.0
    for i in range(1, n):
        a[i] = a[i - 1] * (2 * i - 1) / (2.0 * i)
    return project_to_simplex(a)


def generate_powerlaw(n: int, p: float) -> np.ndarray:
    """a_i ∝ (i+1)^(−p), normalised to sum 1."""
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = idx ** (-p)
    return project_to_simplex(a)


def generate_geometric(n: int, r: float) -> np.ndarray:
    """Geometric distribution a_i = (1‑r)·r^i, normalised to sum 1."""
    i = np.arange(n, dtype=np.float64)
    a = (1.0 - r) * (r ** i)
    return project_to_simplex(a)


def generate_beta(n: int, alpha: float, beta: float) -> np.ndarray:
    """
    Discrete Beta‑shaped distribution.
    Samples the Beta PDF at points (i+0.5)/n to avoid the 0/1 singularities.
    """
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    a = (x ** (alpha - 1.0)) * ((1.0 - x) ** (beta - 1.0))
    a = np.nan_to_num(a, nan=0.0, posinf=0.0, neginf=0.0)
    return project_to_simplex(a)


def mix_vectors(v1: np.ndarray, v2: np.ndarray, lam: float) -> np.ndarray:
    """Convex mixture λ·v1 + (1‑λ)·v2, projected back onto the simplex."""
    return project_to_simplex(lam * v1 + (1.0 - lam) * v2)


# ----------------------------------------------------------------------
# Smooth (log‑sum‑exp) optimisation – Adam
# ----------------------------------------------------------------------


def smooth_adam_optimize(start_a: np.ndarray,
                         time_deadline: float,
                         beta_start: float = 5.0,
                         beta_end: float = 1000.0,
                         step0: float = 0.85,
                         adam_beta1: float = 0.9,
                         adam_beta2: float = 0.999,
                         adam_eps: float = 1e-8) -> tuple[np.ndarray, float]:
    """
    Adam optimiser for the soft‑max surrogate of max(conv(a,a)).
    Stops when the wall‑clock deadline is reached.
    """
    a = start_a.astype(np.float64).copy()
    n = a.shape[0]
    m1 = np.zeros_like(a)
    m2 = np.zeros_like(a)
    best_a = a.copy()
    best_score = compute_score(a)
    it = 0

    while time.time() < time_deadline:
        it += 1
        # linear schedule for the temperature β (geometric annealing)
        t_frac = min(it / 1_000_000.0, 1.0)   # assume up to ~1 M iters
        beta = beta_start * (beta_end / beta_start) ** t_frac

        conv = convolution_fft(a, a)
        maxc = conv.max()
        # soft‑max weights (log‑sum‑exp)
        exp_w = np.exp(beta * (conv - maxc))
        w = exp_w / exp_w.sum()

        rev_a = a[::-1]
        grad_full = convolution_fft(rev_a, w)               # length 3n‑2
        grad = 2.0 * grad_full[n - 1:n - 1 + n]

        # Adam update
        m1 = adam_beta1 * m1 + (1.0 - adam_beta1) * grad
        m2 = adam_beta2 * m2 + (1.0 - adam_beta2) * (grad * grad)
        m1_hat = m1 / (1.0 - adam_beta1 ** it)
        m2_hat = m2 / (1.0 - adam_beta2 ** it)

        step = step0 / np.sqrt(it)
        a = a - step * m1_hat / (np.sqrt(m2_hat) + adam_eps)
        a = project_to_simplex(a)

        if it % 2000 == 0:
            cur_score = compute_score(a)
            if cur_score < best_score:
                best_score = cur_score
                best_a = a.copy()
                print(f"[Adam] iteration {it:,} → best score {best_score:.6f}")

    final_score = compute_score(a)
    if final_score < best_score:
        best_score, best_a = final_score, a.copy()
    return best_a, best_score


# ----------------------------------------------------------------------
# Linear‑programming refinement (active‑set style)
# ----------------------------------------------------------------------


def lp_refine(f: np.ndarray,
              time_deadline: float,
              max_iters: int = 12,
              mix_coeff: float = 0.15) -> tuple[np.ndarray, float]:
    """
    One‑dimensional LP refinement described in the problem statement.
    Repeatedly solves

        maximise Σ g_i  subject to (f * g)_k ≤ M, g_i ≥ 0,

    where M = max(f * f).  A convex combination (1‑t)·f + t·g (projected onto
    the simplex) is taken after each solve.  The procedure never increases
    the max‑convolution and often reduces it.
    """
    n = f.shape[0]
    len_c = 2 * n - 1
    best_f = f.copy()
    best_score = compute_score(f)

    for it in range(1, max_iters + 1):
        if time.time() >= time_deadline:
            break

        # current bound
        conv = convolution_fft(f, f)
        M = float(conv.max())

        # ----- build the full constraint matrix A (size (2n‑1) × n) -----
        # column j contains shifted copy of f
        A = np.zeros((len_c, n), dtype=np.float64)
        for j in range(n):
            A[j:j + n, j] = f

        # ----- solve the LP -----
        # maximise Σ g_i  ⇔ minimise −Σ g_i
        c = -np.ones(n, dtype=np.float64)
        b_ub = np.full(len_c, M, dtype=np.float64)
        bounds = [(0.0, None)] * n

        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds,
                      method='highs', options={'presolve': True})

        if not res.success:
            # If the solver fails we break – the previous f is the best we have.
            print(f"[LP] iteration {it}: solver failed – exiting LP phase.")
            break

        g = res.x

        # ----- mix and re‑project onto the simplex -----
        f = (1.0 - mix_coeff) * f + mix_coeff * g
        f = project_to_simplex(f)

        cur_score = compute_score(f)
        if cur_score < best_score:
            best_score = cur_score
            best_f = f.copy()
            print(f"[LP] iteration {it} improved score to {best_score:.6f}")

    return best_f, best_score


# ----------------------------------------------------------------------
# Exact sub‑gradient optimiser on the true max‑objective
# ----------------------------------------------------------------------


def subgrad_optimize(init_a: np.ndarray,
                     time_deadline: float,
                     step0: float = 0.07,
                     max_iter: int = 500_000) -> tuple[np.ndarray, float]:
    """Projected sub‑gradient descent on 2·n·max(conv(a,a))."""
    a = init_a.astype(np.float64).copy()
    best_a = a.copy()
    best_score = compute_score(a)
    n = a.shape[0]

    for t in range(1, max_iter + 1):
        if time.time() >= time_deadline:
            break
        conv = convolution_fft(a, a)
        k = int(np.argmax(conv))

        # sub‑gradient of the active convolution entry
        idx = np.arange(n)
        j = k - idx
        mask = (j >= 0) & (j < n)
        g = np.zeros_like(a)
        g[mask] = 2.0 * a[j[mask]]

        step = step0 / np.sqrt(t)
        a_new = a - step * g
        a_new = project_to_simplex(a_new)

        cur_score = compute_score(a_new)
        if cur_score < best_score:
            best_score, best_a = cur_score, a_new.copy()

        a = a_new

        # occasional jitter to escape flat plateaus
        if t % 1000 == 0:
            jitter = np.random.randn(n) * (step0 * 0.004 / np.sqrt(t))
            a = project_to_simplex(a + jitter)

    final_score = compute_score(a)
    if final_score < best_score:
        best_score, best_a = final_score, a.copy()
    return best_a, best_score


# ----------------------------------------------------------------------
# Final stochastic hill‑climb
# ----------------------------------------------------------------------


def hill_climb_permutation(seq,
                           time_deadline,
                           max_steps: int = 300_000) -> tuple[list[float], float]:
    """Tiny stochastic moves (swap or fractional transfer) with simplex projection."""
    a = np.array(seq, dtype=np.float64)
    n = a.shape[0]
    best_a = a.copy()
    best_score = compute_score(best_a)

    step_scale = 0.08  # initial fraction of mass moved

    while max_steps > 0 and time.time() < time_deadline:
        cand = best_a.copy()
        if np.random.rand() < 0.5:      # swap two entries
            i, j = np.random.randint(n, size=2)
            cand[i], cand[j] = cand[j], cand[i]
        else:                           # move a fraction from i to j
            i, j = np.random.randint(n, size=2)
            if i != j:
                delta = step_scale * best_a[i]
                cand[i] -= delta
                cand[j] += delta
                if cand[i] < 0:
                    cand[i] = 0.0

        cand = project_to_simplex(cand)
        cur_score = compute_score(cand)

        if cur_score < best_score:
            best_score = cur_score
            best_a = cand
            step_scale = min(step_scale * 1.06, 0.5)
        else:
            step_scale *= 0.97

        max_steps -= 1

    return best_a.tolist(), best_score
