"""
Propose a step‑function (non‑negative vector) that minimises `evaluate_sequence`.

Algorithm
---------
1. Choose a length `n` according to the remaining wall‑clock budget.
2. Scan a handful of analytic families (U‑shape, power‑law, exponential,
   symmetric beta, discrete Gaussian).  Keep the best few candidates.
3. Refine the current best with a *smooth* surrogate of the max‑convolution:
   fβ(a)= (1/β)·log∑exp(β·(a*a)_k).  Optimisation is performed with
   L‑BFGS‑B (if SciPy is present) and a few Adam runs for several β values.
4. Run a long mirror (exponentiated) sub‑gradient descent on the true
   max‑objective.  The step size follows ηt=0.8/√t and the iterate is always
   projected onto the probability simplex.
5. Apply a low‑pass filter in the Fourier domain (several cut‑offs) and
   keep the best smoothed version.
6. Symmetrise the vector (a←½(a+rev(a))) – most optima are symmetric.
7. A tiny simulated‑annealing hill‑climb cleans the last remaining tiny
   imperfections.
8. Return the best sequence found (as a plain Python list of floats).

All stages respect the global `budget_s` deadline (default 1000 s) and the
search never exceeds the allowed time.  The code works with only NumPy;
SciPy is optional (used for L‑BFGS‑B).  No file‑IO or network access is used.
"""

import time
import math
import numpy as np

# optional: SciPy for L‑BFGS‑B
try:
    import scipy.optimize as _opt
    _HAS_SCIPY = True
except Exception:  # pragma: no cover
    _HAS_SCIPY = False


# ----------------------------------------------------------------------
# FFT utilities
# ----------------------------------------------------------------------
def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """
    Linear (aperiodic) convolution via real FFT.
    Returns an array of length len(a)+len(b)-1.
    """
    m = a.size + b.size - 1
    nfft = next_pow2(m)
    A = np.fft.rfft(a, n=nfft)
    B = np.fft.rfft(b, n=nfft)
    C = A * B
    conv = np.fft.irfft(C, n=nfft)[:m]
    return conv


def compute_score(a: np.ndarray) -> float:
    """
    Exact objective for a probability vector (∑a = 1):
        score = 2·n·max(a*a)
    This equals ``evaluate_sequence`` for the same normalised vector.
    """
    n = a.shape[0]
    conv = convolution_fft(a, a)
    return 2.0 * n * float(conv.max())


# ----------------------------------------------------------------------
# Simplex utilities
# ----------------------------------------------------------------------
def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection onto the probability simplex."""
    n = v.shape[0]
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / n)

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.where(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def softmax(z: np.ndarray) -> np.ndarray:
    """Numerically stable soft‑max (maps ℝⁿ → simplex)."""
    z_max = np.max(z)
    e = np.exp(z - z_max)
    return e / e.sum()


# ----------------------------------------------------------------------
# Analytic families (all return a probability vector, sum = 1)
# ----------------------------------------------------------------------
def symmetric_product_family(n: int, alpha: float) -> np.ndarray:
    """U‑shaped: a_i ∝ [(i+½)(n‑½‑i)]^(‑α)."""
    i = np.arange(n, dtype=np.float64)
    # log‑form to avoid overflow/underflow
    logw = -alpha * (np.log(i + 0.5) + np.log(n - 0.5 - i))
    w = np.exp(logw)
    w[np.isinf(w)] = np.finfo(np.float64).max
    a = w / w.sum()
    return a


def left_heavy_power_family(n: int, beta: float) -> np.ndarray:
    """Monotone decreasing power‑law: a_i ∝ (i+½)^(‑β)."""
    i = np.arange(n, dtype=np.float64)
    w = np.exp(-beta * np.log(i + 0.5))
    a = w / w.sum()
    return a


def exponential_decay_family(n: int, gamma: float) -> np.ndarray:
    """Simple exponential decay: a_i ∝ exp(‑γ·i)."""
    i = np.arange(n, dtype=np.float64)
    w = np.exp(-gamma * i)
    a = w / w.sum()
    return a


def beta_distribution_family(n: int, alpha: float) -> np.ndarray:
    """Symmetric beta‑shape: a_i ∝ (i/(n‑1))^α·(1‑i/(n‑1))^α."""
    i = np.arange(n, dtype=np.float64)
    t = i / (n - 1)
    eps = 1e-12
    w = (t + eps) ** alpha * (1 - t + eps) ** alpha
    a = w / w.sum()
    return a


def gaussian_family(n: int, sigma: float) -> np.ndarray:
    """Discrete Gaussian centred in the middle."""
    i = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    w = np.exp(-0.5 * ((i - centre) / sigma) ** 2)
    a = w / w.sum()
    return a


def random_sparse_start(n: int) -> np.ndarray:
    """Sparse probability vector (≈5 % non‑zeros)."""
    sparsity = np.random.uniform(0.03, 0.07)
    k = max(2, int(round(sparsity * n)))
    idx = np.random.choice(n, size=k, replace=False)
    a = np.zeros(n, dtype=np.float64)
    a[idx] = 1.0 / k
    return project_to_simplex(a)


def random_monotone_start(n: int) -> np.ndarray:
    """Random decreasing vector (sorted power‑transformed uniform)."""
    r = np.random.rand(n) ** np.random.uniform(0.5, 5.0)
    r.sort()
    a = r[::-1]               # decreasing
    a = a / a.sum()
    return a


# ----------------------------------------------------------------------
# Smooth surrogate utilities (log‑sum‑exp of convolution)
# ----------------------------------------------------------------------
def _smooth_obj_and_grad(z: np.ndarray, beta: float, n: int):
    """
    Returns (f, grad) where
        f   = (1/β)·log∑_k exp[β·(a*a)_k]
        grad = ∂f/∂z   (z are the log‑soft‑max parameters)
    """
    a = softmax(z)                     # a ∈ simplex, sum = 1
    conv = convolution_fft(a, a)       # length 2n‑1
    maxc = conv.max()
    # stable log‑sum‑exp
    exp_terms = np.exp(beta * (conv - maxc))
    sum_exp = exp_terms.sum()
    f = (np.log(sum_exp) + beta * maxc) / beta

    # gradient wrt a:   ∂f/∂a_i = 2 * Σ_k w_k * a_{k-i}
    w = exp_terms / sum_exp            # shape (2n‑1,)
    a_rev = a[::-1]
    conv_rev = convolution_fft(a_rev, w)   # length 3n‑2
    grad_a = 2.0 * conv_rev[n - 1 : 2 * n - 1]

    # chain rule through soft‑max
    dot_ag = np.dot(a, grad_a)
    grad_z = a * (grad_a - dot_ag)
    return float(f), grad_z.astype(np.float64)


def l_bfgs_refine(start_a: np.ndarray, deadline: float,
                  beta_list=(200.0, 800.0, 2000.0, 5000.0),
                  max_iter_per_beta=1000) -> tuple[np.ndarray, float]:
    """Refine `start_a` by L‑BFGS‑B on the log‑sum‑exp surrogate."""
    if not _HAS_SCIPY:          # pragma: no cover
        return start_a, compute_score(start_a)

    best_a = start_a.copy()
    best_score = compute_score(best_a)
    eps = 1e-12
    z = np.log(best_a + eps)

    for beta in beta_list:
        if time.time() >= deadline - 5.0:
            break
        remaining = deadline - time.time()
        max_iter = min(max_iter_per_beta, int(remaining * 15))

        def func(z_vec):
            f, g = _smooth_obj_and_grad(z_vec, beta, best_a.shape[0])
            return f, g

        try:
            opt_res = _opt.fmin_l_bfgs_b(func,
                                         z,
                                         maxiter=max_iter,
                                         disp=False,
                                         factr=1e7,
                                         pgtol=1e-12)
        except Exception as exc:  # pragma: no cover
            print(f"[{time.time():.1f}] L‑BFGS failed at β={beta}: {exc}")
            continue

        z = opt_res[0]
        a_opt = softmax(z)
        sc_opt = compute_score(a_opt)
        if sc_opt < best_score:
            best_score = sc_opt
            best_a = a_opt.copy()
            print(f"[{time.time():.1f}] L‑BFGS β={beta:.0f} improved score = {best_score:.6f}")

    return best_a, best_score


def adam_optimize(start_a: np.ndarray, deadline: float,
                  beta_list=(200.0, 800.0, 2000.0, 5000.0),
                  lr: float = 0.07,
                  max_iter_per_beta: int = 800) -> tuple[np.ndarray, float]:
    """
    Adam optimiser on the same smoothed surrogate.
    Small learning‑rate works well because the surface is relatively flat.
    """
    n = start_a.shape[0]
    eps = 1e-12
    z = np.log(start_a + eps)
    m = np.zeros_like(z)
    v = np.zeros_like(z)
    beta1 = 0.9
    beta2 = 0.999
    t = 0
    best_a = start_a.copy()
    best_score = compute_score(best_a)

    for beta in beta_list:
        if time.time() >= deadline - 2.0:
            break
        max_iter = min(max_iter_per_beta,
                       int((deadline - time.time()) * 12))
        for _ in range(max_iter):
            if time.time() >= deadline - 1.0:
                break
            t += 1
            f, grad = _smooth_obj_and_grad(z, beta, n)
            m = beta1 * m + (1.0 - beta1) * grad
            v = beta2 * v + (1.0 - beta2) * (grad ** 2)
            m_hat = m / (1.0 - beta1 ** t)
            v_hat = v / (1.0 - beta2 ** t)
            z -= lr * m_hat / (np.sqrt(v_hat) + 1e-8)

            if t % 200 == 0:
                a = softmax(z)
                sc = compute_score(a)
                if sc < best_score:
                    best_score = sc
                    best_a = a.copy()
    return best_a, best_score


# ----------------------------------------------------------------------
# Mirror (exponentiated) sub‑gradient descent on the true max‑objective
# ----------------------------------------------------------------------
def mirror_subgrad_optimize(start_a: np.ndarray, deadline: float,
                           step0: float = 0.8,
                           check_interval: int = 2500) -> tuple[np.ndarray, float]:
    """Multiplicative sub‑gradient descent on F(a)=2·n·max(a*a)."""
    n = start_a.shape[0]
    a = start_a.astype(np.float64).copy()
    best_a = a.copy()
    best_score = compute_score(a)

    t = 0
    while time.time() < deadline:
        t += 1
        conv = convolution_fft(a, a)
        k = int(np.argmax(conv))

        # gradient of active convolution entry
        idx = np.arange(n)
        j = k - idx
        mask = (j >= 0) & (j < n)
        g = np.zeros_like(a)
        g[mask] = 2.0 * a[j[mask]]

        eta = step0 / math.sqrt(t)
        a = a * np.exp(-eta * g)        # multiplicative update

        # project onto simplex (entropy projection)
        s = a.sum()
        if s == 0.0:
            a = np.full_like(a, 1.0 / n)
        else:
            a = a / s

        if t % check_interval == 0:
            sc = compute_score(a)
            if sc < best_score:
                best_score = sc
                best_a = a.copy()
                print(f"[{time.time():.1f}] mirror sub‑grad iter {t} best = {best_score:.6f}")

    # final check
    sc = compute_score(a)
    if sc < best_score:
        best_score = sc
        best_a = a.copy()
    return best_a, best_score


# ----------------------------------------------------------------------
# Low‑pass smoothing in the Fourier domain
# ----------------------------------------------------------------------
def low_pass_smooth(a: np.ndarray, cutoff_ratio: float) -> np.ndarray:
    """
    Zero‑out frequencies above ``cutoff_ratio`` of the Nyquist.
    Afterwards project back onto the simplex.
    """
    n = a.shape[0]
    A = np.fft.rfft(a)
    cutoff = int(cutoff_ratio * A.shape[0])
    A[cutoff:] = 0
    a_s = np.fft.irfft(A, n=n)
    a_s = np.maximum(a_s, 0.0)
    a_s = project_to_simplex(a_s)
    return a_s


# ----------------------------------------------------------------------
# Tiny simulated‑annealing hill‑climb
# ----------------------------------------------------------------------
def hill_climb_sa(start_a: np.ndarray, deadline: float,
                  max_steps: int = 1_500_000,
                  init_temp_factor: float = 1e-4) -> tuple[np.ndarray, float]:
    """
    Very low‑temperature random walk on the simplex.
    """
    n = start_a.size
    cur_a = start_a.copy()
    cur_score = compute_score(cur_a)

    best_a = cur_a.copy()
    best_score = cur_score

    T0 = best_score * init_temp_factor
    T = T0
    step_scale = 0.12

    for step in range(1, max_steps + 1):
        if time.time() >= deadline:
            break

        cand = cur_a.copy()
        if np.random.rand() < 0.5:          # swap two entries
            i, j = np.random.randint(n, size=2)
            cand[i], cand[j] = cand[j], cand[i]
        else:                               # transfer a fraction of mass
            i, j = np.random.randint(n, size=2)
            if i != j:
                delta = step_scale * cur_a[i]
                cand[i] -= delta
                cand[j] += delta
                if cand[i] < 0.0:
                    cand[i] = 0.0

        cand = cand / cand.sum()
        cand_score = compute_score(cand)
        delta = cand_score - cur_score

        if delta < 0.0 or np.random.rand() < np.exp(-delta / (T + 1e-18)):
            cur_a = cand
            cur_score = cand_score
            if cur_score < best_score:
                best_score = cur_score
                best_a = cur_a.copy()
                step_scale = min(step_scale * 1.07, 0.5)
        else:
            step_scale *= 0.97

        T = T0 * (1.0 - step / max_steps)

    return best_a, best_score


# ----------------------------------------------------------------------
# Main driver
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 12345, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative sequence minimising `evaluate_sequence`.
    Returns the sequence as a plain Python list of floats.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 1.5   # tiny safety margin

    # --------------------------------------------------------------
    # Choose a sensible length (larger n gives finer resolution)
    # --------------------------------------------------------------
    if budget_s >= 900:
        n = 16384
    elif budget_s >= 600:
        n = 8192
    elif budget_s >= 300:
        n = 4096
    else:
        n = 2048
    print(f"[{time.time():.1f}] Using length n = {n}")

    best_seq = None
    best_score = float("inf")

    # --------------------------------------------------------------
    # 0.  Optional fallback to a known construction
    # --------------------------------------------------------------
    try:
        fb_vec = [float(x) for x in height_sequence_1]    # type: ignore
        fb = evaluate_sequence(fb_vec)
        print(f"[{time.time():.1f}] fallback score = {fb:.6f}")
        if fb < best_score:
            best_score = fb
            best_seq = fb_vec
    except Exception:
        pass

    # --------------------------------------------------------------
    # 1. Scan analytic families (cheap dense grid)
    # --------------------------------------------------------------
    # helper to test a candidate
    def try_candidate(a: np.ndarray, tag: str):
        nonlocal best_seq, best_score
        a = project_to_simplex(a)
        sc = compute_score(a)
        if sc < best_score:
            best_score = sc
            best_seq = a.tolist()
            print(f"[{time.time():.1f}] new best ({tag}) = {best_score:.6f}")

    # symmetric product / U‑shape
    for alpha in np.linspace(0.0, 20.0, 101):
        if time.time() >= deadline - 40.0:
            break
        try_candidate(symmetric_product_family(n, alpha), f"U‑α={alpha:.2f}")

    # left‑heavy power‑law
    for beta in np.linspace(0.0, 10.0, 81):
        if time.time() >= deadline - 40.0:
            break
        try_candidate(left_heavy_power_family(n, beta), f"P‑β={beta:.2f}")

    # exponential decay
    for gamma in np.linspace(0.0, 10.0, 81):
        if time.time() >= deadline - 40.0:
            break
        try_candidate(exponential_decay_family(n, gamma), f"E‑γ={gamma:.2f}")

    # symmetric beta distribution
    for alpha in np.linspace(0.0, 10.0, 81):
        if time.time() >= deadline - 40.0:
            break
        try_candidate(beta_distribution_family(n, alpha), f"B‑α={alpha:.2f}")

    # discrete Gaussian (different standard deviations)
    for sigma in (n / 12.0, n / 9.0, n / 6.0, n / 4.0):
        if time.time() >= deadline - 40.0:
            break
        try_candidate(gaussian_family(n, sigma), f"G‑σ={sigma:.1f}")

    # random starts (monotone and sparse)
    for _ in range(120):
        if time.time() >= deadline - 40.0:
            break
        try_candidate(random_monotone_start(n), "rand‑mono")
        try_candidate(random_sparse_start(n), "rand‑sparse")

    # --------------------------------------------------------------
    # 2. Refine the current best with L‑BFGS on the smoothed surrogate
    # --------------------------------------------------------------
    if best_seq is not None and time.time() < deadline - 250.0:
        print(f"[{time.time():.1f}] Starting L‑BFGS refinement")
        a_start = np.array(best_seq, dtype=np.float64)
        a_refined, sc_refined = l_bfgs_refine(
            a_start,
            deadline - 150.0,
            beta_list=(200.0, 800.0, 2000.0, 5000.0),
            max_iter_per_beta=1200,
        )
        if sc_refined < best_score:
            best_score = sc_refined
            best_seq = a_refined.tolist()
            print(f"[{time.time():.1f}] after L‑BFGS best = {best_score:.6f}")

    # --------------------------------------------------------------
    # 3. Additional Adam optimisation on the same surrogate
    # --------------------------------------------------------------
    if best_seq is not None and time.time() < deadline - 200.0:
        print(f"[{time.time():.1f}] Adam optimiser on surrogate")
        a_start = np.array(best_seq, dtype=np.float64)
        a_adam, sc_adam = adam_optimize(
            a_start,
            deadline - 120.0,
            beta_list=(200.0, 800.0, 2000.0, 5000.0),
            lr=0.07,
            max_iter_per_beta=1000,
        )
        if sc_adam < best_score:
            best_score = sc_adam
            best_seq = a_adam.tolist()
            print(f"[{time.time():.1f}] after Adam best = {best_score:.6f}")

    # --------------------------------------------------------------
    # 4. Mirror sub‑gradient descent on the exact max‑objective
    # --------------------------------------------------------------
    if best_seq is not None and time.time() < deadline - 80.0:
        print(f"[{time.time():.1f}] Mirror sub‑gradient descent")
        a_sub, sc_sub = mirror_subgrad_optimize(
            np.array(best_seq, dtype=np.float64),
            deadline - 30.0,
            step0=0.8,
        )
        if sc_sub < best_score:
            best_score = sc_sub
            best_seq = a_sub.tolist()
            print(f"[{time.time():.1f}] after mirror sub‑grad best = {best_score:.6f}")

    # --------------------------------------------------------------
    # 5. Low‑pass smoothing (try several cut‑offs)
    # --------------------------------------------------------------
    if best_seq is not None and time.time() < deadline - 20.0:
        a_base = np.array(best_seq, dtype=np.float64)
        for ratio in (0.20, 0.30, 0.40, 0.50):
            if time.time() >= deadline - 5.0:
                break
            a_s = low_pass_smooth(a_base, ratio)
            sc = compute_score(a_s)
            if sc < best_score:
                best_score = sc
                best_seq = a_s.tolist()
                print(f"[{time.time():.1f}] low‑pass ratio={ratio:.2f} improved = {best_score:.6f}")

    # --------------------------------------------------------------
    # 6. Symmetrisation (most solutions are symmetric)
    # --------------------------------------------------------------
    if best_seq is not None:
        a_arr = np.array(best_seq, dtype=np.float64)
        a_sym = (a_arr + a_arr[::-1]) * 0.5
        a_sym = project_to_simplex(a_sym)
        sc_sym = compute_score(a_sym)
        if sc_sym < best_score:
            best_score = sc_sym
            best_seq = a_sym.tolist()
            print(f"[{time.time():.1f}] symmetrisation improved best = {best_score:.6f}")

    # --------------------------------------------------------------
    # 7. Final tiny simulated‑annealing hill‑climb
    # --------------------------------------------------------------
    if best_seq is not None and time.time() < deadline - 5.0:
        print(f"[{time.time():.1f}] Final hill‑climb (SA)")
        a_final, sc_final = hill_climb_sa(
            np.array(best_seq, dtype=np.float64),
            deadline,
            max_steps=1_500_000,
        )
        if sc_final < best_score:
            best_score = sc_final
            best_seq = a_final.tolist()
            print(f"[{time.time():.1f}] hill‑climb improved best = {best_score:.6f}")

    # --------------------------------------------------------------
    # 8. Safeguard – fallback to a simple uniform vector if needed
    # --------------------------------------------------------------
    if best_seq is None:
        n_small = 256
        best_seq = [1.0 / n_small] * n_small
        best_score = evaluate_sequence(best_seq)

    # --------------------------------------------------------------
    # Final official evaluation (the harness will also call evaluate_sequence)
    # --------------------------------------------------------------
    final_score = evaluate_sequence(best_seq)
    print(f"\n[{time.time():.1f}] finished search – final reported score = {final_score:.6f}")

    return best_seq
