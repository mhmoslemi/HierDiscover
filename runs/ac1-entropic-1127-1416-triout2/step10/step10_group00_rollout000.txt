"""
Search for a low‑value step‑function sequence.

The core of the search is an *exponential‑gradient* (EG) descent on a
smoothed version of the max‑convolution objective.
For a probability vector a (∑a=1) we minimise the smooth max

    f_β(a) = (1/β)·log Σ_k exp(β·(a*a)_k)

with β growing from a small value to a large one (≈5000).  The gradient of
f_β is computed analytically via an FFT‑based convolution, and the EG
update

    a ← a ⊙ exp(−η·∇f_β)

is performed with a step‑size η that decays together with β.
Every few hundred iterations the iterate is symmetrised
(a ← (a+a[::-1])/2) and renormalised onto the simplex.

Several analytically‑motivated initialisations (product‑type, power‑type,
U‑shaped, sparse random, monotone random) as well as the fallback
`height_sequence_1` are tried; each is refined with the EG optimiser.
The best candidate is finally polished by a short simulated‑annealing
hill‑climb.

All work respects the 1000 s wall‑clock budget and returns a plain
Python list of non‑negative numbers.
"""
import time
import numpy as np
from typing import List, Tuple

# ----------------------------------------------------------------------
# Basic convolution utilities (full aperiodic convolution via real FFT)
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full aperiodic convolution using real FFT (real inputs)."""
    m = a.size + b.size - 1
    nfft = next_pow2(m)
    A = np.fft.rfft(a, nfft)
    B = np.fft.rfft(b, nfft)
    C = A * B
    conv = np.fft.irfft(C, nfft)[:m]
    return conv


def compute_score(a: np.ndarray) -> float:
    """
    Exact objective for a probability vector (∑a = 1):
        score = 2·n·max_k (a*a)_k .
    """
    n = a.shape[0]
    conv = convolution_fft(a, a)
    return 2.0 * n * float(conv.max())


# ----------------------------------------------------------------------
# Analytic families (used for initial guesses)
# ----------------------------------------------------------------------


def symmetric_product_family(n: int, alpha: float) -> np.ndarray:
    """a_i ∝ [(i+½)(n‑½‑i)]^{‑α}, symmetric, on the simplex."""
    i = np.arange(n, dtype=np.float64)
    logprod = np.log(i + 0.5) + np.log(n - 0.5 - i)
    w = np.exp(-alpha * logprod)
    w[np.isinf(w)] = np.finfo(np.float64).max
    w = np.maximum(w, 0.0)
    return w / w.sum()


def left_heavy_power_family(n: int, beta: float) -> np.ndarray:
    """a_i ∝ (i+½)^{‑β}, monotone decreasing."""
    i = np.arange(n, dtype=np.float64)
    w = np.exp(-beta * np.log(i + 0.5))
    w = np.maximum(w, 0.0)
    return w / w.sum()


def exp_u_family(n: int, gamma: float) -> np.ndarray:
    """U‑shape: a_i ∝ exp(-γ·min(i, n‑1‑i))."""
    i = np.arange(n, dtype=np.float64)
    dist = np.minimum(i, n - 1 - i)
    w = np.exp(-gamma * dist)
    w = np.maximum(w, 0.0)
    return w / w.sum()


def random_sparse_start(n: int) -> np.ndarray:
    """Very sparse uniform‑mass vector (~5 % non‑zeros)."""
    sparsity = np.random.uniform(0.03, 0.07)
    k = max(2, int(round(sparsity * n)))
    idx = np.random.choice(n, size=k, replace=False)
    a = np.zeros(n, dtype=np.float64)
    a[idx] = 1.0 / k
    a = np.maximum(a, 0.0)
    return a / a.sum()


def random_monotone_start(n: int) -> np.ndarray:
    """Decreasing vector obtained by sorting a powered uniform."""
    r = np.random.rand(n) ** np.random.uniform(0.5, 5.0)
    r.sort()
    a = r[::-1]
    a = np.maximum(a, 0.0)
    return a / a.sum()


# ----------------------------------------------------------------------
# Exponential‑gradient optimiser on the smoothed max‑convolution
# ----------------------------------------------------------------------


def eg_smooth_optimize(start_a: np.ndarray,
                       deadline: float,
                       max_iters: int = 150_000,
                       beta_start: float = 1.0,
                       beta_end: float = 5000.0,
                       eta_start: float = 0.05,
                       eta_end: float = 0.001,
                       sym_interval: int = 4000) -> Tuple[np.ndarray, float]:
    """
    Minimise the smooth max‑convolution using exponential‑gradient updates.
    a is kept on the simplex by normalisation after each step.
    """
    a = start_a.copy()
    n = a.shape[0]
    eps = 1e-12

    best_a = a.copy()
    best_score = compute_score(a)

    for it in range(1, max_iters + 1):
        if time.time() >= deadline:
            break

        t = it / max_iters
        beta = beta_start * (beta_end / beta_start) ** t
        eta = eta_start * (eta_end / eta_start) ** t

        # smoothed max (log‑sum‑exp) weights
        conv = convolution_fft(a, a)
        maxc = conv.max()
        exp_w = np.exp(beta * (conv - maxc))
        w = exp_w / exp_w.sum()

        # gradient of the smooth max w.r.t. a
        a_rev = a[::-1]
        conv_rev = convolution_fft(a_rev, w)
        grad = 2.0 * conv_rev[n - 1:n - 1 + n]

        # exponential‑gradient step
        a = a * np.exp(-eta * grad)
        a = np.maximum(a, eps)          # keep strictly positive
        a = a / a.sum()                 # projection onto simplex

        # occasional symmetrisation
        if sym_interval and (it % sym_interval == 0):
            a = (a + a[::-1]) * 0.5
            a = a / a.sum()

        # periodic exact evaluation
        if it % 2000 == 0:
            sc = compute_score(a)
            if sc < best_score:
                best_score = sc
                best_a = a.copy()
                print(f"[{time.time():.1f}] EG improved to {best_score:.6f} (iter {it})")

    # final evaluation
    sc = compute_score(a)
    if sc < best_score:
        best_score = sc
        best_a = a.copy()

    return best_a, best_score


# ----------------------------------------------------------------------
# Simulated‑annealing hill‑climb (lightweight final polishing)
# ----------------------------------------------------------------------


def hill_climb_sa(start_a: np.ndarray,
                  deadline: float,
                  max_steps: int = 2_000_000) -> Tuple[np.ndarray, float]:
    """
    Simple SA on the simplex:
      – 50 % swap two entries,
      – 50 % transfer a fraction of mass.
    Temperature decays linearly.
    """
    n = start_a.size
    cur_a = start_a.copy()
    cur_score = compute_score(cur_a)

    best_a = cur_a.copy()
    best_score = cur_score

    T0 = best_score * 1e-4
    T = T0
    step_scale = 0.12

    for step in range(1, max_steps + 1):
        if time.time() >= deadline:
            break

        cand = cur_a.copy()
        if np.random.rand() < 0.5:            # swap
            i, j = np.random.randint(n, size=2)
            cand[i], cand[j] = cand[j], cand[i]
        else:                                 # mass transfer
            i, j = np.random.randint(n, size=2)
            if i != j:
                delta = step_scale * cur_a[i]
                cand[i] -= delta
                cand[j] += delta
                if cand[i] < 0.0:
                    cand[i] = 0.0

        # project back to simplex (renormalise)
        cand = np.maximum(cand, 0.0)
        cand_sum = cand.sum()
        if cand_sum == 0.0:
            continue
        cand = cand / cand_sum

        cand_score = compute_score(cand)

        delta = cand_score - cur_score
        if delta < 0 or np.random.rand() < np.exp(-delta / (T + 1e-18)):
            cur_a = cand
            cur_score = cand_score
            if cur_score < best_score:
                best_score = cur_score
                best_a = cur_a.copy()
                step_scale = min(step_scale * 1.07, 0.5)
        else:
            step_scale *= 0.97

        # linear cooling
        T = T0 * (1.0 - step / max_steps)

    return best_a, best_score
