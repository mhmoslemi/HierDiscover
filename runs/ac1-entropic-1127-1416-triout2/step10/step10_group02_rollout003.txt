    a = a0.copy()

    n = a.shape[0]

    L = 1 << ((2*n - 2).bit_length())  # next power of 2 >= 2n-1

    m = np.zeros_like(a)  # first moment

    v = np.zeros_like(a)  # second moment

    best_a = a.copy(); best_val = compute_score(a)

    t = 0

    start = time.time()

    while time.time() < deadline:

        t += 1

        # conv
        A_fft = np.fft.rfft(a, n=L)
        # conv via elementwise square
        conv_fft = A_fft * A_fft
        conv = np.fft.irfft(conv_fft, n=L)[:2*n-1]

        # lambda schedule
        elapsed_frac = (time.time() - start) / (deadline - start)
        lam = lambda_start + (lambda_end - lambda_start) * elapsed_frac

        # softmax weights (stable)
        conv_shifted = conv - conv.max()
        w = np.exp(lam * conv_shifted)
        w_sum = w.sum()
        if w_sum == 0 or np.isnan(w_sum):
            # fallback to uniform weights (should not happen)
            w = np.full_like(w, 1.0 / w.shape[0])
        else:
            w /= w_sum

        # Compute using FFT: cross_corr = ifft(fft(a) * conj(fft(w)))

        beta2 = 0.999

   a = project_to_simplex(a0.astype(float))

   t = 0

   start = time.time()

        t += 1

        # compute conv via FFT
        A_fft = np.fft.rfft(a, n=L)
        conv_fft = A_fft * A_fft

        beta2 = 0.999

"""
Non‑negative step‑function optimiser for ``evaluate_sequence``.

Approach
--------
1.  Initialise a reasonable candidate by a power‑law sweep.
2.  Refine it using a smooth approximation of the ‑∞‑norm:
    minimise   f(a)= (1/λ)·log Σ_k exp(λ·(a⋆a)_k)   (⋆ = linear convolution)
    The gradient of f is a weighted sum of shifted copies of a, which we
    compute efficiently with FFTs.  An Adam optimiser drives the updates,
    while the iterate is projected onto the probability simplex after each
    step.  λ is increased slowly from a small value to a large one, so the
    optimisation first sees a smooth landscape and then focuses on the true
    maximum.
3.  A short projected sub‑gradient descent (as in the baseline) cleans up
    any remaining bias.
4.  Finally a cheap greedy hill‑climb performs stochastic mass‑transfer
    moves for a last polish.

All phases respect the global 1000 s wall‑clock budget.  The best sequence
found is returned as a plain list of floats.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Utility functions
# ----------------------------------------------------------------------
def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of a vector onto the probability simplex."""
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / v.shape[0])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, v.shape[0] + 1)
    mask = u - cssv / ind > 0
    rho = np.where(mask)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def full_convolution(a: np.ndarray) -> np.ndarray:
    """
    Full linear convolution a*a using a real FFT.
    Length of the result is 2*len(a)-1.
    """
    n = a.shape[0]
    m = 2 * n - 1
    n_fft = 1 << ((m - 1).bit_length())          # next power of two ≥ m
    A = np.fft.rfft(a, n=n_fft)
    C = A * A
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


def compute_score(a: np.ndarray) -> float:
    """Exact objective for a normalised vector (sum = 1):
       2·n·max(conv(a,a)).
    """
    n = a.shape[0]
    s = a.sum()
    if s < 1e-12:
        return float("inf")
    conv = full_convolution(a)
    return float(2.0 * n * conv.max() / (s * s))


def resample_sequence(seq: np.ndarray, target_len: int) -> np.ndarray:
    """Linear interpolation to change the length of an existing sequence."""
    old_len = seq.shape[0]
    if old_len == target_len:
        return seq.copy()
    old_x = np.linspace(0.0, 1.0, old_len)
    new_x = np.linspace(0.0, 1.0, target_len)
    new_seq = np.interp(new_x, old_x, seq)
    new_seq = np.maximum(new_seq, 0.0)
    new_seq = project_to_simplex(new_seq)
    return new_seq


# ----------------------------------------------------------------------
# 1. Power‑law sweep (cheap global search)
# ----------------------------------------------------------------------
def best_power_law_vector(n: int, deadline: float) -> np.ndarray:
    """Search over a_i ∝ (i+1)^{-p} for p∈[0.5,0.7] and return the best."""
    best_val = float("inf")
    best_a = None
    best_p = 0.5

    # coarse grid
    for p in np.linspace(0.5, 0.7, 41):
        if time.time() >= deadline:
            break
        a = np.arange(1, n + 1, dtype=float) ** (-p)
        a /= a.sum()
        val = compute_score(a)
        if val < best_val:
            best_val, best_a, best_p = val, a, p

    # fine grid around the best p
    if best_a is not None:
        low = max(0.5, best_p - 0.03)
        high = min(0.8, best_p + 0.03)
        for p in np.linspace(low, high, 31):
            if time.time() >= deadline:
                break
            a = np.arange(1, n + 1, dtype=float) ** (-p)
            a /= a.sum()
            val = compute_score(a)
            if val < best_val:
                best_val, best_a, best_p = val, a, p

    return best_a


# ----------------------------------------------------------------------
# 2. Smooth‑max Adam optimisation
# ----------------------------------------------------------------------
def smooth_adam_descent(a0: np.ndarray,
                        deadline: float,
                        lambda_start: float = 5.0,
                        lambda_end: float = 200.0,
                        init_lr: float = 0.05) -> np.ndarray:
    """
    Minimise a smooth approximation of max(conv(a,a)) using Adam.
    The smoothing parameter λ is increased linearly from ``lambda_start``
    to ``lambda_end`` during the run.
    """
    a = project_to_simplex(a0.astype(float).copy())
    n = a.shape[0]
    # FFT length (next power of two ≥ 2n‑1)
    L = 1 << ((2 * n - 2).bit_length())
    # Adam state
    m = np.zeros_like(a)
    v = np.zeros_like(a)

    best_a = a.copy()
    best_val = compute_score(a)

    t = 0
    start = time.time()

    while time.time() < deadline:
        t += 1

        # ----- convolution (full) -----
        A_fft = np.fft.rfft(a, n=L)
        conv_fft = A_fft * A_fft
        conv = np.fft.irfft(conv_fft, n=L)[: 2 * n - 1]

        # ----- λ schedule -----
        frac = (time.time() - start) / (deadline - start)
        lam = lambda_start + (lambda_end - lambda_start) * frac

        # ----- soft‑max weights -----
        max_c = conv.max()
        exps = np.exp(lam * (conv - max_c))      # subtract max for stability
        w = exps / exps.sum()

        # ----- gradient (cross‑correlation a ⋆ w) -----
        W_fft = np.fft.rfft(w, n=L)
        cross = np.fft.irfft(A_fft * np.conj(W_fft), n=L)[:n]
        grad = 2.0 * cross    # ∇_a f = Σ_k p_k * 2·shift(a,k)

        # ----- Adam update -----
        beta1, beta2, eps = 0.9, 0.999, 1e-8
        lr = init_lr / np.sqrt(t)

        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)

        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)

        a = a - lr * m_hat / (np.sqrt(v_hat) + eps)

        # ----- projection onto the simplex -----
        a = project_to_simplex(a)

        # ----- occasional jitter to escape plateaus -----
        if t % 2000 == 0:
            a += np.random.randn(n) * (init_lr * 0.01 / np.sqrt(t))
            a = project_to_simplex(a)

        # ----- track best -----
        cur_val = compute_score(a)
        if cur_val < best_val:
            best_val = cur_val
            best_a = a.copy()
            if t % 500 == 0:
                print(f"[smooth_adam] iter {t:6d}, λ={lam:6.1f}, score={cur_val:.6f}")

    return best_a


# ----------------------------------------------------------------------
# 3. Projected sub‑gradient polishing (as in the baseline)
# ----------------------------------------------------------------------
def subgradient_polish(a0: np.ndarray,
                       deadline: float,
                       step0: float = 0.12) -> np.ndarray:
    """Projected sub‑gradient descent on the exact L∞ objective."""
    a = a0.copy()
    n = a.shape[0]
    best_a = a.copy()
    best_val = compute_score(a)

    t = 0
    while time.time() < deadline:
        t += 1
        conv = full_convolution(a)
        k = int(np.argmax(conv))

        # sub‑gradient of the active entry (2 * shifted copy of a)
        idx = np.arange(n)
        j = k - idx
        mask = (j >= 0) & (j < n)
        g = np.zeros_like(a)
        g[mask] = 2.0 * a[j[mask]]

        step = step0 / np.sqrt(t)
        a = a - step * g
        a = project_to_simplex(a)

        val = compute_score(a)
        if val < best_val:
            best_val, best_a = val, a.copy()

        if t % 1000 == 0:
            jitter = np.random.randn(n) * (step0 * 0.01 / np.sqrt(t))
            a += jitter
            a = project_to_simplex(a)

    return best_a


# ----------------------------------------------------------------------
# 4. Greedy hill‑climb (tiny stochastic moves)
# ----------------------------------------------------------------------
def hill_climb(a_start: np.ndarray,
               deadline: float,
               max_no_improve: int = 500_000) -> np.ndarray:
    """Random mass‑transfer moves with greedy acceptance."""
    a = a_start.copy()
    n = a.shape[0]
    best_a = a.copy()
    best_val = compute_score(a)

    step_scale = 0.05
    it = 0
    no_improve = 0

    while time.time() < deadline and it < max_no_improve:
        i, j = np.random.randint(n, size=2)
        if i == j:
            continue

        delta = step_scale * a[i]
        cand = a.copy()
        cand[i] -= delta
        cand[j] += delta
        cand = np.maximum(cand, 0.0)
        cand = project_to_simplex(cand)

        val = compute_score(cand)
        if val < best_val:
            best_val, best_a = val, cand.copy()
            a = cand
            step_scale = min(step_scale * 1.07, 0.5)
            no_improve = 0
        else:
            step_scale *= 0.985
            no_improve += 1

        it += 1

    return best_a


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 12345,
                     budget_s: int = 1000,
                     **kwargs):
    """
    Search for a non‑negative coefficient sequence that minimises
    ``evaluate_sequence``.  Returns a plain Python list of floats.
    """
    np.random.seed(seed)
    start_time = time.time()
    # small safety margin for the harness bookkeeping
    deadline = start_time + budget_s - 5.0

    # ------------------------------------------------------------------
    # Choose a length – a larger n gives more flexibility.
    # ------------------------------------------------------------------
    n = 4096

    best_seq = None
    best_val = float("inf")

    # ------------------------------------------------------------------
    # 0. Optional fallback from a previously discovered construction
    # ------------------------------------------------------------------
    try:
        fallback_raw = np.array(height_sequence_1, dtype=float)  # type: ignore
        fallback = resample_sequence(fallback_raw, n)
        val = evaluate_sequence(fallback.tolist())
        print(f"[fallback] score = {val:.6f}")
        if val < best_val:
            best_val, best_seq = val, fallback.tolist()
    except Exception:
        pass

    # ------------------------------------------------------------------
    # 1. Power‑law sweep (fast global exploration)
    # ------------------------------------------------------------------
    pll_deadline = deadline - 300.0          # reserve most time for polishing
    if time.time() < pll_deadline:
        a_pow = best_power_law_vector(n, pll_deadline)
        if a_pow is not None:
            val = compute_score(a_pow)
            print(f"[power sweep] best p‑score = {val:.6f}")
            if val < best_val:
                best_val, best_seq = val, a_pow.tolist()

    # ------------------------------------------------------------------
    # 2. Smooth‑max Adam optimisation (main refinement)
    # ------------------------------------------------------------------
    adam_deadline = deadline - 120.0
    if best_seq is not None and time.time() < adam_deadline:
        a0 = np.array(best_seq, dtype=float)
        a_refined = smooth_adam_descent(a0, adam_deadline)
        val = compute_score(a_refined)
        print(f"[smooth_adam] refined score = {val:.6f}")
        if val < best_val:
            best_val, best_seq = val, a_refined.tolist()

    # ------------------------------------------------------------------
    # 3. Sub‑gradient polishing (exact max reduction)
    # ------------------------------------------------------------------
    sub_deadline = deadline - 30.0
    if best_seq is not None and time.time() < sub_deadline:
        a0 = np.array(best_seq, dtype=float)
        a_refined = subgradient_polish(a0, sub_deadline)
        val = compute_score(a_refined)
        print(f"[sub‑grad] refined score = {val:.6f}")
        if val < best_val:
            best_val, best_seq = val, a_refined.tolist()

    # ------------------------------------------------------------------
    # 4. Final greedy hill‑climb
    # ------------------------------------------------------------------
    hill_deadline = deadline - 5.0
    if best_seq is not None and time.time() < hill_deadline:
        a_start = np.array(best_seq, dtype=float)
        a_final = hill_climb(a_start, hill_deadline)
        val = compute_score(a_final)
        print(f"[hill‑climb] final score = {val:.6f}")
        if val < best_val:
            best_val, best_seq = val, a_final.tolist()

    # ------------------------------------------------------------------
    # Safeguard – fall back to a uniform vector if nothing was found
    # ------------------------------------------------------------------
    if best_seq is None:
        best_seq = [1.0 / n] * n
        best_val = evaluate_sequence(best_seq)

    # ------------------------------------------------------------------
    # Final official evaluation
    # ------------------------------------------------------------------
    final_score = evaluate_sequence(best_seq)
    print(f"\nSearch finished. Reported score = {final_score:.6f}")
    return best_seq
