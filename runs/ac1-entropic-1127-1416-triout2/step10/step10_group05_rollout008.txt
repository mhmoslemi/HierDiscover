def smooth_max_optimize(a0, deadline, max_iters=500000, beta_start=0.5, beta_end=20000.0, eta0=0.8):
    a = np.maximum(a0.copy(), 1e-15)
    a /= a.sum()
    n = a.shape[0]
    best_a, best_sc = a.copy(), compute_score(a)

    for it in range(1, max_iters+1):
        if time.time()>=deadline:
            break
        # calculate schedule
        t = it / max_iters
        beta = beta_start * (beta_end / beta_start) ** t
        conv = convolution_fft(a, a)
        maxc = conv.max()
        # compute softmax weights
        # to avoid overflow, subtract maxc
        expw = np.exp(beta * (conv - maxc))
        w = expw / expw.sum()
        # gradient of smooth max: 2 * convolution of reversed a with w
        a_rev = a[::-1]
        grad = 2.0 * convolution_fft(a_rev, w)[n-1:n-1+n]  # length n proper slice

        eta = eta0 / np.sqrt(it)
        a = np.maximum(a * np.exp(-eta * grad), 1e-15)
        a /= a.sum()
        if it % 400 == 0:
            sc = compute_score(a)
            if sc < best_sc:
                best_sc, best_a = sc, a.copy()

"""
Search for a step‑function (non‑negative vector) that minimises
evaluate_sequence(sequence) = 2·n·max(a*a) / (sum a)^2.
The algorithm combines:

1.  A wide grid‑search over several low‑dimensional families
    (exponential, power‑law, Beta, Gaussian, sinusoid) to obtain a good
    initial candidate.
2.  A smooth‑max (log‑sum‑exp) descent using KL‑mirror updates with a
    slowly increasing temperature β.
3.  An Adam optimiser that works directly on the true max‑convolution
    (sub‑gradient of the active entry) while keeping the vector on the
    probability simplex via a soft‑max parametrisation.
4.  A final targeted hill‑climb that moves a small packet of mass away
    from the index that currently dominates the maximal convolution
    entry.

All phases respect the global wall‑clock budget (default 1000 s) and
continually keep track of the best vector seen.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Basic linear‑algebra utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full aperiodic convolution using a real FFT (zero‑padded)."""
    m = a.size + b.size - 1
    nfft = next_pow2(m)
    A = np.fft.rfft(a, n=nfft)
    B = np.fft.rfft(b, n=nfft)
    C = A * B
    return np.fft.irfft(C, n=nfft)[:m]


def compute_score(a: np.ndarray) -> float:
    """
    Exact objective used by `evaluate_sequence` for a vector with
    sum = 1. Returns 2·n·max(a*a) where the convolution is full.
    """
    n = a.shape[0]
    conv = convolution_fft(a, a)
    return 2.0 * n * float(conv.max())


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of v onto the probability simplex.
    """
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / v.shape[0])

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, v.shape[0] + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.where(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


# ----------------------------------------------------------------------
# Parametric families of candidate vectors
# ----------------------------------------------------------------------


def exponential_shape(n: int, lam: float) -> np.ndarray:
    """a_i ∝ exp(-lam·i) (lam≥0), normalised to sum = 1."""
    i = np.arange(n, dtype=float)
    if lam == 0.0:
        w = np.ones_like(i)
    else:
        w = np.exp(-lam * i)
    return w / w.sum()


def power_shape(n: int, p: float) -> np.ndarray:
    """a_i ∝ (i+1)^(-p) (p≥0), normalised to sum = 1."""
    i = np.arange(1, n + 1, dtype=float)
    if p == 0.0:
        w = np.ones_like(i)
    else:
        w = i ** (-p)
    return w / w.sum()


def beta_shape(n: int, a: float, b: float) -> np.ndarray:
    """
    Discrete Beta distribution on {0,…,n‑1}:
        a_i ∝ ((i+0.5)/n)^(a‑1)·(1‑(i+0.5)/n)^(b‑1)
    """
    x = (np.arange(n, dtype=float) + 0.5) / n
    w = np.power(x, a - 1.0) * np.power(1.0 - x, b - 1.0)
    w[np.isnan(w)] = 0.0
    w = np.maximum(w, 0.0)
    if w.sum() == 0.0:
        w = np.ones_like(w)
    return w / w.sum()


def gaussian_shape(n: int, sigma_frac: float) -> np.ndarray:
    """Discrete centred Gaussian, sigma = sigma_frac·n, normalised."""
    mu = (n - 1) / 2.0
    sigma = max(sigma_frac * n, 1e-9)
    i = np.arange(n, dtype=float)
    w = np.exp(-((i - mu) ** 2) / (2.0 * sigma ** 2))
    return w / w.sum()


def sinusoid_shape(n: int, power: float) -> np.ndarray:
    """a_i ∝ sin(π·(i+1)/(n+1))^power, normalised."""
    i = np.arange(1, n + 1, dtype=float)
    w = np.sin(np.pi * i / (n + 1)) ** power
    # Guard against the trivial all‑zero case
    if w.sum() == 0.0:
        w = np.ones_like(w)
    return w / w.sum()

# Optimisation phases

def smooth_max_optimize(start_a: np.ndarray,
                        deadline: float,
                        max_iters: int = 200_000,
                        beta_start: float = 0.5,
                        beta_end: float = 20_000.0,
                        eta0: float = 0.8) -> tuple[np.ndarray, float]:
    """
    Minimise the log‑sum‑exp surrogate of the convolution maximum.
    Updates a with a KL‑mirror step a ← a·exp(-η·∇) and re‑normalises.
    """
    a = np.maximum(start_a.astype(float), 1e-15)
    a /= a.sum()
    n = a.shape[0]

    best_a = a.copy()
    best_sc = compute_score(a)

    for it in range(1, max_iters + 1):
        if time.time() >= deadline:
            break

        # temperature schedule
        t = it / max_iters
        beta = beta_start * (beta_end / beta_start) ** t

        conv = convolution_fft(a, a)
        maxc = conv.max()

        # soft‑max weights (stable)
        expw = np.exp(beta * (conv - maxc))
        w = expw / expw.sum()

        # gradient of the smooth max (see derivation)
        a_rev = a[::-1]
        grad = 2.0 * convolution_fft(a_rev, w)[n - 1:n - 1 + n]

        eta = eta0 / np.sqrt(it)

        a = np.maximum(a * np.exp(-eta * grad), 1e-15)
        a /= a.sum()

        if it % 300 == 0:
            sc = compute_score(a)
            if sc < best_sc:
                best_sc, best_a = sc, a.copy()

    # final check
    sc = compute_score(a)
    if sc < best_sc:
        best_sc, best_a = sc, a.copy()
    return best_a, best_sc


def adam_max_optimize(start_a: np.ndarray,
                      deadline: float,
                      max_iters: int = 300_000,
                      alpha: float = 0.2,
                      beta1: float = 0.9,
                      beta2: float = 0.999,
                      eps: float = 1e-8) -> tuple[np.ndarray, float]:
    """
    Adam optimiser on the true max‑convolution (sub‑gradient).
    The vector is parametrised as a = softmax(b) to stay on the simplex.
    """
    n = start_a.shape[0]
    # initialise b = log a (with a small epsilon)
    b = np.log(np.maximum(start_a, 1e-15))
    m = np.zeros_like(b)
    v = np.zeros_like(b)

    # keep a running best
    a = np.exp(b - np.max(b))
    a /= a.sum()
    best_a = a.copy()
    best_sc = compute_score(best_a)

    for t in range(1, max_iters + 1):
        if time.time() >= deadline:
            break

        # current a from softmax
        b_shift = b - np.max(b)          # numerical stability
        a = np.exp(b_shift)
        a /= a.sum()

        # convolution and active index
        conv = convolution_fft(a, a)
        k_max = int(conv.argmax())

        # sub‑gradient of max(conv) w.r.t a
        grad_a = np.zeros_like(a)
        i_start = max(0, k_max - (n - 1))
        i_end = min(n - 1, k_max)
        idx = np.arange(i_start, i_end + 1)
        grad_a[idx] = 2.0 * a[k_max - idx]

        # convert to gradient w.r.t the log‑parameters (softmax Jacobian)
        mu = np.dot(a, grad_a)
        grad_b = a * (grad_a - mu)

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad_b
        v = beta2 * v + (1.0 - beta2) * (grad_b ** 2)

        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)

        b -= (alpha / (np.sqrt(v_hat) + eps)) * m_hat

        if t % 500 == 0:
            sc = compute_score(a)
            if sc < best_sc:
                best_sc, best_a = sc, a.copy()

    # final evaluation
    a = np.exp(b - np.max(b))
    a /= a.sum()
    sc = compute_score(a)
    if sc < best_sc:
        best_sc, best_a = sc, a.copy()
    return best_a, best_sc


def hill_climb_optimize(start_a: np.ndarray,
                        deadline: float,
                        max_steps: int = 400_000) -> tuple[np.ndarray, float]:
    """
    Greedy walk that moves a small packet of mass away from the
    index contributing most to the current maximal convolution entry.
    Only improving moves are kept.
    """
    a = start_a.copy()
    n = a.size
    best_a = a.copy()
    best_sc = compute_score(best_a)

    step_scale = 0.02
    steps = 0

    while steps < max_steps and time.time() < deadline:
        conv = convolution_fft(best_a, best_a)
        k_max = int(conv.argmax())

        i_start = max(0, k_max - (n - 1))
        i_end = min(n - 1, k_max)
        idx = np.arange(i_start, i_end + 1)
        contrib = best_a[idx] * best_a[k_max - idx]

        i_max = idx[np.argmax(contrib)]

        # random destination different from i_max
        j = np.random.randint(n)
        if j == i_max:
            continue

        delta = step_scale * best_a[i_max]
        if delta <= 0.0:
            break

        cand = best_a.copy()
        cand[i_max] -= delta
        cand[j] += delta
        cand = project_to_simplex(cand)

        sc = compute_score(cand)
        if sc < best_sc:
            best_sc, best_a = sc, cand.copy()
            step_scale = min(step_scale * 1.07, 0.5)
        else:
            step_scale *= 0.96

        steps += 1

    return best_a, best_sc

            print(f"[{time.time():.1f}] after Adam phase best = {best_score:.6f}")

    # -------------------------------------------------
    # 5. Targeted hill‑climb polishing
    # -------------------------------------------------
    if best_seq is not None and time.time() < deadline - 5.0:
        print(f"[{time.time():.1f}] final hill‑climb polishing")
        a_final, sc_final = hill_climb_optimize(
            np.array(best_seq, dtype=float),
            deadline,
            max_steps=300_000,
        )
        if sc_final < best_score:
            best_score, best_seq = sc_final, a_final.tolist()
            print(f"[{time.time():.1f}] hill‑climb improved best = {best_score:.6f}")

    # -------------------------------------------------
    # 6. Safeguard – uniform tiny vector if nothing succeeded
    # -------------------------------------------------
    if best_seq is None:
        n_small = 256
        uniform_small = [1.0 / n_small] * n_small
        best_seq = uniform_small
        best_score = evaluate_sequence(best_seq)

    # -------------------------------------------------
    # 7. Final official evaluation (the harness also calls it)
    # -------------------------------------------------
    final_score = evaluate_sequence(best_seq)
    print(f"\n[{time.time():.1f}] finished – final reported score = {final_score:.6f}")

    # Return a plain Python list
    return best_seq
