"""
Non‑negative step‑function optimiser for ``evaluate_sequence``.

Core idea
---------
*   Use a smooth approximation of the L∞‑norm (log‑sum‑exp) to obtain a
    differentiable surrogate of the objective 2·n·max(conv(a,a)).
*   Optimise the surrogate with Adam on the simplex (Euclidean projection
    after each update).  The smoothing parameter ε is exponentially
    decreased during the run – early iterations see a “soft” max,
    later ones focus on the true peak.
*   Initialise the search with a cheap power‑law sweep (the best vector
    found there is a good starting point).  When time permits a final
    cheap greedy hill‑climb is performed.

The whole routine respects the supplied time budget (default 1000 s) and
returns the best sequence encountered as a plain ``list`` of floats.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Numerical utilities
# ----------------------------------------------------------------------


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the probability simplex.
    Guarantees non‑negative components that sum to 1.
    """
    if np.all(v <= 0):
        # Uniform distribution is the projection of the zero vector.
        return np.full_like(v, 1.0 / v.shape[0])

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, v.shape[0] + 1)
    mask = u - cssv / ind > 0
    rho = np.where(mask)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def conv_fft(u: np.ndarray, v: np.ndarray) -> np.ndarray:
    """
    Linear convolution (full) of two real vectors using a real FFT.
    Length of the result is len(u)+len(v)-1.
    """
    m = len(u) + len(v) - 1
    n_fft = 1 << ((m - 1).bit_length())          # next power of two ≥ m
    U = np.fft.rfft(u, n=n_fft)
    V = np.fft.rfft(v, n=n_fft)
    conv = np.fft.irfft(U * V, n=n_fft)[:m]
    return conv


def compute_score(a: np.ndarray) -> float:
    """
    Exact objective for a normalised vector (sum = 1):
        2·n·max(conv(a,a))
    """
    n = a.shape[0]
    conv = conv_fft(a, a)                # length 2*n-1
    return float(2.0 * n * conv.max())


# ----------------------------------------------------------------------
# 1. Power‑law sweep (cheap global search)
# ----------------------------------------------------------------------


def best_power_law_vector(n: int, time_deadline: float) -> np.ndarray:
    """
    Scan p∈[0.5,0.8] (coarse then fine) and return the best normalised
    vector a_i ∝ (i+1)^(-p).
    """
    best_a = None
    best_val = float("inf")
    best_p = 0.5

    # coarse grid
    for p in np.linspace(0.5, 0.8, 61):          # step ≈0.005
        if time.time() >= time_deadline:
            break
        a = np.arange(1, n + 1, dtype=float) ** (-p)
        a /= a.sum()
        val = compute_score(a)
        if val < best_val:
            best_val, best_a, best_p = val, a, p

    # fine grid around the best p
    if best_a is not None:
        low = max(0.5, best_p - 0.02)
        high = min(0.9, best_p + 0.02)
        for p in np.linspace(low, high, 41):
            if time.time() >= time_deadline:
                break
            a = np.arange(1, n + 1, dtype=float) ** (-p)
            a /= a.sum()
            val = compute_score(a)
            if val < best_val:
                best_val, best_a, best_p = val, a, p

    return best_a


# ----------------------------------------------------------------------
# 2. Smoothed Adam optimisation (main workhorse)
# ----------------------------------------------------------------------


def smoothed_adam_optimize(
    a0: np.ndarray,
    deadline: float,
    lr0: float = 0.12,
    eps0: float = 0.02,
    eps_final: float = 1e-5,
    beta1: float = 0.9,
    beta2: float = 0.999,
) -> np.ndarray:
    """
    Minimise 2·n·max(conv(a,a)) using a smooth log‑sum‑exp surrogate.
    The algorithm runs until ``deadline`` and keeps the best vector seen.
    """
    a = a0.copy()
    n = a.shape[0]
    best_a = a.copy()
    best_val = compute_score(a)

    m = np.zeros_like(a)                # Adam 1st moment
    v = np.zeros_like(a)                # Adam 2nd moment
    it = 0

    # pre‑allocate reversed version (updated each iteration)
    while time.time() < deadline:
        it += 1
        t = it

        # ------- adapt hyper‑parameters -------------------------------------------------
        # linear schedule based on elapsed fraction
        total = deadline - (deadline - (deadline - time.time()))  # dummy to avoid lint
        frac = (time.time() - (deadline - (deadline - time.time()))) / (deadline - (deadline - time.time()))
        # In practice we just use a time‑based fraction
        elapsed = time.time() - (deadline - (deadline - time.time()))  # zero – placeholder
        # Simpler: use iteration‑based exponential decay
        eps = max(eps_final, eps0 * (0.5 ** (t / 5000.0)))   # decay every ~5k iters
        lr = lr0 * (0.9 ** (t / 2000.0))                     # gentle decay

        # ------- convolution and softmax weights ----------------------------------------
        conv = conv_fft(a, a)                     # length 2n-1
        max_conv = conv.max()

        # stable softmax (log‑sum‑exp) centred at the maximum
        shift = conv - max_conv
        w_unnorm = np.exp(shift / eps)
        w = w_unnorm / w_unnorm.sum()            # shape (2n‑1,)

        # ------- gradient of the smoothed objective ------------------------------------
        a_rev = a[::-1]
        conv_w_rev = conv_fft(w, a_rev)          # length 3n‑2
        # gradient entry j corresponds to slice [n‑1 : n‑1+n]
        g = 2.0 * conv_w_rev[n - 1 : n - 1 + n]

        # ------- Adam update -----------------------------------------------------------
        m = beta1 * m + (1.0 - beta1) * g
        v = beta2 * v + (1.0 - beta2) * (g * g)

        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)

        step = lr * m_hat / (np.sqrt(v_hat) + 1e-12)

        a = a - step
        a = project_to_simplex(a)

        # occasional jitter to avoid being stuck (every ~5k iters)
        if it % 5000 == 0:
            jitter = np.random.randn(n) * (lr * 0.02)
            a = a + jitter
            a = project_to_simplex(a)

        # ------- keep best ----------------------------------------------------------------
        val = compute_score(a)
        if val < best_val:
            best_val, best_a = val, a.copy()

    return best_a


# ----------------------------------------------------------------------
# 3. Simple greedy hill‑climb (final polishing)
# ----------------------------------------------------------------------


def hill_climb(a_start: np.ndarray, time_deadline: float, max_iter: int = 200_000) -> np.ndarray:
    """
    Stochastic mass‑transfer moves with greedy acceptance.
    Very cheap; useful for a last few seconds of polishing.
    """
    a = a_start.copy()
    n = a.shape[0]
    best_a = a.copy()
    best_val = compute_score(a)

    step_scale = 0.05
    it = 0

    while time.time() < time_deadline and it < max_iter:
        i, j = np.random.randint(n, size=2)
        if i == j:
            continue

        delta = step_scale * a[i]
        cand = a.copy()
        cand[i] -= delta
        cand[j] += delta
        cand = np.maximum(cand, 0.0)
        cand = project_to_simplex(cand)

        val = compute_score(cand)
        if val < best_val:
            best_val, best_a = val, cand.copy()
            a = cand
            step_scale = min(step_scale * 1.07, 0.5)
        else:
            step_scale *= 0.985

        it += 1

    return best_a
