"""
enhanced_step_search.py

Search for a non‑negative step‑function (vector) that minimizes

    evaluate_sequence(a) = 2·n·max(conv(a,a)) / (sum(a)^2)

The algorithm works in several stages, respecting the 1000 s time budget:

1. Choose a problem size `n` (larger when time permits).
2. Exhaustive grid search over three 1‑parameter families
   (exponential, power‑law, discrete Beta) and a few random Dirichlet starts.
3. Smooth optimisation of a log‑sum‑exp surrogate of the max‑convolution
   using an exponentiated‑gradient (KL‑mirror) update.  The smoothing
   parameter β is increased from 0.5 to 20000, and the learning rate is
   annealed as 1/√t.  The loop stops automatically when the budget runs out.
4. Direct sub‑gradient descent on the true max‑convolution (still with a
   KL‑mirror step) for further refinement.
5. Targeted hill‑climbing: repeatedly move a small packet of mass from the
   index that contributes most to the current maximal convolution entry to
   a destination that most reduces the max.  Several candidate destinations
   are tested per step; the step size adapts automatically.
6. A short simulated‑annealing‑style random perturbation phase for final
   polishing.

All vectors are kept on the probability simplex (non‑negative, sum = 1),
so the objective is invariant under scaling and matches the behaviour of
`evaluate_sequence`.  The best sequence found is returned as a plain list
of floats.

The code is deliberately written without closures or lambdas, following the
challenge’s constraints.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Basic linear‑algebra utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full aperiodic convolution using a real FFT (zero‑padded)."""
    m = a.size + b.size - 1
    nfft = next_pow2(m)
    A = np.fft.rfft(a, n=nfft)
    B = np.fft.rfft(b, n=nfft)
    C = A * B
    return np.fft.irfft(C, n=nfft)[:m]


def compute_score(a: np.ndarray) -> float:
    """Exact objective for a probability vector (sum == 1)."""
    n = a.shape[0]
    conv = convolution_fft(a, a)
    return 2.0 * n * float(conv.max())


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / v.shape[0])

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, v.shape[0] + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.where(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


# ----------------------------------------------------------------------
# Parameterised families of candidate vectors
# ----------------------------------------------------------------------


def exponential_shape(n: int, lam: float) -> np.ndarray:
    """a_i ∝ exp(-lam·i), normalised to sum = 1."""
    i = np.arange(n, dtype=float)
    w = np.exp(-lam * i) if lam > 0.0 else np.ones_like(i)
    return w / w.sum()


def power_shape(n: int, p: float) -> np.ndarray:
    """a_i ∝ (i+1)^(-p), normalised to sum = 1."""
    i = np.arange(1, n + 1, dtype=float)
    w = i ** (-p) if p > 0.0 else np.ones_like(i)
    return w / w.sum()


def beta_shape(n: int, a: float, b: float) -> np.ndarray:
    """
    Discrete Beta distribution on {0,…,n‑1}:
        a_i ∝ ( (i+0.5)/n )^(a‑1) * (1‑(i+0.5)/n)^(b‑1)
    """
    x = (np.arange(n, dtype=float) + 0.5) / n
    w = np.power(x, a - 1.0) * np.power(1.0 - x, b - 1.0)
    w[np.isnan(w)] = 0.0
    w = np.maximum(w, 0.0)
    if w.sum() == 0.0:
        w = np.ones_like(w)
    return w / w.sum()


def random_dirichlet_start(n: int, conc: float = 1.0) -> np.ndarray:
    """Sample a random point on the simplex using Gamma(shape=conc)."""
    if conc <= 0.0:
        # fallback to uniform
        x = np.random.rand(n)
    else:
        x = np.random.gamma(shape=conc, scale=1.0, size=n)
    if x.sum() == 0:
        x = np.ones_like(x)
    return x / x.sum()

# Grid searches for the simple families
# ----------------------------------------------------------------------


def scan_families(n: int, deadline: float):
    """Extensive scan of exponential, power‑law and Beta families."""
    best_a, best_sc = None, float("inf")

    # Exponential
    lam_grid = np.concatenate((np.array([0.0]), np.logspace(-7, -1, 300)))
    for lam in lam_grid:
        if time.time() >= deadline:
            break
        a = exponential_shape(n, lam)
        sc = compute_score(a)
        if sc < best_sc:
            best_sc, best_a = sc, a.copy()

    # Power‑law
    p_grid = np.linspace(0.0, 8.0, 300)
    for p in p_grid:
        if time.time() >= deadline:
            break
        a = power_shape(n, p)
        sc = compute_score(a)
        if sc < best_sc:
            best_sc, best_a = sc, a.copy()

    # Beta family (moderate grid, more points near edges)
    a_grid = np.concatenate((
        np.logspace(-0.8, -0.2, 14),
        np.logspace(-0.2, 0.4, 14),
        np.linspace(0.5, 3.0, 14)
    ))
    b_grid = a_grid.copy()
    for a_par in a_grid:
        for b_par in b_grid:
            if time.time() >= deadline:
                return best_a, best_sc
            a_vec = beta_shape(n, a_par, b_par)
            sc = compute_score(a_vec)
            if sc < best_sc:
                best_sc, best_a = sc, a_vec.copy()

    return best_a, best_sc

def smooth_optimize(start_a: np.ndarray,
                    deadline: float,
                    max_iters: int = 800_000,
                    beta_start: float = 0.5,
                    beta_end: float = 20_000.0,
                    eta0: float = 0.9) -> tuple[np.ndarray, float]:
    """
    KL‑mirror descent on a log‑sum‑exp approximation of the max‑convolution.
    β grows exponentially from `beta_start` to `beta_end`.
    """
    a = np.maximum(start_a.astype(float), 1e-15)
    a /= a.sum()
    n = a.shape[0]

    best_a = a.copy()
    best_sc = compute_score(a)

    for it in range(1, max_iters + 1):
        if time.time() >= deadline:
            break

        # annealing schedule for β
        t = it / max_iters
        beta = beta_start * (beta_end / beta_start) ** t

        conv = convolution_fft(a, a)                # length 2n‑1
        maxc = conv.max()
        # stable soft‑max weights on the convolution entries
        expw = np.exp(beta * (conv - maxc))
        w = expw / expw.sum()                       # still length 2n‑1

        # gradient of the smooth max w.r.t. a (see derivation in description)
        a_rev = a[::-1]
        grad = 2.0 * convolution_fft(a_rev, w)[n - 1:n - 1 + n]

        eta = eta0 / np.sqrt(it)

        # exponentiated‑gradient (KL‑mirror) update
        a = np.maximum(a * np.exp(-eta * grad), 1e-15)
        a /= a.sum()

        if it % 500 == 0:
            sc = compute_score(a)
            if sc < best_sc:
                best_sc, best_a = sc, a.copy()
    # final check
    sc = compute_score(a)
    if sc < best_sc:
        best_sc, best_a = sc, a.copy()
    return best_a, best_sc


def max_descent_optimize(start_a: np.ndarray,
                         deadline: float,
                         max_iters: int = 400_000,
                         eta0: float = 0.5) -> tuple[np.ndarray, float]:
    """
    Sub‑gradient descent directly on the true max‑convolution,
    still using the KL‑mirror update.
    """
    a = np.maximum(start_a.astype(float), 1e-15)
    a /= a.sum()
    n = a.shape[0]

    best_a = a.copy()
    best_sc = compute_score(a)

    for it in range(1, max_iters + 1):
        if time.time() >= deadline:
            break

        conv = convolution_fft(a, a)
        k_max = int(conv.argmax())

        # sub‑gradient for the active maximal entry
        g = np.zeros_like(a)
        i_start = max(0, k_max - (n - 1))
        i_end = min(n - 1, k_max)
        idx = np.arange(i_start, i_end + 1)
        g[idx] = 2.0 * a[k_max - idx]

        eta = eta0 / np.sqrt(it)

        a = np.maximum(a * np.exp(-eta * g), 1e-15)
        a /= a.sum()

        if it % 400 == 0:
            sc = compute_score(a)
            if sc < best_sc:
                best_sc, best_a = sc, a.copy()
    # final check
    sc = compute_score(a)
    if sc < best_sc:
        best_sc, best_a = sc, a.copy()
    return best_a, best_sc


def hill_climb(start_seq: np.ndarray,
               deadline: float,
               max_steps: int = 600_000) -> tuple[np.ndarray, float]:
    """
    Focused hill‑climbing that moves a packet of mass from the
    index contributing most to the current maximal convolution entry
    to a destination that (approximately) minimises the max.
    """
    a = start_seq.copy()
    n = a.size
    best_a = a.copy()
    best_sc = compute_score(best_a)

    step_scale = 0.12
    steps = 0

    while steps < max_steps and time.time() < deadline:
        conv = convolution_fft(best_a, best_a)
        k_max = int(conv.argmax())

        i_start = max(0, k_max - (n - 1))
        i_end = min(n - 1, k_max)
        idx = np.arange(i_start, i_end + 1)

        # contribution of each index to the active sum
        contrib = best_a[idx] * best_a[k_max - idx]
        i_max = idx[contrib.argmax()]

        # build a small set of candidate destinations
        dest_candidates = {np.argmin(best_a)}
        while len(dest_candidates) < 5:
            cand_j = np.random.randint(n)
            if cand_j != i_max:
                dest_candidates.add(cand_j)

        improved = False
        for j in dest_candidates:
            delta = step_scale * best_a[i_max]
            if delta <= 0.0:
                continue
            cand = best_a.copy()
            cand[i_max] -= delta
            cand[j] += delta
            # projection onto simplex keeps the sum exactly 1
            cand = project_to_simplex(cand)

            sc = compute_score(cand)
            if sc < best_sc - 1e-12:
                best_sc, best_a = sc, cand.copy()
                step_scale = min(step_scale * 1.07, 0.5)
                improved = True
                break

        if not improved:
            step_scale *= 0.93  # shrink step when stuck

        steps += 1

    return best_a, best_sc


def random_perturbation(start_seq: np.ndarray,
                        deadline: float,
                        max_iter: int = 200_000) -> tuple[np.ndarray, float]:
    """
    Short simulated‑annealing‑style random walk.
    Small log‑space perturbations are applied and accepted if they improve.
    """
    a = start_seq.copy()
    n = a.size
    best_a = a.copy()
    best_sc = compute_score(a)

    # temperature schedule – starts relatively high, cools down
    T0 = 0.02
    for it in range(1, max_iter + 1):
        if time.time() >= deadline:
            break
        T = T0 * (1.0 - it / max_iter)  # linear cooling

        # propose a multiplicative perturbation in log‑space
        log_a = np.log(a + 1e-15)
        noise = np.random.randn(n) * (0.2 * T)   # scale shrinks with T
        cand_log = log_a + noise
        cand = np.exp(cand_log)
        cand = project_to_simplex(cand)

        sc = compute_score(cand)
        if sc < best_sc:
            best_sc, best_a = sc, cand.copy()
            a = cand
        else:
            # accept uphill with small probability
            if np.random.rand() < np.exp(-(sc - best_sc) / (T + 1e-12)):
                a = cand

    return best_a, best_sc

        print(f"[{time.time():.1f}] final random‑perturbation polish")
        a_final, sc_final = random_perturbation(
            np.array(best_seq, dtype=float),
            deadline,
            max_iter=200_000,
        )
        if sc_final < best_score:
            best_score, best_seq = sc_final, a_final.tolist()
            print(f"[{time.time():.1f}] perturbation improved best = {best_score:.6f}")

    # -------------------------------------------------
    # Safeguard: tiny uniform vector if everything failed
    # -------------------------------------------------
    if best_seq is None:
        n_small = 256
        best_seq = [1.0 / n_small] * n_small
        best_score = evaluate_sequence(best_seq)

    # -------------------------------------------------
    # Final official evaluation (the harness will also call it)
    # -------------------------------------------------
    final_score = evaluate_sequence(best_seq)
    print(f"\n[{time.time():.1f}] finished – final reported score = {final_score:.6f}")

    # The harness expects a plain Python list of floats.
    return best_seq
