def golden_section(func, a, b, tol=1e-12, max_iter=60, deadline=None):
    phi = (np.sqrt(5) - 1) / 2
    c = b - phi * (b - a)
    d = a + phi * (b - a)
    fc = func(c)
    fd = func(d)
    for _ in range(max_iter):
        if deadline is not None and time.time() >= deadline:
            break
        if fc < fd:
            b = d
            d = c
            fd = fc
            c = b - phi * (b - a)
            fc = func(c)
        else:
            a = c
            c = d
            fc = fd
            d = a + phi * (b - a)
            fd = func(d)
        if abs(b - a) < tol:
            break
    # return best point
    if fc < fd:
        return c, fc
    else:
        return d, fd
```

"""
Step‑function optimiser.

The routine builds a non‑negative sequence a whose sum is 1 (the scaling in
`evaluate_sequence` makes the objective 2·n·max(a*a) independent of the
absolute scale).  It proceeds in the following stages:

1. Choose a length n that fits the remaining wall‑clock budget.
2. One‑dimensional golden‑section searches for the best exponential decay
   λ and the best power‑law exponent p (both families are normalised to a sum
   of 1).  The best of the two candidates is kept.
3. Refine the candidate with a momentum‑based mirror descent in the log‑softmax
   (log‑its) domain.  The update uses the exact sub‑gradient of the max‑convolution
   and a simple momentum term (β=0.9).  The learning‑rate decays as 1/√t.
4. If time permits, repeat the descent from a few random Dirichlet starts and
   keep the best result.
5. A targeted hill‑climb moves a small packet of mass away from the index that
   contributes most to the current maximal convolution entry; moves are kept
   only when they improve the exact objective.
6. Return the best sequence found (as a plain Python list).

All heavy work (FFT based convolutions) is done with NumPy’s real FFT
routines.  The code respects the 1000 s wall‑clock limit via frequent
`time.time()` checks.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Basic linear‑algebra utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """
    Full aperiodic convolution using a real FFT (zero‑padded).
    Returns an array of length len(a)+len(b)-1.
    """
    m = a.size + b.size - 1
    nfft = next_pow2(m)
    A = np.fft.rfft(a, n=nfft)
    B = np.fft.rfft(b, n=nfft)
    C = A * B
    conv = np.fft.irfft(C, n=nfft)[:m]
    return conv


def compute_score(a: np.ndarray) -> float:
    """
    Exact objective used by `evaluate_sequence` for a vector that already
    satisfies sum(a)=1.  Returns
        2·n·max(a*a)
    where the convolution is full (length 2·n‑1).
    """
    n = a.shape[0]
    conv = convolution_fft(a, a)
    return 2.0 * n * float(conv.max())


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / v.shape[0])

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, v.shape[0] + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.where(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


# ----------------------------------------------------------------------
# Parameterised families of candidate vectors
# ----------------------------------------------------------------------


def exponential_shape(n: int, lam: float) -> np.ndarray:
    """a_i ∝ exp(-lam·i), normalised to sum = 1."""
    i = np.arange(n, dtype=float)
    if lam > 0.0:
        w = np.exp(-lam * i)
    else:
        w = np.ones_like(i)
    w_sum = w.sum()
    if w_sum == 0.0:
        w = np.ones_like(i)
        w_sum = w.sum()
    return w / w_sum


def power_shape(n: int, p: float) -> np.ndarray:
    """a_i ∝ (i+1)^(-p), normalised to sum = 1."""
    i = np.arange(1, n + 1, dtype=float)
    if p > 0.0:
        w = i ** (-p)
    else:
        w = np.ones_like(i)
    w_sum = w.sum()
    if w_sum == 0.0:
        w = np.ones_like(i)
        w_sum = w.sum()
    return w / w_sum


# ----------------------------------------------------------------------
# One‑dimensional golden‑section search (bounded, monotone on both sides)
# ----------------------------------------------------------------------


def golden_section(func, a, b, tol=1e-12, max_iter=60, deadline=None):
    """
    Minimise `func` on [a,b] using the golden‑section method.
    Returns the best x and its function value.
    """
    phi = (math.sqrt(5) - 1) / 2  # 0.618...
    c = b - phi * (b - a)
    d = a + phi * (b - a)
    fc = func(c)
    fd = func(d)

    for _ in range(max_iter):
        if deadline is not None and time.time() >= deadline:
            break
        if fc < fd:
            b, fd = d, fc
            d = c
            c = b - phi * (b - a)
            fc = func(c)
        else:
            a, fc = c, fd
            c = d
            d = a + phi * (b - a)
            fd = func(d)
        if abs(b - a) < tol:
            break

    if fc < fd:
        return c, fc
    else:
        return d, fd


# ----------------------------------------------------------------------
# Optimisation helpers
# ----------------------------------------------------------------------


def momentum_descent(start_a, deadline, max_iters=200_000,
                     lr0=0.7, beta=0.9, report_every=2000):
    """
    Mirror‑descent in log‑softmax coordinates with simple momentum.
    Returns the best vector seen and its score.
    """
    a = start_a.astype(float).copy()
    a /= a.sum()
    n = a.shape[0]

    # initialise log‑its
    z = np.log(a + 1e-15)
    m = np.zeros_like(z)

    best_a = a.copy()
    best_sc = compute_score(best_a)

    it = 0
    while time.time() < deadline and it < max_iters:
        it += 1

        # current probability vector from logits
        a = np.exp(z - np.max(z))
        a /= a.sum()

        # convolution and location of the current max
        conv = convolution_fft(a, a)
        k = int(conv.argmax())

        # sub‑gradient of the active convolution entry
        grad = np.zeros_like(a)
        i_start = max(0, k - (n - 1))
        i_end = min(n - 1, k)
        idx = np.arange(i_start, i_end + 1)
        grad[idx] = 2.0 * a[k - idx]

        # gradient w.r.t. logits (softmax Jacobian)
        avg = np.dot(a, grad)
        grad_z = a * (grad - avg)

        # momentum update
        m = beta * m + (1.0 - beta) * grad_z

        # diminishing step size
        lr = lr0 / math.sqrt(it)

        z -= lr * m

        # occasional exact evaluation
        if it % report_every == 0:
            a = np.exp(z - np.max(z))
            a /= a.sum()
            sc = compute_score(a)
            if sc < best_sc:
                best_sc, best_a = sc, a.copy()
                print(f"[{time.time():.1f}] descent iter {it:6d}  best {best_sc:.6f}")

    # final check
    a = np.exp(z - np.max(z))
    a /= a.sum()
    sc = compute_score(a)
    if sc < best_sc:
        best_sc, best_a = sc, a.copy()
    return best_a, best_sc


def targeted_hill_climb(start_a, deadline,
                        max_steps=150_000, report_every=5000):
    """
    Greedy hill‑climb: repeatedly move a fraction of mass from the index that
    contributes most to the maximal convolution entry to a random destination.
    Only improving moves are kept.
    """
    a = start_a.copy()
    n = a.size
    best_a = a.copy()
    best_sc = compute_score(best_a)

    step_scale = 0.10
    steps = 0

    while steps < max_steps and time.time() < deadline:
        conv = convolution_fft(best_a, best_a)
        k = int(conv.argmax())

        # contributions to the active entry
        i_start = max(0, k - (n - 1))
        i_end = min(n - 1, k)
        idx = np.arange(i_start, i_end + 1)
        contrib = best_a[idx] * best_a[k - idx]

        i_max = idx[np.argmax(contrib)]

        # pick a destination that is not i_max
        j = np.random.randint(n - 1)
        if j >= i_max:
            j += 1

        delta = step_scale * best_a[i_max]
        if delta <= 0.0:
            break

        cand = best_a.copy()
        cand[i_max] -= delta
        cand[j] += delta
        cand = project_to_simplex(cand)

        sc = compute_score(cand)
        if sc < best_sc:
            best_sc, best_a = sc, cand.copy()
            step_scale = min(step_scale * 1.07, 0.5)
        else:
            step_scale *= 0.96

        steps += 1
        if steps % report_every == 0:
            print(f"[{time.time():.1f}] hill‑climb step {steps:6d}  best {best_sc:.6f}")

    return best_a, best_sc

    # -------------------------------------------------
    remaining = deadline - time.time()
    if remaining > 800:
        n = 32768
    elif remaining > 500:
        n = 16384
    elif remaining > 250:
        n = 8192
    else:
        n = 4096
    print(f"[{time.time():.1f}] selected length = {n}")

    best_seq = None
    best_score = float("inf")

    # -------------------------------------------------

    # -------------------------------------------------
    try:
        fallback = [float(x) for x in height_sequence_1]   # type: ignore
        fb = evaluate_sequence(fallback)

        if fb < best_score:

    except Exception:
        pass

    # -------------------------------------------------
    # 1. Precise 1‑parameter searches (exponential & power)
    # -------------------------------------------------
    # Exponential decay λ ∈ [0, 0.001] (upper bound generous)
    def exp_obj(lam):
        a = exponential_shape(n, lam)
        return compute_score(a)

    lam_opt, lam_sc = golden_section(
        exp_obj, 0.0, 0.001,
        tol=1e-12, max_iter=50,
        deadline=deadline - 200.0
    )
    a_exp = exponential_shape(n, lam_opt)
    if lam_sc < best_score:
        best_score, best_seq = lam_sc, a_exp.copy()
    print(f"[{time.time():.1f}] exponential search λ={lam_opt:.8f}  score={lam_sc:.6f}")

    # Power‑law exponent p ∈ [0, 6]
    def power_obj(p):
        a = power_shape(n, p)
        return compute_score(a)

    p_opt, p_sc = golden_section(
        power_obj, 0.0, 6.0,
        tol=1e-9, max_iter=60,
        deadline=deadline - 150.0
    )
    a_pow = power_shape(n, p_opt)
    if p_sc < best_score:
        best_score, best_seq = p_sc, a_pow.copy()
    print(f"[{time.time():.1f}] power‑law search p={p_opt:.6f}  score={p_sc:.6f}")

    # -------------------------------------------------
    # 2. Momentum descent from the best candidate found so far
    # -------------------------------------------------
    if best_seq is not None and time.time() < deadline - 80.0:

        a_opt, sc_opt = momentum_descent(

            print(f"[{time.time():.1f}] after descent best = {best_score:.6f}")

    return best_seq.tolist()
