"""
Step‑function minimiser for `evaluate_sequence`.

The optimiser works in several stages, always respecting an overall
wall‑clock budget (default 1000 s):

1. **Problem size** – pick a length `n`.  Larger `n` gives more freedom,
   but the FFT cost grows only ~n log n, so we can afford up to 32768
   when time permits.

2. **Parametric families** – exhaustive (but cheap) scans of several
   1‑ and 2‑parameter families that are known to give low convolution
   peaks:
   * one‑sided exponential,
   * power‑law,
   * symmetric edge‑exponential / edge‑power (high weight near the
     boundaries),
   * Beta distribution (U‑shaped when α,β<1),
   * a simple “two‑edge block’’ shape.

3. **Random Dirichlet starts** – a handful of random probability vectors
   for diversification.

4. **Smooth‑max optimisation** – minimise a log‑sum‑exp surrogate of the
   maximum convolution using a KL‑mirror (exponentiated‑gradient) update.
   The smoothing parameter β is increased from 0.5 to a large value so the
   surrogate approaches the true max.

5. **Exact max‑descent** – sub‑gradient descent on the true
   ‑max(convolution) with the same multiplicative update.

6. **Targeted hill‑climb** – repeatedly moves a small packet of mass
   away from the index that contributes most to the current maximal
   convolution entry; only improving moves are kept, with an adaptive
   step‑size.

7. **Simulated‑annealing polishing** – a stochastic walk that accepts
   occasional uphill moves with a temperature that cools exponentially.
   This helps escape shallow local minima.

All vectors are kept on the probability simplex, i.e. non‑negative and
summing to 1, which makes the objective invariant under scaling and matches
the behaviour of `evaluate_sequence`.  The best vector seen during the run is
returned as a plain list of floats.

The implementation is fully self‑contained, uses only NumPy and the
provided `evaluate_sequence`, and respects the 1 CPU / 1000 s budget.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Basic linear‑algebra utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full aperiodic convolution using a real FFT (zero‑padded)."""
    m = a.size + b.size - 1
    nfft = next_pow2(m)
    A = np.fft.rfft(a, n=nfft)
    B = np.fft.rfft(b, n=nfft)
    C = A * B
    return np.fft.irfft(C, n=nfft)[:m]


def compute_score(a: np.ndarray) -> float:
    """
    Exact objective used by `evaluate_sequence` for a vector that already
    satisfies sum(a)=1.  Returns 2·n·max(a∗a).
    """
    n = a.shape[0]
    conv = convolution_fft(a, a)
    return 2.0 * n * float(conv.max())


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    n = v.shape[0]
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / n)

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.where(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def exponential_shape(n: int, lam: float) -> np.ndarray:
    """a_i ∝ exp(-lam·i), normalised to sum = 1."""
    i = np.arange(n, dtype=float)
    w = np.exp(-lam * i) if lam > 0.0 else np.ones_like(i)
    return w / w.sum()


def power_shape(n: int, p: float) -> np.ndarray:
    """a_i ∝ (i+1)^(-p), normalised to sum = 1."""
    i = np.arange(1, n + 1, dtype=float)
    w = i ** (-p) if p > 0.0 else np.ones_like(i)
    return w / w.sum()


def beta_shape(n: int, a: float, b: float) -> np.ndarray:
    """Discrete Beta distribution on {0,…,n‑1}."""
    x = (np.arange(n, dtype=float) + 0.5) / n
    w = np.power(x, a - 1.0) * np.power(1.0 - x, b - 1.0)
    w[np.isnan(w)] = 0.0
    w = np.maximum(w, 0.0)
    if w.sum() == 0.0:
        w = np.ones_like(w)
    return w / w.sum()


def edge_exponential_shape(n: int, lam: float) -> np.ndarray:
    """a_i ∝ exp(-lam·dist_to_edge), where dist_to_edge = min(i,n‑1‑i)."""
    i = np.arange(n, dtype=float)
    dist = np.minimum(i, n - 1 - i)
    w = np.exp(-lam * dist) if lam > 0.0 else np.ones_like(i)
    return w / w.sum()


def edge_power_shape(n: int, p: float) -> np.ndarray:
    """a_i ∝ (dist_to_edge+1)^(-p)."""
    i = np.arange(n, dtype=float)
    dist = np.minimum(i, n - 1 - i) + 1.0
    w = dist ** (-p) if p > 0.0 else np.ones_like(i)
    return w / w.sum()


def two_edge_block_shape(n: int, fraction: float) -> np.ndarray:
    """
    Uniform mass on the first and last `fraction` proportion of indices,
    zero elsewhere.  The two edge blocks have equal total mass.
    """
    m = int(round(fraction * n))
    if m == 0:
        # fallback to uniform
        return np.full(n, 1.0 / n)
    w = np.zeros(n, dtype=float)
    w[:m] = 1.0
    w[-m:] = 1.0
    return w / w.sum()

def exp_smooth_optimize(start_a: np.ndarray,
                        deadline: float,
                        max_iters: int = 400_000,
                        beta_start: float = 0.5,
                        beta_end: float = 8000.0,
                        eta0: float = 0.7) -> tuple[np.ndarray, float]:
    """
    Minimises a log‑sum‑exp surrogate of the max‑convolution using a KL‑mirror.
    β grows exponentially from `beta_start` to `beta_end`.
    """
    a = np.maximum(start_a.astype(float), 1e-16)
    a /= a.sum()
    n = a.shape[0]

    best_a = a.copy()
    best_sc = compute_score(a)

    for it in range(1, max_iters + 1):
        if time.time() >= deadline:
            break

        t = it / max_iters
        beta = beta_start * (beta_end / beta_start) ** t

        conv = convolution_fft(a, a)
        maxc = conv.max()
        expw = np.exp(beta * (conv - maxc))
        w = expw / expw.sum()

        # gradient of smooth max (see derivation)
        grad = 2.0 * convolution_fft(a[::-1], w)[n - 1:n - 1 + n]

        eta = eta0 / np.sqrt(it)

        a = np.maximum(a * np.exp(-eta * grad), 1e-16)
        a /= a.sum()

        # occasional exact check
        if it % 500 == 0:
            sc = compute_score(a)
            if sc < best_sc:
                best_sc, best_a = sc, a.copy()

    sc = compute_score(a)
    if sc < best_sc:
        best_sc, best_a = sc, a.copy()
    return best_a, best_sc


def max_descent_optimize(start_a: np.ndarray,
                         deadline: float,
                         max_iters: int = 300_000,
                         eta0: float = 0.5) -> tuple[np.ndarray, float]:
    """
    Sub‑gradient descent on the true max‑convolution using a KL‑mirror update.
    """
    a = np.maximum(start_a.astype(float), 1e-16)
    a /= a.sum()
    n = a.shape[0]

    best_a = a.copy()
    best_sc = compute_score(a)

    for it in range(1, max_iters + 1):
        if time.time() >= deadline:
            break

        conv = convolution_fft(a, a)
        k_max = int(conv.argmax())

        # sub‑gradient of the active convolution entry
        g = np.zeros_like(a)
        i_start = max(0, k_max - (n - 1))
        i_end = min(n - 1, k_max)
        idx = np.arange(i_start, i_end + 1)
        g[idx] = 2.0 * a[k_max - idx]

        eta = eta0 / np.sqrt(it)

        a = np.maximum(a * np.exp(-eta * g), 1e-16)
        a /= a.sum()

        if it % 400 == 0:
            sc = compute_score(a)
            if sc < best_sc:
                best_sc, best_a = sc, a.copy()

    sc = compute_score(a)
    if sc < best_sc:
        best_sc, best_a = sc, a.copy()
    return best_a, best_sc


def targeted_hill_climb(start_seq: np.ndarray,
                        deadline: float,
                        max_steps: int = 400_000) -> tuple[np.ndarray, float]:
    """
    Greedy hill‑climb that always steals a small packet of mass from the index
    with the largest contribution to the current maximal convolution entry.
    Step size (as a fraction of the source mass) adapts based on recent
    success.
    """
    a = start_seq.copy()
    n = a.size
    best_a = a.copy()
    best_sc = compute_score(best_a)

    step_scale = 0.08
    steps = 0

    while steps < max_steps and time.time() < deadline:
        conv = convolution_fft(best_a, best_a)
        k_max = int(conv.argmax())

        # indices that appear in the active sum
        i_start = max(0, k_max - (n - 1))
        i_end = min(n - 1, k_max)
        idx = np.arange(i_start, i_end + 1)
        contrib = best_a[idx] * best_a[k_max - idx]

        i_max = idx[np.argmax(contrib)]

        # pick a destination that is *not* i_max
        j = np.random.randint(n - 1)
        if j >= i_max:
            j += 1

        delta = step_scale * best_a[i_max]
        if delta <= 0.0:
            break

        cand = best_a.copy()
        cand[i_max] -= delta
        cand[j] += delta
        cand = project_to_simplex(cand)

        sc = compute_score(cand)
        if sc < best_sc:
            best_sc, best_a = sc, cand.copy()
            step_scale = min(step_scale * 1.08, 0.5)
        else:
            step_scale *= 0.96

        steps += 1

    return best_a, best_sc


def simulated_annealing_refine(start_seq: np.ndarray,
                               deadline: float,
                               max_iters: int = 200_000) -> tuple[np.ndarray, float]:
    """
    Lightweight simulated‑annealing walk.  Randomly moves a tiny packet of
    mass between two indices.  Moves that increase the score may be accepted
    with probability exp(-(Δ)/T); the temperature T decays exponentially.
    """
    a = start_seq.copy()
    n = a.size
    best_a = a.copy()
    best_sc = compute_score(a)

    # initial temperature as a small fraction of the current score
    T = best_sc * 0.02
    # geometric cooling factor
    decay = np.exp(np.log(0.001) / max_iters)  # cool to 0.001*initial after max_iters

    for it in range(max_iters):
        if time.time() >= deadline:
            break

        # choose source and target
        i = np.random.randint(n)
        j = np.random.randint(n - 1)
        if j >= i:
            j += 1
        # transfer a small relative amount
        delta = 0.02 * a[i] * (np.random.rand() * 0.5 + 0.5)  # between 1%‑2% of source
        if delta <= 0:
            continue

        cand = a.copy()
        cand[i] -= delta
        cand[j] += delta
        cand = project_to_simplex(cand)

        sc = compute_score(cand)
        delta_sc = sc - best_sc
        if delta_sc < 0 or np.random.rand() < np.exp(-delta_sc / T):
            a = cand
            if sc < best_sc:
                best_sc, best_a = sc, cand.copy()

        T *= decay
    return best_a, best_sc

def propose_candidate(seed: int = 12345, budget_s: int = 1000, **kwargs):
    """Search for a step‑function minimising `evaluate_sequence`."""
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 2.0   # safety margin

    remaining = deadline - start_time
    # Choose problem size based on remaining budget
    if remaining > 800:
        n = 32768
    elif remaining > 600:
        n = 16384
    elif remaining > 350:
        n = 8192
    else:
        n = 4096
    print(f"[{time.time():.1f}] chosen length = {n}")

    best_seq = None
    best_score = float("inf")

    # ------------------------------------------------------------------
    # 0. optional fallback from a previously discovered vector
    # ------------------------------------------------------------------
    try:
        fallback = [float(x) for x in height_sequence_1]   # type: ignore
        fb = evaluate_sequence(fallback)
        print(f"[{time.time():.1f}] fallback score = {fb:.6f}")
        if fb < best_score:
            best_score = fb
            best_seq = fallback
    except Exception:
        pass

    # ------------------------------------------------------------------
    # 1. Grid scans over diverse families
    # ------------------------------------------------------------------
    # allocate a chunk of the budget for exhaustive scans
    scan_deadline = deadline - 250.0
    if time.time() < scan_deadline:
        a1, sc1 = scan_exponential_and_power(n, scan_deadline)
        if a1 is not None and sc1 < best_score:
            best_score, best_seq = sc1, a1.tolist()
            print(f"[{time.time():.1f}] after exp/power scan = {best_score:.6f}")

    if time.time() < scan_deadline:
        a2, sc2 = scan_beta_family(n, scan_deadline)
        if a2 is not None and sc2 < best_score:
            best_score, best_seq = sc2, a2.tolist()
            print(f"[{time.time():.1f}] after beta scan = {best_score:.6f}")

    if time.time() < scan_deadline:
        a3, sc3 = scan_edge_exponential_and_power(n, scan_deadline)
        if a3 is not None and sc3 < best_score:
            best_score, best_seq = sc3, a3.tolist()
            print(f"[{time.time():.1f}] after edge‑exp/power scan = {best_score:.6f}")

    if time.time() < scan_deadline:
        a4, sc4 = scan_two_edge_block(n, scan_deadline)
        if a4 is not None and sc4 < best_score:
            best_score, best_seq = sc4, a4.tolist()
            print(f"[{time.time():.1f}] after edge‑block scan = {best_score:.6f}")

    # ------------------------------------------------------------------
    # 2. Random Dirichlet starts (small quick diversifications)
    # ------------------------------------------------------------------
    for _ in range(8):
        if time.time() >= scan_deadline:
            break
        rand_vec = np.random.rand(n)
        rand_vec /= rand_vec.sum()
        sc = compute_score(rand_vec)
        if sc < best_score:
            best_score, best_seq = sc, rand_vec.tolist()
            print(f"[{time.time():.1f}] random start improved to {best_score:.6f}")

    # ------------------------------------------------------------------
    # 3. Smooth‑max optimisation (log‑sum‑exp surrogate)
    # ------------------------------------------------------------------
    if best_seq is not None and time.time() < deadline - 150.0:
        print(f"[{time.time():.1f}] starting smooth optimisation")
        a_opt, sc_opt = exp_smooth_optimize(
            np.array(best_seq, dtype=float),
            deadline - 130.0,
            max_iters=500_000,
            beta_end=12000.0,
            eta0=0.7,
        )
        if sc_opt < best_score:
            best_score, best_seq = sc_opt, a_opt.tolist()
            print(f"[{time.time():.1f}] after smooth phase best = {best_score:.6f}")

    # ------------------------------------------------------------------
    # 4. Exact max‑descent optimisation
    # ------------------------------------------------------------------
    if best_seq is not None and time.time() < deadline - 80.0:
        print(f"[{time.time():.1f}] starting exact max‑descent")
        a_opt2, sc_opt2 = max_descent_optimize(
            np.array(best_seq, dtype=float),
            deadline - 70.0,
            max_iters=350_000,
            eta0=0.6,
        )
        if sc_opt2 < best_score:
            best_score, best_seq = sc_opt2, a_opt2.tolist()
            print(f"[{time.time():.1f}] after max‑descent best = {best_score:.6f}")

    # ------------------------------------------------------------------
    # 5. Targeted hill‑climb polishing
    # ------------------------------------------------------------------
    if best_seq is not None and time.time() < deadline - 30.0:
        print(f"[{time.time():.1f}] final targeted hill‑climb")
        a_hc, sc_hc = targeted_hill_climb(
            np.array(best_seq, dtype=float),
            deadline - 20.0,
            max_steps=450_000,
        )
        if sc_hc < best_score:
            best_score, best_seq = sc_hc, a_hc.tolist()
            print(f"[{time.time():.1f}] hill‑climb improved best = {best_score:.6f}")

    # ------------------------------------------------------------------
    # 6. Simulated‑annealing polishing (optional, if time left)
    # ------------------------------------------------------------------
    if best_seq is not None and time.time() < deadline - 5.0:
        print(f"[{time.time():.1f}] simulated‑annealing refinement")
        a_sa, sc_sa = simulated_annealing_refine(
            np.array(best_seq, dtype=float),
            deadline - 5.0,
            max_iters=250_000,
        )
        if sc_sa < best_score:
            best_score, best_seq = sc_sa, a_sa.tolist()
            print(f"[{time.time():.1f}] SA improved best = {best_score:.6f}")

    # ------------------------------------------------------------------
    # Safeguard: tiny uniform vector if everything failed
    # ------------------------------------------------------------------
    if best_seq is None:
        n_small = 256
        best_seq = [1.0 / n_small] * n_small
        best_score = evaluate_sequence(best_seq)

    # ------------------------------------------------------------------
    # Final official evaluation (the harness will also call it)
    # ------------------------------------------------------------------
    final_score = evaluate_sequence(best_seq)
    print(f"\n[{time.time():.1f}] finished – final reported score = {final_score:.6f}")

    # The harness expects a plain Python list of floats.
    return best_seq
