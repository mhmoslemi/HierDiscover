"""
Propose a step‑function that minimises `evaluate_sequence`.

The search proceeds in several stages within the 1000 s wall‑clock budget:

1. **Analytic families** – a rich collection of parameterised shapes
   (U‑shaped, power‑law, exponential, symmetric/asymmetric beta).  A dense
   grid of parameters is scanned and the best vector is kept.

2. **Random convex combinations** – the families above are stored as “base”
   vectors.  Random mixtures of a few bases are projected onto the simplex;
   many such candidates are evaluated.

3. **L‑BFGS refinement** – the current best vector is smooth‑optimised on a
   log‑sum‑exp surrogate of the max‑convolution (β‑schedule 200–8000).

4. **Mirror (exponentiated) sub‑gradient descent** – sub‑gradient of the
   active convolution entry is applied with a multiplicative update and
   projection onto the simplex.

5. **Euclidean sub‑gradient descent** – an additive sub‑gradient step with
   Euclidean projection provides a complementary descent direction.

6. **Symmetrisation** – many optima are symmetric; we average the vector
   with its reversal and re‑project.

7. **Targeted simulated‑annealing hill‑climb** – a low‑temperature SA that
   swaps or transfers mass, with a bias toward improving moves.

All stages respect the global deadline; the best sequence found is returned
as a plain list of floats.
"""

# FFT utilities

def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Linear convolution using real FFT (full length)."""
    m = a.size + b.size - 1
    nfft = next_pow2(m)
    A = np.fft.rfft(a, n=nfft)
    B = np.fft.rfft(b, n=nfft)
    C = A * B
    conv = np.fft.irfft(C, n=nfft)[:m]
    return conv


def compute_score(a: np.ndarray) -> float:
    """
    Exact objective for a vector with sum(a)=1:
        score = 2·n·max(a*a)
    """
    n = a.shape[0]
    conv = convolution_fft(a, a)
    return 2.0 * n * float(conv.max())


# ----------------------------------------------------------------------
# Simplex utilities
# ----------------------------------------------------------------------
def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / v.shape[0])

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, v.shape[0] + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.where(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def softmax(z: np.ndarray) -> np.ndarray:
    """Numerically stable soft‑max."""
    z_max = np.max(z)
    e = np.exp(z - z_max)
    return e / e.sum()


# ----------------------------------------------------------------------
# Analytic families (return vectors with sum = 1)
# ----------------------------------------------------------------------
def symmetric_product_family(n: int, alpha: float) -> np.ndarray:
    """U‑shaped family: a_i ∝ [(i+½)(n‑½‑i)]^(‑α)."""
    i = np.arange(n, dtype=np.float64)
    logprod = np.log(i + 0.5) + np.log(n - 0.5 - i)
    w = np.exp(-alpha * logprod)
    w[np.isinf(w)] = np.finfo(np.float64).max
    return w / w.sum()


def left_heavy_power_family(n: int, beta: float) -> np.ndarray:
    """Monotonically decreasing power‑law: a_i ∝ (i+½)^(‑β)."""
    i = np.arange(n, dtype=np.float64)
    w = np.exp(-beta * np.log(i + 0.5))
    return w / w.sum()


def symmetric_power_family(n: int, beta: float) -> np.ndarray:
    """Symmetrised version of the power‑law (U‑shaped)."""
    left = left_heavy_power_family(n, beta)
    return (left + left[::-1]) * 0.5


def exponential_decay_family(n: int, gamma: float) -> np.ndarray:
    """Simple exponential decay: a_i ∝ exp(‑γ·i)."""
    i = np.arange(n, dtype=np.float64)
    w = np.exp(-gamma * i)
    return w / w.sum()


def beta_distribution_family(n: int, alpha: float) -> np.ndarray:
    """Symmetric Beta(α,α) shape."""
    i = np.arange(n, dtype=np.float64)
    t = i / (n - 1)
    eps = 1e-12
    w = (t + eps) ** (alpha - 1) * (1 - t + eps) ** (alpha - 1)
    return w / w.sum()


def asymmetric_beta_family(n: int, alpha: float, beta: float) -> np.ndarray:
    """General Beta(α,β) density on the index grid."""
    i = np.arange(n, dtype=np.float64)
    t = i / (n - 1)
    eps = 1e-12
    w = (t + eps) ** (alpha - 1) * (1 - t + eps) ** (beta - 1)
    return w / w.sum()

# ----------------------------------------------------------------------
# Smooth surrogate (L‑BFGS) utilities
# ----------------------------------------------------------------------
def _smooth_obj_and_grad(z: np.ndarray, beta: float, n: int):
    """
    Smoothed objective:
        f = (1/β)·log Σ exp(β·(a*a)_k)
    Returns (f, ∂f/∂z) where a = softmax(z).
    """
    a = softmax(z)
    conv = convolution_fft(a, a)              # len = 2n‑1
    maxc = conv.max()
    exp_terms = np.exp(beta * (conv - maxc))
    sum_exp = exp_terms.sum()
    f = (np.log(sum_exp) + beta * maxc) / beta

    w = exp_terms / sum_exp                   # same shape as conv
    # Gradient w.r.t. a: ∂f/∂a = 2 * (a * w) convolved (a reversed)
    a_rev = a[::-1]
    conv_rev = convolution_fft(a_rev, w)      # len = 3n‑2
    grad_a = 2.0 * conv_rev[n - 1:2 * n - 1]

    # Chain rule through softmax
    dot_ag = np.dot(a, grad_a)
    grad_z = a * (grad_a - dot_ag)
    return float(f), grad_z.astype(np.float64)


def l_bfgs_refine(start_a: np.ndarray, deadline: float,
                  beta_list=(200.0, 800.0, 2000.0, 5000.0),
                  max_iter_per_beta=1200) -> tuple[np.ndarray, float]:
    """Refine a vector via L‑BFGS‑B on the smoothed surrogate."""
    if not _HAS_SCIPY:
        return start_a, compute_score(start_a)

    best_a = start_a.copy()
    best_score = compute_score(best_a)
    eps = 1e-12
    z = np.log(best_a + eps)                  # initialise log‑soft‑max params

    for beta in beta_list:
        if time.time() >= deadline - 5.0:
            break
        remaining = deadline - time.time()
        max_iter = min(max_iter_per_beta, int(remaining * 12))  # ~12 iters/sec

        def func(z_vec):
            return _smooth_obj_and_grad(z_vec, beta, best_a.shape[0])

        try:
            opt_res = _opt.fmin_l_bfgs_b(
                func,
                z,
                maxiter=max_iter,
                disp=False,
                factr=1e7,
                pgtol=1e-12,
            )
        except Exception as exc:  # pragma: no cover
            print(f"[{time.time():.1f}] L‑BFGS failed at β={beta}: {exc}")
            continue

        z = opt_res[0]
        a_opt = softmax(z)
        sc_opt = compute_score(a_opt)
        if sc_opt < best_score:
            best_score = sc_opt
            best_a = a_opt.copy()
            print(f"[{time.time():.1f}] L‑BFGS β={beta:.0f} improved score = {best_score:.6f}")

    return best_a, best_score

# ----------------------------------------------------------------------
def mirror_subgrad_optimize(start_a: np.ndarray, deadline: float,
                           step0: float = 0.8,
                           check_interval: int = 2000) -> tuple[np.ndarray, float]:
    """Multiplicative sub‑gradient descent on the true max‑objective."""
    n = start_a.shape[0]
    a = start_a.astype(np.float64).copy()
    best_a = a.copy()
    best_score = compute_score(a)

    t = 0
    while time.time() < deadline:
        t += 1
        conv = convolution_fft(a, a)
        k = int(np.argmax(conv))

        # gradient of the active entry
        idx = np.arange(n)
        j = k - idx
        mask = (j >= 0) & (j < n)
        g = np.zeros_like(a)
        g[mask] = 2.0 * a[j[mask]]

        eta = step0 / math.sqrt(t)
        a = a * np.exp(-eta * g)   # multiplicative update

        # project back onto simplex (entropy projection)
        s = a.sum()
        if s == 0.0:
            a = np.full_like(a, 1.0 / n)
        else:
            a = a / s

        if t % check_interval == 0:
            sc = compute_score(a)
            if sc < best_score:
                best_score = sc
                best_a = a.copy()
                print(f"[{time.time():.1f}] mirror sub‑grad iter {t} best = {best_score:.6f}")

    # final check
    sc = compute_score(a)
    if sc < best_score:
        best_score = sc
        best_a = a.copy()
    return best_a, best_score

# Small‑temperature simulated annealing hill‑climb
# ----------------------------------------------------------------------
def hill_climb_sa(start_a: np.ndarray, deadline: float,
                  max_steps: int = 1_200_000,
                  init_temp_factor: float = 2e-4) -> tuple[np.ndarray, float]:
    """
    Random walk on the simplex with a very low temperature.
    """
    n = start_a.size
    cur_a = start_a.copy()
    cur_score = compute_score(cur_a)

    best_a = cur_a.copy()
    best_score = cur_score

    T0 = best_score * init_temp_factor
    T = T0
    step_scale = 0.12

    for step in range(1, max_steps + 1):
        if time.time() >= deadline:
            break

        cand = cur_a.copy()
        if np.random.rand() < 0.5:          # swap two entries
            i, j = np.random.randint(n, size=2)
            cand[i], cand[j] = cand[j], cand[i]
        else:                               # transfer a fraction of mass
            i, j = np.random.randint(n, size=2)
            if i != j:
                delta = step_scale * cur_a[i]
                cand[i] -= delta
                cand[j] += delta
                if cand[i] < 0.0:
                    cand[i] = 0.0

        cand = cand / cand.sum()
        cand_score = compute_score(cand)
        delta = cand_score - cur_score

        if delta < 0.0 or np.random.rand() < np.exp(-delta / (T + 1e-18)):
            cur_a = cand
            cur_score = cand_score
            if cur_score < best_score:
                best_score = cur_score
                best_a = cur_a.copy()
                step_scale = min(step_scale * 1.07, 0.5)
        else:
            step_scale *= 0.97

        T = T0 * (1.0 - step / max_steps)

    return best_a, best_score

            break
        vec = exponential_decay_family(n, gamma)
        sc = compute_score(vec)
        base_vectors.append(vec)
        if sc < best_score:
            best_score, best_seq = sc, vec.copy()
            print(f"[{time.time():.1f}] new best (exp‑γ={gamma:.2f}) = {best_score:.6f}")

    # Symmetric Beta shapes
    for alpha in np.linspace(0.2, 12.0, 31):
        if time.time() >= deadline - 70.0:
            break
        vec = beta_distribution_family(n, alpha)
        sc = compute_score(vec)
        base_vectors.append(vec)
        if sc < best_score:
            best_score, best_seq = sc, vec.copy()
            print(f"[{time.time():.1f}] new best (beta‑α={alpha:.2f}) = {best_score:.6f}")

    # Random asymmetric Beta families (few samples)
    for _ in range(80):
        if time.time() >= deadline - 70.0:
            break
        a = np.random.uniform(0.2, 12.0)
        b = np.random.uniform(0.2, 12.0)
        vec = asymmetric_beta_family(n, a, b)
        sc = compute_score(vec)
        base_vectors.append(vec)
        if sc < best_score:
            best_score, best_seq = sc, vec.copy()
            print(f"[{time.time():.1f}] new best (beta‑α={a:.2f},β={b:.2f}) = {best_score:.6f}")

    # Random monotone start (adds diversity)
    for _ in range(60):
        if time.time() >= deadline - 70.0:
            break
        vec = random_monotone_start(n)
        sc = compute_score(vec)
        base_vectors.append(vec)
        if sc < best_score:
            best_score, best_seq = sc, vec.copy()
            print(f"[{time.time():.1f}] new best (rand‑mono) = {best_score:.6f}")

    # ------------------------------------------------------------------
    # 2. Random convex combinations of the base vectors
    # ------------------------------------------------------------------
    if base_vectors and time.time() < deadline - 50.0:
        base_arr = np.stack(base_vectors, axis=0)   # shape (B, n)
        B = base_arr.shape[0]
        for _ in range(2500):
            if time.time() >= deadline - 50.0:
                break
            # pick three random bases and mix them
            idxs = np.random.choice(B, size=3, replace=False)
            weights = np.random.rand(3)
            weights /= weights.sum()
            cand = (weights[0] * base_arr[idxs[0]] +
                    weights[1] * base_arr[idxs[1]] +
                    weights[2] * base_arr[idxs[2]])
            cand = project_to_simplex(cand)
            sc = compute_score(cand)
            if sc < best_score:
                best_score, best_seq = sc, cand.copy()
                print(f"[{time.time():.1f}] new best (mix) = {best_score:.6f}")

    # ------------------------------------------------------------------
    # 3. L‑BFGS refinement on the current best
    # ------------------------------------------------------------------
    if best_seq is not None and time.time() < deadline - 200.0:
        print(f"[{time.time():.1f}] Starting L‑BFGS refinement")
        a_start = np.array(best_seq, dtype=np.float64)
        a_ref, sc_ref = l_bfgs_refine(
            a_start,
            deadline - 150.0,
            beta_list=(200., 800., 2000., 5000.),
            max_iter_per_beta=1200,
        )
        if sc_ref < best_score:
            best_score, best_seq = sc_ref, a_ref.copy()
            print(f"[{time.time():.1f}] after L‑BFGS best = {best_score:.6f}")

    # ------------------------------------------------------------------
    # 4. Mirror (exponentiated) sub‑gradient descent
    # ------------------------------------------------------------------
    if best_seq is not None and time.time() < deadline - 80.0:
        print(f"[{time.time():.1f}] Starting mirror sub‑gradient descent")
        a_mir, sc_mir = mirror_subgrad_optimize(
            np.array(best_seq, dtype=np.float64),
            deadline - 40.0,
            step0=1.0,
            check_interval=3000,
        )
        if sc_mir < best_score:
            best_score, best_seq = sc_mir, a_mir.copy()
            print(f"[{time.time():.1f}] after mirror sub‑grad best = {best_score:.6f}")

    # ------------------------------------------------------------------
    # 5. Euclidean sub‑gradient descent (adds a different descent path)
    # ------------------------------------------------------------------
    if best_seq is not None and time.time() < deadline - 40.0:
        print(f"[{time.time():.1f}] Starting Euclidean sub‑gradient descent")
        a_euc, sc_euc = euclidean_subgrad_optimize(
            np.array(best_seq, dtype=np.float64),
            deadline - 20.0,
            step0=0.5,
            check_interval=2000,
        )
        if sc_euc < best_score:
            best_score, best_seq = sc_euc, a_euc.copy()
            print(f"[{time.time():.1f}] after Euclid sub‑grad best = {best_score:.6f}")

    # ------------------------------------------------------------------
    # 6. Symmetrisation (many optima are symmetric)
    # ------------------------------------------------------------------
    if best_seq is not None:
        a_arr = np.array(best_seq, dtype=np.float64)
        a_sym = (a_arr + a_arr[::-1]) * 0.5
        a_sym = project_to_simplex(a_sym)
        sc_sym = compute_score(a_sym)
        if sc_sym < best_score:
            best_score, best_seq = sc_sym, a_sym.tolist()
            print(f"[{time.time():.1f}] symmetrisation improved best = {best_score:.6f}")

    # ------------------------------------------------------------------
    # 7. Final low‑temperature hill‑climb
    # ------------------------------------------------------------------
    if best_seq is not None and time.time() < deadline - 5.0:
        print(f"[{time.time():.1f}] Final hill‑climb (SA)")
        a_final, sc_final = hill_climb_sa(
            np.array(best_seq, dtype=np.float64),
            deadline,
            max_steps=800_000,
            init_temp_factor=2e-4,
        )
        if sc_final < best_score:
            best_score, best_seq = sc_final, a_final.tolist()
            print(f"[{time.time():.1f}] hill‑climb improved best = {best_score:.6f}")

    # ------------------------------------------------------------------
    # Safeguard – fall back to uniform if nothing was found
    # ------------------------------------------------------------------
    if best_seq is None:
        best_seq = [1.0 / n] * n
        best_score = evaluate_sequence(best_seq)

    # ------------------------------------------------------------------
    # Final official evaluation (the harness will also call evaluate_sequence)
    # ------------------------------------------------------------------
    final_score = evaluate_sequence(best_seq)
    print(f"\n[{time.time():.1f}] finished search – final reported score = {final_score:.6f}")
    return best_seq
