"""
Propose a step‑function (non‑negative sequence) that minimises

    evaluate_sequence(a) = 2·n·max(conv(a,a)) / (sum(a)²)

The algorithm consists of three stages:

1. **Seeding** – a collection of hand‑crafted families (sine², beta‑polynomials,
   arcsine, random Dirichlet, sparse spikes).  The best seed (exact objective)
   is kept.

2. **Adamised soft‑max minimisation** – we minimise the smooth surrogate

       L_β(a) = log Σ exp(β·conv(a,a))

   using Adam on the projected simplex.  The temperature β is linearly
   increased from a small value to a large one, giving a very accurate
   approximation of the true max‑norm.  The projection onto the probability
   simplex is performed after each update.

3. **Final polishing** – a short sub‑gradient phase on the true max‑objective
   followed by a stochastic hill‑climb that accepts only improving moves.

All work respects a 1000 s wall‑clock budget.  The returned value is a plain
``list[float]`` of non‑negative heights.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Low‑level utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full linear convolution using a real FFT (real input arrays)."""
    m = a.shape[0] + b.shape[0] - 1
    n_fft = next_pow2(m)
    A = np.fft.rfft(a, n=n_fft)
    B = np.fft.rfft(b, n=n_fft)
    C = A * B
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


def compute_score(a: np.ndarray) -> float:
    """
    Exact objective for a probability vector (sum = 1):
        2·n·max(conv(a,a)).
    """
    n = a.shape[0]
    conv = convolution_fft(a, a)
    return 2.0 * n * float(conv.max())


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of `v` onto the probability simplex
    {x | x_i ≥ 0, Σ x_i = 1}.
    """
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / v.shape[0])

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, v.shape[0] + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.where(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


# ----------------------------------------------------------------------
# Seed generators
# ----------------------------------------------------------------------


def generate_deterministic(n: int) -> np.ndarray:
    """Sin² shape a_i = sin²(π·i/(n+1))."""
    i = np.arange(1, n + 1, dtype=float)
    a = np.sin(np.pi * i / (n + 1.0)) ** 2
    a /= a.sum()
    return a


def generate_beta_family(n: int, p: float) -> np.ndarray:
    """Symmetric polynomial shape a_i ∝ ((i+1)*(n-i))**p."""
    i = np.arange(n, dtype=float)
    w = ((i + 1.0) * (n - i)) ** p
    w[w < 0] = 0.0
    a = w / w.sum()
    return a


def generate_arcsine(n: int) -> np.ndarray:
    """Arcsine‑type shape a_i ∝ 1/√( (i+0.5)*(n‑0.5‑i) )."""
    i = np.arange(n, dtype=float) + 0.5
    w = 1.0 / np.sqrt(i * (n - 0.5 - i))
    a = w / w.sum()
    return a


def random_dirichlet(n: int, conc: float = 1.0) -> np.ndarray:
    """Draw a random probability vector from Dirichlet(conc)."""
    if conc == 1.0:
        a = np.random.rand(n)
    else:
        a = np.random.gamma(conc, 1.0, size=n)
    a /= a.sum()
    return a


def random_spike(n: int, k: int = 70) -> np.ndarray:
    """Very sparse vector: k random positions with random positive weights."""
    idx = np.random.choice(n, size=min(k, n), replace=False)
    w = np.random.rand(len(idx))
    a = np.zeros(n, dtype=float)
    a[idx] = w
    a /= a.sum()
    return a


# ----------------------------------------------------------------------
# Optimisation phases
# ----------------------------------------------------------------------


def adam_softmax(
    a: np.ndarray,
    start_score: float,
    deadline: float,
    beta_start: float = 5.0,
    beta_end: float = 8000.0,
    lr: float = 0.05,
    beta1: float = 0.9,
    beta2: float = 0.999,
) -> tuple[np.ndarray, float]:
    """
    Projected Adam minimisation of the smooth surrogate
        Lβ(a)=log Σ exp(β·conv(a,a)).
    The temperature β is annealed linearly from `beta_start` to `beta_end`.
    """
    n = a.shape[0]
    cur = a.copy()
    best = cur.copy()
    best_score = start_score

    m = np.zeros_like(cur)  # first moment
    v = np.zeros_like(cur)  # second moment
    it = 0
    phase_start = time.time()

    while time.time() < deadline:
        # exact objective (for bookkeeping)
        conv = convolution_fft(cur, cur)
        max_conv = conv.max()
        cur_score = 2.0 * n * float(max_conv)
        if cur_score < best_score:
            best_score = cur_score
            best = cur.copy()

        # temperature schedule
        elapsed = time.time() - phase_start
        frac = elapsed / (deadline - phase_start)
        beta = beta_start + frac * (beta_end - beta_start)

        # soft‑max weights (stable)
        shifted = conv - max_conv          # max becomes 0
        exp_vals = np.exp(beta * shifted)  # all ≤ 1
        w = exp_vals / exp_vals.sum()      # probability vector of length 2n‑1

        # gradient of Lβ:  ∇ = 2·β·correlate(w, a)
        # compute correlation via convolution with reversed a
        conv_rev = convolution_fft(w, cur[::-1])
        grad = 2.0 * beta * conv_rev[n - 1 : 2 * n - 1]

        # Adam update
        it += 1
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)

        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        step = lr * m_hat / (np.sqrt(v_hat) + 1e-8)

        cur = cur - step
        cur = project_to_simplex(cur)

    return best, best_score


def subgradient_refine(
    a: np.ndarray,
    start_score: float,
    deadline: float,
    max_iter: int = 300_000,
    step0: float = 0.07,
) -> tuple[np.ndarray, float]:
    """Projected sub‑gradient descent on the true max‑objective."""
    n = a.shape[0]
    cur = a.copy()
    best = cur.copy()
    best_score = start_score

    for t in range(1, max_iter + 1):
        if time.time() >= deadline:
            break

        conv = convolution_fft(cur, cur)
        k = int(np.argmax(conv))

        # sub‑gradient of max(conv) w.r.t. a:
        idx = np.arange(n)
        j = k - idx
        mask = (j >= 0) & (j < n)
        g = np.zeros_like(cur)
        g[mask] = 2.0 * cur[j[mask]]

        step = step0 / np.sqrt(t)
        cur -= step * g
        cur = project_to_simplex(cur)

        cur_score = compute_score(cur)
        if cur_score < best_score:
            best_score = cur_score
            best = cur.copy()

        # occasional tiny jitter to escape plateaus
        if t % 1000 == 0:
            jitter = np.random.randn(n) * (step0 * 0.004 / np.sqrt(t))
            cur += jitter
            cur = project_to_simplex(cur)

    return best, best_score


def hill_climb(
    a: np.ndarray,
    deadline: float,
    max_steps: int = 800_000,
) -> tuple[np.ndarray, float]:
    """Stochastic hill‑climb that keeps a move only if it improves the exact score."""
    n = a.shape[0]
    best = a.copy()
    best_score = compute_score(best)
    step_scale = 0.08

    steps = 0
    while steps < max_steps and time.time() < deadline:
        cand = best.copy()
        # either swap two entries or transfer a fraction from one to another
        if np.random.rand() < 0.5:
            i, j = np.random.randint(n, size=2)
            cand[i], cand[j] = cand[j], cand[i]
        else:
            i, j = np.random.randint(n, size=2)
            if cand[i] > 0.0 and i != j:
                delta = step_scale * cand[i]
                cand[i] -= delta
                cand[j] += delta
                if cand[i] < 0.0:
                    cand[i] = 0.0

        cand /= cand.sum()                     # re‑normalise numerical drift
        cur_score = compute_score(cand)

        if cur_score < best_score:
            best_score = cur_score
            best = cand
            step_scale = min(step_scale * 1.07, 0.5)
        else:
            step_scale *= 0.96

        steps += 1

    return best, best_score

def propose_candidate(seed: int = 12345, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step‑function that minimises `evaluate_sequence`.
    Returns a plain Python list of floats.
    """
    np.random.seed(seed)
    start_time = time.time()
    # safety margin for the external harness
    deadline = start_time + budget_s - 1.0
    total_time = deadline - start_time

    # --------------------------------------------------------------
    # 0. Optional fallback from a previously discovered sequence
    # --------------------------------------------------------------
    best_seq = None
    best_val = float("inf")
    try:
        fallback = [float(x) for x in height_sequence_1]  # type: ignore
        fb = evaluate_sequence(fallback)
        print(f"Fallback score = {fb:.6f}")
        if fb < best_val:
            best_val = fb
            best_seq = fallback
    except Exception:
        pass

    # --------------------------------------------------------------
    # 1. Choose a suitable length (larger lengths give more freedom)
    # --------------------------------------------------------------
    n = 4096 if total_time > 850.0 else 2048
    print(f"Using sequence length = {n}")

    # --------------------------------------------------------------
    # 2. Build a diverse pool of seeds
    # --------------------------------------------------------------
    seeds = []

    # deterministic analytic families
    seeds.append(generate_deterministic(n))
    for p in np.linspace(-2.5, 2.5, 41):            # fine sweep over beta–family
        seeds.append(generate_beta_family(n, p))
    seeds.append(generate_arcsine(n))

    # random Dirichlet draws with several concentrations
    for conc in (0.1, 0.3, 0.5, 1.0, 2.0, 5.0):
        for _ in range(2):
            seeds.append(random_dirichlet(n, conc=conc))

    # very sparse spikes – historically a good start for max‑conv reduction
    for k in (30, 50, 80, 120):
        seeds.append(random_spike(n, k=k))

    # include fallback if it matches the current length
    if best_seq is not None and len(best_seq) == n:
        seeds.append(np.array(best_seq, dtype=float) / np.sum(best_seq))

    # evaluate seeds with the exact objective and keep the best one
    for s in seeds:
        s = np.asarray(s, dtype=float)
        s = project_to_simplex(s)          # safeguard tiny numerical drift
        score = compute_score(s)
        if score < best_val:
            best_val = score
            best_seq = s.tolist()
            print(f"[seed] improvement: score = {best_val:.6f}")

    cur_a = np.array(best_seq, dtype=float)
    cur_a = cur_a / cur_a.sum()
    cur_score = compute_score(cur_a)

    # --------------------------------------------------------------
    # 3. Adam‑based soft‑max gradient descent (major part of the budget)
    # --------------------------------------------------------------
    phase1_deadline = start_time + total_time * 0.55
    print("\n[Phase 1] Adam soft‑max minimisation")
    cur_a, cur_score = adam_softmax(
        cur_a,
        cur_score,
        phase1_deadline,
        beta_start=5.0,
        beta_end=8000.0,
        lr=0.05,
    )
    if cur_score < best_val:
        best_val = cur_score
    print(f"After Phase 1: best score = {best_val:.6f}")

    # --------------------------------------------------------------
    # 4. Sub‑gradient refinement (exact max‑objective)
    # --------------------------------------------------------------
    if time.time() < deadline - 30.0:
        print("\n[Phase 2] Sub‑gradient refinement")
        phase2_deadline = deadline - 30.0
        cur_a, cur_score = subgradient_refine(
            cur_a, cur_score, phase2_deadline, max_iter=600_000, step0=0.07
        )
        if cur_score < best_val:
            best_val = cur_score
            print(f"After sub‑grad: best score = {best_val:.6f}")

    # --------------------------------------------------------------
    # 5. Final hill‑climb polishing
    # --------------------------------------------------------------
    if time.time() < deadline - 1.0:
        print("\n[Phase 3] Hill‑climb polishing")
        cur_a, cur_score = hill_climb(cur_a, deadline, max_steps=1_200_000)
        if cur_score < best_val:
            best_val = cur_score
            print(f"After polishing: best score = {best_val:.6f}")

    # --------------------------------------------------------------
    # 6. Final conversion and official evaluation
    # --------------------------------------------------------------
    final_seq = cur_a.tolist()
    final_val = evaluate_sequence(final_seq)
    print(f"\nFinished search. Final reported score = {final_val:.6f}")

    return final_seq
