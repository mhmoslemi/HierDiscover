"""
Optimisation of the “step‑function” objective.

The evaluation function is
    E(a) = 2·n·max(a ⊛ a) / (∑a)²
which is invariant under positive scaling, so we may restrict the vector to the
probability simplex (∑a = 1).  The goal is therefore to minimise
    2·n·max(conv(a,a)).

The algorithm consists of the following phases, each respecting the 1000 s
wall‑clock budget:

1. Choose a problem size `n`.  Larger `n` gives a tighter theoretical lower
   bound (≈ 2 n/(2 n‑1) → 1), but also increases the cost of each FFT.
2. Quick systematic scans of three one‑parameter families (exponential,
   power‑law, discrete Beta) to obtain good deterministic seeds.
3. A few random Dirichlet seeds to diversify the search.
4. For each promising seed run a long “max‑descent’’ optimisation (sub‑gradient
   descent on the true max‑convolution) using a KL‑mirror (exponentiated‑gradient)
   update.  The method is cheap – only one FFT per iteration – and converges
   reliably.
5. A final intensive targeted hill‑climb that repeatedly extracts a small
   packet of mass from the index that contributes most to the current maximal
   convolution entry and redistributes it elsewhere.  The step size is
   automatically adapted.
6. Return the best sequence found.

All updates keep the vector on the probability simplex (non‑negative,
normalised) which matches the behaviour of `evaluate_sequence`.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Linear‑algebra utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full aperiodic convolution via a real FFT (zero‑padded)."""
    m = a.size + b.size - 1
    nfft = next_pow2(m)
    A = np.fft.rfft(a, n=nfft)
    B = np.fft.rfft(b, n=nfft)
    C = A * B
    return np.fft.irfft(C, n=nfft)[:m]


def compute_score(a: np.ndarray) -> float:
    """
    Exact objective used by `evaluate_sequence` for a vector already on the
    simplex (∑a = 1).  Returns 2·n·max(conv(a,a)).
    """
    n = a.shape[0]
    conv = convolution_fft(a, a)
    return 2.0 * n * float(conv.max())


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / v.shape[0])

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, v.shape[0] + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.where(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


# ----------------------------------------------------------------------
# Parameterised families of candidate vectors
# ----------------------------------------------------------------------


def exponential_shape(n: int, lam: float) -> np.ndarray:
    """a_i ∝ exp(-lam·i), normalised to sum = 1."""
    i = np.arange(n, dtype=float)
    w = np.exp(-lam * i) if lam > 0.0 else np.ones_like(i)
    return w / w.sum()


def power_shape(n: int, p: float) -> np.ndarray:
    """a_i ∝ (i+1)^(-p), normalised to sum = 1."""
    i = np.arange(1, n + 1, dtype=float)
    w = i ** (-p) if p > 0.0 else np.ones_like(i)
    return w / w.sum()


def beta_shape(n: int, a: float, b: float) -> np.ndarray:
    """
    Discrete Beta distribution on {0,…,n‑1}:
        a_i ∝ ((i+0.5)/n)^(a‑1) * (1‑((i+0.5)/n))^(b‑1)
    """
    x = (np.arange(n, dtype=float) + 0.5) / n
    w = np.power(x, a - 1.0) * np.power(1.0 - x, b - 1.0)
    w[np.isnan(w)] = 0.0
    w = np.maximum(w, 0.0)
    if w.sum() == 0.0:
        w = np.ones_like(w)
    return w / w.sum()


# ----------------------------------------------------------------------
# Grid searches for the simple families
# ----------------------------------------------------------------------


def scan_exponential_and_power(n: int, deadline: float):
    """Exhaustive scan of exponential and power‑law families."""
    best_a, best_sc = None, float("inf")

    # exponential λ from 0 (uniform) to about 5 (very steep)
    lam_grid = np.concatenate((np.array([0.0]), np.logspace(-5, 0, 250)))
    for lam in lam_grid:
        if time.time() >= deadline:
            break
        a = exponential_shape(n, lam)
        sc = compute_score(a)
        if sc < best_sc:
            best_sc, best_a = sc, a.copy()

    # power‑law p from 0 (uniform) to 8
    p_grid = np.linspace(0.0, 8.0, 250)
    for p in p_grid:
        if time.time() >= deadline:
            break
        a = power_shape(n, p)
        sc = compute_score(a)
        if sc < best_sc:
            best_sc, best_a = sc, a.copy()

    return best_a, best_sc


def scan_beta_family(n: int, deadline: float):
    """Moderate 2‑parameter Beta scan."""
    best_a, best_sc = None, float("inf")
    # concentrate grids around both edges and centre
    a_grid = np.concatenate((
        np.logspace(-0.7, -0.1, 10),
        np.logspace(-0.1, 0.3, 10),
        np.linspace(0.5, 3.0, 10)))
    b_grid = a_grid.copy()
    for a in a_grid:
        for b in b_grid:
            if time.time() >= deadline:
                return best_a, best_sc
            a_vec = beta_shape(n, a, b)
            sc = compute_score(a_vec)
            if sc < best_sc:
                best_sc, best_a = sc, a_vec.copy()
    return best_a, best_sc

def max_descent_optimize(start_a: np.ndarray,
                         deadline: float,
                         max_iters: int = 300_000,
                         eta0: float = 0.5) -> tuple[np.ndarray, float]:
    """
    Direct sub‑gradient descent on the true max‑convolution.
    Updates use the KL‑mirror (multiplicative) rule.
    """
    a = np.maximum(start_a.astype(float), 1e-15)
    a /= a.sum()
    n = a.shape[0]

    best_a = a.copy()
    best_sc = compute_score(a)

    for it in range(1, max_iters + 1):
        if time.time() >= deadline:
            break

        conv = convolution_fft(a, a)
        k_max = int(conv.argmax())

        # sub‑gradient of the active entry
        g = np.zeros_like(a)
        i_start = max(0, k_max - (n - 1))
        i_end = min(n - 1, k_max)
        idx = np.arange(i_start, i_end + 1)
        g[idx] = 2.0 * a[k_max - idx]

        eta = eta0 / np.sqrt(it)

        a = np.maximum(a * np.exp(-eta * g), 1e-15)
        a /= a.sum()

        if it % 250 == 0:
            sc = compute_score(a)
            if sc < best_sc:
                best_sc, best_a = sc, a.copy()
    # final check
    sc = compute_score(a)
    if sc < best_sc:
        best_sc, best_a = sc, a.copy()
    return best_a, best_sc


def targeted_hill_climb(start_seq: np.ndarray,
                        deadline: float,
                        max_steps: int = 500_000) -> tuple[np.ndarray, float]:
    """
    Greedy hill‑climb: repeatedly extract a fraction of mass from the index that
    contributes most to the current maximal convolution entry and redistribute it.
    Accepts only improving moves; the step size is adapted.
    """
    a = start_seq.copy()
    n = a.size
    best_a = a.copy()
    best_sc = compute_score(best_a)

    step_scale = 0.08   # initial fraction of the selected mass to move
    steps = 0

    while steps < max_steps and time.time() < deadline:
        conv = convolution_fft(best_a, best_a)
        k_max = int(conv.argmax())

        # indices that participate in the active sum
        i_start = max(0, k_max - (n - 1))
        i_end = min(n - 1, k_max)
        idx = np.arange(i_start, i_end + 1)
        contrib = best_a[idx] * best_a[k_max - idx]

        # pick the index with the single largest contribution
        i_max = idx[np.argmax(contrib)]

        # choose a destination distinct from i_max
        j = np.random.randint(n)
        if j == i_max:
            continue

        delta = step_scale * best_a[i_max]
        if delta <= 0.0:
            break

        cand = best_a.copy()
        cand[i_max] -= delta
        cand[j] += delta
        cand = project_to_simplex(cand)

        sc = compute_score(cand)
        if sc < best_sc:
            best_sc, best_a = sc, cand.copy()
            step_scale = min(step_scale * 1.06, 0.5)
        else:
            step_scale *= 0.94

        steps += 1

    return best_a, best_sc

def propose_candidate(seed: int = 12345, budget_s: int = 1000, **kwargs):
    """Search for a step‑function that minimises `evaluate_sequence`."""
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 2.0   # safety margin

    # -------------------------------------------------
    # Choose a sequence length.
    # -------------------------------------------------
    remaining = deadline - start_time
    if remaining > 800:
        n = 32768
    elif remaining > 500:
        n = 24576
    else:
        n = 16384
    print(f"[{time.time():.1f}] chosen length = {n}")

    best_seq = None
    best_score = float("inf")

    # -------------------------------------------------
    # Fallback from a previously discovered vector.
    # -------------------------------------------------
    try:
        fallback = [float(x) for x in height_sequence_1]   # type: ignore
        fb = evaluate_sequence(fallback)
        print(f"[{time.time():.1f}] fallback score = {fb:.6f}")
        if fb < best_score:
            best_score = fb
            best_seq = fallback
    except Exception:
        pass

    # -------------------------------------------------
    # 1. Grid search over simple families.
    # -------------------------------------------------
    a_scan, sc_scan = scan_exponential_and_power(n, deadline - 200.0)
    if a_scan is not None and sc_scan < best_score:
        best_score, best_seq = sc_scan, a_scan.tolist()
        print(f"[{time.time():.1f}] after exp/power scan = {best_score:.6f}")

    a_beta, sc_beta = scan_beta_family(n, deadline - 150.0)
    if a_beta is not None and sc_beta < best_score:
        best_score, best_seq = sc_beta, a_beta.tolist()
        print(f"[{time.time():.1f}] after beta scan = {best_score:.6f}")

    # -------------------------------------------------
    # 2. A handful of random Dirichlet starts.
    # -------------------------------------------------
    for _ in range(4):
        if time.time() >= deadline - 130.0:
            break
        a_rand = np.random.rand(n)
        a_rand /= a_rand.sum()
        sc = compute_score(a_rand)
        if sc < best_score:
            best_score, best_seq = sc, a_rand.tolist()
            print(f"[{time.time():.1f}] random start improved to {best_score:.6f}")

    # -------------------------------------------------
    # 3. Long max‑descent optimisation on the best seed.
    # -------------------------------------------------
    if best_seq is not None and time.time() < deadline - 80.0:
        print(f"[{time.time():.1f}] starting long max‑descent")
        a_opt, sc_opt = max_descent_optimize(
            np.array(best_seq, dtype=float),
            deadline - 30.0,
            max_iters=600_000,
            eta0=0.45,
        )
        if sc_opt < best_score:
            best_score, best_seq = sc_opt, a_opt.tolist()
            print(f"[{time.time():.1f}] after long max‑descent best = {best_score:.6f}")

    # -------------------------------------------------
    # 4. Targeted hill‑climb polishing.
    # -------------------------------------------------
    if best_seq is not None and time.time() < deadline - 5.0:
        print(f"[{time.time():.1f}] final targeted hill‑climb")
        a_final, sc_final = targeted_hill_climb(
            np.array(best_seq, dtype=float),
            deadline,
            max_steps=1_000_000,
        )
        if sc_final < best_score:
            best_score, best_seq = sc_final, a_final.tolist()
            print(f"[{time.time():.1f}] hill‑climb improved best = {best_score:.6f}")

    # -------------------------------------------------
    # Safeguard: tiny uniform vector if everything failed.
    # -------------------------------------------------
    if best_seq is None:
        n_small = 256
        best_seq = [1.0 / n_small] * n_small
        best_score = evaluate_sequence(best_seq)

    # -------------------------------------------------
    # Final official evaluation (the harness will also call it).
    # -------------------------------------------------
    final_score = evaluate_sequence(best_seq)
    print(f"\n[{time.time():.1f}] finished – final reported score = {final_score:.6f}")

    return best_seq
