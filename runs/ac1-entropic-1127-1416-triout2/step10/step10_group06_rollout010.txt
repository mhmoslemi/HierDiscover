def adam_les_optimize(start_a, deadline, beta_schedule, max_iter_per_beta=2000):
    n = start_a.shape[0]
    eps = 1e-12
    # initialize logits
    z = np.log(start_a + eps)
    best_a = start_a.copy()
    best_score = compute_score(best_a)
    # Adam hyperparameters
    b1 = 0.9
    b2 = 0.999
    # timestamp
    t_global = 0
    for beta in beta_schedule:
        if time.time() >= deadline - 2.0: break
        # Initialize Adam moments for this beta
        m = np.zeros_like(z)
        v = np.zeros_like(z)
        t = 0
        # Determine iteration budget based on remaining time
        remaining = deadline - time.time()
        max_iter = min(max_iter_per_beta, int(remaining * 15)) # approximate 15 iters per sec
        for i in range(max_iter):
            if time.time() >= deadline - 2.0: break
            t += 1
            t_global += 1
            loss, grad_z = _smooth_obj_and_grad(z, beta, n)
            # Adam update
            m = b1 * m + (1 - b1) * grad_z
            v = b2 * v + (1 - b2) * (grad_z * grad_z)
            m_hat = m / (1 - b1 ** t)
            v_hat = v / (1 - b2 ** t)
            lr = 0.05 / math.sqrt(beta)  # scale inversely with sqrt(beta)
            z = z - lr * m_hat / (np.sqrt(v_hat) + 1e-8)
            # evaluate periodically
            if i % 200 == 0:
                a = softmax(z)
                a = symmetrize_and_project(a)
                sc = compute_score(a)
                if sc < best_score:
                    best_score = sc
                    best_a = a.copy()
                    print(f"[{time.time():.1f}] Adam β={beta:.0f} iter {i} best = {best_score:.6f}")
                # re-assign z based on updated a to keep consistency (floating log)
                z = np.log(a + eps)  # optional
        # End of beta loop: final evaluate
        a = softmax(z)
        a = symmetrize_and_project(a)
        sc = compute_score(a)
        if sc < best_score:
            best_score = sc
            best_a = a.copy()
            print(f"[{time.time():.1f}] Adam after β={beta:.0f} improved best = {best_score:.6f}")
    return best_a, best_score

def pnorm_exponential_optimize(start_a, deadline, p_list, max_iter_per_p=2000):
    n = start_a.shape[0]
    a = start_a.copy()
    a = symmetrize_and_project(a)
    best_a = a.copy()
    best_score = compute_score(best_a)
    t = 0
    for p in p_list:
        if time.time() >= deadline - 2.0: break
        remaining = deadline - time.time()
        max_iter = min(max_iter_per_p, int(remaining * 15))
        for i in range(max_iter):
            if time.time() >= deadline - 2.0: break
            t += 1
            # compute convolution
            conv = convolution_fft(a, a)   # length 2n-1
            # compute norm_p
            norm_p = np.sum(conv ** p) ** (1.0 / p)
            # compute weights
            w = conv ** (p - 1)
            # compute gradient wrt a
            a_rev = a[::-1]
            conv_rev = convolution_fft(a_rev, w)  # length 3n-2
            grad = 2.0 * conv_rev[n - 1: 2 * n - 1] * (norm_p ** (-(p - 1)))
            # exponentiated update
            eta = 0.8 / math.sqrt(t + 1)
            a = a * np.exp(-eta * grad)
            # project onto simplex
            a = a / a.sum()
            # symmetrize
            a = symmetrize_and_project(a)
            # evaluate periodically
            if i % 200 == 0:
                sc = compute_score(a)
                if sc < best_score:
                    best_score = sc
                    best_a = a.copy()
                    print(f"[{time.time():.1f}] p-norm p={p} iter {i} best = {best_score:.6f}")
        # after finishing p-phase, compute best final
        sc = compute_score(a)
        if sc < best_score:
            best_score = sc
            best_a = a.copy()
            print(f"[{time.time():.1f}] p-norm after p={p} improved best = {best_score:.6f}")
    return best_a, best_score

"""
Propose a step‑function sequence that minimises `evaluate_sequence`.

Algorithm
---------
1. **Problem normalisation** – work with a probability vector (`sum = 1`);
   the target value is `2·n·max(a*a)`.  All updates stay on the simplex.

2. **Broad analytic sweep** – generate many candidate shapes
   (U‑shaped, power‑law, exponential, β‑distribution and a 2‑parameter
   product‑power family).  A fine grid is scanned and the best few
   vectors are kept.

3. **Smooth surrogate optimisation** – minimise the log‑sum‑exp
   approximation of the max‑convolution using Adam on the log‑soft‑max
   (logits) parameters.  The temperature `β` is increased
   (`5 → 20 → 80 → 200 → 800 → 2000 → 5000`) while the learning‑rate
   is scaled as `0.05/√β`.  After each β‑phase the current vector is
   symmetrised and projected onto the simplex.

4. **p‑norm refinement** – replace the LES surrogate by the
   `L_p`‑norm of the convolution (`p = 4,8,16,32,64,128`).  The gradient
   is obtained analytically and an exponentiated‑gradient (mirror) step
   with step‑size `η_t = 0.8/√t` is applied.  Symmetrisation is enforced
   after every update.

5. **Final tiny simulated‑annealing hill‑climb** – a cheap SA moves
   mass between entries (swap or transfer) with a very low temperature
   to escape shallow local minima.

All stages respect the global wall‑clock deadline (default 1000 s).
The code uses only NumPy and the supplied `evaluate_sequence` function.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# FFT utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """
    Linear convolution via real FFT.
    Returns a vector of length len(a)+len(b)-1.
    """
    m = a.size + b.size - 1
    nfft = next_pow2(m)
    A = np.fft.rfft(a, n=nfft)
    B = np.fft.rfft(b, n=nfft)
    C = A * B
    conv = np.fft.irfft(C, n=nfft)[:m]
    return conv


def compute_score(a: np.ndarray) -> float:
    """
    Exact objective for a probability vector (∑a = 1):
        score = 2·n·max(a*a)
    """
    n = a.shape[0]
    conv = convolution_fft(a, a)
    return 2.0 * n * float(conv.max())


# ----------------------------------------------------------------------
# Simplex utilities
# ----------------------------------------------------------------------


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection onto the probability simplex."""
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / v.shape[0])

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, v.shape[0] + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.where(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def softmax(z: np.ndarray) -> np.ndarray:
    """Numerically stable soft‑max (maps ℝⁿ → simplex)."""
    z_max = np.max(z)
    e = np.exp(z - z_max)
    return e / e.sum()


def symmetrize_and_project(a: np.ndarray) -> np.ndarray:
    """Force symmetry (a[i] = a[n‑1‑i]) and project onto the simplex."""
    a = (a + a[::-1]) * 0.5
    a = np.maximum(a, 0.0)
    return project_to_simplex(a)


# ----------------------------------------------------------------------
# Analytic families (all return a probability vector)
# ----------------------------------------------------------------------


def symmetric_product_family(n: int, alpha: float) -> np.ndarray:
    """U‑shaped: a_i ∝ [(i+½)(n‑½‑i)]^(‑α)."""
    i = np.arange(n, dtype=np.float64)
    logprod = np.log(i + 0.5) + np.log(n - 0.5 - i)
    w = np.exp(-alpha * logprod)
    w[np.isinf(w)] = np.finfo(np.float64).max
    a = w / w.sum()
    return a


def left_heavy_power_family(n: int, beta: float) -> np.ndarray:
    """Monotone decreasing power law: a_i ∝ (i+½)^(‑β)."""
    i = np.arange(n, dtype=np.float64)
    w = np.exp(-beta * np.log(i + 0.5))
    a = w / w.sum()
    return a


def exponential_decay_family(n: int, gamma: float) -> np.ndarray:
    """Simple exponential decay: a_i ∝ exp(‑γ·i)."""
    i = np.arange(n, dtype=np.float64)
    w = np.exp(-gamma * i)
    a = w / w.sum()
    return a


def beta_distribution_family(n: int, alpha: float) -> np.ndarray:
    """Symmetric beta‑shape: a_i ∝ (i/(n‑1))^α·(1‑i/(n‑1))^α."""
    i = np.arange(n, dtype=np.float64)
    t = i / (n - 1)
    eps = 1e-12
    w = (t + eps) ** alpha * (1 - t + eps) ** alpha
    a = w / w.sum()
    return a


def combined_product_power_family(n: int, alpha: float, beta: float) -> np.ndarray:
    """
    Two‑parameter family:
        a_i ∝ [(i+½)(n‑½‑i)]^(‑α) · (i+½)^(‑β)
    (symmetric U‑shape modulated by a monotone power law).
    """
    i = np.arange(n, dtype=np.float64)
    logprod = np.log(i + 0.5) + np.log(n - 0.5 - i)
    w = np.exp(-alpha * logprod - beta * np.log(i + 0.5))
    w[np.isinf(w)] = 0.0
    a = w / w.sum()
    return a


# ----------------------------------------------------------------------
# Random initialisers
# ----------------------------------------------------------------------


def random_dirichlet_start(n: int, conc: float) -> np.ndarray:
    """Dirichlet draw with concentration `conc` (≤1 → sparse)."""
    a = np.random.dirichlet(np.full(n, conc))
    return a


def random_sparse_start(n: int) -> np.ndarray:
    """≈5 % non‑zeros, equal mass on the active entries."""
    sparsity = np.random.uniform(0.03, 0.07)
    k = max(2, int(round(sparsity * n)))
    idx = np.random.choice(n, size=k, replace=False)
    a = np.zeros(n, dtype=np.float64)
    a[idx] = 1.0 / k
    return project_to_simplex(a)


def random_monotone_start(n: int) -> np.ndarray:
    """Generate a random decreasing vector."""
    r = np.random.rand(n) ** np.random.uniform(0.5, 5.0)
    r.sort()
    a = r[::-1]            # decreasing
    a = a / a.sum()
    return a


# ----------------------------------------------------------------------
# Smooth surrogate (log‑sum‑exp) gradient (used by Adam)
# ----------------------------------------------------------------------


def _smooth_obj_and_grad(z: np.ndarray, beta: float, n: int):
    """
    Returns (f, grad_z) where
        f   = (1/β)·log∑_k exp[β·(a*a)_k]   (smooth approximation of max)
        grad_z = ∂f/∂z  (z are the logits of the soft‑max)
    """
    a = softmax(z)                     # a ∈ simplex
    conv = convolution_fft(a, a)       # length 2n‑1
    maxc = conv.max()
    # log‑sum‑exp in a stable way
    exp_terms = np.exp(beta * (conv - maxc))
    sum_exp = exp_terms.sum()
    f = (np.log(sum_exp) + beta * maxc) / beta

    # gradient w.r.t. a
    w = exp_terms / sum_exp            # same shape as conv
    a_rev = a[::-1]
    conv_rev = convolution_fft(a_rev, w)   # length 3n‑2
    grad_a = 2.0 * conv_rev[n - 1:2 * n - 1]

    # chain rule through soft‑max
    dot_ag = np.dot(a, grad_a)
    grad_z = a * (grad_a - dot_ag)
    return float(f), grad_z.astype(np.float64)


# ----------------------------------------------------------------------
# Adam optimisation of the LES surrogate
# ----------------------------------------------------------------------


def adam_les_optimize(start_a: np.ndarray,
                      deadline: float,
                      beta_schedule=(5.0, 20.0, 80.0, 200.0, 800.0, 2000.0, 5000.0),
                      max_iter_per_beta=4000) -> tuple[np.ndarray, float]:
    """
    Adam optimiser on the log‑sum‑exp surrogate.
    Returns the best vector found and its exact score.
    """
    n = start_a.shape[0]
    eps = 1e-12
    z = np.log(start_a + eps)

    best_a = start_a.copy()
    best_score = compute_score(best_a)

    # Adam hyper‑parameters
    b1, b2 = 0.9, 0.999

    for beta in beta_schedule:
        if time.time() >= deadline - 2.0:
            break

        m = np.zeros_like(z)
        v = np.zeros_like(z)
        t = 0

        # adapt iteration budget to remaining time (≈15 iters / s)
        remaining = deadline - time.time()
        max_iter = min(max_iter_per_beta, int(remaining * 15))

        for it in range(max_iter):
            if time.time() >= deadline - 2.0:
                break
            t += 1
            loss, grad_z = _smooth_obj_and_grad(z, beta, n)

            # Adam update
            m = b1 * m + (1 - b1) * grad_z
            v = b2 * v + (1 - b2) * (grad_z * grad_z)
            m_hat = m / (1 - b1 ** t)
            v_hat = v / (1 - b2 ** t)

            lr = 0.05 / math.sqrt(beta)          # heuristic scaling
            z = z - lr * m_hat / (np.sqrt(v_hat) + 1e-8)

            # occasional evaluation (every 200 steps)
            if (it & 0x7F) == 0:    # ~ every 128 steps, cheap integer test
                a = softmax(z)
                a = symmetrize_and_project(a)
                sc = compute_score(a)
                if sc < best_score:
                    best_score = sc
                    best_a = a.copy()
                    print(f"[{time.time():.1f}] Adam β={beta:.0f} iter {it} best = {best_score:.6f}")
                # keep the logits consistent with the symmetrised a
                z = np.log(a + eps)

        # final evaluation for this β
        a = softmax(z)
        a = symmetrize_and_project(a)
        sc = compute_score(a)
        if sc < best_score:
            best_score = sc
            best_a = a.copy()
            print(f"[{time.time():.1f}] After β={beta:.0f} improved best = {best_score:.6f}")

    return best_a, best_score


# ----------------------------------------------------------------------
# p‑norm surrogate optimisation with exponentiated (mirror) updates
# ----------------------------------------------------------------------


def pnorm_exponential_optimize(start_a: np.ndarray,
                               deadline: float,
                               p_list=(4, 8, 16, 32, 64, 128),
                               max_iter_per_p=4000) -> tuple[np.ndarray, float]:
    """
    Minimises the L_p‑norm of the convolution (`p` increasing) using
    multiplicative updates (mirror descent) on the simplex.
    Returns the best vector and its exact score.
    """
    n = start_a.shape[0]
    a = symmetrize_and_project(start_a.copy())
    best_a = a.copy()
    best_score = compute_score(best_a)

    t_total = 0

    for p in p_list:
        if time.time() >= deadline - 2.0:
            break

        remaining = deadline - time.time()
        max_iter = min(max_iter_per_p, int(remaining * 15))

        for it in range(max_iter):
            if time.time() >= deadline - 2.0:
                break
            t_total += 1

            conv = convolution_fft(a, a)                 # len = 2n‑1
            norm_p = np.sum(conv ** p) ** (1.0 / p)
            # avoid division by zero when norm is tiny
            if norm_p == 0:
                break
            w = conv ** (p - 1)                         # weights for the gradient
            a_rev = a[::-1]
            conv_rev = convolution_fft(a_rev, w)        # len = 3n‑2
            grad_a = 2.0 * conv_rev[n - 1:2 * n - 1] * (norm_p ** (-(p - 1)))

            # exponentiated (mirror) update
            eta = 0.8 / math.sqrt(t_total + 1)
            a = a * np.exp(-eta * grad_a)

            # renormalise onto the simplex
            s = a.sum()
            if s == 0.0:
                a = np.full_like(a, 1.0 / n)
            else:
                a = a / s

            # enforce symmetry
            a = symmetrize_and_project(a)

            # periodic exact evaluation
            if (it & 0x7F) == 0:      # every ~128 steps
                sc = compute_score(a)
                if sc < best_score:
                    best_score = sc
                    best_a = a.copy()
                    print(f"[{time.time():.1f}] p‑norm p={p} iter {it} best = {best_score:.6f}")

        # final check after finishing this p
        sc = compute_score(a)
        if sc < best_score:
            best_score = sc
            best_a = a.copy()
            print(f"[{time.time():.1f}] After p={p} improved best = {best_score:.6f}")

    return best_a, best_score


# ----------------------------------------------------------------------
# Tiny simulated‑annealing hill‑climb (final polishing)
# ----------------------------------------------------------------------


def hill_climb_sa(start_a: np.ndarray,
                  deadline: float,
                  max_steps: int = 800_000,
                  init_temp_factor: float = 1e-5) -> tuple[np.ndarray, float]:
    """
    Very low‑temperature simulated annealing on the simplex.
    Moves: swap two entries or shift a fraction of mass.
    """
    n = start_a.size
    cur_a = start_a.copy()
    cur_score = compute_score(cur_a)

    best_a = cur_a.copy()
    best_score = cur_score

    T0 = best_score * init_temp_factor
    T = T0
    step_scale = 0.12

    for step in range(1, max_steps + 1):
        if time.time() >= deadline:
            break

        cand = cur_a.copy()
        if np.random.rand() < 0.5:          # swap
            i, j = np.random.randint(n, size=2)
            cand[i], cand[j] = cand[j], cand[i]
        else:                                # mass transfer
            i, j = np.random.randint(n, size=2)
            if i != j:
                delta = step_scale * cand[i]
                cand[i] -= delta
                cand[j] += delta
                if cand[i] < 0.0:
                    cand[i] = 0.0

        cand = project_to_simplex(cand)      # stay on simplex
        cand_score = compute_score(cand)
        delta = cand_score - cur_score

        if delta < 0.0 or np.random.rand() < np.exp(-delta / (T + 1e-18)):
            cur_a = cand
            cur_score = cand_score
            if cur_score < best_score:
                best_score = cur_score
                best_a = cur_a.copy()
                step_scale = min(step_scale * 1.07, 0.5)
        else:
            step_scale *= 0.97

        T = T0 * (1.0 - step / max_steps)

    return best_a, best_score

    return best_seq
