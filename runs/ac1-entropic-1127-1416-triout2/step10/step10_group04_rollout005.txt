"""
Non‑negative step‑function optimiser (symmetry‑aware).

Main idea
---------
* Generate a wide variety of analytic starting shapes (uniform, central‑binomial,
  arcsine, power‑law, geometric, triangular) and a few random Dirichlet draws.
* Enforce exact symmetry a_i = a_{n‑1‑i} – the optimal solution is believed to be
  symmetric, which reduces the search space and stabilises the optimisation.
* Run a long smoothed‑max Adam optimisation with an annealed temperature
  (β from 1 to ≈2000).  After each Adam step we symmetrise and project onto the
  probability simplex.
* Polish the result with a projected sub‑gradient descent on the true
  max‑convolution objective (again respecting symmetry).
* Perform several short random‑restart bursts (Gaussian perturbation + brief
  sub‑gradient run) to escape local minima.
* Finish with a cheap symmetric stochastic hill‑climb that moves a small amount
  of mass between mirrored positions.
* All phases respect the wall‑clock deadline (default 1000 s) and keep the
  best sequence seen.
"""

import time
import numpy as np


# ----------------------------------------------------------------------
# Basic utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """
    Full linear convolution via real FFT.
    Returns length len(a)+len(b)-1 array.
    """
    m = a.shape[0] + b.shape[0] - 1
    n_fft = next_pow2(m)
    A = np.fft.rfft(a, n=n_fft)
    B = np.fft.rfft(b, n=n_fft)
    C = A * B
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


def compute_score(a: np.ndarray) -> float:
    """
    Exact objective for a probability vector (sum = 1).
    Returns 2·n·max(conv(a,a)).
    """
    n = a.shape[0]
    conv = convolution_fft(a, a)
    return 2.0 * n * float(conv.max())


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of v onto the probability simplex (sum = 1, v ≥ 0).
    """
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / v.shape[0])

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, v.shape[0] + 1)
    cond = u - cssv / ind > 0
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def symmetrize(v: np.ndarray) -> np.ndarray:
    """Force exact mirror symmetry: v_i = v_{n-1-i}."""
    return (v + v[::-1]) / 2.0


# ----------------------------------------------------------------------
# Analytic families for initial guesses
# ----------------------------------------------------------------------


def generate_uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def generate_central_binomial(n: int) -> np.ndarray:
    """a_i = C(2i,i)/4^i, monotone decreasing. Not symmetric – we symmetrise later."""
    a = np.empty(n, dtype=np.float64)
    a[0] = 1.0
    for i in range(1, n):
        a[i] = a[i - 1] * (2 * i - 1) / (2.0 * i)
    return a


def generate_powerlaw(n: int, p: float) -> np.ndarray:
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = idx ** (-p)
    return a


def generate_geometric(n: int, r: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = (1.0 - r) * (r ** i)
    return a


def generate_arcsine(n: int) -> np.ndarray:
    """Discrete analogue of the arcsine density: a_i ∝ 1/√((i+1)*(n-i))."""
    i = np.arange(n, dtype=np.float64) + 0.5
    a = 1.0 / np.sqrt(i * (n - i + 0.5))
    return a


def generate_triangular(n: int) -> np.ndarray:
    """Symmetric triangle: linearly rising to the centre then falling."""
    i = np.arange(n, dtype=np.float64)
    a = np.minimum(i + 1, n - i)
    return a


# ----------------------------------------------------------------------
# Symmetry‑aware smoothed‑max (log‑sum‑exp) optimisation – Adam
# ----------------------------------------------------------------------


def smooth_sym_optimize(
    start_a: np.ndarray,
    time_deadline: float,
    max_iters: int = 500_000,
    beta_start: float = 1.0,
    beta_end: float = 2000.0,
    step0: float = 0.9,
    adam_beta1: float = 0.9,
    adam_beta2: float = 0.999,
    adam_eps: float = 1e-8,
) -> tuple[np.ndarray, float]:
    """
    Adam optimiser for the smoothed max‑convolution objective.
    Keeps symmetry a = a[::-1] throughout.
    """
    a = start_a.astype(np.float64).copy()
    a = symmetrize(a)
    a = project_to_simplex(a)

    n = a.shape[0]
    m1 = np.zeros_like(a)
    m2 = np.zeros_like(a)

    best_a = a.copy()
    best_score = compute_score(a)

    for it in range(1, max_iters + 1):
        if time.time() >= time_deadline:
            break

        # anneal temperature
        t_frac = it / max_iters
        beta = beta_start * (beta_end / beta_start) ** t_frac

        conv = convolution_fft(a, a)
        maxc = conv.max()
        # stabilized soft‑max weights
        diff = conv - maxc
        # clip exponent to avoid overflow
        exp_arg = np.clip(beta * diff, -60.0, 60.0)
        exp_w = np.exp(exp_arg)
        w = exp_w / exp_w.sum()

        # gradient of the smooth surrogate
        rev_a = a[::-1]
        grad_full = convolution_fft(rev_a, w)
        grad = 2.0 * grad_full[n - 1 : n - 1 + n]

        # Adam update
        m1 = adam_beta1 * m1 + (1.0 - adam_beta1) * grad
        m2 = adam_beta2 * m2 + (1.0 - adam_beta2) * (grad * grad)
        m1_hat = m1 / (1.0 - adam_beta1 ** it)
        m2_hat = m2 / (1.0 - adam_beta2 ** it)

        step = step0 / np.sqrt(it)
        a_new = a - step * m1_hat / (np.sqrt(m2_hat) + adam_eps)

        # enforce symmetry and simplex feasibility
        a_new = symmetrize(a_new)
        a_new = project_to_simplex(a_new)

        a = a_new

        if it % 500 == 0:
            cur_score = compute_score(a)
            if cur_score < best_score:
                best_score = cur_score
                best_a = a.copy()

    final_score = compute_score(a)
    if final_score < best_score:
        best_score, best_a = final_score, a.copy()
    return best_a, best_score


# ----------------------------------------------------------------------
# Symmetry‑aware projected sub‑gradient optimiser (true max objective)
# ----------------------------------------------------------------------


def subgrad_sym_optimize(
    init_a: np.ndarray,
    time_deadline: float,
    max_iter: int = 300_000,
    step0: float = 0.07,
) -> tuple[np.ndarray, float]:
    """Projected sub‑gradient descent on 2·n·max(conv(a,a)) with symmetry enforcement."""
    a = init_a.astype(np.float64).copy()
    a = symmetrize(a)
    a = project_to_simplex(a)

    best_a = a.copy()
    best_score = compute_score(a)
    n = a.shape[0]

    for t in range(1, max_iter + 1):
        if time.time() >= time_deadline:
            break

        conv = convolution_fft(a, a)
        k = int(np.argmax(conv))

        # sub‑gradient for the active convolution entry
        idx = np.arange(n)
        j = k - idx
        mask = (j >= 0) & (j < n)
        g = np.zeros_like(a)
        g[mask] = 2.0 * a[j[mask]]

        step = step0 / np.sqrt(t)
        a_new = a - step * g

        # symmetry & projection
        a_new = symmetrize(a_new)
        a_new = project_to_simplex(a_new)

        cur_score = compute_score(a_new)
        if cur_score < best_score:
            best_score, best_a = cur_score, a_new.copy()

        a = a_new

        # occasional jitter to escape flats
        if t % 1000 == 0:
            jitter = np.random.randn(n) * (step0 * 0.004 / np.sqrt(t))
            a += jitter
            a = symmetrize(a)
            a = project_to_simplex(a)

    final_score = compute_score(a)
    if final_score < best_score:
        best_score, best_a = final_score, a.copy()
    return best_a, best_score


# ----------------------------------------------------------------------
# Random‑restart helper (small Gaussian perturbations + short sub‑gradient)
# ----------------------------------------------------------------------


def restart_perturbations(
    base_a: np.ndarray,
    time_deadline: float,
    n_restarts: int = 5,
    per_restart_budget: float = 12.0,
) -> tuple[np.ndarray, float]:
    """Generate perturbed copies of ``base_a`` and polish each with a short sub‑gradient run."""
    best_a = base_a.copy()
    best_score = compute_score(best_a)

    n = base_a.shape[0]

    for r in range(n_restarts):
        if time.time() >= time_deadline:
            break

        pert = best_a + 0.02 * np.random.randn(n)
        pert = symmetrize(pert)
        pert = project_to_simplex(pert)

        sub_deadline = time.time() + per_restart_budget
        # limit iterations heuristically (≈2000 it/s)
        max_iter = int(min(200_000, per_restart_budget * 2_000))

        a_tmp, s_tmp = subgrad_sym_optimize(
            pert,
            time_deadline=sub_deadline,
            max_iter=max_iter,
            step0=0.05,
        )
        if s_tmp < best_score:
            best_score, best_a = s_tmp, a_tmp.copy()
            print(f" Restart {r}: improved to {best_score:.6f}")

    return best_a, best_score


# ----------------------------------------------------------------------
# Symmetric stochastic hill‑climb (tiny transfers respecting mirror pairs)
# ----------------------------------------------------------------------


def hill_climb_sym(
    seq,
    time_deadline,
    max_steps: int = 250_000,
) -> tuple[list[float], float]:
    """Tiny symmetric moves (swap or paired mass transfer) with simplex projection."""
    a = np.array(seq, dtype=np.float64)
    a = symmetrize(a)
    a = project_to_simplex(a)
    n = a.shape[0]

    best_a = a.copy()
    best_score = compute_score(best_a)

    step_scale = 0.07  # fraction of mass moved each successful step

    while max_steps > 0 and time.time() < time_deadline:
        cand = best_a.copy()

        if np.random.rand() < 0.5:
            # swap two symmetric pairs
            i, j = np.random.randint(n // 2 + 1, size=2)
            # perform swap on both sides
            cand[i], cand[j] = cand[j], cand[i]
            if i != n - 1 - i:
                cand[n - 1 - i], cand[n - 1 - j] = cand[n - 1 - j], cand[n - 1 - i]
        else:
            # move a small fraction from i to j (mirrored as well)
            i, j = np.random.randint(n // 2 + 1, size=2)
            if i != j:
                delta = step_scale * best_a[i]
                cand[i] -= delta
                cand[j] += delta
                # mirror the same transfer
                mi, mj = n - 1 - i, n - 1 - j
                cand[mi] -= delta
                cand[mj] += delta
                # clip negatives
                cand[i] = max(cand[i], 0.0)
                cand[mi] = max(cand[mi], 0.0)

        cand = project_to_simplex(cand)
        cand = symmetrize(cand)
        cur_score = compute_score(cand)

        if cur_score < best_score:
            best_score = cur_score
            best_a = cand
            step_scale = min(step_scale * 1.07, 0.5)
        else:
            step_scale *= 0.96

        max_steps -= 1

    return best_a.tolist(), best_score


# ----------------------------------------------------------------------
# Main entry point required by the harness
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 12345, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative, symmetric sequence that minimises ``evaluate_sequence``.
    Returns a plain Python list of floats (already on the simplex).
    """
    np.random.seed(seed)
    start_time = time.time()
    # leave a small safety margin for the harness
    deadline = start_time + budget_s - 4.0

    # ------------------------------------------------------------------
    # Determine problem size (length of the sequence)
    # ------------------------------------------------------------------
    try:
        n = len(height_sequence_1)  # type: ignore[name-defined]
        print(f"Using provided height_sequence_1 of length {n}.")
    except Exception:
        n = 2048
        print(f"No external sequence provided – using length {n}.")

    # ------------------------------------------------------------------
    # 0. Assemble a diverse set of analytic starters
    # ------------------------------------------------------------------
    init_candidates = []

    # uniform
    init_candidates.append(generate_uniform(n))

    # central binomial (symmetrised)
    cb = generate_central_binomial(n)
    init_candidates.append(symmetrize(cb))

    # arcsine shape (symmetrised automatically)
    init_candidates.append(generate_arcsine(n))

    # triangular
    init_candidates.append(symmetrize(generate_triangular(n)))

    # power‑law family
    for p in np.linspace(0.2, 1.0, 9):
        init_candidates.append(symmetrize(generate_powerlaw(n, float(p))))

    # geometric family (moderately heavy tails)
    for r in [0.94, 0.96, 0.97, 0.98, 0.99]:
        init_candidates.append(symmetrize(generate_geometric(n, float(r))))

    # a few random Dirichlet draws (α = 0.5 gives heavy‑tailed samples)
    for _ in range(8):
        rand_vec = np.random.dirichlet(alpha=np.full(n, 0.5))
        init_candidates.append(symmetrize(rand_vec))

    # ------------------------------------------------------------------
    # 1. Evaluate starters and pick the best
    # ------------------------------------------------------------------
    best_seq = None
    best_val = float("inf")
    for idx, cand in enumerate(init_candidates):
        cand = project_to_simplex(symmetrize(cand))
        val = compute_score(cand)
        print(f"Starter {idx:02d}: score = {val:.6f}")
        if val < best_val:
            best_val = val
            best_seq = cand.copy()

    print(f"Best starter score = {best_val:.6f}")

    # ------------------------------------------------------------------
    # 2. Smoothed‑max Adam optimisation (symmetry‑aware)
    # ------------------------------------------------------------------
    smooth_deadline = deadline - 260.0
    if time.time() < smooth_deadline:
        remaining = smooth_deadline - time.time()
        # Roughly 1800 it/s on typical hardware → allow up to ~3 M iters if time permits
        max_iters = int(min(3_000_000, remaining * 1800))
        print(f"Smooth Adam phase: up to {max_iters:,} iterations.")
        a_smooth, smooth_score = smooth_sym_optimize(
            best_seq,
            time_deadline=smooth_deadline,
            max_iters=max_iters,
            beta_start=1.0,
            beta_end=2000.0,
            step0=0.85,
        )
        if smooth_score < best_val:
            best_val = smooth_score
            best_seq = a_smooth.copy()
            print(f"  → smooth phase improved to {best_val:.6f}")

    # ------------------------------------------------------------------
    # 3. Projected sub‑gradient polishing (symmetry‑aware)
    # ------------------------------------------------------------------
    subgrad_deadline = deadline - 80.0
    if time.time() < subgrad_deadline:
        remaining = subgrad_deadline - time.time()
        max_iter = int(min(800_000, remaining * 2000))
        print(f"Sub‑gradient phase: up to {max_iter:,} iterations.")
        a_sub, sub_score = subgrad_sym_optimize(
            best_seq,
            time_deadline=subgrad_deadline,
            max_iter=max_iter,
            step0=0.07,
        )
        if sub_score < best_val:
            best_val = sub_score
            best_seq = a_sub.copy()
            print(f"  → sub‑gradient phase improved to {best_val:.6f}")

    # ------------------------------------------------------------------
    # 4. Random restarts with short sub‑gradient bursts
    # ------------------------------------------------------------------
    restart_deadline = deadline - 30.0
    if time.time() < restart_deadline:
        n_restarts = 6
        per_restart = (restart_deadline - time.time()) / max(1, n_restarts)
        print(f"Random‑restart phase: {n_restarts} restarts, ~{per_restart:.1f}s each.")
        a_restart, restart_score = restart_perturbations(
            best_seq,
            time_deadline=restart_deadline,
            n_restarts=n_restarts,
            per_restart_budget=per_restart,
        )
        if restart_score < best_val:
            best_val = restart_score
            best_seq = a_restart.copy()
            print(f"  → restart phase improved to {best_val:.6f}")

    # ------------------------------------------------------------------
    # 5. Final symmetric stochastic hill‑climb
    # ------------------------------------------------------------------
    hill_deadline = deadline - 2.0
    if time.time() < hill_deadline:
        steps = 300_000
        print(f"Hill‑climb phase: up to {steps:,} steps.")
        polished_seq, hill_score = hill_climb_sym(
            best_seq.tolist(),
            time_deadline=hill_deadline,
            max_steps=steps,
        )
        if hill_score < best_val:
            best_val = hill_score
            best_seq = np.array(polished_seq, dtype=np.float64)
            print(f"  → hill‑climb improved to {best_val:.6f}")

    # ------------------------------------------------------------------
    # 6. Safeguard – fall back to uniform if nothing was found
    # ------------------------------------------------------------------
    if best_seq is None:
        best_seq = generate_uniform(n)
        best_val = compute_score(best_seq)

    # ------------------------------------------------------------------
    # 7. Final reporting using the provided evaluate_sequence()
    # ------------------------------------------------------------------
    final_score = evaluate_sequence(best_seq.tolist())
    print(f"\nSearch finished. Final reported score = {final_score:.6f}")

    # Return a plain list of floats (already on the simplex)
    return best_seq.tolist()
