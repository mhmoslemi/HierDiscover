"""
Non‑negative step‑function optimiser for ``evaluate_sequence`` (2026 edition).

Goal
----
Find a vector a of length n (default 2048) with non‑negative entries,
∑a = 1, minimising   2·n·max(conv(a,a)).
The function ``evaluate_sequence`` is scale‑invariant, therefore we keep the
vector on the probability simplex.

Strategy
--------
1.  **Fallback** – try the historic construction (height_sequence_1) if it
    exists.
2.  **Global sweep** – a cheap power‑law sweep (a_i∝(i+1)^‑p) to obtain a
    decent starting point.
3.  **Smooth‑max gradient descent** – minimise a smooth approximation of the
    max‑convolution using a projected gradient method.  
    The smooth max is  (1/β)·log∑exp(β·conv) .  
    The gradient is a weighted sum of the usual sub‑gradients, computed
    efficiently with FFT‑based correlation.  β is increased gradually (continuation)
    so that the optimisation first sees a smooth landscape and later focusses on the
    true maximum.
    Several random Dirichlet restarts are tried within the remaining time.
4.  **Fine greedy hill‑climb** – cheap stochastic mass‑transfer moves with
    greedy acceptance, starting from the best vector found above.
5.  Return the best sequence (as a plain Python list) before the time‑budget
    expires.

All steps respect the global time budget (≈1000 s) and every intermediate
vector is projected onto the simplex to satisfy the constraints required by
``evaluate_sequence``.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Numerical utilities
# ----------------------------------------------------------------------


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """ Euclidean projection of v onto the probability simplex (sum=1, v>=0). """
    if np.all(v <= 0):
        # Uniform distribution is the projection of the zero vector.
        return np.full_like(v, 1.0 / v.shape[0])

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, v.shape[0] + 1)
    mask = u - cssv / ind > 0
    rho = np.where(mask)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def full_convolution(a: np.ndarray) -> np.ndarray:
    """Linear convolution a*a (full mode) using a real FFT."""
    n = a.shape[0]
    m = 2 * n - 1
    n_fft = 1 << ((m - 1).bit_length())          # next power of two ≥ m
    A = np.fft.rfft(a, n=n_fft)
    C = A * A
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


def compute_score(a: np.ndarray) -> float:
    """Exact objective for a (already normalised to sum 1): 2·n·max(conv)."""
    n = a.shape[0]
    conv = full_convolution(a)
    return float(2.0 * n * conv.max())


# ----------------------------------------------------------------------
# 1. Power‑law sweep (cheap global search)
# ----------------------------------------------------------------------


def best_power_law_vector(n: int, time_deadline: float) -> np.ndarray:
    """Search p∈[0.5,0.8] for a_i∝(i+1)^‑p, return the best normalised vector."""
    best_a = None
    best_val = float("inf")
    best_p = 0.5

    # coarse grid
    coarse_ps = np.linspace(0.5, 0.8, 61)  # step ≈0.005
    for p in coarse_ps:
        if time.time() >= time_deadline:
            break
        a = np.arange(1, n + 1, dtype=float) ** (-p)
        a /= a.sum()
        val = compute_score(a)
        if val < best_val:
            best_val, best_a, best_p = val, a, p

    # fine grid around the best p
    if best_a is not None:
        low = max(0.5, best_p - 0.025)
        high = min(0.9, best_p + 0.025)
        fine_ps = np.linspace(low, high, 31)
        for p in fine_ps:
            if time.time() >= time_deadline:
                break
            a = np.arange(1, n + 1, dtype=float) ** (-p)
            a /= a.sum()
            val = compute_score(a)
            if val < best_val:
                best_val, best_a, best_p = val, a, p

    return best_a


# ----------------------------------------------------------------------
# 2. Smooth‑max projected gradient descent
# ----------------------------------------------------------------------


def smooth_gradient_descent(a0: np.ndarray,
                            time_deadline: float,
                            step0: float = 0.05,
                            beta_start: float = 1.0,
                            beta_factor: float = 1.02,
                            beta_max: float = 200.0) -> np.ndarray:
    """
    Projected gradient descent on the smooth approximation of the max‑convolution.

    The smooth max of conv (with parameter β) is
        S_β(conv) = max_c + log(∑exp(β·(conv-max_c))) / β,
    where max_c = max(conv).  Its gradient w.r.t. a is a weighted sum of the
    usual sub‑gradients, which we compute via a correlation using FFTs.
    The algorithm increases β gradually (continuation) so that the landscape
    becomes sharper over time.
    """
    a = a0.copy()
    n = a.shape[0]
    best_a = a.copy()
    best_val = compute_score(a)

    t = 0
    beta = beta_start
    # pre‑allocate buffers for FFTs (size depends only on n)
    m = 2 * n - 1
    L = 1 << ((m + n - 1 - 1).bit_length())   # next power of two ≥ 3·n‑2

    while time.time() < time_deadline:
        t += 1

        # ----- convolution -----
        conv = full_convolution(a)                      # length 2n‑1
        max_c = conv.max()

        # ----- smooth max weight (soft‑max) -----
        # stable computation
        shifted = conv - max_c
        exp_term = np.exp(beta * shifted)
        sum_exp = exp_term.sum()
        w = exp_term / sum_exp                          # normalised weights
        # (optional) logging every 2000 steps
        if t % 2000 == 0:
            smooth_val = max_c + np.log(sum_exp) / beta
            approx_score = 2.0 * n * smooth_val
            print(f"[smooth‑GD] it={t:06d} β={beta:.2f} approx_score={approx_score:.6f}")

        # ----- correlation = weighted sub‑gradient -----
        # Pad to length L for FFT
        w_pad = np.zeros(L, dtype=float)
        w_pad[:m] = w
        a_pad = np.zeros(L, dtype=float)
        a_pad[:n] = a

        W = np.fft.rfft(w_pad)
        A = np.fft.rfft(a_pad)

        # correlation = ifft( W * conj(A) )
        corr_full = np.fft.irfft(W * np.conj(A), n=L)
        corr = corr_full[:n]                     # only first n entries needed

        # gradient of the smooth objective: 4·n·corr
        grad = 4.0 * n * corr
        grad_norm = np.linalg.norm(grad)
        if grad_norm == 0.0:
            break
        direction = grad / grad_norm

        # ----- step size (1/√t schedule) -----
        step = step0 / np.sqrt(t)

        # ----- update and projection -----
        a = a - step * direction
        a = project_to_simplex(a)

        # occasional jitter to avoid stagnation
        if t % 1000 == 0:
            jitter = np.random.randn(n) * (step0 * 0.02 / np.sqrt(t))
            a += jitter
            a = project_to_simplex(a)

        # ----- keep best exact score -----
        val = compute_score(a)
        if val < best_val:
            best_val, best_a = val, a.copy()

        # ----- continuation on β -----
        beta = min(beta_max, beta * beta_factor)

    return best_a


# ----------------------------------------------------------------------
# 3. Simple greedy hill‑climb (fine local search)
# ----------------------------------------------------------------------


def hill_climb(a_start: np.ndarray,
               time_deadline: float,
               max_no_improve: int = 800_000) -> np.ndarray:
    """
    Stochastic mass‑transfer moves with greedy acceptance.
    Move a fraction of the mass from i to j, re‑project onto the simplex,
    keep the move if it improves the exact objective.
    """
    a = a_start.copy()
    n = a.shape[0]
    best_a = a.copy()
    best_val = compute_score(a)

    step_scale = 0.05   # initial fraction of mass to move
    it = 0
    no_improve = 0

    while time.time() < time_deadline and it < max_no_improve:
        i, j = np.random.randint(n, size=2)
        if i == j:
            continue

        delta = step_scale * a[i]          # amount taken from i
        if delta == 0.0:
            it += 1
            continue

        cand = a.copy()
        cand[i] -= delta
        cand[j] += delta

        cand = np.maximum(cand, 0.0)       # safeguard tiny negatives
        cand = project_to_simplex(cand)

        val = compute_score(cand)
        if val < best_val:
            best_val, best_a = val, cand.copy()
            a = cand
            step_scale = min(step_scale * 1.07, 0.5)
            no_improve = 0
        else:
            step_scale *= 0.985
            no_improve += 1

        it += 1
        if no_improve >= 50000:
            # reduce step scale when stuck for a while
            step_scale = max(step_scale * 0.5, 1e-4)
            no_improve = 0

    return best_a


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 12345,
                      budget_s: int = 1000,
                      **kwargs):
    """
    Search for a non‑negative coefficient sequence that minimises
    ``evaluate_sequence``.  Returns a plain Python list of floats.
    """
    np.random.seed(seed)
    start_time = time.time()
    # leave a tiny safety margin for the harness
    deadline = start_time + budget_s - 5.0

    # ------------------------------------------------------------------
    n = 2048  # length of the step function
    # ------------------------------------------------------------------
    best_seq = None
    best_val = float("inf")

    # ------------------------------------------------------------------
    # 0. Optional fallback from historic construction
    # ------------------------------------------------------------------
    try:
        fallback = [float(x) for x in height_sequence_1]  # type: ignore
        val = evaluate_sequence(fallback)
        print(f"[fallback] score = {val:.6f}")
        if val < best_val:
            best_val, best_seq = val, fallback
    except Exception:
        pass

    # ------------------------------------------------------------------
    # 1. Power‑law sweep (cheap global search)
    # ------------------------------------------------------------------
    pll_deadline = deadline - 300.0
    if time.time() < pll_deadline:
        a_pow = best_power_law_vector(n, pll_deadline)
        if a_pow is not None:
            val = compute_score(a_pow)
            print(f"[power‑sweep] best p‑score = {val:.6f}")
            if val < best_val:
                best_val, best_seq = val, a_pow.tolist()

    # ------------------------------------------------------------------
    # 2. Smooth‑max projected gradient descent (multiple restarts)
    # ------------------------------------------------------------------
    gd_deadline = deadline - 150.0
    if time.time() < gd_deadline and best_seq is not None:
        # we will try a few random Dirichlet starts in addition to the best so far
        starts = [np.array(best_seq, dtype=float)]
        # generate up to 4 random Dirichlet vectors
        for _ in range(4):
            # exponential(1) sample -> Dirichlet
            rand = -np.log(np.random.rand(n) + 1e-12)
            rand /= rand.sum()
            starts.append(rand)

        # allocate time roughly equally among the starts
        per_start_deadline = gd_deadline
        for idx, a0 in enumerate(starts):
            if time.time() >= gd_deadline:
                break
            local_deadline = min(per_start_deadline, gd_deadline)
            print(f"[smooth‑GD] start {idx+1}/{len(starts)}")
            a_refined = smooth_gradient_descent(a0, local_deadline)
            val = compute_score(a_refined)
            print(f"[smooth‑GD] refined exact score = {val:.6f}")
            if val < best_val:
                best_val, best_seq = val, a_refined.tolist()
            # shrink remaining budget a little to keep room for hill‑climb
            per_start_deadline = gd_deadline - (time.time() - start_time)

    # ------------------------------------------------------------------
    # 3. Final greedy hill‑climb
    # ------------------------------------------------------------------
    hill_deadline = deadline - 5.0
    if best_seq is not None and time.time() < hill_deadline:
        a_start = np.array(best_seq, dtype=float)
        a_final = hill_climb(a_start, hill_deadline)
        val = compute_score(a_final)
        print(f"[hill‑climb] final exact score = {val:.6f}")
        if val < best_val:
            best_val, best_seq = val, a_final.tolist()

    # ------------------------------------------------------------------
    # Safeguard – fall back to a uniform vector if nothing was found
    # ------------------------------------------------------------------
    if best_seq is None:
        best_seq = [1.0 / n] * n
        best_val = evaluate_sequence(best_seq)

    # ------------------------------------------------------------------
    # Final report (using the official evaluate_sequence)
    # ------------------------------------------------------------------
    final_score = evaluate_sequence(best_seq)
    print(f"\nSearch finished. Reported score = {final_score:.6f}")
    return best_seq
