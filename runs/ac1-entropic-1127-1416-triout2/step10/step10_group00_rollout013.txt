"""
Propose a step‑function sequence that minimises

    2·n·max( a * a )   (with a on the simplex).

The algorithm combines three ideas that were missing in the previous runner:

1. **Fast analytic seeding** – a brief scan of a few low‑dimensional families
   (product, power, exponential‑U) on a fairly large length (2048 or 4096)
   to obtain a good starting point.

2. **L‑BFGS optimisation of a smooth max‑approximation** – we minimise the
   log‑sum‑exp surrogate of the max‑convolution using the exact gradient
   (derived from the soft‑max weighting).  The optimisation is performed on
   the soft‑max parametrisation `a = softmax(z)`, which automatically enforces
   non‑negativity and the simplex constraint.  A schedule of increasing
   smoothing parameters `β` drives the solution towards the true max.

3. **Final simulated‑annealing hill‑climb** – a short SA walk with symmetric
   moves refines the result and removes any remaining asymmetry.

All steps respect the 1000 s wall‑clock budget and continuously keep the best
sequence seen so far.  The code is self‑contained, uses only NumPy,
SciPy and the provided `evaluate_sequence` function, and prints progress
information.
"""

import time
import numpy as np
from scipy.optimize import minimize

# ----------------------------------------------------------------------
# Helper utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full aperiodic convolution using real FFT."""
    m = a.size + b.size - 1
    nfft = next_pow2(m)
    A = np.fft.rfft(a, nfft)
    B = np.fft.rfft(b, nfft)
    C = A * B
    conv = np.fft.irfft(C, nfft)[:m]
    return conv


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / v.shape[0])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, v.shape[0] + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.where(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def softmax(z: np.ndarray) -> np.ndarray:
    """Numerically stable soft‑max."""
    z_max = np.max(z)
    e = np.exp(z - z_max)
    return e / e.sum()


def compute_score(a: np.ndarray) -> float:
    """Exact objective for a probability vector (∑a = 1)."""
    n = a.shape[0]
    conv = convolution_fft(a, a)
    return 2.0 * n * float(conv.max())


# ----------------------------------------------------------------------
# Analytic families (used for fast seeding)
# ----------------------------------------------------------------------


def symmetric_product_family(n: int, alpha: float) -> np.ndarray:
    """ a_i ∝ [(i+½)(n‑½‑i)]^{‑α}, symmetric, on the simplex. """
    i = np.arange(n, dtype=np.float64)
    logprod = np.log(i + 0.5) + np.log(n - 0.5 - i)
    w = np.exp(-alpha * logprod)
    w[np.isinf(w)] = np.finfo(np.float64).max
    return w / w.sum()


def left_heavy_power_family(n: int, beta: float) -> np.ndarray:
    """ a_i ∝ (i+½)^{‑β}, monotone decreasing. """
    i = np.arange(n, dtype=np.float64)
    w = np.exp(-beta * np.log(i + 0.5))
    return w / w.sum()


def exp_u_family(n: int, gamma: float) -> np.ndarray:
    """U‑shape: a_i ∝ exp(-γ·min(i, n‑1‑i))."""
    i = np.arange(n, dtype=np.float64)
    dist = np.minimum(i, n - 1 - i)
    w = np.exp(-gamma * dist)
    return w / w.sum()


# ----------------------------------------------------------------------
# Smooth‑max objective and gradient (used by L‑BFGS)
# ----------------------------------------------------------------------


def smooth_max_and_grad(z: np.ndarray, beta: float, n: int):
    """
    Returns (f, grad) where

        f   = smooth max = max(conv) + log‑sum‑exp(beta·(conv‑max))/beta
        grad = ∂f/∂z  (gradient w.r.t. the soft‑max parameters)

    The gradient is derived analytically, avoiding finite‑difference noise.
    """
    a = softmax(z)
    conv = convolution_fft(a, a)                     # length 2n‑1
    maxc = conv.max()
    # stable soft‑max weights
    expw = np.exp(beta * (conv - maxc))
    sumexp = expw.sum()
    f = maxc + np.log(sumexp) / beta                 # smooth approximation of max
    w = expw / sumexp                                # normalized soft‑max over positions

    # weighted sum of the convolution gradients
    a_rev = a[::-1]
    conv_rev = convolution_fft(a_rev, w)            # length 3n‑2, we need central slice
    grad_a = 2.0 * conv_rev[n - 1:n - 1 + n]        # shape (n,)

    # back‑prop through the soft‑max
    dot_ag = np.dot(a, grad_a)
    grad_z = a * grad_a - dot_ag * a
    return f, grad_z


# ----------------------------------------------------------------------
# L‑BFGS optimisation with a schedule of β values
# ----------------------------------------------------------------------


def l_bfgs_optimize(a_start: np.ndarray,
                    betas,
                    max_iter_per_stage: int,
                    deadline: float):
    """
    Repeated L‑BFGS runs on the smooth‑max surrogate.
    `betas` is a list of smoothing parameters (increasing).
    The function respects the global deadline and returns the best a
    found (still on the simplex and symmetrised).
    """
    n = a_start.shape[0]
    a = a_start.copy()
    best_a = a.copy()
    best_score = compute_score(a)

    for beta in betas:
        if time.time() >= deadline:
            break
        print(f"[{time.time():.1f}] L‑BFGS stage with β={beta}")

        # initialise the optimisation variables
        z0 = np.log(a + 1e-12)

        # objective wrapper for scipy
        def fun(z):
            f, g = smooth_max_and_grad(z, beta, n)
            return f, g

        # limit the number of internal L‑BFGS iterations so we never exceed deadline
        maxit = max_iter_per_stage
        opts = {"maxiter": maxit, "disp": False}

        try:
            res = minimize(fun,
                           z0,
                           method="L-BFGS-B",
                           jac=True,
                           options=opts)
        except Exception as e:
            print(f"  L‑BFGS failure: {e}")
            continue

        a = softmax(res.x)

        # enforce symmetry at every stage (helps numerical stability)
        a = (a + a[::-1]) * 0.5
        a = project_to_simplex(a)

        sc = compute_score(a)
        if sc < best_score:
            best_score = sc
            best_a = a.copy()
            print(f"[{time.time():.1f}]    improved to {best_score:.6f}")

    return best_a, best_score


# ----------------------------------------------------------------------
# Simple simulated‑annealing hill‑climb (final polishing)
# ----------------------------------------------------------------------


def hill_climb_sa(start_a: np.ndarray,
                  deadline: float,
                  max_steps: int = 2_000_000) -> tuple[np.ndarray, float]:
    """
    Random walk on the simplex with occasional uphill moves.
    50 % swap, 50 % mass transfer.  Temperature decays linearly.
    """
    n = start_a.size
    cur_a = start_a.copy()
    cur_score = compute_score(cur_a)

    best_a = cur_a.copy()
    best_score = cur_score

    T0 = best_score * 1e-4
    step_scale = 0.12

    for step in range(1, max_steps + 1):
        if time.time() >= deadline:
            break

        cand = cur_a.copy()
        if np.random.rand() < 0.5:                     # swap two entries
            i, j = np.random.randint(n, size=2)
            cand[i], cand[j] = cand[j], cand[i]
        else:                                          # move a fraction of mass
            i, j = np.random.randint(n, size=2)
            if i != j:
                delta = step_scale * cand[i]
                cand[i] -= delta
                cand[j] += delta
                if cand[i] < 0.0:
                    cand[i] = 0.0

        cand = project_to_simplex(cand)
        cand_score = compute_score(cand)

        delta = cand_score - cur_score
        if delta < 0 or np.random.rand() < np.exp(-delta / (T0 * (1 - step / max_steps) + 1e-18)):
            cur_a = cand
            cur_score = cand_score
            if cur_score < best_score:
                best_score = cur_score
                best_a = cur_a.copy()
                step_scale = min(step_scale * 1.07, 0.5)
        else:
            step_scale *= 0.97

    return best_a, best_score
