"""
Algorithm Overview
------------------
1. Load the previous best construction (global ``height_sequence_1``) if it
   exists and use it as a starting point.
2. Perform a series of *random restarts*:
   – each restart begins from a randomly generated non‑negative vector on
     the probability simplex,
   – the vector is refined by L‑BFGS‑B on a *smooth‑max* surrogate of the
     true objective 2·n·max(a*a) .  The surrogate is
          f_β(a) = max(a*a) + (1/β)·log∑exp(β·((a*a)-max(a*a))),
     which becomes the exact max as β→∞.
   – the variables are parametrised by logits `z` (a = softmax(z))
     so the simplex and non‑negativity constraints are satisfied automatically.
3. Keep the best candidate found over all restarts.
4. *Continuation*: increase the smoothing parameter β step‑wise
   (β = 200 → 800 → 3000 → 8000) and run a short L‑BFGS refinement at each
   level, starting from the current best.
5. Run a short projected sub‑gradient descent on the true max‑objective
   for fine‑tuning.
6. Enforce exact symmetry (a ← (a+a[::-1])/2) and a final simplex projection.
7. Return the resulting list of heights.

The heavy lifting (FFT‑based convolution, softmax, simplex projection,
smooth‑max objective and its gradient) are implemented as top‑level
functions so that no closures are used.
"""

import time
import numpy as np
from scipy.optimize import minimize

# ----------------------------------------------------------------------
# Core utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full aperiodic convolution using a real FFT (zero‑padded)."""
    m = a.size + b.size - 1
    nfft = next_pow2(m)
    A = np.fft.rfft(a, nfft)
    B = np.fft.rfft(b, nfft)
    C = A * B
    conv = np.fft.irfft(C, nfft)[:m]
    return conv


def softmax(z: np.ndarray) -> np.ndarray:
    """Numerically stable soft‑max (produces a probability vector)."""
    z_max = np.max(z)
    e = np.exp(z - z_max)
    return e / np.sum(e)


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex (sum = 1, ≥0)."""
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / v.shape[0])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, v.shape[0] + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.where(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def compute_score(a: np.ndarray) -> float:
    """Exact 2·n·max(a*a) for a probability vector (∑a = 1)."""
    n = a.shape[0]
    conv = convolution_fft(a, a)
    return 2.0 * n * float(conv.max())


# ----------------------------------------------------------------------
# Random initialisers (on the simplex)
# ----------------------------------------------------------------------


def random_sparse_start(n: int) -> np.ndarray:
    """Very sparse vector (~5 % non‑zeros)."""
    sparsity = np.random.uniform(0.03, 0.07)
    k = max(2, int(round(sparsity * n)))
    idx = np.random.choice(n, size=k, replace=False)
    a = np.zeros(n, dtype=np.float64)
    a[idx] = 1.0 / k
    return project_to_simplex(a)


def random_monotone_start(n: int) -> np.ndarray:
    """Decreasing shape obtained by sorting a powered uniform."""
    r = np.random.rand(n) ** np.random.uniform(0.5, 5.0)
    r.sort()
    a = r[::-1]
    return a / a.sum()


def random_u_start(n: int) -> np.ndarray:
    """Exponentially decaying \"U\" shape (large mass at the ends)."""
    gamma = np.random.uniform(0.0, 6.0)
    i = np.arange(n, dtype=np.float64)
    dist = np.minimum(i, n - 1 - i)
    w = np.exp(-gamma * dist)
    return w / w.sum()


# ----------------------------------------------------------------------
# Smooth‑max surrogate objective (used by L‑BFGS)
# ----------------------------------------------------------------------


class SmoothMaxObjective:
    """
    Callable object evaluating the smoothed max of the auto‑convolution
    together with its gradient w.r.t. the logits `z` (where a = softmax(z)).
    """

    def __init__(self, n: int, beta: float):
        self.n = n
        self.beta = beta

    def __call__(self, z: np.ndarray):
        a = softmax(z)                         # probability vector
        conv = convolution_fft(a, a)           # a*a
        maxc = conv.max()

        # smooth approximation of the max (log‑sum‑exp trick)
        exp_term = np.exp(self.beta * (conv - maxc))
        sum_exp = exp_term.sum()
        f = maxc + np.log(sum_exp) / self.beta  # = (1/β)·log Σexp(β·conv)

        # gradient of f w.r.t. a (using the soft‑max weights)
        w = exp_term / sum_exp                 # soft‑max over convolution entries
        a_rev = a[::-1]
        conv_rev = convolution_fft(a_rev, w)   # (a_rev * w)

        # gradient w.r.t. a (2·a_rev * w) for the convolution part
        grad_a = 2.0 * conv_rev[self.n - 1:self.n - 1 + self.n]

        # back‑prop through the soft‑max parametrisation
        dot_ag = np.dot(a, grad_a)
        grad_z = a * grad_a - dot_ag * a
        return f, grad_z


# ----------------------------------------------------------------------
# L‑BFGS optimisation wrapper
# ----------------------------------------------------------------------


def lbfgs_optimize(start_a: np.ndarray,
                   beta: float,
                   max_iter: int,
                   deadline: float) -> tuple[np.ndarray, float]:
    """
    Refine `start_a` (a probability vector) by L‑BFGS on the smoothed‑max
    surrogate with smoothing parameter `beta`.
    Returns the refined vector and its exact score (2·n·max).
    """
    if time.time() >= deadline:
        return start_a, compute_score(start_a)

    n = start_a.shape[0]
    eps = 1e-12
    z0 = np.log(start_a + eps)          # logits, guaranteed finite

    objective = SmoothMaxObjective(n, beta)

    # L‑BFGS‑B with gradient supplied via `objective`; we keep it simple
    res = minimize(fun=objective,
                   x0=z0,
                   method='L-BFGS-B',
                   jac=True,
                   options={'maxiter': max_iter, 'disp': False})

    a_opt = softmax(res.x)
    return a_opt, compute_score(a_opt)


# ----------------------------------------------------------------------
# Projected sub‑gradient descent on the true max‑objective
# ----------------------------------------------------------------------


def subgrad_optimize(start_a: np.ndarray,
                     deadline: float,
                     max_iters: int = 200_000,
                     step0: float = 0.06) -> tuple[np.ndarray, float]:
    """
    Classic sub‑gradient descent for the exact max‑objective.
    The iterate is projected onto the simplex after each step.
    """
    n = start_a.shape[0]
    a = start_a.astype(np.float64).copy()
    best_a = a.copy()
    best_score = compute_score(a)

    for t in range(1, max_iters + 1):
        if time.time() >= deadline:
            break

        conv = convolution_fft(a, a)
        k = int(np.argmax(conv))

        # sub‑gradient of the max‑entry (see analysis)
        idx = np.arange(n)
        j = k - idx
        mask = (j >= 0) & (j < n)
        g = np.zeros_like(a)
        g[mask] = 2.0 * a[j[mask]]

        step = step0 / np.sqrt(t)
        a = project_to_simplex(a - step * g)

        if t % 2000 == 0:
            sc = compute_score(a)
            if sc < best_score:
                best_score = sc
                best_a = a.copy()
                print(f"[{time.time():.1f}] sub‑grad improved to {best_score:.6f}")

    sc = compute_score(a)
    if sc < best_score:
        best_score = sc
        best_a = a.copy()
    return best_a, best_score


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 12345,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function sequence minimising
    ``evaluate_sequence`` under the simplex constraint.
    Uses random restarts + L‑BFGS on a smoothed‑max surrogate,
    a continuation in the smoothing parameter and a final
    sub‑gradient polishing.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 2.0   # safety margin

    # ------------------------------------------------------------------
    # 0. Fallback to the previously found construction (if any)
    # ------------------------------------------------------------------
    best_seq = None
    best_score = float("inf")
    try:
        fb_vec = [float(x) for x in height_sequence_1]  # type: ignore
        fb = evaluate_sequence(fb_vec)
        print(f"[{time.time():.1f}] fallback score = {fb:.6f}")
        if fb < best_score:
            best_score = fb
            best_seq = fb_vec
    except Exception:
        pass

    # ------------------------------------------------------------------
    # 1. Determine problem size (use fallback length if possible)
    # ------------------------------------------------------------------
    if best_seq is not None:
        n = len(best_seq)
    else:
        n = 2048                     # a reasonable default
    print(f"[{time.time():.1f}] problem dimension n = {n}")

    # Ensure we are working with a proper probability vector
    if best_seq is not None:
        a0 = np.array(best_seq, dtype=np.float64)
        a0 = project_to_simplex(a0)
    else:
        a0 = np.full(n, 1.0 / n, dtype=np.float64)

    # ------------------------------------------------------------------
    # 2. Random‑restart L‑BFGS optimisation (coarse β)
    # ------------------------------------------------------------------
    num_restarts = 12
    beta_start = 200.0
    max_iter_restart = 300

    print(f"[{time.time():.1f}] starting {num_restarts} random restarts (β={beta_start})")
    generators = [random_sparse_start, random_monotone_start, random_u_start]

    for r in range(num_restarts):
        if time.time() >= deadline - 100.0:
            break
        init = np.random.choice(generators)(n)
        init = project_to_simplex(init)
        a_opt, sc = lbfgs_optimize(init, beta_start, max_iter_restart, deadline)
        # Symmetrise before evaluation (the optimum appears symmetric)
        a_opt = (a_opt + a_opt[::-1]) * 0.5
        a_opt = project_to_simplex(a_opt)
        sc = compute_score(a_opt)
        if sc < best_score:
            best_score = sc
            best_seq = a_opt.tolist()
            print(f"[{time.time():.1f}] new best after restart #{r+1} = {best_score:.6f}")

    # ------------------------------------------------------------------
    # 3. Continuation in β (refine the current best)
    # ------------------------------------------------------------------
    continuation_betas = [800.0, 3000.0, 8000.0]
    max_iter_refine = 600

    for beta in continuation_betas:
        if best_seq is None or time.time() >= deadline - 80.0:
            break
        print(f"[{time.time():.1f}] continuation refinement with β = {beta}")
        a_start = np.array(best_seq, dtype=np.float64)
        a_opt, sc = lbfgs_optimize(a_start, beta, max_iter_refine, deadline)
        a_opt = (a_opt + a_opt[::-1]) * 0.5
        a_opt = project_to_simplex(a_opt)
        sc = compute_score(a_opt)
        if sc < best_score:
            best_score = sc
            best_seq = a_opt.tolist()
            print(f"[{time.time():.1f}] improved after β={beta} refinement = {best_score:.6f}")

    # ------------------------------------------------------------------
    # 4. Short projected sub‑gradient descent for final polishing
    # ------------------------------------------------------------------
    if best_seq is not None and time.time() < deadline - 30.0:
        print(f"[{time.time():.1f}] final sub‑gradient polishing")
        a_sub, sc_sub = subgrad_optimize(
            np.array(best_seq, dtype=np.float64),
            deadline - 5.0,
            max_iters=150_000,
            step0=0.04,
        )
        if sc_sub < best_score:
            best_score = sc_sub
            best_seq = a_sub.tolist()
            print(f"[{time.time():.1f}] after sub‑grad best = {best_score:.6f}")

    # ------------------------------------------------------------------
    # 5. Explicit symmetrisation and final safety projection
    # ------------------------------------------------------------------
    if best_seq is not None:
        a_arr = np.array(best_seq, dtype=np.float64)
        a_arr = (a_arr + a_arr[::-1]) * 0.5
        a_arr = project_to_simplex(a_arr)
        best_seq = a_arr.tolist()
        best_score = evaluate_sequence(best_seq)  # final true evaluation
        print(f"[{time.time():.1f}] final reported score = {best_score:.6f}")

    # ------------------------------------------------------------------
    # 6. Fallback to a trivial uniform vector if everything failed
    # ------------------------------------------------------------------
    if best_seq is None:
        n_small = 256
        best_seq = [1.0 / n_small] * n_small
        best_score = evaluate_sequence(best_seq)
        print(f"[{time.time():.1f}] fallback to uniform (score = {best_score:.6f})")

    return best_seq
