def simulated_annealing(a_start, time_deadline, init_temp=1e-3, final_temp=1e-6, cooling_factor=0.9995, step_scale_initial=0.05, step_scale_min=1e-4):

    n = len(a)
    cur_val = compute_score(a)
    best_a = a.copy()
    best_val = cur_val
    step_scale = step_scale_initial
    T = init_temp
    iter =0
    while time.time() < time_deadline:
        # propose move
        i = np.random.randint(n)
        j = np.random.randint(n)
        if i == j or a[i] <= 0.0:
            continue
        delta = step_scale * a[i]  # move fraction
        if delta <= 0:
            continue
        cand = a.copy()
        cand[i] -= delta
        cand[j] += delta
        # ensure non-negative due to float error
        cand = np.maximum(cand, 0.0)
        cand = project_to_simplex(cand)
        val = compute_score(cand)

        if delta_val < 0 or np.random.rand() < np.exp(-delta_val / max(T, 1e-12)):
            # accept
            a = cand
            cur_val = val
            # adjust step scale maybe
            step_scale = min(step_scale * 1.02, 0.5)

                best_val = cur_val
                best_a = a.copy()
        else:
            step_scale = max(step_scale * 0.98, step_scale_min)

        vec = np.maximum(vec, 0.0)
        # project to simplex to ensure sum=1
        vec = project_to_simplex(vec)

        sc = compute_score(vec)
        # keep
        if len(best) < max_keep:
            best.append((sc, vec.copy()))
            best.sort(key=lambda x: x[0])
        else:
            if sc < best[-1][0]:
                best[-1] = (sc, vec.copy())
                best.sort(key=lambda x: x[0])

    # 1) power law grid

    for lam in lam_vals:
        if time.time() >= deadline:
            break

"""
Non‑negative step‑function optimiser for ``evaluate_sequence``.

Algorithm
---------
1. **Global exploration** – sweep three parametric families
   (power‑law, exponential decay and beta‑shaped) plus a pool of
   random Dirichlet samples.  The best few candidates (by the exact
   objective 2·n·max(conv(a,a))) are retained.

2. **Projected sub‑gradient polishing** – from the best candidate run a
   sub‑gradient descent on the true max‑convolution.  All active
   indices (those attaining the current maximum up to a tiny tolerance)
   contribute to the gradient, which yields a smoother descent toward a
   flatter autocorrelation.

3. **Simulated annealing** – a Metropolis‑style hill‑climb with a
   decreasing temperature that accepts occasional worsening moves.
   Small random mass‑transfer steps between coordinates are performed,
   always projecting back onto the probability simplex.

The three phases are timed to respect the 1000 s wall‑clock budget.
The final sequence (a plain Python list of floats) is returned.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Numerical utilities
# ----------------------------------------------------------------------


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    if np.all(v <= 0):
        # Uniform distribution is the projection of the zero vector.
        return np.full_like(v, 1.0 / v.shape[0])

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, v.shape[0] + 1)
    mask = u - cssv / ind > 0
    rho = np.where(mask)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def full_convolution(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a using a real FFT (length 2·n‑1)."""
    n = a.shape[0]
    m = 2 * n - 1
    n_fft = 1 << ((m - 1).bit_length())          # next power of two ≥ m
    A = np.fft.rfft(a, n=n_fft)
    C = A * A
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


def compute_score(a: np.ndarray) -> float:
    """Exact objective for a normalised vector (sum = 1):
       2·n·max(conv(a,a)).
    """
    n = a.shape[0]
    conv = full_convolution(a)
    return float(2.0 * n * conv.max())


# ----------------------------------------------------------------------
# Parametric families
# ----------------------------------------------------------------------


def generate_powerlaw(n: int, p: float) -> np.ndarray:
    """a_i ∝ (i+1)^{-p}."""
    a = np.arange(1, n + 1, dtype=float) ** (-p)
    total = a.sum()
    if total == 0 or not np.isfinite(total):
        return np.full(n, 0.0)
    return a / total


def generate_exponential(n: int, lam: float) -> np.ndarray:
    """a_i ∝ exp(-lam·i)."""
    a = np.exp(-lam * np.arange(n, dtype=float))
    total = a.sum()
    if total == 0 or not np.isfinite(total):
        return np.full(n, 0.0)
    return a / total


def generate_beta(n: int, alpha: float, beta: float) -> np.ndarray:
    """a_i ∝ (i+1)^{α-1}·(n-i)^{β-1}  (discrete Beta‑shaped density)."""
    i = np.arange(1, n + 1, dtype=float)
    left = np.power(i, alpha - 1.0)
    right = np.power(n - i + 1.0, beta - 1.0)
    a = left * right
    if not np.all(np.isfinite(a)):
        return np.full(n, 0.0)
    total = a.sum()
    if total == 0 or not np.isfinite(total):
        return np.full(n, 0.0)
    return a / total


# ----------------------------------------------------------------------
# Global search (baseline)
# ----------------------------------------------------------------------


def baseline_search(n: int, deadline: float, max_keep: int = 5):
    """Explore several families and random Dirichlet draws.
    Returns up to ``max_keep`` best vectors (as numpy arrays)."""
    best = []                     # list of (score, vector)

    def try_candidate(vec: np.ndarray):
        if not np.isfinite(vec).all():
            return
        vec = np.maximum(vec, 0.0)
        vec = project_to_simplex(vec)
        sc = compute_score(vec)
        if len(best) < max_keep:
            best.append((sc, vec.copy()))
            best.sort(key=lambda x: x[0])
        elif sc < best[-1][0]:
            best[-1] = (sc, vec.copy())
            best.sort(key=lambda x: x[0])

    # 1) power‑law sweep
    for p in np.linspace(0.4, 0.9, 101):          # 101 points
        if time.time() >= deadline:
            break
        try_candidate(generate_powerlaw(n, p))

    # 2) exponential sweep
    for lam in np.linspace(0.0001, 0.02, 200):
        if time.time() >= deadline:
            break
        try_candidate(generate_exponential(n, lam))

    # 3) Beta sweep (coarse grid)
    al_vals = np.linspace(0.2, 4.0, 19)          # step ≈0.2
    be_vals = np.linspace(0.2, 4.0, 19)
    for alpha in al_vals:
        for beta in be_vals:
            if time.time() >= deadline:
                break
            try_candidate(generate_beta(n, alpha, beta))

    # 4) Random Dirichlet draws – different concentration parameters
    concs = [0.2, 0.5, 1.0, 2.0]
    samples_per = 80                            # total ≈ 320 draws
    for conc in concs:
        for _ in range(samples_per):
            if time.time() >= deadline:
                break
            a = np.random.dirichlet([conc] * n)
            try_candidate(a)

    best.sort(key=lambda x: x[0])
    return [vec for sc, vec in best]


# ----------------------------------------------------------------------
# Projected sub‑gradient polishing
# ----------------------------------------------------------------------


def subgradient_polish(a0: np.ndarray, deadline: float,
                       step0: float = 0.30) -> np.ndarray:
    """Projected sub‑gradient descent on 2·n·max(conv(a,a)).
    All indices attaining the current maximum (within 1e‑9 relative)
    contribute to the gradient."""
    a = a0.copy()
    n = a.shape[0]
    best_a = a.copy()
    best_val = compute_score(a)

    t = 0
    while time.time() < deadline:
        t += 1
        conv = full_convolution(a)
        max_val = conv.max()
        eps = max_val * 1e-9
        active = np.where(conv >= max_val - eps)[0]

        # accumulate gradient from all active entries
        g = np.zeros_like(a)
        idx = np.arange(n)
        for k in active:
            j = k - idx
            mask = (j >= 0) & (j < n)
            g[mask] += 2.0 * a[j[mask]]

        step = step0 / np.sqrt(t)
        a -= step * g
        a = project_to_simplex(a)

        cur_val = compute_score(a)
        if cur_val < best_val:
            best_val = cur_val
            best_a = a.copy()

        # occasional tiny jitter to avoid stagnation
        if t % 2000 == 0:
            jitter = np.random.randn(n) * (step0 * 0.005 / np.sqrt(t))
            a += jitter
            a = project_to_simplex(a)

    return best_a


# ----------------------------------------------------------------------
# Simulated annealing (stochastic hill‑climb with temperature)
# ----------------------------------------------------------------------


def simulated_annealing(a_start: np.ndarray, deadline: float,
                        init_temp: float = 1e-3,
                        final_temp: float = 1e-6,
                        cooling_factor: float = 0.9995,
                        step_scale_init: float = 0.05,
                        step_scale_min: float = 1e-5) -> np.ndarray:
    """Metropolis hill‑climb that can accept worsening moves.
    Small mass‑transfer steps are proposed and the current temperature
    decays exponentially."""
    a = a_start.copy()
    n = a.shape[0]
    cur_val = compute_score(a)
    best_a = a.copy()
    best_val = cur_val

    T = init_temp
    step_scale = step_scale_init
    it = 0

    while time.time() < deadline:
        it += 1
        i = np.random.randint(n)
        if a[i] <= 0.0:
            continue
        j = np.random.randint(n)
        if i == j:
            continue

        delta = step_scale * a[i]
        cand = a.copy()
        cand[i] -= delta
        cand[j] += delta
        cand = np.maximum(cand, 0.0)
        cand = project_to_simplex(cand)

        val = compute_score(cand)
        dval = val - cur_val

        if dval < 0 or np.random.rand() < np.exp(-dval / max(T, 1e-12)):
            # accept
            a = cand
            cur_val = val
            step_scale = min(step_scale * 1.02, 0.5)
            if cur_val < best_val:
                best_val = cur_val
                best_a = a.copy()
        else:
            # reject
            step_scale = max(step_scale * 0.98, step_scale_min)

        # temperature schedule
        T = max(final_temp, T * cooling_factor)

        # every few thousand iterations add a global jitter
        if it % 5000 == 0:
            jitter = np.random.randn(n) * (step_scale * 0.01)
            a += jitter
            a = project_to_simplex(a)
            cur_val = compute_score(a)

    return best_a
