```
def upsample_vector(vec, new_n):
    old_n = len(vec)
    old_x = np.linspace(0, 1, old_n, endpoint=False)
    new_x = np.linspace(0, 1, new_n, endpoint=False)
    new_vec = np.interp(new_x, old_x, vec, left=0.0, right=0.0)
    # ensure non-negative
    new_vec = np.maximum(new_vec, 0.0)
    # project onto simplex (so sum=1)
    return project_to_simplex(new_vec)
```

```
def hill_climb_targeted(start_a, deadline, max_steps=500_000):
    n = start_a.shape[0]
    a = start_a.copy()
    # ensure sum=1
    a = a / a.sum()
    conv = convolution_fft(a, a)
    cur_score = 2.0 * n * conv.max()  # compute internal
    best_a = a.copy()
    best_score = cur_score
    T0 = cur_score * 1e-4
    T = T0
    step_scale = 0.05
    
    for step in range(1, max_steps+1):
        if time.time() >= deadline:
            break
        # compute current max conv index
        conv = convolution_fft(a, a)
        M = conv.max()
        k = int(np.argmax(conv))
        # Build list of contributing indices
        # compute contributions array
        idx = np.arange(n)
        j = k - idx
        mask = (j >=0) & (j < n)
        contrib = np.zeros_like(a)
        contrib[mask] = a[idx[mask]] * a[j[mask]]
        # Choose i with probability proportional to contribution (biased to large contributions)
        if contrib.sum() == 0:
            # fallback random
            i = np.random.randint(n)
        else:
            prob = contrib / contrib.sum()
            i = np.random.choice(n, p=prob)
        j = k - i
        # Ensure valid j
        if not (0 <= j < n):
            # fallback random partner
            j = np.random.randint(n)
        # Choose operation: transfer from i (or j) to random other index r
        if np.random.rand() < 0.5 and a[i] > 1e-12:
            # move epsilon from i
            epsilon = step_scale * a[i]
            r = np.random.randint(n)
            # avoid moving to same index
            while r == i:
                r = np.random.randint(n)
            # make transfer
            cand = a.copy()
            cand[i] -= epsilon
            cand[r] += epsilon
            if cand[i] < 0:
                cand[i] = 0.0
        else:
            # move from j if valid
```

assistantfinal**Solution Overview**

The goal is to minimise  

\[
\text{score}(a)=\frac{2\,n\;\max (a*a)}{(\sum_i a_i)^2},
\]

where `a*a` is the aperiodic convolution of the non‑negative vector `a`.
Because the score is invariant under a global scaling of `a`, we may work on the
simplex `∑a_i = 1`.  In this setting the objective reduces to  

\[
F(a)=2\,n\;\max (a*a) .
\]

The programme follows a **multistage strategy** that repeatedly improves the
best known vector while respecting the time budget (≈ 1000 s).

1. **Coarse analytic scan** – a few well‑known parametric families
   (power, product, exponential‑U, Beta, mixtures) are sampled on
   `n = 1024` and `n = 2048`.  The best candidate is kept.

2. **Smooth‑max Adam optimisation** – we minimise the smooth approximation  

   \[
   f_\beta(a)=\frac1\beta\log\!\Bigl(\sum_k e^{\beta\,(a*a)_k}\Bigr)
   \]

   by Adam on the soft‑max parametrisation `a = softmax(z)`.  
   Two passes are performed:
   * a **long first pass** (β from 5 → 2000) without any enforced symmetry,
   * a **short second pass** (β ≈ 50 → 3000) for fine‑tuning.

3. **Projected sub‑gradient descent** on the true (non‑smooth) max‑objective
   provides a cheap, complementary descent direction.

4. **Upsampling** – the best solution for `n=2048` is linearly interpolated to
   `n=4096`.  The larger discretisation reduces the discretisation error of the
   continuous limit, so a further Adam run (again without symmetrisation) can
   exploit the extra degrees of freedom.

5. **Targeted simulated‑annealing hill‑climb** – instead of random moves,
   we repeatedly locate the index `k*` where the convolution attains its
   maximum and preferentially move a small amount of mass out of the “heavy”
   contributing pair `(i, k*−i)`.  A temperature schedule permits occasional
   uphill moves, allowing the algorithm to escape shallow local minima.

6. **Final evaluation** – the best vector found is handed to the given
   `evaluate_sequence` function (which applies the required clipping).

All steps respect the wall‑clock limit; early termination is performed whenever
the remaining budget is insufficient for the next stage.

```python
"""
Propose a sequence of non‑negative heights that minimises the
auto‑convolution objective used by `evaluate_sequence`.

The search proceeds in several stages:
  1. Scan a few analytic families (power, product, U‑shape, Beta) on
     n = 1024 and n = 2048 to obtain a good initial vector.
  2. Refine the candidate with Adam on a smooth approximation of the
     max‑convolution (no forced symmetry).
  3. A short projected sub‑gradient descent on the true max‑objective.
  4. Upsample the best 2048‑vector to 4096 entries and run a further
     Adam optimisation.
  5. Targeted simulated‑annealing hill‑climb that moves mass out of the
     pair of indices contributing most to the current maximal convolution.
  6. Return the best vector (sum is arbitrary – the score is scale‑invariant).

All helpers are defined at module level; no closures or file I/O are used.
"""

import time
import numpy as np
from typing import List, Tuple

# ----------------------------------------------------------------------
# Core utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full aperiodic convolution using real FFT (real‑valued inputs)."""
    m = a.size + b.size - 1
    nfft = next_pow2(m)
    A = np.fft.rfft(a, nfft)
    B = np.fft.rfft(b, nfft)
    C = A * B
    conv = np.fft.irfft(C, nfft)[:m]
    return conv


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection onto the probability simplex (sum = 1, v ≥ 0)."""
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / v.shape[0])

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, v.shape[0] + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.where(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def softmax(z: np.ndarray) -> np.ndarray:
    """Numerically stable soft‑max."""
    z_max = np.max(z)
    e = np.exp(z - z_max)
    return e / e.sum()


def compute_score(a: np.ndarray) -> float:
    """Exact objective for a probability vector (∑a = 1)."""
    n = a.shape[0]
    conv = convolution_fft(a, a)
    return 2.0 * n * float(conv.max())


# ----------------------------------------------------------------------
# Analytic families (used for the coarse scan)
# ----------------------------------------------------------------------


def symmetric_product_family(n: int, alpha: float) -> np.ndarray:
    """a_i ∝ [(i+½)(n‑½‑i)]^{‑α}, symmetric, on the simplex."""
    i = np.arange(n, dtype=np.float64)
    logprod = np.log(i + 0.5) + np.log(n - 0.5 - i)
    w = np.exp(-alpha * logprod)
    w[np.isinf(w)] = np.finfo(np.float64).max
    return w / w.sum()


def left_heavy_power_family(n: int, beta: float) -> np.ndarray:
    """a_i ∝ (i+½)^{‑β}, monotone decreasing."""
    i = np.arange(n, dtype=np.float64)
    w = np.exp(-beta * np.log(i + 0.5))
    return w / w.sum()


def exp_u_family(n: int, gamma: float) -> np.ndarray:
    """U‑shape: a_i ∝ exp(-γ·min(i, n‑1‑i))."""
    i = np.arange(n, dtype=np.float64)
    dist = np.minimum(i, n - 1 - i)
    w = np.exp(-gamma * dist)
    return w / w.sum()


def mixed_uniform_u_family(n: int, lam: float, gamma: float) -> np.ndarray:
    """Convex mixture of uniform and exponential‑U shape."""
    uniform = np.full(n, 1.0 / n, dtype=np.float64)
    u_shape = exp_u_family(n, gamma)
    return lam * uniform + (1.0 - lam) * u_shape


def beta_family(n: int, p: float, q: float) -> np.ndarray:
    """Beta‑type density ∝ (i+½)^{p‑1}·(n‑½‑i)^{q‑1}."""
    i = np.arange(n, dtype=np.float64)
    w = np.power(i + 0.5, p - 1.0) * np.power(n - 0.5 - i, q - 1.0)
    w[w < 0] = 0.0
    return w / w.sum()


# ----------------------------------------------------------------------
# Random initialisers
# ----------------------------------------------------------------------


def random_sparse_start(n: int) -> np.ndarray:
    """Very sparse start (~5 % non‑zeros)."""
    sparsity = np.random.uniform(0.03, 0.07)
    k = max(2, int(round(sparsity * n)))
    idx = np.random.choice(n, size=k, replace=False)
    a = np.zeros(n, dtype=np.float64)
    a[idx] = 1.0 / k
    return project_to_simplex(a)


def random_monotone_start(n: int) -> np.ndarray:
    """Decreasing vector obtained by sorting a powered uniform."""
    r = np.random.rand(n) ** np.random.uniform(0.5, 5.0)
    r.sort()
    a = r[::-1]
    return a / a.sum()


def random_u_start(n: int) -> np.ndarray:
    """Random exponential‑U shape."""
    gamma = np.random.uniform(0.0, 6.0)
    return exp_u_family(n, gamma)


# ----------------------------------------------------------------------
# Smoothed‑max Adam optimiser (no forced symmetry)
# ----------------------------------------------------------------------


def adam_smooth_optimize(start_a: np.ndarray,
                         deadline: float,
                         max_iters: int = 300_000,
                         beta_start: float = 5.0,
                         beta_end: float = 2000.0,
                         lr: float = 0.018,
                         sym_interval: int = 0) -> Tuple[np.ndarray, float]:
    """
    Minimises the smooth approximation
        f_β(a) = (1/β)·log Σ_k exp(β·(a*a)_k)
    using Adam on the soft‑max parametrisation a = softmax(z).

    Parameters
    ----------
    start_a      – probability vector (∑=1)
    deadline     – wall‑clock time at which to stop
    max_iters    – maximal number of iterations (subject to deadline)
    beta_start   – starting smoothing parameter
    beta_end     – final smoothing parameter (exponential schedule)
    lr           – Adam learning rate
    sym_interval – if >0, symmetrise a every `sym_interval` steps;
                   0 disables symmetrisation (allows asymmetric vectors).

    Returns
    -------
    best_a, best_score on the simplex (score = 2·n·max(conv)).
    """
    n = start_a.shape[0]
    eps = 1e-12
    # initialise log‑probabilities
    z = np.log(start_a + eps)

    m = np.zeros_like(z)
    v = np.zeros_like(z)

    beta1 = 0.9
    beta2 = 0.999
    eps_adam = 1e-12

    best_a = softmax(z)
    best_score = compute_score(best_a)

    for it in range(1, max_iters + 1):
        if time.time() >= deadline:
            break

        # exponential schedule for β
        t = it / max_iters
        beta = beta_start * (beta_end / beta_start) ** t

        a = softmax(z)

        if sym_interval and (it % sym_interval == 0):
            a = (a + a[::-1]) * 0.5
            a = project_to_simplex(a)
            z = np.log(a + eps)

        # full convolution
        conv = convolution_fft(a, a)
        maxc = conv.max()
        exp_w = np.exp(beta * (conv - maxc))
        w = exp_w / exp_w.sum()              # soft‑max weights over the convolution entries

        # gradient of the smooth max w.r.t. a
        a_rev = a[::-1]
        conv_rev = convolution_fft(a_rev, w)
        grad_a = 2.0 * conv_rev[n - 1:n - 1 + n]

        # back‑prop through soft‑max
        dot_ag = np.dot(a, grad_a)
        grad_z = a * grad_a - dot_ag * a

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad_z
        v = beta2 * v + (1.0 - beta2) * (grad_z ** 2)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        z -= lr * m_hat / (np.sqrt(v_hat) + eps_adam)

        # occasional reporting
        if it % 2000 == 0:
            a_tmp = softmax(z)
            sc = compute_score(a_tmp)
            if sc < best_score:
                best_score = sc
                best_a = a_tmp.copy()
                print(f"[{time.time():.1f}] Adam improved to {best_score:.6f}")

    # final projection
    a_fin = softmax(z)
    a_fin = project_to_simplex(a_fin)
    sc_fin = compute_score(a_fin)
    if sc_fin < best_score:
        best_score = sc_fin
        best_a = a_fin.copy()

    return best_a, best_score


# ----------------------------------------------------------------------
# Projected sub‑gradient descent on the true max‑objective
# ----------------------------------------------------------------------


def subgrad_optimize(start_a: np.ndarray,
                     deadline: float,
                     max_iters: int = 1_000_000,
                     step0: float = 0.07) -> Tuple[np.ndarray, float]:
    """
    Classic projected sub‑gradient descent for 2·n·max(a*a)
    with diminishing step ∝ 1/√t.
    """
    n = start_a.shape[0]
    a = start_a.astype(np.float64).copy()
    best_a = a.copy()
    best_score = compute_score(a)

    for t in range(1, max_iters + 1):
        if time.time() >= deadline:
            break

        conv = convolution_fft(a, a)
        k = int(np.argmax(conv))

        idx = np.arange(n)
        j = k - idx
        mask = (j >= 0) & (j < n)
        g = np.zeros_like(a)
        g[mask] = 2.0 * a[j[mask]]

        step = step0 / np.sqrt(t)
        a = project_to_simplex(a - step * g)

        if t % 2000 == 0:
            sc = compute_score(a)
            if sc < best_score:
                best_score = sc
                best_a = a.copy()
                print(f"[{time.time():.1f}] sub‑grad improved to {best_score:.6f}")

    sc = compute_score(a)
    if sc < best_score:
        best_score = sc
        best_a = a.copy()
    return best_a, best_score


# ----------------------------------------------------------------------
# Targeted simulated‑annealing hill‑climb
# ----------------------------------------------------------------------


def hill_climb_targeted(start_a: np.ndarray,
                        deadline: float,
                        max_steps: int = 500_000) -> Tuple[np.ndarray, float]:
    """
    Simulated‑annealing walk that always identifies the convolution index
    k* with the current maximum and preferentially moves a small
    fraction of mass out of the pair of entries (i, k*‑i) that contribute
    most to this maximum.
    """
    n = start_a.size
    a = start_a.copy()
    # ensure we stay on the simplex
    a = a / a.sum()
    conv = convolution_fft(a, a)
    cur_score = compute_score(a)

    best_a = a.copy()
    best_score = cur_score

    # temperature schedule (tiny temperature, enough to accept occasional uphill moves)
    T0 = cur_score * 1e-4
    T = T0
    step_scale = 0.07

    for step in range(1, max_steps + 1):
        if time.time() >= deadline:
            break

        conv = convolution_fft(a, a)
        k = int(np.argmax(conv))

        # contribution of each i to the current maximum
        idx = np.arange(n)
        j = k - idx
        mask = (j >= 0) & (j < n)
        contrib = np.zeros_like(a)
        contrib[mask] = a[idx[mask]] * a[j[mask]]

        # pick a heavy contributor (probability proportional to contribution)
        if contrib.sum() > 0:
            probs = contrib / contrib.sum()
            i = np.random.choice(n, p=probs)
        else:
            i = np.random.randint(n)

        j = k - i
        if not (0 <= j < n):
            # fall back to a random partner if the partner lies outside the range
            j = np.random.randint(n)

        # decide from which side to take mass
        if np.random.rand() < 0.5 and a[i] > 1e-12:
            src = i
        elif a[j] > 1e-12:
            src = j
        else:
            src = i  # any non‑zero entry; if both are zero we will perform a swap later

        # propose a small transfer from src to a random destination
        eps = step_scale * a[src]
        dst = np.random.randint(n)
        while dst == src:
            dst = np.random.randint(n)

        cand = a.copy()
        cand[src] -= eps
        cand[dst] += eps
        cand = np.maximum(cand, 0.0)
        cand = cand / cand.sum()          # re‑project onto simplex

        # evaluate candidate
        cand_score = compute_score(cand)

        delta = cand_score - cur_score
        if delta < 0 or np.random.rand() < np.exp(-delta / (T + 1e-18)):
            a = cand
            cur_score = cand_score
            if cur_score < best_score:
                best_score = cur_score
                best_a = a.copy()
                step_scale = min(step_scale * 1.05, 0.5)   # slowly allow larger moves
        else:
            step_scale = max(step_scale * 0.97, 1e-4)

        # linear cooling
        T = T0 * (1.0 - step / max_steps)

    return best_a, best_score

# Helper for up‑sampling a simplex vector

def upsample_vector(vec: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation of a probability vector to a larger length."""
    old_n = vec.shape[0]
    old_x = np.linspace(0, 1, old_n, endpoint=False)
    new_x = np.linspace(0, 1, new_n, endpoint=False)
    new_vec = np.interp(new_x, old_x, vec, left=0.0, right=0.0)
    new_vec = np.maximum(new_vec, 0.0)
    return project_to_simplex(new_vec)

    # subsequent internal optimisation stages.
    # -----------------------------------------------------------
    if best_seq is not None:
        a_best = np.array(best_seq, dtype=np.float64)
        a_best = project_to_simplex(a_best)
        best_seq = a_best.tolist()

    # -----------------------------------------------------------
    # 2. First Adam refinement (no forced symmetry)
    # -----------------------------------------------------------
    if best_seq is not None and time.time() < deadline - 30.0:
        print(f"[{time.time():.1f}] first Adam refinement (asymmetric)")
        a_start = np.array(best_seq, dtype=np.float64)
        a_opt, sc_opt = adam_smooth_optimize(
            a_start,
            deadline - 20.0,
            max_iters=300_000,
            beta_start=5.0,
            beta_end=2000.0,
            lr=0.018,
            sym_interval=0,               # keep asymmetry
        )
        if sc_opt < best_score:
            best_score = sc_opt
            best_seq = a_opt.tolist()
            print(f"[{time.time():.1f}] after first Adam best = {best_score:.6f}")

    # -----------------------------------------------------------
    # 3. Projected sub‑gradient descent (short)
    # -----------------------------------------------------------
    if best_seq is not None and time.time() < deadline - 20.0:
        print(f"[{time.time():.1f}] projected sub‑gradient descent")
        a_sub, sc_sub = subgrad_optimize(
            np.array(best_seq, dtype=np.float64),
            deadline - 10.0,
            max_iters=800_000,
            step0=0.07,
        )
        if sc_sub < best_score:
            best_score = sc_sub
            best_seq = a_sub.tolist()
            print(f"[{time.time():.1f}] after sub‑grad best = {best_score:.6f}")

    # -----------------------------------------------------------
    # 4. Upsample to a larger length and run another Adam pass
    # -----------------------------------------------------------
    if best_seq is not None and time.time() < deadline - 30.0:
        n_cur = len(best_seq)
        n_new = n_cur * 2
        if n_new <= 8192:
            print(f"[{time.time():.1f}] upsampling from {n_cur} to {n_new}")
            a_ups = upsample_vector(np.array(best_seq, dtype=np.float64), n_new)
            a_opt2, sc_opt2 = adam_smooth_optimize(
                a_ups,
                deadline - 15.0,
                max_iters=250_000,
                beta_start=10.0,
                beta_end=3000.0,
                lr=0.012,
                sym_interval=0,
            )
            if sc_opt2 < best_score:
                best_score = sc_opt2
                best_seq = a_opt2.tolist()
                print(f"[{time.time():.1f}] after upsampled Adam best = {best_score:.6f}")

    # -----------------------------------------------------------
    # 5. Final targeted hill‑climb (simulated annealing)
    # -----------------------------------------------------------
    if best_seq is not None and time.time() < deadline - 5.0:
        print(f"[{time.time():.1f}] final targeted hill‑climb")
        a_final, sc_final = hill_climb_targeted(
            np.array(best_seq, dtype=np.float64),
            deadline,
            max_steps=400_000,
        )
        if sc_final < best_score:
            best_score = sc_final
            best_seq = a_final.tolist()
            print(f"[{time.time():.1f}] hill‑climb improved best = {best_score:.6f}")

    # -----------------------------------------------------------
    # 6. Safeguard default (should never be needed)
    # -----------------------------------------------------------
    if best_seq is None:
        n_small = 256
        best_seq = [1.0 / n_small] * n_small
        best_score = evaluate_sequence(best_seq)

    final_score = evaluate_sequence(best_seq)
    print(f"\n[{time.time():.1f}] finished search – final reported score = {final_score:.6f}")

    return best_seq
```
