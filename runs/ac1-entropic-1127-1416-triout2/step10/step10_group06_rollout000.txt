"""
Optimise a non‑negative step‑function (vector) for the objective

    score = 2·n·max(conv(a, a)) / (sum(a)**2)

(which is exactly `evaluate_sequence` after normalisation).

The optimiser works in several stages:

1. **Exploration on several lengths** – quick scans of a few analytic families
   (U‑shaped, power‑law, exponential, beta, Gaussian, sinusoidal, raised‑cosine)
   on lengths 256 … 8192.  A handful of random initialisations are also tried.
   The best vector (and its length) is kept.

2. **Upsampling** – the best vector is interpolated to the maximal length
   (default 16384) and re‑normalised.

3. **Smooth surrogate optimisation** – minimise a log‑sum‑exp approximation
   of the max‑convolution using L‑BFGS‑B on the soft‑max parametrisation.
   Several β‑values (10 → 200 → 800 → 2000) are used so that the surrogate
   converges to the true max.

4. **Mirror (exponentiated) sub‑gradient descent** – a simple multiplicative
   update on the true max‑convolution with a diminishing step‑size.

5. **Symmetrisation** – many optimal solutions are symmetric; the current
   vector is symmetrised and projected back onto the simplex.

6. **Tiny simulated‑annealing hill‑climb** – occasional swaps / mass‑transfer
   moves remove shallow local minima.

All stages respect the global deadline (default 1000 s).  The function
`propose_candidate` returns a plain Python list of floats ready for
`evaluate_sequence`.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# FFT utilities (linear convolution via real FFT)
# ----------------------------------------------------------------------
def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Linear convolution using real FFT (full length)."""
    m = a.size + b.size - 1
    nfft = next_pow2(m)
    A = np.fft.rfft(a, n=nfft)
    B = np.fft.rfft(b, n=nfft)
    C = A * B
    conv = np.fft.irfft(C, n=nfft)[:m]
    return conv


def compute_score(a: np.ndarray) -> float:
    """
    Exact objective for a probability vector (sum(a)=1):
        score = 2·n·max(conv(a, a))
    """
    n = a.shape[0]
    conv = convolution_fft(a, a)
    return 2.0 * n * float(conv.max())


# ----------------------------------------------------------------------
# Simplex utilities
# ----------------------------------------------------------------------
def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection onto the probability simplex."""
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / v.shape[0])

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, v.shape[0] + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.where(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def softmax(z: np.ndarray) -> np.ndarray:
    """Stable soft‑max mapping ℝⁿ → simplex."""
    z_max = np.max(z)
    e = np.exp(z - z_max)
    return e / e.sum()


# ----------------------------------------------------------------------
# Analytic families – each returns a normalised probability vector
# ----------------------------------------------------------------------
def symmetric_product_family(n: int, alpha: float) -> np.ndarray:
    """U‑shaped: a_i ∝ [(i+½)(n‑½‑i)]^(‑α)."""
    i = np.arange(n, dtype=np.float64)
    # avoid log(0) by adding a tiny epsilon
    eps = 1e-12
    logprod = np.log(i + 0.5 + eps) + np.log(n - 0.5 - i + eps)
    w = np.exp(-alpha * logprod)
    a = w / w.sum()
    return a


def left_heavy_power_family(n: int, beta: float) -> np.ndarray:
    """Monotone decreasing power law: a_i ∝ (i+½)^(‑β)."""
    i = np.arange(n, dtype=np.float64)
    w = np.exp(-beta * np.log(i + 0.5))
    a = w / w.sum()
    return a


def exponential_decay_family(n: int, gamma: float) -> np.ndarray:
    """Simple exponential decay: a_i ∝ exp(‑γ·i)."""
    i = np.arange(n, dtype=np.float64)
    w = np.exp(-gamma * i)
    a = w / w.sum()
    return a


def beta_distribution_family(n: int, alpha: float, beta_param: float = None) -> np.ndarray:
    """Symmetric (or asymmetric) beta shape."""
    if beta_param is None:
        beta_param = alpha
    i = np.arange(n, dtype=np.float64)
    t = i / (n - 1)
    eps = 1e-12
    w = (t + eps) ** alpha * (1 - t + eps) ** beta_param
    a = w / w.sum()
    return a


def gaussian_family(n: int, sigma_frac: float) -> np.ndarray:
    """Gaussian centred at the middle. sigma_frac = sigma / n."""
    sigma = sigma_frac * n
    mid = (n - 1) / 2.0
    i = np.arange(n, dtype=np.float64)
    w = np.exp(-0.5 * ((i - mid) / sigma) ** 2)
    a = w / w.sum()
    return a


def sinusoid_half_wave_family(n: int, power: float = 1.0) -> np.ndarray:
    """a_i ∝ sin(π·(i+0.5)/n)^power, always positive."""
    i = np.arange(n, dtype=np.float64)
    w = np.sin(np.pi * (i + 0.5) / n) ** power
    a = w / w.sum()
    return a


def raised_cosine_family(n: int, power: float = 1.0) -> np.ndarray:
    """a_i ∝ (1 - cos(2π·i/(n‑1)))^power, zero at the ends."""
    i = np.arange(n, dtype=np.float64)
    w = (1.0 - np.cos(2.0 * np.pi * i / (n - 1))) ** power
    a = w / w.sum()
    return a


# ----------------------------------------------------------------------
# Random starts
# ----------------------------------------------------------------------
def random_sparse_start(n: int) -> np.ndarray:
    """Sparse probability vector (≈5 % non‑zeros)."""
    sparsity = np.random.uniform(0.03, 0.07)
    k = max(2, int(round(sparsity * n)))
    idx = np.random.choice(n, size=k, replace=False)
    a = np.zeros(n, dtype=np.float64)
    a[idx] = 1.0 / k
    return project_to_simplex(a)


def random_monotone_start(n: int) -> np.ndarray:
    """Random decreasing vector (sorted power‑transformed uniform)."""
    r = np.random.rand(n) ** np.random.uniform(0.5, 5.0)
    r.sort()
    a = r[::-1]
    a = a / a.sum()
    return a


# ----------------------------------------------------------------------
# Smooth surrogate optimisation (L‑BFGS‑B on log‑sum‑exp)
# ----------------------------------------------------------------------
def _smooth_obj_and_grad(z: np.ndarray, beta: float, n: int):
    """
    Returns (f, grad) for the log‑sum‑exp surrogate
        f   = (1/β)·log∑exp(β·conv)
    and its gradient w.r.t. the log‑soft‑max parameters z.
    """
    a = softmax(z)                     # a ∈ simplex
    conv = convolution_fft(a, a)       # length 2n‑1
    maxc = conv.max()
    # stable log‑sum‑exp
    exp_terms = np.exp(beta * (conv - maxc))
    sum_exp = exp_terms.sum()
    f = (np.log(sum_exp) + beta * maxc) / beta

    w = exp_terms / sum_exp            # weights summing to 1, shape 2n‑1
    a_rev = a[::-1]
    # convolution of a_rev with w gives the gradient wrt a (see derivation)
    conv_rev = convolution_fft(a_rev, w)   # length 3n‑2
    grad_a = 2.0 * conv_rev[n - 1 : 2 * n - 1]

    # chain rule through softmax
    dot_ag = np.dot(a, grad_a)
    grad_z = a * (grad_a - dot_ag)
    return float(f), grad_z.astype(np.float64)


def l_bfgs_refine(start_a: np.ndarray, deadline: float,
                  beta_schedule=(10.0, 50.0, 200.0, 800.0, 2000.0),
                  max_iter_per_beta=1200) -> tuple[np.ndarray, float]:
    """
    Refine `start_a` using L‑BFGS‑B on the smoothed surrogate.
    Returns the best vector (as ndarray) and its exact score.
    """
    try:
        import scipy.optimize as _opt
    except Exception:  # pragma: no cover
        # SciPy unavailable – fall back to a no‑op
        return start_a, compute_score(start_a)

    n = start_a.shape[0]
    eps = 1e-12
    # initialise log‑soft‑max parameters
    z = np.log(start_a + eps)

    best_a = start_a.copy()
    best_score = compute_score(best_a)

    for beta in beta_schedule:
        if time.time() >= deadline - 4.0:
            break

        # allocate iterations dynamically based on remaining time
        remaining = deadline - time.time()
        max_iter = min(max_iter_per_beta, int(remaining * 15))  # ~15 iters / sec

        def func(z_vec):
            return _smooth_obj_and_grad(z_vec, beta, n)

        try:
            opt_res = _opt.fmin_l_bfgs_b(
                func,
                z,
                maxiter=max_iter,
                disp=False,
                factr=1e7,
                pgtol=1e-12,
            )
        except Exception as exc:  # pragma: no cover
            print(f"[{time.time():.1f}] L‑BFGS failed at β={beta}: {exc}")
            continue

        z = opt_res[0]
        a_opt = softmax(z)
        sc_opt = compute_score(a_opt)
        if sc_opt < best_score:
            best_score = sc_opt
            best_a = a_opt.copy()
            print(f"[{time.time():.1f}] L‑BFGS β={beta:.0f} improved score = {best_score:.6f}")

    return best_a, best_score


# ----------------------------------------------------------------------
# Mirror (exponentiated) sub‑gradient descent on the true max‑objective
# ----------------------------------------------------------------------
def mirror_subgrad_optimize(start_a: np.ndarray, deadline: float,
                           step0: float = 0.9,
                           check_interval: int = 1500) -> tuple[np.ndarray, float]:
    """
    Multiplicative sub‑gradient descent on F(a)=2·n·max(conv(a,a)).
    Works directly on the simplex via normalisation after each update.
    """
    n = start_a.shape[0]
    a = start_a.astype(np.float64).copy()
    best_a = a.copy()
    best_score = compute_score(a)

    t = 0
    while time.time() < deadline:
        t += 1
        conv = convolution_fft(a, a)
        k = int(np.argmax(conv))

        # sub‑gradient for the active convolution index
        idx = np.arange(n)
        j = k - idx
        mask = (j >= 0) & (j < n)
        g = np.zeros_like(a)
        g[mask] = 2.0 * a[j[mask]]

        eta = step0 / math.sqrt(t)
        a = a * np.exp(-eta * g)

        # projection back onto simplex (simple normalisation)
        s = a.sum()
        if s == 0.0:
            a = np.full_like(a, 1.0 / n)
        else:
            a = a / s

        if t % check_interval == 0:
            sc = compute_score(a)
            if sc < best_score:
                best_score = sc
                best_a = a.copy()
                print(f"[{time.time():.1f}] mirror sub‑grad iter {t} best = {best_score:.6f}")

    # final check
    sc = compute_score(a)
    if sc < best_score:
        best_score = sc
        best_a = a.copy()
    return best_a, best_score


# ----------------------------------------------------------------------
# Tiny simulated‑annealing hill‑climb
# ----------------------------------------------------------------------
def hill_climb_sa(start_a: np.ndarray, deadline: float,
                  max_steps: int = 800_000,
                  init_temp_factor: float = 8e-5) -> tuple[np.ndarray, float]:
    """
    Very low‑temperature SA on the simplex:
        - with prob 0.5 swap two entries,
        - otherwise transfer a fraction of mass from i→j.
    """
    n = start_a.size
    cur_a = start_a.copy()
    cur_score = compute_score(cur_a)

    best_a = cur_a.copy()
    best_score = cur_score

    T0 = best_score * init_temp_factor
    T = T0
    step_scale = 0.10

    for step in range(1, max_steps + 1):
        if time.time() >= deadline:
            break

        cand = cur_a.copy()
        if np.random.rand() < 0.5:
            i, j = np.random.randint(n, size=2)
            cand[i], cand[j] = cand[j], cand[i]
        else:
            i, j = np.random.randint(n, size=2)
            if i != j:
                delta = step_scale * cur_a[i]
                cand[i] -= delta
                cand[j] += delta
                if cand[i] < 0.0:
                    cand[i] = 0.0

        cand = cand / cand.sum()
        cand_score = compute_score(cand)
        delta = cand_score - cur_score

        if delta < 0.0 or np.random.rand() < np.exp(-delta / (T + 1e-18)):
            cur_a = cand
            cur_score = cand_score
            if cur_score < best_score:
                best_score = cur_score
                best_a = cur_a.copy()
                step_scale = min(step_scale * 1.07, 0.5)
        else:
            step_scale *= 0.97

        T = T0 * (1.0 - step / max_steps)

    return best_a, best_score


# ----------------------------------------------------------------------
# Main driver
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 12345, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative sequence minimising `evaluate_sequence`.
    Returns a plain Python list of floats.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 2.0  # safety margin

    # ------------------------------------------------------------------
    # Choose a set of lengths to explore; larger lengths give finer granularity.
    # ------------------------------------------------------------------
    lengths = [256, 512, 1024, 2048, 4096, 8192]
    # If enough time is left, also try the maximal length.
    if budget_s > 600:
        lengths.append(16384)

    best_vec = None
    best_score = float("inf")
    best_n = None

    # ------------------------------------------------------------------
    # Phase 0 – optional fallback (if a known construction is present)
    # ------------------------------------------------------------------
    try:
        fb_vec = [float(x) for x in height_sequence_1]   # type: ignore
        fb_score = evaluate_sequence(fb_vec)
        print(f"[{time.time():.1f}] fallback score = {fb_score:.6f}")
        if fb_score < best_score:
            best_score = fb_score
            best_vec = fb_vec
            best_n = len(fb_vec)
    except Exception:
        pass

    # ------------------------------------------------------------------
    # Phase 1 – cheap scan of analytic families on each length
    # ------------------------------------------------------------------
    for n in lengths:
        if time.time() >= deadline - 30.0:
            break
        print(f"[{time.time():.1f}] scanning families for n = {n}")

        # ----- symmetric product (U‑shaped) -----
        for alpha in np.linspace(0.0, 8.0, 61):
            vec = symmetric_product_family(n, alpha)
            sc = compute_score(vec)
            if sc < best_score:
                best_score, best_vec, best_n = sc, vec.tolist(), n
                print(f"[{time.time():.1f}] new best (U‑α={alpha:.2f}) = {best_score:.6f}")

        # ----- left‑heavy power law -----
        for beta in np.linspace(0.0, 8.0, 61):
            vec = left_heavy_power_family(n, beta)
            sc = compute_score(vec)
            if sc < best_score:
                best_score, best_vec, best_n = sc, vec.tolist(), n
                print(f"[{time.time():.1f}] new best (pow‑β={beta:.2f}) = {best_score:.6f}")

        # ----- exponential decay -----
        for gamma in np.linspace(0.0, 7.0, 51):
            vec = exponential_decay_family(n, gamma)
            sc = compute_score(vec)
            if sc < best_score:
                best_score, best_vec, best_n = sc, vec.tolist(), n
                print(f"[{time.time():.1f}] new best (exp‑γ={gamma:.2f}) = {best_score:.6f}")

        # ----- symmetric beta distribution -----
        for alpha in np.linspace(0.0, 8.0, 51):
            vec = beta_distribution_family(n, alpha)
            sc = compute_score(vec)
            if sc < best_score:
                best_score, best_vec, best_n = sc, vec.tolist(), n
                print(f"[{time.time():.1f}] new best (beta‑α={alpha:.2f}) = {best_score:.6f}")

        # ----- Gaussian family (sigma as fraction of n) -----
        for sigma_frac in np.linspace(0.03, 0.5, 45):
            vec = gaussian_family(n, sigma_frac)
            sc = compute_score(vec)
            if sc < best_score:
                best_score, best_vec, best_n = sc, vec.tolist(), n
                print(f"[{time.time():.1f}] new best (gauss‑σ={sigma_frac:.3f}) = {best_score:.6f}")

        # ----- Sinusoidal half‑wave (power) -----
        for power in np.linspace(0.5, 4.0, 35):
            vec = sinusoid_half_wave_family(n, power)
            sc = compute_score(vec)
            if sc < best_score:
                best_score, best_vec, best_n = sc, vec.tolist(), n
                print(f"[{time.time():.1f}] new best (sin‑p={power:.2f}) = {best_score:.6f}")

        # ----- Raised cosine (power) -----
        for power in np.linspace(0.5, 4.0, 35):
            vec = raised_cosine_family(n, power)
            sc = compute_score(vec)
            if sc < best_score:
                best_score, best_vec, best_n = sc, vec.tolist(), n
                print(f"[{time.time():.1f}] new best (rcos‑p={power:.2f}) = {best_score:.6f}")

        # ----- random starts (monotone & sparse) -----
        for _ in range(80):
            if time.time() >= deadline - 30.0:
                break
            vec = random_monotone_start(n)
            sc = compute_score(vec)
            if sc < best_score:
                best_score, best_vec, best_n = sc, vec.tolist(), n
                print(f"[{time.time():.1f}] new best (rand‑mono) = {best_score:.6f}")

        for _ in range(80):
            if time.time() >= deadline - 30.0:
                break
            vec = random_sparse_start(n)
            sc = compute_score(vec)
            if sc < best_score:
                best_score, best_vec, best_n = sc, vec.tolist(), n
                print(f"[{time.time():.1f}] new best (rand‑sparse) = {best_score:.6f}")

    # ------------------------------------------------------------------
    # At this point we have a promising vector (maybe at a small n)
    # ------------------------------------------------------------------
    if best_vec is None:
        # fall back to uniform of moderate size
        n_fallback = 512
        best_vec = [1.0 / n_fallback] * n_fallback
        best_score = evaluate_sequence(best_vec)
        best_n = n_fallback

    # ------------------------------------------------------------------
    # Phase 2 – upsample to the maximal length (if needed)
    # ------------------------------------------------------------------
    target_n = max(lengths)  # typically 16384 if budget permits
    if best_n != target_n:
        # linear interpolation on the index domain [0,1]
        src = np.array(best_vec, dtype=np.float64)
        src_x = np.linspace(0.0, 1.0, src.size)
        tgt_x = np.linspace(0.0, 1.0, target_n)
        up = np.interp(tgt_x, src_x, src)
        up = up / up.sum()
        best_vec = up.tolist()
        best_n = target_n
        best_score = compute_score(up)
        print(f"[{time.time():.1f}] up‑sampled best to n={target_n}, score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 3 – smooth surrogate refinement (L‑BFGS) with multiple restarts
    # ------------------------------------------------------------------
    if time.time() < deadline - 300.0:
        a_start = np.array(best_vec, dtype=np.float64)
        # first warm‑start
        a_ref, sc_ref = l_bfgs_refine(a_start, deadline - 200.0)
        if sc_ref < best_score:
            best_score, best_vec = sc_ref, a_ref.tolist()
            print(f"[{time.time():.1f}] L‑BFGS warm‑start improved to {best_score:.6f}")

        # a few random perturbations + L‑BFGS
        for rnd in range(4):
            if time.time() >= deadline - 200.0:
                break
            pert = a_ref * np.exp(0.02 * np.random.randn(target_n))
            pert = pert / pert.sum()
            a_ref2, sc_ref2 = l_bfgs_refine(pert, deadline - 150.0)
            if sc_ref2 < best_score:
                best_score, best_vec = sc_ref2, a_ref2.tolist()
                a_ref = a_ref2
                print(f"[{time.time():.1f}] L‑BFGS restart {rnd} → {best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 4 – mirror sub‑gradient descent on the true max‑objective
    # ------------------------------------------------------------------
    if time.time() < deadline - 100.0:
        a_sub, sc_sub = mirror_subgrad_optimize(
            np.array(best_vec, dtype=np.float64),
            deadline - 70.0,
            step0=0.9,
        )
        if sc_sub < best_score:
            best_score, best_vec = sc_sub, a_sub.tolist()
            print(f"[{time.time():.1f}] mirror sub‑grad improved to {best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 5 – symmetrisation (many optima are symmetric)
    # ------------------------------------------------------------------
    if best_vec is not None:
        a_arr = np.array(best_vec, dtype=np.float64)
        a_sym = (a_arr + a_arr[::-1]) * 0.5
        a_sym = project_to_simplex(a_sym)
        sc_sym = compute_score(a_sym)
        if sc_sym < best_score:
            best_score, best_vec = sc_sym, a_sym.tolist()
            print(f"[{time.time():.1f}] symmetrisation improved best = {best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 6 – final tiny SA hill‑climb
    # ------------------------------------------------------------------
    if time.time() < deadline - 8.0:
        a_final, sc_final = hill_climb_sa(
            np.array(best_vec, dtype=np.float64),
            deadline,
            max_steps=800_000,
        )
        if sc_final < best_score:
            best_score, best_vec = sc_final, a_final.tolist()
            print(f"[{time.time():.1f}] hill‑climb final improved to {best_score:.6f}")

    # ------------------------------------------------------------------
    # Final validation via the provided `evaluate_sequence`
    # ------------------------------------------------------------------
    final_score = evaluate_sequence(best_vec)
    print(f"\n[{time.time():.1f}] finished – final reported score = {final_score:.6f}")
    return best_vec
