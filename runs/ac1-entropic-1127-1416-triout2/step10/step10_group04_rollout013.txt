"""
Optimiser for ``evaluate_sequence`` – aim: push the max‑auto‑convolution
(2·n·max(conv(a,a))) as close as possible to its theoretical lower bound ≈1.
Approach
--------
1.  Build a diverse portfolio of analytic starter vectors:
    * uniform
    * central‑binomial (the classic “binomial” shape)
    * a one‑parameter family a_i ∝ (i+½)^{α‑1}(n‑½‑i)^{α‑1}  (discrete Beta(α,α))
    * a one‑parameter “generalised binomial” a_i ∝ C(2i,i)*(q)^i
    * simple power‑law and geometric families
2.  Pick the best starter according to the exact score (2·n·max(conv)).
3.  Refine it with a long‑running smooth‑max Adam optimiser
    (soft‑max surrogate of the max, Euclidean projection onto the simplex).
4.  Polish the result with projected sub‑gradient descent on the true
    max‑convolution objective.
5.  Finish with a cheap stochastic hill‑climb that moves tiny fractions of
    mass between random positions.
All phases respect a global 1000 s wall‑clock budget and keep the best
vector seen so far.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Utility functions

def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """
    Linear convolution using real FFT (fast for the lengths we use).
    Returns an array of length len(a)+len(b)-1.
    """
    m = a.shape[0] + b.shape[0] - 1
    n_fft = next_pow2(m)
    A = np.fft.rfft(a, n=n_fft)
    B = np.fft.rfft(b, n=n_fft)
    C = A * B
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


def compute_score(a: np.ndarray) -> float:
    """Exact objective for a simplex‑normalised vector."""
    n = a.shape[0]
    conv = convolution_fft(a, a)
    return 2.0 * n * float(conv.max())


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of v onto the probability simplex:
        sum(v) = 1, v ≥ 0.
    """
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / v.shape[0])

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, v.shape[0] + 1)
    cond = u - cssv / ind > 0
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

# Analytic starter families

def generate_uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def generate_central_binomial(n: int) -> np.ndarray:
    """a_i = C(2i,i)/4^i, normalised to a probability vector."""
    a = np.empty(n, dtype=np.float64)
    a[0] = 1.0
    for i in range(1, n):
        a[i] = a[i - 1] * (2.0 * i - 1) / (2.0 * i)
    return project_to_simplex(a)


def generate_powerlaw(n: int, p: float) -> np.ndarray:
    """a_i ∝ (i+1)^{-p}."""
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = idx ** (-p)
    return project_to_simplex(a)


def generate_geometric(n: int, r: float) -> np.ndarray:
    """Geometric distribution (1‑r) r^i, already normalised for r<1."""
    i = np.arange(n, dtype=np.float64)
    a = (1.0 - r) * (r ** i)
    return project_to_simplex(a)


def generate_beta_shape(n: int, alpha: float) -> np.ndarray:
    """
    Discrete symmetric Beta(α,α) on {0,…,n‑1}:
    a_i ∝ (i+½)^{α‑1} (n‑½‑i)^{α‑1}.
    """
    i = np.arange(n, dtype=np.float64)
    a = (i + 0.5) ** (alpha - 1.0) * (n - 0.5 - i) ** (alpha - 1.0)
    return project_to_simplex(a)


def generate_binomial_family(n: int, q: float) -> np.ndarray:
    """
    Generalised binomial shape:
    a_i ∝ C(2i,i) * q^i.
    For q = 0.25 we obtain the classic central‑binomial family.
    """
    a = np.empty(n, dtype=np.float64)
    a[0] = 1.0
    factor = 4.0 * q
    for i in range(1, n):
        a[i] = a[i - 1] * (2.0 * i - 1) / (2.0 * i) * factor
    return project_to_simplex(a)

def smooth_max_adam(
    start_a: np.ndarray,
    deadline: float,
    beta_start: float = 5.0,
    beta_end: float = 400.0,
    step0: float = 0.9,
    adam_beta1: float = 0.9,
    adam_beta2: float = 0.999,
    adam_eps: float = 1e-8,
) -> tuple[np.ndarray, float]:
    """
    Adam optimisation of the soft‑max surrogate of max(conv(a,a)).
    Uses an exponentially increasing temperature β (from beta_start to beta_end).
    """
    a = start_a.copy()
    n = a.shape[0]
    m1 = np.zeros_like(a)
    m2 = np.zeros_like(a)
    it = 0
    best_a = a.copy()
    best_score = compute_score(a)

    # pre‑compute how many iterations we are *likely* to get, to drive β schedule
    # simply map iteration → fraction of total (unknown) iterations via a
    # exponential schedule that reaches beta_end after ~5·10⁵ steps.
    max_schedule_iters = 600_000

    while time.time() < deadline:
        it += 1
        # schedule β
        t_frac = min(it / max_schedule_iters, 1.0)
        beta = beta_start * ((beta_end / beta_start) ** t_frac)

        # full auto‑convolution
        conv = convolution_fft(a, a)
        maxc = conv.max()

        # soft‑max weights
        exp_w = np.exp(beta * (conv - maxc))
        w = exp_w / exp_w.sum()

        # gradient of the smoothed objective: 2·rev(a) * w (restricted to length n)
        rev_a = a[::-1]
        grad_full = convolution_fft(rev_a, w)          # length≈3n‑2
        grad = 2.0 * grad_full[n - 1 : n - 1 + n]

        # Adam update
        m1 = adam_beta1 * m1 + (1.0 - adam_beta1) * grad
        m2 = adam_beta2 * m2 + (1.0 - adam_beta2) * (grad * grad)
        m1_hat = m1 / (1.0 - adam_beta1 ** it)
        m2_hat = m2 / (1.0 - adam_beta2 ** it)

        step = step0 / np.sqrt(it)
        a_new = a - step * m1_hat / (np.sqrt(m2_hat) + adam_eps)
        a_new = project_to_simplex(a_new)

        a = a_new

        # occasional exact evaluation
        if it % 400 == 0:
            cur_score = compute_score(a)
            if cur_score < best_score:
                best_score = cur_score
                best_a = a.copy()

    final_score = compute_score(a)
    if final_score < best_score:
        best_score, best_a = final_score, a.copy()
    return best_a, best_score


def subgradient_descent(
    start_a: np.ndarray,
    deadline: float,
    step0: float = 0.07,
) -> tuple[np.ndarray, float]:
    """
    Projected sub‑gradient descent on the true max‑objective:
        f(a) = 2·n·max(conv(a,a)).
    The sub‑gradient is easy: for the active index k,
    ∂f/∂a_i = 2·a_{k‑i} (if k‑i is a valid index, else 0).
    """
    a = start_a.copy()
    n = a.shape[0]
    best_a = a.copy()
    best_score = compute_score(a)

    it = 0
    while time.time() < deadline:
        it += 1
        conv = convolution_fft(a, a)
        k = int(np.argmax(conv))

        # Build sub‑gradient
        g = np.zeros_like(a)
        # indices i where j = k - i lies in 0..n‑1
        i_vals = np.arange(n)
        j_vals = k - i_vals
        mask = (j_vals >= 0) & (j_vals < n)
        g[mask] = 2.0 * a[j_vals[mask]]

        step = step0 / np.sqrt(it)
        a_new = a - step * g
        a_new = project_to_simplex(a_new)

        cur_score = compute_score(a_new)
        if cur_score < best_score:
            best_score, best_a = cur_score, a_new.copy()

        a = a_new

        # tiny jitter to escape plateaus (every 1000 iters)
        if it % 1000 == 0:
            jitter = np.random.randn(n) * (step0 * 0.004 / np.sqrt(it))
            a += jitter
            a = project_to_simplex(a)

    final_score = compute_score(a)
    if final_score < best_score:
        best_score, best_a = final_score, a.copy()
    return best_a, best_score


def hill_climb(
    start_seq,
    deadline: float,
    max_steps: int = 250_000,
) -> tuple[list[float], float]:
    """
    Tiny stochastic hill‑climb: either swap two entries or move a small
    fraction of mass from i to j, followed by projection onto the simplex.
    """
    a = np.array(start_seq, dtype=np.float64)
    n = a.shape[0]
    best_a = a.copy()
    best_score = compute_score(best_a)

    step_scale = 0.07  # start with moving ~7 % of the source entry

    steps = 0
    while steps < max_steps and time.time() < deadline:
        cand = best_a.copy()
        if np.random.rand() < 0.5:   # swap
            i, j = np.random.randint(n, size=2)
            cand[i], cand[j] = cand[j], cand[i]
        else:                        # fractional transfer
            i, j = np.random.randint(n, size=2)
            if i != j:
                delta = step_scale * best_a[i]
                cand[i] -= delta
                cand[j] += delta
                if cand[i] < 0:
                    cand[i] = 0.0

        cand = project_to_simplex(cand)
        cur_score = compute_score(cand)

        if cur_score < best_score:
            best_score = cur_score
            best_a = cand
            step_scale = min(step_scale * 1.06, 0.5)
        else:
            step_scale *= 0.97

        steps += 1

    return best_a.tolist(), best_score

        init_candidates.append(generate_geometric(n, float(r)))

    # Beta‑shape family (symmetric Beta(α,α))
    for alpha in np.linspace(0.25, 2.0, 10):
        init_candidates.append(generate_beta_shape(n, float(alpha)))

    # Generalised binomial family
    for q in [0.18, 0.20, 0.22, 0.24, 0.25, 0.27, 0.30]:
        init_candidates.append(generate_binomial_family(n, float(q)))

    # ---------------------------------------------------------------
    # 2. Pick the best starter according to the exact score
    # ---------------------------------------------------------------
    best_seq = None
    best_val = float("inf")
    for idx, cand in enumerate(init_candidates):
        cand = project_to_simplex(cand)
        val = compute_score(cand)
        print(f"Starter {idx:02d}: score = {val:.6f}")
        if val < best_val:
            best_val = val
            best_seq = cand.copy()
    print(f"Best starter score = {best_val:.6f}")

    # ---------------------------------------------------------------
    # 3. Smooth‑max Adam optimisation (majority of the time budget)
    # ---------------------------------------------------------------
    now = time.time()
    remaining = deadline - now
    # allocate ~55 % of the remaining budget to the smooth phase
    smooth_deadline = now + 0.55 * remaining
    if time.time() < smooth_deadline:
        print(f"Starting smooth‑max Adam optimisation (until {smooth_deadline - start_time:.1f}s).")
        a_smooth, smooth_score = smooth_max_adam(
            best_seq,
            deadline=smooth_deadline,
            beta_start=5.0,
            beta_end=400.0,
            step0=0.9,
        )
        if smooth_score < best_val:
            best_val = smooth_score
            best_seq = a_smooth.copy()
            print(f"Smooth phase improved score to {best_val:.6f}")

    # ---------------------------------------------------------------
    # 4. Projected sub‑gradient polishing (≈30 % of remaining time)
    # ---------------------------------------------------------------
    now = time.time()
    remaining = deadline - now
    subgrad_deadline = now + 0.30 * remaining
    if time.time() < subgrad_deadline:
        print(f"Running sub‑gradient polishing (until {subgrad_deadline - start_time:.1f}s).")
        a_sub, sub_score = subgradient_descent(
            best_seq,
            deadline=subgrad_deadline,
            step0=0.07,
        )
        if sub_score < best_val:
            best_val = sub_score
            best_seq = a_sub.copy()
            print(f"Sub‑gradient phase improved score to {best_val:.6f}")

    # ---------------------------------------------------------------
    # 5. Random restarts with short sub‑gradient bursts (optional)
    # ---------------------------------------------------------------
    now = time.time()
    remaining = deadline - now
    restart_deadline = now + 0.10 * remaining
    if time.time() < restart_deadline:
        base = best_seq.copy()
        for r in range(7):
            if time.time() >= restart_deadline:
                break
            pert = base + 0.02 * np.random.randn(n)
            pert = project_to_simplex(pert)
            sub_dead = time.time() + 9.0          # give each restart ~9 s
            a_tmp, s_tmp = subgradient_descent(
                pert,
                deadline=sub_dead,
                step0=0.05,
            )
            if s_tmp < best_val:
                best_val = s_tmp
                best_seq = a_tmp.copy()
                print(f"Restart {r} yielded improvement: {best_val:.6f}")

    # ---------------------------------------------------------------
    # 6. Final stochastic hill‑climb (use the rest of the budget)
    # ---------------------------------------------------------------
    hill_deadline = deadline
    if time.time() < hill_deadline:
        steps = 200_000
        print(f"Final hill‑climb (up to {steps} steps).")
        polished_seq, hill_score = hill_climb(
            best_seq.tolist(),
            deadline=hill_deadline,
            max_steps=steps,
        )
        if hill_score < best_val:
            best_val = hill_score
            best_seq = np.array(polished_seq, dtype=np.float64)
            print(f"Hill‑climb improved final score to {best_val:.6f}")

    # ---------------------------------------------------------------
    # 7. Fallback (should never happen)
    # ---------------------------------------------------------------
    if best_seq is None:
        best_seq = np.full(n, 1.0 / n, dtype=np.float64)
        best_val = compute_score(best_seq)

    # ---------------------------------------------------------------
    # 8. Report final score using the provided evaluate_sequence()
    # ---------------------------------------------------------------
    final_score = evaluate_sequence(best_seq.tolist())
    print(f"\nSearch finished. Final reported score = {final_score:.6f}")
    return best_seq.tolist()
