"""
Propose a low‐score step sequence using a linear‑programming “mass‑pushing’’ step
followed by smooth‑max Adam refinement and a short simulated‑annealing hill‑climb.

The LP step fixes the current maximal auto‑convolution M = max(a*a) and solves

    max   Σ g_i
    s.t.  (a * g)_k ≤ M   for all k            (linear constraints)
          g_i ≥ 0

The solution puts as much mass as possible into positions where the current
shape a contributes little to the convolution, i.e. pushes mass towards the
edges.  The new candidate is a convex combination  (1‑t)·a + t·g, symmetrised
and projected onto the simplex.  Repeating this quickly drives the maximal
convolution down.

After a few LP iterations we run a smoothed‑max Adam optimiser (the same as
in the reference code) for fine tuning, then finish with a lightweight
simulated‑annealing walk.

All heavy work uses NumPy / SciPy only, respects a 1000 s wall‑clock budget and
returns a plain list of floats suitable for ``evaluate_sequence``.
"""

import time
import numpy as np
from scipy.linalg import toeplitz
from scipy.optimize import linprog
from typing import List, Tuple

# ----------------------------------------------------------------------
# Core utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full aperiodic convolution using real FFT."""
    m = a.size + b.size - 1
    nfft = next_pow2(m)
    A = np.fft.rfft(a, nfft)
    B = np.fft.rfft(b, nfft)
    C = A * B
    conv = np.fft.irfft(C, nfft)[:m]
    return conv


def compute_score(a: np.ndarray) -> float:
    """
    Exact objective for a probability vector (∑a = 1):
        2·n·max(a*a)
    """
    n = a.shape[0]
    conv = convolution_fft(a, a)
    return 2.0 * n * float(conv.max())


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / v.shape[0])

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, v.shape[0] + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.where(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def softmax(z: np.ndarray) -> np.ndarray:
    """Numerically stable soft‑max."""
    z_max = np.max(z)
    e = np.exp(z - z_max)
    return e / e.sum()


# ----------------------------------------------------------------------
# Analytic families (used only for a quick start)
# ----------------------------------------------------------------------


def symmetric_product_family(n: int, alpha: float) -> np.ndarray:
    """a_i ∝ [(i+½)(n‑½‑i)]^{‑α}, symmetric, on the simplex."""
    i = np.arange(n, dtype=np.float64)
    # use logs to avoid overflow for large alpha
    logprod = np.log(i + 0.5) + np.log(n - 0.5 - i)
    w = np.exp(-alpha * logprod)
    # handle potential infs when alpha is zero
    w[np.isinf(w)] = np.finfo(np.float64).max
    w = w / w.sum()
    return w


# ----------------------------------------------------------------------
# Smoothed‑max Adam optimisation (identical to the reference)
# ----------------------------------------------------------------------


def adam_smooth_optimize(start_a: np.ndarray,
                         deadline: float,
                         max_iters: int = 300_000,
                         beta_start: float = 5.0,
                         beta_end: float = 1200.0,
                         lr: float = 0.018,
                         sym_interval: int = 4000) -> Tuple[np.ndarray, float]:
    """
    Minimise the smooth approximation of max‑convolution
    f_β(a) = (1/β)·log Σ_k exp(β·(a*a)_k)
    via Adam on the soft‑max parametrisation a = softmax(z).
    Every `sym_interval` steps the iterate is symmetrised
    (a ← (a+a[::-1])/2) and re‑projected onto the simplex.
    """
    n = start_a.shape[0]
    eps = 1e-12
    z = np.log(start_a + eps)

    m = np.zeros_like(z)
    v = np.zeros_like(z)

    beta1 = 0.9
    beta2 = 0.999
    eps_adam = 1e-12

    best_a = softmax(z)
    best_score = compute_score(best_a)

    for it in range(1, max_iters + 1):
        if time.time() >= deadline:
            break

        # exponential schedule for β
        t = it / max_iters
        beta = beta_start * (beta_end / beta_start) ** t

        a = softmax(z)

        # occasional symmetrisation
        if sym_interval and (it % sym_interval == 0):
            a = (a + a[::-1]) * 0.5
            a = project_to_simplex(a)
            z = np.log(a + eps)

        conv = convolution_fft(a, a)
        maxc = conv.max()
        # smooth soft‑max weights
        exp_w = np.exp(beta * (conv - maxc))
        w = exp_w / exp_w.sum()

        # gradient wrt a (derived from smooth max)
        a_rev = a[::-1]
        conv_rev = convolution_fft(a_rev, w)
        grad_a = 2.0 * conv_rev[n - 1:n - 1 + n]

        # back‑prop through soft‑max
        dot_ag = np.dot(a, grad_a)
        grad_z = a * grad_a - dot_ag * a

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad_z
        v = beta2 * v + (1.0 - beta2) * (grad_z ** 2)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        z -= lr * m_hat / (np.sqrt(v_hat) + eps_adam)

        if it % 1000 == 0:
            a_tmp = softmax(z)
            a_tmp = (a_tmp + a_tmp[::-1]) * 0.5
            a_tmp = project_to_simplex(a_tmp)
            sc = compute_score(a_tmp)
            if sc < best_score:
                best_score = sc
                best_a = a_tmp.copy()
                print(f"[{time.time():.1f}] Adam improved to {best_score:.6f}")

    # final projection
    a_fin = softmax(z)
    a_fin = (a_fin + a_fin[::-1]) * 0.5
    a_fin = project_to_simplex(a_fin)
    sc_fin = compute_score(a_fin)
    if sc_fin < best_score:
        best_score = sc_fin
        best_a = a_fin.copy()

    return best_a, best_score


# ----------------------------------------------------------------------
# Simulated‑annealing hill‑climb (same as reference)
# ----------------------------------------------------------------------


def hill_climb_sa(start_a: np.ndarray,
                  deadline: float,
                  max_steps: int = 3_000_000) -> Tuple[np.ndarray, float]:
    """
    Simple random walk on the simplex with a tiny temperature schedule.
    50 % swaps, 50 % mass transfers.
    """
    n = start_a.size
    cur_a = start_a.copy()
    cur_score = compute_score(cur_a)

    best_a = cur_a.copy()
    best_score = cur_score

    T0 = best_score * 1e-4
    T = T0
    step_scale = 0.12

    for step in range(1, max_steps + 1):
        if time.time() >= deadline:
            break

        cand = cur_a.copy()
        if np.random.rand() < 0.5:  # swap two entries
            i, j = np.random.randint(n, size=2)
            cand[i], cand[j] = cand[j], cand[i]
        else:                       # transfer a fraction of mass
            i, j = np.random.randint(n, size=2)
            if i != j:
                delta = step_scale * cur_a[i]
                cand[i] -= delta
                cand[j] += delta
                if cand[i] < 0.0:
                    cand[i] = 0.0

        cand = project_to_simplex(cand)
        cand_score = compute_score(cand)

        delta = cand_score - cur_score
        if delta < 0 or np.random.rand() < np.exp(-delta / (T + 1e-18)):
            cur_a = cand
            cur_score = cand_score
            if cur_score < best_score:
                best_score = cur_score
                best_a = cur_a.copy()
                step_scale = min(step_scale * 1.07, 0.5)
        else:
            step_scale *= 0.97

        T = T0 * (1.0 - step / max_steps)

    return best_a, best_score


# ----------------------------------------------------------------------
# Linear‑programming “mass‑pushing’’ iteration
# ----------------------------------------------------------------------


def lp_mass_push(a: np.ndarray,
                 t: float,
                 M: float) -> np.ndarray:
    """
    Given current shape a (∑a = 1) and its maximal auto‑convolution M,
    solve the LP that maximises total mass while respecting (a*g)_k ≤ M.
    Returns the scaled solution g which also satisfies ∑g = 1.
    """
    n = a.shape[0]
    # Build Toeplitz matrix representing the linear map g ↦ a*g
    col = np.concatenate([a, np.zeros(n - 1)], dtype=np.float64)
    row = np.concatenate([a[:1], np.zeros(n - 1)], dtype=np.float64)
    A = toeplitz(col, row)                      # shape (2n‑1, n)

    # Linear programme: min  -sum(g)  ⇔ max sum(g)
    c = -np.ones(n, dtype=np.float64)
    bounds = [(0.0, None)] * n
    b_ub = np.full(2 * n - 1, M, dtype=np.float64)

    # Use the HiGHS interior‑point solver (fast for moderate sizes)
    res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds,
                  method='highs', options={'presolve': True})
    if not res.success:
        raise RuntimeError(f"LP failed: {res.message}")

    g = res.x
    # Scale to simplex
    s = g.sum()
    if s <= 0:
        raise RuntimeError("LP produced zero mass")
    g = g / s
    return g

    4. Finish with a short simulated‑annealing hill‑climb.

    a = np.maximum(a, 1e-12)
    a = project_to_simplex(a)

    best_a = a.copy()
    best_score = compute_score(best_a)
    print(f"[{time.time():.1f}] Initial score = {best_score:.6f}")

    # -----------------------------------------------------------
    # 1. Linear‑programming mass‑pushing iterations
    # -----------------------------------------------------------
    mix_ratio = 0.4          # initial mixing weight (t)
    max_lp_iters = 12
    for it in range(max_lp_iters):
        if time.time() >= deadline - 120.0:
            break

        # current maximal auto‑convolution
        conv = convolution_fft(best_a, best_a)
        M = float(conv.max())

        try:
            g = lp_mass_push(best_a, mix_ratio, M)
        except Exception as e:
            print(f"[{time.time():.1f}] LP iteration {it} failed: {e}")
            break

        # convex combination, symmetrisation, projection
        new_a = (1.0 - mix_ratio) * best_a + mix_ratio * g
        new_a = (new_a + new_a[::-1]) * 0.5
        new_a = project_to_simplex(new_a)

        sc = compute_score(new_a)
        print(f"[{time.time():.1f}] LP iter {it:2d} – score {sc:.6f}")

        if sc < best_score:
            best_score = sc
            best_a = new_a.copy()
            print(f"   *** new best after LP iter {it} = {best_score:.6f}")

        # slowly decay the mixing coefficient – we want finer adjustments later
        mix_ratio = max(0.05, mix_ratio * 0.85)

    # -----------------------------------------------------------
    # 2. Adam smooth‑max refinement
    # -----------------------------------------------------------
    if time.time() < deadline - 150.0:
        print(f"[{time.time():.1f}] Starting Adam refinement")
        a_opt, sc_opt = adam_smooth_optimize(
            best_a,
            deadline - 80.0,
            max_iters=250_000,
            beta_start=5.0,
            beta_end=2500.0,
            lr=0.018,
            sym_interval=3000,
        )
        if sc_opt < best_score:
            best_score = sc_opt
            best_a = a_opt.copy()
            print(f"[{time.time():.1f}] Adam improved best to {best_score:.6f}")

    # -----------------------------------------------------------
    # 3. Final simulated‑annealing hill‑climb
    # -----------------------------------------------------------
    if time.time() < deadline - 5.0:
        print(f"[{time.time():.1f}] Final hill‑climb (SA)")
        a_final, sc_final = hill_climb_sa(best_a, deadline, max_steps=1_000_000)
        if sc_final < best_score:
            best_score = sc_final
            best_a = a_final.copy()
            print(f"[{time.time():.1f}] SA improved best to {best_score:.6f}")

    # -----------------------------------------------------------
    # 4. Return result (list of Python floats)
    # -----------------------------------------------------------
    final_score = evaluate_sequence(best_a.tolist())
    print(f"\n[{time.time():.1f}] Search finished – final reported score = {final_score:.6f}")
    return best_a.tolist()
