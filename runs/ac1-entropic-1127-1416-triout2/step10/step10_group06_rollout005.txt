"""
Propose a step‑function (non‑negative sequence) that minimises
`evaluate_sequence`.  The strategy improves on the previous baseline by

1. Using a larger search space (n up to 32768) and a finer sweep of
   analytic families (U‑shaped, power‑law, exponential, beta).
2. Refining the best candidate with a much tighter log‑sum‑exp surrogate
   (β up to 12000) via L‑BFGS‑B (more iterations per β).
3. Performing an aggressive mirror‑subgradient descent with a larger
   initial step size.
4. A more exploratory simulated‑annealing hill‑climb (higher initial
   temperature and longer schedule) to escape local minima.
5. Symmetrising the result (many optima are symmetric) and a final call
   to `evaluate_sequence`.

All work respects the 1000 s wall‑clock limit: every loop checks the
deadline and aborts early if needed.
"""

import time
import math

# ----------------------------------------------------------------------
# SciPy optional imports
# ----------------------------------------------------------------------
try:

# ----------------------------------------------------------------------
# FFT utilities (linear convolution via real FFT)
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _conv_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Aperiodic convolution of two real vectors using FFT."""
    m = a.size + b.size - 1
    nfft = _next_pow2(m)
    A = np.fft.rfft(a, n=nfft)
    B = np.fft.rfft(b, n=nfft)
    C = A * B
    conv = np.fft.irfft(C, n=nfft)[:m]
    return conv


def _score(a: np.ndarray) -> float:
    """
    Exact objective for a probability vector (sum(a)=1):
        score = 2·n·max(conv(a,a))
    This matches ``evaluate_sequence`` up to clipping.
    """
    n = a.shape[0]
    conv = _conv_fft(a, a)
    return 2.0 * n * float(conv.max())


# ----------------------------------------------------------------------
# Simplex utilities
# ----------------------------------------------------------------------


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection onto the probability simplex."""
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / v.shape[0])

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, v.shape[0] + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.where(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def _softmax(z: np.ndarray) -> np.ndarray:
    """Stable soft‑max mapping ℝⁿ → simplex."""
    z_max = np.max(z)
    e = np.exp(z - z_max)
    return e / e.sum()


# ----------------------------------------------------------------------
# Analytic families (all return a probability vector, i.e. sum to 1)
# ----------------------------------------------------------------------


def _sym_product_family(n: int, alpha: float) -> np.ndarray:
    """U‑shaped family: a_i ∝ [(i+½)(n‑½‑i)]^(‑α)."""
    i = np.arange(n, dtype=np.float64)
    logprod = np.log(i + 0.5) + np.log(n - 0.5 - i)
    w = np.exp(-alpha * logprod)
    # Guard against overflow for very large α
    w[np.isinf(w)] = np.finfo(np.float64).max
    a = w / w.sum()
    return a


def _left_power_family(n: int, beta: float) -> np.ndarray:
    """Monotone decreasing power law: a_i ∝ (i+½)^(‑β)."""
    i = np.arange(n, dtype=np.float64)
    w = np.exp(-beta * np.log(i + 0.5))
    a = w / w.sum()
    return a


def _exp_decay_family(n: int, gamma: float) -> np.ndarray:
    """Simple exponential decay: a_i ∝ exp(‑γ·i)."""
    i = np.arange(n, dtype=np.float64)
    w = np.exp(-gamma * i)
    a = w / w.sum()
    return a


def _beta_family(n: int, alpha: float) -> np.ndarray:
    """Symmetric beta‑distribution (U‑shape for α<1): a_i ∝ t^α·(1‑t)^α."""
    i = np.arange(n, dtype=np.float64)
    t = i / (n - 1)
    eps = 1e-12
    w = (t + eps) ** alpha * (1.0 - t + eps) ** alpha
    a = w / w.sum()
    return a


def _random_sparse_start(n: int) -> np.ndarray:
    """Sparse start: ~5 % non‑zeros, uniform on the chosen indices."""
    sparsity = np.random.uniform(0.03, 0.07)
    k = max(2, int(round(sparsity * n)))
    idx = np.random.choice(n, size=k, replace=False)
    a = np.zeros(n, dtype=np.float64)
    a[idx] = 1.0 / k
    return _project_simplex(a)


def _random_monotone_start(n: int) -> np.ndarray:
    """Random decreasing shape (sorted power‑transformed uniform)."""
    r = np.random.rand(n) ** np.random.uniform(0.5, 5.0)
    r.sort()
    a = r[::-1]                 # decreasing
    a /= a.sum()
    return a


def _mixed_start(n: int) -> np.ndarray:
    """Mixture of uniform and a heavy‑U‑shape."""
    u = np.full(n, 1.0 / n)
    v = _sym_product_family(n, alpha=6.0)
    w = 0.4 * u + 0.6 * v
    return _project_simplex(w)

# Smooth surrogate (log‑sum‑exp) and L‑BFGS refinement
# ----------------------------------------------------------------------


def _smooth_obj_and_grad(z: np.ndarray, beta: float, n: int):
    """
    Returns (f, grad) for the surrogate
        f(z) = (1/β)·log Σ_k exp[β·(a*a)_k] ,  a = softmax(z).
    The gradient is ∂f/∂z.
    """
    a = _softmax(z)                     # a ∈ simplex
    conv = _conv_fft(a, a)               # length 2n‑1
    maxc = conv.max()
    # stable log‑sum‑exp
    exp_terms = np.exp(beta * (conv - maxc))
    sum_exp = exp_terms.sum()
    f = (np.log(sum_exp) + beta * maxc) / beta

    # gradient w.r.t. a
    w = exp_terms / sum_exp               # shape (2n‑1,)
    # compute ∂f/∂a via convolution of w with a (reversed)
    a_rev = a[::-1]
    conv_rev = _conv_fft(a_rev, w)        # length 3n‑2
    grad_a = 2.0 * conv_rev[n - 1: 2 * n - 1]

    # chain rule through soft‑max
    dot_ag = np.dot(a, grad_a)
    grad_z = a * (grad_a - dot_ag)
    return float(f), grad_z.astype(np.float64)


def _l_bfgs_refine(start_a: np.ndarray, deadline: float,
                   beta_list=(200.0, 800.0, 4000.0, 12000.0),
                   max_iter_per_beta=1500) -> tuple[np.ndarray, float]:
    """
    Refine ``start_a`` using L‑BFGS‑B on the smoothed surrogate.
    Returns the best vector found and its exact score.
    """
    if not _HAS_SCIPY:
        return start_a, _score(start_a)

    best_a = start_a.copy()
    best_sc = _score(best_a)
    eps = 1e-12
    z = np.log(best_a + eps)               # log‑soft‑max parameters

    for beta in beta_list:
        if time.time() >= deadline - 5.0:
            break
        remaining = deadline - time.time()
        max_iter = min(max_iter_per_beta, int(remaining * 8))  # ~8 iters / sec
        if max_iter <= 0:
            break

        def fun(z_vec):
            return _smooth_obj_and_grad(z_vec, beta, best_a.shape[0])

        try:
            opt_res = _opt.fmin_l_bfgs_b(
                fun,
                z,
                maxiter=max_iter,
                disp=False,
                factr=1e7,
                pgtol=1e-12,
            )
        except Exception as exc:  # pragma: no cover
            print(f"[{time.time():.1f}] L‑BFGS failed (β={beta}): {exc}")
            continue

        z = opt_res[0]
        a_opt = _softmax(z)
        sc_opt = _score(a_opt)
        if sc_opt < best_sc:
            best_sc = sc_opt
            best_a = a_opt.copy()
            print(f"[{time.time():.1f}] L‑BFGS β={beta:.0f} → score={best_sc:.6f}")

    return best_a, best_sc


# ----------------------------------------------------------------------
# Mirror (exponentiated) sub‑gradient descent on the true max objective
# ----------------------------------------------------------------------


def _mirror_subgrad(start_a: np.ndarray, deadline: float,
                    step0: float = 1.2,
                    check_interval: int = 4000) -> tuple[np.ndarray, float]:
    """
    Multiplicative (exponentiated) sub‑gradient descent on
        F(a) = 2·n·max(a*a)
    with constant‑step schedule η_t = step0 / sqrt(t).
    """
    n = start_a.shape[0]
    a = start_a.astype(np.float64).copy()
    best_a = a.copy()
    best_sc = _score(a)

    t = 0
    while time.time() < deadline:
        t += 1
        conv = _conv_fft(a, a)
        k = int(np.argmax(conv))

        # gradient of the active convolution entry
        idx = np.arange(n)
        j = k - idx
        mask = (j >= 0) & (j < n)
        g = np.zeros_like(a)
        g[mask] = 2.0 * a[j[mask]]

        eta = step0 / math.sqrt(t)
        a = a * np.exp(-eta * g)          # multiplicative update

        # renormalise onto simplex (entropy projection)
        s = a.sum()
        if s == 0.0:
            a = np.full_like(a, 1.0 / n)
        else:
            a = a / s

        if t % check_interval == 0:
            sc = _score(a)
            if sc < best_sc:
                best_sc = sc
                best_a = a.copy()
                print(f"[{time.time():.1f}] mirror iter {t} → best={best_sc:.6f}")

    # final check
    sc = _score(a)
    if sc < best_sc:
        best_sc = sc
        best_a = a.copy()
    return best_a, best_sc


# ----------------------------------------------------------------------
# Simulated‑annealing hill‑climb (more aggressive)
# ----------------------------------------------------------------------


def _hill_climb_sa(start_a: np.ndarray, deadline: float,
                   max_steps: int = 2_000_000,
                   init_temp_factor: float = 0.05) -> tuple[np.ndarray, float]:
    """
    Random walk on the simplex with a modest temperature.
    * swap two entries* or *transfer a fraction of mass*.
    Temperature decays linearly to zero.
    """
    n = start_a.size
    cur = start_a.copy()
    cur_sc = _score(cur)
    best = cur.copy()
    best_sc = cur_sc

    T0 = best_sc * init_temp_factor
    T = T0
    step_scale = 0.12

    for step in range(1, max_steps + 1):
        if time.time() >= deadline:
            break

        cand = cur.copy()
        if np.random.rand() < 0.5:          # swap
            i, j = np.random.randint(n, size=2)
            cand[i], cand[j] = cand[j], cand[i]
        else:                               # transfer a fraction
            i, j = np.random.randint(n, size=2)
            if i != j:
                delta = step_scale * cur[i]
                cand[i] -= delta
                cand[j] += delta
                if cand[i] < 0.0:
                    cand[i] = 0.0

        cand = cand / cand.sum()
        cand_sc = _score(cand)
        delta = cand_sc - cur_sc

        # Metropolis acceptance
        if delta < 0.0 or np.random.rand() < np.exp(-delta / (T + 1e-18)):
            cur = cand
            cur_sc = cand_sc
            if cur_sc < best_sc:
                best_sc = cur_sc
                best = cur.copy()
                step_scale = min(step_scale * 1.07, 0.5)
        else:
            step_scale *= 0.97

        # linear cooling
        T = T0 * (1.0 - step / max_steps)

    return best, best_sc

# High‑level driver

def propose_candidate(seed: int = 12345, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative sequence minimising ``evaluate_sequence``.
    Returns a plain Python list of floats.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 2.0    # safety margin

    # -------------------------
    # Choose a length
    # -------------------------
    if budget_s >= 950:
        n = 32768
    elif budget_s >= 850:
        n = 16384
    else:
        n = 8192
    print(f"[{time.time():.1f}] chosen length n = {n}")

    best_vec = None
    best_score = float("inf")

    # -------------------------------------------------
    # 0. optional fallback (if global variable exists)
    # -------------------------------------------------
    try:
        fb_vec = [float(x) for x in height_sequence_1]   # type: ignore
        fb = evaluate_sequence(fb_vec)
        print(f"[{time.time():.1f}] fallback score = {fb:.6f}")
        if fb < best_score:
            best_score = fb
            best_vec = fb_vec
    except Exception:
        pass

    # -------------------------------------------------
    # 1. Scan analytic families (coarse but cheap)
    # -------------------------------------------------
    # symmetric product (U‑shaped)
    for alpha in np.linspace(0.05, 30.0, 151):
        if time.time() >= deadline - 30.0:
            break
        vec = _sym_product_family(n, alpha)
        sc = _score(vec)
        if sc < best_score:
            best_score = sc
            best_vec = vec.tolist()
            print(f"[{time.time():.1f}] new best (sym‑α={alpha:.2f}) → {best_score:.6f}")

    # left‑heavy power‑law
    for beta in np.linspace(0.05, 30.0, 151):
        if time.time() >= deadline - 30.0:
            break
        vec = _left_power_family(n, beta)
        sc = _score(vec)
        if sc < best_score:
            best_score = sc
            best_vec = vec.tolist()
            print(f"[{time.time():.1f}] new best (pow‑β={beta:.2f}) → {best_score:.6f}")

    # exponential decay
    for gamma in np.linspace(0.01, 15.0, 151):
        if time.time() >= deadline - 30.0:
            break
        vec = _exp_decay_family(n, gamma)
        sc = _score(vec)
        if sc < best_score:
            best_score = sc
            best_vec = vec.tolist()
            print(f"[{time.time():.1f}] new best (exp‑γ={gamma:.2f}) → {best_score:.6f}")

    # beta distribution (U‑shaped for α<1)
    for alpha in np.linspace(0.15, 0.95, 41):
        if time.time() >= deadline - 30.0:
            break
        vec = _beta_family(n, alpha)
        sc = _score(vec)
        if sc < best_score:
            best_score = sc
            best_vec = vec.tolist()
            print(f"[{time.time():.1f}] new best (beta‑α={alpha:.2f}) → {best_score:.6f}")

    # random starts (diverse)
    for _ in range(80):
        if time.time() >= deadline - 30.0:
            break
        vec = _random_monotone_start(n)
        sc = _score(vec)
        if sc < best_score:
            best_score = sc
            best_vec = vec.tolist()
            print(f"[{time.time():.1f}] new best (rand‑mono) → {best_score:.6f}")

    for _ in range(60):
        if time.time() >= deadline - 30.0:
            break
        vec = _random_sparse_start(n)
        sc = _score(vec)
        if sc < best_score:
            best_score = sc
            best_vec = vec.tolist()
            print(f"[{time.time():.1f}] new best (rand‑sparse) → {best_score:.6f}")

    for _ in range(40):
        if time.time() >= deadline - 30.0:
            break
        vec = _mixed_start(n)
        sc = _score(vec)
        if sc < best_score:
            best_score = sc
            best_vec = vec.tolist()
            print(f"[{time.time():.1f}] new best (mixed) → {best_score:.6f}")

    # -------------------------------------------------
    # 2. L‑BFGS refinement on the smooth surrogate
    # -------------------------------------------------
    if best_vec is not None and time.time() < deadline - 300.0:
        print(f"[{time.time():.1f}] starting L‑BFGS refinement")
        a_start = np.array(best_vec, dtype=np.float64)
        a_ref, sc_ref = _l_bfgs_refine(
            a_start,
            deadline - 200.0,
            beta_list=(200.0, 800.0, 4000.0, 12000.0),
            max_iter_per_beta=1500,
        )
        if sc_ref < best_score:
            best_score = sc_ref
            best_vec = a_ref.tolist()
            print(f"[{time.time():.1f}] after L‑BFGS → best={best_score:.6f}")

    # -------------------------------------------------
    # 3. Mirror sub‑gradient descent (true max)
    # -------------------------------------------------
    if best_vec is not None and time.time() < deadline - 80.0:
        print(f"[{time.time():.1f}] starting mirror sub‑gradient")
        a_sub, sc_sub = _mirror_subgrad(
            np.array(best_vec, dtype=np.float64),
            deadline - 20.0,
            step0=1.2,
        )
        if sc_sub < best_score:
            best_score = sc_sub
            best_vec = a_sub.tolist()
            print(f"[{time.time():.1f}] after mirror sub‑grad → best={best_score:.6f}")

    # -------------------------------------------------
    # 4. Symmetrisation (many optima are symmetric)
    # -------------------------------------------------
    if best_vec is not None:
        a_arr = np.array(best_vec, dtype=np.float64)
        a_sym = (a_arr + a_arr[::-1]) * 0.5
        a_sym = _project_simplex(a_sym)
        sc_sym = _score(a_sym)
        if sc_sym < best_score:
            best_score = sc_sym
            best_vec = a_sym.tolist()
            print(f"[{time.time():.1f}] symmetry improved → best={best_score:.6f}")

    # -------------------------------------------------
    # 5. Aggressive hill‑climb (SA)
    # -------------------------------------------------
    if best_vec is not None and time.time() < deadline - 5.0:
        print(f"[{time.time():.1f}] final hill‑climb (SA)")
        a_final, sc_final = _hill_climb_sa(
            np.array(best_vec, dtype=np.float64),
            deadline,
            max_steps=2_000_000,
            init_temp_factor=0.05,   # 5 % of current score → more freedom
        )
        if sc_final < best_score:
            best_score = sc_final
            best_vec = a_final.tolist()
            print(f"[{time.time():.1f}] hill‑climb improved → best={best_score:.6f}")

    # -------------------------------------------------
    # Safeguard: fall back to a modest uniform vector if needed
    # -------------------------------------------------
    if best_vec is None:
        n_small = 256
        best_vec = [1.0 / n_small] * n_small
        best_score = evaluate_sequence(best_vec)

    # -------------------------------------------------
    # Final official evaluation (the harness will also call it)
    # -------------------------------------------------
    final_score = evaluate_sequence(best_vec)
    print(f"\n[{time.time():.1f}] finished – final reported score = {final_score:.6f}")
    return best_vec
