"""
Non‑negative step‑function optimiser for ``evaluate_sequence``.

The algorithm consists of three stages:
1. **Broad parametric exploration** – generate a variety of
   analytically‑parameterised shapes (power‑law, power‑law with a tail,
   symmetric product‑law, geometric, triangular and random Dirichlet
   vectors).  Each candidate is normalised to sum‑to‑1 and scored with
   the exact objective (2·n·max(conv(a,a))).  The best shape found
   becomes the starting point for the next stage.
2. **Long projected sub‑gradient polish with momentum** – from the best
   candidate we run a very large number of projected sub‑gradient steps
   on the non‑smooth objective 2·n·max(conv(a,a)).  The gradient at the
   currently active convolution index k is g_i = 2·a_{k‑i} (zero when
   out of range).  A simple exponential‑moving‑average (momentum) is
   used to accelerate convergence, together with a decaying step‑size
   (∝1/√t).  Occasionally a tiny random jitter is added to escape
   plateaus.  The whole remaining time budget is devoted to this phase.
3. **Final reporting** – the best vector obtained is returned as a plain
   Python list; ``evaluate_sequence`` is finally called to obtain the
   official score.

The routine respects the 1000 s wall‑clock limit, prints occasional
progress messages and never uses filesystem or network I/O.
"""

import time
import numpy as np

# ------------------------------------------------------------
# Numerical utilities
# ------------------------------------------------------------

def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of v onto the probability simplex
    { x >= 0, sum(x) = 1 }.
    """
    if np.all(v <= 0):
        # Uniform distribution is the projection of the zero vector.
        return np.full_like(v, 1.0 / v.shape[0])

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, v.shape[0] + 1)
    mask = u - cssv / ind > 0
    rho = np.where(mask)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def full_convolution(a: np.ndarray) -> np.ndarray:
    """
    Full linear convolution a*a using a real FFT.
    Length of the result is 2*len(a)-1.
    """
    n = a.shape[0]
    m = 2 * n - 1
    # next power of two >= m
    n_fft = 1 << ((m - 1).bit_length())
    A = np.fft.rfft(a, n=n_fft)
    C = A * A
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


def compute_score(a: np.ndarray) -> float:
    """
    Exact objective for a normalised vector (sum = 1):
        2·n·max(conv(a,a))
    """
    n = a.shape[0]
    conv = full_convolution(a)
    return float(2.0 * n * conv.max())


# ------------------------------------------------------------
# Shape generators
# ------------------------------------------------------------

def shape_power(n: int, p: float) -> np.ndarray:
    a = np.arange(1, n + 1, dtype=float) ** (-p)
    a /= a.sum()
    return a


def shape_power_tail(n: int, p: float, r: float) -> np.ndarray:
    """
    a_i ∝ (i+1)^{-p} * (n-i)^{r}
    """
    i = np.arange(1, n + 1, dtype=float)
    tail = (n - np.arange(n, dtype=float)) ** r
    a = (i ** (-p)) * tail
    a /= a.sum()
    return a


def shape_symmetric(n: int, p: float) -> np.ndarray:
    """
    a_i ∝ ((i+1)*(n-i))^{-p}, symmetric w.r.t. centre.
    """
    i = np.arange(1, n + 1, dtype=float)
    prod = i * (n - np.arange(n, dtype=float))
    a = prod ** (-p)
    a /= a.sum()
    return a


def shape_geometric(n: int, r: float) -> np.ndarray:
    """
    a_i ∝ r^i, truncated and normalised.
    """
    a = (r ** np.arange(n, dtype=float))
    a /= a.sum()
    return a


def shape_triangular(n: int) -> np.ndarray:
    """
    Symmetric triangular distribution (linear rise then fall).
    """
    half = n // 2
    if n % 2 == 0:
        up = np.arange(1, half + 1)
        down = up[::-1]
    else:
        up = np.arange(1, half + 2)
        down = up[:-1][::-1]
    a = np.concatenate([up, down]).astype(float)
    a /= a.sum()
    return a


# ------------------------------------------------------------
# Sub‑gradient polish with momentum
# ------------------------------------------------------------

def subgradient_polish_momentum(a0: np.ndarray, deadline: float,
                                step0: float = 0.12,
                                beta: float = 0.9) -> np.ndarray:
    """
    Projected sub‑gradient descent with exponential‑moving‑average
    momentum on the objective 2·n·max(conv(a,a)).
    The whole remaining time budget is used.
    """
    a = a0.copy()
    v = np.zeros_like(a)                # momentum buffer
    best_a = a.copy()
    best_val = compute_score(a)

    t = 0
    n = a.shape[0]

    while time.time() < deadline:
        t += 1

        conv = full_convolution(a)
        k = int(np.argmax(conv))

        # sub‑gradient of the active entry: g_i = 2 * a_{k-i} (if in range)
        idx = np.arange(n)
        j = k - idx
        mask = (j >= 0) & (j < n)
        g = np.zeros_like(a)
        g[mask] = 2.0 * a[j[mask]]

        step = step0 / np.sqrt(t)
        v = beta * v + (1.0 - beta) * g
        a = a - step * v
        a = project_to_simplex(a)

        val = compute_score(a)
        if val < best_val:
            best_val = val
            best_a = a.copy()
            # occasional progress report
            if t % 5000 == 0:
                print(f"[{time.time():.1f}] polish iter {t}, new best {best_val:.6f}")

        # tiny jitter every few thousand iterations to shake out of plateaus
        if t % 4000 == 0:
            jitter = np.random.randn(n) * (step0 * 0.008 / np.sqrt(t))
            a += jitter
            a = project_to_simplex(a)

    return best_a

    param_best_desc = ""

    # Helper to test a candidate shape
    def test_candidate(a_vec, description):
        nonlocal param_best_a, param_best_score, param_best_desc
        score = compute_score(a_vec)
        if score < param_best_score:
            param_best_score, param_best_a, param_best_desc = score, a_vec.copy(), description
            print(f"[sweep] new best {score:.6f} ({description})")

    # Power‑law family
    for p in np.linspace(0.50, 0.78, 57):   # step ≈0.005
        if time.time() >= deadline:
            break
        a = shape_power(n, p)
        test_candidate(a, f"power p={p:.4f}")

    # Power‑law + tail family
    for p in np.linspace(0.50, 0.78, 31):
        if time.time() >= deadline:
            break
        for r in np.linspace(0.0, 0.5, 26):
            if time.time() >= deadline:
                break
            a = shape_power_tail(n, p, r)
            test_candidate(a, f"powTail p={p:.4f} r={r:.4f}")

    # Symmetric product‑law family
    for p in np.linspace(0.50, 0.78, 57):
        if time.time() >= deadline:
            break
        a = shape_symmetric(n, p)
        test_candidate(a, f"sym p={p:.4f}")

    # Geometric family
    for r in np.linspace(0.990, 0.9999, 45):
        if time.time() >= deadline:
            break
        a = shape_geometric(n, r)
        test_candidate(a, f"geom r={r:.6f}")

    # Triangular shape (single candidate)
    a_tri = shape_triangular(n)
    test_candidate(a_tri, "triangular")

    # Random Dirichlet seeds (a few)
    for _ in range(5):
        if time.time() >= deadline:
            break
        a_rand = np.random.rand(n)
        a_rand /= a_rand.sum()
        test_candidate(a_rand, "random")

    # If we found any candidate, initialise best_seq with it
    if param_best_a is not None:
        best_seq = param_best_a.tolist()
        best_val = evaluate_sequence(best_seq)
        print(f"[sweep] best sweep score (official) = {best_val:.6f}")

    # ---------------------------------------------------------
    # 2. Extensive sub‑gradient polishing (majority of time)
    # ---------------------------------------------------------
    if best_seq is not None and time.time() < deadline - 2.0:
        a_start = np.array(best_seq, dtype=float)
        print("[polish] starting long momentum sub‑gradient polish...")
        a_refined = subgradient_polish_momentum(a_start, deadline)
        refined_score = compute_score(a_refined)
        print(f"[polish] refined raw score = {refined_score:.6f}")

        # official evaluation
        final_seq = a_refined.tolist()
        final_score = evaluate_sequence(final_seq)
        print(f"[polish] official score = {final_score:.6f}")

        if final_score < best_val:
            best_val, best_seq = final_score, final_seq

    # ---------------------------------------------------------
    # 3. Fallback to uniform if nothing succeeded
    # ---------------------------------------------------------
    if best_seq is None:
        best_seq = [1.0 / n] * n
        best_val = evaluate_sequence(best_seq)

    # ---------------------------------------------------------
    # 4. Final report
    # ---------------------------------------------------------
    final_score = evaluate_sequence(best_seq)
    print(f"\nSearch finished. Reported score = {final_score:.6f}")
    return best_seq
