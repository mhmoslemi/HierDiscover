def mirror_subgrad_optimize(start_a, time_deadline, max_iters=500000, eta0=1.0):
    a = start_a.copy()
    best_a = a.copy()
    best_score = compute_score(a)
    n = len(a)
    for t in range(1, max_iters+1):
        if time.time() >= time_deadline:
            break
        conv = convolution_fft(a, a)
        k = int(np.argmax(conv))
        # compute g
        idx = np.arange(n)
        j = k - idx
        mask = (j >= 0) & (j < n)
        g = np.zeros_like(a)
        g[mask] = 2.0 * a[j[mask]]
        avg = np.dot(a, g)  # equals 2*max_conv
        eta = eta0 / np.sqrt(t)
        # compute multiplicative update
        # Clip exponent to avoid overflow:
        exp_arg = -eta * (g - avg)
        # Limit exp_arg
        exp_arg = np.clip(exp_arg, -50, 50)
        a = a * np.exp(exp_arg)
        # renormalize
        sum_a = a.sum()
        if sum_a == 0:
            # fallback: add epsilon uniform
            a = np.full(n, 1.0 / n)
        else:
            a = a / sum_a
        # maybe occasionally evaluate score
        if t % 500 == 0:
            cur_score = compute_score(a)
            if cur_score < best_score:
                best_score = cur_score
                best_a = a.copy()

"""
Non‑negative step‑function optimiser for ``evaluate_sequence``.

Main ideas
----------
1. Build a diverse set of analytic seeds (uniform, central‑binomial,
   power‑law, geometric, Beta‑shaped) and keep the best.

2. Refine the best seed with a *mirrored* sub‑gradient descent:
   we minimise 2·n·max(conv(a,a)) on the probability simplex using an
   exponentiated‑gradient (KL‑mirror) update based on the sub‑gradient
   of the max‑convolution (the active peak).

3. Polish the result with a cheap stochastic hill‑climb (swap / mass‑move)
   to squeeze out the last bits of improvement.

All phases respect a wall‑clock deadline (default 1000 s) and keep the
best sequence encountered.  The final sequence is returned as a plain
list of floats (already on the simplex), ready for ``evaluate_sequence``.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Basic linear‑algebra utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """
    Full linear convolution using a real FFT.
    Returns an array of length len(a)+len(b)-1.
    """
    m = a.shape[0] + b.shape[0] - 1
    n_fft = next_pow2(m)
    A = np.fft.rfft(a, n=n_fft)
    B = np.fft.rfft(b, n=n_fft)
    C = A * B
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


def compute_score(a: np.ndarray) -> float:
    """
    Exact objective for a vector already normalised to sum = 1.
    Returns 2·n·max(conv(a,a)).
    """
    n = a.shape[0]
    conv = convolution_fft(a, a)
    return 2.0 * n * float(conv.max())


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of v onto the probability simplex:
        sum(v) = 1, v >= 0.
    """
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / v.shape[0])

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, v.shape[0] + 1)
    cond = u - cssv / ind > 0
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


# ----------------------------------------------------------------------
# Analytic families for initial candidates
# ----------------------------------------------------------------------


def generate_central_binomial(n: int) -> np.ndarray:
    """a_i = C(2i,i)/4^i, normalised to sum 1 (monotonically decreasing)."""
    a = np.empty(n, dtype=np.float64)
    a[0] = 1.0
    for i in range(1, n):
        a[i] = a[i - 1] * (2.0 * i - 1.0) / (2.0 * i)
    return project_to_simplex(a)


def generate_powerlaw(n: int, p: float) -> np.ndarray:
    """a_i ∝ (i+1)^(-p), normalised to sum 1."""
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = idx ** (-p)
    return project_to_simplex(a)


def generate_geometric(n: int, r: float) -> np.ndarray:
    """Geometric distribution a_i = (1‑r)·r^i, normalised (exact if r<1)."""
    i = np.arange(n, dtype=np.float64)
    a = (1.0 - r) * (r ** i)
    return project_to_simplex(a)


def generate_beta_shape(n: int, alpha: float) -> np.ndarray:
    """
    Discrete Beta(α,α) shape:
        a_i ∝ (i+1)^{α‑1}·(n‑i)^{α‑1}
    Symmetric and heavy‑tailed for α<0.5.
    """
    i = np.arange(1, n + 1, dtype=np.float64)          # i = 1 .. n
    left = i ** (alpha - 1.0)
    right = (n - i + 1) ** (alpha - 1.0)
    a = left * right
    return project_to_simplex(a)


# ----------------------------------------------------------------------
# Mirror‑descent optimiser on the true max‑convolution objective
# ----------------------------------------------------------------------


def mirror_subgrad_optimize(
    start_a: np.ndarray,
    time_deadline: float,
    max_iters: int = 500_000,
    eta0: float = 0.8,
) -> tuple[np.ndarray, float]:
    """
    KL‑mirror (exponentiated‑gradient) sub‑gradient descent on
    f(a)=2·n·max(conv(a,a)).  a is kept on the simplex.
    """
    a = start_a.astype(np.float64).copy()
    best_a = a.copy()
    best_score = compute_score(a)
    n = a.shape[0]

    for t in range(1, max_iters + 1):
        if time.time() >= time_deadline:
            break

        # full auto‑convolution and active peak
        conv = convolution_fft(a, a)
        k = int(np.argmax(conv))

        # sub‑gradient of the active entry (see derivation in analysis)
        idx = np.arange(n)
        j = k - idx
        mask = (j >= 0) & (j < n)
        g = np.zeros_like(a)
        g[mask] = 2.0 * a[j[mask]]

        # average of the sub‑gradient under the current a (equals 2·max_conv)
        avg = np.dot(a, g)

        # diminishing stepsize
        eta = eta0 / np.sqrt(t)

        # exponentiated‑gradient update (KL‑mirror)
        exp_arg = -eta * (g - avg)
        # keep exponent in a safe range
        exp_arg = np.clip(exp_arg, -50.0, 50.0)
        a = a * np.exp(exp_arg)

        # renormalise onto the simplex
        sum_a = a.sum()
        if sum_a == 0.0:
            a = np.full_like(a, 1.0 / n)
        else:
            a = a / sum_a

        # occasional progress check
        if t % 500 == 0:
            cur_score = compute_score(a)
            if cur_score < best_score:
                best_score = cur_score
                best_a = a.copy()

    # final evaluation
    final_score = compute_score(a)
    if final_score < best_score:
        best_score, best_a = final_score, a.copy()

    return best_a, best_score


# ----------------------------------------------------------------------
# Final stochastic hill‑climb (tiny local moves)
# ----------------------------------------------------------------------


def hill_climb_permutation(
    seq,
    time_deadline,
    max_steps: int = 300_000,
) -> tuple[list[float], float]:
    """
    Very local stochastic moves (swap or fractional mass transfer) with
    simplex projection.  Keeps the best seen sequence.
    """
    a = np.array(seq, dtype=np.float64)
    n = a.shape[0]
    best_a = a.copy()
    best_score = compute_score(best_a)

    step_scale = 0.08  # initial fraction of mass moved

    while max_steps > 0 and time.time() < time_deadline:
        cand = best_a.copy()
        if np.random.rand() < 0.5:      # swap two entries
            i, j = np.random.randint(n, size=2)
            cand[i], cand[j] = cand[j], cand[i]
        else:                           # move a fraction from i to j
            i, j = np.random.randint(n, size=2)
            if i != j:
                delta = step_scale * best_a[i]
                cand[i] -= delta
                cand[j] += delta
                if cand[i] < 0.0:
                    cand[i] = 0.0

        cand = project_to_simplex(cand)
        cur_score = compute_score(cand)

        if cur_score < best_score:
            best_score = cur_score
            best_a = cand
            step_scale = min(step_scale * 1.06, 0.5)
        else:
            step_scale *= 0.97

        max_steps -= 1

    return best_a.tolist(), best_score

# Main entry point required by the harness
# ----------------------------------------------------------------------

    """

    start_time = time.time()
    # keep a safety margin for the harness
    deadline = start_time + budget_s - 5.0

    # ------------------------------------------------------------------
    # 0. Problem size
    # ------------------------------------------------------------------
    try:
        n = len(height_sequence_1)          # type: ignore[name-defined]
        print(f"Using provided height_sequence_1 of length {n}.")
    except Exception:
        n = 2048
        print(f"No external sequence provided – using length {n}.")

    # ------------------------------------------------------------------
    # 1. Build a collection of analytic starters
    # ------------------------------------------------------------------
    init_candidates = []

    # uniform
    init_candidates.append(np.full(n, 1.0 / n, dtype=np.float64))

    # central binomial (classic)
    init_candidates.append(generate_central_binomial(n))

    # power‑law family
    for p in np.linspace(0.3, 1.0, 9):
        init_candidates.append(generate_powerlaw(n, float(p)))

    # geometric family (r close to 1)
    for r in [0.97, 0.98, 0.985, 0.99, 0.992, 0.995]:
        init_candidates.append(generate_geometric(n, float(r)))

    # Beta‑shaped family (α in (0.1,0.5])
    for alpha in np.linspace(0.10, 0.55, 10):
        init_candidates.append(generate_beta_shape(n, float(alpha)))

    # possibly include the previously found best construction
    try:
        init_candidates.append(np.array(height_sequence_1, dtype=np.float64))
        print("Added provided height_sequence_1 as a starter.")
    except Exception:
        pass

    # ------------------------------------------------------------------
    # 2. Choose the best starter by exact evaluation
    # ------------------------------------------------------------------
    best_seq = None
    best_val = float("inf")
    for idx, cand in enumerate(init_candidates):
        cand = project_to_simplex(cand)
        val = compute_score(cand)
        print(f"Starter {idx:02d}: score = {val:.6f}")
        if val < best_val:
            best_val = val
            best_seq = cand.copy()
    print(f"Best starter score = {best_val:.6f}")

    # ------------------------------------------------------------------
    # 3. Mirror‑descent optimisation (main workhorse)
    # ------------------------------------------------------------------
    mirror_deadline = deadline - 100.0   # reserve time for polishing
    if time.time() < mirror_deadline:
        remaining = mirror_deadline - time.time()
        # very generous iteration budget (assume ~0.0008 s per iteration)
        max_iters = int(min(2_000_000, remaining * 1200))
        print(f"Mirror sub‑gradient phase – up to {max_iters} iterations.")
        a_opt, opt_score = mirror_subgrad_optimize(
            best_seq,
            time_deadline=mirror_deadline,
            max_iters=max_iters,
            eta0=0.8,
        )
        if opt_score < best_val:
            best_val, best_seq = opt_score, a_opt.copy()
            print(f"Mirror phase improved score to {best_val:.6f}")

    # ------------------------------------------------------------------
    # 4. Short random‑restart bursts (optional, cheap)
    # ------------------------------------------------------------------
    restart_deadline = deadline - 50.0
    if time.time() < restart_deadline:
        base = best_seq.copy()
        for r in range(4):
            if time.time() >= restart_deadline:
                break
            pert = base + 0.015 * np.random.randn(n)
            pert = project_to_simplex(pert)
            sub_dead = time.time() + 10.0
            a_tmp, s_tmp = mirror_subgrad_optimize(
                pert,
                time_deadline=sub_dead,
                max_iters=150_000,
                eta0=0.6,
            )
            if s_tmp < best_val:
                best_val, best_seq = s_tmp, a_tmp.copy()
                print(f"Restart {r} yielded improvement: {best_val:.6f}")

    # ------------------------------------------------------------------
    # 5. Final stochastic hill‑climb
    # ------------------------------------------------------------------
    hill_deadline = deadline - 2.0
    if time.time() < hill_deadline:
        steps = 300_000
        print(f"Final hill‑climb for up to {steps} steps.")
        polished_seq, hill_score = hill_climb_permutation(
            best_seq.tolist(),
            time_deadline=hill_deadline,
            max_steps=steps,
        )
        if hill_score < best_val:
            best_val, best_seq = hill_score, np.array(polished_seq, dtype=np.float64)
            print(f"Hill‑climb improved final score to {best_val:.6f}")

    # ------------------------------------------------------------------
    # 6. Safeguard – fall back to uniform if something went wrong
    # ------------------------------------------------------------------
    if best_seq is None:
        best_seq = np.full(n, 1.0 / n, dtype=np.float64)
        best_val = compute_score(best_seq)

    # ------------------------------------------------------------------
    # 7. Final reporting using the provided evaluate_sequence()
    # ------------------------------------------------------------------
    final_score = evaluate_sequence(best_seq.tolist())
    print(f"\nSearch finished. Final reported score = {final_score:.6f}")

    # Return a plain list of floats (already on the simplex)
    return best_seq.tolist()
