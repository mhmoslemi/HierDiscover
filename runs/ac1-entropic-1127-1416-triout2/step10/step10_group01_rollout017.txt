def evaluate_sequence(sequence: list[float]) -> float:
    """
    Evaluates a sequence of coefficients with enhanced security checks.
    Returns np.inf if the input is invalid.
    """
    # --- Security Checks ---

    # Verify that the input is a list
    if not isinstance(sequence, list):
        return np.inf

    # Reject empty lists
    if not sequence:
        return np.inf

    # Check each element in the list for validity
    for x in sequence:
        # Reject boolean types (as they are a subclass of int) and
        # any other non-integer/non-float types (like strings or complex numbers).
        if isinstance(x, bool) or not isinstance(x, (int, float)):
            return np.inf

        # Reject Not-a-Number (NaN) and infinity values.
        if np.isnan(x) or np.isinf(x):
            return np.inf

    # Convert all elements to float for consistency
    sequence = [float(x) for x in sequence]

    # Protect against negative numbers
    sequence = [max(0, x) for x in sequence]

    # Protect against numbers that are too large
    sequence = [min(1000.0, x) for x in sequence]

    n = len(sequence)
    b_sequence = np.convolve(sequence, sequence)
    max_b = max(b_sequence)
    sum_a = np.sum(sequence)

    # Protect against the case where the sum is too close to zero
    if sum_a < 0.01:
        return np.inf

    return float(2 * n * max_b / (sum_a**2))

"""
Propose a step‑function (non‑negative sequence) that minimises

        2·n·max(convolve(a,a)) / (sum(a)**2)

under the simplex constraint.
The implementation follows a multi‑phase strategy:

0.  Load a fallback candidate if the harness supplies ``height_sequence_1``.
1.  Generate many random Dirichlet / analytic starts and keep the best.
2.  Refine the best start with a smooth‑max Adam optimiser (softmax
    parametrisation, exponentially increasing β).
3.  Cheap projected sub‑gradient descent on the exact max‑objective.
4.  An extended LP‑based refinement loop (the “state‑of‑the‑art”
    iteration) with a dynamic mixing weight.
5.  Symmetrisation (a ← (a+rev(a))/2) if it improves the score.
6.  A final long simulated‑annealing hill‑climb.
All phases respect the global wall‑clock budget (default 1000 s) and
the routine always returns a plain list of floats ready for
``evaluate_sequence``.
"""

import time
import math
import numpy as np
from scipy.optimize import linprog

# ----------------------------------------------------------------------
# Core helpers
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full (aperiodic) convolution using real FFT (zero‑padded to a power of 2)."""
    m = a.size + b.size - 1
    nfft = next_pow2(m)
    A = np.fft.rfft(a, n=nfft)
    B = np.fft.rfft(b, n=nfft)
    C = A * B
    conv = np.fft.irfft(C, n=nfft)[:m]
    return conv


def compute_score(a: np.ndarray) -> float:
    """Exact objective for a probability vector (∑a = 1):
           2·n·max(a*a)
    """
    n = a.shape[0]
    conv = convolution_fft(a, a)
    return float(2.0 * n * conv.max())


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex (∑v = 1, v≥0)."""
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / v.shape[0])

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, v.shape[0] + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.where(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def softmax(z: np.ndarray) -> np.ndarray:
    """Stable soft‑max."""
    z_max = np.max(z)
    e = np.exp(z - z_max)
    return e / e.sum()


# ----------------------------------------------------------------------
# Analytic families – used for the quick diverse scan
# ----------------------------------------------------------------------


def symmetric_product_family(n: int, alpha: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    logprod = np.log(i + 0.5) + np.log(n - 0.5 - i)
    w = np.exp(-alpha * logprod)
    a = w / w.sum()
    return a


def left_heavy_power_family(n: int, beta: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    w = np.exp(-beta * np.log(i + 0.5))
    a = w / w.sum()
    return a


def exp_u_family(n: int, gamma: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    dist = np.minimum(i, n - 1 - i)
    w = np.exp(-gamma * dist)
    a = w / w.sum()
    return a


def random_monotone_start(n: int) -> np.ndarray:
    r = np.random.rand(n) ** np.random.uniform(0.5, 5.0)
    r.sort()
    a = r[::-1]
    a /= a.sum()
    return a


def random_u_start(n: int) -> np.ndarray:
    gamma = np.random.uniform(0.0, 6.0)
    return exp_u_family(n, gamma)


# ----------------------------------------------------------------------
# 0.  Fallback candidate (if supplied by the harness)
# ----------------------------------------------------------------------


def load_fallback():
    """If the global variable ``height_sequence_1`` is present, use it."""
    try:
        fb_vec = [float(x) for x in height_sequence_1]  # type: ignore
        return fb_vec
    except Exception:
        return None


# ----------------------------------------------------------------------
# 1.  Quick random / analytic scan
# ----------------------------------------------------------------------


def quick_scan(n: int, deadline: float, best_score: float, best_seq):
    """Generate a few hundred random / analytic vectors and keep the best."""
    # Analytic families
    alphas = np.linspace(0.0, 12.0, 13)
    betas = np.linspace(0.0, 12.0, 13)
    gammas = np.linspace(0.0, 6.0, 13)

    for alpha in alphas:
        if time.time() >= deadline:
            return best_score, best_seq
        vec = symmetric_product_family(n, alpha)
        sc = compute_score(vec)
        if sc < best_score:
            best_score, best_seq = sc, vec.tolist()

    for beta in betas:
        if time.time() >= deadline:
            return best_score, best_seq
        vec = left_heavy_power_family(n, beta)
        sc = compute_score(vec)
        if sc < best_score:
            best_score, best_seq = sc, vec.tolist()

    for gamma in gammas:
        if time.time() >= deadline:
            return best_score, best_seq
        vec = exp_u_family(n, gamma)
        sc = compute_score(vec)
        if sc < best_score:
            best_score, best_seq = sc, vec.tolist()

    # Random draws (Dirichlet, monotone, U‑shaped)
    for _ in range(300):
        if time.time() >= deadline:
            break
        if np.random.rand() < 0.5:
            vec = random_monotone_start(n)
        else:
            vec = random_u_start(n)
        sc = compute_score(vec)
        if sc < best_score:
            best_score, best_seq = sc, vec.tolist()
    return best_score, best_seq


# ----------------------------------------------------------------------
# 2.  Adam optimiser on the smooth‑max surrogate
# ----------------------------------------------------------------------


def adam_smooth_optimize(start_a: np.ndarray,
                         deadline: float,
                         max_iters: int = 200_000,
                         beta_start: float = 4.0,
                         beta_end: float = 800.0,
                         lr: float = 0.018) -> tuple[np.ndarray, float]:
    """Adam on the smooth surrogate of max‑convolution."""
    eps = 1e-12
    z = np.log(start_a + eps)

    m = np.zeros_like(z)
    v = np.zeros_like(z)
    beta1 = 0.9
    beta2 = 0.999
    eps_adam = 1e-12

    best_a = softmax(z)
    best_score = compute_score(best_a)

    for it in range(1, max_iters + 1):
        if time.time() >= deadline:
            break

        # exponential schedule for beta
        t = it / max_iters
        beta = beta_start * (beta_end / beta_start) ** t

        a = softmax(z)
        conv = convolution_fft(a, a)
        maxc = conv.max()

        # smooth‑max weights
        exp_w = np.exp(beta * (conv - maxc))
        w = exp_w / exp_w.sum()

        # gradient wrt a (see derivation in the original script)
        a_rev = a[::-1]
        conv_rev = convolution_fft(a_rev, w)
        grad_a = 2.0 * conv_rev[len(a) - 1:len(a) - 1 + len(a)]

        # back‑prop through soft‑max
        dot_ag = np.dot(a, grad_a)
        grad_z = a * grad_a - dot_ag * a

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad_z
        v = beta2 * v + (1.0 - beta2) * (grad_z ** 2)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        z -= lr * m_hat / (np.sqrt(v_hat) + eps_adam)

        # occasional bookkeeping
        if it % 1000 == 0:
            a_tmp = softmax(z)
            sc = compute_score(a_tmp)
            if sc < best_score:
                best_score, best_a = sc, a_tmp.copy()
    # final check
    a_fin = softmax(z)
    sc_fin = compute_score(a_fin)
    if sc_fin < best_score:
        best_score, best_a = sc_fin, a_fin.copy()
    return best_a, best_score


# ----------------------------------------------------------------------
# 3.  Sub‑gradient descent on the true max‑objective
# ----------------------------------------------------------------------


def subgrad_optimize(start_a: np.ndarray,
                     deadline: float,
                     max_iters: int = 1_000_000,
                     step0: float = 0.07) -> tuple[np.ndarray, float]:
    """Very cheap projected sub‑gradient descent using the exact max‑convolution."""
    n = start_a.size
    a = start_a.astype(np.float64).copy()

    best_a = a.copy()
    best_score = compute_score(a)

    for t in range(1, max_iters + 1):
        if time.time() >= deadline:
            break

        conv = convolution_fft(a, a)
        k = int(np.argmax(conv))

        idx = np.arange(n)
        j = k - idx
        mask = (j >= 0) & (j < n)
        g = np.zeros_like(a)
        g[mask] = 2.0 * a[j[mask]]
        # diagonal terms have coefficient 1 (already covered by the above since j==idx)
        diag = (j == idx) & mask
        if np.any(diag):
            g[diag] = a[diag]

        step = step0 / np.sqrt(t)
        a = project_to_simplex(a - step * g)

        if t % 3000 == 0:
            sc = compute_score(a)
            if sc < best_score:
                best_score, best_a = sc, a.copy()

    sc = compute_score(a)
    if sc < best_score:
        best_score, best_a = sc, a.copy()
    return best_a, best_score


# ----------------------------------------------------------------------
# 4.  LP‑based refinement (the classic “state‑of‑the‑art” iteration)
# ----------------------------------------------------------------------


def lp_refine_step(f: np.ndarray, M: float, mix_weight: float) -> np.ndarray:
    """Solve   max Σ g_j   s.t.   (f * g)_k ≤ M,
    then mix f ← (1‑t)·f + t·g_norm and re‑project onto the simplex."""
    n = f.shape[0]
    m = 2 * n - 1

    # Build convolution matrix A (A @ g = f * g)
    A = np.zeros((m, n), dtype=np.float64)
    for i in range(n):
        A[i:i + n, i] = f[i]

    c = -np.ones(n, dtype=np.float64)          # maximise Σ g ⇒ minimise −Σ g
    bounds = [(0.0, None) for _ in range(n)]
    b = np.full(m, M, dtype=np.float64)

    res = linprog(c, A_ub=A, b_ub=b, bounds=bounds, method='highs',
                  options={'presolve': True, 'time_limit': 15})
    if not res.success:
        return f

    g = np.maximum(res.x, 0.0)
    sum_g = g.sum()
    if sum_g <= 0.0:
        return f

    g_norm = g / sum_g
    f_new = (1.0 - mix_weight) * f + mix_weight * g_norm
    f_new = project_to_simplex(f_new)
    return f_new


def lp_refinement_loop(a: np.ndarray,
                       deadline: float,
                       start_mix: float = 0.2,
                       max_iters: int = 30) -> tuple[np.ndarray, float]:
    """Run a series of LP refinements until time runs out or no further progress."""
    mix = start_mix
    best_a = a.copy()
    best_score = compute_score(a)

    for it in range(max_iters):
        if time.time() >= deadline:
            break
        conv = convolution_fft(a, a)
        M = float(conv.max())
        a_new = lp_refine_step(a, M, mix_weight=mix)
        sc = compute_score(a_new)
        if sc < best_score - 1e-9:
            best_score, best_a = sc, a_new.copy()
            a = a_new
            # success – keep mixing weight moderate
            mix = max(0.05, mix * 0.95)
        else:
            # no improvement – become more conservative
            mix = max(0.02, mix * 0.7)
    return best_a, best_score


# ----------------------------------------------------------------------
# 5.  Symmetrisation helper
# ----------------------------------------------------------------------


def symmetrise(a: np.ndarray) -> np.ndarray:
    """Replace a by a symmetric version (a+rev(a))/2, projected onto the simplex."""
    a_sym = (a + a[::-1]) * 0.5
    return project_to_simplex(a_sym)


# ----------------------------------------------------------------------
# 6.  Final stochastic hill‑climb (simulated annealing)
# ----------------------------------------------------------------------


def hill_climb_sa(start_a: np.ndarray,
                  deadline: float,
                  max_steps: int = 2_000_000) -> tuple[np.ndarray, float]:
    n = start_a.size
    cur_a = start_a.copy()
    cur_score = compute_score(cur_a)

    best_a = cur_a.copy()
    best_score = cur_score

    T0 = best_score * 2e-4
    T = T0
    step_scale = 0.12

    for step in range(1, max_steps + 1):
        if time.time() >= deadline:
            break

        cand = cur_a.copy()
        if np.random.rand() < 0.5:                     # swap two entries
            i, j = np.random.randint(n, size=2)
            cand[i], cand[j] = cand[j], cand[i]
        else:                                          # move a little mass
            i, j = np.random.randint(n, size=2)
            if i != j:
                delta = step_scale * cur_a[i]
                cand[i] -= delta
                cand[j] += delta
                if cand[i] < 0.0:
                    cand[i] = 0.0

        cand = project_to_simplex(cand)
        cand_score = compute_score(cand)

        delta = cand_score - cur_score
        if delta < 0 or np.random.rand() < np.exp(-delta / (T + 1e-18)):
            cur_a, cur_score = cand, cand_score
            if cur_score < best_score:
                best_score, best_a = cur_score, cur_a.copy()
                step_scale = min(step_scale * 1.07, 0.5)
        else:
            step_scale *= 0.97

        # linear cooling
        T = T0 * (1.0 - step / max_steps)

    return best_a, best_score


# ----------------------------------------------------------------------
# Main driver – propose_candidate
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 12345,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """Search for a sequence of non‑negative heights that minimises the
    evaluation function.  The routine respects a wall‑clock budget and
    returns a list of floats ready for ``evaluate_sequence``.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 2.0   # tiny safety margin

    best_seq = None
    best_score = float("inf")

    # ------------------------------------------------------------------
    # 0.  Load fallback if present
    # ------------------------------------------------------------------
    fb = load_fallback()
    if fb is not None:
        fb_score = compute_score(np.asarray(fb, dtype=np.float64))
        print(f"[{time.time():.1f}] fallback score = {fb_score:.6f}")
        if fb_score < best_score:
            best_score, best_seq = fb_score, fb

    # ------------------------------------------------------------------
    # Determine problem size (default 2048)
    # ------------------------------------------------------------------
    n = len(best_seq) if best_seq is not None else 2048

    # ------------------------------------------------------------------
    # 1.  Quick scan of analytic / random families
    # ------------------------------------------------------------------
    if time.time() < deadline - 250.0:
        print(f"[{time.time():.1f}] quick scan (n={n})")
        best_score, best_seq = quick_scan(n, deadline, best_score, best_seq)

    # ------------------------------------------------------------------
    # Normalise the working vector to the simplex
    # ------------------------------------------------------------------
    if best_seq is None:
        a = np.full(n, 1.0 / n, dtype=np.float64)
    else:
        a = np.asarray(best_seq, dtype=np.float64)
    a = np.clip(a, 0.0, None)
    a = a / a.sum()
    best_seq = a.tolist()
    best_score = compute_score(a)

    # ------------------------------------------------------------------
    # 2.  Adam – smooth‑max optimisation
    # ------------------------------------------------------------------
    if time.time() < deadline - 180.0:
        print(f"[{time.time():.1f}] Adam optimisation start")
        a_opt, sc_opt = adam_smooth_optimize(
            a,
            deadline - 60.0,
            max_iters=250_000,
            beta_start=4.0,
            beta_end=900.0,
            lr=0.019,
        )
        if sc_opt < best_score:
            best_score, a = sc_opt, a_opt
            print(f"[{time.time():.1f}] after Adam best = {best_score:.6f}")

    # ------------------------------------------------------------------
    # 3.  Sub‑gradient descent (very cheap)
    # ------------------------------------------------------------------
    if time.time() < deadline - 120.0:
        print(f"[{time.time():.1f}] sub‑gradient descent")
        a_opt, sc_opt = subgrad_optimize(
            a,
            deadline - 30.0,
            max_iters=2_500_000,
            step0=0.07,
        )
        if sc_opt < best_score:

    # ------------------------------------------------------------------
    # 4.  Extended LP refinement loop

    if time.time() < deadline - 60.0:

            max_iters=30,
        )
        if sc_opt < best_score:
            best_score, a = sc_opt, a_opt
            print(f"[{time.time():.1f}] after LP best = {best_score:.6f}")

    # ------------------------------------------------------------------
    # 5.  Symmetrisation
    # ------------------------------------------------------------------
    if time.time() < deadline - 10.0:
        a_sym = symmetrise(a)
        sc_sym = compute_score(a_sym)
        if sc_sym < best_score:

            print(f"[{time.time():.1f}] symmetrisation improved = {best_score:.6f}")

    # ------------------------------------------------------------------
    # 6.  Final hill‑climb (SA) for the remaining budget
    # ------------------------------------------------------------------
    if time.time() < deadline - 5.0:
        print(f"[{time.time():.1f}] final simulated‑annealing hill‑climb")
        a_opt, sc_opt = hill_climb_sa(a, deadline, max_steps=2_000_000)
        if sc_opt < best_score:
            best_score, a = sc_opt, a_opt
            print(f"[{time.time():.1f}] hill‑climb improved = {best_score:.6f}")

    # ------------------------------------------------------------------

    # ------------------------------------------------------------------
    final_seq = a.tolist()
    final_score = evaluate_sequence(final_seq)  # the harness will have this in scope

    return final_seq
