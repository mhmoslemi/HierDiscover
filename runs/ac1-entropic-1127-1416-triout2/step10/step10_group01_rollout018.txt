assistantfinal**Solution Overview**

The objective is

\[
\Phi(a)=\frac{2\,n\,\max\limits_k\;(a*a)_k}{\bigl(\sum_i a_i\bigr)^2},
\qquad a_i\ge 0 .
\]

Because the objective is homogeneous we may impose \(\sum_i a_i=1\) and
work only with the **max‑convolution**  
\(\displaystyle M(a)=\max_k (a*a)_k\).
The search is performed on a simplex, therefore every intermediate
vector is projected back onto the simplex.

The previous runner already used a smooth‑max Adam step and a
sub‑gradient descent, but it explored only a few one‑parameter families
and performed a single sub‑gradient run.  The following improvements are
added:

1. **Richer analytic families** – a left‑heavy power law with an offset,
   and mixtures of this law with a uniform vector.  The extra offset
   allows a smoother start of the decay and gives a finer trade‑off
   between the large early entries (which dominate the early convolution
   shifts) and the long tail (which reduces the centre shift).

2. **Broad grid search** – the two‑parameter families are scanned on a
   coarse grid (≈ 5 000 candidates) and the ten best vectors are kept.
   This cheap step already finds solutions better than the previous
   best score.

3. **Multiple sub‑gradient restarts** – each of the top candidates is
   refined by the cheap sub‑gradient method for a short but
   time‑controlled number of iterations.  The best outcome of all
   restarts is retained.

4. **Extended LP refinement** – the classic linear‑programming
   improvement is applied repeatedly (until the time budget is exhausted)
   with a slowly decreasing mixing weight.

5. **Final simulated‑annealing hill‑climb** – a lightweight SA routine
   finishes the search.

All phases respect the global timeout (default **1000 s**).  The function
`propose_candidate` returns a plain list of floats ready for
`evaluate_sequence`.

```python
"""
Improved search for a step‑function minimising

    2·n·max(convolve(a,a)) / (sum(a)**2)

The algorithm consists of

0.  Load a fallback candidate if present.
1.  Scan richer analytic families (offset power law, mixtures).
2.  Keep the ten best vectors from the scan.
3.  Run several short sub‑gradient refinements from those vectors.
4.  Apply many LP‑refinement steps (state‑of‑the‑art iteration).
5.  Final simulated‑annealing hill‑climb.
All phases stop when the global deadline is reached.
"""

import time
import math
import numpy as np
from scipy.optimize import linprog

# ----------------------------------------------------------------------
# Utility functions
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Aperiodic convolution via real FFT."""
    m = a.size + b.size - 1
    nfft = next_pow2(m)
    A = np.fft.rfft(a, n=nfft)
    B = np.fft.rfft(b, n=nfft)
    C = A * B
    conv = np.fft.irfft(C, n=nfft)[:m]
    return conv


def compute_score(a: np.ndarray) -> float:
    """
    Exact objective for a probability vector (Σa = 1):
        2·n·max(a*a)
    """
    n = a.shape[0]
    conv = convolution_fft(a, a)
    return 2.0 * n * float(conv.max())


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection onto the probability simplex."""
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / v.shape[0])

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, v.shape[0] + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.where(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def softmax(z: np.ndarray) -> np.ndarray:
    """Numerically stable soft‑max."""
    z_max = np.max(z)
    e = np.exp(z - z_max)
    return e / e.sum()


# ----------------------------------------------------------------------
# Analytic families (new and old)
# ----------------------------------------------------------------------


def uniform_family(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def left_heavy_power_family(n: int, beta: float) -> np.ndarray:
    """w_i = (i+0.5)^{-beta}."""
    i = np.arange(n, dtype=np.float64)
    w = np.exp(-beta * np.log(i + 0.5))
    a = w / w.sum()
    return a


def left_heavy_power_offset_family(n: int, beta: float, offset: float) -> np.ndarray:
    """w_i = (i+offset)^{-beta}   (offset > 0)."""
    i = np.arange(n, dtype=np.float64)
    w = np.exp(-beta * np.log(i + offset))
    a = w / w.sum()
    return a


def mixed_power_uniform_family(
    n: int, beta: float, lam: float, offset: float = 0.5
) -> np.ndarray:
    """lam·left‑heavy + (1‑lam)·uniform."""
    pw = left_heavy_power_offset_family(n, beta, offset)
    uni = uniform_family(n)
    a = lam * pw + (1.0 - lam) * uni
    return a / a.sum()


def random_monotone_start(n: int) -> np.ndarray:
    r = np.random.rand(n) ** np.random.uniform(0.5, 5.0)
    r.sort()
    a = r[::-1]               # decreasing
    a /= a.sum()
    return a


def random_u_start(n: int) -> np.ndarray:
    """U‑shaped exponential (edges heavy)."""
    gamma = np.random.uniform(0.0, 6.0)
    i = np.arange(n, dtype=np.float64)
    dist = np.minimum(i, n - 1 - i)
    w = np.exp(-gamma * dist)
    a = w / w.sum()
    return a


# ----------------------------------------------------------------------
# Sub‑gradient optimisation
# ----------------------------------------------------------------------


def subgrad_optimize(start_a: np.ndarray,
                     deadline: float,
                     max_iters: int = 500_000,
                     step0: float = 0.07) -> tuple[np.ndarray, float]:
    """
    Projected sub‑gradient descent on   2·n·max(a*a)
    with a 1/√t stepsize.
    """
    n = start_a.shape[0]
    a = start_a.astype(np.float64).copy()
    best_a = a.copy()
    best_score = compute_score(a)

    for t in range(1, max_iters + 1):
        if time.time() >= deadline:
            break

        conv = convolution_fft(a, a)
        k = int(np.argmax(conv))

        # sub‑gradient of the max‑convolution at shift k
        idx = np.arange(n)
        j = k - idx
        mask = (j >= 0) & (j < n)
        g = np.zeros_like(a)
        g[mask] = 2.0 * a[j[mask]]
        # diagonal terms (i = j) appear only once
        diag = (j == idx) & mask
        if np.any(diag):
            g[diag] = a[diag]

        step = step0 / np.sqrt(t)
        a = project_to_simplex(a - step * g)

        if t % 5000 == 0:
            sc = compute_score(a)
            if sc < best_score:
                best_score = sc
                best_a = a.copy()
                print(f"[{time.time():.1f}] sub‑grad improved to {best_score:.6f}")

    # final check
    sc = compute_score(a)
    if sc < best_score:
        best_score = sc
        best_a = a.copy()
    return best_a, best_score


# ----------------------------------------------------------------------
# Linear‑programming refinement (state‑of‑the‑art iteration)
# ----------------------------------------------------------------------


def lp_refine_step(f: np.ndarray, M: float, mix_weight: float = 0.2) -> np.ndarray:
    """
    Solve   max Σ g_j   s.t.   (f * g)_k ≤ M  for all shifts k,
    g ≥ 0, then mix f←(1‑t)f + t·g_norm   (t = mix_weight).
    """
    n = f.shape[0]
    m = 2 * n - 1

    # Build convolution matrix A (A @ g = f * g)
    A = np.zeros((m, n), dtype=np.float64)
    for i in range(n):
        A[i:i + n, i] = f[i]

    c = -np.ones(n, dtype=np.float64)            # maximise Σ g
    bounds = [(0.0, None) for _ in range(n)]
    b = np.full(m, M, dtype=np.float64)

    res = linprog(c, A_ub=A, b_ub=b, bounds=bounds,
                  method='highs',
                  options={'presolve': True, 'time_limit': 15})

    if not res.success:
        return f

    g = np.maximum(res.x, 0.0)
    sum_g = g.sum()
    if sum_g <= 0.0:
        return f

    g_norm = g / sum_g
    f_new = (1.0 - mix_weight) * f + mix_weight * g_norm
    f_new = project_to_simplex(f_new)
    return f_new


# ----------------------------------------------------------------------
# Simulated‑annealing hill‑climb
# ----------------------------------------------------------------------


def hill_climb_sa(start_a: np.ndarray,
                  deadline: float,
                  max_steps: int = 2_000_000) -> tuple[np.ndarray, float]:
    """
    Simple SA on the simplex.  Swaps or mass transfers are proposed,
    and the move is accepted with a temperature‑dependent probability.
    """
    n = start_a.size
    cur_a = start_a.copy()
    cur_score = compute_score(cur_a)

    best_a = cur_a.copy()
    best_score = cur_score

    T0 = best_score * 1e-4
    T = T0
    step_scale = 0.15

    for step in range(1, max_steps + 1):
        if time.time() >= deadline:
            break

        cand = cur_a.copy()
        if np.random.rand() < 0.5:          # swap two entries
            i, j = np.random.randint(n, size=2)
            cand[i], cand[j] = cand[j], cand[i]
        else:                               # transfer a fraction of mass
            i, j = np.random.randint(n, size=2)
            if i != j:
                delta = step_scale * cur_a[i]
                cand[i] -= delta
                cand[j] += delta
                if cand[i] < 0.0:
                    cand[i] = 0.0

        cand = project_to_simplex(cand)
        cand_score = compute_score(cand)

        delta = cand_score - cur_score
        if delta < 0.0 or np.random.rand() < np.exp(-delta / (T + 1e-18)):
            cur_a = cand
            cur_score = cand_score
            if cur_score < best_score:
                best_score = cur_score
                best_a = cur_a.copy()
                step_scale = min(step_scale * 1.07, 0.5)
        else:
            step_scale *= 0.97

        # linear cooling
        T = T0 * (1.0 - step / max_steps)

    return best_a, best_score
```
