    best_a = start_a.copy()

    for it in range(1, max_iters+1):
        if time.time() >= deadline:
            break
        # Compute a via softmax
        max_u = np.max(u)
        a = np.exp(u - max_u)
        a = a / a.sum()
        # smoothing parameter
        t = it / max_iters
        beta = beta_start * (beta_end / beta_start) ** t
        # gradient
        if beta < beta_end/10:
            # use smooth
            conv = convolution_fft(a, a)
            maxc = conv.max()
            w = np.exp(beta * (conv - maxc))
            w = w / w.sum()

            grad_a = 2.0 * tmp[n-1:n-1+n]

"""
Propose a step‑function (non‑negative vector) that minimizes

    2·n·max(conv(a,a)) / (sum(a)**2)

The objective is scale‑invariant, so we keep the vector normalised
to sum = 1 and minimise 2·n·max(conv).  The optimiser works in three
stages:

1. **Quick family scan** – a few exponential / power‑law candidates.
2. **Annealed Adam optimisation** – minimise a log‑sum‑exp (smooth‑max)
   approximation of the max‑convolution, gradually increasing the
   smoothing parameter β until the true max is reached.  The optimiser
   works on the soft‑max parametrisation `a = softmax(u)`, i.e. the
   update is performed on the unconstrained log‑weights `u` using an
   Adam optimiser (mirrored gradient on the simplex).
3. **Targeted hill‑climb** – a greedy polishing that repeatedly moves a
   small packet of mass away from the index that contributes most to the
   current maximal convolution entry; only improving moves are kept.

The whole routine respects a 1000 s wall‑clock budget, chooses a
reasonable sequence length (larger when time permits) and keeps the
best vector seen.  The final result is returned as a plain Python list.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Basic linear‑algebra utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """
    Full aperiodic convolution using a real FFT (zero‑padded).
    Returns an array of length len(a)+len(b)-1.
    """
    m = a.size + b.size - 1
    nfft = next_pow2(m)
    A = np.fft.rfft(a, n=nfft)
    B = np.fft.rfft(b, n=nfft)
    C = A * B
    return np.fft.irfft(C, n=nfft)[:m]


def compute_score(a: np.ndarray) -> float:
    """
    Exact objective used by `evaluate_sequence` for a probability vector.
    Returns 2·n·max(conv(a,a)) where n = len(a) and sum(a)=1.
    """
    n = a.shape[0]
    conv = convolution_fft(a, a)
    return 2.0 * n * float(conv.max())


# ----------------------------------------------------------------------
# Simple parametric families (used for quick initialisation)
# ----------------------------------------------------------------------


def exponential_shape(n: int, lam: float) -> np.ndarray:
    """a_i ∝ exp(-lam·i), normalised to sum = 1."""
    i = np.arange(n, dtype=float)
    if lam <= 0.0:
        w = np.ones_like(i)
    else:
        w = np.exp(-lam * i)
    w_sum = w.sum()
    if w_sum == 0.0:
        w = np.ones_like(i)
    return w / w_sum


def power_shape(n: int, p: float) -> np.ndarray:
    """a_i ∝ (i+1)^(-p), normalised to sum = 1."""
    i = np.arange(1, n + 1, dtype=float)
    if p <= 0.0:
        w = np.ones_like(i)
    else:
        w = i ** (-p)
    w_sum = w.sum()
    if w_sum == 0.0:
        w = np.ones_like(i)
    return w / w_sum


def beta_shape(n: int, a: float, b: float) -> np.ndarray:
    """
    Discrete Beta distribution on {0,…,n‑1}:
        a_i ∝ ((i+0.5)/n)^(a‑1) * (1‑(i+0.5)/n)^(b‑1)
    The offset 0.5 prevents division‑by‑zero for a,b<1.
    """
    x = (np.arange(n, dtype=float) + 0.5) / n
    w = np.power(x, a - 1.0) * np.power(1.0 - x, b - 1.0)
    w = np.where(np.isnan(w), 0.0, w)
    w = np.maximum(w, 0.0)
    if w.sum() == 0.0:
        w = np.ones_like(w)
    return w / w.sum()


def random_dirichlet(alpha: float, n: int) -> np.ndarray:
    """Draw a random point on the simplex using Gamma(alpha,1)."""
    if alpha <= 0.0:
        # fall back to uniform
        x = np.random.rand(n)
    else:
        x = np.random.gamma(alpha, 1.0, size=n)
    s = x.sum()
    if s == 0.0:
        x = np.ones(n)
        s = n
    return x / s

def grad_smooth_max(a: np.ndarray, beta: float) -> np.ndarray:
    """
    Gradient of the log‑sum‑exp (smooth max) of the convolution
    with respect to the probability vector `a`.
    """
    n = a.shape[0]
    conv = convolution_fft(a, a)
    maxc = conv.max()
    # stable soft‑max over convolution entries
    expw = np.exp(beta * (conv - maxc))
    w = expw / expw.sum()
    a_rev = a[::-1]
    tmp = convolution_fft(a_rev, w)            # length = 3n‑2
    grad = 2.0 * tmp[n - 1:n - 1 + n]
    return grad


def grad_max_subgrad(a: np.ndarray) -> np.ndarray:
    """Sub‑gradient of the true max‑convolution (active index only)."""
    n = a.shape[0]
    conv = convolution_fft(a, a)
    k_max = int(conv.argmax())
    g = np.zeros_like(a)

    i_start = max(0, k_max - (n - 1))
    i_end = min(n - 1, k_max)
    idx = np.arange(i_start, i_end + 1, dtype=int)
    g[idx] = 2.0 * a[k_max - idx]
    return g


def adam_anneal_optimize(start_a: np.ndarray,
                         deadline: float,
                         max_iters: int = 300_000,
                         beta_start: float = 0.5,
                         beta_end: float = 5000.0,
                         base_lr: float = 0.6) -> tuple[np.ndarray, float]:
    """
    Adam optimiser on the simplex using the log‑sum‑exp smooth‑max
    (β grows exponentially from `beta_start` to `beta_end`).  The
    parameters are kept as log‑weights `u` so that `a = softmax(u)`.
    """
    n = start_a.shape[0]
    # initialise log‑weights
    u = np.log(start_a + 1e-30)

    m = np.zeros_like(u)
    v = np.zeros_like(u)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-12

    best_a = start_a.copy()
    best_sc = compute_score(best_a)

    for it in range(1, max_iters + 1):
        if time.time() >= deadline:
            break

        # soft‑max to obtain a probability vector
        max_u = np.max(u)
        a = np.exp(u - max_u)
        a_sum = a.sum()
        a = a / a_sum

        # annealing schedule for β
        t = it / max_iters
        beta = beta_start * (beta_end / beta_start) ** t

        # gradient w.r.t. a
        if beta < beta_end * 0.2:
            grad_a = grad_smooth_max(a, beta)
        else:
            # when β is large the smooth max already approximates the true max,
            # but we switch to the exact sub‑gradient to avoid numerical issues.
            grad_a = grad_max_subgrad(a)

        # mirrored gradient on the simplex (∇_u = a ∘ (∇_a − (a·∇_a)))
        dot = float(np.dot(a, grad_a))
        grad_u = a * (grad_a - dot)

        # Adam update on u
        m = beta1 * m + (1.0 - beta1) * grad_u
        v = beta2 * v + (1.0 - beta2) * (grad_u ** 2)

        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        lr = base_lr / np.sqrt(it)          # simple decay
        u -= lr * m_hat / (np.sqrt(v_hat) + eps)

        # occasional exact evaluation for bookkeeping
        if it % 3000 == 0:
            a_check = np.exp(u - np.max(u))
            a_check = a_check / a_check.sum()
            sc = compute_score(a_check)
            if sc < best_sc:
                best_sc = sc
                best_a = a_check.copy()
                print(f"[{time.time():.1f}] iter {it:7d} – new best {best_sc:.6f}")

    # final evaluation
    a_final = np.exp(u - np.max(u))
    a_final = a_final / a_final.sum()
    sc_final = compute_score(a_final)
    if sc_final < best_sc:
        best_sc = sc_final
        best_a = a_final.copy()
        print(f"[{time.time():.1f}] final   – best {best_sc:.6f}")

    return best_a, best_sc


def targeted_hill_climb(start_a: np.ndarray,
                        deadline: float,
                        max_steps: int = 300_000) -> tuple[np.ndarray, float]:
    """
    Greedy local search: repeatedly take a small packet of mass from the
    index that contributes most to the current maximal convolution entry
    and move it to a random destination.  Only improving moves are kept.
    """
    a = start_a.copy()
    n = a.size
    best_a = a.copy()
    best_sc = compute_score(best_a)

    step_scale = 0.10
    steps = 0

    while steps < max_steps and time.time() < deadline:
        conv = convolution_fft(best_a, best_a)
        k_max = int(conv.argmax())

        # list of indices that actually appear in the active sum
        i_start = max(0, k_max - (n - 1))
        i_end = min(n - 1, k_max)
        idx = np.arange(i_start, i_end + 1, dtype=int)

        # contribution of each participating term a_i * a_{k_max-i}
        contrib = best_a[idx] * best_a[k_max - idx]
        i_max = idx[np.argmax(contrib)]

        # pick a random destination index (different from i_max)
        j = np.random.randint(n)
        if j == i_max:
            continue

        delta = step_scale * best_a[i_max]
        if delta <= 0.0:
            break

        cand = best_a.copy()
        cand[i_max] -= delta
        cand[j] += delta

        # re‑project to the simplex (numerical safety)
        cand = np.maximum(cand, 0.0)
        s = cand.sum()
        if s == 0.0:
            continue
        cand = cand / s

        sc = compute_score(cand)
        if sc < best_sc:
            best_sc = sc
            best_a = cand.copy()
            step_scale = min(step_scale * 1.07, 0.5)   # grow a little
        else:
            step_scale *= 0.96                         # shrink on failure

        steps += 1

    return best_a, best_sc

def propose_candidate(seed: int = 12345, budget_s: int = 1000, **kwargs):
    """Search for a step‑function that minimises `evaluate_sequence`."""
    np.random.seed(seed)
    start_time = time.time()
    # leave a small safety margin for the harness
    deadline = start_time + budget_s - 2.0

    # -------------------------------------------------
    # Choose a reasonable length (larger when time permits)
    # -------------------------------------------------
    remaining = deadline - start_time
    if remaining > 800:
        n = 32768
    elif remaining > 500:
        n = 16384
    elif remaining > 250:
        n = 8192
    else:
        n = 4096
    print(f"[{time.time():.1f}] chosen length = {n}")

    best_seq = None
    best_score = float("inf")

    # -------------------------------------------------
    # Optional fallback from a previously discovered vector
    # -------------------------------------------------
    try:
        fallback = np.array(height_sequence_1, dtype=float)  # type: ignore
        # make it a probability vector of the chosen length
        fallback = np.maximum(fallback, 0.0)
        if fallback.sum() > 0.0:
            fallback = fallback / fallback.sum()
            if fallback.size != n:
                if fallback.size > n:
                    fallback = fallback[:n]
                else:
                    fallback = np.concatenate([fallback, np.zeros(n - fallback.size)])
                fallback = fallback / fallback.sum()
            fb = evaluate_sequence(fallback.tolist())
            print(f"[{time.time():.1f}] fallback score = {fb:.6f}")
            if fb < best_score:
                best_score = fb
                best_seq = fallback.tolist()
    except Exception:
        pass

    # -------------------------------------------------
    # 1. Quick scan of a few parametrised families
    # -------------------------------------------------
    scan_deadline = deadline - 150.0
    if time.time() < scan_deadline:
        # exponential shapes
        lam_grid = np.concatenate((np.array([0.0]),
                                   np.logspace(-5, -2, 100)))
        for lam in lam_grid:
            if time.time() >= scan_deadline:
                break
            a = exponential_shape(n, lam)
            sc = compute_score(a)
            if sc < best_score:
                best_score, best_seq = sc, a.tolist()
                print(f"[{time.time():.1f}] exp lam={lam:.2e} → {best_score:.6f}")

        # power‑law shapes
        p_grid = np.linspace(0.0, 6.0, 100)
        for p in p_grid:
            if time.time() >= scan_deadline:
                break
            a = power_shape(n, p)
            sc = compute_score(a)
            if sc < best_score:
                best_score, best_seq = sc, a.tolist()
                print(f"[{time.time():.1f}] power p={p:.2f} → {best_score:.6f}")

        # a modest beta family scan
        beta_deadline = deadline - 120.0
        a_grid = np.concatenate((np.logspace(-0.7, -0.1, 8),
                                 np.logspace(-0.1, 0.3, 8),
                                 np.linspace(0.5, 3.0, 8)))
        for a_par in a_grid:
            for b_par in a_grid:
                if time.time() >= beta_deadline:
                    break
                a = beta_shape(n, a_par, b_par)
                sc = compute_score(a)
                if sc < best_score:
                    best_score, best_seq = sc, a.tolist()
                    print(f"[{time.time():.1f}] beta a={a_par:.2f} b={b_par:.2f} → {best_score:.6f}")

    # -------------------------------------------------
    # 2. Random Dirichlet starts (cheap diversification)
    # -------------------------------------------------
    rand_deadline = deadline - 100.0
    for _ in range(4):
        if time.time() >= rand_deadline:
            break
        a_rand = random_dirichlet(alpha=0.5, n=n)
        sc = compute_score(a_rand)
        if sc < best_score:
            best_score, best_seq = sc, a_rand.tolist()
            print(f"[{time.time():.1f}] random Dirichlet → {best_score:.6f}")

    # -------------------------------------------------
    # 3. Annealed Adam optimisation on the best current vector
    # -------------------------------------------------
    if best_seq is not None and time.time() < deadline - 30.0:
        print(f"[{time.time():.1f}] starting annealed Adam optimisation")
        # Allocate the bulk of the remaining time to this phase
        opt_deadline = deadline - 20.0
        # decide a safe iteration budget (depends on n)
        est_iters = int(min(300_000, (opt_deadline - time.time()) * (8000 // n)))
        # (8000//n) is a crude estimate of iterations per second for given n
        a_opt, sc_opt = adam_anneal_optimize(
            np.array(best_seq, dtype=float),
            opt_deadline,
            max_iters=est_iters,
            beta_start=0.5,
            beta_end=6000.0,
            base_lr=0.5,
        )
        if sc_opt < best_score:
            best_score, best_seq = sc_opt, a_opt.tolist()
            print(f"[{time.time():.1f}] after Adam optimisation best = {best_score:.6f}")

    # -------------------------------------------------
    # 4. Targeted hill‑climb polishing
    # -------------------------------------------------
    if best_seq is not None and time.time() < deadline - 5.0:
        print(f"[{time.time():.1f}] final hill‑climb polishing")
        a_final, sc_final = targeted_hill_climb(
            np.array(best_seq, dtype=float),
            deadline,
            max_steps=500_000,
        )
        if sc_final < best_score:
            best_score, best_seq = sc_final, a_final.tolist()
            print(f"[{time.time():.1f}] hill‑climb improved best = {best_score:.6f}")

    # -------------------------------------------------
    # Safeguard: uniform vector if everything failed
    # -------------------------------------------------
    if best_seq is None:
        n_small = 512
        best_seq = [1.0 / n_small] * n_small
        best_score = evaluate_sequence(best_seq)

    # -------------------------------------------------
    # Final official evaluation (the harness will also call it)
    # -------------------------------------------------
    final_score = evaluate_sequence(best_seq)
    print(f"\n[{time.time():.1f}] finished – final reported score = {final_score:.6f}")

    # Return a plain Python list of floats.
    return best_seq
