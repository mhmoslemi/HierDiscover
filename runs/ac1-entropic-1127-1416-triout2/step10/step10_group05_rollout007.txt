def adaptive_kl_descent(start_a, deadline, max_iters=200_000, eta0=0.5):
    a = np.maximum(start_a.astype(float), 1e-15)
    a /= a.sum()
    n = a.shape[0]
    best_a = a.copy()
    best_sc = compute_score(a)
    current_sc = best_sc
    eta = eta0
    # optional momentum
    m_vec = np.zeros_like(a)
    beta_mom = 0.9

    for it in range(1, max_iters+1):
        if time.time() >= deadline:
            break
        conv = convolution_fft(a, a)
        k_max = int(conv.argmax())
        # subgradient
        g = np.zeros_like(a)
        i_start = max(0, k_max-(n-1))
        i_end = min(n-1, k_max)
        idx = np.arange(i_start, i_end+1)
        g[idx] = 2.0 * a[k_max - idx]

        # momentum
        m_vec = beta_mom * m_vec + (1 - beta_mom) * g

        # propose update
        a_candidate = np.maximum(a * np.exp(-eta * m_vec), 1e-15)
        a_candidate /= a_candidate.sum()
        cand_sc = compute_score(a_candidate)
        if cand_sc < current_sc:
            a = a_candidate
            current_sc = cand_sc
            # increase step size slightly
            eta = min(eta * 1.05, 5.0)
        else:
            # reject: keep a unchanged
            eta = eta * 0.5
            if eta < 1e-8:
                eta = 1e-8

        # track best
        if current_sc < best_sc:
            best_sc = current_sc
            best_a = a.copy()
            # optionally increase step size more
        # occasionally print progress
        if it % 5000 == 0:
            # update best if current better?
            if current_sc < best_sc:
                best_sc = current_sc
                best_a = a.copy()
            # Print with timestamp maybe
            # Not mandatory
            pass
    # final compare
    if current_sc < best_sc:
        best_sc = current_sc
        best_a = a.copy()
    return best_a, best_sc

"""
Step‑function optimiser (≈ 1 000 s budget)

The optimiser works in three stages:

1️⃣  **Coarse families** – a quick scan of a few parametrised families
    (exponential, power‑law, Beta / arcsine).  The best candidate is kept.

2️⃣  **Many‑restart KL‑mirror descent** – a large number of random Dirichlet
    starts are refined with an exponentiated‑gradient (KL‑mirror) update.
    The update uses the true sub‑gradient of the max‑convolution, a momentum
    term and an adaptive step size.  The best vector from all starts is
    retained.

3️⃣  **Upsample & polish** – the best vector is linearly interpolated to a
    larger dimension (≈ 2×) and refined again with the same mirror descent.
    Finally a targeted hill‑climb moves a small packet of mass away from the
    index that contributes most to the current maximal convolution entry.

All vectors are kept on the probability simplex (non‑negative, sum = 1); for
such vectors the objective `evaluate_sequence` reduces to

    2·n·max(conv(a,a))

which is computed by `compute_score`.  The algorithm is deliberately
stateless (no closures) and prints progress timestamps for the harness.
"""

import time
import numpy as np


# ----------------------------------------------------------------------
# Basic linear‑algebra utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full aperiodic convolution using a real FFT (zero‑padded)."""
    m = a.size + b.size - 1
    nfft = next_pow2(m)
    A = np.fft.rfft(a, n=nfft)
    B = np.fft.rfft(b, n=nfft)
    C = A * B
    return np.fft.irfft(C, n=nfft)[:m]


def compute_score(a: np.ndarray) -> float:
    """
    Exact objective used by `evaluate_sequence` for a vector that already
    satisfies the simplex constraint (sum = 1).  Returns
        2·n·max(a*a)
    where the convolution is full (length 2·n‑1).
    """
    n = a.shape[0]
    conv = convolution_fft(a, a)
    return 2.0 * n * float(conv.max())


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / v.shape[0])

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, v.shape[0] + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.where(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


# ----------------------------------------------------------------------
# Parametrised families of candidate vectors
# ----------------------------------------------------------------------


def exponential_shape(n: int, lam: float) -> np.ndarray:
    """a_i ∝ exp(-lam·i), normalised to sum = 1."""
    i = np.arange(n, dtype=float)
    w = np.exp(-lam * i) if lam > 0.0 else np.ones_like(i)
    return w / w.sum()


def power_shape(n: int, p: float) -> np.ndarray:
    """a_i ∝ (i+1)^(-p), normalised to sum = 1."""
    i = np.arange(1, n + 1, dtype=float)
    w = i ** (-p) if p > 0.0 else np.ones_like(i)
    return w / w.sum()


def beta_shape(n: int, a: float, b: float) -> np.ndarray:
    """
    Discrete Beta distribution on {0,…,n‑1}:
        a_i ∝ ( (i+0.5)/n )^(a‑1) * (1‑(i+0.5)/n)^(b‑1)
    The offset 0.5 prevents division‑by‑zero for a,b<1.
    """
    x = (np.arange(n, dtype=float) + 0.5) / n
    w = np.power(x, a - 1.0) * np.power(1.0 - x, b - 1.0)
    w[np.isnan(w)] = 0.0
    w = np.maximum(w, 0.0)
    if w.sum() == 0.0:
        w = np.ones_like(w)
    return w / w.sum()


# ----------------------------------------------------------------------
# Grid searches for simple families
# ----------------------------------------------------------------------


def scan_exponential_and_power(n: int, deadline: float):
    """Exhaustive scan of exponential and power‑law families."""
    best_a, best_sc = None, float("inf")

    # exponential decay – cover very flat to rather steep shapes
    lam_grid = np.concatenate(
        (np.array([0.0]), np.logspace(-5, 0.0, 300, base=10.0))
    )
    for lam in lam_grid:
        if time.time() >= deadline:
            break
        a = exponential_shape(n, lam)
        sc = compute_score(a)
        if sc < best_sc:
            best_sc, best_a = sc, a.copy()

    # power law – from flat (p≈0) to fairly steep (p≈8)
    p_grid = np.linspace(0.0, 8.0, 300)
    for p in p_grid:
        if time.time() >= deadline:
            break
        a = power_shape(n, p)
        sc = compute_score(a)
        if sc < best_sc:
            best_sc, best_a = sc, a.copy()

    return best_a, best_sc


def scan_beta_family(n: int, deadline: float):
    """Scan a 2‑parameter Beta family (including the arcsine case)."""
    best_a, best_sc = None, float("inf")
    # grids concentrate around the interesting region (0.01 … 3.0)
    a_grid = np.concatenate(
        (
            np.logspace(-2, -0.1, 12),
            np.logspace(-0.1, 0.2, 12),
            np.linspace(0.3, 3.0, 12),
        )
    )
    b_grid = a_grid.copy()
    for a in a_grid:
        for b in b_grid:
            if time.time() >= deadline:
                return best_a, best_sc
            a_vec = beta_shape(n, a, b)
            sc = compute_score(a_vec)
            if sc < best_sc:
                best_sc, best_a = sc, a_vec.copy()
    return best_a, best_sc


# ----------------------------------------------------------------------
# Adaptive KL‑mirror descent (sub‑gradient of the true max)
# ----------------------------------------------------------------------


def adaptive_kl_descent(
    start_a: np.ndarray,
    deadline: float,
    max_iters: int = 200_000,
    eta0: float = 0.5,
    beta_mom: float = 0.9,
    record_every: int = 500,
) -> tuple[np.ndarray, float]:
    """
    Exponentiated‑gradient (KL‑mirror) descent on the exact objective.
    A momentum term (beta_mom) and an adaptive step size (η decays as
    1/√t) are used.  The routine returns the best vector seen.
    """
    a = np.maximum(start_a.astype(float), 1e-15)
    a /= a.sum()
    n = a.shape[0]

    best_a = a.copy()
    best_sc = compute_score(a)
    current_sc = best_sc

    # momentum accumulator
    m_vec = np.zeros_like(a)

    for it in range(1, max_iters + 1):
        if time.time() >= deadline:
            break

        conv = convolution_fft(a, a)
        k_max = int(conv.argmax())

        # sub‑gradient of the maximal entry
        g = np.zeros_like(a)
        i_start = max(0, k_max - (n - 1))
        i_end = min(n - 1, k_max)
        idx = np.arange(i_start, i_end + 1, dtype=int)
        g[idx] = 2.0 * a[k_max - idx]

        # momentum update
        m_vec = beta_mom * m_vec + (1.0 - beta_mom) * g

        # step size schedule (√t decay)
        eta = eta0 / np.sqrt(it)

        # multiplicative KL‑mirror step
        a = np.maximum(a * np.exp(-eta * m_vec), 1e-15)
        a /= a.sum()

        # periodic exact scoring
        if it % record_every == 0:
            sc = compute_score(a)
            current_sc = sc
            if sc < best_sc:
                best_sc, best_a = sc, a.copy()

    # final check
    sc = compute_score(a)
    if sc < best_sc:
        best_sc, best_a = sc, a.copy()
    return best_a, best_sc


# ----------------------------------------------------------------------
# Targeted hill‑climb (greedy mass transfer)
# ----------------------------------------------------------------------


def targeted_hill_climb(
    start_seq: np.ndarray,
    deadline: float,
    max_steps: int = 500_000,
    step_scale_init: float = 0.10,
) -> tuple[np.ndarray, float]:
    """
    Greedy walk that repeatedly removes a small packet of mass from the
    index that contributes most to the current maximal convolution entry
    and adds it to the currently smallest entry.  Only improving moves are
    kept; the step scale is adapted.
    """
    a = start_seq.copy()
    n = a.size
    best_a = a.copy()
    best_sc = compute_score(best_a)

    step_scale = step_scale_init
    steps = 0

    while steps < max_steps and time.time() < deadline:
        conv = convolution_fft(best_a, best_a)
        k_max = int(conv.argmax())

        i_start = max(0, k_max - (n - 1))
        i_end = min(n - 1, k_max)
        idx = np.arange(i_start, i_end + 1, dtype=int)
        contrib = best_a[idx] * best_a[k_max - idx]

        i_max = idx[np.argmax(contrib)]

        # destination = smallest mass (avoid no‑op)
        j = np.argmin(best_a)
        if j == i_max:
            # fallback to a random different index
            candidates = np.arange(n)
            candidates = candidates[candidates != i_max]
            if candidates.size == 0:
                break
            j = np.random.choice(candidates)

        delta = step_scale * best_a[i_max]
        if delta <= 0.0:
            break

        cand = best_a.copy()
        cand[i_max] -= delta
        cand[j] += delta
        # re‑normalise (tiny numerical drift may appear)
        cand = np.maximum(cand, 0.0)
        cand /= cand.sum()

        sc = compute_score(cand)
        if sc < best_sc:
            best_sc, best_a = sc, cand.copy()
            step_scale = min(step_scale * 1.07, 0.5)
        else:
            step_scale *= 0.96

        steps += 1

    return best_a, best_sc

def upscale_vector(a_small: np.ndarray, n_large: int) -> np.ndarray:
    """
    Linear interpolation from length len(a_small) → n_large,
    followed by simplex normalisation.
    """
    x_small = np.linspace(0.0, 1.0, a_small.size)
    x_large = np.linspace(0.0, 1.0, n_large)
    a_large = np.interp(x_large, x_small, a_small)
    a_large = np.maximum(a_large, 0.0)
    if a_large.sum() == 0.0:
        a_large = np.full_like(a_large, 1.0 / n_large)
    else:
        a_large /= a_large.sum()
    return a_large
