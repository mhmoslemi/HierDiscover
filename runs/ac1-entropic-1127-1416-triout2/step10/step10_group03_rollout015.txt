def lp_mixing_step(a, t, epsilon=1e-8):
    n = len(a)
    conv = convolution_fft(a, a)
    M = conv.max()
    # Build sparse matrix
    i = np.arange(n, dtype=np.int32)
    j = np.arange(n, dtype=np.int32)
    rows = np.repeat(i, n) + np.tile(j, n)    # shape n*n
    cols = np.tile(j, n)
    data = np.repeat(a, n)                   # coefficient a_i for each column j
    # construct csr matrix

    B = np.fft.rfft(b, n=n_fft)

def softmax_gradient_descent(
    a: np.ndarray,
    start_score: float,
    deadline: float,
    beta_start: float = 5.0,
    beta_end: float = 4000.0,
    step_coef: float = 0.0015,

    """
    Projected gradient descent on the smooth surrogate
        Lβ(a)=log Σ exp(β·conv(a,a)).
    The temperature β is annealed linearly; the step size decays with √iteration
    and with β.  The best iterate (according to the true objective) is kept.

        max_conv = conv.max()

def lp_mixing_step(
    a: np.ndarray,
    t: float,
    rows: np.ndarray,
    cols: np.ndarray,
    M: float,
    deadline: float,
) -> tuple[np.ndarray, float, bool]:
    """
    One iteration of the LP‑mixing sub‑routine.
    Returns (a_new, new_M, success_flag).
    """
    n = a.shape[0]
    # Build the constraint matrix A_ub for the current a
    data = np.repeat(a, n)                     # a_i appears n times, column‑wise
    A_ub = sparse.csr_matrix((data, (rows, cols)), shape=(2 * n - 1, n))
    b_ub = np.full(2 * n - 1, M, dtype=np.float64)

    # maximise sum(g)  <=> minimise -sum(g)
    c = -np.ones(n, dtype=np.float64)

    res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=(0, None), method="highs")
    if not res.success:

        return a, M, False

    g = res.x
    # avoid exact zeros (helps later LP solves)
    eps = 1e-12
    g = np.maximum(g, eps)
    sum_g = g.sum()
    g_norm = g / sum_g

    # convex combination
    a_new = (1.0 - t) * a + t * g_norm

    a_new = a_new / a_new.sum()

    # evaluate new bound
    new_M = convolution_fft(a_new, a_new).max()
    return a_new, new_M, True

            best_a = cand

def subgradient_refine(
    a: np.ndarray,
    start_score: float,
    deadline: float,
    max_iter: int = 800_000,
    step0: float = 0.07,
) -> tuple[np.ndarray, float]:
    """
    Projected sub‑gradient descent on the exact L∞‑objective.
    """
    n = a.shape[0]
    cur_a = a.copy()
    best_a = cur_a.copy()
    best_score = start_score
    for t in range(1, max_iter + 1):
        if time.time() >= deadline:
            break
        conv = convolution_fft(cur_a, cur_a)
        k = int(np.argmax(conv))

        # sub‑gradient at the active index k
        idx = np.arange(n)
        j = k - idx
        mask = (j >= 0) & (j < n)
        g = np.zeros_like(cur_a)
        g[mask] = 2.0 * cur_a[j[mask]]

        step = step0 / np.sqrt(t)
        cur_a -= step * g
        cur_a = project_to_simplex(cur_a)

        cur_score = compute_score(cur_a)
        if cur_score < best_score:
            best_score = cur_score
            best_a = cur_a.copy()

        if t % 5000 == 0:
            # tiny random jitter to avoid stagnation
            jitter = np.random.randn(n) * (step0 * 0.004 / np.sqrt(t))
            cur_a += jitter
            cur_a = project_to_simplex(cur_a)

    return best_a, best_score


# ----------------------------------------------------------------------
# Phase 4 – stochastic hill‑climb polishing
# ----------------------------------------------------------------------


def hill_climb(
    a: np.ndarray,
    deadline: float,
    max_steps: int = 1_200_000,
    init_step: float = 0.08,
) -> tuple[np.ndarray, float]:
    """Greedy stochastic moves kept only when they improve the exact score."""
    n = a.shape[0]
    best_a = a.copy()
    best_score = compute_score(best_a)
    step_scale = init_step
    steps = 0
    while steps < max_steps and time.time() < deadline:
        cand = best_a.copy()
        if np.random.rand() < 0.5:
            # swap two entries
            i, j = np.random.randint(n, size=2)
            cand[i], cand[j] = cand[j], cand[i]
        else:
            # transfer a small fraction from i to j
            i, j = np.random.randint(n, size=2)
            if cand[i] > 0.0 and i != j:
                delta = step_scale * cand[i]
                cand[i] -= delta
                cand[j] += delta
                if cand[i] < 0.0:
                    cand[i] = 0.0
        cand /= cand.sum()          # numerical guard
        cur_score = compute_score(cand)

        if cur_score < best_score:
            best_score = cur_score
            best_a = cand
            step_scale = min(step_scale * 1.07, 0.5)
        else:
            step_scale *= 0.96

        steps += 1
    return best_a, best_score

# Main entry point required by the harness

def propose_candidate(seed: int = 12345, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step‑function that minimises ``evaluate_sequence``.
    Returns a plain Python list of floats.
    """
    np.random.seed(seed)
    start = time.time()
    deadline = start + budget_s - 1.0          # a small safety margin
    total_time = deadline - start

    # --------------------------------------------------------------
    # 0. Choose the vector length (more entries give a larger search space)
    # --------------------------------------------------------------
    n = 4096 if total_time > 850.0 else 2048
    print(f"[Info] Using sequence length n = {n}")

    # --------------------------------------------------------------
    # 1. Generate a diverse seed pool and keep the best one
    # --------------------------------------------------------------
    seeds = []

    # deterministic families
    seeds.append(generate_deterministic(n))
    for p in np.linspace(-2.5, 2.5, 9):
        seeds.append(generate_beta_family(n, p))

    # random Dirichlet draws with various concentrations
    for conc in (0.2, 0.5, 1.0, 2.0, 5.0):
        for _ in range(3):
            seeds.append(random_dirichlet(n, conc))

    # sparse spikes – historically a good source of low‑max‑convolution vectors
    for k in (30, 50, 80, 120):
        seeds.append(random_spike(n, k=k))

    # fallback from previous best construction, if it matches the current length
    try:
        fallback = [float(x) for x in height_sequence_1]   # type: ignore
        if len(fallback) == n:
            seeds.append(np.array(fallback, dtype=float) / np.sum(fallback))
    except Exception:
        pass

    best_seq = None
    best_score = float("inf")
    for s in seeds:
        s = np.asarray(s, dtype=float)
        s = project_to_simplex(s)          # enforce feasibility
        sc = compute_score(s)
        if sc < best_score:
            best_score = sc
            best_seq = s.tolist()
            print(f"[Seed] improved to {best_score:.6f}")

    cur_a = np.array(best_seq, dtype=float)
    cur_a = cur_a / cur_a.sum()
    cur_score = compute_score(cur_a)

    # --------------------------------------------------------------
    # 2. Soft‑max gradient descent (major part of the budget)
    # --------------------------------------------------------------
    phase1_deadline = start + total_time * 0.35
    print("\n[Phase 1] Soft‑max surrogate gradient descent")
    cur_a, cur_score = softmax_gradient_descent(
        cur_a, cur_score, phase1_deadline,
        beta_start=5.0, beta_end=4000.0, step_coef=0.0015,
    )
    if cur_score < best_score:
        best_score = cur_score
        best_seq = cur_a.tolist()
        print(f"[Phase 1] new best {best_score:.6f}")

    # --------------------------------------------------------------
    # 3. LP mixing – a few aggressive Frank‑Wolfe style steps
    # --------------------------------------------------------------
    if _SCIPY_OK and time.time() < deadline - 30.0:
        print("\n[Phase 2] LP mixing")
        # Pre‑compute index arrays for the Toeplitz constraint matrix (independent of a)
        i = np.arange(n, dtype=np.int32)
        rows = np.repeat(i, n) + np.tile(i, n)          # shape (n*n,)
        cols = np.tile(i, n)

        lp_deadline = deadline - 30.0
        t = 0.5                # start with a relatively large mixing weight
        while time.time() < lp_deadline:
            conv = convolution_fft(cur_a, cur_a)
            M = conv.max()
            cur_a, new_M, ok = lp_mixing_step(cur_a, t, rows, cols, M)
            if not ok:
                break
            cur_score = compute_score(cur_a)
            if cur_score < best_score:
                best_score = cur_score
                best_seq = cur_a.tolist()
                print(f"[LP] improvement to {best_score:.6f}")
            t = max(0.05, t * 0.85)   # decay the step size

    # --------------------------------------------------------------
    # 4. Sub‑gradient refinement (cheap iterations)
    # --------------------------------------------------------------
    if time.time() < deadline - 20.0:
        print("\n[Phase 3] Sub‑gradient refinement")
        phase3_deadline = deadline - 20.0
        cur_a, cur_score = subgradient_refine(
            cur_a, cur_score, phase3_deadline,
            max_iter=800_000, step0=0.07,
        )
        if cur_score < best_score:
            best_score = cur_score
            best_seq = cur_a.tolist()
            print(f"[Sub‑grad] new best {best_score:.6f}")

    # --------------------------------------------------------------
    # 5. Final hill‑climb polishing
    # --------------------------------------------------------------
    if time.time() < deadline - 1.0:
        print("\n[Phase 4] Hill‑climb polishing")
        cur_a, cur_score = hill_climb(cur_a, deadline, max_steps=1_200_000)
        if cur_score < best_score:
            best_score = cur_score
            best_seq = cur_a.tolist()
            print(f"[Hill‑climb] final best {best_score:.6f}")

    # --------------------------------------------------------------
    # 6. Return the best sequence found (as a plain list)
    # --------------------------------------------------------------
    final_seq = best_seq if best_seq is not None else cur_a.tolist()
    final_val = evaluate_sequence(final_seq)   # official evaluation (includes clipping)
    print(f"\n[Done] Reported score = {final_val:.6f}")

    return final_seq
