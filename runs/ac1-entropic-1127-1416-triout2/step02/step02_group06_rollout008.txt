"""
Propose a low‑score step‑function sequence.

Algorithm
---------

1. **Seed generation** – a diverse collection of simple families
   (uniform, geometric, power‑law, triangular, bimodal, random Dirichlet
   and the optional global `height_sequence_1`) is created and projected
   onto the ℓ₁‑simplex Σa = √(2 n).  The best seed (by the fast FFT based
   max‑convolution score) is kept.

2. **Projected (adaptive) sub‑gradient descent** – starting from the best
   seed we minimise the maximal self‑convolution value.
   The active set (indices whose convolution value is within a tiny
   tolerance of the current maximum) supplies a sub‑gradient.
   A normalised step is taken, the result is clipped to the allowed
   range and projected back onto the simplex.  The step size is
   adapted by a simple back‑tracking line‑search and a Nesterov‑type
   momentum term (β≈0.9).  Occasional random jitter helps to escape
   plateaux.

3. **Simulated‑annealing fine‑tuning** – after the deterministic phase a
   short stochastic phase performs random mass transfers between two
   positions.  Moves are accepted according to a temperature schedule
   (Metropolis criterion).  This frequently discovers slight improvements
   that the sub‑gradient method cannot reach.

4. **Linear‑programming refinement** (if SciPy is available) – the current
   vector `f` is used to solve the LP  

        maximise   Σ gᵢ
        subject to (f * g)ₖ ≤ M   for all k,
                     gᵢ ≥ 0 ,

   where *M* is the current maximal convolution value.
   The solution `g` is rescaled to the target ℓ₁‑norm and mixed with `f`.
   A handful of such updates are performed.

5. **Final check** – the best sequence discovered is evaluated with the
   official ``evaluate_sequence`` function and returned.

All numbers are kept non‑negative, bounded by 1000 and the sum is always
fixed to √(2 n), so the evaluator never rejects the sequence.
The routine respects the 1000 s time budget; if the deadline is reached
the current best solution is returned.
"""

import time
import numpy as np

# --------------------------------------------------------------------------- #
# Simplex projection (Duchi et al. 2008)
def projection_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Project ``v`` onto the simplex {x ≥ 0, Σx = s}."""
    if s < 0:
        raise ValueError("simplex sum must be non‑negative")
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho_arr = np.where(u - cssv / (np.arange(n) + 1.0) > 0)[0]
    if rho_arr.size == 0:
        return np.zeros_like(v)
    rho = rho_arr[-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# --------------------------------------------------------------------------- #
def _fast_maxconv(seq: np.ndarray, fft_len: int) -> float:
    """Return max_{k} (seq * seq)[k] using an FFT of length ``fft_len``."""
    F = np.fft.rfft(seq, n=fft_len)
    conv = np.fft.irfft(F * F, n=fft_len)[: 2 * seq.size - 1]
    return float(conv.max())


# --------------------------------------------------------------------------- #
def _build_toeplitz_matrix(f: np.ndarray):
    """Return a sparse (2n‑1) × n matrix A with A·g = f * g."""
    n = f.shape[0]
    rows = np.empty(n * n, dtype=np.int32)
    cols = np.empty(n * n, dtype=np.int32)
    data = np.empty(n * n, dtype=np.float64)

    ptr = 0
    for i in range(n):
        rows[ptr: ptr + n] = np.arange(i, i + n)         # rows i … i+n‑1
        cols[ptr: ptr + n] = i                          # column i
        data[ptr: ptr + n] = f
        ptr += n

    from scipy.sparse import csr_matrix
    return csr_matrix((data, (rows, cols)), shape=(2 * n - 1, n))


# --------------------------------------------------------------------------- #
def propose_candidate(seed: int = 42, budget_s: float = 1000, **kwargs):
    """
    Search for a non‑negative step‑function (list of heights) that minimises
    ``evaluate_sequence`` under the given time budget.
    """
    np.random.seed(seed)
    start = time.time()
    deadline = start + budget_s - 0.3          # safety margin

    # ------------------------------------------------------------------- #
    # Problem size – a few thousand entries give a good trade‑off.
    n = 2500
    target_sum = np.sqrt(2.0 * n)               # Σa = √(2 n)
    fft_len = 1 << ((2 * n - 1).bit_length())   # length for a power‑of‑2 FFT

    # ------------------------------------------------------------------- #
    # 1️⃣  Seed generation
    seeds = []

    # uniform
    seeds.append(np.full(n, target_sum / n))

    # geometric (mirrored) families
    for lam in np.logspace(-5, -1, 12):
        r = 1.0 - lam
        idx = np.arange(n)
        left = r ** idx
        right = r ** (n - 1 - idx)
        a = left + right
        a = a / a.sum() * target_sum
        seeds.append(a)

    # power‑law families
    for p in np.linspace(0.3, 0.9, 12):
        base = 1.0 / np.power(np.arange(1, n + 1, dtype=np.float64), p)
        a = base / base.sum() * target_sum
        seeds.append(a)

    # triangular (linear ramp) – symmetric
    half = n // 2
    rise = np.arange(1, half + 1, dtype=np.float64)
    if n % 2 == 0:
        tri = np.concatenate([rise, rise[::-1]])
    else:
        tri = np.concatenate([rise, [half + 1], rise[::-1]])
    tri = tri / tri.sum() * target_sum
    seeds.append(tri)

    # bimodal – mass concentrated at the two ends
    m_end = max(1, n // 30)
    a = np.zeros(n, dtype=np.float64)
    a[:m_end] = 1.0
    a[-m_end:] = 1.0
    a = a / a.sum() * target_sum
    seeds.append(a)

    # random Dirichlet‑like (sparse) seeds
    for _ in range(6):
        alpha = np.random.uniform(0.05, 0.3)
        a = np.random.gamma(alpha, 1.0, size=n)
        a = a / a.sum() * target_sum
        seeds.append(a)

    # optional global reference sequence
    try:
        hs = globals()["height_sequence_1"]
        if isinstance(hs, (list, tuple, np.ndarray)):
            hs = np.asarray(hs, dtype=np.float64)
            if hs.shape[0] == n:
                hs = np.clip(hs, 0.0, 1000.0)
                hs = hs * (target_sum / hs.sum())
                seeds.append(hs)
    except Exception:
        pass

    # ------------------------------------------------------------------- #
    # 2️⃣  Scan seeds, keep the best one
    best_seq = None
    best_score = np.inf

    for i, cand in enumerate(seeds):
        if time.time() >= deadline:
            break
        cand = projection_simplex(cand, target_sum)
        sc = _fast_maxconv(cand, fft_len)
        if sc < best_score:
            best_score, best_seq = sc, cand.copy()
            print(f"[{time.time() - start:6.1f}s] seed {i:02d} → score {sc:.6f}")

    # ------------------------------------------------------------------- #
    # 3️⃣  Projected sub‑gradient descent with momentum and back‑tracking
    curr = best_seq.copy()
    best = curr.copy()
    best_score = _fast_maxconv(best, fft_len)

    # momentum buffer (Nesterov‐type)
    velocity = np.zeros_like(curr)

    # initial step (absolute, not a fraction of the vector)
    step = 0.7 * target_sum / n               # ≈ 0.7 · average height
    min_step = 1e-9
    momentum = 0.9
    rel_tol = 1e-4                             # active‑set tolerance
    it = 0
    idx_arr = np.arange(n)

    while time.time() < deadline:
        # ----- convolution (FFT) -------------------------------------------------
        F = np.fft.rfft(curr, n=fft_len)
        conv = np.fft.irfft(F * F, n=fft_len)[: 2 * n - 1]
        max_val = conv.max()
        score = max_val                         # because Σa² = 2 n

        # ----- update global best ------------------------------------------------
        if score < best_score - 1e-12:
            best_score, best = score, curr.copy()
            print(f"[{time.time() - start:6.1f}s] iter {it:7d} → new best {score:.6f}")
            step = max(step * 0.9, min_step)   # tighten step after progress

        # ----- active set -------------------------------------------------------
        tol_abs = max_val * rel_tol
        active_idx = np.where(conv >= max_val - tol_abs)[0]

        # ----- sub‑gradient (sum of the gradients of active constraints) -------
        grad = np.zeros_like(curr)
        for k in active_idx:
            j = k - idx_arr
            mask = (j >= 0) & (j < n)
            grad[mask] += 2.0 * curr[j[mask]]
        gnorm = np.linalg.norm(grad)
        if gnorm == 0.0:
            break

        # ----- Nesterov‑type momentum step --------------------------------------
        velocity = momentum * velocity - step * (grad / gnorm)
        tentative = curr + velocity
        tentative = np.clip(tentative, 0.0, 1000.0)
        curr = projection_simplex(tentative, target_sum)

        # ----- occasional jitter (helps to escape flat regions) -----------------
        if np.random.rand() < 0.015:
            jitter = (np.random.randn(n) *
                      step * target_sum / (30.0 * n))
            curr = projection_simplex(
                np.clip(curr + jitter, 0.0, 1000.0), target_sum)

        it += 1
        # ----- step‑size decay --------------------------------------------------
        if it % 400 == 0:
            step = max(step * 0.85, min_step)

    # ------------------------------------------------------------------- #
    # 4️⃣  Simulated‑annealing fine‑tuning (uses the same fast FFT score)
    remaining = deadline - time.time()
    if remaining > 10.0:               # reserve a few seconds for SA if possible
        sa_temp0 = best_score * 0.05   # starting temperature (≈ 5 % of current value)
        sa_temp = sa_temp0
        sa_seq = best.copy()
        sa_score = best_score
        sa_it = 0
        max_sa_iter = 200_000          # hard cap – will be limited by time anyway

        while time.time() < deadline - 0.2 and sa_it < max_sa_iter:
            i, j = np.random.choice(n, size=2, replace=False)

            # move a random *fraction* of the smaller of the two masses
            max_move = min(sa_seq[i], target_sum * 0.02)  # ≤ 2 % of total mass
            if max_move <= 0:
                continue
            delta = np.random.rand() * max_move

            cand = sa_seq.copy()
            cand[i] -= delta
            cand[j] += delta
            # bounds are far from the 1000‑limit, but keep the clipping for safety
            if cand[i] < 0.0 or cand[j] > 1000.0:
                continue

            cand_score = _fast_maxconv(cand, fft_len)

            # Metropolis acceptance
            if cand_score < sa_score or np.random.rand() < np.exp(-(cand_score - sa_score) / sa_temp):
                sa_seq, sa_score = cand, cand_score
                if cand_score < best_score:
                    best_score, best = cand_score, cand.copy()
                    print(f"[{time.time() - start:6.1f}s] SA iter {sa_it:7d} → new best {cand_score:.6f}")

            # exponential cooling
            sa_temp = max(sa_temp * 0.9999, 1e-6)
            sa_it += 1

    # ------------------------------------------------------------------- #
    # 5️⃣  Linear‑programming refinement (if SciPy is available)
    try:
        from scipy.optimize import linprog
        from scipy.sparse import csr_matrix
    except Exception:
        linprog = None

    if linprog is not None and time.time() < deadline - 5.0:
        f = best.copy()
        for lp_iter in range(12):
            if time.time() >= deadline - 5.0:
                break
            # current maximal convolution value
            F = np.fft.rfft(f, n=fft_len)
            conv_f = np.fft.irfft(F * F, n=fft_len)[: 2 * n - 1]
            M = conv_f.max()

            # build Toeplitz constraint matrix A (shape (2n‑1, n))
            A = _build_toeplitz_matrix(f)
            b = np.full(2 * n - 1, M)

            # maximise Σ g_i  →  minimise -Σ g_i
            res = linprog(
                c=-np.ones(n),
                A_ub=A,
                b_ub=b,
                bounds=[(0, None)] * n,
                method="highs",
                options={"presolve": True, "time_limit": 30},
            )
            if not res.success or res.x is None:
                break

            g = res.x
            sum_g = g.sum()
            if sum_g < 1e-12:
                break

            # rescale and mix
            g = g * (target_sum / sum_g)
            t = 0.15  # modest mixing weight
            f_new = (1.0 - t) * f + t * g
            f_new = projection_simplex(f_new, target_sum)

            # new score
            Fnew = np.fft.rfft(f_new, n=fft_len)
            conv_new = np.fft.irfft(Fnew * Fnew, n=fft_len)[: 2 * n - 1]
            Mnew = conv_new.max()
            score_new = Mnew

            if score_new < best_score - 1e-8:
                best_score, best = score_new, f_new.copy()
                print(f"[{time.time() - start:6.1f}s] LP‑iter {lp_iter} → new best {score_new:.6f}")

            # accept the update if it does not worsen the bound
            if Mnew <= M * 0.999:
                f = f_new
            else:
                break

    # ------------------------------------------------------------------- #
    # 6️⃣  Final evaluation with the official evaluator
    final_score = evaluate_sequence(best.tolist())
    print(f"Search finished. Final score reported by evaluate_sequence: {final_score:.6f}")

    return best.tolist()
