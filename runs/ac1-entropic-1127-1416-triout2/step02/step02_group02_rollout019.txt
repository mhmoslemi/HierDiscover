"""
Propose a candidate step function by minimizing the maximum self‑convolution
using a smooth approximation (soft‑max) of the infinity norm and an
Adam‑style projected optimiser on the probability simplex.
The temperature of the soft‑max is increased during the run (continuation
method) so that early iterations see a smooth objective and later
iterations focus on the true peak.
The routine respects the 1000 s time budget and returns the best vector
seen.
"""

import time
import numpy as np

def simplex_projection(v: np.ndarray) -> np.ndarray:
    """Euclidean projection onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1] - 1
    theta = cssv[cond][-1] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w

def next_pow2(x: int) -> int:
    """Small helper returning the next power of two ≥ x."""
    return 1 << ((x - 1).bit_length())

def full_convolution_fft(x: np.ndarray, y: np.ndarray, size: int) -> np.ndarray:
    """Linear convolution via FFT, result is trimmed to len(x)+len(y)-1."""
    fx = np.fft.rfft(x, size)
    fy = np.fft.rfft(y, size)
    conv = np.fft.irfft(fx * fy, size)
    return conv[: len(x) + len(y) - 1]

    n = a.shape[0]

"""
Propose a candidate step function by minimising the peak of its
self‑convolution.  The infinite‑norm of the convolution is approximated
by a smooth soft‑max (log‑sum‑exp) with a temperature that is raised
during the run (continuation).  The gradient of the smooth objective is
computed analytically (it is a weighted cross‑correlation) and an
Adam optimiser is used.  After each update the iterate is projected
onto the probability simplex (non‑negative entries, Σa=1).  Random
jitter and occasional restarts help escape plateaus.  The routine
stops after at most ``budget_s`` seconds and returns the best sequence
found.
"""

import time

def simplex_projection(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of a vector onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1] - 1  # zero‑based index
    theta = cssv[cond][-1] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def next_pow2(x: int) -> int:
    """Return the smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())

    """
    Linear convolution using FFT, truncated to the exact length
    ``len(x) + len(y) - 1``.
    """
