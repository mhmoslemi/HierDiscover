"""
Propose a low‑score step‑function sequence.

The algorithm consists of three phases:

1. **Broad seeding** – many simple parametric families (geometric,
   power‑law, sinusoidal, double‑exponential, etc.) are sampled.
   Each candidate is projected onto the ℓ₁‑simplex
   Σaᵢ = √(2 n) and scored with a fast FFT‑based convolution.
   The best seed is kept.

2. **Smoothed‑max gradient descent** – we minimise the quantity
   `max(conv(a,a))` while the ℓ₁‑norm is fixed.
   The non‑smooth maximum is replaced by a soft‑max
   `w_i ∝ exp(β·(c_i−c_max))`.  The gradient of the weighted sum
   `∑ w_i·c_i` can be written as a single convolution, so each
   iteration needs only three FFTs.  An Nesterov‑type momentum step
   is taken and the iterate is projected back onto the simplex.
   The temperature β is slowly increased so the optimisation
   gradually focuses on the true maximum.

3. **Local random refinement** – after the gradient phase we perform
   a simple simulated‑annealing hill‑climber that moves a small amount
   of mass between two random positions.  Only moves that improve the
   fast‑score are kept (with a tiny probability of uphill moves
   to escape plateaus).

All operations respect the security checks of ``evaluate_sequence``.
The routine stops when the allotted time (≈ 1000 s) is exhausted and
returns the best sequence discovered.
"""

import time
import numpy as np

# --------------------------------------------------------------------------- #
# Simplex projection (Duchi et al. 2008)
def projection_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Project `v` onto the simplex {x ≥ 0, Σx = s}."""
    if s < 0:
        raise ValueError("Simplex sum must be non‑negative")
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho_arr = np.where(u - cssv / (np.arange(n) + 1.0) > 0)[0]
    if rho_arr.size == 0:
        return np.zeros_like(v)
    rho = rho_arr[-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# --------------------------------------------------------------------------- #
def _fast_score(seq: np.ndarray, L: int) -> float:
    """
    Fast evaluation of the objective
        2·n·max(conv(seq,seq)) / (sum(seq)²)
    using an FFT of length L (pre‑computed power‑of‑two).
    """
    n = seq.size
    F = np.fft.rfft(seq, n=L)
    conv_full = np.fft.irfft(F * F, n=L)
    max_conv = conv_full[: 2 * n - 1].max()
    s = seq.sum()
    if s < 0.01:          # safeguard (should never happen)
        return np.inf
    return float(2.0 * n * max_conv / (s * s))


# --------------------------------------------------------------------------- #
def _max_convolution(seq: np.ndarray, L: int):
    """
    Return the full convolution a*a (length 2n‑1) and its maximum.
    """
    n = seq.size
    F = np.fft.rfft(seq, n=L)
    conv_full = np.fft.irfft(F * F, n=L)[: 2 * n - 1]
    return conv_full, conv_full.max()


# --------------------------------------------------------------------------- #
def propose_candidate(seed: int = 42, budget_s: float = 1000, **kwargs):
    """
    Search for a non‑negative sequence that minimises ``evaluate_sequence``.
    Returns the best sequence (as a Python list) found before the time budget
    expires.
    """
    np.random.seed(seed)
    start = time.time()
    deadline = start + budget_s - 0.3          # small safety margin

    # ------------------------------------------------------------------- #
    # Problem size – a few thousand entries give a good trade‑off.
    n = 2500
    target_sum = np.sqrt(2.0 * n)             # ℓ₁‑norm we keep fixed

    # FFT length: need to accommodate linear convolution of length 3·n‑2.
    L_fft = 1 << ((3 * n - 2).bit_length())   # power of two ≥ 3n‑2 (≈8192)

    # ------------------------------------------------------------------- #
    # 1. Build a diverse collection of seed families.
    seeds = []

    # (a) Geometric families a_i ∝ r^i
    lambdas = np.logspace(-5, -0.5, 50)       # λ = 1‑r
    for lam in lambdas:
        r = 1.0 - lam
        r_pow = r ** np.arange(n)
        # normalisation to the target ℓ₁‑norm
        C = target_sum / r_pow.sum()
        a = C * r_pow
        seeds.append(a)

    # (b) Reversed geometric (mass peaks at the right end)
    for lam in lambdas:
        r = 1.0 - lam
        r_pow = r ** np.arange(n)[::-1]
        C = target_sum / r_pow.sum()
        a = C * r_pow
        seeds.append(a)

    # (c) Power‑law families a_i ∝ (i+1)^‑p
    for p in np.linspace(0.2, 1.2, 40):
        base = 1.0 / np.power(np.arange(1, n + 1, dtype=np.float64), p)
        C = target_sum / base.sum()
        a = C * base
        seeds.append(a)

    # (d) Reversed power‑law (mass at the right)
    for p in np.linspace(0.2, 1.2, 40):
        base = 1.0 / np.power(np.arange(1, n + 1, dtype=np.float64)[::-1], p)
        C = target_sum / base.sum()
        a = C * base
        seeds.append(a)

    # (e) Double‑exponential (peaks at both ends)
    for lam in np.logspace(-5, -0.5, 30):
        r = 1.0 - lam
        dist = np.minimum(np.arange(n), np.arange(n - 1, -1, -1))
        a = C = None
        w = r ** dist
        C = target_sum / w.sum()
        a = C * w
        seeds.append(a)

    # (f) Sinusoidal shape (smooth bump in the middle)
    s = np.sin(np.pi * (np.arange(n) + 0.5) / n)
    C = target_sum / s.sum()
    seeds.append(C * s)

    # (g) Uniform (fallback)
    seeds.append(np.full(n, target_sum / n))

    # (h) Load the global reference sequence if it has the right length.
    try:
        global height_sequence_1
        if isinstance(height_sequence_1, (list, tuple, np.ndarray)):
            hs = np.asarray(height_sequence_1, dtype=np.float64)
            if hs.shape[0] == n:
                hs = np.clip(hs, 0.0, 1000.0)
                hs = hs * (target_sum / hs.sum())
                seeds.append(hs)
    except Exception:
        pass

    # ------------------------------------------------------------------- #
    # 2. Scan seeds with the fast scorer and keep the best few.
    best_seq = None
    best_score = float("inf")
    best_seeds = []                 # keep the top‑3 seeds for later restarts

    for idx, cand in enumerate(seeds):
        if time.time() >= deadline:
            break
        cand = projection_simplex(cand, target_sum)
        sc = _fast_score(cand, L_fft)
        if sc < best_score:
            best_score, best_seq = sc, cand.copy()
            print(f"[{time.time() - start:6.1f}s] seed {idx:03d} → score {sc:.6f}")
        # keep a few good seeds
        best_seeds.append((sc, cand.copy()))
        best_seeds.sort(key=lambda x: x[0])
        if len(best_seeds) > 3:
            best_seeds.pop()

    # ------------------------------------------------------------------- #
    # 3. Smoothed‑max gradient descent (momentum + projection).
    # Start from the overall best seed.
    curr = best_seq.copy()
    best_score = _fast_score(curr, L_fft)
    best = curr.copy()

    # hyper‑parameters
    base_step = target_sum / (5.0 * n)      # initial absolute step size
    momentum = 0.9
    velocity = np.zeros_like(curr)

    beta = 5.0          # initial soft‑max temperature
    beta_factor = 1.001
    beta_max = 150.0

    it = 0
    while time.time() < deadline:
        # ----- convolution (full) -----
        conv, max_conv = _max_convolution(curr, L_fft)

        # ----- bookkeeping -----
        score = max_conv                     # denominator is constant (≈1)
        if score < best_score - 1e-12:
            best_score, best = score, curr.copy()
            print(f"[{time.time() - start:6.1f}s] iter {it:07d} → new best {score:.6f}")

        # ----- soft‑max weights -----
        w = np.exp(beta * (conv - max_conv))
        w_sum = w.sum()
        if w_sum == 0:
            # numerical safeguard – fall back to uniform weighting
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        # ----- gradient (2 * correlation of w with reversed a) -----
        a_rev = curr[::-1]
        # FFTs (length L_fft, which is large enough for linear convolution)
        W = np.fft.rfft(w, n=L_fft)
        A_rev = np.fft.rfft(a_rev, n=L_fft)
        grad_full = np.fft.irfft(W * A_rev, n=L_fft)
        grad = 2.0 * grad_full[n - 1:n - 1 + n]   # slice the linear part

        gnorm = np.linalg.norm(grad)
        if gnorm == 0.0:
            break
        grad /= gnorm

        # ----- Nesterov‑type momentum step -----
        step = base_step / np.sqrt(it + 1.0)
        velocity = momentum * velocity - step * grad
        tentative = curr + velocity
        # enforce box constraints before projection
        tentative = np.clip(tentative, 0.0, 1000.0)
        curr = projection_simplex(tentative, target_sum)

        # ----- occasional random jitter (helps to escape flat regions) -----
        if np.random.rand() < 0.01:
            jitter = (np.random.randn(n) *
                      step * target_sum / (30.0 * n))
            curr = projection_simplex(
                np.clip(curr + jitter, 0.0, 1000.0), target_sum)

        # ----- update temperature -----
        beta = min(beta * beta_factor, beta_max)

        it += 1

        # ---- early stop if nothing improved for a while ----
        if it % 2000 == 0 and time.time() > deadline - 200:
            # let the remaining time be spent on local refinement
            break

    # ------------------------------------------------------------------- #
    # 4. Local random refinement (simulated‑annealing style).
    # We keep the current best as the starting point.
    cur = best.copy()
    cur_score = best_score
    temp_start = 0.02
    temp_end = 0.001
    total_hill_iters = 0

    while time.time() < deadline:
        # linear cooling schedule based on remaining fraction of time
        fraction = (deadline - time.time()) / (deadline - start)
        temp = temp_end + (temp_start - temp_end) * fraction

        # propose moving a small amount of mass from i → j
        i, j = np.random.randint(0, n, size=2)
        if i == j:
            continue
        eps = base_step * np.random.rand()
        eps = min(eps, cur[i])                     # cannot go negative
        cand = cur.copy()
        cand[i] -= eps
        cand[j] += eps
        # clipping is unnecessary (values stay ≤ target_sum ≪ 1000)
        cand = projection_simplex(cand, target_sum)   # re‑project to keep exact sum

        cand_score = _fast_score(cand, L_fft)

        # accept if improvement or with a small uphill probability
        if cand_score < cur_score - 1e-12:
            cur, cur_score = cand, cand_score
            if cand_score < best_score - 1e-12:
                best_score, best = cand_score, cand.copy()
                print(f"[{time.time() - start:6.1f}s] hill‑climb → new best {cand_score:.6f}")
        else:
            # Metropolis acceptance
            prob = np.exp(-(cand_score - cur_score) / max(temp, 1e-12))
            if np.random.rand() < prob:
                cur, cur_score = cand, cand_score

        total_hill_iters += 1
        # occasional diagnostics
        if total_hill_iters % 5000 == 0:
            print(f"[{time.time() - start:6.1f}s] hill‑iter {total_hill_iters:07d} best {best_score:.6f}")

    # ------------------------------------------------------------------- #
    # 5. Final evaluation with the official evaluator (exact, includes clamping).
    final_score = evaluate_sequence(best.tolist())
    print(f"Search finished. Final score reported by evaluate_sequence: {final_score:.6f}")

    return best.tolist()
