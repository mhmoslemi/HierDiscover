def adam_opt(a_init, time_deadline):
    a = a_init.copy()
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    t = 0
    best = a.copy()
    best_obj = compute_objective(a)
    while time.time() < time_deadline:
        # compute conv
        fft = np.fft.rfft(a, n=2*n - 1)
        conv = np.fft.irfft(fft * fft, n=2*n - 1)
        max_val = conv.max()
        # identify active ks
        tol = max_val * 1e-12 + 1e-14
        active = np.where(conv >= max_val - tol)[0]
        # compute gradient
        grad = np.zeros_like(a)
        for k in active:
            low = max(0, k - n + 1)
            high = min(n - 1, k)
            idx = np.arange(low, high + 1, dtype=int)
            grad[idx] += 2 * a[k - idx]
        if active.size > 1:
            grad /= active.size
        # Adam update
        t += 1
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad * grad)
        m_hat = m / (1 - beta1 ** t)
        v_hat = v / (1 - beta2 ** t)
        step_size = 0.01  # may tune
        a = a - step_size * m_hat / (np.sqrt(v_hat) + eps)
        # projection onto simplex
        a = simplex_projection(a)
        # evaluate objective
        cur_obj = compute_objective(a)
        if cur_obj < best_obj:
            best_obj = cur_obj
            best = a.copy()
            print(f"Iter {t}, new best {best_obj:.6f}")
        # early exit if best_obj < target
        if best_obj < 1.503:
            break
    return best, best_obj

def adam_optimize(a0, deadline):
    a = a0.copy()
    n = a.shape[0]
    m = 2 * n - 1
    # Adam params
    beta1, beta2, eps = 0.9, 0.999, 1e-8
    lr = 0.05
    m_t = np.zeros_like(a)
    v_t = np.zeros_like(a)
    best = a.copy()
    best_score = compute_objective(a)
    iteration = 0
    last_improve = 0
    while time.time() < deadline:
        # compute gradient
        grad, max_val, conv, active, _ = subgradient(a)
        # Adam update
        iteration += 1
        m_t = beta1 * m_t + (1 - beta1) * grad
        v_t = beta2 * v_t + (1 - beta2) * (grad * grad)
        m_hat = m_t / (1 - beta1 ** iteration)
        v_hat = v_t / (1 - beta2 ** iteration)
        step = lr * m_hat / (np.sqrt(v_hat) + eps)
        a = a - step
        # project onto simplex
        a = simplex_projection(a)
        # evaluate
        cur_score = compute_objective(a)
        if cur_score < best_score - 1e-12:
            best_score = cur_score
            best = a.copy()
            last_improve = iteration
            print(f"Iter {iteration:6d}: best {best_score:.6f}")
        # occasional random jitter to escape
        if iteration % 5000 == 0:
            a = simplex_projection(a + 0.001 * np.random.randn(n))
        # restart if stagnation
        if iteration - last_improve > 50000:
            # perturb around best
            a = simplex_projection(best + 0.01 * np.random.randn(n))
            last_improve = iteration
        # early stop if below target
        if best_score < 1.503:
            print(f"Target reached: {best_score:.6f} at iteration {iteration}")
            break
    return best, best_score

"""
Propose a candidate step‑function sequence via multi‑restart Adam optimisation.

The objective to minimise is
    f(a) = 2·n·max_k (a * a)[k]   (with Σ a_i = 1, a_i ≥ 0)

which is convex on the probability simplex.  We use a projected Adam
optimiser (first‑order adaptive method) together with a simplex projection
(Duchi‑et‑al.) to stay feasible.  Several random restarts (uniform,
Dirichlet‑sparse, exponential‑decay) are performed within the allotted
time‑budget; the best solution found is returned.

The code is self‑contained, uses only NumPy and the supplied
``evaluate_sequence`` function, and prints progress information.
"""

import time
import numpy as np


def simplex_projection(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the probability simplex:
        { w | w_i ≥ 0 , Σ w_i = 1 }.
    Implementation follows Duchi et al. (2008).
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1] - 1
    theta = cssv[cond][-1] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def compute_objective(a: np.ndarray) -> float:
    """
    Objective for a vector with Σ a = 1.
    Equivalent to evaluate_sequence after the same normalisation:
        2·n·max(conv(a, a))
    """
    n = a.shape[0]
    m = 2 * n - 1
    fft = np.fft.rfft(a, n=m)
    conv = np.fft.irfft(fft * fft, n=m)
    return 2.0 * n * conv.max()


def subgradient(a: np.ndarray):
    """
    Returns a sub‑gradient of the objective at a, together with the
    current convolution, its maximum value and the set of active indices.
    """
    n = a.shape[0]
    m = 2 * n - 1
    fft = np.fft.rfft(a, n=m)
    conv = np.fft.irfft(fft * fft, n=m)
    max_val = conv.max()
    # tolerance to collect all (approximately) maximal entries
    tol = max_val * 1e-12 + 1e-14
    active = np.where(conv >= max_val - tol)[0]

    grad = np.zeros_like(a)
    for k in active:
        low = k - n + 1
        if low < 0:
            low = 0
        high = k
        if high > n - 1:
            high = n - 1
        idx = np.arange(low, high + 1, dtype=int)
        # contribution 2·a_{k‑i}
        grad[idx] += 2.0 * a[k - idx]

    if active.size > 1:
        grad /= float(active.size)

    return grad, max_val, conv, active


def adam_optimize(a0: np.ndarray, deadline: float, target: float = 1.503):
    """
    Adam optimiser with projection onto the simplex.
    Stops early if the target value is reached.
    Returns the best vector seen and its objective value.
    """
    a = a0.copy()
    n = a.shape[0]

    # Adam hyper‑parameters
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    lr = 0.05            # base learning rate

    m_t = np.zeros_like(a)
    v_t = np.zeros_like(a)

    best = a.copy()
    best_score = compute_objective(best)

    t = 0
    last_improve = 0

    while time.time() < deadline:
        grad, max_val, conv, active = subgradient(a)

        t += 1
        # Adam moment updates
        m_t = beta1 * m_t + (1.0 - beta1) * grad
        v_t = beta2 * v_t + (1.0 - beta2) * (grad * grad)

        m_hat = m_t / (1.0 - beta1 ** t)
        v_hat = v_t / (1.0 - beta2 ** t)

        step = lr * m_hat / (np.sqrt(v_hat) + eps)
        a = a - step
        a = simplex_projection(a)

        cur_score = compute_objective(a)
        if cur_score < best_score - 1e-12:
            best_score = cur_score
            best = a.copy()
            last_improve = t
            print(f"Iter {t:6d}: improved global best = {best_score:.6f}")

        # occasional jitter to escape flat regions
        if t % 5000 == 0:
            a = simplex_projection(a + 0.001 * np.random.randn(n))

        # restart around the current best if no improvement for a while
        if t - last_improve > 50000:
            a = simplex_projection(best + 0.01 * np.random.randn(n))
            last_improve = t

        if best_score < target:
            print(f"Target {target:.3f} reached with score {best_score:.6f} at iter {t}")
            break

    return best, best_score


def propose_candidate(budget_s: int = 1000) -> list:
    """
    Search for a low‑score step function.
    The function runs up to ``budget_s`` seconds and returns the best
    sequence found.
    """
    deadline_total = time.time() + budget_s - 5.0   # safety margin

    # ------------------------------------------------------------------
    # 1. Problem size and initial sequence (fallback to uniform if needed)
    # ------------------------------------------------------------------
    try:
        init_seq = np.asarray(height_sequence_1, dtype=float)   # type: ignore[name-defined]
        n = init_seq.shape[0]
    except Exception:
        n = 5146
        init_seq = np.full(n, 1.0 / n)

    # Normalise to Σ a = 1 (probability simplex)
    init_seq = init_seq / init_seq.sum()
    best_global = init_seq.copy()
    best_score = compute_objective(best_global)
    print(f"Initial best score = {best_score:.6f} (n = {n})")

    # ------------------------------------------------------------------
    # 2. Multi‑restart optimisation loop
    # ------------------------------------------------------------------
    restart = 0
    while time.time() < deadline_total:
        time_left = deadline_total - time.time()
        if time_left < 30.0:   # not enough time for another run
            break

        # allocate a chunk of the remaining budget to this run
        run_time = min(80.0, time_left * 0.8)
        run_deadline = time.time() + run_time

        # ----- choose a random initialisation -------------------------
        mode = np.random.choice(["uniform", "dirichlet", "exponential"])
        if mode == "uniform":
            a0 = np.full(n, 1.0 / n)
        elif mode == "dirichlet":
            # small alpha yields sparse, asymmetric starts
            alpha = np.full(n, 0.3)
            a0 = np.random.dirichlet(alpha)
        else:  # exponential decay
            lam = np.random.uniform(0.5, 5.0) / n
            a0 = np.exp(-lam * np.arange(n))
            a0 = a0 / a0.sum()

        # occasionally blend with the current global best
        if np.random.rand() < 0.2:
            a0 = 0.5 * a0 + 0.5 * best_global
            a0 = a0 / a0.sum()

        # ----- run Adam optimisation ----------------------------------
        a_opt, score_opt = adam_optimize(a0, run_deadline)

        # ----- keep the best solution ---------------------------------
        if score_opt < best_score:
            best_score = score_opt
            best_global = a_opt.copy()
            print(f"Restart {restart}: new global best = {best_score:.6f}")

        restart += 1

    print(f"Finished after {restart} restarts. Best score = {best_score:.6f}")
    # --------------------------------------------------------------
    # 3. Return the best sequence as a plain Python list.
    # --------------------------------------------------------------
    return [float(v) for v in best_global.tolist()]
