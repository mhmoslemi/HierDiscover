def propose_candidate(seed=42, budget_s=1000, **kwargs):

    lambdas = np.logspace(-5, -0.5, 30) # tune
    for lam in lambdas:
        r = 1.0 - lam
        r_n = r**n

        C = target_sum * lam / denom
        a = C * (r ** np.arange(n))
        seeds.append(a.astype(np.float64))

    # power law
    for p in np.linspace(0.3, 1.0, 30):
        base = 1.0 / np.power(np.arange(1, n+1, dtype=np.float64), p)
        C = target_sum / base.sum()

"""
Propose a low‑score step‑function sequence.

Algorithm
---------
1. **Seeding** – generate a diverse pool of candidates (geometric,
   power‑law, random Dirichlet and the global `height_sequence_1` if
   available).  All candidates are projected onto the simplex
   Σa_i = √(2 n) (the optimal ℓ₁‑norm) and the fastest FFT based score
   is kept.

2. **Smooth‑max gradient descent** – we minimise the smooth approximation
   of the maximum convolution value
       L(a) = (1/α)·log Σ_k exp(α·(a∗a)_k) .
   Its gradient can be written as a convolution:
       ∇L = 2·(a * softmax_α(a∗a)),
   where the soft‑max weights sum to one.  At each iteration we
   * compute `conv = a∗a`,
   * build the soft‑max weights,
   * obtain the gradient via another FFT‑based convolution,
   * take a step opposite the normalised gradient,
   * clip to [0,1000] and project back onto the simplex.
   The “sharpness’’ α is increased slowly (α≈2→8) so that the
   optimisation first follows a smooth landscape and later focuses on
   the true maximum.

   An adaptive step‑size (increase on improvement, decrease otherwise)
   together with occasional random jitter helps escape plateaus.

3. **Linear‑programming polishing** – a short LP phase (if SciPy is
   present) identical to the one used in the reference code is applied
   to the best vector found, further decreasing the ℓ∞‑bound.

4. **Return** – the best sequence (as a plain list) is returned; a final
   call to ``evaluate_sequence`` prints the official score.
"""

import time
import numpy as np

# --------------------------------------------------------------------------- #
def projection_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Project `v` onto the simplex {x ≥ 0, Σx = s}."""
    if s < 0:
        raise ValueError("Simplex sum must be non‑negative")
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho_arr = np.where(u - cssv / (np.arange(n) + 1.0) > 0)[0]
    if rho_arr.size == 0:
        return np.zeros_like(v)
    rho = rho_arr[-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# --------------------------------------------------------------------------- #
def full_convolve(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Return the full (linear) convolution of two real vectors using FFT."""
    L = len(x) + len(y) - 1
    N = 1 << ((L - 1).bit_length())          # next power of two
    X = np.fft.rfft(x, n=N)
    Y = np.fft.rfft(y, n=N)
    return np.fft.irfft(X * Y, n=N)[:L]


# --------------------------------------------------------------------------- #
def _fast_score(seq: np.ndarray) -> float:
    """Fast evaluation of the score 2·n·max(conv(seq,seq))/(sum(seq)²)."""
    n = seq.size
    conv = full_convolve(seq, seq)
    max_conv = conv.max()
    s = seq.sum()
    return float(2.0 * n * max_conv / (s * s))


# --------------------------------------------------------------------------- #
def _build_toeplitz_matrix(f: np.ndarray):
    """Return A (2n‑1)×n such that A·g = f∗g (full convolution)."""
    n = f.shape[0]
    rows = np.empty(n * n, dtype=np.int32)
    cols = np.empty(n * n, dtype=np.int32)
    data = np.empty(n * n, dtype=np.float64)

    ptr = 0
    for i in range(n):
        rows[ptr: ptr + n] = np.arange(i, i + n)
        cols[ptr: ptr + n] = i
        data[ptr: ptr + n] = f
        ptr += n

    from scipy.sparse import csr_matrix
    return csr_matrix((data, (rows, cols)), shape=(2 * n - 1, n))


# --------------------------------------------------------------------------- #
def propose_candidate(seed: int = 42, budget_s: float = 1000, **kwargs):
    """Search for a non‑negative sequence that minimises ``evaluate_sequence``."""
    np.random.seed(seed)
    start = time.time()
    deadline = start + budget_s - 0.5          # a small safety margin

    # --------------------------------------------------------------- #
    # Problem size – 2500 gives a good trade‑off between quality and speed
    n = 2500
    target_sum = np.sqrt(2.0 * n)              # Σa_i is fixed to this value

    # --------------------------------------------------------------- #
    # 1. Build a pool of seeds
    seeds = []

    # geometric families a_i ∝ r^i
    for lam in np.logspace(-5, -0.5, 30):
        r = 1.0 - lam
        denom = 1.0 - r ** n
        if denom <= 0.0:

        base = 1.0 / np.power(np.arange(1, n + 1, dtype=np.float64), p)
        C = target_sum / base.sum()
        a = C * base
        seeds.append(a)

    # a few random Dirichlet‑like candidates
    for _ in range(5):
        rand_vec = np.random.rand(n)
