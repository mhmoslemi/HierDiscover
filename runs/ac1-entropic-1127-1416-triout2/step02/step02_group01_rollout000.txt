"""
Propose a low‑score step‑function sequence.

The algorithm combines several ideas:
* Multiple problem sizes (2048, 4096, 8192) – larger n tends to lower the
  asymptotic constant.
* A handful of diverse initialisations (arcsine, uniform, Dirichlet,
  Gaussian‑shaped, triangular).
* Projected Adam optimisation over a smooth approximation of the
  max‑convolution (the “soft‑max” with a homotopy on the temperature α).
* After the main optimisation a cheap stochastic hill‑climb and a
  random re‑allocation refinement finish the search.
All work is performed inside the time budget (default 1000 s) and the
result is returned as a plain Python list of non‑negative floats.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Basic utilities
# ----------------------------------------------------------------------


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of a vector onto the probability simplex."""
    n = v.shape[0]
    if np.all(v <= 0):
        return np.full(n, 1.0 / n, dtype=v.dtype)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.where(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def next_pow2(x: int) -> int:
    """Next power of two greater than or equal to x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full convolution via real FFT (fast)."""
    m = len(a) + len(b) - 1
    n_fft = next_pow2(m)
    A = np.fft.rfft(a, n=n_fft)
    B = np.fft.rfft(b, n=n_fft)
    C = A * B
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


def compute_score(a: np.ndarray) -> float:
    """
    Score identical to `evaluate_sequence` when ∑a = 1.
    Returns 2·n·max(a*a) / (sum(a)^2).
    """
    n = a.shape[0]
    conv = convolution_fft(a, a)
    max_val = float(conv.max())
    s = float(a.sum())
    if s < 1e-12:
        return float("inf")
    return 2.0 * n * max_val / (s * s)


# ----------------------------------------------------------------------
# Initialisers
# ----------------------------------------------------------------------


def arcsine_initial(n: int) -> np.ndarray:
    """Arcsine‑shaped start: a_i ∝ 1 / sqrt((i+0.5)(n‑0.5‑i))."""
    i = np.arange(n, dtype=float)
    eps = 0.25
    vals = 1.0 / np.sqrt((i + 0.5) * (n - 0.5 - i) + eps)
    return vals / vals.sum()


def gaussian_initial(n: int, sigma_factor: float = 0.22) -> np.ndarray:
    """Smooth Gaussian‑like bell centred in the middle."""
    i = np.arange(n, dtype=float)
    centre = (n - 1) / 2.0
    sigma = sigma_factor * n
    vals = np.exp(-0.5 * ((i - centre) / sigma) ** 2)
    return vals / vals.sum()


def triangular_initial(n: int) -> np.ndarray:
    """Symmetric triangular shape (ramp up then down)."""
    i = np.arange(n, dtype=float)
    centre = (n - 1) / 2.0
    vals = 1.0 - np.abs(i - centre) / centre
    vals[vals < 0] = 0.0
    return vals / vals.sum()


def uniform_initial(n: int) -> np.ndarray:
    """Flat distribution."""
    return np.full(n, 1.0 / n, dtype=float)


def dirichlet_initial(n: int) -> np.ndarray:
    """Random Dirichlet draw (guaranteed sum = 1)."""
    return np.random.dirichlet(np.ones(n))


# ----------------------------------------------------------------------
# Smoothed‑max gradient
# ----------------------------------------------------------------------


def smoothed_gradient(a: np.ndarray, alpha: float):
    """
    Gradient of the smooth max:
        f_α(a) = (1/α)·log Σ_k exp(α·(a∗a)_k)
    Returns (grad, conv) where conv = a∗a (full length 2n‑1).
    """
    n = a.shape[0]
    conv = convolution_fft(a, a)                 # length 2n‑1

    # soft‑max weights
    maxc = conv.max()
    exp_term = np.exp(alpha * (conv - maxc))
    w = exp_term / exp_term.sum()                # sum(w) = 1

    # convolution of reversed a with the weight vector
    a_rev = a[::-1]
    conv_rev = convolution_fft(a_rev, w)        # length 3n‑2
    start = n - 1
    rev_slice = conv_rev[start:start + n]       # needed part (size n)

    # diagonal contribution w_{2i}
    w_diag = w[::2][:n]

    grad = 2.0 * rev_slice - w_diag * a
    return grad, conv

# ----------------------------------------------------------------------

    eps: float = 1e-8,
    noise_every: int = 400,
    noise_scale: float = 2e-4,

    """
    Projected Adam optimisation of the smoothed objective.
    Returns the best vector encountered (as a NumPy array) and its
    unsmoothed score.
    """
    a = project_to_simplex(init_a.astype(float).copy())
    m = np.zeros_like(a)
    v = np.zeros_like(a)

    best_a = a.copy()
    best_score = compute_score(a)

    for t in range(1, max_iter + 1):
        if time.time() >= time_deadline:
            break

        # linear homotopy for the smoothing parameter α
        alpha = alpha_start + (alpha_end - alpha_start) * min(1.0, t / max_iter)

        grad, _ = smoothed_gradient(a, alpha)

        # Adam moment updates
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)

        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)

        # step size schedule – decreasing like 1/√t
        eta = 0.5 / np.sqrt(t)

        a_new = a - eta * m_hat / (np.sqrt(v_hat) + eps)

        # occasional isotropic jitter to escape shallow plateaus
        if noise_every and (t % noise_every == 0):
            a_new += np.random.randn(a.shape[0]) * noise_scale

        a_new = project_to_simplex(a_new)

        cur_score = compute_score(a_new)
        if cur_score < best_score:
            best_score = cur_score
            best_a = a_new.copy()

        a = a_new

        # occasional progress report (quiet for speed)
        if t % 10000 == 0:
            print(f"    Adam iter {t:,d} – score {cur_score:.6f} – best {best_score:.6f}")

    return best_a, best_score


# ----------------------------------------------------------------------
# Local refinement utilities
# ----------------------------------------------------------------------


def hill_climb(seq, time_deadline: float, max_steps: int = 30_000, init_step: float = 0.1):
    """
    Simple stochastic pair‑wise exchange hill‑climb.
    Moves a small fraction of mass from one index to another if it improves
    the true max‑convolution score.
    """
    a = np.array(seq, dtype=float)
    a = project_to_simplex(a)
    best_a = a.copy()
    best_score = compute_score(best_a)
    n = len(a)
    step = init_step

    while max_steps > 0 and time.time() < time_deadline:
        i, j = np.random.randint(n, size=2)
        if i == j:
            continue
        # move a fraction of the current mass at i to j
        delta = step * best_a[i]
        if delta <= 0:
            continue
        cand = best_a.copy()
        cand[i] -= delta
        cand[j] += delta
        if cand[i] < 0:
            continue
        cand = project_to_simplex(cand)
        cur_score = compute_score(cand)
        if cur_score < best_score:
            best_score = cur_score
            best_a = cand
            step = min(step * 1.07, 0.5)
        else:
            step *= 0.97
        max_steps -= 1
    return best_a.tolist(), best_score


def random_reallocation(seq, time_deadline: float, trials: int = 20_000, subset: int = 7):
    """
    Random re‑allocation of mass inside a small subset of coordinates.
    This often discovers improvements missed by simple pair‑wise moves.
    """
    a = np.array(seq, dtype=float)
    a = project_to_simplex(a)
    n = len(a)
    best_a = a.copy()
    best_score = compute_score(best_a)

    while trials > 0 and time.time() < time_deadline:
        idx = np.random.choice(n, size=subset, replace=False)
        total = best_a[idx].sum()
        if total == 0:
            trials -= 1
            continue
        # propose a new Dirichlet distribution on the selected indices
        new_weights = np.random.dirichlet(np.ones(subset)) * total
        cand = best_a.copy()
        cand[idx] = new_weights
        cand = project_to_simplex(cand)
        cur_score = compute_score(cand)
        if cur_score < best_score:
            best_score = cur_score
            best_a = cand
        trials -= 1
    return best_a.tolist(), best_score


# ----------------------------------------------------------------------
# Helper to decide iteration budget per length
# ----------------------------------------------------------------------


def max_iter_for_n(n: int) -> int:
    """Rough iteration budget as a function of n (larger n -> fewer iters)."""
    if n <= 2048:
        return 250_000
    if n <= 4096:
        return 150_000
    return 80_000

# Main entry point required by the competition harness
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative sequence that yields a low `evaluate_sequence`
    value.  The function respects the time budget and returns a plain
    Python list of floats (the step‑function heights).
    """
    np.random.seed(seed)
    start_time = time.time()
    # keep a small safety margin for final checks
    deadline = start_time + budget_s - 4.0

    best_seq = None
    best_val = float("inf")

    # ------------------------------------------------------------------
    # Optional fallback to a known construction (if present)
    # ------------------------------------------------------------------
    try:
        fallback = [float(x) for x in height_sequence_1]
        fallback_val = evaluate_sequence(fallback)
        print(f"Fallback score = {fallback_val:.6f}")
        if fallback_val < best_val:
            best_val = fallback_val
            best_seq = fallback
    except NameError:
        pass

    # ------------------------------------------------------------------
    # Problem sizes to explore.
    # Larger n reduces the asymptotic constant but costs more per iteration.
    # ------------------------------------------------------------------
    lengths = [2048, 4096, 8192]

    for n in lengths:
        if time.time() >= deadline - 5.0:
            break
        print(f"\n=== Optimising length n = {n} ===")

        # A diverse set of initial guesses
        inits = [
            arcsine_initial(n),
            uniform_initial(n),
            dirichlet_initial(n),
            gaussian_initial(n),
            triangular_initial(n),
        ]

        for idx, init_a in enumerate(inits, start=1):
            if time.time() >= deadline - 5.0:
                break
            print(f"  start {idx}/{len(inits)}")
            max_iter = max_iter_for_n(n)
            seq, val = adam_optimize(
                init_a=init_a,
                time_deadline=deadline,
                max_iter=max_iter,
                alpha_start=2.0,
                alpha_end=500.0,
                noise_every=600,
                noise_scale=1e-4,
            )
            print(f"    Adam best score = {val:.6f}")
            if val < best_val:
                best_val = val
                best_seq = seq.tolist()
                print(f"    *** New global best {best_val:.6f} (n={n}) ***")

    # ------------------------------------------------------------------
    # Final refinement phase (if time permits)
    # ------------------------------------------------------------------
    if best_seq is not None and time.time() < deadline - 2.0:
        print("\nRefining best sequence with hill‑climb")
        refined_seq, refined_val = hill_climb(
            best_seq, time_deadline=deadline, max_steps=40_000, init_step=0.08
        )
        print(f"    hill‑climb score = {refined_val:.6f}")
        if refined_val < best_val:
            best_val = refined_val
            best_seq = refined_seq

        print("Running random‑subset re‑allocation")
        refined_seq, refined_val = random_reallocation(
            best_seq, time_deadline=deadline, trials=30_000, subset=9
        )
        print(f"    re‑allocation score = {refined_val:.6f}")
        if refined_val < best_val:
            best_val = refined_val
            best_seq = refined_seq

    # ------------------------------------------------------------------
    # Safety net – if nothing was found, use a trivial uniform sequence.
    # ------------------------------------------------------------------
    if best_seq is None:
        print("No candidate found – falling back to uniform length‑256 sequence.")
        best_seq = uniform_initial(256).tolist()
        best_val = evaluate_sequence(best_seq)

    # ------------------------------------------------------------------
    # Final verification (the harness will also call evaluate_sequence)
    # ------------------------------------------------------------------
    final_score = evaluate_sequence(best_seq)
    print(f"\nFinished search. Final reported score = {final_score:.6f}")

    # The harness expects a plain list of floats.
    return best_seq
