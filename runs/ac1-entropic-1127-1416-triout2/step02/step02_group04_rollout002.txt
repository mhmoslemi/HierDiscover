"""
Propose a step‑function height sequence that minimizes the objective

    evaluate_sequence(a) = 2·n·max(conv(a,a)) / (sum(a)²)

The sum of the coefficients does not affect the value (the numerator
scales quadratically with the sum, the denominator does the same), so we
work on the probability simplex (non‑negative, sum = 1).  
The objective is the maximum entry of the self‑convolution, a
nondifferentiable function.  We replace the max by a smooth
log‑sum‑exp (soft‑max) with an annealed temperature α, compute its exact
gradient via FFT (the gradient of the soft‑max of a self‑convolution is
`2 * correlation(a, softmax_weights)`) and perform an exponentiated‑gradient
update (multiplicative update on the simplex).  The temperature α starts
small (the objective is smooth) and is increased geometrically – a classic
continuation method that helps escape poor local minima.

The algorithm:
  1. Choose an initial distribution (the best of a few simple analytical
     shapes or the globally supplied `height_sequence_1` if present).
  2. Repeatedly
        – compute the self‑convolution by FFT,
        – form soft‑max weights w_i = exp(α·(conv_i‑max))/∑exp(...),
        – obtain the gradient g = 2·corr(a,w) using another FFT,
        – update in the log‑domain:  log a ← log a − η·g,
        – renormalise a to sum = 1,
        – keep the best solution seen (exact objective via `compute_objective`).
     The step size η follows a diminishing schedule η₀/√t and the
     temperature α is multiplied by a factor each iteration, up to a

def simplex_projection(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of a vector onto the probability simplex."""
    v = np.asarray(v, dtype=float)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1] - 1
    theta = cssv[cond][-1] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def compute_objective(a: np.ndarray) -> float:
    """Exact objective 2·n·max(conv(a,a)) assuming sum(a)=1."""
    n = a.shape[0]
    # linear convolution via FFT (length 2·n‑1)
    fft_a = np.fft.rfft(a, n=2 * n - 1)
    conv = np.fft.irfft(fft_a * fft_a, n=2 * n - 1)
    return 2.0 * n * conv.max()


def beta_sequence(alpha: float, beta: float, n: int) -> np.ndarray:
    """Discretised Beta(α,β) density on n points, normalised to sum = 1."""
    x = (np.arange(n) + 0.5) / n
    logw = (alpha - 1.0) * np.log(x) + (beta - 1.0) * np.log(1.0 - x)
    logw = np.clip(logw, -700.0, 700.0)      # avoid overflow
    w = np.exp(logw)
    if w.sum() == 0 or not np.isfinite(w.sum()):
        w = np.ones_like(w)
    return w / w.sum()


def initial_candidates(n: int) -> np.ndarray:
    """
    Return a good starting distribution for length n.
    Tries a decreasing ramp, uniform, a small beta grid and a few random
    Dirichlet draws, keeping the best according to the exact objective.
    """
    best_a = None
    best_score = np.inf

    # 1. linear decreasing ramp (known good baseline)
    a_ramp = np.arange(n, 0, -1, dtype=float)
    a_ramp = 2.0 * a_ramp / (n * (n + 1))
    a_ramp /= a_ramp.sum()
    score = compute_objective(a_ramp)
    best_a, best_score = a_ramp, score

    # 2. uniform
    a_uni = np.full(n, 1.0 / n)

    if score < best_score:

    alphas = np.linspace(0.3, 4.0, 7)

            score = compute_objective(a_beta)
            if score < best_score:

    for _ in range(8):

        a_dir /= a_dir.sum()
        score = compute_objective(a_dir)
