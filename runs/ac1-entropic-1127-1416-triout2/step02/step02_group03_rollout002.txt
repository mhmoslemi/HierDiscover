def _full_convolution_fft_custom(x, y):
    n = len(x)
    m = len(y)
    size = 1 << ((n+m-1).bit_length())
    X = np.fft.rfft(x, n=size)

"""
Propose a step‑function with a very low evaluation value.

The algorithm consists of two phases:

1. **Projected smoothed‑max gradient descent**  
   – The objective  max(conv(a,a))  is approximated by a smooth
     log‑sum‑exp with a temperature parameter α.  
   – The gradient of this smooth objective can be written as a convolution
     of the soft‑max weights with the reversed current vector, which we
     evaluate efficiently by FFTs.  
   – After each gradient step we project onto the simplex
     {x ≥ 0, Σx = √(2·n)} (Duchi et al. algorithm).  
   – α is increased gradually, the step size decays like 1/√t and occasional
     random jitter helps escape plateaus.

2. **Local random pair‑wise re‑allocation** (fine‑tuning)  
   – With the remaining time we repeatedly move a tiny amount of mass
     from one index to another, accept the move only if the maximal
     convolution value decreases.  This stochastic hill‑climbing
     exploits the cheap FFT evaluation of the objective.

Both phases respect the 1000 s budget and keep the sum of the
coefficients exactly √(2·n); consequently the score returned by
``evaluate_sequence`` equals the maximal convolution value.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities
# ----------------------------------------------------------------------


def _project_simplex(v: np.ndarray, target_sum: float) -> np.ndarray:
    """
    Euclidean projection of a vector onto the simplex
        { w | w_i ≥ 0, Σ w_i = target_sum }.
    Implements the algorithm of Duchi et al. (2008).
    """
    if target_sum < 0:
        raise ValueError("target_sum must be non‑negative")
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - target_sum
    ind = np.arange(1, len(v) + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.zeros_like(v)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def _full_convolution_fft(x: np.ndarray, y: np.ndarray | None = None) -> np.ndarray:
    """
    Linear (full) convolution using a real FFT.
    If ``y`` is omitted the function returns x*x.
    The returned array has length len(x)+len(y)-1 (or 2*len(x)-1).
    """
    if y is None:
        y = x
    n = len(x) + len(y) - 1
    size = 1 << (n - 1).bit_length()          # next power of two
    fx = np.fft.rfft(x, n=size)
    fy = np.fft.rfft(y, n=size)
    conv = np.fft.irfft(fx * fy, n=size)[:n]
    return conv


def _max_conv(a: np.ndarray) -> float:
    """Maximum entry of the full self‑convolution a*a."""
    return _full_convolution_fft(a, a).max()


def _make_candidate(n: int, p: float, offset: float, beta: float) -> np.ndarray:
    """
    Build an un‑normalised raw sequence
        a_i = (i + 1 + offset)^(-p) * (1 + beta * i/(n‑1)),
    clip negatives, and project onto the simplex Σa = √(2·n).
    """
    idx = np.arange(n, dtype=float)
    base = np.power(idx + 1.0 + offset, -p)
    factor = 1.0 + beta * idx / (n - 1)
    raw = base * factor
    raw = np.maximum(raw, 0.0)
    target = np.sqrt(2.0 * n)
    return _project_simplex(raw, target)


# ----------------------------------------------------------------------
# Main search routine
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 123, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative sequence that minimises

        2 * n * max(conv(a, a)) / (sum(a) ** 2)

    using smoothed‑max gradient descent followed by a local random
    pair‑wise refinement.  The function respects the 1000 s budget and
    returns the best sequence found as a plain Python list.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5  # small safety margin

    # ------------------------------------------------------------------
    # 0. Choose a reasonable length.
    # ------------------------------------------------------------------
    n = 12000                     # larger than before, still fast with FFTs
    target_sum = np.sqrt(2.0 * n)

    # ------------------------------------------------------------------
    # 1. Initialise the sequence.
    # ------------------------------------------------------------------
    # Try to reuse a previously found sequence (if it has the right length);
    # otherwise start from a simple power‑law.
    best_seq = None
    try:
        old = globals().get("height_sequence_1")
        if old is not None and len(old) == n:
            best_seq = np.array(old, dtype=float)
            best_seq = _project_simplex(best_seq, target_sum)
    except Exception:
        best_seq = None

    if best_seq is None:
        # power‑law start (p = 0.5, zero offset, no tilt)
        best_seq = _make_candidate(n, p=0.5, offset=0.0, beta=0.0)

    # ------------------------------------------------------------------
    # 2. Projected smoothed‑max gradient descent.
    # ------------------------------------------------------------------
    a = best_seq.copy()
    best_M = _max_conv(a)                # because Σa = √(2n) ⇒ score = M
    print(f"[init] n={n}, initial max‑conv = {best_M:.6f}")

    # hyper‑parameters
    step0 = 0.5                          # base step size
    alpha = 10.0                         # temperature for the soft‑max
    max_alpha = 80.0

    it = 0
    # reserve ~80 % of the time budget for the gradient phase
    grad_deadline = time.time() + 0.80 * (deadline - time.time())

    while time.time() < grad_deadline:
        it += 1

        # ----- evaluate current max‑conv -----
        conv = _full_convolution_fft(a, a)
        M = conv.max()
        if M < best_M:
            best_M = M
            best_seq = a.copy()
            print(f"[gd] iter {it:6d}  new best max‑conv = {best_M:.6f}")

        # ----- smooth approximation of the max -----
        # p_k = softmax(α·conv_k)
        shifted = conv - M                         # ≤ 0
        exp_shift = np.exp(alpha * shifted)        # safe because shifted ≤ 0
        weight_sum = exp_shift.sum()
        p = exp_shift / weight_sum                  # probabilities, length 2n‑1

        # ----- gradient of the smooth objective -----
        # grad = 2 * (p * a_rev)_{slice n‑1 … n‑1+n-1}
        a_rev = a[::-1]
        size = 1 << ((len(p) + len(a_rev) - 1).bit_length())
        P = np.fft.rfft(p, n=size)
        A_rev = np.fft.rfft(a_rev, n=size)
        conv_rev = np.fft.irfft(P * A_rev, n=size)[: len(p) + len(a_rev) - 1]
        grad = 2.0 * conv_rev[n - 1 : n - 1 + n]    # shape (n,)

        # ----- step and projection -----
        step = step0 / np.sqrt(it)
        cand = a - step * grad
        cand = _project_simplex(cand, target_sum)

        # occasional jitter to shake out of plateaus
        if it % 100 == 0 and np.random.rand() < 0.5:
            jitter = (np.random.rand(n) - 0.5) * 0.02 * cand
            cand = cand + jitter
            cand = _project_simplex(cand, target_sum)

        a = cand

        # slowly raise the temperature – we want a sharper approximation later
        if it % 2000 == 0:
            alpha = min(alpha * 1.05, max_alpha)

    # ------------------------------------------------------------------
    # 3. Local random pair‑wise refinement (fine‑tuning).
    # ------------------------------------------------------------------
    # Use the remaining budget for cheap stochastic hill‑climbing.
    remaining = deadline - time.time()
    if remaining > 1.0:
        a = best_seq.copy()
        best_M = _max_conv(a)

        # Pre‑compute a random number generator for speed
        rng = np.random.default_rng()

        while time.time() < deadline:

            i, j = rng.integers(0, n, size=2)
            if i == j:

            max_delta = a[i] * 0.15
            if max_delta <= 0:
                continue
            delta = rng.random() * max_delta

            cand = a.copy()
            cand[i] -= delta
            cand[j] += delta
            if cand[i] < 0:            # sanity check (projection not needed)
                continue

            M_cand = _max_conv(cand)
            if M_cand < best_M - 1e-9:     # strict improvement
                best_M = M_cand
                best_seq = cand.copy()
                a = cand
                print(f"[local] new best max‑conv = {best_M:.6f}")

    print(f"Finished – best max‑conv = {best_M:.6f} after {it} gradient steps")
    # The harness expects a plain Python list.
    return best_seq.tolist()
