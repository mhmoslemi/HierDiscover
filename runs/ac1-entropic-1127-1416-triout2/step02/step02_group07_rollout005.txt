"""
Propose a step‑function by minimising the sup‑norm of its self‑convolution.
The method is a projected (and optionally momentum‑accelerated) gradient
descent on the probability simplex.  At each iteration the gradient is
approximated by averaging the sub‑gradients of the K largest convolution
entries – this smoothes the direction and reduces plateaux that appear
when only the single maximal shift is used.  Small random “jitters’’ and
periodic smoothing of the sequence help escape local minima.
The routine runs for at most ``budget_s’’ seconds (default 1000 s) and
returns the best list of heights found.  If nothing useful is discovered
the global ``height_sequence_1'' (if present) is used as a safe fallback.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper functions (no closures, no lambdas)
# ----------------------------------------------------------------------


def project_onto_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the probability simplex:
        { x | x >= 0, sum(x) = 1 }.
    Returns a new array (the input is not modified).
    """
    n = v.shape[0]
    # sort descending
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all entries are zero – return the uniform distribution
        return np.full_like(v, 1.0 / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def full_convolution(a: np.ndarray) -> np.ndarray:
    """
    Compute the full discrete convolution a * a using an FFT.
    Length of the result is 2*len(a) - 1.
    """
    n = a.shape[0]
    size = 2 * n
    fft_len = 1 << ((size - 1).bit_length())      # next power of two
    fft_a = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(fft_a * fft_a, n=fft_len)
    return conv[: 2 * n - 1]


def aggregate_gradient(a: np.ndarray, idxs: np.ndarray) -> np.ndarray:
    """
    Average the sub‑gradient of the convolution entries whose shifts are
    given in ``idxs''.
    For a shift k the sub‑gradient is
        g_i = a_{k-i}   (if i participates, otherwise 0)
    The result is the arithmetic mean of the individual gradients.
    """
    n = a.shape[0]
    grad = np.zeros_like(a)
    for k in idxs:
        low = max(0, k - n + 1)
        high = min(k, n - 1)
        if low > high:
            continue
        idx_range = np.arange(low, high + 1, dtype=int)
        grad[idx_range] += a[k - idx_range]
    if idxs.size > 0:
        grad /= float(idxs.size)
    return grad


def objective_score(a: np.ndarray) -> float:
    """
    Compute 2 * n * max(conv(a, a)) for a vector that already satisfies
    sum(a) == 1.  This matches the value returned by evaluate_sequence.
    """
    n = a.shape[0]
    conv = full_convolution(a)
    return float(2.0 * n * np.max(conv))


def smooth_sequence(a: np.ndarray) -> np.ndarray:
    """
    Apply a tiny 3‑point smoothing filter and renormalise onto the simplex.
    """
    kernel = np.array([0.1, 0.8, 0.1], dtype=float)
    sm = np.convolve(a, kernel, mode="same")
    sm = np.maximum(sm, 0.0)
    sm /= sm.sum()
    # projection is only a safety net for numerical drift
    return project_onto_simplex(sm)


# ----------------------------------------------------------------------
# Main search routine
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: int = 1000) -> list[float]:
    """
    Search for a non‑negative height sequence that minimises

        2 * n * max(conv(a, a)) / (sum(a) ** 2)

    using a projected (momentum) gradient descent with an aggregated
    sub‑gradient.  The routine respects the given time budget and returns
    the best list of heights found.
    """
    np.random.seed(seed)

    # Keep a small safety margin so that we can print the final result.
    deadline = time.time() + budget_s - 2.0

    # ------------------------------------------------------------------
    # Parameters (feel free to tweak)
    # ------------------------------------------------------------------
    n = 5000                              # length of the step‑function
    max_K = 15                            # number of top convolution entries
    step_initial = 0.7                    # initial step size (will decay)
    beta = 0.9                            # momentum factor (0 → no momentum)
    smooth_interval = 400                # how often to apply smoothing
    jitter_interval = 250                # how often to try a random jitter

    # ------------------------------------------------------------------
    # Initialisation – start from the classic 1/sqrt(i+1) shape
    # ------------------------------------------------------------------
    raw = 1.0 / np.sqrt(np.arange(n, dtype=float) + 1.0)
    a = raw / raw.sum()                   # now sum(a) == 1

    # If a previous construction exists, keep it as an alternative start.
    try:
        prev = np.asarray(height_sequence_1, dtype=float)
        if prev.ndim == 1 and prev.size == n:
            # keep a copy; the optimiser will pick the better one
            a_prev = prev / prev.sum()
        else:
            a_prev = None
    except Exception:
        a_prev = None

    best_seq = a.copy()
    best_score = objective_score(a)

    # possibly improve from the previous construction immediately
    if a_prev is not None:
        s_prev = objective_score(a_prev)
        if s_prev < best_score:
            best_score = s_prev
            best_seq = a_prev.copy()
            a = a_prev.copy()
            print(f"Initial improvement from height_sequence_1: score = {best_score:.6f}")

    print(f"Starting optimisation (n={n}) – initial score = {best_score:.6f}")

    # ------------------------------------------------------------------
    # Momentum buffer
    # ------------------------------------------------------------------
    v = np.zeros_like(a)

    iteration = 0
    no_improve_counter = 0

    while time.time() < deadline:
        # ---- 1. compute the convolution and its worst shift(s) ----
        conv = full_convolution(a)
        max_val = np.max(conv)

        # current objective (sum(a) == 1)
        cur_score = 2.0 * n * max_val
        if cur_score < best_score:
            best_score = cur_score
            best_seq = a.copy()
            print(f"Iter {iteration:06d}: new best = {best_score:.6f}")

            # early exit if the target is reached
            if best_score < 1.503:
                break

        # ---- 2. build a descent direction from the top K peaks ----
        # indices of the K largest convolution entries
        if max_K >= conv.size:
            top_idx = np.arange(conv.size, dtype=int)
        else:
            # argpartition gives unsorted K largest indices
            top_idx = np.argpartition(-conv, max_K - 1)[:max_K]

        # average sub‑gradient of those entries
        grad = aggregate_gradient(a, top_idx)

        # ---- 3. momentum update + projection onto simplex ----
        v = beta * v + (1.0 - beta) * grad
        step = step_initial / np.sqrt(iteration + 1.0)
        a_candidate = a - step * v
        a_candidate = project_onto_simplex(a_candidate)

        # ---- 4. evaluate candidate ----
        conv_cand = full_convolution(a_candidate)
        cand_score = 2.0 * n * np.max(conv_cand)

        if cand_score < cur_score - 1e-12:
            # accept the move
            a = a_candidate
            no_improve_counter = 0
        else:
            # reject – keep the old a, but still decay momentum
            no_improve_counter += 1
            # if we have been stuck for a while, reduce momentum a bit
            if no_improve_counter > 2000:
                beta = max(0.0, beta - 0.05)
                no_improve_counter = 0

        # ---- 5. occasional random jitter (helps to leave flat plateaux) ----
        if iteration % jitter_interval == 0 and iteration > 0:
            i, j = np.random.randint(0, n, size=2)
            if i != j:
                delta = 0.005 * a[i] * (np.random.rand() * 2.0 - 1.0)   # ±0.5 % of a[i]
                cand = a.copy()
                cand[i] = max(cand[i] - delta, 0.0)
                cand[j] = cand[j] + delta
                cand = project_onto_simplex(cand)
                conv_j = full_convolution(cand)
                score_j = 2.0 * n * np.max(conv_j)
                if score_j < best_score:
                    a = cand
                    best_seq = cand.copy()
                    best_score = score_j
                    print(f"Iter {iteration:06d}: jitter improvement = {best_score:.6f}")

        # ---- 6. occasional smoothing of the whole sequence ----
        if iteration % smooth_interval == 0 and iteration > 0:
            a = smooth_sequence(a)

        iteration += 1

    # ------------------------------------------------------------------
    # Safety fallback – if nothing was found use the previous best
    # ------------------------------------------------------------------
    if best_seq is None or best_seq.size == 0:
        try:
            best_seq = np.array(height_sequence_1, dtype=float)
        except Exception:
            # uniform distribution as a last resort (always yields 2.0)
            best_seq = np.full(2000, 1.0 / 2000.0, dtype=float)

    print(f"Finished after {iteration} iterations – final best score = {best_score:.6f}")
    return best_seq.tolist()
