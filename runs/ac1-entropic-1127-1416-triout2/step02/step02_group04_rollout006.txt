"""
Propose a step‑function (non‑negative sequence) that minimizes

    2 * n * max(conv(a, a)) / (sum(a) ** 2)

where ``conv`` is the full linear convolution.  Because the objective is
scale‑invariant we restrict the search to the probability simplex
(sum(a)=1, a_i ≥ 0).  The algorithm proceeds in three stages:

1.  Build a modest library of promising seeds (linear ramp, uniform,
    several discretised Beta‑distributions and a few Dirichlet draws)
    and keep the few best of them.

2.  Starting from each of the best seeds run an Adam optimiser with a
    *soft‑max* approximation of the “max’’ in the objective.
    For a smoothing parameter α the weight of convolution entry *k* is

        w_k ∝ exp(α·(conv_k – max(conv))).

    The gradient of the smoothed objective is obtained analytically as a
    cross‑correlation and computed efficiently with FFTs.
    After every step the iterate is projected onto the probability simplex.

3.  If the optimiser stalls for a while we inject a small random
    perturbation of the current best vector (a cheap way to escape shallow
    local minima) and continue.  The best vector seen over the whole run
    is returned.

The whole search respects the 1000 s wall‑clock budget and prints
progress messages.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities
# ----------------------------------------------------------------------


def simplex_projection(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the probability simplex:
    { w | w_i ≥ 0 , sum_i w_i = 1 }.
    Implements the algorithm of Duchi et al. (2008).
    """
    v = np.asarray(v, dtype=float)
    n = v.shape[0]

    # sort v in decreasing order
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)

    # Find the rho index
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all entries non‑positive → uniform distribution
        return np.full_like(v, 1.0 / n)

    rho = ind[cond][-1] - 1                # zero‑based
    theta = cssv[cond][-1] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def compute_objective(a: np.ndarray) -> float:
    """
    Exact objective for a vector normalised to sum=1:
        2·n·max(conv(a,a))

    Uses FFT for O(n log n) convolution.
    """
    n = a.shape[0]
    conv_len = 2 * n - 1
    fft_len = 1 << (conv_len - 1).bit_length()      # next power of two
    A = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(A * A, n=fft_len)[:conv_len]
    return float(2.0 * n * conv.max())


def beta_sequence(alpha: float, beta: float, n: int) -> np.ndarray:
    """
    Discretised Beta(α,β) on n points, normalised to sum to one.
    The midpoint rule avoids the singular endpoints.
    """
    x = (np.arange(n) + 0.5) / n
    # work in the log‑domain to avoid overflow
    logw = (alpha - 1.0) * np.log(x) + (beta - 1.0) * np.log(1.0 - x)
    logw = np.clip(logw, -700.0, 700.0)     # np.exp safe range
    w = np.exp(logw)
    total = w.sum()
    if total == 0.0 or not np.isfinite(total):
        w = np.ones_like(w)
        total = w.sum()
    return w / total

def propose_candidate(seed: int = 0, budget_s: int = 1000) -> list[float]:
    """
    Search for a non‑negative height sequence that minimises the
    evaluation function.  Returns the best sequence found (as a plain
    Python list) before the time budget expires.
    """
    np.random.seed(seed)
    start_time = time.time()
    # safety margin so that we always finish before the hard timeout
    deadline = start_time + budget_s - 5.0

    # --------------------------------------------------------------
    # Problem size
    # --------------------------------------------------------------
    try:
        n = len(height_sequence_1)                 # type: ignore[name-defined]
    except NameError:
        n = 5146                                   # fallback from the prompt

    conv_len = 2 * n - 1
    # FFT length that can accommodate convolution (a*a) *and* correlation (p*a)
    fft_len = 1 << ((conv_len + n - 1) - 1).bit_length()   # ≥ 3n‑2

    # --------------------------------------------------------------
    # 1.  Generate a pool of deterministic / random seeds
    # --------------------------------------------------------------
    seeds = []

    # linear decreasing ramp (used in the previous baseline)
    a_ramp = np.arange(n, 0, -1, dtype=float)
    a_ramp = 2.0 * a_ramp / (n * (n + 1))
    a_ramp /= a_ramp.sum()
    seeds.append(a_ramp)

    # uniform distribution
    seeds.append(np.full(n, 1.0 / n))

    # systematic Beta‑grid (covers decreasing ramps and U‑shapes)
    al_grid = np.linspace(0.2, 4.0, 9)
    be_grid = np.linspace(0.2, 4.0, 9)
    for a in al_grid:
        for b in be_grid:
            if time.time() > deadline:
                break
            seeds.append(beta_sequence(a, b, n))
        if time.time() > deadline:
            break

    # random Beta draws for extra coverage
    for _ in range(150):
        if time.time() > deadline:
            break
        a = np.random.uniform(0.05, 4.0)
        b = np.random.uniform(0.05, 4.0)
        seeds.append(beta_sequence(a, b, n))

    # a handful of Dirichlet‑like vectors
    for _ in range(20):
        if time.time() > deadline:
            break
        conc = np.random.uniform(0.1, 2.0, size=n)
        a = np.random.gamma(conc, 1.0)
        a /= a.sum()
        seeds.append(a)

    # --------------------------------------------------------------
    # 2.  Evaluate seeds and keep the best few
    # --------------------------------------------------------------
    scored_seeds = [(compute_objective(a), a) for a in seeds]
    scored_seeds.sort(key=lambda x: x[0])
    best_score = scored_seeds[0][0]
    best_a = scored_seeds[0][1].copy()
    print(f"Initial best from seeds: score = {best_score:.6f}")

    # keep top‑k seeds for the Adam phase (k=5 balances diversity / time)
    top_k = 5
    initial_vectors = [vec.copy() for (_, vec) in scored_seeds[:top_k]]

    # --------------------------------------------------------------
    # 3.  Adam optimiser with soft‑max smoothing of the max‑operator
    # --------------------------------------------------------------
    beta1, beta2 = 0.9, 0.999
    epsilon = 1e-8
    base_lr = 0.5                # initial learning rate
    max_total_iter = 200_000     # overall iteration upper bound
    iter_cnt = 0
    stagnation_counter = 0
    stagnation_limit = 4000      # after this many non‑improving steps we perturb

    # schedule for the soft‑max temperature (α).  Start smooth, end sharp.
    alpha_start, alpha_end = 5.0, 250.0

    # We run a loop that cycles through the selected starting points.
    # After each full cycle we continue from the current best vector.
    current_vec = best_a.copy()
    m = np.zeros_like(current_vec)   # first moment estimate
    v = np.zeros_like(current_vec)   # second moment estimate

    # Helper: compute convolution (full) of the current vector
    def compute_full_conv(x):
        A = np.fft.rfft(x, n=fft_len)
        conv_full = np.fft.irfft(A * A, n=fft_len)[:conv_len]
        return conv_full, A

    while time.time() < deadline and iter_cnt < max_total_iter:
        # ----------------------------------------------------------
        # 3a) convolution of the current iterate
        # ----------------------------------------------------------
        conv, A_fft = compute_full_conv(current_vec)
        max_b = conv.max()
        cur_score = 2.0 * n * max_b

        # update global best if we improved
        if cur_score < best_score:
            best_score = cur_score
            best_a = current_vec.copy()
            stagnation_counter = 0
            print(f"Iter {iter_cnt:6d}: new best {best_score:.6f}")
        else:
            stagnation_counter += 1

        # ----------------------------------------------------------
        # 3b) soft‑max weights over convolution entries
        # ----------------------------------------------------------
        # linear schedule for the smoothing factor α
        alpha = alpha_start + (alpha_end - alpha_start) * (iter_cnt / max_total_iter)
        # exp(α·(conv – max)) is safe because exponent ≤ 0
        w = np.exp(alpha * (conv - max_b))
        w_sum = w.sum()
        if w_sum == 0.0 or not np.isfinite(w_sum):
            # fallback to uniform weights – should never happen
            w = np.full_like(w, 1.0 / conv_len)
        else:
            w /= w_sum

        # ----------------------------------------------------------
        # 3c) gradient of the smoothed objective
        # ----------------------------------------------------------
        # gradient = 2 * (w ⋆ current_vec)  (cross‑correlation)
        W_fft = np.fft.rfft(w, n=fft_len)
        # cross‑correlation via inverse FFT of W * conj(A)
        corr_full = np.fft.irfft(W_fft * np.conj(A_fft), n=fft_len)
        # slice the part that corresponds to lags 0 … n‑1
        g = 2.0 * corr_full[n - 1 : n - 1 + n]

        # ----------------------------------------------------------
        # 3d) Adam update
        # ----------------------------------------------------------
        iter_cnt += 1
        t = iter_cnt

        m = beta1 * m + (1.0 - beta1) * g
        v = beta2 * v + (1.0 - beta2) * (g * g)

        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)

        step = base_lr * m_hat / (np.sqrt(v_hat) + epsilon)
        new_vec = current_vec - step
        # keep the iterate on the simplex
        current_vec = simplex_projection(new_vec)

        # ----------------------------------------------------------
        # 3e) occasional random perturbation to escape plateaus
        # ----------------------------------------------------------
        if stagnation_counter >= stagnation_limit:
            # inject a small Gaussian noise around the *best* vector seen
            noise = np.random.normal(scale=0.02, size=n)
            perturbed = best_a + noise
            perturbed = np.maximum(perturbed, 0.0)
            perturbed /= perturbed.sum()
            current_vec = perturbed
            m.fill(0.0)
            v.fill(0.0)
            stagnation_counter = 0
            print(f"Iter {iter_cnt:6d}: perturbation applied")

    # --------------------------------------------------------------
    # 4.  Return best solution found
    # --------------------------------------------------------------
    print(f"Finished after {iter_cnt:,d} iterations, best score = {best_score:.6f}")
    # Convert to plain Python list of floats as required by the evaluator
    return [float(v) for v in best_a.tolist()]
