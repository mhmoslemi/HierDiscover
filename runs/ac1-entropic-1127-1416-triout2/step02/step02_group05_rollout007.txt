"""
propose_candidate
-----------------
A mirror‑descent / exponentiated‑gradient optimiser with a soft‑max
approximation of the “max‑convolution’’ term.
The algorithm works directly on the probability simplex (∑aᵢ = 1,
aᵢ ≥ 0) by repeatedly

1. computing a*a via an FFT,
2. turning the convolution into a smooth probability distribution
   (soft‑max with a temperature β that grows with time),
3. forming the gradient  gᵢ = 2·∑ₖ wₖ·a_{k‑i}  (a cross‑correlation,
   efficiently obtained by a second FFT),
4. applying an exponentiated‑gradient update   aᵢ ← aᵢ·exp(‑η·gᵢ)
   and renormalising,
5. enforcing symmetry (a = reverse(a)) and injecting occasional random
   mass‑transfer steps to escape local minima.

The step‑size η decays, while β is increased so the optimisation first
behaves like a smooth L₂‑type problem and gradually focuses on the true
ℓ∞‑objective.  After the main time‑limited loop a short projected‑gradient
refinement (the classic “gradient‑descent / simulated‑annealing’’ step)
is performed from the best point found.

The returned list contains non‑negative heights and is guaranteed to meet
the security checks of ``evaluate_sequence``.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
def binomial_sequence(length: int) -> np.ndarray:
    """Central‑binomial coefficients cₖ = C(2k,k)/4ᵏ, k = 0…length‑1."""
    seq = np.empty(length, dtype=np.float64)
    seq[0] = 1.0
    for k in range(1, length):
        seq[k] = seq[k - 1] * ((2 * k - 1) * (2 * k)) / (4.0 * k * k)
    return seq


# ----------------------------------------------------------------------
def next_pow2(x: int) -> int:
    """Small helper: smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: int = 1000) -> list[float]:
    """
    Search for a non‑negative step‑function (a list of heights) that minimises
    ``evaluate_sequence``.  The routine respects the time budget and returns
    the best sequence discovered.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.1   # safety margin

    # --------------------------------------------------------------
    # 1) Choose the length n.  Prefer the supplied construction if present.
    try:
        # ``height_sequence_1`` may be defined globally by the framework.
        n = len(height_sequence_1)          # type: ignore[name-defined]
        a = np.array(height_sequence_1, dtype=np.float64)
        a = np.maximum(a, 0.0)
        if a.sum() == 0.0:
            raise ValueError
    except Exception:
        # Fallback: a relatively large default size.
        n = 8000
        a = binomial_sequence(n).astype(np.float64)

    # Normalise to a probability vector (∑a = 1) – the objective is scale‑invariant.
    a = a / a.sum()

    # Small random jitter to avoid exact symmetry artefacts.
    a += 1e-8 * np.random.rand(n)
    a = a / a.sum()

    # Best‑so‑far bookkeeping.
    best_seq = a.copy()
    best_score = 2.0 * n * np.convolve(a, a).max()
    print(f"[init] n={n}, initial score≈{best_score:.9f}")

    # --------------------------------------------------------------
    # 2) Optimisation hyper‑parameters.
    eta = 0.5               # initial exponentiated‑gradient step
    eta_decay = 0.9995
    min_eta = 1e-4

    beta = 5.0              # soft‑max “temperature”
    beta_mul = 1.01
    max_beta = 200.0

    # Main optimisation loop – mirror descent on the simplex.
    it = 0
    while time.time() < deadline:
        # ---- convolution a*a (FFT for speed) ----
        L = next_pow2(2 * n - 1)
        A_hat = np.fft.rfft(a, n=L)
        conv = np.fft.irfft(A_hat * A_hat, n=L)[: 2 * n - 1]

        # ---- current objective (exact, no soft‑max) ----
        cur_score = 2.0 * n * conv.max()
        if cur_score < best_score:
            best_score = cur_score
            best_seq = a.copy()
            print(f"[improve] iter={it}, score={best_score:.9f}")

        # ---- soft‑max weights over convolution entries ----
        # shift by the maximum for numerical stability
        max_c = conv.max()
        exp_arg = beta * (conv - max_c)
        w = np.exp(exp_arg)
        w_sum = w.sum()
        if w_sum == 0.0:
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        # ---- gradient gᵢ = 2·∑ₖ wₖ·a_{k-i} (cross‑correlation) ----
        rev_a = a[::-1]
        L2 = next_pow2(len(w) + n - 1)          # length for rev_a * w
        RevA_hat = np.fft.rfft(rev_a, n=L2)
        W_hat = np.fft.rfft(w, n=L2)
        corr = np.fft.irfft(RevA_hat * W_hat, n=L2)
        g = 2.0 * corr[n - 1 : n - 1 + n]       # slice the n relevant entries

        # ---- exponentiated‑gradient update (mirror descent) ----
        # Clip the gradient to avoid overflow in the exponent.
        g_clip = np.clip(g, -10.0, 10.0)
        a = a * np.exp(-eta * g_clip)

        # Enforce non‑negativity and re‑normalise (simplex projection).
        a = np.maximum(a, 0.0)
        total = a.sum()
        if total == 0.0:
            a = np.full(n, 1.0 / n)
        else:
            a /= total

        # Symmetrise the candidate – the optimal shape is known to be symmetric.
        a = 0.5 * (a + a[::-1])
        a /= a.sum()

        # Occasional random mass transfer (simulated‑annealing flavour).
        if np.random.rand() < 0.005:
            i, j = np.random.randint(0, n, size=2)
            if i != j and a[i] > 1e-12:
                amt = 0.05 * a[i] * np.random.rand()
                a[i] -= amt
                a[j] += amt
                a = np.maximum(a, 0.0)
                a /= a.sum()
                a = 0.5 * (a + a[::-1])   # keep symmetry after the perturbation

        # ---- schedule update ----
        eta = max(min_eta, eta * eta_decay)
        beta = min(max_beta, beta * beta_mul)

        it += 1

    # --------------------------------------------------------------
    # 3) Short projected‑gradient refinement from the best point.
    #    This mirrors the earlier gradient‑descent / simulated‑annealing
    #    approach and often squeezes a few more digits.
    a = best_seq.copy()
    step = 0.02
    step_decay = 0.9999
    min_step = 1e-6
    refin_iters = 3000
    for _ in range(refin_iters):
        conv = np.convolve(a, a)
        max_val = conv.max()
        # indices that are (almost) maximal
        tol = max_val - 1e-14
        max_idxs = np.flatnonzero(conv >= tol)

        # build gradient of the (hard) max
        g = np.zeros_like(a)
        for k in max_idxs:
            lo = max(0, k - (n - 1))
            hi = min(n - 1, k)
            idx = np.arange(lo, hi + 1, dtype=int)
            g[idx] += a[k - idx]

        # zero‑sum direction (projected gradient)
        sum_g = g.sum()
        d = -g + (sum_g / n)

        a_candidate = a + step * d
        a_candidate = np.maximum(a_candidate, 0.0)
        s = a_candidate.sum()
        if s == 0.0:
            a_candidate = np.full(n, 1.0 / n)
        else:
            a_candidate /= s

        # keep symmetry
        a_candidate = 0.5 * (a_candidate + a_candidate[::-1])

        a = a_candidate
        step = max(min_step, step * step_decay)

    # Evaluate the final candidate with the official routine.
    final_score = float(2 * n * np.convolve(a, a).max())
    print(f"[final] best score={best_score:.9f} (pre‑refinement), "
          f"post‑refinement≈{final_score:.9f}, iterations={it}")

    # Return the best sequence discovered (pre‑refinement is usually superior,
    # but we keep the refined one if it turned out better).
    if final_score < best_score:
        best_seq = a.copy()

    return best_seq.tolist()
