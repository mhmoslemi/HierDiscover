"""
Propose a step‑function sequence that (almost) minimizes

        2 * n * max(conv(a,a)) / (sum(a)**2)

The algorithm consists of three phases:

1. **Randomised parametric warm‑start** – a short sweep over a 5‑parameter
   family (power‑law with optional offset, linear tilt and exponential tail).
   Each candidate is projected onto the simplex
   `{a_i ≥ 0, Σ a_i = √(2·n)}`.

2. **Momentum‑accelerated sub‑gradient descent** – we minimise the
   maximal self‑convolution value `max(conv(a,a))`.  At each iteration the
   convolution is computed via a real FFT.  All indices whose value is
   within a small relative tolerance of the current maximum are taken as
   an active set; the (sub‑)gradient is the sum of the usual convolution
   contributions over this set.  A heavy‑ball momentum term smoothes the
   updates.  After each step we project back onto the simplex.

3. **Multi‑restart** – if no improvement is seen for a while we re‑initialise
   from a fresh random candidate (again from the parametric family).  The
   best sequence seen over the whole budget is returned.

The routine respects a 1000‑second wall‑clock limit and prints occasional
progress information.  It returns a plain Python list of floats that can be
directly fed to `evaluate_sequence`.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Simplex projection (Duchi et al. 2008)
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray, target_sum: float) -> np.ndarray:
    """Project v onto the simplex { x ≥ 0, Σ x = target_sum }."""
    if target_sum < 0:
        raise ValueError("target_sum must be non‑negative")
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - target_sum
    ind = np.arange(1, len(v) + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.zeros_like(v)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# FFT‑based full convolution
# ----------------------------------------------------------------------
def _full_convolution_fft(a: np.ndarray) -> np.ndarray:
    """Return the full linear convolution a * a using a real FFT."""
    n = len(a)
    size = 1 << ((2 * n - 1).bit_length())  # next power of two
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return conv


# ----------------------------------------------------------------------
# Candidate generator (5‑parameter family)
# ----------------------------------------------------------------------
def _make_candidate(
    n: int,
    p: float,
    offset: float,
    beta: float,
    gamma: float,
) -> np.ndarray:
    """
    Build a raw sequence
        a_i = (i + 1 + offset)^{-p} * (1 + beta * i/(n-1)) *
              exp(-gamma * i / n)
    Clip negatives, then project onto the simplex of sum √(2·n).
    """
    idx = np.arange(n, dtype=float)               # i = 0 … n‑1
    # power‑law part
    base = np.power(idx + 1.0 + offset, -p)
    # linear tilt
    factor = 1.0 + beta * idx / (n - 1)
    # exponential tail
    exponential = np.exp(-gamma * idx / n)
    raw = base * factor * exponential
    raw = np.maximum(raw, 0.0)                     # guard against tiny negatives
    target = np.sqrt(2.0 * n)
    return _project_simplex(raw, target)


# ----------------------------------------------------------------------
# Sub‑gradient for a set of active convolution indices
# ----------------------------------------------------------------------
def _compute_subgradient(a: np.ndarray, active_idx: np.ndarray) -> np.ndarray:
    """
    Return the (sub‑)gradient of max(conv(a,a)) w.r.t. a for the given
    active indices.  The computation follows the exact formula used in the
    reference implementation.
    """
    n = len(a)
    g = np.zeros_like(a)
    for k in active_idx:
        if k < n:
            # contribution of a_{k-i} for i = 0 … k
            g[: k + 1] += a[k :: -1]
        else:
            # contribution for the trailing part (k ≥ n)
            start = k - (n - 1)          # smallest i with a_{k-i} defined
            # a_{k-i} runs from a_{n-1} down to a_{start}
            g[start:] += a[n - 1 : start - 1 : -1]
    return g


# ----------------------------------------------------------------------
# Main search routine
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 123, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative height sequence that minimises the evaluation
    function.  The function obeys the 1000 s time limit and returns the best
    sequence found as a plain list of floats.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5  # tiny safety margin

    # ------------------------------------------------------------------
    # 0. Choose the length of the sequence.
    # ------------------------------------------------------------------
    n = 8500                     # a compromise between resolution and speed
    target_sum = np.sqrt(2.0 * n)

    # ------------------------------------------------------------------
    # 1. Warm‑start: try to reuse a previously found construction.
    # ------------------------------------------------------------------
    best_seq = None
    hs = globals().get("height_sequence_1")
    if isinstance(hs, list) and len(hs) == n:
        best_seq = np.array(hs, dtype=float)
        best_seq = _project_simplex(best_seq, target_sum)

    if best_seq is None:
        # simple power‑law fallback
        best_seq = _make_candidate(
            n=n, p=0.5, offset=0.0, beta=0.0, gamma=0.0
        )

    # ------------------------------------------------------------------
    # 2. Fast random sweep over the 5‑parameter family.
    # ------------------------------------------------------------------
    best_score = _full_convolution_fft(best_seq).max()
    print(f"[init] n={n:5d}  score={best_score:.6f}")

    sweep_deadline = time.time() + 0.20 * (deadline - time.time())
    while time.time() < sweep_deadline:
        p = np.random.uniform(0.30, 0.70)          # exponent
        offset = np.random.uniform(0.0, 15.0)      # additive shift
        beta = np.random.uniform(-0.9, 2.5)        # linear tilt
        gamma = np.random.uniform(0.0, 0.20)       # exponential tail strength
        cand = _make_candidate(n, p, offset, beta, gamma)
        cand_score = _full_convolution_fft(cand).max()
        if cand_score < best_score:

            best_seq = cand

        if best_score < 1.48:                     # early break if lucky

    # ------------------------------------------------------------------

    # ------------------------------------------------------------------
    # Momentum‐parameter

    # Initial step‑size (will decay)

    # Relative tolerance for the active set (starts coarse, shrinks)

    # Restart control

    global_best_score = best_score

    iteration = 0
    stagnation_counter = 0
    momentum = np.zeros_like(best_seq)

    while time.time() < deadline:
        iteration += 1

        # Full convolution and current maximal value
        conv_full = _full_convolution_fft(best_seq)
        M = conv_full.max()

        # Keep global best
        if M < global_best_score:
            global_best_score = M
            global_best_seq = best_seq.copy()
            stagnation_counter = 0
            print(f"[sgd] iter {iteration:6d}  new best {global_best_score:.6f}")
            if global_best_score < 1.503:
                # target reached – exit early
                break
        else:
            stagnation_counter += 1

        # ------------------------------------------------------------------
        # 3a. Determine the active set (indices close to the maximum)
        # ------------------------------------------------------------------
        # Linear interpolation of tolerance between start and end
        t_frac = min(iteration / 200000.0, 1.0)   # safe scaling for first ~200k iters
        rel_tol = tol_start * (1 - t_frac) + tol_end * t_frac
        threshold = M * (1.0 - rel_tol)
        active_idx = np.where(conv_full >= threshold)[0]

        # ------------------------------------------------------------------
        # 3b. Compute the (sub)gradient for the active set
        # ------------------------------------------------------------------
        g = _compute_subgradient(best_seq, active_idx)

        # ------------------------------------------------------------------
        # 3c. Momentum update
        # ------------------------------------------------------------------
        momentum = beta_mom * momentum + (1.0 - beta_mom) * g

        # ------------------------------------------------------------------
        # 3d. Step size (decays like 1/√t)
        # ------------------------------------------------------------------
        step = step0 / np.sqrt(iteration)

        # ------------------------------------------------------------------
        # 3e. Gradient step + projection onto the simplex
        # ------------------------------------------------------------------
        cand = best_seq - step * momentum
        cand = _project_simplex(cand, target_sum)

        # ------------------------------------------------------------------
        # 3f. Small jitter to escape plateaus (every ~100 iterations)
        # ------------------------------------------------------------------
        if iteration % 100 == 0 and np.random.rand() < 0.15:
            jitter = (np.random.rand(n) - 0.5) * 0.02 * cand
            cand = cand + jitter
            cand = _project_simplex(cand, target_sum)

        best_seq = cand

        # ------------------------------------------------------------------
        # 3g. Restart if we have been stuck too long
        # ------------------------------------------------------------------
        if stagnation_counter >= stagnation_limit and restarts_used < max_restarts:
            restarts_used += 1
            stagnation_counter = 0
            # New random warm‑start (reuse the parametric sweep)
            p = np.random.uniform(0.30, 0.70)
            offset = np.random.uniform(0.0, 15.0)
            beta = np.random.uniform(-0.9, 2.5)
            gamma = np.random.uniform(0.0, 0.20)
            best_seq = _make_candidate(n, p, offset, beta, gamma)
            momentum = np.zeros_like(best_seq)
            print(f"[restart] #{restarts_used} after {iteration} iters – "
                  f"new start score {_full_convolution_fft(best_seq).max():.6f}")

    # ------------------------------------------------------------------
    # 4. Return the best sequence seen over the whole run
    # ------------------------------------------------------------------
    print(f"Finished – best score {global_best_score:.6f} after {iteration} iters.")
    # Store the result globally so that a later run can reuse it.

    return global_best_seq.tolist()
