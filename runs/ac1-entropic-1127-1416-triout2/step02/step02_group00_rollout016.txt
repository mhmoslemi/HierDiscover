    conv = convolution_fft(a, a)   # shape 2n-1
    # compute softmax weights w
    maxc = np.max(conv)

"""
Hybrid Adam‑gradient search for a low‑value step‑function.

The optimiser minimises a smooth approximation of the
max‑convolution objective:
    F(a) = 2·n·max(a*a) / (∑a)² .
A soft‑max (temperature τ) is applied to the full convolution,
yielding a differentiable surrogate whose gradient can be
computed by a second FFT‑convolution.  Adam‑style
momentum, a simplex projection and a geometric decay of τ
provide stable, fast descent from many different starting
shapes (uniform, arcsine, random Dirichlet, two‑parameter
family).  Several lengths are examined; the best vector is
returned within the 1000 s budget.
"""

import time
import numpy as np
from typing import List, Tuple

# ----------------------------------------------------------------------
# Basic linear‑algebra utilities (no closures, no lambdas)
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """
    Fast real convolution via FFT (zero‑padded to the next power of two).
    Returns an array of length len(a)+len(b)-1.
    """
    m = len(a) + len(b) - 1
    n_fft = next_pow2(m)
    A = np.fft.rfft(a, n=n_fft)
    B = np.fft.rfft(b, n=n_fft)
    C = A * B
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the probability simplex
    {x | x_i ≥ 0, Σ_i x_i = 1}.
    Implementation of Duchi et al. (2008).
    """
    if np.all(v <= 0):
        # any point in the simplex is OK; use the uniform distribution
        return np.full_like(v, 1.0 / len(v))

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho = np.where(u - cssv / np.arange(1, len(v) + 1) > 0)[0]
    if len(rho) == 0:                     # all entries are zero
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def compute_score(a: np.ndarray) -> float:
    """
    Internal proxy for the official evaluator (the sum of a is assumed ≈1).
    Returns 2·n·max(a*a) / (∑a)² .
    """
    n = len(a)
    conv = convolution_fft(a, a)
    max_val = float(np.max(conv))
    s = float(np.sum(a))
    if s < 1e-12:
        return float("inf")
    return 2.0 * n * max_val / (s * s)


def arcsine_initial(n: int) -> np.ndarray:
    """Classic arcsine‑shaped start: a_i ∝ 1/√((i+½)*(n‑½‑i))."""
    i = np.arange(n, dtype=float)
    eps = 0.25
    vals = 1.0 / np.sqrt((i + 0.5) * (n - 0.5 - i) + eps)
    return vals / np.sum(vals)


def generate_shape(n: int, p: float, eps: float) -> np.ndarray:
    """
    Symmetric density a_i ∝ ((i+eps)*(n‑1‑i+eps))⁻ᵖ, then normalised.
    """
    i = np.arange(n, dtype=float)
    base = ((i + eps) * (n - 1 - i + eps)) ** (-p)
    if base.sum() == 0.0:
        return np.full(n, 1.0 / n)
    return base / base.sum()


# ----------------------------------------------------------------------
# Smoothed gradient (soft‑max of the convolution)
# ----------------------------------------------------------------------


def gradient_softmax(a: np.ndarray, tau: float) -> np.ndarray:
    """
    Gradient of the soft‑max approximation of max(a*a) w.r.t. a.
    Implements
        w_k ∝ exp(conv_k / τ)   (conv = a*a)
        ∇F_i = 2 * Σ_k w_k * a_{k-i}
    using two FFT‑convolutions.
    """
    n = len(a)

    # convolution a*a
    conv = convolution_fft(a, a)               # length 2n‑1
    maxc = float(np.max(conv))

    # soft‑max weights (stable)
    exps = np.exp((conv - maxc) / tau)
    w = exps / np.sum(exps)                     # length 2n‑1, non‑negative

    # second convolution: w * a_rev
    conv_w_rev = convolution_fft(w, a[::-1])   # length 3n‑2
    # extract the n components that correspond to Σ_k w_k * a_{k-i}
    grad = 2.0 * conv_w_rev[n - 1 : 2 * n - 1]   # shape (n,)

    return grad


# ----------------------------------------------------------------------
# Adam‑style projected optimisation
# ----------------------------------------------------------------------


def adam_optimize(
    initial_a: np.ndarray,
    max_iter: int,
    deadline: float,
    eta0: float = 0.5,
    beta1: float = 0.9,
    beta2: float = 0.999,
    epsilon: float = 1e-8,
    tau0: float = 0.12,
    tau_decay: float = 0.9995,
    tau_min: float = 0.001,
    noise_every: int = 0,
    noise_scale: float = 0.0,
) -> Tuple[np.ndarray, float]:
    """
    Projected Adam optimiser using the soft‑max surrogate.
    Returns the best vector seen and its internal score.
    """
    a = project_to_simplex(initial_a.astype(float).copy())
    best_a = a.copy()
    best_sc = compute_score(a)

    n = len(a)
    m = np.zeros_like(a)          # 1st moment
    v = np.zeros_like(a)          # 2nd moment
    tau = tau0

    for it in range(1, max_iter + 1):
        if time.time() >= deadline:
            break

        # --- gradient of the smooth surrogate ---
        grad = gradient_softmax(a, tau)

        # --- Adam updates (descent) ---
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)

        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        step = eta0 * m_hat / (np.sqrt(v_hat) + epsilon)
        a = a - step

        # optional stochastic perturbation (helps escape plateaus)
        if noise_every and (it % noise_every == 0):
            a += np.random.randn(n) * noise_scale

        # keep on the probability simplex
        a = project_to_simplex(a)

        # anneal temperature
        tau = max(tau * tau_decay, tau_min)

        cur_sc = compute_score(a)
        if cur_sc < best_sc:
            best_sc = cur_sc
            best_a = a.copy()

        # occasional progress report
        if it % 5000 == 0:
            print(f"    it={it:,d}  cur={cur_sc:.6f}  best={best_sc:.6f}  τ={tau:.5f}")

    return best_a, best_sc

    # 3.  Optional final polishing of the globally best vector.
    # ------------------------------------------------------------------
    if best_seq and time.time() < deadline - 5.0:
        a = np.array(best_seq, dtype=float)
        remaining = deadline - time.time()
        n = len(a)
        it_budget = int(min(250_000, max(20_000, remaining * (2500.0 / n))))

        print(f"\nFinal polishing (length {n}, max_iter≈{it_budget:,d}) …")
        a_refined, _ = adam_optimize(
            initial_a=a,
            max_iter=it_budget,
            deadline=deadline,
            eta0=0.4,
            beta1=0.9,
            beta2=0.999,
            epsilon=1e-8,
            tau0=0.06,
            tau_decay=0.9999,
            tau_min=0.0005,
            noise_every=600,
            noise_scale=2e-5,
        )
        final_val = evaluate_sequence(a_refined.tolist())
        print(f"Final polished score = {final_val:.6f}")
        if final_val < best_val:
            best_val = final_val
            best_seq = a_refined.tolist()
            print("*** Updated global best after final polish ***")

    # ------------------------------------------------------------------
    # 4.  Guarantee a return value.
    # ------------------------------------------------------------------
    if not best_seq:
        best_seq = [1.0] * 256
        best_val = evaluate_sequence(best_seq)

    print(f"\nSearch finished. Best reported score = {best_val:.6f}")
    return best_seq
