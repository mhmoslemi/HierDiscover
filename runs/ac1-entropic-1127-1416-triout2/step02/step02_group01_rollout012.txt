"""
Propose a step function (non‑negative sequence) that minimizes the
`evaluate_sequence` score.

Algorithm
---------

* Several problem sizes `n` are tried (up to a few × 10⁴).  
* For each `n` we generate three initial sequences:
      – an arcsine‑shaped distribution,
      – a uniform distribution,
      – a random Dirichlet sample.
* Each start is refined by an Adam optimiser that minimises a smooth
  approximation of the objective

        f_α(a) = (1/α)·log Σ_k exp(α·(a*a)_k) ,

  i.e. a soft‑max of the self‑convolution.  The smoothing parameter α
  is increased from a small value to a large one (homotopy).  After each
  Adam step the iterate is projected onto the probability simplex
  (∑a_i = 1, a_i ≥ 0).  The gradient of the smoothed objective is
  computed analytically via two FFT‑based convolutions (O(n log n)).
* The best vector encountered (lowest *true* score
  `2·n·max(a*a) / (∑a)²`) is kept globally.
* When the remaining time is small a cheap stochastic hill‑climbing
  phase moves a small amount of mass between two coordinates, accepting
  only improving moves.
* If a previously discovered construction (`height_sequence_1`) exists,
  it is tried as a fallback.

The whole search respects the 1000 s budget; it stops just before the
deadline and returns the best sequence found.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Basic utilities
# ----------------------------------------------------------------------


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of a vector onto the probability simplex."""
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / v.shape[0])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho = np.where(u - cssv / np.arange(1, v.shape[0] + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full linear convolution using a real FFT (fast)."""
    m = len(a) + len(b) - 1
    n_fft = next_pow2(m)
    A = np.fft.rfft(a, n=n_fft)
    B = np.fft.rfft(b, n=n_fft)
    C = A * B
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


def compute_score(a: np.ndarray) -> float:
    """Score identical to `evaluate_sequence` when ∑a = 1."""
    n = len(a)
    conv = convolution_fft(a, a)
    max_val = float(np.max(conv))
    if max_val < 0:
        max_val = 0.0
    return 2.0 * n * max_val / (np.sum(a) ** 2)


def arcsine_initial(n: int) -> np.ndarray:
    """Arcsine‑shaped start: a_i ∝ 1 / sqrt((i+0.5)(n‑0.5‑i))."""
    i = np.arange(n, dtype=float)
    eps = 0.25
    vals = 1.0 / np.sqrt((i + 0.5) * (n - 0.5 - i) + eps)
    return vals / np.sum(vals)


# ----------------------------------------------------------------------
# Smoothed‑max gradient (corrected)
# ----------------------------------------------------------------------


def smoothed_gradient(a: np.ndarray, alpha: float):
    """
    Gradient of the smoothed maximum
        f_α(a) = (1/α)·log Σ_k exp(α·(a*a)_k)

    Returns the gradient (same shape as a) and the raw convolution a*a.
    """
    n = len(a)
    conv = convolution_fft(a, a)                # length 2n‑1
    maxc = np.max(conv)
    # soft‑max weights
    exp_term = np.exp(alpha * (conv - maxc))
    w = exp_term / np.sum(exp_term)             # Σ_k w_k = 1

    # convolution of reversed a with the weight vector
    a_rev = a[::-1]
    conv_rev = convolution_fft(a_rev, w)        # length 3n‑2
    start = n - 1
    rev_slice = conv_rev[start:start + n]       # needed slice

    # correct gradient (no diagonal correction needed)
    grad = 2.0 * rev_slice
    return grad, conv


# ----------------------------------------------------------------------
# Adam optimiser on the smoothed objective
# ----------------------------------------------------------------------


def adam_optimize(
    initial_a: np.ndarray,
    time_deadline: float,
    max_iter: int = 100_000,
    lr: float = 0.3,
    alpha_start: float = 1.0,
    alpha_end: float = 500.0,
    beta1: float = 0.9,
    beta2: float = 0.999,
    eps: float = 1e-8,
    noise_every: int = 0,
    noise_scale: float = 0.0,
):
    """
    Projected Adam optimisation of the smoothed max objective.
    Returns the best vector encountered (as a NumPy array) and its
    unsmoothed score.
    """
    a = project_to_simplex(initial_a.astype(float).copy())
    best_a = a.copy()
    best_score = compute_score(a)

    n = len(a)
    m = np.zeros_like(a)          # 1st moment
    v = np.zeros_like(a)          # 2nd moment
    b1_pow = 1.0
    b2_pow = 1.0

    for t in range(1, max_iter + 1):
        if time.time() >= time_deadline:
            break

        # update smoothing parameter α (linear homotopy)
        alpha = alpha_start + (alpha_end - alpha_start) * min(1.0, t / max_iter)

        grad, _ = smoothed_gradient(a, alpha)

        # Adam moment updates
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)

        b1_pow *= beta1
        b2_pow *= beta2
        m_hat = m / (1.0 - b1_pow)
        v_hat = v / (1.0 - b2_pow)

        step = lr * m_hat / (np.sqrt(v_hat) + eps)

        a_new = a - step

        if noise_every and (t % noise_every == 0):
            a_new += np.random.randn(n) * noise_scale

        a_new = project_to_simplex(a_new)

        cur_score = compute_score(a_new)
        if cur_score < best_score:
            best_score = cur_score
            best_a = a_new.copy()

        a = a_new

        # occasional progress report
        if t % 5000 == 0:
            print(f"    iter {t:,d}  α={alpha:.2f}  score={cur_score:.6f}  best={best_score:.6f}")

    return best_a, best_score


# ----------------------------------------------------------------------
# Cheap stochastic hill‑climbing refinement
# ----------------------------------------------------------------------


def hill_climb(seq, time_deadline: float, max_steps: int = 60_000, init_step: float = 0.05):
    """
    Stochastic pair‑wise mass exchange hill‑climbing.
    Moves that improve the *true* score are kept.
    """
    a = np.array(seq, dtype=float)
    a = project_to_simplex(a)
    best_a = a.copy()
    best_score = compute_score(best_a)
    n = len(a)
    step = init_step

    while max_steps > 0 and time.time() < time_deadline:
        i, j = np.random.randint(n, size=2)
        if i == j:
            continue

        delta = step * best_a[i]
        if delta <= 0:
            continue

        cand = best_a.copy()
        cand[i] -= delta
        cand[j] += delta
        if cand[i] < 0:
            continue

        cand = project_to_simplex(cand)
        cur_score = compute_score(cand)

        if cur_score < best_score:
            best_score = cur_score
            best_a = cand
            step = min(step * 1.07, 0.5)   # accelerate when successful
        else:
            step *= 0.97                     # otherwise shrink step

        max_steps -= 1

    return best_a.tolist(), best_score

def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):

    best_seq = None
    best_val = float("inf")

    # ------------------------------------------------------------------

    try:
        fallback = [float(x) for x in height_sequence_1]  # noqa: F821

    except Exception:

        pass

    # ------------------------------------------------------------------
    # Problem sizes to explore (larger n gives a better asymptotic constant).
    # ------------------------------------------------------------------
    lengths = [1024, 2048, 4096, 8192]

    for n in lengths:
