    beta1=0.9
    beta2=0.999
    eps=1e-8
    best_a = a.copy()
    best_sc = compute_score(a)
    it = 0
    while time.time() < deadline and it < max_iter:
        it += 1
        # get worst conv index
        _, k_max = max_convolution_and_index(a)
        grad = subgradient_of_max_at_index(a, k_max)
        # Adam update

        v = beta2 * v + (1 - beta2) * (grad**2)

            best_a = a.copy()

def lp_update(a, M):
    n = len(a)
    rows = 2*n - 1
    # Build constraint matrix A (rows x n)
    A = np.zeros((rows, n), dtype=np.float64)
    for j in range(n):
        A[j:j+n, j] = a
    # Objective: maximize sum(b) -> minimize -sum(b)
    c = -np.ones(n)
    bounds = [(0, None)] * n
    b_ub = np.full(rows, M, dtype=np.float64)
    res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
    if not res.success:
        return a  # no improvement
    b = res.x
    # project to simplex
    b = project_to_simplex(b)
    return b

"""
A fast mixed‑strategy optimiser for the step‑function bound.

Main ideas
----------
1. **Broad random exploration** – for several lengths (256 … 8192) we
   generate thousands of candidates from a 2‑parameter family
   a_i ∝ (i+ε)^(-p_left)·(n‑1‑i+ε)^(-p_right).  The family contains the
   classical arcsine shape as a special case and also covers many
   asymmetric possibilities.

2. **Accelerated projected Adam** – every promising candidate is refined
   with a projected Adam optimiser that minimises the non‑smooth
   objective 2·n·max(a*a)/‖a‖_1².  The gradient is the sub‑gradient at the
   maximal convolution entry; Adam’s adaptive step‑sizes give a much
   faster descent than plain sub‑gradient descent.

3. **Linear‑programming “exchange” step** – once a good shape is found
   we solve the LP
          maximise   Σ b_j
          subject to b ≥ 0,
                     (a ∗ b) ≤ M   (M = max(a ∗ a)),
   using SciPy’s ``highs`` solver.  The LP direction is mixed into the
   current vector, projected back onto the simplex and the process is
   repeated a few times.  This is the exact analogue of the update
   described in the statement and typically yields a noticeable drop in
   the maximal self‑convolution.

The whole routine respects the 1000 s budget, always returns a
non‑negative list, and tries to beat the previous best score
(~1.518).  If SciPy is unavailable the LP stage is silently skipped.
"""

import time
import math
import numpy as np
from typing import List, Tuple

# ----------------------------------------------------------------------
# Basic linear‑algebra utilities (no closures)
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Convolution via real FFT (zero‑padded to next power of two)."""
    m = len(a) + len(b) - 1
    n_fft = next_pow2(m)
    A = np.fft.rfft(a, n=n_fft)
    B = np.fft.rfft(b, n=n_fft)
    C = A * B
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


def max_convolution_and_index(a: np.ndarray) -> Tuple[float, int]:
    """Return (max value, index) of a*a."""
    conv = convolution_fft(a, a)
    idx = int(np.argmax(conv))
    return float(conv[idx]), idx


def subgradient_of_max_at_index(a: np.ndarray, k: int) -> np.ndarray:
    """
    Sub‑gradient of the maximal convolution entry (index k) w.r.t. a.
    """
    n = len(a)
    grad = np.zeros_like(a)
    i = np.arange(n)
    j = k - i
    mask = (j >= 0) & (j < n)
    grad[mask] = a[j[mask]]
    if k % 2 == 0:
        diag = k // 2
        if diag < n:
            grad[diag] = 2.0 * a[diag]
    return grad


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection onto the probability simplex {x ≥ 0, Σx = 1}.
    Duchi et al. (2008) implementation.
    """
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / len(v))

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho = np.where(u - cssv / np.arange(1, len(v) + 1) > 0)[0]
    if len(rho) == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def compute_score(a: np.ndarray) -> float:
    """
    Fast evaluation of the objective (scale‑invariant version):
        2·n·max(a*a) / (sum(a)²)
    The input should already be non‑negative.
    """
    n = len(a)
    conv = convolution_fft(a, a)
    max_val = float(np.max(conv))
    s = float(np.sum(a))
    if s < 1e-12:
        return float("inf")
    return 2.0 * n * max_val / (s * s)


# ----------------------------------------------------------------------
# Shape generators
# ----------------------------------------------------------------------


def arcsine_initial(n: int) -> np.ndarray:
    """Classic arcsine start: a_i ∝ 1/√((i+½)*(n‑½‑i))."""
    i = np.arange(n, dtype=float)
    eps = 0.25
    vals = 1.0 / np.sqrt((i + 0.5) * (n - 0.5 - i) + eps)
    return vals / np.sum(vals)


def generate_general_shape(
    n: int, p_left: float, p_right: float, eps: float
) -> np.ndarray:
    """
    Symmetric or asymmetric power‑law shape:
        a_i ∝ (i+ε)^(-p_left)·(n‑1‑i+ε)^(-p_right)
    Normalised to L¹‑norm = 1.
    """
    i = np.arange(n, dtype=float)
    left = (i + eps) ** (-p_left)
    right = (n - 1 - i + eps) ** (-p_right)
    vals = left * right
    total = np.sum(vals)
    if total == 0.0:
        return np.full(n, 1.0 / n)
    return vals / total


# ----------------------------------------------------------------------
# Adam‑based projected sub‑gradient optimiser
# ----------------------------------------------------------------------


def improve_subgradient_adam(
    init_a: np.ndarray,
    deadline: float,
    max_iter: int = 200_000,
    eta0: float = 0.6,
    beta1: float = 0.9,
    beta2: float = 0.999,
    eps: float = 1e-8,
) -> Tuple[np.ndarray, float]:
    """
    Projected Adam on the max‑convolution objective.
    Returns the best vector seen and its internal score.
    """
    a = project_to_simplex(init_a.astype(float).copy())
    m = np.zeros_like(a)
    v = np.zeros_like(a)

    best_a = a.copy()
    best_sc = compute_score(a)

    it = 0
    while time.time() < deadline and it < max_iter:
        it += 1
        # maximal convolution entry
        _, k_max = max_convolution_and_index(a)
        grad = subgradient_of_max_at_index(a, k_max)

        # Adam moments
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)

        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        eta = eta0 / math.sqrt(it)
        a = a - eta * m_hat / (np.sqrt(v_hat) + eps)
        a = project_to_simplex(a)

        sc = compute_score(a)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()
        if it % 5000 == 0:
            print(
                f"    [Adam] iter={it:,d} cur={sc:.6f} best={best_sc:.6f}"
            )
    return best_a, best_sc


# ----------------------------------------------------------------------
# Linear–programming exchange step (optional, needs SciPy)
# ----------------------------------------------------------------------


try:
    from scipy.optimize import linprog
except Exception:  # pragma: no cover
    linprog = None


def lp_update(a: np.ndarray, M: float) -> np.ndarray:
    """
    Solve   maximise Σ b_j   s.t.   b ≥ 0  and  (a ∗ b) ≤ M .
    Return the feasible point b projected onto the simplex.
    If SciPy is not available a copy of ``a`` is returned.
    """
    if linprog is None:
        return a  # fallback – no LP available

    n = len(a)
    rows = 2 * n - 1

    # Build the Toeplitz‑type constraint matrix A (rows × n)
    A = np.zeros((rows, n), dtype=np.float64)
    for j in range(n):
        A[j : j + n, j] = a

    c = -np.ones(n)                       # maximise Σb  → minimise -Σb
    bounds = [(0.0, None)] * n
    b_ub = np.full(rows, M, dtype=np.float64)

    res = linprog(
        c,
        A_ub=A,
        b_ub=b_ub,
        bounds=bounds,
        method="highs",
        options={"presolve": True},
    )
    if not res.success:
        return a
    b = res.x
    b = project_to_simplex(b)
    return b


# ----------------------------------------------------------------------
# Combined refinement: LP steps interleaved with Adam sub‑gradient
# ----------------------------------------------------------------------


def refine_combined(
    a: np.ndarray,
    deadline: float,
    lp_steps: int = 6,
    mix_alpha: float = 0.5,
    sub_max_iter: int = 150_000,
) -> Tuple[np.ndarray, float]:
    """
    Apply a few LP exchange steps, each followed by a short Adam run.
    """
    best_a = a.copy()
    best_sc = compute_score(a)

    for step in range(lp_steps):
        if time.time() >= deadline - 2.0:
            break

        # LP exchange
        M = np.max(convolution_fft(a, a))
        b = lp_update(a, M)
        a = (1.0 - mix_alpha) * a + mix_alpha * b
        a = project_to_simplex(a)

        sc = compute_score(a)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()
        print(f"    [LP] step {step + 1}/{lp_steps}   score={sc:.6f}")

        # Adam sub‑gradient (short burst)
        remaining = deadline - time.time()
        if remaining < 5.0:
            break
        max_iter = int(min(sub_max_iter, remaining * 30))
        a, sc_sub = improve_subgradient_adam(
            a,
            deadline=deadline,
            max_iter=max_iter,
            eta0=0.7,
        )
        if sc_sub < best_sc:
            best_sc = sc_sub
            best_a = a.copy()
        print(f"    [Adam] after LP step score={sc_sub:.6f}")

    return best_a, best_sc


# ----------------------------------------------------------------------
# Main entry point required by the competition harness
# ----------------------------------------------------------------------


def propose_candidate(
    seed: int = 123,
    budget_s: int = 1000,
    **kwargs,
) -> List[float]:
    """
    Search for a non‑negative step‑function with a low
    ``evaluate_sequence`` value.  The routine respects the time budget,
    explores several lengths, performs a broad random scan of a flexible
    two‑parameter family, refines the best candidate with an accelerated
    Adam optimiser and a few LP‑exchange steps, and finally returns the
    best sequence found.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 2.0  # safety margin

    best_seq: List[float] = []
    best_val: float = float("inf")

    # ------------------------------------------------------------------
    # 0.  Optional fallback from the harness
    # ------------------------------------------------------------------
    try:
        fallback = [float(x) for x in height_sequence_1]  # type: ignore
        val = evaluate_sequence(fallback)
        print(f"Fallback (height_sequence_1) score = {val:.6f}")
        if val < best_val:
            best_val = val
            best_seq = fallback
    except Exception:
        pass

    # ------------------------------------------------------------------
    # 1.  Broad random exploration across several lengths
    # ------------------------------------------------------------------
    lengths = [256, 512, 1024, 2048, 4096, 8192]

    for n in lengths:
        if time.time() > deadline - 30.0:
            break
        print(f"\n=== Exploring length n = {n} ===")

        # deterministic starters
        starters = [np.full(n, 1.0 / n), arcsine_initial(n)]
        local_best_a = None
        local_best_sc = float("inf")

        for s in starters:
            sc = compute_score(s)
            if sc < local_best_sc:
                local_best_sc = sc
                local_best_a = s.copy()

        # random sample from the two‑parameter family
        max_samples = 2000 if n <= 2048 else 800
        for _ in range(max_samples):
            if time.time() > deadline - 30.0:
                break
            p_left = rng.uniform(0.1, 1.0)
            p_right = rng.uniform(0.1, 1.0)
            eps = rng.uniform(0.001, max(2.0, 0.02 * n))
            cand = generate_general_shape(n, p_left, p_right, eps)
            sc = compute_score(cand)
            if sc < local_best_sc:
                local_best_sc = sc
                local_best_a = cand.copy()

        # evaluate the raw best candidate (slow evaluator) – optional
        if local_best_a is not None:
            val = evaluate_sequence(local_best_a.tolist())
            print(f"  Raw best for n={n}  score = {val:.6f}")
            if val < best_val:
                best_val = val
                best_seq = local_best_a.tolist()
                print(f"  *** New global best (raw) ***")

    # ------------------------------------------------------------------
    # 2.  Heavy refinement of the current global best
    # ------------------------------------------------------------------
    if best_seq and time.time() < deadline - 10.0:
        a = np.array(best_seq, dtype=float)
        # normalise to L1 = 1 (scale does not affect the official score)
        a = a / np.sum(a)

        remaining = deadline - time.time()
        # allocate the bulk of the remaining time to refinement
        print("\n--- Starting heavy refinement ---")
        # First a long Adam run
        a, _ = improve_subgradient_adam(
            a,
            deadline=deadline,
            max_iter=int(min(350_000, remaining * 30)),
            eta0=0.7,
        )
        # Followed by a few LP‑Adam cycles
        a, _ = refine_combined(a, deadline=deadline)

        final_val = evaluate_sequence(a.tolist())
        print(f"Final polished score = {final_val:.6f}")
        if final_val < best_val:
            best_val = final_val
            best_seq = a.tolist()
            print("*** Updated global best after final polishing ***")

    # ------------------------------------------------------------------
    # 3.  Guard – always return something
    # ------------------------------------------------------------------
    if not best_seq:
        best_seq = [1.0] * 256
        best_val = evaluate_sequence(best_seq)

    print(f"\nFinished search. Best reported score = {best_val:.6f}")
    return best_seq
```

This implementation respects the 1000 s time limit, explores a rich family of shapes, uses an accelerated Adam optimiser, optionally employs a linear‑programming exchange step (when SciPy is present), and finally returns the best non‑negative step‑function it discovered.
