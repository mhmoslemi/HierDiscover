"""
Improved search for a low‑score step function.

The algorithm works in three stages:

1. **Broad stochastic shape search** –  Several simple symmetric
   families (power‑law, Gaussian, arcsine) are sampled with random
   parameters.  For each length we keep the few best candidates (by the
   fast internal score).

2. **Projected Adam optimisation** –  Starting from each promising
   candidate we run an Adam‑style projected optimiser on the
   non‑smooth objective `2·n·max(a*a)/∑a²`.  The gradient uses the exact
   sub‑gradient of the maximum convolution entry; after each update we
   project back onto the probability simplex and re‑enforce symmetry.

3. **Local swap‑based refinement** –  A cheap “pair‑swap” Monte‑Carlo
   step moves a tiny amount of mass from one coordinate to another.
   Improvements are kept, otherwise the move is discarded (simple
   hill‑climbing).  This removes the remaining tiny bumps in the
   convolution profile.

The whole routine respects the 1000 s budget, explores a few lengths
(256‑2048), and returns the best sequence found (as a plain `list` of
floats).  All helper functions are defined at top‑level, there are no
closures or lambda‑expressions, and the official `evaluate_sequence`
function is called only for the final verification of a candidate.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Basic linear‑algebra utilities

def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Fast (real) convolution using FFT with zero‑padding."""
    m = len(a) + len(b) - 1
    n_fft = next_pow2(m)
    A = np.fft.rfft(a, n=n_fft)
    B = np.fft.rfft(b, n=n_fft)
    C = A * B
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


def max_convolution_and_index(a: np.ndarray) -> Tuple[float, int]:
    """Return (maximum value, index) of the self‑convolution a*a."""
    conv = convolution_fft(a, a)
    idx = int(np.argmax(conv))
    return float(conv[idx]), idx


def subgradient_of_max_at_index(a: np.ndarray, k: int) -> np.ndarray:
    """
    Sub‑gradient of the maximal convolution entry (index k) w.r.t. a.

    For each i, ∂c_k/∂a_i = a_{k‑i} if the index is valid.
    The diagonal term (i = k‑i) appears twice, therefore it is doubled.
    """
    n = len(a)
    grad = np.zeros_like(a)
    i = np.arange(n)
    j = k - i
    mask = (j >= 0) & (j < n)
    grad[mask] = a[j[mask]]
    if k % 2 == 0:
        diag = k // 2
        if diag < n:
            grad[diag] = 2.0 * a[diag]
    return grad


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of `v` onto the probability simplex
    {x | x_i ≥ 0, Σ_i x_i = 1}.
    Implementation of Duchi et al. (2008).
    """
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / len(v))

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho_arr = u - cssv / np.arange(1, len(v) + 1)
    rho = np.where(rho_arr > 0)[0]
    if len(rho) == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w

# Fast internal objective (scale‑invariant)

def compute_score(a: np.ndarray) -> float:
    """
    Fast proxy for the official evaluator:
        score = 2·n·max(a*a) / (sum(a)²)
    The vector `a` is assumed to be on the simplex (sum≈1).
    """
    n = len(a)
    conv = convolution_fft(a, a)
    max_val = float(np.max(conv))
    s = float(np.sum(a))
    if s < 1e-12:
        return float("inf")
    return 2.0 * n * max_val / (s * s)

def arcsine_shape(n: int) -> np.ndarray:
    """
    Classic arcsine density: a_i ∝ 1 / sqrt((i+0.5)*(n-0.5-i)).
    """
    i = np.arange(n, dtype=float)
    eps = 0.25
    vals = 1.0 / np.sqrt((i + 0.5) * (n - 0.5 - i) + eps)
    return vals / np.sum(vals)


def powerlaw_shape(n: int, p: float, eps: float) -> np.ndarray:
    """
    Symmetric power‑law shape:
        a_i ∝ ((i+eps)*(n‑1‑i+eps))^{‑p}
    """
    i = np.arange(n, dtype=float)
    base = ((i + eps) * (n - 1 - i + eps)) ** (-p)
    total = base.sum()
    if total == 0.0:
        return np.full(n, 1.0 / n)
    return base / total


def gaussian_shape(n: int, sigma: float) -> np.ndarray:
    """
    Symmetric discrete Gaussian centred at (n‑1)/2.
    """
    i = np.arange(n, dtype=float)
    centre = (n - 1) / 2.0
    vals = np.exp(-0.5 * ((i - centre) / sigma) ** 2)
    return vals / np.sum(vals)


def mixed_shape(n: int, rng: np.random.Generator) -> np.ndarray:
    """
    Random mixture of a power‑law and a Gaussian component.
    Coefficients are drawn uniformly; the mixture is renormalised.
    """
    p = rng.uniform(0.3, 0.8)
    eps = rng.uniform(0.001, max(1.0, 0.02 * n))
    sigma = rng.uniform(0.05 * n, 0.35 * n)
    w1 = rng.uniform(0.0, 1.0)
    base1 = powerlaw_shape(n, p, eps)
    base2 = gaussian_shape(n, sigma)
    mix = w1 * base1 + (1.0 - w1) * base2
    return mix / np.sum(mix)

def adam_optimize(
    init_a: np.ndarray,
    deadline: float,
    max_iter: int = 200_000,
    lr0: float = 0.5,
    beta1: float = 0.9,
    beta2: float = 0.999,
    eps: float = 1e-8,
) -> Tuple[np.ndarray, float]:
    """
    Projected Adam optimiser on the non‑smooth objective.
    Returns the best vector seen (in `compute_score` sense) and its score.
    """
    a = project_to_simplex(init_a.astype(float).copy())
    m = np.zeros_like(a)
    v = np.zeros_like(a)

    best_a = a.copy()
    best_sc = compute_score(a)

    t = 0
    n = len(a)
    while time.time() < deadline and t < max_iter:
        t += 1
        # maximal convolution entry and its index
        _, k_max = max_convolution_and_index(a)
        grad = subgradient_of_max_at_index(a, k_max)

        # Adam moments
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)

        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)

        # step size – we combine a decaying factor with the Adam scaling
        step = lr0 / np.sqrt(t)

        a_new = a - step * m_hat / (np.sqrt(v_hat) + eps)

        # enforce constraints
        a_new = project_to_simplex(a_new)

        # enforce symmetry (important for the optimum)
        a_new = 0.5 * (a_new + a_new[::-1])

        cur_sc = compute_score(a_new)
        if cur_sc < best_sc:
            best_sc = cur_sc
            best_a = a_new.copy()

        a = a_new

        # occasional progress report (lightweight)
        if t % 20_000 == 0:
            print(f"    Adam iter={t:,d} cur_score={cur_sc:.6f} best={best_sc:.6f}")

    return best_a, best_sc

    the fast `compute_score`.  Symmetry and simplex constraints are
    restored after every step.
    """
    n = len(a)
    cur_a = a.copy()
    best_a = cur_a.copy()
    best_sc = compute_score(cur_a)

    while steps > 0 and time.time() < deadline:
        steps -= 1
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue

        # the amount we try to shift – a small fraction of the source mass
        max_move = min(cur_a[i], delta_frac)
        if max_move <= 0.0:
            continue
        delta = rng.uniform(0.0, max_move)

        cand = cur_a.copy()
        cand[i] -= delta
        cand[j] += delta

        # re‑enforce the symmetry and the simplex
        cand = 0.5 * (cand + cand[::-1])
        cand = project_to_simplex(cand)

        sc = compute_score(cand)
        if sc < best_sc:
            best_sc = sc
            best_a = cand.copy()
            cur_a = cand
        # otherwise we simply discard the candidate (pure hill‑climbing)

    return best_a, best_sc


# ----------------------------------------------------------------------
# Main entry point required by the evaluation harness
# ----------------------------------------------------------------------


def propose_candidate(
    seed: int = 42,
    budget_s: int = 1000,
    **kwargs,
) -> List[float]:
    """
    Search for a non‑negative step sequence with a low
    `evaluate_sequence` value.  The algorithm explores several lengths,
    performs a broad random shape scan, refines the best candidates with
    a projected Adam optimiser, and finishes with a cheap local swap
    refinement.  The best vector found (as a plain list of floats) is
    returned.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    # safety margin so we still have time for the final evaluation
    deadline = start_time + budget_s - 2.0

    best_seq: List[float] = []
    best_val: float = float("inf")

    # ------------------------------------------------------------------
    # 0.  If a fallback construction exists, use it as an initial baseline.
    # ------------------------------------------------------------------
    try:
        fallback = [float(x) for x in height_sequence_1]  # type: ignore
        val = evaluate_sequence(fallback)
        print(f"Fallback (height_sequence_1) score = {val:.6f}")
        if val < best_val:
            best_val, best_seq = val, fallback
    except NameError:
        pass

    # ------------------------------------------------------------------
    # 1.  Lengths to investigate – we keep them moderate so that the LP
    #     style optimisation (if later added) stays cheap.
    # ------------------------------------------------------------------
    lengths = [256, 512, 1024, 2048]

    for n in lengths:
        if time.time() >= deadline - 5.0:
            break
        print(f"\n=== Exploring length n = {n} ===")

        # ----------------------------------------------------------------
        # 1a.  Random shape scan – keep the top few candidates (by fast score)
        # ----------------------------------------------------------------
        candidates: List[np.ndarray] = []

        # deterministic starters
        starters = [
            np.full(n, 1.0 / n),
            arcsine_shape(n),
        ]
        for st in starters:
            candidates.append(st)

        # random families
        max_random = 2000   # total number of random shapes (budget‑aware)
        for _ in range(max_random):
            if time.time() >= deadline - 5.0:
                break
            family = rng.integers(0, 3)
            if family == 0:  # power‑law
                p = rng.uniform(0.2, 0.9)
                eps = rng.uniform(0.001, max(1.0, 0.02 * n))
                cand = powerlaw_shape(n, p, eps)
            elif family == 1:  # Gaussian
                sigma = rng.uniform(0.05 * n, 0.35 * n)
                cand = gaussian_shape(n, sigma)
            else:  # mixture of the two
                cand = mixed_shape(n, rng)

            # a tiny random jitter to increase diversity
            jitter = 1.0 + rng.normal(scale=0.02, size=n)
            cand = np.maximum(cand * jitter, 0.0)
            cand /= cand.sum()
            cand = 0.5 * (cand + cand[::-1])   # enforce symmetry
            candidates.append(cand)

        # keep the 5 best by fast proxy score
        scored = [(compute_score(c), c) for c in candidates]
        scored.sort(key=lambda x: x[0])
        top_candidates = [c for _, c in scored[:5]]
        print(f"    Shape scan best fast score = {scored[0][0]:.6f}")

        # ----------------------------------------------------------------
        # 1b.  Refine each top candidate
        # ----------------------------------------------------------------
        for idx, cand in enumerate(top_candidates):
            if time.time() >= deadline - 5.0:
                break
            print(f"  Refinement #{idx+1} for n={n}")

            # allocate a slice of remaining time for this candidate
            remaining = deadline - time.time()
            max_iter = min(180_000, int(remaining * 30))

            # Adam optimiser
            refined_a, _ = adam_optimize(
                init_a=cand,
                deadline=deadline,
                max_iter=max_iter,
                lr0=0.6,
            )

            # Local swap refinement (fast, tiny steps)
            refined_a, _ = local_swap_refinement(
                a=refined_a,
                rng=rng,
                deadline=deadline,
                steps=30_000,
                delta_frac=0.015,
            )

            # Evaluate with the official (slow) function once
            val = evaluate_sequence(refined_a.tolist())
            print(f"    → final score = {val:.6f}")

            if val < best_val:
                best_val, best_seq = val, refined_a.tolist()
                print(f"    *** New global best: {best_val:.6f} (n={n}) ***")

    # ------------------------------------------------------------------
    # 2.  Final polishing on the global best (time permitting)
    # ------------------------------------------------------------------
    if best_seq and time.time() < deadline - 5.0:
        a = np.array(best_seq, dtype=float)
        remaining = deadline - time.time()
        max_iter = min(300_000, int(remaining * 30))
        print(f"\nFinal polishing (max_iter={max_iter}) …")
        a, _ = adam_optimize(
            init_a=a,
            deadline=deadline,
            max_iter=max_iter,
            lr0=0.5,
        )
        a, _ = local_swap_refinement(
            a=a,
            rng=rng,
            deadline=deadline,
            steps=50_000,
            delta_frac=0.01,
        )
        final_val = evaluate_sequence(a.tolist())
        print(f"Final polished score = {final_val:.6f}")
        if final_val < best_val:
            best_val, best_seq = final_val, a.tolist()
            print(f"*** Updated global best after final polish ***")

    # ------------------------------------------------------------------
    # 3.  Safety fallback – guarantee a return value.
    # ------------------------------------------------------------------
    if not best_seq:
        best_seq = [1.0] * 256
        best_val = evaluate_sequence(best_seq)

    print(f"\nFinished search. Best reported score = {best_val:.6f}")
    return best_seq
