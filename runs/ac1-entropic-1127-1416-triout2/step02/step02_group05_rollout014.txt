def compute_lp_direction(a, M):
    n = len(a)
    # Build Toeplitz matrix T of shape (2n-1, n)
    rows = 2*n-1
    T = lil_matrix((rows, n))
    for i in range(n):
        # a_i contributes to rows i .. i+n-1
        T[i:i+n, i] = a[i]   # broadcast

    res = linprog(c, A_ub=T, b_ub=np.full(rows, M), bounds=[(0, None)]*n, method='highs')
    if not res.success:
        # fallback to gradient direction
        return None
    g = res.x
    return g

def subgradient(a, max_idxs):
    n = len(a)
    g = np.zeros_like(a)
    # mask array
    mask = np.zeros(2*n-1, dtype=np.float64)
    mask[max_idxs] = 1.0

    size = 1 << (len(mask) + len(a_rev) - 1).bit_length()
    fm = np.fft.rfft(mask, size)
    fr = np.fft.rfft(a_rev, size)
    conv = np.fft.irfft(fm * fr, size)
    # extract appropriate slice: indices n-1 .. n-1+n-1
    g = conv[n-1: n-1 + n]
    return g

"""
Propose a candidate step‐function using an adaptive projected gradient /
simulated‑annealing scheme.

The algorithm works with a normalized non‑negative vector a (sum(a)=1).
At each iteration it

1. evaluates the self‑convolution a*a and locates the maximal entry(ies);
2. builds the sub‑gradient of this max (the sum of contributions of a to the
   active convolution entries);
3. forms a zero‑sum search direction d = –g + mean(g) (so the L1‑norm stays
   unchanged);
4. blends this direction with a momentum term;
5. tries a step along the direction, projects back onto the probability simplex,
   and accepts the move if the objective 2·n·max(a*a) improves (or,
   occasionally, with a Metropolis probability to escape plateaus);
6. periodically performs a larger deterministic mass transfer from the most
   “harmful’’ index (largest g) to the least harmful one (smallest g).

The step size and a temperature parameter are adapted online, allowing large
updates early on and finer exploration later.  The best sequence seen before the
time budget expires is returned.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
def binomial_sequence(length: int) -> np.ndarray:
    """Central binomial coefficients c_k = C(2k,k)/4**k for k=0..length‑1."""
    seq = np.empty(length, dtype=np.float64)
    seq[0] = 1.0
    for k in range(1, length):
        seq[k] = seq[k - 1] * ((2 * k - 1) * (2 * k)) / (4.0 * k * k)
    return seq


# ----------------------------------------------------------------------
def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a non‑negative vector onto the probability simplex
    (non‑negative, sum = 1).  Simple clipping + renormalisation works because the
    vector is never all‑zero during the optimisation.
    """
    v = np.maximum(v, 0.0)
    s = v.sum()
    if s == 0.0:
        return np.full_like(v, 1.0 / len(v))
    return v / s


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function height vector that minimises

        2 * n * max(conv(a, a))   (the scale‑invariant objective used by
        `evaluate_sequence`).

    A combination of projected gradient descent, momentum, adaptive step‑size,
    and occasional simulated‑annealing moves is employed.  The algorithm runs
    until the supplied time budget is almost exhausted and returns the best
    sequence encountered.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.1   # a tiny safety margin

    # ------------------------------------------------------------------
    # Determine the length of the search space.
    try:
        n = len(height_sequence_1)          # type: ignore[name-defined]
        a = np.array(height_sequence_1, dtype=np.float64)
    except Exception:
        n = 8000   # a comfortable size that keeps the FFT cheap
        a = binomial_sequence(n).astype(np.float64)

    # Normalise to a probability vector (sum = 1) and enforce non‑negativity.
    a = np.maximum(a, 0.0)
    a_sum = a.sum()
    if a_sum == 0.0:
        a = np.full(n, 1.0 / n)
    else:
        a = a / a_sum

    # ------------------------------------------------------------------
    # Helper to compute the convolution maximum (and the full convolution if
    # needed).  FFT based convolution is fast for the sizes we use.
    conv_len = 2 * n - 1
    fft_len = 1 << (conv_len - 1).bit_length()   # next power of two

    def max_convolution(v: np.ndarray) -> float:
        """Return max(v * v) using an FFT based linear convolution."""
        fv = np.fft.rfft(v, fft_len)
        prod = fv * fv
        conv = np.fft.irfft(prod, fft_len)[:conv_len]
        return float(conv.max())

    # Initial objective
    cur_max = max_convolution(a)
    cur_obj = 2.0 * n * cur_max
    best_seq = a.copy()
    best_obj = cur_obj

    print(f"[init] n={n}, initial objective={best_obj:.9f}")

    # ------------------------------------------------------------------
    # Hyper‑parameters
    step = 0.12            # initial step length
    step_inc = 1.02        # increase factor after a successful step
    step_dec = 0.7         # decrease factor after a rejected step
    min_step = 1e-6
    max_step = 1.0
    beta = 0.9             # momentum coefficient
    momentum = np.zeros_like(a)

    temperature = 0.02     # annealing temperature (absolute scale of obj)
    temp_decay = 0.9999    # geometric decay per iteration

    # Frequency of deterministic large transfers
    large_transfer_interval = 4000
    transfer_frac = 0.15   # fraction of the source mass moved in a large step

    it = 0
    last_improve = start_time

    while time.time() < deadline:
        # --------------------------------------------------------------
        # 1) convolution and identification of maximal indices
        fv = np.fft.rfft(a, fft_len)
        prod = fv * fv
        conv_full = np.fft.irfft(prod, fft_len)[:conv_len]
        max_val = float(conv_full.max())
        cur_obj = 2.0 * n * max_val

        # record improvement
        if cur_obj < best_obj - 1e-12:
            best_obj = cur_obj
            best_seq = a.copy()
            elapsed = time.time() - start_time
            print(f"[improve] iter={it}, obj={best_obj:.9f}, time={elapsed:.2f}s")
            last_improve = time.time()

        # --------------------------------------------------------------
        # 2) sub‑gradient of the maximal entry (or entries)
        #    g_i = Σ_{k∈max} a_{k-i}
        max_indices = np.flatnonzero(conv_full >= max_val - 1e-14)

        g = np.zeros_like(a)
        for k in max_indices:
            lo = max(0, k - (n - 1))
            hi = min(n - 1, k)
            idx = np.arange(lo, hi + 1)
            g[idx] += a[k - idx]

        # zero‑sum direction
        d = -g + g.mean()

        # --------------------------------------------------------------
        # 3) momentum blend
        momentum = beta * momentum + (1.0 - beta) * d

        # --------------------------------------------------------------
        # 4) try a projected step
        step_try = step
        accepted = False
        while step_try >= min_step:
            cand = a + step_try * momentum
            cand = project_to_simplex(cand)

            cand_max = max_convolution(cand)
            cand_obj = 2.0 * n * cand_max

            if cand_obj < cur_obj:                     # deterministic improvement
                a = cand
                cur_obj = cand_obj
                step = min(max_step, step_try * step_inc)
                accepted = True
                break
            # Metropolis acceptance (allows uphill moves while temperature > 0)
            delta = cand_obj - cur_obj
            if temperature > 0 and np.random.rand() < np.exp(-delta / temperature):
                a = cand
                cur_obj = cand_obj
                step = max(min_step, step_try * step_dec)
                accepted = True
                break

            step_try *= step_dec

        if not accepted:
            step = max(min_step, step * step_dec)

        # --------------------------------------------------------------
        # 5) occasional large deterministic mass transfer (helps escape flat
        #    plateaus).  We move a chunk from the index with the highest g
        #    (most responsible for the current peak) to the index with the
        #    smallest g (least responsible).
        if (it + 1) % large_transfer_interval == 0:
            i_src = int(np.argmax(g))
            i_dst = int(np.argmin(g))
            if i_src != i_dst and a[i_src] > 0.0:
                amount = transfer_frac * a[i_src]
                a[i_src] -= amount
                a[i_dst] += amount
                a = project_to_simplex(a)
                cur_obj = 2.0 * n * max_convolution(a)

        # --------------------------------------------------------------
        # 6) decay the temperature slowly
        temperature = max(1e-8, temperature * temp_decay)

        it += 1

    # ------------------------------------------------------------------
    total_time = time.time() - start_time
    print(f"[final] best objective={best_obj:.9f}, iterations={it}, time={total_time:.2f}s")
    # Return the best sequence as a plain list of floats.
    return best_seq.tolist()
