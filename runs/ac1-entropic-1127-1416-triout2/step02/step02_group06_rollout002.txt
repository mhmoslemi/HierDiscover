def propose_candidate(...):
    import time, numpy as np, math

    start=time.time()
    deadline = start + budget_s - 0.2

    # Determine n
    try:
        init_seq = np.asarray(height_sequence_1, dtype=np.float64)
        n = init_seq.shape[0]
    except Exception:
        n = 2500
        init_seq = None

    target_sum = math.sqrt(2.0 * n)

    if init_seq is not None and init_seq.shape[0]==n:
        f = np.clip(init_seq, 0.0, 1000.0)
        # project onto simplex with target_sum
        f = projection_simplex(f, target_sum)
    else:
        # random geometric seed
        r = 0.99
        base = np.power(r, np.arange(n))
        base = base / base.sum() * target_sum
        f = projection_simplex(base, target_sum)

    # compute initial score
    curr_score = _fast_score(f)
    best_f = f.copy()
    best_score = curr_score

    # SA parameters
    T0 = max(0.1, (curr_score - 1.4))  # maybe ~0.2 - 0.5
    Tmin = 1e-6
    max_move = target_sum * 0.05   # maximum mass moved in one step (5% of total)
    iteration = 0

    while time.time() < deadline:
        # update temperature
        elapsed = time.time() - start
        frac = elapsed / (budget_s - 0.2)
        T = T0 * (Tmin/T0) ** frac  # exponential cooling

        # propose move
        i_src = np.random.randint(n)
        if f[i_src] <= 1e-12:
            continue
        i_dst = np.random.randint(n-1)
        if i_dst >= i_src:
            i_dst += 1  # guarantee different

        # choose delta
        # delta random uniform up to min(f[i_src], max_move)
        delta = np.random.rand() * min(f[i_src], max_move)
        # apply
        f[i_src] -= delta
        f[i_dst] += delta

        # score
        new_score = _fast_score(f)
        delta_score = new_score - curr_score

        # accept?
        if delta_score <= 0 or np.random.rand() < math.exp(-delta_score / T):
            # accept
            curr_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_f = f.copy()
                print(f"[{time.time()-start:6.1f}s] iter {iteration} new best {best_score:.6f}")
        else:
            # revert
            f[i_src] += delta
            f[i_dst] -= delta

        iteration += 1

        # occasionally reproject to avoid numerical drift
        if iteration % 1000 == 0:
            f = projection_simplex(f, target_sum)

    final_score = evaluate_sequence(best_f.tolist())
    print(f"Search finished. Final score: {final_score:.6f}")
    return best_f.tolist()

"""
Propose a low‑score step‑function sequence.

The algorithm consists of three phases:

1. **Wide seeding** – generate many candidate vectors from simple families
   (geometric, power‑law, uniform) and from random phase spectra.
   Each candidate is projected onto the simplex {x≥0, Σx = √(2 n)}
   and evaluated with a fast FFT‑based convolution.  The best seed is kept.

2. **Simulated‑annealing hill climbing** – starting from the best seed we
   repeatedly move a random amount of mass from one index to another
   (or swap two entries).  The move is accepted if it improves the
   objective (the maximal self‑convolution) or with a temperature‑dependent
   probability, allowing occasional uphill steps.  Temperature and the
   maximum moved mass decay linearly with time.  The current best vector
   is stored.

3. **Linear‑programming refinement** – when the annealing phase finishes
   (≈ 20 % of the time budget remains) we solve the classic LP:
        maximise Σ g_i  subject to g_i ≥ 0, (f * g)_k ≤ M
   where M is the present maximal convolution value.  The optimal g is
   rescaled to the target ℓ₁‑norm and mixed with the current f.
   A few LP updates further lower the bound.

All numbers are kept non‑negative, ≤ 1000 and the ℓ₁‑norm is fixed to
√(2 n), so the evaluation function reduces to the maximal convolution entry.
The routine respects the 1000 s time budget and returns the best sequence
found.
"""

import time
import math
import numpy as np

# --------------------------------------------------------------------------- #
def projection_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Project vector v onto the simplex {x ≥ 0, Σx = s} (Duchi et al. 2008)."""
    if s < 0:
        raise ValueError("Simplex sum must be non‑negative")
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho_arr = np.where(u - cssv / (np.arange(n) + 1.0) > 0)[0]
    if rho_arr.size == 0:
        return np.zeros_like(v)
    rho = rho_arr[-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# --------------------------------------------------------------------------- #
def fast_score(seq: np.ndarray) -> float:
    """Return the objective 2·n·max(conv(seq,seq)) / (sum(seq)²) using FFT."""
    n = seq.size
    fft_len = 1 << ((2 * n - 1).bit_length())          # next power of two
    F = np.fft.rfft(seq, n=fft_len)
    conv = np.fft.irfft(F * F, n=fft_len)[: 2 * n - 1]
    max_conv = conv.max()
    s = seq.sum()
    # sum should equal √(2 n), therefore denominator ≈ 2 n.
    return float(2.0 * n * max_conv / (s * s))


# --------------------------------------------------------------------------- #
def _build_toeplitz_matrix(f: np.ndarray):
    """Return a (2n‑1) × n CSR matrix A with A·g = f * g."""
    n = f.shape[0]
    rows = np.empty(n * n, dtype=np.int32)
    cols = np.empty(n * n, dtype=np.int32)
    data = np.empty(n * n, dtype=np.float64)

    ptr = 0
    for i in range(n):
        rows[ptr: ptr + n] = np.arange(i, i + n)          # rows i … i+n‑1
        cols[ptr: ptr + n] = i
        data[ptr: ptr + n] = f
        ptr += n

    from scipy.sparse import csr_matrix
    return csr_matrix((data, (rows, cols)), shape=(2 * n - 1, n))


# --------------------------------------------------------------------------- #
def _random_phase_seed(n: int, target_sum: float) -> np.ndarray:
    """
    Generate a seed by assigning unit magnitude random phases to the
    discrete Fourier coefficients, inverting, clipping negatives and
    projecting onto the simplex.
    """
    # build a Hermitian‑symmetric spectrum of length n
    X = np.zeros(n, dtype=np.complex128)
    # DC term
    X[0] = 0.0
    half = n // 2
    # random phases for positive frequencies (excluding Nyquist if even)
    phi = np.random.uniform(0.0, 2.0 * np.pi, half - 1)
    X[1:half] = np.exp(1j * phi)
    X[-half + 1:] = np.conj(X[1:half][::-1])
    if n % 2 == 0:
        X[half] = 0.0                     # Nyquist term must be real
    # inverse FFT → real sequence (up to numerical noise)
    f = np.real(np.fft.ifft(X))
    # make non‑negative and obey the bound 0 ≤ f_i ≤ 1000
    f = np.clip(f, 0.0, 1000.0)
    # project onto the target ℓ₁‑norm simplex
    f = projection_simplex(f, target_sum)
    return f

    T_final = 1e-6
    max_move_start = target_sum * 0.20               # up to 20 % of total mass
    max_move_end = target_sum * 0.01                 # shrink to 1 % later
    it = 0
    projection_interval = 4000                       # re‑project every few thousand steps

    # Keep a small buffer for the LP phase (≈ 20 s)
    lp_buffer = 22.0

    while time.time() < deadline - lp_buffer:
        # linear temperature fraction
        elapsed = time.time() - start
        fraction = elapsed / (budget_s - lp_buffer - 0.2)
        T = T_start * (1.0 - fraction) + T_final * fraction

        # linearly decreasing maximal moved mass
        max_move = max_move_start * (1.0 - fraction) + max_move_end * fraction

        # ----- propose a move ------------------------------------------------
        if np.random.rand() < 0.68:          # mass‑shift move
            i_src = np.random.randint(n)
            if curr[i_src] <= 1e-12:        # nothing to give – skip
                continue
            i_dst = np.random.randint(n - 1)
            if i_dst >= i_src:
                i_dst += 1
            delta = np.random.rand() * min(curr[i_src], max_move)
            # apply
            curr[i_src] -= delta
            curr[i_dst] += delta
            revert = False
        else:                               # pure swap move (preserves ℓ₁‑norm)
            i_src = np.random.randint(n)
            i_dst = np.random.randint(n - 1)
            if i_dst >= i_src:
                i_dst += 1
            curr[i_src], curr[i_dst] = curr[i_dst], curr[i_src]
            delta = None
            revert = False

        # ----- evaluate -----
        new_score = fast_score(curr)
        delta_score = new_score - curr_score

        # acceptance test
        if delta_score <= 0.0 or np.random.rand() < math.exp(-delta_score / T):
            # accept
            curr_score = new_score
            if new_score < best_score - 1e-10:
                best_score, best = new_score, curr.copy()
                print(f"[{time.time() - start:6.1f}s] iter {it:07d} → new best {new_score:.6f}")
        else:
            # reject – revert the move
            if delta is None:          # swap
                curr[i_src], curr[i_dst] = curr[i_dst], curr[i_src]
            else:                       # shift
                curr[i_src] += delta
                curr[i_dst] -= delta

        it += 1

        # occasional re‑projection to avoid drift
        if it % projection_interval == 0:
            curr = projection_simplex(curr, target_sum)

    # --------------------------------------------------------------- #
    # 4. Linear‑programming refinement (if SciPy is present)
    try:
        from scipy.optimize import linprog
        from scipy.sparse import csr_matrix
    except Exception:      # SciPy not available – skip this phase
        linprog = None

    if linprog is not None and time.time() < deadline - 2.0:
        f = best.copy()
        fft_len = 1 << ((2 * n - 1).bit_length())
        lp_iters = 12
        for lp_it in range(lp_iters):
            if time.time() >= deadline - 2.0:
                break
            # current maximal convolution value M
            F = np.fft.rfft(f, n=fft_len)
            conv_f = np.fft.irfft(F * F, n=fft_len)[: 2 * n - 1]
            M = conv_f.max()

            # Build Toeplitz constraint matrix (f * g) ≤ M
            A = _build_toeplitz_matrix(f)
            b = np.full(2 * n - 1, M)

            # LP: maximise Σ g_i  ⇔ minimise -Σ g_i
            res = linprog(
                c=-np.ones(n),          # objective
                A_ub=A,
                b_ub=b,
                bounds=[(0, None)] * n,
                method='highs',
                options={'presolve': True, 'time_limit': 30}
            )
            if not res.success or res.x is None:
                break

            g = res.x
            sum_g = g.sum()
            if sum_g < 1e-12:
                break

            # rescale g to the target ℓ₁‑norm and mix
            g = g * (target_sum / sum_g)
            t_mix = 0.15
            f_new = (1.0 - t_mix) * f + t_mix * g
            f_new = projection_simplex(f_new, target_sum)

            # evaluate new candidate
            Fnew = np.fft.rfft(f_new, n=fft_len)
            conv_new = np.fft.irfft(Fnew * Fnew, n=fft_len)[: 2 * n - 1]
            new_score = float(2.0 * n * conv_new.max() / (f_new.sum() ** 2))

            if new_score < best_score - 1e-9:
                best_score, best = new_score, f_new.copy()
                print(f"[{time.time() - start:6.1f}s] LP‑iter {lp_it} → best {new_score:.6f}")

            # accept the mixed vector if it does not increase the bound
            if conv_new.max() <= M * 0.999:
                f = f_new
            else:
                break

    # --------------------------------------------------------------- #
    # 5. Final verification with the official evaluator
    final_score = evaluate_sequence(best.tolist())
    print(f"Search finished. Final score reported by evaluate_sequence: {final_score:.6f}")

    return best.tolist()
