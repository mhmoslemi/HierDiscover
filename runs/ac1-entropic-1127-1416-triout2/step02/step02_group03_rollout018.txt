"""
Simple‐grid optimisation for the step‑function problem.

The algorithm works in three stages:

1️⃣  Warm‑start – a cheap 4‑parameter power‑law family is random‑searched
    on a modest size (n≈2000).  The best candidate is kept.

2️⃣  Smoothing‑gradient descent (Adam) – the objective
    max(conv(a,a)) is approximated by a smooth log‑sum‑exp
    (parameter λ grows from 0.5 to ≈10).  The gradient of this
    surrogate is obtained by one FFT‑based convolution.
    Adam updates are projected onto the simplex
    {x≥0, Σx = √(2n)}.

3️⃣  Upscaling – the small‑n solution is interpolated to a larger
    dimension (n≈8000) and the same Adam‑with‑smoothing loop is
    continued using a larger λ (≈30).  The best sequence over the
    whole run is returned.

The code respects the 1000 s time limit, prints intermediate scores,
and finally returns a plain Python list suitable for ``evaluate_sequence``.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities
# ----------------------------------------------------------------------


def _project_simplex(v: np.ndarray, target_sum: float) -> np.ndarray:
    """Euclidean projection of v onto {x≥0, Σx = target_sum}
    (Duchi et al. 2008)."""
    if target_sum < 0:
        raise ValueError("target_sum must be non‑negative")
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - target_sum
    ind = np.arange(1, len(v) + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.zeros_like(v)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def _full_convolution_fft(a: np.ndarray) -> np.ndarray:
    """Linear convolution a*a using a real FFT (length 2·len(a)-1)."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, size)
    conv = np.fft.irfft(fa * fa, size)[: 2 * n - 1]
    return conv


def _max_conv(a: np.ndarray) -> float:
    """Maximum entry of the self‑convolution of a."""
    return _full_convolution_fft(a).max()


def _make_candidate(n: int, p: float, offset: float, beta: float) -> np.ndarray:
    """Generate a normalised power‑law candidate:
       a_i = (i+1+offset)^{-p} * (1 + beta·i/(n‑1)).
    """
    idx = np.arange(n, dtype=float)
    base = np.power(idx + 1.0 + offset, -p)
    factor = 1.0 + beta * idx / (n - 1)
    raw = np.maximum(base * factor, 0.0)
    target = math.sqrt(2.0 * n)
    return _project_simplex(raw, target)


def _gradient_smooth(a: np.ndarray, w: np.ndarray) -> np.ndarray:
    """
    Gradient of the smooth surrogate  f(a)= Σ_k w_k (a*a)[k] .
    Returns   g = ∂f/∂a  (vector of length len(a)).
    """
    n = a.shape[0]
    a_rev = a[::-1]                     # reverse for the anti‑diagonal sums
    size = 1 << ((n + w.size - 1).bit_length())
    fa_rev = np.fft.rfft(a_rev, size)
    fw = np.fft.rfft(w, size)
    conv = np.fft.irfft(fa_rev * fw, size)
    start = n - 1
    return 2.0 * conv[start : start + n]   # slice of length n


# ----------------------------------------------------------------------
# Main search routine
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 123, budget_s: int = 1000) -> list[float]:
    """
    Find a non‑negative step‑function with a low evaluation value.
    The routine performs a quick parametric sweep, a smoothed Adam
    descent on a small problem, then up‑scales and continues the
    descent on a larger problem.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 2.0   # safety margin

    # ------------------------------------------------------------------
    # 0. Problem sizes
    # ------------------------------------------------------------------
    n_small = 2000          # fast inner optimisation
    n_large = 8000          # final resolution
    target_small = math.sqrt(2.0 * n_small)
    target_large = math.sqrt(2.0 * n_large)

    # ------------------------------------------------------------------
    # 1. Warm‑start on the small instance (random 4‑parameter sweep)
    # ------------------------------------------------------------------
    #   If a previously found sequence exists we reuse it (down‑sampled).
    best_small = None
    hs = globals().get("height_sequence_1")
    if isinstance(hs, list) and len(hs) >= 2:
        # down‑sample / up‑sample to n_small
        arr = np.array(hs, dtype=float)
        if len(arr) != n_small:
            arr = np.interp(
                np.linspace(0, len(arr) - 1, n_small),
                np.arange(len(arr)),
                arr,
            )
        best_small = _project_simplex(arr, target_small)

    if best_small is None:
        # simple power‑law start
        best_small = _make_candidate(n_small, p=0.5, offset=0.0, beta=0.0)

    best_score_small = _max_conv(best_small)
    print(f"[init‑small] n={n_small}  M={best_score_small:.6f}")

    # random sweep – use ~15 % of the total budget
    sweep_deadline = start_time + 0.15 * (deadline - start_time)
    while time.time() < sweep_deadline:
        p = np.random.uniform(0.30, 0.70)
        off = np.random.uniform(0.0, 12.0)
        beta = np.random.uniform(-0.9, 2.5)
        cand = _make_candidate(n_small, p, off, beta)
        cand_score = _max_conv(cand)
        if cand_score < best_score_small:
            best_score_small = cand_score
            best_small = cand
            print(f"[sweep‑small] new M={best_score_small:.6f}  p={p:.3f} α={off:.3f} β={beta:.3f}")
        if best_score_small < 1.45:          # already very good – stop early
            break

    # ------------------------------------------------------------------
    # 2. Smoothed Adam descent on the small instance
    # ------------------------------------------------------------------
    a = best_small.copy()
    m = np.zeros_like(a)      # first moment
    v = np.zeros_like(a)      # second moment
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8
    lr0 = 0.6                 # base learning rate
    iter_small = 0
    max_iter_small = 25000    # hard cap – also limited by time

    while time.time() < deadline and iter_small < max_iter_small:
        iter_small += 1

        # full convolution (needed for the max and for the weights)
        conv = _full_convolution_fft(a)
        M = conv.max()

        # smooth max – λ grows linearly from 0.5 to 10
        lam = 0.5 + 9.5 * (iter_small / max_iter_small)
        # soft‑max weights (numerically stable)
        w = np.exp(lam * (conv - M))
        w /= w.sum()

        # gradient of the smooth surrogate
        grad = _gradient_smooth(a, w)

        # Adam update (step size decays like 1/√t)
        lr = lr0 / math.sqrt(iter_small)
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1 ** iter_small)
        v_hat = v / (1.0 - beta2 ** iter_small)
        step = lr * m_hat / (np.sqrt(v_hat) + eps)

        a = a - step
        a = _project_simplex(a, target_small)

        # keep the best over the whole run
        if M < best_score_small:
            best_score_small = M
            best_small = a.copy()
            print(f"[sgd‑small] iter {iter_small:5d}  new best M={best_score_small:.6f}")

        # tiny random jitter every few hundred steps to escape plateaus
        if iter_small % 500 == 0 and np.random.rand() < 0.2:
            jitter = (np.random.rand(n_small) - 0.5) * 0.02 * a
            a = a + jitter
            a = _project_simplex(a, target_small)

        # early stop if we already beat the target
        if best_score_small < 1.503:
            break

    # ------------------------------------------------------------------
    # 3. Upscale to the large instance
    # ------------------------------------------------------------------
    a_large = np.interp(
        np.linspace(0, n_small - 1, n_large),
        np.arange(n_small),
        best_small,
    )
    a_large = _project_simplex(a_large, target_large)
    best_large = a_large.copy()
    best_score_large = _max_conv(best_large)
    print(f"[init‑large] n={n_large}  M={best_score_large:.6f}")

    # ------------------------------------------------------------------
    # 4. Smoothed Adam descent on the large instance
    # ------------------------------------------------------------------
    a = best_large.copy()
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    lr0 = 0.5
    iter_large = 0
    max_iter_large = 60000   # we stop earlier if time runs out

    while time.time() < deadline and iter_large < max_iter_large:
        iter_large += 1

        conv = _full_convolution_fft(a)
        M = conv.max()

        # λ grows from 2 to 30 (more aggressive sharpening)
        lam = 2.0 + 28.0 * (iter_large / max_iter_large)
        w = np.exp(lam * (conv - M))
        w /= w.sum()

        grad = _gradient_smooth(a, w)

        lr = lr0 / math.sqrt(iter_large)
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1 ** iter_large)
        v_hat = v / (1.0 - beta2 ** iter_large)
        step = lr * m_hat / (np.sqrt(v_hat) + eps)

        a = a - step
        a = _project_simplex(a, target_large)

        if M < best_score_large:
            best_score_large = M
            best_large = a.copy()
            print(f"[sgd‑large] iter {iter_large:5d}  new best M={best_score_large:.6f}")

        if iter_large % 1000 == 0 and np.random.rand() < 0.15:
            jitter = (np.random.rand(n_large) - 0.5) * 0.015 * a
            a = a + jitter
            a = _project_simplex(a, target_large)

        if best_score_large < 1.503:
            break

    # ------------------------------------------------------------------
    # 5. Final report
    # ------------------------------------------------------------------
    total_iters = iter_small + iter_large
    print(f"Finished – total iterations {total_iters}, best M={best_score_large:.6f}")

    # Return a plain Python list as required by the evaluation harness
    return best_large.tolist()
