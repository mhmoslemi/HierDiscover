"""
Improved search for a low‑value step‑function.

The score to minimise is

    E(a) = 2·n·max(a∗a) / (sum(a)²),

which is scale‑invariant, so we restrict the search to the probability
simplex (∑a = 1, a_i ≥ 0).  The algorithm combines three ideas:

1. **SMOOTHED‑MAX GRADIENT DESCENT** – a fast FFT based projected
   gradient descent on the soft‑max surrogate of the max‑convolution.
   Several restarts with different initial shapes are performed.

2. **ADAM‑STYLE MOMENTUM** – the same surrogate is minimised with an
   Adam optimiser, which often reaches a lower value in fewer iterations.

3. **FAST INCREMENTAL HILL‑CLIMB** – after the continuous optimisation a
   specialised local search moves a small amount of mass from one index
   to another.  The convolution is updated analytically in O(n) using
   slice operations, so millions of moves can be attempted in the
   remaining time.  The step size is adapted and occasional non‑improving
   moves are accepted with a simple simulated‑annealing rule.

All heavy work is performed with NumPy (FFT in C), and the total
runtime never exceeds the supplied budget (default 1000 s).  The
function `propose_candidate` returns the best sequence found as a plain
list of floats.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Basic linear‑algebra utilities
# ----------------------------------------------------------------------


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / v.shape[0])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho = np.where(u - cssv / np.arange(1, len(v) + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full (linear) convolution using a real FFT."""
    m = len(a) + len(b) - 1
    n_fft = next_pow2(m)
    A = np.fft.rfft(a, n=n_fft)
    B = np.fft.rfft(b, n=n_fft)
    C = A * B
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


def compute_score(a: np.ndarray) -> float:
    """Score identical to `evaluate_sequence` when ∑a = 1."""
    n = len(a)
    conv = convolution_fft(a, a)
    return float(2.0 * n * np.max(conv) / (np.sum(a) ** 2))


# ----------------------------------------------------------------------
# Initial shapes
# ----------------------------------------------------------------------


def arcsine_initial(n: int) -> np.ndarray:
    """Arcsine‑shaped start: a_i ∝ 1 / sqrt((i+0.5)(n‑0.5‑i))."""
    i = np.arange(n, dtype=float)
    eps = 1e-9
    vals = 1.0 / np.sqrt((i + 0.5) * (n - 0.5 - i) + eps)
    return vals / np.sum(vals)


def uniform_initial(n: int) -> np.ndarray:
    """Uniform distribution."""
    return np.full(n, 1.0 / n)


def cosine_initial(n: int) -> np.ndarray:
    """Cosine‑shaped start: a_i ∝ sin(π·(i+0.5)/n)."""
    i = np.arange(n, dtype=float)
    vals = np.sin(np.pi * (i + 0.5) / n)
    vals = np.maximum(vals, 0.0)
    return vals / np.sum(vals)


def exponential_initial(n: int, lam: float = 0.001) -> np.ndarray:
    """Exponential decay a_i ∝ exp(-lam·i)."""
    i = np.arange(n, dtype=float)
    vals = np.exp(-lam * i)
    return vals / np.sum(vals)


# ----------------------------------------------------------------------
# Smoothed‑max surrogate and its gradient
# ----------------------------------------------------------------------


def smoothed_gradient(a: np.ndarray, alpha: float):
    """
    Gradient of the smoothed maximum
        f_α(a) = (1/α)·log Σ_k exp(α·(a∗a)_k)
    Returns (grad, conv) where conv = a∗a (full length 2n‑1).
    """
    n = len(a)
    conv = convolution_fft(a, a)                 # full convolution, length 2n‑1

    # soft‑max weights
    maxc = np.max(conv)
    exp_term = np.exp(alpha * (conv - maxc))
    w = exp_term / np.sum(exp_term)              # ∑ w = 1

    # grad = 2 * (a_rev * w) - diag(w) * a
    a_rev = a[::-1]
    conv_rev = convolution_fft(a_rev, w)         # length 3n‑2
    start = n - 1
    rev_slice = conv_rev[start:start + n]        # needed n entries

    w_diag = w[::2][:n]                          # w_{2i}
    grad = 2.0 * rev_slice - w_diag * a
    return grad, conv


def improve_smoothed(
    init_a: np.ndarray,
    deadline: float,
    max_iter: int = 120_000,
    eta0: float = 0.8,
    alpha_start: float = 1.0,
    alpha_end: float = 200.0,
    noise_every: int = 600,
    noise_scale: float = 5e-5,
):
    """
    Projected gradient descent on the smoothed surrogate.
    Returns (best_a, best_score).
    """
    a = project_to_simplex(init_a.astype(float).copy())
    best_a = a.copy()
    best_score = compute_score(a)

    n = len(a)
    for t in range(1, max_iter + 1):
        if time.time() >= deadline:
            break

        # linear schedule for smoothing parameter
        alpha = alpha_start + (alpha_end - alpha_start) * min(1.0, t / max_iter)

        grad, _ = smoothed_gradient(a, alpha)

        eta = eta0 / math.sqrt(t)
        a_new = a - eta * grad

        # occasional isotropic noise to help escape plateaus
        if noise_every and (t % noise_every == 0):
            a_new += np.random.randn(n) * noise_scale

        a_new = project_to_simplex(a_new)

        cur_score = compute_score(a_new)
        if cur_score < best_score:
            best_score = cur_score
            best_a = a_new.copy()

        a = a_new

        if t % 5000 == 0:
            print(f"    iter {t:,d}  cur_score {cur_score:.6f}  best {best_score:.6f}")

    return best_a, best_score

def improve_adam(
    init_a: np.ndarray,
    deadline: float,
    max_iter: int = 80_000,
    lr: float = 0.5,
    beta1: float = 0.9,
    beta2: float = 0.999,
    eps: float = 1e-8,
    alpha_start: float = 1.0,
    alpha_end: float = 250.0,
):
    """
    Adam on the smoothed‑max surrogate.
    Returns (best_a, best_score).
    """
    a = project_to_simplex(init_a.astype(float).copy())
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    best_a = a.copy()
    best_score = compute_score(a)

    n = len(a)
    for t in range(1, max_iter + 1):
        if time.time() >= deadline:
            break

        alpha = alpha_start + (alpha_end - alpha_start) * min(1.0, t / max_iter)

        grad, _ = smoothed_gradient(a, alpha)

        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)

        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)

        a_new = a - lr * m_hat / (np.sqrt(v_hat) + eps)
        a_new = project_to_simplex(a_new)

        cur_score = compute_score(a_new)
        if cur_score < best_score:
            best_score = cur_score
            best_a = a_new.copy()

        a = a_new

        if t % 4000 == 0:
            print(f"    Adam iter {t:,d}  cur_score {cur_score:.6f}  best {best_score:.6f}")

    return best_a, best_score

def hill_climb_incremental(start_seq, deadline, max_steps=50_000, init_step=0.08):
    """
    Stochastic pairwise mass transfer.
    The convolution is updated analytically so each step costs O(n).
    """
    a = np.array(start_seq, dtype=float)
    a = project_to_simplex(a)          # ensure we are on the simplex
    n = len(a)

    # initial full convolution
    conv = convolution_fft(a, a)
    cur_max = float(conv.max())
    cur_score = 2.0 * n * cur_max / (np.sum(a) ** 2)

    best_a = a.copy()
    best_conv = conv.copy()
    best_score = cur_score

    step = init_step
    # Simple simulated‑annealing schedule (temperature proportional to step)
    while max_steps > 0 and time.time() < deadline:
        # weighted choice for source index (more mass → higher chance)
        src = np.random.choice(n, p=a)          # source with probability ≈ its mass
        dst = np.random.randint(n - 1)
        if dst >= src:
            dst += 1

        delta = step * a[src]
        if delta <= 0.0:
            max_steps -= 1
            continue

        # ------- incremental update of the convolution -----------------
        # keep a copy of the old slice (required for the linear term)
        a_src = a[src]      # original value
        a_dst = a[dst]

        # copy of the current convolution
        conv_new = best_conv.copy()

        # linear term: 2*delta*(a*e_dst - a*e_src)
        conv_new[src:src + n] -= 2.0 * delta * best_a
        conv_new[dst:dst + n] += 2.0 * delta * best_a

        # quadratic term from (e_dst - e_src)^2
        conv_new[2 * src] += delta * delta
        conv_new[2 * dst] += delta * delta
        conv_new[src + dst] -= 2.0 * delta * delta
        # --------------------------------------------------------------

        # new a after the transfer
        a_new = best_a.copy()
        a_new[src] -= delta
        a_new[dst] += delta

        # no need to re‑project – the move preserves the sum and keeps
        # all entries non‑negative (we already checked delta ≤ a[src]).
        new_max = float(conv_new.max())
        new_score = 2.0 * n * new_max / (np.sum(a_new) ** 2)

        # Acceptance rule: always accept improvement, otherwise accept
        # with probability exp(-(new‑score‑cur)/T).  T is taken as `step`.
        improve = new_score < best_score
        if improve or (np.random.rand() < math.exp(-(new_score - best_score) / max(step, 1e-6))):
            best_a = a_new
            best_conv = conv_new
            best_score = new_score
            # slowly increase step after a success
            step = min(step * 1.03, 0.5)
        else:
            # shrink step after a rejection
            step *= 0.97

        max_steps -= 1

        if max_steps % 5000 == 0:
            print(f"    hill‑climb step {max_steps:,d}  best {best_score:.6f}")

    return best_a.tolist(), best_score

            # Quick hill‑climb on the result of the smoothed‑grad run
            if time.time() < deadline - 4.0:
                seq_ref, val_ref = hill_climb_incremental(
                    best_seq, deadline, max_steps=int(15_000 * 2048 / n), init_step=0.07
                )
                if val_ref < best_val:
                    best_val, best_seq = val_ref, seq_ref
                    print(f"      refined after hill‑climb = {best_val:.6f}")

            if time.time() >= deadline - 5.0:
                break

            # ----------------------------------------------------------------
            # Adam optimiser
            # ----------------------------------------------------------------
            print(f"  start {idx}/{len(inits)} – Adam")
            seq_adam, val_adam = improve_adam(
                init_a=init_vec,
                deadline=deadline,
                max_iter=int(80_000 * 2048 / n),
                lr=0.6,
                alpha_start=0.5,
                alpha_end=300.0,
            )
            print(f"      Adam best score = {val_adam:.6f}")
            if val_adam < best_val:
                best_val = val_adam
                best_seq = seq_adam.tolist()
                print(f"      *** New global best {best_val:.6f} (n={n}) ***")

            if time.time() < deadline - 4.0:
                seq_ref2, val_ref2 = hill_climb_incremental(
                    best_seq, deadline, max_steps=int(12_000 * 2048 / n), init_step=0.06
                )
                if val_ref2 < best_val:
                    best_val, best_seq = val_ref2, seq_ref2
                    print(f"      refined after Adam‑hill‑climb = {best_val:.6f}")

        # End of init loop for this n
    # ------------------------------------------------------------------
    # Final long hill‑climb on the best candidate found so far
    # ------------------------------------------------------------------
    if best_seq is not None and time.time() < deadline - 2.0:
        print("\n=== Final intensive hill‑climb on best candidate ===")
        refined_seq, refined_val = hill_climb_incremental(
            best_seq,
            deadline,
            max_steps=120_000,
            init_step=0.05,
        )
        if refined_val < best_val:
            best_val, best_seq = refined_val, refined_seq
            print(f"      final refined score = {best_val:.6f}")

    # ------------------------------------------------------------------
    # Safety net – if nothing was produced, fall back to a small uniform list
    # ------------------------------------------------------------------
    if best_seq is None:
        best_seq = [1.0] * 256
        best_val = evaluate_sequence(best_seq)  # noqa: F821

    # Final verification using the supplied evaluator (clipping, etc.)
    final_score = evaluate_sequence(best_seq)  # noqa: F821
    print(f"\nSearch finished. Reported final score = {final_score:.6f}")

    # The harness expects a plain Python list of floats
    return best_seq
