        a_fft = np.fft.rfft(a, n=N)
        conv = np.fft.irfft(a_fft * a_fft, n=N)[:2*n-1]
        max_c = conv.max()
        cur_score = 2 * n * max_c
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(...)

        # compute w (softmax)
        # subtract max_c to avoid overflow.
        exp_term = np.exp(alpha * (conv - max_c))

        # compute gradient via conv_rev
        a_rev = a[::-1]

        w_fft = np.fft.rfft(w, n=N)
        conv_rev_full = np.fft.irfft(w_fft * a_rev_fft, n=N)[:(2*n-1)+n-1]

        # exponentiated gradient update
        # Prevent too large exponent: clip grad maybe
        # compute update
        a_new_raw = a * np.exp(-eta * grad)
        # remove extremely small values to maintain numeric stability
        # ensure positivity.
        a_new = a_new_raw / a_new_raw.sum()

        # jitter occasionally

            break

"""
Propose a candidate step‑function using a smoothed‐gradient / exponentiated‑gradient
descent on the simplex.

The true objective is 2·n·max(a*a)/(sum a)^2.  Since scaling cancels out we work
with a probability vector (Σa = 1).  The non‑smooth “max’’ is replaced by a
log‑sum‑exp (soft‑max) with a temperature α that is slowly increased.
For a given α the smoothed objective
    F_α(a) = 2·n· (1/α)·log Σ_k exp(α·(a*a)_k)
is differentiable; its gradient can be written in closed form and is
evaluated efficiently by FFT‑based convolutions.
A mirror‑descent step with the KL‑divergence (exponentiated‑gradient update)
keeps the iterate in the probability simplex without an explicit projection.
A tiny random jitter is injected occasionally to escape flat regions.
The loop runs until the time budget expires (≈1000 s) and the best vector
encountered (according to the exact objective) is returned.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities
# ----------------------------------------------------------------------
def next_pow2(x: int) -> int:
    """Return the smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def simplex_projection(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of a vector onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1] - 1
    theta = cssv[cond][-1] / (rho + 1)
    return np.maximum(v - theta, 0.0)


def compute_objective(a: np.ndarray) -> float:
    """Exact objective 2·n·max(conv(a,a)) for a probability vector."""
    n = a.shape[0]
    L = 2 * n - 1
    N = next_pow2(L)
    a_fft = np.fft.rfft(a, n=N)
    conv = np.fft.irfft(a_fft * a_fft, n=N)[:L]
    return 2.0 * n * conv.max()


# ----------------------------------------------------------------------
# Main search routine
# ----------------------------------------------------------------------
def propose_candidate(budget_s: int = 1000) -> list:
    """
    Search for a low‑score non‑negative step‑function.
    The function respects the given time budget and returns the best
    sequence (as a plain Python list) found so far.
    """
    np.random.seed(0)                     # reproducibility
    deadline = time.time() + budget_s - 5.0   # safety margin

    # ------------------------------------------------------------------
    # 1. Choose problem size and initial point.
    # ------------------------------------------------------------------
    try:
        init = np.asarray(height_sequence_1, dtype=float)   # type: ignore[name-defined]
        n = init.shape[0]
    except Exception:

        init = np.full(n, 1.0 / n)

    # Normalise to a probability vector (Σa = 1) and ensure strict positivity.
    a = init.astype(np.float64)
    total = a.sum()
    if total <= 0:
        a = np.full(n, 1.0 / n)
    else:
        a = a / total
    # avoid exact zeros (exp may otherwise freeze a coordinate)
    eps = 1e-12
    a = np.maximum(a, eps)
    a = a / a.sum()

    # ------------------------------------------------------------------
    # 2. Hyper‑parameters
    # ------------------------------------------------------------------
    # smoothed‑max temperature
    alpha = 10.0          # start soft
    alpha_max = 500.0
    alpha_inc = 1.005     # multiplicative increase per iteration

    # exponentiated‑gradient step size (decreases like 1/√t)
    eta0 = 0.5

    best_a = a.copy()
    best_score = compute_objective(best_a)

    iteration = 0
    # FFT size that works for both the convolution (length 2n‑1)
    # and the gradient convolution (length 3n‑2).
    N = next_pow2(3 * n)

    while time.time() < deadline:
        # --------------------------------------------------------------
        # (a) exact objective for monitoring / best‑keeping
        # --------------------------------------------------------------
        # convolution a*a (full length L = 2n‑1)
        a_fft = np.fft.rfft(a, n=N)
        conv_full = np.fft.irfft(a_fft * a_fft, n=N)[: 2 * n - 1]
        max_conv = conv_full.max()
        cur_score = 2.0 * n * max_conv

        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"Iter {iteration:6d}: improved score = {best_score:.6f}")

        # --------------------------------------------------------------
        # (b) smoothed max – compute softmax weights w_k
        # --------------------------------------------------------------
        # w_k ∝ exp(α·(conv_k – max_conv)), ∑ w = 1
        # (subtract max_conv for numerical stability)

        # --------------------------------------------------------------
        # (c) gradient of the smoothed objective:

        # --------------------------------------------------------------

            break

    # ------------------------------------------------------------------
    return [float(v) for v in best_a.tolist()]
