"""
Propose a candidate step‑function by minimizing the supremum of its
self‑convolution.  The method augments the simple sub‑gradient scheme
used previously with a *smooth‑max* surrogate and an adaptive blend of
the exact sub‑gradient.  A projected (Euclidean) simplex projection keeps
the coefficients non‑negative and normalised (∑a_i = 1), so the
objective that evaluate_sequence returns reduces to 2·n·max(conv).

Algorithm outline
-----------------
1. Initialise a with the classic a_i ∝ 1/√(i+1) shape and normalise.
2. Loop until the time budget is exhausted:
   * Compute the full self‑convolution a∗a by FFT.
   * Build a smooth‑max weight vector  
         w_i ∝ exp( p·(conv_i−max(conv)) )
     with a temperature parameter p that grows slowly.
   * Obtain the gradient of the smooth‑max objective via a second FFT:
         g_smooth = 2·(w ∗ a_rev)[n−1 : n−1+n]
   * Compute the exact sub‑gradient at the current worst shift k_max.
   * Blend the two gradients:
         g = (1−α)·g_smooth + α·g_sub,
     where α ramps from 0 to 1 as the optimisation proceeds.
   * Take a diminishing step in the opposite gradient direction,
     normalise the step by the gradient norm, and project back onto the
     probability simplex.
   * Occasionally perform a tiny random mass redistribution (jitter)
     to escape shallow plateaux.
   * Keep the best sequence seen so far.
3. After the main loop, run a short pure sub‑gradient refinement
   (no smoothing) to polish the solution.
4. Return the best sequence as a plain Python list.

The code uses only NumPy, no external solvers, and respects the 1000 s
budget.  All helper functions are top‑level, contain no closures, and
the main routine prints progress and the final best score.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Helper functions (top‑level, no closures)
# ----------------------------------------------------------------------


def project_onto_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the probability simplex:
        { x | x >= 0, sum(x) = 1 }.
    Returns a new array (the input is not modified).
    """
    n = v.shape[0]
    # sort descending
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all entries are zero – return the uniform distribution
        return np.full_like(v, 1.0 / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def next_pow2(x: int) -> int:
    """Return the smallest power of two greater than or equal to x."""
    return 1 << (x - 1).bit_length()


def full_convolution_fft(u: np.ndarray, v: np.ndarray) -> np.ndarray:
    """
    Compute the full discrete convolution of two real vectors using FFT.
    Length of the result is len(u) + len(v) - 1.
    """
    n = len(u) + len(v) - 1
    fft_len = next_pow2(n)
    fu = np.fft.rfft(u, n=fft_len)
    fv = np.fft.rfft(v, n=fft_len)
    conv = np.fft.irfft(fu * fv, n=fft_len)
    return conv[:n]


def smooth_max_gradient(a: np.ndarray, conv: np.ndarray, p: float) -> np.ndarray:
    """
    Gradient of the smooth‑max surrogate
          f(a) = (1/p)·log ∑_k exp(p·conv_k)
    with respect to the coefficient vector a (∑a_i = 1).
    """
    n = len(a)
    # soft‑max weights (numerically stable)
    max_c = np.max(conv)
    w = np.exp(p * (conv - max_c))
    w_sum = np.sum(w)
    if w_sum == 0.0:
        w = np.full_like(w, 1.0 / w.size)
    else:
        w /= w_sum

    # gradient = 2 * (w ∗ a_rev)[n−1 : n−1+n]
    a_rev = a[::-1]
    conv_wr = full_convolution_fft(w, a_rev)               # length = 2n-1 + n -1 = 3n-2
    start = n - 1
    g = 2.0 * conv_wr[start:start + n]
    return g


def subgradient_max(a: np.ndarray, k_max: int) -> np.ndarray:
    """
    Sub‑gradient of the maximal convolution entry (at shift k_max)
    with respect to the coefficient vector a.
    """
    n = len(a)
    g = np.zeros_like(a)
    low = max(0, k_max - n + 1)
    high = min(k_max, n - 1)
    if low <= high:
        idx = np.arange(low, high + 1, dtype=int)
        g[idx] = a[k_max - idx]
    return g


def initialise_shape(n: int) -> np.ndarray:
    """
    Classic analytic start: a_i ∝ 1/√(i+1), normalised to sum = 1.
    """
    raw = 1.0 / np.sqrt(np.arange(n, dtype=float) + 1.0)
    a = raw / raw.sum()
    return a


# ----------------------------------------------------------------------
# Main search routine
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative height sequence that minimises

        2 * n * max(conv(a, a)) / (sum(a) ** 2)

    using a smooth‑max guided projected (sub‑)gradient descent.
    The function runs for at most `budget_s` seconds and returns the
    best list of floats found.
    """
    np.random.seed(seed)
    start_time = time.time()
    # reserve a few seconds for final printing / clean‑up
    deadline = start_time + budget_s - 5.0

    # ------------------------------------------------------------------
    # Parameters
    # ------------------------------------------------------------------
    n = 5000                  # length of the step‑function
    a = initialise_shape(n)   # start on the probability simplex (sum = 1)

    best_seq = a.copy()
    best_score = float('inf')

    # smoothing temperature (p) schedule
    p = 1.0
    p_max = 180.0             # final temperature – large enough to approximate max
    p_inc_per_iter = (p_max - p) / 200000   # slow linear increase

    # gradient‑blending schedule
    ramp_iters = 30000        # after this many iterations we rely fully on sub‑gradient

    # step‑size schedule
    lr0 = 0.6

    iteration = 0

    # ------------------------------------------------------------------
    # Main optimisation loop
    # ------------------------------------------------------------------
    while time.time() < deadline:
        # ---- convolution and worst shift --------------------------------
        conv = full_convolution_fft(a, a)      # a * a
        max_val = np.max(conv)
        k_max = int(np.argmax(conv))

        # ---- update smoothing temperature --------------------------------
        if p < p_max:
            p = min(p_max, p + p_inc_per_iter)

        # ---- gradients --------------------------------------------------
        g_smooth = smooth_max_gradient(a, conv, p)      # smooth‑max gradient
        g_sub = subgradient_max(a, k_max)               # exact sub‑gradient

        # blend the two gradients (α ramps from 0 to 1)
        alpha = min(1.0, iteration / ramp_iters)
        g = (1.0 - alpha) * g_smooth + alpha * g_sub

        # ---- step with diminishing length --------------------------------
        step = lr0 / math.sqrt(iteration + 1.0)
        # normalise gradient to avoid occasional huge jumps
        grad_norm = np.linalg.norm(g) + 1e-12
        a_candidate = a - step * (g / grad_norm)

        # ---- project back onto the simplex -------------------------------
        a = project_onto_simplex(a_candidate)

        # ---- evaluate current objective (sum(a) == 1) --------------------
        score = 2.0 * n * max_val
        if score < best_score:
            best_score = score
            best_seq = a.copy()
            print(f"Iter {iteration:06d}: improved score = {best_score:.6f} "
                  f"(p={p:.1f}, α={alpha:.3f}, max_shift={k_max})")
            # early exit if we already beat the target (optional)
            if best_score < 1.503:
                break

        # ---- occasional random jitter to escape plateaux -----------------
        if iteration % 250 == 0 and iteration > 0:
            i, j = np.random.randint(0, n, size=2)
            if i != j:
                # move a tiny random fraction of mass from i to j
                delta = 0.004 * a[i] * (np.random.rand() * 2.0 - 1.0)   # ±0.4 %
                cand = a.copy()
                cand[i] = max(cand[i] - delta, 0.0)
                cand[j] = cand[j] + delta
                cand = project_onto_simplex(cand)

                # evaluate jittered candidate (full conv needed)
                conv_cand = full_convolution_fft(cand, cand)
                cand_score = 2.0 * n * np.max(conv_cand)

                if cand_score < best_score:
                    best_score = cand_score
                    best_seq = cand.copy()
                    a = cand
                    print(f"  Jitter improvement: score = {best_score:.6f}")

        iteration += 1

    # ------------------------------------------------------------------
    # Short pure sub‑gradient polishing phase (if time permits)
    # ------------------------------------------------------------------
    while time.time() < deadline:
        conv = full_convolution_fft(a, a)
        max_val = np.max(conv)
        k_max = int(np.argmax(conv))
        g_sub = subgradient_max(a, k_max)

        # smaller step for polishing
        step = (lr0 * 0.25) / math.sqrt(iteration + 1.0)
        grad_norm = np.linalg.norm(g_sub) + 1e-12
        a_candidate = a - step * (g_sub / grad_norm)
        a = project_onto_simplex(a_candidate)

        score = 2.0 * n * max_val
        if score < best_score:
            best_score = score
            best_seq = a.copy()
            print(f"Polish {iteration:06d}: improved score = {best_score:.6f}")

        iteration += 1

    # ------------------------------------------------------------------
    # Fallback – use a previously stored construction if nothing was found
    # ------------------------------------------------------------------
    if best_seq is None or best_seq.size == 0:
        try:
            # height_sequence_1 is expected to be defined globally by the runner
            best_seq = np.array(height_sequence_1, dtype=float)
        except Exception:
            # ultimate fallback: uniform sequence (guaranteed finite score)
            best_seq = np.full(2000, 1.0 / 2000.0, dtype=float)

    print(f"Finished after {iteration} iterations – best score = {best_score:.6f} "
          f"(length {len(best_seq)})")
    # Return a plain Python list as required by the driver
    return best_seq.tolist()
