    n = len(a)
    conv = convolution_fft(a, a)  # length 2n-1
    c_max = np.max(conv)
    # stable softmax

    sum_exp = np.sum(exp_scaled)
    w = exp_scaled / sum_exp
    M_beta = c_max + np.log(sum_exp) / beta

    grad_slice = conv_rev[n-1:n-1+n]
    grad_f = 4 * n * grad_slice  # because grad_M = 2*grad_slice
    return score, grad_f

"""
Propose a low‑score non‑negative step function.

The algorithm consists of three stages:

1.  **Exploration** – generate many candidate vectors for a few
    lengths (256…2048) using simple parametric families (arcsine‑like,
    two‑parameter power‑law, Dirichlet draws).  The fast‐computed
    objective 2·n·max(a*a)/(sum a)² is used to keep the few most promising
    candidates.

2.  **Annealed Adam optimisation** – each retained candidate is refined
    by minimising a smooth surrogate of the maximal convolution:
        M_β(a) = max_k (a*a)_k   (smoothed by a log‑sum‑exp with parameter β).
    Gradients are obtained analytically via FFT‑based convolutions, the
    iterate stays on the probability simplex by Euclidean projection and the
    learning rate follows the Adam scheme.  β is increased linearly from a
    small value (≈0.5) to a large one (≈128) so that the optimisation first
    sees a smooth landscape and then focuses on the true maximum.

3.  **Final polishing** – the currently best vector is run through a longer
    optimisation with an even larger β (≈256) to push the real maximal
    convolution down as far as possible.

All heavy work is done with FFT‑based convolutions, which makes the method
fast enough to explore many iterations within the 1000 s budget.
"""

import time
import numpy as np
from typing import List, Tuple

# Low‑level utilities (top‑level, no closures)
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Linear convolution using a real FFT (fast, zero‑padded)."""
    a = np.asarray(a, dtype=float)
    b = np.asarray(b, dtype=float)
    m = a.size + b.size - 1
    n_fft = next_pow2(m)
    A = np.fft.rfft(a, n=n_fft)
    B = np.fft.rfft(b, n=n_fft)
    C = A * B
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the probability simplex
    {x | x_i ≥ 0 , Σ_i x_i = 1}.
    Implementation of Duchi et al. (2008).
    """
    v = np.asarray(v, dtype=float)
    n = v.shape[0]

    # Shortcut for the all‑zero case
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / n)

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho_arr = np.where(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho_arr.size == 0:
        theta = 0.0
    else:
        rho = rho_arr[-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def compute_score(a: np.ndarray) -> float:
    """
    Fast proxy for the official evaluator:
        2·n·max(a*a) / (sum(a)²)
    (a is assumed to be a probability vector – sum≈1).
    """
    n = a.size
    conv = convolution_fft(a, a)
    max_val = float(np.max(conv))
    s = float(np.sum(a))
    if s < 1e-12:
        return float("inf")
    return 2.0 * n * max_val / (s * s)


def arcsine_initial(n: int) -> np.ndarray:
    """
    Classical arcsine‑shaped start:
        a_i ∝ 1 / sqrt((i+½)*(n‑½‑i)).
    Normalised to unit L¹ norm.
    """
    i = np.arange(n, dtype=float)
    eps = 0.25
    vals = 1.0 / np.sqrt((i + 0.5) * (n - 0.5 - i) + eps)
    return vals / vals.sum()


def generate_shape(n: int, p: float, eps: float) -> np.ndarray:
    """
    Symmetric power‑law family:
        a_i ∝ ((i+eps)*(n‑1‑i+eps))⁻ᵖ .
    The vector is normalised to sum = 1.
    """
    i = np.arange(n, dtype=float)
    base = ((i + eps) * (n - 1 - i + eps)) ** (-p)
    total = base.sum()
    if total == 0.0:
        return np.full(n, 1.0 / n)
    return base / total


# ----------------------------------------------------------------------
# Annealed Adam optimiser for the smooth surrogate of max‑convolution
# ----------------------------------------------------------------------


def _smooth_objective_and_grad(a: np.ndarray, beta: float) -> Tuple[float, np.ndarray]:
    """
    Returns (smooth_score, gradient) for a given smoothing parameter β.

    The smooth score is 2·n·M_β where
        M_β = log‑sum‑exp_β (a*a) / β .
    The gradient is ∂(2·n·M_β)/∂a, projected onto the simplex is handled
    by the caller.
    """
    n = a.size
    conv = convolution_fft(a, a)                       # length 2n‑1
    c_max = np.max(conv)

    # stable softmax of β·conv
    scale = beta * (conv - c_max)                      # prevent overflow
    exp_scaled = np.exp(scale)
    sum_exp = np.sum(exp_scaled)
    w = exp_scaled / sum_exp                           # softmax weights, sum(w)=1

    # gradient of the smooth max:
    #   grad_i = 2 * Σ_k w_k * a_{k‑i}
    a_rev = a[::-1]
    conv_rev = convolution_fft(a_rev, w)               # length 3n‑2
    grad_slice = conv_rev[n - 1 : n - 1 + n]           # Σ_k w_k * a_{k‑i}
    grad_M = 2.0 * grad_slice

    smooth_score = 2.0 * n * (c_max + np.log(sum_exp) / beta)
    grad = 4.0 * n * grad_slice                         # 2·n·∂M/∂a = 4·n·grad_slice
    return smooth_score, grad


def annealed_adam(
    initial_a: np.ndarray,
    time_deadline: float,
    max_iter: int = 150_000,
    beta_start: float = 0.5,
    beta_end: float = 128.0,
    lr: float = 0.03,
    beta1: float = 0.9,
    beta2: float = 0.999,
    eps: float = 1e-8,
) -> Tuple[np.ndarray, float]:
    """
    Adam optimiser on the smooth surrogate of max‑convolution with a linear
    β‑annealing schedule.
    Returns the best vector seen (according to the fast proxy score) and its
    proxy score.
    """
    a = project_to_simplex(initial_a.astype(float).copy())
    n = a.size
    m = np.zeros_like(a)
    v = np.zeros_like(a)

    best_a = a.copy()
    best_score = compute_score(a)

    for t in range(1, max_iter + 1):
        if time.time() >= time_deadline:
            break

        # linear β schedule
        beta = beta_start + (beta_end - beta_start) * (t / max_iter)

        # smooth objective + gradient
        _, grad = _smooth_objective_and_grad(a, beta)

        # Adam moments
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)

        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)

        # Gradient step
        a = a - lr * m_hat / (np.sqrt(v_hat) + eps)

        # Projection onto the simplex (keeps a ≥ 0, Σ a = 1)
        a = project_to_simplex(a)

        # Occasionally evaluate the cheap proxy score and keep the best
        if t % 2000 == 0:
            cur = compute_score(a)
            if cur < best_score:
                best_score = cur
                best_a = a.copy()

    return best_a, best_score
