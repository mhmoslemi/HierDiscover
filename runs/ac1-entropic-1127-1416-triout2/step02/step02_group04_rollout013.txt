    # gradient descent
    a = best_seq.copy()
    prev_a = a.copy()
    eta = 0.9

    iteration = 0
    no_improve_iters = 0
    while time.time() < deadline and iteration < 500000:
        # compute conv etc
        fft = np.fft.rfft(a, n=2*n-1)
        conv = np.fft.irfft(fft*fft, n=2*n-1)
        max_b = conv.max()
        cur_score = 2.0 * n * max_b
        if cur_score < best_score - 1e-12:
            best_score = cur_score
            best_seq = a.copy()
            print(f"Iter {iteration}: new best {best_score:.6f}")
            no_improve_iters = 0
        else:
            no_improve_iters += 1

        eps = max_b * 1e-6
        active = np.where(conv >= max_b - eps)[0]
        if active.size == 0:
            active = np.array([conv.argmax()], dtype=int)

        g = np.zeros_like(a)
        for k in active:
            low = max(0, k - n + 1)
            high = min(n - 1, k)
            idx = np.arange(low, high+1)
            partner = k - idx
            g[idx] += 2.0 * a[partner]
        g /= float(active.size)

        step = eta / np.sqrt(iteration + 1)

        # momentum update
        a_new = a - step * g + mu * (a - prev_a)
        a_new = simplex_projection(a_new)

        a = a_new

"""
Propose a candidate sequence that minimizes the evaluation function.

Algorithm Overview
------------------
1. Determine the problem size ``n`` (length of the step function).  
2. Generate a diverse set of seed sequences:
   * deterministic ramps, uniform, simple exponential and polynomial shapes,
   * a grid of discretised Beta‑distributions,
   * a handful of random Dirichlet vectors.
   All seeds are normalised to sum to 1 and the best few (default 5) are kept.
3. Starting from the best seed, run a long projected (sub‑gradient) descent
   on the simplex:
   * The sub‑gradient is the average of the gradients of every convolution
     entry that is within a tiny relative tolerance of the current maximum.
   * A Nesterov‑style momentum term accelerates convergence.
   * After each step the iterate is projected back onto the probability simplex
     using the Euclidean projection of Duchi et al. (2008).
   * If no improvement is seen for many iterations a small random
     perturbation (noise) is added to escape possible plateaus.
4. The loop runs until the 1000 s budget is exhausted (or a target score
   ≤ 1.503 is reached).  The best sequence encountered is returned as a plain
   ``list[float]``.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper functions (all top‑level – no closures)
# ----------------------------------------------------------------------


def simplex_projection(v):
    """
    Euclidean projection of a vector onto the probability simplex:
    { w | w_i ≥ 0 , Σ w_i = 1 }.
    Implementation follows Duchi et al. (2008).
    """
    v = np.asarray(v, dtype=float)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all entries are non‑positive → uniform distribution
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1] - 1               # zero‑based index
    theta = cssv[cond][-1] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def compute_objective(a):
    """
    Return 2·n·max(a∗a) assuming Σ a_i = 1.
    This is exactly the value returned by ``evaluate_sequence`` after the
    normalisation step.
    """
    n = a.shape[0]
    # linear convolution using FFT (length 2·n‑1)
    fft = np.fft.rfft(a, n=2 * n - 1)
    conv = np.fft.irfft(fft * fft, n=2 * n - 1)
    return 2.0 * n * conv.max()


def beta_sequence(alpha, beta, n):
    """Discretised Beta(α,β) density on ``n`` mid‑points, normalised to sum 1."""
    x = (np.arange(n) + 0.5) / n
    logw = (alpha - 1.0) * np.log(x) + (beta - 1.0) * np.log(1.0 - x)
    logw = np.clip(logw, -700.0, 700.0)          # avoid overflow in exp
    w = np.exp(logw)
    total = w.sum()
    if total == 0 or not np.isfinite(total):
        w = np.ones_like(w)
        total = w.sum()
    return w / total


def exponential_shape(lam, n):
    """a_i ∝ exp(‑lam·i)  (lam ≥ 0), normalised to sum 1."""
    i = np.arange(n, dtype=float)
    w = np.exp(-lam * i)
    w /= w.sum()
    return w


def polynomial_shape(p, n):
    """a_i ∝ 1 - (i/(n‑1))**p  (p ≥ 0), clipped to non‑negative, normalised."""
    i = np.arange(n, dtype=float)
    frac = i / (n - 1)
    w = 1.0 - np.power(frac, p)
    w = np.clip(w, 0.0, None)
    w /= w.sum()
    return w


def subgradient_active(a, active):
    """
    Compute the averaged sub‑gradient of the maximum convolution entry
    over the supplied ``active`` index set.
    """
    n = a.shape[0]
    g = np.zeros_like(a)
    for k in active:
        low = max(0, k - n + 1)
        high = min(n - 1, k)
        idx = np.arange(low, high + 1, dtype=int)
        partner = k - idx
        g[idx] += 2.0 * a[partner]
    g /= float(active.size)
    return g


# ----------------------------------------------------------------------
# Main search routine
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 0, budget_s: int = 1000):
    """
    Search for a non‑negative step‑function (a list of floats) that minimises
    ``evaluate_sequence``.  The function respects the 1000 s time budget and
    returns the best sequence discovered.
    """
    np.random.seed(seed)
    start_time = time.time()
    # keep a 5‑second safety margin for the outer runner
    deadline = start_time + budget_s - 5.0

    # --------------------------------------------------------------
    # 1. Problem size
    # --------------------------------------------------------------
    try:
        n = len(height_sequence_1)   # type: ignore[name-defined]
    except NameError:
        n = 5146                     # fallback length from the prompt

    # --------------------------------------------------------------
    # 2. Helper to remember the globally best and a short “top‑K” list
    # --------------------------------------------------------------
    best_seq = None
    best_score = np.inf
    TOP_K = 5
    top = []                       # list of (score, seq)

    def consider(candidate):
        """Evaluate *candidate* and keep it if it improves the global or top‑K list."""
        nonlocal best_seq, best_score, top
        s = compute_objective(candidate)
        if s < best_score:
            best_score = s
            best_seq = candidate.copy()
            print(f"New global best (init) : {s:.6f}")
        # maintain the top‑K list (sorted by score, smallest first)
        if len(top) < TOP_K:
            top.append((s, candidate.copy()))
            top.sort(key=lambda x: x[0])
        else:
            if s < top[-1][0]:
                top[-1] = (s, candidate.copy())
                top.sort(key=lambda x: x[0])

    # --------------------------------------------------------------
    # 3. Deterministic seed families
    # --------------------------------------------------------------
    # Linear decreasing ramp (the analytic construction used previously)
    a_ramp = np.arange(n, 0, -1, dtype=float)
    a_ramp = 2.0 * a_ramp / (n * (n + 1))
    a_ramp /= a_ramp.sum()
    consider(a_ramp)

    # Uniform distribution
    a_uniform = np.full(n, 1.0 / n)
    consider(a_uniform)

    # Exponential family (λ from 0.001 to 5, 20 values)
    for lam in np.linspace(0.001, 5.0, 20):
        a_exp = exponential_shape(lam, n)
        consider(a_exp)

    # Polynomial family (p ∈ {0.5,1,1.5,2,3})
    for p in [0.5, 1.0, 1.5, 2.0, 3.0]:
        a_poly = polynomial_shape(p, n)
        consider(a_poly)

    # --------------------------------------------------------------
    # 4. Systematic Beta‑distribution sweep (coarse grid)
    # --------------------------------------------------------------
    alphas = np.linspace(0.2, 5.0, 12)
    betas = np.linspace(0.2, 5.0, 12)
    for α in alphas:
        for β in betas:
            if time.time() >= deadline:
                break
            a_beta = beta_sequence(α, β, n)
            consider(a_beta)
        if time.time() >= deadline:
            break

    # --------------------------------------------------------------
    # 5. Randomised seeds (Dirichlet‑like and random Beta)
    # --------------------------------------------------------------
    for _ in range(200):
        if time.time() >= deadline:
            break
        # Dirichlet‑like
        conc = np.random.uniform(0.1, 3.0, size=n)
        a_dir = np.random.gamma(conc, 1.0)
        a_dir /= a_dir.sum()
        consider(a_dir)

        # Random Beta parameters
        α = np.random.uniform(0.05, 5.0)
        β = np.random.uniform(0.05, 5.0)
        a_beta = beta_sequence(α, β, n)
        consider(a_beta)

    # --------------------------------------------------------------
    # 6. Convex combinations of the current top candidates
    #    (helps to explore the neighbourhood of good points)
    # --------------------------------------------------------------
    for _ in range(150):
        if time.time() >= deadline:
            break
        # pick two different seeds from the top‑K list
        i, j = np.random.choice(len(top), 2, replace=False)
        w = np.random.rand()
        a_mix = w * top[i][1] + (1.0 - w) * top[j][1]
        a_mix /= a_mix.sum()          # keep a proper probability vector
        consider(a_mix)

    # --------------------------------------------------------------
    # 7. Projected (sub‑gradient) descent with momentum
    # --------------------------------------------------------------
    # Start from the globally best seed discovered so far
    a = best_seq.copy()
    prev_a = a.copy()
    eta = 0.85                # base step‑size
    mu = 0.55                 # momentum coefficient
    iteration = 0
    stagnation = 0           # number of iterations without improvement
    max_iter = 500000        # hard cap (won’t be reached if time runs out)

    while time.time() < deadline and iteration < max_iter:
        # ----- convolution and max entry -----
        fft = np.fft.rfft(a, n=2 * n - 1)
        conv = np.fft.irfft(fft * fft, n=2 * n - 1)
        max_b = conv.max()
        cur_score = 2.0 * n * max_b

        # keep the best ever seen
        if cur_score < best_score - 1e-12:
            best_score = cur_score
            best_seq = a.copy()
            print(f"Iter {iteration:6d}: new best {best_score:.6f}")
            stagnation = 0
        else:
            stagnation += 1

        # ----- active set (indices within a tiny relative tolerance) -----
        eps = max_b * 1e-6
        active = np.where(conv >= max_b - eps)[0]
        if active.size == 0:          # safety fallback
            active = np.array([conv.argmax()], dtype=int)

        # ----- sub‑gradient -----
        g = subgradient_active(a, active)

        # ----- step‑size (classical 1/√t schedule) -----
        step = eta / np.sqrt(iteration + 1)

        # ----- Nesterov–style momentum update + projection -----
        a_next = a - step * g + mu * (a - prev_a)
        a_next = simplex_projection(a_next)

        prev_a, a = a, a_next
        iteration += 1

        # ----- occasional random perturbation to escape plateaus -----
        if stagnation >= 20000:
            # magnitude of the perturbation slowly decays with time
            sigma = max(0.01, 0.5 / np.sqrt(iteration + 1))
            noise = sigma * np.random.randn(n) * a
            a = a + noise
            a = np.clip(a, 0.0, None)
            a /= a.sum()
            prev_a = a.copy()
            stagnation = 0
            print(f"Iter {iteration:6d}: perturbation applied, current {cur_score:.6f}")

        # early exit if we have already hit the target (1.503)
        if best_score <= 1.503:
            print(f"Target reached at iteration {iteration}, score = {best_score:.6f}")
            break

    # --------------------------------------------------------------
    # 8. Return the best sequence discovered
    # --------------------------------------------------------------
    print(f"Finished after {iteration} iterations, best score = {best_score:.6f}")
    # Convert to plain Python list of floats as required by the external driver.
    return [float(v) for v in best_seq.tolist()]
