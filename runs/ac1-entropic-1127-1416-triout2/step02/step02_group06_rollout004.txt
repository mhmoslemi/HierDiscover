"""
Propose a step‑function (non‑negative vector) that minimizes the
``evaluate_sequence`` score.

The algorithm works in three phases:

1. **Broad seeding** – many simple families (geometric, power‑law,
   linear, triangular, Gaussian‑like) are generated for a chosen length
   `n` (default 5000).  Each candidate is projected onto the simplex
   Σa_i = √(2 n) and scored quickly with an FFT‑based convolution.

2. **Active‑set projected sub‑gradient descent** – starting from the
   best seed, the convolution is recomputed each iteration, the active
   set of (near‑maximal) indices is identified, and the sub‑gradient
   of the max‑functional is assembled.  An Euclidean‑projected
   gradient step with Nesterov‑type momentum is taken; occasional
   random jitter helps escape plateaus.  The step size is decayed
   slowly and the best vector seen so far is retained.

3. **Final check** – after the time budget expires the best vector is
   evaluated with the official ``evaluate_sequence`` function and
   returned.

The code respects the security checks performed by
``evaluate_sequence`` (non‑negative, ≤ 1000).  It uses only a single
CPU and runs for at most the supplied time budget (≈ 1000 s).
"""

import time
import numpy as np
from typing import List

# --------------------------------------------------------------------------- #
def projection_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """
    Project `v` onto the simplex {x ≥ 0, Σx = s}.
    Implements the algorithm of Duchi et al. (2008).
    """
    if s < 0:
        raise ValueError("Simplex sum must be non‑negative")
    n = v.shape[0]
    u = np.sort(v)[::-1]                     # descending sort
    cssv = np.cumsum(u) - s
    rho_arr = np.where(u - cssv / (np.arange(n) + 1.0) > 0)[0]
    if rho_arr.size == 0:
        return np.zeros_like(v)
    rho = rho_arr[-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# --------------------------------------------------------------------------- #
def _fast_score(seq: np.ndarray, fft_len: int) -> float:
    """
    Return the maximal value of the self‑convolution of `seq`.
    For a sequence normalised to Σa_i = √(2 n) the returned value is
    exactly the score reported by ``evaluate_sequence``.
    """
    F = np.fft.rfft(seq, n=fft_len)
    conv = np.fft.irfft(F * F, n=fft_len)[: 2 * seq.size - 1]
    return float(conv.max())

    """
    Search for a low‑score non‑negative step‑function.

    Parameters
    ----------
    seed : int
        Random seed for reproducibility.
    budget_s : float
        Time budget in seconds (default 1000 s).

    Returns
    -------
    List[float]
        The best sequence discovered (as a plain Python list).
    """
    np.random.seed(seed)
    start = time.time()
    deadline = start + budget_s - 0.3               # safety margin

    # ------------------------------------------------------------------- #
    # Problem size – a few thousand entries give a good trade‑off.
    # Larger n often yields a lower max‑convolution, but we keep n moderate
    # to stay far from the memory limits of the LP refinement.
    n = 5000
    target_sum = np.sqrt(2.0 * n)                  # Σa_i = √(2 n)

    # FFT length – next power of two ≥ 2·n‑1 (shared across all calls)
    fft_len = 1 << ((2 * n - 1).bit_length())

    # ------------------------------------------------------------------- #
    # 1. Generate a variety of seeds.
    seeds = []

    # ---- geometric / exponential families (lam = 1‑r) ---------------- #
    # lam ∈ [1e‑6, 0.3] gives r in (0.7, 0.999999)
    for lam in np.logspace(-6, -0.2, 40):
        r = 1.0 - lam
        r_n = r ** n
        denom = 1.0 - r_n
        if denom <= 0.0:
            continue
        C = target_sum * lam / denom
        a = C * (r ** np.arange(n))
        # Clip extreme values before projection (helps numerical stability)
        a = np.clip(a, 0.0, 1000.0)
        seeds.append(a)

    # ---- power‑law families (a_i ∝ (i+1)^‑p) ------------------------ #
    for p in np.linspace(0.2, 1.4, 30):
        base = 1.0 / np.power(np.arange(1, n + 1, dtype=np.float64), p)
        C = target_sum / base.sum()
        a = C * base
        a = np.clip(a, 0.0, 1000.0)
        seeds.append(a)

    # ---- linear decreasing (a_i ∝ n‑i) -------------------------------- #
    lin = np.arange(n, 0, -1, dtype=np.float64)          # n, n‑1, …, 1
    C = target_sum / lin.sum()
    a = C * lin
    a = np.clip(a, 0.0, 1000.0)
    seeds.append(a)

    # ---- triangular (symmetric, peak in the centre) ------------------- #
    tri = np.minimum(np.arange(n) + 1, n - np.arange(n))  # 1,2,…,⌈n/2⌉,…,1
    C = target_sum / tri.sum()
    a = C * tri
    a = np.clip(a, 0.0, 1000.0)
    seeds.append(a)

    # ---- half‑Gaussian (a_i ∝ exp(‑(i/σ)²)) --------------------------- #
    for sigma in np.logspace(0.0, 3.0, 20):               # σ from 1 to 1000
        base = np.exp(- (np.arange(n, dtype=np.float64) / sigma) ** 2)
        if base.sum() == 0:
            continue
        C = target_sum / base.sum()
        a = C * base
        a = np.clip(a, 0.0, 1000.0)
        seeds.append(a)

    # ---- random simplexes (exponential Dirichlet) -------------------- #
    for _ in range(12):
        rand = np.random.exponential(scale=1.0, size=n)
        a = projection_simplex(rand, target_sum)
        seeds.append(a)

    # ---- use the globally supplied sequence, if it matches length ---- #
    try:
        global height_sequence_1
        if isinstance(height_sequence_1, (list, tuple, np.ndarray)):
            hs = np.asarray(height_sequence_1, dtype=np.float64)
            if hs.shape[0] == n:
                hs = np.clip(hs, 0.0, 1000.0)
                hs = projection_simplex(hs, target_sum)
                seeds.append(hs)
    except Exception:
        pass

    # ------------------------------------------------------------------- #
    # 2. Scan seeds – keep the best (exact ℓ₁‑norm enforced).
    best_seq = None
    best_score = float("inf")
    for idx, cand in enumerate(seeds):
        if time.time() >= deadline:
            break
        cand = projection_simplex(cand, target_sum)
        sc = _fast_score(cand, fft_len)
        if sc < best_score:
            best_score, best_seq = sc, cand.copy()
            print(f"[{time.time() - start:6.1f}s] seed {idx:02d} → score {sc:.6f}")

    # If for some reason we have no seed (should not happen), fall back
    if best_seq is None:
        best_seq = np.full(n, target_sum / n)

    # ------------------------------------------------------------------- #
    # 3. Active‑set projected sub‑gradient descent with momentum.
    curr = best_seq.copy()
    best = curr.copy()
    best_score = _fast_score(best, fft_len)

    velocity = np.zeros_like(curr)
    # initial absolute step (roughly 1 % of the average entry)
    step = 0.7 * target_sum / n
    min_step = 1e-9
    momentum = 0.9
    it = 0
    rel_tol = 1e-4                      # tolerance for the active set

    while time.time() < deadline:
        # ----- convolution (FFT) ---------------------------------------
        F = np.fft.rfft(curr, n=fft_len)
        conv = np.fft.irfft(F * F, n=fft_len)[: 2 * n - 1]
        max_val = conv.max()
        score = float(max_val)           # because Σa_i = √(2 n) ⇒ score = max

        # ----- record improvement --------------------------------------
        if score < best_score - 1e-9:
            best_score, best = score, curr.copy()
            print(f"[{time.time() - start:6.1f}s] iter {it:7d} → new best {score:.6f}")
            # tighten step when we make progress
            step = max(step * 0.9, min_step)

        # ----- active set ---------------------------------------------
        tol_abs = max_val * rel_tol
        active_idx = np.where(conv >= max_val - tol_abs)[0]

        # ----- sub‑gradient of the max‑functional ----------------------
        grad = np.zeros_like(curr)
        idx_arr = np.arange(n)
        for k in active_idx:
            j = k - idx_arr
            mask = (j >= 0) & (j < n)
            grad[mask] += 2.0 * curr[j[mask]]
        gnorm = np.linalg.norm(grad)
        if gnorm == 0.0:
            break

        # ----- Nesterov‑type momentum step ----------------------------
        velocity = momentum * velocity - step * (grad / gnorm)
        tentative = curr + velocity
        tentative = np.clip(tentative, 0.0, 1000.0)
        curr = projection_simplex(tentative, target_sum)

        # ----- occasional jitter (helps escaping flat plateaus) ------
        if np.random.rand() < 0.015:
            jitter = (np.random.randn(n) *
                      step * target_sum / (30.0 * n))
            curr = projection_simplex(
                np.clip(curr + jitter, 0.0, 1000.0), target_sum)

        it += 1

        # ----- step‑size decay -----------------------------------------
        if it % 500 == 0:
            step = max(step * 0.85, min_step)

        # ----- random restart if stagnated for a long while -----------
        if it % 5000 == 0 and best_score - score > -1e-6:
            # add a modest random perturbation to current iterate
            pert = np.random.randn(n) * step * 0.5
            curr = projection_simplex(
                np.clip(best + pert, 0.0, 1000.0), target_sum)

    # ------------------------------------------------------------------- #
    # 4. Final sanity check with the official evaluator.
    final_score = evaluate_sequence(best.tolist())
    print(f"Search finished. Final score reported by evaluate_sequence: {final_score:.6f}")

    return best.tolist()
