def propose_candidate(seed=42, budget_s=1000):
    np.random.seed(seed)
    start = time.time()
    deadline = start + budget_s - 5.0

    # Choose length based on budget for decent performance.
    n = 12000   # or 10000. Let's choose 12000.

    # Initialize shape: try a few p values to get best initial.
    best_init_seq = None; best_init_score = float('inf')
    for p in np.linspace(0.2, 0.8, 7):
        a_raw = (np.arange(1, n+1, dtype=float) ** (-p))
        a_raw = a_raw / a_raw.sum()  # normalize sum to 1
        # evaluate
        score = evaluate_sequence(a_raw.tolist())
        if score < best_init_score:
            best_init_score = score
            best_init_seq = a_raw.copy()
    if best_init_seq is None:
        # fallback uniform
        a = np.full(n, 1.0/n)
    else:
        a = best_init_seq

    # define parameters
    step0 = 0.5
    beta = 0.9
    # momentum vector
    m = np.zeros_like(a)
    # keep track best sequence
    best_seq = a.copy()
    best_score = best_init_score

    iter = 0
    top_k = 15
    jitter_interval = 300
    # main loop
    while time.time() < deadline:
        # compute conv
        conv = full_convolution(a)
        max_val = conv.max()
        score = 2.0 * n * max_val
        if score < best_score:
            best_score = score
            best_seq = a.copy()
            print(f"Iter {iter:06d}: new best score {best_score:.6f} (max shift {int(np.argmax(conv))})")

            # early exit condition
            if best_score < 1.503:
                break

        # compute top K shift indices
        if top_k >= conv.size:
            top_idxs = np.arange(conv.size)
        else:
            # partial sort
            top_idxs = np.argpartition(-conv, top_k)[:top_k]

        # compute gradient (sum of gradients for top shifts)
        g = np.zeros_like(a)
        for k in top_idxs:
            low = max(0, k - n + 1)
            high = min(k, n - 1)
            if low > high:
                continue
            idx = np.arange(low, high + 1, dtype=int)
            g[idx] += a[k - idx]

        # momentum update
        m = beta * m + (1 - beta) * g  # simple momentum

        # step size schedule
        step = step0 / np.sqrt(iter + 1.0)

        # update candidate using Euclidean projection
        a_candidate = a - step * m
        a_candidate = project_onto_simplex(a_candidate)

        # optionally evaluate candidate (not necessary but we need for next iteration)
        a = a_candidate

        # occasional random jitter
        if (iter % jitter_interval) == 0 and iter > 0:
            # pick two indices
            i, j = np.random.randint(0, n, size=2)
            if i != j:
                eps = 0.01  # max 1% mass shift
                delta = eps * a[i] * (np.random.rand() * 2 - 1)  # small random fraction
                cand = a.copy()
                cand[i] = max(cand[i] - delta, 0.0)
                cand[j] = cand[j] + delta
                cand = project_onto_simplex(cand)
                cand_score = 2.0 * n * full_convolution(cand).max()
                if cand_score < best_score:
                    best_score = cand_score
                    best_seq = cand.copy()
                    a = cand
                    print(f"Iter {iter:06d}: jitter improved to {best_score:.6f}")

        iter += 1

    # fallback if nothing found
    if best_seq is None or best_seq.size == 0:
        try:
            best_seq = np.array(height_sequence_1, dtype=float)
        except Exception:
            best_seq = np.full(2000, np.sqrt(2.0 * 2000) / 2000, dtype=float)

    print(f"Finished after {iter} iterations, best score {best_score:.6f}, length {len(best_seq)}")
    return best_seq.tolist()

Propose a candidate step‑function using an adaptive sub‑gradient method.

Key ideas
---------
* Work on a relatively large length (default 12 000).  The constant in the
  objective 2·n·max(conv)/sum² is almost independent of *n*, so a larger
  *n* allows a finer discretisation of the optimal continuous shape.
* Initialise the vector with a 1/(i+1) ᵖ shape; a few values of the exponent
  *p* are tried and the best one (according to ``evaluate_sequence``) is
  kept as a warm start.
* At each iteration the full self‑convolution is computed by an FFT.
  Instead of using only the single maximal entry, the gradient is formed
  from the *K* largest convolution values (``top_k``).  This yields a
  smoother descent direction and avoids getting stuck on plateaus.
* Momentum (exponential moving average) is applied to the gradient
  (parameter ``beta``) and the step size decays like 1/√iteration.
* After the Euclidean step the vector is projected back onto the
  probability simplex (non‑negative, sum = 1) using a standard O(n log n)
  projection.
* Every ``jitter_interval`` iterations a tiny random redistribution of mass
  is tried to escape shallow local minima.
* The best sequence seen so far (according to the true ``evaluate_sequence``
  function) is returned.  If something goes wrong the previously known
  global ``height_sequence_1`` is used as a safe fallback.

The routine respects a user‑supplied time budget (default 1000 s) and
always finishes before the deadline.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper functions (all top‑level, no closures)
# ----------------------------------------------------------------------


def project_onto_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the probability simplex:
        { x | x >= 0, sum(x) = 1 }.
    Returns a new array (the input is not modified).
    """
    n = v.shape[0]
    # sort descending
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all entries are zero – return the uniform distribution
        return np.full_like(v, 1.0 / n, dtype=float)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def full_convolution(a: np.ndarray) -> np.ndarray:
    """
    Compute the full discrete convolution a * a using an FFT.
    Length of the result is 2*len(a) - 1.
    """
    n = a.shape[0]
    size = 2 * n
    fft_len = 1 << (size - 1).bit_length()          # next power of two
    fft_a = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(fft_a * fft_a, n=fft_len)
    return conv[: 2 * n - 1]


def gradient_from_top_shifts(a: np.ndarray, top_idxs: np.ndarray) -> np.ndarray:
    """
    Compute a (sub‑)gradient that is the sum of the gradients of the
    convolution entries indexed by ``top_idxs``.
    """
    n = a.shape[0]
    g = np.zeros_like(a)
    for k in top_idxs:
        low = max(0, k - n + 1)
        high = min(k, n - 1)
        if low > high:
            continue
        idx = np.arange(low, high + 1, dtype=int)
        g[idx] += a[k - idx]
    return g

# Main search routine
# ----------------------------------------------------------------------

        2 * n * max(conv(a, a)) / (sum(a) ** 2)

    and returns the best list of floats found.
    """

    start_time = time.time()
    # reserve a few seconds for final printing / cleanup

    # --------------------------------------------------------------

    # If something failed we fall back to the classic 1/√i shape.
