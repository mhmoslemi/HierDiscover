import math
import numpy as np
from scipy.optimize import linprog

# ----------------------------------------------------------------------
# Helper utilities (all top‑level, no closures)
# ----------------------------------------------------------------------


def _project_simplex(v: np.ndarray, target_sum: float) -> np.ndarray:
    """Euclidean projection of v onto {w | w_i ≥ 0, Σ w_i = target_sum}."""
    if target_sum < 0:
        raise ValueError("target_sum must be non‑negative")
    u = np.sort(v)[::-1]                # sorted descending
    cssv = np.cumsum(u) - target_sum
    ind = np.arange(1, len(v) + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.zeros_like(v)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def _full_convolution_fft(a: np.ndarray) -> np.ndarray:
    """Linear convolution a*a (full length 2·len(a)-1) via real FFT."""
    n = len(a)
    size = 1 << ((2 * n - 1).bit_length())      # next power of two
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return conv


def _max_conv(a: np.ndarray) -> float:
    """Maximum entry of the self‑convolution of a."""
    return _full_convolution_fft(a).max()


def _make_candidate(
    n: int, p: float, offset: float, beta: float
) -> np.ndarray:
    """
    Build a raw power‑law candidate

        a_i = (i+1+offset)^(-p) * (1 + beta * i/(n‑1))

    and project it onto the simplex Σa_i = √(2 n).
    """
    idx = np.arange(n, dtype=float)
    base = np.power(idx + 1.0 + offset, -p)
    factor = 1.0 + beta * idx / (n - 1)
    raw = base * factor
    raw = np.maximum(raw, 0.0)               # protect against tiny negatives
    target = math.sqrt(2.0 * n)
    return _project_simplex(raw, target)


def _subgradient_step(
    a: np.ndarray,
    max_inds: np.ndarray,
    step: float,
    target_sum: float,
) -> np.ndarray:
    """
    Perform one (projected) sub‑gradient step for the objective
    max(conv(a,a)).  The gradient is accumulated only over indices
    where the current maximum is attained.
    """
    n = len(a)
    g = np.zeros_like(a)

    for k in max_inds:
        if k < n:
            # i runs from 0 … k  (the slice a[0:k+1] is reversed)
            g[: k + 1] += a[k :: -1]
        else:
            start = k - (n - 1)
            # i runs from start … n‑1, matched with a[k‑i]
            g[start:] += a[k - np.arange(start, n)]

    cand = a - step * g
    cand = _project_simplex(cand, target_sum)
    return cand


def _random_sweep(
    n: int,
    target_sum: float,
    time_limit: float,
    best_seq: np.ndarray,
    best_score: float,
) -> tuple[np.ndarray, float]:
    """
    Very fast random search over the four‑parameter power‑law family.
    It stops when `time_limit` seconds have elapsed.
    Returns the best sequence and its score.
    """
    end = time.time() + time_limit
    while time.time() < end:
        p = np.random.uniform(0.30, 0.70)
        off = np.random.uniform(0.0, 12.0)
        beta = np.random.uniform(-0.9, 2.5)
        cand = _make_candidate(n, p, off, beta)
        s = _max_conv(cand)          # Σa = √(2n) ⇒ score = max‑conv
        if s < best_score:
            best_score = s
            best_seq = cand
    return best_seq, best_score


def _active_set_lp_refinement(
    a: np.ndarray,
    M: float,
    max_inds: np.ndarray,
    target_sum: float,
    mix_factor: float,
) -> np.ndarray:
    """
    Solve the active‑set linear program

        maximise   Σ g_i
        subject to g ≥ 0,
                   (a * g)[k] ≤ M  for all k in max_inds

    and return the convex mixture (1‑t)·a + t·g_scaled,
    where `g_scaled` has sum = target_sum.
    """
    n = len(a)
    m = len(max_inds)
    # Build the (m × n) inequality matrix A_ub·g ≤ b_ub.
    A = np.zeros((m, n), dtype=float)
    for row, k in enumerate(max_inds):
        i_start = max(0, k - (n - 1))
        i_end = min(k, n - 1)
        # reversed slice of a placed on the appropriate columns
        A[row, i_start : i_end + 1] = a[i_start : i_end + 1][::-1]

    # Linprog formulation: minimise -Σ g_i   (i.e. maximise Σ g_i)
    c = -np.ones(n, dtype=float)
    bounds = [(0.0, None)] * n
    b_ub = np.full(m, M, dtype=float)

    # Solve using the HiGHS interior‑point method (fast for sparse dense A)
    res = linprog(
        c,
        A_ub=A,
        b_ub=b_ub,
        bounds=bounds,
        method="highs",
        options={"presolve": True},
    )
    if not res.success:
        # If the LP fails, simply return the original a.
        return a.copy()

    g0 = res.x
    sum_g0 = g0.sum()
    if sum_g0 <= 1e-12:
        return a.copy()

    # Scale to the required simplex sum.
    g_scaled = g0 * (target_sum / sum_g0)

    # Convex mixture.
    new_a = (1.0 - mix_factor) * a + mix_factor * g_scaled
    # Numerical drift may violate the simplex – re‑project.
    new_a = _project_simplex(new_a, target_sum)
    return new_a


# ----------------------------------------------------------------------
# Main search routine
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 123, budget_s: int = 1000, **kwargs):
    """
    Search for a step‑function with a very low evaluation value.
    Returns the best sequence (list of floats) found before the time budget
    expires.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 1.5   # safety buffer

    # --------------------------------------------------------------
    # 0. Choose the discretisation length.
    # --------------------------------------------------------------
    n = 8000                         # a size that balances quality & speed
    target_sum = math.sqrt(2.0 * n)

    # --------------------------------------------------------------
    # 1. Warm‑start – try to reuse a previously discovered sequence.
    # --------------------------------------------------------------
    best_seq = None
    if "height_sequence_1" in globals():
        hs = globals()["height_sequence_1"]
        if isinstance(hs, (list, np.ndarray)) and len(hs) == n:
            best_seq = np.asarray(hs, dtype=float)
            best_seq = _project_simplex(best_seq, target_sum)

    if best_seq is None:
        # Simple power‑law start (p=0.5, zero offset, no tilt).
        best_seq = _make_candidate(n, p=0.5, offset=0.0, beta=0.0)

    best_score = _max_conv(best_seq)          # because Σa = √(2n) ⇒ score = max‑conv
    print(f"[init] n={n}  score={best_score:.6f}")

    # --------------------------------------------------------------
    # 2. Fast random sweep (≈ 10 % of the total budget).
    # --------------------------------------------------------------
    sweep_time = 0.10 * (deadline - time.time())
    best_seq, best_score = _random_sweep(
        n, target_sum, sweep_time, best_seq, best_score
    )
    print(f"[sweep] best after random sweep {best_score:.6f}")

    # --------------------------------------------------------------
    # 3. Projected sub‑gradient descent (momentum‑free) – quick
    #    reduction of the max‑convolution.
    # --------------------------------------------------------------
    a = best_seq.copy()
    step0 = 1.0                      # initial sub‑gradient step size
    momentum = np.zeros_like(a)      # optional simple momentum term
    beta = 0.9
    iteration = 0
    no_improve = 0

    while time.time() < deadline:
        iteration += 1
        conv_full = _full_convolution_fft(a)
        M = conv_full.max()

        # keep the globally best sequence
        if M < best_score:
            best_score = M
            best_seq = a.copy()
            no_improve = 0
            print(f"[sgd] iter {iteration:5d}  new best {best_score:.6f}")
            if best_score < 1.503:
                break
        else:
            no_improve += 1

        # ----------------------------------------------------------
        # Decide whether to perform an LP refinement (every 200 iters
        # or if we have stagnated for a while).
        # ----------------------------------------------------------
        if iteration % 200 == 0 or no_improve > 500:
            eps = 1e-12 * M
            max_inds = np.where(conv_full >= M - eps)[0]

            # Guard against an excessively large active set.
            if len(max_inds) > 300:
                # keep only the most critical indices (those exactly equal to M)
                max_inds = max_inds[conv_full[max_inds] == M]

            # Active‑set LP refinement.
            a = _active_set_lp_refinement(
                a, M, max_inds, target_sum, mix_factor=0.05
            )
            # after mixing we continue the sub‑gradient loop.
            continue

        # ----------------------------------------------------------
        # Normal sub‑gradient step.
        # ----------------------------------------------------------
        eps = 1e-12 * M
        max_inds = np.where(conv_full >= M - eps)[0]

        # compute sub‑gradient (same as in the reference code)
        g = np.zeros_like(a)
        for k in max_inds:
            if k < n:
                g[: k + 1] += a[k :: -1]
            else:
                start = k - (n - 1)
                g[start:] += a[k - np.arange(start, n)]

        # simple momentum update (optional but cheap)
        momentum = beta * momentum + (1.0 - beta) * g

        # step size decays like 1/√t
        step = step0 / math.sqrt(iteration)
        cand = a - step * momentum
        cand = _project_simplex(cand, target_sum)

        # occasional tiny random jitter to escape plateaus
        if iteration % 50 == 0 and np.random.rand() < 0.3:
            jitter = (np.random.rand(n) - 0.5) * 0.02 * cand
            cand += jitter
            cand = _project_simplex(cand, target_sum)

        a = cand

    # --------------------------------------------------------------
    # 4. Final report
    # --------------------------------------------------------------
    print(
        f"Finished after {iteration} iterations – best score {best_score:.6f}"
    )
    # The harness expects a plain Python list.
    return best_seq.tolist()
