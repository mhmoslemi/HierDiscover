"""
Smooth‑gradient projected optimisation for the step‑function problem.

The objective to minimise (with a simplex normalisation Σa = √(2·n))
is   2·n·max(conv(a,a)) / (Σa)²  =  max(conv(a,a)) .
We therefore keep the sum exactly √(2·n) and minimise the peak of the
self‑convolution.

The algorithm consists of three phases:

1.  A quick random sweep over a 4‑parameter power‑law family yields a
    decent starting point.
2.  From this warm‑start a projected Adam optimiser is run.  The true
    max is replaced by a smooth soft‑max surrogate
          fλ(a) = (1/λ)·log Σ exp(λ·convk(a))
    whose gradient can be written in closed form.  After each Adam step
    the vector is projected onto the simplex {x≥0, Σx = √(2·n)}.
    The temperature λ is increased slowly so that the optimisation
    first sees a smooth landscape and later focuses on the actual
    maximum.
3.  The best sequence seen during the run is returned.

All heavy work (FFT‑based convolutions) is performed in compiled code,
so millions of iterations fit comfortably into the 1000 s budget.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities (no closures)
# ----------------------------------------------------------------------


def _project_simplex(v: np.ndarray, target_sum: float) -> np.ndarray:
    """Euclidean projection of v onto {x ≥ 0, Σx = target_sum}."""
    if target_sum < 0:
        raise ValueError("target_sum must be non‑negative")
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - target_sum
    ind = np.arange(1, len(v) + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.zeros_like(v)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def _fft_convolve(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """
    Linear convolution of two real arrays using a real FFT.
    Returns an array of length len(x)+len(y)-1.
    """
    n = len(x)
    m = len(y)
    size = 1 << ((n + m - 1).bit_length())  # next power of two
    fx = np.fft.rfft(x, n=size)
    fy = np.fft.rfft(y, n=size)
    conv = np.fft.irfft(fx * fy, n=size)[: n + m - 1]
    return conv


def _full_convolution_fft(a: np.ndarray) -> np.ndarray:
    """Full self‑convolution a*a (linear, not circular)."""
    return _fft_convolve(a, a)


def _max_conv(a: np.ndarray) -> float:
    """Maximum entry of the linear self‑convolution a*a."""
    return _full_convolution_fft(a).max()


def _make_candidate(n: int, p: float, offset: float, beta: float) -> np.ndarray:
    """
    Build a power‑law candidate
        a_i = (i+1+offset)^{-p}·(1+β·i/(n‑1))
    then project it onto the simplex of sum √(2·n).
    """
    idx = np.arange(n, dtype=float)
    base = np.power(idx + 1.0 + offset, -p)
    factor = 1.0 + beta * idx / (n - 1)
    raw = base * factor
    raw = np.maximum(raw, 0.0)
    target = np.sqrt(2.0 * n)
    return _project_simplex(raw, target)


# ----------------------------------------------------------------------
# Main search routine
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 123, budget_s: int = 1000, **kwargs):
    """
    Return a list of non‑negative heights that (empirically) minimise the
    evaluation function.  The routine respects a 1000 s time budget and
    prints progress information.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 2.0      # safety margin

    # ------------------------------------------------------------------
    # 0. Problem size (larger n typically yields a lower constant)
    # ------------------------------------------------------------------
    n = 15000                                   # feel free to adjust
    target_sum = np.sqrt(2.0 * n)

    # ------------------------------------------------------------------
    # 1. Warm‑start: try to reuse an already‑found sequence if it exists
    # ------------------------------------------------------------------
    best_seq = None
    hs = globals().get("height_sequence_1")
    if hs is not None:
        try:
            arr = np.asarray(hs, dtype=float)
            if arr.size >= n:
                best_seq = arr[:n].copy()
            else:
                best_seq = np.concatenate([arr, np.zeros(n - arr.size)])
            best_seq = _project_simplex(best_seq, target_sum)
        except Exception:
            best_seq = None

    if best_seq is None:
        # fallback: a simple power‑law with p=0.5
        best_seq = _make_candidate(n, p=0.5, offset=0.0, beta=0.0)

    # ------------------------------------------------------------------
    # 2. Fast random sweep over the 4‑parameter family
    # ------------------------------------------------------------------
    best_score = _max_conv(best_seq)          # equal to the evaluation value
    print(f"[init] n={n:,d}  score={best_score:.6f}")

    # allocate ~30 % of the remaining budget to the random sweep
    sweep_deadline = time.time() + 0.30 * (deadline - time.time())
    while time.time() < sweep_deadline:
        p = np.random.uniform(0.35, 0.65)
        off = np.random.uniform(0.0, 15.0)
        beta = np.random.uniform(-0.9, 2.5)
        cand = _make_candidate(n, p, off, beta)
        cand_score = _max_conv(cand)
        if cand_score < best_score:
            best_score = cand_score
            best_seq = cand
            print(f"[sweep] new best {best_score:.6f}  p={p:.3f} α={off:.3f} β={beta:.3f}")
        if best_score < 1.45:   # an early‑stop guard (unlikely to hit)
            break

    # ------------------------------------------------------------------
    # 3. Projected Adam optimisation on a soft‑max surrogate
    # ------------------------------------------------------------------
    # Adam hyper‑parameters
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    base_lr = 0.5               # step size before Adam normalisation
    # soft‑max temperature schedule
    lam0 = 0.5
    lam_inc = 0.0005
    lam_max = 20.0

    a = best_seq.copy()
    m = np.zeros_like(a)
    v = np.zeros_like(a)

    iteration = 0
    while time.time() < deadline:
        iteration += 1

        # current convolution & objective value
        conv_full = _full_convolution_fft(a)
        M = conv_full.max()

        # track the best ever seen
        if M < best_score:
            best_score = M
            best_seq = a.copy()
            print(f"[adam] iter {iteration:6d}  new best {best_score:.6f}")

            # stop early if we already beat the target
            if best_score < 1.503:
                break

        # ----- soft‑max weights (smooth approximation of the max) -----
        lam = min(lam_max, lam0 + lam_inc * iteration)
        shifted = conv_full - M          # ≤ 0
        w = np.exp(lam * shifted)
        w_sum = w.sum()
        if w_sum == 0:
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        # ----- gradient of the smooth surrogate -----
        # g_i = 2 * Σ_j w_{i+j} * a_j  (linear sliding‑dot product)
        corr = _fft_convolve(w, a[::-1])               # length 2n-1 + n-1
        grad = 2.0 * corr[n - 1 : n - 1 + n]           # slice to length n

        # ----- Adam update -----
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)

        m_hat = m / (1.0 - beta1 ** iteration)
        v_hat = v / (1.0 - beta2 ** iteration)

        step = base_lr * m_hat / (np.sqrt(v_hat) + eps)

        a = a - step
        a = _project_simplex(a, target_sum)

        # occasional tiny jitter to escape plateaus
        if iteration % 250 == 0 and np.random.rand() < 0.2:
            jitter = (np.random.rand(n) - 0.5) * 0.001 * a
            a += jitter
            a = _project_simplex(a, target_sum)

    # ------------------------------------------------------------------
    # 4. Final report
    # ------------------------------------------------------------------
    print(f"Finished after {iteration} iterations – best score {best_score:.6f}")
    # Convert to plain Python list as required by the evaluation harness.
    return best_seq.tolist()
