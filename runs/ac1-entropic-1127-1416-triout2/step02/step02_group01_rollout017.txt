"""
Search for a non‑negative step‑function that minimises

    E(a) = 2·n·max(a∗a) / (sum(a)²)

The strategy is deliberately simple yet powerful:

1.  Use a few classic initial shapes (arcsine, uniform, random Dirichlet,
    and a mix of arcsine/uniform) for several problem sizes.
2.  Run a cheap stochastic hill‑climbing phase that repeatedly moves a
    small amount of mass from one index to another and accepts the move
    only if the true evaluation (full convolution) improves.
3.  Allocate the available 1000 s budget across increasing lengths
    (2048 → 4096 → 8192).  Larger n gives a lower asymptotic constant, and
    the hill‑climber is fast enough to perform tens of thousands of moves
    even for the biggest vectors.
4.  Keep the best sequence found so far and finish with a short final
    refinement.

All helper functions are top‑level; no closures or file I/O are used.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Basic linear‑algebra utilities
# ----------------------------------------------------------------------


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of a vector onto the probability simplex."""
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / v.shape[0])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho = np.where(u - cssv / np.arange(1, v.shape[0] + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full convolution using a real FFT (fast)."""
    m = len(a) + len(b) - 1
    n_fft = next_pow2(m)
    A = np.fft.rfft(a, n=n_fft)
    B = np.fft.rfft(b, n=n_fft)
    C = A * B
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


def compute_score(a: np.ndarray) -> float:
    """
    Exact score used by `evaluate_sequence`.  The function is scale‑invariant;
    we simply keep the vector on the simplex (sum = 1) for consistency.
    """
    n = len(a)
    conv = convolution_fft(a, a)           # length 2n‑1
    max_val = float(conv.max())
    s = float(a.sum())
    if s < 1e-12:
        return float("inf")
    return 2.0 * n * max_val / (s * s)


def arcsine_initial(n: int) -> np.ndarray:
    """Arcsine‑shaped start: a_i ∝ 1 / sqrt((i+0.5)(n‑0.5‑i))."""
    i = np.arange(n, dtype=float)
    eps = 0.25
    vals = 1.0 / np.sqrt((i + 0.5) * (n - 0.5 - i) + eps)
    return vals / vals.sum()


# ----------------------------------------------------------------------
# Hill‑climbing optimisation (stochastic pairwise mass transfer)
# ----------------------------------------------------------------------


def hill_climb(start_seq,
               time_deadline: float,
               max_steps: int = 100_000,
               init_step: float = 0.2) -> tuple[list[float], float]:
    """
    Stochastic hill climbing.

    At each iteration a small amount of mass is moved from index i to j.
    Moves that reduce the true evaluation are kept; otherwise the step
    size is slightly decreased.  The routine stops when the step budget
    is exhausted or the global deadline is reached.
    """
    a = np.array(start_seq, dtype=float)
    a = project_to_simplex(a)          # ensure sum = 1
    n = len(a)

    best_a = a.copy()
    best_score = compute_score(best_a)
    step = init_step

    while max_steps > 0 and time.time() < time_deadline:
        i, j = np.random.randint(n, size=2)
        if i == j:
            continue

        delta = step * best_a[i]       # move a proportion of the source mass
        if delta <= 0.0:
            continue

        cand = best_a.copy()
        cand[i] -= delta
        cand[j] += delta
        if cand[i] < 0.0:               # safety (numerical)
            continue

        # No need to re‑project because sum is preserved and non‑negativity holds.
        cur_score = compute_score(cand)

        if cur_score < best_score:
            best_score = cur_score
            best_a = cand
            step = min(step * 1.02, 0.5)    # slowly enlarge an improving step
        else:
            step *= 0.98                    # shrink step on a non‑improving move

        max_steps -= 1

    return best_a.tolist(), best_score

    start_time = time.time()
    # tiny safety margin for the harness

    best_seq = None
    best_val = float("inf")

    # --------------------------------------------------------------

    try:

            best_val = val
            best_seq = seq

    # Fallback in the unlikely event that we never produced a candidate.
    # ------------------------------------------------------------------

        best_val = evaluate_sequence(best_seq)

    final_score = evaluate_sequence(best_seq)
    print(f"\nSearch finished. Final reported score = {final_score:.6f}")

    return best_seq
