"""Mirror‑descent optimizer for the max‑convolution objective.

The method keeps a cumulative (scaled) sub‑gradient and builds the
probability vector by an exponentiated‑gradient update
\(\;a_i\propto\exp(-G_i)\) where \(G\) is the cumulative gradient.
A Polyak step size \((f(x)-L)/\|g\|^2\) (clipped) is used,
combined with a diminishing‑step schedule as a safety net.
Multiple random restarts (including the previously known sequence
if available) are performed until the time budget expires.
The best vector found is returned as a plain Python list.
"""

import time
import numpy as np

def _project_to_simplex(v: np.ndarray) -> np.ndarray:
    """ Euclidean projection onto the probability simplex. """
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1] - 1
    theta = cssv[cond][-1] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def _max_and_subgradient(a: np.ndarray) -> tuple[float, np.ndarray]:
    """
    Return the maximal entry of the (linear) convolution a*a and a
    sub‑gradient of that entry (the same quantity that is used in the
    original sub‑gradient code).
    """
    n = a.shape[0]
    fft = np.fft.rfft(a, n=2 * n - 1)
    conv = np.fft.irfft(fft * fft, n=2 * n - 1)
    max_val = float(conv.max())
    # tolerance to collect all active (nearly maximal) indices
    tol = max_val * 1e-12 + 1e-14
    active = np.where(conv >= max_val - tol)[0]

    grad = np.zeros_like(a)
    for k in active:
        low = max(0, k - n + 1)
        high = min(n - 1, k)
        idx = np.arange(low, high + 1, dtype=int)
        partner = k - idx
        grad[idx] += 2.0 * a[partner]
    if active.size > 1:
        grad /= float(active.size)
    return max_val, grad

        else:

        n = 2048                     # a convenient default length

    # --------------------------------------------------------------

    # --------------------------------------------------------------

    lower_bound = 1.0               # theoretical lower bound of the objective

        return 2.0 * n * max_conv

    # Collect all indices that are numerically tied for the maximum.
    tol = max_val * 1e-12 + 1e-14
    active = np.where(conv >= max_val - tol)[0]

    grad = np.zeros_like(a)
    for k in active:
        low = max(0, k - n + 1)
        high = min(n - 1, k)
        idx = np.arange(low, high + 1, dtype=int)
        partner = k - idx
        grad[idx] += 2.0 * a[partner]

    if active.size > 1:
        grad /= float(active.size)
    return max_val, grad


def propose_candidate(budget_s: int = 1000) -> list[float]:
    """
    Search for a step‑function that minimizes

        2·n·max(a*a) / (sum(a)**2)

    The routine runs for at most ``budget_s`` seconds and returns the best
    sequence encountered.
    """
    # ------------------------------------------------------------------
    # 1. Problem size and a (optional) starting point.
    # ------------------------------------------------------------------
    deadline = time.time() + budget_s - 5.0   # safety margin

    try:
        # Use the historic best sequence if it exists.
        init = np.asarray(height_sequence_1, dtype=float)  # type: ignore[name-defined]
        n = init.shape[0]
        init = np.maximum(init, 0.0)
        s = init.sum()
        if s > 0:
            init = init / s
        else:
            init = None
    except Exception:
        init = None
        n = 2048                     # reasonable default length

    # ------------------------------------------------------------------
    # 2. Optimiser constants.
    # ------------------------------------------------------------------
    lower_bound = 1.0               # theoretical lower bound of the objective
    best_a = None
    best_score = float("inf")
    max_step = 5.0                  # clip for the Polyak step size

    # ------------------------------------------------------------------
    # 3. Main optimisation loop – many independent restarts.
    # ------------------------------------------------------------------
    restart = 0
    while time.time() < deadline:
        rng = np.random.default_rng(
            seed=int(time.time() * 1e6) % (2**32 - 1)
        )
        # initialise a new point (first restart may use the known sequence)
        if init is not None and restart == 0:
            a = init.copy()
        else:
            a = rng.random(n)
        a = np.maximum(a, 1e-30)          # avoid exact zeros
        a = a / a.sum()                   # project onto the simplex

        # cumulative scaled sub‑gradient (mirrored variable)
        g_cum = np.zeros_like(a)

        it = 0
        while time.time() < deadline:
            max_conv, grad = _max_and_subgradient(a)
            cur_score = 2.0 * n * max_conv   # because sum(a)==1

            # keep the globally best solution
            if cur_score < best_score:
                best_score = cur_score
                best_a = a.copy()
                print(
                    f"[{time.time():.1f}] restart {restart}, iter {it}: "
                    f"new best = {best_score:.6f}"
                )
                # early exit if we have already beaten the target
                if best_score < 1.503:
                    print("Target reached – returning early.")
                    return [float(v) for v in best_a.tolist()]

            # --------------------------------------------------------------
            # 3.1 Polyak step size (clipped) and cumulative gradient update.
            # --------------------------------------------------------------
            norm_sq = float(np.dot(grad, grad))
            if norm_sq == 0.0:
                break     # nothing to improve in this restart

            step = (cur_score - lower_bound) / (norm_sq + 1e-12)
            step = max(1e-12, min(step, max_step))

            g_cum += step * grad

            # --------------------------------------------------------------
            # 3.2 Exponentiated‑gradient / mirror‑descent update.
            # --------------------------------------------------------------
            tmp = -g_cum
            tmp -= np.max(tmp)          # stabilise the exponentials
            a = np.exp(tmp)
            s = a.sum()
            if s == 0.0:
                a = np.full(n, 1.0 / n)
            else:
                a /= s

            # --------------------------------------------------------------
            # 3.3 Occasional stochastic jitter (helps escape plateaus).
            # --------------------------------------------------------------
            if it % 5000 == 0 and it > 0:
                eps = 0.001 * rng.standard_normal(n)
                a = np.maximum(a + eps, 0.0)
                a /= a.sum()

            it += 1

        restart += 1

    # ------------------------------------------------------------------
    # 4. Return the best sequence we have found.
    # ------------------------------------------------------------------
    if best_a is None:
        best_a = np.full(n, 1.0 / n)

    return [float(v) for v in best_a.tolist()]
