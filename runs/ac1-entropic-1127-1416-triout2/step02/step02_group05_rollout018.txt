    # Hyperparameters
    step = 0.1
    min_step = 1e-6
    step_decay = 0.99
    alpha = 50.0 # smoothing factor
    iter_cnt = 0

    while time.time() < deadline:
        b = np.convolve(a, a)
        max_b = b.max()
        cur_score = 2.0 * n * max_b
        if cur_score < best_score - 1e-12:
            best_score = cur_score
            best_seq = a.copy()
            print(f"[improve] iter={iter_cnt}, score={best_score:.9f}")

        # smoothing weights
        # compute exp(alpha*(b-max_b))
        w = np.exp(alpha * (b - max_b))
        w_sum = w.sum()
        if w_sum == 0:
            # all zero? shouldn't happen
            break
        p = w / w_sum

        # compute gradient: grad = 2 * y, where y_i = sum_j a_j p_{i+j}
        # using convolution with reversed p
        p_rev = p[::-1]
        conv = np.convolve(a, p_rev)  # length 3n-2

        idx = 2*n - 2 - np.arange(n)
        y = conv[idx]
        grad = 2.0 * y

        # line search
        t = step

        step = max(min_step, t * step_decay)
        iter_cnt += 1

"""
Propose a step‑function sequence that minimises

    2 * n * max(a * a) / (sum(a) ** 2)

The algorithm is a hybrid of:

1. Smoothed‑gradient descent – we minimise a smooth surrogate of the
   ‖a∗a‖∞ norm (log‑sum‑exp of the convolution).  The gradient can be
   written in closed form and is obtained by a single convolution.
   After each trial step the vector is projected onto the probability simplex
   (non‑negative, sum = 1).

2. Periodic linear‑programming refinement – for the current maximal
   convolution entries we solve a tiny LP (maximise Σ g j) that keeps the
   mixed convolution (a∗g) below the current bound.  Mixing the LP solution
   with the current vector provably never raises the ‖a∗a‖∞ value
   (provided the mixing factor is small).

Both parts respect the security checks of ``evaluate_sequence`` and run
well within a 1000 s budget for the typical length (~5 k).  The best
sequence seen is returned as a plain Python list.
"""

import time
import numpy as np
from scipy.optimize import linprog

# ----------------------------------------------------------------------
def binomial_sequence(length: int) -> np.ndarray:
    """
    Central‑binomial coefficients c_k = C(2k, k) / 4**k, k = 0 … length‑1.
    The sequence is already normalised later.
    """
    seq = np.empty(length, dtype=np.float64)
    seq[0] = 1.0
    for k in range(1, length):
        seq[k] = seq[k - 1] * ((2 * k - 1) * (2 * k)) / (4.0 * k * k)
    return seq


# ----------------------------------------------------------------------
def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection onto the probability simplex (non‑negative,
    sum = 1).  For a strictly positive sum the simple scaling works;
    if the vector is all zeros we return the uniform distribution.
    """
    v = np.maximum(v, 0.0)
    s = v.sum()
    if s == 0.0:
        return np.full_like(v, 1.0 / len(v))
    return v / s


# ----------------------------------------------------------------------
def smooth_gradient(a: np.ndarray, alpha: float = 80.0) -> np.ndarray:
    """
    Gradient of the smoothed max (log‑sum‑exp) of the convolution a∗a.
    Returns ∇_a  ( 1/α log Σ_k exp(α·(a∗a)_k) ).
    """
    n = len(a)
    b = np.convolve(a, a)
    # soft‑max weights, numerically stable because we subtract max(b)
    w = np.exp(alpha * (b - b.max()))
    w /= w.sum()                     # now Σ_k w_k = 1

    # y_i = Σ_j a_j w_{i+j}  (i = 0 … n‑1)
    # This can be obtained by a single convolution with the reversed weights.
    w_rev = w[::-1]
    conv = np.convolve(a, w_rev)    # length 3n‑2
    idx = 2 * n - 2 - np.arange(n)   # pick the slice that corresponds to i+j
    y = conv[idx]
    return 2.0 * y                   # ∇_a = 2·y


# ----------------------------------------------------------------------
def lp_refine(a: np.ndarray,
              cur_max: float,
              max_constraints: int = 150,
              mix: float = 0.4) -> np.ndarray:
    """
    Solve a tiny linear programme that adds mass in directions that keep
    the current maximal convolution entry unchanged.

    maximise   Σ g_j
    subject to  (a * g)_k ≤ cur_max  for the most active constraints k
                g_j ≥ 0

    The new vector is  (1‑mix)·a + mix·g (projected onto the simplex).
    """
    n = len(a)
    # pick the indices with the largest convolution entries (active set)
    b = np.convolve(a, a)
    idx = np.argsort(-b)[:max_constraints]            # descending order
    rows = []
    cols = []
    data = []

    for row_id, k in enumerate(idx):
        lo = max(0, k - (n - 1))
        hi = min(k, n - 1)
        cols_range = np.arange(lo, hi + 1, dtype=int)
        rows.extend([row_id] * len(cols_range))
        cols.extend(cols_range.tolist())
        data.extend(a[k - cols_range].tolist())

    A = np.zeros((len(idx), n), dtype=np.float64)
    if rows:
        A[rows, cols] = data

    c = -np.ones(n)                     # maximise Σ g_j  => minimise -Σ g_j
    ub = np.full(len(idx), cur_max)
    bounds = [(0, None)] * n

    res = linprog(c, A_ub=A, b_ub=ub, bounds=bounds,
                  method='highs', options={'presolve': True})
    if not res.success or res.x.sum() == 0.0:
        return a                         # no beneficial direction found

    g = res.x / res.x.sum()             # renormalise to unit L1‑norm
    a_new = (1.0 - mix) * a + mix * g
    return project_to_simplex(a_new)


# ----------------------------------------------------------------------

    """

    """

    best_seq = a.copy()
    best_score = float('inf')
    step = 0.12
    step_decay = 0.9995
    min_step = 1e-6
    it = 0

    # ------------------------------------------------------------------
    while time.time() < deadline:
        # ---- current evaluation ------------------------------------------------
        b = np.convolve(a, a)
        cur_max = b.max()
        cur_score = 2.0 * n * cur_max

        if cur_score < best_score:

            best_seq = a.copy()

        # ---- smoothed gradient descent -----------------------------------------
        grad = smooth_gradient(a, alpha=80.0)
        direction = -grad
        t = step

        while t > min_step:
            cand = a + t * direction
            cand = np.maximum(cand, 0.0)
            cand = project_to_simplex(cand)

            if 2.0 * n * cand_max < cur_score:

        it += 1

    print(f"[final] best score={best_score:.9f}, iterations={it}")
    return best_seq.tolist()
