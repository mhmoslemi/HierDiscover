    return best_a.tolist()

def optimize(a_start, max_iters):
    a = a_start.copy()
    best_local_a = a.copy()
    best_local_score = compute_objective(a)
    for t in range(max_iters):
        # compute conv and active set
        fft = np.fft.rfft(a, n=2*n-1)
        conv = np.fft.irfft(fft*fft, n=2*n-1)
        max_val = conv.max()
        active = np.where(conv >= max_val - tol)[0]
        grad = np.zeros_like(a)
        for k in active:
            low = max(0, k - n + 1)
            high = min(n-1, k)
            idx = np.arange(low, high+1)
            partner = k - idx
            grad[idx] += 2.0 * a[partner]
        if active.size > 1:
            grad /= active.size
        # compute step size
        eta = 0.5 / np.sqrt(t+1)
        # exponentiated update
        a = a * np.exp(-eta * grad)
        # renormalize
        a_sum = a.sum()
        if a_sum == 0:
            a = np.full(n, 1.0/n)
        else:
            a = a / a_sum
        # evaluate score
        cur_score = compute_objective(a)
        if cur_score < best_local_score:
            best_local_score = cur_score
            best_local_a = a.copy()
            # optional print
            print(f"Iter {t}: score = {cur_score:.6f}")
        # optional early stop if cur_score below global best maybe
    return best_local_a, best_local_score

def propose_candidate(budget_s=1000):
    import time, numpy as np

    deadline = time.time() + budget_s - 5.0

    # Step 1: Determine n
    try:
        n = len(height_sequence_1)
    except Exception:
        n = 5000   # default

    # Step 2: Precompute index arrays
    i = np.arange(n)
    center = (n-1)/2

    best_a = None
    best_score = float('inf')

    # Helper to evaluate and update best
    def consider(candidate):
        nonlocal best_a, best_score
        total = candidate.sum()
        if total == 0:
            return
        a = candidate / total
        score = compute_objective(a)
        if score < best_score:
            best_score = score
            best_a = a.copy()
            print(f"Init candidate improved best to {best_score:.6f}")

    # Uniform
    consider(np.full(n, 1.0))

    # Existing height_sequence_1 if available
    try:
        h1 = np.asarray(height_sequence_1, dtype=float)
        consider(h1)
    except Exception:
        pass

    # Cosine power candidates
    cos_base = np.cos(np.pi * (i - center) / (n-1))
    for p in np.linspace(0.2, 5.0, 12):
        a = np.power(cos_base, p)
        consider(a)

    # Beta-like candidates (symmetric)
    x = i / (n-1)
    for alpha in np.linspace(0.5, 5.0, 12):
        a = (x * (1 - x))**alpha
        consider(a)

    # Gaussian candidates
    for sigma_frac in np.linspace(0.1, 0.5, 9):
        sigma = sigma_frac * n
        a = np.exp(-0.5 * ((i - center) / sigma) ** 2)
        consider(a)

    # Random Dirichlet candidates (few)
    for _ in range(5):
        a = np.random.rand(n)
        consider(a)

    # If after all, best_a is None use uniform
    if best_a is None:
        best_a = np.full(n, 1.0/n)

    # Main optimization loop
    a = best_a.copy()
    # Best overall record
    best_overall_a = a.copy()
    best_overall_score = compute_objective(a)
    # optimization parameters
    iteration = 0
    # for exponentiated gradient, we may use step0 = 0.5
    step0 = 0.5

    while time.time() < deadline:
        # compute convolution
        fft = np.fft.rfft(a, n=2*n-1)
        conv = np.fft.irfft(fft*fft, n=2*n-1)
        max_val = conv.max()
        # inclusive active indices near max
        tol = max_val * 1e-12 + 1e-14
        active = np.where(conv >= max_val - tol)[0]

        # compute subgradient
        grad = np.zeros_like(a)
        for k in active:
            low = max(0, k - n + 1)
            high = min(n - 1, k)
            idx = np.arange(low, high + 1)
            partner = k - idx
            grad[idx] += 2.0 * a[partner]

        if active.size > 1:
            grad /= active.size

        # step size
        eta = step0 / np.sqrt(iteration + 1)

        # exponentiated gradient update (multiplicative)
        # protect against overflow by clipping grad
        # compute factor
        factor = np.exp(-eta * grad)
        a = a * factor
        # renormalize to simplex
        sum_a = a.sum()
        if sum_a == 0:
            a = np.full(n, 1.0/n)
        else:
            a = a / sum_a

        # occasional jitter
        if iteration % 4000 == 0 and iteration > 0:
            eps = 0.005 / np.sqrt(iteration + 1)
            a = a + eps * np.random.randn(n)
            a = np.clip(a, 0, None)
            a = a / a.sum()

        # evaluate
        cur_score = compute_objective(a)
        if cur_score < best_overall_score:
            best_overall_score = cur_score
            best_overall_a = a.copy()
            print(f"Iter {iteration:6d}: improved overall score = {best_overall_score:.6f}")
            # early stop if target reached
            if best_overall_score < 1.503:
                print(f"Target achieved: {best_overall_score:.6f} at iter {iteration}")
                break

        iteration += 1
        # optionally limit max iterations e.g., 200000

    # Return best sequence
    return [float(v) for v in best_overall_a.tolist()]

"""
Propose a step‐function coefficients vector that (approximately) minimises

    2·n·max(conv(a,a)) / (sum(a)^2)

The algorithm consists of three stages:

1. **Problem size** – use the length of the previously found
   `height_sequence_1` if it exists, otherwise fall back to a
   moderate size (≈5000) that gives a good trade‑off between solution
   quality and runtime.

2. **Initial‑point search** – generate a handful of deterministic
   candidates (uniform, cosine‑power, symmetric Beta‑shaped,
   truncated Gaussian) together with a few random Dirichlet vectors.
   Each candidate is normalised to sum 1 and evaluated with a fast FFT‑based
   objective. The best among them seeds the optimisation.

3. **Exponentiated‑gradient descent** – starting from the best seed,
   repeatedly:
       * compute the full self‑convolution via FFT,
       * locate the (possibly tied) index/indices attaining the current
         maximum,
       * build a sub‑gradient of this maximum,
       * perform a multiplicative (exponentiated) gradient step
         `a_i ← a_i·exp(-η·grad_i)`,
       * renormalise to the simplex,
       * occasionally inject a tiny random jitter to escape plateaux,
       * keep the best vector seen so far.
   The step‑size η follows a classic diminishing schedule
   `η = η0 / sqrt(t+1)`.  The loop runs until the time budget
   (≈ 1000 s) is exhausted or a target value (1.503) is reached.

The returned list contains non‑negative numbers (already normalised;
scaling does not affect the objective) and can be fed directly to
`evaluate_sequence`.
"""

import time
import numpy as np

def simplex_projection(v: np.ndarray) -> np.ndarray:
    """Project onto the probability simplex {w ≥ 0, Σw = 1}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1] - 1
    theta = cssv[cond][-1] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def compute_objective(a: np.ndarray) -> float:
    """
    Fast version of `evaluate_sequence` for vectors normalised to Σa = 1.
    Returns 2·n·max(conv(a,a)).
    """
    n = a.shape[0]
    # full linear convolution via FFT (length 2n‑1)
    fft = np.fft.rfft(a, n=2 * n - 1)
    conv = np.fft.irfft(fft * fft, n=2 * n - 1)
    max_b = conv.max()
    return 2.0 * n * max_b


def propose_candidate(budget_s: int = 1000) -> list:
    """
    Search for a low‑score step function.
    Runs for up to ``budget_s`` seconds and returns the best sequence found.
    """
    np.random.seed(0)                     # reproducibility
    deadline = time.time() + budget_s - 5.0   # safety margin

    # --------------------------------------------------------------
    # 1. Choose problem size.
    # --------------------------------------------------------------
    try:
        n = len(height_sequence_1)        # type: ignore[name-defined]
        init_from_env = True
    except Exception:
        n = 5146                          # reasonable default
        init_from_env = False

    i = np.arange(n)
    centre = (n - 1) / 2.0

    best_a = None
    best_score = float('inf')

    def consider(candidate: np.ndarray):
        """Normalise candidate, evaluate and update global best."""
        nonlocal best_a, best_score
        total = candidate.sum()
        if total <= 0:
            return
        a = candidate / total
        score = compute_objective(a)
        if score < best_score:
            best_score = score
            best_a = a.copy()
            print(f"Initial candidate improved best score to {best_score:.6f}")

    # ---- deterministic families ------------------------------------
    # uniform
    consider(np.full(n, 1.0))

    # previous best (if available)
    if init_from_env:
        try:
            consider(np.asarray(height_sequence_1, dtype=float))
        except Exception:
            pass

    # cosine‑power shapes  a_i = cos(...)^p   (p>0)
    cos_base = np.cos(np.pi * (i - centre) / (n - 1))
    for p in np.linspace(0.2, 5.0, 12):
        a = np.power(cos_base, p)
        consider(a)

    # symmetric Beta‑like shapes   a_i ∝ (x·(1‑x))^α
    x = i / (n - 1)
    for alpha in np.linspace(0.5, 5.0, 12):
        a = (x * (1.0 - x)) ** alpha
        consider(a)

    # truncated Gaussian   a_i = exp(-0.5*((i‑centre)/σ)^2)
    for sigma_frac in np.linspace(0.1, 0.5, 9):
        sigma = sigma_frac * n
        a = np.exp(-0.5 * ((i - centre) / sigma) ** 2)
        consider(a)

    # a few random Dirichlet seeds
    for _ in range(5):
        a = np.random.rand(n)
        consider(a)

    # --------------------------------------------------------------
    # 2. If nothing succeeded, fall back to uniform.
    # --------------------------------------------------------------
    if best_a is None:
        best_a = np.full(n, 1.0 / n)

    # --------------------------------------------------------------
    # 3. Main optimisation loop – exponentiated‑gradient descent.
    # --------------------------------------------------------------
    a = best_a.copy()
    best_overall_a = a.copy()
    best_overall_score = compute_objective(a)

    iteration = 0
    eta0 = 0.5           # initial step‑size multiplier
    beta_jitter = 0.005  # magnitude of occasional random jitter

    while time.time() < deadline:
        # ----- convolution and location of the current maximum -----
        fft = np.fft.rfft(a, n=2 * n - 1)
        conv = np.fft.irfft(fft * fft, n=2 * n - 1)
        max_val = conv.max()

        # ----- active set (indices tying for the maximum) ----------
        tol = max_val * 1e-12 + 1e-14
        active = np.where(conv >= max_val - tol)[0]

        # ----- sub‑gradient of the max‑entry -----------------------
        grad = np.zeros_like(a)
        for k in active:
            low = max(0, k - n + 1)
            high = min(n - 1, k)
            idx = np.arange(low, high + 1, dtype=int)
            partner = k - idx
            grad[idx] += 2.0 * a[partner]
        if active.size > 1:
            grad /= active.size  # any convex combination is a valid sub‑gradient

        # ----- multiplicative (exponentiated) update ---------------
        eta = eta0 / np.sqrt(iteration + 1)
        # protect against overflow
        factor = np.exp(-eta * grad)
        a = a * factor
        # re‑project onto the simplex (simple renormalisation)
        s = a.sum()
        if s == 0.0:
            a = np.full(n, 1.0 / n)
        else:
            a /= s

        # ----- occasional jitter to escape flat regions ----------
        if iteration % 4000 == 0 and iteration > 0:
            eps = beta_jitter / np.sqrt(iteration + 1)
            a += eps * np.random.randn(n)
            a = np.clip(a, 0.0, None)
            a /= a.sum()

        # ----- evaluate current solution ---------------------------
        cur_score = compute_objective(a)
        if cur_score < best_overall_score:
            best_overall_score = cur_score
            best_overall_a = a.copy()
            print(f"Iter {iteration:6d}: improved overall score = {best_overall_score:.6f}")
            if best_overall_score < 1.503:
                print(f"Target reached (score {best_overall_score:.6f}) at iteration {iteration}")
                break

        iteration += 1

    # --------------------------------------------------------------
    # 4. Return the best vector found as a plain Python list.
    # --------------------------------------------------------------
    return [float(v) for v in best_overall_a.tolist()]
