"""
Propose a candidate step‑height sequence.

The algorithm is a projected sub‑gradient method with *exact line search*.
Given a non‑negative vector a (∑a = 1), the objective is
    F(a) = 2·n·max_k (a * a)[k]          (n = len(a))

which is equivalent to minimising the ℓ∞‑norm of the self‑convolution.
For the current maximal convolution entries we compute a sub‑gradient
g_i = Σ_{k∈M} a_{k‑i} (M = set of maximal indices).  The feasible
descent direction is d = mean(g) – g, which automatically satisfies
∑d = 0 and keeps the sum of a unchanged.

The step length α is chosen *optimally*: for any α≥0 the new convolution
is
    b(α) = a*a + 2α·(a*d) + α²·(d*d)      (convolution of vectors)
and max_k b_k(α) is a convex univariate function of α.
A golden‑section search on the interval [0, α_max] (the largest α that
keeps a non‑negative) yields the optimal step in a few dozen evaluations.

All convolutions are performed with FFTs (full linear convolution) for
speed, even for several‑thousand‑element vectors.  Random pairwise
mass‑transfer moves are injected occasionally to escape plateaus.
The best sequence seen during the allotted time is returned.

The routine runs for at most ``budget_s`` seconds (default 1000 s) and
prints progress information.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Return the smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def full_convolve_fft(u: np.ndarray, v: np.ndarray) -> np.ndarray:
    """
    Full linear convolution of two real vectors using FFT.
    Returns an array of length len(u)+len(v)-1.
    """
    L = len(u) + len(v) - 1
    N = _next_pow2(L)
    fu = np.fft.rfft(u, n=N)
    fv = np.fft.rfft(v, n=N)
    conv = np.fft.irfft(fu * fv, n=N)
    return conv[:L].astype(np.float64)


def _max_convolution(a: np.ndarray, aa: np.ndarray = None) -> tuple[float, np.ndarray]:
    """
    Compute the self‑convolution of a and return (max_value, conv_vector).
    If aa is supplied it is assumed to be a*a, avoiding recomputation.
    """
    if aa is None:
        aa = full_convolve_fft(a, a)
    max_val = aa.max()
    return max_val, aa


def _optimal_alpha(conv_aa: np.ndarray,
                   conv_ad: np.ndarray,
                   conv_dd: np.ndarray,
                   a_max: float,
                   alpha_hi: float,
                   max_iters: int = 30) -> tuple[float, float]:
    """
    Perform a golden‑section search on α∈[0, α_hi] to minimise
        f(α) = max_k (conv_aa + 2α·conv_ad + α²·conv_dd)_k .
    Returns (α_opt, f_opt).
    """
    phi = (np.sqrt(5) - 1) / 2  # 0.618...
    lo, hi = 0.0, alpha_hi
    # initialise interior points
    x1 = hi - phi * (hi - lo)
    x2 = lo + phi * (hi - lo)

    def f(alpha: float) -> float:
        b = conv_aa + 2.0 * alpha * conv_ad + (alpha * alpha) * conv_dd
        return b.max()

    f1 = f(x1)
    f2 = f(x2)

    for _ in range(max_iters):
        if f1 < f2:
            hi, x2, f2 = x2, x1, f1
            x1 = hi - phi * (hi - lo)
            f1 = f(x1)
        else:
            lo, x1, f1 = x1, x2, f2
            x2 = lo + phi * (hi - lo)
            f2 = f(x2)

    # choose the better of the two interior points
    if f1 < f2:
        return x1, f1
    else:
        return x2, f2


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function height vector that minimises
    evaluate_sequence().  Returns the best sequence found before the time
    budget expires.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.1   # tiny safety margin

    # ------------------------------------------------------------------
    # Determine the length of the vector.
    try:
        n = len(height_sequence_1)          # type: ignore[name-defined]
        init_from_prev = True
    except Exception:
        n = 8000
        init_from_prev = False

    # ------------------------------------------------------------------
    # Helper to initialise a vector.
    def _init_vector() -> np.ndarray:
        if init_from_prev:
            a0 = np.array(height_sequence_1, dtype=np.float64)
            a0 = np.maximum(a0, 0.0)
            s = a0.sum()
            if s < 1e-12:
                a0 = np.full(n, 1.0 / n)
            else:
                a0 /= s
            return a0
        # fallback: central binomial coefficients (discrete normal)
        seq = np.empty(n, dtype=np.float64)
        seq[0] = 1.0
        for k in range(1, n):
            seq[k] = seq[k - 1] * ((2.0 * k - 1.0) * (2.0 * k)) / (4.0 * k * k)
        seq = np.maximum(seq, 0.0)
        seq /= seq.sum()
        return seq

    # ------------------------------------------------------------------
    a = _init_vector()
    conv_aa = full_convolve_fft(a, a)
    cur_max, conv_aa = _max_convolution(a, conv_aa)
    best_score = 2.0 * n * cur_max
    best_seq = a.copy()
    best_iter = 0
    iter_cnt = 0
    last_improve = time.time()

    print(f"[init] n={n}, initial score={best_score:.9f}")

    # ------------------------------------------------------------------
    # Main optimisation loop
    while time.time() < deadline:
        # --------------------------------------------------------------
        # 1.  Locate maximal convolution entries.
        cur_max = conv_aa.max()
        # keep track of globally best
        cur_score = 2.0 * n * cur_max
        if cur_score < best_score - 1e-12:
            best_score = cur_score
            best_seq = a.copy()
            best_iter = iter_cnt
            print(f"[improve] iter={iter_cnt}, score={best_score:.9f}, "
                  f"time={time.time() - (deadline - budget_s):.2f}s")
            last_improve = time.time()

        # --------------------------------------------------------------
        # 2.  Sub‑gradient of the max‑entry (g_i = Σ_{k∈M} a_{k‑i}).
        tol = cur_max - 1e-14   # very small absolute tolerance
        max_inds = np.flatnonzero(conv_aa >= tol)

        g = np.zeros_like(a)
        # accumulate contributions of all maximal positions
        for k in max_inds:
            lo = max(0, k - (n - 1))
            hi = min(n - 1, k)
            idx = np.arange(lo, hi + 1, dtype=np.int64)
            g[idx] += a[k - idx]          # a_{k‑i}

        # --------------------------------------------------------------
        # 3.  Feasible descent direction (sum(d)=0, keeps positivity after step)
        mean_g = g.mean()
        d = mean_g - g                     # = -g + mean(g)·1

        # If the direction is (numerically) zero, break.
        if np.linalg.norm(d) < 1e-18:
            break

        # --------------------------------------------------------------
        # 4.  Maximum admissible step (prevent negativity)
        neg_mask = d < 0.0
        if np.any(neg_mask):
            alpha_max = np.min(a[neg_mask] / (-d[neg_mask]))
        else:
            # direction is non‑negative everywhere – we can move arbitrarily,
            # but a huge step would only increase the max; limit it reasonably.
            alpha_max = 1.0

        if alpha_max <= 0.0:
            break

        # --------------------------------------------------------------
        # 5.  Convolutions needed for the quadratic model of the objective
        conv_ad = full_convolve_fft(a, d)         # a*d
        conv_dd = full_convolve_fft(d, d)         # d*d

        # --------------------------------------------------------------
        # 6.  Exact line search on α∈[0, α_max]
        alpha_opt, new_max = _optimal_alpha(conv_aa, conv_ad, conv_dd,
                                            cur_max, alpha_max,
                                            max_iters=30)

        # --------------------------------------------------------------
        # 7.  Update a (the step is guaranteed to keep a ≥ 0)
        a = a + alpha_opt * d

        # Clip any tiny negative values caused by floating‑point noise,
        # then renormalise to keep ∑a = 1 exactly.
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s == 0.0:
            a = np.full_like(a, 1.0 / n)
        else:
            a /= s

        # --------------------------------------------------------------
        # 8.  Refresh the convolution for the new point.
        conv_aa = full_convolve_fft(a, a)

        # --------------------------------------------------------------
        # 9.  Occasionally inject a tiny random redistribution to escape
        #     possible stagnation (≈1 % of iterations).
        if np.random.rand() < 0.01:
            i, j = np.random.choice(n, size=2, replace=False)
            if a[i] > 0.0:
                delta = 0.02 * np.random.rand() * a[i]
                a[i] -= delta
                a[j] += delta
                # re‑normalise (sum is preserved, but numerical errors may appear)
                a /= a.sum()
                conv_aa = full_convolve_fft(a, a)

        iter_cnt += 1

    # ------------------------------------------------------------------
    print(f"[final] best score={best_score:.9f}, found at iter={best_iter}, "
          f"total iterations={iter_cnt}")
    # The evaluation routine clips values to [0,1000]; our numbers are tiny.
    return best_seq.tolist()
