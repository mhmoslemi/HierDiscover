def solve_lp(f, M, active_set):
    n = len(f)
    m = len(active_set)
    A = np.zeros((m, n))
    for i, k in enumerate(active_set):
        start = max(0, k - n + 1)
        end = min(k, n - 1)
        # j index from start to end
        # For each j, coefficient = f[k - j].
        j_indices = np.arange(start, end + 1)
        i_indices = k - j_indices
        A[i, j_indices] = f[i_indices]
    # b_ub = M for all rows.
    # bounds (0, None) for each variable.
    # Objective minimize -sum(b) => c = -np.ones(n)
    res = linprog(c=-np.ones(n), A_ub=A, b_ub=np.full(m, M), bounds=[(0, None)]*n, method='highs')
    # result.x is b
    return res

def _solve_lp(f, M, active_set):
    n = len(f)
    m = len(active_set)
    # Build dense matrix A
    A = np.zeros((m, n))
    for row_idx, k in enumerate(active_set):
        start = max(0, k - n + 1)
        end = min(k, n - 1)
        j = np.arange(start, end + 1)
        i = k - j
        A[row_idx, j] = f[i]
    # objective: minimize -sum(b)
    c = -np.ones(n)
    # bounds
    bounds = [(0, None)] * n
    # solve using linprog
    try:
        res = linprog(c, A_ub=A, b_ub=np.full(m, M), bounds=bounds, method='highs')
        if res.success:
            return res.x
        else:
            # if unbounded or other issue, return None
            # ensure we don't crash
            return None
    except Exception as e:
        print(f"[lp] linprog exception: {e}")
        return None

"""
Propose a step‑function (non‑negative sequence) with a very low value of

        2·n·max(conv(a,a)) / (sum(a)²)

The algorithm has three phases:

1. **Randomised 4‑parameter family** – a quick sweep over
   a_i = (i+α)^(-p)·(1+β·i/(n‑1))  (α≥0, 0.3≤p≤0.7, –0.9≤β≤2.5)
   each candidate is projected onto the simplex Σa_i = √(2n).

2. **Projected sub‑gradient descent** – keep the sum exactly √(2n) and
   move opposite to the sub‑gradient of the maximal convolution entry.
   The step size decays like 1/√t, occasional jitter helps escape plateaus.
   The loop stops when the improvement plateaus.

3. **Linear‑programming refinement** – for the current sequence `f`
   let `M = max(conv(f,f))`.  Solve a small LP

        maximise   Σ b_j
        subject to b_j ≥ 0
                   (f * b)[k] ≤ M   for a handful of “active’’ indices k

   (`f*b` is a linear convolution in `b`).  The LP spreads the mass to
   positions where `f` is small, increasing Σb while keeping the bound.
   After scaling `b` back to the required sum we mix it with `f`
   (convex combination) and re‑project.  The LP is solved with SciPy’s
   HiGHS interior‑point method; only a few constraints (≈15) are used,
   so each solve is fast.

The whole search respects a 1000 s time budget and returns the best
sequence found.  All numbers are non‑negative and the sequence is a plain
Python list as required by the harness.
"""

import time
import numpy as np
from scipy.optimize import linprog

# ----------------------------------------------------------------------
# Helper utilities (all top‑level, no closures)
# ----------------------------------------------------------------------


def _project_simplex(v: np.ndarray, target_sum: float) -> np.ndarray:
    """Euclidean projection of v onto {w ≥ 0, Σ w = target_sum}."""
    if target_sum < 0:
        raise ValueError("target_sum must be non‑negative")
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - target_sum
    ind = np.arange(1, len(v) + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.zeros_like(v)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def _full_convolution_fft(a: np.ndarray) -> np.ndarray:
    """Linear convolution a*a using a real FFT (full length 2·len(a)-1)."""
    n = len(a)
    size = 1 << ((2 * n - 1).bit_length())  # next power of two
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return conv


def _full_convolution_two(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Linear convolution a*b using a real FFT."""
    n = len(a) + len(b) - 1
    size = 1 << (n.bit_length())
    fa = np.fft.rfft(a, n=size)
    fb = np.fft.rfft(b, n=size)
    conv = np.fft.irfft(fa * fb, n=size)[:n]
    return conv


def _max_conv(a: np.ndarray) -> float:
    """Maximum entry of the linear convolution a*a."""
    return _full_convolution_fft(a).max()


def _make_candidate(n: int, p: float, offset: float, beta: float) -> np.ndarray:
    """Return a normalised candidate from the 4‑parameter family."""
    idx = np.arange(n, dtype=float)                     # i = 0 … n‑1
    base = np.power(idx + 1.0 + offset, -p)
    factor = 1.0 + beta * idx / (n - 1)
    raw = base * factor
    raw = np.maximum(raw, 0.0)
    target = np.sqrt(2.0 * n)
    return _project_simplex(raw, target)


def _subgradient_step(a: np.ndarray, max_inds: np.ndarray,
                      step: float, target_sum: float) -> np.ndarray:
    """One projected sub‑gradient step for max(conv(a,a))."""
    n = len(a)
    g = np.zeros_like(a)

    for k in max_inds:
        if k < n:
            # i = 0 … k
            g[: k + 1] += a[k::-1]
        else:
            # i = k-(n‑1) … n‑1
            start = k - (n - 1)
            idx = np.arange(start, n)
            g[start:] += a[k - idx]

    cand = a - step * g
    return _project_simplex(cand, target_sum)


def _solve_lp(f: np.ndarray, M: float, active_set: list[int]) -> np.ndarray | None:
    """
    Solve the LP:
        maximise   Σ b_j
        subject to b_j ≥ 0
                   (f * b)[k] ≤ M   for each k in active_set
    Returns the optimal b (numpy array) or None on failure.
    """
    n = len(f)
    m = len(active_set)
    A = np.zeros((m, n), dtype=float)

    for row, k in enumerate(active_set):
        # convolution row: (f * b)[k] = Σ_i f_i * b_{k-i}
        i_start = max(0, k - (n - 1))
        i_end = min(k, n - 1)
        j = np.arange(i_start, i_end + 1, dtype=int)          # b‑indices
        i = k - j                                            # corresponding f‑indices
        A[row, j] = f[i]

    c = -np.ones(n)          # minimise –Σb  → maximise Σb
    bounds = [(0, None)] * n
    b_ub = np.full(m, M)

    try:
        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds,
                      method='highs', options={"presolve": True})
        if res.success:
            return res.x
        return None
    except Exception as exc:
        print(f"[lp] linprog exception: {exc}")
        return None


def _lp_with_cutting(f: np.ndarray, M: float,
                    init_active: list[int],
                    max_iter: int = 5) -> np.ndarray | None:
    """
    Solve the LP with a simple cutting‑plane loop: after each solution
    check the full convolution f*b; any violated index is added to the
    active set and the LP is resolved (up to max_iter times).
    """
    active = list(set(init_active))
    for _ in range(max_iter):
        b = _solve_lp(f, M, active)
        if b is None:
            return None
        conv_fb = _full_convolution_two(f, b)
        viol = np.where(conv_fb > M + 1e-9)[0]
        if len(viol) == 0:
            return b
        # add only a few new constraints to keep the problem small
        active.extend(int(v) for v in viol[:10])
        active = list(set(active))
    return b

def propose_candidate(seed: int = 123, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative height sequence that minimises the evaluation
    function.  Returns the best sequence found as a plain list of floats.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 2.0          # safety margin

    # --------------------------------------------------------------
    # 0. Problem size – a few ten‑thousands give a finer shape.
    # --------------------------------------------------------------
    n = 12000
    target_sum = np.sqrt(2.0 * n)

    # --------------------------------------------------------------
    # 1. Warm‑start from a previously stored construction if possible.
    # --------------------------------------------------------------
    best_seq = None
    hs = globals().get("height_sequence_1")
    if isinstance(hs, list) and len(hs) == n:
        best_seq = np.array(hs, dtype=float)
        best_seq = _project_simplex(best_seq, target_sum)

    if best_seq is None:
        # simple power‑law start
        best_seq = _make_candidate(n, p=0.5, offset=0.0, beta=0.0)

    best_score = _max_conv(best_seq)                 # Σa = √(2n) ⇒ score = max conv
    print(f"[init] n={n:5d}  score={best_score:.6f}")

    # --------------------------------------------------------------
    # 2. Random sweep over the 4‑parameter family (≈30 % of budget)
    # --------------------------------------------------------------
    sweep_end = time.time() + 0.30 * (deadline - time.time())
    while time.time() < sweep_end:
        p = np.random.uniform(0.35, 0.65)
        off = np.random.uniform(0.0, 30.0)           # larger offset range
        beta = np.random.uniform(-1.2, 3.0)
        cand = _make_candidate(n, p, off, beta)
        sc = _max_conv(cand)
        if sc < best_score:
            best_score = sc
            best_seq = cand
            print(f"[sweep] new best {best_score:.6f}  p={p:.3f} α={off:.3f} β={beta:.3f}")
            if best_score < 1.5:
                break

    # --------------------------------------------------------------
    # 3. Projected sub‑gradient descent (until stagnation or time out)
    # --------------------------------------------------------------
    curr = best_seq.copy()
    step0 = 1.0
    iter_no = 0
    last_improve = 0
    while time.time() < deadline:
        iter_no += 1
        conv_full = _full_convolution_fft(curr)
        M = conv_full.max()

        if M < best_score:
            best_score = M
            best_seq = curr.copy()
            last_improve = iter_no
            print(f"[sgd] iter {iter_no:6d}  best {best_score:.6f}")

            if best_score < 1.5:
                break

        # stop sub‑gradient phase if no progress for a while
        if iter_no - last_improve > 5000:
            break

        eps = 1e-12 * M if M != 0 else 1e-12
        max_inds = np.where(conv_full >= M - eps)[0]

        step = step0 / np.sqrt(iter_no)
        nxt = _subgradient_step(curr, max_inds, step, target_sum)

        # occasional jitter to escape flat regions
        if iter_no % 2000 == 0 and np.random.rand() < 0.3:
            jitter = (np.random.rand(n) - 0.5) * 0.02 * nxt
            nxt = nxt + jitter
            nxt = _project_simplex(nxt, target_sum)

        curr = nxt

    # --------------------------------------------------------------
    # 4. Linear‑programming refinement (remaining time)
    # --------------------------------------------------------------
    while time.time() < deadline:
        seq = best_seq.copy()
        conv_seq = _full_convolution_fft(seq)
        M = conv_seq.max()
        if M < best_score:
            best_score = M
            best_seq = seq.copy()
            print(f"[lp] improved best {best_score:.6f}")

        # pick the K largest convolution entries as active constraints
        K = 15
        # argpartition is O(N) and gives the K biggest indices
        top_k = np.argpartition(conv_seq, -K)[-K:]
        active = list(set(int(v) for v in top_k))

        b = _lp_with_cutting(seq, M, active, max_iter=4)
        if b is None:
            # if LP failed, fall back to a tiny random perturbation
            jitter = (np.random.rand(n) - 0.5) * 0.01 * seq
            seq = _project_simplex(seq + jitter, target_sum)
            best_seq = seq
            continue

        sum_b = b.sum()
        # we need a larger total mass to achieve improvement
        if sum_b <= target_sum * 1.001:
            # no useful gain – apply a small random move and continue
            jitter = (np.random.rand(n) - 0.5) * 0.01 * seq
            seq = _project_simplex(seq + jitter, target_sum)
            best_seq = seq
            continue

        # scale to the required sum and mix with the current sequence
        g = b * (target_sum / sum_b)
        t_mix = 0.35                           # mixing weight
        new_seq = (1 - t_mix) * seq + t_mix * g
        new_seq = _project_simplex(new_seq, target_sum)

        new_score = _max_conv(new_seq)
        if new_score < best_score:
            best_score = new_score
            best_seq = new_seq
            print(f"[lp] mix improved to {best_score:.6f}")

        # update the working sequence for the next LP round
        # (a few sub‑gradient steps are occasionally inserted)
        if np.random.rand() < 0.2:
            # a couple of cheap gradient steps to fine‑tune
            conv_full = _full_convolution_fft(new_seq)
            M = conv_full.max()
            eps = 1e-12 * M if M != 0 else 1e-12
            max_inds = np.where(conv_full >= M - eps)[0]
            step = 0.5 / np.sqrt(iter_no + 1)
            new_seq = _subgradient_step(new_seq, max_inds, step, target_sum)

        best_seq = new_seq
        iter_no += 1

    print(f"Finished – best score {best_score:.6f}")
    return best_seq.tolist()
