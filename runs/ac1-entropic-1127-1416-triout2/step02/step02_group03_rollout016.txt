        if M < best_score:

"""
Propose a step‑function with a very low evaluation value.

The algorithm consists of three phases:

1️⃣  Warm‑start – try to reuse the global `height_sequence_1` if it has the
    right length, otherwise start from a simple power‑law sequence.

2️⃣  Random parametric sweep – explore a 4‑parameter family
    a_i = (i+α)^{-p}·(1+β·i/(n‑1)).  For each candidate the vector is projected
    onto the simplex {x≥0, Σx = √(2·n)}.  The best candidate from this cheap sweep
    becomes the seed for the final optimisation.

3️⃣  Adaptive Adam‑type projected descent – we minimise
    max(conv(a,a)) (which equals the final score because Σa is fixed).
    The sub‑gradient is the sum of the shifted copies of `a` belonging to the
    currently maximal convolution entries.  An Adam update is applied and the
    result is projected back onto the simplex.  The step‑size adapts automatically.
    Periodic jitter helps to escape plateaus.

4️⃣  Optional greedy local mass‑transfer search – while time permits we try
    tiny transfers of mass between two coordinates; the convolution is updated
    incrementally (O(n)) and the move is kept only if it reduces the max
    convolution.

The whole routine respects a 1000 s wall‑time budget and returns the best
sequence found as a plain Python list of floats.
"""

import time
import numpy as np

# ------------------------------------------------------------
# Helper utilities
# ------------------------------------------------------------

def _project_simplex(v: np.ndarray, target_sum: float) -> np.ndarray:
    """ Euclidean projection of v onto {x ≥ 0, Σx = target_sum}. """
    if target_sum < 0:
        raise ValueError("target_sum must be non‑negative")
    # Duchi et al. (2008) – O(n log n)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - target_sum
    ind = np.arange(1, len(v) + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.zeros_like(v)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def _full_convolution_fft(a: np.ndarray) -> np.ndarray:
    """ Linear convolution a*a via a real FFT (returns length 2·len(a)-1). """
    n = len(a)
    size = 1 << ((2 * n - 1).bit_length())        # next power of two
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return conv


def _max_conv(a: np.ndarray) -> float:
    """Maximum entry of the self‑convolution of a."""
    return _full_convolution_fft(a).max()


def _make_candidate(n: int, p: float, offset: float, beta: float) -> np.ndarray:
    """
    Build a raw sequence
        a_i = (i + 1 + offset)^{-p} * (1 + β·i/(n‑1)),
    clip negatives and project onto the simplex Σa = √(2·n).
    """
    idx = np.arange(n, dtype=float)
    base = np.power(idx + 1.0 + offset, -p)
    factor = 1.0 + beta * idx / (n - 1)
    raw = base * factor
    raw = np.maximum(raw, 0.0)
    target = np.sqrt(2.0 * n)
    return _project_simplex(raw, target)


def _subgradient_at_max(a: np.ndarray, max_inds: np.ndarray) -> np.ndarray:
    """
    Compute a sub‑gradient of max(conv(a,a)) using only the indices where
    the convolution attains its current maximum.
    """
    n = len(a)
    g = np.zeros_like(a)
    for k in max_inds:
        if k < n:
            # i = 0 … k
            g[: k + 1] += a[k :: -1]
        else:
            # i = k-(n-1) … n-1
            start = k - (n - 1)
            idx = np.arange(start, n)
            g[start:] += a[k - idx]
    return g


# ------------------------------------------------------------
# Main search routine
# ------------------------------------------------------------

def propose_candidate(seed: int = 123, budget_s: int = 1000) -> list[float]:
    """
    Search for a non‑negative step‑function (list of heights) that minimises

        2 * n * max(conv(a, a)) / (sum(a) ** 2)

    under the constraints a_i ≥ 0 and Σ a_i = √(2·n).
    The function respects the 1000 s time limit and returns the best sequence
    found (as a plain Python list of floats).
    """
    np.random.seed(seed)
    # safety margin – leave a couple of seconds for the harness
    deadline = time.time() + budget_s - 5.0

    # --------------------------------------------------------
    # 0. Choose a problem size.
    # --------------------------------------------------------
    # 7000–9000 elements give a good trade‑off between expressiveness
    # and per‑iteration cost.  8000 is a comfortable default.
    n = 8000
    target_sum = np.sqrt(2.0 * n)

    # --------------------------------------------------------
    # 1. Warm‑start (reuse previous construction if possible)
    # --------------------------------------------------------
    best_seq = None
    hs = globals().get("height_sequence_1")
    if isinstance(hs, list) and len(hs) == n:
        best_seq = np.asarray(hs, dtype=float)
        best_seq = _project_simplex(best_seq, target_sum)

    if best_seq is None:
        # plain power‑law start
        best_seq = _make_candidate(n, p=0.5, offset=0.0, beta=0.0)

    best_score = _max_conv(best_seq)       # because Σa = √(2n), score = max conv
    print(f"[init] n={n:5d}  score={best_score:.6f}")

    # --------------------------------------------------------
    # 2. Random parametric sweep (≈20 % of the total budget)
    # --------------------------------------------------------
    sweep_deadline = time.time() + 0.20 * (deadline - time.time())
    while time.time() < sweep_deadline:
        p = np.random.uniform(0.35, 0.80)               # exponent
        off = np.random.uniform(0.0, 12.0)              # offset α
        beta = np.random.uniform(-1.0, 3.0)             # tilt β
        cand = _make_candidate(n, p, off, beta)

        # also try the symmetric version (mirrored + averaged)
        sym = (cand + cand[::-1]) * 0.5
        sym = _project_simplex(sym, target_sum)

        for trial in (cand, sym):
            sc = _max_conv(trial)
            if sc < best_score:
                best_score = sc
                best_seq = trial
                print(f"[sweep] new best {best_score:.6f}  p={p:.3f} α={off:.3f} β={beta:.3f}")

        if best_score < 1.50:          # early exit if we are already very good
            break

    # --------------------------------------------------------
    # 3. Adaptive Adam‑type projected descent
    # --------------------------------------------------------
    a = best_seq.copy()
    m = np.zeros_like(a)               # first moment
    v = np.zeros_like(a)               # second moment
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8
    step_eta = 0.45                    # base learning rate
    iter_cnt = 0

    while time.time() < deadline:
        iter_cnt += 1
        conv = _full_convolution_fft(a)
        M = conv.max()

        if M < best_score:
            best_score = M
            best_seq = a.copy()
            print(f"[adam] iter {iter_cnt:6d}  new best {best_score:.6f}")
            if best_score < 1.503:
                break

        # indices where the convolution is (essentially) maximal
        tol = 1e-12 * M if M != 0 else 1e-12
        max_inds = np.where(conv >= M - tol)[0]

        # sub‑gradient w.r.t the current max‑set
        g = _subgradient_at_max(a, max_inds)

        # Adam update (projected after the step)
        m = beta1 * m + (1.0 - beta1) * g
        v = beta2 * v + (1.0 - beta2) * (g * g)

        m_hat = m / (1.0 - beta1 ** iter_cnt)
        v_hat = v / (1.0 - beta2 ** iter_cnt)

        step = step_eta * m_hat / (np.sqrt(v_hat) + eps)
        a = a - step
        a = _project_simplex(a, target_sum)

        # occasional jitter to shake the iterate out of plateaus
        if iter_cnt % 80 == 0 and np.random.rand() < 0.35:
            jitter = (np.random.rand(n) - 0.5) * 0.02 * a
            a += jitter
            a = _project_simplex(a, target_sum)

    # --------------------------------------------------------
    # 4. Greedy local mass‑transfer refinement (use any remaining time)
    # --------------------------------------------------------
    remaining = deadline - time.time()
    if remaining > 2.0:
        # keep the current convolution and run a simple hill‑climb
        conv = _full_convolution_fft(a)
        cur_M = conv.max()
        anneal_iter = 0
        while time.time() < deadline:
            anneal_iter += 1
            i = np.random.randint(0, n)
            if a[i] <= 0.0:
                continue
            # transfer a small fraction of mass from i to j
            delta = a[i] * np.random.rand() * 0.08   # up to 8 %
            if delta <= 0.0:
                continue
            j = np.random.randint(0, n - 1)
            if j == i:
                continue

            # incremental update of the convolution (O(n))
            #   new_conv = conv + 2·δ·(a shifted by j – a shifted by i)
            upd = np.zeros_like(conv)
            upd[i:i + n] -= 2.0 * delta * a
            upd[j:j + n] += 2.0 * delta * a
            new_conv = conv + upd
            new_M = new_conv.max()

            # accept only improvements (greedy)
            if new_M < cur_M:
                a[i] -= delta
                a[j] += delta
                conv = new_conv
                cur_M = new_M
                if cur_M < best_score:
                    best_score = cur_M
                    best_seq = a.copy()
                    print(f"[anneal] iter {anneal_iter:6d}  new best {best_score:.6f}")

    print(f"Finished – total iterations {iter_cnt} – best score {best_score:.6f}")
    return best_seq.tolist()
