def propose_candidate(seed=42, budget_s=1000, **kwargs):
    import time, numpy as np
    rng = np.random.default_rng(seed)
    deadline = time.time() + budget_s - 0.1

    # Determine length: maybe try 15000 if possible else fallback to 8000
    n = 15000
    best_score = float('inf')
    best_seq = None

    # Precompute FFT length
    conv_len = 2*n - 1
    fft_len = 1 << (conv_len - 1).bit_length()

    def score_and_conv(a):
        # Ensure sum = 1
        a = a / a.sum()
        A = np.fft.rfft(a, n=fft_len)
        c = np.fft.irfft(A*A, n=fft_len)[:conv_len]
        max_c = c.max()
        # Score = 2 * n * max_c
        return 2.0 * n * max_c, c

    # Start with binomial or uniform as initial
    # We'll include multiple restarts
    restart = 0
    while time.time() < deadline:
        # restart loop maybe each 3000 iterations
        # Choose initial a:
        if restart == 0 and 'height_sequence_1' in globals():
            a = np.array(height_sequence_1, dtype=np.float64)
            if len(a) != n:
                # Resample or interpolate to n length
                # Use linear interpolation
                x_old = np.linspace(0, 1, len(a))
                x_new = np.linspace(0, 1, n)
                a = np.interp(x_new, x_old, a)
        else:
            # random Dirichlet
            # param = 0.5 maybe yields sparse, 1 yields uniform, >1 yields concentrated
            alpha = 0.7
            a = rng.dirichlet(np.full(n, alpha))
        # Normalize
        a = a / a.sum()

        cur_score, cur_conv = score_and_conv(a)
        # local search
        temperature = 1.0
        # step size for random transfer: relative fraction e.g., 0.1
        step_fraction = 0.05
        inner_iter = 0
        while time.time() < deadline and inner_iter < 5000:
            # Keep each inner loop small; break if no improvement for while.
            # Random move: choose i,j
            i, j = rng.integers(0, n, size=2)
            if i == j:
                continue
            # Choose amount
            max_transfer = min(a[i], step_fraction)
            delta = rng.random() * max_transfer
            if delta == 0:
                continue

            a_new = a.copy()
            a_new[i] -= delta
            a_new[j] += delta

            # evaluate
            new_score, _ = score_and_conv(a_new)

            # Accept with Metropolis
            if new_score < cur_score or rng.random() < np.exp(-(new_score - cur_score) / temperature):
                a = a_new
                cur_score = new_score
                # record best

"""
Propose a step‑function sequence that (heuristically) minimises

    2 * n * max(a * a) / (sum(a) ** 2)

where * denotes linear convolution.  
The algorithm is a hybrid of three ideas:

1. **Multiple restarts** – we try several initialisations
   (binomial, uniform, random Dirichlet and, when available,
   the previously discovered `height_sequence_1`).

2. **Projected gradient descent** – the sub‑gradient of the
   current maximal convolution entry is computed exactly
   (by scanning the active indices).  A line‑search along the
   feasible direction (sum‑zero, non‑negative) is performed,
   followed by a Euclidean projection onto the probability simplex.

3. **Simulated‑annealing fine‑tuning** – after the deterministic phase
   we run a classic Metropolis walk that transfers a small random
   mass between two positions.  The temperature decays slowly,
   allowing occasional uphill moves to escape local minima.

All convolutions are evaluated by FFT (real‑valued rfft) which
runs in O(n log n) and makes it feasible to work with long
sequences (≈ 15000 entries).  The best sequence seen before the
time budget expires is returned as a plain Python list.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
def binomial_sequence(length: int) -> np.ndarray:
    """
    Central–binomial coefficients   c_k = C(2k, k) / 4**k,
    k = 0 … length‑1.  They obey the recurrence used in the reference
    implementation.
    """
    seq = np.empty(length, dtype=np.float64)
    seq[0] = 1.0
    for k in range(1, length):
        seq[k] = seq[k - 1] * ((2 * k - 1) * (2 * k)) / (4.0 * k * k)
    return seq


# ----------------------------------------------------------------------
def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a non‑negative vector onto the probability simplex
    (non‑negative entries that sum to one).
    """
    v = np.maximum(v, 0.0)
    s = v.sum()
    if s == 0.0:
        # uniform fallback
        return np.full_like(v, 1.0 / v.size)
    return v / s


# ----------------------------------------------------------------------
def compute_score_and_conv(a: np.ndarray,
                           n: int,
                           fft_len: int,
                           conv_len: int) -> tuple[float, np.ndarray]:
    """
    Returns (score, full convolution) for a vector a that already satisfies
    sum(a) == 1.  Convolution is performed by FFT for speed.
    """
    # real FFT of a (zero‑padded to fft_len)
    A = np.fft.rfft(a, n=fft_len)
    # element‑wise product = convolution in frequency domain
    conv_full = np.fft.irfft(A * A, n=fft_len)[:conv_len]
    max_val = conv_full.max()
    # the evaluation function is scale‑invariant, so sum(a)=1 → score = 2*n*max
    score = 2.0 * n * max_val
    return score, conv_full


# ----------------------------------------------------------------------
def compute_gradient(a: np.ndarray,
                     conv: np.ndarray,
                     max_val: float) -> np.ndarray:
    """
    Sub‑gradient of the maximal convolution entry.
    For every index i we sum a_{k-i} over all active indices k
    (those where conv[k] is (almost) maximal).
    """
    n = a.shape[0]
    # tolerance – we treat values within 1e-12 of the maximum as active
    tol = max_val - 1e-12
    active = np.flatnonzero(conv >= tol)
    g = np.zeros_like(a)
    for k in active:
        lo = max(0, k - (n - 1))
        hi = min(n - 1, k)
        idx = np.arange(lo, hi + 1, dtype=np.int64)
        # a_{k-i} for i in idx
        g[idx] += a[k - idx]
    return g


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑height sequence that (heuristically)
    minimises ``evaluate_sequence``.  The function runs for at most
    ``budget_s`` seconds and returns the best sequence encountered.
    """
    rng = np.random.default_rng(seed)
    deadline = time.time() + budget_s - 0.1   # tiny safety margin

    # --------------------------------------------------------------
    # Choose a problem size.
    # If a previous solution exists we reuse its length; otherwise we take a
    # generous default that still fits comfortably in memory.
    global height_sequence_1  # may be undefined
    try:
        n = len(height_sequence_1)          # type: ignore[name-defined]
    except Exception:
        n = 15000

    conv_len = 2 * n - 1
    fft_len = 1 << (conv_len - 1).bit_length()   # next power of two

    best_score = float('inf')
    best_seq = None

    # --------------------------------------------------------------
    # Different ways to initialise a restart.
    init_methods = ['binomial', 'uniform', 'random']
    # If a previous construction is available we prepend it.
    if 'height_sequence_1' in globals():
        init_methods.insert(0, 'previous')

    restart = 0
    while time.time() < deadline:
        # ---- choose initial vector ---------------------------------
        method = init_methods[restart % len(init_methods)]

        if method == 'previous':
            a = np.array(height_sequence_1, dtype=np.float64)
            if len(a) != n:
                # simple linear interpolation to the target length
                x_old = np.linspace(0.0, 1.0, len(a), endpoint=False)
                x_new = np.linspace(0.0, 1.0, n, endpoint=False)
                a = np.interp(x_new, x_old, a)
        elif method == 'binomial':
            a = binomial_sequence(n).astype(np.float64)
        elif method == 'uniform':
            a = np.full(n, 1.0 / n, dtype=np.float64)
        else:   # random Dirichlet (moderately sparse)
            alpha = 0.7
            a = rng.dirichlet(np.full(n, alpha))

        # Normalise to the probability simplex
        a = project_to_simplex(a)

        # ---- evaluate initial point --------------------------------
        cur_score, cur_conv = compute_score_and_conv(a, n, fft_len, conv_len)

        # ---- deterministic gradient descent phase ------------------
        step_alpha = 0.2          # initial step magnitude
        step_decay = 0.9995
        min_alpha = 1e-5
        gd_iter = 0
        max_gd_iters = 4000

        while time.time() < deadline and gd_iter < max_gd_iters:
            max_val = cur_conv.max()
            # gradient of the current maximum entry (or entries)
            g = compute_gradient(a, cur_conv, max_val)

            # feasible direction: keep the sum constant (zero‑sum)
            sum_g = g.sum()
            d = -g + (sum_g / n)

            # ----- simple backtracking line search -----------------
            improved = False
            for factor in (1.0, 0.5, 0.25, 0.125):
                step = step_alpha * factor
                a_candidate = a + step * d
                a_candidate = np.maximum(a_candidate, 0.0)
                a_candidate = project_to_simplex(a_candidate)

                new_score, new_conv = compute_score_and_conv(
                    a_candidate, n, fft_len, conv_len)

                if new_score < cur_score:
                    a = a_candidate
                    cur_score = new_score
                    cur_conv = new_conv
                    improved = True
                    break

            if not improved:
                # if no improvement, shrink the step more aggressively
                step_alpha *= 0.5

            # decay step size slowly, but never go below the floor
            step_alpha = max(min_alpha, step_alpha * step_decay)

            # keep global best
            if cur_score < best_score:
                best_score = cur_score
                best_seq = a.copy()
                elapsed = time.time() - (deadline - budget_s)
                print(f"[best] restart={restart} gd_iter={gd_iter} "
                      f"score={best_score:.9f} time={elapsed:.2f}s")

            gd_iter += 1

        # ---- simulated‑annealing fine‑tuning phase ---------------
        temperature = 1.0
        anneal_iter = 0
        max_anneal_iters = 8000

        while time.time() < deadline and anneal_iter < max_anneal_iters:
            i, j = rng.integers(0, n, size=2)
            if i == j:
                continue

            # move a tiny random fraction of mass from i to j
            max_transfer = min(a[i], 0.05)          # at most 5 % of a[i]
            delta = rng.random() * max_transfer
            if delta == 0.0:
                continue

            a_candidate = a.copy()
            a_candidate[i] -= delta
            a_candidate[j] += delta

            # (sum unchanged, positivity preserved by construction)
            new_score, new_conv = compute_score_and_conv(
                a_candidate, n, fft_len, conv_len)

            # Metropolis acceptance
            if new_score < cur_score or rng.random() < np.exp(-(new_score - cur_score) / temperature):
                a = a_candidate
                cur_score = new_score
                cur_conv = new_conv
                if cur_score < best_score:
                    best_score = cur_score
                    best_seq = a.copy()
                    elapsed = time.time() - (deadline - budget_s)
                    print(f"[best] restart={restart} anneal_iter={anneal_iter} "
                          f"score={best_score:.9f} time={elapsed:.2f}s")

            # geometric cooling
            temperature *= 0.9999
            anneal_iter += 1

        # early stopping if we have already reached the target bound
        if best_score <= 1.503 + 1e-6:
            break

        restart += 1

    # ------------------------------------------------------------------
    # If for any reason we never stored a candidate, fall back to a
    # normalised binomial sequence.
    if best_seq is None:
        best_seq = binomial_sequence(n)
        best_seq = best_seq / best_seq.sum()

    # The evaluation function will clip values to [0,1000]; ours are far
    # smaller, so the returned list is safe.
    return best_seq.tolist()
