def gradient_descent(a, max_iters, start_time, deadline):
    velocity = np.zeros_like(a)
    beta = 0.9
    for it in range(max_iters):
        if time.time() >= deadline: break
        # compute conv via FFT
        fft_a = np.fft.rfft(a, n=2*n-1)
        conv = np.fft.irfft(fft_a * fft_a, n=2*n-1)
        max_b = conv.max()
        cur_score = 2.0 * n * max_b
        if cur_score < best_score: update best.
        # active indices
        eps = max_b * 1e-5
        active_idx = np.where(conv >= max_b - eps)[0]
        # compute gradient using average as before
        g = np.zeros_like(a)
        for k in active_idx:
            low = max(0, k - n + 1)
            high = min(n - 1, k)
            idx = np.arange(low, high + 1, dtype=int)
            partner = k - idx
            g[idx] += 2.0 * a[partner]
        g /= active_idx.size
        # momentum update
        velocity = beta * velocity + (1 - beta) * g
        step = eta / np.sqrt(it + 1)
        a = simplex_projection(a - step * velocity)
    return a

def simulated_annealing(a, cur_score, start_time, deadline):
    best_a = a.copy()
    best_score = cur_score
    # parameters
    total = deadline - start_time
    T0 = cur_score * 0.1 # initial temperature proportional to objective.
    while time.time() < deadline:
        # time fraction
        elapsed = time.time() - start_time
        frac = elapsed / total
        T = T0 * (1 - frac)  # linear cooling
        # propose move
        i, j = np.random.randint(n, size=2)
        if i == j:
            continue
        eps = np.random.rand() * min(a[i], 0.5) * 0.5  # up to half of a_i
        b = a.copy()
        b[i] -= eps
        b[j] += eps
        # compute objective
        # conv via FFT
        fft_b = np.fft.rfft(b, n=2*n-1)
        convb = np.fft.irfft(fft_b * fft_b, n=2*n-1)
        maxb = convb.max()
        new_score = 2.0 * n * maxb
        delta = new_score - cur_score
        if delta < 0 or np.exp(-delta / (T + 1e-12)) > np.random.rand():
            a = b
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_a = a.copy()
                print(f"Anneal new best {best_score:.6f} at iter {iteration}")
        # optionally limit epsilon smaller as time passes
    return best_a, best_score

"""
Propose a non‑negative height sequence that minimises

    2·n·max(conv(a,a)) / (sum(a)**2)

The algorithm works in three stages:

1. **Diverse deterministic seeds** – uniform, linear ramp, power‑law,
   exponential, asymmetric Gaussian and a grid of Beta‑shaped
   distributions (including random Dirichlet draws).  All seeds are
   normalised to sum to one and the best few are kept.

2. **Momentum projected sub‑gradient descent** – starting from the best
   seeds we run a long gradient loop on the simplex.  The gradient is
   obtained from the *k* largest convolution entries (default k=5) and
   is combined with a simple momentum term.  After each step the iterate
   is projected back onto the probability simplex using the Euclidean
   projection of Duchi et al.

3. **Simulated annealing** – the current best vector is further refined
   by random mass transfers between two indices, accepted with a
   temperature that decreases linearly to zero.  This phase can escape
   shallow local minima left by the deterministic descent.

The whole search respects the 1000 s budget; a safety margin of a few
seconds is kept.  The returned list is a plain Python list of floats,
ready for `evaluate_sequence`.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper functions (top‑level – no closures)
# ----------------------------------------------------------------------
def simplex_projection(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of a vector onto the probability simplex."""
    v = np.asarray(v, dtype=float)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1] - 1               # zero‑based index
    theta = cssv[cond][-1] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def compute_objective(a: np.ndarray) -> float:
    """Return 2·n·max(a*a) assuming sum(a)=1 (exact evaluation)."""
    n = a.shape[0]
    fft = np.fft.rfft(a, n=2 * n - 1)
    conv = np.fft.irfft(fft * fft, n=2 * n - 1)
    return 2.0 * n * conv.max()


def beta_sequence(alpha: float, beta: float, n: int) -> np.ndarray:
    """Discretised Beta(α,β) on n points, normalised to sum‑one."""
    x = (np.arange(n) + 0.5) / n
    logw = (alpha - 1.0) * np.log(x) + (beta - 1.0) * np.log(1.0 - x)
    logw = np.clip(logw, -700.0, 700.0)          # avoid overflow
    w = np.exp(logw)
    total = w.sum()
    if total == 0 or not np.isfinite(total):
        w = np.ones_like(w)
        total = w.sum()
    return w / total


def power_sequence(p: float, n: int) -> np.ndarray:
    """a_i ∝ (n-i)^p, normalised."""
    v = (np.arange(n, 0, -1, dtype=float)) ** p
    return v / v.sum()


def exponential_sequence(r: float, n: int) -> np.ndarray:
    """a_i ∝ r**i, normalised."""
    v = r ** np.arange(n)
    return v / v.sum()


def gaussian_sequence(sigma: float, n: int) -> np.ndarray:
    """Asymmetric Gaussian centred at the left edge."""
    i = np.arange(n)
    v = np.exp(-i ** 2 / (2.0 * sigma * sigma))
    return v / v.sum()


# ----------------------------------------------------------------------
# Optimisation primitives
# ----------------------------------------------------------------------
def run_gradient_descent(seed_vec: np.ndarray,
                         n: int,
                         deadline: float,
                         best_score: float,
                         best_a: np.ndarray) -> tuple[float, np.ndarray]:
    """
    Momentum projected sub‑gradient descent on the simplex.
    Returns the (possibly) improved best_score and best_a.
    """
    a = seed_vec.copy()
    vel = np.zeros_like(a)
    eta = 0.9
    beta_mom = 0.9
    iteration = 0
    # number of active convolution entries to use for the sub‑gradient
    active_k = 5

    while time.time() < deadline:
        # convolution via FFT
        fft_a = np.fft.rfft(a, n=2 * n - 1)
        conv = np.fft.irfft(fft_a * fft_a, n=2 * n - 1)
        max_b = conv.max()
        cur_score = 2.0 * n * max_b

        # keep global best
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"Grad iter {iteration:5d}: new best {best_score:.6f}")

        # indices of the k largest convolution entries
        if active_k < conv.size:
            idx = np.argpartition(conv, -active_k)[-active_k:]
        else:
            idx = np.arange(conv.size)

        # build averaged sub‑gradient for the chosen entries
        g = np.zeros_like(a)
        for k in idx:
            low = max(0, k - n + 1)
            high = min(n - 1, k)
            ind = np.arange(low, high + 1, dtype=int)
            partner = k - ind
            g[ind] += 2.0 * a[partner]
        g /= float(len(idx))

        # momentum update and projected step
        vel = beta_mom * vel + (1.0 - beta_mom) * g
        step = eta / np.sqrt(iteration + 1.0)
        a = simplex_projection(a - step * vel)

        iteration += 1

    return best_score, best_a


def run_simulated_annealing(start_vec: np.ndarray,
                            n: int,
                            deadline: float,
                            best_score: float,
                            best_a: np.ndarray) -> tuple[float, np.ndarray]:
    """
    Small temperature‑controlled random walk that transfers a tiny amount
    of mass between two positions.  Accepts worsening moves with a
    probability decreasing with temperature.
    """
    cur_a = start_vec.copy()
    cur_score = best_score
    total_time = deadline - time.time()
    start_time = time.time()
    # initial temperature – a modest fraction of the current score
    T0 = max(cur_score * 0.2, 1e-8)

    while time.time() < deadline:
        elapsed = time.time() - start_time
        frac = elapsed / total_time
        T = max(T0 * (1.0 - frac), 1e-12)

        # propose a transfer of mass
        i, j = np.random.randint(n, size=2)
        if i == j:
            continue
        max_eps = cur_a[i]
        # the step size shrinks as the annealing proceeds
        eps = np.random.rand() * max_eps * (0.5 * (1.0 - frac) + 0.01)
        new_a = cur_a.copy()
        new_a[i] -= eps
        new_a[j] += eps

        # evaluate candidate (objective is scale‑invariant because sum=1)
        new_score = compute_objective(new_a)

        delta = new_score - cur_score
        if delta < 0 or np.exp(-delta / T) > np.random.rand():
            cur_a = new_a
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_a = cur_a.copy()
                print(f"Anneal iter {int(elapsed)}: new best {best_score:.6f}")

    return best_score, best_a
