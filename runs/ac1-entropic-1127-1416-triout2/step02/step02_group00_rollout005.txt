def compute_softmax_weights(conv: np.ndarray, lam: float) -> np.ndarray:
    # conv shape (2n-1)
    x = lam * conv
    mx = np.max(x)
    w = np.exp(x - mx)
    w /= w.sum()
    return w

def gradient_from_weights(a: np.ndarray, w: np.ndarray) -> np.ndarray:

def optimize_sequence(initial_a: np.ndarray, deadline: float,
                     lam_start=5.0, lam_end=80.0, lam_ramp_iters=5000,
                     eta0=0.5, max_iter=200000,
                     noise_every=0, noise_scale=0.0) -> Tuple[np.ndarray, float]:
    a = initial_a.astype(float).copy()
    a = a / a.sum()
    best_a = a.copy()
    best_sc = compute_score(a)
    n = len(a)
    for it in range(1, max_iter+1):
        if time.time() >= deadline:
            break
        # schedule lambda
        lam = lam_start + (lam_end - lam_start) * min(1.0, it / lam_ramp_iters)
        conv = convolution_fft(a, a)
        w = compute_softmax_weights(conv, lam)
        grad = gradient_from_weights(a, w)
        # step size schedule
        eta = eta0 / np.sqrt(it)
        # multiplicative update:
        a = a * np.exp(-eta * grad)
        # optionally add noise
        if noise_every and (it % noise_every == 0):
            a += np.random.randn(n) * noise_scale
            a = np.maximum(a, 0.0)
        # re-normalize to simplex
        sum_a = a.sum()
        if sum_a <= 0:
            a = np.full(n, 1.0/n)
        else:
            a /= sum_a

        cur_sc = compute_score(a)
        if cur_sc < best_sc:
            best_sc = cur_sc
            best_a = a.copy()
        # occasional logging
        if it % 2000 == 0:
            print(f"    iter={it} lam={lam:.1f} cur={cur_sc:.6f} best={best_sc:.6f}")
    return best_a, best_sc

"""
Hybrid soft‑max gradient descent for the step‑function minimisation problem.

The algorithm works in three stages:

1. **Initial families** –  Several parametric families (uniform,
   arcsine, power‑law, beta‑like, Dirichlet) are sampled for a chosen
   length *n* and the best few are kept.

2. **Soft‑max gradient optimisation** –  Starting from each kept
   candidate we run an exponentiated‑gradient descent on the smooth
   approximation

        Φ_λ(a) = (2·n / Σ a_i²) · ( 1/λ  · log Σ_k exp( λ·(a*a)_k) )

   The gradient of the inner log‑sum‑exp term is obtained by a single
   convolution with the soft‑max weights, computed efficiently via FFT.
   The step size decays as 1/√t and a tiny amount of Gaussian noise is
   injected every few thousand steps to escape shallow plateaus.

3. **Final polishing** –  The current global best sequence is refined
   once more with a longer schedule and a larger final λ.

All operations stay on the probability simplex (positivity + unit sum)
by using multiplicative updates followed by normalisation.
The routine respects the allotted runtime (default 1000 s) and always
returns a feasible sequence of non‑negative floats.
"""

import time
import numpy as np
from typing import List, Tuple

# ----------------------------------------------------------------------
# Low‑level numeric utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Linear convolution via real FFT (full length, zero‑padded to power‑of‑2)."""
    m = len(a) + len(b) - 1
    n_fft = next_pow2(m)
    A = np.fft.rfft(a, n=n_fft)
    B = np.fft.rfft(b, n=n_fft)
    C = A * B
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


def compute_score(a: np.ndarray) -> float:
    """
    Official objective (scale‑invariant):
        2·n·max(a*a) / (sum(a)²)
    a is assumed to be non‑negative; the function works for any vector.
    """
    n = len(a)
    conv = convolution_fft(a, a)
    max_val = float(np.max(conv))
    s = float(np.sum(a))
    if s < 1e-12:
        return float("inf")
    return 2.0 * n * max_val / (s * s)


def arcsine_initial(n: int) -> np.ndarray:
    """Classic arcsine density (discrete, normalised)."""
    i = np.arange(n, dtype=float)
    eps = 0.25
    vals = 1.0 / np.sqrt((i + 0.5) * (n - 0.5 - i) + eps)
    return vals / vals.sum()


def generate_shape(n: int, p: float, eps: float) -> np.ndarray:
    """Symmetric power‑law shape a_i ∝ ((i+ε)(n‑1‑i+ε))⁻ᵖ."""
    i = np.arange(n, dtype=float)
    base = ((i + eps) * (n - 1 - i + eps)) ** (-p)
    total = base.sum()
    if total == 0.0:
        return np.full(n, 1.0 / n)
    return base / total


def softmax_weights(conv: np.ndarray, lam: float) -> np.ndarray:
    """Soft‑max over the convolution entries (numerically stable)."""
    x = lam * conv
    mx = np.max(x)
    w = np.exp(x - mx)
    w /= w.sum()
    return w


def gradient_from_weights(a: np.ndarray, w: np.ndarray) -> np.ndarray:
    """
    Gradient of Σ_k w_k·(a*a)_k w.r.t a.
    Implements   g_j = 2· Σ_m a_m · w_{m+j}.
    Computed via one FFT‑based convolution.
    """
    w_rev = w[::-1]
    conv = convolution_fft(a, w_rev)               # length = 3·n‑2
    idx = 2 * len(a) - 2                            # points to position 2n‑2
    grads = 2.0 * conv[idx - np.arange(len(a))]
    return grads


def normalize(v: np.ndarray) -> np.ndarray:
    """Project a non‑negative vector onto the probability simplex."""
    total = v.sum()
    if total <= 0.0:
        return np.full_like(v, 1.0 / len(v))
    return v / total


# ----------------------------------------------------------------------
# Optimiser (soft‑max + exponentiated gradient)
# ----------------------------------------------------------------------


def optimise_sequence(
    initial_a: np.ndarray,
    deadline: float,
    *,
    lam_start: float = 5.0,
    lam_end: float = 80.0,
    lam_ramp_iters: int = 4000,
    eta0: float = 0.6,
    max_iter: int = 200_000,
    noise_every: int = 0,
    noise_scale: float = 0.0,
) -> Tuple[np.ndarray, float]:
    """
    Exponentiated‑gradient descent on a smooth approximation of the max‑convolution.
    Returns the best vector seen and its (non‑smooth) score.
    """
    a = normalize(initial_a.astype(float).copy())
    best_a = a.copy()
    best_sc = compute_score(a)

    n = len(a)

    for it in range(1, max_iter + 1):
        if time.time() >= deadline:
            break

        # annealing of the soft‑max temperature
        lam = lam_start + (lam_end - lam_start) * min(1.0, it / lam_ramp_iters)

        # current convolution and soft‑max weights
        conv = convolution_fft(a, a)                     # length 2n‑1
        w = softmax_weights(conv, lam)

        # gradient of the weighted sum of convolutions
        grad = gradient_from_weights(a, w)

        # step size (standard 1/√t schedule)
        eta = eta0 / np.sqrt(it)

        # multiplicative (exponentiated) update – guarantees non‑negativity
        a = a * np.exp(-eta * grad)

        # optional exploration noise
        if noise_every and (it % noise_every == 0):
            a += np.random.randn(n) * noise_scale
            a = np.maximum(a, 0.0)

        # re‑project onto the simplex
        a = normalize(a)

        cur_sc = compute_score(a)
        if cur_sc < best_sc:
            best_sc = cur_sc
            best_a = a.copy()

        # occasional progress report
        if it % 2000 == 0:
            print(f"      iter={it:,d}  λ={lam:.1f}  cur={cur_sc:.6f}  best={best_sc:.6f}")

    return best_a, best_sc


# ----------------------------------------------------------------------
# Main entry point required by the competition harness
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs) -> List[float]:
    """
    Search for a non‑negative step‑function with a low ``evaluate_sequence`` value.
    The routine respects the given time budget, examines several lengths,
    samples a variety of initial shapes and refines each with the optimiser
    described above.  The best sequence found is returned as a plain list.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 2.0   # small safety margin

    best_seq: List[float] = []
    best_val: float = float("inf")

    # ------------------------------------------------------------------
    # 0. Optional fallback from a previously discovered construction.
    # ------------------------------------------------------------------
    try:
        fallback = [float(x) for x in height_sequence_1]  # type: ignore
        val = evaluate_sequence(fallback)
        print(f"Fallback (height_sequence_1) score = {val:.6f}")
        if val < best_val:
            best_val = val
            best_seq = fallback
    except NameError:
        pass

    # ------------------------------------------------------------------
    # 1.  Lengths to explore (powers of two – cheap FFTs).
    # ------------------------------------------------------------------
    lengths = [1024, 2048, 4096]

    for n in lengths:
        if time.time() >= deadline - 5.0:
            break
        print(f"\n=== Exploring length n = {n} ===")

        # --------------------------------------------------------------
        # 1a.  Build a modest library of candidate start vectors.
        # --------------------------------------------------------------
        candidates: List[np.ndarray] = []

        # uniform
        candidates.append(np.full(n, 1.0 / n))

        # classic arcsine
        candidates.append(arcsine_initial(n))

        # power‑law family (p, eps)
        for _ in range(5):
            p = rng.uniform(0.2, 0.8)
            eps = rng.uniform(0.001, max(2.0, 0.02 * n))
            candidates.append(generate_shape(n, p, eps))

        # beta‑like shapes
        for _ in range(3):
            alpha = rng.uniform(0.2, 5.0)
            beta = rng.uniform(0.2, 5.0)
            i = np.arange(n, dtype=float)
            base = (i + 0.5) ** (alpha - 1) * (n - 1 - i + 0.5) ** (beta - 1)
            total = base.sum()
            if total == 0.0:
                base = np.full(n, 1.0 / n)
            else:
                base = base / total
            candidates.append(base)

        # Dirichlet / exponential samples
        for _ in range(2):
            samp = rng.exponential(1.0, size=n)
            samp /= samp.sum()
            candidates.append(samp)

        # --------------------------------------------------------------
        # 1b.  Quick sweep with the fast internal score, keep the best few.
        # --------------------------------------------------------------
        scored = [(compute_score(c), c) for c in candidates]
        scored.sort(key=lambda x: x[0])
        top_candidates = [c for _, c in scored[:3]]   # keep the three most promising

        # --------------------------------------------------------------
        # 1c.  Refine each retained candidate.
        # --------------------------------------------------------------
        for idx, cand in enumerate(top_candidates):
            if time.time() >= deadline - 5.0:
                break
            print(f"  candidate {idx + 1}/{len(top_candidates)}")

            remaining = max(0.0, deadline - time.time())
            # rough budget: about 15 optimisation steps per second
            max_iter = int(min(30_000, max(1_000, remaining * 15)))

            refined, _ = optimise_sequence(
                initial_a=cand,
                deadline=deadline,
                lam_start=5.0,
                lam_end=80.0,
                lam_ramp_iters=4000,
                eta0=0.6,
                max_iter=max_iter,
                noise_every=1000,
                noise_scale=1e-5,
            )

            # final (slow) evaluation with the official evaluator
            val = evaluate_sequence(refined.tolist())
            print(f"    evaluated score = {val:.6f}")

            if val < best_val:
                best_val = val
                best_seq = refined.tolist()
                print(f"    *** New global best: {best_val:.6f} (n={n}) ***")

    # ------------------------------------------------------------------
    # 2.  Final polishing of the overall best sequence (if time permits).
    # ------------------------------------------------------------------
    if best_seq and time.time() < deadline - 5.0:
        a = np.array(best_seq, dtype=float)
        remaining = max(0.0, deadline - time.time())
        max_iter = int(min(80_000, remaining * 15))

        print("\nFinal polishing of the global best")
        a_refined, _ = optimise_sequence(
            initial_a=a,
            deadline=deadline,
            lam_start=5.0,
            lam_end=120.0,
            lam_ramp_iters=8000,
            eta0=0.5,
            max_iter=max_iter,
            noise_every=800,
            noise_scale=5e-6,
        )
        final_val = evaluate_sequence(a_refined.tolist())
        print(f"Final polished score = {final_val:.6f}")
        if final_val < best_val:
            best_val = final_val
            best_seq = a_refined.tolist()
            print("*** Updated global best after final polish ***")

    # ------------------------------------------------------------------
    # 3.  Guard – always return a feasible sequence.
    # ------------------------------------------------------------------
    if not best_seq:
        best_seq = [1.0] * 256
        best_val = evaluate_sequence(best_seq)

    print(f"\nSearch finished. Best reported score = {best_val:.6f}")
    return best_seq
