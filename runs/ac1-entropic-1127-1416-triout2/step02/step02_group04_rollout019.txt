"""
Propose a step‑function (non‑negative vector) that minimizes

    2 * n * max(conv(a, a)) / (sum(a) ** 2)

where conv(a,a) is the linear convolution of a with itself.
The objective is scale invariant; we therefore keep the vector on the
probability simplex (sum = 1, all entries ≥ 0).

The search has three phases

1.  A quick sweep of simple analytic families (uniform, linear ramp,
    a few Beta‑shapes, a sine shape, the previously found vector,
    random Dirichlet samples).  The best among them seeds the
    optimisation.

2.  Classic projected sub‑gradient descent on the exact max‑convolution
    (the same method that produced the 1.530293 bound).  A diminishing
    step–size and Euclidean projection onto the simplex keep the
    iterates feasible.

3.  A smoother optimisation based on a *soft‑max* of the convolution
    (log‑sum‑exp).  For a temperature τ the surrogate objective is

        Sτ(a) = (1/τ) * log Σk exp( τ * convk(a) ).

    Its gradient can be written as a pair of convolutions:
        ∇Sτ = 2 * ( a * p  +  a * p_rev )[relevant slice]

    where p_k = exp(τ * convk) / Σj exp(τ * convj) and p_rev is p
    reversed.  Adam is used to take adaptive steps; τ is slowly
    increased so that the surrogate tightens around the true max.
    After each step the iterate is projected back onto the simplex.

All work is performed with FFT‑based convolutions (O(n log n)) and the
total wall‑clock time is limited to 1000 s.
"""

import time
import numpy as np

def simplex_projection(v):
    """Euclidean projection of a vector onto the probability simplex."""
    v = np.asarray(v, dtype=float)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1] - 1
    theta = cssv[cond][-1] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def compute_objective(a):
    """Exact evaluation for a vector with sum = 1:
    returns 2·n·max(conv(a,a)).
    """
    n = a.shape[0]
    # linear convolution via FFT (length 2n‑1)
    fft_len = 1 << ((2 * n - 1 - 1).bit_length())
    fa = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(fa * fa, n=fft_len)[: 2 * n - 1]
    return 2.0 * n * conv.max()


def beta_sequence(alpha, beta, n):
    """Discretised Beta(α,β) on n points, normalised to sum 1."""
    x = (np.arange(n) + 0.5) / n
    logw = (alpha - 1.0) * np.log(x) + (beta - 1.0) * np.log(1.0 - x)
    logw = np.clip(logw, -700.0, 700.0)      # avoid overflow in exp
    w = np.exp(logw)
    if w.sum() == 0 or not np.isfinite(w.sum()):
        w = np.ones_like(w)
    return w / w.sum()


def _fft_len_at_least(m):
    """Small utility: smallest power of two ≥ m."""
    n = 1
    while n < m:
        n <<= 1
    return n


def _self_convolution_fft(a, fft_len):
    """Linear convolution a*a using a pre‑chosen FFT length."""
    fa = np.fft.rfft(a, n=fft_len)
    conv_full = np.fft.irfft(fa * fa, n=fft_len)
    return conv_full[: 2 * len(a) - 1]


def _gradient_softmax(a, conv_self, tau, fft_len):
    """
    Gradient of the log‑sum‑exp surrogate.
    Parameters
    ----------
    a          – current vector (sum = 1)
    conv_self  – linear convolution a*a (length 2n‑1)
    tau        – temperature (>0)
    fft_len    – length of FFT to use (≥ 3n‑2)
    Returns
    -------
    grad – numpy array of size n
    """
    n = a.shape[0]

    # ----- compute probability distribution over convolution indices -----
    # shift by the maximum to keep the exponentials well‑scaled
    shift = conv_self.max()
    w = np.exp(tau * (conv_self - shift))
    p = w / w.sum()

    # ----- convolution a with p and with reversed p -----
    fa = np.fft.rfft(a, n=fft_len)
    fp = np.fft.rfft(p, n=fft_len)
    conv_ap = np.fft.irfft(fa * fp, n=fft_len)          # a * p
    p_rev = p[::-1]
    fp_rev = np.fft.rfft(p_rev, n=fft_len)
    conv_ap_rev = np.fft.irfft(fa * fp_rev, n=fft_len)  # a * p_rev

    # extract the needed slices
    #   term2 = (a * p)[i]  -> first n entries
    term2 = conv_ap[:n]
    #   term1 = (a * p_rev)[i + 2n‑2]  -> slice starting at 2n‑2
    start = 2 * n - 2
    term1 = conv_ap_rev[start : start + n]

    grad = 2.0 * (term1 + term2)
    return grad

            best_a, best_score = a.copy(), s
            print(f"[RndB] α={al:.3f},β={be:.3f} => {s:.6f}")

    # a few Dirichlet‑like draws
    for _ in range(12):
        if time.time() > deadline:
            break
        conc = np.random.uniform(0.1, 3.0, size=n)
        a = np.random.gamma(conc, 1.0)
        a /= a.sum()
        s = compute_objective(a)
        if s < best_score:
            best_a, best_score = a.copy(), s
            print(f"[Dir] random      score = {s:.6f}")

    # ------------------------------------------------------------------
    # 3. Projected sub‑gradient descent (the method that gave 1.53)
    # ------------------------------------------------------------------
    a = best_a.copy()
    it = 0
    eta = 0.8

    while time.time() < deadline and it < 8000:
        # self‑convolution (FFT, length 2n‑1)
        fft_len_s = _fft_len_at_least(2 * n - 1)
        fa = np.fft.rfft(a, n=fft_len_s)
        conv_full = np.fft.irfft(fa * fa, n=fft_len_s)
        conv = conv_full[: 2 * n - 1]

        max_b = conv.max()
        cur_score = 2.0 * n * max_b
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[SGD] it {it:5d} new best {best_score:.6f}")

        # active set (entries close to the current max)
        eps = max_b * 1e-5
        active = np.where(conv >= max_b - eps)[0]
        if active.size == 0:
            active = np.array([conv.argmax()], dtype=int)

        # sub‑gradient (average over the active indices)
        g = np.zeros_like(a)
        for k in active:
            lo = max(0, k - n + 1)
            hi = min(n - 1, k)
            idx = np.arange(lo, hi + 1, dtype=int)
            partner = k - idx
            g[idx] += 2.0 * a[partner]
        g /= float(active.size)

        step = eta / np.sqrt(it + 1)
        a = simplex_projection(a - step * g)
        it += 1

    # ------------------------------------------------------------------
    # 4. Smoothed (log‑sum‑exp) optimisation with Adam
    # ------------------------------------------------------------------
    # pre‑compute a suitable FFT length for the longer convolutions
    fft_len_full = _fft_len_at_least(3 * n - 2)

    # Adam hyper‑parameters
    beta1, beta2 = 0.9, 0.999
    epsilon = 1e-8
    lr = 0.5                     # base learning rate
    m = np.zeros(n)
    v = np.zeros(n)

    # temperature schedule (start soft, end hard)
    tau = 1.0
    max_tau = 300.0

    it2 = 0
    while time.time() < deadline and it2 < 20000:
        # self‑convolution (reuse the same FFT length as in the previous
        # stage – the result is still exact for the first 2n‑1 entries)
        fa = np.fft.rfft(a, n=fft_len_full)
        conv_full = np.fft.irfft(fa * fa, n=fft_len_full)
        conv = conv_full[: 2 * n - 1]

        max_b = conv.max()
        cur_score = 2.0 * n * max_b
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[Adam] it {it2:5d} new best {best_score:.6f}")

        # gradient of the smoothed objective
        grad = _gradient_softmax(a, conv, tau, fft_len_full)

        # Adam update
        it2 += 1
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)
        m_hat = m / (1.0 - beta1 ** it2)
        v_hat = v / (1.0 - beta2 ** it2)
        step = lr * m_hat / (np.sqrt(v_hat) + epsilon)

        a = simplex_projection(a - step)

        # slowly tighten the surrogate (increase τ)
        tau = min(max_tau, tau * 1.02)

    # ------------------------------------------------------------------
    # 5. Return the best sequence discovered
    # ------------------------------------------------------------------
    print(f"Finished – best score = {best_score:.6f} after {it + it2} iterations")
    return [float(v) for v in best_a.tolist()]
