The program searches for a non‑negative sequence `a = (a₀,…,aₙ₋₁)` that minimises  

\[
E(a)=\frac{2\,n\;\max_k (a*a)_k}{\bigl(\sum_i a_i\bigr)^2},
\qquad (a*a)_k=\sum_i a_i a_{k-i}.
\]

`E(a)` is scale invariant, therefore the search is performed on the probability
simplex `∑a_i = 1, a_i ≥ 0`.  
The algorithm consists of three stages

1. **Low‑dimensional parameter sweep** – families of analytically‑motivated
   shapes (`uniform`, `arcsine`, power‑law shapes, mixtures) are generated,
   evaluated and the best few are kept.

2. **Exponentiated‑gradient optimisation** – starting from each retained
   candidate a projected *mirror* gradient descent (exponentiated‑gradient)
   minimises a smooth approximation of the `max` function
   \[
   f_α(a)=\frac{1}{α}\log\!\Bigl(\sum_k \exp\bigl(α\,(a*a)_k\bigr)\Bigr)
   \]
   (α is increased from a small to a large value).  The update

   \[
   a_i \leftarrow a_i\,\exp\!\bigl(-\eta_t\,\nabla_i f_α(a)\bigr),\qquad
   a\leftarrow a/\sum a
   \]

   stays automatically on the simplex and is empirically faster than
   Euclidean projection.

3. **Stochastic hill‑climbing refinement** – a cheap pair‑wise mass‑exchange
   local search finishes the optimisation.

The whole search respects the 1000 s budget, checks the time after every
iteration and always returns the best sequence found.  If a historic solution
`height_sequence_1` exists it is used as a fall‑back.

```python
"""
Hybrid search for a low‑max‑convolution step function.

* Small parametric families (uniform, arcsine, power‑law, mixtures) give
  good starting points.
* Exponentiated‑gradient descent minimises a smoothed max‑convolution
  objective on the probability simplex.
* Final stochastic hill‑climbing cleans up remaining small defects.

The routine returns a list of non‑negative floats that can be fed to
`evaluate_sequence`.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Basic numeric utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full 1‑D convolution via real FFT (fast, length = len(a)+len(b)-1)."""
    m = len(a) + len(b) - 1
    n_fft = next_pow2(m)
    A = np.fft.rfft(a, n=n_fft)
    B = np.fft.rfft(b, n=n_fft)
    C = A * B
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


def compute_score(a: np.ndarray) -> float:
    """
    Exact score of `evaluate_sequence` when the vector is already normalised
    (∑a = 1).  Returns 2·n·max(a*a) .
    """
    n = len(a)
    conv = convolution_fft(a, a)
    max_val = float(np.max(conv))
    return 2.0 * n * max_val


# ----------------------------------------------------------------------
# Analytic families for the initialisation phase
# ----------------------------------------------------------------------


def arcsine_initial(n: int) -> np.ndarray:
    """Arcsine‑shaped start: a_i ∝ 1/√((i+½)(n‑½‑i))."""
    i = np.arange(n, dtype=float)
    eps = 0.25
    vals = 1.0 / np.sqrt((i + 0.5) * (n - 0.5 - i) + eps)
    return vals / np.sum(vals)


def beta_shape(n: int, beta: float) -> np.ndarray:
    """
    Power‑law shape a_i ∝ ((i+½)(n‑½‑i))**beta.
    beta=0 → uniform, beta=-0.5 → arcsine.
    """
    i = np.arange(n, dtype=float)
    base = (i + 0.5) * (n - 0.5 - i)
    # avoid exact zero – base is ≥ 0.5·(n‑0.5) > 0, so safe
    vals = np.exp(beta * np.log(base))
    return vals / np.sum(vals)


# ----------------------------------------------------------------------
# Smoothed‑max gradient (soft‑max) – same as the reference implementation
# ----------------------------------------------------------------------


def smoothed_gradient(a: np.ndarray, alpha: float):
    """
    Gradient of the smooth approximation
        f_α(a) = (1/α)·log Σ_k exp(α·(a∗a)_k)
    Returns (gradient, full self‑convolution).
    """
    n = len(a)
    conv = convolution_fft(a, a)                     # length 2n‑1
    maxc = np.max(conv)
    # soft‑max weights (numerically stable)
    exp_term = np.exp(alpha * (conv - maxc))
    w = exp_term / np.sum(exp_term)                  # Σ w = 1
    # convolution of reversed a with w; needed for gradient
    a_rev = a[::-1]
    conv_rev = convolution_fft(a_rev, w)             # length 3n‑2
    start = n - 1
    rev_slice = conv_rev[start:start + n]            # entries for each i
    # diagonal contribution (k = 2i)
    w_diag = w[::2][:n]
    grad = 2.0 * rev_slice - w_diag * a
    return grad, conv


# ----------------------------------------------------------------------
# Exponentiated‑gradient optimiser (mirror descent on the simplex)
# ----------------------------------------------------------------------


def improve_expgd(
    initial_a: np.ndarray,
    time_deadline: float,
    max_iter: int = 200_000,
    eta0: float = 0.5,
    alpha_start: float = 1.0,
    alpha_end: float = 800.0,
    noise_every: int = 500,
    noise_scale: float = 1e-5,
):
    """
    Projected mirror descent on the smoothed objective.
    Returns the best vector encountered (as a NumPy array) and its
    unsmoothed score.
    """
    a = np.maximum(initial_a.astype(float), 0.0)
    a_sum = a.sum()
    if a_sum == 0:
        a = np.full_like(a, 1.0 / len(a))
    else:
        a = a / a_sum

    best_a = a.copy()
    best_score = compute_score(a)
    n = len(a)

    for t in range(1, max_iter + 1):
        if time.time() >= time_deadline:
            break

        # linear increase of the smoothing parameter α
        alpha = alpha_start + (alpha_end - alpha_start) * min(1.0, t / max_iter)

        grad, _ = smoothed_gradient(a, alpha)

        # step length – classic 1/√t schedule
        eta = eta0 / np.sqrt(t)

        # mirror‑update (exponentiated‑gradient)
        a = a * np.exp(-eta * grad)

        # occasional additive jitter to escape shallow plateaus
        if noise_every and (t % noise_every == 0):
            a += np.random.randn(n) * noise_scale
            a = np.maximum(a, 0.0)

        # re‑normalise onto the simplex
        s = a.sum()
        if s == 0.0:
            a = np.full(n, 1.0 / n)
        else:
            a = a / s

        cur_score = compute_score(a)
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()

        if t % 5000 == 0:
            print(f"    exp‑gd iter {t:,d}  score {cur_score:.6f}  best {best_score:.6f}")

    return best_a, best_score


# ----------------------------------------------------------------------
# Cheap stochastic hill‑climbing refinement
# ----------------------------------------------------------------------


def hill_climb(seq, time_deadline: float, max_steps: int = 30000, init_step: float = 0.08):
    """
    Stochastic pair‑wise mass exchange.
    Moves improving the true (max‑convolution) score are kept.
    """
    a = np.array(seq, dtype=float)
    a = np.maximum(a, 0.0)
    total = a.sum()
    if total == 0:
        a = np.full_like(a, 1.0 / len(a))
    else:
        a = a / total

    best_a = a.copy()
    best_score = compute_score(best_a)
    n = len(a)
    step = init_step

    while max_steps > 0 and time.time() < time_deadline:
        i, j = np.random.randint(n, size=2)
        if i == j:
            continue
        delta = step * best_a[i]
        if delta <= 0:
            continue
        cand = best_a.copy()
        cand[i] -= delta
        cand[j] += delta
        if cand[i] < 0:
            continue
        # renormalise (mass is conserved, only numerical drift)
        cand = np.maximum(cand, 0.0)
        cand_sum = cand.sum()
        if cand_sum == 0:
            continue
        cand = cand / cand_sum

        cur_score = compute_score(cand)
        if cur_score < best_score:
            best_score = cur_score
            best_a = cand
            step = min(step * 1.06, 0.5)
        else:
            step *= 0.97
        max_steps -= 1

    return best_a.tolist(), best_score


# ----------------------------------------------------------------------
# Main entry point expected by the competition harness
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a step‑function (non‑negative sequence) that yields a low
    `evaluate_sequence` value.  The method combines:
      – a quick parametric sweep,
      – exponentiated‑gradient descent on a smooth max‑approximation,
      – a final stochastic hill‑climb.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 2.0   # safety margin

    best_seq = None
    best_val = float("inf")

    # ------------------------------------------------------------------
    # Fallback to a historic construction if it exists.
    # ------------------------------------------------------------------
    try:
        fallback = [float(x) for x in height_sequence_1]
        fallback_val = evaluate_sequence(fallback)
        print(f"Fallback score = {fallback_val:.6f}")
        if fallback_val < best_val:
            best_val = fallback_val
            best_seq = fallback
    except NameError:
        pass

    # ------------------------------------------------------------------
    # Lengths to explore (larger n gives a finer discretisation).
    # ------------------------------------------------------------------
    lengths = [1024, 2048, 4096]

    for n in lengths:
        if time.time() >= deadline - 5.0:
            break
        print(f"\n=== Optimising length n = {n} ===")

        # ---------- 1. generate a modest pool of initial candidates ----------
        candidates = []

        # uniform and pure arcsine
        candidates.append(np.full(n, 1.0 / n))
        candidates.append(arcsine_initial(n))

        # power‑law family (beta ∈ [‑0.55,‑0.30])
        for beta in np.linspace(-0.55, -0.30, 7):
            shape = beta_shape(n, beta)
            candidates.append(shape)
            # mixtures with uniform
            for lam in np.linspace(0.1, 0.9, 5):
                mix = lam * shape + (1.0 - lam) * (1.0 / n)
                candidates.append(mix)

        # a few random Dirichlet draws for diversity
        for _ in range(6):
            candidates.append(np.random.dirichlet(np.ones(n)))

        # ---------- 2. quick evaluation, keep the most promising ----------
        scores = []
        for vec in candidates:
            if time.time() >= deadline - 5.0:
                break
            sc = compute_score(vec)
            scores.append((sc, vec))
        scores.sort(key=lambda x: x[0])
        top_vectors = [vec for _, vec in scores[:3]]
        print(f"  best raw score from pool = {scores[0][0]:.6f}")

        # ---------- 3. refinement with exponentiated‑gradient ----------
        for idx, init_vec in enumerate(top_vectors, start=1):
            if time.time() >= deadline - 5.0:
                break
            print(f"  ++ Refine start #{idx}")
            refined_vec, refined_score = improve_expgd(
                initial_a=init_vec,
                time_deadline=deadline,
                max_iter=200_000,
                eta0=0.7,
                alpha_start=2.0,
                alpha_end=1000.0,
                noise_every=600,
                noise_scale=1e-5,
            )
            print(f"      refined (exp‑gd) score = {refined_score:.6f}")
            if refined_score < best_val:
                best_val = refined_score
                best_seq = refined_vec.tolist()
                print(f"      *** New global best {best_val:.6f} (n={n}) ***")

    # ------------------------------------------------------------------
    # Final cheap stochastic hill‑climbing refinement (if time permits).
    # ------------------------------------------------------------------
    if best_seq is not None and time.time() < deadline - 2.0:
        print("\n>>> Final hill‑climb refinement")
        refined_seq, refined_val = hill_climb(
            best_seq,
            time_deadline=deadline,
            max_steps=50000,
            init_step=0.05,
        )
        print(f"      refined hill‑climb score = {refined_val:.6f}")
        if refined_val < best_val:
            best_val = refined_val
            best_seq = refined_seq

    # ------------------------------------------------------------------
    # Safety net: if we never produced a candidate, fall back to a tiny uniform.
    # ------------------------------------------------------------------
    if best_seq is None:
        best_seq = [1.0] * 256
        best_val = evaluate_sequence(best_seq)

    final_score = evaluate_sequence(best_seq)
    print(f"\nFinished search. Final reported score = {final_score:.6f}")

    # The harness expects a plain Python list of floats.
    return best_seq
```
