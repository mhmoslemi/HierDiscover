"""
Search for a step‑function that beats the previous bound (≈1.570).
The best known analytic family are power‑law sequences a_i ∝ (i+1)^(-p).
For p≈0.55 the theoretical value of the evaluation function is
    2·(1‑p)^2·B(1‑p,1‑p)  ≈ 1.50,
which is below the target 1.503.  The routine therefore performs a
lightweight grid‑search over p, refines the best exponent by a simple
hill‑climbing, and finally runs a few sub‑gradient iterations to polish
the result.  All sequences are projected onto the probability simplex
so that the denominator of evaluate_sequence stays equal to 1.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper functions (no closures)
# ----------------------------------------------------------------------


def project_onto_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the probability simplex:
        { x | x >= 0, sum(x) = 1 }.
    Returns a new array (the input is not modified).
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]                     # sorted descending
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all entries are zero – return uniform distribution
        return np.full_like(v, 1.0 / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def full_convolution(a: np.ndarray) -> np.ndarray:
    """
    Compute the full discrete convolution a * a using an FFT.
    Length of the result is 2*len(a) - 1.
    """
    n = a.shape[0]
    size = 2 * n
    fft_len = 1 << (size - 1).bit_length()          # next power of two
    fft_a = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(fft_a * fft_a, n=fft_len)
    return conv[: 2 * n - 1]


def gradient_at_max(a: np.ndarray, k: int) -> np.ndarray:
    """
    Sub‑gradient of the maximal convolution entry (at shift k)
    with respect to the coefficient vector a.
    """
    n = a.shape[0]
    g = np.zeros_like(a)
    low = max(0, k - n + 1)
    high = min(k, n - 1)
    if low > high:
        return g
    idx = np.arange(low, high + 1, dtype=int)
    g[idx] = a[k - idx]
    return g


def score_from_sequence(a: np.ndarray) -> float:
    """
    Compute the evaluation value that evaluate_sequence would return,
    assuming sum(a) == 1 (which is the case after projection).
    """
    n = a.shape[0]
    conv = full_convolution(a)
    return float(2.0 * n * np.max(conv))

# Main search routine
