"""
Propose a low‑score step‑function sequence.

Algorithm
---------

We work with a probability vector  q (non‑negative, Σq = 1) and later
scale it by √(2 n) to obtain the actual heights.
The score is  2 n·max(q∗q)  (because the denominator of the official
evaluator becomes 1 after the ℓ₁‑normalisation).

1. **Seeding** – generate a modest collection of simple families
   (geometric, power‑law, uniform, random Dirichlet).  The best seed
   (according to the fast FFT based score) is taken as the start point.

2. **Smoothed‑max gradient descent** – we replace the non‑smooth
   max by the smooth approximation  
        f_α(q) = (1/α)·log ∑_k exp(α·(q∗q)_k) .  
   Its gradient is
        ∇f_α(q) = 2·conv(w, q)      with   w_k ∝ exp(α·(q∗q)_k) .
   The gradient is computed completely with FFTs.  An Adam‑style
   update is performed and the iterate is projected back onto the
   simplex {x≥0, Σx = 1}.  The smoothing parameter α is increased
   slowly (homotopy) to approach the true max.

3. **Optional LP polishing** – after the gradient phase we optionally
   run a few linear‑programming refinements exactly as in the reference
   implementation (if SciPy is available).

The whole routine respects the 1000 s budget, prints progress, and
finally returns the best heights found (scaled to √(2 n)).  The result
is validated with the official ``evaluate_sequence`` function.
"""

import time
import numpy as np

# --------------------------------------------------------------------------- #
# Simplex projection (Duchi et al. 2008)
def projection_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Project a vector onto the simplex {x ≥ 0, Σx = s}."""
    if s < 0:
        raise ValueError("Simplex sum must be non‑negative")
    n = v.shape[0]
    # sort v in descending order
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho_arr = np.where(u - cssv / (np.arange(n) + 1.0) > 0)[0]
    if rho_arr.size == 0:
        return np.zeros_like(v)
    rho = rho_arr[-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# --------------------------------------------------------------------------- #
def _fast_score(seq: np.ndarray) -> float:
    """Fast evaluation of 2·n·max(conv(seq,seq)) / (sum(seq)**2) using FFT."""
    n = seq.size
    fft_len = 1 << ((2 * n - 1).bit_length())
    F = np.fft.rfft(seq, n=fft_len)
    conv = np.fft.irfft(F * F, n=fft_len)[: 2 * n - 1]
    max_conv = conv.max()
    s = seq.sum()
    return float(2.0 * n * max_conv / (s * s))


# --------------------------------------------------------------------------- #
def _build_toeplitz_matrix(f: np.ndarray):
    """Sparse Toeplitz matrix A (shape (2n‑1, n)) such that A·g = f * g."""
    n = f.shape[0]
    rows = np.empty(n * n, dtype=np.int32)
    cols = np.empty(n * n, dtype=np.int32)
    data = np.empty(n * n, dtype=np.float64)

    ptr = 0
    for i in range(n):
        rows[ptr : ptr + n] = np.arange(i, i + n)
        cols[ptr : ptr + n] = i
        data[ptr : ptr + n] = f
        ptr += n

    from scipy.sparse import csr_matrix
    return csr_matrix((data, (rows, cols)), shape=(2 * n - 1, n))


# --------------------------------------------------------------------------- #
def propose_candidate(seed: int = 42, budget_s: float = 1000) -> list[float]:
    """
    Search for a non‑negative step‑function that minimises
    ``evaluate_sequence`` within the given time budget.
    Returns the best sequence found (list of floats).
    """
    np.random.seed(seed)
    start = time.time()
    deadline = start + budget_s - 0.2   # keep a safety margin

    # ------------------------------------------------------------------- #
    # Problem size
    n = 2500
    target_sum = np.sqrt(2.0 * n)          # √(2 n) – required ℓ₁‑norm

    # FFT length large enough for all convolutions we use (q∗q and w∗q)
    fft_len = 1 << ((3 * n - 2).bit_length())   # ≥ 3n‑2

    # ------------------------------------------------------------------- #
    # 1. Build a collection of seeds (all expressed as probability vectors)
    seeds = []

    # geometric families (a_i ∝ r^i)
    lambdas = np.logspace(-5, -1, 30)
    for lam in lambdas:
        r = 1.0 - lam
        r_pow_n = r ** n
        denom = 1.0 - r_pow_n
        if denom <= 0.0:
            continue
        coeff = lam / denom
        a = coeff * (r ** np.arange(n))
        seeds.append(a)

    # power‑law families (a_i ∝ (i+1)^‑p)
    for p in np.linspace(0.3, 0.95, 25):
        base = 1.0 / np.power(np.arange(1, n + 1, dtype=np.float64), p)
        seeds.append(base)

    # uniform seed
    seeds.append(np.full(n, 1.0 / n))

    # a few random Dirichlet draws – some concentrated, some flat
    for alpha in (0.5, 1.0, 2.0, 5.0):
        rand = np.random.gamma(alpha, 1.0, size=n)
        seeds.append(rand)

    # include a possible previously discovered sequence
    try:
        global height_sequence_1
        if isinstance(height_sequence_1, (list, tuple, np.ndarray)):
            hs = np.asarray(height_sequence_1, dtype=np.float64)
            if hs.shape[0] == n:
                seeds.append(hs.copy())
    except Exception:
        pass

    # ------------------------------------------------------------------- #
    # 2. Choose the best seed (projected onto the simplex Σq=1)
    best_q = None
    best_score = float("inf")

    for idx, raw in enumerate(seeds):
        if time.time() >= deadline:
            break
        # turn into a probability vector
        q = np.maximum(raw, 0.0)
        q = projection_simplex(q, 1.0)
        sc = _fast_score(q)               # exact objective for Σq=1
        if sc < best_score:
            best_score, best_q = sc, q.copy()
            print(f"[{time.time() - start:6.1f}s] seed {idx:02d} → score {sc:.6f}")

    # ------------------------------------------------------------------- #
    # 3. Smoothed‑max gradient descent (Adam + simplex projection)
    q = best_q.copy()
    best_score = _fast_score(q)
    best_q = q.copy()

    # Adam hyper‑parameters
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8
    m = np.zeros_like(q)
    v = np.zeros_like(q)

    # smoothing (α) schedule
    alpha = 1.0               # start soft
    alpha_max = 200.0
    alpha_mul = 1.0005        # multiplicative increase each iteration

    # learning‑rate schedule
    lr = 0.2

    it = 0
    last_improve = 0

    while time.time() < deadline:
        # --- convolution q * q (full length 2n‑1) -----------------------
        Fq = np.fft.rfft(q, n=fft_len)
        conv = np.fft.irfft(Fq * Fq, n=fft_len)[: 2 * n - 1]

        max_conv = conv.max()

        # --- smoothed max weights --------------------------------------
        # w_k ∝ exp(α·(conv_k - max_conv))   (stable shift)
        exp_vals = np.exp(alpha * (conv - max_conv))
        w = exp_vals / exp_vals.sum()                 # shape (2n‑1,)

        # --- gradient = 2·conv(w, q) ----------------------------------
        Fw = np.fft.rfft(w, n=fft_len)
        grad_full = np.fft.irfft(Fw * Fq, n=fft_len)[:n]   # first n entries
        grad = 2.0 * grad_full

        # --- Adam update ------------------------------------------------
        it += 1
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)

        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        step = lr * m_hat / (np.sqrt(v_hat) + eps)   # descent direction
        q_new = q - step
        q_new = np.clip(q_new, 0.0, None)
        q_new = projection_simplex(q_new, 1.0)

        q = q_new

        # --- increase smoothing parameter --------------------------------
        if alpha < alpha_max:
            alpha = min(alpha_max, alpha * alpha_mul)

        # --- occasional random jitter to avoid stagnation -------------
        if np.random.rand() < 0.012:
            jitter = np.random.randn(n) * 1e-3
            q = projection_simplex(np.clip(q + jitter, 0.0, None), 1.0)

        # --- evaluate every few hundred iterations ----------------------
        if it % 250 == 0:
            sc = _fast_score(q)
            if sc < best_score - 1e-7:
                best_score, best_q = sc, q.copy()
                last_improve = it
                print(f"[{time.time() - start:6.1f}s] iter {it:7d} → new best {sc:.6f}")
            # simple step‑size decay if no improvement for a while
            if it - last_improve > 5000:
                lr *= 0.7
                last_improve = it   # reset counter

    # ------------------------------------------------------------------- #
    # 4. Optional LP polishing (as in the reference code)
    try:
        from scipy.optimize import linprog
    except Exception:
        linprog = None

    if linprog is not None and time.time() < deadline - 5.0:
        f = best_q.copy()
        for lp_iter in range(8):
            if time.time() >= deadline - 5.0:
                break
            # current maximal autocorrelation
            Ff = np.fft.rfft(f, n=fft_len)
            conv_f = np.fft.irfft(Ff * Ff, n=fft_len)[: 2 * n - 1]
            M = conv_f.max()

            # Build Toeplitz matrix for constraints (f * g) ≤ M
            A = _build_toeplitz_matrix(f)          # (2n‑1, n) sparse
            b_vec = np.full(2 * n - 1, M)

            # maximise Σ g_i  ⇔ minimise - Σ g_i
            res = linprog(
                c=-np.ones(n),
                A_ub=A,
                b_ub=b_vec,
                bounds=[(0, None)] * n,
                method="highs",
                options={"presolve": True, "time_limit": 15},
            )
            if not res.success or res.x is None:
                break

            g = res.x
            sum_g = g.sum()
            if sum_g < 1e-12:
                break

            # Rescale g to have ℓ₁‑norm = 1 and mix
            g = g * (1.0 / sum_g)
            t_mix = 0.12
            f_new = (1.0 - t_mix) * f + t_mix * g
            f_new = projection_simplex(f_new, 1.0)

            # compute new objective
            Fnew = np.fft.rfft(f_new, n=fft_len)
            conv_new = np.fft.irfft(Fnew * Fnew, n=fft_len)[: 2 * n - 1]
            score_new = float(2.0 * n * conv_new.max() / (f_new.sum() ** 2))

            if score_new < best_score - 1e-8:
                best_score, best_q = score_new, f_new.copy()
                print(f"[{time.time() - start:6.1f}s] LP‑iter {lp_iter} → best {score_new:.6f}")

            # accept if it does not worsen the bound
            if conv_new.max() <= M * 0.999:
                f = f_new
            else:
                break

    # ------------------------------------------------------------------- #
    # 5. Scale back to the required ℓ₁‑norm and final verification
    best_a = best_q * target_sum
    best_a = np.clip(best_a, 0.0, 1000.0)   # safety clamp

    final_score = evaluate_sequence(best_a.tolist())
    print(f"Search finished. Final score reported by evaluate_sequence: {final_score:.6f}")

    return best_a.tolist()
