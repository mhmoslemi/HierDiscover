assistantfinalBelow is a self‑contained implementation of `propose_candidate`.  
It uses a smooth approximation of the “max‑convolution’’ objective (soft‑max / log‑sum‑exp), an Adam‑style mirror descent on the probability simplex, and an annealing schedule for the smoothing parameter β.  
The algorithm starts from the best known sequence (if available) or from a sparse random Dirichlet draw, repeatedly improves the vector, and keeps the best solution found before the time budget expires.

```python
"""
Propose a step‑function sequence by minimising

    2·n·max( a * a ) / ( Σ a )²

The optimisation proceeds in two stages:

1.  A smoothed version of the max‑convolution is minimised using
    Adam‑style updates on the probability simplex (mirrored via Euclidean
    projection).  The smoothing parameter β starts small (very smooth) and
    is multiplied by a factor >1 every few thousand iterations, gradually
    turning the soft‑max into the true max.

2.  After the β‑schedule saturates the algorithm continues with a plain
    sub‑gradient step (projected onto the simplex) to polish the solution.

Both stages respect the 1000 s deadline, keep the best vector seen, and
return it as a plain Python list of non‑negative floats.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities
# ----------------------------------------------------------------------


def simplex_projection(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the probability simplex:
        { w | w_i ≥ 0 , Σ w_i = 1 }.
    Implementation follows Duchi et al. (2008).
    """
    n = v.shape[0]
    # sort v in decreasing order
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all entries non‑positive → uniform distribution
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1] - 1               # zero‑based index of last positive
    theta = cssv[cond][-1] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def convolve_self(a: np.ndarray) -> np.ndarray:
    """
    Full linear convolution a * a (real‑valued) using FFT.
    Returns an array of length 2·len(a)‑1.
    """
    n = a.shape[0]
    L = 2 * n - 1
    # use a power‑of‑two length for speed
    N = 1 << ((L - 1).bit_length())
    fa = np.fft.rfft(a, N)
    conv_full = np.fft.irfft(fa * fa, N)[:L]
    # rounding may produce tiny negative numbers; clip to zero
    np.maximum(conv_full, 0.0, out=conv_full)
    return conv_full


def correlation_fft(p: np.ndarray, a: np.ndarray) -> np.ndarray:
    """
    Compute the (non‑circular) correlation of p and a:
        (p ⋆ a)[i] = Σ_k p_{k+i} a_k   for i = 0,…,len(a)-1 .
    Implemented via FFT for O(n log n) speed.
    Returns a vector of length len(a).
    """
    n = a.shape[0]
    L = len(p) + n - 1                     # length of full correlation
    N = 1 << ((L - 1).bit_length())        # next power of two
    fp = np.fft.rfft(p, N)
    fa = np.fft.rfft(a, N)
    corr_full = np.fft.irfft(fp * np.conj(fa), N)
    start = n - 1
    return corr_full[start:start + n]


def smoothed_max(conv: np.ndarray, beta: float) -> (float, np.ndarray):
    """
    Soft‑max (log‑sum‑exp) approximation of max(conv) with temperature 1/beta.
    Returns the approximated value and the probability vector
    p_k = exp(beta * (conv_k - max_conv)) / Σ exp(...).
    """
    max_val = conv.max()
    # shift for numerical stability
    shifted = beta * (conv - max_val)
    # avoid overflow/underflow
    exp_shifted = np.exp(shifted - np.max(shifted))
    z = exp_shifted.sum()
    p = exp_shifted / z
    # soft‑max value = max_val + log(z)/beta
    smoothed = max_val + np.log(z) / beta
    return smoothed, p


# ----------------------------------------------------------------------
# Main optimisation routine
# ----------------------------------------------------------------------


def propose_candidate(budget_s: int = 1000) -> list:
    """
    Search for a low‑score step‑function sequence.
    Runs for up to ``budget_s`` seconds (default 1000) and returns the
    best sequence found as a plain Python list of floats.
    """
    np.random.seed(42)                     # reproducibility
    start_time = time.time()
    deadline = start_time + budget_s - 5.0   # safety margin for Python overhead

    # --------------------------------------------------------------
    # 1. Problem size and initial point
    # --------------------------------------------------------------
    try:
        # Use the best known construction if it exists in the environment
        init = np.asarray(height_sequence_1, dtype=float)   # type: ignore[name-defined]
        n = init.shape[0]
    except Exception:
        # Fallback: a moderate size that offers a good trade‑off between
        # expressiveness and speed.
        n = 5146
        init = None

    if init is None:
        # Sparse random start: Dirichlet with small concentration → many near‑zero entries
        alpha = 0.07
        a = np.random.gamma(alpha, 1.0, size=n)
        a /= a.sum()
    else:
        a = init.astype(float)
        total = a.sum()
        if total <= 0.0:
            a = np.full(n, 1.0 / n)
        else:
            a = a / total                     # project to simplex

    # --------------------------------------------------------------
    # 2. Optimisation hyper‑parameters
    # --------------------------------------------------------------
    # Adam‑style parameters
    beta1 = 0.9
    beta2 = 0.999
    eps_adam = 1e-8
    m = np.zeros_like(a)
    v = np.zeros_like(a)

    # Smoothing schedule for the soft‑max
    beta = 2.0                # start very smooth
    beta_max = 2000.0
    beta_inc_every = 4000    # increase β every this many iterations
    beta_mul = 1.02

    # Learning rate (pre‑projection) – tuned for the simplex geometry
    eta0 = 0.12

    # Book‑keeping
    best_a = a.copy()
    conv = convolve_self(a)
    best_score = 2.0 * n * conv.max()          # exact objective for sum(a)=1
    iteration = 0

    # --------------------------------------------------------------
    # 3. Main optimisation loop
    # --------------------------------------------------------------
    while time.time() < deadline:
        # ---- 3.1 Compute convolution and its smooth max ----
        conv = convolve_self(a)                # length 2n-1
        smoothed, p = smoothed_max(conv, beta)

        # ---- 3.2 Gradient of the smoothed objective ----
        # ∂/∂a_i  [ Σ_k p_k * (a * a)_k ]  =  2 * (p ⋆ a)_i   (approx.)
        grad = 2.0 * correlation_fft(p, a)

        # ---- 3.3 Adam update on the (unprojected) vector ----
        iteration += 1
        step_size = eta0 / np.sqrt(iteration)   # simple diminishing global scale
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)
        m_hat = m / (1.0 - beta1 ** iteration)
        v_hat = v / (1.0 - beta2 ** iteration)
        direction = step_size * m_hat / (np.sqrt(v_hat) + eps_adam)

        a_candidate = a - direction
        a_candidate = simplex_projection(a_candidate)

        # ---- 3.4 Occasionally inject a tiny random jitter to break symmetry ----
        if iteration % 2500 == 0:
            jitter = 0.005 * np.random.randn(n)
            a_candidate = simplex_projection(a_candidate + jitter)

        a = a_candidate

        # ---- 3.5 Update best solution seen so far ----
        cur_score = 2.0 * n * conv.max()
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"Iter {iteration:6d}: improved score = {best_score:.6f}")

        # ---- 3.6 Anneal β (make the soft‑max sharper) ----
        if iteration % beta_inc_every == 0 and beta < beta_max:
            beta = min(beta_max, beta * beta_mul)

        # ---- 3.7 Early stop if target met ----
        if best_score < 1.503:
            print(f"Target reached (score {best_score:.6f}) at iteration {iteration}")
            break

    # --------------------------------------------------------------
    # 4. Optional final polishing with a pure sub‑gradient step
    # --------------------------------------------------------------
    # Do a short walk with the exact max‑convolution sub‑gradient to
    # remove any residual smoothing bias (if time permits).
    if time.time() < deadline - 2.0:
        a = best_a.copy()
        velocity = np.zeros_like(a)
        beta_momentum = 0.9
        step0 = 0.5
        iters_polish = 0
        while time.time() < deadline - 2.0:
            # exact convolution
            conv = convolve_self(a)
            max_val = conv.max()
            k_max = int(conv.argmax())

            # active indices (within numerical tolerance)
            tol = max_val * 1e-12 + 1e-14
            active = np.where(conv >= max_val - tol)[0]

            grad = np.zeros_like(a)
            for k in active:
                low = max(0, k - n + 1)
                high = min(n - 1, k)
                idx = np.arange(low, high + 1, dtype=int)
                partner = k - idx
                grad[idx] += 2.0 * a[partner]
            if active.size > 1:
                grad *= 1.0 / active.size

            step = step0 / np.sqrt(iters_polish + 1)
            velocity = beta_momentum * velocity - step * grad
            a = simplex_projection(a + velocity)

            # keep best
            cur_score = 2.0 * n * conv.max()
            if cur_score < best_score:
                best_score = cur_score
                best_a = a.copy()
                print(f"Polish {iters_polish:6d}: new best = {best_score:.6f}")

            if best_score < 1.503:
                print(f"Target reached during polishing at iter {iters_polish}")
                break
            iters_polish += 1

    # --------------------------------------------------------------
    # 5. Return result
    # --------------------------------------------------------------
    # Convert to plain Python list; evaluate_sequence works with any
    # non‑negative numbers, so we keep the values exactly as produced.
    return [float(v) for v in best_a.tolist()]
```
