"""
Propose a step‑function (non‑negative vector) that minimises

        2 * n * max(a * a) / (sum(a) ** 2)

where “*” is the linear convolution and n = len(a).

Algorithm
---------
* Initialise from the best known construction (``height_sequence_1`` if
  available, otherwise from the central‑binomial coefficients) and normalise
  to unit ℓ₁‑norm.
* Enforce symmetry a[i] = a[n‑1‑i] – the optimum is known to be symmetric.
* In each iteration:
    1. Compute the convolution a⋆a by means of an FFT (O(n log n)).
    2. Locate the (few) indices where the convolution attains its maximum.
    3. The gradient of the (piecewise linear) max‑functional is
       g_i = Σ_{k∈max} a_{k‑i}.
       This is obtained as a second convolution a⋆m where m is the
       indicator of the maximal positions (again via FFT).
    4. Form a zero‑sum descent direction   d = mean(g) – g  (moving mass
       from entries that contribute most to the peak towards those that
       contribute the least).
    5. Project a + step·d onto the probability simplex (clip, renormalise)
       and re‑symmetrise.
    6. Decay the step size gradually; occasionally perform a random
       pair‑wise mass transfer to escape flat regions.
    7. Keep the best vector seen so far.
* The loop runs until the allocated time budget expires (default 1000 s).

The FFT acceleration makes it possible to perform many more iterations
than the naïve O(n²) approach used previously, which typically yields a
lower bound on the evaluation function.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
def binomial_sequence(length: int) -> np.ndarray:
    """
    Central binomial coefficients scaled by 4**k:
        c₀ = 1,
        c_{k+1} = c_k * ((2k+1)(2k+2)) / (4 (k+1)²)
    The seq satisfies Σ c_k ≈ √(π k) and is a classical start point.
    """
    seq = np.empty(length, dtype=np.float64)
    seq[0] = 1.0
    for k in range(1, length):
        seq[k] = seq[k - 1] * ((2 * k - 1) * (2 * k)) / (4.0 * k * k)
    return seq


# ----------------------------------------------------------------------
def next_pow2(x: int) -> int:
    """Small utility: smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


# ----------------------------------------------------------------------
def project_onto_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a non‑negative vector onto the probability simplex
    (components ≥0, sum = 1).  The implementation follows the classic O(n)
    algorithm based on sorting.
    """
    n = v.shape[0]
    # make sure we are working with a copy
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all entries are zero – return uniform distribution
        return np.full_like(v, 1.0 / n)
    rho = ind[cond].max()
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑height vector that minimises the
    provided evaluation function.  Returns the best vector found within the
    time budget.

    Parameters
    ----------
    seed : int
        Random seed for reproducibility.
    budget_s : int
        Time budget in seconds (default 1000).

    Returns
    -------
    list[float]
        Best sequence discovered (ℓ₁‑norm = 1, non‑negative, symmetric).
    """
    np.random.seed(seed)
    end_time = time.time() + budget_s - 0.1   # safety margin

    # ------------------------------------------------------------------
    # 1) Choose problem size and initial vector
    # ------------------------------------------------------------------
    try:
        # ``height_sequence_1`` is supplied by the framework if a previous
        # best construction exists.
        init = np.array(height_sequence_1, dtype=np.float64)   # type: ignore[name-defined]
        n = init.size
    except Exception:
        # fallback – a slightly larger size than the previous run (≈5200)
        n = 6000
        init = binomial_sequence(n)

    # Normalise to unit ℓ₁‑norm (the objective is scale‑invariant)
    a = init.astype(np.float64)
    a = a / a.sum()
    # Enforce exact symmetry – this never harms the value of the objective.
    a = (a + a[::-1]) * 0.5

    # ------------------------------------------------------------------
    # 2) FFT preparation
    # ------------------------------------------------------------------
    L = next_pow2(2 * n - 1)   # size for circular convolution (power of two)
    # Pre‑allocate arrays that stay on the device to avoid repeated allocations
    # (NumPy reuses the buffers when possible, but explicit creation is fine).

    # ------------------------------------------------------------------
    # 3) Optimisation parameters
    # ------------------------------------------------------------------
    step = 0.45          # initial step length (in the zero‑sum direction)
    min_step = 1e-6
    max_step = 0.5
    decay = 0.9998       # multiplicative decay per iteration
    random_move_prob = 0.01
    random_move_frac = 0.06   # max fraction of mass moved in a random transfer

    best_score = np.inf
    best_seq = a.copy()
    last_improve = time.time()
    it = 0

    # ------------------------------------------------------------------
    # 4) Main optimisation loop
    # ------------------------------------------------------------------
    while time.time() < end_time:
        # ---------- convolution a⋆a ----------
        A = np.fft.rfft(a, n=L)
        conv_full = np.fft.irfft(A * A, n=L)[: 2 * n - 1]
        max_val = conv_full.max()
        cur_score = 2.0 * n * max_val          # because sum(a) == 1

        # ---------- keep the best ----------
        if cur_score < best_score - 1e-12:
            best_score = cur_score
            best_seq = a.copy()
            last_improve = time.time()
            elapsed = time.time() - (end_time - budget_s)
            print(f"[improve] iter={it}, score={best_score:.9f}, time={elapsed:.2f}s")

        # ---------- gradient of the max entry ----------
        # mask of (almost) maximal positions
        eps = max_val * 1e-12 + 1e-14
        mask = (conv_full >= max_val - eps).astype(np.float64)

        # gradient g = a ⋆ mask   (only first n entries are needed)
        M = np.fft.rfft(mask, n=L)
        g_full = np.fft.irfft(A * M, n=L)[:n]      # length‑n slice

        # zero‑sum descent direction   d = mean(g) – g
        mean_g = g_full.mean()
        d = mean_g - g_full

        # ---------- projected gradient update ----------
        a_candidate = a + step * d
        # clip negatives
        a_candidate = np.maximum(a_candidate, 0.0)
        # renormalise to the simplex
        a_candidate = project_onto_simplex(a_candidate)
        # enforce symmetry again
        a_candidate = (a_candidate + a_candidate[::-1]) * 0.5

        # ---------- optional random perturbation ----------
        if np.random.rand() < random_move_prob:
            i, j = np.random.randint(0, n, size=2)
            if i != j and a_candidate[i] > 0.0:
                delta = random_move_frac * np.random.rand() * a_candidate[i]
                a_candidate[i] -= delta
                a_candidate[j] += delta
                a_candidate = project_onto_simplex(a_candidate)
                a_candidate = (a_candidate + a_candidate[::-1]) * 0.5

        # replace the current iterate
        a = a_candidate

        # ---------- adapt step size ----------
        step = max(min_step, step * decay)

        # ---------- restart if stuck for a while ----------
        if time.time() - last_improve > 30.0:
            # re‑inject a small amount of noise around the best known solution
            a = best_seq.copy()
            jitter = 0.02 * np.random.randn(n) * a
            a = np.maximum(a + jitter, 0.0)
            a = project_onto_simplex(a)
            a = (a + a[::-1]) * 0.5
            step = max(step, 0.04)   # give the optimiser a fresh push
            last_improve = time.time()
            elapsed = time.time() - (end_time - budget_s)
            print(f"[restart] iter={it}, time={elapsed:.2f}s")

        it += 1

    # ------------------------------------------------------------------
    # 5) Final reporting
    # ------------------------------------------------------------------
    print(f"[final] best score={best_score:.9f}, iterations={it}")

    # The evaluation function caps values at 1000 and forces non‑negativity,
    # so returning the unit‑norm vector is safe.
    return best_seq.tolist()
