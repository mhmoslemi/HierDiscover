def gradient_softmax(a, conv, beta):
    L = conv.shape[0]
    maxc = conv.max()
    w = np.exp(beta * (conv - maxc))
    w_sum = w.sum()
    if w_sum == 0:
        w = np.full_like(w, 1.0/L)
    else:
        w /= w_sum
    revw = w[::-1]
    # compute convolution a (n) with revw (2n-1) => length n+2n-2 = 3n-2
    size = 1 << ((3*n-2) - 1).bit_length()
    A = np.fft.rfft(a, n=size)
    R = np.fft.rfft(revw, n=size)
    conv_full = np.fft.irfft(A * R, n=size)
    # Get gradient g_i = conv_full[L-1 - i]
    Lm1 = L - 1 # = 2n-2
    grad = conv_full[Lm1 - np.arange(n)]
    return grad

"""
Propose a candidate step‑function whose self‑convolution
has a small supremum norm.

The algorithm works on a fixed length `n` (default the length of the
previous best construction, otherwise 5000).  It starts from a
reasonable analytic shape (≈1/√i) Normalises it to sum = 1 and then
performs a projected (mirror) gradient descent that minimises a
smooth approximation of the quantity

    max_k (a * a)_k               (a lives on the probability simplex)

The smooth approximation is the soft‑max of the convolution entries.
Initially the temperature is high (β≈1) so many shifts influence the
gradient; β is slowly increased, eventually turning the update into the
usual sub‑gradient step on the maximal entry.  A Nesterov‑type momentum
term is used and after every few thousand iterations a tiny random
redistribution of mass is tried to escape shallow plateaux.

The whole loop respects a time‑budget (`budget_s` seconds) and the best
sequence found during the run is returned (as a plain Python list).
If something goes wrong the previous global construction `height_sequence_1`
is used as a safe fallback.
"""

import time
import numpy as np
from typing import List

# ----------------------------------------------------------------------
# Helper functions – all top‑level, no closures
# ----------------------------------------------------------------------


def project_onto_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the probability simplex:
        { x | x >= 0, sum(x) = 1 }.
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def full_convolution(a: np.ndarray) -> np.ndarray:
    """
    Linear (full) convolution a*a using an FFT.
    Length of the result is 2*len(a)-1.
    """
    n = a.shape[0]
    size = 1 << ((2 * n - 1) - 1).bit_length()   # next power of two
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[: 2 * n - 1]


def gradient_at_max(a: np.ndarray, k: int) -> np.ndarray:
    """
    Sub‑gradient of the maximal convolution entry (at shift k)
    w.r.t. the vector a.
    """
    n = a.shape[0]
    g = np.zeros_like(a)
    low = max(0, k - n + 1)
    high = min(k, n - 1)
    if low <= high:
        idx = np.arange(low, high + 1, dtype=int)
        g[idx] = a[k - idx]
    return g


def softmax_weights(conv: np.ndarray, beta: float) -> np.ndarray:
    """
    Soft‑max on the convolution values.
    Returns a probability vector of the same length as conv.
    """
    maxc = conv.max()
    # shift for numerical stability
    ex = np.exp(beta * (conv - maxc))
    s = ex.sum()
    if s == 0.0:
        return np.full_like(conv, 1.0 / conv.size)
    return ex / s


def gradient_from_weights(a: np.ndarray, w: np.ndarray, fft_size: int) -> np.ndarray:
    """
    Gradient of the soft‑max objective:
        g_i = Σ_k w_k a_{k-i}
    which is a (linear) cross‑correlation of `a` with `w`.
    Computed via FFT, using a pre‑chosen `fft_size` that is at least
    n + len(w) - 1.
    """
    n = a.shape[0]
    L = w.size                 # = 2*n-1
    rev_w = w[::-1]            # needed for the cross‑correlation
    A = np.fft.rfft(a, n=fft_size)
    B = np.fft.rfft(rev_w, n=fft_size)
    full = np.fft.irfft(A * B, n=fft_size)
    # the needed slice starts at offset = L-1 and runs backwards for n entries
    offset = L - 1
    idx = offset - np.arange(n)
    return full[idx]

# Main search routine
# ----------------------------------------------------------------------

        2 * n * max(conv(a, a)) / (sum(a) ** 2)

    using a projected (mirror) gradient descent with a soft‑max
    approximation of the max‑operator.  The function runs for at most
    `budget_s` seconds and returns the best list of floats found.
    """
    np.random.seed(seed)

    # keep a few seconds for final printing/return
    deadline = time.time() + budget_s - 5.0

    # ------------------------------------------------------------------
    # Initialise a – use the previously found construction if available
    # ------------------------------------------------------------------
    try:
        init = np.array(height_sequence_1, dtype=float)  # type: ignore[name-defined]
        n = init.size
        a = init.copy()
        # ensure non‑negativity (the previous run already did this)
        a = np.maximum(a, 0.0)
    except Exception:
        n = 5000
        a = 1.0 / np.sqrt(np.arange(n, dtype=float) + 1.0)

    # Normalise to the simplex (sum = 1)
    a = np.maximum(a, 0.0)
    a_sum = a.sum()
    if a_sum == 0.0:
        a = np.full(n, 1.0 / n)
    else:
        a /= a_sum

    best_seq = a.copy()
    best_score = 2.0 * n * np.max(full_convolution(a))

    # Parameters for the optimisation loop
    # ------------------------------------------------------------------
