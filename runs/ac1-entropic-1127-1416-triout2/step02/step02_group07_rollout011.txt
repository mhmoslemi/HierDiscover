"""
Propose a candidate step‑function by a smoothed sub‑gradient descent on the
probability simplex.

The objective to minimise is

    2 * n * max(conv(a, a)) / (sum(a)**2)

Because the ratio is scale‑invariant we keep the coefficients on the simplex
(sum = 1, a_i ≥ 0).  In each iteration we

1. compute the full self‑convolution a*a via an FFT,
2. build a soft‑max over the convolution values – this yields a weight
   vector w that approximates the “max’’ but stays smooth,
3. the weighted sum of the sub‑gradients of the individual convolution
   entries can be written as a correlation `a ⋆ w`.  This correlation is
   computed with a single extra FFT,
4. take a step opposite to the gradient, project back onto the simplex,
5. use a simple back‑tracking line‑search to guarantee descent,
6. occasional random jitter helps escape tiny plateaux.

We start from several sensible initial shapes (powers 1/(i+1)^p,
Dirichlet random draws and, if available, the previous best sequence) and
run the optimiser for the remaining time budget, keeping the overall best
sequence found.
"""

import time
from typing import List

import numpy as np
from numpy.fft import rfft, irfft


# ----------------------------------------------------------------------
# Helper utilities – all top‑level, no closures
# ----------------------------------------------------------------------


def project_onto_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the probability simplex
    { x | x ≥ 0, sum(x) = 1 }.
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(n, dtype=np.int64) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all zero – return the uniform distribution
        return np.full_like(v, 1.0 / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def full_convolution(a: np.ndarray) -> np.ndarray:
    """
    Full convolution a * a (length 2*len(a)-1) using an FFT.
    The FFT length is the next power of two for speed.
    """
    n = a.shape[0]
    size = 1 << (2 * n - 1).bit_length()          # ≥ 2n
    A = rfft(a, n=size)
    conv_full = irfft(A * A, n=size)
    return conv_full[: 2 * n - 1]


def weighted_correlation(a: np.ndarray, w: np.ndarray) -> np.ndarray:
    """
    Compute the weighted sum of sub‑gradients:
        G[i] = Σ_k w_k * a_{k-i}   (indices out of range give 0)

    This is exactly the linear correlation a ⋆ w.  It is obtained with one
    FFT pair.  The returned vector has length len(a).
    """
    n = a.shape[0]
    L = w.shape[0]               # should be 2*n-1
    size = 1 << (n + L - 1).bit_length()   # ≥ n+L-1
    A = rfft(a, n=size)
    W = rfft(w, n=size)
    corr_full = irfft(A * np.conj(W), n=size)
    return corr_full[:n]


def objective_score(a: np.ndarray) -> float:
    """
    Return the value 2 * n * max(conv(a, a)) for a vector that already
    satisfies sum(a) == 1.  This matches the value returned by
    `evaluate_sequence` (apart from the clamping which does not fire for
    our tiny, non‑negative entries).
    """
    n = a.shape[0]
    conv = full_convolution(a)
    return 2.0 * n * np.max(conv)


# ----------------------------------------------------------------------
# Core optimisation loop for a single start vector
# ----------------------------------------------------------------------


def descent_on_simplex(
    a0: np.ndarray,
    deadline: float,
    *,
    step_initial: float = 0.5,
) -> tuple[np.ndarray, float]:
    """
    Projected (smoothed) sub‑gradient descent on the simplex.
    Returns the best sequence seen and its objective value.
    """
    a = a0.copy()
    n = a.shape[0]

    # ensure a is a proper probability vector
    s = a.sum()
    if s <= 0:
        a = np.full(n, 1.0 / n)
    else:
        a = a / s

    best_seq = a.copy()
    best_score = objective_score(a)

    iteration = 0
    # initial temperature (smoothing parameter)
    # start fairly smooth, then cool down
    tau = max(best_score / (2 * n), 1e-6)   # roughly max(conv)

    # pre‑compute an FFT size that works for both self‑convolution
    # (length 2n‑1) and the correlation (length 3n‑2)
    size = 1 << (3 * n - 2).bit_length()

    while time.time() < deadline:
        # ----- 1. self‑convolution & current score -----
        A_fft = rfft(a, n=size)
        conv_full = irfft(A_fft * A_fft, n=size)
        conv = conv_full[: 2 * n - 1]                # linear part only
        max_val = conv.max()
        score = 2.0 * n * max_val

        if score < best_score:
            best_score = score
            best_seq = a.copy()
            print(f"Iter {iteration:06d}: new best {best_score:.6f}")

        # ----- 2. form a smooth approximation of the max -----
        #   w_k ∝ exp((conv_k - max_val) / tau)
        # shift by max_val for numerical stability
        if tau < 1e-12:
            tau = 1e-12
        shifted = (conv - max_val) / tau
        # avoid overflow/underflow – the max entry becomes exp(0)=1
        exp_vals = np.exp(shifted)
        w = exp_vals / exp_vals.sum()               # length 2n‑1, sum = 1

        # ----- 3. weighted sub‑gradient (correlation) -----
        grad = weighted_correlation(a, w)           # length n, sum(grad)=1

        # ----- 4. back‑tracking line search -----
        step = step_initial / np.sqrt(iteration + 1.0)
        improved = False
        for _ in range(8):
            a_candidate = project_onto_simplex(a - step * grad)
            # evaluate candidate
            A_c = rfft(a_candidate, n=size)
            conv_c_full = irfft(A_c * A_c, n=size)
            conv_c = conv_c_full[: 2 * n - 1]
            score_c = 2.0 * n * conv_c.max()
            if score_c < score:
                a = a_candidate
                improved = True
                step *= 1.1            # a little larger next time
                break
            step *= 0.5                # shrink and try again

        # ----- 5. occasional random jitter (escape plateaux) -----
        if not improved and iteration % 250 == 0 and iteration > 0:
            i, j = np.random.randint(0, n, size=2)
            if i != j:
                delta = 0.001 * a[i] * (np.random.rand() - 0.5)
                cand = a.copy()
                cand[i] = max(cand[i] - delta, 0.0)
                cand[j] = cand[j] + delta
                cand = project_onto_simplex(cand)
                A_c = rfft(cand, n=size)
                conv_c_full = irfft(A_c * A_c, n=size)
                conv_c = conv_c_full[: 2 * n - 1]
                score_c = 2.0 * n * conv_c.max()
                if score_c < best_score:
                    best_score = score_c
                    best_seq = cand.copy()
                    a = cand
                    print(f"Jitter improvement: {best_score:.6f}")

        # ----- 6. cool the temperature -----
        tau *= 0.995                     # very slow cooling
        iteration += 1

    return best_seq, best_score

def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs) -> List[float]:
    """
    Search for a non‑negative step‑function that minimises the evaluation
    function.  The search runs for at most `budget_s` seconds and returns
    a plain Python list of heights.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 5.0            # a few seconds margin

    # ------------------------------------------------------------------
    # Prepare several starting points (different shapes, random draws)
    # ------------------------------------------------------------------
    starts = []

    # 1. classic power‑law a_i ∝ 1/√(i+1)
    n0 = 5000
    raw = 1.0 / np.sqrt(np.arange(n0, dtype=float) + 1.0)
    starts.append(raw / raw.sum())

    # 2. a_i ∝ 1/(i+1)^p for a few p values
    for p in (0.55, 0.6, 0.65, 0.7):
        raw = 1.0 / np.power(np.arange(1, n0 + 1, dtype=float), p)
        starts.append(raw / raw.sum())

    # 3. a few Dirichlet random vectors (moderately sparse)
    for _ in range(3):
        starts.append(np.random.dirichlet(alpha=np.full(n0, 0.2)))

    # 4. previous best sequence if it exists
    try:
        prev = np.array(height_sequence_1, dtype=float)
        if prev.size > 0 and prev.sum() > 0:
            prev = prev / prev.sum()
            starts.append(prev)
    except Exception:
        pass  # variable not defined – ignore

    # Global trackers for the best sequence overall
    global_best_seq = None
    global_best_score = np.inf

    # ------------------------------------------------------------------
    # Run the optimiser on each start until time runs out
    # ------------------------------------------------------------------
    for idx, a_start in enumerate(starts):
        if time.time() >= deadline:
            break
        print(f"--- Optimisation run {idx + 1}/{len(starts)} (len={len(a_start)}) ---")
        # give each run the remaining budget (or a fraction of it)
        local_deadline = deadline
        seq, score = descent_on_simplex(a_start, local_deadline)

        if score < global_best_score:
            global_best_score = score
            global_best_seq = seq.copy()
            print(f">>> New global best after run {idx + 1}: {global_best_score:.6f}")

    # ------------------------------------------------------------------
    # Fallback in the very unlikely case nothing was produced
    # ------------------------------------------------------------------
    if global_best_seq is None:
        n_fallback = 2000
        uniform = np.full(n_fallback, 1.0 / n_fallback, dtype=float)
        global_best_seq = uniform
        global_best_score = objective_score(uniform)

    print(f"Final best score: {global_best_score:.6f} (length {len(global_best_seq)})")
    return global_best_seq.tolist()
