"""
Propose a step‑function sequence that minimises

    2·n·max(conv(a, a)) / (sum(a)^2)

The algorithm has two phases:

1. **Smoothed gradient descent** – a soft‑max approximation of the
   max‑operator is used (`softmax_α`).  The gradient of this smooth
   objective is obtained by a single FFT‑based convolution, then the
   iterate is projected onto the probability simplex.  The smoothing
   parameter `α` is increased slowly so the method behaves like a
   sub‑gradient method for the original problem but converges much
   faster early on.

2. **Exact sub‑gradient refinement** – once the time budget is almost
   exhausted we switch to the classic projected sub‑gradient scheme
   (identical to the reference implementation) to polish the solution.

Both phases keep the best vector seen so far (according to the exact
objective) and return it as a plain Python list.  The routine respects
the 1000 s time limit, prints progress information and works for any
available starting point (`height_sequence_1` or a fresh uniform start).
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper: Euclidean projection onto the probability simplex.
# ----------------------------------------------------------------------
def simplex_projection(v: np.ndarray) -> np.ndarray:
    """Project a vector onto { w | w_i ≥ 0 , Σ w_i = 1 } (Euclidean)."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1] - 1
    theta = cssv[cond][-1] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Exact objective (sum of coefficients is 1, so the denominator drops out).
# ----------------------------------------------------------------------
def exact_objective(a: np.ndarray) -> float:
    """2·n·max(conv(a, a)) for a probability vector."""
    n = a.shape[0]
    N = 1 << ((2 * n - 1) - 1).bit_length()          # next power of two ≥ 2n‑1
    fa = np.fft.rfft(a, N)
    conv = np.fft.irfft(fa * fa, N)[: 2 * n - 1]
    return 2.0 * n * conv.max()

# Main search routine.
# ----------------------------------------------------------------------

    else:

        a = init.astype(float)

    best_score = exact_objective(best_a)

    # 2. Pre‑computed FFT sizes.
    # --------------------------------------------------------------
    N_conv = 1 << ((2 * n - 1) - 1).bit_length()          # power of two ≥ 2n‑1
    N_grad = 1 << ((3 * n - 2) - 1).bit_length()         # power of two ≥ 3n‑2

    # --------------------------------------------------------------
    # 3. Phase 1 – smoothed (soft‑max) gradient descent.
    # --------------------------------------------------------------
    iter1 = 0

    while time.time() < phase1_end:

        #       and for the soft‑max weights) -----
        fa = np.fft.rfft(a, N_conv)
        conv_full = np.fft.irfft(fa * fa, N_conv)[: 2 * n - 1]
        max_c = conv_full.max()

    # --------------------------------------------------------------
    # 4. Phase 2 – exact max‑subgradient polishing.
    # --------------------------------------------------------------
    iter2 = 0
    # momentum variables for the classic sub‑gradient scheme
    velocity = np.zeros_like(a)
    beta = 0.9
    step0 = 0.5

    while time.time() < deadline:
        # ----- current convolution and the maximal entry -----
        fa = np.fft.rfft(a, N_conv)
        conv = np.fft.irfft(fa * fa, N_conv)[: 2 * n - 1]
        max_val = conv.max()
        cur_score = 2.0 * n * max_val
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"Iter {iter1 + iter2:7d} (sub): improved best = {best_score:.6f}")

        # ----- active set (indices where the maximum is attained) -----
        tol = max_val * 1e-12 + 1e-14
        active = np.where(conv >= max_val - tol)[0]

        # ----- sub‑gradient of the exact max (average over the active set) -----
        grad = np.zeros_like(a)
        for k in active:
            lo = max(0, k - n + 1)
            hi = min(n - 1, k)
            idx = np.arange(lo, hi + 1, dtype=int)
            partner = k - idx
            grad[idx] += 2.0 * a[partner]
        if active.size > 1:
            grad *= 1.0 / active.size

        # ----- Nesterov‑style momentum + projection -----
        step = step0 / np.sqrt(iter2 + 1.0)
        velocity = beta * velocity - step * grad
        a = simplex_projection(a + velocity)

        # ----- tiny random perturbation every few thousand steps -----
        if iter2 % 5000 == 0 and iter2 > 0:
            eps = 0.01 / np.sqrt(iter2 + 1.0)
            a = simplex_projection(a + eps * np.random.randn(n))

        iter2 += 1

        if best_score < 1.503:
            print(f"Target reached during phase 2 at iteration {iter1 + iter2}")
            break

    # --------------------------------------------------------------
    # 5. Return the best sequence discovered.
    # --------------------------------------------------------------
    return [float(v) for v in best_a.tolist()]
