```
def proj_simplex(v):
    if v.sum() == 1 and np.all(v >= 0):
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(len(v)) + 1
    cond = u - cssv / ind > 0
    rho = ind[cond].max()
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0)
    return w
```

"""
Optimisation of a non‑negative step‑function (height) sequence.
The objective is   2*n*max(conv(a,a)) / (sum(a)**2) .
Since the expression is homogeneous we fix sum(a)=1 and minimise
max(b) where b = a * a (linear convolution).  The algorithm works in
the Fourier domain:

*  a is zero‑padded to a power‑of‑two length L >= 3·n so that both the
   convolution and the correlation needed for the gradient are obtained
   without circular wrap‑around.
*  b = a⋆a is computed by one FFT and one inverse FFT.
*  The maximal entry of b defines a hard sub‑gradient (indicator of the
   active indices).  In addition we add a smooth variance term
   λ·∑(b−avg)² whose gradient is a correlation with the weight
   2·(b−avg).  The total weight vector w therefore is
        w = 1_{max} + λ·2·(b−avg) .
*  The gradient of the current objective is the linear correlation of w
   with a, which we also compute by FFT.
*  We take a projected gradient step preserving the simplex (non‑negative,
   sum=1).  An Adam‑style decay of the step‑size and occasional small
   random mass‑transfers help to escape shallow local minima.
*  The best sequence seen during the allotted budget (default 1000 s)
   is returned.

The implementation is fully vectorised and runs many thousands of
iterations per second, allowing a substantial improvement over the
baseline π/2≈1.5708.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
def binomial_sequence(length: int) -> np.ndarray:
    """First *length* central‑binomial coefficients c_k = C(2k,k)/4**k."""
    seq = np.empty(length, dtype=np.float64)
    seq[0] = 1.0
    for k in range(1, length):
        seq[k] = seq[k - 1] * ((2 * k - 1) * (2 * k)) / (4.0 * k * k)
    return seq


# ----------------------------------------------------------------------
def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


# ----------------------------------------------------------------------
def simple_normalise(v: np.ndarray) -> np.ndarray:
    """Clip negatives, renormalise to L1‑norm 1 (fallback to uniform)."""
    v = np.maximum(v, 0.0)
    s = v.sum()
    if s == 0.0:
        return np.full_like(v, 1.0 / v.size)
    return v / s


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: int = 1000) -> list[float]:
    """
    Search for a step‑function height vector that minimises
    evaluate_sequence(...).  A fast gradient descent in the frequency
    domain is performed for the whole time budget.

    Parameters
    ----------
    seed : int
        Random seed for reproducibility.
    budget_s : int
        Time budget in seconds.

    Returns
    -------
    list[float]
        The best sequence discovered (as plain Python list).
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.1   # safety margin

    # ------------------------------------------------------------------
    # Length of the sequence – use the provided construction if present.
    try:
        n = len(height_sequence_1)          # type: ignore[name-defined]
    except Exception:
        n = 8000

    # FFT length large enough for both convolution (2n‑1) and correlation
    L = next_pow2(3 * n)                     # > (2n‑1)+(n)‑1

    # Average value of the convolution (sum(b)=1, length=2n‑1)
    avg_b = 1.0 / (2 * n - 1)

    # Hyper‑parameters
    step = 0.5               # initial step size
    step_decay = 0.9995      # exponential decay
    min_step = 1e-6
    lam = 0.35               # weight of the variance‑flattening term
    tol_max = 1e-14          # tolerance for recognising active max entries

    # ------------------------------------------------------------------
    # Initial sequence – binomial shape perturbed a little to break symmetry.
    a = binomial_sequence(n).astype(np.float64)
    a = a * np.random.rand(n)         # random scaling each entry
    a = simple_normalise(a)           # L1 = 1

    best_seq = a.copy()
    best_score = float('inf')
    iter_cnt = 0
    last_imp_time = time.time()

    # ------------------------------------------------------------------
    while time.time() < deadline:
        # ----- 1. convolution a*a (linear) --------------------------------
        a_fft = np.fft.rfft(a, n=L)
        b = np.fft.irfft(a_fft * a_fft, n=L)[: 2 * n - 1]   # length 2n‑1

        cur_max = b.max()
        cur_score = 2.0 * n * cur_max                       # sum(a)=1
        if cur_score < best_score:
            best_score = cur_score
            best_seq = a.copy()
            print(f"[improve] iter={iter_cnt} score={best_score:.9f} "
                  f"time={time.time() - (deadline - budget_s):.2f}s")
            last_imp_time = time.time()

        # ----- 2. build weight vector w for the gradient -----------------
        # indicator of the active maximal entries
        max_idx = np.flatnonzero(b >= cur_max - tol_max)
        w = np.zeros_like(b)
        w[max_idx] = 1.0
        # add the variance‑flattening contribution: λ·2·(b−avg)

        w_fft = np.fft.rfft(w, n=L)

        # ----- 4. zero‑sum direction (keeps sum(a) unchanged) -------------
        sum_g = grad.sum()

        a_candidate = a + step * direction
        a_candidate = simple_normalise(a_candidate)

                a_candidate = simple_normalise(a_candidate)

        a = a_candidate

        # ----- 7. step‑size schedule --------------------------------------
        step = max(min_step, step * step_decay)
        iter_cnt += 1

    # ------------------------------------------------------------------
    print(f"[final] best score={best_score:.9f}, iterations={iter_cnt}")
    return best_seq.tolist()
