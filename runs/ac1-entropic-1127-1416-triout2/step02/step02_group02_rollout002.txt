"""
Propose a candidate step‑function sequence by minimizing

    2·n·max(a*a) / (sum(a)^2)

The algorithm combines three ideas that were not used together before:

1. **Smoothed max** –  instead of the nonsmooth max‑operator we first minimise
    the smooth approximation  fₐ(a)= (1/α)·log Σₖ exp(α·(a∗a)ₖ).  The gradient of
    fₐ is a weighted sum of the individual sub‑gradients and moves the mass
    away from *all* near‑maximal convolution entries at once.

2. **Adam‑style adaptive steps** –  the (sub)gradient is fed to an Adam optimiser,
    which automatically adapts the step size per coordinate.  After each update
    the iterate is projected onto the probability simplex (non‑negative,
    Σa=1) with the Duchi‑et‑al. O(n) Euclidean projection.

3. **Switch to exact sub‑gradient** –  once the smoothing parameter α has grown
    large (or after a fixed time fraction) the algorithm switches back to the
    true max‑subgradient, allowing a final fine‑tuning stage.

The routine respects a 1000 s budget, keeps the best vector seen so far and
occasionally injects a tiny random jitter to escape shallow plateaus.  It can
start from the previously discovered construction (`height_sequence_1`) or fall
back to a uniform distribution.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Simplex projection (Euclidean) – Duchi et al. (2008)
# ----------------------------------------------------------------------
def simplex_projection(v: np.ndarray) -> np.ndarray:
    """Project a vector onto the probability simplex {w ≥ 0, Σw = 1}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all entries are non‑positive – the uniform vector is the projection
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1] - 1               # zero‑based index
    theta = cssv[cond][-1] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Main optimisation routine
# ----------------------------------------------------------------------
def propose_candidate(budget_s: int = 1000) -> list:
    """
    Search for a low‑score step‑function.
    The function runs for up to ``budget_s`` seconds and returns the best
    sequence found (as a plain Python list of floats).
    """
    np.random.seed(0)                     # reproducibility
    deadline = time.time() + budget_s - 5.0   # safety margin

    # --------------------------------------------------------------
    # 1. Initialise a feasible vector of length n
    # --------------------------------------------------------------
    try:
        init = np.asarray(height_sequence_1, dtype=float)   # type: ignore[name-defined]
        n = init.shape[0]
    except Exception:
        # moderate size that fits comfortably into the time‑budget
        n = 5146
        init = np.full(n, 1.0 / n)

    a = init.copy()
    total = a.sum()
    if total <= 0:
        a = np.full(n, 1.0 / n)
    else:
        a = a / total                         # enforce Σa = 1

    # --------------------------------------------------------------
    # 2. Pre‑computed FFT sizes
    # --------------------------------------------------------------
    # length of a*a (linear convolution) = 2n‑1
    N1 = 2 * n - 1
    # length needed for a * reversed(p)   = n + (2n‑1) – 1 = 3n‑2
    N2 = 1 << int(np.ceil(np.log2(3 * n - 2)))   # next power of two (FFT speed)

    # --------------------------------------------------------------
    # 3. Optimiser state
    # --------------------------------------------------------------
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8
    lr = 0.04               # base learning‑rate for Adam
    m = np.zeros_like(a)    # first‑moment estimate
    v = np.zeros_like(a)    # second‑moment estimate

    # smoothing parameter (soft‑max temperature)
    alpha = 5.0
    alpha_growth = 1.001
    alpha_max = 500.0

    # status variables
    best_a = a.copy()
    # objective = 2·n·max(conv)  (sum(a)=1 already)
    fft_a_N2 = np.fft.rfft(a, n=N2)
    conv_full = np.fft.irfft(fft_a_N2 * fft_a_N2, n=N2)
    conv_auto = conv_full[:N1]                     # a*a
    best_score = 2.0 * n * conv_auto.max()
    last_improve = 0
    iteration = 0
    start_time = time.time()
    # fraction of total time to stay in the smoothed phase
    smooth_phase_fraction = 0.65

    while time.time() < deadline:
        # ----------------------------------------------------------
        # a) FFT of current iterate (single FFT, reused below)
        # ----------------------------------------------------------
        fft_a_N2 = np.fft.rfft(a, n=N2)

        # linear convolution a*a (full length N2); we need the first 2n‑1 entries
        conv_full = np.fft.irfft(fft_a_N2 * fft_a_N2, n=N2)
        conv_auto = conv_full[:N1]                # length 2n‑1

        max_val = conv_auto.max()
        cur_score = 2.0 * n * max_val

        # ----------------------------------------------------------
        # b) Choose gradient computation method
        # ----------------------------------------------------------
        elapsed = time.time() - start_time
        use_smooth = elapsed < smooth_phase_fraction * budget_s

        if use_smooth:
            # ---- smoothed max (log‑sum‑exp) ----
            # prevent overflow / underflow
            shift = conv_auto - max_val               # ≤ 0
            # exponentials may underflow to 0 – handle that case
            exp_term = np.exp(alpha * shift)
            sum_exp = exp_term.sum()
            if sum_exp == 0.0:
                # all mass at the exact maximiser
                p = np.zeros_like(conv_auto)
                p[np.argmax(conv_auto)] = 1.0
            else:
                p = exp_term / sum_exp               # soft‑max probabilities

            # ---- gradient = 2·(a ⋆ p) (cross‑correlation) ----
            rev_p = p[::-1]                         # reversed soft‑max vector
            fft_rev_p = np.fft.rfft(rev_p, n=N2)
            cross = np.fft.irfft(fft_a_N2 * fft_rev_p, n=N2)
            # indices 2n‑2, 2n‑3, … , n‑1 give the needed n entries
            grad = 2.0 * cross[(2 * n - 2) - np.arange(n)]

            # increase smoothing strength very slowly
            alpha = min(alpha * alpha_growth, alpha_max)
        else:
            # ---- exact sub‑gradient of the max entry ----
            tol = max_val * 1e-12 + 1e-14
            active = np.where(conv_auto >= max_val - tol)[0]

            grad = np.zeros_like(a)
            for k in active:
                low = max(0, k - n + 1)
                high = min(n - 1, k)
                idx = np.arange(low, high + 1, dtype=int)
                partner = k - idx
                grad[idx] += 2.0 * a[partner]
            if active.size > 1:
                grad *= 1.0 / active.size

        # ----------------------------------------------------------
        # c) Adam update with projection onto the simplex
        # ----------------------------------------------------------
        iteration += 1
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)

        m_hat = m / (1.0 - beta1 ** iteration)
        v_hat = v / (1.0 - beta2 ** iteration)

        step = lr / (np.sqrt(v_hat) + eps)
        a_candidate = simplex_projection(a - step * m_hat)

        a = a_candidate

        # ----------------------------------------------------------
        # d) Keep the best solution seen so far
        # ----------------------------------------------------------
        # (re‑use previously computed max_val if we are still on the same
        #  a – otherwise recompute – cheap relative to an FFT)
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            last_improve = iteration
            print(f"Iter {iteration:7d}: improved score = {best_score:.6f}")

        # ----------------------------------------------------------
        # e) Occasional jitter to escape flat plateaus
        # ----------------------------------------------------------
        if iteration % 8000 == 0:
            jitter = 0.001 / np.sqrt(iteration) * np.random.randn(n)
            a = simplex_projection(a + jitter)

        # ----------------------------------------------------------
        # f) Early exit if we have already beaten the target
        # ----------------------------------------------------------
        if best_score < 1.503:
            print(f"Target reached (score={best_score:.6f}) at iteration {iteration}")
            break

    # --------------------------------------------------------------
    # 4. Return the best sequence (list of Python floats)
    # --------------------------------------------------------------
    return [float(v) for v in best_a.tolist()]
