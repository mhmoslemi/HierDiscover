    a = np.array(seq, dtype=float)
    a = project_to_simplex(a)
    best_a = a.copy()
    best_score = compute_score(best_a)
    n = len(a)
    step = init_step

    while max_steps > 0 and time.time() < time_deadline:
        conv = convolution_fft(a, a)
        k_max = int(np.argmax(conv))
        # Compute contributions = a[i] * a[k_max-i]
        # For valid i:
        i_start = max(0, k_max - (n - 1))
        i_end = min(k_max, n - 1)
        idx_range = np.arange(i_start, i_end + 1)
        contributions = a[idx_range] * a[k_max - idx_range]
        donor_idx = idx_range[np.argmax(contributions)]

        # Determine recipient: choose index r with minimal a[k_max - r] value.
        # candidate recipients set: all indices in [0, n-1] where k_max - r in [0,n-1].
        # We can compute reverse contributions similarly.
        # But some indices may not have valid counterpart.
        # Simpler: choose j = argmin a (global).
        recipient_idx = int(np.argmin(a))

        if donor_idx == recipient_idx:
            # choose another recipient
            sorted_idx = np.argsort(a)
            for cand in sorted_idx:
                if cand != donor_idx:
                    recipient_idx = int(cand)
                    break

        delta = step * a[donor_idx]
        if delta <= 0:
            # reduce step and continue
            step *= 0.9
            max_steps -= 1
            continue
        # ensure not to exceed donor
        delta = min(delta, a[donor_idx])

        cand = a.copy()
        cand[donor_idx] -= delta
        cand[recipient_idx] += delta

        # avoid negative due to rounding
        if cand[donor_idx] < -1e-12:
            max_steps -= 1
            continue

        cand = project_to_simplex(cand)  # keep normalization (but maintain sum)

            best_score = cur_score
            best_a = cand.copy()
            a = cand

            # Slightly revert or try another move
            step *= 0.95

Propose a low‑score sequence for the self‑convolution benchmark.

Algorithm
---------
1.  Work on the probability simplex (∑a_i = 1, a_i ≥ 0).  The
    evaluation function is scale‑invariant, so the score for a simplex
    vector is 2·n·max(a∗a).

2.  For each candidate length n we run several independent restarts:
    * arcsine‑shaped start (the classic analytically‑motivated shape),
    * uniform start,
    * a random Dirichlet sample.

3.  Each restart is refined with a *smoothed‑max* projected gradient
    descent (Adam‑style momentum).  The smoothed objective is

        f_α(a) = (1/α)·log Σ_k exp(α·(a∗a)_k),

    whose gradient can be computed with two FFT‑based convolutions.
    The smoothing parameter α is increased from 2 up to 800,
    so the algorithm gradually focuses on the true max.

4.  After the gradient phase we apply a *directed hill‑climb*:
    the current worst convolution index k_max is identified; mass is
    transferred from a donor index that contributes most to that peak
    to a recipient index that contributes least, accepting only
    improving moves.  This explicitly flattens the convolution.

5.  The best sequence found over all lengths (up to 8192 when time
    permits) is returned.  A final greedy hill‑climb is performed if
    time remains.

The whole search respects the 1000 s time budget and never produces
negative or NaN entries; the harness will see a plain list of floats.

import time
import numpy as np

# ----------------------------------------------------------------------
# Basic linear‑algebraic utilities
# ----------------------------------------------------------------------


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of a vector onto the probability simplex."""
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / v.shape[0])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho = np.where(u - cssv / np.arange(1, v.shape[0] + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full convolution using a real FFT (fast)."""
    m = len(a) + len(b) - 1
    n_fft = next_pow2(m)
    A = np.fft.rfft(a, n=n_fft)
    B = np.fft.rfft(b, n=n_fft)
    C = A * B
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


def compute_score(a: np.ndarray) -> float:
    """
    The same value as `evaluate_sequence` for a simplex vector (∑a = 1).
    """
    n = len(a)
    conv = convolution_fft(a, a)
    max_val = float(np.max(conv))
    return 2.0 * n * max_val


def arcsine_initial(n: int) -> np.ndarray:
    """Arcsine‑shaped start: a_i ∝ 1 / sqrt((i+0.5)(n‑0.5‑i))."""
    i = np.arange(n, dtype=float)
    eps = 0.25
    vals = 1.0 / np.sqrt((i + 0.5) * (n - 0.5 - i) + eps)
    return vals / np.sum(vals)


# ----------------------------------------------------------------------
# Smoothed‑max gradient machinery (Adam optimizer)
# ----------------------------------------------------------------------


def smoothed_gradient(a: np.ndarray, alpha: float):
    """
    Gradient of the smoothed maximum
        f_α(a) = (1/α)·log Σ_k exp(α·(a∗a)_k)

    Returns
    -------
    grad : np.ndarray
        ∂f_α/∂a  (same shape as a)
    conv : np.ndarray
        The full self‑convolution a∗a (length 2n‑1)
    """
    n = len(a)
    conv = convolution_fft(a, a)                # length 2n‑1

    # soft‑max weights over the convolution entries
    maxc = np.max(conv)
    exp_term = np.exp(alpha * (conv - maxc))
    w = exp_term / np.sum(exp_term)             # ∑_k w_k = 1

    # convolution of reversed a with the weight vector
    a_rev = a[::-1]
    conv_rev = convolution_fft(a_rev, w)        # length 3n‑2
    start = n - 1
    rev_slice = conv_rev[start:start + n]       # needed part for each i

    # diagonal contribution w_{2i}
    w_diag = w[::2][:n]

    grad = 2.0 * rev_slice - w_diag * a
    return grad, conv


def improve_sequence_smoothed(
    initial_a: np.ndarray,
    time_deadline: float,
    max_iter: int = 300_000,
    eta0: float = 0.8,
    alpha_start: float = 2.0,
    alpha_end: float = 800.0,
    noise_every: int = 500,
    noise_scale: float = 1e-4,
) -> tuple[np.ndarray, float]:
    """
    Adam‑type projected gradient descent on the smoothed objective.
    Returns the best vector seen (as a NumPy array) and its unsmoothed score.
    """
    a = project_to_simplex(initial_a.astype(float).copy())
    best_a = a.copy()
    best_score = compute_score(a)

    n = len(a)

    # Adam hyper‑parameters
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    m = np.zeros_like(a)
    v = np.zeros_like(a)

    t = 0
    for t in range(1, max_iter + 1):
        if time.time() >= time_deadline:
            break

        # linear schedule for the smoothing parameter
        alpha = alpha_start + (alpha_end - alpha_start) * min(1.0, t / max_iter)

        grad, _ = smoothed_gradient(a, alpha)

        # Adam update (biased moments)
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)

        # bias‑corrected moments
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)

        # step size that decays like 1/√t (via eta0)
        step = eta0 / np.sqrt(t)
        a_new = a - step * m_hat / (np.sqrt(v_hat) + eps)

        if noise_every and (t % noise_every == 0):
            a_new += np.random.randn(n) * noise_scale

        a_new = project_to_simplex(a_new)

        cur_score = compute_score(a_new)
        if cur_score < best_score:
            best_score = cur_score
            best_a = a_new.copy()

        a = a_new

        if t % 5000 == 0:
            print(f"    iter {t:,d}  score {cur_score:.6f}  best {best_score:.6f}")

    return best_a, best_score


# ----------------------------------------------------------------------
# Directed hill‑climbing that explicitly flattens the max convolution
# ----------------------------------------------------------------------


def directed_hill_climb(seq, time_deadline: float,
                        max_steps: int = 30_000,
                        init_step: float = 0.05) -> tuple[list[float], float]:
    """
    Greedy hill‑climb that moves mass from the donor index that
    contributes most to the current worst convolution entry
    to the recipient index that contributes least.
    """
    a = np.array(seq, dtype=float)
    a = project_to_simplex(a)
    best_a = a.copy()
    best_score = compute_score(best_a)
    n = len(a)
    step = init_step

    while max_steps > 0 and time.time() < time_deadline:
        conv = convolution_fft(a, a)
        k_max = int(np.argmax(conv))

        # donor: index i with largest a_i * a_{k_max-i}
        i_start = max(0, k_max - (n - 1))
        i_end = min(k_max, n - 1)
        idx_range = np.arange(i_start, i_end + 1)
        contrib = a[idx_range] * a[k_max - idx_range]
        donor = int(idx_range[np.argmax(contrib)])

        # recipient: index with smallest a (global minimum)
        recipient = int(np.argmin(a))
        if recipient == donor:
            # pick the second‑smallest if necessary
            sorted_idx = np.argsort(a)
            for cand in sorted_idx:
                if cand != donor:
                    recipient = int(cand)
                    break

        delta = step * a[donor]
        if delta <= 0:
            step *= 0.9
            max_steps -= 1
            continue
        delta = min(delta, a[donor])

        cand = a.copy()
        cand[donor] -= delta
        cand[recipient] += delta

        # Projection is cheap and guarantees feasibility
        cand = project_to_simplex(cand)

        cur_score = compute_score(cand)

        if cur_score < best_score:
            best_score = cur_score
            best_a = cand.copy()
            a = cand
            step = min(step * 1.05, 0.5)
        else:
            step *= 0.95

        max_steps -= 1

    return best_a.tolist(), best_score


# ----------------------------------------------------------------------
# Main entry point required by the competition harness
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a step‑function (non‑negative sequence) that yields a low
    `evaluate_sequence` value.  The method mixes smoothed‑max Adam descent,
    a directed hill‑climb, and a final generic hill‑climb.
    """
    np.random.seed(seed)
    start_time = time.time()
    # tiny safety margin
    deadline = start_time + budget_s - 2.0

    best_seq = None
    best_val = float("inf")

    # ------------------------------------------------------------------
    # Optional fallback to a previously discovered construction.
    # ------------------------------------------------------------------
    try:
        fallback = [float(x) for x in height_sequence_1]
        fallback_val = evaluate_sequence(fallback)
        print(f"Fallback score = {fallback_val:.6f}")
        if fallback_val < best_val:
            best_val = fallback_val
            best_seq = fallback
    except NameError:
        pass

    # ------------------------------------------------------------------
    # Lengths to explore – we start small and grow while time permits.
    # ------------------------------------------------------------------
    lengths = [256, 512, 1024, 2048, 4096, 8192]

    for n in lengths:
        if time.time() >= deadline - 5.0:
            break
        print(f"\nOptimising length n = {n}")

        # Decide how many restarts we can afford (more restarts for small n)
        if n <= 1024:
            restarts = 5
        else:
            restarts = 3

        for r in range(restarts):
            if time.time() >= deadline - 5.0:
                break

            # Three initial shapes – we randomise the Dirichlet each restart
            init_choices = [
                arcsine_initial(n),
                np.full(n, 1.0 / n),
                np.random.dirichlet(np.ones(n)),
            ]
            init_a = init_choices[r % len(init_choices)]

            print(f"  restart {r + 1}/{restarts}")

            # ----- smoothed‑max Adam descent -----
            seq, val = improve_sequence_smoothed(
                initial_a=init_a,
                time_deadline=deadline,
                max_iter=300_000,
                eta0=0.8,
                alpha_start=2.0,
                alpha_end=800.0,
                noise_every=400,
                noise_scale=5e-5,
            )
            print(f"    after gradient: score {val:.6f}")

            # ----- directed hill‑climb (focuses on the worst peak) -----
            if time.time() < deadline - 1.0:
                seq, val = directed_hill_climb(
                    seq,
                    time_deadline=deadline,
                    max_steps=30_000,
                    init_step=0.05,
                )
                print(f"    after directed climb: score {val:.6f}")

            # Keep the best candidate seen so far
            if val < best_val:
                best_val = val
                best_seq = seq
                print(f"    *** New global best {best_val:.6f} (n={n}) ***")

    # ------------------------------------------------------------------
    # Final cheap generic hill‑climb (if time permits)
    # ------------------------------------------------------------------
    if best_seq is not None and time.time() < deadline - 1.0:
        print("\nFinal generic hill‑climb refinement")
        # simple pairwise exchange hill‑climb
        def simple_hill(seq, deadline, steps=40_000, step=0.05):
            a = np.array(seq, dtype=float)
            a = project_to_simplex(a)
            best_a = a.copy()
            best_score = compute_score(best_a)
            n = len(a)
            cur_step = step
            while steps > 0 and time.time() < deadline:
                i, j = np.random.randint(n, size=2)
                if i == j:
                    continue
                delta = cur_step * best_a[i]
                if delta <= 0:
                    continue
                cand = best_a.copy()
                cand[i] -= delta
                cand[j] += delta
                cand = project_to_simplex(cand)
                cur_score = compute_score(cand)
                if cur_score < best_score:
                    best_score = cur_score
                    best_a = cand
                    cur_step = min(cur_step * 1.05, 0.5)
                else:
                    cur_step *= 0.97
                steps -= 1
            return best_a.tolist(), best_score

        refined_seq, refined_val = simple_hill(best_seq, deadline, steps=50_000, step=0.05)
        print(f"    refined generic hill‑climb score = {refined_val:.6f}")
        if refined_val < best_val:
            best_val = refined_val
            best_seq = refined_seq

    # ------------------------------------------------------------------
    # Fallback in the unlikely event that we never produced a candidate.
    # ------------------------------------------------------------------
    if best_seq is None:
        best_seq = [1.0] * 256
        best_val = evaluate_sequence(best_seq)

    final_score = evaluate_sequence(best_seq)
    print(f"\nFinished search. Final reported score = {final_score:.6f}")

    # The harness expects a plain Python list of floats.
    return best_seq
