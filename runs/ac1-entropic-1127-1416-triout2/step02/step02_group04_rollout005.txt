"""
propose_candidate()

Search for a non‑negative vector a (sum = 1) that minimizes

    2·n·max(conv(a,a))       ( n = len(a) )

where conv is the ordinary linear convolution.  The optimizer consists of
three stages:

1.  Fast exploration of a few analytical and random seeds (uniform,
    decreasing ramp, a small Beta‑grid and a few Dirichlet draws).  The
    best seed is kept.

2.  Projected Adam / momentum descent on the simplex.  In each iteration
    we compute the self‑convolution with FFT, pick one (or a random
    one of a few) “active’’ index where the maximum is attained and use
    the exact sub‑gradient of that convolution entry.  An Adam‑style
    update is applied and the iterate is Euclidean‑projected onto the
    probability simplex (Duchi et al. 2008).

3.  A short stochastic refinement: repeatedly move a tiny amount of mass
    from a random index to another and keep the move if it improves the
    objective.

The whole routine respects the time budget (`budget_s` seconds) and
returns the best sequence found as a plain list of floats.
"""

import numpy as np

def simplex_projection(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the probability simplex:
        { w | w_i >= 0 , sum_i w_i = 1 }.
    Implementation follows Duchi et al. (2008).
    """
    v = np.asarray(v, dtype=float)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):                # all entries non‑positive → uniform
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1] - 1            # zero‑based
    theta = cssv[cond][-1] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def conv_max(a: np.ndarray) -> float:
    """
    Return max(conv(a,a)) for a vector a (assumed to sum to 1).
    Convolution is performed with FFT for speed.
    """
    n = a.shape[0]
    fft = np.fft.rfft(a, n=2 * n - 1)
    conv = np.fft.irfft(fft * fft, n=2 * n - 1)
    return float(conv.max())


def compute_objective(a: np.ndarray) -> float:
    """
    Objective 2·n·max(conv(a,a))  for a vector a on the simplex.
    """
    n = a.shape[0]
    return 2.0 * n * conv_max(a)


def subgradient_one_active(a: np.ndarray):
    """
    Compute a sub‑gradient of max(conv(a,a)) using a single active index.
    The function returns a pair (g, max_val) where g is the sub‑gradient
    (shape identical to a) and max_val = max(conv(a,a)).
    """
    n = a.shape[0]
    # convolution via FFT
    fft = np.fft.rfft(a, n=2 * n - 1)
    conv = np.fft.irfft(fft * fft, n=2 * n - 1)
    max_val = float(conv.max())

    # indices where the maximum is attained (within a tiny tolerance)
    eps = max_val * 1e-9
    active = np.where(conv >= max_val - eps)[0]

    # pick a random active index (if there are several)
    k = np.random.choice(active) if active.size > 1 else active[0]

    # build the exact gradient of the chosen convolution entry
    low = max(0, k - n + 1)
    high = min(n - 1, k)
    idx = np.arange(low, high + 1, dtype=int)
    partner = k - idx                      # partner indices in the sum
    g = np.zeros_like(a)
    g[idx] = 2.0 * a[partner]              # ∇_i (a * a)_k = 2·a_{k-i}
    return g, max_val


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: int = 1000) -> list[float]:
    """
    Heuristic optimiser for the step‑function evaluation problem.

    Parameters
    ----------
    seed : int
        Random seed for reproducibility.
    budget_s : int
        Time budget in seconds (default 1000).

    Returns
    -------
    list[float]
        The best sequence found (length equals the problem size, sum ≈ 1).
    """
    np.random.seed(seed)
    start_time = time.time()
    # keep a small safety margin so that the caller does not see a timeout
    deadline = start_time + budget_s - 1.0

    # ------------------------------------------------------------------
    # 1. Problem size
    # ------------------------------------------------------------------
    try:
        n = len(height_sequence_1)          # type: ignore[name-defined]
    except NameError:
        n = 5146

    # ------------------------------------------------------------------
    # 2. Build an initial pool of seeds
    # ------------------------------------------------------------------
    best_a = None
    best_score = np.inf

    def try_candidate(a):
        nonlocal best_a, best_score
        a = a.astype(float, copy=False)
        a_sum = a.sum()
        if a_sum <= 0:
            return
        a = a / a_sum                       # normalise to the simplex
        s = compute_objective(a)
        if s < best_score:
            best_score = s
            best_a = a.copy()
            print(f"New seed best = {best_score:.6f}")

    # a) decreasing linear ramp (classic construction)
    ramp = np.arange(n, 0, -1, dtype=float)
    ramp = 2.0 * ramp / (n * (n + 1))
    try_candidate(ramp)

    # b) uniform distribution
    uniform = np.full(n, 1.0 / n)
    try_candidate(uniform)

    # c) a few Beta‑shapes (coarse grid)
    xs = (np.arange(n) + 0.5) / n
    for alpha in np.linspace(0.3, 5.0, 9):
        for beta in np.linspace(0.3, 5.0, 9):
            if time.time() >= deadline:
                break
            logw = (alpha - 1.0) * np.log(xs) + (beta - 1.0) * np.log(1.0 - xs)
            logw = np.clip(logw, -700, 700)            # avoid overflow
            w = np.exp(logw)
            if not np.isfinite(w).all() or w.sum() == 0:
                w = np.ones_like(w)
            try_candidate(w)

    # d) random Dirichlet‑like vectors (few of them)
    for _ in range(30):
        if time.time() >= deadline:
            break
        conc = np.random.uniform(0.1, 3.0, size=n)
        a = np.random.gamma(conc, 1.0)
        try_candidate(a)

    # If a global solution already exists, use it as an additional seed.
    try:
        global height_sequence_1                     # type: ignore[name-defined]
        a0 = np.asarray(height_sequence_1, dtype=float)
        try_candidate(a0)
    except Exception:
        pass

    # ------------------------------------------------------------------
    # 3. Projected Adam‑style (momentum) descent from the best seed
    # ------------------------------------------------------------------
    if best_a is None:
        # Fallback – should never happen
        best_a = uniform.copy()
        best_score = compute_objective(best_a)

    a = best_a.copy()
    m = np.zeros_like(a)          # first‑moment estimate (momentum)
    v = np.zeros_like(a)          # second‑moment estimate (RMS)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    base_lr = 0.8                # base learning rate; tuned empirically
    iteration = 0

    while time.time() < deadline:
        # --------------------------------------------------------------
        # evaluate current point and possibly update the global best
        # --------------------------------------------------------------
        g, max_val = subgradient_one_active(a)
        cur_score = 2.0 * n * max_val
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"Iter {iteration:6d}: new best {best_score:.6f}")

        # --------------------------------------------------------------
        # Adam update (with bias correction) followed by simplex projection
        # --------------------------------------------------------------
        iteration += 1
        lr_t = base_lr / np.sqrt(iteration)          # simple decay

        m = beta1 * m + (1.0 - beta1) * g
        v = beta2 * v + (1.0 - beta2) * (g * g)

        # bias‑corrected moments
        m_hat = m / (1.0 - beta1 ** iteration)
        v_hat = v / (1.0 - beta2 ** iteration)

        step = lr_t * m_hat / (np.sqrt(v_hat) + eps)
        a = simplex_projection(a - step)

    # ------------------------------------------------------------------
    # 4. Short stochastic refinement (tiny mass swaps)
    # ------------------------------------------------------------------
    # use any remaining time (if any)
    while time.time() < deadline:
        # pick two distinct indices and move a small amount of mass
        i, j = np.random.randint(n, size=2)
        if i == j or a[i] <= 0:
            continue
        delta = np.random.uniform(0.0, min(a[i], 1e-3))
        a_candidate = a.copy()
        a_candidate[i] -= delta
        a_candidate[j] += delta
        # a_candidate is already on the simplex (sum unchanged) and non‑negative
        cand_score = compute_objective(a_candidate)
        if cand_score < best_score:
            best_score = cand_score
            best_a = a_candidate.copy()
            a = a_candidate
            print(f"Random tweak improved to {best_score:.6f}")

    # ------------------------------------------------------------------
    # 5. Return the best solution as a plain list of floats
    # ------------------------------------------------------------------
    print(f"Finished – best score = {best_score:.6f}")
    return [float(v) for v in best_a.tolist()]
