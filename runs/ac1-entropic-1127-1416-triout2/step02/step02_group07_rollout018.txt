"""
Propose a candidate step‑function by minimizing the supremum norm of the
self‑convolution.  The method improves on the reference sub‑gradient
descent by

* using the Adam optimiser (adaptive step‑size and momentum),
* aggregating sub‑gradients from all maximising shifts (the true
  sub‑gradient of the point‑wise maximum),
* performing a cheap projection onto the probability simplex,
* injecting small random “jitter’’ moves to escape plateaux,
* keeping the best sequence seen during the whole 1000 s budget.

The algorithm works on a fixed length (default 5000) and keeps the
coefficients on the simplex (non‑negative, sum = 1).  The returned list
can be fed directly to ``evaluate_sequence``.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper functions (top level, no closures)
# ----------------------------------------------------------------------


def project_onto_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the probability simplex:
        { x | x >= 0, sum(x) = 1 }.
    Returns a new array (input is not modified).
    """
    n = v.shape[0]
    # sort descending
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(n) + 1
    # find the rho index
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all entries are zero – return the uniform distribution
        return np.full_like(v, 1.0 / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def full_convolution(a: np.ndarray) -> np.ndarray:
    """
    Compute the full discrete convolution a * a using an FFT.
    Length of the result is 2*len(a) - 1.
    """
    n = a.shape[0]
    # length for FFT (next power of two for speed)
    size = 2 * n
    fft_len = 1 << (size - 1).bit_length()
    fft_a = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(fft_a * fft_a, n=fft_len)
    return conv[: 2 * n - 1]


def gradient_at_shifts(a: np.ndarray, shifts) -> np.ndarray:
    """
    Sub‑gradient of the point‑wise maximum of the self‑convolution.
    ``shifts`` is an iterable of indices k where the convolution attains
    its maximal value (or is very close to it).  The sub‑gradient of the
    max is any convex combination of the individual gradients; we simply
    average them.
    """
    n = a.shape[0]
    g = np.zeros_like(a)
    for k in shifts:
        low = max(0, k - n + 1)
        high = min(k, n - 1)
        if low > high:
            continue
        idx = np.arange(low, high + 1, dtype=int)
        g[idx] += a[k - idx]
    if len(shifts) > 0:
        g /= len(shifts)
    return g


# ----------------------------------------------------------------------
# Main search routine
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 12345, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative height sequence that minimises

        2 * n * max(conv(a, a)) / (sum(a) ** 2)

    using an Adam‑based projected sub‑gradient method with occasional
    random perturbations.  The routine runs for at most ``budget_s`` seconds
    and returns the best list of floats found.
    """
    np.random.seed(seed)

    # keep a few seconds spare for final printing / return
    deadline = time.time() + budget_s - 5.0

    # ------------------------- parameters ------------------------------
    n = 5000                     # length of the step‑function
    # --- initialise with the classic 1/√(i+1) shape (sum == 1) ----------
    raw = 1.0 / np.sqrt(np.arange(n, dtype=float) + 1.0)
    a = raw / raw.sum()          # now sum(a) == 1

    # Adam hyper‑parameters
    lr = 0.5                     # base learning rate
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8

    m = np.zeros_like(a)         # first moment estimate
    v = np.zeros_like(a)         # second moment estimate

    # keep the best sequence seen so far
    best_seq = a.copy()
    best_score = 2.0 * n * np.max(full_convolution(a))
    print(f"Initial score (n={n}): {best_score:.6f}")

    iteration = 0
    # initial convolution (used for the first gradient)
    conv = full_convolution(a)

    while time.time() < deadline:
        # -----------------------------------------------------------------
        # 1. locate the (near) maximal entries of the convolution
        # -----------------------------------------------------------------
        max_val = conv.max()
        # indices where the value is within a tiny tolerance of the maximum
        tol = max(1e-12, max_val * 1e-10)
        max_idx = np.where(conv >= max_val - tol)[0]

        # -----------------------------------------------------------------
        # 2. sub‑gradient (average over all maximal shifts)
        # -----------------------------------------------------------------
        g = gradient_at_shifts(a, max_idx)

        # -----------------------------------------------------------------
        # 3. Adam update (projected on the simplex)
        # -----------------------------------------------------------------
        iteration += 1
        m = beta1 * m + (1.0 - beta1) * g
        v = beta2 * v + (1.0 - beta2) * (g * g)

        m_hat = m / (1.0 - beta1 ** iteration)
        v_hat = v / (1.0 - beta2 ** iteration)

        step = lr * m_hat / (np.sqrt(v_hat) + eps)
        a_candidate = a - step                     # move opposite to gradient
        a = project_onto_simplex(a_candidate)      # keep feasibility

        # -----------------------------------------------------------------
        # 4. evaluate the new point (one FFT per iteration)
        # -----------------------------------------------------------------
        conv = full_convolution(a)
        new_score = 2.0 * n * conv.max()
        if new_score < best_score:
            best_score = new_score
            best_seq = a.copy()
            print(f"Iter {iteration:06d}: improved score = {best_score:.6f}")

        # -----------------------------------------------------------------
        # 5. occasional random jitter to jump out of shallow plateaux
        # -----------------------------------------------------------------
        if iteration % 250 == 0:
            # pick two distinct positions
            i, j = np.random.randint(0, n, size=2)
            if i != j:
                # move a small fraction of mass from i to j
                delta = 0.003 * a[i] * np.random.uniform(-1.0, 1.0)
                cand = a.copy()
                cand[i] = max(cand[i] - delta, 0.0)
                cand[j] = cand[j] + delta
                cand = project_onto_simplex(cand)

                # quick evaluation
                cand_score = 2.0 * n * np.max(full_convolution(cand))
                if cand_score < best_score:
                    best_score = cand_score
                    best_seq = cand.copy()
                    a = cand            # accept the jittered point
                    conv = full_convolution(a)
                    print(f"    jitter improvement: score = {best_score:.6f}")

        # safety guard – avoid infinite loops (should not happen)
        if iteration >= 5_000_000:
            break

    # ----------------------------------------------------------------------
    # Fallback handling (very unlikely to be needed)
    # ----------------------------------------------------------------------
    if best_seq is None or best_seq.size == 0:
        try:
            best_seq = np.array(height_sequence_1, dtype=float)
            best_score = float('inf')
        except Exception:
            # uniform fallback (always yields a score of 2.0)
            best_seq = np.full(2000, 1.0 / 2000, dtype=float)
            best_score = float('inf')

    print(f"Final best score: {best_score:.6f} (length {len(best_seq)})")
    # return a plain Python list for the driver
    return best_seq.tolist()
