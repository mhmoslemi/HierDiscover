"""
candidate_generator.py

Search for a non‑negative sequence that minimizes
    E(a) = 2·n·max(a∗a) / (∑a)²
where ∗ is the ordinary (full) convolution.
The code combines three ideas:

1. **Diversified starts** – a few parametric families (uniform,
   decreasing geometric, power‑law, arcsine) are turned into probability
   vectors (sum = 1) and used as initial points.

2. **Adam optimisation of a smoothed max** – the non‑smooth
   max‑convolution is approximated by a log‑sum‑exp
   `f_α(a) = (1/α)·log Σ_k exp(α·(a∗a)_k)`.  Its gradient can be
   computed with two FFT‑based convolutions.  Adam (first/second moment
   adaptation) gives robust, fast progress even in high dimensions.

3. **Long stochastic hill‑climbing** – after the gradient phase the
   best vector is refined by many cheap pairwise mass‑transfer moves
   (and occasional Gaussian perturbations).  Moves that improve the
   true objective are kept; the step size is adapted automatically.

The whole search is bounded by the supplied budget (default 1000 s).
The function `propose_candidate` follows the competition’s required
signature and returns a plain list of floats.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Basic utilities
# ----------------------------------------------------------------------


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of a vector onto the probability simplex."""
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / v.shape[0])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho = np.where(u - cssv / np.arange(1, v.shape[0] + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full (linear) convolution using real FFT – fast for many calls."""
    m = len(a) + len(b) - 1
    n_fft = next_pow2(m)
    A = np.fft.rfft(a, n=n_fft)
    B = np.fft.rfft(b, n=n_fft)
    C = A * B
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


def compute_score(a: np.ndarray) -> float:
    """
    Score identical to `evaluate_sequence` when the vector sums to 1.
    """
    n = len(a)
    conv = convolution_fft(a, a)
    max_val = float(np.max(conv))
    return 2.0 * n * max_val


# ----------------------------------------------------------------------
# Initial families
# ----------------------------------------------------------------------


def uniform_initial(n: int) -> np.ndarray:
    """Flat distribution – each entry = 1/n."""
    return np.full(n, 1.0 / n)


def arcsine_initial(n: int) -> np.ndarray:
    """a_i ∝ 1 / sqrt((i+0.5)(n‑0.5‑i))."""
    i = np.arange(n, dtype=float)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - 0.5 - i))
    return vals / np.sum(vals)


def geometric_initial(n: int, r: float) -> np.ndarray:
    """Decreasing geometric: a_i ∝ r**i  (0<r<1)."""
    v = r ** np.arange(n, dtype=float)
    return v / np.sum(v)


def powerlaw_initial(n: int, p: float) -> np.ndarray:
    """Power law decreasing: a_i ∝ (i+1)^(-p)."""
    v = 1.0 / np.power(np.arange(1, n + 1, dtype=float), p)
    return v / np.sum(v)


# ----------------------------------------------------------------------
# Smoothed‑max gradient (log‑sum‑exp) and Adam optimiser
# ----------------------------------------------------------------------


def smoothed_gradient(a: np.ndarray, alpha: float):
    """
    Gradient of the smoothed maximum
        f_α(a) = (1/α)·log Σ_k exp(α·(a∗a)_k)
    Returns (grad, conv) where conv = a∗a (full length 2n‑1).
    """
    n = len(a)
    conv = convolution_fft(a, a)                     # length 2n‑1
    maxc = np.max(conv)                             # for numerical stability
    w = np.exp(alpha * (conv - maxc))
    w_sum = np.sum(w)
    w /= w_sum                                      # soft‑max weights, Σ w = 1
    # reversed a for the Jacobian term
    a_rev = a[::-1]
    tmp = convolution_fft(a_rev, w)                 # length 3n‑2
    start = n - 1
    rev_slice = tmp[start:start + n]                # needed component of gradient
    w_diag = w[::2][:n]                             # weights where i=j (even indices)
    grad = 2.0 * rev_slice - w_diag * a
    return grad, conv


def adam_optimize(
    init_a: np.ndarray,
    deadline: float,
    max_iter: int = 200_000,
    eta0: float = 0.8,
    alpha_start: float = 1.0,
    alpha_end: float = 2000.0,
    beta1: float = 0.9,
    beta2: float = 0.999,
    eps: float = 1e-8,
) -> tuple[np.ndarray, float]:
    """
    Adam optimisation of the smoothed max‑convolution objective.
    Returns the best vector (already on the simplex) together with its
    true (unsmoothed) score.
    """
    a = project_to_simplex(init_a.astype(float).copy())
    n = len(a)
    m = np.zeros_like(a)          # first moment
    v = np.zeros_like(a)          # second moment
    best_a = a.copy()
    best_score = compute_score(a)

    for t in range(1, max_iter + 1):
        if time.time() >= deadline:
            break

        # linear homotopy for the smoothing parameter
        alpha = alpha_start + (alpha_end - alpha_start) * min(1.0, t / max_iter)

        grad, _ = smoothed_gradient(a, alpha)

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)

        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)

        a_new = a - eta0 * m_hat / (np.sqrt(v_hat) + eps)
        a_new = project_to_simplex(a_new)

        cur_score = compute_score(a_new)
        if cur_score < best_score:
            best_score = cur_score
            best_a = a_new.copy()

        a = a_new

        # occasional progress dump (very cheap)
        if t % 50000 == 0:
            print(f"    Adam iter {t:,d} – score {cur_score:.6f} – best {best_score:.6f}")

    return best_a, best_score


# ----------------------------------------------------------------------
# Stochastic hill‑climbing refinement
# ----------------------------------------------------------------------


def hill_climb(seq, deadline: float, max_steps: int = 200_000, init_step: float = 0.1):
    """
    Pairwise mass‑transfer hill‑climb with an adaptive step size.
    Occasionally a global Gaussian perturbation is tried.
    """
    a = np.array(seq, dtype=float)
    a = project_to_simplex(a)
    best_a = a.copy()
    best_score = compute_score(best_a)
    n = len(a)
    step = init_step

    while max_steps > 0 and time.time() < deadline:
        max_steps -= 1

        if np.random.rand() < 0.1:                     # global perturbation
            cand = best_a + np.random.normal(scale=step, size=n)
            cand = project_to_simplex(cand)
        else:                                          # pairwise exchange
            i, j = np.random.randint(n, size=2)
            if i == j:
                continue
            delta = step * best_a[i]
            if delta <= 0:
                continue
            cand = best_a.copy()
            cand[i] -= delta
            cand[j] += delta
            if cand[i] < 0:
                continue
            cand = project_to_simplex(cand)

        cur_score = compute_score(cand)

        # accept only improvements (strictly better)
        if cur_score < best_score:
            best_score = cur_score
            best_a = cand
            step = min(step * 1.08, 0.5)   # slowly enlarge successful step
        else:
            step *= 0.985                  # shrink on failure

    return best_a.tolist(), best_score

        if refined_val < best_val:
            best_val = refined_val
            best_seq = refined_seq

    # ------------------------------------------------------------------
    # Fallback in the extremely unlikely event that nothing was built
    # ------------------------------------------------------------------
    if best_seq is None:
        best_seq = [1.0] * 256
        best_val = evaluate_sequence(best_seq)

    final_score = evaluate_sequence(best_seq)
    print(f"\nFinished search. Final reported score = {final_score:.6f}")

    # The harness expects a plain Python list of floats.
    return best_seq
```
