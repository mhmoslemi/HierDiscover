def full_convolve_fft(x, y):
    size = len(x) + len(y) - 1
    nfft = 1 << (size - 1).bit_length()
    X = np.fft.rfft(x, nfft)
    Y = np.fft.rfft(y, nfft)
    z = np.fft.irfft(X * Y, nfft)[:size]
    return z

"""
Propose a better step‑height sequence.

Algorithm
---------
We treat the objective
    F(a) = 2·n·max(a∗a)   (a is non‑negative, Σa = 1)
as a smooth optimisation problem by replacing the hard max with a soft‑max
(temperature β).  For a fixed β the smooth surrogate is

    f_β(a) = (1/β)·log Σ_k exp(β·(a∗a)_k) .

Its gradient can be written in closed form:
    ∂f_β/∂a = (a ⋆ w) + a ,   w_k ∝ exp(β·(a∗a)_k) , Σ w = 1,
where “⋆” denotes the cross‑correlation of w with a.
The gradient points *up* the surrogate, therefore we move in the opposite
direction and project back onto the probability simplex (non‑negative,
unit‑sum).  The temperature β is increased slowly (simulated‑annealing) so
that the method first makes large, smooth moves and later concentrates on
the true max.  All convolutions are performed with FFTs (O(n log n))
which allows us to handle sequences of several thousand entries inside the
1000 s time budget.

The routine also keeps the best sequence seen (according to the original
`evaluate_sequence` function), injects occasional random mass transfers to
avoid stagnation and decays the step size.

The function returns the best sequence found before the deadline.
"""

import time
import numpy as np

# --------------------------------------------------------------
def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of a non‑negative vector onto the simplex Σv = 1."""
    v = np.maximum(v, 0.0)
    s = v.sum()
    if s == 0.0:
        return np.full_like(v, 1.0 / v.size)
    return v / s


# --------------------------------------------------------------
def full_convolve_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution using a single real FFT (size is a power‑of‑2)."""
    size = x.size + y.size - 1
    nfft = 1 << (size - 1).bit_length()      # next power of two
    X = np.fft.rfft(x, n=nfft)
    Y = np.fft.rfft(y, n=nfft)
    z = np.fft.irfft(X * Y, n=nfft)[:size]
    # Numerical noise can produce tiny negative values – clip for safety.
    return np.maximum(z, 0.0)

    """
