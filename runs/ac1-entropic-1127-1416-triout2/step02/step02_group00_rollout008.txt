    if sum_b == 0:

    if not res.success:
        return None, False

"""
candidate_proposer.py

Search for a non‑negative step‑function (vector) that minimizes

    2·n·max(a*a) / (sum(a)**2)

(the evaluation used by the competition).  The method combines three
ingredients:

1.  A fast random scan of a two‑parameter symmetric shape family
    a_i ∝ ((i+ε)(n‑1‑i+ε))⁻ᵖ.  The best shape for each length n is kept.

2.  Projected (Nesterov‑accelerated) sub‑gradient descent on the
    ∞‑norm of the self‑convolution (the same idea as the reference
    implementation, but with momentum and occasional noise).

3.  An occasional linear‑programming “boost” step (when SciPy is
    available).  For a fixed vector a we maximise Σ b_j subject to
    a * b ≤ M (where M is the current maximal self‑convolution).  The
    optimiser b is renormalised to unit L¹‑norm, which strictly reduces
    the objective whenever Σ b_j>1.

The search is performed for several lengths (256 … 2048, optionally
4096) and the best candidate ever seen is returned.  A final polishing
stage (sub‑gradient + optional LP) is applied to the global best.
All helpers are top‑level, no closures or lambdas are used.
"""

import time
import math
import numpy as np
from typing import List, Tuple

# ----------------------------------------------------------------------
# Low‑level utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Convolution via real FFT (zero‑padded to the next power of two)."""
    m = len(a) + len(b) - 1
    n_fft = next_pow2(m)
    A = np.fft.rfft(a, n=n_fft)
    B = np.fft.rfft(b, n=n_fft)
    C = A * B
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


def max_convolution_and_index(a: np.ndarray) -> Tuple[float, int]:
    """Return (max value, arg‑max index) of a*a."""
    conv = convolution_fft(a, a)
    idx = int(np.argmax(conv))
    return float(conv[idx]), idx


def subgradient_of_max_at_index(a: np.ndarray, k: int) -> np.ndarray:
    """
    Sub‑gradient of the maximal self‑convolution entry (index k) with
    respect to a.  For each i, ∂c_k/∂a_i = a_{k‑i} if the index lies
    inside the vector; the diagonal term (k = 2·i) is doubled.
    """
    n = len(a)
    grad = np.zeros_like(a)
    i = np.arange(n)
    j = k - i
    mask = (j >= 0) & (j < n)
    grad[mask] = a[j[mask]]
    if k % 2 == 0:
        diag = k // 2
        if diag < n:
            grad[diag] = 2.0 * a[diag]
    return grad


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the probability simplex
    {x | x_i ≥ 0, Σ_i x_i = 1}.
    Implementation follows Duchi et al. (2008).
    """
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / len(v))

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho = np.where(u - cssv / np.arange(1, len(v) + 1) > 0)[0]
    if len(rho) == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def compute_score(a: np.ndarray) -> float:
    """Scale‑invariant objective used by the official evaluator."""
    n = len(a)
    conv = convolution_fft(a, a)
    max_val = float(np.max(conv))
    s = float(np.sum(a))
    if s < 1e-12:
        return float("inf")
    return 2.0 * n * max_val / (s * s)


def arcsine_initial(n: int) -> np.ndarray:
    """Classic arcsine‑shaped start: a_i ∝ 1/√((i+½)(n‑½‑i))."""
    i = np.arange(n, dtype=float)
    eps = 0.25
    vals = 1.0 / np.sqrt((i + 0.5) * (n - 0.5 - i) + eps)
    return vals / np.sum(vals)


def generate_shape(n: int, p: float, eps: float) -> np.ndarray:
    """
    Produce a symmetric density a_i ∝ ((i+eps)*(n‑1‑i+eps))⁻ᵖ,
    then normalise to unit L¹ norm.
    """
    i = np.arange(n, dtype=float)
    base = ((i + eps) * (n - 1 - i + eps)) ** (-p)
    total = base.sum()
    if total == 0.0:
        return np.full(n, 1.0 / n)
    return base / total

def subgradient_refine(
    initial_a: np.ndarray,
    time_deadline: float,
    max_iter: int = 200_000,
    eta0: float = 0.5,
    noise_every: int = 0,
    noise_scale: float = 0.0,
) -> Tuple[np.ndarray, float]:
    """
    Projected (Nesterov‑accelerated) sub‑gradient descent on the
    max‑convolution objective.  Returns the best vector seen and its
    internal (fast) score.
    """
    a = project_to_simplex(initial_a.astype(float).copy())
    best_a = a.copy()
    best_sc = compute_score(a)

    a_prev = a.copy()
    it = 0
    while time.time() < time_deadline and it < max_iter:
        it += 1
        # worst convolution entry
        _, k_max = max_convolution_and_index(a)
        grad = subgradient_of_max_at_index(a, k_max)

        # diminishing step size
        eta = eta0 / math.sqrt(it)

        # Nesterov momentum (polyak)
        beta = (it - 2) / (it + 1) if it > 1 else 0.0
        a_new = a - eta * grad + beta * (a - a_prev)

        # optional tiny noise to escape plateaus
        if noise_every and (it % noise_every == 0):
            a_new += np.random.randn(len(a)) * noise_scale

        a_new = project_to_simplex(a_new)

        cur_sc = compute_score(a_new)
        if cur_sc < best_sc:
            best_sc = cur_sc
            best_a = a_new.copy()

        a_prev = a
        a = a_new

        if it % 5000 == 0:
            print(f"    subgrad iter={it:,d} cur={cur_sc:.6f} best={best_sc:.6f}")

    return best_a, best_sc


def solve_lp_for_b(a: np.ndarray, M: float) -> Tuple[np.ndarray, bool]:
    """
    Solve  max Σ b_j   subject to   a * b ≤ M   and   b ≥ 0.
    Returns the optimal b and a flag indicating success.
    The function uses SciPy's high‑performance ``highs`` LP solver.
    """
    try:
        import scipy.optimize
        import scipy.sparse
    except Exception:   # SciPy not available
        return np.empty(0), False

    n = len(a)
    # Build sparse constraint matrix for all k = 0 … 2n‑2
    rows = []
    cols = []
    data = []
    for k in range(2 * n - 1):
        i_min = max(0, k - (n - 1))
        i_max = min(n - 1, k)
        i_vals = np.arange(i_min, i_max + 1, dtype=int)
        j_vals = k - i_vals
        rows.extend([k] * len(i_vals))
        cols.extend(j_vals.tolist())
        data.extend(a[i_vals].tolist())
    A = scipy.sparse.csr_matrix((data, (rows, cols)), shape=(2 * n - 1, n))
    c = -np.ones(n)                     # maximise sum(b) == minimise -sum(b)
    b_ub = np.full(2 * n - 1, M, dtype=float)
    bounds = [(0, None)] * n

    try:
        res = scipy.optimize.linprog(
            c, A_ub=A, b_ub=b_ub, bounds=bounds, method="highs", options={"presolve": True}
        )
    except Exception:
        return np.empty(0), False

    if not res.success:
        return np.empty(0), False
    return res.x, True


def lp_boost(a: np.ndarray, time_deadline: float, max_lp_iters: int = 5) -> Tuple[np.ndarray, float]:
    """
    Apply a few LP‑boost steps: each step solves the LP described above,
    renormalises the solution and repeats while an improvement is possible.
    Returns the (possibly) improved vector and its fast score.
    """
    best_a = a.copy()
    best_sc = compute_score(best_a)
    it = 0
    while time.time() < time_deadline and it < max_lp_iters:
        M, _ = max_convolution_and_index(best_a)
        b, ok = solve_lp_for_b(best_a, M)
        if not ok:
            break
        sum_b = b.sum()
        if sum_b <= 1.0 + 1e-8:    # no improvement possible
            break
        candidate = b / sum_b
        cand_sc = compute_score(candidate)
        if cand_sc < best_sc:
            best_sc = cand_sc
            best_a = candidate
            print(f"    LP boost {it+1}: sum(b)={sum_b:.4f}, score={best_sc:.6f}")
        else:
            break
        it += 1
    return best_a, best_sc

def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs) -> List[float]:
    """
    Search for a step‑function (non‑negative sequence) with a low
    ``evaluate_sequence`` value.  The routine respects the time budget,
    tries several lengths, randomly explores a two‑parameter shape
    family, refines the best candidates with accelerated sub‑gradient
    descent, applies occasional LP‑boost steps (when possible), and
    finally returns the best vector found as a plain list of floats.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 1.0          # safety margin

    best_seq: List[float] = []
    best_val: float = float("inf")

    # ------------------------------------------------------------------
    # 0.  Fallback to a previously discovered construction, if any.
    # ------------------------------------------------------------------
    try:
        fallback = [float(x) for x in height_sequence_1]   # type: ignore
        val = evaluate_sequence(fallback)
        print(f"Fallback (height_sequence_1) score = {val:.6f}")
        if val < best_val:
            best_val = val
            best_seq = fallback
    except NameError:
        pass

    # ------------------------------------------------------------------
    # 1.  Exploration over several lengths.
    # ------------------------------------------------------------------
    lengths = [256, 512, 1024, 2048, 4096]

    for n in lengths:
        if time.time() >= deadline - 5.0:
            break
        print(f"\n=== Length n = {n} ===")

        # ----- 1a. Random scan of the (p, eps) family -----------------
        local_best_a = None
        local_best_sc = float("inf")

        # deterministic starters
        starters = [np.full(n, 1.0 / n), arcsine_initial(n)]
        for starter in starters:
            sc = compute_score(starter)
            if sc < local_best_sc:
                local_best_sc = sc
                local_best_a = starter.copy()

        # random samples (budget‑aware)
        max_samples = 800
        for _ in range(max_samples):
            if time.time() >= deadline - 5.0:
                break
            p = rng.uniform(0.1, 1.2)                     # broadened range
            eps = rng.uniform(1e-3, max(0.5, 0.02 * n))
            cand = generate_shape(n, p, eps)
            sc = compute_score(cand)
            if sc < local_best_sc:
                local_best_sc = sc
                local_best_a = cand.copy()

        print(f"  Random scan best score = {local_best_sc:.6f}")

        # ----- 1b. Sub‑gradient refinement ----------------------------
        if local_best_a is not None:
            # allocate a fraction of remaining time to this length
            remaining = deadline - time.time()
            sub_time = min(remaining * 0.4, 30.0)          # at most ~30 s per length
            sub_deadline = time.time() + sub_time
            refined_a, refined_sc = subgradient_refine(
                initial_a=local_best_a,
                time_deadline=sub_deadline,
                max_iter=200_000,
                eta0=0.7,
                noise_every=800,
                noise_scale=5e-5,
            )
            print(f"  After sub‑gradient score = {refined_sc:.6f}")
            local_best_a = refined_a
            local_best_sc = refined_sc

            # ----- 1c. Optional LP boost (only for moderate n) ----------
            if n <= 2048:
                lp_deadline = deadline - 5.0
                local_best_a, local_best_sc = lp_boost(
                    local_best_a, lp_deadline, max_lp_iters=5
                )
                print(f"  After LP boost score = {local_best_sc:.6f}")

        # ----- 1d. Evaluate with the official evaluator ----------------
        if local_best_a is not None:
            val = evaluate_sequence(local_best_a.tolist())
            print(f"  Official evaluation score = {val:.6f}")
            if val < best_val:
                best_val = val
                best_seq = local_best_a.tolist()
                print(f"  *** New global best! n={n}, score={best_val:.6f} ***")

    # ------------------------------------------------------------------
    # 2.  Final polishing on the current global best (time permitting)
    # ------------------------------------------------------------------
    if best_seq and time.time() < deadline - 5.0:
        print("\nFinal polishing of global best")
        a = np.array(best_seq, dtype=float)

        # sub‑gradient polishing
        sub_deadline = time.time() + min(40.0, deadline - time.time() - 5.0)
        a, _ = subgradient_refine(
            initial_a=a,
            time_deadline=sub_deadline,
            max_iter=300_000,
            eta0=0.6,
            noise_every=600,
            noise_scale=3e-5,
        )

        # one more LP boost if size permits
        if len(a) <= 2048:
            a, _ = lp_boost(a, time_deadline=deadline - 5.0, max_lp_iters=3)

        final_val = evaluate_sequence(a.tolist())
        print(f"Final polished score = {final_val:.6f}")
        if final_val < best_val:
            best_val = final_val
            best_seq = a.tolist()
            print("*** Updated global best after final polish ***")

    # ------------------------------------------------------------------
    # 3.  Fallback guard – ensure we always return *something*
    # ------------------------------------------------------------------
    if not best_seq:
        best_seq = [1.0] * 256
        best_val = evaluate_sequence(best_seq)

    print(f"\nFinished search. Best reported score = {best_val:.6f}")
    return best_seq
