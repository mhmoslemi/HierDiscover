"""
Propose a step‐function height sequence that minimises

    2·n·max(conv(a,a)) / (sum(a)^2)

where conv is the full linear convolution.
The algorithm runs a large‐scale projected Adam optimisation on the
smooth approximation of the L∞‑norm (softmax of the convolution
entries).  An exponential decay on the softmax temperature is used so
that the optimisation gradually focuses on the true maximum.
The iterate is always projected onto the probability simplex, which
ensures non‑negativity and a fixed sum (the objective is scale‑invariant).
Multiple random initialisations are tried (including the previous best
if available); the best sequence found before the time budget expires
is returned.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities
# ----------------------------------------------------------------------


def simplex_projection(v):
    """Euclidean projection of a vector onto the probability simplex."""
    v = np.asarray(v, dtype=float)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1] - 1
    theta = cssv[cond][-1] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def next_pow2(x):
    """Return the smallest power of two >= x (x>0)."""
    return 1 << (x - 1).bit_length()


def compute_objective(a):
    """Exact objective 2·n·max(conv(a,a)) for sum(a)=1."""
    n = a.shape[0]
    # use FFT for speed
    m = next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, m)
    conv = np.fft.irfft(fa * fa, n=m)[: 2 * n - 1]
    return 2.0 * n * conv.max()


# ----------------------------------------------------------------------
# Core optimisation routine
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 0, budget_s: int = 1000):
    """
    Search for a non‑negative height sequence with minimal evaluation score.
    The function respects the 1000 s budget and returns the best sequence
    found as a plain Python list of floats.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 5.0  # safety margin

    # --------------------------------------------------------------
    # Determine problem size
    # --------------------------------------------------------------
    try:
        n = len(height_sequence_1)  # type: ignore[name-defined]
        previous = np.array(height_sequence_1, dtype=float)
        previous = previous / previous.sum()
    except NameError:
        n = 5146
        previous = None

    # --------------------------------------------------------------
    # Initial candidates
    # --------------------------------------------------------------
    init_candidates = []

    # 1) previous best, if any
    if previous is not None:
        init_candidates.append(previous)

    # 2) uniform distribution
    init_candidates.append(np.full(n, 1.0 / n))

    # 3) linear decreasing ramp (mirrored to keep sum=1)
    ramp = np.arange(n, 0, -1, dtype=float)
    ramp = ramp / ramp.sum()
    init_candidates.append(ramp)

    # 4) symmetric triangular shape (peak in the centre)
    tri = np.concatenate([np.arange(1, n // 2 + 1), np.arange(n // 2, 0, -1)])
    tri = tri / tri.sum()
    init_candidates.append(tri)

    # 5) a handful of random Dirichlet‑like seeds
    for _ in range(5):
        conc = np.random.uniform(0.1, 3.0, size=n)
        a = np.random.gamma(conc, 1.0)
        a = a / a.sum()
        init_candidates.append(a)

    # --------------------------------------------------------------
    # Global bookkeeping
    # --------------------------------------------------------------
    best_seq = None
    best_score = np.inf

    # pre‑compute FFT length (covers both a*a and p*a convolutions)
    M = next_pow2(3 * n - 2)  # >= len(p)+len(a)-1

    # optimisation hyper‑parameters
    beta1, beta2 = 0.9, 0.999
    eps_adam = 1e-8
    base_lr = 0.5          # base learning rate for Adam
    epsilon_decay = 0.9995  # decay of the softmax temperature
    min_eps_factor = 1e-4   # floor for the temperature relative to max_conv
    active_tol = 1e-12      # tolerance when falling back to the active set
    print_interval = 5000   # progress reports

    # --------------------------------------------------------------
    # Main optimisation loop over different initialisations
    # --------------------------------------------------------------
    for init_id, a0 in enumerate(init_candidates):
        if time.time() > deadline:
            break
        a = a0.copy()
        m = np.zeros_like(a)
        v = np.zeros_like(a)
        t = 0
        eps_factor = 0.5  # start temperature as 0.5·max_conv

        while time.time() < deadline:
            t += 1

            # ----- convolution a*a (full) -----
            fa = np.fft.rfft(a, M)
            conv_full = np.fft.irfft(fa * fa, n=M)[: 2 * n - 1]
            max_conv = conv_full.max()
            obj = 2.0 * n * max_conv

            # ----- keep best -----
            if obj < best_score:

            epsilon = max_conv * eps_factor

            # ----- compute soft‑max probabilities over the convolution entries -----
            if eps_factor > min_eps_factor:

            g = 2.0 * grad_full[:n]

            # ----- Adam update -----
            m = beta1 * m + (1.0 - beta1) * g
            v = beta2 * v + (1.0 - beta2) * (g * g)
            m_hat = m / (1.0 - beta1 ** t)
            v_hat = v / (1.0 - beta2 ** t)
            step = base_lr * m_hat / (np.sqrt(v_hat) + eps_adam)
            a = a - step

            # ----- projection onto the simplex -----
            a = simplex_projection(a)

            # occasional jitter to escape shallow plateaus
            if t % (print_interval * 2) == 0:
                jitter = np.random.normal(scale=0.001, size=n)
                a = np.maximum(a + jitter, 0.0)
                a = simplex_projection(a)

            # progress report
            if t % print_interval == 0:
                print(
                    f"[init {init_id}] iter {t:6d}: obj={obj:.6f}, eps_factor={eps_factor:.6g}"
                )

            # a hard iteration limit to keep each restart bounded
            if t >= 200_000:
                break

    # --------------------------------------------------------------
    # Return the best sequence found
    # --------------------------------------------------------------
    if best_seq is None:
        # should never happen, but fall back to uniform if it does
        best_seq = np.full(n, 1.0 / n)

    print(f"Finished search. Best objective = {best_score:.6f}")
    return [float(v) for v in best_seq.tolist()]
