def weighted_correlation(a, w, fft_len):
    # a_rev = a[::-1]; compute convolution w * a_rev via FFT.
    # Compute FFT of w and a_rev (maybe precompute rev).
    Fw = np.fft.rfft(w, n=fft_len)
    Fa_rev = np.fft.rfft(a[::-1], n=fft_len)
    conv = np.fft.irfft(Fw * Fa_rev, n=fft_len)[:len(w) + len(a) - 1]  # length 3n-2
    # extract relevant slice.
    start = len(a) - 1
    grad = 2.0 * conv[start:start+len(a)]
    return grad

def propose_candidate(seed=42, budget_s=1000, **kwargs):
    import time, numpy as np
    np.random.seed(seed)
    start = time.time()
    deadline = start + budget_s - 0.2

    # Choose n based on time maybe; we can try n=3000.
    n = 3000 # maybe 3000 for better shape.
    target_sum = np.sqrt(2.0 * n)

    # Precompute FFT length.
    fft_len = 1 << ((2*n - 1).bit_length())

    # Define helper functions inside? No closures allowed? Actually we can define functions at top-level. So not needed.

    # 1. Build seeds: geometric, power-law, uniform, random, and maybe previous best.

    seeds = []

    # Geometric (decreasing)
    lambdas = np.logspace(-6, -1, 40)   # adapt to new n
    for lam in lambdas:
        r = 1.0 - lam
        denom = 1.0 - r**n
        if denom <= 0: continue
        C = target_sum * lam / denom
        a = C * (r ** np.arange(n))
        seeds.append(a)

    # Power-law
    for p in np.linspace(0.2, 1.0, 30):
        base = 1.0 / np.power(np.arange(1, n+1), p)
        C = target_sum / base.sum()
        a = C * base
        seeds.append(a)

    # Uniform
    seeds.append(np.full(n, target_sum / n))

    # Random Dirichlet seeds: sample random vector via np.random.rand and project onto simplex.
    for _ in range(20):
        rand = np.random.rand(n)
        a = projection_simplex(rand, target_sum)
        seeds.append(a)

    # Add previous best if exists
    try:
        global height_sequence_1
        if isinstance(height_sequence_1, (list, np.ndarray)):
            hs = np.asarray(height_sequence_1, dtype=np.float64)
            if hs.shape[0] == n:
                hs = np.clip(hs, 0.0, 1000.0)
                hs = hs * (target_sum / hs.sum())
                seeds.append(hs)
    except Exception:
        pass

    best_seq = None
    best_score = float('inf')
    # Evaluate seeds with fast score
    for idx, cand in enumerate(seeds):
        if time.time() >= deadline:
            break
        cand = projection_simplex(cand, target_sum)
        sc = _fast_score(cand)  # using defined earlier
        if sc < best_score:
            best_score, best_seq = sc, cand.copy()
            print(f"[{time.time() - start:6.1f}s] seed {idx:02d} score {sc:.6f}")

    # Now perform iterative optimization on best_seq using smooth gradient descent.
    curr = best_seq.copy()
    best_score = _fast_score(curr)
    best = curr.copy()
    # parameters for smooth method
    alpha = 5.0    # start low
    alpha_max = 200.0
    alpha_inc = 1.02   # multiply each iteration
    step = target_sum / (5.0 * n)  # initial step size
    min_step = 1e-8
    # track iteration
    it = 0
    while time.time() < deadline:
        # compute conv
        F = np.fft.rfft(curr, n=fft_len)
        conv = np.fft.irfft(F * F, n=fft_len)[: 2*n - 1]
        max_c = conv.max()

        # compute smooth objective
        # stable exp
        shifted = conv - max_c  # subtract max to avoid overflow
        exp_vals = np.exp(alpha * shifted)
        sum_exp = exp_vals.sum()
        smooth_obj = (np.log(sum_exp) / alpha) + max_c  # because we shifted: log sum exp(alpha*(c-max)) / alpha + max

        # compute softmax weights
        w = exp_vals / sum_exp  # length 2n-1, sums to 1

        # compute gradient = 2 * correlation(w, curr)
        # compute via FFT using helper
        grad = weighted_correlation(curr, w, fft_len)

        gnorm = np.linalg.norm(grad)
        if gnorm == 0:
            break
        # step size adapt
        # Use simple backtracking
        accepted = False
        step_try = step
        for _ in range(8):
            new = curr - step_try * (grad / gnorm)
            new = np.clip(new, 0.0, 1000.0)
            new = projection_simplex(new, target_sum)
            # compute new smooth objective quickly
            Fnew = np.fft.rfft(new, n=fft_len)
            conv_new = np.fft.irfft(Fnew * Fnew, n=fft_len)[:2*n-1]
            max_c_new = conv_new.max()
            shifted_new = conv_new - max_c_new
            exp_new = np.exp(alpha * shifted_new)
            sum_exp_new = exp_new.sum()
            smooth_new = (np.log(sum_exp_new) / alpha) + max_c_new

            # Use sparse.diags to add contributions.
            data = []
            rows = []
            cols = []
            for i in range(n):

                row_idx = np.arange(i, i + n)

                rows.append(row_idx)
                cols.append(col_idx)

            rows = np.concatenate(rows)
            cols = np.concatenate(cols)
            data = np.concatenate(data)
            A = csr_matrix((data, (rows, cols)), shape=(2*n-1, n))
            b = np.full(2*n-1, M)
            res = linprog(c=-np.ones(n), A_ub=A, b_ub=b, bounds=[(0, None)]*n, method='highs')
            if not res.success:

            sum_g = g.sum()
            if sum_g == 0:

            # accept update if not worsen M
            if Mnew <= M * 0.999:
                f = f_new
            else:
                break

    final_score = evaluate_sequence(best.tolist())
    print(f"Search finished. Final score reported by evaluate_sequence: {final_score:.6f}")
    return best.tolist()

"""
A two‑phase optimizer for the “max self‑convolution” problem.

Phase 1 – Global seeding + smooth (log‑sum‑exp) gradient descent.
          The non‑smooth max is replaced by  fα(a)= (1/α)·log Σexp(α·c_k)
          where c = a∗a.  Its gradient can be written as a weighted
          correlation, which is evaluated efficiently by FFT.  The
          smoothing parameter α is gradually increased so that the
          algorithm initially moves easily and later focuses on the true
          max.

Phase 2 – A short LP polishing step (exactly as in the reference
          implementation).  If SciPy is not available the LP phase is
          skipped.

Both phases respect the security checks of ``evaluate_sequence``:
non‑negative entries, clamped to [0, 1000] and projected onto the
ℓ₁‑simplex {x≥0, Σx = √(2 n)}.
The routine respects a 1000 s time budget and prints progress messages.
"""

import time
import numpy as np

# --------------------------------------------------------------------------- #
def projection_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Project a vector onto the simplex {x≥0, Σx = s} (Duchi et al. 2008)."""
    if s < 0:
        raise ValueError("simplex sum must be non‑negative")
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho_arr = np.where(u - cssv / (np.arange(n) + 1.0) > 0)[0]
    if rho_arr.size == 0:
        return np.zeros_like(v)
    rho = rho_arr[-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# --------------------------------------------------------------------------- #
def fast_score(seq: np.ndarray) -> float:
    """
    Return the exact value of ``evaluate_sequence`` for a sequence that is
    already ℓ₁‑normalised (Σ = √(2 n)).  The routine uses a power‑of‑two FFT.
    """
    n = seq.size
    fft_len = 1 << ((2 * n - 1).bit_length())
    F = np.fft.rfft(seq, n=fft_len)
    conv = np.fft.irfft(F * F, n=fft_len)[: 2 * n - 1]
    max_conv = conv.max()
    # With the normalisation Σ = √(2 n) the score equals max_conv.
    return float(max_conv)


# --------------------------------------------------------------------------- #
def weighted_correlation(a: np.ndarray, w: np.ndarray, fft_len_corr: int) -> np.ndarray:
    """
    Compute g_i = 2 * Σ_k w_k * a_{k-i}   (correlation of w with a)
    using FFT.  ``a`` has length n, ``w`` length 2n‑1.  The result has length n.
    """
    a_rev = a[::-1]
    Fw = np.fft.rfft(w, n=fft_len_corr)
    Fa = np.fft.rfft(a_rev, n=fft_len_corr)
    conv_full = np.fft.irfft(Fw * Fa, n=fft_len_corr)
    start = a.shape[0] - 1               # index of k = 0 in the full convolution
    grad = 2.0 * conv_full[start:start + a.shape[0]]
    return grad


# --------------------------------------------------------------------------- #
def propose_candidate(seed: int = 42, budget_s: float = 1000, **kwargs):
    """
    Search for a non‑negative step‑function that minimises
    ``evaluate_sequence``.  The method combines a rich seed set,
    a smooth‑gradient descent with an increasing temperature α and a
    final LP polishing phase.
    """
    np.random.seed(seed)
    start = time.time()
    deadline = start + budget_s - 0.2   # safety margin

    # ------------------------------------------------------------------- #
    # problem size – 2500 gives a good trade‑off between flexibility and
    # memory consumption for the LP phase.
    n = 2500
    target_sum = np.sqrt(2.0 * n)       # Σ a_i = √(2 n)

    # FFT lengths
    fft_len_conv = 1 << ((2 * n - 1).bit_length())   # for a*a
    fft_len_corr = 1 << ((3 * n - 2).bit_length())   # for correlation w↔a

    # ------------------------------------------------------------------- #
    # 1. generate a diverse collection of seeds
    seeds = []

    # geometric (decreasing) families
    for lam in np.logspace(-5, -1, 40):
        r = 1.0 - lam
        denom = 1.0 - r ** n
        if denom <= 0.0:
            continue
        C = target_sum * lam / denom
        a = C * (r ** np.arange(n))
        seeds.append(a)

    # power‑law families
    for p in np.linspace(0.2, 1.0, 30):
        base = 1.0 / np.power(np.arange(1, n + 1, dtype=np.float64), p)
        C = target_sum / base.sum()
        a = C * base
        seeds.append(a)

    # uniform baseline
    seeds.append(np.full(n, target_sum / n))

    # random Dirichlet‑like seeds
    for _ in range(20):
        rand = np.random.rand(n)
        a = projection_simplex(rand, target_sum)
        seeds.append(a)

    # include the previously discovered construction if it matches the size
    try:
        global height_sequence_1
        if isinstance(height_sequence_1, (list, tuple, np.ndarray)):
            hs = np.asarray(height_sequence_1, dtype=np.float64)
            if hs.shape[0] == n:
                hs = np.clip(hs, 0.0, 1000.0)
                hs = hs * (target_sum / hs.sum())
                seeds.append(hs)
    except Exception:
        pass

    # ------------------------------------------------------------------- #
    # 2. keep the best seed (exact ℓ₁‑normalisation)
    best_seq = None
    best_score = float("inf")
    for idx, cand in enumerate(seeds):
        if time.time() >= deadline:
            break
        cand = projection_simplex(cand, target_sum)
        sc = fast_score(cand)
        if sc < best_score:
            best_score, best_seq = sc, cand.copy()
            print(f"[{time.time() - start:6.1f}s] seed {idx:02d} → score {sc:.6f}")

    # ------------------------------------------------------------------- #
    # 3. Smooth (log‑sum‑exp) projected gradient descent
    curr = best_seq.copy()
    best = curr.copy()
    best_score = fast_score(best)

    # smoothing parameter α : start low, increase slowly
    alpha = 5.0
    alpha_max = 200.0
    alpha_mult = 1.02

    # step size – absolute movement (will be scaled by the unit gradient)
    step = 1.0                # initial guess, will be adapted
    min_step = 1e-8

    it = 0
    while time.time() < deadline:
        # --- compute convolution and smooth objective ---------------------------------
        F = np.fft.rfft(curr, n=fft_len_conv)
        conv = np.fft.irfft(F * F, n=fft_len_conv)[: 2 * n - 1]
        max_c = conv.max()

        # log‑sum‑exp (stable)
        shift = conv - max_c               # prevent overflow
        exp_shift = np.exp(alpha * shift)
        sum_exp = exp_shift.sum()
        smooth_obj = max_c + np.log(sum_exp) / alpha

        # softmax weights for the gradient
        w = exp_shift / sum_exp            # Σ w_k = 1, length 2n‑1

        # --- gradient (weighted correlation) ---------------------------------------
        grad = weighted_correlation(curr, w, fft_len_corr)

        gnorm = np.linalg.norm(grad)
        if gnorm == 0.0:
            break

        # --- back‑tracking line search ------------------------------------------------
        accepted = False
        trial_step = step
        for _ in range(8):
            cand = curr - trial_step * (grad / gnorm)
            cand = np.clip(cand, 0.0, 1000.0)
            cand = projection_simplex(cand, target_sum)

            # evaluate smooth objective at candidate
            Fc = np.fft.rfft(cand, n=fft_len_conv)
            convc = np.fft.irfft(Fc * Fc, n=fft_len_conv)[: 2 * n - 1]
            max_c_cand = convc.max()
            shift_c = convc - max_c_cand
            exp_c = np.exp(alpha * shift_c)
            smooth_c = max_c_cand + np.log(exp_c.sum()) / alpha

            if smooth_c < smooth_obj - 1e-14:
                curr = cand
                smooth_obj = smooth_c
                accepted = True
                break
            trial_step *= 0.5

        if not accepted:
            step = max(step * 0.5, min_step)
        else:
            step = min(step * 1.05, target_sum)   # allow a mild increase

        # --- keep the best true score every few iterations ----------------------------
        if it % 250 == 0:
            true_score = fast_score(curr)         # with Σ normalised this = max conv
            if true_score < best_score - 1e-10:
                best_score, best = true_score, curr.copy()
                print(f"[{time.time() - start:6.1f}s] iter {it:07d} → new best {true_score:.6f}")

        # --- slowly increase the smoothing parameter ---------------------------------
        if alpha < alpha_max:
            alpha = min(alpha * alpha_mult, alpha_max)

        # --- occasional random jitter to escape plateaus ------------------------------
        if np.random.rand() < 0.012:
            jitter = np.random.randn(n) * step * 0.3
            curr = projection_simplex(np.clip(curr + jitter, 0.0, 1000.0), target_sum)

        it += 1

    # ------------------------------------------------------------------- #
    # 4. Linear‑programming polishing (if SciPy is available)
    try:
        from scipy.optimize import linprog
        from scipy.sparse import csr_matrix
    except Exception:                # SciPy not present – skip LP phase
        linprog = None

    if linprog is not None and time.time() < deadline - 5.0:
        f = best.copy()
        for lp_iter in range(12):
            if time.time() >= deadline - 5.0:
                break
            # current maximal convolution value
            F = np.fft.rfft(f, n=fft_len_conv)
            conv_f = np.fft.irfft(F * F, n=fft_len_conv)[: 2 * n - 1]
            M = conv_f.max()

            # build Toeplitz matrix A such that A·g = f * g
            rows = np.empty(n * n, dtype=np.int32)
            cols = np.empty(n * n, dtype=np.int32)
            data = np.empty(n * n, dtype=np.float64)

            ptr = 0
            for i in range(n):
                rows[ptr:ptr + n] = np.arange(i, i + n)
                cols[ptr:ptr + n] = i
                data[ptr:ptr + n] = f
                ptr += n

            A = csr_matrix((data, (rows, cols)), shape=(2 * n - 1, n))
            b = np.full(2 * n - 1, M)

            # maximise Σ g_i  →  minimise -Σ g_i
            res = linprog(
                c=-np.ones(n),
                A_ub=A,
                b_ub=b,
                bounds=[(0, None)] * n,
                method='highs',
                options={'presolve': True, 'time_limit': 30}
            )
            if not res.success:
                break

            g = res.x
            sum_g = g.sum()
            if sum_g < 1e-12:
                break
            g = g * (target_sum / sum_g)

            # small mixing with the current vector
            t_mix = 0.12
            f_new = (1.0 - t_mix) * f + t_mix * g
            f_new = projection_simplex(f_new, target_sum)

            # evaluate new max‑convolution
            Fnew = np.fft.rfft(f_new, n=fft_len_conv)
            conv_new = np.fft.irfft(Fnew * Fnew, n=fft_len_conv)[: 2 * n - 1]
            Mnew = conv_new.max()
            score_new = float(Mnew)       # Σ is normalised → score = max

            if score_new < best_score - 1e-9:
                best_score, best = score_new, f_new.copy()
                print(f"[{time.time() - start:6.1f}s] LP‑iter {lp_iter} → new best {score_new:.6f}")

            # accept the LP update if it does not increase the bound
            if Mnew <= M * 0.999:
                f = f_new
            else:
                break

    # ------------------------------------------------------------------- #
    # 5. Final sanity check with the official evaluator.
    final_score = evaluate_sequence(best.tolist())
    print(f"Search finished. Final score reported by evaluate_sequence: {final_score:.6f}")

    return best.tolist()
