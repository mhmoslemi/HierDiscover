"""
Propose a step‑function height sequence that minimises the provided
evaluation function.

Algorithm
---------
1. Determine the problem size `n`.  If a global `height_sequence_1`
   exists we use its length, otherwise we fall back to 5146 (the length
   used in the previous best solution).

2. Build a quick portfolio of deterministic candidates:
   * a linear decreasing ramp,
   * the uniform distribution,
   * a modest grid of Beta‑distribution shapes,
   * a handful of random Dirichlet‑like vectors.
   All candidates are normalised to sum = 1 and the one with the
   smallest value of `2·n·max(a*a)` is kept as the initial point.

3. Starting from this seed we run a long projected gradient descent
   on the probability simplex.  The gradient is **not** a raw
   sub‑gradient of the max‑convolution; instead we smooth the
   max‑operator with a log‑sum‑exp (temperature `τ`) and use the
   resulting (smooth) gradient:
        g = 2·Z·a ,   Z_{i,j}=w[i+j] ,   w ∝ exp(conv/τ) , Σw=1 .
   The temperature is slowly annealed (geometric decay) so that the
   optimisation first follows a smooth landscape and later focuses on
   the true maximum.

   The update is an additive step followed by the Euclidean projection
   onto the probability simplex (Duchi et al., 2008).  The step size
   follows a classic diminishing schedule `η/√(t+1)`.

4. After the main loop we perform a short random‑perturbation phase:
   the current best vector is mixed with a few random Dirichlet draws
   and a few hundred more gradient steps are taken from each perturbed
   start, keeping any improvement.

5. The best sequence observed during the whole run is returned as a
   plain Python list of floats.  All numbers are non‑negative and sum to
   one, so `evaluate_sequence` will accept them.

The implementation relies only on NumPy and pure Python and respects
the 1000 s time budget.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper: Euclidean projection onto the probability simplex
# ----------------------------------------------------------------------
def simplex_projection(v):
    """Project a vector onto the probability simplex {w| w_i≥0, Σw_i=1}."""
    v = np.asarray(v, dtype=float)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho = np.nonzero(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:                     # all non‑positive → uniform
        return np.full_like(v, 1.0 / n)
    rho = rho[-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Fast linear convolution via FFT (real input → real output)
# ----------------------------------------------------------------------
def full_convolution(x, y):
    """Linear convolution of two real vectors, full length."""
    n = len(x) + len(y) - 1
    n_fft = 1 << (n - 1).bit_length()        # next power of two
    X = np.fft.rfft(x, n=n_fft)
    Y = np.fft.rfft(y, n=n_fft)
    conv = np.fft.irfft(X * Y, n=n_fft)
    return conv[:n]


# ----------------------------------------------------------------------
# Gradient w.r.t. a of the smoothed max of a*a
# ----------------------------------------------------------------------
def gradient_from_weights(w_norm, a):
    """
    Given the normalised weight vector w_norm (length 2n‑1) and the current
    step heights a (length n), compute 2·Z·a where Z_{i,j}=w_norm[i+j].
    The result has length n.
    """
    n = len(a)
    a_rev = a[::-1]
    conv_len = len(w_norm) + n - 1            # = 3n‑2
    n_fft = 1 << (conv_len - 1).bit_length()
    W = np.fft.rfft(w_norm, n=n_fft)
    A_rev = np.fft.rfft(a_rev, n=n_fft)
    full = np.fft.irfft(W * A_rev, n=n_fft)
    start = n - 1                            # slice start index
    grad = 2.0 * full[start:start + n]
    return grad


# ----------------------------------------------------------------------
# Objective for a normalised vector (sum == 1)
# ----------------------------------------------------------------------
def compute_objective(a):
    """Return 2·n·max(a*a) for a vector a with sum = 1."""
    n = a.shape[0]
    conv = full_convolution(a, a)            # length 2n‑1
    return 2.0 * n * conv.max()


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: int = 1000):
    """
    Search for a non‑negative sequence of heights that minimises the
    given evaluation function.  The function respects the 1000 s time
    budget and returns the best sequence found.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 5.0    # a small safety margin

    # --------------------------------------------------------------
    # 1. Problem size
    # --------------------------------------------------------------
    try:
        n = len(height_sequence_1)           # type: ignore[name-defined]
    except Exception:
        n = 5146

    # --------------------------------------------------------------
    # 2. Build deterministic starters
    # --------------------------------------------------------------
    best_a = None
    best_score = np.inf

    # Linear decreasing ramp
    a_ramp = np.arange(n, 0, -1, dtype=float)
    a_ramp = a_ramp * 2.0 / (n * (n + 1))
    a_ramp /= a_ramp.sum()
    score = compute_objective(a_ramp)
    best_a, best_score = a_ramp.copy(), score
    print(f"Initial linear ramp:    score = {score:.6f}")

    # Uniform distribution
    a_uniform = np.full(n, 1.0 / n)
    score = compute_objective(a_uniform)
    if score < best_score:
        best_a, best_score = a_uniform.copy(), score
        print(f"Uniform improved to:   score = {score:.6f}")

    # Helper to generate a Beta‑shaped vector
    def beta_sequence(alpha, beta, size):
        x = (np.arange(size) + 0.5) / size
        logw = (alpha - 1.0) * np.log(x) + (beta - 1.0) * np.log(1.0 - x)
        logw = np.clip(logw, -700.0, 700.0)   # protect against overflow
        w = np.exp(logw)
        if w.sum() == 0 or not np.isfinite(w.sum()):
            w = np.ones_like(w)
        return w / w.sum()

    # Grid of Beta parameters (moderate size)
    alphas = np.linspace(0.2, 5.0, 12)
    betas  = np.linspace(0.2, 5.0, 12)

    for alpha in alphas:
        for beta in betas:
            if time.time() > deadline:
                break
            a = beta_sequence(alpha, beta, n)
            s = compute_objective(a)
            if s < best_score:
                best_a, best_score = a.copy(), s
                print(f"Beta({alpha:.2f},{beta:.2f}) => score {s:.6f}")
        if time.time() > deadline:
            break

    # Random Beta draws for extra coverage
    for _ in range(200):
        if time.time() > deadline:
            break
        a = beta_sequence(np.random.uniform(0.05, 5.0), np.random.uniform(0.05, 5.0), n)
        s = compute_objective(a)
        if s < best_score:
            best_a, best_score = a.copy(), s
            print(f"Random Beta => score {s:.6f}")

    # Random Dirichlet‑like seeds
    for _ in range(15):
        if time.time() > deadline:
            break
        conc = np.random.uniform(0.1, 3.0, size=n)
        a = np.random.gamma(conc, 1.0)
        a /= a.sum()
        s = compute_objective(a)
        if s < best_score:
            best_a, best_score = a.copy(), s
            print(f"Random Dirichlet => score {s:.6f}")

    # If a global height_sequence_1 exists, use it as an additional seed
    try:
        cand = np.asarray(height_sequence_1, dtype=float)
        if cand.shape[0] == n and np.all(np.isfinite(cand)):
            cand = np.maximum(cand, 0.0)
            if cand.sum() > 0:
                cand /= cand.sum()
                s = compute_objective(cand)
                if s < best_score:
                    best_a, best_score = cand.copy(), s
                    print(f"height_sequence_1 => score {s:.6f}")
    except Exception:
        pass

    # --------------------------------------------------------------
    # 3. Smoothed projected gradient descent
    # --------------------------------------------------------------
    a = best_a.copy()
    iteration = 0
    eta = 0.80                               # base step‑size
    tau = None                               # temperature for smoothing

    while time.time() < deadline and iteration < 30000:
        # ----- convolution and current objective -----
        conv = full_convolution(a, a)                # length 2n‑1
        max_conv = conv.max()
        cur_score = 2.0 * n * max_conv

        # keep the best ever seen
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"Iter {iteration:5d}: new best {best_score:.6f}")

        # ----- temperature handling -----
        if tau is None:
            tau = max_conv * 0.20                    # start relatively smooth
        else:
            # geometric decay; never let it become smaller than a
            # tiny fraction of the current peak
            tau = max(tau * 0.9995, max_conv * 1e-5)

        # ----- smoothed max weights -----
        # w ∝ exp((conv - max_conv)/tau) , Σw = 1
        # using clipping to stay in the safe exponential range
        exp_arg = (conv - max_conv) / tau
        exp_arg = np.clip(exp_arg, -700, 0)          # avoid underflow/overflow
        w = np.exp(exp_arg)
        w_sum = w.sum()
        if w_sum == 0 or not np.isfinite(w_sum):
            w_norm = np.full_like(w, 1.0 / w.size)
        else:
            w_norm = w / w_sum

        # ----- gradient of the smoothed objective -----
        grad = gradient_from_weights(w_norm, a)      # shape (n,)

        # Optional clipping of the gradient to keep steps reasonable
        grad = np.clip(grad, -1e3, 1e3)

        # ----- diminishing step size -----
        step = eta / np.sqrt(iteration + 1.0)

        # ----- projected additive update -----
        a = simplex_projection(a - step * grad)

        iteration += 1

    # --------------------------------------------------------------
    # 4. Local random perturbations + short refinements
    # --------------------------------------------------------------
    # use whatever time is still left
    while time.time() < deadline:
        # mix the current best with a fresh Dirichlet draw
        eps = np.random.uniform(0.005, 0.05)          # mixing ratio
        noise = np.random.gamma(1.0, 1.0, size=n)
        noise /= noise.sum()
        a_pert = (1.0 - eps) * best_a + eps * noise
        a_pert /= a_pert.sum()

        # a few gradient steps from this perturbed start
        sub_iter = 0
        max_sub = 800
        while time.time() < deadline and sub_iter < max_sub:
            conv = full_convolution(a_pert, a_pert)
            max_conv = conv.max()
            cur_score = 2.0 * n * max_conv

            if cur_score < best_score:
                best_score = cur_score
                best_a = a_pert.copy()
                print(f"Perturb iter {iteration+sub_iter:5d}: new best {best_score:.6f}")

            # same smoothing machinery as before
            if tau is None:
                tau = max_conv * 0.20
            else:
                tau = max(tau * 0.9995, max_conv * 1e-5)

            exp_arg = (conv - max_conv) / tau
            exp_arg = np.clip(exp_arg, -700, 0)
            w = np.exp(exp_arg)
            w_sum = w.sum()
            w_norm = w / w_sum if w_sum > 0 else np.full_like(w, 1.0 / w.size)

            grad = gradient_from_weights(w_norm, a_pert)
            grad = np.clip(grad, -1e3, 1e3)

            step = eta / np.sqrt(iteration + sub_iter + 1.0)
            a_pert = simplex_projection(a_pert - step * grad)

            sub_iter += 1

        iteration += sub_iter
        # stop if we already exhausted the time budget
        if time.time() + 0.5 >= deadline:
            break

    # --------------------------------------------------------------
    # 5. Return the best sequence found
    # --------------------------------------------------------------
    print(f"Finished after {iteration} iterations, best score = {best_score:.6f}")
    return [float(v) for v in best_a.tolist()]
