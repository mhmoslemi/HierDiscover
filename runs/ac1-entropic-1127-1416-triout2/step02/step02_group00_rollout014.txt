"""
Hybrid random‐shape + projected momentum sub‑gradient search.

1.  Choose a few candidate lengths (default 512…4096).  
2.  For each length sample many two‑sided power‑law shapes
    a_i ∝ (i+ε₁)^(-p₁)*(n‑1‑i+ε₂)^(-p₂)   (p₁,p₂>0, ε>0).  
    The family contains the classic symmetric arcsine density,
    but also non‑symmetric variants that can beat it.
3.  Keep the best shape (according to the fast “compute_score”
    which is the same quantity as ``evaluate_sequence`` up to a
    constant factor) and refine it with a projected sub‑gradient
    optimiser.  A Nesterov‑style momentum term and occasional
    Gaussian noise help to escape plateaus.
4.  The best vector found over all lengths is polished once more
    with the optimiser until the time budget (≈1000 s) is exhausted.
   The final vector (list of floats) is returned.
"""

import time
import numpy as np
from typing import List, Tuple

# ----------------------------------------------------------------------
# Low‑level utilities (no closures)
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Linear convolution via real FFT (fast, zero‑padding to next power of two)."""
    m = len(a) + len(b) - 1
    n_fft = next_pow2(m)
    A = np.fft.rfft(a, n=n_fft)
    B = np.fft.rfft(b, n=n_fft)
    C = A * B
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


def max_convolution_and_index(a: np.ndarray) -> Tuple[float, int]:
    """
    Return the maximal entry of a*a and its index.
    """
    conv = convolution_fft(a, a)
    idx = int(np.argmax(conv))
    return float(conv[idx]), idx


def subgradient_of_max_at_index(a: np.ndarray, k: int) -> np.ndarray:
    """
    Sub‑gradient of the maximal convolution entry (index k) w.r.t. a.

    For each i, ∂c_k/∂a_i = a_{k‑i}   if 0 ≤ k‑i < n,
    and doubled diagonal term when k = 2·i.
    """
    n = len(a)
    grad = np.zeros_like(a)
    i = np.arange(n)
    j = k - i
    mask = (j >= 0) & (j < n)
    grad[mask] = a[j[mask]]
    if k % 2 == 0:
        diag = k // 2
        if diag < n:
            grad[diag] = 2.0 * a[diag]
    return grad


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the probability simplex
    {x | x_i ≥ 0, Σ_i x_i = 1}.
    Implementation of Duchi et al. (2008).
    """
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / len(v))

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho = np.where(u - cssv / np.arange(1, len(v) + 1) > 0)[0]
    if len(rho) == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def compute_score(a: np.ndarray) -> float:
    """
    Fast proxy for ``evaluate_sequence``.
    The input is assumed to be on the probability simplex (sum≈1).
    Returns   2·n·max(a*a) / (sum(a)²) .
    """
    n = len(a)
    conv = convolution_fft(a, a)
    max_val = float(np.max(conv))
    s = float(np.sum(a))
    if s < 1e-12:
        return float("inf")
    return 2.0 * n * max_val / (s * s)


def arcsine_initial(n: int) -> np.ndarray:
    """
    Classic arcsine‑shaped start:
        a_i ∝ 1 / √((i+½)·(n‑½‑i)).
    """
    i = np.arange(n, dtype=float)
    eps = 0.25
    vals = 1.0 / np.sqrt((i + 0.5) * (n - 0.5 - i) + eps)
    return vals / np.sum(vals)


def generate_shape_general(
    n: int,
    p_left: float,
    p_right: float,
    eps_left: float,
    eps_right: float,
) -> np.ndarray:
    """
    Two‑sided power‑law family:
        a_i ∝ (i+eps_left)^(-p_left)·(n‑1‑i+eps_right)^(-p_right) .
    The parameters are kept positive; the result is normalised to L¹‑norm 1.
    """
    i = np.arange(n, dtype=float)
    left = (i + eps_left) ** (-p_left)
    right = (n - 1 - i + eps_right) ** (-p_right)
    base = left * right
    total = base.sum()
    if total == 0.0 or not np.isfinite(total):
        return np.full(n, 1.0 / n)
    return base / total


def improve_sequence(
    initial_a: np.ndarray,
    time_deadline: float,
    max_iter: int = 200_000,
    eta0: float = 0.6,
    noise_every: int = 400,
    noise_scale: float = 1e-5,
    momentum: float = 0.9,
) -> Tuple[np.ndarray, float]:
    """
    Projected (Nesterov‑style) sub‑gradient descent for the max‑convolution objective.
    Returns the best vector seen and its internal score.
    """
    a = project_to_simplex(initial_a.astype(float).copy())
    a_prev = a.copy()
    best_a = a.copy()
    best_sc = compute_score(a)

    it = 0
    while time.time() < time_deadline and it < max_iter:
        it += 1
        # current worst convolution entry
        _, k_max = max_convolution_and_index(a)
        grad = subgradient_of_max_at_index(a, k_max)

        eta = eta0 / np.sqrt(it)          # classic diminishing step‑size

        # Nesterov‑style momentum
        a_new = a - eta * grad + momentum * (a - a_prev)

        if noise_every and (it % noise_every == 0):
            a_new += np.random.randn(len(a)) * noise_scale

        a_new = project_to_simplex(a_new)

        cur_sc = compute_score(a_new)
        if cur_sc < best_sc:
            best_sc = cur_sc
            best_a = a_new.copy()

        a_prev = a
        a = a_new

        # Very light progress output (can be silenced if desired)
        if it % 10000 == 0:
            print(
                f"    iter={it:,d}  cur={cur_sc:.6f}  best={best_sc:.6f}"
            )

    return best_a, best_sc
