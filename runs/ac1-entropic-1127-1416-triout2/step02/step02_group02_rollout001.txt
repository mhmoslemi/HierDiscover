def _softmax(z: np.ndarray) -> np.ndarray:
    """Numerically stable soft‑max."""
    z_max = np.max(z)
    e = np.exp(z - z_max)
    return e / e.sum()


def _conv_full(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full (linear) convolution using FFT, returns length len(a)+len(b)-1."""
    n = len(a) + len(b) - 1
    # rfft works with real input, we pad to the same length for both transforms
    fa = np.fft.rfft(a, n=n)
    fb = np.fft.rfft(b, n=n)
    return np.fft.irfft(fa * fb, n=n)

        w = exp_vals / exp_vals.sum()          # w_k ≥ 0, Σw = 1

        # ----- gradient of smooth objective w.r.t. a -----
        # ∂f/∂a_i = 2 Σ_k w_k a_{k-i}   (cross‑correlation)
        grad_a = 2.0 * _cross_corr(w, a)       # length n

        # ----- push gradient through the soft‑max (y‑space) -----
        avg = np.dot(a, grad_a)                # scalar <a,∇_a f>
        grad_y = a * (grad_a - avg)            # ∂f/∂y_i

        # ----- Adam update on y -----
        m = beta1 * m + (1.0 - beta1) * grad_y
        v = beta2 * v + (1.0 - beta2) * (grad_y ** 2)
        m_hat = m / (1.0 - beta1 ** iteration)
        v_hat = v / (1.0 - beta2 ** iteration)
        y -= lr * m_hat / (np.sqrt(v_hat) + eps_adam)

        # ----- occasional evaluation of the true objective -----
        if iteration % 200 == 0 or time.time() > deadline - 30:
            cur_score = _objective_from_a(a)
            if cur_score < best_score:
                best_score = cur_score
                best_a = a.copy()
                print(f"Iter {iteration:6d}: improved true score = {best_score:.6f}")

        # ----- anneal β (make the smooth max tighter) -----
        if iteration % 1000 == 0:
            beta = min(beta * beta_factor, beta_max)

        # ----- early stop if a very good value is already reached -----
        if best_score < 1.45:          # far below the previous best
            break

    # ------------------------------------------------------------------
    # 3. Return best sequence (fallback to uniform if nothing better)
    # ------------------------------------------------------------------
    if best_a is None:
        best_a = np.full(n, 1.0 / n)

    # Convert to plain Python list of floats (as required by the grader)
    return [float(v) for v in best_a.tolist()]
