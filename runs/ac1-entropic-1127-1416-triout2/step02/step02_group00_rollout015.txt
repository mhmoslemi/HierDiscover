"""
Hybrid LP‑+‑projected‑gradient optimiser for the step‑function bound.

The algorithm proceeds in three phases for each chosen length `n`:

1.  Initialise with a discretised arcsine density (the known continuous
    optimiser whose self‑convolution is uniform).

2.  Repeatedly solve a *linear programme*:
        maximise   Σ b_i
        subject to 0 ≤ b,
                    (a ★ b)_k ≤ M   for every k,
        where `a` is the current candidate and `M = max(a★a)`.
    The LP finds a vector `b` that can be made larger while keeping the
    current worst convolution value `M`.  After normalising `b` to sum 1,
    the new candidate `a ← b / Σb` enjoys a strictly smaller bound
    (`M` is divided by `(Σb)²`).  This step drives the sum‑distribution
    of `a` very close to uniform (the continuous optimum).

3.  A few hundred thousand iterations of projected sub‑gradient descent
    (the optimiser that was previously used) polish the LP‑output.
    Momentum is added via a simple Adam‑style schedule for faster
    convergence.

All steps respect the overall time budget (≈ 1000 s).  The best
candidate encountered across several lengths (256, 512, 1024, 2048)
is finally returned as a plain ``list`` of non‑negative floats.
"""

import time
import numpy as np
from typing import List, Tuple

# ----------------------------------------------------------------------
# Fast convolution utilities
# ----------------------------------------------------------------------
def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Linear convolution using a real FFT (fast and accurate)."""
    m = len(a) + len(b) - 1
    n_fft = next_pow2(m)
    A = np.fft.rfft(a, n=n_fft)
    B = np.fft.rfft(b, n=n_fft)
    C = A * B
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


def max_convolution_and_index(a: np.ndarray) -> Tuple[float, int]:
    """Return the maximum value of a★a and the index where it occurs."""
    conv = convolution_fft(a, a)
    idx = int(np.argmax(conv))
    return float(conv[idx]), idx


# ----------------------------------------------------------------------
# Simple projection onto the probability simplex (Σa_i = 1, a_i ≥ 0)
# ----------------------------------------------------------------------
def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x | x≥0, Σx = 1}."""
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / len(v))

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho = np.where(u - cssv / np.arange(1, len(v) + 1) > 0)[0]
    rho = rho[-1] if rho.size else 0
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Scaled objective used internally (scale‑invariant)
# ----------------------------------------------------------------------
def compute_score(a: np.ndarray) -> float:
    """Return 2·n·max(a★a) (the denominator Σa = 1 is already enforced)."""
    n = len(a)
    conv = convolution_fft(a, a)
    max_val = float(np.max(conv))
    return 2.0 * n * max_val


# ----------------------------------------------------------------------
# Classical arcsine start‑up (continuous optimal distribution)
# ----------------------------------------------------------------------
def arcsine_initial(n: int) -> np.ndarray:
    """Return a normalised discretisation of the arcsine density."""
    i = np.arange(n, dtype=float)
    # avoid exact zeros at the ends – a tiny epsilon works well
    eps = 0.5e-6
    vals = 1.0 / np.sqrt((i + 0.5) * (n - 0.5 - i) + eps)
    return vals / vals.sum()


# ----------------------------------------------------------------------
# Projected sub‑gradient optimiser (used for polishing)
# ----------------------------------------------------------------------
def subgradient_of_max_at_index(a: np.ndarray, k: int) -> np.ndarray:
    """Sub‑gradient of the maximal convolution entry (index k) w.r.t. a."""
    n = len(a)
    grad = np.zeros_like(a)
    i = np.arange(n)
    j = k - i
    mask = (j >= 0) & (j < n)
    grad[mask] = a[j[mask]]
    # double the diagonal term when it exists
    if k % 2 == 0:
        diag = k // 2
        if diag < n:
            grad[diag] = 2.0 * a[diag]
    return grad


def improve_sequence(initial_a: np.ndarray,
                    deadline: float,
                    max_iter: int = 200_000,
                    eta0: float = 0.5,
                    noise_every: int = 500,
                    noise_scale: float = 1e-4) -> Tuple[np.ndarray, float]:
    """
    Adam‑style projected sub‑gradient descent on the max‑convolution objective.
    Returns the best vector seen and its internal score.
    """
    a = project_to_simplex(initial_a.astype(float).copy())
    best_a = a.copy()
    best_sc = compute_score(a)

    n = len(a)
    m = 0
    v = np.zeros_like(a)  # first moment
    s = np.zeros_like(a)  # second moment
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8

    it = 0
    while time.time() < deadline and it < max_iter:
        it += 1
        # current worst convolution entry
        _, k_max = max_convolution_and_index(a)
        grad = subgradient_of_max_at_index(a, k_max)

        # Adam update
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad ** 2)
        m_hat = m / (1 - beta1 ** it)
        v_hat = v / (1 - beta2 ** it)

        step = eta0 / np.sqrt(it)
        a_new = a - step * m_hat / (np.sqrt(v_hat) + eps)

        if noise_every and (it % noise_every == 0):
            a_new += np.random.randn(n) * noise_scale

        a_new = project_to_simplex(a_new)

        cur_sc = compute_score(a_new)
        if cur_sc < best_sc:
            best_sc = cur_sc
            best_a = a_new.copy()

        a = a_new

        if it % 5_000 == 0:
            print(f"    subgrad iter={it:,d} cur={cur_sc:.6f} best={best_sc:.6f}")

        # early exit if already very good
        if best_sc < 1.503:
            break

    return best_a, best_sc


# ----------------------------------------------------------------------
# Linear‑programming utilities (LP that exploits the current bound)
# ----------------------------------------------------------------------
def convolution_pattern_indices(n: int) -> Tuple[np.ndarray, np.ndarray]:
    """
    Return the (row, col) index arrays for the convolution matrix
    C where C @ b = a ★ b .  For a fixed length n the pattern of
    indices never changes; only the data (the actual a‑values) do.
    """
    rows = np.empty(n * n, dtype=np.int32)
    cols = np.empty(n * n, dtype=np.int32)
    pos = 0
    for j in range(n):
        rows[pos:pos + n] = np.arange(j, j + n, dtype=np.int32)
        cols[pos:pos + n] = j
        pos += n
    return rows, cols


def solve_lp_max_sum(a: np.ndarray,
                    M: float,
                    row_idx: np.ndarray,
                    col_idx: np.ndarray) -> Tuple[np.ndarray, float]:
    """
    Solve   maximise Σ b_i   subject to   b_i ≥ 0,
                                           (a ★ b)_k ≤ M   (for all k).
    Returns the optimal b and its ℓ₁‑norm.  If the LP fails,
    ``(None, None)`` is returned.
    """
    from scipy import sparse
    from scipy.optimize import linprog

    n = len(a)
    # data for the sparse matrix – each column repeats the current a‑vector
    data = np.tile(a, n)
    A = sparse.csr_matrix((data, (row_idx, col_idx)),
                          shape=(2 * n - 1, n))

    # Scale the constraints so that the RHS becomes 1.
    epsilon = 1e-12
    A_scaled = A.multiply(1.0 / (M * (1.0 + epsilon)))
    b_ub = np.ones(2 * n - 1)           # (a★b) / M ≤ 1

    # linprog solves a *x ≤ b, we minimise –Σb_i  → maximise Σb_i
    c = -np.ones(n)

    res = linprog(c,
                  A_ub=A_scaled,
                  b_ub=b_ub,
                  bounds=[(0, None)] * n,
                  method='highs',
                  options={'presolve': True})
    if not res.success:
        return None, None
    b = res.x
    return b, b.sum()
