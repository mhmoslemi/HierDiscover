"""
Projected gradient descent on a smooth approximation of the max‑convolution
bound, with a corrected gradient, multi‑start at several lengths
(512‑8192), and a final stochastic hill‑climb.  The algorithm keeps the
sequence on the probability simplex (∑a_i = 1, a_i ≥ 0) so that the
evaluation function is scale‑invariant.  The gradient of the smooth‑max
objective is computed analytically (no diagonal correction needed) and
the step size follows a 1/√t schedule.  Various initialisations
(arcsine‑shape, uniform, sparse Dirichlet) are tried; the best candidate
found before the time budget expires is returned.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Basic utilities
# ----------------------------------------------------------------------


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of a vector onto the probability simplex."""
    if np.all(v <= 0):
        # fall back to uniform distribution
        return np.full_like(v, 1.0 / v.shape[0])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho = np.where(u - cssv / np.arange(1, v.shape[0] + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full convolution using a real FFT (fast)."""
    m = len(a) + len(b) - 1
    n_fft = next_pow2(m)
    A = np.fft.rfft(a, n=n_fft)
    B = np.fft.rfft(b, n=n_fft)
    C = A * B
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


def compute_score(a: np.ndarray) -> float:
    """Score identical to `evaluate_sequence` when ∑a = 1."""
    n = len(a)
    conv = convolution_fft(a, a)
    max_val = float(np.max(conv))
    s = float(np.sum(a))
    if s < 1e-12:
        return float("inf")
    return 2.0 * n * max_val / (s * s)


def arcsine_initial(n: int) -> np.ndarray:
    """Arcsine‑shaped start: a_i ∝ 1 / sqrt((i+0.5)(n‑0.5‑i))."""
    i = np.arange(n, dtype=float)
    eps = 0.25
    vals = 1.0 / np.sqrt((i + 0.5) * (n - 0.5 - i) + eps)
    return vals / np.sum(vals)


def uniform_initial(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n)


def sparse_dirichlet_initial(n: int, concentration: float = 0.3) -> np.ndarray:
    """Dirichlet with small concentration → sparse initial vectors."""
    vec = np.random.dirichlet(np.full(n, concentration))
    return vec / np.sum(vec)


# ----------------------------------------------------------------------
# Smoothed‑max gradient
# ----------------------------------------------------------------------


def smoothed_gradient(a: np.ndarray, alpha: float):
    """
    Gradient of the smoothed maximum
        f_α(a) = (1/α)·log Σ_k exp(α·(a∗a)_k)

    Returns (grad, conv) where grad has the same shape as a,
    and conv = a∗a (full convolution, length 2n‑1).
    """
    n = len(a)
    # self‑convolution
    conv = convolution_fft(a, a)                # length 2n‑1

    # soft‑max weights over the convolution entries
    maxc = np.max(conv)
    exp_term = np.exp(alpha * (conv - maxc))
    w = exp_term / np.sum(exp_term)             # Σ_k w_k = 1

    # convolution of reversed a with the weight vector
    a_rev = a[::-1]
    conv_rev = convolution_fft(a_rev, w)        # length 3n‑2
    start = n - 1
    rev_slice = conv_rev[start:start + n]       # needed slice

    grad = 2.0 * rev_slice
    return grad, conv


# ----------------------------------------------------------------------
# Projected (sub)gradient descent on the smoothed objective
# ----------------------------------------------------------------------


def optimise_sequence(
    init_a: np.ndarray,
    deadline: float,
    max_iter: int = 200_000,
    eta0: float = 0.8,
    alpha_start: float = 2.0,
    alpha_end: float = 500.0,
    noise_every: int = 400,
    noise_scale: float = 2e-5,
):
    """
    Projected gradient descent on the smoothed max‑convolution objective.
    Returns the best vector seen (as a NumPy array) and its unsmoothed score.
    """
    a = project_to_simplex(init_a.astype(float).copy())
    best_a = a.copy()
    best_score = compute_score(a)

    n = len(a)
    momentum = np.zeros_like(a)
    beta = 0.9  # momentum coefficient

    for t in range(1, max_iter + 1):
        if time.time() >= deadline:
            break

        # linearly increase smoothing parameter
        alpha = alpha_start + (alpha_end - alpha_start) * min(1.0, t / max_iter)

        grad, _ = smoothed_gradient(a, alpha)

        # simple momentum (Nesterov‑like)
        momentum = beta * momentum + (1.0 - beta) * grad
        eta = eta0 / np.sqrt(t)

        a_new = a - eta * momentum

        # occasional tiny random perturbation to escape plateaus
        if noise_every and (t % noise_every == 0):
            a_new += np.random.randn(n) * noise_scale

        a_new = project_to_simplex(a_new)

        cur_score = compute_score(a_new)
        if cur_score < best_score:
            best_score = cur_score
            best_a = a_new.copy()
            # early exit if we are already very close to the target
            if best_score < 1.505:
                # give a bit of extra time for final refinement
                pass

        a = a_new

        # occasional progress report
        if t % 5000 == 0:
            print(f"    iter {t:,d}  score {cur_score:.6f}  best {best_score:.6f}")

    return best_a, best_score


# ----------------------------------------------------------------------
# Cheap stochastic refinement (pair‑wise mass exchange)
# ----------------------------------------------------------------------


def hill_climb(seq, deadline: float, max_steps: int = 80_000, init_step: float = 0.08):
    """
    Stochastic pair‑wise exchange hill‑climbing.
    Moves a proportion of mass from one index to another,
    keeping the simplex constraint; improvements are kept.
    """
    a = np.array(seq, dtype=float)
    a = project_to_simplex(a)
    best_a = a.copy()
    best_score = compute_score(best_a)
    n = len(a)
    step = init_step

    while max_steps > 0 and time.time() < deadline:
        i, j = np.random.randint(n, size=2)
        if i == j:
            continue
        # move a fraction of the mass at i towards j
        delta = step * best_a[i]
        if delta <= 0:
            continue
        cand = best_a.copy()
        cand[i] -= delta
        cand[j] += delta
        # enforce non‑negativity & simplex
        cand = project_to_simplex(cand)

        cur_score = compute_score(cand)
        if cur_score < best_score:
            best_score = cur_score
            best_a = cand
            step = min(step * 1.04, 0.5)   # slowly enlarge step after success
        else:
            step *= 0.96                   # shrink step after failure

        max_steps -= 1

    return best_a.tolist(), best_score
