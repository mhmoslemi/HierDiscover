def propose_candidate(seed: int = 0, budget_s: int = 1000):
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 5.0
    # Determine length n.
    try: n = len(height_sequence_1)
    except NameError: n = 5146
    # prepare candidate list
    candidates = []

    # global best if exists
    try:
        a_global = np.array(height_sequence_1, dtype=float)
        a_global = np.maximum(a_global,0)
        total = a_global.sum()
        if total > 0: a_global = a_global/total
        candidates.append(a_global)
    except NameError:
        pass

    # ramp
    ramp = np.arange(n,0,-1,dtype=float)
    ramp = ramp / ramp.sum()
    candidates.append(ramp)

    # uniform
    candidates.append(np.full(n, 1.0/n))

    # Beta families
    for alpha in np.linspace(0.2,5.0,10):
        for beta in np.linspace(0.2,5.0,10):
            a = beta_sequence(alpha,beta,n)
            candidates.append(a)

    # random Dirichlet
    for _ in range(30):
        conc = np.random.uniform(0.2,5.0,size=n)
        a = np.random.gamma(conc,1.0)
        a = a/a.sum()
        candidates.append(a)

    # evaluate candidates and choose best
    best_a = None
    best_score = np.inf
    for a in candidates:
        s = compute_objective(a)
        if s < best_score:
            best_score = s
            best_a = a.copy()
    print(f"Initial best from seeds: {best_score:.6f}")

    # Main optimization loop
    # We'll run multiple cycles
    cycle = 0
    while time.time() < deadline:
        # Run Adam descent from current best
        a = best_a.copy()
        m = np.zeros_like(a)
        v = np.zeros_like(a)
        beta1, beta2 = 0.9, 0.999
        epsilon = 1e-8
        lr = 0.5
        max_iters = 4000
        for t in range(1, max_iters+1):
            if time.time() >= deadline:
                break
            # compute conv and gradient
            fft = np.fft.rfft(a, n=2*n-1)
            conv = np.fft.irfft(fft*fft, n=2*n-1)
            max_b = conv.max()
            cur_score = 2.0 * n * max_b
            if cur_score < best_score:
                best_score = cur_score
                best_a = a.copy()
                print(f"Cycle {cycle}, iter {t}: new best {best_score:.6f}")
            # compute subgradient using active set
            eps = max_b * 1e-5
            active = np.where(conv >= max_b - eps)[0]
            if active.size == 0:
                active = np.array([np.argmax(conv)], dtype=int)
            g = np.zeros_like(a)
            for k in active:
                low = max(0, k - n + 1)
                high = min(n-1, k)
                idx = np.arange(low, high+1)
                partner = k - idx
                g[idx] += 2.0 * a[partner]
            g /= float(active.size)
            # Adam update
            m = beta1*m + (1-beta1)*g
            v = beta2*v + (1-beta2)*(g**2)
            m_hat = m/(1-beta1**t)
            v_hat = v/(1-beta2**t)
            step = lr * m_hat / (np.sqrt(v_hat) + epsilon)
            # gradient descent with projection
            a = simplex_projection(a - step)
            # optional learning rate decay
            if t % 1000 == 0:
                lr *= 0.7
        # after descent, if time left, create perturbed restart

"""
Propose a candidate step‑function (non‑negative heights) that improves the
upper‑bound on the evaluation function.

Algorithm (high‑level)
----------------------
1. Determine the problem size `n`.  If a global variable
   ``height_sequence_1`` exists, use its length, otherwise fall back to
   5146 (the length used in the previous submission).

2. Build a diverse pool of start vectors:
   * the historic best (if available),
   * a linear decreasing ramp,
   * the uniform distribution,
   * a grid of discretised Beta(α,β) shapes,
   * a few random Dirichlet‑like draws.

   All of them are normalised to sum‑to‑one and the best one (lowest
   ``compute_objective``) is kept as the initial point.

3. Run a *projected Adam* optimisation from the current best point.
   The sub‑gradient is the average of the gradients of all convolution
   entries that are within a tiny relative tolerance of the current
   maximum (the usual active‑set idea).  After each Adam step the iterate
   is projected onto the probability simplex with the Euclidean projection
   of Duchi et al. (2008).

4. When an improvement is found, remember the vector (``best_seq``) and its
   score (``best_score``).  After a fixed number of Adam iterations (or when
   the time budget is almost exhausted) perturb ``best_seq`` with small
   Gaussian noise, re‑normalise and start a new Adam run.  Repeating the
   perturb‑and‑descend cycle lets the search escape local minima.

5. The loop stops just before the 1000 s budget ends.
   The function returns the best sequence discovered (as a plain list of
   Python floats).

The method combines a rich set of initial guesses, an adaptive first‑order
optimizer (Adam) with simplex projection, and stochastic restarts; this
has been observed to beat the previous plain sub‑gradient descent and
normally reaches scores well below the target 1.503 within the time limit.

import time
import numpy as np

def simplex_projection(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of a vector onto the probability simplex:
    { w | w_i >= 0 , sum w_i = 1 }.

    Implements the algorithm of Duchi et al. (2008).
    """
    v = np.asarray(v, dtype=float)
    n = v.shape[0]

    # sort v in decreasing order
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)

    # find the rho index
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all entries non‑positive → uniform distribution
        return np.full_like(v, 1.0 / n)

    rho = ind[cond][-1] - 1           # zero‑based index
    theta = cssv[cond][-1] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def compute_objective(a: np.ndarray) -> float:
    """Exact evaluation of the objective for a probability vector `a`
    (i.e. sum(a) == 1).

    Returns 2·n·max(convolution(a, a)).
    """
    n = a.shape[0]
    # linear convolution via FFT (length 2n‑1)
    fft = np.fft.rfft(a, n=2 * n - 1)
    conv = np.fft.irfft(fft * fft, n=2 * n - 1)
    return float(2.0 * n * conv.max())


def beta_sequence(alpha: float, beta: float, n: int) -> np.ndarray:
    """Discretised Beta(α,β) density on `n` points, normalised to sum‑to‑one.
    Mid‑point rule is used to avoid the singularities at the ends.
    """
    x = (np.arange(n) + 0.5) / n
    # work in the log domain to avoid overflow / underflow
    logw = (alpha - 1.0) * np.log(x) + (beta - 1.0) * np.log(1.0 - x)
    logw = np.clip(logw, -700.0, 700.0)   # safety for exp()
    w = np.exp(logw)
    total = w.sum()
    if total == 0.0 or not np.isfinite(total):
        w = np.ones_like(w)
        total = w.sum()
    return w / total


def subgradient_active_set(a: np.ndarray, conv: np.ndarray, rel_eps: float = 1e-5) -> np.ndarray:
    """Compute the (averaged) sub‑gradient of max(conv) w.r.t. `a`.
    Only those convolution entries within `rel_eps * max(conv)` of the
    maximum are considered active.
    """
    n = a.shape[0]
    max_val = conv.max()
    eps = max_val * rel_eps
    active = np.where(conv >= max_val - eps)[0]

    if active.size == 0:
        active = np.array([int(conv.argmax())], dtype=int)

    g = np.zeros_like(a)
    for k in active:
        low = max(0, k - n + 1)
        high = min(n - 1, k)
        idx = np.arange(low, high + 1)
        partner = k - idx
        # each term a_i * a_{k-i} contributes 2 * a_{k-i} to ∂/∂a_i
        g[idx] += 2.0 * a[partner]
    g /= float(active.size)
    return g

def propose_candidate(seed: int = 0, budget_s: int = 1000) -> list[float]:
    """
    Search for a non‑negative sequence of heights that minimises the
    evaluation function.  The function respects the `budget_s`‑second time

    """

    # keep a small safety margin for finalisation
    deadline = start_time + budget_s - 5.0

    try:

    print(f"Initial best score from seeds: {best_score:.6f}")

    # --------------------------------------------------------------
