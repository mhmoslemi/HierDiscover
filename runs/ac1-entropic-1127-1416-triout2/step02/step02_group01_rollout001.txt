"""
Propose a low‑scoring step‑function.

The algorithm is a three‑phase search:

1. **Initialisation** – start from the analytically‑motivated arcsine shape
   (a_i ∝ 1/√((i+½)(n−½−i))) on a few problem sizes (n≈4096,8192).
   The vector is projected onto the probability simplex (∑a_i=1).

2. **Adam optimisation of a smoothed max‑convolution** – we minimise a
   smooth approximation of the objective
        f_α(a) = (1/α) log Σ_k exp(α·(a∗a)_k)
   using Adam (adaptive moments) with a homotopy on the smoothing
   parameter α (gradually increased to a large value so the smoothed
   problem approaches the original max‑norm).  After each update we
   project onto the simplex.  The best vector seen during the run is
   retained.

3. **Stochastic hill‑climbing refinement** – a cheap pair‑wise mass‑transfer
   local search (with a small simulated‑annealing style acceptance rule)
   that fine‑tunes the solution with respect to the true (non‑smoothed)
   evaluation function.

The search runs under the given time budget (default 1000 s).  If a
previously discovered construction ``height_sequence_1`` exists it is used
as a fallback candidate.  The returned object is a plain Python list of
non‑negative floats.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Basic utilities -------------------------------------------------------
# ----------------------------------------------------------------------


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the probability simplex
    {x | x_i ≥ 0, Σ_i x_i = 1}.
    """
    n = v.shape[0]
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / n)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho = np.where(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """
    Full convolution using real FFT (fast).
    Returns a vector of length len(a)+len(b)-1.
    """
    m = len(a) + len(b) - 1
    n_fft = next_pow2(m)
    A = np.fft.rfft(a, n=n_fft)
    B = np.fft.rfft(b, n=n_fft)
    C = A * B
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


def compute_score(a: np.ndarray) -> float:
    """
    Fast score identical to `evaluate_sequence` when Σa = 1.
    """
    n = len(a)
    conv = convolution_fft(a, a)
    max_val = float(np.max(conv))
    s = float(np.sum(a))
    if s < 1e-12:
        return float("inf")
    return 2.0 * n * max_val / (s * s)


def arcsine_initial(n: int) -> np.ndarray:
    """
    Arcsine‑shaped start: a_i ∝ 1 / sqrt((i+0.5)*(n-0.5-i)).
    Returns a probability vector (sum = 1).
    """
    i = np.arange(n, dtype=float)
    eps = 0.25  # small regularisation to avoid singularities at the ends
    vals = 1.0 / np.sqrt((i + 0.5) * (n - 0.5 - i) + eps)
    return vals / np.sum(vals)


# ----------------------------------------------------------------------
# Smoothed‑max gradient (used inside Adam) -----------------------------
# ----------------------------------------------------------------------


def smoothed_gradient(a: np.ndarray, alpha: float):
    """
    Gradient of the smoothed maximum
        f_α(a) = (1/α)·log Σ_k exp(α·(a∗a)_k)

    Returns a pair (grad, conv) where `grad` has the same shape as `a`
    and `conv` is the full self‑convolution a∗a (length 2·n‑1).
    """
    n = len(a)
    # Full self‑convolution
    conv = convolution_fft(a, a)           # length 2n-1

    # Soft‑max weights, numerically stable
    maxc = np.max(conv)
    exp_term = np.exp(alpha * (conv - maxc))
    w = exp_term / np.sum(exp_term)        # Σ w = 1

    # Convolution of reversed a with the weight vector
    a_rev = a[::-1]
    conv_rev = convolution_fft(a_rev, w)   # length 3n-2
    start = n - 1
    rev_slice = conv_rev[start:start + n]  # contribution for each a_i

    # Diagonal weights (those where index of conv equals 2*i)
    w_diag = w[::2][:n]

    # Gradient of the smoothed objective
    grad = 2.0 * rev_slice - w_diag * a
    return grad, conv


# ----------------------------------------------------------------------
# Adam optimiser on the smoothed objective -----------------------------
# ----------------------------------------------------------------------


def improve_sequence_adam(
    initial_a: np.ndarray,
    time_deadline: float,
    max_iter: int = 150_000,
    eta0: float = 0.5,
    alpha_start: float = 1.0,
    alpha_end: float = 800.0,
):
    """
    Projected Adam on the smoothed max‑convolution objective.
    Returns the best vector (as a NumPy array) and its unsmoothed score.
    """
    a = project_to_simplex(initial_a.astype(float).copy())
    best_a = a.copy()
    best_score = compute_score(a)

    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8

    for t in range(1, max_iter + 1):
        if time.time() >= time_deadline:
            break

        # Linear homotopy on α (smoothness parameter)
        alpha = alpha_start + (alpha_end - alpha_start) * min(1.0, t / max_iter)

        grad, _ = smoothed_gradient(a, alpha)

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)

        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)

        # Learning‑rate schedule (decays like 1/√t)
        lr = eta0 / np.sqrt(t)

        a = a - lr * m_hat / (np.sqrt(v_hat) + eps)
        a = project_to_simplex(a)

        cur_score = compute_score(a)
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()

        if t % 10000 == 0:
            print(f"    Adam iter {t:,d}  score {cur_score:.6f}  best {best_score:.6f}")

    return best_a, best_score


# ----------------------------------------------------------------------
# Simple hill‑climbing / simulated‑annealing refinement -----------------
# ----------------------------------------------------------------------


def hill_climb(seq, time_deadline: float, max_steps: int = 30_000, init_step: float = 0.05):
    """
    Stochastic pair‑wise exchange hill‑climb with a tiny annealing component.
    Moves a small fraction of mass from index i to index j.
    """
    a = np.array(seq, dtype=float)
    a = project_to_simplex(a)
    best_a = a.copy()
    best_score = compute_score(best_a)
    n = len(a)

    step = init_step
    temperature = 0.01  # initial temperature for occasional uphill moves

    while max_steps > 0 and time.time() < time_deadline:
        i, j = np.random.randint(n, size=2)
        if i == j:
            continue
        # Propose moving a fraction of the mass at i to j
        delta = step * best_a[i]
        if delta <= 0:
            continue
        cand = best_a.copy()
        cand[i] -= delta
        cand[j] += delta
        # Ensure non‑negativity (project just in case)
        cand = project_to_simplex(cand)
        cur_score = compute_score(cand)

        # Acceptance rule: always accept improvements,
        # otherwise accept with probability exp(-(Δ)/T)
        if cur_score < best_score:
            best_score = cur_score
            best_a = cand
            step = min(step * 1.07, 0.5)      # slowly enlarge successful step
            temperature *= 0.995             # cool down a bit
        else:
            # uphill move with a decreasing probability
            prob = np.exp(-(cur_score - best_score) / max(temperature, 1e-12))
            if np.random.rand() < prob:
                best_a = cand
                best_score = cur_score
                temperature *= 0.997
            step *= 0.98                      # shrink step on failure

        max_steps -= 1

    return best_a.tolist(), best_score


# ----------------------------------------------------------------------
# Main entry point ------------------------------------------------------
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative sequence that minimises ``evaluate_sequence``.
    The routine performs a multi‑scale Adam optimisation followed by a
    cheap stochastic refinement.  The best sequence found before the
    time budget expires is returned as a plain list of floats.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 2.0   # small safety margin

    best_seq = None
    best_val = float("inf")

    # ------------------------------------------------------------------
    # 1.  Fallback to any previously discovered construction.
    # ------------------------------------------------------------------
    try:
        fallback = [float(x) for x in height_sequence_1]   # type: ignore
        fallback_val = evaluate_sequence(fallback)
        print(f"Fallback score = {fallback_val:.6f}")
        if fallback_val < best_val:
            best_val = fallback_val
            best_seq = fallback
    except Exception:
        pass

    # ------------------------------------------------------------------
    # 2.  Multi‑scale Adam optimisation.
    # ------------------------------------------------------------------
    # Problem sizes – larger n leads to a smaller asymptotic constant
    # but costs more per iteration.  We aim for n≈8192 if time permits.
    lengths = [4096, 8192]

    for n in lengths:
        if time.time() >= deadline - 5.0:
            break
        print(f"\nOptimising length n = {n}")

        # Initialise with the arcsine shape (good theoretical start)
        init_a = arcsine_initial(n)

        # Determine a sensible iteration budget based on remaining time.
        # Rough estimate: 0.8 s per 10 k iterations for n≈4096,
        # ~1.5 s per 10 k for n≈8192 on typical hardware.
        remaining = deadline - time.time()
        # Allocate roughly half of the remaining budget to this size.
        target_seconds = remaining * 0.5
        # Empirical per‑iteration cost (seconds):
        cost_per_iter = 0.00008 if n <= 4096 else 0.00015
        max_iter = max(1, int(target_seconds / cost_per_iter))

        # Run the Adam optimiser.
        seq_opt, val_opt = improve_sequence_adam(
            initial_a=init_a,
            time_deadline=deadline,
            max_iter=max_iter,
            eta0=0.7,
            alpha_start=1.0,
            alpha_end=800.0,
        )
        # Evaluate with the true (unsmoothed) objective.
        real_val = evaluate_sequence(seq_opt.tolist())
        print(f"  candidate (n={n}) score = {real_val:.6f}")

        if real_val < best_val:
            best_val = real_val
            best_seq = seq_opt.tolist()
            print(f"  *** New global best {best_val:.6f} (n={n}) ***")

    # ------------------------------------------------------------------
    # 3.  Local stochastic refinement (if time permits).
    # ------------------------------------------------------------------
    if best_seq is not None and time.time() < deadline - 2.0:
        print("\nRefining best sequence with hill‑climb")
        refined_seq, refined_score = hill_climb(
            best_seq,
            time_deadline=deadline,
            max_steps=60_000,
            init_step=0.04,
        )
        refined_val = evaluate_sequence(refined_seq)
        print(f"    refined score = {refined_val:.6f}")
        if refined_val < best_val:
            best_val = refined_val
            best_seq = refined_seq

    # ------------------------------------------------------------------
    # 4.  Fallback in the unlikely event that we have no candidate.
    # ------------------------------------------------------------------
    if best_seq is None:
        best_seq = [1.0] * 256
        best_val = evaluate_sequence(best_seq)

    final_score = evaluate_sequence(best_seq)
    print(f"\nFinished search. Final reported score = {final_score:.6f}")

    # The harness expects a plain Python list of floats.
    return best_seq
