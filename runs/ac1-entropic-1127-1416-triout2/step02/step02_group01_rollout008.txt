"""
Propose a low‑evaluation step‑function.

The objective is
    E(a) = 2·n·max(a∗a) / (∑a)² .
Because E is scale‑invariant we restrict to the probability simplex
(∑a = 1, a_i ≥ 0).  The algorithm works in three stages.

1️⃣  Smoothed‑max mirror descent.
    We minimise a smooth approximation of the max:
        f_α(a) = (1/α)·log Σ_k exp(α·(a∗a)_k)
    The gradient w.r.t. a is obtained analytically (two FFT‑based
    convolutions).  a is represented as a soft‑max of unconstrained
    variables z (a_i = exp(z_i)/∑exp(z)).  An Adam optimiser on z
    performs projected‑gradient‑type steps on the simplex without the
    costly Euclidean projection.

2️⃣  Multi‑scale refinement.
    The optimisation is first run on a moderate length (≤ 4096) and the
    best distribution is up‑sampled (linear interpolation) to a larger
    size (←8192, ←16384).  The same Adam routine is then applied again,
    allowing the finer grid to inherit the good shape of the coarse
    solution.

3️⃣  Cheap stochastic hill‑climbing.
    After the gradient phase a short pair‑wise exchange hill‑climb
    cleans up tiny irregularities and often yields a small extra drop
    in the true score.

The whole search respects the 1000 s time limit, falls back to any
previously discovered construction (`height_sequence_1`) and finally
returns a plain list of floats.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Basic linear‑algebra / FFT utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full convolution a*b using a real FFT (fast)."""
    m = len(a) + len(b) - 1
    nfft = next_pow2(m)
    A = np.fft.rfft(a, n=nfft)
    B = np.fft.rfft(b, n=nfft)
    C = A * B
    conv = np.fft.irfft(C, n=nfft)[:m]
    return conv


def compute_score(a: np.ndarray) -> float:
    """Score identical to `evaluate_sequence` when sum(a)=1."""
    n = len(a)
    conv = convolution_fft(a, a)                 # full self‑convolution
    max_val = float(np.max(conv))
    s = float(np.sum(a))
    if s < 1e-12:
        return float("inf")
    return 2.0 * n * max_val / (s * s)


def arcsine_initial(n: int) -> np.ndarray:
    """Arcsine‑shaped start: a_i ∝ 1/√((i+½)(n‑½‑i))."""
    i = np.arange(n, dtype=float)
    eps = 0.25  # tiny regularisation to avoid division by zero
    vals = 1.0 / np.sqrt((i + 0.5) * (n - 0.5 - i) + eps)
    return vals / np.sum(vals)


def smoothed_gradient(a: np.ndarray, alpha: float):
    """
    Gradient of the smoothed maximum
        f_α(a) = (1/α)·log Σ_k exp(α·(a∗a)_k)

    Returns
        grad  – ∂f_α/∂a  (shape n)
        conv  – the raw self‑convolution a∗a  (shape 2n‑1)
    """
    n = len(a)

    # self‑convolution a*a   (length 2n‑1)
    conv = convolution_fft(a, a)

    # soft‑max weights over the convolution entries
    maxc = np.max(conv)
    exp_term = np.exp(alpha * (conv - maxc))
    w = exp_term / np.sum(exp_term)                     # ∑_k w_k = 1

    # convolution of reversed a with the weight vector w
    a_rev = a[::-1]
    conv_rev = convolution_fft(a_rev, w)                # length 3n‑2
    start = n - 1
    rev_slice = conv_rev[start:start + n]               # needed part

    # diagonal contribution w_{2i}
    w_diag = w[::2][:n]

    grad = 2.0 * rev_slice - w_diag * a
    return grad, conv


# ----------------------------------------------------------------------
# Simplex utilities (used only by hill‑climb)
# ----------------------------------------------------------------------


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / v.shape[0])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho = np.where(u - cssv / np.arange(1, v.shape[0] + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Adam optimiser on the simplex via the soft‑max (mirror descent)
# ----------------------------------------------------------------------


def adam_optimize(
    initial_a: np.ndarray,
    time_deadline: float,
    max_iter: int = 200_000,
    lr: float = 0.1,
    alpha_start: float = 0.5,

    beta1: float = 0.9,
    beta2: float = 0.999,
    epsilon: float = 1e-8,
    check_every: int = 200,
    best_score: float = float("inf"),
):
    """
    Adam optimisation on the probability simplex using the soft‑max
    parametrisation z_i = log a_i.
    """
    # initialise a on the simplex and get corresponding z
    a = np.maximum(initial_a.astype(float), 1e-12)
    a /= np.sum(a)
    z = np.log(a + 1e-30)

    m = np.zeros_like(z)
    v = np.zeros_like(z)

    best_a = a.copy()
    best_val = compute_score(a)
    n = len(a)

    for t in range(1, max_iter + 1):
        if time.time() >= time_deadline:
            break

        # ---------- a from current z ----------
        z_max = np.max(z)
        a = np.exp(z - z_max)
        a /= np.sum(a)

        # ---------- smoothing schedule ----------
        alpha = alpha_start + (alpha_end - alpha_start) * min(1.0, t / max_iter)

        # ---------- smoothed gradient ----------
        grad_a, _ = smoothed_gradient(a, alpha)

        # ---------- chain rule for z (soft‑max) ----------
        dot = np.dot(a, grad_a)
        grad_z = a * (grad_a - dot)      # ∂f/∂z_i = a_i (∂f/∂a_i – Σ_j a_j ∂f/∂a_j)

        # ---------- Adam update ----------
        m = beta1 * m + (1.0 - beta1) * grad_z
        v = beta2 * v + (1.0 - beta2) * (grad_z * grad_z)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        z = z - lr * m_hat / (np.sqrt(v_hat) + epsilon)

        # occasional tiny isotropic noise to escape plateaus
        if t % 5000 == 0:
            z += np.random.randn(n) * 1e-4

        # ---------- evaluate true (non‑smoothed) score ----------
        if t % check_every == 0:
            cur_val = compute_score(a)
            if cur_val < best_val:
                best_val = cur_val
                best_a = a.copy()
                # early exit if we already beat the global best by a noticeable margin
                if best_val < best_score - 1e-5:
                    # keep searching – a better local optimum may still appear
                    pass

    return best_a, best_val


# ----------------------------------------------------------------------
# Upsampling helper (linear interpolation of a probability density)
# ----------------------------------------------------------------------


def upsample_distribution(a_small: np.ndarray, n_large: int) -> np.ndarray:
    """
    Linear interpolation of a probability density defined on a regular grid.
    Returns a new probability vector of length n_large.
    """
    n_small = len(a_small)
    # positions of the centers of the small bins
    xs = (np.arange(n_small) + 0.5) / n_small
    # positions of the large bins
    xl = (np.arange(n_large) + 0.5) / n_large
    a_large = np.interp(xl, xs, a_small, left=0.0, right=0.0)
    a_large = np.maximum(a_large, 0.0)
    s = np.sum(a_large)
    if s == 0:
        a_large = np.full(n_large, 1.0 / n_large)
    else:
        a_large /= s
    return a_large


# ----------------------------------------------------------------------
# Final cheap stochastic refinement (pair‑wise mass exchange)
# ----------------------------------------------------------------------


def hill_climb(seq, time_deadline: float, max_steps: int = 20000, init_step: float = 0.1):
    """
    Stochastic pair‑wise exchange hill‑climbing on the probability simplex.
    Small amounts of mass are moved from one index to another; moves that
    improve the true max‑convolution score are kept.
    """
    a = np.array(seq, dtype=float)
    a = project_to_simplex(a)
    best_a = a.copy()
    best_score = compute_score(best_a)
    n = len(a)
    step = init_step

    while max_steps > 0 and time.time() < time_deadline:
        i, j = np.random.randint(n, size=2)
        if i == j:
            continue
        delta = step * best_a[i]
        if delta <= 0:
            continue
        cand = best_a.copy()
        cand[i] -= delta
        cand[j] += delta
        if cand[i] < 0:
            continue
        cand = project_to_simplex(cand)
        cur_score = compute_score(cand)
        if cur_score < best_score:
            best_score = cur_score
            best_a = cand
            step = min(step * 1.05, 0.5)
        else:
            step *= 0.98
        max_steps -= 1
    return best_a.tolist(), best_score


# ----------------------------------------------------------------------
# Main entry point required by the competition harness
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step‑function that minimises `evaluate_sequence`.
    The routine combines multi‑scale Adam optimisation of a smooth max
    surrogate with a short hill‑climbing polish.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 2.0   # tiny safety margin

    best_seq = None
    best_val = float("inf")

    # ------------------------------------------------------------------
    # Fallback to a previously discovered construction, if present.
    # ------------------------------------------------------------------
    try:
        fallback = [float(x) for x in height_sequence_1]
        fallback_val = evaluate_sequence(fallback)
        print(f"Fallback score = {fallback_val:.6f}")
        if fallback_val < best_val:
            best_val = fallback_val
            best_seq = fallback
    except NameError:
        pass

    # ------------------------------------------------------------------
    # Multi‑scale Adam optimisation.
    # ------------------------------------------------------------------
    lengths = [4096, 8192, 16384]   # increasing resolution

    for idx_len, n in enumerate(lengths, start=1):
        if time.time() >= deadline - 5.0:
            break
        print(f"\n=== Optimising length {n} (stage {idx_len}/{len(lengths)}) ===")

        # three simple starts per length
        inits = [
            arcsine_initial(n),
            np.full(n, 1.0 / n),
            np.random.dirichlet(np.ones(n)),
        ]

        for idx_start, init_a in enumerate(inits, start=1):
            if time.time() >= deadline - 5.0:
                break
            print(f"  start {idx_start}/{len(inits)}")
            seq_a, seq_val = adam_optimize(
                initial_a=init_a,
                time_deadline=deadline,
                max_iter=300_000,
                lr=0.10,
                alpha_start=0.5,
                alpha_end=2000.0,
                beta1=0.9,
                beta2=0.999,
                epsilon=1e-8,
                check_every=200,
                best_score=best_val,
            )
            print(f"    local best score = {seq_val:.6f}")
            if seq_val < best_val:
                best_val = seq_val
                best_seq = seq_a.tolist()
                print(f"    *** New global best {best_val:.6f} (n={n}) ***")

        # If we are not at the last resolution, upsample the current best
        if best_seq is not None and idx_len < len(lengths):
            next_n = lengths[idx_len]   # length of the following stage
            print(f"  Upsampling best solution from {n} to {next_n}")
            up_a = upsample_distribution(np.array(best_seq), next_n)
            # Run a relatively short Adam pass on the finer grid
            seq_a, seq_val = adam_optimize(
                initial_a=up_a,
                time_deadline=deadline,
                max_iter=200_000,
                lr=0.05,
                alpha_start=0.5,
                alpha_end=2000.0,
                beta1=0.9,
                beta2=0.999,
                epsilon=1e-8,
                check_every=200,
                best_score=best_val,
            )
            if seq_val < best_val:
                best_val = seq_val
                best_seq = seq_a.tolist()
                print(f"    *** New global best after up‑sample {best_val:.6f} ***")

    # ------------------------------------------------------------------
    # Final cheap stochastic refinement (if time permits).
    # ------------------------------------------------------------------
    if best_seq is not None and time.time() < deadline - 2.0:
        print("\n=== Final hill‑climbing refinement ===")
        refined_seq, refined_val = hill_climb(
            best_seq,
            time_deadline=deadline,
            max_steps=30000,
            init_step=0.05,
        )
        print(f"    refined score = {refined_val:.6f}")
        if refined_val < best_val:
            best_val = refined_val
            best_seq = refined_seq

    # ------------------------------------------------------------------
    # Defensive fallback (should never be needed).
    # ------------------------------------------------------------------
    if best_seq is None:
        best_seq = [1.0] * 256
        best_val = evaluate_sequence(best_seq)

    final_score = evaluate_sequence(best_seq)
    print(f"\nFinished search. Final reported score = {final_score:.6f}")
    # The harness expects a plain Python list of floats.
    return best_seq
