def propose_candidate(...):
    import time, numpy as np
    deadline = time.time() + budget_s - 5
    n = 5000

    try:
        init_seq = np.array(height_sequence_1, dtype=float)

    except NameError:

        a = raw / raw.sum()
    best_a = a.copy()
    best_score = 2.0 * n * np.max(full_convolution(a))
    print(f"Initial score: {best_score:.6f}")

    # gradient descent parameters
    iteration = 0
    beta = 5.0
    step = 0.5

    while time.time() < deadline:
        # compute conv via FFT
        conv = full_convolution(a)
        max_conv = conv.max()
        # compute softmax weights
        # shift to prevent overflow
        shifted = conv - max_conv
        exp_terms = np.exp(beta * shifted)
        w = exp_terms / exp_terms.sum()
        # compute gradient via correlation using FFT
        L = 1 << ((len(w) + n - 1).bit_length())  # length >= len(w)+n-1 = 3n-2
        A = np.fft.rfft(a, n=L)
        W = np.fft.rfft(w, n=L)
        corr = np.fft.irfft(np.conj(A) * W, n=L)
        # slice central n entries
        grad = 2.0 * corr[n-1 : n-1+n]   # shape (n,)
        # update
        a_new = a - step * grad
        a = project_onto_simplex(a_new)

        # evaluate score
        conv_new = full_convolution(a)
        score = 2.0 * n * conv_new.max()
        if score < best_score:
            best_score = score
            best_a = a.copy()
            print(f"Iter {iteration}: improved score {best_score:.6f}")
            if best_score < 1.503:  # early exit
                break

        # adapt step and beta
        iteration += 1
        # diminishing step
        step = 0.5 / np.sqrt(iteration + 1)
        # increase beta gradually up to maybe 100
        if beta < 200:
            beta = min(beta * 1.005, 200.0)

        # occasional random jitter to escape local minima

"""
Propose a candidate step‑function by minimising the sup‑norm of the
self‑convolution with a smooth approximation.

Algorithm
---------
1. Work on a fixed length `n` (default 5000).  The coefficients are kept
   on the probability simplex (non‑negative, sum = 1); the evaluation
   function then returns `2·n·max(conv(a,a))`.

2. Initialise `a` with a classic decreasing shape (≈1/√i) or, if a
   global `height_sequence_1` is present and has the correct length,
   with that sequence normalised to sum = 1.

3. Repeatedly:
   * Compute the full self‑convolution `c = a * a` with an FFT.
   * Build a smooth approximation of the sup‑norm:
        w_k = exp(β·(c_k – max(c))) / Σ_j exp(β·(c_j – max(c))).
     For β ≈ 5 the weights are almost uniform; they are increased
     slowly up to a few hundred so that the optimisation finally focuses
     on the true maximum.
   * The gradient of the smoothed objective is
        g = 2·corr(a, w)                (correlation, length n)
     which is obtained efficiently with another pair of FFTs.
   * Take a diminishing projected gradient step and project back onto
     the simplex.
   * Every few hundred iterations perform a tiny random jitter of the
     mass to escape shallow plateaux.
   * Keep the best sequence seen so far.

4. The loop stops when the time budget (≈ 1000 s) is exhausted or when a
   score better than the target `1.503` is found.

The routine returns a plain Python list of floats, ready for
`evaluate_sequence`.  If something goes wrong a simple uniform sequence
is returned as a safe fallback.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper functions (top‑level, no closures)
# ----------------------------------------------------------------------


def project_onto_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the probability simplex:
        { x | x >= 0, sum(x) = 1 }.
    Returns a new array (the input is not modified).
    """
    n = v.shape[0]
    # Sort v in descending order
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(n) + 1
    # Find the rho index
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # All entries are zero – return the uniform distribution
        return np.full_like(v, 1.0 / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def full_convolution(a: np.ndarray) -> np.ndarray:
    """
    Compute the full discrete convolution a * a using an FFT.
    Length of the result is 2*len(a) - 1.
    """
    n = a.shape[0]
    size = 2 * n
    # next power of two (speed)
    fft_len = 1 << (size - 1).bit_length()
    fft_a = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(fft_a * fft_a, n=fft_len)
    return conv[: 2 * n - 1]


def smoothed_gradient(a: np.ndarray, beta: float) -> np.ndarray:
    """
    Compute the gradient of the soft‑maximum of the self‑convolution.
    Returns a vector of length len(a).
    """
    n = a.shape[0]
    # 1) self‑convolution
    conv = full_convolution(a)                # length 2n‑1
    max_conv = conv.max()

    # 2) soft‑max weights (avoid overflow)
    shifted = conv - max_conv                 # max becomes 0
    exp_terms = np.exp(beta * shifted)
    w = exp_terms / exp_terms.sum()           # length 2n‑1, sums to 1

    # 3) correlation a ↔ w (g_i = 2·Σ_k w_k·a_{k‑i})
    # linear correlation can be obtained as ifft(conj(fft(a))*fft(w))
    L = 1 << ((len(w) + n - 1).bit_length())  # >= 3n‑2
    A = np.fft.rfft(a, n=L)
    W = np.fft.rfft(w, n=L)
    corr = np.fft.irfft(np.conj(A) * W, n=L)
    # centre part corresponding to i = 0 … n‑1
    grad = 2.0 * corr[n - 1 : n - 1 + n]
    return grad


def objective_score(a: np.ndarray) -> float:
    """
    Compute the exact score returned by evaluate_sequence when sum(a)=1:
        2·n·max(conv(a,a))
    """
    n = a.shape[0]
    return 2.0 * n * full_convolution(a).max()


# ----------------------------------------------------------------------
# Main search routine
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step‑function that minimises
    evaluate_sequence(sequence).

    Parameters
    ----------
    seed : int
        Random seed for reproducibility.
    budget_s : int
        Time budget in seconds (default 1000).  A few seconds are kept
        for final printing, so the actual optimisation stops slightly
        earlier.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 5.0  # safety margin

    # ------------------------------------------------------------------
    # Choose length and initialise a
    # ------------------------------------------------------------------
    n = 5000
    # try to reuse a previous construction if it matches the length
    try:
        init = np.array(height_sequence_1, dtype=float)
        if init.size == n:
            a = init.copy()
        else:
            raise ValueError
    except Exception:
        # classic 1/√(i+1) shape
        raw = 1.0 / np.sqrt(np.arange(n, dtype=float) + 1.0)
        a = raw / raw.sum()

    # ensure we are on the simplex (sum = 1)
    a = a / a.sum()
    best_seq = a.copy()
    best_score = objective_score(a)
    print(f"Initial score (n={n}): {best_score:.6f}")

    # ------------------------------------------------------------------
    # Parameters for the smoothed gradient descent
    # ------------------------------------------------------------------
    iteration = 0
    beta = 5.0                 # start with a gentle smoothing
    step_initial = 0.5         # initial step size factor

    # ------------------------------------------------------------------
    # Main optimisation loop
    # ------------------------------------------------------------------
    while time.time() < deadline:
        # 1) gradient of smoothed sup‑norm
        grad = smoothed_gradient(a, beta)

        # 2) diminishing step and projection onto simplex
        step = step_initial / np.sqrt(iteration + 1.0)
        a_candidate = a - step * grad
        a = project_onto_simplex(a_candidate)

        # 3) evaluate the exact objective
        score = objective_score(a)
        if score < best_score:
            best_score = score
            best_seq = a.copy()
            print(f"Iter {iteration:06d}: improved score = {best_score:.6f}")

            # early exit if we have beaten the target
            if best_score < 1.503:
                break

        # 4) schedule for β (make the approximation sharper)
        if beta < 200.0:
            beta = min(beta * 1.005, 200.0)

        # 5) occasional random jitter to escape tiny plateaux
        if iteration % 500 == 0 and iteration > 0:
            i, j = np.random.randint(0, n, size=2)
            if i != j:
                # move a very small fraction of mass from i to j
                delta = 0.005 * a[i] * (np.random.rand() - 0.5)
                cand = a.copy()
                cand[i] = max(cand[i] - delta, 0.0)
                cand[j] = cand[j] + delta
                cand = project_onto_simplex(cand)
                a = cand

        iteration += 1

    # ------------------------------------------------------------------
    # Fallback in the unlikely event that nothing was found
    # ------------------------------------------------------------------
    if best_seq is None or best_seq.size == 0:
        print("Fallback: returning a uniform sequence.")
        best_seq = np.full(n, 1.0 / n, dtype=float)

    print(f"Final best score: {best_score:.6f} (length {len(best_seq)})")
    # Return a plain Python list as required by the driver
    return best_seq.tolist()
