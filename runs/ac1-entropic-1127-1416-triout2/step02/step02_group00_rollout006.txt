The search works as follows:

1. **Fallback** – use a previously discovered sequence if it exists.
2. **Length sweep** – try several lengths (up to 16384).

def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Convolution via real FFT (zero‑padding to the next power of two)."""
    m = len(a) + len(b) - 1
    n_fft = next_pow2(m)
    A = np.fft.rfft(a, n=n_fft)
    B = np.fft.rfft(b, n=n_fft)
    C = A * B
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv

    """Return (max value, arg‑max index) of a*a."""
    conv = convolution_fft(a, a)
    idx = int(np.argmax(conv))
    return float(conv[idx]), idx


def subgradient_of_max_at_index(a: np.ndarray, k: int) -> np.ndarray:
    """
    Sub‑gradient of the maximal convolution entry (index k) with respect to a.

    For each i, ∂c_k/∂a_i = a_{k‑i} if 0 ≤ k‑i < n,
    and double the diagonal term when k = 2·i.
    """
    n = len(a)
    grad = np.zeros_like(a)
    i = np.arange(n)
    j = k - i
    mask = (j >= 0) & (j < n)
    grad[mask] = a[j[mask]]
    if k % 2 == 0:
        diag = k // 2
        if diag < n:
            grad[diag] = 2.0 * a[diag]
    return grad


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the probability simplex
    {x | x_i ≥ 0, Σ_i x_i = 1}.
    Implementation of Duchi et al. (2008).
    """
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / len(v))

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho = np.where(u - cssv / np.arange(1, len(v) + 1) > 0)[0]
    if len(rho) == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def compute_score(a: np.ndarray) -> float:
    """
    Fast, scale‑invariant objective used by the official evaluator:
        2·n·max(a*a) / (sum(a)²)
    (the input is assumed to be a probability vector, i.e. sum≈1).
    """
    n = len(a)
    conv = convolution_fft(a, a)
    max_val = float(np.max(conv))
    s = float(np.sum(a))
    if s < 1e-12:
        return float("inf")
    return 2.0 * n * max_val / (s * s)


def arcsine_initial(n: int) -> np.ndarray:
    """
    Classic arcsine‑shaped start:
        a_i ∝ 1 / √((i+½)(n‑½‑i)).
    """
    i = np.arange(n, dtype=float)
    eps = 0.25                     # small regularisation to avoid 0·∞
    vals = 1.0 / np.sqrt((i + 0.5) * (n - 0.5 - i) + eps)
    return vals / np.sum(vals)


def generate_shape(n: int, p: float, eps: float) -> np.ndarray:
    """
    Produce a symmetric density   a_i ∝ ((i+ε)(n‑1‑i+ε))⁻ᵖ,
    then normalise to unit L¹ norm.
    """
    i = np.arange(n, dtype=float)
    base = ((i + eps) * (n - 1 - i + eps)) ** (-p)
    total = base.sum()
    if total == 0.0:
        # degenerate case – fall back to uniform
        return np.full(n, 1.0 / n)
    return base / total


def improve_sequence(initial_a: np.ndarray,
                    time_deadline: float,
                    max_iter: int = 200_000,
                    eta0: float = 0.5,
                    noise_every: int = 500,
                    noise_scale: float = 1e-4) -> Tuple[np.ndarray, float]:
    """
    Projected sub‑gradient descent on the max‑convolution objective.
    Occasionally Gaussian noise is added and a tiny smoothing step is
    applied to help escape local plateaus.

    Returns the best vector seen (as a NumPy array) and its internal
    score as measured by ``compute_score``.
    """
    a = project_to_simplex(initial_a.astype(float).copy())
    best_a = a.copy()
    best_sc = compute_score(a)

    n = len(a)
    smooth_kernel = np.array([0.25, 0.5, 0.25], dtype=float)
    smooth_every = 5000

    it = 0
    while time.time() < time_deadline and it < max_iter:
        it += 1

        # worst convolution entry and its sub‑gradient
        _, k_max = max_convolution_and_index(a)
        grad = subgradient_of_max_at_index(a, k_max)

        # diminishing step size
        eta = eta0 / np.sqrt(it)

        a_new = a - eta * grad

        if noise_every and (it % noise_every == 0):
            a_new += np.random.randn(n) * noise_scale

        a_new = project_to_simplex(a_new)

        # occasional light smoothing
        if it % smooth_every == 0:
            a_new = np.convolve(a_new, smooth_kernel, mode='same')
            a_new = project_to_simplex(a_new)

        cur_sc = compute_score(a_new)
        if cur_sc < best_sc:
            best_sc = cur_sc
            best_a = a_new.copy()

        a = a_new

        # occasional progress report (quiet if time is tight)
        if it % 5000 == 0 and time.time() < time_deadline - 1.0:
            print(f"    iter={it:,d}  cur={cur_sc:.6f}  best={best_sc:.6f}")

    return best_a, best_sc

def propose_candidate(seed: int = 42,
                     budget_s: int = 1000,
                     **kwargs) -> List[float]:
    """
    Search for a non‑negative step‑function with a low
    ``evaluate_sequence`` value.  The routine respects the time budget,
    explores several lengths, refines promising candidates with a
    projected sub‑gradient optimiser and finally returns the best vector
    found (as a plain list of floats).
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    # small safety margin so that we always finish before the hard limit
    deadline = start_time + budget_s - 2.0

    best_seq: List[float] = []
    best_val: float = float("inf")
    TARGET = 1.503  # we aim to beat this value

    # ------------------------------------------------------------------
    # 1.  Fallback – use any previously discovered construction if present.
    # ------------------------------------------------------------------
    try:
        fallback = [float(x) for x in height_sequence_1]   # type: ignore
        val = evaluate_sequence(fallback)
        print(f"Fallback (height_sequence_1) score = {val:.6f}")
        if val < best_val:
            best_val = val
            best_seq = fallback
    except NameError:
        pass

    # ------------------------------------------------------------------
    # 2.  Exploration over a range of lengths.
    # ------------------------------------------------------------------
    lengths = [512, 1024, 2048, 4096, 8192, 16384]

    for n in lengths:
        if time.time() >= deadline - 5.0:
            break
        print(f"\n=== Exploring length n = {n} ===")

        # --------------------------------------------------------------
        # 2a.  Find a good shape for this length.
        # --------------------------------------------------------------
        local_best_vec: np.ndarray | None = None
        local_best_sc = float("inf")

        # deterministic starters
        starters = [np.full(n, 1.0 / n), arcsine_initial(n)]
        for starter in starters:
            sc = compute_score(starter)
            if sc < local_best_sc:
                local_best_sc = sc
                local_best_vec = starter.copy()

        # random samples from the (p, eps) family
        max_samples = 1500  # enough to get a good candidate quickly
        for _ in range(max_samples):
            if time.time() >= deadline - 5.0:
                break
            p = rng.uniform(0.15, 0.75)                 # p around the arcsine region
            eps = rng.uniform(0.001, max(2.0, 0.02 * n))
            cand = generate_shape(n, p, eps)
            sc = compute_score(cand)
            if sc < local_best_sc:
                local_best_sc = sc
                local_best_vec = cand

        # a few purely random Dirichlet vectors (useful for very large n)
        for _ in range(200):
            if time.time() >= deadline - 5.0:
                break
            cand = rng.random(n)
            cand /= cand.sum()
            sc = compute_score(cand)
            if sc < local_best_sc:
                local_best_sc = sc
                local_best_vec = cand

        # --------------------------------------------------------------
        # 2b.  Refine the locally best candidate with sub‑gradient descent.
        # --------------------------------------------------------------
        if local_best_vec is not None and time.time() < deadline - 5.0:
            remaining = deadline - time.time()
            # allocate a fraction of the remaining time to this length
            max_iter = min(250_000, int(remaining * 30))
            print(f"  Refining length {n} (max_iter={max_iter}) – "
                  f"pre‑refine score = {local_best_sc:.6f}")
            refined_a, _ = improve_sequence(
                initial_a=local_best_vec,
                time_deadline=deadline,
                max_iter=max_iter,
                eta0=0.7,
                noise_every=800,
                noise_scale=5e-5,
            )
            # evaluate with the official (slow) evaluator once
            val = evaluate_sequence(refined_a.tolist())
            print(f"  → refined score = {val:.6f}")
            if val < best_val:
                best_val = val
                best_seq = refined_a.tolist()
                print(f"  *** New global best: {best_val:.6f} (n={n}) ***")
                if best_val < TARGET:
                    # early exit – we already beat the target, no need to
                    # keep trying larger sizes unless we have ample time.
                    print("Target achieved – will continue polishing only.")
                    # we keep exploring only for polishing later
        else:
            # No time for refinement – just test the raw best shape.
            if local_best_vec is not None:
                val = evaluate_sequence(local_best_vec.tolist())
                print(f"  Raw candidate score = {val:.6f}")
                if val < best_val:
                    best_val = val
                    best_seq = local_best_vec.tolist()
                    print(f"  *** New global best (raw) ***")

    # ------------------------------------------------------------------
    # 3.  Final polishing of the best sequence found so far.
    # ------------------------------------------------------------------
    if best_seq and time.time() < deadline - 5.0:
        a = np.array(best_seq, dtype=float)
        remaining = deadline - time.time()
        max_iter = min(300_000, int(remaining * 30))
        print(f"\nFinal polishing (len={len(a)}) – max_iter={max_iter}")
        polished_a, _ = improve_sequence(
            initial_a=a,
            time_deadline=deadline,
            max_iter=max_iter,
            eta0=0.6,
            noise_every=600,
            noise_scale=3e-5,
        )
        final_val = evaluate_sequence(polished_a.tolist())
        print(f"Final polished score = {final_val:.6f}")
        if final_val < best_val:
            best_val = final_val
            best_seq = polished_a.tolist()
            print("*** Updated global best after final polish ***")

    # ------------------------------------------------------------------
    # 4.  Guard – ensure something is always returned.
    # ------------------------------------------------------------------
    if not best_seq:
        # As a last resort fall back to a modest arcsine shape.
        best_seq = arcsine_initial(2048).tolist()
        best_val = evaluate_sequence(best_seq)

    print(f"\nFinished search. Best reported score = {best_val:.6f}")
    return best_seq
