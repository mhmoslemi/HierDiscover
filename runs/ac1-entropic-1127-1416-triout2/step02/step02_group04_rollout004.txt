"""
Propose a non‑negative height sequence that minimizes
    2·n·max(a*a)   (a is normalised to sum = 1)

The algorithm works in three phases

1.  Initialise a good starting point:
    * use the previously found sequence ``height_sequence_1`` if it exists,
    * a few analytically inspired Beta‑shapes,
    * a handful of random Dirichlet samples.

2.  Run a projected (sub‑)gradient descent on a *smoothed* version of the
    max‑convolution objective.
    The smoothing uses a soft‑max (log‑sum‑exp) over the convolution
    entries, giving a dense gradient that points away from the currently
    largest values.  A simple momentum term and a diminishing step size are
    employed, and after every update the iterate is projected onto the
    probability simplex.

3.  When the time budget is nearly exhausted the best vector seen so far is
    returned as a plain list of Python ``float`` objects.

The whole routine respects the 1000 s budget; if the deadline is reached it
simply returns the best sequence found up to that moment.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities
# ----------------------------------------------------------------------


def simplex_projection(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the probability simplex:
    { w | w_i >= 0, sum w_i = 1 }.
    Implementation follows Duchi et al. (2008).
    """
    v = np.asarray(v, dtype=float)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):                     # all non‑positive → uniform
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1] - 1                  # zero‑based index
    theta = cssv[cond][-1] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def compute_objective(a: np.ndarray) -> float:
    """
    Exact objective (2·n·max(a*a)) assuming sum(a) == 1.
    Uses FFT for an O(n log n) convolution.
    """
    n = a.shape[0]
    fft = np.fft.rfft(a, n=2 * n - 1)
    conv = np.fft.irfft(fft * fft, n=2 * n - 1)
    return 2.0 * n * conv.max()


def beta_sequence(alpha: float, beta: float, n: int) -> np.ndarray:
    """
    Discretised Beta(α,β) density on n points (mid‑point rule),
    normalised to sum = 1.
    """
    x = (np.arange(n) + 0.5) / n
    logw = (alpha - 1.0) * np.log(x) + (beta - 1.0) * np.log(1.0 - x)
    logw = np.clip(logw, -700.0, 700.0)      # avoid overflow
    w = np.exp(logw)
    total = w.sum()
    if total == 0 or not np.isfinite(total):
        w = np.ones_like(w)
        total = w.sum()
    return w / total


def next_pow2(x: int) -> int:
    """Small helper – next power of two greater or equal to x."""
    return 1 << (x - 1).bit_length()

    while time.time() < deadline and iteration < 50000:
        # ---- convolution a*a (linear) ---------------------------------
        A_fft = np.fft.rfft(a, n=L)
        conv_full = np.fft.irfft(A_fft * A_fft, n=L)
        conv = conv_full[: 2 * n - 1]                # length 2n‑1
        max_c = conv.max()
        cur_score = 2.0 * n * max_c

        # maintain the best solution seen so far
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"Iter {iteration:5d}: new best {best_score:.6f}")

        # ---- smoothing (soft‑max) over convolution entries -------------
        if iteration == 0:
            eps = max_c * 0.2 + eps_min               # initial smoothing
        else:
            eps = max(eps * eps_decay, eps_min)

        # w_k ∝ exp((c_k‑max)/eps)  (numerically stable)
        w = np.exp((conv - max_c) / eps)
        w_sum = w.sum()
        if w_sum == 0 or not np.isfinite(w_sum):
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        # ---- gradient of the smoothed objective ------------------------
        W_fft = np.fft.rfft(w, n=L)
        grad_full = np.fft.irfft(W_fft * A_fft, n=L)
        grad = 2.0 * grad_full[:n]                   # only first n entries needed

        # ---- projected gradient step with momentum --------------------
        step = step_base / np.sqrt(iteration + 1)
        velocity = mu * velocity - step * grad
        a_next = simplex_projection(a + velocity)

        # occasional tiny random jitter to help escape plateaus
        if iteration % 1000 == 0 and iteration > 0:
            jitter = np.random.normal(scale=0.001, size=n)
            a_next = simplex_projection(a_next + jitter)

        a = a_next
        iteration += 1

    # ------------------------------------------------------------------
    # 4. Final report
    # ------------------------------------------------------------------
    elapsed = time.time() - start_time
    print(f"Finished after {iteration} iterations ({elapsed:.1f}s), best score = {best_score:.6f}")

    # Return the best sequence as a list of Python floats
    return [float(v) for v in best_a.tolist()]
