from scipy.optimize import linprog
import scipy.sparse as sp

def solve_lp_max_sum(f, M):
    n = len(f)
    # Build sparse matrix H: for each k (0..2n-2), for each i (0..n-1) such that 0 <= k-i < n:
    # j = i index in f? Actually constraint: sum_i f_i * g_{k-i} <= M.
    # Equivalent to: sum_j f_{k-j} * g_j <= M.
    # For each k row, non-zero entries correspond to j where 0 <= k-j < n => max(0, k-(n-1)) <= j <= min(k, n-1).
    # coefficient at column j = f_{k-j}
    row = []
    col = []
    data = []
    for k in range(2*n-1):
        j_start = max(0, k-(n-1))
        j_end = min(k, n-1)
        # indices j_start .. j_end inclusive
        for j in range(j_start, j_end+1):
            row.append(k)
            col.append(j)
            data.append(f[k-j])
    A = sp.csr_matrix((data, (row, col)), shape=(2*n-1, n))
    b = np.full(2*n-1, M)
    c = -np.ones(n)  # minimize -sum
    res = linprog(c, A_ub=A, b_ub=b, bounds=(0, None), method='highs')
    if not res.success:
        return None
    g = res.x
    return g

"""
Propose a step‑function with a very low evaluation value.

The search proceeds in three phases:

0.  Initialise the sequence.  If a historic best ``height_sequence_1`` is
    available it is taken, otherwise a simple power‑law candidate is used.
1.  A fast random sweep over a 4‑parameter family (power law + linear tilt)
    looks for a better warm‑start.
2.  A lengthy projected sub‑gradient descent with momentum
    (instead of the plain sub‑gradient in the reference implementation)
    reduces the maximum self‑convolution.  The step size is adaptive –
    it decays slowly and also reacts to the current optimality gap.
    Small random jitters are injected from time‑to‑time to escape plateaus.

The objective of ``evaluate_sequence`` equals ``max(conv(a,a))`` once the
sum of the coefficients is fixed to ``sqrt(2·n)``; therefore the algorithm
optimises the infinity‑norm of the linear convolution while keeping the
vector on the simplex ``{x≥0, Σx = sqrt(2·n)}``.
The routine respects the 1000 s time limit and returns the best sequence
found.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities – all are top‑level (no closures)
# ----------------------------------------------------------------------


def _project_simplex(v: np.ndarray, target_sum: float) -> np.ndarray:
    """
    Euclidean projection of a vector onto the simplex
        { w | w_i ≥ 0, Σ w_i = target_sum }.
    Implements the algorithm of Duchi et al. (2008).
    """
    if target_sum < 0:
        raise ValueError("target_sum must be non‑negative")
    u = np.sort(v)[::-1]                     # sorted descending
    cssv = np.cumsum(u) - target_sum
    ind = np.arange(1, len(v) + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.zeros_like(v)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def _full_convolution_fft(a: np.ndarray) -> np.ndarray:
    """
    Linear (full) convolution a*a using a real FFT.
    Returns an array of length 2·len(a)-1.
    """
    n = len(a)
    size = 1 << ((2 * n - 1).bit_length())      # next power of two
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return conv


def _max_conv(a: np.ndarray) -> float:
    """Maximum entry of the linear self‑convolution of ``a``."""
    return _full_convolution_fft(a).max()


def _make_candidate(n: int, p: float, offset: float, beta: float) -> np.ndarray:
    """
    Build an un‑normalised raw sequence
        a_i = (i + 1 + offset)^{-p} * (1 + beta * i/(n-1))
    and project it onto the simplex Σ a_i = sqrt(2·n).
    """
    idx = np.arange(n, dtype=float)                # i = 0 … n‑1
    base = np.power(idx + 1.0 + offset, -p)
    factor = 1.0 + beta * idx / (n - 1)            # linear tilt
    raw = base * factor
    raw = np.maximum(raw, 0.0)                     # clip negatives
    target = np.sqrt(2.0 * n)
    return _project_simplex(raw, target)


def _subgradient(a: np.ndarray, max_inds: np.ndarray) -> np.ndarray:
    """
    Compute a (sub)gradient of the functional  max(conv(a,a))
    at the current point.  Only the indices attaining the maximum
    (``max_inds``) contribute.
    """
    n = len(a)
    g = np.zeros_like(a)
    for k in max_inds:
        if k < n:
            # i runs 0 … k
            g[: k + 1] += a[k::-1]                     # reversed prefix
        else:
            # i runs k-(n-1) … n-1
            start = k - (n - 1)
            idx = np.arange(start, n)
            g[start:] += a[k - idx]                    # reversed suffix
    return g


# ----------------------------------------------------------------------
# Main search routine
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 123, budget_s: int = 1000) -> list[float]:
    """
    Search for a non‑negative sequence that minimises
        2·n·max(conv(a,a)) / (sum(a)²)

    The method combines a random parametric sweep with a long
    projected sub‑gradient descent that uses momentum and an adaptive
    step size.  The best sequence encountered before the time budget
    expires is returned as a plain Python list.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5   # small safety margin

    # ------------------------------------------------------------------
    # 0. Choose a sensible length and obtain a warm start.
    # ------------------------------------------------------------------
    base = globals().get("height_sequence_1")
    if base is not None:
        n = len(base)
        a = np.array(base, dtype=float)
    else:
        n = 7000                       # default length used in the reference code
        a = None
    target_sum = np.sqrt(2.0 * n)

    if a is None:
        # simple power‑law start
        a = _make_candidate(n, p=0.5, offset=0.0, beta=0.0)
    else:
        # enforce exact sum for safety
        a = _project_simplex(a, target_sum)

    best_seq = a.copy()
    best_score = _max_conv(best_seq)
    print(f"[init] n={n:5d}  score={best_score:.6f}")

    # ------------------------------------------------------------------
    # 1. Coarse random exploration of the 4‑parameter family.
    #    Use roughly 30 % of the remaining budget.
    # ------------------------------------------------------------------
    sweep_deadline = time.time() + 0.30 * (deadline - time.time())
    while time.time() < sweep_deadline:
        p = np.random.uniform(0.30, 0.80)               # exponent
        off = np.random.uniform(0.0, 12.0)               # offset
        beta = np.random.uniform(-1.0, 3.0)              # tilt
        cand = _make_candidate(n, p, off, beta)
        cand_score = _max_conv(cand)
        if cand_score < best_score:
            best_score = cand_score
            best_seq = cand.copy()
            a = cand.copy()
            print(f"[sweep] new best {best_score:.6f}  p={p:.3f} off={off:.3f} β={beta:.3f}")

        if best_score < 1.503:        # early exit if target already reached
            break

    # ------------------------------------------------------------------
    # 2. Projected sub‑gradient descent with momentum.
    #    The step size decays slowly (power 0.35) and also adapts to
    #    the current optimality gap.
    # ------------------------------------------------------------------
    momentum = np.zeros_like(a)
    mu = 0.9                     # momentum factor
    step0 = 5.0                  # base step size (larger than the reference code)
    iteration = 0
    g_norm_epsilon = 1e-16

    while time.time() < deadline:
        iteration += 1

        # current objective value
        conv_full = _full_convolution_fft(a)
        M = conv_full.max()

        # store best solution seen so far
        if M < best_score - 1e-12:
            best_score = M
            best_seq = a.copy()
            print(f"[sgd] iter {iteration:6d}  new best {best_score:.6f}")
            if best_score < 1.503:
                break

        # indices where the maximum is attained (with a tiny tolerance)
        eps = 1e-12 * M if M != 0 else 1e-12
        max_inds = np.where(conv_full >= M - eps)[0]

        # sub‑gradient of the max‑convolution
        g = _subgradient(a, max_inds)

        # adaptive step size:
        #   - a slowly decaying term for long‑range progress
        #   - a gap‑driven term that shrinks as we approach the current best
        g_norm2 = np.dot(g, g) + g_norm_epsilon
        gap = M - best_score
        step_gap = gap / g_norm2 if gap > 1e-12 else 0.0
        step_decay = step0 / ((iteration + 1) ** 0.35)
        step = max(step_decay, step_gap)

        # momentum update and projection onto the simplex
        momentum = mu * momentum - step * g
        cand = a + momentum
        cand = _project_simplex(cand, target_sum)

        # occasional random jitter to escape flat regions
        if iteration % 100 == 0 and np.random.rand() < 0.25:
            jitter = (np.random.rand(n) - 0.5) * 0.02 * cand
            cand = cand + jitter
            cand = _project_simplex(cand, target_sum)

        a = cand

    print(f"Finished after {iteration} iterations – best score {best_score:.6f}")
    # Convert to plain Python list as required by the evaluation harness.
    return best_seq.tolist()
