def lp_refine(f, M):

    n = len(f)
    # Build constraint matrix A_ub shape (2n-1, n)
    rows = []
    for k in range(2*n-1):
        coeff = np.zeros(n)
        # i from max(0, k-n+1) to min(k, n-1)
        lo = max(0, k - n + 1)
        hi = min(k, n - 1)
        # for each i, coefficient for b_{k-i} is f_i.
        # So for each i in range(lo, hi+1):
        #   idx = k - i
        #   coeff[idx] = f_i
        idx = np.arange(lo, hi+1)
        coeff[k - idx] = f[idx]
        rows.append(coeff)
    A_ub = np.stack(rows, axis=0)  # shape (2n-1,n)
    b_ub = np.full(2*n-1, M)
    c = -np.ones(n)  # maximize sum => minimize -sum

    # bounds
    bounds = [(0, None)] * n

    res = opt.linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
    if not res.success:
        return None
    b = res.x
    # Normalize to sum=1
    if b.sum() == 0:
        return None
    b /= b.sum()
    return b

"""
Fast smoothing‑gradient search for the step‑function minimising
``evaluate_sequence``.  The algorithm runs several independent
initialisations (uniform, arcsine, random symmetric β‑shapes and random
Dirichlet vectors) for a few sequence lengths.  For each candidate a
smooth approximation of the max‑convolution is minimised by a projected
Adam optimiser.  A continuation on the smoothing parameter (β) allows
the optimiser to first move in a gentle landscape and then focus on the
true max.  The best vector found (projected onto the probability simplex)
is returned as a plain list of floats.

The code respects the 1000 s budget, prints intermediate scores and
never returns an empty sequence.
"""

import time
import numpy as np
from typing import List, Tuple

# ----------------------------------------------------------------------
# Low‑level utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Convolution via real FFT (zero‑padding to the next power of two)."""
    m = len(a) + len(b) - 1
    n_fft = next_pow2(m)
    A = np.fft.rfft(a, n=n_fft)
    B = np.fft.rfft(b, n=n_fft)
    C = A * B
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of ``v`` onto the probability simplex
    {x | x_i ≥ 0, Σ_i x_i = 1}.  Implementation of Duchi et al. (2008).
    """
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / len(v))

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho = np.where(u - cssv / np.arange(1, len(v) + 1) > 0)[0]
    if len(rho) == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def compute_score(a: np.ndarray) -> float:
    """
    Fast surrogate of the official objective:
        2·n·max(a*a) / (sum(a)²)
    (a is assumed to be on the simplex, i.e. sum≈1).
    """
    n = len(a)
    conv = convolution_fft(a, a)
    max_val = float(np.max(conv))
    s = float(np.sum(a))
    if s < 1e-12:
        return float("inf")
    return 2.0 * n * max_val / (s * s)


def arcsine_initial(n: int) -> np.ndarray:
    """Classic arcsine density a_i ∝ 1/√((i+½)*(n‑½‑i))."""
    i = np.arange(n, dtype=float)
    eps = 0.25
    vals = 1.0 / np.sqrt((i + 0.5) * (n - 0.5 - i) + eps)
    return vals / np.sum(vals)


def generate_shape(n: int, p: float, eps: float) -> np.ndarray:
    """Symmetric family  a_i ∝ ((i+ε)(n‑1‑i+ε))⁻ᵖ  (p>0, ε>0)."""
    i = np.arange(n, dtype=float)
    base = ((i + eps) * (n - 1 - i + eps)) ** (-p)
    total = base.sum()
    if total == 0.0:
        return np.full(n, 1.0 / n)
    return base / total


def random_dirichlet(n: int) -> np.ndarray:
    """Draw a random point on the simplex (Gamma(1) → Dirichlet)."""
    y = np.random.gamma(1.0, 1.0, size=n)
    if y.sum() == 0.0:
        return np.full(n, 1.0 / n)
    return y / y.sum()


# ----------------------------------------------------------------------
# Smoothed‑max optimiser
# ----------------------------------------------------------------------


def smooth_max_optimizer(
    init_a: np.ndarray,
    deadline: float,
    max_iter: int = 200_000,
    beta_start: float = 5.0,
    beta_end: float = 200.0,
    eta0: float = 0.05,
    noise_every: int = 500,
    noise_scale: float = 2e-4,
) -> Tuple[np.ndarray, float]:
    """
    Projected Adam optimiser for the smooth approximation of
    ``max(a*a)``.  The smoothing parameter β is increased linearly from
    ``beta_start`` to ``beta_end`` (continuation method).
    Returns the best vector seen (still on the simplex) and its
    ``compute_score`` value.
    """
    a = project_to_simplex(init_a.astype(float).copy())
    best_a = a.copy()
    best_sc = compute_score(a)

    n = len(a)
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8

    for it in range(1, max_iter + 1):
        if time.time() >= deadline:
            break

        # current convolution and its maximum
        conv = convolution_fft(a, a)
        max_val = np.max(conv)

        # continuation on β
        prog = it / max_iter
        beta = beta_start + (beta_end - beta_start) * prog

        # soft‑max weights
        w = np.exp(beta * (conv - max_val))
        w_sum = w.sum()
        if w_sum == 0.0:
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        # gradient of the weighted sum  g_i = Σ_k w_k ∂c_k/∂a_i
        # using the convolution trick: grad = (w_rev * a)[2n‑2‑i] + w_{2i}·a_i
        w_rev = w[::-1]
        grad_full = convolution_fft(w_rev, a)                 # length = 3n‑2
        idx = (len(w) - 1) - np.arange(n)                     # 2n‑2‑i
        grad = grad_full[idx]
        diag_idx = 2 * np.arange(n)
        grad += w[diag_idx] * a                               # double the diagonal term

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        step = eta0 * m_hat / (np.sqrt(v_hat) + eps)

        a_new = a - step

        # occasional isotropic noise to escape shallow plateaux
        if noise_every and (it % noise_every == 0):
            a_new += np.random.randn(n) * noise_scale

        a_new = project_to_simplex(a_new)

        cur_sc = compute_score(a_new)
        if cur_sc < best_sc:
            best_sc = cur_sc
            best_a = a_new.copy()

        a = a_new

        if it % 5000 == 0:
            print(
                f"    iter={it:,d}  cur={cur_sc:.6f}  best={best_sc:.6f}  β={beta:.1f}"
            )

    return best_a, best_sc

# Main entry point required by the competition harness

def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs) -> List[float]:
    """
    Search for a non‑negative step‑function with a low ``evaluate_sequence``
    value.  Several sequence lengths are examined, each is initialised with
    a handful of sensible families, then refined by the smoothed‑max
    optimiser.  The best candidate overall is returned.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 2.0           # tiny safety margin

    best_seq: List[float] = []
    best_val: float = float("inf")

    # ------------------------------------------------------------------
    # 0.  Use any previously discovered construction if present.
    # ------------------------------------------------------------------
    try:
        fallback = [float(x) for x in height_sequence_1]   # type: ignore
        val = evaluate_sequence(fallback)
        print(f"Fallback (height_sequence_1) score = {val:.6f}")
        if val < best_val:
            best_val = val
            best_seq = fallback
    except NameError:
        pass

    # ------------------------------------------------------------------
    # 1.  Lengths to try – short ones converge quickly, long ones have
    #     more freedom.  We interleave them to make the best use of time.
    # ------------------------------------------------------------------
    lengths = [1024, 2048, 512, 256]

    for n in lengths:
        if time.time() >= deadline - 10.0:
            break
        print(f"\n=== Exploring length n = {n} ===")

        # --------------------------------------------------------------
        # 1a.  Build a pool of initial vectors.
        # --------------------------------------------------------------
        init_pool: List[np.ndarray] = []

        # deterministic starters
        init_pool.append(np.full(n, 1.0 / n))      # uniform
        init_pool.append(arcsine_initial(n))      # classic arcsine

        # a few random (p, eps) shapes
        for _ in range(12):
            p = rng.uniform(0.05, 1.2)
            eps = rng.uniform(1e-3, max(2.0, 0.02 * n))
            init_pool.append(generate_shape(n, p, eps))

        # a few Dirichlet random vectors
        for _ in range(12):
            init_pool.append(random_dirichlet(n))

        local_best_a = None
        local_best_sc = float("inf")

        # --------------------------------------------------------------
        # 1b.  Quick evaluation of the pool (fast surrogate).
        # --------------------------------------------------------------
        for cand in init_pool:
            sc = compute_score(cand)
            if sc < local_best_sc:
                local_best_sc = sc
                local_best_a = cand.copy()

        # --------------------------------------------------------------
        # 1c.  Refine the most promising candidate with the smooth optimiser.
        # --------------------------------------------------------------
        if local_best_a is not None and time.time() < deadline - 5.0:
            # estimate a generous iteration budget – roughly 30 it/s
            remaining = deadline - time.time()
            max_iter = min(180_000, int(remaining * 30))

            print(f"  Refining (max_iter={max_iter}) …")
            refined_a, _ = smooth_max_optimizer(
                init_a=local_best_a,
                deadline=deadline,
                max_iter=max_iter,
                eta0=0.07,
                noise_every=700,
                noise_scale=1e-4,
                beta_start=8.0,
                beta_end=150.0,
            )
            val = evaluate_sequence(refined_a.tolist())
            print(f"  → refined score = {val:.6f}")
            if val < best_val:
                best_val = val
                best_seq = refined_a.tolist()
                print(f"  *** New global best: {best_val:.6f} (n={n}) ***")
        else:
            # Fallback: evaluate the raw best pooled candidate.
            if local_best_a is not None:
                val = evaluate_sequence(local_best_a.tolist())
                print(f"  Raw candidate score = {val:.6f}")
                if val < best_val:
                    best_val = val
                    best_seq = local_best_a.tolist()
                    print(f"  *** New global best (raw) ***")

    # ------------------------------------------------------------------
    # 2.  Final polishing of the overall best (time‑permitting).
    # ------------------------------------------------------------------
    if best_seq and time.time() < deadline - 5.0:
        a = np.array(best_seq, dtype=float)
        remaining = deadline - time.time()
        max_iter = min(300_000, int(remaining * 30))
        print(f"\nFinal polishing (max_iter={max_iter}) …")
        a_refined, _ = smooth_max_optimizer(
            init_a=a,
            deadline=deadline,
            max_iter=max_iter,
            eta0=0.05,
            noise_every=600,
            noise_scale=5e-5,
            beta_start=10.0,
            beta_end=200.0,
        )
        final_val = evaluate_sequence(a_refined.tolist())
        print(f"Final polished score = {final_val:.6f}")
        if final_val < best_val:
            best_val = final_val
            best_seq = a_refined.tolist()
            print("*** Updated global best after final polish ***")

    # ------------------------------------------------------------------
    # 3.  Safety net – always return *something*.
    # ------------------------------------------------------------------
    if not best_seq:
        best_seq = [1.0] * 256
        best_val = evaluate_sequence(best_seq)

    print(f"\nFinished search. Best reported score = {best_val:.6f}")
    return best_seq
