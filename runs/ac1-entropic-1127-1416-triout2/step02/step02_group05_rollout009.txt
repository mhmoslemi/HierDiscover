def conv_fft(a):
    n = len(a)
    size = 1 << (2*n-1).bit_length()
    # could use np.fft.rfft for real
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)

"""
Alternating linear‑programming optimisation for the min‑peak convolution problem.

The method keeps a non‑negative step‑height vector `a` (∑a=1).  For the
current vector we solve a linear program that finds a partner `x` which
minimises the peak of the convolution `x * a`.  A convex combination
`a←(1‑τ)·a+τ·x` reduces the maximal autocorrelation while staying on the
probability simplex.  Convolutions are evaluated by FFT for speed.
The best vector seen within the time budget is returned.
"""
import time
import numpy as np
from scipy.optimize import linprog

def binomial_sequence(length: int) -> np.ndarray:
    """Return the first *length* central binomial coefficients."""
    seq = np.empty(length, dtype=np.float64)
    seq[0] = 1.0
    for k in range(1, length):
        seq[k] = seq[k - 1] * ((2 * k - 1) * (2 * k)) / (4.0 * k * k)
    return seq

def _next_pow2(x: int) -> int:
    """Small helper returning the smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()

def conv_fft(a: np.ndarray) -> np.ndarray:
    """Convolution of a with itself using a real FFT."""
    n = a.shape[0]
    conv_len = 2 * n - 1
    size = _next_pow2(conv_len)
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)
    # Numerical noise can produce tiny negative values – clip them.
    conv = conv[:conv_len]
    conv[conv < 0] = 0.0
    return conv

def _build_lp_constraints(a: np.ndarray, idx: np.ndarray):
    """Construct A_ub and b_ub for the LP that minimises max(a * x)
    over the given index set `idx`.  The last variable is the auxiliary
    scalar `t` representing the peak value.
    """
    n = a.shape[0]
    k_cnt = idx.shape[0]
    # Allocate dense matrix – k_cnt is tiny (≤~30) so memory is negligible.
    A_ub = np.zeros((k_cnt, n + 1), dtype=np.float64)
    for row, k in enumerate(idx):
        lo = max(0, k - (n - 1))
        hi = min(n - 1, k)
        i_range = np.arange(lo, hi + 1, dtype=int)
        A_ub[row, i_range] = a[k - i_range]
        A_ub[row, -1] = -1.0        # -t ≤ - Σ a_i x_{k-i}
    b_ub = np.zeros(k_cnt, dtype=np.float64)
    return A_ub, b_ub

            raise ValueError
    except Exception:

        n = 8000
        a = binomial_sequence(n).astype(np.float64)
    # Normalise to the probability simplex (the objective is scale‑invariant).
    a = a / a.sum()
    n = a.shape[0]

    cur_conv = conv_fft(a)
    best_score = 2.0 * n * cur_conv.max()
    best_seq = a.copy()
    print(f"[init] length={n}, initial score={best_score:.9f}")

    tau = 0.5                 # initial mixing weight

Alternating linear‑programming optimisation for the min‑peak convolution
problem.

The algorithm keeps a non‑negative step‑height vector `a` with ∑a = 1.
For the current vector we solve a tiny linear programme that finds a
partner `x` which minimises the peak of the convolution `x * a`.  A
convex combination `a ← (1‑τ)·a + τ·x` (with τ slowly decaying) reduces
the maximal autocorrelation while staying on the probability simplex.
Convolutions are evaluated by an FFT, which makes the inner loop fast.
The best vector seen before the time budget expires is returned.
"""

import time

from scipy.optimize import linprog

# ----------------------------------------------------------------------
def binomial_sequence(length: int) -> np.ndarray:
    """Return the first *length* central‑binomial coefficients
    c_k = C(2k, k) / 4**k, k = 0 … length‑1.
    """
    seq = np.empty(length, dtype=np.float64)
    seq[0] = 1.0
    for k in range(1, length):
        seq[k] = seq[k - 1] * ((2 * k - 1) * (2 * k)) / (4.0 * k * k)
    return seq


# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Small helper returning the smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


# ----------------------------------------------------------------------
def conv_fft(a: np.ndarray) -> np.ndarray:
    """Convolution of a with itself via a real FFT (fast, O(n log n))."""
    n = a.shape[0]
    conv_len = 2 * n - 1
    size = _next_pow2(conv_len)
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)
    conv = conv[:conv_len]
    # Numerical noise may generate tiny negatives → clip.
    conv[conv < 0] = 0.0
    return conv


# ----------------------------------------------------------------------
def _build_lp_constraints(a: np.ndarray, idx: np.ndarray):
    """
    Build the inequality matrix for the LP

        minimise t
        s.t.    (a * x)_k ≤ t    for every k in ``idx``
                Σ x_i = 1,  x_i ≥ 0,  t ≥ 0

    The last variable in the LP is the auxiliary scalar ``t``.
    """
    n = a.shape[0]
    k_cnt = idx.shape[0]
    A_ub = np.zeros((k_cnt, n + 1), dtype=np.float64)

    for row, k in enumerate(idx):
        lo = max(0, k - (n - 1))
        hi = min(n - 1, k)
        i_range = np.arange(lo, hi + 1, dtype=int)
        # coefficient of x_i in (a * x)_k is a_{k‑i}
        A_ub[row, i_range] = a[k - i_range]
        A_ub[row, -1] = -1.0          # bring the auxiliary variable t to the LHS

    b_ub = np.zeros(k_cnt, dtype=np.float64)
    return A_ub, b_ub
