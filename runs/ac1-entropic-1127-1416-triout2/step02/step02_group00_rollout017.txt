def smoothed_max_gradient(a, tau):
    # conv = a*a
    conv = convolution_fft(a, a)
    # compute max
    max_c = conv.max()
    # compute weights
    scaled = (conv - max_c) / tau   # max entry becomes 0
    w = np.exp(scaled)
    w_sum = w.sum()
    # compute conv of reversed a with w
    a_rev = a[::-1]
    conv_w = convolution_fft(a_rev, w)
    # slice to get size n
    start = len(a) - 1
    g = conv_w[start:start+len(a)]  # length n
    # add extra diag contribution
    diag_idx = np.arange(len(a))
    extra = w[2*diag_idx] * a
    g = g + extra
    # Normalize gradient by sum of weights
    grad = g / w_sum
    return grad, conv.max()
```

def adam_optimize(initial_a, deadline, max_iter=...):
    a = project_to_simplex(initial_a.copy())
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    best_a = a.copy()
    best_score = compute_score(a)
    t = 0
    tau0 = 0.1
    tau_decay = 0.999 # per iteration
    while time.time() < deadline and t < max_iter:
        t += 1
        # temperature schedule
        tau = max(tau0 * (tau_decay**t), 1e-6)
        grad, max_conv = smoothed_max_gradient(a, tau)
        # Adam update
        m = beta1*m + (1-beta1)*grad
        v = beta2*v + (1-beta2)*(grad*grad)
        m_hat = m / (1 - beta1**t)
        v_hat = v / (1 - beta2**t)
        step = 0.1 / (np.sqrt(v_hat) + eps)  # typical lr scaling
        a_new = a - step * m_hat
        a_new = project_to_simplex(a_new)
        # compute score every 1000 iters maybe
        if t % 1000 == 0:
            sc = compute_score(a_new)
            if sc < best_score:
                best_score = sc
                best_a = a_new.copy()
        a = a_new
    return best_a, best_score
```

"""
Optimisation of a non‑negative step function for the ``evaluate_sequence`` score.

The algorithm proceeds in three stages:

1. **Initial construction** –  a quick random scan of a two‑parameter
   symmetric family  a_i ∝ ((i+ε)(n‑1‑i+ε))⁻ᵖ  (p>0, ε>0) is performed for a
   handful of lengths (512 … 4096).  The best candidate (lowest fast
   ``compute_score``) is kept as a starting point.

2. **Accelerated smoothing descent** –  a smooth approximation of the
   ``max`` in the objective is built with a temperature τ.  The gradient
   of the smoothed max can be obtained by two FFT‑based convolutions,
   after which an Adam‑style update on the probability simplex is taken.
   τ is decreased slowly, giving a fast, global descent.

3. **Fine‑grained projected sub‑gradient descent** –  the vector obtained
   from stage 2 is further refined using the classic sub‑gradient method
   on the true max‑convolution (one FFT per iteration).  The Euclidean
   projection onto the simplex guarantees feasibility.

The whole routine respects a 1000 s budget, falling back to the global
variable ``height_sequence_1`` if it exists.  The returned list of floats
represents the best step‑function found.
"""

import time
import numpy as np
from typing import List, Tuple

# ----------------------------------------------------------------------
# Low‑level utilities (no closures)
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Linear convolution via real FFT (zero‑padded to a power of two)."""
    m = len(a) + len(b) - 1
    n_fft = next_pow2(m)
    A = np.fft.rfft(a, n=n_fft)
    B = np.fft.rfft(b, n=n_fft)
    C = A * B
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


def max_convolution_and_index(a: np.ndarray) -> Tuple[float, int]:
    """Return (max value, arg‑max index) of the full self‑convolution a*a."""
    conv = convolution_fft(a, a)
    idx = int(np.argmax(conv))
    return float(conv[idx]), idx


def subgradient_of_max_at_index(a: np.ndarray, k: int) -> np.ndarray:
    """
    Sub‑gradient of the maximal convolution entry (index k) w.r.t. a.

    For each i, ∂c_k/∂a_i = a_{k‑i} if 0 ≤ k‑i < n,
    and doubled diagonal term when k = 2·i.
    """
    n = len(a)
    grad = np.zeros_like(a)
    i = np.arange(n)
    j = k - i
    mask = (j >= 0) & (j < n)
    grad[mask] = a[j[mask]]
    if k % 2 == 0:
        diag = k // 2
        if diag < n:
            grad[diag] = 2.0 * a[diag]
    return grad


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection onto the simplex {x ≥ 0, Σx = 1}.
    Duchi et al., 2008.
    """
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / len(v))

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho = np.where(u - cssv / np.arange(1, len(v) + 1) > 0)[0]
    if len(rho) == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def compute_score(a: np.ndarray) -> float:
    """
    Fast proxy for the official objective:
        2·n·max(a*a) / (Σa)²
    The vector is assumed to be on the simplex (Σa≈1).
    """
    n = len(a)
    conv = convolution_fft(a, a)
    max_val = float(np.max(conv))
    if max_val < 0:
        max_val = 0.0
    return 2.0 * n * max_val


def arcsine_initial(n: int) -> np.ndarray:
    """Classic arcsine‑shaped start: a_i ∝ 1/√((i+½)*(n‑½‑i))."""
    i = np.arange(n, dtype=float)
    eps = 0.25
    vals = 1.0 / np.sqrt((i + 0.5) * (n - 0.5 - i) + eps)
    return vals / vals.sum()


def generate_shape(n: int, p: float, eps: float) -> np.ndarray:
    """
    Symmetric density a_i ∝ ((i+eps)*(n‑1‑i+eps))⁻ᵖ,
    then normalised to unit L¹ norm.
    """
    i = np.arange(n, dtype=float)
    base = ((i + eps) * (n - 1 - i + eps)) ** (-p)
    total = base.sum()
    if total == 0.0:
        return np.full(n, 1.0 / n)
    return base / total


# ----------------------------------------------------------------------
# Smoothed‑max gradient (used by the Adam optimiser)
# ----------------------------------------------------------------------


def smoothed_max_gradient(a: np.ndarray, tau: float) -> Tuple[np.ndarray, float]:
    """
    Gradient of          log‑sum‑exp((a*a)/τ)      w.r.t. a,
    which is a smooth approximation of the sub‑gradient of max(a*a).
    Returns (grad, max_conv) where max_conv is the true maximum of a*a.
    """
    n = len(a)
    conv = convolution_fft(a, a)                # length 2n‑1
    max_val = float(conv.max())
    # compute softmax‑like weights, shifted for numerical stability
    w = np.exp((conv - max_val) / tau)          # max weight = 1
    w_sum = w.sum()

    # gradient = (1 / w_sum) * [ (a_rev * w)_{centre} + extra_diag ]
    a_rev = a[::-1]
    conv_w = convolution_fft(a_rev, w)          # length 3n‑2
    start = n - 1
    g = conv_w[start:start + n]                 # contribution a_{k-i}
    # double the diagonal term when k = 2*i
    diag_extra = w[2 * np.arange(n)] * a
    g = g + diag_extra
    grad = g / w_sum
    return grad, max_val

# Optimisers
# ----------------------------------------------------------------------


def adam_smooth_optimize(initial_a: np.ndarray,
                         deadline: float,
                         max_iter: int = 200_000) -> Tuple[np.ndarray, float]:
    """
    Adam optimisation on the smoothed‑max objective.
    Returns the best vector seen (according to the fast ``compute_score``)
    and its score.
    """
    a = project_to_simplex(initial_a.astype(float).copy())
    best_a = a.copy()
    best_sc = compute_score(a)

    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-9
    lr = 0.15               # base learning‑rate
    tau = 0.02              # initial temperature

    t = 0
    while time.time() < deadline and t < max_iter:
        t += 1
        # smoothed gradient
        grad, _ = smoothed_max_gradient(a, tau)

        # Adam moments
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)

        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)

        step = lr / (np.sqrt(v_hat) + eps)
        a = a - step * m_hat
        a = project_to_simplex(a)

        # cool the temperature slowly
        tau = max(tau * 0.999, 5e-7)

        # occasional evaluation of the true max‑convolution score
        if t % 1500 == 0:
            sc = compute_score(a)
            if sc < best_sc:
                best_sc = sc
                best_a = a.copy()
    return best_a, best_sc


def subgradient_optimize(initial_a: np.ndarray,
                         deadline: float,
                         max_iter: int = 200_000,
                         eta0: float = 0.7) -> Tuple[np.ndarray, float]:
    """
    Classic projected sub‑gradient descent on the exact max‑convolution.
    """
    a = project_to_simplex(initial_a.astype(float).copy())
    best_a = a.copy()
    best_sc = compute_score(a)

    n = len(a)
    it = 0
    while time.time() < deadline and it < max_iter:
        it += 1
        _, k_max = max_convolution_and_index(a)
        grad = subgradient_of_max_at_index(a, k_max)

        eta = eta0 / np.sqrt(it)
        a = a - eta * grad
        a = project_to_simplex(a)

        if it % 2000 == 0:
            sc = compute_score(a)
            if sc < best_sc:
                best_sc = sc
                best_a = a.copy()
    return best_a, best_sc

def propose_candidate(seed: int = 42,
                     budget_s: int = 1000,
                     **kwargs) -> List[float]:
    """
    Search for a non‑negative step‑function with a low ``evaluate_sequence`` value.
    The routine respects the time budget, explores a few lengths, refines the
    best candidate with an Adam optimiser on a smoothed max, and finally
    performs a projected sub‑gradient polishing.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 3.0   # safety margin

    best_seq: List[float] = []
    best_val: float = float("inf")

    # ------------------------------------------------------------------
    # 0.  Use the global fallback if it is available.
    # ------------------------------------------------------------------
    try:
        fallback = [float(x) for x in height_sequence_1]  # type: ignore
        val = evaluate_sequence(fallback)
        print(f"Fallback sequence score = {val:.6f}")
        if val < best_val:
            best_val = val
            best_seq = fallback
    except NameError:
        pass

    # ------------------------------------------------------------------
    # 1.  Scan a few lengths and pick a promising initial shape.
    # ------------------------------------------------------------------
    lengths = [512, 1024, 2048, 4096]
    global_best_a = None
    global_best_sc = float("inf")
    global_best_n = None

    for n in lengths:
        if time.time() >= deadline:
            break
        print(f"\n--- Length {n} ---")
        # deterministic starters
        starters = [np.full(n, 1.0 / n), arcsine_initial(n)]
        local_best_a = None
        local_best_sc = float("inf")
        for s in starters:
            sc = compute_score(s)
            if sc < local_best_sc:
                local_best_sc = sc
                local_best_a = s

        # random sampling in the (p, eps) family
        max_samples = 1500
        for _ in range(max_samples):
            if time.time() >= deadline:
                break
            p = rng.uniform(0.2, 2.0)               # broadened range
            eps = rng.uniform(1e-4, max(3.0, 0.02 * n))
            cand = generate_shape(n, p, eps)
            sc = compute_score(cand)
            if sc < local_best_sc:
                local_best_sc = sc
                local_best_a = cand

        # keep the best across lengths
        if local_best_a is not None and local_best_sc < global_best_sc:
            global_best_sc = local_best_sc
            global_best_a = local_best_a.copy()
            global_best_n = n
            print(f"New best across lengths: score = {global_best_sc:.6f}")

    # ------------------------------------------------------------------
    # 2.  Refine the globally best shape (if any) with Adam + sub‑gradient.
    # ------------------------------------------------------------------
    if global_best_a is not None and time.time() < deadline - 5:
        n = global_best_n
        print(f"\nRefining best shape of length {n} with Adam")
        # allocate ~60 % of the remaining budget to Adam, the rest to sub‑grad
        remaining = deadline - time.time()
        adam_deadline = time.time() + remaining * 0.58
        subgrad_deadline = deadline

        a_adam, sc_adam = adam_smooth_optimize(global_best_a, adam_deadline)
        val = evaluate_sequence(a_adam.tolist())
        print(f"Score after Adam (fast eval) = {val:.6f}")
        if val < best_val:
            best_val = val
            best_seq = a_adam.tolist()
            print("*** New global best after Adam ***")

        # final fine‑grained sub‑gradient polishing
        if time.time() < subgrad_deadline:
            print("Starting sub‑gradient polishing")
            a_sub, sc_sub = subgradient_optimize(a_adam, subgrad_deadline,
                                                 eta0=0.6)
            val = evaluate_sequence(a_sub.tolist())
            print(f"Score after polishing = {val:.6f}")
            if val < best_val:
                best_val = val
                best_seq = a_sub.tolist()
                print("*** New global best after polishing ***")

    # ------------------------------------------------------------------
    # 3.  Final safety polish on the current best (if time permits).
    # ------------------------------------------------------------------
    if best_seq and time.time() < deadline - 2:
        a = np.array(best_seq, dtype=float)
        remaining = deadline - time.time()
        print("\nFinal safety polish (Adam only)")
        a_final, _ = adam_smooth_optimize(a, time.time() + remaining)
        final_val = evaluate_sequence(a_final.tolist())
        print(f"Final polished score = {final_val:.6f}")
        if final_val < best_val:
            best_val = final_val
            best_seq = a_final.tolist()
            print("*** Updated global best after final polish ***")

    # ------------------------------------------------------------------
    # 4.  Guard – always return something.
    # ------------------------------------------------------------------
    if not best_seq:
        best_seq = [1.0] * 256
        best_val = evaluate_sequence(best_seq)

    print(f"\nSearch finished. Best reported score = {best_val:.6f}")
    return best_seq
