"""
Smooth‑max projected gradient descent for the self‑convolution L∞‑norm.

The evaluation function (with sum(a)=1) reduces to   2·n·max(a*a).
We keep the coefficients on the probability simplex (non‑negative, sum = 1)
and minimise a differentiable approximation of the max:
    f_α(a) = (1/α)·log Σ_k exp(α·(a*a)_k) .
Its gradient is
    ∇f_α = 2·corr(p , a)      with  p_k = exp(α·b_k)/Σ exp(α·b)
where b = a*a (full convolution) and corr denotes the linear correlation.
The algorithm repeatedly:
  * evaluates b and the true score,
  * computes the smoothed gradient,
  * takes a projected gradient step (with a decaying step size),
  * occasionally injects random jitter,
  * slowly raises the temperature parameter α so that the smooth max
    becomes the true max.
The best sequence seen within the allocated time budget is returned.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
def next_pow2(x: int) -> int:
    """Return the smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


# ----------------------------------------------------------------------
def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the probability simplex:
    non‑negative entries that sum to 1.
    """
    v = np.maximum(v, 0.0)
    s = v.sum()
    if s == 0.0:
        # fallback to the uniform distribution
        return np.full_like(v, 1.0 / v.size)
    return v / s


# ----------------------------------------------------------------------
def binomial_sequence(length: int) -> np.ndarray:
    """
    Central‑binomial coefficients  c_k = C(2k,k)/4**k  for k=0…length‑1.
    They satisfy   c_0=1,
    c_{k+1}=c_k * ((2k+1)(2k+2))/(4·(k+1)²).
    """
    seq = np.empty(length, dtype=np.float64)
    seq[0] = 1.0
    for k in range(1, length):
        seq[k] = seq[k - 1] * ((2 * k - 1) * (2 * k)) / (4.0 * k * k)
    return seq


# ----------------------------------------------------------------------
def logsumexp(x: np.ndarray) -> float:
    """Stable log‑sum‑exp."""
    m = np.max(x)
    return m + np.log(np.sum(np.exp(x - m)))


# ----------------------------------------------------------------------
def conv_full_fft(a: np.ndarray, L: int) -> np.ndarray:
    """Full linear convolution a*a using an FFT of length L (real input)."""
    A = np.fft.rfft(a, L)
    B = A * A
    conv = np.fft.irfft(B, L)
    n = a.size
    return conv[: 2 * n - 1]  # keep only the true linear part


# ----------------------------------------------------------------------
def gradient_smoothed_max(a: np.ndarray, p: np.ndarray, L: int) -> np.ndarray:
    """
    Gradient of the smooth max (α‑log‑sum‑exp) w.r.t. a.
    p is the probability vector (weights) of length 2·n‑1, sum(p)=1.
    The formula is   ∇ = 2·corr(p , a) .
    """
    n = a.size
    a_rev = a[::-1]                     # reversed a
    A_rev = np.fft.rfft(a_rev, L)
    P = np.fft.rfft(p, L)
    corr = np.fft.irfft(P * A_rev, L)   # linear correlation (full length)
    start = n - 1
    grad = 2.0 * corr[start : start + n]  # slice to length n
    return grad


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function sequence that minimises
    evaluate_sequence().  The method is a smooth‑max projected gradient
    descent with an annealing schedule for the smoothing parameter α.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.1   # safety margin

    # --------------------------------------------------------------
    # Determine problem size and initialise a.
    try:
        # Use the previously discovered construction if it exists.
        init_seq = np.array(height_sequence_1, dtype=np.float64)  # type: ignore[name-defined]
        n = max(init_seq.size, 8192)          # allow a larger space if possible
        a = init_seq.astype(np.float64)
        # normalise to a probability vector
        a = a / a.sum()
        if n > a.size:                       # pad with zeros if we enlarged n
            a = np.pad(a, (0, n - a.size), constant_values=0.0)
    except Exception:
        # fallback: central‑binomial coefficients, generous length
        n = 16384
        a = binomial_sequence(n).astype(np.float64)
        a = a / a.sum()

    # --------------------------------------------------------------
    # Pre‑compute FFT length (must accommodate 3·n‑2 for the gradient step)
    L = next_pow2(3 * n)

    # Hyper‑parameters
    step_size = 0.05          # initial projected‑gradient step magnitude
    step_decay = 0.9995       # exponential decay of the step size
    min_step = 1e-6
    alpha = 4.0               # smoothing parameter (temperature)
    alpha_factor = 1.001      # multiplicative increase per iteration
    alpha_max = 2000.0

    # bookkeeping
    best_seq = a.copy()
    b = conv_full_fft(a, L)
    best_score = 2.0 * n * b.max()
    print(f"[init] n={n}, initial score={best_score:.9f}, alpha={alpha}")

    iter_cnt = 0
    last_improve = time.time()

    # --------------------------------------------------------------
    # Main optimisation loop
    while time.time() < deadline:
        # ----- current state -------------------------------------------------
        b = conv_full_fft(a, L)
        max_b = b.max()
        cur_score = 2.0 * n * max_b

        # ----- keep the best -------------------------------------------------
        if cur_score < best_score:
            best_score = cur_score
            best_seq = a.copy()
            elapsed = time.time() - (deadline - budget_s)
            print(f"[improve] iter={iter_cnt}, score={best_score:.9f}, time={elapsed:.2f}s")
            last_improve = time.time()

        # ----- smooth‑max and its gradient ----------------------------------
        # p_k = exp(α·b_k) / Σ exp(α·b)
        scaled = alpha * b
        logZ = logsumexp(scaled)
        p = np.exp(scaled - logZ)          # length 2·n‑1, sums to 1

        grad = gradient_smoothed_max(a, p, L)

        # ----- projected gradient step (with simple backtracking) ------------
        eta = step_size
        improved = False
        for _ in range(5):                     # at most 5 back‑track attempts
            a_candidate = a - eta * grad
            a_candidate = project_to_simplex(a_candidate)

            # evaluate candidate (cheap, using FFT)
            b_cand = conv_full_fft(a_candidate, L)
            cand_score = 2.0 * n * b_cand.max()
            if cand_score <= cur_score:
                a = a_candidate
                improved = True
                break
            eta *= 0.5                         # shrink step and retry

        # if none of the back‑tracks improved, keep the old a

        # ----- occasional random jitter (simulated‑annealing flavour) ------
        if np.random.rand() < 0.005:
            i, j = np.random.randint(0, n, size=2)
            if i != j and a[i] > 0.0:
                amount = 0.05 * np.random.rand() * a[i]
                a[i] -= amount
                a[j] += amount
                a = project_to_simplex(a)

        # ----- schedule updates ------------------------------------------------
        step_size = max(min_step, step_size * step_decay)
        alpha = min(alpha_max, alpha * alpha_factor)

        # ----- stagnation handling (re‑initialise around the best) ----------
        if time.time() - last_improve > 180 and (deadline - time.time()) > 200:
            # add a tiny Gaussian noise to the best sequence and resume
            a = best_seq.copy()
            noise = np.random.randn(n) * 0.001
            a = np.maximum(a + noise, 0.0)
            a = project_to_simplex(a)
            last_improve = time.time()
            print(f"[restart] iter={iter_cnt}, re‑initialised around best, alpha={alpha:.2f}")

        iter_cnt += 1

    # ----------------------------------------------------------------------
    print(f"[final] best score={best_score:.9f}, total iterations={iter_cnt}")
    # Return a plain Python list, as required by the external evaluator.
    return best_seq.tolist()
