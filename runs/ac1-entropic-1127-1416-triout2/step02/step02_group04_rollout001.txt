"""
Propose a candidate step‑function sequence.

The algorithm is a smooth, entropy‑regularised optimisation of the
max‑convolution objective.
It works on the probability simplex by parameterising the heights with a
soft‑max (log‑variables `x`).  For a given temperature `eps` the objective

    f_eps(a) = eps * logsumexp( (a*a) / eps )

is a smooth approximation of the original `max(a*a)`.  Its gradient can be
computed analytically using the convolution theorem, which we evaluate
with FFTs.  An Adam optimiser (projected automatically by the soft‑max)
updates `x`.  The temperature is decreased geometrically, driving the
solution towards a true minimiser of the max‑convolution.

The method starts from a collection of sensible seeds (the previous best
solution, a linear ramp, several Beta‑shaped and Dirichlet samples) and
runs the optimiser while respecting the 1000 s time budget.  The best
sequence seen (evaluated with the exact objective `2·n·max(conv)`) is
returned as a plain list of non‑negative floats.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper functions (same as in the previous baseline)
# ----------------------------------------------------------------------
def simplex_projection(v):
    """Euclidean projection of a vector onto the probability simplex."""
    v = np.asarray(v, dtype=float)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1] - 1
    theta = cssv[cond][-1] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def compute_objective(a):
    """Exact objective 2·n·max(a*a) for a non‑negative vector a with sum=1."""
    n = a.shape[0]
    fft = np.fft.rfft(a, n=2 * n - 1)
    conv = np.fft.irfft(fft * fft, n=2 * n - 1)
    return float(2.0 * n * conv.max())


def beta_sequence(alpha, beta, n):
    """Discretised Beta(α,β) distribution on n points, normalised to sum = 1."""
    x = (np.arange(n) + 0.5) / n
    logw = (alpha - 1.0) * np.log(x) + (beta - 1.0) * np.log(1.0 - x)
    logw = np.clip(logw, -700.0, 700.0)     # avoid overflow in exp()
    w = np.exp(logw)
    total = w.sum()
    if total <= 0 or not np.isfinite(total):
        w = np.ones_like(w)
        total = w.sum()
    return w / total


# ----------------------------------------------------------------------
# Main search routine
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: int = 1000):
    """Search for a non‑negative sequence that minimises the evaluation
    function.  Returns the best sequence found as a list of floats."""
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 5.0   # small safety margin

    # ------------------------------------------------------------------
    # Determine problem size
    # ------------------------------------------------------------------
    try:
        n = len(height_sequence_1)   # type: ignore[name-defined]
    except NameError:
        n = 5146                     # fallback length

    # ------------------------------------------------------------------
    # Gather a few deterministic / random seeds
    # ------------------------------------------------------------------
    best_a = None
    best_score = np.inf

    # 1) Ramp (the classic construction)
    ramp = np.arange(n, 0, -1, dtype=float)
    ramp = 2.0 * ramp / (n * (n + 1))
    ramp /= ramp.sum()
    score = compute_objective(ramp)
    best_a, best_score = ramp.copy(), score
    print(f"Ramp seed: score = {score:.6f}")

    # 2) Uniform
    uniform = np.full(n, 1.0 / n)
    score = compute_objective(uniform)
    if score < best_score:
        best_a, best_score = uniform.copy(), score
        print(f"Uniform improved to {score:.6f}")

    # 3) Previous global best, if available
    try:
        prev = np.asarray(height_sequence_1, dtype=float)
        prev = np.maximum(prev, 0.0)
        if prev.sum() > 0:
            prev = prev / prev.sum()
            s = compute_objective(prev)
            if s < best_score:
                best_a, best_score = prev.copy(), s
                print(f"Prev global best improved to {s:.6f}")
    except NameError:
        pass

    # 4) Systematic Beta family
    alphas = np.linspace(0.2, 5.0, 11)
    betas = np.linspace(0.2, 5.0, 11)
    for a in alphas:
        for b in betas:
            if time.time() > deadline:
                break
            cand = beta_sequence(a, b, n)
            s = compute_objective(cand)
            if s < best_score:
                best_a, best_score = cand.copy(), s
                print(f"Beta({a:.2f},{b:.2f}) -> {s:.6f}")
        if time.time() > deadline:
            break

    # 5) Random Beta samples
    for _ in range(150):
        if time.time() > deadline:
            break
        a = np.random.uniform(0.05, 5.0)
        b = np.random.uniform(0.05, 5.0)
        cand = beta_sequence(a, b, n)
        s = compute_objective(cand)
        if s < best_score:
            best_a, best_score = cand.copy(), s
            print(f"Rnd Beta({a:.3f},{b:.3f}) -> {s:.6f}")

    # 6) Random Dirichlet‑like vectors
    for _ in range(12):
        if time.time() > deadline:
            break
        conc = np.random.uniform(0.1, 3.0, size=n)
        cand = np.random.gamma(conc, 1.0)
        cand /= cand.sum()
        s = compute_objective(cand)
        if s < best_score:
            best_a, best_score = cand.copy(), s
            print(f"Rnd Dirichlet improved to {s:.6f}")

    # ------------------------------------------------------------------
    # Smooth optimisation with entropy regularisation (soft‑max + Adam)
    # ------------------------------------------------------------------
    # Hyper‑parameters
    max_iters_total = 25000               # hard cap (actual run will stop earlier)
    base_lr = 0.25                         # initial learning rate
    beta1, beta2 = 0.9, 0.999
    eps_adam = 1e-8
    temperature_initial_factor = 0.5      # eps = factor * current max_conv
    temperature_decay = 0.7                # geometric decay every 800 iters
    decay_interval = 800

    # Keep track of elapsed iterations for the Adam bias corrections
    global_iter = 0

    while time.time() < deadline and global_iter < max_iters_total:
        # ------------------------------------------------------------------
        # Initialise from the current best (with a tiny random perturbation)
        # ------------------------------------------------------------------
        a0 = best_a.copy()
        # Add a small noise to avoid being stuck exactly at a local plateau
        a0 += np.random.normal(scale=1e-4, size=n)
        a0 = np.maximum(a0, 0.0)
        if a0.sum() == 0:
            a0 = np.full(n, 1.0 / n)
        else:
            a0 /= a0.sum()

        # Log‑variables for soft‑max parametrisation
        x = np.log(a0 + 1e-30)           # small offset to stay in domain
        m = np.zeros_like(x)
        v = np.zeros_like(x)

        # Temperature schedule starts from a fraction of the initial max‑conv
        # (we will recompute it after the first convolution).
        temperature = None

        # ------------------------------------------------------------------
        # Inner optimisation loop
        # ------------------------------------------------------------------
        it = 0
        while time.time() < deadline and global_iter < max_iters_total:
            # ----- Soft‑max projection (ensures non‑negativity & sum=1) -----
            max_x = np.max(x)
            exp_x = np.exp(x - max_x)
            a = exp_x / exp_x.sum()

            # ----- Convolution a*a (linear, length 2n‑1) -----------------
            fft_a = np.fft.rfft(a, n=2 * n - 1)
            conv = np.fft.irfft(fft_a * fft_a, n=2 * n - 1)
            max_conv = conv.max()

            # ----- Record improvement ------------------------------------
            cur_score = 2.0 * n * max_conv
            if cur_score < best_score:
                best_score = cur_score
                best_a = a.copy()
                print(f"Iter {global_iter:5d}: new best {best_score:.6f}")

            # ----- Initialise / update temperature -----------------------
            if temperature is None:                         # first iteration
                temperature = max_conv * temperature_initial_factor
            elif it % decay_interval == 0:
                temperature = max(temperature * temperature_decay, 1e-12)

            # ----- Compute soft weights w = softmax(conv/temperature) -----
            # Use shift = conv - max_conv for numerical stability.
            shift = conv - max_conv
            w = np.exp(shift / temperature)
            w_sum = w.sum()
            if w_sum == 0 or not np.isfinite(w_sum):
                w = np.full_like(w, 1.0 / w.size)
            else:
                w /= w_sum

            # ----- Gradient of smooth max w.r.t. a -----------------------
            # g_a_i = 2 * sum_k w_k * a_{k-i}  (linear correlation)
            L = 3 * n - 2                     # length for linear correlation
            w_fft = np.fft.rfft(w, n=L)
            a_fft = np.fft.rfft(a, n=L)
            corr_full = np.fft.irfft(w_fft * np.conj(a_fft), n=L)
            # indices (n‑1) … (n‑1+n‑1) correspond to lags 0 … n‑1
            grad_a = 2.0 * corr_full[(n - 1):(n - 1 + n)]

            # ----- Chain rule through soft‑max ---------------------------
            avg = np.dot(a, grad_a)          # a·grad_a scalar
            grad_x = a * (grad_a - avg)      # ∂f/∂x

            # ----- Adam update -----------------------------------------
            m = beta1 * m + (1.0 - beta1) * grad_x
            v = beta2 * v + (1.0 - beta2) * (grad_x ** 2)
            it += 1
            global_iter += 1
            m_hat = m / (1.0 - beta1 ** it)
            v_hat = v / (1.0 - beta2 ** it)

            # Diminishing learning‑rate factor
            lr = base_lr / (np.sqrt(it) + 1.0)

            # Parameter step (soft‑max automatically projects)
            x = x - lr * m_hat / (np.sqrt(v_hat) + eps_adam)

            # Occasionally inject a little noise to help escape plateaus
            if it % 1200 == 0:
                x += np.random.normal(scale=0.02, size=n)

        # End of inner optimisation for one restart
        # Loop back to start a fresh restart if time permits
    # ------------------------------------------------------------------
    # Return the best sequence found (as a plain Python list)
    # ------------------------------------------------------------------
    print(f"Finished after {global_iter} total iterations, best score = {best_score:.6f}")
    # The evaluation function is scale‑invariant, but keep sum≈1 for safety.
    return [float(v) for v in best_a.tolist()]
