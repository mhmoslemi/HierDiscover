"""
Propose a step‑function height sequence that (heuristically) minimises

        2 * n * max(a * a) / (sum(a)**2)

where “*” is the linear convolution.  The algorithm is a
projected exponentiated‑gradient descent with a soft‑max
approximation of the max‑operator and a simple simulated‑annealing
perturbation.  Convolutions and correlations are performed with FFTs
so that even a few‑‑ten‑thousand‑length vector can be updated many
times within the 1000 s budget.

Key ideas
---------
* Initialise from the best known construction (`height_sequence_1`) if it
  exists, otherwise fall back to the classic binomial sequence.
* Normalise the vector to unit L¹‑norm – the objective is scale invariant.
* At each iteration:
    - Compute the full self‑convolution `b = a * a`.
    - Form a soft‑max weight vector `w` that concentrates on the largest
      convolution entries (parameter `beta` is increased slowly).
    - The gradient of the smoothed maximum is the correlation
      `g = correlate(w, a)`.  Larger `g[i]` means that `a[i]` contributes
      heavily to the peaks, so we reduce it.
    - Update with an exponentiated‑gradient step
          a ← a ⊙ exp(step * v)   (⊙ is element‑wise product)
      where `v` is a momentum accumulator of the negative gradient.
    - Re‑normalise onto the probability simplex.
    - Occasionally perform a random mass‑transfer to help escape local
      minima (simulated annealing).
* Keep the best sequence ever seen and return it when the time budget
  expires.

The whole routine runs in O(n log n) per iteration thanks to the FFT
based convolution/correlation, allowing tens of thousands of updates
within the time limit.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
def binomial_sequence(length: int) -> np.ndarray:
    """Return the first *length* central‑binomial coefficients
    c_k = C(2k, k) / 4**k  (k = 0 … length‑1)."""
    seq = np.empty(length, dtype=np.float64)
    seq[0] = 1.0
    for k in range(1, length):
        seq[k] = seq[k - 1] * ((2 * k - 1) * (2 * k)) / (4.0 * k * k)
    return seq


# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Small helper – smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


# ----------------------------------------------------------------------
def _conv_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Linear convolution via FFT (real‑valued inputs)."""
    size = a.size + b.size - 1
    nfft = _next_pow2(size)
    A = np.fft.rfft(a, n=nfft)
    B = np.fft.rfft(b, n=nfft)
    c = np.fft.irfft(A * B, n=nfft)[:size]
    return c


# ----------------------------------------------------------------------
def _correlate_valid(w: np.ndarray, a: np.ndarray) -> np.ndarray:
    """
    Valid (linear) correlation of w with a.
    Returns length‑len(a) + len(w) - 1 – (len(a)-1) = len(w)-len(a)+1 = len(a)
    values exactly matching   g_i = Σ_j w_{j+i} a_j .
    """
    a_rev = a[::-1]
    conv = _conv_fft(w, a_rev)
    start = a.size - 1
    end = w.size
    return conv[start:end]


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Heuristic optimiser for the step‑function “height” vector.
    Returns the best sequence found before the time budget runs out.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.1      # safety margin

    # ------------------------------------------------------------------
    # 1️⃣  Choose a length and an initial vector.
    try:
        base = np.asarray(height_sequence_1, dtype=np.float64)   # type: ignore[name-defined]
        n_base = base.shape[0]
    except Exception:
        base = None
        n_base = 0

    # target length – a few‑thousand entries give a good trade‑off.
    target_n = max(8000, n_base)

    if base is not None:
        # pad the known construction with a tiny uniform tail (so the sum stays 1)
        a = np.zeros(target_n, dtype=np.float64)
        a[:n_base] = np.maximum(base, 0.0)
        if target_n > n_base:
            tail = np.random.rand(target_n - n_base) * 1e-6
            a[n_base:] = tail
    else:
        a = binomial_sequence(target_n)

    # protect against accidental negatives and normalise to unit L¹.
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s == 0.0 or np.isnan(s):
        a = np.full(target_n, 1.0 / target_n)
    else:
        a /= s

    n = a.size

    # ------------------------------------------------------------------
    # 2️⃣  Helper to compute the objective (sum(a) == 1 ⇒ 2·n·max(b)).
    def _objective(vec: np.ndarray) -> float:
        b = _conv_fft(vec, vec)
        return 2.0 * n * b.max()

    # initial score
    best_score = _objective(a)
    best_seq = a.copy()
    print(f"[init] n={n}, initial score={best_score:.9f}")

    # ------------------------------------------------------------------
    # 3️⃣  Optimisation hyper‑parameters
    step = 0.85                     # initial exponentiated‑gradient step
    step_decay = 0.9999
    min_step = 1e-6

    beta_start, beta_end = 5.0, 50.0      # soft‑max “temperature” schedule
    momentum_decay = 0.90
    v = np.zeros_like(a)                  # momentum accumulator

    iter_cnt = 0
    last_improve = start_time

    # ------------------------------------------------------------------
    # 4️⃣  Main optimisation loop
    while time.time() < deadline:
        # ----- current convolution & score
        b = _conv_fft(a, a)
        max_b = b.max()
        cur_score = 2.0 * n * max_b

        if cur_score < best_score:
            best_score = cur_score
            best_seq = a.copy()
            print(f"[improve] iter={iter_cnt}, score={best_score:.9f}, "
                  f"time={time.time() - start_time:.2f}s")
            last_improve = time.time()

        # ----- soft‑max mask over convolution entries
        elapsed = time.time() - start_time
        beta = beta_start + (beta_end - beta_start) * (elapsed / budget_s)

        # `b` is non‑negative; shift by its maximum to keep exponents ≤ 0.
        delta = b - max_b
        w = np.exp(beta * delta)                 # w[max] = 1
        w_sum = w.sum()
        if w_sum == 0.0 or np.isnan(w_sum):
            # fall back to a hard max mask (should rarely happen)
            w = np.zeros_like(b)
            w[np.argmax(b)] = 1.0
            w_sum = 1.0
        w /= w_sum

        # ----- gradient of the (smoothed) max via correlation
        g = _correlate_valid(w, a)               # shape (n,)

        # ----- momentum‑augmented exponentiated‑gradient step
        v = momentum_decay * v - (1.0 - momentum_decay) * g
        a_candidate = a * np.exp(step * v)       # v ≤ 0 for large g → reduction
        a_candidate = np.maximum(a_candidate, 0.0)

        # ----- projector onto the probability simplex
        s = a_candidate.sum()
        if s == 0.0 or np.isnan(s):
            a_candidate = np.full_like(a, 1.0 / n)
        else:
            a_candidate /= s

        # ----- occasional random mass‑transfer (simulated annealing)
        if np.random.rand() < 0.002:   # ≈ once per 500 iterations
            i, j = np.random.randint(0, n, size=2)
            if i != j and a_candidate[i] > 0.0:
                amount = 0.01 * np.random.rand() * a_candidate[i]   # up‑to 1 % move
                a_candidate[i] -= amount
                a_candidate[j] += amount
                a_candidate /= a_candidate.sum()

        # ----- evaluate candidate (full convolution)
        b_cand = _conv_fft(a_candidate, a_candidate)
        cand_score = 2.0 * n * b_cand.max()

        # ----- Metropolis‑style acceptance
        accept = False
        if cand_score < cur_score:
            accept = True
        else:
            # temperature that slowly cools – we use a tiny scale factor
            temp = 0.001 * cur_score
            if np.random.rand() < np.exp(-(cand_score - cur_score) / temp):
                accept = True

        if accept:
            a = a_candidate
            # update best if the accepted move is also a new record
            if cand_score < best_score:
                best_score = cand_score
                best_seq = a.copy()
                print(f"[improve] iter={iter_cnt} (accept), score={best_score:.9f}, "
                      f"time={time.time() - start_time:.2f}s")

        # ----- step‑size decay
        step = max(min_step, step * step_decay)

        iter_cnt += 1

    # ------------------------------------------------------------------
    print(f"[final] best score={best_score:.9f}, iterations={iter_cnt}")
    # The evaluation routine caps values to [0,1000]; our sequence is far
    # below that bound, hence safe to return directly.
    return best_seq.tolist()
