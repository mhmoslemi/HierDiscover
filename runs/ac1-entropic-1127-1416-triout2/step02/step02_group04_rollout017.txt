    n = len(a)
    conv = np.convolve(a, a)  # length 2n-1
    eps = 1e-300

    w = conv_safe ** (p - 1)

    L = conv.shape[0]

    idx2 = 2 * np.arange(n)
    mask = idx2 < L
    g_temp[mask] -= a[mask] * w[idx2[mask]]
    g = factor * g_temp
    return conv, g

"""
Propose a step‑height sequence that (approximately) minimizes

    2·n·max(conv(a,a)) / (sum(a)^2)

The function is scale‑invariant; we therefore keep the sequence on the
probability simplex (non‑negative, sum = 1).  The search proceeds in three
stages:

1. **Diverse initial seeds** – uniform, linear ramp, a family of
   exponential shapes a_i ∝ r^i and several Beta‑distributions.  The best
   among them (according to the exact objective 2·n·max(conv)) is kept.

2. **Multiplicative sub‑gradient descent** – using the (almost‑active)
   convolution entries we form a sub‑gradient of the max‑convolution.
   Updates are performed in the log‑space (exponential‑gradient /
   mirror‑descent on the simplex) with a diminishing step size.

3. **Smoothed Lp‑gradient descent (continuation)** – we minimise the
   Lp‑norm of the convolution with p gradually increased from 2 up to
   30.  An Adam optimiser is used and after each step the iterate is
   projected back onto the simplex via Euclidean projection.

Both stages keep track of the best sequence seen so far.  The whole
procedure stops a few seconds before the 1000 s budget and returns the
best list of floats.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities
# ----------------------------------------------------------------------


def simplex_projection(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of a vector onto the probability simplex."""
    v = np.asarray(v, dtype=float)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all non‑positive → uniform distribution
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1] - 1                # zero‑based index
    theta = cssv[cond][-1] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def compute_objective(a: np.ndarray) -> float:
    """Exact evaluation (2·n·max(conv)) for a vector that already sums to 1."""
    n = a.shape[0]
    conv = np.convolve(a, a)               # linear convolution, length 2·n‑1
    return 2.0 * n * conv.max()


def beta_sequence(alpha: float, beta: float, n: int) -> np.ndarray:
    """Discretised Beta(α,β) on n interior points, normalised to sum = 1."""
    x = (np.arange(n) + 0.5) / n
    logw = (alpha - 1.0) * np.log(x) + (beta - 1.0) * np.log(1.0 - x)
    logw = np.clip(logw, -700.0, 700.0)    # avoid overflow in exp
    w = np.exp(logw)
    if not np.isfinite(w).all() or w.sum() == 0.0:
        w = np.ones_like(w)
    return w / w.sum()


def exponential_sequence(r: float, n: int) -> np.ndarray:
    """Geometric decay a_i ∝ r^i, normalised to sum = 1."""
    if r <= 0.0:
        return np.full(n, 1.0 / n)
    a = r ** np.arange(n, dtype=float)
    s = a.sum()
    if s == 0.0 or not np.isfinite(s):
        a = np.ones(n, dtype=float)
        s = a.sum()
    return a / s


# ----------------------------------------------------------------------
# Sub‑gradient for the max‑convolution (active set)
# ----------------------------------------------------------------------


def max_conv_subgradient(a: np.ndarray, eps_factor: float = 1e-5) -> tuple:
    """
    Return (conv, g) where conv = a*a (full convolution) and g is a
    sub‑gradient of max(conv) with respect to a (averaged over all
    constraints that are within eps_factor of the maximum).
    """
    n = a.shape[0]
    conv = np.convolve(a, a)                     # length 2n‑1
    max_val = conv.max()
    eps = max_val * eps_factor
    active = np.where(conv >= max_val - eps)[0]

    # sub‑gradient accumulation (exact, O(|active|·n) – tiny in practice)
    g = np.zeros_like(a)
    for k in active:
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        idx = np.arange(lo, hi + 1, dtype=int)
        partner = k - idx
        g[idx] += 2.0 * a[partner]

    # diagonal correction: when k = 2*i we added 2*a_i but only a_i is correct
    diag_idx = 2 * np.arange(n)
    diag_mask = diag_idx < conv.shape[0]
    g[diag_mask] -= a[diag_mask]
    return conv, g


# ----------------------------------------------------------------------
# Smoothed Lp‑gradient (continuation) utilities
# ----------------------------------------------------------------------


def lp_gradient(a: np.ndarray, p: float) -> tuple:
    """
    Compute (conv, g) where conv = a*a and g = ∇‖conv‖_p (the gradient of the
    Lp‑norm of the convolution vector).  The gradient is scaled exactly
    as d/d a ‖conv‖_p = (∑ conv^p)^{1/p‑1} * Σ conv^{p‑1} ∂conv/∂a .
    """
    n = a.shape[0]
    conv = np.convolve(a, a)                     # length L = 2n‑1
    eps = 1e-300
    conv_pos = np.maximum(conv, eps)
    # Lp‑norm related terms
    conv_p = conv_pos ** p
    sum_pow = conv_p.sum()
    if sum_pow <= 0.0:
        sum_pow = eps
    factor = sum_pow ** ((1.0 - p) / p)           # (∑ conv^p)^{1/p‑1}
    w = conv_pos ** (p - 1)                      # length L

    # Cross‑correlation w ⋆ a  (g_temp[i] = Σ_k w_k a_{k‑i})
    L = conv.shape[0]
    N = 1 << ((n + L - 1).bit_length())          # next power of two
    a_fft = np.fft.rfft(a, N)
    w_fft = np.fft.rfft(w, N)
    cross = np.fft.irfft(a_fft * np.conj(w_fft), N)   # linear cross‑corr
    g_temp = 2.0 * cross[:n]                     # 2·Σ_k w_k a_{k‑i}
    # diagonal correction (k = 2·i case)
    diag_idx = 2 * np.arange(n)
    diag_mask = diag_idx < L
    g_temp[diag_mask] -= a[diag_mask] * w[diag_idx[diag_mask]]
    g = factor * g_temp
    return conv, g


# ----------------------------------------------------------------------
# Main search routine
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 0, budget_s: int = 1000) -> list:
    """
    Search for a non‑negative sequence of step heights that minimises the
    evaluation function.  The function respects a 1000 s time budget and
    returns the best sequence found.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 5.0   # a small safety margin

    # --------------------------------------------------------------
    # Problem size
    # --------------------------------------------------------------
    try:
        n = len(height_sequence_1)   # type: ignore[name-defined]
    except Exception:
        n = 5146                     # fallback used in the challenge

    # --------------------------------------------------------------
    # 1. Build a collection of promising seeds
    # --------------------------------------------------------------
    best_a = None
    best_score = np.inf

    def maybe_update(a_candidate: np.ndarray, label: str):
        nonlocal best_a, best_score
        score = compute_objective(a_candidate)
        if score < best_score:
            best_a = a_candidate.copy()
            best_score = score
            print(f"{label:20s} => score {score:.6f}")

    # Uniform
    maybe_update(np.full(n, 1.0 / n), "uniform")

    # Linear decreasing ramp (the classic analytic construction)
    ramp = np.arange(n, 0, -1, dtype=float)
    ramp = ramp / ramp.sum()
    maybe_update(ramp, "linear ramp")

    # Exponential family
    for r in np.linspace(0.5, 0.9999, 12):
        maybe_update(exponential_sequence(r, n), f"exp(r={r:.4f})")

    # Beta‑distribution grid
    alphas = np.linspace(0.2, 5.0, 9)
    betas = np.linspace(0.2, 5.0, 9)
    for a_ in alphas:
        for b_ in betas:
            maybe_update(beta_sequence(a_, b_, n), f"beta({a_:.2f},{b_:.2f})")
            if time.time() >= deadline:
                break
        if time.time() >= deadline:
            break

    # Random Dirichlet‑like seeds
    for _ in range(10):
        if time.time() >= deadline:
            break
        conc = np.random.uniform(0.1, 3.0, size=n)
        a_rand = np.random.gamma(conc, 1.0)
        a_rand /= a_rand.sum()
        maybe_update(a_rand, "random dirichlet")

    if best_a is None:                     # should never happen
        best_a = np.full(n, 1.0 / n)
        best_score = compute_objective(best_a)

    # --------------------------------------------------------------
    # 2. Phase‑1 : multiplicative sub‑gradient descent (active set)
    # --------------------------------------------------------------
    a = best_a.copy()
    eta0 = 0.8                              # initial step size for log‑update
    it_phase1 = 0
    max_phase1_iters = 4000

    while time.time() < deadline and it_phase1 < max_phase1_iters:
        conv, g = max_conv_subgradient(a, eps_factor=1e-5)

        # Diminishing step size in log‑space (mirror descent on simplex)
        it_phase1 += 1
        eta = eta0 / np.sqrt(it_phase1)
        # exponential‑gradient update (log‑space to avoid overflow)
        log_a = np.log(a + 1e-30)
        log_a -= eta * g
        a = np.exp(log_a)
        a /= a.sum()                         # back to simplex

        # occasional exact evaluation
        if it_phase1 % 100 == 0:
            cur = compute_objective(a)
            if cur < best_score:
                best_score = cur
                best_a = a.copy()
                print(f"Phase‑1 iter {it_phase1:5d}: new best {best_score:.6f}")

    # --------------------------------------------------------------
    # 3. Phase‑2 : smoothed Lp‑gradient descent with Adam (continuation)
    # --------------------------------------------------------------
    # Starting point = best from phase‑1 (or original best)
    a = best_a.copy()
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1, beta2 = 0.9, 0.999
    adam_eps = 1e-8
    lr = 0.12                                # base learning rate for Adam
    it_phase2 = 0
    max_phase2_iters = 15000
    p = 2.0                                   # start with L2, increase gradually
    p_target = 30.0
    p_inc_interval = 500                      # increase p every this many iters

    while time.time() < deadline and it_phase2 < max_phase2_iters:
        it_phase2 += 1

        conv, g = lp_gradient(a, p)

        # Adam moments
        m = beta1 * m + (1.0 - beta1) * g
        v = beta2 * v + (1.0 - beta2) * (g ** 2)
        m_hat = m / (1.0 - beta1 ** it_phase2)
        v_hat = v / (1.0 - beta2 ** it_phase2)

        step = lr * m_hat / (np.sqrt(v_hat) + adam_eps)
        a = a - step
        a = simplex_projection(a)            # ensure feasibility

        # Continuation on p
        if it_phase2 % p_inc_interval == 0 and p < p_target:
            p = min(p + 2.0, p_target)
            print(f"Phase‑2 iter {it_phase2:5d}: increased p to {p}")

        # Periodic exact check
        if it_phase2 % 100 == 0:
            cur = 2.0 * n * conv.max()
            if cur < best_score:
                best_score = cur
                best_a = a.copy()
                print(f"Phase‑2 iter {it_phase2:5d}: new best {best_score:.6f}")

    # --------------------------------------------------------------
    # 4. Return the best sequence found
    # --------------------------------------------------------------
    print(f"Finished after {it_phase1 + it_phase2} total iterations, best score = {best_score:.6f}")
    # Convert to plain Python list of floats
    return [float(v) for v in best_a.tolist()]
