def solve_lp_for_g(f: np.ndarray, M: float) -> np.ndarray:
    """
    Solve linear program:
        max sum(g)
        s.t. conv(f, g) <= M    (elementwise)
             g >= 0
    Returns optimal g as np.ndarray (may be None if LP fails).
    """
    n = len(f)
    rows = 2 * n - 1
    # Build A_ub
    A = np.zeros((rows, n), dtype=np.float64)
    # Each column i fills rows i..i+n-1 with f_i (since conv(f,g)_k includes f_i * g_{k-i})
    for i in range(n):
        A[i:i+n, i] = f  # runs within bounds
    # objective: maximize sum(g) => minimize -c^T g, where c = -ones
    c = -np.ones(n, dtype=np.float64)
    b_ub = np.full(rows, M, dtype=np.float64)
    # variable bounds: (0, None)
    bounds = [(0, None)] * n
    res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs', options={'presolve': True})
    if res.success:
        return res.x
    else:
        # fall back: return current f (or uniform) as best guess
        return f.copy()

"""
Hybrid LP‑subgradient optimizer for the step‑function problem.

The algorithm works in three stages:

1. **Parametric shape scan** –  a two‑parameter symmetric family
   a_i ∝ ((i+ε)*(n‑1‑i+ε))⁻ᵖ  (p>0, ε>0) is sampled densely
   (p up to 3, ε down to 1e‑6).  The best shape for each length n
   (256 … 4096) is kept.

2. **LP‑based mixing** –  Starting from the best shape, we repeatedly
   solve a linear program

       max   Σ g_j
       s.t.  (f * g)_k ≤ M   ∀k ,   g ≥ 0 ,

   where M = max(f*f).  The solution g is projected back onto the
   probability simplex and blended with the current f:
       f ← (1−τ)·f + τ·g .
   This step often produces a large drop in the maximal self‑convolution.

3. **Projected sub‑gradient polishing** –  After a few LP updates we
   run a long projected sub‑gradient (or Adam‑style) descent on the
   non‑smooth objective  max(f*f)  while staying on the simplex.
   The final vector is evaluated with the official ``evaluate_sequence``
   and the best overall answer is returned.

The whole routine respects the given time budget (≈1000 s) and falls
back to a trivial uniform vector if no candidate is found.
"""

import time
import numpy as np
from typing import List, Tuple
from scipy.optimize import linprog

# ----------------------------------------------------------------------
# Low‑level utilities (no closures)
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Convolution via real FFT (fast, zero‑padding to next power of two)."""
    m = len(a) + len(b) - 1
    n_fft = next_pow2(m)
    A = np.fft.rfft(a, n=n_fft)
    B = np.fft.rfft(b, n=n_fft)
    C = A * B
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


def max_convolution_and_index(a: np.ndarray) -> Tuple[float, int]:
    """Return (max value, arg‑max index) of a*a."""
    conv = convolution_fft(a, a)
    idx = int(np.argmax(conv))
    return float(conv[idx]), idx


def subgradient_of_max_at_index(a: np.ndarray, k: int) -> np.ndarray:
    """Sub‑gradient of the maximal convolution entry (index k) w.r.t. a."""
    n = len(a)
    grad = np.zeros_like(a)
    i = np.arange(n)
    j = k - i
    mask = (j >= 0) & (j < n)
    grad[mask] = a[j[mask]]
    # double the diagonal term when k = 2·i
    if k % 2 == 0:
        diag = k // 2
        if diag < n:
            grad[diag] = 2.0 * a[diag]
    return grad


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection onto the probability simplex {x ≥ 0, Σx = 1}.
    Implementation of Duchi et al. (2008).
    """
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / len(v))

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho_arr = u - cssv / np.arange(1, len(v) + 1)
    rho = np.where(rho_arr > 0)[0]
    if len(rho) == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def compute_score(a: np.ndarray) -> float:
    """
    Internal (fast) version of the official objective.
    The vector is assumed to be a probability vector (sum≈1).
    Returns 2·n·max(a*a)  (the denominator Σa² = 1 after normalisation).
    """
    n = len(a)
    conv = convolution_fft(a, a)
    max_val = float(np.max(conv))
    return 2.0 * n * max_val


def arcsine_initial(n: int) -> np.ndarray:
    """Classic arcsine‑shaped start: a_i ∝ 1/√((i+½)*(n‑½‑i))."""
    i = np.arange(n, dtype=float)
    eps = 0.25
    vals = 1.0 / np.sqrt((i + 0.5) * (n - 0.5 - i) + eps)
    return vals / np.sum(vals)


def generate_shape(n: int, p: float, eps: float) -> np.ndarray:
    """
    Produce a symmetric density a_i ∝ ((i+eps)*(n‑1‑i+eps))⁻ᵖ,
    then normalise to unit L¹ norm.
    """
    i = np.arange(n, dtype=float)
    base = ((i + eps) * (n - 1 - i + eps)) ** (-p)
    total = base.sum()
    if total == 0.0 or np.isinf(total):
        return np.full(n, 1.0 / n)
    return base / total


# ----------------------------------------------------------------------
# Linear‑program based update
# ----------------------------------------------------------------------


def solve_lp_for_g(f: np.ndarray, M: float) -> np.ndarray:
    """
    Solve the LP:
        max Σ g_j
        s.t. (f * g)_k ≤ M   for all k,
             g_j ≥ 0.
    Returns the optimal g (may be zero if LP fails).
    """
    n = len(f)
    rows = 2 * n - 1

    # Build Toeplitz matrix for the convolution constraints
    A = np.zeros((rows, n), dtype=np.float64)
    for i in range(n):
        A[i:i + n, i] = f  # each column i fills rows i … i+n‑1 with f_i

    c = -np.ones(n, dtype=np.float64)          # maximise sum(g) → minimise -sum(g)
    b_ub = np.full(rows, M, dtype=np.float64) # RHS = M for all constraints
    bounds = [(0.0, None)] * n                  # g ≥ 0

    # Use the high‑performance HiGHS solver (available in recent SciPy)
    res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds,
                  method='highs', options={'presolve': True})

    if res.success:
        return res.x
    # Fallback: return a copy of f (neutral element)
    return f.copy()


def lp_mix_step(f: np.ndarray, tau: float) -> Tuple[np.ndarray, float]:
    """
    Perform one LP‑mixing step:
        * compute current M = max(f*f)
        * solve LP to obtain g (maximising its ℓ₁‑norm under the bound)
        * blend f and the normalised g with weight tau
    Returns the new vector and the old M.
    """
    M, _ = max_convolution_and_index(f)
    g = solve_lp_for_g(f, M)
    g_sum = g.sum()
    if g_sum > 0:
        g = g / g_sum
    else:
        g = f.copy()
    f_new = (1.0 - tau) * f + tau * g
    f_new = project_to_simplex(f_new)
    return f_new, M


# ----------------------------------------------------------------------
# Sub‑gradient / Adam‑style polishing
# ----------------------------------------------------------------------


def polish_subgradient(a: np.ndarray,
                       deadline: float,
                       max_iter: int = 300_000,
                       eta0: float = 0.6,
                       use_adam: bool = True,
                       noise_every: int = 600,
                       noise_scale: float = 3e-5) -> Tuple[np.ndarray, float]:
    """
    Projected (Adam‑style) sub‑gradient descent on max(f*f).
    Returns the best vector seen and its internal (fast) score.
    """
    a = project_to_simplex(a.astype(float).copy())
    best = a.copy()
    best_sc = compute_score(a)

    n = len(a)
    # Adam parameters
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8

    it = 0
    while time.time() < deadline and it < max_iter:
        it += 1

        # find (one of) the maximal convolution indices
        _, k_max = max_convolution_and_index(a)
        grad = subgradient_of_max_at_index(a, k_max)

        if use_adam:
            # Adam update
            m = beta1 * m + (1.0 - beta1) * grad
            v = beta2 * v + (1.0 - beta2) * (grad ** 2)
            m_hat = m / (1.0 - beta1 ** it)
            v_hat = v / (1.0 - beta2 ** it)
            step = eta0 * m_hat / (np.sqrt(v_hat) + eps)
            a_new = a - step
        else:
            # simple diminishing step size
            eta = eta0 / np.sqrt(it)
            a_new = a - eta * grad

        if noise_every and (it % noise_every == 0):
            a_new += np.random.randn(n) * noise_scale

        a_new = project_to_simplex(a_new)

        cur_sc = compute_score(a_new)
        if cur_sc < best_sc:
            best_sc = cur_sc
            best = a_new.copy()

        a = a_new

        # occasional progress report (only if we have time)
        if it % 10_000 == 0:
            print(f"    subgrad it={it:,d} cur={cur_sc:.6f} best={best_sc:.6f}")

    return best, best_sc

def refine_candidate(initial_a: np.ndarray,

        starters = [
            np.full(n, 1.0 / n),   # uniform
            arcsine_initial(n),    # classic arcsine
        ]

        local_best = None
        local_best_sc = float("inf")
        for s in starters:
            sc = compute_score(s)
            if sc < local_best_sc:
                local_best_sc = sc
                local_best = s.copy()

        # ---- 1b.  Random scan of the (p, eps) family --------------------
        # Allocate a modest portion of the remaining time to random scanning
        scan_budget = min(30.0, (deadline - time.time()) * 0.15)   # up to ~30 s
        scan_deadline = time.time() + scan_budget
        max_samples = 1500

        for _ in range(max_samples):
            if time.time() >= scan_deadline or time.time() >= deadline - 5.0:
                break
            p = rng.uniform(0.2, 3.0)                     # broadened p‑range
            eps = rng.uniform(1e-6, max(2.0, 0.02 * n))   # very small eps allowed
            cand = generate_shape(n, p, eps)
            sc = compute_score(cand)
            if sc < local_best_sc:
                local_best_sc = sc
                local_best = cand.copy()

        # ---- 1c.  Refine the locally best candidate --------------------
        if local_best is not None:
            # give a fraction of the remaining time to the heavy refinement
            remaining = deadline - time.time()
            # aggressive but time‑aware configuration
            max_lp_steps = 12
            lp_deadline = time.time() + min(remaining * 0.6, 500.0)  # at most 500 s
            refined, _ = refine_candidate(local_best, lp_deadline,
                                          max_lp_steps=max_lp_steps,
                                          lp_tau_start=0.6,
                                          lp_tau_decay=0.85)
            # Evaluate with the official (slow) evaluator once
            val = evaluate_sequence(refined.tolist())
            print(f"  Refined candidate (n={n}) score = {val:.6f}")
            if val < best_val:
                best_val = val
                best_seq = refined.tolist()
                print(f"  *** New global best: {best_val:.6f} (n={n}) ***")
        else:
            # Should never happen, but keep a safe guard
            continue

    # ------------------------------------------------------------------
    # 2.  Final polishing of the global best (time permitting)
    # ------------------------------------------------------------------
    if best_seq and time.time() < deadline - 5.0:
        a = np.array(best_seq, dtype=float)
        print("\nFinal polishing of the global best ...")
        a_refined, _ = polish_subgradient(a, deadline,
                                          max_iter=400_000,
                                          eta0=0.5,
                                          use_adam=True,
                                          noise_every=500,
                                          noise_scale=2e-5)
        final_val = evaluate_sequence(a_refined.tolist())
        print(f"Final polished score = {final_val:.6f}")
        if final_val < best_val:
            best_val = final_val
            best_seq = a_refined.tolist()
            print("*** Updated global best after final polish ***")

    # ------------------------------------------------------------------
    # 3.  Ensure we return something (fallback uniform vector)
    # ------------------------------------------------------------------
    if not best_seq:
        best_seq = [1.0] * 256
        best_val = evaluate_sequence(best_seq)
        print(f"Fallback uniform vector used, score = {best_val:.6f}")

    print(f"\nSearch finished. Best reported score = {best_val:.6f}")
    return best_seq
