"""
Propose a candidate step‑function sequence.

The algorithm is a *projected stochastic sub‑gradient* method with several
enhancements:

*   The objective (with sum(a)=1) is 2·n·max(conv(a,a)).  The sub‑gradient
    of the max‑convolution is obtained by correlating a with the mask of
    (near‑) maximal entries.
*   A momentum term (heavy‑ball) stabilises the descent.
*   An adaptive “active‑set” tolerance starts large (5 % of the peak) and
    shrinks, giving a smoother direction early on.
*   Simulated‑annealing acceptance (temperature slowly cooling) allows
    occasional uphill moves, helping escape local minima.
*   Small random mass‑transfer moves are injected regularly.
*   When progress stalls a random perturbation of the current best
    solution is performed (restart).

The routine respects the 1000 s budget, prints progress and finally returns
the best sequence found as a plain Python list of floats.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
def binomial_sequence(length: int) -> np.ndarray:
    """Central‑binomial coefficients  C(2k,k)/4^k  for k = 0 … length‑1."""
    seq = np.empty(length, dtype=np.float64)
    seq[0] = 1.0
    for k in range(1, length):
        seq[k] = seq[k - 1] * ((2 * k - 1) * (2 * k)) / (4.0 * k * k)
    return seq


# ----------------------------------------------------------------------
def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a non‑negative vector onto the probability simplex
    (sum = 1, all components ≥ 0).  Since v is already non‑negative we may
    simply renormalise.
    """
    v = np.maximum(v, 0.0)
    s = v.sum()
    if s == 0.0:
        # fallback: uniform distribution
        return np.full_like(v, 1.0 / len(v))
    return v / s

            print(f"[restart] iter={iter_cnt}, elapsed={elapsed:.2f}s")

        iter_cnt += 1

    # --------------------------------------------------------------
    total_elapsed = time.time() - start_time
    print(f"[final] best score={best_score:.9f}, iterations={iter_cnt}, "
          f"time={total_elapsed:.2f}s")
    return best_seq.tolist()
