"""
Propose a step‑function (non‑negative sequence) that minimises the
evaluation function `evaluate_sequence`.  After normalising the sum to
√(2·n) the score is simply `max(conv(a, a))`.  The algorithm works in
two stages:

1.  Exponentiated‑gradient descent on the smooth surrogate of the
    max‑convolution:
        L_λ(a) = (1/λ)·log Σ_k exp(λ·(a*a)_k) .
    The gradient is obtained by a second convolution of the soft‑max
    weights with the reversed vector.  After each update the vector is
    projected back onto the simplex {x ≥ 0, Σ x = √(2·n)}.

2.  A short projected sub‑gradient descent (the same scheme that was
    used in the reference implementation) starting from the best
    candidate found in stage 1, to polish the final result.

All convolutions are performed with a real FFT (O(n log n)).  The
routine respects a 1000 s time budget, prints occasional progress
messages and finally returns the best sequence as a plain Python list.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities
# ----------------------------------------------------------------------


def _project_simplex(v: np.ndarray, target_sum: float) -> np.ndarray:
    """Euclidean projection of a vector onto the simplex {w_i ≥ 0,
    Σ w_i = target_sum}.  Implements the algorithm of Duchi et al. (2008)."""
    if target_sum < 0:
        raise ValueError("target_sum must be non‑negative")
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - target_sum
    rho = np.where(u - cssv / (np.arange(1, len(v) + 1)) > 0)[0]
    if rho.size == 0:
        return np.zeros_like(v)
    rho = rho[-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def _conv_fft(u: np.ndarray, v: np.ndarray) -> np.ndarray:
    """Linear convolution of two real vectors using a real FFT."""
    size = 1 << ((len(u) + len(v) - 1).bit_length())
    fu = np.fft.rfft(u, n=size)
    fv = np.fft.rfft(v, n=size)
    conv = np.fft.irfft(fu * fv, n=size)
    return conv[: len(u) + len(v) - 1]


def _full_convolution_fft(a: np.ndarray) -> np.ndarray:
    """Full (linear) convolution a*a; length = 2·len(a)-1."""
    return _conv_fft(a, a)


def _max_conv(a: np.ndarray) -> float:
    """Maximum entry of the self‑convolution a*a."""
    return _full_convolution_fft(a).max()


def _make_candidate(
    n: int, p: float, offset: float, beta: float
) -> np.ndarray:
    """
    Build an un‑normalised raw sequence
        a_i = (i + 1 + offset)^{-p} * (1 + beta * i/(n‑1))
    and project it onto the simplex Σ a = √(2·n).
    """
    idx = np.arange(n, dtype=float)
    base = np.power(idx + 1.0 + offset, -p)
    factor = 1.0 + beta * idx / (n - 1)
    raw = base * factor
    raw = np.maximum(raw, 0.0)
    target = math.sqrt(2.0 * n)
    return _project_simplex(raw, target)


def _subgradient_step(
    a: np.ndarray, max_inds: np.ndarray, step: float, target_sum: float
) -> np.ndarray:
    """
    One projected sub‑gradient step for the objective max(conv(a,a)).
    The sub‑gradient is summed over all active indices `max_inds`.
    """
    n = len(a)
    g = np.zeros_like(a)

    for k in max_inds:
        if k < n:
            # i runs from 0 … k
            g[: k + 1] += a[k :: -1]
        else:
            start = k - (n - 1)
            # i runs from start … n‑1
            g[start:] += a[k - np.arange(start, n)]

    cand = a - step * g
    cand = _project_simplex(cand, target_sum)
    return cand


# ----------------------------------------------------------------------
# Main search routine
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 123, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative height sequence that minimises the
    evaluation function.  The routine runs for at most `budget_s`
    seconds and returns the best sequence found as a plain list.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 2.0   # safety margin

    # --------------------------------------------------------------
    # 0. Choose length – a few thousand elements give a good trade‑off.
    # --------------------------------------------------------------
    n = 8000
    target_sum = math.sqrt(2.0 * n)

    # --------------------------------------------------------------
    # 1. Warm‑start: reuse a previously discovered sequence if possible.
    # --------------------------------------------------------------
    best_seq = None
    hs = globals().get("height_sequence_1")
    if hs is not None and len(hs) == n:
        best_seq = np.asarray(hs, dtype=float)
        best_seq = _project_simplex(best_seq, target_sum)

    if best_seq is None:
        # simple power‑law start
        best_seq = _make_candidate(n, p=0.5, offset=0.0, beta=0.0)

    # --------------------------------------------------------------
    # 2. Tiny random sweep over the 4‑parameter family for a quick boost.
    # --------------------------------------------------------------
    best_score = _max_conv(best_seq)
    print(f"[init] n={n:5d}  score={best_score:.6f}")

    sweep_deadline = time.time() + 0.10 * (deadline - time.time())
    while time.time() < sweep_deadline:
        p = np.random.uniform(0.3, 0.8)
        off = np.random.uniform(0.0, 12.0)
        beta = np.random.uniform(-0.9, 2.5)
        cand = _make_candidate(n, p, off, beta)
        cand_score = _max_conv(cand)
        if cand_score < best_score:
            best_score = cand_score
            best_seq = cand
            print(f"[sweep] new best {best_score:.6f}  p={p:.3f} α={off:.3f} β={beta:.3f}")
        if best_score < 1.55:   # early exit if we already get reasonably good
            break

    # --------------------------------------------------------------
    # 3. Exponentiated‑gradient descent on a smooth max‑convolution.
    # --------------------------------------------------------------
    a = best_seq.copy()
    eta0 = 0.5          # initial learning‑rate factor
    lam = 0.5           # start with a mild smoothing
    lam_max = 12.0
    lam_inc_every = 3000   # increase λ every this many iterations

    iteration = 0
    while time.time() < deadline:
        iteration += 1

        # current convolution and current score
        conv = _full_convolution_fft(a)
        M = conv.max()
        if M < best_score:
            best_score = M
            best_seq = a.copy()
            print(f"[egdm] iter {iteration:7d}  new best {best_score:.6f}")
            if best_score < 1.503:
                break

        # ---- soft‑max weights for the smooth max -----------------
        # they are all ≤1, so no overflow worries
        conv_shift = conv - M                     # ≤0
        w = np.exp(lam * conv_shift)
        w_sum = w.sum()
        if w_sum == 0:
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        # ---- gradient of the smooth objective --------------------
        # a_rev is a reversed copy of a padded to length 2n‑1
        a_rev = np.zeros(2 * n - 1, dtype=float)
        a_rev[:n] = a[::-1]

        # convolution w * a_rev (full length = 4n‑3)
        grad_full = _conv_fft(w, a_rev)
        # central slice (indices n‑1 … 2n‑2) gives the needed gradient
        g = 2.0 * grad_full[n - 1 : n - 1 + n]

        # ---- step size (decays like 1/√t) ----------------------
        step = eta0 / math.sqrt(iteration)

        # ---- multiplicative (exponentiated) update -------------
        a = a * np.exp(-step * g)

        # ---- re‑normalise onto the simplex ----------------------
        s = a.sum()
        if s == 0 or not np.isfinite(s):
            # fall back to a fresh random start if things went bad
            a = _make_candidate(n, p=0.5, offset=0.0, beta=0.0)
        else:
            a = a * (target_sum / s)

        # ---- anneal λ -------------------------------------------
        if iteration % lam_inc_every == 0 and lam < lam_max:
            lam = min(lam * 1.10, lam_max)

        # ---- occasional jitter to avoid stagnation -------------
        if iteration % 2000 == 0 and np.random.rand() < 0.25:
            jitter = (np.random.rand(n) - 0.5) * 0.02 * a
            a = a + jitter
            a = np.maximum(a, 0.0)
            a = a * (target_sum / a.sum())

    # --------------------------------------------------------------
    # 4. Short projected sub‑gradient polishing (as in the reference)
    # --------------------------------------------------------------
    curr = best_seq.copy()
    step0 = 0.5
    sub_it = 0
    sub_deadline = time.time() + min(60.0, deadline - time.time())
    while time.time() < sub_deadline:
        sub_it += 1
        conv_full = _full_convolution_fft(curr)
        M = conv_full.max()
        if M < best_score:
            best_score = M
            best_seq = curr.copy()
            print(f"[sgd] sub‑iter {sub_it:5d}  new best {best_score:.6f}")
            if best_score < 1.503:
                break
        eps = 1e-12 * M if M != 0 else 1e-12
        max_inds = np.where(conv_full >= M - eps)[0]
        step = step0 / math.sqrt(sub_it)
        cand = _subgradient_step(curr, max_inds, step, target_sum)
        if sub_it % 50 == 0 and np.random.rand() < 0.3:
            jitter = (np.random.rand(n) - 0.5) * 0.02 * cand
            cand = cand + jitter
            cand = _project_simplex(cand, target_sum)
        curr = cand

    print(f"Finished – best score {best_score:.6f} after {iteration} EG "
          f"and {sub_it} SGD iterations")
    return best_seq.tolist()
