"""
Propose a candidate step‑function that minimizes

    2 * n * max(conv(a, a)) / (sum(a)**2)

The algorithm is a mirror‑descent (exponentiated‑gradient) optimiser on the
simplex combined with a few stochastic refinements:

1. Initialise the coefficients with the classic a_i ∝ 1/√(i+1) shape
   (normalised to sum = 1).  The shape is already close to optimal but the
   optimisation refines it.

2. In each iteration
   * compute the full self‑convolution using an FFT,
   * locate the *active* shifts – the top K (default 5) shifts whose
     convolution value is within a small relative tolerance of the
     current maximum,
   * build a sub‑gradient of the maximum entry as the sum of the
     individual gradients for the active shifts,
   * perform an exponentiated‑gradient update (multiplicative
     modification) and renormalise to stay on the probability simplex.

   The learning‑rate decays like η₀/√t, which is a standard schedule for
   sub‑gradient methods on convex Lipschitz functions.

3. Every few hundred iterations a tiny random redistribution of mass
   (jitter) is applied to help the search escape shallow plateaux.

4. Throughout the run the best sequence seen so far (according to the
   exact evaluation function) is remembered.

5. After the main loop a short stochastic hill‑climbing phase is run:
   small amounts of mass are moved between random indices and kept if the
   objective improves.

The whole routine respects a time budget (default 1000 s) and returns a
plain Python list of non‑negative heights.  If something goes wrong it
falls back to the previously discovered global `height_sequence_1`.
"""

import time
import numpy as np


def project_onto_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection onto the probability simplex {x | x >= 0, sum(x)=1}.
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def full_convolution(a: np.ndarray) -> np.ndarray:
    """
    Compute the full discrete self‑convolution a*a using an FFT.
    Length of result is 2*len(a)-1.
    """
    n = a.shape[0]
    size = 2 * n
    fft_len = 1 << (size - 1).bit_length()          # next power of two
    fft_a = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(fft_a * fft_a, n=fft_len)
    return conv[: 2 * n - 1]


def gradient_sum_topk(a: np.ndarray, conv: np.ndarray, topk: int = 5,
                     rel_tol: float = 1e-4) -> np.ndarray:
    """
    Return the sum of sub‑gradients belonging to the `topk` largest
    convolution entries.  To make the search a bit more robust we also
    include any shift whose value lies within `rel_tol` of the global
    maximum.
    """
    n = a.shape[0]
    max_val = conv.max()
    # indices that are “active”
    active = np.where(conv >= max_val * (1.0 - rel_tol))[0]
    # keep at most topk of them (the most extreme)
    if active.size > topk:
        active = active[np.argpartition(-conv[active], topk)[:topk]]
    grad = np.zeros_like(a)
    for k in active:
        low = max(0, k - n + 1)
        high = min(k, n - 1)
        if low > high:
            continue
        idx = np.arange(low, high + 1, dtype=int)
        grad[idx] += a[k - idx]                     # sub‑gradient for this shift
    return grad

    evaluation function defined in the competition.  The method combines
    exponentiated‑gradient descent with occasional random jitter and a
    final hill‑climbing phase.
    """
    np.random.seed(seed)
    # keep a few seconds free for final printing/return
    deadline = time.time() + budget_s - 5.0

    # ------------------------------------------------------------------
    # Parameters
    # ------------------------------------------------------------------
    n = 5000                          # length of the step‑function
    # ----- initial shape ------------------------------------------------
    raw = 1.0 / np.sqrt(np.arange(n, dtype=float) + 1.0)
    a = raw / raw.sum()               # sum(a) == 1

    # ----- optimisation control -----------------------------------------
    step_initial = 0.6                 # base learning rate for EG
    topk = 7                          # number of active shifts to use
    rel_tol = 2e-4                    # tolerance to include near‑max shifts
    jitter_every = 300                # iterations between random jitter
    jitter_frac = 0.005               # max fraction of mass moved in jitter

    best_seq = a.copy()
    best_score = 2.0 * n * np.max(full_convolution(a))

    print(f"Initial objective (n={n}): {best_score:.6f}")

    iteration = 0
    # ------------------------------------------------------------------
    # Main exponentiated‑gradient loop
    # ------------------------------------------------------------------
    while time.time() < deadline:
        conv = full_convolution(a)
        max_val = conv.max()
        cur_score = 2.0 * n * max_val
        if cur_score < best_score:
            best_score = cur_score
            best_seq = a.copy()
            print(f"Iter {iteration:06d}: improved score = {best_score:.6f}")

            if best_score < 1.503:
                # target reached – stop early
                break

        # compute (smoothed) sub‑gradient using the most critical shifts
        g = gradient_sum_topk(a, conv, topk=topk, rel_tol=rel_tol)

        # exponentiated‑gradient update
        eta = step_initial / np.sqrt(iteration + 1.0)
        # avoid underflow / overflow: cap the exponent
        exp_term = np.exp(-eta * g)
        a = a * exp_term
        # enforce positivity and simplex constraint
        a = np.maximum(a, 0.0)
        if a.sum() == 0.0:
            a = np.full_like(a, 1.0 / n)
        else:
            a = a / a.sum()

        # occasional tiny random jitter to increase exploration
        if (iteration % jitter_every) == 0 and iteration > 0:
            i, j = np.random.randint(0, n, size=2)
            if i != j:
                delta = jitter_frac * a[i] * (np.random.rand() - 0.5)
                cand = a.copy()
                cand[i] = max(cand[i] - delta, 0.0)
                cand[j] = cand[j] + delta
                # re‑normalise
                if cand.sum() == 0.0:
                    cand = np.full_like(cand, 1.0 / n)
                else:
                    cand = cand / cand.sum()
                cand_score = 2.0 * n * np.max(full_convolution(cand))
                if cand_score < best_score:
                    best_score = cand_score
                    best_seq = cand.copy()
                    a = cand
                    print(f"  jitter improvement: score = {best_score:.6f}")

        iteration += 1

    # ------------------------------------------------------------------
    # Short stochastic hill‑climbing refinement (uses remaining budget)
    # ------------------------------------------------------------------
    refinement_deadline = time.time() + min(30.0, deadline - time.time())
    while time.time() < refinement_deadline:
        # move a tiny amount of mass from i to j
        i, j = np.random.randint(0, n, size=2)
        if i == j:
            continue
        # fraction of the source mass to shift (very small)
        frac = 0.001 * np.random.rand()
        delta = frac * a[i]
        if delta == 0.0:
            continue
        cand = a.copy()
        cand[i] -= delta
        cand[j] += delta
        # renormalise (should already sum to 1, but numerical errors)
        cand = cand / cand.sum()
        cand_score = 2.0 * n * np.max(full_convolution(cand))
        if cand_score < best_score:
            best_score = cand_score
            best_seq = cand.copy()
            a = cand
            print(f"  hill‑climb improvement: score = {best_score:.6f}")

    # ------------------------------------------------------------------
    # Safety fallback – use the previous construction if nothing found
    # ------------------------------------------------------------------
    if best_seq is None or best_seq.size == 0:
        try:
            best_seq = np.array(height_sequence_1, dtype=float)
        except Exception:
            # final resort: uniform distribution (objective = 2.0)
            best_seq = np.full(2000, np.sqrt(2.0 * 2000) / 2000, dtype=float)

    print(f"Final best score: {best_score:.6f} (length {len(best_seq)})")
    return best_seq.tolist()
