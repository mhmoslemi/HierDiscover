"""
Propose a low‑score step‑function sequence.

The algorithm consists of three stages:

1. **Broad seeding** – generate a diverse pool of candidate vectors (geometric,
   power‑law, uniform and random Dirichlet).  Each candidate is projected onto the
   ℓ₁‑simplex Σa = √(2n) and its maximal self‑convolution is measured with a fast
   FFT.  The best seed is kept.

2. **Projected sub‑gradient descent** – starting from the best seed we minimise
   the maximum of the self‑convolution (which is exactly the value returned by
   ``evaluate_sequence`` for a normalised vector).  In each iteration we locate
   the active set (indices where the convolution attains its current maximum),
   compute the sub‑gradient, take a Nesterov‑type momentum step and project back
   onto the simplex.  The step size decays slowly.

3. **Simulated‑annealing local search** – the vector obtained after the gradient
   phase is refined by a large number of cheap rank‑1 updates.  Moving a small
   amount of mass from a source index *i* to a destination *j* changes the
   convolution by a known analytical expression, which can be updated in O(n)
   time without recomputing an FFT.  An exponential cooling schedule accepts
   uphill moves with probability  exp(−Δ/T).  The best vector seen during the
   annealing is retained.

Finally the best vector is clipped, projected once more onto the simplex (to
eliminate any numerical drift) and evaluated with the official
``evaluate_sequence`` function.

The whole procedure respects the 1000 s time limit.
"""

import time
import numpy as np

# --------------------------------------------------------------------------- #
# Simplex projection (Duchi et al. 2008)
def projection_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Project vector v onto the simplex {x ≥ 0, Σx = s}."""
    if s < 0:
        raise ValueError("Simplex sum must be non‑negative")
    n = v.shape[0]
    # sort v in descending order
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho_arr = np.where(u - cssv / (np.arange(n) + 1.0) > 0)[0]
    if rho_arr.size == 0:
        return np.zeros_like(v)
    rho = rho_arr[-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# --------------------------------------------------------------------------- #
def max_self_convolution(a: np.ndarray, fft_len: int) -> float:
    """Return max_k (a * a)[k] using an FFT of length fft_len."""
    F = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(F * F, n=fft_len)[: 2 * a.size - 1]
    return float(conv.max())


# --------------------------------------------------------------------------- #
def _build_toeplitz_matrix(f: np.ndarray):
    """Return a sparse matrix A (2n‑1, n) such that A·g = f * g."""
    n = f.shape[0]
    rows = np.empty(n * n, dtype=np.int32)
    cols = np.empty(n * n, dtype=np.int32)
    data = np.empty(n * n, dtype=np.float64)

    ptr = 0
    for i in range(n):
        rows[ptr: ptr + n] = np.arange(i, i + n)          # rows i … i+n‑1
        cols[ptr: ptr + n] = i                           # column i
        data[ptr: ptr + n] = f
        ptr += n

    from scipy.sparse import csr_matrix
    return csr_matrix((data, (rows, cols)), shape=(2 * n - 1, n))

def propose_candidate(seed: int = 42, budget_s: float = 1000, **kwargs):
    """
    Search for a non‑negative step‑function (list of floats) that minimises the
    evaluation function.  The routine returns the best sequence found before the
    time budget expires.
    """
    np.random.seed(seed)
    start = time.time()
    deadline = start + budget_s - 0.2      # safety margin before the caller kills us

    # --------------------------------------------------------------- #
    # Problem size – a few thousand entries provide a good trade‑off.
    n = 2500
    target_sum = np.sqrt(2.0 * n)          # we keep the ℓ₁‑norm fixed throughout
    max_val = 1000.0

    # Shared FFT length for all fast convolutions (next power‑of‑two ≥ 2·n‑1)
    fft_len = 1 << ((2 * n - 1).bit_length())

    # --------------------------------------------------------------- #
    # 1. Generate a diverse pool of seeds.
    seeds = []

    # geometric families a_i ∝ λ·(1‑λ)^i
    lambdas = np.logspace(-5, -1, 40)
    for lam in lambdas:
        r = 1.0 - lam
        denom = 1.0 - r ** n
        if denom <= 0.0:
            continue
        coeff = target_sum * lam / denom
        a = coeff * (r ** np.arange(n))
        a = np.clip(a, 0.0, max_val)
        seeds.append(a)

    # power‑law families a_i ∝ (i+1)^‑p
    for p in np.linspace(0.3, 0.95, 30):
        base = 1.0 / np.power(np.arange(1, n + 1, dtype=np.float64), p)
        coeff = target_sum / base.sum()
        a = coeff * base
        a = np.clip(a, 0.0, max_val)
        seeds.append(a)

    # uniform seed (fallback)
    seeds.append(np.full(n, target_sum / n))

    # random Dirichlet‑like seeds – many small components
    for _ in range(10):
        raw = np.random.exponential(scale=1.0, size=n)
        a = projection_simplex(raw, target_sum)
        a = np.clip(a, 0.0, max_val)
        seeds.append(a)

    # if a previous global solution exists and has the right length, use it
    try:
        global height_sequence_1
        if isinstance(height_sequence_1, (list, tuple, np.ndarray)):
            h = np.asarray(height_sequence_1, dtype=np.float64)
            if h.shape[0] == n:
                h = np.clip(h, 0.0, max_val)
                h = projection_simplex(h, target_sum)
                seeds.append(h)
    except Exception:
        pass

    # --------------------------------------------------------------- #
    # 2. Scan the seeds – keep the best according to the exact max‑convolution.
    best_seq = None
    best_score = float("inf")
    for idx, cand in enumerate(seeds):
        if time.time() >= deadline:
            break
        cand = projection_simplex(cand, target_sum)

    curr = best_seq.copy()
    best = curr.copy()
    best_score = max_self_convolution(best, fft_len)

    # momentum buffer
    velocity = np.zeros_like(curr)

    # initial absolute step (about 1 % of the average entry)
    step = 0.7 * target_sum / n
    min_step = 1e-9
    momentum = 0.9
    rel_tol = 1e-4
    it = 0

    grad_deadline = start + 0.20 * budget_s

        # convolution via FFT
        F = np.fft.rfft(curr, n=fft_len)

        score = float(cur_max)               # = evaluate_sequence when Σa = √(2n)

        tol_abs = max_cur = cur_max * rel_tol

        grad = np.zeros_like(curr)
        idx_arr = np.arange(n)
        for k in active_idx:
            j = k - idx_arr
            mask = (j >= 0) & (j < n)
            grad[mask] += 2.0 * curr[j[mask]]

            break

        velocity = momentum * velocity - step * (grad / gnorm)
        tentative = curr + velocity
        tentative = np.clip(tentative, 0.0, max_val)
        curr = projection_simplex(tentative, target_sum)

    # Initialise the convolution of the current best vector.
    a = best.copy()
    F = np.fft.rfft(a, n=fft_len)

    cur_max = float(conv.max())
    best_score = cur_max
    best = a.copy()
    best_conv = conv.copy()

    T0 = best_score * 0.03 + 1e-4           # initial temperature
    T = T0
    alpha = 0.9999                          # exponential cooling
    min_T = 1e-6

    # keep track of total iterations for occasional diagnostics
    sa_iter = 0

    while time.time() < deadline - 0.5:

        if a[i] <= 1e-12:
            continue

        j = np.random.randint(0, n - 1)
        if j >= i:

        # Compute the change of the convolution analytically.
        #   (a‑δ·e_i + δ·e_j) * (a‑δ·e_i + δ·e_j) = a*a + Δ
        # where

        #       + δ²·e_{2i} + δ²·e_{2j} - 2δ²·e_{i+j}
        # shift_i(a)[k] = a[k‑i] (zero outside the support)
        # ----------------------------------------------------------- #
        delta_conv = np.zeros_like(conv)

        delta_conv[2 * i] += delta * delta
        delta_conv[2 * j] += delta * delta

        new_max = float(new_conv.max())

        accept = False

        else:

            if np.random.rand() < prob:
                accept = True

        if accept:
            # apply the move
            a[i] -= delta
            a[j] += delta
            conv = new_conv
            cur_max = new_max

            if cur_max < best_score:
                best_score = cur_max
                best = a.copy()
                best_conv = conv.copy()
                print(f"[{time.time() - start:6.1f}s] SA iter {sa_iter:7d} → new best {best_score:.6f}")

        # temperature update
        T = max(T * alpha, min_T)
        sa_iter += 1

        if np.random.rand() < 0.0005:
            jitter = (np.random.randn(n) *
                      (target_sum / n) * 0.5)

                np.clip(a + jitter, 0.0, max_val), target_sum)

            conv = np.fft.irfft(F * F, n=fft_len)[: 2 * n - 1]
            cur_max = float(conv.max())

    # --------------------------------------------------------------- #

    #    time (removes any accumulated floating‑point drift) and evaluate.
    best = projection_simplex(best, target_sum)
    final_score = evaluate_sequence(best.tolist())
    print(f"Search finished. Final score reported by evaluate_sequence: {final_score:.6f}")

    return best.tolist()
