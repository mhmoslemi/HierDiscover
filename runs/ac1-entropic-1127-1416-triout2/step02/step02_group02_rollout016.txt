"""
Propose a candidate step‑function sequence by *annealed smoothed‑max*
gradient descent on the simplex.

The evaluation function is

    f(a) = 2·n·max(conv(a,a)) / (sum(a)²)

If we keep a on the probability simplex (sum(a)=1) this reduces to
`2·n·max(conv(a,a))`.  The maximum entry of the convolution is a convex,
nondifferentiable function; we replace it by a soft‑max with a temperature
parameter α and minimise the smooth surrogate with Adam.  After each
Adam step we project back onto the simplex (Duchi‑et‑al. 2008).  The
temperature α is slowly increased so the method converges to the true
objective while still being able to escape flat regions.

The routine runs for at most ``budget_s`` seconds, tries two problem
sizes (the size of the known best sequence and a larger one) and keeps
the best vector seen.  All values are non‑negative and sum to 1, so the
returned list can be fed directly to ``evaluate_sequence``.
"""
import time
import numpy as np

# ----------------------------------------------------------------------
# Helper functions (all top‑level – no closures)
# ----------------------------------------------------------------------


def simplex_projection(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the probability simplex:
        { w | w_i ≥ 0 , Σ w_i = 1 }.
    Implementation follows Duchi et al. (2008).
    """
    n = v.shape[0]
    # sort v in decreasing order
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1] - 1            # zero‑based
    theta = cssv[cond][-1] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def next_pow2(x: int) -> int:
    """Small utility – smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def compute_objective(a: np.ndarray) -> float:
    """
    Direct evaluation of the objective for a probability vector a
    (i.e. Σ a = 1).  Returns 2·n·max(conv(a,a)).
    """
    n = a.shape[0]
    N = next_pow2(2 * n - 1)
    fft_a = np.fft.rfft(a, N)
    conv = np.fft.irfft(fft_a * fft_a, n=N)[: 2 * n - 1]
    return 2.0 * n * conv.max()


def run_annealed_adam(
    n: int,
    a0: np.ndarray,
    deadline: float,
    *,
    lr: float = 0.05,
    beta1: float = 0.9,
    beta2: float = 0.999,
    eps: float = 1e-8,
    alpha_start: float = 1.0,
    alpha_max: float = 200.0,
    alpha_inc_every: int = 2000,
    noise_every: int = 5000,
) -> tuple[np.ndarray, float]:
    """
    Core optimisation loop.

    Parameters
    ----------
    n               – length of the sequence.
    a0              – initial probability vector (already normalised).
    deadline        – Unix time after which the loop must stop.
    lr, beta1, …    – Adam hyper‑parameters.
    alpha_start     – initial temperature for the soft‑max.
    alpha_max       – cap for the temperature.
    alpha_inc_every – how many iterations before multiplying α by 1.2.
    noise_every     – iteration interval at which a tiny Gaussian jitter
                      is added before projection (helps escape plateaus).

    Returns
    -------
    best_a, best_score   – the best vector found and its objective value.
    """
    a = a0.copy()
    best_a = a.copy()
    best_score = compute_objective(a)

    # Pre‑compute the FFT length (must accommodate both a*a and w_rev*a)
    N = next_pow2(3 * n - 2)  # >= (2n‑1)+n‑1

    # Adam state
    m = np.zeros_like(a)
    v = np.zeros_like(a)

    alpha = alpha_start
    it = 0
    last_improve = 0

    # bias‑correction denominators (updated each iteration)
    beta1_pow = 1.0
    beta2_pow = 1.0

    while time.time() < deadline:
        # --------------------------------------------------------------
        # 1) convolution a*a  (needed for objective and for the weights)
        # --------------------------------------------------------------
        fft_a = np.fft.rfft(a, N)
        conv_full = np.fft.irfft(fft_a * fft_a, n=N)[: 2 * n - 1]
        max_val = conv_full.max()

        # --------------------------------------------------------------
        # 2) Track the best solution seen so far
        # --------------------------------------------------------------
        cur_score = 2.0 * n * max_val
        if cur_score < best_score - 1e-12:
            best_score = cur_score
            best_a = a.copy()
            last_improve = it
            print(f"n={n:5d} it={it:7d}   improved score = {best_score:.6f}")

        # --------------------------------------------------------------
        # 3) Soft‑max weights for the smoothed max
        # --------------------------------------------------------------
        # (conv_full - max_val) ≤ 0  → safe exponentiation
        w_exp = np.exp(alpha * (conv_full - max_val))
        w_sum = w_exp.sum()
        if w_sum == 0 or not np.isfinite(w_sum):
            # numerical under‑flow – fall back to uniform weights
            w = np.full_like(w_exp, 1.0 / w_exp.size)
        else:
            w = w_exp / w_sum

        # --------------------------------------------------------------
        # 4) Gradient of the smoothed max:
        #    g_i = 2 * Σ_k w_k * a_{k-i}   (zero outside the interval)
        #    This is a convolution of w_rev with a.
        # --------------------------------------------------------------
        w_rev = w[::-1]                      # length L = 2n‑1
        fft_w_rev = np.fft.rfft(w_rev, N)
        # fft_a already computed, reuse it
        conv_wrev_a = np.fft.irfft(fft_w_rev * fft_a, n=N)[: 3 * n - 2]
        # Indices for the gradient:   g_i = 2 * conv_wrev_a[2n‑2‑i]
        grad = 2.0 * conv_wrev_a[(2 * n - 2) - np.arange(n)]

        # --------------------------------------------------------------
        # 5) Adam update and simplex projection
        # --------------------------------------------------------------
        it += 1
        beta1_pow *= beta1
        beta2_pow *= beta2

        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)

        m_hat = m / (1.0 - beta1_pow)
        v_hat = v / (1.0 - beta2_pow)

        step = lr * m_hat / (np.sqrt(v_hat) + eps)
        a = simplex_projection(a - step)

        # --------------------------------------------------------------
        # 6) Occasional jitter (helps to get out of shallow plateaus)
        # --------------------------------------------------------------
        if it % noise_every == 0:
            eps_noise = 0.001 / np.sqrt(it)
            a = simplex_projection(a + eps_noise * np.random.randn(n))

        # --------------------------------------------------------------
        # 7) Anneal the temperature (make the surrogate tighter)
        # --------------------------------------------------------------
        if it % alpha_inc_every == 0 and alpha < alpha_max:
            alpha = min(alpha * 1.2, alpha_max)

        # --------------------------------------------------------------
        # 8) Stagnation handling – random restart if no improvement for a while
        # --------------------------------------------------------------
        if it - last_improve > 20000:
            # re‑initialise around the best point (small perturbation)
            a = best_a + 0.01 * np.random.rand(n)
            a = simplex_projection(a)
            # reset Adam state
            m.fill(0.0)
            v.fill(0.0)
            beta1_pow = 1.0
            beta2_pow = 1.0
            last_improve = it
            print(f"n={n} it={it}: restart due to stagnation")

        # --------------------------------------------------------------
        # 9) Early exit if we are already very close to the theoretical lower bound
        # --------------------------------------------------------------
        # The absolute lower bound for length n is 2·n/(2·n‑1) ≈ 1.0.
        # If we are within 0.01 of that we stop early.
        if best_score <= (2.0 * n) / (2 * n - 1) + 0.01:
            break

    return best_a, best_score


def propose_candidate(budget_s: int = 1000) -> list:
    """
    Search for a low‑score step function.  Returns the best sequence found
    within the allotted time budget.
    """
    # small safety margin so we definitely finish before the external timeout
    deadline = time.time() + budget_s - 5.0
    np.random.seed(1)  # deterministic but not tied to the previous run

    # --------------------------------------------------------------
    # Determine problem size(s) and initialise starts
    # --------------------------------------------------------------
    try:
        # the environment may already provide a good construction
        init_vec = np.asarray(height_sequence_1, dtype=float)  # type: ignore[name-defined]
        n0 = init_vec.size
        # normalise to a probability vector
        if init_vec.sum() > 0:
            init_vec = init_vec / init_vec.sum()
        else:
            init_vec = np.full(n0, 1.0 / n0)
    except Exception:
        # fallback – moderate size, uniform start
        n0 = 6000
        init_vec = np.full(n0, 1.0 / n0)

    # We'll try two lengths: the original one and a larger one
    n_big = min(12000, int(1.5 * n0))

    sizes = [(n0, init_vec)]
    if n_big != n0:
        sizes.append((n_big, np.full(n_big, 1.0 / n_big)))  # uniform start

    best_overall = None
    best_score_overall = float("inf")

    for n, a_start in sizes:
        # If the start vector has a different length, pad or trim it
        if a_start.size != n:
            if a_start.size > n:
                a0 = a_start[:n]
            else:
                a0 = np.concatenate([a_start, np.full(n - a_start.size, 0.0)])
            if a0.sum() == 0:
                a0 = np.full(n, 1.0 / n)
            else:
                a0 = a0 / a0.sum()
        else:
            a0 = a_start.copy()

        # Run the main optimisation loop for this size
        best_a, best_sc = run_annealed_adam(
            n=n,
            a0=a0,
            deadline=deadline,
            lr=0.07,
            alpha_start=2.0,
            alpha_max=150.0,
        )
        if best_sc < best_score_overall:
            best_score_overall = best_sc
            best_overall = best_a.copy()
            print(f"=== New global best (n={n}) = {best_score_overall:.6f} ===")

        # If we are out of time, break early
        if time.time() >= deadline:
            break

    # Convert to plain Python list of floats for the caller
    if best_overall is None:
        # Should never happen – fall back to a uniform sequence
        n = n0
        best_overall = np.full(n, 1.0 / n)

    return [float(v) for v in best_overall.tolist()]
