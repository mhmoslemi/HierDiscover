"""
Propose a candidate step‑function sequence that minimises
    2·n·max(a*a) / (sum(a)^2)

The algorithm works in three stages:

1. **Fast parametric sweep** – we generate a variety of simple
   families (power‑law, exponential, linear, quadratic, asymmetric
   power‑law) on the same length as the current best known sequence.
   The best of these low‑dimensional candidates is kept as a warm‑start.

2. **Projected Adam optimisation** – starting from the best candidate we
   run a long Adam‑style stochastic sub‑gradient descent on the simplex.
   At each iteration we compute the convolution by FFT, locate the active
   (maximal) entries and build a sub‑gradient of the maximum‑convolution.
   The update is projected onto the probability simplex with the
   Euclidean projection of Duchi et al. (2008).  An occasional small
   random jitter helps escape flat regions.

3. **Result selection** – the best vector seen (lowest objective value)
   during the whole run is returned as a plain Python list.  The routine
   respects a user‑provided time budget (default 1000 s) and prints
   progress information.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper: Euclidean projection onto the probability simplex
# ----------------------------------------------------------------------
def simplex_projection(v: np.ndarray) -> np.ndarray:
    """Project vector v onto the simplex {w | w_i >= 0, sum(w)=1}."""
    n = v.shape[0]
    if n == 0:
        return v
    # sort v in decreasing order
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):                     # all non‑positive
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1] - 1                  # zero‑based index
    theta = cssv[cond][-1] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Objective (fast version – works on vectors that already sum to 1)
# ----------------------------------------------------------------------
def compute_objective(a: np.ndarray) -> float:
    """Return 2·n·max(conv(a,a)) for a with sum(a)=1."""
    n = a.shape[0]
    # convolution via FFT (real‑valued)
    fft = np.fft.rfft(a, n=2 * n - 1)
    conv = np.fft.irfft(fft * fft, n=2 * n - 1)
    return 2.0 * n * conv.max()


# ----------------------------------------------------------------------
# Sub‑gradient of the maximum‑convolution entry
# ----------------------------------------------------------------------
def max_conv_subgradient(a: np.ndarray) -> np.ndarray:
    """
    Compute a sub‑gradient of  max_k (a*a)[k]  w.r.t. a.
    Returns a vector of the same size as a.
    """
    n = a.shape[0]
    fft = np.fft.rfft(a, n=2 * n - 1)
    conv = np.fft.irfft(fft * fft, n=2 * n - 1)

    max_val = conv.max()
    # numerical tolerance for active set
    tol = max_val * 1e-12 + 1e-14
    active = np.where(conv >= max_val - tol)[0]

    grad = np.zeros_like(a)
    for k in active:
        low = max(0, k - n + 1)
        high = min(n - 1, k)
        idx = np.arange(low, high + 1, dtype=int)
        partner = k - idx
        # each pair contributes 2*a[partner] to the gradient of a[idx]
        grad[idx] += 2.0 * a[partner]

    if active.size > 1:
        grad *= 1.0 / active.size
    return grad


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(budget_s: int = 1000) -> list:
    """
    Search for a low‑score sequence of non‑negative heights.
    The function runs for up to ``budget_s`` seconds and returns the
    best sequence found (as a plain Python list of floats).
    """
    # ------------------------------------------------------------------
    # 0. Preparations & time limit
    # ------------------------------------------------------------------
    np.random.seed(0)   # reproducibility
    safety = 5.0                     # a small safety margin
    deadline = time.time() + max(budget_s - safety, 1)

    # ------------------------------------------------------------------
    # 1. Problem size and initial point
    # ------------------------------------------------------------------
    try:
        # use the previously discovered construction if available
        init = np.asarray(height_sequence_1, dtype=float)   # type: ignore[name-defined]
        n = init.shape[0]
    except Exception:
        # fall back to a moderate size that works well within the time budget
        n = 5400
        init = np.full(n, 1.0 / n)

    # Normalise to the simplex (Σ a = 1)
    if init.sum() <= 0:
        a = np.full(n, 1.0 / n)
    else:
        a = init / init.sum()

    # ------------------------------------------------------------------
    # 2. Fast parametric sweep (low‑dimensional families)
    # ------------------------------------------------------------------
    best_a = a.copy()
    best_score = compute_objective(best_a)

    # Helper to evaluate and possibly replace the current best
    def consider(candidate: np.ndarray):
        nonlocal best_a, best_score
        s = compute_objective(candidate)
        if s < best_score:
            best_score = s
            best_a = candidate.copy()
            print(f"[param sweep] new best = {best_score:.6f}")

    # (a) Power‑law decreasing a_i ∝ (i+1)^{-p}
    for p in np.linspace(0.0, 4.0, 41):            # 0,0.1,…,4
        w = (np.arange(1, n + 1, dtype=float) ** (-p))
        w /= w.sum()
        consider(w)

    # (b) Exponential decreasing a_i ∝ r^i
    for r in np.linspace(0.995, 0.55, 46):        # ~0.995 … 0.55
        w = (r ** np.arange(n, dtype=float))
        w /= w.sum()
        consider(w)

    # (c) Linear decreasing (triangular) shape
    w = (np.arange(n, 0, -1, dtype=float))       # n, n‑1, …,1
    w /= w.sum()
    consider(w)

    # (d) Quadratic decreasing a_i ∝ (n-i)^2
    w = ((np.arange(n, 0, -1, dtype=float)) ** 2)
    w /= w.sum()
    consider(w)

    # (e) Asymmetric power‑law: a_i ∝ (i+1)^{-p}·(n-i)^{-q}
    alphas = np.linspace(0.0, 2.0, 9)
    betas  = np.linspace(0.0, 2.0, 9)
    for p in alphas:
        for q in betas:
            i = np.arange(n, dtype=float) + 1          # 1 … n
            w = (i ** (-p)) * ((n - i + 1) ** (-q))
            if w.sum() == 0:
                continue
            w /= w.sum()
            consider(w)

    # ------------------------------------------------------------------
    # 3. Projected Adam optimisation (sub‑gradient descent on the simplex)
    # ------------------------------------------------------------------
    a = best_a.copy()
    best_score = compute_objective(a)
    best_a = a.copy()

    # Adam hyper‑parameters
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    lr0 = 0.5                     # base learning rate (will be scaled)

    m = np.zeros_like(a)          # first moment
    v = np.zeros_like(a)          # second moment
    iteration = 0

    while time.time() < deadline:
        # ---- sub‑gradient of the current max‑convolution entry ----
        grad = max_conv_subgradient(a)

        # ---- Adam update (with projection onto the simplex) ----
        iteration += 1
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)

        m_hat = m / (1.0 - beta1 ** iteration)
        v_hat = v / (1.0 - beta2 ** iteration)

        # time‑decaying learning rate – helps with the diminishing‑step‑size nature
        lr = lr0 / np.sqrt(iteration)

        step = lr * m_hat / (np.sqrt(v_hat) + eps)
        a_candidate = simplex_projection(a - step)

        # ---- evaluate ----
        cur_score = compute_objective(a_candidate)

        # keep the best ever seen
        if cur_score < best_score:
            best_score = cur_score
            best_a = a_candidate.copy()
            print(f"Iter {iteration:6d}: improved score = {best_score:.6f}")

        # accept the candidate (even if not improving) to keep the optimiser moving
        a = a_candidate

        # ---- occasional jitter to escape plateaus ----
        if iteration % 3000 == 0:
            jitter = 0.005 * np.random.randn(n)
            a = simplex_projection(a + jitter)

        # ---- early‑stop if we have already reached the target ----
        if best_score < 1.503:
            print(f"Target threshold crossed at iteration {iteration}")
            break

    # ------------------------------------------------------------------
    # 4. Return the best sequence as a plain list
    # ------------------------------------------------------------------
    # Convert to plain Python floats (evaluate_sequence will later clamp)
    return [float(v) for v in best_a.tolist()]
