"""
Propose a step‑function (non‑negative vector) with a low
evaluate_sequence() score.

The algorithm works in two phases:

1. **Smoothed max‑convolution minimisation** – a smooth
   approximation of the objective  max(conv(a,a)) is optimised
   by Adam.  The smoothness is controlled by a temperature β which
   is slowly increased, turning the loss from a soft‑max into the
   true max.  After each Adam step the vector is projected onto the
   simplex {x ≥ 0, Σx = √(2 n)} (the usual normalisation).

2. **Linear‑programming polishing** – the current vector f is used
   to build the Toeplitz matrix A such that A·g = f * g.
   We solve a small LP that maximises Σ g_i under the constraints
   (f * g)_k ≤ M where M = max(f * f).  The solution g is rescaled
   to the target ℓ₁‑norm and mixed with f (line‑search for the best
   mixing weight).  Several LP‑updates are performed until no
   further improvement is observed.

Both phases respect the security checks of ``evaluate_sequence``.
The routine runs for at most the supplied time budget (≈ 1000 s) and
returns the best sequence discovered.
"""

import time
import math
import numpy as np

# --------------------------------------------------------------------------- #
# Simplex projection (Duchi et al. 2008)
def projection_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Project vector v onto the simplex {x ≥ 0, Σx = s}."""
    if s < 0:
        raise ValueError("simplex sum must be non‑negative")
    n = v.shape[0]
    # sort v in descending order
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.where(u - cssv / (np.arange(n) + 1.0) > 0)[0]
    if rho.size == 0:
        return np.zeros_like(v)
    rho = rho[-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# --------------------------------------------------------------------------- #
def _fast_score(seq: np.ndarray) -> float:
    """Fast evaluation of the objective 2·n·max(conv(seq,seq)) / (Σseq)²."""
    n = seq.size
    fft_len = 1 << ((2 * n - 1).bit_length())          # next power of two
    F = np.fft.rfft(seq, n=fft_len)
    conv = np.fft.irfft(F * F, n=fft_len)[: 2 * n - 1]
    max_conv = conv.max()
    s = seq.sum()
    # denominator never vanishes because we keep a proper ℓ₁‑norm
    return float(2.0 * n * max_conv / (s * s))

    The function respects the time budget and returns the best list found.
    """

    fft_len = 1 << ((3 * n - 2).bit_length())

    # --------------------------------------------------------------- #
    # Initialise with a good seed.
    try:

        if hs.shape[0] != n:

        lam = 5e-4
        r = 1.0 - lam
        a = np.array([r ** i for i in range(n)], dtype=np.float64)

    a = projection_simplex(a, target_sum)

    # keep the best vector seen so far
    best_a = a.copy()
    best_score = _fast_score(best_a)

    # --------------------------------------------------------------- #
    # Adam optimiser on a smoothed max‑convolution loss
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8
    m = np.zeros_like(a)
    v = np.zeros_like(a)

    # smoothing (soft‑max) temperature
    beta_s = 5.0                  # start with a fairly smooth approximation
    beta_target = 200.0           # final temperature (close to true max)
    # increase β linearly – we will let it grow for a few thousand steps
    beta_step = (beta_target - beta_s) / 5000.0
