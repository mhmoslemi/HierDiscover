def solve_lp_for_b(a, M):
    n = len(a)
    A = build_A(a) # (2n-1, n)
    b_vec = M * np.ones(2*n-1)
    # Use linprog: minimize c^T x, where c = - np.ones(n) for max sum.
    res = scipy.optimize.linprog(c=-np.ones(n), A_ub=A, b_ub=b_vec, bounds=[(0, None)]*n, method='highs')
    if not res.success:
        # fallback: return zeros
        return np.zeros(n)
    return res.x

"""
Propose a low‑score step‑function using an *alternating linear‑program* 
scheme.  For a current vector f we compute M = max(f*f).  Holding M fixed
we solve the LP

    maximize   Σ b_i
    subject to b_i ≥ 0
               (f * b)_k ≤ M   for all k

which finds a vector g that can be mixed with f without increasing the
self‑convolution peak.  Repeating this update (with a small mixing step
or a full replacement) quickly drives the peak down.  After the LP
phase we polish the result with a cheap stochastic hill‑climb.
The whole search respects the 1000 s budget and works for lengths up
to a few thousand entries.
"""

import time
import math
import numpy as np

# ------------------------------------------------------------
# Basic linear‑algebra utilities
# ------------------------------------------------------------

def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full (linear) convolution using a real FFT."""
    m = len(a) + len(b) - 1
    n_fft = next_pow2(m)
    A = np.fft.rfft(a, n=n_fft)
    B = np.fft.rfft(b, n=n_fft)
    C = A * B
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex Σx_i = 1, x_i ≥ 0."""
    n = v.shape[0]
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / n)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    rho = np.where(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def compute_score(a: np.ndarray) -> float:
    """Score identical to `evaluate_sequence` for vectors on the simplex."""
    n = a.shape[0]
    conv = convolution_fft(a, a)
    max_val = float(np.max(conv))
    s = float(np.sum(a))
    if s < 1e-12:
        return float("inf")
    return 2.0 * n * max_val / (s * s)


def arcsine_initial(n: int) -> np.ndarray:
    """Arcsine‑shaped start: a_i ∝ 1 / sqrt((i+0.5)(n‑0.5‑i))."""
    i = np.arange(n, dtype=float)
    eps = 0.25
    vals = 1.0 / np.sqrt((i + 0.5) * (n - 0.5 - i) + eps)
    return vals / np.sum(vals)

# Linear‑programming core
# ------------------------------------------------------------

try:
    from scipy.optimize import linprog
    _LP_AVAILABLE = True
except Exception:      # pragma: no cover
    _LP_AVAILABLE = False
    linprog = None


def build_convolution_matrix(a: np.ndarray) -> np.ndarray:
    """
    Build the matrix A such that (A @ b)_k = (a * b)_k,
    where * denotes linear convolution.
    Shape: (2·n‑1, n)
    """
    n = a.shape[0]
    rows = 2 * n - 1
    A = np.zeros((rows, n), dtype=np.float64)
    # column‑wise filling: column j contains a shifted by j positions
    for j in range(n):
        A[j:j + n, j] = a
    return A


def solve_lp_for_b(a: np.ndarray, M: float) -> np.ndarray:
    """
    Solve   max Σ b_i   s.t.   b_i ≥ 0,   (a * b)_k ≤ M  (for all k).
    Returns the optimal b (may be all zeros if LP fails).
    """
    n = a.shape[0]
    rows = 2 * n - 1
    A = build_convolution_matrix(a)               # (2n‑1) × n
    ub = np.full(rows, M, dtype=np.float64)       # RHS vector

    # maximise Σ b_i  ⇒ minimise -Σ b_i
    c = -np.ones(n, dtype=np.float64)

    # bounds for each variable: b_i ≥ 0, no upper bound
    bounds = [(0, None)] * n

    res = linprog(c, A_ub=A, b_ub=ub, bounds=bounds,
                  method='highs', options={'presolve': True})
    # free the matrix early
    del A

    if not res.success:
        # LP failed – return a zero vector to keep the caller safe.
        return np.zeros(n, dtype=np.float64)
    return res.x

def improve_via_lp(initial_a: np.ndarray,
                   deadline: float,
                   max_iters: int = 25) -> tuple[np.ndarray, float]:
    """
    Repeatedly:
       * compute M = max(a*a)
       * solve the LP for b with that M
       * either replace a by the normalised b or mix a←(1‑t)a+t·b
    The best vector (lowest true score) encountered is returned.
    """
    a = project_to_simplex(initial_a.astype(float).copy())
    best_a = a.copy()
    best_score = compute_score(a)

    for it in range(1, max_iters + 1):
        if time.time() >= deadline:
            break

        # current bound
        conv_a = convolution_fft(a, a)
        M = float(np.max(conv_a))

        # LP step
        b = solve_lp_for_b(a, M)
        sum_b = float(b.sum())
        if sum_b <= 0.0:
            break

        # normalised LP solution
        g = b / sum_b

        # candidate 1 – direct replacement
        cand1 = project_to_simplex(g)
        score1 = compute_score(cand1)

        # candidate 2 – convex mixing (try a few step sizes)
        improved = False
        for t in (0.5, 0.2, 0.1, 0.05, 0.02):
            cand = project_to_simplex((1.0 - t) * a + t * g)
            sc = compute_score(cand)
            if sc < best_score:
                a = cand
                best_a = cand.copy()
                best_score = sc
                improved = True
                print(f"    LP iter {it:2d} mix t={t:4.2f}  score {sc:.6f}")
                break
        if improved:
            continue

        # try direct replacement if it improves
        if score1 < best_score:
            a = cand1
            best_a = cand1.copy()
            best_score = score1
            print(f"    LP iter {it:2d} replace      score {score1:.6f}")
            continue

        print(f"    LP iter {it:2d} no improvement, stop.")

    return best_a, best_score

def hill_climb(seq, deadline: float, max_steps: int = 30000,
               init_step: float = 0.08) -> tuple[list[float], float]:
    """Small random mass swaps that keep the simplex constraint."""
    a = np.array(seq, dtype=float)
    a = project_to_simplex(a)
    best_a = a.copy()
    best_score = compute_score(best_a)
    n = a.shape[0]
    step = init_step

    while max_steps > 0 and time.time() < deadline:
        i, j = np.random.randint(n, size=2)
        if i == j:
            continue
        # move a fraction of mass from i to j
        delta = step * best_a[i]
        if delta <= 0:
            continue
        cand = best_a.copy()
        cand[i] -= delta
        cand[j] += delta
        if cand[i] < 0:
            continue
        cand = project_to_simplex(cand)
        cur = compute_score(cand)
        if cur < best_score:
            best_score = cur
            best_a = cand
            step = min(step * 1.07, 0.5)   # slowly grow step on success
        else:
            step *= 0.97                    # shrink on failure
        max_steps -= 1
    return best_a.tolist(), best_score

    """
    Search for a non‑negative sequence that minimises `evaluate_sequence`.
    The search combines an alternating LP refinement with a final hill‑climb.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 2.0        # small safety margin

    best_seq = None
    best_val = float("inf")

    # ------------------------------------------------------------------
    # If a previous construction is present, use it as a quick fallback.
    # ------------------------------------------------------------------
    try:
        fallback = [float(x) for x in height_sequence_1]   # type: ignore[name‑defined]
        fallback_val = evaluate_sequence(fallback)
        print(f"Fallback score = {fallback_val:.6f}")
        if fallback_val < best_val:
            best_val = fallback_val
            best_seq = fallback
    except Exception:
        pass

    # ------------------------------------------------------------------
    # Lengths to explore (keep them modest – the LP matrix grows as O(n²)).
    # ------------------------------------------------------------------
    lengths = [512, 1024, 2048]

    for n in lengths:
        if time.time() >= deadline - 5.0:
            break
        print(f"\n=== Optimising length n = {n} ===")

        # Three initialists per length
        inits = [
            arcsine_initial(n),
            np.full(n, 1.0 / n),
            np.random.dirichlet(np.ones(n)),
        ]

        for idx, init_a in enumerate(inits, start=1):
            if time.time() >= deadline - 5.0:
                break
            if not _LP_AVAILABLE:
                print("  SciPy not available – skipping LP optimisation.")
                continue
            print(f"  start {idx}/{len(inits)} (LP phase)")
            seq, val = improve_via_lp(init_a, deadline, max_iters=20)
            print(f"    LP best score = {val:.6f}")
            if val < best_val:

                best_seq = seq.tolist()
                print(f"    *** New global best {best_val:.6f} (n={n}) ***")

    # ------------------------------------------------------------------
    # Final cheap stochastic refinement (if time permits).
    # ------------------------------------------------------------------
    if best_seq is not None and time.time() < deadline - 2.0:
        print("\nRefining best sequence with hill‑climb")
        refined_seq, refined_val = hill_climb(best_seq,
                                              deadline,
                                              max_steps=50000,
                                              init_step=0.05)
        print(f"    refined score = {refined_val:.6f}")
        if refined_val < best_val:
            best_val = refined_val
            best_seq = refined_seq

    # ------------------------------------------------------------------
    # Fallback in the extremely unlikely event that nothing was produced.
    # ------------------------------------------------------------------
    if best_seq is None:
        best_seq = [1.0] * 256
        best_val = evaluate_sequence(best_seq)

    final_score = evaluate_sequence(best_seq)
    print(f"\nFinished search. Final reported score = {final_score:.6f}")

    # The harness expects a plain Python list of floats.
    return best_seq
