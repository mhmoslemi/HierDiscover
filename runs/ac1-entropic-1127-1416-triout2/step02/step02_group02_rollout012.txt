def correlation_grad(a, w, L):
    # a length n, w length 2n-1
    a_rev_fft = np.fft.rfft(a[::-1], n=L)
    w_fft = np.fft.rfft(w, n=L)
    corr_full = np.fft.irfft(a_rev_fft * w_fft, n=L)
    n = a.shape[0]
    return 2.0 * corr_full[n-1:n-1+n]
```

"""
Propose a candidate step‑function sequence that improves the current best bound.
The algorithm starts from the globally stored `height_sequence_1` (if present)
or from a uniform distribution, normalises it to a probability simplex,
and then performs projected Adam optimisation on a smooth approximation of
the max‑convolution objective.

The smoothing uses a soft‑max (temperature `γ`) over the auto‑convolution
entries so that the gradient accounts for many near‑max constraints.
`γ` is initialised as a fraction of the current maximum and annealed
geometrically; this gives a coarse global search early on and refines
to the true sub‑gradient later.  Each Adam step is projected back onto the
simplex (non‑negative entries that sum to one).  Periodic tiny random
jitter helps escape shallow plateaus.  The best vector encountered is
returned as a plain Python list.

The whole routine respects the 1000 s time budget and prints progress.
"""

import time
import numpy as np

def simplex_projection(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1] - 1
    theta = cssv[cond][-1] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def next_pow2(x: int) -> int:
    """Return the smallest power of two >= x."""
    return 1 << (x - 1).bit_length()


def auto_convolution(a: np.ndarray) -> np.ndarray:
    """Full auto‑convolution of a using FFT (real input)."""
    n = a.shape[0]
    fft = np.fft.rfft(a, n=2 * n - 1)
    conv = np.fft.irfft(fft * fft, n=2 * n - 1)
    return conv


def correlation_gradient(a: np.ndarray, w: np.ndarray, L: int) -> np.ndarray:
    """
    Gradient of the smooth max (∑ w_k (a*a)_k) w.r.t. a.
    It equals 2 * (a reversed) ⋆ w, i.e. a correlation.
    """
    # a_rev = a reversed
    a_rev_fft = np.fft.rfft(a[::-1], n=L)
    w_fft = np.fft.rfft(w, n=L)
    corr_full = np.fft.irfft(a_rev_fft * w_fft, n=L)
    n = a.shape[0]
    # correlation entries correspond to indices n-1 … n-1+n-1
    grad = 2.0 * corr_full[n - 1 : n - 1 + n]
    return grad


def propose_candidate(budget_s: int = 1000) -> list:
    """Search for a low‑score step‑function sequence within `budget_s` seconds."""
    deadline = time.time() + budget_s - 5.0  # small safety margin

    # ----------------------------------------------------------------------
    # 1. Problem size & initial point
    # ----------------------------------------------------------------------
    try:
        # Use the best known sequence if the global variable exists
        init = np.asarray(height_sequence_1, dtype=float)  # type: ignore[name-defined]
        n = init.shape[0]
    except Exception:
        # Fallback: moderate size, all‑positive uniform start
        n = 5146
        init = np.full(n, 1.0 / n)

    a = init.copy()
    total = a.sum()
    if total <= 0:
        a = np.full(n, 1.0 / n)
    else:
        a = a / total  # project onto the simplex

    # ----------------------------------------------------------------------
    # 2. Optimisation hyper‑parameters
    # ----------------------------------------------------------------------
    best_a = a.copy()
    conv = auto_convolution(a)
    best_score = 2.0 * n * conv.max()
    print(f"Initial score = {best_score:.6f}")

    # Adam state
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1, beta2 = 0.9, 0.999
    eps = 1e-12
    lr = 0.3  # base learning‑rate

    # smoothing temperature (soft‑max temperature)
    gamma = None
    gamma_min = 1e-12
    gamma_decay = 0.995  # multiplicative decay per iteration

    # length for the correlation FFT (>= 3n)
    L = next_pow2(3 * n)

    iteration = 0
    # ----------------------------------------------------------------------
    # 3. Main optimisation loop
    # ----------------------------------------------------------------------
    while time.time() < deadline:
        # ---- auto‑convolution and current maximum ----
        fft_a = np.fft.rfft(a, n=2 * n - 1)
        conv = np.fft.irfft(fft_a * fft_a, n=2 * n - 1)
        max_val = conv.max()

        # ---- initialise / anneal soft‑max temperature ----
        if gamma is None:
            gamma = max_val * 0.5 + 1e-15
        else:
            gamma = max(gamma * gamma_decay, gamma_min)

        # ---- soft‑max weights over the convolution entries ----
        # (conv - max_val) <= 0, so exponentials are well‑behaved.
        exp_vals = np.exp((conv - max_val) / gamma)
        w = exp_vals / exp_vals.sum()

        # ---- compute gradient via correlation (FFT) ----
        grad = correlation_gradient(a, w, L)

        # ---- Adam update (projected) ----
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)

        m_hat = m / (1.0 - beta1 ** (iteration + 1))
        v_hat = v / (1.0 - beta2 ** (iteration + 1))

        step = lr * m_hat / (np.sqrt(v_hat) + eps)
        a_new = simplex_projection(a - step)

        a = a_new

        # ---- keep the best solution seen so far ----
        cur_score = 2.0 * n * max_val
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"Iter {iteration:6d}: improved score = {best_score:.6f}")

            if best_score < 1.503:
                print("Target achieved – early termination.")
                break

        # ---- occasional tiny random jitter to escape plateaus ----
        if iteration % 2000 == 0 and iteration > 0:
            jitter = np.random.randn(n) * 1e-4
            a = simplex_projection(a + jitter)

        iteration += 1

    # ----------------------------------------------------------------------
    # 4. Return best sequence as plain Python list
    # ----------------------------------------------------------------------
    return [float(x) for x in best_a.tolist()]
