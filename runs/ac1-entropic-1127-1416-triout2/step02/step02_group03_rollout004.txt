"""
Projected gradient descent with a smooth approximation of the ‑max‑convolution.
The algorithm works in two stages.

Stage 1 – warm‑start.
    • If a global variable ``height_sequence_1`` exists we reuse its length and
      values (projected onto the simplex of sum √(2·n)).
    • Otherwise we draw a few random members of the 4‑parameter power‑law family
      a_i = (i+α)^(-p)·(1+β·i/(n‑1)), keep the best (according to the exact
      max‑convolution) and project it onto the simplex.

Stage 2 – optimisation.
    • At each iteration we compute the full linear convolution a*a by a real FFT.
    • A smooth maximum M_τ(a) = τ·log∑_k exp(conv_k/τ) is used; its gradient is
      g = 2·(w * a_rev)[n‑1:n‑1+n] where w are the soft‑max weights
        w_k = exp((conv_k‑max)/τ)/∑ exp(...)
      and a_rev is a reversed copy of a.  The convolution needed for the gradient
      is also performed with a real FFT.
    • An adaptive step size η_t = η₀/√t is taken opposite to the gradient and the
      result is projected back onto the simplex {x≥0, Σx = √(2·n)}.
    • The smoothing parameter τ is exponentially decreased (τ←max(τ·ρ, τ_min))
      so the optimisation gradually focuses on the true maximum.
    • Occasionally a tiny random jitter is added to escape plateaus.

The best sequence found (according to the exact evaluation
2·n·max(conv)/sum², which equals max(conv) because Σa = √(2·n)) is returned.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Simplex projection (Duchi et al. 2008)
# ----------------------------------------------------------------------
def project_simplex(v: np.ndarray, target_sum: float) -> np.ndarray:
    """Euclidean projection of v onto {x ≥ 0, Σx = target_sum}."""
    if target_sum < 0:
        raise ValueError("target_sum must be non‑negative")
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - target_sum
    rho = np.where(u - cssv / np.arange(1, len(v) + 1) > 0)[0]
    if rho.size == 0:
        return np.zeros_like(v)
    rho = rho[-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# FFT‑based convolution utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Return the smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def full_convolution_fft(a: np.ndarray) -> np.ndarray:
    """
    Linear convolution a*a (full length 2·len(a)‑1) via a real FFT.
    """
    n = len(a)
    size = _next_pow2(2 * n - 1)
    Fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(Fa * Fa, n=size)[: 2 * n - 1]
    return conv


def gradient_from_weights(a: np.ndarray, w: np.ndarray, fft_size: int,
                         slice_start: int) -> np.ndarray:
    """
    Compute ∇_a ( Σ_k w_k·(a*a)_k ) = 2·B·a,
    where B_{ij}=w_{i+j}.  The result is obtained by a single
    convolution of w with the reversed a.
    """
    rev = a[::-1]
    Fw = np.fft.rfft(w, n=fft_size)
    Fr = np.fft.rfft(rev, n=fft_size)
    full = np.fft.irfft(Fw * Fr, n=fft_size)
    # slice corresponds to indices n‑1 … n‑1+n‑1
    grad = 2.0 * full[slice_start:slice_start + len(a)]
    return grad


# ----------------------------------------------------------------------
# Candidate generator (power‑law + linear tilt)
# ----------------------------------------------------------------------
def make_candidate(n: int, p: float, offset: float, beta: float) -> np.ndarray:
    """
    Build a raw sequence a_i = (i+1+offset)^{-p}·(1+β·i/(n‑1)),
    clip negatives, and project onto the simplex of sum √(2·n).
    """
    idx = np.arange(n, dtype=float)
    base = np.power(idx + 1.0 + offset, -p)
    factor = 1.0 + beta * idx / (n - 1)
    raw = np.maximum(base * factor, 0.0)
    target = np.sqrt(2.0 * n)
    return project_simplex(raw, target)


# ----------------------------------------------------------------------
# Main search routine
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 123, budget_s: int = 1000, **kwargs):
    """
    Return a list of non‑negative heights that (hopefully) beats the
    current best evaluation value of ≈1.503.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 2.0   # safety margin

    # --------------------------------------------------------------
    # Choose length n (reuse known good length if possible)
    # --------------------------------------------------------------
    hs_global = globals().get("height_sequence_1")
    if isinstance(hs_global, list) and len(hs_global) > 0:
        n = len(hs_global)
    else:
        n = 10000                     # a comfortable size for the budget

    target_sum = np.sqrt(2.0 * n)

    # --------------------------------------------------------------
    # Warm‑start: try to use the global sequence, otherwise a random
    # power‑law candidate (few random draws to pick a good start).
    # --------------------------------------------------------------
    if isinstance(hs_global, list) and len(hs_global) == n:
        a = np.array(hs_global, dtype=float)
        a = project_simplex(a, target_sum)
        best_seq = a.copy()
        best_score = full_convolution_fft(a).max()
        print(f"[init] using global height_sequence_1  n={n}  score={best_score:.6f}")
    else:
        # cheap random sweep over the 4‑parameter family
        best_seq = None
        best_score = np.inf
        for _ in range(80):   # 80 candidates – cheap and fast
            p = np.random.uniform(0.35, 0.65)
            off = np.random.uniform(0.0, 12.0)
            beta = np.random.uniform(-0.9, 2.5)
            cand = make_candidate(n, p, off, beta)
            score = full_convolution_fft(cand).max()
            if score < best_score:
                best_score = score
                best_seq = cand
        a = best_seq.copy()
        print(f"[init] random sweep best  n={n}  score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Pre‑compute FFT size for the gradient step (w length = 2n‑1,
    # a_rev length = n ⇒ convolution length = 3n‑2)
    # ------------------------------------------------------------------
    grad_fft_size = _next_pow2(3 * n - 2)
    slice_start = n - 1   # start index of the slice that yields B·a

    # ------------------------------------------------------------------
    # Optimisation parameters
    # ------------------------------------------------------------------
    step0 = 0.6               # base step size for η_t = step0/√t
    tau = 0.5                 # initial smoothing parameter
    tau_decay = 0.9995        # exponential decay per iteration
    tau_min = 1e-4
    iteration = 0
    best_overall = best_seq.copy()
    best_overall_score = best_score

    # ------------------------------------------------------------------
    # Main optimisation loop
    # ------------------------------------------------------------------
    while time.time() < deadline:
        iteration += 1

        # ----- compute exact convolution and current max -----
        conv = full_convolution_fft(a)
        current_max = conv.max()

        # Keep the best encountered sequence.
        if current_max < best_overall_score:
            best_overall_score = current_max
            best_overall = a.copy()
            print(f"[sgd] iter {iteration:6d}  new best {best_overall_score:.6f}  τ={tau:.6f}")

            if best_overall_score < 1.503:   # target reached
                break

        # ----- smooth maximum (soft‑max) -----
        # stable soft‑max: shift by the current maximum
        shift = (conv - current_max) / tau
        # exponential may under‑flow for very negative values – that is fine
        exp_shift = np.exp(shift)
        w = exp_shift / exp_shift.sum()       # length 2n‑1, Σ w = 1

        # ----- gradient of the smoothed objective -----
        g = gradient_from_weights(a, w, grad_fft_size, slice_start)

        # ----- step size (decaying like 1/√t) -----
        eta = step0 / np.sqrt(iteration)

        # ----- gradient descent + projection onto the simplex -----
        a = a - eta * g
        a = project_simplex(a, target_sum)

        # ----- anneal the smoothing parameter -----
        tau = max(tau_min, tau * tau_decay)

        # ----- occasional tiny jitter to avoid stagnation -----
        if iteration % 250 == 0 and np.random.rand() < 0.3:
            jitter = (np.random.rand(n) - 0.5) * 0.02 * a   # up to ±1 % of current values
            a += jitter
            a = project_simplex(a, target_sum)

    # ------------------------------------------------------------------
    # Final report
    # ------------------------------------------------------------------
    print(f"Finished after {iteration} iterations – best score {best_overall_score:.6f}")
    return best_overall.tolist()
