def propose_candidate(budget_s=1000):

    start = time.time()

    # set initial n based on height_sequence_1 if exists else some default (probably ~5000)

    # function to normalize to simplex
    def normalize(v):
        s = v.sum()
        if s <=0:
            return np.full_like(v, 1.0/v.shape[0])
        return v / s

    # make initial distribution
    if 'height_sequence_1' in globals():
        a = normalize(np.asarray(height_sequence_1, dtype=float))
    else:
        a = np.full(n, 1.0/n)

    # optionally add small random noise:
    a = normalize(a + 0.001 * np.random.rand(n))

    # Adam parameters
    beta1=0.9
    beta2=0.999
    eps=1e-8
    lr=0.02

    m = np.zeros_like(a)
    v = np.zeros_like(a)
    t=0

    best_a = a.copy()
    best_score = compute_objective(a)

    # main loop
    while time.time() < deadline:
        t += 1
        # compute conv, gradient
        fft = np.fft.rfft(a, n=2*n-1)
        conv = np.fft.irfft(fft*fft, n=2*n-1)
        max_val = conv.max()
        k_max = int(conv.argmax())
        # find active set
        tol = max_val*1e-12 + 1e-14
        active = np.where(conv >= max_val - tol)[0]

        # gradient
        grad = np.zeros_like(a)
        for k in active:
            lo = max(0, k-(n-1))
            hi = min(n-1, k)
            idx = np.arange(lo, hi+1, dtype=int)
            partner = k - idx
            grad[idx] += 2.0 * a[partner]
        if active.size > 1:
            grad /= active.size

        # Adam update
        m = beta1 * m + (1-beta1) * grad
        v = beta2 * v + (1-beta2) * (grad*grad)
        m_hat = m / (1 - beta1**t)
        v_hat = v / (1 - beta2**t)
        step = lr * m_hat / (np.sqrt(v_hat) + eps)
        a_new = a - step
        # projection to simplex
        a = simplex_projection(a_new)

        # jitter every 5000 iterations to escape stagnation
        if t % 5000 == 0:
            jitter = 0.005 / np.sqrt(t)
            a = simplex_projection(a + jitter * np.random.randn(n))

        # evaluate objective occasionally

            cur_score = compute_objective(a)
            if cur_score < best_score:
                best_score = cur_score
                best_a = a.copy()

        if best_score < 1.503:
            print(f"Target reached at iter {t}")
            break

    # final result
    return [float(v) for v in best_a.tolist()]

"""
Propose a step‑function sequence that (approximately) minimises

    2·n·max(conv(a, a)) / (sum(a)**2)

The algorithm works in two phases:

1. **Projected Adam optimisation** on the probability simplex.
   An Adam step is taken, then the vector is Euclidean‑projected onto
   {a_i ≥ 0, Σ a_i = 1}.  The sub‑gradient of the max‑entry of the
   self‑convolution is used (the same sub‑gradient as in the reference
   code).  Small stochastic “jitter’’ steps are added periodically
   to escape flat plateaus.

2. **Occasional hill‑climbing moves** – a random transfer of a tiny
   fraction of mass from a source index to a distant target index.
   The move is kept only if it improves the objective.  This provides
   a cheap global‑exploration mechanism that complements the local
   Adam steps.

The optimisation is run on several problem sizes (the size of the
previous best sequence, plus a few powers of two).  At the end of each
stage the current best solution is up‑sampled (linear interpolation) to
the next larger size, so that the search can reuse useful structure.
The routine stops as soon as the target value 1.503 is reached or when
the allotted time (default 1000 s) is exhausted.

The returned list contains non‑negative numbers that sum to 1; the
evaluation function `evaluate_sequence` is scale‑invariant, so the
normalisation does not affect the score.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper functions – all are defined at top level (no closures)
# ----------------------------------------------------------------------


def simplex_projection(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the probability simplex:
        { w | w_i ≥ 0 , Σ w_i = 1 }.
    Implementation follows Duchi et al. (2008).
    """
    n = v.shape[0]
    # sort v in decreasing order
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    # condition for finding the threshold
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all entries non‑positive → uniform distribution
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1] - 1               # zero‑based index
    theta = cssv[cond][-1] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def compute_objective(a: np.ndarray) -> float:
    """
    Return the evaluation function for a vector satisfying Σ a = 1.
    Equivalent to evaluate_sequence after the same normalisation:
        2·n·max(conv(a, a))
    """
    n = a.shape[0]
    fft = np.fft.rfft(a, n=2 * n - 1)
    conv = np.fft.irfft(fft * fft, n=2 * n - 1)
    max_b = conv.max()
    return 2.0 * n * max_b


def convolution_and_active(a: np.ndarray):
    """
    Compute the self‑convolution of a (via FFT) and return:
        conv    – the full convolution array (length 2·n‑1)
        max_val – maximal entry of conv
        active  – indices where conv is (numerically) maximal
    """
    n = a.shape[0]
    fft = np.fft.rfft(a, n=2 * n - 1)
    conv = np.fft.irfft(fft * fft, n=2 * n - 1)
    # numerical noise may produce tiny negative values
    np.maximum(conv, 0.0, out=conv)
    max_val = conv.max()
    tol = max_val * 1e-12 + 1e-14
    active = np.where(conv >= max_val - tol)[0]
    return conv, max_val, active


def subgradient_max_conv(a: np.ndarray, n: int, conv: np.ndarray,
                         active: np.ndarray) -> np.ndarray:
    """
    Sub‑gradient of the maximal convolution entry with respect to a.
    For each active index k we add the contribution 2·a_{k‑i}
    for all i that participate in (a*a)_k, then average over the
    active set.
    """
    grad = np.zeros_like(a)
    for k in active:
        lo = max(0, k - (n - 1))
        hi = min(n - 1, k)
        idx = np.arange(lo, hi + 1, dtype=int)
        partner = k - idx
        grad[idx] += 2.0 * a[partner]
    if active.size > 1:
        grad /= float(active.size)
    return grad


def random_mass_move(a: np.ndarray, eps_factor: float = 0.05) -> np.ndarray:
    """
    Perform a tiny random transfer of probability mass.
    A source index is chosen with probability proportional to a_i,
    a target index uniformly, and a fraction eps_factor of the source
    weight is moved to the target.  The result is projected onto the
    simplex to keep feasibility.
    """
    n = a.shape[0]
    if np.all(a == 0):
        return a
    src = np.random.choice(n, p=a)          # more likely to move from big entries
    tgt = np.random.randint(0, n - 1)       # uniform target
    if tgt >= src:
        tgt += 1
    eps = eps_factor * a[src]
    if eps <= 0.0:
        return a
    a_new = a.copy()
    a_new[src] -= eps
    a_new[tgt] += eps
    # project to enforce non‑negativity & Σ=1
    return simplex_projection(a_new)


def upscale_sequence(seq: np.ndarray, new_n: int) -> np.ndarray:
    """
    Linear interpolation of a probability vector to a larger length.
    The result is renormalised to sum to 1.
    """
    old_n = seq.shape[0]
    if old_n == new_n:
        return seq.copy()
    x_old = np.linspace(0.0, 1.0, old_n)
    x_new = np.linspace(0.0, 1.0, new_n)
    new_seq = np.interp(x_new, x_old, seq)
    # due to interpolation the sum may drift – renormalise
    new_seq = new_seq / new_seq.sum()
    return new_seq


def optimise_for_n(a_start: np.ndarray, n: int,
                   time_budget: float,
                   target_score: float = 1.503) -> tuple[np.ndarray, float]:
    """
    Run a projected Adam optimiser (with jitter and occasional random
    mass moves) for a given problem size n and a time budget.
    Returns the best vector found and its objective value.
    """
    # Adam hyper‑parameters – chosen empirically
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    lr = 0.02                      # base learning rate for the additive update

    # state variables
    a = a_start.copy()
    m = np.zeros_like(a)
    v = np.zeros_like(a)

    best_a = a.copy()
    best_score = compute_objective(a)

    t = 0
    last_improve_iter = 0
    deadline = time.time() + time_budget

    while time.time() < deadline:
        t += 1

        # --------------------------------------------------------------
        # 1) convolution, active set and sub‑gradient
        # --------------------------------------------------------------
        conv, max_val, active = convolution_and_active(a)
        grad = subgradient_max_conv(a, n, conv, active)

        # --------------------------------------------------------------
        # 2) Adam update + projection onto the simplex
        # --------------------------------------------------------------
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        step = lr * m_hat / (np.sqrt(v_hat) + eps)
        a = simplex_projection(a - step)          # we descend, hence ‘- step’

        # --------------------------------------------------------------
        # 3) occasional jitter (helps to leave flat plateaus)
        # --------------------------------------------------------------
        if t % 5000 == 0:
            eps_jitter = 0.01 / np.sqrt(t)
            a = simplex_projection(a + eps_jitter * np.random.randn(n))

        # --------------------------------------------------------------
        # 4) occasional random mass move (cheap hill‑climbing)
        # --------------------------------------------------------------
        if t % 1500 == 0:
            cand = random_mass_move(a, eps_factor=0.04)
            cand_score = compute_objective(cand)
            cur_score = 2.0 * n * max_val
            if cand_score < cur_score:
                a = cand
                # recompute gradient information for the next loop
                # (skip for speed – we will compute it on the next iteration)

        # --------------------------------------------------------------
        # 5) bookkeeping
        # --------------------------------------------------------------
        if t % 100 == 0:
            cur_score = 2.0 * n * max_val
            if cur_score < best_score:
                best_score = cur_score
                best_a = a.copy()
                last_improve_iter = t
                print(f"n={n:5d} iter={t:6d}  score={best_score:.6f}")

            # early exit if we are already below the target
            if best_score < target_score:
                return best_a, best_score

            # if we have been stagnant for a long time, perform a tiny reset
            if t - last_improve_iter > 20000:
                # re‑initialise momentum while keeping the current a
                m.fill(0.0)
                v.fill(0.0)
                last_improve_iter = t

    return best_a, best_score

    # external timeout kills the process.

        a0 = simplex_projection(a0)   # guarantee feasibility

        # ------------------------------------------------------------------
        # Optimise for the current size
        # ------------------------------------------------------------------
        best_a_stage, best_score_stage = optimise_for_n(
            a0, n, this_budget, target_score=1.503
        )

        # keep track of the global optimum
        if best_score_stage < best_overall_score:
            best_overall_score = best_score_stage
            best_overall_a = best_a_stage.copy()
            print(f"=== New global best after size {n}: score={best_overall_score:.6f}")

        # reduce the remaining time budget
        remaining_time = deadline - time.time()
        if remaining_time < 5.0:
            break
        if best_overall_score < 1.503:
            break

    # ------------------------------------------------------------------
    # Fallback: if for some reason nothing was found, return a uniform vector
    # ------------------------------------------------------------------
    if best_overall_a is None:
        n = base_n
        best_overall_a = np.full(n, 1.0 / n)

    # Convert to plain Python list as required by the evaluation harness
    return [float(v) for v in best_overall_a.tolist()]
