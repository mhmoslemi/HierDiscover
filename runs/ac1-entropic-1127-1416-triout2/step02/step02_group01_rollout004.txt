def lp_step(a, M):
    n = len(a)
    # Build conv matrix
    A = np.zeros((2*n-1, n), dtype=float)
    for i in range(n):

    res = linprog(-np.ones(n), A_ub=A, b_ub=b, bounds=[(0, None)]*n, method='highs')

"""
Alternating linear‑programming (LP) refinement + stochastic hill‑climb.

The objective is scale‑invariant, so we keep the sequence on the
probability simplex (∑a = 1).  For a current vector `a` we compute
M = max(a*a).  The LP

    max   ∑ g_j
    s.t.  (a * g)_k ≤ M      for all convolution indices k
          g_j ≥ 0

asks for the distribution with the largest possible total mass that
still respects the current self‑convolution bound.  If the optimum
has sum(g) > 1 we can scale it down to sum 1, which strictly lowers the
bound M.  Mixing the scaled solution with the old vector yields a new
candidate with a strictly smaller `2·n·max(a*a)`.  Repeating a few
times drives the bound down quickly.

After the LP phase we polish the best candidate with a cheap
pair‑wise stochastic hill‑climb, moving tiny amounts of mass between
coordinates and keeping moves that improve the true `evaluate_sequence`
score.

The whole search respects the 1000 s budget, explores several
lengths (512…4096) and a few random starts, and finally returns the
best sequence found.
"""

import time
import math
import numpy as np
from scipy.optimize import linprog

# ----------------------------------------------------------------------
# Basic helpers
# ----------------------------------------------------------------------


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of a vector onto the probability simplex."""
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / v.shape[0])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho = np.where(u - cssv / np.arange(1, v.shape[0] + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full convolution using a real FFT (fast)."""
    m = len(a) + len(b) - 1
    n_fft = next_pow2(m)
    A = np.fft.rfft(a, n=n_fft)
    B = np.fft.rfft(b, n=n_fft)
    C = A * B
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


def compute_score(a: np.ndarray) -> float:
    """
    Score identical to `evaluate_sequence` when ∑a = 1.
    Returns 2·n·max(a*a).
    """
    n = len(a)
    conv = convolution_fft(a, a)
    max_val = float(np.max(conv))
    return 2.0 * n * max_val


def arcsine_initial(n: int) -> np.ndarray:
    """
    Discrete arcsine-shaped start:
        a_i ∝ 1 / sqrt((i+0.5)*(n-0.5-i))
    """
    i = np.arange(n, dtype=float)
    eps = 0.25
    vals = 1.0 / np.sqrt((i + 0.5) * (n - 0.5 - i) + eps)
    return vals / np.sum(vals)


def uniform_initial(n: int) -> np.ndarray:
    """Uniform distribution on the simplex."""
    return np.full(n, 1.0 / n)


def random_dirichlet_initial(n: int) -> np.ndarray:
    """Random Dirichlet(1,…,1) start."""
    return np.random.dirichlet(np.ones(n))


# ----------------------------------------------------------------------
# Linear‑programming step
# ----------------------------------------------------------------------


def solve_lp_for_step(f: np.ndarray, bound: float) -> np.ndarray | None:
    """
    Solve   max sum(g)  s.t.  (f * g)_k ≤ bound  (for all k) , g ≥ 0.
    Returns the optimal g (numpy array) or None if the solver fails.
    """
    n = len(f)
    m = 2 * n - 1

    # Build the convolution matrix A where A[k, j] = f_j if
    # j ≤ k ≤ j+n-1, else 0.
    A = np.zeros((m, n), dtype=float)
    for j in range(n):
        A[j:j + n, j] = f[j]

    b = np.full(m, bound, dtype=float)
    c = -np.ones(n)                     # minimize -sum(g)  == maximize sum(g)
    bounds = [(0.0, None)] * n

    # HiGHS is the default modern LP solver in SciPy.
    res = linprog(c, A_ub=A, b_ub=b, bounds=bounds, method="highs")
    if res.success:
        return res.x
    else:
        return None


def lp_refine(initial_a: np.ndarray,
              time_deadline: float,
              max_outer: int = 15,
              mix_weight: float = 0.5) -> tuple[np.ndarray, float]:
    """
    Perform a few LP‑based refinements starting from `initial_a`
    (which should already be on the simplex).  Returns the best vector
    encountered and its unsmoothed score.
    """
    a = project_to_simplex(initial_a.astype(float).copy())
    best_a = a.copy()
    best_score = compute_score(a)

    n = len(a)

    for it in range(max_outer):
        if time.time() >= time_deadline:
            break

        # current bound M = max self‑convolution
        conv = convolution_fft(a, a)
        M = float(np.max(conv))

        # solve the linear programme
        g0 = solve_lp_for_step(a, M)
        if g0 is None:
            # solver failure – abort the LP refinement for this start
            break

        sum_g0 = float(np.sum(g0))
        # If we cannot increase the total mass, we are at a stationary point.
        if sum_g0 <= 1.0 + 1e-12:
            break

        # Scale the LP solution back to the simplex.
        g_scaled = g0 / sum_g0

        # Mix old and new vectors.
        a = (1.0 - mix_weight) * a + mix_weight * g_scaled
        a = project_to_simplex(a)   # numerical safety

        cur_score = compute_score(a)
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()

        # gradually reduce the mixing weight to stabilize the sequence
        mix_weight = max(0.1, mix_weight * 0.95)

    return best_a, best_score


# ----------------------------------------------------------------------
# Cheap stochastic hill‑climb refinement
# ----------------------------------------------------------------------


def hill_climb(seq, time_deadline: float, max_steps: int = 20000,
               init_step: float = 0.1) -> tuple[list[float], float]:
    """
    Stochastic pair‑wise mass exchange hill‑climb.
    Returns a Python list and its true `evaluate_sequence` score.
    """
    a = np.array(seq, dtype=float)
    a = project_to_simplex(a)
    best_a = a.copy()
    best_score = compute_score(best_a)
    n = len(a)
    step = init_step

    while max_steps > 0 and time.time() < time_deadline:
        i, j = np.random.randint(n, size=2)
        if i == j:
            continue
        delta = step * best_a[i]
        if delta <= 0:
            continue
        cand = best_a.copy()
        cand[i] -= delta
        cand[j] += delta
        if cand[i] < 0:
            continue
        cand = project_to_simplex(cand)
        cur_score = compute_score(cand)
        if cur_score < best_score:
            best_score = cur_score
            best_a = cand
            step = min(step * 1.05, 0.5)
        else:
            step *= 0.98
        max_steps -= 1

    return best_a.tolist(), best_score

# Main entry point
# ----------------------------------------------------------------------
