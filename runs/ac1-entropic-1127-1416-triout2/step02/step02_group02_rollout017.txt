def optimize(a, deadline):
    best_score = compute_objective(a)
    best_a = a.copy()
    iteration=0
    while time < deadline:
        # compute conv
        fft = np.fft.rfft(a, n=2*n-1)
        conv = np.fft.irfft(fft*fft, n=2*n-1)
        max_val = conv.max()
        k_max = conv.argmax()
        # find active set
        tol = max_val * 1e-12 + 1e-14
        active = np.where(conv >= max_val - tol)[0]
        # compute gradient of max conv
        grad = np.zeros_like(a)
        for k in active:
            low = max(0, k - n + 1)
            high = min(n - 1, k)
            idx = np.arange(low, high + 1, dtype=int)
            partner = k - idx
            grad[idx] += 2.0 * a[partner]
        if active.size > 1:
            grad = grad / active.size
        # gradient of objective f = 2*n*max_conv/sum^2
        # Since sum=1, gradient same scaling factor; multiply by 2*n maybe but scaling doesn't affect direction for mirror descent? Actually step size scaling matters.
        # Compute scaled gradient: grad_obj = 2 * n * grad
        grad_obj = 2.0 * n * grad
        # step size schedule
        eta = 0.5 / np.sqrt(iteration + 1)  # can tune
        # Exponential update
        # Avoid overflow: subtract max(eta*grad) from exponent.
        exponent = -eta * grad_obj
        # Clip exponent to reasonable range
        max_exp = np.max(exponent)
        exponent = exponent - max_exp  # bring max to zero
        a = a * np.exp(exponent)
        # Renormalize to sum=1
        a_sum = a.sum()
        if a_sum == 0 or not np.isfinite(a_sum):
            # reset to uniform if degenerate
            a = np.full_like(a, 1.0 / a.shape[0])
        else:
            a = a / a_sum
        # Ensure any nan due to underflow set to small epsilon
        # Evaluate objective
        cur_score = compute_objective(a)
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"Restart {restart_id} iter {iteration}: improved {best_score}")
        # early stop if no improvement for many iterations
        iteration += 1
        if iteration % 2000 ==0:
            # add jitter small random to avoid stagnation
            jitter = 0.001 * np.random.randn(a.shape[0])
            a = np.maximum(a + jitter, 0)
            a /= a.sum()
    return best_a, best_score

def optimize_adam(init_a, deadline, max_iters=None):

   n = init_a.shape[0]

   # if use softmax parametrization, set b = log(init_a) maybe.

   # But init_a may contain zeros; log(0) invalid; we can set b = np.log(init_a + 1e-12)

   b = np.log(init_a + 1e-12)

   m = np.zeros_like(b)

   v = np.zeros_like(b)

   beta1, beta2 = 0.9, 0.999

   epsilon = 1e-8

   best_score = compute_objective(init_a)

   best_a = init_a.copy()

   t = 0

   while time.time() < deadline and (max_iters is None or t < max_iters):

         # compute a via softmax

         max_b = np.max(b)  # to avoid overflow

         a = np.exp(b - max_b)

         a_sum = a.sum()

         a = a / a_sum

         # compute gradient for max conv

         fft = np.fft.rfft(a, n=2*n-1)

         conv = np.fft.irfft(fft * fft, n=2*n-1)

         max_val = conv.max()

         # find active set as before

         tol = max_val * 1e-12 + 1e-14

         active = np.where(conv >= max_val - tol)[0]

         grad_a = np.zeros_like(a)

         for k in active:

              low = max(0, k - n + 1)

              high = min(n-1, k)

              idx = np.arange(low, high+1, dtype=int)

              partner = k - idx

              grad_a[idx] += 2.0 * a[partner]

         if active.size > 1:

              grad_a /= active.size

         # gradient of objective: factor = 2*n (since sum=1)

         grad_obj = 2.0 * n * grad_a

         # gradient wrt logits b (softmax): g_b = a * (grad_obj - np.dot(a, grad_obj))

         avg = np.dot(a, grad_obj)

         g_b = a * (grad_obj - avg)

         # Adam update

         t += 1

         beta1t = beta1 ** t

         beta2t = beta2 ** t

         m = beta1 * m + (1 - beta1) * g_b

         v = beta2 * v + (1 - beta2) * (g_b ** 2)

         m_hat = m / (1 - beta1t)

         v_hat = v / (1 - beta2t)

         lr = 0.1 / (np.sqrt(t))  # or constant?

         b -= lr * m_hat / (np.sqrt(v_hat) + epsilon)

         # Evaluate objective

         cur_score = compute_objective(a)

         if cur_score < best_score:

               best_score = cur_score

               best_a = a.copy()

               print(f"Restart {restart_id} iter {t}: new best {best_score:.6f}")

         # early break if best_score < target

         if best_score < 1.5:

               break

   return best_a, best_score

def get_initial_seq(start_type, n, rng):
    if start_type == 'prev':
        try:
            seq = np.asarray(height_sequence_1, dtype=float)
            if seq.shape[0] != n:
                # if length mismatched, we can truncate or pad?
                # We'll adapt by truncating or resizing. Simpler: if seq longer, truncate; if shorter, pad zeros.
                if seq.shape[0] > n:
                    seq = seq[:n]
                else:
                    seq = np.concatenate([seq, np.zeros(n - seq.shape[0])])
            a = np.maximum(seq, 0.0)
            total = a.sum()
            if total == 0:
                a = np.full(n, 1.0 / n)
            else:
                a = a / total
            return a
        except Exception:
            # fallback
            pass
    if start_type == 'uniform':
        return np.full(n, 1.0 / n)
    elif start_type == 'random_dirichlet':
        # Dirichlet with concentration < 1 to promote sparsity
        alpha = 0.5
        a = rng.gamma(alpha, 1.0, size=n)  # gamma approach to Dirichlet

def adam_optimize(init_a, deadline, restart_id, target=1.503):
    n = init_a.shape[0]
    fft_len = 1 << ((2*n - 2).bit_length())
    # initialize logits
    epsilon = 1e-12
    b = np.log(init_a + epsilon)
    m = np.zeros_like(b)
    v = np.zeros_like(b)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8

    best_a = init_a.copy()
    # compute initial objective using maxconv_and_grad
    max_val, _ = maxconv_and_grad(init_a, fft_len)
    best_score = 2.0 * n * max_val
    stagnation = 0
    t = 0
    while time.time() < deadline:
        t += 1
        # softmax
        max_b = np.max(b)
        a = np.exp(b - max_b)
        a_sum = a.sum()
        a = a / a_sum
        # compute convection and gradient
        max_val, grad_a = maxconv_and_grad(a, fft_len)
        cur_score = 2.0 * n * max_val
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            stagnation = 0
            print(f"Restart {restart_id} iter {t:6d}: new best {best_score:.6f}")
        else:
            stagnation += 1
        if best_score < target:
            break
        if stagnation > 8000:
            # occasional random jitter in log-space
            b += 0.01 * rng.randn(n)
            stagnation = 0
        # gradient for logits
        grad_obj = 2.0 * n * grad_a
        avg = np.dot(a, grad_obj)
        grad_b = a * (grad_obj - avg)
        # Adam update
        m = beta1 * m + (1 - beta1) * grad_b
        v = beta2 * v + (1 - beta2) * (grad_b ** 2)
        m_hat = m / (1 - beta1 ** t)
        v_hat = v / (1 - beta2 ** t)
        # decreasing learning rate
        lr = 0.4 / np.sqrt(t)
        b -= lr * m_hat / (np.sqrt(v_hat) + eps)
    return best_a, best_score

"""
Propose a candidate step‑function sequence that minimizes

    2·n·max(conv(a, a)) / (sum(a)**2)

The implementation uses several Adam‑optimisation restarts with
different initialisations (previous best, uniform, random Dirichlet,
geometric).  The variables are re‑parameterised by logits so that the
simplex constraint Σa_i = 1 is satisfied automatically via a softmax.
For each iteration the aperiodic convolution is obtained with an FFT,
the active indices attaining the maximum are detected and a sub‑gradient
of the max‑convolution is built.  The gradient is transformed to the
logit space, an Adam update with a diminishing learning‑rate is applied,
and occasional random jitter in log‑space is added to escape flat
regions.  The best sequence seen over all restarts is returned as a
plain Python list of floats.

The routine respects a user supplied time budget (default 1000 s) and
prints progress messages.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper functions (top‑level – no closures)
# ----------------------------------------------------------------------


def _maxconv_and_subgrad(a: np.ndarray, fft_len: int):
    """
    Compute the aperiodic convolution a*a, the maximal convolution value,
    and a sub‑gradient of that maximal value w.r.t. a.
    Returns (max_val, grad) where grad has the same shape as a.
    """
    n = a.shape[0]
    # convolution via FFT (only the first 2·n‑1 entries are needed)
    fft = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(fft * fft, n=fft_len)[: 2 * n - 1]

    max_val = conv.max()
    # tolerance for numerically tied maxima
    tol = max_val * 1e-12 + 1e-14
    active = np.where(conv >= max_val - tol)[0]

    grad = np.zeros_like(a)
    for k in active:
        low = max(0, k - n + 1)
        high = min(n - 1, k)
        idx = np.arange(low, high + 1, dtype=int)
        # partner indices such that i + partner = k
        grad[idx] += 2.0 * a[k - idx]

    if active.size > 1:
        grad /= float(active.size)          # any convex combination is ok

    return max_val, grad


def _initial_sequence(kind: str, n: int, rng: np.random.Generator,
                      prev_seq: np.ndarray | None):
    """
    Produce an initial probability vector of length n.
    `kind` can be 'prev', 'uniform', 'random_dirichlet' or 'geometric'.
    """
    if kind == "prev" and prev_seq is not None:
        seq = prev_seq.copy()
        if seq.shape[0] != n:
            # truncate or pad with zeros to match the target length
            if seq.shape[0] > n:
                seq = seq[:n]
            else:
                seq = np.concatenate([seq,
                                      np.zeros(n - seq.shape[0],
                                               dtype=seq.dtype)])
        seq = np.maximum(seq, 0.0)
        total = seq.sum()
        if total == 0.0:
            return np.full(n, 1.0 / n)
        return seq / total

    if kind == "uniform":
        return np.full(n, 1.0 / n)

    if kind == "random_dirichlet":
        # concentration < 1 yields sparse vectors
        alpha = 0.5
        sample = rng.gamma(alpha, 1.0, size=n)
        return sample / sample.sum()

    if kind == "geometric":
        # symmetric geometric decay from both ends
        r = 0.98 + 0.02 * rng.random()      # r ∈ (0.98, 1.0)
        left = r ** np.arange(n)
        right = r ** np.arange(n - 1, -1, -1)
        seq = left + right
        return seq / seq.sum()

    # fallback
    return np.full(n, 1.0 / n)


def _adam_optimize(init_a: np.ndarray,
                   deadline: float,
                   restart_id: int,
                   target: float = 1.503) -> tuple[np.ndarray, float]:
    """
    Adam optimisation on the soft‑max logits of a probability vector.
    Returns (best_a, best_score) seen before the deadline.
    """
    n = init_a.shape[0]
    # length of the zero‑padded FFT (next power of two for speed)
    fft_len = 1 << ((2 * n - 2).bit_length())

    # logits, small epsilon prevents log(0)
    eps_init = 1e-12
    b = np.log(init_a + eps_init)

    # Adam moments
    m = np.zeros_like(b)
    v = np.zeros_like(b)
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8

    # evaluation of the start point
    max_val, _ = _maxconv_and_subgrad(init_a, fft_len)
    best_score = 2.0 * n * max_val
    best_a = init_a.copy()

    stagnation = 0          # counts iterations without improvement
    it = 0

    while time.time() < deadline:
        it += 1

        # ------------------------------------------------------------------
        # 1) obtain the probability vector from the logits
        # ------------------------------------------------------------------
        b_max = np.max(b)
        a = np.exp(b - b_max)                 # shift for numerical stability
        a = a / a.sum()                       # Σa = 1  (soft‑max)

        # ------------------------------------------------------------------
        # 2) convolution, maximal entry and its sub‑gradient
        # ------------------------------------------------------------------
        max_val, grad_a = _maxconv_and_subgrad(a, fft_len)
        cur_score = 2.0 * n * max_val

        # ------------------------------------------------------------------
        # 3) keep the best solution seen so far
        # ------------------------------------------------------------------
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            stagnation = 0
            print(f"Restart {restart_id:02d} iter {it:6d}:   best = {best_score:.6f}")
        else:
            stagnation += 1

        # ------------------------------------------------------------------
        # 4) termination criteria
        # ------------------------------------------------------------------
        if best_score < target:
            break
        if stagnation > 8000:                 # escape possible plateau
            b += 0.01 * np.random.randn(n)    # random jitter in log‑space
            stagnation = 0

        # ------------------------------------------------------------------
        # 5) Adam update on the logits
        # ------------------------------------------------------------------
        # gradient of the original objective (Σa = 1 ⇒ factor 2·n)
        grad_obj = 2.0 * n * grad_a
        # transform to log‑space (soft‑max Jacobian)
        avg = np.dot(a, grad_obj)
        grad_b = a * (grad_obj - avg)

        # moments update
        m = beta1 * m + (1.0 - beta1) * grad_b
        v = beta2 * v + (1.0 - beta2) * (grad_b ** 2)

        # bias‑corrected estimates
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        # diminishing learning rate (helps with the non‑convex landscape)
        lr = 0.4 / np.sqrt(it)

        b -= lr * m_hat / (np.sqrt(v_hat) + eps)

    return best_a, best_score
