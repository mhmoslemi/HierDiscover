```
def lp_step(f, M, sum_f, t=0.1):
    n = len(f)
    # Build A matrix as sparse
    rows = []
    cols = []
    data = []
    for i in range(2*n-1):
        # j range
        j_start = max(0, i-(n-1))
        j_end = min(i, n-1)
        # For k in j_start..j_end:
        for k in range(j_start, j_end+1):
            # coefficient of b_k is f[i-k]
            coeff = f[i - k]
            rows.append(i)
            cols.append(k)
            data.append(coeff)
    A = scipy.sparse.csr_matrix((data, (rows, cols)), shape=(2*n-1, n))
    b_ub = np.full(2*n-1, M)
    # Solve
    res = scipy.optimize.linprog(c=-np.ones(n), A_ub=A, b_ub=b_ub,
                                 bounds=[(0, None)]*n, method='highs')
    if not res.success: return None
    b = res.x
    sum_b = b.sum()
    if sum_b == 0: return None
    g = b * (sum_f / sum_b)
    # new f
    f_new = (1-t)*f + t*g
    return f_new
```

def subgradient_descent(initial, time_deadline, max_iters):
    a = initial.copy()
    a = project_to_simplex(a)
    best = a.copy()
    best_score = compute_score(a)
    n = len(a)
    for it in range(1, max_iters+1):
        if time.time() >= time_deadline:
            break
        max_val, k_max = max_convolution_and_index(a)
        grad = subgradient_of_max_at_index(a, k_max)
        # step size schedule: eta = eta0 / sqrt(it)
        eta = 0.5 / np.sqrt(it)
        a_new = a - eta * grad
        a_new = project_to_simplex(a_new)
        cur_score = compute_score(a_new)
        if cur_score < best_score:
            best_score = cur_score
            best = a_new.copy()
        a = a_new
    return best, best_score
```

"""
Hybrid optimisation for the step‑function bound.

The search proceeds in three stages per sequence length:

1️⃣  Random exploration of a few parametric families (arcsine‑like,
    beta‑type, exponential‑type) – cheap evaluation with a fast
    FFT‑based score.

2️⃣  Long projected sub‑gradient descent on the non‑smooth objective
    (max self‑convolution).  The iteration count is chosen adaptively
    to use the remaining time budget.

3️⃣  An “active‑set” linear‑programming step inspired by the published
    alternating‑LP method: for a fixed f we solve

        maximise   ∑ b_i
        subject to  (f * b)_k ≤ M   for all k,
                    b_i ≥ 0,

    where M = max(f*f).  The solution b (scaled to the same ℓ¹‑norm as f)
    provides a direction that provably reduces M when we move a small
    step towards it.  A handful of such LP steps are performed,
    interleaved with a final polishing sub‑gradient phase.

The routine respects the supplied time budget (≈1000 s) and returns the
best sequence found as a plain ``list`` of floats.  If SciPy is not
available the LP stage is skipped and the algorithm falls back to the
sub‑gradient optimiser only.
"""

import time
import numpy as np
from typing import List, Tuple

# ----------------------------------------------------------------------
# Optional SciPy import for the LP stage
# ----------------------------------------------------------------------
try:
    from scipy.optimize import linprog  # type: ignore
    SCIPY_AVAILABLE = True
except Exception:  # pragma: no cover
    SCIPY_AVAILABLE = False

# ----------------------------------------------------------------------
# Low‑level utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Linear convolution via real FFT (zero‑padded to a power of two)."""
    m = len(a) + len(b) - 1
    n_fft = next_pow2(m)
    A = np.fft.rfft(a, n=n_fft)
    B = np.fft.rfft(b, n=n_fft)
    C = A * B
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


def max_convolution_and_index(a: np.ndarray) -> Tuple[float, int]:
    """Return (maximum value, index) of a*a."""
    conv = convolution_fft(a, a)
    idx = int(np.argmax(conv))
    return float(conv[idx]), idx


def subgradient_of_max_at_index(a: np.ndarray, k: int) -> np.ndarray:
    """
    Sub‑gradient of the maximal convolution entry (index k) w.r.t. a.
    For each i, ∂c_k/∂a_i = a_{k-i} if that index exists,
    and doubled when k = 2·i.
    """
    n = len(a)
    grad = np.zeros_like(a)
    i = np.arange(n)
    j = k - i
    mask = (j >= 0) & (j < n)
    grad[mask] = a[j[mask]]
    if k % 2 == 0:
        diag = k // 2
        if diag < n:
            grad[diag] = 2.0 * a[diag]
    return grad


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection onto the probability simplex {x ≥ 0, Σx = 1}.
    Duchi et al. (2008) implementation.
    """
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / len(v))

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho = np.where(u - cssv / np.arange(1, len(v) + 1) > 0)[0]
    if len(rho) == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def compute_score(a: np.ndarray) -> float:
    """
    Fast internal proxy for ``evaluate_sequence``:
    2·n·max(a*a) / (sum(a)²)  (scale‑invariant, sum≈1 after projection).
    """
    n = len(a)
    conv = convolution_fft(a, a)
    max_val = float(np.max(conv))
    s = float(np.sum(a))
    if s < 1e-12:
        return float("inf")
    return 2.0 * n * max_val / (s * s)


# ----------------------------------------------------------------------
# Parametric shape families
# ----------------------------------------------------------------------


def arcsine_initial(n: int) -> np.ndarray:
    """Classic arcsine‑shaped start: a_i ∝ 1/√((i+½)*(n‑½‑i))."""
    i = np.arange(n, dtype=float)
    eps = 0.25
    vals = 1.0 / np.sqrt((i + 0.5) * (n - 0.5 - i) + eps)
    return vals / np.sum(vals)


def generate_shape(n: int, p: float, eps: float) -> np.ndarray:
    """
    Symmetric power‑law family:
        a_i ∝ ((i+eps)*(n‑1‑i+eps))⁻ᵖ .
    """
    i = np.arange(n, dtype=float)
    base = ((i + eps) * (n - 1 - i + eps)) ** (-p)
    total = base.sum()
    if total == 0.0:
        return np.full(n, 1.0 / n)
    return base / total


def generate_exponential_shape(n: int, decay: float) -> np.ndarray:
    """Peak at the centre, exponential decay towards the edges."""
    i = np.arange(n, dtype=float)
    centre = (n - 1) / 2.0
    vals = np.exp(-decay * np.abs(i - centre))
    return vals / vals.sum()


def generate_beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    """Beta‑distribution‑like shape on the discrete grid."""
    i = np.arange(n, dtype=float) + 0.5
    x = i / n
    # avoid exact 0/1 issues
    x = np.clip(x, 1e-12, 1 - 1e-12)
    vals = (x ** (alpha - 1)) * ((1 - x) ** (beta - 1))
    if vals.sum() == 0.0:
        vals = np.full(n, 1.0 / n)
    return vals / vals.sum()


# ----------------------------------------------------------------------
# Projected sub‑gradient descent (stage 2)
# ----------------------------------------------------------------------


def subgradient_descent(
    initial_a: np.ndarray,
    time_deadline: float,
    max_iter: int = 300_000,
    eta0: float = 0.7,
    noise_every: int = 1000,
    noise_scale: float = 5e-5,
) -> Tuple[np.ndarray, float]:
    """
    Projected sub‑gradient descent on the max‑convolution objective.
    Returns the best vector encountered and its internal score.
    """
    a = project_to_simplex(initial_a.astype(float).copy())
    best_a = a.copy()
    best_sc = compute_score(a)

    n = len(a)
    it = 0
    while time.time() < time_deadline and it < max_iter:
        it += 1
        _, k_max = max_convolution_and_index(a)
        grad = subgradient_of_max_at_index(a, k_max)

        eta = eta0 / np.sqrt(it)          # diminishing step size
        a_new = a - eta * grad

        if noise_every and (it % noise_every == 0):
            a_new += np.random.randn(n) * noise_scale

        a_new = project_to_simplex(a_new)

        cur_sc = compute_score(a_new)
        if cur_sc < best_sc:
            best_sc = cur_sc
            best_a = a_new.copy()
        a = a_new

        # occasional progress report (kept lightweight)
        if it % 50000 == 0:
            print(f"    subgrad iter={it:,d}  cur={cur_sc:.6f}  best={best_sc:.6f}")

    return best_a, best_sc


# ----------------------------------------------------------------------
# Linear‑programming improvement (stage 3)
# ----------------------------------------------------------------------


def build_conv_matrix(f: np.ndarray) -> np.ndarray:
    """
    Construct the (2n‑1) × n matrix A such that
        A @ b  =  f * b   (linear convolution).
    Dense version – acceptable for n ≤ 2048.
    """
    n = len(f)
    m = 2 * n - 1
    A = np.zeros((m, n), dtype=float)
    for col in range(n):
        A[col : col + n, col] = f
    return A


def lp_improve(
    a: np.ndarray,
    time_deadline: float,
    max_steps: int = 20,
    t0: float = 0.25,
) -> Tuple[np.ndarray, float]:
    """
    Perform a few LP‑based improvement steps.
    Each step solves

        maximise   Σ y_i
        subject to   A y ≤ 1 ,  y ≥ 0

    where A is the convolution matrix of the current a.
    The optimal y defines a direction g = y / Σy that,
    after a small interpolation, reduces the maximal self‑convolution.
    """
    a = a.copy()
    n = len(a)
    sum_target = np.sum(a)  # normally 1 after projection
    M = np.max(convolution_fft(a, a))

    for step in range(max_steps):
        if time.time() >= time_deadline:
            break

        # Build constraints for the current a
        A = build_conv_matrix(a)

        # Solve the LP with unit bound; scaling handled later
        if SCIPY_AVAILABLE:
            res = linprog(
                c=-np.ones(n),          # maximise Σy  → minimise –Σy
                A_ub=A,
                b_ub=np.ones(2 * n - 1),
                bounds=[(0, None)] * n,
                method="highs",
            )
        else:
            # SciPy not available – abort the LP stage
            break

        if not res.success:
            print(f"    LP step {step} failed: {res.message}")
            break

        y = res.x
        sum_y = np.sum(y)
        if sum_y <= 0:
            break

        # Candidate g has the same ℓ¹‑norm as a
        g = y / sum_y * sum_target

        # Interpolation – step size decays geometrically
        t = t0 * (0.85 ** step)
        a = (1.0 - t) * a + t * g
        a = project_to_simplex(a)

        # New bound
        M_new = np.max(convolution_fft(a, a))
        improvement = M - M_new
        M = M_new

        print(
            f"    LP step {step+1}/{max_steps}  t={t:.4f}  ΔM={improvement:.6e}  M={M:.6e}"
        )
        if improvement < 1e-12:  # stagnation
            break

    return a, M

def propose_candidate(
    seed: int = 42,
    budget_s: int = 1000,
    **kwargs,
) -> List[float]:
    """
    Search for a non‑negative step‑function with a low
    ``evaluate_sequence`` value.
    The routine explores a few sequence lengths, samples several
    parametric families, refines the best candidates with a long
    sub‑gradient run, applies a few LP‑based improvement steps and
    finally returns the best sequence discovered.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    # tiny safety margin to avoid overshooting the harness limit
    deadline = start_time + budget_s - 2.0

    best_seq: List[float] = []
    best_val: float = float("inf")

    # ------------------------------------------------------------------
    # 0.  Fallback to any pre‑computed construction present in the env.
    # ------------------------------------------------------------------
    try:
        fallback = [float(x) for x in height_sequence_1]  # type: ignore
        val = evaluate_sequence(fallback)
        print(f"Fallback (height_sequence_1) score = {val:.6f}")
        if val < best_val:
            best_val = val
            best_seq = fallback
    except Exception:
        pass

    # ------------------------------------------------------------------
    # 1.  Exploration over a handful of lengths.
    # ------------------------------------------------------------------
    lengths = [1024, 2048]

    for n in lengths:
        if time.time() >= deadline - 5.0:
            break
        print(f"\n=== Exploring length n = {n} ===")

        # ---- 1a. Random scan of shape families ----
        local_best_a = None
        local_best_sc = float("inf")

        starters = [
            np.full(n, 1.0 / n),          # uniform
            arcsine_initial(n),           # classic arcsine
        ]
        for s in starters:
            sc = compute_score(s)
            if sc < local_best_sc:
                local_best_sc = sc
                local_best_a = s.copy()

        # random shapes from three families
        max_samples = 800
        for _ in range(max_samples):
            if time.time() >= deadline - 5.0:
                break
            family = rng.integers(3)
            if family == 0:  # power‑law family
                p = rng.uniform(0.2, 0.9)
                eps = rng.uniform(0.001, max(2.0, 0.02 * n))
                cand = generate_shape(n, p, eps)
            elif family == 1:  # exponential decay
                decay = rng.uniform(0.001, 0.02 * n)
                cand = generate_exponential_shape(n, decay)
            else:  # beta‑type family
                alpha = rng.uniform(0.3, 2.5)
                beta = rng.uniform(0.3, 2.5)
                cand = generate_beta_shape(n, alpha, beta)
            sc = compute_score(cand)
            if sc < local_best_sc:
                local_best_sc = sc
                local_best_a = cand

        # ---- 1b. Long sub‑gradient descent ----
        if local_best_a is not None and time.time() < deadline - 5.0:

            refined_a, _ = subgradient_descent(

                time_deadline=deadline,
                max_iter=max_iters,
                eta0=0.8,
                noise_every=800,

            )
            # evaluate with the official (slow) scorer
            val = evaluate_sequence(refined_a.tolist())
            print(f"  → sub‑gradient score = {val:.6f}")
            if val < best_val:
                best_val = val
                best_seq = refined_a.tolist()
                print(f"  *** New global best after sub‑gradient: {best_val:.6f} (n={n}) ***")
        else:
            # if time ran out, just test the raw candidate
            if local_best_a is not None:
                val = evaluate_sequence(local_best_a.tolist())
                print(f"  Raw candidate score = {val:.6f}")
                if val < best_val:
                    best_val = val
                    best_seq = local_best_a.tolist()
                    print(f"  *** New global best (raw) ***")

        # ---- 1c. LP‑based improvement (if time permits) ----
        if best_seq and time.time() < deadline - 10.0 and SCIPY_AVAILABLE:
            a = np.array(best_seq, dtype=float)
            remaining = deadline - time.time()
            # we allow at most 20 LP steps – each is cheap compared to thousands of sub‑gradient iters
            max_lp_steps = min(20, int(remaining / 15))  # about 15 s per LP step on typical hardware
            print(f"\n  LP improvement (≤{max_lp_steps} steps) …")
            a, final_M = lp_improve(
                a,
                time_deadline=deadline,
                max_steps=max_lp_steps,
                t0=0.3,
            )
            val = evaluate_sequence(a.tolist())
            print(f"  → post‑LP score = {val:.6f}")
            if val < best_val:
                best_val = val
                best_seq = a.tolist()
                print(f"  *** New global best after LP: {best_val:.6f} (n={n}) ***")

            # final polishing with sub‑gradient after LP phase
            if time.time() < deadline - 5.0:
                remaining = deadline - time.time()
                max_iters = int(min(300_000, remaining * 30))
                print(f"\n  Final polishing (sub‑gradient, max_iter={max_iters}) …")
                a_polished, _ = subgradient_descent(
                    initial_a=a,
                    time_deadline=deadline,
                    max_iter=max_iters,
                    eta0=0.6,
                    noise_every=600,
                    noise_scale=3e-5,
                )
                val = evaluate_sequence(a_polished.tolist())
                print(f"  → polished score = {val:.6f}")
                if val < best_val:
                    best_val = val
                    best_seq = a_polished.tolist()
                    print(f"  *** Updated global best after final polish ***")

    # ------------------------------------------------------------------
    # 2.  Guard – ensure we always return *something*.
    # ------------------------------------------------------------------
    if not best_seq:
        best_seq = [1.0] * 256
        best_val = evaluate_sequence(best_seq)

    print(f"\nFinished search. Best reported score = {best_val:.6f}")
    return best_seq
