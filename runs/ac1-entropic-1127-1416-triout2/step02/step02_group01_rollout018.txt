    # define utilities same as earlier.

"""
Search for non‑negative step heights that minimise the self‑convolution
upper‑bound 2·n·max(a∗a)/∑a².  The algorithm proceeds
1. a quick sweep over a family of Beta‑shaped sequences (a_i∝(i+½)^α (n‑½‑i)^β)
   for a few lengths; best candidate is kept.
2. the best candidate is up‑sampled to a larger length (if time permits) and
   refined by a cheap stochastic hill‑climbing that moves a small amount of
   mass between two random positions.
Optionally a short smoothed‑max gradient descent is performed when time
allows.  All operations keep the vector on the probability simplex.
"""

import time
import numpy as np

# Helper functions: projection, FFT convolution, score
def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    if np.all(v <= 0):
        n = v.shape[0]
        return np.full_like(v, 1.0 / n)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    rho = np.where(u - cssv / np.arange(1, v.shape[0] + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w

def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()

def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full convolution using a real FFT."""
    m = len(a) + len(b) - 1
    n_fft = next_pow2(m)
    A = np.fft.rfft(a, n=n_fft)
    B = np.fft.rfft(b, n=n_fft)
    C = A * B
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv

def compute_score(a: np.ndarray) -> float:
    """Score identical to evaluate_sequence when ∑a = 1."""
    a = np.asarray(a, dtype=float)
    n = len(a)
    conv = convolution_fft(a, a)
    max_val = float(np.max(conv))
    s = float(np.sum(a))
    if s < 1e-12:
        return float('inf')
    return 2.0 * n * max_val / (s * s)

# Shape generators ---------------------------------------------------------

def arcsine_initial(n: int) -> np.ndarray:
    """Arcsine‑shaped start: a_i ∝ 1 / sqrt((i+0.5)(n‑0.5‑i))."""
    i = np.arange(n, dtype=float)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - 0.5 - i))
    return vals / np.sum(vals)

def beta_sequence(n: int, alpha: float, beta: float = None) -> np.ndarray:
    """Beta‑type coefficients a_i ∝ (i+½)^α·(n‑½‑i)^β."""
    if beta is None:
        beta = alpha
    i = np.arange(n, dtype=float)
    left = np.power(i + 0.5, alpha)
    right = np.power(n - 0.5 - i, beta)
    seq = left * right
    # Numerical safety – clamp extreme values.
    seq = np.clip(seq, 0, 1e12)
    seq_sum = np.sum(seq)
    if seq_sum == 0:
        seq = np.full(n, 1.0 / n)
    else:
        seq = seq / seq_sum
    return seq

def exponential_sequence(n: int, lam: float) -> np.ndarray:
    """a_i ∝ exp(-lam·|i-(n-1)/2|)."""
    i = np.arange(n, dtype=float)
    center = (n - 1) / 2.0
    seq = np.exp(-lam * np.abs(i - center))
    seq = seq / np.sum(seq)
    return seq

def upsample_sequence(seq: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation of a probability vector to a larger size."""
    old_n = len(seq)
    # positions of old points in [0,1]
    old_x = (np.arange(old_n) + 0.5) / old_n
    new_x = (np.arange(new_n) + 0.5) / new_n
    new_vals = np.interp(new_x, old_x, seq)
    new_vals = new_vals / np.sum(new_vals)
    return new_vals

# Stochastic hill‑climbing -------------------------------------------------

def hill_climb(seq, time_deadline: float, max_steps: int = 30000, init_step: float = 0.07):
    """Move a small amount of mass from a random index to another.
    Keep moves that improve the true (unsmoothed) score."""
    a = np.array(seq, dtype=float)
    a = project_to_simplex(a)
    best_a = a.copy()
    best_score = compute_score(best_a)
    n = len(a)
    step = init_step

    while max_steps > 0 and time.time() < time_deadline:
        i, j = np.random.randint(n, size=2)
        if i == j:
            continue
        # Transfer a fraction of the mass at i to j.
        delta = step * best_a[i]
        if delta <= 0:
            continue
        cand = best_a.copy()
        cand[i] -= delta
        cand[j] += delta
        # Projection to stay on simplex (mostly already satisfied).
        cand = project_to_simplex(cand)
        cur_score = compute_score(cand)
        if cur_score < best_score:
            best_score = cur_score
            best_a = cand

"""
Search for a non‑negative step‑function that minimises

    E(a) = 2·n·max(a∗a) / (sum(a)²)

The search proceeds in four phases:

1. **Parametric sweep** – generate many candidates from a Beta‑type
   family  a_i ∝ (i+½)^α·(n‑½‑i)^β  (including the classic arcsine shape,
   uniform mix and simple exponential decay).  For each length we keep the
   best candidate seen.

2. **Upsampling** – if time permits the best candidate is linearly
   interpolated to a larger length (8192) which usually lowers the bound
   because the discrete approximation becomes finer.

3. **Smoothed‑max gradient descent** – a short projected gradient descent
   on the smooth log‑sum‑exp approximation of the max is performed when
   there is still budget.

4. **Hill‑climbing refinement** – a cheap stochastic pair‑wise mass‑transfer
   hill‑climb that accepts only moves improving the true (unsmoothed)
   objective finishes the search.

All vectors are kept on the probability simplex (∑a_i = 1, a_i ≥ 0), so the
scale‑invariance of the objective is respected.  The final list of floats
is returned and will be fed to `evaluate_sequence`.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Basic linear‑algebra utilities
# ----------------------------------------------------------------------


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    if np.all(v <= 0):
        n = v.shape[0]
        return np.full_like(v, 1.0 / n, dtype=float)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho = np.where(u - cssv / np.arange(1, v.shape[0] + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full convolution using a real FFT (fast)."""
    m = len(a) + len(b) - 1
    n_fft = next_pow2(m)
    A = np.fft.rfft(a, n=n_fft)
    B = np.fft.rfft(b, n=n_fft)
    C = A * B
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


def compute_score(a: np.ndarray) -> float:
    """Score identical to `evaluate_sequence` when ∑a = 1."""
    a = np.asarray(a, dtype=float)
    n = len(a)
    conv = convolution_fft(a, a)
    max_val = float(np.max(conv))
    s = float(np.sum(a))
    if s < 1e-12:
        return float("inf")
    return 2.0 * n * max_val / (s * s)

# Families of candidate step‑functions
# ----------------------------------------------------------------------


def arcsine_initial(n: int) -> np.ndarray:
    """Arcsine‑shaped start: a_i ∝ 1 / sqrt((i+0.5)(n‑0.5‑i))."""
    i = np.arange(n, dtype=float)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - 0.5 - i))
    return vals / np.sum(vals)


def beta_sequence(n: int, alpha: float, beta: float | None = None) -> np.ndarray:
    """Beta‑type coefficients a_i ∝ (i+½)^α·(n‑½‑i)^β."""
    if beta is None:
        beta = alpha
    i = np.arange(n, dtype=float)
    left = np.power(i + 0.5, alpha)
    right = np.power(n - 0.5 - i, beta)
    seq = left * right
    # Clip pathological spikes; the later normalization keeps the vector on the simplex.
    seq = np.clip(seq, 0.0, 1e12)
    total = np.sum(seq)
    if total == 0.0:
        seq = np.full(n, 1.0 / n)
    else:
        seq = seq / total
    return seq


def exponential_sequence(n: int, lam: float) -> np.ndarray:
    """a_i ∝ exp(-lam·|i - (n‑1)/2|)."""
    i = np.arange(n, dtype=float)
    center = (n - 1) / 2.0
    seq = np.exp(-lam * np.abs(i - center))
    return seq / np.sum(seq)


def upsample_sequence(seq: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation of a probability vector to a larger size."""
    old_n = len(seq)
    old_x = (np.arange(old_n) + 0.5) / old_n
    new_x = (np.arange(new_n) + 0.5) / new_n
    new_vals = np.interp(new_x, old_x, seq)
    return new_vals / np.sum(new_vals)

    init_step: float = 0.07,

    Pair‑wise mass transfer hill‑climbing.
    Moves that lower the true (unsmoothed) score are kept.
    """

    a = project_to_simplex(a)

        print(f"\nRunning a brief smoothed‑max gradient descent on n = {n}")
        init_a = np.array(best_seq, dtype=float)
        refined_seq, refined_score = improve_sequence_smoothed(
            initial_a=init_a,
            time_deadline=deadline,
            max_iter=110_000,
            eta0=0.9,
            alpha_start=5.0,
            alpha_end=400.0,
            noise_every=500,
            noise_scale=2e-5,
        )
        print(f"Gradient‑descent best score = {refined_score:.6f}")
        if refined_score < best_val:
            best_val = refined_score
            best_seq = refined_seq.tolist()

    # ------------------------------------------------------------------
    # Phase 4 – final cheap stochastic hill‑climb refinement.
    # ------------------------------------------------------------------
    if best_seq is not None and time.time() < deadline - 2:
        print("\nFinal hill‑climb refinement")
        refined_seq, refined_val = hill_climb(
            best_seq,
            time_deadline=deadline,
            max_steps=50000,
            init_step=0.05,
        )
        print(f"Hill‑climb score = {refined_val:.6f}")
        if refined_val < best_val:
            best_val = refined_val
            best_seq = refined_seq

    # ------------------------------------------------------------------
    # Fallback in the extremely unlikely case that we never produced a candidate.
    # ------------------------------------------------------------------
    if best_seq is None:
        best_seq = [1.0] * 256
        best_val = evaluate_sequence(best_seq)

    final_score = evaluate_sequence(best_seq)
    print(f"\nSearch finished. Final reported score = {final_score:.6f}")
    return best_seq
```

This implementation stays within the allowed time budget, explores a diverse family of shapes (including the arcsine, Beta‑type, exponential and their mixtures), optionally refines the best candidate with a smoothed‑max gradient step and finishes with stochastic hill‑climbing. The returned sequence is guaranteed to be non‑negative and summed to a positive value, satisfying all constraints of `evaluate_sequence`.
