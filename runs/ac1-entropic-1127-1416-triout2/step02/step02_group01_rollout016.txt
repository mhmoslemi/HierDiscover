assistantfinal**Solution Overview**

The target is to minimise  

\[
E(a)=\frac{2\,n\;\max (a*a)}{(\sum_i a_i)^2},
\qquad a_i\ge 0 .
\]

The value is *scale‑invariant*, therefore we keep the vector on the probability
simplex \(\sum_i a_i = 1\).  
The only expensive operation is the full self‑convolution, which is performed
with an FFT in `O(n log n)` time.

The optimiser consists of three stages

1. **Multiple random starts** – vectors are drawn from several Dirichlet
   distributions (uniform,‑medium‑,‑high‑variance) and from the classical
   arcsine shape.
2. **Projected Adam optimisation** of the smooth approximation  

   \[
   f_\alpha(a)=\frac{1}{\alpha}\log\Bigl(\sum_k e^{\alpha\,(a*a)_k}\Bigr),
   \]

   with a homotopy on the smoothing parameter \(\alpha\) (from `1` up to
   `2000`).  The Adam updates are projected onto the simplex after each step.
3. **Permutation hill‑climbing** – the best vector from the Adam runs is
   refined by a large number of random swaps (and occasional small mass
   transfers).  Because the whole convolution is re‑evaluated after each move,
   the true, unsmoothed score `evaluate_sequence` is used.

All three phases respect the 1000 s budget; the code stops early if the target
score `1.503` is already reached.

The implementation uses only `numpy` and standard library modules, follows the
required API (`propose_candidate` returns a plain Python list), and prints a few
progress messages.

```python
"""
Propose a low‑score step‑function.

The algorithm works in three phases:

1.  Several random starts (uniform, Dirichlet, arcsine) for a chosen length.
2.  For each start a projected Adam optimiser minimises a smooth
    approximation of the max‑convolution.  The smoothing parameter α is
    increased from 1 to 2000 (homotopy) and the iterates are projected onto
    the probability simplex.
3.  The best vector is further polished by a pure permutation / mass‑transfer
    hill‑climbing that evaluates the true (non‑smoothed) objective.

All heavy work is done with FFT‑based convolutions, giving an O(n·log n)
per‑iteration cost.  The routine respects the 1000 s time budget and prints
intermediate scores.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Utility functions
# ----------------------------------------------------------------------


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the probability simplex
    (non‑negative entries that sum to 1).
    """
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / v.shape[0])

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, v.shape[0] + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.where(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full convolution a * b using a real FFT (fast)."""
    m = len(a) + len(b) - 1
    n_fft = next_pow2(m)
    A = np.fft.rfft(a, n=n_fft)
    B = np.fft.rfft(b, n=n_fft)
    C = A * B
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


def compute_score(a: np.ndarray) -> float:
    """
    Fast replica of `evaluate_sequence` when ∑a = 1.
    Returns 2·n·max(a*a).
    """
    n = a.shape[0]
    conv = convolution_fft(a, a)
    max_val = float(np.max(conv))
    return 2.0 * n * max_val


def arcsine_initial(n: int) -> np.ndarray:
    """
    Classical arcsine start: a_i ∝ 1 / sqrt( (i+0.5)·(n-0.5-i) ).
    Afterwards projected to the simplex.
    """
    i = np.arange(n, dtype=float)
    eps = 0.25
    vals = 1.0 / np.sqrt((i + 0.5) * (n - 0.5 - i) + eps)
    return project_to_simplex(vals)


# ----------------------------------------------------------------------
# Smoothed‑max gradient (Adam version)
# ----------------------------------------------------------------------


def smoothed_gradient(a: np.ndarray, alpha: float):
    """
    Gradient of the soft‑max approximation of max(a*a):
        f_α(a) = (1/α)·log Σ_k exp(α·(a*a)_k)

    Returns the gradient vector (same shape as a) and the raw convolution.
    """
    n = a.shape[0]
    # self‑convolution (length 2n‑1)
    conv = convolution_fft(a, a)

    # soft‑max weights (stable formulation)
    maxc = np.max(conv)
    exp_term = np.exp(alpha * (conv - maxc))
    w = exp_term / np.sum(exp_term)          # Σ w_k = 1, length 2n‑1

    # convolution of reversed a with the weight vector
    a_rev = a[::-1]
    conv_rev = convolution_fft(a_rev, w)    # length 3n‑2
    start = n - 1
    rev_slice = conv_rev[start:start + n]   # needed for each component

    # diagonal contribution (w_{2i}) inside the smoothed max
    w_diag = w[::2][:n]

    grad = 2.0 * rev_slice - w_diag * a
    return grad, conv


def adam_optimize(
    init_a: np.ndarray,
    time_deadline: float,
    max_iter: int = 200_000,
    eta: float = 0.02,
    beta1: float = 0.9,
    beta2: float = 0.999,
    eps: float = 1e-8,
    alpha_start: float = 1.0,
    alpha_end: float = 2000.0,
    noise_every: int = 800,
    noise_scale: float = 2e-4,
):
    """
    Projected Adam optimiser on the smoothed objective.
    Returns the best vector (ndarray) found and its true (unsmoothed) score.
    """
    a = project_to_simplex(init_a.astype(float).copy())
    m = np.zeros_like(a)
    v = np.zeros_like(a)

    best_a = a.copy()
    best_score = compute_score(a)

    n = len(a)

    for t in range(1, max_iter + 1):
        if time.time() >= time_deadline:
            break

        # homotopy on the smoothing parameter
        alpha = alpha_start + (alpha_end - alpha_start) * min(1.0, t / max_iter)

        grad, _ = smoothed_gradient(a, alpha)

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)

        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)

        step = eta * m_hat / (np.sqrt(v_hat) + eps)
        a_new = a - step

        # occasional isotropic noise to escape shallow plateaus
        if noise_every and (t % noise_every == 0):
            a_new += np.random.randn(n) * noise_scale

        a_new = project_to_simplex(a_new)

        cur_score = compute_score(a_new)
        if cur_score < best_score:
            best_score = cur_score
            best_a = a_new.copy()

        a = a_new

        # tiny progress output every few thousand iterations (no heavy I/O)
        if t % 5000 == 0:
            print(f"    Adam iter {t:,d} – cur {cur_score:.6f} – best {best_score:.6f}")

    return best_a, best_score


# ----------------------------------------------------------------------
# Pure permutation / mass‑transfer hill climbing
# ----------------------------------------------------------------------


def hill_climb_permutation(seq, time_deadline, max_steps=100_000):
    """
    Stochastic hill‑climb that only swaps two entries or moves a small
    fraction of mass.  The true (unsmoothed) objective is used for acceptance.
    """
    a = np.array(seq, dtype=float)
    n = a.shape[0]

    best_a = a.copy()
    best_score = compute_score(best_a)

    step_scale = 0.1  # initial fraction of mass transferred in a move

    while max_steps > 0 and time.time() < time_deadline:
        move_type = np.random.rand()
        cand = best_a.copy()

        if move_type < 0.5:
            # simple swap
            i, j = np.random.randint(n, size=2)
            cand[i], cand[j] = cand[j], cand[i]
        else:
            # move a fraction from i to j
            i, j = np.random.randint(n, size=2)
            if i == j:
                continue
            delta = step_scale * best_a[i]
            if delta <= 0:
                continue
            cand[i] -= delta
            cand[j] += delta
            # keep feasibility without a full projection (tiny negative guard)
            if cand[i] < 0:
                cand[i] = 0.0

        # re‑project only if numeric drift accumulated
        cand = project_to_simplex(cand)

        cur_score = compute_score(cand)

        if cur_score < best_score:
            best_score = cur_score
            best_a = cand
            # success – slowly increase step size
            step_scale = min(step_scale * 1.05, 0.5)
        else:
            # failure – shrink step size
            step_scale *= 0.97

        max_steps -= 1

    return best_a.tolist(), best_score

def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """

    The routine performs several random restarts, runs a projected Adam optimiser

    start_time = time.time()
    # a safety margin of a couple of seconds for the harness
    deadline = start_time + budget_s - 2.0

    best_seq = None
    best_val = float("inf")

    # ------------------------------------------------------------------

    except NameError:
        pass

    # ------------------------------------------------------------------
    # Parameter grid for the multi‑start Adam phase
    # ------------------------------------------------------------------
    lengths = [2048, 4096, 8192]          # larger n → better asymptotic constant
    alphas = [0.5, 1.0, 2.0]             # Dirichlet concentration parameters

    for n in lengths:

        # run a short Adam optimisation for each start
        for idx, init_vec in enumerate(initial_vectors, start=1):
            if time.time() >= deadline - 5.0:

            # allocate iterations proportionally to remaining time

    final_score = evaluate_sequence(best_seq)
```
