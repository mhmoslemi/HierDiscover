def _conv_fft(x, y):
    size = 1 << ((len(x)+len(y)-1).bit_length())
    fx = np.fft.rfft(x, n=size)
    fy = np.fft.rfft(y, n=size)
    conv = np.fft.irfft(fx*fy, n=size)[:len(x)+len(y)-1]
    return conv

def _full_convolution_fft(a):
    n = len(a)
    size = 1 << ((2*n-1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[:2*n-1]
    return conv

def _gradient_softmax_fft(a, w, size=None):
    # assumes w length = 2n-1.
    n = len(a)
    if size is None:
        size = 1 << ((3*n-2).bit_length())
    # FFT of a
    fa = np.fft.rfft(a, n=size)
    # FFT of reversed w
    w_rev = w[::-1]
    fw = np.fft.rfft(w_rev, n=size)
    # Multiply

"""Search for a low‑value step‑function.

The algorithm
----------------
1.   Choose a length *n* (default 10000) and a target ℓ₂‑norm
     `sqrt(2·n)`.  If a global variable ``height_sequence_1`` is present,
     it is used as a warm‑start, otherwise a simple power‑law or a random
     Dirichlet point is taken.

2.   A short random sweep over a 4‑parameter family
     `a_i = (i+α)^(-p)*(1+β·i/(n‑1))` produces a good initial point.

3.   The main optimisation is a **projected gradient descent on a
     smoothed version of the objective** `max(conv(a,a))`.  The smooth
     approximation is the log‑sum‑exp (soft‑max) over the convolution
     entries with a temperature parameter `α`.  The gradient can be
     computed by a single extra FFT convolution, and after each step the
     vector is projected onto the simplex `{a≥0, Σa = sqrt(2·n)}`.

4.   The temperature grows linearly from 5 to 200, turning the smoothed
     objective into the true max as the optimisation proceeds.  The step
     size decays like `η₀ / √t`.  Occasionally a tiny random jitter is added
     to avoid stagnation.

5.   The best sequence seen during the run is returned.

The whole routine respects the 1000 s time budget and prints progress
information.  All numbers are non‑negative, the sum is exactly
`sqrt(2·n)`, and the final evaluation equals `max(conv(a,a))`, i.e.
the value returned by ``evaluate_sequence``.
"""

import time
import numpy as np

# ------------------------------------------------------------
# Helper utilities
# ------------------------------------------------------------

def _project_simplex(v: np.ndarray, target_sum: float) -> np.ndarray:
    """
    Euclidean projection of a vector onto the simplex
        { w | w_i ≥ 0, Σ w_i = target_sum } .
    Implements the algorithm of Duchi et al. (2008).
    """
    if target_sum < 0:
        raise ValueError("target_sum must be non‑negative")
    # Work on a copy
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - target_sum
    ind = np.arange(1, len(v) + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.zeros_like(v)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """
    Linear convolution of 1‑D real arrays using a real FFT.
    Returns an array of length len(x)+len(y)-1.
    """
    size = 1 << ((len(x) + len(y) - 1).bit_length())
    fx = np.fft.rfft(x, n=size)
    fy = np.fft.rfft(y, n=size)
    conv = np.fft.irfft(fx * fy, n=size)[:len(x) + len(y) - 1]
    return conv


def _full_convolution_fft(a: np.ndarray) -> np.ndarray:
    """
    Compute the full (linear) convolution a * a via FFT.
    Result has length 2·len(a)-1.
    """
    return _conv_fft(a, a)


def _max_conv(a: np.ndarray) -> float:
    """Return max_{k} (a*a)[k] using FFT."""
    return _full_convolution_fft(a).max()


def _softmax_weights(conv: np.ndarray, temperature: float) -> np.ndarray:
    """
    Compute the soft‑max weights:
        w_k = exp( temperature * (conv_k - max(conv)) )
    Normalised so that Σ w_k = 1.
    """
    # shift by max to avoid overflow
    max_val = conv.max()
    shifted = conv - max_val
    # Use np.exp on the shifted values
    exp_vals = np.exp(temperature * shifted)
    total = exp_vals.sum()
    if total == 0:
        # fallback to uniform weights (should not happen)
        return np.full_like(conv, 1.0 / conv.size)
    return exp_vals / total


def _gradient_softmax(a: np.ndarray, w: np.ndarray) -> np.ndarray:
    """
    Gradient of the smoothed objective   f(a)= Σ_k w_k (a*a)[k] .
    The formula reduces to   g = 2 * (w_rev * a)  restricted to n entries,
    where w_rev is w reversed.
    """
    n = len(a)
    # Convolution of reversed weights with a
    conv = _conv_fft(w[::-1], a)
    start = len(w) - 1          # = 2*n-2
    g = 2.0 * conv[start:start + n]
    return g


def _make_candidate(n: int, p: float, offset: float, beta: float) -> np.ndarray:
    """
    Build an un‑normalised raw sequence
        a_i = (i + 1 + offset)^{-p} * (1 + beta * i/(n‑1))
    and project it onto the simplex of sum √(2·n).
    """
    idx = np.arange(n, dtype=float)                # i = 0 … n‑1
    base = np.power(idx + 1.0 + offset, -p)
    factor = 1.0 + beta * idx / (n - 1)
    raw = base * factor
    raw = np.maximum(raw, 0.0)                     # enforce non‑negativity
    target = np.sqrt(2.0 * n)
    return _project_simplex(raw, target)

def propose_candidate(seed: int = 123, budget_s: int = 1000, **kwargs):
    """

    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 2.0   # tiny safety margin

    # --------------------------------------------------------
    # 0. Choose a length (larger n usually yields a lower bound)
    # --------------------------------------------------------
    n = 10000                     # feel free to adjust; fits easily in the budget
    target_sum = np.sqrt(2.0 * n)

    # --------------------------------------------------------
    best_seq = None

        best_seq = _project_simplex(best_seq, target_sum)
    else:
        # fallback: a simple power‑law shape
        best_seq = _make_candidate(n, p=0.5, offset=0.0, beta=0.0)

    # --------------------------------------------------------
    sweep_deadline = time.time() + 0.15 * (deadline - time.time())

            jitter = (np.random.rand(n) - 0.5) * 0.01 * cand
            cand += jitter
            cand = _project_simplex(cand, target_sum)

            best_score = cand_score
            best_seq = cand.copy()
            curr = cand
            print(f"[local] iter {iteration:6d}  improved to {best_score:.6f}")

    print(f"Finished after {iteration} iterations – best score {best_score:.6f}")
    # Return as plain Python list
    return best_seq.tolist()

    conv = _conv_fft(w[::-1], a)          # length = 3n‑2
    start = len(w) - 1                    # = 2n‑2
    return 2.0 * conv[start:start + n]


def _make_candidate(n: int, p: float, offset: float, beta: float) -> np.ndarray:
    """Build a raw sequence a_i = (i+1+offset)^(-p)*(1+β·i/(n‑1))
    and project it onto the simplex Σa = sqrt(2·n)."""
    idx = np.arange(n, dtype=float)
    base = np.power(idx + 1.0 + offset, -p)
    factor = 1.0 + beta * idx / (n - 1)
    raw = np.maximum(base * factor, 0.0)
    target = np.sqrt(2.0 * n)
    return _project_simplex(raw, target)


# ------------------------------------------------------------
# Main search routine
# ------------------------------------------------------------

def propose_candidate(seed: int = 123, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative height sequence with a low
    ``evaluate_sequence`` score.

    Returns the best sequence (as a plain list) found before the
    time budget expires.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 2.0   # keep a small safety margin

    # ----------------------------------------------------
    # 0. Length of the sequence (larger n generally yields a lower bound)
    # ----------------------------------------------------
    n = 10000
    target_sum = np.sqrt(2.0 * n)

    # ----------------------------------------------------
    # 1. Warm start from a previously discovered sequence if available
    # ----------------------------------------------------
    best_seq = None
    hs = globals().get("height_sequence_1")
    if isinstance(hs, list) and len(hs) == n:
        best_seq = np.array(hs, dtype=float)
        best_seq = _project_simplex(best_seq, target_sum)
    else:
        # fallback: simple power‑law of exponent 0.5
        best_seq = _make_candidate(n, p=0.5, offset=0.0, beta=0.0)

    best_score = _max_conv(best_seq)
    print(f"[init] n={n}  score={best_score:.6f}")

    # ----------------------------------------------------
    # 2. Coarse random sweep over the 4‑parameter family
    # ----------------------------------------------------
    sweep_deadline = time.time() + 0.15 * (deadline - time.time())
    while time.time() < sweep_deadline:
        p = np.random.uniform(0.35, 0.65)
        off = np.random.uniform(0.0, 12.0)
        beta = np.random.uniform(-0.9, 2.7)
        cand = _make_candidate(n, p, off, beta)
        cand_score = _max_conv(cand)
        if cand_score < best_score:
            best_score = cand_score
            best_seq = cand
            print(f"[sweep] new best {best_score:.6f}  p={p:.3f} α={off:.3f} β={beta:.3f}")
        if best_score < 1.503:   # early exit if we already beat the target
            break

    # ----------------------------------------------------
    # 3. Projected gradient descent on a temperature‑annealed soft‑max
    # ----------------------------------------------------
    cur = best_seq.copy()
    step0 = 0.8          # initial step size (decays like 1/√t)
    iteration = 0

    start_temp = 5.0
    end_temp = 200.0

    while time.time() < deadline:
        iteration += 1

        # current objective value (exact max of the convolution)
        conv_full = _full_convolution_fft(cur)
        M = conv_full.max()
        if M < best_score:
            best_score = M
            best_seq = cur.copy()
            print(f"[sgd] iter {iteration:6d}  new best {best_score:.6f}")
            if best_score < 1.503:
                break

        # temperature for the soft‑max (linearly annealed)
        progress = (time.time() - start_time) / (deadline - start_time)
        temperature = start_temp + progress * (end_temp - start_temp)

        # soft‑max weights and gradient
        w = _softmax_weights(conv_full, temperature)
        grad = _gradient_softmax(cur, w)

        # step size schedule
        step = step0 / np.sqrt(iteration)

        # gradient step + projection onto the simplex
        nxt = cur - step * grad
        nxt = _project_simplex(nxt, target_sum)

        # occasional tiny jitter to escape plateaus
        if iteration % 100 == 0 and np.random.rand() < 0.25:
            jitter = (np.random.rand(n) - 0.5) * 0.01 * nxt
            nxt += jitter
            nxt = _project_simplex(nxt, target_sum)

        cur = nxt

    # ----------------------------------------------------
    # 4. Very cheap local refinement: tiny pairwise transfers
    # ----------------------------------------------------
    while time.time() < deadline - 0.2:
        i, j = np.random.randint(0, n, size=2)
        if i == j:
            continue
        eps = 0.001 * cur[i]          # move 0.1 % of the source height
        if eps <= 0:
            continue
        cand = cur.copy()
        cand[i] -= eps
        cand[j] += eps
        if cand[i] < 0:
            continue
        cand_score = _max_conv(cand)
        if cand_score < best_score:
            best_score = cand_score
            best_seq = cand.copy()
            cur = cand
            print(f"[local] iter {iteration:6d}  improved to {best_score:.6f}")

    print(f"Finished after {iteration} iterations – best score {best_score:.6f}")
    return best_seq.tolist()
