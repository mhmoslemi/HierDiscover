def propose_candidate(seed=0, budget_s=1000):
    import time, numpy as np
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.1

    # Reference sequence
    try:
        init_seq = np.array(height_sequence_1, dtype=np.float64)
    except Exception:
        # fallback: binomial
        n = 5000
        init_seq = binomial_sequence(n)
    n = len(init_seq)

    # Normalize
    a = init_seq.copy()
    a = np.maximum(a, 0.0)
    total = a.sum()
    if total < 1e-12:
        a = np.full(n, 1/n)
    else:
        a = a / total

    # Compute convolution via np.convolve (full)
    b = np.convolve(a, a)
    cur_max = b.max()
    best_max = cur_max
    best_a = a.copy()
    best_score = 2.0 * n * cur_max

    # Parameters
    # Temperature schedule (linear cooling)
    T0 = 5e-5  # initial temperature; adjust after experiments
    T_end = 1e-7
    # step scale (max fraction of source mass transferred per move)
    step_initial = 0.1  # 10% of source mass
    step_min = 0.001
    # number of iterations per print
    print_interval = 5.0
    next_print = time.time() + print_interval

    iter_count = 0
    while time.time() < deadline:
        elapsed = time.time() - (deadline - budget_s)
        # linear temperature
        T = T0 + (T_end - T0) * (elapsed / budget_s)

        # step scale decays
        step_scale = step_initial * (1 - elapsed / budget_s) + step_min * (elapsed / budget_s)

        # Choose source index j (mass donor)
        # Weighted random selection by a (more mass -> larger effect) or uniform
        if np.random.rand() < 0.7:  # weighted by a^p
            probs = a / a.sum()
            j = np.random.choice(n, p=probs)
        else:
            j = np.random.randint(n)
        # Choose target i (different)
        i = np.random.randint(n)
        while i == j:
            i = np.random.randint(n)

        # Determine epsilon
        max_transfer = min(a[j], step_scale * a[j])  # cannot exceed a[j]
        eps = np.random.rand() * max_transfer

        if eps <= 0:
            continue

        # Update b via incremental addition using original a
        # Save copy of a for incremental use
        a_old = a.copy()  # Could avoid copy but we need original values for update; we can compute using a array before modifications.

        # Efficient vectorized updates:
        # b[i:i+n] += 2*eps*a_old
        # b[j:j+n] -= 2*eps*a_old
        # Use slicing
        # Note: For large n, slicing copies view but okay
        # Use b_view = b[i:i+n] etc.
        # But we must ensure dtype is float64.

        b[i:i+n] += 2.0 * eps * a_old
        b[j:j+n] -= 2.0 * eps * a_old

        # Update a
        a[i] += eps
        a[j] -= eps

        # Ensure rounding errors
        # Optionally zero negative small values
        if a[j] < 0:
            # Clip and adjust b corrections? For safety skip move if leads negative (shouldn't)
            a[j] = 0.0

        # Compute new max
        new_max = b.max()
        cur_score = 2.0 * n * new_max

        # Accept/reject based on temperature
        accept = False
        if new_max <= cur_max:
            accept = True
        else:
            # Metropolis criterion
            if np.random.rand() < np.exp(-(new_max - cur_max) / (T + 1e-12)):
                accept = True

        if accept:
            cur_max = new_max
            # update best if improved
            if new_max < best_max:
                best_max = new_max
                best_a = a.copy()
                best_score = 2.0 * n * best_max
                if time.time() >= next_print:
                    print(f"[improve] iter={iter_count}, score={best_score:.9f}, time={time.time() - (deadline - budget_s):.2f}s")
                    next_print = time.time() + print_interval
        else:
            # Reject: revert changes
            # revert a
            a[i] -= eps
            a[j] += eps
            # revert b
            b[i:i+n] -= 2.0 * eps * a_old
            b[j:j+n] += 2.0 * eps * a_old
            # cur_max remains
        iter_count += 1

    # final best
    print(f"[final] best score={best_score:.9f}, iterations={iter_count}")
    return best_a.tolist()

"""
propose_candidate
-----------------
A fast simulated‑annealing / gradient‑guided hill‑climber for the
step‑function optimisation problem.  The algorithm works on a fixed
length `n` (taken from the supplied `height_sequence_1` if present,
otherwise a binomial seed).  The vector `a` is kept on the probability
simplex (non‑negative, sum = 1) – the evaluation is scale‑invariant.

At each iteration the convolution `b = a * a` is maintained incrementally.
The maximal entries of `b` are identified; a sub‑gradient `g` of the
maximum (obtained by convolving the mask of the maximal entries with
the reversed `a`) indicates which coordinates contribute most to the
peak.  Mass is transferred from the index with largest `g` to the one
with smallest `g`.  The transfer size is a fraction of the source mass
and decays slowly.  Moves that increase the maximal convolution value
are rejected (or accepted only with a tiny Metropolis probability) –
this yields a monotone descent with occasional random perturbations to
escape plateaus.  To avoid accumulation of rounding errors the full
convolution is recomputed periodically, and after a long period without
improvement the algorithm restarts from a fresh random Dirichlet
distribution.

The routine runs for the allotted time budget (default ≈ 1000 s) and
returns the best sequence found.  All numbers are non‑negative and the
sum is ≈ 1, so the external ``evaluate_sequence`` sees the correct
objective value.
"""

import time
import numpy as np


def binomial_sequence(length: int) -> np.ndarray:
    """Central binomial coefficients c_k = C(2k,k)/4**k, k=0..length‑1."""
    seq = np.empty(length, dtype=np.float64)
    seq[0] = 1.0
    for k in range(1, length):
        seq[k] = seq[k - 1] * ((2 * k - 1) * (2 * k)) / (4.0 * k * k)
    return seq


def propose_candidate(seed: int = 0, budget_s: int = 1000) -> list[float]:
    """
    Search for a non‑negative step‑function height sequence that
    minimises ``evaluate_sequence``.  The method is a fast gradient‑guided
    stochastic local search as described in the module docstring.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.1   # safety margin

    # ------------------------------------------------------------------
    # Initialise a from the supplied construction if possible.
    try:
        a = np.array(height_sequence_1, dtype=np.float64)  # type: ignore[name-defined]
    except Exception:
        # fallback: a binomial seed of moderate size
        n = 5200
        a = binomial_sequence(n)

    # Ensure non‑negativity and normalise to sum = 1 (scale‑invariant)
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-12:
        a = np.full_like(a, 1.0 / len(a))
    else:
        a = a / s

    n = len(a)

    # Exact convolution (used for the first iteration and periodic resets)
    b = np.convolve(a, a)
    cur_max = b.max()

    best_a = a.copy()
    best_max = cur_max
    best_score = 2.0 * n * best_max

    # ------------------------------------------------------------------
    # Hyper‑parameters
    init_step = 0.07          # initial fraction of donor mass to move
    step = init_step
    step_decay = 0.99996     # exponential decay per iteration
    min_step = 1e-5

    stagnation_limit = 150.0  # seconds without improvement before a restart
    last_improve = start_time

    recompute_interval = 4000    # full recompute of b every this many steps
    print_interval = 10.0       # progress print frequency (seconds)
    next_print = start_time + print_interval

    iter_cnt = 0
    reject_cnt = 0

    while time.time() < deadline:
        # --------------------------------------------------------------
        # 1.  Check for improvement
        if cur_max < best_max - 1e-14:
            best_max = cur_max
            best_a = a.copy()
            best_score = 2.0 * n * best_max
            last_improve = time.time()
            print(
                f"[improve] iter={iter_cnt}, score={best_score:.9f}, "
                f"time={time.time() - start_time:.2f}s"
            )

        # --------------------------------------------------------------
        # 2.  Identify the active set of maximal convolution entries
        max_val = cur_max
        # tiny tolerance to capture all numerically maximal entries
        tol = max_val - 1e-15
        max_idx = np.flatnonzero(b >= tol)

        # --------------------------------------------------------------
        # 3.  Compute a sub‑gradient of the maximum (vector g of length n)
        #     Using convolution of the mask with the reversed a.
        mask = np.zeros(2 * n - 1, dtype=np.float64)
        mask[max_idx] = 1.0
        # convolution: g_i = Σ_k mask_k * a_{k-i}
        conv = np.convolve(mask, a[::-1])
        g = conv[n - 1 : n - 1 + n]  # slice of length n

        # --------------------------------------------------------------
        # 4.  Choose donor (source) and receiver (target) indices
        src = int(np.argmax(g))
        tgt = int(np.argmin(g))

        # If the chosen source has no mass (possible after many moves), fall back
        if src == tgt or a[src] <= 0.0:
            # pick a random donor with positive mass
            positive = np.flatnonzero(a > 0)
            if len(positive) == 0:
                break  # should never happen
            src = int(np.random.choice(positive))
            tgt = int(np.random.randint(n))
            while tgt == src:

        # --------------------------------------------------------------

        max_transfer = step * a[src]          # at most this fraction of the donor mass
        eps = max_transfer * np.random.rand()  # randomise a bit

        # --------------------------------------------------------------
        # 7.  Apply the mass transfer to a

        # 8.  Evaluate the new maximal entry

            reject_cnt = 0

            a[src] += eps
            b[tgt : tgt + n] -= delta
            b[src : src + n] += delta

            # If many consecutive rejections, shrink the step size aggressively
            if reject_cnt > 200:
                step = max(min_step, step * 0.5)
                reject_cnt = 0

        # --------------------------------------------------------------
        # 10.  Decay step size
        step = max(min_step, step * step_decay)

        iter_cnt += 1

        # --------------------------------------------------------------
        # 11.  Periodic full recomputation of the convolution to erase
        #     accumulated O(ε²) drift.
        if (iter_cnt % recompute_interval) == 0:
            b = np.convolve(a, a)
            cur_max = b.max()

        # --------------------------------------------------------------
        # 12.  Restart if we have been stagnant for a while
        if time.time() - last_improve > stagnation_limit:
            # fresh random start (Dirichlet gives a valid simplex point)
            a = np.random.dirichlet(np.ones(n))
            b = np.convolve(a, a)
            cur_max = b.max()
            step = init_step
            last_improve = time.time()

                f"time={time.time() - start_time:.2f}s"
            )

        # 13.  Periodic progress report

                f"elapsed={elapsed:.1f}s"
            )

    print(f"[final] best score={best_score:.9f}, iterations={iter_cnt}")
    # The evaluation function clips values to [0,1000] – our numbers are tiny.
    return best_a.tolist()
