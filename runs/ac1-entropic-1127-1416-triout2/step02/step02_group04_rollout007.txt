"""
Propose a candidate sequence of non‑negative heights that (approximately)
minimises the evaluation function

    2·n·max(conv(a,a)) / (sum(a)^2)

The algorithm works in three stages:

1. **Initial seeding** – a collection of deterministic (ramp, uniform) and
   randomly generated Beta/Dirichlet vectors are examined; the best one is kept.

2. **Projected sub‑gradient descent with a Polyak step size** – starting from the
   best seed, the active constraints (indices where the convolution attains its
   maximal value) are identified, a sub‑gradient of the maximal convolution value
   is assembled and a step length
        (max‑value – theoretical lower bound) / ||g||²
   is used.  After each step the iterate is projected onto the probability
   simplex (non‑negative, sum = 1).  This phase usually yields a large part of the
   improvement.

3. **Simulated‑annealing fine‑tuning** – the final phase performs cheap
   “mass‑transfer’’ moves (take a small amount from one index and give it to
   another) together with occasional global random re‑initialisations.  Moves are
   accepted with a decreasing temperature schedule, allowing the search to escape
   shallow local minima.

The whole search respects the 1000 s time budget and returns the best sequence
found (as a plain Python list of floats).  All numbers are non‑negative, sum to
one and therefore are safe for the provided ``evaluate_sequence`` function.
"""

import time
import math
import numpy as np
from typing import List

# ----------------------------------------------------------------------
# Helper functions (all top‑level, no closures)
# ----------------------------------------------------------------------


def simplex_projection(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the probability simplex:
    { w | w_i ≥ 0 , sum_i w_i = 1 }.
    Implementation follows Duchi et al. (2008).
    """
    v = np.asarray(v, dtype=float)
    n = v.shape[0]
    # sort v in decreasing order
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all entries are non‑positive → uniform distribution
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1] - 1               # zero‑based index
    theta = cssv[cond][-1] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def beta_sequence(alpha: float, beta: float, n: int) -> np.ndarray:
    """
    Discretised Beta(α,β) distribution on ``n`` points,
    normalised to sum to one.
    """
    x = (np.arange(n) + 0.5) / n               # points strictly in (0,1)
    logw = (alpha - 1.0) * np.log(x) + (beta - 1.0) * np.log(1.0 - x)
    # avoid overflow / underflow
    logw = np.clip(logw, -700.0, 700.0)
    w = np.exp(logw)
    total = w.sum()
    if total == 0.0 or not np.isfinite(total):
        w = np.ones_like(w)
        total = w.sum()
    return w / total


def random_mixture_beta(n: int) -> np.ndarray:
    """
    Build a mixture of 1–3 Beta densities with random parameters
    and random mixing weights.  The result is normalised to sum = 1.
    """
    k = np.random.randint(1, 4)               # number of components
    weights = np.random.rand(k)
    weights /= weights.sum()
    a = np.zeros(n, dtype=float)
    for i in range(k):
        alpha = np.random.uniform(0.05, 6.0)
        beta = np.random.uniform(0.05, 6.0)
        a += weights[i] * beta_sequence(alpha, beta, n)
    s = a.sum()
    if s == 0.0:
        a[:] = 1.0 / n
    else:
        a /= s
    return a


def random_dirichlet_like(n: int) -> np.ndarray:
    """
    Generate a random vector with Dirichlet‑like behaviour.
    Concentration parameters are drawn uniformly from [0.1, 3].
    """
    conc = np.random.uniform(0.1, 3.0, size=n)
    a = np.random.gamma(conc, 1.0)
    s = a.sum()
    if s == 0.0:
        a[:] = 1.0 / n
    else:
        a /= s
    return a


def max_convolution(a: np.ndarray) -> float:
    """
    Return max(conv(a, a)) where conv is the linear convolution.
    The vector ``a`` is assumed to sum to one (the scaling does not matter
    for the ratio used in the evaluation function).
    """
    n = a.shape[0]
    # linear convolution via FFT (zero‑padding to length 2·n‑1)
    fft = np.fft.rfft(a, n=2 * n - 1)
    conv = np.fft.irfft(fft * fft, n=2 * n - 1)
    return float(conv.max())


def objective_from_max(max_conv: float, n: int) -> float:
    """Exact objective value from the maximal convolution entry."""
    return 2.0 * n * max_conv


# ----------------------------------------------------------------------
# Main search routine
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 0, budget_s: int = 1000) -> List[float]:
    """
    Search for a non‑negative sequence of heights that minimises the
    provided evaluation function.  The routine respects a 1000 s time
    budget and returns the best sequence it discovers.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 5.0            # safety margin

    # --------------------------------------------------------------
    # 1. Problem size
    # --------------------------------------------------------------
    try:
        n = len(height_sequence_1)   # type: ignore[name-defined]
        prev_exists = True
    except NameError:
        n = 5146                     # fallback length from the prompt
        prev_exists = False

    # --------------------------------------------------------------
    # 2. Utility closures (local for readability)
    # --------------------------------------------------------------
    def norm_one(v: np.ndarray) -> np.ndarray:
        """Clamp to non‑negative and renormalise to sum = 1."""
        v = np.maximum(v, 0.0)
        s = v.sum()
        if s == 0.0:
            v[:] = 1.0 / n
        else:
            v /= s
        return v

    # --------------------------------------------------------------
    # 3. Initial pool of candidates
    # --------------------------------------------------------------
    best_a = None
    best_score = np.inf

    def try_candidate(a: np.ndarray, label: str = "") -> None:
        nonlocal best_a, best_score
        a = norm_one(a)
        mb = max_convolution(a)
        score = objective_from_max(mb, n)
        if score < best_score:
            best_score = score
            best_a = a.copy()
            print(f"{label:30s} -> score = {score:.6f}")

    # a) previous construction (if it exists)
    if prev_exists:
        try_candidate(np.array(height_sequence_1, dtype=float), "previous best")

    # b) deterministic baselines
    a_uniform = np.full(n, 1.0 / n)
    try_candidate(a_uniform, "uniform")

    a_ramp = np.arange(n, 0, -1, dtype=float)
    a_ramp = 2.0 * a_ramp / (n * (n + 1))
    try_candidate(a_ramp, "linear ramp")

    # c) systematic Beta grid (moderately fine)
    alphas = np.linspace(0.2, 6.0, 16)
    betas = np.linspace(0.2, 6.0, 16)
    for alpha in alphas:
        for beta in betas:
            if time.time() > deadline:
                break
            try_candidate(beta_sequence(alpha, beta, n),
                           f"Beta({alpha:.2f},{beta:.2f})")
        if time.time() > deadline:
            break

    # d) random Beta mixtures
    for _ in range(300):
        if time.time() > deadline:
            break
        try_candidate(random_mixture_beta(n), "random Beta‑mix")

    # e) random Dirichlet‑like vectors
    for _ in range(30):
        if time.time() > deadline:
            break
        try_candidate(random_dirichlet_like(n), "random Dirichlet")

    # At this point ``best_a`` holds the best seed we have seen.
    if best_a is None:
        # fall back to uniform (should never happen)
        best_a = a_uniform.copy()
        best_score = objective_from_max(max_convolution(best_a), n)

    # --------------------------------------------------------------
    # 4. Phase 1 – Projected sub‑gradient descent (Polyak step)
    # --------------------------------------------------------------
    a = best_a.copy()
    lower_bound_max = 1.0 / (2 * n - 1)               # theoretical minimum of max(conv)
    max_iterations_phase1 = 20000
    iteration = 0

    while time.time() < deadline and iteration < max_iterations_phase1:
        # compute current convolution and maximal entry
        fft = np.fft.rfft(a, n=2 * n - 1)
        conv = np.fft.irfft(fft * fft, n=2 * n - 1)
        max_b = float(conv.max())
        cur_score = objective_from_max(max_b, n)

        # keep global best
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"Iter {iteration:5d}: new best {best_score:.6f}")

        # stop if we already reached numerical lower bound
        if max_b <= lower_bound_max + 1e-12:
            break

        # active constraints – entries within relative tolerance of max
        eps = max_b * 1e-6
        active = np.where(conv >= max_b - eps)[0]
        if active.size == 0:
            active = np.array([conv.argmax()], dtype=int)

        # sub‑gradient of the maximal convolution value
        g = np.zeros_like(a)
        for k in active:
            low = max(0, k - n + 1)
            high = min(n - 1, k)
            idx = np.arange(low, high + 1, dtype=int)
            partner = k - idx
            g[idx] += 2.0 * a[partner]
        g /= float(active.size)

        # Polyak step size (optimal for convex sub‑gradient with known lower bound)
        g_norm_sq = float(np.dot(g, g))
        if g_norm_sq == 0.0:
            break
        step = (max_b - lower_bound_max) / g_norm_sq

        # safety – never step too far (clip)
        step = max(step, 0.0)
        # try a few reductions if the projection does not improve the objective
        accepted = False
        for _ in range(5):
            a_candidate = simplex_projection(a - step * g)
            mb_cand = max_convolution(a_candidate)
            cand_score = objective_from_max(mb_cand, n)
            if cand_score < cur_score - 1e-12:
                a = a_candidate
                accepted = True
                break
            step *= 0.5       # halve and retry
        if not accepted:
            # if nothing worked, shrink step aggressively and continue
            step *= 0.2

        iteration += 1

    # --------------------------------------------------------------
    # 5. Phase 2 – Simulated annealing fine‑tuning
    # --------------------------------------------------------------
    # Use the best vector from phase 1 as the starting point.
    cur_a = best_a.copy()
    cur_score = best_score
    best_local_a = cur_a.copy()
    best_local_score = cur_score

    # annealing parameters
    total_time = deadline - time.time()
    if total_time > 5.0:
        T0 = (cur_score - 1.0) * 0.2 + 1e-3          # initial temperature (rough)
        Tmin = 1e-6
        step_scale = 0.05                           # initial fraction of mass to move
        iter_anneal = 0
        while time.time() < deadline:
            elapsed = time.time() - (deadline - total_time)
            frac = elapsed / total_time
            # linear cooling
            T = max(Tmin, T0 * (1.0 - frac))

            # decide move type
            if np.random.rand() < 0.02:          # occasional global jump
                if np.random.rand() < 0.5:
                    cand = random_mixture_beta(n)
                else:
                    cand = random_dirichlet_like(n)
            else:
                # tiny transfer move
                i = np.random.randint(n)
                j = np.random.randint(n)
                while j == i:
                    j = np.random.randint(n)
                delta = cur_a[i] * np.random.rand() * step_scale
                if delta > cur_a[i]:
                    delta = cur_a[i]
                cand = cur_a.copy()
                cand[i] -= delta
                cand[j] += delta

            cand_score = objective_from_max(max_convolution(cand), n)

            # Metropolis acceptance
            if cand_score < cur_score or np.random.rand() < math.exp(-(cand_score - cur_score) / T):
                cur_a = cand
                cur_score = cand_score
                if cand_score < best_local_score:
                    best_local_score = cand_score
                    best_local_a = cand.copy()
                    print(f"Anneal {iter_anneal:6d}: new best {best_local_score:.6f}")

            # gradually shrink the move magnitude
            if iter_anneal % 2000 == 0 and iter_anneal > 0:
                step_scale *= 0.9
                step_scale = max(step_scale, 1e-4)

            iter_anneal += 1

        # Keep the best found in the annealing phase if it improved
        if best_local_score < best_score:
            best_score = best_local_score
            best_a = best_local_a.copy()

    # --------------------------------------------------------------
    # 6. Return the best sequence (plain Python list)
    # --------------------------------------------------------------
    print(f"Finished search: best score = {best_score:.6f}")
    return [float(v) for v in best_a.tolist()]
