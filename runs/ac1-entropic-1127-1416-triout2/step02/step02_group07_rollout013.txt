"""
Propose a sequence of non‑negative heights that (approximately) minimizes

    2 * n * max(conv(a, a)) / (sum(a)**2)

The algorithm works in three stages:

1. **Small‑scale LP refinement** (n≈2000) – if SciPy is available.
   We solve a linear program that maximises the total mass `sum(b)`
   under the constraint `conv(a, b) ≤ M` where `M` is the current
   supremum of `conv(a, a)`.  Mixing the solution with the current
   vector strictly reduces the objective.

2. **Upsampling** – the improved small vector is interpolated to a
   larger size (n≈5000) and projected back onto the probability simplex.

3. **Projected sub‑gradient descent with momentum** on the large vector.
   The gradient is the sum of sub‑gradients corresponding to the top‑k
   largest convolution entries; a simple momentum term and occasional
   random jitter help escape plateaux.

The routine respects a 1000 s wall‑clock budget, prints progress and
returns a plain Python list of floats.  A safe fallback uses the global
`height_sequence_1` if it exists, otherwise a uniform distribution.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper functions (top‑level, no closures)
# ----------------------------------------------------------------------


def project_onto_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the probability simplex
    { x | x ≥ 0, sum(x) = 1 }.
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def full_convolution(a: np.ndarray) -> np.ndarray:
    """
    Compute the full discrete convolution a*a using an FFT.
    Result length = 2*len(a) - 1.
    """
    n = a.shape[0]
    size = 2 * n
    fft_len = 1 << (size - 1).bit_length()   # next power of two
    fa = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(fa * fa, n=fft_len)
    return conv[: 2 * n - 1]


def gradient_at_max(a: np.ndarray, k: int) -> np.ndarray:
    """
    Sub‑gradient of the convolution entry at shift k with respect to a.
    For each i contributing to that entry we set grad[i] = a[k-i].
    """
    n = a.shape[0]
    low = max(0, k - n + 1)
    high = min(k, n - 1)
    if low > high:
        return np.zeros_like(a)
    idx = np.arange(low, high + 1, dtype=int)
    g = np.zeros_like(a)
    g[idx] = a[k - idx]
    return g


def combined_gradient(a: np.ndarray, ks) -> np.ndarray:
    """
    Sum of sub‑gradients for a collection of shift indices `ks`.
    """
    g = np.zeros_like(a)
    for k in ks:
        g += gradient_at_max(a, int(k))
    return g


def init_power_shape(n: int) -> np.ndarray:
    """
    Classic decreasing shape a_i ∝ 1 / sqrt(i+1), projected onto the simplex.
    """
    raw = 1.0 / np.sqrt(np.arange(n, dtype=float) + 1.0)
    a = raw / raw.sum()
    return a


def build_conv_ub_matrix(a: np.ndarray):
    """
    Construct the matrix A (size (2n‑1) × n) such that
        (A @ b)[k] = (a * b)[k]
    for all shifts k.  The matrix is built as a sparse CSR matrix
    using constant diagonals, each diagonal d contains the value a[d].
    """
    from scipy.sparse import diags

    n = a.shape[0]
    # each diagonal d (0 ≤ d < n) has length n and constant value a[d]
    data = [np.full(n, a[d]) for d in range(n)]
    offsets = np.arange(0, n)                     # d = row‑col
    A = diags(data, offsets, shape=(2 * n - 1, n), format="csr")
    return A


def solve_lp_for_g(a: np.ndarray, M: float):
    """
    Solve the LP:
        max   sum(b)
        s.t.  conv(a, b) ≤ M   (element‑wise)
              b ≥ 0
    Returns the optimal b (or None if the solver fails).
    """
    try:
        from scipy.optimize import linprog
        A = build_conv_ub_matrix(a)
        n = a.shape[0]
        c = -np.ones(n)                        # maximise sum(b) == minimise -sum(b)
        b_ub = np.full(2 * n - 1, M)
        res = linprog(c, A_ub=A, b_ub=b_ub,
                      bounds=[(0, None)] * n, method="highs")
        if res.success:
            return res.x
    except Exception:
        pass
    return None

        a_candidate = (a_small + b_norm) / 2.0
        a_candidate = project_onto_simplex(a_candidate)
        score_cand = 2.0 * n_small * np.max(full_convolution(a_candidate))
        if score_cand < best_score_small:
            best_score_small = score_cand
            best_small = a_candidate.copy()
            print(f"[LP] iter {it}: improved score {best_score_small:.6f}")
        a_small = a_candidate

    # ------------------------------------------------------------------
    # Stage 2 – upsample to a larger size (n ≈ 5000)
    # ------------------------------------------------------------------
    n_large = 5000
    x_small = np.linspace(0, 1, n_small)
    x_large = np.linspace(0, 1, n_large)
    a_large = np.interp(x_large, x_small, best_small)
    a_large = project_onto_simplex(a_large)

    best_seq = a_large.copy()
    best_score = 2.0 * n_large * np.max(full_convolution(a_large))
    print(f"[Large] after up‑sampling, score {best_score:.6f}")

    # ------------------------------------------------------------------
    # Stage 3 – projected sub‑gradient descent with momentum on the large n
    # ------------------------------------------------------------------
    iteration = 0
    step_initial = 0.8           # initial step‑size multiplier
    beta = 0.9                   # momentum decay
    momentum = np.zeros_like(a_large)
    top_k = 5                    # number of top convolution entries used for the gradient

    while time.time() < deadline:
        conv = full_convolution(a_large)
        max_val = np.max(conv)

        # indices of the top‑k entries (including the absolute maximum)
        top_idx = np.argpartition(conv, -top_k)[-top_k:]

        # compute a combined sub‑gradient
        grad = combined_gradient(a_large, top_idx)

        # diminishing step size
        step = step_initial / np.sqrt(iteration + 1.0)

        # momentum update
        momentum = beta * momentum - step * grad

        # take a step and re‑project onto the simplex
        a_candidate = a_large + momentum
        a_candidate = project_onto_simplex(a_candidate)

        # evaluate the candidate
        score = 2.0 * n_large * np.max(full_convolution(a_candidate))

        if score < best_score:
            best_score = score
            best_seq = a_candidate.copy()
            a_large = a_candidate
            momentum = np.zeros_like(a_large)   # reset after a successful move
            print(f"[SGD] iter {iteration:06d}: improved score {best_score:.6f}")
        else:
            # keep the candidate (it already contains the momentum move)
            a_large = a_candidate

        # --------------------------------------------------------------
        # Random jitter – helps to escape shallow plateaux
        # --------------------------------------------------------------
        if iteration % 250 == 0 and iteration > 0:
            i, j = np.random.randint(0, n_large, size=2)
            if i != j:
                delta = 0.01 * a_large[i] * (np.random.rand() * 2.0 - 1.0)  # up to ±1 % of mass at i
                cand = a_large.copy()
                cand[i] = max(cand[i] - delta, 0.0)
                cand[j] = cand[j] + delta
                cand = project_onto_simplex(cand)
                cand_score = 2.0 * n_large * np.max(full_convolution(cand))
                if cand_score < best_score:
                    best_score = cand_score
                    best_seq = cand.copy()
                    a_large = cand
                    momentum = np.zeros_like(a_large)
                    print(f"[Jitter] iter {iteration:06d}: improved score {best_score:.6f}")

        # early exit if target reached
        if best_score < 1.503:
            print("[Info] target score achieved – early termination")
            break

        iteration += 1

    # ------------------------------------------------------------------
    # Safety fallback – use the previous global construction if needed
    # ------------------------------------------------------------------
    if best_seq is None or len(best_seq) == 0:
        try:
            best_seq = np.array(height_sequence_1, dtype=float)
            print("[Fallback] using global height_sequence_1")
        except Exception:
            best_seq = np.full(2000, 1.0 / 2000.0)
            print("[Fallback] using uniform distribution")

    # final sanity check with the provided evaluator (if available)
    try:
        final_score = evaluate_sequence(best_seq.tolist())
        print(f"[Result] final evaluated score = {final_score:.6f} (length {len(best_seq)})")
    except Exception:
        print(f"[Result] final internal score = {best_score:.6f} (length {len(best_seq)})")

    # driver expects a plain Python list
    return best_seq.tolist()
