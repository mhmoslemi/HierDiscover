"""
Propose a step‑function height sequence that minimises

    2·n·max(conv(a,a)) / (sum(a)^2)

where conv(a,a) is the linear self‑convolution.
The optimisation is performed on the probability simplex
(sum a = 1, a_i ≥ 0) using a smooth approximation of the
max‑operator (soft‑max) together with an Adam optimiser and
Euclidean projection onto the simplex (Duchi et al. 2008).
A short initial exploration (ramp, uniform, beta‑shapes,
Dirichlet draws) supplies a good starting point, after which
the gradient‑based refinement seeks a lower L∞‑norm of the
autocorrelation.  The algorithm respects the 1000 s time budget
and returns the best sequence found.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities
# ----------------------------------------------------------------------


def simplex_projection(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex
    { w | w_i ≥ 0 , sum w_i = 1 }.
    Implementation follows Duchi et al. (2008).
    """
    v = np.asarray(v, dtype=float)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1] - 1
    theta = cssv[cond][-1] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def compute_objective(a: np.ndarray) -> float:
    """Exact value of the evaluation function when sum(a)=1:
    2·n·max(a*a)  (a*a denotes linear convolution).
    """
    n = a.shape[0]
    # use FFT for speed – length 2*n‑1 is enough for a linear conv.
    fft_len = 2 * n - 1
    Fa = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(Fa * Fa, n=fft_len)
    max_b = conv.max()
    return 2.0 * n * max_b


def beta_sequence(alpha: float, beta: float, n: int) -> np.ndarray:
    """Discretised Beta(α,β) density on n mid‑points, normalised to sum 1."""
    x = (np.arange(n) + 0.5) / n
    logw = (alpha - 1.0) * np.log(x) + (beta - 1.0) * np.log(1.0 - x)
    logw = np.clip(logw, -700.0, 700.0)          # protect from overflow
    w = np.exp(logw)
    total = w.sum()
    if total == 0 or not np.isfinite(total):
        w = np.ones_like(w)
        total = w.sum()
    return w / total


# ----------------------------------------------------------------------
# Main search routine
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 0, budget_s: int = 1000) -> list[float]:
    """
    Search for a non‑negative height sequence that minimises the
    evaluation function.  Uses a smoothed max objective and an Adam
    optimiser on the simplex.  Returns the best sequence as a plain
    Python list of floats.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 5.0  # a few seconds safety margin

    # --------------------------------------------------------------
    # 1. Problem size
    # --------------------------------------------------------------
    try:
        n = len(height_sequence_1)   # type: ignore[name-defined]
        # use the existing sequence as a convenient warm‑start
        a_start = np.array([float(v) for v in height_sequence_1], dtype=float)
        a_start = np.maximum(a_start, 0.0)
        a_start /= a_start.sum()
    except NameError:
        # fallback size from the prompt
        n = 5146
        a_start = np.arange(n, 0, -1, dtype=float)
        a_start = a_start / a_start.sum()

    best_a = a_start.copy()
    best_score = compute_objective(best_a)
    print(f"Initial candidate score = {best_score:.6f}")

    # --------------------------------------------------------------
    # 2. Cheap deterministic/beta/Dirichlet seeds
    # --------------------------------------------------------------
    # uniform
    a_uni = np.full(n, 1.0 / n)
    s = compute_objective(a_uni)
    if s < best_score:
        best_a, best_score = a_uni.copy(), s
        print(f"Uniform improved to {s:.6f}")

    # a few Beta distributions (coarse grid)
    al_grid = np.linspace(0.2, 5.0, 12)
    be_grid = np.linspace(0.2, 5.0, 12)
    for al in al_grid:
        for be in be_grid:
            if time.time() > deadline:
                break
            a = beta_sequence(al, be, n)
            s = compute_objective(a)
            if s < best_score:
                best_a, best_score = a.copy(), s
                print(f"Beta({al:.2f},{be:.2f}) → {s:.6f}")
        if time.time() > deadline:
            break

    # random Beta draws
    for _ in range(150):
        if time.time() > deadline:
            break
        al = np.random.uniform(0.05, 5.0)
        be = np.random.uniform(0.05, 5.0)
        a = beta_sequence(al, be, n)
        s = compute_objective(a)
        if s < best_score:
            best_a, best_score = a.copy(), s
            print(f"Rnd‑Beta({al:.2f},{be:.2f}) → {s:.6f}")

    # Dirichlet‑like samples
    for _ in range(12):
        if time.time() > deadline:
            break
        conc = np.random.uniform(0.1, 3.0, size=n)
        a = np.random.gamma(conc, 1.0)
        a /= a.sum()
        s = compute_objective(a)
        if s < best_score:
            best_a, best_score = a.copy(), s
            print(f"Dirichlet seed improved to {s:.6f}")

    # --------------------------------------------------------------
    # 3. Gradient‑based refinement (soft‑max + Adam)
    # --------------------------------------------------------------
    a = best_a.copy()
    # Adam state
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8
    lr = 0.12                     # base learning rate (pre‑projection)

    # smoothing parameter for the max (soft‑max temperature)
    alpha = 8.0
    alpha_max = 5000.0
    alpha_growth = 1.001          # exponential growth per iteration

    # FFT length large enough for both conv (2n‑1) and correlation (n+2n‑2)
    fft_len = 1 << ((3 * n - 2).bit_length())

    iteration = 0
    last_impr = 0

    while time.time() < deadline and iteration < 200000:
        # ----- self‑convolution (FFT) -----
        Fa = np.fft.rfft(a, n=fft_len)
        conv_full = np.fft.irfft(Fa * Fa, n=fft_len)
        conv = conv_full[: 2 * n - 1]               # linear convolution
        max_c = conv.max()

        # ----- smoothed max (log‑sum‑exp) -----
        # shift by max to avoid overflow
        shifted = conv - max_c
        exp_shift = np.exp(alpha * shifted)
        w_sum = exp_shift.sum()
        if w_sum == 0 or not np.isfinite(w_sum):
            w = np.full_like(conv, 1.0 / conv.shape[0])
        else:
            w = exp_shift / w_sum                       # softmax weights

        # ----- gradient of smoothed loss -----
        # correlation a ⋆ w  (cross‑correlation)
        Fw = np.fft.rfft(w, n=fft_len)
        corr_full = np.fft.irfft(np.conj(Fa) * Fw, n=fft_len)
        grad = 2.0 * np.real(corr_full[:n])           # sub‑gradient-like

        # ----- Adam update -----
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1 ** (iteration + 1))
        v_hat = v / (1.0 - beta2 ** (iteration + 1))
        step = lr * m_hat / (np.sqrt(v_hat) + eps)

        a = simplex_projection(a - step)             # stay on the simplex

        # ----- book‑keeping -----
        cur_score = 2.0 * n * max_c
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            last_impr = iteration
            if iteration % 1000 == 0:
                print(f"Iter {iteration:6d}: new best {best_score:.6f}")

        # ----- adapt smoothing parameter -----
        if alpha < alpha_max:
            alpha = min(alpha_max, alpha * alpha_growth)

        # ----- occasional random perturbation if stagnated -----
        if iteration - last_impr > 5000:
            sigma = 0.02
            a = best_a + sigma * np.random.randn(n)
            a = np.maximum(a, 0.0)
            a = simplex_projection(a)
            m.fill(0.0)
            v.fill(0.0)
            last_impr = iteration
            print(f"Stagnation reset at iter {iteration}")

        iteration += 1
        # occasional progress report
        if iteration % 5000 == 0:
            print(f"Iter {iteration:6d}: best score so far {best_score:.6f}")

    print(f"Finished after {iteration} iterations – best score = {best_score:.6f}")
    return [float(v) for v in best_a.tolist()]
