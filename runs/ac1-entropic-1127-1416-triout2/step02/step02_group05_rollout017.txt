"""
propose_candidate
-----------------
Projected sub‑gradient descent with an exact line‑search on the probability
simplex, plus occasional random mass‑transfer moves.  The sub‑gradient of the
objective  2·n·max(a∗a)/(sum a)²  is the sum of the current maximal
convolution entries; moving opposite to this direction reduces the peak.
For each iteration we compute the feasible step length (keeping all a_i ≥ 0)
and perform a ternary search on that interval to find the step that gives the
largest decrease of the maximal convolution entry.  The convolution is
evaluated by an FFT‑based routine (O(n log n)) which makes thousands of
iterations feasible within the 1000 s budget.  Random pairwise transfers are
added to escape possible plateaus, and the best vector seen is returned.
"""

import time
import numpy as np

# --------------------------------------------------------------
def binomial_sequence(length: int) -> np.ndarray:
    """
    Central binomial coefficients scaled by 4**k:
    c_k = C(2k,k)/4**k  (k=0,…,length‑1).
    """
    seq = np.empty(length, dtype=np.float64)
    seq[0] = 1.0
    for k in range(1, length):
        seq[k] = seq[k - 1] * ((2 * k - 1) * (2 * k)) / (4.0 * k * k)
    return seq


# --------------------------------------------------------------
def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection onto the probability simplex (non‑negative, sum = 1)."""
    v = np.maximum(v, 0.0)
    s = v.sum()
    if s <= 0.0:
        return np.full_like(v, 1.0 / v.size)
    return v / s


# --------------------------------------------------------------
def next_pow2(x: int) -> int:
    """Small utility – next power of two ≥ x."""
    return 1 << (x - 1).bit_length()


# --------------------------------------------------------------
def max_convolution_fft(a: np.ndarray, fft_len: int) -> tuple[float, np.ndarray]:
    """
    Compute a∗a via FFT (real spectra) and return (max value, full convolution).
    The length of the returned convolution is 2·len(a)‑1.
    """
    fa = np.fft.rfft(a, n=fft_len)
    fc = fa * fa
    conv = np.fft.irfft(fc, n=fft_len)[: 2 * a.size - 1]
    return float(conv.max()), conv


# --------------------------------------------------------------
def subgradient(a: np.ndarray, conv_full: np.ndarray,
                max_val: float, max_inds: np.ndarray) -> np.ndarray:
    """
    Sub‑gradient of the maximal convolution entry.
    g_i = Σ_{k ∈ max_inds} a_{k-i} .
    (The factor 2 of the true gradient is irrelevant – direction is the same.)
    """
    n = a.size
    g = np.zeros_like(a)
    for k in max_inds:
        lo = max(0, k - (n - 1))
        hi = min(n - 1, k)
        idx = np.arange(lo, hi + 1, dtype=np.int64)
        g[idx] += a[k - idx]          # a_{k-i}
    return g

                a2 = np.maximum(a2, 0.0)
                a1 /= a1.sum()
                a2 /= a2.sum()
                m1, _ = max_convolution_fft(a1, fft_len)
                m2, _ = max_convolution_fft(a2, fft_len)
                f1 = 2.0 * n * m1
                f2 = 2.0 * n * m2
                if f1 < f2:
                    right = t2
                else:
                    left = t1
            alpha_opt = (left + right) / 2.0
            a_new = a + alpha_opt * d
            a_new = np.maximum(a_new, 0.0)
            a_new /= a_new.sum()
            new_max, _ = max_convolution_fft(a_new, fft_len)
            new_score = 2.0 * n * new_max
            if new_score < cur_score - 1e-12:
                a = a_new
                cur_score = new_score
                iter_cnt += 1
                continue        # go to next iteration

        # --------------------------------------------------------------
        #  random pairwise mass transfer (simulated‑annealing style)
        i, j = np.random.randint(0, n, size=2)
        if i != j and a[i] > 1e-12:
            delta = 0.02 * np.random.rand() * a[i]   # move up to 2 % of source
            a_tmp = a.copy()
            a_tmp[i] -= delta
            a_tmp[j] += delta
            a_tmp /= a_tmp.sum()
            tmp_max, _ = max_convolution_fft(a_tmp, fft_len)
            tmp_score = 2.0 * n * tmp_max
            if tmp_score < cur_score:
                a = a_tmp
                cur_score = tmp_score
                iter_cnt += 1
                continue

        # --------------------------------------------------------------
        #  occasional restart / perturbation if stuck for a long time
        if no_improve_cnt > 5000:
            a = best_seq + 0.001 * np.random.randn(n)
            a = np.maximum(a, 0.0)
            a_sum = a.sum()
            if a_sum == 0.0:
                a = np.full(n, 1.0 / n)
            else:
                a /= a_sum
            no_improve_cnt = 0
            # recompute score after the perturbation
            cur_max, _ = max_convolution_fft(a, fft_len)
            cur_score = 2.0 * n * cur_max

        iter_cnt += 1

    # ------------------------------------------------------------------
    print(f"[final] best score={best_score:.9f}, iterations={iter_cnt}")
    return best_seq.tolist()
