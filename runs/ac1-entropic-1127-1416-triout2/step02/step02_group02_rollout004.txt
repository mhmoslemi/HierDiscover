def compute_objective(a):
    n = a.shape[0]
    conv = np.fft.irfft(np.fft.rfft(a, n=2*n-1) * np.fft.rfft(a, n=2*n-1), n=2*n-1)
    return 2.0 * n * conv.max()

def compute_gradient(a):
    n = a.shape[0]
    # compute convolution and find max
    fft_a = np.fft.rfft(a, n=2*n-1)
    conv = np.fft.irfft(fft_a * fft_a, n=2*n-1)
    max_val = conv.max()
    # active indices near max
    tol = max_val * 1e-12 + 1e-14
    active = np.where(conv >= max_val - tol)[0]
    grad = np.zeros_like(a)
    for k in active:
        low = max(0, k - n + 1)
        high = min(n-1, k)
        idx = np.arange(low, high+1)
        partner = k - idx
        grad[idx] += 2.0 * a[partner]
    if active.size > 1:
        grad /= active.size
    return grad, max_val

def generate_shape(alpha, n):
    i = np.arange(n)
    x = (i + 0.5) / n  # midpoints in (0,1)
    # avoid zeros
    # compute weighting
    # Use exponent (alpha-1)
    w = (x ** (alpha - 1)) * ((1 - x) ** (alpha - 1))
    # may encounter overflow for alpha<1; but we restrict alpha>=0.4
    # good.
    # Normalise to sum 1
    w_sum = w.sum()
    if w_sum == 0:
        return np.full(n, 1.0 / n)
    return w / w_sum

"""
Propose a candidate step‑function sequence.

The search consists of three stages:

1. **Parametric U‑shaped family** – a symmetric Beta‑like density
   `a_i ∝ (x_i)^{α-1} (1-x_i)^{α-1}` (with `x_i = (i+0.5)/n`),
   mixed with the uniform distribution.  By sweeping a small grid
   over the shape exponent `α` (≥0.5, i.e. edge‑heavy but finite) and
   the mixing weight `λ ∈ [0,1]` we obtain a good starting point that
   already reduces the convolution peak substantially.

2. **Projected Adam optimisation** – from the best parametric vector we
   run a few hundred Adam steps on the original objective

        f(a) = 2·n·max( a * a ) / (∑ a)² ,

   where `*` denotes the linear convolution.  The gradient of the
   max‑entry is easy to compute (see `compute_gradient`).  After each
   Adam step we project back onto the probability simplex using the
   O(n log n) Euclidean projection of Duchi et al. (2008).

3. **Time‑limited loop** – the whole routine stops a few seconds
   before the supplied time budget, returning the best sequence seen.

The algorithm runs with a single CPU, uses only NumPy (no external
solvers) and respects the 1000 s limit.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper functions (all top‑level – no closures)
# ----------------------------------------------------------------------


def simplex_projection(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the probability simplex:
        { w | w_i ≥ 0 , Σ w_i = 1 }.
    Implementation follows Duchi‑et‑al. (2008).
    """
    n = v.shape[0]
    # sort v in decreasing order
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all entries non‑positive → uniform distribution
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1] - 1               # zero‑based index
    theta = cssv[cond][-1] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def compute_objective(a: np.ndarray) -> float:
    """
    Objective 2·n·max(a*a) / (sum(a)^2).
    The input is assumed to be non‑negative and summed to 1,
    therefore the denominator is 1 and the formula simplifies to
        2·n·max(conv(a, a)).
    The convolution is evaluated by FFT for speed.
    """
    n = a.shape[0]
    # length of the linear convolution
    L = 2 * n - 1
    fft_a = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(fft_a * fft_a, n=L)
    return float(2.0 * n * conv.max())


def compute_gradient(a: np.ndarray):
    """
    Returns (grad, max_val) where `grad` is a sub‑gradient of the
    objective at `a` (with respect to the max‑entry) and `max_val`
    is the current maximal convolution value.
    """
    n = a.shape[0]
    L = 2 * n - 1
    fft_a = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(fft_a * fft_a, n=L)
    max_val = conv.max()
    # indices whose value is (numerically) maximal
    tol = max_val * 1e-12 + 1e-14
    active = np.where(conv >= max_val - tol)[0]

    grad = np.zeros_like(a)
    for k in active:
        low = max(0, k - n + 1)
        high = min(n - 1, k)
        idx = np.arange(low, high + 1, dtype=int)
        partner = k - idx
        grad[idx] += 2.0 * a[partner]          # derivative of a_i * a_{k-i}
    if active.size > 1:
        grad /= float(active.size)            # any convex combination is valid
    return grad, max_val


def beta_like_shape(alpha: float, n: int) -> np.ndarray:
    """
    Symmetric Beta‑like density on the discrete grid {0,…,n‑1}.
    For `alpha = 1` the distribution is uniform.
    The exponent `alpha ≥ 0.5` guarantees that all entries stay below 1000
    even for n = 20000.
    """
    i = np.arange(n, dtype=float)
    # mid‑point representation of x ∈ (0,1)
    x = (i + 0.5) / n
    # weight proportional to x^{α-1} (1-x)^{α-1}
    w = (x ** (alpha - 1.0)) * ((1.0 - x) ** (alpha - 1.0))
    # numeric safety – replace possible infinities (α very close to 0.5)
    w = np.nan_to_num(w, nan=0.0, posinf=0.0, neginf=0.0)
    total = w.sum()
    if total == 0.0:
        # fallback – uniform
        return np.full(n, 1.0 / n)
    return w / total


# ----------------------------------------------------------------------
# Main search routine (the function that will be called)
# ----------------------------------------------------------------------


def propose_candidate(budget_s: int = 1000) -> list:
    """
    Search for a non‑negative step‑function that minimises
        2·n·max(a*a) / (sum(a)^2)

    The routine respects the given time budget (seconds) and returns
    the best sequence found as a plain Python list of floats.
    """
    # ------------------------------------------------------------------
    # 0. Timing and deterministic seed
    # ------------------------------------------------------------------
    np.random.seed(42)
    deadline = time.time() + budget_s - 5.0   # safety margin

    # ------------------------------------------------------------------
    # 1. Problem size – use the length of a known solution if available,
    #    otherwise fall back to a moderately large size.
    # ------------------------------------------------------------------
    try:
        # `height_sequence_1` may be provided by the evaluation harness
        init = np.asarray(height_sequence_1, dtype=float)  # type: ignore[name-defined]
        n = int(init.shape[0])
    except Exception:
        # a size that balances quality and speed
        n = 20000

    # ------------------------------------------------------------------
    # 2. Very cheap initial guess – uniform distribution
    # ------------------------------------------------------------------
    uniform = np.full(n, 1.0 / n)

    # ------------------------------------------------------------------
    # 3. Grid search over the two‑parameter family (α, λ)
    # ------------------------------------------------------------------
    best_score = np.inf
    best_a = uniform.copy()
    best_alpha = None
    best_lambda = None

    alphas = np.linspace(0.5, 1.20, 15)   # shape exponent
    lambdas = np.linspace(0.0, 1.0, 13)   # weight of the Beta‑like shape
    for alpha in alphas:
        shape = beta_like_shape(alpha, n)
        for lam in lambdas:
            a = lam * shape + (1.0 - lam) * uniform
            a /= a.sum()                     # enforce the simplex exactly
            score = compute_objective(a)
            if score < best_score:
                best_score = score
                best_a = a.copy()
                best_alpha = alpha
                best_lambda = lam
    print(f"Grid search finished – best score {best_score:.6f} "
          f"at alpha={best_alpha:.3f}, lambda={best_lambda:.3f}")

    # ------------------------------------------------------------------
    # 4. Refine with projected Adam (adaptive sub‑gradient method)
    # ------------------------------------------------------------------
    a = best_a.copy()
    m = np.zeros_like(a)          # first moment estimate
    v = np.zeros_like(a)          # second moment estimate
    beta1 = 0.90
    beta2 = 0.999
    eps = 1e-8
    base_lr = 0.8                 # base learning rate (will be scaled)
    it = 0

    while time.time() < deadline:
        it += 1
        grad, _ = compute_gradient(a)

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        step = base_lr * m_hat / (np.sqrt(v_hat) + eps)

        a_candidate = simplex_projection(a - step)

        # occasional random jitter to escape shallow flats
        if it % 4000 == 0:
            jitter = 0.01 * np.random.randn(n)
            a_candidate = simplex_projection(a_candidate + jitter)

        cur_score = compute_objective(a_candidate)
        if cur_score < best_score:
            best_score = cur_score
            best_a = a_candidate.copy()
            print(f"Iter {it:6d}: improved score = {best_score:.6f}")

        # early exit if we have already crossed the target
        if best_score < 1.503:
            print(f"Target reached at iteration {it}")
            break

        a = a_candidate

    # ------------------------------------------------------------------
    # 5. Return the best vector as a Python list
    # ------------------------------------------------------------------
    return [float(v) for v in best_a.tolist()]
