    best_a = a.copy(); best_sc = compute_score(a)
    m = np.zeros_like(a); v = np.zeros_like(a)
    t = 0
    while time.time() < deadline and t < max_iter:
        t+=1
        # compute grad via subgradient
        _, k = max_convolution_and_index(a)
        grad = subgradient_of_max_at_index(a, k)
        # update Adam
        m = beta1*m + (1-beta1)*grad

        v_hat = v / (1 - beta2**t)

        if cur_sc < best_sc:
            best_sc = cur_sc
            best_a = a_new.copy()
        a = a_new

"""
Propose a low‑evaluation step‑function.
================================================

The objective is

    score = 2·n·max(a*a) / (sum(a)**2)

where ``a*a`` is the (zero‑padded) self‑convolution.
Because the score is invariant under scaling, we restrict the search to the
probability simplex (sum(a)=1, a_i≥0).  Minimising the score is therefore
equivalent to minimising the ℓ_∞‑norm of the autocorrelation.

The algorithm combines three ideas:

1. **Broad family search** –  several analytically‑motivated families
   (β‑type, exponential, triangular, uniform, “gap’’) are sampled with
   random parameters.  The best few candidates per length are kept.

2. **Fast projected Adam optimisation** –  Starting from a candidate we
   run a projected Adam optimiser on the nonsmooth objective.
   The sub‑gradient of the current worst convolution entry is used.
   Adam’s adaptive step‑size and momentum reach a better local minimum
   faster than the simple sub‑gradient scheme used before.

3. **Global coordination & polishing** –  the procedure is repeated for a
   handful of lengths (256 … 4096).  The best vector among all lengths is
   finally polished again with a longer Adam run.

All heavy work (FFT‑based convolutions, projection onto the simplex) is
vectorised and written as top‑level functions – no closures or lambdas.
The routine respects the given time budget (default 1000 s) and always
returns a valid non‑negative sequence.
"""

import time
import numpy as np
from typing import List, Tuple

# ----------------------------------------------------------------------
# Low‑level utilities (no closures, all top‑level)
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Return the smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Convolution via real FFT with zero‑padding to a power of two."""
    m = len(a) + len(b) - 1
    n_fft = next_pow2(m)
    A = np.fft.rfft(a, n=n_fft)
    B = np.fft.rfft(b, n=n_fft)
    C = A * B
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


def max_convolution_and_index(a: np.ndarray) -> Tuple[float, int]:
    """Return (max value, arg‑max index) of a*a."""
    conv = convolution_fft(a, a)
    idx = int(np.argmax(conv))
    return float(conv[idx]), idx


def subgradient_of_max_at_index(a: np.ndarray, k: int) -> np.ndarray:
    """
    Sub‑gradient of the maximal convolution entry (index k) w.r.t. a.

    For each i, ∂c_k/∂a_i = a_{k‑i} when the index lies in range.
    If k = 2·i, the diagonal term is doubled.
    """
    n = len(a)
    grad = np.zeros_like(a)
    i = np.arange(n)
    j = k - i
    mask = (j >= 0) & (j < n)
    grad[mask] = a[j[mask]]
    if k % 2 == 0:
        diag = k // 2
        if diag < n:
            grad[diag] = 2.0 * a[diag]
    return grad


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the probability simplex
    {x | x_i ≥ 0, Σ_i x_i = 1}.
    Duchi et al., “Efficient Projections onto the L1‑Ball…”, ICML 2008.
    """
    if np.all(v <= 0):
        # corner case – return the centre of the simplex
        return np.full_like(v, 1.0 / len(v))

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho = np.where(u - cssv / np.arange(1, len(v) + 1) > 0)[0]
    if len(rho) == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def compute_score(a: np.ndarray) -> float:
    """
    Fast proxy for the official evaluator when sum(a)=1.
    Returns 2·n·max(a*a).
    """
    n = len(a)
    conv = convolution_fft(a, a)
    max_val = float(np.max(conv))
    return 2.0 * n * max_val


# ----------------------------------------------------------------------
# Parameterised shape families
# ----------------------------------------------------------------------


def generate_beta_shape(n: int, p: float, eps: float) -> np.ndarray:
    """
    Symmetric β‑type density:
        a_i ∝ ((i+eps)*(n‑1‑i+eps))^{-p}.
    Returns a probability vector (sum=1).
    """
    i = np.arange(n, dtype=float)
    base = ((i + eps) * (n - 1 - i + eps)) ** (-p)
    total = base.sum()
    if total == 0.0:
        return np.full(n, 1.0 / n)
    return base / total


def generate_exponential_shape(n: int, scale: float) -> np.ndarray:
    """
    Exponential “tent’’ shape centred at the middle:
        a_i ∝ exp(-scale·|i - (n‑1)/2|).
    """
    mid = (n - 1) / 2.0
    i = np.arange(n, dtype=float)
    vals = np.exp(-scale * np.abs(i - mid))
    return vals / vals.sum()


def generate_triangular_shape(n: int) -> np.ndarray:
    """Standard symmetric triangular density."""
    i = np.arange(n, dtype=float)
    mid = (n - 1) / 2.0
    vals = 1.0 - np.abs(i - mid) / (mid + 1e-12)
    vals = np.clip(vals, 0.0, None)
    return vals / vals.sum()


def generate_gap_shape(n: int, outer_frac: float = 0.20) -> np.ndarray:
    """
    Piecewise‑constant with low mass in a central gap.
    outer_frac defines the fraction of indices that belong to each outer block.
    """
    outer_len = max(1, int(np.round(outer_frac * n)))
    inner_len = n - 2 * outer_len
    if inner_len < 0:
        inner_len = 0
    outer = np.full(outer_len, 1.0)
    inner = np.full(inner_len, 0.1) if inner_len > 0 else np.array([], dtype=float)
    vals = np.concatenate([outer, inner, outer])
    if len(vals) != n:
        # adjust by truncating / padding
        vals = np.resize(vals, n)
    return vals / vals.sum()


def generate_uniform_shape(n: int) -> np.ndarray:
    """Uniform distribution."""
    return np.full(n, 1.0 / n)


# ----------------------------------------------------------------------
# Optimisation – projected Adam on the ℓ_∞‑norm of the autocorrelation
# ----------------------------------------------------------------------


def adam_optimize(initial_a: np.ndarray,
                  deadline: float,
                  max_iter: int = 200_000,
                  lr: float = 0.2,
                  beta1: float = 0.9,
                  beta2: float = 0.999,
                  eps: float = 1e-8,
                  noise_every: int = 800,
                  noise_scale: float = 5e-5) -> Tuple[np.ndarray, float]:
    """
    Adam optimiser with projection onto the simplex after each update.
    Returns the best vector observed and its *compute_score* value.
    """
    a = project_to_simplex(initial_a.astype(float).copy())
    best_a = a.copy()
    best_sc = compute_score(a)

    m = np.zeros_like(a)
    v = np.zeros_like(a)

    t = 0
    n = len(a)

    while time.time() < deadline and t < max_iter:
        t += 1
        # sub‑gradient of the currently maximal convolution entry
        _, k_max = max_convolution_and_index(a)
        grad = subgradient_of_max_at_index(a, k_max)

        # Adam moment updates
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)

        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)

        a_new = a - lr * m_hat / (np.sqrt(v_hat) + eps)

        if noise_every and (t % noise_every == 0):
            a_new += np.random.randn(n) * noise_scale

        a_new = project_to_simplex(a_new)

        cur_sc = compute_score(a_new)
        if cur_sc < best_sc:
            best_sc = cur_sc
            best_a = a_new.copy()

        a = a_new

        if t % 5000 == 0:
            print(f"    Adam iter={t:,d} cur_score={cur_sc:.6f} best_score={best_sc:.6f}")

    return best_a, best_sc

def propose_candidate(seed: int = 42,

    Search for a non‑negative step sequence with low evaluation score.
    The search explores several lengths, samples many analytic shapes,
    refines promising candidates with projected Adam, and finally
    performs a polishing run on the global best.
    """

    deadline = start_time + budget_s - 2.0

        print(f"Fallback (height_sequence_1) score = {val:.6f}")

            best_seq = fallback
    except Exception:
        pass

    # ------------------------------------------------------------------
    # 1.  Lengths to examine (increase if time permits).
    # ------------------------------------------------------------------
    lengths = [256, 512, 1024, 2048, 4096]

    # How many top candidates per length will be refined?
    TOP_K = 3

    for n in lengths:
        if time.time() >= deadline - 5.0:
            break
        print(f"\n=== Exploring length n = {n} ===")

        # ------------------------------------------------------------------
        # 1a.  Random scan over a family of analytic shapes.
        # ------------------------------------------------------------------
        local_candidates: List[np.ndarray] = []
        local_scores: List[float] = []

        # deterministic starters (always useful)
        starters = [
            generate_uniform_shape(n),
            generate_triangular_shape(n),
            generate_beta_shape(n, p=0.5, eps=0.25),   # classic arcsine
            generate_exponential_shape(n, scale=0.001 * n),
        ]

        for cand in starters:
            sc = compute_score(cand)
            local_candidates.append(cand)
            local_scores.append(sc)

        # random exploration
        max_samples = 800
        for _ in range(max_samples):
            if time.time() >= deadline - 5.0:
                break

            family = rng.choice(["beta", "exp", "gap"])
            if family == "beta":
                # p roughly in (0.05, 2.0); eps > 0
                p = rng.uniform(0.05, 2.0)
                eps = rng.uniform(1e-4, max(0.5, 0.02 * n))
                cand = generate_beta_shape(n, p, eps)
            elif family == "exp":
                # scale controls decay; typical 0.001–0.02 * n
                scale = rng.uniform(0.001, 0.02) * n
                cand = generate_exponential_shape(n, scale)
            else:  # gap
                outer_frac = rng.uniform(0.05, 0.30)
                cand = generate_gap_shape(n, outer_frac=outer_frac)

            sc = compute_score(cand)
            # keep the best few
            if len(local_candidates) < TOP_K:
                local_candidates.append(cand)
                local_scores.append(sc)
            else:
                worst_idx = int(np.argmax(local_scores))
                if sc < local_scores[worst_idx]:
                    local_candidates[worst_idx] = cand
                    local_scores[worst_idx] = sc

        # ------------------------------------------------------------------
        # 1b.  Refine the selected candidates with Adam.
        # ------------------------------------------------------------------
        for idx, cand in enumerate(local_candidates):
            if time.time() >= deadline - 5.0:
                break
            # allocate a time slice for this refinement
            remaining = deadline - time.time()
            # roughly 30 Adam updates per second for n≈2048;
            # we are generous with the upper bound to stay within budget.
            max_iter = min(180_000, int(remaining * 30.0 / (n / 2048.0)))
            if max_iter < 5_000:
                # not enough time left for a meaningful run

                deadline=deadline,
                max_iter=max_iter,
                lr=0.25,
                noise_every=800,
                noise_scale=4e-5,
            )
            # final evaluation with the official (slow) evaluator
            val = evaluate_sequence(refined_a.tolist())
            print(f"      refined score = {val:.6f}")
            if val < best_val:
                best_val = val
                best_seq = refined_a.tolist()
                print(f"      *** New global best: {best_val:.6f} (n={n}) ***")

        # ------------------------------------------------------------------
        # 1c.  If no refinement was possible, still test the raw best shape.
        # ------------------------------------------------------------------
        if not best_seq:
            # fallback to the best raw candidate seen for this length
            best_raw_idx = int(np.argmin(local_scores))
            raw_cand = local_candidates[best_raw_idx]
            val = evaluate_sequence(raw_cand.tolist())

                print(f"  *** New global best (raw) ***")

    # 2.  Final polishing of the current best vector (if time permits).
    # ------------------------------------------------------------------
    if best_seq and time.time() < deadline - 5.0:
        a = np.array(best_seq, dtype=float)
        remaining = deadline - time.time()
        max_iter = min(300_000, int(remaining * 30.0 / (len(a) / 2048.0)))
        print(f"\nFinal polishing (n={len(a)}, max_iter={max_iter}) …")
        a_refined, _ = adam_optimize(
            initial_a=a,
            deadline=deadline,
            max_iter=max_iter,
            lr=0.2,
            noise_every=600,
            noise_scale=3e-5,
        )
        final_val = evaluate_sequence(a_refined.tolist())
        print(f"Final polished score = {final_val:.6f}")
        if final_val < best_val:
            best_val = final_val
            best_seq = a_refined.tolist()
            print("*** Updated global best after final polish ***")

    # ------------------------------------------------------------------
    # 3.  Guard – always return something.
    # ------------------------------------------------------------------
    if not best_seq:
        print("No candidate found – returning a simple uniform sequence.")
        best_seq = [1.0] * 256
        best_val = evaluate_sequence(best_seq)

    print(f"\nFinished search. Best reported score = {best_val:.6f}")
    return best_seq
