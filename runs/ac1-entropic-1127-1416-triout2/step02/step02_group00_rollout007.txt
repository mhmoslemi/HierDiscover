def generate_beta_shape(n, alpha, beta, eps):
    i = np.arange(n, dtype=float)
    base = (i + eps)**(alpha - 1) * (n - 1 - i + eps)**(beta - 1)
    base_sum = base.sum()
    if base_sum == 0:
        return np.full(n, 1.0/n)
    return base / base_sum

def smooth_descent(initial_a, deadline, max_iter=200000, init_lr=0.5):
    a = project_to_simplex(initial_a.copy())
    n = len(a)
    best_a = a.copy()

    lam = 2.0

        conv = convolution_fft(a, a)  # length 2n-1

        if iter % 2000 == 0 and lam < 30:

"""
Search for a non‑negative step function that minimizes the official
``evaluate_sequence`` objective.

Algorithm
---------
1.  Try a few deterministic constructions (uniform, arcsine, etc.) for
    several lengths (512 – 4096).  Keep the best one seen.

2.  Randomly generate many candidates from two flexible families
    *Beta‑shape*  a_i ∝ (i+ε)^{α‑1} (n‑1‑i+ε)^{β‑1}
    *Power‑law shape* a_i ∝ ((i+ε)(n‑1‑i+ε))^{-p}
    with parameters drawn from broad intervals.  All candidates are
    normalised to sum 1; the cheap ``compute_score`` (identical to the
    official evaluator up to clamping) is used to keep the current best.

3.  Refine the globally best vector with a smooth‑max gradient descent.
    Instead of the non‑smooth max we minimise
        φ_λ(a) = (1/λ)·log Σ_k exp(λ·(a*a)[k])
    which converges to the true max as λ→∞.
    The gradient of φ_λ is obtained analytically and together with a
    Euclidean projection onto the simplex (Duchi et al.) yields an
    efficient projected‑gradient method.  λ is increased slowly during
    optimisation.

4.  As a final polish, a few hundred thousand steps of the classic
    sub‑gradient optimiser (the “improve_sequence” routine from the
    previous solution) are applied to the refined vector.

5.  The best vector found is returned as a plain list of floats.
"""

import time
import math
from typing import List, Tuple

import numpy as np

# ----------------------------------------------------------------------
# Low‑level utilities (top‑level, no closures)
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """
    Convolution via real FFT (fast, zero‑padding to next power of two).
    Returns a real ndarray of length len(a)+len(b)-1.
    """
    m = len(a) + len(b) - 1
    n_fft = next_pow2(m)
    A = np.fft.rfft(a, n=n_fft)
    B = np.fft.rfft(b, n=n_fft)
    C = A * B
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


def compute_score(a: np.ndarray) -> float:
    """
    Scale‑invariant objective used by the official evaluator:
        2·n·max(a*a) / (sum(a)²)
    (the input is assumed to be a probability vector, i.e. sum≈1).
    """
    n = len(a)
    conv = convolution_fft(a, a)
    max_val = float(np.max(conv))
    s = float(np.sum(a))
    if s < 1e-12:
        return float("inf")
    return 2.0 * n * max_val / (s * s)


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the probability simplex
    {x | x_i ≥ 0, Σ_i x_i = 1}.
    Implementation of Duchi et al. (2008).
    """
    if np.all(v <= 0):
        # equally split the mass
        return np.full_like(v, 1.0 / len(v))

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho = np.where(u - cssv / np.arange(1, len(v) + 1) > 0)[0]
    if len(rho) == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def arcsine_initial(n: int) -> np.ndarray:
    """
    Classic arcsine‑shaped start:
        a_i ∝ 1 / sqrt((i+½)*(n‑½‑i)).
    """
    i = np.arange(n, dtype=float)
    eps = 0.25
    vals = 1.0 / np.sqrt((i + 0.5) * (n - 0.5 - i) + eps)
    return vals / np.sum(vals)


def generate_shape(n: int, p: float, eps: float) -> np.ndarray:
    """
    Symmetric power‑law family:
        a_i ∝ ((i+eps)*(n‑1‑i+eps))^{-p}.
    Normalised to unit L¹ norm.
    """
    i = np.arange(n, dtype=float)
    base = ((i + eps) * (n - 1 - i + eps)) ** (-p)
    total = base.sum()
    if total == 0.0:
        return np.full(n, 1.0 / n)
    return base / total


def generate_beta_shape(n: int, alpha: float, beta: float, eps: float) -> np.ndarray:
    """
    Beta‑type family:
        a_i ∝ (i+eps)^{α‑1}·(n‑1‑i+eps)^{β‑1}.
    """
    i = np.arange(n, dtype=float)
    base = (i + eps) ** (alpha - 1.0) * (n - 1 - i + eps) ** (beta - 1.0)
    total = base.sum()
    if total == 0.0:
        return np.full(n, 1.0 / n)
    return base / total


def generate_exp_decay(n: int, lam: float) -> np.ndarray:
    """
    Symmetric exponential decay centred in the middle:
        a_i ∝ exp( -lam·|i - (n‑1)/2| ).
    """
    i = np.arange(n, dtype=float)
    mid = (n - 1) / 2.0
    base = np.exp(-lam * np.abs(i - mid))
    total = base.sum()
    if total == 0.0:
        return np.full(n, 1.0 / n)
    return base / total

def improve_sequence(
    initial_a: np.ndarray,
    time_deadline: float,
    max_iter: int = 200_000,
    eta0: float = 0.5,
    noise_every: int = 500,
    noise_scale: float = 1e-4,
) -> Tuple[np.ndarray, float]:
    """
    Classic projected sub‑gradient descent on the original max‑convolution
    objective.  Returns the best vector seen and its (fast) score.
    """
    a = project_to_simplex(initial_a.astype(float).copy())
    best_a = a.copy()
    best_sc = compute_score(a)

    n = len(a)
    it = 0
    while time.time() < time_deadline and it < max_iter:
        it += 1
        # worst convolution index
        conv = convolution_fft(a, a)
        k_max = int(np.argmax(conv))

        # sub‑gradient of the maximal entry (see comment in the original code)
        i = np.arange(n)
        j = k_max - i
        mask = (j >= 0) & (j < n)
        grad = np.zeros_like(a)
        grad[mask] = a[j[mask]]
        if k_max % 2 == 0:
            diag = k_max // 2
            if diag < n:
                grad[diag] = 2.0 * a[diag]

        # diminishing step size
        eta = eta0 / np.sqrt(it)
        a_new = a - eta * grad
        if noise_every and (it % noise_every == 0):
            a_new += np.random.randn(n) * noise_scale
        a_new = project_to_simplex(a_new)

        cur_sc = compute_score(a_new)
        if cur_sc < best_sc:
            best_sc = cur_sc
            best_a = a_new.copy()
        a = a_new
    return best_a, best_sc


def smooth_descent(
    initial_a: np.ndarray,
    time_deadline: float,
    max_iter: int = 150_000,
    init_lr: float = 0.6,
    lam_start: float = 2.0,
    lam_factor: float = 1.05,
    lam_max: float = 30.0,
) -> np.ndarray:
    """
    Projected gradient descent on the smooth approximation
        φ_λ(a) = (1/λ)·log Σ_k exp(λ·(a*a)[k]).

    λ is increased slowly from ``lam_start`` up to ``lam_max``.
    The routine returns the best vector encountered (according to the
    true max‑convolution score).
    """
    a = project_to_simplex(initial_a.astype(float).copy())
    best_a = a.copy()
    best_sc = compute_score(a)

    n = len(a)
    len_w = 2 * n - 1
    it = 0
    lam = lam_start

    while time.time() < time_deadline and it < max_iter:
        it += 1

        # self‑convolution
        conv = convolution_fft(a, a)                # length 2n‑1
        max_c = np.max(conv)

        # soft‑max weights (stable version)
        exp_terms = np.exp(lam * (conv - max_c))
        w = exp_terms / np.sum(exp_terms)           # sum to 1

        # gradient = 2·(w_rev * a)[len_w‑1‑i]  (see analysis)
        w_rev = w[::-1]
        grad_full = convolution_fft(w_rev, a)       # length 3n‑2
        grad = 2.0 * grad_full[len_w - 1 - np.arange(n)]

        # step size schedule
        lr = init_lr / np.sqrt(it)

        a_new = a - lr * grad
        a_new = project_to_simplex(a_new)

        cur_sc = compute_score(a_new)
        if cur_sc < best_sc:
            best_sc = cur_sc
            best_a = a_new.copy()

        a = a_new

        # increase λ every few thousand iterations
        if it % 2000 == 0 and lam < lam_max:
            lam *= lam_factor

    return best_a

def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs) -> List[float]:
    """
    Search for a step‑function (non‑negative sequence) with a low
    ``evaluate_sequence`` value.  The routine respects the 1000 s budget,
    explores several sequence lengths, performs a wide random scan of
    flexible families and refines the best candidate with smooth‑max
    gradient descent followed by a short sub‑gradient polish.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    # leave a couple of seconds spare for the final return
    deadline = start_time + budget_s - 2.0

    best_seq: List[float] = []
    best_val: float = float("inf")

    # ------------------------------------------------------------------
    # 0.  Fallback to any previously discovered construction.
    # ------------------------------------------------------------------
    try:
        fallback = [float(x) for x in height_sequence_1]  # type: ignore
        val = evaluate_sequence(fallback)
        print(f"Fallback (height_sequence_1) score = {val:.6f}")
        if val < best_val:
            best_val = val
            best_seq = fallback
    except Exception:
        pass

    # ------------------------------------------------------------------
    # 1.  Deterministic starters for a few lengths.
    # ------------------------------------------------------------------
    lengths = [512, 1024, 2048, 4096]
    for n in lengths:
        if time.time() >= deadline - 5.0:
            break
        print(f"\n=== Deterministic start, n = {n} ===")
        starters = [
            np.full(n, 1.0 / n),      # uniform
            arcsine_initial(n),        # classic arcsine
        ]
        for a in starters:
            sc = compute_score(a)
            val = evaluate_sequence(a.tolist())
            print(f"  deterministic candidate score = {val:.6f}")
            if val < best_val:
                best_val, best_seq = val, a.tolist()
                print(f"  *** New global best (deterministic) ***")

    # ------------------------------------------------------------------
    # 2.  Randomised search over flexible families.
    # ------------------------------------------------------------------
    print("\n=== Randomised exploration ===")
    while time.time() < deadline - 5.0:
        # pick a length (larger lengths give a better theoretical lower bound)
        n = int(rng.choice([1024, 2048, 4096]))
        family = rng.choice(["beta", "power", "exp"])
        if family == "beta":
            alpha = rng.uniform(0.1, 0.9)
            beta = rng.uniform(0.1, 0.9)
            eps = rng.uniform(1e-4, 0.5)
            a = generate_beta_shape(n, alpha, beta, eps)
        elif family == "power":
            p = rng.uniform(0.2, 0.9)
            eps = rng.uniform(1e-3, max(2.0, 0.02 * n))
            a = generate_shape(n, p, eps)
        else:  # exponential decay
            lam = rng.uniform(1e-4, 0.03)
            a = generate_exp_decay(n, lam)

        sc = compute_score(a)
        if sc < best_val:
            # evaluate with the official (slower) evaluator once
            val = evaluate_sequence(a.tolist())
            print(f"  new candidate (family={family}) score = {val:.6f}")
            if val < best_val:
                best_val, best_seq = val, a.tolist()
                print(f"  *** New global best from random search ***")

    # ------------------------------------------------------------------
    # 3.  Refinement of the current global best.
    # ------------------------------------------------------------------
    if best_seq and time.time() < deadline - 5.0:
        a0 = np.array(best_seq, dtype=float)
        print("\n=== Smooth‑max gradient descent refinement ===")
        a_refined = smooth_descent(
            initial_a=a0,
            time_deadline=deadline,
            max_iter=200_000,
            init_lr=0.7,
            lam_start=2.0,
            lam_factor=1.04,
            lam_max=25.0,
        )
        val = evaluate_sequence(a_refined.tolist())
        print(f"  after smooth descent score = {val:.6f}")
        if val < best_val:
            best_val, best_seq = val, a_refined.tolist()
            print("*** Updated global best after smooth descent ***")

        # final brief sub‑gradient polishing
        if time.time() < deadline - 2.0:
            print("\n=== Final sub‑gradient polish ===")
            remaining = deadline - time.time()
            max_iter = min(150_000, int(remaining * 30))
            polished_a, _ = improve_sequence(
                initial_a=np.array(best_seq, dtype=float),
                time_deadline=deadline,
                max_iter=max_iter,
                eta0=0.6,
                noise_every=600,
                noise_scale=3e-5,
            )
            val = evaluate_sequence(polished_a.tolist())
            print(f"  after polishing score = {val:.6f}")
            if val < best_val:
                best_val, best_seq = val, polished_a.tolist()
                print("*** Updated global best after polishing ***")

    # ------------------------------------------------------------------
    # 4.  Fallback guard – ensure a return value.
    # ------------------------------------------------------------------
    if not best_seq:
        best_seq = [1.0] * 256
        best_val = evaluate_sequence(best_seq)

    print(f"\nFinished search. Best reported score = {best_val:.6f}")
    return best_seq
