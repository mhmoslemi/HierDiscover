```
def subgradient(a):
    n = a.shape[0]
    # compute convolution via fft for speed
    fft = np.fft.rfft(a, n=2*n-1)
    conv = np.fft.irfft(fft*fft, n=2*n-1)
    max_val = conv.max()
    tol = max_val * 1e-12 + 1e-14
    active = np.where(conv >= max_val - tol)[0]
    grad = np.zeros_like(a)
    for k in active:
         low = max(0, k - n + 1)
         high = min(n-1, k)
         idx = np.arange(low, high+1)
         partner = k - idx
         grad[idx] += 2.0 * a[partner]
    if active.size > 1:
         grad /= active.size
    return grad, max_val
```

def compute_gradient_and_obj(a):
    n = a.shape[0]
    fft = np.fft.rfft(a, n=2*n-1)
    conv = np.fft.irfft(fft*fft, n=2*n-1)
    max_val = conv.max()
    obj = 2.0 * n * max_val
    # tolerance
    tol = max_val * 1e-12 + 1e-14
    active = np.where(conv >= max_val - tol)[0]
    grad = np.zeros_like(a)
    for k in active:
        low = max(0, k - n + 1)
        high = min(n-1, k)
        idx = np.arange(low, high+1, dtype=int)
        partner = k - idx
        grad[idx] += 2.0 * a[partner]
    if active.size > 1:
        grad /= active.size
    return grad, obj
```

Now `adam_update`:

```
def adam_update(a, grad, m, v, t, alpha):
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    m = beta1*m + (1-beta1)*grad
    v = beta2*v + (1-beta2)*(grad*grad)
    m_hat = m / (1 - beta1**t)
    v_hat = v / (1 - beta2**t)
    step = alpha * m_hat / (np.sqrt(v_hat) + eps)
    a_new = simplex_projection(a - step)
    return a_new, m, v
```

"""
Propose a step‑function (non‑negative vector) that minimizes

    evaluate_sequence(seq) = 2·n·max(conv(seq,seq)) / (sum(seq)**2)

The objective is homogeneous, therefore we optimise on the simplex
Σa_i = 1 (a_i ≥ 0).  The search combines

* projected Adam (adaptive step size, momentum) on the sub‑gradient of
  the maximum convolution entry,
* a mirror‑descent style exponentiated‑gradient (EG) fine‑tuner,
* frequent random restarts (including perturbations of the current best)
  to escape plateaus.

All work is bounded by ``budget_s`` seconds; the best vector found is
returned as a plain list of floats.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper: Euclidean projection onto the probability simplex
# ----------------------------------------------------------------------
def simplex_projection(v: np.ndarray) -> np.ndarray:
    """Project a vector onto { w | w_i ≥ 0 , Σ w_i = 1 } (Duchi et al. 2008)."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1] - 1
    theta = cssv[cond][-1] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Gradient of the max‑convolution entry and the objective value
# ----------------------------------------------------------------------
def grad_and_obj(a: np.ndarray):
    """
    Return (sub‑gradient, objective) for a vector a on the simplex.
    Objective = 2·n·max(conv(a,a)).
    """
    n = a.shape[0]
    # convolution via FFT (zero‑padding to length 2n‑1)
    fft = np.fft.rfft(a, n=2 * n - 1)
    conv = np.fft.irfft(fft * fft, n=2 * n - 1)
    max_val = conv.max()
    obj = 2.0 * n * max_val

    # tolerance to decide which entries are (numerically) maximal
    tol = max_val * 1e-12 + 1e-14
    active = np.where(conv >= max_val - tol)[0]

    grad = np.zeros_like(a)
    for k in active:
        low = max(0, k - n + 1)
        high = min(n - 1, k)
        idx = np.arange(low, high + 1, dtype=int)
        partner = k - idx               # partner = k-i
        grad[idx] += 2.0 * a[partner]   # sub‑gradient contribution

    if active.size > 1:
        grad /= active.size
    return grad, obj


# ----------------------------------------------------------------------
# Adam optimizer step with simplex projection
# ----------------------------------------------------------------------
def adam_step(a, grad, m, v, t, lr=0.2):
    """
    Perform one Adam update (projected onto the simplex).
    Returns the new point and the updated first/second moment vectors.
    """
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8

    m = beta1 * m + (1.0 - beta1) * grad
    v = beta2 * v + (1.0 - beta2) * (grad * grad)

    m_hat = m / (1.0 - beta1 ** t)
    v_hat = v / (1.0 - beta2 ** t)

    step = lr * m_hat / (np.sqrt(v_hat) + eps)
    a_new = simplex_projection(a - step)
    return a_new, m, v


# ----------------------------------------------------------------------
# Exponentiated‑gradient update (mirror descent on the simplex)
# ----------------------------------------------------------------------
def eg_step(a, grad, eta):
    """
    Mirror‑descent update using KL‑divergence (multiplicative weights).
    """
    log_a = np.log(a + 1e-30)          # keep positivity
    log_a -= eta * grad
    a_new = np.exp(log_a)
    # re‑normalise (tiny negatives are impossible after exp)
    a_new /= a_new.sum()
    return a_new

        a0 = np.maximum(init, 0.0)
        s = a0.sum()
        a0 = a0 / s if s > 0 else np.full(n, 1.0 / n)
    else:
        a0 = np.full(n, 1.0 / n)

    # --------------------------------------------------------------
    # 2. Global bookkeeping
    # --------------------------------------------------------------
    best_a = a0.copy()
    best_score = grad_and_obj(best_a)[1]
    total_iters = 0
    restart = 0

    # --------------------------------------------------------------
    # 3. Main optimisation loop (restarts + Adam + EG)
    # --------------------------------------------------------------
    while time.time() < deadline:
        # ---- 3.1 Choose a start point for this restart ----------
        if restart == 0:
            a = best_a.copy()                 # first run starts from the known best
        else:
            # blend the current best with a fresh random Dirichlet vector
            if np.random.rand() < 0.4:
                alpha = np.random.choice([0.2, 0.5, 1.0, 2.0])
                rand_vec = np.random.dirichlet(np.ones(n) * alpha)
                mix = np.random.rand() * 0.25 + 0.05   # 5 % – 30 % random part
                a = (1.0 - mix) * best_a + mix * rand_vec
            else:
                alpha = np.random.choice([0.2, 0.5, 1.0, 2.0])
                a = np.random.dirichlet(np.ones(n) * alpha)

        a = np.clip(a, 0.0, None)
        a /= a.sum()                     # ensure feasibility

        # ---- 3.2 Adam stage ------------------------------------
        m = np.zeros_like(a)
        v = np.zeros_like(a)
        t = 0

        # allocate a generous chunk for the Adam phase (up to ~250 s)
        stage_deadline = time.time() + min(250.0, deadline - time.time() - 5.0)
        while time.time() < stage_deadline and time.time() < deadline:
            t += 1
            grad, cur_obj = grad_and_obj(a)

            # keep the best ever seen
            if cur_obj < best_score:
                best_score = cur_obj
                best_a = a.copy()
                print(f"Iter {total_iters:7d} restart {restart} – new best = {best_score:.6f}")
                if best_score < 1.503:
                    return best_a.tolist()

            # Adam update (projected)
            a, m, v = adam_step(a, grad, m, v, t, lr=0.2)

            # occasional stochastic jitter to inject fresh directions
            if t % 8000 == 0:
                a = simplex_projection(a + 0.001 * np.random.randn(n))

            total_iters += 1

        # ---- 3.3 Exponentiated‑gradient fine‑tuning -------------
        eg_deadline = time.time() + min(60.0, deadline - time.time() - 5.0)
        eg_iter = 0
        eta0 = 0.20
        while time.time() < eg_deadline and time.time() < deadline:
            eg_iter += 1
            grad, cur_obj = grad_and_obj(a)

            if cur_obj < best_score:
                best_score = cur_obj
                best_a = a.copy()
                print(f"Iter {total_iters:7d} restart {restart} (EG) – new best = {best_score:.6f}")
                if best_score < 1.503:
                    return best_a.tolist()

            # decreasing learning rate for EG (1/√t schedule)
            eta = eta0 / np.sqrt(eg_iter)
            a = eg_step(a, grad, eta)

            # tiny additive noise every few thousand steps
            if eg_iter % 5000 == 0:
                a += 0.001 * np.random.randn(n)
                a = np.clip(a, 0.0, None)
                a /= a.sum()

            total_iters += 1

        restart += 1   # next restart will use a fresh random blend

    # --------------------------------------------------------------
    # 4. Return the best sequence found (fallback to uniform if needed)
    # --------------------------------------------------------------
    if best_a is None:
        best_a = np.full(n, 1.0 / n)
    # convert to a plain Python list of floats
    return [float(v) for v in best_a.tolist()]
