"""
Propose a step‑function with a low evaluation value.

Algorithm
---------
1. Initialise a non‑negative sequence of length *n* (default 8000) and
   normalise it to Σ a = √(2·n).  If a previous construction
   ``height_sequence_1`` is available it is up‑sampled, otherwise a short
   random power‑law sweep is performed.

2. Smooth‑max optimisation.
   The objective   max(conv(a,a))   is replaced by a smooth
   approximation
        Sγ(a) = (1/γ)·log Σ_k exp(γ·(a∗a)[k]) .
   Its gradient can be written in closed form using a correlation:
        ∇Sγ = 2·corr(a , p)          where  p_k = softmax(γ·conv)_k .
   A projected Adam step (followed by Euclidean projection onto the
   simplex {x≥0, Σx = √(2·n)}) is performed.  The smoothing parameter γ is
   increased gradually (continuation) to focus the search onto the true
   maximum.

3. Fine‑grained sub‑gradient phase.
   After the smooth phase the current best sequence is refined with a
   classic projected sub‑gradient descent that moves only in the
   directions of the (near‑)maximal convolution entries.

The whole routine respects the 1000 s budget and returns the best
sequence found (as a plain list of floats).  The evaluation of any
candidate can be obtained with ``evaluate_sequence`` which, for the
normalised sum, equals the maximal self‑convolution.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities
# ----------------------------------------------------------------------


def _project_simplex(v: np.ndarray, target_sum: float) -> np.ndarray:
    """
    Euclidean projection of a vector onto the simplex
        { w | w_i ≥ 0, Σ w_i = target_sum }.
    Implements the algorithm of Duchi et al. (2008).
    """
    if target_sum < 0:
        raise ValueError("target_sum must be non‑negative")
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - target_sum
    ind = np.arange(1, len(v) + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.zeros_like(v)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def _full_convolution_fft(a: np.ndarray, size: int) -> np.ndarray:
    """
    Linear (full) convolution a*a using a real FFT.
    Returns an array of length 2·len(a)-1.
    The caller must provide a power‑of‑two size ≥ 2·len(a)-1.
    """
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * len(a) - 1]
    return conv


def _full_correlation_fft(a: np.ndarray, w: np.ndarray, size: int) -> np.ndarray:
    """
    Linear correlation of a with w (both real) using FFT.
    Returns the full correlation array of length len(a)+len(w)-1.
    """
    fa = np.fft.rfft(a, n=size)
    fw = np.fft.rfft(w, n=size)
    corr = np.fft.irfft(fa * np.conj(fw), n=size)
    return corr


def _max_conv(a: np.ndarray, size: int) -> float:
    """Return max(conv(a,a)) via FFT."""
    conv = _full_convolution_fft(a, size)
    return float(conv.max())


def _make_candidate(n: int, p: float, offset: float, beta: float) -> np.ndarray:
    """
    Build a raw power‑law / linear‑tilt sequence
        a_i = (i+1+offset)^{-p}·(1+beta·i/(n-1))
    and project it onto the simplex Σ a = √(2·n).
    """
    idx = np.arange(n, dtype=float)
    base = np.power(idx + 1.0 + offset, -p)
    factor = 1.0 + beta * idx / (n - 1)
    raw = base * factor
    raw = np.maximum(raw, 0.0)
    target = math.sqrt(2.0 * n)
    return _project_simplex(raw, target)


def _upsample_sequence(old: np.ndarray, new_n: int) -> np.ndarray:
    """
    Linear interpolation of a 1‑D array to length new_n.
    """
    old_n = len(old)
    if old_n == new_n:
        return old.copy()
    x_old = np.arange(old_n, dtype=float)
    x_new = np.linspace(0, old_n - 1, new_n)
    new = np.interp(x_new, x_old, old)
    return new

        a = _project_simplex(a, target_sum)
        print("[warm] used up‑sampled previous sequence")
    else:
        # cheap random sweep in the 4‑parameter family
        best_local = None
        best_val = 1e9
        for _ in range(12):
            p = np.random.uniform(0.35, 0.65)
            off = np.random.uniform(0.0, 6.0)
            beta = np.random.uniform(-0.8, 2.5)
            cand = _make_candidate(n, p, off, beta)
            # a quick evaluation (max convolution) – size for FFT:
            size_fft = 1 << ((2 * n - 1).bit_length())
            val = _max_conv(cand, size_fft)
            if val < best_val:
                best_val = val
                best_local = cand
        a = best_local
        print(f"[warm] random sweep best {best_val:.6f}")

    # ------------------------------------------------------------------
    # 2. Smooth‑max Adam optimisation (continuation on γ).
    # ------------------------------------------------------------------
    size_fft = 1 << ((3 * n - 2).bit_length())   # enough for correlation
    # Adam hyper‑parameters
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    base_lr = 0.12
    gamma = 5.0          # start with a smooth approximation
    gamma_max = 80.0
    best_score = _max_conv(a, size_fft)          # actual objective (max conv)
    best_seq = a.copy()
    it = 0

    print(f"[adam] start – score {best_score:.6f}, γ={gamma}")

    while time.time() < deadline:
        it += 1

        # ----- linear convolution (needed for smooth objective) -----
        Fa = np.fft.rfft(a, n=size_fft)
        conv = np.fft.irfft(Fa * Fa, n=size_fft)[: 2 * n - 1]

        # ----- smooth max (log‑sum‑exp) and its softmax weights -----
        maxc = conv.max()
        # stable softmax
        exp_shift = np.exp(gamma * (conv - maxc))
        sum_exp = exp_shift.sum()
        smooth = maxc + (np.log(sum_exp) / gamma)   # Sγ(a)
        p = exp_shift / sum_exp                     # softmax probabilities

        # ----- gradient via correlation (g = 2·corr(a, p)) -----
        Fp = np.fft.rfft(p, n=size_fft)
        corr_full = np.fft.irfft(Fa * np.conj(Fp), n=size_fft)
        # slice the part that corresponds to shifts 0 … n‑1
        start = (2 * n - 1) - 1          # = len(p)‑1
        grad = 2.0 * corr_full[start:start + n]

        # ----- Adam update with projection onto the simplex -----
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        step = base_lr * m_hat / (np.sqrt(v_hat) + eps)
        a = a - step
        a = _project_simplex(a, target_sum)

        # ----- occasional evaluation of the true objective -----
        if it % 7 == 0:
            # re‑compute max‑convolution for the new a
            Fa_new = np.fft.rfft(a, n=size_fft)
            conv_new = np.fft.irfft(Fa_new * Fa_new, n=size_fft)[: 2 * n - 1]
            cur_score = float(conv_new.max())
            if cur_score < best_score:
                best_score = cur_score
                best_seq = a.copy()
                print(f"[adam] iter {it:6d}  new best {best_score:.6f}")
                if best_score < 1.503:
                    break

        # ----- continuation on γ -----
        if it % 200 == 0 and gamma < gamma_max:
            gamma = min(gamma * 1.06, gamma_max)

        # ----- occasional random jitter to escape plateaus -----
        if it % 500 == 0 and np.random.rand() < 0.3:
            jitter = (np.random.rand(n) - 0.5) * 0.02 * a
            a = _project_simplex(a + jitter, target_sum)

    # ------------------------------------------------------------------
    # 3. Final projected sub‑gradient refinement (focus on the true max).
    # ------------------------------------------------------------------
    curr = best_seq.copy()
    step0 = 1.0
    iter_sgd = 0
    while time.time() < deadline:
        iter_sgd += 1
        conv_full = _full_convolution_fft(curr, size_fft)
        M = conv_full.max()
        if M < best_score:
            best_score = M
            best_seq = curr.copy()
            print(f"[sgd]  iter {iter_sgd:6d}  new best {best_score:.6f}")
            if best_score < 1.503:
                break

        eps_max = 1e-12 * M if M != 0 else 1e-12
        max_inds = np.where(conv_full >= M - eps_max)[0]

        step = step0 / np.sqrt(iter_sgd)
        # sub‑gradient for the set of maximal indices
        g = np.zeros_like(curr)
        n_len = n
        for k in max_inds:
            if k < n_len:
                # i = 0 … k
                g[: k + 1] += curr[k::-1]
            else:
                # i = k-(n-1) … n-1
                start = k - (n_len - 1)
                idx = np.arange(start, n_len)
                g[start:] += curr[k - idx]

        cand = curr - step * g
        cand = _project_simplex(cand, target_sum)

        # tiny random perturbation every few hundred steps
        if iter_sgd % 100 == 0 and np.random.rand() < 0.25:
            jitter = (np.random.rand(n) - 0.5) * 0.01 * cand
            cand = _project_simplex(cand + jitter, target_sum)

        curr = cand

    print(f"Finished – best score {best_score:.6f} after {it} Adam + {iter_sgd} SG steps")
    return best_seq.tolist()
