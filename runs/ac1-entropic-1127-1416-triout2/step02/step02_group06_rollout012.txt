"""Search for a step function minimizing max self‑convolution.

The algorithm:
1. Build a set of simple analytic seeds (geometric, power‑law,
   uniform, random Dirichlet) and keep the best after a fast FFT
   score.
2. From the best seed run a long projected sub‑gradient descent
   on the simplex  Σa_i = √(2n).  The gradient is the sum of the
   active convolution constraints (indices whose value is close to
   the current maximum).  Momentum and Nesterov‑type updates are
   used, together with occasional multiplicative‑weight/E​G steps
   and random jitter to escape plateaus.
3. If the search stagnates a fresh Dirichlet sample is drawn.
4. At the end the best sequence is verified with the official
   ``evaluate_sequence`` and returned.
"""
import time
import numpy as np

# --------------------------------------------------------------------------- #
def projection_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Project vector ``v`` onto the simplex {x ≥ 0, Σx = s}."""
    if s < 0:
        raise ValueError("Simplex sum must be non‑negative")
    n = v.shape[0]
    # sort descending
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho_arr = np.where(u - cssv / (np.arange(n, dtype=np.float64) + 1) > 0)[0]
    if rho_arr.size == 0:
        return np.zeros_like(v)
    rho = rho_arr[-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w

# --------------------------------------------------------------------------- #
def _fast_score(seq: np.ndarray, fft_len: int) -> float:
    """Score = max(conv(seq, seq)) (the n‑dependent factor cancels)."""
    F = np.fft.rfft(seq, n=fft_len)
    conv = np.fft.irfft(F * F, n=fft_len)[: 2 * seq.size - 1]
    return float(conv.max())

# --------------------------------------------------------------------------- #
def propose_candidate(seed: int = 0, budget_s: float = 1000) -> list[float]:
    """
    Search for a non‑negative step‑function (list of floats) that
    minimises ``evaluate_sequence``.  The function respects the given
    time budget and returns the best sequence found.
    """
    np.random.seed(seed)
    start = time.time()
    deadline = start + budget_s - 0.2  # safety margin for the return

    # ------------------------------------------------------------------- #
    # Problem size – a few thousand entries give a good trade‑off.
    n = 6000
    target_sum = np.sqrt(2.0 * n)          # we keep the ℓ₁‑norm fixed
    fft_len = 1 << ((2 * n - 1).bit_length())

    # ------------------------------------------------------------------- #
    # 1. Build a coarse set of seeds.
    seeds = []

    # uniform seed
    seeds.append(np.full(n, target_sum / n))

    # geometric families (parameter λ = 1‑r)
    lam_vals = np.logspace(-5, -0.1, 30)
    for lam in lam_vals:
        r = 1.0 - lam
        if r <= 0 or r >= 1:
            continue
        # sum_{i=0}^{n‑1} r^i = (1‑rⁿ)/(1‑r) = (1‑rⁿ)/lam
        denom = 1.0 - r ** n
        if denom <= 0.0:
            continue
        C = target_sum * lam / denom
        a = C * (r ** np.arange(n, dtype=np.float64))
        seeds.append(a)

    # power‑law families  a_i ∝ (i+1)⁻ᵖ
    for p in np.linspace(0.2, 2.0, 30):
        base = 1.0 / np.power(np.arange(1, n + 1, dtype=np.float64), p)
        C = target_sum / base.sum()
        a = C * base
        seeds.append(a)

    # random Dirichlet seeds – heavy‑tailed and near‑uniform
    alphas = [0.05, 0.2, 0.5, 1.0, 2.0]
    for alpha in alphas:
        for _ in range(4):
            vec = np.random.dirichlet(np.full(n, alpha))
            a = vec * target_sum
            seeds.append(a)

    # ------------------------------------------------------------------- #
    # 2. Scan the seeds, keep the best according to the fast score.
    best_seq = None
    best_score = float("inf")
    for idx, cand in enumerate(seeds):
        if time.time() >= deadline:
            break
        cand = projection_simplex(cand, target_sum)
        sc = _fast_score(cand, fft_len)
        if sc < best_score:
            best_score = sc
            best_seq = cand.copy()
            print(f"[{time.time() - start:6.1f}s] seed {idx:03d} → score {sc:.6f}")

    if best_seq is None:
        # fallback – uniform vector
        best_seq = np.full(n, target_sum / n)
        best_score = _fast_score(best_seq, fft_len)

    # ------------------------------------------------------------------- #
    # 3. Main optimisation loop – projected sub‑gradient descent with
    #    momentum, occasional exponentiated‑gradient (EG) steps and
    #    random restarts when we stall.
    curr = best_seq.copy()
    best = curr.copy()
    best_score = _fast_score(best, fft_len)

    velocity = np.zeros_like(curr)
    step = 0.02 * target_sum / n          # initial (absolute) step size
    momentum = 0.85
    rel_tol = 1e-4                       # active‑set tolerance
    it = 0
    last_improve = 0

    while time.time() < deadline:
        # ---- convolution (FFT) ------------------------------------------------
        F = np.fft.rfft(curr, n=fft_len)
        conv = np.fft.irfft(F * F, n=fft_len)[: 2 * n - 1]
        max_val = conv.max()
        score = float(max_val)          # because denominator = target_sum² = 2n

        # ---- record improvement ------------------------------------------------
        if score < best_score - 1e-12:
            best_score = score
            best = curr.copy()
            last_improve = it
            print(f"[{time.time() - start:6.1f}s] iter {it:07d} → new best {score:.6f}")
            # shrink step a bit after progress
            step = max(step * 0.95, 1e-9)

        tol_abs = max_val * rel_tol

        # ---- sub‑gradient of the max‑functional ---------------------------------

        # whenever the index k‑i lies inside the vector.

            i_end = min(k, n - 1)
            idx_slice = slice(i_start, i_end + 1)
            j_vals = k - np.arange(i_start, i_end + 1, dtype=np.int64)
            grad[idx_slice] += 2.0 * curr[j_vals]

        gnorm = np.linalg.norm(grad)
        if gnorm == 0.0:
            break

        # ---- Nesterov‑type momentum step --------------------------------------
        velocity = momentum * velocity - step * (grad / gnorm)
        tentative = curr + velocity
        tentative = np.clip(tentative, 0.0, 1000.0)
        curr = projection_simplex(tentative, target_sum)

        # ---- occasional random jitter to escape plateaus -----------------------
        if np.random.rand() < 0.01:
            jitter = (np.random.randn(n) *
                      step * target_sum / (20.0 * n))
            curr = projection_simplex(
                np.clip(curr + jitter, 0.0, 1000.0), target_sum)

        # ---- every few thousand iterations perform an exponentiated‑gradient step --
        if it % 2500 == 0 and it > 0:
            p = curr / target_sum
            max_abs = np.max(np.abs(grad)) + 1e-12
            eta = 0.5 / max_abs
            p = p * np.exp(-eta * grad)
            p = p / p.sum()
            curr = p * target_sum

        # ---- stagnation detection – random Dirichlet restart --------------------
        if it - last_improve > 8000:
            print(f"[{time.time() - start:6.1f}s] stagnation, restarting")
            vec = np.random.dirichlet(np.full(n, 0.5))
            curr = vec * target_sum
            last_improve = it
            # reset momentum
            velocity.fill(0.0)

        # ---- slowly tighten the active‑set tolerance ----------------------------
        if it % 5000 == 0 and it > 0:
            rel_tol = max(rel_tol * 0.9, 1e-6)

        it += 1

    # ------------------------------------------------------------------- #
    # 4. Final sanity check with the official evaluator.
    final_score = evaluate_sequence(best.tolist())
    print(f"Search finished. Final score reported by evaluate_sequence: {final_score:.6f}")

    return best.tolist()
