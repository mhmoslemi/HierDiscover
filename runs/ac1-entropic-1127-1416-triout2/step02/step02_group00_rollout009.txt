def lp_optimize_g(f, M, solver='ECOS', time_limit=5):
    n = len(f)
    A = build_conv_matrix_coeffs(f)
    g = cp.Variable(n)
    constraints = [g >= 0, A @ g <= M]
    prob = cp.Problem(cp.Maximize(cp.sum(g)), constraints)
    try:
        prob.solve(solver=solver, warm_start=True, max_iters=2000)
    except cp.error.SolverError:
        return None
    if g.value is None:
        return None
    return np.array(g.value).ravel()

def iterative_lp(n, t=0.5, max_iter=20, time_deadline=..., start_f=None):
    if start_f is None:
        f = arcsine_initial(n)
    else:
        f = start_f.copy()
    sum_target = np.sqrt(2 * n)
    f = f * sum_target / f.sum()  # ensure sum = sqrt(2n)
    for it in range(max_iter):
        M, _ = max_convolution_and_index(f)
        g = lp_optimize_g(f, M)
        if g is None:
            break
        # scale g
        sum_g = np.sum(g)
        if sum_g == 0:
            break
        g = g * (sum_target / sum_g)
        # mixing
        f = (1 - t) * f + t * g
        # (optional) renormalize to sum_target precisely
        f = f * (sum_target / np.sum(f))
        # optionally break if improvement small
        new_M, _ = max_convolution_and_index(f)
        if new_M >= M - 1e-8:
            # no improvement
            break
    return f

            subject to  f * g ≤ M   (component‑wise)
                         g ≥ 0
       where M = max(f*f).  The solution g is scaled to the same
       ℓ¹‑norm √(2n) and mixed with f (parameter t=0.5).  Repeating this
       step (≤ 12 times) monotonically reduces the maximal self‑convolution.

  3. **Projected Adam refinement** – the best vector found for a length
     is fine‑tuned with a projected Adam optimiser on the exact
     max‑convolution objective (projecting onto the probability simplex
     after each step).  This often squeezes a few extra points.

All candidates are evaluated with the official ``evaluate_sequence``;
the best sequence seen before the time budget expires is returned.
The code uses only NumPy, SciPy (LP via HiGHS) and standard library.
"""

import time
from typing import List, Tuple

import numpy as np
from scipy.optimize import linprog

# ----------------------------------------------------------------------
#  Low‑level convolution utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Fast convolution via real FFT (zero‑padding to a power of two)."""
    m = len(a) + len(b) - 1
    n_fft = next_pow2(m)
    A = np.fft.rfft(a, n=n_fft)
    B = np.fft.rfft(b, n=n_fft)
    C = A * B
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


def max_convolution_and_index(a: np.ndarray) -> Tuple[float, int]:
    """Return (max value, arg‑max index) of a*a."""
    conv = convolution_fft(a, a)
    idx = int(np.argmax(conv))
    return float(conv[idx]), idx


# ----------------------------------------------------------------------
#  Helper for the sub‑gradient optimiser (used in the final polishing)
# ----------------------------------------------------------------------


def subgradient_of_max_at_index(a: np.ndarray, k: int) -> np.ndarray:
    """
    Sub‑gradient of the maximal convolution entry (index k) w.r.t. a.
    """
    n = len(a)
    grad = np.zeros_like(a)
    i = np.arange(n)
    j = k - i
    mask = (j >= 0) & (j < n)
    grad[mask] = a[j[mask]]
    if k % 2 == 0:
        diag = k // 2
        if diag < n:
            grad[diag] = 2.0 * a[diag]
    return grad


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection onto the probability simplex {x | x_i ≥ 0, Σx_i = 1}.
    Implementation of Duchi et al., 2008.
    """
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / len(v))

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho = np.where(u - cssv / np.arange(1, len(v) + 1) > 0)[0]
    if len(rho) == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def compute_score(a: np.ndarray) -> float:
    """Internal (scale‑invariant) objective used by the optimiser."""
    n = len(a)
    conv = convolution_fft(a, a)
    max_val = float(np.max(conv))
    s = float(np.sum(a))
    if s < 1e-12:
        return float("inf")
    return 2.0 * n * max_val / (s * s)


def improve_sequence(initial_a: np.ndarray,
                    time_deadline: float,
                    max_iter: int = 200_000,
                    eta0: float = 0.5,
                    noise_every: int = 500,
                    noise_scale: float = 1e-4) -> Tuple[np.ndarray, float]:
    """
    Projected Adam‑style sub‑gradient descent on the max‑convolution objective.
    Returns the best vector (projected on the simplex) and its internal score.
    """
    a = project_to_simplex(initial_a.astype(float).copy())
    best_a = a.copy()
    best_sc = compute_score(a)

    n = len(a)
    m1 = 0.9
    m2 = 0.999
    eps = 1e-8
    m = np.zeros_like(a)        # first moment
    v = np.zeros_like(a)        # second moment

    it = 0
    while time.time() < time_deadline and it < max_iter:
        it += 1
        # compute sub‑gradient of the maximal convolution entry
        _, k_max = max_convolution_and_index(a)
        grad = subgradient_of_max_at_index(a, k_max)

        # Adam update
        m = m1 * m + (1.0 - m1) * grad
        v = m2 * v + (1.0 - m2) * (grad ** 2)
        mh = m / (1.0 - m1 ** it)
        vh = v / (1.0 - m2 ** it)
        step = eta0 / np.sqrt(it)   # classic diminishing base step
        a_new = a - step * mh / (np.sqrt(vh) + eps)

        if noise_every and (it % noise_every == 0):
            a_new += np.random.randn(n) * noise_scale

        a_new = project_to_simplex(a_new)

        cur_sc = compute_score(a_new)
        if cur_sc < best_sc:
            best_sc = cur_sc
            best_a = a_new.copy()

        a = a_new

        if it % 5000 == 0:
            print(f"    iter={it:,d}  cur={cur_sc:.6f}  best={best_sc:.6f}")

    return best_a, best_sc


# ----------------------------------------------------------------------
#  Shape families for the initial scan
# ----------------------------------------------------------------------


def arcsine_initial(n: int) -> np.ndarray:
    """
    Classic arcsine‑shaped start:
          a_i ∝ 1 / √((i+½)·(n‑½‑i)).
    """
    i = np.arange(n, dtype=float)
    eps = 0.25
    vals = 1.0 / np.sqrt((i + 0.5) * (n - 0.5 - i) + eps)
    return vals / vals.sum()


def generate_shape(n: int, p: float, eps: float) -> np.ndarray:
    """
    Symmetric density a_i ∝ ((i+ε)(n‑1‑i+ε))⁻ᵖ, normalised to L¹‑norm 1.
    """
    i = np.arange(n, dtype=float)
    base = ((i + eps) * (n - 1 - i + eps)) ** (-p)
    total = base.sum()
    if total == 0.0:
        return np.full(n, 1.0 / n)
    return base / total


# ----------------------------------------------------------------------
#  Linear‑programming step (max‑convolution bound reduction)
# ----------------------------------------------------------------------


def build_conv_matrix(f: np.ndarray) -> np.ndarray:
    """
    Construct the (2n‑1)×n matrix A such that (A @ g)[k] = (f * g)[k].
    Zero‑padding outside the vector range is assumed.
    """
    n = len(f)
    rows = 2 * n - 1
    A = np.zeros((rows, n), dtype=float)
    for i in range(n):
        A[i:i + n, i] = f[i]
    return A


def lp_optimize_g(f: np.ndarray, M: float) -> np.ndarray | None:
    """
    Solve   maximise Σ g_j   s.t.   f * g ≤ M  (component‑wise) , g ≥ 0.
    Returns the optimal g (numpy array) or None on failure.
    """
    n = len(f)
    A = build_conv_matrix(f)
    b = np.full(2 * n - 1, M, dtype=float)
    c = -np.ones(n, dtype=float)          # minimise -Σ g → maximise Σ g
    bounds = [(0.0, None)] * n

    # SciPy's HiGHS backend is fast for dense LPs of this size.
    res = linprog(c, A_ub=A, b_ub=b, bounds=bounds,
                  method='highs', options={"presolve": True})
    if not res.success:
        return None
    return res.x


def iterative_lp(f_init: np.ndarray,
                 n: int,
                 deadline: float,
                 t: float = 0.5,
                 max_outer_iter: int = 12) -> np.ndarray:
    """
    Repeatedly apply the LP update described above.
    The vector is kept on the ℓ¹‑norm √(2n) sphere.
    """
    f = f_init.copy()
    target_sum = np.sqrt(2.0 * n)
    f = f * target_sum / f.sum()               # enforce exact ℓ¹‑norm

    for it in range(max_outer_iter):
        if time.time() >= deadline - 5.0:
            break

        M, _ = max_convolution_and_index(f)

        g = lp_optimize_g(f, M)
        if g is None:
            break

        # Rescale g to have the same ℓ¹‑norm as f
        g = g * (target_sum / g.sum())

        # Convex combination (the mix keeps the ℓ¹‑norm equal to target_sum)
        f = (1.0 - t) * f + t * g
        f = f * (target_sum / f.sum())          # numerical renormalisation

        # stop if the bound does not improve
        new_M, _ = max_convolution_and_index(f)
        if new_M >= M - 1e-12:
            break

    return f


# ----------------------------------------------------------------------
#  Main entry point required by the competition harness
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42,
                     budget_s: int = 1000,
                     **kwargs) -> List[float]:
    """
    Search for a non‑negative step‑function with a low
    ``evaluate_sequence`` value.  The routine respects the time budget,
    explores several lengths, applies the LP‑based active‑set reduction
    and finishes with a short projected‑Adam polishing.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 2.0            # tiny safety margin

    best_seq: List[float] = []
    best_val: float = float("inf")

    # ------------------------------------------------------------------
    # 0.  Optional fallback from the surrounding harness.
    # ------------------------------------------------------------------
    try:
        fallback = [float(x) for x in height_sequence_1]   # type: ignore
        val = evaluate_sequence(fallback)
        print(f"Fallback (height_sequence_1) score = {val:.6f}")
        if val < best_val:
            best_val = val
            best_seq = fallback
    except NameError:
        pass

    # ------------------------------------------------------------------
    # 1.  Explore a few candidate lengths.
    # ------------------------------------------------------------------
    lengths = [256, 512, 1024, 2048]

    for n in lengths:
        if time.time() >= deadline - 5.0:
            break
        print(f"\n=== Exploring length n = {n} ===")
        target_sum = np.sqrt(2.0 * n)

        # ------------------------------------------------------------------
        #   a) Build a modest library of starting shapes.
        # ------------------------------------------------------------------
        seeds: List[np.ndarray] = []

        # deterministic starters
        f_arc = arcsine_initial(n)
        f_arc = f_arc * target_sum / f_arc.sum()
        seeds.append(f_arc)

        f_unif = np.full(n, 1.0 / n)
        f_unif = f_unif * target_sum / f_unif.sum()
        seeds.append(f_unif)

        # random parametrised shapes
        for _ in range(6):
            p = rng.uniform(0.05, 1.0)
            eps = rng.uniform(1e-4, max(2.0, 0.02 * n))
            f_rand = generate_shape(n, p, eps)
            f_rand = f_rand * target_sum / f_rand.sum()
            seeds.append(f_rand)

        # ------------------------------------------------------------------
        #   b) Run the LP‑based active‑set improvement on each seed.
        # ------------------------------------------------------------------
        local_best_vec: np.ndarray | None = None
        local_best_score = float("inf")

        for s_idx, f_start in enumerate(seeds):
            if time.time() >= deadline - 5.0:
                break

            # ----- LP iteration -------------------------------------------------
            f_lp = iterative_lp(f_start, n, deadline,
                                t=0.5, max_outer_iter=12)

            seq = f_lp.tolist()
            val = evaluate_sequence(seq)
            print(f"  seed {s_idx} after LP => score {val:.6f}")

            if val < local_best_score:
                local_best_score = val
                local_best_vec = f_lp.copy()

        # ------------------------------------------------------------------
        #   c) Optional short Adam polishing of the best vector for this n.
        # ------------------------------------------------------------------
        if (local_best_vec is not None) and (time.time() < deadline - 5.0):
            # bring the vector onto the probability simplex (sum=1) for the optimiser
            a_simplex = local_best_vec / local_best_vec.sum()
            remaining = deadline - time.time()
            max_iter = min(120_000, int(remaining * 30))

            refined_a, _ = improve_sequence(
                initial_a=a_simplex,
                time_deadline=deadline,
                max_iter=max_iter,
                eta0=0.6,
                noise_every=600,
                noise_scale=3e-5,
            )
            refined_seq = (refined_a * target_sum).tolist()
            val_ref = evaluate_sequence(refined_seq)
            print(f"  polished (LP+Adam) score = {val_ref:.6f}")

            if val_ref < local_best_score:
                local_best_score = val_ref
                local_best_vec = np.array(refined_seq, dtype=float)

        # ------------------------------------------------------------------
        #   d) Record the best observed sequence for this length.
        # ------------------------------------------------------------------
        if local_best_vec is not None:
            seq = local_best_vec.tolist()
            if local_best_score < best_val:
                best_val = local_best_score
                best_seq = seq
                print(f"*** Global best updated to {best_val:.6f} (n={n}) ***")

    # ------------------------------------------------------------------
    # 2.  Final polishing of the overall best (time permitting).
    # ------------------------------------------------------------------
    if best_seq and time.time() < deadline - 5.0:
        n = len(best_seq)
        target_sum = np.sqrt(2.0 * n)
        a_simplex = np.array(best_seq, dtype=float) / np.sum(best_seq)

        remaining = deadline - time.time()
        max_iter = min(300_000, int(remaining * 30))

        refined_a, _ = improve_sequence(
            initial_a=a_simplex,
            time_deadline=deadline,
            max_iter=max_iter,
            eta0=0.5,
            noise_every=500,
            noise_scale=2e-5,
        )
        final_seq = (refined_a * target_sum).tolist()
        final_val = evaluate_sequence(final_seq)
        print(f"Final polishing score = {final_val:.6f}")

        if final_val < best_val:
            best_val = final_val
            best_seq = final_seq
            print("*** Updated global best after final polishing ***")

    # ------------------------------------------------------------------
    # 3.  Guard – always return a concrete (non‑empty) sequence.
    # ------------------------------------------------------------------
    if not best_seq:
        best_seq = [1.0] * 256
        best_val = evaluate_sequence(best_seq)

    print(f"\nSearch finished. Best reported score = {best_val:.6f}")
    return best_seq
