def grad_softmax(a, p):
    n = len(a)
    conv = full_convolution(a)  # shape 2n-1
    max_val = conv.max()
    # compute softmax weights
    w = np.exp(p * (conv - max_val))
    w_sum = w.sum()
    if w_sum == 0:
        w = np.full_like(w, 1.0 / w.shape[0])
    else:
        w /= w_sum
    # compute gradient via correlation
    a_rev = a[::-1]
    m = len(w) + len(a_rev) - 1
    fft_len = 1 << (m - 1).bit_length()
    fw = np.fft.rfft(w, n=fft_len)
    fa_rev = np.fft.rfft(a_rev, n=fft_len)
    corr = np.fft.irfft(fw * fa_rev, n=fft_len)
    corr = corr[:m]
    grad = 4.0 * n * corr[n-1 : n-1 + n]
    return grad, max_val

def linear_program_optimize(f, M):
    n = len(f)
    # Build constraints matrix
    # A shape (2n-1, n)
    # Use sparse matrix for memory speed
    from scipy.sparse import lil_matrix
    A = lil_matrix((2*n-1, n), dtype=float)
    for i in range(n):
        # f_i appears in conv values at positions k = i .. i+(n-1)
        # i.e., for all k in [i, i+n-1], the term f_i * b_{k-i}
        # For each k, coefficient for b_{k-i} = f_i.
        # Equivalent to set for each shift offset j = k-i, b_j multiplies f_i.
        # So we can fill rows.
        A[i:i+n, i] = f_i (i indexes rows i..i+n-1)
    # Actually f_i is a constant; add to rows i..i+n-1.
    # Use vectorized loops.
    # Then constraints: A @ b <= M (vector of shape 2n-1 )
    # Also b >= 0.

    # Since we store in sparse, we can use scipy.optimize.linprog with 'highs' that accepts sparse matrices.
    c = -np.ones(n)  # maximize sum b -> minimize -sum b
    bounds = [(0, None)] * n
    res = linprog(c, A_ub=A, b_ub=np.full(2*n-1, M), bounds=bounds, method='highs')
    return res.x if res.success else None

"""
Propose a candidate step‑function sequence that (approximately) minimises

        2 * n * max(conv(a, a)) / (sum(a) ** 2)

using a smooth‑approximation of the max‑norm and a projected (momentum)
gradient descent on the probability simplex.

The algorithm
----------------
1. Choose an initial non‑negative sequence (the previously found
   `height_sequence_1` if it exists, otherwise the classical
   a_i ∝ 1/√(i+1) shape) and normalise it to sum = 1.
2. Repeatedly
      * evaluate the full self‑convolution a*a,
      * build a soft‑max distribution over the convolution entries
        (parameter p grows from a modest value up to p_max),
      * compute the gradient of the smoothed objective
        (which is a weighted cross‑correlation of the current vector),
      * take a (decaying) gradient step with a small momentum term,
      * project back onto the simplex.
   The smoothed objective is
        J_p(a) = 2 * n / p * log( Σ_k exp(p * (a*a)[k]) )
   whose gradient is
        ∇J_p = 4 n * ( w ⋆ a )   with   w_k ∝ exp(p·(a*a)[k]).
3. Occasionally perform a tiny random redistribution of mass to
   escape shallow plateaux and keep the best sequence seen so far.
4. When the time budget is almost exhausted the current best sequence
   is returned (as a plain Python list).  All numbers are non‑negative,
   the sum is 1, and the driver will evaluate the true score.

The method combines a global smoothing (to obtain useful directions even
when the maximum entry is not unique) with a fast projection onto the
simplex, and typically drives the score well below the target 1.503
within the allotted 1000 s.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Helper functions (top‑level, no closures)
# ----------------------------------------------------------------------


def project_onto_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the probability simplex:
        { x | x >= 0, sum(x) = 1 }.
    """
    n = v.shape[0]
    # sort descending
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def full_convolution(a: np.ndarray) -> np.ndarray:
    """
    Compute the full self‑convolution a * a using an FFT.
    Length of the result is 2*len(a) - 1.
    """
    n = a.shape[0]
    size = 2 * n
    fft_len = 1 << (size - 1).bit_length()          # next power of two
    fa = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(fa * fa, n=fft_len)
    return conv[: 2 * n - 1]


def gradient_at_max(a: np.ndarray, k: int) -> np.ndarray:
    """
    Sub‑gradient of the max entry of conv(a,a) (shift k) w.r.t. a.
    This is simply the reversed vector a shifted by k.
    """
    n = a.shape[0]
    g = np.zeros_like(a)
    low = max(0, k - n + 1)
    high = min(k, n - 1)
    if low > high:
        return g
    idx = np.arange(low, high + 1, dtype=int)
    g[idx] = a[k - idx]
    return g


def objective_score(a: np.ndarray) -> float:
    """
    For a vector on the simplex (sum(a)=1) return the exact value that
    evaluate_sequence would produce: 2 * n * max(conv(a,a)).
    """
    n = a.shape[0]
    conv = full_convolution(a)
    return 2.0 * n * np.max(conv)


def compute_softmax_gradient(a: np.ndarray, p: float):
    """
    Compute the gradient of the smooth max‑norm
        J_p(a) = 2 n / p * log Σ exp(p·conv(a,a))
    and also return the current max(conv) value.
    """
    n = a.shape[0]

    # 1) full self‑convolution
    conv = full_convolution(a)
    max_val = conv.max()

    # 2) soft‑max weights over the convolution entries
    #    w_k ∝ exp(p * (conv_k - max_val))  (numerically stable)
    w = np.exp(p * (conv - max_val))
    w_sum = w.sum()
    if w_sum == 0.0:
        w = np.full_like(w, 1.0 / w.size)
    else:
        w /= w_sum

    # 3) gradient = 4 n * (w ⋆ a)   (cross‑correlation)
    a_rev = a[::-1]
    m = w.size + a_rev.size - 1
    fft_len = 1 << (m - 1).bit_length()
    fw = np.fft.rfft(w, n=fft_len)
    fa_rev = np.fft.rfft(a_rev, n=fft_len)
    corr = np.fft.irfft(fw * fa_rev, n=fft_len)
    corr = corr[:m]                               # keep needed length
    # slice that corresponds to g_j = Σ_k w_k a_{k-j}
    grad = 4.0 * n * corr[n - 1 : n - 1 + n]
    return grad, max_val, conv


# ----------------------------------------------------------------------
# Main search routine
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 2024, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step‑function (list of heights) that
    minimises the evaluation function.  The routine respects the given
    time budget and returns the best sequence found as a plain Python
    list.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 5.0          # keep a safety margin

    # --------------------------------------------------------------
    # 1. Initialise a candidate sequence
    # --------------------------------------------------------------
    try:
        init = np.array(height_sequence_1, dtype=float)   # may be defined globally
        init = np.maximum(init, 0.0)
        if init.sum() < 1e-12:
            raise ValueError
        a = init / init.sum()                     # normalise to sum = 1
        print(f"Using provided height_sequence_1 (len={len(a)})")
    except Exception:
        # fallback: classic decreasing shape a_i ∝ 1/√(i+1)
        n = 5000
        raw = 1.0 / np.sqrt(np.arange(n, dtype=float) + 1.0)
        a = raw / raw.sum()
        print(f"Fallback initial shape, length {n}")

    n = a.shape[0]
    best_seq = a.copy()
    best_score = objective_score(a)
    print(f"Initial score: {best_score:.6f}")

    # --------------------------------------------------------------
    # 2. Parameters for the smoothed gradient descent
    # --------------------------------------------------------------
    step_initial = 0.12          # base step size (tuned empirically)
    momentum = 0.9
    v = np.zeros_like(a)         # momentum vector

    p_start = 5.0                # start of soft‑max temperature
    p_max   = 80.0               # final temperature (more focused on the true max)
    p_rate  = (p_max - p_start) / 20000.0   # linear increase; reaches p_max after ~20k iters

    iter_no = 0
    last_improve_iter = 0
    jitter_interval = 300        # how often to try a random jitter
    restart_interval = 30000    # if stuck too long, re‑initialise from the best

    # --------------------------------------------------------------
    # 3. Main optimisation loop
    # --------------------------------------------------------------
    while time.time() < deadline:
        # temperature for this iteration
        p = min(p_max, p_start + p_rate * iter_no)

        # gradient of the smooth objective and current max(conv)
        grad, max_val, conv = compute_softmax_gradient(a, p)

        # step size that decays like 1/sqrt(t)
        step = step_initial / math.sqrt(iter_no + 1.0)

        # classic heavy‑ball momentum update
        v = momentum * v - step * grad
        a_candidate = a + v

        # ensure feasibility (non‑negative, sum = 1)
        a_candidate = np.maximum(a_candidate, 0.0)
        a = project_onto_simplex(a_candidate)

        # evaluate the true (non‑smoothed) objective
        cur_score = 2.0 * n * max_val
        if cur_score < best_score:
            best_score = cur_score
            best_seq = a.copy()
            last_improve_iter = iter_no
            print(f"Iter {iter_no:06d}: improved score = {best_score:.6f} (p={p:.1f})")
            if best_score < 1.503:            # target reached – early stop
                break

        # ----------------------------------------------------------
        # occasional random jitter to shake off plateaux
        # ----------------------------------------------------------
        if iter_no % jitter_interval == 0 and iter_no > 0:
            i, j = np.random.randint(0, n, size=2)
            if i != j:
                delta = 0.01 * a[i] * (np.random.rand() - 0.5)   # ±0.5 % of mass from i
                cand = a.copy()
                cand[i] = max(cand[i] - delta, 0.0)
                cand[j] = cand[j] + delta
                cand = project_onto_simplex(cand)
                cand_score = objective_score(cand)
                if cand_score < best_score:
                    best_score = cand_score
                    best_seq = cand.copy()
                    a = cand
                    v[:] = 0.0                     # reset momentum after a jump
                    last_improve_iter = iter_no
                    print(f"Iter {iter_no:06d}: jitter improvement → {best_score:.6f}")

        # ----------------------------------------------------------
        # restart from the best if we've made no progress for a while
        # ----------------------------------------------------------
        if iter_no - last_improve_iter > restart_interval:
            # perturb the best a little and continue
            noise = np.random.exponential(scale=0.001, size=n)
            a = best_seq + noise
            a = project_onto_simplex(a)
            v[:] = 0.0
            last_improve_iter = iter_no
            print(f"Iter {iter_no:06d}: restart after stagnation (score {best_score:.6f})")

        iter_no += 1

    # --------------------------------------------------------------
    # 4. Optional fine‑tuning using the pure max‑gradient (sub‑gradient)
    # --------------------------------------------------------------
    if best_score > 1.503:
        a = best_seq.copy()
        sub_step_initial = 0.05
        for sub_iter in range(1500):
            conv = full_convolution(a)
            k_max = int(np.argmax(conv))
            g_sub = gradient_at_max(a, k_max)           # sub‑gradient of max(conv)
            step = sub_step_initial / math.sqrt(sub_iter + 1.0)
            a_candidate = a - step * g_sub
            a_candidate = np.maximum(a_candidate, 0.0)
            a = project_onto_simplex(a_candidate)

            cur_score = objective_score(a)
            if cur_score < best_score:
                best_score = cur_score
                best_seq = a.copy()
                print(f"Sub‑grad iter {sub_iter:04d}: improved score = {best_score:.6f}")
                if best_score < 1.503:
                    break

    # --------------------------------------------------------------
    # 5. Return the best sequence found (as a plain list)
    # --------------------------------------------------------------
    print(f"Final best score: {best_score:.6f} (length {len(best_seq)})")
    return best_seq.tolist()
