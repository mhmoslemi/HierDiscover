def propose_candidate(...):
    import time, numpy as np
    set seed.

    start = time.time()
    deadline = start + budget_s - 0.1

    n = 2500
    target_sum = np.sqrt(2.0*n)
    fft_len = 1 << ((2*n - 1).bit_length())

    # pre-define function _fast_score_local using closure? Can't close but we can define global function that accepts len argument maybe.

    # Step 1: random low freq sampling
    best_seq = None, best_score = inf

    while time.time() < deadline - 50:  # leave 50 seconds for descent
        cand = random_lowfreq_candidate(n, target_sum)
        # compute score
        # compute conv quickly:
        # using np.fft.rfft
        F = np.fft.rfft(cand, n=fft_len)
        conv = np.fft.irfft(F*F, n=fft_len)[:2*n-1]
        max_conv = conv.max()
        score = float(2.0*n*max_conv/(cand.sum()**2))
        if score < best_score:
            best_score, best_seq = score, cand.copy()
            print(f"[{time.time() - start:6.1f}s] random candidate score {score:.6f}")

    # If best not found maybe use uniform as fallback.

    # Step 2: subgradient descent from best
    curr = best_seq.copy()
    # Initialize velocity, step...
    # as similar to earlier algorithm but maybe simpler:
    velocity = np.zeros_like(curr)
    step = target_sum / n * 0.5 # adjust magnitude
    momentum = 0.9
    rel_tol = 1e-4

    while time.time() < deadline:
        # compute conv, same as before
        F = np.fft.rfft(curr, n=fft_len)
        conv = np.fft.irfft(F*F, n=fft_len)[:2*n-1]
        max_val = conv.max()
        score = float(2.0*n*max_val/(curr.sum()**2))

        if score < best_score - 1e-10:
            best_score, best_seq = score, curr.copy()
            print(f"[{time.time() - start:6.1f}s] descent new best {score:.6f}")

        # active set
        tol_abs = max_val*rel_tol
        active = np.where(conv >= max_val - tol_abs)[0]
        # compute gradient:
        # conv derivative sum as earlier
        grad = np.zeros_like(curr)
        idx_arr = np.arange(n)
        for k in active:
            j = k - idx_arr
            mask = (j >=0) & (j < n)
            grad[mask] += 2.0*curr[j[mask]]
        gnorm = np.linalg.norm(grad)
        if gnorm == 0:
            break
        # step update
        velocity = momentum*velocity - step*(grad/gnorm)
        tentative = curr + velocity
        tentative = np.clip(tentative, 0., 1000.)
        curr = projection_simplex(tentative, target_sum)

        # jitter occasional
        if np.random.rand() < 0.01:
            jitter = np.random.randn(n) * step * 0.1
            curr = projection_simplex(np.clip(curr + jitter, 0., 1000.), target_sum)

        # decay step
        step = max(step*0.999, 1e-8)

    # final result: compute final score via evaluate_sequence
    final_score = evaluate_sequence(best_seq.tolist())
    print(f"Search finished. Final score: {final_score:.6f}")
    return best_seq.tolist()

"""[docstring]

Search for a step‑function (non‑negative sequence) that
minimises ``evaluate_sequence``.  The algorithm consists of
two phases:

1. **Fast stochastic sampling** – random vectors are drawn on the
   probability simplex (Dirichlet) or generated from a low‑frequency
   Fourier model.  Every sample is projected onto the simplex
   ``Σ a_i = √(2 n)`` (the scaling that makes the objective equal to
   ``max(conv(a,a))``) and its score is evaluated with an FFT‑based
   convolution.  The best sample found during the first part of the
   time budget is kept.

2. **Projected sub‑gradient descent** – starting from the best sample,
   we perform a simple sub‑gradient descent on the ``max`` functional.
   The active set (all indices whose convolution value is within a
   tiny tolerance of the current maximum) provides a sub‑gradient.
   Each step is followed by a projection onto the simplex.  Nesterov
   momentum and occasional random jitter help to escape plateaus.

The routine respects the 1000 s budget, prints intermediate scores
and finally returns the best sequence (as a plain list) found.
"""

import time
import numpy as np

# --------------------------------------------------------------------------- #
def projection_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Project ``v`` onto the simplex {x ≥ 0, Σ x = s}."""
    if s < 0:
        raise ValueError("simplex sum must be non‑negative")
    n = v.shape[0]
    # sort in descending order
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho_arr = np.where(u - cssv / (np.arange(n) + 1.0) > 0)[0]
    if rho_arr.size == 0:
        return np.zeros_like(v)
    rho = rho_arr[-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w

# --------------------------------------------------------------------------- #
def _fast_score(seq: np.ndarray, fft_len: int) -> float:
    """Return 2·n·max(conv(seq,seq)) / (sum(seq)²) using FFT."""
    n = seq.size
    F = np.fft.rfft(seq, n=fft_len)
    conv = np.fft.irfft(F * F, n=fft_len)[: 2 * n - 1]
    max_conv = conv.max()
    s = seq.sum()
    return float(2.0 * n * max_conv / (s * s))

# --------------------------------------------------------------------------- #
def _random_dirichlet_candidate(n: int, target_sum: float) -> np.ndarray:
    """Draw a random point on the simplex via a Dirichlet(1,…,1)."""
    cand = np.random.dirichlet(np.ones(n)) * target_sum
    # Dirichlet already respects the simplex, but we project for safety
    return projection_simplex(cand, target_sum)

# --------------------------------------------------------------------------- #
def _random_lowfreq_candidate(n: int, target_sum: float) -> np.ndarray:
    """Draw a random low‑frequency Fourier model and project onto the simplex."""
    m = 2 * n - 1                       # length after zero‑padding
    half = (m + 1) // 2                # size of the non‑redundant half‑spectrum

    # Choose a random cutoff for the number of active frequencies.
    # At least one non‑DC frequency, at most half‑1.
    kmax = np.random.randint(2, half)   # kmax ∈ [2, half‑1]

    # Random Gaussian coefficients for the active frequencies (complex).
    # DC term is set to zero so that the mean of the time‑domain signal is ≈0.
    active = np.random.randn(kmax - 1) + 1j * np.random.randn(kmax - 1)

    # Build the full Hermitian spectrum.
    X_half = np.zeros(half, dtype=complex)
    X_half[0] = 0.0                     # DC component
    X_half[1:kmax] = active

    # Mirror the non‑DC part to obtain a real‑valued inverse FFT.
    X = np.concatenate([X_half, np.conj(X_half[1:][::-1])])

    # Inverse FFT gives a real sequence of length m.
    f_full = np.fft.ifft(X).real
    cand = f_full[:n]

    # Make all entries non‑negative (shift by the minimum value).
    min_val = cand.min()
    if min_val < 0:
        cand = cand - min_val

    # Finally project onto the simplex Σ a_i = target_sum.
    cand = projection_simplex(cand, target_sum)
    return cand

# --------------------------------------------------------------------------- #
def propose_candidate(seed: int = 1234, budget_s: float = 1000.0, **kwargs):
    """
    Search for a non‑negative step‑function that minimises
    ``evaluate_sequence``.  The function respects the supplied time
    budget and returns the best sequence (as a list of floats) it has
    found.
    """
    np.random.seed(seed)
    start = time.time()
    deadline = start + budget_s - 0.2      # safety margin

    # ------------------------------------------------------------------- #
    # Problem size – we keep the vector moderately large (≈ 2500 entries)
    # because the objective is essentially scale‑invariant and a long
    # vector offers more degrees of freedom to flatten the autocorrelation.
    n = 2500
    target_sum = np.sqrt(2.0 * n)          # Σ a_i = √(2 n)
    fft_len = 1 << ((2 * n - 1).bit_length())   # next power‑of‑two ≥ 2n‑1

    # ------------------------------------------------------------------- #
    # 1. Stochastic seeding – Dirichlet + low‑frequency Fourier draws.
    best_seq = None
    best_score = float("inf")

    # Reserve a few seconds for the refinement phase.
    seed_phase_deadline = deadline - 30.0

    while time.time() < seed_phase_deadline:
        # Randomly decide which model to draw.
        if np.random.rand() < 0.5:
            cand = _random_dirichlet_candidate(n, target_sum)
        else:
            cand = _random_lowfreq_candidate(n, target_sum)

        # Fast scoring (the denominator is constant because Σ a_i = target_sum)
        cur_score = _fast_score(cand, fft_len)

        if cur_score < best_score:
            best_score = cur_score
            best_seq = cand.copy()
            print(f"[{time.time() - start:6.1f}s] new seed score {cur_score:.6f}")

    # ------------------------------------------------------------------- #
    # 2. If a global reference (height_sequence_1) exists, try it as an extra seed.
    try:
        global height_sequence_1
        if isinstance(height_sequence_1, (list, tuple, np.ndarray)):
            hs = np.asarray(height_sequence_1, dtype=np.float64)
            if hs.shape[0] == n:
                hs = np.clip(hs, 0.0, 1000.0)
                hs = projection_simplex(hs, target_sum)
                score = _fast_score(hs, fft_len)
                if score < best_score:
                    best_score = score
                    best_seq = hs.copy()
                    print(f"[{time.time() - start:6.1f}s] global seed score {score:.6f}")
    except Exception:
        pass

    # ------------------------------------------------------------------- #
    # 3. Projected sub‑gradient descent starting from the best seed.
    if best_seq is None:
        # safety fallback – uniform distribution
        best_seq = np.full(n, target_sum / n)
        best_score = _fast_score(best_seq, fft_len)

    curr = best_seq.copy()
    velocity = np.zeros_like(curr)

    # Initial absolute step size – a fraction of the average entry.
    step = 0.5 * target_sum / n
    min_step = 1e-9
    momentum = 0.9
    rel_tol = 1e-4
    it = 0

    while time.time() < deadline:
        # --- convolution -------------------------------------------------
        F = np.fft.rfft(curr, n=fft_len)
        conv = np.fft.irfft(F * F, n=fft_len)[: 2 * n - 1]
        max_val = conv.max()
        score = float(2.0 * n * max_val / (curr.sum() ** 2))

        # record improvement
        if score < best_score - 1e-12:
            best_score = score
            best_seq = curr.copy()
            print(f"[{time.time() - start:6.1f}s] iter {it:7d} → new best {score:.6f}")

        # --- active set (indices close to the max) ----------------------
        tol_abs = max_val * rel_tol
        active_idx = np.where(conv >= max_val - tol_abs)[0]

        # --- sub‑gradient (sum of shifted copies) ----------------------
        grad = np.zeros_like(curr)
        idx_arr = np.arange(n)
        for k in active_idx:
            j = k - idx_arr
            mask = (j >= 0) & (j < n)
            grad[mask] += 2.0 * curr[j[mask]]

        gnorm = np.linalg.norm(grad)
        if gnorm == 0.0 or step < min_step:
            break

        # --- Nesterov‑type momentum step -------------------------------
        velocity = momentum * velocity - step * (grad / gnorm)
        tentative = curr + velocity
        tentative = np.clip(tentative, 0.0, 1000.0)
        curr = projection_simplex(tentative, target_sum)

        # --- occasional random jitter to escape flat regions -----------
        if np.random.rand() < 0.015:
            jitter = (np.random.randn(n) *
                      step * target_sum / (30.0 * n))
            curr = projection_simplex(
                np.clip(curr + jitter, 0.0, 1000.0), target_sum)

        it += 1
        # slowly decay the step size
        if it % 500 == 0:
            step = max(step * 0.85, min_step)

    # ------------------------------------------------------------------- #
    # 4. Final sanity check with the official evaluator and return.
    final_score = evaluate_sequence(best_seq.tolist())
    print(f"Search finished. Final score reported by evaluate_sequence: {final_score:.6f}")

    return best_seq.tolist()
