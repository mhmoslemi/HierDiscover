"""
FFT‑accelerated projected gradient descent for the “max‑autocorrelation” problem.

The objective is  2·n·max(a∗a)/(∑a)² .  Because the objective is scale‑invariant,
we keep the sequence on the probability simplex (non‑negative, sum = 1).
At each iteration we

1. compute the full self‑convolution a∗a by a single real FFT,
2. locate the (almost) maximal entries and build a binary mask,
3. obtain the sub‑gradient of the max‑value by convolving that mask with the
   reversed sequence (again via FFT),
4. form a zero‑sum descent direction d = –g + mean(g),
5. perform a back‑tracking line‑search, projecting the trial point onto the
   simplex (Duchi et al. 2008) and accepting the move only if the objective
   does not increase,
6. occasionally inject a tiny random mass transfer to escape plateaus.

All FFT sizes are chosen as powers of two, so each iteration costs only a few
millisecond on a sub‑5000‑element problem.  The routine runs for at most the
given time budget and returns the best sequence seen.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
def next_pow2(x: int) -> int:
    """Return the smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


# ----------------------------------------------------------------------
def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the probability simplex:
    { w | w_i ≥ 0 , Σ w_i = 1 }  (Duchi–Shalev‑Shwartz algorithm).
    """
    n = v.shape[0]
    # sort in descending order
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(n) + 1
    # find rho
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all values are zero – return uniform point
        return np.full_like(v, 1.0 / n)
    rho = ind[cond].max()
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def binomial_sequence(length: int) -> np.ndarray:
    """
    Central‑binomial coefficients scaled by 4⁻ᵏ:
        c₀ = 1,
        cₖ₊₁ = cₖ·((2k+1)(2k+2))/(4(k+1)²).
    The vector is non‑negative and sums to ≈√(π·length/2), but we renormalise
    later to unit L¹ norm.
    """
    seq = np.empty(length, dtype=np.float64)
    seq[0] = 1.0
    for k in range(1, length):
        seq[k] = seq[k - 1] * ((2.0 * k - 1.0) * (2.0 * k)) / (4.0 * k * k)
    return seq
