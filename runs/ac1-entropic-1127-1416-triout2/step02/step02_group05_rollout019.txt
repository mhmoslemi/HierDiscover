"""
propose_candidate
-----------------
Hybrid multiplicative‑gradient / simulated‑annealing search for a
non‑negative step‑height vector that minimises the evaluation function

    2 * n * max(a * a) / (sum(a) ** 2).

The algorithm starts from the classic central‑binomial (C(2k,k)/4**k)
distribution, normalised to unit L1‑norm (the objective is scale‑invariant).
Each iteration:

1. Compute the full linear convolution with FFT (O(n log n)).
2. Locate all indices where the convolution attains its maximum.
3. Build a (sub‑)gradient of the maximal entry w.r.t. the coefficients.
4. Perform a *multiplicative* update a_i ← a_i·exp(η·d_i) where d is the
   zero‑sum direction (moving mass away from positions that contribute most
   to the current peak).  The vector is then renormalised onto the simplex.
5. Occasionally inject a tiny random mass‑transfer (simulated‑annealing) to
   escape shallow plateaus.
6. Keep the best vector seen so far.

The step size η decays slowly; the loop runs until the time budget is
exhausted (≈ 1000 s).  The returned list is guaranteed to contain only
non‑negative numbers and to satisfy the required format of
``evaluate_sequence``.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
def binomial_sequence(length: int) -> np.ndarray:
    """
    First ``length`` central‑binomial coefficients
    c_k = C(2k, k) / 4**k  (k = 0 … length‑1).

    They satisfy   c_0 = 1,
    c_{k+1} = c_k * ((2k+1)(2k+2)) / (4 (k+1)**2).
    """
    seq = np.empty(length, dtype=np.float64)
    seq[0] = 1.0
    for k in range(1, length):
        seq[k] = seq[k - 1] * ((2 * k - 1) * (2 * k)) / (4.0 * k * k)
    return seq


# ----------------------------------------------------------------------
def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a non‑negative vector onto the probability simplex
    (non‑negative, sum = 1).  The implementation follows the classic
    O(n·log n) algorithm.
    """
    v = np.maximum(v, 0.0)
    if v.sum() == 0.0:
        # Uniform fallback – never happens in our use‑case.
        return np.full_like(v, 1.0 / len(v))
    # ------------------------------------------------------------------
    # sort descending, find rho
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(len(v)) + 1
    cond = u - cssv / ind > 0
    rho = ind[cond][-1]                     # last true index
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def full_convolution_fft(a: np.ndarray) -> np.ndarray:
    """
    Compute the full linear convolution of ``a`` with itself using FFT.
    Result length is 2*len(a)-1.
    """
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())   # next power of two
    fa = np.fft.rfft(a, n=size)
    fc = fa * fa
    conv = np.fft.irfft(fc, n=size)[: 2 * n - 1]
    # Numerical noise can create tiny negative values – clip.
    conv[conv < 0] = 0.0
    return conv


# ----------------------------------------------------------------------
def subgradient_max_convolution(a: np.ndarray, conv: np.ndarray) -> np.ndarray:
    """
    ``conv`` must be the full self‑convolution of ``a``.
    Returns a zero‑sum direction ``d`` which points *downhill* for the
    objective (i.e. moving mass away from the positions that dominate the
    current maximum).
    """
    n = a.shape[0]
    max_val = conv.max()
    # All positions whose value is (numerically) maximal.
    max_idxs = np.flatnonzero(conv >= max_val - 1e-14)

    # Gradient of the (selected) maximal entry w.r.t. each a_i:
    #   g_i = Σ_{k∈max} a_{k-i}   (only defined when 0 ≤ k-i < n)
    g = np.zeros_like(a)
    for k in max_idxs:
        lo = max(0, k - (n - 1))
        hi = min(n - 1, k)
        idx = np.arange(lo, hi + 1, dtype=int)
        g[idx] += a[k - idx]          # a_{k-i}
    # Zero‑sum direction: d = -g + (mean(g))·1   (so Σ d_i = 0)
    d = -g + (g.mean())
    return d


# ----------------------------------------------------------------------

                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Return the best step‑function height sequence found within the given
    time budget.  The routine implements the optimisation described in the
    module doc‑string.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.1   # safety margin

    # --------------------- choose problem size ------------------------
    try:
        n = len(height_sequence_1)          # type: ignore[name-defined]
    except Exception:
        # A few thousand entries give a good trade‑off between quality and
        # running time on the 1000 s budget.
        n = 5400

    # --------------------- initialisation ----------------------------
    a = binomial_sequence(n).astype(np.float64)
    a = a / a.sum()                # L1‑normalise → sum = 1 (scale‑invariant)

    # Initial evaluation
    conv = full_convolution_fft(a)
    cur_max = conv.max()
    cur_score = 2.0 * n * cur_max          # because sum(a) == 1
    best_seq = a.copy()
    best_score = cur_score

    print(f"[init] n={n}, score={best_score:.9f}")

    # --------------------- optimisation parameters -------------------
    step_size = 0.6          # multiplicative‑update step (η)

    min_step = 1e-4          # never below this
    iter_cnt = 0

    # MAIN LOOP

            last_improve = time.time()

        # ---- 2. compute descent direction ---------------------------
        d = subgradient_max_convolution(a, conv)

        # ---- 3. multiplicative update (keeps positivity) ----------
        # clip d to avoid overflow in exp

        if np.random.rand() < 0.003:          # ~3 per 1000 iterations

                a_candidate = project_to_simplex(a_candidate)

        a = a_candidate

        # ---- 6. step‑size schedule ---------------------------------
        step_size = max(min_step, step_size * step_decay)

        iter_cnt += 1

    # ----------------------------------------------------------------
    # QUICK local refinement (pure hill‑climbing) using the remaining

                      f"time={elapsed:.2f}s")
        iter_cnt += 1

    # ----------------------------------------------------------------
    print(f"[final] best score={best_score:.9f}, iterations={iter_cnt}")
    # The evaluator clips to [0,1000]; our numbers are far smaller.
