Now mirror update:

```
def mirror_update(p, g, lr):
    # p is probability vector sum 1; g is subgradient.
    # p_new proportional to p * exp(-lr * g)
    # To avoid overflow, subtract max shift
    shift = -lr * g
    max_shift = shift.max()
    exp_shift = np.exp(shift - max_shift)
    p_new = p * exp_shift
    p_new_sum = p_new.sum()
    if p_new_sum <= 0:
        # fallback to uniform
        p_new = np.full_like(p, 1.0/p.shape[0])
    else:
        p_new = p_new / p_new_sum
    return p_new
```

def run_sgd(start_a, max_iters, step_base):
    a = start_a.copy()
    best_local = a.copy()
    best_score = compute_objective(a)
    for t in range(max_iters):
        # compute conv via FFT
        fft_a = np.fft.rfft(a, n=2*n-1)
        conv = np.fft.irfft(fft_a*fft_a, n=2*n-1)
        max_b = conv.max()
        if max_b * 2 * n < best_score:
            best_score = 2 * n * max_b
            best_local = a.copy()
        # subgradient:
        eps = max_b * 1e-5
        active = np.where(conv >= max_b - eps)[0]
        # random weights Dirichlet
        w = np.random.dirichlet(np.ones(active.size))
        g = np.zeros_like(a)
        # For each active idx with weight
        for wi, k in zip(w, active):
            low = max(0, k - n + 1)
            high = min(n-1, k)
            idx = np.arange(low, high+1)
            partner = k - idx
            g[idx] += 2 * wi * a[partner]
        # step size schedule
        step = step_base / np.sqrt(t+1)
        a = simplex_projection(a - step * g)
    return best_local, best_score
```

    best_score = compute_objective(start_a)

    for t in range(max_iters):
        # compute a
        max_x = np.max(x)
        exp_x = np.exp(x - max_x)
        a = exp_x / exp_x.sum()
        # compute conv and active set
        fft = np.fft.rfft(a, n=2*n-1)
        conv = np.fft.irfft(fft * fft, n=2*n-1)
        max_b = conv.max()
        score = 2.0 * n * max_b
        if score < best_score:
            best_score = score
            best_a = a.copy()
            print(f"Adam iter {t}: new best {best_score:.6f}")

        eps_rel = max_b * 1e-5
        active = np.where(conv >= max_b - eps_rel)[0]
        if active.size == 0:

        g = np.zeros_like(a)

            low = max(0, k - n + 1)
            high = min(n-1, k)
            idx = np.arange(low, high+1, dtype=int)
            partner = k - idx
            g[idx] += 2.0 * wi * a[partner]
        # g is subgradient of max_b, not scaled 2n factor.
        # Now compute gradient w.r.t x:
        # grad_x = a * (g - np.dot(a,g))
        ag = np.dot(a, g)
        grad_x = a * (g - ag)

        # Adam update
        m = beta1 * m + (1 - beta1) * grad_x
        v = beta2 * v + (1 - beta2) * (grad_x ** 2)
        m_hat = m / (1 - beta1 ** (t + 1))

def smooth_gradient(a, p):
    # compute conv and gradient for smooth objective f.
    n = len(a)
    L = 2*n-1
    fft_a = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(fft_a * fft_a, n=L)
    # compute powers
    conv_p = conv ** p
    sum_conv_p = conv_p.sum()
    f = sum_conv_p ** (1.0 / p)
    if f == 0:
        return np.zeros_like(a), f
    # w = conv^{p-1}
    w = conv ** (p - 1)
    # compute convolution w * a_rev
    a_rev = a[::-1]
    L2 = next_pow2(L + n - 1)  # maybe use same L? Could zero pad.

    # extract relevant part: indices from n-1 to n-1 + n
    g = 2.0 * conv2[(n-1):(n-1+n)] / (f ** (p - 1))

        ag = np.dot(a, g)
        grad_x = a * (g - ag)
        # Adam update on x
        iteration += 1
        m = beta1 * m + (1 - beta1) * grad_x
        v = beta2 * v + (1 - beta2) * (grad_x ** 2)
        m_hat = m / (1 - beta1 ** iteration)
        v_hat = v / (1 - beta2 ** iteration)
        x = x - lr * m_hat / (np.sqrt(v_hat) + eps_adam)
    return best_a, best_score

"""
Propose a height sequence that minimises the evaluation function.

The algorithm works in three stages:

1. **Seeding** – a collection of analytically motivated or random
   distributions (linear ramp, uniform, Beta‑family, Dirichlet) is
   normalised to sum 1 and the best one according to the exact
   objective `2·n·max(a*a)` is kept.

2. **Stochastic sub‑gradient descent** – starting from the best seed we
   repeatedly compute the self‑convolution, pick the (almost) active
   entries, form a random convex combination of their sub‑gradients,
   and take a projected sub‑gradient step onto the probability simplex.
   A simple momentum term and a diminishing stepsize are used.
   This phase consumes most of the time budget.

3. **Adam on the soft‑max parametrisation** – the current best vector is
   written as `a = softmax(x)`.  An (unbiased) sub‑gradient of the
   original objective w.r.t. `a` is transformed to a gradient w.r.t.
   `x` and optimised with Adam.  The iterate stays on the simplex
   automatically, and a short refinement is performed after the first
   phase.

Both phases constantly keep the best sequence seen so far and finally
return it as a plain list of floats.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper functions (top level, no closures)
# ----------------------------------------------------------------------


def simplex_projection(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of a vector onto the probability simplex."""
    v = np.asarray(v, dtype=float)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1] - 1
    theta = cssv[cond][-1] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def compute_objective(a: np.ndarray) -> float:
    """Exact objective: 2·n·max(a*a)  (assumes sum(a)=1)."""
    n = a.shape[0]
    fft = np.fft.rfft(a, n=2 * n - 1)
    conv = np.fft.irfft(fft * fft, n=2 * n - 1)
    return float(2.0 * n * conv.max())


def beta_sequence(alpha: float, beta: float, n: int) -> np.ndarray:
    """Discretised Beta(α,β) distribution on n points, normalised to sum 1."""
    x = (np.arange(n) + 0.5) / n
    logw = (alpha - 1.0) * np.log(x) + (beta - 1.0) * np.log(1.0 - x)
    logw = np.clip(logw, -700.0, 700.0)
    w = np.exp(logw)
    total = w.sum()
    if total == 0 or not np.isfinite(total):
        w = np.ones_like(w)
        total = w.sum()
    return w / total


def random_dirichlet(n: int) -> np.ndarray:
    """Sample a Dirichlet‑like vector (Gamma(α,1) with random α)."""
    conc = np.random.uniform(0.1, 3.0, size=n)
    a = np.random.gamma(conc, 1.0)
    a_sum = a.sum()
    if a_sum == 0:
        a = np.full_like(a, 1.0 / n)
    else:
        a = a / a_sum
    return a


def subgradient_random(a: np.ndarray, conv: np.ndarray, max_b: float) -> np.ndarray:
    """Return a stochastic sub‑gradient of the max‑convolution term."""
    n = a.shape[0]
    eps = max_b * 1e-5
    active = np.where(conv >= max_b - eps)[0]
    if active.size == 0:
        active = np.array([conv.argmax()], dtype=int)

    # random convex combination of the active gradients
    w = np.random.dirichlet(np.ones(active.size))
    g = np.zeros_like(a)
    for wi, k in zip(w, active):
        low = max(0, k - n + 1)
        high = min(n - 1, k)
        idx = np.arange(low, high + 1, dtype=int)
        partner = k - idx
        g[idx] += 2.0 * wi * a[partner]
    return g


def run_subgradient(a_start: np.ndarray, deadline: float) -> tuple[np.ndarray, float]:
    """Projected stochastic sub‑gradient descent with momentum."""
    n = a_start.shape[0]
    a = a_start.copy()
    best_a = a.copy()
    best_score = compute_objective(a)

    beta_mom = 0.9          # momentum coefficient
    m = np.zeros_like(a)    # momentum accumulator
    eta = 0.8               # stepsize constant
    it = 0

    while time.time() < deadline:
        # convolution and max entry
        fft_a = np.fft.rfft(a, n=2 * n - 1)
        conv = np.fft.irfft(fft_a * fft_a, n=2 * n - 1)
        max_b = conv.max()
        cur_score = 2.0 * n * max_b

        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"SubGrad iter {it:6d}: new best {best_score:.6f}")

        # stochastic sub‑gradient of the max entry
        g = subgradient_random(a, conv, max_b)

        # momentum update and projected step
        m = beta_mom * m + (1.0 - beta_mom) * g
        step = eta / np.sqrt(it + 1)
        a = simplex_projection(a - step * m)

        it += 1
    return best_a, best_score


def run_adam_softmax(a_start: np.ndarray, deadline: float) -> tuple[np.ndarray, float]:
    """Adam optimisation on the soft‑max parametrisation of the simplex."""
    n = a_start.shape[0]
    eps = 1e-12
    x = np.log(a_start + eps)                      # logits
    m = np.zeros_like(x)
    v = np.zeros_like(x)
    beta1, beta2 = 0.9, 0.999
    eps_adam = 1e-8
    lr = 0.5

    best_a = a_start.copy()
    best_score = compute_objective(best_a)
    it = 0

    while time.time() < deadline:
        # soft‑max to obtain a probability vector
        max_x = np.max(x)
        exp_x = np.exp(x - max_x)
        a = exp_x / exp_x.sum()

        # convolution
        fft_a = np.fft.rfft(a, n=2 * n - 1)
        conv = np.fft.irfft(fft_a * fft_a, n=2 * n - 1)
        max_b = conv.max()
        cur_score = 2.0 * n * max_b

        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"Adam   iter {it:6d}: new best {best_score:.6f}")

        # stochastic sub‑gradient for the max entry
        eps_rel = max_b * 1e-5
        active = np.where(conv >= max_b - eps_rel)[0]
        if active.size == 0:
            active = np.array([conv.argmax()], dtype=int)
        w = np.random.dirichlet(np.ones(active.size))
        g = np.zeros_like(a)
        for wi, k in zip(w, active):
            low = max(0, k - n + 1)
            high = min(n - 1, k)
            idx = np.arange(low, high + 1, dtype=int)
            partner = k - idx
            g[idx] += 2.0 * wi * a[partner]

        # gradient w.r.t. logits (soft‑max Jacobian)
        ag = np.dot(a, g)
        grad_x = a * (g - ag)

        # Adam update
        it += 1
        m = beta1 * m + (1 - beta1) * grad_x
        v = beta2 * v + (1 - beta2) * (grad_x ** 2)
        m_hat = m / (1 - beta1 ** it)
        v_hat = v / (1 - beta2 ** it)
        x = x - lr * m_hat / (np.sqrt(v_hat) + eps_adam)

    return best_a, best_score


def run_smooth_descent(a_start: np.ndarray, deadline: float, p: int = 8) -> tuple[np.ndarray, float]:
    """
    Gradient descent on the smoothed objective
        f(a) = ( Σ (a*a)_i^p )^{1/p}
    which approximates the max‑convolution.
    """
    n = a_start.shape[0]
    a = a_start.copy()
    best_a = a.copy()
    best_score = compute_objective(a)

    eta = 0.6
    it = 0
    # length needed for the convolution w * a_rev
    L2 = 3 * n - 2

    while time.time() < deadline:
        # self‑convolution
        fft_a = np.fft.rfft(a, n=2 * n - 1)
        conv = np.fft.irfft(fft_a * fft_a, n=2 * n - 1)

        # smooth objective components
        conv_p = conv ** p
        sum_conv_p = conv_p.sum()
        f = sum_conv_p ** (1.0 / p)
        if f == 0:
            break
        w = conv ** (p - 1)

        # convolution w * a_rev (cross‑correlation)
        a_rev = a[::-1]
        fft_w = np.fft.rfft(w, n=L2)
        fft_arev = np.fft.rfft(a_rev, n=L2)
        conv2 = np.fft.irfft(fft_w * fft_arev, n=L2)

        # gradient of the smoothed objective
        grad = 2.0 * conv2[(n - 1):(n - 1 + n)] / (f ** (p - 1))

        # projected gradient step
        step = eta / np.sqrt(it + 1)
        a = simplex_projection(a - step * grad)

        # monitor exact max‑convolution objective
        cur_score = compute_objective(a)
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"Smooth iter {it:6d}: new best {best_score:.6f}")

        it += 1

    return best_a, best_score
