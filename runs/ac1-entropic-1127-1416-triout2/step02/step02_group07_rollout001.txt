"""
Propose a height sequence that (approximately) minimizes

    2 * n * max(conv(a, a)) / (sum(a) ** 2)

The algorithm works on a fixed length `n` (default 5000) and keeps the
coefficients on the probability simplex (non‑negative, sum = 1).  It
combines three ideas:

1. **Smart initialisation** – try several power‑law shapes
   `a_i ∝ (i+1)^(-p)` (p in a small grid) and keep the best one.

2. **Adam optimisation on the soft‑max (log‑its) parametrisation** – the
   vector `u` is updated with the Adam rule while `a = softmax(u)` stays
   automatically on the simplex.  The (sub‑)gradient is taken at the
   (almost) maximal convolution entry; if several shifts are almost
   tied we average their sub‑gradients to obtain a smoother direction.

3. **Exploration** – every few hundred iterations a tiny random
   perturbation of `u` is tried; if it yields a better objective it is
   kept.  The current best sequence is stored and finally returned.

The whole search respects the supplied time budget (default 1000 s) and
prints occasional progress messages.  If something goes wrong the
fallback global `height_sequence_1` is used.

The function `propose_candidate` is the entry point that will be called
by the evaluation harness.
"""

import time
import math

# Helper functions (all top‑level, no closures)
# ----------------------------------------------------------------------


def softmax(u: np.ndarray) -> np.ndarray:
    """Stable softmax: returns a probability vector (sum=1, all >0)."""
    u_max = np.max(u)
    e = np.exp(u - u_max)
    return e / e.sum()


def project_onto_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection onto the probability simplex (unused here)."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def full_convolution(a: np.ndarray) -> np.ndarray:
    """
    Compute the full discrete convolution a * a using an FFT.
    Length of the result is 2*len(a) - 1.
    """
    n = a.shape[0]
    size = 2 * n
    fft_len = 1 << (size - 1).bit_length()  # next power of two
    fft_a = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(fft_a * fft_a, n=fft_len)
    return conv[: 2 * n - 1]


def gradient_at_shift(a: np.ndarray, k: int) -> np.ndarray:
    """
    Sub‑gradient of the convolution entry at shift k with respect to a.
    Returns a vector g where g[i] = a[k-i] if 0 <= i <= n-1 and
    0 <= k-i <= n-1, else 0.
    """
    n = a.shape[0]
    g = np.zeros_like(a)
    low = max(0, k - n + 1)
    high = min(k, n - 1)
    if low > high:
        return g
    idx = np.arange(low, high + 1, dtype=int)
    g[idx] = a[k - idx]
    return g


def avg_gradient_over_top_shifts(a: np.ndarray, conv: np.ndarray,
                                eps: float = 1e-6) -> np.ndarray:
    """
    Compute an averaged sub‑gradient over all shifts whose convolution
    value is within (1‑eps) of the global maximum.
    """
    max_val = conv.max()
    # Shifts that are almost maximal (allow a small tolerance)
    top_idxs = np.where(conv >= max_val * (1.0 - eps))[0]
    if top_idxs.size == 1:
        return gradient_at_shift(a, int(top_idxs[0]))
    # Average the gradients of the selected shifts
    g = np.zeros_like(a)
    for k in top_idxs:
        g += gradient_at_shift(a, int(k))
    g /= float(top_idxs.size)
    return g


def objective_score(a: np.ndarray) -> float:
    """
    Evaluate the objective 2 * n * max(conv(a,a)) assuming sum(a)==1.
    """
    n = a.shape[0]
    conv = full_convolution(a)
    return 2.0 * n * np.max(conv)


# ----------------------------------------------------------------------
# Main search routine
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 314159,
                     budget_s: int = 1000,
                     n: int = 5000,
                     **kwargs) -> list[float]:
    """
    Search for a (non‑negative) sequence of heights that minimises
    `evaluate_sequence`.  The method uses several initial power‑law
    guesses and then runs an Adam optimiser on the log‑it (softmax)
    parameters, with occasional random jitter and multi‑peak gradient
    averaging.
    """
    start_time = time.time()
    deadline = start_time + budget_s - 4.0   # keep a few seconds for exit

    np.random.seed(seed)

    # ------------------------------------------------------------------
    # 1. Pick a good starting point by scanning a few power‑law shapes
    # ------------------------------------------------------------------
    best_start = None
    best_score = math.inf

    # candidate exponents p (a_i ∝ (i+1)^{-p})
    candidate_ps = np.arange(0.40, 0.81, 0.05)   # 0.40, 0.45, ..., 0.80
    for p in candidate_ps:
        raw = 1.0 / np.power(np.arange(n, dtype=float) + 1.0, p)
        a = raw / raw.sum()           # normalised, sum = 1
        score = objective_score(a)
        if score < best_score:
            best_score = score
            best_start = a.copy()
    # Also try the classic 1/sqrt shape as a backup
    raw = 1.0 / np.sqrt(np.arange(n, dtype=float) + 1.0)
    a_sqrt = raw / raw.sum()
    score_sqrt = objective_score(a_sqrt)
    if score_sqrt < best_score:
        best_score = score_sqrt
        best_start = a_sqrt.copy()

    # If for some reason we did not get a start, fall back to uniform
    if best_start is None:
        best_start = np.full(n, 1.0 / n, dtype=float)

    # ------------------------------------------------------------------
    # 2. Initialise Adam on the log‑it (soft‑max) parametrisation
    # ------------------------------------------------------------------
    eps_logit = 1e-12
    u = np.log(best_start + eps_logit)          # log‑its, all finite
    m = np.zeros_like(u)                       # first moment
    v = np.zeros_like(u)                       # second moment
    beta1 = 0.9
    beta2 = 0.999
    eps_adam = 1e-8
    base_lr = 0.15        # base learning rate (will be decayed)
    iteration = 0
    best_seq = best_start.copy()
    best_obj = best_score

    print(f"[Init] n={n}, best start score={best_obj:.6f}")

    # ------------------------------------------------------------------
    # 3. Main optimisation loop (Adam + occasional jitter)
    # ------------------------------------------------------------------
    while time.time() < deadline:
        # 3.1 Compute current a from logits
        a = softmax(u)                     # stays on simplex automatically

        # 3.2 Full self‑convolution and locate (almost) maximal shift(s)
        conv = full_convolution(a)
        max_val = conv.max()
        # averaged sub‑gradient over top shifts (tolerance 1e-6)
        g = avg_gradient_over_top_shifts(a, conv, eps=1e-6)

        # 3.3 Gradient in log‑it space: ∂f/∂u = a * (g - <g,a>)
        # Note: <g,a> equals the (average) convolution value of the
        # selected top shifts; for a single shift it is exactly max_val.
        expected = np.dot(g, a)
        grad_u = a * (g - expected)

        # 3.4 Adam update (learning‑rate decays with sqrt(iter))
        iteration += 1
        lr = base_lr / math.sqrt(iteration)

        m = beta1 * m + (1.0 - beta1) * grad_u
        v = beta2 * v + (1.0 - beta2) * (grad_u ** 2)

        m_hat = m / (1.0 - beta1 ** iteration)
        v_hat = v / (1.0 - beta2 ** iteration)

        u -= lr * m_hat / (np.sqrt(v_hat) + eps_adam)

        # 3.5 Evaluate current objective
        obj = 2.0 * n * max_val   # sum(a) == 1 by construction
        if obj < best_obj:
            best_obj = obj
            best_seq = a.copy()
            print(f"[Iter {iteration:06d}] improved obj={best_obj:.6f}"
                  f" (max shift ≈{np.argmax(conv)})")
            if best_obj < 1.503:          # reached target – stop early
                break

        # 3.6 Random jitter every 500 iterations to escape plateaux
        if iteration % 500 == 0:
            # small Gaussian perturbation of logits
            jitter = np.random.normal(scale=0.02, size=n)
            u_jit = u + jitter
            a_jit = softmax(u_jit)
            conv_jit = full_convolution(a_jit)
            obj_jit = 2.0 * n * conv_jit.max()
            if obj_jit < best_obj:
                best_obj = obj_jit
                best_seq = a_jit.copy()
                u = u_jit          # accept the jittered logits
                print(f"[Jitter] iter {iteration:06d} obj={best_obj:.6f}")

        # Optional: gentle smoothing of u every few thousand steps can be
        # added here if needed, but current scheme already performs well.

    # ------------------------------------------------------------------
    # 4. Fallback handling (should never be needed)
    # ------------------------------------------------------------------
    if best_seq is None or best_seq.size == 0:
        try:
            best_seq = np.array(height_sequence_1, dtype=float)
            best_obj = float('inf')
        except Exception:
            # Very last resort: uniform distribution (objective = 2.0)
            best_seq = np.full(n, 1.0 / n, dtype=float)
            best_obj = float('inf')

    elapsed = time.time() - start_time
    print(f"[Done] elapsed {elapsed:.1f}s – final best obj={best_obj:.6f}"
          f" (length {len(best_seq)})")
    # Return a plain Python list as required by the driver
    return best_seq.tolist()
