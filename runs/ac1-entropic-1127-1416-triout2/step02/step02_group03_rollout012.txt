"""
propose_candidate – heuristic search for a non‑negative step function
with a low value of

    evaluate_sequence(a) = 2 * n * max(conv(a,a)) / (sum(a)**2)

The algorithm works in three stages:

1.  Fast random exploration of a 5‑parameter family
        a_i ∝ (i+1+α)^(-p) · (1 + β·i/(n‑1)) · exp(-γ·i/(n‑1)),
    (α≥0, 0.3≤p≤0.8, –1≤β≤3, 0≤γ≤5).  After generating a raw vector the
    entries are simply scaled so that Σa = √(2n).  The best candidate from
    this sweep is kept.

2.  Projected Polyak sub‑gradient descent on the objective
    `max(conv(a,a))`.  For the current maximum indices K the sub‑gradient
    is
        g_i = Σ_{k∈K} a_{k-i}     (out‑of‑range terms ignored).
    The step size follows a Polyak rule
        η = (M – M_low) / ||g||²   (clamped to a safe interval),
    where `M` is the current maximal convolution value and
    `M_low = (Σa)²/(2n‑1) = 2n/(2n‑1)` is the absolute lower bound.
    After each step the vector is projected back onto the simplex
    `{x≥0, Σx = √(2n)}` using the Euclidean projection of Duchi et al.

3.  The descent is performed in several “chunks”.  When a chunk finishes
    (or when progress stalls) a new random restart is started, thereby
    exploring many basins of attraction.  The best sequence seen over the
    whole budget (default 1000 s) is returned as a plain Python list.

The implementation uses only NumPy and the supplied `evaluate_sequence`
function; no external files or network access are required.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Simplex utilities
# ----------------------------------------------------------------------


def _project_simplex(v: np.ndarray, target_sum: float) -> np.ndarray:
    """
    Euclidean projection of a vector onto the simplex
        { w | w_i ≥ 0, Σ w_i = target_sum }.
    Implements the algorithm of Duchi, Shalev‑Shwartz, Singer & Chandra
    (2008).  The input vector `v` is not modified.
    """
    if target_sum < 0:
        raise ValueError("target_sum must be non‑negative")
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - target_sum
    ind = np.arange(1, len(v) + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all entries will be zero – should not happen here
        return np.zeros_like(v)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Convolution helpers
# ----------------------------------------------------------------------


def _full_convolution_fft(a: np.ndarray) -> np.ndarray:
    """
    Linear convolution a*a using a real FFT.
    Returns an array of length 2·len(a)‑1.
    """
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())  # next power of two
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return conv


def _max_conv(a: np.ndarray) -> float:
    """Maximum entry of the self‑convolution of a."""
    return _full_convolution_fft(a).max()


# ----------------------------------------------------------------------
# Candidate generation (parameterised family)
# ----------------------------------------------------------------------


def _make_candidate_extended(
    n: int,
    p: float,
    offset: float,
    beta: float,
    gamma: float,
) -> np.ndarray:
    """
    Build a raw (un‑normalised) sequence
        a_i = (i+1+offset)^(-p) *
              (1 + beta * i/(n‑1)) *
              exp(-gamma * i/(n‑1))
    Clip negatives, then scale so that Σ a = √(2·n).
    """
    idx = np.arange(n, dtype=float)                      # i = 0 … n‑1
    base = np.power(idx + 1.0 + offset, -p)              # power‑law decay
    tilt = 1.0 + beta * idx / (n - 1)                    # linear tilt
    expo = np.exp(-gamma * idx / (n - 1))                # exponential tail
    raw = base * tilt * expo
    raw = np.maximum(raw, 0.0)                           # enforce non‑negative
    total = raw.sum()
    if total <= 0.0:
        # fall back to uniform distribution
        return np.full(n, np.sqrt(2.0 * n) / n, dtype=float)
    scale = np.sqrt(2.0 * n) / total
    return raw * scale


# ----------------------------------------------------------------------
# Sub‑gradient step (Polyak step size)
# ----------------------------------------------------------------------


def _subgradient_step_polyak(
    a: np.ndarray,
    max_inds: np.ndarray,
    target_sum: float,
    lower_bound: float,
    step_factor: float,
    min_step: float,
    max_step: float,
) -> np.ndarray:
    """
    Perform one Polyak sub‑gradient step for the objective
    max(conv(a,a)).  The gradient is accumulated over all indices where
    the current maximum is attained.  After the step a Euclidean projection
    onto the simplex `{x ≥ 0, Σx = target_sum}` is applied.
    """
    n = a.shape[0]
    g = np.zeros_like(a)

    # Gradient accumulation – vectorised for each max index
    for k in max_inds:
        if k < n:
            # contributions from a[0] … a[k] (reversed)
            g[: k + 1] += a[k::-1]
        else:
            start = k - (n - 1)                     # first i that contributes
            # indices i = start … n‑1, term a_{k-i}
            idx = np.arange(start, n, dtype=int)
            g[start:] += a[k - idx]

    norm_g = np.linalg.norm(g)
    if norm_g == 0.0:
        # nothing to do – return the projected point unchanged
        return _project_simplex(a, target_sum)

    # Polyak step size, guarded against division by zero
    step = step_factor * (np.max(_full_convolution_fft(a)) - lower_bound) / (
        norm_g * norm_g + 1e-12
    )
    # Clamp step to a sensible interval
    step = max(min(step, max_step), min_step)

    cand = a - step * g
    return _project_simplex(cand, target_sum)


# ----------------------------------------------------------------------
# Main optimisation driver
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 123, budget_s: int = 1000) -> list[float]:
    """
    Searches for a step‑function (list of non‑negative heights) that
    minimises ``evaluate_sequence`` under the 1000 s time budget.
    Returns the best sequence found as a plain Python list.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 2.0           # safety margin

    # --------------------------------------------------------------
    # 0. Problem size – a few‑thousand entries give a good trade‑off.
    # --------------------------------------------------------------
    n = 12000
    target_sum = np.sqrt(2.0 * n)
    lower_conv_bound = (target_sum * target_sum) / (2 * n - 1)   # = 2n/(2n‑1)

    # --------------------------------------------------------------
    # 1. Warm‑start: use a previously computed sequence if available.
    # --------------------------------------------------------------
    best_seq = None
    hs = globals().get("height_sequence_1")
    if isinstance(hs, list) and len(hs) == n:
        best_seq = np.array(hs, dtype=float)
        best_seq = _project_simplex(best_seq, target_sum)

    # fallback: simple power‑law seed
    if best_seq is None:
        best_seq = _make_candidate_extended(
            n=n, p=0.5, offset=0.0, beta=0.0, gamma=0.0
        )

    best_score = _max_conv(best_seq)
    print(f"[init] n={n:5d}  score={best_score:.6f}")

    # --------------------------------------------------------------
    # 2. Random exploration of the 5‑parameter family (≈ 20 % budget)
    # --------------------------------------------------------------
    sweep_deadline = time.time() + 0.20 * (deadline - time.time())
    while time.time() < sweep_deadline:
        # Sample parameters uniformly within generous ranges
        p = np.random.uniform(0.30, 0.80)
        off = np.random.uniform(0.0, 15.0)
        beta = np.random.uniform(-1.0, 3.0)
        gamma = np.random.uniform(0.0, 5.0)

        cand = _make_candidate_extended(n, p, off, beta, gamma)
        cand_score = _max_conv(cand)

        if cand_score < best_score:
            best_score = cand_score
            best_seq = cand
            print(f"[sweep] new best {best_score:.6f}  p={p:.3f} α={off:.3f} β={beta:.3f} γ={gamma:.3f}")

        if best_score < 1.503:       # target reached – early exit
            break

    # --------------------------------------------------------------
    # 3. Projected Polyak sub‑gradient descent (majority of the budget)
    # --------------------------------------------------------------
    # Hyper‑parameters for the Polyak step
    step_factor = 1.0
    min_step = 1e-6
    max_step = 0.05

    # sub‑gradient loop runs in “chunks”; after each chunk we may restart
    chunk_fraction = 0.10        # each chunk gets ≈10 % of remaining time
    iteration = 0
    while time.time() < deadline:
        # ------------------------------------------------------------------
        # (a) Choose a starting point for this chunk.
        #     With high probability we continue from the current best;
        #     occasionally we restart from a fresh random candidate.
        # ------------------------------------------------------------------
        if best_seq is not None and np.random.rand() < 0.7:
            a = best_seq.copy()
        else:
            a = _make_candidate_extended(
                n,
                p=np.random.uniform(0.30, 0.80),
                offset=np.random.uniform(0.0, 15.0),
                beta=np.random.uniform(-1.0, 3.0),
                gamma=np.random.uniform(0.0, 5.0),
            )

        # ------------------------------------------------------------------
        # (b) Run a chunk of sub‑gradient iterations.
        # ------------------------------------------------------------------
        chunk_deadline = min(deadline, time.time() + chunk_fraction * (deadline - time.time()))
        while time.time() < chunk_deadline:
            iteration += 1

            conv_full = _full_convolution_fft(a)
            M = conv_full.max()

            # Record a new global best if we have improved.
            if M < best_score:
                best_score = M
                best_seq = a.copy()
                print(f"[sgd] iter {iteration:6d}  new best {best_score:.6f}")
                if best_score < 1.503:
                    # target reached – stop everything
                    print("[sgd] target reached, terminating early")
                    return best_seq.tolist()

            # Identify all indices attaining (essentially) the maximum.
            eps = 1e-12 * M if M != 0 else 1e-12
            max_inds = np.where(conv_full >= M - eps)[0]

            # Polyak sub‑gradient step.
            a = _subgradient_step_polyak(
                a,
                max_inds,
                target_sum,
                lower_conv_bound,
                step_factor,
                min_step,
                max_step,
            )

            # Small jitter every few hundred iterations to escape plateaus.
            if iteration % 150 == 0 and np.random.rand() < 0.25:
                jitter = (np.random.rand(n) - 0.5) * 0.02 * a
                a = a + jitter
                a = _project_simplex(a, target_sum)

    # ----------------------------------------------------------------------
    # 4. Return the best sequence discovered.
    # ----------------------------------------------------------------------
    print(f"Finished – best score {best_score:.6f} after {iteration} iterations")
    return best_seq.tolist()
