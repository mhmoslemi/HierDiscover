        max_val = conv.max()

        w_sum = w.sum()
        if w_sum > 0:

        else:

"""
Optimise a non‑negative step‑function to minimise

    evaluate_sequence(seq) = 2·n·max(conv(seq,seq)) / (sum(seq)**2)

The scale of the sequence does not matter, so we keep the sum fixed to 1.
The problem is: minimise the peak of the self‑convolution subject to a_i≥0,
∑a_i = 1.

Our approach is a two‑phase optimiser:

1. **Linear‑programming polishing**  
   For a current vector `f` we compute M = max(f∗f).  
   We then solve the LP

        maximise   1ᵀ b
        subject to b ≥ 0
                   f ∗ b ≤ M   (element‑wise)

   The LP finds the “most massive’’ distribution that does not raise the peak
   above the current value M.  After scaling `b` to have sum = 1 we move a
   small step towards it:  f←(1‑t) f + t g.  Repeating this (t is reduced by a
   line‑search) strictly reduces the peak.  With a modest size (n≈1000) the
   LP is tiny and solved in a few hundred milliseconds.

2. **Smooth sub‑gradient refinement**  
   Once the LP phase stalls we finish with a few thousand iterations of a
   smoothed max‑gradient descent (soft‑max weights) and occasional random
   jitter.  This cleans up remaining plateaux.

Both phases respect the 1000 s wall‑clock budget and keep the best sequence
found.  If for any reason cvxpy is unavailable the code falls back to the
sub‑gradient method alone.

The returned list can be fed directly to `evaluate_sequence`.
"""

# Optional import of cvxpy (used for the LP phase)

except Exception:  # pragma: no cover
    _CVX_AVAILABLE = False

def project_onto_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the probability simplex
    {x | x≥0, ∑x = 1}.
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def full_convolution(a: np.ndarray) -> np.ndarray:
    """
    Full linear convolution a * a.  For the modest lengths used (≤2000)
    the plain NumPy implementation is fast enough.
    """
    return np.convolve(a, a)


def solve_lp_max_sum(f: np.ndarray, M: float) -> np.ndarray:
    """
    Solve  maximise   1ᵀ b
           subject to   b ≥ 0,
                        f ∗ b ≤ M   (element‑wise)

    Returns the optimal b (unnormalised).  If cvxpy is not available or the
    solver fails, a uniform vector is returned.
    """
    n = f.shape[0]

    # Build the Toeplitz matrix representing the convolution f ∗ b.
    # Row k, column i contains f[k‑i] if the index is valid, else 0.
    T = np.zeros((2 * n - 1, n), dtype=np.float64)
    for i in range(n):
        T[i : i + n, i] = f

    # Linear program with cvxpy
    b = cp.Variable(n)
    constraints = [b >= 0, T @ b <= M * np.ones(2 * n - 1)]
    prob = cp.Problem(cp.Maximize(cp.sum(b)), constraints)
    try:
        prob.solve(solver=cp.ECOS, warm_start=True, max_iters=5000)
    except Exception:  # pragma: no cover
        return np.full(n, 1.0 / n)

    if b.value is None:
        # fallback – uniform distribution
        return np.full(n, 1.0 / n)

    b_opt = np.array(b.value, dtype=np.float64).flatten()
    b_opt = np.maximum(b_opt, 0.0)      # numerical safety
    return b_opt


def smooth_max_weights(conv: np.ndarray, tau: float) -> np.ndarray:
    """
    Compute soft‑max weights w_i = exp(conv_i / tau) / Σ exp(conv / tau).
    The parameter tau controls the smoothness (smaller → sharper).
    """
    # Guard against overflow/underflow
    shifted = conv - np.max(conv)          # improve conditioning
    exp_vals = np.exp(shifted / tau)
    total = exp_vals.sum()
    if total == 0.0:
        return np.full_like(conv, 1.0 / conv.size)
    return exp_vals / total


def grad_from_weights(a: np.ndarray, w: np.ndarray) -> np.ndarray:
    """
    Gradient of the weighted sum Σ_k w_k (a∗a)_k w.r.t. a.
    Using the identity ∂(a∗a)_k / ∂a_i = 2·a_{k-i}.
    The computation is done via one FFT‑based convolution.
    """
    n = a.shape[0]
    # reverse weights for the cross‑correlation
    w_rev = w[::-1]
    size = 3 * n - 2
    fft_len = 1 << (size - 1).bit_length()
    fft_a = np.fft.rfft(a, n=fft_len)
    fft_w = np.fft.rfft(w_rev, n=fft_len)
    conv = np.fft.irfft(fft_a * fft_w, n=fft_len)[: size]
    # Gradient entry i corresponds to conv[2*n-2 - i] (see analysis)
    grad = 2.0 * conv[2 * n - 2 - np.arange(n)]
    return grad

                if i != j:

            if score < best_score:

                print(f"[SGD] n={n} it {sgd_it:04d}  score={score:.6f}")

                    print("[SGD] target reached – exiting early")
                    return best_seq.tolist()

        # end of current length – keep the best found so far

    return best_seq.tolist()
