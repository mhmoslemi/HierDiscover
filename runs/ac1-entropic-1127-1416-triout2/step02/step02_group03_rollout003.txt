def full_convolution_fft(a):
    n = len(a)
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return conv

Similarly for convolution of arbitrary arrays w and a_rev, we implement:

def conv_fft_any(x, y):
    n = len(x) + len(y) - 1
    size = 1 << ((n).bit_length())
    fx = np.fft.rfft(x, n=size)
    fy = np.fft.rfft(y, n=size)
    conv = np.fft.irfft(fx * fy, n=size)[:n]
    return conv

"""
Propose a non‑negative step function with a very low evaluation value.

Algorithm overview
------------------
1. **Parameterisation** – The vector `a` lives on the simplex
   `{a_i ≥ 0, Σ a_i = √(2·n)}`.  We write
           a = S·softmax(y)   (S = √(2·n))
   with an unconstrained vector `y`.  This guarantees the constraints
   automatically and allows us to use first‑order optimisation.

2. **Smooth surrogate of the max‑convolution** – The true objective is
   `max_k (a∗a)[k]`.  We minimise the smooth proxy
           f_λ(a) = (1/λ)·log Σ_k exp(λ·(a∗a)[k])                (λ>0)
   whose gradient is easy to compute:
           ∂f/∂a_i = 2· Σ_k w_k·a_{k‑i},
   where `w_k = exp(λ·(conv_k−max))/Σ exp(·)` are soft‑max weights.
   The gradient w.r.t. `y` follows from the chain rule:
           ∂f/∂y_i = a_i·(∂f/∂a_i − (∂f/∂a·a)/S).

3. **Adam optimisation** – We run Adam on `y` with an annealing schedule
   for `λ` (slow at the beginning, large at the end).  The loop stops
   when the time budget is exhausted or when a target score (`≈1.503`)
   is reached.

4. **Final refinement** – In the last few seconds we switch to the
   classic projected sub‑gradient method (the one shown in the
   description) to polish the best sequence found.

All FFT‑based convolutions use `numpy.fft.rfft` for speed.
The routine respects the 1000 s time limit and returns the best sequence
as a plain Python list.
"""

import time

def _full_convolution_fft(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a using a real FFT."""
    n = len(a)
    size = 1 << ((2 * n - 1).bit_length())          # next power of two
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return conv


def _conv_fft_any(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """
    Convolution of two real vectors using a real FFT.
    Returns the full linear convolution (len(x)+len(y)-1).
    """
    n = len(x) + len(y) - 1
    size = 1 << (n.bit_length())
    fx = np.fft.rfft(x, n=size)
    fy = np.fft.rfft(y, n=size)
    conv = np.fft.irfft(fx * fy, n=size)[:n]
    return conv


def _project_simplex(v: np.ndarray, target_sum: float) -> np.ndarray:
    """
    Euclidean projection of a vector onto the simplex
        { w | w_i ≥ 0, Σ w_i = target_sum }.
    Implements the algorithm of Duchi et al. (2008).
    """
    if target_sum < 0:
        raise ValueError("target_sum must be non‑negative")
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - target_sum
    ind = np.arange(1, len(v) + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.zeros_like(v)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def _make_candidate(n: int, p: float, offset: float, beta: float) -> np.ndarray:
    """
    Produce a power‑law / tilted candidate and scale it to the simplex.
    Formula:
        a_i = (i+1+offset)^{-p} * (1 + beta * i/(n‑1)).
    """
    idx = np.arange(n, dtype=float)
    base = np.power(idx + 1.0 + offset, -p)
    factor = 1.0 + beta * idx / (n - 1)
    raw = base * factor
    raw = np.maximum(raw, 0.0)
    target = np.sqrt(2.0 * n)
    # Scale to the correct ℓ₁‑norm (simple scaling is enough here)
    raw = raw * (target / raw.sum())
    return raw


def _subgradient_step(a: np.ndarray, max_inds: np.ndarray,
                      step: float, target_sum: float) -> np.ndarray:
    """
    One projected sub‑gradient step for the max‑convolution objective.
    The sub‑gradient is built from the indices where the current maximum
    is attained.
    """
    n = len(a)
    g = np.zeros_like(a)

    for k in max_inds:
        if k < n:                     # left half of the convolution
            g[: k + 1] += a[k :: -1]
        else:                         # right half
            start = k - (n - 1)
            idx = np.arange(start, n)
            g[start:] += a[k - idx]

    cand = a - step * g
    cand = _project_simplex(cand, target_sum)
    return cand

        # ---- Adam update ----
        m = beta1 * m + (1 - beta1) * grad_y
        v = beta2 * v + (1 - beta2) * (grad_y ** 2)
        m_hat = m / (1 - beta1 ** t)
        v_hat = v / (1 - beta2 ** t)
        y -= lr * m_hat / (np.sqrt(v_hat) + adam_eps)

        # ---- occasional jitter to avoid plateaus ----
        if (t % 500 == 0) and (np.random.rand() < 0.2):
            y += np.random.randn(n) * 0.001

    # ------------------------------------------------------------------
    # 4. Final polishing using the classic projected sub‑gradient method.
    #    We spend the remaining few seconds on this phase.
    # ------------------------------------------------------------------
    remaining = deadline - time.time()
    if remaining > 5.0:                     # give it at least a few seconds
        print("[refine] entering sub‑gradient polishing phase")
        a = best_seq.copy()
        step0 = 1.0
        it = 0
        # run until the (short) deadline
        while time.time() < deadline:
            it += 1
            conv_full = _full_convolution_fft(a)
            M = conv_full.max()
            if M < best_score:
                best_score = M
                best_seq = a.copy()
                print(f"[sgd]  it {it:6d}  new best {best_score:.6f}")
                if best_score < 1.503:
                    break
            eps_tol = 1e-12 * M if M != 0 else 1e-12
            max_inds = np.where(conv_full >= M - eps_tol)[0]
            step = step0 / np.sqrt(it)
            a = _subgradient_step(a, max_inds, step, target_sum)

            # occasional tiny random perturbation
            if it % 100 == 0 and np.random.rand() < 0.15:
                jitter = (np.random.rand(n) - 0.5) * 0.02 * a
                a = _project_simplex(a + jitter, target_sum)

    # ------------------------------------------------------------------
    # 5. Return the best sequence found.
    # ------------------------------------------------------------------
    print(f"Finished – best score {best_score:.6f}")
    return best_seq.tolist()
