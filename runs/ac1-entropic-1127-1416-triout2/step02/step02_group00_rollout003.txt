"""
Hybrid optimiser for the max‑self‑convolution problem.

The routine builds on three ideas:

1. **Fast parametric scan** – a two‑parameter symmetric family
   a_i ∝ ((i+ε)(n‑1‑i+ε))⁻ᵖ (p>0, ε>0).  A few random draws give good
   starting points for each length n.

2. **Smoothed max gradient descent** – minimise a smooth approximation of
   the maximal convolution value by Adam on the probability simplex.
   The soft‑max (log‑sum‑exp) with an annealing schedule for γ turns the
   non‑smooth max into a differentiable objective; gradients are obtained
   via a single FFT‑based convolution.

3. **Final sub‑gradient polishing** – the classic projected sub‑gradient
   optimiser is run for a few thousand iterations to fine‑tune the best
   candidate.

All helper functions are top‑level, use NumPy only and respect the
time‑budget (≈1000 s).  The returned list is a plain ``list[float]`` ready
for the official ``evaluate_sequence`` call.
"""

import time
from typing import List, Tuple

import numpy as np

# ----------------------------------------------------------------------
# Basic utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """
    Real‑FFT convolution of two 1‑D arrays.
    Returns the full discrete convolution (length len(a)+len(b)-1).
    """
    m = len(a) + len(b) - 1
    n_fft = next_pow2(m)
    A = np.fft.rfft(a, n=n_fft)
    B = np.fft.rfft(b, n=n_fft)
    C = A * B
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the probability simplex
    {x | x_i ≥ 0, Σ_i x_i = 1}.
    Duchi et al. (2008) implementation.
    """
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / len(v))

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho = np.where(u - cssv / np.arange(1, len(v) + 1) > 0)[0]
    if len(rho) == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def compute_score(a: np.ndarray) -> float:
    """
    Fast surrogate for the official evaluator (scale‑invariant):
        2·n·max(a*a) / (sum(a)²)
    The input is assumed to be a probability vector (sum≈1).
    """
    n = len(a)
    conv = convolution_fft(a, a)
    max_val = float(np.max(conv))
    s = float(np.sum(a))
    if s < 1e-12:
        return float("inf")
    return 2.0 * n * max_val / (s * s)

def arcsine_initial(n: int) -> np.ndarray:
    """
    Classic arcsine‑shaped start:
        a_i ∝ 1/√((i+½)*(n‑½‑i)).
    Normalised to sum = 1.
    """
    i = np.arange(n, dtype=float)
    eps = 0.25
    vals = 1.0 / np.sqrt((i + 0.5) * (n - 0.5 - i) + eps)
    return vals / vals.sum()


def generate_shape(n: int, p: float, eps: float) -> np.ndarray:
    """
    Symmetric density a_i ∝ ((i+eps)*(n‑1‑i+eps))⁻ᵖ.
    Returned vector is normalised to sum = 1.
    """
    i = np.arange(n, dtype=float)
    base = ((i + eps) * (n - 1 - i + eps)) ** (-p)
    total = base.sum()
    if total == 0.0:
        return np.full(n, 1.0 / n)
    return base / total

# Smoothed max‑convolution gradient (Adam optimiser)
# ----------------------------------------------------------------------


def _grad_from_softmax_weights(w: np.ndarray, a: np.ndarray) -> np.ndarray:
    """
    Given soft‑max weights w_k (k=0..2n‑2) and current a,
    compute the gradient of Σ_k w_k·(a*a)_k w.r.t. a.
    Implements the formula
        grad_i = Σ_j w_{i+j} a_j  +  w_{2i}·a_i.
    """
    n = len(a)
    # reverse w for the convolution trick
    w_rev = w[::-1]
    conv = convolution_fft(w_rev, a)            # length = 3n‑2
    # extract slice n‑1 .. 2n‑1 (exclusive) and reverse
    base = conv[n - 1 : 2 * n - 1][::-1]        # shape (n,)
    # diagonal extra term
    extra = np.zeros_like(a)
    diag_idx = 2 * np.arange(n)
    mask = diag_idx < len(w)
    extra[mask] = w[diag_idx[mask]] * a[mask]
    return base + extra


def smoothed_descent(
    initial_a: np.ndarray,
    time_deadline: float,
    max_iter: int = 200_000,
    gamma_start: float = 0.5,
    gamma_end: float = 60.0,
    lr: float = 0.15,
    beta1: float = 0.9,
    beta2: float = 0.999,
    eps: float = 1e-8,
    eval_every: int = 200,
) -> Tuple[np.ndarray, float]:
    """
    Adam optimiser on the log‑sum‑exp approximation of the maximal
    self‑convolution.  The iterates are kept on the probability simplex.
    Returns the best vector seen (according to ``compute_score``) and its
    surrogate score.
    """
    a = project_to_simplex(initial_a.astype(float).copy())
    best_a = a.copy()
    best_sc = compute_score(a)

    m = np.zeros_like(a)
    v = np.zeros_like(a)

    n = len(a)

    for it in range(1, max_iter + 1):
        if time.time() >= time_deadline:
            break

        # exponential schedule for γ
        gamma = gamma_start * (gamma_end / gamma_start) ** (it / max_iter)

        # convolution a*a
        conv = convolution_fft(a, a)                     # length 2n‑1
        # soft‑max weights
        max_c = np.max(conv)
        shifted = gamma * (conv - max_c)
        exp_shift = np.exp(shifted)
        w = exp_shift / exp_shift.sum()

        # gradient of the smoothed objective
        grad = _grad_from_softmax_weights(w, a)

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)

        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        step = lr * m_hat / (np.sqrt(v_hat) + eps)
        a_new = a - step
        a_new = project_to_simplex(a_new)

        # occasional tiny jitter to escape flat regions
        if it % 3000 == 0:
            a_new += np.random.randn(n) * 5e-5
            a_new = project_to_simplex(a_new)

        a = a_new

        # periodic exact evaluation
        if it % eval_every == 0:
            sc = compute_score(a)
            if sc < best_sc:
                best_sc = sc
                best_a = a.copy()
        # light progress report
        if it % 5000 == 0:
            print(
                f"    [smoothed] iter={it:,d}  cur≈{compute_score(a):.6f}  best≈{best_sc:.6f}"
            )
    return best_a, best_sc

def max_convolution_and_index(a: np.ndarray) -> Tuple[float, int]:
    """Return (maximum value, index) of a*a."""
    conv = convolution_fft(a, a)
    idx = int(np.argmax(conv))
    return float(conv[idx]), idx


def subgradient_of_max_at_index(a: np.ndarray, k: int) -> np.ndarray:
    """
    Sub‑gradient of the maximal convolution entry (index k) w.r.t. a.
    Handles the doubled diagonal term when k = 2·i.
    """
    n = len(a)
    grad = np.zeros_like(a)
    i = np.arange(n)
    j = k - i
    mask = (j >= 0) & (j < n)
    grad[mask] = a[j[mask]]
    if k % 2 == 0:
        diag = k // 2
        if diag < n:
            grad[diag] = 2.0 * a[diag]
    return grad


def improve_sequence(
    initial_a: np.ndarray,
    time_deadline: float,
    max_iter: int = 200_000,
    eta0: float = 0.5,
    noise_every: int = 500,
    noise_scale: float = 1e-4,
) -> Tuple[np.ndarray, float]:
    """
    Classic projected sub‑gradient descent on the exact max‑convolution
    objective.  Returns the best vector seen and its exact surrogate score.
    """
    a = project_to_simplex(initial_a.astype(float).copy())
    best_a = a.copy()
    best_sc = compute_score(a)

    n = len(a)
    it = 0
    while time.time() < time_deadline and it < max_iter:
        it += 1
        _, k_max = max_convolution_and_index(a)
        grad = subgradient_of_max_at_index(a, k_max)

        eta = eta0 / np.sqrt(it)

        a_new = a - eta * grad
        if noise_every and (it % noise_every == 0):
            a_new += np.random.randn(n) * noise_scale
        a_new = project_to_simplex(a_new)

        cur_sc = compute_score(a_new)
        if cur_sc < best_sc:
            best_sc = cur_sc
            best_a = a_new.copy()
        a = a_new

        if it % 5000 == 0:
            print(
                f"    [subgrad] iter={it:,d}  cur={cur_sc:.6f}  best={best_sc:.6f}"
            )
    return best_a, best_sc

        best_val = evaluate_sequence(best_seq)

    print(f"\nFinished search. Best reported score = {best_val:.6f}")
    return best_seq
