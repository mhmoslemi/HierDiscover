def conv_max(a):
    n = len(a)
    size = 1 << ((2*n -1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2*n-1]
    return conv.max()

def compute_subgradient(a, max_inds):
    n = len(a)
    g = np.zeros_like(a)
    for k in max_inds:
        if k < n:
            g[:k+1] += a[k::-1]
        else:
            start = k - (n-1)
            g[start:] += a[start:][::-1]
    return g

def subgradient_descent(a0, target_sum, deadline):
    a = a0.copy()
    v = np.zeros_like(a)
    mu = 0.9
    step0 = 5.0  # initial step magnitude (tuned)
    iteration = 0
    best_seq = a.copy()

    while time.time() < deadline:

        M = conv_full.max()
        if M < best_score:
            best_score = M
            best_seq = a.copy()

            if best_score < 1.503:
                break
        eps = 1e-12 * M
        max_inds = np.where(conv_full >= M - eps)[0]

        step = step0 / (iteration ** 0.3)   # slower decay

"""
search for a non‑negative step function that minimises the evaluation
score  2·n·max(conv(a,a)) / (sum(a)**2)

Because we always keep Σa = √(2·n), the score is simply
max (conv(a,a)).  The algorithm consists of two phases

1️⃣ GLOBAL SEARCH
   – Randomly generate candidates from a flexible 5‑parameter power‑law
     family (α, p, β₁, β₂) and from a smoothed‑random construction.
   – Each candidate is projected onto the simplex
     {x ≥ 0, Σx = √(2·n)} and its maximal self‑convolution is evaluated
     with an FFT (O(n log n)).
   – The best candidate seen so far is kept.

2️⃣ LOCAL REFINEMENT
   – Starting from the best global candidate we run a momentum
     sub‑gradient descent on the objective max(conv(a,a)).
   – For the active indices K (where the convolution attains its current
     maximum) the sub‑gradient is
         g_i = Σ_{k∈K} a_{k‑i}
     (out‑of‑range terms are ignored).
   – The update uses a classic momentum scheme
         v ← μ·v – η·g ,   a ← a + v
     followed by a Euclidean projection onto the simplex.
   – The step size η decays like t⁻⁰·³, the momentum μ = 0.9.
   – Small smoothing and random jitter are added occasionally to
     avoid stagnation.

The whole routine respects a 1000 s wall‑clock budget, prints
intermediate scores and finally returns the best sequence (as
plain Python list) that it found.
"""

import time
import numpy as np


# ----------------------------------------------------------------------
# 0.  Basic utilities
# ----------------------------------------------------------------------


def _project_simplex(v: np.ndarray, target_sum: float) -> np.ndarray:
    """Euclidean projection of a vector onto the simplex
    { w ≥ 0, Σ w = target_sum } (Duchi et al. 2008)."""
    if target_sum < 0:
        raise ValueError("target_sum must be non‑negative")
    # work on a copy
    u = np.sort(v)[::-1]                     # descending
    cssv = np.cumsum(u) - target_sum
    ind = np.arange(1, len(v) + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):                     # all zero – should not happen
        return np.zeros_like(v)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def _full_convolution_fft(a: np.ndarray) -> np.ndarray:
    """Linear convolution a*a via a real FFT.  Length = 2·len(a)‑1."""
    n = len(a)
    size = 1 << ((2 * n - 1).bit_length())   # next power of two
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return conv


def _max_conv(a: np.ndarray) -> float:
    """Maximum entry of the self‑convolution of a."""
    return _full_convolution_fft(a).max()

def _make_candidate_extended(
    n: int,
    p: float,
    offset: float,
    beta1: float,
    beta2: float,
) -> np.ndarray:
    """Power‑law with linear + quadratic tilt, projected onto the simplex."""
    idx = np.arange(n, dtype=float)                     # i = 0 … n‑1
    base = np.power(idx + 1.0 + offset, -p)
    tilt = (1.0
            + beta1 * idx / (n - 1)
            + beta2 * (idx / (n - 1)) ** 2)
    raw = base * tilt
    raw = np.maximum(raw, 0.0)
    target = np.sqrt(2.0 * n)
    return _project_simplex(raw, target)


def _make_candidate_smoothed_random(n: int) -> np.ndarray:
    """Produce a positive smooth random vector and project it."""

    raw = np.maximum(raw, 1e-12)        # avoid exact zeros
    target = np.sqrt(2.0 * n)
    return _project_simplex(raw, target)

def _subgradient(a: np.ndarray, max_inds: np.ndarray) -> np.ndarray:
    """Compute a sub‑gradient of max(conv(a,a)) at the active indices."""
    n = len(a)
    g = np.zeros_like(a)
    for k in max_inds:
        if k < n:
            # indices 0 … k
            g[: k + 1] += a[k::-1]           # a[k], a[k‑1], … , a[0]
        else:
            # indices k‑(n‑1) … n‑1
            start = k - (n - 1)
            g[start:] += a[start:][::-1]     # reversed slice
    return g
