"""
Propose a low‑score step‑function using a smooth‑max gradient descent.

The code works with a fixed ℓ₁‑norm S = √(2 n).  Under this
normalisation the evaluation function reduces to the peak of the
self‑convolution, i.e.  score = max_k (a∗a)[k].  We minimise a smooth
approximation of this peak:

    f(a) = (1/α)·log Σ_k exp(α·c_k)   where c = a∗a

for a temperature α that is increased during the run (continuation).
The gradient of f is a simple convolution of the current sequence with a
soft‑max weight derived from the convolution values, hence each iteration
costs only three FFTs (self‑convolution, weight‑convolution and the line‑search).

After every gradient step we project back onto the simplex
{ a ≥ 0 , Σa = √(2 n) } using the O(n log n) Euclidean projection.
Occasional jitter and an adaptive step‑size keep the search away from
poor local minima.  The best vector seen so far is returned; a final
call to ``evaluate_sequence`` verifies the official score.

The routine respects the 1000 s budget and uses only NumPy (no
filesystem or network access).
"""

import time
import numpy as np

# --------------------------------------------------------------------------- #
# Simplex projection (Euclidean, Duchi‑et‑al. 2008)
def projection_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Project vector v onto the simplex {x ≥ 0, Σx = s}."""
    if s < 0:
        raise ValueError("simplex sum must be non‑negative")
    n = v.shape[0]
    # sort in descending order
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.where(u - cssv / (np.arange(n) + 1.0) > 0)[0]
    if rho.size == 0:                     # all entries are zero
        return np.zeros_like(v)
    rho = rho[-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# --------------------------------------------------------------------------- #
def _next_pow2(x: int) -> int:
    """Return the smallest power of two >= x."""
    return 1 << ((x - 1).bit_length())


# --------------------------------------------------------------------------- #
def propose_candidate(seed: int = 42, budget_s: float = 1000) -> list[float]:
    """
    Search for a non‑negative step‑function (vector) that minimises
    ``evaluate_sequence``.  The routine runs for at most ``budget_s`` seconds
    and returns the best sequence found.
    """
    np.random.seed(seed)
    start = time.time()
    deadline = start + budget_s - 0.5          # small safety margin

    # ------------------------------------------------------------------- #
    # Problem size – a few‑thousand entries give a good trade‑off.
    # Larger n brings the theoretical lower bound (≈ 1) closer,
    # while still keeping the per‑iteration cost modest.
    n = 8000
    target_sum = np.sqrt(2.0 * n)              # ℓ₁‑norm that makes the score = max‑conv

    # FFT lengths (next powers of two for efficiency)
    conv_len = 2 * n - 1
    fft_len_conv = _next_pow2(conv_len)        # for a∗a
    grad_len = n + conv_len - 1                # = 3 n‑2
    fft_len_grad = _next_pow2(grad_len)        # for gradient convolution

    # ------------------------------------------------------------------- #
    # Seed: a random positive vector projected onto the simplex.
    a = np.random.rand(n)
    a = projection_simplex(a, target_sum)

    # ------------------------------------------------------------------- #
    # Helper to compute the (full) convolution a∗a via FFT.
    def conv_self(x: np.ndarray) -> np.ndarray:
        F = np.fft.rfft(x, n=fft_len_conv)
        c = np.fft.irfft(F * F, n=fft_len_conv)[:conv_len]
        return c

    # ------------------------------------------------------------------- #
    # State tracking
    best = a.copy()
    best_conv = conv_self(best)
    best_score = best_conv.max()
    print(f"[{time.time() - start:6.2f}s] initial score (max conv) = {best_score:.6f}")

    # Gradient‑descent parameters
    step = target_sum * 0.5 / n                # initial absolute step
    min_step = target_sum * 1e-7 / n
    momentum = 0.0                              # plain gradient (no momentum)
    alpha = 0.6                                 # temperature for smooth‑max
    alpha_max = 30.0
    iter_cnt = 0
    no_improve = 0
    max_no_improve = 2000

    # ------------------------------------------------------------------- #
    while time.time() < deadline:
        # ----- current convolution and score --------------------------------
        conv = conv_self(a)
        cur_max = conv.max()

        # record best
        if cur_max < best_score - 1e-12:
            best_score = cur_max
            best = a.copy()
            no_improve = 0
            print(f"[{time.time() - start:6.2f}s] iter {iter_cnt:6d} → new best {best_score:.6f}")

        else:
            no_improve += 1

        # ----- smooth‑max weights -------------------------------------------
        # w_i = exp(α·(c_i – max_c)), max weight = 1, all ≤ 1
        w_raw = np.exp(alpha * (conv - cur_max))
        sum_w = w_raw.sum()
        if sum_w == 0.0:                 # should never happen, but be safe
            sum_w = 1.0
        w_rev = w_raw[::-1]              # reversed for the gradient convolution

        # ----- gradient of the smooth‑max ------------------------------------
        # g = 2 * (w_rev ∗ a)[(2n‑2)‑j] / Σw   (j = 0,…,n‑1)
        Ga = np.fft.rfft(a, n=fft_len_grad)

        grad_full = np.fft.irfft(Ga * Gw, n=fft_len_grad)[:grad_len]

    print(f"Best peak of convolution  : {best_score:.6f}")
    print(f"Official ``evaluate_sequence`` score : {final_score:.6f}")

    return best.tolist()
