def propose_candidate(seed=123, budget_s=1000, **kwargs):
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5

    # Choose length
    n = 20000
    target_sum = np.sqrt(2.0 * n)

    # Initialize sequence
    best_seq = None
    # try existing height_sequence_1
    hs = globals().get("height_sequence_1")
    if isinstance(hs, list) and len(hs) > 0:
        old = np.array(hs, dtype=float)
        if len(old) <= n:
            extra = n - len(old)

            # project onto simplex target sum
            best_seq = _project_simplex(cand, target_sum)
        else:
            # truncate
            cand = old[:n]
            best_seq = _project_simplex(cand, target_sum)
    else:
        # fallback power-law
        best_seq = _make_candidate(n, p=0.5, offset=0.0, beta=0.0)

    # Random sweep (30% time)
    sweep_end = time.time() + 0.30 * (deadline - time.time())
    best_score = _max_conv(best_seq)  # conv max equals score
    while time.time() < sweep_end:
        p = np.random.uniform(0.35, 0.65)
        off = np.random.uniform(0.0, 15.0)
        beta = np.random.uniform(-0.9, 2.5)
        cand = _make_candidate(n, p, off, beta)
        cand_score = _max_conv(cand)
        if cand_score < best_score:
            best_score = cand_score
            best_seq = cand
            print(f"[sweep] new best {best_score:.6f} p={p:.3f} α={off:.3f} β={beta:.3f}")
        if best_score < 1.45:
            break

    # Smoothed projected gradient descent
    a = best_seq.copy()
    v = np.zeros_like(a)   # momentum
    step0 = 1.0
    beta_m = 0.9
    tau0_factor = 0.5  # initial temperature factor relative to current max
    iter_count = 0
    size_big = 1 << ((3*n - 2).bit_length())
    target_sum = np.sqrt(2.0 * n)

    while time.time() < deadline:
        iter_count += 1
        FA = np.fft.rfft(a, n=size_big)
        conv_full = np.fft.irfft(FA * FA, n=size_big)[: 2*n - 1]
        M = conv_full.max()
        if M < best_score:
            best_score = M
            best_seq = a.copy()
            print(f"[sgd] iter {iter_count:6d}  new best {best_score:.6f}")
            if best_score < 1.45:
                break

        # Temperature schedule
        tau = max(M * tau0_factor * np.exp(-iter_count / 5000.0), 1e-6)

        # Compute softmax weights
        shift = (conv_full - M) / tau  # <=0
        # Use np.exp safely

"""
Fast smoothed projected gradient descent for the self‑convolution L∞ problem.

The algorithm works as follows

1.  Choose a length `n` (default 20 000).  The admissible sum is
    `target_sum = sqrt(2·n)`.  With this normalisation the evaluation
    score returned by ``evaluate_sequence`` equals the maximal entry of
    the discrete convolution `a*a`.

2.  Initialise a feasible sequence:
    * If a global variable ``height_sequence_1`` is present it is
      padded (or truncated) to length `n` and projected onto the simplex.
    * Otherwise a simple power‑law candidate is used.

3.  A short random sweep over the 4‑parameter family  

       a[i] = (i+1+α)^(-p)·(1+β·i/(n‑1))

    searches for a better starting point.

4.  Main optimisation loop (until the time budget expires)
       * Compute the full convolution `c = a*a` with a real FFT.
       * Let `M = max(c)`.  The current score is `M`.
       * Build a *softmax* weight vector  
            w[k] ∝ exp((c[k]‑M)/τ) ,   τ>0.
         For large τ the gradient uses information from all shifts,
         while τ→0 reproduces the usual sub‑gradient that only sees the
         active indices.
       * The gradient of the weighted sum `∑_k w_k·(a*a)[k]` equals
         `2·corr(a,w)`, where `corr` is the linear correlation.
         The correlation is obtained again by FFTs:
            g = 2·ifft( fft(a)·conj(fft(w)) ).
       * Apply momentum (`v←β·v+(1‑β)·g`) and a decaying step size
         `η = η₀/√t`.  The new iterate is projected back onto the simplex.
         Tiny random jitter is occasionally added to escape plateaus.

5.  The best sequence seen during the run is returned as a plain Python
    list.  Because the sum is kept exactly `sqrt(2·n)`, the harness
    evaluates the score as the maximal convolution value.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Return the smallest power of two >= x."""
    return 1 << (x - 1).bit_length()


def _project_simplex(v: np.ndarray, target_sum: float) -> np.ndarray:
    """
    Euclidean projection of a vector onto the simplex
        { w | w_i ≥ 0, Σ w_i = target_sum }.
    Implements the algorithm of Duchi et al. (2008).
    """
    if target_sum < 0:
        raise ValueError("target_sum must be non‑negative")
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - target_sum
    ind = np.arange(1, len(v) + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.zeros_like(v)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def _full_convolution_fft(a: np.ndarray) -> np.ndarray:
    """
    Compute the linear convolution a*a using a real FFT.
    Result length is 2·len(a)‑1.
    """
    n = len(a)
    size = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return conv


def _max_conv(a: np.ndarray) -> float:
    """Return max_k (a*a)[k] via FFT."""
    return _full_convolution_fft(a).max()


def _make_candidate(n: int, p: float, offset: float, beta: float) -> np.ndarray:
    """
    Build an un‑normalised raw sequence
        a_i = (i+1+offset)^(-p) * (1 + beta·i/(n‑1))
    and project it onto the simplex of sum sqrt(2·n).
    """
    idx = np.arange(n, dtype=float)
    base = np.power(idx + 1.0 + offset, -p)
    factor = 1.0 + beta * idx / (n - 1)
    raw = base * factor
    raw = np.maximum(raw, 0.0)          # keep non‑negative
    target = np.sqrt(2.0 * n)
    return _project_simplex(raw, target)


# ----------------------------------------------------------------------
# Main search routine
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 123, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step‑function with a low evaluation value.
    The routine respects a ~1000 s time limit and returns the best
    sequence found as a plain Python list.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 2.0   # safety margin

    # --------------------------------------------------------------
    # 0. Problem size
    # --------------------------------------------------------------
    n = 20000                              # larger n gives a tighter lower bound
    target_sum = np.sqrt(2.0 * n)

    # --------------------------------------------------------------
    # 1. Initial feasible point
    # --------------------------------------------------------------
    best_seq = None
    hs = globals().get("height_sequence_1")
    if isinstance(hs, list) and len(hs) > 0:
        old = np.array(hs, dtype=float)
        if len(old) <= n:
            extra = n - len(old)
            cand = np.concatenate([old, np.zeros(extra, dtype=float)])
        else:
            cand = old[:n]
        best_seq = _project_simplex(cand, target_sum)
    else:
        # a simple power‑law start
        best_seq = _make_candidate(n, p=0.5, offset=0.0, beta=0.0)

    # --------------------------------------------------------------
    # 2. Coarse random sweep over the 4‑parameter family
    # --------------------------------------------------------------
    best_score = _max_conv(best_seq)    # because Σa = √(2n) ⇒ score = max conv
    print(f"[init] n={n:,d}  score={best_score:.6f}")

    sweep_deadline = time.time() + 0.30 * (deadline - time.time())
    while time.time() < sweep_deadline:
        p = np.random.uniform(0.35, 0.65)                     # exponent
        off = np.random.uniform(0.0, 20.0)                    # offset α
        beta = np.random.uniform(-0.9, 2.5)                   # tilt β
        cand = _make_candidate(n, p, off, beta)
        cand_score = _max_conv(cand)
        if cand_score < best_score:
            best_score = cand_score
            best_seq = cand
            print(f"[sweep] new best {best_score:.6f}  p={p:.3f} α={off:.3f} β={beta:.3f}")
        if best_score < 1.45:                                 # early stop
            break

    # --------------------------------------------------------------
    # 3. Smoothed projected sub‑gradient descent
    # --------------------------------------------------------------
    a = best_seq.copy()
    v = np.zeros_like(a)                # momentum buffer
    step0 = 1.0                         # initial step size
    beta_m = 0.9                        # momentum coefficient
    tau_factor = 0.5                    # τ = τ_factor·M at iteration 0
    size_big = _next_pow2(3 * n - 2)    # enough for correlation (n + (2n‑1) -1)

    iteration = 0
    while time.time() < deadline:
        iteration += 1

        # ----- full self‑convolution (FFT) -----
        Fa = np.fft.rfft(a, n=size_big)
        conv_full = np.fft.irfft(Fa * Fa, n=size_big)[: 2 * n - 1]
        M = conv_full.max()

        # ----- keep the best ever -----
        if M < best_score:
            best_score = M
            best_seq = a.copy()
            print(f"[sgd] iter {iteration:6d}  new best {best_score:.6f}")
            if best_score < 1.45:               # stop if a good target is reached
                break

        # ----- temperature (soft‑max smoothing) -----
        tau = max(M * tau_factor * np.exp(-iteration / 5000.0), 1e-8)

        # ----- soft‑max weights over the convolution entries -----
        # shift <= 0  →  exp(shift/τ) ∈ (0,1]
        w = np.exp((conv_full - M) / tau)
        w_sum = w.sum()
        if w_sum == 0.0:                     # numerical underflow → fall back to hard active set
            active = np.where(conv_full >= M - 1e-12)[0]
            w = np.zeros_like(conv_full)
            w[active] = 1.0 / len(active)
        else:
            w /= w_sum

        # ----- gradient of the weighted sum via correlation (FFT) -----
        Fw = np.fft.rfft(w, n=size_big)
        grad = np.fft.irfft(Fa * np.conj(Fw), n=size_big)[:n]
        grad = 2.0 * grad                     # exact gradient of ∑ w_k (a*a)[k]

        # ----- momentum and step -----
        v = beta_m * v + (1.0 - beta_m) * grad
        step = step0 / np.sqrt(iteration)

        # ----- projected update -----
        a_new = a - step * v
        a_new = _project_simplex(a_new, target_sum)

        # ----- occasional jitter to escape plateaus -----
        if iteration % 100 == 0 and np.random.rand() < 0.3:
            jitter = (np.random.rand(n) - 0.5) * 0.02 * a_new
            a_new += jitter
            a_new = _project_simplex(a_new, target_sum)

        a = a_new

    print(f"Finished after {iteration} iterations – best score {best_score:.6f}")
    # Return the best sequence as a plain list of Python floats.
    return best_seq.tolist()
