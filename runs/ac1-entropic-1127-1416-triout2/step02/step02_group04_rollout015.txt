def mirror_descent(a0, max_iters, deadline):
    a = a0.copy()
    best_seq = a.copy()
    best_score = compute_objective(a)
    m1 = np.zeros_like(a) # first moment
    m2 = np.zeros_like(a) # second moment
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    for t in range(max_iters):
        if time.time() > deadline:
            break
        # compute convolution
        fft = np.fft.rfft(a, n=2*n-1)
        conv = np.fft.irfft(fft * fft, n=2*n-1)
        max_b = conv.max()
        cur_score = 2 * n * max_b
        if cur_score < best_score:
            best_score = cur_score
            best_seq = a.copy()
            print(...)
        # active
        eps_tol = max_b * 1e-6
        active_idx = np.where(conv >= max_b - eps_tol)[0]
        if active_idx.size == 0:
            active_idx = np.array([conv.argmax()], dtype=int)
        # compute gradient g (subgradient: sum over active)
        g = np.zeros_like(a)
        for k in active_idx:
            low = max(0, k - n + 1)
            high = min(n - 1, k)
            idx = np.arange(low, high+1)
            partner = k - idx
            g[idx] += 2.0 * a[partner]
        g /= float(active_idx.size)

        # Adam update
        m1 = beta1 * m1 + (1 - beta1) * g
        m2 = beta2 * m2 + (1 - beta2) * (g**2)
        m1_hat = m1 / (1 - beta1**(t+1))
        m2_hat = m2 / (1 - beta2**(t+1))
        step = eta / (np.sqrt(m2_hat) + eps)

        # mirror descent: a_new = a * np.exp(-step * m1_hat)
        a = a * np.exp(-step * m1_hat)
        # add small epsilon to avoid zeros
        a = np.maximum(a, 1e-15)
        a /= a.sum()
    return best_seq, best_score
```

"""
Propose a low‑value step‑function using a multi‑start
exponential‑gradient (mirror‑descent) optimiser.

The algorithm works as follows

1.  Choose a problem size `n`.  If the global variable
    ``height_sequence_1`` exists we reuse its length, otherwise we
    fall back to a modest default (≈ 5000).  Larger `n` gives a better
    theoretical lower bound, but also costs more per iteration.

2.  Build a small collection of *promising* initial distributions:
    – the historic best sequence (if available);
    – a decreasing linear ramp;
    – a uniform distribution;
    – a few classic window functions (Hann, Hamming, Blackman);
    – symmetric triangular / cosine / quadratic shapes;
    – a grid of Beta‑distribution shapes;
    – a handful of random Dirichlet‑like samples.
    All candidates are normalised to sum to one and the best one
    (according to the exact objective) is kept as the starting point.

3.  From that point a **mirror‑descent** (exponential‑gradient) loop is
    performed.  In each iteration:
        * compute the linear convolution `c = a*a` via FFT;
        * locate the *active* indices where `c` is within a tiny
          relative tolerance of its maximum;
        * build a sub‑gradient `g` of the max‑convolution:
              g_i = 2 / |active| · Σ_{k∈active} a_{k‑i}
          (the standard sub‑gradient for the ℓ_∞‑norm of a quadratic form);
        * take an exponential step   a_i ← a_i·exp(‑η·g_i)   with a
          diminishing learning rate η_t = η₀ / √(t+1);
        * re‑normalise to stay on the probability simplex.
          A tiny floor (1e‑15) guarantees positivity and avoids numerical
          under‑flow.
        * keep the best sequence ever seen.

    The loop stops when the time budget (`budget_s` seconds) is exhausted
    or a preset iteration limit is reached.

4.  The best sequence encountered is returned as a plain Python list of
    ``float``.  The function respects the 1000 s global timeout
    (a safety margin of a few seconds is kept).

The implementation is deliberately lightweight (no external LP solvers)
and relies only on NumPy, which makes it fast enough to explore many
iterations within the allowed budget.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities

def simplex_projection(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of a vector onto the probability simplex."""
    v = np.asarray(v, dtype=float)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1] - 1
    theta = cssv[cond][-1] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def compute_objective(a: np.ndarray) -> float:
    """Exact value of the evaluation function for a normalised vector (sum=1)."""
    n = a.shape[0]
    # linear convolution via FFT (length 2n‑1)
    fft = np.fft.rfft(a, n=2 * n - 1)
    conv = np.fft.irfft(fft * fft, n=2 * n - 1)
    return 2.0 * n * conv.max()


def beta_sequence(alpha: float, beta: float, n: int) -> np.ndarray:
    """Discretised beta(α,β) density on `n` points, normalised to sum‑one."""
    x = (np.arange(n) + 0.5) / n
    logw = (alpha - 1.0) * np.log(x) + (beta - 1.0) * np.log(1.0 - x)
    logw = np.clip(logw, -700.0, 700.0)          # avoid overflow in exp
    w = np.exp(logw)
    if w.sum() == 0 or not np.isfinite(w.sum()):
        w = np.ones_like(w)
    return w / w.sum()


def window_sequence(name: str, n: int) -> np.ndarray:
    """Classic signal windows (Hann, Hamming, Blackman)."""
    i = np.arange(n, dtype=float)
    if name == "hann":
        w = 0.5 * (1 - np.cos(2 * np.pi * i / (n - 1)))
    elif name == "hamming":
        w = 0.54 - 0.46 * np.cos(2 * np.pi * i / (n - 1))
    elif name == "blackman":
        w = (0.42
             - 0.5 * np.cos(2 * np.pi * i / (n - 1))
             + 0.08 * np.cos(4 * np.pi * i / (n - 1)))
    else:
        raise ValueError("unknown window")
    return w / w.sum()


def symmetric_triangular(n: int) -> np.ndarray:
    """Symmetric triangular window, peak in the centre."""
    i = np.arange(n, dtype=float)
    centre = (n - 1) / 2.0
    w = np.maximum(0.0, 1.0 - np.abs(i - centre) * 2.0 / (n - 1))
    return w / w.sum()


def symmetric_cosine_sq(n: int) -> np.ndarray:
    """Cosine‑squared shape (smooth bump)."""
    i = np.arange(n, dtype=float)
    centre = (n - 1) / 2.0
    w = np.cos(np.pi * (i - centre) / (n + 1)) ** 2
    w = np.maximum(w, 0.0)
    return w / w.sum()


def symmetric_quadratic(n: int) -> np.ndarray:
    """Quadratic bell: 1‑(distance/centre)², clipped at zero."""
    i = np.arange(n, dtype=float)
    centre = (n - 1) / 2.0
    d = (i - centre) / centre
    w = np.maximum(0.0, 1.0 - d ** 2)
    return w / w.sum()


# ----------------------------------------------------------------------
# Core optimiser (mirror descent)
# ----------------------------------------------------------------------
def mirror_descent(
    a0: np.ndarray,
    deadline: float,
    max_iters: int = 20000,
    eta0: float = 0.8,
    tol_rel: float = 1e-6,
    noise_interval: int = 5000,
    noise_scale: float = 0.02,
) -> tuple[np.ndarray, float]:
    """
    Exponential‑gradient (mirror‑descent) optimisation on the simplex.
    Returns the best sequence seen and its objective value.
    """
    a = a0.copy()
    n = a.shape[0]
    best_a = a.copy()
    best_score = compute_objective(a)

    g_prev = None
    for it in range(max_iters):
        if time.time() > deadline:
            break

        # ----- convolution & current objective -----
        fft = np.fft.rfft(a, n=2 * n - 1)
        conv = np.fft.irfft(fft * fft, n=2 * n - 1)
        max_b = conv.max()
        cur_score = 2.0 * n * max_b

        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            # occasional progress report
            print(f"Iter {it:5d}: new best {best_score:.6f}")

        # ----- active set (indices within a tiny tolerance of the max) -----
        eps = max_b * tol_rel
        active = np.where(conv >= max_b - eps)[0]
        if active.size == 0:
            active = np.array([np.argmax(conv)], dtype=int)

        # ----- sub‑gradient of the max‐convolution -----
        g = np.zeros_like(a)
        for k in active:
            low = max(0, k - n + 1)
            high = min(n - 1, k)
            idx = np.arange(low, high + 1, dtype=int)
            partner = k - idx
            g[idx] += 2.0 * a[partner]
        g /= float(active.size)          # average over all active constraints

        # ----- optional momentum (simple exponential smoothing) -----
        if g_prev is not None:
            g = 0.7 * g_prev + 0.3 * g      # a tiny memory of previous step
        g_prev = g

        # ----- exponential (mirror) update -----
        step = eta0 / np.sqrt(it + 1.0)    # diminishing learning rate
        a = a * np.exp(-step * g)

        # ----- keep positivity, renormalise to the simplex -----
        a = np.maximum(a, 1e-15)
        a /= a.sum()

        # ----- occasional random jitter to escape shallow local minima -----
        if (it + 1) % noise_interval == 0:
            a = a * (1.0 + noise_scale * np.random.randn(n))
            a = np.maximum(a, 0.0)
            a /= a.sum()

    return best_a, best_score

            print(f"Run {run+1}/{runs}: improved global best to {best_score:.6f}")

    # ------------------------------------------------------------------
    # 5. Return the best sequence found (as a plain Python list)
    # ------------------------------------------------------------------
    print(f"Finished search – best objective = {best_score:.6f}")
