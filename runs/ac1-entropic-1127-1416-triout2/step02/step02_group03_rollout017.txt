"""
Propose a step‑function (non‑negative vector) that minimises

    2 * n * max(conv(a, a)) / (sum(a) ** 2)

The strategy improves on the reference implementation in three ways:

1. **Larger search space** – we use a longer vector (default n≈12 000).  
   The FFT‑based convolution makes each iteration cheap (O(n log n)).

2. **Richer random warm‑start** – two parametric families are sampled
   (power‑law and exponential) with wider ranges for the parameters.
   The best of a few ×10⁵ random candidates is taken as the start point.

3. **Momentum sub‑gradient descent** – instead of the plain sub‑gradient
   step we keep an exponential moving average of the sub‑gradients
   (β≈0.9).  The direction is therefore smoother and the method escapes
   shallow plateaus more efficiently.  A small jitter is injected
   periodically, and the step‑size decays like 1/√t.

All operations are performed with NumPy and the Euclidean projection onto
the simplex (sum = √(2·n), a_i ≥ 0).  The routine respects the 1000 s
budget and returns the best sequence it found as a plain Python list.
"""
import time
import numpy as np

# ----------------------------------------------------------------------
# Simplex projection (Duchi et al. 2008)
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray, target_sum: float) -> np.ndarray:
    if target_sum < 0:
        raise ValueError("target_sum must be non‑negative")
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - target_sum
    rho_arr = u - cssv / (np.arange(1, len(v) + 1))
    rho = np.nonzero(rho_arr > 0)[0]
    if rho.size == 0:
        return np.zeros_like(v)
    rho = rho[-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# FFT based full convolution (real input) – returns length 2·n‑1 array
# ----------------------------------------------------------------------
def _full_convolution_fft(a: np.ndarray) -> np.ndarray:
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return conv


def _max_conv(a: np.ndarray) -> float:
    """Maximum entry of the linear convolution a*a."""
    return _full_convolution_fft(a).max()


# ----------------------------------------------------------------------
# Families of candidate vectors
# ----------------------------------------------------------------------
def _candidate_powerlaw(n: int, p: float, offset: float, beta: float) -> np.ndarray:
    """a_i ∝ (i+1+offset)^{-p}·(1+β·i/(n‑1))"""
    i = np.arange(n, dtype=float)
    base = np.power(i + 1.0 + offset, -p)
    tilt = 1.0 + beta * i / (n - 1)
    raw = base * tilt
    raw = np.maximum(raw, 0.0)
    target = np.sqrt(2.0 * n)
    return _project_simplex(raw, target)


def _candidate_exponential(n: int, lam: float, beta: float) -> np.ndarray:
    """a_i ∝ exp(-λ·i)·(1+β·i/(n‑1))"""
    i = np.arange(n, dtype=float)
    base = np.exp(-lam * i)
    tilt = 1.0 + beta * i / (n - 1)
    raw = base * tilt
    raw = np.maximum(raw, 0.0)
    target = np.sqrt(2.0 * n)
    return _project_simplex(raw, target)


# ----------------------------------------------------------------------
# Sub‑gradient + momentum step
# ----------------------------------------------------------------------
def _subgradient_momentum_step(a: np.ndarray,
                               max_inds: np.ndarray,
                               step: float,
                               target_sum: float,
                               mom: np.ndarray,
                               beta: float = 0.9) -> tuple[np.ndarray, np.ndarray]:
    """
    One update of the momentum sub‑gradient method.
    Returns (new_a, new_momentum).
    """
    n = a.shape[0]
    g = np.zeros_like(a)

    for k in max_inds:
        if k < n:
            # i = 0 … k  (the slice a[k::-1] is a[k],a[k-1],…,a[0])
            g[: k + 1] += a[k::-1]
        else:
            start = k - (n - 1)               # i = start … n‑1
            # a[k - idx] where idx runs start … n‑1
            idx = np.arange(start, n)
            g[start:] += a[k - idx]

    # momentum update (exponential moving average)
    mom = beta * mom + (1.0 - beta) * g

    cand = a - step * mom
    cand = _project_simplex(cand, target_sum)
    return cand, mom

# Main search routine
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: int = 1000, **_):
    """
    Search for a non‑negative height sequence that minimises the
    evaluation function.  The routine respects the 1000 s time limit
    and returns the best sequence found.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 2.0   # a safety margin

    # ------------------------------------------------------------------
    # 0. Choose a length that balances speed and expressive power.
    # ------------------------------------------------------------------
    n = 12000                         # ≈ 12 k entries – still cheap with FFT
    target_sum = np.sqrt(2.0 * n)

    # ------------------------------------------------------------------
    # 1. Initialise – try to reuse a previously discovered sequence.
    # ------------------------------------------------------------------
    best_seq = None
    hs = globals().get("height_sequence_1")
    if isinstance(hs, list) and len(hs) == n:

        best_seq = _project_simplex(best_seq, target_sum)

    if best_seq is None:

        best_seq = _candidate_powerlaw(n, p=0.55, offset=0.0, beta=0.0)

    best_score = _max_conv(best_seq)
    print(f"[init] n={n:5d}  score={best_score:.6f}")

    # ------------------------------------------------------------------

    # ------------------------------------------------------------------

        if cand_score < best_score:

    # ------------------------------------------------------------------
    # 3. Momentum sub‑gradient descent from the best candidate
    # ------------------------------------------------------------------
    curr = best_seq.copy()
    momentum = np.zeros_like(curr)

    iteration = 0

        max_inds = np.where(conv_full >= M - eps)[0]

        # step size decays like 1/√t  (slightly slower than 1/√t to keep momentum alive)
        step = step0 / np.sqrt(iteration)

    # ------------------------------------------------------------------
    # 4. Wrap‑up
    # ------------------------------------------------------------------
