"""
Propose a low‑score step‑function.

The algorithm proceeds in two stages.

Stage 1 – small problem (n≈2500):
    • A diverse collection of seeds (geometric, power‑law, uniform,
      Gaussian, random Dirichlet and the previous best) is projected onto
      the simplex Σa_i = √(2 n) and the best seed is chosen.
    • From that seed we run many iterations of *smooth‑max* projected
      gradient descent.  The objective is a smoothed version of the
      maximum convolution value

            L_β(f) = (1/β)·log Σ_k exp(β·(f∗f)_k) .

      Its gradient can be written in closed form as a convolution;
      by annealing the smoothing parameter β (starting at 0.5 and
      growing to ~30) we obtain a stable descent direction that
      gradually focuses on the true maximal index.  After each gradient
      step the vector is projected onto the non‑negative simplex.

Stage 2 – large problem (n≈5000):
    • The best small‑scale vector is interpolated to length 5000,
      re‑projected onto the new simplex and refined again with the same
      smooth‑max descent (more degrees of freedom generally lower the bound).

At the end we evaluate the best sequence with the official
``evaluate_sequence`` function and return it.
"""

import time
import numpy as np

# --------------------------------------------------------------------------- #
# Simplex projection (Duchi et al. 2008)
def projection_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Project `v` onto the simplex {x ≥ 0, Σx = s}."""
    if s < 0:
        raise ValueError("Simplex sum must be non‑negative")
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho_arr = np.where(u - cssv / (np.arange(n) + 1.0) > 0)[0]
    if rho_arr.size == 0:
        return np.zeros_like(v)
    rho = rho_arr[-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# --------------------------------------------------------------------------- #
def _fast_score(seq: np.ndarray) -> float:
    """Exact score = 2·n·max(conv(seq,seq)) / (sum(seq)²) via FFT."""
    n = seq.size
    fft_len = 1 << ((2 * n - 1).bit_length())
    F = np.fft.rfft(seq, n=fft_len)
    conv = np.fft.irfft(F * F, n=fft_len)[: 2 * n - 1]
    max_conv = conv.max()
    s = seq.sum()
    return float(2.0 * n * max_conv / (s * s))


# --------------------------------------------------------------------------- #
def _smooth_max_gradient_descent(
    init_seq: np.ndarray,
    target_sum: float,
    deadline: float,
    start_time: float,
) -> tuple[np.ndarray, float]:
    """
    Projected gradient descent on the smooth‑max surrogate.
    Returns the best sequence seen and its (exact) score.
    """
    seq = init_seq.copy()
    n = seq.shape[0]

    # FFT lengths (powers of two, enough for linear convolutions)
    fft_len_conv = 1 << ((2 * n - 1).bit_length())          # for f*f
    fft_len_grad = 1 << ((3 * n - 2).bit_length())          # for f * rev(w)

    # annealing parameters
    beta = 0.5
    beta_max = 30.0
    beta_factor = 1.005

    # step‑size parameters
    step_scale = target_sum * 0.5   # absolute L2 step before normalisation
    step_decay = 0.9997

    best_seq = seq.copy()
    best_score = _fast_score(best_seq)

    # bookkeeping for stagnation detection
    last_improve = 0
    max_no_improve = 6000
    iteration = 0

    while time.time() < deadline:
        # ---- f * f ---------------------------------------------------------
        f_fft = np.fft.rfft(seq, n=fft_len_grad)           # same length works for both convolutions
        conv_full = np.fft.irfft(f_fft * f_fft, n=fft_len_grad)
        conv = conv_full[: 2 * n - 1]

        max_val = conv.max()
        # exact objective (sum is forced to target_sum)
        cur_score = float(2.0 * n * max_val / (target_sum * target_sum))

        if cur_score < best_score - 1e-12:
            best_score = cur_score
            best_seq = seq.copy()
            last_improve = iteration
            print(f"[{time.time() - start_time:6.1f}s] iter {iteration:7d} → new best {cur_score:.6f}")

        # ---- smooth‑max weights -------------------------------------------
        # stable softmax using max_val as the reference
        exp_weights = np.exp(beta * (conv - max_val))
        w = exp_weights / exp_weights.sum()
        rev_w = w[::-1]                                     # needed for the gradient

        # ---- gradient (2 * f ⋆ w) -----------------------------------------
        rev_w_fft = np.fft.rfft(rev_w, n=fft_len_grad)
        grad_full = np.fft.irfft(f_fft * rev_w_fft, n=fft_len_grad)
        grad_full = grad_full[: 3 * n - 2]                  # length 3n‑2
        grad = 2.0 * grad_full[(2 * n - 2) - np.arange(n)]

        grad_norm = np.linalg.norm(grad)
        if grad_norm == 0.0:
            break

        # ---- step ---------------------------------------------------------
        step = step_scale / (grad_norm + 1e-12)
        new_seq = seq - step * grad
        new_seq = np.clip(new_seq, 0.0, 1000.0)
        seq = projection_simplex(new_seq, target_sum)

        # ---- occasional jitter (helps escape plateaus) --------------------
        if np.random.rand() < 0.01:
            jitter = np.random.randn(n) * step * 0.02
            seq = projection_simplex(np.clip(seq + jitter, 0.0, 1000.0), target_sum)

        # ---- annealing ----------------------------------------------------
        if beta < beta_max:
            beta = min(beta_max, beta * beta_factor)

        # ---- step‑size decay -----------------------------------------------
        step_scale *= step_decay

        iteration += 1

        # ---- stagnation handling -------------------------------------------
        if iteration - last_improve > max_no_improve:
            # shake the current point a little and reset the counter
            mask = np.random.rand(n) < 0.08
            seq[mask] = best_seq[mask] * (1.0 + 0.05 * np.random.randn(mask.sum()))
            seq = projection_simplex(seq, target_sum)
            last_improve = iteration

    # final exact score of the best discovered vector
    best_score = _fast_score(best_seq)
    return best_seq, best_score

def propose_candidate(seed: int = 42, budget_s: float = 1000, **kwargs):
    """
    Search for a non‑negative step‑function minimizing ``evaluate_sequence``.
    The total wall‑clock budget is given by ``budget_s`` (≈ 1000 s).
    """
    np.random.seed(seed)
    start = time.time()
    deadline = start + budget_s - 0.3          # tiny safety margin

    # ------------------------------------------------------------------- #
    # 1. Small‑scale problem (n≈2500)
    n_small = 2500
    target_small = np.sqrt(2.0 * n_small)

    # ---- generate a variety of seeds ------------------------------------
    seeds = []

    # geometric families
    for lam in np.logspace(-6, -0.3, 30):
        r = 1.0 - lam
        denom = 1.0 - r ** n_small
        if denom <= 0.0:
            continue
        C = target_small * lam / denom
        a = C * (r ** np.arange(n_small))
        seeds.append(a)

    # power‑law families
    for p in np.linspace(0.3, 1.0, 30):
        base = 1.0 / np.power(np.arange(1, n_small + 1, dtype=np.float64), p)
        C = target_small / base.sum()
        a = C * base
        seeds.append(a)

    # Gaussian‑like bump (centered)
    xs = np.arange(n_small, dtype=np.float64) / (n_small - 1)
    for sigma in (0.08, 0.15, 0.25):
        a = np.exp(-0.5 * ((xs - 0.5) / sigma) ** 2)
        a = a * (target_small / a.sum())
        seeds.append(a)

    # uniform
    seeds.append(np.full(n_small, target_small / n_small))

    # a few Dirichlet random samples
    for _ in range(6):
        alpha = np.random.rand() * 4 + 0.1
        a = np.random.dirichlet(np.full(n_small, alpha)) * target_small
        seeds.append(a)

    # previous best (if available)
    try:
        global height_sequence_1
        hs = np.asarray(height_sequence_1, dtype=np.float64)
        if hs.shape[0] == n_small:
            hs = np.clip(hs, 0.0, 1000.0)
            hs = hs * (target_small / hs.sum())
            seeds.append(hs)
    except Exception:
        pass

    # ---- scan seeds -------------------------------------------------------
    best_seq = None
    best_score = np.inf
    for idx, cand in enumerate(seeds):
        if time.time() >= deadline:
            break
        cand = projection_simplex(cand, target_small)
        sc = _fast_score(cand)
        if sc < best_score:
            best_score = sc
            best_seq = cand.copy()
            print(f"[{time.time() - start:6.1f}s] seed {idx:02d} → score {sc:.6f}")

    if best_seq is None:
        # fallback – uniform distribution
        best_seq = np.full(n_small, target_small / n_small)
        best_score = _fast_score(best_seq)

    # ---- smooth‑max gradient descent on the small problem -----------------
    time_left = deadline - time.time()
    if time_left > 5.0:
        # give roughly half of the remaining budget to the small phase
        phase_deadline = time.time() + min(time_left * 0.55, 400.0)
        best_seq, best_score = _smooth_max_gradient_descent(
            best_seq, target_small, phase_deadline, start
        )
        print(f"[{time.time() - start:6.1f}s] small‑phase best score {best_score:.6f}")

    # ------------------------------------------------------------------- #
    # 2. Upsample to a larger problem (n≈5000) if time permits
    time_left = deadline - time.time()
    if time_left > 15.0:
        n_large = 5000
        target_large = np.sqrt(2.0 * n_large)

        # linear interpolation (keeps the shape)
        xs_small = np.linspace(0.0, 1.0, n_small)
        xs_large = np.linspace(0.0, 1.0, n_large)
        seq_large = np.interp(xs_large, xs_small, best_seq)
        seq_large = np.clip(seq_large, 0.0, 1000.0)
        seq_large = projection_simplex(seq_large, target_large)

        # gradient descent on the large problem
        phase_deadline = time.time() + time_left - 1.0   # leave a second buffer
        seq_large, large_score = _smooth_max_gradient_descent(
            seq_large, target_large, phase_deadline, start
        )
        print(f"[{time.time() - start:6.1f}s] large‑phase best score {large_score:.6f}")

        # keep the better of the two
        if large_score < best_score:
            best_seq, best_score = seq_large, large_score

    # ------------------------------------------------------------------- #
    # 3. Final official evaluation
    final_score = evaluate_sequence(best_seq.tolist())
    print(f"Search finished. Final score reported by evaluate_sequence: {final_score:.6f}")

    return best_seq.tolist()
