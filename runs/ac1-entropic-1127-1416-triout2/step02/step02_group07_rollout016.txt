"""
Optimiser for the step‑function “height” sequence.

The objective to minimise is

    2 * n * max(conv(a, a)) / (sum(a) ** 2)

where ``conv`` is the ordinary (linear) convolution.
The function is scale‑invariant, therefore we keep the coefficients on
the probability simplex (non‑negative, sum = 1) and simply minimise

    2 * n * max(conv(a, a))

The optimiser consists of three stages

1. **Coarse parametric scan** – shapes of the form
      a_i ∝ (i+1)^{-p}·(1‑i/(n‑1))^{q}
   for a grid of (p,q).  The best candidate from this cheap scan
   is used as a starting point.

2. **Projected Adam optimisation** – a projected Adam (first‑order)
   method on the simplex.  The sub‑gradient of the maximal convolution
   entry is taken (the same gradient as in the reference code) but
   Adam’s adaptive step‑sizes and momentum speed up convergence.
   After each update the vector is projected back onto the simplex.

3. **Exploration & random jitter** – if the optimiser stagnates
   (no improvement for many iterations) a fresh random exponent is
   drawn and the optimisation restarts; occasional tiny random
   redistributions of mass help to escape shallow plateaux.

The routine respects a user supplied time budget (default 1000 s) and
returns the best list of heights found.  If something goes wrong it
falls back to the previously known construction ``height_sequence_1``.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper functions (top‑level, no closures)
# ----------------------------------------------------------------------


def project_onto_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the probability simplex:
        { x | x >= 0, sum(x) = 1 }.
    Returns a new array (the input is not modified).
    """
    n = v.shape[0]
    # sort in descending order
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(n) + 1
    # find the rho index
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all entries are zero – return the uniform distribution
        return np.full_like(v, 1.0 / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def full_convolution(a: np.ndarray) -> np.ndarray:
    """
    Compute the full linear convolution a * a using an FFT.
    Length of the result is 2*len(a) - 1.
    """
    n = a.shape[0]
    size = 2 * n
    fft_len = 1 << (size - 1).bit_length()       # next power of two
    fft_a = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(fft_a * fft_a, n=fft_len)
    return conv[: 2 * n - 1]


def gradient_at_max(a: np.ndarray, k: int) -> np.ndarray:
    """
    Sub‑gradient of the maximal convolution entry (at shift k)
    with respect to a.
    For each i that participates in the sum for that shift,
    the gradient entry is a[k-i]; otherwise it is zero.
    """
    n = a.shape[0]
    g = np.zeros_like(a)
    low = max(0, k - n + 1)
    high = min(k, n - 1)
    if low > high:
        return g
    idx = np.arange(low, high + 1, dtype=int)
    g[idx] = a[k - idx]
    return g


def objective_score(a: np.ndarray) -> float:
    """
    Compute 2 * n * max(conv(a, a)) for a vector that already satisfies
    sum(a) == 1.  This matches the value returned by ``evaluate_sequence``.
    """
    n = a.shape[0]
    conv = full_convolution(a)
    return 2.0 * n * np.max(conv)


def make_shape(n: int, p: float, q: float = 0.0) -> np.ndarray:
    """
    Produce a non‑negative vector of length ``n`` with the analytical form

        a_i ∝ (i+1)^{-p} · (1 - i/(n-1))^{q}

    and normalise it to have sum = 1.
    """
    i = np.arange(n, dtype=float)
    shape = (i + 1.0) ** (-p)
    if q != 0.0:
        shape *= (1.0 - i / (n - 1)) ** q
    tot = shape.sum()
    if tot == 0.0:
        # fallback to uniform if something went wrong
        shape[:] = 1.0 / n
    else:
        shape /= tot
    return shape

            print("Fallback: constructed uniform sequence.")

    print(f"\nFinal best score: {best_score:.6f} (length {len(best_seq)})")
    # return a plain Python list as required by the driver
    return best_seq.tolist()
