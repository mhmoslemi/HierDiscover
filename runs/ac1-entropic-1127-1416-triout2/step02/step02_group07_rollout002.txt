"""
Propose a candidate step‑function by minimizing a smoothed version of
the ‑∞‑norm of the self‑convolution.

The algorithm works on a fixed length (default 5000) and keeps the
coefficients on the probability simplex (non‑negative, sum = 1).  Rather
than using a plain sub‑gradient of the hard max, we minimise the
log‑sum‑exp of the convolution entries:

    F(a) = (1/α) log Σ_k exp(α·(a*a)_k) ,

which approaches max_k (a*a)_k for large α but is smooth for moderate
α.  The gradient of F w.r.t. a can be written as a correlation of the
soft‑max weights with the reversed vector a; this is computed efficiently
with two FFT‑based convolutions per iteration.

Updates are performed with multiplicative (mirror‑descent) steps:
    a_i ← a_i·exp(‑η·∂F/∂a_i) ,   then renormalised to sum‑one.
Both the step size η and the smoothing parameter α are increased
gradually (∝√iteration) which gives a continuation from a smooth to a
sharp approximation of the max.  Several initialisations (power‑law,
uniform, previous best, random Dirichlet) are tried sequentially within
the time budget, keeping the best sequence seen.

The routine runs for at most ``budget_s`` seconds (default 1000) and
returns the best list of heights found.  If nothing is discovered a
uniform fallback is returned.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper functions (top level, no closures)

def project_onto_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def full_convolution(a: np.ndarray) -> np.ndarray:
    """Full convolution a * a using an FFT (real‑to‑real)."""
    n = a.shape[0]
    size = 2 * n - 1
    fft_len = 1 << (size - 1).bit_length()
    fa = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(fa * fa, n=fft_len)
    return conv[:size]


def full_convolution_two(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full convolution of two (possibly different) real vectors."""
    size = len(x) + len(y) - 1
    fft_len = 1 << (size - 1).bit_length()
    fx = np.fft.rfft(x, n=fft_len)
    fy = np.fft.rfft(y, n=fft_len)
    conv = np.fft.irfft(fx * fy, n=fft_len)
    return conv[:size]


def smoothed_max_gradient(a: np.ndarray, alpha: float) -> np.ndarray:
    """
    Return the gradient of the smoothed max:
        F(a) = (1/α) log Σ_k exp(α·conv_k)
    where conv = a * a (full convolution).  The gradient is the
    correlation of the soft‑max weights with the reversed a.
    """
    # 1) compute full convolution a*a
    conv = full_convolution(a)               # length 2n‑1
    # 2) soft‑max weights (stable)
    max_c = conv.max()
    w = np.exp(alpha * (conv - max_c))       # all ≤ 1
    w_sum = w.sum()
    w /= w_sum                               # now Σ w = 1
    # 3) correlation w * a_rev  -> gradient (up to a factor 2)
    rev_a = a[::-1]
    corr = full_convolution_two(w, rev_a)    # length (2n‑1)+n‑1 = 3n‑2
    # extract the central part of length n (indices n‑1 … 2n‑2)
    start = len(w) - 1                       # = 2n‑2
    grad = corr[start:start + len(a)]
    # factor 2 (derivative of a_i a_{k-i}) is absorbed into step size later
    return grad

    return 2.0 * n * float(conv.max())
