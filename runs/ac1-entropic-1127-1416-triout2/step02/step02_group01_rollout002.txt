assistantfinal**Solution Overview**

The goal is to minimise  

\[
E(a)=\frac{2\,n\;\max\limits_k (a*a)_k}{\bigl(\sum_i a_i\bigr)^2},
\qquad a_i\ge 0,
\]

where `(a*a)` is the *full* self‑convolution of the sequence `a`.  
`E(a)` is invariant under scaling, therefore we may (and do) restrict the
search to the probability simplex `∑ a_i = 1`.  
The problem is highly non‑convex; the previous implementation used a
smoothed‑max gradient descent with a simple SGD‑style step‑size.  
Here we improve on that in three ways

1. **Adam optimisation** – an adaptive first‑order method with
   momentum that converges much faster on the smooth surrogate  

   \[
   f_\alpha(a)=\frac{1}{\alpha}\log\!\Bigl(\sum_k e^{\alpha\,(a*a)_k}\Bigr).
   \]

2. **Symmetry enforcement** – the optimal shape is symmetric
   (`a_i = a_{n-1-i}`); after each update we symmetrise the vector and
   project back onto the simplex.  This halves the effective dimension
   and yields a lower peak in the convolution.

3. **Multi‑scale, multi‑restart strategy** – we optimise three lengths
   (`1024, 2048, 4096`).  For every length we try three different
   initialisations (arcsine, uniform, random Dirichlet) and keep the
   best candidate.  After the gradient phase a cheap stochastic
   hill‑climb polish removes any remaining small‑scale irregularities.

The whole search is bounded by the supplied time budget (`budget_s`),
so each outer loop checks the remaining time before starting a new
optimisation run.

The code below follows the described approach and conforms to the
required interface `propose_candidate`.  All helper functions are
defined at module level, there are no closures or filesystem access,
and the only external dependency is **NumPy** (already available in
the evaluation environment).

"""
Optimiser for the step‑function self‑convolution bound.

*   Uses a smoothed approximation of the max (log‑sum‑exp) and the
    Adam optimiser (adaptive moments, bias‑correction).
*   Enforces symmetry a_i = a_{n-1-i} – the optimal shape is known to be
    symmetric; this reduces the effective search space.
*   Performs several restarts with different lengths (1024, 2048, 4096)
    and three initial shapes (arcsine, uniform, random Dirichlet).
*   Finishes with a tiny stochastic hill‑climb that directly minimises
    the true `evaluate_sequence` objective.
*   Respects the time budget (`budget_s`) – the function stops as soon
    as the deadline is reached and returns the best sequence found.
"""

import time
import math

def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of the vector v onto the probability simplex
    (non‑negative, sum = 1).
    """
    if np.all(v <= 0):
        # corner case – return the uniform distribution
        return np.full_like(v, 1.0 / v.shape[0])

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, v.shape[0] + 1)
    rho = np.where(u - cssv / ind > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """
    Full convolution using a real FFT (fast O(n log n)).
    Returns a length (len(a)+len(b)-1) array.
    """
    m = len(a) + len(b) - 1
    n_fft = next_pow2(m)
    A = np.fft.rfft(a, n=n_fft)
    B = np.fft.rfft(b, n=n_fft)
    C = A * B
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


def compute_score(a: np.ndarray) -> float:
    """
    Compute the exact score of a (same definition as evaluate_sequence)
    under the assumption Σ a_i = 1.
    """
    n = len(a)
    conv = convolution_fft(a, a)
    max_val = float(np.max(conv))
    # Σ a_i = 1, so denominator = 1
    return 2.0 * n * max_val


def arcsine_initial(n: int) -> np.ndarray:
    """
    Arcsine‑shaped start: a_i ∝ 1 / sqrt((i+0.5)(n‑0.5‑i)).
    The vector is already normalised to sum = 1.
    """
    i = np.arange(n, dtype=float)
    eps = 0.25  # avoids division by zero at the boundaries
    vals = 1.0 / np.sqrt((i + 0.5) * (n - 0.5 - i) + eps)
    return vals / np.sum(vals)


# ----------------------------------------------------------------------
# Smoothed‑max gradient (log‑sum‑exp surrogate)
# ----------------------------------------------------------------------


def smoothed_gradient(a: np.ndarray, alpha: float):
    """
    Gradient of f_alpha(a) = (1/α)·log Σ_k exp(α·(a*a)_k)
    Returns
        grad  – ∇_a f_alpha  (same shape as a)
        conv  – raw self‑convolution (length 2n‑1) for diagnostics
    """
    n = len(a)

    # self‑convolution a * a
    conv = convolution_fft(a, a)                     # length 2n‑1

    # soft‑max weights (stable)
    maxc = np.max(conv)
    exp_term = np.exp(alpha * (conv - maxc))
    w = exp_term / np.sum(exp_term)                  # Σ w = 1, length 2n‑1

    # we need (a_rev (*) w) and the diagonal contributions
    a_rev = a[::-1]
    conv_rev = convolution_fft(a_rev, w)            # length 3n‑2
    start = n - 1
    rev_slice = conv_rev[start:start + n]           # size n, corresponds to
                                                    # Σ_j w_{i+j} a_j

    # diagonal terms w_{2i} multiply a_i (they appear only once)
    w_diag = w[::2][:n]

    grad = 2.0 * rev_slice - w_diag * a
    return grad, conv


# ----------------------------------------------------------------------
# Adam optimiser on the smoothed surrogate
# ----------------------------------------------------------------------


def adam_optimize(
    init_a: np.ndarray,
    deadline: float,
    max_iter: int = 150_000,
    lr: float = 0.5,
    alpha_start: float = 1.0,
    alpha_end: float = 500.0,
    symmetrize: bool = True,
    noise_every: int = 2000,
    noise_scale: float = 5e-5,
):
    """
    Adam optimiser for the smooth surrogate `f_alpha`.
    Returns the best vector seen (normalised to sum = 1) and its
    exact `compute_score`.
    """
    a = project_to_simplex(init_a.astype(float).copy())
    if symmetrize:
        a = (a + a[::-1]) / 2.0
        a = project_to_simplex(a)

    best_a = a.copy()
    best_score = compute_score(a)

    # Adam parameters
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    m = np.zeros_like(a)
    v = np.zeros_like(a)

    n = len(a)

    for t in range(1, max_iter + 1):
        # time check
        if time.time() >= deadline:
            break

        # smoothing schedule – linear from alpha_start to alpha_end
        alpha = alpha_start + (alpha_end - alpha_start) * min(1.0, t / max_iter)

        grad, _ = smoothed_gradient(a, alpha)

        if symmetrize:
            # enforce symmetry of the gradient as well
            grad = (grad + grad[::-1]) / 2.0

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)

        a_new = a - lr * m_hat / (np.sqrt(v_hat) + eps)

        if symmetrize:
            a_new = (a_new + a_new[::-1]) / 2.0

        a_new = project_to_simplex(a_new)

        # occasional isotropic noise to escape plateaus
        if noise_every and (t % noise_every == 0):
            a_new += np.random.randn(n) * noise_scale
            a_new = np.clip(a_new, 0, None)
            a_new = project_to_simplex(a_new)

        cur_score = compute_score(a_new)
        if cur_score < best_score:
            best_score = cur_score
            best_a = a_new.copy()

        a = a_new

        # occasional progress report
        if t % 5000 == 0:
            print(f"    iter {t:,d}  score {cur_score:.6f}  best {best_score:.6f}")

    return best_a, best_score


# ----------------------------------------------------------------------
# Cheap stochastic hill‑climbing refinement (uses the true objective)
# ----------------------------------------------------------------------


def hill_climb(seq, deadline: float, max_steps: int = 30_000, init_step: float = 0.08):
    """
    Pairwise exchange hill‑climbing.
    At each step a small amount of mass is moved from index i to j.
    Moves that improve the exact `evaluate_sequence` score are kept.
    """
    a = np.array(seq, dtype=float)
    a = project_to_simplex(a)          # ensure proper normalisation
    best_a = a.copy()
    best_score = compute_score(best_a)  # exact score (same as evaluate_sequence)
    n = len(a)
    step = init_step

    while max_steps > 0 and time.time() < deadline:
        i, j = np.random.randint(n, size=2)
        if i == j:
            continue
        # move a fraction of the mass at i to j
        delta = step * best_a[i]
        if delta <= 0:
            continue
        cand = best_a.copy()
        cand[i] -= delta
        cand[j] += delta
        cand = project_to_simplex(cand)   # re‑normalise

        cur_score = compute_score(cand)
        if cur_score < best_score:
            best_score = cur_score
            best_a = cand
            step = min(step * 1.07, 0.5)  # increase step on success
        else:
            step *= 0.96                  # shrink step on failure
        max_steps -= 1

    return best_a.tolist(), best_score


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative sequence that minimises `evaluate_sequence`.

    The routine performs:
      1. optional fallback to a known construction (`height_sequence_1`);
      2. multi‑scale Adam optimisation with symmetry enforcement;
      3. final stochastic hill‑climb polish.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 2.0   # leave a small safety margin

    best_seq = None
    best_val = float("inf")

    # ------------------------------------------------------------------
    # 1.  Use a previously discovered construction if it exists.
    # ------------------------------------------------------------------
    try:
        fallback = [float(x) for x in height_sequence_1]
        fallback_val = evaluate_sequence(fallback)
        print(f"Fallback score = {fallback_val:.6f}")
        if fallback_val < best_val:
            best_val = fallback_val
            best_seq = fallback
    except NameError:
        pass

    # ------------------------------------------------------------------
    # 2.  Multi‑scale Adam optimisation
    # ------------------------------------------------------------------
    lengths = [1024, 2048, 4096]

    for n in lengths:
        if time.time() >= deadline - 5.0:
            break
        print(f"\nOptimising length n = {n}")

        # three different starters
        starters = [
            arcsine_initial(n),
            np.full(n, 1.0 / n),
            np.random.dirichlet(np.ones(n)),
        ]

        # hyper‑parameters that adapt with n
        max_iter = 120_000 if n == 1024 else 150_000
        lr = 0.6 if n == 1024 else 0.45
        alpha_end = 800.0 if n >= 2048 else 500.0

        for idx, init_vec in enumerate(starters, start=1):
            if time.time() >= deadline - 5.0:
                break
            print(f"  start {idx}/{len(starters)}")
            seq, val = adam_optimize(
                init_a=init_vec,
                deadline=deadline,
                max_iter=max_iter,
                lr=lr,
                alpha_start=1.0,
                alpha_end=alpha_end,
                symmetrize=True,
                noise_every=2500,
                noise_scale=1e-4,
            )
            print(f"    local best score = {val:.6f}")
            if val < best_val:
                best_val = val
                best_seq = seq.tolist()
                print(f"    *** New global best {best_val:.6f} (n={n}) ***")

    # ------------------------------------------------------------------
    # 3.  Final cheap stochastic refinement
    # ------------------------------------------------------------------
    if best_seq is not None and time.time() < deadline - 2.0:
        print("\nRefining best sequence with hill‑climb")
        refined_seq, refined_val = hill_climb(
            best_seq,
            deadline=deadline,
            max_steps=60_000,
            init_step=0.07,
        )
        print(f"    refined score = {refined_val:.6f}")
        if refined_val < best_val:
            best_val = refined_val
            best_seq = refined_seq

    # ------------------------------------------------------------------
    # 4.  Safety net – return something if nothing was produced.
    # ------------------------------------------------------------------
    if best_seq is None:
        best_seq = [1.0] * 256
        best_val = evaluate_sequence(best_seq)

    # final validation with the official evaluator (ensures clipping, etc.)
    final_score = evaluate_sequence(best_seq)
    print(f"\nFinished search. Final reported score = {final_score:.6f}")

    # The harness expects a plain Python list of floats.
    return best_seq
