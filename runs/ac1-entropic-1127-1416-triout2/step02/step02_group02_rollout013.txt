"""
Optimise a non‑negative step‑function on the simplex to minimise
    2·n·max(f*f) / (sum f)^2 = 2·n·max(conv(f,f))

The algorithm uses three independent runs (previous best,
uniform, random Dirichlet) allocated a share of the time budget.
Each run performs projected Adam optimisation with an
Euclidean projection onto the probability simplex.  The sub‑gradient
of the active convolution entry (or average over the active set) is
used, occasional jitter is injected and stagnation triggers a
restart around the current best solution.  The best vector seen over
all runs is returned as a plain Python list.
"""

import time
import numpy as np

def simplex_projection(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the probability simplex:
        { w | w_i ≥ 0 , Σ w_i = 1 }.
    Implementation follows Duchi et al. (2008).
    """
    n = v.shape[0]
    # sort v in decreasing order
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all entries non‑positive → uniform distribution
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1] - 1               # zero‑based index
    theta = cssv[cond][-1] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def compute_objective(a: np.ndarray) -> float:
    """
    Return the evaluation function for a vector a that is already
    normalised to sum(a)=1.  By the scale‑invariance of the problem
    the objective is simply 2·n·max(conv(a,a)).
    """
    n = a.shape[0]
    fft_a = np.fft.rfft(a, n=2 * n - 1)
    conv = np.fft.irfft(fft_a * fft_a, n=2 * n - 1)
    max_b = conv.max()
    return 2.0 * n * max_b, conv, max_b


def subgradient(a: np.ndarray, conv: np.ndarray, max_val: float) -> np.ndarray:
    """
    Compute a sub‑gradient of the function  max_k (a*a)[k]  at the
    current point a.  The active set consists of all indices where the
    convolution attains (approximately) its maximum.
    """
    n = a.shape[0]
    tol = max_val * 1e-12 + 1e-14
    active = np.where(conv >= max_val - tol)[0]

    grad = np.zeros_like(a)
    for k in active:
        low = max(0, k - n + 1)
        high = min(n - 1, k)
        idx = np.arange(low, high + 1, dtype=int)
        partner = k - idx
        grad[idx] += 2.0 * a[partner]
    if active.size > 1:
        grad /= float(active.size)
    return grad


def optimise_one(init_a: np.ndarray, run_deadline: float) -> tuple[np.ndarray, float]:
    """
    Run a single Adam‑style projected optimisation starting from
    ``init_a`` (which must already sum to 1).  The function returns the
    best vector found and its objective value.
    """
    a = init_a.copy()
    n = a.shape[0]

    # Adam state
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    eta = 0.2                     # base learning‑rate

    best_a = a.copy()
    best_score, _, _ = compute_objective(a)

    iteration = 0
    last_improve = 0

    while time.time() < run_deadline:
        # ---- convolution, max and gradient ----
        score, conv, max_val = compute_objective(a)

        # keep the best point
        if score < best_score:
            best_score = score
            best_a = a.copy()
            last_improve = iteration
            print(f"Iter {iteration:6d}: improved score = {best_score:.6f}")

        # early exit if we already beat the target
        if best_score < 1.503:
            break

        # sub‑gradient of the max‑entry
        grad = subgradient(a, conv, max_val)

        # ---- Adam update (projected) ----
        iteration += 1
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)

        m_hat = m / (1.0 - beta1 ** iteration)
        v_hat = v / (1.0 - beta2 ** iteration)

        step = eta / (np.sqrt(v_hat) + eps)
        a_new = a - step * m_hat
        a = simplex_projection(a_new)

        # occasional jitter to escape shallow plateaus
        if iteration % 5000 == 0:
            jitter = 0.01 / np.sqrt(iteration)
            a = simplex_projection(a + jitter * np.random.randn(n))

        # restart around the current best when we have been idle for a while
        if iteration - last_improve > 40000:
            # blend best point with a fresh Dirichlet sample
            rng = np.random.default_rng()
            rand = rng.dirichlet(alpha=np.ones(n))
            a = simplex_projection(0.5 * best_a + 0.5 * rand)
            m.fill(0.0)
            v.fill(0.0)
            last_improve = iteration   # avoid immediate second restart

    return best_a, best_score


def propose_candidate(budget_s: int = 1000) -> list:
    """
    Search for a step‑function with a low evaluation score.
    The routine respects the 1000 s time limit and prints
    progress information.  It returns the best sequence found as
    a plain Python list.
    """
    deadline = time.time() + budget_s - 5.0   # safety margin

    # --------------------------------------------------------------
    # 1. Determine the problem size and construct three starting points.
    # --------------------------------------------------------------
    try:
        # The environment may provide a previously discovered construction.
        init_prev = np.asarray(height_sequence_1, dtype=float)   # type: ignore[name-defined]
        n = init_prev.shape[0]
        # Clip negatives (should not happen) and normalise to the simplex.
        init_prev = np.clip(init_prev, 0.0, None)
        if init_prev.sum() <= 0.0:
            init_prev = np.full(n, 1.0 / n)
        else:
            init_prev = init_prev / init_prev.sum()
    except Exception:
        # Fallback: a moderate size that works well with the time budget.
        n = 5000
        init_prev = np.full(n, 1.0 / n)

    # Uniform start
    init_uniform = np.full(n, 1.0 / n)

    # Random Dirichlet start
    rng = np.random.default_rng(12345)
    init_random = rng.dirichlet(alpha=np.ones(n))

    # --------------------------------------------------------------
    # 2. Allocate the time budget among the three runs.
    # --------------------------------------------------------------
    start_time = time.time()
    remaining = deadline - start_time
    fractions = [0.35, 0.35, 0.30]  # prev, uniform, random
    inits = [init_prev, init_uniform, init_random]

    overall_best_a = None
    overall_best_score = np.inf

    for init_vec, frac in zip(inits, fractions):
        run_deadline = time.time() + remaining * frac
        a, score = optimise_one(init_vec, run_deadline)
        if score < overall_best_score:
            overall_best_score = score
            overall_best_a = a.copy()
        # update remaining time and break if we are out of budget
        remaining = deadline - time.time()
        if remaining <= 0.0:
            break
        if overall_best_score < 1.503:
            break

    # If, by any chance, nothing was produced, fall back to uniform.
    if overall_best_a is None:
        overall_best_a = np.full(n, 1.0 / n)

    # --------------------------------------------------------------
    # 3. Return the best sequence as a plain Python list.
    # --------------------------------------------------------------
    return [float(v) for v in overall_best_a.tolist()]

"""
Optimise a non‑negative step‑function on the simplex to minimise
    2·n·max(conv(f,f)) / (sum f)^2
which, after fixing sum(f)=1, is simply 2·n·max(conv(f,f)).

The algorithm runs three independent optimisation sessions (starting
from the previously known construction, from a uniform vector and from
a random Dirichlet sample).  Each session uses projected Adam updates
with the sub‑gradient of the active convolution entry.  Periodic
jitter and a simple restart‑around‑the‑best strategy help escape
plateaus.  The best vector over all sessions is returned as a plain
list of floats.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper functions (all top‑level, no closures)
# ----------------------------------------------------------------------


def simplex_projection(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the probability simplex:
        { w | w_i ≥ 0 , Σ w_i = 1 }.
    Implementation follows Duchi et al. (2008).
    """
    n = v.shape[0]
    # sort v in decreasing order
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all entries non‑positive → uniform distribution
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1] - 1               # zero‑based index
    theta = cssv[cond][-1] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def compute_objective(a: np.ndarray) -> tuple[float, np.ndarray, float]:
    """
    Compute the evaluation score for a vector a that already sums to 1.
    Returns (score, full convolution, maximum convolution value).
    """
    n = a.shape[0]
    fft_a = np.fft.rfft(a, n=2 * n - 1)
    conv = np.fft.irfft(fft_a * fft_a, n=2 * n - 1)
    max_val = conv.max()
    score = 2.0 * n * max_val
    return score, conv, max_val


def subgradient(a: np.ndarray, conv: np.ndarray, max_val: float) -> np.ndarray:
    """
    Sub‑gradient of the function  max_k (a*a)[k]  at the current point.
    The active set contains every index whose convolution value is
    within numerical tolerance of the maximum.
    """
    n = a.shape[0]
    tol = max_val * 1e-12 + 1e-14
    active = np.where(conv >= max_val - tol)[0]

    grad = np.zeros_like(a)
    for k in active:
        low = max(0, k - n + 1)
        high = min(n - 1, k)
        idx = np.arange(low, high + 1, dtype=int)
        partner = k - idx
        grad[idx] += 2.0 * a[partner]
    if active.size > 1:
        grad /= float(active.size)
    return grad


def optimise_one(init_a: np.ndarray, run_deadline: float) -> tuple[np.ndarray, float]:
    """
    Perform a single optimisation run starting from ``init_a`` (already
    normalised to sum 1).  Returns the best vector found and its score.
    """
    a = init_a.copy()
    n = a.shape[0]

    # Adam state
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    eta = 0.2                     # base learning‑rate

    best_a = a.copy()
    best_score, _, _ = compute_objective(a)

    iteration = 0
    last_improve = 0

    while time.time() < run_deadline:
        # ---- current objective and gradient ----
        score, conv, max_val = compute_objective(a)

        if score < best_score:
            best_score = score
            best_a = a.copy()
            last_improve = iteration
            print(f"Iter {iteration:6d}: improved score = {best_score:.6f}")

        # stop early if we already beat the target
        if best_score < 1.503:
            break

        # sub‑gradient of the active max‑entry
        grad = subgradient(a, conv, max_val)

        # ---- Adam update (projected) ----
        iteration += 1
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)

        m_hat = m / (1.0 - beta1 ** iteration)
        v_hat = v / (1.0 - beta2 ** iteration)

        step = eta / (np.sqrt(v_hat) + eps)
        a_new = a - step * m_hat
        a = simplex_projection(a_new)

        # occasional jitter to shake the iterate out of plateaus
        if iteration % 5000 == 0:
            jitter = 0.01 / np.sqrt(iteration)
            a = simplex_projection(a + jitter * np.random.randn(n))

        # restart around the best point if we have been idle too long
        if iteration - last_improve > 40000:
            rng = np.random.default_rng()
            rand = rng.dirichlet(alpha=np.ones(n))
            a = simplex_projection(0.5 * best_a + 0.5 * rand)
            m.fill(0.0)
            v.fill(0.0)
            last_improve = iteration

    return best_a, best_score
