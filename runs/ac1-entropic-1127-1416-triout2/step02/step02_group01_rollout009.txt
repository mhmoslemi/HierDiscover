"""
Propose a low‑value step‑function using a fast parametric sweep,
followed by Adam‑based smoothed‑max gradient descent and a cheap
pairwise hill‑climb.  The search works on several vector lengths,
explores a one‑parameter family of “U‑shaped’’ sequences
a_i ∝ ((i+½)(n‑½‑i))^{‑s} (s≈0.5 corresponds to the classic arcsine
shape) and keeps the best candidate seen before the time budget
expires.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Basic linear‑algebra utilities
# ----------------------------------------------------------------------


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of a vector onto the probability simplex."""
    n = v.shape[0]
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / n)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho = np.where(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full convolution using a real FFT (fast)."""
    m = len(a) + len(b) - 1
    n_fft = next_pow2(m)
    A = np.fft.rfft(a, n=n_fft)
    B = np.fft.rfft(b, n=n_fft)
    C = A * B
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


def compute_score(a: np.ndarray) -> float:
    """
    Exact objective value used by ``evaluate_sequence`` when the vector
    is already normalised to sum = 1.
    """
    n = len(a)
    conv = convolution_fft(a, a)
    max_val = float(np.max(conv))
    s = float(np.sum(a))
    if s < 1e-12:
        return float("inf")
    return 2.0 * n * max_val / (s * s)


# ----------------------------------------------------------------------
# Parametric families of candidate sequences
# ----------------------------------------------------------------------


def edge_power_shape(n: int, s: float, eps: float = 1e-12) -> np.ndarray:
    """
    Symmetric U‑shaped distribution:
        a_i ∝ ((i+½)(n‑½‑i) + eps)^{‑s}
    s=0.5 reproduces the classic arcsine start.
    """
    i = np.arange(n, dtype=float)
    base = (i + 0.5) * (n - 0.5 - i) + eps
    vals = base ** (-s)
    return vals / np.sum(vals)


def triangular_shape(n: int) -> np.ndarray:
    """Deterministic triangular (tent) shape."""
    i = np.arange(n, dtype=float)
    vals = np.minimum(i + 1, n - i)
    return vals / np.sum(vals)


def uniform_shape(n: int) -> np.ndarray:
    """Uniform distribution."""
    return np.full(n, 1.0 / n)


# ----------------------------------------------------------------------
# Smoothed‑max gradient utilities (Adam optimisation)
# ----------------------------------------------------------------------


def smoothed_gradient(a: np.ndarray, alpha: float):
    """
    Gradient of the smoothed maximum:
        f_α(a) = (1/α)·log Σ_k exp(α·(a∗a)_k)
    Returns (grad, conv) where conv = a∗a (full length 2n‑1).
    """
    n = len(a)
    # self‑convolution
    conv = convolution_fft(a, a)                # length 2n‑1

    # soft‑max weights (log‑sum‑exp trick)
    maxc = np.max(conv)
    exp_term = np.exp(alpha * (conv - maxc))
    w = exp_term / np.sum(exp_term)             # ∑ w = 1

    # convolution of reversed a with the weight vector
    a_rev = a[::-1]
    conv_rev = convolution_fft(a_rev, w)        # length 3n‑2
    start = n - 1
    rev_slice = conv_rev[start:start + n]       # n‑vector

    # diagonal contribution w_{2i}
    w_diag = w[::2][:n]

    grad = 2.0 * rev_slice - w_diag * a
    return grad, conv

        if time.time() >= deadline:
            break

        # linear schedule for the smoothing parameter α
        alpha = alpha_start + (alpha_end - alpha_start) * min(1.0, t / max_iter)

        grad, _ = smoothed_gradient(a, alpha)

        # Adam update (bias‑corrected)
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)

        step = eta0 / (np.sqrt(v_hat) + eps)
        a_new = a - step * m_hat

        # occasional isotropic jitter to help escape shallow plateaus
        if noise_every and (t % noise_every == 0):
            a_new += np.random.randn(n) * noise_scale

        a_new = project_to_simplex(a_new)

        cur_score = compute_score(a_new)

        if cur_score < best_score:
            best_score = cur_score
            best_a = a_new.copy()

        a = a_new

        if t % 5000 == 0:
            print(f"    [Adam] iter {t:,d}  score {cur_score:.6f}  best {best_score:.6f}")

    return best_a, best_score


# ----------------------------------------------------------------------
# Tiny stochastic refinement (pairwise mass exchange)
# ----------------------------------------------------------------------


def hill_climb(seq, deadline: float, max_steps: int = 30_000, init_step: float = 0.08):
    """
    Simple stochastic hill climbing that moves a small fraction of mass
    from one index to another whenever the true max‑convolution score
    improves.
    """
    a = np.array(seq, dtype=float)
    a = project_to_simplex(a)
    best_a = a.copy()
    best_score = compute_score(best_a)
    n = len(a)
    step = init_step

    while max_steps > 0 and time.time() < deadline:
        i, j = np.random.randint(n, size=2)
        if i == j:
            continue
        delta = step * best_a[i]
        if delta <= 0.0:
            continue
        cand = best_a.copy()
        cand[i] -= delta
        cand[j] += delta
        if cand[i] < 0:
            continue
        cand = project_to_simplex(cand)
        cur_score = compute_score(cand)
        if cur_score < best_score:
            best_score = cur_score
            best_a = cand
            step = min(step * 1.04, 0.5)
        else:
            step *= 0.96
        max_steps -= 1

    return best_a.tolist(), best_score


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step‑function that minimises ``evaluate_sequence``.
    The routine:
      1. tries a small parametric sweep over the family
         a_i ∝ ((i+½)(n‑½‑i))^{‑s},
      2. refines the best candidate of each length with Adam‑based
         smoothed‑max gradient descent,
      3. finally applies a cheap pairwise hill‑climb.
    The best sequence found before the time budget expires is returned.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 2.0   # safety margin

    # ------------------------------------------------------------------
    # Global fallback if a known good construction is present.
    # ------------------------------------------------------------------
    best_seq = None
    best_val = float("inf")
    try:
        fallback = [float(x) for x in height_sequence_1]   # type: ignore
        fallback_val = evaluate_sequence(fallback)
        print(f"Fallback score = {fallback_val:.6f}")
        if fallback_val < best_val:
            best_val = fallback_val
            best_seq = fallback
    except Exception:
        pass

    # ------------------------------------------------------------------
    # Search parameters
    # ------------------------------------------------------------------
    lengths = [1024, 2048, 4096, 8192]          # larger n ⇒ potentially smaller bound
    s_grid = np.linspace(0.30, 0.80, 11)       # exponent s for the power‑law family

    for n in lengths:
        if time.time() >= deadline - 5.0:
            break
        print(f"\n--- Optimising length n = {n} ---")

        # ----- 1. Parametric sweep -------------------------------------------------
        local_best_a = None
        local_best_score = float("inf")

        for s in s_grid:
            a = edge_power_shape(n, s)
            val = compute_score(a)
            if val < local_best_score:
                local_best_score = val
                local_best_a = a
        # also try a few elementary starters
        for starter, name in [(uniform_shape(n), "uniform"),
                              (triangular_shape(n), "triangular")]:
            val = compute_score(starter)
            if val < local_best_score:
                local_best_score = val
                local_best_a = starter

        print(f"  parametric best score = {local_best_score:.6f}")

        # ----- 2. Adam‑based smoothed‑max refinement -------------------------------
        if time.time() < deadline - 5.0:
            # max_iter is generous; the loop will stop early when time runs out
            refined_a, refined_score = improve_sequence_adam(
                init_a=local_best_a,
                deadline=deadline,
                max_iter=400_000,
                eta0=0.15,
                alpha_start=1.0,
                alpha_end=1200.0,
                noise_every=500,
                noise_scale=8e-5,
            )
            print(f"    after Adam refinement score = {refined_score:.6f}")
            if refined_score < best_val:
                best_val = refined_score
                best_seq = refined_a.tolist()
                print(f"    *** New global best {best_val:.6f} (n={n}) ***")
        else:
            # No time for refinement, just keep the parametric candidate
            if local_best_score < best_val:
                best_val = local_best_score
                best_seq = local_best_a.tolist()
                print(f"    *** New global best (parametric) {best_val:.6f} (n={n}) ***")

    # ------------------------------------------------------------------
    # Final cheap stochastic hill‑climb (if time permits)
    # ------------------------------------------------------------------
    if best_seq is not None and time.time() < deadline - 1.0:
        print("\nRefining globally best sequence with hill‑climb")
        refined_seq, refined_score = hill_climb(
            best_seq, deadline=deadline, max_steps=70_000, init_step=0.07
        )
        print(f"    refined hill‑climb score = {refined_score:.6f}")
        if refined_score < best_val:
            best_val = refined_score
            best_seq = refined_seq

    # ------------------------------------------------------------------
    # Safety net
    # ------------------------------------------------------------------
    if best_seq is None:
        best_seq = [1.0] * 256
        best_val = evaluate_sequence(best_seq)

    final_score = evaluate_sequence(best_seq)
    print(f"\nFinished search. Final reported score = {final_score:.6f}")

    # The harness expects a plain Python list of floats.
    return best_seq
```

This implementation performs a quick parametric exploration of the classic “U‑shaped’’ family, refines the most promising candidate of each problem size with an Adam‑styled smoothed‑max gradient descent, and finishes with a lightweight pairwise hill‑climb.  The search is fully time‑aware (respecting the 1000 s budget) and only uses NumPy, satisfying the competition constraints.
