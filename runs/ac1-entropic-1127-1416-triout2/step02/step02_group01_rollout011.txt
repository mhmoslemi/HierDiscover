"""
Optimise a non‑negative step‑function for the evaluation
`E(a)=2·n·max(a∗a)/(∑a)^2`.

The algorithm works in three stages

1. **Warm‑start generation** – a small portfolio of analytically
   motivated shapes (arcsine, power‑laws, symmetric Beta, triangular,
   uniform and a random Dirichlet) is built for each candidate length.
   The best of these is used as a start point.

2. **Smoothed‑max projected gradient descent** – we minimise a smooth
   approximation  f_α(a)=(1/α)·log Σ_k exp(α·(a∗a)_k)   using FFT‑based
   convolutions.  The smoothing parameter α is increased linearly from a
   modest value (≈1) up to a large value (≈5000) so that the gradient
   becomes that of the true max.  After each step the iterate is
   projected back onto the probability simplex (∑a=1, a_i≥0).  The best
   iterate seen is retained.

3. **Stochastic hill‑climbing refinement** – pairwise mass transfers
   between coordinates are tried; moves that improve the true (unsmoothed)
   score are kept.  The step size adapts up/down depending on success.

The whole search respects the 1000 s budget (a small safety margin is
kept) and returns the best sequence found.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Basic linear‑algebra utilities
# ----------------------------------------------------------------------


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    if np.all(v <= 0):
        # Corner case – return uniform distribution
        return np.full_like(v, 1.0 / v.shape[0])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho = np.where(u - cssv / np.arange(1, v.shape[0] + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full convolution using real FFT (fast for real inputs)."""
    m = len(a) + len(b) - 1
    n_fft = next_pow2(m)
    A = np.fft.rfft(a, n=n_fft)
    B = np.fft.rfft(b, n=n_fft)
    C = A * B
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


def compute_score(a: np.ndarray) -> float:
    """Exact score matching `evaluate_sequence` (sum may deviate slightly from 1)."""
    n = len(a)
    conv = convolution_fft(a, a)
    max_val = float(np.max(conv))
    s = float(np.sum(a))
    if s < 1e-12:
        return float("inf")
    return 2.0 * n * max_val / (s * s)


# ----------------------------------------------------------------------
# Analytic initial shapes
# ----------------------------------------------------------------------


def arcsine_initial(n: int) -> np.ndarray:
    """Arcsine‑type start: a_i ∝ 1/√((i+½)(n‑½‑i))."""
    i = np.arange(n, dtype=float)
    eps = 1e-12
    vals = 1.0 / np.sqrt((i + 0.5) * (n - 0.5 - i) + eps)
    return vals / np.sum(vals)


def power_initial(n: int, p: float) -> np.ndarray:
    """Power‑law family a_i ∝ ((i+½)(n‑½‑i))^p .  p<0 concentrates on the ends."""
    i = np.arange(n, dtype=float)
    eps = 1e-12
    vals = ((i + 0.5) * (n - 0.5 - i) + eps) ** p
    vals = np.maximum(vals, 0.0)
    return vals / np.sum(vals)


def beta_initial(n: int, a: float, b: float) -> np.ndarray:
    """Symmetric Beta‑type shape a_i ∝ x^{a‑1}(1‑x)^{b‑1}  (x∈(0,1))."""
    x = (np.arange(n, dtype=float) + 0.5) / n
    eps = 1e-12
    vals = np.power(x + eps, a - 1.0) * np.power(1.0 - x + eps, b - 1.0)
    vals = np.maximum(vals, 0.0)
    return vals / np.sum(vals)


def triangular_initial(n: int) -> np.ndarray:
    """Simple symmetric triangle: a_i ∝ min(i+1, n‑i)."""
    i = np.arange(n, dtype=float)
    vals = np.minimum(i + 1.0, n - i)
    return vals / np.sum(vals)


# ----------------------------------------------------------------------
# Smoothed‑max gradient utilities
# ----------------------------------------------------------------------


def smoothed_gradient(a: np.ndarray, alpha: float):
    """
    Gradient of f_α(a) = (1/α)·log Σ_k exp(α·(a∗a)_k).
    Returns (grad, conv) where conv = a∗a.
    """
    n = len(a)
    # self‑convolution
    conv = convolution_fft(a, a)                 # length 2n‑1

    # soft‑max weights (numerically stable)
    maxc = np.max(conv)
    exp_term = np.exp(alpha * (conv - maxc))
    w = exp_term / np.sum(exp_term)              # length 2n‑1, Σ w = 1

    # gradient contribution via convolution with reversed a
    a_rev = a[::-1]
    conv_rev = convolution_fft(a_rev, w)         # length 3n‑2
    start = n - 1
    rev_slice = conv_rev[start:start + n]        # needed n entries

    # diagonal contribution (indices where k = 2i)
    w_diag = w[::2][:n]

    grad = 2.0 * rev_slice - w_diag * a
    return grad, conv


def improve_sequence_smoothed(
    initial_a: np.ndarray,
    time_deadline: float,
    max_iter: int = 200_000,
    eta0: float = 0.8,
    alpha_start: float = 1.0,
    alpha_end: float = 5000.0,
    noise_every: int = 0,
    noise_scale: float = 0.0,
):
    """
    Projected (smoothed‑max) gradient descent.
    Returns the best vector seen and its exact score.
    """
    a = project_to_simplex(initial_a.astype(float).copy())
    n = len(a)

    # initial score
    cur_conv = convolution_fft(a, a)
    best_score = 2.0 * n * np.max(cur_conv) / (np.sum(a) ** 2)
    best_a = a.copy()

    for t in range(1, max_iter + 1):
        if time.time() >= time_deadline:
            break

        # linear homotopy on α
        alpha = alpha_start + (alpha_end - alpha_start) * min(1.0, t / max_iter)

        grad, conv = smoothed_gradient(a, alpha)

        eta = eta0 / math.sqrt(t)
        a_new = a - eta * grad

        if noise_every and (t % noise_every == 0):
            a_new += np.random.randn(n) * noise_scale

        a_new = project_to_simplex(a_new)

        # exact score of the candidate
        conv_new = convolution_fft(a_new, a_new)
        cur_score = 2.0 * n * np.max(conv_new) / (np.sum(a_new) ** 2)

        if cur_score < best_score:
            best_score = cur_score
            best_a = a_new.copy()

        a = a_new

        # occasional progress report (quiet enough)
        if t % 5000 == 0:
            print(f"    iter {t:,d}  cur_score {cur_score:.6f}  best {best_score:.6f}")

    return best_a, best_score


# ----------------------------------------------------------------------
# Simple stochastic hill‑climbing refinement
# ----------------------------------------------------------------------


def hill_climb(seq, time_deadline: float, max_steps: int = 50000, init_step: float = 0.1):
    """
    Pair‑wise mass‑transfer hill‑climb.
    Small amounts of probability are moved from i to j;
    moves that improve the exact score are kept.
    """
    a = np.array(seq, dtype=float)
    a = project_to_simplex(a)
    n = len(a)
    best_a = a.copy()
    best_score = compute_score(best_a)
    step = init_step

    while max_steps > 0 and time.time() < time_deadline:
        i, j = np.random.randint(n, size=2)
        if i == j:
            continue
        delta = step * best_a[i]
        if delta <= 0:
            continue
        cand = best_a.copy()
        cand[i] -= delta
        cand[j] += delta
        if cand[i] < 0:
            continue
        cand = project_to_simplex(cand)
        cur_score = compute_score(cand)
        if cur_score < best_score:
            best_score = cur_score
            best_a = cand
            step = min(step * 1.03, 0.5)   # grow a bit when successful
        else:
            step *= 0.97                    # shrink on failure
        max_steps -= 1
    return best_a.tolist(), best_score


# ----------------------------------------------------------------------
# Main entry point (required by the competition framework)
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a step‑function (non‑negative sequence) that minimises
    `evaluate_sequence`.  The routine explores a few problem sizes,
    employs several analytically motivated initialisations, runs a
    smoothed‑max projected gradient descent, and finishes with a
    stochastic hill‑climbing polish.
    """
    np.random.seed(seed)
    start_time = time.time()
    # keep a tiny safety margin so we never overrun the allocated budget
    deadline = start_time + budget_s - 2.0

    best_seq = None
    best_val = float("inf")

    # ------------------------------------------------------------------
    # Use a previously discovered construction if it exists.
    # ------------------------------------------------------------------
    try:
        fallback = [float(x) for x in height_sequence_1]   # type: ignore
        fallback_val = evaluate_sequence(fallback)
        print(f"Fallback score = {fallback_val:.6f}")
        if fallback_val < best_val:
            best_val = fallback_val
            best_seq = fallback
    except NameError:
        pass

    # ------------------------------------------------------------------
    # Lengths to explore – larger n gives a finer discretisation.
    # ------------------------------------------------------------------
    lengths = [1024, 2048, 4096]

    # ------------------------------------------------------------------
    # Helper to build a short list of starting vectors for a given n.
    # ------------------------------------------------------------------
    def build_starts(n):
        starts = []
        # analytic families
        starts.append(arcsine_initial(n))
        starts.append(triangular_initial(n))
        starts.append(np.full(n, 1.0 / n))
        # power‑law family (a handful of exponents)
        for p in (-0.7, -0.6, -0.5, -0.4, -0.3):
            starts.append(power_initial(n, p))
        # symmetric Beta family
        for a in (0.3, 0.4, 0.5, 0.6, 0.8):
            starts.append(beta_initial(n, a, a))
        # a single random Dirichlet start
        starts.append(np.random.dirichlet(np.ones(n)))
        return starts

    # ------------------------------------------------------------------
    # Main optimisation loop over lengths.
    # ------------------------------------------------------------------
    for n in lengths:
        if time.time() >= deadline - 5.0:
            break
        print(f"\nOptimising length n = {n}")

        starts = build_starts(n)

        # tune how many gradient iterations we can afford for this length
        remaining = deadline - time.time()
        # allocate roughly equal time per start, but guarantee a minimum of 30 s per start
        max_iter_per_start = int(min(250_000, max(30_000, remaining / (len(starts) * 0.6) * 1e5)))
        # (the factor 0.6 leaves slack for hill‑climb and printing)

        for idx, init_a in enumerate(starts, start=1):
            if time.time() >= deadline - 5.0:
                break
            print(f"  start {idx}/{len(starts)}")

            # quick sanity check – compute the raw score of the init vector
            init_score = compute_score(init_a)
            print(f"    init score = {init_score:.6f}")

            # gradient descent phase
            seq, val = improve_sequence_smoothed(
                initial_a=init_a,
                time_deadline=deadline,
                max_iter=max_iter_per_start,
                eta0=0.8,
                alpha_start=1.0,
                alpha_end=5000.0,
                noise_every=0,          # keep the walk deterministic
                noise_scale=0.0,
            )
            print(f"    gradient best score = {val:.6f}")

            if val < best_val:
                best_val = val
                best_seq = seq.tolist()
                print(f"    *** New global best {best_val:.6f} (n={n}) ***")

    # ------------------------------------------------------------------
    # Final refinement with hill‑climb (if we have a candidate).
    # ------------------------------------------------------------------
    if best_seq is not None and time.time() < deadline - 2.0:
        print("\nRefining best sequence with hill‑climb")
        refined_seq, refined_val = hill_climb(
            best_seq,
            time_deadline=deadline,
            max_steps=120_000,
            init_step=0.07,
        )
        print(f"    refined score = {refined_val:.6f}")
        if refined_val < best_val:
            best_val = refined_val
            best_seq = refined_seq

    # ------------------------------------------------------------------
    # Fallback if everything failed (very unlikely).
    # ------------------------------------------------------------------
    if best_seq is None:
        best_seq = [1.0] * 256
        best_val = evaluate_sequence(best_seq)

    final_score = evaluate_sequence(best_seq)
    print(f"\nFinished search. Final reported score = {final_score:.6f}")

    # The harness expects a plain Python list of floats.
    return best_seq
```

This implementation builds a diverse set of analytically motivated initial vectors,
runs a robust smoothed‑max gradient descent with a long homotopy on the smoothing
parameter (`α` up to 5000), and finishes with a stochastic hill‑climb that moves mass
between coordinates while keeping the iterate on the probability simplex.  
All steps respect the 1000 s time budget, and the code prints intermediate scores
so you can monitor progress during execution.
