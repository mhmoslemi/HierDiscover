def solve_lp_for_g(a, M):
    n = len(a)
    # Build convolution matrix
    # diag_data = np.tile(a, (n, 1))
    data = np.tile(a, (n, 1))
    offsets = np.arange(n)
    T = sp.diags(data, offsets, shape=(2*n - 1, n), format='csr')
    c = -np.ones(n)  # maximize sum -> minimize -sum
    bounds = [(0, None)] * n
    b_ub = np.full(2*n - 1, M)
    # Use high-performance linear programming
    res = opt.linprog(c, A_ub=T, b_ub=b_ub, bounds=bounds, method='highs')
    if not res.success:
        raise RuntimeError('LP failed: ' + res.message)
    return res.x

"""
Propose a candidate step‑function using a hybrid of
* random parametric search (power‑law + tail taper),
* short projected sub‑gradient refinements,
* occasional LP‑based “mass‑rebalancing” steps.

The algorithm works on a fixed length `n` (default 5000),
keeps the coefficients on the probability simplex (non‑negative,
sum = 1) and tries to minimise

    2 * n * max(conv(a, a))

which is exactly the value returned by `evaluate_sequence` for
a normalised sequence.  The search stops after the allotted
budget (≈1000 s) and returns the best sequence found.
"""
import time
import numpy as np
from scipy import sparse
from scipy.optimize import linprog

# ----------------------------------------------------------------------
# Basic linear‑algebra helpers
# ----------------------------------------------------------------------


def project_onto_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of a vector onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def full_convolution(a: np.ndarray) -> np.ndarray:
    """Full discrete convolution a * a using an FFT."""
    n = a.shape[0]
    size = 2 * n
    fft_len = 1 << (size - 1).bit_length()
    fft_a = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(fft_a * fft_a, n=fft_len)
    return conv[: 2 * n - 1]


def gradient_at_max(a: np.ndarray, k: int) -> np.ndarray:
    """Sub‑gradient of the maximal convolution entry (shift k) w.r.t. a."""
    n = a.shape[0]
    g = np.zeros_like(a)
    low = max(0, k - n + 1)
    high = min(k, n - 1)
    if low > high:
        return g
    idx = np.arange(low, high + 1, dtype=int)
    g[idx] = a[k - idx]
    return g


def objective_score(a: np.ndarray) -> float:
    """Return 2 * n * max(conv(a, a)) assuming sum(a) == 1."""
    n = a.shape[0]
    conv = full_convolution(a)
    return 2.0 * n * float(np.max(conv))


# ----------------------------------------------------------------------
# Sub‑gradient refinement (projected onto the simplex)
# ----------------------------------------------------------------------


def subgradient_refine(a: np.ndarray, max_iter: int,
                       deadline: float) -> np.ndarray:
    """Run a few steps of projected sub‑gradient descent."""
    n = a.shape[0]
    step_initial = 0.5
    it = 0
    while it < max_iter and time.time() < deadline:
        conv = full_convolution(a)
        k_max = int(np.argmax(conv))
        g = gradient_at_max(a, k_max)
        step = step_initial / np.sqrt(it + 1.0)
        a_candidate = a - step * g
        a = project_onto_simplex(a_candidate)
        it += 1
    return a


# ----------------------------------------------------------------------
# LP step – maximise the total mass under the same convolution bound
# ----------------------------------------------------------------------


def solve_lp_for_g(a: np.ndarray, M: float) -> np.ndarray:
    """
    Solve   max sum(g)  s.t.   a * g <= M  (full convolution) , g >= 0.
    Returns the optimal vector g (may be non‑normalised).
    """
    n = a.shape[0]
    # Build the Toeplitz‑like convolution matrix (2n‑1) × n.
    # Each diagonal offset i (0 ≤ i < n) contains the constant a[i].
    diag_data = np.tile(a, (n, 1))               # shape (n, n)
    offsets = np.arange(n)                       # 0 … n‑1
    T = sparse.diags(diag_data, offsets,
                     shape=(2 * n - 1, n), format='csr')

    # Linear programme: minimise -sum(g)  ⇔ maximise sum(g)
    c = -np.ones(n)
    bounds = [(0, None)] * n
    b_ub = np.full(2 * n - 1, M, dtype=float)

    res = linprog(c, A_ub=T, b_ub=b_ub, bounds=bounds,
                  method='highs', options={'presolve': True})
    if not res.success:
        raise RuntimeError(f'LP failed: {res.message}')
    return res.x


# ----------------------------------------------------------------------
# Main search routine
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 123, budget_s: int = 1000, **kwargs):
    """Search for a non‑negative height sequence that minimises the
    evaluation function.  Returns the best sequence as a plain list.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 5.0   # keep a few seconds for cleanup

    # ------------------------------------------------------------------
    # Problem size – 5000 gives a good trade‑off between quality and speed
    # ------------------------------------------------------------------
    n = 5000

    # ------------------------------------------------------------------
    # 1.  Initialise from the classic 1/√i shape (already decent)
    # ------------------------------------------------------------------
    raw = 1.0 / np.sqrt(np.arange(n, dtype=float) + 1.0)
    a = raw / raw.sum()                     # sum(a) == 1
    best_seq = a.copy()
    best_score = objective_score(a)
    print(f"[init] length={n}, score={best_score:.6f}")

    # ------------------------------------------------------------------
    # 2.  Random parametric search (few × 10⁴ evaluations are cheap)
    # ------------------------------------------------------------------
    iteration = 0
    while time.time() < deadline:
        # ---- sample a three‑parameter family ---------------------------------
        #   a_i ∝ (i+α+1)^{-p} * (1 - i/(n‑1))^{γ}
        p = 0.5 + 0.15 * (np.random.rand() - 0.5)          # ≈0.425…0.575
        α = np.random.rand() * (n * 0.5)                   # 0 … n/2

        i = np.arange(n, dtype=float)

        tail_factor[i == n - 1] = 0.0                       # exact zero at the end
        raw = (i + α + 1.0) ** (-p) * tail_factor
        # Guard against accidental all‑zero vectors
        if raw.sum() == 0.0:
            continue
        a_candidate = raw / raw.sum()
        cand_score = objective_score(a_candidate)

        if cand_score < best_score:
            best_score = cand_score
            best_seq = a_candidate.copy()
            a = a_candidate.copy()                         # keep it for refinements
            print(f"[param] iter={iteration:06d} score={best_score:.6f}"

        iteration += 1

        # ---- occasional refinement with a short sub‑gradient run ----------
        if iteration % 250 == 0:

            tmp_score = objective_score(a)
            if tmp_score < best_score:
                best_score = tmp_score
                best_seq = a.copy()
                print(f"[sgd ] iter={iteration:06d} improved to {best_score:.6f}")

        # ---- occasional LP‑based mass‑rebalancing step --------------------
        if iteration % 500 == 0:

                    a = (1.0 - t) * a + t * g_norm
                    a = project_onto_simplex(a)
                    lp_score = objective_score(a)
                    if lp_score < best_score:
                        best_score = lp_score
                        best_seq = a.copy()
                        print(f"[lp  ] iter={iteration:06d} improved to {best_score:.6f}")
            except Exception as exc:                         # robust against occasional failures
                print(f"[lp  ] LP step failed: {exc}")

        # ------------------------------------------------------------------
        # safety break if we already beat the target (1.503)
        # ------------------------------------------------------------------
        if best_score < 1.503:
            print("[info] Target reached, stopping early.")
            break

    # ------------------------------------------------------------------
    # 3.  Final local polishing (a few thousand sub‑gradient steps)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        a = subgradient_refine(best_seq, max_iter=2000, deadline=deadline)
        final_score = objective_score(a)
        if final_score < best_score:
            best_score = final_score
            best_seq = a.copy()
            print(f"[final] polishing gave {best_score:.6f}")
