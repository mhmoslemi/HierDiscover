"""
Propose a step‑function (non‑negative sequence) that minimizes

        2 * n * max(conv(a,a)) / (sum(a)**2)

The algorithm consists of three phases:

1. **Warm‑start** – a brief random search in a 4‑parameter power‑law family
   (a_i = (i+α)^(-p)·(1+β·i/(n‑1))) produces a good initial vector whose
   L1‑norm is fixed to √(2·n).  The best vector from this sweep is kept.

2. **Smoothed gradient descent** – we minimise the smoothed objective

        Fγ(a) = (1/γ)·log Σ_k exp(γ·(a*a)_k)

   whose gradient is
        ∇Fγ(a) = 2·corr(a , w)      with   w_k ∝ exp(γ·((a*a)_k−max(a*a))).

   The gradient is obtained by a single FFT‑based correlation, the step is
   normalised (‖Δa‖∞≤step), and the iterate is projected back onto the
   simplex {x≥0, Σx = √(2·n)}.  The smoothing parameter γ starts small
   (≈0.5) and is multiplied by 1.005 each iteration, ending near 8.0.
   An Adam‑style schedule is *not* needed – a simple 1/√t decay works
   well.

3. **Final reporting** – the best vector seen (lowest max‑convolution) is
   returned as a plain list.  Throughout the run the true score
   max(conv) is printed so that the progress can be monitored.

The whole routine respects the 1000 s budget and never calls
``evaluate_sequence`` inside the optimisation loops (the score equals
max(conv) because the L1‑norm is kept fixed).
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities
# ----------------------------------------------------------------------


def _project_simplex(v: np.ndarray, target_sum: float) -> np.ndarray:
    """
    Euclidean projection of a vector onto the simplex
        { w | w_i ≥ 0, Σ w_i = target_sum }.
    Implements the algorithm of Duchi et al. (2008).
    """
    if target_sum < 0:
        raise ValueError("target_sum must be non‑negative")
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - target_sum
    ind = np.arange(1, len(v) + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.zeros_like(v)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def _full_convolution_fft(a: np.ndarray) -> np.ndarray:
    """
    Linear convolution a * a using a real FFT.
    Returns an array of length 2·len(a)‑1.
    """
    n = len(a)
    size = 1 << ((2 * n - 1).bit_length())        # next power of two
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)[: 2 * n - 1]
    return conv


def _max_conv(a: np.ndarray) -> float:
    """Maximum entry of the self‑convolution of a."""
    return _full_convolution_fft(a).max()


def _make_candidate(n: int, p: float, offset: float, beta: float) -> np.ndarray:
    """
    Build an un‑normalised raw sequence
        a_i = (i + 1 + offset)^{-p} * (1 + beta * i/(n‑1))
    and project it onto the simplex Σa = √(2·n).
    """
    idx = np.arange(n, dtype=float)                 # i = 0 … n‑1
    base = np.power(idx + 1.0 + offset, -p)
    factor = 1.0 + beta * idx / (n - 1)
    raw = base * factor
    raw = np.maximum(raw, 0.0)
    target = np.sqrt(2.0 * n)
    return _project_simplex(raw, target)


def _gradient_corr(a: np.ndarray, w: np.ndarray) -> np.ndarray:
    """
    Gradient of the smoothed max‑convolution objective:
        grad = 2 * corr(a, w)   (only the central n entries are needed)
    where w is a probability vector of length 2·n‑1.
    """
    n = len(a)
    size = 1 << ((2 * n - 1).bit_length())
    A = np.fft.rfft(a, n=size)
    W = np.fft.rfft(w, n=size)
    corr_full = np.fft.irfft(A * np.conj(W), n=size)[: 2 * n - 1]
    # central part corresponds to lags −(n‑1) … +(n‑1)
    grad = 2.0 * corr_full[n - 1 : n - 1 + n]
    return grad


# ----------------------------------------------------------------------
# Main search routine
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 123, budget_s: int = 1000, **kwargs):
    """
    Search for a low‑evaluation step‑function.
    Returns the best sequence (list of floats) found before the time budget
    expires.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 2.0   # safety margin for the harness

    # --------------------------------------------------------------
    # 0. Problem size
    # --------------------------------------------------------------
    n = 8000                                 # a size that balances quality / speed
    target_sum = np.sqrt(2.0 * n)

    # --------------------------------------------------------------
    # 1. Warm‑start (use a previously found sequence if available)
    # --------------------------------------------------------------
    best_seq = None
    hs = globals().get("height_sequence_1")
    if hs is not None and len(hs) == n:
        best_seq = np.array(hs, dtype=float)
        best_seq = _project_simplex(best_seq, target_sum)
    else:
        best_seq = _make_candidate(n, p=0.5, offset=0.0, beta=0.0)

    best_score = _max_conv(best_seq)          # because Σa = √(2n) ⇒ score = max conv
    print(f"[init] n={n}  score={best_score:.6f}")

    # --------------------------------------------------------------
    # 2. Fast random sweep in the 4‑parameter family
    # --------------------------------------------------------------
    sweep_deadline = time.time() + 0.15 * (deadline - time.time())
    while time.time() < sweep_deadline:
        p = np.random.uniform(0.35, 0.65)
        off = np.random.uniform(0.0, 15.0)
        beta = np.random.uniform(-0.9, 3.0)
        cand = _make_candidate(n, p, off, beta)
        cand_score = _max_conv(cand)
        if cand_score < best_score:
            best_score = cand_score
            best_seq = cand.copy()
            print(f"[sweep] new best {best_score:.6f}  p={p:.3f} α={off:.3f} β={beta:.3f}")

        if best_score < 1.503:   # early exit if we are lucky
            break

    # --------------------------------------------------------------
    # 3. Smoothed gradient descent with continuation on γ
    # --------------------------------------------------------------
    a = best_seq.copy()
    step0 = 0.15               # initial step size (max change of any entry)
    gamma = 0.5                # start with a smooth approximation
    gamma_max = 8.0
    gamma_inc = 1.005          # multiplicative increase each iteration
    iteration = 0

    while time.time() < deadline:
        iteration += 1

        # self‑convolution and current score
        conv = _full_convolution_fft(a)
        M = conv.max()

        # record the best sequence seen so far
        if M < best_score:
            best_score = M
            best_seq = a.copy()
            print(f"[grad] iter {iteration:6d}  new best {best_score:.6f}  γ={gamma:.3f}")

            if best_score < 1.503:          # target reached
                break

        # ------- smoothed weights for the max ----------
        # stability: shift by the current maximum
        exp_vals = np.exp(gamma * (conv - M))
        w = exp_vals / exp_vals.sum()        # probability vector, length 2·n‑1

        # ------- gradient (correlation) ----------
        grad = _gradient_corr(a, w)

        # normalise the gradient so that the update never exceeds ``step``
        max_abs = np.max(np.abs(grad)) + 1e-12
        norm_grad = grad / max_abs

        # decaying step size (1/√t)
        step = step0 / np.sqrt(iteration)

        # gradient step + projection onto the simplex
        a = a - step * norm_grad
        a = _project_simplex(a, target_sum)

        # increase γ slowly (continuation)
        if gamma < gamma_max:
            gamma = min(gamma_max, gamma * gamma_inc)

        # occasional tiny random jitter to escape flat plateaus
        if iteration % 120 == 0 and np.random.rand() < 0.25:
            jitter = (np.random.rand(n) - 0.5) * 0.02 * a
            a = _project_simplex(a + jitter, target_sum)

    # --------------------------------------------------------------
    # 4. Final reporting
    # --------------------------------------------------------------
    print(f"Finished after {iteration} gradient iterations – best score {best_score:.6f}")
    return best_seq.tolist()
