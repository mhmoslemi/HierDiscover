"""
Hybrid optimiser for the sup‑norm of a self‑convolution.

1. Build a handful of cheap parametric families (power‑law,
   exponential, mixed, truncated uniform) and pick the best one.
2. Starting from that seed run a projected sub‑gradient descent on the
   objective  2·n·max(conv(a,a))  (the denominator is fixed by
   normalising the vector to sum = 1).  The gradient is taken at all
   shifts that are (almost) maximal, giving a more accurate descent
   direction than using a single shift.
3. Every few hundred iterations a tiny random redistribution of mass
   is performed to escape plateaux.
4. The loop stops when the time budget (default 1000 s) is exhausted
   or when the target value 1.503 is beaten.
5. The best sequence seen is returned as a plain Python list.
"""

# ----------------------------------------------------------------------
# Imports and helper utilities (top‑level, no closures)
# ----------------------------------------------------------------------
import time
import numpy as np

# ----------------------------------------------------------------------
# Convolution utilities
# ----------------------------------------------------------------------


def full_convolution(a: np.ndarray) -> np.ndarray:
    """
    Full discrete convolution a * a using an FFT.
    Length of the result is ``2*len(a) - 1``.
    """
    n = a.shape[0]
    size = 2 * n
    # next power of two (speed)
    fft_len = 1 << (size - 1).bit_length()
    # real‑valued FFT (rfft/irfft) is faster and uses half the memory
    fft_a = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(fft_a * fft_a, n=fft_len)
    return conv[: 2 * n - 1]


def max_convolution_gradient(a: np.ndarray):
    """
    Compute the sub‑gradient of the sup‑norm of ``a*a``.
    Returns (gradient, max_val) where ``gradient`` has the same shape as ``a``.
    The gradient is the sum of the elementary gradients belonging to all
    shifts whose convolution value is within numerical tolerance of the
    maximum.
    """
    n = a.shape[0]
    conv = full_convolution(a)
    max_val = conv.max()
    # Numerical tolerance – we take all shifts that are essentially equal.
    tol = max_val * 1e-12 + 1e-15
    ks = np.where(conv >= max_val - tol)[0]

    g = np.zeros_like(a)
    for k in ks:
        low = max(0, k - n + 1)
        high = min(k, n - 1)
        if low > high:
            continue
        idx = np.arange(low, high + 1, dtype=int)
        g[idx] += a[k - idx]
    return g, max_val


def objective_score(a: np.ndarray) -> float:
    """
    Return the value of the evaluation function for a vector already
    normalised to sum = 1, i.e. ``2 * n * max(conv(a,a))``.
    """
    n = a.shape[0]
    conv = full_convolution(a)
    return 2.0 * n * np.max(conv)


# ----------------------------------------------------------------------
# Simple simplex projection (fallback only)
# ----------------------------------------------------------------------


def project_onto_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the probability simplex
    {x | x >= 0, sum(x) = 1}.  Used only as a safety fallback.
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w

# Main optimisation routine
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative height sequence that minimises
    ``evaluate_sequence``.  The routine respects ``budget_s`` seconds,
    uses a hybrid parametric initialise + sub‑gradient descent
    strategy, and returns the best list of floats found.
    """
    np.random.seed(seed)
    start_time = time.time()
    # keep a small safety margin for printing/return
    deadline = start_time + budget_s - 2.0

    # ------------------------------------------------------------------
    # Choose a length for the step function.
    # Use the length of the previously discovered construction if it
    # exists, otherwise fall back to 5000 (the length used in earlier
    # experiments).
    # ------------------------------------------------------------------
    try:

        print(f"Reusing length from previous construction: n = {n}")
    except Exception:
        n = 5000
        print(f"No previous length found – using default n = {n}")

    # ------------------------------------------------------------------
    best_seq = None

        print("Fallback to 1/√(i+1) shape.")

    # ------------------------------------------------------------------
    # 2️⃣  Sub‑gradient descent from the best parametric seed
    # ------------------------------------------------------------------
    a = best_seq.copy()               # always keep sum(a) == 1
    step_initial = 0.8                # initial step magnitude
    iteration = 0
    patience = 0                      # counts iterations without improvement

        # ----- compute gradient at (near) maximal convolution shift(s) -----
        g, max_val = max_convolution_gradient(a)

        # ----- take a diminishing step, then re‑normalise -----
        step = step_initial / np.sqrt(iteration + 1.0)
        a_candidate = a - step * g

        # enforce non‑negativity
        a_candidate = np.clip(a_candidate, 0.0, None)

        s = a_candidate.sum()
        if s <= 1e-12:                # numerical disaster – re‑initialise

        else:
            a_candidate = a_candidate / s

        a = a_candidate

        # ----- evaluate current point -----
        cur_score = 2.0 * n * max_val  # sum(a) is 1, so denominator = 1
        if cur_score < best_score:
            best_score = cur_score
            best_seq = a.copy()
            patience = 0
            print(
                f"[sgd] Iter {iteration:06d}: improved score = {best_score:.6f} "
                f"(max shift value = {max_val:.6e})"
            )
            if best_score < 1.503:           # target reached – stop early
                break
        else:
            patience += 1

        # ----- occasional random jitter to shake off plateaux -----
        if iteration % 250 == 0 and iteration > 0:
            # move a small random fraction of mass from i → j
            i, j = np.random.randint(0, n, size=2)
            if i != j:
                delta = 0.002 * a[i] * np.random.uniform(-1.0, 1.0)
                jitter = a.copy()
                jitter[i] = max(jitter[i] - delta, 0.0)
                jitter[j] += delta
                # renormalise (the total mass may have changed by the clipping)
                jitter_sum = jitter.sum()
                if jitter_sum > 0:
                    jitter = jitter / jitter_sum
                else:
                    jitter = np.full(n, 1.0 / n)
                # quick check whether the jitter improved the objective
                jitter_score = objective_score(jitter)
                if jitter_score < best_score:
                    best_score = jitter_score
                    best_seq = jitter.copy()
                    a = jitter
                    patience = 0
                    print(
                        f"[jitter] Iter {iteration:06d}: improved score = {best_score:.6f}"
                    )

        # ----- optional early termination if we have been stuck for a while -----
        if patience > 20000:          # 20k iterations without improvement
            # re‑initialise from the current best but add a small random kick
            a = best_seq.copy()
            noise = np.random.normal(scale=0.001, size=n)
            a = np.clip(a + noise, 0.0, None)
            a = a / a.sum()
            patience = 0
            print(f"[reset] No improvement for 20k iterations – randomising a bit.")

        iteration += 1

    # ------------------------------------------------------------------
    # 3️⃣  Final safety fallback
    # ------------------------------------------------------------------
    if best_seq is None or best_seq.size == 0:
        # Try to fall back on the global variable that should contain
        # the previous construction.
        try:
            best_seq = np.array(height_sequence_1, dtype=float)
            best_seq = best_seq / best_seq.sum()
            best_score = objective_score(best_seq)
        except Exception:
            # ultimate fallback: uniform distribution (guaranteed score = 2.0)
            best_seq = np.full(n, 1.0 / n, dtype=float)
            best_score = objective_score(best_seq)

    print(f"Finished optimisation – best score = {best_score:.6f} (n = {len(best_seq)})")
    # Return a plain Python list as required by the driver.
    return best_seq.tolist()
