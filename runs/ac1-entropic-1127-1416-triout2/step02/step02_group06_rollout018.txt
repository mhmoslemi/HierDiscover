def smooth_objective(p, alpha, n, fft_len):
    conv = compute_conv(p, fft_len)
    # compute log-sum-exp
    max_c = conv.max()
    # for stability: sum_exp = np.exp(alpha*(conv - max_c)).sum()
    lse = max_c + np.log(np.exp(alpha*(conv - max_c)).sum()) / alpha
    # lse = logsumexp(alpha*conv)/alpha
    return 2.0 * n * lse / (target_sum**2)

def smooth_gradient(p, alpha, n, fft_len):
    conv = compute_conv(p, fft_len)
    max_c = conv.max()
    # compute weights
    exp_vals = np.exp(alpha * (conv - max_c))
    sum_exp = exp_vals.sum()
    weights = exp_vals / sum_exp   # sum 1
    # gradient = 2 * np.correlate(p, weights, mode='valid')
    grad = 2.0 * np.correlate(p, weights, mode='valid')
    # derivative of lse w.r.t conv is weights
    # Since objective = 2n * lse / target_sum**2; derivative w.r.t conv is (2n/(target_sum**2))*weights
    grad = grad * (2.0 * n / (target_sum**2))
    return grad

def adam_optimize(init_seq, target_sum, n, fft_len, deadline, max_iters=200000, tol_rel=1e-4, lr0=0.5, beta1=0.9, beta2=0.999, eps=1e-8):
   seq = init_seq.copy()
   m = np.zeros_like(seq)
   v = np.zeros_like(seq)
   best_seq = seq.copy()
   best_score = fast_score(seq)
   for t in range(1, max_iters+1):
        if time.time() >= deadline:
            break
        conv = compute_conv(seq, fft_len)
        max_val = conv.max()
        # compute score (since sum fixed)
        score = 2.0 * n * max_val / (target_sum**2)
        if score < best_score - 1e-12:
            best_score = score
            best_seq = seq.copy()
            print(f"[{time.time() - start:6.1f}s] Adam iter {t:6d} → new best {score:.6f}")
        # compute gradient on active set
        tol_abs = max_val * tol_rel
        active_idx = np.where(conv >= max_val - tol_abs)[0]
        if active_idx.size == 0:
            break
        mask = np.zeros_like(conv)
        mask[active_idx] = 1.0
        grad = 2.0 * np.correlate(seq, mask, mode='valid')
        # Adam update
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad ** 2)
        m_hat = m / (1 - beta1 ** t)
        v_hat = v / (1 - beta2 ** t)
        # learning rate decays
        lr = lr0 * (1.0 - t / max_iters)
        step = lr * m_hat / (np.sqrt(v_hat) + eps)
        # move opposite direction (gradient ascend increases max)
        cand = seq - step
        # ensure non-negative and bound
        cand = np.clip(cand, 0.0, 1000.0)
        # project to simplex
        cand = projection_simplex(cand, target_sum)
        seq = cand
        # occasional jitter
        if np.random.rand() < 0.005:
            jitter = (np.random.randn(n) * lr * target_sum / (30.0 * n))
            seq = projection_simplex(np.clip(seq + jitter,0,1000), target_sum)

"""
Hybrid search for a low‑score step‑function.

*  Generates a diverse collection of seeds (geometric, power‑law,
   U‑shaped, uniform, random Dirichlet and – if available – the
   previous best sequence).
*  Refines each seed with an Adam–style projected optimiser that
   minimises the maximum of the self‑convolution (active‑set
   sub‑gradient).  The optimiser runs until the time budget expires.
*  A short LP‑refinement phase (identical to the reference method)
   further reduces the bound.
*  All vectors are kept on the simplex {x≥0, Σx = √(2 n)} so the
   objective simplifies to 2 n·max(conv(x,x)).
"""
import time
import numpy as np

# --------------------------------------------------------------------------- #
# Simplex projection (Duchi et al. 2008)
def projection_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Project ``v`` onto the simplex {x ≥ 0, Σx = s}."""
    if s < 0:
        raise ValueError("Simplex sum must be non‑negative")
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho_arr = np.where(u - cssv / (np.arange(n) + 1.0) > 0)[0]
    if rho_arr.size == 0:
        return np.zeros_like(v)
    rho = rho_arr[-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# --------------------------------------------------------------------------- #
def _fft_len_for_n(n: int) -> int:
    """Length of the zero‑padded FFT used for a convolution of size n."""
    return 1 << ((2 * n - 1).bit_length())


# --------------------------------------------------------------------------- #
def _conv(seq: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution ``seq * seq`` using the given FFT length."""
    F = np.fft.rfft(seq, n=fft_len)
    conv = np.fft.irfft(F * F, n=fft_len)[: 2 * seq.shape[0] - 1]
    return conv


# --------------------------------------------------------------------------- #
def _fast_score(seq: np.ndarray) -> float:
    """Score = 2·n·max(conv(seq,seq)) / (sum(seq)²)."""
    n = seq.size
    conv = _conv(seq, _fft_len_for_n(n))
    max_conv = conv.max()
    s = seq.sum()
    return float(2.0 * n * max_conv / (s * s))


# --------------------------------------------------------------------------- #
def _build_toeplitz_matrix(f: np.ndarray):
    """Sparse Toeplitz matrix A (2n‑1 × n) such that A·g = f * g."""
    n = f.shape[0]
    rows = np.empty(n * n, dtype=np.int32)
    cols = np.empty(n * n, dtype=np.int32)
    data = np.empty(n * n, dtype=np.float64)

    ptr = 0
    for i in range(n):
        rows[ptr: ptr + n] = np.arange(i, i + n)      # rows i … i+n‑1
        cols[ptr: ptr + n] = i
        data[ptr: ptr + n] = f
        ptr += n

    from scipy.sparse import csr_matrix
    return csr_matrix((data, (rows, cols)), shape=(2 * n - 1, n))


# --------------------------------------------------------------------------- #
def _adam_optimize(
    init_seq: np.ndarray,
    target_sum: float,
    n: int,
    fft_len: int,
    deadline: float,
    start_time: float,
    max_iters: int = 200_000,
    tol_rel: float = 1e-4,
    lr0: float = 0.4,
    beta1: float = 0.9,
    beta2: float = 0.999,
    eps: float = 1e-8,
) -> np.ndarray:
    """
    Adam‑style projected optimisation on the simplex.
    Uses the active‑set sub‑gradient (indices where the convolution is
    within ``tol_rel`` of its maximum).
    Returns the best vector encountered during the run.
    """
    seq = init_seq.copy()
    m = np.zeros_like(seq)
    v = np.zeros_like(seq)

    best_seq = seq.copy()
    best_score = _fast_score(seq)

    for t in range(1, max_iters + 1):
        if time.time() >= deadline:
            break

        conv = _conv(seq, fft_len)
        max_val = conv.max()
        score = 2.0 * n * max_val / (target_sum ** 2)

        if score < best_score - 1e-12:
            best_score = score
            best_seq = seq.copy()
            print(f"[{time.time() - start_time:6.1f}s] Adam iter {t:6d} → new best {score:.6f}")

        # --- active set ----------------------------------------------------
        tol_abs = max_val * tol_rel
        active_idx = np.where(conv >= max_val - tol_abs)[0]
        if active_idx.size == 0:
            break

        mask = np.zeros_like(conv)
        mask[active_idx] = 1.0
        grad = 2.0 * np.correlate(seq, mask, mode='valid')

        # --- Adam update ---------------------------------------------------
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)

        # linear decay of the learning rate
        lr = lr0 * (1.0 - t / max_iters)
        step = lr * m_hat / (np.sqrt(v_hat) + eps)

        cand = seq - step                     # move opposite to the gradient
        cand = np.clip(cand, 0.0, 1000.0)
        cand = projection_simplex(cand, target_sum)
        seq = cand

        # occasional random jitter to escape plateaus
        if np.random.rand() < 0.005:
            jitter = (np.random.randn(n) *
                      lr * target_sum / (30.0 * n))
            seq = projection_simplex(np.clip(seq + jitter, 0.0, 1000.0),
                                     target_sum)

    return best_seq

        return best_seq.tolist()

    # Allocate 80 % of the remaining budget to Adam, 20 % to LP.
    adam_budget = time_left * 0.80
    lp_budget = time_left - adam_budget

    # Number of independent Adam runs (including the original best)
    n_restarts = 4
    per_restart_deadline = start + budget_s - lp_budget - 0.2
    # we simply reuse the global deadline for the whole Adam phase;
    # the LP phase will respect the remaining time later.

    for r in range(n_restarts):
        if time.time() >= deadline:
            break
        # perturb the current best to obtain a new start point
        pert = best_seq * (1.0 + 0.01 * np.random.randn(n))
        pert = np.clip(pert, 0.0, 1000.0)
        pert = projection_simplex(pert, target_sum)

        # run Adam optimisation
        seq_opt = _adam_optimize(
            pert,
            target_sum,
            n,
            fft_len,
            deadline,
            start,
            max_iters=200_000,
            tol_rel=1e-4,
            lr0=0.4 * target_sum / n,
        )

        sc_opt = _fast_score(seq_opt)
        if sc_opt < best_score:
            best_score = sc_opt
            best_seq = seq_opt.copy()
            print(f"[{time.time() - start:6.1f}s] restart {r} → new best {sc_opt:.6f}")

    # ------------------------------------------------------------------- #
    # 4. Linear‑programming refinement (if SciPy is available).
    try:
        from scipy.optimize import linprog
        from scipy.sparse import csr_matrix
    except Exception:  # SciPy not present
        linprog = None

    if linprog is not None and time.time() < deadline - 5.0:
        f = best_seq.copy()
        # we keep the same problem size ``n`` and ``fft_len``
        for lp_iter in range(30):               # a few more LP steps than before
            if time.time() >= deadline - 5.0:
                break

            conv_f = _conv(f, fft_len)
            M = conv_f.max()

            # constraint matrix A such that A·g = f * g
            A = _build_toeplitz_matrix(f)
            b = np.full(2 * n - 1, M)

            # maximise Σ g_i  ⇔  minimise −Σ g_i
            res = linprog(
                c=-np.ones(n),
                A_ub=A,
                b_ub=b,
                bounds=[(0, None)] * n,
                method="highs",
                options={"presolve": True, "time_limit": 30},
            )
            if not res.success or res.x is None:
                break

            g = res.x
            sum_g = g.sum()
            if sum_g < 1e-12:
                break

            # rescale g to the target ℓ₁‑norm and mix
            g = g * (target_sum / sum_g)
            mix = 0.12                      # a modest mixing weight
            f_candidate = (1.0 - mix) * f + mix * g
            f_candidate = projection_simplex(f_candidate, target_sum)

            conv_new = _conv(f_candidate, fft_len)
            M_new = conv_new.max()
            score_new = float(2.0 * n * M_new / (target_sum ** 2))

            if score_new < best_score - 1e-8:
                best_score = score_new
                best_seq = f_candidate.copy()
                print(f"[{time.time() - start:6.1f}s] LP‑iter {lp_iter} → best {score_new:.6f}")

            # keep the update if it does not increase the maximal convolution
            if M_new <= M * 0.999:
                f = f_candidate
            else:
                break

    # ------------------------------------------------------------------- #
    # 5. Final sanity check with the official evaluator.
    final_score = evaluate_sequence(best_seq.tolist())
    print(f"Search finished. Final score reported by evaluate_sequence: {final_score:.6f}")

    return best_seq.tolist()
