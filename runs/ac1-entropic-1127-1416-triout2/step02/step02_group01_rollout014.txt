"""\
Optimisation of a step‑function sequence for minimal self‑convolution bound.

The algorithm consists of several stages:
  – Several problem sizes n are examined (up to the time budget).
  – For each n a few initial points (arcsine, uniform, random Dirichlet) are generated.
  – A smoothed‑max gradient descent (log‑sum‑exp) is run with a schedule that first
    uses a modest smoothing (α≈1…200) and then increases α up to a large value
    (α≈1500) so that the smoothed objective converges to the true max.
    Adam‑style adaptive steps and a tiny Gaussian noise help escaping shallow
    plateaux.
  – After the smoothing phase we finish with a few thousand steps of a
    true‑max (hard) sub‑gradient descent, which directly reduces the
    maximal convolution entry.
  – The best sequence found overall is polished with a cheap stochastic
    pair‑wise mass‑exchange hill‑climb.
The whole search is stopped as soon as the allocated time budget (≈1000 s)
is exhausted; the best feasible list of non‑negative heights is returned.
"""

import time
import numpy as np

def project_to_simplex(v):
    """Euclidean projection of a vector onto the probability simplex."""
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / v.shape[0])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho = np.where(u - cssv / np.arange(1, v.shape[0] + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w

def next_pow2(x):
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()

def convolution_fft(a, b):
    """Full convolution using a real FFT (fast)."""
    m = len(a) + len(b) - 1
    n_fft = next_pow2(m)
    A = np.fft.rfft(a, n=n_fft)
    B = np.fft.rfft(b, n=n_fft)
    C = A * B
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv

def compute_score(a):
    """Score identical to `evaluate_sequence` when ∑a = 1."""
    n = len(a)
    conv = convolution_fft(a, a)
    max_val = np.max(conv)
    s = np.sum(a)
    if s < 1e-12:
        return float('inf')
    return 2.0 * n * max_val / (s * s)

def arcsine_initial(n):
    """Arcsine‑shaped start: a_i ∝ 1 / sqrt((i+0.5)(n‑0.5‑i))."""
    i = np.arange(n, dtype=float)
    eps = 0.25
    vals = 1.0 / np.sqrt((i + 0.5) * (n - 0.5 - i) + eps)
    return vals / np.sum(vals)

def smoothed_gradient(a, alpha):
    """
    Gradient of the smoothed maximum
        f_α(a) = (1/α)·log Σ_k exp(α·(a∗a)_k)
    Returns the gradient (same shape as a) and the raw convolution a∗a.
    """
    conv = convolution_fft(a, a)              # length 2n‑1
    maxc = np.max(conv)
    exp_term = np.exp(alpha * (conv - maxc))
    w = exp_term / np.sum(exp_term)           # soft‑max weights, sum 1

    # conv of reversed a with the weight vector
    a_rev = a[::-1]
    conv_rev = convolution_fft(a_rev, w)      # length 3n‑2
    start = len(a) - 1
    rev_slice = conv_rev[start:start + len(a)]

    # diagonal contribution (w_{2i})
    w_diag = w[::2][:len(a)]

    grad = 2.0 * rev_slice - w_diag * a
    return grad, conv

def max_gradient(a, idx):
    """
    Exact gradient of the maximal convolution entry (hard max).
    idx – index where a∗a attains its maximum.
    Returns a vector of shape a.
    """
    n = a.shape[0]
    indices = np.arange(n)
    g = np.zeros_like(a)
    # term a_{k - i}
    mask1 = (idx - indices >= 0) & (idx - indices < n)
    g[mask1] += a[idx - indices[mask1]]
    # term a_{i - k}
    mask2 = (indices - idx >= 0) & (indices - idx < n)
    g[mask2] += a[indices[mask2] - idx]
    return g

def adam_update(x, grad, m, v, t, lr=0.1, beta1=0.9, beta2=0.999, eps=1e-8):
    """Performs one Adam step, returns new x, updated m, v."""
    m = beta1 * m + (1.0 - beta1) * grad
    v = beta2 * v + (1.0 - beta2) * (grad * grad)
    m_hat = m / (1.0 - beta1 ** t)
    v_hat = v / (1.0 - beta2 ** t)
    step = lr * m_hat / (np.sqrt(v_hat) + eps)
    x_new = x - step
    return x_new, m, v

def optimise_one_start(initial_a, deadline,
                       max_iter_smoothed=250_000,
                       alpha_start=1.0,
                       alpha_mid=200.0,
                       alpha_end=1500.0,
                       adam_lr=0.2,
                       noise_scale=5e-5,
                       hard_max_iter=30_000):
    """
    Runs a three‑phase optimisation:
      1. smoothed‑max descent with Adam up to α≈alpha_mid,
      2. continuation with larger α up to α≈alpha_end,
      3. a short hard‑max (sub‑gradient) phase.
    Returns the best vector (as a NumPy array) and its true score.
    """
    a = project_to_simplex(initial_a.astype(float).copy())
    n = len(a)

    # Adam moments
    m = np.zeros_like(a)
    v = np.zeros_like(a)

    best_a = a.copy()
    best_score = compute_score(a)

    # ----- Phase 1: modest smoothing -----
    for t in range(1, max_iter_smoothed + 1):
        if time.time() >= deadline:
            break
        # linear schedule of α from start to alpha_mid
        alpha = alpha_start + (alpha_mid - alpha_start) * min(1.0, t / max_iter_smoothed)
        grad, _ = smoothed_gradient(a, alpha)

        # Adam update
        a, m, v = adam_update(a, grad, m, v, t, lr=adam_lr)

        # tiny isotropic noise to escape flat plateaux
        if t % 700 == 0:
            a += np.random.randn(n) * noise_scale

        a = project_to_simplex(a)

        cur_score = compute_score(a)
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()

    # ----- Phase 2: large smoothing (approach true max) -----
    # continue with the same Adam state, enlarge α further
    for t2 in range(1, max_iter_smoothed + 1):
        if time.time() >= deadline:
            break
        # α grows from alpha_mid to alpha_end
        alpha = alpha_mid + (alpha_end - alpha_mid) * min(1.0, t2 / max_iter_smoothed)
        grad, _ = smoothed_gradient(a, alpha)

        # keep same learning‑rate schedule (scaled down a bit)
        a, m, v = adam_update(a, grad, m, v, t2 + max_iter_smoothed,
                               lr=adam_lr * 0.5)

        if t2 % 500 == 0:   # reduce noise as we tighten the approximation
            a += np.random.randn(n) * (noise_scale * 0.5)

        a = project_to_simplex(a)

        cur_score = compute_score(a)
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()

    # ----- Phase 3: hard‑max sub‑gradient descent -----
    # use a simple decreasing step‑size schedule.
    step = 0.2
    for it in range(1, hard_max_iter + 1):
        if time.time() >= deadline:
            break
        conv = convolution_fft(a, a)
        idx_max = int(np.argmax(conv))
        g = max_gradient(a, idx_max)

        # step size decays as 1/√it
        eta = step / np.sqrt(it)
        a = a - eta * g
        a = project_to_simplex(a)

        cur_score = compute_score(a)
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()

        # occasional restart of step size if we see no improvement for a while
        if it % 5000 == 0:
            # reduce step size slowly
            step = max(step * 0.9, 0.02)

    return best_a, best_score

def hill_climb(seq, deadline, max_steps=30_000, init_step=0.05):
    """
    Stochastic pair‑wise exchange hill‑climbing (same as before but
    with a few more steps).
    """
    a = np.array(seq, dtype=float)
    a = project_to_simplex(a)
    best_a = a.copy()
    best_score = compute_score(best_a)
    n = len(a)
    step = init_step

    while max_steps > 0 and time.time() < deadline:
        i, j = np.random.randint(n, size=2)
        if i == j:
            continue
        delta = step * best_a[i]
        if delta <= 0:
            continue
        cand = best_a.copy()
        cand[i] -= delta
        cand[j] += delta
        if cand[i] < 0:
            continue
        cand = project_to_simplex(cand)
        cur_score = compute_score(cand)
        if cur_score < best_score:
            best_score = cur_score
            best_a = cand
            step = min(step * 1.05, 0.5)
        else:
            step *= 0.98
        max_steps -= 1
    return best_a.tolist(), best_score

  3.  A cheap stochastic pair‑wise mass‑exchange hill‑climb used for
      final polishing.

Several problem sizes (n = 1024, 2048, 4096, 8192) are examined; the
number of inner iterations is scaled with n so that the total time
stays within the 1000 s budget.  The best sequence found is returned as
a plain list of floats – exactly the format required by the evaluation
harness.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Basic linear‑algebra utilities
# ----------------------------------------------------------------------


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of a vector onto the probability simplex."""
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / v.shape[0])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho = np.where(u - cssv / np.arange(1, v.shape[0] + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full convolution using a real FFT (fast)."""
    m = len(a) + len(b) - 1
    n_fft = next_pow2(m)
    A = np.fft.rfft(a, n=n_fft)
    B = np.fft.rfft(b, n=n_fft)
    C = A * B
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


def compute_score(a: np.ndarray) -> float:
    """Score identical to ``evaluate_sequence`` when ∑a = 1."""
    n = len(a)
    conv = convolution_fft(a, a)
    max_val = float(np.max(conv))
    s = float(np.sum(a))
    if s < 1e-12:
        return float("inf")
    return 2.0 * n * max_val / (s * s)


def arcsine_initial(n: int) -> np.ndarray:
    """Arcsine‑shaped start: a_i ∝ 1 / √((i+0.5)(n‑0.5‑i))."""
    i = np.arange(n, dtype=float)
    eps = 0.25
    vals = 1.0 / np.sqrt((i + 0.5) * (n - 0.5 - i) + eps)
    return vals / np.sum(vals)


# ----------------------------------------------------------------------
# Smoothed‑max gradient utilities
# ----------------------------------------------------------------------


def smoothed_gradient(a: np.ndarray, alpha: float):
    """
    Gradient of the smoothed maximum

        f_α(a) = (1/α)·log Σ_k exp(α·(a∗a)_k)

    Returns the gradient (same shape as ``a``) and the raw convolution
    ``a∗a`` (length 2n−1).
    """
    conv = convolution_fft(a, a)                     # length 2n‑1
    maxc = np.max(conv)
    exp_term = np.exp(alpha * (conv - maxc))
    w = exp_term / np.sum(exp_term)                  # soft‑max weights, Σw = 1

    a_rev = a[::-1]
    conv_rev = convolution_fft(a_rev, w)             # length 3n‑2
    start = len(a) - 1
    rev_slice = conv_rev[start:start + len(a)]

    # diagonal contribution – positions where the two indices coincide
    w_diag = w[::2][:len(a)]

    grad = 2.0 * rev_slice - w_diag * a
    return grad, conv


# ----------------------------------------------------------------------
# Hard‑max (sub‑gradient) utilities
# ----------------------------------------------------------------------


def max_gradient(a: np.ndarray, idx: int) -> np.ndarray:
    """
    Exact gradient of the maximal convolution entry.
    ``idx`` is the index where ``a∗a`` attains its maximum.
    """
    n = a.shape[0]
    indices = np.arange(n)
    g = np.zeros_like(a)

    # term a_{k‑i}
    mask1 = (idx - indices >= 0) & (idx - indices < n)
    g[mask1] += a[idx - indices[mask1]]

    # term a_{i‑k}
    mask2 = (indices - idx >= 0) & (indices - idx < n)
    g[mask2] += a[indices[mask2] - idx]

    return g


# ----------------------------------------------------------------------
# Adam optimiser (projected onto the simplex)
# ----------------------------------------------------------------------


def adam_update(x, grad, m, v, t, lr=0.1, beta1=0.9, beta2=0.999, eps=1e-8):
    """One Adam step; returns updated (x, m, v)."""
    m = beta1 * m + (1.0 - beta1) * grad
    v = beta2 * v + (1.0 - beta2) * (grad * grad)

    m_hat = m / (1.0 - beta1 ** t)
    v_hat = v / (1.0 - beta2 ** t)

    step = lr * m_hat / (np.sqrt(v_hat) + eps)
    x_new = x - step
    return x_new, m, v


# ----------------------------------------------------------------------
# Core optimisation routine for a single initialisation
# ----------------------------------------------------------------------


def optimise_one_start(
    initial_a: np.ndarray,
    deadline: float,
    max_iter_smoothed: int = 120_000,
    alpha_start: float = 1.0,
    alpha_mid: float = 200.0,
    alpha_end: float = 1300.0,
    adam_lr: float = 0.15,
    noise_scale: float = 2e-5,
    hard_max_iter: int = 12_000,
):
    """
    Three‑phase optimisation:
      1. Smoothed‑max descent with Adam (α from ``alpha_start`` to ``alpha_mid``).
      2. Continuation with larger α (up to ``alpha_end``) – the objective
         approaches the true max.
      3. A short hard‑max sub‑gradient phase.

    Returns the best vector encountered and its true (unsmoothed) score.
    """
    a = project_to_simplex(initial_a.astype(float).copy())
    n = len(a)

    m = np.zeros_like(a)
    v = np.zeros_like(a)

    best_a = a.copy()
    best_score = compute_score(a)

    # ------------------------------------------------------------------
    # Phase 1 – modest smoothing
    # ------------------------------------------------------------------
    for t in range(1, max_iter_smoothed + 1):
        if time.time() >= deadline:
            break

        alpha = alpha_start + (alpha_mid - alpha_start) * min(1.0, t / max_iter_smoothed)
        grad, _ = smoothed_gradient(a, alpha)

        a, m, v = adam_update(a, grad, m, v, t, lr=adam_lr)

        if t % 700 == 0:                                 # tiny isotropic noise
            a += np.random.randn(n) * noise_scale

        a = project_to_simplex(a)

        if t % 200 == 0:                                 # evaluate only periodically
            cur_score = compute_score(a)
            if cur_score < best_score:
                best_score = cur_score
                best_a = a.copy()

    # ------------------------------------------------------------------
    # Phase 2 – large smoothing (approaching the hard max)
    # ------------------------------------------------------------------
    for t2 in range(1, max_iter_smoothed + 1):
        if time.time() >= deadline:
            break

        alpha = alpha_mid + (alpha_end - alpha_mid) * min(1.0, t2 / max_iter_smoothed)
        grad, _ = smoothed_gradient(a, alpha)

        # keep Adam moments, but shrink the learning rate a little
        a, m, v = adam_update(a, grad, m, v,
                              t2 + max_iter_smoothed,
                              lr=adam_lr * 0.5)

        if t2 % 500 == 0:
            a += np.random.randn(n) * (noise_scale * 0.5)

        a = project_to_simplex(a)

        if t2 % 200 == 0:
            cur_score = compute_score(a)
            if cur_score < best_score:
                best_score = cur_score
                best_a = a.copy()

    # ------------------------------------------------------------------
    # Phase 3 – hard‑max sub‑gradient descent
    # ------------------------------------------------------------------
    step = 0.2
    for it in range(1, hard_max_iter + 1):
        if time.time() >= deadline:
            break
        conv = convolution_fft(a, a)
        idx_max = int(np.argmax(conv))
        g = max_gradient(a, idx_max)

        eta = step / np.sqrt(it)          # decaying step size
        a = a - eta * g
        a = project_to_simplex(a)

        if it % 100 == 0:
            cur_score = compute_score(a)
            if cur_score < best_score:
                best_score = cur_score
                best_a = a.copy()

        if it % 2000 == 0:                # slowly reduce the base step
            step = max(step * 0.9, 0.02)

    return best_a, best_score


# ----------------------------------------------------------------------
# Final cheap stochastic refinement
# ----------------------------------------------------------------------


def hill_climb(seq, deadline, max_steps=60_000, init_step=0.05):
    """
    Stochastic pair‑wise exchange hill‑climbing.
    Moves a small amount of mass from one index to another;
    improvements are kept, otherwise the step size is shrunk.
    """
    a = np.array(seq, dtype=float)
    a = project_to_simplex(a)
    best_a = a.copy()
    best_score = compute_score(best_a)
    n = len(a)
    step = init_step

    while max_steps > 0 and time.time() < deadline:
        i, j = np.random.randint(n, size=2)
        if i == j:
            continue
        delta = step * best_a[i]
        if delta <= 0:
            continue
        cand = best_a.copy()
        cand[i] -= delta
        cand[j] += delta
        if cand[i] < 0:
            continue
        cand = project_to_simplex(cand)
        cur_score = compute_score(cand)
        if cur_score < best_score:
            best_score = cur_score
            best_a = cand
            step = min(step * 1.05, 0.5)
        else:
            step *= 0.98
        max_steps -= 1
    return best_a.tolist(), best_score


# ----------------------------------------------------------------------
# Entry point required by the competition harness
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Searches for a non‑negative step‑function (height sequence) that
    minimises ``evaluate_sequence``.  The routine respects the time
    budget, explores several problem sizes, runs a multi‑start
    optimisation for each, and finally returns the best list of
    floats discovered.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 2.0   # tiny safety margin

    best_seq = None
    best_val = float("inf")

    # ------------------------------------------------------------------
    # Optional fallback from a previously found construction
    # ------------------------------------------------------------------
    try:
        fallback = [float(x) for x in height_sequence_1]   # type: ignore
        fallback_val = evaluate_sequence(fallback)
        print(f"Fallback score = {fallback_val:.6f}")
        if fallback_val < best_val:
            best_val = fallback_val
            best_seq = fallback
    except Exception:
        pass

    # ------------------------------------------------------------------
    # Problem sizes to examine – iteration budget scales with n
    # ------------------------------------------------------------------
    candidate_lengths = [1024, 2048, 4096, 8192]

    for n in candidate_lengths:
        if time.time() >= deadline - 5.0:
            break
        print(f"\n=== Optimising length n = {n} ===")

        # Scale the number of inner iterations so that the total work
        # stays roughly constant irrespective of n.
        base_iter = 150_000
        max_iter = max(10_000, int(base_iter * 1024 / n))
        hard_iter = max(2_000, int(12_000 * 1024 / n))

        # Three different starts per length
        inits = [
            arcsine_initial(n),
            np.full(n, 1.0 / n),
            np.random.dirichlet(np.ones(n)),
        ]

        for start_id, init_vec in enumerate(inits, 1):
            if time.time() >= deadline - 5.0:
                break
            print(f"  start {start_id}/{len(inits)}")
            seq, val = optimise_one_start(
                initial_a=init_vec,
                deadline=deadline,
                max_iter_smoothed=max_iter,
                alpha_start=1.0,
                alpha_mid=200.0,
                alpha_end=1300.0,
                adam_lr=0.15,
                noise_scale=2e-5,
                hard_max_iter=hard_iter,
            )
            print(f"    local best score = {val:.6f}")
            if val < best_val:
                best_val = val
                best_seq = seq.tolist()
                print(f"    *** New global best {best_val:.6f} (n={n}) ***")

    # ------------------------------------------------------------------
    # Final cheap refinement
    # ------------------------------------------------------------------
    if best_seq is not None and time.time() < deadline - 2.0:
        print("\nRefining best sequence with hill‑climb")
        refined_seq, refined_val = hill_climb(
            best_seq, deadline, max_steps=60_000, init_step=0.05
        )
        print(f"    refined score = {refined_val:.6f}")
        if refined_val < best_val:
            best_val = refined_val
            best_seq = refined_seq

    # ------------------------------------------------------------------
    # Fallback if nothing was produced (unlikely)
    # ------------------------------------------------------------------
    if best_seq is None:
        best_seq = [1.0] * 256
        best_val = evaluate_sequence(best_seq)

    final_score = evaluate_sequence(best_seq)
    print(f"\nFinished search. Final reported score = {final_score:.6f}")
    return best_seq
