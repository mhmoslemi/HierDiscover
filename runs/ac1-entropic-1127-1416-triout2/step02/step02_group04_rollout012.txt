"""
Propose a high‑quality step‑function sequence.

Algorithm
---------
1. Determine the problem size ``n`` (length of the global
   ``height_sequence_1`` if it exists, otherwise 5146).

2. Build a few deterministic seeds (global solution, uniform ramp,
   uniform distribution) and keep the best one (objective value
   computed with ``compute_objective`` – the same as ``evaluate_sequence``).

3. **Smooth‑max Adam phase**  
   * Parameterise the heights directly (they always lie on the probability
     simplex, i.e. non‑negative and sum = 1).  
   * Approximate the non‑smooth function
        ``max_k (a*a)[k]``
     by a temperature‑scaled soft‑max:
        ``M_β(a) = (1/β)·log Σ_k exp(β·conv_k)``.
   * The gradient of the smoothed objective
        ``F_β(a)=2·n·M_β(a)`` is
        ``∇F_β(a)=4·n·corr(a,w)`` where
        ``w_k = exp(β·conv_k)/Σ_j exp(β·conv_j)`` and
        ``corr`` denotes the linear cross‑correlation of ``a`` and ``w``.
   * Compute all convolutions / correlations with a single zero‑padded FFT
     (length N ≈ power‑of‑2 ≥ 3·n‑2).  
   * Use Adam with a diminishing learning‑rate and an exponential schedule
     for the temperature β (β grows from 1 up to 200).  After each step the
     iterate is projected onto the simplex (Euclidean projection).  The
     best feasible vector seen so far (measured with the *exact* objective)
     is stored.

4. **Projected sub‑gradient refinement**  
   Starting from the best vector discovered in step 3, run a classic
   projected sub‑gradient descent on the true max‑convolution objective.
   The sub‑gradient is the average of the exact gradients of all
   convolution entries that are within a tiny tolerance of the current
   maximum.  A diminishing step‑size ``η/√t`` is used and every iterate is
   projected onto the simplex.

5. Return the best sequence (plain list of floats) found before the
   time limit (1000 s).  Progress and intermediate bounds are printed.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities
# ----------------------------------------------------------------------


def simplex_projection(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the probability simplex:
    {w | w_i >= 0 , Σ w_i = 1}.
    Duchi et al. (2008) implementation.
    """
    v = np.asarray(v, dtype=float)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1] - 1
    theta = cssv[cond][-1] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def compute_objective(a: np.ndarray) -> float:
    """
    Exact objective for a vector whose entries sum to 1:
        2·n·max(a*a)
    """
    n = a.shape[0]
    # linear convolution a*a, length 2*n-1
    L = 2 * n - 1
    fa = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(fa * fa, n=L)
    return float(2.0 * n * conv.max())


# ----------------------------------------------------------------------
# Main search routine
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 0, budget_s: int = 1000) -> list[float]:
    """
    Search for a non‑negative height sequence that minimises
    ``evaluate_sequence``.  The routine respects the given time budget
    (default 1000 s) and returns the best sequence found.
    """
    np.random.seed(seed)
    start_time = time.time()
    # safety margin (few seconds are lost for printing / cleanup)
    deadline = start_time + budget_s - 5.0

    # --------------------------------------------------------------
    # 1. Problem size
    # --------------------------------------------------------------
    try:
        n = len(height_sequence_1)  # type: ignore[name-defined]
    except NameError:
        n = 5146

    print(f"Problem size: n = {n}")

    # --------------------------------------------------------------
    # 2. Initialise a few deterministic seeds
    # --------------------------------------------------------------
    best_a = None
    best_score = np.inf

    # helper to evaluate and possibly update the incumbent
    def try_candidate(a_vec: np.ndarray, name: str) -> None:
        nonlocal best_a, best_score
        a_norm = np.maximum(a_vec, 0.0)
        if a_norm.sum() == 0.0:
            return
        a_norm = a_norm / a_norm.sum()
        score = compute_objective(a_norm)
        if score < best_score:
            best_score = score
            best_a = a_norm.copy()
            print(f"{name:20s} -> new best {best_score:.6f}")

    # a) use the previously discovered global sequence if available
    try:
        cand = np.asarray(height_sequence_1, dtype=float)  # type: ignore[name-defined]
        try_candidate(cand, "global height_sequence_1")
    except NameError:
        pass

    # b) uniform distribution
    try_candidate(np.full(n, 1.0 / n), "uniform")

    # c) simple decreasing ramp (analytic construction)
    ramp = np.arange(n, 0, -1, dtype=float)
    ramp = ramp / ramp.sum()
    try_candidate(ramp, "linear ramp")

    # --------------------------------------------------------------
    # 3. Adam optimisation on a smooth approximation of the max
    # --------------------------------------------------------------
    if best_a is None:
        # fallback – uniform
        best_a = np.full(n, 1.0 / n)

    a = best_a.copy()
    # pre‑computed FFT length large enough for both convolution (2n‑1)
    # and correlation (n + (2n‑1) - 1 = 3n‑2)
    corr_len = 3 * n - 2
    N = 1 << (corr_len - 1).bit_length()  # next power of two

    # Adam hyper‑parameters
    lr = 0.05
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1_pow, beta2_pow = 1.0, 1.0

    # temperature schedule (soft‑max)
    beta0 = 1.0
    beta_max = 200.0
    max_adam_iters = 120000  # generous upper bound, actual stop is time‑driven

    # allocate once to avoid repeated allocations
    conv_len = 2 * n - 1

    # we split the time: ~65 % for Adam, remainder for sub‑gradient fine‑tuning
    adam_deadline = start_time + 0.65 * (budget_s - 5.0)

    it = 0
    while time.time() < adam_deadline and it < max_adam_iters:
        it += 1

        # ---- FFT of the current iterate ---------------------------------
        fa = np.fft.rfft(a, n=N)

        # ---- linear convolution a*a (full length N, we only need first 2n‑1) ----
        conv_full = np.fft.irfft(fa * fa, n=N)
        conv = conv_full[:conv_len]

        max_b = conv.max()
        cur_score = float(2.0 * n * max_b)

        # keep the best exact score
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"Adam iter {it:6d}: new best {best_score:.6f}")

        # ---- smooth‑max weight vector w ----------------------------------
        # soft‑max over the convolution entries, shifted by the maximum for stability
        max_c = max_b
        exp_term = np.exp(beta0 * (conv - max_c))
        w = exp_term / exp_term.sum()      # shape (2n‑1)

        # ---- cross‑correlation a ⋆ w (linear) via FFT --------------------
        fw = np.fft.rfft(w, n=N)
        corr_full = np.fft.irfft(fa * np.conj(fw), n=N)   # length N
        # valid part of linear correlation starts at offset (len(w)-1) = 2n‑2
        start = conv_len - 1                     # 2*n-2
        grad = 4.0 * n * corr_full[start:start + n]   # shape (n)

        # ---- Adam update (with bias correction) ------------------------
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        beta1_pow *= beta1
        beta2_pow *= beta2
        m_hat = m / (1.0 - beta1_pow)
        v_hat = v / (1.0 - beta2_pow)

        a = a - lr * m_hat / (np.sqrt(v_hat) + eps)

        # projection onto the simplex (guarantees feasibility)
        a = simplex_projection(a)

        # ---- temperature schedule (β grows exponentially) ---------------
        prog = it / max_adam_iters
        beta0 = beta0 * (beta_max / beta0) ** prog
        if beta0 > beta_max:
            beta0 = beta_max

    # --------------------------------------------------------------
    # 4. Projected sub‑gradient descent on the *true* max‑convolution
    # --------------------------------------------------------------
    a = best_a.copy()
    eta = 0.8
    subgrad_iters = 20000
    it = 0
    while time.time() < deadline and it < subgrad_iters:
        # convolution a*a (full length)
        fa = np.fft.rfft(a, n=N)
        conv_full = np.fft.irfft(fa * fa, n=N)
        conv = conv_full[:conv_len]

        max_b = conv.max()
        cur_score = float(2.0 * n * max_b)

        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"Sub‑grad iter {it:6d}: new best {best_score:.6f}")

        # active set (indices where conv is within 1e‑5 relative of the max)
        eps_active = max_b * 1e-5
        active = np.where(conv >= max_b - eps_active)[0]
        if active.size == 0:
            active = np.array([conv.argmax()], dtype=int)

        # exact sub‑gradient of the max‑convolution
        g = np.zeros_like(a)
        for k in active:
            lo = max(0, k - n + 1)
            hi = min(n - 1, k)
            idx = np.arange(lo, hi + 1, dtype=int)
            partner = k - idx
            g[idx] += 2.0 * a[partner]
        g /= float(active.size)

        # diminishing step‑size and projection
        step = eta / np.sqrt(it + 1.0)
        a = simplex_projection(a - step * g)

        it += 1

    # --------------------------------------------------------------
    # 5. Return the best sequence found
    # --------------------------------------------------------------
    print(f"Finished. Best score = {best_score:.6f} after {it} sub‑grad steps "
          f"and {max_adam_iters if it==max_adam_iters else it} Adam steps.")
    return [float(v) for v in best_a.tolist()]
