* It targets the *variance* of the auto‑convolution (f*f) instead of the
  active‑set sub‑gradient of the max‑functional.
* The gradient of the variance can be computed by two FFT‑based
  convolutions, giving an O(n log n) iteration cost.
* Random Dirichlet seeds, a simple momentum scheme and occasional
  jitter are used to avoid local minima.
* A final linear‑programming refinement (optional, only if SciPy is
  available) follows the same idea as the published approach.

The routine respects the 1000 s budget, prints progress and returns the
best sequence as a plain Python list.
"""

import time
import numpy as np

# --------------------------------------------------------------------------- #
def projection_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Project a vector onto the simplex {x ≥ 0, Σx = s} (Duchi et al. 2008)."""
    if s < 0:
        raise ValueError("target sum must be non‑negative")
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho_arr = np.where(u - cssv / (np.arange(n) + 1.0) > 0)[0]
    if rho_arr.size == 0:
        return np.zeros_like(v)
    rho = rho_arr[-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# --------------------------------------------------------------------------- #
def _build_toeplitz_matrix(f: np.ndarray):
    """Sparse Toeplitz matrix A such that A·g = f * g (linear convolution)."""
    n = f.shape[0]
    rows = np.empty(n * n, dtype=np.int32)
    cols = np.empty(n * n, dtype=np.int32)
    data = np.empty(n * n, dtype=np.float64)

    ptr = 0
    for i in range(n):
        rows[ptr:ptr + n] = np.arange(i, i + n)          # row indices i … i+n‑1
        cols[ptr:ptr + n] = i                           # column i
        data[ptr:ptr + n] = f
        ptr += n

    from scipy.sparse import csr_matrix
    return csr_matrix((data, (rows, cols)), shape=(2 * n - 1, n))


# --------------------------------------------------------------------------- #
def _conv_self(f: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution f * f (length 2·n‑1) using a pre‑chosen FFT length."""
    F = np.fft.rfft(f, n=fft_len)
    conv = np.fft.irfft(F * F, n=fft_len)[: 2 * f.shape[0] - 1]
    return conv


# --------------------------------------------------------------------------- #
def propose_candidate(seed: int = 42, budget_s: float = 1000, **kwargs):
    """
    Search for a non‑negative step‑function that minimises ``evaluate_sequence``.

    The search consists of:
      1. Random Dirichlet seeds (sum fixed to √(2 n)).
      2. Gradient descent on the variance of the auto‑convolution.
      3. Optional LP refinement (if SciPy is present).
    """
    np.random.seed(seed)
    start = time.time()
    deadline = start + budget_s - 0.2   # small safety margin

    # ------------------------------------------------------------------- #
    # Problem size – a few thousand entries give a good trade‑off.
    # Larger n reduces the theoretical lower bound (≈1) but costs a bit more.
    n = 3000
    target_sum = np.sqrt(2.0 * n)

    # FFT length sufficient for convolutions with vectors up to length 3·n‑2.
    fft_len = 1 << ((3 * n - 2).bit_length())

    # ------------------------------------------------------------------- #
    # Helper for a very fast score (uses the fact that the ℓ₁‑norm is fixed).
    def fast_score(vec: np.ndarray) -> float:
        conv = _conv_self(vec, fft_len)
        return float(2.0 * n * conv.max() / (target_sum ** 2))

    # ------------------------------------------------------------------- #
    # 1️⃣  Random seeding – keep the best of a modest pool.
    best_seq = None
    best_val = float("inf")
    seed_pool = 60                     # number of random starts

    for i in range(seed_pool):
        if time.time() >= deadline:
            break
        a = np.random.exponential(scale=1.0, size=n)
        a = a / a.sum() * target_sum    # project onto the ℓ₁‑simplex (exactly)
        sc = fast_score(a)
        if sc < best_val:
            best_val, best_seq = sc, a.copy()
            print(f"[{time.time() - start:6.1f}s] seed {i:02d} → score {sc:.6f}")

    # also try the uniform vector and (if available) the previous solution
    uniform = np.full(n, target_sum / n)
    sc = fast_score(uniform)
    if sc < best_val:
        best_val, best_seq = sc, uniform.copy()
        print(f"[{time.time() - start:6.1f}s] uniform seed → score {sc:.6f}")

    try:
        global height_sequence_1
        if isinstance(height_sequence_1, (list, tuple, np.ndarray)):
            hs = np.asarray(height_sequence_1, dtype=np.float64)
            if hs.size == n:
                hs = np.clip(hs, 0.0, 1000.0)
                hs = hs * (target_sum / hs.sum())
                sc = fast_score(hs)
                if sc < best_val:
                    best_val, best_seq = sc, hs.copy()
                    print(f"[{time.time() - start:6.1f}s] previous seed → score {sc:.6f}")
    except Exception:
        pass

    # ------------------------------------------------------------------- #
    # 2️⃣  Gradient descent on the variance of f*f.
    #    Objective: minimise  Σ ( (f*f)_k – μ )²  where μ = mean(conv).
    curr = best_seq.copy()
    best = curr.copy()
    best_val = fast_score(best)

    # momentum & step‑size parameters
    velocity = np.zeros_like(curr)
    step = 0.5 * target_sum / n          # absolute step size (will decay)
    min_step = 1e-12
    momentum = 0.9
    it = 0

    while time.time() < deadline:
        # ----- convolution of the current iterate -------------------------
        # Re‑use the FFT of the current vector as much as possible.
        F_f = np.fft.rfft(curr, n=fft_len)
        conv = np.fft.irfft(F_f * F_f, n=fft_len)[: 2 * n - 1]

        max_val = conv.max()
        score = float(2.0 * n * max_val / (target_sum ** 2))

        # update global best
        if score < best_val - 1e-9:
            best_val, best = score, curr.copy()
            print(f"[{time.time() - start:6.1f}s] iter {it:07d} → new best {score:.6f}")
            # tighten step after a successful improvement
            step = max(step * 0.9, min_step)

        # ----- gradient of the variance ----------------------------------
        mu = conv.mean()
        resid = conv - mu                     # weight vector w_k = conv_k – μ

        # FFT of the weight vector (length 2n‑1)
        F_r = np.fft.rfft(resid, n=fft_len)

        # term a = f * w   (full convolution, then take first n entries)
        conv_a = np.fft.irfft(F_f * F_r, n=fft_len)[: 3 * n - 2]
        term_a = conv_a[:n]

        # term b = f * w_rev ; we need entries (2n‑2‑j)
        resid_rev = resid[::-1]
        F_r_rev = np.fft.rfft(resid_rev, n=fft_len)
        conv_b = np.fft.irfft(F_f * F_r_rev, n=fft_len)[: 3 * n - 2]
        term_b = conv_b[(2 * n - 2) - np.arange(n)]

        grad = term_a + term_b                # ∇ Σ (conv−μ)²  up to a constant factor
        gnorm = np.linalg.norm(grad)
        if gnorm == 0.0:
            break

        # ----- Nesterov‑type momentum step --------------------------------
        velocity = momentum * velocity - step * (grad / gnorm)
        tentative = curr + velocity
        tentative = np.clip(tentative, 0.0, 1000.0)
        curr = projection_simplex(tentative, target_sum)

        # ----- occasional random jitter (helps escape plateaus) ----------
        if np.random.rand() < 0.012:
            jitter = np.random.randn(n) * step * 0.5
            curr = projection_simplex(np.clip(curr + jitter, 0.0, 1000.0),
                                      target_sum)

        it += 1

        # decay step size slowly
        if it % 800 == 0:
            step = max(step * 0.85, min_step)

    # ------------------------------------------------------------------- #
    # 3️⃣  Optional linear‑programming refinement (highs solver from SciPy)
    try:
        from scipy.optimize import linprog
        from scipy.sparse import csr_matrix
    except Exception:
        linprog = None

    if linprog is not None and time.time() < deadline - 5.0:
        f = best.copy()
        for lp_iter in range(12):
            if time.time() >= deadline - 5.0:
                break
            conv_f = _conv_self(f, fft_len)
            M = conv_f.max()
            A = _build_toeplitz_matrix(f)               # shape (2n‑1, n)
            b = np.full(2 * n - 1, M)

            res = linprog(c=-np.ones(n), A_ub=A, b_ub=b,
                          bounds=[(0, None)] * n,
                          method='highs',
                          options={'presolve': True, 'time_limit': 30})
            if not res.success or res.x is None:
                break

            g = res.x
            if g.sum() < 1e-12:
                break
            g = g * (target_sum / g.sum())

            t = 0.18
            f_new = (1.0 - t) * f + t * g
            f_new = projection_simplex(f_new, target_sum)

            conv_new = _conv_self(f_new, fft_len)
            Mnew = conv_new.max()
            score_new = float(2.0 * n * Mnew / (target_sum ** 2))

            if score_new < best_val - 1e-8:
                best_val, best = score_new, f_new.copy()
                print(f"[{time.time() - start:6.1f}s] LP‑iter {lp_iter} → best {score_new:.6f}")

            # accept the LP update if it does not increase the maximal
            # convolution (tiny tolerance to avoid cycling)
            if Mnew <= M * 0.999:
                f = f_new
            else:
                break

    # ------------------------------------------------------------------- #
    # Final check with the official evaluator.
    final_score = evaluate_sequence(best.tolist())
    print(f"Search finished – final score reported by evaluate_sequence: {final_score:.6f}")

    return best.tolist()
