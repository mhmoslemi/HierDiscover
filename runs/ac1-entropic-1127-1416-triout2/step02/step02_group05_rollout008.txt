import numpy as np, time

def project_onto_simplex(v):
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    rho = np.nonzero(u - cssv / np.arange(1, n+1) > 0)[0]
    if rho.size == 0:
        theta = 0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0)
    return w

def smooth_max_gradient(a, beta, L):
    # Compute convolution a*a via FFT
    fft_a = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(fft_a * fft_a, n=L)[:2*len(a)-1]
    max_val = conv.max()
    # Softmax weights
    # shift to improve numeric stability
    exp_weights = np.exp(beta * (conv - max_val))
    w = exp_weights / exp_weights.sum()
    # correlation gradient
    fft_w = np.fft.rfft(w, n=L)
    # correlation a ⋆ w = ifft(fft_a * conj(fft_w))
    corr_full = np.fft.irfft(fft_a * np.conj(fft_w), n=L)[:len(a) + len(w) - 1]  # length 3n-2
    n = len(a)
    # gradient g_k = 2 * corr_full[2*n-2 - k]
    g = 2.0 * corr_full[2*n-2 - np.arange(n)]
    return conv, max_val, w, g

def run_once(n, seed, time_limit):
    np.random.seed(seed)
    # init a: either binomial or random
    if np.random.rand() < 0.5:
        # binomial
        a = np.empty(n, dtype=np.float64)
        a[0] = 1.0
        for k in range(1, n):
            a[k] = a[k-1] * ((2*k - 1)*(2*k)) / (4.0*k*k)
        a = a / a.sum()
    else:
        # random Dirichlet
        a = np.random.rand(n)
        a = a / a.sum()
    # ensure sum ~1
    a = a.astype(np.float64)
    best_a = a.copy()
    best_score = float('inf')
    start = time.time()
    deadline = start + time_limit - 0.05

    # Precompute length for FFT
    L = 1 << int(np.ceil(np.log2(3 * n - 2)))  # power of two >= 3n-2
    step = 0.2
    step_decay = 0.9998
    min_step = 1e-5
    beta = 5.0
    beta_factor = 1.001
    beta_max = 200.0
    momentum = np.zeros_like(a)
    mu = 0.9

    iter_cnt = 0
    while time.time() < deadline:
        conv, max_val, w, grad = smooth_max_gradient(a, beta, L)
        # objective
        sum_a = a.sum()
        cur_score = 2.0 * n * max_val / (sum_a ** 2)
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[improve] iter={iter_cnt}, score={best_score:.9f}, time={time.time() - start:.2f}s")
        # gradient descent step (move opposite to gradient)
        momentum = mu * momentum + step * grad
        a_candidate = a - momentum
        # Random mass transfer for exploration
        if np.random.rand() < 0.002:
            i, j = np.random.randint(0, n, size=2)
            if i != j and a_candidate[i] > 0:
                delta = 0.01 * np.random.rand() * a_candidate[i]
                a_candidate[i] -= delta
                a_candidate[j] += delta
        # project onto simplex
        a = project_onto_simplex(a_candidate)
        # update parameters
        step = max(min_step, step * step_decay)
        beta = min(beta_max, beta * beta_factor)
        iter_cnt += 1
    return best_a, best_score, iter_cnt

"""
Propose a candidate step‑function using a smooth‑max projected gradient
descent with simplex projection and occasional random perturbations.
The algorithm works on the probability simplex (∑a_i = 1, a_i ≥ 0) and
optimises a smooth approximation of   max_i (a*a)_i   (convolution of `a`
with itself).  A softmax with an annealing temperature β turns the
nonsmooth max into a differentiable loss; its gradient is obtained from a
correlation that is computed efficiently with FFTs (O(L log L) per
iteration).  Momentum, step‑size decay, and a simple acceptance rule give
a stable descent while random mass transfers help escape local minima.
The routine runs as many independent restarts as time permits (the
overall budget is given by `budget_s`) and returns the best sequence
found.
"""

import time
import numpy as np


def _project_onto_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of a vector onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    rho = np.nonzero(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def _smooth_max_gradient(a: np.ndarray, beta: float, L: int):
    """
    Compute:
      * the convolution a*a,
      * a soft‑max weight vector w over the convolution entries,
      * the gradient of the smooth‑max loss (log‑sum‑exp) w.r.t. a.

    Returns (conv, max_val, w, grad).
    """
    # FFT of a (real‑valued, zero‑padded to length L)
    fft_a = np.fft.rfft(a, n=L)

    # linear convolution a*a (length 2n‑1)
    conv = np.fft.irfft(fft_a * fft_a, n=L)[: 2 * len(a) - 1]

    max_val = conv.max()

    # soft‑max over convolution entries – shift by max for numerical stability
    exp_w = np.exp(beta * (conv - max_val))
    w = exp_w / exp_w.sum()

    # correlation a ⋆ w = ifft(fft_a * conj(fft_w))
    fft_w = np.fft.rfft(w, n=L)
    corr_full = np.fft.irfft(fft_a * np.conj(fft_w), n=L)
    corr_full = corr_full[: len(a) + len(w) - 1]          # length 3n‑2

    n = len(a)
    # gradient g_k = 2 * corr_full[2n‑2‑k]  (derived from d/d a_k)
    grad = 2.0 * corr_full[2 * n - 2 - np.arange(n)]
    return conv, max_val, w, grad


def _run_single_optimisation(n: int, seed: int, time_limit: float):
    """
    Run the smooth‑max descent for `time_limit` seconds on a vector of length `n`.
    Returns the best sequence, its score and the number of performed iterations.
    """
    np.random.seed(seed)

    # ---- initial point -------------------------------------------------
    if np.random.rand() < 0.5:
        # central‑binomial (Catalan) start
        a = np.empty(n, dtype=np.float64)
        a[0] = 1.0
        for k in range(1, n):
            a[k] = a[k - 1] * ((2 * k - 1) * (2 * k)) / (4.0 * k * k)
        a /= a.sum()
    else:
        # random Dirichlet start – breaks symmetry from the outset
        a = np.random.rand(n)
        a /= a.sum()

    best_a = a.copy()
    best_score = float('inf')

    # ---- optimisation parameters ---------------------------------------
    start = time.time()
    deadline = start + time_limit - 0.05

    # FFT length large enough for all needed linear convolutions (3n‑2)
    L = 1 << int(np.ceil(np.log2(3 * n - 2)))

    step = 0.25          # initial step magnitude
    step_decay = 0.9997
    min_step = 1e-5

    beta = 5.0           # soft‑max temperature (low → smooth)
    beta_factor = 1.001
    beta_max = 200.0

    momentum = np.zeros_like(a)
    mu = 0.9             # momentum coefficient

    it = 0
    while time.time() < deadline:
        # ----- current loss and gradient --------------------------------
        conv, max_val, w, grad = _smooth_max_gradient(a, beta, L)
        cur_score = 2.0 * n * max_val / (a.sum() ** 2)

        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[improve] iter={it}, score={best_score:.9f}, "
                  f"time={time.time() - start:.2f}s")

        # ----- momentum step + projection --------------------------------
        momentum = mu * momentum + step * grad
        a_candidate = a - momentum
        a_candidate = _project_onto_simplex(a_candidate)

        # ----- occasional random mass transfer ---------------------------
        if np.random.rand() < 0.002:
            i, j = np.random.randint(0, n, size=2)
            if i != j and a_candidate[i] > 0.0:
                delta = 0.02 * np.random.rand() * a_candidate[i]
                a_candidate[i] -= delta
                a_candidate[j] += delta
                a_candidate = _project_onto_simplex(a_candidate)

        # ----- compute candidate loss to decide acceptance ---------------
        fft_cand = np.fft.rfft(a_candidate, n=L)
        conv_cand = np.fft.irfft(fft_cand * fft_cand, n=L)[: 2 * n - 1]
        max_cand = conv_cand.max()
        cand_score = 2.0 * n * max_cand / (a_candidate.sum() ** 2)

        # Accept if improvement, otherwise keep the old point (with a tiny
        # chance to accept a worse move for exploration).
        if cand_score < cur_score or np.random.rand() < 0.001:
            a = a_candidate
            # reuse FFT for next iteration
            # (computed again at the top of the loop)
        else:
            # rejected – reset momentum to avoid drifting in a bad direction
            momentum = np.zeros_like(a)

        # ----- schedule updates -------------------------------------------
        step = max(min_step, step * step_decay)
        beta = min(beta_max, beta * beta_factor)
        it += 1

    return best_a, best_score, it

    """
    Main entry point.  Repeatedly launches short optimisation runs with
    different random seeds until the overall time budget expires.
    The best sequence seen is returned as a plain Python list.
    """
    deadline = time.time() + budget_s - 0.1  # tiny safety margin

    # Use the length of a previously found construction if available.
    try:
        n = len(height_sequence_1)  # type: ignore[name-defined]
