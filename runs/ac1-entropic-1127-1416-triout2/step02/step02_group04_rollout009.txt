import numpy as np

def simplex_projection(v):

def compute_objective(a):

def geometric_sequence(n, r):
    i = np.arange(n, dtype=float)
    a = np.power(r, i, dtype=float)   # r**i
    total = a.sum()
    if total == 0:
        a = np.full(n, 1.0/n)
        total = 1.0
    return a / total

def evaluate_r(r):
    a = geometric_sequence(n, r)
    return compute_objective(a), a

def propose_candidate(seed=0, budget_s=1000):
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 2.0

    # determine n
    try:
        n = len(height_sequence_1)
    except NameError:
        n = 5146

    best_a = None
    best_score = np.inf

    # baseline: uniform
    a_uniform = np.full(n, 1.0/n)
    score = compute_objective(a_uniform)
    best_a, best_score = a_uniform, score
    print(f"Uniform baseline score {score:.6f}")

    # geometric candidate
    # analytic guess
    if n > 2:
        r_guess = (n - 1) / (n + 1)
    else:
        r_guess = 0.5
    # ensure r_guess < 1
    r_guess = min(r_guess, 0.999999)
    # search around r_guess
    # We'll evaluate points on log scale perhaps
    r_candidates = [r_guess]
    # generate extra points
    for factor in np.linspace(0.9, 1.1, 9):
        r = r_guess * factor
        if r <= 0:
            continue
        if r >= 1:
            r = 0.999999
        r_candidates.append(r)

    # Also evaluate some fixed values: 0.95, 0.97, 0.99
    for r in [0.95, 0.97, 0.985, 0.99, 0.992, 0.996]:
        r_candidates.append(r)

    # Clip duplicates and bound
    r_candidates = sorted(set(r_candidates))

    for r in r_candidates:
        if time.time() > deadline:
            break
        a = geometric_sequence(n, r)
        s = compute_objective(a)
        if s < best_score:
            best_score, best_a = s, a.copy()
            print(f"Geometric r={r:.6f} improved to {s:.6f}")

    # optionally gradient descent refinement
    a = best_a.copy()
    iteration = 0
    eta = 0.8
    while time.time() < deadline and iteration < 5000:
        # compute conv
        fft = np.fft.rfft(a, n=2*n-1)
        conv = np.fft.irfft(fft*fft, n=2*n-1)
        max_b = conv.max()
        cur_score = 2.0 * n * max_b
        if cur_score < best_score - 1e-9:
            best_score = cur_score
            best_a = a.copy()
            print(f"Iter {iteration:04d}: new best {best_score:.6f}")
        # active indices
        eps = max_b * 1e-6
        active = np.where(conv >= max_b - eps)[0]
        if active.size == 0:
            active = np.array([conv.argmax()], dtype=int)
        # compute subgradient
        g = np.zeros_like(a)
        for k in active:
            low = max(0, k - n + 1)
            high = min(n-1, k)
            idx = np.arange(low, high+1, dtype=int)
            partner = k - idx
            g[idx] += 2.0 * a[partner]
        g /= float(active.size)
        step = eta / np.sqrt(iteration + 1)
        a = simplex_projection(a - step * g)
        iteration += 1

    # final best a
    print(f"Finished after {iteration} iterations, best score = {best_score:.6f}")
    return [float(v) for v in best_a.tolist()]

"""
propose_candidate
-----------------

Generate a step‑function that minimises the evaluation function

    2·n·max(conv(a,a)) / (sum(a)^2)

by exploiting the analytic structure of the problem.  For a fixed length
``n`` the optimum (ignoring the simplex constraint) is attained by a
geometric progression

    a_i ∝ r^i ,   i = 0,…,n‑1

1. Determine the length ``n`` (uses the global ``height_sequence_1`` if

2. Build a small set of geometric candidates around the analytic value
   ``r0 = (n‑1)/(n+1)`` and keep the best one (objective is evaluated

3. Starting from the best geometric seed run a short projected
   sub‑gradient descent on the simplex to polish the solution.
   The sub‑gradient is averaged over all convolution entries that are
   within a tiny relative tolerance of the current maximum.

4. Return the best sequence found as a plain Python list of floats.

import time
import numpy as np


def simplex_projection(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of a vector onto the probability simplex
    { w | w_i >= 0 , sum w_i = 1 }.
    Implements the algorithm of Duchi et al. (2008).
    """
    v = np.asarray(v, dtype=float)
    n = v.shape[0]

    # sort v in decreasing order
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all non‑positive – return the uniform distribution
        return np.full_like(v, 1.0 / n)

    rho = ind[cond][-1] - 1            # zero‑based index
    theta = cssv[cond][-1] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def compute_objective(a: np.ndarray) -> float:
    """Return the exact evaluation value for a vector ``a`` that already
    satisfies ``sum(a) == 1``.
    The objective equals ``2·n·max(conv(a,a))``.
    """
    n = a.shape[0]
    # linear convolution via FFT of length 2·n‑1
    fft = np.fft.rfft(a, n=2 * n - 1)
    conv = np.fft.irfft(fft * fft, n=2 * n - 1)
    return float(2.0 * n * conv.max())


def geometric_sequence(n: int, r: float) -> np.ndarray:
    """Return a normalised geometric sequence a_i ∝ r**i  (i=0…n‑1)."""
    if r <= 0.0:

        return np.full(n, 1.0 / n, dtype=float)

    a = np.exp(i * np.log(r))          # r**i but numerically stable
    total = a.sum()
    if total == 0.0 or not np.isfinite(total):
        # safeguard – uniform distribution
        return np.full(n, 1.0 / n, dtype=float)
    return a / total
