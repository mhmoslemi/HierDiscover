"""
Propose a candidate step‑function by exploiting the analytic form of the
optimal power‑law shape a_i ∝ (i+1)^(-p).  For a fixed length n the
maximal self‑convolution is attained at the centre of the convolution
array and asymptotically equals

    max(a*a) ≈ q²·B(q,q) / n,   q = 1‑p,

where B is the Beta function.  The resulting bound
    2·n·max(a*a) = 2·q²·B(q,q)

is minimized for q≈0.42 (p≈0.58) when n≈5 000.  The routine therefore
searches a small interval of exponents around this value, evaluates
the exact objective using an FFT‑based convolution, keeps the best
candidate, and finally refines it with a few projected sub‑gradient
steps.  The whole search respects the 1000 s budget and returns a
plain Python list suitable for ``evaluate_sequence``.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities (all top level, no closures)
# ----------------------------------------------------------------------


def project_onto_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the probability simplex:
        { x | x >= 0, sum(x) = 1 }.
    Returns a new array (the input is not modified).
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]                # sorted in descending order
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):                # all entries zero – fall back to uniform
        return np.full_like(v, 1.0 / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def full_convolution(a: np.ndarray) -> np.ndarray:
    """
    Compute the linear convolution a * a using an FFT.
    The length of the result is 2*len(a)-1.
    """
    n = a.shape[0]
    size = 2 * n
    fft_len = 1 << (size - 1).bit_length()   # next power of two
    fft_a = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(fft_a * fft_a, n=fft_len)
    return conv[: 2 * n - 1]


def gradient_at_max(a: np.ndarray, k: int) -> np.ndarray:
    """
    Sub‑gradient of the maximal entry of the convolution (at shift k)
    with respect to the coefficient vector a.
    """
    n = a.shape[0]
    g = np.zeros_like(a)
    low = max(0, k - n + 1)
    high = min(k, n - 1)
    if low > high:
        return g
    idx = np.arange(low, high + 1, dtype=int)
    g[idx] = a[k - idx]
    return g


def objective_score(a: np.ndarray) -> float:
    """
    Exact value of the evaluation function for a vector that already
    satisfies sum(a) == 1:
        2 * n * max(conv(a, a))
    """
    n = a.shape[0]
    conv = full_convolution(a)
    return 2.0 * n * np.max(conv)


def power_law_sequence(n: int, p: float, offset: float = 1.0) -> np.ndarray:
    """
    Return a non‑negative sequence of length n with entries proportional to
    (i + offset)^{-p}, normalised so that the sum equals 1.
    """
    x = np.arange(n, dtype=float) + offset
    a = np.power(x, -p)
    a_sum = a.sum()
    a /= a_sum
    return a


# ----------------------------------------------------------------------
# Main search routine
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a height sequence that minimises

        2 * n * max(conv(a, a)) / (sum(a) ** 2)

    using a focused exploration of power‑law shapes followed by a brief
    projected sub‑gradient polishing phase.
    """
    np.random.seed(seed)
    # reserve a few seconds for printing and returning the answer
    deadline = time.time() + budget_s - 5.0

    # ------------------------------------------------------------------
    # Parameters
    # ------------------------------------------------------------------
    n = 5000                                     # length of the step‑function

    best_seq = None
    best_score = np.inf
    best_p = None

    # ------------------------------------------------------------------
    # 1) Coarse grid over the exponent p
    # ------------------------------------------------------------------
    print("=== Coarse grid search over exponent p ===")
    for p in np.linspace(0.45, 0.65, 21):        # step 0.01
        a = power_law_sequence(n, p)
        score = objective_score(a)
        if score < best_score:
            best_score = score
            best_seq = a
            best_p = p
            print(f"new best: p={p:.3f}  score={score:.6f}")

    # ------------------------------------------------------------------
    # 2) Fine refinement around the best exponent found
    # ------------------------------------------------------------------
    if best_p is not None:
        print("\n=== Fine grid refinement around p={:.6f} ===".format(best_p))
        fine_start = max(0.30, best_p - 0.05)
        fine_end = min(0.80, best_p + 0.05)
        # use a very fine step (≈0.001)
        for p in np.arange(fine_start, fine_end + 1e-12, 0.001):
            a = power_law_sequence(n, p)
            score = objective_score(a)
            if score < best_score:
                best_score = score
                best_seq = a
                best_p = p
                print(f"refined best: p={p:.4f}  score={score:.6f}")

    # ------------------------------------------------------------------
    # 3) Optional local random perturbations (tiny jitter)
    # ------------------------------------------------------------------
    print("\n=== Random jitter phase ===")
    jitter_iter = 0
    while time.time() < deadline and jitter_iter < 5000:
        # pick two random indices and move a tiny amount of mass
        i, j = np.random.randint(0, n, size=2)
        if i == j:
            continue
        delta = 0.001 * best_seq[i] * (np.random.rand() - 0.5)   # up to 0.05 % of mass
        cand = best_seq.copy()
        cand[i] = max(cand[i] - delta, 0.0)
        cand[j] = cand[j] + delta
        cand = project_onto_simplex(cand)
        cand_score = objective_score(cand)
        if cand_score < best_score:
            best_score = cand_score
            best_seq = cand
            print(f"jitter improvement at iter {jitter_iter}: score={cand_score:.6f}")
        jitter_iter += 1

    # ------------------------------------------------------------------
    # 4) Short projected sub‑gradient polishing starting from the best
    # ------------------------------------------------------------------
    print("\n=== Sub‑gradient polishing ===")
    a = best_seq.copy()
    step_initial = 0.4
    iteration = 0
    while time.time() < deadline and iteration < 20000:
        conv = full_convolution(a)
        k_max = int(np.argmax(conv))
        g = gradient_at_max(a, k_max)
        step = step_initial / np.sqrt(iteration + 1.0)
        a_candidate = a - step * g
        a = project_onto_simplex(a_candidate)

        # recompute exact score
        score = objective_score(a)
        if score < best_score:
            best_score = score
            best_seq = a.copy()
            print(f"sub‑grad iter {iteration:05d}: improved score = {best_score:.6f}")
            # stop early if we already beat the target comfortably
            if best_score < 1.45:
                break
        iteration += 1

    # ------------------------------------------------------------------
    # 5) Safety fallback – use a previous construction if nothing found
    # ------------------------------------------------------------------
    if best_seq is None or best_seq.size == 0:
        try:
            # ``height_sequence_1`` may be provided by the surrounding environment
            best_seq = np.array(height_sequence_1, dtype=float)
            best_score = float('inf')
        except Exception:
            # very last resort: uniform distribution (always yields 2.0)
            best_seq = np.full(n, 1.0 / n, dtype=float)
            best_score = 2.0

    print(f"\nFinal best score: {best_score:.6f} (length {len(best_seq)})")
    print(f"Best exponent p ≈ {best_p:.6f}")

    # return a plain Python list as required by the driver
    return best_seq.tolist()
