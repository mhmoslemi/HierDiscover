"""
Propose a low‑score step‑function sequence.

The algorithm consists of three phases:

1. **Broad random seeding** – generate thousands of candidates from
   several families (geometric, power‑law, Dirichlet, two‑level)
   and from each family also try a random permutation.
   Every candidate is projected onto the ℓ₁‑simplex
   (sum = √(2 n)) and its score is evaluated with a fast FFT‑based
   convolution.  The best seed is kept.

2. **Active‑set projected sub‑gradient descent** – starting from the
   best seed we repeatedly:
      * compute the full self‑convolution,
      * collect the “active” indices whose value is within a tiny
        relative tolerance of the current maximum,
      * build the sub‑gradient of the max‑functional,
      * take a Nesterov‑type momentum step,
      * project back onto the simplex.
   A small amount of random jitter is added from time to time.
   The loop runs until the time budget allocated to this phase
   expires.

3. **Linear‑programming refinement (optional)** – if SciPy is
   available we solve the LP
        max Σ g_i  s.t. (f * g)_k ≤ M   (∀ k)
   where **f** is the current vector and **M** the present maximal
   convolution value.  The solution is rescaled to the target ℓ₁‑norm
   and mixed back into **f**.  A few LP updates are performed.

All operations respect the security checks performed by
``evaluate_sequence`` (non‑negative, ≤ 1000).  The routine stops
when the allotted time (≈ 1000 s) expires and returns the best
sequence discovered.
"""

import time
import numpy as np

# --------------------------------------------------------------------- #
# Simplex projection (Duchi et al. 2008)
def projection_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Project `v` onto the simplex {x ≥ 0, Σx = s}."""
    if s < 0:
        raise ValueError("Simplex sum must be non‑negative")
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho_arr = np.where(u - cssv / (np.arange(n) + 1.0) > 0)[0]
    if rho_arr.size == 0:
        return np.zeros_like(v)
    rho = rho_arr[-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# --------------------------------------------------------------------- #
def fast_score(seq: np.ndarray) -> float:
    """Score = 2·n·max(conv(seq,seq)) / (sum(seq)²) using FFT."""
    n = seq.size
    fft_len = 1 << ((2 * n - 1).bit_length())
    F = np.fft.rfft(seq, n=fft_len)
    conv = np.fft.irfft(F * F, n=fft_len)[: 2 * n - 1]
    max_conv = conv.max()
    s = seq.sum()
    return float(2.0 * n * max_conv / (s * s))


# --------------------------------------------------------------------- #
def _full_convolution(seq: np.ndarray) -> np.ndarray:
    """Return the full self‑convolution of `seq` (length 2·n‑1)."""
    n = seq.size
    fft_len = 1 << ((2 * n - 1).bit_length())
    F = np.fft.rfft(seq, n=fft_len)
    conv = np.fft.irfft(F * F, n=fft_len)[: 2 * n - 1]
    return conv


# --------------------------------------------------------------------- #
def _active_subgradient(seq: np.ndarray, conv: np.ndarray,
                        max_val: float, rel_tol: float) -> np.ndarray:
    """Sub‑gradient of the max‑functional (active‑set version)."""
    n = seq.size
    tol_abs = max_val * rel_tol
    active_idx = np.where(conv >= max_val - tol_abs)[0]

    grad = np.zeros_like(seq)
    idx_arr = np.arange(n, dtype=np.int32)

    for k in active_idx:
        j = k - idx_arr
        mask = (j >= 0) & (j < n)
        grad[mask] += 2.0 * seq[j[mask]]
    return grad


# --------------------------------------------------------------------- #
def _build_toeplitz_matrix(f: np.ndarray):
    """Sparse Toeplitz matrix A (shape (2n‑1, n)) such that A·g = f * g."""
    n = f.shape[0]
    rows = np.empty(n * n, dtype=np.int32)
    cols = np.empty(n * n, dtype=np.int32)
    data = np.empty(n * n, dtype=np.float64)

    ptr = 0
    for i in range(n):
        rows[ptr:ptr + n] = np.arange(i, i + n)          # rows i … i+n‑1
        cols[ptr:ptr + n] = i
        data[ptr:ptr + n] = f
        ptr += n

    from scipy.sparse import csr_matrix
    return csr_matrix((data, (rows, cols)), shape=(2 * n - 1, n))


# --------------------------------------------------------------------- #
def propose_candidate(seed: int = 42, budget_s: float = 1000, **kwargs):
    """
    Search for a non‑negative step‑function (list of floats) that
    minimises ``evaluate_sequence``.  The function respects the
    1000 s time budget and returns the best sequence found.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.2   # safety margin

    # --------------------------------------------------------------- #
    # Problem size – 4000 entries keep the FFT length at 8192.
    n = 4000
    target_sum = np.sqrt(2.0 * n)   # we keep the ℓ₁‑norm fixed

    # --------------------------------------------------------------- #
    # 1️⃣  Broad random seeding
    best_seq = None
    best_score = float("inf")

    # time allocated to the seeding phase (~25 % of the budget)
    seed_deadline = start_time + 0.25 * budget_s

    # useful helpers -------------------------------------------------
    def evaluate_candidate(a):
        """Project onto the simplex, compute the fast score and update the global best."""
        nonlocal best_seq, best_score
        a = projection_simplex(a, target_sum)
        sc = fast_score(a)
        if sc < best_score:
            best_score, best_seq = sc, a.copy()
            print(f"[{time.time() - start_time:6.1f}s] new seed → score {sc:.6f}")
        return a

    while time.time() < seed_deadline:
        # pick a random family -------------------------------------------------
        family = np.random.choice(["geometric", "power", "dirichlet", "two_level"])
        if family == "geometric":
            # geometric decay a_i = r**i  (0 < r < 1)
            lam = 10 ** np.random.uniform(-5, -1)   # λ = 1‑r
            r = 1.0 - lam
            a = np.power(r, np.arange(n, dtype=np.float64))
        elif family == "power":
            # power‑law a_i ∝ (i+1)⁻ᵖ
            p = np.random.uniform(0.3, 1.2)
            a = 1.0 / np.power(np.arange(1, n + 1, dtype=np.float64), p)
        elif family == "dirichlet":
            # Dirichlet (Gamma) draws with concentration α
            alpha = np.random.uniform(0.1, 5.0)
            a = np.random.gamma(alpha, 1.0, size=n)
        else:   # "two_level"
            # k entries with value h, the rest with value l  (h ≥ l ≥ 0)
            frac = np.random.uniform(0.01, 0.5)          # fraction of 'high' entries
            k = max(1, int(frac * n))
            max_h = target_sum / k
            if max_h <= 0:
                continue
            h = np.random.uniform(0.0, min(1000.0, max_h))
            l = (target_sum - k * h) / (n - k)
            if l < 0.0:
                continue
            a = np.full(n, l, dtype=np.float64)
            high_idx = np.random.choice(n, k, replace=False)
            a[high_idx] = h

        # optionally shuffle the values (adds a lot of diversity)
        if np.random.rand() < 0.7:
            np.random.shuffle(a)

        # project onto the simplex and evaluate
        a = evaluate_candidate(a)

    # --------------------------------------------------------------- #
    # If the seeding phase produced nothing (should not happen), fall back to uniform.
    if best_seq is None:
        best_seq = projection_simplex(np.full(n, target_sum / n), target_sum)
        best_score = fast_score(best_seq)

    # --------------------------------------------------------------- #
    # 2️⃣  Active‑set projected sub‑gradient descent
    descent_deadline = start_time + 0.85 * budget_s   # remaining time for refinement
    curr = best_seq.copy()
    best = best_seq.copy()
    best_score = fast_score(best)
    velocity = np.zeros_like(curr)

    # hyper‑parameters
    step = 0.6 * target_sum / n          # absolute step size (initial)
    min_step = 1e-9
    momentum = 0.9
    rel_tol = 5e-5                       # tolerance for the active set
    it = 0

    while time.time() < descent_deadline:
        # ----- convolution & score ------------------------------------
        conv = _full_convolution(curr)
        max_val = conv.max()
        score = float(2.0 * n * max_val / (curr.sum() ** 2))

        # keep the best ever seen
        if score < best_score - 1e-12:
            best_score, best = score, curr.copy()
            print(f"[{time.time() - start_time:6.1f}s] iter {it:7d} → new best {score:.6f}")
            step = max(step * 0.9, min_step)   # tighten the step when we improve

        # ----- active set & sub‑gradient -------------------------------
        grad = _active_subgradient(curr, conv, max_val, rel_tol)
        gnorm = np.linalg.norm(grad)
        if gnorm == 0.0:
            break

        # ----- Nesterov‑type momentum step -----------------------------
        velocity = momentum * velocity - step * (grad / gnorm)
        tentative = curr + velocity
        tentative = np.clip(tentative, 0.0, 1000.0)
        curr = projection_simplex(tentative, target_sum)

        # ----- occasional jitter (helps escape plateaus) ---------------
        if np.random.rand() < 0.02:
            jitter = (np.random.randn(n) *
                      step * target_sum / (30.0 * n))
            curr = projection_simplex(
                np.clip(curr + jitter, 0.0, 1000.0), target_sum)

        it += 1

        # slowly decay step size
        if it % 500 == 0:
            step = max(step * 0.85, min_step)

    # --------------------------------------------------------------- #
    # 3️⃣  Linear‑programming refinement (if SciPy is present)
    try:
        from scipy.optimize import linprog
        from scipy.sparse import csr_matrix
    except Exception:
        linprog = None

    if linprog is not None and time.time() < descent_deadline - 5.0:
        f = best.copy()
        while time.time() < descent_deadline - 5.0:
            # current maximal convolution value
            conv_f = _full_convolution(f)
            M = conv_f.max()

            # Build Toeplitz matrix for constraints (f * g)_k ≤ M
            A = _build_toeplitz_matrix(f)          # shape (2n‑1, n)
            b = np.full(2 * n - 1, M)

            # maximise Σ g_i  →  minimise −Σ g_i
            res = linprog(
                c=-np.ones(n),
                A_ub=A,
                b_ub=b,
                bounds=[(0, None)] * n,
                method='highs',
                options={'presolve': True, 'time_limit': 30}
            )
            if not res.success or res.x is None:
                break

            g = res.x
            sum_g = g.sum()
            if sum_g < 1e-12:
                break

            # Rescale g to the target ℓ₁‑norm and mix
            g = g * (target_sum / sum_g)
            mix = 0.15
            f_new = (1.0 - mix) * f + mix * g
            f_new = projection_simplex(f_new, target_sum)

            # compute new score
            conv_new = _full_convolution(f_new)
            M_new = conv_new.max()
            score_new = float(2.0 * n * M_new / (f_new.sum() ** 2))

            if score_new < best_score - 1e-9:
                best_score, best = score_new, f_new.copy()
                print(f"[{time.time() - start_time:6.1f}s] LP‑update → best {score_new:.6f}")

            # accept the LP step when it does not increase the bound
            if M_new <= M * 0.999:
                f = f_new
            else:
                break

    # --------------------------------------------------------------- #
    # Final sanity check with the official evaluator.
    final_score = evaluate_sequence(best.tolist())
    print(f"Search finished. Final score reported by evaluate_sequence: {final_score:.6f}")

    return best.tolist()
