"""
propose_candidate
-----------------
Hybrid projected‑gradient / simulated‑annealing search for a non‑negative
step‑height vector that minimises

    2 * n * max(a * a) / (sum(a) ** 2)

(the objective used by evaluate_sequence).  The algorithm works in three
phases:

1.  Initialise from the best known construction (``height_sequence_1``)
    or from the classic central‑binomial coefficients.
2.  Perform momentum‑based projected gradient descent on the simplex.
    The gradient of a smooth approximation of the max‑convolution
    (soft‑max with a temperature ``alpha``) is computed via a second
    convolution, which is cheap thanks to NumPy’s FFT‑based
    implementation.
3.  If progress stalls, inject a small random perturbation and continue.
    A simple Metropolis acceptance rule (linear cooling) lets the
    optimiser occasionally accept uphill moves, helping it escape local
    minima.

The best sequence seen during the allotted wall‑time is returned.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
def binomial_sequence(length: int) -> np.ndarray:
    """
    Central binomial coefficients  C(2k, k) / 4**k  for k = 0 .. length‑1.
    """
    seq = np.empty(length, dtype=np.float64)
    seq[0] = 1.0
    for k in range(1, length):
        seq[k] = seq[k - 1] * ((2 * k - 1) * (2 * k)) / (4.0 * k * k)
    return seq


# ----------------------------------------------------------------------
def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the probability simplex
    (non‑negative coordinates that sum to 1).

    Implements the classic O(n log n) algorithm by sorting.
    """
    if np.all(v >= 0) and np.isclose(v.sum(), 1.0):
        return v  # already on the simplex

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(len(v)) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # All entries would be projected to zero – fall back to uniform.
        return np.full_like(v, 1.0 / len(v))
    rho = ind[cond][-1] - 1
    theta = cssv[cond][-1] / (rho + 1)
    w = np.maximum(v - theta, 0)
    return w


# ----------------------------------------------------------------------
