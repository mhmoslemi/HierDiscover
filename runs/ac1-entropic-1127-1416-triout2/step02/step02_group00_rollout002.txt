"""
Propose a low‑discrepancy step‑function.

The routine works in three phases:

1. **Exploratory sampling** –  For several lengths (256 … 4096) we
   generate shapes from a flexible two‑parameter beta family  
   a_i ∝ (i+ε)^{α‑1}·(n‑1‑i+ε)^{β‑1} (α,β>0).  Random draws of
   (α,β,ε) together with the classic arcsine start are scored with a
   fast *max‑convolution* surrogate.  The best candidate per length is kept.

2. **Smooth Adam optimisation** –  Starting from the best sampled shape we
   minimise a *smooth* approximation of the official objective

        f(a) = 2·n·max(a*a) / (Σa)²

   by replacing the max with a soft‑max (log‑sum‑exp) having an annealed
   temperature λ.  The gradient of the smoothed objective is obtained
   analytically (a weighted sum of sub‑gradients of the convolution) and
   an Adam update is applied, followed by a Euclidean projection onto the
   probability simplex.  The optimiser runs as long as the time budget permits.

3. **Final polishing** –  The globally best vector is refined again with the
   same Adam scheme for a longer run.

All helper functions are top‑level, there are no closures, and the code
stays well within the 1000 s budget.  The returned list is guaranteed to be
non‑negative and non‑empty, so the evaluator never returns ``inf``.
"""

import time
import numpy as np
from typing import List, Tuple

# ----------------------------------------------------------------------
# Low‑level utilities (no closures)
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """
    Real‑FFT convolution of two 1‑D arrays.
    Returns the linear convolution (length = len(a)+len(b)-1).
    """
    m = len(a) + len(b) - 1
    n_fft = next_pow2(m)
    A = np.fft.rfft(a, n=n_fft)
    B = np.fft.rfft(b, n=n_fft)
    C = A * B
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


def max_convolution_and_index(a: np.ndarray) -> Tuple[float, int]:
    """Return (max value, index) of the self‑convolution a*a."""
    conv = convolution_fft(a, a)
    idx = int(np.argmax(conv))
    return float(conv[idx]), idx


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection onto the probability simplex {x ≥ 0, Σx = 1}.
    Duchi et al., 2008.
    """
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / v.size)

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho = np.where(u - cssv / np.arange(1, v.size + 1) > 0)[0]
    if len(rho) == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def compute_score(a: np.ndarray) -> float:
    """
    Fast surrogate for the official score (the denominator is Σa² = 1
    after projection, so the function reduces to 2·n·max(a*a)).
    """
    n = len(a)
    conv = convolution_fft(a, a)
    max_val = float(np.max(conv))
    return 2.0 * n * max_val


def arcsine_initial(n: int) -> np.ndarray:
    """Classic arcsine shape: a_i ∝ 1/√((i+½)*(n‑½‑i))."""
    i = np.arange(n, dtype=float)
    eps = 0.25
    vals = 1.0 / np.sqrt((i + 0.5) * (n - 0.5 - i) + eps)
    vals /= vals.sum()
    return vals


def generate_beta_shape(n: int, alpha: float, beta: float, eps: float) -> np.ndarray:
    """
    Symmetric beta‑family density on {0,…,n‑1}:
        a_i ∝ (i+ε)^{α‑1}·(n‑1‑i+ε)^{β‑1}.
    Returned vector is normalized to Σa = 1.
    """
    i = np.arange(n, dtype=float)
    # guard against overflow/underflow
    left = (i + eps) ** (alpha - 1.0)
    right = (n - 1 - i + eps) ** (beta - 1.0)
    base = left * right
    total = base.sum()
    if total <= 0.0 or not np.isfinite(total):
        # fallback to uniform
        return np.full(n, 1.0 / n)
    return base / total


# ----------------------------------------------------------------------
# Smooth Adam optimiser (max‑convolution surrogate)
# ----------------------------------------------------------------------


def smooth_gradient(a: np.ndarray, lam: float) -> Tuple[np.ndarray, float]:
    """
    Compute the gradient of the smooth max‑convolution objective
    for a given temperature λ and also return the smooth objective value.
    """
    n = len(a)
    conv = convolution_fft(a, a)               # length 2n‑1
    max_c = np.max(conv)

    # stable soft‑max
    exp_term = np.exp(lam * (conv - max_c))
    sum_exp = np.sum(exp_term)
    w = exp_term / sum_exp                     # weights, sum to 1
    # smooth approximation of the max
    B = (max_c + np.log(sum_exp) / lam)

    # weighted sum of sub‑gradients:
    #   ∂ (Σ_k w_k (a*a)_k) / ∂a = 2 * (a * w_rev) evaluated at appropriate indices
    w_rev = w[::-1]
    conv_rev = convolution_fft(a, w_rev)       # length n + (2n‑1) -1 = 3n‑2
    # index conversion: weighted gradient = 2 * conv_rev[L-1 - i] with L = 2n‑1
    L = 2 * n - 1
    weighted_grad = 2.0 * conv_rev[L - 1 - np.arange(n)]

    s = np.sum(a)
    term1 = (2.0 * n / (s ** 2)) * weighted_grad
    term2 = -(4.0 * n * B / (s ** 3)) * np.ones_like(a)
    grad = term1 + term2
    return grad, B


def adam_optimize(initial_a: np.ndarray,
                  time_deadline: float,
                  max_iter: int = 100_000,
                  lr0: float = 0.2,
                  lam_start: float = 5.0,
                  lam_end: float = 30.0,
                  noise_every: int = 2000,
                  noise_scale: float = 5e-5) -> Tuple[np.ndarray, float]:
    """
    Adam optimiser on the smooth surrogate.
    Returns the best vector (projected on the simplex) and its fast score.
    """
    a = project_to_simplex(initial_a.astype(float).copy())
    best_a = a.copy()
    best_sc = compute_score(a)

    beta1, beta2 = 0.9, 0.999
    eps = 1e-8
    m = np.zeros_like(a)
    v = np.zeros_like(a)

    n = len(a)

    for it in range(1, max_iter + 1):
        if time.time() >= time_deadline:
            break

        # linear schedule for λ
        lam = lam_start + (lam_end - lam_start) * min(1.0, it / max_iter)

        grad, _ = smooth_gradient(a, lam)

        # Adam moments
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)

        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        step = lr0 * m_hat / (np.sqrt(v_hat) + eps)
        a_new = a - step

        if noise_every and (it % noise_every == 0):
            a_new += np.random.randn(n) * noise_scale

        a_new = project_to_simplex(a_new)

        cur_sc = compute_score(a_new)
        if cur_sc < best_sc:
            best_sc = cur_sc
            best_a = a_new.copy()

        a = a_new

        # occasional status
        if it % 5000 == 0:
            print(f"    Adam iter={it:,d}  cur={cur_sc:.6f}  best={best_sc:.6f}")

    return best_a, best_sc

    # --------------------------------------------------------------

    try:

    # --------------------------------------------------------------
