"""
Propose a low‑autocorrelation step function.

The algorithm improves on the previous sub‑gradient method by
* using a smooth approximation of the max‑convolution (log‑sum‑exp)
  so that the gradient takes *all* convolution entries into account,
* employing an Adam‑style projected gradient descent on the probability
  simplex,
* gradually increasing the smoothing parameter β (continuation) to drive
  the true maximum down,
* trying several simple initial shapes (arcsine, uniform, random
  Dirichlet and a two‑parameter power family) for a few sequence
  lengths and keeping the best result.

All heavy work (convolutions, gradients) uses FFT‑based O(n log n)
routines, while the official `evaluate_sequence` function is called
only for a small number of promising candidates.  The routine respects
the 1000‑second budget and always returns a valid non‑negative list of
heights.
"""

import time
import numpy as np
from typing import List, Tuple

# ----------------------------------------------------------------------
# Low‑level utilities (no closures)
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Return the smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """
    Real convolution via FFT with zero‑padding to the next power of two.
    Returns an array of length len(a)+len(b)-1.
    """
    m = len(a) + len(b) - 1
    n_fft = next_pow2(m)
    A = np.fft.rfft(a, n=n_fft)
    B = np.fft.rfft(b, n=n_fft)
    C = A * B
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the probability simplex
    {x | x_i ≥ 0, Σ_i x_i = 1}.
    Implementation of Duchi et al. (2008).
    """
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / len(v))

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho = np.where(u - cssv / np.arange(1, len(v) + 1) > 0)[0]
    if len(rho) == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def compute_score(a: np.ndarray) -> float:
    """
    Scale‑invariant objective used by the official evaluator:
        2·n·max(a*a) / (sum(a)²)
    The input is assumed to be a probability vector (sum ≈ 1).
    """
    n = len(a)
    conv = convolution_fft(a, a)
    max_val = float(np.max(conv))
    s = float(np.sum(a))
    if s < 1e-12:
        return float("inf")
    return 2.0 * n * max_val / (s * s)


def arcsine_initial(n: int) -> np.ndarray:
    """
    Classic arcsine‑shaped start:
        a_i ∝ 1/√((i+½)*(n‑½‑i)).
    Normalised to unit L¹ norm.
    """
    i = np.arange(n, dtype=float)
    eps = 0.25
    vals = 1.0 / np.sqrt((i + 0.5) * (n - 0.5 - i) + eps)
    return vals / np.sum(vals)


def generate_shape(n: int, p: float, eps: float) -> np.ndarray:
    """
    Symmetric family a_i ∝ ((i+eps)*(n‑1‑i+eps))⁻ᵖ,
    normalised to unit L¹ norm.
    """
    i = np.arange(n, dtype=float)
    base = ((i + eps) * (n - 1 - i + eps)) ** (-p)
    total = base.sum()
    if total == 0.0:
        return np.full(n, 1.0 / n)
    return base / total


# ----------------------------------------------------------------------
# Optimisation core – smooth max + Adam on the simplex
# ----------------------------------------------------------------------


def smooth_descent(
    initial_a: np.ndarray,
    time_deadline: float,
    max_iter_per_beta: int = 1500,
) -> Tuple[np.ndarray, float]:
    """
    Projected Adam optimisation of the smoothed objective

        L_β(a) = (1/β) * log Σ_k exp(β·c_k),

    where c = a∗a.  The algorithm runs a continuation on β
    (0.5 → … → 80) and returns the best vector seen (with the
    true max‑convolution score).
    """
    a = project_to_simplex(initial_a.astype(float).copy())
    n = len(a)

    # best solution seen so far (according to the true objective)
    c = convolution_fft(a, a)
    best_sc = compute_score(a)
    best_a = a.copy()

    # Adam state
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    t = 0

    # schedule of smoothing parameters
    beta_vals = [0.5, 1.0, 2.0, 5.0, 10.0, 20.0, 40.0, 80.0]

    for beta in beta_vals:
        # perform a fixed number of iterations for each β (or stop early)
        for _ in range(max_iter_per_beta):
            if time.time() >= time_deadline:
                break

            t += 1

            # 1) self‑convolution
            c = convolution_fft(a, a)               # length 2n‑1

            # 2) smooth weights w_k ∝ exp(β·c_k)
            #    subtract max for numerical stability
            shift = beta * c - np.max(beta * c)
            w = np.exp(shift)
            w /= np.sum(w)                           # now Σ w = 1

            # 3) gradient of the smoothed objective:
            #    ∇ L_β(a) = 2 * Σ_k w_k * a_{k‑i}
            #    which is 2·correlation(w, a).  We compute it via FFT.
            w_rev = w[::-1]                          # reverse w
            conv_rev = convolution_fft(w_rev, a)     # length 3n‑2
            grad = 2.0 * conv_rev[2 * n - 2 - np.arange(n)]

            # 4) Adam update + projection onto simplex
            m = beta1 * m + (1.0 - beta1) * grad
            v = beta2 * v + (1.0 - beta2) * (grad ** 2)
            m_hat = m / (1.0 - beta1 ** t)
            v_hat = v / (1.0 - beta2 ** t)
            step = 0.05 / (np.sqrt(v_hat) + eps)     # base learning rate = 0.05
            a = a - step * m_hat
            a = project_to_simplex(a)

            # occasional evaluation of the true objective
            if t % 200 == 0:
                c_now = convolution_fft(a, a)
                cur_sc = compute_score(a)  # uses max(c_now) internally
                if cur_sc < best_sc:
                    best_sc = cur_sc
                    best_a = a.copy()

        if time.time() >= time_deadline:
            break

        # small jitter after each β to help escape shallow plateaus
        a += np.random.default_rng().normal(scale=5e-5, size=n)
        a = project_to_simplex(a)

    return best_a, best_sc

# Main entry point required by the competition harness
# ----------------------------------------------------------------------


def propose_candidate(
    seed: int = 42,
    budget_s: int = 1000,
    **kwargs,
) -> List[float]:
    """
    Search for a non‑negative step function with a low
    ``evaluate_sequence`` value.  The routine respects the time budget,
    explores several lengths and initial shapes, refines them with the
    smooth‑max optimiser described above, and finally returns the best
    sequence found (as a plain list of floats).
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 2.0   # safety margin

    best_seq: List[float] = []
    best_val: float = float("inf")

    # --------------------------------------------------------------
    # 0.  Optional fallback from previous run
    # --------------------------------------------------------------
    try:
        fallback = [float(x) for x in height_sequence_1]   # type: ignore
        val = evaluate_sequence(fallback)
        print(f"Fallback (height_sequence_1) score = {val:.6f}")
        if val < best_val:
            best_val = val
            best_seq = fallback
    except NameError:
        pass

    # --------------------------------------------------------------
    # 1.  Search over a few lengths (larger n gives more flexibility)
    # --------------------------------------------------------------
    lengths = [512, 1024, 2048, 4096]

    for n in lengths:
        if time.time() >= deadline - 5.0:
            break
        print(f"\n=== Exploring length n = {n} ===")

        # ---- 1a. Build a small pool of initial candidates ----------------
        candidates: List[np.ndarray] = [
            arcsine_initial(n),
            np.full(n, 1.0 / n),                     # uniform
        ]

        # random Dirichlet samples
        for _ in range(3):
            candidates.append(rng.dirichlet(np.ones(n)))

        # random members of the two‑parameter family
        for _ in range(3):
            p = rng.uniform(0.15, 1.0)
            eps = rng.uniform(1e-4, max(0.5, 0.02 * n))
            candidates.append(generate_shape(n, p, eps))

        # --------------------------------------------------------------
        # 1b.  Refine each candidate (time‑aware)
        # --------------------------------------------------------------
        remaining_candidates = len(candidates)
        for init_a in candidates:
            if time.time() >= deadline - 5.0:
                break
            # allocate a fair share of the remaining budget for this candidate
            time_left = deadline - time.time()
            cand_deadline = time.time() + time_left / max(remaining_candidates, 1)
            remaining_candidates -= 1

            # run the smoother optimiser
            refined_a, _ = smooth_descent(
                initial_a=init_a,
                time_deadline=cand_deadline,
                max_iter_per_beta=1500,
            )

            # evaluate with the official (slow) evaluator once
            val = evaluate_sequence(refined_a.tolist())
            print(f"  length {n} candidate score = {val:.6f}")

            if val < best_val:
                best_val = val
                best_seq = refined_a.tolist()
                print(f"  *** New global best: {best_val:.6f} (n={n}) ***")

    # --------------------------------------------------------------
    # 2.  Final polishing of the current global best (if time permits)
    # --------------------------------------------------------------
    if best_seq and time.time() < deadline - 5.0:
        a = np.array(best_seq, dtype=float)
        print("\nFinal polishing of the best sequence …")
        a_refined, _ = smooth_descent(
            initial_a=a,
            time_deadline=deadline,
            max_iter_per_beta=2500,
        )
        final_val = evaluate_sequence(a_refined.tolist())
        print(f"Final polished score = {final_val:.6f}")
        if final_val < best_val:
            best_val = final_val
            best_seq = a_refined.tolist()
            print("*** Updated global best after final polish ***")

    # --------------------------------------------------------------
    # 3.  Fallback guard – always return something valid
    # --------------------------------------------------------------
    if not best_seq:
        best_seq = [1.0] * 256
        best_val = evaluate_sequence(best_seq)

    print(f"\nSearch finished. Best reported score = {best_val:.6f}")
    return best_seq
