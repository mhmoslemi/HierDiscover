def simplex_projection(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of a vector onto the probability simplex.
    Implements the algorithm of Duchi et al. (2008).
    """
    v = np.asarray(v, dtype=float)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1] - 1
    theta = cssv[cond][-1] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def compute_objective(a: np.ndarray) -> float:
    """Return the exact value of the evaluation function for a vector
    `a` that is already normalised to sum = 1.
    This is `2·n·max(conv)`, where `conv = a*a` (linear convolution).

    conv = np.fft.irfft(fft * fft, n=2 * n - 1)
    return 2.0 * n * conv.max()


def beta_sequence(alpha: float, beta: float, n: int) -> np.ndarray:
    """Discretised Beta(α,β) density on n mid‑points, normalised to sum = 1."""
    x = (np.arange(n) + 0.5) / n

    w = np.exp(logw)
    total = w.sum()
    if total <= 0 or not np.isfinite(total):
        w = np.ones_like(w)
        total = w.sum()
    return w / total

    # a small number of refined iterations (sub‑gradient method from the baseline)
    for it in range(3000):
        if not time_left():
            break
        a_fft = np.fft.rfft(a, n=2 * n - 1)
        conv = np.fft.irfft(a_fft * a_fft, n=2 * n - 1)
        max_c = conv.max()
        cur_score = 2.0 * n * max_c
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[refine] it={it} new best {best_score:.6f}")

        # active set – entries within a relative tolerance of the current max
        eps = max_c * 1e-5
        active = np.where(conv >= max_c - eps)[0]
        if active.size == 0:
            active = np.array([conv.argmax()], dtype=int)

        # sub‑gradient of the true max‑convolution (average over active shifts)
        g = np.zeros_like(a)
        for k in active:
            lo = max(0, k - n + 1)
            hi = min(n - 1, k)
            idx = np.arange(lo, hi + 1, dtype=int)
            partner = k - idx
            g[idx] += 2.0 * a[partner]
        g /= float(active.size)

        # diminishing step‑size and projection onto the simplex
        step = 0.7 / np.sqrt(it + 1.0)
        a = simplex_projection(a - step * g)

    # --------------------------------------------------------------
    # 7. Return the best sequence (scaled – scaling does not affect the objective)
    # --------------------------------------------------------------
    # Scale to the conventional normalisation sum = sqrt(2·n) used in the
    # reference code; this keeps all entries comfortably below the 1000‑cap.
    scale = np.sqrt(2.0 * n) / best_a.sum()
    best_scaled = (best_a * scale).tolist()

    print(f"Search finished – best objective {best_score:.6f}")
    return [float(v) for v in best_scaled]
