def grad_softmax(a, alpha):
    # compute conv = a * a (full)
    F_a = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(F_a * F_a, n=fft_len)[:2*n-1]
    max_val = conv.max()
    # shift for numerical stability
    exp_arg = alpha * (conv - max_val)
    # to avoid overflow, we can clip
    exp_arg = np.clip(exp_arg, -700, 700)  # limit exponent
    w = np.exp(exp_arg)
    w = w / w.sum()
    # compute cross-correlation w ⋆ a (valid) using FFT
    F_w = np.fft.rfft(w, n=fft_len)
    corr_full = np.fft.irfft(F_w * np.conj(F_a), n=fft_len)[: (2*n-1) + n -1]  # length 3n-2 with zero padding.
    # extract valid part: indices n-1 to n-1 + n -1 => [n-1:2n-1)
    start = n - 1
    grad = 2.0 * corr_full[start : start + n]

    np.random.seed(seed)
    start = time.time()

    # Choose n (if global height_sequence_1 available, use its length)

"""Propose a step‑function (non‑negative vector) with a low maximum auto‑convolution.

The algorithm works in three phases:

1. **Seeding** – a handful of simple families (uniform, geometric,
   power‑law) and random Dirichlet vectors are projected onto the
   ℓ₁‑simplex {x≥0, Σx = √(2 n)}.

2. **Smoothed‑max Adam optimisation** – we minimise the soft‑maximum of
   the auto‑convolution  a∗a  using a differentiable surrogate
   `log‑sum‑exp`.  The gradient is obtained by a single FFT‑based
   correlation, and the parameters are updated with Adam and projected
   back onto the simplex.  The temperature α of the soft‑max is raised
   geometrically from 0.5 to ≈200, turning the surrogate into the true
   maximum towards the end of the run.

3. **Linear‑programming refinement** – for the best vector found,
   a small number of LP updates (as in the reference method) are
   performed to exploit the active‑set structure of the constraints.

All operations respect the security checks of ``evaluate_sequence``.
The routine stops before the allocated 1000 s budget and returns the
best sequence discovered.
"""

import time
import numpy as np

# --------------------------------------------------------------------------- #
# Simplex projection (Duchi et al. 2008)
def projection_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Project the vector `v` onto the simplex {x ≥ 0, Σx = s}."""
    if s < 0:
        raise ValueError("Simplex sum must be non‑negative")
    n = v.shape[0]
    # sort descending
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho_arr = np.where(u - cssv / (np.arange(n) + 1.0) > 0)[0]
    if rho_arr.size == 0:
        return np.zeros_like(v)
    rho = rho_arr[-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# --------------------------------------------------------------------------- #
def _max_convolution(a: np.ndarray, fft_len: int) -> float:
    """Return max_{k} (a * a)[k] using an FFT of length `fft_len`."""
    F = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(F * F, n=fft_len)[: 2 * a.size - 1]
    return float(conv.max())


# --------------------------------------------------------------------------- #
def propose_candidate(budget_s: float = 1000, seed: int = 0):
    """
    Search for a non‑negative sequence that minimises
    `evaluate_sequence`.  The function respects the 1000 s budget and
    returns the best list it found.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.2   # safety margin

    # --------------------------------------------------------------- #
    # Problem size – use the length of the previously known solution if
    # available, otherwise fall back to 2500.
    try:
        global height_sequence_1
        n = len(height_sequence_1) if isinstance(height_sequence_1, (list, tuple, np.ndarray)) else 2500
    except Exception:
        n = 2500

    target_sum = np.sqrt(2.0 * n)           # enforce Σa_i = √(2 n)

    # FFT length – must accommodate the linear correlation of lengths n and 2n‑1
    fft_len = 1 << ((3 * n - 2).bit_length())

    # --------------------------------------------------------------- #
    # 1. Build a small pool of seeds.
    seeds = []

    # uniform
    seeds.append(np.full(n, target_sum / n))

    # geometric families a_i = C·r^i
    for lam in np.logspace(-5, -1, 20):
        r = 1.0 - lam
        r_n = r ** n
        denom = 1.0 - r_n
        if denom <= 0.0:
            continue
        C = target_sum * lam / denom
        a = C * (r ** np.arange(n))
        seeds.append(projection_simplex(a, target_sum))

    # power‑law families a_i ∝ (i+1)^{-p}
    for p in np.linspace(0.3, 0.9, 12):
        base = 1.0 / np.power(np.arange(1, n + 1, dtype=np.float64), p)
        C = target_sum / base.sum()
        a = C * base
        seeds.append(projection_simplex(a, target_sum))

    for _ in range(10):

        F_a = np.fft.rfft(a, n=fft_len)

                print(f"[{time.time() - start_time:6.1f}s] LP‑iter {lp_iter} → best {new_score:.6f}")

            # accept the LP update only if it does not increase the bound
            if new_score <= M * 0.999:
                f = f_new
            else:
                break

    # --------------------------------------------------------------- #
    # 5. Final sanity check with the official evaluator.
    final_score = evaluate_sequence(best_seq_global.tolist())
    print(f"Search finished. Final score reported by evaluate_sequence: {final_score:.6f}")

    return best_seq_global.tolist()
