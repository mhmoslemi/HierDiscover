"""
Propose a low‑score step‑function using a smoothed (soft‑max) surrogate for the
maximum convolution value together with a projected Adam optimizer.
The surrogate weight vector w = softmax(β·conv) puts more emphasis on large
entries while still providing a differentiable direction.  Adam gives an
adaptive step‑size per coordinate and a simplex projection (Duchi‑et‑al.)
keeps the iterates feasible (non‑negative and sum‑to‑one).

The algorithm starts from the best known sequence (``height_sequence_1``) if
available, otherwise from a uniform distribution.  A schedule slowly increases
the soft‑max sharpness β so that the optimisation first smooths the landscape
and later concentrates on the true maximum.  Random jitter is injected when
progress stalls.  The routine stops after the allotted time or as soon as the
target bound 1.503 is reached.
"""

import time
import numpy as np

def simplex_projection(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the probability simplex:
        { w | w_i ≥ 0 , Σ w_i = 1 }.
    Implementation follows Duchi et al. (2008).
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1] - 1
    theta = cssv[cond][-1] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def compute_objective(a: np.ndarray) -> float:
    """
    Return the evaluation value 2·n·max(conv(a,a)) for a vector that is already
    normalised to Σ a = 1 (same expression used by ``evaluate_sequence``).
    """
    n = a.shape[0]
    N = 2 * n - 1
    a_fft = np.fft.rfft(a, n=N)
    conv = np.fft.irfft(a_fft * a_fft, n=N)
    return 2.0 * n * conv.max()

            print(f"Target reached at iteration {iteration}")
            break

        iteration += 1

    # --------------------------------------------------------------
    # 4. Return the best sequence found
    # --------------------------------------------------------------
    return [float(v) for v in best_a.tolist()]
