Implementation of cross_correlation using FFT:

def cross_correlation(a, w):
    n = a.shape[0]
    m = w.shape[0]  # should be 2*n-1
    L = 1 << ((n + m - 1 - 1).bit_length())
    # FFT both
    FA = np.fft.rfft(a, n=L)
    FW = np.fft.rfft(w, n=L)
    corr = np.fft.irfft(FA * np.conj(FW), n=L)
    # Only first n entries are needed
    return corr[:n]

"""
Propose a candidate step‑function by a *smoothed* projected gradient descent.

The classic shape a_i ∝ 1/√(i+1) makes the self‑convolution almost
constant, giving a score ≈ π/2 ≈ 1.5708.  By minimising a smooth
approximation of the L∞‑norm (log‑sum‑exp of the convolution) we can
move away from the exact 1/√i shape and lower the true maximum a little
further.  The optimisation runs on the probability simplex (non‑negative,
unit‑sum) and uses a Nesterov‑type momentum term, occasional random
redistributions and several random restarts.  All operations are O(n log n)
via FFTs, so many thousands of iterations fit easily into the 1000 s
budget.

The returned list is a plain Python list of non‑negative floats ready to
be fed to ``evaluate_sequence``.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities – all are top‑level, no closures
# ----------------------------------------------------------------------


def project_onto_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the probability simplex:
        {x | x >= 0, sum(x) = 1}.
    Returns a new array (the input is not modified).
    """
    n = v.shape[0]
    # sort v in descending order
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # fallback – uniform distribution
        return np.full_like(v, 1.0 / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def full_convolution(a: np.ndarray) -> np.ndarray:
    """
    Compute the (linear) convolution a * a using an FFT.
    Length of the result is 2*len(a) - 1.
    """
    n = a.shape[0]
    # length for FFT – next power of two for speed
    size = 2 * n
    fft_len = 1 << ((size - 1).bit_length())
    fft_a = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(fft_a * fft_a, n=fft_len)
    return conv[: 2 * n - 1]


def cross_correlation(a: np.ndarray, w: np.ndarray) -> np.ndarray:
    """
    Linear cross‑correlation of a (length n) with w (length 2n-1):
        (a ⋆ w)[k] = Σ_i a_i * w_{i+k},   k = 0 … n-1 .
    Implemented by FFT (real‑to‑real) – O(L log L) where L is the next
    power of two ≥ 3n‑2.
    """
    n = a.shape[0]
    m = w.shape[0]                     # = 2*n-1
    L = 1 << ((n + m - 2 - 1).bit_length())  # next power of two ≥ 3n‑2
    fa = np.fft.rfft(a, n=L)
    fw = np.fft.rfft(w, n=L)
    corr = np.fft.irfft(fa * np.conj(fw), n=L)
    return corr[:n]                    # keep only the needed n entries


def objective_score(a: np.ndarray) -> float:
    """
    Compute 2 * n * max(conv(a, a)) for a vector that already satisfies
    sum(a) == 1.  This matches the value returned by evaluate_sequence.
    """
    n = a.shape[0]
    conv = full_convolution(a)
    return 2.0 * n * np.max(conv)


# ----------------------------------------------------------------------
# Search routine
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs):
    """
    Search for a height sequence that minimises the evaluation function.
    The method employs several random restarts from simple parametric
    families, then refines the best candidate with a smoothed projected
    gradient descent (log‑sum‑exp surrogate for the max) using Nesterov
    momentum.  The routine respects the time budget and returns the best
    sequence found as a plain Python list.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 3.0   # leave a few seconds for exit

    # --------------------------------------------------------------
    # Problem size – a few thousand entries give a good trade‑off
    # --------------------------------------------------------------
    n = 5000

    # ------------------------------------------------------------------
    # 1️⃣  Build a small pool of starting points (power‑law, shifted,
    #     exponential, random Dirichlet)
    # ------------------------------------------------------------------
    starts = []

    # classic 1/√(i+1)
    raw = 1.0 / np.sqrt(np.arange(n, dtype=float) + 1.0)
    starts.append(raw / raw.sum())

    # shifted versions of the classic shape
    for shift in (5.0, 10.0, 20.0, 50.0):
        raw_s = 1.0 / np.sqrt(np.arange(n, dtype=float) + 1.0 + shift)
        starts.append(raw_s / raw_s.sum())

    # pure exponential families
    for r in (0.999, 0.9995, 0.9999):
        raw_e = r ** np.arange(n, dtype=float)
        starts.append(raw_e / raw_e.sum())

    # a handful of random Dirichlet draws with varied concentration
    for _ in range(12):
        alpha = np.random.uniform(0.1, 2.0)
        cand = np.random.dirichlet(np.full(n, alpha))
        starts.append(cand)

    # --------------------------------------------------------------
    # 2️⃣  Pick the best among the starters
    # --------------------------------------------------------------
    best_seq = None
    best_score = float('inf')
    for a in starts:
        sc = objective_score(a)
        if sc < best_score:
            best_score = sc
            best_seq = a.copy()
    print(f"Initial best score from the pool: {best_score:.6f}")

    # --------------------------------------------------------------
    # 3️⃣  Refine the best candidate with smoothed projected gradient
    # --------------------------------------------------------------
    a = best_seq.copy()
    velocity = np.zeros_like(a)          # Nesterov / momentum term
    it = 0
    # hyper‑parameters (tuned experimentally)
    step_initial = 0.8
    mu = 0.9                              # momentum coefficient
    beta = 0.5                            # start of the log‑sum‑exp smoothing
    beta_max = 5.0

    while time.time() < deadline:
        # ----- convolution and current score -----
        conv = full_convolution(a)
        max_val = conv.max()
        cur_score = 2.0 * n * max_val

        # ----- record improvement -----
        if cur_score < best_score:
            best_score = cur_score
            best_seq = a.copy()
            print(f"Iter {it:06d}: new best = {best_score:.6f}")
            if best_score < 1.503:          # early exit if we reach the target
                break

        # ----- update smoothing parameter (slowly increase) -----
        beta = min(beta_max, beta + 0.0005)   # linear schedule

        # ----- smoothed gradient of the max‑convolution -----
        # w_i = exp(beta * (conv_i - max_val))  (always ≤ 1)
        w = np.exp(beta * (conv - max_val))
        w /= w.sum()                          # normalise → probability vector
        grad = 2.0 * cross_correlation(a, w)  # ∇ (∑_k w_k * (a*a)_k)

        # ----- Nesterov‑style step on the simplex -----
        step = step_initial / np.sqrt(1 + it)
        velocity = mu * velocity - step * grad
        a_candidate = a + velocity
        a_candidate = project_onto_simplex(a_candidate)

        # ----- accept the candidate if it does not worsen the true objective -----
        conv_cand = full_convolution(a_candidate)
        cand_score = 2.0 * n * conv_cand.max()
        if cand_score <= cur_score:
            a = a_candidate
        else:
            # reject – discard the accumulated momentum for this step
            velocity[:] = 0.0

        # ----- occasional random redistribution to escape plateaux -----
        if it % 500 == 0 and it > 0:
            # pick three indices and move a tiny amount of mass between them
            i0, i1, i2 = np.random.choice(n, size=3, replace=False)
            delta = 0.01 * a[i0] * (np.random.rand() - 0.5)   # ≤ 1 % of a[i0]
            b = a.copy()
            b[i0] = max(b[i0] - delta, 0.0)
            b[i1] += max(delta, 0.0) * 0.6
            b[i2] += max(delta, 0.0) * 0.4
            b = project_onto_simplex(b)
            sc = objective_score(b)
            if sc < best_score:
                best_score = sc
                best_seq = b.copy()
                a = b
                print(f"  Random jitter improvement: score = {best_score:.6f}")

        it += 1

    # ------------------------------------------------------------------
    # Safety fallback – use a previously stored construction if needed
    # ------------------------------------------------------------------
    if best_seq is None or best_seq.size == 0:
        try:
            best_seq = np.array(height_sequence_1, dtype=float)
        except Exception:
            # ultimate fallback: uniform sequence (its score is exactly 2.0)
            best_seq = np.full(n, 1.0 / n, dtype=float)

    print(f"Final best score: {best_score:.6f} (length {len(best_seq)})")
    # return a plain Python list as required by the driver
    return best_seq.tolist()
