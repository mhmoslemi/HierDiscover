            p = rng.uniform(0.1, 2.0)

            seeds.append(generate_power_shape(n, p, eps) * sum_target)

            if time.time() >= deadline:
                break

            f = f * (sum_target / f.sum())
            # compute current obj
            cur_obj = np.max(convolution_fft(f, f))
            if cur_obj < best_obj:
                best_obj = cur_obj
                best_seq = f.copy()
            # set iteration loop
            max_iters = 80
            step = 0.3
            for it in range(1, max_iters+1):
                if time.time() >= deadline:
                    break

                max_conv = np.max(convolution_fft(f, f))

"""
Optimisation of a non‑negative step sequence.

The algorithm has two phases:

1️⃣  **Iterative LP refinement** –  Starting from a normalised vector
    (∑a_i = √(2 n)) we repeatedly solve the linear programme

        maximise   Σ g_i
        subject to  g_i ≥ 0 ,  (a * g)_k ≤ M  for all k=0…2n‑2

    where M = max(a * a).  The LP produces a direction g that pushes
    mass into positions that do **not** increase the current worst
    convolution value.  After scaling g to the same ℓ₁‑norm as a we
    replace a ← (1‑t) a + t g (t is a decaying step size).  With a
    fixed ℓ₁‑norm the official objective simplifies to the maximal entry
    of a*a, so we minimise `max(conv(a, a))` directly.

2️⃣  **Projected sub‑gradient polish** –  The LP step is followed by a
    short Euclidean‑projected sub‑gradient descent on the exact
    non‑smooth objective `max(conv(a, a))`.  The sub‑gradient is obtained
    from the active (max‑attaining) convolution entries and the iterate
    is projected onto the probability simplex and re‑scaled to the
    required ℓ₁‑norm.

Both phases respect the global time budget (default 1000 s).  The best
vector found is finally evaluated with the official `evaluate_sequence`
and returned.  The implementation requires only NumPy and SciPy.
"""

import time
import numpy as np
from scipy.optimize import linprog
from typing import List, Tuple

# ----------------------------------------------------------------------
# Low‑level utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Linear convolution via real FFT (full length)."""
    m = a.size + b.size - 1
    n_fft = next_pow2(m)
    A = np.fft.rfft(a, n=n_fft)
    B = np.fft.rfft(b, n=n_fft)
    C = A * B
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection onto the probability simplex {x≥0, Σx=1}."""
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / v.size)

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho_arr = u - cssv / np.arange(1, v.size + 1)
    rho = np.where(rho_arr > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Shape generators (used for initial seeds)
# ----------------------------------------------------------------------


def arcsine_initial(n: int) -> np.ndarray:
    """Classic arcsine shape a_i ∝ 1/√((i+½)*(n‑½‑i))."""
    i = np.arange(n, dtype=float)
    eps = 0.25
    vals = 1.0 / np.sqrt((i + 0.5) * (n - 0.5 - i) + eps)
    vals /= vals.sum()
    return vals


def generate_beta_shape(n: int, alpha: float, beta: float, eps: float) -> np.ndarray:
    """Beta‑family (i+ε)^{α‑1}·(n‑1‑i+ε)^{β‑1}, normalised to Σ=1."""
    i = np.arange(n, dtype=float)
    left = (i + eps) ** (alpha - 1.0)
    right = (n - 1 - i + eps) ** (beta - 1.0)
    base = left * right
    total = base.sum()
    if total <= 0.0 or not np.isfinite(total):
        return np.full(n, 1.0 / n)
    return base / total


def generate_power_shape(n: int, p: float, eps: float) -> np.ndarray:
    """Generalised arcsine: a_i ∝ ((i+½)*(n‑½‑i)+ε)^{‑p}."""
    i = np.arange(n, dtype=float)
    denom = ((i + 0.5) * (n - 0.5 - i) + eps) ** p
    vals = 1.0 / denom
    total = vals.sum()
    if total <= 0.0 or not np.isfinite(total):
        return np.full(n, 1.0 / n)
    return vals / total


# ----------------------------------------------------------------------
# Sub‑gradient descent on the exact objective
# ----------------------------------------------------------------------


def subgradient_descent_exact(a: np.ndarray,
                              target_sum: float,
                              deadline: float,
                              max_iter: int = 30_000,
                              step0: float = 0.2) -> np.ndarray:
    """
    Projected sub‑gradient descent on f(a)=max(conv(a,a)).
    The iterate is kept on the simplex and scaled to `target_sum`
    after each projection.
    """
    n = a.shape[0]
    L = 2 * n - 1
    # initialise on the simplex (already non‑negative)
    a = project_to_simplex(a) * target_sum

    for it in range(1, max_iter + 1):
        if time.time() >= deadline:
            break

        conv = convolution_fft(a, a)
        max_val = float(np.max(conv))

        # indices attaining the maximum (within tiny tolerance)
        tol = max_val * 1e-12 + 1e-14
        mask = (np.abs(conv - max_val) <= tol).astype(float)

        # sub‑gradient of max(conv) w.r.t. a:
        #  ∂/∂a_i max(conv) = 2 * Σ_{k∈K} a_{k-i}
        mask_rev = mask[::-1]
        conv_rev = convolution_fft(a, mask_rev)
        subgrad = 2.0 * conv_rev[L - 1 - np.arange(n)]

        eta = step0 / np.sqrt(it)
        a = a - eta * subgrad
        # project back onto the simplex and restore the required ℓ₁‑norm
        a = project_to_simplex(a) * target_sum

        if (it % 5000) == 0:
            cur = np.max(convolution_fft(a, a))
            print(f"    Sub‑grad it={it:,d} max‑conv={cur:.6f}")

    return a


# ----------------------------------------------------------------------
# Main entry point required by the competition harness
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42,
                     budget_s: int = 1000,
                     **kwargs) -> List[float]:
    """
    Search for a step sequence with a low ``evaluate_sequence`` value.
    The algorithm consists of an LP‑based refinement phase followed by
    a short sub‑gradient polish.  All vectors are kept on the simplex
    with ℓ₁‑norm √(2 n), which makes the official score equal to
    `max(conv(a, a))`.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 2.0          # safety margin

    best_seq: List[float] = []
    best_obj: float = float("inf")

    # ------------------------------------------------------------------
    # Choose a working length.  1024 offers a good speed/quality trade‑off.
    # ------------------------------------------------------------------
    n = 1024
    target_sum = np.sqrt(2.0 * n)                   # ℓ₁‑norm we keep fixed

    # ------------------------------------------------------------------
    # Build a small diverse set of seed vectors.
    # ------------------------------------------------------------------
    seeds: List[np.ndarray] = []

    # deterministic starters
    seeds.append(arcsine_initial(n) * target_sum)
    seeds.append(np.full(n, target_sum / n))

    # random beta families
    for _ in range(4):
        a = rng.uniform(0.2, 5.0)
        b = rng.uniform(0.2, 5.0)
        eps = rng.uniform(0.0, 1.0)
        seeds.append(generate_beta_shape(n, a, b, eps) * target_sum)

    # random power families
    for _ in range(4):
        p = rng.uniform(0.1, 2.5)
        eps = rng.uniform(0.0, 2.0)
        seeds.append(generate_power_shape(n, p, eps) * target_sum)

    # ------------------------------------------------------------------
    # 1️⃣  Iterative LP refinement for each seed
    # ------------------------------------------------------------------
    for s_idx, a in enumerate(seeds, start=1):
        if time.time() >= deadline:
            break

        # ensure exact ℓ₁‑norm
        a = a * (target_sum / a.sum())
        cur_obj = np.max(convolution_fft(a, a))
        if cur_obj < best_obj:
            best_obj = cur_obj
            best_seq = a.copy()
            print(f"Seed {s_idx}: initial max‑conv = {cur_obj:.6f}")

        # LP parameters
        max_lp_iters = 80                     # how many LP updates per seed
        step = 0.30                           # initial convex‑combination weight

        for it in range(1, max_lp_iters + 1):
            if time.time() >= deadline:
                break

            # current worst convolution value
            M = np.max(convolution_fft(a, a))

            # ----- build the constraint matrix A (shape (2n‑1) × n) -----
            L = 2 * n - 1
            A = np.empty((L, n), dtype=np.float64)
            for i in range(n):
                A[i:i + n, i] = a

            # ----- solve the LP: maximise Σ g_i  subject to A·g ≤ M, g≥0 -----
            c = -np.ones(n)                               # -Σg_i  (linprog minimises)
            b_ub = np.full(L, M)                          # upper bounds
            res = linprog(c,
                          A_ub=A,
                          b_ub=b_ub,
                          bounds=[(0, None)] * n,
                          method='highs',
                          options={'presolve': True})

            if not res.success:
                # LP failed – give up on this seed
                print(f"    LP failed at seed {s_idx}, it {it}")
                break

            g0 = res.x
            sum_g0 = g0.sum()
            if sum_g0 <= 0.0:
                break

            # Scale to the target ℓ₁‑norm
            g = g0 * (target_sum / sum_g0)

            # Convex combination update (step size decays slowly)
            step = max(0.02, step * 0.94)
            a = (1.0 - step) * a + step * g

            # Check improvement
            new_obj = np.max(convolution_fft(a, a))
            if new_obj < best_obj:
                best_obj = new_obj
                best_seq = a.copy()
                print(f"    Seed {s_idx}, LP it {it:02d}: max‑conv = {new_obj:.6f}")

    # ------------------------------------------------------------------
    # 2️⃣  Short sub‑gradient polish on the best candidate found so far
    # ------------------------------------------------------------------
    if best_seq and time.time() < deadline - 5.0:
        print("\nStarting final sub‑gradient polish …")
        a = best_seq.copy()
        # run a modest number of iterations (time‑limited)
        a = subgradient_descent_exact(a,
                                      target_sum=target_sum,
                                      deadline=deadline,
                                      max_iter=30_000,
                                      step0=0.15)
        # final objective
        final_obj = np.max(convolution_fft(a, a))
        if final_obj < best_obj:
            best_obj = final_obj
            best_seq = a.copy()
            print(f"Polish improved max‑conv to {final_obj:.6f}")

    # ------------------------------------------------------------------
    # 3️⃣  Return a safe list (fallback if everything failed)
    # ------------------------------------------------------------------
    if not best_seq:
        # Uniform fallback (guaranteed valid)
        n_fallback = 256
        uniform = np.full(n_fallback, 1.0 / n_fallback)
        best_seq = uniform

    # Convert to plain Python list and ensure non‑negative floats
    seq_list = [float(max(0.0, x)) for x in best_seq]

    # Final official evaluation (only for reporting)
    try:
        final_score = evaluate_sequence(seq_list)
        print(f"\nSearch finished. Final official score = {final_score:.6f}")
    except Exception as e:
        print(f"\nSearch finished. (official evaluation failed: {e})")

    return seq_list
