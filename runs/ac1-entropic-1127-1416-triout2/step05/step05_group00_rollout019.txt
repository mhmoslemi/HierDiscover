"""
Propose a low‑score step function.

The evaluation function is   f(a)= 2·n·max(a⊛a) / (∑a)² .
Because the objective is homogeneous we may normalise the vector to
∑a = 1 and simply minimise  2·n·max(a⊛a).

* 1️⃣  Random initialisations (uniform, Dirichlet, sparse‑subset) for a
   handful of lengths (512‑4096).  Keep the best one.
* 2️⃣  Smooth‑max (log‑sum‑exp) gradient descent.
   For a fixed β the smooth‑max ϕβ(a)=log∑exp(β·(a⊛a))
   has gradient   ∇ϕβ(a)=2·conv(w , a_rev)  (w is a soft‑max over the
   convolution entries).  After each step we project back onto the
   probability simplex.
   β is increased slowly (coarse → fine) so the optimisation first
   flattens the whole convolution and finally concentrates on the
   true maximum.
* 3️⃣  A long stochastic hill‑climb (swap / mass‑transfer moves) that
   accepts only improving moves.  This polishes the solution found by
   the gradient phase.
* 4️⃣  If time runs out early we fall back to a uniform vector.

All heavy work uses FFT‑convolutions ( O(n log n) ) and a simple
Euclidean projection onto the simplex, therefore the whole search fits
comfortably into the 1000 s budget.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# FFT‑based full convolution
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full convolution a ⊛ b using a real FFT (fast)."""
    m = len(a) + len(b) - 1
    n_fft = _next_pow2(m)
    A = np.fft.rfft(a, n=n_fft)
    B = np.fft.rfft(b, n=n_fft)
    C = A * B
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


# ----------------------------------------------------------------------
# Evaluation helpers (fast version – sum(a)=1)
# ----------------------------------------------------------------------


def compute_score(a: np.ndarray) -> float:
    """2·n·max(a⊛a)  (valid when ∑a = 1)."""
    n = a.shape[0]
    conv = convolution_fft(a, a)
    return 2.0 * n * float(np.max(conv))


# ----------------------------------------------------------------------
# Projection onto the probability simplex (∑=1 , a_i≥0)
# ----------------------------------------------------------------------


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    if np.all(v <= 0):
        # all non‑positive – return uniform
        return np.full_like(v, 1.0 / v.shape[0])

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, v.shape[0] + 1)
    cond = u - cssv / ind > 0
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


# ----------------------------------------------------------------------
# Smooth‑max gradient step
# ----------------------------------------------------------------------


def smooth_max_gradient_step(a: np.ndarray, beta: float, step: float) -> np.ndarray:
    """
    One projected gradient step on ϕβ(a)=log∑exp(β·(a⊛a)).
    Returns the updated vector (still on the simplex).
    """
    n = a.shape[0]

    # convolution a⊛a
    conv = convolution_fft(a, a)                     # length 2n‑1

    # soft‑max weights over the convolution entries
    # (subtract the max for numerical stability)
    exp_arg = beta * conv
    exp_arg -= np.max(exp_arg)
    w = np.exp(exp_arg)
    w /= np.sum(w)                                   # shape (2n‑1,)

    # gradient = 2 * conv(w , a_rev)  (see derivation in the docstring)
    a_rev = a[::-1]
    grad_full = convolution_fft(w, a_rev)           # length 3n‑2
    grad = 2.0 * grad_full[n - 1 : n - 1 + n]       # slice of length n

    # gradient descent + projection
    a_new = a - step * grad
    a_new = project_to_simplex(a_new)
    return a_new


# ----------------------------------------------------------------------
# Hill‑climb polishing (swap / mass‑transfer)
# ----------------------------------------------------------------------


def hill_climb(seq, deadline, max_steps=300_000):
    """
    Simple stochastic hill‑climb that only accepts improving moves.
    Uses the true objective compute_score().
    """
    a = np.array(seq, dtype=float)
    n = a.shape[0]

    best = a.copy()
    best_score = compute_score(best)

    step_scale = 0.1  # fraction of mass moved in a transfer

    while max_steps > 0 and time.time() < deadline:
        cand = best.copy()

        if np.random.rand() < 0.5:            # swap two entries
            i, j = np.random.randint(n, size=2)
            cand[i], cand[j] = cand[j], cand[i]
        else:                                 # transfer a fraction from i → j
            i, j = np.random.randint(n, size=2)
            if i != j and cand[i] > 0:
                delta = step_scale * cand[i]
                cand[i] -= delta
                cand[j] += delta

        cand = project_to_simplex(cand)
        cur_score = compute_score(cand)

        if cur_score < best_score:
            best, best_score = cand, cur_score
            step_scale = min(step_scale * 1.05, 0.5)
        else:
            step_scale *= 0.97

        max_steps -= 1

    return best.tolist(), best_score
