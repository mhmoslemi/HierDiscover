"""
Propose a step–function with a low evaluation score.

The evaluation score is 2·n·max(a⊛a)/(∑a)².
For ∑a = 1 this is simply 2·n·max(a⊛a).
The goal is to minimise the peak of the ordinary (non‑reversed) convolution
a⊛a.  The algorithm works in four stages

1.  Random Dirichlet initialisation for several lengths.
2.  Smooth “log‑sum‑exp’’ optimisation.
    *   The exact peak max(·) is approximated by
        fα(a)= (1/α)·log Σk exp(α·convk).  Larger α focuses on the true maximum.
    *   The gradient of fα is a weighted correlation of a with the
        soft‑max weights; it is obtained cheaply with one FFT‑based correlation.
    *   A projected gradient step (onto the probability simplex) is taken.
    *   α is increased slowly, while the step size decays ~1/√t.
3.  A few thousand sub‑gradient steps on the true max (as in the reference code)
    – useful for fine‑tuning.
4.  A final stochastic hill‑climb that swaps or moves mass between
    positions, always keeping the vector on the simplex.

All stages respect the 1000 s time budget, and intermediate progress is printed.
The returned list is a plain Python list of non‑negative floats suitable for
`evaluate_sequence`.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Basic numerical helpers
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full convolution a⊛b using a real FFT (fast)."""
    m = len(a) + len(b) - 1
    n_fft = next_pow2(m)
    A = np.fft.rfft(a, n=n_fft)
    B = np.fft.rfft(b, n=n_fft)
    C = A * B
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


def correlation_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """
    Linear correlation (full) of x with y.
    Returns array of length len(x)+len(y)-1 such that
    out[i+len(y)-1] = Σj x[j]·y[j+i] .
    """
    N = next_pow2(len(x) + len(y) - 1)
    X = np.fft.rfft(x, n=N)
    Y = np.fft.rfft(y, n=N)
    corr = np.fft.irfft(X * np.conj(Y), n=N)[: len(x) + len(y) - 1]
    return corr


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the probability simplex
    (non‑negative entries that sum to 1).
    """
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / v.shape[0])

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, v.shape[0] + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.where(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def compute_score(a: np.ndarray) -> float:
    """Fast replica of `evaluate_sequence` for ∑a = 1.
    Returns 2·n·max(a⊛a)."""
    n = a.shape[0]
    conv = convolution_fft(a, a)
    return 2.0 * n * float(np.max(conv))


# ----------------------------------------------------------------------
# Smooth log‑sum‑exp optimisation
# ----------------------------------------------------------------------


def smooth_optimize(
    a0: np.ndarray,
    deadline: float,
    max_iter: int = 300_000,
    step0: float = 0.12,
    alpha_start: float = 0.5,
    alpha_end: float = 80.0,
    alpha_mult: float = 1.02,
) -> tuple[np.ndarray, float]:
    """
    Minimise the peak of a⊛a using a smooth approximation.
    The objective is
        fα(a) = (1/α)·log Σk exp(α·convk)
    whose gradient is a weighted correlation of a with the soft‑max
    weights w_k ∝ exp(α·convk).  The routine performs projected
    gradient steps on the simplex.
    """
    a = a0.astype(float).copy()
    n = len(a)
    best_a = a.copy()
    best_score = compute_score(a)

    alpha = alpha_start
    it = 0

    while time.time() < deadline and it < max_iter:
        # ----- convolution and current score -----
        conv = convolution_fft(a, a)
        cur_score = 2.0 * n * float(np.max(conv))
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()

        # ----- slowly increase the smoothing parameter -----
        if it % 500 == 0 and alpha < alpha_end:
            alpha = min(alpha * alpha_mult, alpha_end)

        # ----- soft‑max weights on the convolution entries -----
        max_c = np.max(conv)                # for numerical stability
        exp_vals = np.exp(alpha * (conv - max_c))
        w = exp_vals / np.sum(exp_vals)     # shape (2n‑1,)

        # ----- gradient = 2 * correlation(a, w) (see derivation) -----
        # correlation of a (length n) with w (length 2n‑1)
        corr = correlation_fft(a, w)        # length 3n‑2
        # the needed lag range is i = 0 … n‑1 → index = (len(w)-1) + i
        start = len(w) - 1                  # = 2n‑2
        grad = 2.0 * corr[start : start + n]

        # ----- projected gradient step (step size ∝ 1/√t) -----
        step = step0 / np.sqrt(it + 1.0)
        a = project_to_simplex(a - step * grad)

        it += 1

    return best_a, best_score


# ----------------------------------------------------------------------
# Sub‑gradient optimisation of the exact max (as in the reference)
# ----------------------------------------------------------------------


def subgrad_optimize(
    a0: np.ndarray,
    deadline: float,
    max_iter: int = 200_000,
    step0: float = 0.07,
) -> tuple[np.ndarray, float]:
    """Projected sub‑gradient descent on the true objective 2·n·max(a⊛a)."""
    a = a0.astype(float).copy()
    best_a = a.copy()
    best_score = compute_score(a)
    n = len(a)

    for t in range(1, max_iter + 1):
        if time.time() >= deadline:
            break

        conv = convolution_fft(a, a)
        k = int(np.argmax(conv))

        idx = np.arange(n)
        j = k - idx
        mask = (j >= 0) & (j < n)
        g = np.zeros_like(a)
        g[mask] = 2.0 * a[j[mask]]

        step = step0 / np.sqrt(t)
        a = project_to_simplex(a - step * g)

        cur_score = compute_score(a)
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()

        if t % 1000 == 0:
            # small random perturbation to escape flat spots
            a += np.random.randn(n) * (step0 * 0.005 / np.sqrt(t))
            a = project_to_simplex(a)

    return best_a, best_score


# ----------------------------------------------------------------------
# Simple stochastic hill‑climber (swap / mass‑transfer moves)
# ----------------------------------------------------------------------


def hill_climb_permutation(
    seq,
    time_deadline,
    max_steps: int = 300_000,
) -> tuple[list[float], float]:
    """
    Stochastic hill‑climb that swaps entries or transfers a fraction of mass.
    The true (unsmoothed) objective is used for acceptance.
    """
    a = np.array(seq, dtype=float)
    n = a.shape[0]

    best_a = a.copy()
    best_score = compute_score(best_a)

    step_scale = 0.1  # initial fraction of mass moved in a transfer

    while max_steps > 0 and time.time() < time_deadline:
        cand = best_a.copy()

        if np.random.rand() < 0.5:                # simple swap
            i, j = np.random.randint(n, size=2)
            cand[i], cand[j] = cand[j], cand[i]
        else:                                      # move a fraction from i to j
            i, j = np.random.randint(n, size=2)
            if i != j:
                delta = step_scale * cand[i]
                cand[i] -= delta
                cand[j] += delta
                if cand[i] < 0:
                    cand[i] = 0.0

        cand = project_to_simplex(cand)
        cur_score = compute_score(cand)

        if cur_score < best_score:
            best_score = cur_score
            best_a = cand
            step_scale = min(step_scale * 1.05, 0.5)
        else:
            step_scale *= 0.97

        max_steps -= 1

    return best_a.tolist(), best_score

def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative sequence that minimises `evaluate_sequence`.
    The routine combines random Dirichlet starts, a smooth log‑sum‑exp
    gradient descent, a short sub‑gradient refinement and a final hill‑climb.
    """
    np.random.seed(seed)
    start_time = time.time()
    # reserve a small safety margin
    deadline = start_time + budget_s - 2.0

    best_seq = None
    best_val = float("inf")

    # --------------------------------------------------------------
    # (optional) try the known fallback sequence if it exists
    # --------------------------------------------------------------
    try:
        fallback = [float(x) for x in height_sequence_1]  # type: ignore
        fallback_val = evaluate_sequence(fallback)
        print(f"Fallback score = {fallback_val:.6f}")
        if fallback_val < best_val:
            best_val = fallback_val
            best_seq = fallback
    except Exception:
        pass

    # --------------------------------------------------------------
    # Phase 1 : several random Dirichlet initialisations
    # --------------------------------------------------------------
    lengths = [512, 1024, 2048, 4096, 8192]
    while time.time() < deadline - 180.0:
        n = int(np.random.choice(lengths))
        a0 = np.random.rand(n)
        a0 = a0 / np.sum(a0)                     # Dirichlet(1,…,1)
        cur = compute_score(a0)
        if cur < best_val:
            best_val = cur
            best_seq = a0.tolist()
            print(f"Random Dirichlet init improvement: score={best_val:.6f} (n={n})")

    # --------------------------------------------------------------
    # Phase 2 : smooth log‑sum‑exp optimisation on the current best
    # --------------------------------------------------------------
    if best_seq is not None and time.time() < deadline - 120.0:
        print("\nStarting smooth log‑sum‑exp optimisation")
        a = np.array(best_seq, dtype=float)
        a = project_to_simplex(a)

        # use a generous time slice for this phase
        phase_deadline = min(deadline - 60.0, time.time() + 40.0)
        a_opt, opt_score = smooth_optimize(
            a,
            time_deadline=phase_deadline,
            max_iter=400_000,
            step0=0.12,
            alpha_start=0.5,
            alpha_end=80.0,
            alpha_mult=1.02,
        )
        if opt_score < best_val:
            best_val = opt_score
            best_seq = a_opt.tolist()
            print(f"Smooth optimisation improved score to {best_val:.6f}")

    # --------------------------------------------------------------
    # Phase 3 : short sub‑gradient refinement (exact max)
    # --------------------------------------------------------------
    if best_seq is not None and time.time() < deadline - 60.0:
        print("\nSub‑gradient refinement (exact max)")
        a_init = np.array(best_seq, dtype=float)
        refined_a, refined_score = subgrad_optimize(
            a_init,
            time_deadline=deadline - 30.0,
            max_iter=250_000,
            step0=0.07,
        )
        if refined_score < best_val:
            best_val = refined_score
            best_seq = refined_a.tolist()
            print(f"Sub‑gradient improved best to {best_val:.6f}")

    # --------------------------------------------------------------
    # Phase 4 : final hill‑climb polishing
    # --------------------------------------------------------------
    if best_seq is not None and time.time() < deadline - 5.0:
        print("\nHill‑climb polishing")
        polished_seq, polished_val = hill_climb_permutation(
            best_seq,
            time_deadline=deadline,
            max_steps=300_000,
        )
        if polished_val < best_val:
            best_val = polished_val
            best_seq = polished_seq
            print(f"Hill‑climb improved best to {best_val:.6f}")

    # --------------------------------------------------------------
    # Safeguard: uniform vector if nothing succeeded
    # --------------------------------------------------------------
    if best_seq is None:
        n = 256
        best_seq = [1.0 / n] * n
        best_val = evaluate_sequence(best_seq)

    final_score = evaluate_sequence(best_seq)
    print(f"\nFinished search. Final reported score = {final_score:.6f}")
    return best_seq
