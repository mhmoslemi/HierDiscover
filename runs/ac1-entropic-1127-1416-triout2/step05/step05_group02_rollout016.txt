    def fast_score(a):

        return 2.0*n*max_val/(S*S)
    # Start with fallback if any

            best_seq = cand
            best_val = val
        except Exception:
            pass

    # Options of lengths
    lengths = [512, 1024, 2048, 4096]
    # Seed generation
    seed_vectors = []
    for n in lengths:
        # baseline arcsine
        a_arc = arcsine_initial(n)
        seed_vectors.append(a_arc)
        # uniform
        seed_vectors.append(np.full(n, 1.0/n))
        # random Dirichlet
        for _ in range(10):
            rnd = rng.random(n)
            rnd = rnd/ rnd.sum()
            seed_vectors.append(rnd)
        # random beta shape parameters
        for _ in range(20):
            a = rng.uniform(0.05, 5.0)
            b = rng.uniform(0.05, 5.0)
            eps = rng.uniform(0.0, max(1.0, 0.02*n))
            seed_vectors.append(generate_beta_shape(n, a, b, eps))
        # random power shape parameters
        for _ in range(20):

"""
Propose a low‑discrepancy step sequence.

The algorithm follows a multiscale scheme:

1️⃣  Small‑scale optimisation (n=256) – many random seeds are
    generated (uniform, Dirichlet, beta‑ and power‑shapes).  Each seed
    is refined with an Adam‑style sub‑gradient method on the exact
    objective f(a)=2·n·max(a∗a) (the denominator is constant after the
    simplex projection).  A short EG polishing finishes the small
    optimisation and yields a high‑quality prototype shape.

2️⃣  Upscaling – the prototype is interpolated to larger lengths
    (512, 1024, 2048, 4096).  Each up‑scaled vector receives a modest
    Adam refinement (fewer iterations, because the shape is already
    good) and a final exponentiated‑gradient polishing.

3️⃣  Linear‑programming boost – for the best candidate (if its length
    ≤ 2048) we solve a “max‑sum under the same convolution bound” LP:
    maximise Σg  subject to  f∗g ≤ M  (M=max(f∗f)).  Mixing a small
    amount of the LP solution back into f typically raises the sum
    without increasing the bound, which lowers the official score.

4️⃣  A final projected sub‑gradient / EG polish is performed on the
    current best vector, after which the (official) ‑‑evaluate_sequence‑‑
    routine is called and the best sequence is returned.

All updates keep the vector on the probability simplex
{a≥0, Σa=1}, thus the objective is scale‑invariant and the official
score is simply 2·n·max(conv(a,a)).  The whole search is bounded by the
provided time budget (default 1000 s) and never returns an empty
sequence.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Low‑level utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Linear convolution via real FFT (output length = len(a)+len(b)-1)."""
    m = a.shape[0] + b.shape[0] - 1
    n_fft = next_pow2(m)
    A = np.fft.rfft(a, n=n_fft)
    B = np.fft.rfft(b, n=n_fft)
    C = A * B
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


def compute_score(a: np.ndarray) -> float:
    """Fast surrogate for the official score (assumes Σa = 1)."""
    n = a.shape[0]
    conv = convolution_fft(a, a)
    return 2.0 * n * float(np.max(conv))


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection onto the probability simplex {x ≥ 0, Σx = 1}.
    Duchi et al., 2008.
    """
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / v.size)

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho_arr = u - cssv / np.arange(1, v.size + 1)
    rho = np.where(rho_arr > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Parameterised families (used for seeding)
# ----------------------------------------------------------------------


def arcsine_initial(n: int) -> np.ndarray:
    """Classic arcsine shape: a_i ∝ 1/√((i+½)*(n‑½‑i))."""
    i = np.arange(n, dtype=float)
    eps = 0.25
    vals = 1.0 / np.sqrt((i + 0.5) * (n - 0.5 - i) + eps)
    vals /= vals.sum()
    return vals


def generate_beta_shape(
    n: int, alpha: float, beta: float, eps: float
) -> np.ndarray:
    """
    (i+ε)^{α‑1}·(n‑1‑i+ε)^{β‑1} on {0,…,n‑1}, normalised to Σ=1.
    """
    i = np.arange(n, dtype=float)
    left = (i + eps) ** (alpha - 1.0)
    right = (n - 1 - i + eps) ** (beta - 1.0)
    base = left * right
    total = base.sum()
    if total <= 0.0 or not np.isfinite(total):
        return np.full(n, 1.0 / n)
    return base / total


def generate_power_shape(
    n: int, p: float, eps: float
) -> np.ndarray:
    """
    Generalised arcsine: a_i ∝ ((i+½)*(n‑½‑i)+ε)^{‑p}.
    p>0, eps≥0.
    """
    i = np.arange(n, dtype=float)
    denom = ((i + 0.5) * (n - 0.5 - i) + eps) ** p
    vals = 1.0 / denom
    total = vals.sum()
    if total <= 0.0 or not np.isfinite(total):
        return np.full(n, 1.0 / n)
    return vals / total


# ----------------------------------------------------------------------
# Optimisation primitives
# ----------------------------------------------------------------------


def eg_refine(
    a: np.ndarray,
    deadline: float,
    max_iter: int = 100_000,
    eta0: float = 0.3,
) -> np.ndarray:
    """
    Mirror‑descent style polishing using an exponentiated‑gradient update.
    Keeps the iterate on the simplex automatically.
    """
    n = a.shape[0]
    L = 2 * n - 1
    for it in range(1, max_iter + 1):
        if time.time() >= deadline:
            break

        conv = convolution_fft(a, a)
        max_val = float(np.max(conv))
        tol = max_val * 1e-12 + 1e-14
        mask = (np.abs(conv - max_val) <= tol).astype(float)

        subgrad = 2.0 * convolution_fft(a, mask[::-1])[L - 1 - np.arange(n)]

        eta = eta0 / (np.sqrt(it) + 1.0)
        a = a * np.exp(-eta * subgrad)

        total = a.sum()
        if total == 0.0:
            a = np.full_like(a, 1.0 / n)
        else:
            a = a / total

        if it % 5_000 == 0:
            cur = compute_score(a)
            print(f"    EG it={it:,d} score={cur:.6f}")

    return a


def adam_subgradient_descent(
    a0: np.ndarray,
    deadline: float,
    max_iter: int = 200_000,
    base_lr: float = 0.2,
) -> np.ndarray:
    """
    Adam‑style sub‑gradient descent on f(a)=2·n·max(conv(a,a)).
    The iterate is kept on the simplex after each step.
    """
    a = project_to_simplex(a0.copy())
    n = a.shape[0]
    L = 2 * n - 1

    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1 = 0.9
    beta2 = 0.999
    eps_adam = 1e-8

    for it in range(1, max_iter + 1):
        if time.time() >= deadline:
            break

        conv = convolution_fft(a, a)
        max_val = float(np.max(conv))
        tol = max_val * 1e-12 + 1e-14
        mask = (np.abs(conv - max_val) <= tol).astype(float)

        subgrad = 2.0 * convolution_fft(a, mask[::-1])[L - 1 - np.arange(n)]

        # Adam moments
        m = beta1 * m + (1 - beta1) * subgrad
        v = beta2 * v + (1 - beta2) * (subgrad ** 2)

        m_hat = m / (1 - beta1 ** it)
        v_hat = v / (1 - beta2 ** it)

        lr = base_lr / np.sqrt(it)

        a = a - lr * m_hat / (np.sqrt(v_hat) + eps_adam)
        a = project_to_simplex(a)

        if it % 5_000 == 0:
            cur = compute_score(a)
            print(f"    Adam it={it:,d} score={cur:.6f}")

    return a


# ----------------------------------------------------------------------
# Linear‑programming boost (optional, uses SciPy)
# ----------------------------------------------------------------------
def lp_boost(f: np.ndarray, M: float, t: float = 0.2) -> np.ndarray | None:
    """
    Solve   max Σg   s.t.   f∗g ≤ M , g ≥ 0 .
    Returns a mixed vector (1‑t)·f + t·g after projection onto the simplex.
    """
    try:
        import scipy.sparse as sp
        from scipy.optimize import linprog
    except Exception:
        return None

    n = f.shape[0]
    L = 2 * n - 1

    # Build sparse matrix A where A[k,i] = f_i if 0 ≤ k‑i < n else 0.
    rows = []
    cols = []
    data = []
    for i in range(n):
        row_start = i
        row_end = i + n  # exclusive
        rows.extend(range(row_start, row_end))
        cols.extend([i] * n)
        data.extend([f[i]] * n)
    A = sp.csr_matrix((data, (rows, cols)), shape=(L, n))

    b = np.full(L, M)
    c = -np.ones(n)               # maximise Σg  ⇔ minimise -Σg

    res = linprog(
        c,
        A_ub=A,
        b_ub=b,
        bounds=[(0, None)] * n,
        method="highs",
        options={"presolve": True, "time_limit": 30},
    )
    if not res.success:
        return None

    g = res.x
    # mix a little of g into f
    mixed = (1 - t) * f + t * g
    mixed = project_to_simplex(mixed)
    return mixed

def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a non‑negative step sequence with a low ``evaluate_sequence``
    value.  The routine combines a multiscale Adam/EG optimisation,
    optional LP‑based boost and a final polishing phase.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.8  # safety margin

    best_seq: list[float] = []
    best_val: float = float("inf")

    # ------------------------------------------------------------------
    # 0. Optional fallback from a previous run
    # ------------------------------------------------------------------
    if "height_sequence_1" in globals():
        try:
            fallback = [float(x) for x in globals()["height_sequence_1"]]
            val = evaluate_sequence(fallback)
            print(f"Fallback (height_sequence_1) score = {val:.6f}")
            if val < best_val:
                best_val = val
                best_seq = fallback
        except Exception:
            pass

    # ------------------------------------------------------------------
    # 1. Small‑scale optimisation (n = 256)
    # ------------------------------------------------------------------
    n_small = 256
    seeds_small = []

    # deterministic starters
    seeds_small.append(arcsine_initial(n_small))
    seeds_small.append(np.full(n_small, 1.0 / n_small))

    # random Dirichlet seeds
    for _ in range(30):
        vec = rng.random(n_small)
        vec /= vec.sum()
        seeds_small.append(vec)

    # random beta‑shape seeds
    for _ in range(30):
        a = rng.uniform(0.1, 5.0)
        b = rng.uniform(0.1, 5.0)
        eps = rng.uniform(0.0, max(1.0, 0.02 * n_small))
        seeds_small.append(generate_beta_shape(n_small, a, b, eps))

    # random power‑shape seeds
    for _ in range(30):
        p = rng.uniform(0.1, 5.0)
        eps = rng.uniform(0.0, 3.0)
        seeds_small.append(generate_power_shape(n_small, p, eps))

    # keep a handful of the most promising seeds (fast surrogate)
    scored_small = [(compute_score(v), v) for v in seeds_small]
    scored_small.sort(key=lambda x: x[0])
    seeds_small = [v for _, v in scored_small[:5]]

    refined_small = None
    refined_small_score = float("inf")
    print("\n--- Small‑scale optimisation (n=256) ---")
    for idx, init in enumerate(seeds_small, 1):
        if time.time() >= deadline - 5:
            break
        print(f" Refine small seed {idx}/{len(seeds_small)}")
        a = adam_subgradient_descent(
            init,
            deadline,
            max_iter=200_000,
            base_lr=0.25,
        )
        a = eg_refine(a, deadline, max_iter=20_000, eta0=0.2)
        sc = compute_score(a)
        print(f"   final fast score = {sc:.6f}")
        if sc < refined_small_score:
            refined_small_score = sc
            refined_small = a.copy()

    # fallback if nothing succeeded
    if refined_small is None:
        refined_small = arcsine_initial(n_small)

    # ------------------------------------------------------------------
    # 2. Upscale to larger lengths and refine
    # ------------------------------------------------------------------
    larger_lengths = [512, 1024, 2048, 4096]
    print("\n--- Upscaling & refinement ---")
    for n in larger_lengths:
        if time.time() >= deadline - 5:
            break
        # linear interpolation of the prototype
        x_small = np.linspace(0, 1, n_small)
        x_large = np.linspace(0, 1, n)
        up = np.interp(x_large, x_small, refined_small)
        up = up / up.sum()
        print(f"\nLength {n}: start refinement")

        # modest Adam run (cost grows with n)
        a = adam_subgradient_descent(
            up,
            deadline,
            max_iter=80_000,
            base_lr=0.20,
        )
        a = eg_refine(a, deadline, max_iter=20_000, eta0=0.2)

        # official evaluation
        try:
            val = evaluate_sequence(a.tolist())
            print(f"  official score = {val:.6f}")
            if val < best_val:
                best_val = val
                best_seq = a.tolist()
                print(f"  *** New globally best: {best_val:.6f} ***")
        except Exception as e:
            print(f"  evaluation failed: {e}")

    # ------------------------------------------------------------------
    # 3. LP boost on the current best (only if length ≤ 2048)
    # ------------------------------------------------------------------
    if best_seq and time.time() < deadline - 4:
        a = np.array(best_seq, dtype=float)
        n = a.shape[0]
        if n <= 2048:
            conv = convolution_fft(a, a)
            M = float(np.max(conv))
            mixed = lp_boost(a, M, t=0.2)
            if mixed is not None:
                print("\nApplied LP‑boost, polishing result …")
                mixed = eg_refine(mixed, deadline, max_iter=15_000, eta0=0.2)
                try:
                    val = evaluate_sequence(mixed.tolist())
                    print(f"  post‑LP official score = {val:.6f}")
                    if val < best_val:
                        best_val = val
                        best_seq = mixed.tolist()
                        print("*** Updated after LP boost ***")
                except Exception as e:
                    print(f"  LP‑boost evaluation error: {e}")

    # ------------------------------------------------------------------
    # 4. Final global polish (projected sub‑gradient + EG)
    # ------------------------------------------------------------------
    if best_seq and time.time() < deadline - 4:
        print("\nFinal global polishing …")
        a = np.array(best_seq, dtype=float)
        a = adam_subgradient_descent(
            a,
            deadline,
            max_iter=200_000,
            base_lr=0.12,
        )
        if time.time() < deadline - 2:
            a = eg_refine(a, deadline, max_iter=50_000, eta0=0.15)
        try:
            final_val = evaluate_sequence(a.tolist())
            print(f"Final official score = {final_val:.6f}")
            if final_val < best_val:
                best_val = final_val
                best_seq = a.tolist()
                print("*** Updated after final polish ***")
        except Exception as e:
            print(f"final polishing failed: {e}")

    # ------------------------------------------------------------------
    # 5. Guard – guarantee a return value
    # ------------------------------------------------------------------
    if not best_seq:
        n = 256
        best_seq = [1.0 / n] * n
        best_val = evaluate_sequence(best_seq)

    print(f"\nSearch finished. Best official score = {best_val:.6f}")
    return best_seq
