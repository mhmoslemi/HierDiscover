"""
Optimise a step‐function (non‑negative sequence) for the provided
`evaluate_sequence` objective.

The score is 2·n·max(conv(a,a)) / (sum(a)^2).  Because the objective is
scale‑invariant we restrict ourselves to the probability simplex
(sum(a)=1).  Minimising the maximum entry of the self‑convolution is a
non‑smooth min‑max problem.  We use a two‑phase first‑order method:

1. **Soft‑max smoothing** – a smoothed version of the max (softmax)
   is differentiated.  The gradient is a convolution of the soft‑max
   weights with the reversed current vector.  An additive projected
   gradient step (with a decreasing step‑size) is taken and the vector
   is re‑projected onto the simplex.

2. **Top‑K hard max** – once the smoothing parameter is large we switch
   to a hard‑max on a few (K) largest convolution entries.  This gives
   a sub‑gradient of the true max and finishes the optimisation.

Both phases run until the time budget (≈ 1000 s) is exhausted.  The best
vector found is finally polished by a simple stochastic hill‑climber
(swaps / mass transfers).  All operations are O(n log n) using an FFT‑
based convolution, so the routine works comfortably with n≈2000–5000
within the time limit.

The function `propose_candidate` is the entry point required by the
evaluation harness.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Basic utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Return the smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """
    Full convolution of real vectors a and b using a real FFT.
    The result has length len(a)+len(b)-1.
    """
    m = a.size + b.size - 1
    n_fft = next_pow2(m)
    A = np.fft.rfft(a, n=n_fft)
    B = np.fft.rfft(b, n=n_fft)
    C = A * B
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


def compute_score(a: np.ndarray) -> float:
    """
    Fast replica of `evaluate_sequence` for vectors that already sum to 1.
    Returns 2·n·max(convolution(a,a)).
    """
    n = a.shape[0]
    conv = convolution_fft(a, a)
    return 2.0 * n * float(np.max(conv))


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the probability simplex
    (non‑negative, sum = 1).  Implements the O(n) algorithm of
    Duchi et al. (2008).
    """
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / v.shape[0])

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(1, v.shape[0] + 1)
    cond = u - cssv / ind > 0
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


# ----------------------------------------------------------------------
# Optimisation phases
# ----------------------------------------------------------------------


def softmax_phase(a: np.ndarray,
                  deadline: float,
                  beta_start: float = 1.0,
                  beta_factor: float = 1.02,
                  lr_start: float = 0.4,
                  lr_decay: float = 0.9995,
                  max_iter: int = 200_000) -> tuple[np.ndarray, float]:
    """
    Smoothed gradient descent.
    Uses a soft‑max over the convolution entries with a temperature parameter
    beta that grows geometrically.  The gradient of the soft‑max objective is
    computed via a second convolution.
    """
    n = a.shape[0]
    best = a.copy()
    best_score = compute_score(a)

    beta = beta_start
    lr = lr_start

    for it in range(1, max_iter + 1):
        if time.time() >= deadline:
            break

        # convolution a * a
        conv = convolution_fft(a, a)
        max_c = np.max(conv)

        # soft‑max weights (stable implementation)
        w = np.exp(beta * (conv - max_c))
        w_sum = w.sum()
        if w_sum == 0:
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        # gradient = 2 * (w * a) cross‑correlation.
        # cross‑correlation = convolution of w with reversed a.
        grad_full = convolution_fft(w, a[::-1])
        g = 2.0 * grad_full[n - 1: n - 1 + n]

        # additive projected gradient step
        step = lr / np.sqrt(it)
        a_new = a - step * g
        a_new = np.maximum(a_new, 0.0)
        a_new = project_to_simplex(a_new)

        # keep best
        cur_score = compute_score(a_new)
        if cur_score < best_score:
            best_score = cur_score
            best = a_new.copy()
            # a successful step – allow a slightly larger learning rate
            lr = min(lr * 1.02, lr_start)
        else:
            lr = max(lr * lr_decay, 1e-6)

        a = a_new
        beta = min(beta * beta_factor, 1e4)

        # occasional progress report
        if it % 5000 == 0:
            print(f"Soft‑max iter {it:6d} – best score = {best_score:.6f}")

    return best, best_score


def topk_phase(a: np.ndarray,
               deadline: float,
               top_k: int = 5,
               lr_start: float = 0.5,
               lr_decay: float = 0.997,
               min_lr: float = 1e-5,
               max_iter: int = 300_000) -> tuple[np.ndarray, float]:
    """
    Hard‑max optimisation.
    At each iteration only the `top_k` largest convolution entries are kept
    (uniform weight).  The resulting sub‑gradient drives the largest peaks
    down.  An additive step with a simple learning‑rate schedule is used.
    """
    n = a.shape[0]
    best = a.copy()
    best_score = compute_score(a)

    lr = lr_start

    for it in range(1, max_iter + 1):
        if time.time() >= deadline:
            break

        conv = convolution_fft(a, a)

        # indices of the k largest entries
        idx_top = np.argpartition(-conv, top_k)[:top_k]

        # build weight vector (uniform on the chosen indices)
        w = np.zeros_like(conv)
        w[idx_top] = 1.0 / top_k

        # gradient = 2 * (w * a) cross‑correlation
        grad_full = convolution_fft(w, a[::-1])
        g = 2.0 * grad_full[n - 1: n - 1 + n]

        # projected gradient step
        a_new = a - lr * g
        a_new = np.maximum(a_new, 0.0)
        a_new = project_to_simplex(a_new)

        cur_score = compute_score(a_new)
        if cur_score < best_score:
            best_score = cur_score
            best = a_new.copy()
            lr = min(lr * 1.02, lr_start)      # reward improvement
        else:
            lr = max(lr * lr_decay, min_lr)    # otherwise decay

        a = a_new

        if it % 10000 == 0:
            print(f"Top‑K iter {it:6d} – best score = {best_score:.6f}")

    return best, best_score


def hill_climb(seq: list[float],
               deadline: float,
               max_steps: int = 400_000) -> tuple[list[float], float]:
    """
    Simple stochastic hill‑climber.
    Random swaps or fractional mass transfers are accepted only when they
    improve the true `evaluate_sequence` score.
    """
    a = np.array(seq, dtype=float)
    n = a.size

    best_a = a.copy()
    best_score = compute_score(best_a)

    step_scale = 0.08  # typical fraction of mass moved

    while max_steps > 0 and time.time() < deadline:
        cand = best_a.copy()
        if np.random.rand() < 0.5:
            # swap two positions
            i, j = np.random.randint(n, size=2)
            cand[i], cand[j] = cand[j], cand[i]
        else:
            # move a fraction from i to j
            i, j = np.random.randint(n, size=2)
            if i == j:
                continue
            delta = step_scale * cand[i]
            if delta <= 0:
                continue
            cand[i] -= delta
            cand[j] += delta

        cand = project_to_simplex(cand)
        cur_score = compute_score(cand)

        if cur_score < best_score:
            best_score = cur_score
            best_a = cand
            step_scale = min(step_scale * 1.05, 0.5)
        else:
            step_scale *= 0.97

        max_steps -= 1

    return best_a.tolist(), best_score

def propose_candidate(seed: int = 12345, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative sequence that minimises `evaluate_sequence`.
    A quick random‑sparse pre‑search produces a decent start.
    Two gradient‑based phases (soft‑max then hard‑max) are applied,
    followed by a hill‑climber polish.
    The function respects the time budget and returns a plain list of floats.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 2.0          # tiny safety margin

    best_seq = None
    best_val = float("inf")

    # ------------------------------------------------------------------
    # (Optional) start from the known construction if it exists
    # ------------------------------------------------------------------
    try:
        fallback = [float(x) for x in height_sequence_1]   # type: ignore
        fallback_val = evaluate_sequence(fallback)
        print(f"Fallback score = {fallback_val:.6f}")
        if fallback_val < best_val:
            best_val = fallback_val
            best_seq = fallback
    except Exception:
        pass

    # ------------------------------------------------------------------
    # Phase 0 – cheap random sparse initialisation
    # ------------------------------------------------------------------
    lengths = [1024, 2048, 4096]
    while time.time() < deadline - 120.0:
        n = int(np.random.choice(lengths))
        # density between 20 % and 80 %
        p = np.random.uniform(0.2, 0.8)
        k = max(2, int(round(p * n)))
        idx = np.random.choice(n, size=k, replace=False)
        a = np.zeros(n, dtype=float)
        a[idx] = 1.0 / k
        cur_score = compute_score(a)
        if cur_score < best_val:
            best_val = cur_score
            best_seq = a.tolist()
            print(f"Random init – n={n}, k={k}, score={best_val:.6f}")

    # ------------------------------------------------------------------
    # Phase 1 – soft‑max gradient descent
    # ------------------------------------------------------------------
    if best_seq is not None and time.time() < deadline - 60.0:
        a = np.array(best_seq, dtype=float)
        a = project_to_simplex(a)                     # ensure exact simplex
        a, score = softmax_phase(a, deadline - 30.0)
        if score < best_val:
            best_val = score
            best_seq = a.tolist()
            print(f"After soft‑max phase – score={best_val:.6f}")

    # ------------------------------------------------------------------
    # Phase 2 – hard top‑K gradient descent
    # ------------------------------------------------------------------
    if best_seq is not None and time.time() < deadline - 30.0:
        a = np.array(best_seq, dtype=float)
        a, score = topk_phase(a, deadline - 5.0, top_k=5)
        if score < best_val:
            best_val = score
            best_seq = a.tolist()
            print(f"After top‑K phase – score={best_val:.6f}")

    # ------------------------------------------------------------------
    # Phase 3 – hill‑climb polishing
    # ------------------------------------------------------------------
    if best_seq is not None and time.time() < deadline - 2.0:
        polished_seq, polished_score = hill_climb(best_seq, deadline, max_steps=400_000)
        if polished_score < best_val:
            best_val = polished_score
            best_seq = polished_seq
            print(f"Hill‑climb polished – score={best_val:.6f}")

    # ------------------------------------------------------------------
    # Safeguard: if nothing succeeded, fall back to a uniform vector
    # ------------------------------------------------------------------
    if best_seq is None:
        n = 256
        best_seq = [1.0 / n] * n
        best_val = evaluate_sequence(best_seq)

    # Final official evaluation (uses the provided `evaluate_sequence`)
    final_score = evaluate_sequence(best_seq)
    print(f"\nSearch finished – final official score = {final_score:.6f}")

    # The harness expects a plain Python list of floats.
    return best_seq
