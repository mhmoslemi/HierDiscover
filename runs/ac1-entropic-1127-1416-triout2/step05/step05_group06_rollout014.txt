"""
Hybrid Frank‑Wolfe + smooth‑max refinement for the max‑convolution problem.

* Stage 1 – diverse seeds (arcsine, power‑law, Dirichlet, mixtures) are
  generated and the one with the smallest fast score (2·n·max(a∗a)) is kept.

* Stage 2 – a Frank‑Wolfe‑style iteration:
    – Given a current f (∑f=1) compute M = max(f∗f).
    – Solve the linear program
          maximize   1ᵀb
          subject to conv(f,b) ≤ M   (2n‑1 inequalities)
                     b ≥ 0
      The LP is tiny (≈4 000 constraints, 2 000 variables) and solved with
      SciPy’s HiGHS interior‑point algorithm.
    – The optimiser b is normalised to a probability vector g.
    – The optimal mixing weight γ∈[0,1] for the update
          f←(1‑γ)·f + γ·g
      is found by a cheap ternary search on the true (fast) objective.
    – The new f is renormalised (∑f=1).  This step provably never increases
      the max‑convolution and typically reduces it.

* Stage 3 – after the LP loop we polish the best vector with an
  exponentiated‑gradient sub‑gradient method on the *exact* max‑convolution
  (the same routine used in the previous submission).

The whole routine respects the wall‑clock budget (default 1000 s) and always
returns a feasible list of non‑negative floats.
"""

import time
import numpy as np
import scipy.sparse as sp
import scipy.optimize as opt
from typing import List, Tuple

# ----------------------------------------------------------------------
# Low‑level numeric utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full linear convolution using real FFT (zero‑padded to a power of two)."""
    m = len(a) + len(b) - 1
    n_fft = next_pow2(m)
    A = np.fft.rfft(a, n=n_fft)
    B = np.fft.rfft(b, n=n_fft)
    C = A * B
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


def compute_score(a: np.ndarray) -> float:
    """Fast version of the official objective.
    The vector a is assumed non‑negative and ∑a = 1, so the score is simply
    2·n·max(a∗a)."""
    n = len(a)
    conv = convolution_fft(a, a)
    return 2.0 * n * float(conv.max())


# ----------------------------------------------------------------------
# Families of initial shapes
# ----------------------------------------------------------------------


def arcsine_initial(n: int) -> np.ndarray:
    """Discrete arcsine density (normalised to sum = 1)."""
    i = np.arange(n, dtype=float)
    eps = 0.25
    vals = 1.0 / np.sqrt((i + 0.5) * (n - 0.5 - i) + eps)
    return vals / vals.sum()


def generate_powerlaw(n: int, p: float, eps: float) -> np.ndarray:
    """Symmetric power‑law a_i ∝ ((i+ε)(n‑1‑i+ε))⁻ᵖ."""
    i = np.arange(n, dtype=float)
    base = ((i + eps) * (n - 1 - i + eps)) ** (-p)
    total = base.sum()
    if total == 0.0:
        return np.full(n, 1.0 / n)
    return base / total


def random_dirichlet(n: int, concentration: float,
                     rng: np.random.Generator) -> np.ndarray:
    """Sample a Dirichlet‑like vector and normalise to sum = 1."""
    if concentration <= 0:
        samp = rng.exponential(1.0, size=n) ** (1.0 / max(0.1, concentration))
    else:
        samp = rng.gamma(concentration, 1.0, size=n)
    samp += 1e-12
    return samp / samp.sum()


# ----------------------------------------------------------------------
# Linear‑programming sub‑problem (Frank–Wolfe step)
# ----------------------------------------------------------------------


def build_conv_pattern(n: int) -> Tuple[np.ndarray, np.ndarray, Tuple[int, int]]:
    """
    Returns (row_idx, col_idx, shape) describing the non‑zero pattern of the
    Toeplitz matrix that maps b → conv(f,b).  The pattern does not depend on f.
    """
    rows = []
    cols = []
    for k in range(2 * n - 1):
        i0 = max(0, k - (n - 1))
        i1 = min(k, n - 1)
        rows.extend([k] * (i1 - i0 + 1))
        cols.extend(range(i0, i1 + 1))
    return (np.array(rows, dtype=np.int32),
            np.array(cols, dtype=np.int32),
            (2 * n - 1, n))


def linear_program_step(f: np.ndarray,
                        M: float,
                        rows: np.ndarray,
                        cols: np.ndarray,
                        shape: Tuple[int, int]) -> np.ndarray:
    """
    Solve   max 1ᵀb   s.t.  conv(f,b) ≤ M,  b ≥ 0.
    Returns the optimiser b (may be all zeros if the problem is infeasible).
    """
    # construct sparse matrix A where A·b = conv(f,b)
    data = f[cols]                           # each non‑zero equals the appropriate f_i
    A = sp.csr_matrix((data, (rows, cols)), shape=shape)

    n = f.shape[0]
    c = -np.ones(n)                          # maximise 1ᵀb  → minimise –1ᵀb
    ub = np.full(shape[0], M)

    res = opt.linprog(c,
                      A_ub=A,
                      b_ub=ub,
                      bounds=[(0, None)] * n,
                      method='highs',
                      options={'presolve': True})
    if not res.success:
        return np.zeros_like(f)
    return res.x


# ----------------------------------------------------------------------
# Mixing weight optimisation (cheap ternary search on the fast score)
# ----------------------------------------------------------------------


def optimal_gamma(f: np.ndarray, g: np.ndarray,
                  iters: int = 12) -> float:
    """Find γ∈[0,1] minimising the fast objective of (1‑γ)·f + γ·g."""
    lo, hi = 0.0, 1.0
    for _ in range(iters):
        m1 = lo + (hi - lo) / 3.0
        m2 = hi - (hi - lo) / 3.0
        val1 = compute_score((1.0 - m1) * f + m1 * g)
        val2 = compute_score((1.0 - m2) * f + m2 * g)
        if val1 < val2:
            hi = m2
        else:
            lo = m1
    return (lo + hi) / 2.0


# ----------------------------------------------------------------------
# Exponentiated sub‑gradient polish (same as previous top‑performing code)
# ----------------------------------------------------------------------


def _max_conv_gradient(a: np.ndarray, conv: np.ndarray) -> np.ndarray:
    """Sub‑gradient of max_k (a∗a)_k."""
    n = len(a)
    max_val = conv.max()
    mask = np.isclose(conv, max_val, atol=1e-15, rtol=0.0)
    active = np.where(mask)[0]
    grad = np.zeros_like(a)

    for k in active:
        lo = max(0, k - (n - 1))
        hi = min(k, n - 1)
        idx = np.arange(lo, hi + 1, dtype=int)
        grad[idx] += 2.0 * a[k - idx]

    if len(active) > 1:
        grad /= len(active)
    return grad


def optimise_subgradient(initial_a: np.ndarray,
                         deadline: float,
                         *,
                         eta0: float = 0.6,
                         max_iter: int = 200_000) -> Tuple[np.ndarray, float]:
    """Exponentiated sub‑gradient descent directly on max‑convolution."""
    a = np.maximum(initial_a, 1e-30)
    a = a / a.sum()
    best_a = a.copy()
    best_sc = compute_score(a)
    n = len(a)

    for it in range(1, max_iter + 1):
        if time.time() >= deadline:
            break

        conv = convolution_fft(a, a)
        grad = _max_conv_gradient(a, conv)

        eta = eta0 / np.sqrt(it)
        a = a * np.exp(-eta * grad)

        a_sum = a.sum()
        a = a / a_sum if a_sum != 0.0 else np.full_like(a, 1.0 / n)

        cur_sc = compute_score(a)
        if cur_sc < best_sc:
            best_sc = cur_sc
            best_a = a.copy()

        # occasional progress report (does not affect timing)
        if it % 5000 == 0:
            print(f"    subgrad it={it:,d} cur={cur_sc:.6f} best={best_sc:.6f}")

    return best_a, best_sc
