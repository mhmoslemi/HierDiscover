"""
Hybrid LP‑subgradient optimizer for the max‑auto‑convolution problem.

* Start from a smooth seed (arcsine, Dirichlet or a known fallback).
* Repeatedly solve a tiny linear programme:
      maximise   Σ g_i
      s.t.       g_i ≥ 0
                 (f * g)_k ≤ M   for every k
  where M = max(f * f).  The LP tells us how much total mass can be
  redistributed without increasing the current maximum.
* Scale the solution back to the simplex (sum = 1) – this reduces the
  bound by a factor 1/Σg i – and mix it into the current vector.
* After a few dozen LP steps we polish the result with a classic
  exponentiated sub‑gradient descent on the true max‑convolution.
* The whole routine respects the provided wall‑clock budget and always
  returns a feasible non‑negative list of floats.
"""

import time
import numpy as np
from typing import List

# ---- basic numeric utilities -------------------------------------------------

def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full linear convolution using real FFT (zero‑padded to a power of two)."""
    m = len(a) + len(b) - 1
    n_fft = next_pow2(m)
    A = np.fft.rfft(a, n=n_fft)
    B = np.fft.rfft(b, n=n_fft)
    C = A * B
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


def compute_score(a: np.ndarray) -> float:
    """Fast version of the official objective (sum(a)=1 ⇒ 2 n·max(conv))."""
    n = len(a)
    conv = convolution_fft(a, a)
    max_val = float(np.max(conv))
    # sum is assumed to be 1; if not we normalise here.
    s = float(np.sum(a))
    if s < 1e-12:
        return float("inf")
    return 2.0 * n * max_val / (s * s)


# ---- families of initial seeds ----------------------------------------------

def arcsine_initial(n: int) -> np.ndarray:
    """Discrete arcsine density (normalised to sum = 1)."""
    i = np.arange(n, dtype=float)
    eps = 0.25
    vals = 1.0 / np.sqrt((i + 0.5) * (n - 0.5 - i) + eps)
    return vals / vals.sum()


def random_dirichlet(n: int, concentration: float, rng: np.random.Generator) -> np.ndarray:
    """Sample a Dirichlet‑like vector and normalise to sum = 1."""
    if concentration <= 0:
        samp = rng.exponential(1.0, size=n) ** (1.0 / max(0.1, concentration))
    else:
        samp = rng.gamma(concentration, 1.0, size=n)
    samp += 1e-12
    return samp / samp.sum()


def perturb(a: np.ndarray, rng: np.random.Generator, scale: float = 0.02) -> np.ndarray:
    """Add a tiny Gaussian perturbation and renormalise to the simplex."""
    b = a + rng.normal(0.0, scale, size=a.shape)
    b = np.maximum(b, 0.0)
    s = b.sum()
    if s == 0.0:
        return np.full_like(a, 1.0 / len(a))
    return b / s


# ---- linear‑programming step -------------------------------------------------

def _build_conv_matrix(f: np.ndarray) -> np.ndarray:
    """
    Build the (2n‑1)×n matrix that maps g → f∗g.
    Row k contains the coefficients f_i that multiply g_{k−i}.
    """
    from scipy.linalg import toeplitz
    n = len(f)
    col = np.concatenate([f, np.zeros(n - 1)])
    row = np.concatenate([f[:1], np.zeros(n - 1)])
    return toeplitz(col, row)   # dense float64 array


def _solve_lp_max_sum(f: np.ndarray, M: float) -> np.ndarray:
    """
    Linear programme:
        maximise   Σ g_i
        s.t.       g_i ≥ 0
                   (f * g)_k ≤ M  for every k
    Returns the optimal g (may be dense) or, on failure, a copy of f.
    """
    import scipy.optimize

    n = len(f)
    A = _build_conv_matrix(f)                     # shape (2n‑1, n)
    b = np.full(2 * n - 1, M, dtype=float)
    c = -np.ones(n)                               # minimise −Σ g_i  ⇔ maximise Σ g_i
    bounds = [(0, None)] * n

    # ‘highs’ is fast and reliable for medium‑size LPs.
    res = scipy.optimize.linprog(c, A_ub=A, b_ub=b, bounds=bounds,
                                 method='highs', options={'presolve': True})
    if res.success and res.x is not None:
        g = np.maximum(res.x, 0.0)                # numerical safety
    else:
        # fallback – keep the current point unchanged
        g = f.copy()
    return g


def _lp_iterative_optimize(f0: np.ndarray,
                           deadline: float,
                           max_iters: int = 30,
                           mix_start: float = 0.25) -> np.ndarray:
    """
    Repeatedly apply the LP step described above, mixing the solution into
    the current vector.  The loop stops when the wall‑clock deadline is
    reached or after ``max_iters`` iterations.
    """
    f = f0.copy()
    best_f = f.copy()
    best_sc = compute_score(f)
    mix = mix_start
    it = 0

    while time.time() < deadline - 1.0 and it < max_iters:
        it += 1
        conv = convolution_fft(f, f)
        M = float(conv.max())

        g = _solve_lp_max_sum(f, M)
        sum_g = float(g.sum())
        if sum_g <= 1e-12:
            break

        # Scale g back onto the simplex (sum = 1) – this reduces the bound.
        g_scaled = g / sum_g

        # Exponential‑mixing (convex combination) with a slowly decreasing step.
        f = (1.0 - mix) * f + mix * g_scaled
        f /= f.sum()                     # protect against round‑off drift

        cur_sc = compute_score(f)
        if cur_sc < best_sc:
            best_sc = cur_sc
            best_f = f.copy()
            print(f"    LP iter {it:3d}  score {cur_sc:.6f}  M {M:.6f}  Σg {sum_g:.6f}")

        # gently decay the mixing coefficient so we take smaller steps later
        mix = max(0.05, mix * 0.94)

    return best_f


# ---- sub‑gradient polishing (identical to the previous script) -------------

def _max_conv_gradient(a: np.ndarray, conv: np.ndarray) -> np.ndarray:
    """
    Sub‑gradient of max_k (a∗a)_k.
    For each active offset k we add the contribution 2·a_{k‑j}.
    """
    n = len(a)
    max_val = conv.max()
    mask = np.isclose(conv, max_val, atol=1e-15, rtol=0.0)
    active = np.where(mask)[0]
    grad = np.zeros_like(a)

    for k in active:
        lo = max(0, k - (n - 1))
        hi = min(k, n - 1)
        idx = np.arange(lo, hi + 1, dtype=int)
        grad[idx] += 2.0 * a[k - idx]

    if len(active) > 1:
        grad /= len(active)
    return grad


def optimise_subgradient(initial_a: np.ndarray,
                         deadline: float,
                         *,
                         eta0: float = 0.6,
                         max_iter: int = 200_000,
                         noise_every: int = 0,
                         noise_scale: float = 0.0) -> np.ndarray:
    """
    Exponentiated sub‑gradient descent on the true max‑convolution.
    Returns the best vector seen (sum normalised to 1).
    """
    a = np.maximum(initial_a, 1e-30)
    a = a / a.sum()

        cur_sc = compute_score(a)
        if cur_sc < best_sc:
            best_sc = cur_sc
            best_a = a.copy()

        if it % 2000 == 0:
            print(f"    subgrad iter={it:,d} cur={cur_sc:.6f} best={best_sc:.6f}")

    return best_a

"""
Hybrid LP‑subgradient optimizer for the max‑auto‑convolution problem.

* Start from a smooth seed (arcsine, Dirichlet or a known fallback).
* Repeatedly solve a tiny linear programme:
      maximise   Σ g_i
      s.t.       g_i ≥ 0
                 (f * g)_k ≤ M   for every k
  where M = max(f * f).  The LP tells us how much total mass can be
  redistributed without increasing the current maximum.
* Scale the solution back to the simplex (sum = 1) – this reduces the
  bound by a factor 1/Σg_i – and mix it into the current vector.
* After a few dozen LP steps we polish the result with a classic
  exponentiated sub‑gradient descent on the true max‑convolution.
* The whole routine respects the provided wall‑clock budget and always
  returns a feasible non‑negative list of floats.
"""

def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full linear convolution using real FFT (zero‑padded to a power of two)."""
    m = len(a) + len(b) - 1
    n_fft = next_pow2(m)
    A = np.fft.rfft(a, n=n_fft)
    B = np.fft.rfft(b, n=n_fft)
    C = A * B
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv

    """
    Fast version of the official objective.
    The vector is assumed non‑negative; if its sum is 0 the score is infinite.
    """

    conv = convolution_fft(a, a)
    max_val = float(np.max(conv))
    s = float(np.sum(a))
