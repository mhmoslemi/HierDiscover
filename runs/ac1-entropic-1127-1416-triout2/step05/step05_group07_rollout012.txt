def project_to_simplex(v):
    # v: np.ndarray, shape (n,)
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(1, n+1)
    cond = u - cssv / ind > 0
    rho = ind[cond].max()
    theta = cssv[cond].max() / rho
    w = np.maximum(v - theta, 0)
    return w
```

def optimize_smoothed(
    init_a,
    deadline,
    lam: float = 10.0,
    eta0: float = 0.1,
    max_iter: int = 200_000,
):
    a = np.maximum(init_a, 1e-30)
    a = a / a.sum()
    best_a = a.copy()
    best_sc = compute_score(a)

    n = len(a)
    a_rev = a[::-1]
    for it in range(1, max_iter+1):
        if time.time() >= deadline:
            break
        conv = convolution_fft(a, a)
        max_conv = conv.max()
        # compute weights for smoothed max: w_k = exp(lam*(conv_k - max_conv))
        # to avoid overflow, compute w = exp(lam*(conv - max_conv))
        # max weight =1.
        w = np.exp(lam * (conv - max_conv))
        # gradient: 2 * (w * a_rev)[n-1:n-1+n]
        grad_full = convolution_fft(w, a_rev)
        grad = 2.0 * grad_full[n-1:n-1+len(a)]

        # step size schedule; use diminishing step: eta = eta0 / sqrt(it)
        eta = eta0 / np.sqrt(it)

        a = a - eta * grad
        # ensuring non-negativity after step maybe necessary before projection
        # project onto simplex
        # use projection algorithm
        a = project_to_simplex(a)

        cur_sc = compute_score(a)
        if cur_sc < best_sc:
            best_sc = cur_sc
            best_a = a.copy()
    return best_a, best_sc

def project_to_simplex(v):
    # returns projection onto simplex {x >=0, sum x = 1}
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    rho = np.where(u - cssv / np.arange(1, n+1) > 0)[0]
    if len(rho) == 0:
        # all zero?
        return np.full_like(v, 1.0/n)
    rho = rho[-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0)
    return w
```

"""
Propose a low‑score step‑function.

The algorithm works in three stages:

1. **Warm‑start** – a discrete arcsine density (the known
   “flat‑convolution” distribution) is generated for a fairly large
   length (default 8192).  This already yields a convolution that is
   close to uniform.

2. **Multiplicative sub‑gradient descent** – an exponentiated‑gradient
   (mirror‑descent on the simplex) reduces the maximal self‑convolution.
   The update is `a ← a·exp(−η·g)` where ``g`` is a sub‑gradient of the
   max‑convolution.

3. **Smoothed (log‑sum‑exp) refinement** – the non‑smooth objective
   `max_k (a∗a)_k` is approximated by the smooth function
   `φ_λ(a) = (1/λ)·log Σ_k exp(λ·(a∗a)_k)`.  Its gradient can be computed
   efficiently with one extra convolution.  A standard projected
   gradient descent with decreasing step size is applied, gradually
   increasing the smoothing parameter `λ` (continuation) to tighten the
   approximation.

All operations respect the probability simplex (non‑negative,
unit‑sum) and the whole search stops before the 1000 s wall‑clock
budget.  The best sequence seen is finally evaluated with the official
`evaluate_sequence` function and returned.
"""

import time
from typing import List, Tuple

import numpy as np

# ----------------------------------------------------------------------
# Numerical utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """
    Full linear convolution using a real FFT (fast and accurate).
    """
    m = len(a) + len(b) - 1
    n_fft = next_pow2(m)
    A = np.fft.rfft(a, n=n_fft)
    B = np.fft.rfft(b, n=n_fft)
    C = A * B
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


def compute_score(a: np.ndarray) -> float:
    """
    Cheap surrogate of `evaluate_sequence`:   2·n·max(a∗a)/(sum(a)²).
    The vector *a* is assumed to be non‑negative.
    """
    n = len(a)
    conv = convolution_fft(a, a)
    max_val = float(conv.max())
    s = float(a.sum())
    if s < 1e-12:
        return float("inf")
    return 2.0 * n * max_val / (s * s)


# ----------------------------------------------------------------------
# Simplex projection (Euclidean) – Duchi et al. (2008)
# ----------------------------------------------------------------------


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Project a vector onto the probability simplex {x≥0, Σx=1}."""
    if v.ndim != 1:
        raise ValueError("Simplex projection expects a 1‑D array")
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all entries are zero – return the uniform distribution
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1]          # last true index (1‑based)
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Families of initial shapes
# ----------------------------------------------------------------------


def arcsine_initial(n: int) -> np.ndarray:
    """
    Discrete arcsine density a_i ∝ 1/√((i+½)(n‑½‑i)).
    The vector is normalised to sum to 1.
    """
    i = np.arange(n, dtype=float)
    eps = 0.25
    vals = 1.0 / np.sqrt((i + 0.5) * (n - 0.5 - i) + eps)
    vals /= vals.sum()
    return vals


# ----------------------------------------------------------------------
# Sub‑gradient optimiser (exponentiated / mirror‑descent)
# ----------------------------------------------------------------------


def _max_conv_gradient(a: np.ndarray, conv: np.ndarray) -> np.ndarray:
    """
    Sub‑gradient of   max_k (a∗a)_k.
    For each active offset k the gradient is 2·a_{k‑j}.
    If several offsets are active the contribution is averaged.
    """
    n = len(a)
    max_val = conv.max()
    # active offsets – tolerance enough because values differ by ~1e‑7
    active = np.where(np.isclose(conv, max_val, atol=1e-15))[0]
    grad = np.zeros_like(a)

    for k in active:
        lo = max(0, k - (n - 1))
        hi = min(k, n - 1)
        idx = np.arange(lo, hi + 1, dtype=int)
        grad[idx] += 2.0 * a[k - idx]

    if len(active) > 1:
        grad /= len(active)
    return grad


def optimise_exponential(
    init_a: np.ndarray,
    deadline: float,
    *,
    eta0: float = 0.6,
    max_iter: int = 200_000,
    noise_every: int = 0,
    noise_scale: float = 0.0,
) -> Tuple[np.ndarray, float]:
    """
    Exponentiated‑gradient (mirror‑descent) on the non‑smooth objective
    Φ(a)=max_k (a∗a)_k, constrained to the probability simplex.
    Returns the best vector seen and its surrogate score.
    """
    a = np.maximum(init_a, 1e-30)
    a = a / a.sum()
    best_a = a.copy()
    best_sc = compute_score(a)

    n = len(a)

    for it in range(1, max_iter + 1):
        if time.time() >= deadline:
            break

        conv = convolution_fft(a, a)
        grad = _max_conv_gradient(a, conv)

        eta = eta0 / np.sqrt(it)
        a = a * np.exp(-eta * grad)          # multiplicative update

        if noise_every and (it % noise_every == 0):
            a += np.random.randn(n) * noise_scale
            a = np.maximum(a, 0.0)

        a = project_to_simplex(a)

        cur_sc = compute_score(a)
        if cur_sc < best_sc:
            best_sc = cur_sc
            best_a = a.copy()

        if it % 10_000 == 0:
            print(f"    exp‑grad it={it:,d} cur={cur_sc:.6f} best={best_sc:.6f}")

    return best_a, best_sc


# ----------------------------------------------------------------------
# Smoothed (log‑sum‑exp) optimiser
# ----------------------------------------------------------------------


def optimise_smoothed(
    init_a: np.ndarray,
    deadline: float,
    *,
    lam_start: float = 5.0,
    lam_end: float = 30.0,
    eta0: float = 0.3,
    max_iter: int = 200_000,
) -> Tuple[np.ndarray, float]:
    """
    Gradient descent on the smooth approximation
        φ_λ(a) = (1/λ)·log Σ_k exp(λ·(a∗a)_k).

    The smoothing parameter λ is increased linearly from ``lam_start``
    to ``lam_end`` (continuation).  After each step the vector is
    projected back onto the simplex.
    """
    a = np.maximum(init_a, 1e-30)
    a = a / a.sum()
    best_a = a.copy()
    best_sc = compute_score(a)

    n = len(a)
    a_rev = a[::-1]            # constant reverse for the whole run

    for it in range(1, max_iter + 1):
        if time.time() >= deadline:
            break

        # ------- current smoothing parameter -------
        lam = lam_start + (lam_end - lam_start) * (it - 1) / max_iter

        # ------- convolution a*a -------
        conv = convolution_fft(a, a)
        max_conv = conv.max()

        # ------- smoothed weights (log‑sum‑exp) -------
        # stability: subtract max_conv before exponentiation
        w = np.exp(lam * (conv - max_conv))
        w /= w.sum()                               # normalised weight vector

        # ------- gradient of the smoothed objective -------
        grad_full = convolution_fft(w, a_rev)      # length = 3n‑2
        grad = 2.0 * grad_full[n - 1 : n - 1 + n]  # slice to length n

        # ------- step (projected gradient descent) -------
        eta = eta0 / np.sqrt(it)
        a = a - eta * grad
        a = project_to_simplex(a)

        cur_sc = compute_score(a)
        if cur_sc < best_sc:
            best_sc = cur_sc
            best_a = a.copy()

        if it % 10_000 == 0:
            print(
                f"    smooth it={it:,d} lam={lam:.1f} cur={cur_sc:.6f} best={best_sc:.6f}"
            )

    return best_a, best_sc

def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs) -> List[float]:
    """
    Search for a non‑negative step‑function with a low ``evaluate_sequence`` score.
    The routine:
        * optionally uses a fallback construction if present,
        * builds a large arcsine‑shape vector (default n = 8192),
        * refines it with exponentiated‑gradient sub‑gradient descent,
        * performs a continuation of smoothed gradient descent,
        * finally evaluates the best candidate and returns it.
    """
    rng = np.random.default_rng(seed)
    start = time.time()
    # keep a tiny safety margin for the final evaluation call
    deadline = start + budget_s - 2.0

    best_seq: List[float] = []
    best_val: float = float("inf")
    best_vec: np.ndarray = None

    # ------------------------------------------------------------------
    # 0.  Optional fallback from a previously discovered construction.
    # ------------------------------------------------------------------
    try:
        fallback = [float(x) for x in height_sequence_1]  # type: ignore
        val = evaluate_sequence(fallback)
        print(f"Fallback (height_sequence_1) score = {val:.6f}")
        if val < best_val:
            best_val = val
            best_seq = fallback
            best_vec = np.array(fallback, dtype=float)
            best_vec = best_vec / best_vec.sum()
    except Exception:
        pass

    # ------------------------------------------------------------------
    # 1.  Choose a reasonably large problem size.
    # ------------------------------------------------------------------
    # If there is still ample time we go for n=16384, otherwise n=8192.
    # This heuristic keeps the total runtime inside the budget.
    remaining = deadline - time.time()
    if remaining > 300.0:
        n = 16384
    else:
        n = 8192
    print(f"\n=== Main optimisation length = {n} ===")

    # ------------------------------------------------------------------
    # 2.  Warm‑start with the discrete arcsine density.
    # ------------------------------------------------------------------
    a0 = arcsine_initial(n)

    # ------------------------------------------------------------------
    # 3.  Exponentiated‑gradient sub‑gradient descent.
    # ------------------------------------------------------------------
    time_left = deadline - time.time()
    # empirical budget: ≈ 250 iterations per second for n≈8k,
    #   150 it/s for n≈16k – allocate roughly half of the remaining time.
    iters_sub = int(min(300_000, max(20_000, time_left * (250 if n <= 8192 else 150) / 2)))
    print(f"Running exponential sub‑gradient for ≈{iters_sub} iterations")
    a_sub, sc_sub = optimise_exponential(
        init_a=a0,
        deadline=deadline,
        eta0=0.7,
        max_iter=iters_sub,
        noise_every=0,
        noise_scale=0.0,
    )
    print(f"  After sub‑gradient best surrogate score = {sc_sub:.6f}")

    # ------------------------------------------------------------------
    # 4.  Smoothed (log‑sum‑exp) refinement with continuation on λ.
    # ------------------------------------------------------------------
    time_left = deadline - time.time()
    iters_smooth = int(min(400_000, max(10_000, time_left * (250 if n <= 8192 else 120))))
    print(f"Running smoothed refinement for ≈{iters_smooth} iterations")
    a_refined, sc_refined = optimise_smoothed(
        init_a=a_sub,
        deadline=deadline,
        lam_start=5.0,
        lam_end=30.0,
        eta0=0.4,
        max_iter=iters_smooth,
    )
    print(f"  After smoothing best surrogate score = {sc_refined:.6f}")

    # ------------------------------------------------------------------
    # 5.  Official evaluation of the best candidate.
    # ------------------------------------------------------------------
    val = evaluate_sequence(a_refined.tolist())
    print(f"Official evaluation of final candidate = {val:.6f}")
    if val < best_val:
        best_val = val
        best_seq = a_refined.tolist()
        best_vec = a_refined.copy()
        print(f"*** New overall best = {best_val:.6f} ***")

    # ------------------------------------------------------------------
    # 6.  (Optional) tiny final polishing with a higher λ if time permits.
    # ------------------------------------------------------------------
    if best_vec is not None and time.time() < deadline - 5.0:
        time_left = deadline - time.time()
        iters_final = int(min(200_000, max(5_000, time_left * 200)))
        print("\nFinal polishing with very strong smoothing")
        a_final, sc_final = optimise_smoothed(
            init_a=best_vec,
            deadline=deadline,
            lam_start=30.0,
            lam_end=50.0,
            eta0=0.2,
            max_iter=iters_final,
        )
        final_val = evaluate_sequence(a_final.tolist())
        print(f"Final polishing score = {final_val:.6f}")
        if final_val < best_val:
            best_val = final_val
            best_seq = a_final.tolist()
            print("*** Updated best after final polishing ***")

    # ------------------------------------------------------------------
    # 7.  Guard – always return a feasible sequence.
    # ------------------------------------------------------------------
    if not best_seq:
        best_seq = [1.0] * 256
        best_val = evaluate_sequence(best_seq)

    print(f"\nSearch finished. Best reported score = {best_val:.6f}")
    return best_seq
