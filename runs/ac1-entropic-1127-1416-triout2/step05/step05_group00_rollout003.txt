"""
Propose a step‑function (non‑negative vector) with a low score for
`evaluate_sequence`.  The algorithm is a multi‑stage continuation
gradient descent on a smooth approximation of the objective
    f(a) = max_i (a * a)[i]        (a on the probability simplex)
using a soft‑max (log‑sum‑exp) smoothing.  The vector is parametrised by
log‑its `z` (so that a = softmax(z) automatically satisfies a ≥ 0 and
∑a = 1).  For a sequence of decreasing smoothing parameters ε we run a
few L‑BFGS‑B optimisations, each starting from the result of the previous
stage.  Several random restarts (including sparse‑support starts) are
tried; the best result is finally polished by a fast hill‑climbing
procedure.

The code uses only top‑level functions (no closures, no lambdas) and
relies on NumPy and SciPy (already available in the execution
environment).  All work respects the 1000 s budget.
"""

import time
import math
import numpy as np
from scipy.optimize import minimize

# ----------------------------------------------------------------------
# Helper utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Return the smallest power of two greater than or equal to x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """
    Full linear convolution of two real vectors using an FFT of length
    a power of two (fast and accurate for the sizes we use).
    """
    m = a.size + b.size - 1
    n_fft = next_pow2(m)
    A = np.fft.rfft(a, n=n_fft)
    B = np.fft.rfft(b, n=n_fft)
    C = A * B
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


def compute_score(a: np.ndarray) -> float:
    """
    Fast replica of `evaluate_sequence` for the case ∑a = 1.
    Returns 2·n·max(a⊛a).
    """
    n = a.shape[0]
    conv = convolution_fft(a, a)
    return 2.0 * n * float(np.max(conv))


def softmax(z: np.ndarray) -> np.ndarray:
    """Numerically stable softmax (returns a probability vector)."""
    z_max = np.max(z)
    e = np.exp(z - z_max)
    return e / np.sum(e)


# ----------------------------------------------------------------------
# Smoothed max objective and its gradient (w.r.t. the logits ``z``)
# ----------------------------------------------------------------------


def smoothed_max_and_grad(z: np.ndarray,
                          eps: float,
                          n: int) -> tuple[float, np.ndarray]:
    """
    Given logits `z` (length n) compute
        f(z) = (1/eps) * logsumexp(eps * conv(a,a)),
    where a = softmax(z).  Return the scalar value and the gradient
    ∂f/∂z (same shape as ``z``).  The gradient is derived analytically
    using the chain rule and the convolution structure.
    """
    # a on the simplex
    a = softmax(z)

    # convolution a * a  (length 2n‑1)
    conv = convolution_fft(a, a)

    # ----- smoothed max (log‑sum‑exp) -----
    c_max = np.max(conv)                     # for numerical stability
    exp_term = np.exp(eps * (conv - c_max))   # non‑negative, no overflow
    w_sum = np.sum(exp_term)                 # denominator of the softmax
    # f = (log Σ exp(eps*conv)) / eps
    f_val = (c_max + math.log(w_sum)) / eps

    # ----- gradient w.r.t. a -----
    # normalized weights w_i = exp(eps*conv_i)/ Σ exp(...)
    w = exp_term / w_sum  # shape (2n-1,)

    # convolution of w with the reversed a gives Σ_i w_i a_{i-k}
    a_rev = a[::-1]
    conv_w_a_rev = convolution_fft(w, a_rev)          # length 3n‑2
    grad_a = 2.0 * conv_w_a_rev[n - 1:n - 1 + n]      # slice of length n

    # ----- gradient w.r.t. logits z (softmax Jacobian) -----
    # ∂f/∂z = a * (grad_a - <a, grad_a>)
    dot_ag = np.dot(a, grad_a)
    grad_z = a * (grad_a - dot_ag)

    return f_val, grad_z


# ----------------------------------------------------------------------
# Continuation optimisation (multiple ε values)
# ----------------------------------------------------------------------


def optimise_from_logits(init_z: np.ndarray,
                        n: int,
                        eps_schedule: list[float],
                        time_deadline: float) -> np.ndarray:
    """
    Run a cascade of L‑BFGS‑B optimisations, each with a different smoothing
    parameter ε taken from ``eps_schedule`` (largest → smallest).
    The optimisation stops early if the global deadline is reached.
    Returns the logits of the best vector found.
    """
    z = init_z.copy()

    for eps in eps_schedule:
        if time.time() >= time_deadline:
            break

        # wrapper required by ``scipy.optimize.minimize`` (the extra args
        # are passed unchanged to the objective function).
        args = (eps, n)

        # minimise f(z) – note that we set ``jac=True`` and the function
        # returns both value and gradient.
        res = minimize(fun=smoothed_max_and_grad,
                       x0=z,
                       args=args,
                       method="L-BFGS-B",
                       jac=True,
                       options={"maxiter": 300,
                                "ftol": 1e-12,
                                "disp": False})

        # keep the latest point as the starting point for the next ε
        z = res.x

    return z


# ----------------------------------------------------------------------
# Hill‑climbing polishing (swap / mass transfer)
# ----------------------------------------------------------------------


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the probability simplex
    (non‑negative entries that sum to 1).  Standard O(n log n) algorithm.
    """
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / v.shape[0])

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, v.shape[0] + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.where(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def hill_climb(seq: list[float],
               time_deadline: float,
               max_steps: int = 200_000) -> tuple[list[float], float]:
    """
    Simple stochastic hill‑climber that either swaps two entries or moves a
    fraction of mass from one index to another.  Only moves that improve
    the true (unsmoothed) score are kept.
    """
    a = np.array(seq, dtype=float)
    n = a.shape[0]

    best_a = a.copy()
    best_score = compute_score(best_a)

    step_scale = 0.15  # start with modest transfers

    while max_steps > 0 and time.time() < time_deadline:
        cand = best_a.copy()
        if np.random.rand() < 0.5:   # swap
            i, j = np.random.randint(n, size=2)
            cand[i], cand[j] = cand[j], cand[i]
        else:                        # transfer a fraction of mass
            i, j = np.random.randint(n, size=2)
            if i != j:
                delta = step_scale * best_a[i]
                if delta > 0:
                    cand[i] -= delta
                    cand[j] += delta
                    if cand[i] < 0:
                        cand[i] = 0.0

        cand = project_to_simplex(cand)
        cur_score = compute_score(cand)

        if cur_score < best_score:
            best_score = cur_score
            best_a = cand
            step_scale = min(step_scale * 1.08, 0.5)
        else:
            step_scale *= 0.97

        max_steps -= 1

    return best_a.tolist(), best_score
