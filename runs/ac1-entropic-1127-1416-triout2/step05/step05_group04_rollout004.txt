def propose_candidate(seed=123, budget_s=1000):
    start_time = time.time()
    deadline = start_time + budget_s - 0.5  # safety

    rng = np.random.default_rng(seed)

    # Pre-known fallback
    best_seq = []
    best_val = float('inf')
    if 'height_sequence_1' in globals():
        try:
            seq = [float(x) for x in globals()['height_sequence_1']]
            val = evaluate_sequence(seq)
            if val < best_val:
                best_seq, best_val = seq, val
        except:
            pass

    # Hyperparameters
    ns = [1024, 2048, 4096]  # maybe also 8192 if time
    top_k_total = 12
    best_pool = []  # list of (score, vector)

    def maybe_insert(score, vec):
        if len(best_pool) < top_k_total:
            best_pool.append((score, vec))
        else:
            # find worst
            worst_idx = max(range(len(best_pool)), key=lambda i: best_pool[i][0])
            if score < best_pool[worst_idx][0]:
                best_pool[worst_idx] = (score, vec)

    # exploration per n
    for n in ns:
        # deterministic seeds
        maybe_insert(compute_score(np.full(n, 1.0/n)), np.full(n, 1.0/n))
        maybe_insert(compute_score(arcsine_initial(n)), arcsine_initial(n))
        # loops sampling
        max_candidates_per_n = 25000  # adjust
        cnt = 0
        while time.time() < deadline - 5 and cnt < max_candidates_per_n:
            cnt += 1
            # choose shape family based on random
            r = rng.random()
            if r < 0.4:
                # Beta symmetric alpha=beta
                a = rng.uniform(0.05, 5.0)
                eps = rng.uniform(1e-6, 0.5)
                cand = generate_beta_shape(n, a, a, eps)
            elif r < 0.7:
                p = rng.uniform(0.05, 5.0)
                eps = rng.uniform(0.0, 1.0)
                cand = generate_power_shape(n, p, eps)
            elif r < 0.85:
                w = rng.random()
                cand = generate_mixture_shape(n, w)
            else:
                # triangular bump maybe shifted
                peak = rng.integers(0, n)
                cand = generate_triangular_shape(n, peak)
            sc = compute_score(cand)
            maybe_insert(sc, cand)

    # sort best seeds
    best_pool.sort(key=lambda x: x[0])
    seed_vectors = [vec for _, vec in best_pool[:top_k_total]]

    # refine each seed
    for idx, seed_vec in enumerate(seed_vectors, start=1):
        if time.time() >= deadline - 5:
            break
        n = len(seed_vec)
        # start Adam
        iter_est = int((deadline - time.time()) * 30)  # allocate 30 iter per sec approx
        if iter_est < 5000:
            iter_est = 5000
        refined, fast_sc = adam_optimize_smooth(seed_vec,
                                                deadline,
                                                max_iter=iter_est,
                                                lr0=0.2,
                                                lam_start=2.0,
                                                lam_end=200.0,
                                                noise_every=5000,
                                                noise_scale=5e-5)
        # subgradient
        if time.time() < deadline - 2:
            sub_iter = int((deadline - time.time()) * 25)
            if sub_iter < 5000:
                sub_iter = 5000
            refined = subgradient_descent_exact(refined,
                                                deadline,
                                                max_iter=sub_iter,
                                                step0=0.02)
        # final annealing perhaps
        if time.time() < deadline - 1:
            refined = simulated_annealing(refined,
                                          rng,
                                          deadline,
                                          max_iters=20000,
                                          init_temp=0.02,
                                          decay=0.99995)
        # evaluate official
        try:
            val = evaluate_sequence(refined.tolist())
        except Exception:
            val = float('inf')
        if val < best_val:
            best_val, best_seq = val, refined.tolist()
            print(f"New best after seed {idx}: {best_val:.6f}")

    # final polishing if time
    if best_seq and time.time() < deadline - 2:
        a = np.array(best_seq, dtype=float)
        # subgradient
        sub_iter = int((deadline - time.time()) * 20)
        if sub_iter > 0:
            a = subgradient_descent_exact(a, deadline, max_iter=sub_iter, step0=0.015)
        # Adam
        if time.time() < deadline - 1:
            adam_iter = int((deadline - time.time()) * 20)
            if adam_iter > 0:
                a, _ = adam_optimize_smooth(a,
                                            deadline,
                                            max_iter=adam_iter,
                                            lr0=0.1,
                                            lam_start=2.0,
                                            lam_end=150.0,
                                            noise_every=5000,
                                            noise_scale=1e-5)
        try:
            val = evaluate_sequence(a.tolist())
            if val < best_val:
                best_val, best_seq = val, a.tolist()
                print(f"Better after final polishing: {best_val:.6f}")
        except Exception:
            pass

    # guard
    if not best_seq:
        n = 2048
        best_seq = [1.0/n] * n
        best_val = evaluate_sequence(best_seq)

    print(f"Search finished. Best score {best_val:.6f}")
    return best_seq

"""
propose_candidate
=================

Search for a non‑negative step sequence that minimises the official
``evaluate_sequence`` objective.

The algorithm works in three phases:

1. **Broad random exploration** of several parametric families
   (arcsine, symmetric Beta, power‑law, exponential, mixtures, triangular
   bumps).  For each candidate the fast surrogate
   ``score = 2·n·max(conv(a,a))`` (with Σa = 1) is evaluated and the
   best *K* vectors are kept in a small heap.

2. **Local refinement** of every kept seed:
   * Adam optimisation on a soft‑max approximation of the max‑convolution,
   * exact sub‑gradient descent on the true objective,
   * a short simulated‑annealing polishing step.
   The budget is split dynamically according to the remaining time.

3. **Global polishing** of the overall best vector (if time permits).

All operations respect the simplex constraint (non‑negative entries,
sum = 1) by Euclidean projection.  The routine is fully self‑contained,
uses only NumPy and the standard library and respects the 1000 s time
limit.

The returned list can be handed directly to ``evaluate_sequence``.
"""

import time
import numpy as np
from typing import List, Tuple

# ----------------------------------------------------------------------
# Low‑level utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """
    Linear convolution via real FFT.
    Length = len(a) + len(b) - 1.
    """
    m = len(a) + len(b) - 1
    n_fft = next_pow2(m)
    A = np.fft.rfft(a, n=n_fft)
    B = np.fft.rfft(b, n=n_fft)
    C = A * B
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


def compute_score(a: np.ndarray) -> float:
    """
    Fast surrogate for the official score.
    ``a`` is assumed to be already normalised to Σa = 1,
    therefore the denominator disappears and the score is simply
    ``2·n·max(conv(a,a))``.
    """
    n = a.shape[0]
    conv = convolution_fft(a, a)
    return 2.0 * n * float(np.max(conv))


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of ``v`` onto the probability simplex
    {x ≥ 0, Σx = 1}.  Implements the algorithm of
    Duchi et al. (2008).
    """
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / v.size)

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho = np.where(u - cssv / np.arange(1, v.size + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Shape factories (all return a vector that already sums to 1)
# ----------------------------------------------------------------------


def arcsine_initial(n: int) -> np.ndarray:
    """Classic arcsine shape: a_i ∝ 1/√((i+½)*(n‑½‑i))."""
    i = np.arange(n, dtype=float)
    eps = 0.25
    vals = 1.0 / np.sqrt((i + 0.5) * (n - 0.5 - i) + eps)
    vals /= vals.sum()
    return vals


def generate_beta_shape(n: int, alpha: float, beta: float, eps: float) -> np.ndarray:
    """
    Symmetric (or not) Beta shape:
    (i+ε)^{α‑1}·(n‑1‑i+ε)^{β‑1}, normalised to Σ=1.
    """
    i = np.arange(n, dtype=float)
    left = (i + eps) ** (alpha - 1.0)
    right = (n - 1 - i + eps) ** (beta - 1.0)
    base = left * right
    total = base.sum()
    if not np.isfinite(total) or total <= 0.0:
        return np.full(n, 1.0 / n)
    return base / total


def generate_power_shape(n: int, p: float, eps: float) -> np.ndarray:
    """
    Generalised arcsine: a_i ∝ ((i+½)*(n‑½‑i)+ε)^{-p},   p>0.
    """
    i = np.arange(n, dtype=float)
    denom = ((i + 0.5) * (n - 0.5 - i) + eps) ** p
    vals = 1.0 / denom
    total = vals.sum()
    if not np.isfinite(total) or total <= 0.0:
        return np.full(n, 1.0 / n)
    return vals / total


def generate_exponential_shape(n: int, decay: float) -> np.ndarray:
    """Geometric decay: a_i ∝ exp(-decay * i)."""
    i = np.arange(n, dtype=float)
    vals = np.exp(-decay * i)
    total = vals.sum()
    if total <= 0.0:
        return np.full(n, 1.0 / n)
    return vals / total


def generate_triangular_shape(n: int, peak: int) -> np.ndarray:
    """Symmetric triangular bump with its apex at ``peak``."""
    left = np.arange(peak + 1, dtype=float)
    right = np.arange(n - peak - 1, dtype=float)[::-1]
    seq = np.concatenate([left, right])
    seq /= seq.sum()
    return seq


def generate_mixture_shape(n: int, weight_arc: float) -> np.ndarray:
    """Mixture (weight_arc)·arcsine + (1‑weight_arc)·uniform."""
    arc = arcsine_initial(n)
    uni = np.full(n, 1.0 / n)
    mix = weight_arc * arc + (1.0 - weight_arc) * uni
    mix /= mix.sum()
    return mix

def smooth_gradient(a: np.ndarray, lam: float) -> np.ndarray:
    """
    Gradient of the soft‑max approximation of ``max(conv(a,a))``.
    Returns ∇_a (2·n·softmax_max) where the simplex constraint is handled
    later by projection.
    """
    n = a.shape[0]
    conv = convolution_fft(a, a)                # length 2·n‑1
    max_c = np.max(conv)

    # stable soft‑max over the convolution values
    exp_term = np.exp(lam * (conv - max_c))
    w = exp_term / np.sum(exp_term)             # weights sum to 1

    # Weighted sub‑gradient: 2·∑_k w_k shift(a,k)
    w_rev = w[::-1]
    conv_rev = convolution_fft(a, w_rev)
    L = 2 * n - 1
    weighted_grad = 2.0 * conv_rev[L - 1 - np.arange(n)]

    return 2.0 * n * weighted_grad


def adam_optimize_smooth(initial_a: np.ndarray,
                         deadline: float,
                         max_iter: int = 100_000,
                         lr0: float = 0.12,
                         lam_start: float = 2.0,
                         lam_end: float = 200.0,
                         noise_every: int = 2000,
                         noise_scale: float = 5e-5) -> Tuple[np.ndarray, float]:
    """
    Adam optimisation on the smooth surrogate of the max‑convolution.
    Returns the best vector (projected on the simplex) and its fast score.
    """
    a = project_to_simplex(initial_a.astype(float).copy())
    best_a = a.copy()
    best_sc = compute_score(a)

    beta1, beta2 = 0.9, 0.999
    eps = 1e-8
    m = np.zeros_like(a)
    v = np.zeros_like(a)

    for it in range(1, max_iter + 1):
        if time.time() >= deadline:
            break

        # linearly increase the softmax temperature
        lam = lam_start + (lam_end - lam_start) * min(1.0, it / max_iter)

        grad = smooth_gradient(a, lam)

        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)

        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        step = lr0 * m_hat / (np.sqrt(v_hat) + eps)
        a_new = a - step

        if noise_every and (it % noise_every == 0):
            a_new += np.random.randn(a.size) * noise_scale

        a_new = project_to_simplex(a_new)

        cur_sc = compute_score(a_new)
        if cur_sc < best_sc:
            best_sc = cur_sc
            best_a = a_new.copy()

        a = a_new

        if it % 5000 == 0:
            print(f"    AdamSmooth it={it:,d} cur={cur_sc:.6f} best={best_sc:.6f}")

    return best_a, best_sc


def subgradient_descent_exact(initial_a: np.ndarray,
                              deadline: float,
                              max_iter: int = 80_000,
                              step0: float = 0.02) -> np.ndarray:
    """
    Sub‑gradient descent on the exact objective
    f(a) = 2·n·max(conv(a,a)).
    The step size follows η_t = step0 / √t.
    """
    a = project_to_simplex(initial_a.copy())
    best_a = a.copy()
    best_sc = compute_score(a)
    n = a.shape[0]
    L = 2 * n - 1

    for it in range(1, max_iter + 1):
        if time.time() >= deadline:
            break

        conv = convolution_fft(a, a)
        max_val = np.max(conv)

        # mask of positions attaining (or numerically close to) the maximum
        tol = max_val * 1e-12 + 1e-14
        mask = (np.abs(conv - max_val) <= tol).astype(float)

        mask_rev = mask[::-1]
        conv_rev = convolution_fft(a, mask_rev)
        weighted_grad = 2.0 * conv_rev[L - 1 - np.arange(n)]

        eta = step0 / np.sqrt(it)
        a = a - eta * weighted_grad
        a = project_to_simplex(a)

        cur_sc = compute_score(a)
        if cur_sc < best_sc:
            best_sc = cur_sc
            best_a = a.copy()

        if it % 5000 == 0:
            print(f"    SubGrad it={it:,d} cur={cur_sc:.6f} best={best_sc:.6f}")

    return best_a


def simulated_annealing(a: np.ndarray,
                        rng: np.random.Generator,
                        deadline: float,
                        max_iters: int = 120_000,
                        init_temp: float = 0.02,
                        decay: float = 0.9999) -> np.ndarray:
    """
    Simple Metropolis‑style local search.
    A small amount of mass is moved between two coordinates.
    The fast surrogate (2·n·max) is used for acceptance.
    """
    cur = a.copy()
    cur_sc = compute_score(cur)
    best = cur.copy()
    best_sc = cur_sc
    n = cur.size
    temp = init_temp
    it = 0

    while it < max_iters and time.time() < deadline:
        it += 1
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue

        # move up to 20 % of the mass at i
        delta = cur[i] * 0.2 * rng.random()
        if delta <= 1e-15 or cur[i] - delta < 0:
            continue

        cand = cur.copy()
        cand[i] -= delta
        cand[j] += delta

        cand_sc = compute_score(cand)

        if cand_sc < cur_sc or rng.random() < np.exp(-(cand_sc - cur_sc) / max(temp, 1e-12)):
            cur = cand
            cur_sc = cand_sc
            if cur_sc < best_sc:
                best = cur.copy()
                best_sc = cur_sc

        temp = max(temp * decay, 1e-8)

        if it % 5000 == 0:
            print(f"    SA it={it:,d} cur={cur_sc:.6f} best={best_sc:.6f}")

    return best

    """

    The function respects the 1000 s time budget, returns a non‑empty list,
    and prints progress information.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5      # safety margin

    best_seq: List[float] = []
    best_val: float = float("inf")

    # ------------------------------------------------------------------
    # 0.  Load fallback if a previous run produced a sequence
    # ------------------------------------------------------------------
    if "height_sequence_1" in globals():
        try:
            fallback = [float(x) for x in globals()["height_sequence_1"]]
            val = evaluate_sequence(fallback)
            print(f"Fallback (height_sequence_1) score = {val:.6f}")
            if val < best_val:
                best_val = val
                best_seq = fallback
        except Exception:
            pass

    # ------------------------------------------------------------------
    # 1.  Wide random exploration (top‑K heap)
    # ------------------------------------------------------------------
    # Lengths to try – 2048 gave good results previously,
    # we also probe 1024 and 4096.
    lengths = [1024, 2048, 4096]
    top_k = 12                                 # total number of seeds kept
    best_pool: List[Tuple[float, np.ndarray]] = []   # (score, vector)

    def maybe_insert(score: float, vec: np.ndarray) -> None:
        """Insert (score,vec) into best_pool keeping its size ≤ top_k."""
        if len(best_pool) < top_k:
            best_pool.append((score, vec))
        else:
            # locate worst entry
            worst_idx = max(range(len(best_pool)), key=lambda i: best_pool[i][0])
            if score < best_pool[worst_idx][0]:
                best_pool[worst_idx] = (score, vec)

    # deterministic starters (uniform + arcsine) for each length
    for n in lengths:
        uniform = np.full(n, 1.0 / n)
        maybe_insert(compute_score(uniform), uniform)

        arc = arcsine_initial(n)
        maybe_insert(compute_score(arc), arc)

    # systematic grid for a few shape families (few dozen candidates per length)
    for n in lengths:
        # symmetric Beta (α = β) on a log‑scale grid
        for a in np.geomspace(0.05, 5.0, num=12):
            cand = generate_beta_shape(n, a, a, eps=1e-6)
            maybe_insert(compute_score(cand), cand)

        # power‑law family
        for p in np.linspace(0.1, 5.0, num=12):
            cand = generate_power_shape(n, p, eps=0.0)
            maybe_insert(compute_score(cand), cand)

        # exponential decay
        for decay in np.linspace(0.0005, 0.02, num=8):
            cand = generate_exponential_shape(n, decay)
            maybe_insert(compute_score(cand), cand)

        # mixture of arcsine & uniform
        for w in np.linspace(0.0, 1.0, num=9):
            cand = generate_mixture_shape(n, w)
            maybe_insert(compute_score(cand), cand)

    # random sampling (high‑variance families) – stop ~30 s before deadline
    max_random = 120_000
    cnt = 0
    while time.time() < deadline - 30.0 and cnt < max_random:
        cnt += 1
        # pick a random length
        n = int(rng.choice(lengths))
        r = rng.random()
        if r < 0.30:                     # symmetric beta
            a = rng.uniform(0.02, 5.0)
            eps = rng.uniform(1e-6, 1.0)
            cand = generate_beta_shape(n, a, a, eps)
        elif r < 0.55:                   # power law
            p = rng.uniform(0.05, 5.0)
            eps = rng.uniform(0.0, 1.0)
            cand = generate_power_shape(n, p, eps)
        elif r < 0.70:                   # exponential
            decay = rng.uniform(0.0005, 0.03)
            cand = generate_exponential_shape(n, decay)
        elif r < 0.85:                   # mixture arcsine/uniform
            w = rng.random()
            cand = generate_mixture_shape(n, w)
        else:                            # triangular bump (random peak)
            peak = rng.integers(0, n)
            cand = generate_triangular_shape(n, peak)

        maybe_insert(compute_score(cand), cand)

    # ------------------------------------------------------------------
    # 2.  Refinement of each seed (ordered by current score)
    # ------------------------------------------------------------------
    best_pool.sort(key=lambda x: x[0])           # best first
    seed_vectors = [vec for _, vec in best_pool[:top_k]]

    for idx, seed_vec in enumerate(seed_vectors, start=1):
        if time.time() >= deadline - 5.0:
            break
        n = seed_vec.shape[0]
        print(f"\n--- Refining seed {idx}/{len(seed_vectors)} (n={n}) ---")

        # ---- Adam on smooth surrogate ----
        remaining = deadline - time.time()
        max_iter_adam = max(5_000, int(min(250_000, remaining * 35)))
        print(f"    Adam smooth: max_iter={max_iter_adam}")
        refined, _ = adam_optimize_smooth(
            initial_a=seed_vec,
            deadline=deadline,
            max_iter=max_iter_adam,
            lr0=0.10,
            lam_start=2.0,
            lam_end=200.0,
            noise_every=3000,
            noise_scale=3e-5,
        )

        # ---- Exact sub‑gradient descent ----
        if time.time() < deadline - 4.0:
            remaining = deadline - time.time()
            max_iter_sub = max(5_000, int(min(250_000, remaining * 30)))
            print(f"    Sub‑gradient descent: max_iter={max_iter_sub}")
            refined = subgradient_descent_exact(
                refined,
                deadline,
                max_iter=max_iter_sub,
                step0=0.018,
            )

        # ---- Simulated annealing polishing ----
        if time.time() < deadline - 2.0:
            remaining = deadline - time.time()
            max_iter_sa = max(5_000, int(min(150_000, remaining * 25)))
            print(f"    Simulated annealing: max_iter={max_iter_sa}")
            refined = simulated_annealing(
                refined,
                rng,
                deadline,
                max_iters=max_iter_sa,
                init_temp=0.015,
                decay=0.99995,
            )

        # ----- Official evaluation -----
        try:
            val = evaluate_sequence(refined.tolist())
            print(f"    Official score = {val:.6f}")
            if val < best_val:
                best_val = val
                best_seq = refined.tolist()
                print(f"    *** New global best: {best_val:.6f} ***")
        except Exception:
            pass

    # ------------------------------------------------------------------
    # 3.  Global polishing of the overall best (if time permits)
    # ------------------------------------------------------------------
    if best_seq and time.time() < deadline - 3.0:
        print("\n--- Final global polishing ---")
        a = np.array(best_seq, dtype=float)

        # sub‑gradient
        remaining = deadline - time.time()
        sub_iters = max(5_000, int(min(300_000, remaining * 25)))
        a = subgradient_descent_exact(a, deadline, max_iter=sub_iters, step0=0.015)

        # Adam on smooth surrogate
        if time.time() < deadline - 1.0:
            remaining = deadline - time.time()
            adam_iters = max(5_000, int(min(300_000, remaining * 25)))
            a, _ = adam_optimize_smooth(
                a,
                deadline,
                max_iter=adam_iters,
                lr0=0.08,
                lam_start=2.0,
                lam_end=150.0,
                noise_every=3000,
                noise_scale=1e-5,
            )

        try:
            val = evaluate_sequence(a.tolist())
            print(f"Final polished official score = {val:.6f}")
            if val < best_val:
                best_val = val
                best_seq = a.tolist()
                print("*** Updated global best after final polish ***")
        except Exception:
            pass

    # ------------------------------------------------------------------
    # 4.  Guard – ensure a non‑empty return value
    # ------------------------------------------------------------------
    if not best_seq:
        n = 2048
        best_seq = [1.0 / n] * n
        best_val = evaluate_sequence(best_seq)

    print(f"\nSearch finished. Best reported score = {best_val:.6f}")
    return best_seq
