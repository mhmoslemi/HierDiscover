"""
search_algorithm.py
~~~~~~~~~~~~~~~~~~~
A multi‑phase optimiser for the “max‑convolution” objective used by
`evaluate_sequence`.  The goal is to minimise

    2·n·max_i (a * a)[i]   with   a_i ≥ 0 ,   Σ a_i = 1

where * denotes the linear convolution.  The algorithm consists of:

1️⃣  Several random Dirichlet starts → keep the best.
2️⃣  A linear‑programming (LP) refinement based on the classic
    “alternating LP” idea (maximise Σ g_j under the constraint
    (a * g) ≤ max(a * a)).  After each LP the current distribution is
    mixed slightly with the LP solution; the max‑convolution never
    increases.
3️⃣  Exponentiated sub‑gradient descent (the method used in the
    reference implementation) – now with a larger time budget.
4️⃣  A short stochastic hill‑climber (swap / mass‑transfer moves).

All phases respect the 1000 s time budget of the evaluator.  The final
sequence is returned as a plain Python list of non‑negative floats.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# FFT‑based convolution – fast and exact for linear convolution
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Linear convolution a * b using a real FFT (fast)."""
    m = a.shape[0] + b.shape[0] - 1
    n_fft = _next_pow2(m)
    A = np.fft.rfft(a, n=n_fft)
    B = np.fft.rfft(b, n=n_fft)
    C = A * B
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


def _compute_score(a: np.ndarray) -> float:
    """
    Fast replica of `evaluate_sequence` when Σ a = 1.
    Returns 2·n·max(a*a) (exact for a normalised vector).
    """
    n = a.shape[0]
    conv = _convolution_fft(a, a)
    return 2.0 * n * float(np.max(conv))


# ----------------------------------------------------------------------
# Euclidean projection onto the probability simplex
# ----------------------------------------------------------------------


def _project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Project a vector onto the simplex {x ≥ 0, Σ x = 1}."""
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / v.shape[0])

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, v.shape[0] + 1)
    cond = u - cssv / ind > 0
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


# ----------------------------------------------------------------------
# Exponentiated sub‑gradient optimisation (same idea as the reference)
# ----------------------------------------------------------------------


def _exp_subgrad_optimize(
    init_a: np.ndarray,
    deadline: float,
    max_iter: int = 5_000_000,
    eta0: float = 0.5,
) -> tuple[np.ndarray, float]:
    """Exponentiated sub‑gradient descent on the exact objective."""
    a = init_a.astype(float).copy()
    best_a = a.copy()
    best_score = _compute_score(a)
    n = len(a)

    for t in range(1, max_iter + 1):
        if time.time() >= deadline:
            break

        conv = _convolution_fft(a, a)
        k = int(np.argmax(conv))

        # sub‑gradient for the maximal convolution index
        idx = np.arange(n)
        j = k - idx
        mask = (j >= 0) & (j < n)
        g = np.zeros_like(a)
        g[mask] = 2.0 * a[j[mask]]

        eta = eta0 / np.sqrt(t)
        a = a * np.exp(-eta * g)
        a = _project_to_simplex(a)

        cur_score = _compute_score(a)
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()

        # occasional tiny jitter to escape plateaus
        if t % 5000 == 0:
            a += np.random.randn(n) * (eta0 * 1e-4 / np.sqrt(t))
            a = _project_to_simplex(a)

    return best_a, best_score


# ----------------------------------------------------------------------
# Linear‑programming refinement (alternating LP from the paper)
# ----------------------------------------------------------------------


def _solve_lp_for_g(f: np.ndarray, M: float) -> np.ndarray:
    """
    Solve   maximise Σ g_j   subject to   f * g ≤ M   and   g ≥ 0 .
    Returns a normalised version (Σ g = 1).  If SciPy is unavailable,
    falls back to the current vector.
    """
    try:
        from scipy.optimize import linprog
    except Exception:                     # pragma: no cover
        # No SciPy – return the current vector unchanged.
        return f.copy()

    n = f.shape[0]
    m = 2 * n - 1

    # Build convolution matrix A where (A @ g)[k] = (f * g)[k].
    A = np.zeros((m, n))
    for j in range(n):
        A[j:j + n, j] = f

    # Upper bound for every convolution entry
    b_ub = np.full(m, M)

    # maximise Σ g  ⇒  minimise -Σ g
    c = -np.ones(n)

    bounds = [(0, None)] * n
    res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method="highs")
    if not res.success:
        # a safe fallback – keep the current distribution
        return f.copy()

    g0 = res.x
    s = np.sum(g0)
    if s <= 0.0:
        return f.copy()
    return g0 / s           # normalise to a probability vector


def _lp_refinement(
    start_a: np.ndarray,
    deadline: float,
    max_outer: int = 30,
) -> tuple[np.ndarray, float]:
    """
    Repeatedly tighten the bound M = max(a*a) using the LP described
    in the statement.  After each LP we mix the current a with the LP
    solution (step size 0.2, reduced if the bound would increase).
    """
    a = start_a.copy()
    best_score = _compute_score(a)
    best_a = a.copy()

    while time.time() < deadline and max_outer > 0:
        conv = _convolution_fft(a, a)
        M = float(np.max(conv))

        g = _solve_lp_for_g(a, M)          # feasible, Σg = 1
        step = 0.2
        improved = False

        while step > 1e-4:
            candidate = (1 - step) * a + step * g
            candidate = _project_to_simplex(candidate)
            cand_score = _compute_score(candidate)

            if cand_score <= best_score + 1e-12:
                a = candidate
                if cand_score < best_score:
                    best_score = cand_score
                    best_a = a.copy()
                improved = True
                break
            step *= 0.5

        if not improved:
            # no acceptable mixing step – stop early
            break

        max_outer -= 1

    return best_a, best_score


# ----------------------------------------------------------------------
# Simple stochastic hill‑climber (swap / mass‑transfer moves)
# ----------------------------------------------------------------------


def _hill_climb(seq, deadline, max_steps=200_000):
    """Local search that swaps entries or transfers a fraction of mass."""
    a = np.array(seq, dtype=float)
    n = a.shape[0]

    best_a = a.copy()
    best_score = _compute_score(best_a)

    step_scale = 0.1

    while max_steps > 0 and time.time() < deadline:
        move_type = np.random.rand()
        cand = best_a.copy()

        if move_type < 0.5:
            i, j = np.random.randint(n, size=2)
            cand[i], cand[j] = cand[j], cand[i]
        else:
            i, j = np.random.randint(n, size=2)
            if i != j:
                delta = step_scale * best_a[i]
                cand[i] -= delta
                cand[j] += delta
                cand[i] = max(cand[i], 0.0)

        cand = _project_to_simplex(cand)
        cur_score = _compute_score(cand)

        if cur_score < best_score:
            best_score = cur_score
            best_a = cand
            step_scale = min(step_scale * 1.05, 0.5)
        else:
            step_scale *= 0.97

        max_steps -= 1

    return best_a.tolist(), best_score

    # --------------------------------------------------------------
    if time.time() < deadline - 80.0:
        print("\nRunning exponentiated sub‑gradient optimisation")
        sub_seq, sub_val = _exp_subgrad_optimize(
            best_seq,
            time_deadline=deadline - 60.0,
            max_iter=5_000_000,
            eta0=0.6,
        )
        if sub_val < best_val:
            best_val = sub_val
            best_seq = sub_seq.copy()
            print(f"Sub‑gradient optimisation improved best to {best_val:.6f}")

    # --------------------------------------------------------------
    # Phase 4 – final stochastic hill‑climb polishing
    # --------------------------------------------------------------
    if time.time() < deadline - 5.0:
        print("\nFinal hill‑climb polishing")
        polished_seq, polished_val = _hill_climb(best_seq.tolist(), deadline, max_steps=250_000)
        if polished_val < best_val:
            best_val = polished_val
            best_seq = np.array(polished_seq, dtype=float)
            print(f"Hill‑climb improved best to {best_val:.6f}")

    # --------------------------------------------------------------
    # Final safety – ensure non‑negativity and normalisation
    # --------------------------------------------------------------
    final_a = np.maximum(best_seq, 0.0)
    final_a = _project_to_simplex(final_a)   # enforce Σ = 1 exactly

    final_score = evaluate_sequence(final_a.tolist())
    print(f"\nSearch finished. Final reported score = {final_score:.6f}")

    return final_a.tolist()
