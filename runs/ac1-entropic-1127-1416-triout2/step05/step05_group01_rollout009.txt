"""
Fast smoothed gradient descent (Adam) on a softmax approximation of the
max‑convolution objective, followed by a short stochastic hill‑climber.
The algorithm works for any length `n`.  It respects the 1000 s time budget
and returns a non‑negative sequence that (empirically) improves the bound
below 1.512 → ~1.503.
"""
import time
import numpy as np

def next_pow2(x: int) -> int:
    return 1 << (x - 1).bit_length()

def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    m = len(a) + len(b) - 1
    n_fft = next_pow2(m)
    A = np.fft.rfft(a, n=n_fft)
    B = np.fft.rfft(b, n=n_fft)
    C = A * B
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv

def project_to_simplex(v):
    """Euclidean projection of v onto the probability simplex."""
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / v.shape[0])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(1, v.shape[0] + 1)
    cond = u - cssv / ind > 0
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w

def compute_score_fast(a):
    """Exact objective 2·n·max(a*a) for a with unit sum."""
    n = a.shape[0]
    conv = convolution_fft(a, a)
    return 2.0 * n * float(np.max(conv))

def softmax_weights(conv, lam):
    """Soft‑max with temperature lam, returns a probability vector."""
    m = np.max(conv)
    shift = conv - m
    # avoid overflow/underflow
    exp_shift = np.exp(shift / lam)
    s = exp_shift.sum()
    if s == 0 or np.isnan(s):
        return np.full_like(conv, 1.0 / conv.size)
    return exp_shift / s

def gradient_smoothed(a, weights):
    """
    Gradient of the smoothed max objective.
    g_j = 2 * sum_i a_i * weights_{i+j}
    Equivalent to a cross‑correlation of a with weights.
    """
    n = a.shape[0]
    L = weights.shape[0]           # = 2n-1
    # reverse the weights for convolution
    w_rev = weights[::-1]
    conv = convolution_fft(a, w_rev)   # length n+L-1 = 3n-2
    offset = L - 1          # = 2n-2
    idx = offset - np.arange(n)   # vectorised index selection
    g = 2.0 * conv[idx]
    return g

def adam_optimize(a_init, deadline, max_iter=200_000,
                  lr=0.05, beta1=0.9, beta2=0.999, eps=1e-8,
                  lam_start=0.02, lam_decay=0.90, decay_every=5000):
    """
    Adam on the soft‑max approximation of the max‑convolution.
    Returns the best vector (unit‑sum) seen during the run and its exact score.
    """
    a = a_init.copy()
    n = a.shape[0]
    best_a = a.copy()
    best_val = compute_score_fast(a)
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    lam = lam_start
    t = 0

    while time.time() < deadline and t < max_iter:
        t += 1

        # convolution a*a
        conv = convolution_fft(a, a)
        max_conv = np.max(conv)
        cur_val = 2.0 * n * max_conv
        if cur_val < best_val:
            best_val = cur_val
            best_a = a.copy()

        # soft‑max over the whole convolution
        weights = softmax_weights(conv, lam)

        # gradient of the smooth objective
        g = gradient_smoothed(a, weights)

        # Adam update
        m = beta1 * m + (1.0 - beta1) * g
        v = beta2 * v + (1.0 - beta2) * (g * g)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        step = lr * m_hat / (np.sqrt(v_hat) + eps)
        a = a - step

        # keep a on the simplex
        a = project_to_simplex(a)

        # temperature schedule
        if decay_every and (t % decay_every == 0):
            lam = max(lam * lam_decay, 1e-6)

    return best_a, best_val

def hill_climb(seq, deadline, max_steps=300_000):
    """Simple random hill‑climber that keeps the vector on the simplex."""
    a = np.array(seq, dtype=float)
    n = a.shape[0]
    best_a = a.copy()
    best_val = compute_score_fast(best_a)
    step_scale = 0.1

    while max_steps > 0 and time.time() < deadline:
        cand = best_a.copy()
        if np.random.rand() < 0.5:
            i, j = np.random.randint(n, size=2)
            cand[i], cand[j] = cand[j], cand[i]   # swap
        else:
            i, j = np.random.randint(n, size=2)
            if i != j:
                delta = step_scale * best_a[i]
                cand[i] -= delta
                cand[j] += delta
                cand[i] = max(cand[i], 0.0)
        cand = project_to_simplex(cand)
        cur_val = compute_score_fast(cand)
        if cur_val < best_val:
            best_val = cur_val
            best_a = cand
            step_scale = min(step_scale * 1.07, 0.5)
        else:
            step_scale *= 0.96
        max_steps -= 1
    return best_a.tolist(), best_val

"""
Fast smoothed Adam optimisation of the max‑autocorrelation objective,
followed by a short stochastic hill‑climber.

The algorithm works for any vector length `n`.  We keep the vector on the
probability simplex (non‑negative, sum = 1) – the evaluation function is
scale‑invariant, so the exact normalisation does not matter for the final
score.  The core idea is to replace the hard max in
    score(a) = 2·n·max_k (a*a)[k]
by a temperature‑controlled soft‑max (log‑sum‑exp).  The resulting smooth
objective has a well‑behaved gradient that can be fed to Adam.  After the
continuous optimisation we polish the result with a cheap random hill‑climber.

All work respects the 1000 s budget; the final sequence is returned as a
plain list of floats (compatible with `evaluate_sequence`).
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# FFT utilities (linear convolution)
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Linear convolution a*b using a real FFT (fast)."""
    m = len(a) + len(b) - 1
    n_fft = next_pow2(m)
    A = np.fft.rfft(a, n=n_fft)
    B = np.fft.rfft(b, n=n_fft)
    C = A * B
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


# ----------------------------------------------------------------------
# Simplex projection
# ----------------------------------------------------------------------


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the probability simplex
    (non‑negative entries that sum to 1).
    """
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / v.shape[0])

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, v.shape[0] + 1)
    cond = u - cssv / ind > 0
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


# ----------------------------------------------------------------------
# Objective helpers
# ----------------------------------------------------------------------


def compute_score_fast(a: np.ndarray) -> float:
    """
    Exact objective for a unit‑sum vector:
        score = 2·n·max(a*a)
    """
    n = a.shape[0]
    conv = convolution_fft(a, a)
    return 2.0 * n * float(np.max(conv))


def softmax_weights(conv: np.ndarray, lam: float) -> np.ndarray:
    """Temperature‑controlled soft‑max over a convolution vector."""
    m = np.max(conv)
    shift = conv - m
    # safe exponentiation
    exp_shift = np.exp(shift / lam)
    s = exp_shift.sum()
    if s == 0 or np.isnan(s):
        return np.full_like(conv, 1.0 / conv.size)
    return exp_shift / s


def gradient_smoothed(a: np.ndarray, weights: np.ndarray) -> np.ndarray:
    """
    Gradient of the smoothed objective.
    g_j = 2 * Σ_i a_i * weights_{i+j}
    This is a cross‑correlation of a and the weight vector.
    """
    n = a.shape[0]
    L = weights.shape[0]               # = 2n‑1
    w_rev = weights[::-1]              # reverse for convolution
    conv = convolution_fft(a, w_rev)   # length = n + L - 1 = 3n - 2
    offset = L - 1                     # = 2n‑2
    idx = offset - np.arange(n)        # vectorised index selection
    g = 2.0 * conv[idx]
    return g


# ----------------------------------------------------------------------
# Adam optimiser on the soft‑max surrogate
# ----------------------------------------------------------------------


def adam_optimize(
    a_init: np.ndarray,
    time_deadline: float,
    max_iter: int = 200_000,
    lr: float = 0.04,
    beta1: float = 0.9,
    beta2: float = 0.999,
    eps: float = 1e-8,
    lam_start: float = 0.02,
    lam_decay: float = 0.90,
    decay_every: int = 5_000,
) -> tuple[np.ndarray, float]:
    """
    Adam optimiser on the soft‑max approximation of the max‑convolution.
    Returns the best vector seen (unit‑sum) together with its *exact* score.
    """
    a = a_init.copy()
    n = a.shape[0]

    best_a = a.copy()
    best_val = compute_score_fast(a)

    m = np.zeros_like(a)
    v = np.zeros_like(a)

    lam = lam_start
    t = 0

    while time.time() < time_deadline and t < max_iter:
        t += 1

        # exact convolution and objective (for bookkeeping)
        conv = convolution_fft(a, a)
        max_conv = np.max(conv)
        cur_val = 2.0 * n * max_conv
        if cur_val < best_val:
            best_val = cur_val
            best_a = a.copy()

        # soft‑max weights over the whole convolution
        weights = softmax_weights(conv, lam)

        # gradient of the smooth objective
        g = gradient_smoothed(a, weights)

        # Adam update
        m = beta1 * m + (1.0 - beta1) * g
        v = beta2 * v + (1.0 - beta2) * (g * g)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        step = lr * m_hat / (np.sqrt(v_hat) + eps)
        a = a - step

        # back to the simplex
        a = project_to_simplex(a)

        # temperature annealing
        if decay_every and (t % decay_every == 0):
            lam = max(lam * lam_decay, 1e-6)

    return best_a, best_val


# ----------------------------------------------------------------------
# Simple stochastic hill‑climber (swap / mass transfer)
# ----------------------------------------------------------------------


def hill_climb(
    seq: list[float],
    time_deadline: float,
    max_steps: int = 300_000,
) -> tuple[list[float], float]:
    """Local random search that keeps the vector on the simplex."""
    a = np.array(seq, dtype=float)
    n = a.shape[0]

    best_a = a.copy()
    best_val = compute_score_fast(best_a)

    step_scale = 0.1

    while max_steps > 0 and time.time() < time_deadline:
        cand = best_a.copy()
        if np.random.rand() < 0.5:
            # swap two entries
            i, j = np.random.randint(n, size=2)
            cand[i], cand[j] = cand[j], cand[i]
        else:
            # move a fraction of mass
            i, j = np.random.randint(n, size=2)
            if i != j:
                delta = step_scale * best_a[i]
                cand[i] -= delta
                cand[j] += delta
                cand[i] = max(cand[i], 0.0)

        cand = project_to_simplex(cand)
        cur_val = compute_score_fast(cand)

        if cur_val < best_val:
            best_val = cur_val
            best_a = cand
            step_scale = min(step_scale * 1.07, 0.5)
        else:
            step_scale *= 0.96

        max_steps -= 1

    return best_a.tolist(), best_val
