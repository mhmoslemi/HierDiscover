```
def simulated_annealing(a, time_deadline):
    cur = a.copy()
    cur_sc = compute_score(cur)
    best = cur.copy()
    best_sc = cur_sc
    n = len(a)

    # final temp small
    while time.time() < time_deadline:
        # compute fraction of time left
```

```
    # fallback attempt from height_sequence_1

         val = evaluate_sequence(fallback)

    except Exception: pass

    # choose length(s)
    lengths = [8192, 4096, 2048]   # try largest first
    for n in lengths:
        if time.time() >= deadline - 10: break
        print(f"\n=== Optimizing length {n} ===")
        # Stage A: grid search
        alphas = np.linspace(0.15, 1.5, 30)
        eps_vals = np.geomspace(1e-8, 0.5, 12)

            for eps in eps_vals:
                a = generate_beta_shape(n, alpha, alpha, eps)

                if sc < local_best_sc:

                    local_best = a.copy()

            if time.time() >= deadline - 8: break

            weights = rng.random(k)
            weights /= weights.sum()
            comp = np.zeros(n)
            for i in range(k):
                alpha = rng.uniform(0.05, 2.0)

                comp += weights[i] * generate_beta_shape(n, alpha, alpha, eps)
            cand = comp / comp.sum()
            sc = compute_score(cand)
            if sc < local_best_sc:
                local_best_sc = sc
                local_best = cand.copy()
        # Stage C: if we have candidate, refine via simulated annealing + subgradient
        if local_best is not None and time.time() < deadline - 5:
            # Simulated annealing
```

assistantfinal**Solution Overview**

The goal is to minimise  

\[
\text{score}(a)=\frac{2\,n\;\max (a*a)}{\bigl(\sum a\bigr)^2},
\qquad a_i\ge 0 .
\]

Because the score is homogeneous we can **normalise** the vector to a
probability distribution \(\sum a_i = 1\); the denominator becomes a
constant and minimising the score is exactly the same as minimising the
maximum entry of the self‑convolution.

The search strategy is a four‑phase pipeline that fits into the 1000 s
budget:

1. **Exploratory parametrised sampling** – a dense grid of symmetric
   Beta‑shapes (including the classic arcsine, sine, Hann and other
   window functions) and many random mixtures of up to three Beta
   components.  
   The cheap surrogate  
   `2·n·max(conv(a,a))` (implemented by `compute_score`) is used to keep
   the best few candidates.

2. **Simulated annealing on the simplex** – starting from the best
   candidate we run a temperature‑controlled Gaussian walk, always
   re‑projecting onto the probability simplex.  The temperature is
   decreased linearly, allowing occasional uphill steps early on and
   converging to a pure descent later.

3. **Exact sub‑gradient descent** – a short, back‑tracking line‑search
   optimisation on the *true* maximum convolution (not the smooth
   surrogate).  This sharpens the solution obtained by the annealer.

4. **Global polishing** – a final short annealing round on the overall
   best vector.

All phases stop as soon as the global deadline is reached, guaranteeing
a valid (non‑empty) return.  The algorithm works for any vector length;
larger lengths (up to 8192) give a tighter theoretical bound
(\(2n\max\le 2n/(2n-1)\to 1\)) while still being cheap thanks to the FFT
based convolution.

```python
"""
search for a non‑negative step sequence that minimises
    2·n·max(conv(a,a)) / (sum a)²
using a parametric beta‑family sampler, simulated annealing on the simplex,
and a final exact sub‑gradient descent.  Larger vectors (up to 8192) are
tried first – the bound improves with n.
"""

import time
import numpy as np
from typing import List

# ----------------------------------------------------------------------
# Core linear‑algebra utilities (FFT convolution)
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """
    Linear convolution using a real FFT.
    Returned length is len(a)+len(b)-1.
    """
    m = a.size + b.size - 1
    n_fft = next_pow2(m)
    A = np.fft.rfft(a, n=n_fft)
    B = np.fft.rfft(b, n=n_fft)
    C = A * B
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


def max_convolution(a: np.ndarray) -> float:
    """Maximum entry of the self‑convolution of a."""
    return float(np.max(convolution_fft(a, a)))


# ----------------------------------------------------------------------
# Simplex projection (Euclidean)
# ----------------------------------------------------------------------


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of v onto the probability simplex
    {x ≥ 0, Σ x = 1}.  Implementation of Duchi et al. (2008).
    """
    n = v.shape[0]
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / n)

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho = np.where(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Fast surrogate (used during the search)
# ----------------------------------------------------------------------


def compute_score(a: np.ndarray) -> float:
    """
    Fast surrogate: 2·n·max(conv(a,a)).
    a is assumed to be normalised (Σ a = 1), therefore this equals the
    official evaluation value.
    """
    n = a.shape[0]
    return 2.0 * n * max_convolution(a)


# ----------------------------------------------------------------------
# Shape generators
# ----------------------------------------------------------------------


def generate_beta_shape(n: int, alpha: float, beta: float, eps: float) -> np.ndarray:
    """Symmetric Beta‑family density on {0,…,n‑1} and normalised."""
    i = np.arange(n, dtype=float)
    left = (i + eps) ** (alpha - 1.0)
    right = (n - 1 - i + eps) ** (beta - 1.0)
    base = left * right
    total = base.sum()
    if total <= 0.0 or not np.isfinite(total):
        return np.full(n, 1.0 / n)
    return base / total


def generate_beta_mixture(
    n: int,
    alphas: List[float],
    betas: List[float],
    epsilons: List[float],
    weights: List[float],
) -> np.ndarray:
    """
    Convex combination of several Beta‑shapes.
    The result is re‑normalised to Σ = 1.
    """
    mix = np.zeros(n, dtype=float)
    for a, b, e, w in zip(alphas, betas, epsilons, weights):
        mix += w * generate_beta_shape(n, a, b, e)
    mix_sum = mix.sum()
    if mix_sum == 0.0 or not np.isfinite(mix_sum):
        return np.full(n, 1.0 / n)
    return mix / mix_sum


def arcsine_initial(n: int) -> np.ndarray:
    """Classic arcsine (Beta(½,½)) shape."""
    i = np.arange(n, dtype=float)
    eps = 0.25
    vals = 1.0 / np.sqrt((i + 0.5) * (n - 0.5 - i) + eps)
    return vals / vals.sum()


def sine_initial(n: int) -> np.ndarray:
    """Positive half‑sine window."""
    i = np.arange(n, dtype=float)
    vals = np.sin(np.pi * (i + 0.5) / n)
    return vals / vals.sum()


def hann_initial(n: int) -> np.ndarray:
    """Hann (raised cosine) window – zero at both ends."""
    i = np.arange(n, dtype=float)
    vals = 0.5 * (1.0 - np.cos(2.0 * np.pi * i / (n - 1)))
    return vals / vals.sum()


# ----------------------------------------------------------------------
# Exact sub‑gradient descent (with back‑tracking line‑search)
# ----------------------------------------------------------------------


def subgradient_optimize(
    initial_a: np.ndarray,
    time_deadline: float,
    step0: float = 0.3,
    max_iter: int = 200_000,
) -> np.ndarray:
    """
    Projected sub‑gradient descent on the true
    max‑convolution.  Returns the best vector seen.
    """
    a = initial_a.copy()
    best_a = a.copy()
    best_sc = compute_score(a)

    n = a.shape[0]

    for it in range(1, max_iter + 1):
        if time.time() >= time_deadline:
            break

        conv = convolution_fft(a, a)
        max_val = float(np.max(conv))
        idx = int(np.argmax(conv))

        # sub‑gradient of the active quadratic term
        i = np.arange(n)
        j = idx - i
        mask = (j >= 0) & (j < n)
        grad = np.zeros_like(a)
        grad[mask] = 2.0 * a[j[mask]]

        # step size with a simple 1/√t decay
        step = step0 / np.sqrt(it)

        # back‑tracking line‑search
        for _ in range(12):
            a_new = a - step * grad
            a_new = project_to_simplex(a_new)
            if max_convolution(a_new) < max_val - 1e-14:
                break
            step *= 0.5

        cur_sc = compute_score(a_new)
        if cur_sc < best_sc:
            best_sc = cur_sc
            best_a = a_new.copy()

        a = a_new

    return best_a


# ----------------------------------------------------------------------
# Simulated annealing on the simplex (fast surrogate)
# ----------------------------------------------------------------------


def simulated_annealing(
    start_a: np.ndarray,
    time_deadline: float,
    init_temp: float = None,
) -> np.ndarray:
    """
    Temperature‑controlled Gaussian walk on the simplex.
    Uses the fast surrogate `compute_score`.  Returns the best vector found.
    """
    rng = np.random.default_rng()
    a_cur = start_a.copy()
    cur_sc = compute_score(a_cur)

    best_a = a_cur.copy()
    best_sc = cur_sc

    n = a_cur.shape[0]
    start_time = time.time()
    total_budget = time_deadline - start_time
    T0 = init_temp if init_temp is not None else cur_sc

    while time.time() < time_deadline:
        # linear cooling: temperature proportional to remaining time
        remaining = time_deadline - time.time()
        frac = remaining / total_budget
        T = T0 * frac

        # step size also shrinks with the temperature
        step_scale = 0.03 * (0.5 + frac)   # ~0.045 → 0.015
        noise = rng.normal(scale=step_scale, size=n)
        a_prop = a_cur + noise
        a_prop = project_to_simplex(a_prop)

        prop_sc = compute_score(a_prop)
        delta = prop_sc - cur_sc
        if delta < 0 or rng.random() < np.exp(-delta / max(T, 1e-12)):
            a_cur = a_prop
            cur_sc = prop_sc
            if cur_sc < best_sc:
                best_sc = cur_sc
                best_a = a_cur.copy()

    return best_a


# ----------------------------------------------------------------------
# Main entry point required by the evaluation harness
# ----------------------------------------------------------------------


def propose_candidate(
    seed: int = 42,
    budget_s: int = 1000,
    **kwargs,
) -> List[float]:
    """
    Search for a non‑negative step sequence with a low official score.
    The routine
      1. tries a few deterministic window shapes,
      2. performs a dense grid over symmetric Beta shapes,
      3. explores random mixtures of up to three Beta components,
      4. refines the current best with simulated annealing,
      5. finishes with a short exact sub‑gradient descent.
    The longest feasible vector length (up to 8192) is used first,
    because the theoretical lower bound improves with n.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 2.0   # safety margin

    best_seq: List[float] = []
    best_val: float = float("inf")

    # ------------------------------------------------------------------
    # 0.  Fallback – the previously known construction if it exists
    # ------------------------------------------------------------------
    try:
        fallback = [float(x) for x in height_sequence_1]  # type: ignore
        val = evaluate_sequence(fallback)
        print(f"Fallback (height_sequence_1) official score = {val:.6f}")
        if val < best_val:
            best_val, best_seq = val, fallback
    except Exception:
        pass

    # ------------------------------------------------------------------
    # 1.  Lengths – try the longest first
    # ------------------------------------------------------------------
    lengths = [8192, 4096, 2048, 1024]

    for n in lengths:
        if time.time() >= deadline - 10.0:
            break
        print(f"\n=== Optimising length n = {n} ===")

        # ------------------------------------------------------------------
        # Deterministic starters (window functions)
        # ------------------------------------------------------------------
        starters = [
            np.full(n, 1.0 / n),     # uniform
            arcsine_initial(n),
            sine_initial(n),
            hann_initial(n),
        ]
        local_best = None
        local_best_sc = float("inf")
        for st in starters:
            sc = compute_score(st)
            if sc < local_best_sc:
                local_best_sc, local_best = sc, st.copy()

        # ------------------------------------------------------------------
        # 1a. Grid over symmetric Beta shapes (alpha = beta)
        # ------------------------------------------------------------------
        alphas = np.linspace(0.15, 1.5, 30)          # exponent range
        eps_vals = np.geomspace(1e-8, 0.5, 12)      # tiny to moderate smoothing
        for alpha in alphas:
            for eps in eps_vals:
                cand = generate_beta_shape(n, alpha, alpha, eps)
                sc = compute_score(cand)
                if sc < local_best_sc:
                    local_best_sc, local_best = sc, cand.copy()

        # ------------------------------------------------------------------
        # 1b. Random mixtures of up to three Beta components
        # ------------------------------------------------------------------
        max_samples = int(5000 * (budget_s / 1000.0))
        for _ in range(max_samples):
            if time.time() >= deadline - 6.0:
                break
            k = rng.integers(2, 4)                 # 2 or 3 components
            weights = rng.random(k)
            weights /= weights.sum()
            alphas_mix = rng.uniform(0.05, 2.0, size=k)
            eps_mix = rng.uniform(1e-8, 0.5, size=k)
            cand = generate_beta_mixture(
                n,
                alphas=alphas_mix.tolist(),
                betas=alphas_mix.tolist(),
                epsilons=eps_mix.tolist(),
                weights=weights.tolist(),
            )
            sc = compute_score(cand)
            if sc < local_best_sc:
                local_best_sc, local_best = sc, cand.copy()

        # ------------------------------------------------------------------
        # 2.  Simulated annealing starting from the locally best vector
        # ------------------------------------------------------------------
        if local_best is not None and time.time() < deadline - 4.0:
            print(f"  Annealing phase (len={n}) …")
            a_anneal = simulated_annealing(
                start_a=local_best,
                time_deadline=deadline,
            )
            # ------------------------------------------------------------------
            # 3.  Short sub‑gradient polishing (exact max)
            # ------------------------------------------------------------------
            if time.time() < deadline - 2.0:
                remaining = deadline - time.time()
                max_iter_sub = int(min(80_000, remaining * 30))
                a_refined = subgradient_optimize(
                    a_anneal,
                    time_deadline=deadline,
                    step0=0.30,
                    max_iter=max_iter_sub,
                )
            else:
                a_refined = a_anneal

            # ------------------------------------------------------------------
            # 4.  Official evaluation
            # ------------------------------------------------------------------
            val = evaluate_sequence(a_refined.tolist())
            print(f"  → official score for length {n} = {val:.6f}")
            if val < best_val:
                best_val, best_seq = val, a_refined.tolist()
                print(f"  *** New global best (len={n}) = {best_val:.6f} ***")
        else:
            # In the unlikely case we never entered the annealing phase,
            # evaluate the raw best candidate.
            if local_best is not None:
                val = evaluate_sequence(local_best.tolist())
                if val < best_val:
                    best_val, best_seq = val, local_best.tolist()
                    print("*** Updated global best (raw) ***")

    # ------------------------------------------------------------------
    # 5.  Global polishing of the overall best vector (if time permits)
    # ------------------------------------------------------------------
    if best_seq and time.time() < deadline - 2.0:
        print("\n=== Global polishing of the current best sequence ===")
        a = np.array(best_seq, dtype=float)

        # a short annealing run
        a = simulated_annealing(a, time_deadline=deadline)

        # final sub‑gradient descent (few iterations)
        if time.time() < deadline - 1.0:
            sub_iters = int(min(40_000, (deadline - time.time()) * 30))
            a = subgradient_optimize(a, time_deadline=deadline,
                                     step0=0.25, max_iter=sub_iters)

        final_val = evaluate_sequence(a.tolist())
        print(f"Final polished official score = {final_val:.6f}")
        if final_val < best_val:
            best_val, best_seq = final_val, a.tolist()
            print("*** Updated global best after final polish ***")

    # ------------------------------------------------------------------
    # 6.  Guard – guarantee a non‑empty return
    # ------------------------------------------------------------------
    if not best_seq:
        best_seq = [1.0] * 256
        best_val = evaluate_sequence(best_seq)
        print("Fallback to uniform 256‑vector.", best_val)

    print(f"\nSearch finished. Best reported official score = {best_val:.6f}")
    return best_seq
```
