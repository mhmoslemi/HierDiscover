"""
Improved step‑function search.

The algorithm works in several phases:

1.  **Seed generation** – start from the provided fallback (if any),
    otherwise from a uniform distribution.  A few random Dirichlet
    vectors are tried to get a better initial point.

2.  **Alternating LP optimisation** – for the current sequence `a`
    compute `M = max(a⊛a)`.  Solve the linear programme

           maximise   Σ b_j
           subject to   (a⊛b)[k] ≤ M   (k = 0,…,2n‑2)
                        b ≥ 0

    using SciPy’s high‑performance `linprog`.  Mix the solution `b`
    into `a` with a diminishing step size and project back onto the
    probability simplex.  Repeating this monotone step shrinks the
    peak `M`.

3.  **Smooth soft‑max descent** – minimise a differentiable surrogate
    for the peak, namely the log‑sum‑exp of the convolution,
    by projected gradient descent.  The gradient can be computed
    efficiently as a “valid” convolution of the soft‑max weights with
    the current vector.

4.  **Sub‑gradient refinement** – a classic sub‑gradient descent on
    the exact maximum (as in the previous state‑of‑the‑art).

5.  **Hill‑climb polishing** – stochastic swaps / mass‑transfer
    moves accepted only when they improve the true `evaluate_sequence`
    score.

All phases respect the 1000 s budget; progress and intermediate scores
are printed.  The returned object is a plain list of non‑negative
floats (sum ≈ 1) ready for `evaluate_sequence`.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Linear‑algebra utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full convolution a⊛b using a real FFT (fast)."""
    m = len(a) + len(b) - 1
    n_fft = next_pow2(m)
    A = np.fft.rfft(a, n=n_fft)
    B = np.fft.rfft(b, n=n_fft)
    C = A * B
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


def compute_score(a: np.ndarray) -> float:
    """
    Fast replica of `evaluate_sequence` when Σa = 1.
    Returns 2·n·max(a⊛a).
    """
    n = a.shape[0]
    conv = convolution_fft(a, a)
    return 2.0 * n * float(np.max(conv))


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the probability simplex:
    non‑negative entries that sum to 1.
    """
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / v.shape[0])

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, v.shape[0] + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.where(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


# ----------------------------------------------------------------------
# Linear programming (SciPy) – a ⊛ b ≤ M
# ----------------------------------------------------------------------
try:
    import scipy.optimize as opt
    _have_scipy = True
except Exception:               # pragma: no cover
    _have_scipy = False


def solve_lp_for_b_scipy(a: np.ndarray, M: float):
    """
    Solve   maximise   Σ b_j
    subject to b ≥ 0,
               (a ⊛ b)[k] ≤ M   for every shift k.
    Returns the optimal b (may have sum < 1) or None on failure.
    """
    if not _have_scipy:
        return None

    n = a.shape[0]
    L = 2 * n - 1                     # length of the full convolution

    # Build the constraint matrix A (shape L×n) where
    #   A[k, j] = a[k‑j]  if 0 ≤ k‑j < n, else 0.
    A = np.zeros((L, n), dtype=float)
    for j in range(n):
        A[j:j + n, j] = a               # slice assignment is fast

    c = -np.ones(n)                     # maximise Σb ⇔ minimise −Σb
    bounds = [(0, None)] * n
    b_ub = np.full(L, M, dtype=float)

    try:
        res = opt.linprog(c, A_ub=A, b_ub=b_ub,
                          bounds=bounds, method='highs',
                          options={'disp': False})
    except Exception as e:               # pragma: no cover
        print(f"Linprog exception: {e}")
        return None

    if res.success and res.x is not None:
        return res.x
    # On failure we simply give up on the LP step.
    return None


# ----------------------------------------------------------------------
# Smooth soft‑max gradient descent (convex surrogate of max)
# ----------------------------------------------------------------------


def smooth_optimize(a_init: np.ndarray,
                    time_deadline: float,
                    max_iter: int = 300_000,
                    step0: float = 0.05,
                    alpha_start: float = 1.0,
                    alpha_end: float = 50.0) -> tuple[np.ndarray, float]:
    """
    Projected gradient descent on   φ(a) = (1/α)·log Σ exp(α·(a⊛a)).
    The gradient is 2·(w ⊛ a) where w are the soft‑max weights.
    """
    a = a_init.astype(float).copy()
    best_a = a.copy()
    best_score = compute_score(a)
    n = len(a)

    for it in range(1, max_iter + 1):
        if time.time() >= time_deadline:
            break

        conv = convolution_fft(a, a)

        # increase α linearly so that the surrogate approaches the true max
        alpha = alpha_start + (alpha_end - alpha_start) * min(1.0, it / max_iter)

        # soft‑max weights (stable implementation)
        tmp = alpha * conv
        shift = np.max(tmp)
        exp_vals = np.exp(tmp - shift)
        w = exp_vals / np.sum(exp_vals)          # length 2n‑1, Σw = 1

        # gradient = 2·(valid convolution of w with a)
        g = 2.0 * np.convolve(w, a, mode='valid')   # shape n

        # step size with 1/√t decay
        step = step0 / np.sqrt(it)
        a_new = a - step * g
        a_new = project_to_simplex(a_new)

        cur_score = compute_score(a_new)
        if cur_score < best_score:
            best_score = cur_score
            best_a = a_new.copy()

        # occasional jitter to escape shallow plateaus
        if it % 1000 == 0:
            a_new += np.random.randn(n) * (step0 * 0.005 / np.sqrt(it))
            a_new = project_to_simplex(a_new)

        a = a_new

    return best_a, best_score


# ----------------------------------------------------------------------
# Classic sub‑gradient descent on the exact max‑convolution
# ----------------------------------------------------------------------


def subgrad_optimize(init_a: np.ndarray,
                     time_deadline: float,
                     max_iter: int = 200_000,
                     step0: float = 0.07) -> tuple[np.ndarray, float]:
    """
    Projected sub‑gradient descent on 2·n·max(a⊛a).
    The sub‑gradient at the current maximal shift k is
        g[i] = 2·a[k‑i]  (if index valid) .
    """
    a = init_a.astype(float).copy()
    best_a = a.copy()
    best_score = compute_score(a)
    n = len(a)

    for t in range(1, max_iter + 1):
        if time.time() >= time_deadline:
            break

        conv = convolution_fft(a, a)
        k = int(np.argmax(conv))

        # build sub‑gradient for the active shift k
        idx = np.arange(n)
        j = k - idx
        mask = (j >= 0) & (j < n)
        g = np.zeros_like(a)
        g[mask] = 2.0 * a[j[mask]]

        step = step0 / np.sqrt(t)
        a_new = a - step * g
        a_new = project_to_simplex(a_new)

        cur_score = compute_score(a_new)
        if cur_score < best_score:
            best_score = cur_score
            best_a = a_new.copy()

        a = a_new

        # small random perturbation every thousand steps
        if t % 1000 == 0:
            a += np.random.randn(n) * (step0 * 0.005 / np.sqrt(t))
            a = project_to_simplex(a)

    return best_a, best_score


# ----------------------------------------------------------------------
# Hill‑climb polishing (swap / mass transfer)
# ----------------------------------------------------------------------


def hill_climb_permutation(seq, time_deadline, max_steps=300_000):
    """
    Stochastic hill‑climb that swaps two entries or transfers a fraction
    of mass from one position to another.  The true objective (via
    `compute_score`) is used for acceptance.
    """
    a = np.array(seq, dtype=float)
    n = a.shape[0]

    best_a = a.copy()
    best_score = compute_score(best_a)

    step_scale = 0.1  # initial fraction of mass moved in a transfer

    while max_steps > 0 and time.time() < time_deadline:
        move_type = np.random.rand()
        cand = best_a.copy()

        if move_type < 0.5:
            # simple swap
            i, j = np.random.randint(n, size=2)
            cand[i], cand[j] = cand[j], cand[i]
        else:
            # move a fraction from i to j
            i, j = np.random.randint(n, size=2)
            if i == j:
                continue
            delta = step_scale * best_a[i]
            if delta <= 0:
                continue
            cand[i] -= delta
            cand[j] += delta
            if cand[i] < 0:
                cand[i] = 0.0

        cand = project_to_simplex(cand)
        cur_score = compute_score(cand)

        if cur_score < best_score:
            best_score = cur_score
            best_a = cand
            step_scale = min(step_scale * 1.05, 0.5)
        else:
            step_scale *= 0.97

        max_steps -= 1

    return best_a.tolist(), best_score
