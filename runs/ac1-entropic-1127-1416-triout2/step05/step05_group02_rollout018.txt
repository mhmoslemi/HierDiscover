```
def smooth_gradient_descent(initial_a, deadline, max_iter=100000, step0=0.1, tau=0.1):
    a = project_to_simplex(initial_a.copy())
    n = len(a)
    L = 2*n - 1
    for it in range(1, max_iter+1):
        if time.time() >= deadline:
            break
        conv = convolution_fft(a, a)
        max_val = float(np.max(conv))
        # compute weights for softmax with temperature tau
        # subtract max for numeric stability
        scaled = (conv - max_val) / tau  # negative numbers
        # exponentials
        exp_vals = np.exp(scaled)
        w = exp_vals / np.sum(exp_vals)

        w_rev = w[::-1]
        conv_rev = convolution_fft(a, w_rev)

        # step size
        eta = step0 / np.sqrt(it)
        a = a - eta * subgrad
        a = project_to_simplex(a)
        if it % 5000 == 0:
            cur_score = compute_score(a)
            print(f"    SmoothGD it={it:,d} τ={tau:.4f} score={cur_score:.6f}")
    return a
```

```
def smooth_gradient_descent(initial_a, deadline, max_iter=100_000, step0=0.2, tau_start=0.5, tau_end=0.02):
    a = project_to_simplex(initial_a.copy())
    n = len(a); L = 2*n-1
    half = max_iter // 2
    for it in range(1, max_iter+1):
        if time.time() >= deadline:
            break
        conv = convolution_fft(a, a)
        max_val = float(np.max(conv))
        if it <= half:
            tau = tau_start
        else:
            tau = tau_end
        # compute softmax weights
        scaled = (conv - max_val) / tau
        w = np.exp(scaled)
        w_sum = np.sum(w)
        if w_sum == 0:
            w = np.ones_like(w) / w.size
        else:
            w = w / w_sum
        # subgrad
        w_rev = w[::-1]
        conv_rev = convolution_fft(a, w_rev)
        subgrad = 2.0 * conv_rev[L-1 - np.arange(n)]
        eta = step0 / np.sqrt(it)
        a = a - eta * subgrad
        a = project_to_simplex(a)
        if it % 5000 == 0:
            cur_sc = compute_score(a)
            print(f"    SmoothGD it={it:,d} tau={tau:.4f} score={cur_sc:.6f}")
    return a
```

assistantfinal**Solution Overview**

The bound that is minimised is  

```
score = 2·n·max(conv(a,a)) / (sum(a)²)
```

`conv(a,a)` is the *linear* convolution, therefore the maximum is obtained at the lag
where the overlap of the vector with a shifted (or reversed) copy is largest.
The problem is non‑convex, but the objective is scale‑invariant, so we may safely
optimise over the probability simplex `{a ≥ 0, Σa = 1}` and use the fast surrogate  

```
fast_score(a) = 2·n·max(conv(a,a))
```

The previous implementation used a plain sub‑gradient descent.  
The new code improves the search in three ways

1. **Richer seed pool** – additional parametric families (exponential, linear,
   logistic) and random permutations of high‑variance vectors are added.
2. **Smoothed gradient descent** – a temperature‑controlled log‑sum‑exp
   approximation of the `max` is used.  
   The gradient of `log‑sum‑exp(conv)` is a convex combination of the individual
   shift‑gradients and provides a much more informative direction than the
   binary sub‑gradient.
3. **Adaptive schedule & polishing** – the temperature is gradually lowered,
   the step‑size follows `ηₜ = η₀/√t`, and a final exponentiated‑gradient phase
   (mirror‑descent) cleans the tail.

All heavy work is performed with FFT‑based convolutions (`O(n log n)`), and the
algorithm respects the 1000 s wall‑clock limit by continuously checking the
deadline.

The function `propose_candidate` returns the best sequence found; it can be
called directly by the competition harness.

```python
"""
Improved search for a step‑function minimising
    2·n·max(conv(a,a)) / (sum(a)**2)

The algorithm builds a diverse pool of seed vectors (including monotone
exponential, linear, logistic families and random permutations), then refines
the most promising seeds with a *smoothed* gradient descent that minimises a
log‑sum‑exp surrogate of the max‑convolution.  The temperature τ is reduced
during the run, the step size follows ηₜ = η₀/√t, and each iterate is projected
onto the probability simplex.  A final exponentiated‑gradient polishing step
further reduces the official score.  The whole search stops after the allotted
time budget (≈1000 s) and returns the best sequence discovered.
"""

import time
import numpy as np
from typing import List, Tuple

# ----------------------------------------------------------------------
# Low‑level utilities (no closures)
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Linear convolution via real FFT (length = len(a)+len(b)-1)."""
    m = len(a) + len(b) - 1
    n_fft = next_pow2(m)
    A = np.fft.rfft(a, n=n_fft)
    B = np.fft.rfft(b, n=n_fft)
    C = A * B
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


def compute_score(a: np.ndarray) -> float:
    """Fast surrogate: assumes Σa = 1 → score = 2·n·max(conv(a,a))."""
    n = a.shape[0]
    conv = convolution_fft(a, a)
    return 2.0 * n * float(np.max(conv))


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection onto the probability simplex {x ≥ 0, Σx = 1}.
    Duchi et al., 2008.
    """
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / v.size)

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho_arr = u - cssv / np.arange(1, v.size + 1)
    rho = np.where(rho_arr > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Shape generators (additional families)
# ----------------------------------------------------------------------


def arcsine_initial(n: int) -> np.ndarray:
    """Classic arcsine shape: a_i ∝ 1/√((i+½)*(n‑½‑i))."""
    i = np.arange(n, dtype=float)
    eps = 0.25
    vals = 1.0 / np.sqrt((i + 0.5) * (n - 0.5 - i) + eps)
    vals /= vals.sum()
    return vals


def generate_beta_shape(n: int, alpha: float, beta: float, eps: float) -> np.ndarray:
    """(i+ε)^{α‑1}·(n‑1‑i+ε)^{β‑1} normalised."""
    i = np.arange(n, dtype=float)
    left = (i + eps) ** (alpha - 1.0)
    right = (n - 1 - i + eps) ** (beta - 1.0)
    base = left * right
    total = base.sum()
    if total <= 0.0 or not np.isfinite(total):
        return np.full(n, 1.0 / n)
    return base / total


def generate_power_shape(n: int, p: float, eps: float) -> np.ndarray:
    """Generalised arcsine: a_i ∝ ((i+½)*(n‑½‑i)+ε)^{‑p}."""
    i = np.arange(n, dtype=float)
    denom = ((i + 0.5) * (n - 0.5 - i) + eps) ** p
    vals = 1.0 / denom
    total = vals.sum()
    if total <= 0.0 or not np.isfinite(total):
        return np.full(n, 1.0 / n)
    return vals / total


def generate_exponential_shape(n: int, rate: float) -> np.ndarray:
    """Monotonically decreasing exponential a_i ∝ exp(-rate·i)."""
    i = np.arange(n, dtype=float)
    vals = np.exp(-rate * i)
    vals /= vals.sum()
    return vals


def generate_linear_shape(n: int, ascending: bool = True) -> np.ndarray:
    """Linear ramp a_i ∝ i (or reversed)."""
    i = np.arange(n, dtype=float)
    if not ascending:
        i = i[::-1]
    vals = i + 1.0
    vals /= vals.sum()
    return vals


def generate_logistic_shape(n: int, k: float) -> np.ndarray:
    """Monotone logistic: a_i ∝ 1/(1+exp(-k·(i‑mid)))."""
    i = np.arange(n, dtype=float)
    mid = (n - 1) / 2.0
    vals = 1.0 / (1.0 + np.exp(-k * (i - mid)))
    vals /= vals.sum()
    return vals


def generate_mixture_shape(n: int, weight_arc: float) -> np.ndarray:
    """Weight_arc·arcsine + (1‑weight_arc)·uniform."""
    arc = arcsine_initial(n)
    uni = np.full(n, 1.0 / n)
    mix = weight_arc * arc + (1.0 - weight_arc) * uni
    mix /= mix.sum()
    return mix


def generate_random_permutation(base: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """Randomly permutes the entries of a base distribution."""
    perm = rng.permutation(len(base))
    vec = base[perm]
    # re‑normalise (numerical safety)
    vec = np.maximum(vec, 0.0)
    if vec.sum() == 0.0:
        vec = np.full_like(vec, 1.0 / len(vec))
    else:
        vec = vec / vec.sum()
    return vec


# ----------------------------------------------------------------------
# Optimisation primitives
# ----------------------------------------------------------------------


def smooth_gradient_descent(initial_a: np.ndarray,
                            deadline: float,
                            max_iter: int = 120_000,
                            step0: float = 0.2,
                            tau_start: float = 0.5,
                            tau_end: float = 0.02) -> np.ndarray:
    """
    Gradient descent on the smooth surrogate
        φ_τ(a) = τ·log Σ_k exp(conv_k(a)/τ)

    τ is gradually lowered (first half of the run uses tau_start,
    second half uses tau_end).  The gradient of φ_τ is a convex combination of
    the individual shift‑gradients, which yields a more stable direction than
    the binary sub‑gradient.  After each step we project onto the simplex.
    """
    a = project_to_simplex(initial_a.copy())
    n = len(a)
    L = 2 * n - 1
    half = max_iter // 2

    for it in range(1, max_iter + 1):
        if time.time() >= deadline:
            break

        conv = convolution_fft(a, a)
        max_val = float(np.max(conv))

        # temperature schedule
        tau = tau_start if it <= half else tau_end

        # soft‑max weights over the convolution entries
        scaled = (conv - max_val) / tau          # ≤ 0, avoids overflow
        w = np.exp(scaled)
        w_sum = w.sum()
        if w_sum == 0.0:
            w = np.full_like(w, 1.0 / w.size)
        else:
            w = w / w_sum

        # gradient = 2·∑_k w_k·shift(a,k)
        w_rev = w[::-1]
        conv_rev = convolution_fft(a, w_rev)
        subgrad = 2.0 * conv_rev[L - 1 - np.arange(n)]

        eta = step0 / np.sqrt(it)
        a = a - eta * subgrad
        a = project_to_simplex(a)

        if it % 5_000 == 0:
            cur_sc = compute_score(a)
            print(f"    SmoothGD it={it:,d} τ={tau:.4f} score={cur_sc:.6f}")

    return a


def eg_refine(a: np.ndarray,
              deadline: float,
              max_iter: int = 80_000,
              eta0: float = 0.25) -> np.ndarray:
    """
    Mirror‑descent polishing using an exponentiated‑gradient update.
    Keeps the iterate on the simplex automatically.
    """
    a = a.copy()
    n = len(a)
    L = 2 * n - 1

    for it in range(1, max_iter + 1):
        if time.time() >= deadline:
            break

        conv = convolution_fft(a, a)
        max_val = float(np.max(conv))
        tol = max_val * 1e-12 + 1e-14
        mask = (np.abs(conv - max_val) <= tol).astype(float)

        mask_rev = mask[::-1]
        conv_rev = convolution_fft(a, mask_rev)
        subgrad = 2.0 * conv_rev[L - 1 - np.arange(n)]

        eta = eta0 / (np.sqrt(it) + 1.0)
        a = a * np.exp(-eta * subgrad)

        total = a.sum()
        if total == 0.0:
            a = np.full_like(a, 1.0 / n)
        else:
            a = a / total

        if it % 5_000 == 0:
            cur_sc = compute_score(a)
            print(f"    EG it={it:,d} score={cur_sc:.6f}")

    return a

def propose_candidate(seed: int = 42,
                     budget_s: int = 1000,
                     **kwargs) -> List[float]:
    """
    Search for a non‑negative step sequence with a low ``evaluate_sequence``
    value.  The routine builds a large pool of seed vectors (various parametric
    families, random permutations, mixtures), selects the best few, refines
    each with a smoothed gradient descent + EG polish, and finally applies a
    global polish to the current best.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 2.0   # safety margin

    best_seq: List[float] = []
    best_val: float = float("inf")

    # ------------------------------------------------------------------
    # Optional fallback from a previous run
    # ------------------------------------------------------------------
    if "height_sequence_1" in globals():
        try:
            fallback = [float(x) for x in globals()["height_sequence_1"]]
            val = evaluate_sequence(fallback)
            print(f"Fallback (height_sequence_1) score = {val:.6f}")
            if val < best_val:
                best_val = val
                best_seq = fallback
        except Exception:
            pass

    # ------------------------------------------------------------------
    # 1️⃣  Build a diverse pool of candidates
    # ------------------------------------------------------------------
    lengths = [256, 512, 1024, 2048]   # larger sizes improve the bound but cost more
    global_candidates: List[Tuple[float, np.ndarray]] = []

    for n in lengths:
        if time.time() >= deadline - 20.0:
            break
        print(f"\n=== Generating candidates for length n = {n} ===")

        # deterministic starters
        starters = [np.full(n, 1.0 / n), arcsine_initial(n)]
        for st in starters:
            sc = compute_score(st)
            global_candidates.append((sc, st.copy()))

        # ----- random beta‑family -----
        for _ in range(180):
            if time.time() >= deadline - 5.0:
                break
            a = rng.uniform(0.1, 5.0)
            b = rng.uniform(0.1, 5.0)
            eps = rng.uniform(0.0, max(1.0, 0.02 * n))
            cand = generate_beta_shape(n, a, b, eps)
            sc = compute_score(cand)
            global_candidates.append((sc, cand))

        # ----- random power‑shape -----
        for _ in range(180):
            if time.time() >= deadline - 5.0:
                break
            p = rng.uniform(0.1, 2.5)
            eps = rng.uniform(0.0, 2.0)
            cand = generate_power_shape(n, p, eps)
            sc = compute_score(cand)
            global_candidates.append((sc, cand))

        # ----- exponential families (monotone) -----
        for _ in range(100):
            if time.time() >= deadline - 5.0:
                break
            rate = rng.uniform(0.001, 5.0)
            cand = generate_exponential_shape(n, rate)
            sc = compute_score(cand)
            global_candidates.append((sc, cand))

        # ----- linear ramps (ascending / descending) -----
        for asc in (True, False):
            cand = generate_linear_shape(n, ascending=asc)
            sc = compute_score(cand)
            global_candidates.append((sc, cand))

        # ----- logistic shapes -----
        for _ in range(80):
            if time.time() >= deadline - 5.0:
                break
            k = rng.uniform(0.05, 5.0)
            cand = generate_logistic_shape(n, k)
            sc = compute_score(cand)
            global_candidates.append((sc, cand))

        # ----- mixtures of arcsine & uniform -----
        for _ in range(80):
            if time.time() >= deadline - 5.0:
                break
            w = rng.random()
            cand = generate_mixture_shape(n, w)
            sc = compute_score(cand)
            global_candidates.append((sc, cand))

        # ----- random permutations of a high‑variance base vector -----
        # base vector: exponential with a random rate (keeps variance high)
        base = generate_exponential_shape(n, rng.uniform(0.5, 3.0))
        for _ in range(80):
            if time.time() >= deadline - 5.0:
                break
            cand = generate_random_permutation(base, rng)
            sc = compute_score(cand)
            global_candidates.append((sc, cand))

    # Keep the very best few seeds for intensive refinement
    global_candidates.sort(key=lambda x: x[0])
    top_seeds = [vec for _, vec in global_candidates[:12]]
    print(f"\nSelected {len(top_seeds)} top seeds for refinement "
          f"(score range {global_candidates[0][0]:.6f} – {global_candidates[-1][0]:.6f})")

    # ------------------------------------------------------------------
    # 2️⃣  Refine each seed with smoothed GD + EG polish
    # ------------------------------------------------------------------
    for idx, seed_vec in enumerate(top_seeds, start=1):
        if time.time() >= deadline - 5.0:
            break
        n = seed_vec.shape[0]
        print(f"\nRefining seed {idx}/{len(top_seeds)} (n={n}) …")

        # Allocate time for this seed (roughly equal share of the remainder)
        remaining = deadline - time.time()
        time_for_seed = remaining / (len(top_seeds) - idx + 1)
        max_iter = int(min(200_000, time_for_seed * 30))

        refined = smooth_gradient_descent(seed_vec,
                                          deadline,
                                          max_iter=max_iter,
                                          step0=0.24,
                                          tau_start=0.5,
                                          tau_end=0.02)

        # EG polishing (lighter, but often helps)
        if time.time() < deadline - 2.0:
            remaining = deadline - time.time()
            max_iter_eg = int(min(90_000, remaining * 30))
            refined = eg_refine(refined, deadline,
                                max_iter=max_iter_eg,
                                eta0=0.22)

        # Official evaluation
        try:
            val = evaluate_sequence(refined.tolist())
            print(f"  → official score = {val:.6f}")
            if val < best_val:
                best_val = val
                best_seq = refined.tolist()
                print(f"  *** New global best: {best_val:.6f} ***")
        except Exception as e:
            print(f"  Evaluation failed: {e}")

    # ------------------------------------------------------------------
    # 3️⃣  Global polish on the overall best (if time remains)
    # ------------------------------------------------------------------
    if best_seq and time.time() < deadline - 3.0:
        print("\nFinal global polish …")
        a = np.array(best_seq, dtype=float)

        remaining = deadline - time.time()
        max_iter_sub = int(min(300_000, remaining * 30))
        a = smooth_gradient_descent(a,
                                    deadline,
                                    max_iter=max_iter_sub,
                                    step0=0.18,
                                    tau_start=0.4,
                                    tau_end=0.01)

        if time.time() < deadline - 2.0:
            remaining = deadline - time.time()
            max_iter_eg = int(min(150_000, remaining * 30))
            a = eg_refine(a, deadline,
                          max_iter=max_iter_eg,
                          eta0=0.20)

        final_val = evaluate_sequence(a.tolist())
        print(f"Final polished official score = {final_val:.6f}")
        if final_val < best_val:
            best_val = final_val
            best_seq = a.tolist()
            print("*** Updated global best after final polish ***")

    # ------------------------------------------------------------------
    # 4️⃣  Guard – ensure a non‑empty return value
    # ------------------------------------------------------------------
    if not best_seq:
        fallback_n = 256
        best_seq = [1.0 / fallback_n] * fallback_n
        best_val = evaluate_sequence(best_seq)

    print(f"\nSearch finished. Best reported official score = {best_val:.6f}")
    return best_seq
```
