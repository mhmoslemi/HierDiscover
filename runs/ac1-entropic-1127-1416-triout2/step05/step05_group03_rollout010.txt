"""
Improved step‑function search for the objective
    2·n·max(conv(a,a)) / (Σ a)²

The algorithm consists of four phases:

1. **Fast exploratory sampling** – draws many candidates from a variety of
   parametric families (beta, power‑law, mixtures).  A small “elite” set
   (the best K vectors) is kept.

2. **Upsampling & polishing** – each elite candidate is up‑sampled to a
   larger length (default 4096) and refined by three optimisation stages:
      • Adam on a smooth log‑sum‑exp surrogate (high temperature → close to
        the true max).  The temperature is annealed from 1 to 800.
      • Projected sub‑gradient descent on the exact max‑convolution with a
        back‑tracking line‑search.
      • A cheap stochastic Gaussian walk that can accept occasional uphill
        moves.

3. **Global polishing** – the best vector found overall is polished again
   at the final length with the same three stages, using a larger iteration
   budget.

4. **Safety fall‑back** – if nothing was found a uniform vector of length
   256 is returned.

All vectors are kept on the probability simplex (Σ a = 1); thus the official
score equals `2·n·max(conv(a,a))`.  The search respects the 1000 s budget and
prints progress information.
"""

import time
import numpy as np
from typing import List, Tuple

# ----------------------------------------------------------------------
# Basic linear‑algebra utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """
    Linear convolution via real FFT.
    Returns an array of length len(a)+len(b)-1.
    """
    m = a.size + b.size - 1
    n_fft = next_pow2(m)
    A = np.fft.rfft(a, n=n_fft)
    B = np.fft.rfft(b, n=n_fft)
    C = A * B
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


def max_convolution(a: np.ndarray) -> float:
    """Maximum entry of the self‑convolution of a."""
    return float(np.max(convolution_fft(a, a)))


def compute_score(a: np.ndarray) -> float:
    """
    Fast surrogate (official score for Σa = 1):
        2·n·max(conv(a,a))
    """
    n = a.shape[0]
    return 2.0 * n * max_convolution(a)


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection onto the probability simplex {x ≥ 0, Σx = 1}.
    Duchi et al. (2008) implementation.
    """
    n = v.shape[0]
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / n)

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.where(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w

# Shape generators
# ----------------------------------------------------------------------


def generate_beta_shape(
    n: int, alpha: float, beta: float, eps: float
) -> np.ndarray:
    """
    Symmetric beta‑family density on {0,…,n‑1}:
        a_i ∝ (i+ε)^{α‑1}·(n‑1‑i+ε)^{β‑1}.
    The vector is normalised to Σa = 1.
    """
    i = np.arange(n, dtype=float)
    left = (i + eps) ** (alpha - 1.0)
    right = (n - 1 - i + eps) ** (beta - 1.0)
    base = left * right
    total = base.sum()
    if total <= 0.0 or not np.isfinite(total):
        return np.full(n, 1.0 / n)
    return base / total


def generate_beta_mixture(
    n: int,
    alpha1: float,
    beta1: float,
    eps1: float,
    alpha2: float,
    beta2: float,
    eps2: float,
    w: float,
) -> np.ndarray:
    """Convex combination of two beta‑family densities."""
    a1 = generate_beta_shape(n, alpha1, beta1, eps1)
    a2 = generate_beta_shape(n, alpha2, beta2, eps2)
    mix = w * a1 + (1.0 - w) * a2
    mix_sum = mix.sum()
    if mix_sum == 0.0 or not np.isfinite(mix_sum):
        return np.full(n, 1.0 / n)
    return mix / mix_sum


def generate_power_shape(n: int, p: float, eps: float) -> np.ndarray:
    """
    Power‑law “U‑shaped” density:
        a_i ∝ (i+ε)^{-p}·(n‑1‑i+ε)^{-p}.
    The vector is normalised to Σa = 1.
    """
    i = np.arange(n, dtype=float)
    left = (i + eps) ** (-p)
    right = (n - 1 - i + eps) ** (-p)
    base = left * right
    total = base.sum()
    if total <= 0.0 or not np.isfinite(total):
        return np.full(n, 1.0 / n)
    return base / total


def arcsine_initial(n: int) -> np.ndarray:
    """Classic arcsine shape (symmetrised)."""
    i = np.arange(n, dtype=float)
    eps = 0.25
    vals = 1.0 / np.sqrt((i + 0.5) * (n - 0.5 - i) + eps)
    vals /= vals.sum()
    return vals

# Smooth surrogate (log‑sum‑exp) and its gradient (used by Adam)
# ----------------------------------------------------------------------


def smooth_gradient(a: np.ndarray, lam: float) -> Tuple[np.ndarray, float]:
    """
    Gradient of the log‑sum‑exp surrogate of max(conv(a,a)).
    Returns (gradient, surrogate value B).
    """
    n = a.shape[0]
    conv = convolution_fft(a, a)  # length 2n‑1
    max_c = np.max(conv)

    exp_term = np.exp(lam * (conv - max_c))
    sum_exp = np.sum(exp_term)
    w = exp_term / sum_exp                     # soft‑max over convolution entries
    B = max_c + np.log(sum_exp) / lam          # smooth approximation of the true max

    # Weighted sum of sub‑gradients:
    # ∂/∂a Σ_k w_k (a*a)_k = 2 * (a ⋆ w_rev)
    w_rev = w[::-1]
    conv_rev = convolution_fft(a, w_rev)       # length 3n‑2
    L = 2 * n - 1
    weighted_grad = 2.0 * conv_rev[L - 1 - np.arange(n)]

    s = np.sum(a)
    term1 = (2.0 * n / (s ** 2)) * weighted_grad
    term2 = -(4.0 * n * B / (s ** 3)) * np.ones_like(a)
    grad = term1 + term2
    return grad, B


def adam_optimize(
    initial_a: np.ndarray,
    time_deadline: float,
    max_iter: int = 150_000,
    lr0: float = 0.12,
    lam_start: float = 1.0,
    lam_end: float = 800.0,
    noise_every: int = 1500,
    noise_scale: float = 4e-5,
) -> Tuple[np.ndarray, float]:
    """
    Adam optimiser on the smooth log‑sum‑exp surrogate.
    Returns the best vector (projected on the simplex) and its fast score.
    """
    a = project_to_simplex(initial_a.astype(float).copy())
    best_a = a.copy()
    best_sc = compute_score(a)

    beta1, beta2 = 0.9, 0.999
    eps_adam = 1e-8
    m = np.zeros_like(a)
    v = np.zeros_like(a)

    n = a.shape[0]

    for it in range(1, max_iter + 1):
        if time.time() >= time_deadline:
            break

        # quadratic annealing of the temperature
        t = it / max_iter
        lam = lam_start + (lam_end - lam_start) * (t ** 0.5)

        grad, _ = smooth_gradient(a, lam)

        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)

        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        step = lr0 * m_hat / (np.sqrt(v_hat) + eps_adam)
        a_new = a - step

        if noise_every and (it % noise_every == 0):
            a_new += np.random.randn(n) * noise_scale

        a_new = project_to_simplex(a_new)

        # we evaluate the *fast* surrogate score only occasionally to save time
        if it % 200 == 0:
            cur_sc = compute_score(a_new)
            if cur_sc < best_sc:
                best_sc = cur_sc
                best_a = a_new.copy()

        a = a_new

        if it % 5000 == 0:
            cur_sc = compute_score(a)
            print(
                f"    Adam iter={it:,d} cur={cur_sc:.6f} best={best_sc:.6f}"
            )

    # final evaluation of the best point we have
    final_sc = compute_score(best_a)
    return best_a, final_sc


# ----------------------------------------------------------------------
# Exact sub‑gradient descent with back‑tracking line‑search
# ----------------------------------------------------------------------


def subgradient_optimize(
    initial_a: np.ndarray,
    time_deadline: float,
    step0: float = 0.25,
    max_iter: int = 120_000,
) -> Tuple[np.ndarray, float]:
    """
    Projected sub‑gradient descent on the *true* max‑convolution.
    Uses a simple back‑tracking line‑search to guarantee descent.
    Returns the best vector and its fast score.
    """
    a = initial_a.copy()
    best_a = a.copy()
    best_sc = compute_score(a)

    n = a.shape[0]

    for it in range(1, max_iter + 1):
        if time.time() >= time_deadline:
            break

        conv = convolution_fft(a, a)
        max_val = float(np.max(conv))
        idx = int(np.argmax(conv))

        # gradient of the active quadratic term (pairwise product)
        i = np.arange(n)
        j = idx - i
        mask = (j >= 0) & (j < n)
        grad = np.zeros_like(a)
        grad[mask] = 2.0 * a[j[mask]]

        # step size schedule (Polyak‑like scaling)
        step = step0 / np.sqrt(it)

        # back‑tracking line‑search
        for _ in range(12):
            a_new = a - step * grad
            a_new = project_to_simplex(a_new)
            new_val = max_convolution(a_new)
            if new_val < max_val - 1e-14:
                break
            step *= 0.5

        cur_sc = compute_score(a_new)
        if cur_sc < best_sc:
            best_sc = cur_sc
            best_a = a_new.copy()

        a = a_new

        if it % 30000 == 0:
            print(
                f"    Sub‑grad iter={it:,d} cur={cur_sc:.6f} best={best_sc:.6f}"
            )

    return best_a, best_sc


# ----------------------------------------------------------------------
# Stochastic local walk (Gaussian perturbations)
# ----------------------------------------------------------------------


def random_local_search(
    initial_a: np.ndarray,
    time_deadline: float,
    init_step: float = 0.012,
    max_iter: int = 200_000,
) -> Tuple[np.ndarray, float]:
    """
    Gaussian walk on the simplex with a tiny uphill‑move probability.
    Returns the best vector seen and its fast score.
    """
    a = initial_a.copy()
    best_a = a.copy()
    best_sc = compute_score(a)

    n = a.shape[0]
    rng = np.random.default_rng()

    for it in range(1, max_iter + 1):
        if time.time() >= time_deadline:
            break

        step = init_step / np.sqrt(it)
        noise = rng.normal(scale=step, size=n)
        a_proposed = a + noise
        a_proposed = project_to_simplex(a_proposed)

        cur_sc = compute_score(a_proposed)

        # Accept if improvement; otherwise accept with a tiny probability
        if cur_sc < best_sc or (rng.random() < 1e-4 and cur_sc < 10 * best_sc):
            a = a_proposed
            if cur_sc < best_sc:
                best_sc = cur_sc
                best_a = a_proposed.copy()

        if it % 60000 == 0:
            print(
                f"    Random local iter={it:,d} cur={cur_sc:.6f} best={best_sc:.6f}"
            )

    return best_a, best_sc


# ----------------------------------------------------------------------
# Utility helpers
# ----------------------------------------------------------------------


def upsample_vector(v: np.ndarray, new_n: int) -> np.ndarray:
    """
    Linear interpolation up‑sampling to a new size `new_n`.
    The result is re‑normalised to Σ = 1.
    """
    old_n = v.shape[0]
    if old_n == new_n:
        return v.copy()
    x_old = np.linspace(0.0, 1.0, old_n)
    x_new = np.linspace(0.0, 1.0, new_n)
    v_new = np.interp(x_new, x_old, v)
    v_new = v_new / v_new.sum()
    return v_new


def update_top_k(top_k: List[Tuple[float, np.ndarray]],
                 score: float,
                 vec: np.ndarray,
                 max_k: int) -> List[Tuple[float, np.ndarray]]:
    """Maintain a sorted list of at most `max_k` best (score,vector) pairs."""
    if len(top_k) < max_k:
        top_k.append((score, vec.copy()))
        top_k.sort(key=lambda x: x[0])
    else:
        if score < top_k[-1][0]:
            top_k[-1] = (score, vec.copy())
            top_k.sort(key=lambda x: x[0])
    return top_k

# Main entry point required by the competition harness
# ----------------------------------------------------------------------


def propose_candidate(
    seed: int = 42,
    budget_s: int = 1000,
    **kwargs,
) -> List[float]:
    """
    Search for a non‑negative step sequence with a low official score.
    Returns the best list of heights found within the allotted budget.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 2.0   # safety margin

    best_seq: List[float] = []
    best_val: float = float("inf")

    # ------------------------------------------------------------------
    # 0.  Fallback (if a previous construction is available)
    # ------------------------------------------------------------------
    try:
        fallback = [float(x) for x in height_sequence_1]  # type: ignore
        val = evaluate_sequence(fallback)
        print(f"Fallback (height_sequence_1) official score = {val:.6f}")
        if val < best_val:
            best_val = val
            best_seq = fallback
    except Exception:
        pass

    # ------------------------------------------------------------------
    # 1.  Fast exploratory sampling on a few modest lengths
    # ------------------------------------------------------------------
    # small lengths (fast to evaluate)
    small_lengths = [512, 1024]
    elite_per_len = 6                     # keep this many elites for polishing
    for n in small_lengths:
        if time.time() >= deadline - 30.0:
            break
        print(f"\n=== Sampling length {n} ===")
        top_k: List[Tuple[float, np.ndarray]] = []

        # deterministic starters
        starters = [
            np.full(n, 1.0 / n),
            arcsine_initial(n),
        ]
        for st in starters:
            sc = compute_score(st)
            top_k = update_top_k(top_k, sc, st, elite_per_len)

        max_samples = int(4000 * (budget_s / 1000.0))
        for _ in range(max_samples):
            if time.time() >= deadline - 30.0:
                break
            r = rng.random()
            if r < 0.25:   # symmetric beta
                log_alpha = rng.uniform(np.log10(0.001), np.log10(5.0))
                alpha = 10 ** log_alpha
                eps = rng.uniform(1e-8, 0.02 * n)
                cand = generate_beta_shape(n, alpha, alpha, eps)
            elif r < 0.5:  # generic beta
                log_alpha = rng.uniform(np.log10(0.001), np.log10(5.0))
                log_beta = rng.uniform(np.log10(0.001), np.log10(5.0))
                alpha = 10 ** log_alpha
                beta = 10 ** log_beta
                eps = rng.uniform(1e-8, 0.02 * n)
                cand = generate_beta_shape(n, alpha, beta, eps)
            elif r < 0.75:  # power‑law U‑shape
                log_p = rng.uniform(np.log10(0.1), np.log10(5.0))
                p = 10 ** log_p
                eps = rng.uniform(1e-8, 0.02 * n)
                cand = generate_power_shape(n, p, eps)
            else:          # mixture of two power shapes
                log_p1 = rng.uniform(np.log10(0.1), np.log10(5.0))
                log_p2 = rng.uniform(np.log10(0.1), np.log10(5.0))
                p1 = 10 ** log_p1
                p2 = 10 ** log_p2
                eps1 = rng.uniform(1e-8, 0.02 * n)
                eps2 = rng.uniform(1e-8, 0.02 * n)
                w = rng.random()
                a1 = generate_power_shape(n, p1, eps1)
                a2 = generate_power_shape(n, p2, eps2)
                cand = w * a1 + (1.0 - w) * a2
                cand = cand / cand.sum()
            sc = compute_score(cand)
            top_k = update_top_k(top_k, sc, cand, elite_per_len)

        # ------------------------------------------------------------------
        # 2.  Upsample elites and polish them at a larger length
        # ------------------------------------------------------------------
        final_n = 4096
        for idx, (sc, vec) in enumerate(top_k):
            if time.time() >= deadline - 5.0:
                break
            print(f"  Polishing elite #{idx+1} (n={n} → {final_n})")
            a = upsample_vector(vec, final_n)

            # Adam stage
            remaining = deadline - time.time()
            max_iter_adam = int(min(150_000, remaining * 25))
            a, _ = adam_optimize(
                initial_a=a,
                time_deadline=deadline,
                max_iter=max_iter_adam,
                lr0=0.12,
                lam_start=1.0,
                lam_end=800.0,
                noise_every=1300,
                noise_scale=5e-5,
            )

            # Sub‑gradient stage
            if time.time() < deadline - 3.0:
                max_iter_sub = int(min(80_000, (deadline - time.time()) * 30))
                a, _ = subgradient_optimize(
                    a,
                    time_deadline=deadline,
                    step0=0.25,
                    max_iter=max_iter_sub,
                )

            # Final stochastic walk
            if time.time() < deadline - 2.0:
                max_iter_loc = int(min(120_000, (deadline - time.time()) * 35))
                a, _ = random_local_search(
                    a,
                    time_deadline=deadline,
                    init_step=0.010,
                    max_iter=max_iter_loc,
                )

            # Official evaluation
            try:
                val = evaluate_sequence(a.tolist())
                print(f"  → official score (elite #{idx+1}) = {val:.6f}")
                if val < best_val:
                    best_val = val
                    best_seq = a.tolist()
                    print(f"  *** New global best = {best_val:.6f} ***")
            except Exception:
                pass

    # ------------------------------------------------------------------
    # 3.  Global polishing of the overall best vector (if any)
    # ------------------------------------------------------------------
    if best_seq and time.time() < deadline - 5.0:
        print("\n=== Global polishing of the best sequence ===")
        a = np.array(best_seq, dtype=float)

        # Ensure we are at the final length (might already be)
        final_n = max(len(a), 4096)
        if len(a) != final_n:
            a = upsample_vector(a, final_n)

        # Stage A – long Adam run (high temperature)
        remaining = deadline - time.time()
        max_iter_adam = int(min(300_000, remaining * 30))
        print(f"  Adam polishing (max_iter={max_iter_adam}) …")
        a, _ = adam_optimize(
            initial_a=a,
            time_deadline=deadline,
            max_iter=max_iter_adam,
            lr0=0.10,
            lam_start=1.0,
            lam_end=800.0,
            noise_every=1500,
            noise_scale=4e-5,
        )

        # Stage B – exact sub‑gradient descent
        if time.time() < deadline - 4.0:
            max_iter_sub = int(min(150_000, (deadline - time.time()) * 30))
            print(f"  Sub‑gradient polishing (max_iter={max_iter_sub}) …")
            a, _ = subgradient_optimize(
                a,
                time_deadline=deadline,
                step0=0.20,
                max_iter=max_iter_sub,
            )

        # Stage C – stochastic local walk
        if time.time() < deadline - 2.0:
            max_iter_loc = int(min(250_000, (deadline - time.time()) * 40))
            print(f"  Random local polishing (max_iter={max_iter_loc}) …")
            a, _ = random_local_search(
                a,
                time_deadline=deadline,
                init_step=0.008,
                max_iter=max_iter_loc,
            )

        try:
            final_val = evaluate_sequence(a.tolist())
            print(f"Final polished official score = {final_val:.6f}")
            if final_val < best_val:
                best_val = final_val
                best_seq = a.tolist()
                print("*** Updated global best after final polish ***")
        except Exception:
            pass

    # ------------------------------------------------------------------
    # 4.  Guard – guarantee a non‑empty return
    # ------------------------------------------------------------------
    if not best_seq:
        # Safe uniform fallback (length 256)
        best_seq = [1.0] * 256
        best_val = evaluate_sequence(best_seq)
        print("Fallback to uniform 256‑vector.", best_val)

    print(f"\nSearch finished. Best reported official score = {best_val:.6f}")
    return best_seq
