def softmax_eg_descent(initial_a, deadline, max_iter, lr0, tau_start, tau_end):
    a = project_to_simplex(initial_a.copy())
    n = len(a)
    L = 2*n-1
    for it in range(1, max_iter+1):
        if time.time() >= deadline: break
        # compute conv
        conv = convolution_fft(a, a)
        maxconv = float(np.max(conv))
        # temperature schedule linear:
        tau = tau_start + (tau_end - tau_start) * (it / max_iter)
        # compute weights
        # subtract maxconv for stability
        w = np.exp(tau * (conv - maxconv))
        w_sum = w.sum()
        if w_sum == 0:
            w = np.full_like(w, 1.0 / w.size)
        else:
            w = w / w_sum
        # gradient via convolution with w_rev
        w_rev = w[::-1]
        conv_rev = convolution_fft(a, w_rev)
        # subgrad = 2 * conv_rev[L-1 - np.arange(n)]
        g = 2.0 * conv_rev[L - 1 - np.arange(n)]
        # step size
        eta = lr0 / np.sqrt(it)
        # EG update
        a = a * np.exp(-eta * g)
        # normalize to simplex
        total = a.sum()
        if total == 0:
            a = np.full_like(a, 1.0 / n)
        else:
            a = a / total
        # occasionally log surrogates
        if it % 2000 == 0:
            cur_s = compute_score(a)
            print(f"    Softmax EG it={it} score={cur_s:.6f}")
    return a

"""
Improved search for a low‑∞‑norm self‑convolution step function.

The algorithm combines three complementary ideas:

1️⃣  **Diverse parametric seeds** –  Samples from many families
    (arcsine, beta, generalized power, Gaussian, mixtures,
    random Dirichlet).  The fast surrogate `2·n·max(conv(a,a))`
    (with Σa = 1) is used to keep only the most promising seeds.

2️⃣  **Smooth‑max gradient descent** –  Instead of the raw sub‑gradient
    of the max, we minimise the smooth approximation  
    `gτ(a) = (1/τ)·log Σₖ exp(τ·convₖ(a))`.  The gradient can be written
    as a weighted convolution and is obtained with two FFTs per
    iteration.  An exponentiated‑gradient (mirror‑descent) update keeps
    the iterates on the probability simplex automatically.
    The temperature τ is annealed from a small value (smooth landscape)
    to a large value (approaching the true max).

3️⃣  **Final polishing** –  A short exact sub‑gradient phase followed
    by the classic exponentiated‑gradient polish (binary mask) as in
    the previous state‑of‑the‑art code.

All phases respect a global time budget (default 1000 s).  The best
vector found is evaluated with the official ``evaluate_sequence`` and
returned.  The implementation uses only NumPy and the standard library.
"""

import time
import heapq
from typing import List, Tuple

import numpy as np

# ----------------------------------------------------------------------
# Low‑level utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Linear convolution via real FFT (length = len(a)+len(b)-1)."""
    m = a.shape[0] + b.shape[0] - 1
    n_fft = next_pow2(m)
    A = np.fft.rfft(a, n=n_fft)
    B = np.fft.rfft(b, n=n_fft)
    C = A * B
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


def compute_fast_score(a: np.ndarray) -> float:
    """
    Fast surrogate for the official score.
    The vector is assumed to be normalised to Σa = 1,
    therefore the denominator disappears and the score is simply
    2·n·max(conv(a,a)).
    """
    n = a.shape[0]
    conv = convolution_fft(a, a)
    return 2.0 * n * float(np.max(conv))


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection onto the probability simplex {x ≥ 0, Σx = 1}."""
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / v.size)

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho_arr = u - cssv / np.arange(1, v.size + 1)
    rho = np.where(rho_arr > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Shape generators
# ----------------------------------------------------------------------


def arcsine_initial(n: int) -> np.ndarray:
    """Classic arcsine shape a_i ∝ 1/√((i+½)*(n‑½‑i))."""
    i = np.arange(n, dtype=float)
    eps = 0.25
    vals = 1.0 / np.sqrt((i + 0.5) * (n - 0.5 - i) + eps)
    vals /= vals.sum()
    return vals


def generate_power_shape(n: int, p: float, eps: float) -> np.ndarray:
    """Generalised arcsine: a_i ∝ ((i+½)*(n‑½‑i)+ε)^{‑p}."""
    i = np.arange(n, dtype=float)
    denom = ((i + 0.5) * (n - 0.5 - i) + eps) ** p
    vals = 1.0 / denom
    total = vals.sum()
    if total <= 0.0 or not np.isfinite(total):
        return np.full(n, 1.0 / n)
    return vals / total


def generate_beta_shape(n: int, alpha: float, beta: float, eps: float) -> np.ndarray:
    """Beta‑family: (i+ε)^{α‑1}·(n‑1‑i+ε)^{β‑1}."""
    i = np.arange(n, dtype=float)
    left = (i + eps) ** (alpha - 1.0)
    right = (n - 1 - i + eps) ** (beta - 1.0)
    base = left * right
    total = base.sum()
    if total <= 0.0 or not np.isfinite(total):
        return np.full(n, 1.0 / n)
    return base / total


def generate_gaussian_shape(n: int, sigma: float) -> np.ndarray:
    """Discrete Gaussian centred in the middle, variance σ²."""
    i = np.arange(n, dtype=float)
    centre = (n - 1) / 2.0
    vals = np.exp(-0.5 * ((i - centre) / sigma) ** 2)
    total = vals.sum()
    if total <= 0.0 or not np.isfinite(total):
        return np.full(n, 1.0 / n)
    return vals / total


def generate_mixture_shape(n: int, weight_arc: float) -> np.ndarray:
    """Mixture (weight_arc)·arcsine + (1‑weight_arc)·uniform."""
    arc = arcsine_initial(n)
    uni = np.full(n, 1.0 / n)
    mix = weight_arc * arc + (1.0 - weight_arc) * uni
    mix /= mix.sum()
    return mix


def generate_triangular_shape(n: int, peak: int) -> np.ndarray:
    """Symmetric triangular bump, apex at index ``peak``."""
    left = np.arange(peak + 1, dtype=float)
    right = np.arange(n - peak - 1, dtype=float)[::-1]
    seq = np.concatenate([left, right])
    seq /= seq.sum()
    return seq


def generate_dirichlet_random(n: int, rng: np.random.Generator) -> np.ndarray:
    """Sample from Dirichlet(1,…,1) – i.e. normalised exponentials."""
    sample = rng.exponential(scale=1.0, size=n)
    sample_sum = sample.sum()
    if sample_sum == 0.0:
        return np.full(n, 1.0 / n)
    return sample / sample_sum

def subgradient_descent_exact(initial_a: np.ndarray,
                              deadline: float,
                              max_iter: int = 80_000,
                              step0: float = 0.05) -> np.ndarray:
    """
    Sub‑gradient descent on  f(a)=2·n·max(conv(a,a)).
    The step size is ηₜ = step0 / √t and we project onto the simplex.
    """
    a = project_to_simplex(initial_a.copy())
    n = a.shape[0]
    L = 2 * n - 1

    for it in range(1, max_iter + 1):
        if time.time() >= deadline:
            break

        conv = convolution_fft(a, a)
        max_val = float(np.max(conv))
        # binary mask for (almost) max entries
        tol = max_val * 1e-12 + 1e-14
        mask = (np.abs(conv - max_val) <= tol).astype(float)

        # sub‑gradient = 2·∑_{k∈K} shift(a,k)
        mask_rev = mask[::-1]
        conv_rev = convolution_fft(a, mask_rev)
        subgrad = 2.0 * conv_rev[L - 1 - np.arange(n)]

        eta = step0 / np.sqrt(it)
        a = a - eta * subgrad
        a = project_to_simplex(a)

        if it % 5000 == 0:
            cur_sc = compute_fast_score(a)
            print(f"    SubGrad it={it:,d} score={cur_sc:.6f}")

    return a


def eg_refine(a: np.ndarray,
              deadline: float,
              max_iter: int = 100_000,
              eta0: float = 0.3) -> np.ndarray:
    """
    Classic exponentiated‑gradient polish using a binary mask.
    """
    n = a.shape[0]
    L = 2 * n - 1

    for it in range(1, max_iter + 1):
        if time.time() >= deadline:
            break

        conv = convolution_fft(a, a)
        max_val = float(np.max(conv))
        tol = max_val * 1e-12 + 1e-14
        mask = (np.abs(conv - max_val) <= tol).astype(float)

        mask_rev = mask[::-1]
        conv_rev = convolution_fft(a, mask_rev)
        subgrad = 2.0 * conv_rev[L - 1 - np.arange(n)]

        eta = eta0 / (np.sqrt(it) + 1.0)
        a = a * np.exp(-eta * subgrad)

        total = a.sum()
        if total == 0.0:
            a = np.full_like(a, 1.0 / n)
        else:
            a = a / total

        if it % 5000 == 0:
            cur_sc = compute_fast_score(a)
            print(f"    EG polish it={it:,d} score={cur_sc:.6f}")

    return a


def softmax_eg_descent(initial_a: np.ndarray,
                       deadline: float,
                       max_iter: int = 60_000,
                       lr0: float = 0.25,
                       tau_start: float = 0.01,
                       tau_end: float = 20.0) -> np.ndarray:
    """
    Mirror‑descent on a smooth approximation of the max:
        gτ(a) = (1/τ)·log Σₖ exp(τ·convₖ(a))

    The gradient is a weighted sum of the standard sub‑gradients,
    computed with a single convolution.  An exponentiated‑gradient
    update keeps the iterate on the simplex automatically.
    """
    a = project_to_simplex(initial_a.copy())
    n = a.shape[0]
    L = 2 * n - 1

    for it in range(1, max_iter + 1):
        if time.time() >= deadline:
            break

        # current convolution
        conv = convolution_fft(a, a)
        max_conv = float(np.max(conv))

        # temperature schedule (linear annealing)
        tau = tau_start + (tau_end - tau_start) * (it / max_iter)

        # soft‑max weights (stable formulation)
        exps = np.exp(tau * (conv - max_conv))
        w = exps / exps.sum()

        # gradient = 2·∑ₖ wₖ·shift(a,k)
        w_rev = w[::-1]
        conv_rev = convolution_fft(a, w_rev)
        grad = 2.0 * conv_rev[L - 1 - np.arange(n)]

        eta = lr0 / np.sqrt(it)
        a = a * np.exp(-eta * grad)
        total = a.sum()
        if total == 0.0:
            a = np.full_like(a, 1.0 / n)
        else:
            a = a / total

        if it % 5000 == 0:
            cur_sc = compute_fast_score(a)
            print(f"    Soft‑max EG it={it:,d} score={cur_sc:.6f}")

    return a

def propose_candidate(seed: int = 42,
                     budget_s: int = 1000,
                     **kwargs) -> List[float]:
    """
    Search for a non‑negative step‑function (list of heights) that
    minimises ``evaluate_sequence``.
    """
    rng = np.random.default_rng(seed)
    start = time.time()
    deadline = start + budget_s - 0.5   # safety margin

    best_seq: List[float] = []
    best_val: float = float("inf")

    # ------------------------------------------------------------------
    # 1️⃣  Generate a large pool of diverse seeds (fast surrogate)
    # ------------------------------------------------------------------
    lengths = [512, 1024, 2048, 4096]
    keep_per_len = 40                # how many best vectors we keep per length
    global_heap: List[Tuple[float, np.ndarray]] = []   # max‑heap on (-score, vec)

    def push_candidate(score: float, vec: np.ndarray) -> None:
        """Maintain a max‑heap of limited size (keeps the smallest scores)."""
        if len(global_heap) < keep_per_len * len(lengths):
            heapq.heappush(global_heap, (-score, vec.copy()))
        else:
            # heap[0] holds the *worst* (largest) score because we store negative values
            if -global_heap[0][0] > score:
                heapq.heapreplace(global_heap, (-score, vec.copy()))

    print("\n=== Stage 1 – randomised seed generation ===")
    for n in lengths:
        if time.time() >= deadline - 10.0:
            break
        print(f"\n--- length = {n} ---")

        # deterministic starters
        push_candidate(compute_fast_score(np.full(n, 1.0 / n)),
                       np.full(n, 1.0 / n))
        push_candidate(compute_fast_score(arcsine_initial(n)), arcsine_initial(n))

        # random parametric families
        for _ in range(180):
            if time.time() >= deadline - 5.0:
                break
            a = rng.uniform(0.1, 5.0)
            b = rng.uniform(0.1, 5.0)
            eps = rng.uniform(0.0, max(1.0, 0.02 * n))
            cand = generate_beta_shape(n, a, b, eps)
            push_candidate(compute_fast_score(cand), cand)

        for _ in range(180):
            if time.time() >= deadline - 5.0:
                break
            p = rng.uniform(0.1, 3.0)
            eps = rng.uniform(0.0, 2.0)
            cand = generate_power_shape(n, p, eps)
            push_candidate(compute_fast_score(cand), cand)

        for _ in range(120):
            if time.time() >= deadline - 5.0:
                break
            sigma = rng.uniform(0.1, n / 4.0)
            cand = generate_gaussian_shape(n, sigma)
            push_candidate(compute_fast_score(cand), cand)

        for _ in range(120):
            if time.time() >= deadline - 5.0:
                break
            w = rng.random()
            cand = generate_mixture_shape(n, w)
            push_candidate(compute_fast_score(cand), cand)

        for _ in range(120):
            if time.time() >= deadline - 5.0:
                break
            peak = rng.integers(0, n)
            cand = generate_triangular_shape(n, peak)
            push_candidate(compute_fast_score(cand), cand)

        for _ in range(120):
            if time.time() >= deadline - 5.0:
                break
            cand = generate_dirichlet_random(n, rng)
            push_candidate(compute_fast_score(cand), cand)

    # Extract the globally best seeds
    global_candidates = sorted([(-s, v) for s, v in global_heap], key=lambda x: x[0])
    top_seeds = [vec for _, vec in global_candidates[:12]]
    print(f"\nSelected {len(top_seeds)} top seeds for intensive refinement "
          f"(fast score range: {global_candidates[0][0]:.6f} – "
          f"{global_candidates[-1][0]:.6f})")

    # ------------------------------------------------------------------
    # 2️⃣  Intensive refinement of each seed
    # ------------------------------------------------------------------
    for idx, seed_vec in enumerate(top_seeds, start=1):
        if time.time() >= deadline - 5.0:
            break
        n = seed_vec.shape[0]
        print(f"\nRefining seed {idx}/{len(top_seeds)} (n={n}) …")

        # 2a – sub‑gradient descent (coarse)
        remaining = deadline - time.time()
        max_iter_sub = int(min(150_000, remaining * 30))
        a = subgradient_descent_exact(seed_vec, deadline,
                                      max_iter=max_iter_sub,
                                      step0=0.18)

        # 2b – smooth‑max (soft‑max) EG descent
        remaining = deadline - time.time()
        max_iter_soft = int(min(80_000, remaining * 30))
        if max_iter_soft > 0:
            a = softmax_eg_descent(a, deadline,
                                   max_iter=max_iter_soft,
                                   lr0=0.25,
                                   tau_start=0.01,
                                   tau_end=25.0)

        # 2c – classic EG polish (binary mask)
        remaining = deadline - time.time()
        max_iter_eg = int(min(80_000, remaining * 30))
        if max_iter_eg > 0:
            a = eg_refine(a, deadline, max_iter=max_iter_eg, eta0=0.22)

        # official evaluation
        try:
            val = evaluate_sequence(a.tolist())
            print(f"  → official score = {val:.6f}")
            if val < best_val:
                best_val = val
                best_seq = a.tolist()
                print(f"  *** New global best: {best_val:.6f} ***")
        except Exception as e:
            print(f"  Evaluation failed: {e}")

    # ------------------------------------------------------------------
    # 3️⃣  Global polish of the current best (if time permits)
    # ------------------------------------------------------------------
    if best_seq and time.time() < deadline - 3.0:
        print("\n=== Final global polish ===")
        a = np.array(best_seq, dtype=float)

        # sub‑gradient phase (larger budget)
        remaining = deadline - time.time()
        max_iter_sub = int(min(250_000, remaining * 30))
        if max_iter_sub > 0:
            a = subgradient_descent_exact(a, deadline,
                                          max_iter=max_iter_sub,
                                          step0=0.12)

        # soft‑max EG descent with a slower annealing
        remaining = deadline - time.time()
        max_iter_soft = int(min(150_000, remaining * 30))
        if max_iter_soft > 0:
            a = softmax_eg_descent(a, deadline,
                                   max_iter=max_iter_soft,
                                   lr0=0.20,
                                   tau_start=0.01,
                                   tau_end=30.0)

        # final EG polish
        remaining = deadline - time.time()
        max_iter_eg = int(min(150_000, remaining * 30))
        if max_iter_eg > 0:
            a = eg_refine(a, deadline, max_iter=max_iter_eg, eta0=0.18)

        final_val = evaluate_sequence(a.tolist())
        print(f"Final polished official score = {final_val:.6f}")
        if final_val < best_val:
            best_val = final_val
            best_seq = a.tolist()
            print("*** Updated global best after final polish ***")

    # ------------------------------------------------------------------
    # 4️⃣  Fallback in the extremely unlikely case that nothing was found
    # ------------------------------------------------------------------
    if not best_seq:
        fallback_n = 256
        best_seq = [1.0 / fallback_n] * fallback_n
        best_val = evaluate_sequence(best_seq)
        print("Fallback (uniform) used.")

    print(f"\nSearch finished. Best reported official score = {best_val:.6f}")
    return best_seq
