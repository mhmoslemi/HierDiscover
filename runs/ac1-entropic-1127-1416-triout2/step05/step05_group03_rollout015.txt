"""
Improved step‑function search.

The algorithm explores several lengths (powers of two up to 8192) and
generates a large variety of candidate shapes:
    – uniform, arcsine, exponential decay, power‑law decay,
    – β‑family densities and mixtures of two β‑densities.
The best candidate for each length is refined by three stages:

1. Adam optimisation of a smooth log‑sum‑exp surrogate (λ annealed up
   to 800 to approximate the true max).  The iterate is projected onto
   the probability simplex after each step.

2. Exact sub‑gradient descent on the true max‑convolution with a
   back‑tracking line‑search guaranteeing descent.

3. A cheap stochastic local walk (Gaussian perturbations) that also
   accepts occasional uphill moves.

Whenever a longer length is considered, the current global best vector
is upsampled to initialise the search, which provides a good warm‑start.
All stages respect the global 1000 s budget; the routine always returns
a non‑empty vector (fallback to a uniform 256‑vector).  The code uses only
NumPy and the provided ``evaluate_sequence`` function.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Basic linear‑algebra utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """
    Linear convolution via real FFT.
    Returns an array of length len(a)+len(b)-1.
    """
    m = a.size + b.size - 1
    n_fft = next_pow2(m)
    A = np.fft.rfft(a, n=n_fft)
    B = np.fft.rfft(b, n=n_fft)
    C = A * B
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


def max_convolution(a: np.ndarray) -> float:
    """Maximum entry of the self‑convolution of a."""
    return float(np.max(convolution_fft(a, a)))


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection onto the probability simplex
    {x ≥ 0, Σx = 1}.  Implementation of Duchi et al. (2008).
    """
    n = v.shape[0]
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / n)

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho = np.where(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def compute_score(a: np.ndarray) -> float:
    """
    Fast surrogate used during the search.
    The vector a is assumed to be normalised (Σa = 1), in which case
    the surrogate equals the official evaluation value.
    """
    n = a.shape[0]
    return 2.0 * n * max_convolution(a)


# ----------------------------------------------------------------------
# Shape generators
# ----------------------------------------------------------------------


def generate_beta_shape(
    n: int, alpha: float, beta: float, eps: float
) -> np.ndarray:
    """
    Symmetric β‑family density on {0,…,n‑1}:
        a_i ∝ (i+ε)^{α‑1}·(n‑1‑i+ε)^{β‑1}.
    Returned vector is normalised to Σa = 1.
    """
    i = np.arange(n, dtype=float)
    left = (i + eps) ** (alpha - 1.0)
    right = (n - 1 - i + eps) ** (beta - 1.0)
    base = left * right
    total = base.sum()
    if total <= 0.0 or not np.isfinite(total):
        return np.full(n, 1.0 / n)
    return base / total


def generate_beta_mixture(
    n: int,
    alpha1: float,
    beta1: float,
    eps1: float,
    alpha2: float,
    beta2: float,
    eps2: float,
    w: float,
) -> np.ndarray:
    """
    Convex combination of two β‑family densities.
    The result is re‑normalised to Σ = 1.
    """
    a1 = generate_beta_shape(n, alpha1, beta1, eps1)
    a2 = generate_beta_shape(n, alpha2, beta2, eps2)
    mix = w * a1 + (1.0 - w) * a2
    mix_sum = mix.sum()
    if mix_sum == 0.0 or not np.isfinite(mix_sum):
        return np.full(n, 1.0 / n)
    return mix / mix_sum


def arcsine_initial(n: int) -> np.ndarray:
    """Classic arcsine shape (symmetrised)."""
    i = np.arange(n, dtype=float)
    eps = 0.25
    vals = 1.0 / np.sqrt((i + 0.5) * (n - 0.5 - i) + eps)
    vals /= vals.sum()
    return vals


def generate_exponential_decay(n: int, rate: float) -> np.ndarray:
    """Monotonically decreasing exponential (geometric) shape."""
    i = np.arange(n, dtype=float)
    base = np.exp(-rate * i)
    total = base.sum()
    if total == 0.0:
        return np.full(n, 1.0 / n)
    return base / total


def generate_powerlaw_shape(n: int, alpha: float, eps: float = 1e-8) -> np.ndarray:
    """Power‑law decay a_i ∝ (i+eps)^{-α}, normalised."""
    i = np.arange(1, n + 1, dtype=float) + eps
    base = i ** (-alpha)
    total = base.sum()
    if total == 0.0:
        return np.full(n, 1.0 / n)
    return base / total


def upsample_sequence(seq: np.ndarray, new_len: int) -> np.ndarray:
    """Linear interpolation of a normalised vector onto a new length."""
    old_x = np.linspace(0.0, 1.0, len(seq))
    new_x = np.linspace(0.0, 1.0, new_len)
    new_vals = np.interp(new_x, old_x, seq)
    new_vals /= new_vals.sum()
    return new_vals


# ----------------------------------------------------------------------
# Smooth surrogate (log‑sum‑exp) and its gradient (used by Adam)
# ----------------------------------------------------------------------


def smooth_gradient(a: np.ndarray, lam: float):
    """
    Gradient of the log‑sum‑exp surrogate of max(conv(a,a)).
    Returns (gradient, surrogate_value).
    """
    n = a.shape[0]
    conv = convolution_fft(a, a)          # length 2n‑1
    max_c = np.max(conv)

    # stable log‑sum‑exp over the convolution entries
    exp_term = np.exp(lam * (conv - max_c))
    sum_exp = np.sum(exp_term)
    w = exp_term / sum_exp                # soft‑max weights
    B = max_c + np.log(sum_exp) / lam     # smooth approximation of the max

    # weighted sum of sub‑gradients:
    # ∂/∂a  Σ_k w_k (a*a)_k = 2 * (a ⋆ w_rev)
    w_rev = w[::-1]
    conv_rev = convolution_fft(a, w_rev)      # length 3n‑2
    L = 2 * n - 1
    weighted_grad = 2.0 * conv_rev[L - 1 - np.arange(n)]

    s = np.sum(a)
    term1 = (2.0 * n / (s ** 2)) * weighted_grad
    term2 = -(4.0 * n * B / (s ** 3)) * np.ones_like(a)
    grad = term1 + term2
    return grad, B


def adam_optimize(
    initial_a: np.ndarray,
    time_deadline: float,
    max_iter: int = 150_000,
    lr0: float = 0.15,
    lam_start: float = 1.0,
    lam_end: float = 200.0,
    noise_every: int = 1000,
    noise_scale: float = 4e-5,
) -> tuple[np.ndarray, float]:
    """
    Adam optimiser on the smooth log‑sum‑exp surrogate.
    Returns the best vector (projected on the simplex) and its fast score.
    """
    a = project_to_simplex(initial_a.astype(float).copy())
    best_a = a.copy()
    best_sc = compute_score(a)

    beta1, beta2 = 0.9, 0.999
    eps_adam = 1e-8
    m = np.zeros_like(a)
    v = np.zeros_like(a)

    n = a.shape[0]

    for it in range(1, max_iter + 1):
        if time.time() >= time_deadline:
            break

        # Quadratic annealing of λ (more aggressive towards the end)
        t = it / max_iter
        lam = lam_start + (lam_end - lam_start) * (t ** 0.5)

        grad, _ = smooth_gradient(a, lam)

        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)

        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        step = lr0 * m_hat / (np.sqrt(v_hat) + eps_adam)
        a_new = a - step

        if noise_every and (it % noise_every == 0):
            a_new += np.random.randn(n) * noise_scale

        a_new = project_to_simplex(a_new)

        cur_sc = compute_score(a_new)
        if cur_sc < best_sc:
            best_sc = cur_sc
            best_a = a_new.copy()

        a = a_new

        if it % 5000 == 0:
            print(f"    Adam it={it:,d} cur={cur_sc:.6f} best={best_sc:.6f}")

    return best_a, best_sc


# ----------------------------------------------------------------------
# Exact sub‑gradient descent with back‑tracking line‑search
# ----------------------------------------------------------------------


def subgradient_optimize(
    initial_a: np.ndarray,
    time_deadline: float,
    step0: float = 0.3,
    max_iter: int = 200_000,
) -> tuple[np.ndarray, float]:
    """
    Projected sub‑gradient descent on the *true* max‑convolution.
    Guarantees descent via a simple back‑tracking line‑search.
    """
    a = initial_a.copy()
    best_a = a.copy()
    best_sc = compute_score(a)

    n = a.shape[0]

    for it in range(1, max_iter + 1):
        if time.time() >= time_deadline:
            break

        conv = convolution_fft(a, a)
        max_val = float(np.max(conv))
        idx = int(np.argmax(conv))

        # gradient of the active quadratic term (pairwise product)
        i = np.arange(n)
        j = idx - i
        mask = (j >= 0) & (j < n)
        grad = np.zeros_like(a)
        grad[mask] = 2.0 * a[j[mask]]

        # back‑tracking line‑search
        step = step0 / np.sqrt(it)
        for _ in range(12):
            a_new = a - step * grad
            a_new = project_to_simplex(a_new)
            new_val = max_convolution(a_new)
            if new_val < max_val - 1e-14:
                break
            step *= 0.5

        cur_sc = compute_score(a_new)
        if cur_sc < best_sc:
            best_sc = cur_sc
            best_a = a_new.copy()

        a = a_new

        if it % 40000 == 0:
            print(f"    Sub‑grad it={it:,d} cur={cur_sc:.6f} best={best_sc:.6f}")

    return best_a, best_sc


# ----------------------------------------------------------------------
# Stochastic local walk (Gaussian perturbations)
# ----------------------------------------------------------------------


def random_local_search(
    initial_a: np.ndarray,
    time_deadline: float,
    init_step: float = 0.015,
    max_iter: int = 250_000,
) -> tuple[np.ndarray, float]:
    """
    Cheap Gaussian walk on the simplex with a tiny uphill‑move probability.
    Returns the best vector seen and its fast score.
    """
    a = initial_a.copy()
    best_a = a.copy()
    best_sc = compute_score(a)

    n = a.shape[0]
    rng = np.random.default_rng()

    for it in range(1, max_iter + 1):
        if time.time() >= time_deadline:
            break

        step = init_step / np.sqrt(it)
        noise = rng.normal(scale=step, size=n)
        a_prop = a + noise
        a_prop = project_to_simplex(a_prop)

        cur_sc = compute_score(a_prop)

        # accept if improvement; otherwise occasional uphill move
        if cur_sc < best_sc or (rng.random() < 1e-4 and cur_sc < 10 * best_sc):
            a = a_prop
            if cur_sc < best_sc:
                best_sc = cur_sc
                best_a = a_prop.copy()

        if it % 50000 == 0:
            print(f"    Random walk it={it:,d} cur={cur_sc:.6f} best={best_sc:.6f}")

    return best_a, best_sc

def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a non‑negative step sequence with a low official score.
    The routine mixes extensive shape sampling, Adam refinement,
    exact sub‑gradient descent, and a final stochastic walk.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5   # small safety margin

    best_seq: list[float] = []
    best_val: float = float("inf")

    # ------------------------------------------------------------------
    # Optional fallback from a previously discovered construction
    # ------------------------------------------------------------------
    try:
        fallback = [float(x) for x in height_sequence_1]   # type: ignore
        val = evaluate_sequence(fallback)
        print(f"Fallback (height_sequence_1) official score = {val:.6f}")
        if val < best_val:
            best_val = val
            best_seq = fallback
    except Exception:
        pass

    # ------------------------------------------------------------------
    # Lengths to explore – powers of two (larger n gives more flexibility)
    # ------------------------------------------------------------------
    lengths = [1024, 2048, 4096, 8192]

    # keep the best vector seen so far for up‑sampling to larger lengths
    global_best_vec: np.ndarray | None = None

    for n in lengths:
        if time.time() >= deadline - 5.0:
            break
        print(f"\n=== Exploring length n = {n} ===")

        # --------------------------------------------------------------
        # 0. warm‑start from the best vector of a previous (smaller) length
        # --------------------------------------------------------------
        if global_best_vec is not None:
            local_best = upsample_sequence(global_best_vec, n)
            local_best_sc = compute_score(local_best)
        else:
            local_best = None
            local_best_sc = float("inf")

        # --------------------------------------------------------------
        # 1. Deterministic starters
        # --------------------------------------------------------------
        starters = [
            np.full(n, 1.0 / n),                     # uniform
            arcsine_initial(n),                      # arcsine
            generate_exponential_decay(n, rate=0.001),  # very shallow expo
            generate_exponential_decay(n, rate=0.01),   # moderate expo
            generate_powerlaw_shape(n, alpha=0.5),      # mild power‑law
            generate_powerlaw_shape(n, alpha=2.0),      # steeper power‑law
        ]
        for st in starters:
            sc = compute_score(st)
            if sc < local_best_sc:
                local_best_sc = sc
                local_best = st.copy()

        # --------------------------------------------------------------
        # 2. Random beta‑family shapes (plain and mixtures)
        # --------------------------------------------------------------
        max_samples = int(4000 * (budget_s / 1000.0))
        for _ in range(max_samples):
            if time.time() >= deadline - 5.0:
                break
            if rng.random() < 0.3:
                # mixture of two beta shapes
                w = rng.random()
                alpha1 = rng.uniform(0.02, 4.0)
                beta1 = rng.uniform(0.02, 4.0)
                eps1 = rng.uniform(1e-8, max(0.01, 0.02 * n))
                alpha2 = rng.uniform(0.02, 4.0)
                beta2 = rng.uniform(0.02, 4.0)
                eps2 = rng.uniform(1e-8, max(0.01, 0.02 * n))
                cand = generate_beta_mixture(
                    n, alpha1, beta1, eps1, alpha2, beta2, eps2, w
                )
            else:
                # simple beta
                alpha = rng.uniform(0.02, 4.0)
                beta = rng.uniform(0.02, 4.0)
                eps = rng.uniform(1e-8, max(0.01, 0.02 * n))
                cand = generate_beta_shape(n, alpha, beta, eps)

            sc = compute_score(cand)
            if sc < local_best_sc:
                local_best_sc = sc
                local_best = cand.copy()

        # --------------------------------------------------------------
        # 3. Additional parametric families (exponential & power‑law)
        # --------------------------------------------------------------
        # Exponential rates (log‑space)
        for rate in np.geomspace(1e-5, 5.0, num=12):
            cand = generate_exponential_decay(n, rate)
            sc = compute_score(cand)
            if sc < local_best_sc:
                local_best_sc = sc
                local_best = cand.copy()

        # Power‑law exponents
        for alpha in np.linspace(0.2, 5.0, num=12):
            cand = generate_powerlaw_shape(n, alpha)

            if sc < local_best_sc:

        # --------------------------------------------------------------

        # --------------------------------------------------------------
        if local_best is not None and time.time() < deadline - 5.0:

            max_iter_adam = int(min(250_000, remaining * 30))

        else:

    # ------------------------------------------------------------------
    # 5. Global polishing of the overall best vector (if time permits)
    # ------------------------------------------------------------------
    if best_seq and time.time() < deadline - 4.0:

        a = np.array(best_seq, dtype=float)

        remaining = deadline - time.time()
        max_iter_adam = int(min(400_000, remaining * 30))
        print(f"  Adam polishing (max_iter={max_iter_adam}) …")
        a, _ = adam_optimize(
            initial_a=a,
            time_deadline=deadline,

            lr0=0.10,

            noise_scale=3e-5,
        )

        # Stage B – exact sub‑gradient descent
        if time.time() < deadline - 3.0:
            max_iter_sub = int(min(200_000, (deadline - time.time()) * 30))
            print(f"  Sub‑gradient polishing (max_iter={max_iter_sub}) …")
            a, _ = subgradient_optimize(
                a,
                time_deadline=deadline,
                step0=0.18,
                max_iter=max_iter_sub,
            )

        # Stage C – stochastic local walk
        if time.time() < deadline - 2.0:

            print(f"  Random walk polishing (max_iter={max_iter_loc}) …")

            )

        if final_val < best_val:

            best_seq = a.tolist()
            print("*** Updated global best after final polish ***")

    # 6. Guard – guarantee a non‑empty return
    # ------------------------------------------------------------------

        best_val = evaluate_sequence(best_seq)
        print("Fallback to uniform 256‑vector.", best_val)

    return best_seq
