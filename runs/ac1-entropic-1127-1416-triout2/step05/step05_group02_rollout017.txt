def adam_optimize(initial_a, deadline, max_iter=50000, lr=0.05, beta1=0.9, beta2=0.999, eps=1e-8, tau=1.0, tau_increase=False):
    a = project_to_simplex(initial_a.copy())
    n = len(a)
    L = 2*n - 1
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    for it in range(1, max_iter+1):
        if time.time() >= deadline:
            break
        conv = convolution_fft(a, a)
        max_val = np.max(conv)
        # compute softmax weights
        if tau_increase:
            tau_cur = tau * min(1.0, it / max_iter)  # or increase linearly
        else:
            tau_cur = tau
        # compute softmax over conv: w_k = exp(tau_cur * conv_k) / sum_k exp(...)
        # to avoid overflow: subtract max conv
        exp_term = np.exp(tau_cur * (conv - max_val))
        w = exp_term / exp_term.sum()
        # compute subgradient: 2 * convolution of a and reversed w
        w_rev = w[::-1]
        conv_rev = convolution_fft(a, w_rev)
        subgrad = 2.0 * conv_rev[L-1 - np.arange(n)]
        # Adam update
        m = beta1 * m + (1 - beta1) * subgrad
        v = beta2 * v + (1 - beta2) * (subgrad ** 2)
        m_hat = m / (1 - beta1 ** it)
        v_hat = v / (1 - beta2 ** it)
        a = a - lr * m_hat / (np.sqrt(v_hat) + eps)
        a = project_to_simplex(a)
        # optionally print progress
    return a
```

"""
Propose a low‑discrepancy step sequence.

The search consists of three phases:

1️⃣  Generate a diverse pool of candidate vectors using several
    parametric families (beta, power‑shape, exponential, mixtures,
    triangular bumps, uniform and arcsine).  A cheap hill‑climb on the
    two main families adds a few locally optimised seeds.

2️⃣  Refine every seed with a long Adam‑style stochastic‑gradient
    descent on the exact objective  
        f(a)=2·n·max(conv(a,a))  (the denominator disappears for Σa=1).  
    The gradient of the smooth “soft‑max” surrogate of the max is
    computed by a single FFT‑based convolution, and the iterate is
    projected onto the probability simplex after each step.

3️⃣  Polishing with exponentiated‑gradient (mirror‑descent) and a final
    global Adam run on the best vector found.

All phases respect a global 1000 s budget.  The routine returns the
best sequence (list of non‑negative floats) it discovered.
"""

import time
import numpy as np
from typing import List, Tuple

# ----------------------------------------------------------------------
# Low‑level utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """
    Linear convolution via real FFT (length = len(a)+len(b)-1).
    """
    m = a.shape[0] + b.shape[0] - 1
    n_fft = next_pow2(m)
    A = np.fft.rfft(a, n=n_fft)
    B = np.fft.rfft(b, n=n_fft)
    C = A * B
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


def compute_score(a: np.ndarray) -> float:
    """
    Fast surrogate for the official score.
    The vector is assumed to be normalised to Σa = 1,
    therefore the denominator disappears and the score is
        2·n·max(conv(a,a)).
    """
    n = a.shape[0]
    conv = convolution_fft(a, a)
    return 2.0 * n * float(np.max(conv))


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection onto the probability simplex
    {x ≥ 0, Σx = 1}.  Duchi et al., 2008.
    """
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / v.size)

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho_arr = u - cssv / np.arange(1, v.size + 1)
    rho = np.where(rho_arr > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w

# Shape generators
# ----------------------------------------------------------------------


def arcsine_initial(n: int) -> np.ndarray:
    """Classic arcsine shape."""
    i = np.arange(n, dtype=float)
    eps = 0.25
    vals = 1.0 / np.sqrt((i + 0.5) * (n - 0.5 - i) + eps)
    vals /= vals.sum()
    return vals


def generate_beta_shape(n: int, alpha: float, beta: float, eps: float) -> np.ndarray:
    """Beta family: (i+ε)^{α‑1}·(n‑1‑i+ε)^{β‑1}."""
    i = np.arange(n, dtype=float)
    left = (i + eps) ** (alpha - 1.0)
    right = (n - 1 - i + eps) ** (beta - 1.0)
    base = left * right
    total = base.sum()
    if total <= 0.0 or not np.isfinite(total):
        return np.full(n, 1.0 / n)
    return base / total


def generate_power_shape(n: int, p: float, eps: float) -> np.ndarray:
    """Generalised arcsine: a_i ∝ ((i+½)*(n‑½‑i)+ε)^{‑p}."""
    i = np.arange(n, dtype=float)
    denom = ((i + 0.5) * (n - 0.5 - i) + eps) ** p
    vals = 1.0 / denom
    total = vals.sum()
    if total <= 0.0 or not np.isfinite(total):
        return np.full(n, 1.0 / n)
    return vals / total


def generate_exponential_shape(n: int, decay: float) -> np.ndarray:
    """Symmetric exponential decay."""
    i = np.arange(n, dtype=float)
    vals = np.exp(-decay * i) + np.exp(-decay * (n - 1 - i))
    total = vals.sum()
    if total <= 0.0 or not np.isfinite(total):
        return np.full(n, 1.0 / n)
    return vals / total


def generate_triangular_shape(n: int, peak: int) -> np.ndarray:
    """Symmetric triangle with apex at `peak`."""
    left = np.arange(peak + 1, dtype=float)
    right = np.arange(n - peak - 1, dtype=float)[::-1]
    seq = np.concatenate([left, right])
    seq /= seq.sum()
    return seq


def generate_mixture_shape(n: int, weight_arc: float) -> np.ndarray:
    """Mixture (weight_arc)·arcsine + (1‑weight_arc)·uniform."""
    arc = arcsine_initial(n)
    uni = np.full(n, 1.0 / n)
    mix = weight_arc * arc + (1.0 - weight_arc) * uni
    mix /= mix.sum()
    return mix

    n: int,

    init_eps: float,
    deadline: float,
    max_iters: int = 300,

    for _ in range(max_iters):

        sc = compute_score(cand)
        if sc < best_sc:
            p, eps, best_seq, best_sc = p_new, eps_new, cand, sc
    return best_seq, best_sc

    init_alpha: float,
    init_beta: float,
    init_eps: float,

    max_iters: int = 300,

    alpha, beta, eps = init_alpha, init_beta, init_eps
    best_seq = generate_beta_shape(n, alpha, beta, eps)

        eps_new = max(0.0, eps + np.random.uniform(-0.1, 0.1))

def adam_optimize(
    initial_a: np.ndarray,
    deadline: float,
    max_iter: int = 80_000,
    lr: float = 0.05,
    beta1: float = 0.9,
    beta2: float = 0.999,
    eps: float = 1e-8,
    tau_start: float = 0.5,
    tau_end: float = 5.0,
) -> np.ndarray:
    """
    Adam optimiser on the smoothed max objective.
    The gradient uses the soft‑max weights
        w_k ∝ exp(tau * conv_k) .
    The step‑size decays implicitly through the increasing temperature
    (`tau`) which focuses the gradient onto the true maximum.
    """
    a = project_to_simplex(initial_a.copy())
    n = a.shape[0]
    L = 2 * n - 1
    m_vec = np.zeros_like(a)
    v_vec = np.zeros_like(a)

    for it in range(1, max_iter + 1):
        if time.time() >= deadline:
            break

        conv = convolution_fft(a, a)                     # length L
        max_val = float(np.max(conv))

        # linear temperature schedule
        tau = tau_start + (tau_end - tau_start) * (it / max_iter)

        # soft‑max weights (stable computation)
        exp_term = np.exp(tau * (conv - max_val))
        w = exp_term / exp_term.sum()                    # length L

        # weighted sub‑gradient  = 2·∑_k w_k·shift(a,k)
        w_rev = w[::-1]
        conv_rev = convolution_fft(a, w_rev)             # length n+L‑1
        subgrad = 2.0 * conv_rev[L - 1 - np.arange(n)]

        # Adam update
        m_vec = beta1 * m_vec + (1 - beta1) * subgrad
        v_vec = beta2 * v_vec + (1 - beta2) * (subgrad ** 2)

        m_hat = m_vec / (1 - beta1 ** it)
        v_hat = v_vec / (1 - beta2 ** it)

        a = a - lr * m_hat / (np.sqrt(v_hat) + eps)

        a = project_to_simplex(a)

        if it % 5000 == 0:
            cur_sc = compute_score(a)
            print(f"    Adam it={it:,d}  score={cur_sc:.6f}")

    return a


def eg_refine(
    a: np.ndarray,
    deadline: float,
    max_iter: int = 100_000,
    eta0: float = 0.3,
) -> np.ndarray:
    """
    Exponentiated‑gradient polishing (mirror‑descent on the simplex).
    Keeps the iterate non‑negative automatically.
    """
    n = a.shape[0]
    L = 2 * n - 1

    for it in range(1, max_iter + 1):
        if time.time() >= deadline:
            break

        conv = convolution_fft(a, a)
        max_val = float(np.max(conv))
        tol = max_val * 1e-12 + 1e-14
        mask = (np.abs(conv - max_val) <= tol).astype(float)

        # sub‑gradient for the true max (same formula as in the original code)
        mask_rev = mask[::-1]
        conv_rev = convolution_fft(a, mask_rev)
        subgrad = 2.0 * conv_rev[L - 1 - np.arange(n)]

        eta = eta0 / (np.sqrt(it) + 1.0)
        a = a * np.exp(-eta * subgrad)

        total = a.sum()
        if total == 0.0:
            a = np.full_like(a, 1.0 / n)
        else:
            a = a / total

        if it % 5000 == 0:
            cur_sc = compute_score(a)
            print(f"    EG   it={it:,d}  score={cur_sc:.6f}")

    return a

        pool.append((sc, cand))

    # ------------------------------------------------------------------
    # Keep the most promising seeds
    # ------------------------------------------------------------------
    pool.sort(key=lambda x: x[0])
    top_seeds = [vec for _, vec in pool[:12]]
    print(f"\nSelected {len(top_seeds)} top seeds for refinement "
          f"(score range {pool[0][0]:.6f} – {pool[-1][0]:.6f})")

    # ------------------------------------------------------------------
    # 2.  Refine each seed with Adam + optional EG polish
    # ------------------------------------------------------------------
    for idx, seed_vec in enumerate(top_seeds, start=1):
        if time.time() >= deadline - 5.0:
            break
        n = seed_vec.shape[0]
        print(f"\nRefining seed {idx}/{len(top_seeds)} (n={n}) …")

        remaining = deadline - time.time()
        # allocate up to ~60 k Adam iterations per seed (adjust to remaining time)
        max_it = int(min(80_000, remaining * 30))
        refined = adam_optimize(
            seed_vec,
            deadline,
            max_iter=max_it,
            lr=0.08,
            tau_start=0.5,
            tau_end=6.0,
        )

        # Light EG polishing (helps escape tiny plateaus)
        if time.time() < deadline - 2.0:
            refined = eg_refine(refined, deadline, max_iter=20000, eta0=0.2)

        # Official evaluation
        try:
            val = evaluate_sequence(refined.tolist())
            print(f"  → official score = {val:.6f}")
            if val < best_val:
                best_val = val
                best_seq = refined.tolist()
                print(f"  *** New global best: {best_val:.6f} ***")
        except Exception as e:
            print(f"  Evaluation failed: {e}")

    # ------------------------------------------------------------------
    # 3.  Global polish on the current best (if time allows)
    # ------------------------------------------------------------------
    if best_seq and time.time() < deadline - 3.0:
        print("\nFinal global polish …")
        a = np.array(best_seq, dtype=float)

        remaining = deadline - time.time()
        max_it = int(min(200_000, remaining * 30))
        a = adam_optimize(
            a,
            deadline,
            max_iter=max_it,
            lr=0.04,
            tau_start=0.5,
            tau_end=8.0,
        )

        if time.time() < deadline - 2.0:
            a = eg_refine(a, deadline, max_iter=50000, eta0=0.15)

        final_val = evaluate_sequence(a.tolist())
        print(f"Final polished official score = {final_val:.6f}")
        if final_val < best_val:
            best_val = final_val
            best_seq = a.tolist()
            print("*** Updated global best after final polish ***")

    # ------------------------------------------------------------------
    # 4.  Guard – ensure a non‑empty return value
    # ------------------------------------------------------------------
    if not best_seq:
        fallback_n = 256
        best_seq = [1.0 / fallback_n] * fallback_n
        best_val = evaluate_sequence(best_seq)

    print(f"\nSearch finished. Best reported official score = {best_val:.6f}")
    return best_seq
