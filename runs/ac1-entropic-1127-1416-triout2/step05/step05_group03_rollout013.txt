def l_bfgs_optimize(initial_a, time_deadline, lam=200.0, max_iter=2000):
    n = len(initial_a)
    # Ensure positivity
    eps = 1e-12
    x0 = np.log(initial_a + eps)

    def fun_and_grad(x):
        # compute a via softmax
        max_x = np.max(x)
        exp_x = np.exp(x - max_x)
        a = exp_x / exp_x.sum()
        grad_a, B = smooth_gradient(a, lam)
        # construct objective f = 2 * n * B / (s**2) (s = 1)
        f = 2.0 * n * B
        # chain rule for gradient w.r.t x
        dot = np.dot(a, grad_a)
        grad_x = a * (grad_a - dot)
        return f, grad_x

    # Use SciPy minimize
    from scipy.optimize import minimize
    res = minimize(fun=lambda x: fun_and_grad(x)[0],
                   x0=x0,
                   jac=lambda x: fun_and_grad(x)[1],
                   method='L-BFGS-B',
                   options={'maxiter': max_iter, 'disp': False})
    # get final a
    max_x = np.max(res.x)
    exp_x = np.exp(res.x - max_x)
    a_opt = exp_x / exp_x.sum()
    return a_opt, compute_score(a_opt)

def propose_candidate(seed=42, budget_s=1000, **kwargs):
    - rng = np.random.default_rng(seed)
    - start_time = time.time()
    - deadline = start_time + budget_s - 2

    best_seq = None; best_val = np.inf

    # fallback: try height_sequence_1 if exists
    try:
        fallback = [float(x) for x in height_sequence_1]
        val = evaluate_sequence(fallback)
        if val < best_val:
            best_val = val; best_seq = fallback
    except Exception:
        pass

    # set lengths to try based on time left
    candidate_lengths = [2048, 4096, 8192]  # prefer longer
    for n in candidate_lengths:
        now = time.time()
        if now >= deadline - 30: break

        print(f"\n=== Working on length {n} ===")
        # Phase 1: sampling
        best_local = None
        best_local_sc = np.inf

        # deterministic starters
        starters = [np.full(n, 1.0/n), arcsine_initial(n)]
        for st in starters:
            sc = compute_score(st)
            if sc < best_local_sc:
                best_local_sc = sc; best_local = st

        # number of random candidates based on remaining budget
        remaining = deadline - time.time()
        # allocate a proportion of random draws: say up to 6000 for each length
        max_random = min(8000, int(remaining * 4))  # rough scaling
        for _ in range(max_random):
            if time.time() >= deadline - 30: break
            if rng.random() < 0.25:
                # mixture
                w = rng.random()
                alpha1 = rng.uniform(0.01, 5.0)
                beta1 = rng.uniform(0.01, 5.0)
                eps1 = rng.uniform(1e-9, 0.01 * n)
                alpha2 = rng.uniform(0.01, 5.0)
                beta2 = rng.uniform(0.01, 5.0)
                eps2 = rng.uniform(1e-9, 0.01 * n)
                cand = generate_beta_mixture(n, alpha1, beta1, eps1, alpha2, beta2, eps2, w)
            else:
                # single beta
                alpha = rng.uniform(0.01, 5.0)
                beta = rng.uniform(0.01, 5.0)
                eps = rng.uniform(1e-9, 0.01 * n)
                cand = generate_beta_shape(n, alpha, beta, eps)
            # optionally also exponential shape
            if rng.random() < 0.1:
                decay = rng.uniform(0.0005, 0.01)
                i = np.arange(n)
                cand = np.exp(-decay * i)
                cand = cand / cand.sum()
            sc = compute_score(cand)
            if sc < best_local_sc:
                best_local_sc = sc; best_local = cand

        # if no candidate (unlikely), continue
        if best_local is None:
            continue

        # Phase 2: refine top candidates
        # produce a list of top candidates including best_local and a few random others
        top_candidates = [best_local]
        # create additional candidates by perturbing best_local
        for _ in range(2):
            pert = best_local + rng.normal(scale=0.01, size=n)
            pert = np.maximum(pert, 0)
            pert = pert / pert.sum()
            top_candidates.append(pert)

        # refine each with L-BFGS
        refined_candidates = []
        for idx, cand in enumerate(top_candidates):
            if time.time() >= deadline - 20: break
            # run L-BFGS at increasing lambda schedule
            a_ref = cand
            for lam in [50.0, 200.0, 500.0]:
                if time.time() >= deadline - 20: break
                a_ref, _ = l_bfgs_optimize(a_ref, deadline, lam=lam, max_iter=300)
                # optional subgradient after each lam
                a_ref, _ = subgradient_optimize(a_ref, deadline, step0=0.2, max_iter=5000)
            refined_candidates.append(a_ref)

        # Evaluate refined candidates officially
        for cand in refined_candidates:
            if time.time() >= deadline - 5: break
            val = evaluate_sequence(cand.tolist())
            print(f"Length {n} candidate official score = {val:.6f}")
            if val < best_val:
                best_val = val
                best_seq = cand.tolist()
                print(f"*** New global best = {best_val:.6f} ***")

        # also evaluate best_local as fallback
        val = evaluate_sequence(best_local.tolist())
        if val < best_val:
            best_val = val; best_seq = best_local.tolist()
            print(f"*** Updated best from raw local (len={n}) = {best_val:.6f} ***")

    # final global polishing if time allows
    if best_seq is not None and time.time() < deadline - 5:
        print("\n=== Global polishing ===")
        a = np.array(best_seq, dtype=float)
        # longer L-BFGS runs with higher lambda
        for lam in [200.0, 500.0, 800.0]:
            if time.time() >= deadline - 5: break
            a, _ = l_bfgs_optimize(a, deadline, lam=lam, max_iter=800)
            # subgradient refinement
            a, _ = subgradient_optimize(a, deadline, step0=0.15, max_iter=50000)
        final_val = evaluate_sequence(a.tolist())
        print(f"Final polished official score = {final_val:.6f}")
        if final_val < best_val:
            best_val = final_val
            best_seq = a.tolist()

    # guarantee non-empty
    if not best_seq:
        best_seq = [1.0] * 256
        best_val = evaluate_sequence(best_seq)
        print("Using uniform fallback, score:", best_val)

    print(f"\nSearch finished. Best official score = {best_val:.6f}")
    return best_seq

"""
step‑function optimizer
-----------------------

The goal is to minimise

        2·n·max(conv(a,a)) / (Σa)²

for a non‑negative vector `a`.  The optimiser works in three stages:

1. **Coarse sampling (n≈1024).**  
   A large number of beta‑family and mixture shapes are generated.
   The best candidate according to the fast surrogate
   `2·n·max(conv(a,a))` (sum normalised to 1) is kept.

2. **Progressive refinement.**  
   The best shape is up‑sampled to longer lengths (2048, 4096) and
   refined with a smooth log‑sum‑exp surrogate of the max‑convolution.
   The surrogate is optimised by Adam, with an annealed temperature
   that ends at a large λ (≈800) so the surrogate is almost the true
   max.  After each Adam phase a few exact sub‑gradient steps are
   performed to guarantee monotone improvement.

3. **Global polishing.**  
   The overall best vector is polished again with a long Adam run
   (high λ) followed by a sub‑gradient descent and a cheap stochastic
   walk.  All stages respect the 1000 s budget; the routine always
   returns a non‑empty sequence.

The code uses only NumPy and SciPy (for `minimize`).  All helper
functions are top‑level and avoid closures or lambda expressions.
"""

import time
import numpy as np
from typing import List, Tuple
from scipy.optimize import minimize

# ----------------------------------------------------------------------
# Basic linear‑algebra utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """
    Linear convolution using a real FFT.
    Returns an array of length len(a)+len(b)-1.
    """
    m = a.size + b.size - 1
    n_fft = next_pow2(m)
    A = np.fft.rfft(a, n=n_fft)
    B = np.fft.rfft(b, n=n_fft)
    C = A * B
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


def max_convolution(a: np.ndarray) -> float:
    """Maximum entry of the self‑convolution of a."""
    return float(np.max(convolution_fft(a, a)))


def compute_score(a: np.ndarray) -> float:
    """
    Fast surrogate: 2·n·max(conv(a,a)).
    The vector is assumed to be normalised to Σa = 1,
    so this equals the official evaluation value.
    """
    n = a.shape[0]
    return 2.0 * n * max_convolution(a)


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection onto the probability simplex
    {x ≥ 0, Σx = 1}.  Implementation of Duchi et al. (2008).
    """
    n = v.shape[0]
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / n)

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho = np.where(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Shape generators
# ----------------------------------------------------------------------


def generate_beta_shape(n: int, alpha: float, beta: float, eps: float) -> np.ndarray:
    """
    Symmetric beta‑family density on {0,…,n‑1}:
        a_i ∝ (i+eps)^{α‑1}·(n‑1‑i+eps)^{β‑1}.
    Returned vector is normalised to Σa = 1.
    """
    i = np.arange(n, dtype=float)
    left = (i + eps) ** (alpha - 1.0)
    right = (n - 1 - i + eps) ** (beta - 1.0)
    base = left * right
    total = base.sum()
    if total <= 0.0 or not np.isfinite(total):
        return np.full(n, 1.0 / n)
    return base / total


def generate_beta_mixture(
    n: int,
    alpha1: float,
    beta1: float,
    eps1: float,
    alpha2: float,
    beta2: float,
    eps2: float,
    w: float,
) -> np.ndarray:
    """
    Convex combination of two beta‑family densities.
    The result is renormalised to Σa = 1.
    """
    a1 = generate_beta_shape(n, alpha1, beta1, eps1)
    a2 = generate_beta_shape(n, alpha2, beta2, eps2)
    mix = w * a1 + (1.0 - w) * a2
    mix_sum = mix.sum()
    if mix_sum == 0.0 or not np.isfinite(mix_sum):
        return np.full(n, 1.0 / n)
    return mix / mix_sum


def arcsine_initial(n: int) -> np.ndarray:
    """Classic arcsine shape (symmetrised)."""
    i = np.arange(n, dtype=float)
    eps = 0.25
    vals = 1.0 / np.sqrt((i + 0.5) * (n - 0.5 - i) + eps)
    vals /= vals.sum()
    return vals


def upsample_sequence(a: np.ndarray, n_target: int) -> np.ndarray:
    """
    Linear interpolation from length len(a) to n_target,
    followed by re‑normalisation to Σ = 1.
    """
    n_src = a.shape[0]
    x_src = np.linspace(0.0, 1.0, n_src)
    x_tgt = np.linspace(0.0, 1.0, n_target)
    a_tgt = np.interp(x_tgt, x_src, a)
    a_tgt /= a_tgt.sum()
    return a_tgt


# ----------------------------------------------------------------------
# Smooth surrogate (log‑sum‑exp) and its gradient
# ----------------------------------------------------------------------


def smooth_gradient(a: np.ndarray, lam: float) -> Tuple[np.ndarray, float]:
    """
    Gradient of the log‑sum‑exp surrogate of max(conv(a,a)).
    Returns (gradient, surrogate value B).
    The returned gradient corresponds to the objective
        f(a) = 2·n·B / (Σa)² .
    """
    n = a.shape[0]
    conv = convolution_fft(a, a)                     # length 2n‑1
    max_c = np.max(conv)

    # stable log‑sum‑exp
    exp_term = np.exp(lam * (conv - max_c))
    sum_exp = np.sum(exp_term)
    w = exp_term / sum_exp                            # soft‑max weights
    B = max_c + np.log(sum_exp) / lam                 # smoothed max

    # weighted sum of sub‑gradients:
    # ∂/∂a  Σ_k w_k (a*a)_k = 2 * (a ⋆ w_rev)
    w_rev = w[::-1]
    conv_rev = convolution_fft(a, w_rev)            # length 3n‑2
    L = 2 * n - 1
    weighted_grad = 2.0 * conv_rev[L - 1 - np.arange(n)]

    s = np.sum(a)
    term1 = (2.0 * n / (s ** 2)) * weighted_grad
    term2 = -(4.0 * n * B / (s ** 3)) * np.ones_like(a)
    grad = term1 + term2
    return grad, B


# ----------------------------------------------------------------------
# Adam optimisation on the smooth surrogate
# ----------------------------------------------------------------------


def adam_optimize(
    initial_a: np.ndarray,
    time_deadline: float,
    max_iter: int = 100_000,
    lr0: float = 0.12,
    lam_start: float = 1.0,
    lam_end: float = 400.0,
    noise_every: int = 800,
    noise_scale: float = 5e-5,
) -> Tuple[np.ndarray, float]:
    """
    Adam optimiser for the smoothed objective.
    Returns the best vector seen (projected onto the simplex) and its
    fast surrogate value.
    """
    a = project_to_simplex(initial_a.astype(float).copy())
    n = a.shape[0]

    best_a = a.copy()
    best_sc = compute_score(a)

    beta1, beta2 = 0.9, 0.999
    eps_adam = 1e-8
    m = np.zeros_like(a)
    v = np.zeros_like(a)

    for it in range(1, max_iter + 1):
        if time.time() >= time_deadline:
            break

        # quadratic temperature schedule
        t = it / max_iter
        lam = lam_start + (lam_end - lam_start) * (t ** 0.5)

        grad, _ = smooth_gradient(a, lam)

        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)

        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        step = lr0 * m_hat / (np.sqrt(v_hat) + eps_adam)
        a_new = a - step

        if noise_every and (it % noise_every == 0):
            a_new += np.random.randn(n) * noise_scale

        a_new = project_to_simplex(a_new)

        cur_sc = compute_score(a_new)
        if cur_sc < best_sc:
            best_sc = cur_sc
            best_a = a_new.copy()

        a = a_new

        if it % 5000 == 0:
            print(
                f"    Adam iter={it:,d} cur={cur_sc:.6f} best={best_sc:.6f}"
            )

    return best_a, best_sc


# ----------------------------------------------------------------------
# Exact sub‑gradient descent with back‑tracking line‑search
# ----------------------------------------------------------------------


def subgradient_optimize(
    initial_a: np.ndarray,
    time_deadline: float,
    step0: float = 0.25,
    max_iter: int = 60_000,
) -> Tuple[np.ndarray, float]:
    """
    Projected sub‑gradient descent on the true max‑convolution.
    Returns the best vector seen (projected) and its fast surrogate
    value.
    """
    a = initial_a.copy()
    n = a.shape[0]

    best_a = a.copy()
    best_sc = compute_score(a)

    for it in range(1, max_iter + 1):
        if time.time() >= time_deadline:
            break

        conv = convolution_fft(a, a)
        max_val = float(np.max(conv))
        idx = int(np.argmax(conv))

        # gradient of the active quadratic term
        i = np.arange(n)
        j = idx - i
        mask = (j >= 0) & (j < n)
        grad = np.zeros_like(a)
        grad[mask] = 2.0 * a[j[mask]]

        step = step0 / np.sqrt(it)
        for _ in range(12):
            a_new = a - step * grad
            a_new = project_to_simplex(a_new)
            new_val = max_convolution(a_new)
            if new_val < max_val - 1e-14:
                break
            step *= 0.5

        cur_sc = compute_score(a_new)
        if cur_sc < best_sc:
            best_sc = cur_sc
            best_a = a_new.copy()

        a = a_new

        if it % 15000 == 0:
            print(
                f"    Sub‑grad iter={it:,d} cur={cur_sc:.6f} best={best_sc:.6f}"
            )

    return best_a, best_sc


# ----------------------------------------------------------------------
# Stochastic local walk (Gaussian perturbations)
# ----------------------------------------------------------------------


def random_local_search(
    initial_a: np.ndarray,
    time_deadline: float,
    init_step: float = 0.018,
    max_iter: int = 150_000,
) -> Tuple[np.ndarray, float]:
    """
    Cheap Gaussian walk on the simplex with a tiny uphill‑move
    probability.  Returns the best vector seen and its fast surrogate.
    """
    a = initial_a.copy()
    n = a.shape[0]
    rng = np.random.default_rng()

    best_a = a.copy()
    best_sc = compute_score(a)

    for it in range(1, max_iter + 1):
        if time.time() >= time_deadline:
            break

        step = init_step / np.sqrt(it)
        noise = rng.normal(scale=step, size=n)
        a_prop = a + noise
        a_prop = project_to_simplex(a_prop)

        cur_sc = compute_score(a_prop)

        # accept if improvement; otherwise accept a rare uphill move
        if cur_sc < best_sc or (rng.random() < 1e-4 and cur_sc < 10 * best_sc):
            a = a_prop
            if cur_sc < best_sc:
                best_sc = cur_sc
                best_a = a_prop.copy()

        if it % 50000 == 0:
            print(
                f"    Random walk iter={it:,d} cur={cur_sc:.6f} best={best_sc:.6f}"
            )

    return best_a, best_sc
