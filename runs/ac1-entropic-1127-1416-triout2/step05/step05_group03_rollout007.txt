"""
Propose a low‑score step‑function.

Main ideas
----------

* The objective after the internal clipping is   2·n·max(conv(a,a)),
  where the vector `a` is normalized to Σa = 1.
  Scaling does not change the value – we only have to optimise the *shape*.

* The search works in three stages
    1. **Massive random sampling** of simple beta‑family shapes (and a few
       mixtures).  This cheap “global” phase quickly finds a decent starting
       point for a given length.
    2. **Gradient‑based polishing**  
       – Adam on a smooth log‑sum‑exp surrogate (λ annealed up to a large
         value, e.g. 800).  The update is always projected onto the simplex.  
       – An optional L‑BFGS‑B call (soft‑max re‑parameterisation) that uses the
         same smooth surrogate; it often pushes the solution a little further.  
       – Exact sub‑gradient descent on the true max‑convolution with a simple
         back‑tracking line‑search.
    3. **Stochastic walk** – tiny Gaussian perturbations on the simplex with a
       slight uphill‑move probability, to escape shallow basins.

* The whole routine is *time‑aware*: every optimisation routine receives the
  global deadline and stops automatically when the budget (≈1000 s) is exhausted.
  The algorithm first concentrates on length 2048 (the length that gave the
  current best bound) and, if time permits, repeats the whole pipeline for a
  longer vector (4096) – longer vectors sometimes allow a lower constant.

* All operations are performed with FFT‑based convolution (`np.fft.rfft`), which
  is O(n log n) and fast enough for the dimensions we use.

The function `propose_candidate` returns the best vector (list of floats) found
before the deadline.  If nothing is found a safe uniform fallback is returned.
"""

import time
import numpy as np
from typing import List, Tuple

# ----------------------------------------------------------------------
# Linear‑algebra utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """
    Linear convolution via real FFT.
    Returns an array of length len(a)+len(b)-1.
    """
    m = a.size + b.size - 1
    n_fft = next_pow2(m)
    A = np.fft.rfft(a, n=n_fft)
    B = np.fft.rfft(b, n=n_fft)
    C = A * B
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


def max_convolution(a: np.ndarray) -> float:
    """Maximum entry of the self‑convolution of a."""
    return float(np.max(convolution_fft(a, a)))


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of `v` onto the probability simplex {x≥0, Σx=1}.
    Implementation of Duchi et al. (2008).
    """
    n = v.shape[0]
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / n)

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho = np.where(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def compute_score(a: np.ndarray) -> float:
    """
    Fast surrogate: 2·n·max(conv(a,a)).  For a normalized vector (Σa = 1) this
    equals the official evaluation value.
    """
    n = a.shape[0]
    return 2.0 * n * max_convolution(a)


# ----------------------------------------------------------------------
# Shape generators
# ----------------------------------------------------------------------


def generate_beta_shape(
    n: int, alpha: float, beta: float, eps: float
) -> np.ndarray:
    """
    Symmetric beta‑family density on {0,…,n‑1}:
        a_i ∝ (i+ε)^{α‑1}·(n‑1‑i+ε)^{β‑1}.
    The vector is normalised to Σa = 1.
    """
    i = np.arange(n, dtype=float)
    left = (i + eps) ** (alpha - 1.0)
    right = (n - 1 - i + eps) ** (beta - 1.0)
    base = left * right
    total = base.sum()
    if total <= 0.0 or not np.isfinite(total):
        return np.full(n, 1.0 / n)
    return base / total


def generate_beta_mixture(
    n: int,
    alpha1: float,
    beta1: float,
    eps1: float,
    alpha2: float,
    beta2: float,
    eps2: float,
    w: float,
) -> np.ndarray:
    """
    Convex combination of two beta‑family densities, re‑normalised to Σ=1.
    """
    a1 = generate_beta_shape(n, alpha1, beta1, eps1)
    a2 = generate_beta_shape(n, alpha2, beta2, eps2)
    mix = w * a1 + (1.0 - w) * a2
    ms = mix.sum()
    if ms == 0.0 or not np.isfinite(ms):
        return np.full(n, 1.0 / n)
    return mix / ms


def arcsine_initial(n: int) -> np.ndarray:
    """Classic arcsine shape (symmetrised)."""
    i = np.arange(n, dtype=float)
    eps = 0.25
    vals = 1.0 / np.sqrt((i + 0.5) * (n - 0.5 - i) + eps)
    vals /= vals.sum()
    return vals


# ----------------------------------------------------------------------
# Smooth surrogate utilities (used by Adam and optional L‑BFGS)
# ----------------------------------------------------------------------


def smooth_gradient(a: np.ndarray, lam: float) -> Tuple[np.ndarray, float]:
    """
    Gradient of the log‑sum‑exp surrogate of max(conv(a,a)).
    Returns (gradient_of_objective, smooth_max_value).
    The gradient already incorporates the outer factor 2·n.
    """
    n = a.shape[0]
    conv = convolution_fft(a, a)               # length 2n‑1
    max_c = np.max(conv)

    # stable log‑sum‑exp
    exp_term = np.exp(lam * (conv - max_c))
    sum_exp = np.sum(exp_term)
    w = exp_term / sum_exp                     # soft‑max over convolution entries
    B = max_c + np.log(sum_exp) / lam          # smooth approximation of the max

    # weighted sum of sub‑gradients:
    # ∂/∂a  Σ_k w_k (a*a)_k = 2 * (a ⋆ w_rev)
    w_rev = w[::-1]
    conv_rev = convolution_fft(a, w_rev)       # length 3n‑2
    L = 2 * n - 1
    weighted_grad = 2.0 * conv_rev[L - 1 - np.arange(n)]

    s = np.sum(a)
    term1 = (2.0 * n / (s ** 2)) * weighted_grad
    term2 = -(4.0 * n * B / (s ** 3)) * np.ones_like(a)
    grad = term1 + term2
    return grad, B


# ----------------------------------------------------------------------
# Adam optimiser on the smooth surrogate
# ----------------------------------------------------------------------


def adam_optimize(
    initial_a: np.ndarray,
    time_deadline: float,
    max_iter: int = 150_000,
    lr0: float = 0.12,
    lam_start: float = 1.0,
    lam_end: float = 800.0,
    noise_every: int = 1200,
    noise_scale: float = 4e-5,
) -> Tuple[np.ndarray, float]:
    """
    Adam optimiser on the log‑sum‑exp surrogate of max(conv).
    Returns the best vector (projected on the simplex) and its fast surrogate
    score (2·n·smooth_max).
    """
    a = project_to_simplex(initial_a.astype(float).copy())
    best_a = a.copy()
    best_sc = compute_score(a)

    beta1, beta2 = 0.9, 0.999
    eps_adam = 1e-8
    m = np.zeros_like(a)
    v = np.zeros_like(a)

    n = a.shape[0]

    for it in range(1, max_iter + 1):
        if time.time() >= time_deadline:
            break

        # annealed temperature (quadratic‑ish schedule)
        t = it / max_iter
        lam = lam_start + (lam_end - lam_start) * (t ** 0.5)

        grad, _ = smooth_gradient(a, lam)

        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)

        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        step = lr0 * m_hat / (np.sqrt(v_hat) + eps_adam)
        a_new = a - step

        if noise_every and (it % noise_every == 0):
            a_new += np.random.randn(n) * noise_scale

        a_new = project_to_simplex(a_new)

        cur_sc = compute_score(a_new)
        if cur_sc < best_sc:
            best_sc = cur_sc
            best_a = a_new.copy()

        a = a_new

        if it % 5000 == 0:
            print(
                f"    Adam iter={it:,d} cur={cur_sc:.6f} best={best_sc:.6f} λ={lam:.1f}"
            )
    return best_a, best_sc


# ----------------------------------------------------------------------
# Exact sub‑gradient descent (true max) with back‑tracking line search
# ----------------------------------------------------------------------


def subgradient_optimize(
    initial_a: np.ndarray,
    time_deadline: float,
    step0: float = 0.25,
    max_iter: int = 150_000,
) -> Tuple[np.ndarray, float]:
    """
    Projected sub‑gradient descent on the *true* max‑convolution.
    Returns the best vector (projected) and its fast surrogate score.
    """
    a = initial_a.copy()
    best_a = a.copy()
    best_sc = compute_score(a)

    n = a.shape[0]

    for it in range(1, max_iter + 1):
        if time.time() >= time_deadline:
            break

        conv = convolution_fft(a, a)
        max_val = float(np.max(conv))
        idx = int(np.argmax(conv))

        # sub‑gradient of the active quadratic term
        i = np.arange(n)
        j = idx - i
        mask = (j >= 0) & (j < n)
        grad = np.zeros_like(a)
        grad[mask] = 2.0 * a[j[mask]]

        step = step0 / np.sqrt(it)
        # simple back‑tracking
        for _ in range(12):
            a_new = a - step * grad
            a_new = project_to_simplex(a_new)
            new_val = max_convolution(a_new)
            if new_val < max_val - 1e-15:
                break
            step *= 0.5

        cur_sc = compute_score(a_new)
        if cur_sc < best_sc:
            best_sc = cur_sc
            best_a = a_new.copy()

        a = a_new

        if it % 40000 == 0:
            print(f"    Sub‑grad iter={it:,d} cur={cur_sc:.6f} best={best_sc:.6f}")

    return best_a, best_sc


# ----------------------------------------------------------------------
# Stochastic local walk (tiny Gaussian perturbations)
# ----------------------------------------------------------------------


def random_local_search(
    initial_a: np.ndarray,
    time_deadline: float,
    init_step: float = 0.018,
    max_iter: int = 200_000,
) -> Tuple[np.ndarray, float]:
    """
    Gaussian walk on the simplex with a tiny uphill‑move probability.
    Returns the best vector seen and its fast surrogate score.
    """
    a = initial_a.copy()
    best_a = a.copy()
    best_sc = compute_score(a)

    n = a.shape[0]
    rng = np.random.default_rng()

    for it in range(1, max_iter + 1):
        if time.time() >= time_deadline:
            break

        step = init_step / np.sqrt(it)
        noise = rng.normal(scale=step, size=n)
        a_prop = a + noise
        a_prop = project_to_simplex(a_prop)

        cur_sc = compute_score(a_prop)

        # Accept improvement, otherwise accept with tiny probability
        if cur_sc < best_sc or (rng.random() < 1e-4 and cur_sc < 10 * best_sc):
            a = a_prop
            if cur_sc < best_sc:
                best_sc = cur_sc
                best_a = a_prop.copy()

        if it % 50000 == 0:
            print(
                f"    Random walk iter={it:,d} cur={cur_sc:.6f} best={best_sc:.6f}"
            )
    return best_a, best_sc


# ----------------------------------------------------------------------
# Optional L‑BFGS‑B refinement (soft‑max re‑parameterisation)
# ----------------------------------------------------------------------


def lbfgs_optimize(
    initial_a: np.ndarray,
    time_deadline: float,
    lam: float = 800.0,
    maxfun: int = 5000,
) -> Tuple[np.ndarray, float]:
    """
    L‑BFGS‑B on the smooth surrogate (log‑sum‑exp) using a soft‑max
    representation a_i = exp(x_i) / Σ exp(x).  Returns the best vector and its
    fast surrogate score.  If SciPy is unavailable the routine is silently
    skipped.
    """
    try:
        from scipy.optimize import minimize
    except Exception:
        return initial_a, compute_score(initial_a)

    # initialise x = log(a) (shifted to keep numbers moderate)
    x0 = np.log(initial_a + 1e-14)
    x0 = x0 - np.mean(x0)

    n = len(initial_a)

    def fun_and_grad(x):
        if time.time() >= time_deadline:
            # raise a custom exception that will be caught outside
            raise RuntimeError("Time limit hit")

        a = np.exp(x)
        a_sum = a.sum()
        a = a / a_sum

        grad_a, B = smooth_gradient(a, lam)   # grad of 2·n·B, B = smooth max
        # chain rule for soft‑max
        dot = np.dot(a, grad_a)
        grad_x = a * (grad_a - dot)

        val = 2.0 * n * B
        return val, grad_x.astype(np.float64)

    # SciPy's minimize expects a scalar return; we wrap the exception.
    options = {"maxfun": maxfun, "disp": False}
    try:
        res = minimize(
            fun_and_grad,
            x0,
            method="L-BFGS-B",
            jac=True,
            options=options,
        )
    except RuntimeError:
        # time limit – fall back to the initial point
        return initial_a, compute_score(initial_a)

    x_opt = res.x
    a_opt = np.exp(x_opt)
    a_opt = a_opt / a_opt.sum()
    return a_opt, compute_score(a_opt)


# ----------------------------------------------------------------------
# Main entry point required by the competition harness
# ----------------------------------------------------------------------


def propose_candidate(
    seed: int = 42,
    budget_s: int = 1000,
    **kwargs,
) -> List[float]:
    """
    Search for a non‑negative step sequence with a low official score.
    The routine performs:

    1. Random beta‑family sampling (including mixtures) for a chosen length.
    2. Long Adam optimisation on a smooth surrogate (λ up to 800).
    3. Optional L‑BFGS‑B refinement.
    4. Exact sub‑gradient descent on the true max‑convolution.
    5. A final stochastic Gaussian walk.

    If time permits the whole pipeline is repeated for a longer length
    (4096).  The best vector seen (according to evaluate_sequence) is returned.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 2.0   # safety margin

    best_seq: List[float] = []
    best_val: float = float("inf")

    # ------------------------------------------------------------------
    # 0.  Optional fallback from a previously known construction
    # ------------------------------------------------------------------
    try:
        fallback = [float(x) for x in height_sequence_1]  # type: ignore
        val = evaluate_sequence(fallback)
        print(f"Fallback (height_sequence_1) official score = {val:.6f}")
        if val < best_val:
            best_val = val
            best_seq = fallback
    except Exception:
        pass

    # ------------------------------------------------------------------
    # Helper: evaluate a candidate and update the global best
    # ------------------------------------------------------------------
    def maybe_update(candidate: np.ndarray, descr: str):
        nonlocal best_seq, best_val
        val = evaluate_sequence(candidate.tolist())
        print(f"    [{descr}] official score = {val:.6f}")
        if val < best_val:
            best_val = val
            best_seq = candidate.tolist()
            print(f"    *** New global best (= {best_val:.6f}) ***")

    # ------------------------------------------------------------------
    # Core optimisation pipeline for a given length
    # ------------------------------------------------------------------
    def optimise_length(n: int):
        nonlocal deadline
        print(f"\n=== Optimising length n = {n} ===")
        # ---------- 1. Exploratory sampling ----------
        local_best = None
        local_best_sc = float("inf")

        # deterministic starters
        starters = [np.full(n, 1.0 / n), arcsine_initial(n)]
        for s in starters:
            sc = compute_score(s)
            if sc < local_best_sc:
                local_best_sc = sc
                local_best = s.copy()

        # random beta family (including occasional mixtures)
        max_samples = 8000   # a generous amount – cheap because only fast score
        for _ in range(max_samples):
            if time.time() >= deadline - 30.0:
                break
            if rng.random() < 0.25:  # mixture case
                w = rng.random()
                alpha1 = rng.uniform(0.02, 3.5)
                beta1 = rng.uniform(0.02, 3.5)
                eps1 = rng.uniform(0.0, max(0.01, 0.02 * n))
                alpha2 = rng.uniform(0.02, 3.5)
                beta2 = rng.uniform(0.02, 3.5)
                eps2 = rng.uniform(0.0, max(0.01, 0.02 * n))
                cand = generate_beta_mixture(
                    n, alpha1, beta1, eps1, alpha2, beta2, eps2, w
                )
            else:
                alpha = rng.uniform(0.02, 3.5)
                beta = rng.uniform(0.02, 3.5)
                eps = rng.uniform(0.0, max(0.01, 0.02 * n))
                cand = generate_beta_shape(n, alpha, beta, eps)

            sc = compute_score(cand)
            if sc < local_best_sc:
                local_best_sc = sc
                local_best = cand.copy()

        if local_best is None:
            # should never happen, but guard
            local_best = np.full(n, 1.0 / n)

        print(f"  Sampling best fast score = {local_best_sc:.6f}")

        # ---------- 2. Adam polishing ----------
        if time.time() < deadline - 5.0:
            remaining = deadline - time.time()
            max_iter_adam = int(min(120_000, remaining * 25))
            print(f"  Adam polishing (max_iter={max_iter_adam}) …")
            polished_a, _ = adam_optimize(
                initial_a=local_best,
                time_deadline=deadline,
                max_iter=max_iter_adam,
                lr0=0.12,
                lam_start=1.0,
                lam_end=800.0,
                noise_every=1200,
                noise_scale=4e-5,
            )
        else:
            polished_a = local_best

        # ---------- 3. Optional L‑BFGS‑B ----------
        if time.time() < deadline - 4.0:
            remaining = deadline - time.time()
            maxfun_lbfgs = int(min(3000, remaining * 5))
            print(f"  L‑BFGS‑B refinement (maxfun={maxfun_lbfgs}) …")
            polished_a, _ = lbfgs_optimize(
                polished_a,
                time_deadline=deadline,
                lam=800.0,
                maxfun=maxfun_lbfgs,
            )

        # ---------- 4. Exact sub‑gradient descent ----------
        if time.time() < deadline - 3.0:
            remaining = deadline - time.time()
            max_iter_sub = int(min(120_000, remaining * 30))
            print(f"  Sub‑gradient descent (max_iter={max_iter_sub}) …")
            polished_a, _ = subgradient_optimize(
                polished_a,
                time_deadline=deadline,
                step0=0.25,
                max_iter=max_iter_sub,
            )

        # ---------- 5. Stochastic local walk ----------
        if time.time() < deadline - 2.0:
            remaining = deadline - time.time()
            max_iter_loc = int(min(180_000, remaining * 35))
            print(f"  Random local walk (max_iter={max_iter_loc}) …")
            polished_a, _ = random_local_search(
                polished_a,
                time_deadline=deadline,
                init_step=0.018,
                max_iter=max_iter_loc,
            )

        # ---------- 6. Official evaluation ----------
        maybe_update(polished_a, f"length {n}")

    # ------------------------------------------------------------------
    # Phase 1 – optimise the length that historically gave the best bound
    # ------------------------------------------------------------------
    optimise_length(2048)

    # ------------------------------------------------------------------
    # Phase 2 – if enough budget remains, try a longer vector (4096)
    # ------------------------------------------------------------------
    if time.time() < deadline - 30.0:
        optimise_length(4096)

    # ------------------------------------------------------------------
    # Phase 3 – final polishing of the globally best vector (if any)
    # ------------------------------------------------------------------
    if best_seq and time.time() < deadline - 5.0:
        print("\n=== Global polishing of the overall best sequence ===")
        a = np.array(best_seq, dtype=float)

        # Adam polishing (more iterations, higher λ)
        remaining = deadline - time.time()
        max_iter_adam = int(min(200_000, remaining * 30))
        print(f"  Adam polishing (max_iter={max_iter_adam}) …")
        a, _ = adam_optimize(
            initial_a=a,
            time_deadline=deadline,
            max_iter=max_iter_adam,
            lr0=0.10,
            lam_start=1.0,
            lam_end=1000.0,
            noise_every=1500,
            noise_scale=3e-5,
        )

        # Exact sub‑gradient descent
        if time.time() < deadline - 4.0:
            remaining = deadline - time.time()
            max_iter_sub = int(min(150_000, remaining * 30))
            print(f"  Sub‑gradient polishing (max_iter={max_iter_sub}) …")
            a, _ = subgradient_optimize(
                a,
                time_deadline=deadline,
                step0=0.20,
                max_iter=max_iter_sub,
            )

        # Final random walk
        if time.time() < deadline - 2.0:
            remaining = deadline - time.time()
            max_iter_loc = int(min(250_000, remaining * 40))
            print(f"  Random walk final polish (max_iter={max_iter_loc}) …")
            a, _ = random_local_search(
                a,
                time_deadline=deadline,
                init_step=0.012,
                max_iter=max_iter_loc,
            )

        final_val = evaluate_sequence(a.tolist())
        print(f"Final polished official score = {final_val:.6f}")
        if final_val < best_val:
            best_val = final_val
            best_seq = a.tolist()
            print("*** Updated global best after final polish ***")

    # ------------------------------------------------------------------
    # 4. Guard – guarantee a non‑empty return
    # ------------------------------------------------------------------
    if not best_seq:
        # safe uniform fallback (length 256)
        best_seq = [1.0] * 256
        best_val = evaluate_sequence(best_seq)
        print("Fallback to uniform 256‑vector.", best_val)

    print(f"\nSearch finished. Best reported official score = {best_val:.6f}")
    return best_seq
