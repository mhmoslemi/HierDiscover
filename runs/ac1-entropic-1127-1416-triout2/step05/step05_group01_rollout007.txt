def exp_subgrad_optimize_topK(init_a, time_deadline, top_k=1, max_iter=10_000_000, eta0=0.5, mu=0.0):
    a = init_a.copy()
    n = len(a)
    best_a = a.copy()
    best_score = compute_score(a)
    m = np.zeros_like(a) if mu>0 else None
    for t in range(1, max_iter+1):
        if time.time() >= time_deadline:
            break
        conv = convolution_fft(a, a)
        # get top_k indices
        if top_k == 1:
            ks = [int(np.argmax(conv))]
        else:
            # argsort descending
            ks = np.argpartition(conv, -top_k)[-top_k:]
        g = np.zeros_like(a)
        for k in ks:
            # compute shift gradient contribution
            # i ranges intersect
            i_start = max(0, k - (n-1))
            i_end = min(n-1, k)  # inclusive
            # slice
            i_range = np.arange(i_start, i_end+1)
            g[i_range] += 2.0 * a[k - i_range]
        if mu > 0:
            if m is None:
                m = np.zeros_like(a)
            m = mu * m + (1 - mu) * g
            g_use = m
        else:
            g_use = g
        eta = eta0 / np.sqrt(t)
        a = a * np.exp(-eta * g_use)
        # projection onto simplex
        a = project_to_simplex(a)
        cur_score = compute_score(a)
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
    return best_a, best_score

def smooth_desc(init_a, time_deadline, beta_start=5.0, beta_end=200.0, max_iter=200_000, eta0=0.5):
    a = init_a.copy()
    n = len(a)
    best_a = a.copy()
    best_score = compute_score(a)
    t = 0
    while time.time() < time_deadline and t < max_iter:
        t += 1
        conv = convolution_fft(a, a)
        # schedule beta linearly
        beta = beta_start + (beta_end - beta_start) * (t / max_iter)
        # stable softmax
        maxc = np.max(beta * conv)
        w = np.exp(beta * conv - maxc)
        w = w / np.sum(w)

        grad = 2.0 * convolution_fft(w, a)
        eta = eta0 / np.sqrt(t)
        a = a * np.exp(-eta * grad)  # multiplicative update
        a = project_to_simplex(a)
        cur_score = compute_score(a)
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
    return best_a, best_score

def hill_climb(seq, time_deadline, max_steps=None):
    a = np.array(seq, dtype=float)
    n = a.shape[0]

    best_a = a.copy()
    best_score = compute_score(best_a)

    step_scale = 0.1
    steps = 0
    # If max_steps None compute based on time left
    if max_steps is None:
        # allow ~ (time left / 0.0005) steps estimated
        max_steps = int((time_deadline - time.time()) / 0.0005)

    while steps < max_steps and time.time() < time_deadline:
        # Choose move randomly: 0-0.4 swap, 0.4-0.8 transfer, 0.8-1 random perturb
        mv = np.random.rand()
        cand = best_a.copy()
        if mv < 0.4:
            i, j = np.random.randint(n, size=2)
            cand[i], cand[j] = cand[j], cand[i]
        elif mv < 0.8:
            i, j = np.random.randint(n, size=2)
            if i != j:
                delta = step_scale * cand[i]
                cand[i] -= delta
                cand[j] += delta
                cand[i] = max(cand[i], 0.0)
        else:
            # small Gaussian perturbation on a few entries
            idxs = np.random.choice(n, size=3, replace=False)
            delta = step_scale * np.random.randn(3)
            cand[idxs] += delta
            cand = np.maximum(cand, 0.0)
        cand = project_to_simplex(cand)
        cur_score = compute_score(cand)

        if cur_score < best_score:
            best_score = cur_score
            best_a = cand
            step_scale = min(step_scale * 1.05, 0.5)
        else:
            step_scale *= 0.97
        steps += 1

    return best_a.tolist(), best_score

"""
Optimisation of a non‑negative step‑function for the
`evaluate_sequence` objective.

The algorithm consists of three stages under a 1000 s budget:

1️⃣  Random Dirichlet initialisation (length n = 4096) and a fast
    exponentiated sub‑gradient descent.  The sub‑gradient is computed
    from the *k* largest convolution entries (default k = 3) and a
    simple momentum term is used to stabilise the multiplicative update.
    The step size η decays as η₀ / √t.

2️⃣  A soft‑max (smooth‑max) refinement.  The convolution values are turned
    into a probability distribution with a temperature β that is increased
    linearly from β₀ to β₁.  The gradient of the smooth‑max is exactly
    2·(w * a) where w are the soft‑max weights; a multiplicative update
    with a diminishing step size further reduces the peak of the
    autocorrelation.

3️⃣  An aggressive stochastic hill‑climber that performs swaps,
    proportional mass transfers and small Gaussian perturbations,
    always projecting back onto the probability simplex.

All phases respect the global time limit; the best sequence found is
returned.  The returned vector is already normalised (∑aᵢ = 1) so the
value reported by `evaluate_sequence` equals `2·n·max(a*a)`.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# FFT‑based linear convolution (real signals)
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Linear convolution of two real vectors using a real FFT."""
    m = len(a) + len(b) - 1
    n_fft = _next_pow2(m)
    A = np.fft.rfft(a, n=n_fft)
    B = np.fft.rfft(b, n=n_fft)
    C = A * B
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


# ----------------------------------------------------------------------
# Simple utilities
# ----------------------------------------------------------------------


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the probability simplex
    (non‑negative entries that sum to 1).
    """
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / v.shape[0])

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, v.shape[0] + 1)
    cond = u - cssv / ind > 0
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def compute_score(a: np.ndarray) -> float:
    """
    Fast replica of `evaluate_sequence` when the vector is already
    normalised (∑a = 1).  Returns 2·n·max(a*a).
    """
    n = a.shape[0]
    conv = convolution_fft(a, a)
    return 2.0 * n * float(np.max(conv))


# ----------------------------------------------------------------------
# 1️⃣ Exponentiated sub‑gradient descent (multiple top‑k peaks)
# ----------------------------------------------------------------------


def exp_subgrad_optimize_topk(init_a: np.ndarray,
                              time_deadline: float,
                              top_k: int = 3,
                              max_iter: int = 5_000_000,
                              eta0: float = 1.0,
                              mu: float = 0.8) -> tuple[np.ndarray, float]:
    """
    Multiplicative (exponentiated) sub‑gradient descent.
    The sub‑gradient is the sum of the gradients coming from the
    ``top_k`` largest convolution entries.  A simple exponential moving
    average (parameter ``mu``) supplies momentum.
    """
    a = init_a.astype(float).copy()
    n = len(a)
    best_a = a.copy()
    best_score = compute_score(a)

    # momentum buffer, zero‑initialised
    m_buf = np.zeros_like(a) if mu > 0 else None

    for t in range(1, max_iter + 1):
        if time.time() >= time_deadline:
            break

        conv = convolution_fft(a, a)

        # indices of the top‑k peaks (unsorted, possibly with ties)
        if top_k == 1:
            ks = [int(np.argmax(conv))]
        else:
            # argpartition is O(n) and gives the k largest unsorted
            ks = np.argpartition(conv, -top_k)[-top_k:]

        # build the summed sub‑gradient
        g = np.zeros_like(a)
        for k in ks:
            # valid i satisfy 0 ≤ i ≤ n‑1 and 0 ≤ k‑i ≤ n‑1
            i_start = max(0, k - (n - 1))
            i_end = min(n - 1, k)
            if i_start > i_end:
                continue
            i_range = np.arange(i_start, i_end + 1)
            g[i_range] += 2.0 * a[k - i_range]

        if mu > 0:
            m_buf = mu * m_buf + (1.0 - mu) * g
            use_g = m_buf
        else:
            use_g = g

        eta = eta0 / np.sqrt(t)

        # multiplicative update (always stays non‑negative)
        a = a * np.exp(-eta * use_g)

        # project back onto the simplex
        a = project_to_simplex(a)

        cur_score = compute_score(a)
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()

    return best_a, best_score


# ----------------------------------------------------------------------
# 2️⃣ Soft‑max (smooth‑max) gradient descent with increasing β
# ----------------------------------------------------------------------


def smooth_max_descent(init_a: np.ndarray,
                       time_deadline: float,
                       beta_start: float = 5.0,
                       beta_end: float = 400.0,
                       max_iter: int = 500_000,
                       eta0: float = 0.3) -> tuple[np.ndarray, float]:
    """
    Gradient descent on the smooth approximation
        f_β(a) = (1/β)·log Σ_k exp(β·(a*a)[k]).
    The gradient is 2·(w*a) where w are the soft‑max weights.
    β is increased linearly from ``beta_start`` to ``beta_end``.
    """
    a = init_a.astype(float).copy()
    n = len(a)
    best_a = a.copy()
    best_score = compute_score(a)

    for t in range(1, max_iter + 1):
        if time.time() >= time_deadline:
            break

        conv = convolution_fft(a, a)

        # linear schedule for β
        beta = beta_start + (beta_end - beta_start) * (t / max_iter)

        # stable soft‑max on the convolution values
        # max over β·conv is the same as max(conv)·β
        max_c = np.max(beta * conv)
        w = np.exp(beta * conv - max_c)
        w /= np.sum(w)                     # w now a probability vector, length 2n‑1

        # gradient = 2·(w*a) – we need only the first n components
        full_grad = convolution_fft(w, a)   # length 3n‑2
        grad = 2.0 * full_grad[:n]

        eta = eta0 / np.sqrt(t)

        # multiplicative update + simplex projection
        a = a * np.exp(-eta * grad)
        a = project_to_simplex(a)

        cur_score = compute_score(a)
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()

    return best_a, best_score


# ----------------------------------------------------------------------
# 3️⃣ Stochastic hill‑climbing refinement
# ----------------------------------------------------------------------


def hill_climb(seq: list[float],
               time_deadline: float,
               max_steps: int | None = None) -> tuple[list[float], float]:
    """
    Simple hill‑climber that repeatedly applies three types of moves:
      * swap two entries,
      * transfer a fraction of mass from one index to another,
      * add a small Gaussian perturbation on three random indices.
    After every move the vector is projected onto the simplex.
    The best vector seen is returned.
    """
    a = np.array(seq, dtype=float)
    n = a.shape[0]

    best_a = a.copy()
    best_score = compute_score(best_a)

    # estimate a safe number of steps from remaining time if not supplied
    if max_steps is None:
        remaining = max(0.0, time_deadline - time.time())
        # one step is cheap (< 0.5 ms); be generous
        max_steps = int(remaining / 0.0004)

    step_scale = 0.1  # initial fraction of mass moved in a transfer

    steps_done = 0
    while steps_done < max_steps and time.time() < time_deadline:
        move_type = np.random.rand()
        cand = best_a.copy()

        if move_type < 0.4:
            # ---- swap ----
            i, j = np.random.randint(n, size=2)
            cand[i], cand[j] = cand[j], cand[i]

        elif move_type < 0.8:
            # ---- mass transfer ----
            i, j = np.random.randint(n, size=2)
            if i != j:
                delta = step_scale * cand[i]
                cand[i] -= delta
                cand[j] += delta
                cand[i] = max(cand[i], 0.0)

        else:
            # ---- small Gaussian perturbation on three entries ----
            idxs = np.random.choice(n, size=3, replace=False)
            delta = step_scale * np.random.randn(3)
            cand[idxs] += delta
            cand = np.maximum(cand, 0.0)

        # re‑project onto the simplex
        cand = project_to_simplex(cand)
        cur_score = compute_score(cand)

        if cur_score < best_score:
            best_score = cur_score
            best_a = cand
            # success → be a bit bolder
            step_scale = min(step_scale * 1.05, 0.5)
        else:
            # failure → shrink step size
            step_scale *= 0.97

        steps_done += 1

    return best_a.tolist(), best_score
