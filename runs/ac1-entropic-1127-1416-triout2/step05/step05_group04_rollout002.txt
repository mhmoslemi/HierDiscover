"""
Improved search for a low‑score step sequence.

The algorithm proceeds in three phases:

1. **Parametric sweep** – a dense grid over a few low‑dimensional families
   (generalised arcsine / power shapes, mixtures with uniform and a random
   Beta‑family).  The best vector (normalised to Σ a=1) is retained.

2. **Continuous refinement** – the selected seed is polished with three
   optimisation stages:
        • Adam on a smooth soft‑max surrogate of max (conv(a,a)).
        • Exact sub‑gradient descent on the true objective.
        • A short simulated‑annealing run for escaping tiny local traps.

3. **Final validation** – the best candidate is evaluated with the official
   `evaluate_sequence` function; if it improves the current best it is kept.

All heavy work uses an FFT‑based convolution (`O(n log n)`) and a simplex
projection, which keeps the runtime well below the 1000 s limit even for
`n = 2048`.  The code is self‑contained and only relies on NumPy and the
standard library.
"""

import time
import numpy as np
from typing import List, Tuple

# ----------------------------------------------------------------------
# Low‑level utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """
    Linear convolution via real FFT.
    Length = len(a)+len(b)-1.
    """
    m = len(a) + len(b) - 1
    n_fft = next_pow2(m)
    A = np.fft.rfft(a, n=n_fft)
    B = np.fft.rfft(b, n=n_fft)
    C = A * B
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


def compute_score(a: np.ndarray) -> float:
    """
    Fast surrogate for the official score.
    The vector is assumed to sum to 1 (simplex), therefore the official
    denominator disappears and the score reduces to ``2·n·max(conv(a,a))``.
    """
    n = a.shape[0]
    conv = convolution_fft(a, a)
    return 2.0 * n * float(np.max(conv))


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection onto the probability simplex {x ≥ 0, Σx = 1}.
    Duchi et al., 2008.
    """
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / v.size)

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho_arr = np.where(u - cssv / np.arange(1, v.size + 1) > 0)[0]
    if rho_arr.size == 0:
        theta = 0.0
    else:
        rho = rho_arr[-1]
        theta = cssv[rho] / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


# ----------------------------------------------------------------------
# Shape factories (all return a vector that already sums to 1)
# ----------------------------------------------------------------------


def arcsine_initial(n: int) -> np.ndarray:
    """Classic arcsine shape: a_i ∝ 1 / sqrt((i+½)*(n‑½‑i))."""
    i = np.arange(n, dtype=float)
    eps = 0.25
    vals = 1.0 / np.sqrt((i + 0.5) * (n - 0.5 - i) + eps)
    vals /= vals.sum()
    return vals


def generate_power_shape(n: int, p: float, eps: float) -> np.ndarray:
    """
    Generalised arcsine: a_i ∝ ((i+½)*(n‑½‑i)+ε)^{-p},   p>0.
    The resulting vector is normalised to Σ = 1.
    """
    i = np.arange(n, dtype=float)
    denom = ((i + 0.5) * (n - 0.5 - i) + eps) ** p
    vals = 1.0 / denom
    total = vals.sum()
    if not np.isfinite(total) or total <= 0.0:
        return np.full(n, 1.0 / n)
    return vals / total


def generate_beta_shape(n: int, alpha: float, beta: float, eps: float) -> np.ndarray:
    """
    (i+ε)^{α‑1}·(n‑1‑i+ε)^{β‑1} on {0,…,n‑1}, normalised to Σ=1.
    """
    i = np.arange(n, dtype=float)
    left = (i + eps) ** (alpha - 1.0)
    right = (n - 1 - i + eps) ** (beta - 1.0)
    base = left * right
    total = base.sum()
    if not np.isfinite(total) or total <= 0.0:
        return np.full(n, 1.0 / n)
    return base / total


# ----------------------------------------------------------------------
# Optimisers
# ----------------------------------------------------------------------


def smooth_gradient(a: np.ndarray, lam: float) -> np.ndarray:
    """
    Gradient of the soft‑max approximation of max(conv(a,a)).
    Returns ∇_a (2·n·softmax_max) where the simplex constraint is handled
    by the subsequent projection.
    """
    n = a.shape[0]
    conv = convolution_fft(a, a)               # length 2·n‑1
    max_c = np.max(conv)

    # Numerically stable soft‑max (weights sum to 1)
    exp_term = np.exp(lam * (conv - max_c))
    w = exp_term / np.sum(exp_term)

    # Weighted sub‑gradient: 2·∑_k w_k shift(a,k)
    w_rev = w[::-1]
    conv_rev = convolution_fft(a, w_rev)
    L = 2 * n - 1
    weighted_grad = 2.0 * conv_rev[L - 1 - np.arange(n)]

    # Objective factor 2·n
    return 2.0 * n * weighted_grad


def adam_optimize_smooth(initial_a: np.ndarray,
                         deadline: float,
                         max_iter: int = 200_000,
                         lr0: float = 0.12,
                         lam_start: float = 2.0,
                         lam_end: float = 200.0,
                         noise_every: int = 2000,
                         noise_scale: float = 1e-5) -> Tuple[np.ndarray, float]:
    """
    Adam optimiser on the smooth surrogate.
    Returns the best vector (projected on the simplex) and its fast score.
    """
    a = project_to_simplex(initial_a.astype(float).copy())
    best_a = a.copy()
    best_sc = compute_score(a)

    beta1, beta2 = 0.9, 0.999
    eps = 1e-8
    m = np.zeros_like(a)
    v = np.zeros_like(a)

    for it in range(1, max_iter + 1):
        if time.time() >= deadline:
            break

        # Linear schedule for temperature λ
        lam = lam_start + (lam_end - lam_start) * min(1.0, it / max_iter)

        grad = smooth_gradient(a, lam)

        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)

        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        step = lr0 * m_hat / (np.sqrt(v_hat) + eps)
        a_new = a - step

        if noise_every and (it % noise_every == 0):
            a_new += np.random.randn(a.size) * noise_scale

        a_new = project_to_simplex(a_new)

        cur_sc = compute_score(a_new)
        if cur_sc < best_sc:
            best_sc = cur_sc
            best_a = a_new.copy()

        a = a_new

        # occasional progress report
        if it % 50000 == 0:
            print(f"    AdamSmooth it={it:,d} cur={cur_sc:.6f} best={best_sc:.6f}")

    return best_a, best_sc


def subgradient_descent_exact(initial_a: np.ndarray,
                              deadline: float,
                              max_iter: int = 500_000,
                              step0: float = 0.02) -> np.ndarray:
    """
    Sub‑gradient descent on the exact objective f(a)=2·n·max(conv(a,a)).
    Step size η_t = step0 / sqrt(t).  Projection onto the simplex keeps the
    sum fixed to 1.
    """
    a = project_to_simplex(initial_a.copy())
    best_a = a.copy()
    best_sc = compute_score(a)
    n = a.shape[0]
    L = 2 * n - 1

    for it in range(1, max_iter + 1):
        if time.time() >= deadline:
            break

        conv = convolution_fft(a, a)
        max_val = np.max(conv)

        # mask of entries attaining (or almost attaining) the maximum
        tol = max_val * 1e-12 + 1e-14
        mask = (np.abs(conv - max_val) <= tol).astype(float)

        mask_rev = mask[::-1]
        conv_rev = convolution_fft(a, mask_rev)
        weighted_grad = 2.0 * conv_rev[L - 1 - np.arange(n)]

        eta = step0 / np.sqrt(it)
        a = a - eta * weighted_grad
        a = project_to_simplex(a)

        cur_sc = compute_score(a)
        if cur_sc < best_sc:
            best_sc = cur_sc
            best_a = a.copy()

        if it % 50000 == 0:
            print(f"    SubGrad it={it:,d} cur={cur_sc:.6f} best={best_sc:.6f}")

    return best_a


def simulated_annealing(a: np.ndarray,
                        rng: np.random.Generator,
                        deadline: float,
                        max_iters: int = 200_000,
                        init_temp: float = 0.02,
                        decay: float = 0.9999) -> np.ndarray:
    """
    Simple Metropolis‑style local search.
    Moves a small fraction of mass from one coordinate to another.
    The fast surrogate (2·n·max) is used for the acceptance test.
    """
    cur = a.copy()
    cur_sc = compute_score(cur)
    best = cur.copy()
    best_sc = cur_sc
    n = cur.size
    temp = init_temp
    it = 0

    while it < max_iters and time.time() < deadline:
        it += 1
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue

        # move up to 20 % of the mass at i
        delta = cur[i] * 0.2 * rng.random()
        if delta <= 1e-15 or cur[i] - delta < 0:
            continue

        cand = cur.copy()
        cand[i] -= delta
        cand[j] += delta

        cand_sc = compute_score(cand)

        if cand_sc < cur_sc or rng.random() < np.exp(-(cand_sc - cur_sc) / max(temp, 1e-12)):
            cur = cand
            cur_sc = cand_sc
            if cur_sc < best_sc:
                best = cur.copy()
                best_sc = cur_sc

        temp = max(temp * decay, 1e-8)

        if it % 50000 == 0:
            print(f"    SA it={it:,d} cur={cur_sc:.6f} best={best_sc:.6f}")

    return best

def propose_candidate(seed: int = 42,
                     budget_s: int = 1000,
                     **kwargs) -> List[float]:
    """
    Search for a non‑negative step sequence with a low ``evaluate_sequence``
    value.  The routine consists of a parametric sweep, followed by
    Adam → sub‑gradient → simulated‑annealing refinement.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.8    # small safety margin

    # --------------------------------------------------------------
    # 0.  Possible fallback from a previous run
    # --------------------------------------------------------------
    best_seq: List[float] = []
    best_val: float = float("inf")
    if "height_sequence_1" in globals():
        try:
            fallback = [float(x) for x in globals()["height_sequence_1"]]
            val = evaluate_sequence(fallback)
            print(f"Fallback (height_sequence_1) score = {val:.6f}")
            if val < best_val:
                best_val = val
                best_seq = fallback
        except Exception:
            pass

    # --------------------------------------------------------------
    # 1.  Parametric exploration (very cheap, dense grid)
    # --------------------------------------------------------------
    n = 2048
    cur_best_score = float("inf")
    cur_best_vec = None

    # a) power / generalized‑arcsine family
    eps_vals = [0.0, 0.01, 0.1, 0.5, 1.0]
    p_grid = np.linspace(0.20, 1.00, 161)           # step ≈0.005
    for eps in eps_vals:
        for p in p_grid:
            vec = generate_power_shape(n, float(p), float(eps))
            sc = compute_score(vec)
            if sc < cur_best_score:
                cur_best_score = sc
                cur_best_vec = vec

    # b) mixture of classic arcsine and uniform
    arc = arcsine_initial(n)
    uniform = np.full(n, 1.0 / n)
    for w in np.linspace(0.0, 1.0, 21):
        vec = w * arc + (1.0 - w) * uniform
        vec /= vec.sum()
        sc = compute_score(vec)
        if sc < cur_best_score:
            cur_best_score = sc
            cur_best_vec = vec

    # c) random Beta‑family (few hundred samples)
    for _ in range(2000):
        if time.time() >= deadline - 30.0:
            break
        alpha = rng.uniform(0.01, 8.0)
        beta = rng.uniform(0.01, 8.0)
        eps = rng.uniform(1e-6, 2.0)
        vec = generate_beta_shape(n, alpha, beta, eps)
        sc = compute_score(vec)
        if sc < cur_best_score:
            cur_best_score = sc
            cur_best_vec = vec

    # Store the best vector from the sweep
    if cur_best_vec is not None:
        best_seq = cur_best_vec.tolist()
        best_val = cur_best_score
        print(f"Parametric sweep best surrogate score = {best_val:.6f}")

    # --------------------------------------------------------------
    # 2.  Continuous refinement of the selected seed
    # --------------------------------------------------------------
    if not best_seq:
        # Should never happen, but guard against empty fallback.
        best_seq = [1.0] * n

    a = np.array(best_seq, dtype=float)

    # --- Adam on smooth surrogate ---------------------------------
    remaining = deadline - time.time()
    if remaining > 8.0:
        max_iter_adam = int(min(300_000, remaining * 150))
        print(f"Running Adam (smooth) for up to {max_iter_adam} iters")
        a, _ = adam_optimize_smooth(
            initial_a=a,
            deadline=deadline,
            max_iter=max_iter_adam,
            lr0=0.12,
            lam_start=2.0,
            lam_end=300.0,
            noise_every=2500,
            noise_scale=2e-5,
        )

    # --- Exact sub‑gradient descent -------------------------------
    remaining = deadline - time.time()
    if remaining > 8.0:
        max_iter_sub = int(min(600_000, remaining * 120))
        print(f"Running exact sub‑gradient descent for up to {max_iter_sub} iters")
        a = subgradient_descent_exact(
            a,
            deadline,
            max_iter=max_iter_sub,
            step0=0.015,
        )

    # --- Simulated annealing polishing ----------------------------
    remaining = deadline - time.time()
    if remaining > 6.0:
        max_iter_sa = int(min(250_000, remaining * 100))
        print(f"Running simulated annealing for up to {max_iter_sa} iters")
        a = simulated_annealing(
            a,
            rng,
            deadline,
            max_iters=max_iter_sa,
            init_temp=0.018,
            decay=0.99995,
        )

    # --------------------------------------------------------------
    # 3.  Official evaluation of the polished candidate
    # --------------------------------------------------------------
    try:
        official_score = evaluate_sequence(a.tolist())
        print(f"Official evaluation of polished candidate = {official_score:.6f}")
        if official_score < best_val:
            best_val = official_score
            best_seq = a.tolist()
    except Exception:
        pass

    # --------------------------------------------------------------
    # 4.  Guard – ensure a non‑empty return value
    # --------------------------------------------------------------
    if not best_seq:
        best_seq = [1.0] * n
        best_val = evaluate_sequence(best_seq)

    print(f"\nSearch finished. Best reported official score = {best_val:.6f}")
    return best_seq
