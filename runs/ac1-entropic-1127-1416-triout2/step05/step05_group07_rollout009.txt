def adam_step(v, grad_v, t, lr=0.02, b1=0.9, b2=0.999, eps=1e-8):
    m = b1*m + (1-b1)*grad_v
    v_hat = b2*v2 + (1-b2)*(grad_v**2)
    m_hat = m/(1-b1**t)
    v_hat_corr = v_hat/(1-b2**t)
    v = v - lr * m_hat / (np.sqrt(v_hat_corr)+eps)
    return v, m, v_hat
```

def smooth_loss_and_grad(a, beta):
    conv = convolution_fft(a, a)   # length 2n-1

def run_optimize_logit(logits, n, deadline, rng, extra_iters=None):
    # parameters
    max_iter_default = 5000
    if extra_iters is not None:
        max_iter = extra_iters
    else:
        # estimate iterations based on time left
        remaining = max(0.0, deadline - time.time())
        # assume ~3000 iter per second for n~8192 maybe slower; we can approximate as 2000.
        # Provide at least 1000.
        max_iter = int(min(100_000, max(2000, remaining * 2000)))
    # Adam hyperparams
    lr = 0.15
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8

    # maintain momentum
    m = np.zeros_like(logits)
    v2 = np.zeros_like(logits)

    best_a = None
    best_obj = float('inf')
    # schedule for beta
    beta0 = 1.0
    beta_max = 300.0
    # linear increase
    # compute per-iteration increment:
    beta_inc = (beta_max - beta0) / max_iter

    for t in range(1, max_iter + 1):
        if time.time() >= deadline:
            break

        # compute a via softmax
        max_logit = np.max(logits)
        exps = np.exp(logits - max_logit)
        a = exps / exps.sum()

        beta = beta0 + beta_inc * t
        if beta > beta_max:
            beta = beta_max

        # compute loss and gradient wrt a
        loss, grad_a = smooth_loss_and_grad(a, beta)

        # gradient wrt logits
        dot = np.dot(a, grad_a)
        grad_logit = a * (grad_a - dot)

        # Adam update
        m = beta1 * m + (1 - beta1) * grad_logit
        v2 = beta2 * v2 + (1 - beta2) * (grad_logit ** 2)

        m_hat = m / (1 - beta1 ** t)
        v2_hat = v2 / (1 - beta2 ** t)

        logits = logits - lr * m_hat / (np.sqrt(v2_hat) + eps)

        # compute current objective using conv (full conv)
        conv = convolution_fft(a, a)
        cur_obj = 2.0 * n * conv.max()
        # record best
        if cur_obj < best_obj:
            best_obj = cur_obj
            best_a = a.copy()
        # optionally print progress occasionally
        if t % 2000 == 0:
            print(f"      iter={t} cur_obj={cur_obj:.6f} best_obj={best_obj:.6f} beta={beta:.1f}")

    # after loop, if best_a is None fallback to initial a
    if best_a is None:
        best_a = a

    return best_a, best_obj

"""
Propose a low‑value step‑function (non‑negative sequence) for the
max–self‑convolution objective.

Algorithm
---------
1.  Build a hierarchy of problem sizes (powers of two).  For each size n
    we generate several sensible starting points:
        – up‑sampled best solution from the previous size,
        – the discrete arcsine density (known to give a flat convolution),
        – a few Dirichlet draws and slight perturbations of the arcsine.
2.  Each start is turned into logits  v  (soft‑max parametrisation
    a_i = exp(v_i)/∑exp(v) ) and optimised with an Adam‑style mirror‑descent
    on the *smooth* surrogate
        L_β(a) = (1/β)·log∑_k exp(β·(a∗a)_k) .
    The gradient of L_β w.r.t. a is
        ∇_a L = 2·corr(a , w)   with   w_k = softmax(β·(a∗a)_k) .
    Using the chain‑rule we obtain the gradient w.r.t. the logits.
    The temperature β is increased linearly from 1 to a large value
    (≈200) so the surrogate gradually approaches the true max.
3.  During optimisation we keep the vector that yields the smallest
    *true* objective 2·n·max(a∗a)/(∑a)^2 (the denominator is 1 because
    a lives on the simplex).  The true objective can be read from the
    convolution already computed for the surrogate, so no extra cost is
    incurred.
4.  After the hierarchy is finished we optionally run a final polishing
    stage with a finer learning‑rate schedule.
5.  The best feasible sequence found is returned.  The routine always
    finishes with a valid list (fallback to the arcsine shape if needed)
    and respects the supplied wall‑clock budget (default 1000 s).

The implementation is self‑contained, uses only NumPy and the provided
``evaluate_sequence`` function, and prints progress / best scores.
"""

import time
import numpy as np
from typing import List, Tuple

# ----------------------------------------------------------------------
# Basic numeric helpers
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """
    Full linear convolution using a real FFT.
    This is much faster than numpy.convolve for the lengths we need.
    """
    m = len(a) + len(b) - 1
    n_fft = next_pow2(m)
    A = np.fft.rfft(a, n=n_fft)
    B = np.fft.rfft(b, n=n_fft)
    C = A * B
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


def compute_objective(a: np.ndarray) -> float:
    """
    Exact objective used by `evaluate_sequence` for a vector that
    already sums to 1: 2·n·max(a∗a).
    """
    n = len(a)
    conv = convolution_fft(a, a)
    return 2.0 * n * conv.max()


# ----------------------------------------------------------------------
# Families of initial shapes
# ----------------------------------------------------------------------


def arcsine_initial(n: int) -> np.ndarray:
    """Discrete arcsine density a_i ∝ 1/√((i+½)*(n‑½‑i))."""
    i = np.arange(n, dtype=float)
    eps = 0.25  # tiny regulariser to avoid division‑by‑zero
    vals = 1.0 / np.sqrt((i + 0.5) * (n - 0.5 - i) + eps)
    return vals / vals.sum()


def random_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    """Gamma‑based Dirichlet‑like draw (α=conc).  conc<0 gives heavy‑tailed draws."""
    if conc <= 0.0:
        # heavy‑tailed “near‑sparse” draws
        samp = rng.exponential(1.0, size=n) ** (1.0 / max(0.1, -conc))
    else:
        samp = rng.gamma(conc, 1.0, size=n)
    samp += 1e-12
    return samp / samp.sum()


def perturb(a: np.ndarray, rng: np.random.Generator, scale: float = 0.02) -> np.ndarray:
    """Add tiny Gaussian jitter and renormalise."""
    b = a + rng.normal(0.0, scale, size=a.shape)
    b = np.maximum(b, 0.0)
    if b.sum() == 0.0:
        return np.full_like(a, 1.0 / len(a))
    return b / b.sum()


def upsample(prev: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation from len(prev) → new_n (followed by re‑normalisation)."""
    x_old = np.linspace(0.0, 1.0, len(prev))
    x_new = np.linspace(0.0, 1.0, new_n)
    interp = np.interp(x_new, x_old, prev)
    tot = interp.sum()
    if tot == 0.0:
        return np.full(new_n, 1.0 / new_n)
    return interp / tot


# ----------------------------------------------------------------------
# Smooth surrogate (log‑sum‑exp) and its gradient
# ----------------------------------------------------------------------


def smooth_loss_and_grad(a: np.ndarray, beta: float) -> Tuple[float, np.ndarray]:
    """
    Return (loss, grad) where
        loss = (1/β)·log∑_k exp(β·(a∗a)_k)
    and grad = ∂loss/∂a .
    """
    n = len(a)

    # a∗a
    conv = convolution_fft(a, a)                     # length 2n‑1
    maxc = conv.max()

    # soft‑max weights w_k ∝ exp(β·conv_k)
    shifted = conv - maxc                # stabilise exponentials
    exp_term = np.exp(beta * shifted)
    w = exp_term / exp_term.sum()

    # loss = maxc + (log∑exp(β·(conv-maxc)))/β
    loss = maxc + np.log(exp_term.sum()) / beta

    # gradient w.r.t. a : 2·corr(a , w)
    # correlation can be obtained as convolution of a with reversed w.
    w_rev = w[::-1]
    corr_full = convolution_fft(a, w_rev)          # length 3n‑2

    # indices 2n‑2 , 2n‑3 , … , n‑2  (exactly n entries)
    grad_a = 2.0 * corr_full[2 * n - 2 : n - 2 : -1]

    return loss, grad_a


# ----------------------------------------------------------------------
# Adam optimiser on the simplex (soft‑max parametrisation)
# ----------------------------------------------------------------------


def optimise_logits(
    init_logits: np.ndarray,
    n: int,
    deadline: float,
    *,
    max_iter: int = 100_000,
    lr: float = 0.15,
    beta1: float = 0.9,
    beta2: float = 0.999,
    eps: float = 1e-8,
    beta_start: float = 1.0,
    beta_end: float = 200.0,
    report_every: int = 2000,
) -> Tuple[np.ndarray, float]:
    """
    Runs Adam on the smooth surrogate.
    Returns the best vector (as probabilities) and its *exact* objective.
    """
    logits = init_logits.copy()
    m = np.zeros_like(logits)
    v2 = np.zeros_like(logits)

    best_a = None
    best_obj = float("inf")

    # linear temperature schedule
    beta_inc = (beta_end - beta_start) / max_iter

    for t in range(1, max_iter + 1):
        # early‑out if we run out of wall‑clock time
        if time.time() >= deadline:
            break

        # --- soft‑max (keeps us on the simplex) ---
        max_logit = np.max(logits)
        exps = np.exp(logits - max_logit)
        a = exps / exps.sum()          # a_i ≥ 0, sum = 1

        # --- temperature for this iteration ---
        beta = beta_start + beta_inc * t
        if beta > beta_end:
            beta = beta_end

        # --- smooth loss and gradient wrt a ---
        _, grad_a = smooth_loss_and_grad(a, beta)

        # --- chain rule: gradient wrt logits ---
        dot = np.dot(a, grad_a)
        grad_logit = a * (grad_a - dot)

        # --- Adam update ---
        m = beta1 * m + (1.0 - beta1) * grad_logit
        v2 = beta2 * v2 + (1.0 - beta2) * (grad_logit ** 2)

        m_hat = m / (1.0 - beta1 ** t)
        v2_hat = v2 / (1.0 - beta2 ** t)

        logits -= lr * m_hat / (np.sqrt(v2_hat) + eps)

        # --- evaluate the *true* objective (no extra convolution needed) ---
        conv = convolution_fft(a, a)
        cur_obj = 2.0 * n * conv.max()
        if cur_obj < best_obj:
            best_obj = cur_obj
            best_a = a.copy()

        if t % report_every == 0:
            print(
                f"      iter={t:,d} beta={beta:.1f} cur_obj={cur_obj:.6f} best_obj={best_obj:.6f}"
            )

    if best_a is None:
        # fallback – should never happen
        best_a = a
        best_obj = 2.0 * n * convolution_fft(best_a, best_a).max()

    return best_a, best_obj

# Main entry point required by the platform
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: int = 1000) -> List[float]:
    """
    Search for a non‑negative step‑function with a low ``evaluate_sequence`` value.
    The routine respects the provided time budget and returns a feasible list.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    # keep a small safety margin for the final call to ``evaluate_sequence``
    deadline = start_time + budget_s - 2.0

    best_seq: List[float] = []
    best_val: float = float("inf")
    best_a: np.ndarray = None   # best vector on the simplex (sum=1)

    # ------------------------------------------------------------------
    # 0.  Optional fallback from a previously discovered construction.
    # ------------------------------------------------------------------
    try:
        fallback = [float(x) for x in height_sequence_1]  # type: ignore
        val = evaluate_sequence(fallback)
        print(f"Fallback (height_sequence_1) score = {val:.6f}")
        if val < best_val:
            best_val = val
            best_seq = fallback
            best_a = np.array(fallback, dtype=float)
            best_a /= best_a.sum()
    except Exception:
        pass

    # ------------------------------------------------------------------
    # 1.  Hierarchical optimisation over increasing lengths.
    # ------------------------------------------------------------------
    # The lengths are powers of two – feel free to prune if time runs out.
    lengths = [512, 1024, 2048, 4096, 8192, 16384]

    for n in lengths:
        if time.time() >= deadline - 5.0:
            break
        print(f"\n=== Optimising length n = {n} ===")

        # --------------------------------------------------------------
        # 1a. Gather a small library of start vectors for this size.
        # --------------------------------------------------------------
        starts: List[np.ndarray] = []

        if best_a is not None:
            # Upsample the current best – strong warm‑start for the larger size.
            up = upsample(best_a, n)
            starts.append(up)

        # Classic arcsine shape (always a solid baseline).
        starts.append(arcsine_initial(n))

        # A few Dirichlet draws – give diversity.
        for conc in (0.5, 1.0, 2.0):
            starts.append(random_dirichlet(n, conc, rng))

        # Small perturbations of the arcsine shape.
        for _ in range(2):
            starts.append(perturb(arcsine_initial(n), rng, scale=0.02))

        # --------------------------------------------------------------
        # 1b.  Run the Adam‑based optimiser on each start and keep the best.
        # --------------------------------------------------------------
        for idx, init in enumerate(starts):
            if time.time() >= deadline - 5.0:
                break
            print(f"  start {idx + 1}/{len(starts)} (n={n})")

            # Initialise logits – using log(probability) is a natural choice.
            eps = 1e-12
            init = np.clip(init, eps, None)  # avoid log(0)
            logits = np.log(init)

            # Estimate a reasonable iteration budget from the remaining time.
            remaining = max(0.0, deadline - time.time())
            # Rough empirical figure: ≈2000 iters/s for n≈8 k, slower for larger n.
            max_iter = int(min(200_000, max(5_000, remaining * 1500)))

            a_opt, _ = optimise_logits(
                init_logits=logits,
                n=n,
                deadline=deadline,
                max_iter=max_iter,
                lr=0.12,               # a bit smaller for stability on larger n
                beta_start=1.0,
                beta_end=250.0,
                report_every=3000,
            )

            # Official evaluation (expensive, called only for promising candidates)
            val = evaluate_sequence(a_opt.tolist())
            print(f"    evaluated score = {val:.6f}")

            if val < best_val:
                best_val = val
                best_seq = a_opt.tolist()
                best_a = a_opt.copy()
                print(f"    *** New global best: {best_val:.6f} (n={n}) ***")

    # ------------------------------------------------------------------
    # 2.  Final polishing of the overall best (if time permits).
    # ------------------------------------------------------------------
    if best_a is not None and time.time() < deadline - 5.0:
        n = len(best_a)
        print("\nFinal polishing of the global best (fine‑grained Adam run)")
        eps = 1e-12
        logits = np.log(np.clip(best_a, eps, None))

        remaining = max(0.0, deadline - time.time())
        max_iter = int(min(300_000, max(10_000, remaining * 1200)))

        a_refined, _ = optimise_logits(
            init_logits=logits,
            n=n,
            deadline=deadline,
            max_iter=max_iter,
            lr=0.06,               # smaller step for a delicate final phase
            beta_start=150.0,
            beta_end=400.0,
            report_every=5000,
        )

        final_val = evaluate_sequence(a_refined.tolist())
        print(f"Final polished score = {final_val:.6f}")
        if final_val < best_val:
            best_val = final_val
            best_seq = a_refined.tolist()
            print("*** Updated global best after final polish ***")

    # ------------------------------------------------------------------
    # 3.  Guarantee a feasible return.
    # ------------------------------------------------------------------
    if not best_seq:
        # As a last resort return a uniform vector of modest size.
        best_seq = [1.0] * 256
        best_val = evaluate_sequence(best_seq)

    print(f"\nSearch finished. Best reported score = {best_val:.6f}")
    return best_seq
