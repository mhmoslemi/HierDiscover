def generate_candidates(n, rng):
    cand = []
    # uniform
    cand.append(np.full(n, 1.0/n))
    # arcsine
    cand.append(arcsine_initial(n))
    # semicircle
    cand.append(semicircle_initial(n))
    # powerlaw random
    for _ in range(4):
        p = rng.uniform(0.2, 1.0)
        eps = rng.uniform(0.0, 5.0)
        cand.append(generate_powerlaw(n, p, eps))
    # beta-like random
    for _ in range(3):
        alpha = rng.uniform(0.4, 2.0)
        beta = rng.uniform(0.4, 2.0)
        cand.append(generate_beta_like(n, alpha, beta))
    # dirichlet draws
    for conc in [0.3, 1.0, 2.0]:
        cand.append(random_dirichlet(n, conc, rng))
    # mixture arcsine+uniform
    for _ in range(3):
        w = rng.random()
        mix = w * arcsine_initial(n) + (1 - w) * np.full(n, 1.0/n)
        cand.append(mix / mix.sum())
    # mixture arcsine+semicircle
    for _ in range(2):
        w = rng.random()
        mix = w * arcsine_initial(n) + (1 - w) * semicircle_initial(n)
        cand.append(mix / mix.sum())
    return cand

"""
propose_candidate
-----------------

A time‑bounded multi‑stage search for a non‑negative step‑function that minimizes

    2·n·max(a∗a) / (sum(a)²)

The algorithm works in three phases:

1. **Broad exploration** – several sequence lengths (powers of two) are examined.
   For each length a diverse family of seed vectors is generated
   (uniform, arcsine, semicircle, beta‑like, power‑law, Dirichlet and mixtures).
   The cheap “fast score’’ (the same objective without the security checks)
   ranks the seeds.

2. **Focused refinement** – the ten best seeds (across all lengths) are polished.
   * Simulated annealing moves a tiny amount of mass between two random bins.
   * A soft‑max gradient descent (log‑sum‑exp surrogate) performs many
     exponentiated‑gradient updates, preserving non‑negativity and the simplex.
   * A short exponentiated sub‑gradient descent on the true max‑convolution
     removes any remaining bias.

   All three refinements stop as soon as the global wall‑clock deadline is
   reached.

3. **Final validation** – the best vector found is finally evaluated with the
   official ``evaluate_sequence`` function and returned.

The code respects the 1000 s budget, never exceeds the budget by more than a
few seconds, and always returns a feasible list of floats.
"""

import time
import numpy as np
from typing import List, Tuple

# ----------------------------------------------------------------------
# Low‑level numeric utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full linear convolution using a real FFT (zero‑padded to a power of two)."""
    m = a.size + b.size - 1
    n_fft = next_pow2(m)
    A = np.fft.rfft(a, n=n_fft)
    B = np.fft.rfft(b, n=n_fft)
    C = A * B
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


def compute_score(a: np.ndarray) -> float:
    """
    Fast version of the official objective (scale‑invariant):
        2·n·max(a∗a) / (sum(a)²)
    The vector a is assumed non‑negative.
    """
    n = a.size
    conv = convolution_fft(a, a)
    max_val = float(conv.max())
    s = float(a.sum())
    if s < 1e-12:
        return float("inf")
    return 2.0 * n * max_val / (s * s)


# ----------------------------------------------------------------------
# Families of seed vectors
# ----------------------------------------------------------------------


def arcsine_initial(n: int) -> np.ndarray:
    """Discrete arcsine‑like density (no singularities, normalised to sum = 1)."""
    i = np.arange(n, dtype=float)
    # eps avoids the exact singularity at the endpoints.
    eps = 0.25
    vals = 1.0 / np.sqrt((i + 0.5) * (n - 0.5 - i) + eps)
    return vals / vals.sum()


def semicircle_initial(n: int) -> np.ndarray:
    """Semicircle shape a_i ∝ √(i·(n‑1‑i))."""
    i = np.arange(n, dtype=float)
    vals = np.sqrt(i * (n - 1 - i))
    # avoid the all‑zero vector when n=1
    if vals.sum() == 0.0:
        return np.full(n, 1.0 / n)
    return vals / vals.sum()


def generate_powerlaw(n: int, p: float, eps: float) -> np.ndarray:
    """a_i ∝ ((i+ε)(n‑1‑i+ε))⁻ᵖ."""
    i = np.arange(n, dtype=float)
    base = ((i + eps) * (n - 1 - i + eps)) ** (-p)
    total = base.sum()
    if total == 0.0:
        return np.full(n, 1.0 / n)
    return base / total


def generate_beta_like(n: int, alpha: float, beta: float) -> np.ndarray:
    """Beta‑like shape a_i ∝ (i+½)^{α‑1}·(n‑1‑i+½)^{β‑1}."""
    i = np.arange(n, dtype=float)
    base = (i + 0.5) ** (alpha - 1) * (n - 1 - i + 0.5) ** (beta - 1)
    total = base.sum()
    if total == 0.0:
        return np.full(n, 1.0 / n)
    return base / total


def random_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    """Dirichlet‑like sample (Gamma draws) and normalise."""
    if conc <= 0.0:
        samp = rng.exponential(1.0, size=n) ** (1.0 / max(0.1, conc))
    else:
        samp = rng.gamma(conc, 1.0, size=n)
    samp += 1e-12
    return samp / samp.sum()


def mixture(arcsine: np.ndarray, other: np.ndarray, w: float) -> np.ndarray:
    """Simple convex mixture of two vectors."""
    mix = w * arcsine + (1.0 - w) * other
    s = mix.sum()
    if s == 0.0:
        return np.full_like(mix, 1.0 / mix.size)
    return mix / s


# ----------------------------------------------------------------------
# Local stochastic improvement (simulated annealing)
# ----------------------------------------------------------------------


def propose_neighbor(a: np.ndarray,
                     rng: np.random.Generator,
                     max_step: float = 0.02) -> np.ndarray:
    """Transfer a small amount of mass from one random index to another."""
    n = a.size
    i, j = rng.integers(0, n, size=2)
    if i == j or a[i] == 0.0:
        return a
    delta = a[i] * rng.random() * max_step
    b = a.copy()
    b[i] -= delta
    b[j] += delta
    return b


def simulated_annealing(a0: np.ndarray,
                        deadline: float,
                        rng: np.random.Generator,
                        max_iter: int = 2000,
                        start_temp: float = 0.005,
                        cooling: float = 0.9995,
                        max_step: float = 0.02) -> Tuple[np.ndarray, float]:
    """
    Very light SA that stays on the simplex.  Returns the best vector seen
    together with its fast score.
    """
    a = a0.copy()
    cur_sc = compute_score(a)
    best_a = a.copy()
    best_sc = cur_sc
    temp = start_temp

    for _ in range(max_iter):
        if time.time() >= deadline:
            break
        cand = propose_neighbor(a, rng, max_step)
        cand_sc = compute_score(cand)
        delta = cand_sc - cur_sc
        if delta < 0.0 or rng.random() < np.exp(-delta / max(1e-12, temp)):
            a = cand
            cur_sc = cand_sc
            if cur_sc < best_sc:
                best_sc = cur_sc
                best_a = a.copy()
        temp *= cooling
    return best_a, best_sc


# ----------------------------------------------------------------------
# Smooth‑max (log‑sum‑exp) gradient descent
# ----------------------------------------------------------------------


def softmax_gradient_descent(initial_a: np.ndarray,
                             deadline: float,
                             betas: List[float],
                             iter_per_beta: int = 300,
                             eta0: float = 0.5) -> Tuple[np.ndarray, float]:
    """
    Minimises the smooth surrogate
        Φ_β(a) = (1/β)·log Σ_k exp(β·(a∗a)_k)

    using an exponentiated‑gradient update and a projection onto the simplex.
    """
    a = np.maximum(initial_a, 1e-30)
    a = a / a.sum()
    best_a = a.copy()
    best_sc = compute_score(a)
    n = a.size

    for beta in betas:
        for it in range(1, iter_per_beta + 1):
            if time.time() >= deadline:
                return best_a, best_sc

            # --- convolution a∗a -------------------------------------------------
            conv = convolution_fft(a, a)               # length 2n‑1
            max_c = conv.max()
            # stable soft‑max weights
            exp_vals = np.exp(beta * (conv - max_c))
            p = exp_vals / exp_vals.sum()              # Σ p = 1

            # --- cross‑correlation p ⋆ a  (for the gradient) ------------------
            N = next_pow2(p.size + n - 1)
            P = np.fft.fft(p, N)
            A = np.fft.fft(a, N)
            cross = np.fft.ifft(P * np.conj(A)).real[:n]
            cross = np.maximum(cross, 0.0)

            # gradient of Φ_β
            grad = 4.0 * n * cross                     # non‑negative

            # exponentiated‑gradient step (step size ∝ 1/√t)
            eta = eta0 / np.sqrt(it)
            a = a * np.exp(-eta * grad)

            # projection onto the probability simplex (simple renormalisation)
            s = a.sum()
            if s == 0.0:
                a = np.full_like(a, 1.0 / n)
            else:
                a = a / s

            cur_sc = compute_score(a)
            if cur_sc < best_sc:
                best_sc = cur_sc
                best_a = a.copy()

    return best_a, best_sc


# ----------------------------------------------------------------------
# Exponentiated sub‑gradient on the true max‑convolution
# ----------------------------------------------------------------------


def _max_conv_gradient(a: np.ndarray, conv: np.ndarray) -> np.ndarray:
    """
    Sub‑gradient of max_k (a∗a)_k.
    For every active shift k (those attaining the maximum) we add
    the vector 2·a_{k‑j} to the gradient and finally average over the
    number of active shifts.
    """
    n = a.size
    max_val = conv.max()
    mask = np.isclose(conv, max_val, atol=1e-15, rtol=0.0)
    active = np.where(mask)[0]
    grad = np.zeros_like(a)

    for k in active:
        lo = max(0, k - (n - 1))
        hi = min(k, n - 1)
        idx = np.arange(lo, hi + 1, dtype=int)
        grad[idx] += 2.0 * a[k - idx]

    if active.size > 1:
        grad /= float(active.size)
    return grad


def optimise_subgradient(initial_a: np.ndarray,
                         deadline: float,
                         *,
                         eta0: float = 0.4,
                         max_iter: int = 100_000) -> Tuple[np.ndarray, float]:
    """
    Exponentiated sub‑gradient descent directly on Φ(a)=max_k (a∗a)_k.
    Returns the best vector seen (according to the fast score) and its score.
    """
    a = np.maximum(initial_a, 1e-30)
    a = a / a.sum()
    best_a = a.copy()
    best_sc = compute_score(a)
    n = a.size

    for it in range(1, max_iter + 1):
        if time.time() >= deadline:
            break

        conv = convolution_fft(a, a)
        grad = _max_conv_gradient(a, conv)

        eta = eta0 / np.sqrt(it)
        a = a * np.exp(-eta * grad)

        s = a.sum()
        a = a / s if s != 0.0 else np.full_like(a, 1.0 / n)

        cur_sc = compute_score(a)
        if cur_sc < best_sc:
            best_sc = cur_sc
            best_a = a.copy()

        # occasional progress report (useful when the budget is large)
        if it % 5000 == 0:
            print(f"    sub‑grad iter={it:,d} cur={cur_sc:.6f} best={best_sc:.6f}")

    return best_a, best_sc

def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs) -> List[float]:
    """
    Search for a non‑negative step‑function with a low ``evaluate_sequence``
    value. The routine respects the supplied wall‑clock budget and always
    returns a feasible list of floats.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 2.0          # safety margin

    best_seq: List[float] = []
    best_val: float = float("inf")

    # ------------------------------------------------------------------
    # 0.  optional fallback from a previously discovered construction
    # ------------------------------------------------------------------
    try:
        fallback = [float(x) for x in height_sequence_1]   # type: ignore
        val = evaluate_sequence(fallback)
        print(f"Fallback (height_sequence_1) score = {val:.6f}")
        if val < best_val:
            best_val = val
            best_seq = fallback
    except NameError:
        pass

    # ------------------------------------------------------------------
    # 1.  generate a large pool of cheap candidates and keep the best few
    # ------------------------------------------------------------------
    lengths = [256, 512, 1024, 2048]                # powers of two – FFT‑friendly
    all_candidates: List[Tuple[float, np.ndarray]] = []   # (fast_score, vector)

    for n in lengths:
        if time.time() >= deadline - 5.0:
            break
        print(f"\n=== Exploring length n = {n} ===")

        # ----- diverse seeds ------------------------------------------------
        seeds: List[np.ndarray] = []

        # uniform
        seeds.append(np.full(n, 1.0 / n))

        # classic arcsine
        seeds.append(arcsine_initial(n))

        # semicircle
        seeds.append(semicircle_initial(n))

        # random power‑law shapes
        for _ in range(4):
            p = rng.uniform(0.2, 1.2)
            eps = rng.uniform(0.0, max(1.0, 0.02 * n))
            seeds.append(generate_powerlaw(n, p, eps))

        # beta‑like shapes
        for _ in range(3):
            alpha = rng.uniform(0.4, 5.0)
            beta = rng.uniform(0.4, 5.0)
            seeds.append(generate_beta_like(n, alpha, beta))

        # Dirichlet / exponential draws
        for conc in [0.3, 1.0, 2.0]:
            seeds.append(random_dirichlet(n, conc, rng))

        # mixtures with arcsine (helps to escape local basins)
        arc = arcsine_initial(n)
        for _ in range(3):
            w = rng.random()
            mix = mixture(arc, np.full(n, 1.0 / n), w)
            seeds.append(mix)

        for _ in range(2):
            w = rng.random()
            mix = mixture(arc, semicircle_initial(n), w)
            seeds.append(mix)

        # ----- cheap ranking -------------------------------------------------
        scored = [(compute_score(s), s) for s in seeds]
        scored.sort(key=lambda x: x[0])
        # keep the five most promising from this length
        for sc, vec in scored[:5]:
            all_candidates.append((sc, vec))

    # ------------------------------------------------------------------
    # 2.  keep the global ten best cheap candidates (after the length loop)
    # ------------------------------------------------------------------
    all_candidates.sort(key=lambda x: x[0])
    top_global = [vec for _, vec in all_candidates[:10]]

    # ------------------------------------------------------------------
    # 3.  refine each top candidate with slower, more accurate methods
    # ------------------------------------------------------------------
    for idx, cand in enumerate(top_global):
        if time.time() >= deadline - 3.0:
            break
        n = cand.size
        print(f"\n--- Refinement of candidate {idx + 1}/{len(top_global)} (n={n}) ---")

        # ---- Simulated annealing ------------------------------------------------
        max_iter_sa = int(min(3000,
                              max(500, (deadline - time.time()) * 250 / n)))
        sa_start_temp = max(0.005, 0.02 * compute_score(cand))
        sa_vec, _ = simulated_annealing(cand, deadline, rng,
                                        max_iter=max_iter_sa,
                                        start_temp=sa_start_temp,
                                        cooling=0.9995,
                                        max_step=0.02)

        # evaluate with the official routine (only a few times!)
        val_sa = evaluate_sequence(sa_vec.tolist())
        print(f"    SA evaluated score = {val_sa:.6f}")
        if val_sa < best_val:
            best_val = val_sa
            best_seq = sa_vec.tolist()
            print(f"    *** New global best (SA) {best_val:.6f} ***")

        # ---- Smooth‑max gradient descent ----------------------------------------
        betas = [5, 15, 30, 60, 120, 240]          # increasingly tight approximation
        it_per_beta = max(100,
                          int(min(800,
                                  (deadline - time.time()) * 120 / (len(betas) * n))))
        sg_vec, _ = softmax_gradient_descent(sa_vec, deadline,
                                             betas=betas,
                                             iter_per_beta=it_per_beta,
                                             eta0=0.4)

        val_sg = evaluate_sequence(sg_vec.tolist())
        print(f"    smooth‑gd evaluated score = {val_sg:.6f}")
        if val_sg < best_val:
            best_val = val_sg
            best_seq = sg_vec.tolist()
            print(f"    *** New global best (smooth‑gd) {best_val:.6f} ***")

    # ------------------------------------------------------------------
    # 4.  Final polishing on the overall best vector (if any time left)
    # ------------------------------------------------------------------
    if best_seq and time.time() < deadline - 2.0:
        print("\nFinal polishing with true‑max sub‑gradient")
        a0 = np.array(best_seq, dtype=float)
        remaining = max(0.0, deadline - time.time())
        max_iter = int(min(150_000, remaining * 250))
        polished, _ = optimise_subgradient(a0, deadline,
                                           eta0=0.35,
                                           max_iter=max_iter)
        final_val = evaluate_sequence(polished.tolist())
        print(f"Final polished score = {final_val:.6f}")
        if final_val < best_val:
            best_val = final_val
            best_seq = polished.tolist()
            print("*** Updated global best after final polishing ***")

    # ------------------------------------------------------------------
    # 5.  Guard – always return a feasible sequence.
    # ------------------------------------------------------------------
    if not best_seq:
        print("No feasible candidate found – falling back to uniform vector.")
        n = 256
        best_seq = np.full(n, 1.0 / n).tolist()
        best_val = evaluate_sequence(best_seq)

    print(f"\nSearch finished. Best reported score = {best_val:.6f}")
    return best_seq
