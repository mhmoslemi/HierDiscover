```
def exp_grad_adam(a, ...):
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1=0.9, beta2=0.999
    eps=1e-8
    for t in range(1, max_iter+1):
        conv = convolution_fft(a, a)
        max_val = np.max(conv)
        # compute softmax weights
        w = np.exp(beta * (conv - max_val))
        w /= w.sum()
        grad = gradient_from_weights(a, w)
        # Adam update
        m = beta1*m + (1-beta1)*grad
        v = beta2*v + (1-beta2)*grad**2
        m_hat = m/(1-beta1**t)
        v_hat = v/(1-beta2**t)
        step = eta/(np.sqrt(v_hat)+eps)
        a = a * np.exp(-step*m_hat)
        a = normalize(a)
```

import time, numpy as np

def next_pow2(x): return 1 << (x-1).bit_length()

def convolution_fft(a,b):
    m = len(a)+len(b)-1
    n_fft = next_pow2(m)
    A = np.fft.rfft(a, n=n_fft)
    B = np.fft.rfft(b, n=n_fft)
    C = A*B
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv

def normalize(v):
    s = v.sum()
    if s <= 0:
        return np.full_like(v, 1.0/len(v))
    return v / s

def compute_score(a):
    n = len(a)
    conv = convolution_fft(a, a)
    max_val = np.max(conv)
    s = a.sum()
    if s < 1e-12:
        return np.inf
    return 2.0*n*max_val/(s*s)

def gradient_from_weights(a, w):
    # w length = 2*len(a)-1
    w_rev = w[::-1]
    conv = convolution_fft(a, w_rev)
    idx = 2*len(a)-2
    grads = 2.0 * conv[idx - np.arange(len(a))]
    return grads

def arcsine_initial(n):
    i = np.arange(n, dtype=float)
    eps = 0.25
    vals = 1.0/np.sqrt((i+0.5)*(n-0.5-i)+eps)
    return normalize(vals)

def powerlaw_shape(n, p, eps):
    i = np.arange(n, dtype=float)
    base = ((i+eps)*(n-1-i+eps))**(-p)
    return normalize(base)

def uniform_shape(n):
    return np.full(n, 1.0/n)

def random_dirichlet_alpha(n, rng, alpha):
    # Dirichlet with same concentration for all entries
    samp = rng.gamma(alpha, 1.0, size=n)
    return normalize(samp)

def smooth_descent(a, deadline, eta0=1.0, beta_start=2.0, beta_end=30.0):
    a = normalize(a.astype(float).copy())
    n = len(a)
    best_a = a.copy()
    conv = convolution_fft(a, a)
    best_sc = 2.0 * n * np.max(conv)
    it = 0
    start = time.time()
    while time.time() < deadline:
        it += 1
        elapsed = time.time() - start
        # linear schedule for beta
        beta = beta_start + (beta_end - beta_start) * min(1.0, elapsed / (deadline - start))
        conv = convolution_fft(a, a)
        max_val = np.max(conv)
        # soft weights
        exp_vals = np.exp(beta * (conv - max_val))
        w = exp_vals / exp_vals.sum()
        grad = gradient_from_weights(a, w)
        eta = eta0 / np.sqrt(it)
        a = a * np.exp(-eta * grad)
        a = normalize(a)
        cur_sc = 2.0 * n * max_val
        if cur_sc < best_sc:
            best_sc = cur_sc
            best_a = a.copy()
    return best_a, best_sc

def subgradient_descent(a, deadline, eta0=0.7):
    a = normalize(a.astype(float).copy())
    n = len(a)
    best_a = a.copy()
    conv = convolution_fft(a, a)
    best_sc = 2.0 * n * np.max(conv)
    it = 0
    while time.time() < deadline:
        it += 1
        conv = convolution_fft(a, a)
        max_val = np.max(conv)
        # indices achieving max (within tolerance)
        tol = 1e-14
        max_idx = np.where(conv >= max_val - tol)[0]
        w = np.zeros_like(conv)
        if max_idx.size > 0:
            w[max_idx] = 1.0 / max_idx.size
        grad = gradient_from_weights(a, w)
        eta = eta0 / np.sqrt(it)
        a = a * np.exp(-eta * grad)
        a = normalize(a)
        cur_sc = 2.0 * n * max_val
        if cur_sc < best_sc:
            best_sc = cur_sc
            best_a = a.copy()
    return best_a, best_sc

def local_random_perturb(a, iters, step, rng):
    # a sum=1, length n
    n = len(a)
    best = a.copy()
    best_sc = compute_score(best)
    for _ in range(iters):
        i, j = rng.choice(n, size=2, replace=False)
        # propose moving up to step * best[i] from i to j
        max_move = min(step, best[i])
        if max_move <= 0:
            continue
        delta = rng.uniform(0, max_move)
        cand = best.copy()
        cand[i] -= delta
        cand[j] += delta
        # ensure nonnegative
        if cand[i] < 0:
            continue
        sc = compute_score(cand)
        if sc < best_sc:
            best_sc = sc
            best = cand
    return best, best_sc

                print("[fallback] added length", n)

    scored.sort(key=lambda x: x[0])
    # keep top K for deeper refinement

                best_score = val2

                print(f"New global best after hard polish (n={n}) score={val2:.6f}")

"""
Improved multi‑start optimisation for the step‑function objective

    2·n·max(conv(a,a)) / (sum(a)²)

The algorithm works as follows:

1. **Candidate pool** – for several lengths (1024, 2048, 4096) we create a
   diverse set of initial vectors: arcsine shape, uniform, random
   Dirichlet draws with many concentration parameters and a few random
   power‑law shapes.  If a historic construction `height_sequence_1` is
   present it is interpolated to the considered lengths.

2. **Quick screening** – the candidates are ranked with a fast FFT‑based
   score (`compute_score`).  The ten best are kept for deeper work.

3. **Fast soft‑max mirror descent** – each kept candidate is refined with
   an exponentiated‑gradient (mirror‑descent) method whose “temperature”
   β is annealed from a small to a larger value while the step‑size
   η decays as 1/√t.  This yields a smooth approximation of the true
   sub‑gradient and quickly reduces the objective.

4. **Hard sub‑gradient polishing** – the best of the smooth runs is
   further optimised with a true max‑sub‑gradient descent (uniform weight
   on all maximising convolution entries).  The same exponentiated‑gradient
   update rule is used but with the exact max‑weight.

5. **Brief random perturbation** – a last short stochastic hill‑climb
   (mass transfer between two random coordinates) may squeeze a little
   more out of the final vector.

All work respects the supplied time budget (default 1000 s) and always
returns a feasible non‑negative list of floats.
"""

import time
import numpy as np
from typing import List, Tuple

# ----------------------------------------------------------------------
# Low‑level numeric utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """
    Linear convolution via an FFT of size a power of two.
    Returns the full (len(a)+len(b)-1) result.
    """
    m = len(a) + len(b) - 1
    n_fft = next_pow2(m)
    A = np.fft.rfft(a, n=n_fft)
    B = np.fft.rfft(b, n=n_fft)
    C = A * B
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


def normalize(v: np.ndarray) -> np.ndarray:
    """Project a non‑negative vector onto the simplex (sum = 1)."""
    s = v.sum()
    if s <= 0.0:
        return np.full_like(v, 1.0 / len(v))
    return v / s


def compute_score(a: np.ndarray) -> float:
    """
    Fast evaluation of the official objective (scale‑invariant):
        2·n·max(conv(a,a)) / (sum(a)²)
    The vector a is assumed non‑negative.
    """
    n = len(a)
    conv = convolution_fft(a, a)
    max_val = float(np.max(conv))
    s = float(np.sum(a))
    if s < 1e-12:
        return float("inf")
    return 2.0 * n * max_val / (s * s)


def gradient_from_weights(a: np.ndarray, w: np.ndarray) -> np.ndarray:
    """
    Gradient of Σ_k w_k·(a*a)_k with respect to a.
    Implements   g_j = 2· Σ_i a_i·w_{i+j}.
    Computed via one FFT‑based convolution.
    """
    w_rev = w[::-1]
    conv = convolution_fft(a, w_rev)               # length = 3·n‑2
    idx = 2 * len(a) - 2                            # position 2n‑2
    grads = 2.0 * conv[idx - np.arange(len(a))]
    return grads


# ----------------------------------------------------------------------
# Parametric families for the initial pool
# ----------------------------------------------------------------------


def arcsine_initial(n: int) -> np.ndarray:
    """Discrete arcsine distribution ≈ 1/√(i·(n‑1‑i)) (normalised)."""
    i = np.arange(n, dtype=float)
    eps = 0.25
    vals = 1.0 / np.sqrt((i + 0.5) * (n - 0.5 - i) + eps)
    return normalize(vals)


def powerlaw_shape(n: int, p: float, eps: float) -> np.ndarray:
    """Symmetric power‑law: a_i ∝ ((i+ε)(n‑1‑i+ε))⁻ᵖ."""
    i = np.arange(n, dtype=float)
    base = ((i + eps) * (n - 1 - i + eps)) ** (-p)
    return normalize(base)


def uniform_shape(n: int) -> np.ndarray:
    """Uniform vector on n points."""
    return np.full(n, 1.0 / n)


def random_dirichlet_alpha(n: int, rng: np.random.Generator,
                          alpha: float) -> np.ndarray:
    """
    Dirichlet draw with a common concentration parameter `alpha`.
    Larger `alpha` → smoother, smaller → spikier.
    """
    if alpha <= 0.0:
        alpha = 0.1
    samp = rng.gamma(alpha, 1.0, size=n)
    return normalize(samp)


# ----------------------------------------------------------------------
# Refinement methods
# ----------------------------------------------------------------------


def smooth_descent(a: np.ndarray, deadline: float,
                   *, eta0: float = 1.0,
                   beta_start: float = 2.0,
                   beta_end: float = 30.0) -> Tuple[np.ndarray, float]:
    """
    Exponentiated‑gradient descent using a *soft* max weight:
        w_i ∝ exp(β·(conv_i – max(conv))).

    β is linearly annealed from `beta_start` to `beta_end` while the
    step‑size η follows η₀/√t.  The best iterate (exact objective) is
    returned together with its exact score.
    """
    a = normalize(a.astype(float).copy())
    n = len(a)
    best_a = a.copy()
    conv = convolution_fft(a, a)
    best_sc = 2.0 * n * np.max(conv)               # exact score (sum=1)

    it = 0
    start = time.time()
    total = max(deadline - start, 1e-12)

    while time.time() < deadline:
        it += 1
        # annealed temperature
        beta = beta_start + (beta_end - beta_start) * min(1.0,
                                                          (time.time() - start) / total)

        conv = convolution_fft(a, a)
        max_val = float(np.max(conv))

        # softmax‑type weights on the convolution entries
        exp_vals = np.exp(beta * (conv - max_val))   # max entry gets weight 1
        w = exp_vals / np.sum(exp_vals)             # ∑ w = 1

        # gradient of the smoothed objective
        grad = gradient_from_weights(a, w)

        # step‑size schedule
        eta = eta0 / np.sqrt(it)

        # exponentiated‑gradient update + simplex projection
        a = a * np.exp(-eta * grad)
        a = normalize(a)

        cur_sc = 2.0 * n * max_val
        if cur_sc < best_sc:
            best_sc = cur_sc
            best_a = a.copy()

    return best_a, best_sc


def subgradient_descent(a: np.ndarray, deadline: float,
                        *, eta0: float = 0.7) -> Tuple[np.ndarray, float]:
    """
    Classic sub‑gradient descent on the *true* max‑convolution objective.
    Uses the exact max‑weight (uniform on all maximisers) and an
    exponentiated‑gradient update with step‑size η₀/√t.
    Returns the best iterate and its exact score.
    """
    a = normalize(a.astype(float).copy())
    n = len(a)
    best_a = a.copy()
    conv = convolution_fft(a, a)
    best_sc = 2.0 * n * float(np.max(conv))

    it = 0
    while time.time() < deadline:
        it += 1
        conv = convolution_fft(a, a)
        max_val = float(np.max(conv))

        # hard max‑weight
        w = np.zeros_like(conv)
        max_idx = np.where(conv >= max_val - 1e-14)[0]
        if max_idx.size:
            w[max_idx] = 1.0 / max_idx.size

        grad = gradient_from_weights(a, w)

        eta = eta0 / np.sqrt(it)
        a = a * np.exp(-eta * grad)
        a = normalize(a)

        cur_sc = 2.0 * n * max_val
        if cur_sc < best_sc:
            best_sc = cur_sc
            best_a = a.copy()

    return best_a, best_sc


def local_random_perturb(a: np.ndarray, moves: int, step: float,
                         rng: np.random.Generator) -> Tuple[np.ndarray, float]:
    """
    Very cheap stochastic hill‑climbing: repeatedly move a small amount
    of mass from a random coordinate to another.  The sum stays exactly
    one, so `compute_score` can be used directly.
    """
    best = a.copy()
    best_sc = compute_score(best)
    n = len(a)

    for _ in range(moves):
        i, j = rng.choice(n, size=2, replace=False)
        # take at most `step` of the mass from i
        max_move = min(step, best[i])
        if max_move <= 0.0:
            continue
        delta = rng.uniform(0.0, max_move)
        cand = best.copy()
        cand[i] -= delta
        cand[j] += delta
        sc = compute_score(cand)
        if sc < best_sc:
            best_sc = sc
            best = cand
    return best, best_sc


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs) -> List[float]:
    """
    Search for a non‑negative step‑function minimising the official
    evaluation metric.  A pool of diverse initial vectors is built,
    the ten most promising are refined with a temperature‑annealed
    mirror‑descent, the best of those is finally polished with a hard
    sub‑gradient method and a tiny random perturbation.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    # small safety margin to avoid a hard timeout
    deadline = start_time + budget_s - 2.0

    best_seq: List[float] = []
    best_val: float = float("inf")

    # ------------------------------------------------------------------
    # 1. Build a diverse pool of initial candidates
    # ------------------------------------------------------------------
    n_candidates = [1024, 2048, 4096]
    raw_candidates: List[Tuple[int, np.ndarray]] = []

    for n in n_candidates:
        # deterministic families
        raw_candidates.append((n, arcsine_initial(n)))
        raw_candidates.append((n, uniform_shape(n)))

        # Dirichlet draws with several concentration parameters
        for alpha in [0.2, 0.5, 1.0, 2.0, 5.0]:
            raw_candidates.append((n, random_dirichlet_alpha(n, rng, alpha)))

        # random power‑law shapes
        for _ in range(3):
            p = rng.uniform(0.2, 0.8)
            eps = rng.uniform(0.0, max(5.0, 0.02 * n))
            raw_candidates.append((n, powerlaw_shape(n, p, eps)))

    # optional historic construction
    try:
        prev = np.array([float(x) for x in height_sequence_1], dtype=float)
        if prev.ndim == 1:
            for n in n_candidates:
                # linear interpolation to the target length
                if len(prev) == n:
                    cand = normalize(prev)
                else:
                    x_old = np.linspace(0, 1, len(prev))
                    x_new = np.linspace(0, 1, n)
                    cand = np.interp(x_new, x_old, prev)
                    cand = normalize(cand)
                raw_candidates.append((n, cand))
                print("[fallback] interpolated length =", n)
    except NameError:
        pass

    # ------------------------------------------------------------------
    # 2. Quick ranking with the fast internal score
    # ------------------------------------------------------------------
    scored = [(compute_score(c), n, c) for (n, c) in raw_candidates]
    scored.sort(key=lambda x: x[0])
    top_k = 10
    top_initial = scored[:top_k]

    print(f"Quick ranking – best fast score = {top_initial[0][0]:.6f}")

    # ------------------------------------------------------------------
    # 3. Refine each of the top initial candidates
    # ------------------------------------------------------------------
    for fast_sc, n, cand in top_initial:
        if time.time() >= deadline - 5.0:
            break

        # allocate roughly half of the remaining time to the smooth phase,
        # the rest will be used for the final hard polish (outside the loop)
        remaining = max(0.0, deadline - time.time())
        smooth_budget = min(30.0, remaining * 0.5)      # at most 30 s per candidate
        smooth_deadline = time.time() + smooth_budget

        a_smooth, _ = smooth_descent(
            a=cand,
            deadline=smooth_deadline,
            eta0=1.0,
            beta_start=2.0,
            beta_end=30.0,
        )

        # evaluate with the official routine; this is the first reliable
        # bound we obtain for this candidate
        val = evaluate_sequence(a_smooth.tolist())
        if val < best_val:
            best_val = val
            best_seq = a_smooth.tolist()
            print(f"*** New global best after smooth (n={n}) score={val:.6f}")

    # ------------------------------------------------------------------
    # 4. Hard sub‑gradient polishing of the currently best vector
    # ------------------------------------------------------------------
    if best_seq and time.time() < deadline - 5.0:
        a_start = np.array(best_seq, dtype=float)
        a_polished, _ = subgradient_descent(
            a=a_start,
            deadline=deadline,
            eta0=0.7,
        )
        val = evaluate_sequence(a_polished.tolist())
        if val < best_val:
            best_val = val
            best_seq = a_polished.tolist()
            print(f"*** Updated global best after hard polish score={val:.6f}")

    # ------------------------------------------------------------------
    # 5. Very short random perturbation (few thousand moves)
    # ------------------------------------------------------------------
    if best_seq and time.time() < deadline - 1.0:
        a_best = np.array(best_seq, dtype=float)
        # number of moves proportional to the time that is left
        moves = int(2000 * (deadline - time.time()))
        if moves > 0:
            a_best, _ = local_random_perturb(a_best, moves=moves,
                                             step=0.01, rng=rng)
            val = evaluate_sequence(a_best.tolist())
            if val < best_val:
                best_val = val
                best_seq = a_best.tolist()
                print(f"*** Updated best after random tweak score={val:.6f}")

    # ------------------------------------------------------------------
    # 6. Guard – always return a feasible sequence.
    # ------------------------------------------------------------------
    if not best_seq:
        best_seq = uniform_shape(256).tolist()
        best_val = evaluate_sequence(best_seq)
        print("*** Fallback to uniform sequence ***")

    print(f"\nSearch finished. Best reported score = {best_val:.6f}")
    return best_seq
