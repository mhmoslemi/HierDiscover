```
def propose_candidate(seed=0, budget_s=1000):
    start = time.time()
    deadline = start + budget_s - 2.0

    n = 2048   # can try 4096 if time.

    np.random.seed(seed)

    a = arcsine_initial(n)  # shape already sum=1
    best_a = a
    best_score = compute_score(a)

    # LP refinement loop
    iteration = 0
    while time.time() < deadline - 10.0 and iteration < 30:
        M = max_convolution(a)
        # Build constraints matrix (sparse)
        A = build_convolution_constraints(a)
        # Solve LP
        res = linprog(c=-np.ones(n), A_ub=A, b_ub=np.full(2*n-1, M), bounds=[(0, None)]*n, method='highs')
        if not res.success:
            print("LP failed:", res.message)
            break
        b = res.x
        B = b.sum()
        if B <= 1.000001:
            print(f"LP iteration {iteration}: no mass increase (B={B}).")
            break
        scale = 1.0 / B
        a_new = b * scale
        sc = compute_score(a_new)
        print(f"LP iteration {iteration}: B={B:.6f}, scale={scale:.6f}, score={sc:.6f}")
        if sc < best_score:
            best_score = sc
            best_a = a_new.copy()
        a = a_new
        iteration += 1

    # Additional polishing with Adam (fast surrogate)
    if time.time() < deadline - 5.0:
        remaining = deadline - time.time()
        max_iter_adam = int(min(200_000, remaining * 30))
        print(f"Polishing with Adam, max_iter={max_iter_adam}")
        a_polished, sc_adam = adam_optimize(best_a, deadline, max_iter=max_iter_adam)
        if sc_adam < best_score:
            best_score = sc_adam
            best_a = a_polished.copy()

    # Subgradient descent for true max
    if time.time() < deadline - 3.0:
        remaining = deadline - time.time()
        max_iter_sub = int(min(100_000, remaining * 30))
        print(f"Subgradient polishing, max_iter={max_iter_sub}")
        a_polished, sc_sub = subgradient_optimize(best_a, deadline, max_iter=max_iter_sub)
        if sc_sub < best_score:
            best_score = sc_sub
            best_a = a_polished.copy()

    # Final evaluation
    final_val = evaluate_sequence(best_a.tolist())
    print(f"Final official score = {final_val:.6f}")
    return best_a.tolist()
```

```
def build_convolution_constraints(a):
    n = a.shape[0]
    rows = []
    cols = []
    data = []
    for k in range(2*n-1):
        i_start = max(0, k-(n-1))
        i_end = min(n-1, k)
        # i runs from i_start to i_end inclusive
        for i in range(i_start, i_end+1):
            rows.append(k)
            j = k - i
            cols.append(j)
            data.append(a[i])
    return csr_matrix((data, (rows, cols)), shape=(2*n-1, n))
```

assistantfinal**Solution Overview**

The score to minimise is  

\[
\text{score}(a)=\frac{2n\;\max (a*a)}{(\sum_i a_i)^2},
\qquad a_i\ge 0 .
\]

If we keep the vector on the simplex (`∑a_i = 1`) the denominator disappears and we only have to minimise  
`max(a*a)`.  
For a fixed vector `a` the following linear programme is exact:

* maximise `∑ b_i`  
* subject to `b_i ≥ 0` and `a * b ≤ M` element‑wise,  

where `M = max(a*a)`.  
If `B = ∑ b_i > 1` we can scale `b` back to the simplex (`b ← b / B`).  
All constraints are multiplied by `1/B`, therefore the new maximum convolution becomes `M/B < M`.  
Repeating this LP‑step monotonically decreases the objective and converges to a (near) optimal shape.

The algorithm is:

1. **Initial shape** – an arcsine distribution, normalised to the simplex.
2. **LP refinement** – up to 15 LP iterations (the matrix is built on the fly, size `≈ n²`).
3. **Polishing** – a short Adam optimisation on a smooth log‑sum‑exp surrogate,
   followed by a few true sub‑gradient steps.
4. **Final evaluation** – using the provided `evaluate_sequence`.

All steps respect the global 1000 s budget and always return a non‑empty vector.

```python
"""
Propose a step‑function (non‑negative vector) with a low official score.

Algorithm
---------

1.  Start from an arcsine shape (sum = 1).
2.  Repeatedly solve the linear programme

        maximise   Σ b_i
        subject to  b_i ≥ 0
                    (a * b)[k] ≤ M   for every k

    where M = max(a*a).  The feasible region depends on `a`, therefore after each
    LP we obtain a vector `b` with larger ℓ₁‑norm `B`.  Scaling `b` back to the
    simplex reduces the maximal self‑convolution to `M / B`.  This monotone
    improvement is iterated until no more increase in ℓ₁‑norm is possible.

3.  The resulting vector is refined with a few Adam steps on a smoothed
    surrogate (log‑sum‑exp of the convolution) and a short projected
    sub‑gradient descent on the true max‑convolution.

4.  The best vector found is returned as a plain Python list.
"""

import time
import math
from typing import List, Tuple

import numpy as np
from scipy.optimize import linprog

# ----------------------------------------------------------------------
# Basic linear‑algebra utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Linear convolution using a real FFT."""
    m = a.size + b.size - 1
    n_fft = next_pow2(m)
    A = np.fft.rfft(a, n=n_fft)
    B = np.fft.rfft(b, n=n_fft)
    C = A * B
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


def max_convolution(a: np.ndarray) -> float:
    """Maximum entry of the self‑convolution of `a`."""
    return float(np.max(convolution_fft(a, a)))


def compute_score(a: np.ndarray) -> float:
    """
    Fast surrogate (valid when Σ a = 1):
        2·n·max(a*a)
    """
    n = a.shape[0]
    return 2.0 * n * max_convolution(a)


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection onto the probability simplex {x ≥ 0, Σx = 1}.
    Implementation of Duchi et al. (2008).
    """
    n = v.shape[0]
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / n)

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho = np.where(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Shape generators
# ----------------------------------------------------------------------


def arcsine_initial(n: int) -> np.ndarray:
    """
    Classic arcsine shape (symmetrised) and normalised to the simplex.
    """
    i = np.arange(n, dtype=float)
    eps = 0.25
    vals = 1.0 / np.sqrt((i + 0.5) * (n - 0.5 - i) + eps)
    vals /= vals.sum()
    return vals


# ----------------------------------------------------------------------
# Linear‑programming refinement
# ----------------------------------------------------------------------


def build_conv_matrix(a: np.ndarray) -> np.ndarray:
    """
    Construct the dense matrix `A` (size (2n‑1) × n) such that A @ b = a * b.
    Column i contains the vector `a` starting at row i.
    """
    n = a.shape[0]
    m = 2 * n - 1
    A = np.zeros((m, n), dtype=float)
    for i in range(n):
        A[i:i + n, i] = a
    return A


def lp_refine(a: np.ndarray, deadline: float,
              max_iters: int = 15) -> Tuple[np.ndarray, float]:
    """
    Perform up‑to `max_iters` LP refinements.
    Returns the best vector (still on the simplex) and its fast score.
    """
    n = a.shape[0]
    best_a = a.copy()
    best_sc = compute_score(a)

    for it in range(max_iters):
        if time.time() >= deadline - 5.0:
            break

        M = max_convolution(a)

        # Build the convolution matrix for the current a
        A = build_conv_matrix(a)
        rhs = np.full(2 * n - 1, M)

        # maximise Σ b_i  → minimise -Σ b_i
        c = -np.ones(n)

        res = linprog(c, A_ub=A, b_ub=rhs,
                      bounds=[(0, None)] * n,
                      method='highs')

        if not res.success:
            print(f"LP iteration {it}: solver failed ({res.message})")
            break

        b = res.x
        B = b.sum()

        if B <= 1.0 + 1e-8:
            print(f"LP iteration {it}: no ℓ₁‑increase (B={B:.6f})")
            break

        # Scale back onto the simplex
        a_new = b / B
        sc = compute_score(a_new)

        print(f"LP iteration {it}: B={B:.6f}, score={sc:.6f}")

        if sc < best_sc:
            best_sc = sc
            best_a = a_new.copy()

        a = a_new

    return best_a, best_sc


# ----------------------------------------------------------------------
# Smooth surrogate (log‑sum‑exp) and Adam optimiser
# ----------------------------------------------------------------------


def smooth_gradient(a: np.ndarray, lam: float) -> Tuple[np.ndarray, float]:
    """
    Gradient of the log‑sum‑exp surrogate of max(conv(a,a)).
    Returns (gradient, surrogate value).
    """
    n = a.shape[0]
    conv = convolution_fft(a, a)          # length 2n‑1
    max_c = np.max(conv)

    exp_term = np.exp(lam * (conv - max_c))
    sum_exp = np.sum(exp_term)
    w = exp_term / sum_exp                # soft‑max weights on convolution entries
    B = max_c + np.log(sum_exp) / lam     # smooth approximation of the max

    # Derivative of Σ_k w_k (a*a)_k  w.r.t. a:
    #   ∂/∂a_i   = 2 * Σ_k w_k a_{k-i}
    # This can be written as a convolution with the reversed weight vector.
    w_rev = w[::-1]
    conv_rev = convolution_fft(a, w_rev)   # length 3n‑2
    L = 2 * n - 1
    weighted_grad = 2.0 * conv_rev[L - 1 - np.arange(n)]

    s = np.sum(a)
    term1 = (2.0 * n / (s ** 2)) * weighted_grad
    term2 = -(4.0 * n * B / (s ** 3)) * np.ones_like(a)
    grad = term1 + term2
    return grad, B


def adam_optimize(initial_a: np.ndarray,
                  deadline: float,
                  max_iter: int = 150_000,
                  lr0: float = 0.15,
                  lam_start: float = 1.0,
                  lam_end: float = 200.0,
                  noise_every: int = 1000,
                  noise_scale: float = 4e-5) -> Tuple[np.ndarray, float]:
    """
    Adam optimiser on the smooth log‑sum‑exp surrogate.
    Returns the best vector (still on the simplex) and its fast score.
    """
    a = project_to_simplex(initial_a.astype(float).copy())
    best_a = a.copy()
    best_sc = compute_score(a)

    beta1, beta2 = 0.9, 0.999
    eps_adam = 1e-8
    m = np.zeros_like(a)
    v = np.zeros_like(a)

    n = a.shape[0]

    for it in range(1, max_iter + 1):
        if time.time() >= deadline:
            break

        t = it / max_iter
        lam = lam_start + (lam_end - lam_start) * (t ** 0.5)

        grad, _ = smooth_gradient(a, lam)

        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)

        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        step = lr0 * m_hat / (np.sqrt(v_hat) + eps_adam)
        a_new = a - step

        if noise_every and (it % noise_every == 0):
            a_new += np.random.randn(n) * noise_scale

        a_new = project_to_simplex(a_new)

        cur_sc = compute_score(a_new)
        if cur_sc < best_sc:
            best_sc = cur_sc
            best_a = a_new.copy()

        a = a_new

        if it % 5_000 == 0:
            print(f"    Adam iter={it:,d} cur={cur_sc:.6f} best={best_sc:.6f}")

    return best_a, best_sc

# Exact sub‑gradient descent (true max‑convolution)
# ----------------------------------------------------------------------


def subgradient_optimize(initial_a: np.ndarray,
                         deadline: float,
                         step0: float = 0.3,
                         max_iter: int = 200_000) -> Tuple[np.ndarray, float]:
    """
    Projected sub‑gradient descent on the *true* max‑convolution.
    Returns the best vector (projected) and its fast score.
    """
    a = initial_a.copy()
    best_a = a.copy()
    best_sc = compute_score(a)

    n = a.shape[0]

    for it in range(1, max_iter + 1):
        if time.time() >= deadline:
            break

        conv = convolution_fft(a, a)
        max_val = float(np.max(conv))
        idx = int(np.argmax(conv))

        # Sub‑gradient for the active quadratic term
        i = np.arange(n)
        j = idx - i
        mask = (j >= 0) & (j < n)
        grad = np.zeros_like(a)
        grad[mask] = 2.0 * a[j[mask]]

        step = step0 / np.sqrt(it)
        for _ in range(12):
            a_new = a - step * grad
            a_new = project_to_simplex(a_new)
            new_val = max_convolution(a_new)
            if new_val < max_val - 1e-14:
                break
            step *= 0.5

        cur_sc = compute_score(a_new)
        if cur_sc < best_sc:
            best_sc = cur_sc
            best_a = a_new.copy()

        a = a_new

        if it % 40_000 == 0:
            print(f"    Sub‑grad iter={it:,d} cur={cur_sc:.6f} best={best_sc:.6f}")

    return best_a, best_sc

    The search consists of:
    * an arcsine start,

    * a short Adam polishing on a smooth surrogate,
    * a few true sub‑gradient steps,
    * final official evaluation.
    """

    start_time = time.time()

    # ------------------------------------------------------------------

    # ------------------------------------------------------------------

    # ------------------------------------------------------------------
    if time.time() < deadline - 8.0:

        max_iter_adam = int(min(120_000, remaining * 30))
        print(f"Adam polishing (max_iter={max_iter_adam}) …")
        a, adam_score = adam_optimize(
            initial_a=a,
            deadline=deadline,
            max_iter=max_iter_adam,
            lr0=0.12,
            lam_start=1.0,
            lam_end=250.0,
            noise_every=1500,
            noise_scale=4e-5,
        )
        print(f"  Adam best fast score = {adam_score:.6f}")

    # ------------------------------------------------------------------
    # 4.  Exact sub‑gradient descent (true max‑convolution)
    # ------------------------------------------------------------------
    if time.time() < deadline - 4.0:
        remaining = deadline - time.time()
        max_iter_sub = int(min(100_000, remaining * 30))

            a,
            deadline=deadline,
            step0=0.20,
            max_iter=max_iter_sub,
        )

    # ------------------------------------------------------------------
    # 5.  Final official evaluation
    # ------------------------------------------------------------------
    final_val = evaluate_sequence(a.tolist())
    print(f"\nFinal official score = {final_val:.6f}")

    # Guard – always return a non‑empty list
    if not a.size:
        a = np.ones(256) / 256.0
        final_val = evaluate_sequence(a.tolist())
        print("Returned fallback uniform vector (score: {:.6f})".format(final_val))

    return a.tolist()
```
