"""
Propose a non‑negative step sequence with a low value of
    2·n·max(conv(a,a)) / (Σ a)²

The algorithm proceeds in four stages:

1. **Exploratory sampling** – for several vector lengths we draw many
   beta‑family shapes (including mixtures) and keep the few best according
   to a fast surrogate 2·n·max(conv(a,a)) (the denominator is 1 because we
   project onto the simplex).

2. **Long Adam optimisation** – each survivor is refined by Adam on a
   soft‑max surrogate of the max‑convolution.  The temperature is annealed
   up to a large value (λ≈400) so the surrogate is very close to the true
   max.  The whole run is limited only by the remaining time budget.

3. **Exact sub‑gradient descent** – after Adam we switch to the true
   max‑convolution and perform a projected sub‑gradient descent with a
   back‑tracking line‑search.  This guarantees that the official objective
   never increases.

4. **Final polishing** – a cheap Gaussian walk (with occasional uphill
   moves) is applied as a last fine‑tuning step.

All stages respect the global 1000 s budget and keep track of the best
vector seen (according to the official ``evaluate_sequence``).  If nothing
is found a safe uniform fallback is returned.
"""

import time
import numpy as np
from typing import List, Tuple

# ----------------------------------------------------------------------
# Linear‑algebra utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Linear convolution using a real FFT (fast for all n)."""
    m = a.size + b.size - 1
    n_fft = next_pow2(m)
    A = np.fft.rfft(a, n=n_fft)
    B = np.fft.rfft(b, n=n_fft)
    C = A * B
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


def max_convolution(a: np.ndarray) -> float:
    """Maximum entry of the self‑convolution of a."""
    return float(np.max(convolution_fft(a, a)))


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection onto the probability simplex {x≥0, Σx=1}.
    Duchi et al. (2008) implementation.
    """
    n = v.shape[0]
    # if every entry is ≤0 we return the uniform point
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / n)

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho = np.where(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def fast_score(a: np.ndarray) -> float:
    """
    Fast surrogate for the official objective.
    The vector is assumed to be normalised to Σa = 1, therefore the
    denominator is 1 and the score equals 2·n·max(conv(a,a)).
    """
    n = a.shape[0]
    return 2.0 * n * max_convolution(a)


# ----------------------------------------------------------------------
# Shape generators (beta families and simple arcsine)
# ----------------------------------------------------------------------


def generate_beta_shape(n: int, alpha: float, beta: float, eps: float) -> np.ndarray:
    """Beta‑family: a_i ∝ (i+ε)^{α-1}·(n-1-i+ε)^{β-1}."""
    i = np.arange(n, dtype=float)
    left = (i + eps) ** (alpha - 1.0)
    right = (n - 1 - i + eps) ** (beta - 1.0)
    base = left * right
    total = base.sum()
    if total <= 0.0 or not np.isfinite(total):
        return np.full(n, 1.0 / n)
    return base / total


def generate_beta_mixture(
    n: int,
    alpha1: float,
    beta1: float,
    eps1: float,
    alpha2: float,
    beta2: float,
    eps2: float,
    w: float,
) -> np.ndarray:
    """Convex combination of two beta‑family densities."""
    a1 = generate_beta_shape(n, alpha1, beta1, eps1)
    a2 = generate_beta_shape(n, alpha2, beta2, eps2)
    mix = w * a1 + (1.0 - w) * a2
    s = mix.sum()
    if s == 0.0 or not np.isfinite(s):
        return np.full(n, 1.0 / n)
    return mix / s


def arcsine_shape(n: int) -> np.ndarray:
    """Classic arcsine density (the shape used in the original baseline)."""
    i = np.arange(n, dtype=float)
    eps = 0.25  # tiny regulariser to avoid division by zero
    vals = 1.0 / np.sqrt((i + 0.5) * (n - 0.5 - i) + eps)
    vals /= vals.sum()
    return vals


# ----------------------------------------------------------------------
# Smooth surrogate (log‑sum‑exp) and its gradient (for Adam)
# ----------------------------------------------------------------------


def smooth_gradient(a: np.ndarray, lam: float) -> Tuple[np.ndarray, float]:
    """
    Gradient of the log‑sum‑exp surrogate of max(conv(a,a)).
    Returns (grad, surrogate_value).
    """
    n = a.shape[0]
    conv = convolution_fft(a, a)                 # length 2n‑1
    max_c = np.max(conv)

    # stable soft‑max over the convolution entries
    exp_term = np.exp(lam * (conv - max_c))
    sum_exp = np.sum(exp_term)
    w = exp_term / sum_exp                       # soft‑max weights
    B = max_c + np.log(sum_exp) / lam            # smooth approximation of the max

    # d/da Σ_k w_k (a*a)_k = 2 * (a ★ w_rev)
    w_rev = w[::-1]
    conv_rev = convolution_fft(a, w_rev)        # length 3n‑2
    L = 2 * n - 1
    weighted_grad = 2.0 * conv_rev[L - 1 - np.arange(n)]

    s = np.sum(a)
    term1 = (2.0 * n / (s ** 2)) * weighted_grad
    term2 = -(4.0 * n * B / (s ** 3)) * np.ones_like(a)
    grad = term1 + term2
    return grad, B


def adam_optimize(
    init_a: np.ndarray,
    deadline: float,
    max_iter: int = 300_000,
    lr0: float = 0.12,
    lam_start: float = 1.0,
    lam_end: float = 400.0,
    noise_every: int = 1500,
    noise_scale: float = 4e-5,
) -> Tuple[np.ndarray, float]:
    """
    Adam optimiser on the smooth log‑sum‑exp surrogate.
    Returns the best vector found (projected on the simplex) together with
    its fast score.
    """
    a = project_to_simplex(init_a.astype(float).copy())
    best_a = a.copy()
    best_sc = fast_score(a)

    beta1, beta2 = 0.9, 0.999
    eps_adam = 1e-8
    m = np.zeros_like(a)
    v = np.zeros_like(a)

    for it in range(1, max_iter + 1):
        if time.time() >= deadline:
            break

        # quadratic annealing of the temperature λ
        t = it / max_iter
        lam = lam_start + (lam_end - lam_start) * (t ** 0.5)

        grad, _ = smooth_gradient(a, lam)

        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)

        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        step = lr0 * m_hat / (np.sqrt(v_hat) + eps_adam)
        a_new = a - step

        if noise_every and (it % noise_every == 0):
            a_new += np.random.randn(a.size) * noise_scale

        a_new = project_to_simplex(a_new)

        cur_sc = fast_score(a_new)
        if cur_sc < best_sc:
            best_sc = cur_sc
            best_a = a_new.copy()

        a = a_new

        if it % 50_000 == 0:
            print(
                f"    Adam iter={it:,d} cur={cur_sc:.6f} best={best_sc:.6f}"
            )

    return best_a, best_sc


# ----------------------------------------------------------------------
# Exact sub‑gradient descent with simple back‑tracking line‑search
# ----------------------------------------------------------------------


def subgradient_optimize(
    init_a: np.ndarray,
    deadline: float,
    step0: float = 0.30,
    max_iter: int = 200_000,
) -> Tuple[np.ndarray, float]:
    """
    Projected sub‑gradient descent on the true max‑convolution.
    Guarantees non‑increase of the true objective.
    """
    a = init_a.copy()
    best_a = a.copy()
    best_sc = fast_score(a)

    n = a.shape[0]

    for it in range(1, max_iter + 1):
        if time.time() >= deadline:
            break

        conv = convolution_fft(a, a)
        max_val = float(np.max(conv))
        idx = int(np.argmax(conv))

        # gradient of the active quadratic term
        i = np.arange(n)
        j = idx - i
        mask = (j >= 0) & (j < n)
        grad = np.zeros_like(a)
        grad[mask] = 2.0 * a[j[mask]]

        # step size that decays like 1/√t
        step = step0 / np.sqrt(it)
        for _ in range(12):
            a_new = a - step * grad
            a_new = project_to_simplex(a_new)
            new_val = max_convolution(a_new)
            if new_val < max_val - 1e-14:
                break
            step *= 0.5

        cur_sc = fast_score(a_new)
        if cur_sc < best_sc:
            best_sc = cur_sc
            best_a = a_new.copy()

        a = a_new

        if it % 40_000 == 0:
            print(
                f"    Sub‑grad iter={it:,d} cur={cur_sc:.6f} best={best_sc:.6f}"
            )

    return best_a, best_sc


# ----------------------------------------------------------------------
# Stochastic local walk (Gaussian perturbations)
# ----------------------------------------------------------------------


def random_local_search(
    init_a: np.ndarray,
    deadline: float,
    init_step: float = 0.015,
    max_iter: int = 250_000,
) -> Tuple[np.ndarray, float]:
    """
    Cheap Gaussian walk on the simplex.
    Occasionally accepts a small uphill move to escape plateaus.
    """
    a = init_a.copy()
    best_a = a.copy()
    best_sc = fast_score(a)

    n = a.size
    rng = np.random.default_rng()

    for it in range(1, max_iter + 1):
        if time.time() >= deadline:
            break

        step = init_step / np.sqrt(it)
        noise = rng.normal(scale=step, size=n)
        a_prop = a + noise
        a_prop = project_to_simplex(a_prop)

        cur_sc = fast_score(a_prop)

        # accept if strictly better, otherwise with tiny probability
        if cur_sc < best_sc or (rng.random() < 1e-4 and cur_sc < 10 * best_sc):
            a = a_prop
            if cur_sc < best_sc:
                best_sc = cur_sc
                best_a = a_prop.copy()

        if it % 50_000 == 0:
            print(
                f"    Random walk iter={it:,d} cur={cur_sc:.6f} best={best_sc:.6f}"
            )

    return best_a, best_sc

            print("*** Updated global best after final polish ***")

    # ------------------------------------------------------------------
    # 4. Guard – guarantee a non‑empty return
    # ------------------------------------------------------------------
    if not best_seq:
        # safe uniform fallback (length 256)
        best_seq = [1.0] * 256
        best_val = official_score(best_seq)
        print("Fallback to uniform 256‑vector.", best_val)

    print(f"\nSearch finished. Best reported official score = {best_val:.6f}")
    return best_seq
