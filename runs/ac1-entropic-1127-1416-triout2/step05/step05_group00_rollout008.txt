#!/usr/bin/env python
"""
Step‑function optimizer (2026).

We minimise 2·n·max(a⊛a)  (the value returned by `evaluate_sequence`
when the vector is normalised to sum = 1).  The algorithm consists of

1.  a fast smooth‑max gradient descent on the convolution,
    using a soft‑max approximation of the L∞‑norm.
2.  occasional projection onto the probability simplex.
3.  an optional hill‑climb that swaps or moves mass between positions.
4.  a cheap fallback to a uniform or previously known candidate.

The whole search stops before the provided time budget (default 1000 s)
and returns the best sequence found as a plain list of floats.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities (no closures, no lambdas)
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full linear convolution a ⊛ b using a real FFT (fast)."""
    m = len(a) + len(b) - 1
    n_fft = next_pow2(m)
    A = np.fft.rfft(a, n=n_fft)
    B = np.fft.rfft(b, n=n_fft)
    C = A * B
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


def compute_score(a: np.ndarray) -> float:
    """
    Fast replica of `evaluate_sequence` when the vector is normalised to sum=1.
    Returns 2·n·max(a⊛a) (the exact value used for optimisation).
    """
    n = a.shape[0]
    conv = convolution_fft(a, a)
    return 2.0 * n * float(np.max(conv))


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the probability simplex
    (non‑negative entries that sum to 1).
    """
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / v.shape[0])

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, v.shape[0] + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.where(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def linear_correlation(a: np.ndarray, w: np.ndarray) -> np.ndarray:
    """
    Linear (non‑circular) correlation r[i] = Σ_j a[j]·w[i+j] for i≥0.
    Returned length is len(a)+len(w)-1.
    Implemented via FFT with zero‑padding.
    """
    n = a.shape[0]
    m = w.shape[0]
    L = next_pow2(n + m - 1)
    A = np.fft.fft(a, L)
    W = np.fft.fft(w, L)
    r = np.fft.ifft(A * np.conj(W)).real[: n + m - 1]
    return r


# ----------------------------------------------------------------------
# Smooth‑max gradient descent
# ----------------------------------------------------------------------


def smooth_max_descent(
    init_a: np.ndarray,
    deadline: float,
    max_iters: int = 200_000,
    step0: float = 0.15,
    alpha_start: float = 5.0,
    alpha_target: float = 200.0,
) -> tuple[np.ndarray, float]:
    """
    Projected gradient descent on the smooth approximation
        F(a) = 2·n·log‑sum‑exp(α·(a⊛a)) / α
    The gradient of the inner log‑sum‑exp is a soft‑max over the
    convolution entries, giving a simple linear‑time (FFT‑based) expression.
    """
    a = init_a.astype(float).copy()
    a = project_to_simplex(a)            # guarantee exact simplex
    best_a = a.copy()
    best_score = compute_score(a)

    n = len(a)
    m = 2 * n - 1               # convolution length
    alpha = alpha_start
    alpha_inc = (alpha_target / alpha_start) ** (1.0 / max_iters)

    for t in range(1, max_iters + 1):
        if time.time() >= deadline:
            break

        # ----- convolution and current score -----
        conv = convolution_fft(a, a)          # length 2n‑1
        cur_score = 2.0 * n * float(np.max(conv))
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()

        # ----- smooth max weights (soft‑max over convolution) -----
        maxc = np.max(conv)
        # scale by α, subtract maxc for numerical stability
        exp_term = np.exp(alpha * (conv - maxc))
        w = exp_term / np.sum(exp_term)       # w.shape == (2n‑1,)

        # ----- gradient of the smoothed objective -----
        # grad = 2 * Σ_j a_j w_{i+j}  (i = 0 … n‑1)
        corr = linear_correlation(a, w)       # length n+m‑1
        grad = 2.0 * corr[:n]

        # ----- step size (classical 1/√t schedule) -----
        step = step0 / np.sqrt(t)
        a = a - step * grad
        a = project_to_simplex(a)

        # ----- slowly increase α (approach true max) -----
        alpha = min(alpha * alpha_inc, alpha_target)

        # ----- occasional tiny random perturbation (helps escape plateaus) -----
        if t % 500 == 0:
            sigma = 0.005 / np.sqrt(t)
            a += np.random.randn(n) * sigma
            a = project_to_simplex(a)

    final_score = compute_score(best_a)
    return best_a, final_score


# ----------------------------------------------------------------------
# Simple hill‑climber (swap / mass transfer)
# ----------------------------------------------------------------------


def hill_climb(seq, deadline, max_steps=200_000):
    """
    Stochastic hill‑climb that either swaps two entries or transfers a
    fraction of mass from one position to another.  The true (unsmoothed)
    objective is used for acceptance.
    """
    a = np.array(seq, dtype=float)
    n = a.shape[0]

    best = a.copy()
    best_val = compute_score(best)

    transfer_frac = 0.1

    while max_steps > 0 and time.time() < deadline:
        cand = best.copy()

        if np.random.rand() < 0.5:
            # swap two positions
            i, j = np.random.randint(n, size=2)
            cand[i], cand[j] = cand[j], cand[i]
        else:
            # move a fraction of mass from i to j
            i, j = np.random.randint(n, size=2)
            if i != j and cand[i] > 0:
                delta = transfer_frac * cand[i]
                cand[i] -= delta
                cand[j] += delta

        cand = project_to_simplex(cand)
        val = compute_score(cand)

        if val < best_val:
            best_val = val
            best = cand
            # slowly increase the step size when we improve
            transfer_frac = min(transfer_frac * 1.05, 0.5)
        else:
            # otherwise shrink it
            transfer_frac *= 0.97

        max_steps -= 1

    return best.tolist(), best_val


# ----------------------------------------------------------------------
# Main entry point required by the evaluation harness
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative sequence that minimises `evaluate_sequence`.
    The routine combines a smooth‑max gradient descent with a final
    hill‑climbing polish.  It respects the given time budget.
    """
    np.random.seed(seed)
    start = time.time()
    # leave a small safety margin for the harness
    deadline = start + budget_s - 2.0

    best_seq = None
    best_val = float("inf")

    # --------------------------------------------------------------
    # (optional) use a previously known candidate if it exists
    # --------------------------------------------------------------
    try:
        fallback = [float(x) for x in height_sequence_1]  # type: ignore
        fallback_val = evaluate_sequence(fallback)
        print(f"Fallback score = {fallback_val:.6f}")
        if fallback_val < best_val:
            best_val = fallback_val
            best_seq = fallback
    except Exception:
        pass

    # --------------------------------------------------------------
    # Choose a few problem sizes (larger n gives more freedom)
    # --------------------------------------------------------------
    lengths = [2048, 4096]   # both are powers of two → FFT friendly
    for n in lengths:
        if time.time() >= deadline - 30.0:
            break
        print(f"\n=== Optimising length {n} ===")

        # ------ conduct a few random restarts of the smooth‑max descent ------
        restarts = 3
        for r in range(restarts):
            if time.time() >= deadline - 20.0:
                break

            # random Dirichlet start (sum to 1)
            a0 = np.random.rand(n)
            a0 /= a0.sum()

            a_opt, score_opt = smooth_max_descent(
                a0,
                deadline - 5.0,
                max_iters=120_000,
                step0=0.13,
                alpha_start=5.0,
                alpha_target=150.0,
            )
            print(
                f"  restart {r + 1}/{restarts}, best smooth‑max score "
                f"{score_opt:.6f}"
            )
            if score_opt < best_val:
                best_val = score_opt
                best_seq = a_opt.tolist()
                print(f"  >>> New global best = {best_val:.6f}")

        # ------ final hill‑climb polish for this length ---------------
        if best_seq is not None and time.time() < deadline - 5.0:
            print("\nHill‑climb polishing")
            polished, val_polished = hill_climb(
                best_seq,
                deadline,
                max_steps=250_000,
            )
            if val_polished < best_val:
                best_val = val_polished
                best_seq = polished
                print(f"Hill‑climb improved best to {best_val:.6f}")

    # --------------------------------------------------------------
    # Safeguard: fallback to a uniform vector if nothing was found
    # --------------------------------------------------------------
    if best_seq is None:
        n = 256
        best_seq = [1.0 / n] * n
        best_val = evaluate_sequence(best_seq)

    # final official evaluation (uses the provided `evaluate_sequence`)
    final_score = evaluate_sequence(best_seq)
    print(f"\nFinished search. Final reported score = {final_score:.6f}")

    # The harness expects a plain Python list of floats.
    return best_seq
