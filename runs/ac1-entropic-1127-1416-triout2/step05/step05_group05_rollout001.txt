def refine_annealed(a, deadline, eta0=0.5, beta0=1.0, beta_max=40.0, max_iter=10_000, report_every=0):
    a = normalize(a.astype(float).copy())
    n = len(a)
    best_a = a.copy()
    best_score = compute_score(a)   # score = 2*n*max(conv)
    it = 0
    while time.time() < deadline and it < max_iter:
        it += 1
        conv = convolution_fft(a, a)
        max_val = np.max(conv)
        
        # schedule beta: increase linearly with iteration up to max_iter
        beta = beta0 + (beta_max - beta0) * min(1.0, it / max_iter)
        exp_vals = np.exp(beta * (conv - max_val))
        w = exp_vals / np.sum(exp_vals)
        grad = gradient_from_weights(a, w)
        eta = eta0 / np.sqrt(it)
        a = a * np.exp(-eta * grad)
        a = normalize(a)
        score = compute_score(a)
        if score < best_score:
            best_score = score
            best_a = a.copy()
        if report_every and it % report_every == 0:
            print(f"    iter {it} score {score:.6f} best {best_score:.6f} beta {beta:.2f}")
    return best_a, best_score
```

"""
Propose a low‑dimensional non‑negative step function (probability vector)
that minimizes

        2·n·max(conv(a,a)) / (sum(a)²)

The implementation follows a multi‑start, multi‑scale scheme:

1.  **Parameter sweep** – generate a rich portfolio of candidate
    distributions (arcsine‑like, power‑law, uniform, random Dirichlet,
    optional previous best).  For the arcsine and power‑law families a
    small grid of shape parameters is explored and the fast internal
    score (2·n·max(conv) with sum = 1) is used to keep the few best
    candidates.

2.  **Annealed mirror‑descent** – each retained candidate is refined
    by an exponentiated‑gradient (multiplicative) update on the simplex.
    The gradient is taken w.r.t. a *soft* max of the convolution:
    `w_i ∝ exp(β·(conv_i – max(conv)))`.  The temperature β starts
    small (β≈1) and is slowly increased up to β≈30, i.e. a deterministic
    annealing that moves from a smooth to the true max‑objective.
    The step‑size follows ηₜ = η₀/√t.  The best iterate (exact score) is
    remembered.

3.  **Hard sub‑gradient polishing** – the best refined vector is finally
    polished with the classic sub‑gradient method that uses the exact
    max‑weight (uniform on maximisers).  This finishes the optimisation
    within the remaining time budget.

All arithmetic is performed with NumPy, convolutions use a real FFT for
speed, and the simplex projection is the standard `max(v,0)/sum(v)`
procedure.  The routine respects the given time budget (default 1000 s)
and always returns a feasible list of floats.
"""

import time
import numpy as np
from typing import List, Tuple

# ----------------------------------------------------------------------
# Low‑level utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """
    Linear convolution via real FFT (full length, zero‑padded to a power‑of‑2).
    """
    m = len(a) + len(b) - 1
    n_fft = next_pow2(m)
    A = np.fft.rfft(a, n=n_fft)
    B = np.fft.rfft(b, n=n_fft)
    C = A * B
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


def normalize(v: np.ndarray) -> np.ndarray:
    """Project a non‑negative vector onto the simplex (sum = 1)."""
    s = v.sum()
    if s <= 0.0:
        return np.full_like(v, 1.0 / len(v))
    return v / s


def compute_score(a: np.ndarray) -> float:
    """
    Fast evaluation of the scale‑invariant objective
        2·n·max(conv(a,a))   (a is assumed to sum to 1)
    """
    n = len(a)
    conv = convolution_fft(a, a)
    max_val = float(np.max(conv))
    if max_val < 0:
        return float("inf")
    return 2.0 * n * max_val


def gradient_from_weights(a: np.ndarray, w: np.ndarray) -> np.ndarray:
    """
    Gradient of Σ_k w_k·(a*a)_k w.r.t. a.
    Implements   g_j = 2· Σ_m a_m · w_{m+j}
    Computed via one FFT‑based convolution.
    """
    # convolution of a with reversed w
    w_rev = w[::-1]
    conv = convolution_fft(a, w_rev)               # length = 3·n‑2
    idx_center = 2 * len(a) - 2
    grads = 2.0 * conv[idx_center - np.arange(len(a))]
    return grads


# ----------------------------------------------------------------------
# Parametric families (all return a probability vector, i.e. sum=1)
# ----------------------------------------------------------------------


def arcsine_shape_eps(n: int, eps: float = 0.25) -> np.ndarray:
    """Discrete arcsine‑like shape: 1/√((i+½)(n‑½‑i)+eps)."""
    i = np.arange(n, dtype=float)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - 0.5 - i) + eps)
    return normalize(vals)


def powerlaw_shape(n: int, p: float, eps: float) -> np.ndarray:
    """Symmetric power‑law: a_i ∝ ((i+ε)(n‑1‑i+ε))⁻ᵖ."""
    i = np.arange(n, dtype=float)
    base = ((i + eps) * (n - 1 - i + eps)) ** (-p)
    return normalize(base)


def uniform_shape(n: int) -> np.ndarray:
    """Uniform distribution on n points."""
    return np.full(n, 1.0 / n)


def random_dirichlet(n: int, rng: np.random.Generator) -> np.ndarray:
    """Sample a random point on the (n‑dim) simplex."""
    samp = rng.exponential(1.0, size=n)
    return normalize(samp)


# ----------------------------------------------------------------------
# Refinement methods
# ----------------------------------------------------------------------


def refine_annealed(
    a: np.ndarray,
    deadline: float,
    *,
    eta0: float = 0.8,
    beta0: float = 1.0,
    beta_max: float = 30.0,
    max_iter: int = 200_000,
    report_every: int = 0,
) -> Tuple[np.ndarray, float]:
    """
    Exponentiated‑gradient mirror‑descent with a linearly increasing
    inverse temperature β (annealing).  The weight vector for the gradient
    is the softmax over (conv – max(conv)).  The best iterate (exact score)
    is returned.
    """
    a = normalize(a.astype(float).copy())
    n = len(a)
    best_a = a.copy()
    best_score = compute_score(a)

    it = 0
    while time.time() < deadline and it < max_iter:
        it += 1

        # convolution of current iterate with itself
        conv = convolution_fft(a, a)
        max_val = float(np.max(conv))

        # annealing schedule for β
        beta = beta0 + (beta_max - beta0) * min(1.0, it / max_iter)

        # soft‑max weights (max entry gets weight ≈1/|max_set|)
        exp_vals = np.exp(beta * (conv - max_val))
        w = exp_vals / np.sum(exp_vals)

        # gradient of the smoothed objective
        grad = gradient_from_weights(a, w)

        # step‑size schedule
        eta = eta0 / np.sqrt(it)

        # multiplicative update + simplex projection
        a = a * np.exp(-eta * grad)
        a = normalize(a)

        cur_score = compute_score(a)
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()

        if report_every and (it % report_every == 0):
            print(
                f"    annealed iter={it:,d} β={beta:.2f} score={cur_score:.6f} best={best_score:.6f}"
            )
    return best_a, best_score


def refine_subgradient(
    a: np.ndarray,
    deadline: float,
    *,
    max_iter: int = 100_000,
    eta0: float = 0.7,
    report_every: int = 0,
) -> Tuple[np.ndarray, float]:
    """
    Classic sub‑gradient descent on the exact max‑convolution objective.
    Uses multiplicative updates and a 1/√t step‑size.
    Returns the best vector and its exact score.
    """
    a = normalize(a.astype(float).copy())
    best_a = a.copy()
    best_score = compute_score(a)

    for it in range(1, max_iter + 1):
        if time.time() >= deadline:
            break

        conv = convolution_fft(a, a)
        max_val = float(np.max(conv))

        # weight = uniform on (near) maximisers
        w = np.zeros_like(conv)
        max_idx = np.where(conv >= max_val - 1e-14)[0]
        w[max_idx] = 1.0 / len(max_idx)

        grad = gradient_from_weights(a, w)

        eta = eta0 / np.sqrt(it)
        a = a * np.exp(-eta * grad)
        a = normalize(a)

        cur_score = 2.0 * len(a) * max_val
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()

        if report_every and (it % report_every == 0):
            print(f"    sub‑grad iter={it:,d} cur={cur_score:.6f} best={best_score:.6f}")

    return best_a, best_score


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------


def propose_candidate(
    seed: int = 42,
    budget_s: int = 1000,
    **kwargs,
) -> List[float]:
    """
    Search for a non‑negative step‑function with a low score.
    The routine respects the given time budget, explores several
    parametric families, refines the most promising candidates with an
    annealed mirror‑descent, and ends with a hard sub‑gradient polish.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    # safety margin to avoid hard timeout
    deadline = start_time + budget_s - 2.0

    # --------------------------------------------------------------
    # Choose a working length (larger n gives a finer discretisation).
    # 4096 is a good compromise between resolution and speed.
    # --------------------------------------------------------------
    n = 4096

    # --------------------------------------------------------------
    # 1.  Build a rich set of initial candidates.
    # --------------------------------------------------------------
    candidates: List[np.ndarray] = []

    # a) arcsine family with several epsilons
    for eps in [0.0, 1e-6, 1e-3, 1e-2, 0.1, 0.25, 0.5, 1.0, 5.0]:
        candidates.append(arcsine_shape_eps(n, eps))

    # b) power‑law family on a coarse grid
    for p in np.linspace(0.30, 0.70, 9):
        for eps in [0.0, 1e-4, 1e-2, 0.1, 1.0, 5.0]:
            candidates.append(powerlaw_shape(n, float(p), float(eps)))

    # c) a few random Dirichlet draws
    for _ in range(8):
        candidates.append(random_dirichlet(n, rng))

    # d) uniform distribution (baseline)
    candidates.append(uniform_shape(n))

    # e) optionally reuse a previously discovered construction
    try:
        fallback = np.array([float(x) for x in height_sequence_1], dtype=float)
        if fallback.ndim == 1:
            fallback = normalize(fallback)
            candidates.append(fallback)
            print("[fallback] incorporated previous best")
    except NameError:
        pass

    # --------------------------------------------------------------
    # 2.  Fast screening – keep only the few most promising ones.
    # --------------------------------------------------------------
    scored = [(compute_score(c), c) for c in candidates]
    scored.sort(key=lambda t: t[0])
    top_k = 6
    top_candidates = [vec for _, vec in scored[:top_k]]
    print(f"Initial fast best score = {scored[0][0]:.6f} (top {top_k} kept)")

    best_seq: List[float] = []
    best_score: float = float("inf")

    # --------------------------------------------------------------
    # 3.  Annealed mirror‑descent refinement for each top candidate.
    # --------------------------------------------------------------
    for idx, cand in enumerate(top_candidates):
        if time.time() >= deadline - 5.0:
            break
        remaining = max(0.0, deadline - time.time())
        # allocate roughly 30 % of the remaining budget to this candidate
        sub_deadline = time.time() + remaining * 0.30
        print(f"Refining candidate {idx+1}/{len(top_candidates)} (time left {remaining:.1f}s)")
        refined, _ = refine_annealed(
            cand,
            sub_deadline,
            eta0=0.8,
            beta0=1.0,
            beta_max=30.0,
            max_iter=200_000,
            report_every=0,
        )
        official = evaluate_sequence(refined.tolist())
        print(f"  refined official score = {official:.6f}")
        if official < best_score:
            best_score = official
            best_seq = refined.tolist()
            print("*** New global best ***")

    # --------------------------------------------------------------
    # 4.  Final hard sub‑gradient polishing (if time permits).
    # --------------------------------------------------------------
    if best_seq and time.time() < deadline - 5.0:
        print("\nStarting final hard sub‑gradient polish")
        polished, _ = refine_subgradient(
            np.array(best_seq),
            deadline,
            max_iter=200_000,
            eta0=0.5,
            report_every=0,
        )
        final_val = evaluate_sequence(polished.tolist())
        print(f"  final official score = {final_val:.6f}")
        if final_val < best_score:
            best_score = final_val
            best_seq = polished.tolist()
            print("*** Updated global best after polish ***")

    # --------------------------------------------------------------
    # 5.  Safety fallback – always return a feasible vector.
    # --------------------------------------------------------------
    if not best_seq:
        best_seq = uniform_shape(256).tolist()
        best_score = evaluate_sequence(best_seq)

    print(f"\nSearch finished. Best reported score = {best_score:.6f}")
    return best_seq
