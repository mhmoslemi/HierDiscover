def lp_refine(f: np.ndarray, iters: int, t: float, deadline: float) -> np.ndarray:
    n = f.shape[0]
    for it in range(iters):
        if time.time() >= deadline:
            break
        M = max_convolution(f)
        A = build_convolution_matrix(f)
        # RHS
        b_ub = np.full(2 * n - 1, M, dtype=np.float64)
        # solve
        res = linprog(c=-np.ones(n), A_ub=A, b_ub=b_ub,
                      bounds=(0, None), method='highs',
                      options={'presolve': True, 'time_limit': max(0, deadline - time.time())})
        if not res.success:
            break
        b_opt = res.x
        sum_b = b_opt.sum()
        if sum_b <= 1e-12:
            break
        g = b_opt / sum_b
        # convex combine with backtracking
        t_local = t
        for _ in range(5):
            f_new = (1 - t_local) * f + t_local * g
            f_new = f_new / np.sum(f_new)
            M_new = max_convolution(f_new)
            if M_new <= M * (1 + 1e-12):
                f = f_new
                break
            t_local *= 0.5
        # else if not improved, break early

"""
LP‑based iterative refinement for the Linf‑norm of a self‑convolution.

The algorithm:
1. Start from a classic arcsine‑shaped vector (sum = 1).
2. Repeatedly solve a linear program:
       maximise   Σ b_i
       subject to 0 ≤ b_i ,   (f * b)_k ≤ M   for all k,
   where M = max(f∗f) and f is the current vector.
   The LP finds a candidate b with a larger ℓ₁‑mass while keeping the
   convolution bound unchanged.
3. Convex‑combine f and the normalised solution g = b/Σb, backing off
   the step size if the combined vector does not improve the bound.
4. After a few LP steps run a short projected sub‑gradient descent on
   the *true* max‑convolution, then a light stochastic walk to polish.
5. Keep the best vector (according to the official ``evaluate_sequence``)
   encountered over several lengths (1024 and 2048).  The whole search
   respects the 1000 s budget and always returns a non‑empty sequence.
"""

import time
import numpy as np
import scipy.sparse as sp
from scipy.optimize import linprog
from typing import List, Tuple

# ----------------------------------------------------------------------
# Basic linear‑algebra utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """
    Linear convolution via real FFT.
    Returns an array of length len(a)+len(b)-1.
    """
    m = a.size + b.size - 1
    n_fft = next_pow2(m)
    A = np.fft.rfft(a, n=n_fft)
    B = np.fft.rfft(b, n=n_fft)
    C = A * B
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


def max_convolution(a: np.ndarray) -> float:
    """Maximum entry of the self‑convolution of a."""
    return float(np.max(convolution_fft(a, a)))


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection onto the probability simplex {x ≥ 0, Σx = 1}.
    Implementation of Duchi et al. (2008).
    """
    n = v.shape[0]
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / n)

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho = np.where(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def compute_score(a: np.ndarray) -> float:
    """
    Fast surrogate (valid when Σa = 1):
        2·n·max(conv(a,a)).
    """
    n = a.shape[0]
    return 2.0 * n * max_convolution(a)


def arcsine_initial(n: int) -> np.ndarray:
    """Classic arcsine shape (symmetrised)."""
    i = np.arange(n, dtype=float)
    eps = 0.25
    vals = 1.0 / np.sqrt((i + 0.5) * (n - 0.5 - i) + eps)
    vals /= vals.sum()
    return vals


# ----------------------------------------------------------------------
# Build convolution matrix (sparse) for the linear program
# ----------------------------------------------------------------------


def build_convolution_matrix(f: np.ndarray) -> sp.csr_matrix:
    """
    Build a sparse matrix A such that (A @ b) = f ∗ b
    (linear convolution, zero padding outside the interval).
    """
    n = f.shape[0]
    i = np.arange(n, dtype=np.int64)[:, None]       # (n,1)
    j = np.arange(n, dtype=np.int64)[None, :]       # (1,n)
    rows = (i + j).ravel()                          # length n*n
    cols = np.tile(j, (n, 1)).ravel()               # length n*n
    data = np.repeat(f, n)                          # length n*n
    return sp.csr_matrix((data, (rows, cols)), shape=(2 * n - 1, n))


# ----------------------------------------------------------------------
# LP‑based refinement (alternating linear program)
# ----------------------------------------------------------------------


def lp_refine(
    f: np.ndarray,
    max_iters: int,
    step_size: float,
    deadline: float,
) -> np.ndarray:
    """
    Perform a few iterations of the alternating LP scheme.
    ``step_size`` is the initial convex combination weight (t in the paper).
    The procedure stops early if the time budget expires or if no improvement
    can be made.
    """
    n = f.shape[0]
    for it in range(max_iters):
        if time.time() >= deadline:
            break

        M = max_convolution(f)                     # current bound
        A = build_convolution_matrix(f)            # (2n‑1) × n sparse matrix
        b_ub = np.full(2 * n - 1, M, dtype=np.float64)

        # maximise Σb  <=> minimise −Σb
        res = linprog(
            c=-np.ones(n, dtype=np.float64),
            A_ub=A,
            b_ub=b_ub,
            bounds=(0, None),
            method="highs",
            options={"presolve": True},
        )
        if not res.success:
            # LP failed – give up the refinement
            break

        b_opt = res.x
        sum_b = b_opt.sum()
        if sum_b <= 1e-12:
            break

        # Normalise to sum = 1
        g = b_opt / sum_b

        # Convex combination with back‑tracking if the bound does not improve
        t_local = step_size
        for _ in range(5):
            f_new = (1.0 - t_local) * f + t_local * g
            f_new = f_new / f_new.sum()          # enforce Σ=1
            M_new = max_convolution(f_new)
            if M_new <= M * (1.0 + 1e-12):
                f = f_new
                break
            t_local *= 0.5
        # If none of the back‑tracking steps improved, abort further LP work
        else:
            break
    return f


# ----------------------------------------------------------------------
# Projected sub‑gradient descent on the true max‑convolution
# ----------------------------------------------------------------------


def subgradient_optimize(
    a: np.ndarray,
    deadline: float,
    max_iter: int = 80_000,
    step0: float = 0.25,
) -> np.ndarray:
    """
    Simple projected sub‑gradient descent on
        2·n·max(conv(a,a))  (with Σa = 1).
    Uses a back‑tracking line‑search to guarantee monotone descent.
    """
    n = a.shape[0]
    best = a.copy()
    best_sc = compute_score(a)

    for it in range(1, max_iter + 1):
        if time.time() >= deadline:
            break

        conv = convolution_fft(a, a)
        max_val = float(conv.max())
        idx = int(conv.argmax())

        # Sub‑gradient of the active quadratic term
        grad = np.zeros_like(a)
        i = np.arange(n)
        j = idx - i
        mask = (j >= 0) & (j < n)
        grad[mask] = 2.0 * a[j[mask]]

        step = step0 / np.sqrt(it)

        # Back‑tracking line search
        for _ in range(12):
            a_new = a - step * grad
            a_new = project_to_simplex(a_new)
            if max_convolution(a_new) < max_val - 1e-14:
                break
            step *= 0.5

        cur_sc = compute_score(a_new)
        if cur_sc < best_sc:
            best_sc = cur_sc
            best = a_new.copy()

        a = a_new

    return best


# ----------------------------------------------------------------------
# Light stochastic walk (Gaussian perturbations)
# ----------------------------------------------------------------------


def random_local_search(
    a: np.ndarray,
    deadline: float,
    init_step: float = 0.015,
    max_iter: int = 150_000,
) -> np.ndarray:
    """
    Cheap Gaussian walk on the simplex with a tiny uphill‑move probability.
    """
    n = a.shape[0]
    best = a.copy()
    best_sc = compute_score(a)
    rng = np.random.default_rng()

    for it in range(1, max_iter + 1):
        if time.time() >= deadline:
            break

        step = init_step / np.sqrt(it)
        noise = rng.normal(scale=step, size=n)
        cand = a + noise
        cand = project_to_simplex(cand)

        cur_sc = compute_score(cand)

        if cur_sc < best_sc or (rng.random() < 1e-4 and cur_sc < 10 * best_sc):
            a = cand
            if cur_sc < best_sc:
                best_sc = cur_sc
                best = cand.copy()

    return best

        # ------------------------------------------------------------------
        lp_iters = 12                       # a dozen LP solves are cheap enough
        f = lp_refine(f, max_iters=lp_iters, step_size=0.3, deadline=deadline)

        val = evaluate_sequence(f.tolist())
        print(f"  after LP refinement official score = {val:.6f}")
        if val < best_val:
            best_val = val
            best_seq = f.tolist()

        # ------------------------------------------------------------------
        # Phase 2 – short projected sub‑gradient polishing
        # ------------------------------------------------------------------
        remaining = deadline - time.time()
        if remaining > 5.0:
            max_sub_iters = int(min(60_000, remaining * 28))
            print(f"  sub‑gradient polishing (max_iter={max_sub_iters}) …")
            f = subgradient_optimize(f, deadline=deadline, max_iter=max_sub_iters, step0=0.25)
            val = evaluate_sequence(f.tolist())
            print(f"  after sub‑gradient official score = {val:.6f}")
            if val < best_val:
                best_val = val
                best_seq = f.tolist()

        # ------------------------------------------------------------------
        # Phase 3 – final cheap stochastic walk
        # ------------------------------------------------------------------
        remaining = deadline - time.time()
        if remaining > 3.0:
            max_loc_iters = int(min(120_000, remaining * 30))
            print(f"  random local walk (max_iter={max_loc_iters}) …")
            f = random_local_search(f, deadline=deadline, init_step=0.012, max_iter=max_loc_iters)
            val = evaluate_sequence(f.tolist())
            print(f"  after random walk official score = {val:.6f}")
            if val < best_val:
                best_val = val
                best_seq = f.tolist()

    # ------------------------------------------------------------------
    # Guard – guarantee a non‑empty return
    # ------------------------------------------------------------------
    if not best_seq:
        # safe uniform fallback (length 256)
        best_seq = [1.0] * 256
        best_val = evaluate_sequence(best_seq)
        print("Fallback to uniform 256‑vector.", best_val)

    print(f"\nSearch finished. Best reported official score = {best_val:.6f}")
    return best_seq
