def simplex_projection(v):
    # Project onto simplex {x>=0, sum x = 1}
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    rho = np.where(u - cssv / (np.arange(len(v)) + 1) > 0)[0][-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0)
    return w

```python
"""
Propose a sequence of non‑negative heights that minimises
2·n·max(conv(a,a))/(sum(a)²).

The algorithm:
1. Generate a rich pool of candidate shapes (arcsine, power‑law,
   exponential decay, random Dirichlet, uniform, tent).
   Candidates are normalised to a probability vector (sum = 1) and
   quickly ranked by a fast FFT based proxy for the objective.
2. Take the best few candidates and refine them with two stages:
   a) a smooth exponentiated‑gradient descent where the “soft‑max”
      temperature β is increased during the run;
   b) a hard additive sub‑gradient descent with projection onto the
      simplex (the exact max‑convolution sub‑gradient).
   The best vector found is evaluated with the official
   ``evaluate_sequence`` routine.
3. The best sequence found within the time budget is returned.
"""

import time
import numpy as np
from typing import List, Tuple

def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()

def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Linear convolution via real FFT (full length)."""
    m = len(a) + len(b) - 1
    n_fft = next_pow2(m)
    A = np.fft.rfft(a, n=n_fft)
    B = np.fft.rfft(b, n=n_fft)
    C = A * B
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv

def normalize(v: np.ndarray) -> np.ndarray:
    """Project a non‑negative vector onto the simplex (sum = 1)."""
    total = v.sum()
    if total <= 0.0:
        return np.full_like(v, 1.0 / len(v))
    return v / total

def simplex_projection(v: np.ndarray) -> np.ndarray:
    """Euclidean projection onto the probability simplex."""
    # Reference: Duchi et al. (2008)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    rho = np.nonzero(u - cssv / (np.arange(n) + 1) > 0)[0]
    if len(rho) == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w

def compute_score(a: np.ndarray) -> float:
    """Fast proxy for the official objective (sum(a) = 1)."""
    n = len(a)
    conv = convolution_fft(a, a)
    max_val = float(np.max(conv))
    return 2.0 * n * max_val

def gradient_from_weights(a: np.ndarray, w: np.ndarray) -> np.ndarray:
    """∇ₐ Σ_k w_k (a*a)_k = 2·conv(a, w_rev)."""
    # w must have length 2·len(a)‑1
    w_rev = w[::-1]
    conv = convolution_fft(a, w_rev)               # length = 3·n‑2
    idx = 2 * len(a) - 2                            # position 2n‑2
    grads = 2.0 * conv[idx - np.arange(len(a))]
    return grads

# ----------------------------------------------------------------------
# Families of candidate shapes
# ----------------------------------------------------------------------
def arcsine_initial(n: int) -> np.ndarray:
    i = np.arange(n, dtype=float)
    eps = 0.25
    vals = 1.0 / np.sqrt((i + 0.5) * (n - 0.5 - i) + eps)
    return normalize(vals)

def powerlaw_shape(n: int, p: float, eps: float) -> np.ndarray:
    i = np.arange(n, dtype=float)
    base = ((i + eps) * (n - 1 - i + eps)) ** (-p)
    return normalize(base)

def uniform_shape(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n)

def tent_shape(n: int) -> np.ndarray:
    i = np.arange(n, dtype=float)
    vals = np.minimum(i + 1, n - i)
    return normalize(vals)

def random_dirichlet(n: int, rng: np.random.Generator) -> np.ndarray:
    samp = rng.exponential(1.0, size=n)
    return normalize(samp)

def exponential_decay_shape(n: int, lam: float) -> np.ndarray:
    i = np.arange(n, dtype=float)
    center = (n - 1) / 2.0
    vals = np.exp(-lam * np.abs(i - center))
    return normalize(vals)

# ----------------------------------------------------------------------
# Refinement routines
# ----------------------------------------------------------------------
def refine_smooth_schedule(
    a: np.ndarray,
    deadline: float,
    *,
    eta0: float = 1.0,
    beta_start: float = 5.0,
    beta_end: float = 40.0,
    report_every: int = 0,
) -> Tuple[np.ndarray, float]:
    """
    Exponentiated‑gradient descent with a slowly increasing temperature.
    ``beta`` controls the softness of the max‑approximation; it is
    increased from ``beta_start`` to ``beta_end`` during the run.
    """
    a = normalize(a.astype(float).copy())
    n = len(a)
    best_a = a.copy()
    conv = convolution_fft(a, a)
    best_score = 2.0 * n * float(np.max(conv))

    it = 0
    while time.time() < deadline:
        it += 1
        conv = convolution_fft(a, a)
        max_val = float(np.max(conv))

        # linearly increase beta (capped)
        beta = beta_start + (beta_end - beta_start) * min(1.0, it * 1e-4)

        # soft‑max weights on the convolution entries
        exp_vals = np.exp(beta * (conv - max_val))
        w = exp_vals / np.sum(exp_vals)

        grad = gradient_from_weights(a, w)

        eta = eta0 / np.sqrt(it)

        a = a * np.exp(-eta * grad)
        a = normalize(a)

        cur_score = 2.0 * n * max_val
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()

        if report_every and (it % report_every == 0):
            print(f"    smooth iter={it:,d} cur={cur_score:.6f} best={best_score:.6f}")

    return best_a, best_score


def refine_subgradient_additive(
    a: np.ndarray,
    deadline: float,
    *,
    eta0: float = 0.7,
    max_iter: int = 200_000,
    report_every: int = 0,
) -> Tuple[np.ndarray, float]:
    """
    Classical additive sub‑gradient descent on the exact max‑convolution
    objective.  After each gradient step the vector is projected back onto
    the probability simplex.
    """
    a = normalize(a.astype(float).copy())
    n = len(a)
    best_a = a.copy()
    conv = convolution_fft(a, a)
    best_score = 2.0 * n * float(np.max(conv))

    for it in range(1, max_iter + 1):
        if time.time() >= deadline:
            break

        conv = convolution_fft(a, a)
        max_val = float(np.max(conv))

        # uniform weight on (near) maximisers
        w = np.zeros_like(conv)
        max_idx = np.where(conv >= max_val - 1e-14)[0]
        if max_idx.size > 0:
            w[max_idx] = 1.0 / max_idx.size
        else:
            w[np.argmax(conv)] = 1.0

        grad = gradient_from_weights(a, w)

        eta = eta0 / np.sqrt(it)
        a = a - eta * grad
        a = simplex_projection(a)

        cur_score = 2.0 * n * max_val
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()

        if report_every and (it % report_every == 0):
            print(f"    sub‑grad iter={it:,d} cur={cur_score:.6f} best={best_score:.6f}")

    return best_a, best_score

def propose_candidate(
    seed: int = 42,
    budget_s: int = 1000,
    **kwargs,
) -> List[float]:
    """
    Search for a non‑negative step‑function with a low evaluation score.
    The routine respects the time budget, builds a large pool of simple
    parametric families, selects the most promising one with a fast FFT‑based
    proxy, then refines it with a smooth exponential‑gradient phase followed
    by a hard additive sub‑gradient polishing phase.
    The best sequence found (as list of floats) is returned.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    # leave a small safety margin
    deadline = start_time + budget_s - 2.0

    best_seq: List[float] = []
    best_val: float = float("inf")

    # --------------------------------------------------------------
    # 1️⃣  Build a diverse pool of candidates (different lengths)
    # --------------------------------------------------------------
    length_options = [2048, 4096]   # 2 k and 4 k are a good trade‑off
    raw_candidates: List[np.ndarray] = []

    # fallback construction if the harness provides one
    try:
        fallback = np.array([float(x) for x in height_sequence_1], dtype=float)
        if fallback.ndim == 1:
            fallback = normalize(fallback)
            raw_candidates.append(fallback)
            print("[fallback] length =", len(fallback))
    except NameError:
        pass

    for n in length_options:
        raw_candidates.append(arcsine_initial(n))
        raw_candidates.append(uniform_shape(n))
        raw_candidates.append(tent_shape(n))

        # a few random Dirichlet draws
        for _ in range(4):
            raw_candidates.append(random_dirichlet(n, rng))

        # exponential‑decay shapes
        for lam in (0.0005, 0.001, 0.002, 0.005):
            raw_candidates.append(exponential_decay_shape(n, lam))

        # power‑law sweep
        for p in np.linspace(0.30, 1.00, 8):
            eps = rng.uniform(0.0, max(5.0, 0.02 * n))
            raw_candidates.append(powerlaw_shape(n, float(p), float(eps)))

    # ----------------------------------------------------------------
    # 2️⃣  Fast ranking using the inexpensive proxy (sum = 1)
    # ----------------------------------------------------------------
    scored = [(compute_score(c), c) for c in raw_candidates]
    scored.sort(key=lambda x: x[0])
    # keep the top few for an expensive refinement
    top_k = 3
    refinement_seeds = [c for (_, c) in scored[:top_k]]
    print(f"Initial fast best score = {scored[0][0]:.6f}, refining {len(refinement_seeds)} candidates")

    # ----------------------------------------------------------------
    # 3️⃣  Refine each selected candidate
    # ----------------------------------------------------------------
    for idx, cand in enumerate(refinement_seeds):
        if time.time() >= deadline - 5.0:
            break

        n = len(cand)
        print(f"\n--- Refinement of candidate #{idx+1} (n={n}) ---")

        # ----- smooth exponentiated‑gradient phase -----
        remaining = deadline - time.time()
        smooth_deadline = time.time() + remaining * 0.45  # ~45 % of what remains
        smooth_a, smooth_proxy = refine_smooth_schedule(
            a=cand,
            deadline=smooth_deadline,
            eta0=1.0,
            beta_start=5.0,
            beta_end=40.0,
            report_every=0,

"""
Propose a sequence of non‑negative heights that minimises

    2·n·max(conv(a,a)) / (sum(a)²)

The algorithm works as follows:

1️⃣  Build a large pool of simple candidates (arcsine, power‑law,
    exponential‑decay, tent, uniform, random Dirichlet).  All vectors are
    normalised to a probability vector (sum = 1) and quickly ranked by a
    fast FFT‑based proxy for the objective.

2️⃣  Take the best few candidates and refine each of them in two stages:

   • **Smooth exponentiated‑gradient descent** – a multiplicative update
     with a *soft‑max* weight on the convolution entries.
     The temperature β is increased from ``beta_start`` to ``beta_end``
     during the run, which gradually turns the smooth approximation
     into the true max.

   • **Hard additive sub‑gradient descent** – the exact sub‑gradient of
     the max‑convolution is used (uniform weight on the maximisers) and
     after every step the vector is projected back onto the simplex.

   The best vector seen (according to the official ``evaluate_sequence``
   routine) is kept as the global optimum.

3️⃣  If the time budget expires before any refinement finishes, a safe
   uniform vector is returned.

The whole routine respects a user‑provided time limit (default 1000 s)
and always returns a feasible list of floats.
"""

import time
import numpy as np
from typing import List, Tuple

# ----------------------------------------------------------------------
# Low‑level numeric utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """
    Linear convolution via real FFT (full length, zero‑padded to a power‑of‑2).
    """
    m = len(a) + len(b) - 1
    n_fft = next_pow2(m)
    A = np.fft.rfft(a, n=n_fft)
    B = np.fft.rfft(b, n=n_fft)
    C = A * B
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


def normalize(v: np.ndarray) -> np.ndarray:
    """Project a non‑negative vector onto the simplex (sum = 1)."""
    total = v.sum()
    if total <= 0.0:
        return np.full_like(v, 1.0 / len(v))
    return v / total


def simplex_projection(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the probability simplex.
    Implements the algorithm of Duchi et al. (2008).
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    rho = np.where(u - cssv / (np.arange(n) + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def compute_score(a: np.ndarray) -> float:
    """
    Fast proxy of the official objective.
    The vector a is assumed to be non‑negative and already normalised
    (∑a = 1), therefore the score reduces to 2·n·max(conv(a,a)).
    """
    n = len(a)
    conv = convolution_fft(a, a)
    max_val = float(np.max(conv))
    return 2.0 * n * max_val


def gradient_from_weights(a: np.ndarray, w: np.ndarray) -> np.ndarray:
    """
    Gradient of Σ_k w_k·(a*a)_k w.r.t. a.
    Implements   g = 2·conv(a, w_rev)   (only the central n entries are needed).
    """
    w_rev = w[::-1]
    conv = convolution_fft(a, w_rev)               # length = 3·n‑2
    idx = 2 * len(a) - 2                           # position 2n‑2
    grads = 2.0 * conv[idx - np.arange(len(a))]
    return grads


# ----------------------------------------------------------------------
# Parametric families for the initial pool
# ----------------------------------------------------------------------


def arcsine_initial(n: int) -> np.ndarray:
    """Discrete arcsine distribution (≈ 1/√(i·(n‑1‑i))) normalised."""
    i = np.arange(n, dtype=float)
    eps = 0.25
    vals = 1.0 / np.sqrt((i + 0.5) * (n - 0.5 - i) + eps)
    return normalize(vals)


def powerlaw_shape(n: int, p: float, eps: float) -> np.ndarray:
    """Symmetric power‑law shape a_i ∝ ((i+ε)(n‑1‑i+ε))⁻ᵖ."""
    i = np.arange(n, dtype=float)
    base = ((i + eps) * (n - 1 - i + eps)) ** (-p)
    return normalize(base)


def uniform_shape(n: int) -> np.ndarray:
    """Uniform probability vector on n points."""
    return np.full(n, 1.0 / n)


def tent_shape(n: int) -> np.ndarray:
    """Symmetric triangular (tent) shape."""
    i = np.arange(n, dtype=float)
    vals = np.minimum(i + 1, n - i)
    return normalize(vals)


def random_dirichlet(n: int, rng: np.random.Generator) -> np.ndarray:
    """Draw a random point on the simplex (exponential sampling)."""
    samp = rng.exponential(1.0, size=n)
    return normalize(samp)


def exponential_decay_shape(n: int, lam: float) -> np.ndarray:
    """Symmetric exponential decay from the centre."""
    i = np.arange(n, dtype=float)
    centre = (n - 1) / 2.0
    vals = np.exp(-lam * np.abs(i - centre))
    return normalize(vals)


# ----------------------------------------------------------------------
# Refinement methods
# ----------------------------------------------------------------------


def refine_smooth_schedule(
    a: np.ndarray,
    deadline: float,
    *,
    eta0: float = 1.0,
    beta_start: float = 5.0,
    beta_end: float = 40.0,
    report_every: int = 0,
) -> Tuple[np.ndarray, float]:
    """
    Multiplicative mirror‑descent with a temperature that gradually rises.
    The weight vector for the gradient is a softened max:
        w_i ∝ exp(β·(conv_i – max(conv))).
    """
    a = normalize(a.astype(float).copy())
    n = len(a)
    best_a = a.copy()
    best_score = 2.0 * n * float(np.max(convolution_fft(a, a)))

    it = 0
    while time.time() < deadline:
        it += 1
        conv = convolution_fft(a, a)
        max_val = float(np.max(conv))

        # linearly raise β (capped at beta_end)
        beta = beta_start + (beta_end - beta_start) * min(1.0, it * 1e-4)

        # soft‑max weights
        exp_vals = np.exp(beta * (conv - max_val))
        w = exp_vals / np.sum(exp_vals)

        grad = gradient_from_weights(a, w)

        eta = eta0 / np.sqrt(it)

        a = a * np.exp(-eta * grad)
        a = normalize(a)

        cur_score = 2.0 * n * max_val
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()

        if report_every and (it % report_every == 0):
            print(f"    smooth iter={it:,d} cur={cur_score:.6f} best={best_score:.6f}")

    return best_a, best_score


def refine_subgradient_additive(
    a: np.ndarray,
    deadline: float,
    *,
    eta0: float = 0.7,
    max_iter: int = 200_000,
    report_every: int = 0,
) -> Tuple[np.ndarray, float]:
    """
    Classical additive sub‑gradient descent on the actual max‑convolution.
    After each step the iterate is projected back onto the simplex.
    """
    a = normalize(a.astype(float).copy())
    n = len(a)
    best_a = a.copy()
    best_score = 2.0 * n * float(np.max(convolution_fft(a, a)))

    for it in range(1, max_iter + 1):
        if time.time() >= deadline:
            break

        conv = convolution_fft(a, a)
        max_val = float(np.max(conv))

        # uniform weight on (near) maximisers
        w = np.zeros_like(conv)
        max_idx = np.where(conv >= max_val - 1e-14)[0]
        if max_idx.size > 0:
            w[max_idx] = 1.0 / max_idx.size
        else:
            w[np.argmax(conv)] = 1.0

        grad = gradient_from_weights(a, w)

        eta = eta0 / np.sqrt(it)
        a = a - eta * grad
        a = simplex_projection(a)

        cur_score = 2.0 * n * max_val
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()

        if report_every and (it % report_every == 0):
            print(f"    sub‑grad iter={it:,d} cur={cur_score:.6f} best={best_score:.6f}")

    return best_a, best_score
