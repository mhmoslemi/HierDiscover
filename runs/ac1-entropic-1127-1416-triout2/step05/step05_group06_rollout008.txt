def smooth_max_descent(a, betas, it_per_beta, deadline):
    best_a=a.copy()
    best_sc=compute_score(a)
    n=len(a)
    for beta in betas:
        for it in range(it_per_beta):
            if time.time()>=deadline: return best_a,best_sc
            conv = convolution_fft(a,a)
            max_c=conv.max()
            # stable softmax: exp(beta*(conv - max_c))
            exp_vals = np.exp(beta * (conv - max_c))
            p = exp_vals / exp_vals.sum()
            # gradient: 4*n * cross-correlation of p and a
            N = next_pow2(len(p) + n - 1)
            P = np.fft.fft(p, N)
            A = np.fft.fft(a, N)
            cross = np.fft.ifft(P * np.conj(A)).real[:n]
            grad = 4.0 * n * cross
            # step size
            eta = 0.5 / np.sqrt(it + 1)
            a = a * np.exp(-eta * grad)
            a_sum = a.sum()
            if a_sum == 0: a = np.full_like(a, 1.0/n)
            else: a = a / a_sum
            sc=compute_score(a)
            if sc<best_sc: best_sc=sc; best_a=a.copy()
    return best_a,best_sc

"""
Multi‑scale exponential‑gradient optimisation for the max‑convolution bound.

The algorithm builds a hierarchy of increasingly fine step functions.
At a very low resolution (n≈256) cheap exponentiated‑gradient sub‑gradient
iterations are run on several diverse initial shapes (arcsine, Dirichlet,
beta‑like, power‑law).  The best low‑resolution shape is up‑sampled to the
next resolution and refined further.  This proceeds up to a final length
(≈4096–8192 depending on the remaining wall‑clock budget).  The update
rule is

    a ← a ⊙ exp(‑η·∇Φ(a)),   a ← a / Σ a,

where Φ(a) = max (a∗a) and ∇Φ is taken as the average sub‑gradient over all
active convolution indices.  The learning‑rate η decays as 1/√t.
A final call to `evaluate_sequence` returns the officially scored sequence.

The method stays inside the 1000 s wall‑clock budget, always returns a
feasible non‑negative list and aims at beating the previous bound
≈1.514.
"""

import time
import numpy as np
from typing import List, Tuple

# ----------------------------------------------------------------------
# Basic numeric utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full linear convolution via real FFT (zero‑padded to a power of two)."""
    m = len(a) + len(b) - 1
    n_fft = next_pow2(m)
    A = np.fft.rfft(a, n=n_fft)
    B = np.fft.rfft(b, n=n_fft)
    C = A * B
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


def compute_score(a: np.ndarray) -> float:
    """Fast surrogate of the official objective (scale‑invariant)."""
    n = len(a)
    conv = convolution_fft(a, a)
    max_val = float(conv.max())
    s = float(a.sum())
    if s < 1e-12:
        return float("inf")
    return 2.0 * n * max_val / (s * s)


# ----------------------------------------------------------------------
# Families of seed shapes (all are normalised to sum‑to‑one)
# ----------------------------------------------------------------------


def arcsine_initial(n: int) -> np.ndarray:
    """Discrete arcsine density (symmetrised, ε prevents division by zero)."""
    i = np.arange(n, dtype=float)
    eps = 0.25
    vals = 1.0 / np.sqrt((i + 0.5) * (n - 0.5 - i) + eps)
    return vals / vals.sum()


def generate_powerlaw(n: int, p: float, eps: float) -> np.ndarray:
    """Symmetric power‑law a_i ∝ ((i+ε)(n‑1‑i+ε))⁻ᵖ."""
    i = np.arange(n, dtype=float)
    base = ((i + eps) * (n - 1 - i + eps)) ** (-p)
    total = base.sum()
    if total == 0.0:
        return np.full(n, 1.0 / n)
    return base / total


def generate_beta_like(n: int, alpha: float, beta: float) -> np.ndarray:
    """Beta‑like shape a_i ∝ (i+½)^{α‑1} (n‑1‑i+½)^{β‑1}."""
    i = np.arange(n, dtype=float)
    base = (i + 0.5) ** (alpha - 1) * (n - 1 - i + 0.5) ** (beta - 1)
    total = base.sum()
    if total == 0.0:
        return np.full(n, 1.0 / n)
    return base / total


def random_dirichlet(n: int, concentration: float, rng: np.random.Generator) -> np.ndarray:
    """Dirichlet‑like draw (Gamma when concentration>0, otherwise stretched exponential)."""
    if concentration <= 0:
        samp = rng.exponential(1.0, size=n) ** (1.0 / max(0.1, concentration))
    else:
        samp = rng.gamma(concentration, 1.0, size=n)
    samp += 1e-12
    return samp / samp.sum()


# ----------------------------------------------------------------------
# Sub‑gradient of the true max‑convolution
# ----------------------------------------------------------------------


def _max_conv_gradient(a: np.ndarray, conv: np.ndarray) -> np.ndarray:
    """
    Sub‑gradient of Φ(a)=max_k (a∗a)_k.
    For every active index k we add 2·a_{k‑j} to the gradient entry j.
    The result is averaged over the number of active indices.
    """
    n = len(a)
    max_val = conv.max()
    # active offsets (tolerance accounts for floating‑point noise)
    mask = np.isclose(conv, max_val, atol=1e-15, rtol=0.0)
    active = np.where(mask)[0]
    grad = np.zeros_like(a)

    for k in active:
        lo = max(0, k - (n - 1))
        hi = min(k, n - 1)
        idx = np.arange(lo, hi + 1, dtype=int)
        grad[idx] += 2.0 * a[k - idx]

    if len(active) > 1:
        grad /= len(active)
    return grad


# ----------------------------------------------------------------------
# Exponentiated‑gradient optimiser (sub‑gradient on the true max)
# ----------------------------------------------------------------------


def subgradient_optimize(a: np.ndarray,
                         deadline: float,
                         *,
                         max_iters: int,
                         eta0: float = 0.6,
                         stage_name: str = "") -> Tuple[np.ndarray, float]:
    """
    Runs exponentiated‑gradient descent on Φ(a)=max (a∗a).
    Returns the best vector seen and its fast (compute_score) value.
    """
    a = np.maximum(a, 1e-30)
    a = a / a.sum()
    best_a = a.copy()
    best_sc = compute_score(a)
    n = len(a)

    it = 0
    while it < max_iters and time.time() < deadline:
        conv = convolution_fft(a, a)
        grad = _max_conv_gradient(a, conv)

        it += 1
        eta = eta0 / np.sqrt(it)
        a = a * np.exp(-eta * grad)
        a = a / a.sum()

        cur_sc = compute_score(a)
        if cur_sc < best_sc:
            best_sc = cur_sc
            best_a = a.copy()
            print(f"{stage_name} iter={it:,d} NEW BEST score={best_sc:.6f}")

        if it % 2000 == 0:
            print(f"{stage_name} iter={it:,d} cur={cur_sc:.6f} best={best_sc:.6f}")

    return best_a, best_sc


# ----------------------------------------------------------------------
# Helper: upscale a low‑resolution shape to a higher resolution
# ----------------------------------------------------------------------


def upscale(arr: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation from len(arr) to new_n (preserves symmetry)."""
    old_n = len(arr)
    # positions of bin centres in [0,1)
    x_old = np.linspace(0.0, 1.0, old_n, endpoint=False) + 0.5 / old_n
    x_new = np.linspace(0.0, 1.0, new_n, endpoint=False) + 0.5 / new_n
    return np.interp(x_new, x_old, arr)


# ----------------------------------------------------------------------
# Main entry point required by the competition harness
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs) -> List[float]:
    """
    Searches for a non‑negative step function with a low evaluation score.
    The routine respects the wall‑clock budget and always returns a feasible
    list of floats.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    # keep a small safety margin for the final evaluation call
    deadline = start_time + budget_s - 2.0

    best_seq: List[float] = []
    best_val: float = float("inf")

    # ------------------------------------------------------------------
    # (optional) fallback to a previously known construction
    # ------------------------------------------------------------------
    try:
        fallback = [float(x) for x in height_sequence_1]  # type: ignore
        val = evaluate_sequence(fallback)
        print(f"Fallback (height_sequence_1) score = {val:.6f}")
        if val < best_val:
            best_val, best_seq = val, fallback
    except Exception:
        pass

    # ------------------------------------------------------------------
    # Stage 0 – low resolution exploration (n≈256)
    # ------------------------------------------------------------------
    n0 = 256
    init_candidates = [
        arcsine_initial(n0),
        random_dirichlet(n0, 0.5, rng),
        random_dirichlet(n0, 1.0, rng),
        random_dirichlet(n0, 2.0, rng),
        generate_beta_like(n0, 0.5, 0.5),
        generate_powerlaw(n0, 0.5, eps=1e-3),
    ]

    print("\n=== Stage 0: low‑resolution optimisation (n=256) ===")
    best_a0 = None
    best_sc0 = float("inf")

    # allocate ~150 k sub‑gradient iterations total for stage 0
    max_iters_per_candidate = 30000
    for idx, cand in enumerate(init_candidates):
        if time.time() >= deadline - 5.0:
            break
        print(f"  Candidate {idx+1}/{len(init_candidates)}")
        a_opt, sc_opt = subgradient_optimize(
            cand,
            deadline,
            max_iters=max_iters_per_candidate,
            eta0=0.65,
            stage_name=f"subgrad0-{idx+1}"
        )
        if sc_opt < best_sc0:
            best_sc0, best_a0 = sc_opt, a_opt

    if best_a0 is None:
        # sanity fallback – uniform vector
        best_a0 = np.full(n0, 1.0 / n0)
        best_sc0 = compute_score(best_a0)

    print(f"Best low‑res score after stage 0: {best_sc0:.6f}")

    # ------------------------------------------------------------------
    # Multi‑scale refinement: 256 → 1024 → 4096 (→ 8192 if time permits)
    # ------------------------------------------------------------------
    # Define the hierarchy of target resolutions and iteration budgets.
    stages = [
        (1024, 80000, 0.55),   # (target_n, max_iters, eta0)
        (4096, 40000, 0.45),
        (8192, 15000, 0.35),   # final stage, may be skipped if time runs out
    ]

    a_current = best_a0.copy()
    for stage_idx, (n_target, max_iters, eta0) in enumerate(stages, start=1):
        if time.time() >= deadline - 5.0:
            break
        print(f"\n=== Stage {stage_idx}: upscale to n={n_target} ===")
        a_current = upscale(a_current, n_target)
        a_current = a_current / a_current.sum()

        # Run a modest number of sub‑gradient steps at the new resolution.
        a_current, sc_current = subgradient_optimize(
            a_current,
            deadline,
            max_iters=max_iters,
            eta0=eta0,
            stage_name=f"subgrad{stage_idx}"
        )
        # keep track of the global best
        if sc_current < best_sc0:
            best_sc0, best_seq = sc_current, a_current.copy()
            print(f"*** New global best after stage {stage_idx}: {best_sc0:.6f} ***")

    # ------------------------------------------------------------------
    # Final polishing (if any time remains)
    # ------------------------------------------------------------------
    if time.time() < deadline - 5.0:
        print("\n=== Final polishing (sub‑gradient) ===")
        remaining_iters = 200000
        a_current, sc_current = subgradient_optimize(
            a_current,
            deadline,
            max_iters=remaining_iters,
            eta0=0.3,
            stage_name="final"
        )
        if sc_current < best_sc0:
            best_sc0, best_seq = sc_current, a_current.copy()
            print(f"*** Updated global best after final polishing: {best_sc0:.6f} ***")

    # ------------------------------------------------------------------
    # Convert to plain list, evaluate with the official scorer and return.
    # ------------------------------------------------------------------
    if not best_seq:
        # should never happen, but keep a safe fallback
        best_seq = [1.0] * 256

    final_val = evaluate_sequence(best_seq.tolist())
    print(f"\nSearch finished. Official evaluated score = {final_val:.6f}")

    # keep the best official score (might be different from fast score)
    if final_val < best_val:
        best_val = final_val
        best_seq = best_seq

    return best_seq.tolist()
