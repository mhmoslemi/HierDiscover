def gradient_smooth_max(a, gamma):
    conv = convolution_fft(a, a)
    maxv = np.max(conv)
    # subtract max for stability
    exps = np.exp(gamma * (conv - maxv))
    w = exps / np.sum(exps)
    w_rev = w[::-1]
    L = 2 * len(a) - 1
    grad = 2.0 * convolution_fft(a, w_rev)[L - 1 - np.arange(len(a))]
    return grad

def adam_optimize(a0, deadline, max_iter, lr0=0.05, gamma0=10.0, gamma_factor=1.02, gamma_max=300.0):
    a = a0.copy()
    n = len(a)
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    t = 0
    gamma = gamma0
    best = a.copy()
    best_score = compute_score(a)

    while t < max_iter and time.time() < deadline:
        t += 1
        # optionally increase gamma gradually
        gamma = min(gamma_max, gamma * gamma_factor)
        grad = gradient_smooth_max(a, gamma)
        # Adam update
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad * grad)
        m_hat = m / (1 - beta1**t)
        v_hat = v / (1 - beta2**t)
        step = lr0 * m_hat / (np.sqrt(v_hat) + eps)
        a = a - step
        a = project_to_simplex(a)
        # evaluate every some iterations
        if t % 2000 == 0:
            sc = compute_score(a)
            if sc < best_score:
                best_score = sc
                best = a.copy()
            # optional: print progress
    return best

"""
Propose a low‑discrepancy step function.

The algorithm works in three phases:

1️⃣  Generate a diverse pool of candidates with a few parametric families
    (arcsine, beta, power‑shape, mixtures, triangular bumps).  The best
    few seeds (according to the fast surrogate 2·n·max(conv)) are kept.

2️⃣  For each seed a two‑stage refinement is performed:
    – a short sub‑gradient descent on the exact L∞‑objective,
    – a longer Adam optimisation on a *smooth* approximation of the
      max (log‑sum‑exp of the convolution).  After each update the
      vector is projected onto the probability simplex.

    Finally an exponentiated‑gradient polish (mirror descent) is applied.

3️⃣  If time permits the current global best undergoes a final polish.

All heavy work respects a global time budget (default 1000 s).  The
returned list is a non‑negative step sequence that minimises the
official ``evaluate_sequence`` objective.
"""

import time
import numpy as np
from typing import List, Tuple

# ----------------------------------------------------------------------
# Low‑level utilities (no closures)
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Linear convolution via real FFT (length = len(a)+len(b)-1)."""
    m = len(a) + len(b) - 1
    n_fft = next_pow2(m)
    A = np.fft.rfft(a, n=n_fft)
    B = np.fft.rfft(b, n=n_fft)
    C = A * B
    conv = np.fft.ifft(C, n=n_fft)[:m].real
    return conv


def compute_score(a: np.ndarray) -> float:
    """
    Fast surrogate for the official score.
    The vector is assumed to be normalised to Σa = 1,
    therefore the denominator disappears and the score is simply
    2·n·max(conv(a,a)).
    """
    n = a.shape[0]
    conv = convolution_fft(a, a)
    return 2.0 * n * float(np.max(conv))


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection onto the probability simplex {x ≥ 0, Σx = 1}.
    Duchi et al., 2008.
    """
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / v.size)

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho_arr = u - cssv / np.arange(1, v.size + 1)
    rho = np.where(rho_arr > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w

# Shape generators
# ----------------------------------------------------------------------


def arcsine_initial(n: int) -> np.ndarray:
    """Classic arcsine shape: a_i ∝ 1/√((i+½)*(n‑½‑i))."""
    i = np.arange(n, dtype=float)
    eps = 0.25
    vals = 1.0 / np.sqrt((i + 0.5) * (n - 0.5 - i) + eps)
    vals /= vals.sum()
    return vals


def generate_beta_shape(n: int, alpha: float, beta: float, eps: float) -> np.ndarray:
    """
    (i+ε)^{α‑1}·(n‑1‑i+ε)^{β‑1} on {0,…,n‑1}, normalised to Σ=1.
    """
    i = np.arange(n, dtype=float)
    left = (i + eps) ** (alpha - 1.0)
    right = (n - 1 - i + eps) ** (beta - 1.0)
    base = left * right
    total = base.sum()
    if total <= 0.0 or not np.isfinite(total):
        return np.full(n, 1.0 / n)
    return base / total


def generate_power_shape(n: int, p: float, eps: float) -> np.ndarray:
    """
    Generalised arcsine: a_i ∝ ((i+½)*(n‑½‑i)+ε)^{‑p}.
    p>0, eps≥0.
    """
    i = np.arange(n, dtype=float)
    denom = ((i + 0.5) * (n - 0.5 - i) + eps) ** p
    vals = 1.0 / denom
    total = vals.sum()
    if total <= 0.0 or not np.isfinite(total):
        return np.full(n, 1.0 / n)
    return vals / total


def generate_triangular_shape(n: int, peak: int) -> np.ndarray:
    """Symmetric triangular bump with apex at ``peak``."""
    left = np.arange(peak + 1, dtype=float)
    right = np.arange(n - peak - 1, dtype=float)[::-1]
    seq = np.concatenate([left, right])
    seq /= seq.sum()
    return seq


def generate_mixture_shape(n: int, weight_arc: float) -> np.ndarray:
    """Mixture (weight_arc)·arcsine + (1‑weight_arc)·uniform."""
    arc = arcsine_initial(n)
    uni = np.full(n, 1.0 / n)
    mix = weight_arc * arc + (1.0 - weight_arc) * uni
    mix /= mix.sum()
    return mix

def subgradient_descent_exact(initial_a: np.ndarray,
                              deadline: float,
                              max_iter: int = 80_000,
                              step0: float = 0.05) -> np.ndarray:
    """
    Sub‑gradient descent on the exact objective f(a)=2·n·max(conv(a,a)).
    The step size is η_t = step0 / √t.  Projection onto the simplex enforces Σa = 1.
    """
    a = project_to_simplex(initial_a.copy())
    n = len(a)
    L = 2 * n - 1

    for it in range(1, max_iter + 1):
        if time.time() >= deadline:
            break

        conv = convolution_fft(a, a)
        max_val = float(np.max(conv))

        # mask of all (near‑max) indices
        tol = max_val * 1e-12 + 1e-14
        mask = (np.abs(conv - max_val) <= tol).astype(float)

        # sub‑gradient = 2·sum_{k∈K} shift(a,k)
        mask_rev = mask[::-1]
        conv_rev = convolution_fft(a, mask_rev)
        subgrad = 2.0 * conv_rev[L - 1 - np.arange(n)]

        eta = step0 / np.sqrt(it)
        a = a - eta * subgrad
        a = project_to_simplex(a)

        if it % 5000 == 0:
            cur_sc = compute_score(a)
            print(f"    SubGrad it={it:,d} score={cur_sc:.6f}")

    return a


def gradient_smooth_max(a: np.ndarray, gamma: float) -> np.ndarray:
    """
    Gradient of the smooth approximation
        f_gamma(a) = (1/γ)·log Σ_k exp(γ·conv_k)
    where conv = a*a (linear convolution).  Returns a vector of length len(a).
    """
    n = len(a)
    L = 2 * n - 1
    conv = convolution_fft(a, a)
    maxv = np.max(conv)
    # stable softmax
    exps = np.exp(gamma * (conv - maxv))
    w = exps / np.sum(exps)          # weights sum to 1
    # gradient = 2 * (a * w_rev) convolution
    w_rev = w[::-1]
    conv_rev = convolution_fft(a, w_rev)
    grad = 2.0 * conv_rev[L - 1 - np.arange(n)]
    return grad


def adam_optimize(initial_a: np.ndarray,
                  deadline: float,
                  max_iter: int = 100_000,
                  lr0: float = 0.05,
                  gamma0: float = 10.0,
                  gamma_factor: float = 1.02,
                  gamma_max: float = 250.0) -> np.ndarray:
    """
    Adam optimiser on the smooth‑max surrogate.
    The iterate is kept on the probability simplex by Euclidean projection.
    """
    a = project_to_simplex(initial_a.copy())
    n = len(a)
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8

    gamma = gamma0
    best_a = a.copy()
    best_sc = compute_score(a)

    for it in range(1, max_iter + 1):
        if time.time() >= deadline:
            break

        # increase smoothing gradually
        gamma = min(gamma_max, gamma * gamma_factor)

        grad = gradient_smooth_max(a, gamma)

        # Adam updates
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad * grad)

        m_hat = m / (1 - beta1 ** it)
        v_hat = v / (1 - beta2 ** it)

        step = lr0 * m_hat / (np.sqrt(v_hat) + eps)
        a = a - step
        a = project_to_simplex(a)

        if it % 3000 == 0:
            sc = compute_score(a)
            if sc < best_sc:
                best_sc = sc
                best_a = a.copy()
            print(f"    Adam it={it:,d} score={sc:.6f}")

    return best_a


def eg_refine(a: np.ndarray,
              deadline: float,
              max_iter: int = 100_000,
              eta0: float = 0.3) -> np.ndarray:
    """
    Mirror‑descent style polishing using an exponentiated‑gradient update.
    Keeps the iterate on the simplex automatically.
    """
    a = a.copy()
    n = len(a)
    L = 2 * n - 1

    for it in range(1, max_iter + 1):
        if time.time() >= deadline:
            break

        conv = convolution_fft(a, a)
        max_val = float(np.max(conv))
        tol = max_val * 1e-12 + 1e-14
        mask = (np.abs(conv - max_val) <= tol).astype(float)

        mask_rev = mask[::-1]
        conv_rev = convolution_fft(a, mask_rev)
        subgrad = 2.0 * conv_rev[L - 1 - np.arange(n)]

        eta = eta0 / (np.sqrt(it) + 1.0)
        a = a * np.exp(-eta * subgrad)

        total = a.sum()
        if total == 0.0:
            a = np.full_like(a, 1.0 / n)
        else:
            a = a / total

        if it % 5000 == 0:
            cur_sc = compute_score(a)
            print(f"    EG it={it:,d} score={cur_sc:.6f}")

    return a


def hillclimb_power_shape(n: int,
                          init_p: float,
                          init_eps: float,
                          deadline: float,
                          max_iters: int = 300) -> Tuple[np.ndarray, float]:
    """
    Simple random‑walk hill‑climbing on (p,eps) of the generalized arcsine family.
    Uses the fast surrogate.
    """
    p, eps = init_p, init_eps
    best_seq = generate_power_shape(n, p, eps)
    best_sc = compute_score(best_seq)

    for _ in range(max_iters):
        if time.time() >= deadline:
            break
        p_new = max(0.01, p * (1.0 + np.random.uniform(-0.15, 0.15)))
        eps_new = max(0.0, eps + np.random.uniform(-0.1, 0.1))
        cand = generate_power_shape(n, p_new, eps_new)
        sc = compute_score(cand)
        if sc < best_sc:
            p, eps, best_seq, best_sc = p_new, eps_new, cand, sc
    return best_seq, best_sc


def hillclimb_beta_shape(n: int,
                         init_alpha: float,
                         init_beta: float,
                         init_eps: float,
                         deadline: float,
                         max_iters: int = 300) -> Tuple[np.ndarray, float]:
    """
    Random‑walk hill‑climbing on the three parameters of the beta family.
    """
    alpha, beta, eps = init_alpha, init_beta, init_eps
    best_seq = generate_beta_shape(n, alpha, beta, eps)

        alpha_new = max(0.01, alpha * (1.0 + np.random.uniform(-0.15, 0.15)))

        eps_new = max(0.0, eps + np.random.uniform(-0.1, 0.1))
        cand = generate_beta_shape(n, alpha_new, beta_new, eps_new)

                                            step0=0.22)

        # Phase B – Adam on smooth‑max surrogate
        remaining = deadline - time.time()
        max_iter_adam = int(min(150_000, remaining * 28))
        refined = adam_optimize(refined, deadline,
                                max_iter=max_iter_adam,
                                lr0=0.07,
                                gamma0=12.0,
                                gamma_factor=1.02,
                                gamma_max=300.0)

        # Phase C – exponentiated‑gradient polish
        remaining = deadline - time.time()
        if remaining > 2.0:
            max_iter_eg = int(min(90_000, remaining * 28))
            refined = eg_refine(refined, deadline,
                                max_iter=max_iter_eg,
                                eta0=0.18)

        # Official evaluation
        try:
            val = evaluate_sequence(refined.tolist())
            print(f"  → official score = {val:.6f}")
            if val < best_val:
                best_val = val
                best_seq = refined.tolist()
                print(f"  *** New global best: {best_val:.6f} ***")
        except Exception as e:
            print(f"  Evaluation failed: {e}")

    # ------------------------------------------------------------------
    # 3.  Global polish on the current best (if time permits)
    # ------------------------------------------------------------------
    if best_seq and time.time() < deadline - 3.0:
        print("\nFinal global polish …")
        a = np.array(best_seq, dtype=float)

        remaining = deadline - time.time()
        max_iter_sg = int(min(200_000, remaining * 28))
        a = subgradient_descent_exact(a, deadline,
                                      max_iter=max_iter_sg,
                                      step0=0.12)

        remaining = deadline - time.time()
        if remaining > 2.0:
            max_iter_eg = int(min(120_000, remaining * 28))
            a = eg_refine(a, deadline,
                          max_iter=max_iter_eg,
                          eta0=0.16)

        final_val = evaluate_sequence(a.tolist())
        print(f"Final polished official score = {final_val:.6f}")
        if final_val < best_val:
            best_val = final_val
            best_seq = a.tolist()
            print("*** Updated global best after final polish ***")

    # ------------------------------------------------------------------
    # 4.  Guard – ensure a non‑empty return value
    # ------------------------------------------------------------------
    if not best_seq:
        fallback_n = 1024
        best_seq = [1.0 / fallback_n] * fallback_n
        best_val = evaluate_sequence(best_seq)

    print(f"\nSearch finished. Best reported official score = {best_val:.6f}")
    return best_seq
