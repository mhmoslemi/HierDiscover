"""
Search for a non‑negative step‑function (probability vector) that minimizes

    score = 2·n·max(conv(a,a))      (with ∑a = 1)

The algorithm consists of three phases:

1️⃣  Initialise several promising candidates (uniform, a discrete arcsine
    distribution and a random Dirichlet draw).  Keep the best one.

2️⃣  Iteratively improve the current vector by solving a *linear programme*:

       max   ∑ b_j
       s.t.  f * b  ≤  M   (component‑wise),
             b ≥ 0

    where ``f`` is the current vector, ``M = max(conv(f,f))`` and ``*``
    denotes linear convolution.  The optimal ``b`` is scaled to a
    probability vector ``g`` and mixed with ``f``:

       f ← (1‑t)·f + t·g   (t ≈ 0.2, adapted on the fly).

    Each LP is solved with ``scipy.optimize.linprog`` (HiGHS).  The loop
    continues until the time budget is exhausted.

3️⃣  A short exponentiated sub‑gradient descent (the method used in the
    reference code) and a simple hill‑climber polish finish the search.

All work respects the 1000 s limit; the function returns the best sequence
found.
"""

import time
import math
import numpy as np

# SciPy is only needed for the linear programme; we import lazily
try:
    import scipy.optimize as _opt
except Exception:  # pragma: no cover
    _opt = None

# ----------------------------------------------------------------------
# FFT‑based convolution (exact, fast for the sizes we use)
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Linear convolution a*b using a real FFT."""
    m = a.shape[0] + b.shape[0] - 1
    n_fft = _next_pow2(m)
    A = np.fft.rfft(a, n=n_fft)
    B = np.fft.rfft(b, n=n_fft)
    C = A * B
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


def compute_score(a: np.ndarray) -> float:
    """
    Fast replica of `evaluate_sequence` when ∑a = 1.
    Returns 2·n·max(a*a) exactly.
    """
    n = a.shape[0]
    conv = convolution_fft(a, a)
    return 2.0 * n * float(np.max(conv))


# ----------------------------------------------------------------------
# Euclidean projection onto the probability simplex
# ----------------------------------------------------------------------


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Project a vector onto the probability simplex (non‑negative, sum = 1)."""
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / v.shape[0])

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, v.shape[0] + 1)
    cond = u - cssv / ind > 0
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


# ----------------------------------------------------------------------
# Exponentiated sub‑gradient optimisation (used for polishing)
# ----------------------------------------------------------------------


def exp_subgrad_optimize(init_a: np.ndarray,
                         time_deadline: float,
                         max_iter: int = 2_000_000,
                         eta0: float = 0.5) -> tuple[np.ndarray, float]:
    """Exponentiated sub‑gradient descent on 2·n·max(conv(a,a))."""
    a = init_a.astype(float).copy()
    best_a = a.copy()
    best_score = compute_score(a)
    n = a.shape[0]

    for t in range(1, max_iter + 1):
        if time.time() >= time_deadline:
            break

        conv = convolution_fft(a, a)
        k = int(np.argmax(conv))

        # sub‑gradient of the active convolution index
        idx = np.arange(n)
        j = k - idx
        mask = (j >= 0) & (j < n)
        g = np.zeros_like(a)
        g[mask] = 2.0 * a[j[mask]]

        eta = eta0 / math.sqrt(t)
        a = a * np.exp(-eta * g)

        # re‑normalise onto the simplex
        s = a.sum()
        if s == 0.0:
            a = np.full_like(a, 1.0 / n)
        else:
            a = a / s

        cur_score = compute_score(a)
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()

        # tiny jitter every few thousand steps
        if t % 5000 == 0:
            jitter = np.random.randn(n) * (eta0 * 1e-4 / math.sqrt(t))
            a = project_to_simplex(a + jitter)

    return best_a, best_score


# ----------------------------------------------------------------------
# Simple stochastic hill‑climber (swap / mass transfer)
# ----------------------------------------------------------------------


def hill_climb(seq, time_deadline, max_steps=150_000):
    """Local search that swaps or transfers a fraction of mass."""
    a = np.array(seq, dtype=float)
    n = a.shape[0]

    best_a = a.copy()
    best_score = compute_score(best_a)

    step_scale = 0.1  # typical fraction moved

    while max_steps > 0 and time.time() < time_deadline:
        move_type = np.random.rand()
        cand = best_a.copy()

        if move_type < 0.5:
            # swap two positions
            i, j = np.random.randint(n, size=2)
            cand[i], cand[j] = cand[j], cand[i]
        else:
            # transfer a fraction from i to j
            i, j = np.random.randint(n, size=2)
            if i != j:
                delta = step_scale * best_a[i]
                cand[i] -= delta
                cand[j] += delta
                cand[i] = max(cand[i], 0.0)

        cand = project_to_simplex(cand)
        cur_score = compute_score(cand)

        if cur_score < best_score:
            best_score = cur_score
            best_a = cand
            step_scale = min(step_scale * 1.05, 0.5)
        else:
            step_scale *= 0.97

        max_steps -= 1

    return best_a.tolist(), best_score


# ----------------------------------------------------------------------
# Linear‑programming sub‑problem (maximising sum under convolution bound)
# ----------------------------------------------------------------------


def _build_conv_matrix(f: np.ndarray) -> np.ndarray:
    """
    Build the (2n‑1) × n matrix A with A[k, j] = f[k‑j]  (zero outside).
    Multiplying A @ b yields conv(f,b).
    """
    n = f.shape[0]
    m = 2 * n - 1
    A = np.zeros((m, n), dtype=float)
    # each column j is f placed starting at row j
    for j in range(n):
        A[j:j + n, j] = f
    return A


def lp_step(current_f: np.ndarray, M: float) -> np.ndarray:
    """
    Solve the LP:
        max   sum(b)
        s.t.  conv(current_f, b) ≤ M
              b ≥ 0
    Returns the normalised vector g (∑g = 1).  If the LP cannot be solved,
    returns the current_f unchanged.
    """
    if _opt is None:
        # SciPy not available – fall back to identity
        return current_f.copy()

    A = _build_conv_matrix(current_f)
    m = A.shape[0]
    n = current_f.shape[0]

    c = -np.ones(n)                     # maximise sum(b)  ⇔ minimise –sum(b)
    b_ub = np.full(m, M, dtype=float)  # right‑hand side
    bounds = [(0, None)] * n

    # HiGHS (default in recent SciPy) is fast for our size
    res = _opt.linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
    if not res.success or res.x is None:
        # In the unlikely event of failure we simply keep the current vector
        return current_f.copy()

    b_vec = res.x
    s = b_vec.sum()
    if s <= 0.0:
        return current_f.copy()
    g = b_vec / s
    return g

# Main entry point required by the evaluation harness
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 1234, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative sequence that minimises `evaluate_sequence`.
    The strategy mixes a linear‑programming sub‑problem, exponentiated
    sub‑gradient descent and a final hill‑climber.  It respects the 1000 s
    budget and returns the best sequence found.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 1.5   # a small safety margin

    # --------------------------------------------------------------
    # Length of the vector – 1024 is a good compromise between quality
    # and computational cost.
    # --------------------------------------------------------------
    n = 1024

    # --------------------------------------------------------------
    # Build a few deterministic starters (uniform & discrete arcsine)
    # --------------------------------------------------------------
    uniform = np.full(n, 1.0 / n)
    best_a = uniform.copy()
    best_score = compute_score(best_a)

    # Discrete arcsine distribution (Beta(½,½) analogue)
    idx = np.arange(n, dtype=float)
    arcsine_raw = 1.0 / np.sqrt((idx + 0.5) * (n - idx - 0.5))
    arcsine = arcsine_raw / arcsine_raw.sum()
    arcsine_score = compute_score(arcsine)
    if arcsine_score < best_score:
        best_score = arcsine_score
        best_a = arcsine.copy()
        print(f"Arcsine start score = {best_score:.6f}")

    # Optional fallback sequence supplied by the harness
    try:
        fallback = [float(x) for x in height_sequence_1]  # type: ignore
        fallback_a = np.array(fallback, dtype=float)
        if fallback_a.sum() == 0.0:
            fallback_a = np.full_like(fallback_a, 1.0 / fallback_a.shape[0])
        else:
            fallback_a = fallback_a / fallback_a.sum()
        fb_score = compute_score(fallback_a)
        print(f"Fallback score = {fb_score:.6f}")
        if fb_score < best_score:
            best_score = fb_score
            best_a = fallback_a.copy()
    except Exception:
        pass

    # Random Dirichlet start (adds stochastic diversity)
    rand_start = np.random.dirichlet(np.ones(n))
    rand_score = compute_score(rand_start)
    if rand_score < best_score:
        best_score = rand_score
        best_a = rand_start.copy()
        print(f"Random start score = {best_score:.6f}")

    # --------------------------------------------------------------
    # Phase 1 – iterative LP‑based improvement
    # --------------------------------------------------------------
    f = best_a.copy()
    t_step = 0.20               # mixing factor (will be adapted)
    iter_cnt = 0
    max_iter = 5000             # hard cap – the loop also stops on time

    while time.time() < deadline - 30.0 and iter_cnt < max_iter:
        conv_f = convolution_fft(f, f)
        M = float(conv_f.max())

        # solve the linear sub‑problem
        g = lp_step(f, M)

        # convex combination
        new_f = (1.0 - t_step) * f + t_step * g
        # re‑normalise (numerical drift)
        new_f = new_f / new_f.sum()

        new_score = compute_score(new_f)
        if new_score < best_score:
            best_score = new_score
            best_a = new_f.copy()
            f = new_f
            # successful step → we can be slightly bolder
            t_step = min(t_step * 1.03, 0.5)
        else:
            # no improvement – become more cautious
            t_step = max(t_step * 0.92, 0.01)
            # keep the old f (discard the non‑improving move)

        iter_cnt += 1
        if iter_cnt % 100 == 0:
            remaining = deadline - time.time()
            print(f"LP iter {iter_cnt:04d} – best score {best_score:.6f} – t {t_step:.3f} – time left {remaining:.1f}s")

    # --------------------------------------------------------------
    # Phase 2 – exponentiated sub‑gradient polishing (few thousand iters)
    # --------------------------------------------------------------
    if time.time() < deadline - 5.0:
        print("\nStarting exponentiated sub‑gradient polishing")
        polished_f, polished_score = exp_subgrad_optimize(
            best_a,
            time_deadline=deadline - 5.0,
            max_iter=1_000_000,
            eta0=0.5,
        )
        if polished_score < best_score:
            best_score = polished_score
            best_a = polished_f.copy()
            print(f"Sub‑gradient improved best to {best_score:.6f}")

    # --------------------------------------------------------------
    # Phase 3 – final hill‑climber refinement
    # --------------------------------------------------------------
    if time.time() < deadline - 1.0:
        print("\nFinal hill‑climb refinement")
        final_seq, final_val = hill_climb(
            best_a.tolist(),
            time_deadline=deadline,
            max_steps=200_000,
        )
        if final_val < best_score:
            best_score = final_val
            best_a = np.array(final_seq, dtype=float)
            print(f"Hill‑climb improved best to {best_score:.6f}")

    # --------------------------------------------------------------
    # Return the best sequence (evaluate_sequence will be called by the harness)
    # --------------------------------------------------------------
    final_score = evaluate_sequence(best_a.tolist())
    print(f"\nSearch finished. Final reported score = {final_score:.6f}")
    return best_a.tolist()
