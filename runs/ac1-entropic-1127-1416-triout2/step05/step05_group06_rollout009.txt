def solve_lp_max_sum_under_conv(f, M):
    n = len(f)
    # Build convolution matrix C (2n-1 x n)
    # using toeplitz
    from scipy.linalg import toeplitz
    from scipy.optimize import linprog

    # first column c: f then zeros length n-1
    c_col = np.concatenate([f, np.zeros(n-1)])
    # first row r: [f[0], zeros]
    r_row = np.concatenate([np.array([f[0]]), np.zeros(n-1)])
    C = toeplitz(c_col, r_row)  # shape (2n-1, n)
    # constraints: C @ g <= M
    A_ub = C
    b_ub = np.full(2*n - 1, M)
    # objective: maximize sum(g) -> linprog minimize -sum(g)
    obj = -np.ones(n)
    bounds = [(0, None)] * n

    # Use high-performance method 'highs' (default)
    res = linprog(c=obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
    if not res.success:
        # fallback: return original f (normalized)
        return f
    g = res.x
    # clip small negatives due to numeric
    g = np.maximum(g, 0.0)
    return g

"""
Hybrid heuristic for the step‑function minimisation problem.

Main ideas
----------

1. **Rich seed pool** – arcsine, beta‑like, power‑law and Dirichlet draws
   for a few well‑chosen lengths (powers of two, FFT‑friendly).

2. **Linear‑program “lift”** – given a current vector *f* with
   M = max( f∗f ), we solve the LP
        maximise   Σ g_i
        subject   f∗g ≤ M   (component‑wise)
                   g_i ≥ 0 .
   The solution *g* tries to put as much ℓ₁‑mass as possible while staying
   under the current maximal convolution.  Mixing *f*←(1−τ)f+τ g
   (τ≈0.3‑0.5) provably never increases the objective and often produces a
   noticeable drop.

3. **Smooth‑max gradient descent** – a log‑sum‑exp surrogate of the max
   is minimised with exponentiated‑gradient updates (FFT‑based cross‑correlation).

4. **Exact max‑convolution sub‑gradient polish** – the final candidate is
   refined with the true non‑smooth objective.

The routine respects the wall‑clock budget (default 1000 s) and always
returns a feasible non‑negative list of floats.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Basic numeric utilities (FFT based convolution, fast score)
# ----------------------------------------------------------------------
def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full linear convolution using a real FFT (zero‑padded to a power of two)."""
    m = len(a) + len(b) - 1
    n_fft = next_pow2(m)
    A = np.fft.rfft(a, n=n_fft)
    B = np.fft.rfft(b, n=n_fft)
    C = A * B
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


def compute_score(a: np.ndarray) -> float:
    """Fast version of the official objective (scale‑invariant)."""
    n = len(a)
    conv = convolution_fft(a, a)
    max_val = float(np.max(conv))
    s = float(np.sum(a))
    if s < 1e-12:
        return float("inf")
    return 2.0 * n * max_val / (s * s)


# ----------------------------------------------------------------------
# Families of parametric initial shapes
# ----------------------------------------------------------------------
def arcsine_initial(n: int) -> np.ndarray:
    """Discrete arcsine density (normalised to sum = 1)."""
    i = np.arange(n, dtype=float)
    eps = 0.25
    vals = 1.0 / np.sqrt((i + 0.5) * (n - 0.5 - i) + eps)
    return vals / vals.sum()


def generate_powerlaw(n: int, p: float, eps: float) -> np.ndarray:
    """Symmetric power‑law a_i ∝ ((i+ε)(n‑1‑i+ε))⁻ᵖ."""
    i = np.arange(n, dtype=float)
    base = ((i + eps) * (n - 1 - i + eps)) ** (-p)
    total = base.sum()
    if total == 0.0:
        return np.full(n, 1.0 / n)
    return base / total


def generate_beta_like(n: int, alpha: float, beta: float) -> np.ndarray:
    """Beta‑like shape a_i ∝ (i+½)^{α‑1} (n‑1‑i+½)^{β‑1}."""
    i = np.arange(n, dtype=float)
    base = (i + 0.5) ** (alpha - 1) * (n - 1 - i + 0.5) ** (beta - 1)
    total = base.sum()
    if total == 0.0:
        return np.full(n, 1.0 / n)
    return base / total


def random_dirichlet(n: int, concentration: float, rng: np.random.Generator) -> np.ndarray:
    """Sample a Dirichlet‑like vector (Gamma draws) and normalise."""
    if concentration <= 0:
        samp = rng.exponential(1.0, size=n) ** (1.0 / max(0.1, concentration))
    else:
        samp = rng.gamma(concentration, 1.0, size=n)
    samp += 1e-12
    return samp / samp.sum()


# ----------------------------------------------------------------------
# Linear‑program “lift” step
# ----------------------------------------------------------------------
def _solve_lp_max_sum_under_conv(f: np.ndarray, M: float) -> np.ndarray:
    """
    Solve   maximise Σ g_i   subject to   f∗g ≤ M,   g ≥ 0.
    Returns the optimal g (may be the unchanged f on failure).
    """
    from scipy.linalg import toeplitz
    from scipy.optimize import linprog

    n = len(f)
    # convolution matrix C (shape (2n‑1, n)) so that C @ g = f∗g
    c_col = np.concatenate([f, np.zeros(n - 1)])
    r_row = np.concatenate([np.array([f[0]]), np.zeros(n - 1)])
    C = toeplitz(c_col, r_row)                     # dense, but n≤4096 is OK
    b = np.full(2 * n - 1, M)

    # maximise Σ g_i  →  minimise −Σ g_i
    obj = -np.ones(n)
    bounds = [(0, None)] * n

    res = linprog(c=obj, A_ub=C, b_ub=b, bounds=bounds, method="highs", options={"presolve": True})
    if not res.success:
        # fallback: keep the current vector
        return f.copy()
    g = np.maximum(res.x, 0.0)
    return g


def lp_refine(initial_f: np.ndarray,
              mixing: float,
              max_iter: int,
              deadline: float) -> tuple[np.ndarray, float]:
    """
    Repeated LP lift – each iteration solves the LP, mixes the solution
    with the current vector and renormalises to unit ℓ₁‑norm.
    Returns the best vector found and its fast score.
    """
    f = initial_f.copy()
    f = f / f.sum()
    best_f = f.copy()
    best_sc = compute_score(f)

    for it in range(max_iter):
        if time.time() >= deadline:
            break
        conv = convolution_fft(f, f)
        M = float(conv.max())
        g = _solve_lp_max_sum_under_conv(f, M)
        # normalise g to unit sum (scaling down never violates the bound)
        s = g.sum()
        if s == 0.0:
            continue
        g = g / s
        # convex combination
        f = (1.0 - mixing) * f + mixing * g
        f = f / f.sum()
        sc = compute_score(f)
        if sc < best_sc:
            best_sc = sc
            best_f = f.copy()
        # gradually shrink the mixing to stabilise
        mixing = max(0.05, mixing * 0.94)
    return best_f, best_sc


# ----------------------------------------------------------------------
# Smooth‑max gradient descent (log‑sum‑exp surrogate)
# ----------------------------------------------------------------------
def softmax_gradient_descent(initial_a: np.ndarray,
                             deadline: float,
                             betas: list[float],
                             iter_per_beta: int = 400,
                             eta0: float = 0.5) -> tuple[np.ndarray, float]:
    """
    Minimise a smooth approximation of max‑convolution:
        Φ_β(a) = (1/β)·log Σ_k exp(β·(a∗a)_k).

    Uses an exponentiated‑gradient update and a simple simplex projection.
    """
    a = np.maximum(initial_a, 1e-30)
    a = a / a.sum()
    best_a = a.copy()
    best_sc = compute_score(a)

    n = len(a)
    for beta in betas:
        for it in range(1, iter_per_beta + 1):
            if time.time() >= deadline:
                return best_a, best_sc
            conv = convolution_fft(a, a)                  # length 2n‑1
            max_c = conv.max()
            exp_vals = np.exp(beta * (conv - max_c))
            p = exp_vals / exp_vals.sum()                 # soft‑max weights
            # cross‑correlation p ⋆ a  (length n)
            N = next_pow2(len(p) + n - 1)
            P = np.fft.fft(p, N)
            A = np.fft.fft(a, N)
            cross = np.fft.ifft(P * np.conj(A)).real[:n]
            cross = np.maximum(cross, 0.0)
            grad = 4.0 * n * cross                         # always ≥0
            eta = eta0 / np.sqrt(it)
            a = a * np.exp(-eta * grad)
            s = a.sum()
            a = a / s if s > 0 else np.full_like(a, 1.0 / n)
            cur_sc = compute_score(a)
            if cur_sc < best_sc:
                best_sc = cur_sc
                best_a = a.copy()
    return best_a, best_sc


# ----------------------------------------------------------------------
# Exact max‑convolution sub‑gradient polishing
# ----------------------------------------------------------------------
def _max_conv_gradient(a: np.ndarray, conv: np.ndarray) -> np.ndarray:
    """Sub‑gradient of max_k (a∗a)_k (averaged over all active indices)."""
    n = len(a)
    max_val = conv.max()
    mask = np.isclose(conv, max_val, atol=1e-15, rtol=0.0)
    active = np.where(mask)[0]
    grad = np.zeros_like(a)

    for k in active:
        lo = max(0, k - (n - 1))
        hi = min(k, n - 1)
        idx = np.arange(lo, hi + 1, dtype=int)
        grad[idx] += 2.0 * a[k - idx]

    if len(active) > 1:
        grad /= len(active)
    return grad


def optimise_subgradient(initial_a: np.ndarray,
                         deadline: float,
                         eta0: float = 0.3,
                         max_iter: int = 200_000) -> tuple[np.ndarray, float]:
    """
    Exponentiated sub‑gradient descent on the true max‑convolution.
    Returns the best vector seen and its (fast) score.
    """
    a = np.maximum(initial_a, 1e-30)
    a = a / a.sum()
    best_a = a.copy()
    best_sc = compute_score(a)
    n = len(a)

    for it in range(1, max_iter + 1):
        if time.time() >= deadline:
            break
        conv = convolution_fft(a, a)
        grad = _max_conv_gradient(a, conv)
        eta = eta0 / np.sqrt(it)
        a = a * np.exp(-eta * grad)
        s = a.sum()
        a = a / s if s > 0 else np.full_like(a, 1.0 / n)
        cur_sc = compute_score(a)
        if cur_sc < best_sc:
            best_sc = cur_sc
            best_a = a.copy()
        if it % 2000 == 0:
            print(f"    subgrad iter={it:,d} cur={cur_sc:.6f} best={best_sc:.6f}")
    return best_a, best_sc
