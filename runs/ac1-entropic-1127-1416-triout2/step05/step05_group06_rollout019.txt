"""
Hybrid optimizer for the “max–convolution” step‑function challenge.

The search works in several stages, all respecting the wall‑clock budget
(1000 s by default).

1.  **Seed generation** – a rich pool of starting vectors is created:
    uniform, arcsine, power‑law, beta‑like (including very heavy tails
    α,β∈[0.05,0.5]), exponential decays and Dirichlet draws.  The cheap
    internal score (2·n·max(a∗a)/(∑a)²) is used to keep the most promising
    few.

2.  **Local stochastic improvement** – a short simulated‑annealing run
    moves a tiny amount of mass between random bins while staying on the
    simplex.

3.  **Smooth‑max gradient descent** – the non‑smooth max is replaced by a
    log‑sum‑exp surrogate.  A schedule of increasing β (5 → 960) and several
    hundred inner iterations per β performs an exponentiated‑gradient
    update, always projected back onto the simplex.

4.  **Exact sub‑gradient polishing** – the true max‑convolution is
    minimised by an exponentiated‑gradient scheme with a 1/√t step‑size.
    The best vector found so far is continuously updated.

5.  **Upscaling** – once a good shape is found for a modest length
    (n≈1024), it is linearly interpolated to a larger length (2048 → 4096
    → 8192) and the whole refinement pipeline (2‑4) is repeated.  Upscaling
    lets the optimiser escape discretisation artefacts and approach the
    theoretical limit.

All stages stop as soon as the global deadline is reached.  The routine
always returns a feasible non‑negative list of floats; if nothing works a
fallback uniform vector is emitted.
"""

import time
import numpy as np
from typing import List, Tuple
from scipy.optimize import differential_evolution

# ----------------------------------------------------------------------
# Basic numeric utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full linear convolution using real FFT (zero‑padded to a power of two)."""
    m = a.size + b.size - 1
    n_fft = next_pow2(m)
    A = np.fft.rfft(a, n=n_fft)
    B = np.fft.rfft(b, n=n_fft)
    C = A * B
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


def compute_score(a: np.ndarray) -> float:
    """
    Fast surrogate of the official objective (scale‑invariant):
        2·n·max(a∗a) / (∑a)²
    The vector a is assumed non‑negative.
    """
    n = a.size
    conv = convolution_fft(a, a)
    max_val = float(conv.max())
    s = float(a.sum())
    if s < 1e-12:
        return float("inf")
    return 2.0 * n * max_val / (s * s)


# ----------------------------------------------------------------------
# Families of initial shapes (all returned *already normalised* to sum=1)
# ----------------------------------------------------------------------


def uniform_initial(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n)


def arcsine_initial(n: int) -> np.ndarray:
    i = np.arange(n, dtype=float)
    eps = 0.25
    vals = 1.0 / np.sqrt((i + 0.5) * (n - 0.5 - i) + eps)
    return vals / vals.sum()


def generate_powerlaw(n: int, p: float, eps: float) -> np.ndarray:
    i = np.arange(n, dtype=float)
    base = ((i + eps) * (n - 1 - i + eps)) ** (-p)
    total = base.sum()
    if total == 0.0:
        return uniform_initial(n)
    return base / total


def generate_beta_like(n: int, alpha: float, beta: float) -> np.ndarray:
    """Beta‑like shape a_i ∝ (i+½)^{α‑1} (n‑1‑i+½)^{β‑1}."""
    i = np.arange(n, dtype=float)
    # work in log‑space for stability when α or β < 1
    log_base = (alpha - 1) * np.log(i + 0.5) + (beta - 1) * np.log(n - 1 - i + 0.5)
    base = np.exp(log_base)
    total = base.sum()
    if total == 0.0 or np.any(np.isnan(base)):
        return uniform_initial(n)
    return base / total


def exponential_decay(n: int, rate: float) -> np.ndarray:
    """Monotonically decreasing shape a_i ∝ exp(‑rate·i)."""
    i = np.arange(n, dtype=float)
    vals = np.exp(-rate * i)
    return vals / vals.sum()


def random_dirichlet(n: int, concentration: float, rng: np.random.Generator) -> np.ndarray:
    """Sample a Dirichlet‑like vector (Gamma draws) and normalise."""
    if concentration <= 0:
        samp = rng.exponential(1.0, size=n) ** (1.0 / max(0.1, concentration))
    else:
        samp = rng.gamma(concentration, 1.0, size=n)
    samp += 1e-12
    return samp / samp.sum()


def upsample(a: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation from len(a) → new_n, then re‑normalise."""
    old_x = np.linspace(0.0, 1.0, a.size)
    new_x = np.linspace(0.0, 1.0, new_n)
    up = np.interp(new_x, old_x, a)
    s = up.sum()
    return up / s if s > 0 else uniform_initial(new_n)


# ----------------------------------------------------------------------
# Simulated annealing (light “mass‑swap” neighbourhood)
# ----------------------------------------------------------------------


def propose_neighbor(a: np.ndarray, rng: np.random.Generator, max_step: float = 0.02) -> np.ndarray:
    """Transfer a small amount of mass from one random bin to another."""
    n = a.size
    i, j = rng.integers(0, n, size=2)
    if i == j or a[i] == 0.0:
        return a
    delta = a[i] * rng.random() * max_step
    b = a.copy()
    b[i] -= delta
    b[j] += delta
    return b


def simulated_annealing(a0: np.ndarray,
                        deadline: float,
                        rng: np.random.Generator,
                        max_iter: int = 2000,
                        start_temp: float = 0.005,
                        cooling: float = 0.9995,
                        max_step: float = 0.02) -> Tuple[np.ndarray, float]:
    """Very light SA that preserves the simplex."""
    a = a0.copy()
    cur_sc = compute_score(a)
    best_a = a.copy()
    best_sc = cur_sc
    temp = start_temp

    for _ in range(max_iter):
        if time.time() >= deadline:
            break
        cand = propose_neighbor(a, rng, max_step)
        cand_sc = compute_score(cand)
        delta = cand_sc - cur_sc
        if delta < 0.0 or rng.random() < np.exp(-delta / max(1e-12, temp)):
            a = cand
            cur_sc = cand_sc
            if cur_sc < best_sc:
                best_sc = cur_sc
                best_a = a.copy()
        temp *= cooling
    return best_a, best_sc


# ----------------------------------------------------------------------
# Smooth‑max gradient descent (log‑sum‑exp surrogate)
# ----------------------------------------------------------------------


def softmax_gradient_descent(initial_a: np.ndarray,
                             deadline: float,
                             betas: List[float],
                             iter_per_beta: int = 400,
                             eta0: float = 0.5) -> Tuple[np.ndarray, float]:
    """
    Minimise a smooth approximation of the max‑convolution:
        Φ_β(a) = (1/β)·log Σ_k exp(β·(a∗a)_k).

    Uses an exponentiated‑gradient step followed by a simplex projection.
    """
    a = np.maximum(initial_a, 1e-30)
    a = a / a.sum()
    best_a = a.copy()
    best_sc = compute_score(a)

    for beta in betas:
        for it in range(1, iter_per_beta + 1):
            if time.time() >= deadline:
                return best_a, best_sc

            conv = convolution_fft(a, a)                     # length 2n‑1
            max_c = conv.max()
            # numerically stable soft‑max weights
            exp_vals = np.exp(beta * (conv - max_c))
            p = exp_vals / exp_vals.sum()                    # Σ p = 1

            # cross‑correlation p ⋆ a  (length n)
            N = next_pow2(p.size + a.size - 1)
            P = np.fft.fft(p, N)
            A = np.fft.fft(a, N)
            cross = np.fft.ifft(P * np.conj(A)).real[:a.size]
            cross = np.maximum(cross, 0.0)

            # gradient of the smooth objective (non‑negative)
            grad = 4.0 * a.size * cross

            # exponentiated‑gradient step (η decays like 1/√t)
            eta = eta0 / np.sqrt(it)
            a = a * np.exp(-eta * grad)

            # projection onto the simplex
            s = a.sum()
            if s == 0.0:
                a = uniform_initial(a.size)
            else:
                a = a / s

            cur_sc = compute_score(a)
            if cur_sc < best_sc:
                best_sc = cur_sc
                best_a = a.copy()
    return best_a, best_sc


# ----------------------------------------------------------------------
# Exact max‑convolution sub‑gradient (exponentiated updates)
# ----------------------------------------------------------------------


def _max_conv_gradient(a: np.ndarray, conv: np.ndarray) -> np.ndarray:
    """
    Sub‑gradient of max_k (a∗a)_k:
        For each active offset k, contribution = 2·a_{k‑j}.
    If several offsets share the maximal value they are averaged.
    """
    n = a.size
    max_val = conv.max()
    mask = np.isclose(conv, max_val, atol=1e-15)
    active = np.where(mask)[0]
    grad = np.zeros_like(a)

    for k in active:
        lo = max(0, k - (n - 1))
        hi = min(k, n - 1)
        idx = np.arange(lo, hi + 1, dtype=int)
        grad[idx] += 2.0 * a[k - idx]

    if active.size > 1:
        grad /= active.size
    return grad


def optimise_subgradient(initial_a: np.ndarray,
                         deadline: float,
                         eta0: float = 0.6,
                         max_iter: int = 200_000,
                         noise_every: int = 0,
                         noise_scale: float = 0.0) -> Tuple[np.ndarray, float]:
    """
    Exponentiated sub‑gradient descent directly on Φ(a)=max_k (a∗a)_k.
    Returns the best vector seen and its *fast* score.
    """
    a = np.maximum(initial_a, 1e-30)
    a = a / a.sum()
    best_a = a.copy()
    best_sc = compute_score(a)
    n = a.size

    for it in range(1, max_iter + 1):
        if time.time() >= deadline:
            break

        conv = convolution_fft(a, a)
        grad = _max_conv_gradient(a, conv)

        eta = eta0 / np.sqrt(it)
        a = a * np.exp(-eta * grad)

        if noise_every and (it % noise_every == 0):
            a += np.random.randn(n) * noise_scale
            a = np.maximum(a, 0.0)

        s = a.sum()
        a = a / s if s != 0.0 else uniform_initial(n)

        cur_sc = compute_score(a)
        if cur_sc < best_sc:
            best_sc = cur_sc
            best_a = a.copy()

    return best_a, best_sc

    # 3.  Upscale to larger lengths (2048 → 4096 → 8192) if time permits.
    # ------------------------------------------------------------------
    for new_n in [2048, 4096, 8192]:
        if time.time() >= deadline - 5.0:
            break
        if not best_seq:
            continue
        print(f"\n=== Upscaling to n = {new_n} ===")
        a_up = upsample(np.array(best_seq, dtype=float), new_n)

        # short SA on the upscaled vector
        it_sa = int(min(3000, (deadline - time.time()) * 300))
        sa_up, _ = simulated_annealing(a_up, deadline, rng,
                                       max_iter=it_sa,
                                       start_temp=0.01,
                                       cooling=0.9995,
                                       max_step=0.025)

        # smooth‑max GD with a tighter β schedule
        betas = [10, 30, 60, 120, 240, 480]
        remaining = max(0.0, deadline - time.time())
        it_per_beta = max(150, int(min(1000, remaining * 180) / len(betas)))
        sg_up, _ = softmax_gradient_descent(sa_up, deadline,
                                            betas=betas,
                                            iter_per_beta=it_per_beta,
                                            eta0=0.35)

        # final exact sub‑gradient polishing
        remaining = max(0.0, deadline - time.time())
        max_iter_sub = int(min(100_000, remaining * 250))
        final_up, _ = optimise_subgradient(sg_up, deadline,
                                           eta0=0.3,
                                           max_iter=max_iter_sub)

        val_up = evaluate_sequence(final_up.tolist())
        print(f"    Upscaled n={new_n} final score = {val_up:.6f}")
        if val_up < best_val:
            best_val = val_up
            best_seq = final_up.tolist()
            print(f"    *** Updated global best after upscaling ({new_n}) ***")

    # ------------------------------------------------------------------
    # 4.  Optional low‑dimensional fine‑tuning via DE on beta parameters.
    # ------------------------------------------------------------------
    if time.time() < deadline - 5.0:
        # optimise a beta‑shape for the current best length
        cur_n = len(best_seq) if best_seq else 1024
        print("\n=== Differential‑evolution fine‑tuning of a Beta shape ===")

        def de_obj(params: np.ndarray) -> float:
            a, b = params  # both in (0.01, 2.5)
            shape = generate_beta_like(cur_n, a, b)
            return compute_score(shape)

        bounds = [(0.01, 2.5), (0.01, 2.5)]
        # limit evaluations to stay within budget
        max_evals = 300
        try:
            result = differential_evolution(
                de_obj,
                bounds,
                rng=rng,
                maxiter=20,
                popsize=15,
                polish=False,
                updating='deferred',
                workers=1,
                disp=False,
                atol=0,
                mutation=(0.5, 1),
                recombination=0.7,
                init='latinhypercube',
                strategy='best1bin',
                callback=lambda *_: time.time() >= deadline,
                constraints=()
            )
            if result.success and result.nfev <= max_evals:
                beta_opt = generate_beta_like(cur_n, *result.x)
                val_beta = evaluate_sequence(beta_opt.tolist())
                print(f"    DE‑beta shape score = {val_beta:.6f}")
                if val_beta < best_val:
                    best_val = val_beta
                    best_seq = beta_opt.tolist()
                    print("*** Updated global best after DE beta fine‑tuning ***")
        except Exception as e:
            print(f"    DE optimisation failed: {e}")

    # ------------------------------------------------------------------
    # 5.  Guard – always return a feasible sequence.
    # ------------------------------------------------------------------
    if not best_seq:
        best_seq = uniform_initial(256).tolist()
        best_val = evaluate_sequence(best_seq)

    print(f"\nSearch finished. Best reported score = {best_val:.6f}")
    return best_seq
```

This implementation extends the original hybrid search by:

* Adding heavy‑tail beta families (α,β as low as 0.05) and exponential decays.
* Introducing a lightweight **differential‑evolution** step that directly optimises the two‑parameter beta shape.
* Using a richer β‑schedule and more inner iterations in the smooth‑max GD phase.
* Upscaling the best vector successively (1024 → 2048 → 4096 → 8192) and re‑optimising at each scale.
* Applying modest additive noise only when explicitly requested.

All stages respect the wall‑clock deadline, continuously track the best feasible vector, and finally return the best sequence found.
