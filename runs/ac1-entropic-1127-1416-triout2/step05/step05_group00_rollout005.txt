"""
Optimisation of a non‑negative step function for the given evaluation
metric  2·n·max(conv(a,a)) / (sum(a))².

The algorithm is a multi‑phase search:

1. **Random starts** – many Dirichlet / sparse‑subset vectors.
2. **Soft‑max gradient descent** – minimise a smooth approximation of the
   peak of the self‑convolution by exponentiated‑gradient updates.
   The temperature β is slowly increased, turning the smooth objective
   into the true max.
3. **Final hill‑climb** – simple stochastic moves (swap / mass transfer)
   that are accepted only when they improve the true score.

The whole search respects the 1000 s budget and never returns an empty
list.  All helper functions are top‑level and use only NumPy and the
provided ``evaluate_sequence`` function.
"""

from __future__ import annotations
import time
import numpy as np

# ----------------------------------------------------------------------
# Basic linear‑algebra utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """
    Full linear convolution a ⊛ b using a real FFT.
    Returned length = len(a) + len(b) - 1.
    """
    m = len(a) + len(b) - 1
    n_fft = next_pow2(m)
    A = np.fft.rfft(a, n=n_fft)
    B = np.fft.rfft(b, n=n_fft)
    C = A * B
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


def compute_score(a: np.ndarray) -> float:
    """
    Fast replica of ``evaluate_sequence`` when the vector is already
    normalised to sum 1.
    Returns 2·n·max(a⊛a).
    """
    n = a.shape[0]
    conv = convolution_fft(a, a)
    return float(2.0 * n * np.max(conv))


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the probability simplex
    (non‑negative entries that sum to 1).
    """
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / v.shape[0])

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, v.shape[0] + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.where(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


# ----------------------------------------------------------------------
# Random initialisation utilities
# ----------------------------------------------------------------------


def random_sparse_vector(n: int, density: float) -> np.ndarray:
    """
    Produce a random vector of length n with roughly ``density`` of its
    entries non‑zero.  The non‑zero values are drawn from a Gamma(1,1)
    distribution and the whole vector is normalised to sum 1.
    """
    k = max(2, int(round(density * n)))
    idx = np.random.choice(n, size=k, replace=False)
    vals = np.random.gamma(shape=1.0, scale=1.0, size=k)
    a = np.zeros(n, dtype=float)
    a[idx] = vals
    a_sum = a.sum()
    if a_sum == 0:
        a[:] = 1.0 / n
    else:
        a /= a_sum
    return a


# ----------------------------------------------------------------------
# Soft‑max gradient descent (exponentiated‑gradient)
# ----------------------------------------------------------------------


def _softmax_weights(conv: np.ndarray, beta: float) -> np.ndarray:
    """
    Compute the soft‑max weights w_i = exp(beta·conv_i) / Σ exp(...).
    The subtraction of the maximum ensures numerical stability.
    """
    maxc = np.max(conv)
    exps = np.exp(beta * (conv - maxc))
    w = exps / np.sum(exps)
    return w


def _gradient_of_softmax_maxconv(a: np.ndarray, beta: float) -> tuple[np.ndarray, np.ndarray]:
    """
    For a given vector a (sum = 1) and temperature beta, return
    the gradient of the soft‑max approximation of max(conv(a,a))
    (a vector of length n) and the raw convolution (for diagnostics).
    """
    # full self‑convolution
    conv = convolution_fft(a, a)                     # length 2n‑1
    # soft‑max weights over all shifts
    w = _softmax_weights(conv, beta)                # same length as conv
    # gradient: 2 * (w ⋆ a_rev) evaluated at the centre
    a_rev = a[::-1]
    corr_full = convolution_fft(w, a_rev)           # length 3n‑2
    n = a.shape[0]
    offset = n - 1                                   # centre index
    g = 2.0 * corr_full[offset:offset + n]          # shape (n,)
    return g, conv


def optimise_with_softmax(
    a0: np.ndarray,
    deadline: float,
    max_iter: int = 200_000,
    eta0: float = 0.5,
    beta0: float = 0.5,
) -> tuple[np.ndarray, float]:
    """
    Exponentiated‑gradient descent on a smooth approximation of the L∞
    norm of the convolution.  The temperature β starts small and is
    slowly increased so that the objective morphs into the true max.
    """
    a = a0.astype(float).copy()
    n = a.shape[0]

    best = a.copy()
    best_score = compute_score(a)

    t = 0
    beta = beta0

    # main loop – stops when time is exhausted or the iteration budget
    while time.time() < deadline and t < max_iter:
        t += 1

        # gradient of the smoothed objective
        g, conv = _gradient_of_softmax_maxconv(a, beta)

        # step‑size (decreasing like 1/√t)
        eta = eta0 / np.sqrt(t)

        # exponentiated‑gradient update (keeps all entries non‑negative)
        a *= np.exp(-eta * g)

        # renormalise onto the probability simplex (simple ℓ₁‑normalisation)
        s = a.sum()
        if s <= 0:
            a[:] = 1.0 / n
        else:
            a /= s

        # occasional evaluation and temperature adaptation
        if t % 200 == 0:
            cur_score = compute_score(a)
            if cur_score < best_score:
                best_score = cur_score
                best = a.copy()
                # when we improve we become more aggressive – raise β
                beta = min(beta * 1.02, 30.0)
            else:
                # otherwise cool a little – helps to avoid premature focus
                beta = max(beta * 0.99, 0.5)

        # small random jitter prevents getting stuck in symmetric plateaus
        if t % 1000 == 0:
            jitter = np.random.randn(n) * (0.001 / np.sqrt(t))
            a += jitter
            a = np.maximum(a, 0.0)
            a_sum = a.sum()
            if a_sum > 0:
                a /= a_sum
            else:
                a[:] = 1.0 / n

    return best, best_score


# ----------------------------------------------------------------------
# Final hill‑climb polishing (swap / mass‑transfer moves)
# ----------------------------------------------------------------------


def hill_climb(
    seq: list[float],
    time_deadline: float,
    max_steps: int = 400_000,
) -> tuple[list[float], float]:
    """
    Stochastic hill‑climb that either swaps two entries or transfers a
    fraction of mass from one position to another.  All candidates are
    projected back onto the simplex.
    """
    a = np.array(seq, dtype=float)
    n = a.shape[0]

    best_a = a.copy()
    best_score = compute_score(best_a)

    step_scale = 0.12  # typical fraction of mass moved in a transfer

    while max_steps > 0 and time.time() < time_deadline:
        move_type = np.random.rand()
        cand = best_a.copy()

        if move_type < 0.5:
            # simple swap
            i, j = np.random.randint(n, size=2)
            cand[i], cand[j] = cand[j], cand[i]
        else:
            # move a fraction from i to j
            i, j = np.random.randint(n, size=2)
            if i != j:
                delta = step_scale * cand[i]
                cand[i] -= delta
                cand[j] += delta
                cand[i] = max(cand[i], 0.0)

        # renormalise onto the simplex
        s = cand.sum()
        if s > 0:
            cand /= s
        else:
            cand[:] = 1.0 / n

        cur_score = compute_score(cand)

        if cur_score < best_score:
            best_score = cur_score
            best_a = cand
            step_scale = min(step_scale * 1.07, 0.5)
        else:
            step_scale *= 0.96

        max_steps -= 1

    return best_a.tolist(), best_score
