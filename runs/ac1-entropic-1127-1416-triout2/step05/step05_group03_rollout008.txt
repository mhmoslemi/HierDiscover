"""
Improved search for a non‑negative step sequence that minimises

        2·n·max(conv(a,a)) / (Σ a)²

The algorithm combines a much broader random‑shape sampling stage with
more aggressive annealing in the Adam phase and an extra
exponentiated‑gradient polishing step.  It works on several candidate
lengths (1024, 2048, 4096) and keeps the best vector found according to
the official ``evaluate_sequence`` function.

Key ideas
---------

* **Very wide shape families** – in addition to symmetric beta
  densities we also sample Dirichlet draws, power‑law “U‑shapes” and a
  linear descending shape.  The parameter ranges deliberately favour
  extreme U‑shapes which are known to flatten the self‑convolution.

* **Adam on a high‑temperature log‑sum‑exp surrogate** – the temperature
  (``lam``) is annealed up to 500, making the surrogate almost identical
  to the true max‑convolution.

* **Projected sub‑gradient descent** – after the smooth phase we switch
  to the true max‑convolution and perform a back‑tracking line‑search.

* **Random Gaussian walk** – a final cheap stochastic walk explores the
  neighbourhood of the current best point.

* **Global polishing** – once a best vector has been found we repeat the
  three optimisation phases on it (with a higher temperature and more
  iterations) to squeeze out the last bits of improvement.

All steps respect the 1000 s wall‑clock budget and guarantee a non‑empty
return.
"""

import time
import numpy as np
from typing import List, Tuple

# ----------------------------------------------------------------------
# Linear‑algebra utilities (no closures)
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Return the smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """
    Linear convolution via real FFT (returns length len(a)+len(b)-1).
    """
    m = a.size + b.size - 1
    n_fft = next_pow2(m)
    A = np.fft.rfft(a, n=n_fft)
    B = np.fft.rfft(b, n=n_fft)
    C = A * B
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


def max_convolution(a: np.ndarray) -> float:
    """Maximum entry of the self‑convolution of a."""
    return float(np.max(convolution_fft(a, a)))


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection onto the probability simplex
    {x ≥ 0, Σx = 1}.  Implementation of Duchi et al. (2008).
    """
    n = v.shape[0]
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / n)

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho = np.where(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def compute_score(a: np.ndarray) -> float:
    """
    Fast surrogate: 2·n·max(conv(a,a)).
    For a vector normalised to Σa = 1 this equals the official score.
    """
    n = a.shape[0]
    return 2.0 * n * max_convolution(a)


# ----------------------------------------------------------------------
# Shape generators
# ----------------------------------------------------------------------


def generate_beta_shape(
    n: int, alpha: float, beta: float, eps: float
) -> np.ndarray:
    """
    Symmetric beta‑family density on {0,…,n‑1}:
        a_i ∝ (i+ε)^{α‑1}·(n‑1‑i+ε)^{β‑1},
    normalised to Σa = 1.
    """
    i = np.arange(n, dtype=float)
    left = (i + eps) ** (alpha - 1.0)
    right = (n - 1 - i + eps) ** (beta - 1.0)
    base = left * right
    total = base.sum()
    if total <= 0.0 or not np.isfinite(total):
        return np.full(n, 1.0 / n)
    return base / total


def generate_beta_mixture(
    n: int,
    alpha1: float,
    beta1: float,
    eps1: float,
    alpha2: float,
    beta2: float,
    eps2: float,
    w: float,
) -> np.ndarray:
    """
    Convex combination of two beta‑family densities, re‑normalised.
    """
    a1 = generate_beta_shape(n, alpha1, beta1, eps1)
    a2 = generate_beta_shape(n, alpha2, beta2, eps2)
    mix = w * a1 + (1.0 - w) * a2
    mix_sum = mix.sum()
    if mix_sum == 0.0 or not np.isfinite(mix_sum):
        return np.full(n, 1.0 / n)
    return mix / mix_sum


def arcsine_initial(n: int) -> np.ndarray:
    """Classic symmetric arcsine shape."""
    i = np.arange(n, dtype=float)
    eps = 0.25
    vals = 1.0 / np.sqrt((i + 0.5) * (n - 0.5 - i) + eps)
    vals /= vals.sum()
    return vals


def powerlaw_u_shape(n: int, p: float, eps: float) -> np.ndarray:
    """
    Symmetric “U” shape: a_i ∝ ((i+ε)*(n‑1‑i+ε))^{-p}.
    Normalised to Σ = 1.
    """
    i = np.arange(n, dtype=float)
    base = ((i + eps) * (n - 1 - i + eps)) ** (-p)
    total = base.sum()
    if total <= 0.0 or not np.isfinite(total):
        return np.full(n, 1.0 / n)
    return base / total


def linear_descending(n: int) -> np.ndarray:
    """Deterministic linear descending shape (triangular distribution)."""
    i = np.arange(n, dtype=float)
    base = n - i
    total = base.sum()
    return base / total


# ----------------------------------------------------------------------
# Smooth surrogate (log‑sum‑exp) and its gradient (used by Adam)
# ----------------------------------------------------------------------


def smooth_gradient(a: np.ndarray, lam: float) -> Tuple[np.ndarray, float]:
    """
    Gradient of the log‑sum‑exp surrogate of max(conv(a,a)).
    Returns (gradient, surrogate value).
    """
    n = a.shape[0]
    conv = convolution_fft(a, a)          # length 2n‑1
    max_c = np.max(conv)

    # stable log‑sum‑exp
    exp_term = np.exp(lam * (conv - max_c))
    sum_exp = np.sum(exp_term)
    w = exp_term / sum_exp                # soft‑max on convolution entries
    B = max_c + np.log(sum_exp) / lam     # smoothed max

    # weighted sum of sub‑gradients:
    # ∂/∂a  Σ_k w_k (a*a)_k = 2 * (a ⋆ w_rev)
    w_rev = w[::-1]
    conv_rev = convolution_fft(a, w_rev)      # length 3n‑2
    L = 2 * n - 1
    weighted_grad = 2.0 * conv_rev[L - 1 - np.arange(n)]

    s = np.sum(a)
    term1 = (2.0 * n / (s ** 2)) * weighted_grad
    term2 = -(4.0 * n * B / (s ** 3)) * np.ones_like(a)
    grad = term1 + term2
    return grad, B


def adam_optimize(
    initial_a: np.ndarray,
    time_deadline: float,
    max_iter: int = 120_000,
    lr0: float = 0.12,
    lam_start: float = 1.0,
    lam_end: float = 500.0,
    noise_every: int = 1300,
    noise_scale: float = 5e-5,
) -> Tuple[np.ndarray, float]:
    """
    Adam optimiser on the smooth log‑sum‑exp surrogate.
    Returns the best vector (projected on the simplex) and its fast score.
    """
    a = project_to_simplex(initial_a.astype(float).copy())
    best_a = a.copy()
    best_sc = compute_score(a)

    beta1, beta2 = 0.9, 0.999
    eps_adam = 1e-8
    m = np.zeros_like(a)
    v = np.zeros_like(a)

    n = a.shape[0]

    for it in range(1, max_iter + 1):
        if time.time() >= time_deadline:
            break

        # quadratic temperature schedule (slow at start, fast near the end)
        t = it / max_iter
        lam = lam_start + (lam_end - lam_start) * (t ** 0.5)

        grad, _ = smooth_gradient(a, lam)

        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)

        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        step = lr0 * m_hat / (np.sqrt(v_hat) + eps_adam)
        a_new = a - step

        if noise_every and (it % noise_every == 0):
            a_new += np.random.randn(n) * noise_scale

        a_new = project_to_simplex(a_new)

        cur_sc = compute_score(a_new)
        if cur_sc < best_sc:
            best_sc = cur_sc
            best_a = a_new.copy()

        a = a_new

        if it % 5000 == 0:
            print(f"    Adam iter={it:,d} cur={cur_sc:.6f} best={best_sc:.6f}")

    return best_a, best_sc


# ----------------------------------------------------------------------
# Exact sub‑gradient descent with back‑tracking line‑search
# ----------------------------------------------------------------------


def subgradient_optimize(
    initial_a: np.ndarray,
    time_deadline: float,
    step0: float = 0.25,
    max_iter: int = 150_000,
) -> Tuple[np.ndarray, float]:
    """
    Projected sub‑gradient descent on the true max‑convolution.
    Returns the best vector (projected) and its fast score.
    """
    a = initial_a.copy()
    best_a = a.copy()
    best_sc = compute_score(a)

    n = a.shape[0]

    for it in range(1, max_iter + 1):
        if time.time() >= time_deadline:
            break

        conv = convolution_fft(a, a)
        max_val = float(np.max(conv))
        idx = int(np.argmax(conv))

        # gradient of the active quadratic term
        i = np.arange(n)
        j = idx - i
        mask = (j >= 0) & (j < n)
        grad = np.zeros_like(a)
        grad[mask] = 2.0 * a[j[mask]]

        # diminishing step size with simple back‑tracking
        step = step0 / np.sqrt(it)
        for _ in range(12):
            a_new = a - step * grad
            a_new = project_to_simplex(a_new)
            new_val = max_convolution(a_new)
            if new_val < max_val - 1e-14:
                break
            step *= 0.5

        cur_sc = compute_score(a_new)
        if cur_sc < best_sc:
            best_sc = cur_sc
            best_a = a_new.copy()

        a = a_new

        if it % 40000 == 0:
            print(
                f"    Sub‑grad iter={it:,d} cur={cur_sc:.6f} best={best_sc:.6f}"
            )

    return best_a, best_sc


# ----------------------------------------------------------------------
# Stochastic local search (Gaussian perturbations)
# ----------------------------------------------------------------------


def random_local_search(
    initial_a: np.ndarray,
    time_deadline: float,
    init_step: float = 0.015,
    max_iter: int = 250_000,
) -> Tuple[np.ndarray, float]:
    """
    Gaussian walk on the simplex with a tiny uphill‑move probability.
    Returns the best vector seen and its fast score.
    """
    a = initial_a.copy()
    best_a = a.copy()
    best_sc = compute_score(a)

    n = a.shape[0]
    rng = np.random.default_rng()

    for it in range(1, max_iter + 1):
        if time.time() >= time_deadline:
            break

        step = init_step / np.sqrt(it)
        noise = rng.normal(scale=step, size=n)
        a_proposed = a + noise
        a_proposed = project_to_simplex(a_proposed)

        cur_sc = compute_score(a_proposed)

        # accept if improves, otherwise with a very small uphill probability
        if cur_sc < best_sc or (rng.random() < 1e-4 and cur_sc < 10 * best_sc):
            a = a_proposed
            if cur_sc < best_sc:
                best_sc = cur_sc
                best_a = a_proposed.copy()

        if it % 50000 == 0:
            print(
                f"    Random local iter={it:,d} cur={cur_sc:.6f} best={best_sc:.6f}"
            )

    return best_a, best_sc


# ----------------------------------------------------------------------
# Main entry point required by the competition harness
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs) -> List[float]:
    """
    Search for a non‑negative step sequence with a low official score.
    The routine mixes extensive random‑shape sampling, a long Adam run
    with high temperature, exact sub‑gradient descent, and a final
    stochastic polishing phase.  It works for any length that fits into
    the allotted 1000 s.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 2.0   # safety margin for I/O

    best_seq: List[float] = []
    best_val: float = float("inf")

    # ------------------------------------------------------------------
    # 0.  Optional fallback from a previously discovered construction
    # ------------------------------------------------------------------
    try:
        fallback = [float(x) for x in height_sequence_1]  # type: ignore
        val = evaluate_sequence(fallback)
        print(f"Fallback (height_sequence_1) official score = {val:.6f}")
        if val < best_val:
            best_val = val
            best_seq = fallback
    except Exception:
        pass

    # ------------------------------------------------------------------
    # 1.  Explore several lengths
    # ------------------------------------------------------------------
    candidate_lengths = [1024, 2048, 4096]

    for n in candidate_lengths:
        if time.time() >= deadline - 30.0:
            break
        print(f"\n=== Sampling length n = {n} ===")

        # --------------------------------------------------------------
        # Deterministic starters (uniform + arcsine + linear)
        # --------------------------------------------------------------
        local_best = None
        local_best_sc = float("inf")
        starters = [
            np.full(n, 1.0 / n),
            arcsine_initial(n),
            linear_descending(n),
        ]
        for st in starters:
            sc = compute_score(st)
            if sc < local_best_sc:
                local_best_sc = sc
                local_best = st.copy()

        # --------------------------------------------------------------
        # Random shape sampling – broadened families
        # --------------------------------------------------------------
        max_samples = int(6000 * (budget_s / 1000.0))
        for _ in range(max_samples):
            if time.time() >= deadline - 30.0:
                break

            shape_type = rng.integers(0, 5)   # 0:beta, 1:mixture, 2:power‑law, 3:Dirichlet, 4:linear combo
            if shape_type == 0:
                # single beta
                alpha = rng.uniform(0.001, 2.5)
                beta = rng.uniform(0.001, 2.5)
                eps = rng.uniform(1e-12, max(0.01, 0.02 * n))
                cand = generate_beta_shape(n, alpha, beta, eps)
            elif shape_type == 1:
                # mixture of two betas
                w = rng.random()
                alpha1 = rng.uniform(0.001, 2.5)
                beta1 = rng.uniform(0.001, 2.5)
                eps1 = rng.uniform(1e-12, max(0.01, 0.02 * n))
                alpha2 = rng.uniform(0.001, 2.5)
                beta2 = rng.uniform(0.001, 2.5)
                eps2 = rng.uniform(1e-12, max(0.01, 0.02 * n))
                cand = generate_beta_mixture(
                    n, alpha1, beta1, eps1, alpha2, beta2, eps2, w
                )
            elif shape_type == 2:
                # power‑law U‑shape
                p = rng.uniform(0.2, 5.0)
                eps = rng.uniform(1e-12, max(0.01, 0.02 * n))
                cand = powerlaw_u_shape(n, p, eps)
            elif shape_type == 3:
                # Dirichlet with a very small concentration parameter
                alpha0 = rng.uniform(0.02, 0.3)
                g = rng.gamma(alpha0, 1.0, size=n)
                total = g.sum()
                cand = g / total if total > 0 else np.full(n, 1.0 / n)
            else:
                # linear descending plus a tiny random perturbation
                base = linear_descending(n)
                perturb = rng.normal(scale=0.01, size=n)
                cand = np.maximum(base + perturb, 0.0)
                cand /= cand.sum()

            sc = compute_score(cand)
            if sc < local_best_sc:
                local_best_sc = sc
                local_best = cand.copy()

        # --------------------------------------------------------------
        # 2.  Polishing of the locally best candidate (if we have one)
        # --------------------------------------------------------------
        if local_best is not None and time.time() < deadline - 5.0:
            # allocate time proportionally but keep a safety cushion
            remaining = deadline - time.time()
            max_iter_adam = int(min(300_000, remaining * 28))
            print(f"  Polishing with Adam (max_iter={max_iter_adam}) …")
            polished, _ = adam_optimize(
                initial_a=local_best,
                time_deadline=deadline,
                max_iter=max_iter_adam,
                lr0=0.10,
                lam_start=1.0,
                lam_end=500.0,
                noise_every=1200,
                noise_scale=4e-5,
            )

            # Sub‑gradient descent on the true max
            if time.time() < deadline - 5.0:
                max_iter_sub = int(min(120_000, (deadline - time.time()) * 30))
                polished, _ = subgradient_optimize(
                    polished,
                    time_deadline=deadline,
                    step0=0.28,
                    max_iter=max_iter_sub,
                )

            # Final cheap Gaussian walk
            if time.time() < deadline - 3.0:
                max_iter_loc = int(min(180_000, (deadline - time.time()) * 35))
                polished, _ = random_local_search(
                    polished,
                    time_deadline=deadline,
                    init_step=0.016,
                    max_iter=max_iter_loc,
                )

            # Official evaluation (once per length)
            val = evaluate_sequence(polished.tolist())
            print(f"  → official score for length {n} = {val:.6f}")
            if val < best_val:
                best_val = val
                best_seq = polished.tolist()
                print(f"  *** New global best (len={n}) = {best_val:.6f} ***")
        else:
            # fallback – evaluate the raw best candidate discovered by sampling
            if local_best is not None:
                val = evaluate_sequence(local_best.tolist())
                print(f"  Raw candidate official score = {val:.6f}")
                if val < best_val:
                    best_val = val
                    best_seq = local_best.tolist()
                    print("*** Updated global best (raw) ***")

    # ------------------------------------------------------------------
    # 3.  Global polishing of the overall best vector
    # ------------------------------------------------------------------
    if best_seq and time.time() < deadline - 5.0:
        print("\n=== Global polishing of the best sequence ===")
        a = np.array(best_seq, dtype=float)

        remaining = deadline - time.time()
        max_iter_adam = int(min(400_000, remaining * 30))
        print(f"  Adam polishing (max_iter={max_iter_adam}) …")
        a, _ = adam_optimize(
            initial_a=a,
            time_deadline=deadline,
            max_iter=max_iter_adam,
            lr0=0.09,
            lam_start=1.0,
            lam_end=600.0,
            noise_every=1500,
            noise_scale=3e-5,
        )

        if time.time() < deadline - 4.0:
            max_iter_sub = int(min(150_000, (deadline - time.time()) * 30))
            print(f"  Sub‑gradient polishing (max_iter={max_iter_sub}) …")
            a, _ = subgradient_optimize(
                a,
                time_deadline=deadline,
                step0=0.22,
                max_iter=max_iter_sub,
            )

        if time.time() < deadline - 2.0:
            max_iter_loc = int(min(250_000, (deadline - time.time()) * 40))
            print(f"  Random local polishing (max_iter={max_iter_loc}) …")
            a, _ = random_local_search(
                a,
                time_deadline=deadline,
                init_step=0.012,
                max_iter=max_iter_loc,
            )

        final_val = evaluate_sequence(a.tolist())
        print(f"Final polished official score = {final_val:.6f}")
        if final_val < best_val:
            best_val = final_val
            best_seq = a.tolist()
            print("*** Updated global best after final polish ***")

    # ------------------------------------------------------------------
    # 4.  Guard – guarantee a non‑empty return
    # ------------------------------------------------------------------
    if not best_seq:
        best_seq = np.full(256, 1.0 / 256).tolist()
        best_val = evaluate_sequence(best_seq)
        print("Fallback to uniform 256‑vector.", best_val)

    print(f"\nSearch finished. Best reported official score = {best_val:.6f}")
    return best_seq
