"""
A fast multi‑stage optimiser for the “step‑function” challenge.

Stage 1 – coarsely explore several low‑dimensional parametric families
(beta, power‑law, exponential, and mixtures with uniform) for several
different lengths n.  The optimisation is performed with a few
Nelder‑Mead runs on the log‑transformed parameters – the objective is
the exact score 2·n·max(conv(a,a)) after normalising Σa=1.

Stage 2 – take the best seed found and polish it with three
gradient‑based refinements that were already proven effective:
    * Adam on a soft‑max surrogate of the max‑convolution,
    * exact sub‑gradient descent on the true objective,
    * a short simulated‑annealing walk.

All heavy lifting stays within the 1000 s time budget, uses only NumPy,
SciPy (if available) and the already‑imported ``evaluate_sequence``.
The returned list is ready for the official testing harness.
"""

import time
import numpy as np
from typing import List, Tuple

# ----------------------------------------------------------------------
# Low‑level utilities (FFT‑based convolution, projection)
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """
    Linear convolution via real FFT.
    Length = len(a)+len(b)-1.
    """
    m = len(a) + len(b) - 1
    n_fft = next_pow2(m)
    A = np.fft.rfft(a, n=n_fft)
    B = np.fft.rfft(b, n=n_fft)
    C = A * B
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


def compute_score(a: np.ndarray) -> float:
    """
    Exact objective for a vector already normalised to Σa = 1:
        2·n·max(conv(a,a))
    """
    n = a.size
    conv = convolution_fft(a, a)
    return 2.0 * n * float(np.max(conv))


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection onto the probability simplex {x ≥ 0, Σx = 1}.
    (Duchi et al., 2008)
    """
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / v.size)

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho = np.where(u - cssv / np.arange(1, v.size + 1) > 0)[0]
    if len(rho) == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


# ----------------------------------------------------------------------
# Parametric families – all return a vector that already sums to 1
# ----------------------------------------------------------------------


def generate_beta_shape(n: int, alpha: float, beta: float, eps: float) -> np.ndarray:
    """ (i+ε)^{α‑1}·(n‑1‑i+ε)^{β‑1} , normalised to Σ=1. """
    i = np.arange(n, dtype=float)
    left = (i + eps) ** (alpha - 1.0)
    right = (n - 1 - i + eps) ** (beta - 1.0)
    base = left * right
    total = base.sum()
    if not np.isfinite(total) or total <= 0.0:
        return np.full(n, 1.0 / n)
    return base / total


def generate_power_shape(n: int, p: float, eps: float) -> np.ndarray:
    """ a_i ∝ ((i+½)*(n‑½‑i)+ε)^{‑p} , p>0 """
    i = np.arange(n, dtype=float)
    denom = ((i + 0.5) * (n - 0.5 - i) + eps) ** p
    vals = 1.0 / denom
    total = vals.sum()
    if not np.isfinite(total) or total <= 0.0:
        return np.full(n, 1.0 / n)
    return vals / total


def generate_exponential_shape(n: int, lam: float) -> np.ndarray:
    """ a_i ∝ exp(-lam·i) , normalised. """
    i = np.arange(n, dtype=float)
    vals = np.exp(-lam * i)
    total = vals.sum()
    if total <= 0.0:
        return np.full(n, 1.0 / n)
    return vals / total


def generate_triangular_shape(n: int, peak: int) -> np.ndarray:
    """ Symmetric triangular bump, peak at index ``peak``. """
    left = np.arange(peak + 1, dtype=float)
    right = np.arange(n - peak - 1, dtype=float)[::-1]
    seq = np.concatenate([left, right])
    seq /= seq.sum()
    return seq


def uniform_shape(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n)


# ----------------------------------------------------------------------
# Gradient‑based refiners (same as in the previous state‑of‑the‑art)
# ----------------------------------------------------------------------


def smooth_gradient(a: np.ndarray, lam: float) -> np.ndarray:
    """
    Gradient of the soft‑max approximation of max(conv(a,a)).
    Returned gradient is for 2·n·softmax(max).
    """
    n = a.size
    conv = convolution_fft(a, a)          # length 2·n‑1
    max_c = np.max(conv)

    # numerically stable soft‑max weights
    exp_term = np.exp(lam * (conv - max_c))
    w = exp_term / np.sum(exp_term)

    # weighted sub‑gradient = 2·shift(a,k) weighted by w
    w_rev = w[::-1]
    conv_rev = convolution_fft(a, w_rev)
    L = 2 * n - 1
    weighted_grad = 2.0 * conv_rev[L - 1 - np.arange(n)]

    return 2.0 * n * weighted_grad


def adam_optimize_smooth(initial_a: np.ndarray,
                         deadline: float,
                         max_iter: int = 100_000,
                         lr0: float = 0.15,
                         lam_start: float = 2.0,
                         lam_end: float = 80.0,
                         noise_every: int = 2000,
                         noise_scale: float = 5e-5) -> Tuple[np.ndarray, float]:
    """Adam optimiser on the smooth surrogate."""
    a = project_to_simplex(initial_a.astype(float).copy())
    best_a = a.copy()
    best_sc = compute_score(a)

    beta1, beta2 = 0.9, 0.999
    eps = 1e-8
    m = np.zeros_like(a)
    v = np.zeros_like(a)

    for it in range(1, max_iter + 1):
        if time.time() >= deadline:
            break

        lam = lam_start + (lam_end - lam_start) * min(1.0, it / max_iter)

        grad = smooth_gradient(a, lam)

        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)

        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        step = lr0 * m_hat / (np.sqrt(v_hat) + eps)
        a_new = a - step

        if noise_every and (it % noise_every == 0):
            a_new += np.random.randn(a.size) * noise_scale

        a_new = project_to_simplex(a_new)

        cur_sc = compute_score(a_new)
        if cur_sc < best_sc:
            best_sc = cur_sc
            best_a = a_new.copy()

        a = a_new

        # occasional progress report – cheap enough not to affect timing
        if it % 5000 == 0:
            print(f"    AdamSmooth it={it:,d} cur={cur_sc:.6f} best={best_sc:.6f}")

    return best_a, best_sc


def subgradient_descent_exact(initial_a: np.ndarray,
                              deadline: float,
                              max_iter: int = 80_000,
                              step0: float = 0.05) -> np.ndarray:
    """Sub‑gradient descent on the exact objective."""
    a = project_to_simplex(initial_a.copy())
    best_a = a.copy()
    best_sc = compute_score(a)
    n = a.size
    L = 2 * n - 1

    for it in range(1, max_iter + 1):
        if time.time() >= deadline:
            break

        conv = convolution_fft(a, a)
        max_val = np.max(conv)

        # mask of (almost) maximising entries
        tol = max_val * 1e-12 + 1e-14
        mask = (np.abs(conv - max_val) <= tol).astype(float)

        mask_rev = mask[::-1]
        conv_rev = convolution_fft(a, mask_rev)
        weighted_grad = 2.0 * conv_rev[L - 1 - np.arange(n)]

        eta = step0 / np.sqrt(it)
        a = a - eta * weighted_grad
        a = project_to_simplex(a)

        cur_sc = compute_score(a)
        if cur_sc < best_sc:
            best_sc = cur_sc
            best_a = a.copy()

        if it % 5000 == 0:
            print(f"    SubGrad it={it:,d} cur={cur_sc:.6f} best={best_sc:.6f}")

    return best_a


def simulated_annealing(a: np.ndarray,
                        rng: np.random.Generator,
                        deadline: float,
                        max_iters: int = 120_000,
                        init_temp: float = 0.02,
                        decay: float = 0.9999) -> np.ndarray:
    """Simple Metropolis‑style local search on the exact surrogate."""
    cur = a.copy()
    cur_sc = compute_score(cur)
    best = cur.copy()
    best_sc = cur_sc
    n = cur.size
    temp = init_temp
    it = 0

    while it < max_iters and time.time() < deadline:
        it += 1
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue

        # move up to 20 % of the mass at i
        delta = cur[i] * 0.2 * rng.random()
        if delta <= 1e-15 or cur[i] - delta < 0:
            continue

        cand = cur.copy()
        cand[i] -= delta
        cand[j] += delta

        cand_sc = compute_score(cand)

        if cand_sc < cur_sc or rng.random() < np.exp(-(cand_sc - cur_sc) / max(temp, 1e-12)):
            cur = cand
            cur_sc = cand_sc
            if cur_sc < best_sc:
                best = cur.copy()
                best_sc = cur_sc

        temp = max(temp * decay, 1e-8)

        if it % 5000 == 0:
            print(f"    SA it={it:,d} cur={cur_sc:.6f} best={best_sc:.6f}")

    return best

def _opt_beta(n: int, deadline: float, rng: np.random.Generator) -> Tuple[np.ndarray, float]:
    """Optimise the beta family (α,β,ε)."""
    try:
        import scipy.optimize as opt
    except Exception:          # SciPy unavailable – skip
        return None, float("inf")

    best_vec = None
    best_sc = float("inf")
    attempts = 6
    for _ in range(attempts):
        if time.time() >= deadline:
            break
        # random start (log‑space)
        x0 = np.log([rng.uniform(0.02, 5.0),
                     rng.uniform(0.02, 5.0),
                     rng.uniform(1e-6, 2.0)])

        def fun(p):
            alpha, beta, eps = np.exp(p)
            vec = generate_beta_shape(n, alpha, beta, eps)
            return compute_score(vec)

        res = opt.minimize(fun, x0, method="Nelder-Mead",
                           options={"maxiter": 300, "maxfev": 500, "disp": False})
        vec = generate_beta_shape(n,
                                  np.exp(res.x[0]),
                                  np.exp(res.x[1]),
                                  np.exp(res.x[2]))
        sc = compute_score(vec)
        if sc < best_sc:
            best_sc = sc
            best_vec = vec
    return best_vec, best_sc

def _opt_beta_uniform_mix(n: int, deadline: float, rng: np.random.Generator) -> Tuple[np.ndarray, float]:
    """Beta + uniform mixture (α,β,ε,w)."""
    try:
        import scipy.optimize as opt
    except Exception:
        return None, float("inf")

    uniform = uniform_shape(n)
    best_vec = None
    best_sc = float("inf")
    attempts = 5
    for _ in range(attempts):
        if time.time() >= deadline:
            break
        # log‑param for α,β,ε and logit for w ∈ (0,1)
        x0 = np.concatenate([
            np.log([rng.uniform(0.02, 5.0),
                    rng.uniform(0.02, 5.0),
                    rng.uniform(1e-6, 2.0)]),
            [np.log(rng.random() + 1e-12) - np.log(1 - rng.random() + 1e-12)]
        ])

        def fun(p):
            alpha, beta, eps = np.exp(p[:3])
            w = 1.0 / (1.0 + np.exp(-p[3]))
            beta_vec = generate_beta_shape(n, alpha, beta, eps)
            vec = w * beta_vec + (1.0 - w) * uniform
            return compute_score(vec)

        res = opt.minimize(fun, x0, method="Nelder-Mead",
                           options={"maxiter": 250, "maxfev": 400, "disp": False})
        w_opt = 1.0 / (1.0 + np.exp(-res.x[3]))
        beta_vec = generate_beta_shape(n, np.exp(res.x[0]), np.exp(res.x[1]), np.exp(res.x[2]))
        vec = w_opt * beta_vec + (1.0 - w_opt) * uniform
        sc = compute_score(vec)
        if sc < best_sc:
            best_sc = sc
            best_vec = vec
    return best_vec, best_sc

def propose_candidate(seed: int = 42,
                     budget_s: int = 1000,
                     **kwargs) -> List[float]:
    """
    Search for a non‑negative step sequence with a low ``evaluate_sequence``
    value.  The routine explores several parametric families for a handful
    of lengths, refines the best seed with Adam → sub‑gradient → simulated
    annealing, and finally returns the best vector found.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 1.0          # small safety margin

    best_seq: List[float] = []
    best_val: float = float("inf")
    best_n: int = 0

    # ------------------------------------------------------------------
    # 0.  Use a fallback if the previous run produced a vector
    # ------------------------------------------------------------------
    if "height_sequence_1" in globals():
        try:
            fb = [float(x) for x in globals()["height_sequence_1"]]
            val = evaluate_sequence(fb)
            print(f"Fallback (height_sequence_1) score = {val:.6f}")
            if val < best_val:
                best_val = val
                best_seq = fb
                best_n = len(fb)
        except Exception:
            pass

    # ------------------------------------------------------------------
    # 1.  Coarse exploration over a few lengths and families
    # ------------------------------------------------------------------
    lengths = [256, 512, 1024, 2048]      # larger lengths increase the 2·n factor
    for n in lengths:
        if time.time() >= deadline:
            break
        print(f"\n--- Exploring length n = {n} ---")

        # uniform baseline (score = 2.0)
        uniform = uniform_shape(n)
        sc = compute_score(uniform)
        if sc < best_val:
            best_val = sc
            best_seq = uniform.tolist()
            best_n = n
            print(f"Uniform baseline improves best → {best_val:.6f}")

        # arcsine (used in the original script)
        arc = arcsine_initial(n) if 'arcsine_initial' in globals() else uniform
        sc = compute_score(arc)
        if sc < best_val:
            best_val = sc
            best_seq = arc.tolist()
            best_n = n
            print(f"Arcsine baseline improves best → {best_val:.6f}")

        # ----- beta family -----
        vec, sc = _opt_beta(n, deadline, rng)
        if vec is not None and sc < best_val:
            best_val = sc
            best_seq = vec.tolist()
            best_n = n
            print(f"Beta family improves best → {best_val:.6f}")

        # ----- power‑law family -----
        vec, sc = _opt_power(n, deadline, rng)
        if vec is not None and sc < best_val:
            best_val = sc
            best_seq = vec.tolist()
            best_n = n
            print(f"Power‑law family improves best → {best_val:.6f}")

        # ----- exponential family -----
        vec, sc = _opt_exponential(n, deadline, rng)
        if vec is not None and sc < best_val:
            best_val = sc
            best_seq = vec.tolist()
            best_n = n
            print(f"Exponential family improves best → {best_val:.6f}")

        # ----- beta + uniform mixture -----
        vec, sc = _opt_beta_uniform_mix(n, deadline, rng)
        if vec is not None and sc < best_val:
            best_val = sc
            best_seq = vec.tolist()
            best_n = n
            print(f"Beta+Uniform mix improves best → {best_val:.6f}")

        # ----- power + uniform mixture -----
        vec, sc = _opt_power_uniform_mix(n, deadline, rng)
        if vec is not None and sc < best_val:
            best_val = sc
            best_seq = vec.tolist()
            best_n = n
            print(f"Power+Uniform mix improves best → {best_val:.6f}")

    # ------------------------------------------------------------------
    # 2.  Heavy local polishing of the overall best seed
    # ------------------------------------------------------------------
    if best_seq and time.time() < deadline - 5.0:
        a = np.array(best_seq, dtype=float)

        # A – Adam on the smooth surrogate (long run)
        remaining = deadline - time.time()
        max_iter_adam = int(max(10_000, remaining * 30))
        print(f"\nPolishing – Adam smooth (max_iter={max_iter_adam})")
        a, _ = adam_optimize_smooth(
            a,
            deadline=deadline,
            max_iter=max_iter_adam,
            lr0=0.13,
            lam_start=2.0,
            lam_end=120.0,
            noise_every=2000,
            noise_scale=2e-5,
        )

        # B – exact sub‑gradient descent
        if time.time() < deadline - 5.0:
            remaining = deadline - time.time()
            max_iter_sub = int(max(10_000, remaining * 25))
            print(f"Polishing – sub‑gradient descent (max_iter={max_iter_sub})")
            a = subgradient_descent_exact(a, deadline, max_iter=max_iter_sub, step0=0.025)

        # C – simulated‑annealing polishing
        if time.time() < deadline - 5.0:
            remaining = deadline - time.time()
            max_iter_sa = int(max(10_000, remaining * 25))
            print(f"Polishing – simulated annealing (max_iter={max_iter_sa})")
            a = simulated_annealing(a, rng, deadline, max_iters=max_iter_sa,
                                    init_temp=0.015, decay=0.99995)

        # Official evaluation (for logging only)
        try:
            val = evaluate_sequence(a.tolist())
            print(f"Polished official score = {val:.6f}")
            if val < best_val:
                best_val = val
                best_seq = a.tolist()
                best_n = a.size
                print("*** New global best after polishing ***")
        except Exception:
            pass

    # ------------------------------------------------------------------
    # 3.  Guard – ensure non‑empty return
    # ------------------------------------------------------------------
    if not best_seq:
        # Very last resort – uniform of a modest length
        n = 512
        best_seq = uniform_shape(n).tolist()
        best_val = evaluate_sequence(best_seq)

    print(f"\nSearch finished. Best reported score = {best_val:.6f} (length {best_n})")
    return best_seq
