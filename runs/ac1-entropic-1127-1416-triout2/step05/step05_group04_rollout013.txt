def build_conv_matrix(f):
    n = len(f)

def propose_candidate(seed=42, budget_s=1000, **kwargs):
    import time, numpy as np; from scipy.optimize import linprog; from scipy.sparse import diags
    start = time.time()
    deadline = start + budget_s - 1.0
    rng = np.random.default_rng(seed)

    n = 2048
    best_seq = None
    best_val = np.inf

    # initial seeds list
    seeds = []

    # uniform
    uniform = np.full(n, 1.0/n)
    seeds.append(uniform)
    # arcsine
    seeds.append(arcsine_initial(n))
    # random betas
    for _ in range(5):
        alpha = rng.uniform(0.1, 10)
        beta = rng.uniform(0.1, 10)
        eps = 1e-6
        seeds.append(generate_beta_shape(n, alpha, beta, eps))
    # mixture param
    for _ in range(5):
        w = rng.random()
        seeds.append(generate_mixture_shape(n, w))

    # Evaluate initial seeds, keep best
    for s in seeds:
        val = compute_score(s)
        if val < best_val:
            best_val = val
            best_seq = s

    # Run LP-based refinement on best seq
    f = best_seq.copy()
    iteration = 0
    while time.time() < deadline - 5.0 and iteration < 8:
        iteration += 1
        M = np.max(convolution_fft(f, f))
        # Build constraint matrix
        A = diags([np.full(n, f[i]) for i in range(n)], offsets=np.arange(n), shape=(2*n-1, n), format='csr')
        b = np.full(2*n-1, M)
        res = linprog(c=-np.ones(n), A_ub=A, b_ub=b, bounds=[(0, None)]*n, method='highs')
        if not res.success:
            print("LP failed", res.message)
            break
        g = res.x
        sumg = g.sum()
        if sumg == 0:
            break
        g = g / sumg
        t = 0.5  # mixing weight
        f_new = (1 - t)*f + t*g
        # project onto simplex (should already sum to 1, but small rounding errors)
        f_new = project_to_simplex(f_new)
        cur_val = compute_score(f_new)
        print(f"Iter {iteration}: obj {cur_val:.6f} (mix {t})")
        if cur_val < best_val:
            best_val = cur_val
            best_seq = f_new.copy()
        # If improvement not significant, reduce t
        if cur_val >= compute_score(f):
            t = max(0.1, t*0.7)
        f = f_new

    # final polishing using Adam + subgradient (optional)
    # use same functions as before if time left.
    if time.time() < deadline - 2.0:

"""
Propose a non‑negative step sequence that minimizes the official
``evaluate_sequence`` score.

The search consists of four stages

1. **Wide random exploration** of several parametric families
   (Beta‑shape, power‑law, mixture, triangular).  Only the best few
   seeds are retained (a small top‑K heap).

2. **LP‑based refinement** – for a current vector ``f`` we compute
   ``M = max(f∗f)`` and solve the linear program

        maximise   Σ g_j
        subject to  (f∗g)_k ≤ M   for all k
                    g_j ≥ 0

   The solution ``g`` is normalised to sum = 1 and mixed with ``f``.
   A few iterations of this step move the distribution away from the
   places that cause the largest autocorrelation.

3. **Smooth Adam optimisation** of a log‑sum‑exp surrogate of the
   max‑convolution, followed by an exact sub‑gradient descent and a
   short simulated‑annealing polish.

4. **Final evaluation** with the official ``evaluate_sequence``.  The
   best vector seen during the whole run is returned.  All operations
   respect the 1000 s time budget and rely only on NumPy/SciPy.
"""

import time
import numpy as np
from typing import List, Tuple

def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """
    Linear convolution via real FFT.
    Length = len(a)+len(b)-1.
    """
    m = len(a) + len(b) - 1
    n_fft = next_pow2(m)
    A = np.fft.rfft(a, n=n_fft)
    B = np.fft.rfft(b, n=n_fft)
    C = A * B
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


def compute_score(a: np.ndarray) -> float:
    """
    Fast exact objective for vectors normalised to Σa = 1:
        2·n·max(conv(a,a))
    (identical to the official score because the denominator is 1.)
    """
    n = len(a)
    conv = convolution_fft(a, a)
    return 2.0 * n * float(np.max(conv))


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection onto the probability simplex {x ≥ 0, Σx = 1}.
    Duchi et al., 2008.
    """
    if np.all(v <= 0):
        # Uniform fallback – never happens in practice
        return np.full_like(v, 1.0 / v.size)

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho = np.where(u - cssv / np.arange(1, v.size + 1) > 0)[0]
    if len(rho) == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def arcsine_initial(n: int) -> np.ndarray:
    """Classic arcsine shape: a_i ∝ 1/√((i+½)*(n‑½‑i))."""
    i = np.arange(n, dtype=float)
    eps = 0.25
    vals = 1.0 / np.sqrt((i + 0.5) * (n - 0.5 - i) + eps)
    vals /= vals.sum()
    return vals


def generate_beta_shape(n: int, alpha: float, beta: float, eps: float) -> np.ndarray:
    """
    (i+ε)^{α‑1}·(n‑1‑i+ε)^{β‑1} on {0,…,n‑1}, normalised to Σ=1.
    """
    i = np.arange(n, dtype=float)
    left = (i + eps) ** (alpha - 1.0)
    right = (n - 1 - i + eps) ** (beta - 1.0)
    base = left * right
    total = base.sum()
    if not np.isfinite(total) or total <= 0.0:
        return np.full(n, 1.0 / n)
    return base / total


def generate_power_shape(n: int, p: float, eps: float) -> np.ndarray:
    """
    Generalised arcsine: a_i ∝ ((i+½)*(n‑½‑i)+ε)^{-p},   p>0, eps≥0.
    """
    i = np.arange(n, dtype=float)
    denom = ((i + 0.5) * (n - 0.5 - i) + eps) ** p
    vals = 1.0 / denom
    total = vals.sum()
    if not np.isfinite(total) or total <= 0.0:
        return np.full(n, 1.0 / n)
    return vals / total


def generate_triangular_shape(n: int, peak: int) -> np.ndarray:
    """Symmetric triangular bump with its apex at ``peak``."""
    left = np.arange(peak + 1, dtype=float)
    right = np.arange(n - peak - 1, dtype=float)[::-1]
    seq = np.concatenate([left, right])
    seq /= seq.sum()
    return seq


def generate_mixture_shape(n: int, weight_arc: float) -> np.ndarray:
    """Mixture (weight_arc)·arcsine + (1‑weight_arc)·uniform."""
    arc = arcsine_initial(n)
    uni = np.full(n, 1.0 / n)
    mix = weight_arc * arc + (1.0 - weight_arc) * uni
    mix /= mix.sum()
    return mix

def smooth_gradient(a: np.ndarray, lam: float) -> np.ndarray:
    """
    Gradient of the log‑sum‑exp surrogate of max(conv(a,a)).
    Returns ∇_a (2·n·softmax_max) where the simplex constraint is
    enforced later by a projection.
    """
    n = len(a)
    conv = convolution_fft(a, a)               # length L = 2·n‑1
    max_c = np.max(conv)

    # Soft‑max weights (numerically stable)
    exp_term = np.exp(lam * (conv - max_c))
    w = exp_term / np.sum(exp_term)            # Σ w = 1

    # Weighted sub‑gradient 2·Σ_k w_k shift(a,k)
    # Using convolution: (a * w_rev) evaluated at positions that line up with a.
    w_rev = w[::-1]
    conv_rev = convolution_fft(a, w_rev)       # length n+L-1
    L = 2 * n - 1
    weighted_grad = 2.0 * conv_rev[L - 1 - np.arange(n)]

    return 2.0 * n * weighted_grad


def adam_optimize_smooth(initial_a: np.ndarray,
                         deadline: float,
                         max_iter: int = 100_000,
                         lr0: float = 0.12,
                         lam_start: float = 2.0,
                         lam_end: float = 150.0,
                         noise_every: int = 3000,
                         noise_scale: float = 5e-5) -> Tuple[np.ndarray, float]:
    """
    Adam optimisation on the smooth surrogate.
    Returns the best vector (projected on the simplex) and its fast score.
    """
    a = project_to_simplex(initial_a.astype(float).copy())
    best_a = a.copy()
    best_sc = compute_score(a)

    beta1, beta2 = 0.9, 0.999
    eps = 1e-8
    m = np.zeros_like(a)
    v = np.zeros_like(a)

    for it in range(1, max_iter + 1):
        if time.time() >= deadline:
            break

        lam = lam_start + (lam_end - lam_start) * min(1.0, it / max_iter)

        grad = smooth_gradient(a, lam)

        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)

        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        step = lr0 * m_hat / (np.sqrt(v_hat) + eps)
        a_new = a - step

        if noise_every and (it % noise_every == 0):
            a_new += np.random.randn(a.size) * noise_scale

        a_new = project_to_simplex(a_new)

        cur_sc = compute_score(a_new)
        if cur_sc < best_sc:
            best_sc = cur_sc
            best_a = a_new.copy()

        a = a_new

        if it % 5000 == 0:
            print(f"    AdamSmooth it={it:,d} cur={cur_sc:.6f} best={best_sc:.6f}")

    return best_a, best_sc


def subgradient_descent_exact(initial_a: np.ndarray,
                              deadline: float,
                              max_iter: int = 80_000,
                              step0: float = 0.04) -> np.ndarray:
    """
    Sub‑gradient descent on the exact objective f(a)=2·n·max(conv(a,a)).
    Step size η_t = step0 / √t.
    """
    a = project_to_simplex(initial_a.copy())
    best_a = a.copy()
    best_sc = compute_score(a)
    n = len(a)
    L = 2 * n - 1

    for it in range(1, max_iter + 1):
        if time.time() >= deadline:
            break

        conv = convolution_fft(a, a)
        max_val = np.max(conv)

        # tolerance to pick all active (max‑attaining) shifts
        tol = max_val * 1e-12 + 1e-14
        mask = (np.abs(conv - max_val) <= tol).astype(float)

        # Gradient = 2 * Σ_{k active} shift(a,k)
        conv_rev = convolution_fft(a, mask[::-1])
        weighted_grad = 2.0 * conv_rev[L - 1 - np.arange(n)]

        eta = step0 / np.sqrt(it)
        a = a - eta * weighted_grad
        a = project_to_simplex(a)

        cur_sc = compute_score(a)
        if cur_sc < best_sc:
            best_sc = cur_sc
            best_a = a.copy()

        if it % 5000 == 0:
            print(f"    SubGrad it={it:,d} cur={cur_sc:.6f} best={best_sc:.6f}")

    return best_a


def simulated_annealing(a: np.ndarray,
                        rng: np.random.Generator,
                        deadline: float,
                        max_iters: int = 120_000,
                        init_temp: float = 0.02,
                        decay: float = 0.9999) -> np.ndarray:
    """
    Very light Metropolis‑style local search using the fast surrogate.
    """
    cur = a.copy()
    cur_sc = compute_score(cur)
    best = cur.copy()
    best_sc = cur_sc
    n = len(cur)
    temp = init_temp
    it = 0

    while it < max_iters and time.time() < deadline:
        it += 1
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue

        delta = cur[i] * 0.2 * rng.random()
        if delta <= 1e-15 or cur[i] - delta < 0:
            continue

        cand = cur.copy()
        cand[i] -= delta
        cand[j] += delta

        cand_sc = compute_score(cand)

        if cand_sc < cur_sc or rng.random() < np.exp(-(cand_sc - cur_sc) / max(temp, 1e-12)):
            cur = cand
            cur_sc = cand_sc
            if cur_sc < best_sc:
                best = cur.copy()
                best_sc = cur_sc

        temp = max(temp * decay, 1e-8)

        if it % 5000 == 0:
            print(f"    SA it={it:,d} cur={cur_sc:.6f} best={best_sc:.6f}")

    return best

from scipy.sparse import diags
from scipy.optimize import linprog


def build_conv_matrix(f: np.ndarray):
    """
    Return a sparse matrix A (shape (2n‑1, n)) such that A @ g = conv(f,g).
    Each row k contains the coefficients f_i that contribute to (f∗g)_k.
    """
    n = len(f)
    # For offset d (= i) the diagonal consists of the constant value f[d]
    data = [np.full(n, f[d]) for d in range(n)]
    offsets = np.arange(n)                     # 0 … n‑1
    A = diags(data, offsets, shape=(2 * n - 1, n), format="csr")
    return A


def lp_refine(f: np.ndarray,
              deadline: float,
              max_outer: int = 6,
              mix_weight: float = 0.5) -> np.ndarray:
    """
    Perform at most ``max_outer`` iterations of the LP‑based refinement.
    Each iteration solves

        maximise Σ g_j
        s.t.   conv(f,g) ≤ M
               g ≥ 0

    where M = max(conv(f,f)).  The solution is projected onto the simplex,
    mixed with the current vector and returned.
    """
    cur = f.copy()
    n = len(cur)

    for outer in range(1, max_outer + 1):
        if time.time() >= deadline - 0.5:
            break

        M = np.max(convolution_fft(cur, cur))
        A = build_conv_matrix(cur)
        b = np.full(2 * n - 1, M)

        # maximise Σ g → minimise -Σ g
        res = linprog(c=-np.ones(n), A_ub=A, b_ub=b,
                      bounds=[(0, None)] * n,
                      method="highs")
        if not res.success:
            print(f"    LP iteration {outer} failed: {res.message}")
            break

        g = res.x
        sumg = g.sum()
        if sumg <= 0:
            break
        g = g / sumg                     # back to the simplex

        # convex mixing
        cur = (1.0 - mix_weight) * cur + mix_weight * g
        cur = project_to_simplex(cur)

        cur_sc = compute_score(cur)
        print(f"    LP‑refine iter {outer}: obj={cur_sc:.6f}")

    return cur

def propose_candidate(seed: int = 42,
                     budget_s: int = 1000,
                     **kwargs) -> List[float]:
    """
    Search for a low‑score non‑negative step sequence.
    The routine blends a fast random parametrised sweep, an LP‑based
    refinement, and heavy local optimisation (Adam → sub‑gradient → SA).
    The best vector found (as a plain Python list) is returned.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 1.0   # 1 s safety margin

    # ------------------------------------------------------------------
    # Global bookkeeping
    # ------------------------------------------------------------------
    best_seq: List[float] = []
    best_val: float = float("inf")

    # optional fallback from a previous run
    if "height_sequence_1" in globals():
        try:
            fallback = [float(x) for x in globals()["height_sequence_1"]]
            val = evaluate_sequence(fallback)
            print(f"Fallback (height_sequence_1) score = {val:.6f}")
            if val < best_val:
                best_val, best_seq = val, fallback
        except Exception:
            pass

    # ------------------------------------------------------------------
    # 0.  Choose a working length (2048 gives a good trade‑off)
    # ------------------------------------------------------------------
    n = 2048

    # ------------------------------------------------------------------
    # 1.  Wide random exploration – keep a tiny top‑K heap
    # ------------------------------------------------------------------
    top_k = 10
    best_pool: List[Tuple[float, np.ndarray]] = []   # (score, vector)

    def maybe_insert(score: float, vec: np.ndarray):
        """Insert (score,vec) keeping pool size ≤ top_k."""
        if len(best_pool) < top_k:
            best_pool.append((score, vec))
        else:
            worst_idx = max(range(len(best_pool)),
                            key=lambda i: best_pool[i][0])
            if score < best_pool[worst_idx][0]:
                best_pool[worst_idx] = (score, vec)

    # Deterministic starters
    maybe_insert(compute_score(np.full(n, 1.0 / n)), np.full(n, 1.0 / n))
    maybe_insert(compute_score(arcsine_initial(n)), arcsine_initial(n))

    # Random parametric families
    max_candidates = 120_000
    cnt = 0
    while time.time() < deadline - 30.0 and cnt < max_candidates:
        cnt += 1
        r = rng.random()
        if r < 0.30:                     # beta family (wide α,β)
            alpha = rng.uniform(0.05, 12.0)
            beta = rng.uniform(0.05, 12.0)
            eps = rng.uniform(1e-6, 3.0)
            cand = generate_beta_shape(n, alpha, beta, eps)
        elif r < 0.55:                   # power / generalized arcsine
            p = rng.uniform(0.05, 6.0)
            eps = rng.uniform(0.0, 3.0)
            cand = generate_power_shape(n, p, eps)
        elif r < 0.70:                   # mixture arcsine / uniform
            w = rng.random()
            cand = generate_mixture_shape(n, w)
        elif r < 0.85:                   # triangular bump
            peak = rng.integers(0, n)
            cand = generate_triangular_shape(n, int(peak))
        else:                            # Dirichlet‑like scatter
            alphas = rng.exponential(scale=1.0, size=n) + 0.01
            cand = alphas / alphas.sum()

        sc = compute_score(cand)
        maybe_insert(sc, cand)

    # Extract the best seeds in order of increasing score
    best_pool.sort(key=lambda x: x[0])
    seed_vectors = [vec for _, vec in best_pool]

    # ------------------------------------------------------------------
    # 2.  Refine each seed (skip if time is almost gone)
    # ------------------------------------------------------------------
    for idx, seed_vec in enumerate(seed_vectors, 1):
        if time.time() >= deadline - 8.0:
            break
        print(f"\n--- Refining seed {idx}/{len(seed_vectors)} ---")

        # a) LP‑based refinement (few fast iterations)
        f = seed_vec.copy()
        f = lp_refine(f, deadline, max_outer=4, mix_weight=0.45)

        # b) Adam on the smooth surrogate (allow many iterations)
        if time.time() < deadline - 5.0:
            remaining = deadline - time.time()
            max_iter_adam = int(min(250_000, remaining * 40))
            f, _ = adam_optimize_smooth(
                initial_a=f,
                deadline=deadline,
                max_iter=max_iter_adam,
                lr0=0.12,
                lam_start=2.0,
                lam_end=120.0,
                noise_every=2500,
                noise_scale=2e-5,
            )

        # c) Exact sub‑gradient descent
        if time.time() < deadline - 5.0:
            remaining = deadline - time.time()
            max_iter_sub = int(min(200_000, remaining * 30))
            f = subgradient_descent_exact(
                f,
                deadline,
                max_iter=max_iter_sub,
                step0=0.03,
            )

        # d) Very short simulated annealing polish
        if time.time() < deadline - 5.0:
            remaining = deadline - time.time()
            max_iter_sa = int(min(150_000, remaining * 30))
            f = simulated_annealing(
                f,
                rng,
                deadline,
                max_iters=max_iter_sa,
                init_temp=0.015,
                decay=0.99995,
            )

        # Official evaluation
        try:
            val = evaluate_sequence(f.tolist())
            print(f"    Official score = {val:.6f}")
            if val < best_val:
                best_val, best_seq = val, f.tolist()
                print(f"    *** New global best: {best_val:.6f} ***")
        except Exception:
            pass

    # ------------------------------------------------------------------
    # 3.  Global polishing of the overall best (if any time left)
    # ------------------------------------------------------------------
    if best_seq and time.time() < deadline - 4.0:
        a = np.array(best_seq, dtype=float)
        remaining = deadline - time.time()
        # a few more sub‑gradient steps
        sub_iters = int(min(250_000, remaining * 30))
        a = subgradient_descent_exact(a, deadline, max_iter=sub_iters, step0=0.02)
        # a final Adam run with a stronger soft‑max
        a, _ = adam_optimize_smooth(
            a,
            deadline,
            max_iter=int(min(200_000, (deadline - time.time()) * 30)),
            lr0=0.08,
            lam_start=2.0,
            lam_end=150.0,
            noise_every=2500,
            noise_scale=1e-5,
        )
        final_val = evaluate_sequence(a.tolist())
        print(f"Final polished official score = {final_val:.6f}")
        if final_val < best_val:
            best_val, best_seq = final_val, a.tolist()
            print("*** Updated global best after final polish ***")

    # ------------------------------------------------------------------
    # 4.  Guard – return something sensible
    # ------------------------------------------------------------------
    if not best_seq:
        best_seq = [1.0] * n
        best_val = evaluate_sequence(best_seq)

    print(f"\nSearch finished. Best reported score = {best_val:.6f}")
    return best_seq
