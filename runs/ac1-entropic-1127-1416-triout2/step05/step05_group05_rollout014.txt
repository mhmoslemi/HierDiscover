"""
Propose a non‑negative step function (a probability vector) that minimizes

        2·n·max(conv(a,a)) / (sum(a)²)

The algorithm proceeds in three stages within the given time budget:

1.  **Initial pool** – a handful of simple families (arcsine, uniform,
    random Dirichlet, symmetric power‑law, symmetric Beta) are generated
    for a relatively large length (default n = 4096).  The best of these
    (according to a fast FFT‑based surrogate score) seeds the search.

2.  **Randomised shape search** – while a small fraction of the remaining
    time is left, random parameters for the power‑law and Beta families are
    sampled.  Each candidate is evaluated with the fast surrogate and the
    best is kept.  This often discovers a shape slightly better than the
    deterministic families.

3.  **Annealed mirror‑descent** – a multiplicative (exponentiated‑gradient)
    update is performed on the simplex.  The weight vector for the gradient
    is a soft‑max over the convolution entries, with a temperature that
    increases linearly from `beta_start` to `beta_end`.  This “smooths’’ the
    max‑operator initially and gradually hardens it, allowing stable
    progress toward the true objective.  The step‑size follows η = η₀/√t.
    The best iterate (according to the exact objective) is retained.

4.  **Hard sub‑gradient polish** – the final candidate is refined with the
    classic sub‑gradient method that uses the genuine max‑weight (uniform on
    the set of maximisers).  This finishes the optimisation with the remaining
    time budget.

All intermediate scores are computed with a fast FFT‑based routine;
only the final candidate is evaluated with the official ``evaluate_sequence``
function.  The routine always returns a feasible list of non‑negative
floats and respects the supplied time limit.
"""

import time
import numpy as np
from typing import List, Tuple

# ----------------------------------------------------------------------
# Low‑level numeric utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """
    Linear convolution via real FFT (full length), zero‑padded to a power‑of‑2.
    """
    m = len(a) + len(b) - 1
    n_fft = next_pow2(m)
    A = np.fft.rfft(a, n=n_fft)
    B = np.fft.rfft(b, n=n_fft)
    C = A * B
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


def normalize(v: np.ndarray) -> np.ndarray:
    """Project a non‑negative vector onto the simplex (sum = 1)."""
    v = np.maximum(v, 0.0)          # safety
    total = v.sum()
    if total <= 0.0:
        return np.full_like(v, 1.0 / len(v))
    return v / total


def compute_score(a: np.ndarray) -> float:
    """
    Fast evaluation of the official objective (scale‑invariant):
        2·n·max(conv(a,a)) / (sum(a)²)
    The vector a is assumed non‑negative and already normalised (sum = 1).
    """
    n = len(a)
    conv = convolution_fft(a, a)
    max_val = float(np.max(conv))
    return 2.0 * n * max_val


def gradient_from_weights(a: np.ndarray, w: np.ndarray) -> np.ndarray:
    """
    Gradient of Σ_k w_k·(a*a)_k with respect to a.
    Implements   g_j = 2· Σ_m a_m · w_{m+j}.
    Both a and w are assumed to be real, a ≥ 0, w has length 2·len(a)-1.
    """
    w_rev = w[::-1]
    conv = convolution_fft(a, w_rev)               # length = 3·n‑2
    idx = 2 * len(a) - 2                            # position 2n‑2
    grads = 2.0 * conv[idx - np.arange(len(a))]
    return grads


# ----------------------------------------------------------------------
# Parametric families for the initial pool and random search
# ----------------------------------------------------------------------


def arcsine_initial(n: int) -> np.ndarray:
    """Discrete arcsine distribution ≈ 1/√(i·(n‑1‑i)) (with a tiny ε)."""
    i = np.arange(n, dtype=float)
    eps = 0.25
    vals = 1.0 / np.sqrt((i + 0.5) * (n - 0.5 - i) + eps)
    return normalize(vals)


def powerlaw_shape(n: int, p: float, eps: float) -> np.ndarray:
    """Symmetric power‑law a_i ∝ ((i+ε)(n‑1‑i+ε))⁻ᵖ."""
    i = np.arange(n, dtype=float)
    base = ((i + eps) * (n - 1 - i + eps)) ** (-p)
    return normalize(base)


def beta_shape(n: int, alpha: float, eps: float = 1e-8) -> np.ndarray:
    """
    Symmetric Beta(α,α) density sampled on n points:
        a_i ∝ (i+ε)^{α‑1}·(n‑1‑i+ε)^{α‑1}
    """
    i = np.arange(n, dtype=float)
    vals = ((i + eps) ** (alpha - 1)) * ((n - 1 - i + eps) ** (alpha - 1))
    return normalize(vals)


def uniform_shape(n: int) -> np.ndarray:
    """Uniform probability vector on n points."""
    return np.full(n, 1.0 / n)


def random_dirichlet(n: int, rng: np.random.Generator) -> np.ndarray:
    """Draw a random point on the simplex (exponential sampling)."""
    samp = rng.exponential(1.0, size=n)
    return normalize(samp)


# ----------------------------------------------------------------------
# Refinement methods
# ----------------------------------------------------------------------


def refine_annealed(
    a: np.ndarray,
    deadline: float,
    *,
    eta0: float = 1.0,
    beta_start: float = 2.0,
    beta_end: float = 80.0,
    rng: np.random.Generator = None,
    report_every: int = 0,
) -> Tuple[np.ndarray, float]:
    """
    Exponentiated‑gradient (mirror‑descent) with a *soft* max‑weight that
    linearly hardens from ``beta_start`` to ``beta_end``.
    Returns the best iterate (exact fast score) and its score.
    """
    a = normalize(a.astype(float).copy())
    n = len(a)
    best_a = a.copy()
    conv = convolution_fft(a, a)
    best_sc = 2.0 * n * float(np.max(conv))

    it = 0
    start = time.time()
    total_time = max(1e-9, deadline - start)

    while time.time() < deadline:
        it += 1
        # elapsed fraction for the beta schedule
        elapsed = time.time() - start
        beta = beta_start + (beta_end - beta_start) * (elapsed / total_time)

        # current convolution
        conv = convolution_fft(a, a)
        max_val = float(np.max(conv))

        # soft‑max‑type weights (more uniform for small beta)
        exp_vals = np.exp(beta * (conv - max_val))  # max entry -> weight = 1
        w = exp_vals / np.sum(exp_vals)

        # gradient of the smoothed objective
        grad = gradient_from_weights(a, w)

        # step‑size schedule (classic η = η₀ / √t)
        eta = eta0 / np.sqrt(it)

        # multiplicative update and projection onto simplex
        a = a * np.exp(-eta * grad)
        a = normalize(a)

        cur_sc = 2.0 * n * max_val
        if cur_sc < best_sc:
            best_sc = cur_sc
            best_a = a.copy()

        if report_every and (it % report_every == 0):
            print(f"    annealed iter={it:,d} cur={cur_sc:.6f} best={best_sc:.6f}")

        # occasional tiny random jitter to escape plateaus
        if rng and it % 7000 == 0:
            jitter = rng.normal(scale=0.005, size=n)
            a = a * (1.0 + jitter)
            a = np.clip(a, 0.0, None)
            a = normalize(a)

    return best_a, best_sc


def refine_subgradient(
    a: np.ndarray,
    deadline: float,
    *,
    max_iter: int = 300_000,
    eta0: float = 0.7,
    report_every: int = 0,
) -> Tuple[np.ndarray, float]:
    """
    Classic sub‑gradient descent on the *exact* max‑convolution objective.
    Multiplicative updates with step‑size ηₜ = η₀ / √t.
    Returns the best iterate seen and its fast score.
    """
    a = normalize(a.astype(float).copy())
    best_a = a.copy()
    n = len(a)

    conv = convolution_fft(a, a)
    best_sc = 2.0 * n * float(np.max(conv))

    for it in range(1, max_iter + 1):
        if time.time() >= deadline:
            break

        conv = convolution_fft(a, a)
        max_val = float(np.max(conv))

        # Uniform weight on (near) maximisers
        w = np.zeros_like(conv)
        max_idx = np.where(conv >= max_val - 1e-14)[0]
        w[max_idx] = 1.0 / len(max_idx)

        grad = gradient_from_weights(a, w)

        eta = eta0 / np.sqrt(it)
        a = a * np.exp(-eta * grad)
        a = normalize(a)

        cur_sc = 2.0 * n * max_val
        if cur_sc < best_sc:
            best_sc = cur_sc
            best_a = a.copy()

        if report_every and (it % report_every == 0):
            print(f"    sub‑grad iter={it:,d} cur={cur_sc:.6f} best={best_sc:.6f}")

    return best_a, best_sc
