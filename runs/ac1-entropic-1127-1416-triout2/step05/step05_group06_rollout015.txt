def lp_refine(a0, n, deadline, rng, t_initial=0.3, max_iter=15):
    f = a0.astype(float).copy()
    # ensure sum_f > 0, else set uniform
    if f.sum() == 0:
        f = np.full(n, 1.0 / n)
    sum_f = f.sum()
    best_f = f.copy()
    best_sc = compute_score(f)
    # precompute pattern
    rows, cols, diffs = conv_pattern_cache.get(n) else build and store.

    for it in range(max_iter):
        if time.time() >= deadline - 2:
            break
        # compute M
        conv_f = convolution_fft(f, f)
        M = conv_f.max()
        # build A matrix using pattern and f
        data = f[diffs]  # vectorized
        A = sparse.csr_matrix((data, (rows, cols)), shape=(2*n-1, n))
        # solve LP
        c = -np.ones(n)  # maximize sum(g)
        b_ub = np.full(2*n-1, M)
        res = scipy.optimize.linprog(c, A_ub=A, b_ub=b_ub, bounds=(0, None), method='highs')
        if not res.success:
            # fallback: break
            print(f"LP failed at iter {it} with status {res.message}")
            break
        g = res.x
        # scale g to have same sum as f
        sum_g = g.sum()
        if sum_g == 0:
            break
        g_scaled = g * (sum_f / sum_g)
        # update t schedule: maybe decreasing t
        t = t_initial / np.sqrt(it + 1)
        f_new = (1 - t) * f + t * g_scaled
        f = f_new
        # evaluate
        cur_sc = compute_score(f)
        if cur_sc < best_sc:
            best_sc = cur_sc
            best_f = f.copy()
    return best_f, best_sc

def lp_refine(f, n, deadline, rng, max_iters=12, t0=0.2):
    # pattern
    rows, cols, diffs = get_conv_pattern(n)
    f = np.array(f, dtype=float)
    sum_f = f.sum()
    best_f = f.copy()
    best_score = compute_score(f)
    it = 0
    while it < max_iters and time.time() < deadline - 2:
        # compute M
        conv_f = convolution_fft(f, f)
        M = conv_f.max()
        # prepare data for constraint matrix
        data = f[diffs]  # diff shape same as rows length
        A = sparse.csr_matrix((data, (rows, cols)), shape=(2*n-1, n))
        # Linear program
        c = -np.ones(n)  # maximize sum(g)
        b_ub = np.full(2*n-1, M)
        # Solve
        res = scipy.optimize.linprog(c, A_ub=A, b_ub=b_ub,
                                    bounds=[(0, None)]*n,
                                    method='highs', options={'presolve': True})
        if not res.success:
            # maybe fallback to previous f
            print(f"LP solve failed at iter {it}: {res.message}")
            break
        g = res.x
        # scale g to match sum_f
        sum_g = g.sum()
        if sum_g == 0:
            break
        g_scaled = g * (sum_f / sum_g)
        # update step size
        t = t0 / np.sqrt(it + 1)
        f = (1 - t) * f + t * g_scaled
        # normalize small negative due to numeric
        f = np.maximum(f, 0.0)
        # optionally renormalize sum exactly
        f = f * (sum_f / f.sum())
        cur_score = compute_score(f)
        if cur_score < best_score:
            best_score = cur_score
            best_f = f.copy()
        it += 1
    return best_f, best_score
```

"""
Hybrid optimizer for the step‑function max‑convolution objective.

1.  A diverse library of seed shapes (uniform, arcsine, power‑law,
    beta‑like, Dirichlet draws, simple mixtures) is generated for a
    chosen length (powers of two – FFT‑friendly).

2.  Each seed is refined by an *alternating linear‑program* scheme:
    – Let f be the current vector, M = max(f∗f).
    – Solve the LP max Σ g_j s.t. f∗g ≤ M and g ≥ 0.
      The convolution constraint is a Toeplitz matrix; it is built
      once per length as a sparse pattern and the coefficients are
      updated each iteration.
    – Scale g to keep the ℓ₁‑sum unchanged and replace f by a convex
      combination f←(1‑t)f+t·g (t decays like 1/√it).  The max‑convolution
      never increases, and a few iterations already lower the objective
      appreciably.

3.  The best vector obtained from the LP stage is finally polished
    with an exponentiated‑sub‑gradient descent on the true max‑convolution
    (the same update that was used in the previous baseline) – many
    thousands of cheap FFT‑based steps are performed while respecting the
    time budget.

All operations stay inside the supplied wall‑clock limit (default 1000 s)
and the routine always returns a feasible non‑negative list of floats.
"""

import time
import math
import numpy as np
from scipy import sparse
from scipy.optimize import linprog

# ----------------------------------------------------------------------
# Global cache for the sparse Toeplitz pattern used in the LP constraints
# ----------------------------------------------------------------------
_conv_pattern_cache = {}


def _conv_pattern(n: int):
    """Return (rows, cols, diffs) describing the Toeplitz pattern of f∗g."""
    if n not in _conv_pattern_cache:
        rows = []
        cols = []
        diffs = []                      # row index minus column index = k‑i
        m = 2 * n - 1
        for k in range(m):
            lo = max(0, k - (n - 1))
            hi = min(k, n - 1)
            for i in range(lo, hi + 1):
                rows.append(k)
                cols.append(i)
                diffs.append(k - i)   # coefficient will be f[diffs]
        _conv_pattern_cache[n] = (
            np.array(rows, dtype=np.int32),
            np.array(cols, dtype=np.int32),
            np.array(diffs, dtype=np.int32),
        )
    return _conv_pattern_cache[n]


# ----------------------------------------------------------------------
# Basic numeric utilities (FFT‑based convolution, fast objective)
# ----------------------------------------------------------------------
def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full linear convolution via real FFT (zero‑padded to a power of two)."""
    m = len(a) + len(b) - 1
    n_fft = next_pow2(m)
    A = np.fft.rfft(a, n=n_fft)
    B = np.fft.rfft(b, n=n_fft)
    conv = np.fft.irfft(A * B, n=n_fft)[:m]
    return conv


def compute_score(a: np.ndarray) -> float:
    """
    Fast surrogate of evaluate_sequence:
        2·n·max(a∗a) / (sum(a)²)
    The vector a is assumed non‑negative.
    """
    n = a.shape[0]
    conv = convolution_fft(a, a)
    max_val = float(conv.max())
    s = float(a.sum())
    if s < 1e-12:
        return float("inf")
    return 2.0 * n * max_val / (s * s)


# ----------------------------------------------------------------------
# Families of initial shapes
# ----------------------------------------------------------------------
def arcsine_initial(n: int) -> np.ndarray:
    """Discrete arcsine density (sum = 1)."""
    i = np.arange(n, dtype=float)
    eps = 0.25
    vals = 1.0 / np.sqrt((i + 0.5) * (n - 0.5 - i) + eps)
    return vals / vals.sum()


def generate_powerlaw(n: int, p: float, eps: float) -> np.ndarray:
    """Symmetric power‑law a_i ∝ ((i+ε)(n‑1‑i+ε))⁻ᵖ."""
    i = np.arange(n, dtype=float)
    base = ((i + eps) * (n - 1 - i + eps)) ** (-p)
    total = base.sum()
    if total == 0.0:
        return np.full(n, 1.0 / n)
    return base / total


def generate_beta_like(n: int, alpha: float, beta: float) -> np.ndarray:
    """Beta‑like shape a_i ∝ (i+½)^{α‑1}(n‑1‑i+½)^{β‑1}."""
    i = np.arange(n, dtype=float)
    base = (i + 0.5) ** (alpha - 1) * (n - 1 - i + 0.5) ** (beta - 1)
    total = base.sum()
    if total == 0.0:
        return np.full(n, 1.0 / n)
    return base / total


def random_dirichlet(n: int, concentration: float, rng: np.random.Generator) -> np.ndarray:
    """Sample a Dirichlet‑like vector (Gamma draws) and normalise."""
    if concentration <= 0:
        samp = rng.exponential(1.0, size=n) ** (1.0 / max(0.1, concentration))
    else:
        samp = rng.gamma(concentration, 1.0, size=n)
    samp += 1e-12
    return samp / samp.sum()


def mix_arcsine_uniform(n: int, rng: np.random.Generator) -> np.ndarray:
    """Simple convex mixture of arcsine and the uniform distribution."""
    w = rng.random()
    mix = w * arcsine_initial(n) + (1.0 - w) * np.full(n, 1.0 / n)
    return mix / mix.sum()


def generate_initial_candidates(n: int, rng: np.random.Generator) -> list[np.ndarray]:
    """Return a modest library of diverse starting vectors."""
    cand = []

    # uniform
    cand.append(np.full(n, 1.0 / n))

    # classic arcsine
    cand.append(arcsine_initial(n))

    # random Dirichlet draws (various concentrations)
    for conc in [0.1, 0.3, 0.8, 2.0, 5.0]:
        cand.append(random_dirichlet(n, conc, rng))

    # random power‑law shapes
    for _ in range(4):
        p = rng.uniform(0.2, 1.3)
        eps = rng.uniform(0.001, max(1.0, 0.02 * n))
        cand.append(generate_powerlaw(n, p, eps))

    # random beta‑like shapes
    for _ in range(4):
        alpha = rng.uniform(0.4, 5.0)
        beta = rng.uniform(0.4, 5.0)
        cand.append(generate_beta_like(n, alpha, beta))

    # mixtures arcsine ↔ uniform
    for _ in range(2):
        cand.append(mix_arcsine_uniform(n, rng))

    return cand


# ----------------------------------------------------------------------
# Linear‑program (LP) refinement of a vector
# ----------------------------------------------------------------------
def lp_refine(
    a0: np.ndarray,
    n: int,
    deadline: float,
    rng: np.random.Generator,
    max_iters: int = 12,
    t0: float = 0.4,
) -> tuple[np.ndarray, float]:
    """
    Alternating LP scheme:

    * M = max(f∗f)
    * solve   max Σ g   s.t.   f∗g ≤ M , g ≥ 0
    * scale g to keep the ℓ₁‑sum equal to that of f
    * update f ← (1‑t)f + t·g   with t decaying like t0/√it

    The process never increases the max‑convolution and typically
    reduces it after a handful of iterations.
    """
    f = np.array(a0, dtype=float)
    if f.sum() == 0.0:
        f = np.full(n, 1.0 / n)

    sum_f = f.sum()
    best_f = f.copy()
    best_sc = compute_score(f)

    rows, cols, diffs = _conv_pattern(n)

    for it in range(max_iters):
        if time.time() >= deadline - 1.0:
            break

        # current max‑convolution value
        conv_f = convolution_fft(f, f)
        M = float(conv_f.max())

        # Build the sparse matrix A where (A·g)_k = Σ_i f_{k-i}·g_i
        data = f[diffs]  # vectorised coefficient update
        A = sparse.csr_matrix((data, (rows, cols)), shape=(2 * n - 1, n))

        # Linear program: maximise Σ g_i  ⇔  minimise -Σ g_i
        c = -np.ones(n)
        b_ub = np.full(2 * n - 1, M)
        bounds = [(0.0, None)] * n

        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method="highs")
        if not res.success:
            # In practice this rarely happens; abort the inner loop.
            print(f"   LP failed (iter={it}) – {res.message}")
            break

        g = res.x
        sum_g = g.sum()
        if sum_g == 0.0:
            break

        # Scale g to keep the ℓ₁‑sum unchanged
        g = g * (sum_f / sum_g)

        # Convex update (step size decays)
        t = t0 / math.sqrt(it + 1.0)
        f = (1.0 - t) * f + t * g
        # Numerical hygiene
        f = np.maximum(f, 0.0)
        f = f * (sum_f / f.sum())

        cur_sc = compute_score(f)
        if cur_sc < best_sc:
            best_sc = cur_sc
            best_f = f.copy()

    return best_f, best_sc


# ----------------------------------------------------------------------
# Exponentiated sub‑gradient descent on the true max‑convolution
# ----------------------------------------------------------------------
def _max_conv_gradient(a: np.ndarray, conv: np.ndarray) -> np.ndarray:
    """
    Sub‑gradient of max_k (a∗a)_k.
    For every offset attaining the maximum we add the corresponding
    cross‑correlation contribution; the average is taken if several
    offsets are active.
    """
    n = a.shape[0]
    max_val = conv.max()
    mask = np.isclose(conv, max_val, atol=1e-15, rtol=0.0)
    active = np.where(mask)[0]
    grad = np.zeros_like(a)

    for k in active:
        lo = max(0, k - (n - 1))
        hi = min(k, n - 1)
        idx = np.arange(lo, hi + 1, dtype=int)
        grad[idx] += 2.0 * a[k - idx]

    if len(active) > 1:
        grad /= len(active)
    return grad


def optimise_subgradient(
    initial_a: np.ndarray,
    deadline: float,
    *,
    eta0: float = 0.5,
    max_iter: int = 300_000,
) -> tuple[np.ndarray, float]:
    """
    Exponentiated sub‑gradient descent directly on
    Φ(a)=max_k (a∗a)_k.
    """
    a = np.maximum(initial_a, 1e-30)
    a = a / a.sum()
    best_a = a.copy()
    best_sc = compute_score(a)
    n = a.shape[0]

    for it in range(1, max_iter + 1):
        if time.time() >= deadline - 1.0:
            break

        conv = convolution_fft(a, a)
        grad = _max_conv_gradient(a, conv)

        eta = eta0 / math.sqrt(it)
        a = a * np.exp(-eta * grad)

        # renormalise onto the simplex
        s = a.sum()
        if s == 0.0:
            a = np.full_like(a, 1.0 / n)
        else:
            a = a / s

        cur_sc = compute_score(a)
        if cur_sc < best_sc:
            best_sc = cur_sc
            best_a = a.copy()

        # occasional progress report (does not affect timing much)
        if it % 50000 == 0:
            print(f"    subgrad iter={it:,d} cur={cur_sc:.6f} best={best_sc:.6f}")

    return best_a, best_sc
