def refine_power(a, deadline, *, eta0=0.8, p_start=2.0, p_end=60.0, report_every=0):

    a = normalize(a.astype(float).copy())
    n = len(a)
    best_a = a.copy()
    conv = convolution_fft(a,a)
    max_val = float(np.max(conv))
    best_sc = 2.0 * n * max_val   # sum = 1

    it = 0
    start = time.time()

    while True:
        cur_time = time.time()
        if cur_time >= deadline:
            break
        it += 1

        conv = convolution_fft(a,a)
        max_val = float(np.max(conv))
        # determine p for this iteration based on elapsed fraction
        elapsed_frac = (cur_time - start) / (deadline - start)
        p = p_start + (p_end - p_start) * elapsed_frac

        # avoid overflow issues: conv may be zero at ends; add small epsilon
        eps = 1e-20
        weights = np.maximum(conv, eps) ** (p - 1)
        sum_w = np.sum(weights)
        if sum_w == 0:
            w = np.full_like(weights, 1.0/len(weights))
        else:
            w = weights / sum_w

        grad = gradient_from_weights(a, w)

        eta = eta0 / np.sqrt(it)   # schedule
        a = a * np.exp(-eta * grad)

        # enforce symmetry (optional)
        a = (a + a[::-1]) / 2.0

        a = normalize(a)

        cur_sc = 2.0 * n * max_val
        if cur_sc < best_sc:
            best_sc = cur_sc
            best_a = a.copy()
            if report_every and it % report_every == 0:
                print(f"    iter={it} p={p:.2f} cur={cur_sc:.6f} best={best_sc:.6f}")

        if report_every and it % report_every == 0:
            print(f"    iter={it} p={p:.2f} cur={cur_sc:.6f} best={best_sc:.6f}")

    return best_a, best_sc

def refine_pow(a, deadline, *, eta0=0.9, beta_start=5.0, beta_end=30.0, p_start=2.0, p_end=100.0, report_every=0):
    a = normalize(a.astype(float).copy())
    n = len(a)
    best_a = a.copy()
    conv = convolution_fft(a, a)
    max_val = float(np.max(conv))
    best_sc = 2.0 * n * max_val

    it = 0
    start = time.time()
    # optional momentum
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8

    while time.time() < deadline:
        it += 1
        # compute conv and max
        conv = convolution_fft(a, a)
        max_val = float(np.max(conv))
        # compute p and beta based on elapsed fraction
        fraction = (time.time() - start) / (deadline - start)
        p = p_start + (p_end - p_start) * fraction
        beta = beta_start + (beta_end - beta_start) * fraction

        # compute weight w using power relative to max
        scale = max_val if max_val > 0 else 1.0
        rel = conv / scale
        # ensure non-negative (they are)
        # Avoid numerical issues for large p:
        if p >= 1.0:
            w_raw = rel ** (p - 1)   # max entry becomes 1
        else:
            # p<1 not used, just fallback to uniform
            w_raw = np.ones_like(rel)

        w = w_raw / np.sum(w_raw)

        # gradient
        grad = gradient_from_weights(a, w)

        # momentum updates
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad ** 2)
        m_hat = m / (1 - beta1 ** it)
        v_hat = v / (1 - beta2 ** it)

        # step size schedule
        eta = eta0 / np.sqrt(it)

        # exponentiated Adam update
        step = eta * m_hat / (np.sqrt(v_hat) + eps)
        a = a * np.exp(-step)

        # enforce symmetry (optional)
        a = (a + a[::-1]) / 2.0

        a = normalize(a)

        cur_sc = 2.0 * n * max_val
        if cur_sc < best_sc:
            best_sc = cur_sc
            best_a = a.copy()
            if report_every and it % report_every == 0:
                print(f"    iter={it} p={p:.2f} cur={cur_sc:.6f} best={best_sc:.6f}")

        if report_every and it % report_every == 0:
            print(f"    iter={it} p={p:.2f} cur={cur_sc:.6f} best={best_sc:.6f}")

    return best_a, best_sc

def refine_pow(a, deadline, *, eta0=0.9, p_start=2.0, p_end=120.0, max_iter=None, report_every=0):
    # a normalized sum = 1
    a = normalize(a.astype(float).copy())
    n = len(a)
    best_a = a.copy()
    conv = convolution_fft(a, a)
    best_sc = 2.0 * n * np.max(conv)
    best_val = best_sc

    it = 0
    start = time.time()
    # Adam params
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8

    # maximum number of iterations limited by time or optional max_iter
    while time.time() < deadline and (max_iter is None or it < max_iter):
        it += 1
        conv = convolution_fft(a, a)
        max_val = float(np.max(conv))

        # schedule p, increasing
        elapsed_frac = (time.time() - start) / (deadline - start)
        p = p_start + (p_end - p_start) * elapsed_frac

        # compute weight w via power relative to max
        # avoid division by zero
        rel = conv / max_val if max_val > 0 else np.zeros_like(conv)
        # numerically stable: for p large, pow of small numbers underflow => zero, fine
        w_raw = rel ** (p - 1)  # max entry becomes 1
        w_sum = np.sum(w_raw)
        if w_sum == 0:
            w = np.full_like(w_raw, 1.0 / len(w_raw))
        else:
            w = w_raw / w_sum

        # gradient of weighted sum
        grad = gradient_from_weights(a, w)

        # Adam update
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad ** 2)

        m_hat = m / (1 - beta1 ** it)
        v_hat = v / (1 - beta2 ** it)

        eta = eta0 / np.sqrt(it)

        step = eta * m_hat / (np.sqrt(v_hat) + eps)

        # Clip step to avoid extreme updates
        step = np.clip(step, -5.0, 5.0)

        a = a * np.exp(-step)

        # enforce symmetry
        a = (a + a[::-1]) / 2.0

        a = normalize(a)

        cur_sc = 2.0 * n * max_val
        if cur_sc < best_sc:
            best_sc = cur_sc
            best_a = a.copy()
            if report_every and it % report_every == 0:
                print(f"    iter={it} p={p:.2f} cur={cur_sc:.6f} best={best_sc:.6f}")

        if report_every and it % report_every == 0:
            print(f"    iter={it} p={p:.2f} cur={cur_sc:.6f} best={best_sc:.6f}")

    return best_a, best_sc

"""
Search for a low‑score non‑negative step function.

Strategy
--------
1. Build a diverse pool of initial families (arcsine, uniform, Hann,
   Blackman, Gaussian, random Dirichlet, power‑law, plus the previous
   best construction if available).  All candidates are normalised to
   have sum = 1.

2. Pick the three cheapest candidates according to a fast FFT‑based
   proxy (2·n·max(conv) with sum = 1).  Each of them is refined with
   *exponentiated‑gradient Adam* using a *power‑law soft‑max* weighting:
   w_i ∝ (conv_i / max(conv))^(p−1) with p growing from 2 to ~120.
   Symmetry a_i = a_{n‑1‑i} is enforced after every step.
   The iteration stops when the time budget for that candidate expires.

3. The best refined vector is polished with the classic hard sub‑gradient
   method (exact max‑weight) for the remaining time.

All numbers are kept non‑negative, normalised to a simplex, and the
function respects the supplied time limit (default 1000 s).  The final
list of floats is returned; it can be fed directly to ``evaluate_sequence``.
"""

import time
import numpy as np
from typing import List, Tuple

# ----------------------------------------------------------------------
# Low‑level numeric utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """
    Linear convolution via real FFT (full length, zero‑padded to a power‑of‑2).
    """
    m = len(a) + len(b) - 1
    n_fft = next_pow2(m)
    A = np.fft.rfft(a, n=n_fft)
    B = np.fft.rfft(b, n=n_fft)
    C = A * B
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


def normalize(v: np.ndarray) -> np.ndarray:
    """Project a non‑negative vector onto the simplex (sum = 1)."""
    total = v.sum()
    if total <= 0.0:
        return np.full_like(v, 1.0 / len(v))
    return v / total


def compute_score(a: np.ndarray) -> float:
    """
    Fast (scale‑invariant) proxy for the official objective:
        2·n·max(conv(a,a))   with   sum(a)=1 .
    """
    n = len(a)
    conv = convolution_fft(a, a)
    max_val = float(np.max(conv))
    return 2.0 * n * max_val


def gradient_from_weights(a: np.ndarray, w: np.ndarray) -> np.ndarray:
    """
    Gradient of Σ_k w_k·(a*a)_k w.r.t. a.
    Implements   g_j = 2· Σ_m a_m · w_{m+j}.
    Computed via one FFT‑based convolution.
    """
    w_rev = w[::-1]
    conv = convolution_fft(a, w_rev)               # length = 3·n‑2
    idx = 2 * len(a) - 2                            # position 2n‑2
    grads = 2.0 * conv[idx - np.arange(len(a))]
    return grads


# ----------------------------------------------------------------------
# Parametric families for the initial sweep
# ----------------------------------------------------------------------


def arcsine_initial(n: int) -> np.ndarray:
    """Discrete arcsine distribution (≈ 1/√(i·(n‑1‑i))) normalised."""
    i = np.arange(n, dtype=float)
    eps = 0.25
    vals = 1.0 / np.sqrt((i + 0.5) * (n - 0.5 - i) + eps)
    return normalize(vals)


def uniform_shape(n: int) -> np.ndarray:
    """Uniform probability vector on n points."""
    return np.full(n, 1.0 / n)


def hann_window(n: int) -> np.ndarray:
    """Hann window (raised cosine), normalised."""
    i = np.arange(n, dtype=float)
    w = 0.5 - 0.5 * np.cos(2.0 * np.pi * i / (n - 1))
    return normalize(w)


def blackman_window(n: int) -> np.ndarray:
    """Blackman window, normalised."""
    i = np.arange(n, dtype=float)
    w = (
        0.42
        - 0.5 * np.cos(2.0 * np.pi * i / (n - 1))
        + 0.08 * np.cos(4.0 * np.pi * i / (n - 1))
    )
    return normalize(w)


def gaussian_shape(n: int, sigma: float | None = None) -> np.ndarray:
    """Discrete symmetric Gaussian, normalised."""
    if sigma is None:
        sigma = n / 6.0
    i = np.arange(n, dtype=float)
    mid = (n - 1) / 2.0
    w = np.exp(-0.5 * ((i - mid) / sigma) ** 2)
    return normalize(w)


def random_dirichlet(n: int, rng: np.random.Generator) -> np.ndarray:
    """Draw a random point on the simplex (exponential sampling)."""
    samp = rng.exponential(1.0, size=n)
    return normalize(samp)


def powerlaw_shape(n: int, p: float, eps: float) -> np.ndarray:
    """Symmetric power‑law shape a_i ∝ ((i+ε)(n‑1‑i+ε))⁻ᵖ."""
    i = np.arange(n, dtype=float)
    base = ((i + eps) * (n - 1 - i + eps)) ** (-p)
    return normalize(base)


# ----------------------------------------------------------------------
# Refinement methods
# ----------------------------------------------------------------------


def refine_pow(
    a: np.ndarray,
    deadline: float,
    *,
    eta0: float = 0.9,
    p_start: float = 2.0,
    p_end: float = 120.0,
    report_every: int = 0,
) -> Tuple[np.ndarray, float]:
    """
    Exponentiated‑gradient with Adam moments and a *power‑law* soft‑max.
    The weight on convolution entry k is
        w_k ∝ (conv_k / max(conv))^(p‑1)
    with p linearly increasing from ``p_start`` to ``p_end``.
    Symmetry a_i = a_{n‑1‑i} is enforced after each update.
    Returns the best iterate (exact objective) and its score.
    """
    a = normalize(a.astype(float).copy())
    n = len(a)
    best_a = a.copy()
    # exact score (sum(a)=1 → score = 2·n·max(conv))
    conv = convolution_fft(a, a)
    best_sc = 2.0 * n * float(np.max(conv))

    # Adam bookkeeping
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8

    it = 0
    start = time.time()
    while time.time() < deadline:
        it += 1

        conv = convolution_fft(a, a)
        max_val = float(np.max(conv))

        # schedule p (smooth → hard)
        frac = (time.time() - start) / (deadline - start)
        p = p_start + (p_end - p_start) * frac

        # compute power‑law weights
        if max_val > 0:
            rel = conv / max_val
        else:
            rel = np.zeros_like(conv)
        # for p>1, max entry becomes 1, others shrink towards 0
        w_raw = rel ** (p - 1)
        w_sum = np.sum(w_raw)
        if w_sum == 0:
            w = np.full_like(w_raw, 1.0 / len(w_raw))
        else:
            w = w_raw / w_sum

        # gradient of the weighted sum
        grad = gradient_from_weights(a, w)

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)

        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        eta = eta0 / np.sqrt(it)

        step = eta * m_hat / (np.sqrt(v_hat) + eps)
        # safety clipping
        step = np.clip(step, -5.0, 5.0)

        a = a * np.exp(-step)

        # enforce symmetry and project back onto the simplex
        a = (a + a[::-1]) / 2.0
        a = normalize(a)

        cur_sc = 2.0 * n * max_val
        if cur_sc < best_sc:
            best_sc = cur_sc
            best_a = a.copy()
            if report_every and it % report_every == 0:
                print(
                    f"    pow‑iter={it:,d} p={p:.2f} cur={cur_sc:.6f} best={best_sc:.6f}"
                )

        if report_every and it % report_every == 0:
            print(
                f"    pow‑iter={it:,d} p={p:.2f} cur={cur_sc:.6f} best={best_sc:.6f}"
            )

    return best_a, best_sc


def refine_subgradient(
    a: np.ndarray,
    deadline: float,

    max_iter: int = 200_000,
    eta0: float = 0.5,

) -> Tuple[np.ndarray, float]:
    """
    Classical sub‑gradient descent on the exact max‑convolution objective.
    Uses multiplicative (exponentiated‑gradient) updates with a 1/√t step‑size.
    """
    a = normalize(a.astype(float).copy())
    best_a = a.copy()
    n = len(a)
    conv = convolution_fft(a, a)
    best_sc = 2.0 * n * float(np.max(conv))

    for it in range(1, max_iter + 1):

        max_val = float(np.max(conv))

        # uniform weight on the (near) maximisers
        w = np.zeros_like(conv)
        max_idx = np.where(conv >= max_val - 1e-14)[0]
        w[max_idx] = 1.0 / len(max_idx)

        grad = gradient_from_weights(a, w)

        eta = eta0 / np.sqrt(it)
        a = a * np.exp(-eta * grad)
        # enforce symmetry (optional but helpful)
        a = (a + a[::-1]) / 2.0
        a = normalize(a)

        cur_sc = 2.0 * n * max_val
        if cur_sc < best_sc:
            best_sc = cur_sc
            best_a = a.copy()
            if report_every and it % report_every == 0:
                print(
                    f"    sub‑grad iter={it:,d} cur={cur_sc:.6f} best={best_sc:.6f}"
                )

        if report_every and it % report_every == 0:
            print(
                f"    sub‑grad iter={it:,d} cur={cur_sc:.6f} best={best_sc:.6f}"
            )

    return best_a, best_sc
