def adam_optimize(initial_a, deadline, max_iter=5000, alpha0=0.5, beta1=0.9, beta2=0.999, eps=1e-8):
    w = np.log(initial_a + 1e-30) # use stable
    m = np.zeros_like(w)
    v = np.zeros_like(w)
    best_a = np.exp(w - np.max(w))
    best_a /= best_a.sum()
    best_score = compute_score(best_a)
    t = 0
    for it in range(1, max_iter+1):
        if time.time() >= deadline:
            break
        t = it
        # compute a
        w_shifted = w - np.max(w)  # prevent overflow
        a = np.exp(w_shifted)
        a_sum = a.sum()
        if a_sum == 0.0:
            a = np.full_like(w, 1.0 / len(w))
        else:
            a = a / a_sum

        # compute conv
        conv = convolution_fft(a, a)
        # compute subgradient
        grad_a = _max_conv_gradient(a, conv)
        # compute grad w
        dot = np.dot(a, grad_a)
        grad_w = a * (grad_a - dot)

        # Adam update
        m = beta1 * m + (1 - beta1) * grad_w
        v = beta2 * v + (1 - beta2) * (grad_w ** 2)
        m_hat = m / (1 - beta1 ** it)
        v_hat = v / (1 - beta2 ** it)
        eta = alpha0 / (np.sqrt(it) + 1e-8)  # or just alpha0
        w -= eta * m_hat / (np.sqrt(v_hat) + eps)

        # evaluate current a (after update)
        cur_score = compute_score(a)
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
    return best_a, best_score

"""
Hybrid “continuous‑beta → smooth‑max → Adam” search for a low‑max‑convolution step
function.

Main ideas
----------

* **Beta‑family seeding** – the optimal shape is close to a symmetric Beta
  density on the interval.  We sweep a fine grid of the shape parameter
  α (α=β) for a given length, evaluate the cheap scale‑invariant score
  ``2·n·max(a∗a)/(∑a)²`` and keep the best few candidates.

* **Smooth‑max surrogate** – instead of the non‑smooth max we minimise
  ``Φβ(a) = (1/β)·log∑k exp(β·(a∗a)k)``.  Its gradient is
  ``∇Φβ(a) = 2·(p⋆a)`` where ``p`` is the soft‑max weight vector
  ``p_k ∝ exp(β·(a∗a)_k)``.  The factor 4·n that appears in the official
  objective (``2·n·max``) is folded into the gradient.

* **Exponentiated‑gradient Adam** – we work in the log‑domain
  ``w_i = log a_i`` (so the simplex constraint is automatic via the
  softmax).  Adam’s adaptive moments accelerate convergence on the
  highly non‑convex landscape.

* **Hierarchical up‑sampling** – a good shape on a modest grid (e.g.
  n=4096) is linearly interpolated to a finer grid (2·n) and refined
  again.  This lets us exploit the fact that the objective’s asymptotic
  value is independent of n while a finer discretisation reduces the
  discretisation error.

* **Final polishing** – the best vector found is polished with the
  exact sub‑gradient method from the reference implementation.

The routine respects the wall‑clock budget (default 1000 s) and always
returns a feasible list of non‑negative floats.
"""

import time
import numpy as np
from typing import List, Tuple

# ----------------------------------------------------------------------
# Basic numeric utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full linear convolution using a real FFT (zero‑padded to a power of two)."""
    m = len(a) + len(b) - 1
    n_fft = next_pow2(m)
    A = np.fft.rfft(a, n=n_fft)
    B = np.fft.rfft(b, n=n_fft)
    C = A * B
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


def compute_score(a: np.ndarray) -> float:
    """
    Fast version of the official objective (scale‑invariant):
        2·n·max(a∗a) / (sum(a)²)
    """
    n = len(a)
    conv = convolution_fft(a, a)
    max_val = float(conv.max())
    s = float(a.sum())
    if s < 1e-12:
        return float("inf")
    return 2.0 * n * max_val / (s * s)


# ----------------------------------------------------------------------
# Families of initial shapes
# ----------------------------------------------------------------------


def arcsine_initial(n: int) -> np.ndarray:
    """Discrete arcsine density (normalised to sum = 1)."""
    i = np.arange(n, dtype=float)
    eps = 0.25
    vals = 1.0 / np.sqrt((i + 0.5) * (n - 0.5 - i) + eps)
    return vals / vals.sum()


def generate_beta_like(n: int, alpha: float, beta: float) -> np.ndarray:
    """Beta‑like shape a_i ∝ (i+½)^{α‑1} (n‑1‑i+½)^{β‑1}."""
    i = np.arange(n, dtype=float)
    base = (i + 0.5) ** (alpha - 1) * (n - 1 - i + 0.5) ** (beta - 1)
    # guard against NaNs when alpha or beta < 1
    base = np.where(np.isfinite(base), base, 0.0)
    total = base.sum()
    if total == 0.0:
        return np.full(n, 1.0 / n)
    return base / total


def random_dirichlet(n: int, concentration: float, rng: np.random.Generator) -> np.ndarray:
    """Sample a Dirichlet‑like vector and normalise."""
    if concentration <= 0:
        samp = rng.exponential(1.0, size=n) ** (1.0 / max(0.1, concentration))
    else:
        samp = rng.gamma(concentration, 1.0, size=n)
    samp += 1e-12
    return samp / samp.sum()


# ----------------------------------------------------------------------
# Sub‑gradient of the true max‑convolution (used for final polishing)
# ----------------------------------------------------------------------


def _max_conv_gradient(a: np.ndarray, conv: np.ndarray) -> np.ndarray:
    """
    Sub‑gradient of max_k (a∗a)_k.
    For each active offset k where the maximum is attained, we add the
    contribution 2·a_{k‑j}.  When several offsets are active we average.
    """
    n = len(a)
    max_val = conv.max()
    mask = np.isclose(conv, max_val, atol=1e-15, rtol=0.0)
    active = np.where(mask)[0]
    grad = np.zeros_like(a)

    for k in active:
        lo = max(0, k - (n - 1))
        hi = min(k, n - 1)
        idx = np.arange(lo, hi + 1, dtype=int)
        grad[idx] += 2.0 * a[k - idx]

    if len(active) > 1:
        grad /= len(active)
    return grad


# ----------------------------------------------------------------------
# Adam optimiser on the smooth‑max surrogate
# ----------------------------------------------------------------------


def adam_smooth_optimize(initial_a: np.ndarray,
                         deadline: float,
                         max_iter: int,
                         *,
                         alpha0: float = 0.6,
                         beta1: float = 0.9,
                         beta2: float = 0.999,
                         eps: float = 1e-8,
                         beta_max: float = 2500.0,
                         eval_every: int = 50) -> Tuple[np.ndarray, float]:
    """
    Adam optimiser on the smooth surrogate
        Φβ(a) = (1/β)·log Σ_k exp(β·(a∗a)_k)
    The vector a is kept on the simplex via a log‑softmax
    (w = log a, a = softmax(w)).  β is increased linearly from 1
    to *beta_max* during the run.
    """
    n = len(initial_a)
    # initialise logits
    w = np.log(initial_a + 1e-30)
    m = np.zeros_like(w)
    v = np.zeros_like(w)

    # best bookkeeping
    best_a = initial_a.copy()
    best_sc = compute_score(best_a)

    for it in range(1, max_iter + 1):
        if time.time() >= deadline:
            break

        # ----- obtain current a from logits -----
        w_shift = w - np.max(w)                # for numerical stability
        a = np.exp(w_shift)
        a_sum = a.sum()
        if a_sum == 0.0:
            a = np.full_like(w, 1.0 / n)
        else:
            a = a / a_sum

        # ----- convolution a∗a -----
        conv = convolution_fft(a, a)          # length 2n‑1
        max_c = conv.max()

        # ----- soft‑max weights (smooth max) -----
        # linear schedule for β
        beta = 1.0 + (beta_max - 1.0) * it / max_iter
        # stable exponentiation
        exp_vals = np.exp(beta * (conv - max_c))
        p = exp_vals / exp_vals.sum()          # Σ p = 1

        # ----- gradient of the smooth surrogate -----
        # cross‑correlation p ⋆ a  (length n)
        N = next_pow2(len(p) + n - 1)
        P = np.fft.rfft(p, N)
        A = np.fft.rfft(a, N)
        cross = np.fft.irfft(P * np.conj(A), N)[:n].real
        grad_a = 4.0 * n * cross               # ∇_a (2·n·Φβ)

        # ----- convert to gradient w.r.t. logits (softmax Jacobian) -----
        dot = np.dot(a, grad_a)
        grad_w = a * (grad_a - dot)

        # ----- Adam update on w -----
        m = beta1 * m + (1.0 - beta1) * grad_w
        v = beta2 * v + (1.0 - beta2) * (grad_w ** 2)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        step = alpha0 / (np.sqrt(it) + 1e-8)   # a mild 1/√t schedule
        w -= step * m_hat / (np.sqrt(v_hat) + eps)

        # ----- periodic evaluation with the exact (non‑smooth) score -----
        if it % eval_every == 0:
            # recompute a after the update for an accurate score
            w_shift = w - np.max(w)
            a_eval = np.exp(w_shift)
            a_eval /= a_eval.sum()
            sc = compute_score(a_eval)
            if sc < best_sc:
                best_sc = sc
                best_a = a_eval.copy()

    return best_a, best_sc


# ----------------------------------------------------------------------
# Final polishing using the exact max‑convolution sub‑gradient
# ----------------------------------------------------------------------


def optimise_subgradient(initial_a: np.ndarray,
                         deadline: float,
                         *,
                         eta0: float = 0.6,
                         max_iter: int = 200_000,
                         noise_every: int = 0,
                         noise_scale: float = 0.0) -> Tuple[np.ndarray, float]:
    """
    Exponentiated‑gradient sub‑gradient descent on the true objective
    ``max_k (a∗a)_k``.  The implementation is identical to the one used
    in the reference code; it serves as a polishing step.
    """
    a = np.maximum(initial_a, 1e-30)
    a = a / a.sum()
    best_a = a.copy()
    best_sc = compute_score(best_a)
    n = len(a)

    for it in range(1, max_iter + 1):
        if time.time() >= deadline:
            break

        conv = convolution_fft(a, a)
        grad = _max_conv_gradient(a, conv)

        eta = eta0 / np.sqrt(it)
        a = a * np.exp(-eta * grad)

        if noise_every and (it % noise_every == 0):
            a += np.random.randn(n) * noise_scale
            a = np.maximum(a, 0.0)

        s = a.sum()
        a = a / s if s != 0.0 else np.full_like(a, 1.0 / n)

        cur_sc = compute_score(a)
        if cur_sc < best_sc:
            best_sc = cur_sc
            best_a = a.copy()

    return best_a, best_sc

def upsample_vector(v: np.ndarray, new_len: int) -> np.ndarray:
    """Linear interpolation from len(v) to new_len, then renormalise."""
    old_x = np.linspace(0, 1, len(v))
    new_x = np.linspace(0, 1, new_len)
    up = np.interp(new_x, old_x, v)
    up = np.maximum(up, 0.0)
    if up.sum() == 0.0:
        up = np.full(new_len, 1.0 / new_len)
    else:
        up = up / up.sum()
    return up
