"""
Propose a low‑discrepancy step sequence that minimizes the official
evaluate_sequence() score.

Algorithm
---------

1. Choose a moderate length `n` (default 4096 – larger `n` gives a lower
   theoretical bound).  All vectors are normalised to sum = 1, which is
   safe because the official objective is homogeneous.

2. Generate a handful of parametric “seed” vectors (arcsine,
   beta‑shaped, mixtures, triangular bumps, random Dirichlet).  Their
   fast surrogate score

        S(a) = 2·n·max(conv(a,a))

   is evaluated with an FFT‑based convolution.  The three best seeds are
   kept.

3. For each retained seed perform a three‑stage local optimisation,
   always respecting the simplex constraint:

   * **Adam on a smooth max** – we replace the hard max by a
     log‑sum‑exp with an annealed temperature `λ`.  The gradient is
     obtained analytically and the iterate is projected onto the
     probability simplex.

   * **Exact sub‑gradient descent** – after the Adam phase we switch to
     the true non‑smooth objective and use a diminishing step size
     ηₜ = step0/√t.

   * **Simulated annealing polishing** – a cheap Metropolis move that
     swaps a short‑range fraction of mass between two coordinates,
     accepted according to the fast surrogate.

   Each phase respects a global deadline (the supplied time budget).

4. After each refinement the official `evaluate_sequence` function is
   called; the best result is stored.

5. If time permits, a final global polish (sub‑gradient + Adam) is run
   on the current overall best vector.

All heavy work uses only NumPy; the routine fits comfortably inside the
1000 s limit on a single CPU and finally returns a list of non‑negative
heights ready for the official evaluator.
"""

import time
import numpy as np
from typing import List, Tuple

# ----------------------------------------------------------------------
# Low‑level utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """
    Linear convolution via real FFT.
    Length = len(a) + len(b) - 1.
    """
    m = len(a) + len(b) - 1
    n_fft = next_pow2(m)
    A = np.fft.rfft(a, n=n_fft)
    B = np.fft.rfft(b, n=n_fft)
    C = A * B
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


def compute_score(a: np.ndarray) -> float:
    """
    Fast surrogate for the official score.
    The vector is assumed to sum to 1, therefore the denominator of the
    official formula disappears and the score reduces to
        2·n·max(conv(a,a)).
    """
    n = a.shape[0]
    conv = convolution_fft(a, a)
    return 2.0 * n * float(np.max(conv))


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    if np.all(v <= 0):
        # all non‑positive – return the uniform distribution
        return np.full_like(v, 1.0 / v.size)

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho_arr = u - cssv / np.arange(1, v.size + 1)
    rho = np.nonzero(rho_arr > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Shape factories  (all vectors sum to 1)
# ----------------------------------------------------------------------


def arcsine_shape(n: int, eps: float = 0.25) -> np.ndarray:
    """Discrete arcsine‑type shape with a small epsilon regulariser."""
    i = np.arange(n, dtype=float)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - 0.5 - i) + eps)
    vals /= vals.sum()
    return vals


def beta_shape(n: int, alpha: float, beta: float, eps: float = 1e-6) -> np.ndarray:
    """
    (i+ε)^{α‑1}·(n‑1‑i+ε)^{β‑1} on {0,…,n‑1}, normalised to Σ=1.
    """
    i = np.arange(n, dtype=float)
    left = (i + eps) ** (alpha - 1.0)
    right = (n - 1 - i + eps) ** (beta - 1.0)
    base = left * right
    total = base.sum()
    if not np.isfinite(total) or total <= 0.0:
        return np.full(n, 1.0 / n)
    return base / total


def mixture_shape(n: int, weight_arc: float = 0.5) -> np.ndarray:
    """Mixture (weight_arc)·arcsine + (1‑weight_arc)·uniform."""
    arc = arcsine_shape(n)
    uni = np.full(n, 1.0 / n)
    mix = weight_arc * arc + (1.0 - weight_arc) * uni
    mix /= mix.sum()
    return mix


def triangular_shape(n: int, peak: int) -> np.ndarray:
    """Symmetric triangular bump with its apex at `peak`."""
    left = np.arange(peak + 1, dtype=float)
    right = np.arange(n - peak - 1, dtype=float)[::-1]
    seq = np.concatenate([left, right])
    seq /= seq.sum()
    return seq


def random_dirichlet(n: int, rng: np.random.Generator) -> np.ndarray:
    """Full Dirichlet sample – all components stay positive."""
    alphas = rng.exponential(scale=1.0, size=n) + 0.01
    vec = alphas / alphas.sum()
    return vec


# ----------------------------------------------------------------------
# Optimisers
# ----------------------------------------------------------------------


def smooth_gradient(a: np.ndarray, lam: float) -> np.ndarray:
    """
    Gradient of the soft‑max approximation of max(conv(a,a)).
    Returns ∇_a (2·n·softmax_max) where the simplex constraint is
    enforced afterwards by a projection.
    """
    n = a.shape[0]
    conv = convolution_fft(a, a)               # length 2·n‑1
    max_c = np.max(conv)

    # stable soft‑max weights (sum to 1)
    exp_term = np.exp(lam * (conv - max_c))
    w = exp_term / np.sum(exp_term)

    # convolution of a with reversed weight vector gives the summed
    # sub‑gradient for each component
    w_rev = w[::-1]
    conv_rev = convolution_fft(a, w_rev)
    L = 2 * n - 1
    weighted_grad = 2.0 * conv_rev[L - 1 - np.arange(n)]

    # factor 2·n from the surrogate objective
    return 2.0 * n * weighted_grad


def adam_optimize_smooth(initial_a: np.ndarray,
                         deadline: float,
                         max_iter: int = 200_000,
                         lr0: float = 0.14,
                         lam_start: float = 2.0,
                         lam_end: float = 200.0,
                         noise_every: int = 0,
                         noise_scale: float = 0.0) -> Tuple[np.ndarray, float]:
    """
    Adam optimiser on the smooth surrogate (log‑sum‑exp max).
    Returns the best vector (projected onto the simplex) and its fast
    surrogate score.
    """
    a = project_to_simplex(initial_a.astype(float).copy())
    best_a = a.copy()
    best_sc = compute_score(a)

    beta1, beta2 = 0.9, 0.999
    eps = 1e-8
    m = np.zeros_like(a)
    v = np.zeros_like(a)

    for it in range(1, max_iter + 1):
        if time.time() >= deadline:
            break

        # linear schedule for λ
        lam = lam_start + (lam_end - lam_start) * min(1.0, it / max_iter)

        grad = smooth_gradient(a, lam)

        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)

        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        step = lr0 * m_hat / (np.sqrt(v_hat) + eps)
        a_new = a - step

        if noise_every and (it % noise_every == 0):
            a_new += np.random.randn(a.size) * noise_scale

        a_new = project_to_simplex(a_new)

        cur_sc = compute_score(a_new)
        if cur_sc < best_sc:
            best_sc = cur_sc
            best_a = a_new.copy()

        a = a_new

        if it % 5000 == 0:
            print(f"    AdamSmooth it={it:,d} cur={cur_sc:.6f} best={best_sc:.6f}")

    return best_a, best_sc


def subgradient_descent_exact(initial_a: np.ndarray,
                              deadline: float,
                              max_iter: int = 120_000,
                              step0: float = 0.03) -> np.ndarray:
    """
    Sub‑gradient descent on the true objective
        f(a) = 2·n·max(conv(a,a)).
    Step size ηₜ = step0 / sqrt(t).  Each iterate is projected onto the
    simplex.
    """
    a = project_to_simplex(initial_a.copy())
    best_a = a.copy()
    best_sc = compute_score(a)
    n = a.shape[0]
    L = 2 * n - 1

    for it in range(1, max_iter + 1):
        if time.time() >= deadline:
            break

        conv = convolution_fft(a, a)
        max_val = np.max(conv)

        # tolerance to collect all (approximately) active constraints
        tol = max_val * 1e-12 + 1e-14
        mask = (np.abs(conv - max_val) <= tol).astype(float)

        # weighted sub‑gradient (see smooth case with λ→∞)
        mask_rev = mask[::-1]
        conv_rev = convolution_fft(a, mask_rev)
        weighted_grad = 2.0 * conv_rev[L - 1 - np.arange(n)]

        eta = step0 / np.sqrt(it)
        a = a - eta * weighted_grad
        a = project_to_simplex(a)

        cur_sc = compute_score(a)
        if cur_sc < best_sc:
            best_sc = cur_sc
            best_a = a.copy()

        if it % 5000 == 0:
            print(f"    SubGrad it={it:,d} cur={cur_sc:.6f} best={best_sc:.6f}")

    return best_a


def simulated_annealing(a: np.ndarray,
                        rng: np.random.Generator,
                        deadline: float,
                        max_iters: int = 80_000,
                        init_temp: float = 0.015,
                        decay: float = 0.9999) -> np.ndarray:
    """
    Very cheap Metropolis‑style annealing using the fast surrogate score.
    A fraction (≤20 %) of the mass at a random source index is moved to a
    random destination index.
    """
    cur = a.copy()
    cur_sc = compute_score(cur)
    best = cur.copy()
    best_sc = cur_sc
    n = cur.size
    temp = init_temp
    it = 0

    while it < max_iters and time.time() < deadline:
        it += 1
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue

        # choose a small chunk of mass from i
        delta = cur[i] * 0.2 * rng.random()
        if delta <= 1e-15:
            continue

        cand = cur.copy()
        cand[i] -= delta
        cand[j] += delta

        cand_sc = compute_score(cand)

        # Metropolis acceptance
        if cand_sc < cur_sc or rng.random() < np.exp(-(cand_sc - cur_sc) / max(temp, 1e-12)):
            cur = cand
            cur_sc = cand_sc
            if cur_sc < best_sc:
                best = cur.copy()
                best_sc = cur_sc

        temp = max(temp * decay, 1e-8)

        if it % 5000 == 0:
            print(f"    SA it={it:,d} cur={cur_sc:.6f} best={best_sc:.6f}")

    return best

def propose_candidate(seed: int = 42,
                     budget_s: int = 1000,
                     **kwargs) -> List[float]:
    """
    Search for a non‑negative step sequence with a low ``evaluate_sequence``
    value.  The routine combines a modest random sampling of parametrised
    families with a heavy local optimisation (Adam → sub‑gradient → SA).
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 2.0      # safety margin for final eval

    best_seq: List[float] = []
    best_val: float = float("inf")

    # ------------------------------------------------------------------
    # 0.  Optional fallback from a previous run
    # ------------------------------------------------------------------
    if "height_sequence_1" in globals():
        try:
            fallback = [float(x) for x in globals()["height_sequence_1"]]
            val = evaluate_sequence(fallback)
            print(f"Fallback (height_sequence_1) score = {val:.6f}")
            if val < best_val:
                best_val = val
                best_seq = fallback
        except Exception:
            pass

    # ------------------------------------------------------------------
    # 1.  Choose a length – larger n gives a smaller theoretical bound.
    # ------------------------------------------------------------------
    # 4096 is a good compromise between quality and speed.
    n = 4096
    print(f"Using vector length n = {n}")

    # ------------------------------------------------------------------
    # 2.  Generate a few parametric seeds
    # ------------------------------------------------------------------
    seed_vectors = []

    # deterministic baselines
    seed_vectors.append(arcsine_shape(n, eps=0.25))
    seed_vectors.append(arcsine_shape(n, eps=1.0))
    seed_vectors.append(mixture_shape(n, weight_arc=0.6))

    # randomised families
    for _ in range(7):
        r = rng.random()
        if r < 0.35:               # beta family
            alpha = rng.uniform(0.1, 0.9)
            beta = rng.uniform(0.1, 0.9)
            eps = rng.uniform(1e-6, 0.5)
            cand = beta_shape(n, alpha, beta, eps)
        elif r < 0.65:             # power / generalized arcsine
            p = rng.uniform(0.5, 2.0)
            eps = rng.uniform(0.0, 1.0)
            # power shape = ((i+0.5)*(n-0.5-i)+eps)^{-p}
            i = np.arange(n, dtype=float)
            denom = ((i + 0.5) * (n - 0.5 - i) + eps) ** p
            cand = 1.0 / denom
            cand /= cand.sum()
        elif r < 0.80:             # triangular bump
            peak = rng.integers(0, n)
            cand = triangular_shape(n, int(peak))
        else:                      # full Dirichlet sample
            cand = random_dirichlet(n, rng)
        seed_vectors.append(cand)

    # ------------------------------------------------------------------
    # 3.  Keep the best few seeds according to the fast surrogate
    # ------------------------------------------------------------------
    top_k = 5
    scored_seeds: List[Tuple[float, np.ndarray]] = []
    for vec in seed_vectors:
        sc = compute_score(vec)
        scored_seeds.append((sc, vec))

    scored_seeds.sort(key=lambda x: x[0])
    selected = [v for _, v in scored_seeds[:top_k]]
    print(f"Selected {len(selected)} seeds for refinement (best fast score = {scored_seeds[0][0]:.6f})")

    # ------------------------------------------------------------------
    # 4.  Refine each selected seed
    # ------------------------------------------------------------------
    for idx, seed_vec in enumerate(selected, 1):
        if time.time() >= deadline - 5.0:
            break
        print(f"\n--- Refining seed {idx}/{len(selected)} ---")

        # Adam (smooth surrogate)
        remaining = deadline - time.time()
        # Rough estimate: ~0.001 s per iteration for n=4096 → allocate up to 100 k it.
        max_iter_adam = int(min(150_000, remaining * 180))
        print(f"    Adam (smooth) max_iter={max_iter_adam}")
        refined, _ = adam_optimize_smooth(
            initial_a=seed_vec,
            deadline=deadline,
            max_iter=max_iter_adam,
            lr0=0.13,
            lam_start=2.0,
            lam_end=180.0,
            noise_every=3000,
            noise_scale=2e-5,
        )

        # Sub‑gradient on the exact objective
        if time.time() < deadline - 5.0:
            remaining = deadline - time.time()
            max_iter_sub = int(min(120_000, remaining * 150))
            print(f"    Sub‑gradient descent max_iter={max_iter_sub}")
            refined = subgradient_descent_exact(
                refined,
                deadline,
                max_iter=max_iter_sub,
                step0=0.025,
            )

        # Simulated annealing polishing
        if time.time() < deadline - 5.0:
            remaining = deadline - time.time()
            max_iter_sa = int(min(100_000, remaining * 130))
            print(f"    Simulated annealing max_iter={max_iter_sa}")
            refined = simulated_annealing(
                refined,
                rng,
                deadline,
                max_iters=max_iter_sa,
                init_temp=0.010,
                decay=0.99995,
            )

        # Official evaluation
        try:
            val = evaluate_sequence(refined.tolist())
            print(f"    Official score = {val:.6f}")
            if val < best_val:
                best_val = val
                best_seq = refined.tolist()
                print(f"    *** New global best: {best_val:.6f} ***")
        except Exception as exc:
            print(f"    Evaluation failed: {exc}")

    # ------------------------------------------------------------------
    # 5.  Final global polishing (if we have a candidate)
    # ------------------------------------------------------------------
    if best_seq and time.time() < deadline - 3.0:
        print("\n--- Final global polishing ---")
        a = np.array(best_seq, dtype=float)

        remaining = deadline - time.time()
        max_iter_sub = int(min(150_000, remaining * 140))
        a = subgradient_descent_exact(a, deadline, max_iter=max_iter_sub, step0=0.02)

        remaining = deadline - time.time()
        max_iter_adam = int(min(150_000, remaining * 140))
        a, _ = adam_optimize_smooth(
            a,
            deadline,
            max_iter=max_iter_adam,
            lr0=0.10,
            lam_start=2.0,
            lam_end=150.0,
            noise_every=2500,
            noise_scale=1e-5,
        )

        final_val = evaluate_sequence(a.tolist())
        print(f"Final polished official score = {final_val:.6f}")
        if final_val < best_val:
            best_val = final_val
            best_seq = a.tolist()
            print("*** Updated global best after final polish ***")

    # ------------------------------------------------------------------
    # 6.  Guard – guarantee a return value
    # ------------------------------------------------------------------
    if not best_seq:
        # fallback to a simple arcsine shape (always feasible)
        best_seq = arcsine_shape(n, eps=0.25).tolist()
        best_val = evaluate_sequence(best_seq)
        print(f"Fallback arcsine final score = {best_val:.6f}")

    print(f"\nSearch finished. Best reported official score = {best_val:.6f}")
    return best_seq
