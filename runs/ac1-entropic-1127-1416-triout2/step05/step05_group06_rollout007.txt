"""
Hybrid multi‑length, multi‑family search for a low‑max‑convolution step
function.

*  A set of diverse analytic “seed” families is generated for each
   candidate length (powers of two up to 8192).  Families include the
   classic arcsine, several window functions (Hann, Hamming, Blackman),
   Beta‑like shapes with random exponents, Dirichlet draws and simple
   mixtures.

*  All seeds are ranked with a fast “scale‑invariant” score
   2·n·max(a∗a)/(sum(a)²).  The five best seeds for a given length are
   refined.

*  Refinement consists of three stages, each respecting the 1000 s wall
   clock:

    1.  Light simulated‑annealing that moves a small amount of mass
        between random bins while keeping the vector on the simplex.
    2.  Smooth‑max gradient descent on the log‑sum‑exp surrogate
        Φβ(a) = (1/β)·log Σ exp(β·(a∗a)k) with an exponentially growing
        β‑schedule.  An exponentiated‑gradient update together with a
        simplex renormalisation is used – this is cheap (FFT‑based) and
        converges quickly.
    3.  Final polish by exponentiated sub‑gradient descent on the true
        max‑convolution (the exact nonsmooth objective).

*  After each refinement the *official* `evaluate_sequence` is called;
   the best sequence found so far is stored.  The routine always returns
   a feasible (non‑negative) list of floats before the time budget
   expires.

The search explores larger lengths (up to 8192) where the discrete
approximation of the optimal continuous distribution can be finer,
while keeping the total number of expensive gradient steps modest.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Basic numeric utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full linear convolution using a real FFT (zero‑padded to a power of two)."""
    m = len(a) + len(b) - 1
    n_fft = next_pow2(m)
    A = np.fft.rfft(a, n=n_fft)
    B = np.fft.rfft(b, n=n_fft)
    C = A * B
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


def compute_score(a: np.ndarray) -> float:
    """
    Fast, scale‑invariant version of the official objective:
        2·n·max(a∗a) / (sum(a)²)
    The vector a is assumed non‑negative.
    """
    n = len(a)
    conv = convolution_fft(a, a)
    max_val = float(np.max(conv))
    s = float(np.sum(a))
    if s < 1e-12:
        return float("inf")
    return 2.0 * n * max_val / (s * s)


# ----------------------------------------------------------------------
# Families of analytic seeds
# ----------------------------------------------------------------------


def arcsine_initial(n: int) -> np.ndarray:
    """Discrete arcsine density (normalised to sum = 1)."""
    i = np.arange(n, dtype=float)
    eps = 0.25
    vals = 1.0 / np.sqrt((i + 0.5) * (n - 0.5 - i) + eps)
    return vals / vals.sum()


def generate_beta_like(n: int, alpha: float, beta: float) -> np.ndarray:
    """Beta‑like shape a_i ∝ (i+½)^{α‑1} (n‑1‑i+½)^{β‑1}."""
    i = np.arange(n, dtype=float)
    base = (i + 0.5) ** (alpha - 1) * (n - 1 - i + 0.5) ** (beta - 1)
    total = base.sum()
    if total == 0.0:
        return np.full(n, 1.0 / n)
    return base / total


def random_dirichlet(n: int, concentration: float, rng: np.random.Generator) -> np.ndarray:
    """Draw a Dirichlet‑like vector (Gamma draws) and normalise."""
    if concentration <= 0.0:
        # heavy‑tailed exponential‑like draws
        samp = rng.exponential(1.0, size=n) ** (1.0 / max(0.1, concentration))
    else:
        samp = rng.gamma(concentration, 1.0, size=n)
    samp = np.maximum(samp, 1e-12)
    return samp / samp.sum()


def hann_window(n: int) -> np.ndarray:
    """Classic Hann (raised‑cosine) window, normalised to sum = 1."""
    i = np.arange(n, dtype=float)
    w = 0.5 - 0.5 * np.cos(2.0 * np.pi * i / (n - 1))
    return w / w.sum()


def hamming_window(n: int) -> np.ndarray:
    """Hamming window, normalised to sum = 1."""
    i = np.arange(n, dtype=float)
    w = 0.54 - 0.46 * np.cos(2.0 * np.pi * i / (n - 1))
    return w / w.sum()


def blackman_window(n: int) -> np.ndarray:
    """Blackman window, normalised to sum = 1."""
    i = np.arange(n, dtype=float)
    w = (
        0.42
        - 0.5 * np.cos(2.0 * np.pi * i / (n - 1))
        + 0.08 * np.cos(4.0 * np.pi * i / (n - 1))
    )
    return w / w.sum()


# ----------------------------------------------------------------------
# Simulated annealing (cheap local improvement)
# ----------------------------------------------------------------------


def propose_neighbor(a: np.ndarray, rng: np.random.Generator, max_step: float = 0.02) -> np.ndarray:
    """Move a small amount of mass from one random bin to another."""
    n = len(a)
    i, j = rng.integers(0, n, size=2)
    if i == j or a[i] == 0.0:
        return a
    delta = a[i] * rng.random() * max_step
    b = a.copy()
    b[i] -= delta
    b[j] += delta
    return b


def simulated_annealing(
    a0: np.ndarray,
    deadline: float,
    rng: np.random.Generator,
    max_iter: int = 2000,
    start_temp: float = 0.005,
    cooling: float = 0.9995,
    max_step: float = 0.02,
) -> tuple[np.ndarray, float]:
    """Very light SA that stays on the probability simplex."""
    a = a0.copy()
    cur_sc = compute_score(a)
    best_a = a.copy()
    best_sc = cur_sc
    temp = start_temp

    for _ in range(max_iter):
        if time.time() >= deadline:
            break
        cand = propose_neighbor(a, rng, max_step)
        cand_sc = compute_score(cand)
        delta = cand_sc - cur_sc
        if delta < 0.0 or rng.random() < np.exp(-delta / max(1e-12, temp)):
            a = cand
            cur_sc = cand_sc
            if cur_sc < best_sc:
                best_sc = cur_sc
                best_a = a.copy()
        temp *= cooling
    return best_a, best_sc


# ----------------------------------------------------------------------
# Smooth‑max gradient descent (log‑sum‑exp surrogate)
# ----------------------------------------------------------------------


def softmax_gradient_descent(
    initial_a: np.ndarray,
    deadline: float,
    betas: list[float],
    iter_per_beta: int = 500,
    eta0: float = 0.5,
) -> tuple[np.ndarray, float]:
    """
    Minimise Φβ(a) = (1/β)·log Σ exp(β·(a∗a)k) using an exponentiated‑gradient
    update and a simplex projection (simple renormalisation).
    """
    a = np.maximum(initial_a, 1e-30)
    a = a / a.sum()
    best_a = a.copy()
    best_sc = compute_score(a)

    for beta in betas:
        for it in range(1, iter_per_beta + 1):
            if time.time() >= deadline:
                return best_a, best_sc

            # full auto‑convolution
            conv = convolution_fft(a, a)          # length 2n‑1

            # soft‑max weights (stable)
            m = conv.max()
            exp_vals = np.exp(beta * (conv - m))
            p = exp_vals / exp_vals.sum()         # Σ p = 1

            # cross‑correlation p ⋆ a (linear, length n)
            N = next_pow2(len(p) + len(a) - 1)
            P = np.fft.fft(p, N)
            A = np.fft.fft(a, N)
            cross = np.fft.ifft(P * np.conj(A)).real[: len(a)]

            # gradient of the surrogate (non‑negative)
            grad = 4.0 * len(a) * cross

            # exponentiated‑gradient step
            eta = eta0 / np.sqrt(it)
            a = a * np.exp(-eta * grad)

            # projection onto the simplex (simple renorm)
            s = a.sum()
            if s == 0.0:
                a = np.full_like(a, 1.0 / len(a))
            else:
                a = a / s

            cur_sc = compute_score(a)
            if cur_sc < best_sc:
                best_sc = cur_sc
                best_a = a.copy()

    return best_a, best_sc


# ----------------------------------------------------------------------
# Exact sub‑gradient on the true max‑convolution (final polish)
# ----------------------------------------------------------------------


def _max_conv_gradient(a: np.ndarray, conv: np.ndarray) -> np.ndarray:
    """
    Sub‑gradient of max_k (a∗a)_k .
    For each active offset k the gradient contribution is 2·a_{k−j}.
    If several offsets share the maximal value they are averaged.
    """
    n = len(a)
    max_val = conv.max()
    mask = np.isclose(conv, max_val, atol=1e-15, rtol=0.0)
    active = np.where(mask)[0]
    grad = np.zeros_like(a)

    for k in active:
        lo = max(0, k - (n - 1))
        hi = min(k, n - 1)
        idx = np.arange(lo, hi + 1, dtype=int)
        grad[idx] += 2.0 * a[k - idx]

    if len(active) > 1:
        grad /= len(active)
    return grad


def optimise_subgradient(
    initial_a: np.ndarray,
    deadline: float,
    *,
    eta0: float = 0.6,
    max_iter: int = 200_000,
    noise_every: int = 0,
    noise_scale: float = 0.0,
) -> tuple[np.ndarray, float]:
    """
    Exponentiated sub‑gradient descent directly on Φ(a)=max_k (a∗a)_k.
    Returns the best vector seen (as a NumPy array) and its fast score.
    """
    a = np.maximum(initial_a, 1e-30)
    a = a / a.sum()
    best_a = a.copy()
    best_sc = compute_score(a)
    n = len(a)

    for it in range(1, max_iter + 1):
        if time.time() >= deadline:
            break

        conv = convolution_fft(a, a)
        grad = _max_conv_gradient(a, conv)

        eta = eta0 / np.sqrt(it)
        a = a * np.exp(-eta * grad)

        if noise_every and (it % noise_every == 0):
            a += np.random.randn(n) * noise_scale
            a = np.maximum(a, 0.0)

        s = a.sum()
        a = a / s if s != 0.0 else np.full_like(a, 1.0 / n)

        cur_sc = compute_score(a)
        if cur_sc < best_sc:
            best_sc = cur_sc
            best_a = a.copy()

        if it % 5000 == 0:
            print(f"    subgrad iter={it:,d} cur={cur_sc:.6f} best={best_sc:.6f}")

    return best_a, best_sc

# Main entry point required by the competition harness

def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function with a low
    ``evaluate_sequence`` value.  The routine respects a wall‑clock budget
    (default 1000 s) and always returns a feasible list of floats.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    # keep a small safety margin for the final call to evaluate_sequence
    deadline = start_time + budget_s - 2.0

    best_seq: list[float] = []
    best_val: float = float("inf")

    # ------------------------------------------------------------------
    # 0.  Fallback – reuse a previously discovered construction if present.
    # ------------------------------------------------------------------
    try:
        fallback = [float(x) for x in height_sequence_1]  # type: ignore
        val = evaluate_sequence(fallback)
        print(f"Fallback (height_sequence_1) score = {val:.6f}")
        if val < best_val:
            best_val = val
            best_seq = fallback
    except NameError:
        pass

    # ------------------------------------------------------------------
    # 1.  Explore a handful of lengths (powers of two are FFT‑friendly).
    # ------------------------------------------------------------------
    lengths = [8192, 4096, 2048, 1024]

    for n in lengths:
        if time.time() >= deadline - 5.0:
            break
        print(f"\n=== Exploring length n = {n} ===")

        # --------------------------------------------------------------
        # 1a. Build a diverse library of seed vectors.
        # --------------------------------------------------------------
        seeds: list[np.ndarray] = []

        # uniform
        seeds.append(np.full(n, 1.0 / n))

        # classic arcsine
        seeds.append(arcsine_initial(n))

        # window families
        seeds.append(hann_window(n))
        seeds.append(hamming_window(n))
        seeds.append(blackman_window(n))

        # random Beta‑like shapes (edge‑heavy and interior‑heavy)
        for _ in range(30):
            alpha = rng.uniform(0.15, 1.4)   # α<1 → edges, α>1 → centre
            beta = rng.uniform(0.15, 1.4)
            seeds.append(generate_beta_like(n, alpha, beta))

        # random Dirichlet draws (various concentrations)
        for conc in [0.2, 0.5, 1.0, 2.0]:
            seeds.append(random_dirichlet(n, conc, rng))

        # simple mixtures of arcsine and uniform
        for _ in range(5):
            w = rng.random()
            mix = w * arcsine_initial(n) + (1.0 - w) * np.full(n, 1.0 / n)
            seeds.append(mix / mix.sum())

        # --------------------------------------------------------------
        # 1b. Quick ranking with the cheap internal score.
        # --------------------------------------------------------------
        scored = [(compute_score(c), c) for c in seeds]
        scored.sort(key=lambda x: x[0])
        top_candidates = [c for _, c in scored[:5]]  # keep the five most promising

        # --------------------------------------------------------------
        # 1c. Refine each retained candidate.
        # --------------------------------------------------------------
        for idx, cand in enumerate(top_candidates):
            if time.time() >= deadline - 4.0:
                break
            print(f"  candidate {idx + 1}/{len(top_candidates)} (n={n})")

            # ---- Simulated annealing -------------------------------------------------
            max_iter_sa = int(min(4000, (deadline - time.time()) * 350))
            start_temp = max(0.005, 0.02 * compute_score(cand))
            sa_a, _ = simulated_annealing(
                cand,
                deadline,
                rng,
                max_iter=max_iter_sa,
                start_temp=start_temp,
                cooling=0.9995,
                max_step=0.08,
            )
            val_sa = evaluate_sequence(sa_a.tolist())
            print(f"    SA evaluated score = {val_sa:.6f}")
            if val_sa < best_val:
                best_val = val_sa
                best_seq = sa_a.tolist()
                print(f"    *** New global best (SA) {best_val:.6f} ***")

            # ---- Smooth‑max gradient descent ----------------------------------------
            betas = [2, 5, 10, 20, 40, 80, 160]
            # allocate iterations per β depending on remaining time
            remaining = max(0.0, deadline - time.time())
            it_per_beta = max(100, int(min(1200, remaining * 250) / len(betas)))
            gd_a, _ = softmax_gradient_descent(
                sa_a,
                deadline,
                betas=betas,
                iter_per_beta=it_per_beta,
                eta0=0.45,
            )
            val_gd = evaluate_sequence(gd_a.tolist())
            print(f"    smooth‑gd evaluated score = {val_gd:.6f}")
            if val_gd < best_val:
                best_val = val_gd
                best_seq = gd_a.tolist()
                print(f"    *** New global best (smooth‑gd) {best_val:.6f} ***")

    # ------------------------------------------------------------------
    # 2.  Final polishing of the overall best (if time permits).
    # ------------------------------------------------------------------
    if best_seq and time.time() < deadline - 5.0:
        print("\nFinal polishing with max‑convolution sub‑gradient")
        a = np.array(best_seq, dtype=float)
        remaining = max(0.0, deadline - time.time())
        max_iter = int(min(200_000, remaining * 300))
        a_refined, _ = optimise_subgradient(
            initial_a=a,
            deadline=deadline,
            eta0=0.3,
            max_iter=max_iter,
            noise_every=0,
            noise_scale=0.0,
        )
        final_val = evaluate_sequence(a_refined.tolist())
        print(f"Final polished score = {final_val:.6f}")
        if final_val < best_val:
            best_val = final_val
            best_seq = a_refined.tolist()
            print("*** Updated global best after final polishing ***")

    # ------------------------------------------------------------------
    # 3.  Guard – always return a feasible sequence.
    # ------------------------------------------------------------------
    if not best_seq:
        best_seq = [1.0] * 256
        best_val = evaluate_sequence(best_seq)

    print(f"\nSearch finished. Best reported score = {best_val:.6f}")
    return best_seq
