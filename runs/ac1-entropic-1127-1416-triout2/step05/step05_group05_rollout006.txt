def refine_smooth_dynamic(a, deadline, eta0=1.0, beta_start=2.0, beta_end=30.0, momentum=0.9, report_every=0):
   a = normalize(a.astype(float).copy())
   n = len(a)
   best_a = a.copy()
   best_score = 2.0 * n * np.max(convolution_fft(a, a))
   it = 0
   m = np.zeros_like(a)
   start = time.time()
   while time.time() < deadline:
       it += 1
       conv = convolution_fft(a, a)
       max_val = float(np.max(conv))
       # compute beta based on elapsed fraction
       elapsed = time.time() - start
       total = deadline - start
       frac = elapsed / total if total > 0 else 0.0
       beta = beta_start + (beta_end - beta_start) * frac

       # softmax weights
       exp_vals = np.exp(beta * (conv - max_val))
       w = exp_vals / np.sum(exp_vals)

       grad = gradient_from_weights(a, w)

       # momentum update
       m = momentum * m + (1 - momentum) * grad

       # step size
       eta = eta0 / np.sqrt(it)

       a = a * np.exp(-eta * m)
       a = normalize(a)

       cur_score = 2.0 * n * max_val
       if cur_score < best_score:
           best_score = cur_score
           best_a = a.copy()
   return best_a, best_score

"""
Optimisation of a non‑negative step‑function for the objective

    2·n·max(conv(a,a)) / (sum(a)²)

The algorithm works in three phases within the given time budget.

1.  **Candidate generation** – Several parametric families (arcsine,
    beta‑distribution, power‑law, uniform and random Dirichlet) are
    sampled.  The best candidate from this cheap sweep (using the fast
    FFT‑based score) becomes the seed for a more serious optimisation.

2.  **Dynamic smooth mirror‑descent** – An exponentiated‑gradient update
    on the probability simplex using a temperature‑schedule for the
    soft‑max weights (β grows from a small to a large value).  A momentum
    term and an explicit symmetry projection (a←(a+a[::-1])/2) keep the
    iterates well‑behaved and exploit the symmetry of the problem.

3.  **Hard sub‑gradient polishing** – The final candidate is refined with
    a classic sub‑gradient method that uses the true maximising indices
    of the convolution.  The last iterate with the smallest exact
    objective (computed via ``evaluate_sequence``) is returned.

The whole search respects the time limit, tries several random restarts
and always falls back to a feasible uniform vector if nothing else is
found.
"""

import time
import numpy as np
from typing import List, Tuple

# ----------------------------------------------------------------------
# Low‑level numeric utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """
    Linear (full) convolution via real FFT.
    """
    m = len(a) + len(b) - 1
    n_fft = next_pow2(m)
    A = np.fft.rfft(a, n=n_fft)
    B = np.fft.rfft(b, n=n_fft)
    C = A * B
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


def normalize(v: np.ndarray) -> np.ndarray:
    """Project a non‑negative vector onto the simplex (sum = 1)."""
    total = v.sum()
    if total <= 0.0:
        return np.full_like(v, 1.0 / len(v))
    return v / total


def symmetrize(v: np.ndarray) -> np.ndarray:
    """Force symmetry: v_i = v_{n-1-i}."""
    return (v + v[::-1]) / 2.0


def compute_score(a: np.ndarray) -> float:
    """
    Fast evaluation of the official objective (scale‑invariant):
        2·n·max(conv(a,a)) / (sum(a)²)
    The vector a is assumed non‑negative and (approximately) on the simplex.
    """
    n = len(a)
    conv = convolution_fft(a, a)
    max_val = float(np.max(conv))
    s = float(np.sum(a))
    if s < 1e-12:
        return float("inf")
    return 2.0 * n * max_val / (s * s)


def gradient_from_weights(a: np.ndarray, w: np.ndarray) -> np.ndarray:
    """
    Gradient of Σ_k w_k·(a*a)_k with respect to a.
    Implements   g_j = 2· Σ_m a_m · w_{m+j}.
    Computed via one FFT‑based convolution.
    """
    w_rev = w[::-1]
    conv = convolution_fft(a, w_rev)               # length = 3·n‑2
    idx = 2 * len(a) - 2                            # position 2n‑2
    grads = 2.0 * conv[idx - np.arange(len(a))]
    return grads


# ----------------------------------------------------------------------
# Parametric families for the initial sweep
# ----------------------------------------------------------------------


def arcsine_initial(n: int) -> np.ndarray:
    """Discrete arcsine distribution (≈ 1/√(i·(n‑1‑i))) normalised."""
    i = np.arange(n, dtype=float)
    eps = 0.25
    vals = 1.0 / np.sqrt((i + 0.5) * (n - 0.5 - i) + eps)
    return normalize(vals)


def powerlaw_shape(n: int, p: float, eps: float) -> np.ndarray:
    """Symmetric power‑law shape a_i ∝ ((i+ε)(n‑1‑i+ε))⁻ᵖ."""
    i = np.arange(n, dtype=float)
    base = ((i + eps) * (n - 1 - i + eps)) ** (-p)
    return normalize(base)


def beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    """
    Discretised symmetric Beta distribution.
    a_i ∝ x^{α‑1} (1‑x)^{β‑1} with x = i/(n‑1).
    """
    i = np.arange(n, dtype=float)
    x = i / (n - 1)
    eps = 1e-12
    # avoid division by zero when α or β < 1
    vals = (x + eps) ** (alpha - 1) * ((1 - x) + eps) ** (beta - 1)
    # replace possible infinities by zero
    vals = np.where(np.isfinite(vals), vals, 0.0)
    return normalize(vals)


def uniform_shape(n: int) -> np.ndarray:
    """Uniform probability vector on n points."""
    return np.full(n, 1.0 / n)


def random_dirichlet(n: int, rng: np.random.Generator) -> np.ndarray:
    """Draw a random point on the simplex (exponential sampling)."""
    samp = rng.exponential(1.0, size=n)
    return normalize(samp)


# ----------------------------------------------------------------------
# Refinement methods
# ----------------------------------------------------------------------


def refine_smooth_dynamic(
    a: np.ndarray,
    deadline: float,
    *,
    eta0: float = 1.0,
    beta_start: float = 2.0,
    beta_end: float = 40.0,
    momentum: float = 0.9,
    report_every: int = 0,
) -> Tuple[np.ndarray, float]:
    """
    Exponentiated‑gradient mirror descent with a *growing* soft‑max
    temperature.  A momentum term smooths the updates and a symmetry
    projection keeps the iterate within the natural sub‑space.
    Returns the best iterate (evaluated with the exact objective) and its
    score.
    """
    a = normalize(a.astype(float).copy())
    a = symmetrize(a)
    n = len(a)
    best_a = a.copy()
    conv = convolution_fft(a, a)
    best_score = 2.0 * n * float(np.max(conv))

    it = 0
    m = np.zeros_like(a)  # momentum accumulator
    start = time.time()

    while time.time() < deadline:
        it += 1

        conv = convolution_fft(a, a)
        max_val = float(np.max(conv))

        # temperature schedule (linear in elapsed fraction)
        elapsed = time.time() - start
        total = deadline - start
        frac = elapsed / total if total > 0 else 0.0
        beta = beta_start + (beta_end - beta_start) * frac

        # soft‑max weights on the convolution entries
        exp_vals = np.exp(beta * (conv - max_val))
        w = exp_vals / np.sum(exp_vals)   # ∑ w = 1

        # gradient of the smoothed objective
        grad = gradient_from_weights(a, w)

        # momentum update
        m = momentum * m + (1.0 - momentum) * grad

        # step size schedule
        eta = eta0 / np.sqrt(it)

        # multiplicative (exponentiated‑gradient) update
        a = a * np.exp(-eta * m)

        # enforce non‑negativity, symmetry and simplex constraints
        a = symmetrize(a)
        a = normalize(a)

        cur_score = 2.0 * n * max_val
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()

        if report_every and (it % report_every == 0):
            print(f"    smooth it={it:,d} cur={cur_score:.6f} best={best_score:.6f}")

    return best_a, best_score


def refine_subgradient(
    a: np.ndarray,
    deadline: float,
    *,
    max_iter: int = 30_000,
    eta0: float = 0.7,
    report_every: int = 0,
) -> Tuple[np.ndarray, float]:
    """
    Classical sub‑gradient descent on the true max‑convolution objective.
    Uses multiplicative updates, a 1/√t step‑size and a symmetry
    projection.  Returns the best vector seen and its exact score.
    """
    a = normalize(a.astype(float).copy())
    a = symmetrize(a)
    n = len(a)

    conv = convolution_fft(a, a)
    best_score = 2.0 * n * float(np.max(conv))
    best_a = a.copy()

    for it in range(1, max_iter + 1):
        if time.time() >= deadline:
            break

        conv = convolution_fft(a, a)
        max_val = float(np.max(conv))

        # uniform weighting over (near) maximisers
        w = np.zeros_like(conv)
        max_idx = np.where(conv >= max_val - 1e-14)[0]
        w[max_idx] = 1.0 / len(max_idx)

        grad = gradient_from_weights(a, w)

        eta = eta0 / np.sqrt(it)
        a = a * np.exp(-eta * grad)

        # keep feasible
        a = symmetrize(a)
        a = normalize(a)

        cur_score = 2.0 * n * max_val
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()

        if report_every and (it % report_every == 0):
            print(f"    sub‑grad it={it:,d} cur={cur_score:.6f} best={best_score:.6f}")

    return best_a, best_score

            elif typ == "beta":
                a = rng.uniform(0.1, 1.0)
                b = rng.uniform(0.1, 1.0)
                seed_vec = beta_shape(n, a, b)
            else:  # dirichlet
                seed_vec = random_dirichlet(n, rng)

            # small random perturbation for diversity
            seed_vec = seed_vec + rng.normal(scale=0.01, size=n)
            seed_vec = np.maximum(seed_vec, 0.0)
            seed_vec = normalize(seed_vec)

        # smooth refinement
        smooth_vec, _ = refine_smooth_dynamic(
            a=seed_vec,
            deadline=per_restart_deadline,
            eta0=1.0,
            beta_start=2.0,
            beta_end=40.0,
            momentum=0.9,
            report_every=0,
        )

        # official evaluation of the smoothed vector
        val = evaluate_sequence(smooth_vec.tolist())
        print(f"[restart {restart_counter}] official score after smooth = {val:.6f}")

        if val < best_val:
            best_val = val
            best_seq = smooth_vec.tolist()
            print("*** New global best after smooth refinement ***")

        restart_counter += 1

    # ------------------------------------------------------------------
    # 4.  Hard sub‑gradient polishing using the remaining budget
    # ------------------------------------------------------------------
    time_left = deadline - time.time()
    if best_seq and time_left > 5.0:
        print("\nStarting hard sub‑gradient polishing")
        a_start = np.array(best_seq, dtype=float)
        polished_vec, _ = refine_subgradient(
            a=a_start,
            deadline=deadline,
            max_iter=200_000,
            eta0=0.7,
            report_every=0,
        )
        final_val = evaluate_sequence(polished_vec.tolist())
        print(f"Polished official score = {final_val:.6f}")

        if final_val < best_val:
            best_val = final_val
            best_seq = polished_vec.tolist()
            print("*** Updated global best after hard polish ***")

    # ------------------------------------------------------------------
    # 5.  Fallback – always return a feasible sequence.
    # ------------------------------------------------------------------
    if not best_seq:
        # very unlikely, but guarantees a valid return value
        best_seq = uniform_shape(256).tolist()
        best_val = evaluate_sequence(best_seq)
        print("[fallback] returning uniform sequence")

    print(f"\nSearch finished. Best reported score = {best_val:.6f}")
    return best_seq
