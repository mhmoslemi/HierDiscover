from scipy.optimize import linprog

def solve_lp(b):
    n = len(b)
    m = 2*n - 1
    # Build A_ub shape (m+1,n+1): first m constraints M @ a - t <= 0
    M = np.zeros((m, n))
    for i in range(n):
        M[i:i+n, i] = b
    A_ub = np.hstack([M, -np.ones((m, 1))])
    b_ub = np.zeros(m)
    # equality: sum a = 1
    A_eq = np.hstack([np.ones((1, n)), np.zeros((1, 1))])
    b_eq = np.array([1.0])
    # bounds: a_i >=0, t free maybe >=0
    bounds = [(0, None)] * n + [(0, None)]  # t >=0
    # objective: minimize t => c = [0...0, 1]
    c = np.zeros(n+1)
    c[-1] = 1.0
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                  bounds=bounds, method='highs', options={'presolve': True})
    if not res.success:
        return None
    x = res.x[:n]  # a variables
    t_val = res.x[-1]
    return x, t_val
```

def solve_lp_with_b(b, symmetric=False):
    n = len(b)
    m = 2*n - 1
    # construct M matrix
    M = np.zeros((m, n), dtype=np.float64)
    for i in range(n):
        M[i:i+n, i] = b
    # A_ub = [M, -1]
    A_ub = np.hstack([M, -np.ones((m, 1))])
    b_ub = np.zeros(m)
    # equality constraints
    # sum a = 1
    A_eq = np.hstack([np.ones((1, n)), np.zeros((1, 1))])
    b_eq = np.array([1.0])
    if symmetric:
        # add constraints a[i] - a[n-1-i] = 0
        eq_rows = []
        for i in range(n//2):
            row = np.zeros(n+1)
            row[i] = 1
            row[n-1-i] = -1
            eq_rows.append(row)
        if eq_rows:
            A_eq = np.vstack([A_eq] + eq_rows)
            b_eq = np.concatenate([b_eq, np.zeros(len(eq_rows))])
    # variable bounds: a_i >=0, t >=0
    bounds = [(0, None)] * n + [(0, None)]
    c = np.zeros(n+1)
    c[-1] = 1.0
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                  bounds=bounds, method='highs')
    if not res.success:
        return None
    a_sol = res.x[:n]
    t_sol = res.x[-1]
    return a_sol, t_sol
```

"""
Hybrid optimizer for the max‑convolution step‑function problem.

The search proceeds in three stages:

1.  **Diverse seeds** – a handful of analytically‑motivated shapes
    (arcsine, power‑law, beta‑like, Dirichlet) are generated for a
    modest length (1024 or 2048).  All vectors are normalised to sum = 1.

2.  **LP refinement** – for a current vector b we solve the linear
    program

        minimise   t
        subject to M(b)·a ≤ t ,   Σ a = 1 ,   a ≥ 0

    where M(b) encodes the convolution (a∗b).  The solution a reduces
    the maximal *cross*‑convolution with b.  Repeating the LP a few
    times (using the new a as the next b) drives the sequence toward a
    self‑consistent low‑peak convolution.

3.  **Smooth‑max Adam optimisation** – the nonsmooth objective
    ``max_k (a∗a)_k`` is approximated by a log‑sum‑exp surrogate.
    A softmax weight vector p is formed from the current convolution,
    its gradient is obtained via one FFT‑based cross‑correlation and
    Adam is used to update a while projecting onto the probability
    simplex after each step.

A short final polishing with an exponentiated‑sub‑gradient method
further lowers the exact max‑convolution.  The routine respects the
wall‑clock budget (default 1000 s) and always returns a feasible
non‑negative list of floats.
"""

import time
from typing import List, Tuple

import numpy as np
from scipy.optimize import linprog

# ----------------------------------------------------------------------
# Basic numeric utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Linear convolution using real FFT (zero‑padded to a power of two)."""
    m = len(a) + len(b) - 1
    n_fft = next_pow2(m)
    A = np.fft.rfft(a, n=n_fft)
    B = np.fft.rfft(b, n=n_fft)
    C = A * B
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


def compute_score(a: np.ndarray) -> float:
    """
    Fast surrogate of the official objective (scale‑invariant):
        2·n·max(a∗a) / (sum(a)²)
    The vector a is assumed non‑negative and already normalised to sum = 1.
    """
    n = len(a)
    conv = convolution_fft(a, a)
    max_val = float(np.max(conv))
    return 2.0 * n * max_val


# ----------------------------------------------------------------------
# Seed generators
# ----------------------------------------------------------------------


def arcsine_initial(n: int) -> np.ndarray:
    """Discrete arcsine density (normalised to sum = 1)."""
    i = np.arange(n, dtype=float)
    eps = 0.25
    vals = 1.0 / np.sqrt((i + 0.5) * (n - 0.5 - i) + eps)
    return vals / vals.sum()


def generate_powerlaw(n: int, p: float, eps: float) -> np.ndarray:
    """Symmetric power‑law a_i ∝ ((i+ε)(n‑1‑i+ε))⁻ᵖ."""
    i = np.arange(n, dtype=float)
    base = ((i + eps) * (n - 1 - i + eps)) ** (-p)
    total = base.sum()
    if total == 0.0:
        return np.full(n, 1.0 / n)
    return base / total


def generate_beta_like(n: int, alpha: float, beta: float) -> np.ndarray:
    """Beta‑like shape a_i ∝ (i+½)^{α‑1} (n‑1‑i+½)^{β‑1}."""
    i = np.arange(n, dtype=float)
    base = (i + 0.5) ** (alpha - 1) * (n - 1 - i + 0.5) ** (beta - 1)
    total = base.sum()
    if total == 0.0:
        return np.full(n, 1.0 / n)
    return base / total


def random_dirichlet(n: int, concentration: float, rng: np.random.Generator) -> np.ndarray:
    """Sample a Dirichlet‑like vector and normalise."""
    if concentration <= 0:
        samp = rng.exponential(1.0, size=n) ** (1.0 / max(0.1, concentration))
    else:
        samp = rng.gamma(concentration, 1.0, size=n)
    samp += 1e-12
    return samp / samp.sum()


# ----------------------------------------------------------------------
# Linear‑programming refinement (cross‑convolution)
# ----------------------------------------------------------------------


def _build_conv_matrix(b: np.ndarray) -> np.ndarray:
    """
    Build the dense matrix M (shape (2n‑1, n)) such that
    M @ a  ==  a ∗ b   (full linear convolution).
    """
    n = len(b)
    m = 2 * n - 1
    M = np.zeros((m, n), dtype=np.float64)
    for i in range(n):
        M[i:i + n, i] = b
    return M


def solve_lp(b: np.ndarray, symmetric: bool = False) -> Tuple[np.ndarray, float] | None:
    """
    Solve the LP:
        minimise   t
        subject to M(b)·a ≤ t ,   Σ a = 1 ,   a ≥ 0 ,   t ≥ 0 .
    Returns (a_opt, t_opt) or None on failure.
    """
    n = len(b)
    M = _build_conv_matrix(b)                     # (2n‑1) × n
    m = M.shape[0]

    # Inequality: M·a - t ≤ 0
    A_ub = np.hstack([M, -np.ones((m, 1))])
    b_ub = np.zeros(m)

    # Equality: Σ a = 1   (t does not appear)
    A_eq = np.hstack([np.ones((1, n)), np.zeros((1, 1))])
    b_eq = np.array([1.0])

    if symmetric:
        # Add a_i = a_{n‑1‑i}
        rows = []
        for i in range(n // 2):
            row = np.zeros(n + 1)
            row[i] = 1.0
            row[n - 1 - i] = -1.0
            rows.append(row)
        if rows:
            A_eq = np.vstack([A_eq] + rows)
            b_eq = np.concatenate([b_eq, np.zeros(len(rows))])

    # Bounds for a_i and t (all ≥ 0)
    bounds = [(0.0, None)] * n + [(0.0, None)]

    # Objective: minimise t  →  c = [0,…,0,1]
    c = np.zeros(n + 1)
    c[-1] = 1.0

    res = linprog(
        c,
        A_ub=A_ub,
        b_ub=b_ub,
        A_eq=A_eq,
        b_eq=b_eq,
        bounds=bounds,
        method="highs",
        options={"presolve": True},
    )
    if not res.success:
        return None
    a_opt = np.maximum(res.x[:n], 0.0)
    a_opt /= a_opt.sum() if a_opt.sum() > 0 else 1.0
    t_opt = float(res.x[-1])
    return a_opt, t_opt


def lp_refine(initial: np.ndarray, deadline: float, max_iters: int = 6) -> np.ndarray:
    """
    Repeatedly solve the LP using the current vector as the convolution partner.
    Stops early if no improvement or the time budget expires.
    """
    a = initial.copy()
    best_sc = compute_score(a)
    for it in range(max_iters):
        if time.time() >= deadline:
            break
        res = solve_lp(a, symmetric=False)
        if res is None:
            break
        a_new, _ = res
        sc_new = compute_score(a_new)
        if sc_new < best_sc - 1e-12:          # strict improvement
            a = a_new
            best_sc = sc_new
        else:
            break
    return a


# ----------------------------------------------------------------------
# Smooth‑max Adam optimisation
# ----------------------------------------------------------------------


def adam_smooth_opt(initial: np.ndarray, deadline: float, total_iters: int) -> np.ndarray:
    """
    Minimise the soft‑max surrogate Φ_β(a) = (1/β)·log Σ exp(β·(a∗a)_k)
    using Adam with projection onto the simplex after every step.
    """
    a = np.maximum(initial, 1e-30)
    a = a / a.sum()
    n = len(a)

    # Adam state
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    b1 = 0.9
    b2 = 0.999
    eps = 1e-8
    lr = 0.05

    # β schedule – start modest, end very large (close to true max)
    betas = [10, 30, 100, 300, 1000, 3000]
    it_per_beta = max(1, total_iters // len(betas))

    best_a = a.copy()
    best_sc = compute_score(a)

    t_iter = 0
    for beta in betas:
        for _ in range(it_per_beta):
            if time.time() >= deadline:
                return best_a
            # ----- smooth max weight -----
            conv = convolution_fft(a, a)                # length 2n‑1
            max_c = conv.max()
            exp_vals = np.exp(beta * (conv - max_c))
            p = exp_vals / exp_vals.sum()               # Σ p = 1

            # ----- gradient of Φ_β -----
            N = next_pow2(len(p) + n - 1)
            P = np.fft.fft(p, N)
            A = np.fft.fft(a, N)
            cross = np.fft.ifft(P * np.conj(A)).real[:n]   # p ⋆ a
            grad = 4.0 * n * cross

            # ----- Adam update (additive) -----
            t_iter += 1
            m = b1 * m + (1.0 - b1) * grad
            v = b2 * v + (1.0 - b2) * (grad ** 2)
            m_hat = m / (1.0 - b1 ** t_iter)
            v_hat = v / (1.0 - b2 ** t_iter)
            step = lr * m_hat / (np.sqrt(v_hat) + eps)

            a = a - step
            a = np.maximum(a, 0.0)
            s = a.sum()
            if s == 0.0:
                a = np.full_like(a, 1.0 / n)
            else:
                a = a / s

            cur_sc = compute_score(a)
            if cur_sc < best_sc:
                best_sc = cur_sc
                best_a = a.copy()

    return best_a


# ----------------------------------------------------------------------
# Exponentiated sub‑gradient polishing (exact max)
# ----------------------------------------------------------------------


def _max_conv_gradient(a: np.ndarray, conv: np.ndarray) -> np.ndarray:
    """Sub‑gradient of max_k (a∗a)_k (averaged over all active ks)."""
    n = len(a)
    max_val = conv.max()
    mask = np.isclose(conv, max_val, atol=1e-15, rtol=0.0)
    active = np.where(mask)[0]
    grad = np.zeros_like(a)

    for k in active:
        lo = max(0, k - (n - 1))
        hi = min(k, n - 1)
        idx = np.arange(lo, hi + 1, dtype=int)
        grad[idx] += 2.0 * a[k - idx]

    if len(active) > 1:
        grad /= len(active)
    return grad


def subgradient_polish(initial: np.ndarray, deadline: float, max_iter: int = 200_000) -> np.ndarray:
    """Exponentiated sub‑gradient descent directly on the true max‑convolution."""
    a = np.maximum(initial, 1e-30)
    a = a / a.sum()
    n = len(a)

    best_a = a.copy()
    best_sc = compute_score(a)

    for it in range(1, max_iter + 1):
        if time.time() >= deadline:
            break
        conv = convolution_fft(a, a)
        grad = _max_conv_gradient(a, conv)
        eta = 0.6 / np.sqrt(it)
        a = a * np.exp(-eta * grad)

        s = a.sum()
        a = a / s if s != 0.0 else np.full_like(a, 1.0 / n)

        cur_sc = compute_score(a)
        if cur_sc < best_sc:
            best_sc = cur_sc
            best_a = a.copy()
        if it % 5000 == 0:
            print(f"    subgrad iter={it:,d} cur={cur_sc:.6f} best={best_sc:.6f}")

    return best_a
