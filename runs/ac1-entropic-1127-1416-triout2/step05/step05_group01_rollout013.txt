def smooth_gradient_descent(init_a, time_deadline, beta0=1.0, eta0=0.5, beta_inc=1.02, beta_max=200.0):
    a = init_a.copy()
    n = len(a)
    best_a = a.copy()
    best_score = compute_score(a)
    G = np.zeros_like(a)
    beta = beta0
    t = 0
    while time.time() < time_deadline:
        t += 1
        # compute convolution
        conv = convolution_fft(a, a)
        max_conv = float(conv.max())

        # compute softmax weights
        # stable: shift by max
        m = max_conv
        exp_vals = np.exp(beta * (conv - m))
        w = exp_vals / exp_vals.sum()

        # compute gradient using cross-correlation
        w_rev = w[::-1]
        corr = convolution_fft(a, w_rev)  # length = n+ (2n-1)-1
        idx = 2*n - 2 - np.arange(n)
        grad = 2.0 * corr[idx]

        # update AdaGrad
        G += grad**2
        eta = eta0 / (np.sqrt(G) + 1e-8)

        # multiplicative update
        a = a * np.exp(-eta * grad)

        # renorm to sum=1 (simple Euclidean projection onto simplex)
        a = project_to_simplex(a)

        # update best
        cur_score = 2.0 * n * max_conv
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            # optionally small print
            if t % 5000 == 0:
                print(f"Iter {t}: improved score {best_score:.6f}")

        # increase beta gradually
        if beta < beta_max:
            beta = min(beta * beta_inc, beta_max)

        # add jitter occasionally
        if t % 5000 == 0:
            a += np.random.randn(n) * (eta0 * 1e-4 / np.sqrt(t))
            a = project_to_simplex(a)

    return best_a, best_score

"""
Propose a low‑score step function.

Algorithm
---------
1.  Choose a fairly large support (n = 2048).  The objective
    `score = 2·n·max(conv)` (the sum of the sequence is normalised to 1)
    can be made arbitrarily close to 1, but the search space is highly
    non‑convex.

2.  **Smooth weighted‑gradient descent**  
    Instead of using the sub‑gradient of the single maximal convolution
    entry, we use a soft‑max (temperature‑controlled) weighting of *all*
    convolution values:

        w_k ∝ exp(β·(conv_k – max(conv)))

    The gradient of the (smoothed) objective w.r.t. the coefficients is

        g_i = 2·∑_k w_k·a_{k‑i}     (cross‑correlation of a and w)

    The update is performed in the exponentiated (mirror‑descent) domain:

        a_i ← a_i·exp(–η·g_i)   and then projected back onto the simplex.

    The step‑size η follows an AdaGrad schedule
    (`η = η0 / sqrt(∑g²)`) and the temperature β is slowly increased
    (β ← β·β_inc) so that the optimisation starts smooth and ends
    focused on the true ℓ∞‑norm.

3.  Several independent restarts (random Dirichlet draws or the provided
    fallback solution) are run for a fixed fraction of the time budget.
    The best‑scoring vector of all runs is kept.

4.  **Hill‑climbing polish** – a cheap stochastic local search that swaps
    two positions or transfers a fraction of mass between two entries.
    Every move is projected onto the simplex; improvements are kept and
    the step‑size is adapted.

The whole search respects the 1000 s limit; all numbers are non‑negative,
hence the returned sequence can be fed directly to `evaluate_sequence`.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# FFT‑based convolution utilities (fast, linear convolution)
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Linear convolution a*b using a real FFT (fast)."""
    m = len(a) + len(b) - 1
    n_fft = next_pow2(m)
    A = np.fft.rfft(a, n=n_fft)
    B = np.fft.rfft(b, n=n_fft)
    C = A * B
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


# ----------------------------------------------------------------------
# Simplex projection (Euclidean)
# ----------------------------------------------------------------------


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the probability simplex
    (non‑negative entries that sum to 1).
    """
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / v.shape[0])

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, v.shape[0] + 1)
    cond = u - cssv / ind > 0
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


# ----------------------------------------------------------------------
# Scoring (fast replica of evaluate_sequence for sum‑normalised vectors)
# ----------------------------------------------------------------------


def compute_score(a: np.ndarray) -> float:
    """
    Return 2·n·max(a*a) for a vector whose entries sum to 1.
    This equals the value returned by `evaluate_sequence`.
    """
    n = a.shape[0]
    conv = convolution_fft(a, a)
    return 2.0 * n * float(np.max(conv))


# ----------------------------------------------------------------------
# Weighted (soft‑max) gradient of the ℓ∞‑norm of the convolution
# ----------------------------------------------------------------------


def weighted_gradient(a: np.ndarray, beta: float):
    """
    Compute the gradient of the soft‑max‑smoothed peak convolution.

    Returns
    -------
    grad : ndarray, shape (n,)
        The gradient w.r.t. a (before projection).
    max_conv : float
        The current maximal (unsmoothed) convolution value.
    """
    n = a.shape[0]

    # 1) compute the convolution a*a
    conv = convolution_fft(a, a)               # length 2n‑1
    max_conv = float(np.max(conv))

    # 2) soft‑max weights on the convolution entries
    #    w_k ∝ exp(β·(conv_k – max_conv))
    #    using a stable formulation (subtract the max first).
    if beta == 0.0:
        w = np.full_like(conv, 1.0 / conv.size)
    else:
        shifted = conv - max_conv               # ≤ 0
        exp_vals = np.exp(beta * shifted)
        w = exp_vals / np.sum(exp_vals)         # length 2n‑1, sums to 1

    # 3) gradient = 2·cross‑correlation(a, w)
    #    cross‑correlation can be obtained as convolution with the reversed w.
    w_rev = w[::-1]
    corr = convolution_fft(a, w_rev)           # length 3n‑2
    idx = 2 * n - 2 - np.arange(n)             # maps back to the n coefficients
    grad = 2.0 * corr[idx]

    return grad, max_conv


# ----------------------------------------------------------------------
# Smooth weighted exponentiated gradient descent (AdaGrad)
# ----------------------------------------------------------------------


def smooth_exp_gradient_descent(init_a: np.ndarray,
                               time_deadline: float,
                               beta0: float = 1.0,
                               beta_inc: float = 1.02,
                               beta_max: float = 200.0,
                               eta0: float = 0.5) -> tuple[np.ndarray, float]:
    """
    Gradient descent on the soft‑max‑smoothed objective.

    Parameters
    ----------
    init_a : ndarray
        Starting point (already normalised to sum‑1).
    time_deadline : float
        Absolute time (seconds since the epoch) at which the optimisation
        must stop.
    beta0, beta_inc, beta_max : float
        Temperature schedule for the soft‑max.
    eta0 : float
        Initial AdaGrad step‑size.

    Returns
    -------
    best_a, best_score
        The best coefficient vector found and its (exact) score.
    """
    a = init_a.astype(float).copy()
    n = a.shape[0]

    best_a = a.copy()
    best_score = compute_score(a)

    # AdaGrad accumulator
    G = np.zeros_like(a)

    beta = beta0
    iteration = 0

    while time.time() < time_deadline:
        iteration += 1

        grad, max_conv = weighted_gradient(a, beta)

        # AdaGrad per‑coordinate step‑size
        G += grad ** 2
        eta = eta0 / (np.sqrt(G) + 1e-12)     # avoid division by zero

        # multiplicative update (clip the exponent to avoid overflow)
        factor = np.exp(-np.clip(eta * grad, 0.0, 50.0))
        a = a * factor

        # project back onto the simplex (ensures sum‑1)
        a = project_to_simplex(a)

        # exact score (no smoothing)
        cur_score = 2.0 * n * max_conv
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            if iteration % 2000 == 0:
                print(f"Iter {iteration:>7d}: improved score = {best_score:.6f}")

        # temperature annealing
        if beta < beta_max:
            beta = min(beta * beta_inc, beta_max)

        # occasional tiny jitter to help escape plateaus
        if iteration % 5000 == 0:
            jitter = np.random.randn(n) * (eta0 * 1e-4 / np.sqrt(iteration))
            a = a + jitter
            a = project_to_simplex(a)

    return best_a, best_score


# ----------------------------------------------------------------------
# Simple stochastic hill‑climber
# ----------------------------------------------------------------------


def hill_climb(seq, time_deadline, max_steps=200_000):
    """
    Stochastic hill‑climber that performs swaps or fractional mass moves.
    Only improving moves are kept; the step‑size is adapted according to
    recent success rate.
    """
    a = np.array(seq, dtype=float)
    n = a.shape[0]

    best_a = a.copy()
    best_score = compute_score(best_a)

    step_scale = 0.1  # fraction of mass transferred in a move

    while max_steps > 0 and time.time() < time_deadline:
        move_type = np.random.rand()
        cand = best_a.copy()

        if move_type < 0.5:
            # swap two positions
            i, j = np.random.randint(n, size=2)
            cand[i], cand[j] = cand[j], cand[i]
        else:
            # transfer a fraction from i to j
            i, j = np.random.randint(n, size=2)
            if i != j:
                delta = step_scale * cand[i]
                cand[i] -= delta
                cand[j] += delta
                cand[i] = max(cand[i], 0.0)

        # re‑normalise onto the simplex
        cand = project_to_simplex(cand)
        cur_score = compute_score(cand)

        if cur_score < best_score:
            best_score = cur_score
            best_a = cand
            step_scale = min(step_scale * 1.05, 0.5)   # success → larger moves
        else:
            step_scale *= 0.97                         # failure → smaller moves

        max_steps -= 1

    return best_a.tolist(), best_score

    deadline = start_time + budget_s - 1.5

    best_seq = None
    best_val = float("inf")

    # ------------------------------------------------------------------
    # Try to use the provided fallback (if any)
    # ------------------------------------------------------------------
    try:
        fallback_raw = np.array(height_sequence_1, dtype=float)  # type: ignore
        if fallback_raw.size > 0:
            fallback_norm = fallback_raw / fallback_raw.sum()
            fallback_score = compute_score(fallback_norm)
            print(f"Fallback (normalised) score = {fallback_score:.6f}")
            if fallback_score < best_val:
                best_val = fallback_score
                best_seq = fallback_norm.tolist()
    except Exception:
        pass

    # ------------------------------------------------------------------
    # Parameters for the search
    # ------------------------------------------------------------------
    n = 2048                          # length of the candidate vector
    beta0 = 1.0                       # start temperature for soft‑max
    beta_inc = 1.02
    beta_max = 200.0
    eta0 = 0.5                        # initial AdaGrad step‑size

    # ------------------------------------------------------------------
    # Phase 1 – multiple random restarts with smooth gradient descent
    # ------------------------------------------------------------------
    # reserve ~200 s for the final hill‑climb
    hill_climb_time = 200.0
    while time.time() < deadline - hill_climb_time:
        remaining = deadline - hill_climb_time - time.time()
        # give each restart at most 150 s (or whatever is left)
        run_time = min(150.0, remaining)
        if run_time <= 0:
            break

        # random Dirichlet start
        a0 = np.random.dirichlet(np.ones(n))
        # run the optimiser
        refined_a, refined_score = smooth_exp_gradient_descent(
            a0,
            time_deadline=time.time() + run_time,
            beta0=beta0,
            beta_inc=beta_inc,
            beta_max=beta_max,
            eta0=eta0,
        )
        if refined_score < best_val:
            best_val = refined_score
            best_seq = refined_a.tolist()
            print(f"New best after gradient descent: {best_val:.6f}")

    # ------------------------------------------------------------------
    # Phase 2 – final hill‑climb polishing (uses remaining time)
    # ------------------------------------------------------------------
    if best_seq is not None and time.time() < deadline:
        print("\nStarting hill‑climb polishing")
        polished_seq, polished_val = hill_climb(
            best_seq,
            time_deadline=deadline,
            max_steps=500_000,
        )
        if polished_val < best_val:
            best_val = polished_val
            best_seq = polished_seq
            print(f"Hill‑climb improved best to {best_val:.6f}")

    # ------------------------------------------------------------------
    # Fallback in the unlikely case that nothing succeeded
    # ------------------------------------------------------------------
    if best_seq is None:
        n_small = 256
        best_seq = [1.0 / n_small] * n_small
        best_val = compute_score(np.array(best_seq))

    # verify with the official evaluator (scales are invariant)
    final_score = evaluate_sequence(best_seq)
    print(f"\nSearch finished. Final reported score = {final_score:.6f}")
    return best_seq
