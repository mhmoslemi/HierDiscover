2. **Smooth exponentiated‑gradient descent** –  we repeatedly
   *a ← a·exp(‑η·∇)* and project back onto the simplex.
   The gradient is taken w.r.t. the *soft* max of the convolution:
   w_i ∝ exp(β·(c_i‑max(c))) .  β grows linearly, η decays like 1/√t.
   This quickly drives the maximal convolution down while staying stable.

3. **Hard sub‑gradient polishing** –  once the time budget is half‑used we
   switch to the true sub‑gradient (uniform weight on all (near) maximisers)
   and continue with the same exponentiated‑gradient update.  The best exact
   score (using the official `evaluate_sequence`) seen during the whole run
   is returned.

All operations are performed on the simplex (∑a=1) so the objective is
scale‑invariant; the final sequence is clipped to the allowed range by the
official evaluator.  The routine respects the supplied time limit
(`budget_s`, default 1000 s) and always returns a feasible list of floats.
"""

import time
import numpy as np
from typing import List, Tuple

# ----------------------------------------------------------------------
# Low‑level numeric utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """
    Linear convolution via real FFT (full length, zero‑padded to a power‑of‑2).
    """
    m = len(a) + len(b) - 1
    n_fft = next_pow2(m)
    A = np.fft.rfft(a, n=n_fft)
    B = np.fft.rfft(b, n=n_fft)
    C = A * B
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


def normalize(v: np.ndarray) -> np.ndarray:
    """Project a non‑negative vector onto the simplex (sum = 1)."""
    total = v.sum()
    if total <= 0.0:
        # fallback to uniform if something went wrong
        return np.full_like(v, 1.0 / len(v))
    return v / total


def compute_score(a: np.ndarray) -> float:
    """
    Fast proxy for the official objective.
    The vector a is assumed non‑negative and sum(a)=1, therefore the score
    equals 2·n·max(conv(a,a)).
    """
    n = len(a)
    conv = convolution_fft(a, a)
    max_val = float(np.max(conv))
    return 2.0 * n * max_val


def gradient_from_weights(a: np.ndarray, w: np.ndarray) -> np.ndarray:
    """
    Gradient of Σ_k w_k·(a*a)_k w.r.t. a.
    Implements   g_j = 2· Σ_m a_m · w_{m+j}.
    Computed via a single FFT‑based convolution.
    """
    # reverse w because convolution with reversed weight yields the needed sum
    w_rev = w[::-1]
    conv = convolution_fft(a, w_rev)               # length = 3·n‑2
    idx = 2 * len(a) - 2                           # position of zero shift
    grads = 2.0 * conv[idx - np.arange(len(a))]
    return grads


# ----------------------------------------------------------------------
# Parametric families (initial candidates)
# ----------------------------------------------------------------------


def arcsine_initial(n: int) -> np.ndarray:
    """Discrete arcsine distribution: a_i ∝ 1/√((i+½)(n‑½‑i))."""
    i = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - 0.5 - i))
    return normalize(vals)


def powerlaw_shape(n: int, p: float, eps: float = 0.5) -> np.ndarray:
    """
    Symmetric power‑law shape a_i ∝ ((i+eps)(n‑1‑i+eps))⁻ᵖ.
    eps prevents division by zero.
    """
    i = np.arange(n, dtype=np.float64)
    base = ((i + eps) * (n - 1 - i + eps)) ** (-p)
    return normalize(base)


def uniform_shape(n: int) -> np.ndarray:
    """Uniform vector (sum = 1)."""
    return np.full(n, 1.0 / n)


def random_dirichlet(n: int, rng: np.random.Generator) -> np.ndarray:
    """Sample a random point on the simplex."""
    samp = rng.exponential(1.0, size=n)
    return normalize(samp)


# ----------------------------------------------------------------------
# Refinement methods
# ----------------------------------------------------------------------


def smooth_descent(
    a: np.ndarray,
    deadline: float,
    *,
    eta0: float = 0.9,
    beta_start: float = 6.0,
    beta_end: float = 25.0,
) -> Tuple[np.ndarray, float]:
    """
    Exponentiated‑gradient descent using a softened max‑weight.
    The temperature β grows linearly from beta_start to beta_end,
    while the step size follows η_t = η0/√t.
    Returns the best iterate (according to the exact score) and its score.
    """
    a = normalize(a.astype(np.float64).copy())
    n = len(a)

    best_a = a.copy()
    best_sc = compute_score(best_a)

    it = 0
    while time.time() < deadline:
        it += 1
        # current convolution
        conv = convolution_fft(a, a)
        max_val = np.max(conv)

        # linear schedule for beta
        elapsed = (time.time() - (deadline - (deadline - (time.time()))))
        # simpler: use fraction of total smooth time
        # we approximate fraction by it / (it + 1) (monotone)
        beta = beta_start + (beta_end - beta_start) * min(1.0, it / (it + 1000.0))

        # soft‑max‑type weights (the maximal entry gets weight 1)
        exp_vals = np.exp(beta * (conv - max_val))
        w = exp_vals / np.sum(exp_vals)

        # gradient for this smoothed objective
        grad = gradient_from_weights(a, w)

        # step‑size schedule
        eta = eta0 / np.sqrt(it)

        # exponentiated‑gradient update + simplex projection
        a = a * np.exp(-eta * grad)
        a = normalize(a)

        cur_sc = compute_score(a)
        if cur_sc < best_sc:
            best_sc = cur_sc
            best_a = a.copy()

    return best_a, best_sc


def hard_subgradient(
    a: np.ndarray,
    deadline: float,
    *,
    eta0: float = 0.7,
    tol: float = 1e-13,
) -> Tuple[np.ndarray, float]:
    """
    Classical sub‑gradient descent on the true max‑convolution objective.
    Uses a uniform weight on all (near) maximisers and the same
    exponentiated‑gradient update as the smooth phase.
    Returns the best iterate seen and its exact score.
    """
    a = normalize(a.astype(np.float64).copy())
    n = len(a)

    best_a = a.copy()
    best_sc = compute_score(a)

    it = 0
    while time.time() < deadline:
        it += 1
        conv = convolution_fft(a, a)
        max_val = np.max(conv)

        # binary weight on the set of maximisers (within a tiny tolerance)
        w = np.zeros_like(conv)
        max_idx = np.where(conv >= max_val - tol)[0]
        w[max_idx] = 1.0 / len(max_idx)

        grad = gradient_from_weights(a, w)

        eta = eta0 / np.sqrt(it)
        a = a * np.exp(-eta * grad)
        a = normalize(a)

        cur_sc = compute_score(a)
        if cur_sc < best_sc:
            best_sc = cur_sc
            best_a = a.copy()

    return best_a, best_sc


# ----------------------------------------------------------------------
# Main entry point required by the competition harness
# ----------------------------------------------------------------------


def propose_candidate(
    seed: int = 42,
    budget_s: int = 1000,
    **kwargs,
) -> List[float]:
    """
    Search for a non‑negative step‑function minimizing the official objective.
    The routine respects the supplied time budget, builds several high‑resolution
    initial candidates (arcsine, power‑law sweeps, randomness), refines the
    best one with a smooth exponentiated‑gradient phase and a final hard
    sub‑gradient polish, and finally returns the best sequence found.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    # tiny safety margin to avoid hard timeout
    deadline = start_time + budget_s - 2.0

    # ------------------------------------------------------------------
    # 1.  Choose a working length.
    # ------------------------------------------------------------------
    # A larger n reduces discretisation error of the arcsine shape.
    # 8192 is a good compromise between accuracy and evaluation cost.
    n = 8192

    # ------------------------------------------------------------------
    # 2.  Build an initial set of candidates and keep the best one.
    # ------------------------------------------------------------------
    candidates: List[np.ndarray] = []

    # a) arcsine family (theoretical optimum)
    candidates.append(arcsine_initial(n))

    # b) power‑law sweep – varying the exponent p around 0.5
    for p in np.linspace(0.35, 0.65, 7):
        candidates.append(powerlaw_shape(n, p, eps=0.5))

    # c) a few random Dirichlet draws for extra diversity
    for _ in range(3):
        candidates.append(random_dirichlet(n, rng))

    # d) uniform baseline (useful as a sanity check)
    candidates.append(uniform_shape(n))

    # If a previously discovered construction exists, use it as a fallback.
    try:
        fallback = np.array([float(x) for x in height_sequence_1], dtype=float)
        if fallback.ndim == 1:
            fallback = normalize(fallback)
            candidates.append(fallback)
            print("[fallback] length =", len(fallback))
    except NameError:
        pass

    # Fast ranking with the FFT‑based score (sum already = 1)
    scored = [(compute_score(c), c) for c in candidates]
    scored.sort(key=lambda x: x[0])
    best_candidate = scored[0][1]
    best_fast = scored[0][0]
    print(f"Initial best fast (proxy) score = {best_fast:.6f}")

    # ------------------------------------------------------------------
    # 3.  Smooth exponentiated‑gradient refinement (≈ half of remaining time)
    # ------------------------------------------------------------------
    remaining = max(0.0, deadline - time.time())
    smooth_deadline = time.time() + remaining * 0.55
    print(f"Smooth phase deadline in {remaining * 0.55:.1f}s")

    refined_smooth, smooth_score = smooth_descent(
        a=best_candidate,
        deadline=smooth_deadline,
        eta0=0.9,
        beta_start=6.0,
        beta_end=25.0,
    )
    print(f"Smooth phase best exact score = {smooth_score:.6f}")

    # Full official evaluation of the current best
    val = evaluate_sequence(refined_smooth.tolist())
    print(f"Official score after smooth phase = {val:.6f}")

    best_seq: List[float] = refined_smooth.tolist()
    best_val: float = val

    # ------------------------------------------------------------------
    # 4.  Hard sub‑gradient polishing (use the remaining budget)
    # ------------------------------------------------------------------
    if time.time() < deadline - 4.0:
        print("\nStarting hard sub‑gradient polish")
        polish_deadline = deadline
        polished, _ = hard_subgradient(
            a=refined_smooth,
            deadline=polish_deadline,
            eta0=0.7,
        )
        final_val = evaluate_sequence(polished.tolist())
        print(f"Official score after polish = {final_val:.6f}")

        if final_val < best_val:
            best_val = final_val
            best_seq = polished.tolist()
            print("*** New global best after polishing ***")
    else:
        print("Not enough time left for polishing; returning smooth result.")

    # ------------------------------------------------------------------
    # 5.  Guard – always return a feasible sequence.
    # ------------------------------------------------------------------
    if not best_seq:
        # Extremely unlikely; fall back to a safe uniform vector of modest size.
        best_seq = uniform_shape(256).tolist()
        best_val = evaluate_sequence(best_seq)

    print(f"\nSearch finished. Best reported official score = {best_val:.6f}")
    print(f"Returned sequence length = {len(best_seq)}")
    return best_seq
