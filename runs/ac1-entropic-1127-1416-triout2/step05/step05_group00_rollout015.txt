def smooth_max_gradient(a, alpha):
    # a: np.ndarray shape (n,)
    # compute conv = a*a via convolution_fft
    conv = convolution_fft(a, a)
    max_val = np.max(conv)
    # compute weight w
    # w_i = exp(alpha * (conv_i - max_val))
    w = np.exp(alpha * (conv - max_val))
    # compute gradient
    a_rev = a[::-1]
    grad_full = convolution_fft(w, a_rev)
    grad = 2 * grad_full[n-1 : n-1 + n]   # length n
    return max_val, grad
```

"""
Fast heuristic for minimizing the evaluation score of a step function.

The score for a non‑negative vector `a` (with sum = 1) is
    score = 2·n·max(conv(a,a)),
where `conv` is the full linear convolution.
The goal is to flatten the convolution so that its maximum is as close
to the average 1/(2n) as possible (the theoretical lower bound).

Algorithm (single‑thread, <1000 s):
1.  Generate several simple initial vectors (uniform, Dirichlet,
    random sparse, symmetric exponential).
2.  Refine each candidate with a *smooth max* gradient descent:
        – use a soft‑max weighting `w = exp(α·(conv‑max))`,
        – the gradient of the smooth max is `2·(w * a_rev)`,
          which is obtained by a single FFT‑based convolution.
        – additive projected gradient steps with a decreasing step size.
3.  Optionally improve the best vector by solving a linear programme
    (SciPy `linprog`) that tries to spread the mass while keeping the
    current convolution peak `M` unchanged.
4.  Finish with a short stochastic hill‑climb (swaps / mass transfers)
    that accepts only improving moves.
5.  Return the best sequence found as a plain Python list.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Basic linear‑algebra helpers
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """
    Full convolution a ⊛ b using a real FFT.  Both inputs are 1‑D
    NumPy arrays of dtype float.
    """
    m = a.shape[0] + b.shape[0] - 1
    n_fft = next_pow2(m)
    A = np.fft.rfft(a, n=n_fft)
    B = np.fft.rfft(b, n=n_fft)
    C = A * B
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


def compute_score(a: np.ndarray) -> float:
    """
    Fast replica of `evaluate_sequence` when ∑a == 1.
    Returns 2·n·max(a⊛a).
    """
    n = a.shape[0]
    conv = convolution_fft(a, a)
    return 2.0 * n * float(np.max(conv))


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the probability simplex
    (non‑negative entries that sum to 1).
    """
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / v.shape[0])

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, v.shape[0] + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.where(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


# ----------------------------------------------------------------------
# Candidate generators
# ----------------------------------------------------------------------


def init_uniform(n: int) -> np.ndarray:
    """Uniform probability vector of length n."""
    return np.full(n, 1.0 / n)


def init_dirichlet(n: int, concentration: float = 1.0) -> np.ndarray:
    """Sample from a Dirichlet distribution (Gamma with given concentration)."""
    a = np.random.gamma(concentration, 1.0, n)
    a /= a.sum()
    return a


def init_random_sparse(n: int) -> np.ndarray:
    """Randomly choose a dense support and give equal mass to the chosen indices."""
    p = np.random.uniform(0.20, 0.90)                     # density of support
    k = max(2, int(round(p * n)))
    idx = np.random.choice(n, size=k, replace=False)
    a = np.zeros(n, dtype=float)
    a[idx] = 1.0 / k
    return a


def init_symmetric_exponential(n: int) -> np.ndarray:
    """Symmetric exponential decay from both ends."""
    beta = np.random.uniform(0.0005, 0.005)               # decay rate
    i = np.arange(n, dtype=float)
    a = np.exp(-beta * i) + np.exp(-beta * (n - 1 - i))
    a /= a.sum()
    return a


# ----------------------------------------------------------------------
# Smooth‑max gradient descent (core refinement)
# ----------------------------------------------------------------------


def smooth_max_optimize(a_init: np.ndarray,
                        time_deadline: float,
                        max_iter: int = 25000,
                        alpha_start: float = 0.5,
                        alpha_end: float = 5000.0,
                        step0: float = 0.5) -> tuple[np.ndarray, float]:
    """
    Minimise the smooth approximation of the max‑convolution.
    The smoothness parameter α is increased from ``alpha_start`` to
    ``alpha_end`` during the run.  An additive projected gradient step
    with step size step0/√t is used.
    Returns the best vector seen (sum = 1) and its score.
    """
    a = a_init.astype(float).copy()
    n = a.shape[0]
    best_a = a.copy()
    best_score = compute_score(a)

    for it in range(1, max_iter + 1):
        if time.time() >= time_deadline:
            break

        # linear schedule for α
        alpha = alpha_start + (alpha_end - alpha_start) * (it - 1) / (max_iter - 1)

        # convolution a * a  (full)
        conv = convolution_fft(a, a)
        max_val = float(conv.max())

        # soft‑max weights (w_i ∈ (0,1])
        w = np.exp(alpha * (conv - max_val))

        # gradient of the smooth max: 2·(w ⋆ a)      (cross‑correlation)
        a_rev = a[::-1]
        grad_full = convolution_fft(w, a_rev)
        grad = 2.0 * grad_full[n - 1 : n - 1 + n]

        # decreasing step size
        step = step0 / np.sqrt(it)

        # additive update + projection onto simplex
        a = a - step * grad
        a = project_to_simplex(a)

        cur_score = compute_score(a)
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()

    return best_a, best_score


# ----------------------------------------------------------------------
# Linear‑programming mixing (SciPy) – optional
# ----------------------------------------------------------------------


def solve_lp_scipy(a: np.ndarray, M: float):
    """
    Maximise   1ᵀb   subject to   b ≥ 0,
                (a ⊛ b) ≤ M  (element‑wise)

    Returns the optimal b (may be non‑normalised) or None on failure.
    """
    try:
        from scipy.optimize import linprog
    except Exception:
        return None

    n = a.shape[0]
    L = 2 * n - 1
    # Build the constraint matrix A where (a ⊛ b)[ℓ] = Σ_j a[ℓ‑j]·b_j
    A = np.zeros((L, n), dtype=float)
    for j in range(n):
        A[j : j + n, j] = a  # shift a by j positions

    c = -np.ones(n)                 # maximise Σ b  ⇔ minimise −Σ b
    b_ub = np.full(L, M)
    bounds = [(0, None)] * n

    res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds,
                  method='highs', options={'presolve': True, 'time_limit': 30})
    if not res.success or res.x is None:
        return None
    return np.asarray(res.x, dtype=float)


# ----------------------------------------------------------------------
# Simple hill‑climber (swap / mass transfer)
# ----------------------------------------------------------------------


def hill_climb(seq, time_deadline, max_steps=300_000):
    """
    Stochastic hill‑climb that swaps entries or transfers a fraction of mass.
    Accepts a move only when the true score improves.
    """
    a = np.array(seq, dtype=float)
    n = a.shape[0]
    best_a = a.copy()
    best_score = compute_score(best_a)

    step_scale = 0.1  # fraction of mass moved when doing a transfer

    while max_steps > 0 and time.time() < time_deadline:
        move_type = np.random.rand()
        cand = best_a.copy()

        if move_type < 0.5:
            # swap two positions
            i, j = np.random.randint(n, size=2)
            cand[i], cand[j] = cand[j], cand[i]
        else:
            # transfer a fraction from i to j
            i, j = np.random.randint(n, size=2)
            if i == j:
                continue
            delta = step_scale * best_a[i]
            if delta <= 0:
                continue
            cand[i] -= delta
            cand[j] += delta
            if cand[i] < 0:
                cand[i] = 0.0

        cand = project_to_simplex(cand)
        cur_score = compute_score(cand)

        if cur_score < best_score:
            best_score = cur_score
            best_a = cand
            step_scale = min(step_scale * 1.05, 0.5)
        else:
            step_scale *= 0.97

        max_steps -= 1

    return best_a.tolist(), best_score
