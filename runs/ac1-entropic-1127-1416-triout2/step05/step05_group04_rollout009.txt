def propose_candidate(seed=42, budget_s=1000, **kwargs):

    start = time.time()
    deadline = start + budget_s - 1.0
    best_seq = None
    best_val = float('inf')
    n = 8192
    # seeds
    arcsine = arcsine_initial(n)
    power = generate_power_shape(n, p=0.9, eps=0.1)

    uniform = np.full(n, 1.0/n)
    seeds = [arcsine, power, mix, uniform]
    # optionally add random Dirichlet seeds if time
    # loop seeds
    for idx, seed_vec in enumerate(seeds, 1):
        if time.time() >= deadline - 5:
            break
        print(f"\nRefining seed {idx}/{len(seeds)}")
        a = seed_vec
        # Stage 1: flatten
        remaining = deadline - time.time()
        max_iter_flat = int(min(250000, remaining * 200))
        print(f"   Flatten descent iters={max_iter_flat}")
        a, cur_sc = flatten_descent(a, deadline, max_iter=max_iter_flat, step0=0.05)
        # Stage 2: Adam smoothing
        remaining = deadline - time.time()
        max_iter_adam = int(min(200000, remaining * 100))
        print(f"   Adam smooth iters={max_iter_adam}")
        a, _ = adam_optimize_smooth(a, deadline, max_iter=max_iter_adam, lr0=0.1, lam_start=5.0, lam_end=150.0, noise_every=5000, noise_scale=1e-5)
        # Stage 3: Subgradient
        remaining = deadline - time.time()
        max_iter_sub = int(min(200000, remaining * 150))
        print(f"   Subgradient iters={max_iter_sub}")
        a = subgradient_descent_exact(a, deadline, max_iter=max_iter_sub, step0=0.02)
        # Stage 4: SA maybe
        remaining = deadline - time.time()
        if remaining > 10:
            max_sa = int(min(150000, remaining * 150))
            print(f"   Simulated annealing iters={max_sa}")
            a = simulated_annealing(a, rng, deadline, max_iters=max_sa, init_temp=0.01, decay=0.9999)
        # Evaluate official
        try:
            val = evaluate_sequence(a.tolist())
            print(f"   Official score = {val:.6f}")
            if val < best_val:
                best_val = val
                best_seq = a.tolist()

        except Exception:
            pass
    # final polishing
    if best_seq is not None and time.time() < deadline - 3:
        print("\nFinal polishing of best sequence")
        a = np.array(best_seq, dtype=float)
        remaining = deadline - time.time()
        # Flatten again
        it_flat = int(min(150000, remaining * 150))
        a, _ = flatten_descent(a, deadline, max_iter=it_flat, step0=0.03)

        print(f"Final polished official score = {final_val:.6f}")
        if final_val < best_val:
            best_seq = a.tolist()
            best_val = final_val
    # fallback
    if not best_seq:
        best_seq = [1.0] * n
        best_val = evaluate_sequence(best_seq)
    print(f"\nSearch finished, best score = {best_val:.6f}")

"""
propose_candidate
-----------------

Search for a non‑negative step sequence that minimises the official
``evaluate_sequence`` value.

Algorithm
~~~~~~~~~
1. **Initial seeds** – a few analytically‑motivated shapes (arcsine,
   power‑law, mixture with uniform) and the optional fallback
   ``height_sequence_1`` from a previous run.  All seeds are projected
   onto the probability simplex (Σa = 1).

2. **Flattening phase** – minimise the L2‑error of the convolution to its
   average value 1/(2n‑1).  The gradient of this smooth objective is cheap
   (two FFTs) and drives the whole autocorrelation towards a flat shape.
   This often reduces the supremum far below what the simple arcsine
   already achieves.

3. **Smooth Adam phase** – a continuation method that replaces the hard
   max by a soft‑max (log‑sum‑exp) with an increasing temperature λ.
   Adam optimisation with projection onto the simplex quickly refines
   the distribution.

4. **Exact sub‑gradient descent** – optimisation of the true objective
   ``2·n·max(conv(a,a))`` using the sub‑gradient given by the active
   (maximal) convolution entries.

5. **Simulated‑annealing polish** – a cheap Metropolis move that swaps a
   small amount of mass between two coordinates, accepted according to
   the surrogate objective.

6. **Global polish** – the best vector found is run through the three
   phases again with a tighter budget.

All heavy work is performed with FFT‑based convolutions (real FFT) and
projection onto the simplex, so the algorithm scales to large vectors
(e.g. n = 8192) while staying well within the 1000 s time limit.

The routine returns a list of non‑negative heights ready for
``evaluate_sequence``.
"""

import time
import numpy as np
from typing import List, Tuple

# ----------------------------------------------------------------------
# Low‑level utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """
    Linear convolution via real FFT.
    Length = len(a) + len(b) - 1.
    """
    m = len(a) + len(b) - 1
    n_fft = next_pow2(m)
    A = np.fft.rfft(a, n=n_fft)
    B = np.fft.rfft(b, n=n_fft)
    C = A * B
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


def compute_score(a: np.ndarray) -> float:
    """
    Fast surrogate for the official score.
    The vector is assumed to lie on the simplex (Σa = 1);
    the denominator of ``evaluate_sequence`` therefore disappears
    and the score equals ``2·n·max(conv(a,a))``.
    """
    n = len(a)
    conv = convolution_fft(a, a)
    return 2.0 * n * float(np.max(conv))


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection onto the probability simplex {x ≥ 0, Σx = 1}.
    Duchi et al., 2008.
    """
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / v.size)

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho = np.where(u - cssv / np.arange(1, v.size + 1) > 0)[0]
    if len(rho) == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


# ----------------------------------------------------------------------
# Shape factories (all return a vector that already sums to 1)
# ----------------------------------------------------------------------


def arcsine_initial(n: int) -> np.ndarray:
    """Classic arcsine shape, softened by a tiny epsilon."""
    i = np.arange(n, dtype=float)
    eps = 0.25
    vals = 1.0 / np.sqrt((i + 0.5) * (n - 0.5 - i) + eps)
    vals /= vals.sum()
    return vals


def generate_power_shape(n: int, p: float, eps: float) -> np.ndarray:
    """Generalised arcsine: a_i ∝ ((i+½)*(n‑½‑i)+ε)^{-p},   p>0."""
    i = np.arange(n, dtype=float)
    denom = ((i + 0.5) * (n - 0.5 - i) + eps) ** p
    vals = 1.0 / denom
    total = vals.sum()
    if not np.isfinite(total) or total <= 0.0:
        return np.full(n, 1.0 / n)
    return vals / total


def generate_mixture_shape(n: int, weight_arc: float) -> np.ndarray:
    """Mixture (weight_arc)·arcsine + (1‑weight_arc)·uniform."""
    arc = arcsine_initial(n)
    uni = np.full(n, 1.0 / n)
    mix = weight_arc * arc + (1.0 - weight_arc) * uni
    mix /= mix.sum()
    return mix


# ----------------------------------------------------------------------
# Optimisers
# ----------------------------------------------------------------------


def smooth_gradient(a: np.ndarray, lam: float) -> np.ndarray:
    """
    Gradient of the soft‑max approximation of max(conv(a,a)).
    Returns ∇_a (2·n·softmax_max) where the sum constraint is handled
    later by projection onto the simplex.
    """
    n = len(a)
    conv = convolution_fft(a, a)               # length L = 2·n‑1
    max_c = np.max(conv)

    # Numerically stable soft‑max (weights sum to 1)
    exp_term = np.exp(lam * (conv - max_c))
    w = exp_term / np.sum(exp_term)

    # Weighted sub‑gradient: 2·∑_k w_k shift(a,k)
    w_rev = w[::-1]
    conv_rev = convolution_fft(a, w_rev)
    L = 2 * n - 1
    weighted_grad = 2.0 * conv_rev[L - 1 - np.arange(n)]

    return 2.0 * n * weighted_grad


def adam_optimize_smooth(initial_a: np.ndarray,
                         deadline: float,
                         max_iter: int = 100_000,
                         lr0: float = 0.12,
                         lam_start: float = 2.0,
                         lam_end: float = 100.0,
                         noise_every: int = 2000,
                         noise_scale: float = 5e-5) -> Tuple[np.ndarray, float]:
    """
    Adam optimiser on the smooth (log‑sum‑exp) surrogate of the max.
    Returns the best vector (projected) and its fast score.
    """
    a = project_to_simplex(initial_a.astype(float).copy())
    best_a = a.copy()
    best_sc = compute_score(a)

    beta1, beta2 = 0.9, 0.999
    eps = 1e-8
    m = np.zeros_like(a)
    v = np.zeros_like(a)

    for it in range(1, max_iter + 1):
        if time.time() >= deadline:
            break

        lam = lam_start + (lam_end - lam_start) * min(1.0, it / max_iter)

        grad = smooth_gradient(a, lam)

        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)

        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        step = lr0 * m_hat / (np.sqrt(v_hat) + eps)
        a_new = a - step

        if noise_every and (it % noise_every == 0):
            a_new += np.random.randn(a.size) * noise_scale

        a_new = project_to_simplex(a_new)

        cur_sc = compute_score(a_new)
        if cur_sc < best_sc:
            best_sc = cur_sc
            best_a = a_new.copy()

        a = a_new

        if it % 5000 == 0:
            print(f"    AdamSmooth it={it:,d} cur={cur_sc:.6f} best={best_sc:.6f}")

    return best_a, best_sc


def subgradient_descent_exact(initial_a: np.ndarray,
                              deadline: float,
                              max_iter: int = 80_000,
                              step0: float = 0.05) -> np.ndarray:
    """
    Sub‑gradient descent on the exact objective f(a)=2·n·max(conv(a,a)).
    Step size η_t = step0 / sqrt(t).  Projection onto the simplex keeps Σa=1.
    """
    a = project_to_simplex(initial_a.copy())
    best_a = a.copy()
    best_sc = compute_score(a)
    n = len(a)
    L = 2 * n - 1

    for it in range(1, max_iter + 1):
        if time.time() >= deadline:
            break

        conv = convolution_fft(a, a)
        max_val = np.max(conv)

        # tolerance to catch all indices attaining (or almost attaining) the max
        tol = max_val * 1e-12 + 1e-14
        mask = (np.abs(conv - max_val) <= tol).astype(float)

        mask_rev = mask[::-1]
        conv_rev = convolution_fft(a, mask_rev)
        weighted_grad = 2.0 * conv_rev[L - 1 - np.arange(n)]

        eta = step0 / np.sqrt(it)
        a = a - eta * weighted_grad
        a = project_to_simplex(a)

        cur_sc = compute_score(a)
        if cur_sc < best_sc:
            best_sc = cur_sc
            best_a = a.copy()

        if it % 5000 == 0:
            print(f"    SubGrad it={it:,d} cur={cur_sc:.6f} best={best_sc:.6f}")

    return best_a


def simulated_annealing(a: np.ndarray,
                        rng: np.random.Generator,
                        deadline: float,
                        max_iters: int = 120_000,
                        init_temp: float = 0.02,
                        decay: float = 0.9999) -> np.ndarray:
    """
    Simple Metropolis‑style local search using the fast surrogate score.
    """
    cur = a.copy()
    cur_sc = compute_score(cur)
    best = cur.copy()
    best_sc = cur_sc
    n = len(cur)
    temp = init_temp
    it = 0

    while it < max_iters and time.time() < deadline:
        it += 1
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue

        delta = cur[i] * 0.2 * rng.random()
        if delta <= 1e-15 or cur[i] - delta < 0:
            continue

        cand = cur.copy()
        cand[i] -= delta
        cand[j] += delta

        cand_sc = compute_score(cand)

        if cand_sc < cur_sc or rng.random() < np.exp(-(cand_sc - cur_sc) / max(temp, 1e-12)):
            cur = cand
            cur_sc = cand_sc
            if cur_sc < best_sc:
                best = cur.copy()
                best_sc = cur_sc

        temp = max(temp * decay, 1e-8)

        if it % 5000 == 0:
            print(f"    SA it={it:,d} cur={cur_sc:.6f} best={best_sc:.6f}")

    return best


def flatten_descent(initial_a: np.ndarray,
                    deadline: float,
                    max_iter: int = 150_000,
                    step0: float = 0.05) -> Tuple[np.ndarray, float]:
    """
    Gradient descent on the L2‑error of the convolution to its average.
    Objective: ½‖conv(a,a) – 1/(2n‑1)·1‖².
    This drives the whole autocorrelation towards a flat shape,
    which is a very good proxy for lowering the supremum.
    """
    a = project_to_simplex(initial_a.copy())
    best_a = a.copy()
    best_sc = compute_score(a)

    n = len(a)
    L = 2 * n - 1
    avg = 1.0 / L

    for it in range(1, max_iter + 1):
        if time.time() >= deadline:
            break

        conv = convolution_fft(a, a)          # length L
        err = conv - avg                       # deviation from the target constant

        # gradient = 2 * (a * err_rev) restricted to the first n entries
        err_rev = err[::-1]
        grad_full = 2.0 * convolution_fft(a, err_rev)   # length n+L‑1 = 3n‑2
        grad = grad_full[L - 1 - np.arange(n)]

        eta = step0 / np.sqrt(it)
        a = a - eta * grad
        a = project_to_simplex(a)

        cur_sc = compute_score(a)
        if cur_sc < best_sc:
            best_sc = cur_sc
            best_a = a.copy()

        if it % 5000 == 0:
            print(f"    Flatten it={it:,d} cur={cur_sc:.6f} best={best_sc:.6f}")

    return best_a, best_sc

def propose_candidate(seed: int = 42,

    """
    Search for a non‑negative step sequence with a low official score.
    The routine combines a large‑scale flattening phase, smooth Adam,
    exact sub‑gradient descent and a final annealing polish.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()

    best_seq: List[float] = []

    if "height_sequence_1" in globals():
        try:

            val = evaluate_sequence(fallback)

            if val < best_val:

        except Exception:

    # ------------------------------------------------------------------

    n = 8192                      # larger n gives a tighter asymptotic constant
    seeds: List[np.ndarray] = []

    seeds.append(arcsine_initial(n))

    seeds.append(np.full(n, 1.0 / n))

    # a few random Dirichlet seeds (if time permits)
    while time.time() < deadline - 30.0 and len(seeds) < 8:
        alphas = rng.exponential(scale=1.0, size=n) + 0.01

    for sid, seed_vec in enumerate(seeds, 1):
