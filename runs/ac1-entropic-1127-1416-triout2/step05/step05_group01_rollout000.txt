def hill_climb_permutation(seq, time_deadline, max_steps=200_000):
    a = np.array(seq, dtype=float)
    n = a.shape[0]
    best_a = a.copy()
    best_score = compute_score(a)

    cur_a = a.copy()
    cur_score = best_score

    temp0 = 0.01  # initial temperature maybe 0.1 scaling from score dimension.
    for step in range(max_steps):
        if time.time() >= time_deadline:
            break
        # Temperature schedule linear decay
        t_frac = step / max_steps
        temp = temp0 * (1 - t_frac)

        # Choose move: swap or transfer
        if np.random.rand() < 0.5:
            i, j = np.random.randint(n, size=2)
            cand = cur_a.copy()
            cand[i], cand[j] = cand[j], cand[i]
        else:
            i, j = np.random.randint(n, size=2)
            if i == j:
                continue
            # transfer fraction
            frac = np.random.rand() * 0.2  # up to 20% of mass at i
            delta = frac * cur_a[i]
            cand = cur_a.copy()
            cand[i] -= delta
            cand[j] += delta
            cand[i] = max(cand[i], 0.0)

        cand = project_to_simplex(cand)
        cand_score = compute_score(cand)

        if cand_score < cur_score:
            cur_a = cand
            cur_score = cand_score
            if cand_score < best_score:
                best_score = cand_score
                best_a = cand.copy()
        else:
            # accept with probability exp(-(cand_score - cur_score)/temp)
            if temp > 0 and np.random.rand() < np.exp(-(cand_score - cur_score) / temp):
                cur_a = cand
                cur_score = cand_score

    return best_a.tolist(), best_score

"""
Propose a low‑score non‑negative sequence for `evaluate_sequence`.

Algorithm
---------
1. **Initialization**
   * Try the provided fallback `height_sequence_1` (if any).
   * Generate a random point on the probability simplex (Dirichlet draw) for a
     chosen length `n`.  Symmetrising the start often helps because the
     convolution kernel is symmetric.

2. **Top‑K exponentiated sub‑gradient descent**
   * At each iteration compute the full convolution `c = a * a`.
   * Take the `K` largest convolution entries (the current “active set”).
   * The sub‑gradient of `max(c)` at these indices is
         g_i = 2 * Σ_{k∈active} a_{k-i}
     (zero when the index is out of range).  This is accumulated with fast
     NumPy slice operations.
   * Perform a multiplicative (exponentiated) update
         a ← a · exp(-η·g)
     with a diminishing step size `η = η0 / sqrt(t)`.
   * Project back onto the probability simplex.
   * Keep the best vector seen so far.

   The loop stops when the time budget expires.

3. **Local stochastic hill‑climbing**
   * Starting from the best vector, apply a large number of cheap moves:
     – swap two coordinates,
     – transfer a random fraction of mass from one coordinate to another.
   * After each move re‑project onto the simplex.
   * Accept every improving move; optionally accept worsening moves with a
     simple temperature schedule (simulated‑annealing style) to escape plateaus.

4. **Optional symmetrisation**
   * After hill‑climbing, replace the vector by its symmetric part
     `(a + a[::-1]) / 2` and re‑project onto the simplex.  This often reduces
     the maximal autocorrelation further.

5. **Multiple restarts / lengths**
   * The procedure is executed for a couple of lengths (2048 and, if time
     permits, 4096).  The best sequence over all runs is returned.

All heavy operations (convolution, gradient accumulation) are performed with
NumPy/FFT and stay well within the 1000 s budget.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Basic utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Linear convolution a*b using a real FFT (fast)."""
    m = a.size + b.size - 1
    n_fft = next_pow2(m)
    A = np.fft.rfft(a, n=n_fft)
    B = np.fft.rfft(b, n=n_fft)
    C = A * B
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


def compute_score(a: np.ndarray) -> float:
    """
    Fast replica of `evaluate_sequence` when Σa = 1.
    Returns 2·n·max(a*a) (exactly the value returned by evaluate_sequence
    for a normalised vector).
    """
    n = a.shape[0]
    conv = convolution_fft(a, a)
    return float(2.0 * n * np.max(conv))


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the probability simplex
    (non‑negative entries that sum to 1).
    """
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / v.shape[0])

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, v.shape[0] + 1)
    cond = u - cssv / ind > 0
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


# ----------------------------------------------------------------------
# Top‑K exponentiated sub‑gradient optimizer
# ----------------------------------------------------------------------


def topk_subgrad_optimize(init_a: np.ndarray,
                          time_deadline: float,
                          max_iter: int = 500_000,
                          top_k: int = 12,
                          eta0: float = 0.6) -> tuple[np.ndarray, float]:
    """
    Exponentiated sub‑gradient descent on f(a)=2·n·max(a*a).

    Instead of a single active index we take the `top_k` largest convolution
    entries and sum their sub‑gradients, which provides a more informative
    descent direction.
    """
    a = init_a.astype(float).copy()
    n = a.shape[0]

    best_a = a.copy()
    best_score = compute_score(a)

    for t in range(1, max_iter + 1):
        if time.time() >= time_deadline:
            break

        conv = convolution_fft(a, a)
        max_val = np.max(conv)

        # indices of the K largest entries (unsorted, faster)
        if top_k >= conv.size:
            active = np.arange(conv.size)
        else:
            active = np.argpartition(conv, -top_k)[-top_k:]

        # accumulate sub‑gradient
        g = np.zeros_like(a)
        for k in active:
            # valid i such that 0 ≤ i < n and 0 ≤ k-i < n
            start_i = max(0, k - (n - 1))
            end_i = min(n - 1, k)          # inclusive
            i_idx = np.arange(start_i, end_i + 1)
            j_idx = k - i_idx
            g[i_idx] += 2.0 * a[j_idx]

        # diminishing step size
        eta = eta0 / np.sqrt(t)

        # exponentiated update (keeps a non‑negative)
        a = a * np.exp(-eta * g)

        # projection onto the simplex
        a = project_to_simplex(a)

        cur_score = compute_score(a)
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()

        # occasional tiny random perturbation to escape plateaus
        if t % 5000 == 0:
            jitter = np.random.randn(n) * (eta0 * 1e-4 / np.sqrt(t))
            a = project_to_simplex(a + jitter)

    return best_a, best_score


# ----------------------------------------------------------------------
# Stochastic hill‑climber with optional simulated‑annealing acceptance
# ----------------------------------------------------------------------


def hill_climb(seq,
               time_deadline: float,
               max_steps: int = 200_000) -> tuple[list[float], float]:
    """
    Local search that swaps entries or transfers a fraction of mass.
    Accepts worsening moves with a simple temperature schedule.
    """
    a = np.array(seq, dtype=float)
    n = a.shape[0]

    cur_a = a.copy()
    cur_score = compute_score(cur_a)

    best_a = cur_a.copy()
    best_score = cur_score

    # temperature parameters
    T0 = 0.03
    T_end = 1e-5

    for step in range(max_steps):
        if time.time() >= time_deadline:
            break

        # linear temperature decay
        T = T0 + (T_end - T0) * (step / max_steps)

        # propose a move
        cand = cur_a.copy()
        if np.random.rand() < 0.5:
            # swap two positions
            i, j = np.random.randint(n, size=2)
            cand[i], cand[j] = cand[j], cand[i]
        else:
            # transfer a random fraction of mass from i to j
            i, j = np.random.randint(n, size=2)
            if i != j:
                frac = np.random.rand() * 0.2   # up to 20 % of the source mass
                delta = frac * cand[i]
                cand[i] -= delta
                cand[j] += delta
                cand[i] = max(cand[i], 0.0)

        # re‑project onto the simplex
        cand = project_to_simplex(cand)
        cand_score = compute_score(cand)

        # accept if better, otherwise accept with prob exp(-(Δ)/T)
        if cand_score < cur_score:
            cur_a = cand
            cur_score = cand_score
            if cand_score < best_score:
                best_score = cand_score
                best_a = cand.copy()
        else:
            if T > 0 and np.random.rand() < np.exp(-(cand_score - cur_score) / T):
                cur_a = cand
                cur_score = cand_score

    return best_a.tolist(), best_score


# ----------------------------------------------------------------------
# Optional symmetrisation helper
# ----------------------------------------------------------------------


def symmetrize_and_project(a: np.ndarray) -> np.ndarray:
    """Make the vector symmetric (a_i = a_{n-1-i}) and re‑project onto simplex."""
    sym = (a + a[::-1]) * 0.5
    sym = np.maximum(sym, 0.0)
    return project_to_simplex(sym)


# ----------------------------------------------------------------------
# Main entry point required by the evaluation harness
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 123, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative sequence that minimises `evaluate_sequence`.
    The search runs for up to `budget_s` seconds, trying lengths 2048 and,
    if time permits, 4096, using a top‑K exponentiated sub‑gradient phase
    followed by a stochastic hill‑climber and a final symmetrisation step.
    """
    np.random.seed(seed)
    start_time = time.time()
    # leave a small safety margin for the harness
    deadline = start_time + budget_s - 1.0

    best_seq = None
    best_val = float("inf")

    # --------------------------------------------------------------
    # 0.  Attempt to use the provided fallback sequence (if any)
    # --------------------------------------------------------------
    try:
        fallback = [float(x) for x in height_sequence_1]  # type: ignore
        fallback_val = evaluate_sequence(fallback)
        print(f"Fallback score = {fallback_val:.6f}")
        if fallback_val < best_val:
            best_val = fallback_val
            best_seq = fallback
    except Exception:
        pass

    # --------------------------------------------------------------
    # 1.  Search over a few candidate lengths
    # --------------------------------------------------------------
    candidate_lengths = [2048, 4096]  # start with 2048, try 4096 if time permits

    for n in candidate_lengths:
        if time.time() >= deadline - 20.0:
            break  # not enough time for another full run

        print(f"\n=== Optimising length n = {n} ===")
        attempts = 0
        # allocate at most three independent attempts per length
        while attempts < 3 and time.time() < deadline - 10.0:
            attempts += 1
            # ---- random symmetric start on the simplex ----
            half = n // 2
            half_vec = np.random.dirichlet(np.ones(half))
            a0 = np.concatenate([half_vec,
                                 half_vec[::-1] if n % 2 == 0 else half_vec[-2::-1]])
            a0 = project_to_simplex(a0)   # ensure exact simplex membership

            # ---- Sub‑gradient optimisation (top‑K) ----
            sub_opt_deadline = deadline - 5.0
            a_opt, sub_score = topk_subgrad_optimize(
                a0,
                time_deadline=sub_opt_deadline,
                max_iter=500_000,
                top_k=12,
                eta0=0.6,
            )
            print(f"  Attempt {attempts}: sub‑grad best score = {sub_score:.6f}")

            # ---- Hill‑climbing refinement ----
            hc_deadline = deadline - 2.0
            polished_seq, polished_score = hill_climb(
                a_opt.tolist(),
                time_deadline=hc_deadline,
                max_steps=150_000,
            )
            print(f"  Attempt {attempts}: hill‑climb score = {polished_score:.6f}")

            # ---- Symmetrisation (often beneficial) ----
            arr = np.array(polished_seq, dtype=float)
            sym_arr = symmetrize_and_project(arr)
            sym_score = compute_score(sym_arr)
            if sym_score < polished_score:
                polished_seq = sym_arr.tolist()
                polished_score = sym_score
                print(f"  Attempt {attempts}: after symmetrisation score = {polished_score:.6f}")

            # ---- Evaluate using the official evaluator ----
            true_score = evaluate_sequence(polished_seq)
            print(f"  Attempt {attempts}: true evaluation score = {true_score:.6f}")

            if true_score < best_val:
                best_val = true_score
                best_seq = polished_seq
                print(f"  *** New overall best = {best_val:.6f} (n={n}) ***")

    # --------------------------------------------------------------
    # 2.  Fallback in case nothing was found (should never happen)
    # --------------------------------------------------------------
    if best_seq is None:
        n = 256
        best_seq = [1.0 / n] * n
        best_val = evaluate_sequence(best_seq)

    final_score = evaluate_sequence(best_seq)
    print(f"\nSearch finished. Final reported score = {final_score:.6f} "
          f"with length = {len(best_seq)}")
    return best_seq
