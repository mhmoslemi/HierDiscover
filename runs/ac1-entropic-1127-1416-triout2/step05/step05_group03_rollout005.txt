"""
Propose a step‑function (non‑negative vector) with a low official score.

Algorithm
---------
1. **Broad sampling** – generate many low‑dimensional beta‑family shapes
   (including simple beta densities and convex mixtures) for a fixed
   length `n`.  Keep the top few candidates according to the fast
   surrogate `2·n·max(conv(a,a))` (which equals the official score after
   normalising the sum to 1).

2. **Smooth Adam optimisation** – starting from each kept candidate,
   minimise the log‑sum‑exp smooth approximation of the maximum
   convolution (`max(conv)`).  The temperature `λ` is annealed from 1
   up to a large final value (≈ 500) so that the surrogate becomes
   very close to the true max.  After every Adam step the vector is
   projected onto the probability simplex to keep it non‑negative and
   normalised.

3. **Exact sub‑gradient descent** – switched to the true objective
   `2·n·max(conv)`.  A projected sub‑gradient step with a back‑tracking
   line‑search guarantees that the true max never increases.

4. **Mass‑transfer local search** – a cheap specialised local walk.
   In each move a small amount of mass is transferred from a randomly
   chosen index to another (preserving the sum).  The move is kept if
   it improves the fast score; otherwise it is accepted with a tiny
   uphill probability to escape plateaus.

All phases respect the overall time budget (default 1000 s) and keep
track of the best sequence seen according to the official
`evaluate_sequence`.  If nothing is found a safe uniform fallback is
returned.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Basic linear‑algebra utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Linear convolution via FFT (real inputs)."""
    m = a.size + b.size - 1
    n_fft = next_pow2(m)
    A = np.fft.rfft(a, n=n_fft)
    B = np.fft.rfft(b, n=n_fft)
    C = A * B
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


def max_convolution(a: np.ndarray) -> float:
    """Maximum entry of the self‑convolution of `a`."""
    return float(np.max(convolution_fft(a, a)))


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection onto the probability simplex
    {x ≥ 0, Σx = 1}.  Implementation of Duchi et al. (2008).
    """
    n = v.shape[0]
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / n)

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho = np.where(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def compute_score(a: np.ndarray) -> float:
    """Fast surrogate = 2·n·max(conv(a,a)) (assumes Σa = 1)."""
    n = a.shape[0]
    return 2.0 * n * max_convolution(a)


# ----------------------------------------------------------------------
# Shape generators (beta families and mixtures)
# ----------------------------------------------------------------------


def generate_beta_shape(n: int, alpha: float, beta: float, eps: float) -> np.ndarray:
    """
    Symmetric beta‑family density on {0,…,n‑1}:
        a_i ∝ (i+ε)^{α‑1}·(n‑1‑i+ε)^{β‑1}.
    Returned vector is normalised so that Σa = 1.
    """
    i = np.arange(n, dtype=float)
    left = (i + eps) ** (alpha - 1.0)
    right = (n - 1 - i + eps) ** (beta - 1.0)
    base = left * right
    total = base.sum()
    if total <= 0.0 or not np.isfinite(total):
        return np.full(n, 1.0 / n)
    return base / total


def generate_beta_mixture(
    n: int,
    alpha1: float,
    beta1: float,
    eps1: float,
    alpha2: float,
    beta2: float,
    eps2: float,
    w: float,
) -> np.ndarray:
    """
    Convex combination of two beta‑family densities,
    renormalised to Σ = 1.
    """
    a1 = generate_beta_shape(n, alpha1, beta1, eps1)
    a2 = generate_beta_shape(n, alpha2, beta2, eps2)
    mix = w * a1 + (1.0 - w) * a2
    mix_sum = mix.sum()
    if mix_sum == 0.0 or not np.isfinite(mix_sum):
        return np.full(n, 1.0 / n)
    return mix / mix_sum


def arcsine_initial(n: int) -> np.ndarray:
    """Classic arcsine shape (discrete version)."""
    i = np.arange(n, dtype=float)
    eps = 0.25
    vals = 1.0 / np.sqrt((i + 0.5) * (n - 0.5 - i) + eps)
    vals /= vals.sum()
    return vals


# ----------------------------------------------------------------------
# Smooth surrogate (log‑sum‑exp) and its gradient (Adam)
# ----------------------------------------------------------------------


def smooth_gradient(a: np.ndarray, lam: float) -> tuple[np.ndarray, float]:
    """
    Gradient of the log‑sum‑exp surrogate of max(conv(a,a)).
    Returns (gradient, surrogate_value).
    The objective (for Σa=1) is 2·n·B, where B ≈ max(conv).
    """
    n = a.shape[0]

    conv = convolution_fft(a, a)          # length 2n‑1
    max_c = np.max(conv)

    # soft‑max weights on convolution entries
    exp_term = np.exp(lam * (conv - max_c))
    sum_exp = np.sum(exp_term)
    w = exp_term / sum_exp                # shape (2n‑1,)

    # weighted gradient of the quadratic form a*a
    w_rev = w[::-1]
    conv_rev = convolution_fft(a, w_rev)       # length 3n‑2
    L = 2 * n - 1
    weighted_grad = 2.0 * conv_rev[L - 1 - np.arange(n)]

    grad = 2.0 * n * weighted_grad                # gradient of 2·n·B
    B = max_c + np.log(sum_exp) / lam
    return grad, B


def adam_optimize(
    initial_a: np.ndarray,
    time_deadline: float,
    max_iter: int = 150_000,
    lr0: float = 0.15,
    lam_start: float = 1.0,
    lam_end: float = 500.0,
    noise_every: int = 1000,
    noise_scale: float = 3e-5,
) -> tuple[np.ndarray, float]:
    """
    Adam optimiser on the smooth log‑sum‑exp surrogate.
    Returns the best vector (projected on the simplex) and its fast score.
    """
    a = project_to_simplex(initial_a.astype(float).copy())
    best_a = a.copy()
    best_sc = compute_score(a)

    beta1, beta2 = 0.9, 0.999
    eps_adam = 1e-8
    m = np.zeros_like(a)
    v = np.zeros_like(a)

    n = a.shape[0]

    for it in range(1, max_iter + 1):
        if time.time() >= time_deadline:
            break

        # annealed temperature λ
        t = it / max_iter
        lam = lam_start + (lam_end - lam_start) * (t ** 0.5)

        grad, _ = smooth_gradient(a, lam)

        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)

        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        step = lr0 * m_hat / (np.sqrt(v_hat) + eps_adam)
        a_new = a - step

        if noise_every and (it % noise_every == 0):
            a_new += np.random.randn(n) * noise_scale

        a_new = project_to_simplex(a_new)

        cur_sc = compute_score(a_new)
        if cur_sc < best_sc:
            best_sc = cur_sc
            best_a = a_new.copy()

        a = a_new

        if it % 5000 == 0:
            print(
                f"    Adam iter={it:,d} cur={cur_sc:.6f} best={best_sc:.6f}"
            )

    return best_a, best_sc


# ----------------------------------------------------------------------
# Exact sub‑gradient descent with back‑tracking line‑search
# ----------------------------------------------------------------------


def subgradient_optimize(
    initial_a: np.ndarray,
    time_deadline: float,
    step0: float = 0.3,
    max_iter: int = 200_000,
) -> tuple[np.ndarray, float]:
    """
    Projected sub‑gradient descent on the true max‑convolution.
    Guarantees monotone descent by a simple back‑tracking line‑search.
    """
    a = initial_a.copy()
    best_a = a.copy()
    best_sc = compute_score(a)

    n = a.shape[0]

    for it in range(1, max_iter + 1):
        if time.time() >= time_deadline:
            break

        conv = convolution_fft(a, a)
        max_val = float(np.max(conv))
        idx = int(np.argmax(conv))

        # gradient of max(conv) = 2 * (a shifted by idx)
        i = np.arange(n)
        j = idx - i
        mask = (j >= 0) & (j < n)
        grad = np.zeros_like(a)
        grad[mask] = 2.0 * a[j[mask]]

        step = step0 / np.sqrt(it)
        # back‑track until the max actually decreases
        for _ in range(12):
            a_new = a - step * grad
            a_new = project_to_simplex(a_new)
            new_max = max_convolution(a_new)
            if new_max < max_val - 1e-14:
                break
            step *= 0.5

        cur_sc = compute_score(a_new)
        if cur_sc < best_sc:
            best_sc = cur_sc
            best_a = a_new.copy()

        a = a_new

        if it % 40000 == 0:
            print(
                f"    Sub‑grad iter={it:,d} cur={cur_sc:.6f} best={best_sc:.6f}"
            )

    return best_a, best_sc


# ----------------------------------------------------------------------
# Mass‑transfer local search (pairwise moves)
# ----------------------------------------------------------------------


def mass_transfer_local_search(
    initial_a: np.ndarray,
    time_deadline: float,
    max_iter: int = 200_000,
    rng: np.random.Generator | None = None,
) -> tuple[np.ndarray, float]:
    """
    Very cheap local walk: repeatedly move a small amount of mass from one
    index to another while keeping the vector on the simplex.
    Occasionally accepts a worse move to avoid local traps.
    """
    if rng is None:
        rng = np.random.default_rng()

    a_cur = initial_a.copy()
    best = a_cur.copy()
    best_sc = compute_score(best)
    n = a_cur.shape[0]

    for it in range(1, max_iter + 1):
        if time.time() >= time_deadline:
            break

        # pick a source proportional to its mass (so we usually take from a larger entry)
        src = rng.choice(n, p=a_cur)
        dst = rng.choice(n)
        if src == dst:
            continue

        # amount to transfer – a tiny fraction of the source mass
        eps = rng.uniform(0.0, min(0.02, a_cur[src]))
        if eps <= 0.0:
            continue

        a_new = a_cur.copy()
        a_new[src] -= eps
        a_new[dst] += eps

        # numerical safety – re‑project every few iterations
        if it % 5000 == 0:
            a_new = project_to_simplex(a_new)

        cur_sc = compute_score(a_new)

        if cur_sc < best_sc:
            best_sc = cur_sc
            best = a_new.copy()
            a_cur = a_new.copy()
        else:
            # tiny uphill acceptance probability
            if rng.random() < 1e-4 * np.exp(-(cur_sc - best_sc)):
                a_cur = a_new.copy()

        if it % 50000 == 0:
            print(
                f"    Mass‑local iter={it:,d} cur={cur_sc:.6f} best={best_sc:.6f}"
            )

    return best, best_sc


# ----------------------------------------------------------------------
# Main entry point required by the competition harness
# ----------------------------------------------------------------------


def propose_candidate(
    seed: int = 42,
    budget_s: int = 1000,
    **kwargs,
) -> list[float]:
    """
    Search for a non‑negative step sequence with a low official score.
    The routine uses a quick sampling phase, a long Adam refinement,
    an exact sub‑gradient polishing, and a specialised mass‑transfer
    local walk.  The whole process respects the given time budget.
    """
    rng = np.random.default_rng(seed)
    start = time.time()
    deadline = start + budget_s - 1.0   # leave a safety margin

    best_seq: list[float] = []
    best_val: float = float("inf")

    # --------------------------------------------------------------
    # (optional) use a previously known construction if it exists
    # --------------------------------------------------------------
    try:
        fallback = [float(x) for x in height_sequence_1]  # type: ignore
        val = evaluate_sequence(fallback)
        print(f"Fallback (height_sequence_1) official score = {val:.6f}")
        if val < best_val:
            best_val = val
            best_seq = fallback
    except Exception:
        pass

    # --------------------------------------------------------------
    # 1️⃣  Broad exploratory sampling (fixed length)
    # --------------------------------------------------------------
    n = 2048                     # length that gave good results before
    candidates: list[np.ndarray] = []

    # deterministic starters
    candidates.append(arcsine_initial(n))
    candidates.append(np.full(n, 1.0 / n))

    # pure beta families
    for _ in range(1200):
        if time.time() >= deadline - 5.0:
            break
        alpha = rng.uniform(0.02, 0.9)
        beta  = rng.uniform(0.02, 0.9)
        eps   = rng.uniform(1e-8, 0.03)
        cand = generate_beta_shape(n, alpha, beta, eps)
        candidates.append(cand)

    # mixtures of two betas (more expressive)
    for _ in range(300):
        if time.time() >= deadline - 5.0:
            break
        w = rng.random()
        alpha1 = rng.uniform(0.02, 0.9)
        beta1  = rng.uniform(0.02, 0.9)
        eps1   = rng.uniform(1e-8, 0.03)
        alpha2 = rng.uniform(0.02, 0.9)
        beta2  = rng.uniform(0.02, 0.9)
        eps2   = rng.uniform(1e-8, 0.03)
        cand = generate_beta_mixture(
            n, alpha1, beta1, eps1, alpha2, beta2, eps2, w
        )
        candidates.append(cand)

    # keep the k best according to the fast surrogate
    k = 3
    scored = [(compute_score(c), c) for c in candidates]
    scored.sort(key=lambda x: x[0])
    top_candidates = [c for _, c in scored[:k]]
    print(f"Sampling finished – best surrogate score = {scored[0][0]:.6f}")

    # --------------------------------------------------------------
    # 2️⃣  Refine each top candidate
    # --------------------------------------------------------------
    for idx, cand in enumerate(top_candidates, start=1):
        if time.time() >= deadline - 5.0:
            break
        print(f"\n--- Optimising candidate {idx}/{len(top_candidates)} "
              f"(init score {compute_score(cand):.6f}) ---")

        # ---- Adam on smooth surrogate ----
        remaining = deadline - time.time()
        max_iter_adam = int(min(250_000, remaining * 30))
        a_adam, _ = adam_optimize(
            initial_a=cand,
            time_deadline=deadline,
            max_iter=max_iter_adam,
            lr0=0.18,
            lam_start=1.0,
            lam_end=500.0,
            noise_every=1200,
            noise_scale=4e-5,
        )
        print(f"    After Adam: fast score = {compute_score(a_adam):.6f}")

        # ---- Exact sub‑gradient descent ----
        if time.time() < deadline - 2.0:
            remaining = deadline - time.time()
            max_iter_sub = int(min(120_000, remaining * 20))
            a_sub, _ = subgradient_optimize(
                a_adam,
                time_deadline=deadline,
                step0=0.22,
                max_iter=max_iter_sub,
            )
            print(f"    After sub‑grad: fast score = {compute_score(a_sub):.6f}")
        else:
            a_sub = a_adam

        # ---- Mass‑transfer local walk ----
        if time.time() < deadline - 1.0:
            remaining = deadline - time.time()
            max_iter_loc = int(min(200_000, remaining * 35))
            a_loc, _ = mass_transfer_local_search(
                a_sub,
                time_deadline=deadline,
                max_iter=max_iter_loc,
                rng=rng,
            )
            print(f"    After local walk: fast score = {compute_score(a_loc):.6f}")
        else:
            a_loc = a_sub

        # ---- Official evaluation of the polished candidate ----
        val = evaluate_sequence(a_loc.tolist())
        print(f"    Official score for candidate {idx} = {val:.6f}")
        if val < best_val:
            best_val = val
            best_seq = a_loc.tolist()
            print(f"    *** New global best (score={best_val:.6f}) ***")

    # --------------------------------------------------------------
    # 3️⃣  Global polishing of the overall best (if time permits)
    # --------------------------------------------------------------
    if best_seq and time.time() < deadline - 2.0:
        print("\n=== Global polishing of the current best ===")
        a = np.array(best_seq, dtype=float)

        remaining = deadline - time.time()
        max_iter_adam = int(min(300_000, remaining * 30))
        a, _ = adam_optimize(
            initial_a=a,
            time_deadline=deadline,
            max_iter=max_iter_adam,
            lr0=0.12,
            lam_start=1.0,
            lam_end=600.0,
            noise_every=1500,
            noise_scale=3e-5,
        )
        print(f"    After global Adam: fast score = {compute_score(a):.6f}")

        if time.time() < deadline - 2.0:
            remaining = deadline - time.time()
            max_iter_sub = int(min(150_000, remaining * 25))
            a, _ = subgradient_optimize(
                a,
                time_deadline=deadline,
                step0=0.18,
                max_iter=max_iter_sub,
            )
            print(f"    After global sub‑grad: fast score = {compute_score(a):.6f}")

        if time.time() < deadline - 1.0:
            remaining = deadline - time.time()
            max_iter_loc = int(min(250_000, remaining * 35))
            a, _ = mass_transfer_local_search(
                a,
                time_deadline=deadline,
                max_iter=max_iter_loc,
                rng=rng,
            )
            print(f"    After global local walk: fast score = {compute_score(a):.6f}")

        final_val = evaluate_sequence(a.tolist())
        print(f"Final official score after global polish = {final_val:.6f}")
        if final_val < best_val:
            best_val = final_val
            best_seq = a.tolist()
            print("*** Updated global best after final polish ***")

    # --------------------------------------------------------------
    # 4️⃣  Guard – guarantee a non‑empty return
    # --------------------------------------------------------------
    if not best_seq:
        # Safe uniform fallback (length 256)
        best_seq = [1.0] * 256
        best_val = evaluate_sequence(best_seq)
        print("Fallback to uniform 256‑vector. Official score =", best_val)

    print(f"\nSearch finished. Best reported official score = {best_val:.6f}")
    return best_seq
