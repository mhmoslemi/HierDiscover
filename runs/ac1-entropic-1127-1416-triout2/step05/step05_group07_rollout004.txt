    a = np.maximum(init_a, 1e-30)
    a = a / a.sum()
    best_a = a.copy()
    best_score = compute_score(a)
    n = len(a)
    for it in range(1, max_iter+1):
        if time.time() >= deadline:
            break
        conv = convolution_fft(a, a)  # full conv length 2n-1
        # compute smooth weights
        alpha = alpha_start + (alpha_max - alpha_start) * min(1.0, it / max_iter)  # linear ramp

        factor = np.exp(-eta * grad)

        if it % 2000 == 0:

"""
Hybrid hierarchical search with smoothed‐max gradient descent.

The search works in stages:
  1.  Start from a modest length (≈1024).  For each length we build a small
      library of promising initial vectors (upsampled best, discrete arcsine,
      a few Dirichlet draws and tiny perturbations).
  2.  The candidates are screened with a fast surrogate score
      (FFT based convolution).  The best few are refined with a
      *smoothed‑max* mirror‑descent optimizer:
          •  The objective Φ(a)=max_k (a∗a)_k is approximated by
             Φ_α(a)= (1/α)·log∑_k exp(α·(a∗a)_k).
          •  The gradient of Φ_α is obtainable by a single
             correlation (FFT) with the soft‑max weights.
          •  A multiplicative (exponentiated‑gradient) update keeps the
             iterate on the probability simplex.
          •  α is increased linearly (continuation) so the optimisation
             first sees a smooth landscape and finishes by tightening
             onto the true max.
  3.  After each length the best vector is up‑sampled to the next size.
  4.  When time permits a final polishing with a larger α and a smaller
      learning‑rate is performed.
All work respects the 1000 s wall‑clock budget; the routine always returns
a feasible non‑negative sequence.
"""

# Basic numeric helpers

def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """
    Full linear convolution using a real FFT.
    Returns an array of length len(a)+len(b)-1.
    """
    m = len(a) + len(b) - 1
    nfft = next_pow2(m)
    A = np.fft.rfft(a, n=nfft)
    B = np.fft.rfft(b, n=nfft)
    C = A * B
    conv = np.fft.irfft(C, n=nfft)[:m]
    return conv


def compute_score(a: np.ndarray) -> float:
    """
    Fast surrogate for the official `evaluate_sequence`.
    The value is invariant to scaling of ``a``.
    """
    n = len(a)
    conv = convolution_fft(a, a)
    max_val = float(conv.max())
    s = float(a.sum())
    if s < 1e-12:
        return float("inf")
    return 2.0 * n * max_val / (s * s)


# ----------------------------------------------------------------------
# Families of initial shapes
# ----------------------------------------------------------------------


def arcsine_initial(n: int) -> np.ndarray:
    """Discrete arcsine density a_i ∝ 1/√((i+½)(n‑½‑i))."""
    i = np.arange(n, dtype=float)
    eps = 0.25  # tiny regulariser to avoid division‑by‑zero
    vals = 1.0 / np.sqrt((i + 0.5) * (n - 0.5 - i) + eps)
    return vals / vals.sum()


def random_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    """Draw a Dirichlet‑like vector (Gamma(α,1) with α=conc)."""
    if conc <= 0:
        # heavy‑tailed, near‑sparse draws
        samp = rng.exponential(1.0, size=n) ** (1.0 / max(0.1, -conc))
    else:
        samp = rng.gamma(conc, 1.0, size=n)
    samp += 1e-12
    return samp / samp.sum()


def perturb(a: np.ndarray, rng: np.random.Generator, scale: float = 0.02) -> np.ndarray:
    """Add small Gaussian noise and renormalise."""
    b = a + rng.normal(0.0, scale, size=a.shape)
    b = np.maximum(b, 0.0)
    if b.sum() == 0.0:
        return np.full_like(a, 1.0 / len(a))
    return b / b.sum()


def upsample(prev: np.ndarray, new_n: int) -> np.ndarray:
    """
    Linear interpolation from len(prev) → new_n, followed by re‑normalisation.
    """
    x_old = np.linspace(0.0, 1.0, len(prev))
    x_new = np.linspace(0.0, 1.0, new_n)
    interp = np.interp(x_new, x_old, prev)
    total = interp.sum()
    if total == 0.0:
        return np.full(new_n, 1.0 / new_n)
    return interp / total


# ----------------------------------------------------------------------
# Smoothed‑max (entropy‑regularised) mirror‑descent optimiser
# ----------------------------------------------------------------------


def optimise_smoothed(
    init_a: np.ndarray,
    deadline: float,
    *,
    eta0: float = 0.6,
    alpha_start: float = 1.0,
    alpha_max: float = 200.0,
    max_iter: int = 100_000,
    rng: np.random.Generator = None,
    noise_every: int = 0,
    noise_scale: float = 0.0,
) -> Tuple[np.ndarray, float]:
    """
    Mirror‑descent on the smoothed max objective

        Φ_α(a) = (1/α)·log Σ_k exp(α·(a∗a)_k) ,

    with multiplicative (exponentiated‑gradient) updates.
    The iterate stays on the probability simplex.
    Returns the best vector seen (according to the cheap surrogate
    ``compute_score``) and its surrogate value.
    """
    a = np.maximum(init_a, 1e-30)
    a = a / a.sum()

    best_a = a.copy()
    best_sc = compute_score(a)

    n = len(a)

    for it in range(1, max_iter + 1):
        if time.time() >= deadline:
            break

        # ----- convolution of a with itself -----
        conv = convolution_fft(a, a)          # length 2n‑1

        # ----- continuation on the smoothing parameter -----
        alpha = alpha_start + (alpha_max - alpha_start) * min(1.0, it / max_iter)

        # ----- soft‑max weights (stable) -----
        scaled = alpha * conv
        shift = np.max(scaled)
        exp_vals = np.exp(scaled - shift)
        w = exp_vals / exp_vals.sum()         # shape 2n‑1, sums to 1

        # ----- gradient of Φ_α (correlation) -----
        rev_a = a[::-1]                       # reversed a
        corr = convolution_fft(rev_a, w)      # length 3n‑2
        grad = 2.0 * corr[n - 1 : n - 1 + n]   # vector length n

        if noise_every and (it % noise_every == 0):
            grad += rng.normal(0.0, noise_scale, size=n)

        # ----- step‑size schedule -----
        eta = eta0 / np.sqrt(it)

        # ----- multiplicative update + re‑normalisation -----
        factor = np.exp(-eta * grad)
        a = a * factor
        s = a.sum()
        if s == 0.0:
            a = np.full_like(a, 1.0 / n)
        else:
            a = a / s

        # ----- keep best according to surrogate score -----
        cur_sc = compute_score(a)
        if cur_sc < best_sc:
            best_sc = cur_sc
            best_a = a.copy()

        if it % 2000 == 0:
            print(
                f"    smooth iter={it:,d} cur={cur_sc:.6f} best={best_sc:.6f} "
                f"alpha={alpha:.2f}"
            )

    return best_a, best_sc


# ----------------------------------------------------------------------
# Main entry point required by the platform
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs) -> List[float]:
    """
    Hierarchical search for a non‑negative step‑function with a low
    ``evaluate_sequence`` value.  The routine respects the supplied time
    budget, prints progress, and finally returns the best sequence it
    discovered.
    """
    rng = np.random.default_rng(seed)
    start = time.time()
    deadline = start + budget_s - 2.0  # tiny safety margin

    best_seq: List[float] = []
    best_val: float = float("inf")
    best_internal: np.ndarray = None  # cached best vector on the simplex

    # ------------------------------------------------------------------
    # 0.  Optional fallback from a previously discovered construction.
    # ------------------------------------------------------------------
    try:
        fallback = [float(x) for x in height_sequence_1]  # type: ignore
        val = evaluate_sequence(fallback)
        print(f"Fallback (height_sequence_1) score = {val:.6f}")
        if val < best_val:
            best_val = val
            best_seq = fallback
            best_internal = np.array(fallback, dtype=float)
            best_internal /= best_internal.sum()
    except Exception:
        pass

    # ------------------------------------------------------------------
    # 1.  Length hierarchy – start small, enlarge progressively.
    # ------------------------------------------------------------------
    lengths = [1024, 2048, 4096, 8192, 16384]  # larger values give finer grids

    for n in lengths:
        if time.time() >= deadline - 5.0:
            break
        print(f"\n=== Exploring length n = {n} ===")

        # ----- 1a. Build a small library of start points -----
        candidates: List[np.ndarray] = []

        if best_internal is not None:
            candidates.append(upsample(best_internal, n))

        candidates.append(arcsine_initial(n))

        for conc in (0.3, 0.7, 1.0, 2.0):
            candidates.append(random_dirichlet(n, conc, rng))

        for _ in range(2):
            candidates.append(perturb(arcsine_initial(n), rng, scale=0.02))

        # ----- 1b. Quick screening with the cheap surrogate -----
        scored = [(compute_score(c), c) for c in candidates]
        scored.sort(key=lambda x: x[0])
        top_candidates = [c for _, c in scored[:3]]

        # ----- 1c. Refine each retained candidate -----
        for idx, cand in enumerate(top_candidates):
            if time.time() >= deadline - 5.0:
                break
            print(f"  candidate {idx + 1}/{len(top_candidates)} (n={n})")

            # crude estimate of how many iterations we can still afford
            remaining = max(0.0, deadline - time.time())
            max_iter = int(min(80_000, max(5_000, remaining * 120)))

            refined, _ = optimise_smoothed(
                init_a=cand,
                deadline=deadline,
                eta0=0.6,
                alpha_start=1.0,
                alpha_max=200.0,
                max_iter=max_iter,
                rng=rng,
                noise_every=800,
                noise_scale=5e-6,
            )

            # official evaluation (expensive) – called only for the few best candidates
            val = evaluate_sequence(refined.tolist())
            print(f"    evaluated score = {val:.6f}")

            if val < best_val:
                best_val = val
                best_seq = refined.tolist()
                best_internal = refined.copy()
                print(f"    *** New global best: {best_val:.6f} (n={n}) ***")

    # ------------------------------------------------------------------
    # 2.  Final polishing of the overall best (if time permits).
    # ------------------------------------------------------------------
    if best_seq and time.time() < deadline - 5.0:
        a = np.array(best_seq, dtype=float)
        remaining = max(0.0, deadline - time.time())
        max_iter = int(min(300_000, remaining * 120))

        print("\nFinal polishing of the global best")
        a_ref, _ = optimise_smoothed(
            init_a=a,
            deadline=deadline,
            eta0=0.3,          # finer steps
            alpha_start=10.0,
            alpha_max=400.0,
            max_iter=max_iter,
            rng=rng,
            noise_every=0,
            noise_scale=0.0,
        )
        final_val = evaluate_sequence(a_ref.tolist())
        print(f"Final polished score = {final_val:.6f}")
        if final_val < best_val:
            best_val = final_val
            best_seq = a_ref.tolist()
            print("*** Updated global best after final polish ***")

    # ------------------------------------------------------------------
    # 3.  Guard – always return a feasible sequence.
    # ------------------------------------------------------------------
    if not best_seq:
        best_seq = [1.0] * 256
        best_val = evaluate_sequence(best_seq)

    print(f"\nSearch finished. Best reported score = {best_val:.6f}")
    return best_seq
