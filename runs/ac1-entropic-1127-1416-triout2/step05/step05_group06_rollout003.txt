def build_conv_matrix(f):
    n = len(f)
    rows = []
    cols = []
    data = []
    for k in range(2*n-1):
        lo = max(0, k-(n-1))
        hi = min(k, n-1)
        # iterate over j range
        for j in range(lo, hi+1):
            i = k - j
            rows.append(k)
            cols.append(j)
            data.append(f[i])
    A = scipy.sparse.csr_matrix((data, (rows, cols)), shape=(2*n-1, n), dtype=np.float64)
    return A

def lp_optimize(f, M, rng):
    import scipy.sparse
    from scipy.optimize import linprog

    n = len(f)
    A = build_conv_matrix(f)  # shape (2n-1, n)
    b = np.full(2*n-1, M, dtype=np.float64)
    # objective maximize sum g => minimize -sum
    c = -np.ones(n, dtype=np.float64)
    bounds = [(0, None)] * n
    try:
        res = linprog(c, A_ub=A, b_ub=b, bounds=bounds, method='highs')
    except Exception as e:
        return None, False
    if res.success:
        g = res.x
        if g.sum() == 0:
            return None, False
        g = g / g.sum()
        return g, True
    else:
        return None, False
```

"""
Hybrid optimizer for the “max‑convolution” step‑function problem.

The algorithm proceeds as follows (within the 1000 s wall‑clock budget).

1. **Seed generation** – a diverse library of symmetric candidate vectors is
   built for a few FFT‑friendly lengths (1024, 2048, 4096).  Families include
   arcsine, power‑law, beta‑like, Dirichlet, uniform and simple mixtures.
   All candidates are symmetrised and normalised to sum = 1.

2. **Fast refinement** – each seed is refined with a smooth‑max gradient
   descent (log‑sum‑exp surrogate).  This gives a quick reduction of the
   objective using O(n log n) FFTs per iteration.

3. **LP‑based improvement** – given the current vector `f`, we form the linear
   program

        maximise   Σ_j g_j
        subject to  (f * g)_k ≤ M   for every k,   g_j ≥ 0,

   where `M = max_k (f * f)_k`.  The matrix of the convolution constraints is
   a (2n‑1)×n Toeplitz matrix built with a sparse CSR representation.
   Solving the LP (SciPy HiGHS) yields a feasible `g` with the largest possible
   ℓ₁‑norm under the same max‑convolution bound.  After normalising `g`,
   we try several convex combinations `(1−t)f + t g` (t∈{0.25,0.5,0.75,1.0})
   and keep the best improvement.  This step is repeated a few times or until
   no further gain is observed.

4. **Final polishing** – the best vector found so far is polished with the
   exponentiated‑sub‑gradient method that works directly on the true max‑convolution.
   This step is cheap (single FFT per iteration) and provides the final
   refinement.

All intermediate vectors are forced to be non‑negative, symmetric and
renormalised to have unit sum; the objective is invariant under scaling, so
this does not affect the final score.  Progress and timing information are
printed for transparency.  The routine returns a feasible list of floats that
minimises `evaluate_sequence`.
"""

import time
import numpy as np
from typing import List, Tuple

# ----------------------------------------------------------------------
# Core numeric utilities
# ----------------------------------------------------------------------
def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full linear convolution of real vectors using an FFT of power‑of‑2 length."""
    m = len(a) + len(b) - 1
    n_fft = next_pow2(m)
    A = np.fft.rfft(a, n=n_fft)
    B = np.fft.rfft(b, n=n_fft)
    C = A * B
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


def compute_score(a: np.ndarray) -> float:
    """Fast surrogate of the official objective (scale‑invariant)."""
    n = len(a)
    conv = convolution_fft(a, a)
    max_val = float(np.max(conv))
    s = float(np.sum(a))
    if s < 1e-12:
        return float("inf")
    return 2.0 * n * max_val / (s * s)


def symmetrize(a: np.ndarray) -> np.ndarray:
    """Enforce a ↔ a[::-1] symmetry and renormalise to sum = 1."""
    b = (a + a[::-1]) * 0.5
    s = b.sum()
    if s == 0.0:
        return np.full_like(b, 1.0 / len(b))
    return b / s


# ----------------------------------------------------------------------
# Families of initial shapes (all returned normalised to sum = 1 and symmetric)
# ----------------------------------------------------------------------
def arcsine_initial(n: int) -> np.ndarray:
    i = np.arange(n, dtype=float)
    eps = 0.25
    vals = 1.0 / np.sqrt((i + 0.5) * (n - 0.5 - i) + eps)
    vals = vals / vals.sum()
    return symmetrize(vals)


def generate_powerlaw(n: int, p: float, eps: float) -> np.ndarray:
    i = np.arange(n, dtype=float)
    base = ((i + eps) * (n - 1 - i + eps)) ** (-p)
    total = base.sum()
    if total == 0.0:
        base = np.full(n, 1.0 / n)
    else:
        base = base / total
    return symmetrize(base)


def generate_beta_like(n: int, alpha: float, beta: float) -> np.ndarray:
    i = np.arange(n, dtype=float)
    base = (i + 0.5) ** (alpha - 1) * (n - 1 - i + 0.5) ** (beta - 1)
    total = base.sum()
    if total == 0.0:
        base = np.full(n, 1.0 / n)
    else:
        base = base / total
    return symmetrize(base)


def random_dirichlet(n: int, concentration: float, rng: np.random.Generator) -> np.ndarray:
    if concentration <= 0:
        samp = rng.exponential(1.0, size=n) ** (1.0 / max(0.1, concentration))
    else:
        samp = rng.gamma(concentration, 1.0, size=n)
    samp = np.maximum(samp, 1e-12)
    return symmetrize(samp)


def mixture(a: np.ndarray, b: np.ndarray, w: float) -> np.ndarray:
    c = w * a + (1.0 - w) * b
    return symmetrize(c)


# ----------------------------------------------------------------------
# Smooth‑max (log‑sum‑exp) gradient descent
# ----------------------------------------------------------------------
def softmax_gradient_descent(initial_a: np.ndarray,
                             deadline: float,
                             betas: List[float],
                             iter_per_beta: int = 500,
                             eta0: float = 0.5) -> Tuple[np.ndarray, float]:
    """Minimise the smooth surrogate Φ_β via exponentiated‑gradient updates."""
    a = np.maximum(initial_a, 1e-30)
    a = a / a.sum()
    best_a = a.copy()
    best_sc = compute_score(a)
    n = len(a)

    for beta in betas:
        for it in range(1, iter_per_beta + 1):
            if time.time() >= deadline:
                return best_a, best_sc

            # conv = a * a
            conv = convolution_fft(a, a)                       # length 2n‑1
            mval = conv.max()
            # soft‑max weights (stable)
            exp_vals = np.exp(beta * (conv - mval))
            p = exp_vals / exp_vals.sum()                      # Σ p = 1

            # cross‑correlation p ⋆ a  (size n)
            N = next_pow2(len(p) + n - 1)
            P = np.fft.fft(p, N)
            A = np.fft.fft(a, N)
            cross = np.fft.ifft(P * np.conj(A)).real[:n]
            cross = np.maximum(cross, 0.0)

            # gradient of Φ_β
            grad = 4.0 * n * cross

            eta = eta0 / np.sqrt(it)
            a = a * np.exp(-eta * grad)

            # project onto simplex
            s = a.sum()
            if s == 0.0:
                a = np.full_like(a, 1.0 / n)
            else:
                a = a / s

            cur_sc = compute_score(a)
            if cur_sc < best_sc:
                best_sc = cur_sc
                best_a = a.copy()

    return best_a, best_sc


# ----------------------------------------------------------------------
# Exponentiated sub‑gradient on the true max‑convolution (polishing)
# ----------------------------------------------------------------------
def _max_conv_gradient(a: np.ndarray, conv: np.ndarray) -> np.ndarray:
    """Sub‑gradient of max_k (a∗a)_k."""
    n = len(a)
    max_val = conv.max()
    mask = np.isclose(conv, max_val, atol=1e-15, rtol=0.0)
    active = np.where(mask)[0]
    grad = np.zeros_like(a)

    for k in active:
        lo = max(0, k - (n - 1))
        hi = min(k, n - 1)
        idx = np.arange(lo, hi + 1, dtype=int)
        grad[idx] += 2.0 * a[k - idx]

    if len(active) > 1:
        grad /= len(active)
    return grad


def optimise_subgradient(initial_a: np.ndarray,
                         deadline: float,
                         *,
                         eta0: float = 0.6,
                         max_iter: int = 200_000,
                         noise_every: int = 0,
                         noise_scale: float = 0.0) -> Tuple[np.ndarray, float]:
    """Exponentiated sub‑gradient descent directly on Φ(a)=max_k (a∗a)_k."""
    a = np.maximum(initial_a, 1e-30)
    a = a / a.sum()
    best_a = a.copy()
    best_sc = compute_score(a)
    n = len(a)

    for it in range(1, max_iter + 1):
        if time.time() >= deadline:
            break

        conv = convolution_fft(a, a)
        grad = _max_conv_gradient(a, conv)

        eta = eta0 / np.sqrt(it)
        a = a * np.exp(-eta * grad)

        if noise_every and (it % noise_every == 0):
            a += np.random.randn(n) * noise_scale
            a = np.maximum(a, 0.0)

        s = a.sum()
        a = a / s if s != 0.0 else np.full_like(a, 1.0 / n)

        cur_sc = compute_score(a)
        if cur_sc < best_sc:
            best_sc = cur_sc
            best_a = a.copy()

        if it % 2000 == 0:
            print(f"    subgrad iter={it:,d} cur={cur_sc:.6f} best={best_sc:.6f}")

    return best_a, best_sc


# ----------------------------------------------------------------------
# Linear‑programming improvement (core novelty)
# ----------------------------------------------------------------------
def _build_conv_matrix(f: np.ndarray):
    """Construct the (2n‑1)×n convolution matrix for vector f (sparse CSR)."""
    import scipy.sparse as sp
    n = len(f)
    rows = []
    cols = []
    data = []
    # For each output index k of conv(f,g) we add the appropriate f[i] coefficients.
    for k in range(2 * n - 1):
        lo = max(0, k - (n - 1))
        hi = min(k, n - 1)
        # j runs over columns where the overlap exists
        js = np.arange(lo, hi + 1, dtype=int)
        is_ = k - js
        rows.extend([k] * len(js))
        cols.extend(js.tolist())
        data.extend(f[is_].tolist())
    return sp.csr_matrix((data, (rows, cols)), shape=(2 * n - 1, n), dtype=np.float64)


def lp_optimize(f: np.ndarray, M: float, rng: np.random.Generator) -> Tuple[np.ndarray, bool]:
    """
    Solve the LP:
        maximise Σ g_j
        s.t.   (f * g)_k ≤ M   ∀k,   g_j ≥ 0.
    Returns a normalized feasible vector `g` (unit ℓ₁ norm) and a success flag.
    """
    import scipy.optimize as opt
    n = len(f)
    # Build the constraint matrix only once per LP solve.
    A = _build_conv_matrix(f)
    b = np.full(2 * n - 1, M, dtype=np.float64)

    # maximise Σ g_j  ⇔ minimise −Σ g_j
    c = -np.ones(n, dtype=np.float64)
    bounds = [(0, None)] * n

    try:
        res = opt.linprog(c, A_ub=A, b_ub=b, bounds=bounds, method="highs")
    except Exception as e:
        # In the very unlikely event HiGHS is not available.
        return np.empty(0), False

    if not res.success or res.x is None:
        return np.empty(0), False

    g = res.x
    if g.sum() <= 0.0:
        return np.empty(0), False
    g = g / g.sum()
    return g, True


def lp_improvement_loop(f: np.ndarray,
                        deadline: float,
                        rng: np.random.Generator,
                        max_iters: int = 6) -> np.ndarray:
    """
    Repeatedly apply the LP improvement to `f`, trying several convex mixes.
    Stops when no improvement is observed or the time budget is exhausted.
    """
    best = f.copy()
    best_sc = compute_score(best)
    for it in range(max_iters):
        if time.time() >= deadline:
            break
        conv = convolution_fft(best, best)
        M = float(conv.max())

        g, ok = lp_optimize(best, M, rng)
        if not ok:
            break

        # Try a few mixing coefficients.
        candidates = []
        for t in (0.25, 0.5, 0.75, 1.0):
            cand = (1.0 - t) * best + t * g
            cand = symmetrize(cand)  # keep symmetry invariant
            candidates.append(cand)

        improved = False
        for cand in candidates:
            sc = compute_score(cand)
            if sc < best_sc - 1e-12:       # strict improvement
                best = cand
                best_sc = sc
                improved = True
                # print progress
                print(f"    LP iter {it+1}, mix t={t:.2f}, new score {sc:.6f}")
                break
        if not improved:
            break
    return best
