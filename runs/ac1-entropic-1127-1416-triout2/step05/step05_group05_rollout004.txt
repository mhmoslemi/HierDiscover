"""
Search for a low‑max‑auto‑correlation step function.

The algorithm performs a multi‑start, temperature‑annealed exponentiated‑gradient
search on the simplex (a = softmax(x)).  Each iterate enforces symmetry
(a_i = a_{n‑1‑i}) because the objective is invariant under reversal.
A first short smoothing phase is run on many initial candidates
(arcsine, uniform, random Dirichlet, power‑law, previous best).  The best
candidate is then refined with a longer smoothing run where the soft‑max
temperature β is slowly increased, finally a hard sub‑gradient polish using the
exact max‑weight is applied.  All steps respect the overall time budget and
fallback to a uniform vector if nothing is found.
"""

import time
import numpy as np
from typing import List, Tuple

# ----------------------------------------------------------------------
# Low‑level numeric utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """
    Linear convolution via real FFT (full length, zero‑padded to a power‑of‑2).
    """
    m = len(a) + len(b) - 1
    n_fft = next_pow2(m)
    A = np.fft.rfft(a, n=n_fft)
    B = np.fft.rfft(b, n=n_fft)
    C = A * B
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


def normalize(v: np.ndarray) -> np.ndarray:
    """Project a non‑negative vector onto the simplex (sum = 1)."""
    total = v.sum()
    if total <= 0.0:
        return np.full_like(v, 1.0 / len(v))
    return v / total


def compute_score(a: np.ndarray) -> float:
    """
    Fast evaluation of the official objective (scale‑invariant):
        2·n·max(conv(a,a)) / (sum(a)²)
    The vector a is assumed non‑negative and summed to 1, so the formula reduces
    to 2·n·max(conv).
    """
    n = len(a)
    conv = convolution_fft(a, a)
    max_val = float(np.max(conv))
    return 2.0 * n * max_val


def gradient_from_weights(a: np.ndarray, w: np.ndarray) -> np.ndarray:
    """
    Gradient of Σ_k w_k·(a*a)_k w.r.t. a.
    Implements   g_j = 2· Σ_m a_m · w_{m+j}.
    Computed via one FFT‑based convolution.
    """
    # reverse w because convolution with reversed weight yields the needed sums
    w_rev = w[::-1]
    conv = convolution_fft(a, w_rev)               # length = 3·n‑2
    idx = 2 * len(a) - 2                            # position 2n‑2
    grads = 2.0 * conv[idx - np.arange(len(a))]
    return grads


# ----------------------------------------------------------------------
# Parametric families for the initial sweep
# ----------------------------------------------------------------------


def arcsine_initial(n: int) -> np.ndarray:
    """Discrete arcsine distribution a_i ∝ 1/√((i+½)(n‑½‑i))."""
    i = np.arange(n, dtype=float)
    eps = 0.25
    vals = 1.0 / np.sqrt((i + 0.5) * (n - 0.5 - i) + eps)
    return normalize(vals)


def powerlaw_shape(n: int, p: float, eps: float) -> np.ndarray:
    """Symmetric power‑law shape a_i ∝ ((i+ε)(n‑1‑i+ε))⁻ᵖ."""
    i = np.arange(n, dtype=float)
    base = ((i + eps) * (n - 1 - i + eps)) ** (-p)
    return normalize(base)


def uniform_shape(n: int) -> np.ndarray:
    """Uniform probability vector on n points."""
    return np.full(n, 1.0 / n)


def random_dirichlet(n: int, rng: np.random.Generator) -> np.ndarray:
    """Draw a random point on the simplex (exponential sampling)."""
    samp = rng.exponential(1.0, size=n)
    return normalize(samp)


def enforce_symmetry(a: np.ndarray) -> np.ndarray:
    """Project onto symmetric vectors (a_i = a_{n‑1‑i}) and renormalise."""
    sym = (a + a[::-1]) * 0.5
    return normalize(sym)


# ----------------------------------------------------------------------
# Refinement methods
# ----------------------------------------------------------------------


def smooth_descent(
    a_init: np.ndarray,
    deadline: float,
    *,
    beta_start: float = 0.5,
    beta_end: float = 30.0,
    eta0: float = 1.0,
    jitter: float = 0.0,
    rng: np.random.Generator,
    enforce_sym: bool = True,
) -> Tuple[np.ndarray, float]:
    """
    Temperature‑annealed exponentiated‑gradient descent.
    Weight vector w is a soft‑max over the convolution entries:
        w_i ∝ exp(β·(conv_i – max(conv))).

    Returns the best iterate (according to the exact objective) and its
    score 2·n·max(conv).  The temperature β grows linearly from beta_start to
    beta_end over the allocated time.  Optional jitter adds a tiny
    random perturbation every few hundred steps to help escape plateaus.
    """
    a = normalize(a_init.astype(float).copy())
    n = len(a)

    best_a = a.copy()
    conv = convolution_fft(a, a)
    best_sc = 2.0 * n * float(np.max(conv))

    start = time.time()
    it = 0
    while time.time() < deadline:
        it += 1

        # --------------------------------------------------------------
        # 1) convolution and temperature schedule
        # --------------------------------------------------------------
        conv = convolution_fft(a, a)
        max_val = float(np.max(conv))

        elapsed = time.time() - start
        total = deadline - start
        beta = beta_start + (beta_end - beta_start) * (elapsed / total)

        # soft‑max weights (max entry gets weight 1)
        exp_vals = np.exp(beta * (conv - max_val))
        w_sum = np.sum(exp_vals)
        if w_sum == 0:
            w = np.full_like(conv, 1.0 / len(conv))
        else:
            w = exp_vals / w_sum

        # --------------------------------------------------------------
        # 2) gradient of the smoothed objective
        # --------------------------------------------------------------
        grad = gradient_from_weights(a, w)

        # --------------------------------------------------------------
        # 3) multiplicative update (exponentiated gradient)
        # --------------------------------------------------------------
        eta = eta0 / np.sqrt(it)
        a = a * np.exp(-eta * grad)

        # optional tiny random jitter every 200 iterations
        if jitter > 0.0 and (it % 200 == 0):
            a = a * (1.0 + jitter * rng.standard_normal(n))
            a = np.maximum(a, 0.0)

        # --------------------------------------------------------------
        # 4) enforce symmetry + renormalise
        # --------------------------------------------------------------
        if enforce_sym:
            a = enforce_symmetry(a)
        else:
            a = normalize(a)

        # --------------------------------------------------------------
        # 5) record best exact score
        # --------------------------------------------------------------
        cur_sc = 2.0 * n * float(np.max(conv))
        if cur_sc < best_sc:
            best_sc = cur_sc
            best_a = a.copy()

    return best_a, best_sc


def subgradient_descent(
    a_init: np.ndarray,
    deadline: float,
    *,
    eta0: float = 0.7,
    jitter: float = 0.0,
    rng: np.random.Generator,
    enforce_sym: bool = True,
) -> Tuple[np.ndarray, float]:
    """
    Classical sub‑gradient descent on the exact max‑convolution objective.
    The weight vector w is uniform on the (near‑)maximising entries.
    Uses exponentiated‑gradient updates with step size η_t = η₀ / √t.
    """
    a = normalize(a_init.astype(float).copy())
    n = len(a)

    best_a = a.copy()
    conv = convolution_fft(a, a)
    best_sc = 2.0 * n * float(np.max(conv))

    it = 0
    while time.time() < deadline:
        it += 1

        conv = convolution_fft(a, a)
        max_val = float(np.max(conv))

        # binary weight on maximisers (within a tiny tolerance)
        w = np.zeros_like(conv)
        max_idx = np.where(conv >= max_val - 1e-15)[0]
        if len(max_idx) == 0:  # very unlikely
            w[:] = 1.0 / len(conv)
        else:
            w[max_idx] = 1.0 / len(max_idx)

        grad = gradient_from_weights(a, w)

        eta = eta0 / np.sqrt(it)
        a = a * np.exp(-eta * grad)

        if jitter > 0.0 and (it % 200 == 0):
            a = a * (1.0 + jitter * rng.standard_normal(n))
            a = np.maximum(a, 0.0)

        if enforce_sym:
            a = enforce_symmetry(a)
        else:
            a = normalize(a)

        cur_sc = 2.0 * n * float(np.max(conv))
        if cur_sc < best_sc:
            best_sc = cur_sc
            best_a = a.copy()

    return best_a, best_sc

    if not best_seq:

        best_seq = uniform_shape(256).tolist()
        best_val = evaluate_sequence(best_seq)

    print(f"\nSearch finished. Best reported official score = {best_val:.6f}")
    return best_seq
