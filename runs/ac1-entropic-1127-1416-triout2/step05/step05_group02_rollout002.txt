def freq_mutate(v, sigma):
    # compute FFT of v (real)
    ft = np.fft.rfft(v)
    # apply multiplicative Gaussian noise to each coefficient (except the DC component maybe)
    noise = np.random.normal(1.0, sigma, size=ft.shape)
    ft_mut = ft * noise
    v_mut = np.fft.irfft(ft_mut, n=len(v))
    # Clip negative values to zero
    v_mut = np.maximum(v_mut, 0.0)
    # Projection onto simplex
    v_mut = project_to_simplex(v_mut)
    return v_mut

"""
Propose a low‑discrepancy step function.

Algorithm
---------
1.  **Generate a diverse set of seeds** – a mixture of analytic families
    (arcsine, beta, power, Gaussian, cosine, triangular, uniform) and a
    pool of random Dirichlet vectors, all normalised to Σa = 1.

2.  **Evolutionary search** – a small population (≈30) evolves on the
    fast surrogate ``score = 2·n·max(conv(a,a))``.
    * elitist selection (keep the best 5)
    * crossover (average of two parents)
    * two mutation operators:
        - entry‑wise Gaussian noise,
        - multiplicative noise in the Fourier domain (preserves smoothness)
    * mutation strength decays with the number of generations.
    The loop runs until only a few seconds are left.

3.  **Smoothed gradient descent** – starting from the best vector,
    minimise the smooth approximation of the max:
        fβ(a) = (2·n/β)·log Σk exp(β·convk(a,a)).
    The gradient is computed analytically via one extra convolution.
    β is increased from a small to a large value, turning the objective
    gradually into the true max.  A diminishing step‑size ηt = η0/√t
    together with Euclidean projection onto the simplex drives the iterate.

4.  **Exact sub‑gradient descent + exponentiated‑gradient polish** –
    finish the optimisation with the same primitives that were used in
    the previous best solution.

The whole routine respects a global time budget (default 1000 s) and
always returns a non‑empty list of non‑negative numbers.  The final
candidate is evaluated with the official ``evaluate_sequence`` and
the best one found is returned.
"""

import time
import numpy as np
from typing import List, Tuple

# ----------------------------------------------------------------------
# Low‑level utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Linear convolution via real FFT (length = len(a)+len(b)-1)."""
    m = len(a) + len(b) - 1
    n_fft = next_pow2(m)
    A = np.fft.rfft(a, n=n_fft)
    B = np.fft.rfft(b, n=n_fft)
    C = A * B
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


def compute_score(a: np.ndarray) -> float:
    """
    Fast surrogate for the official score.
    The vector is assumed to be normalised to Σa = 1, therefore the
    denominator disappears and the score is simply 2·n·max(conv(a,a)).
    """
    n = a.shape[0]
    conv = convolution_fft(a, a)
    return 2.0 * n * float(np.max(conv))


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection onto the probability simplex {x ≥ 0, Σx = 1}.
    Duchi et al., 2008.
    """
    if np.all(v <= 0):
        # Uniform fallback – should never happen in practice
        return np.full_like(v, 1.0 / v.size)

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho_arr = u - cssv / np.arange(1, v.size + 1)
    rho = np.where(rho_arr > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w

# Analytic shape generators
# ----------------------------------------------------------------------


def arcsine_initial(n: int) -> np.ndarray:
    """Classic arcsine shape: a_i ∝ 1/√((i+½)*(n‑½‑i))."""
    i = np.arange(n, dtype=float)
    eps = 0.25
    vals = 1.0 / np.sqrt((i + 0.5) * (n - 0.5 - i) + eps)
    vals /= vals.sum()
    return vals


def generate_beta_shape(n: int, alpha: float, beta: float, eps: float) -> np.ndarray:
    """Beta‑type shape (i+ε)^{α‑1}·(n‑1‑i+ε)^{β‑1}."""
    i = np.arange(n, dtype=float)
    left = (i + eps) ** (alpha - 1.0)
    right = (n - 1 - i + eps) ** (beta - 1.0)
    base = left * right
    total = base.sum()
    if total <= 0.0 or not np.isfinite(total):
        return np.full(n, 1.0 / n)
    return base / total


def generate_power_shape(n: int, p: float, eps: float) -> np.ndarray:
    """Generalised arcsine: a_i ∝ ((i+½)*(n‑½‑i)+ε)^{‑p}."""
    i = np.arange(n, dtype=float)
    denom = ((i + 0.5) * (n - 0.5 - i) + eps) ** p
    vals = 1.0 / denom
    total = vals.sum()
    if total <= 0.0 or not np.isfinite(total):
        return np.full(n, 1.0 / n)
    return vals / total


def generate_gaussian_shape(n: int, sigma: float) -> np.ndarray:
    """Discrete Gaussian centred at the middle."""
    i = np.arange(n, dtype=float)
    centre = (n - 1) / 2.0
    vals = np.exp(-0.5 * ((i - centre) / sigma) ** 2)
    total = vals.sum()
    if total <= 0.0 or not np.isfinite(total):
        return np.full(n, 1.0 / n)
    return vals / total


def generate_cosine_shape(n: int, power: float) -> np.ndarray:
    """Cosine‑type bump: a_i ∝ sin^power(π·(i+½)/n)."""
    i = np.arange(n, dtype=float)
    vals = np.sin(np.pi * (i + 0.5) / n) ** power
    total = vals.sum()
    if total <= 0.0 or not np.isfinite(total):
        return np.full(n, 1.0 / n)
    return vals / total


def generate_triangular_shape(n: int, peak: int) -> np.ndarray:
    """Symmetric triangular bump with its apex at ``peak``."""
    left = np.arange(peak + 1, dtype=float)
    right = np.arange(n - peak - 1, dtype=float)[::-1]
    seq = np.concatenate([left, right])
    seq /= seq.sum()
    return seq


def generate_mixture_shape(n: int, weight_arc: float) -> np.ndarray:
    """Mixture (weight_arc)·arcsine + (1‑weight_arc)·uniform."""
    arc = arcsine_initial(n)
    uni = np.full(n, 1.0 / n)
    mix = weight_arc * arc + (1.0 - weight_arc) * uni
    mix /= mix.sum()
    return mix


def random_dirichlet(n: int, alpha: float = 1.0) -> np.ndarray:
    """Sample from a Dirichlet distribution (simplex)."""
    draws = np.random.gamma(alpha, 1.0, size=n)
    draws /= draws.sum()
    return draws


# ----------------------------------------------------------------------
# Mutation operators
# ----------------------------------------------------------------------


def mutate_entry(v: np.ndarray, sigma: float) -> np.ndarray:
    """Add entry‑wise Gaussian noise and project back onto the simplex."""
    pert = v + np.random.normal(0.0, sigma, size=v.shape)
    pert = np.maximum(pert, 0.0)
    return project_to_simplex(pert)


def mutate_freq(v: np.ndarray, sigma: float) -> np.ndarray:
    """
    Apply multiplicative Gaussian noise to the (real) Fourier coefficients,
    inverse‑transform, clip negatives and project onto the simplex.
    """
    ft = np.fft.rfft(v)
    noise = np.random.normal(1.0, sigma, size=ft.shape)
    ft_mut = ft * noise
    v_mut = np.fft.irfft(ft_mut, n=v.shape[0])
    v_mut = np.maximum(v_mut, 0.0)
    return project_to_simplex(v_mut)

# Exact optimisation primitives (as in the reference solution)
# ----------------------------------------------------------------------


def subgradient_descent_exact(initial_a: np.ndarray,
                              deadline: float,
                              max_iter: int = 80_000,
                              step0: float = 0.05) -> np.ndarray:
    """Projected sub‑gradient descent on   f(a)=2·n·max(conv(a,a))."""
    a = project_to_simplex(initial_a.copy())
    n = len(a)
    L = 2 * n - 1

    for it in range(1, max_iter + 1):
        if time.time() >= deadline:
            break

        conv = convolution_fft(a, a)
        max_val = float(np.max(conv))

        # tolerance for equality (floating point)
        tol = max_val * 1e-12 + 1e-14
        mask = (np.abs(conv - max_val) <= tol).astype(float)

        # sub‑gradient = 2·∑_{k∈K} shift(a,k)
        mask_rev = mask[::-1]
        conv_rev = convolution_fft(a, mask_rev)
        subgrad = 2.0 * conv_rev[L - 1 - np.arange(n)]

        eta = step0 / np.sqrt(it)
        a = a - eta * subgrad
        a = project_to_simplex(a)

    return a


def eg_refine(a: np.ndarray,
              deadline: float,
              max_iter: int = 100_000,
              eta0: float = 0.3) -> np.ndarray:
    """Exponentiated‑gradient polishing (mirror descent on the simplex)."""
    n = len(a)
    L = 2 * n - 1

    for it in range(1, max_iter + 1):
        if time.time() >= deadline:
            break

        conv = convolution_fft(a, a)
        max_val = float(np.max(conv))
        tol = max_val * 1e-12 + 1e-14
        mask = (np.abs(conv - max_val) <= tol).astype(float)

        mask_rev = mask[::-1]
        conv_rev = convolution_fft(a, mask_rev)
        subgrad = 2.0 * conv_rev[L - 1 - np.arange(n)]

        eta = eta0 / (np.sqrt(it) + 1.0)
        a = a * np.exp(-eta * subgrad)

        # normalise (keeps us on the simplex)
        total = a.sum()
        if total == 0.0:
            a = np.full_like(a, 1.0 / n)
        else:
            a = a / total

    return a


# ----------------------------------------------------------------------
# Smoothed gradient descent (log‑sum‑exp surrogate)
# ----------------------------------------------------------------------


def smoothed_gradient_descent(a: np.ndarray,
                              deadline: float,
                              total_iters: int = 30_000,
                              beta_start: float = 5.0,
                              beta_end: float = 200.0,
                              eta0: float = 0.2) -> np.ndarray:
    """
    Gradient descent on the smoothed objective
        fβ(a) = (2·n/β)·log Σk exp(β·convk(a,a)).
    β is increased linearly from ``beta_start`` to ``beta_end``.
    """
    n = len(a)
    L = 2 * n - 1

    for it in range(1, total_iters + 1):
        if time.time() >= deadline:
            break

        # current temperature
        beta = beta_start + (beta_end - beta_start) * (it - 1) / (total_iters - 1)

        # convolution of a with itself
        conv = convolution_fft(a, a)

        # stable soft‑max weights
        maxc = np.max(conv)
        w = np.exp(beta * (conv - maxc))
        w /= w.sum()                     # shape (2n‑1,)

        # weighted sub‑gradient (same pattern as the exact sub‑gradient)
        w_rev = w[::-1]
        conv_rev = convolution_fft(a, w_rev)
        subgrad = 2.0 * conv_rev[L - 1 - np.arange(n)]

        # diminishing step‑size
        eta = eta0 / np.sqrt(it)
        a = a - eta * subgrad
        a = project_to_simplex(a)

        # occasional logging
        if it % 5000 == 0:
            # the true (non‑smoothed) surrogate score
            cur = compute_score(a)
            print(f"    SmoothedGD it={it:,d} score={cur:.6f}")

    return a

# Evolutionary search
# ----------------------------------------------------------------------


def evolutionary_search(n: int,
                        deadline: float,
                        pop_size: int = 30,
                        elite_size: int = 5,
                        max_gens: int = 10_000) -> Tuple[np.ndarray, float]:
    """
    Simple elitist evolutionary algorithm on the fast surrogate score.
    Returns the best vector found and its surrogate score.
    """
    rng = np.random.default_rng()
    # ------------------------------------------------------------------
    # 1. Initialise a diverse population
    # ------------------------------------------------------------------
    population = []
    # a few analytic seeds
    seeds = [
        arcsine_initial(n),
        generate_mixture_shape(n, 0.5),
        generate_gaussian_shape(n, sigma=n / 8.0),
        generate_cosine_shape(n, power=2.0),
        np.full(n, 1.0 / n),
    ]
    # random Dirichlet seeds
    for _ in range(pop_size - len(seeds)):
        seeds.append(random_dirichlet(n, alpha=0.7))

    # normalise (should already be) and store
    for s in seeds:
        population.append(project_to_simplex(s))

    # ------------------------------------------------------------------
    # 2. Evolution loop
    # ------------------------------------------------------------------
    best_vec = None
    best_score = float("inf")
    gen = 0
    while time.time() < deadline - 2.0 and gen < max_gens:
        gen += 1

        # evaluate current population
        scores = [compute_score(v) for v in population]

        # update global best
        min_idx = int(np.argmin(scores))
        if scores[min_idx] < best_score:
            best_score = scores[min_idx]
            best_vec = population[min_idx].copy()
            print(f"Gen {gen:4d}: new best surrogate score = {best_score:.6f}")

        # sort by fitness (lower is better)
        order = np.argsort(scores)
        elites = [population[i] for i in order[:elite_size]]

        # mutation strength decays with generations
        sigma_entry = max(0.02, 0.15 * (1.0 - gen / max_gens))
        sigma_freq = max(0.01, 0.10 * (1.0 - gen / max_gens))

        # produce offspring
        offspring = []
        while len(offspring) < pop_size - elite_size:
            # select parents
            p1 = elites[rng.integers(0, elite_size)]
            if rng.random() < 0.5:
                # crossover
                p2 = elites[rng.integers(0, elite_size)]
                child = 0.5 * (p1 + p2)
            else:
                child = p1.copy()

            # mutate
            if rng.random() < 0.6:
                child = mutate_entry(child, sigma_entry)
            else:
                child = mutate_freq(child, sigma_freq)

            offspring.append(child)

        # new generation = elites + offspring
        population = elites + offspring

    # final best (if loop never executed, fallback to a uniform vector)
    if best_vec is None:
        best_vec = np.full(n, 1.0 / n)
        best_score = compute_score(best_vec)

    return best_vec, best_score

def propose_candidate(seed: int = 42,
                     budget_s: int = 1000,
                     **kwargs) -> List[float]:
    """
    Search for a non‑negative step sequence with a low ``evaluate_sequence``
    value.  The routine combines a diverse analytic seeding, an
    evolutionary search, a smoothed‑gradient phase, and a final
    exact sub‑gradient + EG polishing.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 1.0   # small safety margin

    best_seq: List[float] = []
    best_val: float = float("inf")

    # ------------------------------------------------------------------
    # Optional fallback from a previous run
    # ------------------------------------------------------------------
    if "height_sequence_1" in globals():
        try:
            fallback = [float(x) for x in globals()["height_sequence_1"]]
            val = evaluate_sequence(fallback)
            print(f"Fallback (height_sequence_1) official score = {val:.6f}")
            if val < best_val:
                best_val = val
                best_seq = fallback
        except Exception:
            pass

    # ------------------------------------------------------------------
    # 1. Choose a length – larger sizes tend to give lower constants.
    #    2048 is a good trade‑off between quality and speed.
    # ------------------------------------------------------------------
    n = 2048
    print(f"\n=== Starting search with length n = {n} ===")

    # ------------------------------------------------------------------
    # 2. Evolutionary search for a strong seed
    # ------------------------------------------------------------------
    seed_vec, seed_score = evolutionary_search(n, deadline,
                                                pop_size=30,
                                                elite_size=5,
                                                max_gens=12_000)
    print(f"\nEvolutionary phase best surrogate score = {seed_score:.6f}")

    # ------------------------------------------------------------------
    # 3. Smoothed gradient descent (log‑sum‑exp surrogate)
    # ------------------------------------------------------------------
    if time.time() < deadline - 5.0:
        print("\nSmoothed gradient descent …")
        remaining = deadline - time.time()
        # allocate roughly half of the remaining budget to this phase
        total_iters = int(min(30_000, remaining * 30))
        seed_vec = smoothed_gradient_descent(seed_vec, deadline,
                                             total_iters=total_iters,
                                             beta_start=5.0,
                                             beta_end=200.0,
                                             eta0=0.25)

    # ------------------------------------------------------------------
    # 4. Exact sub‑gradient descent (bulk of remaining time)
    # ------------------------------------------------------------------
    if time.time() < deadline - 5.0:
        remaining = deadline - time.time()
        max_iter_sub = int(min(180_000, remaining * 30))
        print("\nExact sub‑gradient descent …")
        seed_vec = subgradient_descent_exact(seed_vec, deadline,
                                             max_iter=max_iter_sub,
                                             step0=0.18)

    # ------------------------------------------------------------------
    # 5. Exponentiated‑gradient polishing
    # ------------------------------------------------------------------
    if time.time() < deadline - 3.0:
        remaining = deadline - time.time()
        max_iter_eg = int(min(120_000, remaining * 30))
        print("\nExponentiated‑gradient polishing …")
        seed_vec = eg_refine(seed_vec, deadline,
                             max_iter=max_iter_eg,
                             eta0=0.20)

    # ------------------------------------------------------------------
    # 6. Official evaluation of the final candidate
    # ------------------------------------------------------------------
    try:
        final_val = evaluate_sequence(seed_vec.tolist())
        print(f"\nFinal official score = {final_val:.6f}")
        best_seq = seed_vec.tolist()
        best_val = final_val
    except Exception as e:
        print(f"Official evaluation failed: {e}")

    # ------------------------------------------------------------------
    # 7. Guard – ensure non‑empty return
    # ------------------------------------------------------------------
    if not best_seq:
        fallback_n = 256
        best_seq = [1.0 / fallback_n] * fallback_n
        best_val = evaluate_sequence(best_seq)

    print(f"\nSearch finished. Best reported official score = {best_val:.6f}")
    # expose the best sequence for a possible fallback in a subsequent run
    globals()["height_sequence_1"] = best_seq
    return best_seq
