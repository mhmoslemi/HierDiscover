def shape_from_params(params: np.ndarray, n: int) -> np.ndarray:
    param_vals = np.exp(params) maybe to unbounded positive.
    K = len(params)
    half_len = n // 2
    pos_control = np.linspace(0, half_len-1, K)
    idx = np.arange(half_len)
    half = np.interp(idx, pos_control, param_vals)
    full = np.concatenate([half, half[::-1]])
    # ensure non-negative
    full = np.maximum(full, 0.0)
    # Normalize sum to 1
    s = full.sum()
    if s > 0:
        full /= s
    else:
        full = np.full(n, 1.0/n)
    return full

"""
Improved search for a low‑value step function.

Strategy
--------
1.  For each candidate length `n` (1024, 2048, 4096) perform a **global**
    optimisation in a low‑dimensional *symmetric* parameter space using
    `scipy.optimize.differential_evolution`.  The parameters are the
    heights of a few control points; the full vector is obtained by a
    piecewise‑linear interpolation, mirrored to obtain a symmetric shape,
    and finally normalised to sum 1 (the official objective is invariant
    under scaling).

2.  The best candidate of the DE phase is refined with a long **Adam**
    optimisation on a smooth log‑sum‑exp surrogate of the max‑convolution
    (temperature annealed up to a large value so the surrogate is tight).

3.  After Adam we perform an exact **projected sub‑gradient descent**
    (using the true max‑convolution) with a back‑tracking line‑search.

4.  A cheap **stochastic local walk** (Gaussian perturbations) finally
    polishes the solution.

All stages respect the overall 1000 s budget and keep track of the best
vector according to the official `evaluate_sequence`.  If nothing is
found a uniform fallback is returned.
"""

import time
from typing import List, Tuple

import numpy as np

# ----------------------------------------------------------------------
# Linear‑algebra / convolution utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """
    Linear convolution via real FFT.
    Returns an array of length len(a)+len(b)-1.
    """
    m = a.size + b.size - 1
    n_fft = next_pow2(m)
    A = np.fft.rfft(a, n=n_fft)
    B = np.fft.rfft(b, n=n_fft)
    C = A * B
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


def max_convolution(a: np.ndarray) -> float:
    """Maximum entry of the self‑convolution of a."""
    return float(np.max(convolution_fft(a, a)))


def compute_score(a: np.ndarray) -> float:
    """
    Fast surrogate for the official score.
    `a` is assumed to be normalised to Σa = 1,
    consequently `2·n·max(conv(a,a))` equals the official value.
    """
    n = a.shape[0]
    return 2.0 * n * max_convolution(a)


# ----------------------------------------------------------------------
# Simplex projection (Euclidean)
# ----------------------------------------------------------------------


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection onto the probability simplex
    {x ≥ 0, Σx = 1}.  Implementation of Duchi et al. (2008).
    """
    n = v.shape[0]
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / n)

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho = np.where(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Symmetric shape generation from a low‑dimensional parameter vector
# ----------------------------------------------------------------------


def shape_from_params(params: np.ndarray, n: int) -> np.ndarray:
    """
    Build a symmetric, non‑negative, unit‑sum vector of length `n`
    from `params` (length K).  The parameters are interpreted as log‑heights
    of `K` control points on the first half of the vector; linear
    interpolation yields a piecewise‑linear shape which is mirrored.
    """
    # ensure positivity
    heights = np.exp(params)                     # shape (K,)
    K = heights.shape[0]
    half_len = n // 2                           # length of one symmetric half

    # positions of control points on the half‑grid
    ctrl_pos = np.linspace(0, half_len - 1, K)

    # linear interpolation on the half‑grid
    idx = np.arange(half_len, dtype=float)
    half = np.interp(idx, ctrl_pos, heights)

    # enforce non‑negativity (should already hold)
    half = np.maximum(half, 0.0)

    # mirror to obtain full vector
    full = np.concatenate([half, half[::-1]])

    # normalise to unit L1‑norm
    total = full.sum()
    if total > 0.0:
        full /= total
    else:
        full = np.full(n, 1.0 / n)
    return full


# ----------------------------------------------------------------------
# Global optimisation via differential evolution (low‑dimensional)
# ----------------------------------------------------------------------


def de_optimize_shape(
    n: int,
    time_deadline: float,
    K: int,
    seed: int,
) -> np.ndarray:
    """
    Run a quick differential‑evolution search for a symmetric shape of
    length `n`.  The population lives in `R^K` (log‑heights of control
    points) with bounds [-5, 5].  The routine returns the best full‑length
    vector found (already normalised to sum 1).  The internal budget is a
    small, fixed number of generations – sufficient to discover a good
    region for the subsequent gradient phases.
    """
    from scipy.optimize import differential_evolution

    # objective for DE: the fast surrogate (no need for the expensive clipping)
    def obj(p):
        a = shape_from_params(p, n)
        return compute_score(a)

    bounds = [(-5.0, 5.0)] * K

    # a modest number of generations; we rely on later gradient phases
    # for fine tuning.
    result = differential_evolution(
        obj,
        bounds,
        strategy="best1bin",
        maxiter=30,
        popsize=15,
        tol=1e-7,
        mutation=(0.5, 1.0),
        recombination=0.7,
        seed=seed,
        polish=False,
        disp=False,
        init="latinhypercube",
    )
    best_params = result.x
    best_vec = shape_from_params(best_params, n)
    return best_vec


# ----------------------------------------------------------------------
# Smooth surrogate (log‑sum‑exp) and its gradient (used by Adam)
# ----------------------------------------------------------------------


def smooth_gradient(a: np.ndarray, lam: float) -> Tuple[np.ndarray, float]:
    """
    Gradient of the log‑sum‑exp surrogate of `max(conv(a,a))`.
    Returns (gradient, surrogate_value).
    """
    n = a.shape[0]
    conv = convolution_fft(a, a)          # length 2n‑1
    max_c = np.max(conv)

    # stable log‑sum‑exp on the convolution entries
    exp_term = np.exp(lam * (conv - max_c))
    sum_exp = np.sum(exp_term)
    w = exp_term / sum_exp                # soft‑max on convolution entries
    B = max_c + np.log(sum_exp) / lam     # smooth approximation of the max

    # weighted sum of sub‑gradients:
    # ∂/∂a Σ_k w_k (a*a)_k = 2 * (a ⋆ w_rev)
    w_rev = w[::-1]
    conv_rev = convolution_fft(a, w_rev)      # length 3n‑2
    L = 2 * n - 1
    weighted_grad = 2.0 * conv_rev[L - 1 - np.arange(n)]

    s = np.sum(a)
    term1 = (2.0 * n / (s ** 2)) * weighted_grad
    term2 = -(4.0 * n * B / (s ** 3)) * np.ones_like(a)
    grad = term1 + term2
    return grad, B


def adam_optimize(
    initial_a: np.ndarray,
    time_deadline: float,
    max_iter: int = 300_000,
    lr0: float = 0.12,
    lam_start: float = 1.0,
    lam_end: float = 300.0,
    noise_every: int = 1500,
    noise_scale: float = 3e-5,
) -> Tuple[np.ndarray, float]:
    """
    Adam optimiser on the smooth log‑sum‑exp surrogate.
    Returns the best vector (projected onto the simplex) and its fast score.
    """
    a = project_to_simplex(initial_a.astype(float).copy())
    best_a = a.copy()
    best_sc = compute_score(a)

    beta1, beta2 = 0.9, 0.999
    eps_adam = 1e-8
    m = np.zeros_like(a)
    v = np.zeros_like(a)

    n = a.shape[0]

    for it in range(1, max_iter + 1):
        if time.time() >= time_deadline:
            break

        # annealed temperature (square‑root schedule gives a gentle rise)
        t = it / max_iter
        lam = lam_start + (lam_end - lam_start) * (t ** 0.5)

        grad, _ = smooth_gradient(a, lam)

        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)

        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        step = lr0 * m_hat / (np.sqrt(v_hat) + eps_adam)

        a_new = a - step

        if noise_every and (it % noise_every == 0):
            a_new += np.random.randn(n) * noise_scale

        a_new = project_to_simplex(a_new)

        cur_sc = compute_score(a_new)
        if cur_sc < best_sc:
            best_sc = cur_sc
            best_a = a_new.copy()

        a = a_new

        if it % 5000 == 0:
            print(
                f"    Adam iter={it:,d} cur={cur_sc:.6f} best={best_sc:.6f}"
            )

    return best_a, best_sc


# ----------------------------------------------------------------------
# Exact sub‑gradient descent with back‑tracking line‑search
# ----------------------------------------------------------------------


def subgradient_optimize(
    initial_a: np.ndarray,
    time_deadline: float,
    step0: float = 0.2,
    max_iter: int = 150_000,
) -> Tuple[np.ndarray, float]:
    """
    Projected sub‑gradient descent on the *true* max‑convolution.
    Uses a simple back‑tracking line‑search to guarantee descent.
    Returns the best vector (projected) and its fast score.
    """
    a = initial_a.copy()
    best_a = a.copy()
    best_sc = compute_score(a)

    n = a.shape[0]

    for it in range(1, max_iter + 1):
        if time.time() >= time_deadline:
            break

        conv = convolution_fft(a, a)
        max_val = float(np.max(conv))
        idx = int(np.argmax(conv))

        # gradient of the active quadratic term
        i = np.arange(n)
        j = idx - i
        mask = (j >= 0) & (j < n)
        grad = np.zeros_like(a)
        grad[mask] = 2.0 * a[j[mask]]

        # decreasing step size + back‑tracking
        step = step0 / np.sqrt(it)
        for _ in range(12):
            a_new = a - step * grad
            a_new = project_to_simplex(a_new)
            new_val = max_convolution(a_new)
            if new_val < max_val - 1e-14:
                break
            step *= 0.5

        cur_sc = compute_score(a_new)
        if cur_sc < best_sc:
            best_sc = cur_sc
            best_a = a_new.copy()

        a = a_new

        if it % 40000 == 0:
            print(
                f"    Sub‑grad iter={it:,d} cur={cur_sc:.6f} best={best_sc:.6f}"
            )

    return best_a, best_sc


# ----------------------------------------------------------------------
# Stochastic local walk (Gaussian perturbations)
# ----------------------------------------------------------------------


def random_local_search(
    initial_a: np.ndarray,
    time_deadline: float,
    init_step: float = 0.012,
    max_iter: int = 250_000,
) -> Tuple[np.ndarray, float]:
    """
    Gaussian walk on the simplex with a tiny uphill‑move probability.
    Returns the best vector seen and its fast score.
    """
    a = initial_a.copy()
    best_a = a.copy()
    best_sc = compute_score(a)

    n = a.shape[0]
    rng = np.random.default_rng()

    for it in range(1, max_iter + 1):
        if time.time() >= time_deadline:
            break

        step = init_step / np.sqrt(it)
        noise = rng.normal(scale=step, size=n)
        a_proposed = a + noise
        a_proposed = project_to_simplex(a_proposed)

        cur_sc = compute_score(a_proposed)

        # accept improvements; allow rare uphill moves
        if cur_sc < best_sc or (rng.random() < 1e-4 and cur_sc < 10 * best_sc):
            a = a_proposed
            if cur_sc < best_sc:
                best_sc = cur_sc
                best_a = a_proposed.copy()

        if it % 50000 == 0:
            print(
                f"    Random local iter={it:,d} cur={cur_sc:.6f} best={best_sc:.6f}"
            )

    return best_a, best_sc


# ----------------------------------------------------------------------
# Main entry point required by the competition harness
# ----------------------------------------------------------------------


def propose_candidate(
    seed: int = 42,
    budget_s: int = 1000,
    **kwargs,
) -> List[float]:
    """
    Search for a non‑negative step sequence with a low official score.
    The routine consists of:
        * (optional) fallback to a known construction,
        * global DE in a low‑dimensional symmetric space for a few lengths,
        * Adam → sub‑gradient → stochastic polishing,
        * final official evaluation.
    Returns the best sequence found (as a plain list of floats).
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 2.0   # safety margin

    best_seq: List[float] = []
    best_val: float = float("inf")

    # ------------------------------------------------------------------
    # 0.  Fallback to a previously known sequence if it exists
    # ------------------------------------------------------------------
    try:
        fallback = [float(x) for x in height_sequence_1]  # type: ignore
        val = evaluate_sequence(fallback)
        print(f"Fallback (height_sequence_1) official score = {val:.6f}")
        if val < best_val:
            best_val = val
            best_seq = fallback
    except Exception:
        pass

    # ------------------------------------------------------------------
    # 1.  Optimise a few candidate lengths
    # ------------------------------------------------------------------
    lengths = [4096, 2048, 1024]   # try larger lengths first

    for n in lengths:
        if time.time() >= deadline - 15.0:
            break
        print(f"\n=== Optimising length n = {n} ===")

        # control‑point count grows with n but stays modest
        K = max(8, min(40, n // 200))

        # ---------- Differential Evolution (global) ----------
        de_deadline = deadline - 5.0
        print(f"  DE phase (K={K}) …")
        de_vec = de_optimize_shape(
            n=n,
            time_deadline=de_deadline,
            K=K,
            seed=int(rng.integers(1_000_000)),
        )
        val = evaluate_sequence(de_vec.tolist())
        print(f"  DE candidate official score = {val:.6f}")
        if val < best_val:
            best_val = val
            best_seq = de_vec.tolist()

        # ---------- Adam refinement ----------
        if time.time() >= deadline - 10.0:
            break
        print("  Adam refinement …")
        adam_deadline = deadline - 5.0
        polished_a, _ = adam_optimize(
            initial_a=de_vec,
            time_deadline=adam_deadline,
            max_iter=400_000,
            lr0=0.10,
            lam_start=1.0,
            lam_end=350.0,
            noise_every=1400,
            noise_scale=2.5e-5,
        )

        # ---------- Sub‑gradient descent ----------
        if time.time() < deadline - 5.0:
            print("  Sub‑gradient polishing …")
            sub_deadline = deadline - 5.0
            polished_a, _ = subgradient_optimize(
                polished_a,
                time_deadline=sub_deadline,
                step0=0.18,
                max_iter=120_000,
            )

        # ---------- Stochastic local walk ----------
        if time.time() < deadline - 3.0:
            print("  Random local walk …")
            loc_deadline = deadline - 3.0
            polished_a, _ = random_local_search(
                polished_a,
                time_deadline=loc_deadline,
                init_step=0.010,
                max_iter=180_000,
            )

        # ---------- Official evaluation ----------
        val = evaluate_sequence(polished_a.tolist())
        print(f"  Polished official score for length {n} = {val:.6f}")
        if val < best_val:
            best_val = val
            best_seq = polished_a.tolist()
            print(f"  *** New global best (len={n}) = {best_val:.6f} ***")

    # ------------------------------------------------------------------
    # 2.  Global polishing of the overall best vector (if time permits)
    # ------------------------------------------------------------------
    if best_seq and time.time() < deadline - 4.0:
        print("\n=== Global polishing of the best sequence ===")
        a = np.array(best_seq, dtype=float)

        # Adam (high‑temperature, long run)
        remaining = deadline - time.time()
        adam_deadline = time.time() + remaining * 0.5
        print(f"  Global Adam polishing (deadline in {remaining*0.5:.1f}s) …")
        a, _ = adam_optimize(
            initial_a=a,
            time_deadline=adam_deadline,
            max_iter=500_000,
            lr0=0.08,
            lam_start=1.0,
            lam_end=400.0,
            noise_every=1500,
            noise_scale=2e-5,
        )

        # Sub‑gradient
        if time.time() < deadline - 2.0:
            sub_deadline = deadline - 2.0
            print("  Global sub‑gradient polishing …")
            a, _ = subgradient_optimize(
                a,
                time_deadline=sub_deadline,
                step0=0.15,
                max_iter=180_000,
            )

        # Random walk
        if time.time() < deadline - 1.0:
            loc_deadline = deadline - 1.0
            print("  Global random walk polishing …")
            a, _ = random_local_search(
                a,
                time_deadline=loc_deadline,
                init_step=0.008,
                max_iter=220_000,
            )

        final_val = evaluate_sequence(a.tolist())
        print(f"Final polished official score = {final_val:.6f}")
        if final_val < best_val:
            best_val = final_val
            best_seq = a.tolist()
            print("*** Updated global best after final polish ***")

    # ------------------------------------------------------------------
    # 3.  Guard – guarantee a non‑empty return
    # ------------------------------------------------------------------
    if not best_seq:
        best_seq = [1.0] * 256
        best_val = evaluate_sequence(best_seq)
        print("Fallback to uniform 256‑vector.", best_val)

    print(f"\nSearch finished. Best reported official score = {best_val:.6f}")
    return best_seq
