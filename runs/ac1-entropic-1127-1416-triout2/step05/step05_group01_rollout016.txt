def hill_climb(seq, time_deadline):
    a = np.array(seq, dtype=float)
    n = len(a)
    best_a = a.copy()
    best_score = compute_score(best_a)
    # set temperature schedule: initial step_scale = 0.2 maybe.
    step_scale = 0.2
    while time.time() < time_deadline:
        # propose move
        i, j = np.random.randint(n, size=2)
        if i == j:
            continue
        # transfer fraction from i to j (positive mass move)
        delta = step_scale * best_a[i]
        # ensure small transfer
        cand = best_a.copy()
        cand[i] -= delta
        cand[j] += delta
        # ensure nonnegative
        if cand[i] < 0:
            # adjust delta to be at most current value
            delta = best_a[i]
            cand[i] = 0
            cand[j] = best_a[j] + delta
        # project onto simplex
        cand = project_to_simplex(cand)
        cur_score = compute_score(cand)
        if cur_score < best_score:
            best_score = cur_score
            best_a = cand
            step_scale = min(step_scale * 1.1, 0.5)
        else:
            step_scale *= 0.98
    return best_a.tolist(), best_score

"""
Hybrid optimizer for the step‑function attack.

The goal is to minimise
    score(a) = 2·n·max(conv(a,a)) / (sum(a))**2
which is invariant under scaling.  We therefore work with
vectors on the probability simplex (sum == 1, a_i ≥ 0) and minimise
    2·n·max(conv(a,a))

The algorithm consists of three stages executed within the
1000 s time budget:

1️⃣ **Random restarts** – a Dirichlet draw (concentration α drawn
   uniformly in [0.1, 5]) provides a diverse initial point.
2️⃣ **Smoothed exponentiated gradient descent** – instead of
   following only the single worst convolution shift, a softmax
   over all shifts is used:
        w_k ∝ exp( λ·conv_k )
   The gradient of the smoothed objective is
        g = 2·(a_rev ∗ w)          (central part only)
   Background updates are multiplicative (a ← a·exp(−η·g))
   with a decreasing step‑size η_t = η₀/√t and a λ‑schedule that
   slowly sharpens (λ rises from ~1 to ~30).  After each step the
   vector is projected back onto the probability simplex.
3️⃣ **Fine‑grained hill‑climbing** – a stochastic mass‑transfer
   move (move a fraction of the mass from one index to another)
   is accepted whenever it lowers the true objective.  The size of
   the transferred fraction adapts to the success rate.

The routine keeps the best sequence seen across all restarts,
prints intermediate scores and finally returns the best list
of floats.  All helper functions are defined at top level and
no external I/O is performed.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# FFT‑based convolution utilities (linear convolution via real FFT)
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """
    Linear convolution a*b using a real FFT (fast).
    The result length is len(a)+len(b)-1.
    """
    m = a.shape[0] + b.shape[0] - 1
    n_fft = _next_pow2(m)
    A = np.fft.rfft(a, n=n_fft)
    B = np.fft.rfft(b, n=n_fft)
    C = A * B
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


# ----------------------------------------------------------------------
# Objective and helpers
# ----------------------------------------------------------------------


def compute_score(a: np.ndarray) -> float:
    """
    Fast replica of `evaluate_sequence` when Σa = 1.
    Returns 2·n·max(conv(a,a)).
    """
    n = a.shape[0]
    conv = convolution_fft(a, a)
    return 2.0 * n * float(np.max(conv))


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the probability simplex
    {x | x_i ≥ 0, Σx_i = 1}.
    """
    if np.all(v <= 0):
        # uniform point if everything is non‑positive
        return np.full_like(v, 1.0 / v.shape[0])

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, v.shape[0] + 1)
    cond = u - cssv / ind > 0
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


# ----------------------------------------------------------------------
# Smoothed exponentiated gradient descent
# ----------------------------------------------------------------------


def smooth_exp_optimize(
    init_a: np.ndarray,
    deadline: float,
    eta0: float = 0.5,
    lam_start: float = 1.0,
    lam_end: float = 30.0,
    eval_interval: int = 400,
) -> tuple[np.ndarray, float]:
    """
    Smoothed exponentiated gradient descent on
        f(a) = max_k (a*a)[k]   (with a on the simplex).

    A softmax over the convolution entries yields a smooth surrogate.
    The gradient of the surrogate is
        g = 2 * (a_rev  *  soft_weights)
    where the convolution is taken over the full length (3·n‑2) and the
    central n entries are kept.

    Returns the best vector encountered and its score.
    """
    a = init_a.astype(float).copy()
    n = a.shape[0]

    best_a = a.copy()
    best_score = compute_score(a)

    it = 0
    while True:
        if time.time() >= deadline:
            break
        it += 1

        # linear convolution a*a
        conv = convolution_fft(a, a)                # length 2n‑1
        M = np.max(conv)                           # current max value

        # softmax temperature (λ) slowly increases
        lam = lam_start + (lam_end - lam_start) * min(1.0, it / 2_000_000)

        # compute softmax weights (stable version)
        exp_vals = np.exp(lam * (conv - M))
        w = exp_vals / np.sum(exp_vals)            # length 2n‑1, sums to 1

        # gradient = 2 * (a_rev * w)[n-1 : n-1+n]
        a_rev = a[::-1]
        g_full = convolution_fft(a_rev, w)         # length 3n‑2
        g = 2.0 * g_full[n - 1 : n - 1 + n]

        # step size
        eta = eta0 / np.sqrt(it)

        # multiplicative update with clipping for numeric safety
        step = -eta * g
        step = np.clip(step, -50, 50)              # avoid overflow in exp
        a = a * np.exp(step)

        # back onto the simplex
        a = project_to_simplex(a)

        # periodic evaluation of the true (non‑smoothed) score
        if it % eval_interval == 0:
            cur_score = compute_score(a)
            if cur_score < best_score:
                best_score = cur_score
                best_a = a.copy()
                print(f"[{time.time():.1f}]  iter {it:,}  new best {best_score:.6f}")

    # final check
    final_score = compute_score(a)
    if final_score < best_score:
        best_score = final_score
        best_a = a.copy()

    return best_a, best_score


# ----------------------------------------------------------------------
# Simple hill‑climber (mass‑transfer moves)
# ----------------------------------------------------------------------


def hill_climb(seq, deadline, max_steps: int = 1_000_000):
    """
    Stochastic hill‑climber that moves a fraction of mass from one index
    to another.  Moves are always projected onto the simplex.
    """
    a = np.array(seq, dtype=float)
    n = a.shape[0]

    best_a = a.copy()
    best_score = compute_score(best_a)

    # start with a modest transfer fraction, adapt during the run
    step_scale = 0.15

    steps = 0
    while steps < max_steps and time.time() < deadline:
        i, j = np.random.randint(n, size=2)
        if i == j:
            continue

        cand = best_a.copy()
        delta = step_scale * best_a[i]               # amount to move
        if delta <= 0:
            continue
        cand[i] -= delta
        cand[j] += delta

        # keep non‑negativity before projection
        cand = np.maximum(cand, 0.0)
        cand = project_to_simplex(cand)

        cur_score = compute_score(cand)

        if cur_score < best_score:
            best_score = cur_score
            best_a = cand
            # success → magnify step a bit
            step_scale = min(step_scale * 1.07, 0.5)
        else:
            # failure → shrink step
            step_scale *= 0.98

        steps += 1

    return best_a.tolist(), best_score

# Main entry point required by the harness
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 2024, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative sequence that minimises `evaluate_sequence`.
    The procedure performs several random restarts, a smoothed
    exponentiated gradient descent on each, and finishes with a short
    hill‑climbing polishing phase.
    """
    np.random.seed(seed)
    start_time = time.time()
    # small safety margin for the harness
    deadline = start_time + budget_s - 1.0

    # ------------------------------------------------------------------
    # Helper to obtain a reasonable vector length (use the length of the
    # provided fallback if present, otherwise 1024)
    # ------------------------------------------------------------------
    try:
        n = len(height_sequence_1)          # type: ignore[name-defined]
    except Exception:
        n = 1024

    best_seq = None
    best_val = float("inf")

    # ------------------------------------------------------------------
    # Use existing fallback if available
    # ------------------------------------------------------------------
    try:
        fallback = [float(x) for x in height_sequence_1]  # type: ignore[name-defined]
        fallback_val = evaluate_sequence(fallback)
        print(f"Fallback score = {fallback_val:.6f}")
        if fallback_val < best_val:
            best_val = fallback_val
            best_seq = fallback
    except Exception:
        pass

    # ------------------------------------------------------------------
    # Main optimisation loop – keep trying new starts until time runs out
    # ------------------------------------------------------------------
    restart_counter = 0
    while time.time() < deadline - 15.0:          # keep ~15 s for final polishing
        restart_counter += 1
        # draw a Dirichlet vector with a random concentration (to vary sparsity)
        alpha = np.random.uniform(0.1, 5.0)
        a0 = np.random.dirichlet(np.full(n, alpha))

        # give each start roughly the same slice of remaining time
        remaining = deadline - time.time()
        # keep at least 8 s for the hill‑climber; the rest for gradient descent
        sub_deadline = time.time() + max(8.0, remaining * 0.6)

        print(f"[{time.time():.1f}] Restart {restart_counter}, α={alpha:.2f}")

        refined_a, refined_score = smooth_exp_optimize(
            a0,
            deadline=sub_deadline,
            eta0=0.55,
            lam_start=0.8,
            lam_end=28.0,
            eval_interval=500,
        )
        if refined_score < best_val:
            best_val = refined_score
            best_seq = refined_a.tolist()
            print(f"  → new global best after opt: {best_val:.6f}")

    # ------------------------------------------------------------------
    # Final hill‑climb polishing (uses the exact objective)
    # ------------------------------------------------------------------
    if best_seq is not None and time.time() < deadline - 3.0:
        print("\nFinal hill‑climb polishing")
        polished_seq, polished_val = hill_climb(
            best_seq,
            deadline=deadline,
            max_steps=500_000,
        )
        if polished_val < best_val:
            best_val = polished_val
            best_seq = polished_seq
            print(f"Hill‑climb improved best to {best_val:.6f}")

    # ------------------------------------------------------------------
    # Fallback in the unlikely event that nothing was produced
    # ------------------------------------------------------------------
    if best_seq is None:
        best_seq = [1.0 / n] * n
        best_val = evaluate_sequence(best_seq)

    final_score = evaluate_sequence(best_seq)
    print(f"\nSearch finished. Final reported score = {final_score:.6f}")

    return best_seq
