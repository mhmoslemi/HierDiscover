def solve_lp_max_sum(f, M):
    n = len(f)
    L = 2*n - 1
    # Build matrix A
    A = np.zeros((L, n), dtype=np.float64)
    # Use for loop
    for i in range(n):
        A[i:i+n, i] = f
    import cvxpy as cp
    g = cp.Variable(n, nonneg=True)
    constraints = [A @ g <= M + 1e-12]  # small epsilon tolerance
    objective = cp.Maximize(cp.sum(g))
    prob = cp.Problem(objective, constraints)
    prob.solve(solver=cp.ECOS, max_iters=5000, abstol=1e-9, reltol=1e-9, feastol=1e-9)
    # On failure, fallback
    if g.value is None:
        return None
    g_opt = np.array(g.value).flatten()
    return g_opt

"""
Propose a low‑discrepancy step function.

The search combines four ideas
  1️⃣  a diverse random sampling of parametric families (arcsine, beta,
      power‑shape, mixtures, triangular bumps);
  2️⃣  a long Euclidean‑projected sub‑gradient descent on the fast
      surrogate  f(a)=2·n·max(conv(a,a)) (the denominator Σa² is fixed by
      normalising the vector to Σa=1);
  3️⃣  exponentiated‑gradient polishing (mirror descent on the simplex);
  4️⃣  a *linear‑program polishing* phase: for the current best vector
      f we solve
            maximise   Σg
            subject to  f∗g ≤ M   (M = max(conv(f,f))) , g ≥ 0 .
      The optimum g has a larger ℓ₁‑norm while keeping the same convolution
      bound.  After normalising g to Σg=1 we perform a small line search
      over convex combinations (1‑t)·f + t·g and keep the best.  Repeating
      this LP‑step a handful of times typically pushes the bound a few
      thousandths lower.

All phases respect a global time budget (default 1000 s).  The best
vector found is returned as a plain list of floats; it is guaranteed
to be non‑negative and to have a non‑zero sum, so ``evaluate_sequence``
never returns ``inf``.
"""

import time
import numpy as np
import cvxpy as cp
from typing import List, Tuple

def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Linear convolution via real FFT (length = len(a)+len(b)-1)."""
    m = len(a) + len(b) - 1
    n_fft = next_pow2(m)
    A = np.fft.rfft(a, n=n_fft)
    B = np.fft.rfft(b, n=n_fft)
    C = A * B
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


def compute_score(a: np.ndarray) -> float:
    """
    Fast surrogate for the official score.
    The vector is assumed to be normalised to Σa = 1,
    therefore the denominator disappears and the score is simply
    2·n·max(conv(a,a)).
    """
    n = a.shape[0]
    conv = convolution_fft(a, a)
    return 2.0 * n * float(np.max(conv))


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection onto the probability simplex {x ≥ 0, Σx = 1}.
    Duchi et al., 2008.
    """
    if np.all(v <= 0):
        # uniform fallback – should never happen in practice
        return np.full_like(v, 1.0 / v.size)

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho_arr = u - cssv / np.arange(1, v.size + 1)
    rho = np.where(rho_arr > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w

def arcsine_initial(n: int) -> np.ndarray:
    """Classic arcsine shape: a_i ∝ 1/√((i+½)*(n‑½‑i))."""
    i = np.arange(n, dtype=float)
    eps = 0.25
    vals = 1.0 / np.sqrt((i + 0.5) * (n - 0.5 - i) + eps)
    vals /= vals.sum()
    return vals


def generate_beta_shape(n: int, alpha: float, beta: float, eps: float) -> np.ndarray:
    """
    (i+ε)^{α‑1}·(n‑1‑i+ε)^{β‑1} on {0,…,n‑1}, normalised to Σ=1.
    """
    i = np.arange(n, dtype=float)
    left = (i + eps) ** (alpha - 1.0)
    right = (n - 1 - i + eps) ** (beta - 1.0)
    base = left * right
    total = base.sum()
    if total <= 0.0 or not np.isfinite(total):
        return np.full(n, 1.0 / n)
    return base / total


def generate_power_shape(n: int, p: float, eps: float) -> np.ndarray:
    """
    Generalised arcsine: a_i ∝ ((i+½)*(n‑½‑i)+ε)^{‑p}.
    p>0, eps≥0.
    """
    i = np.arange(n, dtype=float)
    denom = ((i + 0.5) * (n - 0.5 - i) + eps) ** p
    vals = 1.0 / denom
    total = vals.sum()
    if total <= 0.0 or not np.isfinite(total):
        return np.full(n, 1.0 / n)
    return vals / total


def generate_triangular_shape(n: int, peak: int) -> np.ndarray:
    """Symmetric triangular bump with its apex at ``peak``."""
    left = np.arange(peak + 1, dtype=float)
    right = np.arange(n - peak - 1, dtype=float)[::-1]
    seq = np.concatenate([left, right])
    seq /= seq.sum()
    return seq


def generate_mixture_shape(n: int, weight_arc: float) -> np.ndarray:
    """Mixture (weight_arc)·arcsine + (1‑weight_arc)·uniform."""
    arc = arcsine_initial(n)
    uni = np.full(n, 1.0 / n)
    mix = weight_arc * arc + (1.0 - weight_arc) * uni
    mix /= mix.sum()
    return mix

def subgradient_descent_exact(initial_a: np.ndarray,
                              deadline: float,
                              max_iter: int = 80_000,
                              step0: float = 0.05) -> np.ndarray:
    """
    Sub‑gradient descent on the exact objective f(a)=2·n·max(conv(a,a)).
    The step size is η_t = step0 / √t.  Projection onto the simplex
    enforces Σa = 1.
    """
    a = project_to_simplex(initial_a.copy())
    n = len(a)
    L = 2 * n - 1

    for it in range(1, max_iter + 1):
        if time.time() >= deadline:
            break

        conv = convolution_fft(a, a)
        max_val = float(np.max(conv))

        # active set (indices attaining the max, with tolerance)
        tol = max_val * 1e-12 + 1e-14
        mask = (np.abs(conv - max_val) <= tol).astype(float)

        # sub‑gradient = 2·∑_{k∈K} shift(a,k)
        mask_rev = mask[::-1]
        conv_rev = convolution_fft(a, mask_rev)
        subgrad = 2.0 * conv_rev[L - 1 - np.arange(n)]

        eta = step0 / np.sqrt(it)
        a = a - eta * subgrad
        a = project_to_simplex(a)

        if it % 5000 == 0:
            cur_sc = compute_score(a)
            print(f"    SubGrad it={it:,d} score={cur_sc:.6f}")

    return a


def eg_refine(a: np.ndarray,
              deadline: float,
              max_iter: int = 100_000,
              eta0: float = 0.3) -> np.ndarray:
    """
    Mirror‑descent style polishing using an exponentiated‑gradient update.
    Keeps the iterate on the simplex automatically and often reduces
    the bound a bit further.
    """
    n = a.shape[0]
    L = 2 * n - 1

    for it in range(1, max_iter + 1):
        if time.time() >= deadline:
            break

        conv = convolution_fft(a, a)
        max_val = float(np.max(conv))
        tol = max_val * 1e-12 + 1e-14
        mask = (np.abs(conv - max_val) <= tol).astype(float)

        mask_rev = mask[::-1]
        conv_rev = convolution_fft(a, mask_rev)
        subgrad = 2.0 * conv_rev[L - 1 - np.arange(n)]

        eta = eta0 / (np.sqrt(it) + 1.0)
        a = a * np.exp(-eta * subgrad)

        total = a.sum()
        if total == 0.0:
            a = np.full_like(a, 1.0 / n)
        else:
            a = a / total

        if it % 5000 == 0:
            cur_sc = compute_score(a)
            print(f"    EG it={it:,d} score={cur_sc:.6f}")

    return a


def hillclimb_power_shape(n: int,
                          init_p: float,
                          init_eps: float,
                          deadline: float,
                          max_iters: int = 300) -> Tuple[np.ndarray, float]:
    """
    Random‑walk hill‑climbing on the (p,eps) parameters of the
    generalized arcsine family.  Uses the fast surrogate.
    """
    p, eps = init_p, init_eps
    best_seq = generate_power_shape(n, p, eps)
    best_sc = compute_score(best_seq)

    for _ in range(max_iters):
        if time.time() >= deadline:
            break
        # multiplicative perturbation for p (keeps it positive)
        p_new = max(0.01, p * (1.0 + np.random.uniform(-0.15, 0.15)))
        eps_new = max(0.0, eps + np.random.uniform(-0.1, 0.1))
        cand = generate_power_shape(n, p_new, eps_new)
        sc = compute_score(cand)
        if sc < best_sc:
            p, eps, best_seq, best_sc = p_new, eps_new, cand, sc
    return best_seq, best_sc


def hillclimb_beta_shape(n: int,
                         init_alpha: float,
                         init_beta: float,
                         init_eps: float,
                         deadline: float,
                         max_iters: int = 300) -> Tuple[np.ndarray, float]:
    """
    Random‑walk hill‑climbing on the three parameters of the beta family.
    """
    alpha, beta, eps = init_alpha, init_beta, init_eps
    best_seq = generate_beta_shape(n, alpha, beta, eps)
    best_sc = compute_score(best_seq)

    for _ in range(max_iters):
        if time.time() >= deadline:
            break
        alpha_new = max(0.01, alpha * (1.0 + np.random.uniform(-0.15, 0.15)))
        beta_new  = max(0.01, beta  * (1.0 + np.random.uniform(-0.15, 0.15)))
        eps_new   = max(0.0, eps + np.random.uniform(-0.1, 0.1))
        cand = generate_beta_shape(n, alpha_new, beta_new, eps_new)
        sc = compute_score(cand)
        if sc < best_sc:
            alpha, beta, eps, best_seq, best_sc = alpha_new, beta_new, eps_new, cand, sc
    return best_seq, best_sc

def solve_lp_max_sum(f: np.ndarray, M: float) -> np.ndarray | None:
    """
    Given a current distribution f (Σf = 1) and the bound M = max(conv(f,f)),
    solve   maximise Σg   s.t.   f∗g ≤ M   ,   g ≥ 0 .
    Returns the optimal g (not normalised).  If the solver fails, returns None.
    """
    n = f.shape[0]
    L = 2 * n - 1

    # Build the (L × n) convolution matrix A with rows representing
    # (f * g)[k] = Σ_i f_{k-i}·g_i .
    A = np.zeros((L, n), dtype=np.float64)
    for i in range(n):
        A[i:i + n, i] = f

    g = cp.Variable(n, nonneg=True)
    # Slight tolerance to avoid numerical infeasibility
    constraints = [A @ g <= M + 1e-12]
    prob = cp.Problem(cp.Maximize(cp.sum(g)), constraints)

    try:
        prob.solve(solver=cp.ECOS, warm_start=True, max_iters=5000,
                   abstol=1e-9, reltol=1e-9, feastol=1e-9)
    except Exception as e:
        print(f"   LP solve exception: {e}")
        return None

    if g.value is None:
        print("   LP returned no solution.")
        return None

    return np.asarray(g.value).flatten()


def lp_refinement(start_vec: np.ndarray,
                  deadline: float,
                  max_iter: int = 12) -> np.ndarray:
    """
    Perform a handful of LP‑based polishing steps on ``start_vec``.
    Each step solves the LP described in ``solve_lp_max_sum`` and then
    tries a small line‑search over convex combinations (1‑t)·f + t·g.
    The process stops when the time budget is exhausted or when no
    improvement is observed.
    """
    a = start_vec.copy()
    a = a / a.sum()                     # ensure Σa = 1
    best_score = compute_score(a)

    for it in range(1, max_iter + 1):
        if time.time() >= deadline - 2.0:
            break

        conv_a = convolution_fft(a, a)
        M = float(np.max(conv_a))

        g_opt = solve_lp_max_sum(a, M)
        if g_opt is None:
            break

        sum_g = g_opt.sum()
        if sum_g <= 0.0:
            break
        g_norm = g_opt / sum_g            # Σg_norm = 1

        # line‑search over a few mixing coefficients
        improved = False
        for t in (0.25, 0.5, 0.75, 1.0):
            cand = (1.0 - t) * a + t * g_norm
            cand = cand / cand.sum()
            cand_score = compute_score(cand)
            if cand_score < best_score - 1e-12:
                best_score = cand_score
                a = cand
                improved = True
                # optional debug print
                print(f"    LP iter {it}, t={t:.2f} → fast score {cand_score:.6f}")
                break   # accept first improvement, continue outer loop

        if not improved:
            # no improvement → stop early
            break

    return a

def propose_candidate(seed: int = 42,

    if "height_sequence_1" in globals():

            val = evaluate_sequence(fallback)

            if val < best_val:

        except Exception:

            eps = rng.uniform(0.0, 2.0)
            cand = generate_power_shape(n, p, eps)
            sc = compute_score(cand)
            global_candidates.append((sc, cand))

        # mixtures of arcsine & uniform

            if time.time() >= deadline - 5.0:
                break
            w = rng.random()
            cand = generate_mixture_shape(n, w)
            sc = compute_score(cand)
            global_candidates.append((sc, cand))

        # simple triangular bumps
        for _ in range(80):
            if time.time() >= deadline - 5.0:
                break
            peak = rng.integers(0, n)
            cand = generate_triangular_shape(n, peak)
            sc = compute_score(cand)
            global_candidates.append((sc, cand))

        # 1b️⃣  Local hill‑climb on the two main families
        cand, sc = hillclimb_power_shape(n, 0.5, 0.0, deadline, max_iters=250)
        global_candidates.append((sc, cand))

        cand, sc = hillclimb_beta_shape(n, 0.5, 0.5, 0.0, deadline, max_iters=250)
        global_candidates.append((sc, cand))

    # Keep the very best few seeds for intensive refinement
    global_candidates.sort(key=lambda x: x[0])
    top_seeds = [vec for _, vec in global_candidates[:8]]
    print(f"\nSelected {len(top_seeds)} top seeds for refinement (fast score range "
          f"{global_candidates[0][0]:.6f} – {global_candidates[-1][0]:.6f})")

    # ------------------------------------------------------------------
    # 2️⃣  Intensive refinement of each seed (sub‑gradient + EG)
    # ------------------------------------------------------------------
    for idx, seed_vec in enumerate(top_seeds, start=1):
        if time.time() >= deadline - 5.0:
            break
        n = seed_vec.shape[0]
        print(f"\nRefining seed {idx}/{len(top_seeds)} (n={n}) …")

        # Sub‑gradient descent (uses the exact objective)
        remaining = deadline - time.time()
        max_iter_sub = int(min(120_000, remaining * 30))
        refined = subgradient_descent_exact(seed_vec, deadline,
                                            max_iter=max_iter_sub,
                                            step0=0.18)

        # Exponentiated‑gradient polishing
        if time.time() < deadline - 2.0:
            remaining = deadline - time.time()
            max_iter_eg = int(min(80_000, remaining * 30))
            refined = eg_refine(refined, deadline,
                                max_iter=max_iter_eg,
                                eta0=0.25)

        # Official evaluation
        try:
            val = evaluate_sequence(refined.tolist())
            print(f"  → official score = {val:.6f}")
            if val < best_val:
                best_val = val
                best_seq = refined.tolist()
                print(f"  *** New global best: {best_val:.6f} ***")
        except Exception as e:
            print(f"  Evaluation failed: {e}")

    # ------------------------------------------------------------------
    # 3️⃣  LP‑based polishing on the current best (if time permits)
    # ------------------------------------------------------------------
    if best_seq and time.time() < deadline - 5.0:
        print("\nStarting LP‑based polishing …")
        a = np.array(best_seq, dtype=float)
        a = a / a.sum()                      # ensure simplex
        a = lp_refinement(a, deadline, max_iter=12)

        # Final official evaluation after LP polishing
        try:
            val = evaluate_sequence(a.tolist())
            print(f"LP final official score = {val:.6f}")
            if val < best_val:
                best_val = val
                best_seq = a.tolist()
                print("*** Updated global best after LP polishing ***")
        except Exception as e:
            print(f"LP final evaluation failed: {e}")

    # ------------------------------------------------------------------
    # 4️⃣  Optional final EG polish (if any time left)
    # ------------------------------------------------------------------
    if best_seq and time.time() < deadline - 3.0:
        print("\nFinal exponentiated‑gradient polish …")
        a = np.array(best_seq, dtype=float)
        a = eg_refine(a, deadline, max_iter=80_000, eta0=0.20)
        try:
            val = evaluate_sequence(a.tolist())
            print(f"Final polished official score = {val:.6f}")
            if val < best_val:
                best_val = val
                best_seq = a.tolist()
                print("*** Updated global best after final EG polish ***")
        except Exception as e:
            print(f"Final polish failed: {e}")

    # ------------------------------------------------------------------
    # 5️⃣  Guard – ensure a non‑empty return value
    # ------------------------------------------------------------------
    if not best_seq:
        fallback_n = 512
        best_seq = [1.0 / fallback_n] * fallback_n
        best_val = evaluate_sequence(best_seq)

    print(f"\nSearch finished. Best reported official score = {best_val:.6f}")
    return best_seq
