"""
Propose a low‑score step‑function.

Algorithm
---------
1.  Build a hierarchy of lengths (1024 → 2048 → 4096 → 8192 → 16384).
    For each length we start from a few sensible seeds
    (continuous arcsine density, a few Dirichlet draws and an up‑sampled
    best solution from the previous level).

2.  Each seed is refined using a *smoothed* version of the true objective

        Φ(a) = 2·n·max(a∗a)/(∑a)²

    The max is approximated by a log‑sum‑exp with a temperature τ :

        Mτ(a) = (1/τ)·log∑ₖ exp(τ·(a∗a)ₖ).

    For fixed τ the function is smooth; its gradient can be written
    analytically in terms of a single FFT‑based convolution:

        ∇Φ(a) = 4·n·( g – Mτ·1 ),       gⱼ = Σₖ wₖ·a_{k‑j},
        wₖ = softmax(τ·(a∗a)ₖ).

    The gradient step is performed on the probability simplex
    (non‑negative entries, sum = 1) by a simple Euclidean projection
    (clip negatives and renormalise).

    We start with a low temperature (τ≈2) – the landscape is almost
    quadratic – and then increase τ (2 → 6 → 15 → 30 → 60).  This
    continuation strategy helps to escape poor local minima and to
    fine‑tune the final max‑convolution.

3.  The best vector over **all** lengths is kept.  After the hierarchy
    a final polishing phase runs a longer optimisation at the best
    length with a high temperature (τ=120) and a smaller step size.

All heavy lifting uses FFT‑based convolutions, therefore each iteration
runs in O(n log n) time and easily fits into the 1000 s wall‑clock
budget.  The routine always returns a feasible (non‑negative, sum>0)
list; the official ``evaluate_sequence`` is called only for the few
promising candidates, keeping the total running time low.
"""

import time
import numpy as np
from typing import List, Tuple

# ----------------------------------------------------------------------
# Basic numeric helpers
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Return the smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """
    Full linear convolution using a real FFT.
    Result length = len(a) + len(b) – 1.
    """
    m = len(a) + len(b) - 1
    n_fft = next_pow2(m)
    A = np.fft.rfft(a, n=n_fft)
    B = np.fft.rfft(b, n=n_fft)
    C = A * B
    out = np.fft.irfft(C, n=n_fft)[:m]
    return out


def compute_score(a: np.ndarray) -> float:
    """
    Fast surrogate for the official evaluator.
    The official ``evaluate_sequence`` is exactly
    2·n·max(conv)/sum(a)² (apart from harmless clipping);
    this function reproduces it.
    """
    n = len(a)
    conv = convolution_fft(a, a)
    max_val = float(conv.max())
    s = float(a.sum())
    if s < 1e-12:
        return float("inf")
    return 2.0 * n * max_val / (s * s)


# ----------------------------------------------------------------------
# Initial shapes
# ----------------------------------------------------------------------


def arcsine_initial(n: int) -> np.ndarray:
    """Discrete arcsine density: a_i ∝ 1/√((i+½)(n‑½‑i))."""
    i = np.arange(n, dtype=float)
    eps = 0.25  # tiny regulariser
    vals = 1.0 / np.sqrt((i + 0.5) * (n - 0.5 - i) + eps)
    return vals / vals.sum()


def random_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    """Draw a Dirichlet‑like vector with concentration `conc`."""
    if conc <= 0.0:
        # heavy‑tailed near‑sparse draws
        samp = rng.exponential(1.0, size=n) ** (1.0 / max(0.1, -conc))
    else:
        samp = rng.gamma(conc, 1.0, size=n)
    samp += 1e-12
    return samp / samp.sum()


def perturb(a: np.ndarray, rng: np.random.Generator, scale: float = 0.02) -> np.ndarray:
    """Add a tiny Gaussian jitter and renormalise."""
    b = a + rng.normal(0.0, scale, size=a.shape)
    b = np.maximum(b, 0.0)
    s = b.sum()
    if s == 0.0:
        return np.full_like(a, 1.0 / len(a))
    return b / s


def upsample(prev: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation from ``len(prev)`` to ``new_n`` and re‑normalise."""
    x_old = np.linspace(0.0, 1.0, len(prev))
    x_new = np.linspace(0.0, 1.0, new_n)
    interp = np.interp(x_new, x_old, prev)
    total = interp.sum()
    if total == 0.0:
        return np.full(new_n, 1.0 / new_n)
    return interp / total


# ----------------------------------------------------------------------
# Smoothed optimisation utilities
# ----------------------------------------------------------------------


def _softmax_weights(conv: np.ndarray, tau: float) -> Tuple[np.ndarray, float]:
    """
    Return the soft‑max weights wₖ = exp(τ·cₖ)/∑exp(τ·c) and the smoothed
    maximum Mτ = (1/τ)·log∑exp(τ·cₖ) using a numerically stable formulation.
    """
    c_max = conv.max()
    # shift for stability, never overflow
    shifted = tau * (conv - c_max)
    exp_shifted = np.exp(shifted)
    Z = exp_shifted.sum()
    w = exp_shifted / Z
    M_tau = c_max + np.log(Z) / tau
    return w, M_tau


def _cross_corr(w: np.ndarray, a: np.ndarray) -> np.ndarray:
    """
    Compute gⱼ = Σₖ wₖ·a_{k‑j} for all j (the cross‑correlation of w and a).
    This is performed via a single FFT‑based convolution.
    """
    # a_rev[j] = a[n‑1‑j]
    a_rev = a[::-1]
    conv_full = convolution_fft(w, a_rev)          # length = (2n‑1) + n – 1 = 3n‑2
    n = len(a)
    # The relevant slice starts at index n‑1
    return conv_full[n - 1 : n - 1 + n]


def smooth_optimize(
    init_a: np.ndarray,
    deadline: float,
    schedule: List[Tuple[float, int]],
    eta0_factor: float = 0.5,
) -> Tuple[np.ndarray, float]:
    """
    Optimise the smoothed objective using projected gradient descent.
    ``schedule`` is a list of (τ, iterations) pairs.  The step size follows
    ηₜ = (η₀ / √t) with η₀ = eta0_factor / n.
    """
    a = np.maximum(init_a, 1e-30)
    a /= a.sum()
    best_a = a.copy()
    best_sc = compute_score(a)
    n = len(a)
    it_global = 0
    eta0 = eta0_factor / n

    for tau, max_iter in schedule:
        for it in range(1, max_iter + 1):
            if time.time() >= deadline:
                return best_a, best_sc
            it_global += 1

            # convolution a*a
            conv = convolution_fft(a, a)

            # soft‑max weights and smoothed max
            w, M_tau = _softmax_weights(conv, tau)

            # g = Σₖ wₖ·a_{k‑j}
            g = _cross_corr(w, a)

            # gradient of the smoothed objective (s = 1 because we stay on simplex)
            grad = 4.0 * n * (g - M_tau)   # shape (n,)

            # step size
            eta = eta0 / np.sqrt(it_global)

            # gradient step + projection onto simplex (clip + renormalise)
            a = a - eta * grad
            a = np.maximum(a, 0.0)
            s = a.sum()
            if s == 0.0:
                a = np.full_like(a, 1.0 / n)
            else:
                a = a / s

            # keep the best seeing so far (using the exact surrogate)
            cur_sc = compute_score(a)
            if cur_sc < best_sc:
                best_sc = cur_sc
                best_a = a.copy()

            # occasional progress report
            if it_global % 2000 == 0:
                print(
                    f"       smooth iter={it_global:,d} τ={tau:.1f} cur={cur_sc:.6f} best={best_sc:.6f}"
                )

    return best_a, best_sc
