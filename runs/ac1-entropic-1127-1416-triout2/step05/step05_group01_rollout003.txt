`evaluate_sequence`.  The method combines three ideas:

1️⃣  A short exponentiated sub‑gradient descent on the exact (scale‐invariant)
    objective `score = 2·n·max(a*a)`.  This quickly flattens the
    autocorrelation.

2️⃣  A Frank–Wolfe‑style linear‑program refinement:
    given the current vector `f` with maximal convolution value `M`,
    solve the LP

        maximise  Σ b_i
        s.t.      (f * b)_k ≤ M   for all k,   b_i ≥ 0 .

    normalising `b` back to sum 1 the self‑convolution peak drops.
    Repeating this step yields monotone improvement.

    polishes the result.

`height_sequence_1` that may be provided by the harness; otherwise a fresh

def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Linear convolution via a real FFT (fast)."""
    m = a.shape[0] + b.shape[0] - 1
    n_fft = next_pow2(m)
    A = np.fft.rfft(a, n=n_fft)
    B = np.fft.rfft(b, n=n_fft)
    C = A * B
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


def compute_score(a: np.ndarray) -> float:
    """
    Normalised score used for internal comparisons.
    The sequence is assumed to be non‑negative and sum(a) == 1.
    The returned value equals the one produced by `evaluate_sequence`.
    """
    n = a.shape[0]
    conv = convolution_fft(a, a)
    return 2.0 * n * float(np.max(conv))


# ----------------------------------------------------------------------
# Simple Euclidean projection onto the probability simplex
# ----------------------------------------------------------------------


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Project a vector onto the simplex {x ≥ 0, Σx = 1}."""
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / v.shape[0])

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, v.shape[0] + 1)
    cond = u - cssv / ind > 0
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


# ----------------------------------------------------------------------
# 1️⃣  Exponentiated sub‑gradient optimisation
# ----------------------------------------------------------------------


def exp_subgrad_optimize(init_a: np.ndarray,
                         time_deadline: float,
                         max_iter: int = 5_000_000,
                         eta0: float = 0.5) -> tuple[np.ndarray, float]:
    """
    Exponentiated sub‑gradient descent on   f(a)=2·n·max(a*a).
    The update is   a ← a·exp(−η·g)   with a projection onto the simplex.
    """
    a = init_a.copy()
    best_a = a.copy()
    best_score = compute_score(a)
    n = a.shape[0]

    for t in range(1, max_iter + 1):
        if time.time() >= time_deadline:
            break

        conv = convolution_fft(a, a)
        k = int(np.argmax(conv))

        # build sub‑gradient for the active shift k
        idx = np.arange(n)
        j = k - idx
        mask = (j >= 0) & (j < n)
        g = np.zeros_like(a)
        g[mask] = 2.0 * a[j[mask]]

        eta = eta0 / math.sqrt(t)

        a = a * np.exp(-eta * g)
        a = project_to_simplex(a)

        cur_score = compute_score(a)
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()

        # tiny jitter every few thousand iterations
        if t % 5000 == 0:
            a += np.random.randn(n) * (eta0 * 1e-4 / math.sqrt(t))
            a = project_to_simplex(a)

    return best_a, best_score


# ----------------------------------------------------------------------
# 2️⃣  Linear‑program refinement (Frank–Wolfe style)
# ----------------------------------------------------------------------


def lp_refinement(start_seq: np.ndarray,
                  time_deadline: float,
                  max_iters: int = 12,
                  eps: float = 1e-12) -> tuple[np.ndarray, float]:
    """
    Repeatedly solve

        maximise Σ b_i
        s.t.   (start * b)_k ≤ M   (∀k),   b_i ≥ 0

    where M = max(start * start).  The solution b is normalised back to sum 1,
    yielding a strictly improved vector if possible.  The process stops when
    no further improvement is observed or the time budget expires.
    """
    a = start_seq.copy()
    n = a.shape[0]
    best_a = a.copy()
    best_score = compute_score(a)

    for it in range(max_iters):
        if time.time() >= time_deadline:
            break

        # current maximal convolution value
        conv = convolution_fft(a, a)
        M = float(np.max(conv))

        # Build the (2n‑1) × n Toeplitz matrix that implements linear convolution
        col = np.concatenate([a, np.zeros(n - 1, dtype=a.dtype)])
        row = np.concatenate([a[:1], np.zeros(n - 1, dtype=a.dtype)])
        T = toeplitz(col, row)                     # shape (2n‑1, n)

        # Linear programme: maximise sum(b)  ⇔ minimise -sum(b)
        c = -np.ones(n, dtype=a.dtype)

        # Inequalities T·b ≤ M
        A_ub = T
        b_ub = np.full(T.shape[0], M, dtype=a.dtype)

        bounds = [(0.0, None)] * n

        res = linprog(c,
                      A_ub=A_ub,
                      b_ub=b_ub,
                      bounds=bounds,
                      method='highs',
                      options={'presolve': True})

        if not res.success:
            print(f"LP iteration {it}: failure – {res.message}")
            break

        b = res.x
        sum_b = b.sum()
        if sum_b <= eps:
            print(f"LP iteration {it}: degenerate solution (sum≈0).")
            break

        # Normalise back to the simplex
        g = b / sum_b
        # Ensure a tiny epsilon to keep strict positivity (helps the next LP)
        g = np.maximum(g, eps)
        g = project_to_simplex(g)

        new_score = compute_score(g)

        if new_score + 1e-12 < best_score:
            best_score = new_score
            best_a = g.copy()
            a = g
            print(f"LP iteration {it}: improved score → {best_score:.6f}")
        else:
            print(f"LP iteration {it}: no improvement (score {new_score:.6f})")
            break

    return best_a, best_score


# ----------------------------------------------------------------------
# 3️⃣  Stochastic hill‑climber (swap / mass‑transfer moves)
# ----------------------------------------------------------------------


def hill_climb(seq: np.ndarray,
               time_deadline: float,
               max_steps: int = 500_000) -> tuple[np.ndarray, float]:
    """
    Simple, long‑running hill‑climber that attempts to lower the score.
    Moves are either swaps of two entries or a transfer of a small fraction
    of mass from one index to another.  The candidate is always projected
    onto the probability simplex.
    """
    n = seq.shape[0]
    cur = seq.copy()
    cur_score = compute_score(cur)
    best = cur.copy()
    best_score = cur_score

    # initial transfer fraction (relative to source mass)
    step_frac = 0.08

    while max_steps > 0 and time.time() < time_deadline:
        move_type = np.random.rand()
        cand = best.copy()

        if move_type < 0.5:
            # swap two indices
            i, j = np.random.randint(n, size=2)
            cand[i], cand[j] = cand[j], cand[i]
        else:
            # transfer a fraction of mass from i → j
            i, j = np.random.randint(n, size=2)
            if i != j and cand[i] > 0.0:
                delta = step_frac * cand[i]
                cand[i] -= delta
                cand[j] += delta

        cand = project_to_simplex(cand)
        score = compute_score(cand)

        if score < best_score:
            best_score = score
            best = cand
            # successful step → try a slightly larger step
            step_frac = min(step_frac * 1.07, 0.5)
        else:
            # unsuccessful → shrink step size
            step_frac *= 0.97

        max_steps -= 1

    return best, best_score


# ----------------------------------------------------------------------
# Main entry point for the evaluation harness
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative sequence that minimises `evaluate_sequence`.
    The algorithm proceeds in three phases (sub‑gradient, LP refinement,
    hill‑climb) while respecting the 1000 s budget.
    """
    np.random.seed(seed)
    start = time.time()
    deadline = start + budget_s - 1.5               # safety margin

    # ------------------------------------------------------------------
    # 0️⃣  Choose an initial vector (fallback or fresh random Dirichlet)
    # ------------------------------------------------------------------
    try:
        # may be supplied by the harness
        fallback = np.array([float(x) for x in height_sequence_1], dtype=float)
        print(f"Using provided fallback of length {fallback.size}.")
    except Exception:
        fallback = None

    if fallback is not None and fallback.size >= 256:
        a0 = fallback
    else:
        # a reasonably large dimension – 2048 gives a tighter theoretical bound
        n = 2048
        a0 = np.random.dirichlet(np.ones(n))

    # Normalise to the probability simplex (sum = 1) and guard against exact zeros
    eps = 1e-12
    a0 = np.maximum(a0, eps)
    a0 = a0 / a0.sum()

    best_seq = a0.copy()
    best_score = compute_score(best_seq)
    print(f"Initial score = {best_score:.6f}")

    # ------------------------------------------------------------------
    # 1️⃣  Exponentiated sub‑gradient descent (quick flattening)
    # ------------------------------------------------------------------
    if time.time() < deadline - 30.0:
        subgrad_time = min(deadline - time.time() - 10.0, 200.0)   # at most ~200 s
        sub_deadline = time.time() + subgrad_time
        a_sub, sub_score = exp_subgrad_optimize(best_seq,
                                                time_deadline=sub_deadline,
                                                max_iter=2_000_000,
                                                eta0=0.7)
        if sub_score < best_score:
            best_seq = a_sub
            best_score = sub_score
            print(f"Sub‑gradient phase improved score to {best_score:.6f}")

    # ------------------------------------------------------------------
    # 2️⃣  Linear‑program refinement (Frank–Wolfe style)
    # ------------------------------------------------------------------
    if time.time() < deadline - 20.0:
        lp_seq, lp_score = lp_refinement(best_seq,
                                         time_deadline=deadline - 10.0,
                                         max_iters=12)
        if lp_score < best_score:
            best_seq = lp_seq
            best_score = lp_score
            print(f"LP refinement improved score to {best_score:.6f}")

    # ------------------------------------------------------------------
    # 3️⃣  Final hill‑climb polishing
    # ------------------------------------------------------------------
    if time.time() < deadline - 5.0:
        hill_seq, hill_score = hill_climb(best_seq,
                                          time_deadline=deadline,
                                          max_steps=500_000)
        if hill_score < best_score:
            best_seq = hill_seq
            best_score = hill_score
            print(f"Hill‑climb final improvement → {best_score:.6f}")

    # ------------------------------------------------------------------
    # Safety fallback (should never be needed)
    # ------------------------------------------------------------------
    if best_seq is None or best_seq.size == 0:
        n = 256
        best_seq = np.full(n, 1.0 / n, dtype=float)

    # ------------------------------------------------------------------
    # Return the sequence as a plain Python list (the harness expects this)
    # ------------------------------------------------------------------
    final_score = evaluate_sequence(best_seq.tolist())
    print(f"\nSearch finished. Final reported score = {final_score:.6f}")
    return best_seq.tolist()
