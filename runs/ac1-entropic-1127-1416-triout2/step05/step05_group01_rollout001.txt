"""
Propose a low‑score step function.

The algorithm works in three stages:

1️⃣  **Fast Adam on the simplex** – we parametrize a probability vector
    `a` by a soft‑max of unconstrained real parameters `w`.  The exact
    objective `score(a)=2·n·max(a∗a)` is differentiable except at ties;
    we use a sub‑gradient at the current maximal convolution index.
    Adam (adaptive moment estimation) with a tiny Gaussian jitter on `w`
    performs a mirror‑descent on the simplex.  The best vector seen so
    far is kept.

2️⃣  **Progressive refinement** – the best vector of length `n0=1024`
    is interpolated to a longer length (`n1=2048`) and the Adam step is
    repeated.  The extra degrees of freedom allow the convolution to be
    spread more uniformly.

3️⃣  **Stochastic hill‑climber** – a short combinatorial polishing step
    (swaps and small mass‑transfers) with projection onto the simplex
    removes any remaining local bumps.

All work respects a 1000 s budget.  The returned sequence is a plain
list of non‑negative floats; its score is printed for the harness.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# FFT‑based linear convolution (fast, using a single FFT)
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _autocorr_fft(a: np.ndarray, n_fft: int) -> np.ndarray:
    """Linear autocorrelation a*a using a single real FFT."""
    A = np.fft.rfft(a, n=n_fft)
    C = A * A
    conv = np.fft.irfft(C, n=n_fft)[: 2 * a.shape[0] - 1]
    return conv


def compute_score(a: np.ndarray) -> float:
    """
    Fast replica of ``evaluate_sequence`` for a vector whose entries sum to 1.
    Returns 2·n·max(a*a), exactly the value returned by evaluate_sequence.
    """
    n = a.shape[0]
    m = 2 * n - 1
    n_fft = _next_pow2(m)
    conv = _autocorr_fft(a, n_fft)
    return 2.0 * n * float(np.max(conv))


# ----------------------------------------------------------------------
# Soft‑max utilities (mirror‑descent on the simplex)
# ----------------------------------------------------------------------


def _softmax(w: np.ndarray) -> np.ndarray:
    """Numerically stable soft‑max."""
    z = w - np.max(w)
    e = np.exp(z)
    return e / e.sum()


# ----------------------------------------------------------------------
# Adam optimiser on the simplex (mirrored Adam)
# ----------------------------------------------------------------------


def adam_optimize_simplex(init_a: np.ndarray,
                          deadline: float,
                          max_iter: int = 5_000_000,
                          lr: float = 0.5,
                          beta1: float = 0.9,
                          beta2: float = 0.999,
                          eps: float = 1e-8,
                          jitter_scale: float = 1e-3,
                          jitter_every: int = 5000) -> tuple[np.ndarray, float]:
    """
    Mirror‑descent Adam on the probability simplex.

    Parameters
    ----------
    init_a : np.ndarray
        Initial probability vector (sums to 1, all >0).
    deadline : float
        Time (seconds since epoch) after which the optimiser stops.
    max_iter : int
        Upper bound on the number of iterations (soft, time limit dominates).
    lr : float
        Base learning rate for Adam (acts on the soft‑max parameters).
    beta1, beta2, eps : float
        Standard Adam hyper‑parameters.
    jitter_scale : float
        Standard deviation of Gaussian noise added to the parameters
        every ``jitter_every`` iterations (helps escape plateaus).
    jitter_every : int
        Iteration interval for jitter injection.

    Returns
    -------
    best_a, best_score
        The best probability vector seen and its (scaled) score.
    """
    n = init_a.shape[0]

    # Initialise the unconstrained parameters w = log(a)
    w = np.log(np.maximum(init_a, 1e-30))

    # Adam state
    m = np.zeros_like(w)
    v = np.zeros_like(w)
    beta1_pow = 1.0
    beta2_pow = 1.0

    # bookkeeping
    best_a = _softmax(w)
    best_score = compute_score(best_a)

    # pre‑compute FFT size (independent of w)
    m_len = 2 * n - 1
    n_fft = _next_pow2(m_len)

    t = 0
    while time.time() < deadline and t < max_iter:
        t += 1

        # current probability vector
        a = _softmax(w)

        # autocorrelation and maximal index
        conv = _autocorr_fft(a, n_fft)
        k = int(np.argmax(conv))

        # sub‑gradient of max‑convolution w.r.t. a
        g = np.zeros_like(a)
        idx = np.arange(n)
        j = k - idx
        mask = (j >= 0) & (j < n)
        g[mask] = 2.0 * a[j[mask]]

        # gradient w.r.t. the soft‑max parameters w
        #   ∂f/∂w_i = a_i * (g_i - ⟨a,g⟩)
        avg_g = np.dot(a, g)
        grad_w = a * (g - avg_g)

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad_w
        v = beta2 * v + (1.0 - beta2) * (grad_w * grad_w)

        beta1_pow *= beta1
        beta2_pow *= beta2

        m_hat = m / (1.0 - beta1_pow)
        v_hat = v / (1.0 - beta2_pow)

        w -= lr * m_hat / (np.sqrt(v_hat) + eps)

        # occasional tiny Gaussian jitter to shake the iterates
        if jitter_every and (t % jitter_every == 0):
            w += np.random.randn(n) * (jitter_scale / np.sqrt(t))

        # evaluate current score
        cur_score = compute_score(a)
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()

    return best_a, best_score


# ----------------------------------------------------------------------
# Simple hill‑climber (swap / mass‑transfer moves)
# ----------------------------------------------------------------------


def _project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection onto the probability simplex."""
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / v.shape[0])

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, v.shape[0] + 1)
    cond = u - cssv / ind > 0
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def hill_climb(seq: list[float],
               deadline: float,
               max_steps: int = 200_000) -> tuple[list[float], float]:
    """
    Local stochastic improvement: random swaps or small mass transfers,
    each time projecting back onto the simplex.
    """
    a = np.array(seq, dtype=float)
    n = a.shape[0]

    best_a = a.copy()
    best_score = compute_score(best_a)

    step_scale = 0.07  # how much mass is moved in a transfer move

    while max_steps > 0 and time.time() < deadline:
        cand = best_a.copy()
        if np.random.rand() < 0.5:
            # swap two entries
            i, j = np.random.randint(n, size=2)
            cand[i], cand[j] = cand[j], cand[i]
        else:
            # transfer a fraction of mass from i to j
            i, j = np.random.randint(n, size=2)
            if i != j:
                delta = step_scale * cand[i]
                cand[i] -= delta
                cand[j] += delta
        # re‑project onto the simplex
        cand = _project_to_simplex(cand)

        cur_score = compute_score(cand)
        if cur_score < best_score:
            best_score = cur_score
            best_a = cand
            step_scale = min(step_scale * 1.06, 0.5)   # success → larger steps
        else:
            step_scale *= 0.97                         # failure → smaller steps

        max_steps -= 1

    return best_a.tolist(), best_score


# ----------------------------------------------------------------------
# Utility: up‑sample a probability vector to a larger length
# ----------------------------------------------------------------------


def _upscale(a: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation of a probability vector to a larger size."""
    old_n = a.shape[0]
    if new_n == old_n:
        return a.copy()
    x_old = np.linspace(0.0, 1.0, old_n)
    x_new = np.linspace(0.0, 1.0, new_n)
    a_new = np.interp(x_new, x_old, a)
    a_new /= a_new.sum()                     # re‑normalise
    return a_new

# Main entry point required by the evaluation harness

def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step‑function that minimises ``evaluate_sequence``.
    The search consists of:
        * a few random Dirichlet starts,
        * Adam optimisation on the simplex (mirrored Adam),
        * progressive up‑scaling to a longer vector,
        * a final stochastic hill‑climb polish.
    All work respects the 1000 s budget.
    """
    np.random.seed(seed)
    start_time = time.time()
    # tiny safety margin for the harness
    deadline = start_time + budget_s - 1.5

    best_seq = None
    best_val = float("inf")

    # --------------------------------------------------------------
    # 0️⃣  Use a fallback if the notebook already provides one
    # --------------------------------------------------------------
    try:
        fallback = [float(x) for x in height_sequence_1]  # type: ignore
        fallback_val = evaluate_sequence(fallback)
        print(f"Fallback score = {fallback_val:.6f}")
        if fallback_val < best_val:
            best_val = fallback_val
            best_seq = fallback
    except Exception:
        pass

    # --------------------------------------------------------------
    # 1️⃣  First optimisation round on a moderate length (n=1024)
    # --------------------------------------------------------------
    n0 = 1024
    init_a = np.random.dirichlet(np.ones(n0))
    cur_score = compute_score(init_a)
    if cur_score < best_val:
        best_val = cur_score
        best_seq = init_a.tolist()
        print(f"Random start (n={n0}) score = {best_val:.6f}")

    # Run Adam on the simplex for the remaining budget (leaving ~300 s for later phases)
    deadline_phase1 = deadline - 300.0
    if time.time() < deadline_phase1:
        print("\nPhase 1 – Adam optimisation (n=1024)")
        a_opt, score_opt = adam_optimize_simplex(
            np.array(best_seq, dtype=float),
            deadline=deadline_phase1,
            max_iter=4_000_000,
            lr=0.6,
            jitter_scale=1e-3,
            jitter_every=4000,
        )
        if score_opt < best_val:
            best_val = score_opt
            best_seq = a_opt.tolist()
            print(f"Phase 1 improved best to {best_val:.6f}")

    # --------------------------------------------------------------
    # 2️⃣  Upscale to a longer vector (n=2048) and refine again
    # --------------------------------------------------------------
    n1 = 2048
    if time.time() < deadline - 150.0:
        print("\nPhase 2 – Upscaling to n=2048 and re‑optimising")
        a_up = _upscale(np.array(best_seq, dtype=float), n1)
        a_opt2, score_opt2 = adam_optimize_simplex(
            a_up,
            deadline=deadline - 120.0,
            max_iter=2_500_000,
            lr=0.5,
            jitter_scale=8e-4,
            jitter_every=3000,
        )
        if score_opt2 < best_val:
            best_val = score_opt2
            best_seq = a_opt2.tolist()
            print(f"Phase 2 improved best to {best_val:.6f}")

    # --------------------------------------------------------------
    # 3️⃣  Final short hill‑climb polishing
    # --------------------------------------------------------------
    if best_seq is not None and time.time() < deadline - 5.0:
        print("\nPhase 3 – Hill‑climb polishing")
        polished_seq, polished_val = hill_climb(
            best_seq,
            deadline=deadline,
            max_steps=250_000,
        )
        if polished_val < best_val:
            best_val = polished_val
            best_seq = polished_seq
            print(f"Hill‑climb improved best to {best_val:.6f}")

    # --------------------------------------------------------------
    # 4️⃣  Safeguard – if nothing succeeded, fall back to a tiny uniform vector
    # --------------------------------------------------------------
    if best_seq is None:
        n = 256
        best_seq = [1.0 / n] * n
        best_val = evaluate_sequence(best_seq)

    final_score = evaluate_sequence(best_seq)
    print(f"\nSearch finished. Final reported score = {final_score:.6f}")
    return best_seq
