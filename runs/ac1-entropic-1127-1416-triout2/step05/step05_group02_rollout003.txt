```
def smooth_optimize(initial_a, deadline, max_iter=20000,
                    lr=0.05, beta1=0.9, beta2=0.999, eps=1e-8,
                    alpha_start=1.0, alpha_end=64.0,
                    print_prefix=""):
    a = project_to_simplex(initial_a.copy())
    n = len(a)
    L = 2*n - 1
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    for t in range(1, max_iter+1):
        if time.time() >= deadline:
            break
        # compute convolution
        conv = convolution_fft(a, a)
        # compute smoothing weights
        max_conv = float(np.max(conv))
        # compute softmax weighting
        alpha = alpha_start * (alpha_end/alpha_start) ** ((t-1)/(max_iter-1)) if max_iter>1 else alpha_end
        # For stability subtract max_conv
        exp_vals = np.exp(alpha * (conv - max_conv))
        w = exp_vals / np.sum(exp_vals)
        # gradient
        w_rev = w[::-1]
        conv_rev = convolution_fft(a, w_rev)
        grad = 2.0 * conv_rev[L - 1 - np.arange(n)]
        # Adam step
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad ** 2)
        m_hat = m / (1 - beta1 ** t)
        v_hat = v / (1 - beta2 ** t)
        step = lr * m_hat / (np.sqrt(v_hat) + eps)
        a = a - step
        a = project_to_simplex(a)
        if t % 2000 == 0:
            cur_score = compute_score(a)
            print(f"{print_prefix}Iter {t:,d} smoothed alpha={alpha:.2f} score={cur_score:.6f}")
    return a
```

"""
Propose a low‑discrepancy step function.

The algorithm consists of four stages:

1️⃣   **Broad random sampling** –  generate many vectors on the simplex
     (Dirichlet draws, arcsine shape, Beta‑ and power‑families) for several
     lengths (256 … 4096).  Their fast surrogate score
     `2·n·max(conv(a,a))` is computed with an FFT.  The best few seeds are kept.

2️⃣   **Smooth Adam optimisation** –  each seed is refined by minimising a
     smoothed approximation of the max‑convolution:
        `Sα(a)= (1/α)·log Σ_i exp(α·(a*a)_i)`.
     The gradient of `Sα` is obtained analytically via a second convolution,
     and an Adam update on the probability simplex (projection after each step)
     is performed.  The smoothing parameter `α` is increased from 1 to a large
     value (≈64) during the run.

3️⃣   **Exact sub‑gradient descent + EG polishing** –  the refined vector is
     further improved with the non‑smooth sub‑gradient method used in the
     reference solution and a final exponentiated‑gradient (mirror‑descent)
     pass.

4️⃣   **Optional local perturbation** –  if time permits, a cheap
     random‑swap local search on the official objective is applied.

All stages respect a global time budget (default 1000 s).  The best vector
found is returned as a plain Python list of floats.
"""

import time
import numpy as np
from typing import List, Tuple

# ----------------------------------------------------------------------
# Low‑level utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Linear convolution via real FFT (length = len(a)+len(b)-1)."""
    m = len(a) + len(b) - 1
    n_fft = next_pow2(m)
    A = np.fft.rfft(a, n=n_fft)
    B = np.fft.rfft(b, n=n_fft)
    C = A * B
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


def compute_score(a: np.ndarray) -> float:
    """
    Fast surrogate for the official score.
    Vector is assumed to be normalised to Σa = 1, therefore the
    denominator disappears and the score is simply 2·n·max(conv(a,a)).
    """
    n = a.shape[0]
    conv = convolution_fft(a, a)
    return 2.0 * n * float(np.max(conv))


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection onto the probability simplex {x ≥ 0, Σx = 1}.
    Duchi et al., 2008.
    """
    if np.all(v <= 0):
        # uniform fallback – should never happen in practice
        return np.full_like(v, 1.0 / v.size)

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho_arr = u - cssv / np.arange(1, v.size + 1)
    rho = np.where(rho_arr > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Shape generators (used for the initial random pool)
# ----------------------------------------------------------------------


def arcsine_initial(n: int) -> np.ndarray:
    """Classic arcsine shape: a_i ∝ 1/√((i+½)*(n‑½‑i))."""
    i = np.arange(n, dtype=float)
    eps = 0.25
    vals = 1.0 / np.sqrt((i + 0.5) * (n - 0.5 - i) + eps)
    vals /= vals.sum()
    return vals


def generate_beta_shape(n: int, alpha: float, beta: float, eps: float) -> np.ndarray:
    """
    (i+ε)^{α‑1}·(n‑1‑i+ε)^{β‑1} on {0,…,n‑1}, normalised to Σ=1.
    """
    i = np.arange(n, dtype=float)
    left = (i + eps) ** (alpha - 1.0)
    right = (n - 1 - i + eps) ** (beta - 1.0)
    base = left * right
    total = base.sum()
    if total <= 0.0 or not np.isfinite(total):
        return np.full(n, 1.0 / n)
    return base / total


def generate_power_shape(n: int, p: float, eps: float) -> np.ndarray:
    """
    Generalised arcsine: a_i ∝ ((i+½)*(n‑½‑i)+ε)^{‑p}.
    p>0, eps≥0.
    """
    i = np.arange(n, dtype=float)
    denom = ((i + 0.5) * (n - 0.5 - i) + eps) ** p
    vals = 1.0 / denom
    total = vals.sum()
    if total <= 0.0 or not np.isfinite(total):
        return np.full(n, 1.0 / n)
    return vals / total


def generate_random_simplex(n: int, rng: np.random.Generator) -> np.ndarray:
    """Draw a random point uniformly from the simplex using Gamma(1) variables."""
    x = rng.gamma(1.0, size=n)
    x_sum = x.sum()
    if x_sum == 0.0:
        return np.full(n, 1.0 / n)
    return x / x_sum

def subgradient_descent_exact(initial_a: np.ndarray,
                              deadline: float,
                              max_iter: int = 80_000,
                              step0: float = 0.05) -> np.ndarray:
    """
    Sub‑gradient descent on the exact objective f(a)=2·n·max(conv(a,a)).
    The step size is η_t = step0 / √t.  Projection onto the simplex enforces Σa = 1.
    """
    a = project_to_simplex(initial_a.copy())
    n = len(a)
    L = 2 * n - 1

    for it in range(1, max_iter + 1):
        if time.time() >= deadline:
            break

        conv = convolution_fft(a, a)
        max_val = float(np.max(conv))

        # mask of (approx.) maximisers
        tol = max_val * 1e-12 + 1e-14
        mask = (np.abs(conv - max_val) <= tol).astype(float)

        mask_rev = mask[::-1]
        conv_rev = convolution_fft(a, mask_rev)
        subgrad = 2.0 * conv_rev[L - 1 - np.arange(n)]

        eta = step0 / np.sqrt(it)
        a = a - eta * subgrad
        a = project_to_simplex(a)

        if it % 5000 == 0:
            cur_sc = compute_score(a)
            print(f"    SubGrad it={it:,d} score={cur_sc:.6f}")

    return a


def eg_refine(a: np.ndarray,
              deadline: float,
              max_iter: int = 100_000,
              eta0: float = 0.3) -> np.ndarray:
    """
    Mirror‑descent style polishing using an exponentiated‑gradient update.
    Keeps the iterate on the probability simplex automatically.
    """
    a = a.copy()
    n = len(a)
    L = 2 * n - 1

    for it in range(1, max_iter + 1):
        if time.time() >= deadline:
            break

        conv = convolution_fft(a, a)
        max_val = float(np.max(conv))
        tol = max_val * 1e-12 + 1e-14
        mask = (np.abs(conv - max_val) <= tol).astype(float)

        mask_rev = mask[::-1]
        conv_rev = convolution_fft(a, mask_rev)
        subgrad = 2.0 * conv_rev[L - 1 - np.arange(n)]

        eta = eta0 / (np.sqrt(it) + 1.0)
        a = a * np.exp(-eta * subgrad)

        total = a.sum()
        if total == 0.0:
            a = np.full_like(a, 1.0 / n)
        else:
            a = a / total

        if it % 5000 == 0:
            cur_sc = compute_score(a)
            print(f"    EG it={it:,d} score={cur_sc:.6f}")

    return a


def smooth_optimize(initial_a: np.ndarray,
                    deadline: float,
                    max_iter: int = 20_000,
                    lr: float = 0.05,
                    beta1: float = 0.9,
                    beta2: float = 0.999,
                    eps: float = 1e-8,
                    alpha_start: float = 1.0,
                    alpha_end: float = 64.0,
                    prefix: str = "") -> np.ndarray:
    """
    Adam optimisation on a smoothed max‑convolution:

        Sα(a) = (1/α)·log Σ_i exp(α·(a*a)_i)

    The gradient ∇Sα is a weighted sum of the individual convolution gradients,
    which can be obtained via a second FFT.  The iterate is projected onto the
    probability simplex after each step.
    """
    a = project_to_simplex(initial_a.copy())
    n = len(a)
    L = 2 * n - 1

    m = np.zeros_like(a)
    v = np.zeros_like(a)

    for t in range(1, max_iter + 1):
        if time.time() >= deadline:
            break

        conv = convolution_fft(a, a)
        max_conv = float(np.max(conv))

        # smoothing parameter – geometric schedule
        if max_iter > 1:
            alpha = alpha_start * (alpha_end / alpha_start) ** ((t - 1) / (max_iter - 1))
        else:
            alpha = alpha_end

        # softmax weights for the current α
        # subtraction of max_conv avoids overflow
        exp_vals = np.exp(alpha * (conv - max_conv))
        w = exp_vals / np.sum(exp_vals)

        # gradient of the smoothed objective
        w_rev = w[::-1]
        conv_rev = convolution_fft(a, w_rev)
        grad = 2.0 * conv_rev[L - 1 - np.arange(n)]

        # Adam update (minimisation → subtract step)
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad ** 2)
        m_hat = m / (1 - beta1 ** t)
        v_hat = v / (1 - beta2 ** t)
        step = lr * m_hat / (np.sqrt(v_hat) + eps)

        a = a - step
        a = project_to_simplex(a)

        if t % 2000 == 0:
            cur_sc = compute_score(a)
            print(f"{prefix}Smooth it={t:,d} α={alpha:.2f} score={cur_sc:.6f}")

    return a


def local_perturbation(seq: np.ndarray,
                       deadline: float,
                       rng: np.random.Generator,
                       attempts: int = 500,
                       delta_frac: float = 0.02) -> np.ndarray:
    """
    Very cheap random local search on the official objective.
    Transfers a small amount of mass from one coordinate to another.
    """
    a = seq.copy()
    n = len(a)
    best_val = evaluate_sequence(a.tolist())
    for _ in range(attempts):
        if time.time() >= deadline:
            break
        i, j = rng.integers(0, n), rng.integers(0, n)
        if i == j:
            continue
        # pick a fraction of a[i] to move
        delta = delta_frac * rng.random() * a[i]
        if delta <= 0.0:
            continue
        a_new = a.copy()
        a_new[i] -= delta
        a_new[j] += delta
        # sum unchanged, non‑negativity preserved by construction
        val = evaluate_sequence(a_new.tolist())
        if val < best_val:
            best_val = val
            a = a_new
    return a
