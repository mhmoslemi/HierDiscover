def refine_beta_params(n, α, β, deadline):
    best_score = compute_score(beta_initial)
    best = (α, β)
    step = 0.5
    while time.time() < deadline and step > 1e-4:
        improved = False
        for dα in [-step, 0, step]:
            for dβ in [-step, 0, step]:
                if dα == 0 and dβ == 0: continue
                a_new = β_shape(n, best[0] + dα, best[1] + dβ)
                sc = compute_score(a_new)
                if sc < best_score:
                    best_score = sc
                    best = (best[0] + dα, best[1] + dβ)
                    improved = True
        if not improved:
            step *= 0.5
    return beta_shape(n, *best), best_score

def propose_candidate(seed=42,budget_s=1000,**kwargs):
    start = time.time()
    deadline = start + budget_s - 5.0

    rng = np.random.default_rng(seed)

    best_seq = None
    best_val = float('inf')

    try fallback etc.

    lengths = [1024, 2048, 4096]

    for n in lengths:
        if time.time() >= deadline - 10: break

        print(f"\n--- Length {n} ---")

        # Build candidates
        candidates = []  # list of (score, vector)
        # uniform
        uniform_vec = np.full(n, 1.0 / n)
        candidates.append((compute_score(uniform_vec), uniform_vec))

        # arcsine
        arcs_vec = arcsine_initial(n)
        candidates.append((compute_score(arcs_vec), arcs_vec))

        # random beta sampling
        num_beta = 800 if n<=2048 else 400   # reduce for larger n to stay within time
        for _ in range(num_beta):
            α = 10**rng.uniform(-1.5, 0.7)  # 0.03 to 5
            β = 10**rng.uniform(-1.5, 0.7)
            vec = generate_beta_like(n, α, β)
            candidates.append((compute_score(vec), vec))

        # random powerlaw
        num_pow = 400
        for _ in range(num_pow):
            p = rng.uniform(0.2, 1.5)
            eps = rng.uniform(0.001, max(1, 0.02*n))
            vec = generate_powerlaw(n, p, eps)
            candidates.append((compute_score(vec), vec))

        for conc in [0.3,0.8,1.5,3.0]:

        for _ in range(5):

            candidates.append((compute_score(vec), vec))

"""
Hybrid global‑parameter + local‑gradient search for a low‑max‑convolution step
function.

The algorithm works in three stages:

1. **Global parametric sweep** – For each candidate length we sample many
   Beta‑type (a_i ∝ (i+½)^{α‑1}(n‑1‑i+½)^{β‑1}), power‑law, Dirichlet and a few
   handcrafted shapes (uniform, arcsine, perturbed).  The cheap fast score
   ``2·n·max(a∗a)`` (sum is normalised to 1) is used to keep only the
   most promising vectors.

2. **Local refinement** – Each kept vector is improved by a short
   simulated‑annealing run, then by a smooth‑max (log‑sum‑exp) gradient
   descent that works on the full‑size vector.  The best result of this
   phase becomes the candidate for the next step.

3. **Final polishing** – The overall best vector is fed to a long
   exponentiated‑sub‑gradient optimisation on the true max‑convolution.
   This step runs only if time permits and yields the final answer.

All operations respect the simplex (non‑negative entries, unit sum) and
the wall‑clock budget (default 1000 s).  The routine returns a feasible
list of floats ready for ``evaluate_sequence``.
"""

import time
import numpy as np
from typing import List, Tuple

# ----------------------------------------------------------------------
# Basic numerical utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full linear convolution using real FFT (zero‑padded to a power of two)."""
    m = len(a) + len(b) - 1
    n_fft = next_pow2(m)
    A = np.fft.rfft(a, n=n_fft)
    B = np.fft.rfft(b, n=n_fft)
    C = A * B
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


def compute_score(a: np.ndarray) -> float:
    """
    Fast version of the official objective (scale‑invariant):
        2·n·max(a∗a) / (sum(a)²)
    The vector a is assumed non‑negative and already normalised to sum 1.
    """
    n = len(a)
    conv = convolution_fft(a, a)
    max_val = float(np.max(conv))
    s = float(np.sum(a))
    if s < 1e-12:
        return float("inf")
    return 2.0 * n * max_val / (s * s)


# ----------------------------------------------------------------------
# Families for generating seed vectors
# ----------------------------------------------------------------------


def arcsine_initial(n: int) -> np.ndarray:
    """Discrete arcsine density (normalised to sum = 1)."""
    i = np.arange(n, dtype=float)
    eps = 0.25
    vals = 1.0 / np.sqrt((i + 0.5) * (n - 0.5 - i) + eps)
    return vals / vals.sum()


def generate_powerlaw(n: int, p: float, eps: float) -> np.ndarray:
    """Symmetric power‑law a_i ∝ ((i+ε)(n‑1‑i+ε))⁻ᵖ."""
    i = np.arange(n, dtype=float)
    base = ((i + eps) * (n - 1 - i + eps)) ** (-p)
    total = base.sum()
    if total == 0.0:
        return np.full(n, 1.0 / n)
    return base / total


def generate_beta_like(n: int, alpha: float, beta: float) -> np.ndarray:
    """Beta‑like shape a_i ∝ (i+½)^{α‑1} (n‑1‑i+½)^{β‑1}."""
    i = np.arange(n, dtype=float)
    base = (i + 0.5) ** (alpha - 1) * (n - 1 - i + 0.5) ** (beta - 1)
    total = base.sum()
    if total == 0.0:
        return np.full(n, 1.0 / n)
    return base / total


def random_dirichlet(n: int, concentration: float, rng: np.random.Generator) -> np.ndarray:
    """Sample a Dirichlet‑like vector (Gamma draws) and normalise."""
    if concentration <= 0:
        samp = rng.exponential(1.0, size=n) ** (1.0 / max(0.1, concentration))
    else:
        samp = rng.gamma(concentration, 1.0, size=n)
    samp += 1e-12
    return samp / samp.sum()


def perturb(a: np.ndarray, rng: np.random.Generator, scale: float = 0.02) -> np.ndarray:
    """Add a tiny Gaussian perturbation and renormalise onto the simplex."""
    b = a + rng.normal(0.0, scale, size=a.shape)
    b = np.maximum(b, 0.0)
    s = b.sum()
    if s == 0.0:
        return np.full_like(a, 1.0 / len(a))
    return b / s


# ----------------------------------------------------------------------
# Simulated annealing (lightweight local stochastic improvement)
# ----------------------------------------------------------------------


def propose_neighbor(a: np.ndarray,
                    rng: np.random.Generator,
                    max_step: float = 0.02) -> np.ndarray:
    """Move a small amount of mass from one random bin to another."""
    n = len(a)
    i, j = rng.integers(0, n, size=2)
    if i == j or a[i] == 0.0:
        return a
    delta = a[i] * rng.random() * max_step
    b = a.copy()
    b[i] -= delta
    b[j] += delta
    return b


def simulated_annealing(a0: np.ndarray,
                        deadline: float,
                        rng: np.random.Generator,
                        max_iter: int = 2000,
                        start_temp: float = 0.005,
                        cooling: float = 0.9995,
                        max_step: float = 0.02) -> Tuple[np.ndarray, float]:
    """Very light SA that stays on the probability simplex."""
    a = a0.copy()
    cur_sc = compute_score(a)
    best_a = a.copy()
    best_sc = cur_sc
    temp = start_temp

    for _ in range(max_iter):
        if time.time() >= deadline:
            break
        cand = propose_neighbor(a, rng, max_step)
        cand_sc = compute_score(cand)
        delta = cand_sc - cur_sc
        if delta < 0.0 or rng.random() < np.exp(-delta / max(1e-12, temp)):
            a = cand
            cur_sc = cand_sc
            if cur_sc < best_sc:
                best_sc = cur_sc
                best_a = a.copy()
        temp *= cooling
    return best_a, best_sc


# ----------------------------------------------------------------------
# Smooth‑max (log‑sum‑exp) gradient descent
# ----------------------------------------------------------------------


def softmax_gradient_descent(initial_a: np.ndarray,
                             deadline: float,
                             betas: List[float],
                             iter_per_beta: int = 500,
                             eta0: float = 0.5) -> Tuple[np.ndarray, float]:
    """
    Minimise a smooth surrogate
        Φ_β(a) = (1/β)·log Σ_k exp(β·(a∗a)_k)
    with an exponentiated‑gradient update followed by a projection onto the simplex.
    """
    a = np.maximum(initial_a, 1e-30)
    a = a / a.sum()
    best_a = a.copy()
    best_sc = compute_score(a)

    for beta in betas:
        for it in range(1, iter_per_beta + 1):
            if time.time() >= deadline:
                return best_a, best_sc

            conv = convolution_fft(a, a)               # length 2n‑1
            max_c = conv.max()
            exp_vals = np.exp(beta * (conv - max_c))
            p = exp_vals / exp_vals.sum()              # soft‑max weights, Σp=1

            # cross‑correlation p ⋆ a   (linear, length n)
            N = next_pow2(len(p) + len(a) - 1)
            P = np.fft.fft(p, N)
            A = np.fft.fft(a, N)
            cross = np.fft.ifft(P * np.conj(A)).real[:len(a)]
            cross = np.maximum(cross, 0.0)

            grad = 4.0 * len(a) * cross                # non‑negative gradient

            eta = eta0 / np.sqrt(it)
            a = a * np.exp(-eta * grad)

            s = a.sum()
            if s == 0.0:
                a = np.full_like(a, 1.0 / len(a))
            else:
                a = a / s

            cur_sc = compute_score(a)
            if cur_sc < best_sc:
                best_sc = cur_sc
                best_a = a.copy()

    return best_a, best_sc


# ----------------------------------------------------------------------
# Exponentiated sub‑gradient on the true max‑convolution (final polishing)
# ----------------------------------------------------------------------


def _max_conv_gradient(a: np.ndarray, conv: np.ndarray) -> np.ndarray:
    """
    Sub‑gradient of max_k (a∗a)_k.
    For each active offset k, the gradient contribution is 2·a_{k‑j}.
    When several offsets share the maximal value they are averaged.
    """
    n = len(a)
    max_val = conv.max()
    mask = np.isclose(conv, max_val, atol=1e-15, rtol=0.0)
    active = np.where(mask)[0]
    grad = np.zeros_like(a)

    for k in active:
        lo = max(0, k - (n - 1))
        hi = min(k, n - 1)
        idx = np.arange(lo, hi + 1, dtype=int)
        grad[idx] += 2.0 * a[k - idx]

    if len(active) > 1:
        grad /= len(active)
    return grad


def optimise_subgradient(initial_a: np.ndarray,
                         deadline: float,
                         *,
                         eta0: float = 0.6,
                         max_iter: int = 200_000,
                         noise_every: int = 0,
                         noise_scale: float = 0.0) -> Tuple[np.ndarray, float]:
    """
    Exponentiated sub‑gradient descent directly on Φ(a)=max_k (a∗a)_k.
    Returns the best vector seen and its fast score.
    """
    a = np.maximum(initial_a, 1e-30)
    a = a / a.sum()
    best_a = a.copy()
    best_sc = compute_score(a)
    n = len(a)

    for it in range(1, max_iter + 1):
        if time.time() >= deadline:
            break

        conv = convolution_fft(a, a)
        grad = _max_conv_gradient(a, conv)

        eta = eta0 / np.sqrt(it)
        a = a * np.exp(-eta * grad)

        if noise_every and (it % noise_every == 0):
            a += np.random.randn(n) * noise_scale
            a = np.maximum(a, 0.0)

        s = a.sum()
        a = a / s if s != 0.0 else np.full_like(a, 1.0 / n)

        cur_sc = compute_score(a)
        if cur_sc < best_sc:
            best_sc = cur_sc
            best_a = a.copy()

        if it % 2000 == 0:
            print(f"    subgrad iter={it:,d} cur={cur_sc:.6f} best={best_sc:.6f}")

    return best_a, best_sc

def refine_beta_params(n: int,
                       alpha0: float,
                       beta0: float,
                       deadline: float) -> Tuple[np.ndarray, float]:
    """
    Simple coordinate‑wise hill‑climbing on (α,β) for the Beta‑like family.
    The returned vector is already normalised (sum = 1).
    """
    best_alpha, best_beta = alpha0, beta0
    best_vec = generate_beta_like(n, best_alpha, best_beta)
    best_sc = compute_score(best_vec)
    step = 0.5

    while step > 1e-4 and time.time() < deadline:
        improved = False
        for dα in (-step, 0.0, step):
            for dβ in (-step, 0.0, step):
                if dα == 0.0 and dβ == 0.0:
                    continue
                α = best_alpha + dα
                β = best_beta + dβ
                if α <= 0.01 or β <= 0.01:
                    continue
                vec = generate_beta_like(n, α, β)
                sc = compute_score(vec)
                if sc < best_sc:
                    best_sc = sc
                    best_alpha, best_beta = α, β
                    best_vec = vec
                    improved = True
        if not improved:
            step *= 0.5

    return best_vec, best_sc

            start_temp = max(0.005, 0.02 * compute_score(cand))
            sa_vec, _ = simulated_annealing(
                cand,
                deadline,
                rng,
                max_iter=max_iter_sa,
                start_temp=start_temp,
                cooling=0.9995,
                max_step=0.015,
            )

            # ---- Smooth‑max gradient descent (modest schedule) ------------------
            betas = [5, 15, 30, 70, 150]
            remaining = max(0.0, deadline - time.time())
            it_per_beta = max(150, int(min(1200, remaining * 180) / len(betas)))
            sg_vec, _ = softmax_gradient_descent(
                sa_vec,
                deadline,
                betas=betas,
                iter_per_beta=it_per_beta,
                eta0=0.45,
            )

            # ---- Quick evaluation with the official scoring function ----------
            cur_val = evaluate_sequence(sg_vec.tolist())
            if cur_val < best_val:
                best_val = cur_val
                best_seq = sg_vec.tolist()
                print(f"    *** New global best (candidate {idx + 1}) = {best_val:.6f} ***")

    # ------------------------------------------------------------------
    # 2.  Final polishing of the overall best sequence (if time permits).
    # ------------------------------------------------------------------
    if best_seq and time.time() < deadline - 5.0:
        print("\nFinal polishing with exponentiated sub‑gradient")
        a = np.array(best_seq, dtype=float)
        remaining = max(0.0, deadline - time.time())
        max_iter = int(min(300_000, remaining * 250))
        refined_a, _ = optimise_subgradient(
            initial_a=a,
            deadline=deadline,
            eta0=0.3,
            max_iter=max_iter,
            noise_every=0,
            noise_scale=0.0,
        )
        final_val = evaluate_sequence(refined_a.tolist())
        print(f"Final polished score = {final_val:.6f}")
        if final_val < best_val:
            best_val = final_val
            best_seq = refined_a.tolist()
            print("*** Updated global best after final polishing ***")

    # ------------------------------------------------------------------
    # 3.  Guard – always return a feasible sequence.
    # ------------------------------------------------------------------
    if not best_seq:
        best_seq = [1.0] * 256
        best_val = evaluate_sequence(best_seq)

    print(f"\nSearch finished. Best reported score = {best_val:.6f}")
    return best_seq
