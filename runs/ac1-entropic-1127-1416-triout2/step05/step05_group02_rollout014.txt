```
def adam_optimize(initial_a: np.ndarray, deadline: float, max_iter: int,
                  lr: float = 0.08, beta1: float = 0.9, beta2: float = 0.999,
                  eps: float = 1e-8, beta_init: float = 5.0, beta_max: float = 200.0,
                  beta_inc: float = 0.001) -> np.ndarray:
    a = project_to_simplex(initial_a.copy())
    n = a.shape[0]
    L = 2*n - 1
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    for it in range(1, max_iter+1):
        if time.time() >= deadline:
            break
        conv = convolution_fft(a, a)
        max_val = float(np.max(conv))
        # smooth weights
        beta = beta_init + beta_inc * it
        if beta > beta_max:
            beta = beta_max
        # offsets
        d = conv - max_val   # <=0
        # stable calculation: exp(beta*d) may underflow for large negative, but okay
        w = np.exp(beta * d)
        w_sum = w.sum()
        if w_sum == 0.0:
            w = np.full_like(w, 1.0 / w.size)
        else:
            w = w / w_sum
        # gradient
        rev = w[::-1]
        conv_rev = convolution_fft(a, rev)
        grad = 2.0 * conv_rev[L - 1 - np.arange(n)]
        # Adam update
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad * grad)
        m_hat = m / (1 - beta1**it)
        v_hat = v / (1 - beta2**it)
        step = lr * m_hat / (np.sqrt(v_hat) + eps)
        a = a - step
        a = project_to_simplex(a)
        if it % 5000 == 0:
            cur_sc = compute_score(a)
            print(f"    Adam it={it:,d} beta={beta:.1f} score={cur_sc:.6f}")
    return a
```

assistantfinal**Solution Overview**

We search for a non‑negative step sequence `a = (a0,…,a_{n‑1})` that minimises  

\[
\text{score}(a)=\frac{2\,n\;\max (a*a)}{(\sum_i a_i)^2}\;,
\]

where `a*a` is the linear convolution of `a` with itself.
The score is *scale invariant* – we may always normalise the vector to
the probability simplex `Σa = 1`.  After normalisation the objective is
simply  

\[
f(a)=2\,n\;\max (a*a).
\]

The programme consists of four stages

1. **Broad sampling** – generate many candidates from a few parametric
   families (arcsine‑like, β‑family, power‑family, mixtures and simple
   triangular bumps).  The fast surrogate `f(a)` (computed with an FFT
   based convolution) is used for ranking.

2. **Seed enrichment** – keep the best few candidates and create a few
   extra mixtures of them.

3. **Refinement** – for every seed we run a *projected Adam optimiser*
   on the exact objective.  
   The Adam step uses the *smoothed* max, i.e. the gradient of  

   \[
   g_\beta(a)=\frac{1}{\beta}\log\!\bigl(\sum_k
   e^{\beta\,(a*a)_k}\bigr) ,
   \]

   with a temperature β that slowly increases from a small value
   (smooth landscape) to a large value (focus on the true maximum).
   After Adam we apply a short exponentiated‑gradient polish (mirror
   descent on the simplex).

4. **Final polish** – if time permits the current global best is
   refined once more with a longer Adam run and a final EG polishing.

All steps respect the global time‑budget (`budget_s` – default 1000 s).
The best vector found is finally evaluated with the official
`evaluate_sequence` and returned.

```python
"""
Improved search for a low‑discrepancy step function.

The algorithm combines wide random sampling of several parametric
families with a long “projected Adam” optimisation on the exact
objective (max of the auto‑convolution).  A smooth approximation of the
max is used during optimisation; the temperature β is increased
gradually so that the optimiser first sees a smooth landscape and
later concentrates on the true maximum.  After Adam a short
exponentiated‑gradient polish is applied.  The whole pipeline respects
the 1000 s time limit and returns the best sequence found.
"""
import time
import numpy as np
from typing import List, Tuple

# ----------------------------------------------------------------------
# Low‑level utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """
    Linear convolution via real FFT (length = len(a)+len(b)-1).
    Both inputs are 1‑D float arrays.
    """
    m = a.shape[0] + b.shape[0] - 1
    n_fft = next_pow2(m)
    A = np.fft.rfft(a, n=n_fft)
    B = np.fft.rfft(b, n=n_fft)
    C = A * B
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


def compute_score(a: np.ndarray) -> float:
    """
    Fast surrogate: the vector is assumed to be normalised (Σa = 1)
    and the score equals 2·n·max(conv(a,a)).
    """
    n = a.shape[0]
    conv = convolution_fft(a, a)
    return 2.0 * n * float(np.max(conv))


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of v onto the probability simplex
    { x ≥ 0, Σx = 1 }  (Duchi et al. 2008).
    """
    if np.all(v <= 0):
        # fallback – uniform distribution
        return np.full_like(v, 1.0 / v.size)

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho_arr = u - cssv / np.arange(1, v.size + 1)
    rho = np.where(rho_arr > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Parametric families
# ----------------------------------------------------------------------


def arcsine_initial(n: int, eps: float = 0.25) -> np.ndarray:
    """Classic arcsine shape (optionally regularised by eps)."""
    i = np.arange(n, dtype=float)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - 0.5 - i) + eps)
    vals /= vals.sum()
    return vals


def generate_beta_shape(n: int, alpha: float, beta: float, eps: float) -> np.ndarray:
    """Beta‑family: (i+ε)^{α‑1}(n‑1‑i+ε)^{β‑1} on {0,…,n‑1}."""
    i = np.arange(n, dtype=float)
    left = (i + eps) ** (alpha - 1.0)
    right = (n - 1 - i + eps) ** (beta - 1.0)
    base = left * right
    total = base.sum()
    if total <= 0.0 or not np.isfinite(total):
        return np.full(n, 1.0 / n)
    return base / total


def generate_power_shape(n: int, p: float, eps: float) -> np.ndarray:
    """Generalised arcsine: a_i ∝ ((i+½)*(n‑½‑i)+ε)^{‑p}."""
    i = np.arange(n, dtype=float)
    denom = ((i + 0.5) * (n - 0.5 - i) + eps) ** p
    vals = 1.0 / denom
    total = vals.sum()
    if total <= 0.0 or not np.isfinite(total):
        return np.full(n, 1.0 / n)
    return vals / total


def generate_triangular_shape(n: int, peak: int) -> np.ndarray:
    """Symmetric triangular bump with apex at position `peak`."""
    left = np.arange(peak + 1, dtype=float)
    right = np.arange(n - peak - 1, dtype=float)[::-1]
    seq = np.concatenate([left, right])
    seq /= seq.sum()
    return seq


def generate_mixture_shape(n: int, weight_arc: float) -> np.ndarray:
    """Mixture (weight_arc)·arcsine + (1‑weight_arc)·uniform."""
    arc = arcsine_initial(n)
    uni = np.full(n, 1.0 / n)
    mix = weight_arc * arc + (1.0 - weight_arc) * uni
    mix /= mix.sum()
    return mix


# ----------------------------------------------------------------------
# Optimisation primitives
# ----------------------------------------------------------------------


def subgradient_mask(a: np.ndarray, beta: float) -> np.ndarray:
    """
    Returns a probability mask w on the convolution index set that
    approximates the soft‑max with temperature β.  The mask sums to 1.
    """
    conv = convolution_fft(a, a)
    max_val = float(np.max(conv))
    # ‑β·(max‑conv) ≥ 0, thus we exponentiate a non‑positive vector.
    d = conv - max_val          # ≤ 0
    w = np.exp(beta * d)
    s = w.sum()
    if s == 0.0:
        # numerically should not happen, fall back to uniform mask
        return np.full_like(w, 1.0 / w.size)
    return w / s


def adam_optimize(initial_a: np.ndarray,
                  deadline: float,
                  max_iter: int,
                  lr: float = 0.08,
                  beta1: float = 0.9,
                  beta2: float = 0.999,
                  eps: float = 1e-8,
                  beta_start: float = 5.0,
                  beta_max: float = 200.0,
                  beta_inc: float = 0.001) -> np.ndarray:
    """
    Projected Adam optimiser on the exact objective.
    The gradient is the (smooth) sub‑gradient of the max‑convolution.
    """
    a = project_to_simplex(initial_a.copy())
    n = a.shape[0]
    L = 2 * n - 1
    m = np.zeros_like(a)
    v = np.zeros_like(a)

    for it in range(1, max_iter + 1):
        if time.time() >= deadline:
            break

        # ----- smooth mask (temperature β grows slowly) -----
        beta = beta_start + beta_inc * it
        if beta > beta_max:
            beta = beta_max
        w = subgradient_mask(a, beta)

        # ----- gradient of smooth max -----
        rev = w[::-1]
        conv_rev = convolution_fft(a, rev)
        grad = 2.0 * conv_rev[L - 1 - np.arange(n)]

        # ----- Adam update -----
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)

        # bias‑corrected estimates (for very large it the bias term
        # is essentially 1, which is fine)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        step = lr * m_hat / (np.sqrt(v_hat) + eps)
        a = a - step
        a = project_to_simplex(a)

        # occasional logging
        if it % 5000 == 0:
            cur = compute_score(a)
            print(f"    Adam it={it:,d} β={beta:.1f} score={cur:.6f}")

    return a


def eg_refine(a: np.ndarray,
              deadline: float,
              max_iter: int = 80_000,
              eta0: float = 0.25) -> np.ndarray:
    """
    Mirror‑descent (exponentiated‑gradient) polishing.
    Keeps the iterate on the simplex automatically.
    """
    a = a.copy()
    n = a.shape[0]
    L = 2 * n - 1

    for it in range(1, max_iter + 1):
        if time.time() >= deadline:
            break

        conv = convolution_fft(a, a)
        max_val = float(np.max(conv))
        # exact mask (all maximisers)
        tol = max_val * 1e-12 + 1e-14
        mask = (np.abs(conv - max_val) <= tol).astype(float)

        rev = mask[::-1]
        conv_rev = convolution_fft(a, rev)
        subgrad = 2.0 * conv_rev[L - 1 - np.arange(n)]

        eta = eta0 / (np.sqrt(it) + 1.0)
        a = a * np.exp(-eta * subgrad)

        total = a.sum()
        if total == 0.0:
            a = np.full_like(a, 1.0 / n)
        else:
            a = a / total

        if it % 5000 == 0:
            cur = compute_score(a)
            print(f"    EG it={it:,d} score={cur:.6f}")

    return a


def hillclimb_power_shape(n: int,
                          init_p: float,
                          init_eps: float,
                          deadline: float,
                          max_iters: int = 2_000) -> Tuple[np.ndarray, float]:
    """Random‑walk hill‑climb on the (p,eps) parameters of the power family."""
    p, eps = init_p, init_eps
    best_seq = generate_power_shape(n, p, eps)
    best_sc = compute_score(best_seq)

    for _ in range(max_iters):
        if time.time() >= deadline:
            break
        p_new = max(0.01, p * (1.0 + np.random.uniform(-0.12, 0.12)))
        eps_new = max(0.0, eps + np.random.uniform(-0.08, 0.08))
        cand = generate_power_shape(n, p_new, eps_new)
        sc = compute_score(cand)
        if sc < best_sc:
            p, eps, best_seq, best_sc = p_new, eps_new, cand, sc
    return best_seq, best_sc


def hillclimb_beta_shape(n: int,
                         init_alpha: float,
                         init_beta: float,
                         init_eps: float,
                         deadline: float,
                         max_iters: int = 2_000) -> Tuple[np.ndarray, float]:
    """Random‑walk hill‑climb on the three parameters of the beta family."""
    alpha, beta, eps = init_alpha, init_beta, init_eps
    best_seq = generate_beta_shape(n, alpha, beta, eps)
    best_sc = compute_score(best_seq)

    for _ in range(max_iters):
        if time.time() >= deadline:
            break
        alpha_new = max(0.01, alpha * (1.0 + np.random.uniform(-0.12, 0.12)))
        beta_new  = max(0.01, beta  * (1.0 + np.random.uniform(-0.12, 0.12)))
        eps_new   = max(0.0, eps + np.random.uniform(-0.08, 0.08))
        cand = generate_beta_shape(n, alpha_new, beta_new, eps_new)
        sc = compute_score(cand)
        if sc < best_sc:
            alpha, beta, eps, best_seq, best_sc = alpha_new, beta_new, eps_new, cand, sc
    return best_seq, best_sc


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42,
                     budget_s: int = 1000,
                     **kwargs) -> List[float]:
    """
    Search for a non‑negative step sequence with a low `evaluate_sequence`
    value.  The routine mixes massive random sampling, local hill‑climbing
    on a few parametric families and a long projected‑Adam refinement.
    The final polish is performed with exponentiated‑gradient updates.
    """
    rng = np.random.default_rng(seed)
    start = time.time()
    deadline = start + budget_s - 1.0          # 1 s safety margin

    best_seq: List[float] = []
    best_val: float = float("inf")

    # ------------------------------------------------------------------
    # 0.  Optional fallback from a previous run
    # ------------------------------------------------------------------
    if "height_sequence_1" in globals():
        try:
            fallback = [float(x) for x in globals()["height_sequence_1"]]
            val = evaluate_sequence(fallback)
            print(f"Fallback (height_sequence_1) score = {val:.6f}")
            if val < best_val:
                best_val, best_seq = val, fallback
        except Exception:
            pass

    # ------------------------------------------------------------------
    # 1.  Wide random sampling of parametric families
    # ------------------------------------------------------------------
    lengths = [256, 512, 1024, 2048, 4096]   # 4096 is still cheap for the fast surrogate
    candidates: List[Tuple[float, np.ndarray]] = []

    for n in lengths:
        if time.time() >= deadline - 5.0:
            break
        print(f"\n=== Sampling length {n} ===")

        # 1a. deterministic starters
        uniform = np.full(n, 1.0 / n)
        candidates.append((compute_score(uniform), uniform.copy()))
        candidates.append((compute_score(arcsine_initial(n)), arcsine_initial(n)))

        # 1b. random β‑family
        for _ in range(300):
            if time.time() >= deadline - 2.0:
                break
            a = rng.uniform(0.1, 5.0)
            b = rng.uniform(0.1, 5.0)
            eps = rng.uniform(0.0, max(1.0, 0.02 * n))
            seq = generate_beta_shape(n, a, b, eps)
            candidates.append((compute_score(seq), seq))

        # 1c. random power‑family (generalised arcsine)
        for _ in range(300):
            if time.time() >= deadline - 2.0:
                break
            p = rng.uniform(0.1, 2.5)
            eps = rng.uniform(0.0, 2.0)
            seq = generate_power_shape(n, p, eps)
            candidates.append((compute_score(seq), seq))

        # 1d. mixtures of arcsine & uniform
        for _ in range(120):
            if time.time() >= deadline - 2.0:
                break
            w = rng.random()
            seq = generate_mixture_shape(n, w)
            candidates.append((compute_score(seq), seq))

        # 1e. simple triangular bumps
        for _ in range(120):
            if time.time() >= deadline - 2.0:
                break
            peak = rng.integers(0, n)
            seq = generate_triangular_shape(n, peak)
            candidates.append((compute_score(seq), seq))

        # 1f. local hill‑climb on the two main families
        cand, sc = hillclimb_power_shape(n, 0.5, 0.0, deadline, max_iters=1500)
        candidates.append((sc, cand))
        cand, sc = hillclimb_beta_shape(n, 0.5, 0.5, 0.0, deadline, max_iters=1500)
        candidates.append((sc, cand))

    # ------------------------------------------------------------------
    # 2.  Keep the best few seeds and enrich them with mixtures
    # ------------------------------------------------------------------
    candidates.sort(key=lambda x: x[0])
    top_k = 12
    top_seeds = [vec for _, vec in candidates[:top_k]]
    print(f"\nSelected {len(top_seeds)} top seeds (fast score range "
          f"{candidates[0][0]:.6f} – {candidates[-1][0]:.6f})")

    # add a few random convex combinations of the top seeds
    for _ in range(30):
        i, j = rng.integers(0, len(top_seeds), size=2)
        w = rng.random()
        combo = w * top_seeds[i] + (1.0 - w) * top_seeds[j]
        combo = project_to_simplex(combo)
        top_seeds.append(combo)

    # ------------------------------------------------------------------
    # 3.  Refinement of each seed (Adam + EG)
    # ------------------------------------------------------------------
    for idx, seed_vec in enumerate(top_seeds, start=1):
        if time.time() >= deadline - 4.0:
            break
        n = seed_vec.shape[0]
        print(f"\nRefining seed {idx}/{len(top_seeds)} (n={n}) …")

        # allocate a reasonable number of iterations – we aim at ~30 M
        # elementary operations per remaining second
        remaining = deadline - time.time()
        max_iter_adam = int(min(200_000, max(20_000, remaining * 25)))

        refined = adam_optimize(seed_vec, deadline,
                                max_iter=max_iter_adam,
                                lr=0.08,
                                beta_start=5.0,
                                beta_max=200.0,
                                beta_inc=0.001)

        # short EG polish
        if time.time() < deadline - 2.0:
            remaining = deadline - time.time()
            max_iter_eg = int(min(80_000, max(5_000, remaining * 25)))
            refined = eg_refine(refined, deadline,
                                max_iter=max_iter_eg,
                                eta0=0.22)

        # official evaluation
        try:
            val = evaluate_sequence(refined.tolist())
            print(f"  → official score = {val:.6f}")
            if val < best_val:
                best_val, best_seq = val, refined.tolist()
                print(f"  *** New global best: {best_val:.6f} ***")
        except Exception as e:
            print(f"  Evaluation error: {e}")

    # ------------------------------------------------------------------
    # 4.  Global polish of the current best (if time permits)
    # ------------------------------------------------------------------
    if best_seq and time.time() < deadline - 6.0:
        print("\nFinal global polish …")
        a = np.array(best_seq, dtype=float)

        remaining = deadline - time.time()
        max_iter_adam = int(min(300_000, max(50_000, remaining * 25)))
        a = adam_optimize(a, deadline,
                          max_iter=max_iter_adam,
                          lr=0.06,
                          beta_start=8.0,
                          beta_max=250.0,
                          beta_inc=0.0008)

        if time.time() < deadline - 2.0:
            remaining = deadline - time.time()
            max_iter_eg = int(min(150_000, max(20_000, remaining * 25)))
            a = eg_refine(a, deadline,
                          max_iter=max_iter_eg,
                          eta0=0.18)

        final_val = evaluate_sequence(a.tolist())
        print(f"Final polished official score = {final_val:.6f}")
        if final_val < best_val:
            best_val, best_seq = final_val, a.tolist()
            print("*** Updated global best after final polish ***")

    # ------------------------------------------------------------------
    # 5.  Guard – return a non‑empty sequence
    # ------------------------------------------------------------------
    if not best_seq:
        # fallback to a simple uniform vector of length 256
        n0 = 256
        best_seq = [1.0 / n0] * n0
        best_val = evaluate_sequence(best_seq)
        print(f"Fallback uniform vector score = {best_val:.6f}")

    print(f"\nSearch finished. Best reported official score = {best_val:.6f}")
    return best_seq
```
