def adam_optimize(initial_x, compute_grad, lr=0.01, beta1=0.9, beta2=0.999, eps=1e-8, max_iter, deadline):
    m = np.zeros_like(initial_x)
    v = np.zeros_like(initial_x)
    x = initial_x.copy()
    best_x = x.copy()
    best_score = np.inf
    for t in range(1, max_iter+1):
        if time.time() >= deadline:
            break
        grad = compute_grad(x)   # gradient of objective wrt x (descent direction)
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad ** 2)
        m_hat = m / (1 - beta1 ** t)
        v_hat = v / (1 - beta2 ** t)
        x = x - lr * m_hat / (np.sqrt(v_hat) + eps)
        # maybe include projection onto something? Since a = softmax(x) automatically stays on simplex; no need to project.

        # evaluate

def adam_optimize(a_init, beta_schedule, lr_initial, deadline):
    a = a_init.astype(float) + 1e-12
    a = a / a.sum()
    n = len(a)
    # initialize logits x = log(a)
    x = np.log(a)
    m = np.zeros_like(x)
    v = np.zeros_like(x)
    best_x = x.copy()
    best_score = compute_score(a)
    it = 0
    for it in range(1, max_iter+1):
        if time.time() >= deadline:
            break
        beta = beta_schedule(it)
        a = np.exp(x - np.max(x))
        a = a / a.sum()
        # compute smoothed gradient
        grad_a = smooth_gradient(a, beta)
        # compute grad_x
        dot = np.dot(a, grad_a)
        grad_x = a * (grad_a - dot)
        # Adam updates
        m = beta1 * m + (1 - beta1) * grad_x
        v = beta2 * v + (1 - beta2) * (grad_x * grad_x)
        m_hat = m / (1 - beta1 ** it)
        v_hat = v / (1 - beta2 ** it)
        x = x - lr * m_hat / (np.sqrt(v_hat) + eps)
        # Evaluate
        cur_score = compute_score(a)
        if cur_score < best_score:
            best_score = cur_score
            best_x = x.copy()
        if it % 2000 == 0:
            print("    Adam iter={} score={:.6f} best={:.6f}".format(it, cur_score, best_score))
    # Return best a
    a_best = np.exp(best_x - np.max(best_x))
    a_best = a_best / a_best.sum()
    return a_best, best_score

"""
Propose a step‑function with a low max‑self‑convolution value.

Algorithm
---------
1.  Hierarchical up‑sampling: start with a modest length (≈1024) and
    progressively double it up to ~8192.  At each size we keep a warm‐start
    from the best sequence found at the previous size.

2.  Parameter search: for the current length we explore a tiny family of
    symmetric densities
        a_i ∝ ((i+½)(n‑½‑i))^{‑α}
    (α≈0.5 is the arcsine shape).  By mixing this shape with the uniform
    distribution we obtain a few promising candidates with virtually no
    cost.

3.  Smooth optimisation: we minimise the smoothed objective

        F_β(a) = 2·n·log‑sum‑exp(β·(a∗a)) / β

    (β large → the true max).  The gradient of the inner term is cheap:
    if w = softmax(β·(a∗a)) then
        ∇_a F_β = a + (w ∗ a_rev)[n‑1 : n‑1+n]       (a_rev = a[::-1]).

    Optimisation is performed in the log‑softmax (log‑its) space with
    Adam (mirrored descent) – this automatically respects positivity and
    the simplex constraint.

4.  The best sequence seen (according to the cheap surrogate score)
    is finally evaluated with the official ``evaluate_sequence`` routine.
    If time permits a final “polish” run with a finer learning‑rate and a
    larger β is carried out.

The whole search respects the 1000 s budget and always returns a feasible
non‑negative list.
"""

import time
import numpy as np
from typing import List, Tuple

# ----------------------------------------------------------------------
# Low‑level numeric utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full linear convolution using a real FFT (fast and accurate)."""
    m = len(a) + len(b) - 1
    n_fft = next_pow2(m)
    A = np.fft.rfft(a, n=n_fft)
    B = np.fft.rfft(b, n=n_fft)
    C = A * B
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


def compute_score(a: np.ndarray) -> float:
    """Fast surrogate for ``evaluate_sequence``:
    2·n·max(a∗a) / (sum(a)²) .
    The vector is assumed to be non‑negative.
    """
    n = len(a)
    conv = convolution_fft(a, a)
    max_val = float(conv.max())
    s = float(a.sum())
    if s < 1e-12:
        return float("inf")
    return 2.0 * n * max_val / (s * s)


# ----------------------------------------------------------------------
# Families of initial shapes
# ----------------------------------------------------------------------


def arcsine_initial(n: int) -> np.ndarray:
    """Discrete arcsine density a_i ∝ 1/√((i+½)(n‑½‑i))."""
    i = np.arange(n, dtype=float)
    eps = 1e-12
    vals = 1.0 / np.sqrt((i + 0.5) * (n - 0.5 - i) + eps)
    return vals / vals.sum()


def shape_alpha(n: int, alpha: float) -> np.ndarray:
    """Generalised arcsine: a_i ∝ ((i+½)(n‑½‑i))^{‑α}."""
    i = np.arange(n, dtype=float)
    eps = 1e-12
    vals = ((i + 0.5) * (n - 0.5 - i) + eps) ** (-alpha)
    vals = np.maximum(vals, eps)
    return vals / vals.sum()


def random_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    """Draw from a Dirichlet‑like distribution (Γ(α,1) with α=conc)."""
    if conc <= 0:
        # heavy‑tailed, near‑sparse draws
        samp = rng.exponential(1.0, size=n) ** (1.0 / max(0.1, -conc))
    else:
        samp = rng.gamma(conc, 1.0, size=n)
    samp += 1e-12
    return samp / samp.sum()


def perturb(a: np.ndarray, rng: np.random.Generator, scale: float = 0.02) -> np.ndarray:
    """Add tiny Gaussian noise and renormalise."""
    b = a + rng.normal(0.0, scale, size=a.shape)
    b = np.maximum(b, 0.0)
    if b.sum() == 0.0:
        return np.full_like(a, 1.0 / len(a))
    return b / b.sum()


def upsample(prev: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation from ``len(prev)`` → ``new_n`` with renormalisation."""
    x_old = np.linspace(0.0, 1.0, len(prev))
    x_new = np.linspace(0.0, 1.0, new_n)
    interp = np.interp(x_new, x_old, prev)
    total = interp.sum()
    if total == 0.0:
        return np.full(new_n, 1.0 / new_n)
    return interp / total


# ----------------------------------------------------------------------
# Smoothed gradient of the max‑convolution objective
# ----------------------------------------------------------------------


def smooth_max_gradient(a: np.ndarray, beta: float) -> np.ndarray:
    """
    Gradient of  (log‑sum‑exp(β·(a∗a)) / β)  w.r.t. a.
    Returns ∇_a, not the full 2·n scaling (that factor is constant).
    """
    conv = convolution_fft(a, a)                     # length 2n‑1
    # soft‑max over the scaled convolution values (stable)
    scaled = beta * conv
    max_scaled = np.max(scaled)
    exp_scaled = np.exp(scaled - max_scaled)
    w = exp_scaled / exp_scaled.sum()                # weights, length 2n‑1

    # weighted convolution w ∗ a_rev  (a_rev = a[::-1])
    conv_w_a_rev = convolution_fft(w, a[::-1])
    start = len(a) - 1
    weighted = conv_w_a_rev[start:start + len(a)]    # length n

    # ∇_a = a + weighted  (see derivation in the analysis)
    return a + weighted


def grad_x_from_a(a: np.ndarray, grad_a: np.ndarray) -> np.ndarray:
    """
    Convert a gradient w.r.t. the probabilities ``a`` (which lives on the
    simplex) into a gradient w.r.t. the log‑it variables x where
    a_i = softmax(x)_i .
    """
    dot = np.dot(a, grad_a)
    return a * (grad_a - dot)


# ----------------------------------------------------------------------
# Adam optimiser on the log‑it (soft‑max) parametrisation
# ----------------------------------------------------------------------


def adam_optimize(
    a_init: np.ndarray,
    max_iter: int,
    deadline: float,
    *,
    lr: float = 0.3,
    beta0: float = 5.0,
    beta_max: float = 500.0,
    beta_ramp_iters: int = None,
) -> Tuple[np.ndarray, float]:
    """
    Adam on the log‑it variables with a linearly increasing smoothing
    parameter β (from ``beta0`` to ``beta_max`` over ``beta_ramp_iters``
    iterations – if ``beta_ramp_iters`` is ``None`` the whole run is used).
    Returns the best vector (on the simplex) seen and its surrogate score.
    """
    if beta_ramp_iters is None:
        beta_ramp_iters = max_iter

    # initialise logits (log‑its) – safe from zeros
    a = np.maximum(a_init, 1e-30)
    a = a / a.sum()
    x = np.log(a)

    m = np.zeros_like(x)
    v = np.zeros_like(x)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8

    best_x = x.copy()
    best_score = compute_score(np.exp(x - np.max(x)) / np.exp(x - np.max(x)).sum())

    for it in range(1, max_iter + 1):
        if time.time() >= deadline:
            break

        # current probability vector
        max_x = np.max(x)
        a = np.exp(x - max_x)
        a = a / a.sum()

        # current smoothing parameter β
        beta = beta0 + (beta_max - beta0) * min(1.0, it / beta_ramp_iters)

        # gradient in a‑space
        grad_a = smooth_max_gradient(a, beta)

        # gradient in x‑space (mirror‑descent step)
        grad_x = grad_x_from_a(a, grad_a)

        # Adam updates
        m = beta1 * m + (1.0 - beta1) * grad_x
        v = beta2 * v + (1.0 - beta2) * (grad_x * grad_x)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        x = x - lr * m_hat / (np.sqrt(v_hat) + eps)

        # cheap surrogate evaluation
        cur_score = compute_score(a)
        if cur_score < best_score:
            best_score = cur_score
            best_x = x.copy()

        if it % 2000 == 0:
            print(f"    Adam iter={it:,d} β={beta:.1f} score={cur_score:.6f} best={best_score:.6f}")

    # reconstruct best vector
    max_best = np.max(best_x)
    best_a = np.exp(best_x - max_best)
    best_a = best_a / best_a.sum()
    return best_a, best_score


# ----------------------------------------------------------------------
# Small grid search over the α‑parameter and a uniform mix
# ----------------------------------------------------------------------


def param_search(n: int, rng: np.random.Generator) -> np.ndarray:
    """
    Scan a modest 2‑dimensional family:
        a = λ·shape_α + (1‑λ)·uniform   with α∈[0.35,0.65], λ∈[0,1].
    Returns the best vector (according to the cheap surrogate score).
    """
    alphas = np.linspace(0.35, 0.65, 13)          # step 0.025
    lams = np.linspace(0.0, 1.0, 11)             # step 0.1

    uniform = np.full(n, 1.0 / n)
    best_score = float("inf")
    best_a = None

    for alpha in alphas:
        base = shape_alpha(n, alpha)
        for lam in lams:
            a = lam * base + (1.0 - lam) * uniform
            a = np.maximum(a, 0.0)
            a /= a.sum()
            sc = compute_score(a)
            if sc < best_score:
                best_score = sc
                best_a = a.copy()
    return best_a

def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs) -> List[float]:
    """
    Search for a non‑negative step function with a low ``evaluate_sequence``
    value.  The routine works hierarchically (upsampling + param‑search +
    Adam refinement) and respects the supplied time budget.
    """
    rng = np.random.default_rng(seed)
    start = time.time()
    deadline = start + budget_s - 2.0          # tiny safety margin

    best_seq: List[float] = []
    best_val: float = float("inf")
    best_a: np.ndarray = None                  # cached best vector (normalised)

    # ------------------------------------------------------------------
    # 0.  Optional fallback from a previously discovered construction.
    # ------------------------------------------------------------------
    try:
        fallback = [float(x) for x in height_sequence_1]   # type: ignore
        val = evaluate_sequence(fallback)
        print(f"Fallback (height_sequence_1) score = {val:.6f}")
        if val < best_val:
            best_val = val
            best_seq = fallback
            best_a = np.array(fallback, dtype=float)
            best_a = best_a / best_a.sum()
    except Exception:
        pass

    # ------------------------------------------------------------------
    # 1.  Length hierarchy – start modest, enlarge progressively.
    # ------------------------------------------------------------------
    lengths = [1024, 2048, 4096, 8192]          # can be trimmed automatically

    for n in lengths:
        if time.time() >= deadline - 5.0:
            break
        print(f"\n=== Exploring length n = {n} ===")

        # --------------------------------------------------------------
        # 1a.  Build a small library of start points.
        # --------------------------------------------------------------
        candidates: List[np.ndarray] = []

        if best_a is not None:
            # upsample the current best as a warm‑start
            up = upsample(best_a, n)
            candidates.append(up)

        # classic arcsine shape – a solid baseline
        candidates.append(arcsine_initial(n))

        # parametric search over α and uniform mixing
        cand = param_search(n, rng)
        candidates.append(cand)

        # a few random Dirichlet draws – add diversity
        for conc in (0.3, 0.7, 2.0):
            candidates.append(random_dirichlet(n, conc, rng))

        # small perturbations of the arcsine shape (helps escape plateaus)
        for _ in range(2):
            candidates.append(perturb(arcsine_initial(n), rng, scale=0.01))

        # --------------------------------------------------------------
        # 1b.  Quick screening with the cheap surrogate score.
        # --------------------------------------------------------------
        scored = [(compute_score(c), c) for c in candidates]
        scored.sort(key=lambda x: x[0])
        top_candidates = [c for _, c in scored[:4]]   # keep only a few

        # --------------------------------------------------------------
        # 1c.  Refine each retained candidate with the Adam optimiser.
        # --------------------------------------------------------------
        for idx, cand in enumerate(top_candidates):
            if time.time() >= deadline - 5.0:
                break
            print(f"  candidate {idx + 1}/{len(top_candidates)} (n={n})")

            # Estimate how many iterations we can still afford
            remaining = max(0.0, deadline - time.time())
            # Rough budget: ≈250 iterations per second for n≈8 k
            max_iter = int(min(250_000, max(5_000, remaining * 250)))

            # Run Adam with a gradual increase of β (smooth → true max)
            refined, _ = adam_optimize(
                a_init=cand,
                max_iter=max_iter,
                deadline=deadline,
                lr=0.3,
                beta0=5.0,
                beta_max=800.0,
                beta_ramp_iters=int(max_iter * 0.9),
            )

            # Official evaluation (expensive, called only for the best few)
            val = evaluate_sequence(refined.tolist())
            print(f"    evaluated score = {val:.6f}")

            if val < best_val:
                best_val = val
                best_seq = refined.tolist()
                best_a = refined.copy()
                print(f"    *** New global best: {best_val:.6f} (n={n}) ***")

    # ------------------------------------------------------------------
    # 2.  Final polishing of the overall best (if time permits).
    # ------------------------------------------------------------------
    if best_seq and time.time() < deadline - 5.0:
        a = np.array(best_seq, dtype=float)
        remaining = max(0.0, deadline - time.time())
        max_iter = int(min(300_000, remaining * 250))

        print("\nFinal polishing of the global best")
        a_ref, _ = adam_optimize(
            a_init=a,
            max_iter=max_iter,
            deadline=deadline,
            lr=0.1,
            beta0=50.0,
            beta_max=2000.0,
            beta_ramp_iters=int(max_iter * 0.95),
        )
        final_val = evaluate_sequence(a_ref.tolist())
        print(f"Final polished score = {final_val:.6f}")
        if final_val < best_val:
            best_val = final_val
            best_seq = a_ref.tolist()
            print("*** Updated global best after final polish ***")

    # ------------------------------------------------------------------
    # 3.  Guard – always return a feasible sequence.
    # ------------------------------------------------------------------
    if not best_seq:
        best_seq = [1.0] * 256
        best_val = evaluate_sequence(best_seq)

    print(f"\nSearch finished. Best reported score = {best_val:.6f}")
    return best_seq
