def adam_step(a, grad, m, v, t, lr):
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    m = beta1 * m + (1 - beta1) * grad

import time, numpy as np, math
from typing import List

def next_pow2(x):
    return 1 << (x - 1).bit_length()

def convolution_fft(a, b):
    m = len(a) + len(b) - 1
    n_fft = next_pow2(m)
    A = np.fft.rfft(a, n=n_fft)
    B = np.fft.rfft(b, n=n_fft)
    C = A * B
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv

def compute_score(a):
    n = len(a)
    conv = convolution_fft(a, a)
    return 2.0 * n * conv.max() / (a.sum()**2)

def project_simplex(v):
    """ Euclidean projection of v onto the probability simplex {x >=0, sum x = 1}. """
    n = len(v)
    if v.sum() == 1 and np.all(v >= 0):
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all non-positive, return uniform distribution
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w

def arcsine_initial_p(n, p=0.5):
    i = np.arange(n, dtype=float)
    eps = 1e-12
    vals = 1.0 / ((i + 0.5) * (n - 0.5 - i) + eps) ** p
    vals = vals / vals.sum()
    return vals

def random_dirichlet(n, conc, rng):
    if conc <= 0:
        # heavy-tailed
        samp = rng.exponential(1.0, size=n) ** (1.0 / max(0.1, -conc))
    else:
        samp = rng.gamma(conc, 1.0, size=n)
    samp = np.maximum(samp, 1e-12)
    return samp / samp.sum()

def upsample(prev, new_n):
    x_old = np.linspace(0.0, 1.0, len(prev))
    x_new = np.linspace(0.0, 1.0, new_n)
    interp = np.interp(x_new, x_old, prev)
    total = interp.sum()
    if total == 0:
        return np.full(new_n, 1.0 / new_n)
    return interp / total

    len_w = len(w)
    N = next_pow2(len_w + n - 1)
    # correlation using FFT: corr = ifft( conj(FFT(w)) * FFT(a) )
    F_w = np.fft.rfft(w, n=N)
    F_a = np.fft.rfft(a, n=N)
    corr_full = np.fft.irfft(np.conj(F_w) * F_a, n=N)
    # Take first n entries (lags 0..n-1)
    grad = 2.0 * corr_full[:n]
    return smooth_val, grad

def adam_update(a, grad, m, v, t, lr):
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    m = beta1 * m + (1 - beta1) * grad
    v = beta2 * v + (1 - beta2) * (grad * grad)
    m_hat = m / (1 - beta1**t)
    v_hat = v / (1 - beta2**t)
    step = lr * m_hat / (np.sqrt(v_hat) + eps)
    a_new = a - step
    a_proj = project_simplex(a_new)
    return a_proj, m, v

def optimise_smooth(a0, deadline, max_iter=100_000, c_start=5.0, c_end=200.0, lr0=0.05):
    a = a0.copy()
    n = len(a)
    best_a = a.copy()
    best_score = compute_score(a)
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    t = 0
    c = c_start
    # schedule: increase c after each block of steps
    steps_per_c = max(2000, n // 4)  # at least a few thousand per c
    while time.time() < deadline and t < max_iter:
        if t % steps_per_c == 0 and c < c_end:
            c = min(c * 2.0, c_end)
        smooth_val, grad = smooth_max_and_grad(a, c)
        t += 1
        lr = lr0 / np.sqrt(t)
        a, m, v = adam_update(a, grad, m, v, t, lr)
        # Periodically evaluate cheap surrogate
        if t % 2000 == 0:
            cur_score = compute_score(a)
            if cur_score < best_score:
                best_score = cur_score
                best_a = a.copy()
                print(f"        iter={t:,d} c={c:.1f} cur_score={cur_score:.6f} best={best_score:.6f}")
    # final check
    cur_score = compute_score(a)
    if cur_score < best_score:
        best_score = cur_score
        best_a = a.copy()
    return best_a, best_score

"""
Hybrid smoothed‑max Adam optimiser for the max‑self‑convolution problem.

The algorithm builds a hierarchy of lengths (1024 → … → 16384),
starts from a discretised arcsine shape (the known analytic solution)
and a few random Dirichlet draws, then refines each candidate with a
log‑sum‑exp (smooth max) surrogate.  The surrogate is differentiable;
its gradient is obtained by a single FFT‑based correlation, which makes
each iteration O(n log n).  Adam with a decaying learning‑rate together
with a Euclidean projection onto the probability simplex drives the
shape towards a flatter self‑convolution.  The best vector found is
finally polished with a tighter schedule and evaluated with the
official ``evaluate_sequence`` function.

The routine respects a 1000 s wall‑clock budget and always returns a
feasible non‑negative list.
"""

import time
import numpy as np
from typing import List, Tuple

# ----------------------------------------------------------------------
# Basic numeric helpers
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full linear convolution using a real FFT (fast and accurate)."""
    m = len(a) + len(b) - 1
    n_fft = next_pow2(m)
    A = np.fft.rfft(a, n=n_fft)
    B = np.fft.rfft(b, n=n_fft)
    C = A * B
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


def compute_score(a: np.ndarray) -> float:
    """
    Fast surrogate for the official evaluator:
        2·n·max(a*a) / (sum(a)²)
    """
    n = len(a)
    conv = convolution_fft(a, a)
    max_val = float(conv.max())
    s = float(a.sum())
    if s < 1e-12:
        return float("inf")
    return 2.0 * n * max_val / (s * s)


# ----------------------------------------------------------------------
# Projection onto the simplex
# ----------------------------------------------------------------------


def project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex {x≥0, Σx=1}."""
    n = v.shape[0]
    # Fast path
    if np.all(v >= 0) and np.isclose(v.sum(), 1.0):
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all non‑positive – return the uniform distribution
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Families of initial shapes
# ----------------------------------------------------------------------


def arcsine_initial_p(n: int, p: float = 0.5) -> np.ndarray:
    """Discrete arcsine family a_i ∝ ((i+½)(n‑½‑i))^{-p}."""
    i = np.arange(n, dtype=float)
    eps = 1e-12
    vals = 1.0 / ((i + 0.5) * (n - 0.5 - i) + eps) ** p
    vals /= vals.sum()
    return vals


def random_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    """Dirichlet‑like draw (Γ(α,1) with α=conc)."""
    if conc <= 0:
        # heavy‑tailed draws
        samp = rng.exponential(1.0, size=n) ** (1.0 / max(0.1, -conc))
    else:
        samp = rng.gamma(conc, 1.0, size=n)
    samp = np.maximum(samp, 1e-12)
    return samp / samp.sum()


def upsample(prev: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation from len(prev) → new_n and renormalisation."""
    x_old = np.linspace(0.0, 1.0, len(prev))
    x_new = np.linspace(0.0, 1.0, new_n)
    interp = np.interp(x_new, x_old, prev)
    total = interp.sum()
    if total == 0.0:
        return np.full(new_n, 1.0 / new_n)
    return interp / total


# ----------------------------------------------------------------------
# Smoothed‑max surrogate and its gradient
# ----------------------------------------------------------------------


def smooth_max_and_grad(a: np.ndarray, c: float) -> Tuple[float, np.ndarray]:
    """
    Returns the smoothed max value
        f_c(a) = (1/c)·log Σ_k exp(c·(a∗a)_k)
    and its gradient w.r.t. a (on the simplex).

    The gradient is 2·(w ⋆ a) where w_k ∝ exp(c·(a∗a)_k) and
    ⋆ denotes the (non‑circular) correlation.
    """
    n = len(a)

    # 1) self‑convolution
    conv = convolution_fft(a, a)          # length 2n‑1
    max_conv = conv.max()

    # 2) smooth max (log‑sum‑exp) – numerically stable
    shifted = conv - max_conv
    exp_terms = np.exp(c * shifted)
    sum_exp = exp_terms.sum()
    smooth_val = max_conv + (1.0 / c) * np.log(sum_exp)

    # 3) normalised weights for the gradient
    w = exp_terms / sum_exp               # length 2n‑1, sums to 1

    # 4) gradient = 2·correlation(w, a) (valid mode)
    len_w = len(w)
    N = next_pow2(len_w + n - 1)          # zero‑pad for linear correlation
    F_w = np.fft.rfft(w, n=N)
    F_a = np.fft.rfft(a, n=N)
    # correlation via ifft(conj(F_w) * F_a)
    corr_full = np.fft.irfft(np.conj(F_w) * F_a, n=N)
    # first n entries correspond to lags 0 … n‑1
    grad = 2.0 * corr_full[:n]

    return smooth_val, grad


# ----------------------------------------------------------------------
# Adam optimiser on the simplex
# ----------------------------------------------------------------------


def adam_update(
    a: np.ndarray,
    grad: np.ndarray,
    m: np.ndarray,
    v: np.ndarray,
    t: int,
    lr: float,
) -> Tuple[np.ndarray, np.ndarray, np.ndarray]:
    """Adam step followed by projection onto the simplex."""
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8

    m = beta1 * m + (1.0 - beta1) * grad
    v = beta2 * v + (1.0 - beta2) * (grad * grad)

    m_hat = m / (1.0 - beta1 ** t)
    v_hat = v / (1.0 - beta2 ** t)

    step = lr * m_hat / (np.sqrt(v_hat) + eps)
    a_new = a - step
    a_proj = project_simplex(a_new)
    return a_proj, m, v


# ----------------------------------------------------------------------
# Main optimisation driver (smooth‑max + Adam)
# ----------------------------------------------------------------------


def optimise_smooth(
    a0: np.ndarray,
    deadline: float,
    *,
    max_iter: int = 100_000,
    c_start: float = 5.0,
    c_end: float = 200.0,
    lr0: float = 0.05,
) -> Tuple[np.ndarray, float]:
    """
    Adam optimiser on the log‑sum‑exp surrogate.
    Returns the best vector seen (according to the cheap surrogate
    ``compute_score``) and its surrogate value.
    """
    a = a0.copy()
    n = len(a)

    best_a = a.copy()
    best_sc = compute_score(a)

    m = np.zeros_like(a)
    v = np.zeros_like(a)

    t = 0
    c = c_start
    # increase c after a modest block of iterations
    steps_per_c = max(2000, n // 4)

    while time.time() < deadline and t < max_iter:
        if t % steps_per_c == 0 and c < c_end:
            c = min(c * 2.0, c_end)

        _, grad = smooth_max_and_grad(a, c)

        t += 1
        lr = lr0 / np.sqrt(t)

        a, m, v = adam_update(a, grad, m, v, t, lr)

        # occasional inexpensive surrogate check
        if t % 2000 == 0:
            cur_sc = compute_score(a)
            if cur_sc < best_sc:
                best_sc = cur_sc
                best_a = a.copy()
                print(
                    f"        iter={t:,d} c={c:.1f} cur_sc={cur_sc:.6f} best_sc={best_sc:.6f}"
                )

    # final cheap check
    cur_sc = compute_score(a)
    if cur_sc < best_sc:
        best_sc = cur_sc
        best_a = a.copy()

    return best_a, best_sc

    # 1.  Length hierarchy – start small, enlarge progressively.
    # ------------------------------------------------------------------
    lengths = [1024, 2048, 4096, 8192, 16384]

    for n in lengths:
        if time.time() >= deadline - 5.0:
            break
        print(f"\n=== Length {n} ===")

        # --------------------------------------------------------------
        # 1a. Build a modest library of start points.
        # --------------------------------------------------------------
        starts: List[np.ndarray] = []

        if best_vec is not None:
            starts.append(upsample(best_vec, n))

        # arcsine family with a few exponents
        for p in (0.45, 0.5, 0.55):
            starts.append(arcsine_initial_p(n, p))

        # random Dirichlet draws (different concentrations)
        for conc in (0.3, 0.7, 1.5):
            starts.append(random_dirichlet(n, conc, rng))

        # small perturbations of the canonical arcsine
        base_arc = arcsine_initial_p(n, 0.5)
        for _ in range(2):
            pert = base_arc + rng.normal(0.0, 0.02, size=n)
            pert = np.maximum(pert, 0.0)
            pert = pert / pert.sum()
            starts.append(pert)

        # --------------------------------------------------------------
        # 1b. Quick screening with the cheap surrogate.
        # --------------------------------------------------------------
        scored = [(compute_score(s), s) for s in starts]
        scored.sort(key=lambda x: x[0])
        top_candidates = [s for _, s in scored[:3]]  # keep three most promising

        # --------------------------------------------------------------
        # 1c. Refine each retained candidate with smoothed‑max Adam.
        # --------------------------------------------------------------
        for idx, cand in enumerate(top_candidates):
            if time.time() >= deadline - 5.0:
                break
            print(f"  candidate {idx + 1}/{len(top_candidates)} (n={n})")

            remaining = max(0.0, deadline - time.time())
            # budget for this candidate (≈ 300 iterations / s for n≈8 k)
            max_iters = int(min(200_000, max(5_000, remaining * 300)))

            refined, _ = optimise_smooth(
                a0=cand,
                deadline=deadline,
                max_iter=max_iters,
                c_start=5.0,
                c_end=200.0,
                lr0=0.05,
            )

            val = evaluate_sequence(refined.tolist())
            print(f"    evaluated score = {val:.6f}")

            if val < best_val:
                best_val = val
                best_seq = refined.tolist()
                best_vec = refined.copy()
                print(f"    *** New global best: {best_val:.6f} (n={n}) ***")

    # ------------------------------------------------------------------
    # 2.  Final polishing of the overall best (time permitting).
    # ------------------------------------------------------------------
    if best_vec is not None and time.time() < deadline - 5.0:
        print("\nFinal polishing phase")
        remaining = max(0.0, deadline - time.time())
        max_iters = int(min(300_000, remaining * 300))

        polished, _ = optimise_smooth(
            a0=best_vec,
            deadline=deadline,
            max_iter=max_iters,
            c_start=20.0,
            c_end=400.0,
            lr0=0.03,
        )
        final_val = evaluate_sequence(polished.tolist())
        print(f"Final polished score = {final_val:.6f}")
        if final_val < best_val:
            best_val = final_val
            best_seq = polished.tolist()
            print("*** Updated global best after final polish ***")

    # ------------------------------------------------------------------
    # 3.  Guard – always return a feasible sequence.
    # ------------------------------------------------------------------
    if not best_seq:
        best_seq = [1.0] * 256
        best_val = evaluate_sequence(best_seq)

    print(f"\nSearch finished. Best reported score = {best_val:.6f}")
    return best_seq
