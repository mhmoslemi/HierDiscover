import scipy.sparse as sp

def make_conv_matrix(f):
    n = len(f)
    rows = []
    cols = []
    data = []
    for k in range(2*n-1):
        i_start = max(0, k - (n-1))
        i_end = min(k, n-1)
        # row indices for b indices j from i_start..i_end; coefficient f_{k - j}
        for j in range(i_start, i_end+1):
            rows.append(k)
            cols.append(j)
            data.append(f[k - j])
    A = sp.csr_matrix((data, (rows, cols)), shape=(2*n-1, n))
    return A

The driver calls `propose_candidate()` and expects a list of non‑negative numbers that (after the internal sanitising steps) minimises  

```
    2·n·max(a∗a) / (sum(a)²)
```

The search consists of four independent stages that together explore the space far more aggressively than the reference implementation:

1. **Broad random seeding** – a few thousand vectors are generated from cheap, expressive families  
   (uniform, arcsine, power‑law, Beta‑like, Dirichlet and random mixtures of a few Beta components).  
   The cheap “fast score’’ (`compute_score`) is used to keep only the best few dozen seeds.

2. **Local stochastic improvement** – each retained seed is refined by a lightweight simulated‑annealing
   that moves a small amount of mass between random bins while staying on the probability simplex.

3. **Smooth‑max gradient descent** – the non‑smooth `max` is replaced by a log‑sum‑exp surrogate.  
   An exponentiated‑gradient update together with a schedule of increasingly sharp β‑values drives the
   vector towards a (near) minimiser of the true objective.

4. **Linear‑programming boost** – given the current vector `f`, we solve the linear program  

```
    max   Σ b_i
    s.t.  (f * b)_k ≤ M   ∀k          (M = max_k (f*f)_k)
          b_i ≥ 0
```

   The solution `b` concentrates the mass where `f` is small, therefore after normalising `b`
   the maximal autocorrelation drops.  The LP is built with a sparse Toeplitz matrix and solved
   by SciPy’s HiGHS interior‑point solver.  One or two LP steps are performed per candidate.

5. **Final polishing** – the best vector found so far is polished with an exponentiated
   sub‑gradient method that works directly on the true `max`‑convolution.  The iteration count
   is chosen automatically from the remaining wall‑clock budget.

All steps respect the 1000 s budget, constantly checking the deadline and aborting as soon as
the time limit is reached.  If nothing is found a simple uniform vector is returned, guaranteeing
a feasible solution.

1.  Randomly generate thousands of candidate vectors from several expressive families

    The fast “scale‑invariant’’ score 2·n·max(a∗a)/(sum(a)²) is used to keep the few best
    seeds.

    (β = 10, 20, 40, 80, 160, 320) drives the vector towards a (near) minimiser of the true

    derived from the remaining wall‑clock budget.

def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full linear convolution using a real FFT (zero‑padded to a power of two)."""
    m = len(a) + len(b) - 1
    n_fft = next_pow2(m)
    A = np.fft.rfft(a, n=n_fft)
    B = np.fft.rfft(b, n=n_fft)
    C = A * B
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


def compute_score(a: np.ndarray) -> float:
    """
    Fast scale‑invariant objective:
        2·n·max(a∗a) / (sum(a)²)
    The vector a is assumed non‑negative.
    """
    n = a.shape[0]
    conv = convolution_fft(a, a)
    max_val = float(np.max(conv))
    s = float(np.sum(a))
    if s < 1e-12:
        return float("inf")
    return 2.0 * n * max_val / (s * s)

def arcsine_initial(n: int) -> np.ndarray:
    """Discrete arcsine density (normalised to sum = 1)."""
    i = np.arange(n, dtype=float)
    eps = 0.25
    vals = 1.0 / np.sqrt((i + 0.5) * (n - 0.5 - i) + eps)
    return vals / vals.sum()


def generate_powerlaw(n: int, p: float, eps: float) -> np.ndarray:
    """Symmetric power‑law a_i ∝ ((i+ε)(n‑1‑i+ε))⁻ᵖ."""
    i = np.arange(n, dtype=float)
    base = ((i + eps) * (n - 1 - i + eps)) ** (-p)
    total = base.sum()
    if total == 0.0:
        return np.full(n, 1.0 / n)
    return base / total


def generate_beta_like(n: int, alpha: float, beta: float) -> np.ndarray:
    """Beta‑like shape a_i ∝ (i+½)^{α‑1} (n‑1‑i+½)^{β‑1}."""
    i = np.arange(n, dtype=float)
    base = (i + 0.5) ** (alpha - 1) * (n - 1 - i + 0.5) ** (beta - 1)
    total = base.sum()
    if total == 0.0:
        return np.full(n, 1.0 / n)
    return base / total


def random_dirichlet(n: int, concentration: float, rng: np.random.Generator) -> np.ndarray:
    """Sample a Dirichlet‑like vector (Gamma draws) and normalise."""
    if concentration <= 0:
        samp = rng.exponential(1.0, size=n) ** (1.0 / max(0.1, concentration))
    else:
        samp = rng.gamma(concentration, 1.0, size=n)
    samp += 1e-12
    return samp / samp.sum()


def random_beta_mixture(n: int, rng: np.random.Generator) -> np.ndarray:
    """
    Random mixture of 2–3 Beta‑like components.
    Parameters of each component are drawn uniformly from a wide range.
    """
    ncomp = rng.integers(2, 4)          # 2 or 3 components
    alphas = rng.uniform(0.1, 6.0, size=ncomp)
    betas = rng.uniform(0.1, 6.0, size=ncomp)
    weights = rng.dirichlet(np.ones(ncomp))
    mix = np.zeros(n, dtype=float)
    for w, a, b in zip(weights, alphas, betas):
        mix += w * generate_beta_like(n, a, b)
    mix = mix / mix.sum()
    return mix


def perturb(a: np.ndarray, rng: np.random.Generator, scale: float = 0.02) -> np.ndarray:
    """Add a tiny Gaussian perturbation and renormalise onto the simplex."""
    b = a + rng.normal(0.0, scale, size=a.shape)
    b = np.maximum(b, 0.0)
    s = b.sum()
    if s == 0.0:
        return np.full_like(a, 1.0 / a.shape[0])
    return b / s

def propose_neighbor(a: np.ndarray, rng: np.random.Generator, max_step: float = 0.02) -> np.ndarray:
    """Move a small amount of mass from one random bin to another."""
    n = a.shape[0]
    i, j = rng.integers(0, n, size=2)
    if i == j or a[i] == 0.0:
        return a
    delta = a[i] * rng.random() * max_step
    b = a.copy()
    b[i] -= delta
    b[j] += delta
    return b


def simulated_annealing(a0: np.ndarray,
                        deadline: float,
                        rng: np.random.Generator,
                        max_iter: int = 2000,
                        start_temp: float = 0.005,
                        cooling: float = 0.9995,
                        max_step: float = 0.02) -> np.ndarray:
    """
    Very light SA that respects the simplex.
    Returns the best vector seen (its score is obtained with compute_score).
    """
    a = a0.copy()
    cur_sc = compute_score(a)
    best_a = a.copy()
    best_sc = cur_sc
    temp = start_temp

    for it in range(max_iter):
        if time.time() >= deadline:
            break
        cand = propose_neighbor(a, rng, max_step)
        cand_sc = compute_score(cand)
        delta = cand_sc - cur_sc
        if delta < 0.0 or rng.random() < math.exp(-delta / max(1e-12, temp)):
            a = cand
            cur_sc = cand_sc
            if cur_sc < best_sc:
                best_sc = cur_sc
                best_a = a.copy()
        temp *= cooling
    return best_a

# Smooth‑max gradient descent (log‑sum‑exp surrogate)
# ----------------------------------------------------------------------


def softmax_gradient_descent(initial_a: np.ndarray,
                             deadline: float,
                             betas: list[float],
                             iter_per_beta: int = 500,
                             eta0: float = 0.5) -> np.ndarray:
    """
    Minimise Φ_β(a) = (1/β)·log Σ_k exp(β·(a∗a)_k)
    using an exponentiated‑gradient update and a simplex projection.
    """
    a = np.maximum(initial_a, 1e-30)
    a = a / a.sum()
    best_a = a.copy()
    best_sc = compute_score(a)

    n = a.shape[0]

    for beta in betas:
        for it in range(1, iter_per_beta + 1):
            if time.time() >= deadline:
                return best_a
            # convolution a∗a
            conv = convolution_fft(a, a)            # length 2n‑1
            # soft‑max weights (stable)
            max_c = conv.max()
            exp_vals = np.exp(beta * (conv - max_c))
            p = exp_vals / exp_vals.sum()           # Σ p = 1

            # cross‑correlation p ⋆ a  (linear, length n)
            N = next_pow2(len(p) + n - 1)
            P = np.fft.fft(p, N)
            A = np.fft.fft(a, N)
            cross = np.fft.ifft(P * np.conj(A)).real[:n]
            cross = np.maximum(cross, 0.0)

            # gradient of the smooth surrogate (non‑negative)
            grad = 4.0 * n * cross

            # exponentiated‑gradient step
            eta = eta0 / np.sqrt(it)
            a = a * np.exp(-eta * grad)

            # projection onto the simplex (renormalisation)
            s = a.sum()
            if s == 0.0:
                a = np.full_like(a, 1.0 / n)
            else:
                a = a / s

            cur_sc = compute_score(a)
            if cur_sc < best_sc:
                best_sc = cur_sc
                best_a = a.copy()
    return best_a


# ----------------------------------------------------------------------
# Linear‑programming boost (f‑fixed, maximise Σb_i under (f*b)≤M)
# ----------------------------------------------------------------------


def build_conv_matrix(f: np.ndarray) -> sp.csr_matrix:
    """
    Build a sparse Toeplitz matrix A such that A @ b = f * b (linear convolution).
    Shape of A: (2·n‑1) × n
    """
    n = f.shape[0]
    rows = []
    cols = []
    data = []
    # Each output index k (0 … 2n‑2) receives contributions from b_j with coefficient f_{k‑j}
    for k in range(2 * n - 1):
        i_start = max(0, k - (n - 1))
        i_end = min(k, n - 1)
        # j runs from i_start … i_end ; coefficient = f_{k‑j}
        for j in range(i_start, i_end + 1):
            rows.append(k)
            cols.append(j)
            data.append(f[k - j])
    return sp.csr_matrix((data, (rows, cols)), shape=(2 * n - 1, n))


def lp_improve(f: np.ndarray, deadline: float) -> np.ndarray | None:
    """
    Perform one LP step:
        max Σ b_i   s.t. (f * b)_k ≤ M  ∀k,   b_i ≥ 0
    where M = max_k (f * f)_k.
    The solution b is returned after normalising to sum = 1.
    Returns None if time is exhausted or LP fails.
    """
    if time.time() >= deadline:
        return None

    n = f.shape[0]
    M = float(np.max(convolution_fft(f, f)))
    A = build_conv_matrix(f)

    # Inequality: A @ b ≤ M
    b_ub = np.full(A.shape[0], M, dtype=float)

    # Linear objective: maximise Σb_i  ⇒ minimise – Σb_i
    c = -np.ones(n, dtype=float)

    # Bounds for each variable: b_i ≥ 0
    bounds = [(0, None)] * n

    # Solve with HiGHS (fast interior‑point) – accepts sparse matrices
    try:
        res = opt.linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method="highs", options={"presolve": True})
    except Exception:
        return None

    if not res.success:
        return None

    b = np.maximum(res.x, 0.0)
    total = b.sum()
    if total == 0.0:
        return None
    b = b / total
    return b


# ----------------------------------------------------------------------
# Exponentiated sub‑gradient on the true max‑convolution (final polishing)
# ----------------------------------------------------------------------


def _max_conv_gradient(a: np.ndarray, conv: np.ndarray) -> np.ndarray:
    """
    Sub‑gradient of  max_k (a∗a)_k.
    For each k attaining the maximum, the contribution is 2·a_{k‑j}.
    The contributions are averaged if several k are active.
    """
    n = a.shape[0]
    max_val = conv.max()
    mask = np.isclose(conv, max_val, atol=1e-15, rtol=0.0)
    active = np.where(mask)[0]
    grad = np.zeros_like(a)

    for k in active:
        lo = max(0, k - (n - 1))
        hi = min(k, n - 1)
        idx = np.arange(lo, hi + 1, dtype=int)
        grad[idx] += 2.0 * a[k - idx]

    if len(active) > 1:
        grad /= len(active)
    return grad


def optimise_subgradient(initial_a: np.ndarray,
                         deadline: float,
                         eta0: float = 0.6,
                         max_iter: int = 200_000) -> np.ndarray:
    """
    Exponentiated sub‑gradient descent directly on Φ(a)=max_k (a∗a)_k.
    Returns the best vector seen before the deadline.
    """
    a = np.maximum(initial_a, 1e-30)
    a = a / a.sum()
    best_a = a.copy()
    best_sc = compute_score(a)
    n = a.shape[0]

    for it in range(1, max_iter + 1):
        if time.time() >= deadline:
            break

        conv = convolution_fft(a, a)
        grad = _max_conv_gradient(a, conv)

        eta = eta0 / np.sqrt(it)
        a = a * np.exp(-eta * grad)

        # project back onto simplex
        s = a.sum()
        if s == 0.0:
            a = np.full_like(a, 1.0 / n)
        else:
            a = a / s

        cur_sc = compute_score(a)
        if cur_sc < best_sc:
            best_sc = cur_sc
            best_a = a.copy()

        # occasional diagnostics (lightweight)
        if (it % 5000) == 0:
            print(f"    subgrad iter={it:,d} cur={cur_sc:.6f} best={best_sc:.6f}")

    return best_a

def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function that minimises the official
    ``evaluate_sequence`` objective.  The routine respects a wall‑clock budget
    (default 1000 s) and always returns a feasible list of floats.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    # Reserve a small safety margin for the final call to evaluate_sequence
    deadline = start_time + budget_s - 2.0

    best_seq: list[float] = []
    best_val: float = float("inf")

    # ------------------------------------------------------------------
    # 0.  Optional fallback if the notebook already produced a good construction.
    # ------------------------------------------------------------------
    try:
        fallback = [float(x) for x in height_sequence_1]  # type: ignore
        v = evaluate_sequence(fallback)
        print(f"Fallback (height_sequence_1) score = {v:.6f}")
        if v < best_val:
            best_val = v
            best_seq = fallback
    except Exception:
        pass

    # ------------------------------------------------------------------
    # 1.  Random seeding (large, cheap, diverse)
    # ------------------------------------------------------------------
    # Choose a single length – 2048 gave the best result so far and keeps the
    # convolution cheap enough for many inner loops.
    n = 2048
    all_candidates = []

    # a few deterministic seeds
    all_candidates.append(np.full(n, 1.0 / n))                 # uniform
    all_candidates.append(arcsine_initial(n))                # classic arcsine

    # random draws from expressive families
    N_random = 2000
    for _ in range(N_random):
        kind = rng.choice(["powerlaw", "beta", "dirichlet", "mixture"])
        if kind == "powerlaw":
            p = rng.uniform(0.1, 1.5)
            eps = rng.uniform(0.0, max(1.0, 0.02 * n))
            cand = generate_powerlaw(n, p, eps)
        elif kind == "beta":
            a = rng.uniform(0.1, 6.0)
            b = rng.uniform(0.1, 6.0)
            cand = generate_beta_like(n, a, b)
        elif kind == "dirichlet":
            conc = rng.uniform(0.1, 3.0)
            cand = random_dirichlet(n, conc, rng)
        else:  # mixture
            cand = random_beta_mixture(n, rng)
        # small random perturbation helps avoid exact zeros
        cand = perturb(cand, rng, scale=0.005)
        all_candidates.append(cand)

    # --------------------------------------------------------------
    # 1b.  Fast scoring, keep the best few dozen seeds.
    # --------------------------------------------------------------
    scored = [(compute_score(c), c) for c in all_candidates]
    scored.sort(key=lambda x: x[0])
    top_k = 30
    top_candidates = [c for _, c in scored[:top_k]]
    print(f"Generated {len(all_candidates)} seeds, keeping top {top_k} (best fast score = {scored[0][0]:.6f})")

    # ------------------------------------------------------------------
    # 2.  Refine each top candidate.
    # ------------------------------------------------------------------
    for idx, seed_vec in enumerate(top_candidates):
        if time.time() >= deadline - 10.0:
            break
        print(f"\n--- Refining candidate {idx + 1}/{len(top_candidates)} ---")

        # 2a. Simulated annealing (light)
        max_iter_sa = int(min(3000, (deadline - time.time()) * 200))
        sa_start_temp = max(0.005, 0.02 * compute_score(seed_vec))
        a_sa = simulated_annealing(seed_vec, deadline, rng,
                                   max_iter=max_iter_sa,
                                   start_temp=sa_start_temp,
                                   cooling=0.9995,
                                   max_step=0.025)

        # evaluate after SA
        val_sa = evaluate_sequence(a_sa.tolist())
        print(f"    SA -> score {val_sa:.6f}")
        if val_sa < best_val:
            best_val = val_sa
            best_seq = a_sa.tolist()
            print(f"    *** New global best (SA) {best_val:.6f} ***")

        # 2b. Smooth‑max gradient descent
        betas = [10, 20, 40, 80, 160, 320]
        # allocate iterations proportionally to remaining time
        remaining = max(0.0, deadline - time.time())
        it_per_beta = max(150, int(min(1200, remaining * 250) / len(betas)))
        a_sg = softmax_gradient_descent(a_sa, deadline,
                                        betas=betas,
                                        iter_per_beta=it_per_beta,
                                        eta0=0.5)

        val_sg = evaluate_sequence(a_sg.tolist())
        print(f"    Smooth‑GD -> score {val_sg:.6f}")
        if val_sg < best_val:
            best_val = val_sg
            best_seq = a_sg.tolist()
            print(f"    *** New global best (smooth‑GD) {best_val:.6f} ***")

        # 2c. One LP improvement step (optional, cheap)
        b_lp = lp_improve(a_sg, deadline)
        if b_lp is not None:
            val_lp = evaluate_sequence(b_lp.tolist())
            print(f"    LP boost -> score {val_lp:.6f}")
            if val_lp < best_val:
                best_val = val_lp
                best_seq = b_lp.tolist()
                print(f"    *** New global best (LP) {best_val:.6f} ***")

        # 2d. Second LP boost after the first – sometimes a second pass helps
        if b_lp is not None and time.time() < deadline - 2.0:
            b_lp2 = lp_improve(b_lp, deadline)
            if b_lp2 is not None:
                val_lp2 = evaluate_sequence(b_lp2.tolist())
                print(f"    Second LP boost -> score {val_lp2:.6f}")
                if val_lp2 < best_val:
                    best_val = val_lp2
                    best_seq = b_lp2.tolist()
                    print(f"    *** New global best (2nd LP) {best_val:.6f} ***")

    # ------------------------------------------------------------------
    # 3.  Final polishing of the overall best sequence (if we still have time)
    # ------------------------------------------------------------------
    if best_seq and time.time() < deadline - 5.0:
        print("\nFinal polishing with true max‑convolution sub‑gradient")
        a0 = np.array(best_seq, dtype=float)
        remaining = max(0.0, deadline - time.time())
        max_iter = int(min(300_000, remaining * 300))
        a_refined = optimise_subgradient(initial_a=a0,
                                         deadline=deadline,
                                         eta0=0.4,
                                         max_iter=max_iter)
        final_val = evaluate_sequence(a_refined.tolist())
        print(f"Final polished score = {final_val:.6f}")
        if final_val < best_val:
            best_val = final_val
            best_seq = a_refined.tolist()
            print("*** Updated global best after final polishing ***")

    # ------------------------------------------------------------------
    # 4.  Guard – always return a feasible sequence.
    # ------------------------------------------------------------------
    if not best_seq:
        best_seq = [1.0] * 256
        best_val = evaluate_sequence(best_seq)

    print(f"\nSearch finished. Best reported score = {best_val:.6f}")
    return best_seq
