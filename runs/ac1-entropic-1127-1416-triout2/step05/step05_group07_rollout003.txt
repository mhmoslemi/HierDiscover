    # Compute full cross-correlation (like np.correlate(a, w, mode='full')).
    # correlation = ifft(fft(a, N) * conj(fft(w, N))) (since correlation = a ⋆ w = ifft( A * conj(W) ), where A = fft(a), W = fft(w)

def smooth_max_gradient(a, beta):
    conv = convolution_fft(a, a)
    # compute softmax weights
    max_conv = conv.max()
    # subtract max to avoid overflow
    exp_vals = np.exp(beta * (conv - max_conv))
    w = exp_vals / exp_vals.sum()
    # correlation = a ⋆ w (full)
    # Use np.fft.rfft to compute correlation.
    n = len(a)
    m = len(w)
    N = next_pow2(n + m - 1)
    A = np.fft.rfft(a, N)
    W = np.fft.rfft(w, N)
    corr_full = np.fft.irfft(A * np.conj(W), N)[:n + m - 1]
    # shift index start = m - 1
    grad = 2.0 * corr_full[m - 1 : m - 1 + n]
    return grad, conv

The objective  

\[
\Phi(a)=\frac{2\,n\;\max_k (a*a)_k}{(\sum_i a_i)^2}
\]

is scale‑invariant, therefore we restrict the search to the probability
simplex  
\(\sum_i a_i = 1,\;a_i\ge 0\).
For a fixed shape the numerator is the maximum entry of the
self‑convolution.  The optimal shape makes the whole convolution as flat
as possible – the discrete analogue of the continuous arcsine law.

The previous code used a sub‑gradient / exponentiated‑gradient scheme.
Here we replace it by a **smooth‑max Adam optimiser**:

* the non‑smooth $\max$ is replaced by a soft‑max (log‑sum‑exp) with a
temperature β that is increased during the run (annealing);
* the gradient of the smooth objective can be written in closed form as
a *cross‑correlation* of the current vector with the soft‑max weights;
  it is computed by two FFTs per iteration (self‑convolution + correlation);
* Adam (first‑order moment + variance adaptation) together with a
projection onto the simplex (zero out negatives, renormalise, and enforce
symmetry) gives a fast and stable descent;
* a hierarchical scheme (512 → 1024 → 2048 → 4096 → 8192) upsamples the
best vector found so far and refines it again, reusing the same optimiser.

All heavy work respects the 1000 s wall‑clock budget.  The final
candidate is evaluated with the official `evaluate_sequence` function.

```python
"""
Smooth‑max Adam optimiser for the max‑self‑convolution problem.

The algorithm works hierarchically:
    1.  start with a modest length (512) and a good initial shape
        (discrete arcsine law, random Dirichlet draws, etc.);
    2.  optimise the shape on the simplex using a soft‑max
        approximation of the max‑convolution and Adam updates;
    3.  up‑sample the best shape to the next length and repeat.
A small symmetry enforcement (a_i = a_{n‑1‑i}) is applied after every
projection – the optimal solution is known to be symmetric.
The optimiser uses two FFTs per iteration (self‑convolution + correlation)
and runs as many iterations as the remaining time permits.
"""

import time
import numpy as np
from typing import List, Tuple

# ----------------------------------------------------------------------
# Helper utilities (FFT, next power of two, simplex projection, etc.)
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """
    Full linear convolution a * b using a real FFT.
    """
    m = len(a) + len(b) - 1
    n_fft = next_pow2(m)
    A = np.fft.rfft(a, n=n_fft)
    B = np.fft.rfft(b, n=n_fft)
    C = A * B
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


def correlate_fft(a: np.ndarray, w: np.ndarray) -> np.ndarray:
    """
    Full cross‑correlation a ⋆ w (the same result as np.correlate
    with mode='full'), computed via FFT.
    """
    n = len(a)
    m = len(w)
    N = next_pow2(n + m - 1)
    A = np.fft.rfft(a, N)
    W = np.fft.rfft(w, N)
    corr = np.fft.irfft(A * np.conj(W), N)[: n + m - 1]
    return corr


def project_onto_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of v onto the probability simplex
    {x ≥ 0, Σ x = 1}.  Implements the O(n) algorithm of
    Duchi et al. (2008).
    """
    if v.sum() == 1.0 and np.all(v >= 0):
        return v
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # numerical fallback
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


def symmetrize(v: np.ndarray) -> np.ndarray:
    """Force mirror symmetry v_i = v_{n-1-i}."""
    return (v + v[::-1]) / 2.0


# ----------------------------------------------------------------------
# Fast surrogate of the official evaluator (exact for sum‑normalised vectors)
# ----------------------------------------------------------------------


def compute_score(a: np.ndarray) -> float:
    """
    Exact objective for a normalised vector (Σa = 1):
        2·n·max(a*a)
    """
    n = len(a)
    conv = convolution_fft(a, a)
    max_val = float(conv.max())
    return 2.0 * n * max_val


# ----------------------------------------------------------------------
# Families of initial shapes
# ----------------------------------------------------------------------


def arcsine_initial(n: int) -> np.ndarray:
    """
    Discrete arcsine density: a_i ∝ 1 / sqrt((i+½)(n‑½‑i)).
    Returns a normalised probability vector.
    """
    i = np.arange(n, dtype=float)
    eps = 0.25                          # prevents division by zero at the ends
    vals = 1.0 / np.sqrt((i + 0.5) * (n - 0.5 - i) + eps)
    vals /= vals.sum()
    return vals


def random_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    """
    Simple Dirichlet‑like draw.
    conc > 0  → gamma(shape=conc, scale=1)
    conc ≤ 0  → heavy‑tailed (exponential power)
    """
    if conc > 0:
        samp = rng.gamma(conc, 1.0, size=n)
    else:
        # heavy tail, tends to sparse vectors
        alpha = max(0.1, -conc)
        samp = rng.exponential(1.0, size=n) ** (1.0 / alpha)
    samp = np.maximum(samp, 1e-12)
    return samp / samp.sum()


def perturb(a: np.ndarray, rng: np.random.Generator, scale: float = 0.02) -> np.ndarray:
    """Add i.i.d. Gaussian noise and re‑normalise."""
    b = a + rng.normal(0.0, scale, size=a.shape)
    b = np.maximum(b, 0.0)
    if b.sum() == 0.0:
        return np.full_like(a, 1.0 / len(a))
    return b / b.sum()


def upsample(prev: np.ndarray, new_n: int) -> np.ndarray:
    """
    Linear interpolation from len(prev) → new_n, followed by normalisation.
    """
    x_old = np.linspace(0.0, 1.0, len(prev))
    x_new = np.linspace(0.0, 1.0, new_n)
    interp = np.interp(x_new, x_old, prev)
    total = interp.sum()
    if total == 0.0:
        return np.full(new_n, 1.0 / new_n)
    return interp / total


# ----------------------------------------------------------------------
# Smooth‑max Adam optimiser (core of the new search)
# ----------------------------------------------------------------------


def optimise_adam_softmax(
    init_a: np.ndarray,
    deadline: float,
    max_iter: int,
    *,
    lr: float = 0.5,
    beta0: float = 0.1,
    beta_max: float = 500.0,
    beta_mult: float = 1.001,
    eps: float = 1e-8,
) -> Tuple[np.ndarray, float]:
    """
    Adam optimiser on the smooth surrogate
        Φ_β(a) = (1/β) log Σ_k exp(β·(a*a)_k)
    constrained to the probability simplex.
    Returns the best vector seen and its (exact) score.
    """
    a = np.maximum(init_a, 1e-30)
    a = a / a.sum()

    m = np.zeros_like(a)          # first moment
    v = np.zeros_like(a)          # second moment

    best_a = a.copy()
    best_sc = compute_score(a)

    beta = beta0

    for it in range(1, max_iter + 1):
        if time.time() >= deadline:
            break

        # self‑convolution
        conv = convolution_fft(a, a)

        # soft‑max weights (stable formulation)
        max_conv = conv.max()
        exp_vals = np.exp(beta * (conv - max_conv))
        w = exp_vals / exp_vals.sum()        # Σ w = 1, all w ≥ 0

        # gradient of Φ_β :  g = 2 * (a ⋆ w)  (valid entries only)
        corr_full = correlate_fft(a, w)
        # cross‑correlation length = n + (2n‑1) - 1 = 3n‑2.
        # The required slice starts at (len(w)-1) = 2n‑2.
        n = len(a)
        start = len(w) - 1
        grad = 2.0 * corr_full[start : start + n]

        # Adam update
        m = 0.9 * m + 0.1 * grad
        v = 0.999 * v + 0.001 * (grad * grad)

        m_hat = m / (1.0 - 0.9 ** it)
        v_hat = v / (1.0 - 0.999 ** it)

        a -= lr * m_hat / (np.sqrt(v_hat) + eps)

        # projection onto the simplex + symmetry enforcement
        a = np.maximum(a, 0.0)
        a = symmetrize(a)
        a = project_onto_simplex(a)

        # keep the best (exact) score seen so far
        cur_sc = compute_score(a)
        if cur_sc < best_sc:
            best_sc = cur_sc
            best_a = a.copy()

        # anneal β towards β_max
        if beta < beta_max:
            beta = min(beta_max, beta * beta_mult)

        # occasional progress report (does not affect timing much)
        if it % 20000 == 0:
            print(
                f"    ADAM iter={it:,d} β={beta:.2f} cur={cur_sc:.6f} best={best_sc:.6f}"
            )

    return best_a, best_sc


# ----------------------------------------------------------------------
# Main entry point required by the competition framework
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs) -> List[float]:
    """
    Search for a non‑negative step‑function with a low
    ``evaluate_sequence`` value.
    The routine follows a multiscale scheme and uses the smooth‑max
    Adam optimiser described above.
    """
    rng = np.random.default_rng(seed)
    start = time.time()
    deadline = start + budget_s - 2.0            # small safety margin

    best_seq: List[float] = []
    best_val: float = float("inf")
    best_a: np.ndarray = None                    # normalised best vector

    # ------------------------------------------------------------------
    # 0.  Optional fallback from a previously discovered construction.
    # ------------------------------------------------------------------
    try:
        fallback = [float(x) for x in height_sequence_1]   # type: ignore
        val = evaluate_sequence(fallback)
        print(f"Fallback (height_sequence_1) score = {val:.6f}")
        if val < best_val:
            best_val = val
            best_seq = fallback
            best_a = np.array(fallback, dtype=float)
            best_a = best_a / best_a.sum()
    except Exception:
        pass

    # ------------------------------------------------------------------
    # 1.   Length hierarchy – start small, enlarge progressively.
    # ------------------------------------------------------------------
    lengths = [512, 1024, 2048, 4096, 8192]     # stop earlier if time runs out

    for n in lengths:
        if time.time() >= deadline - 5.0:
            break
        print(f"\n=== Exploring length n = {n} ===")

        # --------------------------------------------------------------
        # 1a. Build a pool of initial vectors.
        # --------------------------------------------------------------
        pool: List[np.ndarray] = []

        if best_a is not None:
            up = upsample(best_a, n)
            pool.append(up)

        pool.append(arcsine_initial(n))

        for conc in (0.5, 1.0, 2.0):
            pool.append(random_dirichlet(n, conc, rng))

        for _ in range(2):
            pool.append(perturb(arcsine_initial(n), rng, scale=0.02))

        # --------------------------------------------------------------
        # 1b. Quick screening with the fast exact score.
        # --------------------------------------------------------------
        scored = [(compute_score(c), c) for c in pool]
        scored.sort(key=lambda x: x[0])
        top_cands = [c for _, c in scored[:3]]          # keep the three most promising

        # --------------------------------------------------------------
        # 1c. Refine each candidate with the Adam optimiser.
        # --------------------------------------------------------------
        for idx, cand in enumerate(top_cands):
            if time.time() >= deadline - 5.0:
                break
            print(f"  Candidate {idx + 1}/{len(top_cands)} (n={n})")

            # Estimate how many iterations we can still afford.
            remaining = max(0.0, deadline - time.time())
            # Empirical budget: ~250 iterations per second for n≈8k.
            max_iter = int(min(300_000, max(5_000, remaining * 250)))

            refined, _ = optimise_adam_softmax(
                init_a=cand,
                deadline=deadline,
                max_iter=max_iter,
                lr=0.5,
                beta0=0.1,
                beta_max=500.0,
                beta_mult=1.001,
            )

            # Official evaluation (expensive, but we call it only for the few
            # best candidates after optimisation).
            val = evaluate_sequence(refined.tolist())
            print(f"    evaluate_sequence = {val:.6f}")

            if val < best_val:
                best_val = val
                best_seq = refined.tolist()
                best_a = refined.copy()
                print(f"    *** New global best: {best_val:.6f} (n={n}) ***")

    # ------------------------------------------------------------------
    # 2.   Final polishing of the overall best (time permitting).
    # ------------------------------------------------------------------
    if best_seq and time.time() < deadline - 5.0:
        a = np.array(best_seq, dtype=float)
        remaining = max(0.0, deadline - time.time())
        max_iter = int(min(500_000, remaining * 250))

        print("\nFinal polishing of the global best")
        a_ref, _ = optimise_adam_softmax(
            init_a=a,
            deadline=deadline,
            max_iter=max_iter,
            lr=0.3,               # smaller step size for fine tuning
            beta0=0.5,
            beta_max=1000.0,
            beta_mult=1.001,
        )
        final_val = evaluate_sequence(a_ref.tolist())
        print(f"Final polished score = {final_val:.6f}")
        if final_val < best_val:
            best_val = final_val
            best_seq = a_ref.tolist()
            print("*** Updated global best after final polish ***")

    # ------------------------------------------------------------------
    # 3.   Guard – always return a feasible sequence.
    # ------------------------------------------------------------------
    if not best_seq:
        best_seq = [1.0] * 256
        best_val = evaluate_sequence(best_seq)

    print(f"\nSearch finished. Best reported score = {best_val:.6f}")
    return best_seq
```
