def build_sidon_set(n, time_limit):
    best_set=[]
    best_size=0
    start=time.time()
    max_possible = int((np.sqrt(16*n -7) -1)//2)
    while time.time() < time_limit:
        perm=np.random.permutation(n)
        used=np.zeros(2*n-1, dtype=bool)
        cur=[]
        for i in perm:
            # check self sum
            if used[2*i]:
                continue
            if cur:
                sums = i + np.array(cur, dtype=int)
                if used[sums].any():
                    continue
            # add
            cur.append(int(i))
            used[2*i]=True
            if cur[:-1]:
                used[i+np.array(cur[:-1], dtype=int)] = True
        if len(cur) > best_size:
            best_set = cur
            best_size = len(cur)
            # early break if reached theoretical max
            if best_size == max_possible:
                break
        # optionally break after many iterations.
    # attempt to extend greedily to maximal
    # compute used sums for best_set
    used = np.zeros(2*n-1, dtype=bool)
    for i in best_set:
        for j in best_set:
            used[i+j] = True
    best_set_set = set(best_set)
    improved=True
    while improved and time.time() < time_limit:
        improved=False
        for i in range(n):

def solve_lp_for_b_fast(a: np.ndarray, M: float):
    import scipy.optimize as opt
    n = a.shape[0]
    rows = 2*n -1
    # Build matrix A of shape (rows, n)
    A = np.zeros((rows, n), dtype=float)
    # For each column j, fill a from row j to j+n-1
    for j in range(n):
        A[j:j+n, j] = a
    # constraints: A @ b <= M
    c = -np.ones(n)  # maximize sum, actually minimize -sum
    ub = np.full(rows, M)
    bounds = [(0, None)] * n
    res = opt.linprog(c, A_ub=A, b_ub=ub, bounds=bounds, method='highs')
    if res.success:
        return res.x
    else:
        return None

"""
Candidate generator for the step‑function challenge.

Strategy
--------
1. Initialise a sparse non‑negative vector and normalise it to sum = 1.
2. Alternating optimisation:
   – For the current vector a compute M = max(a⊛a).
   – Solve the linear program
        maximise   Σ b_j
        subject to b ≥ 0,
                   (a ⊛ b) ≤ M   (element‑wise)
     using SciPy’s high‑performance “highs” LP solver.
   – Normalise b to the simplex, mix a←(1‑t)·a+t·b (t≈0.2) and re‑project.
   This step monotonically drives the peak of the convolution down.
3. Refine the result with a projected sub‑gradient method on the exact
   objective 2·n·max(a⊛a) (the same as `evaluate_sequence` after fixing
   Σa = 1).  The sub‑gradient is taken from the current maximal shift.
4. Final stochastic hill‑climb (swaps / mass transfers) that accepts a
   move only when the true evaluation score improves.
All phases respect the 1000 s budget; the algorithm falls back to a
uniform vector if nothing else succeeds.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Fast convolution utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full convolution a ⊛ b using a real FFT (fast)."""
    m = len(a) + len(b) - 1
    n_fft = next_pow2(m)
    A = np.fft.rfft(a, n=n_fft)
    B = np.fft.rfft(b, n=n_fft)
    C = A * B
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


def compute_score(a: np.ndarray) -> float:
    """
    Fast replica of `evaluate_sequence` when Σa == 1.
    Returns 2·n·max(a⊛a).
    """
    n = a.shape[0]
    conv = convolution_fft(a, a)
    return 2.0 * n * float(np.max(conv))


# ----------------------------------------------------------------------
# Projection onto the probability simplex
# ----------------------------------------------------------------------


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the probability simplex
    (non‑negative entries that sum to 1).
    """
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / v.shape[0])

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, v.shape[0] + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.where(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


# ----------------------------------------------------------------------
# Linear programming step (a ⊛ b ≤ M)
# ----------------------------------------------------------------------


def solve_lp_for_b(a: np.ndarray, M: float):
    """
    Solve   maximise   Σ b_j   subject to   b ≥ 0,
                (a ⊛ b) ≤ M   (element‑wise)

    Uses scipy.optimize.linprog with the HiGHS solver.
    Returns the optimal b as a NumPy array, or None on failure.
    """
    try:
        import scipy.optimize as opt
    except Exception:  # pragma: no cover
        return None

    n = a.shape[0]
    rows = 2 * n - 1

    # Build the constraint matrix A where A[l, j] = a[l‑j] if 0≤l‑j<n else 0.
    A = np.zeros((rows, n), dtype=float)
    for j in range(n):
        A[j:j + n, j] = a

    # Linear programme: minimise -Σ b_j  (i.e. maximise Σ b_j)
    c = -np.ones(n, dtype=float)               # minimise –sum(b)
    ub = np.full(rows, M, dtype=float)         # (a⊛b) ≤ M
    bounds = [(0.0, None)] * n                  # b ≥ 0

    res = opt.linprog(
        c,
        A_ub=A,
        b_ub=ub,
        bounds=bounds,
        method="highs",
        options={"presolve": True},
    )
    if res.success and res.x is not None:
        return np.asarray(res.x, dtype=float)
    return None


# ----------------------------------------------------------------------
# Sub‑gradient optimiser (exact max‑convolution)
# ----------------------------------------------------------------------


def subgrad_optimize(init_a: np.ndarray,
                     time_deadline: float,
                     max_iter: int = 200_000,
                     step0: float = 0.07) -> tuple[np.ndarray, float]:
    """
    Projected sub‑gradient descent on 2·n·max(a⊛a) with Σa = 1.
    The sub‑gradient at the maximal shift k is
    g[i] = 2·a[k‑i] (if the index is valid, otherwise 0).
    """
    a = init_a.astype(float).copy()
    a = project_to_simplex(a)           # ensure exact simplex
    best_a = a.copy()
    best_score = compute_score(a)
    n = a.shape[0]

    for t in range(1, max_iter + 1):
        if time.time() >= time_deadline:
            break

        conv = convolution_fft(a, a)
        k = int(np.argmax(conv))

        # sub‑gradient of the maximal entry
        idx = np.arange(n)
        j = k - idx
        mask = (j >= 0) & (j < n)
        g = np.zeros_like(a)
        g[mask] = 2.0 * a[j[mask]]

        step = step0 / np.sqrt(t)
        a_new = a - step * g
        a_new = project_to_simplex(a_new)

        cur_score = compute_score(a_new)
        if cur_score < best_score:
            best_score = cur_score
            best_a = a_new.copy()

        a = a_new

        # occasional jitter to escape plateaus
        if t % 1000 == 0:
            a += np.random.randn(n) * (step0 * 0.005 / np.sqrt(t))
            a = project_to_simplex(a)

    return best_a, best_score


# ----------------------------------------------------------------------
# Hill‑climber (swap / mass transfer)
# ----------------------------------------------------------------------


def hill_climb_permutation(seq, time_deadline, max_steps=300_000):
    """
    Stochastic hill‑climb that swaps entries or transfers a fraction of mass.
    The true (unsmoothed) objective is used for acceptance.
    """
    a = np.array(seq, dtype=float)
    n = a.shape[0]

    best_a = a.copy()
    best_score = compute_score(best_a)

    step_scale = 0.1  # fraction of mass moved in a transfer

    while max_steps > 0 and time.time() < time_deadline:
        # choose move type
        move_type = np.random.rand()
        cand = best_a.copy()

        if move_type < 0.5:
            # swap two positions
            i, j = np.random.randint(n, size=2)
            cand[i], cand[j] = cand[j], cand[i]
        else:
            # transfer a fraction from i to j
            i, j = np.random.randint(n, size=2)
            if i == j:
                continue
            delta = step_scale * best_a[i]
            if delta <= 0:
                continue
            cand[i] -= delta
            cand[j] += delta
            if cand[i] < 0:
                cand[i] = 0.0

        cand = project_to_simplex(cand)
        cur_score = compute_score(cand)

        if cur_score < best_score:
            best_score = cur_score
            best_a = cand
            step_scale = min(step_scale * 1.05, 0.5)
        else:
            step_scale *= 0.97

        max_steps -= 1

    return best_a.tolist(), best_score
