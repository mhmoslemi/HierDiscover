"""
Step‑function optimizer for `evaluate_sequence`.

The search proceeds in several stages within the 1000 s budget:

1. **Parametric sweep** – try simple analytic families (uniform,
   power‑law, exponential, cosine) on a few lengths (512‑2048).  The best
   candidate is kept.

2. **Random sparse seeds** – generate many sparse vectors with equal
   non‑zero mass; keep any improvement.

3. **Linear‑programming mixing** – using `scipy.optimize.linprog`
   (available in the execution environment) we solve
   maximise Σ b_j  s.t.  a ⊛ b ≤ M (M = max(a⊛a)) and b ≥ 0.
   The optimal b is normalised and mixed into the current a.
   Several iterations are performed; each iteration strictly reduces the
   score or stops.

4. **Adam‑style projected sub‑gradient descent** – the exact sub‑gradient
   of the peak convolution entry is used with an Adam optimiser; after
   each step the vector is projected onto the probability simplex.

5. **Hill‑climb polishing** – stochastic swaps / small mass transfers
   are accepted only when they lower the true score.  A large number of
   moves are performed until time runs out.

All intermediate vectors are kept on the simplex (non‑negative,
sum‑to‑1) so the objective simplifies to `2·n·max(a⊛a)`.  The final
vector is returned as a plain Python list of floats.
"""

import time
import math
import numpy as np

# --------------------------------------------------------------
# Optional SciPy imports (used for the LP step)
# --------------------------------------------------------------
try:
    from scipy import sparse
    from scipy.optimize import linprog
    _SCIPY_AVAILABLE = True
except Exception:   # pragma: no cover
    _SCIPY_AVAILABLE = False

# --------------------------------------------------------------
# Basic helpers
# --------------------------------------------------------------
def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full convolution using a real FFT (fast)."""
    m = a.shape[0] + b.shape[0] - 1
    n_fft = next_pow2(m)
    A = np.fft.rfft(a, n=n_fft)
    B = np.fft.rfft(b, n=n_fft)
    C = A * B
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


def compute_score(a: np.ndarray) -> float:
    """
    Fast replica of `evaluate_sequence` when sum(a) == 1.
    Returns 2·n·max(a⊛a).
    """
    n = a.shape[0]
    conv = convolution_fft(a, a)
    return 2.0 * n * float(np.max(conv))


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the probability simplex
    (non‑negative entries that sum to 1).
    """
    if np.all(v <= 0):
        # Return the uniform distribution
        return np.full_like(v, 1.0 / v.shape[0])

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, v.shape[0] + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.where(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    # Numerical safety: renormalise
    w_sum = w.sum()
    if w_sum == 0:
        return np.full_like(v, 1.0 / v.shape[0])
    return w / w_sum


# --------------------------------------------------------------
# LP step (uses SciPy if available)
# --------------------------------------------------------------
def solve_lp_for_b_scipy(a: np.ndarray, M: float) -> np.ndarray | None:
    """
    Solve the linear program:
        maximise   Σ b_j
        subject to b ≥ 0,
                   (a ⊛ b)_k ≤ M   for all k.
    Returns the optimal b as a NumPy array, or None on failure.
    """
    if not _SCIPY_AVAILABLE:
        return None

    n = a.shape[0]
    L = 2 * n - 1

    # Build (dense -> sparse) convolution matrix C such that C @ b = a ⊛ b
    # Row k, column i (i from 0..n-1) gets a_i if 0 ≤ k-i < n.
    # We construct it with simple vectorised indexing.
    i = np.arange(n, dtype=np.int64)[:, None]          # shape (n,1)
    j = np.arange(n, dtype=np.int64)[None, :]          # shape (1,n)
    rows = (i + j).ravel()                             # length n*n
    cols = np.repeat(np.arange(n, dtype=np.int64), n)  # length n*n
    data = np.repeat(a, n)                             # a_i repeated n times

    C = sparse.csr_matrix((data, (rows, cols)), shape=(L, n))

    # Linear objective: maximise sum(b)  <=> minimise -sum(b)
    c = -np.ones(n)

    # Upper‑bound constraints: C @ b ≤ M
    ub = np.full(L, M)

    bounds = [(0, None) for _ in range(n)]

    try:
        res = linprog(c, A_ub=C, b_ub=ub, bounds=bounds,
                      method='highs', options={'presolve': True})
    except Exception:  # pragma: no cover
        return None

    if not res.success or res.x is None:
        return None
    return np.asarray(res.x, dtype=float)


# --------------------------------------------------------------
# Adam‑style sub‑gradient optimiser
# --------------------------------------------------------------
def adam_optimize(a0: np.ndarray,
                  time_deadline: float,
                  max_iter: int = 200_000,
                  lr0: float = 0.08) -> tuple[np.ndarray, float]:
    """
    Projected Adam optimiser on the exact objective 2·n·max(a⊛a).
    Returns the best vector seen and its score.
    """
    a = a0.copy()
    n = a.shape[0]

    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8

    best_a = a.copy()
    best_score = compute_score(a)

    for t in range(1, max_iter + 1):
        if time.time() >= time_deadline:
            break

        conv = convolution_fft(a, a)
        M = float(conv.max())
        k = int(conv.argmax())

        # sub‑gradient of the max entry (simple case: single argmax)
        g = np.zeros_like(a)
        idx = np.arange(n)
        j = k - idx
        mask = (j >= 0) & (j < n)
        g[mask] = 2.0 * a[j[mask]]

        # Adam update
        lr = lr0 / math.sqrt(t)
        m = beta1 * m + (1 - beta1) * g
        v = beta2 * v + (1 - beta2) * (g * g)

        m_hat = m / (1 - beta1 ** t)
        v_hat = v / (1 - beta2 ** t)

        step = lr * m_hat / (np.sqrt(v_hat) + eps)
        a = a - step
        a = project_to_simplex(a)

        cur_score = compute_score(a)
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()

        # occasional tiny random perturbation to escape plateaus
        if t % 1000 == 0:
            noise = np.random.randn(n) * (lr0 * 0.004 / math.sqrt(t))
            a = project_to_simplex(a + noise)

    return best_a, best_score


# --------------------------------------------------------------
# Hill‑climb polishing (stochastic swaps / mass transfers)
# --------------------------------------------------------------
def hill_climb(a0: list[float],
               time_deadline: float,
               max_steps: int = 500_000) -> tuple[list[float], float]:
    """
    Simple stochastic hill‑climber that only accepts improving moves.
    Works directly with the true objective (`evaluate_sequence`) via
    `compute_score`.
    """
    a = np.array(a0, dtype=float)
    n = a.shape[0]

    best_a = a.copy()
    best_score = compute_score(best_a)

    # step size as a fraction of the mass at the source index
    base_frac = 0.12

    while max_steps > 0 and time.time() < time_deadline:
        move_type = np.random.rand()
        cand = best_a.copy()

        if move_type < 0.5:
            # swap two positions
            i, j = np.random.randint(n, size=2)
            cand[i], cand[j] = cand[j], cand[i]
        else:
            # move a fraction of mass from i to j
            i, j = np.random.randint(n, size=2)
            if i == j:
                max_steps -= 1
                continue
            delta = base_frac * best_a[i]
            if delta <= 0:
                max_steps -= 1
                continue
            cand[i] -= delta
            cand[j] += delta
            if cand[i] < 0:
                cand[i] = 0.0

        cand = project_to_simplex(cand)
        cur_score = compute_score(cand)

        if cur_score < best_score:
            best_score = cur_score
            best_a = cand
            # slowly increase the transferred fraction when we succeed
            base_frac = min(base_frac * 1.07, 0.5)
        else:
            # shrink when we fail
            base_frac *= 0.96

        max_steps -= 1

    return best_a.tolist(), best_score


# --------------------------------------------------------------
# Parametric families (used in the initial sweep)
# --------------------------------------------------------------
def make_uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n)


def make_power(n: int, p: float) -> np.ndarray:
    i = np.arange(1, n + 1, dtype=float)
    a = i ** p
    a /= a.sum()
    return a


def make_exponential(n: int, lam: float) -> np.ndarray:
    i = np.arange(n, dtype=float)
    a = np.exp(-lam * i)
    a /= a.sum()
    return a


def make_cosine(n: int) -> np.ndarray:
    # non‑negative cosine window that is zero at the ends
    i = np.arange(n, dtype=float)
    a = 0.5 * (1 - np.cos(2 * math.pi * i / (n - 1)))
    a += 1e-12                     # avoid exact zeros before projection
    a /= a.sum()
    return a

# Main entry point expected by the harness
# --------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step‑function that minimises
    `evaluate_sequence`.  The routine respects a 1000 s wall‑clock
    budget and returns a plain Python list of floats.
    """
    np.random.seed(seed)
    start_time = time.time()
    # keep a small safety margin for the harness
    deadline = start_time + budget_s - 2.0

    best_vec = None
    best_score = float("inf")

    # ----------------------------------------------------------
    # Phase 0 – convenient fallback (if a known solution exists)
    # ----------------------------------------------------------
    try:
        fallback = [float(x) for x in height_sequence_1]  # type: ignore
        fb_score = evaluate_sequence(fallback)
        print(f"Fallback score = {fb_score:.6f}")
        if fb_score < best_score:
            best_score = fb_score
            best_vec = fallback
    except Exception:
        pass

    # ----------------------------------------------------------
    # Phase 1 – parametric sweep on a few modest lengths
    # ----------------------------------------------------------
    lengths = [512, 1024, 2048]
    power_exps = [0.0, 0.2, 0.5, 0.8, 1.0, 1.3, 1.6, 2.0, 3.0, 4.0]
    exp_lams = np.logspace(-3, 0.5, 15)   # from 0.001 to ~3.16
    for n in lengths:
        if time.time() >= deadline - 200:
            break

        # uniform
        cand = make_uniform(n)
        sc = compute_score(cand)
        if sc < best_score:
            best_score = sc
            best_vec = cand.tolist()
            print(f"Uniform n={n} improved: {best_score:.6f}")

        # cosine window
        cand = make_cosine(n)
        sc = compute_score(cand)
        if sc < best_score:
            best_score = sc
            best_vec = cand.tolist()
            print(f"Cosine n={n} improved: {best_score:.6f}")

        # power law family
        for p in power_exps:
            cand = make_power(n, p)
            sc = compute_score(cand)
            if sc < best_score:
                best_score = sc
                best_vec = cand.tolist()
                print(f"Power p={p:.2f} n={n} improved: {best_score:.6f}")

        # exponential family
        for lam in exp_lams:
            cand = make_exponential(n, lam)
            sc = compute_score(cand)
            if sc < best_score:
                best_score = sc
                best_vec = cand.tolist()
                print(f"Exp λ={lam:.4f} n={n} improved: {best_score:.6f}")

    # ----------------------------------------------------------
    # Phase 2 – random sparse‑subset initialisations
    # ----------------------------------------------------------
    # generate many cheap candidates; we only keep improvements.
    sparse_lengths = [512, 1024, 2048, 4096]
    while time.time() < deadline - 150.0:
        n = int(np.random.choice(sparse_lengths))
        # choose a density between 20 % and 80 %
        density = np.random.uniform(0.20, 0.80)
        k = max(2, int(round(density * n)))
        idx = np.random.choice(n, size=k, replace=False)
        cand = np.zeros(n, dtype=float)
        cand[idx] = 1.0 / k
        sc = compute_score(cand)
        if sc < best_score:
            best_score = sc
            best_vec = cand.tolist()
            print(f"Sparse init n={n}, k={k} improved: {best_score:.6f}")

    # ----------------------------------------------------------
    # Phase 3 – LP‑based alternating optimisation (SciPy)
    # ----------------------------------------------------------
    if _SCIPY_AVAILABLE and best_vec is not None and time.time() < deadline - 80.0:
        print("\nStarting LP‑based alternating optimisation (SciPy)")
        a = np.array(best_vec, dtype=float)
        a = project_to_simplex(a)          # enforce exact simplex

        lp_iters = 0
        max_lp_iters = 20
        while time.time() < deadline - 40.0 and lp_iters < max_lp_iters:
            conv = convolution_fft(a, a)
            M = float(conv.max())

            b = solve_lp_for_b_scipy(a, M)
            if b is None:
                print("LP solve failed – exiting LP loop")
                break

            b_sum = b.sum()
            if b_sum <= 0:
                print("LP returned zero mass – exiting LP loop")
                break
            b = b / b_sum

            # mix a towards b (moderate step)
            mixing = 0.25
            a = (1.0 - mixing) * a + mixing * b
            a = project_to_simplex(a)

            cur_score = compute_score(a)
            if cur_score < best_score:
                best_score = cur_score
                best_vec = a.tolist()
                print(f"LP iter {lp_iters}: improved score = {best_score:.6f}")

            lp_iters += 1

    # ----------------------------------------------------------
    # Phase 4 – Adam sub‑gradient optimisation
    # ----------------------------------------------------------
    if best_vec is not None and time.time() < deadline - 30.0:
        print("\nAdam sub‑gradient refinement")
        a_init = np.array(best_vec, dtype=float)
        refined_a, refined_score = adam_optimize(
            a_init,
            time_deadline=deadline - 15.0,
            max_iter=200_000,
            lr0=0.08,
        )
        if refined_score < best_score:
            best_score = refined_score
            best_vec = refined_a.tolist()
            print(f"Adam refinement improved best to {best_score:.6f}")

    # ----------------------------------------------------------
    # Phase 5 – final hill‑climb polishing
    # ----------------------------------------------------------
    if best_vec is not None and time.time() < deadline - 5.0:
        print("\nHill‑climb polishing")
        polished_seq, polished_score = hill_climb(
            best_vec,
            time_deadline=deadline,
            max_steps=600_000,
        )
        if polished_score < best_score:
            best_score = polished_score
            best_vec = polished_seq
            print(f"Hill‑climb improved best to {best_score:.6f}")

    # ----------------------------------------------------------
    # Safety fallback – if nothing was found, use a uniform vector.
    # ----------------------------------------------------------
    if best_vec is None:
        n = 256
        best_vec = [1.0 / n] * n
        best_score = evaluate_sequence(best_vec)

    # Final verification using the official evaluator
    final_score = evaluate_sequence(best_vec)
    print(f"\nFinished search. Final reported score = {final_score:.6f}")

    # Return a plain Python list of floats as required by the harness
    return best_vec
