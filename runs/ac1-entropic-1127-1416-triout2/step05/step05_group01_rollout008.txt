def propose_candidate(seed=123, budget_s=1000):
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 1.5

    # parameters
    n = 2048
    a = np.random.dirichlet(np.ones(n))
    best_a = a.copy()
    best_score = compute_score(best_a)

    # schedule parameters
    eta0 = 0.5
    tau0 = 0.1   # maybe ratio to range of conv
    # we will gradually decay tau

    iteration = 0
    while time.time() < deadline:
        iteration += 1
        t = iteration
        eta = eta0 / np.sqrt(t)

        conv = convolution_fft(a, a)
        max_c = conv.max()
        # compute temperature adaptively:
        tau = max(1e-12, tau0 * (max_c - conv.min()))
        # compute softmax weights
        w = np.exp((conv - max_c) / tau)
        w_sum = w.sum()
        if w_sum == 0:
            w = np.full_like(w, 1.0 / len(w))
        else:
            w = w / w_sum
        # gradient
        w_rev = w[::-1]
        conv_rev = convolution_fft(w_rev, a)
        # slice:
        slice_start = n-1
        slice_end = slice_start + n
        g = 2.0 * conv_rev[slice_start:slice_end][::-1]

        # exponentiated update
        a = a * np.exp(-eta * g)

        # project onto simplex
        a = project_to_simplex(a)

        # compute score occasionally
        if iteration % 200 == 0 or time.time() > deadline - 5:
            cur_score = compute_score(a)
            if cur_score < best_score:
                best_score = cur_score
                best_a = a.copy()
                print(f"Iter {iteration}: improved best score {best_score:.6f}")

    # final hill climb with swaps and mass transfers
    if time.time() < deadline - 5:
        polished_seq, polished_val = hill_climb_permutation(best_a.tolist(),
                                                            time_deadline=deadline,
                                                            max_steps=500000)
        if polished_val < best_score:
            best_score = polished_val
            best_a = np.array(polished_seq)

    final_seq = best_a.tolist()
    final_val = evaluate_sequence(final_seq)
    print(f"Search finished. Final reported score = {final_val:.6f}")
    return final_seq

"""
Propose a low‑score step function.

Algorithm
---------
1. Initialise a probability vector a (∑a = 1) of length n by a Dirichlet draw.
2. Repeatedly minimise the smoothed maximum of the autocorrelation
   c = a * a using exponentiated mirror‑descent.
   *   c is the linear convolution of a with itself.
   *   A soft‑max weight w≈exp((c‑max(c))/τ) spreads the influence of
       *all* large autocorrelation entries (τ > 0 is a temperature).
   *   The gradient of the smoothed objective is
           g = 2·conv_rev[w_rev , a]       (a sliding weighted dot‑product)
       where w_rev is w reversed.  The slice conv_rev[n‑1:2n‑1] (reversed)
       yields g.
   *   Update: a ← a·exp(‑η·g)  and project back onto the probability simplex.
   *   η decays like η₀/√t, τ decays slowly so the optimisation first
       flattens the whole autocorrelation and later concentrates on the
       true peak.
3. After the gradient phase, a short stochastic hill‑climber (swap / mass‑transfer
   moves) finishes the polishing.
4. The best vector found is returned unchanged – evaluate_sequence rescales
   the result internally, so keeping ∑a≈1 is fine.

The whole procedure respects the 1000 s budget.  If a historic good
construction ``height_sequence_1`` exists it is used as a warm start.
"""
import time
import numpy as np

# ----------------------------------------------------------------------
# FFT‑based convolution utilities (real FFT, linear convolution)
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """
    Linear convolution a*b using a real FFT (fast).
    Returns a 1‑D array of length len(a)+len(b)-1.
    """
    m = a.shape[0] + b.shape[0] - 1
    n_fft = _next_pow2(m)
    A = np.fft.rfft(a, n=n_fft)
    B = np.fft.rfft(b, n=n_fft)
    C = A * B
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


# ----------------------------------------------------------------------
# Simple Euclidean projection onto the probability simplex
# ----------------------------------------------------------------------


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the probability simplex
    (non‑negative entries that sum to 1).  Implements the algorithm of
    Wang & Carreira‑Perpiñán (2013).
    """
    if np.all(v <= 0):
        # equally spread mass if everything is non‑positive
        return np.full_like(v, 1.0 / v.shape[0])

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, v.shape[0] + 1)
    cond = u - cssv / ind > 0
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Fast replica of evaluate_sequence for vectors with sum = 1
# ----------------------------------------------------------------------


def compute_score(a: np.ndarray) -> float:
    """
    Return the exact value that evaluate_sequence would produce
    for a vector with sum(a)=1 (i.e. 2·n·max(a*a)).
    """
    n = a.shape[0]
    conv = convolution_fft(a, a)
    return float(2.0 * n * np.max(conv))


# ----------------------------------------------------------------------
# Stochastic hill‑climber (swap / mass‑transfer moves)
# ----------------------------------------------------------------------


def hill_climb_permutation(seq, time_deadline, max_steps=200_000):
    """
    Simple hill‑climber that swaps two entries or moves a fraction of mass.
    The objective used is the exact evaluate_sequence score (via compute_score).
    """
    a = np.array(seq, dtype=float)
    n = a.shape[0]

    best_a = a.copy()
    best_score = compute_score(best_a)

    step_scale = 0.1  # fraction of mass moved in a transfer

    while max_steps > 0 and time.time() < time_deadline:
        # generate a candidate
        cand = best_a.copy()
        if np.random.rand() < 0.5:
            i, j = np.random.randint(n, size=2)
            cand[i], cand[j] = cand[j], cand[i]          # swap
        else:
            i, j = np.random.randint(n, size=2)
            if i != j:
                delta = step_scale * best_a[i]
                cand[i] -= delta
                cand[j] += delta
                cand[i] = max(cand[i], 0.0)               # keep non‑negative

        # re‑project onto simplex
        cand = project_to_simplex(cand)
        cur_score = compute_score(cand)

        # accept if improvement
        if cur_score < best_score:
            best_score = cur_score
            best_a = cand
            step_scale = min(step_scale * 1.07, 0.5)      # a little more aggressive
        else:
            step_scale *= 0.96

        max_steps -= 1

    return best_a.tolist(), best_score


# ----------------------------------------------------------------------
# Smoothed exponentiated mirror‑descent on the exact objective
# ----------------------------------------------------------------------


def smoothed_exp_descent(
    a: np.ndarray,
    time_deadline: float,
    eta0: float = 0.6,
    tau0: float = 0.05,
    tau_decay: float = 0.9995,
) -> tuple[np.ndarray, float]:
    """
    Minimise 2·n·max(a*a) by minimising a smoothed version of the max.
    The temperature τ controls the smoothing and is slowly decreased.
    Returns the best vector seen and its score.
    """
    n = a.shape[0]
    best_a = a.copy()
    best_score = compute_score(best_a)

    iteration = 0
    tau = tau0

    while time.time() < time_deadline:
        iteration += 1
        # diminishing learning rate
        eta = eta0 / np.sqrt(iteration)

        # current autocorrelation
        conv = convolution_fft(a, a)                     # length 2n‑1
        max_c = conv.max()
        # adapt temperature – we keep a small positive lower bound
        range_c = max_c - conv.min()
        if range_c <= 0:
            range_c = 1e-12
        tau = max(tau * tau_decay, 1e-8)                # slowly cool down

        # soft‑max weights over the shifts
        w = np.exp((conv - max_c) / (tau * range_c))    # normalise by range to keep τ scale‑invariant
        w_sum = w.sum()
        if w_sum == 0.0:
            w = np.full_like(w, 1.0 / w.shape[0])
        else:
            w = w / w_sum

        # gradient = 2· (a * w) where * is the sliding dot‑product.
        # Efficiently obtained via one extra FFT.
        w_rev = w[::-1]
        conv_rev = convolution_fft(w_rev, a)            # length 3n‑2
        slice_start = n - 1
        slice_end = slice_start + n
        g = 2.0 * conv_rev[slice_start:slice_end][::-1]   # shape (n,)

        # exponentiated update (mirror‑descent on the simplex)
        # Clip exponent to avoid overflow/underflow
        exp_arg = -eta * g
        exp_arg = np.clip(exp_arg, -50.0, 50.0)
        a = a * np.exp(exp_arg)

        # projection back onto the simplex
        a = project_to_simplex(a)

        # occasional logging and best‑vector update
        if iteration % 500 == 0 or time.time() > time_deadline - 5.0:
            cur_score = compute_score(a)
            if cur_score < best_score:
                best_score = cur_score
                best_a = a.copy()
                print(f"Iter {iteration:>6d}: improved best score = {best_score:.6f}")

    return best_a, best_score
