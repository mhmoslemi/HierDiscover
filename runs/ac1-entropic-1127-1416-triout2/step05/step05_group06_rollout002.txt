def propose_candidate(seed=42, budget_s=1000, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()
    deadline = start + budget_s - 2

    best_val = float('inf')
    best_seq = None

    # fallback if present
    try:
        fallback = [float(x) for x in height_sequence_1] # may be defined globally
        val = evaluate_sequence(fallback)
        if val < best_val:
            best_val = val
            best_seq = fallback
    except NameError:
        pass

    # choose lengths
    base_lengths = [2048, 4096, 8192, 16384]
    # maybe start from smaller if time short.

    for n in base_lengths:
        if time.time() >= deadline - 5:
            break
        print(f"\n=== Length {n} ===")

        # generate seeds list
        seeds = []
        seeds.append(np.full(n, 1.0/n))
        seeds.append(arcsine_initial(n))
        # random power law
        for _ in range(4):
            p = rng.uniform(0.2, 2.0)
            eps = rng.uniform(0.0001, 0.01*n)  # maybe random
            seeds.append(generate_powerlaw(n, p, eps))
        # beta
        for _ in range(3):
            alpha = rng.uniform(0.2, 5.0)
            beta = rng.uniform(0.2, 5.0)
            seeds.append(generate_beta_like(n, alpha, beta))
        # Dirichlet
        for conc in [0.1, 0.5, 1.0, 2.0]:
            seeds.append(random_dirichlet(n, conc, rng))
        # perturbed arcsine
        for _ in range(2):
            seeds.append(perturb(arcsine_initial(n), rng, scale=rng.uniform(0.01,0.05)))
        # mixture arcsine/uniform
        for _ in range(2):
            w = rng.random()
            mix = w * arcsine_initial(n) + (1.0-w) * np.full(n, 1.0/n)
            seeds.append(mix / mix.sum())

        # Evaluate fast scores
        scored = [(compute_score(s), s) for s in seeds]
        scored.sort(key=lambda x: x[0])
        top = [s for _, s in scored[:4]]  # keep four best

        for idx, cand in enumerate(top):
            if time.time() >= deadline - 5:
                break
            print(f"candidate {idx+1}/{len(top)} for n={n}")

            # SA quick refinement
            max_iter_sa = min(2000, int((deadline - time.time()) * 300))
            if max_iter_sa > 0:
                start_temp = max(0.005, 0.02 * compute_score(cand))
                cand_sa, _ = simulated_annealing(cand, deadline, rng,
                                                max_iter=max_iter_sa,
                                                start_temp=start_temp,
                                                cooling=0.9995,
                                                max_step=0.03)
                cand = cand_sa
                # evaluate true score after SA optionally
                true_val = evaluate_sequence(cand.tolist())
                if true_val < best_val:
                    best_val = true_val
                    best_seq = cand.tolist()
                    print(f"  New best after SA: {best_val:.6f}")

            # Smooth max gradient descent
            # Determine remaining time for this candidate
            remaining = deadline - time.time()
            if remaining <= 0:
                break
            # Determine schedule of betas.
            betas = [1, 2, 4, 8, 16, 32, 64, 128, 256]
            # allocate iterations per beta: enough to consume half the remaining budget maybe
            total_iters = int(min(50000, remaining * 300))  # approximate 300 iter per sec
            it_per_beta = max(1, total_iters // len(betas))
            # Run
            a_gd, _ = softmax_gradient_descent(cand, deadline,
                                               betas=betas,
                                               iter_per_beta=it_per_beta,
                                               eta0=0.5)
            val_gd = evaluate_sequence(a_gd.tolist())
            print(f"  GD evaluated score = {val_gd:.6f}")
            if val_gd < best_val:
                best_val = val_gd
                best_seq = a_gd.tolist()
                print(f"  *** New global best (GD) {best_val:.6f}")

    # final polishing
    if best_seq and time.time() < deadline - 5:
        print("\nFinal polishing with subgradient")
        a0 = np.array(best_seq, dtype=float)
        remaining = deadline - time.time()
        max_iter = int(min(200_000, remaining * 300))
        a_ref, _ = optimise_subgradient(initial_a=a0,
                                        deadline=deadline,
                                        eta0=0.3,
                                        max_iter=max_iter)
        final_val = evaluate_sequence(a_ref.tolist())
        if final_val < best_val:
            best_val = final_val
            best_seq = a_ref.tolist()
            print(f"  Final polished score = {final_val:.6f}")

    if not best_seq:
        best_seq = [1.0] * 256
        best_val = evaluate_sequence(best_seq)

    print(f"Search finished. Best score {best_val:.6f}")
    return best_seq

"""
Hybrid multi‑scale optimizer for the max‑convolution objective.

The routine works in three stages:

1. **Global exploration** – for several sequence lengths (powers of two) a
   diverse set of analytically‑inspired seeds (arcsine, power‑law,
   beta‑like, Dirichlet, mixtures) is generated.  The fast surrogate score
   `2·n·max(a∗a)/(sum(a)²)` (scale‑invariant) is used to keep only the
   most promising few seeds.

2. **Refinement** – each kept seed is first improved by a very cheap
   simulated‑annealing move that conserves the simplex, then a
   *smooth‑max* gradient descent is run.  The non‑smooth max is replaced by
   a log‑sum‑exp surrogate with a schedule of increasing β; the gradient
   is obtained via a single FFT, and an exponentiated‑gradient update
   preserves non‑negativity while a renormalisation step projects onto the
   probability simplex.

3. **Final polishing** – the best sequence found so far is polished by an
   exponentiated sub‑gradient descent on the true max‑convolution (no
   smoothing).  All steps stop as soon as the overall time budget (default
   1000 s) is exhausted, guaranteeing that a feasible sequence is always
   returned.

The algorithm deliberately tries larger lengths (up to 16384) because the
theoretical optimum approaches a constant convolution in the limit, which
reduces the objective towards the lower bound of 1.
"""

import time
import numpy as np
from typing import List, Tuple

# ----------------------------------------------------------------------
# Low‑level numeric utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Return the smallest power of two ≥ x (x > 0)."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """
    Linear convolution using real FFT.
    Returns a vector of length len(a)+len(b)-1.
    """
    m = len(a) + len(b) - 1
    n_fft = next_pow2(m)
    A = np.fft.rfft(a, n=n_fft)
    B = np.fft.rfft(b, n=n_fft)
    C = A * B
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


def compute_score(a: np.ndarray) -> float:
    """
    Fast surrogate for the official objective (scale invariant).
    a is assumed non‑negative and need not be normalised.
    """
    n = len(a)
    conv = convolution_fft(a, a)
    max_val = float(conv.max())
    s = float(a.sum())
    if s < 1e-12:
        return float("inf")
    return 2.0 * n * max_val / (s * s)


# ----------------------------------------------------------------------
# Families of initial shapes (all returned normalised to sum = 1)
# ----------------------------------------------------------------------


def arcsine_initial(n: int) -> np.ndarray:
    """Discrete arcsine density on {0,…,n‑1}."""
    i = np.arange(n, dtype=float)
    eps = 1e-6
    vals = 1.0 / np.sqrt((i + 0.5) * (n - 0.5 - i) + eps)
    return vals / vals.sum()


def generate_powerlaw(n: int, p: float, eps: float) -> np.ndarray:
    """Symmetric power‑law a_i ∝ ((i+ε)(n‑1‑i+ε))⁻ᵖ."""
    i = np.arange(n, dtype=float)
    base = ((i + eps) * (n - 1 - i + eps)) ** (-p)
    total = base.sum()
    if total == 0.0:
        return np.full(n, 1.0 / n)
    return base / total


def generate_beta_like(n: int, alpha: float, beta: float) -> np.ndarray:
    """Beta‑like shape a_i ∝ (i+½)^{α‑1} (n‑1‑i+½)^{β‑1}."""
    i = np.arange(n, dtype=float)
    base = (i + 0.5) ** (alpha - 1) * (n - 1 - i + 0.5) ** (beta - 1)
    total = base.sum()
    if total == 0.0:
        return np.full(n, 1.0 / n)
    return base / total


def random_dirichlet(n: int, concentration: float, rng: np.random.Generator) -> np.ndarray:
    """Dirichlet‑like sample (Gamma draws, then normalise)."""
    if concentration <= 0:
        # heavy‑tailed exponential‑like draws
        samp = rng.exponential(1.0, size=n) ** (1.0 / max(0.1, concentration))
    else:
        samp = rng.gamma(concentration, 1.0, size=n)
    samp += 1e-12
    return samp / samp.sum()


def perturb(a: np.ndarray, rng: np.random.Generator, scale: float = 0.02) -> np.ndarray:
    """Add a tiny Gaussian perturbation and re‑normalise."""
    b = a + rng.normal(0.0, scale, size=a.shape)
    b = np.maximum(b, 0.0)
    s = b.sum()
    if s == 0.0:
        return np.full_like(a, 1.0 / len(a))
    return b / s


# ----------------------------------------------------------------------
# Simple stochastic neighbour move (used by simulated annealing)
# ----------------------------------------------------------------------


def propose_neighbor(a: np.ndarray, rng: np.random.Generator, max_step: float = 0.02) -> np.ndarray:
    """Transfer a small amount of mass from one index to another."""
    n = len(a)
    i, j = rng.integers(0, n, size=2)
    if i == j or a[i] == 0.0:
        return a
    delta = a[i] * rng.random() * max_step
    b = a.copy()
    b[i] -= delta
    b[j] += delta
    return b


# ----------------------------------------------------------------------
# Very light simulated annealing (keeps the simplex)
# ----------------------------------------------------------------------


def simulated_annealing(a0: np.ndarray,
                        deadline: float,
                        rng: np.random.Generator,
                        max_iter: int = 2000,
                        start_temp: float = 0.005,
                        cooling: float = 0.9995,
                        max_step: float = 0.02) -> Tuple[np.ndarray, float]:
    """
    Light SA that respects the simplex.  Returns the best vector seen
    together with its fast surrogate score.
    """
    a = a0.copy()
    cur_sc = compute_score(a)
    best_a = a.copy()
    best_sc = cur_sc
    temp = start_temp

    for _ in range(max_iter):
        if time.time() >= deadline:
            break
        cand = propose_neighbor(a, rng, max_step)
        cand_sc = compute_score(cand)
        delta = cand_sc - cur_sc
        if delta < 0.0 or rng.random() < np.exp(-delta / max(1e-12, temp)):
            a = cand
            cur_sc = cand_sc
            if cur_sc < best_sc:
                best_sc = cur_sc
                best_a = a.copy()
        temp *= cooling
    return best_a, best_sc


# ----------------------------------------------------------------------
# Smooth‑max (log‑sum‑exp) gradient descent
# ----------------------------------------------------------------------


def softmax_gradient_descent(initial_a: np.ndarray,
                             deadline: float,
                             betas: List[float],
                             iter_per_beta: int = 500,
                             eta0: float = 0.5) -> Tuple[np.ndarray, float]:
    """
    Minimise  Φ_β(a) = (1/β)·log Σ_k exp(β·(a∗a)_k)  using an
    exponentiated‑gradient update and simplex projection.
    """
    a = np.maximum(initial_a, 1e-30)
    a = a / a.sum()
    best_a = a.copy()
    best_sc = compute_score(a)
    n = len(a)

    for beta in betas:
        for it in range(1, iter_per_beta + 1):
            if time.time() >= deadline:
                return best_a, best_sc

            # full self‑convolution
            conv = convolution_fft(a, a)                     # length 2n‑1

            # soft‑max weights
            max_c = conv.max()
            exp_vals = np.exp(beta * (conv - max_c))
            p = exp_vals / exp_vals.sum()                    # Σ p = 1

            # cross‑correlation p ⋆ a  (length n)
            N = next_pow2(len(p) + n - 1)
            P = np.fft.fft(p, N)
            A = np.fft.fft(a, N)
            cross = np.fft.ifft(P * np.conj(A)).real[:n]     # = Σ_k p_k·a_{k‑j}
            cross = np.maximum(cross, 0.0)                   # safety

            # gradient of the smooth surrogate
            grad = 4.0 * n * cross

            # exponentiated‑gradient step (AdaGrad‑like schedule)
            eta = eta0 / np.sqrt(it)
            a = a * np.exp(-eta * grad)

            # project onto the probability simplex
            s = a.sum()
            if s == 0.0:
                a = np.full_like(a, 1.0 / n)
            else:
                a = a / s

            cur_sc = compute_score(a)
            if cur_sc < best_sc:
                best_sc = cur_sc
                best_a = a.copy()

    return best_a, best_sc


# ----------------------------------------------------------------------
# Sub‑gradient for the true max‑convolution (final polishing)
# ----------------------------------------------------------------------


def _max_conv_gradient(a: np.ndarray, conv: np.ndarray) -> np.ndarray:
    """
    Sub‑gradient of  max_k (a∗a)_k .
    For each active offset k the contribution is 2·a_{k‑j}; average if
    several offsets share the maximum.
    """
    n = len(a)
    max_val = conv.max()
    mask = np.isclose(conv, max_val, atol=1e-15, rtol=0.0)
    active = np.where(mask)[0]
    grad = np.zeros_like(a)

    for k in active:
        lo = max(0, k - (n - 1))
        hi = min(k, n - 1)
        idx = np.arange(lo, hi + 1, dtype=int)
        grad[idx] += 2.0 * a[k - idx]

    if len(active) > 1:
        grad /= len(active)
    return grad


def optimise_subgradient(initial_a: np.ndarray,
                         deadline: float,
                         *,
                         eta0: float = 0.6,
                         max_iter: int = 200_000,
                         noise_every: int = 0,
                         noise_scale: float = 0.0) -> Tuple[np.ndarray, float]:
    """
    Exponentiated sub‑gradient descent on the true max‑convolution.
    Returns the best vector seen (and its fast surrogate score).
    """
    a = np.maximum(initial_a, 1e-30)
    a = a / a.sum()
    best_a = a.copy()
    best_sc = compute_score(a)
    n = len(a)

    for it in range(1, max_iter + 1):
        if time.time() >= deadline:
            break

        conv = convolution_fft(a, a)
        grad = _max_conv_gradient(a, conv)

        eta = eta0 / np.sqrt(it)
        a = a * np.exp(-eta * grad)

        if noise_every and (it % noise_every == 0):
            a += np.random.randn(n) * noise_scale
            a = np.maximum(a, 0.0)

        s = a.sum()
        a = a / s if s != 0.0 else np.full_like(a, 1.0 / n)

        cur_sc = compute_score(a)
        if cur_sc < best_sc:
            best_sc = cur_sc
            best_a = a.copy()

        # occasional progress report (does not affect timing)
        if it % 2000 == 0:
            print(f"    subgrad iter={it:,d} cur={cur_sc:.6f} best={best_sc:.6f}")

    return best_a, best_sc
