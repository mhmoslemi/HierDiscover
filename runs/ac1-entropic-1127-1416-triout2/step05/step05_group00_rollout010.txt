def solve_lp_scipy(a, M):
    n = len(a)
    import scipy.linalg
    from scipy.optimize import linprog

    # Build A matrix of shape (2n-1, n)
    # first column: a[0], a[1], ..., a[n-1], zeros for rest
    c_col = np.concatenate([a, np.zeros(n-1)])
    # first row: a[0], zeros...
    c_row = np.concatenate([a[:1], np.zeros(n-1)])
    A = scipy.linalg.toeplitz(c_col, c_row)   # shape (2n-1, n)

    # bounds for b
    bounds = [(0, None)] * n

    # Objective: maximize sum(b) -> minimize -sum(b)
    c = -np.ones(n)

    # Upper bounds on constraints
    ub = np.full(2*n-1, M)

    # Solve LP
    res = linprog(c, A_ub=A, b_ub=ub, bounds=bounds, method='highs')
    if not res.success:
        return None
    b = res.x
    # Clip small negatives due to numerical
    b = np.maximum(b, 0.0)
    return b

"""
Improved search for a low‑score non‑negative step function.

The algorithm consists of three stages

1. **Random Dirichlet initialisation** – generate a few random probability
   vectors (simplex‑projected) and keep the best one.

2. **Alternating linear‑programming refinement** – for a fixed vector `a`
   compute its peak autocorrelation `M = max(a⊛a)`.  The linear program

        maximise   1ᵀ b
        subject to  a ⊛ b ≤ M          (element‑wise)
                    b ≥ 0

   is linear in `b`.  Using `scipy.optimize.linprog` we obtain a new vector
   `b` that spreads the mass into positions where `a` is small while keeping
   the convolution bounded by `M`.  After normalising `b` to the simplex we
   mix `b` back into `a` (convex combination) and repeat – each iteration
   provably does not increase the peak and usually lowers it.

3. **Finishing polish** – a short projected sub‑gradient descent on the
   true objective `max(a⊛a)` followed by a stochastic hill‑climber that
   swaps or transfers mass between coordinates.  Both respect the simplex
   constraint.

All steps are stopped well before the 1000 s budget expires.  If SciPy
is unavailable the code falls back to the random‑initialisation + sub‑gradient
strategy from the previous baseline.

The returned value is a plain list of floats ready for `evaluate_sequence`.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Optional SciPy imports (required for the LP step)
# ----------------------------------------------------------------------
try:
    import scipy.linalg
    from scipy.optimize import linprog
    SCIPY_OK = True
except Exception:                 # pragma: no cover
    SCIPY_OK = False

# ----------------------------------------------------------------------
# Utility functions
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full convolution a⊛b using a real FFT (fast)."""
    m = len(a) + len(b) - 1
    n_fft = next_pow2(m)
    A = np.fft.rfft(a, n=n_fft)
    B = np.fft.rfft(b, n=n_fft)
    C = A * B
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


def compute_score(a: np.ndarray) -> float:
    """
    Fast replica of `evaluate_sequence` for vectors that sum to 1.
    Returns 2·n·max(a⊛a).
    """
    n = a.shape[0]
    conv = convolution_fft(a, a)
    return 2.0 * n * float(np.max(conv))


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the probability simplex
    (non‑negative entries that sum to 1).
    """
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / v.shape[0])

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, v.shape[0] + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.where(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


# ----------------------------------------------------------------------
# Linear‑programming sub‑routine (SciPy based)
# ----------------------------------------------------------------------
def solve_lp_scipy(a: np.ndarray, M: float):
    """
    Solve   maximise   Σ b_j
    subject to   a ⊛ b ≤ M   (element‑wise) ,  b ≥ 0.

    Returns the optimal b as a NumPy array or ``None`` on failure.
    """
    if not SCIPY_OK:
        return None

    n = a.shape[0]

    # Build the Toeplitz matrix that encodes the convolution a⊛b.
    # Row ℓ has coefficients a[ℓ‑j] (when 0 ≤ ℓ‑j < n) and 0 otherwise.
    first_col = np.concatenate([a, np.zeros(n - 1)])
    first_row = np.concatenate([a[:1], np.zeros(n - 1)])
    A = scipy.linalg.toeplitz(first_col, first_row)   # shape (2n‑1, n)

    # Linear program data
    c = -np.ones(n)                                 # maximise Σb → minimise –Σb
    ub = np.full(2 * n - 1, M)                      # a⊛b ≤ M
    bounds = [(0, None)] * n                        # b_j ≥ 0

    # Solve with HiGHS (fast interior‑point simplex hybrid)
    res = linprog(
        c,
        A_ub=A,
        b_ub=ub,
        bounds=bounds,
        method="highs",
        options={"presolve": True, "time_limit": 30},
    )
    if not res.success or res.x is None:
        return None

    b = np.maximum(res.x, 0.0)   # clip tiny negatives caused by numerics
    return b

# Sub‑gradient optimiser (same idea as the previous baseline)
# ----------------------------------------------------------------------
def subgrad_optimize(init_a: np.ndarray,
                     time_deadline: float,
                     max_iter: int = 200_000,
                     step0: float = 0.07) -> tuple[np.ndarray, float]:
    """Projected sub‑gradient descent on  max(a⊛a)."""
    a = init_a.copy()
    best_a = a.copy()
    best_score = compute_score(a)
    n = len(a)

    for t in range(1, max_iter + 1):
        if time.time() >= time_deadline:
            break

        conv = convolution_fft(a, a)
        k = int(np.argmax(conv))

        # sub‑gradient of max‑entry at shift k:
        # (∂/∂a_i) conv[k] = a_{k-i} + a_{k-i} = 2*a_{k-i} when index valid.
        idx = np.arange(n)
        j = k - idx
        mask = (j >= 0) & (j < n)
        g = np.zeros_like(a)
        g[mask] = 2.0 * a[j[mask]]

        step = step0 / np.sqrt(t)
        a_new = a - step * g
        a_new = project_to_simplex(a_new)

        cur_score = compute_score(a_new)
        if cur_score < best_score:
            best_score = cur_score
            best_a = a_new.copy()

        a = a_new

        # occasional jitter to escape plateaus
        if t % 1000 == 0:
            a += np.random.randn(n) * (step0 * 0.005 / np.sqrt(t))
            a = project_to_simplex(a)

    return best_a, best_score


# ----------------------------------------------------------------------
# Simple stochastic hill‑climber (swap / mass‑transfer moves)
# ----------------------------------------------------------------------
def hill_climb(seq, time_deadline, max_steps=300_000):
    a = np.array(seq, dtype=float)
    n = a.shape[0]

    best = a.copy()
    best_score = compute_score(best)
    step_scale = 0.1

    while max_steps > 0 and time.time() < time_deadline:
        cand = best.copy()
        if np.random.rand() < 0.5:                     # swap
            i, j = np.random.randint(n, size=2)
            cand[i], cand[j] = cand[j], cand[i]
        else:                                          # transfer
            i, j = np.random.randint(n, size=2)
            if i != j:
                delta = step_scale * best[i]
                if delta > 0:
                    cand[i] -= delta
                    cand[j] += delta
                    cand[i] = max(cand[i], 0.0)

        cand = project_to_simplex(cand)
        cur_score = compute_score(cand)

        if cur_score < best_score:
            best_score = cur_score
            best = cand
            step_scale = min(step_scale * 1.05, 0.5)
        else:
            step_scale *= 0.97

        max_steps -= 1

    return best.tolist(), best_score

            best_seq = a.tolist()
            print(f"  New best (rand)  score={best_val:.6f}")

    # ------------------------------------------------------------------
    # Phase 1 – alternating LP refinement (SciPy)
    # ------------------------------------------------------------------
    if SCIPY_OK and best_seq is not None and time.time() < deadline - 120.0:
        print("\nPhase 1 – alternating LP refinement")
        a = np.array(best_seq, dtype=float)
        a = project_to_simplex(a)
        lp_iter = 0
        # a simple schedule: keep mixing factor 0.2, but shrink after a few passes
        mix = 0.2

        while time.time() < deadline - 80.0 and lp_iter < 12:
            conv = convolution_fft(a, a)
            M = float(np.max(conv))

            b = solve_lp_scipy(a, M)
            if b is None:
                print("  LP solve failed – exiting LP loop")
                break

            # normalise b to the simplex and mix
            b = b / np.sum(b)
            a = (1 - mix) * a + mix * b
            a = project_to_simplex(a)

            cur_val = compute_score(a)
            if cur_val < best_val:
                best_val = cur_val
                best_seq = a.tolist()
                print(f"  LP iter {lp_iter:02d} improved score={best_val:.6f}")

            # slowly decay the mixing parameter to fine‑tune later
            mix = max(mix * 0.85, 0.05)
            lp_iter += 1

    # ------------------------------------------------------------------
    # Phase 2 – sub‑gradient refinement
    # ------------------------------------------------------------------
    if best_seq is not None and time.time() < deadline - 30.0:
        print("\nPhase 2 – sub‑gradient refinement")
        a0 = np.array(best_seq, dtype=float)
        refined_a, refined_score = subgrad_optimize(
            a0,
            time_deadline=deadline - 10.0,
            max_iter=200_000,
            step0=0.07,
        )
        if refined_score < best_val:
            best_val = refined_score
            best_seq = refined_a.tolist()
            print(f"  Sub‑gradient improved best to {best_val:.6f}")

    # ------------------------------------------------------------------
    # Phase 3 – final hill‑climb polishing
    # ------------------------------------------------------------------
    if best_seq is not None and time.time() < deadline - 5.0:
        print("\nPhase 3 – hill‑climb polishing")
        polished_seq, polished_score = hill_climb(
            best_seq,
            time_deadline=deadline,
            max_steps=300_000,
        )
        if polished_score < best_val:
            best_val = polished_score
            best_seq = polished_seq
            print(f"  Hill‑climb improved best to {best_val:.6f}")

    # ------------------------------------------------------------------
    # Fallback (should never happen)
    # ------------------------------------------------------------------
    if best_seq is None:
        print("Fallback – uniform vector")
        best_seq = [1.0 / n] * n
        best_val = compute_score(np.array(best_seq))

    # Final official evaluation (the harness will also call evaluate_sequence)
    final_score = compute_score(np.array(best_seq))
    print(f"\nSearch finished. Final reported score = {final_score:.6f}")

    # Return a plain Python list as required by the harness
    return best_seq
