def build_convolution_matrix(f):
    n = len(f)
    m = 2*n - 1
    rows = []
    cols = []
    data = []
    for k in range(m):
        # for g_j where 0<=j<n, coefficient is f_{k-j} if in range

"""
Hybrid LP‑+‑smooth‑gradient optimisation for the max‑convolution step‑function.

Goal
----
Find a non‑negative vector `a` (sum(a) > 0) that minimises

    score(a) = 2·n·max_k (a∗a)_k / (sum(a)²)

where `∗` denotes linear convolution.  The scale‑invariance of the objective
allows us to keep `sum(a)=1` throughout the search.

Algorithm
---------
1. **Initial seeds** – a few diverse families (uniform, arcsine, Dirichlet,
   power‑law, perturbed versions).  All seeds are normalised to sum = 1.

2. **LP‑based Frank‑Wolfe step** – for a current vector `f` we compute
   `M = max(f∗f)`.  Keeping this bound, we solve

        maximise   Σ_j g_j
        subject to  (f ∗ g)_k ≤ M   for all k,
                    g_j ≥ 0.

   The linear programme is solved with `scipy.optimize.linprog` (HiGHS
   solver) on a sparse convolution matrix.  The optimal `g` is scaled to
   sum = 1 and mixed with `f` using a diminishing step size `t = 2/(i+2)`.
   Repeating a handful of times pushes the mass into a shape that can
   achieve the same bound with larger ℓ¹‑norm – this is exactly the
   “active‑set” improvement used in the state‑of‑the‑art method but here
   we obtain it by an exact LP instead of a heuristic linear programme.

3. **Smooth‑max gradient descent** – after the LP phase we refine the
   vector with an exponentiated‑gradient scheme on the log‑sum‑exp
   surrogate

        Φ_β(a) = (1/β)·log Σ_k exp(β·(a∗a)_k)

   The gradient is a cross‑correlation that can be evaluated with a
   single FFT, giving O(n·log n) per iteration.  We increase `β`
   geometrically (5 → 480) and run enough inner iterations to fully
   exploit the remaining time budget.

4. **Exact sub‑gradient polishing** – finally we switch to the true
   non‑smooth objective and perform exponentiated‑gradient descent on
   the sub‑gradient of `max(a∗a)`.  This usually brings a small but
   consistent improvement.

5. **Upsampling** – the LP phase is cheap for moderate lengths
   (n=256 – 512).  After optimisation we linearly interpolate the best
   vector to a larger length (2048 or 4096) before the smooth‑gradient
   and sub‑gradient stages; this yields finer control of the shape
   without paying the LP cost at the final resolution.

All stages respect the global wall‑clock budget (default 1000 s) and
always return a feasible list of non‑negative floats.

"""

import time
import numpy as np
from typing import List, Tuple
from scipy.optimize import linprog
from scipy.sparse import csr_matrix

# ----------------------------------------------------------------------
# Basic convolution utilities (FFT based, O(n log n))
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full linear convolution using a real FFT (zero‑padded to a power of two)."""
    m = len(a) + len(b) - 1
    n_fft = next_pow2(m)
    A = np.fft.rfft(a, n=n_fft)
    B = np.fft.rfft(b, n=n_fft)
    C = A * B
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


def compute_score(a: np.ndarray) -> float:
    """
    Fast version of the official objective (scale‑invariant):
        2·n·max(a∗a) / (sum(a)²)
    The vector a is assumed non‑negative.
    """
    n = len(a)
    conv = convolution_fft(a, a)
    max_val = float(np.max(conv))
    s = float(np.sum(a))
    # Defensive guard – should never happen for valid inputs
    if s < 1e-12:
        return float("inf")
    return 2.0 * n * max_val / (s * s)


# ----------------------------------------------------------------------
# Families of initial shapes
# ----------------------------------------------------------------------


def arcsine_initial(n: int) -> np.ndarray:
    """Discrete arcsine density (regularised, normalised to sum = 1)."""
    i = np.arange(n, dtype=float)
    eps = 0.25
    vals = 1.0 / np.sqrt((i + 0.5) * (n - 0.5 - i) + eps)
    return vals / vals.sum()


def generate_powerlaw(n: int, p: float, eps: float) -> np.ndarray:
    """Symmetric power‑law a_i ∝ ((i+ε)(n‑1‑i+ε))⁻ᵖ."""
    i = np.arange(n, dtype=float)
    base = ((i + eps) * (n - 1 - i + eps)) ** (-p)
    total = base.sum()
    if total == 0.0:
        return np.full(n, 1.0 / n)
    return base / total


def random_dirichlet(n: int, concentration: float, rng: np.random.Generator) -> np.ndarray:
    """Sample a Dirichlet‑like vector and normalise to sum = 1."""
    if concentration <= 0:
        samp = rng.exponential(1.0, size=n) ** (1.0 / max(0.1, concentration))
    else:
        samp = rng.gamma(concentration, 1.0, size=n)
    samp += 1e-12
    return samp / samp.sum()


def perturb(a: np.ndarray, rng: np.random.Generator, scale: float = 0.02) -> np.ndarray:
    """Add a tiny Gaussian perturbation and renormalise."""
    b = a + rng.normal(0.0, scale, size=a.shape)
    b = np.maximum(b, 0.0)
    s = b.sum()
    if s == 0.0:
        return np.full_like(a, 1.0 / len(a))
    return b / s


# ----------------------------------------------------------------------
# Construction of the (f ∗ g) ≤ M linear constraints (sparse matrix)
# ----------------------------------------------------------------------


def build_convolution_matrix(f: np.ndarray) -> csr_matrix:
    """
    Build the (2n‑1)×n matrix A such that (A @ g) = f ∗ g.
    Each row corresponds to a shift k and contains the coefficients f_{k‑j}
    where they are defined.
    """
    n = len(f)
    m = 2 * n - 1
    # Build dense coefficient arrays via double loop (n is ≤ 512 for LP stage)
    rows = []
    cols = []
    data = []
    for i in range(n):
        fi = float(f[i])
        row_start = i
        for j in range(n):
            rows.append(row_start + j)
            cols.append(j)
            data.append(fi)
    A = csr_matrix((data, (rows, cols)), shape=(m, n), dtype=np.float64)
    return A


# ----------------------------------------------------------------------
# LP‑based Frank‑Wolfe improvement
# ----------------------------------------------------------------------


def lp_frank_wolfe_step(f: np.ndarray,
                        max_iter: int,
                        deadline: float) -> np.ndarray:
    """
    Perform a few Frank‑Wolfe–type steps where the search direction is
    obtained from an exact linear programme:

        maximise Σ g_j   s.t.   (f ∗ g)_k ≤ M  ∀k,   g ≥ 0

    The returned vector is the final iterate (still normalised to sum = 1).
    """
    n = len(f)
    a = f.copy()
    for it in range(1, max_iter + 1):
        if time.time() >= deadline:
            break

        # current bound M = max_k (a∗a)_k
        conv = convolution_fft(a, a)
        M = float(conv.max())

        # build convolution matrix for current a
        A = build_convolution_matrix(a)

        # linear programme: maximise sum(g)  ⇔ minimise -sum(g)
        c = -np.ones(n, dtype=np.float64)                # objective
        b_ub = M * np.ones(2 * n - 1, dtype=np.float64) # RHS of constraints
        bounds = [(0.0, None)] * n                     # g_j >= 0

        # solve with HiGHS (fast for our moderate sizes)
        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds,
                      method='highs', options={'presolve': True, 'time_limit': max(0.0, deadline - time.time())})

        if not res.success:
            # LP failed – abort the LP phase for this vector
            break

        g = res.x
        sum_g = g.sum()
        if sum_g <= 1e-15:
            break

        # normalise to the same ℓ¹‑norm (we keep sum = 1 throughout)
        g = g / sum_g

        # Frank‑Wolfe diminishing step size
        t = 2.0 / (it + 2.0)

        # convex combination
        a = (1.0 - t) * a + t * g

    # final renormalisation (numerical safety)
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s == 0.0:
        a = np.full_like(a, 1.0 / n)
    else:
        a = a / s
    return a


# ----------------------------------------------------------------------
# Smooth‑max gradient descent (log‑sum‑exp surrogate)
# ----------------------------------------------------------------------


def softmax_gradient_descent(initial_a: np.ndarray,
                             deadline: float,
                             betas: List[float],
                             iter_per_beta: int = 500,
                             eta0: float = 0.5) -> Tuple[np.ndarray, float]:
    """
    Minimise a smooth approximation of the max‑convolution:

        Φ_β(a) = (1/β)·log Σ_k exp(β·(a∗a)_k)

    An exponentiated‑gradient update (multiplicative) maintains non‑negativity;
    a simple renormalisation enforces the simplex constraint.
    """
    a = np.maximum(initial_a, 1e-30)
    a = a / a.sum()
    best_a = a.copy()
    best_sc = compute_score(a)

    for beta in betas:
        for it in range(1, iter_per_beta + 1):
            if time.time() >= deadline:
                return best_a, best_sc

            # linear convolution a∗a
            conv = convolution_fft(a, a)                     # length 2n‑1
            max_c = conv.max()
            # stable soft‑max weights
            exp_vals = np.exp(beta * (conv - max_c))
            p = exp_vals / exp_vals.sum()                    # Σ p = 1

            # cross‑correlation p ⋆ a  (size n)
            N = next_pow2(len(p) + len(a) - 1)
            P = np.fft.fft(p, N)
            A_f = np.fft.fft(a, N)
            cross = np.fft.ifft(P * np.conj(A_f)).real[:len(a)]
            cross = np.maximum(cross, 0.0)

            # gradient of the smooth surrogate
            grad = 4.0 * len(a) * cross  # non‑negative

            # exponentiated‑gradient step with diminishing η
            eta = eta0 / np.sqrt(it)
            a = a * np.exp(-eta * grad)

            # project back onto simplex
            a_sum = a.sum()
            if a_sum == 0.0:
                a = np.full_like(a, 1.0 / len(a))
            else:
                a = a / a_sum

            cur_sc = compute_score(a)
            if cur_sc < best_sc:
                best_sc = cur_sc
                best_a = a.copy()

    return best_a, best_sc


# ----------------------------------------------------------------------
# Exact max‑convolution sub‑gradient (final polishing)
# ----------------------------------------------------------------------


def _max_conv_gradient(a: np.ndarray, conv: np.ndarray) -> np.ndarray:
    """
    Sub‑gradient of max_k (a∗a)_k.
    For each active shift k (where the max is attained) the contribution to
    the gradient at index j is 2·a_{k‑j} (if the index is valid).  Contributions
    from multiple active shifts are averaged.
    """
    n = len(a)
    max_val = conv.max()
    mask = np.isclose(conv, max_val, atol=1e-15, rtol=0.0)
    active = np.where(mask)[0]
    grad = np.zeros_like(a)

    for k in active:
        lo = max(0, k - (n - 1))
        hi = min(k, n - 1)
        idx = np.arange(lo, hi + 1, dtype=int)
        grad[idx] += 2.0 * a[k - idx]

    if len(active) > 1:
        grad /= float(len(active))
    return grad


def optimise_subgradient(initial_a: np.ndarray,
                         deadline: float,
                         *,
                         eta0: float = 0.6,
                         max_iter: int = 200_000) -> Tuple[np.ndarray, float]:
    """
    Exponentiated sub‑gradient descent directly on Φ(a)=max_k (a∗a)_k.
    Returns the best vector seen and its (true) fast score.
    """
    a = np.maximum(initial_a, 1e-30)
    a = a / a.sum()
    best_a = a.copy()
    best_sc = compute_score(a)
    n = len(a)

    for it in range(1, max_iter + 1):
        if time.time() >= deadline:
            break

        conv = convolution_fft(a, a)
        grad = _max_conv_gradient(a, conv)

        eta = eta0 / np.sqrt(it)
        a = a * np.exp(-eta * grad)

        # simplex projection
        s = a.sum()
        a = a / s if s != 0.0 else np.full_like(a, 1.0 / n)

        cur_sc = compute_score(a)
        if cur_sc < best_sc:
            best_sc = cur_sc
            best_a = a.copy()

        # occasional progress report (does not affect timing)
        if it % 2000 == 0:
            print(f"    subgrad iter={it:,d} cur={cur_sc:.6f} best={best_sc:.6f}")

    return best_a, best_sc

# Main entry point required by the competition harness

def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs) -> List[float]:
    """
    Search for a non‑negative step‑function with a low ``evaluate_sequence``
    value.  The routine respects a wall‑clock budget (default 1000 s) and
    always returns a feasible list of floats.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    # safety margin for the final call to evaluate_sequence
    deadline = start_time + budget_s - 2.0

    best_seq: List[float] = []
    best_val: float = float("inf")

    # ------------------------------------------------------------------
    # 0. Optional fallback from a previously discovered construction.
    # ------------------------------------------------------------------
    try:
        fallback = [float(x) for x in height_sequence_1]  # type: ignore
        val = evaluate_sequence(fallback)
        print(f"Fallback (height_sequence_1) score = {val:.6f}")
        if val < best_val:
            best_val = val
            best_seq = fallback
    except NameError:
        pass

    # ------------------------------------------------------------------
    # 1.  Coarse optimisation on modest lengths (LP stage).
    # ------------------------------------------------------------------
    lp_lengths = [256, 512]               # cheap to solve LP on
    target_lengths = [2048, 4096]         # final refinement resolution

    for n in lp_lengths:
        if time.time() >= deadline - 30.0:
            break
        print(f"\n=== LP phase, base length n = {n} ===")

        # Build a diverse pool of seed vectors (all normalised to sum = 1)
        seeds: List[np.ndarray] = []

        # (i) uniform
        seeds.append(np.full(n, 1.0 / n))

        # (ii) arcsine
        seeds.append(arcsine_initial(n))

        # (iii) random Dirichlet draws
        for conc in [0.2, 0.5, 1.0, 2.0]:
            seeds.append(random_dirichlet(n, conc, rng))

        # (iv) random power‑law shapes
        for _ in range(4):
            p = rng.uniform(0.2, 1.5)
            eps = rng.uniform(0.001, 0.05)
            seeds.append(generate_powerlaw(n, p, eps))

        # (v) perturbed versions of the above (helps escape tiny basins)
        perturbed = [perturb(s, rng, scale=0.05) for s in seeds[:]]
        seeds.extend(perturbed)

        # ------------------------------------------------------------------
        #   Run a few LP‑based Frank‑Wolfe steps for each seed.
        # ------------------------------------------------------------------
        for sid, seed_vec in enumerate(seeds):
            if time.time() >= deadline - 30.0:
                break
            print(f"  seed {sid + 1}/{len(seeds)} (n={n})")
            a = seed_vec.copy()
            a = np.maximum(a, 0.0)
            a = a / a.sum()

            # number of LP iterations – a small constant is sufficient
            a = lp_frank_wolfe_step(a, max_iter=15, deadline=deadline)

            cur_sc = compute_score(a)
            val = evaluate_sequence(a.tolist())
            print(f"    after LP phase score = {val:.6f}")

            if val < best_val:
                best_val = val
                best_seq = a.tolist()
                print(f"    *** New global best after LP {best_val:.6f} ***")

            # ----------------------------------------------------------------
            #   Upsample to the target resolution(s) and perform fine‑tuning.
            # ----------------------------------------------------------------
            for N in target_lengths:
                if time.time() >= deadline - 15.0:
                    break
                # linear interpolation (keeps the overall shape)
                x_old = np.linspace(0.0, 1.0, n)
                x_new = np.linspace(0.0, 1.0, N)
                a_big = np.interp(x_new, x_old, a)
                a_big = np.maximum(a_big, 0.0)
                a_big = a_big / a_big.sum()

                # --- smooth‑max gradient descent ---------------------------------
                remaining = max(0.0, deadline - time.time())
                # allocate iterations proportional to remaining time
                it_per_beta = max(200, int(min(2000, remaining * 250) / 7))
                betas = [5, 15, 30, 60, 120, 240, 480]

                a_big, _ = softmax_gradient_descent(a_big,
                                                    deadline,
                                                    betas=betas,
                                                    iter_per_beta=it_per_beta,
                                                    eta0=0.4)

                val2 = evaluate_sequence(a_big.tolist())
                print(f"    after smooth‑gd (N={N}) score = {val2:.6f}")

                if val2 < best_val:
                    best_val = val2
                    best_seq = a_big.tolist()
                    print(f"    *** New global best after smooth‑gd {best_val:.6f} ***")

                # --- final sub‑gradient polishing --------------------------------
                if time.time() < deadline - 5.0:
                    remaining = max(0.0, deadline - time.time())
                    max_iter = int(min(200_000, remaining * 300))
                    a_refined, _ = optimise_subgradient(a_big,
                                                        deadline,
                                                        eta0=0.3,
                                                        max_iter=max_iter)
                    final_val = evaluate_sequence(a_refined.tolist())
                    print(f"    final polished (N={N}) score = {final_val:.6f}")
                    if final_val < best_val:
                        best_val = final_val
                        best_seq = a_refined.tolist()
                        print(f"    *** Updated global best after polishing {best_val:.6f} ***")

    # ------------------------------------------------------------------
    # 2.  Guard – always return a feasible sequence.
    # ------------------------------------------------------------------
    if not best_seq:
        best_seq = [1.0] * 256
        best_val = evaluate_sequence(best_seq)

    print(f"\nSearch finished. Best reported score = {best_val:.6f}")
    return best_seq
