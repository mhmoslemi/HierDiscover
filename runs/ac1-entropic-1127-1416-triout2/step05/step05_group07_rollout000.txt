"""
Propose a low‑score step‑function by exploiting the discrete arcsine
distribution.  The arcsine density has the property that the sum of two
independent copies is (asymptotically) uniform, which drives the
objective 2·n·max(conv)/sum² → 1 as the length grows.  The algorithm
therefore

1. builds a moderate‑size arcsine shape,
2. refines it with a cheap exponentiated‑gradient (mirror‑descent) sub‑
   gradient loop,
3. upsamples the refined shape to a much larger length (≈ 1.6·10⁴)
   where the discretisation error is already well below the target
   1.503,
4. performs a short final polishing step,
5. finally evaluates the candidate with the official ``evaluate_sequence``
   function (the only expensive call).

All work respects the supplied wall‑clock budget; the routine always
returns a feasible non‑negative list.
"""

import time
import numpy as np
from typing import List, Tuple

# ----------------------------------------------------------------------
# Helper utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full linear convolution via real FFT (fast and accurate)."""
    m = len(a) + len(b) - 1
    n_fft = next_pow2(m)
    A = np.fft.rfft(a, n=n_fft)
    B = np.fft.rfft(b, n=n_fft)
    C = A * B
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


def compute_score(a: np.ndarray) -> float:
    """
    Fast surrogate for the official evaluator.
    Returns 2·n·max(a*a)/sum(a)² (the scaling factor 2·n is kept,
    the denominator is 1 since we keep a on the simplex).
    """
    n = len(a)
    conv = convolution_fft(a, a)
    max_val = float(conv.max())
    s = float(a.sum())
    if s < 1e-12:
        return float("inf")
    return 2.0 * n * max_val / (s * s)


def arcsine_initial(n: int, eps: float = 1e-12) -> np.ndarray:
    """
    Discrete analogue of the arcsine law:
        a_i ∝ 1/√((i+½)(n‑½‑i)).
    The tiny epsilon avoids division‑by‑zero at the ends.
    The vector is normalised to sum to 1 (probability simplex).
    """
    i = np.arange(n, dtype=float)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - 0.5 - i) + eps)
    vals /= vals.sum()
    return vals


def upsample(prev: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation from len(prev) → new_n followed by renormalisation."""
    x_old = np.linspace(0.0, 1.0, len(prev))
    x_new = np.linspace(0.0, 1.0, new_n)
    interp = np.interp(x_new, x_old, prev)
    total = interp.sum()
    if total == 0.0:
        return np.full(new_n, 1.0 / new_n)
    return interp / total


def perturb(a: np.ndarray, rng: np.random.Generator, scale: float = 0.02) -> np.ndarray:
    """Add small Gaussian jitter and renormalise onto the simplex."""
    b = a + rng.normal(0.0, scale, size=a.shape)
    b = np.maximum(b, 0.0)
    if b.sum() == 0.0:
        return np.full_like(a, 1.0 / len(a))
    return b / b.sum()


def _max_conv_gradient(a: np.ndarray, conv: np.ndarray) -> np.ndarray:
    """
    Sub‑gradient of max_k (a∗a)_k.
    We pick a single maximising offset (the first argmax) and set
        g_i = 2·a_{k‑i}   for indices that are in range.
    This yields a valid sub‑gradient for the non‑smooth max.
    """
    n = len(a)
    k = int(conv.argmax())                  # first maximiser
    lo = max(0, k - (n - 1))
    hi = min(k, n - 1)
    idx = np.arange(lo, hi + 1, dtype=int)  # indices i where a_i participates
    grad = np.zeros_like(a)
    grad[idx] = 2.0 * a[k - idx]
    return grad


def optimise_subgradient(
    init_a: np.ndarray,
    deadline: float,
    *,
    eta0: float = 0.6,
    max_iter: int = 200_000,
    noise_every: int = 0,
    noise_scale: float = 0.0,
) -> Tuple[np.ndarray, float]:
    """
    Exponentiated‑gradient (mirror‑descent) on the non‑smooth objective
    Φ(a)=max_k (a∗a)_k  with the simplex constraint.
    Returns the best vector seen together with its surrogate score.
    """
    a = np.maximum(init_a, 1e-30)
    a = a / a.sum()                     # enforce simplex
    best_a = a.copy()
    best_sc = compute_score(a)

    n = len(a)

    for it in range(1, max_iter + 1):
        # time check
        if time.time() >= deadline:
            break

        conv = convolution_fft(a, a)
        grad = _max_conv_gradient(a, conv)

        eta = eta0 / np.sqrt(it)                # classic 1/√t schedule
        # multiplicative (exponentiated) update keeps positivity
        a = a * np.exp(-eta * grad)

        if noise_every and (it % noise_every == 0):
            a += np.random.randn(n) * noise_scale
            a = np.maximum(a, 0.0)

        # project back onto the simplex (renormalise)
        s = a.sum()
        if s == 0.0:
            a = np.full_like(a, 1.0 / n)
        else:
            a = a / s

        cur_sc = compute_score(a)
        if cur_sc < best_sc:
            best_sc = cur_sc
            best_a = a.copy()

        # occasional progress report (does not affect timing)
        if it % 5000 == 0:
            print(f"      subgrad it={it:,d} cur={cur_sc:.6f} best={best_sc:.6f}")

    return best_a, best_sc

# ----------------------------------------------------------------------

    wall‑clock budget and always returns a feasible list.
    """
    rng = np.random.default_rng(seed)
    start = time.time()
    deadline = start + budget_s - 1.0          # safety margin

    best_seq: List[float] = []
    best_val: float = float("inf")
    best_internal: np.ndarray = None          # cached best vector (normalised)

    except Exception:

    small_lengths = [512, 1024, 2048, 4096]
