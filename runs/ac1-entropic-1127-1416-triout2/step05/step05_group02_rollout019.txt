    L = 2*n-1

    v = np.zeros_like(a) # second moment

"""
Propose a low‑discrepancy step sequence.
-------------------------------------------------
This version explores a different optimisation scheme:
* initialise several seeds (uniform, arcsine, random Dirichlet);
* run a smooth‑max Adam optimiser (soft‑max of the convolution) that
  directly follows the true max‑convolution objective;
* finish with a short exact sub‑gradient pass (as in the reference
  implementation) and an optional exponentiated‑gradient polish.
The smooth‑max surrogate (soft‑max with growing β) gives a better
gradient signal than the binary active‑set used by plain sub‑gradient
descent and often finds a slightly lower value of
`evaluate_sequence`.  The whole search respects the 1000 s budget.
"""

import time
import numpy as np
from typing import List, Tuple

# --------------------------------------------------------------
# Simple utilities
# --------------------------------------------------------------

def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Linear convolution via real FFT (full length)."""
    m = a.size + b.size - 1
    n_fft = next_pow2(m)
    A = np.fft.rfft(a, n=n_fft)
    B = np.fft.rfft(b, n=n_fft)
    C = A * B
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection onto the probability simplex {x ≥ 0, Σx = 1}.
    Duchi et al., 2008.
    """
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / v.size)

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho_arr = u - cssv / np.arange(1, v.size + 1)
    rho = np.where(rho_arr > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def compute_score(a: np.ndarray) -> float:
    """Fast surrogate for the official score (Σa = 1)."""
    n = a.shape[0]
    conv = convolution_fft(a, a)
    return 2.0 * n * float(np.max(conv))


# --------------------------------------------------------------
# Shape generators (used for seeding)
# --------------------------------------------------------------

def arcsine_initial(n: int) -> np.ndarray:
    """Classic arcsine shape: a_i ∝ 1/√((i+½)*(n‑½‑i))."""
    i = np.arange(n, dtype=float)
    eps = 0.25
    vals = 1.0 / np.sqrt((i + 0.5) * (n - 0.5 - i) + eps)
    vals /= vals.sum()
    return vals


def random_dirichlet(n: int, alpha: float = 1.0) -> np.ndarray:
    """Draw a random point of the simplex (Gamma/Dirichlet)."""
    x = np.random.gamma(alpha, 1.0, size=n)
    return x / x.sum()

# Optimisation primitives
# --------------------------------------------------------------

def adam_smoothmax(a: np.ndarray,
                   deadline: float,
                   max_iter: int = 200_000,
                   beta_start: float = 2.0,
                   beta_end: float = 30.0,
                   lr: float = 0.05) -> np.ndarray:
    """
    Adam optimiser on a smooth‑max surrogate of the max‑convolution.
    The surrogate is   softmax_β(conv) = Σ conv_k * exp(β conv_k) / Σ exp(β conv_k).
    As β → ∞ the surrogate converges to max(conv).
    """
    n = a.shape[0]
    L = 2 * n - 1
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1, beta2 = 0.9, 0.999
    eps = 1e-12

    for it in range(1, max_iter + 1):
        if time.time() >= deadline:
            break

        # linear schedule for β (smoothness → true max)
        beta = beta_start + (beta_end - beta_start) * (it / max_iter)

        conv = convolution_fft(a, a)

        # stable soft‑max weights
        maxc = np.max(conv)
        w = np.exp(beta * (conv - maxc))
        w_sum = np.sum(w)
        w = w / w_sum

        # gradient of the weighted sum ∑ w_k conv_k
        w_rev = w[::-1]
        conv_rev = convolution_fft(a, w_rev)
        grad = 2.0 * conv_rev[L - 1 - np.arange(n)]

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        step = lr / (np.sqrt(v_hat) + eps)
        a = a - step * m_hat
        a = project_to_simplex(a)

        if it % 10_000 == 0:
            print(f"    Adam it={it:,d} β={beta:.2f} score={compute_score(a):.6f}")

    return a


def subgradient_descent_exact(initial_a: np.ndarray,
                              deadline: float,
                              max_iter: int = 30_000,
                              step0: float = 0.10) -> np.ndarray:
    """
    A short exact sub‑gradient pass (as in the reference code) to
    polish the solution obtained by Adam.
    """
    a = project_to_simplex(initial_a.copy())
    n = len(a)
    L = 2 * n - 1

    for it in range(1, max_iter + 1):
        if time.time() >= deadline:
            break

        conv = convolution_fft(a, a)
        max_val = float(np.max(conv))
        tol = max_val * 1e-12 + 1e-14
        mask = (np.abs(conv - max_val) <= tol).astype(float)

        mask_rev = mask[::-1]
        conv_rev = convolution_fft(a, mask_rev)
        subgrad = 2.0 * conv_rev[L - 1 - np.arange(n)]

        eta = step0 / np.sqrt(it)
        a = a - eta * subgrad
        a = project_to_simplex(a)

        if it % 5_000 == 0:
            print(f"    SubGrad it={it:,d} score={compute_score(a):.6f}")

    return a


def eg_refine(a: np.ndarray,
              deadline: float,
              max_iter: int = 5_000,
              eta0: float = 0.30) -> np.ndarray:
    """
    Exponentiated‑gradient polishing (mirror‑descent on the simplex).
    """
    n = a.shape[0]
    L = 2 * n - 1

    for it in range(1, max_iter + 1):
        if time.time() >= deadline:
            break

        conv = convolution_fft(a, a)
        max_val = float(np.max(conv))
        tol = max_val * 1e-12 + 1e-14
        mask = (np.abs(conv - max_val) <= tol).astype(float)

        mask_rev = mask[::-1]
        conv_rev = convolution_fft(a, mask_rev)
        subgrad = 2.0 * conv_rev[L - 1 - np.arange(n)]

        eta = eta0 / (np.sqrt(it) + 1.0)
        a = a * np.exp(-eta * subgrad)

        total = a.sum()
        if total == 0.0:
            a = np.full_like(a, 1.0 / n)
        else:
            a = a / total

        if it % 2_500 == 0:
            print(f"    EG it={it:,d} score={compute_score(a):.6f}")

    return a

    # 0.  Optional fallback from previous runs (if available)
    # ------------------------------------------------------------------
    if "height_sequence_1" in globals():
        try:
            fallback = [float(x) for x in globals()["height_sequence_1"]]
            val = evaluate_sequence(fallback)
            print(f"Fallback (height_sequence_1) score = {val:.6f}")
            if val < best_val:
                best_val = val
                best_seq = fallback
        except Exception:
            pass

    # ------------------------------------------------------------------
    # 1.  Choose a single length that offers a good trade‑off.
    #     (Experiments show n=1024 works well for the budget.)
    # ------------------------------------------------------------------
    n = 1024
    print(f"\n=== Optimising length n = {n} ===")

    # ------------------------------------------------------------------
    # 2.  Build a small pool of diverse seeds.
    # ------------------------------------------------------------------
    seed_vectors: List[np.ndarray] = []

    # deterministic starters
    seed_vectors.append(np.full(n, 1.0 / n))      # uniform
    seed_vectors.append(arcsine_initial(n))      # arcsine‑like

    # random Dirichlet seeds (different concentration)
    for a in (0.2, 0.5, 1.0, 2.0):
        for _ in range(5):
            if time.time() >= deadline - 5.0:
                break
            seed_vectors.append(random_dirichlet(n, alpha=a))

    # ------------------------------------------------------------------
    # 3.  Optimise each seed with Adam (smooth‑max) then polish.
    # ------------------------------------------------------------------
    for idx, a0 in enumerate(seed_vectors, start=1):
        if time.time() >= deadline - 5.0:
            break
        print(f"\nOptimising seed {idx}/{len(seed_vectors)} …")

        # Adam optimiser on the smoothed max‑convolution
        remaining = deadline - time.time()
        max_iter_adam = int(min(150_000, remaining * 40))
        a = adam_smoothmax(a0, deadline,
                           max_iter=max_iter_adam,
                           beta_start=2.0,
                           beta_end=25.0,
                           lr=0.05)

        # Short exact sub‑gradient refinement
        if time.time() < deadline - 2.0:
            remaining = deadline - time.time()
            max_iter_sub = int(min(30_000, remaining * 30))
            a = subgradient_descent_exact(a, deadline,
                                          max_iter=max_iter_sub,
                                          step0=0.12)

        # Final exponentiated‑gradient polish (lightweight)
        if time.time() < deadline - 1.0:
            remaining = deadline - time.time()
            max_iter_eg = int(min(8_000, remaining * 30))
            a = eg_refine(a, deadline,
                          max_iter=max_iter_eg,
                          eta0=0.25)

        # Fast surrogate score – keep the best candidate
        sc = compute_score(a)
        print(f"    → surrogate score = {sc:.6f}")
        if sc < best_val:
            best_seq = a.tolist()
            best_val = sc

    # ------------------------------------------------------------------
    # 4.  Evaluate the best candidate with the official scorer.
    # ------------------------------------------------------------------
    try:
        official_val = evaluate_sequence(best_seq)
        print(f"\nOfficial evaluation of current best = {official_val:.6f}")
        if official_val < best_val:
            best_val = official_val
    except Exception as e:
        print(f"Official evaluation failed: {e}")

    # ------------------------------------------------------------------
    # 5.  Guard – guarantee a non‑empty return value.
    # ------------------------------------------------------------------
    if not best_seq:
        fallback_n = 256
        best_seq = [1.0 / fallback_n] * fallback_n
        best_val = evaluate_sequence(best_seq)

    print(f"\nSearch finished. Best reported official score = {best_val:.6f}")
    return best_seq
