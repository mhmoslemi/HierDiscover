def adam_optimize(a0, betas, deadline, total_iters_per_beta, lr_base):
    a = a0.copy()
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    t = 0
    best_a = a.copy()
    best_sc = compute_score(a)
    for beta in betas:
        # maybe adjust learning rate scaling with beta
        lr = lr_base / np.sqrt(beta+1)
        for _ in range(total_iters_per_beta):
            if time.time() >= deadline:
                break
            t += 1
            # conv
            conv = convolution_fft(a, a)
            # compute weighted w
            beta_conv = beta * conv
            max_beta = np.max(beta_conv)
            exp_terms = np.exp(beta_conv - max_beta)  # stable
            w = exp_terms / np.sum(exp_terms)
            # gradient
            grad = 2.0 * correlation_fft(a, w)  # length n
            # Adam updates
            m = 0.9 * m + 0.1 * grad
            v = 0.999 * v + 0.001 * (grad * grad)
            m_hat = m / (1 - 0.9**t)
            v_hat = v / (1 - 0.999**t)
            step = lr * m_hat / (np.sqrt(v_hat) + 1e-8)
            a = a - step
            # project onto simplex
            a = project_simplex(a)
            # track best using surrogate (compute_score)
            cur_sc = compute_score(a)
            if cur_sc < best_sc:
                best_sc = cur_sc
                best_a = a.copy()
            # occasional print
            if t % 2000 == 0:
                print(f"      iter {t} beta {beta} cur={cur_sc:.6f} best={best_sc:.6f}")
    return best_a, best_sc

def project_simplex(v):
    """Project vector v onto the probability simplex {x >= 0, sum x = 1}."""
    # Works for any shape.
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(len(v)) + 1
    rho = np.where(u - cssv / ind > 0)[0]
    if len(rho) == 0:
        theta = 0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0)
    return w

def correlation_fft(a, w):

    start = m - 1

"""
Propose a low‑score step‑function.

The algorithm works hierarchically.  For a modest size (≈ 512–4096) we
start from a classic arcsine shape, optionally perturb it, and from a few
random Dirichlet draws.  Each start is refined with a *smooth* Adam
optimiser that minimises a log‑sum‑exp surrogate of the maximal
self‑convolution.  The surrogate becomes sharper (larger β) during the
run – a continuation method that lets the optimiser first move quickly
and then focus on the true max.  After the smooth phase we finish with a
few thousand iterations of the original exponentiated‑gradient
sub‑gradient method (the “sub‑grad” code used in the reference) to
polish the actual max‑convolution.

All updates keep the vector on the probability simplex (non‑negative,
unit‑sum) – the objective is scale‑invariant, so the simplex is a
convenient feasible set.  The routine respects a wall‑clock budget,
always returns a feasible list and prints progress and current best
score.
"""

import time
import numpy as np
from typing import List, Tuple

# Basic numeric helpers

def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray) -> np.ndarray:
    """
    Full linear convolution a∗a using a real FFT.
    Returns an array of length 2·len(a)-1.
    """
    n = len(a)
    m = 2 * n - 1
    L = next_pow2(m)
    A = np.fft.rfft(a, L)
    conv = np.fft.irfft(A * A, L)[:m]
    return conv


def correlation_fft(a: np.ndarray, w: np.ndarray) -> np.ndarray:
    """
    Correlation (a ⋆ w) for non‑negative lags.
    Returns an array of length len(a) where
        (a ⋆ w)[j] = Σ_i a_i * w_{i+j}.
    Both inputs are real; the computation is performed via FFT.
    """
    n = len(a)
    m = len(w)
    L = next_pow2(n + m - 1)
    A = np.fft.fft(a, L)
    W = np.fft.fft(w, L)
    corr = np.fft.ifft(A * np.conj(W)).real
    start = m - 1                     # zero‑lag index
    return corr[start:start + n]      # lags 0 … n‑1


def project_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of v onto the probability simplex
    {x | x_i ≥ 0, Σ_i x_i = 1}.
    """
    if np.all(v >= 0) and np.isclose(v.sum(), 1.0):
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(len(v)) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # Numerical fallback – return uniform distribution
        return np.full_like(v, 1.0 / len(v))
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    # re‑normalise to protect against round‑off
    w /= w.sum()
    return w


def compute_score(a: np.ndarray) -> float:
    """
    Fast surrogate for the official evaluator:
        2·n·max(a*a) / (sum(a)²)
    """
    n = len(a)
    conv = convolution_fft(a)
    max_val = float(conv.max())
    s = float(a.sum())
    if s < 1e-12:
        return float("inf")
    return 2.0 * n * max_val / (s * s)


# ----------------------------------------------------------------------
# Families of initial shapes
# ----------------------------------------------------------------------


def arcsine_initial(n: int) -> np.ndarray:
    """
    Discrete arcsine density a_i ∝ 1/√((i+½)(n‑½‑i)).
    Normalised to unit sum.
    """
    i = np.arange(n, dtype=float)
    eps = 0.25
    vals = 1.0 / np.sqrt((i + 0.5) * (n - 0.5 - i) + eps)
    vals /= vals.sum()
    return vals


def random_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    """Dirichlet‑like draw (Γ(α,1) with α=conc).  Conc can be <0 for heavy‑tails."""
    if conc <= 0.0:
        # heavy‑tailed near‑sparse draws
        samp = rng.exponential(1.0, size=n) ** (1.0 / max(0.1, -conc))
    else:
        samp = rng.gamma(conc, 1.0, size=n)
    samp += 1e-12
    return samp / samp.sum()


def perturb(a: np.ndarray, rng: np.random.Generator, scale: float = 0.02) -> np.ndarray:
    """Add a tiny Gaussian jitter and renormalise."""
    b = a + rng.normal(0.0, scale, size=a.shape)
    b = np.maximum(b, 0.0)
    if b.sum() == 0.0:
        return np.full_like(a, 1.0 / len(a))
    return b / b.sum()


def upsample(prev: np.ndarray, new_n: int) -> np.ndarray:
    """
    Linear interpolation from len(prev) → new_n, followed by
    renormalisation onto the simplex.
    """
    x_old = np.linspace(0.0, 1.0, len(prev))
    x_new = np.linspace(0.0, 1.0, new_n)
    interp = np.interp(x_new, x_old, prev)
    if interp.sum() == 0.0:
        return np.full(new_n, 1.0 / new_n)
    return interp / interp.sum()

# Smooth Adam optimiser (log‑sum‑exp surrogate)
# ----------------------------------------------------------------------


def _smooth_step(a: np.ndarray, beta: float) -> Tuple[np.ndarray, np.ndarray, np.ndarray]:
    """
    One evaluation of the smooth surrogate:
        - conv   : a∗a
        - w      : softmax(β·conv)   (weights summing to 1)
        - grad   : ∇_a  (1/β)·log Σ exp(β·conv)
    """
    conv = convolution_fft(a)                   # length 2n‑1
    z = beta * conv
    max_z = np.max(z)
    exp_z = np.exp(z - max_z)                  # stable
    w = exp_z / exp_z.sum()                    # softmax weights
    grad = 2.0 * correlation_fft(a, w)         # formula derived in analysis
    return conv, w, grad


def adam_optimize(
    init_a: np.ndarray,
    betas: List[float],
    deadline: float,
    max_iter_per_beta: int,
    lr_base: float = 0.4,
    symmetrize: bool = True,
    noise_scale: float = 0.0,
) -> Tuple[np.ndarray, float]:
    """
    Adam‑style optimisation of the smooth surrogate.
    Returns the best vector (w.r.t. the cheap surrogate score) seen.
    """
    a = init_a.copy()
    n = len(a)
    m = np.zeros_like(a)          # first moment
    v = np.zeros_like(a)          # second moment
    t = 0
    best_a = a.copy()
    best_sc = compute_score(a)

    for beta in betas:
        # a slightly smaller learning‑rate for larger β
        lr = lr_base / np.sqrt(beta + 1.0)

        for _ in range(max_iter_per_beta):
            if time.time() >= deadline:
                return best_a, best_sc

            t += 1
            conv, w, grad = _smooth_step(a, beta)

            if noise_scale > 0.0:
                grad += np.random.randn(n) * noise_scale

            # Adam update
            m = 0.9 * m + 0.1 * grad
            v = 0.999 * v + 0.001 * (grad * grad)
            m_hat = m / (1.0 - 0.9 ** t)
            v_hat = v / (1.0 - 0.999 ** t)

            step = lr * m_hat / (np.sqrt(v_hat) + 1e-8)
            a = a - step

            if symmetrize:
                a = (a + a[::-1]) * 0.5

            a = project_simplex(a)

            cur_sc = compute_score(a)
            if cur_sc < best_sc:
                best_sc = cur_sc
                best_a = a.copy()

            if t % 2000 == 0:
                print(
                    f"        iter {t:>6d}  β={beta:<5g}  cur={cur_sc:.6f}  best={best_sc:.6f}"
                )

    return best_a, best_sc


# ----------------------------------------------------------------------
# Sub‑gradient optimiser (original “smooth‑max” optimiser)
# ----------------------------------------------------------------------


def _max_conv_gradient(a: np.ndarray, conv: np.ndarray) -> np.ndarray:
    """
    Sub‑gradient of max_k (a∗a)_k.
    For each active offset k we add 2·a_{k‑j} to the gradient.
    The average is taken if several offsets are active.
    """
    n = len(a)
    max_val = conv.max()
    mask = np.isclose(conv, max_val, atol=1e-15)
    active = np.where(mask)[0]
    grad = np.zeros_like(a)

    for k in active:
        lo = max(0, k - (n - 1))
        hi = min(k, n - 1)
        idx = np.arange(lo, hi + 1, dtype=int)
        grad[idx] += 2.0 * a[k - idx]

    if len(active) > 1:
        grad /= len(active)
    return grad


def optimise_subgradient(
    init_a: np.ndarray,
    deadline: float,
    *,
    eta0: float = 0.6,
    max_iter: int = 100_000,
    noise_every: int = 0,
    noise_scale: float = 0.0,
) -> Tuple[np.ndarray, float]:
    """
    Exponentiated‑gradient (mirror‑descent) on the non‑smooth objective.
    Returns the best vector (according to the cheap surrogate) seen.
    """
    a = np.maximum(init_a, 1e-30)
    a = a / a.sum()
    best_a = a.copy()
    best_sc = compute_score(a)
    n = len(a)

    for it in range(1, max_iter + 1):
        if time.time() >= deadline:
            break

        conv = convolution_fft(a)
        grad = _max_conv_gradient(a, conv)

        eta = eta0 / np.sqrt(it)
        a = a * np.exp(-eta * grad)

        if noise_every and (it % noise_every == 0):
            a += np.random.randn(n) * noise_scale
            a = np.maximum(a, 0.0)

        a_sum = a.sum()
        if a_sum == 0.0:
            a = np.full_like(a, 1.0 / n)
        else:
            a = a / a_sum

        cur_sc = compute_score(a)
        if cur_sc < best_sc:
            best_sc = cur_sc
            best_a = a.copy()

        if it % 2000 == 0:
            print(f"    subgrad iter={it:,d} cur={cur_sc:.6f} best={best_sc:.6f}")

    return best_a, best_sc


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------


def propose_candidate(budget_s: int = 1000, seed: int = 42) -> List[float]:
    """
    Search for a non‑negative step‑function with a low ``evaluate_sequence``
    value.  The routine works hierarchically: it first optimises modest
    lengths, then repeatedly upsamples and refines the result for larger
    sizes.  Both a smooth Adam optimiser (log‑sum‑exp surrogate) and a
    final sub‑gradient polish are employed.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 2.0          # tiny safety margin

    best_seq: List[float] = []
    best_val: float = float("inf")
    best_internal: np.ndarray = None                # stored on the simplex

    # ------------------------------------------------------------------
    # 0.  Optional fallback from a known good construction.
    # ------------------------------------------------------------------
    try:
        fallback = [float(x) for x in height_sequence_1]   # type: ignore
        val = evaluate_sequence(fallback)
        print(f"Fallback (height_sequence_1) score = {val:.6f}")
        if val < best_val:
            best_val = val
            best_seq = fallback
            best_internal = np.array(fallback, dtype=float)
            best_internal = best_internal / best_internal.sum()
    except Exception:
        pass

    # ------------------------------------------------------------------
    # 1.  Length hierarchy – start small, enlarge progressively.
    # ------------------------------------------------------------------
    lengths = [512, 1024, 2048, 4096, 8192, 16384, 32768]

    for n in lengths:
        if time.time() >= deadline - 5.0:
            break
        print(f"\n=== Exploring length n = {n} ===")

        # --------------------------------------------------------------
        # 1a.  Build a small library of start points.
        # --------------------------------------------------------------
        candidates: List[np.ndarray] = []

        if best_internal is not None:
            # warm‑start from previous best (upsampled)
            up = upsample(best_internal, n)
            candidates.append(up)

        # classic arcsine shape – always a strong baseline
        candidates.append(arcsine_initial(n))

        # a few random Dirichlet draws for diversity
        for conc in (0.2, 0.5, 1.0, 2.0):
            candidates.append(random_dirichlet(n, conc, rng))

        # perturbations of the arcsine shape (helps escape tiny plateaus)
        for _ in range(2):
            cand = perturb(arcsine_initial(n), rng, scale=0.01)
            candidates.append(cand)

        # --------------------------------------------------------------
        # 1b.  Quick surrogate screening.
        # --------------------------------------------------------------
        scored = [(compute_score(c), c) for c in candidates]
        scored.sort(key=lambda x: x[0])
        top_candidates = [c for _, c in scored[:3]]   # keep only the few best

        # --------------------------------------------------------------
        # 1c.  Refine each retained start with the smooth Adam optimiser.
        # --------------------------------------------------------------
        for idx, cand in enumerate(top_candidates):
            if time.time() >= deadline - 5.0:
                break
            print(f"  candidate {idx + 1}/{len(top_candidates)} (n={n})")

            remaining = max(0.0, deadline - time.time())
            # Rough budget: ≈ 250 it/s for n≈8 k (conservative)
            max_iter_total = int(min(150_000, max(5_000, remaining * 250)))
            # split roughly equally among the β‑steps
            betas = [5.0, 20.0, 50.0, 100.0, 200.0]
            it_per_beta = max(1, max_iter_total // len(betas))

            refined, _ = adam_optimize(
                init_a=cand,
                betas=betas,
                deadline=deadline,
                max_iter_per_beta=it_per_beta,
                lr_base=0.4,
                symmetrize=True,
                noise_scale=0.0,
            )

            # official (expensive) evaluation – only for the refined result
            val = evaluate_sequence(refined.tolist())
            print(f"    evaluated score = {val:.6f}")

            if val < best_val:
                best_val = val
                best_seq = refined.tolist()
                best_internal = refined.copy()
                print(f"    *** New global best: {best_val:.6f} (n={n}) ***")

    # ------------------------------------------------------------------
    # 2.  Final polishing with the original sub‑gradient routine.
    # ------------------------------------------------------------------
    if best_seq and time.time() < deadline - 5.0:
        print("\nFinal polishing with sub‑gradient optimiser")
        a = np.array(best_seq, dtype=float)
        remaining = max(0.0, deadline - time.time())
        max_iter = int(min(300_000, remaining * 250))
        a_ref, _ = optimise_subgradient(
            init_a=a,
            deadline=deadline,
            eta0=0.3,
            max_iter=max_iter,
            noise_every=0,
            noise_scale=0.0,
        )
        final_val = evaluate_sequence(a_ref.tolist())
        print(f"Final polished score = {final_val:.6f}")
        if final_val < best_val:
            best_val = final_val
            best_seq = a_ref.tolist()
            print("*** Updated global best after final polish ***")

    # ------------------------------------------------------------------
    # 3.  Guard – always return a feasible sequence.
    # ------------------------------------------------------------------
    if not best_seq:
        best_seq = [1.0] * 256
        best_val = evaluate_sequence(best_seq)

    print(f"\nSearch finished. Best reported score = {best_val:.6f}")
    return best_seq
