"""
Propose a non‑negative step function that minimises

    2·n·max(conv(a,a)) / (sum(a)²)

The search proceeds in three phases:

1️⃣  **Parametric pre‑search** – Randomly explore a few low‑dimensional families
    (edge‑heavy power‑law, arcsine with tunable smoothing, mixtures with a little
    uniform mass).  Each candidate is evaluated with the fast FFT‑based score
    `compute_score`.  The best parameters are kept.

2️⃣  **Smooth mirror‑descent** –  Exponentiated‑gradient updates on the simplex
    using a softened max‑weight (`β` controls the softness).  This quickly
    reduces the objective while keeping the iterates well‑behaved.

3️⃣  **Hard sub‑gradient polishing** –  Classical sub‑gradient descent with the
    true max‑weight (uniform on the maximisers) refines the solution to a
    high‑quality local optimum.  If time permits a second polish is started from
    a slightly perturbed point to escape possible shallow minima.

All stages respect the supplied time budget (`budget_s`).  The routine always
returns a feasible list of non‑negative floats; if nothing is found a uniform
fallback of length 256 is used.
"""

import time
import numpy as np
from typing import List, Tuple, Dict, Any

# ----------------------------------------------------------------------
# Low‑level numeric utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """
    Linear convolution via real FFT (full length, zero‑padded to a power‑of‑2).
    """
    m = len(a) + len(b) - 1
    n_fft = next_pow2(m)
    A = np.fft.rfft(a, n=n_fft)
    B = np.fft.rfft(b, n=n_fft)
    C = A * B
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


def normalize(v: np.ndarray) -> np.ndarray:
    """Project a non‑negative vector onto the probability simplex (sum = 1)."""
    total = v.sum()
    if total <= 0.0:
        # fallback to uniform if something went terribly wrong
        return np.full_like(v, 1.0 / len(v))
    return v / total


def compute_score(a: np.ndarray) -> float:
    """
    Fast evaluation of the official objective (scale‑invariant):
        2·n·max(conv(a,a)) / (sum(a)²)
    The vector a is assumed non‑negative.
    """
    n = len(a)
    conv = convolution_fft(a, a)
    max_val = float(np.max(conv))
    s = float(np.sum(a))
    if s < 1e-12:
        return float("inf")
    return 2.0 * n * max_val / (s * s)


def gradient_from_weights(a: np.ndarray, w: np.ndarray) -> np.ndarray:
    """
    Gradient of Σ_k w_k·(a*a)_k with respect to a.
    Implements   g_j = 2· Σ_m a_m · w_{m+j}.
    Computed via one FFT‑based convolution.
    """
    w_rev = w[::-1]
    conv = convolution_fft(a, w_rev)  # length = 3·n‑2
    idx = 2 * len(a) - 2               # position of the centre term
    grads = 2.0 * conv[idx - np.arange(len(a))]
    return grads


# ----------------------------------------------------------------------
# Parametric families for the initial sweep
# ----------------------------------------------------------------------


def arcsine_eps_shape(n: int, eps: float) -> np.ndarray:
    """Discrete arcsine‑type shape with a smoothing epsilon."""
    i = np.arange(n, dtype=float)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - 0.5 - i) + eps)
    return normalize(vals)


def powerlaw_shape(n: int, p: float, eps: float) -> np.ndarray:
    """Symmetric power‑law shape a_i ∝ ((i+ε)(n‑1‑i+ε))⁻ᵖ."""
    i = np.arange(n, dtype=float)
    base = ((i + eps) * (n - 1 - i + eps)) ** (-p)
    return normalize(base)


def uniform_shape(n: int) -> np.ndarray:
    """Uniform probability vector on n points."""
    return np.full(n, 1.0 / n)


# ----------------------------------------------------------------------
# Refinement methods
# ----------------------------------------------------------------------


def refine_smooth(
    a: np.ndarray,
    deadline: float,
    *,
    eta0: float = 0.5,
    beta: float = 20.0,
    report_every: int = 0,
) -> Tuple[np.ndarray, float]:
    """
    Multiplicative mirror‑descent with a softened max‑weight.
    Returns the best iterate (exact objective) and its score.
    """
    a = normalize(a.astype(float).copy())
    n = len(a)
    best_a = a.copy()
    conv = convolution_fft(a, a)
    best_sc = 2.0 * n * float(np.max(conv))

    it = 0
    while time.time() < deadline:
        it += 1
        conv = convolution_fft(a, a)
        max_val = float(np.max(conv))

        # softmax‑type weights
        exp_vals = np.exp(beta * (conv - max_val))
        w = exp_vals / np.sum(exp_vals)

        grad = gradient_from_weights(a, w)

        eta = eta0 / np.sqrt(it)

        a = a * np.exp(-eta * grad)
        a = normalize(a)

        cur_sc = 2.0 * n * max_val
        if cur_sc < best_sc:
            best_sc = cur_sc
            best_a = a.copy()

        if report_every and (it % report_every == 0):
            print(f"[smooth] iter={it:,d} cur={cur_sc:.6f} best={best_sc:.6f}")

    return best_a, best_sc


def refine_subgradient(
    a: np.ndarray,
    deadline: float,
    *,
    max_iter: int = 200_000,
    eta0: float = 0.7,
    report_every: int = 0,
) -> Tuple[np.ndarray, float]:
    """
    Classical sub‑gradient descent on the true max‑convolution objective.
    Uses exponentiated‑gradient updates and a 1/√t step‑size.
    Returns the best vector seen and its exact score.
    """
    a = normalize(a.astype(float).copy())
    n = len(a)
    best_a = a.copy()
    best_sc = 2.0 * n * float(np.max(convolution_fft(a, a)))

    for it in range(1, max_iter + 1):
        if time.time() >= deadline:
            break

        conv = convolution_fft(a, a)
        max_val = float(np.max(conv))

        # uniform weight on (near) maximisers
        w = np.zeros_like(conv)
        max_idx = np.where(conv >= max_val - 1e-14)[0]
        w[max_idx] = 1.0 / len(max_idx)

        grad = gradient_from_weights(a, w)

        eta = eta0 / np.sqrt(it)
        a = a * np.exp(-eta * grad)
        a = normalize(a)

        cur_sc = 2.0 * n * max_val
        if cur_sc < best_sc:
            best_sc = cur_sc
            best_a = a.copy()

        if report_every and (it % report_every == 0):
            print(f"[sub‑grad] iter={it:,d} cur={cur_sc:.6f} best={best_sc:.6f}")

    return best_a, best_sc


# ----------------------------------------------------------------------
# Helper for up‑sampling a discrete distribution
# ----------------------------------------------------------------------


def upscale(arr: np.ndarray, new_n: int) -> np.ndarray:
    """
    Linear interpolation from `len(arr)` points to `new_n` points.
    The result is re‑normalised to sum‑to‑one.
    """
    if len(arr) == new_n:
        return normalize(arr)
    old_x = np.linspace(0.0, 1.0, len(arr), endpoint=False)
    new_x = np.linspace(0.0, 1.0, new_n, endpoint=False)
    up = np.interp(new_x, old_x, arr)
    up = np.maximum(0.0, up)
    return normalize(up)

            print("[fallback] loaded length =", len(fallback_arr))
    except NameError:
        pass

    # ------------------------------------------------------------------
    # 1.  Parametric pre‑search (≈ 15 % of the budget)
    # ------------------------------------------------------------------
    n_small = 2048               # resolution during the quick search
    candidates: List[Tuple[np.ndarray, float, Dict[str, Any]]] = []

    # seed candidates
    candidates.append((arcsine_eps_shape(n_small, eps=0.25), None, {"type": "arcsine", "eps": 0.25, "lam": 0.0}))
    candidates.append((uniform_shape(n_small), None, {"type": "uniform"}))
    if fallback_params:
        candidates.append((fallback_params["data"], None, fallback_params))

    # evaluate seed candidates
    for idx, (cand, _, _) in enumerate(candidates):
        sc = compute_score(cand)
        candidates[idx] = (cand, sc, candidates[idx][2])

    # keep the best among the seeds
    best_cand, best_sc, best_params = min(candidates, key=lambda x: x[1] if x[1] is not None else float("inf"))
    print(f"[search] initial best fast score = {best_sc:.6f}")

    # allocate time for random param exploration
    remaining = max(0.0, deadline - time.time())
    search_time = min(30.0, remaining * 0.15)      # cap at 30 s or 15 % of remaining
    search_deadline = time.time() + search_time
    print(f"[search] random param sweep will run for ≈{search_time:.1f}s")

    while time.time() < search_deadline:
        # randomly pick a family
        if rng.random() < 0.55:
            # power‑law family (captures arcsine as p≈0.5)
            p = rng.uniform(0.25, 0.9)          # exponent < 1 to stay integrable
            eps = rng.uniform(0.0, 5.0)
            base = powerlaw_shape(n_small, p, eps)
            family = "powerlaw"
        else:
            # pure arcsine with tunable epsilon
            eps = rng.uniform(0.0, 5.0)
            base = arcsine_eps_shape(n_small, eps)
            family = "arcsine"
            p = None

        lam = rng.uniform(0.0, 0.4)             # weight of a tiny uniform component
        shape = (1.0 - lam) * base + lam * uniform_shape(n_small)
        shape = normalize(shape)

        sc = compute_score(shape)
        if sc < best_sc:
            best_sc = sc
            best_cand = shape
            best_params = {
                "type": family,
                "p": p,
                "eps": eps,
                "lam": lam,
            }
            print(f"[search] new best fast score {best_sc:.6f} (type={family}, eps={eps:.3f}, "
                  f"{'p=' + f'{p:.3f}' if p is not None else ''} lam={lam:.3f})")

    print(f"[search] finished with best fast score = {best_sc:.6f}")

    # ------------------------------------------------------------------
    # 2.  Build the full‑resolution start vector
    # ------------------------------------------------------------------
    target_n = 4096          # a larger resolution for the final optimisation
    if best_params.get("type") == "fallback":
        # simply upscale the stored fallback array
        a0 = upscale(best_params["data"], target_n)
    else:
        # reconstruct the shape at the higher resolution using the saved params
        typ = best_params.get("type")
        eps = best_params.get("eps", 0.25)
        lam = best_params.get("lam", 0.0)
        if typ == "powerlaw":
            p = best_params["p"]
            base = powerlaw_shape(target_n, p, eps)
        elif typ == "arcsine":
            base = arcsine_eps_shape(target_n, eps)
        else:   # should not happen, fall back to uniform
            base = uniform_shape(target_n)

        if lam > 0.0:
            base = (1.0 - lam) * base + lam * uniform_shape(target_n)
        a0 = normalize(base)

    # ------------------------------------------------------------------
    # 3.  Smooth mirror‑descent (a modest slice of the remaining budget)
    # ------------------------------------------------------------------
    remaining = max(0.0, deadline - time.time())
    smooth_time = max(5.0, remaining * 0.07)           # at least 5 s, about 7 % of what is left
    smooth_deadline = time.time() + smooth_time
    print(f"[refine] starting smooth descent, allocated ≈{smooth_time:.1f}s")

    refined_smooth, smooth_score = refine_smooth(
        a=a0,
        deadline=smooth_deadline,
        eta0=0.5,
        beta=20.0,
        report_every=0,
    )
    print(f"[refine] smooth best exact score = {smooth_score:.6f}")

    val = evaluate_sequence(refined_smooth.tolist())
    best_seq = refined_smooth.tolist()
    best_val = val
    print(f"[refine] smooth official score = {val:.6f}")

    # ------------------------------------------------------------------
    # 4.  Hard sub‑gradient polishing (use the rest of the time)
    # ------------------------------------------------------------------
    if time.time() < deadline - 5.0:
        print("\n[polish] starting hard sub‑gradient polish")
        polished, _ = refine_subgradient(
            a=refined_smooth,
            deadline=deadline,
            max_iter=300_000,
            eta0=0.7,
            report_every=0,
        )
        final_val = evaluate_sequence(polished.tolist())
        print(f"[polish] polished official score = {final_val:.6f}")

        if final_val < best_val:
            best_val = final_val
            best_seq = polished.tolist()
            print("*** Updated global best after hard polish ***")

        # If there is still a sliver of time left, try a brief second polish
        # from a slightly perturbed point (helps escape shallow minima).
        if time.time() < deadline - 5.0:
            perturb = np.maximum(0.0, polished + rng.normal(scale=0.005, size=polished.shape))
            perturb = normalize(perturb)
            print("[polish] second pass from a perturbed start")
            second, _ = refine_subgradient(
                a=perturb,
                deadline=deadline,
                max_iter=200_000,
                eta0=0.7,
                report_every=0,
            )
            second_val = evaluate_sequence(second.tolist())
            print(f"[polish] second pass official score = {second_val:.6f}")
            if second_val < best_val:
                best_val = second_val
                best_seq = second.tolist()
                print("*** Updated global best after second polish ***")

    # ------------------------------------------------------------------
    # 5.  Guard – always return a feasible sequence.
    # ------------------------------------------------------------------
    if not best_seq:
        # Extremely unlikely fallback
        best_seq = uniform_shape(256).tolist()
        best_val = evaluate_sequence(best_seq)

    print(f"\n[final] Search finished. Best reported official score = {best_val:.6f}")
    return best_seq
