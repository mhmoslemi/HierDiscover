```
    best = a.copy()
    best_score = compute_score(best)
    cur_score = best_score

    step = 0.1  # fraction max mass to transfer per move, adapt
    temp_start = 0.1
    temp_end = 1e-5
    # we can set total SA steps based on time, not predetermined.
    while time.time() < deadline:
        # compute temperature linearly decreasing with time
        t = (time.time() - start) / (deadline - start)
        temp = temp_start * (1 - t) + temp_end * t
        # propose move
        cand = a.copy()
        if np.random.rand() < 0.5:
            # swap
            i, j = np.random.randint(n, size=2)

            if i == j: continue

            cand[j] += delta
```

```
        t += 1

        candidates = np.where(conv >= max_conv - eps)[0]
```

```
def simulated_annealing(a_init, time_deadline):
    a = a_init.copy()
    best = a.copy()
    best_score = compute_score(a)
    cur_score = best_score
    n = len(a)
    start = time.time()
    init_temp = 0.1
    final_temp = 1e-5
    step = 0.1   # fractional transfer step size
    while time.time() < time_deadline:
        elapsed = time.time() - start
        frac = elapsed / (time_deadline - start)
        temp = init_temp * (1 - frac) + final_temp * frac
        # propose move
        cand = a.copy()
        if np.random.rand() < 0.5:
            # swap
            i, j = np.random.randint(n, size=2)
            cand[i], cand[j] = cand[j], cand[i]
        else:
            i, j = np.random.randint(n, size=2)
            if i != j:
                delta = step * cand[i]  # move portion of mass from i to j
                cand[i] -= delta
                cand[j] += delta
                # ensure nonnegative (if delta > value)
                if cand[i] < 0:
                    cand[j] += cand[i]  # add back negative part, making cand[i]=0
                    cand[i] = 0.0
        # compute score
        new_score = compute_score(cand)
        # accept if better or probabilistically
        if new_score < cur_score or np.random.rand() < np.exp((cur_score - new_score) / max(temp, 1e-12)):
            a = cand
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best = cand.copy()
                # adapt step on improvement
                step = min(step * 1.05, 0.5)
        else:
            # maybe adjust step
            step = max(step * 0.97, 1e-5)
    return best, best_score
```

The code below implements a three‑phase optimiser for the objective  

\[
\text{score}(a)=\frac{2\,n\;\max\limits_k (a*a)_k}{\bigl(\sum_i a_i\bigr)^2},
\qquad a_i\ge 0,
\]

where \((a*a)_k\) is the ordinary (full) convolution of `a` with itself.  
All sequences are kept on the probability simplex (∑ a = 1), because the
objective is homogeneous of degree 0.

* **Phase 0 – Structured initialization**  
  A Dirichlet draw is sorted decreasingly and then placed on the line in an
  “interleaved‑ends’’ order (largest → left‑most, then right‑most, then the next
  largest, …).  This spreads large entries far from their mirror positions and
  gives a much better starting point than a plain random vector.

* **Phase 1 – Exponentiated sub‑gradient descent**  
  The exact sub‑gradient of the max‑convolution term is used, but rather than
  always taking the unique maximiser we randomly pick one of the indices whose
  value lies within a small relative tolerance of the maximum.  This
  “soft‑max’’ behaviour provides a smoothing effect and helps the iterate leave
  flat plateaus.  An adaptive step size \( \eta_t = \eta_0/\sqrt t \) together
  with Euclidean projection onto the simplex is employed (the same update as
  the reference implementation, but with the stochastic index choice).

* **Phase 2 – Simulated‑annealing polishing**  
  A fast stochastic hill‑climber that accepts worsening moves with a
  temperature schedule.  Three move types are used:

  1. swapping two coordinates,
  2. transferring a fraction of mass from one coordinate to another,
  3. transferring mass between a coordinate and its mirror position
     (i ↔ n‑1‑i) – this directly attacks the term that creates the large
     central convolution value.

  The move size adapts to recent successes.

All phases respect the overall 1000 s budget – each phase stops as soon as the
deadline is reached.  Multiple restarts with different lengths (1024, 2048,
4096) are performed; the best sequence found is returned.

```python
"""
Optimiser for the “max‑self‑convolution” score.

The algorithm proceeds in three phases:
  0. Structured Dirichlet start (interleaved‑ends permutation).
  1. Exponentiated sub‑gradient descent with a stochastic choice of
     a near‑maximal convolution index (adds smoothing).
  2. Simulated‑annealing polishing (swap / transfer / mirror‑transfer moves).

All work stays on the probability simplex (∑a = 1), because the score is
scale‑invariant.  The routine respects the 1000 s budget and returns the best
sequence found.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# FFT‑based convolution utilities (fast linear convolution)
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Linear convolution a*b using real FFT (fast)."""
    m = len(a) + len(b) - 1
    n_fft = next_pow2(m)
    A = np.fft.rfft(a, n=n_fft)
    B = np.fft.rfft(b, n=n_fft)
    C = A * B
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


def compute_score(a: np.ndarray) -> float:
    """
    Fast replica of `evaluate_sequence` for a vector normalised to sum = 1.
    Returns 2·n·max(conv) where conv = a * a.
    """
    n = a.shape[0]
    conv = convolution_fft(a, a)
    return 2.0 * n * float(np.max(conv))


# ----------------------------------------------------------------------
# Projection onto the probability simplex
# ----------------------------------------------------------------------


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the probability simplex
    (non‑negative entries that sum to 1).
    """
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / v.shape[0])

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, v.shape[0] + 1)
    cond = u - cssv / ind > 0
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


# ----------------------------------------------------------------------
# Structured initialisation (interleaved ends)
# ----------------------------------------------------------------------


def interleaved_initial_vector(n: int, rng: np.random.Generator) -> np.ndarray:
    """
    Draw a Dirichlet vector, sort it decreasingly and place the values in
    an interleaved‑ends order: largest at position 0, second largest at n‑1,
    third largest at 1, fourth at n‑2, … .
    The resulting vector sums to 1 and is already on the simplex.
    """
    a = rng.dirichlet(np.ones(n))
    a_sorted = np.sort(a)[::-1]          # descending
    order = np.empty(n, dtype=int)
    left, right, pos, toggle = 0, n - 1, 0, True
    while left <= right:
        if toggle:
            order[pos] = left
            left += 1
        else:
            order[pos] = right
            right -= 1
        pos += 1
        toggle = not toggle
    a_permuted = np.empty_like(a_sorted)
    a_permuted[order] = a_sorted
    return a_permuted


# ----------------------------------------------------------------------
# Exponentiated sub‑gradient optimisation (smoothed max‑choice)
# ----------------------------------------------------------------------


def exp_subgrad_optimize(init_a: np.ndarray,
                         deadline: float,
                         eta0: float = 0.6,
                         eps_factor: float = 1e-3) -> tuple[np.ndarray, float]:
    """
    Exponentiated sub‑gradient descent on f(a)=2·n·max(a*a).
    The maximising index is chosen randomly among all indices whose
    convolution value lies within `eps_factor` of the absolute maximum,
    providing a smoothing effect.
    """
    a = init_a.astype(float).copy()
    best_a = a.copy()
    best_score = compute_score(a)
    n = len(a)
    t = 0

    while time.time() < deadline:
        t += 1
        conv = convolution_fft(a, a)
        max_val = float(conv.max())
        # pick a near‑maximal index
        eps = eps_factor * max_val
        cand_idx = np.where(conv >= max_val - eps)[0]
        k = int(rng.choice(cand_idx))

        # sub‑gradient for shift k: g_i = 2 * a_{k-i} (if index valid)
        idx = np.arange(n)
        j = k - idx
        mask = (j >= 0) & (j < n)
        g = np.zeros_like(a)
        g[mask] = 2.0 * a[j[mask]]

        # step size
        eta = eta0 / np.sqrt(t)

        # multiplicative update and projection
        a = a * np.exp(-eta * g)
        a = project_to_simplex(a)

        cur_score = compute_score(a)
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()

        # occasional tiny jitter to escape tiny plateaus
        if t % 5000 == 0:
            a += rng.standard_normal(n) * (eta0 * 1e-4 / np.sqrt(t))
            a = project_to_simplex(a)

    return best_a, best_score


# ----------------------------------------------------------------------
# Simulated‑annealing polishing
# ----------------------------------------------------------------------


def simulated_annealing(a_start: np.ndarray,
                        deadline: float,
                        init_temp: float = 0.12,
                        final_temp: float = 1e-5,
                        init_step: float = 0.07) -> tuple[np.ndarray, float]:
    """
    Simple SA that performs three types of moves:
        * swap two coordinates,
        * transfer a fraction of mass from i to j,
        * transfer a fraction between a coordinate and its mirror (i ↔ n‑1‑i).
    The temperature is linearly annealed from `init_temp` to `final_temp`.
    """
    a = a_start.copy()
    best = a.copy()
    best_score = compute_score(a)
    cur_score = best_score
    n = len(a)

    start = time.time()
    step = init_step

    while time.time() < deadline:
        # temperature schedule (linear in elapsed fraction)
        elapsed = time.time() - start
        frac = elapsed / (deadline - start)
        temp = init_temp * (1.0 - frac) + final_temp * frac

        cand = a.copy()
        move_type = rng.random()

        if move_type < 0.33:
            # swap two positions
            i, j = rng.integers(n, size=2)
            cand[i], cand[j] = cand[j], cand[i]
        elif move_type < 0.66:
            # ordinary transfer i → j
            i, j = rng.integers(n, size=2)
            if i != j:
                delta = step * cand[i]
                cand[i] -= delta
                cand[j] += delta
                if cand[i] < 0.0:                     # safety clamp
                    cand[j] += cand[i]
                    cand[i] = 0.0
        else:
            # mirror transfer i ↔ n‑1‑i
            i = rng.integers(n)
            j = n - 1 - i
            if i != j:
                delta = step * cand[i]
                cand[i] -= delta
                cand[j] += delta
                if cand[i] < 0.0:
                    cand[j] += cand[i]
                    cand[i] = 0.0

        # projection not needed – sum is preserved by the moves
        cand = project_to_simplex(cand)   # keep numerical drift in check

        new_score = compute_score(cand)

        # Metropolis acceptance
        if new_score < cur_score or rng.random() < np.exp((cur_score - new_score) / max(temp, 1e-12)):
            a = cand
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best = cand.copy()
                step = min(step * 1.06, 0.5)   # enlarge step after success
        else:
            step = max(step * 0.96, 1e-5)      # shrink step after failure

    return best, best_score


# ----------------------------------------------------------------------
# Main entry point required by the harness
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative sequence that minimises `evaluate_sequence`.
    The routine performs several restarts with different lengths, each
    consisting of a structured start, exponentiated sub‑gradient descent,
    and a simulated‑annealing polish.  The best sequence found within the
    time budget is returned.
    """
    global rng
    rng = np.random.default_rng(seed)

    start_time = time.time()
    # Small safety margin for the external harness
    deadline = start_time + budget_s - 1.0

    best_seq = None
    best_val = float("inf")

    # ------------------------------------------------------------------
    # Try a few different lengths (more degrees of freedom → potentially
    # lower scores) while respecting the remaining time.
    # ------------------------------------------------------------------
    lengths = [1024, 2048, 4096]
    for n in lengths:
        if time.time() >= deadline:
            break

        # ----- Phase 0 : initialise -------------------------------------------------
        a0 = interleaved_initial_vector(n, rng)
        cur_val = compute_score(a0)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a0.tolist()
            print(f"Initial (n={n}) score = {best_val:.6f}")

        # ----- Phase 1 : exponentiated sub‑gradient ----------------------------------
        # allocate roughly half of the remaining budget to this phase
        time_now = time.time()
        remaining = deadline - time_now
        subgrad_deadline = time_now + 0.45 * remaining
        if subgrad_deadline > time_now:
            a_opt, score_opt = exp_subgrad_optimize(
                a0,
                deadline=subgrad_deadline,
                eta0=0.6,
                eps_factor=1e-3,
            )
            if score_opt < best_val:
                best_val = score_opt
                best_seq = a_opt.tolist()
                print(f"Sub‑grad (n={n}) improved best to {best_val:.6f}")

        # ----- Phase 2 : simulated‑annealing polishing -------------------------------
        time_now = time.time()
        if time_now < deadline - 0.2:      # leave a tiny buffer
            sa_deadline = deadline
            a_poll, poll_val = simulated_annealing(
                np.array(best_seq, dtype=float) if best_seq is not None else a_opt,
                deadline=sa_deadline,
                init_temp=0.12,
                final_temp=1e-5,
                init_step=0.07,
            )
            if poll_val < best_val:
                best_val = poll_val
                best_seq = a_poll.tolist()
                print(f"SA polish (n={n}) improved best to {best_val:.6f}")

    # ----------------------------------------------------------------------
    # Fallback – uniform vector (should never be needed, but kept for safety)
    # ----------------------------------------------------------------------
    if best_seq is None:
        n = 256
        best_seq = [1.0 / n] * n
        best_val = evaluate_sequence(best_seq)

    final_score = evaluate_sequence(best_seq)
    print(f"\nSearch finished. Final reported score = {final_score:.6f}")
    return best_seq
```
