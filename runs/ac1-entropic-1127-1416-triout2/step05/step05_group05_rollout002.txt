    n = len(f)

"""
Improved non‑negative step‑function search.

The search proceeds in three phases:

1. **Initialisation** – start from the well‑known arcsine shape (and a
   possible fallback `height_sequence_1`).  The vector is normalised to
   unit sum (the objective is scale‑invariant).

2. **LP‑driven Frank‑Wolfe steps** – repeatedly solve the linear program

        max    Σ b_i
        s.t.   (a * b)_k ≤ M   for all k
               b_i ≥ 0

   where `M = max(a*a)` for the current iterate `a`.  The LP finds a
   direction that adds as much mass as possible without increasing the
   current worst‑case convolution value.  A convex combination
   `a ← (1‑t)·a + t·b` (with a decreasing step‑size *t*) moves the
   iterate towards a better shape while keeping the sum fixed by a
   final normalisation.  Only a handful of LP solves (each on a dense
   Toeplitz matrix of size (2n‑1)×n) are needed; with `n=1024` the total
   time stays well below the 1000 s budget.

3. **Fine polishing** – the result of the LP phase is refined with the
   exponentiated‑gradient / sub‑gradient scheme from the previous
   implementation (soft‑max mirror descent followed by a hard sub‑gradient
   pass).  This removes any remaining “noise’’ and typically yields the
   best bound observed.

If the `cvxpy` package is unavailable the algorithm skips the LP phase
and directly runs the smooth‑then‑hard gradient refinement (the previous
state‑of‑the‑art method).

All helper functions are top‑level; `propose_candidate` is the entry
point used by the competition harness.
"""

import time
import math
import numpy as np
from typing import List, Tuple

# ----------------------------------------------------------------------
# Optional import of cvxpy (used for the LP phase)
# ----------------------------------------------------------------------
try:
    import cvxpy as cp
    _CVXPY_AVAILABLE = True
except Exception:  # pragma: no cover
    _CVXPY_AVAILABLE = False

# ----------------------------------------------------------------------
# Basic numeric utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Linear convolution via real FFT (full length, zero‑padded)."""
    m = len(a) + len(b) - 1
    n_fft = next_pow2(m)
    A = np.fft.rfft(a, n=n_fft)
    B = np.fft.rfft(b, n=n_fft)
    C = A * B
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


def normalize_to_sum(v: np.ndarray, target: float = 1.0) -> np.ndarray:
    """Project a non‑negative vector onto the simplex of a given total."""
    total = float(v.sum())
    if total <= 0.0:
        return np.full_like(v, target / len(v))
    return v * (target / total)


def compute_fast_score(a: np.ndarray) -> float:
    """
    Fast (scale‑invariant) evaluation of 2·n·max(conv(a,a)) / (sum(a)²).
    The vector `a` is assumed non‑negative.
    """
    n = len(a)
    conv = convolution_fft(a, a)
    max_val = float(conv.max())
    s = float(a.sum())
    if s < 1e-12:
        return float("inf")
    return 2.0 * n * max_val / (s * s)


def gradient_from_weights(a: np.ndarray, w: np.ndarray) -> np.ndarray:
    """
    Gradient of Σ_k w_k·(a*a)_k w.r.t. a.
    Implements g_j = 2· Σ_m a_m·w_{m+j}.
    Computed via one FFT‑based convolution.
    """
    w_rev = w[::-1]
    conv = convolution_fft(a, w_rev)                # length = 3·n‑2
    idx = 2 * len(a) - 2
    grads = 2.0 * conv[idx - np.arange(len(a))]
    return grads


# ----------------------------------------------------------------------
# Simple families for the initial candidate
# ----------------------------------------------------------------------


def arcsine_initial(n: int) -> np.ndarray:
    """Discrete arcsine distribution (≈ 1/√(i·(n‑1‑i)))."""
    i = np.arange(n, dtype=float)
    eps = 0.25
    vals = 1.0 / np.sqrt((i + 0.5) * (n - 0.5 - i) + eps)
    return normalize_to_sum(vals, target=1.0)


def uniform_shape(n: int) -> np.ndarray:
    """Uniform vector."""
    return np.full(n, 1.0 / n)


def random_dirichlet(n: int, rng: np.random.Generator) -> np.ndarray:
    """Sample uniformly from the simplex."""
    samp = rng.exponential(1.0, size=n)
    return normalize_to_sum(samp, target=1.0)


# ----------------------------------------------------------------------
# Linear‑programming sub‑routine (LP‑phase)
# ----------------------------------------------------------------------


def build_convolution_matrix(f: np.ndarray) -> np.ndarray:
    """
    Construct the (2n‑1) × n matrix T such that (T @ b) = f * b.
    Entry (k, j) = f_{k‑j} if 0 ≤ k‑j < n, otherwise 0.
    """
    n = len(f)
    rows = np.arange(2 * n - 1)[:, None]          # (2n‑1, 1)
    cols = np.arange(n)                           # (n,)
    idx = rows - cols                              # (2n‑1, n)
    mask = (idx >= 0) & (idx < n)
    T = np.zeros((2 * n - 1, n), dtype=float)
    T[mask] = f[idx[mask]]
    return T


def solve_lp_max_sum(f: np.ndarray, M: float) -> np.ndarray | None:
    """
    Solve   max Σ b_i   s.t.   (f * b)_k ≤ M  for all k,   b ≥ 0.
    Returns the optimal vector `b` or ``None`` on failure.
    """
    if not _CVXPY_AVAILABLE:                       # pragma: no cover
        return None

    n = len(f)
    T = build_convolution_matrix(f)                # shape (2n‑1, n)
    b = cp.Variable(n, nonneg=True)

    constraints = [
        T @ b <= np.full(2 * n - 1, float(M) + 1e-12)  # tiny epsilon
    ]

    prob = cp.Problem(cp.Maximize(cp.sum(b)), constraints)
    try:
        prob.solve(solver=cp.ECOS, abstol=1e-9, reltol=1e-9, feastol=1e-9,
                   max_iters=5000, warm_start=False)
    except Exception:                              # pragma: no cover
        return None

    if b.value is None:
        return None
    # Numerical cleanup
    sol = np.maximum(np.asarray(b.value, dtype=float), 0.0)
    return sol


def lp_refinement(
    a0: np.ndarray,
    deadline: float,
    max_iters: int = 30,
    t0: float = 0.5,
) -> Tuple[np.ndarray, float]:
    """
    Frank‑Wolfe‑style refinement using the LP described above.
    Returns the best vector found (as a numpy array) and its official
    score obtained via ``evaluate_sequence``.
    """
    a = normalize_to_sum(a0, target=1.0)
    best_a = a.copy()
    best_val = float("inf")
    t = t0
    it = 0

    while time.time() < deadline and it < max_iters:
        it += 1

        # Current bound M = max(a*a)
        conv = convolution_fft(a, a)
        M = float(conv.max())

        b = solve_lp_max_sum(a, M)
        if b is None:
            # fall back to a simple convex combination with the uniform
            b = uniform_shape(len(a))
        # Mix – convex combination with decreasing step size
        a = (1.0 - t) * a + t * b
        a = normalize_to_sum(a, target=1.0)

        # Exact evaluation (the official, more expensive routine)
        try:
            cur_val = evaluate_sequence(a.tolist())
        except Exception:  # Should never happen, but be safe
            cur_val = float("inf")

        if cur_val < best_val:
            best_val = cur_val
            best_a = a.copy()
            # make the step a bit larger after a good improvement
            # (helps escape flat regions)
            t = min(t0, t * 1.05)

        # Decay the step size gently
        t = max(0.01, t * 0.95)

        # occasional progress report
        if it % 5 == 0:
            print(f"  LP iter={it:3d}  t={t:.3f}  cur_score={cur_val:.6f}  best={best_val:.6f}")

    return best_a, best_val


# ----------------------------------------------------------------------
# Smooth‑then‑hard gradient polishing (same as earlier implementation)
# ----------------------------------------------------------------------


def refine_smooth(
    a: np.ndarray,
    deadline: float,
    *,
    eta0: float = 1.0,
    beta: float = 12.0,
) -> Tuple[np.ndarray, float]:
    """
    Multiplicative mirror‑descent with soft‑max weighting.
    Returns the best iterate (exact score) seen before the deadline.
    """
    a = normalize_to_sum(a, target=1.0)
    n = len(a)
    best_a = a.copy()
    conv = convolution_fft(a, a)
    best_score = 2.0 * n * float(conv.max())

    it = 0
    while time.time() < deadline:
        it += 1
        conv = convolution_fft(a, a)
        max_val = float(conv.max())

        # softmax‑type weights
        exp_vals = np.exp(beta * (conv - max_val))
        w = exp_vals / exp_vals.sum()

        grad = gradient_from_weights(a, w)
        eta = eta0 / math.sqrt(it)

        a = a * np.exp(-eta * grad)
        a = normalize_to_sum(a, target=1.0)

        cur_score = 2.0 * n * max_val
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()

    return best_a, best_score


def refine_subgradient(
    a: np.ndarray,
    deadline: float,
    *,
    max_iter: int = 200_000,
    eta0: float = 0.7,
) -> Tuple[np.ndarray, float]:
    """
    Classical sub‑gradient descent on the true max‑convolution objective.
    Uses multiplicative updates and a 1/√t step‑size schedule.
    """
    a = normalize_to_sum(a, target=1.0)
    n = len(a)
    best_a = a.copy()
    conv = convolution_fft(a, a)
    best_score = 2.0 * n * float(conv.max())

    for it in range(1, max_iter + 1):
        if time.time() >= deadline:
            break

        conv = convolution_fft(a, a)
        max_val = float(conv.max())

        # uniform weight on the (near‑)maximisers
        w = np.zeros_like(conv)
        idx_max = np.where(conv >= max_val - 1e-14)[0]
        w[idx_max] = 1.0 / len(idx_max)

        grad = gradient_from_weights(a, w)
        eta = eta0 / math.sqrt(it)

        a = a * np.exp(-eta * grad)
        a = normalize_to_sum(a, target=1.0)

        cur_score = 2.0 * n * max_val
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()

    return best_a, best_score

def propose_candidate(
    seed: int = 42,
    budget_s: int = 1000,
    **kwargs,
) -> List[float]:
    """
    Search for a non‑negative step‑function that minimises the official
    evaluation function.  The routine respects the given time budget,
    first performs a few LP‑driven Frank‑Wolfe steps (if cvxpy is
    available) and finally polishes the result with a smooth
    exponentiated‑gradient phase followed by a hard sub‑gradient pass.
    The best sequence found is returned as a list of floats.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    # small safety margin
    deadline = start_time + budget_s - 2.0

    # ------------------------------------------------------------------
    # 0. Choose problem size (trade‑off between richness and speed)
    # ------------------------------------------------------------------
    # 1024 works comfortably within the time limit; 2048 gives a slightly
    # better theoretical bound but costs roughly 4× the work.
    n = 1024

    # ------------------------------------------------------------------
    # 1. Initial candidate (fallback + simple families)
    # ------------------------------------------------------------------
    candidates: List[np.ndarray] = []

    # a) previously discovered construction, if present
    try:
        fallback = np.array([float(x) for x in height_sequence_1], dtype=float)
        if fallback.ndim == 1:
            fallback = normalize_to_sum(fallback, target=1.0)
            candidates.append(fallback)
            print("[fallback] length =", len(fallback))
    except NameError:
        pass

    # b) deterministic families
    candidates.append(arcsine_initial(n))
    candidates.append(uniform_shape(n))

    # c) a few random Dirichlet draws
    for _ in range(3):
        candidates.append(random_dirichlet(n, rng))

    # ------------------------------------------------------------------
    # 2. Quick ranking (fast FFT‑based score, scale‑invariant)
    # ------------------------------------------------------------------
    scored = [(compute_fast_score(c), c) for c in candidates]
    scored.sort(key=lambda x: x[0])
    best_fast_seq = scored[0][1].copy()
    print(f"Initial best fast score = {scored[0][0]:.6f}")

    # ------------------------------------------------------------------
    # 3. LP‑driven refinement (if possible)
    # ------------------------------------------------------------------
    best_seq: List[float] = []
    best_val: float = float("inf")

    if _CVXPY_AVAILABLE:
        # allocate roughly half of the remaining budget to the LP phase
        time_left = max(0.0, deadline - time.time())
        lp_deadline = time.time() + 0.5 * time_left
        print(f"Starting LP refinement (deadline in {time_left * 0.5:.1f}s)")

        refined_lp, lp_score = lp_refinement(
            best_fast_seq,
            deadline=lp_deadline,
            max_iters=20,
            t0=0.5,
        )

        # official evaluation of the LP result
        val = evaluate_sequence(refined_lp.tolist())
        print(f"LP refined official score = {val:.6f}")
        if val < best_val:
            best_val = val
            best_seq = refined_lp.tolist()
            print("*** New global best after LP refinement ***")

        current = refined_lp.copy()
    else:
        # No LP support – start polishing from the fast candidate
        current = best_fast_seq.copy()
        print("cvxpy not available – skipping LP phase")

    # ------------------------------------------------------------------
    # 4. Smooth mirror‑descent (soft‑max) – uses roughly half of the rest
    # ------------------------------------------------------------------
    remaining = max(0.0, deadline - time.time())
    smooth_deadline = time.time() + 0.45 * remaining
    print(f"Starting smooth mirror‑descent (deadline in {remaining * 0.45:.1f}s)")

    smooth_res, smooth_score = refine_smooth(
        a=current,
        deadline=smooth_deadline,
        eta0=1.0,
        beta=12.0,
    )
    val = evaluate_sequence(smooth_res.tolist())
    print(f"Smooth refined official score = {val:.6f}")
    if val < best_val:
        best_val = val
        best_seq = smooth_res.tolist()
        print("*** New global best after smooth refinement ***")

    # ------------------------------------------------------------------
    # 5. Hard sub‑gradient polishing – uses the remaining budget
    # ------------------------------------------------------------------
    if time.time() < deadline - 1.0:
        print("\nStarting hard sub‑gradient polish")
        sub_deadline = deadline
        polished_res, _ = refine_subgradient(
            a=smooth_res,
            deadline=sub_deadline,
            max_iter=200_000,
            eta0=0.7,
        )
        final_val = evaluate_sequence(polished_res.tolist())
        print(f"Polished official score = {final_val:.6f}")

        if final_val < best_val:
            best_val = final_val
            best_seq = polished_res.tolist()
            print("*** Updated global best after hard polish ***")

    # ------------------------------------------------------------------
    # 6. Safety net – ensure a feasible return value
    # ------------------------------------------------------------------
    if not best_seq:
        best_seq = uniform_shape(256).tolist()
        best_val = evaluate_sequence(best_seq)

    print(f"\nSearch finished. Best reported score = {best_val:.6f}")
    return best_seq
