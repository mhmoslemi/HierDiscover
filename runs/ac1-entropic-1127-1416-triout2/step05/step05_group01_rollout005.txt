    smoothed objective is
        g = 2·corr( softmax_α(conv) , a )
    where `corr` is the linear correlation (implemented via one FFT).  The
    temperature `α` is increased logarithmically from a small value (≈ 5) to a
    large value (≈ 2000), thus first smoothing the landscape and later focusing
    on the true max.  The update is a *mirrored exponential* step
        a ← a·exp(−η·g)   (η∝1/√t)
    followed by Euclidean projection onto the probability simplex.
    The loop runs until the allotted time is exhausted (≈ 80 % of the budget)
    while keeping the best vector seen.

3️⃣  **Simulated‑annealing polish** – a short stochastic hill‑climber that
    performs random mass‑transfer moves and accepts worsening moves with a
    temperature schedule.  This helps to escape tiny plateaus left by the
    deterministic phase.

All arithmetic is performed with `float64`.  The final candidate is returned
as a plain list of floats; `evaluate_sequence` (provided by the harness) is
called only once at the end to obtain the official score.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# FFT utilities (real convolution)
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Linear convolution a * b using real FFT (fast)."""
    m = len(a) + len(b) - 1
    n_fft = next_pow2(m)
    A = np.fft.rfft(a, n=n_fft)
    B = np.fft.rfft(b, n=n_fft)
    C = A * B
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


# ----------------------------------------------------------------------
# Simplex projection (Euclidean)
# ----------------------------------------------------------------------


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Project a vector onto the probability simplex (non‑negative, sum = 1)."""
    if np.all(v <= 0):
        # all zero or negative → uniform distribution
        return np.full_like(v, 1.0 / v.shape[0])

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, v.shape[0] + 1)
    cond = u - cssv / ind > 0
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Objective utilities
# ----------------------------------------------------------------------


def compute_score(a: np.ndarray) -> float:
    """Exact score used by `evaluate_sequence` when sum(a)=1."""
    n = a.shape[0]
    conv = convolution_fft(a, a)
    return 2.0 * n * float(np.max(conv))


def correlation_with_a(w: np.ndarray, a: np.ndarray) -> np.ndarray:
    """
    Compute the correlation g_i = ∑_k w_k * a_{k-i}
    using a single FFT.  Result length = len(a).
    """
    # correlation = conv(w, reverse(a))
    conv = convolution_fft(w, a[::-1])
    n = a.shape[0]
    start = n - 1
    return conv[start:start + n]


# ----------------------------------------------------------------------
# Continuation gradient descent (smooth max → actual max)
# ----------------------------------------------------------------------


def continuation_descent(
    init_a: np.ndarray,
    deadline: float,
    max_iter: int = 500_000,
    eta0: float = 0.8,
    alpha_start: float = 5.0,
    alpha_end: float = 2000.0,
) -> tuple[np.ndarray, float]:
    """
    Smoothed gradient descent with a temperature (α) schedule.
    Returns the best vector encountered and its score.
    """
    a = init_a.astype(float).copy()
    n = a.shape[0]

    best_a = a.copy()
    best_score = compute_score(a)

    # pre‑compute the schedule factor for α
    if max_iter > 0:
        log_alpha_factor = (np.log(alpha_end) - np.log(alpha_start)) / max_iter
    else:
        log_alpha_factor = 0.0

    t = 1
    last_check = time.time()
    while t <= max_iter and time.time() < deadline:
        # ----- convolution and current max -----
        conv = convolution_fft(a, a)               # length 2n‑1
        cur_max = np.max(conv)
        cur_score = 2.0 * n * float(cur_max)

        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()

        # ----- temperature α (log‑linear schedule) -----
        alpha = np.exp(np.log(alpha_start) + log_alpha_factor * t)

        # ----- soft‑max weights -----
        # stability: shift by max(conv)
        shift = cur_max
        exp_terms = np.exp(alpha * (conv - shift))
        w = exp_terms / np.sum(exp_terms)          # ∑w = 1, w ≥ 0

        # ----- gradient g = 2 * corr(w, a) -----
        g = 2.0 * correlation_with_a(w, a)

        # ----- diminishing step size -----
        eta = eta0 / np.sqrt(t)

        # ----- exponentiated (mirrored) update -----
        log_update = -eta * g
        # shift to avoid under‑/overflow while preserving ratios
        log_update -= np.max(log_update)
        a = a * np.exp(log_update)

        # ----- project back onto simplex -----
        a = project_to_simplex(a)

        # ----- occasional jitter to escape flat regions -----
        if t % 5000 == 0:
            jitter = np.random.randn(n) * (eta0 * 1e-4 / np.sqrt(t))
            a += jitter
            a = project_to_simplex(a)

        # ----- loop bookkeeping -----
        t += 1
        # cheap periodic time‑check
        if t % 1000 == 0:
            now = time.time()
            if now - last_check > 0.5:   # avoid calling time.time() every iteration
                last_check = now
                if now >= deadline:
                    break

    return best_a, best_score


# ----------------------------------------------------------------------
# Simulated‑annealing fine‑tuner
# ----------------------------------------------------------------------


def simulated_annealing_refine(
    start_a: np.ndarray,
    deadline: float,
    max_steps: int = 300_000,
) -> tuple[np.ndarray, float]:
    """
    Simple SA that moves a fraction of mass between random positions.
    Accepts worsening moves with probability exp(−Δ/ T) where T decays
    linearly to zero.
    """
    a_cur = start_a.copy()
    n = a_cur.shape[0]
    cur_score = compute_score(a_cur)

    best_a = a_cur.copy()
    best_score = cur_score

    # Temperature schedule (linear decay)
    total_time = deadline - time.time()
    if total_time <= 0.0:
        return best_a, best_score
    T_start = max(0.01, cur_score * 0.01)   # start a few percent of the current score
    steps_done = 0
    last_check = time.time()

    while steps_done < max_steps and time.time() < deadline:
        # linear temperature based on elapsed fraction
        elapsed = time.time() - (deadline - total_time)
        T = T_start * max(0.0, 1.0 - elapsed / total_time)

        # ----- propose a move -----
        cand = a_cur.copy()
        i, j = np.random.randint(n, size=2)
        if i == j:
            continue
        # transfer a random fraction of the mass at i to j
        frac = np.random.rand() * 0.3          # up to 30 % of source mass
        delta = frac * cand[i]
        cand[i] -= delta
        cand[j] += delta

        cand = project_to_simplex(cand)
        new_score = compute_score(cand)

        # ----- Metropolis acceptance -----
        accept = False
        if new_score < cur_score:
            accept = True
        else:
            # probability of accepting a worsening move
            if T > 0.0 and np.random.rand() < np.exp(-(new_score - cur_score) / T):
                accept = True

        if accept:
            a_cur = cand
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_a = cand.copy()

        steps_done += 1

        # periodic time check (every 2000 steps)
        if steps_done % 2000 == 0:
            if time.time() >= deadline:
                break

    return best_a, best_score


# ----------------------------------------------------------------------
# Main entry point required by the evaluation harness
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative sequence that minimises `evaluate_sequence`.
    The function respects the given time budget (seconds) and returns a list
    of floats (the step heights).
    """
    np.random.seed(seed)
    start_time = time.time()
    # keep a small safety margin for the harness
    deadline = start_time + budget_s - 1.5

    best_seq = None
    best_val = float("inf")

    # --------------------------------------------------------------
    # 0️⃣  Try the supplied fallback (if any)
    # --------------------------------------------------------------
    try:
        fallback = [float(x) for x in height_sequence_1]      # type: ignore
        fallback_val = evaluate_sequence(fallback)
        print(f"Fallback score = {fallback_val:.6f}")
        if fallback_val < best_val:
            best_val = fallback_val
            best_seq = fallback
    except Exception:
        pass

    # --------------------------------------------------------------
    # 1️⃣  Random Dirichlet start (concentration chosen at random)
    # --------------------------------------------------------------
    n = 2048                         # a bit larger than the previous 1024
    concentration = 10 ** np.random.uniform(-1, 1)   # ≈0.1 … 10
    a0 = np.random.dirichlet(np.full(n, concentration))

    cur_val = compute_score(a0)
    print(f"Random start (conc={concentration:.3f}) score = {cur_val:.6f}")
    if cur_val < best_val:
        best_val = cur_val
        best_seq = a0.tolist()

    # --------------------------------------------------------------
    # 2️⃣  Continuation smoothed gradient descent
    # --------------------------------------------------------------
    # Reserve ~70 % of the time for this phase
    phase2_deadline = min(deadline - 12.0, start_time + budget_s * 0.70)
    if time.time() < phase2_deadline:
        print("\n--- Phase 2: continuation gradient descent ---")
        init_vec = np.array(best_seq, dtype=float) if best_seq is not None else a0
        refined_a, refined_score = continuation_descent(
            init_vec,
            time_deadline=phase2_deadline,
            max_iter=600_000,
            eta0=0.8,
            alpha_start=5.0,
            alpha_end=2000.0,
        )
        if refined_score < best_val:
            best_val = refined_score
            best_seq = refined_a.tolist()
            print(f"Phase‑2 improved score to {best_val:.6f}")

    # --------------------------------------------------------------
    # 3️⃣  Simulated‑annealing polishing
    # --------------------------------------------------------------
    if best_seq is not None and time.time() < deadline - 2.0:
        print("\n--- Phase 3: simulated‑annealing polish ---")
        cur_vec = np.array(best_seq, dtype=float)
        polished_a, polished_score = simulated_annealing_refine(
            cur_vec,
            time_deadline=deadline,
            max_steps=300_000,
        )
        if polished_score < best_val:
            best_val = polished_score
            best_seq = polished_a.tolist()
            print(f"Phase‑3 improved score to {best_val:.6f}")

    # --------------------------------------------------------------
    # Fallback: a tiny uniform vector (should never be used)
    # --------------------------------------------------------------
    if best_seq is None:
        n = 256
        best_seq = [1.0 / n] * n
        best_val = evaluate_sequence(best_seq)

    # Final official evaluation (the harness will also do this, but we print it)
    final_score = evaluate_sequence(best_seq)
    print(f"\nSearch finished. Final reported score = {final_score:.6f}")
    return best_seq
