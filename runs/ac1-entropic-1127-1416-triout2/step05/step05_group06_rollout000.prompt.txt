   descent often misses.

3. **Smooth max‑gradient descent** – the non‑smooth max is replaced by a
   soft‑max (log‑sum‑exp) with a schedule of increasing β.  The exact
   gradient of the smooth surrogate is a cross‑correlation that can be
   evaluated with a single FFT, so each iteration is O(n log n).  An
   exponentiated‑gradient step (multiplicative update) maintains non‑negativity
   and is projected onto the simplex after each step.

The best vector found is finally polished with the original exponentiated

The whole search respects the supplied time budget (default 1000 s) and

def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full linear convolution using real FFT (zero‑padded to a power of two)."""
    m = len(a) + len(b) - 1
    n_fft = next_pow2(m)
    A = np.fft.rfft(a, n=n_fft)
    B = np.fft.rfft(b, n=n_fft)
    C = A * B
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


def compute_score(a: np.ndarray) -> float:
    """
    Fast version of the official objective (scale‑invariant):
        2·n·max(a∗a) / (sum(a)²)
    The vector a is assumed non‑negative; the function works for any shape.
    """
    n = len(a)
    conv = convolution_fft(a, a)
    max_val = float(np.max(conv))
    s = float(np.sum(a))
    if s < 1e-12:
        return float("inf")
    return 2.0 * n * max_val / (s * s)

def arcsine_initial(n: int) -> np.ndarray:
    """Discrete arcsine density (normalised to sum = 1)."""
    i = np.arange(n, dtype=float)
    eps = 0.25
    vals = 1.0 / np.sqrt((i + 0.5) * (n - 0.5 - i) + eps)
    return vals / vals.sum()


def generate_powerlaw(n: int, p: float, eps: float) -> np.ndarray:
    """Symmetric power‑law a_i ∝ ((i+ε)(n‑1‑i+ε))⁻ᵖ."""
    i = np.arange(n, dtype=float)
    base = ((i + eps) * (n - 1 - i + eps)) ** (-p)
    total = base.sum()
    if total == 0.0:
        return np.full(n, 1.0 / n)
    return base / total


def generate_beta_like(n: int, alpha: float, beta: float) -> np.ndarray:
    """Beta‑like shape a_i ∝ (i+½)^{α‑1} (n‑1‑i+½)^{β‑1}."""
    i = np.arange(n, dtype=float)
    base = (i + 0.5) ** (alpha - 1) * (n - 1 - i + 0.5) ** (beta - 1)
    total = base.sum()
    if total == 0.0:
        return np.full(n, 1.0 / n)
    return base / total


def random_dirichlet(n: int, concentration: float, rng: np.random.Generator) -> np.ndarray:
    """Sample a Dirichlet‑like vector (Gamma draws) and normalise."""
    if concentration <= 0:
        samp = rng.exponential(1.0, size=n) ** (1.0 / max(0.1, concentration))
    else:
        samp = rng.gamma(concentration, 1.0, size=n)
    samp += 1e-12
    return samp / samp.sum()


def perturb(a: np.ndarray, rng: np.random.Generator, scale: float = 0.02) -> np.ndarray:
    """Add a tiny Gaussian perturbation and renormalise."""
    b = a + rng.normal(0.0, scale, size=a.shape)

def propose_neighbor(a: np.ndarray, rng: np.random.Generator, max_step: float = 0.02) -> np.ndarray:
    """Move a small amount of mass from one random bin to another."""
    n = len(a)
    i, j = rng.integers(0, n, size=2)
    if i == j or a[i] == 0.0:
        return a  # nothing to move
    delta = a[i] * rng.random() * max_step
    b = a.copy()
    b[i] -= delta
    b[j] += delta
    return b


def simulated_annealing(a0: np.ndarray,
                        deadline: float,
                        rng: np.random.Generator,
                        max_iter: int = 2000,
                        start_temp: float = 0.005,

def softmax_gradient_descent(initial_a: np.ndarray,
                             deadline: float,
                             betas: List[float],
                             iter_per_beta: int = 500,
                             eta0: float = 0.5) -> Tuple[np.ndarray, float]:
    """
    Minimise a smooth approximation of the max‑convolution:
        Φ_β(a) = (1/β)·log Σ_k exp(β·(a∗a)_k).

    An exponentiated‑gradient update is performed, followed by a projection
    onto the probability simplex (simple renormalisation).
    """
    a = np.maximum(initial_a, 1e-30)
    a = a / a.sum()
    best_a = a.copy()
    best_sc = compute_score(a)
    n = len(a)

    for beta in betas:
        for it in range(1, iter_per_beta + 1):
            if time.time() >= deadline:
                return best_a, best_sc

            # convolution a∗a
            conv = convolution_fft(a, a)                     # length 2n‑1
            # soft‑max weights (stable version)
            max_c = conv.max()
            exp_vals = np.exp(beta * (conv - max_c))
            p = exp_vals / exp_vals.sum()                    # Σ p = 1

            # cross‑correlation p ⋆ a  (linear, length n)
            N = next_pow2(len(p) + n - 1)
            P = np.fft.fft(p, N)
            A = np.fft.fft(a, N)
            cross = np.fft.ifft(P * np.conj(A)).real[:n]     # = Σ_k p_k·a_{k‑j}
            cross = np.maximum(cross, 0.0)                   # numerical safety

            # gradient of the smooth objective
            grad = 4.0 * n * cross                           # non‑negative

            # exponentiated‑gradient step with 1/√t schedule
            eta = eta0 / np.sqrt(it)
            a = a * np.exp(-eta * grad)

            # renormalise onto the simplex
            a_sum = a.sum()
            if a_sum == 0.0:
                a = np.full_like(a, 1.0 / n)
            else:
                a = a / a_sum

            cur_sc = compute_score(a)
            if cur_sc < best_sc:
                best_sc = cur_sc
                best_a = a.copy()

    return best_a, best_sc

def _max_conv_gradient(a: np.ndarray, conv: np.ndarray) -> np.ndarray:
    """
    Sub‑gradient of max_k (a∗a)_k.
    For each active offset k, the gradient contribution is 2·a_{k‑j}.
    When several offsets share the maximal value they are averaged.
    """
    n = len(a)
    max_val = conv.max()
    # active offsets (exact equality; tolerance is tiny because values are exact floats)
    mask = np.isclose(conv, max_val, atol=1e-15, rtol=0.0)
    active = np.where(mask)[0]
    grad = np.zeros_like(a)

    for k in active:
        lo = max(0, k - (n - 1))
        hi = min(k, n - 1)
        idx = np.arange(lo, hi + 1, dtype=int)
        grad[idx] += 2.0 * a[k - idx]

    if len(active) > 1:
        grad /= len(active)
    return grad


def optimise_subgradient(initial_a: np.ndarray,
                         deadline: float,
                         *,
                         eta0: float = 0.6,

                         noise_every: int = 0,

    """
    a = np.maximum(initial_a, 1e-30)
    a = a / a.sum()

    best_sc = compute_score(a)
    n = len(a)

    for it in range(1, max_iter + 1):

        conv = convolution_fft(a, a)
        grad = _max_conv_gradient(a, conv)

        eta = eta0 / np.sqrt(it)
        a = a * np.exp(-eta * grad)

        if noise_every and (it % noise_every == 0):
            a += np.random.randn(n) * noise_scale
            a = np.maximum(a, 0.0)

        s = a.sum()
        a = a / s if s != 0.0 else np.full_like(a, 1.0 / n)

        cur_sc = compute_score(a)
        if cur_sc < best_sc:
            best_sc = cur_sc
            best_a = a.copy()

        if it % 2000 == 0:
            print(f"    subgrad iter={it:,d} cur={cur_sc:.6f} best={best_sc:.6f}")

    return best_a, best_sc
