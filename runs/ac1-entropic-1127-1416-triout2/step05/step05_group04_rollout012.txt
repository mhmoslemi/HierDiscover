"""
search_candidate.py – 1000 s heuristic for minimise ``evaluate_sequence``.

The algorithm is deliberately simple, reproducible and fits in the 1 CPU
budget.

1. **Wide random exploration** – many parametric families (Beta, power,
   mixtures, tilted arcsine, sparse Dirichlet, triangular bumps).  Each
   candidate is projected onto the probability simplex (Σa = 1) and the
   cheap surrogate ``2·n·max(conv(a,a))`` is evaluated with an FFT‑based
   convolution.  The best *K* seeds are retained (K≈20).

2. **Quick local improvement** – a short exact sub‑gradient descent
   (step ≈ 1/√t) is applied to every seed.  The surrogate score is used
   for ranking; the globally best seed after this phase is kept.

3. **Heavy polishing** – the current best vector is refined by a long
   Adam optimisation of a smooth soft‑max approximation of the max‑norm,
   followed by a longer exact sub‑gradient run and a brief simulated‑
   annealing polishing.  All steps respect the simplex constraint.

All heavy loops are protected by time‑checks; the routine stops
gracefully before the 1000 s deadline and always returns a non‑empty
list of non‑negative heights ready for ``evaluate_sequence``.
"""

import time
import numpy as np
from typing import List, Tuple

# -------------------------------------------------------------
# FFT‑based convolution utilities (fast, O(n log n))
# -------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Linear convolution via real FFT (length = len(a)+len(b)-1)."""
    m = len(a) + len(b) - 1
    n_fft = next_pow2(m)
    A = np.fft.rfft(a, n=n_fft)
    B = np.fft.rfft(b, n=n_fft)
    C = A * B
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


def compute_score(a: np.ndarray) -> float:
    """
    Fast surrogate for the official score.
    a is assumed to sum to 1, therefore the denominator disappears and
    the score is simply ``2·n·max(conv(a,a))``.
    """
    n = a.shape[0]
    conv = convolution_fft(a, a)
    return 2.0 * n * float(conv.max())


# -------------------------------------------------------------
# Simplex projection (Euclidean) – Duchi et al. 2008.
# -------------------------------------------------------------


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Project v onto {x ≥ 0, Σx = 1}."""
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / v.size)

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho = np.where(u - cssv / np.arange(1, v.size + 1) > 0)[0]
    if len(rho) == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# -------------------------------------------------------------
# Parametric shape factories (all return a probability vector)
# -------------------------------------------------------------


def arcsine_initial(n: int) -> np.ndarray:
    """Classic arcsine shape a_i ∝ 1/√((i+½)*(n‑½‑i))."""
    i = np.arange(n, dtype=float)
    eps = 0.25  # tiny regulariser to avoid division by zero
    vals = 1.0 / np.sqrt((i + 0.5) * (n - 0.5 - i) + eps)
    vals /= vals.sum()
    return vals


def generate_beta_shape(n: int, alpha: float, beta: float, eps: float) -> np.ndarray:
    """Beta(α,β) shape on {0,…,n‑1} (α,β > 0)."""
    i = np.arange(n, dtype=float)
    left = (i + eps) ** (alpha - 1.0)
    right = (n - 1 - i + eps) ** (beta - 1.0)
    vals = left * right
    total = vals.sum()
    if total <= 0 or not np.isfinite(total):
        return np.full(n, 1.0 / n)
    return vals / total


def generate_power_shape(n: int, p: float, eps: float) -> np.ndarray:
    """Generalised arcsine a_i ∝ ((i+½)*(n‑½‑i)+ε)^{-p}, p>0."""
    i = np.arange(n, dtype=float)
    denom = ((i + 0.5) * (n - 0.5 - i) + eps) ** p
    vals = 1.0 / denom
    total = vals.sum()
    if total <= 0 or not np.isfinite(total):
        return np.full(n, 1.0 / n)
    return vals / total


def generate_triangular_shape(n: int, peak: int) -> np.ndarray:
    """Symmetric triangular bump with apex at ``peak``."""
    left = np.arange(peak + 1, dtype=float)
    right = np.arange(n - peak - 1, dtype=float)[::-1]
    seq = np.concatenate([left, right])
    seq /= seq.sum()
    return seq


def generate_mixture_shape(n: int, weight_arc: float) -> np.ndarray:
    """Mixture (weight)·arcsine + (1‑weight)·uniform."""
    arc = arcsine_initial(n)
    uni = np.full(n, 1.0 / n)
    mix = weight_arc * arc + (1.0 - weight_arc) * uni
    return mix / mix.sum()


def generate_tilted(shape: np.ndarray, gamma: float) -> np.ndarray:
    """
    Apply a power tilt  (i+1)^γ  to an existing shape and re‑normalise.
    Positive γ pushes mass to the right, negative γ to the left.
    """
    i = np.arange(shape.size, dtype=float) + 1.0
    tilted = shape * (i ** gamma)
    tilted = np.maximum(tilted, 0.0)
    if tilted.sum() == 0.0:
        tilted = np.full_like(shape, 1.0 / shape.size)
    else:
        tilted /= tilted.sum()
    return tilted


def generate_sparse_dirichlet(n: int, rng: np.random.Generator) -> np.ndarray:
    """Few non‑zero entries (k≈[2, 0.05·n]) drawn from a Dirichlet."""
    max_k = max(2, int(0.05 * n))
    k = rng.integers(2, max_k + 1)
    idx = rng.choice(n, size=k, replace=False)
    vals = rng.exponential(scale=1.0, size=k) + 0.01
    vec = np.zeros(n, dtype=float)
    vec[idx] = vals
    vec = vec / vec.sum()
    return vec


# -------------------------------------------------------------
# Optimisers (all work on vectors that already live on the simplex)
# -------------------------------------------------------------


def smooth_gradient(a: np.ndarray, lam: float) -> np.ndarray:
    """
    Gradient of the soft‑max approximation of max(conv(a,a)).
    Returns ∇_a (2·n·softmax_max); the sum‑constraint is later enforced
    by projecting back onto the simplex.
    """
    n = a.shape[0]
    conv = convolution_fft(a, a)               # length L = 2·n‑1
    max_c = conv.max()

    # Soft‑max weights (numerically stable)
    exp_term = np.exp(lam * (conv - max_c))
    w = exp_term / exp_term.sum()

    # Weighted sub‑gradient of the convolution
    w_rev = w[::-1]
    conv_rev = convolution_fft(a, w_rev)
    L = 2 * n - 1
    weighted_grad = 2.0 * conv_rev[L - 1 - np.arange(n)]

    # Final factor 2·n
    return 2.0 * n * weighted_grad


def adam_optimize_smooth(initial_a: np.ndarray,
                         deadline: float,
                         max_iter: int = 200_000,
                         lr0: float = 0.12,
                         lam_start: float = 2.0,
                         lam_end: float = 200.0,
                         noise_every: int = 2_000,
                         noise_scale: float = 2e-5) -> Tuple[np.ndarray, float]:
    """
    Adam on the soft‑max surrogate.  Returns the best vector (projected
    onto the simplex) together with its fast surrogate score.
    """
    a = project_to_simplex(initial_a.astype(float).copy())
    best_a = a.copy()
    best_sc = compute_score(a)

    beta1, beta2 = 0.9, 0.999
    eps = 1e-8
    m = np.zeros_like(a)
    v = np.zeros_like(a)

    for it in range(1, max_iter + 1):
        if time.time() >= deadline:
            break

        # linear schedule for the soft‑max temperature
        lam = lam_start + (lam_end - lam_start) * min(1.0, it / max_iter)

        grad = smooth_gradient(a, lam)

        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)

        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        step = lr0 * m_hat / (np.sqrt(v_hat) + eps)
        a_new = a - step

        # occasional jitter to escape shallow plateaus
        if noise_every and (it % noise_every == 0):
            a_new += np.random.randn(a.size) * noise_scale

        a_new = project_to_simplex(a_new)

        cur_sc = compute_score(a_new)
        if cur_sc < best_sc:
            best_sc = cur_sc
            best_a = a_new.copy()

        a = a_new

        if it % 5_000 == 0:
            print(f"    AdamSmooth it={it:,d} cur={cur_sc:.6f} best={best_sc:.6f}")

    return best_a, best_sc


def subgradient_descent_exact(initial_a: np.ndarray,
                              deadline: float,
                              max_iter: int = 80_000,
                              step0: float = 0.03) -> Tuple[np.ndarray, float]:
    """
    Exact sub‑gradient descent on f(a)=2·n·max(conv(a,a)).
    Step size η_t = step0 / sqrt(t).  The iterate is always projected onto
    the simplex.
    """
    a = project_to_simplex(initial_a.copy())
    best_a = a.copy()
    best_sc = compute_score(a)

    n = a.shape[0]
    L = 2 * n - 1

    for it in range(1, max_iter + 1):
        if time.time() >= deadline:
            break

        conv = convolution_fft(a, a)
        max_val = conv.max()

        # indices where the max is attained (within a tiny tolerance)
        tol = max_val * 1e-12 + 1e-14
        mask = (np.abs(conv - max_val) <= tol).astype(float)

        # sub‑gradient of the max‑norm
        mask_rev = mask[::-1]
        conv_rev = convolution_fft(a, mask_rev)
        weighted_grad = 2.0 * conv_rev[L - 1 - np.arange(n)]

        eta = step0 / np.sqrt(it)
        a = a - eta * weighted_grad
        a = project_to_simplex(a)

        cur_sc = compute_score(a)
        if cur_sc < best_sc:
            best_sc = cur_sc
            best_a = a.copy()

        if it % 5_000 == 0:
            print(f"    SubGrad it={it:,d} cur={cur_sc:.6f} best={best_sc:.6f}")

    return best_a, best_sc


def simulated_annealing(a: np.ndarray,
                        rng: np.random.Generator,
                        deadline: float,
                        max_iters: int = 120_000,
                        init_temp: float = 0.015,
                        decay: float = 0.99995) -> Tuple[np.ndarray, float]:
    """
    Simple Metropolis local search using the fast surrogate score.
    Moves a small fraction of mass from one coordinate to another.
    """
    cur = a.copy()
    cur_sc = compute_score(cur)
    best = cur.copy()
    best_sc = cur_sc
    n = cur.size
    temp = init_temp
    it = 0

    while it < max_iters and time.time() < deadline:
        it += 1
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue

        # Move up to 20 % of the mass at i
        delta = cur[i] * 0.2 * rng.random()
        if delta <= 1e-15 or cur[i] - delta < 0.0:
            continue

        cand = cur.copy()
        cand[i] -= delta
        cand[j] += delta

        cand_sc = compute_score(cand)

        # Metropolis acceptance
        if cand_sc < cur_sc or rng.random() < np.exp(-(cand_sc - cur_sc) / max(temp, 1e-12)):
            cur = cand
            cur_sc = cand_sc
            if cur_sc < best_sc:
                best = cur.copy()
                best_sc = cur_sc

        temp = max(temp * decay, 1e-8)

        if it % 5_000 == 0:
            print(f"    SA it={it:,d} cur={cur_sc:.6f} best={best_sc:.6f}")

    return best, best_sc

    ``evaluate_sequence`` score.  The routine is time‑aware and returns
    the best sequence discovered before the deadline.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 1.0          # a safety margin

    # ------------------------------------------------------------------
    # 0.  Initialise best with a known fallback (if any) and with the
    #     classic arcsine shape.
    # ------------------------------------------------------------------
    best_seq: List[float] = []
    best_fast: float = float("inf")

    # fallback variable may be supplied by the runner
    if "height_sequence_1" in globals():
        try:
            fallback = [float(x) for x in globals()["height_sequence_1"]]
            val = evaluate_sequence(fallback)
            print(f"Fallback (height_sequence_1) official score = {val:.6f}")
            if val < best_fast:
                best_fast = val
                best_seq = fallback
        except Exception:
            pass

    n = 4096                     # length of the vector we optimise
    print(f"Using vector length n = {n}")

    # baseline arcsine (always useful)
    arc0 = arcsine_initial(n)
    arc0_sc = compute_score(arc0)
    print(f"Baseline arcsine surrogate score = {arc0_sc:.6f}")
    if arc0_sc < best_fast:
        best_fast = arc0_sc
        best_seq = arc0.tolist()

    # ------------------------------------------------------------------
    # 1.  Wide‑range random exploration (top‑K pool)
    # ------------------------------------------------------------------
    top_k = 20
    best_pool: List[Tuple[float, np.ndarray]] = []   # (score, vector)

    def maybe_insert(score: float, vec: np.ndarray) -> None:
        """Keep up to top_k vectors with smallest scores."""
        if len(best_pool) < top_k:
            best_pool.append((score, vec))
        else:
            worst_idx = max(range(len(best_pool)),
                            key=lambda i: best_pool[i][0])
            if score < best_pool[worst_idx][0]:
                best_pool[worst_idx] = (score, vec)

    # deterministic seeds
    maybe_insert(arc0_sc, arc0)
    maybe_insert(compute_score(np.full(n, 1.0 / n)), np.full(n, 1.0 / n))

    # random sampling loop – stop when we need ~350 s for refinement
    min_time_for_refine = 350.0
    max_candidates = 600_000
    cnt = 0
    while time.time() < deadline - min_time_for_refine and cnt < max_candidates:
        cnt += 1
        r = rng.random()
        if r < 0.30:                     # Beta family (asymmetric allowed)
            alpha = rng.uniform(0.02, 5.0)
            beta = rng.uniform(0.02, 5.0)
            eps = rng.uniform(1e-6, 3.0)
            cand = generate_beta_shape(n, alpha, beta, eps)

        elif r < 0.55:                   # Generalised power / arcsine‑like
            p = rng.uniform(0.1, 5.0)
            eps = rng.uniform(0.0, 3.0)
            cand = generate_power_shape(n, p, eps)

        elif r < 0.70:                   # Mixture arcsine / uniform
            w = rng.random()
            cand = generate_mixture_shape(n, w)

        elif r < 0.80:                   # Triangular bump
            peak = rng.integers(0, n)
            cand = generate_triangular_shape(n, peak)

        elif r < 0.90:                   # Tilted arcsine
            base = arcsine_initial(n)
            gamma = rng.uniform(-3.0, 3.0)
            cand = generate_tilted(base, gamma)

        elif r < 0.95:                   # Tilted Beta
            alpha = rng.uniform(0.02, 5.0)
            beta = rng.uniform(0.02, 5.0)
            eps = rng.uniform(1e-6, 3.0)
            base = generate_beta_shape(n, alpha, beta, eps)
            gamma = rng.uniform(-2.5, 2.5)
            cand = generate_tilted(base, gamma)

        else:                            # Sparse Dirichlet (few non‑zeros)
            cand = generate_sparse_dirichlet(n, rng)

        sc = compute_score(cand)
        maybe_insert(sc, cand)

    # Sort the pool – best first
    best_pool.sort(key=lambda x: x[0])
    seed_vectors = [vec for _, vec in best_pool]
    print(f"Random exploration produced {cnt} candidates; "
          f"kept top‑{top_k} seeds.")

    # ------------------------------------------------------------------
    # 2.  Quick sub‑gradient refinement of each seed
    # ------------------------------------------------------------------
    for idx, seed_vec in enumerate(seed_vectors, 1):
        if time.time() >= deadline - 10.0:
            break
        print(f"\n--- Quick refinement of seed {idx}/{len(seed_vectors)} ---")
        # Very short sub‑gradient run (≈15 k iterations) – cheap but often
        # yields noticeable improvements.
        remaining = deadline - time.time()
        max_iter_sub = int(min(15_000, remaining * 30))   # heuristic
        refined, refined_fast = subgradient_descent_exact(
            seed_vec,
            deadline,
            max_iter=max_iter_sub,
            step0=0.04,
        )
        # Use the surrogate fast score for comparison
        if refined_fast < best_fast:
            best_fast = refined_fast
            best_seq = refined.tolist()
            print(f"    New overall best (surrogate) = {best_fast:.6f}")

    # ------------------------------------------------------------------
    # 3.  Heavy polishing of the current best vector
    # ------------------------------------------------------------------
    if best_seq and time.time() < deadline - 5.0:
        a = np.array(best_seq, dtype=float)
        print("\n--- Heavy polishing of the global best ---")

        # A – Adam on smooth surrogate (long run)
        remaining = deadline - time.time()
        max_iter_adam = int(min(300_000, remaining * 45))
        print(f"    AdamSmooth max_iter={max_iter_adam}")
        a, _ = adam_optimize_smooth(
            a,
            deadline,
            max_iter=max_iter_adam,
            lr0=0.10,
            lam_start=2.0,
            lam_end=200.0,
            noise_every=2_000,
            noise_scale=1e-5,
        )

        # B – exact sub‑gradient descent (medium run)
        if time.time() < deadline - 3.0:
            remaining = deadline - time.time()
            max_iter_sub = int(min(250_000, remaining * 40))
            print(f"    Sub‑gradient descent max_iter={max_iter_sub}")
            a, _ = subgradient_descent_exact(
                a,
                deadline,
                max_iter=max_iter_sub,
                step0=0.02,
            )

        # C – simulated annealing polish (short run)
        if time.time() < deadline - 2.0:
            remaining = deadline - time.time()
            max_iter_sa = int(min(150_000, remaining * 35))
            print(f"    Simulated annealing max_iter={max_iter_sa}")
            a, _ = simulated_annealing(
                a,
                rng,
                deadline,
                max_iters=max_iter_sa,
                init_temp=0.012,
                decay=0.99996,
            )

        # Final official evaluation
        try:
            final_val = evaluate_sequence(a.tolist())
            print(f"Final official score = {final_val:.6f}")
            if final_val < best_fast:
                best_fast = final_val
                best_seq = a.tolist()
                print("*** Updated global best after polishing ***")
        except Exception as exc:
            print(f"Evaluation error after polishing: {exc}")

    # ------------------------------------------------------------------
    # 4.  Guard – ensure a non‑empty return value
    # ------------------------------------------------------------------
    if not best_seq:
        best_seq = [1.0] * n
        best_fast = evaluate_sequence(best_seq)

    print(f"\nSearch finished. Best surrogate score = {best_fast:.6f}")
    return best_seq
