    a = normalize(a)
    best_a = a.copy()

        it += 1
        conv = convolution_fft(a,a)
        max_val = np.max(conv)
        # weight on max entries
        w = np.zeros_like(conv)
        # indices within tolerance 1e-12
        idxs = np.where(conv >= max_val - 1e-13)[0]
        w[idxs] = 1.0/len(idxs)
        grad = gradient_from_weights(a,w)
        eta = 0.5/np.sqrt(it)
        a = a * np.exp(-eta*grad)
        a = normalize(a)
        cur_sc = compute_score(a)
        if cur_sc < best_score:
            best_score = cur_sc
            best_a = a.copy()
    return best_a, best_score

"""
Multi‑stage search for a non‑negative step‑function that minimises

    2·n·max(conv(a,a)) / (sum(a)²)

The algorithm proceeds as follows:

1. **Candidate generation** – a rich family of deterministic shapes
   (arcsine, beta‑distributions, symmetric power‑laws, the uniform vector)
   and a few random Dirichlet draws.  All vectors are normalised to the
   simplex (∑a = 1).  Mixtures of the two best deterministic shapes are also
   examined.

2. **Fast pre‑selection** – the candidates are ranked by a cheap FFT‑based
   score (exactly the official objective for ∑a = 1).  The best one becomes
   the seed for the optimisation.

3. **Smooth exponentiated‑gradient descent** – a multiplicative mirror‑
   descent on the simplex with a *soft‑max* weighting of the convolution
   entries.  The temperature β is fairly large so that the update focuses
   on the near‑maximal entries while remaining smooth.

4. **Hard sub‑gradient polishing** – the same multiplicative update but
   using the true (uniform) weight on the maximising indices.  A
   1/√t step‑size schedule is employed; the best iterate is retained.

5. **Stochastic hill‑climbing** – a short random local‑search that moves a
   tiny amount of mass from a randomly chosen coordinate to another.
   Accepts only improving moves; this often removes the last few
   percents of the objective.

The whole routine respects the supplied time limit (default 1000 s) and
always returns a feasible list of non‑negative floats.
"""

import time
import numpy as np
from typing import List, Tuple

# ----------------------------------------------------------------------
# Low‑level numeric utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """
    Linear convolution via real FFT (full length, zero‑padded to a power‑of‑2).
    """
    m = len(a) + len(b) - 1
    n_fft = next_pow2(m)
    A = np.fft.rfft(a, n=n_fft)
    B = np.fft.rfft(b, n=n_fft)
    C = A * B
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


def normalize(v: np.ndarray) -> np.ndarray:
    """Project a non‑negative vector onto the simplex (sum = 1)."""
    total = v.sum()
    if total <= 0.0:
        return np.full_like(v, 1.0 / len(v))
    return v / total


def compute_score(a: np.ndarray) -> float:
    """
    Fast evaluation of the official objective for a vector that already
    satisfies ∑a = 1 (the denominator therefore equals 1).

    Returns 2·n·max(conv(a,a)).
    """
    n = len(a)
    conv = convolution_fft(a, a)
    max_val = float(np.max(conv))
    return 2.0 * n * max_val


def gradient_from_weights(a: np.ndarray, w: np.ndarray) -> np.ndarray:
    """
    Gradient of   Σ_k w_k·(a*a)_k   with respect to a.
    Implements   g_j = 2· Σ_m a_m · w_{m+j}.
    Computed via a single FFT‑based convolution.
    """
    w_rev = w[::-1]
    conv = convolution_fft(a, w_rev)               # length = 3·n‑2
    idx = 2 * len(a) - 2
    grads = 2.0 * conv[idx - np.arange(len(a))]
    return grads


# ----------------------------------------------------------------------
# Parametric families for the initial sweep
# ----------------------------------------------------------------------


def arcsine_initial(n: int) -> np.ndarray:
    """Discrete arcsine distribution (≈ 1/√(i·(n‑1‑i))) normalised."""
    i = np.arange(n, dtype=float)
    eps = 0.25
    vals = 1.0 / np.sqrt((i + 0.5) * (n - 0.5 - i) + eps)
    return normalize(vals)


def powerlaw_shape(n: int, p: float, eps: float) -> np.ndarray:
    """Symmetric power‑law shape a_i ∝ ((i+ε)(n‑1‑i+ε))⁻ᵖ."""
    i = np.arange(n, dtype=float)
    base = ((i + eps) * (n - 1 - i + eps)) ** (-p)
    return normalize(base)


def beta_shape(n: int, alpha: float, eps: float = 1e-12) -> np.ndarray:
    """
    Symmetric beta density (α‑1 on each side).  The small epsilon prevents
    division by zero when α < 1.
    """
    i = np.arange(n, dtype=float) + 0.5
    x = i / n
    vals = (x ** (alpha - 1)) * ((1.0 - x) ** (alpha - 1))
    # replace any infinities / NaNs caused by extreme exponents
    vals = np.where(np.isfinite(vals), vals, eps)
    vals = np.maximum(vals, eps)
    return normalize(vals)


def uniform_shape(n: int) -> np.ndarray:
    """Uniform probability vector on n points."""
    return np.full(n, 1.0 / n)


def random_dirichlet(n: int, rng: np.random.Generator) -> np.ndarray:
    """Draw a random point on the simplex (exponential sampling)."""
    samp = rng.exponential(1.0, size=n)
    return normalize(samp)


# ----------------------------------------------------------------------
# Refinement methods
# ----------------------------------------------------------------------


def refine_smooth(
    a: np.ndarray,
    deadline: float,
    *,
    eta0: float = 1.0,
    beta: float = 20.0,
    report_every: int = 0,
) -> Tuple[np.ndarray, float]:
    """
    Exponentiated‑gradient (mirror‑descent) with a softened max‑weight.
    Returns the best iterate seen (exact objective) and its score.
    """
    a = normalize(a.astype(float).copy())
    n = len(a)
    best_a = a.copy()
    best_sc = compute_score(a)

    it = 0
    while time.time() < deadline:
        it += 1
        conv = convolution_fft(a, a)
        max_val = float(np.max(conv))

        # softmax‑type weights, the maximal entry receives weight 1.
        exp_vals = np.exp(beta * (conv - max_val))
        w = exp_vals / np.sum(exp_vals)

        grad = gradient_from_weights(a, w)

        eta = eta0 / np.sqrt(it)
        a = a * np.exp(-eta * grad)
        a = normalize(a)

        cur_sc = compute_score(a)
        if cur_sc < best_sc:
            best_sc = cur_sc
            best_a = a.copy()

        if report_every and (it % report_every == 0):
            print(f"    [smooth] iter={it:,d} cur={cur_sc:.6f} best={best_sc:.6f}")

    return best_a, best_sc


def refine_subgradient(
    a: np.ndarray,
    deadline: float,
    *,
    max_iter: int = 200_000,
    eta0: float = 0.7,
    report_every: int = 0,
) -> Tuple[np.ndarray, float]:
    """
    Classic sub‑gradient descent (exact max‑weight) with multiplicative
    updates.  Returns the best iterate and its exact score.
    """
    a = normalize(a.astype(float).copy())
    best_a = a.copy()
    best_sc = compute_score(a)
    n = len(a)

    for it in range(1, max_iter + 1):
        if time.time() >= deadline:
            break

        conv = convolution_fft(a, a)
        max_val = float(np.max(conv))

        # uniform weight on all maximisers (numerical tolerance)
        w = np.zeros_like(conv)
        tol = 1e-14
        max_idx = np.where(conv >= max_val - tol)[0]
        w[max_idx] = 1.0 / len(max_idx)

        grad = gradient_from_weights(a, w)

        eta = eta0 / np.sqrt(it)
        a = a * np.exp(-eta * grad)
        a = normalize(a)

        cur_sc = compute_score(a)
        if cur_sc < best_sc:
            best_sc = cur_sc
            best_a = a.copy()

        if report_every and (it % report_every == 0):
            print(f"    [sub‑grad] iter={it:,d} cur={cur_sc:.6f} best={best_sc:.6f}")

    return best_a, best_sc


def local_random_search(
    a: np.ndarray,
    deadline: float,
    rng: np.random.Generator,
    step_factor: float = 0.02,
    report_every: int = 0,
) -> Tuple[np.ndarray, float]:
    """
    Tiny stochastic hill‑climbing: move a small fraction of the mass from a
    random source index to a random target index, always keeping the vector
    on the simplex.  Only improving moves are kept.
    """
    a = a.copy()
    best_a = a.copy()
    best_sc = compute_score(a)
    n = len(a)

    it = 0
    while time.time() < deadline:
        it += 1
        i = rng.integers(n)
        if a[i] <= 1e-14:
            continue
        j = rng.integers(n)
        if i == j:
            continue

        # move a random fraction of the source mass
        move = a[i] * rng.random() * step_factor
        if move <= 0.0:
            continue

        a_candidate = a.copy()
        a_candidate[i] -= move
        a_candidate[j] += move
        # No need to renormalise – total mass unchanged – but we protect
        # against rounding errors.
        a_candidate = normalize(a_candidate)

        sc = compute_score(a_candidate)
        if sc < best_sc:
            best_sc = sc
            best_a = a_candidate.copy()
            a = a_candidate  # continue hill‑climbing from the new point

        if report_every and (it % report_every == 0):
            print(f"    [local] iter={it:,d} cur={sc:.6f} best={best_sc:.6f}")

    return best_a, best_sc
