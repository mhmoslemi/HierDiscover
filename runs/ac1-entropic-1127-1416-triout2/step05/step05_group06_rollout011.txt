def adam_exponential(a, grad, m, v, t, lr=0.5, beta1=0.9, beta2=0.999, eps=1e-8):
    m = beta1*m + (1-beta1)*grad
    v = beta2*v + (1-beta2)*(grad**2)
    m_hat = m / (1 - beta1**t)
    v_hat = v / (1 - beta2**t)
    step = lr * m_hat / (np.sqrt(v_hat) + eps)
    a = a * np.exp(-step)
    a = a / a.sum()
    return a, m, v

def smoothmax_adam(a, deadline, betas, iter_per_beta=2000, lr0=0.5):
    n = len(a)
    a = np.maximum(a, 1e-30)
    a = a / a.sum()
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    t = 0
    best_a = a.copy()
    best_sc = compute_score(a)
    beta_pow = 1.0 # not needed.

    for beta in betas:
        # maybe increase iter_per_beta dynamic based on time.
        for _ in range(iter_per_beta):
            if time.time() >= deadline:
                return best_a, best_sc
            # compute convolution
            conv = convolution_fft(a, a)
            # compute softmax weights
            max_c = conv.max()
            exp_vals = np.exp(beta * (conv - max_c))
            p = exp_vals / exp_vals.sum()
            # cross correlation (p ⋆ a)
            # Usually cross_corr_j = sum_k p_k * a_{k-j} where indices condition.
            # Efficiently compute by FFT: Multiply FFT(p) * conj(FFT(a)) and take real.
            N = next_pow2(len(p) + n - 1)
            P = np.fft.fft(p, N)
            A = np.fft.fft(a, N)
            cross = np.fft.ifft(P * np.conj(A)).real[:n]  # length n
            grad = 4.0 * n * cross  # always non-negative
            # Adam update
            t += 1
            m = 0.9 * m + 0.1 * grad
            v = 0.999 * v + 0.001 * (grad**2)
            m_hat = m / (1 - 0.9**t)
            v_hat = v / (1 - 0.999**t)
            step = lr0 * m_hat / (np.sqrt(v_hat) + 1e-8)
            # multiplicative update
            a = a * np.exp(-step)
            # mitigate zeros
            a = np.maximum(a, 1e-30)
            a = a / a.sum()
            cur_sc = compute_score(a)
            if cur_sc < best_sc:
                best_sc = cur_sc
                best_a = a.copy()
    return best_a, best_sc

"""
Hybrid evolutionary‑gradient search for a low‑max‑convolution step function.

The routine works in three phases:

1. **Broad random sampling** – a large, diverse set of candidates is built
   from several parametric families (arcsine, power‑law, beta‑like,
   Dirichlet, mixtures).  All candidates are normalised to sum‑to‑1 and
   the cheap “fast score” (the exact objective without the security guards)
   is used to keep the best few.

2. **Smooth‑max Adam refinement** – each retained candidate is optimised
   with a smooth log‑sum‑exp surrogate of the max‑convolution.
   The gradient of the surrogate is computed by a single FFT and an
   Adam‑style exponentiated‑gradient update (mirror descent on the
   probability simplex) is applied.  The inverse‑temperature β is
   increased geometrically so the surrogate tightens progressively.

3. **Final sub‑gradient polishing** – the best vector from the previous
   stage is polished with a true sub‑gradient method on the
   max‑convolution, again using an exponentiated‑gradient step and a
   simplex projection.  Light Gaussian noise is injected from time to
   time to help escape shallow plateaus.

The whole search respects a wall‑clock budget (default 1000 s) and always
returns a feasible list of non‑negative floats.
"""

import time
import numpy as np
from typing import List, Tuple

# ----------------------------------------------------------------------
# Numerical utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full linear convolution using a real FFT (zero‑padded to a power of two)."""
    m = len(a) + len(b) - 1
    n_fft = next_pow2(m)
    A = np.fft.rfft(a, n=n_fft)
    B = np.fft.rfft(b, n=n_fft)
    C = A * B
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


def compute_score(a: np.ndarray) -> float:
    """
    Fast version of the official objective (scale‑invariant):
        2·n·max(a∗a) / (sum(a)²)
    The vector a is assumed non‑negative.
    """
    n = len(a)
    conv = convolution_fft(a, a)
    max_val = float(np.max(conv))
    s = float(np.sum(a))
    if s < 1e-12:
        return float("inf")
    return 2.0 * n * max_val / (s * s)


# ----------------------------------------------------------------------
# Families of initial shapes
# ----------------------------------------------------------------------


def arcsine_initial(n: int) -> np.ndarray:
    """Discrete arcsine density (normalised to sum = 1)."""
    i = np.arange(n, dtype=float)
    eps = 0.25
    vals = 1.0 / np.sqrt((i + 0.5) * (n - 0.5 - i) + eps)
    return vals / vals.sum()


def generate_powerlaw(n: int, p: float, eps: float) -> np.ndarray:
    """U‑shaped power law a_i ∝ ((i+ε)(n‑1‑i+ε))⁻ᵖ."""
    i = np.arange(n, dtype=float)
    base = ((i + eps) * (n - 1 - i + eps)) ** (-p)
    total = base.sum()
    if total == 0.0:
        return np.full(n, 1.0 / n)
    return base / total


def generate_beta_like(n: int, alpha: float, beta: float) -> np.ndarray:
    """Beta‑like shape a_i ∝ (i+½)^{α‑1}·(n‑1‑i+½)^{β‑1}."""
    i = np.arange(n, dtype=float)
    base = (i + 0.5) ** (alpha - 1) * (n - 1 - i + 0.5) ** (beta - 1)
    total = base.sum()
    if total == 0.0:
        return np.full(n, 1.0 / n)
    return base / total


def random_dirichlet(n: int, concentration: float, rng: np.random.Generator) -> np.ndarray:
    """Sample a Dirichlet‑like vector (Gamma draws) and normalise."""
    if concentration <= 0:
        # Approximating a very spiky distribution
        samp = rng.exponential(1.0, size=n) ** (1.0 / max(0.1, concentration))
    else:
        samp = rng.gamma(concentration, 1.0, size=n)
    samp += 1e-15
    return samp / samp.sum()


def perturb(a: np.ndarray, rng: np.random.Generator, scale: float = 0.02) -> np.ndarray:
    """Add a tiny Gaussian perturbation and renormalise."""
    b = a + rng.normal(0.0, scale, size=a.shape)
    b = np.maximum(b, 0.0)
    s = b.sum()
    if s == 0.0:
        return np.full_like(a, 1.0 / len(a))
    return b / s


# ----------------------------------------------------------------------
# Smooth‑max optimisation (Adam‑style exponentiated mirror descent)
# ----------------------------------------------------------------------


def smoothmax_adam(
    initial_a: np.ndarray,
    deadline: float,
    betas: List[float],
    iter_per_beta: int = 1500,
    lr0: float = 0.6,
) -> Tuple[np.ndarray, float]:
    """
    Minimise a smooth log‑sum‑exp surrogate of the max‑convolution
    using an Adam‑style exponentiated‑gradient (mirror descent) update.
    """
    a = np.maximum(initial_a, 1e-30)
    a = a / a.sum()
    best_a = a.copy()
    best_sc = compute_score(a)

    n = len(a)
    m = np.zeros_like(a)      # first moment
    v = np.zeros_like(a)      # second moment
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    t = 0

    for beta in betas:
        # adapt iteration count if we are running out of time
        max_it = iter_per_beta
        for _ in range(max_it):
            if time.time() >= deadline:
                return best_a, best_sc

            # ----- 1. convolution a∗a -----
            conv = convolution_fft(a, a)          # length 2n‑1

            # ----- 2. soft‑max weights (stable) -----
            max_c = conv.max()
            exp_vals = np.exp(beta * (conv - max_c))
            p = exp_vals / exp_vals.sum()          # Σ p = 1

            # ----- 3. cross‑correlation p ⋆ a -----
            N = next_pow2(len(p) + n - 1)
            P = np.fft.rfft(p, N)
            A = np.fft.rfft(a, N)
            cross = np.fft.irfft(P * np.conj(A), n=N)[:n]
            cross = np.maximum(cross, 0.0)

            # ----- 4. gradient of the surrogate (non‑negative) -----
            grad = 4.0 * n * cross

            # ----- 5. Adam‑style exponentiated update -----
            t += 1
            m = beta1 * m + (1.0 - beta1) * grad
            v = beta2 * v + (1.0 - beta2) * (grad ** 2)
            m_hat = m / (1.0 - beta1 ** t)
            v_hat = v / (1.0 - beta2 ** t)

            step = lr0 * m_hat / (np.sqrt(v_hat) + eps)   # per‑coordinate step
            a = a * np.exp(-step)                         # multiplicative update
            a = np.maximum(a, 1e-30)
            a = a / a.sum()

            cur_sc = compute_score(a)
            if cur_sc < best_sc:
                best_sc = cur_sc
                best_a = a.copy()

    return best_a, best_sc


# ----------------------------------------------------------------------
# Sub‑gradient optimisation on the true max‑convolution (final polish)
# ----------------------------------------------------------------------


def _max_conv_gradient(a: np.ndarray, conv: np.ndarray) -> np.ndarray:
    """
    Sub‑gradient of Φ(a)=max_k (a∗a)_k.
    For each active offset k (where the maximum is attained) the gradient
    contribution is 2·a_{k‑j}. The contributions are averaged over all
    active offsets.
    """
    n = len(a)
    max_val = conv.max()
    mask = np.isclose(conv, max_val, atol=1e-15, rtol=0.0)
    active = np.where(mask)[0]
    grad = np.zeros_like(a)

    for k in active:
        lo = max(0, k - (n - 1))
        hi = min(k, n - 1)
        idx = np.arange(lo, hi + 1, dtype=int)
        grad[idx] += 2.0 * a[k - idx]

    if len(active) > 1:
        grad /= len(active)
    return grad


def optimise_subgradient(
    initial_a: np.ndarray,
    deadline: float,
    *,
    eta0: float = 0.5,
    max_iter: int = 200_000,
    noise_every: int = 5000,
    noise_scale: float = 1e-3,
) -> Tuple[np.ndarray, float]:
    """
    Exponentiated sub‑gradient descent directly on Φ(a)=max_k (a∗a)_k.
    Light Gaussian noise is injected every ``noise_every`` steps to escape
    shallow plateaus.
    """
    a = np.maximum(initial_a, 1e-30)
    a = a / a.sum()
    best_a = a.copy()
    best_sc = compute_score(a)
    n = len(a)

    for it in range(1, max_iter + 1):
        if time.time() >= deadline:
            break

        conv = convolution_fft(a, a)
        grad = _max_conv_gradient(a, conv)

        eta = eta0 / np.sqrt(it)
        a = a * np.exp(-eta * grad)

        if noise_every and (it % noise_every == 0):
            a += np.random.randn(n) * noise_scale
            a = np.maximum(a, 0.0)

        s = a.sum()
        if s == 0.0:
            a = np.full_like(a, 1.0 / n)
        else:
            a = a / s

        cur_sc = compute_score(a)
        if cur_sc < best_sc:
            best_sc = cur_sc
            best_a = a.copy()

        # occasional status line (keeps the harness happy)
        if it % 5000 == 0:
            print(f"    subgrad iter={it:,d} cur={cur_sc:.6f} best={best_sc:.6f}")

    return best_a, best_sc

def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs) -> List[float]:
    """
    Search for a non‑negative step‑function with a low ``evaluate_sequence``
    value.  The routine respects a wall‑clock budget (default 1000 s) and
    always returns a feasible list of floats.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    # keep a small safety margin for the final call to evaluate_sequence
    deadline = start_time + budget_s - 2.0

    best_seq: List[float] = []
    best_val: float = float("inf")

    # ------------------------------------------------------------------
    # 0.  Fallback to a known construction if it exists.
    # ------------------------------------------------------------------
    try:
        fallback = [float(x) for x in height_sequence_1]  # type: ignore
        val = evaluate_sequence(fallback)
        print(f"Fallback (height_sequence_1) score = {val:.6f}")
        if val < best_val:
            best_val = val
            best_seq = fallback
    except NameError:
        pass

    # ------------------------------------------------------------------
    # 1.  Broad random sampling of many parametric families.
    # ------------------------------------------------------------------
    lengths = [1024, 2048, 4096, 8192]          # powers of two – FFT friendly
    TOP_K_GLOBAL = 12                         # how many promising candidates to keep

    # global top‑K list (score, vector)
    global_top: List[Tuple[float, np.ndarray]] = []

    def try_add_candidate(vec: np.ndarray) -> None:
        """Insert a candidate into the global top‑K list if it is good enough."""
        nonlocal global_top
        vec = np.maximum(vec, 0.0)
        s = vec.sum()
        if s == 0.0:
            return
        vec = vec / s
        sc = compute_score(vec)
        if len(global_top) < TOP_K_GLOBAL:
            global_top.append((sc, vec.copy()))
            global_top.sort(key=lambda x: x[0])
        else:
            if sc < global_top[-1][0]:
                global_top[-1] = (sc, vec.copy())
                global_top.sort(key=lambda x: x[0])

    for n in lengths:
        if time.time() >= deadline - 10.0:
            break
        print(f"\n=== Sampling length n = {n} ===")

        # ----- deterministic seeds -------------------------------------------------
        try_add_candidate(np.full(n, 1.0 / n))               # uniform
        try_add_candidate(arcsine_initial(n))                # classic arcsine

        # ----- random power‑law shapes --------------------------------------------
        for _ in range(30):
            p = rng.uniform(0.2, 5.0)
            eps = 10 ** rng.uniform(np.log10(1e-3), np.log10(max(1.0, 0.1 * n)))
            try_add_candidate(generate_powerlaw(n, p, eps))

        # ----- beta‑like shapes (allow α,β < 1 for sharper U‑shape) -------------
        for _ in range(30):
            alpha = 10 ** rng.uniform(np.log10(0.1), np.log10(5.0))
            beta = 10 ** rng.uniform(np.log10(0.1), np.log10(5.0))
            try_add_candidate(generate_beta_like(n, alpha, beta))

        # ----- Dirichlet draws (different concentrations) -------------------------
        for conc in [0.01, 0.05, 0.2, 0.5, 1.0, 2.0]:
            try_add_candidate(random_dirichlet(n, conc, rng))

        # ----- perturbed arcsine (helps to leave the immediate basin) -------------
        for _ in range(5):
            try_add_candidate(perturb(arcsine_initial(n), rng, scale=0.04))

        # ----- simple mixtures of arcsine and uniform -------------------------------
        for _ in range(5):
            w = rng.random()
            mix = w * arcsine_initial(n) + (1.0 - w) * np.full(n, 1.0 / n)
            try_add_candidate(mix)

        # ----- random mixtures of two Dirichlet draws ------------------------------
        for _ in range(5):
            a = random_dirichlet(n, rng.uniform(0.01, 2.0), rng)
            b = random_dirichlet(n, rng.uniform(0.01, 2.0), rng)
            w = rng.random()
            try_add_candidate(w * a + (1 - w) * b)

    # ------------------------------------------------------------------
    # 2.  Refine the most promising candidates with a smooth‑max Adam optimiser.
    # ------------------------------------------------------------------
    if not global_top:
        # as a safety net fall back to a uniform vector
        best_seq = [1.0] * 256
        best_val = evaluate_sequence(best_seq)

    # sort the retained candidates by their fast score (ascending)
    global_top.sort(key=lambda x: x[0])

    # geometric schedule for the temperature beta (tightening the surrogate)
    betas = list(np.geomspace(5.0, 2000.0, num=9))

    for idx, (score, vec) in enumerate(global_top):
        if time.time() >= deadline - 5.0:
            break
        print(f"\nRefining candidate #{idx + 1}/{len(global_top)} (n={len(vec)}) "
              f"initial fast score = {score:.6f}")

        # how many inner iterations per beta can we afford?
        remaining = max(0.0, deadline - time.time())

        val = evaluate_sequence(refined_vec.tolist())
        print(f"  after Adam refinement: fast={refined_sc:.6f} eval={val:.6f}")
        if val < best_val:
            best_val = val
            best_seq = refined_vec.tolist()
            print(f"  *** New global best after Adam {best_val:.6f} ***")

    # ------------------------------------------------------------------
    # 3.  Final polishing of the overall best sequence (if time permits).
    # ------------------------------------------------------------------
    if best_seq and time.time() < deadline - 5.0:
        print("\nFinal polishing with max‑convolution sub‑gradient")
        a = np.array(best_seq, dtype=float)

    if not best_seq:
        best_seq = [1.0] * 256
        best_val = evaluate_sequence(best_seq)

    print(f"\nSearch finished. Best reported score = {best_val:.6f}")
    return best_seq
