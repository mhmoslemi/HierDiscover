def propose_candidate(seed=123, budget_s=1000):
    rng = np.random.default_rng(seed)
    start = time.time()
    deadline = start + budget_s - 0.5 # small margin

    best_seq = None
    best_score = float('inf')
    
    # base length
    lengths = [256, 512, 1024, 2048, 4096, 8192]
    prev_vec = None
    for n in lengths:
        if time.time() >= deadline:
            break
        print(f'\n=== Length {n} ===')
        # initial candidate
        if prev_vec is None:
            # random sampling many
            best_local = None
            best_local_sc = float('inf')

            starters = [np.full(n, 1.0/n), arcsine_initial(n), np.linspace(0,2,n) ] # maybe linear shape scaled

                st = st / st.sum()
                sc = compute_fast_score(st)
                if sc < best_local_sc:
                    best_local_sc = sc
                    best_local = st

            max_samples = 20000 if n <= 1024 else 10000 if n <= 2048 else 5000 if n <= 4096 else 3000
            for i in range(max_samples):
                if time.time() >= deadline - 5:
                    break
                # choose mixture or simple beta
                if rng.random() < 0.4:
                    # mixture
                    w = rng.random()
                    alpha1 = rng.uniform(0.05, 5.0)

                else:
                    alpha = rng.uniform(0.05, 5.0)

"""
Propose a low‑score step‑function.

Algorithm
---------
1. Multi‑scale exploration:
   * Start with a modest length (256) and progressively double it up to
     8192.  At each scale a large set of beta‑family shapes (including
     mixtures) is sampled; the best by the fast surrogate
     `2·n·max(conv(a,a))` (sum of a is kept 1) is kept.
   * The best shape of a scale is linearly up‑sampled to the next scale
     providing a warm‑start.

2. Polishing for each scale:
   * Adam optimisation on a log‑sum‑exp surrogate of the max‑convolution.
     Temperature `lam` is annealed from 1 to 300 (or 400 for the final
     polish) so the surrogate becomes tight.
   * Projected sub‑gradient descent on the true max‑convolution with a
     back‑tracking line‑search.
   * Short stochastic Gaussian walk on the simplex (accepting occasional
     uphill moves).

3. After the longest scale a final polishing phase (again Adam,
   sub‑gradient, random walk) is performed on the global best vector.

All steps respect the overall 1000 s budget, regularly check the deadline
and keep the best vector according to the official `evaluate_sequence`
function.  A uniform fallback is returned if no candidate is produced.
"""

import time
import numpy as np
from typing import List, Tuple

# ----------------------------------------------------------------------
# Linear‑algebra utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """
    Linear convolution via real FFT.
    Returns an array of length len(a)+len(b)-1.
    """
    m = a.size + b.size - 1
    n_fft = next_pow2(m)
    A = np.fft.rfft(a, n=n_fft)
    B = np.fft.rfft(b, n=n_fft)
    C = A * B
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


def max_convolution(a: np.ndarray) -> float:
    """Maximum entry of the self‑convolution of a."""
    return float(np.max(convolution_fft(a, a)))


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection onto the probability simplex {x ≥ 0, Σx = 1}.
    Implementation of Duchi et al. (2008).
    """
    n = v.shape[0]
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / n)

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho = np.where(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def compute_fast_score(a: np.ndarray) -> float:
    """
    Fast surrogate: 2·n·max(conv(a,a)).
    For vectors normalised to Σa = 1 this equals the official score.
    """
    n = a.shape[0]
    return 2.0 * n * max_convolution(a)


# ----------------------------------------------------------------------
# Shape generators (beta families, mixtures, arcsine)
# ----------------------------------------------------------------------


def generate_beta_shape(n: int, alpha: float, beta: float, eps: float) -> np.ndarray:
    """
    Symmetric beta‑family density on {0,…,n‑1}:
        a_i ∝ (i+eps)^{α‑1}·(n‑1‑i+eps)^{β‑1}.
    Returned vector is normalised to Σa = 1.
    """
    i = np.arange(n, dtype=float)
    left = (i + eps) ** (alpha - 1.0)
    right = (n - 1 - i + eps) ** (beta - 1.0)
    base = left * right
    total = base.sum()
    if total <= 0.0 or not np.isfinite(total):
        return np.full(n, 1.0 / n)
    return base / total


def generate_beta_mixture(
    n: int,
    alpha1: float,
    beta1: float,
    eps1: float,
    alpha2: float,
    beta2: float,
    eps2: float,
    w: float,
) -> np.ndarray:
    """
    Convex combination of two beta‑family densities, renormalised to Σ=1.
    """
    a1 = generate_beta_shape(n, alpha1, beta1, eps1)
    a2 = generate_beta_shape(n, alpha2, beta2, eps2)
    mix = w * a1 + (1.0 - w) * a2
    mix_sum = mix.sum()
    if mix_sum == 0.0 or not np.isfinite(mix_sum):
        return np.full(n, 1.0 / n)
    return mix / mix_sum


def arcsine_initial(n: int) -> np.ndarray:
    """Classic (symmetrised) arcsine shape, normalised."""
    i = np.arange(n, dtype=float)
    eps = 0.25
    vals = 1.0 / np.sqrt((i + 0.5) * (n - 0.5 - i) + eps)
    vals /= vals.sum()
    return vals


def upsample_vector(vec: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation from length len(vec) to new_n, then renormalise."""
    old_n = vec.shape[0]
    if old_n == new_n:
        return vec.copy()
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    new_vals = np.interp(new_x, old_x, vec)
    new_vals = np.maximum(new_vals, 0.0)
    s = new_vals.sum()
    if s == 0.0:
        return np.full(new_n, 1.0 / new_n)
    return new_vals / s


# ----------------------------------------------------------------------
# Smooth surrogate (log‑sum‑exp) and its gradient (used by Adam)
# ----------------------------------------------------------------------


def smooth_gradient(a: np.ndarray, lam: float) -> Tuple[np.ndarray, float]:
    """
    Gradient of the log‑sum‑exp surrogate of max(conv(a,a)).
    Returns (gradient, surrogate_value).
    """
    n = a.shape[0]
    conv = convolution_fft(a, a)  # length 2n‑1
    max_c = np.max(conv)

    # stable soft‑max over convolution entries
    exp_term = np.exp(lam * (conv - max_c))
    sum_exp = np.sum(exp_term)
    w = exp_term / sum_exp                 # soft‑max weights
    B = max_c + np.log(sum_exp) / lam      # smooth approximation of the max

    # weighted sum of sub‑gradients:
    # ∂/∂a Σ_k w_k (a*a)_k = 2 * (a ⋆ w_rev)
    w_rev = w[::-1]
    conv_rev = convolution_fft(a, w_rev)  # length 2n‑1
    L = 2 * n - 1
    weighted_grad = 2.0 * conv_rev[L - 1 - np.arange(n)]

    s = np.sum(a)
    term1 = (2.0 * n / (s ** 2)) * weighted_grad
    term2 = -(4.0 * n * B / (s ** 3)) * np.ones_like(a)
    grad = term1 + term2
    return grad, B


def adam_optimize(
    initial_a: np.ndarray,
    time_deadline: float,
    max_iter: int = 150_000,
    lr0: float = 0.15,
    lam_start: float = 1.0,
    lam_end: float = 200.0,
    noise_every: int = 1000,
    noise_scale: float = 4e-5,
) -> Tuple[np.ndarray, float]:
    """
    Adam optimiser on the smooth log‑sum‑exp surrogate.
    Returns the best vector (projected on the simplex) and its fast score.
    """
    a = project_to_simplex(initial_a.astype(float).copy())
    best_a = a.copy()
    best_sc = compute_fast_score(a)

    beta1, beta2 = 0.9, 0.999
    eps_adam = 1e-8
    m = np.zeros_like(a)
    v = np.zeros_like(a)

    n = a.shape[0]

    for it in range(1, max_iter + 1):
        if time.time() >= time_deadline:
            break

        # temperature schedule (quadratic root annealing)
        t = it / max_iter
        lam = lam_start + (lam_end - lam_start) * (t ** 0.5)

        grad, _ = smooth_gradient(a, lam)

        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)

        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        step = lr0 * m_hat / (np.sqrt(v_hat) + eps_adam)
        a_new = a - step

        if noise_every and (it % noise_every == 0):
            a_new += np.random.randn(n) * noise_scale

        a_new = project_to_simplex(a_new)

        cur_sc = compute_fast_score(a_new)
        if cur_sc < best_sc:
            best_sc = cur_sc
            best_a = a_new.copy()

        a = a_new

        # occasional progress output (lightweight)
        if it % 20000 == 0:
            print(f"    Adam iter={it:,d} cur={cur_sc:.6f} best={best_sc:.6f}")

    return best_a, best_sc


# ----------------------------------------------------------------------
# Exact sub‑gradient descent with back‑tracking line‑search
# ----------------------------------------------------------------------


def subgradient_optimize(
    initial_a: np.ndarray,
    time_deadline: float,
    step0: float = 0.3,
    max_iter: int = 200_000,
) -> Tuple[np.ndarray, float]:
    """
    Projected sub‑gradient descent on the true max‑convolution.
    Returns the best vector (projected) and its fast score.
    """
    a = project_to_simplex(initial_a.copy())
    best_a = a.copy()
    best_sc = compute_fast_score(a)

    n = a.shape[0]

    for it in range(1, max_iter + 1):
        if time.time() >= time_deadline:
            break

        conv = convolution_fft(a, a)
        max_val = float(np.max(conv))
        idx = int(np.argmax(conv))

        # gradient of the active quadratic term
        i = np.arange(n)
        j = idx - i
        mask = (j >= 0) & (j < n)
        grad = np.zeros_like(a)
        grad[mask] = 2.0 * a[j[mask]]

        # simple diminishing step size with back‑tracking
        step = step0 / np.sqrt(it)
        for _ in range(12):
            a_new = a - step * grad
            a_new = project_to_simplex(a_new)
            new_val = max_convolution(a_new)
            if new_val < max_val - 1e-14:
                break
            step *= 0.5

        cur_sc = compute_fast_score(a_new)
        if cur_sc < best_sc:
            best_sc = cur_sc
            best_a = a_new.copy()

        a = a_new

        if it % 50000 == 0:
            print(f"    Sub‑grad iter={it:,d} cur={cur_sc:.6f} best={best_sc:.6f}")

    return best_a, best_sc


# ----------------------------------------------------------------------
# Stochastic local search (Gaussian perturbations)
# ----------------------------------------------------------------------


def random_local_search(
    initial_a: np.ndarray,
    time_deadline: float,
    init_step: float = 0.015,
    max_iter: int = 250_000,
) -> Tuple[np.ndarray, float]:
    """
    Gaussian walk on the simplex with a tiny uphill‑move probability.
    Returns the best vector seen and its fast score.
    """
    a = project_to_simplex(initial_a.copy())
    best_a = a.copy()
    best_sc = compute_fast_score(a)

    n = a.shape[0]
    rng = np.random.default_rng()

    for it in range(1, max_iter + 1):
        if time.time() >= time_deadline:
            break

        step = init_step / np.sqrt(it)
        noise = rng.normal(scale=step, size=n)
        a_proposed = a + noise
        a_proposed = project_to_simplex(a_proposed)

        cur_sc = compute_fast_score(a_proposed)

        # accept better solutions; rarely accept worse ones to escape plateaus
        if cur_sc < best_sc or (rng.random() < 1e-4 and cur_sc < 10 * best_sc):
            a = a_proposed
            if cur_sc < best_sc:
                best_sc = cur_sc
                best_a = a_proposed.copy()

        if it % 50000 == 0:
            print(f"    Random walk iter={it:,d} cur={cur_sc:.6f} best={best_sc:.6f}")

    return best_a, best_sc

        adam_iters = int(min(400_000, remaining * 130))
        print(f"  Adam polishing (max_iter={adam_iters}) …")
        a, _ = adam_optimize(
            initial_a=a,
            time_deadline=deadline,
            max_iter=adam_iters,
            lr0=0.10,
            lam_start=1.0,
            lam_end=400.0,
            noise_every=1500,
            noise_scale=2e-5,
        )

        # Stage B – exact sub‑gradient descent
        now = time.time()
        remaining = deadline - now
        if remaining > 1.5:
            sub_iters = int(min(180_000, remaining * 80))
            print(f"  Sub‑gradient polishing (max_iter={sub_iters}) …")
            a, _ = subgradient_optimize(
                a,
                time_deadline=deadline,
                max_iter=sub_iters,
                step0=0.22,
            )

        # Stage C – final stochastic walk
        now = time.time()
        remaining = deadline - now
        if remaining > 1.0:
            loc_iters = int(min(250_000, remaining * 100))
            print(f"  Random walk polishing (max_iter={loc_iters}) …")
            a, _ = random_local_search(
                a,
                time_deadline=deadline,
                init_step=0.012,
                max_iter=loc_iters,
            )

        final_val = evaluate_sequence(a.tolist())
        print(f"Final polished official score = {final_val:.6f}")
        if final_val < best_official:
            best_official = final_val
            best_seq = a.tolist()
            print("*** Updated global best after final polish ***")

    # ------------------------------------------------------------------
    # 5.  Guard – guarantee a non‑empty return
    # ------------------------------------------------------------------
    if not best_seq:
        best_seq = [1.0] * 256
        best_official = evaluate_sequence(best_seq)
        print("Fallback to uniform 256‑vector.", best_official)

    print(f"\nSearch finished. Best reported official score = {best_official:.6f}")
    return best_seq
