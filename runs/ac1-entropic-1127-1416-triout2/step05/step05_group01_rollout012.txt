def grad_smooth(a, conv, gamma):
    # conv = a*a (full conv)
    max_conv = np.max(conv)
    # w_i = exp(gamma * (conv_i - max_conv))
    w = np.exp(gamma * (conv - max_conv))
    w_sum = w.sum()
    if w_sum == 0:
        w = np.full_like(w, 1.0 / w.size)
    else:
        w /= w_sum
    a_rev = a[::-1]
    full = np.convolve(w, a_rev)  # length 3n-2
    g = 2 * full[len(a)-1:len(a)-1+len(a)]  # n entries
    return g
```

Pseudo:

```
def smooth_expgd(a, deadline):
    n = len(a)
    eta0 = 0.5
    gamma0 = 5.0
    gamma_max = 200.0
    max_iter = 1000000   # large bound; break when deadline reached
    best_a = a.copy()
    best_score = compute_score(a)
    for t in range(1, max_iter+1):
        if time.time() >= deadline:
            break
        # conv compute
        conv = convolution_fft(a, a)
        max_conv = np.max(conv)
        # compute weights using softmax with gamma
        gamma = gamma0 + (gamma_max - gamma0) * (t / max_iter)
        # stable softmax
        shifted = conv - max_conv
        w = np.exp(gamma * shifted)
        w_sum = w.sum()
        if w_sum == 0:
            w = np.full_like(w, 1.0 / w.size)
        else:
            w = w / w_sum
        # gradient:
        # compute conv of w with reversed a
        a_rev = a[::-1]
        grad_full = convolution_fft(w, a_rev)   # length >= 2n-1? Our convolution_fft returns linear convolution length len(w)+len(a_rev)-1 (should be 3n-2). It will return full array, not truncated.
        # need slice:
        start = n-1
        grad = 2.0 * grad_full[start:start+n]
        # multiplicative update
        eta = eta0 / np.sqrt(t)
        a = a * np.exp(-eta * grad)
        # project onto simplex
        a_sum = a.sum()
        if a_sum == 0:
            a = np.full_like(a, 1.0 / n)
        else:
            a = a / a_sum
        # keep best
        cur_score = compute_score(a)
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
    return best_a, best_score
```

def smooth_expgd(a0, deadline):
    n = len(a0)
    a = a0.copy()
    best_a = a.copy()
    best_score = compute_score(a)
    v = np.zeros_like(a)
    beta = 0.9
    eta0 = 0.4
    gamma_start = 3.0
    gamma_end = 100.0
    max_iter = 500_000
    for t in range(1, max_iter+1):
        if time.time() >= deadline:
            break
        conv = convolution_fft(a, a)
        max_val = np.max(conv)
        # softmax weights
        gamma = gamma_start + (gamma_end - gamma_start) * (t / max_iter)
        shifted = conv - max_val
        w = np.exp(gamma * shifted)
        w_sum = np.sum(w)
        if w_sum == 0:
            w = np.full_like(w, 1.0 / w.size)
        else:
            w = w / w_sum
        # gradient
        a_rev = a[::-1]
        grad_full = convolution_fft(w, a_rev)
        grad = 2.0 * grad_full[n-1 : n-1+n]
        # momentum
        v = beta * v + (1 - beta) * grad
        # step size
        eta = eta0 / np.sqrt(t)
        # multiplicative update
        a = a * np.exp(-eta * v)
        # projection onto simplex
        s = a.sum()
        if s == 0:
            a = np.full_like(a, 1.0 / n)
        else:
            a = a / s
        cur_score = compute_score(a)
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
    return best_a, best_score

                cand[i] -= delta
                cand[j] += delta

            cand[i+1] += delta

        cand = project_to_simplex(cand)
        cur = compute_score(cand)
        if cur < best_score:
            best = cand
            best_score = cur
            step = min(step * 1.1, 0.5)
        else:
            step *= 0.97
        max_steps -= 1
    return best.tolist(), best_score

"""
Proposes a low‑score step function for the secret ``evaluate_sequence`` routine.

Algorithm
---------
1. **Length choice** – use a longer sequence (2048 or 4096) when time permits.
2. **Multiple restarts** – initialise the vector on the probability simplex
   with a Dirichlet draw (different concentration parameters give diverse
   starting points).
3. **Smooth exponentiated gradient descent**  
   * Objective f(a)=2·n·max(a∗a) (a∗a is the linear autocorrelation).  
   * Approximate the max by a soft‑max (temperature γ) and compute the
     exact gradient of the smoothed objective:
        w_i ∝ exp(γ·(a∗a)_i) ,  Σw_i=1
        ∇f ≈ 2·conv(w, a_rev)[n‑1:n‑1+n]     (a_rev = a[::-1])
   * Apply a mirror‑descent (exponentiated) update
        a ← a·exp(−η·v)   with momentum v,
     and renormalise onto the simplex (division by the sum).
   * γ grows from a small value to a large one, turning the smooth
     surrogate into the true max.  η decays as 1/√t.
4. **Hill‑climbing polish** – stochastic swaps / mass transfers /
   neighbour moves accepted only when the exact score improves.
5. **Fallback** – if nothing succeeded, returns a tiny uniform vector.

All steps respect the 1000 s wall‑time budget.  The final list of
non‑negative floats is returned for ``evaluate_sequence``.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# FFT‑based linear convolution (fast, uses real‑valued FFT)
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Linear convolution of two real vectors using FFT."""
    m = len(a) + len(b) - 1
    n_fft = next_pow2(m)
    A = np.fft.rfft(a, n=n_fft)
    B = np.fft.rfft(b, n=n_fft)
    C = A * B
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


# ----------------------------------------------------------------------
# Objective helpers (fast evaluation when Σa = 1)
# ----------------------------------------------------------------------


def compute_score(a: np.ndarray) -> float:
    """
    Returns 2·n·max(a*a) for a vector a that sums to 1.
    This is exactly the value returned by ``evaluate_sequence`` for the
    same vector (because the objective is homogeneous).
    """
    n = a.shape[0]
    conv = convolution_fft(a, a)
    return 2.0 * n * float(np.max(conv))


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the probability simplex
    (non‑negative entries that sum to 1).
    """
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / v.shape[0])

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, v.shape[0] + 1)
    cond = u - cssv / ind > 0
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


# ----------------------------------------------------------------------
# Smooth (soft‑max) exponentiated gradient descent
# ----------------------------------------------------------------------


def smooth_expgd(
    init_a: np.ndarray,
    deadline: float,
    eta0: float = 0.4,
    beta: float = 0.9,
    gamma_start: float = 3.0,
    gamma_end: float = 200.0,
    max_iter: int = 500_000,
) -> tuple[np.ndarray, float]:
    """
    Mirror descent on a smooth surrogate of the max‑convolution objective.
    Returns the best vector seen and its score.
    """
    a = init_a.astype(float).copy()
    n = a.shape[0]

    best_a = a.copy()
    best_score = compute_score(a)

    # momentum term for the gradient
    v = np.zeros_like(a)

    for t in range(1, max_iter + 1):
        if time.time() >= deadline:
            break

        # ----- current autocorrelation -----
        conv = convolution_fft(a, a)          # length 2n‑1
        max_val = np.max(conv)

        # ----- soft‑max weights -----
        gamma = gamma_start + (gamma_end - gamma_start) * (t / max_iter)
        shifted = conv - max_val               # for numerical stability
        w = np.exp(gamma * shifted)
        w_sum = np.sum(w)
        if w_sum == 0.0:
            w = np.full_like(w, 1.0 / w.size)
        else:
            w = w / w_sum                       # now Σ w = 1

        # ----- gradient of the smoothed objective -----
        a_rev = a[::-1]
        grad_full = convolution_fft(w, a_rev)  # length 3n‑2
        grad = 2.0 * grad_full[n - 1 : n - 1 + n]   # slice of length n

        # ----- momentum -----
        v = beta * v + (1.0 - beta) * grad

        # ----- step size (diminishing) -----
        eta = eta0 / np.sqrt(t)

        # ----- multiplicative (exponentiated) update -----
        # Clip to avoid overflow/underflow in the exponent
        update = np.exp(-eta * np.clip(v, -20.0, 20.0))
        a = a * update

        # ----- renormalise onto the simplex (division is enough) -----
        s = a.sum()
        if s == 0.0:
            a = np.full_like(a, 1.0 / n)
        else:
            a = a / s

        # ----- keep the best -----
        cur_score = compute_score(a)
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()

        # occasional jitter to escape plateaus
        if t % 5000 == 0:
            a += np.random.randn(n) * (eta0 * 1e-4 / np.sqrt(t))
            a = project_to_simplex(a)

    return best_a, best_score


# ----------------------------------------------------------------------
# Simple stochastic hill‑climber (swap / mass moves)
# ----------------------------------------------------------------------


def hill_climb(seq, deadline, max_steps=300_000):
    """
    Stochastic local search that tries swaps, direct mass transfers and
    neighbour transfers.  Uses the exact objective for acceptance.
    """
    a = np.array(seq, dtype=float)
    n = a.shape[0]

    best_a = a.copy()
    best_score = compute_score(best_a)

    step_scale = 0.1  # fraction of mass moved in a transfer

    while max_steps > 0 and time.time() < deadline:
        move_type = np.random.rand()
        cand = best_a.copy()

        if move_type < 0.33:
            # swap two entries
            i, j = np.random.randint(n, size=2)
            cand[i], cand[j] = cand[j], cand[i]
        elif move_type < 0.66:
            # transfer a fraction from i to j
            i, j = np.random.randint(n, size=2)
            if i != j:
                delta = step_scale * cand[i]
                cand[i] -= delta
                cand[j] += delta
                cand[i] = max(cand[i], 0.0)
        else:
            # neighbour transfer (i -> i+1)
            i = np.random.randint(0, n - 1)
            delta = step_scale * cand[i]
            cand[i] -= delta
            cand[i + 1] += delta

        # project back onto the simplex to keep feasibility
        cand = project_to_simplex(cand)
        cur_score = compute_score(cand)

        if cur_score < best_score:
            best_score = cur_score
            best_a = cand
            step_scale = min(step_scale * 1.08, 0.5)   # grow step after success
        else:
            step_scale *= 0.97                         # shrink after failure

        max_steps -= 1

    return best_a.tolist(), best_score

            best_val = fallback_val
            best_seq = fallback
    except Exception:
        pass

    # --------------------------------------------------------------
    # 1.  Choose a suitable length (longer vectors give a better theoretical
    #     lower bound, but are slower to optimise)
    # --------------------------------------------------------------
    time_left = deadline - time.time()
    if time_left > 600:
        n = 4096
    elif time_left > 300:
        n = 3072
    else:
        n = 2048
    print(f"Using vector length n = {n}")

    # --------------------------------------------------------------
    # 2.  Multiple random restarts with smooth SGD
    # --------------------------------------------------------------
    alphas = [0.5, 1.0, 2.0]          # Dirichlet concentration parameters
    max_restarts = 3
    for restart_id in range(max_restarts):
        if time.time() >= deadline - 40:    # keep ~40 s for polishing
            break

        # allocate a slice of the remaining budget for this restart
        remaining_time = deadline - time.time() - 20   # reserve a bit for hill‑climb

    # 3.  Final hill‑climbing refinement (exact objective)
    # --------------------------------------------------------------

        polished_seq, polished_val = hill_climb(
            best_seq,
            deadline=deadline,
            max_steps=500_000,
        )
        if polished_val < best_val:
            best_val = polished_val
            best_seq = polished_seq
            print(f"Hill‑climb improved best to {best_val:.6f}")

    # --------------------------------------------------------------
    # 4.  Safeguard – fall back to a tiny uniform vector if everything failed
    # --------------------------------------------------------------
    if best_seq is None:
        n = 256
        best_seq = [1.0 / n] * n
        best_val = evaluate_sequence(best_seq)

    final_score = evaluate_sequence(best_seq)
    print(f"\nSearch finished. Final reported score = {final_score:.6f}")
