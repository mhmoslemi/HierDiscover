"""
Propose a non‑negative step‑function (vector) with a low
``evaluate_sequence`` score.

The algorithm works in three stages and is deliberately written to
run under the 1000 s budget:

1. **Candidate generation** – a handful of parametric families
   (arcsine, Beta‑distribution, power‑law, random Dirichlet) together
   with the previously discovered construction (if the global
   variable ``height_sequence_1`` exists).  All candidates are
   normalised to the simplex (∑a = 1) and a cheap FFT‑based score
   ``2·n·max(conv(a,a))`` is used to pick the most promising one.

2. **Smooth mirror‑descent** – a multiplicative (exponentiated‑gradient)
   update on the simplex.  The gradient is taken w.r.t. a *soft*
   max‑weight  ``w_i ∝ exp(β·(conv_i−max(conv)))``; this gives a
   smooth approximation of the true sub‑gradient while staying stable.
   Step‑size follows ηₜ = η₀/√t.  The best iterate (according to the
   *exact* objective) is kept.

3. **Hard sub‑gradient polishing** – the soft weights are replaced by
   the true uniform weight on the (near) maximisers of the convolution.
   The same exponentiated‑gradient scheme with a 1/√t schedule is run
   until the time budget is exhausted.

All operations are performed with FFT‑based convolutions (O(n log n)).
The final answer is returned as a plain ``list[float]``; the official
``evaluate_sequence`` routine will re‑scale it internally, therefore we
optimise under the simplex constraint (∑a = 1) and never hit the
clipping limits.
"""

import time
import numpy as np
from typing import List, Tuple

# ----------------------------------------------------------------------
# Low‑level numeric utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """
    Linear convolution using real FFT (full length, zero‑padded to a power‑of‑2).
    Returns an array of length len(a)+len(b)-1.
    """
    m = len(a) + len(b) - 1
    n_fft = next_pow2(m)
    A = np.fft.rfft(a, n=n_fft)
    B = np.fft.rfft(b, n=n_fft)
    C = A * B
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


def normalize_to_simplex(v: np.ndarray) -> np.ndarray:
    """Project non‑negative vector onto the simplex (sum = 1)."""
    total = v.sum()
    if total <= 0.0:
        # fallback to uniform distribution
        return np.full_like(v, 1.0 / len(v))
    return v / total


def fast_score(a: np.ndarray) -> float:
    """
    Fast surrogate of the official objective.
    The vector a is assumed to be non‑negative and already normalised (sum = 1).

    Returns 2·n·max(conv(a,a))   (the denominator is 1 by construction).
    """
    n = len(a)
    conv = convolution_fft(a, a)
    return 2.0 * n * float(np.max(conv))


def gradient_from_weights(a: np.ndarray, w: np.ndarray) -> np.ndarray:
    """
    Compute ∇_a Σ_k w_k·(a*a)_k   where * denotes linear convolution.

    The formula is   g_j = 2· Σ_m a_m·w_{m+j}.
    Implemented with a single FFT‑based convolution.
    """
    # reverse w so that linear convolution produces the needed shifted sums
    w_rev = w[::-1]
    conv = convolution_fft(a, w_rev)          # length = 3·n‑2
    idx = 2 * len(a) - 2                       # position 2n‑2
    grads = 2.0 * conv[idx - np.arange(len(a))]
    return grads


# ----------------------------------------------------------------------
# Parametric families for the initial sweep
# ----------------------------------------------------------------------


def arcsine_shape(n: int) -> np.ndarray:
    """Discrete arcsine distribution (≈ 1/√(i·(n‑1‑i))) on the simplex."""
    i = np.arange(n, dtype=float)
    eps = 0.25
    vals = 1.0 / np.sqrt((i + 0.5) * (n - 0.5 - i) + eps)
    return normalize_to_simplex(vals)


def beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    """Beta(α,β) density sampled on a grid and projected on the simplex."""
    x = np.linspace(0.0, 1.0, n)
    # add a tiny epsilon to avoid infinities when α<1 or β<1
    eps = 1e-12
    vals = (x + eps) ** (alpha - 1.0) * (1.0 - x + eps) ** (beta - 1.0)
    return normalize_to_simplex(vals)


def powerlaw_shape(n: int, p: float, eps: float) -> np.ndarray:
    """Symmetric power‑law: a_i ∝ ((i+ε)(n‑1‑i+ε))⁻ᵖ."""
    i = np.arange(n, dtype=float)
    base = ((i + eps) * (n - 1 - i + eps)) ** (-p)
    return normalize_to_simplex(base)


def uniform_shape(n: int) -> np.ndarray:
    """Uniform distribution on the simplex."""
    return np.full(n, 1.0 / n)


def random_dirichlet(n: int, rng: np.random.Generator) -> np.ndarray:
    """Draw a random point on the simplex using exponential sampling."""
    samp = rng.exponential(1.0, size=n)
    return normalize_to_simplex(samp)


# ----------------------------------------------------------------------
# Refinement methods
# ----------------------------------------------------------------------


def refine_smooth(
    a: np.ndarray,
    deadline: float,
    *,
    eta0: float = 1.0,
    beta: float = 12.0,
    report_every: int = 0,
) -> Tuple[np.ndarray, float]:
    """
    Multiplicative mirror‑descent with a *soft* max‑weight.
    The weight on each convolution entry is
        w_i ∝ exp(β·(conv_i – max(conv))).
    This gives a smooth approximation of the true sub‑gradient.
    Returns the best iterate (according to the exact objective) and its
    exact score.
    """
    a = normalize_to_simplex(a.astype(float).copy())
    n = len(a)
    best_a = a.copy()
    conv = convolution_fft(a, a)
    best_sc = 2.0 * n * float(np.max(conv))

    it = 0
    while time.time() < deadline:
        it += 1

        conv = convolution_fft(a, a)
        max_val = float(np.max(conv))

        # softmax–type weights (kept numerically stable)
        exp_vals = np.exp(beta * (conv - max_val))
        w = exp_vals / np.sum(exp_vals)

        grad = gradient_from_weights(a, w)

        # step‑size schedule ηₜ = η₀ / √t
        eta = eta0 / np.sqrt(it)

        # exponentiated‑gradient update, with clipping to avoid overflow
        grad = np.clip(grad, -20.0, 20.0)
        a = a * np.exp(-eta * grad)
        a = normalize_to_simplex(a)

        cur_sc = 2.0 * n * max_val
        if cur_sc < best_sc:
            best_sc = cur_sc
            best_a = a.copy()

        if report_every and (it % report_every == 0):
            print(f"[smooth] iter={it:,d} cur={cur_sc:.6f} best={best_sc:.6f}")

    return best_a, best_sc


def refine_subgradient(
    a: np.ndarray,
    deadline: float,
    *,
    max_iter: int = 150_000,
    eta0: float = 0.7,
    report_every: int = 0,
) -> Tuple[np.ndarray, float]:
    """
    Classical (hard) sub‑gradient descent using the true max‑weight:
    w = uniform on the set of (near) maximisers of the convolution.
    Returns the best iterate seen and its exact score.
    """
    a = normalize_to_simplex(a.astype(float).copy())
    n = len(a)
    best_a = a.copy()
    conv = convolution_fft(a, a)
    best_sc = 2.0 * n * float(np.max(conv))

    for it in range(1, max_iter + 1):
        if time.time() >= deadline:
            break

        conv = convolution_fft(a, a)
        max_val = float(np.max(conv))

        # binary weight on (almost) maximisers
        w = np.zeros_like(conv)
        # tolerance allows for numerical noise
        tol = 1e-14
        max_idx = np.where(conv >= max_val - tol)[0]
        w[max_idx] = 1.0 / len(max_idx)

        grad = gradient_from_weights(a, w)
        grad = np.clip(grad, -20.0, 20.0)

        eta = eta0 / np.sqrt(it)
        a = a * np.exp(-eta * grad)
        a = normalize_to_simplex(a)

        cur_sc = 2.0 * n * max_val
        if cur_sc < best_sc:
            best_sc = cur_sc
            best_a = a.copy()

        if report_every and (it % report_every == 0):
            print(f"[subgrad] iter={it:,d} cur={cur_sc:.6f} best={best_sc:.6f}")

    return best_a, best_sc


# ----------------------------------------------------------------------
# Helper for up‑sampling an old construction
# ----------------------------------------------------------------------


def upsample_to_length(old: np.ndarray, new_n: int) -> np.ndarray:
    """
    Linear interpolation of a vector ``old`` (length m) to length ``new_n``.
    The result is projected onto the simplex.
    """
    if len(old) == new_n:
        return normalize_to_simplex(old)

    old_x = np.linspace(0.0, 1.0, len(old))
    new_x = np.linspace(0.0, 1.0, new_n)
    interp = np.interp(new_x, old_x, old)
    return normalize_to_simplex(interp)


# ----------------------------------------------------------------------
# Main entry point required by the competition harness
# ----------------------------------------------------------------------


def propose_candidate(
    seed: int = 42,
    budget_s: int = 1000,
    **kwargs,
) -> List[float]:
    """
    Search for a non‑negative step‑function with a low ``evaluate_sequence``
    score.  The routine respects the supplied time budget (default 1000 s) and
    always returns a feasible list of non‑negative floats.

    The search uses a few simple families for initialisation,
    a smooth mirror‑descent phase, and a final hard sub‑gradient polishing.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    # keep a safety margin from the hard timeout
    deadline = start_time + budget_s - 2.0

    # ------------------------------------------------------------------
    # Choose a working length.
    # Larger lengths give a finer discretisation of the continuous problem.
    # 4096 is a good trade‑off for the 1000 s budget.
    # ------------------------------------------------------------------
    n = 4096

    # ------------------------------------------------------------------
    # 1.  Gather a diverse set of initial candidates
    # ------------------------------------------------------------------
    candidates: List[np.ndarray] = []

    # a)  previously discovered construction (if it exists)
    try:
        fallback = np.array([float(x) for x in height_sequence_1], dtype=float)
        if fallback.ndim == 1 and fallback.size > 0:
            fallback = upsample_to_length(fallback, n)
            candidates.append(fallback)
            print("[fallback] up‑sampled to length =", n)
    except NameError:
        pass

    # b)  classic families
    candidates.append(arcsine_shape(n))
    candidates.append(uniform_shape(n))

    # c)  random Dirichlet draws
    for _ in range(4):
        candidates.append(random_dirichlet(n, rng))

    # d)  random Beta families (α,β ∈ (0.1, 2.5))
    for _ in range(5):
        a = rng.uniform(0.1, 2.5)
        b = rng.uniform(0.1, 2.5)
        candidates.append(beta_shape(n, a, b))

    # e)  random power‑law shapes
    for _ in range(3):
        p = rng.uniform(0.30, 0.80)
        eps = rng.uniform(0.0, max(5.0, 0.015 * n))
        candidates.append(powerlaw_shape(n, p, eps))

    # ------------------------------------------------------------------
    # 2.  Fast ranking – keep the best few according to the cheap score
    # ------------------------------------------------------------------
    scored = [(fast_score(c), c) for c in candidates]
    scored.sort(key=lambda x: x[0])
    # keep the top‑3 for possible multi‑restart later
    top_candidates = [c for _, c in scored[:3]]
    best_fast_score = scored[0][0]
    print(f"[initial] best fast score = {best_fast_score:.6f}")

    # ------------------------------------------------------------------
    # 3.  Smooth mirror‑descent (soft max) – allocate ~40 % of remaining time
    # ------------------------------------------------------------------
    time_used = time.time() - start_time
    time_left = max(0.0, deadline - time.time())
    smooth_deadline = time.time() + 0.40 * time_left

    # start from the overall best fast candidate
    current = top_candidates[0].copy()
    refined_smooth, smooth_score = refine_smooth(
        a=current,
        deadline=smooth_deadline,
        eta0=1.0,
        beta=12.0,
        report_every=0,
    )
    print(f"[smooth] best exact score = {smooth_score:.6f}")

    # evaluate with the official routine (expensive but done only once)
    val = evaluate_sequence(refined_smooth.tolist())
    print(f"[smooth] official score = {val:.6f}")

    best_seq: List[float] = refined_smooth.tolist()
    best_val: float = float(val)

    # ------------------------------------------------------------------
    # 4.  Hard sub‑gradient polishing – use the rest of the budget
    # ------------------------------------------------------------------
    if time.time() < deadline - 5.0:
        print("[subgrad] start hard polishing")
        polished, _ = refine_subgradient(
            a=refined_smooth,
            deadline=deadline,
            max_iter=200_000,
            eta0=0.7,
            report_every=0,
        )
        final_val = evaluate_sequence(polished.tolist())
        print(f"[subgrad] official score = {final_val:.6f}")

        if final_val < best_val:
            best_val = final_val
            best_seq = polished.tolist()
            print("*** new global best after hard polish ***")

    # ------------------------------------------------------------------
    # 5.  (Optional) try a second top candidate if time permits
    # ------------------------------------------------------------------
    if time.time() < deadline - 3.0:
        # a quick run from the second candidate, limited to the leftover time
        spare = deadline - time.time()
        if spare > 5.0:
            print("[extra] polishing second best candidate")
            second = top_candidates[1].copy()
            # short smooth phase
            tmp_dead = time.time() + 0.5 * spare
            tmp_smooth, _ = refine_smooth(
                a=second,
                deadline=tmp_dead,
                eta0=1.0,
                beta=12.0,
                report_every=0,
            )
            # remaining time for hard polish
            tmp_final, _ = refine_subgradient(
                a=tmp_smooth,
                deadline=deadline,
                max_iter=150_000,
                eta0=0.7,
                report_every=0,
            )
            val2 = evaluate_sequence(tmp_final.tolist())
            print(f"[extra] official score = {val2:.6f}")
            if val2 < best_val:
                best_val = val2
                best_seq = tmp_final.tolist()
                print("*** new global best from extra run ***")

    # ------------------------------------------------------------------
    # 6.  Guard – ensure we always return a valid sequence
    # ------------------------------------------------------------------
    if not best_seq:
        # fallback to a safe uniform vector (should never happen)
        best_seq = uniform_shape(256).tolist()
        best_val = evaluate_sequence(best_seq)

    print(f"\n[finished] best reported score = {best_val:.6f}")
    return best_seq
