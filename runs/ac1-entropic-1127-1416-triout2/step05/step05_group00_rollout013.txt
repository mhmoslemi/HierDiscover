def smoothed_descent(init_a, time_deadline, max_iter=5000, step0=0.1, start_alpha=1.0, alpha_factor=1.02):
    a = init_a.copy()
    best_a = a.copy()
    best_score = compute_score(a)
    n = len(a)
    ones_n = np.ones(n, dtype=float)
    alpha = start_alpha
    for it in range(1, max_iter+1):
        if time.time() >= time_deadline:
            break
        conv = convolution_fft(a, a)
        # smoothing: use softmax
        maxc = np.max(conv)
        exp_vals = np.exp(alpha * (conv - maxc))
        w = exp_vals / np.sum(exp_vals)  # size 2n-1
        # gradient
        a_rev = a[::-1]
        conv_rev = convolution_fft(w, a_rev)  # length 3n-2
        grad_first = conv_rev[n-1:n-1+n]  # length n
        conv_win = convolution_fft(w, ones_n)  # length 3n-2
        window_sum = conv_win[n-1:n-1+n]
        grad = grad_first + a * window_sum
        # step size schedule
        step = step0 / np.sqrt(it)
        a_new = a - step * grad
        # project onto simplex
        a_new = project_to_simplex(a_new)
        # evaluate
        cur_score = compute_score(a_new)
        if cur_score < best_score:
            best_score = cur_score
            best_a = a_new.copy()
        # update
        a = a_new
        # increase alpha gradually
        alpha = min(alpha * alpha_factor, 100.0)
        # optional random perturbation occasional
        if it % 500 == 0:
            # small random jitter
            a += np.random.randn(n) * (step0 * 0.01 / np.sqrt(it))
            a = project_to_simplex(a)
    return best_a, best_score

def random_init(n):
    # random support size k between 0.1*n and 0.5*n
    k = max(2, int(np.random.uniform(0.1, 0.5) * n))
    idx = np.random.choice(n, size=k, replace=False)
    # generate random positive weights via Dirichlet
    weights = np.random.rand(k)
    weights = weights / np.sum(weights)
    a = np.zeros(n)
    a[idx] = weights
    return a

"""
Propose a low‑score step function.

Algorithm
---------
1. Random‐initialisation:
   * Generate a handful of sequences on several lengths (256, 512, 1024, 2048)
   * The support size is random, the masses are Dirichlet‑distributed.
   * Keep the best few candidates (lowest `compute_score`).

2. Smoothed‑max gradient descent (annealed soft‑max):
   * Use the smooth approximation  f(a)=log‑sum‑exp(α·(a⊛a))/α .
   * Gradient of f is a linear combination of the gradients of the individual
     convolution entries; this can be computed with three FFT‑based convolutions.
   * Start with a small α (smooth) and increase it gradually – this avoids
     getting stuck in a bad local maximum.
   * After each step the vector is projected onto the probability simplex
     (non‑negative, sum = 1).

3. Hill‑climb polishing:
   * Stochastic swaps / mass‑transfer moves accepted only when they improve the
     *exact* score (the original `evaluate_sequence` objective).

All phases respect the 1000 s wall‑time budget; the most time‑consuming part
    – the gradient descent – works on the smallest promising length (usually 256)
    and therefore can perform many thousands of iterations.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Basic utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full convolution a ⊛ b using a real FFT (fast)."""
    m = len(a) + len(b) - 1
    n_fft = next_pow2(m)
    A = np.fft.rfft(a, n=n_fft)
    B = np.fft.rfft(b, n=n_fft)
    C = A * B
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


def compute_score(a: np.ndarray) -> float:
    """
    Fast replica of `evaluate_sequence` when the vector lies on the simplex
    (∑a = 1). Returns 2·n·max(a⊛a).
    """
    n = a.shape[0]
    conv = convolution_fft(a, a)
    return 2.0 * n * float(np.max(conv))


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection onto the probability simplex."""
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / v.shape[0])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, v.shape[0] + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.where(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


# ----------------------------------------------------------------------
# Random initialisation
# ----------------------------------------------------------------------


def random_init(n: int) -> np.ndarray:
    """
    Produce a random non‑negative vector of length n on the simplex.
    The support size k is drawn uniformly in [0.1·n, 0.5·n].
    Masses on the support are Dirichlet(1,…,1) distributed.
    """
    k = max(2, int(np.random.uniform(0.10, 0.50) * n))
    idx = np.random.choice(n, size=k, replace=False)
    weights = np.random.rand(k)
    weights /= np.sum(weights)
    a = np.zeros(n, dtype=float)
    a[idx] = weights
    return a


# ----------------------------------------------------------------------
# Smoothed‑max gradient descent (annealed soft‑max)
# ----------------------------------------------------------------------


def smoothed_descent(
    a0: np.ndarray,
    time_deadline: float,
    max_iter: int = 20_000,
    step0: float = 0.15,
    start_alpha: float = 1.0,
    alpha_factor: float = 1.02,
) -> tuple[np.ndarray, float]:
    """
    Minimise the maximal entry of a⊛a while staying on the simplex.
    The objective is approximated by a soft‑max with temperature 1/α.
    """
    a = a0.copy()
    n = a.shape[0]
    best_a = a.copy()
    best_score = compute_score(a)

    # helpers that stay constant for the whole run
    ones_n = np.ones(n, dtype=float)

    alpha = start_alpha

    for it in range(1, max_iter + 1):
        if time.time() >= time_deadline:
            break

        # full convolution of current iterate with itself
        conv = convolution_fft(a, a)               # length 2n‑1
        maxc = np.max(conv)

        # smoothed max (soft‑max) weights
        exp_vals = np.exp(alpha * (conv - maxc))    # stability trick
        w = exp_vals / np.sum(exp_vals)            # shape (2n‑1,)

        # ----- gradient of the smoothed objective -----
        # term 1 : sum_k w_k * a_{k‑j}
        a_rev = a[::-1]
        conv_rev = convolution_fft(w, a_rev)       # length 3n‑2
        grad_term1 = conv_rev[n - 1 : n - 1 + n]    # slice of length n

        # term 2 : a_j * sum_{k=j}^{j+n‑1} w_k
        conv_win = convolution_fft(w, ones_n)      # length 3n‑2
        window_sum = conv_win[n - 1 : n - 1 + n]   # length n
        grad_term2 = a * window_sum

        grad = grad_term1 + grad_term2

        # step size schedule
        step = step0 / np.sqrt(it)
        a_new = a - step * grad
        a_new = project_to_simplex(a_new)

        cur_score = compute_score(a_new)
        if cur_score < best_score:
            best_score = cur_score
            best_a = a_new.copy()

        a = a_new

        # anneal α (make the soft‑max sharper)
        alpha = min(alpha * alpha_factor, 200.0)

        # occasional tiny random jitter to escape flat regions
        if it % 500 == 0:
            a += np.random.randn(n) * (step0 * 0.005 / np.sqrt(it))
            a = project_to_simplex(a)

    return best_a, best_score


# ----------------------------------------------------------------------
# Hill‑climber (swap / mass transfer)
# ----------------------------------------------------------------------


def hill_climb_permutation(seq, time_deadline, max_steps=400_000):
    """
    Stochastic hill‑climb that swaps entries or transfers a fraction of mass.
    The true (unsmoothed) objective is used for acceptance.
    """
    a = np.array(seq, dtype=float)
    n = a.shape[0]

    best_a = a.copy()
    best_score = compute_score(best_a)

    step_scale = 0.12

    while max_steps > 0 and time.time() < time_deadline:
        move_type = np.random.rand()
        cand = best_a.copy()

        if move_type < 0.5:
            # simple swap
            i, j = np.random.randint(n, size=2)
            cand[i], cand[j] = cand[j], cand[i]
        else:
            # move a fraction from i to j
            i, j = np.random.randint(n, size=2)
            if i != j and cand[i] > 0:
                delta = step_scale * cand[i]
                cand[i] -= delta
                cand[j] += delta

        cand = project_to_simplex(cand)
        cur_score = compute_score(cand)

        if cur_score < best_score:
            best_score = cur_score
            best_a = cand
            step_scale = min(step_scale * 1.07, 0.5)
        else:
            step_scale *= 0.96

        max_steps -= 1

    return best_a.tolist(), best_score
