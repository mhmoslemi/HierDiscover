"""
propose_candidate – 2024‑11‑27 version
-------------------------------------------------
Goal: produce a non‑negative step sequence that minimizes the official
``evaluate_sequence`` objective

    2·n·max(conv(a,a)) / (sum(a)**2)

The value is homogeneous, therefore we work with vectors that sum to 1.
The core of the search is an *alternating linear‑program* scheme
described in the statement:

    • start from a simple parametric shape (arcsine, beta, power …)
    • compute M = max(conv(f,f))
    • solve the LP   maximise   Σ g_j
                     s.t.      f * g ≤ M   (component‑wise)
                     g_j ≥ 0
      The LP pushes mass to positions where f is small, thus lowering the
      worst convolution value.
    • renormalise g to Σ g = 1 and replace f ← (1‑t)·f + t·g
      (t ∈ (0,1) controls the step size)
    • repeat until M stops decreasing.

The LP is solved with CVXPY (ECOS).  After a few LP iterations we
polish the best vector with a smooth‑max Adam optimiser and a short
sub‑gradient descent, both projected onto the probability simplex.
All computations stay well inside the 1000 s limit.

If cvxpy is unavailable the routine falls back to the earlier random‑
search + Adam pipeline, so the code always returns a valid sequence.
"""

import time
import numpy as np
from typing import List, Tuple

# ----------------------------------------------------------------------
# Optional import of cvxpy – the LP engine
# ----------------------------------------------------------------------
try:
    import cvxpy as cp
    _HAS_CVX = True
except Exception:               # pragma: no cover
    _HAS_CVX = False
    print("cvxpy not available – falling back to pure‑gradient search.")

# ----------------------------------------------------------------------
# Low‑level utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Linear convolution via real FFT (fast for the lengths we use)."""
    m = a.size + b.size - 1
    n_fft = next_pow2(m)
    A = np.fft.rfft(a, n=n_fft)
    B = np.fft.rfft(b, n=n_fft)
    C = A * B
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


def compute_score(a: np.ndarray) -> float:
    """
    Fast surrogate for the official score.
    The vector is assumed to sum to 1, therefore the denominator disappears.
    """
    n = a.size
    conv = convolution_fft(a, a)
    return 2.0 * n * float(np.max(conv))


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection onto the probability simplex {x ≥ 0, Σx = 1}.
    Duchi et al., 2008.
    """
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / v.size)

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho = np.where(u - cssv / np.arange(1, v.size + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

# Shape factories (already normalised to sum = 1)
# ----------------------------------------------------------------------


def arcsine_initial(n: int) -> np.ndarray:
    """Classic arcsine law a_i ∝ 1/√((i+½)*(n‑½‑i))."""
    i = np.arange(n, dtype=float)
    eps = 0.25
    vals = 1.0 / np.sqrt((i + 0.5) * (n - 0.5 - i) + eps)
    vals /= vals.sum()
    return vals


def generate_beta_shape(n: int, alpha: float, beta: float, eps: float) -> np.ndarray:
    """
    (i+ε)^{α‑1}·(n‑1‑i+ε)^{β‑1} on {0,…,n‑1}, normalised to Σ=1.
    """
    i = np.arange(n, dtype=float)
    left = (i + eps) ** (alpha - 1.0)
    right = (n - 1 - i + eps) ** (beta - 1.0)
    base = left * right
    total = base.sum()
    if not np.isfinite(total) or total <= 0.0:
        return np.full(n, 1.0 / n)
    return base / total


def generate_power_shape(n: int, p: float, eps: float) -> np.ndarray:
    """
    Generalised arcsine a_i ∝ ((i+½)*(n‑½‑i)+ε)^{-p},   p>0.
    """
    i = np.arange(n, dtype=float)
    denom = ((i + 0.5) * (n - 0.5 - i) + eps) ** p
    vals = 1.0 / denom
    total = vals.sum()
    if not np.isfinite(total) or total <= 0.0:
        return np.full(n, 1.0 / n)
    return vals / total


def generate_triangular_shape(n: int, peak: int) -> np.ndarray:
    """Symmetric triangular bump whose apex sits at ``peak``."""
    left = np.arange(peak + 1, dtype=float)
    right = np.arange(n - peak - 1, dtype=float)[::-1]
    seq = np.concatenate([left, right])
    seq /= seq.sum()
    return seq


def generate_mixture_shape(n: int, weight_arc: float) -> np.ndarray:
    """Mixture (weight_arc)·arcsine + (1‑weight_arc)·uniform."""
    arc = arcsine_initial(n)
    uni = np.full(n, 1.0 / n)
    mix = weight_arc * arc + (1.0 - weight_arc) * uni
    mix /= mix.sum()
    return mix


# ----------------------------------------------------------------------
# Optimisers – smooth Adam + exact sub‑gradient (identical to the previous
# implementation, retained for polishing after the LP phase).
# ----------------------------------------------------------------------


def smooth_gradient(a: np.ndarray, lam: float) -> np.ndarray:
    """
    Gradient of the soft‑max approximation of max(conv(a,a)).
    Returns ∇_a (2·n·softmax_max) – the sum‑to‑one constraint is handled
    later by a projection onto the simplex.
    """
    n = a.size
    conv = convolution_fft(a, a)
    cmax = np.max(conv)

    # stable soft‑max over the (2n‑1) convolution entries
    exp_term = np.exp(lam * (conv - cmax))
    w = exp_term / np.sum(exp_term)

    # weighted sub‑gradient: 2·∑_k w_k shift(a,k)
    w_rev = w[::-1]
    conv_rev = convolution_fft(a, w_rev)          # length 2n‑1
    L = 2 * n - 1
    weighted_grad = 2.0 * conv_rev[L - 1 - np.arange(n)]

    return 2.0 * n * weighted_grad


def adam_optimize_smooth(initial_a: np.ndarray,
                         deadline: float,
                         max_iter: int = 100_000,
                         lr0: float = 0.12,
                         lam_start: float = 2.0,
                         lam_end: float = 80.0,
                         noise_every: int = 2000,
                         noise_scale: float = 2e-5) -> Tuple[np.ndarray, float]:
    """
    Adam optimiser on the smooth surrogate.
    Returns the best vector (projected onto the simplex) and its fast score.
    """
    a = project_to_simplex(initial_a.astype(float).copy())
    best_a = a.copy()
    best_sc = compute_score(a)

    beta1, beta2 = 0.9, 0.999
    eps = 1e-8
    m = np.zeros_like(a)
    v = np.zeros_like(a)

    for it in range(1, max_iter + 1):
        if time.time() >= deadline:
            break

        lam = lam_start + (lam_end - lam_start) * min(1.0, it / max_iter)

        grad = smooth_gradient(a, lam)

        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)

        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        step = lr0 * m_hat / (np.sqrt(v_hat) + eps)
        a_new = a - step

        if noise_every and (it % noise_every == 0):
            a_new += np.random.randn(a.size) * noise_scale

        a_new = project_to_simplex(a_new)

        cur_sc = compute_score(a_new)
        if cur_sc < best_sc:
            best_sc = cur_sc
            best_a = a_new.copy()

        a = a_new

        if it % 5000 == 0:
            print(f"    AdamSmooth it={it:,d} cur={cur_sc:.6f} best={best_sc:.6f}")

    return best_a, best_sc


def subgradient_descent_exact(initial_a: np.ndarray,
                              deadline: float,
                              max_iter: int = 80_000,
                              step0: float = 0.03) -> np.ndarray:
    """
    Sub‑gradient descent on the exact objective f(a)=2·n·max(conv(a,a)).
    Uses a diminishing step η_t = step0 / sqrt(t) and maintains feasibility
    via a projection onto the simplex.
    """
    a = project_to_simplex(initial_a.copy())
    best_a = a.copy()
    best_sc = compute_score(a)
    n = a.size
    L = 2 * n - 1

    for it in range(1, max_iter + 1):
        if time.time() >= deadline:
            break

        conv = convolution_fft(a, a)
        max_val = float(conv.max())

        # tolerance to collect all near‑max entries (sub‑gradient set)
        tol = max_val * 1e-12 + 1e-14
        mask = (np.abs(conv - max_val) <= tol).astype(float)

        # weighted sub‑gradient as described in the analysis
        mask_rev = mask[::-1]
        conv_rev = convolution_fft(a, mask_rev)
        weighted_grad = 2.0 * conv_rev[L - 1 - np.arange(n)]

        eta = step0 / np.sqrt(it)
        a = a - eta * weighted_grad
        a = project_to_simplex(a)

        cur_sc = compute_score(a)
        if cur_sc < best_sc:
            best_sc = cur_sc
            best_a = a.copy()

        if it % 5000 == 0:
            print(f"    SubGrad it={it:,d} cur={cur_sc:.6f} best={best_sc:.6f}")

    return best_a


# ----------------------------------------------------------------------
# Linear‑program machinery (only used when cvxpy is present)
# ----------------------------------------------------------------------


def _build_conv_matrix(f: np.ndarray) -> np.ndarray:
    """
    Build the matrix A (size (2n‑1) × n) such that   A @ g = conv(f,g)
    (ordinary linear convolution, not circular).
    """
    n = f.size
    rows = 2 * n - 1
    A = np.zeros((rows, n), dtype=float)
    for j in range(n):
        A[j:j + n, j] = f
    return A


def _solve_lp_max_sum(f: np.ndarray, M: float) -> np.ndarray | None:
    """
    Solve
          maximise   Σ g_j
          subject to  A(g) ≤ M   (component‑wise)
                      g ≥ 0
    where A(g) = conv(f,g).  Returns the optimal g (as a dense ndarray) or
    None on failure.
    """
    if not _HAS_CVX:                 # pragma: no cover
        return None

    n = f.size
    A = _build_conv_matrix(f)

    g = cp.Variable(n, nonneg=True)
    constraints = [A @ g <= M]
    prob = cp.Problem(cp.Maximize(cp.sum(g)), constraints)

    try:
        prob.solve(solver=cp.ECOS, warm_start=True, max_iters=5000,
                   abstol=1e-9, reltol=1e-9)
    except Exception as e:           # pragma: no cover
        print("LP solve exception:", e)
        return None

    if g.value is None:
        return None
    # Numerical noise may produce tiny negatives – clip them.
    return np.maximum(g.value, 0.0)


def lp_iterative_refine(start_f: np.ndarray,
                        deadline: float,
                        max_iters: int = 25,
                        t: float = 0.25) -> Tuple[np.ndarray, float]:
    """
    Perform the alternating LP refinement described in the editorial.
    ``start_f`` must already be a probability vector (sum=1).
    Returns the refined vector and its fast score.
    """
    if not _HAS_CVX:                 # pragma: no cover
        # No LP engine – just return the start.
        return start_f, compute_score(start_f)

    f = start_f.copy()
    n = f.size

    for it in range(max_iters):
        if time.time() >= deadline - 1.0:
            break

        M = float(np.max(convolution_fft(f, f)))          # current upper bound
        g0 = _solve_lp_max_sum(f, M)

        if g0 is None or g0.sum() <= 0:
            break

        # Normalise the LP output to a probability vector.
        g = g0 / g0.sum()

        # Convex combination (keeps Σ = 1)
        f = (1.0 - t) * f + t * g

        new_M = float(np.max(convolution_fft(f, f)))
        print(f"LP iter {it + 1:2d}: M = {new_M:.6e}")

        # Stop if progress becomes negligible.
        if M - new_M < 1e-12:
            break

    # Final normalisation (safety) and fast score.
    f = f / f.sum()
    return f, compute_score(f)

def propose_candidate(seed: int = 42,
                     budget_s: int = 1000,
                     **kwargs) -> List[float]:
    """
    Search for a low‑score non‑negative step sequence.
    The routine combines a handful of parametric seeds,
    an LP‑based global improvement phase (when cvxpy is available)
    and a final Adam + sub‑gradient polishing stage.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 1.0     # tiny safety margin

    # ------------------------------------------------------------------
    # Configuration
    # ------------------------------------------------------------------
    n = 2048                                   # length of the step function
    best_vec: List[float] = []
    best_score: float = float("inf")

    # ------------------------------------------------------------------
    # 0.  Re‑use a previously stored vector if present (optional fallback)
    # ------------------------------------------------------------------
    if "height_sequence_1" in globals():
        try:
            fallback = [float(x) for x in globals()["height_sequence_1"]]
            val = evaluate_sequence(fallback)
            print(f"Fallback (height_sequence_1) official score = {val:.6f}")
            if val < best_score:
                best_score = val
                best_vec = fallback
        except Exception:
            pass

    # ------------------------------------------------------------------
    # 1.  Generate a small pool of promising seeds
    # ------------------------------------------------------------------
    seed_functions = [
        arcsine_initial,
        lambda n: generate_mixture_shape(n, weight_arc=rng.random()),
        lambda n: generate_beta_shape(n,
                                      alpha=rng.uniform(0.2, 4.0),
                                      beta=rng.uniform(0.2, 4.0),
                                      eps=rng.uniform(1e-6, 2.0)),
        lambda n: generate_power_shape(n,
                                       p=rng.uniform(0.2, 4.0),
                                       eps=rng.uniform(0.0, 2.0)),
        lambda n: generate_triangular_shape(n, peak=rng.integers(0, n)),
    ]

    # we keep the best 5 seeds (sorted by the fast surrogate score)
    seed_pool: List[Tuple[float, np.ndarray]] = []

    def try_insert(sc: float, vec: np.ndarray) -> None:
        if len(seed_pool) < 5:
            seed_pool.append((sc, vec))
        else:
            # replace the worst if the new one is better
            worst_idx = max(range(len(seed_pool)),
                            key=lambda i: seed_pool[i][0])
            if sc < seed_pool[worst_idx][0]:
                seed_pool[worst_idx] = (sc, vec)

    for f_gen in seed_functions:
        vec = f_gen(n).astype(float)
        vec = vec / vec.sum()                     # enforce Σ=1
        sc = compute_score(vec)
        try_insert(sc, vec)

    # ------------------------------------------------------------------
    # 2.  LP refinement for each seed (if the LP engine is present)
    # ------------------------------------------------------------------
    for idx, (_, seed_vec) in enumerate(sorted(seed_pool, key=lambda x: x[0]), 1):
        if time.time() >= deadline - 5.0:
            break
        print(f"\n--- LP refinement of seed {idx}/{len(seed_pool)} ---")
        refined, fast_sc = lp_iterative_refine(seed_vec,
                                               deadline,
                                               max_iters=20,
                                               t=0.25)

        # ----------------------------------------------------------------
        # 3.  Local polishing (Adam → sub‑gradient)
        # ----------------------------------------------------------------
        if time.time() < deadline - 2.0:
            # a) Adam on the smooth max surrogate
            refined, _ = adam_optimize_smooth(
                initial_a=refined,
                deadline=deadline,
                max_iter=120_000,
                lr0=0.09,
                lam_start=2.0,
                lam_end=100.0,
                noise_every=3000,
                noise_scale=1e-5,
            )
        if time.time() < deadline - 2.0:
            refined = subgradient_descent_exact(
                refined,
                deadline,
                max_iter=60_000,
                step0=0.025,
            )

        # ----------------------------------------------------------------
        # 4.  Official evaluation
        # ----------------------------------------------------------------
        try:
            val = evaluate_sequence(refined.tolist())
            print(f"    Official score = {val:.6f}")
            if val < best_score:
                best_score = val
                best_vec = refined.tolist()
                print(f"    *** New global best: {best_score:.6f} ***")
        except Exception:
            pass

    # ------------------------------------------------------------------
    # 5.  Global polishing of the overall best (time permitting)
    # ------------------------------------------------------------------
    if best_vec and time.time() < deadline - 2.0:
        a = np.array(best_vec, dtype=float)
        remaining = deadline - time.time()
        # a short Adam run followed by a sub‑gradient pass
        a, _ = adam_optimize_smooth(
            a,
            deadline,
            max_iter=int(min(200_000, remaining * 30)),
            lr0=0.07,
            lam_start=2.0,
            lam_end=80.0,
            noise_every=2500,
            noise_scale=5e-6,
        )
        a = subgradient_descent_exact(
            a,
            deadline,
            max_iter=int(min(150_000, (deadline - time.time()) * 30)),
            step0=0.02,
        )
        final_val = evaluate_sequence(a.tolist())
        print(f"Final polished official score = {final_val:.6f}")
        if final_val < best_score:
            best_score = final_val
            best_vec = a.tolist()
            print("*** Updated global best after final polish ***")

    # ------------------------------------------------------------------
    # 6.  Safety net – guarantee a non‑empty return
    # ------------------------------------------------------------------
    if not best_vec:
        best_vec = [1.0] * n
        best_score = evaluate_sequence(best_vec)

    print(f"\nSearch finished. Best reported score = {best_score:.6f}")
    return best_vec
