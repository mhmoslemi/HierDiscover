"""
Hybrid arcsine‑parameter search + Adam‑mirrored gradient descent.

The algorithm works in stages:
1.  Build a hierarchy of lengths (1024 → 2048 → 4096 → 8192 → 16384 …).
2.  For each length generate a modest library of start vectors:
    – up‑sampled best from the previous stage,
    – the pure discrete arcsine law,
    – a few parametric “generalised arcsine” shapes  p_i ∝ ((i+γ)*(n‑1‑i+γ))^{-α},
    – random Dirichlet draws,
    – tiny perturbations of the above.
    The surrogate objective 2·n·max(p∗p) (p is normalised to sum 1) is used to keep only
    the most promising candidates.
3.  Refine every retained candidate with an Adam‑style mirror‑descent on the
    simplex.  The true objective (the maximal self‑convolution) is smoothed by a
    soft‑max with a schedule that gradually focusses on the true maximum.
    The update is performed on the log‑probabilities (mirror descent) and the
    resulting vector is projected back onto the probability simplex after each step.
4.  After each length the best vector is evaluated with the official
    ``evaluate_sequence`` routine (which is inexpensive for the lengths we use).
5.  A final polishing phase runs a longer Adam optimisation on the overall best
    vector.

All operations are O(n log n) thanks to FFT‑based convolutions, and the routine
stops well before the 1000 s budget expires, always returning a feasible list of
non‑negative heights.

def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """
    Full linear convolution (a * b) using real FFT.
    Returns an array of length len(a)+len(b)-1.
    """
    m = len(a) + len(b) - 1
    n_fft = next_pow2(m)
    A = np.fft.rfft(a, n=n_fft)
    B = np.fft.rfft(b, n=n_fft)
    C = A * B
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


def compute_score(p: np.ndarray) -> float:
    """
    Fast surrogate used during the search.
    p is assumed to be normalised (sum(p) == 1).
    Returns 2·n·max(p∗p)  (exactly the quantity the official evaluator
    returns after cancelling the sum² term).
    """
    n = len(p)
    conv = convolution_fft(p, p)
    return 2.0 * n * conv.max()

# Families of start shapes
# ----------------------------------------------------------------------


def arcsine_initial(n: int) -> np.ndarray:
    """Discrete arcsine density a_i ∝ 1/√((i+½)(n‑½‑i))."""
    i = np.arange(n, dtype=float)
    eps = 0.25  # tiny regulariser to avoid division‑by‑zero
    vals = 1.0 / np.sqrt((i + 0.5) * (n - 0.5 - i) + eps)
    return vals / vals.sum()


def parametric_shape(n: int, alpha: float = 0.5, gamma: float = 0.5) -> np.ndarray:
    """
    Generalised arcsine family:
        p_i ∝ ((i+γ)*(n‑1‑i+γ))^{-α}
    The classic arcsine corresponds to alpha=0.5, gamma=0.5.
    """
    i = np.arange(n, dtype=float)
    # avoid zero denominator for i=0 or i=n‑1 when gamma==0
    denom = (i + gamma) * (n - 1 - i + gamma)
    # add a tiny epsilon to keep everything finite
    vals = np.where(denom > 0, denom ** (-alpha), 0.0)
    vals += 1e-30
    return vals / vals.sum()


def random_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    """Draw a Dirichlet‑like vector with concentration ``conc``."""
    if conc <= 0:
        # heavy‑tailed, near‑sparse draws
        samp = rng.exponential(1.0, size=n) ** (1.0 / max(0.1, -conc))
    else:
        samp = rng.gamma(conc, 1.0, size=n)
    samp += 1e-12
    return samp / samp.sum()


def perturb(a: np.ndarray, rng: np.random.Generator, scale: float = 0.02) -> np.ndarray:
    """Add Gaussian jitter and re‑normalise."""
    b = a + rng.normal(0.0, scale, size=a.shape)
    b = np.maximum(b, 0.0)
    s = b.sum()
    if s == 0.0:
        return np.full_like(a, 1.0 / len(a))
    return b / s


def upsample(prev: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation from len(prev) → new_n, followed by re‑normalisation."""
    x_old = np.linspace(0.0, 1.0, len(prev))
    x_new = np.linspace(0.0, 1.0, new_n)
    interp = np.interp(x_new, x_old, prev)
    total = interp.sum()
    if total == 0.0:
        return np.full(new_n, 1.0 / new_n)
    return interp / total


# ----------------------------------------------------------------------
# Gradient utilities (soft‑max smoothed max‑convolution)
# ----------------------------------------------------------------------


def softmax_weights(conv: np.ndarray, alpha: float) -> np.ndarray:
    """Compute a smooth soft‑max weight vector for a convolution array."""
    # shift to avoid overflow
    shift = conv.max()
    exps = np.exp(alpha * (conv - shift))
    total = exps.sum()
    if total == 0.0:
        # fall back to uniform weights
        return np.full_like(conv, 1.0 / conv.size)
    return exps / total


def weighted_gradient(p: np.ndarray, w: np.ndarray) -> np.ndarray:
    """
    Gradient of  Σ_k w_k (p∗p)_k  w.r.t. p.
    It equals  2·(p ⋆ w)  where ⋆ denotes cross‑correlation.
    Implemented via FFT – O(n log n).
    """
    n = len(p)
    # cross‑correlation via convolution with reversed weights
    w_rev = w[::-1]                         # length 2n‑1
    conv = convolution_fft(p, w_rev)        # length 3n‑2
    idx_start = 2 * n - 2                   # position of w_rev[0] contribution to p[0]
    grad = 2.0 * conv[idx_start - np.arange(n)]
    return grad


# ----------------------------------------------------------------------
# Adam optimiser on the probability simplex (mirror‑descent version)
# ----------------------------------------------------------------------


def optimise_adam(
    init_p: np.ndarray,
    deadline: float,
    max_iter: int,
    *,
    eta: float = 0.12,
    beta1: float = 0.9,
    beta2: float = 0.999,
    eps: float = 1e-8,
    alpha_start: float = 2.0,
    alpha_end: float = 30.0,
    noise_every: int = 0,
    noise_scale: float = 0.0,
) -> Tuple[np.ndarray, float]:
    """
    Adam‑style mirror descent on the simplex.
    The optimiser works on log‑probabilities (mirror map = negative entropy).
    Returns the best vector seen and its surrogate score.
    """
    n = len(init_p)
    # initialise log‑probabilities (add tiny epsilon to avoid log(0))
    logp = np.log(np.maximum(init_p, 1e-30))
    m = np.zeros_like(logp)
    v = np.zeros_like(logp)

    # keep the best
    p = np.exp(logp)
    p /= p.sum()
    best_p = p.copy()
    best_score = compute_score(p)

    for it in range(1, max_iter + 1):
        if time.time() >= deadline:
            break

        # ----- evaluate current point -----
        p = np.exp(logp)
        p_sum = p.sum()
        if p_sum == 0.0:
            p = np.full_like(p, 1.0 / n)
        else:
            p /= p_sum

        conv = convolution_fft(p, p)
        cur_score = 2.0 * n * conv.max()          # exact surrogate (sum(p)=1)

        if cur_score < best_score:
            best_score = cur_score
            best_p = p.copy()

        # ----- compute smooth gradient -----
        # linear schedule for the soft‑max temperature
        alpha = alpha_start + (alpha_end - alpha_start) * (it - 1) / max(1, max_iter - 1)
        w = softmax_weights(conv, alpha)          # length 2n‑1, sums to 1
        grad = weighted_gradient(p, w)            # gradient of weighted sum

        # ----- Adam update on log‑probabilities -----
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)

        # bias‑corrected moments
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        step = eta / (np.sqrt(v_hat) + eps)
        logp -= step * m_hat

        # ----- occasional noise injection (helps escape plateaus) -----
        if noise_every and (it % noise_every == 0):
            logp += rng.normal(0.0, noise_scale, size=n)

        # ----- progress report -----
        if it % 2000 == 0:
            print(
                f"        Adam iter={it:,d} α={alpha:.2f} cur={cur_score:.6f} best={best_score:.6f}"
            )

    return best_p, best_score


# ----------------------------------------------------------------------
# Main entry point required by the platform
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: int = 1000) -> List[float]:
    """
    Search for a non‑negative step‑function with a low value of
    ``evaluate_sequence``.  The routine respects the supplied time budget,
    builds candidates hierarchically, refines them with Adam‑mirrored descent,
    and finally returns the best sequence found.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    # keep a small safety margin for the final return
    deadline = start_time + budget_s - 2.0

    best_seq: List[float] = []
    best_val: float = float("inf")
    best_p: np.ndarray = None

    # ------------------------------------------------------------------
    # 0.  Optional fall‑back from a previously discovered construction.
    # ------------------------------------------------------------------
    try:
        fallback = [float(x) for x in height_sequence_1]  # type: ignore
        val = evaluate_sequence(fallback)
        print(f"Fallback (height_sequence_1) score = {val:.6f}")
        if val < best_val:
            best_val = val
            best_seq = fallback
            best_p = np.array(fallback, dtype=float)
            best_p /= best_p.sum()
    except Exception:
        pass

    # ------------------------------------------------------------------
    # 1.  Length hierarchy – start modest, enlarge progressively.
    # ------------------------------------------------------------------
    lengths = [1024, 2048, 4096, 8192, 16384]

    for n in lengths:
        if time.time() >= deadline - 5.0:
            break
        print(f"\n=== Exploring length n = {n} ===")

        # --------------------------------------------------------------
        # 1a. Build a small library of start vectors.
        # --------------------------------------------------------------
        cand_list: List[np.ndarray] = []

        if best_p is not None:
            cand_list.append(upsample(best_p, n))

        cand_list.append(arcsine_initial(n))

        # a small grid of (alpha,gamma) parameters around the arcsine law
        alphas = np.linspace(0.40, 0.60, 5)      # 0.40,0.45,0.50,0.55,0.60
        gammas = [0.0, 0.25, 0.5, 1.0, 2.0]
        for a in alphas:
            for g in gammas:
                cand_list.append(parametric_shape(n, alpha=a, gamma=g))

        # a handful of random Dirichlet draws
        for conc in (0.3, 0.7, 1.5):
            cand_list.append(random_dirichlet(n, conc, rng))

        # perturbations of the arcsine shape – useful for jumping off plateaus
        for _ in range(2):
            cand_list.append(perturb(arcsine_initial(n), rng, scale=0.015))

        # --------------------------------------------------------------
        # 1b. Quick screening with the cheap surrogate score.
        # --------------------------------------------------------------
        scored = [(compute_score(c), c) for c in cand_list]
        scored.sort(key=lambda x: x[0])
        top_candidates = [c for _, c in scored[:3]]   # keep only a few

        # --------------------------------------------------------------
        # 1c. Refine each retained candidate with Adam‑mirrored descent.
        # --------------------------------------------------------------
        for idx, cand in enumerate(top_candidates):
            if time.time() >= deadline - 5.0:
                break
            print(f"  candidate {idx + 1}/{len(top_candidates)} (n={n})")

            # estimate remaining time and allocate iterations
            remaining = max(0.0, deadline - time.time())
            # empirical budget: ≈ 250 iterations per second for n≈10 k
            max_iter = int(min(250_000, max(5_000, remaining * 250)))

            refined, _ = optimise_adam(
                init_p=cand,
                deadline=deadline,
                max_iter=max_iter,
                eta=0.12,
                alpha_start=2.0,
                alpha_end=30.0,
                noise_every=800,
                noise_scale=5e-6,
            )

            # official evaluation (expensive – only on the refined vectors)
            val = evaluate_sequence(refined.tolist())
            print(f"    evaluated score = {val:.6f}")

            if val < best_val:
                best_val = val
                best_seq = refined.tolist()
                best_p = refined.copy()
                print(f"    *** New global best: {best_val:.6f} (n={n}) ***")

    # ------------------------------------------------------------------
    # 2.  Final polishing of the overall best (if time permits).
    # ------------------------------------------------------------------
    if best_seq and time.time() < deadline - 5.0:
        a = np.array(best_seq, dtype=float)
        remaining = max(0.0, deadline - time.time())
        max_iter = int(min(300_000, remaining * 250))

        print("\nFinal polishing of the global best")
        a_ref, _ = optimise_adam(
            init_p=a,
            deadline=deadline,
            max_iter=max_iter,
            eta=0.08,
            alpha_start=5.0,
            alpha_end=60.0,
            noise_every=0,
            noise_scale=0.0,
        )
        final_val = evaluate_sequence(a_ref.tolist())
        print(f"Final polished score = {final_val:.6f}")
        if final_val < best_val:
            best_val = final_val
            best_seq = a_ref.tolist()
            print("*** Updated global best after final polish ***")

    # ------------------------------------------------------------------
    # 3.  Guard – always return a feasible sequence.
    # ------------------------------------------------------------------
    if not best_seq:
        best_seq = [1.0] * 256
        best_val = evaluate_sequence(best_seq)

    print(f"\nSearch finished. Best reported score = {best_val:.6f}")
    return best_seq
