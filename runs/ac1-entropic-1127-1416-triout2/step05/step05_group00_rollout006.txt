def gradient_descent_refine(a_init, time_deadline, max_iter=2000, step0=0.3, eps_factor=0.1):

    n = len(a_init)
    a = a_init.copy()
    best_a = a.copy()
    best_sc = compute_score(a)

    # initial epsilon set to 0.2*max_conv? Actually compute conv each iteration.
    for t in range(1, max_iter+1):
        if time.time() >= time_deadline:
            break

        conv = convolution_fft(a, a)
        maxc = np.max(conv)
        # epsilon schedule: start at epsilon0 = maxc * 0.2, gradually decays
        eps = maxc * max(0.01, 0.2 * (0.99 ** (t-1)))  # ensures positive

assistantfinal**Solution Overview**

The program searches for a non‑negative step‑function (a probability vector) that minimizes  

\[
\text{score}= \frac{2\,n\;\max (a*a)}{(\sum a)^2},
\]

where `a*a` is the full convolution of `a` with itself.
Because the score is invariant under scaling of the vector we can (and do) keep
\(\sum a = 1\).  In that case  

\[
\text{score}=2\,n\; \max\_{k}(a*a)_k .
\]

The algorithm consists of four independent phases that run inside the allowed
time budget (≈ 1000 s).

| Phase | Idea |
|------|------|
| **1 – Random sparse start** | Generate many sparse probability vectors of various
lengths and keep the best one. |
| **2 – Smooth soft‑max gradient descent** | Replace the non‑smooth “max’’ by a
soft‑max (log‑sum‑exp).  The gradient of the smoothed objective can be written
as a single convolution, so each iteration costs only one FFT.  After every
step the vector is projected back onto the simplex. |
| **3 – Exact sub‑gradient refinement** | When the soft‑max has become sharp
(`ε` is tiny) we switch to the true sub‑gradient (the gradient at the
maximal shift) for a short finishing optimisation. |
| **4 – Hill‑climb polishing** | Stochastic swaps / mass‑transfer moves are
accepted only when they improve the exact score.  A final symmetric
averaging step is tried as a cheap post‑processor. |

All helpers are defined at module level (no closures) and only `numpy`,
`time` and the supplied `evaluate_sequence` are used.  The code respects the
1000 s limit – it constantly checks the deadline and stops early if necessary.

```python
"""
Search for a low‑score step function.

The algorithm uses
  1. random sparse starts,
  2. a smooth soft‑max surrogate of the max‑convolution (gradient descent
     with a single FFT per iteration),
  3. exact sub‑gradient descent for fine tuning,
  4. stochastic hill‑climbing (swap / mass‑transfer) plus a final
     symmetry‑averaging polish.

All vectors are kept on the probability simplex (non‑negative,
sum = 1), which makes the evaluation fast (score = 2·n·max(conv)).
"""
import time
import numpy as np

# ----------------------------------------------------------------------
# Fast utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Return the smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """
    Full linear convolution a ⊛ b using real FFT.
    Returns an array of length len(a)+len(b)-1.
    """
    m = len(a) + len(b) - 1
    n_fft = next_pow2(m)
    A = np.fft.rfft(a, n=n_fft)
    B = np.fft.rfft(b, n=n_fft)
    C = A * B
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


def compute_score(a: np.ndarray) -> float:
    """
    Fast replica of evaluate_sequence for vectors with sum(a)=1.
    Returns 2·n·max(a⊛a).
    """
    n = a.shape[0]
    conv = convolution_fft(a, a)
    return 2.0 * n * float(np.max(conv))


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the probability simplex.
    Guarantees non‑negativity and sum(v)=1.
    """
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / v.shape[0])

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, v.shape[0] + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.where(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


# ----------------------------------------------------------------------
# 1. Random sparse initialisation
# ----------------------------------------------------------------------


def random_sparse_vector(n: int) -> np.ndarray:
    """
    Generate a random probability vector of length n.
    A random subset of the coordinates (density 0.1–0.9) gets equal mass.
    """
    density = np.random.uniform(0.10, 0.90)
    k = max(2, int(round(density * n)))
    idx = np.random.choice(n, size=k, replace=False)
    a = np.zeros(n, dtype=float)
    a[idx] = 1.0 / k
    return a


# ----------------------------------------------------------------------
# 2. Smooth soft‑max gradient descent
# ----------------------------------------------------------------------


def smooth_gradient_descent(
    a0: np.ndarray,
    time_deadline: float,
    max_iter: int = 5000,
    step0: float = 0.5,
    eps_factor: float = 0.2,
) -> tuple[np.ndarray, float]:
    """
    Minimise a smooth approximation of the max‑convolution:

        f(a) = ε·log∑_k exp(conv_k / ε)

    where conv = a⊛a.
    Gradient is obtained via a single convolution:
        grad = 2 * (w_rev ⊛ a)    with   w = softmax(conv/ε)

    After each step the vector is projected onto the simplex.
    Returns the best vector (lowest exact score) seen.
    """
    a = a0.astype(float).copy()
    best_a = a.copy()
    best_sc = compute_score(a)
    n = len(a)

    for t in range(1, max_iter + 1):
        if time.time() >= time_deadline:
            break

        conv = convolution_fft(a, a)
        maxc = np.max(conv)

        # ε shrinks with the iteration count, but never smaller than 1 % of maxc
        eps = maxc * max(eps_factor * (0.99 ** (t - 1)), 0.01)

        # soft‑max weights (stable formulation)
        exp_vals = np.exp((conv - maxc) / eps)
        w = exp_vals / np.sum(exp_vals)          # ∑ w = 1

        # gradient: g_i = 2 * Σ_j a_j * w_{i+j}
        w_rev = w[::-1]                          # reverse for cross‑correlation
        conv_rev = convolution_fft(a, w_rev)    # length 3n‑2
        idx_start = 2 * n - 2
        grad = 2.0 * conv_rev[idx_start - np.arange(n)]

        # step size (standard SGD schedule)
        step = step0 / np.sqrt(t)
        a_new = a - step * grad
        a_new = project_to_simplex(a_new)

        cur_sc = compute_score(a_new)
        if cur_sc < best_sc:
            best_sc = cur_sc
            best_a = a_new.copy()

        a = a_new

        # occasional tiny random jitter to escape flat regions
        if t % 400 == 0:
            jitter = np.random.randn(n) * (step0 * 0.004 / np.sqrt(t))
            a += jitter
            a = project_to_simplex(a)

    return best_a, best_sc


# ----------------------------------------------------------------------
# 3. Exact sub‑gradient descent (max‑convolution only)
# ----------------------------------------------------------------------


def subgrad_optimize(
    init_a: np.ndarray,
    time_deadline: float,
    max_iter: int = 200_000,
    step0: float = 0.07,
) -> tuple[np.ndarray, float]:
    """
    Classic sub‑gradient descent on the exact objective
        2·n·max(a⊛a) .
    The sub‑gradient is taken at the shift where the convolution attains its
    maximum (see the comment in the original baseline code).
    """
    a = init_a.astype(float).copy()
    best_a = a.copy()
    best_sc = compute_score(a)
    n = len(a)

    for t in range(1, max_iter + 1):
        if time.time() >= time_deadline:
            break

        conv = convolution_fft(a, a)
        k = int(np.argmax(conv))

        # build sub‑gradient g_i = 2·a_{k-i}  (zero outside the support)
        idx = np.arange(n)
        j = k - idx
        mask = (j >= 0) & (j < n)
        g = np.zeros_like(a)
        g[mask] = 2.0 * a[j[mask]]

        step = step0 / np.sqrt(t)
        a_new = a - step * g
        a_new = project_to_simplex(a_new)

        cur_sc = compute_score(a_new)
        if cur_sc < best_sc:
            best_sc = cur_sc
            best_a = a_new.copy()

        a = a_new

        # small random shake every thousand iterations
        if t % 1000 == 0:
            a += np.random.randn(n) * (step0 * 0.005 / np.sqrt(t))
            a = project_to_simplex(a)

    return best_a, best_sc


# ----------------------------------------------------------------------
# 4. Hill‑climb polishing
# ----------------------------------------------------------------------


def hill_climb_permutation(
    seq,
    time_deadline,
    max_steps: int = 300_000,
) -> tuple[list, float]:
    """
    Stochastic hill‑climber that performs swaps or transfers a fraction of
    mass between two positions.  Only moves that improve the *exact* score are
    accepted.
    """
    a = np.array(seq, dtype=float)
    n = a.shape[0]

    best_a = a.copy()
    best_sc = compute_score(best_a)

    step_scale = 0.10   # initial fraction of mass moved in a transfer

    while max_steps > 0 and time.time() < time_deadline:
        move_type = np.random.rand()
        cand = best_a.copy()

        if move_type < 0.5:
            # simple swap
            i, j = np.random.randint(n, size=2)
            cand[i], cand[j] = cand[j], cand[i]
        else:
            # transfer a fraction from i to j
            i, j = np.random.randint(n, size=2)
            if i == j:
                continue
            delta = step_scale * best_a[i]
            if delta <= 0:
                continue
            cand[i] -= delta
            cand[j] += delta
            if cand[i] < 0.0:
                cand[i] = 0.0

        cand = project_to_simplex(cand)
        cur_sc = compute_score(cand)

        if cur_sc < best_sc:
            best_sc = cur_sc
            best_a = cand
            step_scale = min(step_scale * 1.05, 0.5)   # be bolder
        else:
            step_scale *= 0.97                         # be more cautious

        max_steps -= 1

    return best_a.tolist(), best_sc


def final_symmetry_average(a: np.ndarray) -> np.ndarray:
    """
    Produce a symmetric version (a + reverse(a))/2 and re‑project.
    Symmetric vectors often have flatter autocorrelations.
    """
    a_sym = (a + a[::-1]) * 0.5
    return project_to_simplex(a_sym)

    # ------------------------------------------------------------------
```
