"""
Propose a low‑score step function.

Algorithm (≈ 1000 s):
1️⃣  Initialise a probability vector on the simplex (Dirichlet with low
    concentration → sparsity).
2️⃣  Fast exponentiated sub‑gradient descent on the exact non‑smooth
    objective  f(a)=2·n·max(a*a).  This quickly lowers the maximal
    convolution.
3️⃣  Refinement with Adam‑optimised *sub‑gradient* updates (still
    respecting the simplex).  Using an adaptive optimiser helps escape
    shallow plateaus that the pure exponential update can get stuck in.
4️⃣  A short stochastic hill‑climber (swap / mass‑transfer moves) polish
    the final candidate.

All phases respect the 1000 s budget, keep the sequence non‑negative and
normalised (∑a = 1), and return the best sequence found.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# FFT‑based linear convolution utilities (real FFT)
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Linear convolution a*b using a real FFT (fast)."""
    m = len(a) + len(b) - 1
    n_fft = next_pow2(m)
    A = np.fft.rfft(a, n=n_fft)
    B = np.fft.rfft(b, n=n_fft)
    C = A * B
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


# ----------------------------------------------------------------------
# Objective helpers
# ----------------------------------------------------------------------


def compute_score(a: np.ndarray) -> float:
    """
    Fast replica of `evaluate_sequence` when ∑a = 1.
    Returns 2·n·max(a*a) (exactly the value returned by evaluate_sequence
    for a normalised vector).
    """
    n = a.shape[0]
    conv = convolution_fft(a, a)
    return 2.0 * n * float(np.max(conv))


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the probability simplex:
    non‑negative entries that sum to 1.
    """
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / v.shape[0])

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, v.shape[0] + 1)
    cond = u - cssv / ind > 0
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


# ----------------------------------------------------------------------
# Sub‑gradient (exact max) utilities
# ----------------------------------------------------------------------


def subgradient_maxconv(a: np.ndarray) -> np.ndarray:
    """
    Sub‑gradient of  max(a*a) at the current maximal convolution index.
    Returns a vector g with g[i] = 2·a[k‑i] for all i where the index is
    inside the array (otherwise 0), where k = argmax conv.
    """
    n = a.shape[0]
    conv = convolution_fft(a, a)
    k = int(np.argmax(conv))

    idx = np.arange(n)
    j = k - idx
    mask = (j >= 0) & (j < n)
    g = np.zeros_like(a)
    g[mask] = 2.0 * a[j[mask]]
    return g, conv.max()


def exp_subgrad_optimize(init_a: np.ndarray,
                         time_deadline: float,
                         max_iter: int = 3_000_000,
                         eta0: float = 0.7) -> tuple[np.ndarray, float]:
    """Exponentiated sub‑gradient descent (multiplicative updates)."""
    a = init_a.copy()
    best_a = a.copy()
    best_score = compute_score(a)

    n = len(a)
    for t in range(1, max_iter + 1):
        if time.time() >= time_deadline:
            break

        g, cur_max = subgradient_maxconv(a)

        eta = eta0 / np.sqrt(t)          # diminishing step size
        a = a * np.exp(-eta * g)         # multiplicative update

        # projection onto the simplex
        a = project_to_simplex(a)

        cur_score = compute_score(a)
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()

        # thin jitter every few thousand steps (helps escape plateaus)
        if t % 5000 == 0:
            jitter = np.random.randn(n) * (eta0 * 1e-4 / np.sqrt(t))
            a = project_to_simplex(a + jitter)

    return best_a, best_score


# ----------------------------------------------------------------------
# Adam optimiser on the exact sub‑gradient (additive updates)
# ----------------------------------------------------------------------


def adam_subgrad_optimize(init_a: np.ndarray,
                          time_deadline: float,
                          max_iter: int = 200_000,
                          lr: float = 0.08,
                          beta1: float = 0.9,
                          beta2: float = 0.999,
                          eps: float = 1e-8) -> tuple[np.ndarray, float]:
    """Adam optimiser applied to the (non‑smooth) sub‑gradient of max‑conv."""
    a = init_a.copy()
    best_a = a.copy()
    best_score = compute_score(a)

    n = len(a)
    m = np.zeros_like(a)   # first moment
    v = np.zeros_like(a)   # second moment

    for t in range(1, max_iter + 1):
        if time.time() >= time_deadline:
            break

        g, cur_max = subgradient_maxconv(a)

        # Adam update (additive)
        m = beta1 * m + (1.0 - beta1) * g
        v = beta2 * v + (1.0 - beta2) * (g ** 2)

        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)

        a = a - lr * m_hat / (np.sqrt(v_hat) + eps)

        # keep on the simplex
        a = project_to_simplex(a)

        cur_score = compute_score(a)
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()

    return best_a, best_score


# ----------------------------------------------------------------------
# Stochastic hill‑climber (swap / mass‑transfer moves)
# ----------------------------------------------------------------------


def hill_climb_permutation(seq, time_deadline, max_steps=200_000):
    """
    Simple hill‑climber that swaps entries or moves a fraction of mass.
    Uses the *exact* objective for acceptance.
    """
    a = np.array(seq, dtype=float)
    n = a.shape[0]

    best_a = a.copy()
    best_score = compute_score(best_a)

    step_scale = 0.1   # fraction of mass moved in a transfer

    while max_steps > 0 and time.time() < time_deadline:
        cand = best_a.copy()

        if np.random.rand() < 0.5:
            # swap two positions
            i, j = np.random.randint(n, size=2)
            cand[i], cand[j] = cand[j], cand[i]
        else:
            # transfer a fraction from i to j
            i, j = np.random.randint(n, size=2)
            if i != j:
                delta = step_scale * best_a[i]
                cand[i] -= delta
                cand[j] += delta
                cand[i] = max(cand[i], 0.0)

        # re‑normalise onto the simplex
        cand = project_to_simplex(cand)
        cur_score = compute_score(cand)

        if cur_score < best_score:
            best_score = cur_score
            best_a = cand
            step_scale = min(step_scale * 1.05, 0.5)   # success → larger steps
        else:
            step_scale *= 0.97                         # failure → smaller steps

        max_steps -= 1

    return best_a.tolist(), best_score

# Main entry point required by the evaluation harness
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative sequence that minimises `evaluate_sequence`.
    The search consists of three phases:
        1. A sparse Dirichlet start (α<1 → many very small entries).
        2. Exponentiated sub‑gradient descent → rapid reduction.
        3. Adam refinement on the exact sub‑gradient → fine‑tuning.
        4. A short hill‑climber polish.
    All work respects the 1000 s budget.
    """
    np.random.seed(seed)
    start_time = time.time()
    # leave a small safety margin for the harness
    deadline = start_time + budget_s - 1.5

    best_seq = None
    best_val = float("inf")

    # ------------------------------------------------------------------
    # 0️⃣  Fallback (if a previous construction is available)
    # ------------------------------------------------------------------
    try:
        fallback = [float(x) for x in height_sequence_1]  # type: ignore
        fallback_val = evaluate_sequence(fallback)
        print(f"Fallback score = {fallback_val:.6f}")
        if fallback_val < best_val:
            best_val, best_seq = fallback_val, fallback
    except Exception:
        pass

    # ------------------------------------------------------------------
    # 1️⃣  Initialise a sparse Dirichlet vector (n = 2048)
    # ------------------------------------------------------------------
    n = 2048
    concentration = 0.35                     # <1 ⇒ sparse start
    a0 = np.random.dirichlet(np.full(n, concentration))
    cur_val = compute_score(a0)
    if cur_val < best_val:
        best_val, best_seq = cur_val, a0.tolist()
        print(f"Random sparse start score = {best_val:.6f}")

    # ------------------------------------------------------------------
    # 2️⃣  Exponentiated sub‑gradient descent (fast phase)
    # ------------------------------------------------------------------
    if time.time() < deadline - 30.0:
        print("\n--- Phase 2: Exponentiated sub‑gradient optimisation ---")
        a_init = np.array(best_seq, dtype=float) if best_seq is not None else a0
        refined_a, refined_score = exp_subgrad_optimize(
            a_init,
            time_deadline=deadline - 20.0,
            max_iter=4_000_000,
            eta0=0.7,
        )
        if refined_score < best_val:
            best_val, best_seq = refined_score, refined_a.tolist()
            print(f"Exp‑subgrad improved best to {best_val:.6f}")

    # ------------------------------------------------------------------
    # 3️⃣  Adam optimisation on the exact sub‑gradient (fine phase)
    # ------------------------------------------------------------------
    if time.time() < deadline - 10.0:
        print("\n--- Phase 3: Adam sub‑gradient refinement ---")
        a_init = np.array(best_seq, dtype=float)
        refined_a, refined_score = adam_subgrad_optimize(
            a_init,
            time_deadline=deadline - 5.0,
            max_iter=250_000,
            lr=0.08,
        )
        if refined_score < best_val:
            best_val, best_seq = refined_score, refined_a.tolist()
            print(f"Adam refinement improved best to {best_val:.6f}")

    # ------------------------------------------------------------------
    # 4️⃣  Final hill‑climb polishing
    # ------------------------------------------------------------------
    if best_seq is not None and time.time() < deadline - 2.0:
        print("\n--- Phase 4: Hill‑climb polish ---")
        polished_seq, polished_val = hill_climb_permutation(
            best_seq,
            time_deadline=deadline,
            max_steps=300_000,
        )
        if polished_val < best_val:
            best_val, best_seq = polished_val, polished_seq
            print(f"Hill‑climb improved best to {best_val:.6f}")

    # ------------------------------------------------------------------
    # Safeguard – if nothing succeeded, fall back to a tiny uniform vector
    # ------------------------------------------------------------------
    if best_seq is None:
        n = 256
        best_seq = [1.0 / n] * n
        best_val = evaluate_sequence(best_seq)

    final_score = evaluate_sequence(best_seq)
    print(f"\nSearch finished. Final reported score = {final_score:.6f}")
    print(f"Final sequence length = {len(best_seq)}")
    return best_seq
