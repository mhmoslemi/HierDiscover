"""
propose_candidate
=================

Search for a non‑negative step sequence that minimizes

    2·n·max(conv(a,a)) / (Σ a)²

The algorithm consists of four stages:

1. **Focused random sampling**  
   The search space is restricted to the symmetric beta family  

        a_i ∝ (i+ε)^{α-1}·(n−1−i+ε)^{α-1},

   which contains the classic arcsine shape (α≈0.5).  
   α is sampled from a tight log‑uniform interval around 0.5 and
   ε from a log‑uniform interval that favours very small values.
   A few mixtures of two beta components are also tried.  The best
   candidate according to the fast surrogate `2·n·max(conv(a,a))`
   (with Σa = 1) is kept.

2. **High‑temperature Adam optimisation**  
   The best sampled vector is refined by Adam on a smooth
   log‑sum‑exp surrogate of the max‑convolution.  Temperature λ is
   annealed up to 500, which makes the surrogate essentially the true
   max.  The optimiser runs as long as time permits.

3. **Exact sub‑gradient descent** – a projected sub‑gradient step on
   the true max‑convolution with a back‑tracking line‑search guarantees
   monotone descent.

4. **Stochastic local walk** – a cheap Gaussian perturbation phase
   (with a tiny uphill‑move probability) performs a final polishing.

All stages respect the global 1000 s budget and keep the best vector
according to the *official* `evaluate_sequence`.  If nothing is found,
a safe uniform fallback is returned.
"""

import time
import numpy as np
from typing import List, Tuple

# ----------------------------------------------------------------------
# Basic utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """
    Linear convolution via real FFT.
    Returns an array of length len(a)+len(b)-1.
    """
    m = a.size + b.size - 1
    n_fft = next_pow2(m)
    A = np.fft.rfft(a, n=n_fft)
    B = np.fft.rfft(b, n=n_fft)
    C = A * B
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


def max_convolution(a: np.ndarray) -> float:
    """Maximum entry of the self‑convolution of a."""
    return float(np.max(convolution_fft(a, a)))


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection onto the probability simplex {x ≥ 0, Σx = 1}.
    Implementation of Duchi et al. (2008).
    """
    n = v.shape[0]
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / n)

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho = np.where(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def compute_fast_score(a: np.ndarray) -> float:
    """
    Fast surrogate (valid when Σa = 1):
        2·n·max(conv(a,a))
    """
    n = a.shape[0]
    return 2.0 * n * max_convolution(a)


# ----------------------------------------------------------------------
# Shape generators (beta families and mixtures)
# ----------------------------------------------------------------------


def generate_beta_shape(
    n: int, alpha: float, beta: float, eps: float
) -> np.ndarray:
    """
    Symmetric beta‑family density on {0,…,n‑1}:
        a_i ∝ (i+ε)^{α‑1}·(n‑1‑i+ε)^{β‑1}.
    Returned vector is normalised to Σa = 1.
    """
    i = np.arange(n, dtype=float)
    left = (i + eps) ** (alpha - 1.0)
    right = (n - 1 - i + eps) ** (beta - 1.0)
    base = left * right
    total = base.sum()
    if total <= 0.0 or not np.isfinite(total):
        return np.full(n, 1.0 / n)
    return base / total


def generate_beta_mixture(
    n: int,
    alpha1: float,
    beta1: float,
    eps1: float,
    alpha2: float,
    beta2: float,
    eps2: float,
    w: float,
) -> np.ndarray:
    """
    Convex combination of two beta‑family densities.
    The result is re‑normalised to Σ = 1.
    """
    a1 = generate_beta_shape(n, alpha1, beta1, eps1)
    a2 = generate_beta_shape(n, alpha2, beta2, eps2)
    mix = w * a1 + (1.0 - w) * a2
    mix_sum = mix.sum()
    if mix_sum == 0.0 or not np.isfinite(mix_sum):
        return np.full(n, 1.0 / n)
    return mix / mix_sum


def arcsine_shape_eps(n: int, eps: float) -> np.ndarray:
    """
    Arcsine‑like shape with a tunable smoothing constant ε.
    a_i ∝ 1 / sqrt((i+0.5+ε)·(n-0.5-i+ε)).
    """
    i = np.arange(n, dtype=float)
    vals = 1.0 / np.sqrt((i + 0.5 + eps) * (n - 0.5 - i + eps))
    vals = np.maximum(vals, 0.0)
    vals /= vals.sum()
    return vals


# ----------------------------------------------------------------------
# Smooth surrogate (log‑sum‑exp) and gradient (Adam)
# ----------------------------------------------------------------------


def smooth_gradient(a: np.ndarray, lam: float) -> Tuple[np.ndarray, float]:
    """
    Gradient of the log‑sum‑exp surrogate of max(conv(a,a)).
    Returns (gradient, surrogate_value).
    """
    n = a.shape[0]
    conv = convolution_fft(a, a)  # length 2n‑1
    max_c = np.max(conv)

    # stable log‑sum‑exp
    exp_term = np.exp(lam * (conv - max_c))
    sum_exp = np.sum(exp_term)
    w = exp_term / sum_exp                # soft‑max on convolution entries
    B = max_c + np.log(sum_exp) / lam     # smooth approximation of the max

    # weighted sum of sub‑gradients:
    # ∂/∂a  Σ_k w_k (a*a)_k = 2 * (a ⋆ w_rev)
    w_rev = w[::-1]
    conv_rev = convolution_fft(a, w_rev)        # length 3n‑2
    L = 2 * n - 1
    weighted_grad = 2.0 * conv_rev[L - 1 - np.arange(n)]

    s = np.sum(a)
    term1 = (2.0 * n / (s ** 2)) * weighted_grad
    term2 = -(4.0 * n * B / (s ** 3)) * np.ones_like(a)
    grad = term1 + term2
    return grad, B


def adam_optimize(
    initial_a: np.ndarray,
    time_deadline: float,
    max_iter: int = 300_000,
    lr0: float = 0.13,
    lam_start: float = 1.0,
    lam_end: float = 500.0,
    noise_every: int = 1500,
    noise_scale: float = 4e-5,
) -> Tuple[np.ndarray, float]:
    """
    Adam optimisation on the smooth log‑sum‑exp surrogate.
    Returns the best vector seen (projected onto the simplex) and its
    fast score.
    """
    a = project_to_simplex(initial_a.astype(float).copy())
    best_a = a.copy()
    best_sc = compute_fast_score(a)

    beta1, beta2 = 0.9, 0.999
    eps_adam = 1e-8
    m = np.zeros_like(a)
    v = np.zeros_like(a)

    n = a.shape[0]

    for it in range(1, max_iter + 1):
        if time.time() >= time_deadline:
            break

        # annealed temperature (quadratic schedule, a little more aggressive)
        t = it / max_iter
        lam = lam_start + (lam_end - lam_start) * (t ** 0.5)

        grad, _ = smooth_gradient(a, lam)

        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)

        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        step = lr0 * m_hat / (np.sqrt(v_hat) + eps_adam)
        a_new = a - step

        if noise_every and (it % noise_every == 0):
            a_new += np.random.randn(n) * noise_scale

        a_new = project_to_simplex(a_new)

        cur_sc = compute_fast_score(a_new)
        if cur_sc < best_sc:
            best_sc = cur_sc
            best_a = a_new.copy()

        a = a_new

        # occasional progress report (kept lightweight)
        if it % 20000 == 0:
            print(
                f"[Adam] iter={it:,d} cur={cur_sc:.6f} best={best_sc:.6f}"
            )

    return best_a, best_sc


# ----------------------------------------------------------------------
# Exact sub‑gradient descent with back‑tracking line‑search
# ----------------------------------------------------------------------


def subgradient_optimize(
    initial_a: np.ndarray,
    time_deadline: float,
    step0: float = 0.28,
    max_iter: int = 250_000,
) -> Tuple[np.ndarray, float]:
    """
    Projected sub‑gradient descent on the *true* max‑convolution.
    Returns the best vector (projected) and its fast score.
    """
    a = initial_a.copy()
    best_a = a.copy()
    best_sc = compute_fast_score(a)

    n = a.shape[0]

    for it in range(1, max_iter + 1):
        if time.time() >= time_deadline:
            break

        conv = convolution_fft(a, a)
        max_val = float(np.max(conv))
        idx = int(np.argmax(conv))

        # gradient of the active quadratic term
        i = np.arange(n)
        j = idx - i
        mask = (j >= 0) & (j < n)
        grad = np.zeros_like(a)
        grad[mask] = 2.0 * a[j[mask]]

        # decreasing step size & back‑tracking
        step = step0 / np.sqrt(it)
        for _ in range(10):
            a_new = a - step * grad
            a_new = project_to_simplex(a_new)
            new_val = max_convolution(a_new)
            if new_val < max_val - 1e-14:
                break
            step *= 0.5

        cur_sc = compute_fast_score(a_new)
        if cur_sc < best_sc:
            best_sc = cur_sc
            best_a = a_new.copy()

        a = a_new

        if it % 50000 == 0:
            print(
                f"[Sub‑grad] iter={it:,d} cur={cur_sc:.6f} best={best_sc:.6f}"
            )

    return best_a, best_sc


# ----------------------------------------------------------------------
# Stochastic local walk (Gaussian perturbations)
# ----------------------------------------------------------------------


def random_local_search(
    initial_a: np.ndarray,
    time_deadline: float,
    init_step: float = 0.020,
    max_iter: int = 300_000,
) -> Tuple[np.ndarray, float]:
    """
    Gaussian walk on the simplex with a tiny uphill‑move probability.
    Returns the best vector seen and its fast score.
    """
    a = initial_a.copy()
    best_a = a.copy()
    best_sc = compute_fast_score(a)

    n = a.shape[0]
    rng = np.random.default_rng()

    for it in range(1, max_iter + 1):
        if time.time() >= time_deadline:
            break

        step = init_step / np.sqrt(it)
        noise = rng.normal(scale=step, size=n)
        a_proposed = a + noise
        a_proposed = project_to_simplex(a_proposed)

        cur_sc = compute_fast_score(a_proposed)

        # Accept if improvement; otherwise accept with a tiny probability
        if cur_sc < best_sc or (rng.random() < 1e-4 and cur_sc < 10 * best_sc):
            a = a_proposed
            if cur_sc < best_sc:
                best_sc = cur_sc
                best_a = a_proposed.copy()

        if it % 60000 == 0:
            print(
                f"[Random] iter={it:,d} cur={cur_sc:.6f} best={best_sc:.6f}"
            )

    return best_a, best_sc


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------


def propose_candidate(
    seed: int = 42,
    budget_s: int = 1000,
    **kwargs,
) -> List[float]:
    """
    Main driver.  Returns a list of non‑negative heights (the step function)
    with a low official score.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 2.0  # safety margin

    best_seq: List[float] = []
    best_val: float = float("inf")

    # ------------------------------------------------------------------
    # 0.  Fallback from a previous construction, if available
    # ------------------------------------------------------------------
    try:
        fallback = [float(x) for x in height_sequence_1]  # type: ignore
        val = evaluate_sequence(fallback)
        print(f"[Fallback] official score = {val:.6f}")
        if val < best_val:
            best_val = val
            best_seq = fallback
    except Exception:
        pass

    # ------------------------------------------------------------------
    # 1.  Focused random sampling (length = 2048)
    # ------------------------------------------------------------------
    n = 2048
    print(f"\n=== Sampling {n}‑dimensional candidates ===")
    local_best = None
    local_best_sc = float("inf")

    # deterministic starters
    starters = [
        np.full(n, 1.0 / n),          # uniform
        arcsine_shape_eps(n, eps=0.2)  # decent arcsine variant
    ]
    for st in starters:
        sc = compute_fast_score(st)
        if sc < local_best_sc:
            local_best_sc = sc
            local_best = st.copy()

    # sampling loop – stop a few seconds before deadline
    sample_phase_end = deadline - 30.0
    sample_count = 0
    while time.time() < sample_phase_end and sample_count < 300_000:
        sample_count += 1
        # Choose a generator
        if rng.random() < 0.25:
            # mixture of two betas (potentially asymmetric)
            w = rng.random()
            alpha1 = rng.uniform(0.35, 0.65)   # narrow band around 0.5
            beta1 = rng.uniform(0.35, 0.65)
            eps1 = 10 ** rng.uniform(-8, np.log10(1.0))
            alpha2 = rng.uniform(0.35, 0.65)
            beta2 = rng.uniform(0.35, 0.65)
            eps2 = 10 ** rng.uniform(-8, np.log10(1.0))
            cand = generate_beta_mixture(
                n, alpha1, beta1, eps1, alpha2, beta2, eps2, w
            )
        else:
            # symmetric beta or arcsine‑like with custom ε
            if rng.random() < 0.4:
                # pure beta (α = β)
                alpha = rng.uniform(0.35, 0.65)
                eps = 10 ** rng.uniform(-8, np.log10(1.0))
                cand = generate_beta_shape(n, alpha, alpha, eps)
            else:
                # arcsine‑style with independent ε
                eps = 10 ** rng.uniform(-8, np.log10(1.0))
                cand = arcsine_shape_eps(n, eps)

        sc = compute_fast_score(cand)
        if sc < local_best_sc:
            local_best_sc = sc
            local_best = cand.copy()

    print(f"Sampling finished after {sample_count} samples, best fast score = {local_best_sc:.6f}")

    # ------------------------------------------------------------------
    # 2.  Adam polishing of the locally best candidate
    # ------------------------------------------------------------------
    if local_best is not None and time.time() < deadline - 5.0:
        print("\n--- Adam polishing (high temperature) ---")
        polished_a, _ = adam_optimize(
            initial_a=local_best,
            time_deadline=deadline,
            max_iter=500_000,
            lr0=0.13,
            lam_start=1.0,
            lam_end=500.0,
            noise_every=1500,
            noise_scale=4e-5,
        )
    else:
        polished_a = local_best if local_best is not None else np.full(n, 1.0 / n)

    # ------------------------------------------------------------------
    # 3.  Exact sub‑gradient descent
    # ------------------------------------------------------------------
    if time.time() < deadline - 5.0:
        print("\n--- Sub‑gradient descent ---")
        polished_a, _ = subgradient_optimize(
            polished_a,
            time_deadline=deadline,
            step0=0.28,
            max_iter=250_000,
        )

    # ------------------------------------------------------------------
    # 4.  Final stochastic local walk
    # ------------------------------------------------------------------
    if time.time() < deadline - 3.0:
        print("\n--- Final random local search ---")
        polished_a, _ = random_local_search(
            polished_a,
            time_deadline=deadline,
            init_step=0.020,
            max_iter=300_000,
        )

    # ------------------------------------------------------------------
    # 5.  Official evaluation of the polished vector
    # ------------------------------------------------------------------
    final_vec = polished_a.tolist()
    try:
        final_val = evaluate_sequence(final_vec)
        print(f"Official score of final candidate = {final_val:.6f}")
        if final_val < best_val:
            best_val = final_val
            best_seq = final_vec
    except Exception as e:
        print(f"Evaluation failed: {e}")

    # ------------------------------------------------------------------
    # 6.  Guard – guarantee non‑empty return
    # ------------------------------------------------------------------
    if not best_seq:
        # safe uniform fallback (length 256)
        best_seq = [1.0] * 256
        best_val = evaluate_sequence(best_seq)
        print("Fallback to uniform 256‑vector, score =", best_val)

    print(f"\nSearch finished. Best official score = {best_val:.6f}")
    return best_seq
