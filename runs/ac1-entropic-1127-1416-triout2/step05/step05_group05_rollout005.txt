def adam_optimize(a0, deadline, beta_start=2.0, beta_end=30.0, lr=0.5):
    a = a0.copy()
    z = np.log(a + 1e-30)
    n = len(a)
    best_a = a.copy()
    best_score = compute_score(a)
    m = np.zeros_like(z)
    v = np.zeros_like(z)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    t = 0
    total_iters_est = 200000  # guidance for beta schedule
    while time.time() < deadline:
        t += 1
        # compute conv and max
        conv = convolution_fft(a, a)
        max_val = conv.max()
        # smooth max weight
        beta = beta_start + (beta_end - beta_start) * min(1.0, t / total_iters_est)
        w = np.exp(beta * (conv - max_val))
        w /= w.sum()
        # gradient based on weights
        grad_a = gradient_from_weights(a, w)  # shape n
        # gradient w.r.t z
        grad_z = a * grad_a
        # Adam update
        m = beta1 * m + (1 - beta1) * grad_z
        v = beta2 * v + (1 - beta2) * (grad_z ** 2)
        m_hat = m / (1 - beta1 ** t)
        v_hat = v / (1 - beta2 ** t)
        step = lr * m_hat / (np.sqrt(v_hat) + eps)
        z = z - step
        a = np.exp(z)
        a = normalize(a)
        # evaluate score
        score = compute_score(a)  # exact score = 2*n*max_conv
        if score < best_score:
            best_score = score
            best_a = a.copy()
    return best_a, best_score
```

"""
Efficient search for a non‑negative step function that minimises

    2·n·max(conv(a,a)) / (sum(a)²)

Algorithm
---------
1. Build a modest set of diversified initial candidates (arcsine,
   uniform, random Dirichlet, random power‑law, optional fallback).
   Length n is chosen as 4096 (larger than the previous 2048) – this
   gives the optimiser many degrees of freedom while keeping the FFT
   cost modest.

2. For each candidate run a **log‑Adam** optimiser:
   * Keep the vector on the probability simplex (sum = 1) by normalising
     after every update.
   * Work in the log‑domain:  z = log(a).  The gradient w.r.t. z is
     a · ∂f/∂a.
   * Approximate the non‑smooth max in the objective with a soft‑max
     weight w_i ∝ exp(β·(conv_i−max(conv))) .  The temperature β is
     increased from a small value to a larger one during optimisation,
     slowly turning the surrogate into the true max.
   * Adam updates (β₁=0.9, β₂=0.999) give adaptive step sizes and
     usually converge faster than simple exponentiated‑gradient steps.

   The best iterate seen (according to the exact 2·n·max(conv) score)
   is kept.

3. After the Adam phase, a short **hard sub‑gradient polish** (the
   classic exponentiated‑gradient/sub‑gradient method) is applied to
   fine‑tune the solution with the true max‑weight.

4. The final vector is returned as a plain Python list; the official
   ``evaluate_sequence`` routine will be used only for the final
   verification (the inner loops use a fast FFT‑based score).

The routine respects the supplied time budget (default 1000 s) and
always returns a feasible non‑negative sequence.
"""

import time
from typing import List
import numpy as np

# ----------------------------------------------------------------------
# Low‑level numeric utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Return the smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Linear convolution via real FFT (full length, zero‑padded to a power‑of‑2)."""
    m = len(a) + len(b) - 1
    n_fft = next_pow2(m)
    A = np.fft.rfft(a, n=n_fft)
    B = np.fft.rfft(b, n=n_fft)
    C = A * B
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


def normalize(v: np.ndarray) -> np.ndarray:
    """Project a non‑negative vector onto the simplex (sum = 1)."""
    total = v.sum()
    if total <= 0.0:
        return np.full_like(v, 1.0 / len(v))
    return v / total


def gradient_from_weights(a: np.ndarray, w: np.ndarray) -> np.ndarray:
    """
    Gradient of Σ_k w_k·(a*a)_k w.r.t. a.
    Implements   g_j = 2· Σ_m a_m · w_{m+j}.
    Computed via one FFT‑based convolution.
    """
    # w is defined on the linear convolution index set [0,2n‑2].
    # Reverse it and convolve with a to obtain the required sums.
    w_rev = w[::-1]
    conv = convolution_fft(a, w_rev)               # length = 3·n‑2
    idx = 2 * len(a) - 2                            # position 2n‑2
    grads = 2.0 * conv[idx - np.arange(len(a))]
    return grads


# ----------------------------------------------------------------------
# Parametric families for the initial sweep
# ----------------------------------------------------------------------


def arcsine_initial(n: int) -> np.ndarray:
    """Discrete arcsine‑shape distribution (≈1/√(i·(n‑1‑i))) normalised."""
    i = np.arange(n, dtype=float)
    eps = 0.25
    vals = 1.0 / np.sqrt((i + 0.5) * (n - 0.5 - i) + eps)
    return normalize(vals)


def powerlaw_shape(n: int, p: float, eps: float) -> np.ndarray:
    """Symmetric power‑law shape a_i ∝ ((i+ε)(n‑1‑i+ε))⁻ᵖ."""
    i = np.arange(n, dtype=float)
    base = ((i + eps) * (n - 1 - i + eps)) ** (-p)
    return normalize(base)


def uniform_shape(n: int) -> np.ndarray:
    """Uniform probability vector on n points."""
    return np.full(n, 1.0 / n)


def random_dirichlet(n: int, rng: np.random.Generator) -> np.ndarray:
    """Draw a random point on the simplex (exponential sampling)."""
    samp = rng.exponential(1.0, size=n)
    return normalize(samp)


# ----------------------------------------------------------------------
# Adam optimiser with soft‑max smoothing of the max‑convolution
# ----------------------------------------------------------------------


def adam_optimize(
    a0: np.ndarray,
    deadline: float,
    *,
    beta_start: float = 2.0,
    beta_end: float = 30.0,
    lr: float = 0.05,
) -> tuple[np.ndarray, float]:
    """
    Optimise a probability vector in the log‑domain using Adam.
    The objective is 2·n·max(conv(a,a)) (sum(a)=1 ⇒ denominator = 1).

    Parameters
    ----------
    a0 : ndarray
        Initial probability vector (already normalised, non‑negative).
    deadline : float
        Unix timestamp – stop optimisation when time.time() >= deadline.
    beta_start, beta_end : float
        Temperature for the soft‑max weight (linear schedule).
    lr : float
        Base learning rate for Adam (applied to the log‑variables).

    Returns
    -------
    best_a : ndarray
        The best vector seen (normalised, non‑negative).
    best_score : float
        Exact score 2·n·max(conv) of best_a.
    """
    a = a0.copy()
    # work in log‑space to guarantee positivity
    eps_log = 1e-30
    z = np.log(a + eps_log)

    n = len(a)
    # bookkeeping for best iterate
    conv = convolution_fft(a, a)
    max_val = float(conv.max())
    best_score = 2.0 * n * max_val
    best_a = a.copy()

    # Adam parameters
    beta1 = 0.9
    beta2 = 0.999
    adam_eps = 1e-8
    m = np.zeros_like(z)
    v = np.zeros_like(z)

    t = 0
    # rough estimate of total iterations for the temperature schedule
    total_iters_est = 200_000

    while time.time() < deadline:
        t += 1

        # ---- convolution and softened max weight ----
        conv = convolution_fft(a, a)
        max_val = float(conv.max())

        # linear temperature schedule
        beta = beta_start + (beta_end - beta_start) * min(1.0, t / total_iters_est)

        # soft‑max weights (max entry gets weight 1, others < 1)
        w = np.exp(beta * (conv - max_val))
        w /= w.sum()                     # ∑ w = 1

        # ---- gradient of the smoothed objective ----
        grad_a = gradient_from_weights(a, w)   # ∂/∂a
        grad_z = a * grad_a                    # chain rule for z = log a

        # optional clipping to keep the Adam updates stable
        clip_val = 10.0
        grad_z = np.clip(grad_z, -clip_val, clip_val)

        # ---- Adam update for z ----
        m = beta1 * m + (1.0 - beta1) * grad_z
        v = beta2 * v + (1.0 - beta2) * (grad_z ** 2)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)

        step = lr * m_hat / (np.sqrt(v_hat) + adam_eps)
        z = z - step

        # back‑to the probability simplex
        a = np.exp(z)
        a = normalize(a)

        # exact score for this iterate (sum(a)=1 ⇒ denominator = 1)
        cur_score = 2.0 * n * max_val
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()

    return best_a, best_score


# ----------------------------------------------------------------------
# Hard sub‑gradient polishing (same as the previous implementation)
# ----------------------------------------------------------------------


def refine_subgradient(
    a: np.ndarray,
    deadline: float,
    *,
    max_iter: int = 30_000,
    eta0: float = 0.7,
    report_every: int = 0,
) -> tuple[np.ndarray, float]:
    """
    Classical sub‑gradient descent using a true max‑weight (uniform over
    maximisers).  Multiplicative updates (exponentiated‑gradient) keep the
    iterate on the simplex.  Returns the best vector seen and its exact
    score 2·n·max(conv).
    """
    a = normalize(a.astype(float).copy())
    best_a = a.copy()
    n = len(a)

    conv = convolution_fft(a, a)
    best_sc = 2.0 * n * float(np.max(conv))

    for it in range(1, max_iter + 1):
        if time.time() >= deadline:
            break

        conv = convolution_fft(a, a)
        max_val = float(np.max(conv))

        # weight uniformly on the (near) maximisers
        w = np.zeros_like(conv)
        max_idx = np.where(conv >= max_val - 1e-14)[0]
        w[max_idx] = 1.0 / len(max_idx)

        grad = gradient_from_weights(a, w)

        eta = eta0 / np.sqrt(it)
        a = a * np.exp(-eta * grad)
        a = normalize(a)

        cur_sc = 2.0 * n * max_val
        if cur_sc < best_sc:
            best_sc = cur_sc
            best_a = a.copy()

        if report_every and (it % report_every == 0):
            print(f"    sub‑grad iter={it:,d} cur={cur_sc:.6f} best={best_sc:.6f}")

    return best_a, best_sc

    print(f"\nSearch finished. Best reported score = {best_score:.6f}")
    return best_a.tolist()
