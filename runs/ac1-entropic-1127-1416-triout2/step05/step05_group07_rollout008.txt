def local_search(a: np.ndarray, conv: np.ndarray, deadline: float, rng: np.random.Generator):
    n = len(a)
    best_a = a.copy()
    best_conv = conv.copy()
    best_max = best_conv.max()
    best_obj = 2 * n * best_max / (a.sum() ** 2)   # sum =1

    # compute conv length: L = 2*n -1
    L = 2 * n -1

    # precompute Dirac arrays for later (optional)
    # but we can produce them on the fly.

    # Determine number of iterations based on time.
    while time.time() < deadline - 0.1:
        # choose random i with a_i > epsilon
        i = rng.integers(n)
        if a[i] < 1e-12:
            continue
        j = rng.integers(n)
        while j == i:
            j = rng.integers(n)

        # choose delta random fraction
        # to avoid too small steps, scale delta relative to a_i
        delta = a[i] * rng.random() * 0.5    # up to 50% of a_i
        if delta <= 0:
            continue

        # update temporary conv
        # compute shift_i = np.concatenate([np.zeros(i), a, np.zeros(n-1-i)])
        # shift_j similarly; but we can compute as slices.

        # creating new conv quick:
        conv_new = best_conv.copy()
        # shift for j
        start_j = j
        conv_new[start_j : start_j + n] += 2 * delta * a
        # shift for i
        start_i = i
        conv_new[start_i : start_i + n] -= 2 * delta * a
        # delta^2 term
        idx_2j = 2 * j
        idx_2i = 2 * i
        idx_ij = i + j
        if idx_2j < L:
            conv_new[idx_2j] += delta * delta
        if idx_2i < L:
            conv_new[idx_2i] += delta * delta
        if idx_ij < L:
            conv_new[idx_ij] -= 2 * delta * delta

        # compute new max
        new_max = conv_new.max()
        # calculate new objective
        # sum stays 1
        new_obj = 2 * n * new_max / (a.sum() ** 2)  # sum constant 1

        if new_obj < best_obj - 1e-10:  # accept improvement
            # apply update to a and conv
            a[i] -= delta
            a[j] += delta
            best_conv = conv_new
            best_max = new_max
            best_obj = new_obj
            best_a = a.copy()
            # maybe re-normalize due to floating errors: sum close to 1.
            # also if a[i] becomes very small, zero.
        # else maybe accept with small probability? optional.

        # optionally recompute conv from scratch every X iterations to reduce drift.
        # can keep a counter.

    return best_a, best_obj

def local_search(a, conv, deadline, rng):
    n = len(a)
    L = 2*n-1
    best_a = a.copy()
    best_conv = conv.copy()
    best_max = best_conv.max()
    best_obj = 2*n*best_max / (a.sum()**2)

    iter = 0
    recompute_interval = 5000
    while time.time() < deadline - 0.05:
        # random i with a_i > 1e-12
        i = rng.integers(n)
        if a[i] < 1e-8:
            continue
        # random j
        j = rng.integers(n)
        while j == i:
            j = rng.integers(n)
        # choose delta: a fraction of a_i (maybe geometric distribution)
        frac = rng.random() ** 2  # bias towards small moves
        delta = a[i] * frac * 0.5  # up to 50% of a_i
        if delta <= 0:
            continue

        # compute conv_new
        conv_new = best_conv.copy()
        start_i = i
        start_j = j
        # shift contributions: 2*delta*(a - ...?). important: must use a before move.
        conv_new[start_j:start_j + n] += 2.0 * delta * a
        conv_new[start_i:start_i + n] -= 2.0 * delta * a

        idx_2i = 2*i
        idx_ij = i + j

            conv_new[idx_2j] += delta * delta
        if idx_2i < L:
            conv_new[idx_2i] += delta * delta
        if idx_ij < L:
            conv_new[idx_ij] -= 2.0 * delta * delta

        # clamp negative values to zero if any; not strictly necessary.
        # compute new max
        new_max = conv_new.max()
        new_obj = 2.0 * n * new_max / (a.sum() ** 2)  # sum approx 1

        # Accept if better

            # update a, conv (in place)
            a[i] -= delta
            a[j] += delta
            # maintain sum float maybe tiny drift; after update, we could renorm

    # iteration parameters
    it = 0
    grad = None
    while time.time() < deadline - 0.01:
        # recompute gradient every 500 steps
        if it % 500 == 0:
            # compute gradient w.r.t max conv
            conv_full = best_conv  # conv already known
            grad = _max_conv_gradient(best_a, conv_full)
        # pick i from top 10% gradient, j from bottom 10%
        # create index arrays sorted.
        # but for speed, just pick argmax and argmin.
        i = int(np.argmax(grad))
        j = int(np.argmin(grad))
        if i == j:
            # fallback to random index
            i = rng.integers(n)
            j = (i + rng.integers(n-1) + 1) % n
        if best_a[i] <= 1e-12:
            # cannot move, pick random i
            i = rng.integers(n)
            continue
        delta = best_a[i] * rng.random() * 0.5

        conv_new = best_conv.copy()
        conv_new[j:j+n] += 2.0 * delta * best_a
        conv_new[i:i+n] -= 2.0 * delta * best_a
        idx_2j = 2*j
        idx_2i = 2*i
        idx_ij = i+j
        if idx_2j < L:
            conv_new[idx_2j] += delta*delta
        if idx_2i < L:
            conv_new[idx_2i] += delta*delta
        if idx_ij < L:
            conv_new[idx_ij] -= 2.0 * delta*delta
        new_max = conv_new.max()
        new_obj = 2.0 * n * new_max / (best_a.sum() ** 2)

      that most increases the max‑convolution (gradient of the active
      constraints) to an index that contributes the least;
    - update the convolution exactly (O(n) per move) and accept the move
      only when the objective improves.
    Periodically the convolution is recomputed from scratch to avoid
    drift.
4.  If time permits, up‑sample the current best vector to a larger size
    (linear interpolation, re‑normalisation) and repeat steps 2‑3.
5.  Return the best feasible list found.  The routine respects the
    supplied 1000 s wall‑clock budget and prints progress.

The implementation uses only NumPy and SciPy (for the optional linear
program that is *not* required here) and is fully self‑contained.
"""

import time
import math
from typing import List, Tuple

import numpy as np


# ----------------------------------------------------------------------
# Low‑level numeric helpers
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Linear convolution using a real FFT (fast and accurate)."""
    m = len(a) + len(b) - 1
    n_fft = next_pow2(m)
    A = np.fft.rfft(a, n=n_fft)
    B = np.fft.rfft(b, n=n_fft)
    C = A * B
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


def compute_score(a: np.ndarray) -> float:
    """
    Fast surrogate for the official evaluator:
        2·n·max(a*a) / (sum(a)²)
    """
    n = len(a)
    conv = convolution_fft(a, a)
    max_val = float(conv.max())
    s = float(a.sum())
    if s < 1e-12:
        return float("inf")
    return 2.0 * n * max_val / (s * s)


# ----------------------------------------------------------------------
# Families of initial shapes
# ----------------------------------------------------------------------


def arcsine_initial(n: int) -> np.ndarray:
    """
    Discrete arcsine density a_i ∝ 1 / √((i+½)(n‑½‑i)).
    Returned vector is already normalised to sum = 1.
    """
    i = np.arange(n, dtype=float)
    eps = 0.25                     # tiny regulariser
    vals = 1.0 / np.sqrt((i + 0.5) * (n - 0.5 - i) + eps)
    vals /= vals.sum()
    return vals


def random_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    """
    Simple Dirichlet‑like draw.
    conc > 0 → Gamma(α=conc,1); conc ≤ 0 → heavy‑tailed exponential power.
    Returned vector is normalised.
    """
    if conc > 0:
        samp = rng.gamma(conc, 1.0, size=n)
    else:
        # heavy‑tailed: exponentiate an exponential variable
        samp = rng.exponential(1.0, size=n) ** (1.0 / max(0.1, -conc))
    samp += 1e-12
    samp /= samp.sum()
    return samp


def perturb(a: np.ndarray, rng: np.random.Generator, scale: float = 0.02) -> np.ndarray:
    """Add a tiny Gaussian jitter and re‑normalise."""
    b = a + rng.normal(0.0, scale, size=a.shape)
    b = np.maximum(b, 0.0)
    if b.sum() == 0.0:
        return np.full_like(a, 1.0 / len(a))
    return b / b.sum()


def upsample(prev: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation from len(prev) → new_n followed by normalisation."""
    x_old = np.linspace(0.0, 1.0, len(prev))
    x_new = np.linspace(0.0, 1.0, new_n)
    interp = np.interp(x_new, x_old, prev)
    total = interp.sum()
    if total == 0.0:
        return np.full(new_n, 1.0 / new_n)
    return interp / total


# ----------------------------------------------------------------------
# Sub‑gradient optimiser (exponentiated‑gradient / mirror descent)
# ----------------------------------------------------------------------


def _max_conv_gradient(a: np.ndarray, conv: np.ndarray) -> np.ndarray:
    """
    Sub‑gradient of max_k (a∗a)_k.
    For every active offset k the gradient contribution is
        g_j = 2·a_{k‑j}   (zero when the index is out of range).
    If several offsets are active we average their contributions.
    """
    n = len(a)
    max_val = conv.max()
    mask = np.isclose(conv, max_val, atol=1e-15, rtol=0.0)
    active = np.where(mask)[0]
    grad = np.zeros_like(a)

    for k in active:
        lo = max(0, k - (n - 1))
        hi = min(k, n - 1)
        idx = np.arange(lo, hi + 1, dtype=int)
        grad[idx] += 2.0 * a[k - idx]

    if len(active) > 1:
        grad /= len(active)
    return grad


def optimise_subgradient(
    init_a: np.ndarray,
    deadline: float,
    *,
    eta0: float = 0.6,
    max_iter: int = 100_000,
    noise_every: int = 0,
    noise_scale: float = 0.0,
) -> Tuple[np.ndarray, float]:
    """
    Exponentiated‑gradient (mirror‑descent) on the surrogate objective
    Φ(a)=max_k (a∗a)_k, constrained to the probability simplex.
    Returns the best vector seen and its surrogate score.
    """
    a = np.maximum(init_a, 1e-30)
    a /= a.sum()
    best_a = a.copy()
    best_sc = compute_score(a)

    n = len(a)

    for it in range(1, max_iter + 1):
        if time.time() >= deadline:
            break

        conv = convolution_fft(a, a)
        grad = _max_conv_gradient(a, conv)

        eta = eta0 / math.sqrt(it)
        a = a * np.exp(-eta * grad)          # multiplicative update

        if noise_every and (it % noise_every == 0):
            a += np.random.randn(n) * noise_scale
            a = np.maximum(a, 0.0)

        s = a.sum()
        if s == 0.0:
            a = np.full_like(a, 1.0 / n)
        else:
            a = a / s

        cur_sc = compute_score(a)
        if cur_sc < best_sc:
            best_sc = cur_sc
            best_a = a.copy()

        if it % 5000 == 0:
            print(f"    subgrad it={it:,d} cur={cur_sc:.6f} best={best_sc:.6f}")

    return best_a, best_sc


# ----------------------------------------------------------------------
# Directed local‑search optimiser (greedy mass transfer)
# ----------------------------------------------------------------------


def _local_search_directed(
    a_init: np.ndarray,
    conv_init: np.ndarray,
    deadline: float,
    rng: np.random.Generator,
) -> Tuple[np.ndarray, float]:
    """
    Greedy mass‑transfer search.

    Repeatedly picks the index i that most *increases* the current maximal
    convolution (largest entry of the sub‑gradient) and moves a small
    amount of probability mass from i to the index j that contributes the
    *least* (smallest gradient entry).  The convolution is updated exactly
    in O(n) time using the known shift structure.
    """
    a = a_init.copy()
    conv = conv_init.copy()
    n = len(a)
    L = 2 * n - 1

    # current objective (sum(a) should stay ≈1)
    best_obj = 2.0 * n * conv.max() / (a.sum() ** 2)
    best_a = a.copy()
    best_conv = conv.copy()

    it = 0
    grad = None

    while time.time() < deadline - 0.01:
        # recompute gradient every few iterations
        if it % 200 == 0:
            grad = _max_conv_gradient(a, conv)

        # indices with extreme gradient values
        i = int(np.argmax(grad))
        j = int(np.argmin(grad))
        if i == j:
            # fall back to a random pair
            i = rng.integers(n)
            j = rng.integers(n - 1)
            if j >= i:
                j += 1

        if a[i] <= 1e-14:
            # cannot take mass from an (almost) empty cell – skip
            it += 1
            continue

        # move a random fraction (up to 50 %) of the mass at i
        delta = a[i] * rng.random() * 0.5

        # ---- update the convolution exactly (linear time) ----
        conv_new = conv.copy()
        # contribution of the cross terms 2·δ·a shifted
        conv_new[j : j + n] += 2.0 * delta * a
        conv_new[i : i + n] -= 2.0 * delta * a

        # δ² terms (very small, kept for correctness)
        idx_2j = 2 * j
        idx_2i = 2 * i
        idx_ij = i + j
        if idx_2j < L:
            conv_new[idx_2j] += delta * delta
        if idx_2i < L:
            conv_new[idx_2i] += delta * delta
        if idx_ij < L:
            conv_new[idx_ij] -= 2.0 * delta * delta

        new_max = conv_new.max()
        new_obj = 2.0 * n * new_max / (a.sum() ** 2)

        # accept only improving moves
        if new_obj < best_obj - 1e-12:
            a[i] -= delta
            a[j] += delta
            conv = conv_new
            best_obj = new_obj
            best_a = a.copy()
            best_conv = conv.copy()

        # occasional full recomputation to remove accumulated numerical drift
        if it % 5000 == 0:
            conv = convolution_fft(a, a)
            best_obj = 2.0 * n * conv.max() / (a.sum() ** 2)
            best_a = a.copy()
            best_conv = conv.copy()

        it += 1

    return best_a, best_obj


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: int = 1000) -> List[float]:
    """
    Search for a non‑negative step‑function with a low ``evaluate_sequence``
    value (the official objective).  The routine respects the supplied
    time budget, prints progress, and always returns a feasible list.
    """
    rng = np.random.default_rng(seed)
    start = time.time()
    deadline = start + budget_s - 2.0            # tiny safety margin

    best_seq: List[float] = []
    best_val: float = float("inf")
    best_a: np.ndarray = None                    # normalised best vector

    # ------------------------------------------------------------------
    # 0.  Optional fallback from a previously discovered construction.
    # ------------------------------------------------------------------
    try:
        fallback = [float(x) for x in height_sequence_1]   # type: ignore
        val = evaluate_sequence(fallback)
        print(f"Fallback (height_sequence_1) score = {val:.6f}")
        if val < best_val:
            best_val = val
            best_seq = fallback
            best_a = np.array(fallback, dtype=float)
            best_a /= best_a.sum()
    except Exception:
        pass

    # ------------------------------------------------------------------
    # 1.  Length hierarchy – start modest, enlarge progressively.
    # ------------------------------------------------------------------
    lengths = [1024, 2048, 4096, 8192, 16384]   # stop earlier if time runs out

    for n in lengths:
        if time.time() >= deadline - 5.0:
            break
        print(f"\n=== Working at length n = {n} ===")

        # --------------------------------------------------------------
        # 1a. Build a pool of starting vectors.
        # --------------------------------------------------------------
        candidates: List[np.ndarray] = []

        if best_a is not None:
            candidates.append(upsample(best_a, n))          # warm‑start

        candidates.append(arcsine_initial(n))                # classic baseline

        # Dirichlet draws with different concentrations
        for conc in (0.5, 1.0, 2.0):
            candidates.append(random_dirichlet(n, conc, rng))

        # a few jittered copies of the arcsine shape
        for _ in range(2):
            candidates.append(perturb(arcsine_initial(n), rng, scale=0.02))

        # --------------------------------------------------------------
        # 1b. Quick surrogate screening.
        # --------------------------------------------------------------
        scored = [(compute_score(c), c) for c in candidates]
        scored.sort(key=lambda x: x[0])
        top_candidates = [c for _, c in scored[:3]]       # keep only the most promising

        # --------------------------------------------------------------
        # 1c. Refine each retained candidate with the sub‑gradient routine.
        # --------------------------------------------------------------
        for idx, cand in enumerate(top_candidates):
            if time.time() >= deadline - 5.0:
                break
            print(f"  Sub‑gradient refinement of candidate {idx + 1}/{len(top_candidates)}")

            remaining = max(0.0, deadline - time.time())
            # allocate a modest number of iterations (depends on remaining time)
            max_iter = int(min(120_000, max(5_000, remaining * 150)))

            refined, _ = optimise_subgradient(
                init_a=cand,
                deadline=deadline,
                eta0=0.6,
                max_iter=max_iter,
                noise_every=600,
                noise_scale=4e-6,
            )
            val = evaluate_sequence(refined.tolist())
            print(f"    official score = {val:.6f}")

            if val < best_val:
                best_val = val
                best_seq = refined.tolist()
                best_a = refined.copy()
                print(f"    *** New global best: {best_val:.6f} ***")

    # ------------------------------------------------------------------
    # 2.  Directed local‑search polishing (the heavy hitter).
    # ------------------------------------------------------------------
    if best_a is not None and time.time() < deadline - 5.0:
        n = len(best_a)
        print("\n=== Starting directed local search ===")
        conv = convolution_fft(best_a, best_a)

        # run until the wall‑clock expires
        refined, _ = _local_search_directed(best_a, conv, deadline, rng)

        val = evaluate_sequence(refined.tolist())
        print(f"Local‑search final score = {val:.6f}")
        if val < best_val:
            best_val = val
            best_seq = refined.tolist()
            best_a = refined
            print("*** Updated global best after local search ***")

    # ------------------------------------------------------------------
    # 3.  Optional up‑sampling + second local‑search pass if time permits.
    # ------------------------------------------------------------------
    if best_a is not None and time.time() < deadline - 5.0:
        # try one more level (double the length) – often yields a tiny extra gain
        new_n = len(best_a) * 2
        if new_n <= 65536:                     # keep memory usage reasonable
            print(f"\n=== Upsampling to {new_n} and polishing again ===")
            up = upsample(best_a, new_n)
            conv_up = convolution_fft(up, up)

            # a short sub‑gradient round (very cheap at this stage)
            up_refined, _ = optimise_subgradient(
                init_a=up,
                deadline=deadline,
                eta0=0.4,
                max_iter=30_000,
                noise_every=0,
                noise_scale=0.0,
            )
            # final directed local search on the enlarged vector
            up_refined, _ = _local_search_directed(
                up_refined, convolution_fft(up_refined, up_refined), deadline, rng
            )
            val = evaluate_sequence(up_refined.tolist())
            print(f"Upsampled final score = {val:.6f}")
            if val < best_val:
                best_val = val
                best_seq = up_refined.tolist()
                best_a = up_refined
                print("*** New best after up‑sampling ***")

    # ------------------------------------------------------------------
    # 4.  Guard – always return a feasible sequence.
    # ------------------------------------------------------------------
    if not best_seq:
        best_seq = [1.0] * 256
        best_val = evaluate_sequence(best_seq)

    print(f"\nSearch finished. Best reported score = {best_val:.6f}")
    return best_seq
