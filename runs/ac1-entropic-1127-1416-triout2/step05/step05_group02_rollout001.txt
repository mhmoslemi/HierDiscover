def gradient_descent_smooth(initial_a, deadline, max_iters=200000, beta_start=2.0, beta_end=80.0, beta_steps=6, step0=0.2):
    a = project_to_simplex(initial_a.copy())
    n = len(a)
    it = 0
    # Determine schedule for beta: linear or exponential
    betas = np.geomspace(beta_start, beta_end, num=beta_steps)
    for beta in betas:
        # allocate some iterations for each beta level based on remaining time and max_iters
        while time.time() < deadline and it < max_iters:
            it += 1
            # compute smooth objective and gradient
            conv = convolution_fft(a, a)  # len 2n-1
            # Compute softmax
            beta_conv = beta * conv
            max_bc = beta_conv.max()
            # stable softmax
            exp_vals = np.exp(beta_conv - max_bc)
            sum_exp = exp_vals.sum()
            w = exp_vals / sum_exp
            # smooth objective (only for logging)
            obj = (max_bc + np.log(sum_exp)) / beta
            # gradient
            a_rev = a[::-1]
            conv_rev = convolution_fft(w, a_rev)
            grad = 2.0 * conv_rev[n-1:n-1+n]  # shape n
            # step size
            eta = step0 / np.sqrt(it)
            a = a - eta * grad
            a = project_to_simplex(a)
            # optional logging occasional
            if it % 5000 == 0:
                sc = compute_score(a)
                print(f"    beta={beta:.2f} iter={it} surrogate={obj:.6f} score={sc:.6f}")
    return a

"""
Improved search for a low‑value step function.

The algorithm works in three phases:

1️⃣  **Seed generation** –  a diverse pool of vectors of different
    lengths (1024, 2048, 4096) built from analytically‑motivated
    families (arcsine, beta, power, mixtures) and from the previous
    best construction if it is available.

2️⃣  **Smooth‑max gradient descent** –  minimise a differentiable
    surrogate of the true objective,
        fβ(a) = (1/β)·log∑exp(β·(a*a)),
    using projected gradient descent.  The temperature β is increased
    geometrically (5 → 80) so that the optimisation first sees a very
    smooth landscape and later focuses on the actual maximum.  The
    gradient of the surrogate is obtained analytically via a second
    convolution; the iterate is always projected onto the probability
    simplex.

3️⃣  **Exact polishing** –  a short run of the sub‑gradient method from
    the reference solution plus a final exponentiated‑gradient (EG)
    polishing step.  The best vector seen throughout the run is kept
    and finally evaluated with the official ``evaluate_sequence``.

All steps respect the global 1000 s budget; the routine returns a
non‑empty list of non‑negative heights.
"""

import time
import numpy as np
from typing import List, Tuple

# ----------------------------------------------------------------------
# Low‑level utilities (no closures)
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Linear convolution via real FFT (returns length len(a)+len(b)-1)."""
    m = a.size + b.size - 1
    n_fft = next_pow2(m)
    A = np.fft.rfft(a, n=n_fft)
    B = np.fft.rfft(b, n=n_fft)
    C = A * B
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection onto the probability simplex."""
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / v.size)

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho_arr = u - cssv / np.arange(1, v.size + 1)
    rho_candidates = np.where(rho_arr > 0)[0]
    if rho_candidates.size == 0:
        theta = 0.0
    else:
        rho = rho_candidates[-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def compute_score(a: np.ndarray) -> float:
    """Fast surrogate: 2·n·max(conv(a,a)) for a with Σa=1."""
    n = a.shape[0]
    conv = convolution_fft(a, a)
    return 2.0 * n * float(np.max(conv))

# Parametric families (used for seeding)

def arcsine_initial(n: int) -> np.ndarray:
    """Classic arcsine shape."""
    i = np.arange(n, dtype=float)
    eps = 0.25
    vals = 1.0 / np.sqrt((i + 0.5) * (n - 0.5 - i) + eps)
    vals /= vals.sum()
    return vals


def generate_beta_shape(n: int, alpha: float, beta: float, eps: float) -> np.ndarray:
    """(i+eps)^{α‑1}·(n‑1‑i+eps)^{β‑1}, normalised."""
    i = np.arange(n, dtype=float)
    left = (i + eps) ** (alpha - 1.0)
    right = (n - 1 - i + eps) ** (beta - 1.0)
    base = left * right
    total = base.sum()
    if total <= 0.0 or not np.isfinite(total):
        return np.full(n, 1.0 / n)
    return base / total


def generate_power_shape(n: int, p: float, eps: float) -> np.ndarray:
    """((i+½)*(n‑½‑i)+eps)^{‑p}, normalised."""
    i = np.arange(n, dtype=float)
    denom = ((i + 0.5) * (n - 0.5 - i) + eps) ** p
    vals = 1.0 / denom
    total = vals.sum()
    if total <= 0.0 or not np.isfinite(total):
        return np.full(n, 1.0 / n)
    return vals / total


def generate_mixture_shape(n: int, weight_arc: float) -> np.ndarray:
    """Mixture (weight_arc)·arcsine + (1‑weight_arc)·uniform."""
    arc = arcsine_initial(n)
    uni = np.full(n, 1.0 / n)
    mix = weight_arc * arc + (1.0 - weight_arc) * uni
    mix /= mix.sum()
    return mix

def subgradient_descent_exact(initial_a: np.ndarray,
                              deadline: float,
                              max_iter: int = 80_000,
                              step0: float = 0.05) -> np.ndarray:
    """Projected sub‑gradient descent on the true objective."""
    a = project_to_simplex(initial_a.copy())
    n = a.size
    L = 2 * n - 1

    for it in range(1, max_iter + 1):
        if time.time() >= deadline:
            break

        conv = convolution_fft(a, a)
        max_val = float(np.max(conv))
        tol = max_val * 1e-12 + 1e-14
        mask = (np.abs(conv - max_val) <= tol).astype(float)

        # sub‑gradient = 2·(mask * a) convolved and shifted
        mask_rev = mask[::-1]
        conv_rev = convolution_fft(a, mask_rev)
        subgrad = 2.0 * conv_rev[L - 1 - np.arange(n)]

        eta = step0 / np.sqrt(it)
        a = a - eta * subgrad
        a = project_to_simplex(a)

        if it % 5000 == 0:
            print(f"    SubGrad it={it:,d} surrogate={compute_score(a):.6f}")

    return a


def eg_refine(a: np.ndarray,
              deadline: float,
              max_iter: int = 100_000,
              eta0: float = 0.3) -> np.ndarray:
    """Exponentiated‑gradient polishing (mirror descent)."""
    n = a.size
    L = 2 * n - 1

    for it in range(1, max_iter + 1):
        if time.time() >= deadline:
            break

        conv = convolution_fft(a, a)
        max_val = float(np.max(conv))
        tol = max_val * 1e-12 + 1e-14
        mask = (np.abs(conv - max_val) <= tol).astype(float)

        mask_rev = mask[::-1]
        conv_rev = convolution_fft(a, mask_rev)
        subgrad = 2.0 * conv_rev[L - 1 - np.arange(n)]

        eta = eta0 / (np.sqrt(it) + 1.0)
        a = a * np.exp(-eta * subgrad)

        total = a.sum()
        if total == 0.0:
            a = np.full_like(a, 1.0 / n)
        else:
            a = a / total

        if it % 5000 == 0:
            print(f"    EG it={it:,d} surrogate={compute_score(a):.6f}")

    return a

# Smooth‑max gradient descent (new component)
# ----------------------------------------------------------------------


def smooth_max_gradient_descent(initial_a: np.ndarray,
                               deadline: float,
                               max_iters: int = 200_000,
                               beta_start: float = 5.0,
                               beta_end: float = 80.0,
                               beta_steps: int = 6,
                               step0: float = 0.3) -> np.ndarray:
    """
    Projected gradient descent on the smooth surrogate
        fβ(a) = (1/β)·log∑exp(β·conv(a,a)).
    The temperature β is increased geometrically from beta_start to
    beta_end in `beta_steps` stages.
    """
    a = project_to_simplex(initial_a.copy())
    n = a.size
    a_rev = a[::-1]                     # will be updated each iteration
    it = 0
    betas = np.geomspace(beta_start, beta_end, num=beta_steps)

    for beta in betas:
        while it < max_iters and time.time() < deadline:
            it += 1
            # ----- surrogate value & soft‑max weights -----
            conv = convolution_fft(a, a)          # length 2n‑1
            bc = beta * conv
            bc_max = bc.max()
            exp_shift = np.exp(bc - bc_max)      # safe exponentiation
            Z = exp_shift.sum()
            w = exp_shift / Z                     # soft‑max (∑w = 1)

            # ----- gradient of the surrogate -----
            # grad = 2·(w * a) convolved (see derivation)
            a_rev = a[::-1]                       # keep up‑to‑date
            conv_rev = convolution_fft(w, a_rev)  # length 3n‑2
            grad = 2.0 * conv_rev[n - 1:n - 1 + n]  # central slice

            # ----- projected update -----
            eta = step0 / np.sqrt(it)
            a = a - eta * grad
            a = project_to_simplex(a)

            # occasional logging
            if it % 5000 == 0:
                surrogate = (bc_max + np.log(Z)) / beta
                print(f"    β={beta:.2f} it={it:,d} surrogate={surrogate:.6f} "
                      f"fast_score={compute_score(a):.6f}")

    return a
