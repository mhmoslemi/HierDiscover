4️⃣  **Final polish** – a last round of sub‑gradient descent and

def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Linear convolution via real FFT (length = len(a)+len(b)-1)."""
    m = len(a) + len(b) - 1
    n_fft = next_pow2(m)
    A = np.fft.rfft(a, n=n_fft)
    B = np.fft.rfft(b, n=n_fft)
    C = A * B
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


def compute_score(a: np.ndarray) -> float:
    """
    Fast surrogate for the official score.
    The vector is assumed to be normalised to Σa = 1,
    therefore the denominator disappears and the score is
    simply 2·n·max(conv(a,a)).
    """
    n = a.shape[0]
    conv = convolution_fft(a, a)
    return 2.0 * n * float(np.max(conv))


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection onto the probability simplex {x ≥ 0, Σx = 1}.
    Duchi et al., 2008.
    """
    if np.all(v <= 0):
        # uniform fallback – should never happen in practice
        return np.full_like(v, 1.0 / v.size)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho_arr = u - cssv / np.arange(1, v.size + 1)
    rho = np.where(rho_arr > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Shape generators
# ----------------------------------------------------------------------


def arcsine_initial(n: int) -> np.ndarray:
    """Classic arcsine shape: a_i ∝ 1/√((i+½)*(n‑½‑i))."""
    i = np.arange(n, dtype=float)
    eps = 0.25
    vals = 1.0 / np.sqrt((i + 0.5) * (n - 0.5 - i) + eps)
    vals /= vals.sum()
    return vals


def generate_beta_shape(n: int, alpha: float, beta: float, eps: float) -> np.ndarray:
    """
    (i+ε)^{α‑1}·(n‑1‑i+ε)^{β‑1} on {0,…,n‑1}, normalised to Σ=1.
    """
    i = np.arange(n, dtype=float)
    left = (i + eps) ** (alpha - 1.0)
    right = (n - 1 - i + eps) ** (beta - 1.0)
    base = left * right
    total = base.sum()
    if total <= 0.0 or not np.isfinite(total):
        return np.full(n, 1.0 / n)
    return base / total


def generate_power_shape(n: int, p: float, eps: float) -> np.ndarray:
    """
    Generalised arcsine: a_i ∝ ((i+½)*(n‑½‑i)+ε)^{‑p}.
    p>0, eps≥0.
    """
    i = np.arange(n, dtype=float)
    denom = ((i + 0.5) * (n - 0.5 - i) + eps) ** p
    vals = 1.0 / denom
    total = vals.sum()
    if total <= 0.0 or not np.isfinite(total):
        return np.full(n, 1.0 / n)
    return vals / total


def generate_mixture_shape(n: int, weight_arc: float) -> np.ndarray:
    """Mixture (weight_arc)·arcsine + (1‑weight_arc)·uniform."""
    arc = arcsine_initial(n)
    uni = np.full(n, 1.0 / n)
    mix = weight_arc * arc + (1.0 - weight_arc) * uni
    mix /= mix.sum()
    return mix


# ----------------------------------------------------------------------
# Optimisation primitives
# ----------------------------------------------------------------------


def subgradient_descent_exact(
    initial_a: np.ndarray,
    deadline: float,
    max_iter: int = 80_000,
    step0: float = 0.05,
) -> np.ndarray:
    """
    Projected sub‑gradient descent on the exact objective
    f(a)=2·n·max(conv(a,a)).  The step size is ηₜ = step0/√t.
    """
    a = project_to_simplex(initial_a.copy())
    n = len(a)
    L = 2 * n - 1

    for it in range(1, max_iter + 1):
        if time.time() >= deadline:
            break

        conv = convolution_fft(a, a)
        max_val = float(np.max(conv))
        # active set (tolerance for floating point)
        tol = max_val * 1e-12 + 1e-14
        mask = (np.abs(conv - max_val) <= tol).astype(float)

        # sub‑gradient = 2·∑_{k∈K} shift(a,k)
        mask_rev = mask[::-1]
        conv_rev = convolution_fft(a, mask_rev)
        subgrad = 2.0 * conv_rev[L - 1 - np.arange(n)]

        eta = step0 / np.sqrt(it)
        a = a - eta * subgrad
        a = project_to_simplex(a)

        # occasional progress report
        if it % 5000 == 0:
            cur_sc = compute_score(a)
            print(f"    SubGrad it={it:,d} score={cur_sc:.6f}")

    return a


def eg_refine(
    a: np.ndarray,
    deadline: float,
    max_iter: int = 100_000,
    eta0: float = 0.3,
) -> np.ndarray:
    """
    Mirror‑descent (exponentiated‑gradient) polishing.
    Keeps the iterate on the simplex automatically.
    """
    n = len(a)
    L = 2 * n - 1
    for it in range(1, max_iter + 1):
        if time.time() >= deadline:
            break
        conv = convolution_fft(a, a)
        max_val = float(np.max(conv))
        tol = max_val * 1e-12 + 1e-14
        mask = (np.abs(conv - max_val) <= tol).astype(float)

        mask_rev = mask[::-1]
        conv_rev = convolution_fft(a, mask_rev)
        subgrad = 2.0 * conv_rev[L - 1 - np.arange(n)]

        eta = eta0 / (np.sqrt(it) + 1.0)
        a = a * np.exp(-eta * subgrad)

        total = a.sum()
        if total == 0.0:
            a = np.full_like(a, 1.0 / n)
        else:
            a = a / total

        if it % 5000 == 0:
            cur_sc = compute_score(a)
            print(f"    EG it={it:,d} score={cur_sc:.6f}")

    return a


# ----------------------------------------------------------------------
# Main entry point required by the competition harness
# ----------------------------------------------------------------------


def propose_candidate(
    seed: int = 42,
    budget_s: int = 1000,
    **kwargs,
) -> List[float]:
    """
    Search for a non‑negative step sequence with a low
    ``evaluate_sequence`` value.  The routine combines
    diversified seed generation, long projected sub‑gradient
    descent, exponentiated‑gradient polishing and a jitter‑restart
    local search.  All work respects the global time budget.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 1.0  # safety margin

    best_seq: List[float] = []
    best_val: float = float("inf")
    best_fast: float = float("inf")

    # ------------------------------------------------------------------
    # 0.  Optional fallback from a previous run
    # ------------------------------------------------------------------
    if "height_sequence_1" in globals():
        try:
            fallback = [float(x) for x in globals()["height_sequence_1"]]
            val = evaluate_sequence(fallback)
            print(f"Fallback (height_sequence_1) official score = {val:.6f}")
            if val < best_val:
                best_val = val
                best_seq = fallback
                best_fast = compute_score(np.array(fallback))
        except Exception:
            pass

    # ------------------------------------------------------------------
    # 1.  Seed generation (two lengths)
    # ------------------------------------------------------------------
    lengths = [2048, 4096]  # 4096 gives a finer discretisation at modest cost
    raw_seeds: List[np.ndarray] = []

    for n in lengths:
        if time.time() >= deadline - 10.0:
            break
        print(f"\n--- Generating seeds for length {n} ---")
        # deterministic starters
        raw_seeds.append(arcsine_initial(n))
        raw_seeds.append(np.full(n, 1.0 / n))

        # random beta family
        for _ in range(30):
            if time.time() >= deadline - 5.0:
                break
            a = rng.uniform(0.05, 1.0)
            b = rng.uniform(0.05, 1.0)
            eps = rng.uniform(0.0, max(1.0, 0.02 * n))
            raw_seeds.append(generate_beta_shape(n, a, b, eps))

        # random power‑shape family
        for _ in range(30):
            if time.time() >= deadline - 5.0:
                break
            p = rng.uniform(0.2, 3.5)
            eps = rng.uniform(0.0, 5.0)
            raw_seeds.append(generate_power_shape(n, p, eps))

        # arcsine‑uniform mixtures
        for _ in range(20):
            if time.time() >= deadline - 5.0:
                break
            w = rng.random()
            raw_seeds.append(generate_mixture_shape(n, w))

    # ------------------------------------------------------------------
    # 2.  Keep only the most promising seeds (fast surrogate)
    # ------------------------------------------------------------------
    scored_seeds = [(compute_score(v), v) for v in raw_seeds]
    scored_seeds.sort(key=lambda x: x[0])
    top_seeds = [vec for _, vec in scored_seeds[:min(12, len(scored_seeds))]]
    print(
        f"\nSelected {len(top_seeds)} top seeds for intensive refinement "
        f"(fast scores range {scored_seeds[0][0]:.6f} – {scored_seeds[-1][0]:.6f})"
    )

    # ------------------------------------------------------------------
    # 3.  Intensive refinement of each top seed
    # ------------------------------------------------------------------
    for idx, seed_vec in enumerate(top_seeds, start=1):
        if time.time() >= deadline - 4.0:
            break
        n = seed_vec.shape[0]
        print(f"\nRefining seed {idx}/{len(top_seeds)} (n={n}) …")

        # ---- sub‑gradient descent ----
        remaining = deadline - time.time()
        max_iter_sub = int(min(200_000, remaining * 30))
        refined = subgradient_descent_exact(
            seed_vec,
            deadline,
            max_iter=max_iter_sub,
            step0=0.22,  # a relatively large initial step
        )

        # ---- EG polishing ----
        if time.time() < deadline - 2.0:
            remaining = deadline - time.time()
            max_iter_eg = int(min(120_000, remaining * 30))
            refined = eg_refine(
                refined,
                deadline,
                max_iter=max_iter_eg,
                eta0=0.22,
            )

        fast_sc = compute_score(refined)
        print(f"    Fast surrogate after polish: {fast_sc:.6f}")

        # Only invoke the official evaluator when we have a chance to improve
        if fast_sc < best_fast - 1e-6:
            try:
                val = evaluate_sequence(refined.tolist())
                print(f"    Official score = {val:.6f}")
                if val < best_val:
                    best_val = val
                    best_seq = refined.tolist()
                    best_fast = fast_sc
                    print("    *** New global best ***")
            except Exception as e:
                print(f"    Evaluation failed: {e}")

    # ------------------------------------------------------------------
    # 4.  Jitter‑restart local search from the current best
    # ------------------------------------------------------------------
    while time.time() < deadline - 3.0 and best_seq:
        n = len(best_seq)
        base = np.array(best_seq, dtype=float)

        # decreasing jitter magnitude as the budget is spent
        elapsed = time.time() - start_time
        frac = elapsed / budget_s
        sigma = 0.30 * (1.0 - frac) + 0.02 * frac

        jitter = rng.normal(scale=sigma, size=n)
        noisy = base * (1.0 + jitter)
        noisy = np.maximum(noisy, 0.0)
        noisy = project_to_simplex(noisy)

        remaining = deadline - time.time()
        if remaining <= 1.0:
            break

        max_iter_sub = int(min(100_000, remaining * 25))
        refined = subgradient_descent_exact(
            noisy,
            deadline,
            max_iter=max_iter_sub,
            step0=0.20,
        )
        if time.time() < deadline - 1.0:
            max_iter_eg = int(min(80_000, (deadline - time.time()) * 25))
            refined = eg_refine(
                refined,
                deadline,
                max_iter=max_iter_eg,
                eta0=0.20,
            )

        fast_sc = compute_score(refined)
        if fast_sc < best_fast - 1e-6:
            try:
                val = evaluate_sequence(refined.tolist())
                print(f"Jitter restart official score = {val:.6f}")
                if val < best_val:
                    best_val = val
                    best_seq = refined.tolist()
                    best_fast = fast_sc
                    print("*** Updated global best from jitter restart ***")
            except Exception as e:
                print(f"    Evaluation error: {e}")

    # ------------------------------------------------------------------
    # 5.  Final polish on the best vector (if time permits)
    # ------------------------------------------------------------------
    if best_seq and time.time() < deadline - 2.0:
        print("\nFinal global polish …")
        a = np.array(best_seq, dtype=float)
        remaining = deadline - time.time()
        max_iter_sub = int(min(200_000, remaining * 30))
        a = subgradient_descent_exact(a, deadline, max_iter=max_iter_sub, step0=0.12)

        if time.time() < deadline - 1.0:
            remaining = deadline - time.time()
            max_iter_eg = int(min(120_000, remaining * 30))
            a = eg_refine(a, deadline, max_iter=max_iter_eg, eta0=0.12)

        try:
            final_val = evaluate_sequence(a.tolist())
            print(f"Final polished official score = {final_val:.6f}")
            if final_val < best_val:
                best_val = final_val
                best_seq = a.tolist()
                print("*** Updated global best after final polish ***")
        except Exception:
            pass

    # ------------------------------------------------------------------
    # 6.  Guard – guarantee a non‑empty return value
    # ------------------------------------------------------------------
    if not best_seq:
        fallback_n = 256
        best_seq = [1.0 / fallback_n] * fallback_n
        best_val = evaluate_sequence(best_seq)
        print("Fallback to uniform vector used.")

    print(f"\nSearch finished. Best official score = {best_val:.6f}")
    return best_seq
