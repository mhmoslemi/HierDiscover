def flatten_gradient(a):
    n = a.shape[0]
    conv = convolution_fft(a, a)
    avg = 1.0 / (2 * n - 1)
    diff = conv - avg
    loss = np.sum(diff ** 2)
    # compute gradient
    diff_rev = diff[::-1]
    conv_rev = convolution_fft(a, diff_rev)
    L = 2 * n - 1
    grad = 4.0 * conv_rev[L-1 - np.arange(n)]   # factor 4
    return grad, loss

2️⃣  For the best sampled vector of each length we run a *two‑stage* local
    optimisation:
       – **Flattening stage** – minimise the L₂ distance of the self‑convolution
         to the (theoretical) flat level 1/(2n‑1).  This is performed with an
         Adam optimiser on the exact L₂ loss; the gradient is obtained via a
         single extra FFT per iteration.
       – **Max‑stage** – minimise the true max‑convolution using a smooth
         log‑sum‑exp surrogate (annealed temperature `λ`) and Adam, then a
         projected sub‑gradient descent with back‑tracking, and finally a cheap
         stochastic Gaussian walk.

3️⃣  The overall best vector is polished once more with the same three
    stages (longer Adam runs, a higher temperature `λ≈800`) before the final
    call to the official `evaluate_sequence`.

All iterates are kept on the probability simplex (non‑negative, sum = 1) by
projecting after each update.  The code is deliberately time‑driven: every
loop checks the remaining budget and stops early if needed, guaranteeing a
non‑empty return (a uniform 256‑vector) even in the worst case.
"""

import time
import numpy as np
from typing import List, Tuple

# ----------------------------------------------------------------------
# Linear‑algebra utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Return the smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Linear convolution via real FFT (returns length len(a)+len(b)-1)."""
    m = a.size + b.size - 1
    n_fft = next_pow2(m)
    A = np.fft.rfft(a, n=n_fft)
    B = np.fft.rfft(b, n=n_fft)
    C = A * B
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


def max_convolution(a: np.ndarray) -> float:
    """Maximum entry of the self‑convolution of a (full length)."""
    return float(np.max(convolution_fft(a, a)))


def compute_score(a: np.ndarray) -> float:
    """
    Fast (surrogate) score: 2·n·max(conv(a,a)).
    Works for vectors already normalised to Σa = 1.
    """
    n = a.shape[0]
    return 2.0 * n * max_convolution(a)


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of v onto the probability simplex {x≥0, Σx=1}.
    Duchi et al. (2008) implementation – O(n log n).
    """
    n = v.shape[0]
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / n)

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho_arr = u - cssv / np.arange(1, n + 1)
    rho = np.where(rho_arr > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Shape generators (beta families and mixtures)
# ----------------------------------------------------------------------


def generate_beta_shape(n: int, alpha: float, beta: float, eps: float) -> np.ndarray:
    """Symmetric beta‑family density on {0,…,n‑1} (unnormalised)."""
    i = np.arange(n, dtype=float)
    left = (i + eps) ** (alpha - 1.0)
    right = (n - 1 - i + eps) ** (beta - 1.0)
    base = left * right
    total = base.sum()
    if total <= 0.0 or not np.isfinite(total):
        return np.full(n, 1.0 / n)
    return base / total


def generate_beta_mixture(
    n: int,
    alpha1: float,
    beta1: float,
    eps1: float,
    alpha2: float,
    beta2: float,
    eps2: float,
    w: float,
) -> np.ndarray:
    """Convex combination of two beta‑family densities (renormalised)."""
    a1 = generate_beta_shape(n, alpha1, beta1, eps1)
    a2 = generate_beta_shape(n, alpha2, beta2, eps2)
    mix = w * a1 + (1.0 - w) * a2
    mix_sum = mix.sum()
    if mix_sum == 0.0 or not np.isfinite(mix_sum):
        return np.full(n, 1.0 / n)
    return mix / mix_sum


def arcsine_initial(n: int) -> np.ndarray:
    """Classic arcsine (U‑shaped) density."""
    i = np.arange(n, dtype=float)
    eps = 0.25
    vals = 1.0 / np.sqrt((i + 0.5) * (n - 0.5 - i) + eps)
    vals /= vals.sum()
    return vals


# ----------------------------------------------------------------------
# Gradient of the *flattening* loss (L2 distance to a flat convolution)
# ----------------------------------------------------------------------


def flatten_gradient(a: np.ndarray) -> Tuple[np.ndarray, float]:
    """
    Gradient of ‖conv(a,a) – avg‖₂² where avg = 1/(2n‑1) and Σa = 1.
    Returns (grad, loss).
    """
    n = a.shape[0]
    conv = convolution_fft(a, a)                     # length 2n‑1
    avg = 1.0 / (2 * n - 1)
    diff = conv - avg
    loss = float(np.sum(diff ** 2))

    # Gradient: 4 * (a ⊛ diff_rev)  (⊛ = convolution)
    diff_rev = diff[::-1]
    conv_rev = convolution_fft(a, diff_rev)          # length 3n‑2
    L = 2 * n - 1
    grad = 4.0 * conv_rev[L - 1 - np.arange(n)]
    return grad, loss


def adam_flatten(
    initial_a: np.ndarray,
    time_deadline: float,
    max_iter: int = 150_000,
    lr0: float = 0.25,
) -> Tuple[np.ndarray, float]:
    """
    Adam optimiser on the *flattening* L₂ loss.
    Returns the best vector (projected) and its fast score.
    """
    a = project_to_simplex(initial_a.astype(float).copy())
    best_a = a.copy()
    best_sc = compute_score(a)

    beta1, beta2 = 0.9, 0.999
    eps_adam = 1e-8
    m = np.zeros_like(a)
    v = np.zeros_like(a)

    for it in range(1, max_iter + 1):
        if time.time() >= time_deadline:
            break

        grad, loss = flatten_gradient(a)

        # Adam update (learning rate decays like 1/√it)
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)

        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        step = (lr0 / np.sqrt(it)) * m_hat / (np.sqrt(v_hat) + eps_adam)
        a_new = a - step
        a_new = project_to_simplex(a_new)

        cur_sc = compute_score(a_new)
        if cur_sc < best_sc:
            best_sc = cur_sc
            best_a = a_new.copy()

        a = a_new

        # occasional progress report
        if it % 20000 == 0:
            print(
                f"    Adam‑flatten iter={it:,d} loss={loss:.6e} cur_sc={cur_sc:.6f} best_sc={best_sc:.6f}"
            )
    return best_a, best_sc


# ----------------------------------------------------------------------
# Smooth surrogate of the max (log‑sum‑exp) and its gradient
# ----------------------------------------------------------------------


def smooth_gradient(a: np.ndarray, lam: float) -> Tuple[np.ndarray, float]:
    """
    Gradient of the log‑sum‑exp surrogate of max(conv(a,a)).
    Returns (gradient, surrogate value B) where
        B = max(conv) + (1/lam)·log Σ exp(lam·(conv‑max)).
    """
    n = a.shape[0]
    conv = convolution_fft(a, a)          # length 2n‑1
    max_c = np.max(conv)

    # stable log‑sum‑exp
    exp_term = np.exp(lam * (conv - max_c))
    sum_exp = np.sum(exp_term)
    w = exp_term / sum_exp                # soft‑max weights on convolution entries
    B = max_c + np.log(sum_exp) / lam

    # weighted sum of sub‑gradients:
    # ∂/∂a Σ_k w_k (a*a)_k = 2 * (a ⊛ w_rev)
    w_rev = w[::-1]
    conv_rev = convolution_fft(a, w_rev)      # length 3n‑2
    L = 2 * n - 1
    weighted_grad = 2.0 * conv_rev[L - 1 - np.arange(n)]

    s = np.sum(a)
    term1 = (2.0 * n / (s ** 2)) * weighted_grad
    term2 = -(4.0 * n * B / (s ** 3)) * np.ones_like(a)
    grad = term1 + term2
    return grad, B


def adam_max(
    initial_a: np.ndarray,
    time_deadline: float,
    max_iter: int = 200_000,
    lr0: float = 0.12,
    lam_start: float = 1.0,
    lam_end: float = 500.0,
    noise_every: int = 1000,
    noise_scale: float = 4e-5,
) -> Tuple[np.ndarray, float]:
    """
    Adam optimiser on the smoothed max‑convolution surrogate.
    Returns the best vector (projected) and its fast score.
    """
    a = project_to_simplex(initial_a.astype(float).copy())
    best_a = a.copy()
    best_sc = compute_score(a)

    beta1, beta2 = 0.9, 0.999
    eps_adam = 1e-8
    m = np.zeros_like(a)
    v = np.zeros_like(a)

    n = a.shape[0]

    for it in range(1, max_iter + 1):
        if time.time() >= time_deadline:
            break

        # quadratic temperature schedule
        t = it / max_iter
        lam = lam_start + (lam_end - lam_start) * (t ** 0.5)

        grad, _ = smooth_gradient(a, lam)

        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)

        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        step = lr0 * m_hat / (np.sqrt(v_hat) + eps_adam)
        a_new = a - step

        if noise_every and (it % noise_every == 0):
            a_new += np.random.randn(n) * noise_scale

        a_new = project_to_simplex(a_new)

        cur_sc = compute_score(a_new)
        if cur_sc < best_sc:
            best_sc = cur_sc
            best_a = a_new.copy()

        a = a_new

        if it % 5000 == 0:
            print(
                f"    Adam‑max iter={it:,d} cur={cur_sc:.6f} best={best_sc:.6f} lam={lam:.1f}"
            )
    return best_a, best_sc


# ----------------------------------------------------------------------
# Exact sub‑gradient descent on the true max; back‑tracking line‑search
# ----------------------------------------------------------------------


def subgradient_optimize(
    initial_a: np.ndarray,
    time_deadline: float,
    step0: float = 0.3,
    max_iter: int = 200_000,
) -> Tuple[np.ndarray, float]:
    """
    Projected sub‑gradient descent on the *true* max‑convolution.
    Simple back‑tracking line‑search guarantees non‑increase.
    """
    a = initial_a.copy()
    best_a = a.copy()
    best_sc = compute_score(a)
    n = a.shape[0]

    for it in range(1, max_iter + 1):
        if time.time() >= time_deadline:
            break

        conv = convolution_fft(a, a)
        max_val = float(np.max(conv))
        idx = int(np.argmax(conv))

        # gradient of the active quadratic term
        i = np.arange(n)
        j = idx - i
        mask = (j >= 0) & (j < n)
        grad = np.zeros_like(a)
        grad[mask] = 2.0 * a[j[mask]]

        # back‑tracking line‑search
        step = step0 / np.sqrt(it)
        for _ in range(12):
            a_new = a - step * grad
            a_new = project_to_simplex(a_new)
            new_val = max_convolution(a_new)
            if new_val < max_val - 1e-14:
                break
            step *= 0.5

        cur_sc = compute_score(a_new)
        if cur_sc < best_sc:
            best_sc = cur_sc
            best_a = a_new.copy()

        a = a_new

        if it % 40000 == 0:
            print(
                f"    Sub‑grad iter={it:,d} cur={cur_sc:.6f} best={best_sc:.6f}"
            )
    return best_a, best_sc


# ----------------------------------------------------------------------
# Stochastic local walk (cheap Gaussian perturbations)
# ----------------------------------------------------------------------


def random_local_search(
    initial_a: np.ndarray,
    time_deadline: float,
    init_step: float = 0.015,
    max_iter: int = 250_000,
) -> Tuple[np.ndarray, float]:
    """
    Gaussian walk on the simplex with a tiny uphill‑move probability.
    """
    a = initial_a.copy()
    best_a = a.copy()
    best_sc = compute_score(a)
    n = a.shape[0]
    rng = np.random.default_rng()

    for it in range(1, max_iter + 1):
        if time.time() >= time_deadline:
            break

        step = init_step / np.sqrt(it)
        noise = rng.normal(scale=step, size=n)
        a_proposed = a + noise
        a_proposed = project_to_simplex(a_proposed)

        cur_sc = compute_score(a_proposed)

        # accept if improvement, otherwise with tiny probability
        if cur_sc < best_sc or (rng.random() < 1e-4 and cur_sc < 10 * best_sc):
            a = a_proposed
            if cur_sc < best_sc:
                best_sc = cur_sc
                best_a = a_proposed.copy()

        if it % 50000 == 0:
            print(
                f"    Random walk iter={it:,d} cur={cur_sc:.6f} best={best_sc:.6f}"
            )
    return best_a, best_sc


# ----------------------------------------------------------------------
# Main entry point required by the competition harness
# ----------------------------------------------------------------------


def propose_candidate(
    seed: int = 42,
    budget_s: int = 1000,
    **kwargs,
) -> List[float]:
    """
    Search for a non‑negative step sequence with a low official score.
    The routine mixes extensive beta‑shape sampling, a dedicated flattening
    stage, a smooth‑max Adam run (high temperature), exact sub‑gradient
    descent and a final stochastic walk.  All phases respect the global
    time budget.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 1.0   # small safety margin

    best_seq: List[float] = []
    best_val: float = float("inf")

    # ------------------------------------------------------------------
    # 0.  Use previously known construction if present
    # ------------------------------------------------------------------
    try:
        fallback = [float(x) for x in height_sequence_1]  # type: ignore
        val = evaluate_sequence(fallback)
        print(f"Fallback (height_sequence_1) official score = {val:.6f}")
        if val < best_val:
            best_val = val
            best_seq = fallback
    except Exception:
        pass

    # ------------------------------------------------------------------
    # 1.  Exploratory sampling on a few lengths
    # ------------------------------------------------------------------
    lengths = [1024, 2048, 4096]

    for n in lengths:
        if time.time() >= deadline - 30.0:
            break
        print(f"\n=== Sampling length n = {n} ===")

        # ---- deterministic starters -------------------------------------------------
        local_best = None
        local_best_sc = float("inf")
        starters = [
            np.full(n, 1.0 / n),    # uniform
            arcsine_initial(n),    # classical arcsine
        ]
        # tiny grid on symmetric beta exponent (α = β)
        for alpha in np.linspace(0.15, 0.85, 15):
            cand = generate_beta_shape(n, alpha, alpha, eps=1e-6)
            starters.append(cand)

        for st in starters:
            sc = compute_score(st)
            if sc < local_best_sc:
                local_best_sc = sc
                local_best = st.copy()

        # ---- random beta‑family shapes (including mixtures) -------------------------
        max_samples = int(6000 * (budget_s / 1000.0))
        for _ in range(max_samples):
            if time.time() >= deadline - 30.0:
                break
            if rng.random() < 0.35:          # mixture branch
                w = rng.random()
                alpha1 = rng.uniform(0.02, 3.5)
                beta1 = rng.uniform(0.02, 3.5)
                eps1 = rng.uniform(1e-8, max(0.01, 0.02 * n))
                alpha2 = rng.uniform(0.02, 3.5)
                beta2 = rng.uniform(0.02, 3.5)
                eps2 = rng.uniform(1e-8, max(0.01, 0.02 * n))
                cand = generate_beta_mixture(
                    n, alpha1, beta1, eps1, alpha2, beta2, eps2, w
                )
            else:                            # plain beta
                alpha = rng.uniform(0.02, 3.5)
                beta = rng.uniform(0.02, 3.5)
                eps = rng.uniform(1e-8, max(0.01, 0.02 * n))
                cand = generate_beta_shape(n, alpha, beta, eps)

            sc = compute_score(cand)
            if sc < local_best_sc:
                local_best_sc = sc
                local_best = cand

        # ------------------------------------------------------------------
        # 2.  Polishing of the locally best candidate (if any)
        # ------------------------------------------------------------------
        if local_best is None:
            continue

        # allocate remaining time proportionally (but always keep a few seconds)
        remaining = deadline - time.time()
        if remaining < 5.0:
            break

        # ----- stage A – flattening (L2 loss) ---------------------------------
        max_iter_flat = int(min(120_000, remaining * 22))
        print(f"  Flattening stage (max_iter={max_iter_flat}) …")
        a_flat, _ = adam_flatten(
            initial_a=local_best,
            time_deadline=deadline,
            max_iter=max_iter_flat,
            lr0=0.30,
        )
        # ----- stage B – smooth‑max Adam ------------------------------------
        if time.time() < deadline - 5.0:
            remaining = deadline - time.time()
            max_iter_adam = int(min(250_000, remaining * 28))
            print(f"  Smooth‑max Adam (max_iter={max_iter_adam}) …")
            a_max, _ = adam_max(
                initial_a=a_flat,
                time_deadline=deadline,
                max_iter=max_iter_adam,
                lr0=0.12,
                lam_start=1.0,
                lam_end=800.0,
                noise_every=1300,
                noise_scale=5e-5,
            )
        else:
            a_max = a_flat

        # ----- stage C – exact sub‑gradient descent -------------------------
        if time.time() < deadline - 5.0:
            remaining = deadline - time.time()
            max_iter_sub = int(min(150_000, remaining * 30))
            print(f"  Sub‑gradient descent (max_iter={max_iter_sub}) …")
            a_sub, _ = subgradient_optimize(
                a_max,
                time_deadline=deadline,
                step0=0.28,
                max_iter=max_iter_sub,
            )
        else:
            a_sub = a_max

        # ----- stage D – stochastic local walk --------------------------------
        if time.time() < deadline - 3.0:
            remaining = deadline - time.time()
            max_iter_loc = int(min(200_000, remaining * 35))
            print(f"  Random walk polishing (max_iter={max_iter_loc}) …")
            a_final, _ = random_local_search(
                a_sub,
                time_deadline=deadline,
                init_step=0.017,
                max_iter=max_iter_loc,
            )
        else:
            a_final = a_sub

        # ----- official evaluation -------------------------------------------
        val = evaluate_sequence(a_final.tolist())
        print(f"  → official score for length {n} = {val:.6f}")
        if val < best_val:
            best_val = val
            best_seq = a_final.tolist()
            print(f"  *** New global best (len={n}) = {best_val:.6f} ***")

    # ------------------------------------------------------------------
    # 3.  Global polishing of the overall best vector (if time permits)
    # ------------------------------------------------------------------
    if best_seq and time.time() < deadline - 5.0:
        print("\n=== Global polishing of the best sequence ===")
        a = np.array(best_seq, dtype=float)

        # Stage A – long flattening run
        remaining = deadline - time.time()
        max_iter_flat = int(min(180_000, remaining * 20))
        print(f"  Global flattening (max_iter={max_iter_flat}) …")
        a, _ = adam_flatten(
            initial_a=a,
            time_deadline=deadline,
            max_iter=max_iter_flat,
            lr0=0.35,
        )

        # Stage B – high‑temperature Adam on the max‑surrogate
        if time.time() < deadline - 4.0:
            remaining = deadline - time.time()
            max_iter_adam = int(min(350_000, remaining * 28))
            print(f"  Global Adam‑max (max_iter={max_iter_adam}) …")
            a, _ = adam_max(
                initial_a=a,
                time_deadline=deadline,
                max_iter=max_iter_adam,
                lr0=0.10,
                lam_start=1.0,
                lam_end=1000.0,
                noise_every=1500,
                noise_scale=4e-5,
            )

        # Stage C – sub‑gradient descent
        if time.time() < deadline - 4.0:
            remaining = deadline - time.time()
            max_iter_sub = int(min(180_000, remaining * 30))
            print(f"  Global sub‑gradient (max_iter={max_iter_sub}) …")
            a, _ = subgradient_optimize(
                a,
                time_deadline=deadline,
                step0=0.22,
                max_iter=max_iter_sub,
            )

        # Stage D – final random walk
        if time.time() < deadline - 2.0:
            remaining = deadline - time.time()
            max_iter_loc = int(min(250_000, remaining * 40))
            print(f"  Global random walk (max_iter={max_iter_loc}) …")
            a, _ = random_local_search(
                a,
                time_deadline=deadline,
                init_step=0.012,
                max_iter=max_iter_loc,
            )

        final_val = evaluate_sequence(a.tolist())
        print(f"Final polished official score = {final_val:.6f}")
        if final_val < best_val:
            best_val = final_val
            best_seq = a.tolist()
            print("*** Updated global best after final polish ***")

    # ------------------------------------------------------------------
    # 4.  Guarantee a non‑empty return
    # ------------------------------------------------------------------
    if not best_seq:
        best_seq = [1.0] * 256
        best_val = evaluate_sequence(best_seq)
        print("Fallback to uniform 256‑vector.", best_val)

    print(f"\nSearch finished. Best reported official score = {best_val:.6f}")
    return best_seq
