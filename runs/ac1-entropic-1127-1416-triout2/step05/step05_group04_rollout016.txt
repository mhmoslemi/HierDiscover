def adam_optimize_l2(initial_a, deadline, max_iter=200000, lr0=0.2):
    a=project_to_simplex(initial_a.copy())
    best_a = a.copy()
    best_sc = compute_score(a)
    beta1,beta2=0.9,0.999
    eps=1e-8
    m=np.zeros_like(a)
    v=np.zeros_like(a)
    avg = 1.0/(2*len(a)-1)
    for it in range(1, max_iter+1):
        if time.time() >= deadline: break
        # compute conv and gradient
        conv = convolution_fft(a,a)
        diff = conv - avg
        w_rev = diff[::-1]
        conv_rev = convolution_fft(a, w_rev)
        L = len(conv)
        grad = 4.0 * conv_rev[L-1 - np.arange(len(a))]
        # Adam update
        m = beta1*m + (1-beta1)*grad
        v = beta2*v + (1-beta2)*(grad*grad)
        m_hat = m / (1-beta1**it)
        v_hat = v / (1-beta2**it)
        step = lr0 * m_hat / (np.sqrt(v_hat) + eps)
        a = a - step
        a = project_to_simplex(a)
        cur_sc = compute_score(a)
        if cur_sc < best_sc:
            best_sc = cur_sc
            best_a = a.copy()
        if it % 5000 == 0:
            print(f' AdamL2 it={it} cur={cur_sc:.6f} best={best_sc:.6f}')
    return best_a, best_sc

   scale‑invariant, so larger n gives more degrees of freedom without
   changing the target value.

2. Generate a diverse collection of parametric seeds (arcsine, symmetric
   Beta, power‑law, mixtures and a few random Dirichlet vectors).  The
   cheap surrogate `compute_score` (2·n·max(conv) after simplex
   projection) is used to keep the *top_k* most promising seeds.

3. Starting from the best seed, run a long Adam optimisation on the
   **L2‑flattening loss**
        L(a) = Σ (conv(a,a)_k – μ)^2 ,   μ = 1/(2·n‑1)
   The gradient of L is
        ∇L = 4·a * (conv(a,a) – μ)ᵀ   (convolution form).
   Adam’s adaptive steps and the projection onto the probability
   simplex keep Σ a_i = 1 while aggressively equalising the
   convolution values.

4. A final sub‑gradient descent on the true objective
        f(a) = 2·n·max(conv(a,a))
   removes any remaining peaks.

5. The best vector found is evaluated with the official
   `evaluate_sequence`.  If time permits a short simulated‑annealing
   polishing is performed.

All heavy work uses FFT‑based convolution (O(n log n)) so that even
hundreds of thousands of iterations finish well inside the 1000 s
budget.
"""

import time
import numpy as np
from typing import List, Tuple

# ----------------------------------------------------------------------
# Low‑level utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Linear convolution via real FFT (length = len(a)+len(b)-1)."""
    m = len(a) + len(b) - 1
    n_fft = next_pow2(m)
    A = np.fft.rfft(a, n=n_fft)
    B = np.fft.rfft(b, n=n_fft)
    C = A * B
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


def compute_score(a: np.ndarray) -> float:
    """
    Fast surrogate of the official scoring function.
    The vector is assumed to already sum to 1, therefore the denominator
    disappears and the score equals 2·n·max(conv(a,a)).
    """
    n = a.size
    conv = convolution_fft(a, a)
    return 2.0 * n * float(np.max(conv))


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection onto the probability simplex {x ≥ 0, Σx = 1}.
    Duchi et al., 2008.
    """
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / v.size)

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho = np.where(u - cssv / np.arange(1, v.size + 1) > 0)[0]
    if len(rho) == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Shape factories (all vectors are already normalised to Σ = 1)
# ----------------------------------------------------------------------


def arcsine_initial(n: int) -> np.ndarray:
    """Classic arcsine shape: a_i ∝ 1/√((i+½)*(n‑½‑i))."""
    i = np.arange(n, dtype=float)
    eps = 0.25
    vals = 1.0 / np.sqrt((i + 0.5) * (n - 0.5 - i) + eps)
    vals /= vals.sum()
    return vals


def generate_beta_shape(
    n: int, alpha: float, beta: float, eps: float = 1e-8
) -> np.ndarray:
    """
    Symmetric (or not) Beta family:
        a_i ∝ (i+ε)^{α‑1}·(n‑1‑i+ε)^{β‑1}
    """
    i = np.arange(n, dtype=float)
    left = (i + eps) ** (alpha - 1.0)
    right = (n - 1 - i + eps) ** (beta - 1.0)
    base = left * right
    total = base.sum()
    if not np.isfinite(total) or total <= 0.0:
        return np.full(n, 1.0 / n)
    return base / total


def generate_power_shape(
    n: int, p: float, eps: float = 0.0
) -> np.ndarray:
    """
    Generalised arcsine:
        a_i ∝ ((i+½)*(n‑½‑i)+ε)^{-p},  p>0.
    """
    i = np.arange(n, dtype=float)
    denom = ((i + 0.5) * (n - 0.5 - i) + eps) ** p
    vals = 1.0 / denom
    total = vals.sum()
    if not np.isfinite(total) or total <= 0.0:
        return np.full(n, 1.0 / n)
    return vals / total


def generate_triangular_shape(n: int, peak: int) -> np.ndarray:
    """Symmetric triangular bump with apex at `peak`."""
    left = np.arange(peak + 1, dtype=float)
    right = np.arange(n - peak - 1, dtype=float)[::-1]
    seq = np.concatenate([left, right])
    seq /= seq.sum()
    return seq


def generate_mixture_shape(n: int, weight_arc: float) -> np.ndarray:
    """Mixture weight_arc·arcsine + (1‑weight_arc)·uniform."""
    arc = arcsine_initial(n)
    uni = np.full(n, 1.0 / n)
    mix = weight_arc * arc + (1.0 - weight_arc) * uni
    mix /= mix.sum()
    return mix


# ----------------------------------------------------------------------
# Optimisers
# ----------------------------------------------------------------------


def adam_optimize_l2(
    initial_a: np.ndarray,
    deadline: float,
    max_iter: int = 500_000,
    lr0: float = 0.2,
) -> Tuple[np.ndarray, float]:
    """
    Adam optimiser on the L2 flattening loss
        L(a) = Σ_k (conv(a,a)_k – μ)^2,
    where μ = 1/(2·n‑1).  The method returns the best vector seen.
    """
    a = project_to_simplex(initial_a.astype(float).copy())
    best_a = a.copy()
    best_sc = compute_score(a)

    n = a.size
    L_len = 2 * n - 1
    avg = 1.0 / L_len

    beta1, beta2 = 0.9, 0.999
    eps = 1e-8
    m = np.zeros_like(a)
    v = np.zeros_like(a)

    for it in range(1, max_iter + 1):
        if time.time() >= deadline:
            break

        conv = convolution_fft(a, a)          # length = L_len
        diff = conv - avg                     # centred vector
        w_rev = diff[::-1]
        conv_rev = convolution_fft(a, w_rev)  # length = L_len
        grad = 4.0 * conv_rev[L_len - 1 - np.arange(n)]

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)

        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        step = lr0 * m_hat / (np.sqrt(v_hat) + eps)
        a = a - step
        a = project_to_simplex(a)

        cur_sc = compute_score(a)
        if cur_sc < best_sc:
            best_sc = cur_sc
            best_a = a.copy()

        if it % 10_000 == 0:
            print(f"    AdamL2 it={it:,d} cur={cur_sc:.6f} best={best_sc:.6f}")

    return best_a, best_sc


def subgradient_descent_exact(
    initial_a: np.ndarray,
    deadline: float,
    max_iter: int = 200_000,
    step0: float = 0.04,
) -> Tuple[np.ndarray, float]:
    """
    Sub‑gradient descent on the true objective
        f(a)=2·n·max(conv(a,a)).
    Returns the best vector encountered.
    """
    a = project_to_simplex(initial_a.copy())
    best_a = a.copy()
    best_sc = compute_score(a)

    n = a.size
    L = 2 * n - 1

    for it in range(1, max_iter + 1):
        if time.time() >= deadline:
            break

        conv = convolution_fft(a, a)
        max_val = np.max(conv)

        # mask of entries that are (numerically) maximal
        tol = max_val * 1e-12 + 1e-14
        mask = (np.abs(conv - max_val) <= tol).astype(float)

        # weighted sub‑gradient: 2·∑_k mask_k·shift(a,k)
        mask_rev = mask[::-1]
        conv_rev = convolution_fft(a, mask_rev)
        weighted_grad = 2.0 * conv_rev[L - 1 - np.arange(n)]

        eta = step0 / np.sqrt(it)
        a = a - eta * weighted_grad
        a = project_to_simplex(a)

        cur_sc = compute_score(a)
        if cur_sc < best_sc:
            best_sc = cur_sc
            best_a = a.copy()

        if it % 10_000 == 0:
            print(f"    SubGrad it={it:,d} cur={cur_sc:.6f} best={best_sc:.6f}")

    return best_a, best_sc


def simulated_annealing(
    a: np.ndarray,
    rng: np.random.Generator,
    deadline: float,
    max_iters: int = 150_000,
    init_temp: float = 0.02,
    decay: float = 0.99995,
) -> np.ndarray:
    """
    Simple Metropolis search using the fast surrogate score.
    Moves a small fraction of mass from one coordinate to another.
    """
    cur = a.copy()
    cur_sc = compute_score(cur)
    best = cur.copy()
    best_sc = cur_sc
    n = len(cur)
    temp = init_temp
    it = 0

    while it < max_iters and time.time() < deadline:
        it += 1
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue

        delta = cur[i] * 0.2 * rng.random()
        if delta <= 1e-15 or cur[i] - delta < 0:
            continue

        cand = cur.copy()
        cand[i] -= delta
        cand[j] += delta

        cand_sc = compute_score(cand)

        if cand_sc < cur_sc or rng.random() < np.exp(-(cand_sc - cur_sc) / max(temp, 1e-12)):
            cur = cand
            cur_sc = cand_sc
            if cur_sc < best_sc:
                best = cur.copy()
                best_sc = cur_sc

        temp = max(temp * decay, 1e-8)

        if it % 20_000 == 0:
            print(f"    SA it={it:,d} cur={cur_sc:.6f} best={best_sc:.6f}")

    return best


# ----------------------------------------------------------------------
# Main entry point required by the competition harness
# ----------------------------------------------------------------------


def propose_candidate(
    seed: int = 123,
    budget_s: int = 1000,
) -> List[float]:
    """
    Search for a low‑score step sequence.
    The routine builds a set of parametric seeds, refines the best one
    with a long Adam optimisation on the L2 flattening loss, follows
    up with exact sub‑gradient descent, optionally polishes with SA,
    and finally returns the best vector found.
    """
    rng = np.random.default_rng(seed)
    start = time.time()
    deadline = start + budget_s - 2.0   # small safety margin

    # ------------------------------------------------------------------
    # 0.  Fallback from any previously stored sequence
    # ------------------------------------------------------------------
    best_seq: List[float] = []
    best_val: float = float("inf")
    if "height_sequence_1" in globals():
        try:
            fallback = [float(x) for x in globals()["height_sequence_1"]]
            val = evaluate_sequence(fallback)
            print(f"Fallback (height_sequence_1) score = {val:.6f}")
            if val < best_val:
                best_val = val
                best_seq = fallback
        except Exception:
            pass

    # ------------------------------------------------------------------
    # 1.  Build a rich pool of seeds (all normalised to Σ = 1)
    # ------------------------------------------------------------------
    n = 4096                          # power‑of‑two length works well with FFT
    top_k = 5
    best_pool: List[Tuple[float, np.ndarray]] = []   # (score, vector)

    def maybe_insert(vec: np.ndarray):
        sc = compute_score(vec)
        if len(best_pool) < top_k:
            best_pool.append((sc, vec))
        else:
            # replace worst if better
            worst_idx = max(range(len(best_pool)),
                            key=lambda i: best_pool[i][0])
            if sc < best_pool[worst_idx][0]:
                best_pool[worst_idx] = (sc, vec)

    # deterministic starters
    maybe_insert(np.full(n, 1.0 / n))        # uniform
    maybe_insert(arcsine_initial(n))        # arcsine

    # ----- systematic beta‑grid -------------------------------------------------
    alist = np.concatenate(
        [np.linspace(0.15, 0.5, 15), np.linspace(0.6, 2.0, 15)]
    )
    for a in alist:
        cand = generate_beta_shape(n, a, a, eps=1e-8)
        maybe_insert(cand)

    # ----- power‑law family ----------------------------------------------------
    for p in np.linspace(0.2, 5.0, 25):
        cand = generate_power_shape(n, p, eps=0.0)
        maybe_insert(cand)

    # ----- mixture of arcsine & uniform ----------------------------------------
    for w in np.linspace(0.0, 1.0, 11):
        cand = generate_mixture_shape(n, w)
        maybe_insert(cand)

    # ----- few random Dirichlet draws -----------------------------------------
    for _ in range(30):
        alphas_dir = rng.exponential(scale=1.0, size=n) + 0.01
        cand = alphas_dir / alphas_dir.sum()
        maybe_insert(cand)

    # ------------------------------------------------------------------
    # 2.  Pick the best seed and launch a deep optimisation
    # ------------------------------------------------------------------
    best_pool.sort(key=lambda x: x[0])
    seed_vec = best_pool[0][1]
    print(f"Initial best surrogate score = {best_pool[0][0]:.6f}")

    # 2A – long Adam optimisation on the L2 flattening loss
    remaining = deadline - time.time()
    max_iter_adam = int(min(800_000, remaining * 45))
    print(f"Running Adam L2 optimisation, max_iter={max_iter_adam}")
    a_opt, _ = adam_optimize_l2(
        seed_vec,
        deadline,
        max_iter=max_iter_adam,
        lr0=0.2,
    )

    # 2B – exact sub‑gradient descent for final peak shaving
    if time.time() < deadline - 5.0:
        remaining = deadline - time.time()
        max_iter_sub = int(min(300_000, remaining * 40))
        print(f"Running sub‑gradient descent, max_iter={max_iter_sub}")
        a_opt, _ = subgradient_descent_exact(
            a_opt,
            deadline,
            max_iter=max_iter_sub,
            step0=0.04,
        )

    # 2C – optional simulated‑annealing polishing (cheap)
    if time.time() < deadline - 2.0:
        remaining = deadline - time.time()
        max_iter_sa = int(min(150_000, remaining * 30))
        print(f"Running SA polishing, max_iters={max_iter_sa}")
        a_opt = simulated_annealing(
            a_opt,
            rng,
            deadline,
            max_iters=max_iter_sa,
            init_temp=0.015,
            decay=0.99995,
        )

    # ------------------------------------------------------------------
    # 3.  Evaluate officially
    # ------------------------------------------------------------------
    try:
        val = evaluate_sequence(a_opt.tolist())
        print(f"Official score of refined candidate = {val:.6f}")
        if val < best_val:
            best_val = val
            best_seq = a_opt.tolist()
    except Exception as e:
        print("Official evaluation failed:", e)

    # ------------------------------------------------------------------
    # 4.  Global polish (run a short extra Adam if time permits)
    # ------------------------------------------------------------------
    if best_seq and time.time() < deadline - 1.0:
        a = np.array(best_seq, dtype=float)
        remaining = deadline - time.time()
        max_iter_adam2 = int(min(200_000, remaining * 30))
        print(f"Final short Adam L2 polish, max_iter={max_iter_adam2}")
        a, _ = adam_optimize_l2(
            a,
            deadline,
            max_iter=max_iter_adam2,
            lr0=0.15,
        )
        final_val = evaluate_sequence(a.tolist())
        print(f"Final polished official score = {final_val:.6f}")
        if final_val < best_val:
            best_val = final_val
            best_seq = a.tolist()
            print("*** Updated global best after final polish ***")

    # ------------------------------------------------------------------
    # 5.  Guard – ensure a non‑empty return
    # ------------------------------------------------------------------
    if not best_seq:
        best_seq = [1.0] * n
        best_val = evaluate_sequence(best_seq)

    print(f"\nSearch finished. Best reported score = {best_val:.6f}")
    return best_seq
