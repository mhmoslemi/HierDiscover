def coordinate_descent_optimize(initial_a, deadline, step0=0.2, max_iter=500_000):

- a = initial_a.copy()

- best_a = a.copy(); best_score = compute_score(a)

- step_factor = step0

- for it in range(1, max_iter+1):

   if time.time() >= deadline: break

   # compute conv, max, mask, gradient

   conv = convolution_fft(a, a)

   max_val = np.max(conv)

   # tolerance for active set

   tol = max_val * 1e-12 + 1e-14

   mask = (np.abs(conv - max_val) <= tol).astype(float) # shape L

   # gradient

   mask_rev = mask[::-1]

   conv_rev = convolution_fft(a, mask_rev)

   L = 2 * len(a) - 1

   grad = 2.0 * conv_rev[L - 1 - np.arange(len(a))]

   # find high and low

   i_up = int(np.argmax(grad))

   i_down = int(np.argmin(grad))

"""
Search for a non‑negative step sequence with a low official score.

The algorithm works in three phases:

1. **Seeding** – a handful of parametric families (arcsine, beta,
   uniform) are sampled and the best few are kept.

2. **Coordinate‑wise mass‑transfer descent** – at each iteration the
   exact sub‑gradient of the objective `max(conv(a,a))` is computed.
   Mass is moved from the index with the largest gradient to the one
   with the smallest gradient.  This simple greedy step rapidly drives
   the maximum of the self‑convolution down.

3. **Smooth Adam refinement** – the binary‑max is replaced by a
   temperature‑controlled soft‑max (log‑sum‑exp) and a few hundred
   thousand Adam updates are performed.  Finally the best vector is
   evaluated with the official `evaluate_sequence` function.

All vectors are kept on the probability simplex (Σa = 1) – scaling does
not affect the official score because the numerator and denominator
both grow quadratically.  The implementation respects the 1000 s time
budget and prints intermediate progress.
"""

import time
import numpy as np
from typing import List, Tuple

# ----------------------------------------------------------------------
# Low‑level utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """
    Linear convolution via real FFT.
    Length = len(a) + len(b) - 1.
    """
    m = len(a) + len(b) - 1
    n_fft = next_pow2(m)
    A = np.fft.rfft(a, n=n_fft)
    B = np.fft.rfft(b, n=n_fft)
    C = A * B
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


def compute_score(a: np.ndarray) -> float:
    """
    Fast surrogate for the official score.
    For vectors normalised to Σa = 1 the denominator disappears and the
    score is exactly ``2·n·max(conv(a,a))``.
    """
    n = len(a)
    conv = convolution_fft(a, a)
    return 2.0 * n * float(np.max(conv))


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection onto the probability simplex {x ≥ 0, Σx = 1}.
    Duchi et al., 2008.
    """
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / v.size)

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho = np.where(u - cssv / np.arange(1, v.size + 1) > 0)[0]
    if len(rho) == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


# ----------------------------------------------------------------------
# Shape factories (all return a vector that already sums to 1)
# ----------------------------------------------------------------------


def arcsine_initial(n: int) -> np.ndarray:
    """Classic arcsine shape: a_i ∝ 1/√((i+½)*(n‑½‑i))."""
    i = np.arange(n, dtype=float)
    eps = 0.25
    vals = 1.0 / np.sqrt((i + 0.5) * (n - 0.5 - i) + eps)
    vals /= vals.sum()
    return vals


def generate_beta_shape(n: int, alpha: float, beta: float, eps: float) -> np.ndarray:
    """
    (i+ε)^{α‑1}·(n‑1‑i+ε)^{β‑1} on {0,…,n‑1}, normalised to Σ=1.
    A tiny epsilon (≥1e‑6) is required when α or β < 1 to avoid division‑by‑zero.
    """
    i = np.arange(n, dtype=float)
    left = (i + eps) ** (alpha - 1.0)
    right = (n - 1 - i + eps) ** (beta - 1.0)
    base = left * right
    total = base.sum()
    if not np.isfinite(total) or total <= 0.0:
        return np.full(n, 1.0 / n)
    return base / total


# ----------------------------------------------------------------------
# Optimisers
# ----------------------------------------------------------------------


def smooth_gradient(a: np.ndarray, lam: float) -> np.ndarray:
    """
    Gradient of the soft‑max approximation of max(conv(a,a)).
    Returns ∇_a (2·n·softmax_max) – the simplex constraint is handled
    by a later projection.
    """
    n = len(a)
    conv = convolution_fft(a, a)                     # length 2·n‑1
    max_c = np.max(conv)

    # stable soft‑max weights
    exp_term = np.exp(lam * (conv - max_c))
    w = exp_term / np.sum(exp_term)

    # weighted sub‑gradient: 2·∑_k w_k shift(a,k)
    w_rev = w[::-1]
    conv_rev = convolution_fft(a, w_rev)
    L = 2 * n - 1
    weighted_grad = 2.0 * conv_rev[L - 1 - np.arange(n)]

    return 2.0 * n * weighted_grad


def adam_optimize_smooth(initial_a: np.ndarray,
                         deadline: float,
                         max_iter: int = 150_000,
                         lr0: float = 0.12,
                         lam_start: float = 2.0,
                         lam_end: float = 200.0,
                         noise_every: int = 1000,
                         noise_scale: float = 1e-5) -> Tuple[np.ndarray, float]:
    """
    Adam optimiser on the smooth surrogate.
    Returns the best vector (projected onto the simplex) and its fast score.
    """
    a = project_to_simplex(initial_a.astype(float).copy())
    best_a = a.copy()
    best_sc = compute_score(a)

    beta1, beta2 = 0.9, 0.999
    eps = 1e-8
    m = np.zeros_like(a)
    v = np.zeros_like(a)

    for it in range(1, max_iter + 1):
        if time.time() >= deadline:
            break

        lam = lam_start + (lam_end - lam_start) * min(1.0, it / max_iter)

        grad = smooth_gradient(a, lam)

        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)

        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        step = lr0 * m_hat / (np.sqrt(v_hat) + eps)
        a_new = a - step

        if noise_every and (it % noise_every == 0):
            a_new += np.random.randn(a.size) * noise_scale

        a_new = project_to_simplex(a_new)

        cur_sc = compute_score(a_new)
        if cur_sc < best_sc:
            best_sc = cur_sc
            best_a = a_new.copy()

        a = a_new

        if it % 5000 == 0:
            print(f"    AdamSmooth it={it:,d} cur={cur_sc:.6f} best={best_sc:.6f}")

    return best_a, best_sc


def exact_subgradient(a: np.ndarray) -> np.ndarray:
    """
    Exact sub‑gradient of max(conv(a,a)) at the current active set.
    """
    n = len(a)
    conv = convolution_fft(a, a)
    max_val = np.max(conv)
    # tolerance to collect the active set (indices attaining the max)
    tol = max_val * 1e-12 + 1e-14
    mask = (np.abs(conv - max_val) <= tol).astype(float)   # shape 2n‑1

    mask_rev = mask[::-1]
    conv_rev = convolution_fft(a, mask_rev)
    L = 2 * n - 1
    grad = 2.0 * conv_rev[L - 1 - np.arange(n)]
    return grad


def coordinate_descent_optimize(initial_a: np.ndarray,
                                deadline: float,
                                step0: float = 0.3,
                                max_iter: int = 400_000) -> np.ndarray:
    """
    Greedy coordinate descent: move a small amount of mass from the
    index with the largest sub‑gradient to the one with the smallest.
    The sum stays equal to 1 and all entries stay non‑negative.
    """
    a = initial_a.copy()
    n = len(a)
    best_a = a.copy()
    best_sc = compute_score(a)

    for it in range(1, max_iter + 1):
        if time.time() >= deadline:
            break

        grad = exact_subgradient(a)
        i_up = int(np.argmax(grad))
        i_down = int(np.argmin(grad))

        if i_up == i_down:
            # pathologically flat – break early
            break

        # step size decays like 1/√t
        step = step0 * a[i_up] / np.sqrt(it)

        eps = min(step, a[i_up])
        if eps <= 0:
            break

        a[i_up] -= eps
        a[i_down] += eps

        # tiny numerical drift correction every few thousand iterations
        if it % 5000 == 0:
            # ensure non‑negativity and renormalise
            a = np.maximum(a, 0.0)
            total = a.sum()
            if total > 0:
                a /= total

            cur_sc = compute_score(a)
            if cur_sc < best_sc:
                best_sc = cur_sc
                best_a = a.copy()
                print(f"    CD it={it:,d} cur={cur_sc:.6f} best={best_sc:.6f}")

    # final projection to clean up any rounding errors
    a = np.maximum(a, 0.0)
    a /= a.sum()
    if compute_score(a) < best_sc:
        best_a = a.copy()

    return best_a


def simulated_annealing_opt(a: np.ndarray,
                            rng: np.random.Generator,
                            deadline: float,
                            max_iters: int = 120_000,
                            init_temp: float = 0.02,
                            decay: float = 0.9999) -> np.ndarray:
    """
    Simple Metropolis‑style annealing using the fast surrogate score.
    """
    cur = a.copy()
    cur_sc = compute_score(cur)
    best = cur.copy()
    best_sc = cur_sc
    n = len(cur)
    temp = init_temp
    it = 0

    while it < max_iters and time.time() < deadline:
        it += 1
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue

        # move at most 20 % of the mass at i
        delta = cur[i] * 0.2 * rng.random()
        if delta <= 1e-15:
            continue

        cand = cur.copy()
        cand[i] -= delta
        cand[j] += delta

        cand_sc = compute_score(cand)

        if cand_sc < cur_sc or rng.random() < np.exp(-(cand_sc - cur_sc) / max(temp, 1e-12)):
            cur = cand
            cur_sc = cand_sc
            if cur_sc < best_sc:
                best = cur.copy()
                best_sc = cur_sc

        temp = max(temp * decay, 1e-8)

        if it % 5000 == 0:
            print(f"    SA it={it:,d} cur={cur_sc:.6f} best={best_sc:.6f}")

    return best
