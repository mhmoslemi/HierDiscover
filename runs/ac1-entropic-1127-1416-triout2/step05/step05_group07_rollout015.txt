    # v is vector, project onto {x >=0, sum(x)=1}
    u = np.sort(v)[::-1]

def optimize_smooth_expgrad(a0, beta_schedule, eta0, deadline):
    a = np.copy(a0)
    best = a.copy()
    best_score = compute_score(a)
    it = 0
    while True:
        it += 1
        if time.time() >= deadline:
            break
        # get beta for current iteration
        beta = beta_schedule(it)
        # compute conv
        conv = convolution_fft(a, a)   # length 2n-1
        # compute softmax weights
        maxc = conv.max()
        exp_vals = np.exp(beta * (conv - maxc))
        w = exp_vals / exp_vals.sum()
        # compute gradient = 2 * correlation(a, w)
        # correlation:
        n = len(a)
        L = len(w)  # 2n-1
        N = next_pow2(n + L - 1)
        a_pad = np.concatenate([a, np.zeros(N - n)])
        w_pad = np.concatenate([w, np.zeros(N - L)])
        A = np.fft.rfft(a_pad)
        W = np.fft.rfft(w_pad)
        corr = np.fft.irfft(A * np.conj(W), n=N)
        # extract gradient (first n elements after offset)
        grad = 2.0 * corr[L-1:L-1 + n]
        # update a = a * exp(-eta * grad)
        eta = eta0 / np.sqrt(it)   # or schedule
        a = a * np.exp(-eta * grad)
        # renormalize to sum=1
        a_sum = a.sum()
        if a_sum == 0:
            a = np.full_like(a, 1.0 / len(a))
        else:
            a = a / a_sum
        # compute cheap score
        cur_score = compute_score(a)
        if cur_score < best_score:
            best_score = cur_score
            best = a.copy()
        # optionally print progress every few thousand steps
        if it % 5000 == 0:
            print(f"      iter {it:,d} cur={cur_score:.6f} best={best_score:.6f} beta={beta:.2f}")
    return best, best_score

def propose_candidate(seed=123, budget_s=1000):
    rng = np.random.default_rng(seed)
    start = time.time()
    deadline = start + budget_s - 0.5
    best_seq = None
    best_val = float('inf')
    best_vec = None
    # fallback from height_sequence_1
    try:
        fallback = [float(x) for x in height_sequence_1]
        val = evaluate_sequence(fallback)
        print(f"fallback score = {val:.6f}")
        if val < best_val:
            best_val = val
            best_seq = fallback
            best_vec = np.array(fallback, dtype=float)
            best_vec /= best_vec.sum()
    except Exception:
        pass
    # define lengths
    lengths = [512, 1024, 2048, 4096, 8192]
    for n in lengths:
        if time.time() >= deadline - 5:
            break
        print(f"\n=== Optimizing length {n} ===")
        if best_vec is not None:
            a0 = upsample(best_vec, n)
        else:
            a0 = arcsine_initial(n)
        # run optimizer
        a_opt, sc = run_optimizer(a0, n, deadline)
        # evaluate with official evaluator
        val = evaluate_sequence(a_opt.tolist())
        print(f"official score for n={n} = {val:.6f}")
        if val < best_val:
            best_val = val
            best_seq = a_opt.tolist()
            best_vec = a_opt.copy()
    # final polishing if time remains
    if best_seq is not None and time.time() < deadline - 5:
        print("\nFinal polishing")
        a_opt, sc = run_optimizer(np.array(best_seq), len(best_seq), deadline)
        val = evaluate_sequence(a_opt.tolist())
        print(f"final official score = {val:.6f}")
        if val < best_val:
            best_val = val
            best_seq = a_opt.tolist()
    # safety fallback
    if not best_seq:
        best_seq = [1.0] * 256
        best_val = evaluate_sequence(best_seq)
    print(f"Search finished. Best value = {best_val:.6f}")
    return best_seq

"""
propose_candidate – hierarchical optimisation of a step‑function.

The routine starts from a symmetric arcsine distribution (or from the
fallback `height_sequence_1` if it exists) and repeatedly up‑samples the
current best vector to larger sizes.  At each size a smooth approximation
to the objective

    Φ(a) = 2·n·max_k (a∗a)_k / (∑a)^2

is minimised.  The max is replaced by a soft‑max
    S_β(a) = (1/β)·log∑_k exp(β·(a∗a)_k)

so that the gradient is well‑defined and cheap to compute via a single
FFT‑based correlation.  An exponentiated‑gradient (mirror‑descent)
update keeps the iterate non‑negative; a simple renormalisation enforces
the simplex constraint (∑a = 1).  The temperature β is increased in
stages so that the optimisation first follows a smooth landscape and
later focuses on the true maximum.

The whole search respects the 1000 s wall‑clock budget, logs progress,
and always returns a feasible list of non‑negative heights.
"""

import time
import numpy as np
from typing import List, Tuple

# ----------------------------------------------------------------------
# Basic numeric helpers
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Linear convolution using a real FFT (fast and accurate)."""
    m = len(a) + len(b) - 1
    n_fft = next_pow2(m)
    A = np.fft.rfft(a, n=n_fft)
    B = np.fft.rfft(b, n=n_fft)
    C = A * B
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


def correlation_fft(a: np.ndarray, w: np.ndarray) -> np.ndarray:
    """
    Linear correlation a ⋆ w (full length = len(a)+len(w)-1)
    using real FFTs.
    """
    n = len(a)
    L = len(w)
    N = next_pow2(n + L - 1)
    a_pad = np.concatenate([a, np.zeros(N - n, dtype=a.dtype)])
    w_pad = np.concatenate([w, np.zeros(N - L, dtype=w.dtype)])
    A = np.fft.rfft(a_pad)
    W = np.fft.rfft(w_pad)
    corr = np.fft.irfft(A * np.conj(W), n=N)
    return corr  # length N (meaningful part is first n+L-1 entries)


def compute_score(a: np.ndarray) -> float:
    """
    Cheap surrogate of the official evaluator:
        2·n·max(a*a) / (sum(a)²)
    """
    n = len(a)
    conv = convolution_fft(a, a)
    max_val = float(conv.max())
    s = float(a.sum())
    if s < 1e-12:
        return float("inf")
    return 2.0 * n * max_val / (s * s)


# ----------------------------------------------------------------------
# Helpful constructions
# ----------------------------------------------------------------------


def arcsine_initial(n: int) -> np.ndarray:
    """Discrete arcsine density a_i ∝ 1/√((i+½)(n‑½‑i))."""
    i = np.arange(n, dtype=float)
    eps = 0.25
    vals = 1.0 / np.sqrt((i + 0.5) * (n - 0.5 - i) + eps)
    return vals / vals.sum()


def upsample(prev: np.ndarray, new_n: int) -> np.ndarray:
    """
    Linear interpolation from len(prev) → new_n, followed by re‑normalisation.
    """
    x_old = np.linspace(0.0, 1.0, len(prev))
    x_new = np.linspace(0.0, 1.0, new_n)
    interp = np.interp(x_new, x_old, prev)
    total = interp.sum()
    if total == 0.0:
        return np.full(new_n, 1.0 / new_n)
    return interp / total


# ----------------------------------------------------------------------
# Optimisation core – smooth‑max exponentiated‑gradient descent
# ----------------------------------------------------------------------


def beta_for_iter(it: int) -> float:
    """Piece‑wise schedule for the soft‑max temperature."""
    if it < 2_000:
        return 5.0
    if it < 10_000:
        return 20.0
    if it < 30_000:
        return 80.0
    return 200.0  # final temperature (still far from overflow)


def run_optimizer(
    a0: np.ndarray,
    deadline: float,
    *,
    eta0: float = 0.7,
    max_iter: int = 300_000,
) -> Tuple[np.ndarray, float]:
    """
    Exponentiated‑gradient descent on the smooth approximation
    Φ_β(a) = 2·n·S_β(a∗a) where S_β is the soft‑max of the self‑convolution.
    Returns the best vector seen (normalised to sum = 1) and its cheap
    surrogate score.
    """
    a = np.maximum(a0, 1e-30)   # avoid exact zeros
    a = a / a.sum()
    n = len(a)
    L = 2 * n - 1                # length of the full convolution
    N = next_pow2(3 * n)         # FFT size large enough for both conv & corr

    best_a = a.copy()
    best_sc = compute_score(a)

    it = 0
    while time.time() < deadline and it < max_iter:
        it += 1

        # ----- FFT of a (once per iteration) -----
        a_pad = np.concatenate([a, np.zeros(N - n, dtype=a.dtype)])
        A = np.fft.rfft(a_pad)

        # ----- Self‑convolution (for the soft‑max) -----
        conv_full = np.fft.irfft(A * A, n=N)[:L]       # length 2n‑1
        max_conv = conv_full.max()

        # ----- Soft‑max weights -----
        beta = beta_for_iter(it)
        # stable softmax: subtract max
        exp_term = np.exp(beta * (conv_full - max_conv))
        w = exp_term / exp_term.sum()                 # sum_w == 1

        # ----- Correlation a ⋆ w (gradient of soft‑max) -----
        w_pad = np.concatenate([w, np.zeros(N - L, dtype=w.dtype)])
        W = np.fft.rfft(w_pad)
        corr_full = np.fft.irfft(A * np.conj(W), n=N)
        offset = L - 1                                 # position of lag 0
        grad = 2.0 * corr_full[offset:offset + n]      # length n

        # ----- Exponentiated‑gradient update -----
        eta = eta0 / np.sqrt(it)                      # classic 1/√t schedule
        a = a * np.exp(-eta * grad)                    # multiplicative step

        # ----- Re‑normalise onto the simplex -----
        total = a.sum()
        if total == 0.0:
            a = np.full_like(a, 1.0 / n)
        else:
            a = a / total

        # ----- Track the best surrogate score -----
        cur_sc = compute_score(a)
        if cur_sc < best_sc:
            best_sc = cur_sc
            best_a = a.copy()

        # occasional progress report
        if it % 5_000 == 0:
            print(
                f"      iter {it:,d}  β={beta:.0f}  cur={cur_sc:.6f}  best={best_sc:.6f}"
            )

    return best_a, best_sc


# ----------------------------------------------------------------------
# Main entry point required by the evaluation harness
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs) -> List[float]:
    """
    Search for a non‑negative step‑function with a low ``evaluate_sequence``
    value.  A hierarchical scheme (512 → 1024 → 2048 → 4096 → 8192) is used.
    At each scale a smooth‑max exponentiated‑gradient descent is run.
    The routine respects the supplied time budget and always returns a
    feasible list.
    """
    rng = np.random.default_rng(seed)
    start = time.time()
    # keep a small safety margin for the final official evaluation
    deadline = start + budget_s - 0.5

    best_seq: List[float] = []
    best_val: float = float("inf")
    best_internal: np.ndarray = None  # normalised (sum = 1)

    # ------------------------------------------------------------------
    # 0.  Fallback: use the previously discovered construction if present.
    # ------------------------------------------------------------------
    try:
        fallback = [float(x) for x in height_sequence_1]  # type: ignore
        val = evaluate_sequence(fallback)
        print(f"Fallback (height_sequence_1) score = {val:.6f}")
        if val < best_val:
            best_val = val
            best_seq = fallback
            best_internal = np.array(fallback, dtype=float)
            best_internal = best_internal / best_internal.sum()
    except Exception:
        # No fallback available – that's fine.
        pass

    # ------------------------------------------------------------------
    # 1.  Hierarchical optimisation – increase the length progressively.
    # ------------------------------------------------------------------
    # The set of lengths is chosen so that each step roughly doubles the
    # problem size while staying well within the time budget.
    lengths = [512, 1024, 2048, 4096, 8192]

    for n in lengths:
        if time.time() >= deadline - 5.0:
            break
        print(f"\n=== Optimising length n = {n} ===")

        # ----- Initialise the candidate for this size -----
        if best_internal is not None:
            a0 = upsample(best_internal, n)
        else:
            a0 = arcsine_initial(n)

        # Occasionally inject a random Dirichlet‑type perturbation to
        # diversify the search (only if we have enough time left).
        if time.time() < deadline - 10.0 and rng.random() < 0.3:
            # heavy‑tailed draw, then re‑normalise
            tmp = rng.gamma(0.3, 1.0, size=n)
            a0 = np.maximum(tmp, 0.0)
            a0 = a0 / a0.sum()

        # ----- Run the smooth‑max EG optimiser -----
        a_opt, surrogate = run_optimizer(a0, deadline)

        # ----- Official (expensive) evaluation of the best candidate -----
        val = evaluate_sequence(a_opt.tolist())
        print(f"    evaluated official score = {val:.6f}")

        if val < best_val:
            best_val = val
            best_seq = a_opt.tolist()
            best_internal = a_opt.copy()
            print(f"    *** New global best: {best_val:.6f} (n={n}) ***")

    # ------------------------------------------------------------------
    # 2.  Final polishing of the global best (if time permits).
    # ------------------------------------------------------------------
    if best_seq and time.time() < deadline - 5.0:
        print("\nFinal polishing of the global best")
        a_start = np.array(best_seq, dtype=float)
        a_refined, _ = run_optimizer(
            a_start,
            deadline,
            eta0=0.4,      # smaller steps for fine‑tuning
        )
        final_val = evaluate_sequence(a_refined.tolist())
        print(f"Final polished official score = {final_val:.6f}")
        if final_val < best_val:
            best_val = final_val
            best_seq = a_refined.tolist()
            print("*** Updated global best after final polish ***")

    # ------------------------------------------------------------------
    # 3.  Guard – always return a feasible sequence.
    # ------------------------------------------------------------------
    if not best_seq:
        best_seq = [1.0] * 256
        best_val = evaluate_sequence(best_seq)

    print(f"\nSearch finished. Best reported official score = {best_val:.6f}")
    return best_seq
