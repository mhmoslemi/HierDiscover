def gradient_smoothed(a, conv, eps):
    # conv computed.
    # compute softmax weights of conv values.
    max_c = np.max(conv)  # for stability.
    shifted = conv - max_c  # negative values.
    exp_vals = np.exp(eps * shifted)
    sum_exp = np.sum(exp_vals)
    w = exp_vals / sum_exp
    # reversed w for convolution
    w_rev = w[::-1]
    grad = 2.0 * convolution_fft(a, w_rev)
    return grad

def smoothed_gradient_descent(a: np.ndarray,
                              deadline: float,
                              total_iters: int = 200_000,
                              eps_schedule: List[float] = [5.0, 20.0, 80.0, 200.0],
                              step0: float = 0.2) -> np.ndarray:
    a = project_to_simplex(a.copy())
    n = a.shape[0]
    L = 2*n - 1
    it = 0
    # divide total_iters among eps steps proportionally (equal)
    per_eps = max(1, total_iters // len(eps_schedule))
    for eps in eps_schedule:
        for _ in range(per_eps):
            if time.time() >= deadline:
                return a
            it += 1
            conv = convolution_fft(a, a)
            max_c = np.max(conv)
            # compute softmax weights
            # shift conv for stability
            shifted = conv - max_c
            # Clip extreme values to avoid overflow
            # epsilon may be large, exponent may overflow; use np.exp with clipping
            # we can use np.exp(np.clip(eps*shifted, -700, 700))
            exp_arg = np.clip(eps * shifted, -700, 700)
            exp_vals = np.exp(exp_arg)
            sum_exp = np.sum(exp_vals)
            if sum_exp == 0:
                w = np.full_like(conv, 1.0/conv.size)
            else:
                w = exp_vals / sum_exp
            # gradient = 2 * convolution of a with reversed w
            grad = 2.0 * convolution_fft(a, w[::-1])
            eta = step0 / np.sqrt(it)
            a = a - eta * grad
            a = project_to_simplex(a)
            # occasional logging
            if it % 5000 == 0:
                cur = compute_score(a)
                print(f"    SmoothGrad it={it:,d} eps={eps} score={cur:.6f}")
    return a

1️⃣  **Seed** – we start from the globally stored `height_sequence_1`
    (if it exists) and normalise it to the probability simplex.
    When unavailable we fall back to the classic arcsine distribution.

2️⃣  **Heavy sub‑gradient descent** – the exact sub‑gradient of the
    objective `f(a)=2·n·max(conv(a,a))` is cheap to compute (one extra FFT).
    We run this loop for as many iterations as the remaining time permits,
    using a fairly aggressive initial step‑size (`step0≈0.5`) and the classic
    `η_t = step0/√t` schedule.  The iterate is always projected back onto the
    simplex with the Euclidean projection of Duchi et al. (2008).

3️⃣  **Smooth‑gradient refinement** – once the sub‑gradient phase
    exhausts most of the budget we switch to a smooth approximation of the
    `max` using a soft‑max with a schedule of growing temperatures
    (`ε ∈ {5,20,80,200}`).  The gradient of the smoothed objective is a
    weighted convolution and is handled with the same FFT routine.

4️⃣  **Exponentiated‑gradient polishing** – a final short mirror‑descent
    pass (`a_i ← a_i·exp(-η·g_i)`) automatically respects the simplex.

All heavy work is protected by deadline checks, so the function always
returns a valid (non‑empty) sequence before the 1000 s limit.
"""

import time
import numpy as np
from typing import List, Tuple

# ----------------------------------------------------------------------
# Low‑level utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Linear convolution via real FFT (length = len(a)+len(b)-1)."""
    m = len(a) + len(b) - 1
    n_fft = next_pow2(m)
    A = np.fft.rfft(a, n=n_fft)
    B = np.fft.rfft(b, n=n_fft)
    C = A * B
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection onto the probability simplex {x ≥ 0, Σx = 1}.
    Duchi et al., 2008.
    """
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / v.size)

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho_arr = u - cssv / np.arange(1, v.size + 1)
    rho = np.where(rho_arr > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1.0)

    w = np.maximum(v - theta, 0.0)
    return w


def compute_score(a: np.ndarray) -> float:
    """
    Fast surrogate for the official score.
    The vector is assumed to be normalised to Σa = 1, therefore the
    denominator disappears and the score is simply 2·n·max(conv(a,a)).
    """
    n = a.shape[0]
    conv = convolution_fft(a, a)
    return 2.0 * n * float(np.max(conv))


# ----------------------------------------------------------------------
# Shape generators (used only for fall‑back initialisation)
# ----------------------------------------------------------------------


def arcsine_initial(n: int) -> np.ndarray:
    """Classic arcsine shape: a_i ∝ 1/√((i+½)*(n‑½‑i))."""
    i = np.arange(n, dtype=float)
    eps = 0.25
    vals = 1.0 / np.sqrt((i + 0.5) * (n - 0.5 - i) + eps)
    vals /= vals.sum()
    return vals


def generate_power_shape(n: int, p: float, eps: float) -> np.ndarray:
    """
    Generalised arcsine: a_i ∝ ((i+½)*(n‑½‑i)+ε)^{‑p}.
    p>0, eps≥0.
    """
    i = np.arange(n, dtype=float)
    denom = ((i + 0.5) * (n - 0.5 - i) + eps) ** p
    vals = 1.0 / denom
    total = vals.sum()
    if total <= 0.0 or not np.isfinite(total):
        return np.full(n, 1.0 / n)
    return vals / total


# ----------------------------------------------------------------------
# Optimisation primitives
# ----------------------------------------------------------------------


def subgradient_descent_exact(initial_a: np.ndarray,
                              deadline: float,
                              max_iter: int = 10_000_000,
                              step0: float = 0.5) -> np.ndarray:
    """
    Exact sub‑gradient descent on   f(a)=2·n·max(conv(a,a)).
    The step size follows η_t = step0 / √t.
    """
    a = project_to_simplex(initial_a.copy())
    n = len(a)
    L = 2 * n - 1

    for it in range(1, max_iter + 1):
        if time.time() >= deadline:
            break

        conv = convolution_fft(a, a)
        max_val = float(np.max(conv))

        # indices where the maximum is attained
        tol = max_val * 1e-12 + 1e-14
        mask = (np.abs(conv - max_val) <= tol).astype(float)

        # sub‑gradient = 2·∑_{k∈K} shift(a,k)
        mask_rev = mask[::-1]
        conv_rev = convolution_fft(a, mask_rev)          # length = n + (2n‑1)‑1 = 3n‑2
        subgrad = 2.0 * conv_rev[L - 1 - np.arange(n)]

        eta = step0 / np.sqrt(it)
        a = a - eta * subgrad
        a = project_to_simplex(a)

        # occasional progress report
        if it % 10_000 == 0:
            cur_sc = compute_score(a)
            print(f"    SubGrad it={it:,d}  score={cur_sc:.6f}")

    return a


def smoothed_gradient_descent(a: np.ndarray,
                              deadline: float,
                              total_iters: int = 300_000,
                              eps_schedule: List[float] = (5.0, 20.0, 80.0, 200.0),
                              step0: float = 0.2) -> np.ndarray:
    """
    Gradient descent on the smooth approximation of the max:
        f_ε(a) = (1/ε)·log(∑_k exp(ε·conv_k(a))).

    The gradient is 2·conv(a, w_rev) where w are soft‑max weights.
    """
    a = project_to_simplex(a.copy())
    n = a.shape[0]
    L = 2 * n - 1
    it = 0
    per_eps = max(1, total_iters // len(eps_schedule))

    for eps in eps_schedule:
        for _ in range(per_eps):
            if time.time() >= deadline:
                return a
            it += 1

            conv = convolution_fft(a, a)
            max_c = np.max(conv)

            # soft‑max weights (stable)
            shifted = conv - max_c
            exp_arg = np.clip(eps * shifted, -700, 700)   # avoid overflow
            exp_vals = np.exp(exp_arg)
            w = exp_vals / exp_vals.sum()

            # gradient = 2·∑_k w_k·shift(a,k)
            w_rev = w[::-1]
            conv_rev = convolution_fft(a, w_rev)
            grad = 2.0 * conv_rev[L - 1 - np.arange(n)]

            eta = step0 / np.sqrt(it)
            a = a - eta * grad
            a = project_to_simplex(a)

            if it % 10_000 == 0:
                cur_sc = compute_score(a)
                print(f"    SmoothGrad it={it:,d}  eps={eps:.1f}  score={cur_sc:.6f}")

    return a


def eg_refine(a: np.ndarray,
              deadline: float,
              max_iter: int = 200_000,
              eta0: float = 0.4) -> np.ndarray:
    """
    Exponentiated‑gradient polishing (mirror‑descent on the simplex).
    """
    a = a.copy()
    n = a.shape[0]
    L = 2 * n - 1

    for it in range(1, max_iter + 1):
        if time.time() >= deadline:
            break

        conv = convolution_fft(a, a)
        max_val = float(np.max(conv))
        tol = max_val * 1e-12 + 1e-14
        mask = (np.abs(conv - max_val) <= tol).astype(float)

        mask_rev = mask[::-1]
        conv_rev = convolution_fft(a, mask_rev)
        subgrad = 2.0 * conv_rev[L - 1 - np.arange(n)]

        eta = eta0 / (np.sqrt(it) + 1.0)
        a = a * np.exp(-eta * subgrad)

        total = a.sum()
        if total == 0.0:
            a = np.full_like(a, 1.0 / n)
        else:
            a = a / total

        if it % 10_000 == 0:
            cur_sc = compute_score(a)
            print(f"    EG it={it:,d}  score={cur_sc:.6f}")

    return a


# ----------------------------------------------------------------------
# Main entry point required by the competition harness
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42,
                     budget_s: int = 1000,
                     **kwargs) -> List[float]:
    """
    Search for a non‑negative step sequence with a low official score.
    The routine concentrates the whole time budget on a single high‑quality
    seed (fallback or arcsine) and performs a long sub‑gradient descent,
    followed by a smooth‑gradient refinement and a short EG polish.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5   # safety margin

    # ------------------------------------------------------------------
    # 1.  Choose a good starting seed
    # ------------------------------------------------------------------
    fallback = globals().get("height_sequence_1")
    if fallback is not None:
        try:
            seq = np.array([float(x) for x in fallback], dtype=float)
            if seq.size == 0:
                raise ValueError
            seq = np.maximum(seq, 0.0)
            seq = seq / seq.sum()
            print(f"Using global fallback sequence (len={len(seq)})")
        except Exception:
            seq = arcsine_initial(2048)
            print("Fallback corrupted – using arcsine shape.")
    else:
        seq = arcsine_initial(2048)
        print("No fallback found – using arcsine shape.")

    # ------------------------------------------------------------------
    # 2.  Intensive optimisation of the chosen seed
    # ------------------------------------------------------------------
    # Phase A – heavy sub‑gradient descent
    if time.time() < deadline - 5.0:
        print("\n--- Phase A: Sub‑gradient descent (heavy) ---")
        seq = subgradient_descent_exact(seq, deadline,
                                        max_iter=10_000_000,
                                        step0=0.5)
        cur_sc = compute_score(seq)
        print(f"Phase A finished – fast score = {cur_sc:.6f}")

    # Phase B – smooth‑gradient refinement
    if time.time() < deadline - 5.0:
        print("\n--- Phase B: Smoothed gradient descent ---")
        remaining = deadline - time.time()
        # crude estimate of how many iterations we can afford (≈0.00015 s per it)
        possible_iters = int(remaining / 0.00015)
        total_iters = min(500_000, possible_iters)
        seq = smoothed_gradient_descent(seq, deadline,
                                        total_iters=total_iters,
                                        eps_schedule=(5.0, 20.0, 80.0, 200.0),
                                        step0=0.2)
        cur_sc = compute_score(seq)
        print(f"Phase B finished – fast score = {cur_sc:.6f}")

    # Phase C – final exponentiated‑gradient polish
    if time.time() < deadline - 2.0:
        print("\n--- Phase C: Exponentiated‑gradient polish ---")
        seq = eg_refine(seq, deadline,
                        max_iter=150_000,
                        eta0=0.4)
        cur_sc = compute_score(seq)
        print(f"Phase C finished – fast score = {cur_sc:.6f}")

    # ------------------------------------------------------------------
    # 3.  Official evaluation (once, at the end)
    # ------------------------------------------------------------------
    try:
        final_val = evaluate_sequence(seq.tolist())
        print(f"\nOfficial final score = {final_val:.6f}")
    except Exception as e:
        print(f"Official evaluation failed ({e}), falling back to uniform.")
        n = 256
        seq = np.full(n, 1.0 / n)
        final_val = evaluate_sequence(seq.tolist())
        print(f"Fallback uniform score = {final_val:.6f}")

    # ------------------------------------------------------------------
    # 4.  Guard – guarantee a non‑empty return
    # ------------------------------------------------------------------
    if not isinstance(seq, np.ndarray) or seq.size == 0:
        n = 256
        seq = np.full(n, 1.0 / n)

    return seq.tolist()
