        a = project_to_simplex(a)
    return a

"""
Propose a step‑function (non‑negative vector) with a very low
2·n·max(conv(a,a)) / (Σa)² score.

Algorithm
---------
1. **Broad random search** over several lengths (1024 … 16384).  Shapes
   are drawn from a few analytically‑motivated families:
   * plain arcsine,
   * arcsine with a tunable ε,
   * power‑law ( (x(1‑x)+ε)^‑α ) with random α,ε,
   * mixtures with the uniform distribution,
   * beta‑family densities (including mixtures of two betas).

   For each length the best raw shape (according to the fast surrogate
   `compute_score`) is kept.

2. **Heavy local polishing** of the globally best raw candidate:
   * Adam optimisation of a smoothed log‑sum‑exp surrogate
     (temperature annealed up to 250).
   * Exact sub‑gradient descent with back‑tracking line‑search on the
     true max‑convolution.
   * A cheap stochastic Gaussian walk.

3. **Linear‑programming refinement** (if SciPy is available):
   * With the current vector `a` compute `M = max(a*a)`.
   * Solve the LP   maximise ∑g   s.t.   (a * g) ≤ M ,   g ≥ 0 .
     The constraint matrix is a Toeplitz matrix that we build as a
     sparse CSR matrix.
   * If the LP yields a vector with a larger ℓ₁‑norm, rescale it to the
     same sum as `a` and replace `a ← (1‑t)a + t·g_scaled` (t≈0.5).
   * Iterate the LP step a few times – each iteration can only lower
     the objective.

4. Final official evaluation and return of the best sequence found.
   A uniform fallback of length 256 is guaranteed.

All steps respect the supplied time budget (default 1000 s) and print
progress information.
"""

import time
import numpy as np

# ----------------------------------------------------------------------

    from scipy.optimize import linprog  # type: ignore

except Exception:  # pragma: no cover
    SCIPY_AVAILABLE = False

# Linear‑algebra utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """
    Linear convolution via real FFT.
    Returns an array of length len(a)+len(b)-1.
    """
    m = a.size + b.size - 1
    n_fft = next_pow2(m)
    A = np.fft.rfft(a, n=n_fft)
    B = np.fft.rfft(b, n=n_fft)
    C = A * B
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


def max_convolution(a: np.ndarray) -> float:
    """Maximum entry of the self‑convolution of a."""
    return float(np.max(convolution_fft(a, a)))


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection onto the probability simplex
    {x ≥ 0, Σx = 1}.
    Implementation of Duchi et al. (2008).
    """
    n = v.shape[0]
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / n)

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho = np.where(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def compute_score(a: np.ndarray) -> float:
    """
    Fast surrogate: 2·n·max(conv(a,a)).
    For a vector normalised to Σa = 1 this equals the official score.
    """
    n = a.shape[0]
    return 2.0 * n * max_convolution(a)

# Shape generators

def generate_beta_shape(
    n: int, alpha: float, beta: float, eps: float
) -> np.ndarray:
    """Symmetric beta‑family density on {0,…,n‑1}."""
    i = np.arange(n, dtype=float)
    left = (i + eps) ** (alpha - 1.0)
    right = (n - 1 - i + eps) ** (beta - 1.0)
    base = left * right
    total = base.sum()
    if total <= 0.0 or not np.isfinite(total):
        return np.full(n, 1.0 / n)
    return base / total


def generate_beta_mixture(
    n: int,
    alpha1: float,
    beta1: float,
    eps1: float,
    alpha2: float,
    beta2: float,
    eps2: float,
    w: float,
) -> np.ndarray:
    """Convex combination of two beta‑family densities."""
    a1 = generate_beta_shape(n, alpha1, beta1, eps1)
    a2 = generate_beta_shape(n, alpha2, beta2, eps2)
    mix = w * a1 + (1.0 - w) * a2
    mix_sum = mix.sum()
    if mix_sum == 0.0 or not np.isfinite(mix_sum):
        return np.full(n, 1.0 / n)
    return mix / mix_sum


def arcsine_initial(n: int) -> np.ndarray:
    """Classic arcsine shape (symmetrised), ε = 0.25."""
    i = np.arange(n, dtype=float)
    eps = 0.25
    vals = 1.0 / np.sqrt((i + 0.5) * (n - 0.5 - i) + eps)
    vals /= vals.sum()
    return vals


def arcsine_eps_shape(n: int, eps: float) -> np.ndarray:
    """Arcsine‑like shape with a tunable ε."""
    i = np.arange(n, dtype=float)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - 0.5 - i) + eps)
    vals /= vals.sum()
    return vals


def power_law_shape(n: int, alpha: float, eps: float) -> np.ndarray:
    """Density proportional to ((x·(1‑x))+ε)^(-α) on a discrete grid."""
    i = np.arange(n, dtype=float)
    vals = ((i + 0.5) * (n - 0.5 - i) + eps) ** (-alpha)
    vals /= vals.sum()
    return vals


# ----------------------------------------------------------------------
# Smooth surrogate (log‑sum‑exp) and its gradient (used by Adam)
# ----------------------------------------------------------------------


def smooth_gradient(a: np.ndarray, lam: float):
    """
    Gradient of the log‑sum‑exp surrogate of max(conv(a,a)).
    Returns (gradient, surrogate_value).
    """
    n = a.shape[0]
    conv = convolution_fft(a, a)          # length 2n‑1
    max_c = np.max(conv)

    # stable log‑sum‑exp
    exp_term = np.exp(lam * (conv - max_c))
    sum_exp = np.sum(exp_term)
    w = exp_term / sum_exp                # soft‑max on convolution entries
    B = max_c + np.log(sum_exp) / lam     # smooth approximation of the max

    # weighted sum of sub‑gradients:
    # ∂/∂a  Σ_k w_k (a*a)_k = 2 * (a ⋆ w_rev)
    w_rev = w[::-1]
    conv_rev = convolution_fft(a, w_rev)      # length 3n‑2
    L = 2 * n - 1
    weighted_grad = 2.0 * conv_rev[L - 1 - np.arange(n)]

    s = np.sum(a)
    term1 = (2.0 * n / (s ** 2)) * weighted_grad
    term2 = -(4.0 * n * B / (s ** 3)) * np.ones_like(a)
    grad = term1 + term2
    return grad, B


def adam_optimize(
    initial_a: np.ndarray,
    time_deadline: float,
    max_iter: int = 150_000,
    lr0: float = 0.12,
    lam_start: float = 1.0,
    lam_end: float = 250.0,
    noise_every: int = 1200,
    noise_scale: float = 4e-5,
) -> tuple[np.ndarray, float]:
    """
    Adam optimisation on the smooth log‑sum‑exp surrogate.
    Returns the best vector (projected on the simplex) and its fast score.
    """
    a = project_to_simplex(initial_a.astype(float).copy())
    best_a = a.copy()
    best_sc = compute_score(a)

    beta1, beta2 = 0.9, 0.999
    eps_adam = 1e-8
    m = np.zeros_like(a)
    v = np.zeros_like(a)

    n = a.shape[0]

    for it in range(1, max_iter + 1):
        if time.time() >= time_deadline:
            break

        # annealed temperature (square‑root schedule)
        t = it / max_iter
        lam = lam_start + (lam_end - lam_start) * (t ** 0.5)

        grad, _ = smooth_gradient(a, lam)

        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)

        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        step = lr0 * m_hat / (np.sqrt(v_hat) + eps_adam)
        a_new = a - step

        if noise_every and (it % noise_every == 0):
            a_new += np.random.randn(n) * noise_scale

        a_new = project_to_simplex(a_new)

        cur_sc = compute_score(a_new)
        if cur_sc < best_sc:
            best_sc = cur_sc
            best_a = a_new.copy()

        a = a_new

        if it % 5000 == 0:
            print(
                f"    Adam iter={it:,d} cur={cur_sc:.6f} best={best_sc:.6f}"
            )

    return best_a, best_sc

# Exact sub‑gradient descent with back‑tracking line‑search
# ----------------------------------------------------------------------


def subgradient_optimize(
    initial_a: np.ndarray,
    time_deadline: float,
    step0: float = 0.25,
    max_iter: int = 150_000,
) -> tuple[np.ndarray, float]:
    """
    Projected sub‑gradient descent on the true max‑convolution.
    Returns the best vector (projected) and its fast score.
    """
    a = initial_a.copy()
    best_a = a.copy()
    best_sc = compute_score(a)

    n = a.shape[0]

    for it in range(1, max_iter + 1):
        if time.time() >= time_deadline:
            break

        conv = convolution_fft(a, a)
        max_val = float(np.max(conv))
        idx = int(np.argmax(conv))

        # Gradient of the active quadratic term
        i = np.arange(n)
        j = idx - i
        mask = (j >= 0) & (j < n)
        grad = np.zeros_like(a)
        grad[mask] = 2.0 * a[j[mask]]

        # back‑tracking line search
        step = step0 / np.sqrt(it)
        for _ in range(12):
            a_new = a - step * grad
            a_new = project_to_simplex(a_new)
            new_val = max_convolution(a_new)
            if new_val < max_val - 1e-14:
                break
            step *= 0.5

        cur_sc = compute_score(a_new)
        if cur_sc < best_sc:
            best_sc = cur_sc
            best_a = a_new.copy()

        a = a_new

        if it % 40000 == 0:
            print(
                f"    Sub‑grad iter={it:,d} cur={cur_sc:.6f} best={best_sc:.6f}"
            )

    return best_a, best_sc

def random_local_search(
    initial_a: np.ndarray,
    time_deadline: float,
    init_step: float = 0.012,

    """
    Cheap Gaussian walk on the simplex with a tiny uphill‑move probability.
    Returns the best vector seen and its fast score.
    """
    a = initial_a.copy()
    best_a = a.copy()
    best_sc = compute_score(a)

    n = a.shape[0]
    rng = np.random.default_rng()

    for it in range(1, max_iter + 1):
        if time.time() >= time_deadline:
            break

        step = init_step / np.sqrt(it)
        noise = rng.normal(scale=step, size=n)
        a_proposed = a + noise
        a_proposed = project_to_simplex(a_proposed)

        cur_sc = compute_score(a_proposed)

        # accept improving moves; occasionally allow tiny uphill jumps
        if cur_sc < best_sc or (rng.random() < 1e-4 and cur_sc < 10 * best_sc):
            a = a_proposed
            if cur_sc < best_sc:
                best_sc = cur_sc
                best_a = a_proposed.copy()

        if it % 50000 == 0:
            print(
                f"    Random walk iter={it:,d} cur={cur_sc:.6f} best={best_sc:.6f}"
            )

    return best_a, best_sc


# ----------------------------------------------------------------------
# Optional LP‑based refinement (requires SciPy)
# ----------------------------------------------------------------------


if SCIPY_AVAILABLE:
    def _conv_matrix(f: np.ndarray):
        """Toeplitz matrix that maps g ↦ f * g (linear convolution)."""
        n = f.shape[0]
        rows = 2 * n - 1
        # each offset d (0…n‑1) has a constant diagonal = f[d]
        data = [np.full(n, fd, dtype=float) for fd in f]
        offsets = np.arange(n)
        return diags(data, offsets, shape=(rows, n), format="csr")

    def lp_iterative_refine(
        a: np.ndarray,
        deadline: float,
        steps: int = 3,
        t: float = 0.5,
    ) -> np.ndarray:
        """
        Repeated LP refinement:
        maximise Σg   s.t.   (a * g) ≤ M,   g ≥ 0.
        If the LP yields a strictly larger ℓ₁‑norm, blend the scaled solution
        back into `a`.  The procedure can only lower the max‑convolution.
        """
        n = a.shape[0]
        for s in range(steps):
            if time.time() >= deadline:
                break
            M = max_convolution(a)
            A = _conv_matrix(a)                     # (2n‑1)×n sparse
            b = np.full(2 * n - 1, M, dtype=float)
            c = -np.ones(n, dtype=float)            # minimise -Σg  ≡ maximise Σg
            try:
                res = linprog(
                    c,
                    A_ub=A,
                    b_ub=b,
                    bounds=(0, None),
                    method="highs",
                )
            except Exception:  # pragma: no cover
                break
            if not res.success:
                break
            g = res.x
            sum_g = g.sum()
            sum_a = a.sum()
            # no improvement possible?
            if sum_g <= sum_a * (1.0 + 1e-12):
                break
            # scale to keep the sum constant
            g_scaled = g * (sum_a / sum_g)
            # convex combination
            a = (1.0 - t) * a + t * g_scaled
            a = project_to_simplex(a)
        return a
else:  # pragma: no cover
    # Dummy placeholder when SciPy is unavailable
    def lp_iterative_refine(a, deadline, steps=3, t=0.5):
        return a


# ----------------------------------------------------------------------
# Main entry point required by the competition harness
# ----------------------------------------------------------------------


def propose_candidate(
    seed: int = 42,
    budget_s: int = 1000,
    **kwargs,
) -> list[float]:
    """
    Search for a non‑negative step sequence with a low official score.
    The routine combines a broad random sampling of analytically‑motivated
    families, heavy local optimisation (Adam → sub‑gradient → random walk)
    and, when SciPy is available, a small LP‑based polishing step.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 1.0  # small safety margin

    best_seq: list[float] = []
    best_val: float = float("inf")

    # ------------------------------------------------------------------
    # 0.  Existing fallback (if provided by the surrounding harness)
    # ------------------------------------------------------------------
    try:
        fallback = [float(x) for x in height_sequence_1]  # type: ignore
        val = evaluate_sequence(fallback)
        print(f"Fallback (height_sequence_1) official score = {val:.6f}")
        if val < best_val:
            best_val, best_seq = val, fallback
    except Exception:
        pass

    # ------------------------------------------------------------------
    # 1.  Broad random sampling over several lengths
    # ------------------------------------------------------------------
    lengths = [1024, 2048, 4096, 8192, 16384]

    for n in lengths:
        if time.time() >= deadline - 5.0:
            break
        print(f"\n--- Sampling length {n} ---")
        # number of random candidates – larger n gets a smaller budget
        base_samples = 8000
        factor = max(0.2, 2048 / n)  # 1 for 2048, smaller for larger n
        max_samples = int(base_samples * factor * (budget_s / 1000.0))
        max_samples = max(2000, max_samples)

        # keep a local best
        local_best = None
        local_best_score = float("inf")

        # deterministic starters
        starters = [
            np.full(n, 1.0 / n),          # uniform
            arcsine_initial(n),           # classic arcsine (ε = 0.25)
        ]
        for st in starters:
            sc = compute_score(st)
            if sc < local_best_score:
                local_best_score = sc
                local_best = st.copy()

        # randomised families
        for _ in range(max_samples):
            if time.time() >= deadline - 5.0:
                break
            r = rng.random()
            if r < 0.2:
                # power‑law family
                alpha = rng.uniform(0.2, 1.2)
                eps = 10 ** rng.uniform(-6, 2)   # 1e‑6 … 1e2
                cand = power_law_shape(n, alpha, eps)
            elif r < 0.4:
                # power‑law mixed with uniform
                alpha = rng.uniform(0.2, 1.2)
                eps = 10 ** rng.uniform(-6, 2)
                w = rng.random()
                base = power_law_shape(n, alpha, eps)
                cand = w * base + (1.0 - w) * (1.0 / n)
                cand /= cand.sum()
            elif r < 0.55:
                # arcsine with tunable ε
                eps = 10 ** rng.uniform(-6, 2)
                cand = arcsine_eps_shape(n, eps)
            elif r < 0.75:
                # single beta density
                alpha = rng.uniform(0.02, 3.0)
                beta = rng.uniform(0.02, 3.0)
                eps = rng.uniform(1e-8, max(0.01, 0.02 * n))
                cand = generate_beta_shape(n, alpha, beta, eps)
            else:
                # mixture of two betas
                w = rng.random()
                alpha1 = rng.uniform(0.02, 3.0)
                beta1 = rng.uniform(0.02, 3.0)
                eps1 = rng.uniform(1e-8, max(0.01, 0.02 * n))
                alpha2 = rng.uniform(0.02, 3.0)
                beta2 = rng.uniform(0.02, 3.0)
                eps2 = rng.uniform(1e-8, max(0.01, 0.02 * n))
                cand = generate_beta_mixture(
                    n, alpha1, beta1, eps1, alpha2, beta2, eps2, w
                )

            sc = compute_score(cand)
            if sc < local_best_score:
                local_best_score = sc
                local_best = cand.copy()

        # evaluate the best raw candidate of this length (official score)
        if local_best is not None:
            val = evaluate_sequence(local_best.tolist())
            print(f"Length {n} raw best official score = {val:.6f}")
            if val < best_val:
                best_val, best_seq = val, local_best.tolist()
                print(f"*** New global best (len={n}) = {best_val:.6f} ***")

    # ------------------------------------------------------------------
    # 2.  Heavy refinement of the overall best candidate
    # ------------------------------------------------------------------
    if best_seq and time.time() < deadline - 12.0:
        print("\n=== Polishing the globally best sequence ===")
        a = np.array(best_seq, dtype=float)

        # Adam stage
        remaining = deadline - time.time()
        max_iter_adam = int(min(350_000, remaining * 30))
        a, _ = adam_optimize(
            initial_a=a,
            time_deadline=deadline,
            max_iter=max_iter_adam,
            lr0=0.10,
            lam_start=1.0,
            lam_end=250.0,
            noise_every=1300,
            noise_scale=4e-5,
        )

        # Sub‑gradient stage
        if time.time() < deadline - 6.0:
            max_iter_sub = int(min(180_000, (deadline - time.time()) * 30))
            a, _ = subgradient_optimize(
                a,
                time_deadline=deadline,
                step0=0.20,
                max_iter=max_iter_sub,
            )

        # Random walk stage
        if time.time() < deadline - 4.0:
            max_iter_loc = int(min(250_000, (deadline - time.time()) * 35))
            a, _ = random_local_search(
                a,
                time_deadline=deadline,
                init_step=0.012,
                max_iter=max_iter_loc,
            )

        # Optional LP‑based polishing (only for moderate sizes)
        if SCIPY_AVAILABLE and len(a) <= 4096 and time.time() < deadline - 2.0:
            print("LP‑based refinement …")
            a = lp_iterative_refine(a, deadline, steps=3, t=0.5)

        final_val = evaluate_sequence(a.tolist())
        print(f"Final polished official score = {final_val:.6f}")
        if final_val < best_val:
            best_val, best_seq = final_val, a.tolist()
            print("*** Updated global best after polishing ***")

    # ------------------------------------------------------------------
    # 3.  Guard – guarantee a non‑empty return
    # ------------------------------------------------------------------
    if not best_seq:
        best_seq = [1.0] * 256
        best_val = evaluate_sequence(best_seq)
        print("Fallback to uniform 256‑vector.", best_val)

    print(f"\nSearch finished. Best reported official score = {best_val:.6f}")
    return best_seq
