"""
Improved step‑function search.

The algorithm works in three phases:

1. **Exploration (small size)** – generate a diverse set of candidate
   vectors (arcsine, uniform, random Dirichlet, power‑law families and
   the previously found construction if available).  All candidates are
   normalised to the simplex and evaluated with a fast FFT‑based score.
   The best candidate is kept.

2. **Annealed smooth optimisation** – the best vector is up‑sampled to a
   larger dimension (4096 points by default).  A smooth approximation of
   the max‑convolution is minimised by an Adam‑style exponentiated‑gradient
   scheme.  The “temperature’’ of the soft‑max (β) is increased linearly
   from a low value to a high one, providing a continuation method that
   helps escape shallow local minima.

3. **Hard sub‑gradient polishing** – the result of phase 2 is further
   refined using the true max‑convolution (hard weights) with the same
   Adam‑based multiplicative updates.  The best iterate seen (according
   to the official `evaluate_sequence` routine) is finally returned.

All updates stay on the probability simplex (sum = 1) and all numbers
remain non‑negative, satisfying the security checks of the evaluator.
The routine respects the given time budget (default 1000 s) and always
returns a feasible sequence.
"""

import time
import numpy as np
from typing import List, Tuple

# ----------------------------------------------------------------------
# Low‑level numeric utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """
    Linear convolution using a real FFT (full length, zero‑padded to a
    power‑of‑2).  Returns the linear convolution of a and b.
    """
    m = len(a) + len(b) - 1
    n_fft = next_pow2(m)
    A = np.fft.rfft(a, n=n_fft)
    B = np.fft.rfft(b, n=n_fft)
    C = A * B
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


def normalize(v: np.ndarray) -> np.ndarray:
    """Project a non‑negative vector onto the simplex (sum = 1)."""
    total = v.sum()
    if total <= 0.0:
        # fallback to uniform distribution
        return np.full_like(v, 1.0 / len(v))
    return v / total


def compute_score(a: np.ndarray) -> float:
    """
    Fast evaluation of the official objective (scale‑invariant):
        2·n·max(conv(a,a)) / (sum(a)²)
    The vector a is assumed non‑negative and normalised to sum = 1.
    """
    n = len(a)
    conv = convolution_fft(a, a)
    max_val = float(np.max(conv))
    if max_val < 0:
        return float("inf")
    return 2.0 * n * max_val  # because sum(a) == 1


def gradient_from_weights(a: np.ndarray, w: np.ndarray) -> np.ndarray:
    """
    Gradient of Σ_k w_k·(a*a)_k w.r.t. a.
    Implements   g_j = 2· Σ_i a_i·w_{i+j}.
    Computed via one FFT‑based convolution.
    """
    # reverse w for convolution trick
    w_rev = w[::-1]
    conv = convolution_fft(a, w_rev)               # length = 3·n‑2
    idx = 2 * len(a) - 2
    grads = 2.0 * conv[idx - np.arange(len(a))]
    return grads


# ----------------------------------------------------------------------
# Parametric families for the initial sweep
# ----------------------------------------------------------------------


def arcsine_initial(n: int) -> np.ndarray:
    """Discrete arcsine distribution ≈ 1/√(i·(n‑1‑i))."""
    i = np.arange(n, dtype=float)
    eps = 0.25
    vals = 1.0 / np.sqrt((i + 0.5) * (n - 0.5 - i) + eps)
    return normalize(vals)


def powerlaw_shape(n: int, p: float, eps: float) -> np.ndarray:
    """Symmetric power‑law a_i ∝ ((i+ε)(n‑1‑i+ε))⁻ᵖ."""
    i = np.arange(n, dtype=float)
    base = ((i + eps) * (n - 1 - i + eps)) ** (-p)
    return normalize(base)


def uniform_shape(n: int) -> np.ndarray:
    """Uniform probability vector on n points."""
    return np.full(n, 1.0 / n)


def random_dirichlet(n: int, rng: np.random.Generator) -> np.ndarray:
    """Sample a random point on the simplex (exponential method)."""
    samp = rng.exponential(1.0, size=n)
    return normalize(samp)


def upsample(a_small: np.ndarray, n_new: int) -> np.ndarray:
    """
    Linear interpolation from a_small (size m) to a vector of length n_new,
    followed by simplex normalisation.
    """
    if len(a_small) == n_new:
        return a_small.copy()
    x_old = np.linspace(0.0, 1.0, len(a_small))
    x_new = np.linspace(0.0, 1.0, n_new)
    a_new = np.interp(x_new, x_old, a_small)
    a_new = np.maximum(a_new, 0.0)
    return normalize(a_new)

def refine_annealed_adam(
    a: np.ndarray,
    deadline: float,
    *,
    beta_start: float = 2.0,
    beta_end: float = 30.0,
    eta0: float = 0.5,
    beta1: float = 0.9,
    beta2: float = 0.999,
    eps: float = 1e-8,
    report_every: int = 0,
) -> Tuple[np.ndarray, float]:
    """
    Annealed, smoothed optimisation using an Adam‑style exponentiated
    gradient.  The softmax temperature β is increased linearly from
    ``beta_start`` to ``beta_end`` during the allotted time.
    Returns the best iterate (according to the exact max‑convolution
    objective) and its fast internal score.
    """
    a = normalize(a.astype(float).copy())
    n = len(a)

    # Adam state
    m = np.zeros_like(a)
    v = np.zeros_like(a)

    best_a = a.copy()
    best_sc = compute_score(a)

    it = 0
    start = time.time()
    total_time = max(deadline - start, 1e-9)

    while time.time() < deadline:
        it += 1
        # linear schedule for beta
        elapsed = time.time() - start
        beta = beta_start + (beta_end - beta_start) * min(1.0, elapsed / total_time)

        # current convolution
        conv = convolution_fft(a, a)
        max_val = float(np.max(conv))

        # softmax‑like weights (smooth approx. of the max)
        exp_vals = np.exp(beta * (conv - max_val))
        w = exp_vals / np.sum(exp_vals)

        # gradient of the smoothed objective
        grad = gradient_from_weights(a, w)

        # Adam moments
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)

        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        # step size schedule
        eta = eta0 / np.sqrt(it)

        # Adam‑scaled update (multiplicative)
        step = eta * m_hat / (np.sqrt(v_hat) + eps)

        a = a * np.exp(-step)
        a = normalize(a)

        cur_sc = compute_score(a)
        if cur_sc < best_sc:
            best_sc = cur_sc
            best_a = a.copy()

        if report_every and (it % report_every == 0):
            print(
                f"    annealed iter={it:,d} beta={beta:.2f} cur={cur_sc:.6f} best={best_sc:.6f}"
            )

    return best_a, best_sc


def refine_hard_adam(
    a: np.ndarray,
    deadline: float,
    *,
    eta0: float = 0.7,
    beta1: float = 0.9,
    beta2: float = 0.999,
    max_iter: int = 200_000,
    eps: float = 1e-8,
    report_every: int = 0,
) -> Tuple[np.ndarray, float]:
    """
    Classical sub‑gradient polishing using the true max‑convolution
    (hard weights) combined with Adam‑style exponentiated updates.
    Returns the best vector and its *official* score (via
    ``evaluate_sequence``).
    """
    a = normalize(a.astype(float).copy())
    n = len(a)

    # Adam state
    m = np.zeros_like(a)
    v = np.zeros_like(a)

    best_a = a.copy()
    best_sc = compute_score(a)

    for it in range(1, max_iter + 1):
        if time.time() >= deadline:
            break

        conv = convolution_fft(a, a)
        max_val = float(np.max(conv))

        # uniform weight on (almost) maximisers
        tol = 1e-14
        max_idx = np.where(conv >= max_val - tol)[0]
        w = np.zeros_like(conv)
        if max_idx.size > 0:

        grad = gradient_from_weights(a, w)

        # Adam moments
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)

        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        eta = eta0 / np.sqrt(it)
        step = eta * m_hat / (np.sqrt(v_hat) + eps)

        a = a * np.exp(-step)
        a = normalize(a)

        cur_sc = compute_score(a)
        if cur_sc < best_sc:

            best_a = a.copy()

        if report_every and (it % report_every == 0):
            print(f"    hard sub‑grad iter={it:,d} cur={cur_sc:.6f} best={best_sc:.6f}")

    # final official evaluation
    final_score = evaluate_sequence(best_a.tolist())
    return best_a, final_score
