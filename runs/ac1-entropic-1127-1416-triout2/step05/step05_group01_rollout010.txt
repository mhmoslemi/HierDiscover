assistantfinal**Solution Overview**

The score of a step‑function  

\[
S(a)=\frac{2\,n\;\max (a*a)}{(\sum a)^2}
\]

is *scale invariant*: multiplying the whole vector by a constant does **not**
change the score.  Hence we can always keep the vector on the probability
simplex (`∑a = 1`) and only minimise  

\[
\max (a*a) .
\]

For a probability distribution `a` the convolution `b = a*a` is exactly the
distribution of the sum of two independent draws from `a`.  
The theoretical lower bound is obtained when `b` is (almost) uniform, i.e.  

\[
\max b \approx \frac{1}{2n-1}\qquad\Longrightarrow\qquad S_{\min}\approx 1 .
\]

The programme below searches for a distribution whose self‑convolution is as
flat as possible.  It consists of three stages  

1. **Initialisation** – several diverse starting points (uniform, U‑shaped,
   random Dirichlet, a saved fallback) are evaluated and the best one is kept.
2. **Two optimisation phases**  
   *Phase A – L²‑error minimisation* – we minimise  
   \[
   J(a)=\|a*a-u\|_2^2 , \qquad u_k=\frac1{2n-1}
   \]
   by exponentiated gradient descent.  The gradient of `J` can be written as
   two FFT‑based convolutions, therefore every iteration is cheap (`O(n log n)`).
   *Phase B – direct max‑minimisation* – after the error has been reduced we
   switch to the classic sub‑gradient update for the true objective
   `max(a*a)`.  This further pushes the highest peak down.
3. **Polishing** – a simple stochastic hill‑climber (swap / mass transfer) that
   accepts only improving moves, finally projected back onto the simplex.

All work respects the 1000 s budget.  The best vector found during the run is
returned; `evaluate_sequence` will compute the exact score.

```python
"""
Search for a non‑negative step function with a small
max‑convolution (equivalently a small value of
evaluate_sequence).  The algorithm works in three phases:

1.  Initialise with several diverse candidates (fallback, uniform,
    U‑shaped, random Dirichlet) and keep the best.
2.  Optimise first against an L2‑error to a flat target distribution,
    then directly against the max‑convolution using a sub‑gradient.
    Both phases use exponentiated updates and an exact projection onto
    the probability simplex (∑a = 1).  FFT‑based convolutions make each
    iteration fast.
3.  Finish with a short stochastic hill‑climber (swap / mass‑transfer
    moves) that only accepts improvements.

The whole search stops before the 1000 s external timeout and returns the
best sequence found.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# FFT‑based convolution utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """
    Linear convolution a * b using real FFT.
    Returns a vector of length len(a)+len(b)-1.
    """
    m = len(a) + len(b) - 1
    n_fft = next_pow2(m)
    A = np.fft.rfft(a, n=n_fft)
    B = np.fft.rfft(b, n=n_fft)
    C = A * B
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


# ----------------------------------------------------------------------
# Simple utilities
# ----------------------------------------------------------------------


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the probability simplex
    (non‑negative entries that sum to 1).
    """
    if np.all(v <= 0):
        # all non‑positive – return the uniform distribution
        return np.full_like(v, 1.0 / v.shape[0])

    # sort descending
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho = np.where(u - cssv / np.arange(1, len(v) + 1) > 0)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def compute_full_score(a: np.ndarray) -> float:
    """
    Exact replica of ``evaluate_sequence`` for a generic vector a.
    Returns 2·n·max(a*a)/(sum(a)**2).
    """
    n = a.shape[0]
    conv = convolution_fft(a, a)
    max_b = np.max(conv)
    s = np.sum(a)
    if s < 1e-12:
        return float("inf")
    return float(2.0 * n * max_b / (s * s))


def compute_score_simplex(a: np.ndarray) -> float:
    """
    Fast score when the vector already lies on the simplex (∑a = 1).
    Returns 2·n·max(a*a).
    """
    n = a.shape[0]
    conv = convolution_fft(a, a)
    return float(2.0 * n * np.max(conv))


# ----------------------------------------------------------------------
# Phase A – minimise L2 error to a flat target distribution
# ----------------------------------------------------------------------


def l2_error_optimize(
    init_a: np.ndarray,
    target: np.ndarray,
    deadline: float,
    eta0: float = 0.5,
    max_iter: int = 10_000_000,
) -> tuple[np.ndarray, float]:
    """
    Exponentiated gradient descent on J(a)=||a*a - target||_2^2.
    Returns the best vector seen and its (max‑convolution) score.
    """
    a = init_a.copy()
    n = a.shape[0]
    best_a = a.copy()
    best_score = compute_score_simplex(best_a)

    # pre‑allocate buffers to avoid repeated allocations
    for t in range(1, max_iter + 1):
        if time.time() >= deadline:
            break

        # b = a * a
        b = convolution_fft(a, a)

        # error = b - target
        e = b - target

        # gradient = 2 * (e * a_rev + e_rev * a)   (restricted to the n coordinates)
        a_rev = a[::-1]
        # first term
        g1 = convolution_fft(e, a_rev)[n - 1 : n - 1 + n]
        # second term (using the reversed error)
        e_rev = e[::-1]
        g2 = convolution_fft(e_rev, a)[n - 1 : n - 1 + n]
        g = 2.0 * (g1 + g2)

        # step size (decreasing)
        eta = eta0 / math.sqrt(t)

        # multiplicative update + projection onto the simplex
        a = a * np.exp(-eta * g)
        a = project_to_simplex(a)

        # occasional bookkeeping
        if t % 2000 == 0:
            cur_score = compute_score_simplex(a)
            if cur_score < best_score:
                best_score = cur_score
                best_a = a.copy()

    return best_a, best_score


# ----------------------------------------------------------------------
# Phase B – direct sub‑gradient descent on max(a*a)
# ----------------------------------------------------------------------


def max_subgrad_optimize(
    init_a: np.ndarray,
    deadline: float,
    eta0: float = 0.5,
    max_iter: int = 5_000_000,
) -> tuple[np.ndarray, float]:
    """
    Exponentiated sub‑gradient descent on the true objective
    2·n·max(a*a) (the vector stays on the simplex).
    """
    a = init_a.copy()
    n = a.shape[0]
    best_a = a.copy()
    best_score = compute_score_simplex(a)

    for t in range(1, max_iter + 1):
        if time.time() >= deadline:
            break

        conv = convolution_fft(a, a)
        k = int(np.argmax(conv))        # index of the current maximal shift

        # sub‑gradient for the maximal shift
        idx = np.arange(n)
        j = k - idx
        mask = (j >= 0) & (j < n)
        g = np.zeros_like(a)
        g[mask] = 2.0 * a[j[mask]]

        eta = eta0 / math.sqrt(t)
        a = a * np.exp(-eta * g)
        a = project_to_simplex(a)

        # bookkeeping
        cur_score = compute_score_simplex(a)
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()

        # tiny random jitter every few thousand steps to escape plateaus
        if t % 5000 == 0:
            a += np.random.randn(n) * (eta0 * 1e-4 / math.sqrt(t))
            a = project_to_simplex(a)

    return best_a, best_score


# ----------------------------------------------------------------------
# Phase C – stochastic hill‑climber (swap / mass transfer)
# ----------------------------------------------------------------------


def hill_climb(
    seq,
    deadline: float,
    max_steps: int = 300_000,
) -> tuple[list[float], float]:
    """
    Simple hill‑climber that swaps two entries or transfers a fraction of
    mass from one position to another.  Every candidate is projected onto the
    simplex and accepted only if it improves the true score.
    """
    a = np.array(seq, dtype=float)
    n = a.shape[0]

    best_a = a.copy()
    best_score = compute_full_score(best_a)

    # step size controls the fraction of transferred mass
    step_scale = 0.08

    for _ in range(max_steps):
        if time.time() >= deadline:
            break

        cand = best_a.copy()
        if np.random.rand() < 0.5:          # swap
            i, j = np.random.randint(n, size=2)
            cand[i], cand[j] = cand[j], cand[i]
        else:                               # transfer
            i, j = np.random.randint(n, size=2)
            if i != j:
                delta = step_scale * cand[i]
                cand[i] -= delta
                cand[j] += delta
                cand[i] = max(cand[i], 0.0)

        cand = project_to_simplex(cand)
        cur_score = compute_full_score(cand)

        if cur_score < best_score:
            best_score = cur_score
            best_a = cand
            step_scale = min(step_scale * 1.05, 0.5)   # successful → bigger moves
        else:
            step_scale *= 0.97                         # otherwise shrink

    return best_a.tolist(), best_score


# ----------------------------------------------------------------------
# Helper to build a simple U‑shaped start vector
# ----------------------------------------------------------------------


def u_shaped_start(n: int, power: float = 0.7) -> np.ndarray:
    """
    Returns a distribution that is larger at the two ends and smaller in the
    centre:   a[i] ∝ ((i+1)*(n-i))^{-power}.
    """
    idx = np.arange(n)
    w = ((idx + 1) * (n - idx)) ** (-power)
    w = w / w.sum()
    return w
```
