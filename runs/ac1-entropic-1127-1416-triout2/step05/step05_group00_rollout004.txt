import scipy.sparse as sp
def solve_lp_for_b(a, M):
    n = len(a)
    # Build A_ub (2n-1) x n sparse matrix
    rows = []
    cols = []
    data = []
    for shift in range(2*n-1):
        # row index shift
        # for each i in [0,n-1] such that j = shift - i in [0,n-1] => i in max(0, shift-(n-1))..min(n-1, shift)
        i_min = max(0, shift-(n-1))
        i_max = min(n-1, shift)
        for i in range(i_min, i_max+1):
            # a_i * b_{shift-i}
            rows.append(shift)
            cols.append(shift - i)
            data.append(a[i])
    A_ub = sp.csr_matrix((data, (rows, cols)), shape=(2*n-1, n))
    # RHS
    b_ub = np.full(2*n-1, M)
    # objective c = -1 (since maximize sum b)
    c = -np.ones(n)
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=[(0, None)]*n, method='highs')
    if not res.success:
        return None
    return res.x

"""
Propose a low‑score step function.

The evaluation metric is
    score = 2·n·max(conv(a,a)) / (sum(a))**2
which is invariant to scaling, therefore we restrict ourselves to the
probability simplex (a ≥ 0, sum(a)=1).  The goal is to minimise the peak of
the autocorrelation (full convolution).

Algorithm
---------
* Initialise several candidates:
    – use the global fallback `height_sequence_1` when available,
    – a few random sparse vectors of different lengths.
* For each candidate run a multi‑stage projected gradient descent on a
  smooth approximation of the max‑operator:
        smooth_max_α(x)= (1/α)·log Σ exp(α·x_i)
  The gradient of the smooth objective can be written analytically as a
  (scaled) cross‑correlation, which is evaluated with an FFT‑based
  convolution (O(n log n)).
  The schedule gradually increases α (more smoothing → easier global
  moves, larger α → sharper focus on the true maximum) and uses a
  decreasing step‑size.  After each update the iterate is projected back
  onto the simplex.
* Keep the best refined vector (lowest true score).
* Finish with a short stochastic hill‑climb that swaps entries or moves a
  small fraction of mass between positions; each move is kept only if it
  improves the exact evaluation.
* Return the best sequence found (as a plain Python list of floats).

All helper functions are defined at the top level; no closures are used.
The routine respects a 1000 s wall‑time budget.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Basic linear‑algebra utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """
    Full convolution using a real FFT (fast, O(n log n)).
    Returned length = len(a) + len(b) - 1 .
    """
    m = a.shape[0] + b.shape[0] - 1
    n_fft = next_pow2(m)
    A = np.fft.rfft(a, n=n_fft)
    B = np.fft.rfft(b, n=n_fft)
    C = A * B
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


def compute_score(a: np.ndarray) -> float:
    """
    Fast replica of `evaluate_sequence` for a vector that already sums to 1.
    Returns 2·n·max(a⊛a).
    """
    n = a.shape[0]
    conv = convolution_fft(a, a)
    return 2.0 * n * float(np.max(conv))


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the probability simplex
    (non‑negative entries that sum to 1).  Implements the algorithm of
    Wang & Carreira‑Perpiñán (2013).
    """
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / v.shape[0])

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, v.shape[0] + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.where(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


# ----------------------------------------------------------------------
# Smooth max and its gradient (exact, via cross‑correlation)
# ----------------------------------------------------------------------


def smooth_max_and_gradient(a: np.ndarray, alpha: float):
    """
    Returns
        s   = smooth approximation of max(conv(a,a))  (scalar)
        g   = gradient of s w.r.t. a                (numpy array)
    The gradient formula is:
        ∇s = α · 2 · corr(w, a)
    where w_i = softmax(α·conv_i) and corr(w, a)[t] = Σ_i w_i a_{i‑t}.
    """
    n = a.shape[0]

    # 1) full autocorrelation
    conv = convolution_fft(a, a)                 # length 2n‑1
    m = conv.shape[0]                            # = 2n‑1
    max_val = np.max(conv)

    # 2) softmax weights (stable via subtraction of max)
    shifted = conv - max_val                     # ≤ 0
    exp_alpha = np.exp(alpha * shifted)          # max element = 1
    sum_exp = np.sum(exp_alpha)
    w = exp_alpha / sum_exp                      # sums to 1

    # 3) cross‑correlation w ⋆ a  (equivalently convolution of w_rev with a)
    w_rev = w[::-1]
    corr = convolution_fft(w_rev, a)            # length 3n‑2
    # extract the n entries needed for the gradient
    # corr index = m-1 - t   (t = 0 … n‑1)
    g0 = 2.0 * corr[m - 1 - np.arange(n)]

    # 4) smooth max value
    smooth_max = max_val + np.log(sum_exp) / alpha

    # 5) gradient of the smooth max (α·g0)
    grad = alpha * g0

    return smooth_max, grad


# ----------------------------------------------------------------------
# Projected gradient descent on the smooth objective
# ----------------------------------------------------------------------


def refine_candidate(a0: np.ndarray, deadline: float) -> tuple[np.ndarray, float]:
    """
    Multi‑stage projected gradient descent.
    Starts from a0 (already on the simplex) and returns a refined vector
    and its exact score (via `compute_score`).  The procedure stops early
    if the wall‑time budget is exhausted.
    """
    a = a0.copy()
    best = a.copy()
    best_score = compute_score(best)

    # schedule of smoothing parameters (small → large)
    alphas = [0.5, 2.0, 5.0, 10.0, 20.0, 40.0]
    max_iters_per_alpha = 3000           # enough for convergence in practice
    step0 = 0.4                          # base step size

    n = a.shape[0]

    for alpha in alphas:
        for it in range(max_iters_per_alpha):
            if time.time() >= deadline:
                return best, best_score

            # gradient of the smooth max
            smax, grad_smax = smooth_max_and_gradient(a, alpha)

            # full objective = 2·n·smax, gradient = 2·n·grad_smax
            grad = 2.0 * n * grad_smax

            # step size decays with sqrt(iter) and with α (large α → smaller steps)
            step = step0 / (alpha * np.sqrt(it + 1) + 1e-12)

            a_new = a - step * grad
            a_new = project_to_simplex(a_new)

            cur_score = compute_score(a_new)
            if cur_score < best_score:
                best_score = cur_score
                best = a_new.copy()
                # keep the improved iterate as the base for further steps
                a = a_new.copy()
            else:
                # even if not improved, continue from the candidate (allows escape)
                a = a_new

    return best, best_score


# ----------------------------------------------------------------------
# Simple stochastic hill‑climb (swap / mass‑transfer moves)
# ----------------------------------------------------------------------


def hill_climb_permutation(seq, time_deadline, max_steps=200_000):
    """
    Stochastic hill‑climb that swaps two positions or transfers a fraction
    of mass from one entry to another.  Every candidate is projected onto
    the simplex.  The move is kept only if the true score improves.
    """
    a = np.array(seq, dtype=float)
    n = a.shape[0]

    best_a = a.copy()
    best_score = compute_score(best_a)

    step_scale = 0.12   # initial fraction of mass moved in a transfer

    while max_steps > 0 and time.time() < time_deadline:
        move_type = np.random.rand()
        cand = best_a.copy()

        if move_type < 0.5:
            # swap two positions
            i, j = np.random.randint(n, size=2)
            cand[i], cand[j] = cand[j], cand[i]
        else:
            # transfer a fraction from i to j
            i, j = np.random.randint(n, size=2)
            if i != j:
                delta = step_scale * cand[i]
                cand[i] -= delta
                cand[j] += delta
                # keep non‑negativity
                if cand[i] < 0:
                    cand[i] = 0.0

        cand = project_to_simplex(cand)
        cur_score = compute_score(cand)

        if cur_score < best_score:
            best_score = cur_score
            best_a = cand
            step_scale = min(step_scale * 1.07, 0.5)
        else:
            step_scale *= 0.96

        max_steps -= 1

    return best_a.tolist(), best_score
