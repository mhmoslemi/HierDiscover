def greedy_mass_transfer(a0, deadline, rng, max_iter, step_frac_start=0.1, step_frac_decay=0.99, tol=1e-12):
    a = a0.copy()
    best_a = a.copy()
    best_sc = compute_score(a)
    for it in range(max_iter):
        if time.time() >= deadline:
            break
        conv = convolution_fft(a, a)
        max_c = conv.max()
        # gradient subgradient
        grad = _max_conv_gradient(a, conv)  # shape n
        # Determine indices
        i = np.argmax(grad)  # high gradient
        j = np.argmin(grad)
        if i == j:
            # choose random pair
            i, j = rng.integers(0, len(a), size=2)
            if i==j: continue
        # amount to move: delta = step_frac * a[i]
        step_frac = step_frac_start * (step_frac_decay ** it)
        delta = a[i] * step_frac
        if delta <= 0:
            continue
        a[i] -= delta
        a[j] += delta
        # Normalize? sum constant.
        # Ensure nonnegativity
        if a[i] < 0:
            a[j] += a[i]  # add negative part back, ensuring a[i]=0
            a[i] = 0.0
        cur_sc = compute_score(a)
        if cur_sc < best_sc:
            best_sc = cur_sc
            best_a = a.copy()
    return best_a, best_sc

def _max_conv_gradient(a, conv):
    n = len(a)
    max_val = conv.max()
    mask = np.isclose(conv, max_val, atol=1e-15, rtol=0.0)
    active = np.where(mask)[0]
    grad = np.zeros_like(a)
    for k in active:
        lo = max(0, k - (n - 1))
        hi = min(k, n - 1)
        idx = np.arange(lo, hi + 1, dtype=int)
        grad[idx] += 2.0 * a[k - idx]
    if len(active) > 1:
        grad /= len(active)
    return grad

"""
Hybrid search for a low‑max‑convolution step function (improved).

The algorithm proceeds in three phases:

1. **Diverse parametric seeds** – a rich library of analytic shapes
   (arcsine, power‑law, beta, sinusoidal, Hann‑window, triangular,
   Dirichlet draws, mixtures).  For each candidate length we evaluate a
   cheap “compute_score’’ (the scale‑invariant objective) and keep the
   best few.

2. **Local refinement** – each retained seed is polished by a cascade
   of cheap optimisers:
        • Simulated‑annealing (mass‑transfer SA on the simplex).
        • Smooth‑max (log‑sum‑exp) gradient descent with an aggressive
          β‑schedule.
        • Greedy pairwise mass transfer driven by the exact sub‑gradient
          of the max‑convolution.
        • Exponentiated sub‑gradient descent on the true max‑convolution
          (final polish).

   All updates respect non‑negativity and keep the total mass constant,
   so the ratio 2·n·max / sum² is unchanged by scaling.

3. **Global bookkeeping** – after each improvement the sequence is
   evaluated with the official ``evaluate_sequence`` function; the best
   ever seen is retained.  The routine stops well before the time budget
   expires and always returns a feasible list of floats.

The search explores several lengths (powers of two up to 8192).  Longer
vectors give a tighter asymptotic bound but cost more per iteration; the
code adapts the number of iterations to the remaining time.
"""

import time
import math
import numpy as np
from typing import List, Tuple

# ----------------------------------------------------------------------
# Numerical utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full linear convolution using real FFT.  Returns length len(a)+len(b)-1."""
    m = len(a) + len(b) - 1
    n_fft = next_pow2(m)
    A = np.fft.rfft(a, n=n_fft)
    B = np.fft.rfft(b, n=n_fft)
    C = A * B
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


def compute_score(a: np.ndarray) -> float:
    """
    Fast version of the official objective (scale‑invariant):
        2·n·max(a∗a) / (sum(a)²)
    The vector a is assumed non‑negative.
    """
    n = len(a)
    conv = convolution_fft(a, a)
    max_val = float(conv.max())
    s = float(a.sum())
    if s < 1e-12:
        return float("inf")
    return 2.0 * n * max_val / (s * s)


# ----------------------------------------------------------------------
# Families of analytic shapes (all return a probability vector)
# ----------------------------------------------------------------------


def arcsine_initial(n: int) -> np.ndarray:
    """Discrete arcsine density (∝ 1/√[i·(n‑i)])."""
    i = np.arange(n, dtype=float)
    eps = 0.25
    vals = 1.0 / np.sqrt((i + 0.5) * (n - 0.5 - i) + eps)
    vals /= vals.sum()
    return vals


def generate_powerlaw(n: int, p: float, eps: float = 1e-6) -> np.ndarray:
    """Power‑law a_i ∝ ((i+ε)(n‑1‑i+ε))⁻ᵖ."""
    i = np.arange(n, dtype=float)
    base = ((i + eps) * (n - 1 - i + eps)) ** (-p)
    base_sum = base.sum()
    if base_sum == 0.0:
        return np.full(n, 1.0 / n)
    return base / base_sum


def generate_beta_like(n: int, alpha: float, beta: float) -> np.ndarray:
    """Beta‑like shape a_i ∝ (i+½)^{α‑1} (n‑1‑i+½)^{β‑1}."""
    i = np.arange(n, dtype=float)
    base = (i + 0.5) ** (alpha - 1) * (n - 1 - i + 0.5) ** (beta - 1)
    total = base.sum()
    if total == 0.0:
        return np.full(n, 1.0 / n)
    return base / total


def sine_initial(n: int) -> np.ndarray:
    """Half‑sine shape: sin(π·(i+½)/n)."""
    i = np.arange(n, dtype=float) + 0.5
    vals = np.sin(np.pi * i / n)
    vals /= vals.sum()
    return vals


def hann_initial(n: int) -> np.ndarray:
    """Hann window (raised cosine)."""
    i = np.arange(n, dtype=float)
    vals = 0.5 - 0.5 * np.cos(2.0 * np.pi * i / (n - 1))
    vals /= vals.sum()
    return vals


def triangular_initial(n: int) -> np.ndarray:
    """Symmetric triangular shape, zero at the ends."""
    i = np.arange(n, dtype=float)
    centre = (n - 1) / 2.0
    vals = 1.0 - np.abs(i - centre) / (centre + 1.0)
    vals = np.maximum(vals, 0.0)
    vals /= vals.sum()
    return vals


def random_dirichlet(n: int, concentration: float, rng: np.random.Generator) -> np.ndarray:
    """Sample a Dirichlet‑type vector and normalise."""
    if concentration <= 0.0:
        samp = rng.exponential(1.0, size=n)
    else:
        samp = rng.gamma(concentration, 1.0, size=n)
    samp = np.maximum(samp, 1e-12)
    return samp / samp.sum()


def perturb(a: np.ndarray, rng: np.random.Generator, scale: float = 0.02) -> np.ndarray:
    """Add small Gaussian noise and renormalise."""
    b = a + rng.normal(0.0, scale, size=a.shape)
    b = np.maximum(b, 0.0)
    s = b.sum()
    if s == 0.0:
        return np.full_like(a, 1.0 / len(a))
    return b / s


# ----------------------------------------------------------------------
# Simulated annealing (mass‑transfer SA)
# ----------------------------------------------------------------------


def propose_neighbor(a: np.ndarray, rng: np.random.Generator, max_step: float = 0.02) -> np.ndarray:
    """Move a small amount of mass from one random bin to another."""
    n = len(a)
    i, j = rng.integers(0, n, size=2)
    if i == j or a[i] == 0.0:
        return a
    delta = a[i] * rng.random() * max_step
    b = a.copy()
    b[i] -= delta
    b[j] += delta
    return b


def simulated_annealing(a0: np.ndarray,
                        deadline: float,
                        rng: np.random.Generator,
                        max_iter: int = 2000,
                        start_temp: float = 0.01,
                        cooling: float = 0.9995,
                        max_step: float = 0.02) -> Tuple[np.ndarray, float]:
    """Very light SA that respects the simplex."""
    a = a0.copy()
    cur_sc = compute_score(a)
    best_a = a.copy()
    best_sc = cur_sc
    temp = start_temp

    for it in range(max_iter):
        if time.time() >= deadline:
            break
        cand = propose_neighbor(a, rng, max_step)
        cand_sc = compute_score(cand)
        delta = cand_sc - cur_sc
        if delta < 0.0 or rng.random() < math.exp(-delta / max(1e-12, temp)):
            a = cand
            cur_sc = cand_sc
            if cur_sc < best_sc:
                best_sc = cur_sc
                best_a = a.copy()
        temp *= cooling
    return best_a, best_sc


# ----------------------------------------------------------------------
# Smooth‑max (log‑sum‑exp) gradient descent
# ----------------------------------------------------------------------


def softmax_gradient_descent(initial_a: np.ndarray,
                             deadline: float,
                             betas: List[float],
                             iter_per_beta: int = 300,
                             eta0: float = 0.5) -> Tuple[np.ndarray, float]:
    """
    Minimise Φ_β(a) = (1/β)·log Σ_k exp(β·(a∗a)_k) via exponentiated gradient.
    """
    a = np.maximum(initial_a, 1e-30)
    a /= a.sum()
    best_a = a.copy()
    best_sc = compute_score(a)

    for beta in betas:
        for it in range(1, iter_per_beta + 1):
            if time.time() >= deadline:
                return best_a, best_sc

            conv = convolution_fft(a, a)      # length 2n‑1
            cmax = conv.max()
            # stable softmax weights
            exp_vals = np.exp(beta * (conv - cmax))
            p = exp_vals / exp_vals.sum()

            # cross‑correlation p ⋆ a  (length n)
            N = next_pow2(len(p) + len(a) - 1)
            P = np.fft.fft(p, N)
            A = np.fft.fft(a, N)
            cross = np.fft.ifft(P * np.conj(A)).real[:len(a)]
            cross = np.maximum(cross, 0.0)

            grad = 4.0 * len(a) * cross       # non‑negative

            eta = eta0 / math.sqrt(it)
            a = a * np.exp(-eta * grad)

            # project onto simplex (simple renorm)
            s = a.sum()
            if s == 0.0:
                a = np.full_like(a, 1.0 / len(a))
            else:
                a /= s

            cur_sc = compute_score(a)
            if cur_sc < best_sc:
                best_sc = cur_sc
                best_a = a.copy()
    return best_a, best_sc


# ----------------------------------------------------------------------
# Exact sub‑gradient of max‑convolution
# ----------------------------------------------------------------------


def _max_conv_gradient(a: np.ndarray, conv: np.ndarray) -> np.ndarray:
    """
    Sub‑gradient of max_k (a∗a)_k.
    For each active k, the contribution is 2·a_{k‑j}.
    Averaged if several k are active.
    """
    n = len(a)
    max_val = conv.max()
    mask = np.isclose(conv, max_val, atol=1e-15, rtol=0.0)
    active = np.where(mask)[0]
    grad = np.zeros_like(a)

    for k in active:
        lo = max(0, k - (n - 1))
        hi = min(k, n - 1)
        idx = np.arange(lo, hi + 1, dtype=int)
        grad[idx] += 2.0 * a[k - idx]

    if len(active) > 1:
        grad /= len(active)
    return grad


# ----------------------------------------------------------------------
# Greedy pairwise mass transfer driven by exact sub‑gradient
# ----------------------------------------------------------------------


def greedy_mass_transfer(a0: np.ndarray,
                         deadline: float,
                         rng: np.random.Generator,
                         max_iter: int = 5000,
                         start_frac: float = 0.10,
                         decay: float = 0.995) -> Tuple[np.ndarray, float]:
    """
    At each step move a fraction of mass from the index with largest
    sub‑gradient to the one with smallest sub‑gradient.
    """
    a = a0.copy()
    best_a = a.copy()
    best_sc = compute_score(a)
    frac = start_frac

    for it in range(max_iter):
        if time.time() >= deadline:
            break
        conv = convolution_fft(a, a)
        grad = _max_conv_gradient(a, conv)

        i = int(grad.argmax())
        j = int(grad.argmin())
        if i == j or a[i] <= 0.0:
            # fallback to a random pair
            i, j = rng.integers(0, len(a), size=2)
            if i == j or a[i] <= 0.0:
                continue

        delta = a[i] * frac
        if delta <= 0.0:
            continue
        a[i] -= delta
        a[j] += delta

        # keep sum constant, guarantee non‑negativity
        if a[i] < 0.0:
            a[j] += a[i]
            a[i] = 0.0

        cur_sc = compute_score(a)
        if cur_sc < best_sc:
            best_sc = cur_sc
            best_a = a.copy()

        # slowly reduce the step size
        frac *= decay
        if frac < 1e-5:
            frac = 1e-5
    return best_a, best_sc


# ----------------------------------------------------------------------
# Exponentiated sub‑gradient descent on the true max‑convolution
# ----------------------------------------------------------------------


def optimise_subgradient(initial_a: np.ndarray,
                         deadline: float,
                         *,
                         eta0: float = 0.6,
                         max_iter: int = 200_000,
                         noise_every: int = 0,
                         noise_scale: float = 0.0) -> Tuple[np.ndarray, float]:
    """
    Mirror‑descent on Φ(a)=max_k (a∗a)_k with multiplicative updates.
    """
    a = np.maximum(initial_a, 1e-30)
    a /= a.sum()
    best_a = a.copy()
    best_sc = compute_score(a)
    n = len(a)

    for it in range(1, max_iter + 1):
        if time.time() >= deadline:
            break

        conv = convolution_fft(a, a)
        grad = _max_conv_gradient(a, conv)

        eta = eta0 / math.sqrt(it)
        a = a * np.exp(-eta * grad)

        if noise_every and (it % noise_every == 0):
            a += rng.normal(0.0, noise_scale, size=n)
            a = np.maximum(a, 0.0)

        s = a.sum()
        a = a / s if s != 0.0 else np.full_like(a, 1.0 / n)

        cur_sc = compute_score(a)
        if cur_sc < best_sc:
            best_sc = cur_sc
            best_a = a.copy()

        # occasional progress report (quiet unless many iterations)
        if it % 2000 == 0:
            print(f"    subgrad iter={it:,d} cur={cur_sc:.6f} best={best_sc:.6f}")

    return best_a, best_sc

def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs) -> List[float]:

    always returns a feasible list of floats.

        print(f"Fallback (height_sequence_1) score = {val:.6f}")
        if val < best_val:
            best_val = val
            best_seq = fallback
    except Exception:
        pass

    # ------------------------------------------------------------------
    # 1.  Explore several lengths (powers of two up to 8192).
    # ------------------------------------------------------------------
    # Longer vectors tend to give tighter bounds but cost more per step.
    # The list is ordered from largest (most promising) downwards.
    lengths = [8192, 4096, 2048, 1024]

    for n in lengths:
        if time.time() >= deadline - 10.0:
            break
        print(f"\n=== Exploring length n = {n} ===")

        # --------------------------------------------------------------
        # 1a. Build a diverse library of starting vectors.
        # --------------------------------------------------------------
        candidates: List[np.ndarray] = []

        # Uniform
        candidates.append(np.full(n, 1.0 / n))

        # Classic arcsine
        candidates.append(arcsine_initial(n))

        # Power‑law grid
        for p in np.linspace(0.25, 1.25, 7):
            eps = max(1e-6, 0.001 * n)
            candidates.append(generate_powerlaw(n, float(p), eps))

        # Beta‑like grid
        for alpha in [0.7, 1.0, 1.5, 2.5, 4.0]:
            for beta in [0.7, 1.0, 1.5, 2.5, 4.0]:
                candidates.append(generate_beta_like(n, float(alpha), float(beta)))

        # Sinusoidal and Hann families
        candidates.append(sine_initial(n))
        candidates.append(hann_initial(n))
        candidates.append(triangular_initial(n))

        # Dirichlet draws (different concentrations)
        for conc in [0.2, 0.5, 1.0, 2.0]:
            candidates.append(random_dirichlet(n, float(conc), rng))

        # Mixtures of arcsine and uniform (different mixing weights)
        for _ in range(3):
            w = rng.random()
            mix = w * arcsine_initial(n) + (1.0 - w) * np.full(n, 1.0 / n)
            mix /= mix.sum()
            candidates.append(mix)

        # Small Gaussian perturbations of a few seeds
        for base in (arcsine_initial(n), generate_powerlaw(n, 0.6, 0.001)):
            for _ in range(2):
                candidates.append(perturb(base, rng, scale=0.015))

        # --------------------------------------------------------------
        # 1b. Quick evaluation with the fast internal score, keep the best.
        # --------------------------------------------------------------
        scored = [(compute_score(c), c) for c in candidates]
        scored.sort(key=lambda x: x[0])
        # Keep the five most promising seeds for deeper optimisation.
        top_candidates = [c for _, c in scored[:5]]

        # --------------------------------------------------------------
        # 1c. Refine each retained candidate.

        for idx, cand in enumerate(top_candidates):

        a = np.array(best_seq, dtype=float)
        remaining = max(0.0, deadline - time.time())
        max_iter = int(min(250_000, remaining * 350))
        a_refined, _ = optimise_subgradient(a, deadline,

                                            max_iter=max_iter,
                                            noise_every=0,
                                            noise_scale=0.0)
        final_val = evaluate_sequence(a_refined.tolist())

        best_seq = [1.0] * 256
        best_val = evaluate_sequence(best_seq)

    print(f"\nSearch finished. Best reported score = {best_val:.6f}")
    return best_seq
