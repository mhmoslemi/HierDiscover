def compute_gradient_smooth(a, lam):
    n = len(a)
    L = next_pow2(3*n)   # at least len(a)+len(w)-1
    A = np.fft.rfft(a, n=L)

"""
Propose a low‑score step‑function using a multi‑scale Adam optimiser with a
soft‑max smoothing of the max‑self‑convolution objective.

Key ideas
---------
* The score 2·n·max(conv) (with sum(a)=1) depends only on the shape of a.
* Directly minimising the non‑smooth max is hard – we replace it by the
  smooth surrogate
        f_λ(a) = (1/λ)·log( Σ_k exp( λ·(a∗a)_k ) )
  whose gradient is a weighted sum of the individual convolution gradients.
* The gradient can be evaluated in O(n log n) by FFT‑based correlation.
* We optimise the surrogate with Adam on the log‑softmax parametrisation
  (a_i = exp(z_i)/Σ_j exp(z_j)), which guarantees non‑negativity and the
  simplex constraint.
* A schedule of increasing λ (from very smooth to near‑max) guides the
  optimisation to good local minima.
* A hierarchy of lengths (1024 → 2048 → 4096 → 8192) is used: the best
  distribution found so far is up‑sampled to the next size and refined.
* Only a handful of cheap surrogate evaluations are needed; the expensive
  official ``evaluate_sequence`` is called for the few top candidates.
* The whole search respects a 1000 s wall‑clock budget and always returns a
  feasible sequence.

The function ``propose_candidate`` is the entry point required by the
challenge framework.
"""

import time
import math
from typing import List, Tuple
import numpy as np

# ----------------------------------------------------------------------
# Basic numeric helpers
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full linear convolution using a real FFT."""
    m = len(a) + len(b) - 1
    n_fft = next_pow2(m)
    A = np.fft.rfft(a, n=n_fft)
    B = np.fft.rfft(b, n=n_fft)
    C = A * B
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


def compute_score(a: np.ndarray) -> float:
    """
    Fast surrogate for the official evaluator:
        2·n·max(a∗a)   (a is already normalised to sum = 1)
    """
    n = len(a)
    conv = convolution_fft(a, a)
    max_val = float(conv.max())
    return 2.0 * n * max_val


# ----------------------------------------------------------------------
# Families of initial shapes
# ----------------------------------------------------------------------


def arcsine_initial(n: int) -> np.ndarray:
    """
    Discrete arcsine density a_i ∝ 1/√((i+½)(n‑½‑i)).
    This shape yields a nearly flat self‑convolution.
    """
    i = np.arange(n, dtype=float)
    eps = 1e-12
    vals = 1.0 / np.sqrt((i + 0.5) * (n - 0.5 - i) + eps)
    vals /= vals.sum()
    return vals


def powerlaw_initial(n: int, expo: float) -> np.ndarray:
    """
    Generalised power‑law shape a_i ∝ ((i+½)(n‑½‑i))^expo.
    expo = –0.5 reproduces the arcsine distribution.
    """
    i = np.arange(n, dtype=float)
    eps = 1e-12
    vals = ((i + 0.5) * (n - 0.5 - i) + eps) ** expo
    vals = np.maximum(vals, 0.0)
    vals /= vals.sum()
    return vals


def random_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    """Draw a random point on the simplex (α‑parameter = conc)."""
    if conc <= 0:
        # heavy‑tailed, near‑sparse draws
        sample = rng.exponential(1.0, size=n) ** (1.0 / max(0.1, -conc))
    else:
        sample = rng.gamma(conc, 1.0, size=n)
    sample += 1e-12
    sample /= sample.sum()
    return sample


def perturb(a: np.ndarray, rng: np.random.Generator, scale: float = 0.02) -> np.ndarray:
    """Add a tiny Gaussian jitter and re‑normalise."""
    b = a + rng.normal(0.0, scale, size=a.shape)
    b = np.maximum(b, 0.0)
    if b.sum() == 0.0:
        return np.full_like(a, 1.0 / len(a))
    return b / b.sum()


def upsample(prev: np.ndarray, new_n: int) -> np.ndarray:
    """
    Linear interpolation from len(prev) → new_n, followed by re‑normalisation.
    """
    x_old = np.linspace(0.0, 1.0, len(prev))
    x_new = np.linspace(0.0, 1.0, new_n)
    interp = np.interp(x_new, x_old, prev)
    total = interp.sum()
    if total == 0.0:
        return np.full(new_n, 1.0 / new_n)
    return interp / total


# ----------------------------------------------------------------------
# Gradient of the smoothed max‑self‑convolution objective
# ----------------------------------------------------------------------


def compute_gradient_smooth(a: np.ndarray, lam: float) -> np.ndarray:
    """
    Gradient of  f_λ(a) = (1/λ)·log Σ_k exp( λ·(a∗a)_k )
    with respect to a, assuming sum(a)=1.
    Implements
        grad = 2·corr( w , a )
    where w is the soft‑max over the convolution entries.
    """
    n = len(a)

    # size large enough for both the convolution (2n‑1) and the correlation (3n‑2)
    L = next_pow2(3 * n)

    # FFT of a (re‑used for convolution and correlation)
    A = np.fft.rfft(a, n=L)

    # ----- compute the self‑convolution (full length 2n‑1) -----
    conv_full = np.fft.irfft(A * A, n=L)[: 2 * n - 1]

    # ----- soft‑max weights -----
    max_c = conv_full.max()
    # stability: subtract the maximum before exponentiation
    exp_vals = np.exp(lam * (conv_full - max_c))
    w = exp_vals / exp_vals.sum()          # shape (2n‑1,)

    # ----- correlation w ⋆ a (valid part gives the gradient) -----
    W = np.fft.rfft(w, n=L)
    corr = np.fft.irfft(W * np.conj(A), n=L)   # length L (real)

    # valid region for correlation : indices n‑1 … n‑1+n‑1
    grad = 2.0 * corr[n - 1 : n - 1 + n]      # shape (n,)

    return grad


# ----------------------------------------------------------------------
# Adam optimiser on the log‑softmax parametrisation
# ----------------------------------------------------------------------


def adam_optimize(
    a_init: np.ndarray,
    deadline: float,
    *,
    lam_schedule: List[float],
    lr: float = 0.5,
    max_iters_per_lam: int = 5000,
    beta1: float = 0.9,
    beta2: float = 0.999,
    eps_adam: float = 1e-8,
) -> Tuple[np.ndarray, float]:
    """
    Adam optimisation of the smoothed objective.
    Parameters
    ----------
    a_init : np.ndarray
        Starting point (already normalised to sum = 1, non‑negative).
    deadline : float
        Unix time at which the optimiser must stop.
    lam_schedule : list of float
        Sequence of smoothing parameters λ (increasing).
    lr : float
        Base learning‑rate for Adam (applied on the log‑it variables).
    max_iters_per_lam : int
        Upper bound on the number of Adam steps per λ value.
    Returns
    -------
    best_a : np.ndarray
        The best distribution seen (normalised).
    best_score : float
        Surrogate score 2·n·max(a∗a) of ``best_a``.
    """
    n = len(a_init)
    # initialise log‑it variables (avoid log(0))
    eps = 1e-30
    z = np.log(np.maximum(a_init, eps))

    # bookkeeping of the best surrogate value seen so far
    best_a = a_init.copy()
    best_score = compute_score(best_a)

    # Adam state
    m = np.zeros_like(z)
    v = np.zeros_like(z)
    t = 0  # global iteration counter for bias‑correction

    for lam in lam_schedule:
        # decide how many iterations we can still afford for this λ
        if time.time() >= deadline:
            break
        # crude estimate: many thousands of iterations per second are possible;
        # we cap by ``max_iters_per_lam`` and the remaining wall‑clock time.
        remaining = max(0.0, deadline - time.time())
        it_budget = int(min(max_iters_per_lam, remaining * 1500))
        if it_budget <= 0:
            break

        for it in range(it_budget):
            if time.time() >= deadline:
                break

            # ---------- rebuild a from logits ----------
            max_z = np.max(z)
            exp_z = np.exp(z - max_z)          # subtraction for numerical stability
            a = exp_z / exp_z.sum()            # automatically on the simplex

            # ---------- gradient of the smoothed objective ----------
            grad_a = compute_gradient_smooth(a, lam)          # shape (n,)

            # transform gradient to log‑it space:
            #   g_z = a * (grad_a - (a·grad_a))
            dot = np.dot(a, grad_a)
            grad_z = a * (grad_a - dot)

            # ---------- Adam update ----------
            t += 1
            m = beta1 * m + (1.0 - beta1) * grad_z
            v = beta2 * v + (1.0 - beta2) * (grad_z ** 2)
            m_hat = m / (1.0 - beta1 ** t)
            v_hat = v / (1.0 - beta2 ** t)
            step = lr * m_hat / (np.sqrt(v_hat) + eps_adam)
            z = z - step

            # ---------- occasional surrogate evaluation ----------
            if (it + 1) % 250 == 0 or it + 1 == it_budget:
                max_z = np.max(z)
                exp_z = np.exp(z - max_z)
                a_tmp = exp_z / exp_z.sum()
                cur_sc = compute_score(a_tmp)
                if cur_sc < best_score:
                    best_score = cur_sc
                    best_a = a_tmp.copy()
        # proceed to the next λ value
    return best_a, best_score

def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs) -> List[float]:
    """
    Search for a non‑negative step‑function with a low ``evaluate_sequence``
    value.  The routine works hierarchically (1024 → 2048 → 4096 → 8192)
    and uses a multi‑scale Adam optimiser on a smoothed max‑self‑convolution
    objective.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 2.0          # tiny safety margin

    best_seq: List[float] = []
    best_val: float = float("inf")
    best_internal: np.ndarray = None                # cached normalised vector

    # ------------------------------------------------------------------
    # 0.  Optional fallback from a previously discovered construction.
    # ------------------------------------------------------------------
    try:
        fallback = [float(x) for x in height_sequence_1]   # type: ignore
        val = evaluate_sequence(fallback)
        print(f"Fallback (height_sequence_1) score = {val:.6f}")
        if val < best_val:
            best_val = val
            best_seq = fallback
            best_internal = np.array(fallback, dtype=float)
            best_internal /= best_internal.sum()
    except Exception:
        pass

    # ------------------------------------------------------------------
    # 1.  Length hierarchy – start small, enlarge progressively.
    # ------------------------------------------------------------------
    lengths = [1024, 2048, 4096, 8192]  # feel free to trim if time runs out

    for n in lengths:
        if time.time() >= deadline - 10.0:
            break
        print(f"\n=== Exploring length n = {n} ===")

        # --------------------------------------------------------------
        # 1a.  Build a small library of start points.
        # --------------------------------------------------------------
        candidates: List[np.ndarray] = []

        # warm‑start from the best vector found so far (if any)
        if best_internal is not None:
            up = upsample(best_internal, n)
            candidates.append(up)

        # classic arcsine shape (very strong baseline)
        candidates.append(arcsine_initial(n))

        # a few power‑law variants around the arcsine exponent
        for expo in (-0.45, -0.55):
            candidates.append(powerlaw_initial(n, expo))

        # random Dirichlet draws – add diversity
        for conc in (0.5, 1.0, 2.0):
            candidates.append(random_dirichlet(n, conc, rng))

        # small perturbations of the arcsine shape
        for _ in range(2):
            cand = perturb(arcsine_initial(n), rng, scale=0.03)
            candidates.append(cand)

        # --------------------------------------------------------------
        # 1b.  Quick screening with the cheap surrogate score.
        # --------------------------------------------------------------
        scored = [(compute_score(c), c) for c in candidates]
        scored.sort(key=lambda x: x[0])
        top_candidates = [c for _, c in scored[:3]]   # keep the three most promising

        # --------------------------------------------------------------
        # 1c.  Refine each retained candidate with Adam + smoothing.
        # --------------------------------------------------------------
        for idx, cand in enumerate(top_candidates):
            if time.time() >= deadline - 5.0:
                break
            print(f"  Refining candidate {idx + 1}/{len(top_candidates)} (n={n})")

            # smoothing schedule – start very smooth, finish almost exact
            lam_schedule = [1.0, 5.0, 20.0, 80.0, 300.0, 1200.0]

            # estimate remaining time and allocate iterations per λ
            remaining = max(0.0, deadline - time.time())
            # roughly 2000 iterations per second for these sizes
            it_per_lam = max(10, int(min(8000, remaining * 1800 / len(lam_schedule))))

            refined, _ = adam_optimize(
                a_init=cand,
                deadline=deadline,
                lam_schedule=lam_schedule,
                lr=0.6,
                max_iters_per_lam=it_per_lam,
            )

            # official (expensive) evaluation – only for the few final candidates
            val = evaluate_sequence(refined.tolist())
            print(f"    evaluated score = {val:.6f}")

            if val < best_val:
                best_val = val
                best_seq = refined.tolist()
                best_internal = refined.copy()
                print(f"    *** New global best: {best_val:.6f} (n={n}) ***")

    # ------------------------------------------------------------------
    # 2.  Final polishing of the overall best (if time permits).
    # ------------------------------------------------------------------
    if best_seq and time.time() < deadline - 5.0:
        a = np.array(best_seq, dtype=float)
        print("\nFinal polishing of the global best")
        # a slightly finer λ schedule focused on the high‑λ regime
        lam_schedule = [200.0, 600.0, 1800.0, 5400.0]
        remaining = max(0.0, deadline - time.time())
        it_per_lam = max(10, int(min(20000, remaining * 1500 / len(lam_schedule))))

        a_ref, _ = adam_optimize(
            a_init=a,
            deadline=deadline,
            lam_schedule=lam_schedule,
            lr=0.3,
            max_iters_per_lam=it_per_lam,
        )
        final_val = evaluate_sequence(a_ref.tolist())
        print(f"Final polished score = {final_val:.6f}")
        if final_val < best_val:
            best_val = final_val
            best_seq = a_ref.tolist()
            print("*** Updated global best after final polish ***")

    # ------------------------------------------------------------------
    # 3.  Guard – always return a feasible sequence.
    # ------------------------------------------------------------------
    if not best_seq:
        # Very small fallback – guaranteed to be feasible.
        best_seq = [1.0] * 256
        best_val = evaluate_sequence(best_seq)

    print(f"\nSearch finished. Best reported score = {best_val:.6f}")
    return best_seq
