def build_conv_matrix(f):
    n = f.shape[0]
    rows = 2*n - 1
    T = np.zeros((rows, n), dtype=np.float64)
    for i in range(n):
        start = i
        end = i + n
        T[start:end, i] = f
    return T
```

def solve_lp_max_g(f, M, solver='ECOS'):
    n = f.shape[0]
    T = build_conv_matrix(f)
    var = cp.Variable(n, nonneg=True)
    constraints = [T @ var <= M]
    obj = cp.Maximize(cp.sum(var))
    prob = cp.Problem(obj, constraints)
    prob.solve(solver=solver, warm_start=True, max_iters=5000)
    if var.value is None:
        # fallback
        return np.full(n, 1.0/n)
    return np.array(var.value, dtype=np.float64)
```

"""
Search for a low‑sup‑norm step function.

The routine combines three complementary phases:

1️⃣  **Start** – a deterministic arcsine shape (known good for the
    continuous problem) and a few random Dirichlet seeds.

2️⃣  **Iterative linear‑programming refinement** – for a fixed vector `f`
    we solve the LP

          maximise   Σ g_j
          subject to  (f * g)_k ≤ M   ∀k,
                      g ≥ 0 ,

    where `M = max (f*f)`.  The LP finds a direction `g` that can be added
    to `f` without increasing the worst auto‑correlation.  After normalising
    `g` we move partially towards it (line‑search on a few step sizes) and
    repeat.  This mimics the “alternating LP” scheme from the statement
    and typically lowers the bound by a few 10⁻³.

3️⃣  **Exact polishing** – a long sub‑gradient descent on the true
    objective `2·n·max(conv(a,a))` (the denominator is fixed by keeping the
    iterate on the probability simplex) followed by an exponentiated‑gradient
    mirror‑descent.  These phases clean up the remaining tiny irregularities.

All phases respect a global time budget (default 1000 s).  The best vector
found is finally evaluated with the official ``evaluate_sequence`` and
returned.
"""

import time
import numpy as np
import cvxpy as cp
from typing import List

# ----------------------------------------------------------------------
# Low‑level utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Linear convolution via real FFT (length = len(a)+len(b)-1)."""
    m = len(a) + len(b) - 1
    n_fft = next_pow2(m)
    A = np.fft.rfft(a, n=n_fft)
    B = np.fft.rfft(b, n=n_fft)
    C = A * B
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


def compute_score(a: np.ndarray) -> float:
    """
    Fast surrogate for the official score.
    The vector is assumed to be normalised to Σa = 1, therefore the
    denominator disappears and the score is simply 2·n·max(conv(a,a)).
    """
    n = a.shape[0]
    conv = convolution_fft(a, a)
    return 2.0 * n * float(np.max(conv))


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection onto the probability simplex {x ≥ 0, Σx = 1}.
    Duchi et al., 2008.
    """
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / v.size)

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho_arr = u - cssv / np.arange(1, v.size + 1)
    rho = np.where(rho_arr > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Shape generators (used for diverse initial seeds)
# ----------------------------------------------------------------------


def arcsine_initial(n: int) -> np.ndarray:
    """Classic arcsine shape: a_i ∝ 1/√((i+½)*(n‑½‑i))."""
    i = np.arange(n, dtype=float)
    eps = 0.25
    vals = 1.0 / np.sqrt((i + 0.5) * (n - 0.5 - i) + eps)
    vals /= vals.sum()
    return vals


def random_simplex(n: int, rng: np.random.Generator) -> np.ndarray:
    """Draw a random point on the probability simplex (Dirichlet(1,…,1))."""
    x = rng.gamma(1.0, 1.0, size=n)
    return x / x.sum()


# ----------------------------------------------------------------------
# Sub‑gradient descent (exact objective) and EG polishing
# ----------------------------------------------------------------------


def subgradient_descent_exact(initial_a: np.ndarray,
                              deadline: float,
                              max_iter: int = 120_000,
                              step0: float = 0.20) -> np.ndarray:
    """Projected sub‑gradient descent on 2·n·max(conv(a,a))."""
    a = project_to_simplex(initial_a.copy())
    n = len(a)
    L = 2 * n - 1

    for it in range(1, max_iter + 1):
        if time.time() >= deadline:
            break

        conv = convolution_fft(a, a)
        max_val = float(np.max(conv))

        # indices attaining the maximum (with tolerance)
        tol = max_val * 1e-12 + 1e-14
        mask = (np.abs(conv - max_val) <= tol).astype(float)

        # sub‑gradient = 2·∑_{k∈K} shift(a,k)
        mask_rev = mask[::-1]
        conv_rev = convolution_fft(a, mask_rev)
        subgrad = 2.0 * conv_rev[L - 1 - np.arange(n)]

        eta = step0 / np.sqrt(it)
        a = a - eta * subgrad
        a = project_to_simplex(a)

        # occasional progress report
        if it % 5000 == 0:
            print(f"    SubGrad it={it:,d} score={compute_score(a):.6f}")

    return a


def eg_refine(a: np.ndarray,
              deadline: float,
              max_iter: int = 90_000,
              eta0: float = 0.25) -> np.ndarray:
    """Exponentiated‑gradient polishing (mirror descent on the simplex)."""
    n = len(a)
    L = 2 * n - 1

    for it in range(1, max_iter + 1):
        if time.time() >= deadline:
            break

        conv = convolution_fft(a, a)
        max_val = float(np.max(conv))
        tol = max_val * 1e-12 + 1e-14
        mask = (np.abs(conv - max_val) <= tol).astype(float)

        mask_rev = mask[::-1]
        conv_rev = convolution_fft(a, mask_rev)
        subgrad = 2.0 * conv_rev[L - 1 - np.arange(n)]

        eta = eta0 / (np.sqrt(it) + 1.0)
        a = a * np.exp(-eta * subgrad)

        # renormalise onto the simplex (numerical safety)
        total = a.sum()
        if total == 0.0:
            a = np.full_like(a, 1.0 / n)
        else:
            a = a / total

        if it % 5000 == 0:
            print(f"    EG it={it:,d} score={compute_score(a):.6f}")

    return a


# ----------------------------------------------------------------------
# Linear‑programming refinement
# ----------------------------------------------------------------------


def build_conv_matrix(f: np.ndarray) -> np.ndarray:
    """
    Build the (2·n‑1) × n matrix T such that T @ g = f * g (linear convolution).
    For each column i, the vector f is placed starting at row i.
    """
    n = f.shape[0]
    rows = 2 * n - 1
    T = np.zeros((rows, n), dtype=np.float64)
    for i in range(n):
        T[i:i + n, i] = f
    return T


def solve_lp_max_g(f: np.ndarray, M: float) -> np.ndarray:
    """
    Solve   maximise Σ g_j   s.t.   (f * g)_k ≤ M   ∀k,   g ≥ 0.
    Returns the optimal g (may have arbitrary sum).
    """
    n = f.shape[0]
    T = build_conv_matrix(f)

    g = cp.Variable(n, nonneg=True)
    # slight tolerance to avoid infeasibility due to floating‑point noise
    constraints = [T @ g <= M + 1e-12]
    prob = cp.Problem(cp.Maximize(cp.sum(g)), constraints)
    try:
        prob.solve(solver=cp.ECOS, warm_start=True, max_iters=5000, verbose=False)
    except Exception:
        # fallback – any feasible point will do
        prob.solve(solver=cp.SCS, warm_start=True, max_iters=2500, verbose=False)

    if g.value is None:
        # worst‑case fallback – uniform distribution
        return np.full(n, 1.0 / n)
    return np.array(g.value, dtype=np.float64)


def lp_refine(a: np.ndarray, deadline: float, max_iters: int = 12) -> np.ndarray:
    """
    Iteratively improve a probability vector by the LP scheme described
    in the problem statement.  The function stops when no improvement
    is observed or the time budget expires.
    """
    a = project_to_simplex(a)
    best = a.copy()
    best_sc = compute_score(best)

    for it in range(max_iters):
        if time.time() >= deadline - 1.0:
            break
        conv = convolution_fft(best, best)
        M = float(np.max(conv))

        g = solve_lp_max_g(best, M)
        sum_g = g.sum()
        if sum_g <= 1e-12:
            break
        g_norm = g / sum_g          # enforce Σ g = 1

        improved = False
        # try a few convex‑combination step sizes
        for t in (0.4, 0.6, 0.8):
            cand = (1.0 - t) * best + t * g_norm
            sc = compute_score(cand)
            if sc < best_sc - 1e-12:
                best = cand
                best_sc = sc
                improved = True
        if not improved:
            break
    return best

def propose_candidate(seed: int = 42,
                     budget_s: int = 1000,
                     **kwargs) -> List[float]:
    """
    Search for a non‑negative step sequence with a low
    ``evaluate_sequence`` value.
    The algorithm uses arcsine start → LP refinement → sub‑gradient descent →
    exponentiated‑gradient polishing.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 1.0   # small safety margin

    best_seq: List[float] = []
    best_val: float = float("inf")

    # ------------------------------------------------------------------
    # Optional fallback from a previous run
    # ------------------------------------------------------------------
    if "height_sequence_1" in globals():
        try:
            fallback = [float(x) for x in globals()["height_sequence_1"]]
            val = evaluate_sequence(fallback)
            print(f"Fallback (height_sequence_1) score = {val:.6f}")
            if val < best_val:
                best_val = val
                best_seq = fallback
        except Exception as e:
            print("Fallback failed:", e)

    # ------------------------------------------------------------------
    # 1.  Build a small pool of initial seeds (length 2048 works well)
    # ------------------------------------------------------------------
    n = 2048
    seeds = [
        arcsine_initial(n),
        np.full(n, 1.0 / n),          # uniform
    ]
    # a few random Dirichlet points for diversity
    for _ in range(5):
        seeds.append(random_simplex(n, rng))

    # keep only the most promising few (based on the cheap surrogate)
    scored = [(compute_score(s), s) for s in seeds]
    scored.sort(key=lambda x: x[0])
    top_seeds = [vec for _, vec in scored[:3]]
    print(f"\nSelected {len(top_seeds)} top seeds for refinement (fast scores "
          f"{scored[0][0]:.6f} – {scored[-1][0]:.6f})")

    # ------------------------------------------------------------------
    # 2.  Refine each seed
    # ------------------------------------------------------------------
    for idx, seed_vec in enumerate(top_seeds, start=1):
        if time.time() >= deadline - 5.0:
            break
        print(f"\nRefining seed {idx}/{len(top_seeds)} …")
        a = seed_vec.astype(np.float64)

        # 2a. LP refinement (the core improvement step)
        a = lp_refine(a, deadline, max_iters=12)

        # 2b. Sub‑gradient descent (more iterations if time permits)
        if time.time() < deadline - 5.0:
            remaining = deadline - time.time()
            max_iter_sub = int(min(150_000, remaining * 30))
            a = subgradient_descent_exact(a, deadline,
                                          max_iter=max_iter_sub,
                                          step0=0.22)

        # 2c. Exponentiated‑gradient polishing
        if time.time() < deadline - 2.0:
            remaining = deadline - time.time()
            max_iter_eg = int(min(80_000, remaining * 30))
            a = eg_refine(a, deadline,
                          max_iter=max_iter_eg,
                          eta0=0.25)

        # Official evaluation of the refined candidate
        seq = a.tolist()
        try:
            val = evaluate_sequence(seq)
            print(f"  → official score = {val:.6f}")
            if val < best_val:
                best_val = val
                best_seq = seq
                print(f"  *** New global best: {best_val:.6f} ***")
        except Exception as e:
            print("  Evaluation failed:", e)

    # ------------------------------------------------------------------
    # 3.  Global polish on the current best (if time left)
    # ------------------------------------------------------------------
    if best_seq and time.time() < deadline - 3.0:
        print("\nFinal global polish …")
        a = np.array(best_seq, dtype=float)

        remaining = deadline - time.time()
        max_iter_sub = int(min(200_000, remaining * 30))
        a = subgradient_descent_exact(a, deadline,
                                      max_iter=max_iter_sub,
                                      step0=0.15)

        if time.time() < deadline - 2.0:
            remaining = deadline - time.time()
            max_iter_eg = int(min(120_000, remaining * 30))
            a = eg_refine(a, deadline,
                          max_iter=max_iter_eg,
                          eta0=0.20)

        final_seq = a.tolist()
        try:
            final_val = evaluate_sequence(final_seq)
            print(f"Final polished official score = {final_val:.6f}")
            if final_val < best_val:
                best_val = final_val
                best_seq = final_seq
                print("*** Updated global best after final polish ***")
        except Exception as e:
            print("Final polishing evaluation failed:", e)

    # ------------------------------------------------------------------
    # 4.  Guard – ensure a non‑empty return value
    # ------------------------------------------------------------------
    if not best_seq:
        # fallback to a simple uniform vector (guaranteed valid)
        fallback_n = 256
        best_seq = [1.0 / fallback_n] * fallback_n
        best_val = evaluate_sequence(best_seq)

    print(f"\nSearch finished. Best reported official score = {best_val:.6f}")
    return best_seq
