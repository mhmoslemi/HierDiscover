"""
Improved search for a non‑negative step sequence that minimizes
``evaluate_sequence``.  The new version expands on the previous solver by

* keeping a larger “top‑K” pool (12 instead of 8),
* adding an exponential‑decay family (a_i ∝ exp(−λ·i)),
* allocating a longer optimisation budget (more Adam → sub‑gradient → SA steps),
* performing a final intensive simulated‑annealing polishing phase with a
  slower cooling schedule.

All heavy work is limited to the 1000 s budget; the routine returns the
best sequence found (as a plain ``list[float]``) ready for the official
evaluator.
"""

import time
import math
from typing import List, Tuple

import numpy as np

# ----------------------------------------------------------------------
# Low‑level utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """
    Linear convolution using a real FFT.  Length = len(a)+len(b)-1.
    """
    m = len(a) + len(b) - 1
    n_fft = next_pow2(m)
    A = np.fft.rfft(a, n=n_fft)
    B = np.fft.rfft(b, n=n_fft)
    C = A * B
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


def compute_score(a: np.ndarray) -> float:
    """
    Fast surrogate for the official score.
    ``a`` is assumed to lie on the simplex Σa = 1.
    """
    n = a.size
    conv = convolution_fft(a, a)
    return 2.0 * n * float(np.max(conv))


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection onto the probability simplex {x ≥ 0, Σx = 1}.
    Duchi et al., 2008.
    """
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / v.size)

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho = np.where(u - cssv / np.arange(1, v.size + 1) > 0)[0]
    if len(rho) == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

# Shape factories (all return a vector that already sums to 1)
# ----------------------------------------------------------------------


def arcsine_initial(n: int) -> np.ndarray:
    """Classic arcsine shape used in earlier work."""
    i = np.arange(n, dtype=float)
    eps = 0.25
    vals = 1.0 / np.sqrt((i + 0.5) * (n - 0.5 - i) + eps)
    vals /= vals.sum()
    return vals


def generate_beta_shape(n: int, alpha: float, beta: float, eps: float) -> np.ndarray:
    """(i+ε)^{α‑1}·(n‑1‑i+ε)^{β‑1} (Beta‑type) and normalise."""
    i = np.arange(n, dtype=float)
    left = (i + eps) ** (alpha - 1.0)
    right = (n - 1 - i + eps) ** (beta - 1.0)
    base = left * right
    total = base.sum()
    if not np.isfinite(total) or total <= 0.0:
        return np.full(n, 1.0 / n)
    return base / total


def generate_power_shape(n: int, p: float, eps: float) -> np.ndarray:
    """Generalised arcsine: a_i ∝ ((i+½)*(n‑½‑i)+ε)^{-p}."""
    i = np.arange(n, dtype=float)
    denom = ((i + 0.5) * (n - 0.5 - i) + eps) ** p
    vals = 1.0 / denom
    total = vals.sum()
    if not np.isfinite(total) or total <= 0.0:
        return np.full(n, 1.0 / n)
    return vals / total


def generate_triangular_shape(n: int, peak: int) -> np.ndarray:
    """Symmetric triangular bump."""
    left = np.arange(peak + 1, dtype=float)
    right = np.arange(n - peak - 1, dtype=float)[::-1]
    seq = np.concatenate([left, right])
    seq /= seq.sum()
    return seq


def generate_mixture_shape(n: int, weight_arc: float) -> np.ndarray:
    """Mixture (weight_arc)·arcsine + (1‑weight_arc)·uniform."""
    arc = arcsine_initial(n)
    uni = np.full(n, 1.0 / n)
    mix = weight_arc * arc + (1.0 - weight_arc) * uni
    mix /= mix.sum()
    return mix


def generate_exponential_shape(n: int, rate: float) -> np.ndarray:
    """Exponential decay a_i ∝ exp(−rate·i).  Normalised to Σ=1."""
    i = np.arange(n, dtype=float)
    vals = np.exp(-rate * i)
    total = vals.sum()
    if total == 0.0 or not np.isfinite(total):
        return np.full(n, 1.0 / n)
    return vals / total


def generate_dirichlet_shape(n: int, rng: np.random.Generator) -> np.ndarray:
    """Sample a Dirichlet distribution (many tiny entries)."""
    alphas = rng.exponential(scale=1.0, size=n) + 0.01
    vec = alphas / alphas.sum()
    return vec

def smooth_gradient(a: np.ndarray, lam: float) -> np.ndarray:
    """
    Gradient of the soft‑max approximation of max(conv(a,a)).
    The sum‑to‑one constraint is enforced later by projection.
    """
    n = a.size
    conv = convolution_fft(a, a)
    max_c = np.max(conv)

    # soft‑max weights (numerically stable)
    exp_term = np.exp(lam * (conv - max_c))
    w = exp_term / np.sum(exp_term)

    # weighted sub‑gradient – convolution with reversed weights
    w_rev = w[::-1]
    conv_rev = convolution_fft(a, w_rev)
    L = 2 * n - 1
    weighted_grad = 2.0 * conv_rev[L - 1 - np.arange(n)]

    return 2.0 * n * weighted_grad


def adam_optimize_smooth(initial_a: np.ndarray,
                         deadline: float,
                         max_iter: int = 200_000,
                         lr0: float = 0.12,
                         lam_start: float = 2.0,
                         lam_end: float = 200.0,
                         noise_every: int = 2000,
                         noise_scale: float = 2e-5) -> Tuple[np.ndarray, float]:
    """
    Adam optimiser on the smooth surrogate.
    Returns (best_vector, best_score).
    """
    a = project_to_simplex(initial_a.astype(float).copy())
    best_a = a.copy()
    best_sc = compute_score(a)

    beta1, beta2 = 0.9, 0.999
    eps = 1e-8
    m = np.zeros_like(a)
    v = np.zeros_like(a)

    for it in range(1, max_iter + 1):
        if time.time() >= deadline:
            break

        lam = lam_start + (lam_end - lam_start) * min(1.0, it / max_iter)

        grad = smooth_gradient(a, lam)

        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)

        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        step = lr0 * m_hat / (np.sqrt(v_hat) + eps)
        a_new = a - step

        if noise_every and (it % noise_every == 0):
            a_new += np.random.randn(a.size) * noise_scale

        a_new = project_to_simplex(a_new)

        cur_sc = compute_score(a_new)
        if cur_sc < best_sc:
            best_sc = cur_sc
            best_a = a_new.copy()

        a = a_new

        if it % 5000 == 0:
            print(f"    AdamSmooth it={it:,d} cur={cur_sc:.6f} best={best_sc:.6f}")

    return best_a, best_sc


def subgradient_descent_exact(initial_a: np.ndarray,
                              deadline: float,
                              max_iter: int = 300_000,
                              step0: float = 0.05) -> np.ndarray:
    """
    Sub‑gradient descent on the true objective f(a)=2·n·max(conv(a,a)).
    The step size is η_t = step0 / sqrt(t).  Projection onto the simplex
    preserves Σa = 1.
    """
    a = project_to_simplex(initial_a.copy())
    best_a = a.copy()
    best_sc = compute_score(a)
    n = a.size
    L = 2 * n - 1

    for it in range(1, max_iter + 1):
        if time.time() >= deadline:
            break

        conv = convolution_fft(a, a)
        max_val = np.max(conv)

        # mask of entries attaining the maximum (within a tiny tolerance)
        tol = max_val * 1e-12 + 1e-14
        mask = (np.abs(conv - max_val) <= tol).astype(float)

        # sub‑gradient: 2·∑_k mask_k·shift(a,k)
        mask_rev = mask[::-1]
        conv_rev = convolution_fft(a, mask_rev)
        weighted_grad = 2.0 * conv_rev[L - 1 - np.arange(n)]

        eta = step0 / np.sqrt(it)
        a = a - eta * weighted_grad
        a = project_to_simplex(a)

        cur_sc = compute_score(a)
        if cur_sc < best_sc:
            best_sc = cur_sc
            best_a = a.copy()

        if it % 5000 == 0:
            print(f"    SubGrad it={it:,d} cur={cur_sc:.6f} best={best_sc:.6f}")

    return best_a


def simulated_annealing(a: np.ndarray,
                        rng: np.random.Generator,
                        deadline: float,
                        max_iters: int = 600_000,
                        init_temp: float = 0.05,
                        decay: float = 0.99995) -> np.ndarray:
    """
    Simple Metropolis‑style local search.
    Moves a fraction of mass from one coordinate to another.
    The fast surrogate (2·n·max) is used for acceptance.
    """
    cur = a.copy()
    cur_sc = compute_score(cur)
    best = cur.copy()
    best_sc = cur_sc
    n = cur.size
    temp = init_temp
    it = 0

    while it < max_iters and time.time() < deadline:
        it += 1
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue

        # relocate up to 20 % of the mass at i
        delta = cur[i] * 0.2 * rng.random()
        if delta <= 1e-15 or cur[i] - delta < 0:
            continue

        cand = cur.copy()
        cand[i] -= delta
        cand[j] += delta

        cand_sc = compute_score(cand)

        # Metropolis criterion
        if cand_sc < cur_sc or rng.random() < math.exp(-(cand_sc - cur_sc) / max(temp, 1e-12)):
            cur = cand
            cur_sc = cand_sc
            if cur_sc < best_sc:
                best = cur.copy()
                best_sc = cur_sc

        temp = max(temp * decay, 1e-9)

        if it % 10_000 == 0:
            print(f"    SA it={it:,d} cur={cur_sc:.6f} best={best_sc:.6f}")

    return best

def propose_candidate(seed: int = 42,
                     budget_s: int = 1000,
                     **kwargs) -> List[float]:
    """
    Search for a non‑negative step sequence with a low official score.
    The routine follows three stages:
        1) wide random sampling of several parametric families,
        2) heavy local refinement of the best few seeds (Adam → sub‑grad → SA),
        3) a final polished simulated‑annealing run on the overall best.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 1.0   # safety margin

    best_seq: List[float] = []
    best_val: float = float("inf")

    # --------------------------------------------------------------
    # 0.  Load a fallback if the notebook already produced one.
    # --------------------------------------------------------------
    if "height_sequence_1" in globals():
        try:
            fallback = [float(x) for x in globals()["height_sequence_1"]]
            val = evaluate_sequence(fallback)
            print(f"Fallback (height_sequence_1) score = {val:.6f}")
            if val < best_val:
                best_val = val
                best_seq = fallback
        except Exception:
            pass

    # --------------------------------------------------------------
    # 1.  Wide random exploration (top‑K pool)
    # --------------------------------------------------------------
    n = 2048                     # length used in the previous best run
    top_k = 12                  # keep this many promising seeds
    best_pool: List[Tuple[float, np.ndarray]] = []  # (score, vector)

    def maybe_insert(score: float, vec: np.ndarray) -> None:
        """Insert (score,vec) into best_pool keeping its size ≤ top_k."""
        if len(best_pool) < top_k:
            best_pool.append((score, vec))
        else:
            worst_idx = max(range(len(best_pool)),
                            key=lambda i: best_pool[i][0])
            if score < best_pool[worst_idx][0]:
                best_pool[worst_idx] = (score, vec)

    # deterministic starters
    uniform = np.full(n, 1.0 / n)
    maybe_insert(compute_score(uniform), uniform)

    arc = arcsine_initial(n)
    maybe_insert(compute_score(arc), arc)

    # random sampling loop
    max_candidates = 250_000
    cnt = 0
    while time.time() < deadline - 30.0 and cnt < max_candidates:
        cnt += 1
        r = rng.random()
        if r < 0.30:                    # beta family
            alpha = rng.uniform(0.02, 5.0)
            beta = rng.uniform(0.02, 5.0)
            eps = rng.uniform(1e-6, 3.0)
            cand = generate_beta_shape(n, alpha, beta, eps)
        elif r < 0.55:                  # power / generalised arcsine
            p = rng.uniform(0.1, 5.0)
            eps = rng.uniform(0.0, 3.0)
            cand = generate_power_shape(n, p, eps)
        elif r < 0.70:                  # mixture arcsine / uniform
            w = rng.random()
            cand = generate_mixture_shape(n, w)
        elif r < 0.80:                  # exponential decay
            rate = rng.uniform(0.0005, 0.01)   # slow decay -> long tail
            cand = generate_exponential_shape(n, rate)
        elif r < 0.90:                  # triangular bump
            peak = rng.integers(0, n)
            cand = generate_triangular_shape(n, peak)
        else:                           # Dirichlet sample
            cand = generate_dirichlet_shape(n, rng)

        sc = compute_score(cand)
        maybe_insert(sc, cand)

    # Extract seed vectors (best first)
    best_pool.sort(key=lambda x: x[0])
    seed_vectors = [vec for _, vec in best_pool]

    print(f"Random exploration finished – kept {len(seed_vectors)} seeds.")

    # --------------------------------------------------------------
    # 2.  Refinement of each seed
    # --------------------------------------------------------------
    for idx, seed_vec in enumerate(seed_vectors, 1):
        if time.time() >= deadline - 5.0:
            break
        print(f"\n--- Refining seed {idx}/{len(seed_vectors)} ---")

        # Adam‑smooth optimisation
        remaining = deadline - time.time()
        max_iter_adam = int(min(300_000, remaining * 45))
        refined, _ = adam_optimize_smooth(
            initial_a=seed_vec,
            deadline=deadline,
            max_iter=max_iter_adam,
            lr0=0.12,
            lam_start=2.0,
            lam_end=200.0,
            noise_every=2000,
            noise_scale=2e-5,
        )

        # Exact sub‑gradient descent
        if time.time() < deadline - 5.0:
            remaining = deadline - time.time()
            max_iter_sub = int(min(300_000, remaining * 40))
            refined = subgradient_descent_exact(
                refined,
                deadline,
                max_iter=max_iter_sub,
                step0=0.05,
            )

        # Simulated‑annealing polishing
        if time.time() < deadline - 5.0:
            remaining = deadline - time.time()
            max_iter_sa = int(min(600_000, remaining * 40))
            refined = simulated_annealing(
                refined,
                rng,
                deadline,
                max_iters=max_iter_sa,
                init_temp=0.05,
                decay=0.99995,
            )

        # Official evaluation
        try:
            val = evaluate_sequence(refined.tolist())
            print(f"    Official score = {val:.6f}")
            if val < best_val:
                best_val = val
                best_seq = refined.tolist()
                print(f"    *** New global best: {best_val:.6f} ***")
        except Exception:
            pass

    # --------------------------------------------------------------
    # 3.  Final polishing on the overall best vector
    # --------------------------------------------------------------
    if best_seq and time.time() < deadline - 3.0:
        a = np.array(best_seq, dtype=float)
        remaining = deadline - time.time()
        # a few more sub‑gradient steps
        sub_iters = int(min(400_000, remaining * 30))
        a = subgradient_descent_exact(a, deadline, max_iter=sub_iters, step0=0.04)

        # long simulated annealing run with very slow cooling
        sa_iters = int(min(800_000, (deadline - time.time()) * 30))
        a = simulated_annealing(
            a,
            rng,
            deadline,
            max_iters=sa_iters,
            init_temp=0.07,
            decay=0.99998,
        )

        final_val = evaluate_sequence(a.tolist())
        print(f"\nFinal polished official score = {final_val:.6f}")
        if final_val < best_val:
            best_val = final_val
            best_seq = a.tolist()
            print("*** Updated global best after final polish ***")

    # --------------------------------------------------------------
    # 4.  Guard – ensure a non‑empty return value
    # --------------------------------------------------------------
    if not best_seq:
        best_seq = [1.0] * n
        best_val = evaluate_sequence(best_seq)

    print(f"\nSearch finished. Best reported score = {best_val:.6f}")
    return best_seq
