"""
Multi‑scale optimisation for the max‑convolution step‑function problem.

The search proceeds in three stages:

1. **Coarse hierarchy** – start with a tiny vector (n≈64) and progressively
   upscale (linear interpolation) to larger lengths (up to 2048).  Each level
   inherits the best shape from the previous one, giving a good warm‑start.

2. **Linear‑programming improvement** – for modest sizes (n≤512) we solve the
   LP described in the statement:
        maximise   Σ b_j
        subject to b_j≥0   and   (f*b)_k ≤ M   ∀k,
   where f is the current vector and M = max_k (f*f)_k.
   The LP finds a direction that provably does not increase the max‑convolution.
   We take a short convex combination f←(1‑t)f + t·b̂ (b̂ is normalised) and repeat.

3. **Fine‑grained optimisation** – a mirror‑descent (exponentiated‑gradient)
   optimiser with Adam‑style moment estimates minimises the true objective
   Φ(a)=max_k (a∗a)_k.  The update is multiplicative, keeps the vector non‑negative
   and a simple renormalisation projects onto the simplex.
   The optimiser is run for many cheap iterations; a short simulated‑annealing
   pass is added as a final “polish”.

All stages respect the wall‑clock budget (default 1000 s).  The routine always
returns a feasible non‑negative list of floats; the best sequence seen is
reported via the official ``evaluate_sequence`` call at the end.
"""

import time
import numpy as np
from typing import List, Tuple

# optional LP solver
try:
    from scipy.optimize import linprog  # type: ignore
except Exception:  # scipy not available in the execution environment
    linprog = None  # type: ignore


# ----------------------------------------------------------------------
# Basic numeric tools
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Linear convolution via real FFT (zero‑padded to a power of two)."""
    m = len(a) + len(b) - 1
    n_fft = next_pow2(m)
    A = np.fft.rfft(a, n=n_fft)
    B = np.fft.rfft(b, n=n_fft)
    C = A * B
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


def compute_score(a: np.ndarray) -> float:
    """Fast surrogate of the official objective (scale‑invariant)."""
    n = len(a)
    conv = convolution_fft(a, a)
    max_val = float(np.max(conv))
    s = float(np.sum(a))
    if s < 1e-12:
        return float("inf")
    return 2.0 * n * max_val / (s * s)


# ----------------------------------------------------------------------
# Simple families of seeds
# ----------------------------------------------------------------------


def arcsine_initial(n: int) -> np.ndarray:
    """Discrete arcsine‑like density (normalised to sum = 1)."""
    i = np.arange(n, dtype=float)
    eps = 1e-4
    vals = 1.0 / np.sqrt((i + 0.5) * (n - 0.5 - i) + eps)
    vals /= vals.sum()
    return vals


def uniform_initial(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n)


def random_dirichlet(n: int, rng: np.random.Generator, conc: float = 1.0) -> np.ndarray:
    """Sample a Dirichlet vector and normalise."""
    samp = rng.gamma(conc, 1.0, size=n)
    samp = np.maximum(samp, 1e-12)
    return samp / samp.sum()


# ----------------------------------------------------------------------
# Linear‑programming improvement (the “alternating LP” step)
# ----------------------------------------------------------------------


def build_conv_matrix(f: np.ndarray) -> np.ndarray:
    """
    Build matrix A ∈ ℝ^{(2n‑1)×n} such that (f * b) = A @ b.
    Row k corresponds to shift k (0‑based).
    """
    n = len(f)
    m = 2 * n - 1
    A = np.zeros((m, n), dtype=float)
    # fill by hand – still cheap for n up to 512
    for k in range(m):
        lo = max(0, k - (n - 1))
        hi = min(k, n - 1)
        # indices j = lo … hi, coefficient = f[k‑j]
        # we can assign in a vectorised way
        j = np.arange(lo, hi + 1, dtype=int)
        A[k, j] = f[k - j]
    return A


def lp_step(f: np.ndarray, M: float) -> np.ndarray | None:
    """
    Solve   maximise Σ b_j   s.t.   b≥0,  (f * b)_k ≤ M  ∀k.
    Returns the optimiser b (un‑normalised) or None if the LP fails.
    """
    if linprog is None:
        return None
    n = len(f)
    A = build_conv_matrix(f)               # (2n‑1) × n
    b_ub = np.full(A.shape[0], M)          # right‑hand side
    c = -np.ones(n)                        # maximise Σ b_j  → minimise -Σ b_j
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds,
                      method="highs", options={"presolve": True})
    except Exception:  # safety net
        return None
    if not res.success:
        return None
    return res.x


def lp_improve(f: np.ndarray, deadline: float,
               max_iters: int = 20, t: float = 0.2) -> Tuple[np.ndarray, float]:
    """
    Repeated LP improvement on f.  Returns the best vector and its fast score.
    """
    best_f = f.copy()
    best_sc = compute_score(f)
    n = len(f)
    for it in range(max_iters):
        if time.time() >= deadline:
            break
        conv = convolution_fft(f, f)
        M = float(conv.max())
        b = lp_step(f, M)
        if b is None:
            break
        # normalise b to a probability vector
        b_sum = b.sum()
        if b_sum == 0.0:
            b = uniform_initial(n)
        else:
            b = b / b_sum
        # convex blend and renormalise
        f = (1.0 - t) * f + t * b
        f = f / f.sum()
        sc = compute_score(f)
        if sc < best_sc:
            best_sc = sc
            best_f = f.copy()
    return best_f, best_sc


# ----------------------------------------------------------------------
# Adam‑style mirror descent on the true max‑convolution
# ----------------------------------------------------------------------


def _max_conv_gradient(a: np.ndarray, conv: np.ndarray) -> np.ndarray:
    """
    Sub‑gradient of  Φ(a)=max_k (a*a)_k .
    For each active shift k we add the contribution 2·a_{k‑j}.
    When several shifts attain the maximum we average the contributions.
    """
    n = len(a)
    max_val = conv.max()
    mask = np.isclose(conv, max_val, atol=1e-15)
    active = np.where(mask)[0]
    grad = np.zeros_like(a)

    for k in active:
        lo = max(0, k - (n - 1))
        hi = min(k, n - 1)
        idx = np.arange(lo, hi + 1, dtype=int)
        grad[idx] += 2.0 * a[k - idx]

    if len(active) > 1:
        grad /= float(len(active))
    return grad


def adam_maxconv(initial_a: np.ndarray,
                 deadline: float,
                 max_iter: int = 200_000,
                 beta1: float = 0.9,
                 beta2: float = 0.999,
                 eps: float = 1e-8,
                 t0: float = 0.5) -> Tuple[np.ndarray, float]:
    """
    Mirror‑descent (exponentiated‑gradient) with Adam‑style moment adaptation.
    Operates directly on the true max‑convolution objective.
    """
    a = np.maximum(initial_a, 1e-30)
    a = a / a.sum()
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    best_a = a.copy()
    best_sc = compute_score(a)

    for it in range(1, max_iter + 1):
        if time.time() >= deadline:
            break

        conv = convolution_fft(a, a)
        grad = _max_conv_gradient(a, conv)

        # Adam moments on the (non‑negative) gradient
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)

        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        # diminishing step size (inverse‑sqrt schedule) multiplied by a base t0
        eta = t0 / (np.sqrt(it) + 1.0)

        # multiplicative (exponentiated) update, projected onto the simplex
        a = a * np.exp(-eta * m_hat / (np.sqrt(v_hat) + eps))
        s = a.sum()
        if s == 0.0:
            a = uniform_initial(len(a))
        else:
            a = a / s

        cur_sc = compute_score(a)
        if cur_sc < best_sc:
            best_sc = cur_sc
            best_a = a.copy()

    return best_a, best_sc


# ----------------------------------------------------------------------
# Simple stochastic neighbour move (used for a quick final polish)
# ----------------------------------------------------------------------


def propose_neighbor(a: np.ndarray,
                    rng: np.random.Generator,
                    max_step: float = 0.02) -> np.ndarray:
    """Transfer a small fraction of mass from one bin to another."""
    n = len(a)
    i, j = rng.integers(0, n, size=2)
    if i == j or a[i] == 0.0:
        return a
    delta = a[i] * rng.random() * max_step
    b = a.copy()
    b[i] -= delta
    b[j] += delta
    return b


def simulated_annealing(a0: np.ndarray,
                        deadline: float,
                        rng: np.random.Generator,
                        max_iter: int = 2000,
                        start_temp: float = 0.005,
                        cooling: float = 0.9995,
                        max_step: float = 0.02) -> Tuple[np.ndarray, float]:
    """Very light SA that respects the simplex."""
    a = a0.copy()
    cur_sc = compute_score(a)
    best_a = a.copy()
    best_sc = cur_sc
    temp = start_temp

    for it in range(max_iter):
        if time.time() >= deadline:
            break
        cand = propose_neighbor(a, rng, max_step)
        cand_sc = compute_score(cand)
        delta = cand_sc - cur_sc
        if delta < 0.0 or rng.random() < np.exp(-delta / max(1e-12, temp)):
            a = cand
            cur_sc = cand_sc
            if cur_sc < best_sc:
                best_sc = cur_sc
                best_a = a.copy()
        temp *= cooling
    return best_a, best_sc


# ----------------------------------------------------------------------
# Upscaling helper (linear interpolation)
# ----------------------------------------------------------------------


def upscale(a: np.ndarray, new_n: int) -> np.ndarray:
    """Interpolate a vector to a larger length while preserving the simplex."""
    old_n = len(a)
    if old_n == new_n:
        return a.copy()
    # centre points of the bins
    x_old = (np.arange(old_n) + 0.5) / old_n
    x_new = (np.arange(new_n) + 0.5) / new_n
    new = np.interp(x_new, x_old, a, left=0.0, right=0.0)
    new = np.maximum(new, 0.0)
    s = new.sum()
    if s == 0.0:
        new = uniform_initial(new_n)
    else:
        new = new / s
    return new

# Main entry point
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42,
                     budget_s: int = 1000,
                     **kwargs) -> List[float]:
    """
    Search for a non‑negative step‑function with a low ``evaluate_sequence`` value.
    The routine respects the wall‑clock budget and always returns a feasible list.
    """
    rng = np.random.default_rng(seed)
    start = time.time()
    # leave a small safety margin for the final evaluate_sequence call
    deadline = start + budget_s - 2.0

    best_seq: List[float] = []
    best_val: float = float("inf")

    # ------------------------------------------------------------------
    # 0.  Optional fallback if a global construction is already available.
    # ------------------------------------------------------------------
    try:
        fallback = [float(x) for x in height_sequence_1]  # type: ignore
        val = evaluate_sequence(fallback)
        print(f"Fallback (height_sequence_1) score = {val:.6f}")
        if val < best_val:
            best_val = val
            best_seq = fallback
    except Exception:
        pass

    # ------------------------------------------------------------------
    # 1.  Multi‑scale optimisation (64 → 128 → … → 2048).
    # ------------------------------------------------------------------
    sizes = [64, 128, 256, 512, 1024, 2048]   # powers of two – FFT‑friendly
    current: np.ndarray | None = None

    for n in sizes:
        if time.time() >= deadline - 5.0:
            break
        print(f"\n=== Resolution n = {n} ===")

        # ---- 1a. initialise ----
        if current is None:
            # generate a few cheap seeds and keep the best according to the fast score
            seeds = [
                arcsine_initial(n),
                uniform_initial(n),
                random_dirichlet(n, rng, conc=0.3),
                random_dirichlet(n, rng, conc=1.0),
                random_dirichlet(n, rng, conc=2.0),
            ]
            scores = [(compute_score(s), s) for s in seeds]
            scores.sort(key=lambda x: x[0])
            cand = scores[0][1].copy()
        else:
            # upscale the best vector from the previous scale
            cand = upscale(current, n)

        # ---- 1b. optional LP refinement (only for modest n) ----
        if linprog is not None and n <= 512:

            cand, _ = lp_improve(cand, deadline, max_iters=lp_iters, t=0.2)
            print(f"   after LP improvement score = {compute_score(cand):.6f}")

        # ---- 1c. Adam mirror‑descent on the true max‑convolution ----
        # allocate iterations roughly proportional to remaining time and inversely to n
        remaining = max(0.0, deadline - time.time())
        # we cap the number of iterations to keep the whole run inside the budget

        print(f"   after Adam descent   score = {compute_score(cand):.6f}")

                                          max_iter=sa_iters,
                                          start_temp=0.01,
                                          cooling=0.999,
                                          max_step=0.03)
            print(f"   after SA polishing   score = {compute_score(cand):.6f}")

        # ---- 1e. evaluate with the official expensive routine ----
        try:
            val = evaluate_sequence(cand.tolist())
            print(f"   official eval score = {val:.6f}")
            if val < best_val:
                best_val = val
                best_seq = cand.tolist()
                print(f"   *** New global best (score={best_val:.6f}) ***")
        except Exception as exc:
            print(f"   evaluation failed: {exc}")

        # keep the vector for the next scale
        current = cand.copy()

    # ------------------------------------------------------------------
    # 2. Final polishing on the best full‑size candidate (if time permits).
    # ------------------------------------------------------------------
    if best_seq and time.time() < deadline - 5.0:
        print("\nFinal polishing on the best full‑size candidate")
        a = np.array(best_seq, dtype=float)
        remaining = max(0.0, deadline - time.time())
        max_iter = int(min(150_000, remaining * 300))
        a, _ = adam_maxconv(a, deadline, max_iter=max_iter, t0=0.3)
        # one more brief SA sweep
        a, _ = simulated_annealing(a, deadline, rng, max_iter=2000,
                                   start_temp=0.008, cooling=0.998, max_step=0.02)
        final_val = evaluate_sequence(a.tolist())
        print(f"Final polished score = {final_val:.6f}")
        if final_val < best_val:
            best_val = final_val
            best_seq = a.tolist()
            print("*** Updated global best after final polishing ***")

    # ------------------------------------------------------------------
    # 3. Guard – always return something feasible.
    # ------------------------------------------------------------------
    if not best_seq:
        best_seq = uniform_initial(256).tolist()
        best_val = evaluate_sequence(best_seq)

    print(f"\nSearch finished. Best reported score = {best_val:.6f}")
    return best_seq
