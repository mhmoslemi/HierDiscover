def refine_adam(a, deadline):
    a = normalize(a)
    log_a = np.log(a + 1e-30)  # small epsilon
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    t = 0
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    best_a = a.copy()
    best_score = compute_score(a)

    while time.time() < deadline:
        t += 1
        # compute conv and max
        conv = convolution_fft(a, a)
        max_val = conv.max()
        # weight: softmax with temperature schedule (increase)
        # we can implement w_i = exp(beta * (conv - max_val)) / sum()
        # dynamic beta: increase gradually
        beta = min(30.0, 0.1 * t) # e.g., increase
        exp_vals = np.exp(beta * (conv - max_val))
        w = exp_vals / exp_vals.sum()
        grad = gradient_from_weights(a, w)
        # Adam update on log a (gradient of objective w.r.t a)
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad ** 2)
        m_hat = m / (1 - beta1 ** t)
        v_hat = v / (1 - beta2 ** t)
        step = m_hat / (np.sqrt(v_hat) + eps)
        # update log a
        log_a = log_a - (0.5/np.sqrt(t)) * step  # step size schedule
        # map to a and normalize
        a = np.exp(log_a)
        a = normalize(a)
        # compute actual objective (exact)
        cur_score = 2 * len(a) * conv.max()  # sum=1
        # may compute exact each iteration or only occasional
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
    return best_a, best_score
```

def refine_momentum(a, deadline, eta0=0.8, beta_start=5.0, beta_end=30.0, momentum_coeff=0.9):
    start = time.time()
    n = len(a)
    a = normalize(a.astype(float).copy())
    best_a = a.copy()
    best_sc = compute_score(a)
    grad_avg = np.zeros_like(a)

    it = 0
    while time.time() < deadline:
        it += 1
        # compute convolution and max
        conv = convolution_fft(a, a)
        max_val = float(np.max(conv))

        # compute temperature
        elapsed = time.time() - start
        total = deadline - start
        frac = elapsed / total if total > 0 else 1.0
        beta = beta_start + (beta_end - beta_start) * frac

        # compute softmax weights for gradient
        exp_vals = np.exp(beta * (conv - max_val))
        w = exp_vals / np.sum(exp_vals)

        grad = gradient_from_weights(a, w)
        # clip grad to avoid overflow
        grad = np.clip(grad, -50.0, 50.0)

        # momentum update
        grad_avg = momentum_coeff * grad_avg + (1 - momentum_coeff) * grad

        # step size
        eta = eta0 / np.sqrt(it)

        # exponentiated gradient update
        a = a * np.exp(-eta * grad_avg)
        a = normalize(a)

        cur_sc = 2.0 * n * max_val  # exact objective with sum=1
        if cur_sc < best_sc:
            best_sc = cur_sc
            best_a = a.copy()
    return best_a, best_sc

"""
Propose a non‑negative step‑function (list of heights) that minimises

        2·n·max(conv(a,a)) / (sum(a)²)

The search consists of three phases:

1. **Rich initial sweep** – generate a diverse set of candidates from
   analytic families (arcsine, power‑law, beta‑distribution, uniform,
   random Dirichlet) and, if present, the previously best construction
   `height_sequence_1`.  The fast FFT‑based score is used to keep the
   ten most promising candidates.

2. **Momentum mirror‑descent** – a multiplicative‑gradient (exponentiated‑
   gradient) update on the simplex with a *soft‑max* approximation of the
   max‑convolution.  The temperature β is increased linearly from a small
   value to a larger one, while the step size follows η₀/√t.  A momentum
   term (exponential moving average of the gradient) stabilises the
   dynamics.  The best iterate (exact objective) is retained.

3. **Hard sub‑gradient polishing** – a classic sub‑gradient method that
   uses the true max‑convolution (uniform weight on the maximisers) and
   the same momentum‑augmented multiplicative updates.  This stage
   consumes the remaining time budget.

Both refinement stages respect the global deadline and always return a
feasible probability vector (non‑negative, sum = 1).  The final vector is
passed to the official `evaluate_sequence` routine; the best one found
is returned.

The algorithm runs on a single CPU, uses only NumPy and standard
library modules, and stays well within the 1000 s time limit.
"""
import time
from typing import List, Tuple

import numpy as np

# ----------------------------------------------------------------------
# Low‑level numeric utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """
    Linear convolution via real FFT (full length, zero‑padded to a power‑of‑2).
    """
    m = len(a) + len(b) - 1
    n_fft = next_pow2(m)
    A = np.fft.rfft(a, n=n_fft)
    B = np.fft.rfft(b, n=n_fft)
    C = A * B
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


def normalize(v: np.ndarray) -> np.ndarray:
    """Project a non‑negative vector onto the simplex (sum = 1)."""
    total = v.sum()
    if total <= 0.0:
        return np.full_like(v, 1.0 / len(v))
    return v / total


def compute_score(a: np.ndarray) -> float:
    """
    Fast evaluation of the official objective (scale‑invariant):
        2·n·max(conv(a,a)) / (sum(a)²)
    The vector a is assumed non‑negative and already normalised (sum≈1).
    """
    n = len(a)
    conv = convolution_fft(a, a)
    max_val = float(np.max(conv))
    s = float(np.sum(a))
    if s < 1e-12:
        return float("inf")
    return 2.0 * n * max_val / (s * s)


def gradient_from_weights(a: np.ndarray, w: np.ndarray) -> np.ndarray:
    """
    Gradient of Σ_k w_k·(a*a)_k with respect to a.
    Implements   g_j = 2· Σ_m a_m · w_{m+j}.
    Computed via one FFT‑based convolution.
    """
    w_rev = w[::-1]
    # length of the convolution is len(a)+len(w)-1 = 3·n‑2
    conv = convolution_fft(a, w_rev)
    idx = 2 * len(a) - 2  # position of the last element of the full conv
    grads = 2.0 * conv[idx - np.arange(len(a))]
    return grads


# ----------------------------------------------------------------------
# Parametric families for the initial sweep
# ----------------------------------------------------------------------


def arcsine_initial(n: int) -> np.ndarray:
    """Discrete arcsine distribution (≈ 1/√(i·(n‑1‑i))) normalised."""
    i = np.arange(n, dtype=float)
    eps = 0.25
    vals = 1.0 / np.sqrt((i + 0.5) * (n - 0.5 - i) + eps)
    return normalize(vals)


def powerlaw_shape(n: int, p: float, eps: float) -> np.ndarray:
    """Symmetric power‑law shape a_i ∝ ((i+ε)(n‑1‑i+ε))⁻ᵖ."""
    i = np.arange(n, dtype=float)
    base = ((i + eps) * (n - 1 - i + eps)) ** (-p)
    return normalize(base)


def beta_shape(n: int, alpha: float, beta_param: float, eps: float = 1e-8) -> np.ndarray:
    """Beta‑distribution shape on a discrete grid, normalised."""
    x = np.linspace(0.0, 1.0, n, dtype=float)
    vals = (x + eps) ** (alpha - 1) * (1.0 - x + eps) ** (beta_param - 1)
    return normalize(vals)


def uniform_shape(n: int) -> np.ndarray:
    """Uniform probability vector on n points."""
    return np.full(n, 1.0 / n)


def random_dirichlet(n: int, rng: np.random.Generator) -> np.ndarray:
    """Draw a random point on the simplex (exponential sampling)."""
    samp = rng.exponential(1.0, size=n)
    return normalize(samp)

# Refinement methods

def refine_momentum(
    a: np.ndarray,
    deadline: float,
    *,
    eta0: float = 0.8,
    beta_start: float = 5.0,
    beta_end: float = 30.0,
    momentum_coeff: float = 0.9,
    report_every: int = 0,
) -> Tuple[np.ndarray, float]:
    """
    Exponentiated‑gradient with momentum and a linearly increasing
    temperature for the soft‑max approximation of the max‑convolution.
    Returns the best iterate (exact score) and its score.
    """
    a = normalize(a.astype(float).copy())
    n = len(a)
    best_a = a.copy()
    best_sc = compute_score(a)

    grad_avg = np.zeros_like(a)
    it = 0
    start = time.time()

    while time.time() < deadline:
        it += 1

        # convolution and max value
        conv = convolution_fft(a, a)
        max_val = float(np.max(conv))

        # temperature schedule
        elapsed = time.time() - start
        total = deadline - start
        frac = elapsed / total if total > 0 else 1.0
        beta = beta_start + (beta_end - beta_start) * frac

        # soft‑max weights (smooth approximation of the max)
        exp_vals = np.exp(beta * (conv - max_val))  # max entry gives 1
        w = exp_vals / np.sum(exp_vals)

        # gradient of the smoothed objective
        grad = gradient_from_weights(a, w)
        grad = np.clip(grad, -50.0, 50.0)  # protect the exponential

        # momentum (exponential moving average)
        grad_avg = momentum_coeff * grad_avg + (1.0 - momentum_coeff) * grad

        # step size schedule
        eta = eta0 / np.sqrt(it)

        # multiplicative update + simplex projection
        a = a * np.exp(-eta * grad_avg)
        a = normalize(a)

        cur_sc = 2.0 * n * max_val  # exact objective (sum≈1)
        if cur_sc < best_sc:
            best_sc = cur_sc
            best_a = a.copy()

        if report_every and (it % report_every == 0):
            print(f"    momentum iter={it:,d} cur={cur_sc:.6f} best={best_sc:.6f}")

    return best_a, best_sc


def refine_hard_subgradient(
    a: np.ndarray,
    deadline: float,
    *,
    max_iter: int = 200_000,
    eta0: float = 0.7,
    momentum_coeff: float = 0.9,
    report_every: int = 0,
) -> Tuple[np.ndarray, float]:
    """
    Classical sub‑gradient descent on the true max‑convolution objective,
    enhanced with momentum and an exponentiated‑gradient update.
    Returns the best vector seen and its exact score.
    """
    a = normalize(a.astype(float).copy())
    n = len(a)
    best_a = a.copy()
    best_sc = compute_score(a)

    grad_m = np.zeros_like(a)
    it = 0

    while time.time() < deadline and it < max_iter:
        it += 1

        conv = convolution_fft(a, a)
        max_val = float(np.max(conv))

        # weight = uniform on the (near) maximisers
        w = np.zeros_like(conv)
        max_idx = np.where(conv >= max_val - 1e-14)[0]
        w[max_idx] = 1.0 / len(max_idx)

        grad = gradient_from_weights(a, w)
        grad = np.clip(grad, -50.0, 50.0)

        # momentum
        grad_m = momentum_coeff * grad_m + (1.0 - momentum_coeff) * grad

        eta = eta0 / np.sqrt(it)

        a = a * np.exp(-eta * grad_m)
        a = normalize(a)

        cur_sc = 2.0 * n * max_val
        if cur_sc < best_sc:
            best_sc = cur_sc
            best_a = a.copy()

        if report_every and (it % report_every == 0):
            print(f"    sub‑grad iter={it:,d} cur={cur_sc:.6f} best={best_sc:.6f}")

    return best_a, best_sc

def propose_candidate(
    seed: int = 42,
    budget_s: int = 1000,
    **kwargs,
) -> List[float]:
    """
    Search for a non‑negative step‑function with a low ``evaluate_sequence``
    score.  The routine respects the time budget, builds a rich initial pool,
    refines the best candidate with momentum mirror‑descent and finishes with
    a hard sub‑gradient polish.  The best sequence found is returned.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    # tiny safety margin to avoid a hard timeout
    deadline = start_time + budget_s - 2.0

    # ------------------------------------------------------------------
    # 1.  Generate a diverse pool of initial candidates (length = 2048)
    # ------------------------------------------------------------------
    n = 2048
    candidates: List[np.ndarray] = []

    # a) previously discovered construction, if available
    try:
        fallback = np.array([float(x) for x in height_sequence_1], dtype=float)
        if fallback.ndim == 1:
            fallback = normalize(fallback)
            candidates.append(fallback)
            print("[fallback] length =", len(fallback))
    except NameError:
        pass

    # b) analytic families
    base_arc = arcsine_initial(n)
    base_uni = uniform_shape(n)
    candidates.append(base_arc)
    candidates.append(base_uni)

    # c) random Dirichlet draws
    for _ in range(4):
        candidates.append(random_dirichlet(n, rng))

    # d) power‑law shapes (sweep p and ε)
    for p in np.linspace(0.35, 0.65, 7):
        for eps in (0.0, 0.01 * n, 0.025 * n):
            candidates.append(powerlaw_shape(n, p, eps))

    # e) symmetric beta distributions (α = β)
    for alpha in np.linspace(0.35, 0.75, 5):
        candidates.append(beta_shape(n, alpha, alpha, eps=1e-6))

    # f) asymmetric beta distributions
    alpha_vals = [0.4, 0.6]
    beta_vals = [0.8, 1.2]
    for a_ in alpha_vals:
        for b_ in beta_vals:
            candidates.append(beta_shape(n, a_, b_, eps=1e-6))

    # g) convex mixtures of arcsine & uniform
    for lam in np.linspace(0.0, 0.5, 6):
        mix = (1.0 - lam) * base_arc + lam * base_uni
        candidates.append(normalize(mix))

    # ------------------------------------------------------------------
    # 2.  Fast ranking and keep the most promising ones
    # ------------------------------------------------------------------
    scored = [(compute_score(c), c) for c in candidates]
    scored.sort(key=lambda x: x[0])
    top_k = 10
    top_candidates = [c for _, c in scored[:top_k]]
    print(f"Initial best fast score = {scored[0][0]:.6f}")
    print(f"Exploring {top_k} top candidates with refinement")

    # ------------------------------------------------------------------
    # 3.  Allocate time: momentum stage (~30%) then hard sub‑gradient (~70%)
    # ------------------------------------------------------------------
    now = time.time()
    if now >= deadline:
        # fallback – should never happen
        return uniform_shape(256).tolist()
    remaining = deadline - now
    momentum_deadline = now + 0.30 * remaining
    subgrad_deadline = deadline

    # start from the absolute best fast candidate
    current = scored[0][1].copy()

    # ---------------------- momentum mirror‑descent --------------------
    refined_mom, mom_score = refine_momentum(
        a=current,
        deadline=momentum_deadline,
        eta0=0.8,
        beta_start=5.0,
        beta_end=30.0,
        momentum_coeff=0.9,
        report_every=0,
    )
    val = evaluate_sequence(refined_mom.tolist())
    print(f"After momentum refinement official score = {val:.6f}")

    best_seq = refined_mom.tolist()
    best_val = val

    # ------------------------ hard sub‑gradient -----------------------
    if time.time() < subgrad_deadline - 5.0:
        print("\nStarting hard sub‑gradient polishing")
        polished, _ = refine_hard_subgradient(
            a=refined_mom,
            deadline=subgrad_deadline,
            max_iter=300_000,
            eta0=0.7,
            momentum_coeff=0.9,
            report_every=0,
        )
        final_val = evaluate_sequence(polished.tolist())
        print(f"Polished official score = {final_val:.6f}")
        if final_val < best_val:
            best_val = final_val
            best_seq = polished.tolist()
            print("*** Updated global best after hard polish ***")

    # ------------------------------------------------------------------
    # 4.  Guard – ensure a feasible (non‑empty) sequence is returned.
    # ------------------------------------------------------------------
    if not best_seq:
        best_seq = uniform_shape(256).tolist()
        best_val = evaluate_sequence(best_seq)

    print(f"\nSearch finished. Best reported score = {best_val:.6f}")
    return best_seq
