def solve_lp_max_sum(a, M):
    n = len(a)
    A_toeplitz = scipy.linalg.toeplitz(np.concatenate([a, np.zeros(n-1)]), np.concatenate([[a[0]], np.zeros(n-1)]))
    # A_toeplitz shape: (2n-1, n)
    b = cp.Variable(n)
    constraints = [b >= 0, A_toeplitz @ b <= M]
    prob = cp.Problem(cp.Maximize(cp.sum(b)), constraints)
    prob.solve(solver=cp.ECOS, warm_start=True, max_iters=5000)
    if prob.status not in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE]:
        return None
    return b.value
```

from scipy.optimize import linprog

def solve_lp_max_sum_linear(a, M):
    n = len(a)
    # Build Toeplitz matrix: shape (2*n-1, n)
    first_col = np.concatenate([a, np.zeros(n-1)])
    first_row = np.concatenate([[a[0]], np.zeros(n-1)])
    A = scipy.linalg.toeplitz(first_col, first_row)
    # constraints A @ b <= M
    # Note: A shape (2n-1, n)
    # linprog expects A_ub @ x <= b_ub
    res = linprog(c=-np.ones(n), A_ub=A, b_ub=M * np.ones(2*n-1),
                  bounds=[(0, None)] * n,
                  method='highs',
                  options={'presolve': True, 'time_limit': 30}) # maybe time limit 30 secs
    if not res.success:
        return None
    return res.x
```

def propose_candidate(seed=...):
    set seed
    start time, deadline
    n = 256
    a = np.random.dirichlet(np.ones(n))
    best_score = compute_score(a)
    best_a = a.copy()
    iteration = 0

    while time.time() < deadline - 5:
        M = np.max(convolution_fft(a, a))
        # LP step
        b = solve_lp_max_sum_linear(a, M)
        if b is None:
            print("LP failed")
            break
        b_norm = b / np.sum(b)
        new_score = compute_score(b_norm)
        if new_score + 1e-8 < best_score:
            best_score = new_score
            best_a = b_norm.copy()
            a = b_norm
            print(f"LP iteration {iteration} improved score to {best_score:.6f}")
        else:
            # no improvement; break or try subgradient
            # try subgradient as local refinement
            a2, sg_score = exp_subgrad_optimize(a, time_deadline=deadline, max_iter=200_000, eta0=0.5)
            if sg_score < best_score:
                best_score = sg_score
                best_a = a2.copy()
                a = a2
                print(f"Subgrad further improved to {best_score:.6f}")
            else:
                # break if neither improves
                break
        iteration += 1
        if iteration >= 30: # limit iterations
            break

    # final hill climb
    if time.time() < deadline - 2:
        polished_seq, polished_val = hill_climb_permutation(best_a.tolist(), time_deadline=deadline, max_steps=200_000)
        if polished_val < best_score:
            best_score = polished_val
            best_a = np.array(polished_seq)
            print(f"Hill climb final improved to {best_score:.6f}")

    final_seq = best_a.tolist()
    print(f"\nSearch finished. Final reported score = {evaluate_sequence(final_seq):.6f}")
    return final_seq
```

""" 
Propose a low‑score step function by combining exponentiated sub‑gradient descent,
a linear‑programming “sum‑increase” step and a short stochastic hill‑climber.
The LP step finds a non‑negative vector that maximises the total mass while keeping
the current maximal autocorrelation unchanged (a*b ≤ M). After renormalising,
the maximal autocorrelation typically drops, yielding a lower evaluation score.
The whole search respects the 1000 s budget.
"""

import time
import numpy as np
import scipy.linalg
from scipy.optimize import linprog

# ----------------------------------------------------------------------
# Fast convolution utilities (real FFT)
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Linear convolution a*b using a real FFT (fast)."""
    m = len(a) + len(b) - 1
    n_fft = next_pow2(m)
    A = np.fft.rfft(a, n=n_fft)
    B = np.fft.rfft(b, n=n_fft)
    C = A * B
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


def max_convolution(a: np.ndarray) -> float:
    """Return max value of the linear autocorrelation a*a."""
    return float(np.max(convolution_fft(a, a)))


def compute_score(a: np.ndarray) -> float:
    """
    Fast version of `evaluate_sequence` for a vector that already sums to 1.
    Returns 2·n·max(a*a).
    """
    n = a.shape[0]
    conv = convolution_fft(a, a)
    return 2.0 * n * float(np.max(conv))


# ----------------------------------------------------------------------
# Euclidean projection onto the probability simplex (sum == 1, non‑negative)
# ----------------------------------------------------------------------


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the probability simplex.
    """
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / v.shape[0])

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, v.shape[0] + 1)
    cond = u - cssv / ind > 0
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


# ----------------------------------------------------------------------
# Exponentiated sub‑gradient descent on the exact objective
# ----------------------------------------------------------------------


def exp_subgrad_optimize(init_a: np.ndarray,
                         time_deadline: float,
                         max_iter: int = 5_000_000,
                         eta0: float = 0.5) -> tuple[np.ndarray, float]:
    """
    Perform exponentiated sub‑gradient descent on the exact objective
    f(a)=2·n·max(a*a).  The update is a ← a·exp(−η·g) where g is a
    sub‑gradient at the index giving the current maximal autocorrelation.
    After each update a is projected back onto the simplex.
    Returns the best vector seen and its score.
    """
    a = init_a.astype(float).copy()
    n = len(a)
    best_a = a.copy()
    best_score = compute_score(a)

    for t in range(1, max_iter + 1):
        if time.time() >= time_deadline:
            break

        conv = convolution_fft(a, a)
        k = int(np.argmax(conv))

        # sub‑gradient for the maximal convolution index k
        idx = np.arange(n)
        j = k - idx
        mask = (j >= 0) & (j < n)
        g = np.zeros_like(a)
        g[mask] = 2.0 * a[j[mask]]

        eta = eta0 / np.sqrt(t)

        # multiplicative update (always stays non‑negative)
        a = a * np.exp(-eta * g)

        # re‑normalise onto the simplex
        s = a.sum()
        if s == 0.0:
            a = np.full_like(a, 1.0 / n)
        else:
            a = a / s

        cur_score = compute_score(a)
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()

        # occasional jitter to help escape plateaus
        if t % 5000 == 0:
            jitter = np.random.randn(n) * (eta0 * 1e-4 / np.sqrt(t))
            a = project_to_simplex(a + jitter)

    return best_a, best_score


# ----------------------------------------------------------------------
# Linear programming “sum‑increase” step
# ----------------------------------------------------------------------


def solve_lp_max_sum_linear(a: np.ndarray, M: float) -> np.ndarray | None:
    """
    Solve
        maximise   Σ b_i
        subject to a * b ≤ M   (elementwise linear convolution)
                  b_i ≥ 0.
    Returns the optimal b (as a dense NumPy array) or None if the LP
    fails.
    """
    n = len(a)
    # Build Toeplitz matrix A such that (A @ b)[i] = (a * b)[i]
    first_col = np.concatenate([a, np.zeros(n - 1)])
    first_row = np.concatenate([[a[0]], np.zeros(n - 1)])
    A = scipy.linalg.toeplitz(first_col, first_row)  # shape (2n‑1, n)

    # Linear constraints: A @ b ≤ M
    ub = np.full(2 * n - 1, M, dtype=float)

    # linprog solves minimise cᵀx; we maximise Σb → minimise −Σb
    c = -np.ones(n, dtype=float)

    # Bounds: b_i ≥ 0, no explicit upper bound
    bounds = [(0, None) for _ in range(n)]

    # Use the HiGHS interior‑point solver (fast for medium‑size LPs)
    res = linprog(c, A_ub=A, b_ub=ub, bounds=bounds,
                  method='highs', options={'presolve': True})
    if res.success:
        return res.x
    return None


# ----------------------------------------------------------------------
# Stochastic hill‑climber (swap / mass‑transfer moves)
# ----------------------------------------------------------------------


def hill_climb_permutation(seq: list[float],
                           time_deadline: float,
                           max_steps: int = 200_000) -> tuple[list[float], float]:
    """
    Simple stochastic hill‑climber that swaps two entries or moves a
    fraction of mass from one position to another.  The walk stays on the
    probability simplex (sum = 1) and always accepts moves that lower the
    true score.
    """
    a = np.array(seq, dtype=float)
    n = a.shape[0]

    best_a = a.copy()
    best_score = compute_score(best_a)

    step_scale = 0.1  # typical fraction of mass moved

    while max_steps > 0 and time.time() < time_deadline:
        move_type = np.random.rand()
        cand = best_a.copy()

        if move_type < 0.5:
            i, j = np.random.randint(n, size=2)
            cand[i], cand[j] = cand[j], cand[i]
        else:
            i, j = np.random.randint(n, size=2)
            if i != j:
                delta = step_scale * best_a[i]
                cand[i] -= delta
                cand[j] += delta
                cand[i] = max(cand[i], 0.0)

        cand = project_to_simplex(cand)
        cur_score = compute_score(cand)

        if cur_score < best_score:
            best_score = cur_score
            best_a = cand
            step_scale = min(step_scale * 1.05, 0.5)   # successful → larger steps
        else:
            step_scale *= 0.97                         # otherwise shrink

        max_steps -= 1

    return best_a.tolist(), best_score

        cur_score = compute_score(a)
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()

        # occasional jitter to escape plateaus
        if t % 5000 == 0:
            jitter = np.random.randn(n) * (eta0 * 1e-4 / np.sqrt(t))
            a = project_to_simplex(a + jitter)

    return best_a, best_score


# ----------------------------------------------------------------------
# Linear‑programming “sum‑increase’’ step
# ----------------------------------------------------------------------


def solve_lp_max_sum_linear(a: np.ndarray, M: float) -> np.ndarray | None:
    """
    Solve
        maximise   Σ b_i
        subject to a * b ≤ M   (element‑wise linear convolution)
                  b_i ≥ 0.
    Returns the optimal b or None if the LP fails.
    """
    n = len(a)

    # Build the Toeplitz matrix encoding the convolution a * b.
    first_col = np.concatenate([a, np.zeros(n - 1)])
    first_row = np.concatenate([[a[0]], np.zeros(n - 1)])
    A = scipy.linalg.toeplitz(first_col, first_row)  # (2n‑1) × n

    # Linear constraints: A @ b ≤ M
    ub = np.full(2 * n - 1, M, dtype=float)

    # linprog minimises cᵀx, so we maximise Σb via c = -1.
    c = -np.ones(n, dtype=float)
    bounds = [(0, None) for _ in range(n)]

    res = linprog(c, A_ub=A, b_ub=ub, bounds=bounds,
                  method='highs', options={'presolve': True})
    if res.success:
        return res.x
    return None


# ----------------------------------------------------------------------
# Stochastic hill‑climber (swap / mass‑transfer moves)
# ----------------------------------------------------------------------


def hill_climb_permutation(seq: list[float],
                           time_deadline: float,
                           max_steps: int = 200_000) -> tuple[list[float], float]:
    """
    Simple hill‑climber that swaps two entries or moves a fraction of mass.
    The walk stays on the probability simplex (sum = 1) and always accepts
    moves that lower the true objective.
    """
    a = np.array(seq, dtype=float)
    n = a.shape[0]

    best_a = a.copy()
    best_score = compute_score(best_a)

    step_scale = 0.1  # typical fraction of mass moved in a transfer

    while max_steps > 0 and time.time() < time_deadline:
        # choose move type
        move_type = np.random.rand()
        cand = best_a.copy()

        if move_type < 0.5:
            # swap two positions
            i, j = np.random.randint(n, size=2)
            cand[i], cand[j] = cand[j], cand[i]
        else:
            # transfer a fraction from i to j
            i, j = np.random.randint(n, size=2)
            if i != j:
                delta = step_scale * best_a[i]
                cand[i] -= delta
                cand[j] += delta
                cand[i] = max(cand[i], 0.0)

        # re‑project onto the simplex
        cand = project_to_simplex(cand)
        cur_score = compute_score(cand)

        if cur_score < best_score:
            best_score = cur_score
            best_a = cand
            step_scale = min(step_scale * 1.05, 0.5)
        else:
            step_scale *= 0.97

        max_steps -= 1

    return best_a.tolist(), best_score


# ----------------------------------------------------------------------
# Main entry point required by the evaluation harness
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative sequence that minimises `evaluate_sequence`.
    The search proceeds through:
        * a random Dirichlet start (length 256),
        * exponentiated sub‑gradient descent,
        * repeated LP “sum‑increase’’ steps (with occasional sub‑gradient
          fine‑tuning),
        * a final hill‑climber polishing.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 1.0  # tiny safety margin

    best_seq = None
    best_val = float("inf")

    # ------------------------------------------------------------------
    # 0.  Use supplied fallback if it exists (may already be good)
    # ------------------------------------------------------------------
    try:
        fallback = [float(x) for x in height_sequence_1]  # type: ignore
        fallback_val = evaluate_sequence(fallback)
        print(f"Fallback score = {fallback_val:.6f}")
        if fallback_val < best_val:
            best_val = fallback_val
            best_seq = fallback
    except Exception:
        pass

    # ------------------------------------------------------------------
    # 1. Random start (length 256)
    # ------------------------------------------------------------------
    n = 256
    a = np.random.dirichlet(np.ones(n))
    cur_val = compute_score(a)
    if cur_val < best_val:
        best_val = cur_val
        best_seq = a.tolist()
        print(f"Random start (n={n}) score = {best_val:.6f}")

    # ------------------------------------------------------------------
    # Main optimisation loop (only if we have a viable start)
    # ------------------------------------------------------------------
    if best_seq is not None:
        a = np.array(best_seq, dtype=float)

    # --------------------------------------------------------------
    # 2. Exponentiated sub‑gradient descent – quick flattening
    # --------------------------------------------------------------
    if time.time() < deadline - 30.0:
        print("\nRunning exponentiated sub‑gradient descent")
        a, sg_score = exp_subgrad_optimize(
            a,
            time_deadline=deadline - 20.0,
            max_iter=3_000_000,
            eta0=0.6,
        )
        if sg_score < best_val:
            best_val = sg_score
            best_seq = a.tolist()
            print(f"Sub‑gradient improved best to {best_val:.6f}")

    # --------------------------------------------------------------
    # 3. Linear‑programming “sum‑increase’’ loop
    # --------------------------------------------------------------
    lp_iter = 0
    while time.time() < deadline - 5.0:
        M = max_convolution(a)
        b = solve_lp_max_sum_linear(a, M)
        if b is None:
            print("LP solver failed – exiting LP loop")
            break

        b_norm = b / np.sum(b)                # re‑normalise to sum = 1
        new_score = compute_score(b_norm)

        if new_score + 1e-9 < best_val:
            # Accept the LP‑produced vector
            best_val = new_score
            best_seq = b_norm.tolist()
            a = b_norm
            lp_iter += 1
            print(f"LP iteration {lp_iter} improved score to {best_val:.6f}")

            # A short sub‑gradient burst after a successful LP often helps
            if time.time() < deadline - 5.0:
                a, sg_score = exp_subgrad_optimize(
                    a,
                    time_deadline=deadline - 2.0,
                    max_iter=200_000,
                    eta0=0.5,
                )
                if sg_score < best_val:
                    best_val = sg_score
                    best_seq = a.tolist()
                    print(f"  Sub‑gradient fine‑tune improved to {best_val:.6f}")
            continue
        else:
            # No improvement – try a few sub‑gradient steps to escape plateau
            a_tmp, sg_score2 = exp_subgrad_optimize(
                a,
                time_deadline=deadline - 2.0,
                max_iter=200_000,
                eta0=0.5,
            )
            if sg_score2 < best_val - 1e-9:
                best_val = sg_score2
                best_seq = a_tmp.tolist()
                a = a_tmp
                print(f"  Sub‑gradient after stalled LP improved to {best_val:.6f}")
                continue
            else:
                print("LP step stalled and sub‑gradient gave no gain – leaving LP loop")
                break

    # --------------------------------------------------------------
    # 4. Final hill‑climber polishing
    # --------------------------------------------------------------
    if best_seq is not None and time.time() < deadline - 1.0:
        print("\nRunning final hill‑climber")
        polished_seq, polished_val = hill_climb_permutation(
            best_seq,
            time_deadline=deadline,
            max_steps=200_000,
        )
        if polished_val < best_val:
            best_val = polished_val
            best_seq = polished_seq
            print(f"Hill‑climber improved final score to {best_val:.6f}")

    # --------------------------------------------------------------
    # Safety fallback – tiny uniform vector if everything else failed
    # --------------------------------------------------------------
    if best_seq is None:
        n_small = 128
        best_seq = [1.0 / n_small] * n_small
        best_val = evaluate_sequence(best_seq)

    # --------------------------------------------------------------
    # Report final result
    # --------------------------------------------------------------
    final_score = evaluate_sequence(best_seq)
    print(f"\nSearch finished. Final reported score = {final_score:.6f}")
    return best_seq
