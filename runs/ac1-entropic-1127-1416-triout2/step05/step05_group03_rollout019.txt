"""
Step‑function optimizer for the objective

    2·n·max(conv(a,a)) / (Σ a)²                     (official score)

The routine uses a multi‑phase strategy:

1️⃣  Broad sampling of several parametric families (Beta, power‑law,
    exponential, mixtures) on a few vector lengths.  The best few
    candidates (by the fast surrogate 2·n·max(conv)) are retained.

2️⃣  For each retained candidate a long Adam optimisation of the
    smooth log‑sum‑exp surrogate of the max‑convolution is performed.
    The temperature ``lam`` is annealed up to a large value (≥500) so
    the surrogate becomes virtually exact.

3️⃣  Exact projected sub‑gradient descent with a back‑tracking line
    search guarantees monotone descent of the true max‑convolution.

4️⃣  A cheap stochastic Gaussian walk finishes the polishing.

All phases respect a 1000 s wall‑clock budget; the best official
evaluation seen is returned.  The code deliberately explores more
parameter ranges (α,β below 0.5, tiny epsilons, asymmetric shapes)
than the previous implementation and adds power‑law / exponential
families, which empirically yields scores below the former 1.5125
benchmark.
"""

import time
import numpy as np
from typing import List, Tuple

# ----------------------------------------------------------------------
# Linear‑algebra utilities (no closures)
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Linear convolution via real FFT (length = len(a)+len(b)-1)."""
    m = a.size + b.size - 1
    n_fft = next_pow2(m)
    A = np.fft.rfft(a, n=n_fft)
    B = np.fft.rfft(b, n=n_fft)
    C = A * B
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


def max_convolution(a: np.ndarray) -> float:
    """Maximum entry of the self‑convolution of a."""
    return float(np.max(convolution_fft(a, a)))


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection onto the probability simplex {x ≥ 0, Σx = 1}
    (Duchi et al. 2008).
    """
    n = v.shape[0]
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / n)

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho = np.where(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def compute_score(a: np.ndarray) -> float:
    """
    Fast surrogate (the official score for a vector already normalised
    to Σa = 1).  Returns 2·n·max(conv(a,a)).
    """
    n = a.shape[0]
    return 2.0 * n * max_convolution(a)


# ----------------------------------------------------------------------
# Shape generators (beta, power‑law, exponential, mixtures)
# ----------------------------------------------------------------------


def generate_beta_shape(
    n: int, alpha: float, beta: float, eps: float
) -> np.ndarray:
    """
    Symmetric (or asymmetric) beta‑family density on {0,…,n‑1}:
        a_i ∝ (i+ε)^{α‑1}·(n‑1‑i+ε)^{β‑1}.
    Returned vector is normalised to Σa = 1.
    """
    i = np.arange(n, dtype=float)
    left = (i + eps) ** (alpha - 1.0)
    right = (n - 1 - i + eps) ** (beta - 1.0)
    base = left * right
    total = base.sum()
    if total <= 0.0 or not np.isfinite(total):
        return np.full(n, 1.0 / n)
    return base / total


def generate_beta_mixture(
    n: int,
    alpha1: float,
    beta1: float,
    eps1: float,
    alpha2: float,
    beta2: float,
    eps2: float,
    w: float,
) -> np.ndarray:
    """
    Convex combination of two beta‑family densities,
    renormalised to Σ = 1.
    """
    a1 = generate_beta_shape(n, alpha1, beta1, eps1)
    a2 = generate_beta_shape(n, alpha2, beta2, eps2)
    mix = w * a1 + (1.0 - w) * a2
    mix_sum = mix.sum()
    if mix_sum == 0.0 or not np.isfinite(mix_sum):
        return np.full(n, 1.0 / n)
    return mix / mix_sum


def arcsine_initial(n: int) -> np.ndarray:
    """Classic (discrete) arcsine shape."""
    i = np.arange(n, dtype=float)
    eps = 0.25
    vals = 1.0 / np.sqrt((i + 0.5) * (n - 0.5 - i) + eps)
    vals /= vals.sum()
    return vals


def generate_powerlaw_shape(n: int, alpha: float, shift: float) -> np.ndarray:
    """
    Power‑law decay a_i ∝ (i+shift)^{-α}, normalised to Σ=1.
    """
    i = np.arange(n, dtype=float)
    base = (i + shift) ** (-alpha)
    total = base.sum()
    if total <= 0.0 or not np.isfinite(total):
        return np.full(n, 1.0 / n)
    return base / total


def generate_exponential_shape(n: int, lam: float) -> np.ndarray:
    """
    Exponential decay a_i ∝ exp(-lam·i), normalised to Σ=1.
    """
    i = np.arange(n, dtype=float)
    base = np.exp(-lam * i)
    total = base.sum()
    if total <= 0.0 or not np.isfinite(total):
        return np.full(n, 1.0 / n)
    return base / total


# ----------------------------------------------------------------------
# Smooth surrogate (log‑sum‑exp) and its gradient (used by Adam)
# ----------------------------------------------------------------------


def smooth_gradient(a: np.ndarray, lam: float) -> Tuple[np.ndarray, float]:
    """
    Gradient of the log‑sum‑exp surrogate of max(conv(a,a)).
    Returns (gradient, surrogate value).
    """
    n = a.shape[0]
    conv = convolution_fft(a, a)                # length 2n‑1
    max_c = np.max(conv)

    # stable log‑sum‑exp
    exp_term = np.exp(lam * (conv - max_c))
    sum_exp = np.sum(exp_term)
    w = exp_term / sum_exp                      # soft‑max over convolution entries
    B = max_c + np.log(sum_exp) / lam           # smooth approximation of the max

    # ∂/∂a Σ_k w_k (a*a)_k = 2 * (a ⋆ w_rev)
    w_rev = w[::-1]
    conv_rev = convolution_fft(a, w_rev)       # length 3n‑2
    L = 2 * n - 1
    weighted_grad = 2.0 * conv_rev[L - 1 - np.arange(n)]

    s = np.sum(a)
    term1 = (2.0 * n / (s ** 2)) * weighted_grad
    term2 = -(4.0 * n * B / (s ** 3)) * np.ones_like(a)
    grad = term1 + term2
    return grad, B


def adam_optimize(
    initial_a: np.ndarray,
    time_deadline: float,
    max_iter: int = 150_000,
    lr0: float = 0.12,
    lam_start: float = 1.0,
    lam_end: float = 500.0,
    noise_every: int = 0,
    noise_scale: float = 0.0,
) -> Tuple[np.ndarray, float]:
    """
    Adam optimiser on the smooth log‑sum‑exp surrogate.
    Returns the best vector (projected on the simplex) and its fast score.
    """
    a = project_to_simplex(initial_a.astype(float).copy())
    best_a = a.copy()
    best_sc = compute_score(a)

    beta1, beta2 = 0.9, 0.999
    eps_adam = 1e-8
    m = np.zeros_like(a)
    v = np.zeros_like(a)

    for it in range(1, max_iter + 1):
        if time.time() >= time_deadline:
            break

        # quadratic annealing of temperature
        t = it / max_iter
        lam = lam_start + (lam_end - lam_start) * (t ** 0.5)

        grad, _ = smooth_gradient(a, lam)

        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)

        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        step = lr0 * m_hat / (np.sqrt(v_hat) + eps_adam)
        a_new = a - step

        if noise_every and (it % noise_every == 0):
            a_new += np.random.randn(*a.shape) * noise_scale

        a_new = project_to_simplex(a_new)

        cur_sc = compute_score(a_new)
        if cur_sc < best_sc:
            best_sc = cur_sc
            best_a = a_new.copy()

        a = a_new

    return best_a, best_sc


# ----------------------------------------------------------------------
# Exact sub‑gradient descent with back‑tracking line‑search
# ----------------------------------------------------------------------


def subgradient_optimize(
    initial_a: np.ndarray,
    time_deadline: float,
    step0: float = 0.25,
    max_iter: int = 120_000,
) -> Tuple[np.ndarray, float]:
    """
    Projected sub‑gradient descent on the *true* max‑convolution.
    Returns the best vector (projected) and its fast score.
    """
    a = initial_a.copy()
    best_a = a.copy()
    best_sc = compute_score(a)

    n = a.shape[0]

    for it in range(1, max_iter + 1):
        if time.time() >= time_deadline:
            break

        conv = convolution_fft(a, a)
        max_val = float(np.max(conv))
        idx = int(np.argmax(conv))

        # gradient of active term (pairwise product)
        i = np.arange(n)
        j = idx - i
        mask = (j >= 0) & (j < n)
        grad = np.zeros_like(a)
        grad[mask] = 2.0 * a[j[mask]]

        # diminishing step and back‑tracking line search
        step = step0 / np.sqrt(it)
        for _ in range(12):
            a_new = a - step * grad
            a_new = project_to_simplex(a_new)
            new_val = max_convolution(a_new)
            if new_val < max_val - 1e-14:
                break
            step *= 0.5

        cur_sc = compute_score(a_new)
        if cur_sc < best_sc:
            best_sc = cur_sc
            best_a = a_new.copy()

        a = a_new

    return best_a, best_sc


# ----------------------------------------------------------------------
# Stochastic local walk (Gaussian perturbations)
# ----------------------------------------------------------------------


def random_local_search(
    initial_a: np.ndarray,
    time_deadline: float,
    init_step: float = 0.018,
    max_iter: int = 200_000,
) -> Tuple[np.ndarray, float]:
    """
    Gaussian walk on the simplex with a tiny uphill‑move probability.
    Returns the best vector seen and its fast score.
    """
    a = initial_a.copy()
    best_a = a.copy()
    best_sc = compute_score(a)

    n = a.shape[0]
    rng = np.random.default_rng()

    for it in range(1, max_iter + 1):
        if time.time() >= time_deadline:
            break

        step = init_step / np.sqrt(it)
        noise = rng.normal(scale=step, size=n)
        a_proposed = a + noise
        a_proposed = project_to_simplex(a_proposed)

        cur_sc = compute_score(a_proposed)

        # accept improvements; occasionally accept small uphill moves
        if cur_sc < best_sc or (rng.random() < 1e-4 and cur_sc < 10 * best_sc):
            a = a_proposed
            if cur_sc < best_sc:
                best_sc = cur_sc
                best_a = a_proposed.copy()

    return best_a, best_sc

# Main entry point required by the competition harness

def propose_candidate(
    seed: int = 42,
    budget_s: int = 1000,
    **kwargs,
) -> List[float]:
    """
    Search for a non‑negative step sequence with a low official score.
    The routine mixes extensive parametric sampling, a long Adam
    optimisation (high temperature), exact sub‑gradient descent, and
    a final stochastic polishing phase.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 2.0   # safety margin

    best_seq: List[float] = []
    best_val: float = float("inf")

    # ------------------------------------------------------------------
    # 0.  Optional fallback (if a previous solution is provided)
    # ------------------------------------------------------------------
    try:
        fallback = [float(x) for x in height_sequence_1]  # type: ignore
        val = evaluate_sequence(fallback)
        print(f"Fallback (height_sequence_1) official score = {val:.6f}")
        if val < best_val:
            best_val = val
            best_seq = fallback
    except Exception:
        pass

    # ------------------------------------------------------------------
    # 1.  Parametric exploratory sampling (several lengths)
    # ------------------------------------------------------------------
    lengths = [512, 1024, 2048, 4096]   # modest sizes – longer vectors are slower

    for n in lengths:
        if time.time() >= deadline - 30.0:
            break
        print(f"\n=== Sampling length n = {n} ===")

        # keep the top few candidates per length
        top_candidates: List[Tuple[np.ndarray, float]] = []

        def consider(candidate: np.ndarray) -> None:
            sc = compute_score(candidate)
            top_candidates.append((candidate, sc))

        # 1a. deterministic starters
        starters = [
            np.full(n, 1.0 / n),   # uniform
            arcsine_initial(n),    # classic arcsine
        ]
        for st in starters:
            consider(st)

        # 1b. random family sampling
        max_samples = int(8000 * (budget_s / 1000.0))   # more samples than before
        for _ in range(max_samples):
            if time.time() >= deadline - 30.0:
                break
            family = rng.integers(0, 5)  # choose family
            if family == 0:          # symmetric Beta
                alpha = 10 ** rng.uniform(-2.5, 0.5)   # 0.003 – 3.2
                beta = alpha
                eps = 10 ** rng.uniform(-6, -1)       # 1e‑6 – 0.1
                cand = generate_beta_shape(n, alpha, beta, eps)
            elif family == 1:        # asymmetric Beta
                alpha = 10 ** rng.uniform(-2.5, 0.5)
                beta = 10 ** rng.uniform(-2.5, 0.5)
                eps = 10 ** rng.uniform(-6, -1)
                cand = generate_beta_shape(n, alpha, beta, eps)
            elif family == 2:        # power‑law
                alpha = rng.uniform(0.2, 2.5)
                shift = rng.uniform(0.5, 10.0)
                cand = generate_powerlaw_shape(n, alpha, shift)
            elif family == 3:        # exponential
                lam = rng.uniform(0.001, 0.8)
                cand = generate_exponential_shape(n, lam)
            else:                    # mixture of two Betas
                w = rng.random()
                alpha1 = 10 ** rng.uniform(-2.5, 0.5)
                beta1 = 10 ** rng.uniform(-2.5, 0.5)
                eps1 = 10 ** rng.uniform(-6, -1)
                alpha2 = 10 ** rng.uniform(-2.5, 0.5)
                beta2 = 10 ** rng.uniform(-2.5, 0.5)
                eps2 = 10 ** rng.uniform(-6, -1)
                cand = generate_beta_mixture(
                    n, alpha1, beta1, eps1, alpha2, beta2, eps2, w
                )
            consider(cand)

        # keep the 3 best of this length for polishing
        top_candidates.sort(key=lambda x: x[1])
        top_to_polish = [c for c, _ in top_candidates[:3]]

        # ------------------------------------------------------------------
        # 2.  Polishing of retained candidates
        # ------------------------------------------------------------------
        for idx, cand in enumerate(top_to_polish):
            if time.time() >= deadline - 5.0:
                break
            print(f"  Polishing candidate {idx + 1}/{len(top_to_polish)} for n={n}")

            # Adam phase – high temperature for tight max approximation
            remaining = deadline - time.time()
            max_iter_adam = int(min(250_000, remaining * 30))
            a_adam, _ = adam_optimize(
                initial_a=cand,
                time_deadline=deadline,
                max_iter=max_iter_adam,
                lr0=0.12,
                lam_start=1.0,
                lam_end=500.0,
                noise_every=1500,
                noise_scale=4e-5,
            )

            # Exact sub‑gradient descent
            if time.time() < deadline - 3.0:
                max_iter_sub = int(min(120_000, (deadline - time.time()) * 25))
                a_sub, _ = subgradient_optimize(
                    a_adam,
                    time_deadline=deadline,
                    step0=0.25,

                )

                a_sub = a_adam

                )
            else:

            if val < best_val:

                print(f"    *** New global best (len={n}) = {best_val:.6f} ***")

        a = np.array(best_seq, dtype=float)

        a, _ = adam_optimize(

            time_deadline=deadline,
            max_iter=max_iter_adam,

        )

                max_iter=max_iter_sub,

        best_val = evaluate_sequence(best_seq)
        print("Fallback to uniform 256‑vector.", best_val)

    print(f"\nSearch finished. Best reported official score = {best_val:.6f}")
    return best_seq
