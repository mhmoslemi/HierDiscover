def optimise_adam(initial_a, beta_schedule, steps, deadline):
    a = initial_a.copy()
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    t = 0
    best_a = a.copy()
    best_obj = compute_score(a)  # actual objective based on max conv (maybe compute surrogate)
    while t < steps and time.time() < deadline:
        t += 1
        # compute conv of a with itself
        conv = convolution_fft(a, a) # full conv length 2n-1
        # compute smoothed max and gradient
        # using beta = beta_schedule(t)
        beta = beta_schedule(t)
        max_conv = conv.max()
        # compute softmax weights w = exp(beta*(conv - max_conv)) / sum_exp
        # for stability
        # store w array (2n-1)
        exp_term = np.exp(beta * (conv - max_conv))
        sum_exp = exp_term.sum()
        w = exp_term / sum_exp  # shape 2n-1
        # compute gradient: grad = 2 * (a * w_rev). (as earlier)
        w_rev = w[::-1]
        # convolution a with w_rev
        # Use FFT to compute conv_rev = np.convolve(a, w_rev) but faster via FFT
        conv_rev = convolution_fft(a, w_rev)  # length n + 2n-1 -1 = 3n-2, same function works.
        L = len(conv)  # 2n-1
        # extract slice
        start = len(a) - 1  # n-1
        end = start + len(a)  # 2n-1? Wait: conv_rev length = len(a)+len(w_rev)-1 = n + (2n-1) - 1 = 3n-2
        # The slice we need is conv_rev[start:end][::-1]
        grad = 2.0 * conv_rev[start:end][::-1]  # shape n
        # Now Adam update
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad ** 2)
        m_hat = m / (1 - beta1 ** t)
        v_hat = v / (1 - beta2 ** t)
        # step size: eta_t = lr / (sqrt(v_hat) + eps)
        a = a - eta_t * m_hat / (np.sqrt(v_hat) + 1e-8)
        # enforce non-neg and project onto simplex
        # we could allow small negative values then clip to zero and project.
        # Use simple projection.
        a = np.maximum(a, 0)
        a_sum = a.sum()
        if a_sum == 0:
            a = np.full_like(a, 1.0 / len(a))
        else:
            a /= a_sum

        # optionally compute surrogate objective (max conv) after a few steps
        # Evaluate actual evaluate_sequence rarely.
        if t % eval_interval == 0:

                best_obj = cur_obj
                best_a = a.copy()
                print(f"Iter {t}: best surrogate {best_obj:.6f}")

    return best_a, best_obj

def adam_optimize(init_a, max_iter, deadline, lr=0.1, beta1=0.9, beta2=0.999, eps=1e-8, schedule='linear', beta0=10, beta_max=5000):
   a = init_a.copy()
   m = np.zeros_like(a); v = np.zeros_like(a)
   best_a = a.copy()
   best_obj = compute_score(a)  # same as 2n maxconv (since sum=1)
   n = len(a)
   L = 2*n - 1
   for t in range(1, max_iter+1):
        if time.time() >= deadline:
            break
        conv = convolution_fft(a, a)
        # choose beta
        if schedule == 'exp':
            progress = t / max_iter
            beta = beta0 * (beta_max / beta0) ** progress
        else:
            progress = t / max_iter
            beta = beta0 + (beta_max - beta0) * progress
        # compute weights for softmax
        max_conv = conv.max()
        # compute stable softmax
        exp_term = np.exp(beta * (conv - max_conv))
        # sum may underflow if beta large, but max shift ensures exp_term <= 1. (since conv - max_conv <= 0)
        sum_exp = exp_term.sum()
        w = exp_term / sum_exp
        # gradient
        w_rev = w[::-1]
        # convolve a with w_rev
        conv_rev = convolution_fft(a, w_rev)  # length n+L-1 = 3n-2
        # slice
        start = n-1
        grad = 2.0 * conv_rev[start:start+n][::-1]  # shape n
        # Adam updates
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad * grad)
        m_hat = m / (1 - beta1 ** t)
        v_hat = v / (1 - beta2 ** t)
        a = a - lr * m_hat / (np.sqrt(v_hat) + eps)
        # clip and project onto simplex
        a = np.maximum(a, 0.0)
        sum_a = a.sum()
        if sum_a <= 0:
            a = np.full_like(a, 1.0 / n)
        else:
            a = a / sum_a
        # track best surrogate
        current_obj = compute_score(a)
        if current_obj < best_obj:
            best_obj = current_obj
            best_a = a.copy()
        # optional prints every some iterations
        if t % 5000 == 0:
            print(f"    Adam iter {t}/{max_iter}, beta={beta:.1f}, best surrogate {best_obj:.6f}")
   return best_a, best_obj

def convolution_fft(a,b):
    n = len(a) + len(b) - 1
    N = next_pow2(n)
    A = np.fft.rfft(a, n=N)
    B = np.fft.rfft(b, n=N)
    C = A * B
    conv = np.fft.irfft(C, n=N)[:n]
    return conv

"""
Propose a low‑value step‑function using a smoothed max‑convolution objective.

Algorithm outline
-----------------
* The true objective is 2·n·max(a∗a) / (sum(a)^2).  By keeping the vector on the
  probability simplex (sum = 1, a_i ≥ 0) the denominator is 1, so we only have
  to minimise the maximal entry of the self‑convolution.
* A smooth approximation of the max (log‑sum‑exp) is used:
      fβ(a) = (1/β)·log Σ_k exp(β·(a∗a)_k)
  Its gradient can be expressed in closed form using a single FFT‑based
  convolution, which makes each optimisation step O(n log n).
* Adam (adaptive‑momentum) updates are performed on the smooth objective,
  followed by a projection onto the probability simplex (clip ≥0, renormalise).
* Optimisation proceeds hierarchically: start with a modest length,
  optimise, linearly up‑sample the best vector and continue on a larger grid.
  This exploits the fact that the optimal continuous distribution (arcsine)
  becomes more accurate when the discretisation is refined.
* Only a few calls to the expensive ``evaluate_sequence`` are made – after
  each length and for a final polish – so the wall‑clock budget is respected.
"""

import time
import numpy as np
from typing import List, Tuple

# ----------------------------------------------------------------------
# FFT helpers
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Return the smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray, n_fft: int = None) -> np.ndarray:
    """
    Full linear convolution using a real FFT.
    If ``n_fft`` is provided, the transform size is forced to that value.
    """
    if n_fft is None:
        n_fft = next_pow2(len(a) + len(b) - 1)
    A = np.fft.rfft(a, n=n_fft)
    B = np.fft.rfft(b, n=n_fft)
    C = A * B
    conv = np.fft.irfft(C, n=n_fft)
    return conv[: len(a) + len(b) - 1]


# ----------------------------------------------------------------------
# Simple families of starting shapes
# ----------------------------------------------------------------------


def arcsine_initial(n: int) -> np.ndarray:
    """Discrete arcsine density a_i ∝ 1/√((i+½)(n‑½‑i))."""
    i = np.arange(n, dtype=float)
    eps = 0.25
    vals = 1.0 / np.sqrt((i + 0.5) * (n - 0.5 - i) + eps)
    return vals / vals.sum()


def random_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    """Draw a Dirichlet‑like vector (Gamma(α,1) with α=conc)."""
    if conc <= 0:
        # heavy‑tailed, near‑sparse draws
        samp = rng.exponential(1.0, size=n) ** (1.0 / max(0.1, -conc))
    else:
        samp = rng.gamma(conc, 1.0, size=n)
    samp += 1e-12
    return samp / samp.sum()


def perturb(a: np.ndarray, rng: np.random.Generator, scale: float = 0.02) -> np.ndarray:
    """Add small Gaussian jitter and renormalise."""
    b = a + rng.normal(0.0, scale, size=a.shape)
    b = np.maximum(b, 0.0)
    s = b.sum()
    if s == 0.0:
        return np.full_like(a, 1.0 / len(a))
    return b / s


def upsample(prev: np.ndarray, new_n: int) -> np.ndarray:
    """
    Linear interpolation from len(prev) → new_n, followed by re‑normalisation.
    """
    x_old = np.linspace(0.0, 1.0, len(prev))
    x_new = np.linspace(0.0, 1.0, new_n)
    interp = np.interp(x_new, x_old, prev)
    s = interp.sum()
    if s == 0.0:
        return np.full(new_n, 1.0 / new_n)
    return interp / s


# ----------------------------------------------------------------------
# Surrogate objective (smooth max) and its gradient
# ----------------------------------------------------------------------


def smoothed_max_and_grad(
    a: np.ndarray,
    beta: float,
    n_fft: int,
) -> Tuple[float, np.ndarray]:
    """
    Compute the smoothed max fβ(a) = (1/β)·log Σ exp(β·(a∗a))
    and its gradient ∇fβ(a).

    Returns
    -------
    obj : float
        Smooth approximation of the objective (already normalised because sum(a)=1).
    grad : np.ndarray
        Gradient of the smooth objective w.r.t. ``a`` (same shape as ``a``).
    """
    n = len(a)

    # ----- convolution a*a (full) ------------------------------------
    A = np.fft.rfft(a, n=n_fft)
    conv_full = np.fft.irfft(A * A, n=n_fft)[: 2 * n - 1]

    # ----- smooth max (log‑sum‑exp) ---------------------------------
    # stability: shift by max before exponentiating
    max_val = conv_full.max()
    shifted = beta * (conv_full - max_val)
    exp_shifted = np.exp(shifted)
    sum_exp = exp_shifted.sum()
    # objective:  (max_val + log(sum_exp)/beta)
    smooth_obj = max_val + np.log(sum_exp) / beta

    # ----- weights for the gradient ---------------------------------
    w = exp_shifted / sum_exp                 # length 2n‑1, sums to 1
    w_rev = w[::-1]                           # reverse for correlation

    # ----- grad = 2 * (a ⋆ w) --------------------------------------
    # correlation via convolution with reversed w
    B = np.fft.rfft(w_rev, n=n_fft)
    corr_full = np.fft.irfft(A * B, n=n_fft)[: 3 * n - 2]   # a * w_rev

    # slice indices n‑1 … 2n‑2 (inclusive) and reverse
    grad = 2.0 * corr_full[n - 1 : 2 * n - 1][::-1]

    # The smooth objective corresponds to the (unnormalised) max term.
    # Because the vector stays on the simplex, the true objective value is
    # 2·n·max(a∗a).  Here we return the smooth approximation of max(a∗a).
    return smooth_obj, grad


# ----------------------------------------------------------------------
# Adam optimiser on the smooth objective
# ----------------------------------------------------------------------


def adam_optimize(
    init_a: np.ndarray,
    max_iter: int,
    deadline: float,
    lr: float = 0.2,
    beta1: float = 0.9,
    beta2: float = 0.999,
    eps: float = 1e-8,
    beta0: float = 5.0,
    beta_max: float = 2000.0,
    schedule: str = "linear",
) -> Tuple[np.ndarray, float]:
    """
    Adam‑style optimisation of the smoothed max‑convolution objective.
    Returns the best vector seen (projected onto the simplex) and its
    surrogate objective value (2·n·max(conv) because the sum stays ≈1).

    Parameters
    ----------
    init_a :
        Starting probability vector (already normalised).
    max_iter :
        Upper bound on the number of iterations.
    deadline :
        Wall‑clock time after which the routine aborts.
    lr :
        Base learning‑rate for Adam.
    beta0 , beta_max :
        Schedule for the smoothing parameter β.  ``schedule`` can be ``'linear'``
        (β grows linearly) or ``'exp'`` (exponential growth).
    """
    a = np.maximum(init_a, 1e-30)      # avoid exact zeros for numerical stability
    a = a / a.sum()
    n = len(a)
    # FFT size that covers both the convolution a∗a (size 2n‑1) and
    # the correlation a⋆w (size 3n‑2)
    n_fft = next_pow2(3 * n - 2)

    m = np.zeros_like(a)          # first moment
    v = np.zeros_like(a)          # second moment

    best_a = a.copy()
    # quick surrogate based on the true max (no smoothing)
    conv0 = convolution_fft(a, a, n_fft=n_fft)
    best_score = 2.0 * n * conv0.max()

    for it in range(1, max_iter + 1):
        if time.time() >= deadline:
            break

        # ----- smoothing parameter β ---------------------------------
        progress = it / max_iter
        if schedule == "exp":
            beta = beta0 * (beta_max / beta0) ** progress
        else:  # linear
            beta = beta0 + (beta_max - beta0) * progress

        # ----- smooth objective and gradient -------------------------
        smooth_obj, grad = smoothed_max_and_grad(a, beta, n_fft)

        # ----- Adam update -------------------------------------------
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)

        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        step = lr * m_hat / (np.sqrt(v_hat) + eps)
        a = a - step

        # ----- projection onto the probability simplex -----------------
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s <= 0.0:
            a = np.full_like(a, 1.0 / n)
        else:
            a = a / s

        # ----- keep the best surrogate seen so far -------------------
        conv = convolution_fft(a, a, n_fft=n_fft)
        cur_score = 2.0 * n * conv.max()  # because sum(a)=1
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()

        # ----- occasional progress report -----------------------------
        if it % 5000 == 0:
            print(
                f"      Adam iter={it:,d}  β={beta:6.1f}  surrogate={best_score:.6f}"
            )

    return best_a, best_score
