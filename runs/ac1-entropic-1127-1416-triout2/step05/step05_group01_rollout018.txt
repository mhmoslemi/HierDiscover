    order = np.random.permutation(n)
    mask = np.zeros(n, dtype=np.int8)
    sum_counts = np.zeros(2*n-1, dtype=np.int16)
    for i in order:
        # if i already in mask (should not), continue
        # check conflict
        if sum_counts[2*i] != 0:
            continue
        if mask.any():
            # compute i + indices where mask==1
            idxs = np.where(mask)[0]
            sums = i + idxs
            if np.any(sum_counts[sums] != 0):
                continue
        # add i
        mask[i] = 1
        sum_counts[2*i] = 1
        if np.sum(mask) > 1:

            sum_counts[i + idxs] = 2
    s = mask.sum()

    return mask, score, max_count

"""
Sidon‑set based construction for the step‑function minimisation problem.

The evaluation function is

    score = 2·n·max(conv(a,a)) / (sum(a)^2),

where `conv` is the linear convolution.
If we restrict the sequence to a *support* S ⊂ {0,…,n‑1} and put the same
weight 1/|S| on each selected index, then

    max(conv) = (2 / |S|²)          (every unordered pair contributes 2)
    score     = 4·n / |S|² .

Hence the problem reduces to *making the support as large as possible while
keeping every sum i+j (0 ≤ i,j < n) used at most once*.  Such a set is a
Sidon (B₂) set.  For a given length n the trivial bound

    |S|(|S|+1)/2 ≤ 2·n-1               (all possible sums)

gives a theoretical maximum |S|max = ⌊(-1+√(1+8·(2n-1)))/2⌋.
For n=1024 this bound is 63, for n=2048 it is 90.

The algorithm repeatedly builds a maximal Sidon set by a very cheap greedy
process on a random permutation of the indices; the best set found (largest
|S|) is turned into a sequence of uniform weights and returned.
Because the score depends only on |S| (and the unavoidable factor 2 for
ordered pairs) we can evaluate candidates without any FFT calls,
allowing thousands of trials well within the 1000 s budget.

The search stops early if a set reaching the theoretical optimum (or
within 1 % of it) is obtained.
"""
import time
import numpy as np

# ---------------------------------------------------------------
# Greedy construction of a Sidon set (no duplicated sums)
# ---------------------------------------------------------------

def greedy_sid_set(n: int) -> np.ndarray:
    """
    Return a NumPy array of indices forming a maximal Sidon set for the
    given length `n`.  All unordered sums i+j are distinct, i.e. each
    possible sum is used at most once.
    """
    used = np.zeros(2 * n - 1, dtype=np.bool_)   # used[sum] == True  ⇔  sum already taken
    S = []                                        # list of chosen indices

    # iterate a random permutation of 0 … n‑1
    for i in np.random.permutation(n):
        # self‑sum must be free
        if used[2 * i]:
            continue

        # check every already chosen element
        conflict = False
        for j in S:
            if used[i + j]:
                conflict = True
                break
        if conflict:
            continue

        # i can be added – mark all newly occupied sums
        S.append(i)
        used[2 * i] = True               # i+i
        # i + each earlier element
        for j in S[:-1]:
            used[i + j] = True

    return np.asarray(S, dtype=np.int64)


# ---------------------------------------------------------------
# Helper: turn a support set into the final float sequence
# ---------------------------------------------------------------

def support_to_sequence(support: np.ndarray, n: int) -> list[float]:
    """
    Uniform weights on `support` (sum of the sequence equals 1).
    """
    s = support.size
    if s == 0:
        # degenerate case – never happens in normal use
        return [0.0] * n
    weight = 1.0 / s
    seq = np.zeros(n, dtype=np.float64)
    seq[support] = weight
    return seq.tolist()
