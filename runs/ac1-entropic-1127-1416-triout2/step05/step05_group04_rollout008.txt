"""
Propose a step‑function of non‑negative heights that beats the previous
record (≈1.513).  The method is deliberately simple:

1.  Build the classic arcsine shape (the best analytically known family) for
    several lengths.  As the length grows the discretisation error shrinks and
    the evaluation value approaches the theoretical limit 1.5.

2.  If the arcsine candidates are not good enough, perform a short
    randomised search over the 2‑parameter Beta family
    a_i ∝ (i+ε)^{α‑1}(N‑1‑i+ε)^{β‑1} (α,β>0, ε≈0).  The fast surrogate
    2·N·max(conv) is used to prune candidates; the best few are checked with
    the official ``evaluate_sequence`` function.

3.  Finally a few steps of exact sub‑gradient descent are applied to the
    best vector found.  All operations stay well within the 1000 s budget.

The code contains only top‑level helpers (no closures) and uses only numpy
and the standard library, as required by the competition harness.
"""

import time
import numpy as np
from typing import List, Tuple

# ----------------------------------------------------------------------
# Low‑level utilities (FFT‑based convolution, projection, fast score)
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _conv_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Linear convolution using a real FFT (length = len(a)+len(b)-1)."""
    m = len(a) + len(b) - 1
    n_fft = _next_pow2(m)
    A = np.fft.rfft(a, n=n_fft)
    B = np.fft.rfft(b, n=n_fft)
    C = A * B
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


def _fast_score(a: np.ndarray) -> float:
    """
    Fast surrogate for the official score.
    The vector is assumed to sum to 1, therefore the denominator of
    ``evaluate_sequence`` disappears and the score reduces to
    ``2·N·max(conv(a,a))``.
    """
    n = a.shape[0]
    conv = _conv_fft(a, a)
    return 2.0 * n * float(np.max(conv))


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of ``v`` onto the probability simplex
    {x ≥ 0, Σx = 1}.  Implementation of Duchi‑et‑al. (2008).
    """
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / v.size)

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho = np.where(u - cssv / np.arange(1, v.size + 1) > 0)[0]
    if len(rho) == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


# ----------------------------------------------------------------------
# Shape factories
# ----------------------------------------------------------------------


def arcsine_initial(n: int, eps: float = 0.25) -> np.ndarray:
    """
    Classic “arcsine” shape:
        a_i ∝ 1 / √((i+½)(n‑½‑i) + eps)
    The vector is normalised to Σa_i = 1.
    """
    i = np.arange(n, dtype=float)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - 0.5 - i) + eps)
    vals /= vals.sum()
    return vals


def beta_shape(n: int, alpha: float, beta: float, eps: float = 1e-6) -> np.ndarray:
    """
    (i+ε)^{α‑1}·(n‑1‑i+ε)^{β‑1} on {0,…,n‑1}, normalised to Σ=1.
    ``eps`` prevents division‑by‑zero when α or β < 1.
    """
    i = np.arange(n, dtype=float)
    left = (i + eps) ** (alpha - 1.0)
    right = (n - 1 - i + eps) ** (beta - 1.0)
    base = left * right
    total = base.sum()
    if not np.isfinite(total) or total <= 0.0:
        return np.full(n, 1.0 / n)
    return base / total


# ----------------------------------------------------------------------
# Exact sub‑gradient descent on the true objective
# ----------------------------------------------------------------------


def subgradient_descent_exact(
    init_a: np.ndarray,
    deadline: float,
    max_iter: int = 80_000,
    step0: float = 0.05,
) -> np.ndarray:
    """
    Sub‑gradient descent for   f(a)=2·N·max(conv(a,a)).
    The step size η_t = step0 / √t  and the iterate is projected onto the
    probability simplex after every update.
    """
    a = _project_simplex(init_a.copy())
    n = a.shape[0]
    L = 2 * n - 1
    best_a = a.copy()
    best_sc = _fast_score(a)

    for it in range(1, max_iter + 1):
        if time.time() >= deadline:
            break

        conv = _conv_fft(a, a)
        max_val = np.max(conv)

        # mask of indices attaining (or almost attaining) the maximum
        tol = max_val * 1e-12 + 1e-14
        mask = (np.abs(conv - max_val) <= tol).astype(float)

        # weighted sub‑gradient : 2·conv(a, mask_rev) evaluated at the
        # positions that correspond to the original vector.
        mask_rev = mask[::-1]
        grad_rev = _conv_fft(a, mask_rev)
        weighted_grad = 2.0 * grad_rev[L - 1 - np.arange(n)]

        eta = step0 / np.sqrt(it)
        a = a - eta * weighted_grad
        a = _project_simplex(a)

        cur_sc = _fast_score(a)
        if cur_sc < best_sc:
            best_sc = cur_sc
            best_a = a.copy()

        # occasional progress report (lightweight)
        if it % 10_000 == 0:
            print(f"    SubGrad it={it:,d} cur={cur_sc:.6f} best={best_sc:.6f}")

    return best_a

def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs) -> List[float]:
    """
    Search for a non‑negative step sequence with a low
    ``evaluate_sequence`` value.

    Strategy:
      * test the analytically strong arcsine shape for several lengths;
      * if needed, perform a brief random search inside the Beta family;
      * polish the current best with a few sub‑gradient steps;
      * return the best vector found before the time budget expires.
    """
    rng = np.random.default_rng(seed)
    start = time.time()
    deadline = start + budget_s - 1.0  # leave a small safety margin

    best_seq: List[float] = []
    best_val: float = float("inf")

    # ------------------------------------------------------------------
    # 0.  Optional fallback from a previous run
    # ------------------------------------------------------------------
    if "height_sequence_1" in globals():
        try:
            fallback = [float(x) for x in globals()["height_sequence_1"]]
            val = evaluate_sequence(fallback)
            print(f"Fallback (height_sequence_1) score = {val:.6f}")
            if val < best_val:
                best_val = val
                best_seq = fallback
        except Exception:
            pass

    # ------------------------------------------------------------------
    # 1.  Test the arcsine shape for a few lengths
    # ------------------------------------------------------------------
    for n in (4096, 8192, 12288):
        if time.time() >= deadline:
            break
        a = arcsine_initial(n, eps=0.25)
        fast_sc = _fast_score(a)
        print(f"Arcsine n={n:5d} fast score = {fast_sc:.6f}")

        try:
            val = evaluate_sequence(a.tolist())
        except Exception:
            continue

        print(f"  Official score = {val:.6f}")
        if val < best_val:
            best_val = val
            best_seq = a.tolist()
            print(f"  *** New global best (arcsine) = {best_val:.6f} ***")

        if best_val <= 1.503:            # target already met
            return best_seq

    # ------------------------------------------------------------------
    # 2.  Light random search in the Beta family (only if time permits)
    # ------------------------------------------------------------------
    if time.time() < deadline - 20.0:
        n = 8192
        # keep only the best few candidates (score, vector)
        top_k = 5
        pool: List[Tuple[float, np.ndarray]] = []

        def _maybe_insert(score: float, vec: np.ndarray):
            if len(pool) < top_k:
                pool.append((score, vec))
            else:
                # replace the worst entry if the new one is better
                worst_idx = max(range(len(pool)), key=lambda i: pool[i][0])
                if score < pool[worst_idx][0]:
                    pool[worst_idx] = (score, vec)

        attempts = 0
        while time.time() < deadline - 15.0 and attempts < 10_000:
            attempts += 1
            alpha = rng.uniform(0.05, 5.0)
            beta = rng.uniform(0.05, 5.0)
            eps = rng.uniform(1e-6, 0.5)
            cand = beta_shape(n, alpha, beta, eps)
            sc = _fast_score(cand)
            _maybe_insert(sc, cand)

        # rank the kept candidates and evaluate them officially
        pool.sort(key=lambda x: x[0])
        for score, vec in pool:
            if time.time() >= deadline:
                break
            try:
                val = evaluate_sequence(vec.tolist())
            except Exception:
                continue
            if val < best_val:
                best_val = val
                best_seq = vec.tolist()
                print(f"  *** New global best (beta) = {best_val:.6f} ***")
            if best_val <= 1.503:
                return best_seq

    # ------------------------------------------------------------------
    # 3.  Final polishing of the best candidate (if one exists)
    # ------------------------------------------------------------------
    if best_seq and time.time() < deadline - 5.0:
        a_np = np.array(best_seq, dtype=float)
        # a modest number of sub‑gradient steps – cheap enough even for large n
        a_np = subgradient_descent_exact(a_np, deadline, max_iter=60_000, step0=0.02)
        try:
            val = evaluate_sequence(a_np.tolist())
            print(f"Polished score = {val:.6f}")
            if val < best_val:
                best_val = val
                best_seq = a_np.tolist()
                print(f"  *** Updated global best after polishing = {best_val:.6f} ***")
        except Exception:
            pass

    # ------------------------------------------------------------------
    # 4.  Guard – ensure a non‑empty return value
    # ------------------------------------------------------------------
    if not best_seq:
        # as a last resort return a uniform vector of modest length
        n = 2048
        a = np.full(n, 1.0 / n)
        best_seq = a.tolist()
        best_val = evaluate_sequence(best_seq)
        print(f"Fallback to uniform: score = {best_val:.6f}")

    print(f"\nSearch finished. Best reported score = {best_val:.6f}")
    return best_seq
