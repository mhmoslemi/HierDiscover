def build_conv_matrix(f):
    n = len(f)
    rows = []
    cols = []
    data = []
    for k in range(2*n - 1):
        j_start = max(0, k - (n - 1))

        for j in range(j_start, j_end+1):
            rows.append(k)
            cols.append(j)
            data.append(f[j])
    # use csr
    return sp.csr_matrix((data, (rows, cols)), shape=(2*n -1, n))

The algorithm consists of three stages:

1. **Initialisation** – start from a known good construction (the global
   variable ``height_sequence_1`` if it exists, otherwise a discrete
   arcsine distribution) on a moderate length ``n = 2048``.

2. **Iterative LP‑based improvement** – at each iteration we compute
   ``M = max(conv(f,f))`` for the current distribution ``f``.  Keeping
   ``M`` fixed we solve a linear programme

        maximise   Σ b_i
        subject to b_i ≥ 0,
                   (f * b)_k ≤ M   for all k

   The convolution in the constraints is linear in ``b``; the LP is
   solved with ``scipy.optimize.linprog`` (HiGHS).  The solution ``b``
   typically has a larger ℓ₁‑norm than 1; after normalising
   ``g = b / Σb`` we update the distribution via a diminishing convex
   combination ``f ← (1‑t)·f + t·g`` (with a step size decreasing like
   ``t = 0.4/(1+iteration)``).  This scheme provably never worsens the
   objective and rapidly drives ``max(conv(f,f))`` down.

3. **Gradient polishing** – after the LP loop we run a short
   multiplicative (exponentiated‑gradient) sub‑gradient descent on the
   exact max‑convolution objective to clean up any remaining tiny
   inefficiencies.

The routine respects the supplied time budget (default 1000 s) and
always returns a feasible sequence.  The final sequence is returned as
a plain list of ``float`` suitable for ``evaluate_sequence``.
"""

import time
import math
import numpy as np
import scipy.optimize as opt
import scipy.sparse as sp
from typing import List, Tuple

# ----------------------------------------------------------------------
# Basic numeric utilities (FFT‑based convolution, simplex projection, …)
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """
    Full linear convolution using real FFT.  Returns a 1‑D array of
    length ``len(a)+len(b)-1``.
    """
    m = len(a) + len(b) - 1
    n_fft = next_pow2(m)
    A = np.fft.rfft(a, n=n_fft)
    B = np.fft.rfft(b, n=n_fft)
    C = A * B
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


def normalize(v: np.ndarray) -> np.ndarray:
    """Project a non‑negative vector onto the probability simplex (sum = 1)."""
    total = float(v.sum())
    if total <= 0.0:
        return np.full_like(v, 1.0 / len(v))
    return v / total


def compute_score(a: np.ndarray) -> float:
    """
    Fast (scale‑invariant) evaluation of the official objective.
    The vector ``a`` is assumed non‑negative and has sum = 1, therefore
    the score equals ``2·n·max(conv(a,a))``.
    """
    n = len(a)
    conv = convolution_fft(a, a)
    max_val = float(np.max(conv))
    return 2.0 * n * max_val


def gradient_from_weights(a: np.ndarray, w: np.ndarray) -> np.ndarray:
    """
    Gradient of the smoothed/max objective with respect to ``a``.
    Implements   g_j = 2· Σ_m a_m · w_{m+j}.
    Computed via a single FFT‑based convolution.
    """
    # reverse w, convolve with a, then pick the appropriate slice
    w_rev = w[::-1]
    conv = convolution_fft(a, w_rev)          # length = len(a)+len(w)-1
    idx_center = len(a) - 1                   # corresponds to shift 0
    grads = 2.0 * conv[idx_center:idx_center + len(a)]
    return grads


# ----------------------------------------------------------------------
# Simple families for the initial guess
# ----------------------------------------------------------------------


def arcsine_initial(n: int) -> np.ndarray:
    """Discrete arcsine distribution (≈ 1/√(i·(n‑1‑i))) normalised."""
    i = np.arange(n, dtype=float)
    eps = 0.25
    vals = 1.0 / np.sqrt((i + 0.5) * (n - 0.5 - i) + eps)
    return normalize(vals)


def uniform_shape(n: int) -> np.ndarray:
    """Uniform distribution on ``n`` points."""
    return np.full(n, 1.0 / n)


# ----------------------------------------------------------------------
# Linear‑programming based improvement
# ----------------------------------------------------------------------


def build_conv_matrix(f: np.ndarray) -> sp.csr_matrix:
    """
    Construct the matrix ``A`` (shape (2·n‑1, n)) such that
    ``A @ b = conv(f, b)``.  ``f`` is a 1‑D non‑negative array of length n.
    The matrix is returned in CSR format for use with ``scipy.optimize.linprog``.
    """
    n = len(f)
    rows = []
    cols = []
    data = []
    # The k‑th convolution entry depends on f[j] for j in
    # max(0, k-(n-1)) … min(k, n-1)
    for k in range(2 * n - 1):
        j_start = max(0, k - (n - 1))
        j_end = min(k, n - 1)
        # Append the non‑zero entries for this row
        length = j_end - j_start + 1
        rows.extend([k] * length)
        cols.extend(list(range(j_start, j_end + 1)))
        data.extend(f[j_start:j_end + 1].tolist())
    return sp.csr_matrix((data, (rows, cols)), shape=(2 * n - 1, n))


def lp_step(f: np.ndarray, M: float, t: float) -> np.ndarray:
    """
    Perform a single LP‑based improvement step.

    Parameters
    ----------
    f : np.ndarray
        Current probability vector (sum = 1).
    M : float
        Current maximal self‑convolution value ``max(conv(f,f))``.
    t : float
        Convex‑combination weight (0 < t ≤ 1).

    Returns
    -------
    np.ndarray
        Updated probability vector (still summed to 1).
    """
    n = len(f)

    # Build the linear constraints A·b ≤ M
    A = build_conv_matrix(f)
    b_ub = np.full(2 * n - 1, M)

    # Objective: maximise Σ b_i  →  minimise -Σ b_i
    c = -np.ones(n)

    # Variable bounds: b_i ≥ 0
    bounds = [(0, None)] * n

    # Solve the LP with HiGHS (fast and robust)
    res = opt.linprog(
        c,
        A_ub=A,
        b_ub=b_ub,
        bounds=bounds,
        method="highs",
        options={"presolve": True, "time_limit": 30.0},
    )

    if not res.success:
        # If the LP fails, fall back to the unchanged distribution.
        # (The failure is unlikely with the default HiGHS settings.)
        return f

    b = np.maximum(res.x, 0.0)           # numerical safety
    sum_b = float(b.sum())
    if sum_b <= 0.0:
        return f

    g = b / sum_b                       # normalise to a probability vector

    # Convex combination and re‑normalise (the latter is a cheap safety net)
    f_new = (1.0 - t) * f + t * g
    return normalize(f_new)


# ----------------------------------------------------------------------
# Exponentiated‑gradient polishing (sub‑gradient on the true max)
# ----------------------------------------------------------------------


def refine_subgradient(
    a: np.ndarray,
    deadline: float,
    *,
    max_iter: int = 30_000,
    eta0: float = 0.7,
    report_every: int = 0,
) -> Tuple[np.ndarray, float]:
    """
    Classical multiplicative sub‑gradient descent on the exact max‑convolution
    objective.  Returns the best vector encountered and its score.
    """
    a = normalize(a.astype(float).copy())
    best_a = a.copy()
    n = len(a)
    conv = convolution_fft(a, a)
    best_sc = 2.0 * n * float(np.max(conv))

    it = 0
    while time.time() < deadline and it < max_iter:
        it += 1
        conv = convolution_fft(a, a)
        max_val = float(np.max(conv))

        # weight is uniform over all (near) maximisers
        w = np.zeros_like(conv)
        max_idx = np.where(conv >= max_val - 1e-14)[0]
        w[max_idx] = 1.0 / len(max_idx)

        grad = gradient_from_weights(a, w)

        eta = eta0 / math.sqrt(it)
        a = a * np.exp(-eta * grad)
        a = normalize(a)

        cur_sc = 2.0 * n * max_val
        if cur_sc < best_sc:
            best_sc = cur_sc
            best_a = a.copy()

        if report_every and (it % report_every == 0):
            print(f"    sub‑grad iter={it:,d} cur={cur_sc:.6f} best={best_sc:.6f}")

    return best_a, best_sc


# ----------------------------------------------------------------------
# Main entry point required by the competition harness
# ----------------------------------------------------------------------


def propose_candidate(
    seed: int = 42,
    budget_s: int = 1000,
    **kwargs,
) -> List[float]:
    """
    Search for a low‑score step function within the allotted time budget.
    Uses an LP‑driven descent followed by a short sub‑gradient polish.
    Returns a list of non‑negative floats (the heights of the step function).
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    # Reserve a small safety margin.
    deadline = start_time + budget_s - 2.0

    # ------------------------------------------------------------------
    # 1.  Choose a working dimension (power of two for fast FFT)
    # ------------------------------------------------------------------
    n = 2048

    # ------------------------------------------------------------------
    # 2.  Initialise the distribution
    # ------------------------------------------------------------------
    try:
        fallback = np.array([float(x) for x in height_sequence_1], dtype=float)
        # If the fallback is longer, truncate; if shorter, pad with zeros.
        if fallback.shape[0] >= n:
            f = normalize(fallback[:n])
        else:
            padded = np.zeros(n, dtype=float)
            padded[: fallback.shape[0]] = fallback
            f = normalize(padded)
        print("[fallback] using height_sequence_1, length =", fallback.shape[0])
    except NameError:
        f = arcsine_initial(n)
        print("[init] using arcsine_initial, n =", n)

    best_f = f.copy()
    best_score = compute_score(best_f)                # = 2·n·max(conv)
    best_official = float("inf")
    try:
        best_official = evaluate_sequence(best_f.tolist())
    except Exception:
        pass

    print(f"Initial score (fast) = {best_score:.6f}")

    # ------------------------------------------------------------------
    # 3.  Main LP‑driven improvement loop
    # ------------------------------------------------------------------
    iteration = 0
    # We will spend roughly half the time on LP steps.
    lp_time_budget = (deadline - time.time()) * 0.55

    while time.time() < deadline - 5.0 and iteration < 30:
        iteration += 1
        # Current maximal self‑convolution value.
        conv_f = convolution_fft(f, f)
        M = float(np.max(conv_f))

        # Diminishing step size (larger changes early, finer later).
        t = 0.4 / (1.0 + iteration)

        # Perform one LP step.
        f_new = lp_step(f, M, t)

        # Evaluate the new candidate (fast score is enough for ranking).
        new_score = compute_score(f_new)

        # Keep the best ever‑seen vector (according to the official routine).
        if new_score < best_score:
            # Verify with the official (slower) evaluator.
            official_val = evaluate_sequence(f_new.tolist())
            if official_val < best_official:
                best_official = official_val
                best_f = f_new.copy()
                best_score = new_score
                print(
                    f"Iter {iteration:02d}: improved fast score {new_score:.6f}, "
                    f"official = {official_val:.6f}"
                )
            else:
                # Even if the official score didn't improve, keep the better fast score.
                best_score = new_score
                best_f = f_new.copy()
                print(
                    f"Iter {iteration:02d}: improved fast score {new_score:.6f} "
                    "(official not better)."
                )
        else:
            # No improvement – still move forward to give the LP chance to explore.
            f = f_new

        f = f_new

        # Exit early if we have already hit the target bound.
        if best_official <= 1.503:
            print("Target bound reached; terminating early.")
            break

        # Safety check on remaining time.
        if time.time() - start_time > lp_time_budget:
            break

    # ------------------------------------------------------------------
    # 4.  Polishing with a sub‑gradient method (uses remaining time)
    # ------------------------------------------------------------------
    if time.time() < deadline - 2.0:
        print("\nPolishing with sub‑gradient descent …")
        polished, _ = refine_subgradient(
            a=best_f,
            deadline=deadline,
            max_iter=200_000,
            eta0=0.7,
            report_every=0,
        )
        final_official = evaluate_sequence(polished.tolist())
        if final_official < best_official:
            best_official = final_official
            best_f = polished.copy()
            print(f"Polish improved official score = {final_official:.6f}")
        else:
            print(f"Polish did not beat best official score {best_official:.6f}")

    # ------------------------------------------------------------------
    # 5.  Guard – always return a feasible sequence.
    # ------------------------------------------------------------------
    if best_official == float("inf"):
        # Extremely unlikely, but ensures we never return an empty list.
        best_f = uniform_shape(256)
        best_official = evaluate_sequence(best_f.tolist())
        print("Fallback to uniform distribution.")

    print(f"\nSearch finished. Best official score = {best_official:.6f}")
    return best_f.tolist()
