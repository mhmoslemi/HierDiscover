    n = len(a)
    try:

        from scipy.sparse import diags, csr_matrix

    data = [a[d] * np.ones(n) for d in range(n)]

        return None
    b_val = np.array(res.x)
    # ensure non-negative
    b_val = np.maximum(b_val, 0)
    return b_val

def solve_lp_for_b(a, M):
    """
    Solve the linear program:
        maximise   sum(b)
        subject to b >= 0,
                   (a * b) <= M   (full convolution)
    Returns b array or None on failure.
    """
    try:
        from scipy.optimize import linprog
        from scipy.sparse import diags, csr_matrix
    except Exception:
        return None

    n = len(a)

    # Build sparse convolution matrix A with shape (2*n-1, n)
    # each diagonal offset d has constant a[d]
    diagonals = [a[d] * np.ones(n) for d in range(n)]
    offsets = np.arange(n)
    A = diags(diagonals, offsets, shape=(2 * n - 1, n), format='csr')

    # Upper bound constraints: A @ b <= M
    b_ub = np.full(2 * n - 1, M)

    # Objective: maximise sum(b) -> minimise -sum(b)
    c = -np.ones(n)

    bounds = [(0, None)] * n

    # Solve with HiGHS (default)
    try:
        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs', options={'presolve': True})
    except Exception:
        return None

    if not res.success:
        return None

    b = np.asarray(res.x, dtype=float)
    # Ensure non-negative
    b = np.maximum(b, 0.0)

"""
A fast heuristic for minimising the evaluation function
`evaluate_sequence`.  The algorithm combines several classic ideas:

* **Random initialisations** – uniform, Dirichlet, sparse and sine‑shaped
  vectors on the probability simplex.

* **Simulated annealing** – stochastic mass transfers with a temperature
  schedule and Metropolis acceptance.  This explores the non‑convex landscape
  and can escape local minima.

* **Linear‑programming step (optional)** – if SciPy is available we solve
  the LP  maximise Σb  s.t.  b≥0 and (a⊛b)≤M, where M is the current peak.
  The solution is mixed with the current vector and projected back onto the
  simplex.  This mimics the alternating‑LP scheme from the reference paper.

* **Projected sub‑gradient descent** – a deterministic refinement that follows
  the gradient of the currently maximal convolution entry.

* **Hill‑climbing polishing** – greedy swaps / mass transfers accepted only
  when they improve the objective.

All heavy work is done with an FFT‑based convolution (`convolution_fft`) and
the fast surrogate `compute_score(a) = 2·n·max(a⊛a)` (valid when Σa=1).  The
search is time‑bounded (1000 s) and always returns a plain Python list of
non‑negative floats ready for `evaluate_sequence`.
"""

import time
import numpy as np

# Basic linear‑algebra utilities

def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full convolution a⊛b using a real FFT (fast)."""
    m = len(a) + len(b) - 1
    n_fft = next_pow2(m)
    A = np.fft.rfft(a, n=n_fft)
    B = np.fft.rfft(b, n=n_fft)
    C = A * B
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


def compute_score(a: np.ndarray) -> float:
    """
    Fast replica of `evaluate_sequence` when Σa = 1.
    Returns 2·n·max(a⊛a).
    """
    n = a.shape[0]
    conv = convolution_fft(a, a)
    return 2.0 * n * float(np.max(conv))


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the probability simplex
    (non‑negative entries that sum to 1).
    """
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / v.shape[0])

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, v.shape[0] + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.where(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def random_dirichlet(n: int) -> np.ndarray:
    """Draw a random point on the simplex using an exponential/Dirichlet trick."""
    a = np.random.rand(n)
    a /= a.sum()
    return a


def random_sparse(n: int, density: float) -> np.ndarray:
    """Uniform mass on a randomly chosen subset of size ≈ density·n."""
    k = max(2, int(round(density * n)))
    idx = np.random.choice(n, size=k, replace=False)
    a = np.zeros(n, dtype=float)
    a[idx] = 1.0 / k
    return a


def sine_shaped(n: int) -> np.ndarray:
    """Sinusoidal (positive) shape, normalised to sum 1."""
    x = np.arange(1, n + 1)
    a = np.sin(np.pi * x / (n + 1))
    a = np.maximum(a, 0.0)
    a /= a.sum()
    return a

def simulated_annealing(a_init: np.ndarray, deadline: float) -> tuple[np.ndarray, float]:
    """
    Stochastic mass‑transfer annealing on the objective 2·n·max(a⊛a).
    Returns the best vector found and its score.
    """
    a = a_init.copy()
    n = a.shape[0]
    cur_score = compute_score(a)
    best_a = a.copy()
    best_score = cur_score

    start = time.time()
    T0 = 5e-3          # initial temperature
    T_final = 1e-7     # final temperature (practically zero)

    while time.time() < deadline:
        elapsed = time.time() - start
        total = deadline - start
        if total <= 0:
            break
        frac = elapsed / total
        T = T0 * ((T_final / T0) ** frac)  # exponential cooling

        # choose source proportionally to current mass (large entries move more)
        i = np.random.choice(n, p=a)
        j = np.random.randint(n)
        if i == j:
            continue

        # transfer a fraction of mass from i to j
        delta = min(a[i], T * np.random.rand())
        if delta <= 0:
            continue

        a_new = a.copy()
        a_new[i] -= delta
        a_new[j] += delta
        # tiny numerical drift → renormalise occasionally
        a_new = np.maximum(a_new, 0.0)
        a_new /= a_new.sum()

        new_score = compute_score(a_new)

        if new_score < cur_score or np.random.rand() < np.exp(-(new_score - cur_score) / max(T, 1e-12)):
            a = a_new
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_a = a_new.copy()
    return best_a, best_score

def solve_lp_for_b(a: np.ndarray, M: float):
    """
    Solve   maximise   Σ b_j   subject to   b ≥ 0,
                (a ⊛ b)_ℓ ≤ M   for all ℓ.

    Returns the optimal b as a NumPy array, or None on failure.
    """
    if not SCIPY_AVAILABLE:
        return None
    n = a.shape[0]

    # Build the convolution matrix A (shape (2n‑1, n)) as a sparse Toeplitz matrix.
    # Each diagonal offset d (0 … n‑1) is constant a[d].
    diagonals = [a[d] * np.ones(n) for d in range(n)]
    offsets = np.arange(n)               # diagonal offsets
    A = diags(diagonals, offsets, shape=(2 * n - 1, n), format='csr')

    # Upper‑bound constraints: A @ b ≤ M
    b_ub = np.full(2 * n - 1, M)

    # Objective: maximise Σ b → minimise -Σ b
    c = -np.ones(n)

    # Bounds: b ≥ 0
    bounds = [(0, None)] * n

    try:
        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds,
                      method='highs', options={'presolve': True})
    except Exception:
        return None

    if not res.success:
        return None

    b = np.asarray(res.x, dtype=float)
    b = np.maximum(b, 0.0)
    return b


def lp_alternating_opt(a_init: np.ndarray, deadline: float):
    """
    Perform a few LP‑alternating iterations (the “alternating‑LP” scheme).
    Returns the improved vector and its score.
    """
    a = project_to_simplex(a_init.copy())
    best_a = a.copy()
    best_score = compute_score(a)
    n = a.shape[0]
    iteration = 0

    while time.time() < deadline and iteration < 15:
        conv = convolution_fft(a, a)
        M = float(np.max(conv))

        b = solve_lp_for_b(a, M)
        if b is None:
            break

        # normalise b to the simplex (its sum may be >1)
        b_sum = b.sum()
        if b_sum <= 0:
            break
        b = b / b_sum

        # small mixing step
        t = 0.20
        a = (1 - t) * a + t * b
        a = project_to_simplex(a)

        cur_score = compute_score(a)
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()

        iteration += 1
    return best_a, best_score

def subgrad_optimize(init_a: np.ndarray,
                     time_deadline: float,
                     max_iter: int = 200_000,
                     step0: float = 0.07) -> tuple[np.ndarray, float]:
    """
    Simple projected sub‑gradient descent on f(a)=2·n·max(a⊛a).
    The sub‑gradient corresponds to the shift attaining the maximal value.
    """
    a = init_a.astype(float).copy()
    best_a = a.copy()
    best_score = compute_score(a)
    n = len(a)

    for t in range(1, max_iter + 1):
        if time.time() >= time_deadline:
            break

        conv = convolution_fft(a, a)
        k = int(np.argmax(conv))

        idx = np.arange(n)
        j = k - idx
        mask = (j >= 0) & (j < n)
        g = np.zeros_like(a)
        g[mask] = 2.0 * a[j[mask]]

        step = step0 / np.sqrt(t)
        a_new = a - step * g
        a_new = project_to_simplex(a_new)

        cur_score = compute_score(a_new)
        if cur_score < best_score:
            best_score = cur_score
            best_a = a_new.copy()

        a = a_new

        # occasional jitter to escape plateaus
        if t % 1000 == 0:
            a += np.random.randn(n) * (step0 * 0.005 / np.sqrt(t))
            a = project_to_simplex(a)

    return best_a, best_score

def hill_climb_permutation(seq, time_deadline, max_steps=300_000):
    """
    Greedy hill‑climb that either swaps two entries or transfers a fraction
    of mass from one position to another.  Only moves that strictly improve
    the objective are kept.
    """
    a = np.array(seq, dtype=float)
    n = a.shape[0]

    best_a = a.copy()
    best_score = compute_score(best_a)

    step_scale = 0.1  # starting fraction of mass moved

    while max_steps > 0 and time.time() < time_deadline:
        # decide move type
        if np.random.rand() < 0.5:
            # simple swap
            i, j = np.random.randint(n, size=2)
            cand = best_a.copy()
            cand[i], cand[j] = cand[j], cand[i]
        else:
            # move a fraction from i to j
            i, j = np.random.randint(n, size=2)
            if i == j:
                continue
            delta = step_scale * best_a[i]
            if delta <= 0:
                continue
            cand = best_a.copy()
            cand[i] -= delta
            cand[j] += delta
            cand[i] = max(cand[i], 0.0)

        # re‑project onto simplex to keep the sum exactly 1
        cand = project_to_simplex(cand)
        cur_score = compute_score(cand)

        if cur_score < best_score:
            best_score = cur_score
            best_a = cand
            step_scale = min(step_scale * 1.05, 0.5)   # become more aggressive
        else:
            step_scale *= 0.97                         # cool down

        max_steps -= 1
    return best_a.tolist(), best_score

def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative coefficient vector that minimises the given
    evaluation function.  The routine respects the 1000 s time budget and
    returns a plain Python list of floats.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 2.0   # small safety margin

    best_seq = None
    best_val = float("inf")

    # ------------------------------------------------------------------
    # Optional fallback to a previously known construction
    # ------------------------------------------------------------------
    try:
        fallback = [float(x) for x in height_sequence_1]  # type: ignore
        fallback_val = evaluate_sequence(fallback)
        print(f"Fallback score = {fallback_val:.6f}")
        if fallback_val < best_val:
            best_val = fallback_val
            best_seq = fallback
    except Exception:
        pass

    # ------------------------------------------------------------------
    # Lengths to try – larger lengths give more freedom but cost more time
    # ------------------------------------------------------------------
    length_choices = [1024, 2048, 4096]

    for n in length_choices:
        if time.time() >= deadline - 30:
            break

        print(f"\n=== Optimising length {n} ===")
        # ------------------------------------------------------------------
        # Generate a battery of diverse initial vectors
        # ------------------------------------------------------------------
        init_vectors = []

        # uniform
        init_vectors.append(np.full(n, 1.0 / n))

        # a few random Dirichlet draws
        for _ in range(3):
            init_vectors.append(random_dirichlet(n))

        # random sparse supports with various densities
        for _ in range(4):
            density = np.random.uniform(0.15, 0.45)
            init_vectors.append(random_sparse(n, density))

        # sine‑shaped vectors
        for _ in range(2):
            init_vectors.append(sine_shaped(n))

        # evaluate and keep the best of the initial pool
        for vec in init_vectors:
            cur = compute_score(vec)
            if cur < best_val:
                best_val = cur
                best_seq = vec.tolist()
                print(f"Initial improvement: score={best_val:.6f} (n={n})")

        # ------------------------------------------------------------------
        # Choose the best vector of the correct length as the starting point
        # ------------------------------------------------------------------
        if best_seq is not None and len(best_seq) == n:
            a_curr = np.array(best_seq, dtype=float)
        else:
            a_curr = np.full(n, 1.0 / n)

        a_curr = project_to_simplex(a_curr)

        # ------------------------------------------------------------------
        # Phase 1 – Simulated annealing (exploratory)
        # ------------------------------------------------------------------
        sa_deadline = min(deadline - 5.0,
                          time.time() + max(30.0, (deadline - time.time()) * 0.35))
        a_curr, cur_score = simulated_annealing(a_curr, sa_deadline)
        print(f"SA finished: score={cur_score:.6f}")
        if cur_score < best_val:
            best_val = cur_score
            best_seq = a_curr.tolist()

        # ------------------------------------------------------------------
        # Phase 2 – Optional LP alternating improvement (if SciPy available)
        # ------------------------------------------------------------------
        if SCIPY_AVAILABLE and time.time() < deadline - 10.0:
            try:
                a_curr, cur_score = lp_alternating_opt(a_curr, deadline - 10.0)
                print(f"LP‑alternating finished: score={cur_score:.6f}")
                if cur_score < best_val:
                    best_val = cur_score
                    best_seq = a_curr.tolist()
            except Exception as e:                                 # pragma: no cover
                print(f"LP stage failed: {e}")

        # ------------------------------------------------------------------
        # Phase 3 – Sub‑gradient refinement (deterministic)
        # ------------------------------------------------------------------
        if time.time() < deadline - 15.0:
            a_curr, cur_score = subgrad_optimize(
                a_curr,
                time_deadline=deadline - 10.0,
                max_iter=200_000,
                step0=0.07,
            )
            print(f"Sub‑gradient refined: score={cur_score:.6f}")
            if cur_score < best_val:
                best_val = cur_score
                best_seq = a_curr.tolist()

        # ------------------------------------------------------------------
        # Phase 4 – Hill‑climb polishing
        # ------------------------------------------------------------------
        if time.time() < deadline - 5.0:
            a_curr, cur_score = hill_climb_permutation(
                a_curr.tolist(),
                time_deadline=deadline,
                max_steps=300_000,
            )
            print(f"Hill‑climb polished: score={cur_score:.6f}")
            if cur_score < best_val:
                best_val = cur_score
                best_seq = a_curr

        print(f"Best score for length {n}: {best_val:.6f}")

    # ------------------------------------------------------------------
    # Safeguard – fall back to a uniform vector if nothing succeeded
    # ------------------------------------------------------------------
    if best_seq is None:
        n = 256
        best_seq = [1.0 / n] * n
        best_val = evaluate_sequence(best_seq)

    # Final official evaluation (uses the provided `evaluate_sequence`)
    final_score = evaluate_sequence(best_seq)
    print(f"\nFinished search. Final reported score = {final_score:.6f}")

    # The harness expects a plain Python list of floats.
    return best_seq
