def propose_candidate(seed=42, budget_s=1000, **kwargs):
    import time, numpy as np
    rng = np.random.default_rng(seed)
    start = time.time()
    deadline = start + budget_s - 2

    best_seq = None
    best_score = float('inf')

    # base lengths
    lengths = [256, 1024, 4096, 16384]  # may shrink if time limited

    current_a = None
    for idx, n in enumerate(lengths):
        # if no time left, break
        if time.time() >= deadline - 5:
            break

        # if we have a current vector, upsample; else init candidates
        if current_a is None:
            # generate candidates for this n
            candidates = []
            candidates.append(arcsine_initial(n))
            candidates.append(uniform_shape(n))
            for _ in range(5):
                candidates.append(random_dirichlet(n, rng))
            # fallback
            try:
                fallback = np.array([float(x) for x in height_sequence_1], dtype=float)
                if fallback.shape[0] != n:
                    # resample to n
                    fallback = upsample(fallback, n)
                fallback = normalize(fallback)
                candidates.append(fallback)
            except NameError:
                pass
            # pick best by compute_score
            scores = [(compute_score(c), c) for c in candidates]
            scores.sort(key=lambda x: x[0])
            current_a = scores[0][1]
            best_stage_score = scores[0][0]
        else:
            # upsample
            current_a = upsample(current_a, n)
            current_a = normalize(current_a)
            best_stage_score = compute_score(current_a)

        # Determine stage deadline: allocate proportion of remaining time
        remaining = deadline - time.time()
        # allocate for stage; maybe 0.5 for early, 0.75 for later etc.
        # For simplicity, allocate remaining * (1/(len(lengths) - idx))
        # but ensure at least some min time.
        stage_time = remaining * 0.8 / (len(lengths) - idx)  # crude
        stage_deadline = time.time() + stage_time

        print(f"\nStage {idx+1}/{len(lengths)}: n={n}, initial score={best_stage_score:.6f}, allocated {stage_time:.1f}s")

        # Run smooth refinement
        smooth_end = time.time() + stage_time * 0.4
        a_smooth, sc_smooth = refine_smooth(
            a=current_a,
            deadline=smooth_end,
            eta0=1.0,
            beta=15.0,
            report_every=0,
        )
        print(f"  After smooth: score {sc_smooth:.6f}")

        # Run hard subgradient polish for remaining part of stage
        sub_end = stage_deadline
        a_sub, sc_sub = refine_subgradient(
            a=a_smooth,
            deadline=sub_end,
            max_iter=100_000,
            eta0=0.7,
            report_every=0,
        )
        print(f"  After subgradient: score {sc_sub:.6f}")

        # Run random local search (optional) until stage_deadline
        a_local, sc_local = local_search(
            a=a_sub,
            deadline=stage_deadline,
            rng=rng,
        )

        # update current

        # evaluate official if better than global best
        val = evaluate_sequence(current_a.tolist())
        if val < best_score:

            best_seq = current_a.tolist()
            print(f"*** New global best: {best_score:.6f} ***")

        best_seq = uniform_shape(256).tolist()
        best_score = evaluate_sequence(best_seq)

def upsample(a: np.ndarray, n: int) -> np.ndarray:
    m = len(a)
    if n == m:
        return a.copy()
    if n > m:
        x_old = np.linspace(0, 1, m)
        x_new = np.linspace(0, 1, n)
        return np.interp(x_new, x_old, a)
    else: # n < m
        # aggregate bins
        factor = m // n
        # simple average each group
        a_agg = np.mean(a[:factor*n].reshape(n, factor), axis=1)
        return a_agg
```

def local_search(a: np.ndarray, deadline: float, rng: np.random.Generator,
                 max_moves: int = 1_000_000, step_initial: float = 0.2) -> Tuple[np.ndarray, float]:
    best_a = a.copy()
    best_sc = compute_score(best_a)
    n = len(a)
    it = 0
    while time.time() < deadline and it < max_moves:
        it += 1
        # random indices
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        # choose delta as a fraction of source amplitude
        delta = step_initial * best_a[i]
        # optional random scale
        delta *= rng.random()
        if delta <= 0:
            continue
        if best_a[i] < delta:
            delta = best_a[i] * 0.5
        # propose new vector
        a_new = best_a.copy()
        a_new[i] -= delta
        a_new[j] += delta
        # ensure nonnegative
        if a_new[i] < 0:
            continue
        sc_new = compute_score(a_new)
        if sc_new < best_sc:
            best_sc = sc_new
            best_a = a_new
    return best_a, best_sc

"""
Propose a non‑negative step‑function that minimizes

        2·n·max(conv(a,a)) / (sum(a)²)

The algorithm is a multi‑scale exponential‑gradient search:

1.  Work on a sequence of increasing lengths (256 → 1024 → 4096 → 8192 →
    16384).  At each resolution we start from the best shape found on the
    previous level (linear interpolation) and run a fast *soft* mirror
    descent (softmax‑weighted gradient) followed by a *hard* sub‑gradient
    phase that uses the exact max‑convolution.

2.  Both phases use exponentiated‑gradient updates and project back onto the
    probability simplex (∑a = 1, a ≥ 0).  The soft phase employs a
    temperature β ≈ 15 to spread the gradient over near‑maximal entries,
    while the hard phase uses a uniform weight on the true maximisers.

3.  After the gradient phases we perform a cheap stochastic local‑search
    that moves a small amount of mass between random coordinates and keeps
    any improving move.

The routine honours the supplied time budget (default 1000 s), prints a
compact log, and always returns a feasible list of floats.
"""

import time
import numpy as np
from typing import List, Tuple

# ----------------------------------------------------------------------
# Low‑level numeric utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """
    Linear convolution via a real FFT (full length, zero‑padded to a power‑of‑2).
    """
    m = len(a) + len(b) - 1
    n_fft = next_pow2(m)
    A = np.fft.rfft(a, n=n_fft)
    B = np.fft.rfft(b, n=n_fft)
    C = A * B
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


def normalize(v: np.ndarray) -> np.ndarray:
    """Project a non‑negative vector onto the simplex (sum = 1)."""
    v = np.maximum(v, 0.0)
    total = v.sum()
    if total <= 0.0:
        return np.full_like(v, 1.0 / len(v))
    return v / total


def compute_score(a: np.ndarray) -> float:
    """
    Fast objective (scale‑invariant).  The vector a is assumed non‑negative
    and already normalized (sum = 1).  Returns 2·n·max(conv(a,a)).
    """
    n = len(a)
    conv = convolution_fft(a, a)
    max_val = float(np.max(conv))
    return 2.0 * n * max_val


def gradient_from_weights(a: np.ndarray, w: np.ndarray) -> np.ndarray:
    """
    Gradient of Σ_k w_k·(a*a)_k w.r.t. a.
    Implements   g_j = 2· Σ_m a_m · w_{m+j}.
    Computed via one FFT‑based convolution.
    """
    w_rev = w[::-1]
    conv = convolution_fft(a, w_rev)               # length = 3·n‑2
    idx = 2 * len(a) - 2
    grads = 2.0 * conv[idx - np.arange(len(a))]
    return grads


# ----------------------------------------------------------------------
# Parametric families for the initial sweep
# ----------------------------------------------------------------------


def arcsine_initial(n: int) -> np.ndarray:
    """Discrete arcsine‑like shape a_i ∝ 1/√((i+½)(n-½-i)+ε)."""
    i = np.arange(n, dtype=float)
    eps = 0.25
    vals = 1.0 / np.sqrt((i + 0.5) * (n - 0.5 - i) + eps)
    return normalize(vals)


def powerlaw_shape(n: int, p: float, eps: float) -> np.ndarray:
    """Symmetric power‑law a_i ∝ ((i+ε)(n‑1‑i+ε))⁻ᵖ."""
    i = np.arange(n, dtype=float)
    base = ((i + eps) * (n - 1 - i + eps)) ** (-p)
    return normalize(base)


def uniform_shape(n: int) -> np.ndarray:
    """Uniform probability vector."""
    return np.full(n, 1.0 / n)


def random_dirichlet(n: int, rng: np.random.Generator) -> np.ndarray:
    """Sample a random point on the simplex."""
    samp = rng.exponential(1.0, size=n)
    return normalize(samp)


def upsample(a: np.ndarray, n: int) -> np.ndarray:
    """Linear interpolation (or simple averaging) to change the length."""
    m = len(a)
    if n == m:
        return a.copy()
    if n > m:
        x_old = np.linspace(0.0, 1.0, m)
        x_new = np.linspace(0.0, 1.0, n)
        return np.interp(x_new, x_old, a)
    # n < m  – aggregate consecutive entries
    factor = m // n
    trimmed = a[: factor * n]
    return trimmed.reshape(n, factor).mean(axis=1)


# ----------------------------------------------------------------------
# Refinement methods (smooth → hard → local search)
# ----------------------------------------------------------------------


def refine_smooth(
    a: np.ndarray,
    deadline: float,
    *,
    eta0: float = 1.0,
    beta: float = 15.0,
    report_every: int = 0,
) -> Tuple[np.ndarray, float]:
    """
    Multiplicative mirror‑descent with a softmax weight on the convolution.
    Returns the best iterate (exact score) before ``deadline``.
    """
    a = normalize(a.astype(float).copy())
    n = len(a)
    best_a = a.copy()
    best_sc = compute_score(best_a)

    it = 0
    while time.time() < deadline:
        it += 1
        conv = convolution_fft(a, a)
        max_val = float(np.max(conv))

        # softmax‑type weights
        exp_vals = np.exp(beta * (conv - max_val))
        w = exp_vals / exp_vals.sum()

        grad = gradient_from_weights(a, w)

        eta = eta0 / np.sqrt(it)

        a = a * np.exp(-eta * grad)
        a = normalize(a)

        cur_sc = compute_score(a)
        if cur_sc < best_sc:
            best_sc = cur_sc
            best_a = a.copy()

        if report_every and (it % report_every == 0):
            print(f"    smooth iter={it:,d} cur={cur_sc:.6f} best={best_sc:.6f}")

    return best_a, best_sc


def refine_subgradient(
    a: np.ndarray,
    deadline: float,
    *,
    max_iter: int = 200_000,
    eta0: float = 0.7,
    eps: float = 1e-12,
    report_every: int = 0,
) -> Tuple[np.ndarray, float]:
    """
    Classic sub‑gradient descent on the true max‑convolution objective.
    Uses exponentiated‑gradient updates and a 1/√t step schedule.
    """
    a = normalize(a.astype(float).copy())
    best_a = a.copy()
    best_sc = compute_score(a)

    n = len(a)
    for it in range(1, max_iter + 1):
        if time.time() >= deadline:
            break

        conv = convolution_fft(a, a)
        max_val = float(np.max(conv))

        # uniform weight on all (near) maximisers
        w = np.zeros_like(conv)
        max_idx = np.where(conv >= max_val - eps)[0]
        w[max_idx] = 1.0 / len(max_idx)

        grad = gradient_from_weights(a, w)

        eta = eta0 / np.sqrt(it)
        a = a * np.exp(-eta * grad)
        a = normalize(a)

        cur_sc = compute_score(a)
        if cur_sc < best_sc:
            best_sc = cur_sc
            best_a = a.copy()

        if report_every and (it % report_every == 0):
            print(f"    subgrad iter={it:,d} cur={cur_sc:.6f} best={best_sc:.6f}")

    return best_a, best_sc


def local_search(
    a: np.ndarray,
    deadline: float,
    rng: np.random.Generator,
    max_moves: int = 500_000,
    init_step: float = 0.2,
) -> Tuple[np.ndarray, float]:
    """
    Very cheap stochastic coordinate‑wise mass‑transfer search.
    Moves a random fraction of the mass from one index to another.
    Accepts only improving moves (hill‑climbing).
    """
    best_a = a.copy()
    best_sc = compute_score(best_a)
    n = len(a)

    it = 0
    while time.time() < deadline and it < max_moves:
        it += 1
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue

        # amount to move – a random fraction of the source entry
        delta = init_step * best_a[i] * rng.random()
        if delta <= 0.0:
            continue
        if delta > best_a[i]:
            delta = best_a[i] * 0.5

        a_new = best_a.copy()
        a_new[i] -= delta
        a_new[j] += delta

        # maintain non‑negativity
        if a_new[i] < 0.0:
            continue

        sc_new = compute_score(a_new)
        if sc_new < best_sc:
            best_sc = sc_new
            best_a = a_new

    return best_a, best_sc
