```
def refine_sequence(a_initial, deadline, n_iters, gamma_start=1.0, gamma_end=100.0):
    a = a_initial.copy()
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    lr = 0.05  # base learning rate; may adapt schedule.

    total_iters = n_iters
    for it in range(1, total_iters+1):
        if time.time() >= deadline:
            break
        # schedule gamma linearly or exponentially
        gamma = gamma_start + (gamma_end - gamma_start) * (it-1) / (total_iters-1)
        # compute conv and gradient
        conv = convolution_fft(a, a)  # length L
        # compute weights for smooth max
        maxc = np.max(conv)
        exp_vals = np.exp(gamma * (conv - maxc))  # prevent overflow
        sum_exp = np.sum(exp_vals)
        # smooth objective value (optional, not used)
        # grad calculation:
        w = exp_vals / sum_exp  # sum(w)=1
        # compute convolution of a with reversed w
        w_rev = w[::-1]
        conv_rev = convolution_fft(a, w_rev)
        L = len(conv)
        grad = 2.0 * conv_rev[L-1 - np.arange(len(a))]
        # Adam update
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad ** 2)
        m_hat = m / (1 - beta1 ** it)
        v_hat = v / (1 - beta2 ** it)
        step = lr * m_hat / (np.sqrt(v_hat) + eps)
        a = a - step
        a = project_to_simplex(a)
        if it % 5000 == 0:

    return a
```

```
def solve_max_sum_g(f, M):
    n = len(f)
    L = 2*n-1
    # Build sparse matrix A shape (L, n)
    # Each column j corresponds to f shifted by j.
    # Using diags: diagonals of A correspond to f entries.
    # For shift j (0-index), the diagonal offset = -j
    diags_data = []
    diags_offsets = []
    for j in range(n):
        # diagonal offset -j: entries A[i+j, j] = f[i] where i from 0..n-1 => i+j from j..j+n-1
        # So along diagonal offset j? Let's verify.
        # Using scipy.sparse.diags takes a list of diagonals for each offset.
        # We want matrix where column j has f[0] at row j, f[1] at row j+1, ..., f[n-1] at row j+n-1.
        # That is a diagonal with offset = j (row index - column index). Because row = column + offset.
        # So offset = j.
        diag = f  # length n
        diags_data.append(diag)
        diags_offsets.append(j)
    # Build matrix with shape (L, n) maybe each diagonal will expand beyond boundaries automatically.
    A = sp.diags(diags_data, diags_offsets, shape=(L, n), format='csr')
    # RHS b (M vector)

    res = linprog(c, A_ub=A, b_ub=b, bounds=[(0, None)]*n, method='highs')
```

```
        return None
    g = res.x
    if np.sum(g) == 0:
        return None
    g_norm = g / np.sum(g)
    return g_norm
```

The goal is to minimise  

\[
\text{score}(a)=\frac{2\,n\;\max\limits_{k}(a*a)_k}{\bigl(\sum_i a_i\bigr)^2},
\qquad a_i\ge 0 .
\]

Because the score is homogeneous of degree 0 we may fix the normalisation  
\(\sum_i a_i = 1\); the denominator then disappears and the problem reduces to

\[
\min_{a\ge 0,\;\sum a=1}\; 2n\;\max_k (a*a)_k .
\]

The function we optimise is the **maximum of a family of convex quadratic
functions**.  
A simple sub‑gradient descent works, but it converges slowly.
The programme below improves on the previous approach in three ways

1. **Smooth approximation of the max** – we replace  
   \(\max_k z_k\) by the log‑sum‑exp  

   \[
   S_\gamma(z)=\frac{1}{\gamma}\log\!\bigl(\sum_k e^{\gamma z_k}\bigr) ,\qquad 
   \gamma>0,
   \]

   whose gradient is a weighted average of the individual sub‑gradients.
   For a moderate \(\gamma\) the landscape is smooth, allowing fast
   optimisation with Adam; \(\gamma\) is increased during the run so that,
   at the end, the approximation is essentially the true max.

2. **Adam optimisation on the simplex** – we apply the Adam
   adaptive‑gradient method and project the iterate back onto the
   probability simplex after every step.  This gives a much more
   aggressive search direction than a plain sub‑gradient step.

3. **Linear‑programming “boost”** – after Adam (or sub‑gradient) polishing we
   solve the linear programme  

   \[
   \max_{g\ge0}\ \mathbf 1^\top g \quad
   \text{s.t.}\; (f*g)_k \le M\ \forall k,
   \]

   where \(f\) is the current best vector and
   \(M=\max_k (f*f)_k\).  The optimal \(g\) pushes mass into the
   positions where \(f\) is small without enlarging the convolution bound.
   Normalising \(g\) often yields a strictly better candidate.

All phases respect the global time budget (default 1000 s).  
Only NumPy, SciPy and the standard library are used.

```python
"""
Propose a low‑discrepancy step function.

The algorithm works in four stages:

1.  Generate a diverse pool of seeds (arcsine, beta‑families, power‑families,
    uniform, triangular bumps) for several lengths.
2.  Refine every seed with a smooth‑max Adam optimiser that gradually
    increases the soft‑max temperature (γ).  After each Adam step the vector
    is projected onto the probability simplex.
3.  A short exponentiated‑gradient polishing (the same update as the
    reference code) removes the last remaining spikes.
4.  For the best candidates we solve a tiny linear programme that
    maximises the L1‑mass that can be added without raising the
    convolution bound.  The normalised solution is a fresh candidate.

The best candidate found before the deadline is returned.
"""

import time
import numpy as np
from typing import List, Tuple

# ----------------------------------------------------------------------
# Low‑level utilities (FFT convolution, simplex projection, etc.)
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two that is ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full linear convolution using a real FFT."""
    m = a.size + b.size - 1
    n_fft = next_pow2(m)
    A = np.fft.rfft(a, n=n_fft)
    B = np.fft.rfft(b, n=n_fft)
    C = A * B
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


def compute_score(a: np.ndarray) -> float:
    """
    Fast surrogate for the official score.
    The vector is assumed to be normalised to Σa = 1, hence the denominator
    disappears and the score is simply 2·n·max(conv(a,a)).
    """
    n = a.shape[0]
    conv = convolution_fft(a, a)
    return 2.0 * n * float(np.max(conv))


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection onto the probability simplex
    {x ≥ 0, Σx = 1}.  Duchi, Shalev‑Shwartz, Singer & Chandra, 2008.
    """
    if np.all(v <= 0):
        # fallback – should never happen
        return np.full_like(v, 1.0 / v.size)

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho_arr = u - cssv / np.arange(1, v.size + 1)
    rho = np.where(rho_arr > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Shape generators (used for the initial pool)
# ----------------------------------------------------------------------


def arcsine_initial(n: int) -> np.ndarray:
    """Discrete arcsine shape: a_i ∝ 1/√((i+½)*(n‑½‑i))."""
    i = np.arange(n, dtype=float)
    eps = 0.25
    vals = 1.0 / np.sqrt((i + 0.5) * (n - 0.5 - i) + eps)
    vals /= vals.sum()
    return vals


def generate_beta_shape(n: int, alpha: float, beta: float, eps: float) -> np.ndarray:
    """Beta‑family: (i+ε)^{α‑1}·(n‑1‑i+ε)^{β‑1}."""
    i = np.arange(n, dtype=float)
    left = (i + eps) ** (alpha - 1.0)
    right = (n - 1 - i + eps) ** (beta - 1.0)
    base = left * right
    total = base.sum()
    if total <= 0.0 or not np.isfinite(total):
        return np.full(n, 1.0 / n)
    return base / total


def generate_power_shape(n: int, p: float, eps: float) -> np.ndarray:
    """Generalised arcsine: a_i ∝ ((i+½)*(n‑½‑i)+ε)^{‑p}."""
    i = np.arange(n, dtype=float)
    denom = ((i + 0.5) * (n - 0.5 - i) + eps) ** p
    vals = 1.0 / denom
    total = vals.sum()
    if total <= 0.0 or not np.isfinite(total):
        return np.full(n, 1.0 / n)
    return vals / total


def generate_triangular_shape(n: int, peak: int) -> np.ndarray:
    """Symmetric triangular bump, apex at `peak`."""
    left = np.arange(peak + 1, dtype=float)
    right = np.arange(n - peak - 1, dtype=float)[::-1]
    seq = np.concatenate([left, right])
    seq /= seq.sum()
    return seq


def generate_mixture_shape(n: int, weight_arc: float) -> np.ndarray:
    """Mixture weight_arc·arcsine + (1‑weight_arc)·uniform."""
    arc = arcsine_initial(n)
    uni = np.full(n, 1.0 / n)
    mix = weight_arc * arc + (1.0 - weight_arc) * uni
    mix /= mix.sum()
    return mix


# ----------------------------------------------------------------------
# Smooth‑max and Adam optimiser
# ----------------------------------------------------------------------


def smooth_max_and_grad(a: np.ndarray, gamma: float) -> Tuple[float, np.ndarray]:
    """
    Returns the smooth approximation Sγ(conv) and its gradient.
    Sγ = (1/γ) * log Σ_k exp(γ·conv_k) .
    Gradient = Σ_k w_k·(2·shift(a,k)), where w_k are the soft‑max weights.
    """
    n = a.shape[0]
    conv = convolution_fft(a, a)               # length L = 2n‑1
    max_c = np.max(conv)

    # soft‑max weights (stable)
    exp_vals = np.exp(gamma * (conv - max_c))
    sum_exp = exp_vals.sum()
    w = exp_vals / sum_exp

    # gradient = 2 * (a * w_rev) reversed
    w_rev = w[::-1]
    conv_rev = convolution_fft(a, w_rev)
    L = conv.shape[0]
    grad = 2.0 * conv_rev[L - 1 - np.arange(n)]

    # smooth value (not used for optimisation, only for diagnostics)
    smooth_val = max_c + np.log(sum_exp) / gamma
    return smooth_val, grad


def adam_refine(
    a0: np.ndarray,
    deadline: float,
    init_lr: float = 0.12,
    gamma_start: float = 1.0,
    gamma_end: float = 80.0,
    max_iter: int = 200_000,
) -> np.ndarray:
    """
    Adam optimiser on the smooth‑max surrogate.
    The learning‑rate schedule is η_t = init_lr / sqrt(t).
    γ grows linearly from gamma_start to gamma_end during the run.
    """
    a = a0.copy()
    n = a.shape[0]

    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8

    for it in range(1, max_iter + 1):
        if time.time() >= deadline:
            break

        # linear schedule for γ (slowly approach the true max)
        progress = min(1.0, it / max_iter)
        gamma = gamma_start + progress * (gamma_end - gamma_start)

        _, grad = smooth_max_and_grad(a, gamma)

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)

        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        step = init_lr * m_hat / (np.sqrt(v_hat) + eps)
        a = a - step
        a = project_to_simplex(a)

        if (it % 5_000) == 0:
            cur = compute_score(a)
            print(f"    Adam it={it:,d} γ={gamma:.2f} surrogate={cur:.6f}")

    return a

# Exponentiated‑gradient polish (identical to the reference implementation)
# ----------------------------------------------------------------------


def eg_refine(
    a: np.ndarray,
    deadline: float,
    max_iter: int = 80_000,
    eta0: float = 0.25,
) -> np.ndarray:
    """Mirror‑descent style polishing using the true sub‑gradient."""
    n = a.shape[0]
    L = 2 * n - 1

    for it in range(1, max_iter + 1):
        if time.time() >= deadline:
            break

        conv = convolution_fft(a, a)
        max_val = float(np.max(conv))
        tol = max_val * 1e-12 + 1e-14
        mask = (np.abs(conv - max_val) <= tol).astype(float)

        mask_rev = mask[::-1]
        conv_rev = convolution_fft(a, mask_rev)
        subgrad = 2.0 * conv_rev[L - 1 - np.arange(n)]

        eta = eta0 / (np.sqrt(it) + 1.0)
        a = a * np.exp(-eta * subgrad)

        total = a.sum()
        if total == 0.0:
            a = np.full_like(a, 1.0 / n)
        else:
            a = a / total

        if (it % 5_000) == 0:
            cur = compute_score(a)
            print(f"    EG   it={it:,d} score={cur:.6f}")

    return a


# ----------------------------------------------------------------------
# Linear‑programming “boost” (adds a fresh candidate)
# ----------------------------------------------------------------------


def lp_improve(f: np.ndarray, deadline: float) -> np.ndarray | None:
    """
    Solve   max Σ g_i   s.t.   (f * g)_k ≤ M  ∀k,   g_i ≥ 0,
    where M = max_k (f * f)_k.
    The feasible g is normalised to Σ g = 1 and returned.
    If the LP fails or yields the zero vector, ``None`` is returned.
    """
    try:
        from scipy.optimize import linprog
        import scipy.sparse as sp
    except Exception as exc:
        print("LP step unavailable:", exc)
        return None

    n = f.shape[0]
    L = 2 * n - 1
    M = float(np.max(convolution_fft(f, f)))

    # Build the Toeplitz constraint matrix A (shape L×n) as a sparse matrix.
    # Column j contains f shifted by j positions.
    diags_data = [f] * n
    diags_offsets = np.arange(0, n)          # offsets 0 … n‑1
    A = sp.diags(diags_data, diags_offsets, shape=(L, n), format="csr")

    c = -np.ones(n)                          # maximise Σ g_i  → minimise –Σ g_i
    b = np.full(L, M)
    bounds = [(0, None)] * n

    res = linprog(c, A_ub=A, b_ub=b, bounds=bounds, method="highs")
    if not res.success:
        print("  LP solve failed:", res.message)
        return None

    g = res.x
    if g.sum() == 0.0:
        return None
    g_norm = g / g.sum()
    return g_norm

# Main entry point required by the competition harness

def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs) -> List[float]:
    """
    Search for a non‑negative step sequence with a low ``evaluate_sequence``
    value.  The routine builds a pool of parametric seeds, refines them with
    a smooth‑max Adam optimiser, applies a short EG polish and finally tries
    a tiny linear‑programming boost.  The best official score is returned.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 2.0       # small safety margin

    best_seq: List[float] = []
    best_val: float = float("inf")

    # ------------------------------------------------------------------
    # 0.  Optional fallback from a previous run
    # ------------------------------------------------------------------
    if "height_sequence_1" in globals():
        try:
            fallback = [float(x) for x in globals()["height_sequence_1"]]
            val = evaluate_sequence(fallback)
            print(f"Fallback (height_sequence_1) score = {val:.6f}")
            if val < best_val:
                best_val, best_seq = val, fallback
        except Exception:
            pass

    # ------------------------------------------------------------------
    # 1.  Build a diverse pool of seeds (different lengths)
    # ------------------------------------------------------------------
    lengths = [512, 1024, 2048]          # larger sizes are slower but give better constants
    seeds: List[np.ndarray] = []

    for n in lengths:
        if time.time() >= deadline - 10.0:
            break
        print(f"\n=== Generating seeds for n = {n} ===")

        # deterministic starters
        seeds.append(arcsine_initial(n))
        seeds.append(np.full(n, 1.0 / n))

        # random beta‑family
        for _ in range(40):
            if time.time() >= deadline - 5.0:
                break
            a = rng.uniform(0.1, 5.0)
            b = rng.uniform(0.1, 5.0)
            eps = rng.uniform(0.0, 0.02 * n)
            seeds.append(generate_beta_shape(n, a, b, eps))

        # random power‑family
        for _ in range(40):
            if time.time() >= deadline - 5.0:
                break
            p = rng.uniform(0.1, 2.5)
            eps = rng.uniform(0.0, 2.0)
            seeds.append(generate_power_shape(n, p, eps))

        # mixtures of arcsine & uniform
        for _ in range(30):
            if time.time() >= deadline - 5.0:
                break
            w = rng.random()
            seeds.append(generate_mixture_shape(n, w))

        # triangular bumps
        for _ in range(30):
            if time.time() >= deadline - 5.0:
                break
            peak = rng.integers(0, n)
            seeds.append(generate_triangular_shape(n, peak))

    # ------------------------------------------------------------------
    # 2.  Keep only the most promising seeds (fast surrogate ranking)
    # ------------------------------------------------------------------
    scored = [(compute_score(v), v) for v in seeds]
    scored.sort(key=lambda x: x[0])
    top_seeds = [vec for _, vec in scored[:12]]
    print(f"\nSelected {len(top_seeds)} top seeds for refinement.")

    # ------------------------------------------------------------------
    # 3.  Refine each top seed (Adam → EG → LP boost)
    # ------------------------------------------------------------------
    for idx, seed_vec in enumerate(top_seeds, start=1):
        if time.time() >= deadline - 5.0:
            break
        n = seed_vec.shape[0]
        print(f"\nRefining seed {idx}/{len(top_seeds)} (n = {n}) …")

        # ----- Adam refinement -----
        sub_deadline = time.time() + min(30.0, deadline - time.time() - 5.0)
        refined = adam_refine(seed_vec, sub_deadline,
                              init_lr=0.12,
                              gamma_start=1.0,
                              gamma_end=80.0,
                              max_iter=150_000)

        # ----- Exponentiated‑gradient polish -----
        if time.time() < deadline - 4.0:
            eg_deadline = time.time() + min(10.0, deadline - time.time() - 4.0)
            refined = eg_refine(refined, eg_deadline,
                                max_iter=30_000,
                                eta0=0.22)

        # ----- Linear‑programming boost (optional) -----
        if time.time() < deadline - 3.0:
            g = lp_improve(refined, deadline)
            if g is not None:
                # try the pure LP solution and a modest blend
                for cand in (g, 0.5 * refined + 0.5 * g):
                    cand = project_to_simplex(cand)
                    try:
                        val = evaluate_sequence(cand.tolist())
                        print(f"  LP‑candidate official score = {val:.6f}")
                        if val < best_val:
                            best_val, best_seq = val, cand.tolist()
                            print(f"  *** New global best via LP: {best_val:.6f} ***")
                    except Exception as e:
                        print(f"  LP candidate evaluation failed: {e}")

        # ----- Official evaluation of the refined vector -----
        try:
            val = evaluate_sequence(refined.tolist())
            print(f"  → official score = {val:.6f}")
            if val < best_val:
                best_val, best_seq = val, refined.tolist()
                print(f"  *** New global best: {best_val:.6f} ***")
        except Exception as e:
            print(f"  Evaluation failed: {e}")

    # ------------------------------------------------------------------
    # 4.  Global polish on the current best (if time permits)
    # ------------------------------------------------------------------
    if best_seq and time.time() < deadline - 5.0:
        print("\nFinal global polish …")
        a = np.array(best_seq, dtype=float)

        # longer Adam run
        sub_deadline = time.time() + min(60.0, deadline - time.time() - 5.0)
        a = adam_refine(a, sub_deadline,
                        init_lr=0.09,
                        gamma_start=1.0,
                        gamma_end=100.0,
                        max_iter=250_000)

        # short EG polish
        if time.time() < deadline - 3.0:
            a = eg_refine(a, deadline,
                          max_iter=60_000,
                          eta0=0.18)

        try:
            final_val = evaluate_sequence(a.tolist())
            print(f"Final polished official score = {final_val:.6f}")
            if final_val < best_val:
                best_val, best_seq = final_val, a.tolist()
                print("*** Updated global best after final polish ***")
        except Exception as e:
            print("Final evaluation error:", e)

    # ------------------------------------------------------------------
    # 5.  Guard – guarantee a non‑empty return value
    # ------------------------------------------------------------------
    if not best_seq:
        fallback_n = 256
        best_seq = [1.0 / fallback_n] * fallback_n
        best_val = evaluate_sequence(best_seq)

    print(f"\nSearch finished. Best reported official score = {best_val:.6f}")
    return best_seq
```
