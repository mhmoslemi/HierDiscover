def smooth_adam_optimize(z0, deadline, n, beta0, beta_max, ramp, lr):
    """
    Optimize vector a = softmax(z) to minimize max conv using smooth
    approximation with exponential weighting and Adam.
    beta parameter grows from beta0 to beta_max across `ramp` iterations.
    """
    # ensure z is copy
    z = np.array(z0, dtype=float)
    m = np.zeros_like(z)
    v = np.zeros_like(z)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    t = 0
    while time.time() < deadline:
        t += 1
        # compute a via stable softmax
        z_max = np.max(z)
        exp_z = np.exp(z - z_max)
        a = exp_z / exp_z.sum()
        # compute convolution a*a
        conv = convolution_fft(a, a)
        max_c = np.max(conv)

        # schedule beta
        beta = beta0 + (beta_max - beta0) * min(1.0, t / float(ramp))
        # compute weight w
        shifted = conv - max_c
        # scaling for numeric stability: may use exponent with beta
        w = np.exp(beta * shifted)
        w_sum = w.sum()
        if w_sum == 0.0:
            w_norm = np.full_like(w, 1.0 / w.size)
        else:
            w_norm = w / w_sum
        # compute gradient wrt a
        w_rev = w_norm[::-1]
        conv_rev = convolution_fft(a, w_rev)
        # extract appropriate slice:
        grad_a = 2.0 * conv_rev[n-1 : n-1 + n]

        # chain rule gradient wrt z (softmax)
        dot = np.dot(grad_a, a)
        grad_z = a * (grad_a - dot)

        # Adam update
        m = beta1 * m + (1 - beta1) * grad_z
        v = beta2 * v + (1 - beta2) * (grad_z * grad_z)
        m_hat = m / (1 - beta1 ** t)
        v_hat = v / (1 - beta2 ** t)
        z = z - lr * m_hat / (np.sqrt(v_hat) + eps)

        # optionally print progress
        if t % 5000 == 0:
            cur_score = compute_score(a)
            print(f"   Adam it={t} beta={beta:.1f} score={cur_score:.6f}")

    # final a
    z_max = np.max(z)
    exp_z = np.exp(z - z_max)
    a = exp_z / exp_z.sum()
    return a

Main ideas
----------

* **Smooth max surrogate** – replace the non‑differentiable
  ``max(conv)`` by a log‑sum‑exp with a temperature β.
  For a fixed β the surrogate is convex and its gradient can be
  computed analytically via a weighted convolution.  As β grows the
  surrogate approaches the true maximum.

* **Soft‑max parametrisation** – the vector is represented as
  ``a = softmax(z)``.  Positivity and the simplex constraint Σa=1 are
  fulfilled automatically, so we can use an unconstrained optimiser.

* **Adam optimisation** – the smooth surrogate is minimised with the

    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Linear convolution via real FFT."""
    m = a.size + b.size - 1
    n_fft = next_pow2(m)
    A = np.fft.rfft(a, n=n_fft)
    B = np.fft.rfft(b, n=n_fft)
    C = A * B
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


def compute_score(a: np.ndarray) -> float:
    """Fast surrogate: 2·n·max(conv(a,a))   (a already sums to 1)."""
    n = a.shape[0]
    conv = convolution_fft(a, a)
    return 2.0 * n * float(np.max(conv))


# ----------------------------------------------------------------------
# Simple generators for diverse starting points
# ----------------------------------------------------------------------


def arcsine_initial(n: int) -> np.ndarray:
    i = np.arange(n, dtype=float)
    eps = 0.25
    vals = 1.0 / np.sqrt((i + 0.5) * (n - 0.5 - i) + eps)
    vals /= vals.sum()
    return vals


def generate_beta_shape(n: int, alpha: float, beta: float, eps: float) -> np.ndarray:
    i = np.arange(n, dtype=float)
    left = (i + eps) ** (alpha - 1.0)
    right = (n - 1 - i + eps) ** (beta - 1.0)
    base = left * right
    total = base.sum()
    if not np.isfinite(total) or total <= 0.0:
        return np.full(n, 1.0 / n)
    return base / total


def generate_power_shape(n: int, p: float, eps: float) -> np.ndarray:
    i = np.arange(n, dtype=float)
    denom = ((i + 0.5) * (n - 0.5 - i) + eps) ** p
    vals = 1.0 / denom
    total = vals.sum()
    if not np.isfinite(total) or total <= 0.0:
        return np.full(n, 1.0 / n)
    return vals / total


def generate_mixture_shape(n: int, weight_arc: float) -> np.ndarray:
    arc = arcsine_initial(n)
    uni = np.full(n, 1.0 / n)
    mix = weight_arc * arc + (1.0 - weight_arc) * uni
    mix /= mix.sum()
    return mix


# ----------------------------------------------------------------------
# Exact sub‑gradient descent (used for final polishing)
# ----------------------------------------------------------------------


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection onto the probability simplex."""
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / v.size)

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho_arr = u - cssv / np.arange(1, v.size + 1)
    rho = np.where(rho_arr > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def subgradient_descent_exact(
    init_a: np.ndarray,
    deadline: float,
    max_iter: int = 80_000,
    step0: float = 0.05,
) -> np.ndarray:
    """
    Projected sub‑gradient descent on f(a)=2·n·max(conv(a,a)).
    The iterate stays on the simplex.
    """
    a = project_to_simplex(init_a.copy())
    n = a.size
    L = 2 * n - 1

    for it in range(1, max_iter + 1):
        if time.time() >= deadline:
            break

        conv = convolution_fft(a, a)
        mx = float(np.max(conv))
        tol = mx * 1e-12 + 1e-14
        mask = (np.abs(conv - mx) <= tol).astype(float)

        # sub‑gradient = 2 * shift(a, K) summed over all tight indices K
        mask_rev = mask[::-1]
        rev = convolution_fft(a, mask_rev)
        subgrad = 2.0 * rev[L - 1 - np.arange(n)]

        eta = step0 / np.sqrt(it)
        a = a - eta * subgrad
        a = project_to_simplex(a)

        if it % 5000 == 0:
            print(f"    SubGrad it={it:,d} score={compute_score(a):.6f}")

    return a


# ----------------------------------------------------------------------
# Smooth max + Adam optimiser
# ----------------------------------------------------------------------


def softmax(z: np.ndarray) -> np.ndarray:
    """Numerically stable softmax."""
    z_max = np.max(z)
    e = np.exp(z - z_max)
    return e / e.sum()


def smooth_adam_optimize(
    z0: np.ndarray,
    deadline: float,
    n: int,
    beta0: float = 5.0,
    beta_max: float = 120.0,
    ramp_iters: int = 20000,
    lr: float = 0.05,
) -> np.ndarray:
    """
    Minimise the smooth approximation of max(conv) using Adam.
    The iterates are kept on the simplex via the soft‑max map.
    """
    z = np.array(z0, copy=True, dtype=float)
    m = np.zeros_like(z)
    v = np.zeros_like(z)
    beta1, beta2 = 0.9, 0.999
    eps = 1e-12
    t = 0

    while time.time() < deadline:
        t += 1
        a = softmax(z)                     # a ≥0 , Σa=1
        conv = convolution_fft(a, a)
        mx = float(np.max(conv))

        # linearly increase β
        beta = beta0 + (beta_max - beta0) * min(1.0, t / float(ramp_iters))

        # weight vector w ∝ exp(β·(conv – mx))
        shifted = conv - mx
        w = np.exp(beta * shifted)
        w_sum = w.sum()
        if w_sum == 0.0:
            w_norm = np.full_like(w, 1.0 / w.size)
        else:
            w_norm = w / w_sum

        # gradient w.r.t. a : 2 * (a ⋆ w_rev) restricted to original length
        w_rev = w_norm[::-1]
        rev = convolution_fft(a, w_rev)            # length = n + 2n‑2 - 1 = 3n‑2
        grad_a = 2.0 * rev[n - 1 : n - 1 + n]       # slice of length n

        # chain rule for soft‑max parametrisation
        dot = np.dot(grad_a, a)
        grad_z = a * (grad_a - dot)

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad_z
        v = beta2 * v + (1.0 - beta2) * (grad_z * grad_z)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        z = z - lr * m_hat / (np.sqrt(v_hat) + eps)

        if t % 5000 == 0:
            print(f"    Adam it={t:,d} β={beta:.1f} score={compute_score(a):.6f}")

    # final a
    a = softmax(z)
    return a

# Main entry point required by the harness
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs) -> List[float]:
    """
    Search for a step sequence with a low ``evaluate_sequence`` value.
    The routine mixes diverse parametric seeds, a smooth‑max Adam phase
    (the main optimiser) and a short exact sub‑gradient polishing phase.
    """
    rng = np.random.default_rng(seed)
    start = time.time()
    deadline = start + budget_s - 0.5          # tiny safety margin

    best_seq: List[float] = []
    best_val = float("inf")

    # ------------------------------------------------------------------
    # (optional) reuse a previously found sequence
    # ------------------------------------------------------------------
    if "height_sequence_1" in globals():
        try:
            cand = [float(x) for x in globals()["height_sequence_1"]]
            val = evaluate_sequence(cand)
            print(f"Fallback (height_sequence_1) score = {val:.6f}")
            if val < best_val:
                best_val, best_seq = val, cand
        except Exception:
            pass

    # ------------------------------------------------------------------
    # 1.  Choose lengths (larger → potentially better)
    # ------------------------------------------------------------------
    lengths = [4096, 2048, 1024]       # try the biggest first

    for n in lengths:
        if time.time() >= deadline - 10.0:
            break
        print(f"\n=== Optimising length {n} ===")

        # --------------------------------------------------------------
        # 1a. generate a modest pool of seeds
        # --------------------------------------------------------------
        seeds: List[np.ndarray] = [
            arcsine_initial(n),
            np.full(n, 1.0 / n),
            generate_mixture_shape(n, 0.5),
        ]

        # random beta‑family
        for _ in range(5):
            a = rng.uniform(0.1, 5.0)
            b = rng.uniform(0.1, 5.0)
            eps = rng.uniform(0.0, max(0.5, 0.02 * n))
            seeds.append(generate_beta_shape(n, a, b, eps))

        # random power‑shape
        for _ in range(5):
            p = rng.uniform(0.1, 3.0)
            eps = rng.uniform(0.0, 2.0)
            seeds.append(generate_power_shape(n, p, eps))

        # keep the most promising seeds according to the fast surrogate
        scored = [(compute_score(s), s) for s in seeds]
        scored.sort(key=lambda x: x[0])
        top_seeds = [vec for _, vec in scored[:4]]     # at most four per length

        # --------------------------------------------------------------
        # 1b. allocate a slice of time for each seed
        # --------------------------------------------------------------
        per_seed_time = max(2.0, (deadline - time.time()) / (len(top_seeds) * 2))

        for idx, seed_vec in enumerate(top_seeds, start=1):
            if time.time() >= deadline - 5.0:
                break

            print(f"\nSeed {idx}/{len(top_seeds)} (n={n}) – time budget ≈{per_seed_time:.1f}s")
            seed_deadline = time.time() + per_seed_time

            # initialise Adam via log‑softmax (avoids division by zero)
            eps = 1e-12
            z0 = np.log(seed_vec + eps)

            # ----------------------------------------------------------
            # 2.  Smooth‑max Adam optimisation
            # ----------------------------------------------------------
            a_opt = smooth_adam_optimize(
                z0,
                seed_deadline,
                n=n,
                beta0=5.0,
                beta_max=120.0,
                ramp_iters=20000,
                lr=0.05,
            )

            # ----------------------------------------------------------
            # 3.  Official evaluation
            # ----------------------------------------------------------
            try:
                val = evaluate_sequence(a_opt.tolist())
                print(f"   Official score = {val:.6f}")
                if val < best_val:
                    best_val, best_seq = val, a_opt.tolist()
                    print(f"   *** New global best: {best_val:.6f} ***")
            except Exception as e:
                print(f"   Evaluation failed: {e}")

            # ----------------------------------------------------------
            # 4.  Light exact polishing (if time permits)
            # ----------------------------------------------------------
            if time.time() < deadline - 5.0:
                polish_deadline = deadline - 2.0
                cand_arr = np.array(best_seq, dtype=float)
                cand_arr = subgradient_descent_exact(
                    cand_arr,
                    polish_deadline,
                    max_iter=50000,
                    step0=0.12,
                )
                try:
                    val2 = evaluate_sequence(cand_arr.tolist())
                    print(f"   Polished score = {val2:.6f}")
                    if val2 < best_val:
                        best_val, best_seq = val2, cand_arr.tolist()
                        print(f"   *** Updated best after polish: {best_val:.6f} ***")
                except Exception:
                    pass

    # ------------------------------------------------------------------
    # 5.  Final global polishing (if any time left)
    # ------------------------------------------------------------------
    if best_seq and time.time() < deadline - 3.0:
        print("\nFinal global polishing …")
        a_arr = np.array(best_seq, dtype=float)
        a_arr = subgradient_descent_exact(
            a_arr,
            deadline,
            max_iter=150_000,
            step0=0.08,
        )
        try:
            final_val = evaluate_sequence(a_arr.tolist())
            print(f"   Final official score = {final_val:.6f}")
            if final_val < best_val:
                best_val, best_seq = final_val, a_arr.tolist()
                print("*** Global best improved by final polish ***")
        except Exception:
            pass

    # ------------------------------------------------------------------
    # 6.  Guard – guarantee a non‑empty return
    # ------------------------------------------------------------------
    if not best_seq:
        n0 = 512
        best_seq = [1.0 / n0] * n0
        best_val = evaluate_sequence(best_seq)

    print(f"\nSearch finished. Best reported official score = {best_val:.6f}")
    return best_seq
