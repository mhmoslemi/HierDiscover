def compute_soft_subgrad(a, tau):
    conv = convolution_fft(a,a)
    # compute softmax of conv/tau
    max_c = conv.max()
    # subtract max for numerical stability.
    exps = np.exp((conv - max_c)/tau)
    w = exps / exps.sum()
    # reversed
    w_rev = w[::-1]
    subgrad = 2.0 * convolution_fft(a, w_rev)
    return subgrad

Then we do Adam update using subgrad (no extra scaling with (4n/τ)). That is essentially gradient of smoothed max times factor tau/(2n) (some scaling), but we treat step size accordingly.

"""
Improved optimisation for the step‑function bound.

The algorithm works in three layers:

1️⃣  **Broad seeding** –  Generate many diverse candidates for several
    lengths (256, 512, 1024, 2048).  Apart from the classic arcsine and
    uniform starters we sample families (beta, power, mixtures) and
    random Dirichlet draws with a wide range of concentration
    parameters.

2️⃣  **Soft‑subgradient Adam descent** –  Starting from the best seeds we
    run a full‑precision optimiser that minimises the smoothed objective

        φτ(a) = 2·n·log‑sum‑exp( (a∗a)/τ ),

    i.e. a smooth approximation of the original max‑norm.
    The gradient can be written as a *soft* sub‑gradient

        g = 2·conv(a, w_rev) ,

    where `w` is the soft‑max of the current autocorrelation and
    `w_rev` its reversal.  We embed this direction in the Adam optimiser
    (first‑ and second‑moment adaptation) and project the iterate back to
    the probability simplex after each step.  The temperature τ is
    decreased exponentially from a relatively large τ₀ (0.2) to a tiny
    τ_f (≈5·10⁻⁴) so the search first explores a smooth landscape and
    then concentrates on the true max‑norm.

3️⃣  **Exponentiated‑gradient polish** –  As in the reference solution we
    finish each run with a short mirror‑descent (EG) pass, which keeps
    the iterate on the simplex automatically and often squeezes a few
    more hundredths out of the bound.

All phases respect a global time budget (default 1000 s).  The best
vector found is finally evaluated with the official ``evaluate_sequence``
and returned.  The implementation uses only NumPy and the standard
library, stays well below the time limit and is fully self‑contained.
"""

import time
import numpy as np
from typing import List, Tuple

# ----------------------------------------------------------------------
# Low‑level utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Linear convolution via real FFT (length = len(a)+len(b)-1)."""
    m = len(a) + len(b) - 1
    n_fft = next_pow2(m)
    A = np.fft.rfft(a, n=n_fft)
    B = np.fft.rfft(b, n=n_fft)
    C = A * B
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


def compute_score(a: np.ndarray) -> float:
    """
    Fast surrogate for the official score.
    The vector is assumed to be normalised to Σa = 1, therefore the
    denominator disappears and the score is simply 2·n·max(conv(a,a)).
    """
    n = a.shape[0]
    conv = convolution_fft(a, a)
    return 2.0 * n * float(np.max(conv))


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection onto the probability simplex {x ≥ 0, Σx = 1}.
    Duchi et al., 2008.
    """
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / v.size)

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho_arr = u - cssv / np.arange(1, v.size + 1)
    rho = np.where(rho_arr > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w

# Shape generators
# ----------------------------------------------------------------------


def arcsine_initial(n: int) -> np.ndarray:
    """Classic arcsine shape: a_i ∝ 1/√((i+½)*(n‑½‑i))."""
    i = np.arange(n, dtype=float)
    eps = 0.25
    vals = 1.0 / np.sqrt((i + 0.5) * (n - 0.5 - i) + eps)
    vals /= vals.sum()
    return vals


def generate_beta_shape(n: int, alpha: float, beta: float, eps: float) -> np.ndarray:
    """
    (i+ε)^{α‑1}·(n‑1‑i+ε)^{β‑1} on {0,…,n‑1}, normalised to Σ=1.
    """
    i = np.arange(n, dtype=float)
    left = (i + eps) ** (alpha - 1.0)
    right = (n - 1 - i + eps) ** (beta - 1.0)
    base = left * right
    total = base.sum()
    if total <= 0.0 or not np.isfinite(total):
        return np.full(n, 1.0 / n)
    return base / total


def generate_power_shape(n: int, p: float, eps: float) -> np.ndarray:
    """
    Generalised arcsine: a_i ∝ ((i+½)*(n‑½‑i)+ε)^{‑p}.
    p>0, eps≥0.
    """
    i = np.arange(n, dtype=float)
    denom = ((i + 0.5) * (n - 0.5 - i) + eps) ** p
    vals = 1.0 / denom
    total = vals.sum()
    if total <= 0.0 or not np.isfinite(total):
        return np.full(n, 1.0 / n)
    return vals / total


def generate_triangular_shape(n: int, peak: int) -> np.ndarray:
    """Symmetric triangular bump with its apex at ``peak``."""
    left = np.arange(peak + 1, dtype=float)
    right = np.arange(n - peak - 1, dtype=float)[::-1]
    seq = np.concatenate([left, right])
    seq /= seq.sum()
    return seq


def generate_mixture_shape(n: int, weight_arc: float) -> np.ndarray:
    """Mixture (weight_arc)·arcsine + (1‑weight_arc)·uniform."""
    arc = arcsine_initial(n)
    uni = np.full(n, 1.0 / n)
    mix = weight_arc * arc + (1.0 - weight_arc) * uni
    mix /= mix.sum()
    return mix


def generate_dirichlet_shape(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    """Random Dirichlet with scalar concentration parameter."""
    gamma = rng.gamma(conc, 1.0, size=n)
    if gamma.sum() == 0.0:
        return np.full(n, 1.0 / n)
    return gamma / gamma.sum()

# Optimisation primitives

def soft_subgradient(a: np.ndarray, tau: float) -> np.ndarray:
    """
    Gradient of the smoothed max‑norm.
    Returns a vector of the same length as ``a``.
    """
    n = a.shape[0]
    conv = convolution_fft(a, a)                     # length L = 2n‑1
    max_c = np.max(conv)
    # stable soft‑max
    exps = np.exp((conv - max_c) / tau)
    w = exps / np.sum(exps)
    w_rev = w[::-1]
    L = conv.shape[0]
    conv_rev = convolution_fft(a, w_rev)             # length n+L‑1
    subgrad = 2.0 * conv_rev[L - 1 - np.arange(n)]
    return subgrad


def adam_optimize_soft(a0: np.ndarray,
                       deadline: float,
                       max_iter: int = 200_000,
                       step0: float = 0.08,
                       tau0: float = 0.20,
                       tau_end: float = 5e-4,
                       beta1: float = 0.9,
                       beta2: float = 0.999,
                       eps: float = 1e-8) -> np.ndarray:
    """
    Adam optimisation on the smoothed objective.
    The temperature τ decays exponentially from ``tau0`` to ``tau_end``.
    The iterate is projected onto the simplex after each step.
    """
    a = project_to_simplex(a0.copy())
    n = len(a)

    m = np.zeros_like(a)
    v = np.zeros_like(a)

    for it in range(1, max_iter + 1):
        if time.time() >= deadline:
            break

        # exponential decay of the temperature
        tau = max(tau_end,
                  tau0 * (tau_end / tau0) ** ((it - 1) / (max_iter - 1)))

        grad = soft_subgradient(a, tau)

        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)

        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        # Adam step (no extra τ scaling – τ influences the direction)
        a = a - step0 * m_hat / (np.sqrt(v_hat) + eps)

        a = project_to_simplex(a)

        if it % 5000 == 0:
            cur_sc = compute_score(a)
            print(f"    Adam it={it:,d} τ={tau:.6f} score={cur_sc:.6f}")

    return a


def eg_refine(a: np.ndarray,
              deadline: float,
              max_iter: int = 80_000,
              eta0: float = 0.25) -> np.ndarray:
    """
    Mirror‑descent style polishing using an exponentiated‑gradient update.
    Keeps the iterate on the simplex automatically.
    """
    n = a.shape[0]
    L = 2 * n - 1

    for it in range(1, max_iter + 1):
        if time.time() >= deadline:
            break

        conv = convolution_fft(a, a)
        max_val = float(np.max(conv))
        tol = max_val * 1e-12 + 1e-14
        mask = (np.abs(conv - max_val) <= tol).astype(float)

        # Gradient for the true max (hard mask)
        mask_rev = mask[::-1]
        conv_rev = convolution_fft(a, mask_rev)
        subgrad = 2.0 * conv_rev[L - 1 - np.arange(n)]

        eta = eta0 / (np.sqrt(it) + 1.0)
        a = a * np.exp(-eta * subgrad)

        total = a.sum()
        if total == 0.0:
            a = np.full_like(a, 1.0 / n)
        else:
            a = a / total

        if it % 5000 == 0:
            cur_sc = compute_score(a)
            print(f"    EG it={it:,d} score={cur_sc:.6f}")

    return a


def hillclimb_power_shape(n: int,
                          init_p: float,
                          init_eps: float,
                          deadline: float,
                          max_iters: int = 300) -> Tuple[np.ndarray, float]:
    """
    Random‑walk hill‑climbing on the (p, eps) parameters of the
    generalized arcsine family – uses the fast surrogate.
    """
    p, eps = init_p, init_eps
    best_seq = generate_power_shape(n, p, eps)
    best_sc = compute_score(best_seq)

    for _ in range(max_iters):
        if time.time() >= deadline:
            break
        p_new = max(0.01, p * (1.0 + np.random.uniform(-0.15, 0.15)))
        eps_new = max(0.0, eps + np.random.uniform(-0.1, 0.1))
        cand = generate_power_shape(n, p_new, eps_new)
        sc = compute_score(cand)
        if sc < best_sc:
            p, eps, best_seq, best_sc = p_new, eps_new, cand, sc
    return best_seq, best_sc


def hillclimb_beta_shape(n: int,
                         init_alpha: float,
                         init_beta: float,
                         init_eps: float,
                         deadline: float,
                         max_iters: int = 300) -> Tuple[np.ndarray, float]:
    """
    Random‑walk hill‑climbing on the three parameters of the beta family.
    """
    alpha, beta, eps = init_alpha, init_beta, init_eps
    best_seq = generate_beta_shape(n, alpha, beta, eps)
    best_sc = compute_score(best_seq)

    for _ in range(max_iters):
        if time.time() >= deadline:
            break
        alpha_new = max(0.01, alpha * (1.0 + np.random.uniform(-0.15, 0.15)))
        beta_new = max(0.01, beta * (1.0 + np.random.uniform(-0.15, 0.15)))
        eps_new = max(0.0, eps + np.random.uniform(-0.1, 0.1))
        cand = generate_beta_shape(n, alpha_new, beta_new, eps_new)
        sc = compute_score(cand)
        if sc < best_sc:
            alpha, beta, eps, best_seq, best_sc = alpha_new, beta_new, eps_new, cand, sc
    return best_seq, best_sc

# Main entry point required by the competition harness
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42,
                     budget_s: int = 1000,
                     **kwargs) -> List[float]:
    """
    Search for a non‑negative step sequence with a low
    ``evaluate_sequence`` value.  The routine mixes
    (i) a broad pool of parametrised seeds,
    (ii) soft‑subgradient Adam optimisation with a temperature schedule,
    (iii) a final exponentiated‑gradient polish.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 1.0      # safety margin

    best_seq: List[float] = []
    best_val: float = float("inf")

    # ------------------------------------------------------------------
    # 0.  Fallback to a known good construction if present
    # ------------------------------------------------------------------
    if "height_sequence_1" in globals():
        try:
            fallback = [float(x) for x in globals()["height_sequence_1"]]
            val = evaluate_sequence(fallback)
            print(f"Fallback (height_sequence_1) score = {val:.6f}")
            if val < best_val:
                best_val = val
                best_seq = fallback
        except Exception:
            pass

    # ------------------------------------------------------------------
    # 1.  Build a rich candidate pool (different lengths)
    # ------------------------------------------------------------------
    lengths = [256, 512, 1024, 2048]   # increasing resolution
    global_candidates: List[Tuple[float, np.ndarray]] = []

    for n in lengths:
        if time.time() >= deadline - 12.0:
            break
        print(f"\n=== Length n = {n} ===")

        # deterministic starters
        starters = [np.full(n, 1.0 / n), arcsine_initial(n)]
        for st in starters:
            sc = compute_score(st)
            global_candidates.append((sc, st.copy()))

        # random beta‑family
        for _ in range(120):
            if time.time() >= deadline - 5.0:
                break
            a = rng.uniform(0.05, 4.0)
            b = rng.uniform(0.05, 4.0)
            eps = rng.uniform(0.0, max(1.0, 0.02 * n))
            cand = generate_beta_shape(n, a, b, eps)
            sc = compute_score(cand)
            global_candidates.append((sc, cand))

        # random power‑shape
        for _ in range(120):
            if time.time() >= deadline - 5.0:
                break
            p = rng.uniform(0.1, 2.0)
            eps = rng.uniform(0.0, 2.0)
            cand = generate_power_shape(n, p, eps)
            sc = compute_score(cand)
            global_candidates.append((sc, cand))

        # mixtures of arcsine & uniform
        for _ in range(80):
            if time.time() >= deadline - 5.0:
                break
            w = rng.random()
            cand = generate_mixture_shape(n, w)
            sc = compute_score(cand)
            global_candidates.append((sc, cand))

        # Dirichlet draws with varied concentration
        for _ in range(80):
            if time.time() >= deadline - 5.0:
                break
            # log‑uniform concentration between 0.01 and 10
            conc = 10 ** rng.uniform(-2.0, 1.0)
            cand = generate_dirichlet_shape(n, conc, rng)
            sc = compute_score(cand)
            global_candidates.append((sc, cand))

        # simple triangular bumps
        for _ in range(60):
            if time.time() >= deadline - 5.0:
                break
            peak = rng.integers(0, n)
            cand = generate_triangular_shape(n, peak)
            sc = compute_score(cand)
            global_candidates.append((sc, cand))

        # ------------------------------------------------------------------
        # 1b.  Local optimisation on the two main parametric families
        # ------------------------------------------------------------------
        cand, sc = hillclimb_power_shape(n, 0.5, 0.0, deadline, max_iters=250)
        global_candidates.append((sc, cand))

        cand, sc = hillclimb_beta_shape(n, 0.5, 0.5, 0.0, deadline, max_iters=250)
        global_candidates.append((sc, cand))

    # keep the most promising seeds
    global_candidates.sort(key=lambda x: x[0])
    top_seeds = [vec for _, vec in global_candidates[:10]]
    print(f"\nSelected {len(top_seeds)} top seeds for intensive refinement "
          f"(surrogate range {global_candidates[0][0]:.6f} – "
          f"{global_candidates[-1][0]:.6f})")

    # ------------------------------------------------------------------
    # 2.  Intensive refinement (Adam on smoothed max + EG polish)
    # ------------------------------------------------------------------
    for idx, seed_vec in enumerate(top_seeds, start=1):
        if time.time() >= deadline - 5.0:
            break
        n = seed_vec.shape[0]
        print(f"\nRefining seed {idx}/{len(top_seeds)} (n={n}) …")

        # Allocate a generous amount of iterations – the loop will stop early if out of time
        remaining = deadline - time.time()
        max_iter_adam = int(min(150_000, remaining * 30))

        refined = adam_optimize_soft(seed_vec, deadline,
                                     max_iter=max_iter_adam,
                                     step0=0.08,
                                     tau0=0.20,
                                     tau_end=5e-4)

        # EG polishing – cheap but often helpful
        if time.time() < deadline - 2.0:
            remaining = deadline - time.time()
            max_iter_eg = int(min(80_000, remaining * 30))
            refined = eg_refine(refined, deadline,
                                max_iter=max_iter_eg,
                                eta0=0.25)

        # Evaluate with the official metric
        try:
            val = evaluate_sequence(refined.tolist())
            print(f"  → official score = {val:.6f}")
            if val < best_val:
                best_val = val
                best_seq = refined.tolist()
                print(f"  *** New global best: {best_val:.6f} ***")
        except Exception as e:
            print(f"  Evaluation failed: {e}")

    # ------------------------------------------------------------------
    # 3.  Global polish on the current best (if time permits)
    # ------------------------------------------------------------------
    if best_seq and time.time() < deadline - 3.0:
        print("\nFinal global polish …")
        a = np.array(best_seq, dtype=float)

        remaining = deadline - time.time()
        max_iter_adam = int(min(200_000, remaining * 30))
        a = adam_optimize_soft(a, deadline,
                               max_iter=max_iter_adam,
                               step0=0.06,
                               tau0=0.12,
                               tau_end=5e-4)

        if time.time() < deadline - 2.0:
            remaining = deadline - time.time()
            max_iter_eg = int(min(120_000, remaining * 30))
            a = eg_refine(a, deadline,
                          max_iter=max_iter_eg,
                          eta0=0.20)

        final_val = evaluate_sequence(a.tolist())
        print(f"Final polished official score = {final_val:.6f}")
        if final_val < best_val:
            best_val = final_val
            best_seq = a.tolist()
            print("*** Updated global best after final polish ***")

    # ------------------------------------------------------------------
    # 4.  Fallback guard
    # ------------------------------------------------------------------
    if not best_seq:
        fallback_n = 256
        best_seq = [1.0 / fallback_n] * fallback_n
        best_val = evaluate_sequence(best_seq)

    print(f"\nSearch finished. Best reported official score = {best_val:.6f}")
    return best_seq
