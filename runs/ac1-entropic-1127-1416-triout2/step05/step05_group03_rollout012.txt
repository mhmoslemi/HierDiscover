def adam_optimize_exp(initial_a: np.ndarray, time_deadline: float, max_iter=..., lr0=0.1, lam_start=1., lam_end=400., noise_every=..., noise_scale=...):

   w = np.log(initial_a + 1e-30)  # logit

   m = np.zeros_like(w)

   v = np.zeros_like(w)

   best_a = None; best_sc = inf

   For iteration it=1..max_iter:

      if time.time() >= time_deadline: break

      a = softmax(w)

      # optionally noise addition: a_before = a, then perturb w with noise after step?

      lam = lam_start + (lam_end - lam_start) * (it/max_iter)**0.5

      grad_a, _ = smooth_gradient(a, lam)   # gradient of objective (w.r.t a)

      # compute grad_w = a * (grad_a - dot(a, grad_a))
      avg = np.dot(a, grad_a)
      grad_w = a * (grad_a - avg)

      # Adam updates
      m = beta1 * m + (1-beta1) * grad_w

      v = beta2 * v + (1-beta2) * (grad_w**2)

      m_hat = m / (1 - beta1**it)

      v_hat = v / (1 - beta2**it)

      step = lr0 * m_hat / (np.sqrt(v_hat) + eps_adam)

      w = w - step

      # optional noise
      if noise_every and (it % noise_every == 0):
          w += np.random.randn(w.shape[0]) * noise_scale

      a = softmax(w)  # recompute after update

      cur_sc = compute_score(a)
      if cur_sc < best_sc:
         best_sc = cur_sc
         best_a = a.copy()

   return best_a, best_sc

def subgradient_optimize_exp(initial_a, time_deadline, step0=0.3, max_iter=...):

    a = initial_a.copy()
    best_a = a.copy()
    best_sc = compute_score(a)

    n = len(a)
    for it in range(1, max_iter+1):
        if time.time() >= time_deadline: break
        conv = convolution_fft(a, a)
        max_val = np.max(conv)
        idx = int(np.argmax(conv))
        # gradient of max conv w.r.t a
        i = np.arange(n)
        j = idx - i
        mask = (j >= 0) & (j < n)
        grad = np.zeros_like(a)
        grad[mask] = 2.0 * a[j[mask]]  # derivative of maxconv term

        # scaled step size
        step = step0 / np.sqrt(it)  # maybe less
        # exponentiated update
        a = a * np.exp(-step * grad)
        sum_a = a.sum()
        if sum_a == 0 or not np.isfinite(sum_a):
            a = best_a.copy()  # revert to previous best? maybe add fallback
            continue
        a = a / sum_a

        cur_sc = compute_score(a)
        if cur_sc < best_sc:
            best_sc = cur_sc
            best_a = a.copy()
    return best_a, best_sc

"""
Propose a step‑function (non‑negative vector) with a low value of

    2·n·max(conv(a,a)) / (Σ a)²

The algorithm:

1. **Sample** many Beta‑family, mixture‑Beta and Dirichlet shapes for a few
   candidate lengths (512 … 4096).  Keep the best according to the fast
   surrogate 2·n·max(conv).

2. **Smooth Adam optimisation** – work on the log‑its (softmax) of the
   vector, i.e.  a_i = exp(w_i) / Σ_j exp(w_j).  This avoids the costly
   Euclidean projection onto the simplex.  The Adam optimiser is run on
   the log‑sum‑exp surrogate of the max‑convolution with a temperature
   λ that is annealed up to 400.

3. **Exact sub‑gradient descent** – a multiplicative (exponentiated)
   update on the true max‑convolution gradient is performed; the step
   size decays as 1/√t.

4. **Stochastic polishing** – a cheap Gaussian perturbation walk on the
   simplex (multiplicative noise) followed by a simple simulated‑annealing
   phase.

5. The best vector found for each length is officially evaluated
   (``evaluate_sequence``).  The overall best is finally polished with
   longer Adam / sub‑gradient / annealing runs.

All steps respect the global 1000 s budget and guarantee a non‑empty
return.
"""

import time
import numpy as np
from typing import List, Tuple

# ----------------------------------------------------------------------
# Linear‑algebra utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Linear convolution via real FFT (returns length len(a)+len(b)-1)."""
    m = a.size + b.size - 1
    n_fft = next_pow2(m)
    A = np.fft.rfft(a, n=n_fft)
    B = np.fft.rfft(b, n=n_fft)
    C = A * B
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


def max_convolution(a: np.ndarray) -> float:
    """Maximum entry of the self‑convolution of a."""
    return float(np.max(convolution_fft(a, a)))


def compute_score(a: np.ndarray) -> float:
    """
    Fast surrogate: 2·n·max(conv(a,a)).
    For a normalized vector (Σa = 1) this equals the official score.
    """
    n = a.shape[0]
    return 2.0 * n * max_convolution(a)


# ----------------------------------------------------------------------
# Shape generators
# ----------------------------------------------------------------------


def generate_beta_shape(n: int, alpha: float, beta: float, eps: float) -> np.ndarray:
    """Beta‑family density on {0,…,n‑1} (symmetric).  Returned vector is normalised."""
    i = np.arange(n, dtype=float)
    left = (i + eps) ** (alpha - 1.0)
    right = (n - 1 - i + eps) ** (beta - 1.0)
    base = left * right
    total = base.sum()
    if total <= 0.0 or not np.isfinite(total):
        return np.full(n, 1.0 / n)
    return base / total


def generate_beta_mixture(
    n: int,
    alpha1: float,
    beta1: float,
    eps1: float,
    alpha2: float,
    beta2: float,
    eps2: float,
    w: float,
) -> np.ndarray:
    """Convex combination of two beta‑family densities, renormalised."""
    a1 = generate_beta_shape(n, alpha1, beta1, eps1)
    a2 = generate_beta_shape(n, alpha2, beta2, eps2)
    mix = w * a1 + (1.0 - w) * a2
    if mix.sum() == 0.0 or not np.isfinite(mix.sum()):
        return np.full(n, 1.0 / n)
    return mix / mix.sum()


def arcsine_initial(n: int) -> np.ndarray:
    """Classic discrete arcsine shape (symmetrised)."""
    i = np.arange(n, dtype=float)
    eps = 0.25
    vals = 1.0 / np.sqrt((i + 0.5) * (n - 0.5 - i) + eps)
    return vals / vals.sum()


def generate_dirichlet_shape(n: int, concentration: float, rng: np.random.Generator) -> np.ndarray:
    """Sample a Dirichlet distribution with a single concentration (alpha)."""
    draws = rng.gamma(concentration, 1.0, size=n)
    total = draws.sum()
    if total <= 0.0 or not np.isfinite(total):
        return np.full(n, 1.0 / n)
    return draws / total


# ----------------------------------------------------------------------
# Smooth surrogate (log‑sum‑exp) and its gradient
# ----------------------------------------------------------------------


def smooth_gradient(a: np.ndarray, lam: float) -> Tuple[np.ndarray, float]:
    """
    Gradient of the log‑sum‑exp surrogate of max(conv(a,a)).
    Returns (gradient w.r.t a, surrogate value B).
    """
    n = a.shape[0]
    conv = convolution_fft(a, a)              # length 2n‑1
    max_c = np.max(conv)

    # stable log‑sum‑exp
    exp_term = np.exp(lam * (conv - max_c))
    sum_exp = np.sum(exp_term)
    w = exp_term / sum_exp                   # soft‑max over convolution entries
    B = max_c + np.log(sum_exp) / lam        # smooth approximation of the max

    # weighted sum of sub‑gradients:
    # ∂/∂a Σ_k w_k (a*a)_k = 2 * (a ⋆ w_rev)
    w_rev = w[::-1]
    conv_rev = convolution_fft(a, w_rev)     # length 3n‑2
    L = 2 * n - 1
    weighted_grad = 2.0 * conv_rev[L - 1 - np.arange(n)]

    s = np.sum(a)
    term1 = (2.0 * n / (s ** 2)) * weighted_grad
    term2 = -(4.0 * n * B / (s ** 3)) * np.ones_like(a)
    grad = term1 + term2
    return grad, B


# ----------------------------------------------------------------------
# Softmax utilities (exponentiated‑gradient representation)
# ----------------------------------------------------------------------


def softmax(w: np.ndarray) -> np.ndarray:
    """Numerically stable softmax."""
    w_max = np.max(w)
    exp_w = np.exp(w - w_max)
    sum_exp = exp_w.sum()
    return exp_w / sum_exp


# ----------------------------------------------------------------------
# Optimisers working on the log‑its w (so the simplex constraint is free)
# ----------------------------------------------------------------------


def adam_optimize_exp(
    initial_a: np.ndarray,
    time_deadline: float,
    max_iter: int = 200_000,
    lr0: float = 0.08,
    lam_start: float = 1.0,
    lam_end: float = 400.0,
    noise_every: int = 1500,
    noise_scale: float = 5e-5,
    rng: np.random.Generator = None,
) -> Tuple[np.ndarray, float]:
    """
    Adam on the log‑sum‑exp surrogate, optimisation in the log‑softmax space.
    Returns the best vector (as an ndarray) and its fast score.
    """
    eps = 1e-30
    w = np.log(np.maximum(initial_a, eps))
    n = w.shape[0]

    # Adam moments
    m = np.zeros_like(w)
    v = np.zeros_like(w)

    beta1, beta2 = 0.9, 0.999
    eps_adam = 1e-8

    best_a = softmax(w)
    best_sc = compute_score(best_a)

    for it in range(1, max_iter + 1):
        if time.time() >= time_deadline:
            break

        a = softmax(w)

        # temperature schedule (quadratic-root annealing)
        t = it / max_iter
        lam = lam_start + (lam_end - lam_start) * (t ** 0.5)

        grad_a, _ = smooth_gradient(a, lam)           # ∂obj/∂a
        avg = np.dot(a, grad_a)                        # scalar ⟨a,∇⟩
        grad_w = a * (grad_a - avg)                    # chain rule for w

        # Adam update on w
        m = beta1 * m + (1.0 - beta1) * grad_w
        v = beta2 * v + (1.0 - beta2) * (grad_w ** 2)

        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        step = lr0 * m_hat / (np.sqrt(v_hat) + eps_adam)

        w = w - step

        # optional stochastic perturbation
        if noise_every and (it % noise_every == 0):
            if rng is None:
                rng = np.random.default_rng()
            w += rng.standard_normal(n) * noise_scale

        # evaluate current vector
        cur_sc = compute_score(softmax(w))
        if cur_sc < best_sc:
            best_sc = cur_sc
            best_a = softmax(w).copy()

        if it % 5000 == 0:
            print(f"    Adam‑exp iter={it:,d} cur={cur_sc:.6f} best={best_sc:.6f}")

    return best_a, best_sc


def subgradient_optimize_exp(
    initial_a: np.ndarray,
    time_deadline: float,
    step0: float = 0.25,
    max_iter: int = 200_000,
) -> Tuple[np.ndarray, float]:
    """
    Projected (multiplicative) sub‑gradient descent on the true max‑convolution.
    Works directly on the probability simplex via exponentiated updates.
    """
    a = initial_a.copy()
    best_a = a.copy()
    best_sc = compute_score(a)

    n = a.shape[0]

    for it in range(1, max_iter + 1):
        if time.time() >= time_deadline:
            break

        conv = convolution_fft(a, a)
        idx = int(np.argmax(conv))

        # gradient of max‑convolution w.r.t. a
        i = np.arange(n)
        j = idx - i
        mask = (j >= 0) & (j < n)
        grad = np.zeros_like(a)
        grad[mask] = 2.0 * a[j[mask]]          # ∂maxconv/∂a_i

        # step size decays as 1/√t
        step = step0 / np.sqrt(it)

        # exponentiated (multiplicative) update
        a = a * np.exp(-step * grad)
        a_sum = a.sum()
        if a_sum <= 0.0 or not np.isfinite(a_sum):
            a = best_a.copy()                 # fall back to the best seen so far
            continue
        a = a / a_sum

        cur_sc = compute_score(a)
        if cur_sc < best_sc:
            best_sc = cur_sc
            best_a = a.copy()

        if it % 40000 == 0:
            print(f"    Sub‑grad‑exp iter={it:,d} cur={cur_sc:.6f} best={best_sc:.6f}")

    return best_a, best_sc


def random_local_search_exp(
    initial_a: np.ndarray,
    time_deadline: float,
    rng: np.random.Generator,
    init_step: float = 0.015,
    max_iter: int = 250_000,
) -> Tuple[np.ndarray, float]:
    """
    Gaussian multiplicative walk on the simplex with a tiny uphill‑move probability.
    """
    a = initial_a.copy()
    best_a = a.copy()
    best_sc = compute_score(a)

    n = a.shape[0]

    for it in range(1, max_iter + 1):
        if time.time() >= time_deadline:
            break

        step = init_step / np.sqrt(it)
        noise = rng.normal(scale=step, size=n)
        a_prop = a * np.exp(noise)               # multiplicative perturbation
        a_prop = a_prop / a_prop.sum()           # re‑normalise

        cur_sc = compute_score(a_prop)

        # Accept if better; otherwise accept with a tiny probability
        if cur_sc < best_sc or (rng.random() < 1e-4 and cur_sc < 10 * best_sc):
            a = a_prop
            if cur_sc < best_sc:
                best_sc = cur_sc
                best_a = a_prop.copy()

        if it % 50000 == 0:
            print(f"    Random‑exp iter={it:,d} cur={cur_sc:.6f} best={best_sc:.6f}")

    return best_a, best_sc


def simulated_annealing_exp(
    initial_a: np.ndarray,
    time_deadline: float,
    rng: np.random.Generator,
    init_temp: float = 0.05,
    decay: float = 0.9995,
    max_iter: int = 200_000,
) -> Tuple[np.ndarray, float]:
    """
    Simple Metropolis annealing on the simplex (multiplicative Gaussian proposals).
    """
    a_cur = initial_a.copy()
    best_a = a_cur.copy()
    best_sc = compute_score(a_cur)

    n = a_cur.shape[0]

    for it in range(1, max_iter + 1):
        if time.time() >= time_deadline:
            break
        T = init_temp * (decay ** it)
        if T < 1e-7:
            break

        noise = rng.normal(scale=np.sqrt(T), size=n)
        a_prop = a_cur * np.exp(noise)
        a_prop = a_prop / a_prop.sum()

        cur_sc = compute_score(a_prop)
        delta = cur_sc - best_sc
        if delta < 0 or rng.random() < np.exp(-delta / T):
            a_cur = a_prop
            if cur_sc < best_sc:
                best_sc = cur_sc
                best_a = a_prop.copy()

        if it % 50000 == 0:
            print(f"    Anneal‑exp iter={it:,d} T={T:.5f} best={best_sc:.6f}")

    return best_a, best_sc
