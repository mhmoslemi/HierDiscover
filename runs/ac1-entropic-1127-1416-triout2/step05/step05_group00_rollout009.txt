"""
Improved search for a low‑score step function.

The algorithm consists of four phases:

1. **Random Dirichlet restarts** – generate many probability vectors
   (sum = 1) of length ``n`` using Dirichlet distributions with both
   sparse (α<1) and smooth (α>1) shapes.  Keep the best candidate.

2. **Linear‑programming update** – maximise the total mass of a new
   vector ``b`` under the constraints ``a⊛b ≤ M`` where ``M`` is the
   current peak of ``a⊛a``.  The LP is solved with SciPy's ``linprog``
   (the HiGHS solver, which is available in the execution environment).
   The obtained ``b`` is mixed into ``a`` with a small step and projected
   back onto the simplex.

3. **Adam‑accelerated sub‑gradient descent** – minimise the true
   objective ``2·n·max(a⊛a)`` (the sum is fixed to 1) by following the
   sub‑gradient of the maximal convolution entry.  Adam‑style momentum
   and bias‑correction give a more stable descent and allow many
   iterations within the time budget.

4. **Simulated‑annealing hill‑climb** – stochastic mass‑transfer moves
   are accepted according to a temperature schedule, allowing the
   search to escape shallow local minima.  The best vector found in any
   phase is returned.

All phases respect the 1000 s budget, use only NumPy/SciPy and perform
no I/O.  The returned list can be fed directly to ``evaluate_sequence``.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full convolution a ⊛ b using a real FFT (fast)."""
    m = a.shape[0] + b.shape[0] - 1
    n_fft = next_pow2(m)
    A = np.fft.rfft(a, n=n_fft)
    B = np.fft.rfft(b, n=n_fft)
    C = A * B
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


def compute_score(a: np.ndarray) -> float:
    """
    Fast replica of ``evaluate_sequence`` when ∑a = 1.
    Returns 2·n·max(a⊛a).
    """
    n = a.shape[0]
    conv = convolution_fft(a, a)
    return 2.0 * n * float(np.max(conv))


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection onto the probability simplex (non‑negative, sum = 1)."""
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / v.shape[0])

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, v.shape[0] + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.where(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


# ----------------------------------------------------------------------
# Linear programming (SciPy) – maximise ∑b subject to a⊛b ≤ M, b ≥ 0
# ----------------------------------------------------------------------
try:
    from scipy.optimize import linprog
except Exception:  # pragma: no cover
    linprog = None


def build_convolution_matrix(a: np.ndarray) -> np.ndarray:
    """
    Construct the matrix C (shape (2n‑1, n)) such that C @ b = a⊛b
    (full convolution).  Column i contains `a` shifted by i positions.
    """
    n = a.shape[0]
    rows = 2 * n - 1
    C = np.zeros((rows, n), dtype=float)
    # each column i contributes a shifted copy of `a`
    for i in range(n):
        C[i:i + n, i] = a
    return C


def solve_lp_for_b(a: np.ndarray, M: float):
    """
    Solve   maximise   1ᵀb   subject to   b ≥ 0,  a⊛b ≤ M.
    Returns the optimal b (sum(b) > 0) or ``None`` on failure.
    """
    if linprog is None:
        return None

    n = a.shape[0]
    C = build_convolution_matrix(a)

    # linprog minimises, so we use -1 as objective to maximise sum(b)
    c = -np.ones(n, dtype=float)

    bounds = [(0.0, None) for _ in range(n)]

    # Upper bounds for each convolution entry
    ub = np.full(C.shape[0], M, dtype=float)

    # Solve
    res = linprog(
        c,
        A_ub=C,
        b_ub=ub,
        bounds=bounds,
        method="highs",
        options={"presolve": True, "time_limit": 30},
    )

    if not res.success or res.x is None:
        return None
    return np.asarray(res.x, dtype=float)


# ----------------------------------------------------------------------
# Adam‑accelerated sub‑gradient descent on max(a⊛a)
# ----------------------------------------------------------------------
def subgrad_adam(
    a_init: np.ndarray,
    deadline: float,
    max_iter: int = 300_000,
    lr: float = 0.08,
    beta1: float = 0.9,
    beta2: float = 0.999,
    eps: float = 1e-8,
) -> tuple[np.ndarray, float]:
    """
    Adam‑style projected sub‑gradient descent.
    The objective is 2·n·max(a⊛a) (sum(a) = 1 is enforced by the projection).
    Returns the best vector seen and its score.
    """
    a = a_init.astype(float).copy()
    best_a = a.copy()
    best_score = compute_score(a)
    n = a.shape[0]

    m = np.zeros_like(a)   # first moment
    v = np.zeros_like(a)   # second moment

    pow_b1 = 1.0
    pow_b2 = 1.0

    for t in range(1, max_iter + 1):
        if time.time() >= deadline:
            break

        conv = convolution_fft(a, a)
        max_val = np.max(conv)
        # indices where the maximum is attained (within a tiny tolerance)
        max_inds = np.where(conv >= max_val - 1e-13)[0]

        # sub‑gradient = sum_{k∈max_inds} 2·shift(a, k)
        g = np.zeros_like(a)
        idx = np.arange(n)
        for k in max_inds:
            j = k - idx
            mask = (j >= 0) & (j < n)
            g[mask] += 2.0 * a[j[mask]]

        # Adam update
        pow_b1 *= beta1
        pow_b2 *= beta2
        m = beta1 * m + (1.0 - beta1) * g
        v = beta2 * v + (1.0 - beta2) * (g ** 2)

        m_hat = m / (1.0 - pow_b1)
        v_hat = v / (1.0 - pow_b2)

        step = lr * m_hat / (np.sqrt(v_hat) + eps)
        a_new = a - step
        a_new = project_to_simplex(a_new)

        cur_score = compute_score(a_new)
        if cur_score < best_score:
            best_score = cur_score
            best_a = a_new.copy()

        # occasional small random perturbation to avoid stagnation
        if t % 2000 == 0:
            a_new += np.random.randn(n) * (lr * 0.01 / np.sqrt(t))
            a_new = project_to_simplex(a_new)

        a = a_new

    return best_a, best_score


# ----------------------------------------------------------------------
# Simulated‑annealing hill‑climb on the true objective
# ----------------------------------------------------------------------
def annealing_hill_climb(
    a_start: np.ndarray,
    deadline: float,
    max_steps: int = 350_000,
    start_temp: float = 1e-1,
    end_temp: float = 1e-5,
) -> tuple[np.ndarray, float]:
    """
    Stochastic hill‑climb that accepts worsening moves with a temperature
    schedule.  Operates directly on the objective used by ``evaluate_sequence``.
    """
    a = a_start.copy()
    n = a.shape[0]
    best_a = a.copy()
    best_score = compute_score(a)

    # linear schedule in log‑space
    log_start = np.log(start_temp)
    log_end = np.log(end_temp)

    step_scale = 0.12  # typical fraction of mass moved

    for step in range(max_steps):
        if time.time() >= deadline:
            break

        # temperature for this step (exponential decay)
        tau = step / max_steps
        T = np.exp(log_start + tau * (log_end - log_start))

        cand = a.copy()
        i, j = np.random.randint(n, size=2)
        if i == j:
            continue

        # transfer a random fraction of mass from i to j
        delta = step_scale * cand[i] * np.random.rand()
        if delta <= 0.0:
            continue
        cand[i] -= delta
        cand[j] += delta

        # projection onto simplex
        cand = project_to_simplex(cand)

        cur_score = compute_score(cand)
        delta_score = cur_score - best_score

        if delta_score < 0 or np.random.rand() < np.exp(-delta_score / T):
            a = cand
            if cur_score < best_score:
                best_score = cur_score
                best_a = cand.copy()

        # adapt step_scale slightly
        if cur_score < best_score:
            step_scale = min(step_scale * 1.03, 0.5)
        else:
            step_scale *= 0.996

    return best_a.tolist(), best_score

    """
    Search for a non‑negative step function that minimises `evaluate_sequence`.

    The routine is time‑bounded (≈ budget_s seconds) and combines
    Dirichlet restarts, an LP‑based update (SciPy), Adam sub‑gradient
    descent and a simulated‑annealing hill‑climb.
    """
    np.random.seed(seed)
    start_time = time.time()
    # leave a small safety margin for the harness
    deadline = start_time + budget_s - 2.0

    best_seq = None
    best_val = float("inf")

    # --------------------------------------------------------------
    #   (optional) use a known fallback if it exists
    # --------------------------------------------------------------
    try:
        fallback = [float(x) for x in height_sequence_1]  # type: ignore
        fallback_val = evaluate_sequence(fallback)
        print(f"Fallback score = {fallback_val:.6f}")
        if fallback_val < best_val:
            best_val = fallback_val
            best_seq = fallback
    except Exception:
        pass

    # --------------------------------------------------------------
    # Phase 1 – Random Dirichlet restarts (fast)
    # --------------------------------------------------------------
    n = 2048
    print("\nPhase 1 – random Dirichlet restarts")
    dirichlet_alphas = [0.3, 0.7, 1.5, 5.0]  # mixture of sparse / smooth shapes
    while time.time() < deadline - 250.0:
        alpha = np.random.choice(dirichlet_alphas)
        a = np.random.dirichlet(np.full(n, alpha)).astype(float)  # sum = 1
        cur_score = compute_score(a)
        if cur_score < best_val:
            best_val = cur_score
            best_seq = a.tolist()
            print(f"  new best (Dirichlet α={alpha:.2f}) score = {best_val:.6f}")

    # --------------------------------------------------------------
    # Phase 2 – LP based alternating optimisation (SciPy)
    # --------------------------------------------------------------
    if linprog is not None and best_seq is not None and time.time() < deadline - 150.0:
        print("\nPhase 2 – LP‑based update")
        a = np.array(best_seq, dtype=float)
        a = project_to_simplex(a)  # ensure exact sum=1

        lp_iter = 0
        while time.time() < deadline - 80.0 and lp_iter < 12:
            conv_a = convolution_fft(a, a)
            M = float(np.max(conv_a))

            b_opt = solve_lp_for_b(a, M)
            if b_opt is None:
                print("  LP solve failed – aborting LP phase")
                break

            b_sum = b_opt.sum()
            if b_sum <= 0:
                print("  LP returned zero mass – aborting LP phase")
                break
            b_opt /= b_sum

            # small convex combination
            t_mix = 0.12
            a = (1.0 - t_mix) * a + t_mix * b_opt
            a = project_to_simplex(a)

            cur_score = compute_score(a)
            if cur_score < best_val:
                best_val = cur_score
                best_seq = a.tolist()
                print(f"  LP iter {lp_iter}: improved score = {best_val:.6f}")

            lp_iter += 1

    # --------------------------------------------------------------
    # Phase 3 – Adam sub‑gradient refinement
    # --------------------------------------------------------------
    if best_seq is not None and time.time() < deadline - 70.0:
        print("\nPhase 3 – Adam sub‑gradient refinement")
        a0 = np.array(best_seq, dtype=float)
        refined_a, refined_score = subgrad_adam(
            a0,
            deadline=deadline - 50.0,
            max_iter=400_000,
            lr=0.09,
        )
        if refined_score < best_val:
            best_val = refined_score
            best_seq = refined_a.tolist()
            print(f"  Adam refinement improved best to {best_val:.6f}")

    # --------------------------------------------------------------
    # Phase 4 – Simulated‑annealing hill‑climb
    # --------------------------------------------------------------
    if best_seq is not None and time.time() < deadline - 5.0:
        print("\nPhase 4 – final annealing hill‑climb")
        a_start = np.array(best_seq, dtype=float)
        polished_seq, polished_score = annealing_hill_climb(
            a_start,
            deadline=deadline,
            max_steps=300_000,
        )
        if polished_score < best_val:
            best_val = polished_score
            best_seq = polished_seq
            print(f"  Hill‑climb improved best to {best_val:.6f}")

    # --------------------------------------------------------------
    # Safeguard – if nothing was found, fall back to uniform vector
    # --------------------------------------------------------------
    if best_seq is None:
        n = 256
        best_seq = [1.0 / n] * n
        best_val = evaluate_sequence(best_seq)

    # Final official evaluation
    final_score = evaluate_sequence(best_seq)
    print(f"\nFinished search. Final reported score = {final_score:.6f}")

    return best_seq
```

This implementation replaces the unavailable ECOS solver with SciPy's HiGHS linear‑programming backend, adds an Adam‑accelerated sub‑gradient optimizer, and finishes with a simulated‑annealing hill‑climb.  All phases respect the 1000 s budget and should be able to push the score below the previous 1.512 benchmark, aiming for the target ≈ 1.503.
