def evaluate_sequence(sequence: list[float]) -> float:

"""
Improved search for a non‑negative step sequence that minimizes the official
``evaluate_sequence`` score.

The algorithm works in two phases.

1️⃣  Random exploration of several parametric families (beta, power‑law,
    mixtures, Gaussian bumps, a spectral “root‑uniform’’ construction).
    Only the best *top_k* candidates (default 4) are kept.

2️⃣  Heavy local refinement of each seed:
    – Adam optimisation on a soft‑max approximation of the max‑convolution.
    – Exact sub‑gradient descent on the true objective.
    – A short simulated‑annealing polishing phase.
    The best vector is polished once more after all seeds are processed.

All vectors are kept on the probability simplex (Σa=1, a_i≥0) because the
official objective is 2·n·max(conv(a,a)) after this normalisation.
The code respects the 1000 s time limit, uses only NumPy and the standard
library, and prints intermediate scores for monitoring.
"""

import time
from typing import List, Tuple

import numpy as np

# ----------------------------------------------------------------------
# Low‑level utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """
    Linear convolution via real FFT.
    Length = len(a)+len(b)-1.
    """
    m = len(a) + len(b) - 1
    n_fft = next_pow2(m)
    A = np.fft.rfft(a, n=n_fft)
    B = np.fft.rfft(b, n=n_fft)
    C = A * B
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


def compute_score(a: np.ndarray) -> float:
    """
    Fast exact score (identical to the official one when Σa = 1):
    2·n·max(conv(a,a)).
    """
    n = a.shape[0]
    conv = convolution_fft(a, a)
    return 2.0 * n * float(np.max(conv))


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection onto the simplex {x ≥ 0, Σx = 1}.
    Implements the algorithm of Duchi et al. (2008).
    """
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / v.size)

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho_arr = np.where(u - cssv / np.arange(1, v.size + 1) > 0)[0]
    if rho_arr.size == 0:
        theta = 0.0
    else:
        rho = rho_arr[-1]
        theta = cssv[rho] / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


# ----------------------------------------------------------------------
# Shape factories (all return a vector that already sums to 1)
# ----------------------------------------------------------------------


def arcsine_initial(n: int) -> np.ndarray:
    """Classic arcsine shape: a_i ∝ 1/√((i+½)*(n‑½‑i) + 0.25)."""
    i = np.arange(n, dtype=float)
    eps = 0.25
    vals = 1.0 / np.sqrt((i + 0.5) * (n - 0.5 - i) + eps)
    vals /= vals.sum()
    return vals


def generate_beta_shape(n: int, alpha: float, beta: float, eps: float) -> np.ndarray:
    """
    (i+ε)^{α‑1}·(n‑1‑i+ε)^{β‑1}, normalised.
    α,β > 0; eps > 0 to avoid division by zero.
    """
    i = np.arange(n, dtype=float)
    left = (i + eps) ** (alpha - 1.0)
    right = (n - 1 - i + eps) ** (beta - 1.0)
    base = left * right
    total = base.sum()
    if not np.isfinite(total) or total <= 0.0:
        return np.full(n, 1.0 / n)
    return base / total


def generate_power_shape(n: int, p: float, eps: float) -> np.ndarray:
    """
    Generalised arcsine: a_i ∝ ((i+½)*(n‑½‑i)+ε)^{‑p}, p>0, eps≥0.
    """
    i = np.arange(n, dtype=float)
    denom = ((i + 0.5) * (n - 0.5 - i) + eps) ** p
    vals = 1.0 / denom
    total = vals.sum()
    if not np.isfinite(total) or total <= 0.0:
        return np.full(n, 1.0 / n)
    return vals / total


def generate_triangular_shape(n: int, peak: int) -> np.ndarray:
    """Symmetric triangular bump with apex at ``peak``."""
    left = np.arange(peak + 1, dtype=float)
    right = np.arange(n - peak - 1, dtype=float)[::-1]
    seq = np.concatenate([left, right])
    seq /= seq.sum()
    return seq


def generate_mixture_shape(n: int, weight_arc: float) -> np.ndarray:
    """Mixture (weight_arc)·arcsine + (1‑weight_arc)·uniform."""
    arc = arcsine_initial(n)
    uni = np.full(n, 1.0 / n)
    mix = weight_arc * arc + (1.0 - weight_arc) * uni
    mix /= mix.sum()
    return mix


def generate_gaussian_shape(n: int, sigma_frac: float = 0.15) -> np.ndarray:
    """Discrete Gaussian centred in the interval."""
    i = np.arange(n, dtype=float)
    centre = (n - 1) / 2.0
    sigma = sigma_frac * n
    vals = np.exp(-0.5 * ((i - centre) / sigma) ** 2)
    vals /= vals.sum()
    return vals


def generate_spectral_root_shape(n: int) -> np.ndarray:
    """
    Heuristic “root‑uniform’’ construction:
    take the uniform target on length 2·n‑1,
    compute its FFT, take the element‑wise square‑root of the magnitude,
    invert the transform and keep the first n entries.
    Afterwards enforce non‑negativity and project onto the simplex.
    """
    L = 2 * n - 1
    target = np.full(L, 1.0 / L)   # uniform target convolution
    m = next_pow2(L)
    T = np.fft.rfft(target, n=m)
    # square‑root of the magnitude (phase set to zero)
    Spec = np.sqrt(np.abs(T))
    seq = np.fft.irfft(Spec, n=m)[:n]
    seq = np.maximum(seq, 0.0)
    s = seq.sum()
    if s <= 0.0:
        return np.full(n, 1.0 / n)
    return seq / s

# Optimisers
# ----------------------------------------------------------------------


def smooth_gradient(a: np.ndarray, lam: float) -> np.ndarray:
    """
    Gradient of the soft‑max surrogate of max(conv(a,a)).
    Returns ∇_a (2·n·softmax_max) before projection.
    """
    n = a.shape[0]
    conv = convolution_fft(a, a)               # length 2·n‑1
    max_c = np.max(conv)

    # stable soft‑max weights
    exp_term = np.exp(lam * (conv - max_c))
    w = exp_term / np.sum(exp_term)

    # weighted sub‑gradient: 2·∑_k w_k shift(a,k)
    w_rev = w[::-1]
    conv_rev = convolution_fft(a, w_rev)
    L = 2 * n - 1
    weighted_grad = 2.0 * conv_rev[L - 1 - np.arange(n)]

    return 2.0 * n * weighted_grad


def adam_optimize_smooth(initial_a: np.ndarray,
                         deadline: float,
                         max_iter: int = 200_000,
                         lr0: float = 0.12,
                         lam_start: float = 2.0,
                         lam_end: float = 200.0,
                         noise_every: int = 2000,
                         noise_scale: float = 2e-5) -> Tuple[np.ndarray, float]:
    """
    Adam optimiser on the smooth surrogate.
    Returns the best vector (projected onto the simplex) and its fast score.
    """
    a = project_to_simplex(initial_a.astype(float).copy())
    best_a = a.copy()
    best_sc = compute_score(a)

    beta1, beta2 = 0.9, 0.999
    eps = 1e-8
    m = np.zeros_like(a)
    v = np.zeros_like(a)

    for it in range(1, max_iter + 1):
        if time.time() >= deadline:
            break

        lam = lam_start + (lam_end - lam_start) * min(1.0, it / max_iter)

        grad = smooth_gradient(a, lam)

        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)

        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        step = lr0 * m_hat / (np.sqrt(v_hat) + eps)
        a_new = a - step

        if noise_every and (it % noise_every == 0):
            a_new += np.random.randn(a.size) * noise_scale

        a_new = project_to_simplex(a_new)

        cur_sc = compute_score(a_new)
        if cur_sc < best_sc:
            best_sc = cur_sc
            best_a = a_new.copy()

        a = a_new

        if it % 5000 == 0:
            print(f"        AdamSmooth it={it:,d} cur={cur_sc:.6f} best={best_sc:.6f}")

    return best_a, best_sc


def subgradient_descent_exact(initial_a: np.ndarray,
                              deadline: float,
                              max_iter: int = 100_000,
                              step0: float = 0.025) -> np.ndarray:
    """
    Exact sub‑gradient descent on f(a)=2·n·max(conv(a,a)).
    Step size η_t = step0 / sqrt(t).  Projection onto the simplex keeps Σa=1.
    """
    a = project_to_simplex(initial_a.copy())
    best_a = a.copy()
    best_sc = compute_score(a)
    n = a.shape[0]
    L = 2 * n - 1

    for it in range(1, max_iter + 1):
        if time.time() >= deadline:
            break

        conv = convolution_fft(a, a)
        max_val = np.max(conv)

        # mask of all (approximately) maximal entries
        tol = max_val * 1e-12 + 1e-14
        mask = (np.abs(conv - max_val) <= tol).astype(float)

        # Sub‑gradient for the max‑operator = weighted sum of shifts
        mask_rev = mask[::-1]
        conv_rev = convolution_fft(a, mask_rev)
        weighted_grad = 2.0 * conv_rev[L - 1 - np.arange(n)]

        eta = step0 / np.sqrt(it)
        a = a - eta * weighted_grad
        a = project_to_simplex(a)

        cur_sc = compute_score(a)
        if cur_sc < best_sc:
            best_sc = cur_sc
            best_a = a.copy()

        if it % 5000 == 0:
            print(f"        SubGrad it={it:,d} cur={cur_sc:.6f} best={best_sc:.6f}")

    return best_a


def simulated_annealing(a: np.ndarray,
                        rng: np.random.Generator,
                        deadline: float,
                        max_iters: int = 120_000,
                        init_temp: float = 0.015,
                        decay: float = 0.9999) -> np.ndarray:
    """
    Simple Metropolis‑style annealing.
    Moves a small fraction of mass from one coordinate to another.
    The fast surrogate (2·n·max) decides acceptance.
    """
    cur = a.copy()
    cur_sc = compute_score(cur)
    best = cur.copy()
    best_sc = cur_sc
    n = cur.size
    temp = init_temp
    it = 0

    while it < max_iters and time.time() < deadline:
        it += 1
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue

        delta = cur[i] * 0.2 * rng.random()
        if delta <= 1e-15 or cur[i] - delta < 0.0:
            continue

        cand = cur.copy()
        cand[i] -= delta
        cand[j] += delta

        cand_sc = compute_score(cand)

        if cand_sc < cur_sc or rng.random() < np.exp(-(cand_sc - cur_sc) / max(temp, 1e-12)):
            cur = cand
            cur_sc = cand_sc
            if cur_sc < best_sc:
                best = cur.copy()
                best_sc = cur_sc

        temp = max(temp * decay, 1e-8)

        if it % 5000 == 0:
            print(f"        SA it={it:,d} cur={cur_sc:.6f} best={best_sc:.6f}")

    return best

def propose_candidate(seed: int = 42,
                     budget_s: int = 1000,
                     **kwargs) -> List[float]:
    """
    Search for a non‑negative step sequence with a low official score.
    Uses a broad random sampling of parametric families followed by
    intensive local optimisation (Adam → sub‑gradient → SA) on the best
    few seeds.  The routine respects the 1000 s budget and returns a list
    ready for ``evaluate_sequence``.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5   # tiny safety margin

    best_seq: List[float] = []
    best_val: float = float("inf")

    # ------------------------------------------------------------------
    # 0️⃣  Optional fallback from a previous run
    # ------------------------------------------------------------------
    if "height_sequence_1" in globals():
        try:
            fallback = [float(x) for x in globals()["height_sequence_1"]]
            val = evaluate_sequence(fallback)
            print(f"Fallback (height_sequence_1) score = {val:.6f}")
            if val < best_val:
                best_val = val
                best_seq = fallback
        except Exception:
            pass

    # ------------------------------------------------------------------
    # 1️⃣  Wide random exploration – keep the best *top_k* seeds
    # ------------------------------------------------------------------
    n = 2048                         # length used in the reference run
    top_k = 4                        # a small deep‑optimisation pool
    best_pool: List[Tuple[float, np.ndarray]] = []   # (score, vector)

    # deterministic starters
    uniform = np.full(n, 1.0 / n)
    best_pool.append((compute_score(uniform), uniform))

    arc = arcsine_initial(n)
    best_pool.append((compute_score(arc), arc))

    gauss = generate_gaussian_shape(n, sigma_frac=0.12)
    best_pool.append((compute_score(gauss), gauss))

    try:
        spectral = generate_spectral_root_shape(n)
        best_pool.append((compute_score(spectral), spectral))
    except Exception:
        pass

    # Helper to keep only the *top_k* best entries
    def insert_candidate(score: float, vec: np.ndarray):
        if len(best_pool) < top_k:
            best_pool.append((score, vec))
            return
        # find current worst
        worst_idx = max(range(len(best_pool)), key=lambda i: best_pool[i][0])
        if score < best_pool[worst_idx][0]:
            best_pool[worst_idx] = (score, vec)

    max_samples = 80_000
    sample_cnt = 0
    while time.time() < deadline - 30.0 and sample_cnt < max_samples:
        sample_cnt += 1
        r = rng.random()
        if r < 0.30:                # beta family, favour strong U‑shapes
            alpha = rng.uniform(0.02, 0.8)
            beta = rng.uniform(0.02, 0.8)
            eps = rng.uniform(1e-6, 0.5)
            cand = generate_beta_shape(n, alpha, beta, eps)
        elif r < 0.55:              # power / generalized arcsine
            p = rng.uniform(0.1, 2.0)
            eps = rng.uniform(0.0, 1.0)
            cand = generate_power_shape(n, p, eps)
        elif r < 0.70:              # mixture arcsine / uniform
            w = rng.random()
            cand = generate_mixture_shape(n, w)
        elif r < 0.85:              # triangular bump
            peak = rng.integers(0, n)
            cand = generate_triangular_shape(n, peak)
        else:                       # Dirichlet‑like (many small pieces)
            conc = rng.uniform(0.1, 4.0)
            alphas = rng.gamma(shape=conc, scale=1.0, size=n) + 0.001
            cand = alphas / alphas.sum()

        sc = compute_score(cand)
        insert_candidate(sc, cand)

    # sort the pool (best first) and extract the seed vectors
    best_pool.sort(key=lambda x: x[0])
    seed_vectors = [vec for _, vec in best_pool]

    # ------------------------------------------------------------------
    # 2️⃣  Intensive refinement of each seed
    # ------------------------------------------------------------------
    for idx, seed_vec in enumerate(seed_vectors, 1):
        if time.time() >= deadline - 5.0:
            break
        print(f"\n--- Refining seed {idx}/{len(seed_vectors)} ---")

        # A – smooth Adam optimisation (high λ for tighter max approximation)
        remaining = deadline - time.time()
        max_iter_adam = int(min(300_000, remaining * 45))
        print(f"    AdamSmooth  max_iter={max_iter_adam}")
        refined, _ = adam_optimize_smooth(
            initial_a=seed_vec,
            deadline=deadline,
            max_iter=max_iter_adam,
            lr0=0.11,
            lam_start=2.0,
            lam_end=250.0,
            noise_every=2000,
            noise_scale=1e-5,
        )

        # B – exact sub‑gradient descent (fine‑grained step decay)
        if time.time() < deadline - 5.0:
            remaining = deadline - time.time()
            max_iter_sub = int(min(250_000, remaining * 40))
            print(f"    Sub‑gradient max_iter={max_iter_sub}")
            refined = subgradient_descent_exact(
                refined,
                deadline,
                max_iter=max_iter_sub,
                step0=0.018,
            )

        # C – simulated‑annealing polish
        if time.time() < deadline - 5.0:
            remaining = deadline - time.time()
            max_iter_sa = int(min(200_000, remaining * 30))
            print(f"    Simulated annealing max_iters={max_iter_sa}")
            refined = simulated_annealing(
                refined,
                rng,
                deadline,
                max_iters=max_iter_sa,
                init_temp=0.012,
                decay=0.99995,
            )

        # Evaluate with the official function
        try:
            val = evaluate_sequence(refined.tolist())
            print(f"    Official score = {val:.6f}")
            if val < best_val:
                best_val = val
                best_seq = refined.tolist()
                print(f"    *** New global best: {best_val:.6f} ***")
        except Exception:
            pass

    # ------------------------------------------------------------------
    # 3️⃣  Global polishing of the overall best (if time permits)
    # ------------------------------------------------------------------
    if best_seq and time.time() < deadline - 3.0:
        a = np.array(best_seq, dtype=float)
        remaining = deadline - time.time()
        sub_iters = int(min(300_000, remaining * 30))
        adam_iters = int(min(250_000, remaining * 30))

        print("\n--- Final global polishing ---")
        a = subgradient_descent_exact(a, deadline, max_iter=sub_iters, step0=0.015)
        a, _ = adam_optimize_smooth(
            a,
            deadline,
            max_iter=adam_iters,
            lr0=0.07,
            lam_start=5.0,
            lam_end=400.0,
            noise_every=2000,
            noise_scale=5e-6,
        )
        final_val = evaluate_sequence(a.tolist())
        print(f"Final polished official score = {final_val:.6f}")
        if final_val < best_val:
            best_val = final_val
            best_seq = a.tolist()
            print("*** Updated global best after final polish ***")

    # ------------------------------------------------------------------
    # 4️⃣  Safeguard – guarantee a non‑empty return value
    # ------------------------------------------------------------------
    if not best_seq:
        best_seq = np.full(n, 1.0 / n).tolist()
        best_val = evaluate_sequence(best_seq)

    print(f"\nSearch finished. Best reported score = {best_val:.6f}")
    return best_seq
