def adam_logit_optimize(initial_a, time_deadline, max_iter=200000, lr0=0.15, smooth_iters=0):
    # initial logits
    eps = 1e-12
    logits = np.log(np.maximum(initial_a, eps))
    # Adam moments
    m = np.zeros_like(logits)
    v = np.zeros_like(logits)
    beta1, beta2 = 0.9, 0.999
    eps_adam = 1e-8
    n = initial_a.shape[0]
    best_a = initial_a.copy()
    best_val = compute_score(initial_a)
    for it in range(1, max_iter+1):
        if time.time() >= time_deadline:
            break
        a = np.exp(logits)
        a = a / a.sum()
        if smooth_iters and it <= smooth_iters:
            grad_a, _ = smooth_gradient(a, lam=1 + (200-1)*(it/smooth_iters)**0.5) # maybe similar schedule
        else:
            grad_max = active_gradient(a)
            grad_a = 2.0 * n * grad_max   # gradient of objective O w.r.t. a
        # chain rule for logits
        grad_logit = a * (grad_a - np.dot(a, grad_a))
        # Adam updates
        m = beta1 * m + (1 - beta1) * grad_logit
        v = beta2 * v + (1 - beta2) * (grad_logit **2)
        m_hat = m / (1 - beta1**it)
        v_hat = v / (1 - beta2**it)
        step = lr0 * m_hat / (np.sqrt(v_hat) + eps_adam)
        logits -= step
        # evaluate get current a and compute score for best tracking
        cur_a = np.exp(logits)
        cur_a /= cur_a.sum()
        cur_sc = compute_score(cur_a)
        if cur_sc < best_val:
            best_val = cur_sc
            best_a = cur_a.copy()
        if it % 5000 == 0:
            print(f"    AdamLogit iter={it} cur={cur_sc:.6f} best={best_val:.6f}")
    return best_a, best_val

"""
Improved step‑function search.

The programme follows three stages:

1. **Exploratory sampling** – a few thousand beta‑family or simple cosine
   shapes are drawn for a chosen length `n`.  The best sample according to
   the fast (scaled) objective `2·n·max(conv(a,a))` is kept.

2. **Logit‑Adam optimisation** – the selected vector is re‑parameterised
   by soft‑max logits `x` (so `a = softmax(x)`).  Adam is run on the
   logits with two gradient regimes:
        • a smooth log‑sum‑exp surrogate (low temperature) for the first
          30 % of the iterations,
        • the exact sub‑gradient of the max‑convolution afterwards.
   Using logits guarantees non‑negativity and exact simplex satisfaction,
   and avoids the Euclidean projection used in the reference code.

3. **Global polishing** – the best vector found for any length is
   polished again with a longer Adam run and a short stochastic walk.
   The official `evaluate_sequence` is called only for final candidates,
   keeping the total running time well below the 1000 s budget.

All steps respect the time limit, keep the best official score and
fallback to a safe uniform vector if nothing else succeeds.
"""

import time
import numpy as np
from typing import List, Tuple

# ----------------------------------------------------------------------
# Basic utilities (FFT convolution, simplex, etc.)
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """
    Linear convolution via real FFT.
    Returns an array of length len(a)+len(b)-1.
    """
    m = a.size + b.size - 1
    n_fft = next_pow2(m)
    A = np.fft.rfft(a, n=n_fft)
    B = np.fft.rfft(b, n=n_fft)
    C = A * B
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


def max_convolution(a: np.ndarray) -> float:
    """Maximum entry of the self‑convolution of a."""
    return float(np.max(convolution_fft(a, a)))


def compute_fast_score(a: np.ndarray) -> float:
    """
    Fast surrogate 2·n·max(conv(a,a)).
    The vector is assumed to be normalised to sum 1,
    therefore this equals the official score.
    """
    n = a.shape[0]
    return 2.0 * n * max_convolution(a)


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection onto the probability simplex
    {x ≥ 0, Σx = 1}.  Implementation of Duchi et al. (2008).
    """
    n = v.shape[0]
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / n)

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho = np.where(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def softmax(x: np.ndarray) -> np.ndarray:
    """Numerically stable soft‑max."""
    m = np.max(x)
    e = np.exp(x - m)
    return e / e.sum()

    """
    Symmetric beta‑family density on {0,…,n‑1}:
        a_i ∝ (i+ε)^{α‑1}·(n‑1‑i+ε)^{β‑1}.
    Returned vector is normalised to Σa = 1.
    """
    i = np.arange(n, dtype=float)
    left = (i + eps) ** (alpha - 1.0)
    right = (n - 1 - i + eps) ** (beta - 1.0)
    base = left * right
    total = base.sum()
    if total <= 0.0 or not np.isfinite(total):
        return np.full(n, 1.0 / n)
    return base / total


def generate_beta_mixture(
    n: int,
    alpha1: float,
    beta1: float,
    eps1: float,
    alpha2: float,
    beta2: float,
    eps2: float,
    w: float,
) -> np.ndarray:
    """
    Convex combination of two beta‑family densities.
    The result is re‑normalised to Σ = 1.
    """
    a1 = generate_beta_shape(n, alpha1, beta1, eps1)
    a2 = generate_beta_shape(n, alpha2, beta2, eps2)
    mix = w * a1 + (1.0 - w) * a2
    mix_sum = mix.sum()
    if mix_sum == 0.0 or not np.isfinite(mix_sum):
        return np.full(n, 1.0 / n)
    return mix / mix_sum


def arcsine_initial(n: int) -> np.ndarray:
    """Classic discrete arcsine shape (symmetrised)."""
    i = np.arange(n, dtype=float)
    eps = 0.25
    vals = 1.0 / np.sqrt((i + 0.5) * (n - 0.5 - i) + eps)
    vals /= vals.sum()
    return vals


def cosine_window(n: int) -> np.ndarray:
    """
    Simple raised‑cosine (Hann) window, normalised to sum 1.
    """
    i = np.arange(n, dtype=float)
    vals = 0.5 - 0.5 * np.cos(2.0 * np.pi * i / (n - 1))
    vals /= vals.sum()
    return vals


# ----------------------------------------------------------------------
# Gradient utilities
# ----------------------------------------------------------------------


def smooth_gradient(a: np.ndarray, lam: float) -> Tuple[np.ndarray, float]:
    """
    Gradient of the log‑sum‑exp surrogate of max(conv(a,a)).
    Returns (gradient_of_objective, surrogate_max).
    """
    n = a.shape[0]
    conv = convolution_fft(a, a)          # length 2n‑1
    max_c = np.max(conv)

    # stable log‑sum‑exp
    exp_term = np.exp(lam * (conv - max_c))
    sum_exp = np.sum(exp_term)
    w = exp_term / sum_exp                # soft‑max on convolution entries
    B = max_c + np.log(sum_exp) / lam     # smooth approximation of the max

    # weighted sum of sub‑gradients:
    w_rev = w[::-1]
    conv_rev = convolution_fft(a, w_rev)      # length 3n‑2
    L = 2 * n - 1
    weighted_grad = 2.0 * conv_rev[L - 1 - np.arange(n)]

    # since we keep Σa = 1, the denominator is constant;
    # the objective is O = 2·n·B
    grad = 2.0 * n * weighted_grad
    return grad, B


def active_gradient(a: np.ndarray) -> np.ndarray:
    """
    Exact sub‑gradient of max(conv(a,a)) w.r.t. a.
    Handles a possibly non‑unique maximiser by averaging.
    """
    n = a.shape[0]
    conv = convolution_fft(a, a)
    max_val = float(np.max(conv))
    # active set (tiny tolerance to capture ties)
    tol = max_val - 1e-12
    active = (conv >= tol).astype(float)
    cnt = active.sum()
    if cnt == 0:
        active = np.ones_like(conv) / conv.size
        cnt = conv.size
    w = active / cnt                # uniform weight over active indices
    w_rev = w[::-1]
    L = 2 * n - 1
    conv_rev = convolution_fft(a, w_rev)      # length 3n‑2
    grad_max = 2.0 * conv_rev[L - 1 - np.arange(n)]   # ∂max/∂a
    # full objective gradient (Σa = 1) is 2·n·grad_max
    return 2.0 * n * grad_max


# ----------------------------------------------------------------------
# Adam optimiser on logits (soft‑max parametrisation)
# ----------------------------------------------------------------------


def adam_logit_optimize(
    initial_a: np.ndarray,
    deadline: float,
    max_iter: int = 500_000,
    lr0: float = 0.12,
    smooth_frac: float = 0.30,
    lam_start: float = 1.0,
    lam_end: float = 400.0,
    noise_every: int = 2000,
    noise_scale: float = 4e-5,
) -> Tuple[np.ndarray, float]:
    """
    Projected Adam on the probability simplex via logits.
    The first ``smooth_frac`` of the iterations use the smooth surrogate,
    the remainder use the true max‑convolution sub‑gradient.
    Returns the best vector (sum = 1) and its fast score.
    """
    # initialise logits safely
    eps = 1e-12
    logits = np.log(np.maximum(initial_a, eps))

    n = initial_a.shape[0]
    best_a = softmax(logits)
    best_sc = compute_fast_score(best_a)

    # Adam state
    m = np.zeros_like(logits)
    v = np.zeros_like(logits)
    beta1, beta2 = 0.9, 0.999
    eps_adam = 1e-8

    smooth_iters = int(max_iter * smooth_frac)

    for it in range(1, max_iter + 1):
        if time.time() >= deadline:
            break

        # current probability vector
        a = softmax(logits)

        # choose gradient regime
        if it <= smooth_iters:
            # temperature annealing (quadratic schedule)
            t = it / smooth_iters
            lam = lam_start + (lam_end - lam_start) * (t ** 0.5)
            grad_a, _ = smooth_gradient(a, lam)      # gradient of objective
        else:
            grad_a = active_gradient(a)               # exact sub‑gradient

        # chain rule for logits (soft‑max Jacobian)
        dot = np.dot(a, grad_a)
        grad_logit = a * (grad_a - dot)

        # Adam update on logits
        m = beta1 * m + (1.0 - beta1) * grad_logit
        v = beta2 * v + (1.0 - beta2) * (grad_logit ** 2)

        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        step = lr0 * m_hat / (np.sqrt(v_hat) + eps_adam)
        logits -= step

        # occasional tiny isotropic noise (helps escape flat plateaus)
        if noise_every and (it % noise_every == 0):
            logits += np.random.randn(n) * noise_scale

        # evaluate fast score for best‑track
        cur_sc = compute_fast_score(a)
        if cur_sc < best_sc:
            best_sc = cur_sc
            best_a = a.copy()

        if it % 5000 == 0:
            print(
                f"    AdamLogit iter={it:,d} cur={cur_sc:.6f} best={best_sc:.6f}"
            )

    return best_a, best_sc


# ----------------------------------------------------------------------
# Light stochastic walk (final polishing)
# ----------------------------------------------------------------------


def stochastic_walk(
    a0: np.ndarray,
    deadline: float,
    init_step: float = 0.012,
    max_iter: int = 300_000,
) -> Tuple[np.ndarray, float]:
    """
    Tiny Gaussian walk on the simplex (directly on probabilities).
    Accepts downhill moves; occasional uphill moves are allowed.
    Returns the best vector seen and its fast score.
    """
    a = a0.copy()
    best_a = a.copy()
    best_sc = compute_fast_score(a)
    n = a.shape[0]
    rng = np.random.default_rng()

    for it in range(1, max_iter + 1):
        if time.time() >= deadline:
            break
        step = init_step / np.sqrt(it)
        noise = rng.normal(scale=step, size=n)
        a_prop = a + noise
        a_prop = project_to_simplex(a_prop)

        cur_sc = compute_fast_score(a_prop)
        if cur_sc < best_sc or (rng.random() < 1e-4 and cur_sc < 10 * best_sc):
            a = a_prop
            if cur_sc < best_sc:
                best_sc = cur_sc
                best_a = a_prop.copy()

        if it % 50000 == 0:
            print(
                f"    Walk iter={it:,d} cur={cur_sc:.6f} best={best_sc:.6f}"
            )
    return best_a, best_sc


# ----------------------------------------------------------------------
# Main entry point required by the competition harness
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs) -> List[float]:
    """
    Search for a non‑negative step sequence with a low official score.
    The routine performs:
        * a quick uniform / arcsine baseline,
        * extensive random beta‑family sampling,
        * a long Adam optimisation on logits,
        * a final stochastic polishing.
    The best sequence found (according to `evaluate_sequence`) is returned.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 2.0  # safety margin

    best_seq: List[float] = []
    best_val: float = float("inf")

    # ------------------------------------------------------------------
    # 0. optional fallback from the previous run
    # ------------------------------------------------------------------
    try:
        fallback = [float(x) for x in height_sequence_1]  # type: ignore
        val = evaluate_sequence(fallback)
        print(f"Fallback (height_sequence_1) official score = {val:.6f}")
        if val < best_val:
            best_val = val
            best_seq = fallback
    except Exception:
        pass

    # ------------------------------------------------------------------
    # 1.  Choose a few lengths (larger lengths give more expressive power)
    # ------------------------------------------------------------------
    lengths = [2048, 4096, 8192]

    for n in lengths:
        if time.time() >= deadline - 30.0:
            break
        print(f"\n=== Optimising length n = {n} ===")

        # ------------------------------------------------------------------
        # 1a. deterministic starters
        # ------------------------------------------------------------------
        starters = [
            np.full(n, 1.0 / n),      # uniform
            arcsine_initial(n),       # classic arcsine
            cosine_window(n),         # Hann window
        ]

        # ------------------------------------------------------------------
        # 1b. random beta‑family shapes (including mixtures)
        # ------------------------------------------------------------------
        n_samples = 3000
        for _ in range(n_samples):
            if time.time() >= deadline - 30.0:
                break
            if rng.random() < 0.3:       # mixture
                w = rng.random()
                alpha1 = rng.uniform(0.02, 3.0)
                beta1 = rng.uniform(0.02, 3.0)
                eps1 = rng.uniform(1e-8, max(0.01, 0.02 * n))
                alpha2 = rng.uniform(0.02, 3.0)
                beta2 = rng.uniform(0.02, 3.0)
                eps2 = rng.uniform(1e-8, max(0.01, 0.02 * n))
                cand = generate_beta_mixture(
                    n, alpha1, beta1, eps1, alpha2, beta2, eps2, w
                )
            else:                        # single beta
                alpha = rng.uniform(0.02, 3.0)
                beta = rng.uniform(0.02, 3.0)
                eps = rng.uniform(1e-8, max(0.01, 0.02 * n))
                cand = generate_beta_shape(n, alpha, beta, eps)
            starters.append(cand)

        # ------------------------------------------------------------------
        # 1c. pick the best raw sample (fast surrogate)
        # ------------------------------------------------------------------
        local_best = None
        local_best_sc = float("inf")
        for cand in starters:
            sc = compute_fast_score(cand)
            if sc < local_best_sc:
                local_best_sc = sc
                local_best = cand

        # ------------------------------------------------------------------
        # 2.  Adam optimisation on logits (logit‑Adam)
        # ------------------------------------------------------------------
        if local_best is None:
            continue

        remaining = deadline - time.time()
        # allocate a generous number of iterations; the loop itself will stop
        # as soon as the time budget is exhausted.
        max_iter_adam = int(600_000)   # upper bound; actual count limited by time

        print(f"  Adam optimisation (max_iter≈{max_iter_adam}) …")
        polished_a, _ = adam_logit_optimize(
            initial_a=local_best,
            deadline=deadline,
            max_iter=max_iter_adam,
            lr0=0.13,
            smooth_frac=0.30,
            lam_start=1.0,
            lam_end=500.0,
            noise_every=1500,
            noise_scale=5e-5,
        )

        # ------------------------------------------------------------------
        # 3.  Final random walk to polish
        # ------------------------------------------------------------------
        if time.time() < deadline - 5.0:
            max_iter_walk = int(300_000)
            polished_a, _ = stochastic_walk(
                polished_a,
                deadline=deadline,
                init_step=0.015,
                max_iter=max_iter_walk,
            )

        # ------------------------------------------------------------------
        # 4.  Official evaluation for this length
        # ------------------------------------------------------------------
        val = evaluate_sequence(polished_a.tolist())
        print(f"  → official score for length {n} = {val:.6f}")
        if val < best_val:
            best_val = val
            best_seq = polished_a.tolist()
            print(f"  *** New global best (len={n}) = {best_val:.6f} ***")

    # ------------------------------------------------------------------
    # 5.  Global polishing of the overall best vector (if any)
    # ------------------------------------------------------------------
    if best_seq and time.time() < deadline - 5.0:
        print("\n=== Global polishing of the current best sequence ===")
        a = np.array(best_seq, dtype=float)

        # longer Adam run with a higher final temperature
        remaining = deadline - time.time()
        max_iter_adam = int(800_000)
        a, _ = adam_logit_optimize(
            initial_a=a,
            deadline=deadline,
            max_iter=max_iter_adam,
            lr0=0.10,
            smooth_frac=0.25,
            lam_start=1.0,
            lam_end=800.0,
            noise_every=2000,
            noise_scale=4e-5,
        )

        # short stochastic walk
        if time.time() < deadline - 3.0:
            a, _ = stochastic_walk(
                a,
                deadline=deadline,
                init_step=0.012,
                max_iter=250_000,
            )

        final_val = evaluate_sequence(a.tolist())
        print(f"Final polished official score = {final_val:.6f}")
        if final_val < best_val:
            best_val = final_val
            best_seq = a.tolist()
            print("*** Updated global best after final polish ***")

    # ------------------------------------------------------------------
    # 6.  Guard – guarantee a non‑empty return
    # ------------------------------------------------------------------
    if not best_seq:
        best_seq = [1.0] * 256
        best_val = evaluate_sequence(best_seq)
        print("Fallback to uniform 256‑vector.", best_val)

    print(f"\nSearch finished. Best reported official score = {best_val:.6f}")
    return best_seq
