def propose_candidate(...):
    import time, numpy as np
    start = time.time()
    deadline = start + budget_s - 5  # keep 5 seconds margin.

    best_seq, best_val = None, float('inf')
    rng = np.random.default_rng(seed)
    lengths = [2048, 4096, 8192]   # maybe include previously found length as fallback.

    for n in lengths:
        if time.time() >= deadline:
            break
        # Sample families:
        seeds = []
        # uniform
        seeds.append(np.full(n, 1.0/n))
        # arcsine
        seeds.append(arcsine_initial(n))
        # arcsine general p in {0.3,0.5,0.7,1.0}, eps in small values
        for p in [0.3,0.5,0.7,1.0]:
            for eps in [0.001, 0.01, 0.1, 0.5]:
                seeds.append(generate_powerlaw(n, p, eps))  # note variance shape
        # random dirichlet (vartious conc)
        for conc in [0.2,0.5,1.0,2.0]:
            seeds.append(random_dirichlet(n, conc, rng))
        # random mixture arcsine + uniform
        for _ in range(3):
            w = rng.random()
            mix = w * arcsine_initial(n) + (1-w) * np.full(n, 1.0/n)
            seeds.append(mix / mix.sum())
        # perturbed seeds
        for seed_vec in seeds[:5]:
            seeds.append(perturb(seed_vec, rng, scale=0.05))

        # Evaluate fast scores
        scored = [(compute_score(v), v) for v in seeds]
        scored.sort(key=lambda x:x[0])
        top = [v for _,v in scored[:3]]

        # For each top, run gradient refinement.
        for idx, a0 in enumerate(top):
            if time.time() >= deadline:
                break
            # Simulated annealing optional
            # Instead, we start with softmax GD.
            betas = [10, 30, 60, 120, 240]  # gradually increase
            # allocate iterations per beta based on remaining time
            remaining = deadline - time.time()
            total_iter = int(min(4000, remaining * 200))  # 200 it/s roughly.
            # divide equally among betas
            it_per_beta = max(100, total_iter // len(betas))
            a_ref, _ = softmax_gradient_descent(a0, deadline, betas, it_per_beta, eta0=0.5)

            # final subgradient polish
            remaining2 = deadline - time.time()
            if remaining2 > 2:
                max_iter = int(min(150_000, remaining2 * 300))
                a_ref, _ = optimise_subgradient(a_ref, deadline, eta0=0.4,
                                                max_iter=max_iter)
            # evaluate via official function
            seq = a_ref.tolist()
            val = evaluate_sequence(seq)
            print(f"candidate n={n} idx={idx} val={val}")
            if val < best_val:
                best_val, best_seq = val, seq
                print(f"New global best: {best_val}")

    # fallback
    if best_seq is None:
        best_seq = [1.0] * 256
        best_val = evaluate_sequence(best_seq)

    print(f"Return best_val={best_val}")
    return best_seq

"""
Hybrid search for a low‑max‑convolution step function.

The routine explores several lengths (powers of two up to 8192) and a
large family of parametric shapes (arcsine, power‑law, beta‑like,
Dirichlet).  For each promising seed we run a two‑stage optimisation:

1.  A smooth‑max gradient descent.  The non‑smooth objective
    2·n·max(a∗a) is approximated by a log‑sum‑exp surrogate with an
    increasing β‑schedule.  An exponentiated‑gradient (mirror‑descent)
    update keeps the vector non‑negative and on the probability simplex.

2.  A final polish with an exponentiated sub‑gradient descent on the
    true max‑convolution.

Both stages respect the wall‑clock budget (default 1000 s).  The best
feasible sequence found is returned; if nothing is discovered a simple
fallback is used.
"""

import time
import math
from typing import List, Tuple

import numpy as np

# ----------------------------------------------------------------------
# Low‑level utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full linear convolution using a real FFT (zero‑padded to a power of two)."""
    m = len(a) + len(b) - 1
    n_fft = next_pow2(m)
    A = np.fft.rfft(a, n=n_fft)
    B = np.fft.rfft(b, n=n_fft)
    C = A * B
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


def compute_score(a: np.ndarray) -> float:
    """
    Fast (scale‑invariant) version of the official objective:
        2·n·max(a∗a) / (sum(a)²)

    The vector a is assumed non‑negative; the function works for any shape.
    """
    n = len(a)
    conv = convolution_fft(a, a)
    max_val = float(conv.max())
    s = float(a.sum())
    if s < 1e-12:
        return float("inf")
    return 2.0 * n * max_val / (s * s)


# ----------------------------------------------------------------------
# Parametric families for seed generation
# ----------------------------------------------------------------------


def arcsine_initial(n: int) -> np.ndarray:
    """Discrete arcsine density (normalised to sum = 1)."""
    i = np.arange(n, dtype=float)
    eps = 0.25
    vals = 1.0 / np.sqrt((i + 0.5) * (n - 0.5 - i) + eps)
    return vals / vals.sum()


def generate_powerlaw(n: int, p: float, eps: float) -> np.ndarray:
    """Symmetric power‑law a_i ∝ ((i+ε)(n‑1‑i+ε))⁻ᵖ."""
    i = np.arange(n, dtype=float)
    base = ((i + eps) * (n - 1 - i + eps)) ** (-p)
    total = base.sum()
    if total == 0.0:
        return np.full(n, 1.0 / n)
    return base / total


def generate_beta_like(n: int, alpha: float, beta: float) -> np.ndarray:
    """Beta‑like shape a_i ∝ (i+½)^{α‑1} (n‑1‑i+½)^{β‑1}."""
    i = np.arange(n, dtype=float)
    base = (i + 0.5) ** (alpha - 1) * (n - 1 - i + 0.5) ** (beta - 1)
    total = base.sum()
    if total == 0.0:
        return np.full(n, 1.0 / n)
    return base / total


def random_dirichlet(n: int, concentration: float, rng: np.random.Generator) -> np.ndarray:
    """Sample a Dirichlet‑like vector and normalise."""
    if concentration <= 0:
        samp = rng.exponential(1.0, size=n) ** (1.0 / max(0.1, -concentration))
    else:
        samp = rng.gamma(concentration, 1.0, size=n)
    samp += 1e-12
    return samp / samp.sum()


def perturb(a: np.ndarray, rng: np.random.Generator, scale: float = 0.02) -> np.ndarray:
    """Add a tiny Gaussian perturbation and renormalise onto the simplex."""
    b = a + rng.normal(0.0, scale, size=a.shape)
    b = np.maximum(b, 0.0)
    s = b.sum()
    if s == 0.0:
        return np.full_like(a, 1.0 / len(a))
    return b / s


# ----------------------------------------------------------------------
# Smooth‑max (log‑sum‑exp) gradient descent
# ----------------------------------------------------------------------


def softmax_gradient_descent(initial_a: np.ndarray,
                             deadline: float,
                             betas: List[float],
                             iter_per_beta: int = 500,
                             eta0: float = 0.5) -> Tuple[np.ndarray, float]:
    """
    Minimise the smooth surrogate
        Φ_β(a) = (1/β)·log Σ_k exp(β·(a∗a)_k)

    using exponentiated‑gradient updates and projection onto the simplex.
    """
    a = np.maximum(initial_a, 1e-30)
    a = a / a.sum()
    best_a = a.copy()
    best_sc = compute_score(a)

    for beta in betas:
        for it in range(1, iter_per_beta + 1):
            if time.time() >= deadline:
                return best_a, best_sc

            # (a∗a)
            conv = convolution_fft(a, a)               # length 2n‑1

            # soft‑max weights
            max_c = conv.max()
            exp_vals = np.exp(beta * (conv - max_c))
            p = exp_vals / exp_vals.sum()               # Σ p = 1

            # cross‑correlation p ⋆ a (analytic gradient of Φ_β)
            N = next_pow2(len(p) + len(a) - 1)
            P = np.fft.fft(p, N)
            A = np.fft.fft(a, N)
            cross = np.fft.ifft(P * np.conj(A)).real[: len(a)]
            cross = np.maximum(cross, 0.0)

            # gradient of the surrogate (up to a positive factor)
            grad = 4.0 * len(a) * cross

            # exponentiated‑gradient step
            eta = eta0 / math.sqrt(it)
            a = a * np.exp(-eta * grad)

            # simplex projection (simple renorm works for exponentiated updates)
            s = a.sum()
            if s == 0.0:
                a = np.full_like(a, 1.0 / len(a))
            else:
                a = a / s

            cur_sc = compute_score(a)
            if cur_sc < best_sc:
                best_sc = cur_sc
                best_a = a.copy()

    return best_a, best_sc


# ----------------------------------------------------------------------
# Sub‑gradient descent on the true max‑convolution (final polishing)
# ----------------------------------------------------------------------


def _max_conv_gradient(a: np.ndarray, conv: np.ndarray) -> np.ndarray:
    """
    Sub‑gradient of max_k (a∗a)_k.
    For each active offset k we add the contribution 2·a_{k‑j}.
    When several offsets are active we average the result.
    """
    n = len(a)
    max_val = conv.max()
    mask = np.isclose(conv, max_val, atol=1e-15, rtol=0.0)
    active = np.where(mask)[0]
    grad = np.zeros_like(a)

    for k in active:
        lo = max(0, k - (n - 1))
        hi = min(k, n - 1)
        idx = np.arange(lo, hi + 1, dtype=int)
        grad[idx] += 2.0 * a[k - idx]

    if len(active) > 1:
        grad /= len(active)
    return grad


def optimise_subgradient(initial_a: np.ndarray,
                         deadline: float,
                         *,
                         eta0: float = 0.4,
                         max_iter: int = 200_000) -> Tuple[np.ndarray, float]:
    """
    Exponentiated sub‑gradient descent on Φ(a)=max_k (a∗a)_k.
    Returns the best vector seen and its fast score.
    """
    a = np.maximum(initial_a, 1e-30)
    a = a / a.sum()
    best_a = a.copy()
    best_sc = compute_score(a)
    n = len(a)

    for it in range(1, max_iter + 1):
        if time.time() >= deadline:
            break

        conv = convolution_fft(a, a)
        grad = _max_conv_gradient(a, conv)

        eta = eta0 / math.sqrt(it)
        a = a * np.exp(-eta * grad)

        s = a.sum()
        a = a / s if s != 0.0 else np.full_like(a, 1.0 / n)

        cur_sc = compute_score(a)
        if cur_sc < best_sc:
            best_sc = cur_sc
            best_a = a.copy()

        # occasional progress print (optional, harmless)
        if it % 5000 == 0:
            print(f"    subgrad iter={it:,d} cur={cur_sc:.6f} best={best_sc:.6f}")

    return best_a, best_sc
