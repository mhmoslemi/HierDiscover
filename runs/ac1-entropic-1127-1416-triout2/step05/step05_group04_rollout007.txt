"""
Propose a low‑autocorrelation step sequence.

The algorithm consists of four phases:

1. **Broad random exploration** of several parametric families
   (Beta, power‑/generalised‑arcsine, mixtures, triangular bumps,
   Dirichlet samples).  The best `top_k` candidates are kept in a
   small heap.

2. **Deterministic fine‑grid search** for the most promising families
   (power‑shape and beta‑shape) – this often discovers a seed that is
   already better than the best from the random stage.

3. **Heavy local refinement** of every seed:
      – Adam optimisation on a soft‑max surrogate of the max‑convolution,
        with a slow increase of the smoothing parameter `lam`.
      – Exact sub‑gradient descent on the true objective.
      – A short simulated‑annealing polishing phase.
   The budget is split dynamically according to the remaining time.

4. **Global polishing** of the overall best vector (if time permits).

All intermediate vectors are projected onto the probability simplex
so that the official score `2·n·max(conv)/sum²` reduces to the surrogate
`2·n·max(conv)`.  The routine respects the 1000 s time limit and always
returns a non‑empty list of non‑negative heights ready for
`evaluate_sequence`.
"""

import time
import numpy as np
from typing import List, Tuple

# ----------------------------------------------------------------------
# Low‑level utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """
    Linear convolution via real FFT.
    Length = len(a)+len(b)-1.
    """
    m = len(a) + len(b) - 1
    n_fft = next_pow2(m)
    A = np.fft.rfft(a, n=n_fft)
    B = np.fft.rfft(b, n=n_fft)
    C = A * B
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


def compute_score(a: np.ndarray) -> float:
    """
    Fast surrogate for the official score.
    The vector is assumed to belong to the simplex (Σa=1).  The returned
    value equals the official evaluation because the denominator disappears.
    """
    n = len(a)
    conv = convolution_fft(a, a)
    return 2.0 * n * float(np.max(conv))


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection onto the probability simplex {x ≥ 0, Σx = 1}.
    Implementation of Duchi et al. (2008).
    """
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / v.size)

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho_arr = np.where(u - cssv / np.arange(1, v.size + 1) > 0)[0]
    if len(rho_arr) == 0:
        theta = 0.0
    else:
        rho = rho_arr[-1]
        theta = cssv[rho] / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


# ----------------------------------------------------------------------
# Shape factories (all return a vector that already sums to 1)
# ----------------------------------------------------------------------


def arcsine_initial(n: int) -> np.ndarray:
    """Classic arcsine shape: a_i ∝ 1/√((i+½)*(n‑½‑i))."""
    i = np.arange(n, dtype=float)
    eps = 0.25
    vals = 1.0 / np.sqrt((i + 0.5) * (n - 0.5 - i) + eps)
    vals /= vals.sum()
    return vals


def generate_beta_shape(n: int, alpha: float, beta: float, eps: float) -> np.ndarray:
    """
    Beta family on {0,…,n‑1}:
    a_i ∝ (i+ε)^{α‑1}·(n‑1‑i+ε)^{β‑1}.
    The tiny epsilon avoids division‑by‑zero when α or β < 1.
    """
    i = np.arange(n, dtype=float)
    left = (i + eps) ** (alpha - 1.0)
    right = (n - 1 - i + eps) ** (beta - 1.0)
    base = left * right
    total = base.sum()
    if not np.isfinite(total) or total <= 0.0:
        return np.full(n, 1.0 / n)
    return base / total


def generate_power_shape(n: int, p: float, eps: float) -> np.ndarray:
    """
    Generalised arcsine/power shape:
    a_i ∝ ((i+½)*(n‑½‑i)+ε)^{‑p},   p>0, eps≥0.
    """
    i = np.arange(n, dtype=float)
    denom = ((i + 0.5) * (n - 0.5 - i) + eps) ** p
    vals = 1.0 / denom
    total = vals.sum()
    if not np.isfinite(total) or total <= 0.0:
        return np.full(n, 1.0 / n)
    return vals / total


def generate_triangular_shape(n: int, peak: int) -> np.ndarray:
    """Symmetric triangular bump with its apex at ``peak``."""
    left = np.arange(peak + 1, dtype=float)
    right = np.arange(n - peak - 1, dtype=float)[::-1]
    seq = np.concatenate([left, right])
    seq /= seq.sum()
    return seq


def generate_mixture_shape(n: int, weight_arc: float) -> np.ndarray:
    """Mixture (weight_arc)·arcsine + (1‑weight_arc)·uniform."""
    arc = arcsine_initial(n)
    uni = np.full(n, 1.0 / n)
    mix = weight_arc * arc + (1.0 - weight_arc) * uni
    mix /= mix.sum()
    return mix


def generate_dirichlet_shape(n: int, rng: np.random.Generator) -> np.ndarray:
    """Dirichlet sample with random concentration (all >0)."""
    alphas = rng.exponential(scale=1.0, size=n) + 0.01
    vec = rng.gamma(alphas, 1.0)
    vec /= vec.sum()
    return vec

# Optimisers
# ----------------------------------------------------------------------


def smooth_gradient(a: np.ndarray, lam: float) -> np.ndarray:
    """
    Gradient of the soft‑max approximation of max(conv(a,a)).
    Returns ∇_a (2·n·softmax_max) where the sum constraint is later
    enforced by projecting onto the simplex.
    """
    n = len(a)
    conv = convolution_fft(a, a)               # length 2·n‑1
    max_c = np.max(conv)

    # Numerically stable soft‑max (weights sum to 1)
    exp_term = np.exp(lam * (conv - max_c))
    w = exp_term / np.sum(exp_term)

    # Weighted sub‑gradient.
    # Using the identity   (∂/∂a) Σ_k w_k (a * a)_k = 2·(a ⋆ w_rev)
    w_rev = w[::-1]
    conv_rev = convolution_fft(a, w_rev)
    L = 2 * n - 1
    weighted_grad = 2.0 * conv_rev[L - 1 - np.arange(n)]

    return 2.0 * n * weighted_grad


def adam_optimize_smooth(initial_a: np.ndarray,
                         deadline: float,
                         max_iter: int = 200_000,
                         lr0: float = 0.12,
                         lam_start: float = 2.0,
                         lam_end: float = 200.0,
                         noise_every: int = 3000,
                         noise_scale: float = 5e-5) -> Tuple[np.ndarray, float]:
    """
    Adam optimiser on the smooth surrogate.  Returns the best vector
    (projected on the simplex) together with its surrogate score.
    """
    a = project_to_simplex(initial_a.astype(float).copy())
    best_a = a.copy()
    best_sc = compute_score(a)

    beta1, beta2 = 0.9, 0.999
    eps = 1e-8
    m = np.zeros_like(a)
    v = np.zeros_like(a)

    for it in range(1, max_iter + 1):
        if time.time() >= deadline:
            break

        # Linear schedule for lam (soft‑max sharpness)
        lam = lam_start + (lam_end - lam_start) * min(1.0, it / max_iter)

        grad = smooth_gradient(a, lam)

        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)

        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        step = lr0 * m_hat / (np.sqrt(v_hat) + eps)
        a_new = a - step

        if noise_every and (it % noise_every == 0):
            a_new += np.random.randn(a.size) * noise_scale

        a_new = project_to_simplex(a_new)

        cur_sc = compute_score(a_new)
        if cur_sc < best_sc:
            best_sc = cur_sc
            best_a = a_new.copy()

        a = a_new

        if it % 5000 == 0:
            print(f"    AdamSmooth it={it:,d} cur={cur_sc:.6f} best={best_sc:.6f}")

    return best_a, best_sc


def subgradient_descent_exact(initial_a: np.ndarray,
                              deadline: float,
                              max_iter: int = 200_000,
                              step0: float = 0.04) -> np.ndarray:
    """
    Sub‑gradient descent on the exact objective f(a)=2·n·max(conv(a,a)).
    Step size η_t = step0 / sqrt(t).  Projection on the simplex preserves Σa=1.
    """
    a = project_to_simplex(initial_a.copy())
    best_a = a.copy()
    best_sc = compute_score(a)
    n = len(a)
    L = 2 * n - 1

    for it in range(1, max_iter + 1):
        if time.time() >= deadline:
            break

        conv = convolution_fft(a, a)
        max_val = np.max(conv)

        # identify all positions attaining (approximately) the maximum
        tol = max_val * 1e-12 + 1e-14
        mask = (np.abs(conv - max_val) <= tol).astype(float)

        mask_rev = mask[::-1]
        conv_rev = convolution_fft(a, mask_rev)
        weighted_grad = 2.0 * conv_rev[L - 1 - np.arange(n)]

        eta = step0 / np.sqrt(it)
        a = a - eta * weighted_grad
        a = project_to_simplex(a)

        cur_sc = compute_score(a)
        if cur_sc < best_sc:
            best_sc = cur_sc
            best_a = a.copy()

        if it % 5000 == 0:
            print(f"    SubGrad it={it:,d} cur={cur_sc:.6f} best={best_sc:.6f}")

    return best_a


def simulated_annealing(a: np.ndarray,
                        rng: np.random.Generator,
                        deadline: float,
                        max_iters: int = 150_000,
                        init_temp: float = 0.018,
                        decay: float = 0.99993) -> np.ndarray:
    """
    Simple Metropolis‑style local search using the fast surrogate.
    Small fractions of mass are moved between random coordinates.
    """
    cur = a.copy()
    cur_sc = compute_score(cur)
    best = cur.copy()
    best_sc = cur_sc
    n = len(cur)
    temp = init_temp
    it = 0

    while it < max_iters and time.time() < deadline:
        it += 1
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue

        # move up to 20 % of the mass from i to j
        delta = cur[i] * 0.2 * rng.random()
        if delta <= 1e-15 or cur[i] - delta < 0:
            continue

        cand = cur.copy()
        cand[i] -= delta
        cand[j] += delta

        cand_sc = compute_score(cand)

        # Accept if better or with Metropolis probability
        if cand_sc < cur_sc or rng.random() < np.exp(-(cand_sc - cur_sc) / max(temp, 1e-12)):
            cur = cand
            cur_sc = cand_sc
            if cur_sc < best_sc:
                best = cur.copy()
                best_sc = cur_sc

        temp = max(temp * decay, 1e-8)

        if it % 5000 == 0:
            print(f"    SA it={it:,d} cur={cur_sc:.6f} best={best_sc:.6f}")

    return best


def local_perturbation(a: np.ndarray,
                       rng: np.random.Generator,
                       sigma: float = 1e-3) -> np.ndarray:
    """
    Add a tiny Gaussian perturbation and project back to the simplex.
    """
    pert = a + sigma * rng.standard_normal(a.size)
    return project_to_simplex(pert)

def propose_candidate(seed: int = 12345,
                     budget_s: int = 1000,
                     **kwargs) -> List[float]:
    """
    Search for a non‑negative step sequence with a low ``evaluate_sequence``
    value.  The routine mixes wide random sampling, deterministic grid
    refinement and aggressive local optimisation (Adam → sub‑gradient → SA).
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 1.0          # safety margin

    # ------------------------------------------------------------------
    # 0.  Global best bookkeeping
    # ------------------------------------------------------------------
    best_seq: List[float] = []
    best_val: float = float("inf")

    # Fallback in case a previous run produced a useful candidate.
    if "height_sequence_1" in globals():
        try:
            fallback = [float(x) for x in globals()["height_sequence_1"]]
            val = evaluate_sequence(fallback)
            print(f"Fallback (height_sequence_1) score = {val:.6f}")
            if val < best_val:
                best_val = val
                best_seq = fallback
        except Exception:
            pass

    # ------------------------------------------------------------------
    # 1.  Random exploration (keep a small top‑K heap)
    # ------------------------------------------------------------------
    n = 2048
    top_k = 12
    best_pool: List[Tuple[float, np.ndarray]] = []   # (score, vector)

    def maybe_insert(score: float, vec: np.ndarray):
        """Insert (score,vec) into best_pool keeping its size ≤ top_k."""
        if len(best_pool) < top_k:
            best_pool.append((score, vec))
        else:
            worst_idx = max(range(len(best_pool)),
                            key=lambda i: best_pool[i][0])
            if score < best_pool[worst_idx][0]:
                best_pool[worst_idx] = (score, vec)

    # deterministic starters
    uniform = np.full(n, 1.0 / n)
    maybe_insert(compute_score(uniform), uniform)

    arc = arcsine_initial(n)
    maybe_insert(compute_score(arc), arc)

    # random sampling loop
    max_candidates = 300_000
    cnt = 0
    while time.time() < deadline - 120.0 and cnt < max_candidates:
        cnt += 1
        r = rng.random()
        if r < 0.40:               # Beta family – sample on log‑scale
            alpha = 10 ** rng.uniform(-3, 1.5)   # 0.001 … 31.6
            beta = 10 ** rng.uniform(-3, 1.5)
            eps = rng.uniform(1e-7, 2.0)
            cand = generate_beta_shape(n, alpha, beta, eps)
        elif r < 0.80:             # Power / generalized arcsine
            p = rng.uniform(0.1, 8.0)
            eps = rng.uniform(0.0, 3.0)
            cand = generate_power_shape(n, p, eps)
        elif r < 0.90:             # Mixture arcsine / uniform
            w = rng.random()
            cand = generate_mixture_shape(n, w)
        elif r < 0.95:             # Triangular bump
            peak = rng.integers(0, n)
            cand = generate_triangular_shape(n, peak)
        else:                      # Dirichletish sample
            cand = generate_dirichlet_shape(n, rng)

        sc = compute_score(cand)
        maybe_insert(sc, cand)

    # ------------------------------------------------------------------
    # 2.  Deterministic grid search (tiny but often decisive)
    # ------------------------------------------------------------------
    # Power shape grid
    p_grid = np.linspace(0.2, 3.0, 30)          # 30 values
    eps_grid = [0.0, 0.05, 0.2, 0.5, 1.0, 2.0]  # 6 values
    for p in p_grid:
        for eps in eps_grid:
            cand = generate_power_shape(n, float(p), float(eps))
            sc = compute_score(cand)
            maybe_insert(sc, cand)

    # Beta shape grid (log‑uniform sampling on a small lattice)
    alpha_grid = [0.05, 0.2, 0.5, 1.0, 2.0, 5.0]
    beta_grid = [0.05, 0.2, 0.5, 1.0, 2.0, 5.0]
    for a_val in alpha_grid:
        for b_val in beta_grid:
            cand = generate_beta_shape(n, float(a_val), float(b_val), eps=1e-6)
            sc = compute_score(cand)
            maybe_insert(sc, cand)

    # Sort the pool (best first) and extract seed vectors
    best_pool.sort(key=lambda x: x[0])
    seed_vectors = [vec for _, vec in best_pool]
    print(f"Random / grid search produced {len(seed_vectors)} seed vectors (top-{top_k})")

    # ------------------------------------------------------------------
    # 3.  Heavy local refinement of each seed
    # ------------------------------------------------------------------
    for idx, seed_vec in enumerate(seed_vectors, 1):
        if time.time() >= deadline - 30.0:
            break
        print(f"\n--- Refining seed {idx}/{len(seed_vectors)} ---")

        # A) Adam optimisation on smooth surrogate
        remaining = deadline - time.time()
        max_iter_adam = int(min(400_000, remaining * 50))
        refined, _ = adam_optimize_smooth(
            initial_a=seed_vec,
            deadline=deadline,
            max_iter=max_iter_adam,
            lr0=0.12,
            lam_start=2.0,
            lam_end=250.0,
            noise_every=3000,
            noise_scale=2e-5,
        )

        # B) Exact sub‑gradient descent (more iterations than before)
        if time.time() < deadline - 20.0:
            remaining = deadline - time.time()
            max_iter_sub = int(min(400_000, remaining * 45))
            refined = subgradient_descent_exact(
                refined,
                deadline,
                max_iter=max_iter_sub,
                step0=0.03,
            )

        # C) Simulated‑annealing polishing
        if time.time() < deadline - 10.0:
            remaining = deadline - time.time()
            max_iter_sa = int(min(400_000, remaining * 45))
            refined = simulated_annealing(
                refined,
                rng,
                deadline,
                max_iters=max_iter_sa,
                init_temp=0.015,
                decay=0.99995,
            )

        # D) Very small random perturbations (can escape shallow traps)
        if time.time() < deadline - 5.0:
            for _ in range(5):
                perturbed = local_perturbation(refined, rng, sigma=2e-4)
                perturbed = subgradient_descent_exact(
                    perturbed,
                    deadline,
                    max_iter=30_000,
                    step0=0.015,
                )
                if compute_score(perturbed) < compute_score(refined):
                    refined = perturbed

        # Official evaluation
        try:
            val = evaluate_sequence(refined.tolist())
            print(f"    Official score = {val:.6f}")
            if val < best_val:
                best_val = val
                best_seq = refined.tolist()
                print(f"    *** New global best: {best_val:.6f} ***")
        except Exception:
            pass

    # ------------------------------------------------------------------
    # 4.  Global polishing of the overall best (budget permitting)
    # ------------------------------------------------------------------
    if best_seq and time.time() < deadline - 5.0:
        a = np.array(best_seq, dtype=float)
        remaining = deadline - time.time()

        # A final sub‑gradient phase
        sub_iters = int(min(600_000, remaining * 60))
        a = subgradient_descent_exact(a, deadline, max_iter=sub_iters, step0=0.02)

        # Followed by a finishing Adam run with a very sharp soft‑max
        adam_iters = int(min(400_000, remaining * 50))
        a, _ = adam_optimize_smooth(
            a,
            deadline,
            max_iter=adam_iters,
            lr0=0.08,
            lam_start=5.0,
            lam_end=300.0,
            noise_every=4000,
            noise_scale=1e-5,
        )

        final_val = evaluate_sequence(a.tolist())
        print(f"Final polished official score = {final_val:.6f}")
        if final_val < best_val:
            best_val = final_val
            best_seq = a.tolist()
            print("*** Updated global best after final polish ***")

    # ------------------------------------------------------------------
    # 5.  Safety net – guarantee a non‑empty return value
    # ------------------------------------------------------------------
    if not best_seq:
        best_seq = [1.0] * n
        best_val = evaluate_sequence(best_seq)

    print(f"\nSearch finished. Best reported score = {best_val:.6f}")
    return best_seq
