"""
Propose a low‑score step‑function for the official objective

    2·n·max(conv(a,a)) / (sum(a)²)

The strategy is:

1.  Build a fairly large pool of *structured* candidates:
    - Uniform, arcsine, power‑law, and the flexible Beta(α,β) family.
    - A few random Dirichlet draws.
    - If a previous best construction (global ``height_sequence_1``)
      exists it is also added.

2.  Rank the pool with a cheap FFT‑based score (the objective with
    sum(a)=1, i.e. ``2·n·max(conv)``).  The top few candidates are
    taken forward.

3.  For each selected candidate we run a short **smooth mirror‑descent**
    (soft‑max weighting of the convolution) to obtain a refined vector.
    The best refined vector is kept.

4.  The kept vector is *polished* with a standard sub‑gradient method
    that uses the true max‑weight (uniform over the maximisers).  This
    runs until the remaining time budget is exhausted.

All updates are *multiplicative* (exponentiated‑gradient) and the vector
is projected back onto the simplex after each step.
The routine respects the supplied time budget (default 1000 s) and
always returns a feasible list of non‑negative floats.
"""

import time
import numpy as np
from typing import List, Tuple

# ----------------------------------------------------------------------
# Low‑level numeric utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """
    Linear convolution via real FFT (full length, zero‑padded to a power‑of‑2).
    """
    m = len(a) + len(b) - 1
    n_fft = next_pow2(m)
    A = np.fft.rfft(a, n=n_fft)
    B = np.fft.rfft(b, n=n_fft)
    C = A * B
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


def normalize(v: np.ndarray) -> np.ndarray:
    """Project a non‑negative vector onto the simplex (sum = 1)."""
    total = v.sum()
    if total <= 0.0:
        return np.full_like(v, 1.0 / len(v))
    return v / total


def compute_score(a: np.ndarray) -> float:
    """
    Fast evaluation of the official objective when sum(a)=1:
        2·n·max(conv(a,a))
    The vector a must be non‑negative.
    """
    n = len(a)
    conv = convolution_fft(a, a)
    max_val = float(np.max(conv))
    return 2.0 * n * max_val


def gradient_from_weights(a: np.ndarray, w: np.ndarray) -> np.ndarray:
    """
    Gradient of Σ_k w_k·(a*a)_k w.r.t. a.
    Implements   g_j = 2· Σ_m a_m · w_{m+j}.
    Computed with a single FFT‑based convolution.
    """
    w_rev = w[::-1]                      # reverse for convolution
    conv = convolution_fft(a, w_rev)      # length = 3·n‑2
    idx = 2 * len(a) - 2                  # position of the centre (2n‑2)
    grads = 2.0 * conv[idx - np.arange(len(a))]
    return grads


# ----------------------------------------------------------------------
# Parametric families (all return a *probability* vector, i.e. sum=1)
# ----------------------------------------------------------------------


def arcsine_initial(n: int) -> np.ndarray:
    """Discrete arcsine distribution (≈ 1/√(i·(n‑1‑i))) normalised."""
    i = np.arange(n, dtype=float)
    eps = 0.25
    vals = 1.0 / np.sqrt((i + 0.5) * (n - 0.5 - i) + eps)
    return normalize(vals)


def powerlaw_shape(n: int, p: float, eps: float) -> np.ndarray:
    """
    Symmetric power‑law shape a_i ∝ ((i+ε)(n‑1‑i+ε))⁻ᵖ.
    The parameter ``p`` controls the strength of the singularities.
    """
    i = np.arange(n, dtype=float)
    base = ((i + eps) * (n - 1 - i + eps)) ** (-p)
    return normalize(base)


def uniform_shape(n: int) -> np.ndarray:
    """Uniform probability vector on n points."""
    return np.full(n, 1.0 / n)


def random_dirichlet(n: int, rng: np.random.Generator) -> np.ndarray:
    """Sample a random point on the simplex (exponential method)."""
    samp = rng.exponential(1.0, size=n)
    return normalize(samp)


def beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    """
    Discrete Beta(α,β) density on {0,…,n‑1} (after a simple centring).
    a_i ∝ (i+δ)^(α‑1)·(n‑1‑i+δ)^(β‑1) with a tiny δ to avoid 0^neg.
    """
    i = np.arange(n, dtype=float)
    delta = 1e-12
    vals = ((i + delta) ** (alpha - 1.0)) * ((n - 1 - i + delta) ** (beta - 1.0))
    # Guard against NaN/Inf that can appear when α<1 or β<1
    vals = np.where(np.isfinite(vals), vals, 0.0)
    return normalize(vals)


# ----------------------------------------------------------------------
# Refinement methods
# ----------------------------------------------------------------------


def refine_smooth(
    a: np.ndarray,
    deadline: float,
    *,
    eta0: float = 1.0,
    beta: float = 12.0,
    report_every: int = 0,
) -> Tuple[np.ndarray, float]:
    """
    Multiplicative mirror‑descent with a *soft* max‑weight.
    The weight vector is a softmax over the convolution entries.
    Returns the best iterate (exact score) and the corresponding score.
    """
    a = normalize(a.astype(float).copy())
    n = len(a)

    best_a = a.copy()
    conv = convolution_fft(a, a)
    best_sc = compute_score(a)

    it = 0
    while time.time() < deadline:
        it += 1
        conv = convolution_fft(a, a)
        max_val = float(np.max(conv))

        # softmax weights (the entry achieving the max receives weight ≈1)
        exp_vals = np.exp(beta * (conv - max_val))
        w = exp_vals / np.sum(exp_vals)  # ∑ w = 1

        grad = gradient_from_weights(a, w)

        eta = eta0 / np.sqrt(it)

        a = a * np.exp(-eta * grad)
        a = normalize(a)

        cur_sc = compute_score(a)
        if cur_sc < best_sc:
            best_sc = cur_sc
            best_a = a.copy()

        if report_every and (it % report_every == 0):
            print(f"    smooth iter={it:,d} cur={cur_sc:.6f} best={best_sc:.6f}")

    return best_a, best_sc


def refine_subgradient(
    a: np.ndarray,
    deadline: float,
    *,
    max_iter: int = 30_000,
    eta0: float = 0.7,
    report_every: int = 0,
) -> Tuple[np.ndarray, float]:
    """
    Classic sub‑gradient descent on the true max‑convolution objective.
    Uses multiplicative (exponentiated‑gradient) updates and a 1/√t step‑size.
    Returns the best vector seen and its exact score.
    """
    a = normalize(a.astype(float).copy())
    best_a = a.copy()
    best_sc = compute_score(a)

    n = len(a)
    for it in range(1, max_iter + 1):
        if time.time() >= deadline:
            break

        conv = convolution_fft(a, a)
        max_val = float(np.max(conv))

        # uniform weight on the (near) maximisers
        w = np.zeros_like(conv)
        max_idx = np.where(conv >= max_val - 1e-14)[0]
        w[max_idx] = 1.0 / len(max_idx)

        grad = gradient_from_weights(a, w)

        eta = eta0 / np.sqrt(it)
        a = a * np.exp(-eta * grad)
        a = normalize(a)

        cur_sc = compute_score(a)
        if cur_sc < best_sc:
            best_sc = cur_sc
            best_a = a.copy()

        if report_every and (it % report_every == 0):
            print(f"    sub‑grad iter={it:,d} cur={cur_sc:.6f} best={best_sc:.6f}")

    return best_a, best_sc

    mirror‑descent phase and finally polishes the best with a hard
    sub‑gradient descent.  The best sequence found (as a plain list of

    """

    start_time = time.time()
    # small safety margin to avoid hard timeout

    #    Larger n gives a finer search space, but also costs more.
    #    4096 works comfortably within the time limit.
    # ------------------------------------------------------------------
    n = 4096

    # 2. Assemble a pool of diverse initial candidates
    # ------------------------------------------------------------------
    candidates: List[np.ndarray] = []

        if fallback.ndim == 1:
