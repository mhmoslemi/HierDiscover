def local_search_simulated_annealing(a, deadline, max_steps=20000):
    best = a.copy()
    best_score = compute_score(best)
    cur = a.copy()
    cur_score = best_score
    n = len(a)
    rng = np.random.default_rng()
    step = 0
    while time.time() < deadline and step < max_steps:
        step += 1
        # propose: pick i,j distinct, move delta from i to j
        i, j = rng.choice(n, size=2, replace=False)
        delta = rng.uniform(0, min(cur[i], 0.01))  # small shift
        if delta == 0:
            continue
        cur[i] -= delta
        cur[j] += delta
        cur = normalize(cur)  # ensure sum 1 and nonneg as sum is unchanged anyway
        new_score = compute_score(cur)
        # acceptance if better, else revert with some prob maybe
        if new_score < cur_score or rng.random() < np.exp((cur_score - new_score) / (0.001 + step*1e-5)):
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best = cur.copy()
        else:
            # revert
            cur[j] -= delta
            cur[i] += delta
    return best, best_score
```

def refine_smooth_adam(a, deadline, eta0=1.0, beta_start=5.0, beta_end=20.0, ramp_iters=5000, report_every=0):
    a = normalize(a.copy())
    n = len(a)
    best_a = a.copy()
    best_score = compute_score(a)
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta = beta_start
    it = 0
    while time.time() < deadline:
        it += 1
        conv = convolution_fft(a, a)
        max_val = float(np.max(conv))
        # temperature schedule
        beta = beta_start + (beta_end - beta_start) * min(1.0, it / ramp_iters)
        exp_vals = np.exp(beta * (conv - max_val))
        w = exp_vals / exp_vals.sum()
        grad = gradient_from_weights(a, w)
        # Adam moments
        m = 0.9 * m + 0.1 * grad
        v = 0.999 * v + 0.001 * (grad * grad)
        m_hat = m / (1 - 0.9 ** it)
        v_hat = v / (1 - 0.999 ** it)
        step = (eta0 / np.sqrt(it)) * (m_hat / (np.sqrt(v_hat) + 1e-8))
        a = a * np.exp(-step)
        a = normalize(a)
        cur_score = 2.0 * n * max_val  # same as compute_score because sum=1
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
        if report_every and (it % report_every == 0):
            print(f"    smooth iter={it} cur={cur_score:.6f} best={best_score:.6f}")
    return best_a, best_score
```

def refine_subgradient_simple(a, deadline, eta0=0.7, max_iter=200000):
    a = normalize(a.copy())
    n = len(a)
    best_a = a.copy()
    best_score = 2.0 * n * np.max(convolution_fft(a,a))
    it =0
    while time.time() < deadline and it < max_iter:
        it+=1
        conv = convolution_fft(a, a)
        max_val = float(np.max(conv))
        # w uniform on maximizers
        max_indices = np.where(conv >= max_val - 1e-12)[0]
        w = np.zeros_like(conv)
        if len(max_indices) > 0:
            w[max_indices] = 1.0 / len(max_indices)
        grad = gradient_from_weights(a,w)
        eta = eta0 / np.sqrt(it)
        a = a * np.exp(-eta * grad)
        a = normalize(a)
        cur_score = 2.0 * n * max_val
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
        # optional logging
    return best_a, best_score

def refine_lp(f: np.ndarray, deadline: float, t: float = 0.3, max_iter: int = 5) -> np.ndarray:
    n = len(f)
    best_f = f.copy()
    M = np.max(convolution_fft(f, f))
    it = 0
    while time.time() < deadline and it < max_iter:
        it += 1
        rows = 2*n - 1
        A = np.zeros((rows, n), dtype=float)
        for j in range(n):
            A[j:j+n, j] = f
        b = np.full(rows, M)
        res = linprog(-np.ones(n), A_ub=A, b_ub=b, bounds=[(0, None)]*n, method='highs')
        if not res.success:
            print(f"LP failed at iteration {it}")
            break
        g = res.x
        sum_g = g.sum()
        if sum_g <= 0:
            break
        # scale to same total sum as f
        g = g * (f.sum() / sum_g)
        # convex combination
        f_new = (1 - t) * f + t * g
        f_new = normalize(f_new)
        M_new = np.max(convolution_fft(f_new, f_new))
        if M_new < M - 1e-9:
            f = f_new
            M = M_new
            best_f = f.copy()
            print(f"LP iter {it}: improved M to {M:.6f}")
        else:
            # no improvement; break or reduce t
            print(f"LP iter {it}: no improvement")
            break
    return best_f

def upsample_to(f_small: np.ndarray, target_n: int) -> np.ndarray:
    if f_small.shape[0] == target_n:
        return f_small.copy()
    x_old = np.linspace(0, 1, f_small.shape[0])
    x_new = np.linspace(0, 1, target_n)
    interpolator = interp1d(x_old, f_small, kind='linear', fill_value='extrapolate')
    f_new = interpolator(x_new)
    f_new = np.maximum(f_new, 0.0)  # ensure non-negative due to interpolation artifact
    return normalize(f_new)

def next_pow2(x):
    return 1 << (x - 1).bit_length()
def convolution_fft(a,b):
    m = len(a) + len(b) - 1
    n_fft = next_pow2(m)
    A = np.fft.rfft(a, n=n_fft)
    B = np.fft.rfft(b, n=n_fft)
    C = A * B
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv
```

"""
Propose a low‑discrepancy step‑function (non‑negative vector) that
minimises

    2·n·max(conv(a,a)) / (sum(a)²)

The implementation follows three ideas:

1. **Length selection & cheap families** – try several lengths
   (1024 … 16384) and a few parametric families (arcsine, power‑law,
   uniform, random Dirichlet).  The best shape by the fast internal
   score is kept.

2. **Linear‑programming polishing at modest size** – for a moderate
   size (2048) we solve, repeatedly, a linear program

        maximise   Σ g_i
        subject to  (f * g)_k ≤ M   (for all k)
                    g_i ≥ 0,

   where *f* is the current vector and M = max(conv(f,f)).
   The solution spreads mass into positions that do not increase the
   peak of the self‑convolution.  The resulting vector is up‑sampled
   to the final length.

3. **Adam‑accelerated exponentiated‑gradient descent** – a smooth
   mirror‑descent with a soft‑max weight on the convolution entries,
   followed by a hard sub‑gradient polish, uses the official objective
   (exact via `evaluate_sequence`).  The whole search respects the
   given time budget and always returns a feasible list of floats.
"""

import time
from typing import List
import numpy as np
from scipy.optimize import linprog
from scipy.interpolate import interp1d

# ----------------------------------------------------------------------
# Basic numeric utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Return the smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Linear convolution using a real FFT (full length)."""
    m = a.size + b.size - 1
    n_fft = next_pow2(m)
    A = np.fft.rfft(a, n=n_fft)
    B = np.fft.rfft(b, n=n_fft)
    C = A * B
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


def normalize(v: np.ndarray) -> np.ndarray:
    """Project a non‑negative vector onto the probability simplex."""
    total = v.sum()
    if total <= 0.0:
        return np.full_like(v, 1.0 / v.size)
    return v / total


def compute_score(a: np.ndarray) -> float:
    """
    Fast evaluation of the official objective for a vector which already
    satisfies sum(a)=1 (scale‑invariant):
        2·n·max(conv(a,a))
    """
    n = a.size
    conv = convolution_fft(a, a)
    return 2.0 * n * float(conv.max())


def gradient_from_weights(a: np.ndarray, w: np.ndarray) -> np.ndarray:
    """
    Gradient of Σ_k w_k·(a*a)_k  w.r.t. a.
    Implements   g_j = 2· Σ_m a_m·w_{m+j}.
    """
    w_rev = w[::-1]
    conv = convolution_fft(a, w_rev)          # length = 3·n‑2
    idx = 2 * a.size - 2
    grads = 2.0 * conv[idx - np.arange(a.size)]
    return grads


# ----------------------------------------------------------------------
# Parametric families for the initial sweep
# ----------------------------------------------------------------------


def arcsine_initial(n: int) -> np.ndarray:
    """Discrete arcsine distribution (≈ 1/√(i·(n‑1‑i))) normalised."""
    i = np.arange(n, dtype=float)
    eps = 0.25
    vals = 1.0 / np.sqrt((i + 0.5) * (n - 0.5 - i) + eps)
    return normalize(vals)


def powerlaw_shape(n: int, p: float, eps: float) -> np.ndarray:
    """Symmetric power‑law shape a_i ∝ ((i+ε)(n‑1‑i+ε))⁻ᵖ."""
    i = np.arange(n, dtype=float)
    base = ((i + eps) * (n - 1 - i + eps)) ** (-p)
    return normalize(base)


def uniform_shape(n: int) -> np.ndarray:
    """Uniform probability vector on n points."""
    return np.full(n, 1.0 / n)


def random_dirichlet(n: int, rng: np.random.Generator) -> np.ndarray:
    """Random point on the simplex (exponential sampling)."""
    samp = rng.exponential(1.0, size=n)
    return normalize(samp)


# ----------------------------------------------------------------------
# Helpers for resizing / up‑sampling
# ----------------------------------------------------------------------


def upsample_to(shape: np.ndarray, target_n: int) -> np.ndarray:
    """Linear interpolation of a probability vector to a new length."""
    if shape.size == target_n:
        return shape.copy()
    xs = np.linspace(0.0, 1.0, shape.size)
    xs_new = np.linspace(0.0, 1.0, target_n)
    f = interp1d(xs, shape, kind="linear", fill_value="extrapolate")
    new_shape = np.maximum(f(xs_new), 0.0)
    return normalize(new_shape)


# ----------------------------------------------------------------------
# Linear‑programming refinement (small size, cheap)
# ----------------------------------------------------------------------


def refine_lp(f: np.ndarray, deadline: float, *,
              t: float = 0.3,
              max_iter: int = 5) -> np.ndarray:
    """
    Repeatedly solve the LP

        maximise   Σ g_i
        subject to (f * g)_k ≤ M   (∀k)
                   g_i ≥ 0

    where M = max(conv(f,f)).  The solution is mixed with the current
    vector using a convex combination with weight ``t``.
    The routine stops when time runs out or no improvement occurs.
    """
    n = f.size
    best_f = f.copy()
    M = np.max(convolution_fft(f, f))
    it = 0
    while time.time() < deadline and it < max_iter:
        it += 1
        rows = 2 * n - 1
        # Build the convolution matrix A (dense, but n≤2048)
        A = np.zeros((rows, n), dtype=float)
        for j in range(n):
            A[j:j + n, j] = f
        b = np.full(rows, M, dtype=float)

        # Linear program: minimise -Σ g_i   (i.e. maximise Σ g_i)
        res = linprog(-np.ones(n),
                      A_ub=A, b_ub=b,
                      bounds=[(0, None)] * n,
                      method="highs")
        if not res.success:
            print(f"[LP] iteration {it}: solver failed – stopping.")
            break

        g = res.x
        sum_g = g.sum()
        if sum_g <= 0.0:
            break
        # Rescale g to have the same total mass as f (sum = 1)
        g = g * (f.sum() / sum_g)

        # Convex combination and re‑normalisation
        f_new = (1.0 - t) * f + t * g
        f_new = normalize(f_new)

        M_new = np.max(convolution_fft(f_new, f_new))
        if M_new < M - 1e-12:
            f = f_new
            M = M_new
            best_f = f.copy()
            print(f"[LP] iteration {it}: improved M to {M:.6f}")
        else:
            print(f"[LP] iteration {it}: no improvement – stopping.")
            break
    return best_f


# ----------------------------------------------------------------------
# Smooth Adam‑accelerated mirror descent
# ----------------------------------------------------------------------


def refine_smooth_adam(a: np.ndarray, deadline: float, *,
                       eta0: float = 1.0,
                       beta_start: float = 5.0,
                       beta_end: float = 25.0,
                       ramp_iters: int = 5000,
                       report_every: int = 0) -> (np.ndarray, float):
    """
    Exponentiated‑gradient (mirror‑descent) with Adam‑style adaptive
    step sizes.  The weight vector is a soft‑max over the convolution
    entries: w_i ∝ exp(β·(conv_i – max(conv))).  Returns the best point
    seen (internal score) and its score.
    """
    a = normalize(a.astype(float).copy())
    n = a.size
    best_a = a.copy()
    best_score = compute_score(a)

    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8

    it = 0
    while time.time() < deadline:
        it += 1
        conv = convolution_fft(a, a)
        max_val = float(conv.max())

        # temperature schedule: start soft, become hard
        beta = beta_start + (beta_end - beta_start) * min(1.0, it / ramp_iters)

        # soft‑max weights on the convolution values
        exp_vals = np.exp(beta * (conv - max_val))
        w = exp_vals / exp_vals.sum()

        grad = gradient_from_weights(a, w)

        # Adam moments
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)

        # bias‑corrected estimates
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        # learning rate schedule
        eta = eta0 / np.sqrt(it)

        step = eta * (m_hat / (np.sqrt(v_hat) + eps))
        a = a * np.exp(-step)
        a = normalize(a)

        cur_score = 2.0 * n * max_val
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()

        if report_every and (it % report_every == 0):
            print(f"    smooth‑adam iter={it:,d} cur={cur_score:.6f} best={best_score:.6f}")

    return best_a, best_score


# ----------------------------------------------------------------------
# Hard sub‑gradient polishing (exact max weight)
# ----------------------------------------------------------------------


def refine_subgradient_simple(a: np.ndarray, deadline: float, *,
                              eta0: float = 0.7,
                              max_iter: int = 200_000,
                              report_every: int = 0) -> (np.ndarray, float):
    """
    Classical sub‑gradient descent on the true max‑convolution
    objective using exponentiated‑gradient updates.
    """
    a = normalize(a.astype(float).copy())
    n = a.size
    best_a = a.copy()
    best_score = compute_score(a)

    for it in range(1, max_iter + 1):
        if time.time() >= deadline:
            break

        conv = convolution_fft(a, a)
        max_val = float(conv.max())

        # uniform weight on all (near) maximisers
        mask = conv >= max_val - 1e-14
        w = np.zeros_like(conv)
        nb = mask.sum()
        if nb > 0:
            w[mask] = 1.0 / nb

        grad = gradient_from_weights(a, w)

        eta = eta0 / np.sqrt(it)
        a = a * np.exp(-eta * grad)
        a = normalize(a)

        cur_score = 2.0 * n * max_val
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()

        if report_every and (it % report_every == 0):
            print(f"    subgrad iter={it:,d} cur={cur_score:.6f} best={best_score:.6f}")

    return best_a, best_score
