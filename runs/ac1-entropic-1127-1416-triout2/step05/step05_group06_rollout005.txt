    a = np.maximum(initial_a, 1e-30)
    a = a / a.sum()
    best_a = a.copy()
    best_score = compute_score(a)
    n = len(a)
    grad_sq_acc = np.zeros(n) # accumulated squared (per iteration)
    eps = 1e-8
    it = 0
    while time.time() < deadline:
        it += 1
        conv = convolution_fft(a, a)
        grad = _max_conv_gradient(a, conv)  # shape n
        grad_sq_acc += grad**2
        # compute adaptive step: eta = base / sqrt(grad_sq_acc + eps)

        a = a * np.exp(-step * grad)

        a_sum = a.sum()
        if a_sum == 0:

"""
Hybrid optimizer for the max‑convolution bound.

The routine builds a diverse set of seeds (arcsine, power‑law, beta‑like,
Dirichlet draws, mixtures) for a few sequence lengths (powers of two).
For every promising seed we run two phases:

1.  A smooth‑max gradient descent (log‑sum‑exp surrogate) with an
    increasingly aggressive β‑schedule.  The gradient is obtained by a
    single FFT per iteration; an exponentiated‑gradient step keeps the
    entries non‑negative and a simple renormalisation enforces the
    simplex constraint.

2.  An AdaGrad‑style exponentiated sub‑gradient descent on the true
    max‑convolution (using the exact sub‑gradient).  The per‑coordinate
    learning rates shrink with the accumulated squared gradient, which
    yields fine‑grained updates in the later stages.

Both phases are run as long as the global time budget (default 1000 s) permits.
The best vector seen (according to the fast internal score, identical to the
official objective) is returned.  A final polishing pass is performed if time
remains.

All sequences are non‑negative, summed to one (scaling does not affect the
objective) and are clipped to the allowed range by ``evaluate_sequence``.
"""

import time
import numpy as np
from typing import List, Tuple

# ----------------------------------------------------------------------
# Basic numeric utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full linear convolution using a real FFT (zero‑padded to a power of two)."""
    m = len(a) + len(b) - 1
    n_fft = next_pow2(m)
    A = np.fft.rfft(a, n=n_fft)
    B = np.fft.rfft(b, n=n_fft)
    C = A * B
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


def compute_score(a: np.ndarray) -> float:
    """
    Fast version of the official objective (scale‑invariant):
        2·n·max(a∗a) / (sum(a)²)
    The vector a is assumed non‑negative.
    """
    n = len(a)
    conv = convolution_fft(a, a)
    max_val = float(conv.max())
    s = float(a.sum())
    if s < 1e-12:
        return float("inf")
    return 2.0 * n * max_val / (s * s)


# ----------------------------------------------------------------------
# Families for generating initial candidates
# ----------------------------------------------------------------------


def arcsine_initial(n: int) -> np.ndarray:
    """Discrete arcsine density, normalised to sum = 1."""
    i = np.arange(n, dtype=float)
    eps = 0.25
    vals = 1.0 / np.sqrt((i + 0.5) * (n - 0.5 - i) + eps)
    return vals / vals.sum()


def generate_powerlaw(n: int, p: float, eps: float) -> np.ndarray:
    """Symmetric power‑law a_i ∝ ((i+ε)(n‑1‑i+ε))⁻ᵖ."""
    i = np.arange(n, dtype=float)
    base = ((i + eps) * (n - 1 - i + eps)) ** (-p)
    total = base.sum()
    if total == 0.0:
        return np.full(n, 1.0 / n)
    return base / total


def generate_beta_like(n: int, alpha: float, beta: float) -> np.ndarray:
    """Beta‑like shape a_i ∝ (i+½)^{α‑1} (n‑1‑i+½)^{β‑1}."""
    i = np.arange(n, dtype=float)
    base = (i + 0.5) ** (alpha - 1) * (n - 1 - i + 0.5) ** (beta - 1)
    total = base.sum()
    if total == 0.0:
        return np.full(n, 1.0 / n)
    return base / total


def random_dirichlet(n: int, concentration: float, rng: np.random.Generator) -> np.ndarray:
    """Draw a Dirichlet‑like vector and normalise."""
    if concentration <= 0:
        samp = rng.exponential(1.0, size=n) ** (1.0 / max(0.1, concentration))
    else:
        samp = rng.gamma(concentration, 1.0, size=n)
    samp = np.maximum(samp, 1e-12)
    return samp / samp.sum()


def perturb(a: np.ndarray, rng: np.random.Generator, scale: float = 0.02) -> np.ndarray:
    """Add a tiny Gaussian perturbation and renormalise."""
    b = a + rng.normal(0.0, scale, size=a.shape)
    b = np.maximum(b, 0.0)
    s = b.sum()
    if s == 0.0:
        return np.full_like(a, 1.0 / len(a))
    return b / s


# ----------------------------------------------------------------------
# Smooth‑max surrogate optimisation (log‑sum‑exp)
# ----------------------------------------------------------------------


def softmax_gradient_descent(initial_a: np.ndarray,
                             deadline: float,
                             betas: List[float],
                             iter_per_beta: int = 500,
                             eta0: float = 0.5) -> Tuple[np.ndarray, float]:
    """
    Minimise a smooth approximation of the max‑convolution:
        Φ_β(a) = (1/β)·log Σ_k exp(β·(a∗a)_k).

    Exponentiated‑gradient steps are taken; after each step the vector is
    projected back onto the probability simplex by simple renormalisation.
    """
    a = np.maximum(initial_a, 1e-30)
    a = a / a.sum()
    best_a = a.copy()
    best_sc = compute_score(a)
    n = len(a)

    for beta in betas:
        for it in range(1, iter_per_beta + 1):
            if time.time() >= deadline:
                return best_a, best_sc

            conv = convolution_fft(a, a)                # length 2n‑1
            max_c = conv.max()
            # stable soft‑max weights
            exp_vals = np.exp(beta * (conv - max_c))
            p = exp_vals / exp_vals.sum()                # Σ p = 1

            # cross‑correlation p ⋆ a  (linear, length n)
            N = next_pow2(len(p) + n - 1)
            P = np.fft.rfft(p, N)
            A = np.fft.rfft(a, N)
            cross = np.fft.irfft(P * np.conj(A), N)[:n].real
            cross = np.maximum(cross, 0.0)

            grad = 4.0 * n * cross                        # non‑negative

            eta = eta0 / np.sqrt(it)
            a = a * np.exp(-eta * grad)

            # simplex projection
            s = a.sum()
            if s == 0.0:
                a = np.full_like(a, 1.0 / n)
            else:
                a = a / s

            cur_sc = compute_score(a)
            if cur_sc < best_sc:
                best_sc = cur_sc
                best_a = a.copy()
    return best_a, best_sc


# ----------------------------------------------------------------------
# Exact max‑convolution sub‑gradient (AdaGrad version)
# ----------------------------------------------------------------------


def _max_conv_gradient(a: np.ndarray, conv: np.ndarray) -> np.ndarray:
    """
    Sub‑gradient of max_k (a∗a)_k.
    For each active offset k (where the maximum is attained) the gradient
    contribution is 2·a_{k‑j} for the appropriate indices j.
    """
    n = len(a)
    max_val = conv.max()
    mask = np.isclose(conv, max_val, atol=1e-15, rtol=0.0)
    active = np.where(mask)[0]
    grad = np.zeros_like(a)

    for k in active:
        lo = max(0, k - (n - 1))
        hi = min(k, n - 1)
        idx = np.arange(lo, hi + 1, dtype=int)
        grad[idx] += 2.0 * a[k - idx]

    if len(active) > 1:
        grad /= len(active)
    return grad


def optimise_adagrad_subgradient(initial_a: np.ndarray,
                                 deadline: float,
                                 rng: np.random.Generator,
                                 *,
                                 base_lr: float = 0.4,
                                 max_iter: int = 200_000,
                                 noise_every: int = 0,
                                 noise_scale: float = 0.0) -> Tuple[np.ndarray, float]:
    """
    AdaGrad‑style exponentiated sub‑gradient descent on the true max‑convolution.
    A per‑coordinate step size  η_i = base_lr / (√(G_i) + 1)  is used, where
    G_i accumulates the squared gradients.  Optional additive Gaussian noise
    (controlled by ``noise_every``) helps to escape shallow plateaus.
    """
    a = np.maximum(initial_a, 1e-30)
    a = a / a.sum()
    best_a = a.copy()
    best_sc = compute_score(a)
    n = len(a)

    grad_sq_acc = np.zeros_like(a)
    eps = 1e-12

    for it in range(1, max_iter + 1):
        if time.time() >= deadline:
            break

        conv = convolution_fft(a, a)
        grad = _max_conv_gradient(a, conv)

        grad_sq_acc += grad ** 2
        step = base_lr / (np.sqrt(grad_sq_acc) + 1.0)

        a = a * np.exp(-step * grad)

        # optional noise injection
        if noise_every and (it % noise_every == 0):
            a += rng.normal(0.0, noise_scale, size=n)
            a = np.maximum(a, 0.0)

        s = a.sum()
        if s == 0.0:
            a = np.full_like(a, 1.0 / n)
        else:
            a = a / s

        cur_sc = compute_score(a)
        if cur_sc < best_sc:
            best_sc = cur_sc
            best_a = a.copy()

        if it % 50000 == 0:
            print(f"    adagrad iter={it:,d} cur={cur_sc:.6f} best={best_sc:.6f}")

    return best_a, best_sc

        candidates.append(arcsine_initial(n))

        # random power‑law shapes
        for _ in range(6):
            p = rng.uniform(0.1, 1.0)
            eps = rng.uniform(0.0, max(1.0, 0.02 * n))
            candidates.append(generate_powerlaw(n, p, eps))

        # beta‑like shapes
        for _ in range(4):
            alpha = rng.uniform(0.2, 4.0)
            beta = rng.uniform(0.2, 4.0)
            candidates.append(generate_beta_like(n, alpha, beta))

        # Dirichlet / exponential draws
        for conc in [0.2, 0.5, 1.0, 2.0]:
            candidates.append(random_dirichlet(n, conc, rng))

        # perturbed arcsine (helps to leave the immediate basin)
        for _ in range(3):
            candidates.append(perturb(arcsine_initial(n), rng, scale=0.04))

        # simple mixtures of arcsine and uniform
        for _ in range(2):
            w = rng.random()
            mix = w * arcsine_initial(n) + (1.0 - w) * np.full(n, 1.0 / n)
            candidates.append(mix / mix.sum())

        # --------------------------------------------------------------
        # 1b.  Quick sweep with the fast internal score, keep the best few.
        # --------------------------------------------------------------
        scored = [(compute_score(c), c) for c in candidates]
        scored.sort(key=lambda x: x[0])
        top_candidates = [c for _, c in scored[:5]]   # keep the five most promising
        print(f"  Selected {len(top_candidates)} top seeds (fast score).")

        # --------------------------------------------------------------
        # 1c.  Refine each retained candidate.
        # --------------------------------------------------------------
        for idx, cand in enumerate(top_candidates):
            if time.time() >= deadline - 5.0:
                break
            print(f"  candidate {idx + 1}/{len(top_candidates)} (n={n})")

            # ---- Smooth‑max gradient descent ---------------------------------
            betas = [5, 15, 30, 60, 120, 240, 480, 720]
            remaining = max(0.0, deadline - time.time())
            it_per_beta = max(150, int(min(1500, remaining * 200) / len(betas)))
            sg_a, _ = softmax_gradient_descent(
                cand,
                deadline,
                betas=betas,
                iter_per_beta=it_per_beta,
                eta0=0.5,
            )
            cur_fast = compute_score(sg_a)
            if cur_fast < best_val:
                best_val = cur_fast
                best_seq = sg_a.tolist()
                print(f"    ** New global best after soft‑max GD: {best_val:.6f}")

            # ---- AdaGrad sub‑gradient (exact max) ---------------------------
            remaining = max(0.0, deadline - time.time())
            if remaining < 1.0:
                continue
            # allocate up to a few hundred thousand iterations depending on time left
            max_iter = int(min(300_000, remaining * 350))
            sub_a, _ = optimise_adagrad_subgradient(
                sg_a,
                deadline,
                rng,
                base_lr=0.45,
                max_iter=max_iter,
                noise_every=0,
                noise_scale=0.0,
            )
            cur_fast = compute_score(sub_a)
            if cur_fast < best_val:
                best_val = cur_fast
                best_seq = sub_a.tolist()
                print(f"    ** New global best after AdaGrad: {best_val:.6f}")

    # ------------------------------------------------------------------
    # 2.  Final polishing (if time permits).
    # ------------------------------------------------------------------
    if best_seq and time.time() < deadline - 3.0:
        print("\nFinal polishing with AdaGrad sub‑gradient")
        a = np.array(best_seq, dtype=float)
        remaining = max(0.0, deadline - time.time())
        max_iter = int(min(500_000, remaining * 400))
        final_a, _ = optimise_adagrad_subgradient(
            a,
            deadline,
            rng,
            base_lr=0.4,
            max_iter=max_iter,
            noise_every=0,
            noise_scale=0.0,
        )
        final_fast = compute_score(final_a)
        if final_fast < best_val:
            best_val = final_fast
            best_seq = final_a.tolist()
            print(f"    *** Updated global best after final polishing: {best_val:.6f}")

    # ------------------------------------------------------------------
    # 3.  Guard – always return a feasible sequence.
    # ------------------------------------------------------------------
    if not best_seq:
        best_seq = [1.0] * 256
        best_val = compute_score(np.array(best_seq, dtype=float))

    # Convert to plain Python floats (required by the harness)
    best_seq = [float(x) for x in best_seq]

    # One final official evaluation for reporting purposes (does not affect the returned value)
    try:
        official_val = evaluate_sequence(best_seq)
        print(f"\nOfficial ``evaluate_sequence`` score = {official_val:.6f}")
    except Exception as exc:
        print(f"\nOfficial evaluation failed: {exc}")

    print(f"\nSearch finished. Best fast‑score ≈ {best_val:.6f}")
    return best_seq
