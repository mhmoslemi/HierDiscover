    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """
    Linear convolution via real FFT.
    Length = len(a)+len(b)-1.
    """
    m = len(a) + len(b) - 1
    n_fft = next_pow2(m)
    A = np.fft.rfft(a, n=n_fft)
    B = np.fft.rfft(b, n=n_fft)
    C = A * B
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


def compute_score(a: np.ndarray) -> float:
    """
    Fast evaluation used during the search.
    The vector is assumed to sum to one, therefore the denominator of the
    official score disappears and the objective reduces to 2·n·max(conv(a,a)).
    """
    n = a.shape[0]
    conv = convolution_fft(a, a)
    return 2.0 * n * float(np.max(conv))


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the probability simplex
    {x ≥ 0, Σ x = 1}.  Implements the algorithm of Duchi et al. (2008).
    """
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / v.size)

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho = np.where(u - cssv / np.arange(1, v.size + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def arcsine_initial(n: int) -> np.ndarray:
    """Discrete arcsine shape: a_i ∝ 1/√((i+½)*(n‑½‑i))."""
    i = np.arange(n, dtype=float)
    eps = 0.25
    vals = 1.0 / np.sqrt((i + 0.5) * (n - 0.5 - i) + eps)
    vals /= vals.sum()
    return vals


def uniform_initial(n: int) -> np.ndarray:
    """Uniform distribution on n points."""
    return np.full(n, 1.0 / n)


def generate_beta_shape(n: int, alpha: float, beta: float, eps: float) -> np.ndarray:
    """
    Beta‑type shape: (i+ε)^{α‑1}·(n‑1‑i+ε)^{β‑1} on {0,…,n‑1},
    normalised to Σ=1.
    """
    i = np.arange(n, dtype=float)
    left = (i + eps) ** (alpha - 1.0)
    right = (n - 1 - i + eps) ** (beta - 1.0)
    base = left * right
    total = base.sum()
    if not np.isfinite(total) or total <= 0.0:
        return uniform_initial(n)
    return base / total


def generate_power_shape(n: int, p: float, eps: float) -> np.ndarray:
    """
    Generalised arcsine shape: a_i ∝ ((i+½)*(n‑½‑i)+ε)^{-p}.
    """
    i = np.arange(n, dtype=float)
    denom = ((i + 0.5) * (n - 0.5 - i) + eps) ** p
    vals = 1.0 / denom
    total = vals.sum()
    if not np.isfinite(total) or total <= 0.0:
        return uniform_initial(n)
    return vals / total


def generate_mixture_shape(n: int, w_arc: float) -> np.ndarray:
    """Mixture (w)·arcsine + (1‑w)·uniform, normalised to one."""
    arc = arcsine_initial(n)
    uni = uniform_initial(n)
    mix = w_arc * arc + (1.0 - w_arc) * uni
    mix /= mix.sum()
    return mix


def generate_dirichlet_shape(n: int, rng: np.random.Generator) -> np.ndarray:
    """Full Dirichlet sample with exponential parameters (≈ many tiny entries)."""
    alphas = rng.exponential(scale=1.0, size=n) + 0.01
    vec = alphas / alphas.sum()
    return vec

def smooth_gradient(a: np.ndarray, lam: float) -> np.ndarray:
    """
    Gradient of the soft‑max approximation of max(conv(a,a)).
    Returns ∇_a (2·n·softmax_max) – the sum‑to‑one constraint is handled
    later by the simplex projection.
    """
    n = a.shape[0]
    conv = convolution_fft(a, a)                     # length 2n‑1
    max_c = np.max(conv)

    # Soft‑max weights (stable)
    exp_term = np.exp(lam * (conv - max_c))
    w = exp_term / np.sum(exp_term)                 # shape (2n‑1,)

    # Weighted gradient: 2· Σ_k w_k·shift(a,k)
    # This can be written as a convolution of a with the reversed weight vector.
    w_rev = w[::-1]
    conv_rev = convolution_fft(a, w_rev)            # length 2n‑1
    L = 2 * n - 1
    weighted_grad = 2.0 * conv_rev[L - 1 - np.arange(n)]

    # Final factor 2·n
    return 2.0 * n * weighted_grad


def adam_optimize_smooth(initial_a: np.ndarray,
                         deadline: float,
                         max_iter: int = 200_000,
                         lr0: float = 0.10,
                         lam_start: float = 2.0,
                         lam_end: float = 200.0,
                         noise_every: int = 3_000,
                         noise_scale: float = 5e-5) -> Tuple[np.ndarray, float]:
    """
    Adam optimiser on the smooth‑max surrogate.
    Returns the best vector seen (projected onto the simplex) together
    with its fast score.
    """
    a = project_to_simplex(initial_a.astype(float).copy())
    best_a = a.copy()
    best_sc = compute_score(a)

    beta1, beta2 = 0.9, 0.999
    eps = 1e-8
    m = np.zeros_like(a)
    v = np.zeros_like(a)

    for it in range(1, max_iter + 1):
        if time.time() >= deadline:
            break

        # Linear schedule for λ (soft‑max sharpness)
        lam = lam_start + (lam_end - lam_start) * min(1.0, it / max_iter)

        grad = smooth_gradient(a, lam)

        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)

        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        step = lr0 * m_hat / (np.sqrt(v_hat) + eps)
        a_new = a - step

        if noise_every and (it % noise_every == 0):
            a_new += np.random.randn(a.size) * noise_scale

        a_new = project_to_simplex(a_new)

        cur_sc = compute_score(a_new)
        if cur_sc < best_sc:
            best_sc = cur_sc
            best_a = a_new.copy()

        a = a_new

        if it % 5_000 == 0:
            print(f"    AdamSmooth it={it:,d} cur={cur_sc:.6f} best={best_sc:.6f}")

    return best_a, best_sc


def subgradient_descent_exact(initial_a: np.ndarray,
                              deadline: float,
                              max_iter: int = 120_000,
                              step0: float = 0.05) -> np.ndarray:
    """
    Sub‑gradient descent on the exact objective 2·n·max(conv).
    The stepsize follows ηₜ = step0 / √t.
    """
    a = project_to_simplex(initial_a.copy())
    best_a = a.copy()
    best_sc = compute_score(a)
    n = len(a)
    L = 2 * n - 1

    for it in range(1, max_iter + 1):
        if time.time() >= deadline:
            break

        conv = convolution_fft(a, a)
        max_val = np.max(conv)

        # Mask of entries attaining (approximately) the maximum.
        tol = max_val * 1e-12 + 1e-14
        mask = (np.abs(conv - max_val) <= tol).astype(float)

        # Sub‑gradient of max(conv) is the sum of shifted masks (see the paper).
        mask_rev = mask[::-1]
        conv_rev = convolution_fft(a, mask_rev)
        weighted_grad = 2.0 * conv_rev[L - 1 - np.arange(n)]

        eta = step0 / np.sqrt(it)
        a = a - eta * weighted_grad
        a = project_to_simplex(a)

        cur_sc = compute_score(a)
        if cur_sc < best_sc:
            best_sc = cur_sc
            best_a = a.copy()

        if it % 5_000 == 0:
            print(f"    SubGrad it={it:,d} cur={cur_sc:.6f} best={best_sc:.6f}")

    return best_a


def simulated_annealing(a: np.ndarray,
                        rng: np.random.Generator,
                        deadline: float,
                        max_iters: int = 80_000,
                        init_temp: float = 0.015,
                        decay: float = 0.9999) -> np.ndarray:
    """
    Simple Metropolis‑style local search that moves a small amount of mass
    from one coordinate to another.  The fast surrogate score is used for the
    acceptance test.
    """
    cur = a.copy()
    cur_sc = compute_score(cur)
    best = cur.copy()
    best_sc = cur_sc
    n = len(cur)
    temp = init_temp
    it = 0

    while it < max_iters and time.time() < deadline:
        it += 1
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue

        # Transfer up to 20 % of the mass at i.
        delta = cur[i] * 0.2 * rng.random()
        if delta <= 1e-15 or cur[i] - delta < 0:
            continue

        cand = cur.copy()
        cand[i] -= delta
        cand[j] += delta

        cand_sc = compute_score(cand)

        if cand_sc < cur_sc or rng.random() < np.exp(-(cand_sc - cur_sc) / max(temp, 1e-12)):
            cur = cand
            cur_sc = cand_sc
            if cur_sc < best_sc:
                best = cur.copy()
                best_sc = cur_sc

        temp = max(temp * decay, 1e-8)

        if it % 5_000 == 0:
            print(f"    SA it={it:,d} cur={cur_sc:.6f} best={best_sc:.6f}")

    return best

def propose_candidate(seed: int = 43,
                     budget_s: int = 1000,
                     **kwargs) -> List[float]:
    """
    Search for a non‑negative step sequence with a low ``evaluate_sequence``
    value.  The routine blends a large random exploration of parametric
    families with an intensive local optimisation (Adam → sub‑gradient →
    simulated‑annealing).  All work is performed on vectors that are
    already normalised to sum = 1, hence the fast surrogate ``2·n·max(conv)``
    can be used throughout the search.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 1.0   # 1 s safety margin

    best_seq: List[float] = []
    best_val: float = float("inf")

    # ------------------------------------------------------------------
    # 0.  Fallback from a previous run (if present)
    # ------------------------------------------------------------------
    if "height_sequence_1" in globals():
        try:
            fallback = [float(x) for x in globals()["height_sequence_1"]]
            val = evaluate_sequence(fallback)
            print(f"Fallback (height_sequence_1) score = {val:.6f}")
            if val < best_val:
                best_val = val
                best_seq = fallback
        except Exception:
            pass

    # ------------------------------------------------------------------
    # 1.  Wide random exploration – keep the top‑k most promising seeds
    # ------------------------------------------------------------------
    n = 4096                     # a larger length gives a tighter theoretical bound
    top_k = 16
    best_pool: List[Tuple[float, np.ndarray]] = []

    def maybe_insert(score: float, vec: np.ndarray):
        """Insert a candidate into the bounded top‑k heap."""
        if len(best_pool) < top_k:
            best_pool.append((score, vec))
        else:
            # find worst (largest) score in pool
            worst_idx = max(range(len(best_pool)),
                            key=lambda i: best_pool[i][0])
            if score < best_pool[worst_idx][0]:
                best_pool[worst_idx] = (score, vec)

    # deterministic starters
    maybe_insert(compute_score(uniform_initial(n)), uniform_initial(n))
    maybe_insert(compute_score(arcsine_initial(n)), arcsine_initial(n))

    # Randomised sampling loop (time‑controlled)
    max_candidates = 300_000
    cnt = 0
    while time.time() < deadline - 30.0 and cnt < max_candidates:
        cnt += 1
        r = rng.random()
        if r < 0.30:                 # beta family
            alpha = rng.uniform(0.02, 5.0)
            beta = rng.uniform(0.02, 5.0)
            eps = rng.uniform(1e-6, 2.0)
            cand = generate_beta_shape(n, alpha, beta, eps)
        elif r < 0.55:               # power / generalized arcsine
            p = rng.uniform(0.1, 5.0)
            eps = rng.uniform(0.0, 2.0)
            cand = generate_power_shape(n, p, eps)
        elif r < 0.70:               # mixture arcsine / uniform
            w = rng.random()
            cand = generate_mixture_shape(n, w)
        elif r < 0.85:               # Dirichlet‑type (many tiny entries)
            cand = generate_dirichlet_shape(n, rng)
        else:                        # simple triangular bump (random peak)
            peak = rng.integers(0, n)
            left = np.arange(peak + 1, dtype=float)
            right = np.arange(n - peak - 1, dtype=float)[::-1]
            tri = np.concatenate([left, right])
            cand = tri / tri.sum()

        sc = compute_score(cand)
        maybe_insert(sc, cand)

    # Sort the kept seeds by score (best first)
    best_pool.sort(key=lambda x: x[0])
    seed_vectors = [vec for _, vec in best_pool]

    # ------------------------------------------------------------------
    # 2.  Refinement of each promising seed
    # ------------------------------------------------------------------
    for idx, seed_vec in enumerate(seed_vectors, 1):
        if time.time() >= deadline - 5.0:
            break
        print(f"\n--- Refining seed {idx}/{len(seed_vectors)} (score {compute_score(seed_vec):.6f}) ---")

        # ---- Adam on smooth surrogate ----
        remaining = deadline - time.time()
        max_iter_adam = int(min(300_000, remaining * 45))
        print(f"    Adam (smooth), max_iter={max_iter_adam}")
        refined, _ = adam_optimize_smooth(
            initial_a=seed_vec,
            deadline=deadline,
            max_iter=max_iter_adam,
            lr0=0.12,
            lam_start=2.0,
            lam_end=200.0,
            noise_every=3_000,
            noise_scale=2e-5,
        )

        # ---- Exact sub‑gradient descent ----
        if time.time() < deadline - 5.0:
            remaining = deadline - time.time()
            max_iter_sub = int(min(250_000, remaining * 40))
            print(f"    Sub‑gradient descent, max_iter={max_iter_sub}")
            refined = subgradient_descent_exact(
                refined,
                deadline,
                max_iter=max_iter_sub,
                step0=0.04,
            )

        # ---- Simulated annealing polishing ----
        if time.time() < deadline - 5.0:
            remaining = deadline - time.time()
            max_iter_sa = int(min(200_000, remaining * 35))
            print(f"    Simulated annealing, max_iters={max_iter_sa}")
            refined = simulated_annealing(
                refined,
                rng,
                deadline,
                max_iters=max_iter_sa,
                init_temp=0.012,
                decay=0.99995,
            )

        # Official evaluation
        try:
            val = evaluate_sequence(refined.tolist())
            print(f"    Official score = {val:.6f}")
            if val < best_val:
                best_val = val
                best_seq = refined.tolist()
                print(f"    *** New global best: {best_val:.6f} ***")
        except Exception:
            pass

    # ------------------------------------------------------------------
    # 3.  Global polishing of the current best vector (if time permits)
    # ------------------------------------------------------------------
    if best_seq and time.time() < deadline - 3.0:
        a = np.array(best_seq, dtype=float)
        remaining = deadline - time.time()
        sub_iters = int(min(300_000, remaining * 30))
        adam_iters = int(min(250_000, remaining * 30))

        print("\n--- Final global polishing ---")
        a = subgradient_descent_exact(a, deadline, max_iter=sub_iters, step0=0.03)
        a, _ = adam_optimize_smooth(
            a,
            deadline,
            max_iter=adam_iters,
            lr0=0.08,
            lam_start=2.0,
            lam_end=200.0,
            noise_every=2_000,
            noise_scale=1e-5,
        )
        final_val = evaluate_sequence(a.tolist())
        print(f"Final polished official score = {final_val:.6f}")
        if final_val < best_val:
            best_val = final_val
            best_seq = a.tolist()
            print("*** Updated global best after final polish ***")

    # ------------------------------------------------------------------
    # 4.  Guard – ensure a non‑empty return value
    # ------------------------------------------------------------------
    if not best_seq:
        # A safe fallback (uniform) – always valid.
        best_seq = uniform_initial(n).tolist()
        best_val = evaluate_sequence(best_seq)

    print(f"\nSearch finished. Best reported score = {best_val:.6f}")
    return best_seq

Improved gradient‑based and evolutionary search for a non‑negative step
sequence that minimizes the official evaluation function.

The algorithm works in three stages:

1. **Broad random exploration** – many candidate vectors are drawn from a
   variety of parametric families (uniform, arcsine, beta‑shapes,
   power‑shapes, mixtures and full Dirichlet draws).  The best `top_k`
   candidates (by the fast surrogate 2·n·max(conv)) are kept.

   succession of three optimisers:

   a longer sub‑gradient run followed by a second Adam pass.

by a Euclidean projection (`project_to_simplex`).  The function finally
returns a plain Python list ready for `evaluate_sequence`.

        grad = smooth_gradient(a, lam)

        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)

        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        step = lr0 * m_hat / (np.sqrt(v_hat) + eps)
        a_new = a - step

        if noise_every and (it % noise_every == 0):
            a_new += np.random.randn(a.size) * noise_scale

        a_new = project_to_simplex(a_new)

        cur_sc = compute_score(a_new)
        if cur_sc < best_sc:
            best_sc = cur_sc
            best_a = a_new.copy()

        a = a_new

        if it % 5_000 == 0:
            print(f"    AdamSmooth it={it:,d} cur={cur_sc:.6f} best={best_sc:.6f}")

    return best_a, best_sc


def subgradient_descent_exact(initial_a: np.ndarray,
                              deadline: float,
                              max_iter: int = 120_000,
                              step0: float = 0.05) -> np.ndarray:
    """
    Sub‑gradient descent on the exact objective 2·n·max(conv).
    The stepsize follows ηₜ = step0 / √t.
    """
    a = project_to_simplex(initial_a.copy())
    best_a = a.copy()
    best_sc = compute_score(a)
    n = len(a)
    L = 2 * n - 1

    for it in range(1, max_iter + 1):
        if time.time() >= deadline:
            break

        conv = convolution_fft(a, a)
        max_val = np.max(conv)

        # Mask of (approximately) maximal entries
        tol = max_val * 1e-12 + 1e-14
        mask = (np.abs(conv - max_val) <= tol).astype(float)

        mask_rev = mask[::-1]
        conv_rev = convolution_fft(a, mask_rev)
        weighted_grad = 2.0 * conv_rev[L - 1 - np.arange(n)]

        eta = step0 / np.sqrt(it)
        a = a - eta * weighted_grad
        a = project_to_simplex(a)

        cur_sc = compute_score(a)
        if cur_sc < best_sc:
            best_sc = cur_sc
            best_a = a.copy()

        if it % 5_000 == 0:
            print(f"    SubGrad it={it:,d} cur={cur_sc:.6f} best={best_sc:.6f}")

    return best_a


def simulated_annealing(a: np.ndarray,
                        rng: np.random.Generator,
                        deadline: float,
                        max_iters: int = 80_000,
                        init_temp: float = 0.015,
                        decay: float = 0.9999) -> np.ndarray:
    """
    Simple Metropolis‑style local search that moves a small amount of mass
    from one coordinate to another.  The fast surrogate score is used for the
    acceptance test.
    """
    cur = a.copy()
    cur_sc = compute_score(cur)
    best = cur.copy()
    best_sc = cur_sc
    n = len(cur)
    temp = init_temp
    it = 0

    while it < max_iters and time.time() < deadline:
        it += 1
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue

        delta = cur[i] * 0.2 * rng.random()
        if delta <= 1e-15 or cur[i] - delta < 0:
            continue

        cand = cur.copy()
        cand[i] -= delta
        cand[j] += delta

        cand_sc = compute_score(cand)

        if cand_sc < cur_sc or rng.random() < np.exp(-(cand_sc - cur_sc) / max(temp, 1e-12)):
            cur = cand
            cur_sc = cand_sc
            if cur_sc < best_sc:
                best = cur.copy()
                best_sc = cur_sc

        temp = max(temp * decay, 1e-8)

        if it % 5_000 == 0:
            print(f"    SA it={it:,d} cur={cur_sc:.6f} best={best_sc:.6f}")

    return best

def propose_candidate(seed: int = 43,
                     budget_s: int = 1000,
                     **kwargs) -> List[float]:
    """
    Search for a non‑negative step sequence with a low ``evaluate_sequence``
    value.  The routine blends a large random exploration of parametric
    families with an intensive local optimisation (Adam → sub‑gradient →
    simulated‑annealing).  All work is performed on vectors that are
    already normalised to sum = 1, hence the fast surrogate ``2·n·max(conv)``
    can be used throughout the search.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 1.0   # 1 s safety margin

    best_seq: List[float] = []
    best_val: float = float("inf")

    # ------------------------------------------------------------------
    # 0.  Fallback from a previous run (if present)
    # ------------------------------------------------------------------
    if "height_sequence_1" in globals():
        try:
            fallback = [float(x) for x in globals()["height_sequence_1"]]
            val = evaluate_sequence(fallback)
            print(f"Fallback (height_sequence_1) score = {val:.6f}")
            if val < best_val:
                best_val = val
                best_seq = fallback
        except Exception:
            pass

    # ------------------------------------------------------------------
    # 1.  Wide random exploration – keep the top‑k most promising seeds
    # ------------------------------------------------------------------
    n = 4096                     # larger length → tighter theoretical bound
    top_k = 16
    best_pool: List[Tuple[float, np.ndarray]] = []

    def maybe_insert(score: float, vec: np.ndarray):
        """Insert a candidate into the bounded top‑k heap."""
        if len(best_pool) < top_k:
            best_pool.append((score, vec))
        else:
            worst_idx = max(range(len(best_pool)),
                            key=lambda i: best_pool[i][0])
            if score < best_pool[worst_idx][0]:
                best_pool[worst_idx] = (score, vec)

    # Deterministic starters
    maybe_insert(compute_score(uniform_initial(n)), uniform_initial(n))
    maybe_insert(compute_score(arcsine_initial(n)), arcsine_initial(n))

    # Random sampling loop (time‑controlled)
    max_candidates = 300_000
    cnt = 0
    while time.time() < deadline - 30.0 and cnt < max_candidates:
        cnt += 1
        r = rng.random()
        if r < 0.30:                     # beta family
            alpha = rng.uniform(0.02, 5.0)
            beta = rng.uniform(0.02, 5.0)
            eps = rng.uniform(1e-6, 2.0)
            cand = generate_beta_shape(n, alpha, beta, eps)
        elif r < 0.55:                   # power / generalized arcsine
            p = rng.uniform(0.1, 5.0)
            eps = rng.uniform(0.0, 2.0)
            cand = generate_power_shape(n, p, eps)
        elif r < 0.70:                   # mixture arcsine / uniform
            w = rng.random()
            cand = generate_mixture_shape(n, w)
        elif r < 0.85:                   # Dirichlet‑type (many tiny entries)
            cand = generate_dirichlet_shape(n, rng)
        else:                            # triangular bump (random peak)
            peak = rng.integers(0, n)
            left = np.arange(peak + 1, dtype=float)
            right = np.arange(n - peak - 1, dtype=float)[::-1]
            tri = np.concatenate([left, right])
            cand = tri / tri.sum()

        sc = compute_score(cand)
        maybe_insert(sc, cand)

    # Sort the kept seeds by fitness (best first)
    best_pool.sort(key=lambda x: x[0])
    seed_vectors = [vec for _, vec in best_pool]

    # ------------------------------------------------------------------
    # 2.  Refinement of each promising seed
    # ------------------------------------------------------------------
    for idx, seed_vec in enumerate(seed_vectors, 1):
        if time.time() >= deadline - 5.0:
            break
        print(f"\n--- Refining seed {idx}/{len(seed_vectors)} (score {compute_score(seed_vec):.6f}) ---")

        # ---- Adam on smooth surrogate ----
        remaining = deadline - time.time()
        max_iter_adam = int(min(300_000, remaining * 45))
        print(f"    Adam (smooth), max_iter={max_iter_adam}")
        refined, _ = adam_optimize_smooth(
            initial_a=seed_vec,
            deadline=deadline,
            max_iter=max_iter_adam,
            lr0=0.12,
            lam_start=2.0,
            lam_end=200.0,
            noise_every=3_000,
            noise_scale=2e-5,
        )

        # ---- Exact sub‑gradient descent ----
        if time.time() < deadline - 5.0:
            remaining = deadline - time.time()
            max_iter_sub = int(min(250_000, remaining * 40))
            print(f"    Sub‑gradient descent, max_iter={max_iter_sub}")
            refined = subgradient_descent_exact(
                refined,
                deadline,
                max_iter=max_iter_sub,
                step0=0.04,
            )

        # ---- Simulated annealing polishing ----
        if time.time() < deadline - 5.0:
            remaining = deadline - time.time()
            max_iter_sa = int(min(200_000, remaining * 35))
            print(f"    Simulated annealing, max_iters={max_iter_sa}")
            refined = simulated_annealing(
                refined,
                rng,
                deadline,
                max_iters=max_iter_sa,
                init_temp=0.012,
                decay=0.99995,
            )

        # Official evaluation
        try:
            val = evaluate_sequence(refined.tolist())
            print(f"    Official score = {val:.6f}")
            if val < best_val:
                best_val = val
                best_seq = refined.tolist()
                print(f"    *** New global best: {best_val:.6f} ***")
        except Exception:
            pass

    # ------------------------------------------------------------------
    # 3.  Global polishing of the current best vector (if time permits)
    # ------------------------------------------------------------------
    if best_seq and time.time() < deadline - 3.0:
        a = np.array(best_seq, dtype=float)
        remaining = deadline - time.time()
        sub_iters = int(min(300_000, remaining * 30))
        adam_iters = int(min(250_000, remaining * 30))

        print("\n--- Final global polishing ---")
        a = subgradient_descent_exact(a, deadline, max_iter=sub_iters, step0=0.03)
        a, _ = adam_optimize_smooth(
            a,
            deadline,
            max_iter=adam_iters,
            lr0=0.08,
            lam_start=2.0,
            lam_end=200.0,
            noise_every=2_000,
            noise_scale=1e-5,
        )
        final_val = evaluate_sequence(a.tolist())
        print(f"Final polished official score = {final_val:.6f}")
        if final_val < best_val:
            best_val = final_val
            best_seq = a.tolist()
            print("*** Updated global best after final polish ***")

    # ------------------------------------------------------------------
    # 4.  Guard – ensure a non‑empty return value
    # ------------------------------------------------------------------
    if not best_seq:
        best_seq = uniform_initial(n).tolist()
        best_val = evaluate_sequence(best_seq)

    print(f"\nSearch finished. Best reported score = {best_val:.6f}")
    return best_seq
