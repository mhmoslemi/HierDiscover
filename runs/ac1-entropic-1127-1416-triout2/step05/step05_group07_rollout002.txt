"""
Propose a step‑function (non‑negative list of heights) that minimizes
the official ``evaluate_sequence`` objective.

Algorithm
---------
1.  Choose a length `n` (default 8192, reduced if time is short).
2.  Search a small 2‑parameter family that approximates the discrete
    arcsine density:
        a_i ∝ ((i+δ)·(n‑1‑i+δ)+ε)^(‑p) .
    Parameters   p ∈ [0.4,0.6]   (shape exponent)
                δ ∈ [0,1]     (endpoint shift)
    The family is evaluated with a fast surrogate
    ``compute_score`` (exactly the official objective for a normalised
    vector).  A coarse grid and a brief random local search give a
    good starting vector.
3.  Refine the best vector with a *smooth* mirror‑descent optimiser.
    The non‑smooth max is replaced by a soft‑max with a temperature
    γ that is increased during optimisation.  The gradient of the
    smoothed objective is
        ∇Φ = 2·corr(a , softmax_γ(a*a)),
    which is computed with two FFT‑based convolutions.
    Multiplicative (exponentiated‑gradient) updates keep the vector
    on the probability simplex.
4.  When the time budget is almost exhausted the current best vector
    is evaluated with the official ``evaluate_sequence`` and returned.
    If nothing was found a tiny uniform fallback is used.

The whole routine respects the supplied wall‑clock budget
(`budget_s` seconds, default 1000) and always returns a feasible
non‑negative list.
"""

import time
from typing import List, Tuple

import numpy as np

# ----------------------------------------------------------------------
# Fast helpers (identical to the ones used in the reference solution)
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two >= x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full linear convolution using a real FFT (fast and accurate)."""
    m = len(a) + len(b) - 1
    n_fft = next_pow2(m)
    A = np.fft.rfft(a, n=n_fft)
    B = np.fft.rfft(b, n=n_fft)
    C = A * B
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


def compute_score(a: np.ndarray) -> float:
    """
    Exact surrogate of the official evaluator.
    Assumes ``a`` is a non‑negative vector (no need to normalise here;
    the score is scale‑invariant).
    """
    n = len(a)
    conv = convolution_fft(a, a)
    max_val = float(conv.max())
    s = float(a.sum())
    if s < 1e-12:
        return float("inf")
    return 2.0 * n * max_val / (s * s)


# ----------------------------------------------------------------------
# Parametric arcsine‑like family
# ----------------------------------------------------------------------


def arcsine_family(n: int, p: float, shift: float, eps: float = 1e-12) -> np.ndarray:
    """
    Return a normalised vector of length ``n`` with entries

        a_i ∝ ((i+shift)·(n‑1‑i+shift) + eps)^(‑p) .

    For ``p = 0.5`` and ``shift = 0.5`` this is the discrete arcsine
    density.
    """
    i = np.arange(n, dtype=float)
    denom = (i + shift) * (n - 1 - i + shift) + eps
    a = denom ** (-p)
    a = np.maximum(a, 0.0)
    total = a.sum()
    if total == 0.0:
        a = np.full(n, 1.0 / n)
    else:
        a = a / total
    return a


# ----------------------------------------------------------------------
# Smooth mirror‑descent optimiser (soft‑max of the self‑convolution)
# ----------------------------------------------------------------------


def optimise_smooth(
    init_a: np.ndarray,
    deadline: float,
    max_iter: int = 200_000,
    eta0: float = 0.5,
    gamma_start: float = 5.0,
    gamma_mul: float = 1.001,
) -> Tuple[np.ndarray, float]:
    """
    Entropic mirror‑descent on the smoothed objective

        Φ_γ(a) = (1/γ)·log Σ_k exp(γ·(a*a)_k),

    which converges to the original max when γ → ∞.
    Updates are multiplicative (exponentiated‑gradient) and the vector
    is projected onto the probability simplex after each step.
    Returns the best vector seen (according to the exact surrogate
    ``compute_score``) and its score.
    """
    a = np.maximum(init_a, 1e-30)
    a = a / a.sum()
    best_a = a.copy()
    best_sc = compute_score(a)

    n = len(a)
    gamma = gamma_start

    for it in range(1, max_iter + 1):
        if time.time() >= deadline:
            break

        # self‑convolution (length 2n‑1)
        conv = convolution_fft(a, a)

        # soft‑max weights (numerically stable)
        maxc = conv.max()
        weights = np.exp(gamma * (conv - maxc))
        weights /= weights.sum()                     # length 2n‑1, non‑negative

        # gradient = 2·corr(a , weights)
        # correlation can be obtained as a convolution with the reversed weight vector
        w_rev = weights[::-1]
        corr_full = convolution_fft(a, w_rev)       # length 3n‑2
        start = len(w_rev) - 1                      # = 2n‑2
        grad = 2.0 * corr_full[start - np.arange(n)]

        # step size schedule
        eta = eta0 / np.sqrt(it)

        # multiplicative update and simplex projection
        a = a * np.exp(-eta * grad)

        s = a.sum()
        if s == 0.0:
            a = np.full_like(a, 1.0 / n)
        else:
            a = a / s

        cur_sc = compute_score(a)
        if cur_sc < best_sc:
            best_sc = cur_sc
            best_a = a.copy()

        # slowly increase the soft‑max temperature
        gamma *= gamma_mul

        # occasional progress report
        if it % 2000 == 0:
            print(
                f"    smooth iter={it:,d} cur={cur_sc:.6f} best={best_sc:.6f} gamma={gamma:.2f}"
            )

    return best_a, best_sc

    """
    Search for a low‑scoring step function under the given time budget.
    The method first explores a 2‑parameter arcsine family, then refines
    the best candidate with a smooth mirror‑descent optimiser.
    """
    rng = np.random.default_rng(seed)
    start = time.time()
    deadline = start + budget_s - 2.0   # tiny safety margin

    best_seq: List[float] = []
    best_val: float = float("inf")
    best_vec: np.ndarray = None

    # ------------------------------------------------------------------
    # Optional fallback from a previously discovered construction.
    # ------------------------------------------------------------------
    try:
        fallback = [float(x) for x in height_sequence_1]   # type: ignore
        val = evaluate_sequence(fallback)
        print(f"Fallback (height_sequence_1) score = {val:.6f}")
        if val < best_val:
            best_val = val
            best_seq = fallback
            best_vec = np.array(fallback, dtype=float)
            best_vec = best_vec / best_vec.sum()
    except Exception:
        pass

    # ------------------------------------------------------------------
    # Choose a working length.  Larger n gives a better asymptotic bound
    # but also makes each iteration slower.
    # ------------------------------------------------------------------
    remaining = deadline - time.time()
    if remaining > 600:
        n = 8192
    elif remaining > 300:
        n = 4096
    elif remaining > 120:
        n = 2048
    else:
        n = 1024
    print(f"Working length n = {n}")

    # ------------------------------------------------------------------
    # Stage 1 – coarse grid search in the (p,shift) family.
    # ------------------------------------------------------------------
    p_grid = np.linspace(0.40, 0.60, 21)      # 0.40 … 0.60
    shift_grid = np.linspace(0.0, 1.0, 21)    # 0 … 1
    best_score = float("inf")
    best_params = (0.5, 0.5)

    for p in p_grid:
        for sh in shift_grid:
            if time.time() >= deadline:
                break
            cand = arcsine_family(n, p, sh)
            sc = compute_score(cand)
            if sc < best_score:
                best_score = sc
                best_vec = cand.copy()
                best_params = (p, sh)

    print(
        f"Best params after coarse grid: p={best_params[0]:.3f}, "
        f"shift={best_params[1]:.3f}, surrogate score={best_score:.6f}"
    )

    # ------------------------------------------------------------------
    # Stage 2 – tiny random local search around the best parameters.
    # ------------------------------------------------------------------
    for _ in range(80):
        if time.time() >= deadline:
            break
        p = best_params[0] + rng.normal(0.0, 0.01)
        sh = best_params[1] + rng.normal(0.0, 0.05)
        p = float(np.clip(p, 0.01, 5.0))
        sh = float(np.clip(sh, 0.0, 1.0))
        cand = arcsine_family(n, p, sh)
        sc = compute_score(cand)
        if sc < best_score:
            best_score = sc
            best_vec = cand.copy()
            best_params = (p, sh)

    print(
        f"After local perturbations: p={best_params[0]:.3f}, shift={best_params[1]:.3f}, "
        f"surrogate score={best_score:.6f}"
    )

    # ------------------------------------------------------------------
    # Stage 3 – smooth mirror‑descent refinement.
    # ------------------------------------------------------------------
    if best_vec is not None and time.time() < deadline - 5.0:
        # remaining time roughly determines number of allowed iterations
        time_left = deadline - time.time()
        max_iter = int(min(300_000, max(10_000, time_left * 250)))  # ~250 it/s for n≈8k
        print(f"Starting smooth refinement (max_iter={max_iter})")
        refined, refined_sc = optimise_smooth(
            init_a=best_vec,
            deadline=deadline,
            max_iter=max_iter,
            eta0=0.5,
            gamma_start=5.0,
            gamma_mul=1.001,
        )
        if refined_sc < best_score:
            best_score = refined_sc
            best_vec = refined.copy()
            print(f"  Improvement after smooth descent: surrogate score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Final official evaluation (expensive, called only once).
    # ------------------------------------------------------------------
    if best_vec is not None:
        final_val = evaluate_sequence(best_vec.tolist())
        best_seq = best_vec.tolist()
        best_val = final_val
        print(f"Final official score = {best_val:.6f}")
    else:
        # ultimate safeguard – uniform tiny vector
        best_seq = [1.0] * 256
        best_val = evaluate_sequence(best_seq)
        print("Fallback to uniform vector (should never happen).")

    print(f"\nSearch finished. Best reported score = {best_val:.6f}")
    return best_seq
