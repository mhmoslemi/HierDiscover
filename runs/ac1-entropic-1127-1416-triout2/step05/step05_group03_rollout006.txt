"""
Hybrid search for a low‑scoring non‑negative step function.

The algorithm works in four stages:

1.  **Broad sampling** – generate many candidates of length `n` using
    beta‑family densities, mixtures of several betas and Dirichlet draws.
    The best candidate (according to the fast surrogate
    `2·n·max(conv(a,a))`) is retained.

2.  **Long Adam optimisation** – the surrogate is smoothed with a
    temperature‑controlled log‑sum‑exp.  Adam is run with a large number
    of iterations (the loop stops automatically when the global budget
    is exhausted).  The temperature is annealed up to a large value
    (`lam_end≈800`) so that the surrogate becomes almost the exact max.

3.  **Exact sub‑gradient descent** – after Adam we switch to the true
    max‑convolution and perform a projected sub‑gradient descent with a
    simple back‑tracking line‑search.  This guarantees that the official
    score never increases.

4.  **Stochastic polishing** – a cheap Gaussian walk on the simplex
    with a tiny uphill‑move probability finishes the optimisation.

All vectors are kept on the probability simplex (∑a = 1), therefore the
fast surrogate `2·n·max(conv(a,a))` is exactly the value returned by the
official `evaluate_sequence`.  The routine respects the 1000 s budget,
falls back to a uniform vector if nothing is found and prints a short
progress log.
"""

import time
from typing import List

import numpy as np

# ----------------------------------------------------------------------
# Linear‑algebra utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Linear convolution via real FFT (fast for the sizes we use)."""
    m = a.size + b.size - 1
    n_fft = next_pow2(m)
    A = np.fft.rfft(a, n=n_fft)
    B = np.fft.rfft(b, n=n_fft)
    C = A * B
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


def max_convolution(a: np.ndarray) -> float:
    """Maximum entry of the self‑convolution of a."""
    return float(np.max(convolution_fft(a, a)))


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection onto the probability simplex {x ≥ 0, Σx = 1}."""
    n = v.shape[0]
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / n)

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho = np.where(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def compute_score(a: np.ndarray) -> float:
    """
    Fast surrogate: 2·n·max(conv(a,a)).
    For a vector already normalised (Σa = 1) this equals the official score.
    """
    n = a.shape[0]
    return 2.0 * n * max_convolution(a)


# ----------------------------------------------------------------------
# Shape generators (beta families, mixtures, Dirichlet)
# ----------------------------------------------------------------------


def generate_beta_shape(n: int, alpha: float, beta: float, eps: float) -> np.ndarray:
    """Symmetric beta‑family density on {0,…,n‑1} (unnormalised)."""
    i = np.arange(n, dtype=float)
    left = (i + eps) ** (alpha - 1.0)
    right = (n - 1 - i + eps) ** (beta - 1.0)
    base = left * right
    total = base.sum()
    if total <= 0.0 or not np.isfinite(total):
        return np.full(n, 1.0 / n)
    return base / total


def generate_beta_mixture(
    n: int,
    alphas: List[float],
    betas: List[float],
    epsilons: List[float],
    weights: List[float],
) -> np.ndarray:
    """Convex combination of several beta‑family densities."""
    mix = np.zeros(n, dtype=float)
    for a, b, e, w in zip(alphas, betas, epsilons, weights):
        mix += w * generate_beta_shape(n, a, b, e)
    mix_sum = mix.sum()
    if mix_sum == 0.0 or not np.isfinite(mix_sum):
        return np.full(n, 1.0 / n)
    return mix / mix_sum


def arcsine_initial(n: int) -> np.ndarray:
    """Classic arcsine shape (symmetrised)."""
    i = np.arange(n, dtype=float)
    eps = 0.25
    vals = 1.0 / np.sqrt((i + 0.5) * (n - 0.5 - i) + eps)
    vals /= vals.sum()
    return vals


def dirichlet_shape(rng: np.random.Generator, n: int, concentration: float) -> np.ndarray:
    """Sample a point from Dirichlet(concentration,…,concentration)."""
    raw = rng.gamma(concentration, 1.0, size=n)
    if raw.sum() == 0:
        raw = np.ones(n)
    return raw / raw.sum()


# ----------------------------------------------------------------------
# Smooth surrogate (log‑sum‑exp) and its gradient – used by Adam
# ----------------------------------------------------------------------


def smooth_gradient(a: np.ndarray, lam: float) -> tuple[np.ndarray, float]:
    """
    Gradient of the log‑sum‑exp surrogate of max(conv(a,a)).
    Returns (gradient, surrogate value B).
    """
    n = a.shape[0]
    conv = convolution_fft(a, a)          # length 2n‑1
    max_c = np.max(conv)

    # stable log‑sum‑exp
    exp_term = np.exp(lam * (conv - max_c))
    sum_exp = np.sum(exp_term)
    w = exp_term / sum_exp                # soft‑max over convolution entries
    B = max_c + np.log(sum_exp) / lam     # smooth approximation of max

    # derivative of B w.r.t. a:
    w_rev = w[::-1]
    conv_rev = convolution_fft(a, w_rev)      # length 3n‑2
    L = 2 * n - 1
    weighted_grad = 2.0 * conv_rev[L - 1 - np.arange(n)]

    s = np.sum(a)          # should be 1 because of the projection
    term1 = (2.0 * n / (s ** 2)) * weighted_grad
    term2 = -(4.0 * n * B / (s ** 3)) * np.ones_like(a)
    grad = term1 + term2
    return grad, B


def adam_optimize(
    initial_a: np.ndarray,
    time_deadline: float,
    max_iter: int = 1_000_000,
    lr0: float = 0.15,
    lam_start: float = 1.0,
    lam_end: float = 800.0,
    noise_every: int = 800,
    noise_scale: float = 3e-5,
) -> tuple[np.ndarray, float]:
    """
    Adam optimiser on the smooth log‑sum‑exp surrogate.
    The routine respects ``time_deadline`` and returns the best vector
    (projected onto the simplex) together with its fast score.
    """
    a = project_to_simplex(initial_a.astype(float).copy())
    best_a = a.copy()
    best_sc = compute_score(a)

    beta1, beta2 = 0.9, 0.999
    eps_adam = 1e-8
    m = np.zeros_like(a)
    v = np.zeros_like(a)

    n = a.shape[0]

    for it in range(1, max_iter + 1):
        if time.time() >= time_deadline:
            break

        # annealed temperature (sqrt schedule gives a gentle rise)
        t = it / max_iter
        lam = lam_start + (lam_end - lam_start) * (t ** 0.5)

        grad, _ = smooth_gradient(a, lam)

        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)

        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        step = lr0 * m_hat / (np.sqrt(v_hat) + eps_adam)
        a_new = a - step

        if noise_every and (it % noise_every == 0):
            a_new += np.random.randn(n) * noise_scale

        a_new = project_to_simplex(a_new)

        cur_sc = compute_score(a_new)
        if cur_sc < best_sc:
            best_sc = cur_sc
            best_a = a_new.copy()

        a = a_new

        if it % 5000 == 0:
            print(
                f"    Adam iter={it:,d} cur={cur_sc:.6f} best={best_sc:.6f}"
            )

    return best_a, best_sc


# ----------------------------------------------------------------------
# Exact sub‑gradient descent with back‑tracking line‑search
# ----------------------------------------------------------------------


def subgradient_optimize(
    initial_a: np.ndarray,
    time_deadline: float,
    step0: float = 0.25,
    max_iter: int = 500_000,
) -> tuple[np.ndarray, float]:
    """
    Projected sub‑gradient descent on the *true* max‑convolution.
    Returns the best vector seen.
    """
    a = initial_a.copy()
    best_a = a.copy()
    best_sc = compute_score(a)

    n = a.shape[0]

    for it in range(1, max_iter + 1):
        if time.time() >= time_deadline:
            break

        conv = convolution_fft(a, a)
        max_val = float(np.max(conv))
        idx = int(np.argmax(conv))

        # sub‑gradient of the active quadratic term
        i = np.arange(n)
        j = idx - i
        mask = (j >= 0) & (j < n)
        grad = np.zeros_like(a)
        grad[mask] = 2.0 * a[j[mask]]

        step = step0 / np.sqrt(it)   # diminishing step size
        # simple back‑tracking
        for _ in range(12):
            a_new = a - step * grad
            a_new = project_to_simplex(a_new)
            new_val = max_convolution(a_new)
            if new_val < max_val - 1e-14:
                break
            step *= 0.5

        cur_sc = compute_score(a_new)
        if cur_sc < best_sc:
            best_sc = cur_sc
            best_a = a_new.copy()

        a = a_new

        if it % 40000 == 0:
            print(
                f"    Sub‑grad iter={it:,d} cur={cur_sc:.6f} best={best_sc:.6f}"
            )

    return best_a, best_sc


# ----------------------------------------------------------------------
# Stochastic local search (Gaussian perturbations)
# ----------------------------------------------------------------------


def random_local_search(
    initial_a: np.ndarray,
    time_deadline: float,
    init_step: float = 0.015,
    max_iter: int = 500_000,
) -> tuple[np.ndarray, float]:
    """
    Gaussian walk on the simplex with a tiny uphill‑move probability.
    Returns the best vector seen.
    """
    a = initial_a.copy()
    best_a = a.copy()
    best_sc = compute_score(a)

    n = a.shape[0]
    rng = np.random.default_rng()

    for it in range(1, max_iter + 1):
        if time.time() >= time_deadline:
            break

        step = init_step / np.sqrt(it)
        noise = rng.normal(scale=step, size=n)
        a_proposed = a + noise
        a_proposed = project_to_simplex(a_proposed)

        cur_sc = compute_score(a_proposed)

        # Accept if improvement; otherwise accept with tiny probability
        if cur_sc < best_sc or (rng.random() < 1e-4 and cur_sc < 10 * best_sc):
            a = a_proposed
            if cur_sc < best_sc:
                best_sc = cur_sc
                best_a = a_proposed.copy()

        if it % 50000 == 0:
            print(
                f"    Random local iter={it:,d} cur={cur_sc:.6f} best={best_sc:.6f}"
            )

    return best_a, best_sc


# ----------------------------------------------------------------------
# Main entry point required by the competition harness
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs) -> List[float]:
    """
    Search for a non‑negative step sequence with a low official score.
    The routine performs large‑scale sampling, a long Adam run,
    exact sub‑gradient descent and a final stochastic polishing.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    # a tiny safety margin to avoid hard termination
    deadline = start_time + budget_s - 2.0

    best_seq: List[float] = []
    best_val: float = float("inf")

    # ------------------------------------------------------------------
    # 0.  Fallback to a previously known construction if available
    # ------------------------------------------------------------------
    try:
        fallback = [float(x) for x in height_sequence_1]  # type: ignore
        val = evaluate_sequence(fallback)
        print(f"Fallback (height_sequence_1) official score = {val:.6f}")
        if val < best_val:
            best_val = val
            best_seq = fallback
    except Exception:
        pass

    # ------------------------------------------------------------------
    # 1.  Choose a single length (2048) and devote the majority of the budget
    # ------------------------------------------------------------------
    n = 2048
    print(f"\n=== Working with length n = {n} ===")

    # ---------- deterministic starters ----------
    starters = [
        np.full(n, 1.0 / n),
        arcsine_initial(n),
    ]
    local_best = None
    local_best_sc = float("inf")
    for st in starters:
        sc = compute_score(st)
        if sc < local_best_sc:
            local_best_sc = sc
            local_best = st.copy()

    # ---------- random sampling phase ----------
    # give roughly 15 % of the remaining budget to sampling
    sampling_deadline = time.time() + 0.15 * (deadline - time.time())
    print(f"Sampling candidates until {sampling_deadline - start_time:.1f}s from start")
    while time.time() < sampling_deadline:
        # 40 % chance we generate a mixture of up to three betas
        if rng.random() < 0.4:
            k = rng.integers(1, 4)            # number of components 1‑3
            alphas = rng.uniform(0.02, 2.5, size=k)
            betas = rng.uniform(0.02, 2.5, size=k)
            epsilons = rng.uniform(1e-8, max(0.01, 0.02 * n), size=k)
            weights = rng.random(k)
            weights /= weights.sum()
            cand = generate_beta_mixture(n, alphas.tolist(), betas.tolist(),
                                         epsilons.tolist(), weights.tolist())
        else:
            # Dirichlet draw – concentration chosen uniformly in a useful range
            conc = rng.uniform(0.05, 3.0)
            cand = dirichlet_shape(rng, n, conc)

        sc = compute_score(cand)
        if sc < local_best_sc:
            local_best_sc = sc
            local_best = cand.copy()

    # ---------- intensive optimisation ----------
    if local_best is None:
        # this should never happen, but keep the code safe
        local_best = np.full(n, 1.0 / n)

    # Adam (long run)
    print("\n--- Adam optimisation (smooth surrogate) ---")
    a_opt, _ = adam_optimize(
        initial_a=local_best,
        time_deadline=deadline,
        max_iter=1_000_000,
        lr0=0.16,
        lam_start=1.0,
        lam_end=800.0,
        noise_every=700,
        noise_scale=2e-5,
    )

    # Exact sub‑gradient descent
    if time.time() < deadline - 5.0:
        print("\n--- Sub‑gradient descent (exact max) ---")
        a_opt, _ = subgradient_optimize(
            a_opt,
            time_deadline=deadline,
            step0=0.22,
            max_iter=600_000,
        )

    # Final stochastic walk
    if time.time() < deadline - 2.0:
        print("\n--- Random local polishing ---")
        a_opt, _ = random_local_search(
            a_opt,
            time_deadline=deadline,
            init_step=0.012,
            max_iter=600_000,
        )

    # ------------------------------------------------------------------
    # 2.  Official evaluation of the final candidate
    # ------------------------------------------------------------------
    final_val = evaluate_sequence(a_opt.tolist())
    print(f"\nFinal official score for length {n} = {final_val:.6f}")
    if final_val < best_val:
        best_val = final_val
        best_seq = a_opt.tolist()
        print("*** New global best ***")

    # ------------------------------------------------------------------
    # 3.  Guard – guarantee a non‑empty return
    # ------------------------------------------------------------------
    if not best_seq:
        best_seq = [1.0] * 256
        best_val = evaluate_sequence(best_seq)
        print("Fallback to uniform 256‑vector.", best_val)

    print(f"\nSearch finished. Best reported official score = {best_val:.6f}")
    return best_seq
