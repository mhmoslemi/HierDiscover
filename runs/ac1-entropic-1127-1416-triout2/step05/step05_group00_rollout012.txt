"""
Minimising the peak of a self‑convolution.

The evaluation function is homogeneous, so we fix the L¹‑norm of the
candidate to 1 (a probability simplex).  The best possible value is
≈1 (the convolution would be perfectly flat).  Uniform vectors give
2, while distributions of the form (i·(n‑i))^β (β≈‑0.5, i.e. the
arcsine distribution) are known to flatten the sum‑distribution.
We therefore

1. Scan a few lengths and a dense grid of β‑exponents, keeping the
   distribution with the smallest 2·n·max(a⊛a).

2. Starting from the best vector we run a simulated‑annealing style
   local search: repeatedly transfer a random fraction of the mass from
   one index to another.  Moves that lower the objective are always
   accepted; worsening moves are accepted with probability
   exp(‑Δ/temperature).  The temperature linearly decays to zero, the
   step size also shrinks.

The whole search respects the 1000 s budget and finally returns the
best list of non‑negative floats.  The harness will call
`evaluate_sequence` on the returned list.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, n_fft: int) -> np.ndarray:
    """
    Compute the full convolution a ⊛ a using a real FFT.
    `n_fft` must be a power of two ≥ 2·len(a)-1.
    """
    A = np.fft.rfft(a, n=n_fft)
    conv = np.fft.irfft(A * A, n=n_fft)[: 2 * a.size - 1]
    return conv


def compute_score(a: np.ndarray, n_fft: int) -> float:
    """
    Fast replica of `evaluate_sequence` for sum(a)=1.
    Returns 2·n·max(a⊛a) .
    """
    n = a.shape[0]
    conv = convolution_fft(a, n_fft)
    return 2.0 * n * float(conv.max())


def generate_power_distribution(n: int, beta: float) -> np.ndarray:
    """
    Symmetric distribution a_i ∝ ((i+0.5)*(n-0.5-i))**beta,
    normalised to sum 1.
    """
    i = np.arange(n, dtype=float)
    x = (i + 0.5) * (n - 0.5 - i)          # product (distance from edges)
    # raise to the (possibly negative) power
    a = np.power(x, beta, dtype=float)
    a_sum = a.sum()
    if a_sum == 0.0:                     # fallback to uniform if something went wrong
        a = np.full(n, 1.0 / n, dtype=float)
    else:
        a /= a_sum
    return a


def random_simplex_vector(n: int) -> np.ndarray:
    """Draw a random point on the probability simplex (Dirichlet(1))."""
    x = np.random.exponential(scale=1.0, size=n)
    return x / x.sum()


# ----------------------------------------------------------------------
# Simulated‑annealing refinement
# ----------------------------------------------------------------------


def anneal(initial: np.ndarray,
           deadline: float,
           n_fft: int,
           max_iter: int = 2_000_000) -> tuple[np.ndarray, float]:
    """
    Stochastic annealing that conserves the L¹‑norm.
    Returns the best vector seen and its score.
    """
    a = initial.copy()
    cur_score = compute_score(a, n_fft)
    best = a.copy()
    best_score = cur_score

    n = a.size
    it = 0
    start = time.time()

    # start with a relatively warm temperature; it will decay linearly.
    init_temp = max(cur_score * 0.02, 1e-3)
    min_temp = cur_score * 1e-5

    # initial step scale (fraction of the source entry that may be moved)
    step_scale = 0.2

    while time.time() < deadline and it < max_iter:
        # linear temperature schedule
        frac = (deadline - time.time()) / (deadline - start)
        temp = max(min_temp, init_temp * frac)

        # propose a move: take ε from i and give it to j
        i, j = np.random.randint(n, size=2)
        if i == j:
            continue
        eps = step_scale * a[i] * np.random.rand()
        if eps <= 0.0:
            continue

        cand = a.copy()
        cand[i] -= eps
        cand[j] += eps
        # non‑negativity is guaranteed by construction (eps ≤ a[i])
        cand_score = compute_score(cand, n_fft)

        delta = cand_score - cur_score
        if delta < 0.0 or np.random.rand() < np.exp(-delta / temp):
            # accept
            a = cand
            cur_score = cand_score
            if cur_score < best_score:
                best_score = cur_score
                best = a.copy()
                # tighten step size a little when we improve
                step_scale = max(step_scale * 0.98, 1e-4)

        # slowly decay the step scale independent of acceptance
        if it % 10_000 == 0:
            step_scale = max(step_scale * 0.995, 1e-4)

        it += 1

    return best, best_score


# ----------------------------------------------------------------------
# Main entry point required by the evaluation harness
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative sequence that minimises `evaluate_sequence`.
    The routine combines a parametric grid search (power‑law families) and
    a long simulated‑annealing refinement.
    """
    np.random.seed(seed)
    start_time = time.time()
    # leave a small safety margin
    deadline = start_time + budget_s - 2.0

    best_seq = None
    best_val = float("inf")

    # --------------------------------------------------------------
    # 0.  Use any previously known candidate if it exists
    # --------------------------------------------------------------
    try:
        fallback = [float(x) for x in height_sequence_1]   # type: ignore
        fallback_val = evaluate_sequence(fallback)
        print(f"Fallback score = {fallback_val:.6f}")
        if fallback_val < best_val:
            best_val = fallback_val
            best_seq = fallback
    except Exception:
        pass

    # --------------------------------------------------------------
    # 1.  Grid search over lengths and power‑law exponents (β)
    # --------------------------------------------------------------
    lengths = [1024, 2048, 4096]          # larger lengths give more freedom
    betas = np.linspace(-0.90, 0.90, 37)   # fine grid, includes the arcsine case β≈‑0.5

    for n in lengths:
        if time.time() >= deadline - 200.0:
            break
        n_fft = next_pow2(2 * n - 1)

        # uniform vector (β = 0) is a useful baseline
        uniform = np.full(n, 1.0 / n, dtype=float)
        uni_score = compute_score(uniform, n_fft)
        if uni_score < best_val:
            best_val = uni_score
            best_seq = uniform.tolist()
            print(f"Uniform ({n}) score={best_val:.6f}")

        # sweep β
        for beta in betas:
            a = generate_power_distribution(n, beta)
            sc = compute_score(a, n_fft)
            if sc < best_val:
                best_val = sc
                best_seq = a.tolist()
                print(f"β={beta:.2f}, n={n} → score={best_val:.6f}")

        # a few random simplex points for diversity
        for _ in range(12):
            a = random_simplex_vector(n)
            sc = compute_score(a, n_fft)
            if sc < best_val:
                best_val = sc
                best_seq = a.tolist()
                print(f"Random simplex n={n} → score={best_val:.6f}")

    # --------------------------------------------------------------
    # 2.  Simulated‑annealing refinement on the best candidate found
    # --------------------------------------------------------------
    if best_seq is not None and time.time() < deadline - 5.0:
        a0 = np.asarray(best_seq, dtype=float)
        n = a0.shape[0]
        n_fft = next_pow2(2 * n - 1)
        print("\nStarting simulated‑annealing refinement")
        refined_a, refined_score = anneal(
            a0,
            deadline - 2.0,    # leave a couple of seconds for the final call
            n_fft,
        )
        if refined_score < best_val:
            best_val = refined_score
            best_seq = refined_a.tolist()
            print(f"Annealing improved best to {best_val:.6f}")

    # --------------------------------------------------------------
    # 3.  Final safeguard – if nothing succeeded, return a uniform vector
    # --------------------------------------------------------------
    if best_seq is None:
        n = 256
        best_seq = [1.0 / n] * n
        best_val = evaluate_sequence(best_seq)

    # --------------------------------------------------------------
    # 4.  Official evaluation (the harness will also call it)
    # --------------------------------------------------------------
    final_score = evaluate_sequence(best_seq)
    print(f"\nFinished search. Final reported score = {final_score:.6f}")

    # Return a plain Python list of floats.
    return best_seq
