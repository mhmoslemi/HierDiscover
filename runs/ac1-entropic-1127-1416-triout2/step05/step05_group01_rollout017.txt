"""
Propose a low‑score non‑negative step function.

Algorithm
---------
1.  Choose a relatively large length (default 2048) – the
    theoretical lower bound on the score is ≈ 1 for any length,
    and a longer vector gives more freedom.
2.  Perform several random restarts.  Each restart starts from a
    Dirichlet draw (concentration α is randomised) and then runs a
    mirror‑descent style optimiser that minimises a smooth approximation
    of the ∞‑norm of the self‑convolution:
        fβ(a) = (2·n/β)·log∑ₖ exp(β·(a*a)[k]).
    The gradient of the soft‑max is computed via FFT‑based convolution,
    and the iterate is updated with an exponentiated (multiplicative)
    step followed by projection onto the probability simplex.
    The “temperature’’ β is increased with √t, so the optimisation
    gradually focuses on the true maximum.
3.  After the time‑limited optimisation phase a short stochastic
    hill‑climber (swap / mass‑transfer moves) is applied to the best
    vector discovered.
4.  The best sequence found (or a tiny uniform fallback) is returned.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# FFT utilities (linear convolution, real‑valued)
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """
    Linear convolution using a real FFT.
    Returns a vector of length len(a)+len(b)-1.
    """
    m = len(a) + len(b) - 1
    n_fft = next_pow2(m)
    A = np.fft.rfft(a, n=n_fft)
    B = np.fft.rfft(b, n=n_fft)
    C = A * B
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


# ----------------------------------------------------------------------
# Simplex projection (Euclidean)
# ----------------------------------------------------------------------


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the probability simplex:
    {x | x_i ≥ 0, Σ x_i = 1}.
    """
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / v.shape[0])

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, v.shape[0] + 1)
    cond = u - cssv / ind > 0
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


# ----------------------------------------------------------------------
# Fast score (the exact objective when Σa = 1)
# ----------------------------------------------------------------------


def fast_score(a: np.ndarray) -> float:
    """
    Compute 2·n·max(a*a) for a vector whose entries sum to 1.
    This is exactly the value returned by `evaluate_sequence`.
    """
    n = a.shape[0]
    conv = convolution_fft(a, a)
    return 2.0 * n * float(np.max(conv))


# ----------------------------------------------------------------------
# Hill‑climber (swap / mass‑transfer moves)
# ----------------------------------------------------------------------


def hill_climb_permutation(seq, time_deadline, max_steps=200_000):
    """
    Simple stochastic hill‑climber that swaps two entries or moves a
    fraction of mass from one position to another.
    The true objective `evaluate_sequence` is used for acceptance.
    """
    a = np.array(seq, dtype=float)
    n = a.shape[0]

    best_a = a.copy()
    best_score = fast_score(best_a)

    step_scale = 0.1  # fraction of mass moved in a transfer

    while max_steps > 0 and time.time() < time_deadline:
        cand = best_a.copy()

        if np.random.rand() < 0.5:
            # swap two positions
            i, j = np.random.randint(n, size=2)
            cand[i], cand[j] = cand[j], cand[i]
        else:
            # transfer a fraction from i to j
            i, j = np.random.randint(n, size=2)
            if i != j:
                delta = step_scale * best_a[i]
                cand[i] -= delta
                cand[j] += delta
                cand[i] = max(cand[i], 0.0)

        cand = project_to_simplex(cand)
        cur_score = fast_score(cand)

        if cur_score < best_score:
            best_score = cur_score
            best_a = cand
            step_scale = min(step_scale * 1.05, 0.5)   # success → larger steps
        else:
            step_scale *= 0.97                         # failure → smaller steps

        max_steps -= 1

    return best_a.tolist(), best_score


# ----------------------------------------------------------------------
# Smooth (soft‑max) optimisation with exponentiated updates
# ----------------------------------------------------------------------


def smooth_optimize(
    a: np.ndarray,
    time_deadline: float,
    iters: int = 200_000,
    beta0: float = 0.5,
    beta_max: float = 30.0,
    eta0: float = 0.4,
    report_every: int = 5_000,
) -> tuple[np.ndarray, float]:
    """
    Minimise the soft‑max approximation of the max‑convolution.
    Returns the best vector seen and its exact score.
    """
    n = a.shape[0]
    L = next_pow2(3 * n - 2)          # length needed for all FFTs (a, a_rev, w)
    best_a = a.copy()
    best_val = fast_score(best_a)

    for t in range(1, iters + 1):
        if time.time() >= time_deadline - 1.0:
            break

        # ----- a*a ----------------------------------------------------
        A = np.fft.rfft(a, n=L)
        conv_full = np.fft.irfft(A * A, n=L)
        conv = conv_full[: 2 * n - 1]
        max_c = np.max(conv)

        # ----- soft‑max weights ---------------------------------------
        beta = min(beta_max, beta0 * np.sqrt(t))
        # stabilize the exponentials
        exp_terms = np.exp(beta * (conv - max_c))
        w = exp_terms / np.sum(exp_terms)

        # ----- gradient = 2·(w ⋆ a_rev) -------------------------------
        a_rev = a[::-1]
        A_rev = np.fft.rfft(a_rev, n=L)
        W = np.fft.rfft(w, n=L)
        grad_full = np.fft.irfft(A_rev * W, n=L)
        # the needed slice starts at index n‑1
        grad = 2.0 * grad_full[n - 1 : n - 1 + n]

        # ----- exponentiated (mirror) update -------------------------
        eta = eta0 / np.sqrt(t)
        a = a * np.exp(-eta * grad)
        a = project_to_simplex(a)

        # ----- occasional reporting ------------------------------------
        if t % report_every == 0:
            cur_val = fast_score(a)
            if cur_val < best_val:
                best_val = cur_val
                best_a = a.copy()
                print(f"  iter {t:8d} – new best score {cur_val:.6f}")

    # final check
    final_val = fast_score(a)
    if final_val < best_val:
        best_val = final_val
        best_a = a.copy()

    return best_a, best_val
