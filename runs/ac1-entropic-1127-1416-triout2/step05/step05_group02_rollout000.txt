"""
Optimisation of a non‑negative step function for the official
``evaluate_sequence`` bound.

The algorithm consists of four stages:

1️⃣  **Diverse seeding** –  generate many candidate vectors of several
    lengths (512 … 4096) using simple parametric families (arcsine,
    beta‑shape, power‑shape, mixtures, random Dirichlet, point spikes).

2️⃣  **Fast pre‑selection** –  keep the few best candidates according to
    the cheap surrogate ``2·n·max(conv(a,a))`` (the denominator is 1
    after normalising the vector to the probability simplex).

3️⃣  **Refinement** –  apply three complementary optimisation blocks
    *sub‑gradient descent*, *Adam‑style adaptive momentum* and
    *exponentiated‑gradient polishing*.  All updates are followed by a
    Euclidean projection onto the simplex to keep the iterate a valid
    probability distribution.

4️⃣  **Final polishing & return** –  the current global best is polished
    again and finally evaluated with the official ``evaluate_sequence``
    function.  The routine respects the global time budget (default
    1000 s) and always returns a non‑empty sequence.

The search is deliberately stochastic (a fixed RNG seed can be supplied)
and makes heavy use of FFT‑based convolution for speed.
"""

import time
import numpy as np
from typing import List, Tuple

# ----------------------------------------------------------------------
# Low‑level utilities (no closures)

def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """
    Linear convolution via real FFT (length = len(a)+len(b)-1).
    Both inputs are assumed to be 1‑D ``float64`` arrays.
    """
    m = a.shape[0] + b.shape[0] - 1
    n_fft = next_pow2(m)
    A = np.fft.rfft(a, n=n_fft)
    B = np.fft.rfft(b, n=n_fft)
    C = A * B
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


def compute_score(a: np.ndarray) -> float:
    """
    Fast surrogate for the official score.
    The vector is assumed to be normalised to Σa = 1, therefore the
    denominator disappears and the score is simply ``2·n·max(conv(a,a))``.
    """
    n = a.shape[0]
    conv = convolution_fft(a, a)
    return 2.0 * n * float(np.max(conv))


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection onto the probability simplex {x ≥ 0, Σx = 1}.
    Duchi et al., 2008.
    """
    if np.all(v <= 0):
        # uniform fallback – should never happen after a proper update
        return np.full_like(v, 1.0 / v.size)

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho_arr = u - cssv / np.arange(1, v.size + 1)
    rho = np.where(rho_arr > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def subgradient(a: np.ndarray) -> np.ndarray:
    """
    Sub‑gradient of the surrogate ``f(a)=2·n·max(conv(a,a))``.
    Returns a vector of length n.
    """
    n = a.shape[0]
    L = 2 * n - 1
    conv = convolution_fft(a, a)
    max_val = float(np.max(conv))

    # Active set (indices attaining the maximum, tolerant to fp noise)
    tol = max_val * 1e-12 + 1e-14
    mask = (np.abs(conv - max_val) <= tol).astype(float)

    # convolution of a with the reversed mask yields the summed shifts
    mask_rev = mask[::-1]
    conv_rev = convolution_fft(a, mask_rev)
    subgrad = 2.0 * conv_rev[L - 1 - np.arange(n)]
    return subgrad

def arcsine_initial(n: int) -> np.ndarray:
    """Classic arcsine shape: a_i ∝ 1/√((i+½)*(n‑½‑i))."""
    i = np.arange(n, dtype=float)
    eps = 0.25               # regulariser to avoid division by zero
    vals = 1.0 / np.sqrt((i + 0.5) * (n - 0.5 - i) + eps)
    vals /= vals.sum()
    return vals


def generate_beta_shape(n: int, alpha: float, beta: float, eps: float) -> np.ndarray:
    """
    (i+ε)^{α‑1}·(n‑1‑i+ε)^{β‑1} on {0,…,n‑1}, normalised to Σ=1.
    """
    i = np.arange(n, dtype=float)
    left = (i + eps) ** (alpha - 1.0)
    right = (n - 1 - i + eps) ** (beta - 1.0)
    base = left * right
    total = base.sum()
    if total <= 0.0 or not np.isfinite(total):
        return np.full(n, 1.0 / n)
    return base / total


def generate_power_shape(n: int, p: float, eps: float) -> np.ndarray:
    """
    Generalised arcsine: a_i ∝ ((i+½)*(n‑½‑i)+ε)^{‑p}.
    p>0, eps≥0.
    """
    i = np.arange(n, dtype=float)
    denom = ((i + 0.5) * (n - 0.5 - i) + eps) ** p
    vals = 1.0 / denom
    total = vals.sum()
    if total <= 0.0 or not np.isfinite(total):
        return np.full(n, 1.0 / n)
    return vals / total


def generate_mixture_shape(n: int, weight_arc: float) -> np.ndarray:
    """Mixture (weight_arc)·arcsine + (1‑weight_arc)·uniform."""
    arc = arcsine_initial(n)
    uni = np.full(n, 1.0 / n)
    mix = weight_arc * arc + (1.0 - weight_arc) * uni
    mix /= mix.sum()
    return mix


def generate_spike_shape(n: int, mass_left: float, mass_right: float) -> np.ndarray:
    """
    Put ``mass_left`` on the first index and ``mass_right`` on the last
    index, distribute the remainder uniformly.
    """
    remain = max(0.0, 1.0 - mass_left - mass_right)
    uniform = np.full(n, remain / n)
    uniform[0] += mass_left
    uniform[-1] += mass_right
    uniform /= uniform.sum()
    return uniform


def random_dirichlet(n: int, rng: np.random.Generator, concentration: float) -> np.ndarray:
    """
    Sample from Dirichlet(concentration,…,concentration) and return
    a normalised vector.  ``concentration`` may be ≪1 to obtain
    sparse vectors.
    """
    x = rng.gamma(concentration, 1.0, size=n)
    if x.sum() == 0.0:
        x = np.ones(n)
    return x / x.sum()

def subgradient_descent(initial: np.ndarray,
                        deadline: float,
                        max_iter: int = 80_000,
                        step0: float = 0.05) -> np.ndarray:
    """
    Classical sub‑gradient descent with step size ηₜ = step0 / √t.
    Projected onto the simplex after each step.
    """
    a = project_to_simplex(initial.copy())
    for it in range(1, max_iter + 1):
        if time.time() >= deadline:
            break
        sg = subgradient(a)
        eta = step0 / np.sqrt(it)
        a = a - eta * sg
        a = project_to_simplex(a)

        if it % 10_000 == 0:
            cur = compute_score(a)
            print(f"    SubGrad it={it:,d}  surrogate={cur:.6f}")
    return a


def adam_descent(initial: np.ndarray,
                 deadline: float,
                 max_iter: int = 120_000,
                 step0: float = 0.08,
                 beta1: float = 0.9,
                 beta2: float = 0.999,
                 eps: float = 1e-8) -> np.ndarray:
    """
    Adam optimiser applied to the same convex‑concave surrogate.
    The update direction is the sub‑gradient returned by ``subgradient``.
    """
    a = project_to_simplex(initial.copy())
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    t = 0

    for it in range(1, max_iter + 1):
        if time.time() >= deadline:
            break
        t += 1
        sg = subgradient(a)

        m = beta1 * m + (1 - beta1) * sg
        v = beta2 * v + (1 - beta2) * (sg ** 2)

        m_hat = m / (1 - beta1 ** t)
        v_hat = v / (1 - beta2 ** t)

        eta = step0 / np.sqrt(t)
        a = a - eta * m_hat / (np.sqrt(v_hat) + eps)
        a = project_to_simplex(a)

        if it % 15_000 == 0:
            cur = compute_score(a)
            print(f"    Adam it={it:,d}  surrogate={cur:.6f}")
    return a


def eg_refine(initial: np.ndarray,
              deadline: float,
              max_iter: int = 80_000,
              eta0: float = 0.25) -> np.ndarray:
    """
    Mirror‑descent (exponentiated‑gradient) polishing.
    Keeps the iterate on the simplex automatically.
    """
    a = initial.copy()
    for it in range(1, max_iter + 1):
        if time.time() >= deadline:
            break
        sg = subgradient(a)
        eta = eta0 / (np.sqrt(it) + 1.0)
        a = a * np.exp(-eta * sg)
        total = a.sum()
        if total == 0.0:
            a = np.full_like(a, 1.0 / a.size)
        else:
            a = a / total

        if it % 10_000 == 0:
            cur = compute_score(a)
            print(f"    EG it={it:,d}  surrogate={cur:.6f}")
    return a

def hillclimb_power_shape(n: int,
                          init_p: float,
                          init_eps: float,
                          deadline: float,
                          max_iters: int = 300) -> Tuple[np.ndarray, float]:
    """
    Random‑walk hill‑climbing on the (p,eps) parameters of the
    generalised arcsine family, using the fast surrogate.
    """
    p, eps = init_p, init_eps
    best_seq = generate_power_shape(n, p, eps)
    best_sc = compute_score(best_seq)

    for _ in range(max_iters):
        if time.time() >= deadline:
            break
        p_new = max(0.01, p * (1.0 + np.random.uniform(-0.2, 0.2)))
        eps_new = max(0.0, eps + np.random.uniform(-0.15, 0.15))
        cand = generate_power_shape(n, p_new, eps_new)
        sc = compute_score(cand)
        if sc < best_sc:
            p, eps, best_seq, best_sc = p_new, eps_new, cand, sc
    return best_seq, best_sc

            if time.time() >= deadline - 5.0:
                break
            conc = rng.uniform(0.05, 5.0)
            cand = random_dirichlet(n, rng, conc)
            sc = compute_score(cand)
            all_candidates.append((sc, cand))

        # c) random beta‑family
        for _ in range(120):
            if time.time() >= deadline - 5.0:
                break
            a = rng.uniform(0.1, 6.0)
            b = rng.uniform(0.1, 6.0)
            eps = rng.uniform(0.0, max(1.0, 0.02 * n))
            cand = generate_beta_shape(n, a, b, eps)
            sc = compute_score(cand)
            all_candidates.append((sc, cand))

        # d) random power‑shape (generalised arcsine)
        for _ in range(120):
            if time.time() >= deadline - 5.0:
                break
            p = rng.uniform(0.1, 3.0)
            eps = rng.uniform(0.0, 3.0)
            cand = generate_power_shape(n, p, eps)
            sc = compute_score(cand)
            all_candidates.append((sc, cand))

        # e) mixtures of arcsine & uniform
        for _ in range(80):
            if time.time() >= deadline - 5.0:
                break
            w = rng.random()
            cand = generate_mixture_shape(n, w)
            sc = compute_score(cand)
            all_candidates.append((sc, cand))

        # f) spike‑heavy shapes (mass on the two ends)
        for _ in range(60):
            if time.time() >= deadline - 5.0:
                break
            left = rng.uniform(0.0, 0.6)
            right = rng.uniform(0.0, 0.6 - left)  # keep total ≤ 0.6
            cand = generate_spike_shape(n, left, right)
            sc = compute_score(cand)
            all_candidates.append((sc, cand))

        # g) local hill‑climbing on the main two families
        #    (start from a generic point, run a short hill‑climb)
        cand, sc = hillclimb_power_shape(n, 0.5, 0.0, deadline, max_iters=250)
        all_candidates.append((sc, cand))

        cand, sc = hillclimb_beta_shape(n, 0.5, 0.5, 0.0, deadline, max_iters=250)
        all_candidates.append((sc, cand))

    # ------------------------------------------------------------------
    # 2.  Keep a small set of promising seeds (fast surrogate)
    # ------------------------------------------------------------------
    all_candidates.sort(key=lambda x: x[0])
    top_seeds = [vec for _, vec in all_candidates[:12]]
    print(f"\nSelected {len(top_seeds)} top seeds (surrogate range "
          f"{all_candidates[0][0]:.6f} – {all_candidates[-1][0]:.6f})")

    # ------------------------------------------------------------------
    # 3.  Intensive refinement of each seed
    # ------------------------------------------------------------------
    for idx, seed_vec in enumerate(top_seeds, start=1):
        if time.time() >= deadline - 4.0:
            break
        n = seed_vec.shape[0]
        print(f"\nRefining seed {idx}/{len(top_seeds)} (n={n}) …")

        #--- Sub‑gradient descent (bulk of remaining time per seed) ---
        remaining = deadline - time.time()
        max_iter_sub = int(min(70_000, remaining * 30))
        refined = subgradient_descent(seed_vec, deadline,
                                      max_iter=max_iter_sub,
                                      step0=0.18)

        #--- Adam refinement (complementary dynamics) ---
        if time.time() < deadline - 2.0:
            remaining = deadline - time.time()
            max_iter_adam = int(min(90_000, remaining * 30))
            refined = adam_descent(refined, deadline,
                                   max_iter=max_iter_adam,
                                   step0=0.09)

        #--- EG polish (final push) ---
        if time.time() < deadline - 2.0:
            remaining = deadline - time.time()
            max_iter_eg = int(min(70_000, remaining * 30))
            refined = eg_refine(refined, deadline,
                                max_iter=max_iter_eg,
                                eta0=0.22)

        #--- Official evaluation ---
        try:
            val = evaluate_sequence(refined.tolist())
            print(f"  → official score = {val:.6f}")
            if val < best_val:
                best_val = val
                best_seq = refined.tolist()
                print(f"  *** New global best: {best_val:.6f} ***")
        except Exception as e:
            print(f"  Evaluation failed: {e}")

    # ------------------------------------------------------------------
    # 4.  Global polish on the current best (if time permits)
    # ------------------------------------------------------------------
    if best_seq and time.time() < deadline - 5.0:
        print("\nFinal global polish …")
        a = np.array(best_seq, dtype=float)

        remaining = deadline - time.time()
        max_iter_sub = int(min(200_000, remaining * 30))
        a = subgradient_descent(a, deadline,
                                max_iter=max_iter_sub,
                                step0=0.12)

        if time.time() < deadline - 3.0:
            remaining = deadline - time.time()
            max_iter_adam = int(min(150_000, remaining * 30))
            a = adam_descent(a, deadline,
                             max_iter=max_iter_adam,
                             step0=0.07)

        if time.time() < deadline - 2.0:
            remaining = deadline - time.time()
            max_iter_eg = int(min(120_000, remaining * 30))
            a = eg_refine(a, deadline,
                          max_iter=max_iter_eg,
                          eta0=0.18)

        final_val = evaluate_sequence(a.tolist())
        print(f"Final polished official score = {final_val:.6f}")
        if final_val < best_val:
            best_val = final_val
            best_seq = a.tolist()
            print("*** Updated global best after final polish ***")

    # ------------------------------------------------------------------
    # 5.  Guard – ensure a non‑empty return value
    # ------------------------------------------------------------------
    if not best_seq:
        fallback_n = 1024
        best_seq = [1.0 / fallback_n] * fallback_n
        best_val = evaluate_sequence(best_seq)

    print(f"\nSearch finished. Best reported official score = {best_val:.6f}")
    return best_seq
