def run_optimization(n, time_deadline):
    # Step 1 random start
    a = np.random.dirichlet(np.ones(n))
    best_a = a.copy()
    best_score = compute_score(best_a)
    # Subgradient descent
    # compute maximum allowed steps based on time.
    t = 0
    beta = 0.9 # momentum or something
    m = np.zeros(n)
    while time.time() < time_deadline - 10:  # reserve time for hillclimb
        t += 1
        eta = 1.0 / np.sqrt(t)  # use larger base.
        conv = convolution_fft(a, a)
        k = int(np.argmax(conv))
        idx = np.arange(n)
        j = k - idx
        mask = (j >= 0) & (j < n)
        g = np.zeros_like(a)
        g[mask] = 2.0 * a[j[mask]]
        # Update momentum
        m = 0.9 * m + 0.1 * g
        a = a * np.exp(-eta * (g + 0.5 * m))
        a = project_to_simplex(a)
        cur_score = compute_score(a)
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
        # jitter rare
        if t % 4000 == 0:
            jitter = np.random.randn(n) * 1e-5 / np.sqrt(t)
            a = project_to_simplex(a + jitter)
    # Hill climbing with simulated annealing
    cur_a = best_a.copy()
    cur_score = best_score
    T0 = 0.02
    Tmin = 1e-4
    total_hc_time = time_deadline - time.time()
    start_hc = time.time()
    while time.time() < time_deadline:
        # temperature linear cooling based on elapsed fraction
        elapsed = time.time() - start_hc
        frac = elapsed / total_hc_time
        T = max(Tmin, T0 * (1 - frac))
        # propose move
        i, j = np.random.randint(n, size=2)
        if i == j:
            continue
        # amount to transfer: random fraction of a[i]
        delta_frac = np.random.rand() * 0.2  # up to 20% of mass at i
        delta = delta_frac * cur_a[i]
        cand = cur_a.copy()
        cand[i] -= delta
        cand[j] += delta
        # enforce nonnegative
        if cand[i] < 0:
            continue
        cand = project_to_simplex(cand) # small re-normalization to sum 1

        cand_score = compute_score(cand)
        dscore = cand_score - cur_score
        if dscore < 0 or np.random.rand() < np.exp(-dscore / T):
            cur_a = cand
            cur_score = cand_score
            if cur_score < best_score:
                best_score = cur_score
                best_a = cur_a.copy()
    return best_a.tolist(), best_score

def hillclimb_sa(a, time_deadline, T0=0.02):
    n = a.shape[0]
    best = a.copy()
    best_score = compute_score(best)
    cur = a.copy()
    cur_score = best_score
    start = time.time()
    while time.time() < time_deadline:
        elapsed = time.time() - start
        frac = elapsed / (time_deadline - start)
        T = max(1e-5, T0 * (1 - frac))
        # propose random mass transfer
        i, j = np.random.randint(n, size=2)
        if i == j:
            continue
        # choose delta as random proportion of cur[i]
        delta = cur[i] * np.random.rand() * 0.5  # up to half mass of i
        cand = cur.copy()
        cand[i] -= delta
        cand[j] += delta
        # ensure no negative due to rounding
        if cand[i] < 0:
            continue
        cand = cand / cand.sum()
        cand_score = compute_score(cand)
        d = cand_score - cur_score
        if d < 0 or np.random.rand() < np.exp(-d / T):
            cur = cand
            cur_score = cand_score
            if cur_score < best_score:
                best_score = cur_score
                best = cur.copy()
    return best, best_score

"""
Propose a low‑score step function.

The algorithm performs several independent runs (restarts) inside the
overall time budget.  For each run:

1.  Initialise a random probability vector `a` on the simplex
    (Dirichlet draw, length `n` – we use `n = 2048` which gives the
    optimiser more freedom than the previous 1024‑length code).

2.  Apply exponentiated (mirror) sub‑gradient descent on the exact
    objective  
    `score(a) = 2·n·max(conv(a,a))`.  The gradient used is the
    sub‑gradient at the current maximal convolution index, i.e.  
    `g_i = 2·a_{k-i}` (zero when the index is out of range).  The step
    size follows `η = η0/√t` with a relatively large start `η0 = 1.0`.
    The iterates are projected back onto the probability simplex simply
    by normalising the positive entries.

3.  A simulated‑annealing hill‑climber refines the best vector from the
    sub‑gradient phase.  Moves consist of transferring a random fraction
    of the mass from one position to another (or swapping two values);
    acceptance follows the Metropolis rule with a linearly cooling
    temperature.

All runs share the same global time budget (default 1000 s).  The best
vector found over all runs is returned; if for any reason no vector is
produced a tiny uniform distribution is used as a safe fallback.

The code deliberately avoids any file or network I/O and uses only
`numpy` and the supplied `evaluate_sequence` for the final score.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# FFT‑based linear convolution (real input – fast)
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Return the smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Linear convolution a*b using a real FFT (fast)."""
    m = len(a) + len(b) - 1
    n_fft = _next_pow2(m)
    A = np.fft.rfft(a, n=n_fft)
    B = np.fft.rfft(b, n=n_fft)
    C = A * B
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


# ----------------------------------------------------------------------
# Objective (fast replica of evaluate_sequence for a probability vector)
# ----------------------------------------------------------------------


def compute_score(a: np.ndarray) -> float:
    """
    Return exactly the value that `evaluate_sequence` would produce for a
    vector whose entries sum to 1.  The function is `2·n·max(conv(a,a))`.
    """
    n = a.shape[0]
    conv = convolution_fft(a, a)
    return 2.0 * n * float(np.max(conv))


# ----------------------------------------------------------------------
# Simple projection onto the probability simplex (keep non‑negative and
# normalise).  Multiplicative updates never produce negative numbers,
# therefore a plain normalisation suffices, but the helper also clips
# any accidental negatives.
# ----------------------------------------------------------------------


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Clip negatives, then renormalise to sum‑1."""
    v = np.maximum(v, 0.0)
    s = v.sum()
    if s <= 0.0:
        # fallback to the uniform distribution
        return np.full_like(v, 1.0 / v.shape[0])
    return v / s


# ----------------------------------------------------------------------
# Exponentiated sub‑gradient descent on the exact objective
# ----------------------------------------------------------------------


def exp_subgrad_optimize(
    a0: np.ndarray,
    deadline: float,
    eta0: float = 1.0,
) -> tuple[np.ndarray, float]:
    """
    Mirror (exponentiated) sub‑gradient descent on
    f(a)=2·n·max(conv(a,a)).
    Returns the best vector found and its score.
    """
    a = a0.astype(float).copy()
    best_a = a.copy()
    best_score = compute_score(a)
    n = len(a)

    t = 0
    # momentum is optional – a tiny exponential moving average
    momentum = np.zeros_like(a)
    beta = 0.9

    while time.time() < deadline:
        t += 1
        # current convolution and the index of the maximal entry
        conv = convolution_fft(a, a)
        k = int(np.argmax(conv))

        # sub‑gradient of the max‑convolution (ordered pairs)
        idx = np.arange(n)
        j = k - idx
        mask = (j >= 0) & (j < n)
        g = np.zeros_like(a)
        g[mask] = 2.0 * a[j[mask]]

        # simple exponential moving average of gradients (helps stability)
        momentum = beta * momentum + (1.0 - beta) * g

        # diminishing step size
        eta = eta0 / np.sqrt(t)

        # multiplicative update (mirror descent with KL divergence)
        a = a * np.exp(-eta * (g + 0.5 * momentum))

        # renormalise onto the simplex
        a = project_to_simplex(a)

        cur_score = compute_score(a)
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()

        # tiny random jitter every few thousand iterations to break plateaus
        if t % 4000 == 0:
            jitter = np.random.randn(n) * (1e-5 / np.sqrt(t))
            a = project_to_simplex(a + jitter)

    return best_a, best_score


# ----------------------------------------------------------------------
# Simulated‑annealing hill‑climber (mass‑transfer moves)
# ----------------------------------------------------------------------


def hillclimb_simulated_annealing(
    a0: np.ndarray,
    deadline: float,
    T0: float = 0.02,
    Tmin: float = 1e-5,
    move_prob_swap: float = 0.2,
) -> tuple[np.ndarray, float]:
    """
    Stochastic hill‑climber that accepts uphill moves with a temperature.
    The move set consists of:
        * swapping two entries,
        * transferring a random fraction of mass from i to j.
    Returns the best vector encountered and its score.
    """
    cur = a0.copy()
    cur_score = compute_score(cur)
    best = cur.copy()
    best_score = cur_score

    n = len(cur)
    start = time.time()
    total = max(1.0, deadline - start)

    while time.time() < deadline:
        # linearly decreasing temperature
        elapsed = time.time() - start
        T = max(Tmin, T0 * (1.0 - elapsed / total))

        cand = cur.copy()
        if np.random.rand() < move_prob_swap:
            # simple swap
            i, j = np.random.randint(n, size=2)
            cand[i], cand[j] = cand[j], cand[i]
        else:
            # transfer a random fraction of mass from i to j
            i, j = np.random.randint(n, size=2)
            if i == j:
                continue
            frac = np.random.rand() * 0.5          # up to 50 % of source mass
            delta = frac * cand[i]
            cand[i] -= delta
            cand[j] += delta

        cand = project_to_simplex(cand)
        cand_score = compute_score(cand)

        delta_score = cand_score - cur_score
        if delta_score < 0 or np.random.rand() < np.exp(-delta_score / T):
            cur = cand
            cur_score = cand_score
            if cur_score < best_score:
                best_score = cur_score
                best = cur.copy()

    return best, best_score


# ----------------------------------------------------------------------
# One complete optimisation run (random start → sub‑gradient → SA)
# ----------------------------------------------------------------------


def optimise_one_run(
    n: int,
    run_deadline: float,
) -> tuple[list[float], float]:
    """
    Perform a full optimisation for a given length `n` within the supplied
    time limit.  Returns a Python list (the candidate) and its score.
    """
    # ---------- random Dirichlet start ----------
    a = np.random.dirichlet(np.ones(n))

    # ---------- sub‑gradient phase ----------
    # reserve a small fraction for the final hill‑climb
    subgrad_deadline = run_deadline - 5.0
    best_a, best_score = exp_subgrad_optimize(a, subgrad_deadline)

    # ---------- simulated‑annealing polishing ----------
    final_deadline = run_deadline
    polished_a, polished_score = hillclimb_simulated_annealing(
        best_a, final_deadline
    )
    return polished_a.tolist(), polished_score
