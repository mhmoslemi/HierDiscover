"""
Step‑function optimizer for the ``evaluate_sequence`` challenge.

The score is invariant under a positive scaling of the coefficients,
so we work on the probability simplex (non‑negative entries that sum to 1).
The algorithm consists of three stages:

1️⃣  **Smoothed exponentiated gradient descent** – we minimise a
    smooth approximation of the max‑convolution using a soft‑max
    (temperature β).  The update is multiplicative
    ``a ← a·exp(‑η·∇)`` and we re‑project onto the simplex after each step.
    β is slowly increased, so the optimisation transitions from a
    global smooth landscape to the true non‑smooth objective.

2️⃣  **Optional linear‑programming pivot** – the classic “LP‑pivot”
    (solve max∑b s.t. f∗b≤M, b≥0) is used only if time permits,
    providing a large constructive move that keeps the peak bound M.

3️⃣  **Stochastic hill‑climber** – a short, cheap polish that swaps
    entries or transfers a fraction of mass between positions while
    always re‑projecting onto the simplex.

All work respects a 1000 s budget; the best sequence found is returned.
"""

import time
import numpy as np
from typing import List, Tuple

# ----------------------------------------------------------------------
# FFT‑based convolution utilities (fast, linear convolution)
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """
    Linear convolution a * b using a real FFT (fast).  Works for arbitrary
    lengths of ``a`` and ``b``.
    """
    m = a.shape[0] + b.shape[0] - 1
    n_fft = _next_pow2(m)
    A = np.fft.rfft(a, n=n_fft)
    B = np.fft.rfft(b, n=n_fft)
    C = A * B
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


def compute_score(a: np.ndarray) -> float:
    """
    Fast replica of ``evaluate_sequence`` for a vector that already sums to 1.
    Returns ``2 * n * max(conv)`` where ``conv = a * a``.
    """
    n = a.shape[0]
    conv = convolution_fft(a, a)
    return 2.0 * n * float(np.max(conv))


# ----------------------------------------------------------------------
# Euclidean projection onto the probability simplex
# ----------------------------------------------------------------------


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the probability simplex
    (non‑negative entries that sum to 1).
    """
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / v.shape[0])

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, v.shape[0] + 1)
    cond = u - cssv / ind > 0
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    # Numerical safety: enforce exact sum 1
    return w / w.sum()


# ----------------------------------------------------------------------
# Gradient of the soft‑max approximated max‑convolution
# ----------------------------------------------------------------------


def weighted_gradient(a: np.ndarray, w: np.ndarray) -> np.ndarray:
    """
    Gradient of  J(a)=log∑exp(β·(a∗a))  w.r.t. ``a``.
    ``w`` must be a probability vector of length ``2·n‑1`` (the soft‑max
    weights for the convolution entries).  The returned gradient has
    shape ``(n,)``.
    """
    n = a.shape[0]
    rev_w = w[::-1]                     # reverse for correlation
    conv = convolution_fft(a, rev_w)   # length = 3·n‑2
    # slice indices n‑1 … 2·n‑2  (inclusive) → exactly n entries
    slice_part = conv[n - 1 : 2 * n - 1]
    grad = 2.0 * slice_part[::-1]       # flip to obtain correct order
    return grad


# ----------------------------------------------------------------------
# Smoothed exponentiated gradient descent
# ----------------------------------------------------------------------


def exp_smooth_optimize(
    init_a: np.ndarray,
    time_deadline: float,
    eta0: float = 0.5,
    beta0: float = 0.5,
    beta_max: float = 50.0,
) -> Tuple[np.ndarray, float]:
    """
    Minimises 2·n·max(a*a) using a soft‑max surrogate.
    ``eta0`` controls the initial learning rate, ``beta0`` the initial
    temperature of the soft‑max and ``beta_max`` the final temperature.
    Returns the best vector encountered and its (exact) score.
    """
    a = init_a.astype(float).copy()
    n = a.shape[0]

    best_a = a.copy()
    best_score = compute_score(a)

    t = 0
    beta = beta0

    while True:
        if time.time() >= time_deadline:
            break
        t += 1

        conv = convolution_fft(a, a)          # length 2·n‑1
        max_conv = conv.max()
        cur_score = 2.0 * n * float(max_conv)

        # keep the best ever seen (exact objective)
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()

        # ----- soft‑max weights -----
        # (β·(conv – max_conv)) ≤ 0, so we can safely clip
        exp_arg = beta * (conv - max_conv)    # max element maps to 0
        exp_arg = np.clip(exp_arg, -50.0, 0.0)
        w = np.exp(exp_arg)
        w_sum = w.sum()
        if w_sum == 0.0:
            w = np.full_like(conv, 1.0 / conv.shape[0])
        else:
            w = w / w_sum

        # ----- gradient -----
        grad = weighted_gradient(a, w)

        # ----- multiplicative update (exponentiated gradient) -----
        eta = eta0 / np.sqrt(t)
        # Clip the exponent to avoid under‑/overflow
        step = np.exp(-np.clip(eta * grad, -20.0, 20.0))
        a = a * step

        # ----- projection onto simplex -----
        a = project_to_simplex(a)

        # ----- temperature schedule (increase β) -----
        beta = min(beta * 1.0015, beta_max)

        # ----- occasional jitter to escape plateaus -----
        if t % 5000 == 0:
            jitter = np.random.randn(n) * (eta0 * 1e-4 / np.sqrt(t))
            a += jitter
            a = project_to_simplex(a)

    return best_a, best_score


# ----------------------------------------------------------------------
# Stochastic hill‑climber (swap / mass‑transfer moves)
# ----------------------------------------------------------------------


def hill_climb_permutation(
    seq: List[float], time_deadline: float, max_steps: int = 200_000
) -> Tuple[List[float], float]:
    """
    Very cheap hill‑climber that performs either a swap of two entries
    or a transfer of a fraction of mass between two indices.
    The exact objective is used for acceptance.
    """
    a = np.array(seq, dtype=float)
    n = a.shape[0]

    best_a = a.copy()
    best_score = compute_score(best_a)

    step_scale = 0.1  # initial fraction of mass transferred

    steps = 0
    while steps < max_steps and time.time() < time_deadline:
        move_type = np.random.rand()
        cand = best_a.copy()

        if move_type < 0.5:
            # swap two positions
            i, j = np.random.randint(n, size=2)
            cand[i], cand[j] = cand[j], cand[i]
        else:
            # transfer a fraction from i to j
            i, j = np.random.randint(n, size=2)
            if i != j:
                delta = step_scale * cand[i]
                cand[i] -= delta
                cand[j] += delta
                cand[i] = max(cand[i], 0.0)

        cand = project_to_simplex(cand)
        cur_score = compute_score(cand)

        if cur_score < best_score:
            best_score = cur_score
            best_a = cand
            step_scale = min(step_scale * 1.05, 0.5)   # success → larger steps
        else:
            step_scale *= 0.97                         # failure → smaller steps

        steps += 1

    return best_a.tolist(), best_score
