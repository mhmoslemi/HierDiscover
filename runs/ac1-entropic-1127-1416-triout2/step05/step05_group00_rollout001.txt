def exp_subgradient_optimize(init_a, deadline, max_iter=200000, eta0=0.5):

    a = init_a.copy()
    best_a = a.copy()
    best_score = compute_score(a)
    n = len(a)
    for t in range(1, max_iter+1):
        if time.time()>deadline: break
        conv = convolution_fft(a, a)
        k = int(np.argmax(conv))
        # gradient
        g = np.zeros(n)
        # for i where j = k - i in [0,n-1], g[i] = 2*a[j]
        # so loop j values direct:
        # for j in range(max(0, k-n+1), min(n, k+1)):
        #    i = k - j
        #    g[i] = 2*a[j]
        j_start = max(0, k - n + 1)
        j_end = min(k, n-1)
        # compute indices array
        j_idx = np.arange(j_start, j_end+1)
        i_idx = k - j_idx
        g[i_idx] = 2*a[j_idx]

        # step size
        eta = eta0 / np.sqrt(t)
        # exponentiated update
        a = a * np.exp(-eta * g)
        # add small epsilon to avoid zeros?
        a = np.maximum(a, 1e-16)
        a = a / np.sum(a)  # project to simplex (normalization)
        # evaluate
        score = compute_score(a)
        if score < best_score:
            best_score = score
            best_a = a.copy()
        # occasional random perturbation
        if t % 5000 == 0:
            # jitter small random noise
            a = a * (1 + 0.01*np.random.randn(n))
            a = np.maximum(a, 0)
            a /= np.sum(a)
    return best_a, best_score

def solve_lp_for_b(a, M):
    n = len(a)
    # Build sparse A_ub
    rows = []
    cols = []
    data = []
    for k in range(2*n-1):
        i_start = max(0, k - (n-1))
        i_end = min(k, n-1)
        # vectorized:
        i_range = np.arange(i_start, i_end+1)
        j_range = k - i_range
        rows.extend([k] * len(i_range))
        cols.extend(j_range.tolist())
        data.extend(a[i_range].tolist())
    A_ub = scipy.sparse.csr_matrix((data, (rows, cols)), shape=(2*n-1, n))
    b_ub = np.full(2*n-1, M)
    bounds = [(0, None)] * n
    # Use method='highs' which handles sparse.
    res = linprog(c=-np.ones(n), A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
    if res.success:
        return np.maximum(res.x, 0.0)   # ensure non-negative
    else:
        return None

def exp_subgradient_optimize(init_a, deadline, max_iter=200000, eta0=0.4):
    a = init_a.copy()
    n = len(a)
    best_a = a.copy()
    best_score = compute_score(a)
    t = 1
    while t <= max_iter and time.time() < deadline:
        # Calculate conv
        conv = convolution_fft(a, a)
        k = int(np.argmax(conv))
        # compute gradient g using vectorized approach.
        # Choose i and j mapping: As earlier with j indices.
        # We can compute i_arr and j_arr vectorized using slicing.
        # Determine j_start/j_end
        j_start = max(0, k - n + 1)
        j_end = min(k, n - 1)
        # j range
        j_idx = np.arange(j_start, j_end + 1)
        i_idx = k - j_idx
        g = np.zeros_like(a)
        g[i_idx] = 2.0 * a[j_idx]

        # step size
        eta = eta0 / np.sqrt(t)
        # exponentiated update
        a = a * np.exp(-eta * g)
        # avoid underflow: add epsilon
        eps = 1e-16
        a = np.maximum(a, eps)
        a = a / np.sum(a)
        # compute score
        cur_score = compute_score(a)
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
        # occasional perturbation
        if t % 5000 == 0:
            # small random jitter
            a = a * (1.0 + 0.01 * np.random.randn(n))
            a = np.maximum(a, 0)
            a = a / np.sum(a)
        t += 1
    return best_a, best_score

def solve_lp_for_b(a, M):
    n = len(a)
    rows = 2*n - 1
    # Use dense matrix if n <= 2048 else return None.
    if n > 2048:
        return None
    A = np.zeros((rows, n), dtype=float)
    for j in range(n):
        A[j:j+n, j] = a
    b_ub = np.full(rows, M)
    c = -np.ones(n)
    res = linprog(c, A_ub=A, b_ub=b_ub, bounds=[(0, None)]*n, method='highs')
    if res.success:
        b = res.x
        # ensure non-negative
        b = np.maximum(b, 0.0)
        return b
    else:
        return None

def hill_climb(a, deadline, max_steps=200000):
    n = len(a)
    best_a = a.copy()
    best_score = compute_score(best_a)
    step_scale = 0.05
    step_counter = 0
    while step_counter < max_steps and time.time() < deadline:
        move_type = np.random.rand()
        cand = best_a.copy()
        i, j = np.random.randint(n, size=2)
        if move_type < 0.33:
            # swap
            cand[i], cand[j] = cand[j], cand[i]
        elif move_type < 0.66:
            # transfer
            if i != j:
                delta = step_scale * best_a[i]
                if delta > 0:
                    cand[i] -= delta
                    cand[j] += delta
        else:
            # random shuffle of two values
            # select a subset of size 3 and permute
            idxs = np.random.choice(n, size=3, replace=False)
            vals = cand[idxs]
            np.random.shuffle(vals)
            cand[idxs] = vals
        # ensure non-negativity, renorm
        cand = np.maximum(cand, 0)
        cand_sum = np.sum(cand)
        if cand_sum == 0:
            continue
        cand = cand / cand_sum
        cur_score = compute_score(cand)
        if cur_score < best_score:
            best_score = cur_score
            best_a = cand
            # increase step_scale small
            step_scale = min(step_scale * 1.05, 0.5)
        else:
            step_scale *= 0.95
        step_counter += 1
    return best_a, best_score

"""
Propose a low‑score step function.

Algorithm
---------

* **Random sparse initialisation** – generate many candidates of varying
  length and support density.  We draw the non‑zero values from a Dirichlet
  distribution (uniform weights are also tried) and keep the best one.

* **Mirror‑descent (exponentiated sub‑gradient) optimisation** – the objective
  is 2·n·max(a∗a) (the sum of the sequence is normalised to 1).  The
  sub‑gradient of the maximal convolution entry is
  g_i = 2·a_{k−i} (where k is the index attaining the maximum).  Using a
  KL‑mirror step (a_i ← a_i·exp(−η·g_i)) automatically respects the
  simplex constraint.  A decreasing step‑size η_t = η₀/√t is used and the
  best iterate is retained.

* **Linear‑programming refinement** – for moderate lengths (≤2048) we solve
  a small LP with ``scipy.optimize.linprog``:
      maximise   1ᵀb   subject to   b ≥ 0,
                 (a ⊛ b) ≤ M .
  The solution b is mixed back into a (small mixing factor) and the process
  is repeated a few times.

* **Targeted hill‑climbing** – stochastic moves (swap, mass transfer,
  random permutation) are accepted only when they improve the exact score.
  Additionally, on some iterations we explicitly move a fraction of mass
  away from the pair of indices that currently contribute most to the peak
  convolution value.

The whole search respects the 1000 s budget (a small safety margin is kept)
and returns a plain list of non‑negative floats ready for ``evaluate_sequence``.
"""

def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full convolution using a real FFT (fast)."""
    m = len(a) + len(b) - 1
    n_fft = next_pow2(m)
    A = np.fft.rfft(a, n=n_fft)
    B = np.fft.rfft(b, n=n_fft)
    C = A * B
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


def compute_score(a: np.ndarray) -> float:
    """
    Fast replica of ``evaluate_sequence`` under the assumption sum(a) = 1.
    Returns 2·n·max(a∗a).
    """
    n = a.shape[0]
    conv = convolution_fft(a, a)
    return 2.0 * n * float(np.max(conv))


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the probability simplex
    (non‑negative entries that sum to 1).
    """
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / v.shape[0])

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, v.shape[0] + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.where(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

    weights = np.random.dirichlet(np.ones(k))

    a = np.zeros(n, dtype=float)
    a[idx] = weights
    return a


# ----------------------------------------------------------------------
# Mirror‑descent (exponentiated sub‑gradient) optimisation
# ----------------------------------------------------------------------


def exp_subgradient_optimize(init_a: np.ndarray,
                             deadline: float,
                             max_iter: int = 200_000,
                             eta0: float = 0.40) -> tuple[np.ndarray, float]:
    """
    Mirror descent on the exact objective 2·n·max(a∗a).
    Uses exponentiated updates so the simplex constraint is satisfied
    without an explicit projection.
    """
    a = np.maximum(init_a, 1e-16)          # avoid exact zeros
    a /= np.sum(a)

    n = a.shape[0]
    best_a = a.copy()
    best_score = compute_score(a)

    for t in range(1, max_iter + 1):
        if time.time() >= deadline:
            break

        conv = convolution_fft(a, a)
        k = int(np.argmax(conv))

        # sub‑gradient of the maximal entry
        g = np.zeros_like(a)
        j_start = max(0, k - n + 1)
        j_end = min(k, n - 1)
        if j_start <= j_end:
            j_idx = np.arange(j_start, j_end + 1)
            i_idx = k - j_idx
            g[i_idx] = 2.0 * a[j_idx]

        eta = eta0 / np.sqrt(t)
        a = a * np.exp(-eta * g)

        # protect against underflow and renormalise
        a = np.maximum(a, 1e-16)
        a /= np.sum(a)

        cur_score = 2.0 * n * float(np.max(conv))   # same as compute_score(a)
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()

        # occasional small random jitter to escape flat plateaux
        if t % 5000 == 0:
            a = a * (1.0 + 0.01 * np.random.randn(n))
            a = np.maximum(a, 0.0)
            if np.sum(a) == 0:
                a = np.full(n, 1.0 / n)
            else:
                a /= np.sum(a)

    return best_a, best_score


# ----------------------------------------------------------------------
# Linear‑programming refinement (scipy.linprog)
# ----------------------------------------------------------------------


def solve_lp_for_b(a: np.ndarray, M: float):
    """
    Solve   maximise   sum(b)   subject to   b ≥ 0,
                           (a ∗ b) ≤ M .
    Returns the optimal b (array) or None on failure.
    """
    if not SCIPY_AVAILABLE:
        return None

    n = a.shape[0]
    # Restrict to moderate sizes to keep memory usage reasonable
    if n > 2048:
        return None

    rows = 2 * n - 1
    # Build the full convolution matrix A where (A·b)_k = (a∗b)_k .
    # Each column j contributes the vector a shifted by j.
    A = np.zeros((rows, n), dtype=float)
    for j in range(n):
        A[j:j + n, j] = a

    b_ub = np.full(rows, M, dtype=float)
    c = -np.ones(n, dtype=float)                     # maximise sum(b) == minimise -sum(b)
    bounds = [(0, None)] * n

    try:
        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds,
                      method='highs', options={'max_iter': 10000})
    except Exception:   # pragma: no cover
        return None

    if res.success and res.x is not None:
        b = np.maximum(res.x, 0.0)
        return b
    return None


def lp_refine(a: np.ndarray,
              deadline: float,
              max_iter: int = 5,
              mix: float = 0.20) -> tuple[np.ndarray, float]:
    """
    Repeatedly solve the LP described above, mix the solution back into ``a``
    and keep the best iterate.
    """
    best_a = a.copy()
    best_score = compute_score(best_a)

    for it in range(max_iter):
        if time.time() >= deadline:
            break

        conv = convolution_fft(best_a, best_a)
        M = float(np.max(conv))
        b = solve_lp_for_b(best_a, M)
        if b is None:
            break

        # normalise b to the simplex
        b_sum = np.sum(b)
        if b_sum <= 0:
            break
        b /= b_sum

        # small convex combination and re‑normalise
        anew = (1.0 - mix) * best_a + mix * b
        anew = np.maximum(anew, 0.0)
        anew /= np.sum(anew)

        sc = compute_score(anew)
        if sc < best_score:
            best_score, best_a = sc, anew
            print(f"  LP refinement iter {it}: score={best_score:.6f}")
        else:
            # no improvement – stop early
            break

    return best_a, best_score


# ----------------------------------------------------------------------
# Hill‑climbing polish (including a targeted move)
# ----------------------------------------------------------------------


def hill_climb(a: np.ndarray,
               deadline: float,
               max_steps: int = 300_000) -> tuple[np.ndarray, float]:
    """
    Stochastic hill‑climb that accepts only improving moves.
    Three move types are used:
        1) swap two entries,
        2) transfer a fraction of mass,
        3) a targeted move that reduces the dominant convolution contribution.
    """
    n = a.shape[0]
    best_a = a.copy()
    best_score = compute_score(best_a)

    step_scale = 0.10
    steps = 0

    while steps < max_steps and time.time() < deadline:
        cand = best_a.copy()
        move_type = np.random.rand()

        if move_type < 0.33:
            # simple swap
            i, j = np.random.randint(n, size=2)
            cand[i], cand[j] = cand[j], cand[i]

        elif move_type < 0.66:
            # transfer a fraction from i to j
            i, j = np.random.randint(n, size=2)
            if i != j:
                delta = step_scale * best_a[i]
                if delta > 0:
                    cand[i] -= delta
                    cand[j] += delta
                    cand[i] = max(cand[i], 0.0)

        else:
            # targeted move based on the current peak of the convolution
            conv = convolution_fft(best_a, best_a)
            k = int(np.argmax(conv))
            # indices that actually realise the peak
            js = np.arange(max(0, k - n + 1), min(k, n - 1) + 1)
            is_ = k - js
            prod = best_a[is_] * best_a[js]
            # source: where the product is largest
            src = is_[np.argmax(prod)]
            # destination: a position with (near) minimal weight
            dst = np.argmin(best_a)
            if src != dst:
                delta = step_scale * best_a[src]
                cand[src] -= delta
                cand[dst] += delta
                cand[src] = max(cand[src], 0.0)

        # re‑normalise onto the simplex
        cand = np.maximum(cand, 0.0)
        s = np.sum(cand)
        if s == 0:
            continue
        cand /= s

        cur_score = compute_score(cand)

        if cur_score < best_score:
            best_score = cur_score
            best_a = cand
            step_scale = min(step_scale * 1.05, 0.5)  # accelerate a little
        else:
            step_scale *= 0.95  # become more conservative

        steps += 1

    return best_a, best_score


# ----------------------------------------------------------------------
# Main entry point required by the evaluation harness
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative sequence that minimises ``evaluate_sequence``.
    The procedure follows the outline described in the module docstring.
    """
    np.random.seed(seed)
    start = time.time()
    # keep a small safety margin for the harness
    deadline = start + budget_s - 5.0

    best_seq = None
    best_score = float("inf")

    # ------------------------------------------------------------------
    # Optional fallback to a previously known construction
    # ------------------------------------------------------------------
    try:
        fallback = [float(x) for x in height_sequence_1]   # type: ignore
        fallback_val = evaluate_sequence(fallback)
        print(f"Fallback score = {fallback_val:.6f}")
        if fallback_val < best_score:
            best_score = fallback_val
            best_seq = fallback
    except Exception:
        pass

    # ------------------------------------------------------------------
    # Phase 1 – random sparse initialisation
    # ------------------------------------------------------------------
    lengths = [256, 512, 1024, 2048, 4096]

    print("\nPhase 1 – random search")
    while time.time() < deadline - 150.0:          # reserve time for later phases
        n = int(np.random.choice(lengths))
        a = random_candidate(n)

        cur_score = compute_score(a)
        if cur_score < best_score:
            best_score = cur_score
            best_seq = a.tolist()
            print(f"  Random improvement: n={n}, score={best_score:.6f}")

        # also try the same support but uniform weights (often competitive)
        if np.random.rand() < 0.15:
            supp = np.where(a > 0)[0]
            if len(supp) >= 2:
                a_uniform = np.zeros_like(a)
                a_uniform[supp] = 1.0 / len(supp)
                cur_score = compute_score(a_uniform)
                if cur_score < best_score:
                    best_score = cur_score
                    best_seq = a_uniform.tolist()
                    print(f"  Uniform‑support improvement: n={n}, score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 2 – mirror‑descent (exponentiated sub‑gradient) optimisation
    # ------------------------------------------------------------------
    if best_seq is not None:
        print("\nPhase 2 – exponentiated sub‑gradient optimisation")
        a0 = np.array(best_seq, dtype=float)
        a0 = np.maximum(a0, 1e-16)
        a0 /= np.sum(a0)

        sub_deadline = deadline - 60.0          # leave some time for LP & hill‑climb
        a_opt, score_opt = exp_subgradient_optimize(
            a0,
            deadline=sub_deadline,
            max_iter=200_000,
            eta0=0.40,
        )
        if score_opt < best_score:
            best_score = score_opt
            best_seq = a_opt.tolist()
            print(f"  Sub‑gradient improved best to {best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 3 – LP refinement (if applicable)
    # ------------------------------------------------------------------
    if best_seq is not None and time.time() < deadline - 30.0:
        n = len(best_seq)
        if SCIPY_AVAILABLE and n <= 2048:
            print("\nPhase 3 – linear‑programming refinement")
            a_lp, lp_score = lp_refine(
                np.array(best_seq, dtype=float),
                deadline=deadline - 30.0,
                max_iter=6,
                mix=0.20,
            )
            if lp_score < best_score:
                best_score = lp_score
                best_seq = a_lp.tolist()
                print(f"  LP refinement improved best to {best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 4 – final hill‑climb polish
    # ------------------------------------------------------------------
    if best_seq is not None and time.time() < deadline - 5.0:
        print("\nPhase 4 – hill‑climb polishing")
        a_hc, hc_score = hill_climb(
            np.array(best_seq, dtype=float),
            deadline=deadline,
            max_steps=300_000,
        )
        if hc_score < best_score:
            best_score = hc_score
            best_seq = a_hc.tolist()
            print(f"  Hill‑climb improved best to {best_score:.6f}")

    # ------------------------------------------------------------------
    # Safeguard: fall back to a uniform vector if nothing was found
    # ------------------------------------------------------------------
    if best_seq is None:
        n = 256
        best_seq = [1.0 / n] * n
        best_score = evaluate_sequence(best_seq)

    # Final official evaluation (uses the provided `evaluate_sequence`)
    final_score = evaluate_sequence(best_seq)
    print(f"\nFinished search. Final reported score = {final_score:.6f}")

    # The harness expects a plain Python list of floats.
    return best_seq
