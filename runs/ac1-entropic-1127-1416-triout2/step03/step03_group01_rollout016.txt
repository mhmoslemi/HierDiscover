def lp_iterative_optimize(f, max_iters, t, time_deadline):
    n = len(f)
    best_f = f.copy()
    best_sc = compute_score(f)
    f_cur = f.copy()
    for it in range(max_iters):
        if time.time() >= time_deadline:
            break
        # compute M = max conv self
        conv_self = convolution_fft(f_cur, f_cur)
        M = conv_self.max()
        # build A_ub
        rows = 2*n - 1
        A = np.zeros((rows, n), dtype=np.float64)
        for i in range(n):
            A[i:i+n, i] = f_cur  # reuse f_cur each iteration (maybe not stable)
        b_ub = np.full(rows, M, dtype=np.float64)
        # solve LP
        res = linprog(-np.ones(n), A_ub=A, b_ub=b_ub, bounds=[(0, None)]*n, method='highs', options={'presolve': True})
        if not res.success:
            print("LP failed:", res.message)
            break
        g = res.x
        sum_g = g.sum()
        if sum_g <= 1e-12:
            break
        g_norm = g / sum_g
        # update f
        f_new = (1 - t) * f_cur + t * g_norm
        # symmetrize
        f_new = (f_new + f_new[::-1]) / 2
        # renorm to sum=1 (should already be)
        s = f_new.sum()
        if s <= 0:
            break
        f_new /= s
        # compute score
        sc = compute_score(f_new)
        if sc < best_sc:
            best_sc = sc
            best_f = f_new.copy()
        # convergence check: if improvement < some epsilon, break
        if best_sc - sc < 1e-8:
            # minimal improvement
            pass
        f_cur = f_new
    return best_f, best_sc

def convolution_fft(a,b):
    # compute linear convolution using FFT
    m = len(a) + len(b) - 1
    n_fft = next_pow2(m)
    A = np.fft.rfft(a, n_fft)
    B = np.fft.rfft(b, n_fft)
    C = A * B
    conv = np.fft.irfft(C, n_fft)[:m]
    return conv

def max_subgradient_optimize(initial_a, time_deadline, max_iters=50000, step0=0.1, step_decay=5e-5):
    a = project_to_simplex(initial_a.copy())
    best_a = a.copy()
    best_sc = compute_score(a)
    n = len(a)

    for it in range(max_iters):
        if time.time() >= time_deadline:
            break
        conv = convolution_fft(a, a)
        k = int(np.argmax(conv))
        # Build gradient
        # compute valid i range.
        i_min = max(0, k - (n - 1))
        i_max = min(n - 1, k)
        i_vals = np.arange(i_min, i_max + 1)
        j_vals = k - i_vals
        g = np.zeros_like(a)
        g[j_vals] = 2 * a[i_vals]  # assignment may have multiple assignments? Actually each j occurs unique.
        # step size
        step = step0 / (1.0 + step_decay * it)
        a_new = a - step * g
        a_new = project_to_simplex(a_new)
        # symmetrize
        a_new = (a_new + a_new[::-1]) / 2
        a_new = project_to_simplex(a_new)
        sc = compute_score(a_new)
        if sc < best_sc:
            best_sc = sc
            best_a = a_new.copy()
        a = a_new
    return best_a, best_sc

def propose_candidate(seed=123, budget_s=1000, **kwargs):
    """
    Search for a non‑negative step sequence minimizing ``evaluate_sequence``.
    Combines random sampling of symmetric Beta shapes, a long Adam optimisation
    on a smooth surrogate of the objective and a final hard‑max subgradient
    refinement.  The routine respects the time budget and always returns a
    feasible sequence.
    """
    import time, numpy as np
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 2.0

    # container for best candidates (fast score)
    candidates = []  # list of tuples (fast_score, vector)

    # deterministic starters
    lengths = [256, 512, 1024, 2048, 4096]

    for n in lengths:
        # ensure we have time
        if time.time() >= deadline - 10.0:
            break
        print(f"\n=== Length {n} exploration ===")
        # deterministic:
        starters = [np.full(n, 1.0 / n), arcsine_initial(n)]
        for st in starters:
            sc = compute_score(st)
            candidates.append((sc, st))

        # random beta shapes
        max_samples = 2000  # may adjust based on remaining time
        for s in range(max_samples):
            if time.time() >= deadline - 10.0:
                break
            # sample exponents
            alpha = rng.uniform(0.01, 3.0)  # symmetric -> beta = alpha
            eps = rng.uniform(1e-4, max(0.1, 0.02 * n))
            cand = generate_beta_shape(n, alpha, alpha, eps)
            sc = compute_score(cand)
            candidates.append((sc, cand))

    # keep only top K candidates (lowest fast scores)
    K = 5
    candidates.sort(key=lambda x: x[0])
    top_candidates = candidates[:K]
    print(f"\nSelected {len(top_candidates)} top candidates for Adam refinement.")

    best_seq = None
    best_val = float('inf')

    for idx, (sc, a) in enumerate(top_candidates):
        if time.time() >= deadline - 10.0:
            break
        n = len(a)
        print(f"\n--- Adam refinement of candidate #{idx+1} (n={n})  initial fast score={sc:.6f} ---")
        # allocate remaining time for this candidate
        remaining = deadline - time.time()
        # allocate a fraction (e.g., 30% of remaining) for Adam
        max_iter = int(min(200_000, remaining * 30))
        refined_a, refined_sc = adam_optimize(
            initial_a=a,
            time_deadline=deadline,
            max_iter=max_iter,
            lr0=0.15,
            lam_start=5.0,
            lam_end=120.0,
            noise_every=1500,
            noise_scale=3e-5,
        )

"""
Propose a step‐function sequence with a very low ``evaluate_sequence`` value.

Algorithm (all steps respect the global 1000 s budget)

1. **Exploratory sampling**  
   For several lengths (256 … 4096) we generate symmetric beta‑family
   densities

        a_i ∝ (i+ε)^{α‑1}·(n‑1‑i+ε)^{α‑1}          (α>0)

   together with the classic arcsine shape and the uniform vector.
   The fast surrogate ``2·n·max(a*a)`` (exact when Σa=1) is used to keep
   only the 5 best candidates overall.

2. **Long Adam optimisation** – each of the selected candidates is run
   through a smooth approximation of the objective.  The max is replaced
   by a log‑sum‑exp with an annealed temperature λ from 5 up to 120.
   Adam (β₁=0.9, β₂=0.999) with projection onto the probability simplex
   (and optional symmetrisation) is executed for as many iterations as
   the remaining time permits (up to 200 k).  The best vector from this
   phase is kept.

3. **Hard‑max subgradient polishing** – starting from the Adam result we
   minimise the true max‑convolution by a projected sub‑gradient method:
   the gradient of the active convolution index is ``g_j = 2·a_{k‑j}``
   (zero when the index is out of range).  A decreasing step size
   (≈0.08 / (1+10⁻⁴·t)) and symmetrisation are used; the loop runs until
   the time budget is exhausted.

4. **Final evaluation** – the best vector found is fed to the official
   ``evaluate_sequence`` function exactly once.  A fallback to a uniform
   256‑vector (or to the previously supplied ``height_sequence_1``) is
   provided so that the function always returns a feasible answer.

All helper functions are top‑level, there are no closures or lambdas,
and the implementation uses only ``numpy`` (no filesystem/network IO).
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Low‑level utilities (no closures)
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """
    Linear convolution via real FFT.
    Returns an array of length len(a)+len(b)-1.
    """
    m = len(a) + len(b) - 1
    n_fft = next_pow2(m)
    A = np.fft.rfft(a, n=n_fft)
    B = np.fft.rfft(b, n=n_fft)
    C = A * B
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of v onto the probability simplex
    {x ≥ 0, Σx = 1}.  Duchi et al., 2008.
    """
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / v.size)

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho = np.where(u - cssv / np.arange(1, v.size + 1) > 0)[0]
    if len(rho) == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def compute_score(a: np.ndarray) -> float:
    """
    Fast exact surrogate for the official score.
    The input is assumed to be normalised to Σa = 1, thus the objective
    equals 2·n·max(a*a).
    """
    n = len(a)
    conv = convolution_fft(a, a)
    max_val = float(np.max(conv))
    return 2.0 * n * max_val


def arcsine_initial(n: int) -> np.ndarray:
    """Classic arcsine shape: a_i ∝ 1/√((i+½)*(n‑½‑i))."""
    i = np.arange(n, dtype=float)
    eps = 0.25
    vals = 1.0 / np.sqrt((i + 0.5) * (n - 0.5 - i) + eps)
    vals /= vals.sum()
    return vals


def generate_beta_shape(n: int, alpha: float, beta: float, eps: float) -> np.ndarray:
    """
    Symmetric Beta‑family density on {0,…,n‑1}:
        a_i ∝ (i+ε)^{α‑1}·(n‑1‑i+ε)^{β‑1}.
    The returned vector is normalised to Σa = 1.
    """
    i = np.arange(n, dtype=float)
    left = (i + eps) ** (alpha - 1.0)
    right = (n - 1 - i + eps) ** (beta - 1.0)
    base = left * right
    total = base.sum()
    if total <= 0.0 or not np.isfinite(total):
        return np.full(n, 1.0 / n)
    return base / total


# ----------------------------------------------------------------------
# Smooth Adam optimiser (log‑sum‑exp surrogate)
# ----------------------------------------------------------------------


def smooth_gradient(a: np.ndarray, lam: float):
    """
    Gradient of the smooth max‑convolution surrogate for temperature λ.
    Returns (grad, smoothed_max).
    """
    n = len(a)
    conv = convolution_fft(a, a)               # length 2n‑1
    max_c = np.max(conv)

    # stable soft‑max over the convolution values
    exp_term = np.exp(lam * (conv - max_c))
    sum_exp = np.sum(exp_term)
    w = exp_term / sum_exp                     # weights, sum to 1
    B = max_c + np.log(sum_exp) / lam          # smooth approximation of max

    # gradient of Σ_k w_k (a*a)_k  w.r.t. a:
    w_rev = w[::-1]
    conv_rev = convolution_fft(a, w_rev)       # length 3n‑2, but we only need part
    L = 2 * n - 1
    weighted_grad = 2.0 * conv_rev[L - 1 - np.arange(n)]

    s = np.sum(a)
    term1 = (2.0 * n / (s ** 2)) * weighted_grad
    term2 = -(4.0 * n * B / (s ** 3)) * np.ones_like(a)
    grad = term1 + term2
    return grad, B


def adam_optimize(initial_a: np.ndarray,
                  time_deadline: float,
                  max_iter: int = 100_000,
                  lr0: float = 0.2,
                  lam_start: float = 5.0,
                  lam_end: float = 120.0,
                  noise_every: int = 1500,
                  noise_scale: float = 3e-5) -> tuple[np.ndarray, float]:
    """
    Adam optimiser on the smooth surrogate.
    Returns the best vector (projected on the simplex) and its fast score.
    """
    a = project_to_simplex(initial_a.astype(float).copy())
    best_a = a.copy()
    best_sc = compute_score(a)

    beta1, beta2 = 0.9, 0.999
    eps = 1e-8
    m = np.zeros_like(a)
    v = np.zeros_like(a)

    for it in range(1, max_iter + 1):
        if time.time() >= time_deadline:
            break

        # linear schedule for λ
        lam = lam_start + (lam_end - lam_start) * min(1.0, it / max_iter)

        grad, _ = smooth_gradient(a, lam)

        # Adam moments
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)

        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        step = lr0 * m_hat / (np.sqrt(v_hat) + eps)
        a_new = a - step

        if noise_every and (it % noise_every == 0):
            a_new += np.random.randn(*a.shape) * noise_scale

        a_new = project_to_simplex(a_new)

        # Optional symmetrisation (the problem is symmetric)
        a_new = (a_new + a_new[::-1]) / 2.0
        a_new = project_to_simplex(a_new)

        cur_sc = compute_score(a_new)
        if cur_sc < best_sc:
            best_sc = cur_sc
            best_a = a_new.copy()

        a = a_new

        if it % 5_000 == 0:
            print(f"    Adam iter={it:,d}  cur={cur_sc:.6f}  best={best_sc:.6f}")

    return best_a, best_sc


# ----------------------------------------------------------------------
# Hard‑max sub‑gradient optimiser (directly minimises max‑convolution)
# ----------------------------------------------------------------------


def max_subgradient_optimize(initial_a: np.ndarray,
                             time_deadline: float,
                             max_iters: int = 80_000,
                             step0: float = 0.08,
                             step_decay: float = 1e-4) -> tuple[np.ndarray, float]:
    """
    Projected sub‑gradient descent on the true max‑convolution objective.
    At each step the gradient is taken only from the active convolution
    index (the one attaining the maximum).  The iterate is projected onto
    the simplex and symmetrised.
    """
    a = project_to_simplex(initial_a.copy())
    best_a = a.copy()
    best_sc = compute_score(a)
    n = len(a)

    for it in range(max_iters):
        if time.time() >= time_deadline:
            break

        conv = convolution_fft(a, a)
        k = int(np.argmax(conv))

        # Sub‑gradient of max(conv) w.r.t. a
        i_min = max(0, k - (n - 1))
        i_max = min(n - 1, k)
        i_vals = np.arange(i_min, i_max + 1, dtype=int)
        j_vals = k - i_vals
        g = np.zeros_like(a)
        g[j_vals] = 2.0 * a[i_vals]

        step = step0 / (1.0 + step_decay * it)
        a_new = a - step * g
        a_new = project_to_simplex(a_new)

        # Keep the distribution symmetric
        a_new = (a_new + a_new[::-1]) / 2.0
        a_new = project_to_simplex(a_new)

        sc = compute_score(a_new)
        if sc < best_sc:
            best_sc = sc
            best_a = a_new.copy()

        a = a_new

        if (it + 1) % 10_000 == 0:
            print(f"    Subgrad iter={it+1:,d}  cur={sc:.6f}  best={best_sc:.6f}")

    return best_a, best_sc


# ----------------------------------------------------------------------
# Main entry point required by the competition harness
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Search for a non‑negative step sequence with a low ``evaluate_sequence``
    value.  The routine performs random sampling of symmetric beta shapes,
    a long Adam refinement on a smooth surrogate of the objective and a
    final hard‑max sub‑gradient polishing.  The best vector found within the
    allotted time budget is returned.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 2.0   # safety margin

    # ------------------------------------------------------------------
    # 0.  Try the historic fallback if it exists
    # ------------------------------------------------------------------
    best_seq: list[float] = []
    best_val: float = float("inf")
    try:
        fallback = [float(x) for x in height_sequence_1]  # type: ignore
        val = evaluate_sequence(fallback)
        print(f"Fallback (height_sequence_1) score = {val:.6f}")
        if val < best_val:
            best_val = val
            best_seq = fallback
    except Exception:
        pass

    # ------------------------------------------------------------------
    # 1.  Exploratory sampling (keep only the top few candidates)
    # ------------------------------------------------------------------
    lengths = [256, 512, 1024, 2048, 4096]
    all_candidates: list[tuple[float, np.ndarray]] = []

    for n in lengths:
        if time.time() >= deadline - 10.0:
            break
        print(f"\n=== Exploring length n = {n} ===")

        # deterministic starters
        starters = [np.full(n, 1.0 / n), arcsine_initial(n)]
        for st in starters:
            sc = compute_score(st)
            all_candidates.append((sc, st))

        # random symmetric beta-family sampling
        max_samples = 2000
        for _ in range(max_samples):
            if time.time() >= deadline - 10.0:
                break
            alpha = rng.uniform(0.01, 3.0)          # symmetric => beta = alpha
            eps = rng.uniform(1e-4, max(0.1, 0.02 * n))
            cand = generate_beta_shape(n, alpha, alpha, eps)
            sc = compute_score(cand)
            all_candidates.append((sc, cand))

    # Keep only the K best candidates for the expensive polishing stages
    K = 5
    all_candidates.sort(key=lambda x: x[0])
    top_candidates = all_candidates[:K]
    print(f"\nSelected {len(top_candidates)} top candidates for Adam refinement.")

    # ------------------------------------------------------------------
    # 2.  Adam refinement of the elite candidates
    # ------------------------------------------------------------------
    for idx, (init_sc, a0) in enumerate(top_candidates):
        if time.time() >= deadline - 10.0:
            break
        n = len(a0)
        print(f"\n--- Adam refinement of candidate #{idx+1} (n={n})  init_score={init_sc:.6f} ---")
        remaining = deadline - time.time()
        max_iter = int(min(200_000, remaining * 30))
        refined_a, refined_sc = adam_optimize(
            initial_a=a0,
            time_deadline=deadline,
            max_iter=max_iter,
            lr0=0.15,
            lam_start=5.0,
            lam_end=120.0,
            noise_every=1500,
            noise_scale=3e-5,
        )
        print(f"    After Adam: fast score = {refined_sc:.6f}")

        # ------------------------------------------------------------------
        # 3.  Hard‑max sub‑gradient polishing
        # ------------------------------------------------------------------
        if time.time() < deadline - 5.0:
            remaining = deadline - time.time()
            max_sub_iters = int(min(80_000, remaining * 20))
            polished_a, polished_sc = max_subgradient_optimize(
                initial_a=refined_a,
                time_deadline=deadline,
                max_iters=max_sub_iters,
                step0=0.08,
                step_decay=1e-4,
            )
            print(f"    After sub‑gradient: fast score = {polished_sc:.6f}")
        else:
            polished_a = refined_a
            polished_sc = refined_sc

        # ------------------------------------------------------------------
        # 4.  Official evaluation (once per candidate)
        # ------------------------------------------------------------------
        try:
            val = evaluate_sequence(polished_a.tolist())
            print(f"    Official score = {val:.6f}")
        except Exception as e:
            print(f"    Evaluation failed: {e}")
            val = float("inf")

        if val < best_val:
            best_val = val
            best_seq = polished_a.tolist()
            print(f"    *** New global best: {best_val:.6f} (n={n}) ***")

    # ------------------------------------------------------------------
    # 5.  Guard – guarantee a return value
    # ------------------------------------------------------------------
    if not best_seq:
        # Very crude safe fallback (uniform 256‑vector)
        n = 256
        best_seq = [1.0 / n] * n
        best_val = evaluate_sequence(best_seq)
        print(f"Fallback uniform sequence score = {best_val:.6f}")

    print(f"\nFinished search. Best reported score = {best_val:.6f}")
    return best_seq
