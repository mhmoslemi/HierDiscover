        # ---- Adam update ----
        m = beta1 * m + (1.0 - beta1) * grad_z
        v = beta2 * v + (1.0 - beta2) * (grad_z ** 2)
        m_hat = m / (1.0 - beta1 ** (iter_cnt + 1))
        v_hat = v / (1.0 - beta2 ** (iter_cnt + 1))
        z = z - lr * m_hat / (np.sqrt(v_hat) + eps_adam)

        # ---- occasional true‑objective evaluation ----
        if iter_cnt % 1000 == 0:
            cur_score = compute_objective(a)
            if cur_score < best_score - 1e-12:
                best_score = cur_score
                best_a = a.copy()
                print(f"Iter {iter_cnt:7d}: new best {best_score:.6f} (smooth {smooth_obj:.6f})")
                if best_score <= 1.503:
                    print("Target reached during Adam phase")
                    break

        # ---- gradually sharpen the surrogate ----
        if iter_cnt % 2000 == 0 and beta_smooth < 5000:
            beta_smooth *= beta_growth

        iter_cnt += 1

    # --------------------------------------------------------------
    # 4. Polishing phase – short projected sub‑gradient descent
    # --------------------------------------------------------------
    remaining = deadline - time.time()
    if remaining > 5.0 and best_score > 1.503:
        a = best_a.copy()
        prev = a.copy()
        eta = 0.8
        mu = 0.3
        it_polish = 0
        stagn = 0
        max_polish = 200000
        while time.time() < deadline and it_polish < max_polish:
            # current convolution
            fft_a = np.fft.rfft(a, n=2 * n - 1)
            conv = np.fft.irfft(fft_a * fft_a, n=2 * n - 1)
            conv = np.maximum(conv, 0.0)
            max_b = conv.max()
            cur = 2.0 * n * max_b
            if cur < best_score - 1e-12:
                best_score = cur
                best_a = a.copy()
                print(f"Polish iter {it_polish:7d}: new best {best_score:.6f}")
                stagn = 0
            else:
                stagn += 1

            # active set (constraints that are tight)
            eps = max_b * 1e-6
            active = np.where(conv >= max_b - eps)[0]
            if active.size == 0:
                active = np.array([conv.argmax()], dtype=int)

            # sub‑gradient of the active max entry
            g = subgradient_active(a, active)

            # step‑size schedule
            step = eta / np.sqrt(it_polish + 1)

            # Nesterov‑style momentum + projection onto simplex
            a_next = a - step * g + mu * (a - prev)
            a_next = simplex_projection(a_next)

            prev, a = a, a_next
            it_polish += 1

            # occasional random perturbation to escape plateaus
            if stagn >= 5000:
                sigma = max(0.01, 0.3 / np.sqrt(it_polish + 1))
                noise = sigma * np.random.randn(n) * a
                a = a + noise
                a = np.clip(a, 0.0, None)
                a = a / a.sum()
                prev = a.copy()
                stagn = 0
                print(f"Polish iter {it_polish}: perturbation applied")

    # --------------------------------------------------------------
    # 5. Return the best sequence found
    # --------------------------------------------------------------
    print(f"Finished optimisation – best objective {best_score:.6f}")
    return [float(v) for v in best_a.tolist()]
