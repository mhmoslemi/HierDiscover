    for iteration in range(n_iters):
        if time.time() >= deadline: break
        # compute M
        M = compute_max_conv(f)

def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    if np.all(v <= 0):
        n = v.shape[0]
        return np.full_like(v, 1.0 / n, dtype=float)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho = np.where(u - cssv / np.arange(1, v.shape[0] + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full convolution via real FFT (fast for large arrays)."""
    m = len(a) + len(b) - 1
    n_fft = next_pow2(m)
    A = np.fft.rfft(a, n=n_fft)
    B = np.fft.rfft(b, n=n_fft)
    C = A * B
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv

    Score identical to `evaluate_sequence` when ∑a = 1.
    Returns 2·n·max(a*a) (the denominator is 1).
    """
    a = np.asarray(a, dtype=float)
    n = len(a)
    conv = convolution_fft(a, a)
    max_val = float(np.max(conv))
    if max_val < 0:
        max_val = 0.0
    return 2.0 * n * max_val


# ----------------------------------------------------------------------
# Families of candidate step‑functions
# ----------------------------------------------------------------------


def arcsine_initial(n: int) -> np.ndarray:
    """Arcsine‑shaped start: a_i ∝ 1 / sqrt((i+0.5)(n‑0.5‑i))."""
    i = np.arange(n, dtype=float)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - 0.5 - i))
    return vals / np.sum(vals)


def beta_sequence(n: int, alpha: float, beta: float | None = None) -> np.ndarray:
    """Beta‑type coefficients a_i ∝ (i+½)^α·(n‑½‑i)^β."""
    if beta is None:
        beta = alpha
    i = np.arange(n, dtype=float)
    left = np.power(i + 0.5, alpha)
    right = np.power(n - 0.5 - i, beta)
    seq = left * right
    seq = np.clip(seq, 0.0, 1e12)        # guard against overflow
    total = np.sum(seq)
    if total == 0.0:
        seq = np.full(n, 1.0 / n)
    else:
        seq = seq / total
    return seq


def exponential_sequence(n: int, lam: float) -> np.ndarray:
    """a_i ∝ exp(-lam·|i - (n‑1)/2|)."""
    i = np.arange(n, dtype=float)
    center = (n - 1) / 2.0
    seq = np.exp(-lam * np.abs(i - center))
    return seq / np.sum(seq)


def upsample_sequence(seq: np.ndarray, new_n: int) -> np.ndarray:
    """
    Linear interpolation of a probability vector to a larger size.
    The result is re‑normalised to sum to 1.
    """
    old_n = len(seq)
    old_x = (np.arange(old_n) + 0.5) / old_n
    new_x = (np.arange(new_n) + 0.5) / new_n
    new_vals = np.interp(new_x, old_x, seq)
    return new_vals / np.sum(new_vals)


# ----------------------------------------------------------------------
# Smoothed‑max gradient descent (corrected)
# ----------------------------------------------------------------------


def smoothed_gradient(a: np.ndarray, alpha: float):
    """
    Gradient of the smooth log‑sum‑exp approximation of the max:

        fα(a) = (1/α)·log Σ_k exp(α·(a*a)_k)

    Returns (grad, conv) where conv = a*a (full length 2n‑1).
    """
    n = len(a)
    conv = convolution_fft(a, a)                     # length 2n‑1
    maxc = np.max(conv)
    # soft‑max weights (numerically stable)
    exp_term = np.exp(alpha * (conv - maxc))
    w = exp_term / np.sum(exp_term)                  # shape (2n‑1,)

    # grad_i = 2 * Σ_k w_k * a_{k-i}  (indices outside [0,n) give 0)
    a_rev = a[::-1]
    conv_rev = convolution_fft(w, a_rev)             # length 3n‑2
    start = n - 1
    grad = 2.0 * conv_rev[start:start + n]           # vector of length n
    return grad, conv


def improve_sequence_smoothed(
    initial_a: np.ndarray,
    time_deadline: float,
    max_iter: int = 150_000,
    eta0: float = 1.2,
    alpha_start: float = 5.0,
    alpha_end: float = 400.0,
    noise_every: int = 500,
    noise_scale: float = 1e-5,
):
    """
    Projected gradient descent on the smoothed objective.
    Returns the best vector encountered (as a list) and its true score.
    """
    a = project_to_simplex(initial_a.astype(float).copy())
    best_a = a.copy()
    best_score = compute_score(a)
    n = len(a)

    for t in range(1, max_iter + 1):
        if time.time() >= time_deadline:
            break

        # schedule for the smoothing parameter
        alpha = alpha_start + (alpha_end - alpha_start) * min(1.0, t / max_iter)

        grad, _ = smoothed_gradient(a, alpha)

        eta = eta0 / np.sqrt(t)
        a_new = a - eta * grad

        # optional tiny isotropic noise to escape flat regions
        if noise_every and (t % noise_every == 0):
            a_new += np.random.randn(n) * noise_scale

        a_new = project_to_simplex(a_new)

        cur_score = compute_score(a_new)
        if cur_score < best_score:
            best_score = cur_score
            best_a = a_new.copy()

        a = a_new

        if t % 5000 == 0:
            print(f"    grad‑descent iter {t:,d}  curScore {cur_score:.6f}  best {best_score:.6f}")

    return best_a.tolist(), best_score


# ----------------------------------------------------------------------
# Stochastic hill‑climbing refinement
# ----------------------------------------------------------------------


def hill_climb(
    seq,
    time_deadline: float,
    max_steps: int = 50000,
    init_step: float = 0.05,
):
    """
    Pair‑wise mass‑transfer hill climbing.
    Moves that improve the *true* objective are kept.
    """
    a = np.array(seq, dtype=float)
    a = project_to_simplex(a)
    best_a = a.copy()
    best_score = compute_score(best_a)
    n = len(a)
    step = init_step

    while max_steps > 0 and time.time() < time_deadline:
        i, j = np.random.randint(n, size=2)
        if i == j:
            continue
        delta = step * best_a[i]
        if delta <= 0.0:
            continue
        cand = best_a.copy()
        cand[i] -= delta
        cand[j] += delta
        cand = project_to_simplex(cand)
        cur_score = compute_score(cand)
        if cur_score < best_score:
            best_score = cur_score
            best_a = cand
            step = min(step * 1.07, 0.5)   # slowly grow step on success
        else:
            step *= 0.96                    # shrink step on failure
        max_steps -= 1
    return best_a.tolist(), best_score


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step‑function that yields a low
    `evaluate_sequence` value.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 2.0  # tiny safety margin

    best_seq = None
    best_val = float("inf")

    # ------------------------------------------------------------------
    # 0. optional fallback (previous best construction if stored)
    # ------------------------------------------------------------------
    try:
        fallback = [float(x) for x in height_sequence_1]
        fallback_val = evaluate_sequence(fallback)
        print(f"Fallback score = {fallback_val:.6f}")
        if fallback_val < best_val:
            best_val = fallback_val
            best_seq = fallback
    except Exception:
        pass

    # ------------------------------------------------------------------
    # 1. Parametric sweep on modest lengths
    # ------------------------------------------------------------------
    lengths = [256, 512, 1024]
    for n in lengths:
        if time.time() >= deadline - 10:
            break
        print(f"\n=== Exploring length n = {n} ===")

        # ---- arcsine (already good seed) -------------------------------
        seq = arcsine_initial(n)
        score = compute_score(seq)
        if score < best_val:
            best_val, best_seq = score, seq.tolist()
            print(f"New best (arcsine) n={n}  score={score:.6f}")

        # ---- uniform ----------------------------------------------------
        uni = np.full(n, 1.0 / n)
        score = compute_score(uni)
        if score < best_val:
            best_val, best_seq = score, uni.tolist()
            print(f"New best (uniform) n={n}  score={score:.6f}")

        # ---- mixture arcsine‑uniform ------------------------------------
        for w in np.linspace(0.0, 1.0, 21):
            if time.time() >= deadline - 10:
                break
            seq = (1.0 - w) * arcsine_initial(n) + w * uni
            seq = seq / np.sum(seq)
            score = compute_score(seq)
            if score < best_val:
                best_val, best_seq = score, seq.tolist()
                print(f"New best (mix) n={n} w={w:.2f}  score={score:.6f}")

        # ---- beta family (symmetric) -----------------------------------
        alphas = np.linspace(-0.99, 2.5, 110)
        for alpha in alphas:
            if time.time() >= deadline - 10:
                break
            seq = beta_sequence(n, alpha)
            score = compute_score(seq)
            if score < best_val:
                best_val, best_seq = score, seq.tolist()
                print(f"New best (β sym) n={n} α={alpha:.3f}  score={score:.6f}")

        # ---- beta family (asymmetric random) ---------------------------
        for _ in range(150):
            if time.time() >= deadline - 10:
                break
            a = np.random.uniform(-0.99, 2.5)
            b = np.random.uniform(-0.99, 2.5)
            seq = beta_sequence(n, a, b)
            score = compute_score(seq)
            if score < best_val:
                best_val, best_seq = score, seq.tolist()
                print(f"New best (β asym) n={n} a={a:.3f} b={b:.3f}  score={score:.6f}")

        # ---- exponential decay -----------------------------------------
        for lam in np.linspace(0.001, 0.03, 15):
            if time.time() >= deadline - 10:
                break
            seq = exponential_sequence(n, lam)
            score = compute_score(seq)
            if score < best_val:
                best_val, best_seq = score, seq.tolist()
                print(f"New best (exp) n={n} λ={lam:.4f}  score={score:.6f}")

        # ---- random Dirichlet draws ------------------------------------
        for _ in range(200):
            if time.time() >= deadline - 10:
                break
            # concentration <1 yields sparser vectors
            concentration = np.random.uniform(0.1, 1.0)
            seq = np.random.dirichlet(np.full(n, concentration))
            score = compute_score(seq)
            if score < best_val:
                best_val, best_seq = score, seq.tolist()
                print(f"New best (Dir) n={n} conc={concentration:.2f}  score={score:.6f}")

    # ------------------------------------------------------------------
    # 2. Smoothed‑max gradient descent on the best candidate so far
    # ------------------------------------------------------------------
    if best_seq is not None and time.time() < deadline - 12:
        n = len(best_seq)
        print(f"\nRunning smoothed‑max gradient descent on n = {n}")
        init_a = np.array(best_seq, dtype=float)
        refined_seq, refined_score = improve_sequence_smoothed(
            initial_a=init_a,
            time_deadline=deadline,
            max_iter=140_000,
            eta0=1.2,
            alpha_start=5.0,
            alpha_end=400.0,
            noise_every=500,
            noise_scale=1e-5,
        )
        print(f"Gradient‑descent best score = {refined_score:.6f}")
        if refined_score < best_val:
            best_val, best_seq = refined_score, refined_seq

    # ------------------------------------------------------------------
    # 3. Final stochastic hill‑climb refinement
    # ------------------------------------------------------------------
    if best_seq is not None and time.time() < deadline - 2:
        print("\nFinal hill‑climb refinement")
        refined_seq, refined_val = hill_climb(
            best_seq,
            time_deadline=deadline,
            max_steps=60000,
            init_step=0.05,
        )
        print(f"Hill‑climb score = {refined_val:.6f}")
        if refined_val < best_val:
            best_val, best_seq = refined_val, refined_seq

    # ------------------------------------------------------------------
    # 4. Optional up‑sampling (keeps shape, may improve bound)
    # ------------------------------------------------------------------
    if best_seq is not None and time.time() < deadline - 1:
        up_n = 4096
        print(f"\nAttempting up‑sampling to n = {up_n}")
        arr = np.array(best_seq, dtype=float)
        up_seq = upsample_sequence(arr, up_n)
        up_score = compute_score(up_seq)
        print(f"Upsampled score = {up_score:.6f}")
        if up_score < best_val:
            best_val, best_seq = up_score, up_seq.tolist()

    # ------------------------------------------------------------------
    # 5. Safety net
    # ------------------------------------------------------------------
    if best_seq is None:
        best_seq = [1.0] * 256
        best_val = evaluate_sequence(best_seq)

    final_score = evaluate_sequence(best_seq)
    print(f"\nSearch finished. Final reported score = {final_score:.6f}")
    return best_seq
