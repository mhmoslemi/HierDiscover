"""
Propose a sequence of non‑negative heights that minimizes the
``evaluate_sequence`` function.

Algorithm
---------
1. **Initialisation** – start from the previously found best sequence
   ( ``height_sequence_1`` ) or from a symmetric Beta distribution.
   The vector is normalised to the probability simplex
   (sum = 1).

2. **Smooth max gradient descent** – minimise the
   infinity–norm of the auto‑convolution by approximating the
   ``max`` with a soft‑max (temperature λ).  In each iteration:
      * compute the full convolution a∗a (FFT);
      * build a soft‑max weight vector w over the 2n‑1 convolution entries;
      * compute the gradient of the weighted sum
        ``∑ w_k (a∗a)[k]`` – this is a simple convolution of ``a`` with the
        reversed weight vector;
      * take a projected (onto the simplex) momentum step.
   λ is increased slowly so the optimisation focusses more on the true
   maximum as the search proceeds.

3. **Local refinement – Simulated annealing** – after the gradient phase
   the best vector is fine‑tuned with a classic mass‑transfer SA that keeps
   the sum exactly 1.  The temperature is exponentially cooled.

The whole routine respects the supplied time budget (default 1000 s) and
always returns the best sequence seen so far.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Helper functions
# ----------------------------------------------------------------------


def simplex_projection(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of a vector onto the probability simplex."""
    v = np.asarray(v, dtype=float)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all entries non‑positive – fall back to uniform
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1] - 1          # zero‑based
    theta = cssv[cond][-1] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def fft_convolve_full(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """
    Linear convolution of two real vectors using FFT.
    Returns a vector of length len(a)+len(b)-1.
    """
    la = a.shape[0]
    lb = b.shape[0]
    l = la + lb - 1
    # next power of two for speed
    nfft = 1 << ((l - 1).bit_length())
    fa = np.fft.rfft(a, n=nfft)
    fb = np.fft.rfft(b, n=nfft)
    conv = np.fft.irfft(fa * fb, n=nfft)[:l]
    return conv


def compute_score(a: np.ndarray) -> float:
    """
    Exact score for a vector that already satisfies sum(a) == 1.
    Formula: 2 * n * max( a * a ).
    """
    n = a.shape[0]
    fft_a = np.fft.rfft(a, n=2 * n - 1)
    conv = np.fft.irfft(fft_a * fft_a, n=2 * n - 1)
    return 2.0 * n * conv.max()


def beta_sequence(alpha: float, beta: float, n: int) -> np.ndarray:
    """
    Discretised Beta(α,β) density on n points, normalised to sum to 1.
    Mid‑point rule avoids the singular endpoints.
    """
    x = (np.arange(n) + 0.5) / n
    logw = (alpha - 1.0) * np.log(x) + (beta - 1.0) * np.log(1.0 - x)
    logw = np.clip(logw, -700.0, 700.0)          # avoid overflow
    w = np.exp(logw)
    if not np.isfinite(w.sum()) or w.sum() == 0.0:
        w = np.ones_like(w)
    return w / w.sum()

# Main candidate generator
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 0, budget_s: int = 1000) -> list[float]:
    """
    Search for a non‑negative height sequence that minimises
    ``evaluate_sequence``.  Uses up to ``budget_s`` seconds; any leftover
    time is spent on a simulated‑annealing refinement.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5          # safety margin

    # ------------------------------
    # 1. Initialise problem size
    # ------------------------------

    except NameError:

        init = beta_sequence(2.5, 2.5, n)
        print(f"No previous sequence – using symmetric beta start, length {n}.")

    # Normalise to simplex (sum == 1)
    a = init / init.sum()
    best_a = a.copy()
    best_score = compute_score(best_a)
    print(f"Initial score = {best_score:.6f}")

    # ------------------------------
    # 2. Smooth‑max gradient descent
    # ------------------------------
    eta = 1.0                 # base step size
    beta_mom = 0.9            # momentum coefficient
    mom = np.zeros_like(a)

    lam = 5.0                 # soft‑max temperature (larger → closer to true max)
    lam_max = 200.0
    lam_factor = 1.001        # increase per iteration

    iter_grad = 0
    no_improve = 0
    while time.time() < deadline - 30.0:
        # a∗a
        conv = fft_convolve_full(a, a)                 # length 2n‑1
        max_conv = conv.max()
        cur_score = 2.0 * n * max_conv

        # record improvement
        if cur_score < best_score - 1e-12:
            best_score = cur_score
            best_a = a.copy()
            no_improve = 0
            if iter_grad % 2000 == 0:
                print(f"Grad iter {iter_grad:6d}: new best {best_score:.6f}")
        else:
            no_improve += 1

        # soft‑max weights (stable computation)
        # shift by max to avoid under/overflow
        shifted = conv - max_conv
        w = np.exp(lam * shifted)
        w_sum = w.sum()
        if w_sum == 0.0:            # numerical safeguard
            w = np.full_like(w, 1.0 / w.size)
        else:
            w = w / w_sum

        # gradient of weighted sum = 2 * (a * w_rev)[2n‑2‑j] for each j
        w_rev = w[::-1]
        grad_full = fft_convolve_full(a, w_rev)        # length 3n‑2
        idx = 2 * n - 2 - np.arange(n)                 # indices corresponding to j
        g = 2.0 * grad_full[idx]                       # shape (n,)

        # momentum update
        mom = beta_mom * mom + (1.0 - beta_mom) * g

        # diminishing step size
        step = eta / math.sqrt(iter_grad + 1.0)

        # projected gradient step
        a = simplex_projection(a - step * mom)

        # increase λ slowly

        # occasional random restart if stuck for long

            mom = np.zeros_like(a)

    print(f"Gradient phase finished after {iter_grad} iterations; "

    # ------------------------------
    # 3. Simulated‑annealing fine‑tuning
    # ------------------------------
    cur_a = a.copy()

    T_start = cur_score * 0.01

    it_sa = 0

        # exponential cooling
        T = max(T_end, T_start * (0.9995 ** it_sa))

        # ---- propose a mass‑transfer move (keeps the sum exactly 1) ----
        src = np.random.choice(n, p=cur_a)     # favour larger entries
        tgt = np.random.randint(n - 1)
        if tgt >= src:
            tgt += 1

        if cur_a[src] <= 0.0:
            continue

        delta = cur_a[src] * np.random.rand()   # fraction of source mass
        cand = cur_a.copy()
        cand[src] -= delta
        cand[tgt] += delta

        cand_score = compute_score(cand)

        # Metropolis acceptance
        if cand_score < cur_score:
            accept = True
        else:

            accept = np.random.rand() < accept_prob

            cur_score = cand_score
            if cand_score < best_score:

                best_a = cand.copy()

            cur_a = beta_sequence(np.random.uniform(0.5, 5.0),
                                  np.random.uniform(0.5, 5.0), n)
            cur_score = compute_score(cur_a)

        it_sa += 1

    print(f"Simulated‑annealing finished after {it_sa} iterations; "
          f"final best score = {best_score:.6f}")

    # ------------------------------
    # 4. Return best sequence (normalised to sum == 1)
    # ------------------------------
    best_a = best_a / best_a.sum()
    return [float(v) for v in best_a.tolist()]
