def run_mirror_descent(a0, time_limit):
    a = a0.copy()
    n = a.shape[0]
    g_cum = np.zeros_like(a)
    lower_bound = 1.0
    max_step = 5.0
    it = 0
    while time.time() < time_limit:
        max_conv, grad = _max_and_subgradient(a)
        # conv objective
        # current score = 2 * n * max_conv
        # update best maybe inside calling function
        norm_sq = float(np.dot(grad, grad))
        if norm_sq == 0.0:
            break
        cur_score = 2.0 * n * max_conv
        step = (cur_score - lower_bound) / (norm_sq + 1e-12)
        step = max(1e-12, min(step, max_step))
        g_cum += step * grad
        tmp = -g_cum
        tmp -= np.max(tmp)
        a = np.exp(tmp)
        a_sum = a.sum()
        if a_sum == 0:
            a = np.full(n, 1.0 / n)
        else:
            a /= a_sum
        it += 1
        # occasional jitter
        if it % 1000 == 0:
            eps = 0.001 * np.random.randn(n)
            a = np.maximum(a + eps, 0.0)
            a /= a.sum()
    return a

def local_hill_climb(a, time_limit, num_trials=1000):
    n = len(a)
    best_a = a.copy()
    best_score = evaluate_sequence(best_a.tolist())
    for _ in range(num_trials):
        if time.time() >= time_limit:
            break
        i, j = rng.choice(n, size=2, replace=False)
        # move epsilon from i to j (if a[i] > epsilon)
        epsilon = min(best_a[i], 0.01)  # small
        if epsilon <= 0:
            continue
        a_new = best_a.copy()
        a_new[i] -= epsilon
        a_new[j] += epsilon
        # ensure nonnegative
        # no need to renormalize because total mass unchanged.
        score = evaluate_sequence(a_new.tolist())
        if score < best_score:
            best_score = score
            best_a = a_new
    return best_a
```

def _greedy_set(n, max_rep, rng, time_limit):
    S = []
    in_set = np.zeros(n, dtype=bool)
    sum_counts = np.zeros(2 * n - 1, dtype=np.int16)  # maybe int16 fine
    fails = 0
    max_fails = n * 10  # or 10000
    while time.time() < time_limit and fails < max_fails:
        i = rng.integers(0, n)
        if in_set[i]:
            continue
        if not S:
            # adding first element always ok if max_rep >=1
            if 1 <= max_rep:
                in_set[i] = True
                S.append(i)
                sum_counts[2 * i] = 1
                fails = 0
                continue
        # compute max after addition
        # contributions to sums with existing S
        idx = np.array(S, dtype=int) + i
        # new counts would be sum_counts[idx] + 2
        # potential max is max of these and sum_counts[2*i] + 1
        # compute max quickly using np.max
        if idx.size > 0:
            candidate_max = sum_counts[idx] + 2
            candidate_max = int(candidate_max.max())
        else:
            candidate_max = 0
        cand_center = sum_counts[2 * i] + 1
        if cand_center > candidate_max:
            candidate_max = int(cand_center)
        if candidate_max <= max_rep:
            # accept
            for j in S:
                sum_counts[i + j] += 2
            sum_counts[2 * i] += 1
            in_set[i] = True
            S.append(i)
            fails = 0
        else:
            fails += 1
    return S, sum_counts

def _run_mirror_descent(a_init, time_limit):
    a = a_init.copy()
    n = a.shape[0]
    # ensure positivity
    epsilon = 1e-30
    a = np.maximum(a, epsilon)
    a = a / a.sum()
    g_cum = np.zeros_like(a)
    lower_bound = 1.0
    max_step = 5.0
    it = 0
    while time.time() < time_limit:
        max_conv, grad = _max_and_subgradient(a)
        cur_score = 2.0 * n * max_conv
        norm_sq = float(np.dot(grad, grad))
        if norm_sq == 0.0:
            break
        step = (cur_score - lower_bound) / (norm_sq + 1e-12)
        step = max(1e-12, min(step, max_step))
        g_cum += step * grad
        tmp = -g_cum
        tmp -= np.max(tmp)
        a = np.exp(tmp)
        a_sum = a.sum()
        if a_sum == 0.0:
            a = np.full(n, 1.0 / n)
        else:
            a /= a_sum
        it += 1
        if it % 2000 == 0:
            # jitter
            eps = 0.001 * np.random.randn(n)
            a = np.maximum(a + eps, 0.0)
            a /= a.sum()
    return a

"""
A fast multi‑phase search for a step function that minimises

    2·n·max(conv(a,a)) / (sum(a)**2)

The algorithm works as follows:

1.  **Sparse “Sidon‑like” construction** –  We greedily build a set of
    indices S ⊂ {0,…,n‑1} such that the number of ordered pairs (i,j)∈S²
    with the same sum i+j is bounded by a small integer K (4‑6).  A
    uniform distribution on S already gives a much smaller max‑convolution
    than the full‑length uniform vector because cross‑terms are very rare.

2.  **Mirror‑descent refinement** –  Starting from the uniform‑on‑S vector
    (or from random restarts) we run an exponentiated‑gradient / mirror‑descent
    iteration that follows the exact sub‑gradient of the max‑convolution.
    The Polyak step‑size with a theoretical lower‑bound (1.0) gives fast
    convergence.

3.  **Random restarts & local hill‑climbing** –  When time permits we launch
    additional random‑initialised mirror‑descent runs and finally perform a
    cheap coordinate‑wise hill‑climb that moves a tiny amount of mass between
    two positions if it improves the objective.

The whole search stops after the allotted time (default 1000 s) and returns
the best sequence found.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
#  Helper: exact max‑convolution and a sub‑gradient of the active entry.
# ----------------------------------------------------------------------
def _max_and_subgradient(a: np.ndarray):
    """
    Returns (max entry of a*a, a sub‑gradient of the max entry).

    The gradient for a particular active sum k is
        ∂/∂a_i (a*a)[k] = 2·a_{k‑i}
    summed over all active k and averaged if several k tie.
    """
    n = a.shape[0]
    fft = np.fft.rfft(a, n=2 * n - 1)
    conv = np.fft.irfft(fft * fft, n=2 * n - 1)
    max_val = float(conv.max())
    # indices attaining (or numerically close to) the maximum
    tol = max_val * 1e-12 + 1e-14
    active = np.where(conv >= max_val - tol)[0]

    grad = np.zeros_like(a)
    for k in active:
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        idx = np.arange(lo, hi + 1, dtype=int)
        partner = k - idx
        grad[idx] += 2.0 * a[partner]
    if active.size > 1:
        grad /= float(active.size)
    return max_val, grad


# ----------------------------------------------------------------------
#  Greedy construction of a sparse set with bounded representation count.
# ----------------------------------------------------------------------
def _greedy_set(n: int, max_rep: int, rng: np.random.Generator, time_limit: float):
    """
    Build a set S ⊂ [0,n) such that for the full convolution of the
    indicator of S the number of ordered pairs with the same sum never exceeds
    max_rep.  Returns the list of indices and the full count array.
    """
    S = []
    in_set = np.zeros(n, dtype=bool)
    # ordered‑pair count for each possible sum (0 … 2n‑2)
    sum_counts = np.zeros(2 * n - 1, dtype=np.int16)

    fails = 0
    max_fails = n * 10                # stop after many consecutive failures
    while time.time() < time_limit and fails < max_fails:
        i = rng.integers(0, n)
        if in_set[i]:
            continue

        if not S:                     # first element – always OK if max_rep≥1
            if max_rep >= 1:
                in_set[i] = True
                S.append(i)
                sum_counts[2 * i] = 1
                fails = 0
            continue

        # compute the maximal count that would appear after inserting i
        idx = i + np.array(S, dtype=int)          # sums with the already chosen points
        cand_counts = sum_counts[idx] + 2         # each ordered pair (i,j) and (j,i)
        cand_max = int(cand_counts.max())
        cand_center = int(sum_counts[2 * i] + 1)   # the (i,i) ordered pair
        cand_max = max(cand_max, cand_center)

        if cand_max <= max_rep:
            # accept i
            for j in S:
                sum_counts[i + j] += 2
            sum_counts[2 * i] += 1
            in_set[i] = True
            S.append(i)
            fails = 0
        else:
            fails += 1
    return S, sum_counts


# ----------------------------------------------------------------------
#  Mirror‑descent optimiser (exponentiated‑gradient version).
# ----------------------------------------------------------------------
def _run_mirror_descent(a0: np.ndarray, time_limit: float):
    """
    Perform the exponentiated‑gradient / mirror‑descent iteration described in
    the reference code.  a0 must be a non‑negative vector; the routine keeps it
    normalised to a probability vector.
    """
    a = np.maximum(a0, 1e-30).astype(np.float64)
    a /= a.sum()
    n = a.shape[0]
    g_cum = np.zeros_like(a)
    lower_bound = 1.0        # theoretical lower bound of the objective
    max_step = 5.0
    it = 0

    while time.time() < time_limit:
        max_conv, grad = _max_and_subgradient(a)
        cur_score = 2.0 * n * max_conv          # because sum(a)==1
        norm_sq = float(np.dot(grad, grad))
        if norm_sq == 0.0:
            break
        # Polyak step size (clipped)
        step = (cur_score - lower_bound) / (norm_sq + 1e-12)
        step = max(1e-12, min(step, max_step))

        g_cum += step * grad
        tmp = -g_cum
        tmp -= np.max(tmp)                      # for numerical stability
        a = np.exp(tmp)
        s = a.sum()
        if s == 0.0:
            a = np.full(n, 1.0 / n)
        else:
            a /= s

        it += 1
        # occasional stochastic jitter to escape plateaus
        if it % 2000 == 0:
            eps = 0.001 * np.random.randn(n)
            a = np.maximum(a + eps, 0.0)
            a /= a.sum()
    return a


# ----------------------------------------------------------------------
#  Simple hill‑climbing that moves a tiny amount of mass between two entries.
# ----------------------------------------------------------------------
def _local_hill_climb(a: np.ndarray, time_limit: float, max_trials: int = 500):
    """
    Very cheap coordinate‑wise search: relocate a small epsilon of probability
    from a randomly chosen index i to another j if the move improves the
    exact objective.  Stops after max_trials attempts or when the time budget
    expires.
    """
    n = a.shape[0]
    best_a = a.copy()
    best_score = evaluate_sequence(best_a.tolist())
    rng = np.random.default_rng()
    trials = 0

    while time.time() < time_limit and trials < max_trials:
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        # move at most 10 % of the mass of i (or a tiny absolute amount)
        eps = min(best_a[i] * 0.1, 0.005)
        if eps <= 0.0:
            continue
        new_a = best_a.copy()
        new_a[i] -= eps
        new_a[j] += eps
        # sum is unchanged, all entries stay non‑negative
        new_score = evaluate_sequence(new_a.tolist())
        if new_score < best_score:
            best_score = new_score
            best_a = new_a
        trials += 1
    return best_a
