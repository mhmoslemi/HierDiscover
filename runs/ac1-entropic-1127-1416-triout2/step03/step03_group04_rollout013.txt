"""
Propose a low‑score non‑negative step function.
The algorithm works in four stages:

1.  A fast parametric sweep (beta‑type families, arcsine, uniform) over several
    lengths (up to 8192) to obtain a decent starting shape.
2.  If the best shape is smaller than the target length it is linearly up‑sampled.
3.  A projected Adam optimizer minimises a smooth log‑sum‑exp approximation of
    the max‑convolution (α is increased from 5 to 2000).  The true objective is
    evaluated at every iteration and the best iterate is kept.
4.  A cheap stochastic pair‑wise mass‑transfer hill‑climber finishes the search.

All vectors are kept on the probability simplex (∑a = 1, a_i ≥ 0) so the scale‑
invariance of the original objective is respected.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Linear‑algebra utilities (simple, no external dependencies)
# ----------------------------------------------------------------------


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / v.shape[0], dtype=float)

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho = np.where(u - cssv / np.arange(1, v.shape[0] + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def next_pow2(x: int) -> int:
    """Return the smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full (linear) convolution using a real FFT."""
    m = len(a) + len(b) - 1
    n_fft = next_pow2(m)
    A = np.fft.rfft(a, n=n_fft)
    B = np.fft.rfft(b, n=n_fft)
    C = A * B
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


def compute_score(a: np.ndarray) -> float:
    """Exact score identical to `evaluate_sequence` for a normalised vector."""
    a = np.asarray(a, dtype=float)
    n = a.shape[0]
    conv = convolution_fft(a, a)                 # length 2·n‑1
    max_val = np.max(conv)
    s = np.sum(a)
    if s < 1e-12:
        return float("inf")
    return 2.0 * n * max_val / (s * s)


# ----------------------------------------------------------------------
# Families of candidate step‑functions
# ----------------------------------------------------------------------


def beta_sequence(n: int, alpha: float, beta: float | None = None) -> np.ndarray:
    """Beta‑type coefficients a_i ∝ (i+½)^α·(n‑½‑i)^β."""
    if beta is None:
        beta = alpha
    i = np.arange(n, dtype=float)
    left = np.power(i + 0.5, alpha)
    right = np.power(n - 0.5 - i, beta)
    seq = left * right
    # Clip pathological spikes (just a safety net)
    seq = np.clip(seq, 0.0, 1e12)
    total = np.sum(seq)
    if total == 0.0:
        seq = np.full(n, 1.0 / n)
    else:
        seq = seq / total
    return seq


def arcsine_sequence(n: int) -> np.ndarray:
    """Arcsine shape a_i ∝ 1/√((i+½)(n‑½‑i))."""
    i = np.arange(n, dtype=float)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - 0.5 - i))
    return vals / np.sum(vals)


def upsample_sequence(seq: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation of a probability vector to a larger size."""
    old_n = len(seq)
    old_x = (np.arange(old_n) + 0.5) / old_n
    new_x = (np.arange(new_n) + 0.5) / new_n
    new_vals = np.interp(new_x, old_x, seq)
    return new_vals / np.sum(new_vals)


# ----------------------------------------------------------------------
# Smoothed‑max gradient (used inside Adam)
# ----------------------------------------------------------------------


def smoothed_grad(a: np.ndarray, alpha: float):
    """
    Gradient of the smoothed max:
        f_α(a) = (1/α)·log Σ_k exp(α·(a∗a)_k)
    Returns the gradient (shape a) and the raw max of the convolution.
    """
    n = a.shape[0]
    conv = convolution_fft(a, a)                # length L = 2n‑1
    maxc = np.max(conv)

    # soft‑max weights
    exp_term = np.exp(alpha * (conv - maxc))    # stability by subtracting maxc
    w = exp_term / np.sum(exp_term)            # Σ w = 1

    # gradient = 2·(a ⋆ w_rev) at the appropriate slice
    w_rev = w[::-1]
    conv2 = convolution_fft(a, w_rev)          # length 3n‑2
    L = conv.shape[0]                           # = 2n‑1
    grad = 2.0 * conv2[L - n : L][::-1]         # slice of length n, reversed
    return grad, maxc


# ----------------------------------------------------------------------
# Cheap stochastic hill‑climbing refinement
# ----------------------------------------------------------------------


def hill_climb(seq, time_deadline: float, max_steps: int = 60000, init_step: float = 0.05):
    """
    Pair‑wise mass‑transfer hill‑climb.
    Only moves that reduce the true (unsmoothed) objective are kept.
    """
    a = np.array(seq, dtype=float)
    a = project_to_simplex(a)
    best_a = a.copy()
    best_score = compute_score(best_a)
    n = len(a)
    step = init_step

    while max_steps > 0 and time.time() < time_deadline:
        i, j = np.random.randint(n, size=2)
        if i == j:
            continue
        delta = step * best_a[i]
        if delta <= 0.0:
            continue
        cand = best_a.copy()
        cand[i] -= delta
        cand[j] += delta
        cand = project_to_simplex(cand)
        cur_score = compute_score(cand)
        if cur_score < best_score:
            best_score = cur_score
            best_a = cand
            step = min(step * 1.07, 0.5)
        else:
            step *= 0.96
        max_steps -= 1

    return best_a.tolist(), best_score

# Main entry point required by the competition harness

def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a step‑function with a low `evaluate_sequence` value.
    The routine respects the time budget and combines a parametric sweep,
    up‑sampling, a projected Adam optimiser on the smooth log‑sum‑exp
    approximation of the max, and a final hill‑climb.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 2.0   # tiny safety margin

    best_seq = None
    best_val = float("inf")

    # ------------------------------------------------------------------
    # Optional fallback (if a previously discovered construction is present)
    # ------------------------------------------------------------------
    try:
        fallback = [float(x) for x in height_sequence_1]   # type: ignore
        fb_val = evaluate_sequence(fallback)
        print(f"Fallback score = {fb_val:.6f}")
        if fb_val < best_val:
            best_val = fb_val
            best_seq = fallback
    except Exception:
        pass

    # ------------------------------------------------------------------
    # Phase 1 – parametric sweep (beta families, arcsine, uniform)
    # ------------------------------------------------------------------
    target_len = 8192                     # final length we aim for
    lengths = [1024, 2048, 4096, target_len]

    # coarse grid for symmetric beta exponents
    sym_alphas = np.linspace(-0.95, 2.5, 180)

    for n in lengths:
        if time.time() > deadline - 5:
            break
        print(f"\n=== Exploring length n = {n} ===")

        # 1) arcsine
        seq = arcsine_sequence(n)
        score = compute_score(seq)
        if score < best_val:
            best_val = score
            best_seq = seq.tolist()
            print(f"New best (arcsine) n={n}  score={score:.6f}")

        # 2) uniform
        seq = np.full(n, 1.0 / n)
        score = compute_score(seq)
        if score < best_val:
            best_val = score
            best_seq = seq.tolist()
            print(f"New best (uniform) n={n}  score={score:.6f}")

        # 3) symmetric beta sweep
        for a in sym_alphas:
            if time.time() > deadline - 5:
                break
            seq = beta_sequence(n, a)
            score = compute_score(seq)
            if score < best_val:
                best_val = score
                best_seq = seq.tolist()
                print(f"New best (sym β) n={n} α={a:.3f}  score={score:.6f}")

        # 4) random asymmetric beta pairs
        for _ in range(500):
            if time.time() > deadline - 5:
                break
            a = np.random.uniform(-0.95, 2.5)
            b = np.random.uniform(-0.95, 2.5)
            seq = beta_sequence(n, a, b)
            score = compute_score(seq)
            if score < best_val:
                best_val = score
                best_seq = seq.tolist()
                print(f"New best (asym) n={n} a={a:.3f} b={b:.3f}  score={score:.6f}")

    # ------------------------------------------------------------------
    # Phase 2 – up‑sample to the target length if needed
    # ------------------------------------------------------------------
    if best_seq is not None and len(best_seq) < target_len and time.time() < deadline - 8:
        print(f"\nUpsampling best candidate from {len(best_seq)} → {target_len}")
        best_arr = upsample_sequence(np.array(best_seq, dtype=float), target_len)
        up_score = compute_score(best_arr)
        print(f"Upsampled score = {up_score:.6f}")
        if up_score < best_val:
            best_val = up_score
            best_seq = best_arr.tolist()
    else:
        # make sure we are working with a numpy array of the final length
        best_arr = np.array(best_seq, dtype=float) if best_seq is not None else np.full(target_len, 1.0 / target_len)

    # ------------------------------------------------------------------
    # Phase 3 – projected Adam optimiser on the smoothed max
    # ------------------------------------------------------------------
    if time.time() < deadline - 6:
        a = project_to_simplex(best_arr.copy())
        n = a.shape[0]

        m = np.zeros_like(a)
        v = np.zeros_like(a)
        beta1, beta2 = 0.9, 0.999
        eps = 1e-8
        lr0 = 0.07                      # base learning rate
        max_iter = 150_000
        cur_score = compute_score(a)
        best_local = a.copy()
        best_local_score = cur_score

        print(f"\nStarting Adam optimisation on n = {n}")

        for t in range(1, max_iter + 1):
            if time.time() >= deadline - 2:
                break

            # continuation of the smoothing parameter
            alpha = 5.0 + (2000.0 - 5.0) * min(1.0, t / max_iter)

            grad, maxc = smoothed_grad(a, alpha)

            # Adam update
            m = beta1 * m + (1.0 - beta1) * grad
            v = beta2 * v + (1.0 - beta2) * (grad * grad)
            m_hat = m / (1.0 - beta1 ** t)
            v_hat = v / (1.0 - beta2 ** t)

            step = lr0 * m_hat / (np.sqrt(v_hat) + eps)
            a_new = a - step
            a_new = project_to_simplex(a_new)

            cur_score = compute_score(a_new)
            if cur_score < best_local_score:
                best_local_score = cur_score
                best_local = a_new.copy()
                print(f"    iter {t:,d}  new best {cur_score:.6f}")

            a = a_new

            if t % 5000 == 0:
                print(f"    iter {t:,d}  curScore {cur_score:.6f}  best {best_local_score:.6f}")

        best_seq = best_local.tolist()
        best_val = best_local_score
        print(f"Adam optimisation finished. Best score = {best_val:.6f}")

    # ------------------------------------------------------------------
    # Phase 4 – final cheap hill‑climb refinement
    # ------------------------------------------------------------------
    if time.time() < deadline - 2:
        print("\nFinal hill‑climb refinement")
        refined_seq, refined_val = hill_climb(best_seq, time_deadline=deadline, max_steps=60000, init_step=0.05)
        print(f"Hill‑climb score = {refined_val:.6f}")
        if refined_val < best_val:
            best_val = refined_val
            best_seq = refined_seq

    # ------------------------------------------------------------------
    # Return the best sequence found (the harness will call `evaluate_sequence`)
    # ------------------------------------------------------------------
    final_score = evaluate_sequence(best_seq)
    print(f"\nSearch finished. Final reported score = {final_score:.6f}")
    return best_seq
