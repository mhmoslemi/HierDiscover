def adam_descent(a0, time_deadline, max_iter=200000):
    a = a0.copy()
    n = len(a)
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    lr0 = 0.05 # base learning rate
    best_local_score = compute_objective(a)
    best_local_seq = a.copy()
    stagnation = 0
    for it in range(1, max_iter+1):
        if time.time() >= time_deadline:
            break
        # convolve
        fft = np.fft.rfft(a, n=2*n-1)
        conv = np.fft.irfft(fft * fft, n=2*n-1)
        max_b = conv.max()
        cur_score = 2.0 * n * max_b
        # update best
        if cur_score < best_local_score - 1e-14:
            best_local_score = cur_score
            best_local_seq = a.copy()
            stagnation = 0
            # also update global best (outer closure will compare)
        else:
            stagnation += 1
        # active set
        eps_active = max_b * 1e-6
        active = np.where(conv >= max_b - eps_active)[0]
        if active.size == 0:
            active = np.array([conv.argmax()], dtype=int)
        g = subgradient_active(a, active)
        # Adam update
        m = beta1 * m + (1 - beta1) * g
        v = beta2 * v + (1 - beta2) * (g * g)
        m_hat = m / (1 - beta1**it)
        v_hat = v / (1 - beta2**it)
        step = lr0 / (np.sqrt(v_hat) + eps)
        a_next = a - step * m_hat
        # Project onto simplex
        a_next = simplex_projection(a_next)
        a = a_next
        # occasional perturbation if stagnation high
        if stagnation >= 20000:
            sigma = max(0.001, 0.2 / np.sqrt(it + 1))
            noise = sigma * np.random.randn(n) * a
            a = a + noise
            a = np.clip(a, 0.0, None)
            a /= a.sum()
            # reset Adam stats
            m.fill(0)
            v.fill(0)
            stagnation = 0
            print(f"Adam iter {it} : perturbation applied, cur_score {cur_score:.6f}")
        # Optionally check early exit if cur_score very close to best global? Not needed
    return best_local_seq, best_local_score

"""
Propose a candidate sequence that minimizes ``evaluate_sequence``.

Algorithm
---------
1. Determine the length *n* (use the global ``height_sequence_1`` if present,
   otherwise fall back to 5146).
2. Build a diverse pool of seed sequences:
   * uniform, linearly decreasing ramp,
   * exponential families,
   * power‑law families a_i ∝ (i+offset)^{-p},
   * discretised Beta distributions,
   * a few random Dirichlet draws.
   All seeds are normalised to sum 1 and the five best are retained.
3. For each of the retained seeds run a *projected Adam* descent on the
   probability simplex.
   – At each iteration we compute the convolution a*a,
   – identify the active set (indices whose value is within 1e‑6 relative
     tolerance of the current maximum),
   – compute the sub‑gradient of the max‑convolution (average over the
     active set),
   – perform an Adam update and project back onto the simplex.
   – If the method stagnates for many steps a small Gaussian perturbation
     is added to escape flat regions.
4. After the Adam phase a lightweight hill‑climbing (pairwise mass transfer)
   is performed on the best vector found so far.
5. The best vector encountered during the whole run is returned as a plain
   ``list[float]``.  The routine respects a 1000 s time budget (a 5 s safety
   margin is kept for the outer judge).
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Basic utilities
# ----------------------------------------------------------------------


def simplex_projection(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of a vector onto the probability simplex."""
    v = np.asarray(v, dtype=float)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho_arr = np.arange(1, n + 1)
    cond = u - cssv / rho_arr > 0
    if not np.any(cond):
        # all non‑positive → uniform distribution
        return np.full_like(v, 1.0 / n)
    rho = rho_arr[cond][-1] - 1          # zero‑based index
    theta = cssv[cond][-1] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def compute_objective(a: np.ndarray) -> float:
    """
    Return the value of ``evaluate_sequence`` for a vector *a* that is already
    normalised to sum 1 (i.e. 2·n·max(conv(a,a))).
    """
    n = a.shape[0]
    # linear convolution via FFT (length 2·n‑1)
    fft = np.fft.rfft(a, n=2 * n - 1)
    conv = np.fft.irfft(fft * fft, n=2 * n - 1)
    return 2.0 * n * conv.max()


def subgradient_active(a: np.ndarray, active: np.ndarray) -> np.ndarray:
    """
    Sub‑gradient of the max‑convolution with respect to *a*,
    averaged over the supplied active index set.
    """
    n = a.shape[0]
    g = np.zeros_like(a)
    for k in active:
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        idx = np.arange(lo, hi + 1, dtype=int)
        partner = k - idx
        g[idx] += 2.0 * a[partner]
    if active.size:
        g /= float(active.size)
    return g


# ----------------------------------------------------------------------
# Shape generators (all return a *probability* vector, i.e. sum = 1)
# ----------------------------------------------------------------------


def uniform_shape(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n)


def ramp_shape(n: int) -> np.ndarray:
    """Linearly decreasing ramp a_i ∝ (n-i)."""
    a = np.arange(n, 0, -1, dtype=float)
    a = 2.0 * a / (n * (n + 1))
    a /= a.sum()
    return a


def exponential_shape(lam: float, n: int) -> np.ndarray:
    """a_i ∝ exp(-lam·i)."""
    i = np.arange(n, dtype=float)
    w = np.exp(-lam * i)
    w_sum = w.sum()
    if w_sum == 0 or not np.isfinite(w_sum):
        w = np.ones_like(w)
        w_sum = w.sum()
    return w / w_sum


def powerlaw_shape(p: float, offset: float, n: int) -> np.ndarray:
    """a_i ∝ (i+offset)^{-p} (offset>0)."""
    i = np.arange(n, dtype=float) + offset
    w = np.power(i, -p)
    w_sum = w.sum()
    if w_sum == 0 or not np.isfinite(w_sum):
        w = np.ones_like(w)
        w_sum = w.sum()
    return w / w_sum


def beta_shape(alpha: float, beta: float, n: int) -> np.ndarray:
    """Discretised Beta(α,β) density on the mid‑points of n cells."""
    x = (np.arange(n) + 0.5) / n
    # work in log‑space to avoid overflow/underflow
    logw = (alpha - 1.0) * np.log(x) + (beta - 1.0) * np.log(1.0 - x)
    logw = np.clip(logw, -700.0, 700.0)
    w = np.exp(logw)
    w_sum = w.sum()
    if w_sum == 0 or not np.isfinite(w_sum):
        w = np.ones_like(w)
        w_sum = w.sum()
    return w / w_sum


def random_dirichlet(n: int) -> np.ndarray:
    """Draw a random point on the simplex via a Gamma (Dirichlet) draw."""
    conc = np.random.uniform(0.1, 3.0, size=n)
    w = np.random.gamma(conc, 1.0)
    w_sum = w.sum()
    if w_sum == 0:
        w = np.ones_like(w)
        w_sum = w.sum()
    return w / w_sum


# ----------------------------------------------------------------------
# Core optimisation primitives
# ----------------------------------------------------------------------


def adam_descent(
    a0: np.ndarray,
    deadline: float,
    max_iter: int = 200_000,
) -> tuple[np.ndarray, float]:
    """
    Projected Adam optimisation of the max‑convolution objective.
    Returns the best vector found and its objective value.
    """
    a = a0.copy()
    n = a.shape[0]

    # Adam state
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    lr0 = 0.08  # base learning rate (tuned empirically)

    best_a = a.copy()
    best_score = compute_objective(best_a)

    stagnation = 0

    for it in range(1, max_iter + 1):
        if time.time() >= deadline:
            break

        # convolution and max
        fft = np.fft.rfft(a, n=2 * n - 1)
        conv = np.fft.irfft(fft * fft, n=2 * n - 1)
        max_b = conv.max()
        cur_score = 2.0 * n * max_b

        # keep local best
        if cur_score < best_score - 1e-14:
            best_score = cur_score
            best_a = a.copy()
            stagnation = 0
            # progress report
            print(f"Adam it {it:6d}: new local best {best_score:.6f}")
        else:
            stagnation += 1

        # active set (within 1e‑6 relative tolerance)
        eps_act = max_b * 1e-6
        active = np.where(conv >= max_b - eps_act)[0]
        if active.size == 0:
            active = np.array([conv.argmax()], dtype=int)

        # sub‑gradient
        g = subgradient_active(a, active)

        # Adam update
        m = beta1 * m + (1.0 - beta1) * g
        v = beta2 * v + (1.0 - beta2) * (g * g)

        # bias correction
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        step = lr0 / (np.sqrt(v_hat) + eps)
        a_next = a - step * m_hat
        a_next = simplex_projection(a_next)

        a = a_next

        # occasional random kick to break out of plateaus
        if stagnation >= 20_000:
            sigma = max(0.001, 0.2 / np.sqrt(it + 1))
            noise = sigma * np.random.randn(n) * a
            a = a + noise
            a = np.clip(a, 0.0, None)
            a /= a.sum()
            m.fill(0.0)
            v.fill(0.0)
            stagnation = 0
            print(f"Adam it {it:6d}: perturbation applied, cur_score {cur_score:.6f}")

        # optional very cheap smoothing every 5000 steps (helps stability)
        if it % 5_000 == 0:
            # simple 3‑point moving average, then project again
            a = np.convolve(a, np.ones(3) / 3.0, mode="same")
            a = simplex_projection(a)

    return best_a, best_score


def hill_climb(
    a_start: np.ndarray,
    deadline: float,
    max_trials: int = 8_000,
) -> tuple[np.ndarray, float]:
    """
    Tiny coordinate‑wise hill‑climbing: repeatedly move a random
    fraction of mass from one index to another.  Accepts only improving
    moves.  Returns the best vector seen and its objective.
    """
    a = a_start.copy()
    best_a = a.copy()
    best_score = compute_objective(best_a)
    n = a.shape[0]
    trial = 0

    while trial < max_trials and time.time() < deadline:
        trial += 1
        i, j = np.random.choice(n, 2, replace=False)
        if a[i] == 0.0:
            continue
        # transfer a random fraction (up to 15 %) of the smaller bucket
        frac = np.random.rand() * 0.15
        delta = frac * a[i]
        a_new = best_a.copy()
        a_new[i] -= delta
        a_new[j] += delta
        # keep on the simplex automatically (sum unchanged, non‑negativity ensured)
        score = compute_objective(a_new)
        if score < best_score - 1e-12:
            best_score = score
            best_a = a_new
            print(f"Hill‑climb trial {trial:5d}: improved to {best_score:.6f}")
    return best_a, best_score

def propose_candidate(seed: int = 0, budget_s: int = 1000) -> list[float]:
    """
    Search for a non‑negative step function (list of floats) that minimises
    ``evaluate_sequence`` within the given time budget.
    """
    np.random.seed(seed)
    start_time = time.time()
    # keep 5 s safety margin for the outer driver
    deadline = start_time + budget_s - 5.0

    # ------------------------------------------------------------------
    # 1. Problem size
    # ------------------------------------------------------------------
    try:
        n = len(height_sequence_1)  # type: ignore[name-defined]
    except NameError:
        n = 5146

    # ------------------------------------------------------------------
    # 2. Book‑keeping
    # ------------------------------------------------------------------
    best_seq: np.ndarray | None = None
    best_score = np.inf
    TOP_K = 10
    top: list[tuple[float, np.ndarray]] = []

    def consider(candidate: np.ndarray) -> None:
        """Evaluate *candidate* (assumed normalised) and maintain globals."""
        nonlocal best_seq, best_score, top
        score = compute_objective(candidate)
        if score < best_score:
            best_score = score
            best_seq = candidate.copy()
            print(f"Global best updated: {best_score:.6f}")
        # maintain a sorted top‑K list (smallest score first)
        if len(top) < TOP_K:
            top.append((score, candidate.copy()))
            top.sort(key=lambda x: x[0])
        else:
            if score < top[-1][0]:
                top[-1] = (score, candidate.copy())
                top.sort(key=lambda x: x[0])

    # ------------------------------------------------------------------
    # 3. Seed generation (deterministic families)
    # ------------------------------------------------------------------
    consider(uniform_shape(n))
    consider(ramp_shape(n))

    # Exponential family
    for lam in np.linspace(0.001, 5.0, 22):
        if time.time() >= deadline:
            break
        consider(exponential_shape(lam, n))

    # Power‑law family (p, offset)
    p_vals = np.linspace(0.25, 2.5, 25)
    offset_vals = np.linspace(0.01, 4.5, 20)
    for p in p_vals:
        for off in offset_vals:
            if time.time() >= deadline:
                break
            consider(powerlaw_shape(p, off, n))

    # Beta grid (coarse)
    alphas = np.linspace(0.2, 5.0, 12)
    betas = np.linspace(0.2, 5.0, 12)
    for a in alphas:
        for b in betas:
            if time.time() >= deadline:
                break
            consider(beta_shape(a, b, n))

    # Random Dirichlet draws
    for _ in range(150):
        if time.time() >= deadline:
            break
        consider(random_dirichlet(n))

    # ------------------------------------------------------------------
    # 4. Convex combinations of the current top candidates
    # ------------------------------------------------------------------
    for _ in range(120):
        if time.time() >= deadline:
            break
        i, j = np.random.choice(len(top), 2, replace=False)
        w = np.random.rand()
        mix = w * top[i][1] + (1.0 - w) * top[j][1]
        mix /= mix.sum()
        consider(mix)

    # ------------------------------------------------------------------
    # 5. Refine the best seeds with projected Adam descent
    # ------------------------------------------------------------------
    # Sort top list again (just in case)
    top.sort(key=lambda x: x[0])

    max_refine = min(5, len(top))  # limit the number of full Adam runs
    for idx in range(max_refine):
        if time.time() >= deadline:
            break
        init_seq = top[idx][1]
        print(f"--- Adam refinement #{idx+1} (init score {top[idx][0]:.6f}) ---")
        refined_seq, refined_score = adam_descent(init_seq, deadline)
        consider(refined_seq)

    # ------------------------------------------------------------------
    # 6. Final fine‑tuning via tiny hill‑climbing moves
    # ------------------------------------------------------------------
    if best_seq is not None and time.time() < deadline:
        remaining_time = deadline - time.time()
        # Allow up to a few seconds for the hill‑climb (the bulk is cheap)
        hc_deadline = time.time() + min(20.0, remaining_time)
        print("--- Hill‑climbing final polish ---")
        polished_seq, polished_score = hill_climb(best_seq, hc_deadline)
        consider(polished_seq)

    # ------------------------------------------------------------------
    # 7. Return result
    # ------------------------------------------------------------------
    if best_seq is None:
        # This should never happen, but the external driver expects a list.
        raise RuntimeError("No candidate was generated.")
    print(f"Finished search. Best score = {best_score:.6f}")
    return [float(v) for v in best_seq.tolist()]
