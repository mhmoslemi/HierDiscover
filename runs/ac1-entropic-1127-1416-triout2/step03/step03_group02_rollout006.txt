"""
search_step_function.py

A heuristic search for a non‑negative step‑function that minimises

    2·n·max(conv(a,a)) / (sum(a)²)

(the same quantity computed by evaluate_sequence).  The algorithm works
in three phases:

1.  **Broad parametric sweep** –  candidate shapes are generated from a
    few low‑dimensional families (arcsine, power‑law, symmetric beta,
    mixtures with the uniform distribution and random Dirichlet draws).
    For each length ``n`` the best few candidates (by a fast FFT based
    score) are retained.

2.  **Mirror‑descent refinement** –  each retained candidate is improved
    by a sub‑gradient (mirror‑descent) method on the simplex.  The
    true non‑smooth objective ``max(conv)`` is handled directly by using
    the binary indicator of the maximal convolution entry as the weight
    vector, i.e. a genuine sub‑gradient of the max‑function.  Updates are
    multiplicative (exponentiated‑gradient) and the iterate is re‑projected
    onto the probability simplex after each step.

3.  **Final polishing** –  the current global best is refined once more
    with a longer run of the same sub‑gradient method.

All operations use FFT‑based convolutions (``np.fft.rfft``) for speed.
The routine respects the supplied time budget (default 1000 s) and
always returns a feasible list of non‑negative floats.
"""

import time
import numpy as np
from typing import List, Tuple

# ----------------------------------------------------------------------
# Low‑level numeric utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two >= x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Linear convolution via real FFT (full length, zero‑padded to a power‑of‑2)."""
    m = len(a) + len(b) - 1
    n_fft = next_pow2(m)
    A = np.fft.rfft(a, n=n_fft)
    B = np.fft.rfft(b, n=n_fft)
    C = A * B
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


def compute_score(a: np.ndarray) -> float:
    """
    Fast evaluation of the official objective (scale‑invariant):
        2·n·max(conv(a,a)) / (sum(a)²)
    a is assumed to be non‑negative; the function works for any vector.
    """
    n = len(a)
    conv = convolution_fft(a, a)
    max_val = float(np.max(conv))
    s = float(np.sum(a))
    if s < 1e-12:
        return float("inf")
    return 2.0 * n * max_val / (s * s)


def normalize(v: np.ndarray) -> np.ndarray:
    """Project a non‑negative vector onto the probability simplex (sum = 1)."""
    total = v.sum()
    if total <= 0.0:
        return np.full_like(v, 1.0 / len(v))
    return v / total


def gradient_from_weights(a: np.ndarray, w: np.ndarray) -> np.ndarray:
    """
    Gradient of Σ_k w_k·(a*a)_k w.r.t. a.
    Implements   g_j = 2· Σ_m a_m · w_{m+j}.
    Computed via one FFT‑based convolution.
    """
    w_rev = w[::-1]
    conv = convolution_fft(a, w_rev)               # length = 3·n‑2
    idx = 2 * len(a) - 2                            # points to position 2n‑2
    grads = 2.0 * conv[idx - np.arange(len(a))]
    return grads


# ----------------------------------------------------------------------
# Parametric families for the initial sweep
# ----------------------------------------------------------------------


def arcsine_initial(n: int) -> np.ndarray:
    """Discrete arcsine distribution (≈ 1/sqrt(i·(n‑1‑i))) normalised."""
    i = np.arange(n, dtype=float)
    eps = 0.25
    vals = 1.0 / np.sqrt((i + 0.5) * (n - 0.5 - i) + eps)
    return normalize(vals)


def powerlaw_shape(n: int, p: float, eps: float) -> np.ndarray:
    """Symmetric power‑law shape a_i ∝ ((i+ε)(n‑1‑i+ε))⁻ᵖ."""
    i = np.arange(n, dtype=float)
    base = ((i + eps) * (n - 1 - i + eps)) ** (-p)
    return normalize(base)


def beta_shape(n: int, a: float) -> np.ndarray:
    """Symmetric Beta‑like shape with both parameters equal to ``a``."""
    i = np.arange(n, dtype=float)
    base = (i + 0.5) ** (a - 1) * (n - 1 - i + 0.5) ** (a - 1)
    return normalize(base)


def uniform_shape(n: int) -> np.ndarray:
    """Uniform probability vector."""
    return np.full(n, 1.0 / n)


def random_dirichlet(n: int, rng: np.random.Generator) -> np.ndarray:
    """Draw a random point on the simplex (exponential sampling)."""
    samp = rng.exponential(1.0, size=n)
    return normalize(samp)


# ----------------------------------------------------------------------
# Mirror‑descent (sub‑gradient) refinement
# ----------------------------------------------------------------------


def refine_subgradient(
    a: np.ndarray,
    deadline: float,
    *,
    max_iter: int = 30_000,
    eta0: float = 0.7,
    report_every: int = 0,
) -> Tuple[np.ndarray, float]:
    """
    Sub‑gradient descent on the true objective ``max(conv(a,a))``.
    Multiplicative (exponentiated‑gradient) updates guarantee non‑negativity.
    The iterate is always kept on the simplex.
    Returns the best vector seen and its (exact) score.
    """
    a = normalize(a.astype(float).copy())
    best_a = a.copy()
    best_sc = compute_score(a)

    n = len(a)
    conv_len = 2 * n - 1

    for it in range(1, max_iter + 1):
        if time.time() >= deadline:
            break

        conv = convolution_fft(a, a)
        max_val = np.max(conv)
        # binary weight vector for the (sub‑)gradient of the max‑function
        w = np.zeros(conv_len, dtype=float)
        # treat all maximising indices (ties) uniformly
        max_idx = np.where(conv >= max_val - 1e-14)[0]
        w[max_idx] = 1.0 / len(max_idx)

        grad = gradient_from_weights(a, w)

        eta = eta0 / np.sqrt(it)          # classic 1/√t schedule
        a = a * np.exp(-eta * grad)       # multiplicative update
        a = normalize(a)                  # project to simplex

        cur_sc = compute_score(a)
        if cur_sc < best_sc:
            best_sc = cur_sc
            best_a = a.copy()

        if report_every and (it % report_every == 0):
            print(f"    sub‑grad iter={it:,d} cur={cur_sc:.6f} best={best_sc:.6f}")

    return best_a, best_sc

        # --------------------------------------------------------------
        scored = [(compute_score(c), c) for c in candidates]
        scored.sort(key=lambda x: x[0])
        top_k = 5
        top_candidates = [c for _, c in scored[:top_k]]
        print("  top initial scores:",
              " ".join(f"{sc:.6f}" for sc, _ in scored[:top_k]))

        # --------------------------------------------------------------
        # 3.  Refine each retained candidate with sub‑gradient descent.
        # --------------------------------------------------------------
        for idx, cand in enumerate(top_candidates):
            if time.time() >= deadline - 5.0:
                break
            print(f"  candidate {idx + 1}/{len(top_candidates)} start_score = {compute_score(cand):.6f}")

            # decide a modest iteration budget based on remaining time
            remaining = max(0.0, deadline - time.time())
            # empirically ~15 iterations per second for n≈2000
            max_iter = int(min(30_000, max(2_000, remaining * 15)))

            refined, _ = refine_subgradient(
                a=cand,
                deadline=deadline,
                max_iter=max_iter,
                eta0=0.7,
                report_every=0,
            )

            # final (official) evaluation
            val = evaluate_sequence(refined.tolist())
            print(f"    evaluated score = {val:.6f}")

            if val < best_val:
                best_val = val
                best_seq = refined.tolist()
                print(f"    *** New global best: {best_val:.6f} (n={n}) ***")

    # ------------------------------------------------------------------
    # 4.  Final polishing of the overall best sequence (if time permits).
    # ------------------------------------------------------------------
    if best_seq and time.time() < deadline - 5.0:
        print("\nFinal polishing of the global best")
        a = np.array(best_seq, dtype=float)
        remaining = max(0.0, deadline - time.time())
        max_iter = int(min(80_000, max(5_000, remaining * 15)))

        polished, _ = refine_subgradient(
            a=a,
            deadline=deadline,
            max_iter=max_iter,
            eta0=0.5,
            report_every=0,
        )
        final_val = evaluate_sequence(polished.tolist())
        print(f"Polished score = {final_val:.6f}")

        if final_val < best_val:
            best_val = final_val
            best_seq = polished.tolist()
            print("*** Updated global best after final polish ***")

    # ------------------------------------------------------------------
    # 5.  Guard – always return a feasible sequence.
    # ------------------------------------------------------------------
    if not best_seq:
        # fallback to a safe uniform vector
        best_seq = uniform_shape(256).tolist()
        best_val = evaluate_sequence(best_seq)

    print(f"\nSearch finished. Best reported score = {best_val:.6f}")
    return best_seq
