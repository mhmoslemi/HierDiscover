The goal is to find a non‑negative vector `a` (step‑function heights) that minimises  

```
F(a) = 2·n·maxₖ (a * a)[k]          (∑ aᵢ = 1, aᵢ ≥ 0)
```  

where `*` is the discrete convolution.  
The program works in three stages

1. **Rich seeding** – a large and diverse set of deterministic, parametric
   (exponential, power‑law, Beta) and random Dirichlet‑type vectors is created;
   the best few are kept.
2. **Smoothed gradient descent** – we minimise a *smooth* surrogate of the max
   (`τ·log‑sum‑exp`) by an Adam‑style update on the probability simplex.
   A decreasing temperature `τ` turns the surrogate gradually into the true
   objective.  The gradient of the surrogate can be written as a convolution,
   therefore each iteration costs only a few FFTs.
3. **Active‑set LP refinement** – after the gradient phase we solve a tiny
   linear programme that maximises the total mass of a new vector `g` under the
   *active* convolution constraints `(a * g)ₖ ≤ current M`.  Mixing `g` into the
   current iterate strictly reduces the maximum convolution value.

All steps respect the 1000 s budget, keep the vector on the simplex by Euclidean
projection (Duchi et al.) and return the best sequence found.

```python
"""
Propose a candidate step‑function that minimises ``evaluate_sequence``.
The algorithm consists of
  1) generating many diverse seed vectors,
  2) performing a smoothed max‑gradient descent on the simplex (Adam,
     decreasing temperature τ → 0),
  3) a short active‑set linear‑program refinement.
All operations are vectorised and use FFTs for speed.
"""

import time
import numpy as np
from scipy.optimize import linprog
from scipy.sparse import lil_matrix

# ----------------------------------------------------------------------
# 1.  Basic utilities – all top‑level, no closures
# ----------------------------------------------------------------------


def simplex_projection(v):
    """Euclidean projection of a vector onto the probability simplex.
    Implementation follows Duchi et al. (2008)."""
    v = np.asarray(v, dtype=float)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1] - 1
    theta = cssv[cond][-1] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def compute_objective(a):
    """Return 2·n·max(a∗a) assuming Σ aᵢ = 1."""
    n = a.shape[0]
    fft_a = np.fft.rfft(a, n=2 * n - 1)
    conv = np.fft.irfft(fft_a * fft_a, n=2 * n - 1)
    return 2.0 * n * conv.max()


def beta_sequence(alpha, beta, n):
    """Discretised Beta(α,β) density on the mid‑points, normalised to sum 1."""
    x = (np.arange(n) + 0.5) / n
    # work in log‑space to avoid overflow/underflow
    logw = (alpha - 1.0) * np.log(x) + (beta - 1.0) * np.log(1.0 - x)
    logw = np.clip(logw, -700.0, 700.0)
    w = np.exp(logw)
    total = w.sum()
    if total == 0 or not np.isfinite(total):
        w = np.ones_like(w)
        total = w.sum()
    return w / total


def exponential_shape(lam, n):
    """aᵢ ∝ exp(‑lam·i),  lam ≥ 0."""
    i = np.arange(n, dtype=float)
    w = np.exp(-lam * i)
    w /= w.sum()
    return w


def power_law_shape(p, n):
    """aᵢ ∝ (i+1)⁻ᵖ,  p ≥ 0."""
    i = np.arange(1, n + 1, dtype=float)
    w = 1.0 / np.power(i, p)
    w /= w.sum()
    return w


def subgradient_active(a, active):
    """Average sub‑gradient of the max‑convolution entry over *active* indices."""
    n = a.shape[0]
    g = np.zeros_like(a)
    for k in active:
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        idx = np.arange(lo, hi + 1, dtype=int)
        partner = k - idx
        g[idx] += 2.0 * a[partner]
    g /= float(active.size)
    return g


# ----------------------------------------------------------------------
# 2.  Main search routine
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 0, budget_s: int = 1000):
    """
    Search for a non‑negative step‑function (list of floats) that minimises
    ``evaluate_sequence``.  The function respects the 1000 s budget and returns
    the best sequence discovered.
    """
    np.random.seed(seed)
    start = time.time()
    # keep a small safety margin
    deadline = start + budget_s - 5.0

    # --------------------------------------------------------------
    # Problem size
    # --------------------------------------------------------------
    try:
        n = len(height_sequence_1)   # type: ignore[name-defined]
    except NameError:
        n = 5146                     # fallback length from the prompt

    # --------------------------------------------------------------
    # Helper to keep the globally best sequence
    # --------------------------------------------------------------
    best_seq = None
    best_score = np.inf
    TOP_K = 7
    top = []          # list of (score, seq) sorted increasingly

    def consider(candidate):
        """Evaluate *candidate* and update the global / top‑K list."""
        nonlocal best_seq, best_score, top
        s = compute_objective(candidate)
        if s < best_score:
            best_score = s
            best_seq = candidate.copy()
            print(f"Initial global best : {s:.6f}")
        # keep the top‑K
        if len(top) < TOP_K:
            top.append((s, candidate.copy()))
            top.sort(key=lambda x: x[0])
        else:
            if s < top[-1][0]:
                top[-1] = (s, candidate.copy())
                top.sort(key=lambda x: x[0])

    # --------------------------------------------------------------
    # 2. Seed families (deterministic + random)
    # --------------------------------------------------------------
    # Uniform
    consider(np.full(n, 1.0 / n))

    # Linear decreasing ramp
    ramp = np.arange(n, 0, -1, dtype=float)
    ramp = ramp / ramp.sum()
    consider(ramp)

    # Exponential family
    for lam in np.logspace(-3, 1, 18):       # 0.001 … 10
        consider(exponential_shape(lam, n))

    # Power‑law family
    for p in [0.2, 0.5, 1.0, 1.5, 2.0, 3.0]:
        consider(power_law_shape(p, n))

    # Systematic Beta sweep (coarse)
    alphas = np.linspace(0.05, 5.0, 12)
    betas = np.linspace(0.05, 5.0, 12)
    for a_ in alphas:
        for b_ in betas:
            if time.time() >= deadline:
                break
            consider(beta_sequence(a_, b_, n))
        if time.time() >= deadline:
            break

    # Random Dirichlet‑like vectors + random Beta parameters
    for _ in range(150):
        if time.time() >= deadline:
            break
        # Dirichlet‑like
        conc = np.random.uniform(0.1, 3.0, size=n)
        vec = np.random.gamma(conc, 1.0)
        vec /= vec.sum()
        consider(vec)

        # Random Beta
        a_ = 10 ** np.random.uniform(-2, 1)   # 0.01 … 10
        b_ = 10 ** np.random.uniform(-2, 1)
        consider(beta_sequence(a_, b_, n))

    # --------------------------------------------------------------
    # 3. Convex combinations of current top candidates (exploration)
    # --------------------------------------------------------------
    for _ in range(120):
        if time.time() >= deadline:
            break
        i, j = np.random.choice(len(top), 2, replace=False)
        w = np.random.rand()
        mix = w * top[i][1] + (1.0 - w) * top[j][1]
        mix /= mix.sum()
        consider(mix)

    # --------------------------------------------------------------
    # 4. Smoothed gradient descent (Adam + temperature annealing)
    # --------------------------------------------------------------
    a = best_seq.copy()
    a = np.maximum(a, 1e-12)   # avoid exact zeros for log‑related ops
    a /= a.sum()

    # Adam state
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8

    # temperature schedule for the soft‑max approximation of the max
    tau = 0.2
    min_tau = 1e-5
    tau_decay = 0.9995

    stagnation = 0
    iteration = 0
    max_iter = 250000

    while time.time() < deadline and iteration < max_iter:
        # ----- convolution and current objective -----
        fft_a = np.fft.rfft(a, n=2 * n - 1)
        conv = np.fft.irfft(fft_a * fft_a, n=2 * n - 1)
        max_b = conv.max()
        cur_score = 2.0 * n * max_b

        if cur_score < best_score - 1e-12:
            best_score = cur_score
            best_seq = a.copy()
            stagnation = 0
            print(f"Iter {iteration:6d}: new best {best_score:.6f}")
        else:
            stagnation += 1

        # ----- smooth approximation of the max (soft‑max) -----
        tau = max(min_tau, tau * tau_decay)
        w = np.exp(conv / tau)
        w_sum = w.sum()
        if w_sum == 0 or not np.isfinite(w_sum):
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        # ----- gradient of the smoothed objective (vectorised) -----
        # g_i = 2 * Σ_j a_j * w_{i+j}
        rev_w = w[::-1]                     # length 2n‑1
        needed = 3 * n - 2                  # length of full a⨂rev_w
        size = 1
        while size < needed:
            size <<= 1
        A = np.fft.rfft(a, size)
        W = np.fft.rfft(rev_w, size)
        conv_full = np.fft.irfft(A * W, size)[:needed]
        g = 2.0 * conv_full[(2 * n - 2) - np.arange(n)]

        # ----- Adam update (step size decreases automatically) -----
        m = beta1 * m + (1.0 - beta1) * g
        v = beta2 * v + (1.0 - beta2) * (g * g)
        m_hat = m / (1.0 - beta1 ** (iteration + 1))
        v_hat = v / (1.0 - beta2 ** (iteration + 1))
        lr = 0.12 / np.sqrt(iteration + 1)   # base learning rate
        step = lr * m_hat / (np.sqrt(v_hat) + eps)

        a_next = a - step
        a_next = simplex_projection(a_next)
        a = a_next
        iteration += 1

        # ----- occasional random perturbation to escape plateaus -----
        if stagnation >= 25000:
            sigma = max(0.01, 0.5 / np.sqrt(iteration + 1))
            noise = sigma * np.random.randn(n) * a
            a = a + noise
            a = np.clip(a, 0.0, None)
            a /= a.sum()
            m.fill(0.0)
            v.fill(0.0)
            stagnation = 0
            print(f"Iter {iteration:6d}: perturbation applied, cur {cur_score:.6f}")

        if best_score <= 1.503:
            print(f"Target reached at iteration {iteration}, score={best_score:.6f}")
            break

    # --------------------------------------------------------------
    # 5. Active‑set linear‑program refinement (few cheap LPs)
    # --------------------------------------------------------------
    a = best_seq.copy()
    lp_iter = 0
    while time.time() < deadline and lp_iter < 12:
        # compute current convolution and the active set
        fft_a = np.fft.rfft(a, n=2 * n - 1)
        conv = np.fft.irfft(fft_a * fft_a, n=2 * n - 1)
        M = conv.max()
        eps = M * 1e-6
        active = np.where(conv >= M - eps)[0]
        if active.size == 0:
            active = np.array([conv.argmax()], dtype=int)

        # Build the constraint matrix for the active indices only
        rows = len(active)
        A = lil_matrix((rows, n), dtype=np.float64)
        for row, k in enumerate(active):
            j_low = max(0, k - (n - 1))
            j_high = min(k, n - 1)
            # for j in [j_low, j_high] the coefficient of g_j is a_{k-j}
            # fill the row
            idx_range = np.arange(j_low, j_high + 1, dtype=int)
            coeffs = a[k - idx_range]
            A.rows[row] = idx_range.tolist()
            A.data[row] = coeffs.tolist()

        A = A.tocsr()
        b = np.full(rows, M, dtype=np.float64)

        # maximise sum(g)  ⇔ minimise –1·g
        c = -np.ones(n, dtype=np.float64)
        res = linprog(c, A_ub=A, b_ub=b,
                      bounds=[(0, None)] * n,
                      method='highs',
                      options={'presolve': True, 'time_limit': 30})
        if not res.success:
            break
        g = res.x
        total_g = g.sum()
        if total_g <= 0:
            break
        g /= total_g                         # renormalise to sum 1

        # Mix the new direction into the current iterate
        t = 0.12   # modest mixing factor
        a_new = (1.0 - t) * a + t * g
        a_new /= a_new.sum()
        a = a_new
        lp_iter += 1

        # keep the best ever seen
        cur_score = compute_objective(a)
        if cur_score < best_score - 1e-12:
            best_score = cur_score
            best_seq = a.copy()
            print(f"LP‑iter {lp_iter}: improved to {best_score:.6f}")

        if best_score <= 1.503:
            break

    # --------------------------------------------------------------
    # 6. Return the best sequence discovered
    # --------------------------------------------------------------
    print(f"Finished: best score = {best_score:.6f} after {iteration} SGD steps "
          f"+ {lp_iter} LP refinements.")
    return [float(v) for v in best_seq.tolist()]
```
