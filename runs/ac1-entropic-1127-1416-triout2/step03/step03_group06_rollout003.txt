"""
Propose a step‑function (list of non‑negative heights) that minimizes
``evaluate_sequence``.  The method builds a diverse set of parametric
seed shapes (uniform, ramps, exponentials, power‑laws, Beta‑densities,
logistic, polynomial), keeps the best few, then refines the current
best by a projected (sub‑gradient) descent on the probability simplex
with Nesterov‑style momentum and occasional random perturbations.
The whole search respects the supplied time budget (default 1000 s).
"""
import time
import numpy as np
import math

# ----------------------------------------------------------------------
# Basic utilities
# ----------------------------------------------------------------------


def simplex_projection(v):
    """
    Euclidean projection onto the probability simplex { w | w_i >= 0, Σw_i = 1 }.
    Implementation of Duchi et al. (2008).
    """
    v = np.asarray(v, dtype=float)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all entries non‑positive → uniform distribution
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1] - 1               # zero‑based index
    theta = cssv[cond][-1] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def compute_objective(a):
    """
    Compute the objective 2·n·max(a*a) under the hypothesis Σ a_i = 1.
    Uses FFT for fast convolution.
    """
    a = np.asarray(a, dtype=float)
    n = a.shape[0]
    # linear convolution via zero‑padding to length 2·n‑1
    fft = np.fft.rfft(a, n=2 * n - 1)
    conv = np.fft.irfft(fft * fft, n=2 * n - 1)
    return 2.0 * n * conv.max()


def subgradient_active(a, active):
    """
    Return the averaged sub‑gradient of the maximum convolution entry.
    ``active`` is an array of indices where the convolution attains (or
    nearly attains) its maximum.
    """
    n = a.shape[0]
    g = np.zeros_like(a)
    for k in active:
        low = max(0, k - n + 1)
        high = min(n - 1, k)
        idx = np.arange(low, high + 1, dtype=int)
        partner = k - idx
        g[idx] += 2.0 * a[partner]
    # average over the active set
    g /= float(active.size)
    return g

def uniform_shape(n):
    return np.full(n, 1.0 / n)


def linear_ramp_shape(n):
    """Decreasing linear ramp a_i ∝ (n-i)."""
    w = np.arange(n, 0, -1, dtype=float)
    w /= w.sum()
    return w


def exponential_shape(lam, n):
    """a_i ∝ exp(-lam·i).  lam >= 0."""
    i = np.arange(n, dtype=float)
    w = np.exp(-lam * i)
    w /= w.sum()
    return w


def powerlaw_shape(p, n):
    """a_i ∝ (i+1)^{-p}   (p > 0)."""
    i = np.arange(1, n + 1, dtype=float)
    w = i ** (-p)
    w /= w.sum()
    return w


def polynomial_shape(p, n):
    """a_i ∝ 1 - (i/(n‑1))**p   (p >= 0)."""
    i = np.arange(n, dtype=float)
    frac = i / (n - 1)
    w = 1.0 - np.power(frac, p)
    w = np.clip(w, 0.0, None)
    w /= w.sum()
    return w


def beta_shape(alpha, beta, n):
    """Discretised Beta(α,β) density on the mid‑points of n bins."""
    x = (np.arange(n) + 0.5) / n
    # work in log‑space to avoid overflow
    logw = (alpha - 1.0) * np.log(x) + (beta - 1.0) * np.log(1.0 - x)
    logw = np.clip(logw, -700.0, 700.0)
    w = np.exp(logw)
    if w.sum() == 0 or not np.isfinite(w.sum()):
        w = np.ones_like(w)
    w /= w.sum()
    return w


def logistic_shape(mu, sigma, n):
    """a_i ∝ 1 / (1 + exp((i-mu)/sigma)).  mu in [0,n), sigma>0."""
    i = np.arange(n, dtype=float)
    w = 1.0 / (1.0 + np.exp((i - mu) / sigma))
    w = np.clip(w, 0.0, None)
    w /= w.sum()
    return w

def propose_candidate(seed: int = 0, budget_s: int = 1000):
    """
    Search for a non‑negative step‑function (list of floats) that minimises
    ``evaluate_sequence``.  The routine respects a time budget and returns
    the best sequence discovered.
    """
    np.random.seed(seed)
    start = time.time()
    # keep a small safety margin for the outer harness
    deadline = start + budget_s - 5.0

    # --------------------------------------------------------------
    # Determine the length ``n`` of the step function.
    # --------------------------------------------------------------
    try:
        n = len(height_sequence_1)      # type: ignore[name-defined]
    except Exception:
        n = 5146                         # fallback from the prompt

    # --------------------------------------------------------------
    # Book‑keeping of the global best and a small top‑K list.
    # --------------------------------------------------------------
    best_seq = None
    best_score = np.inf
    TOP_K = 5
    top = []   # list of (score, sequence)

    def consider(candidate):
        """Evaluate *candidate* (already normalised) and update the best lists."""
        nonlocal best_seq, best_score, top
        score = compute_objective(candidate)
        if score < best_score:
            best_score = score
            best_seq = candidate.copy()
            print(f"NEW GLOBAL BEST: {best_score:.6f}")
        # maintain a sorted top‑K list
        if len(top) < TOP_K:
            top.append((score, candidate.copy()))
            top.sort(key=lambda x: x[0])
        else:
            if score < top[-1][0]:
                top[-1] = (score, candidate.copy())
                top.sort(key=lambda x: x[0])

    # --------------------------------------------------------------
    # 1) Deterministic families of seeds
    # --------------------------------------------------------------
    consider(uniform_shape(n))
    consider(linear_ramp_shape(n))

    # Exponential – sweep a wide range (log‑space)
    for lam in np.logspace(-3, 1.5, 15):   # 10^{-3} … 10^{1.5} (~31)
        consider(exponential_shape(lam, n))

    # Power‑law – several exponents
    for p in np.linspace(0.1, 5.0, 20):
        consider(powerlaw_shape(p, n))

    # Polynomial shape (the previous family)
    for p in [0.5, 1.0, 1.5, 2.0, 3.0]:
        consider(polynomial_shape(p, n))

    # Logistic – a few hand‑picked parameters
    for mu in [n * 0.2, n * 0.5, n * 0.8]:
        for sigma in [n * 0.05, n * 0.1, n * 0.2]:
            consider(logistic_shape(mu, sigma, n))

    # --------------------------------------------------------------
    # 2) Systematic Beta‑grid (coarse)
    # --------------------------------------------------------------
    alphas = np.linspace(0.2, 5.0, 12)
    betas = np.linspace(0.2, 5.0, 12)
    for a in alphas:
        for b in betas:
            if time.time() >= deadline:
                break
            consider(beta_shape(a, b, n))
        if time.time() >= deadline:
            break

    # --------------------------------------------------------------
    # 3) Randomised seeds (Dirichlet‑like and random Beta)
    # --------------------------------------------------------------
    for _ in range(200):
        if time.time() >= deadline:
            break
        # Dirichlet‑like (Gamma with random concentration)
        conc = np.random.uniform(0.1, 3.0, size=n)
        g = np.random.gamma(conc, 1.0)
        g /= g.sum()
        consider(g)

        # Random Beta parameters (allow very small α,β)
        a = np.random.uniform(0.05, 10.0)
        b = np.random.uniform(0.05, 10.0)
        consider(beta_shape(a, b, n))

    # --------------------------------------------------------------
    # 4) Convex combinations of the current top candidates
    # --------------------------------------------------------------
    for _ in range(150):
        if time.time() >= deadline:
            break
        i, j = np.random.choice(len(top), 2, replace=False)
        w = np.random.rand()
        mix = w * top[i][1] + (1.0 - w) * top[j][1]
        mix /= mix.sum()
        consider(mix)

    # --------------------------------------------------------------
    # 5) Projected sub‑gradient descent with momentum.
    # --------------------------------------------------------------
    if best_seq is None:
        # fallback – uniform (should never happen)
        best_seq = uniform_shape(n)
        best_score = compute_objective(best_seq)

    a = best_seq.copy()
    prev_a = a.copy()
    eta0 = 0.9               # base step size
    momentum = 0.9
    iteration = 0
    stagnation = 0
    max_iter = 800000        # generous cap; will be cut by deadline

    while time.time() < deadline and iteration < max_iter:
        # ----- convolution and current objective -----
        fft = np.fft.rfft(a, n=2 * n - 1)
        conv = np.fft.irfft(fft * fft, n=2 * n - 1)
        cur_max = conv.max()
        cur_score = 2.0 * n * cur_max

        # keep global best
        if cur_score < best_score - 1e-12:
            best_score = cur_score
            best_seq = a.copy()
            print(f"Iter {iteration:6d}: NEW BEST {best_score:.6f}")
            stagnation = 0
        else:
            stagnation += 1

        # ----- active set (indices within a tiny tolerance of the max) -----
        eps = cur_max * 1e-6
        active = np.where(conv >= cur_max - eps)[0]
        if active.size == 0:               # safety fallback
            active = np.array([np.argmax(conv)], dtype=int)

        # ----- sub‑gradient for the active set -----
        g = subgradient_active(a, active)

        # ----- step‑size schedule -----
        step = eta0 / np.sqrt(iteration + 1)

        # ----- Nesterov‑style update + simplex projection -----
        a_next = a - step * g + momentum * (a - prev_a)
        a_next = simplex_projection(a_next)

        prev_a, a = a, a_next
        iteration += 1

        # ----- occasional random perturbation to escape plateaus -----
        if stagnation >= 25000:
            sigma = max(0.01, 0.5 / np.sqrt(iteration + 1))
            noise = sigma * np.random.randn(n) * a
            a = a + noise
            a = np.clip(a, 0.0, None)
            a /= a.sum()
            prev_a = a.copy()
            stagnation = 0
            print(f"Iter {iteration:6d}: perturbation, score {cur_score:.6f}")

        # early exit if we already hit the target bound
        if best_score <= 1.503:
            print(f"Target reached at iteration {iteration}, score = {best_score:.6f}")
            break

    # --------------------------------------------------------------
    # 6) Return the best candidate as a plain Python list.
    # --------------------------------------------------------------
    print(f"Finished after {iteration} iterations, best score = {best_score:.6f}")
    # Convert to list[float] – the external driver expects a plain list.
    return [float(v) for v in best_seq.tolist()]
