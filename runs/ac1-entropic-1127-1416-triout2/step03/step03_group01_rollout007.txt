"""
Search for a low‐max self‑convolution (step) sequence.

* Exploration *
    – Try several lengths (1024 … 8192).
    – Sample many “beta‑family’’ shapes with α,β drawn log‑uniformly in
      (0.01, 5) and a tiny ε.  Also sample sparse uniform vectors and the
      classic arcsine shape.
    – Keep the 3 best candidates (by the cheap surrogate 2·n·max(a*a)) per
      length.

* Refinement *
    – For each kept candidate run an Adam optimiser on the smooth
      soft‑max surrogate of the max‑convolution.
          λ (temperature) is annealed from 0.5 → 200, giving a tighter
      approximation to the true max.
    – After Adam a few hundred sub‑gradient steps on the *exact* max
      (indicator weight) are performed; this often pushes the solution
      a little further.
    – The best refined vector (still normalised to Σa=1) is evaluated with
      the official ``evaluate_sequence`` function.

* Final polishing *
    – The globally best vector is polished once more with a longer Adam run
      (λ → 400) and a final sub‑gradient phase.

The routine respects the 1000 s time limit, never returns an empty list,
and prints progress information.
"""

import time
import numpy as np
from typing import List, Tuple

# ----------------------------------------------------------------------
# Low‑level utilities (no closures)
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Linear convolution of two 1‑D real vectors using FFT."""
    m = len(a) + len(b) - 1
    n_fft = next_pow2(m)
    A = np.fft.rfft(a, n=n_fft)
    B = np.fft.rfft(b, n=n_fft)
    C = A * B
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the probability simplex
    {x ≥ 0, Σx = 1}.  Implements the algorithm of Duchi et al. (2008).
    """
    if np.all(v <= 0):
        # all non‑positive – return the uniform distribution
        return np.full_like(v, 1.0 / v.size)

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho_arr = np.where(u - cssv / np.arange(1, v.size + 1) > 0)[0]
    if rho_arr.size == 0:
        theta = 0.0
    else:
        rho = rho_arr[-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def compute_score(a: np.ndarray) -> float:
    """
    Fast surrogate for the official score.
    The vector is assumed to be normalised to Σa = 1, so the
    denominator disappears and the score reduces to 2·n·max(a*a).
    """
    n = len(a)
    conv = convolution_fft(a, a)
    max_val = float(np.max(conv))
    return 2.0 * n * max_val


def arcsine_initial(n: int) -> np.ndarray:
    """Classic arcsine shape a_i ∝ 1/√((i+½)(n‑½‑i))."""
    i = np.arange(n, dtype=float)
    eps = 0.25
    vals = 1.0 / np.sqrt((i + 0.5) * (n - 0.5 - i) + eps)
    vals /= vals.sum()
    return vals


def generate_beta_shape(n: int, alpha: float, beta: float, eps: float) -> np.ndarray:
    """
    Symmetric beta‑family density on {0,…,n‑1}:
        a_i ∝ (i+ε)^{α‑1}·(n‑1‑i+ε)^{β‑1}.
    The result is normalised to Σa = 1.
    """
    i = np.arange(n, dtype=float)
    left = (i + eps) ** (alpha - 1.0)
    right = (n - 1 - i + eps) ** (beta - 1.0)
    base = left * right
    total = base.sum()
    if total <= 0.0 or not np.isfinite(total):
        return np.full(n, 1.0 / n)
    return base / total


def generate_sparse_uniform(n: int, support_frac: float = 0.02, rng: np.random.Generator = None) -> np.ndarray:
    """
    Uniform mass on a random sparse support.
    support_frac – fraction of indices that receive weight (≈ √n / n works well).
    """
    if rng is None:
        rng = np.random.default_rng()
    s = max(2, int(np.ceil(support_frac * n)))
    idx = rng.choice(n, size=s, replace=False)
    a = np.zeros(n, dtype=float)
    a[idx] = 1.0 / s
    return a


# ----------------------------------------------------------------------
# Smooth‑max gradient (used by Adam)
# ----------------------------------------------------------------------


def smooth_gradient(a: np.ndarray, lam: float) -> Tuple[np.ndarray, float]:
    """
    Gradient of the smooth approximation of the objective
    f(a) = 2·n·max(a*a) / (Σa)², where the max is replaced by the
    log‑sum‑exp with temperature λ.
    Returns (gradient, smoothed_max_value).
    """
    n = len(a)
    conv = convolution_fft(a, a)                     # length 2n‑1
    max_c = np.max(conv)

    # stable soft‑max weights
    exp_term = np.exp(lam * (conv - max_c))
    sum_exp = np.sum(exp_term)
    w = exp_term / sum_exp                           # Σw = 1

    # smoothed max (the surrogate for max conv)
    B = max_c + np.log(sum_exp) / lam

    # Weighted gradient of Σ_k w_k (a*a)_k
    w_rev = w[::-1]
    conv_rev = convolution_fft(a, w_rev)            # length 3n‑2
    L = 2 * n - 1
    weighted_grad = 2.0 * conv_rev[L - 1 - np.arange(n)]

    s = np.sum(a)
    term1 = (2.0 * n / (s ** 2)) * weighted_grad
    term2 = -(4.0 * n * B / (s ** 3)) * np.ones_like(a)
    grad = term1 + term2
    return grad, B

# Adam optimiser (soft‑max surrogate)

def adam_optimize(initial_a: np.ndarray,
                  deadline: float,
                  max_iter: int = 60_000,
                  lr0: float = 0.5,
                  lam_start: float = 0.5,
                  lam_end: float = 200.0,
                  noise_every: int = 3000,
                  noise_scale: float = 1e-4) -> Tuple[np.ndarray, float]:
    """
    Adam optimiser on the smooth surrogate.
    Returns the best vector (projected on the simplex) and its fast score.
    """
    a = project_to_simplex(initial_a.astype(float).copy())
    best_a = a.copy()
    best_sc = compute_score(a)

    beta1, beta2 = 0.9, 0.999
    eps = 1e-8
    m = np.zeros_like(a)
    v = np.zeros_like(a)

    n = len(a)

    for it in range(1, max_iter + 1):
        if time.time() >= deadline:
            break

        # Linear schedule for λ
        lam = lam_start + (lam_end - lam_start) * min(1.0, it / max_iter)

        grad, _ = smooth_gradient(a, lam)

        # Adam moments
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)

        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        step = lr0 * m_hat / (np.sqrt(v_hat) + eps)
        a_new = a - step

        if noise_every and (it % noise_every == 0):
            a_new += np.random.randn(n) * noise_scale

        a_new = project_to_simplex(a_new)

        cur_sc = compute_score(a_new)
        if cur_sc < best_sc:
            best_sc = cur_sc
            best_a = a_new.copy()

        a = a_new

        # occasional status
        if it % 10_000 == 0:
            print(f"    Adam iter={it:,d}  cur={cur_sc:.6f}  best={best_sc:.6f}")

    return best_a, best_sc

def max_convolution(a: np.ndarray) -> Tuple[np.ndarray, int]:
    """Return (full self‑convolution, index of the maximal entry)."""
    conv = convolution_fft(a, a)
    idx = int(np.argmax(conv))
    return conv, idx


def subgradient_of_max(a: np.ndarray, max_idx: int) -> np.ndarray:
    """
    Sub‑gradient of the *exact* maximum convolution term (indicator weight).
    Returns a vector of size n.
    """
    n = len(a)
    L = 2 * n - 1
    w = np.zeros(L, dtype=float)
    w[max_idx] = 1.0
    w_rev = w[::-1]
    conv_rev = convolution_fft(a, w_rev)          # length 3n‑2
    grad = 2.0 * conv_rev[L - 1 - np.arange(n)]
    return grad


def subgradient_descent(a: np.ndarray,
                        deadline: float,
                        max_iter: int = 30_000,
                        step0: float = 0.3,
                        patience: int = 2500) -> Tuple[np.ndarray, float]:
    """
    Simple projected sub‑gradient descent on the true max‑convolution.
    Uses a decreasing step size when no improvement is seen.
    """
    a = project_to_simplex(a.copy())
    best_a = a.copy()
    best_sc = compute_score(a)

    step = step0
    stale = 0

    for it in range(1, max_iter + 1):
        if time.time() >= deadline:
            break

        conv, max_idx = max_convolution(a)
        grad = subgradient_of_max(a, max_idx)

        a_new = a - step * grad
        a_new = project_to_simplex(a_new)

        cur_sc = compute_score(a_new)
        if cur_sc < best_sc:
            best_sc = cur_sc
            best_a = a_new.copy()
            stale = 0
        else:
            stale += 1
            if stale >= patience:
                step *= 0.5
                stale = 0
                if step < 1e-6:
                    break

        a = a_new

        if it % 5000 == 0:
            print(f"    Sub‑grad iter={it:,d}  cur={cur_sc:.6f}  best={best_sc:.6f}")

    return best_a, best_sc


# ----------------------------------------------------------------------
# Candidate generation (exploration)
# ----------------------------------------------------------------------


def sample_beta_candidates(n: int,
                           rng: np.random.Generator,
                           n_samples: int = 300) -> List[np.ndarray]:
    """Return a list of beta‑family candidates (not normalised)."""
    cand = []
    for _ in range(n_samples):
        # log‑uniform sampling for α,β
        log_a = rng.uniform(np.log(0.01), np.log(5.0))
        log_b = rng.uniform(np.log(0.01), np.log(5.0))
        alpha = np.exp(log_a)
        beta = np.exp(log_b)
        # ε between 1e‑6 and 0.5 (log‑uniform)
        eps = 10 ** rng.uniform(-6, -0.3)  # ≈ 1e‑6 … 0.5
        vec = generate_beta_shape(n, alpha, beta, eps)
        cand.append(vec)
    return cand


def sample_sparse_candidates(n: int,
                             rng: np.random.Generator,
                             n_samples: int = 30) -> List[np.ndarray]:
    """Return a list of sparse‑uniform candidates."""
    cand = []
    for _ in range(n_samples):
        vec = generate_sparse_uniform(n, support_frac=0.02, rng=rng)
        cand.append(vec)
    return cand


# ----------------------------------------------------------------------
# Refinement pipeline for a single candidate
# ----------------------------------------------------------------------


def refine_candidate(a: np.ndarray, deadline: float) -> Tuple[np.ndarray, float]:
    """
    Run Adam on the soft‑max surrogate, then a few sub‑gradient steps.
    Returns the best refined vector and its fast score.
    """
    remaining = deadline - time.time()
    if remaining <= 0:
        return a, compute_score(a)

    # -------- Adam phase -------------------------------------------------
    max_iter_adam = int(min(80_000, remaining * 20))
    a_adam, _ = adam_optimize(
        initial_a=a,
        deadline=deadline,
        max_iter=max_iter_adam,
        lr0=0.5,
        lam_start=0.5,
        lam_end=200.0,
        noise_every=2500,
        noise_scale=5e-5,
    )

    # -------- Sub‑gradient phase -----------------------------------------
    remaining = deadline - time.time()
    if remaining > 0.5:
        max_iter_sub = int(min(40_000, remaining * 25))
        a_sub, _ = subgradient_descent(
            a=a_adam,
            deadline=deadline,
            max_iter=max_iter_sub,
            step0=0.4,
            patience=3000,
        )
    else:
        a_sub = a_adam

    return a_sub, compute_score(a_sub)

# Main entry point required by the competition harness
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 123,
                     budget_s: int = 1000,
                     **kwargs) -> List[float]:
    """
    Search for a non‑negative step sequence with a low ``evaluate_sequence``
    value.  The approach combines broad random sampling, Adam optimisation,
    a final sub‑gradient polishing, and a short global polishing run.
    """
    rng = np.random.default_rng(seed)
    start = time.time()
    deadline = start + budget_s - 2.0          # keep a small safety margin

    best_seq: List[float] = []
    best_val: float = float("inf")

    # --------------------------------------------------------------
    # 0.  Optional fallback from a previous run (if available)
    # --------------------------------------------------------------
    try:
        fallback = [float(x) for x in height_sequence_1]  # type: ignore
        val = evaluate_sequence(fallback)
        print(f"Fallback (height_sequence_1) score = {val:.6f}")
        if val < best_val:
            best_val = val
            best_seq = fallback
    except Exception:
        pass

    # --------------------------------------------------------------
    # 1.  Exploration over several lengths
    # --------------------------------------------------------------
    lengths = [1024, 2048, 4096, 8192]

    for n in lengths:
        if time.time() >= deadline - 40.0:
            break
        print(f"\n=== Exploring length n = {n} ===")

        # list of (candidate, fast_score)
        pool: List[Tuple[np.ndarray, float]] = []

        # deterministic starters
        for starter in (np.full(n, 1.0 / n), arcsine_initial(n)):
            sc = compute_score(starter)
            pool.append((starter, sc))

        # random beta‑family samples
        for cand in sample_beta_candidates(n, rng, n_samples=300):
            sc = compute_score(cand)
            pool.append((cand, sc))

        # sparse uniform samples
        for cand in sample_sparse_candidates(n, rng, n_samples=30):
            sc = compute_score(cand)
            pool.append((cand, sc))

        # keep the top‑k candidates for refinement
        pool.sort(key=lambda x: x[1])
        top_k = 3
        keep = pool[:top_k]
        print(f"  Best fast scores (pre‑refine): {[f'{s:.6f}' for _, s in keep]}")

        # --------------------------------------------------------------
        # 2.  Refine each kept candidate
        # --------------------------------------------------------------
        for idx, (cand, fast_sc) in enumerate(keep, 1):
            if time.time() >= deadline - 30.0:
                break
            print(f"  Refining candidate {idx}/{len(keep)} (fast score = {fast_sc:.6f}) …")
            refined, refined_fast = refine_candidate(cand, deadline)

            # evaluate with the *official* (slow) evaluator
            val = evaluate_sequence(refined.tolist())
            print(f"    → official score = {val:.6f}")

            if val < best_val:
                best_val = val
                best_seq = refined.tolist()
                print(f"    *** New global best: {best_val:.6f} (n={n}) ***")

    # --------------------------------------------------------------
    # 3.  Final polishing of the overall best vector (if any)
    # --------------------------------------------------------------
    if best_seq and time.time() < deadline - 5.0:
        print("\nFinal polishing of the global best …")
        a = np.array(best_seq, dtype=float)
        # a longer Adam run with a higher temperature ceiling
        a_refined, _ = adam_optimize(
            initial_a=a,
            deadline=deadline,
            max_iter=int(min(250_000, (deadline - time.time()) * 30)),
            lr0=0.3,
            lam_start=0.5,
            lam_end=400.0,
            noise_every=2000,
            noise_scale=3e-5,
        )
        # one more sub‑gradient pass
        a_refined, _ = subgradient_descent(
            a_refined,
            deadline=deadline,
            max_iter=int(min(80_000, (deadline - time.time()) * 30)),
            step0=0.2,
            patience=4000,
        )
        final_val = evaluate_sequence(a_refined.tolist())
        print(f"Final polished official score = {final_val:.6f}")

        if final_val < best_val:
            best_val = final_val
            best_seq = a_refined.tolist()
            print("*** Updated global best after final polishing ***")

    # --------------------------------------------------------------
    # 4.  Guard – guarantee a return value
    # --------------------------------------------------------------
    if not best_seq:
        # Very crude safe fallback (uniform 256‑vector)
        best_seq = [1.0 / 256.0] * 256
        best_val = evaluate_sequence(best_seq)

    print(f"\nFinished search. Best reported official score = {best_val:.6f}")
    return best_seq
