The score  

\[
S(a)=\frac{2\,n\; \max (a*a)}{(\sum_i a_i)^2}
\]

is invariant under scaling, therefore we may restrict ourselves to probability
vectors (`∑ a_i = 1`).  
`evaluate_sequence` then equals `2·n·max(a*a)`.  
The task is to find a non‑negative vector that makes the self‑convolution as
flat as possible.

The previous implementation used an *incorrect* gradient for the soft‑max
approximation of the max‑convolution – the diagonal contribution was subtracted
instead of being kept.  With the correct gradient the projected Adam optimiser
converges to a much better shape.

The new algorithm consists of three phases

1. **Multiple random starts** (uniform, arcsine, triangular and Dirichlet draws)
   for several problem sizes (`n = 2048, 4096, 8192`).
2. **Projected Adam on the smoothed max**  
   – soft‑max smoothing parameter `α` is increased linearly from `1` to `2000`.  
   – the true gradient of the smoothed objective is used (no diagonal subtraction).

3. **Stochastic hill‑climbing (mass transfer)** on the best vector found so far,

"""
Fast optimisation of the self‑convolution peak.

*   Uses a *correct* gradient of the soft‑max approximation of `max(a*a)`.
*   Projects on the probability simplex after each update.
*   Runs several multi‑start Adam optimisations for lengths 2048‑8192.
*   Finishes with an adaptive hill‑climber that moves small amounts of mass.
*   Returns the best sequence discovered before the time budget expires.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Basic utilities
# ----------------------------------------------------------------------


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of a vector onto the probability simplex."""
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / v.shape[0])

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, v.shape[0] + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.where(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full convolution a * b using a real FFT (fast)."""
    m = len(a) + len(b) - 1
    n_fft = next_pow2(m)
    A = np.fft.rfft(a, n=n_fft)
    B = np.fft.rfft(b, n=n_fft)
    C = A * B
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


def compute_score(a: np.ndarray) -> float:
    """Fast replica of `evaluate_sequence` for a probability vector (∑a = 1)."""
    n = a.shape[0]
    conv = convolution_fft(a, a)
    return 2.0 * n * float(np.max(conv))

# Initial shapes
# ----------------------------------------------------------------------


def arcsine_initial(n: int) -> np.ndarray:
    """U‑shaped arcsine start (β = ½)."""
    i = np.arange(n, dtype=float)
    eps = 0.25
    vals = 1.0 / np.sqrt((i + 0.5) * (n - 0.5 - i) + eps)
    return project_to_simplex(vals)


def triangular_initial(n: int) -> np.ndarray:
    """Symmetric tent shape – grows linearly to the centre then decreases."""
    i = np.arange(n, dtype=float)
    mid = (n - 1) / 2.0
    vals = np.minimum(i, n - 1 - i) + 1.0
    return project_to_simplex(vals)


def uniform_initial(n: int) -> np.ndarray:
    """Flat distribution."""
    return np.full(n, 1.0 / n)

def smoothed_gradient(a: np.ndarray, alpha: float):

    Gradient of the soft‑max approximation of max(a*a).

    maxc = np.max(conv)
    exp_term = np.exp(alpha * (conv - maxc))
    w = exp_term / np.sum(exp_term)                 # Σ w = 1, length 2n‑1

    # Gradient = 2 * (a_rev * w)[n‑1 : n‑1+n]
    a_rev = a[::-1]
    conv_rev = convolution_fft(a_rev, w)            # length 3n‑2
    start = n - 1
    rev_slice = conv_rev[start:start + n]

    grad = 2.0 * rev_slice
    return grad, conv


# ----------------------------------------------------------------------
# Projected Adam optimiser (uses the correct gradient)
# ----------------------------------------------------------------------


def adam_optimize(
    init_a: np.ndarray,
    deadline: float,
    max_iter: int = 200_000,
    eta: float = 0.02,
    beta1: float = 0.9,
    beta2: float = 0.999,
    eps: float = 1e-8,
    alpha_start: float = 1.0,
    alpha_end: float = 2000.0,
    noise_every: int = 800,
    noise_scale: float = 2e-4,
):
    """
    Projected Adam on the smoothed max‑convolution.

    Returns the best vector (as ndarray) found and its exact score.
    """
    a = project_to_simplex(init_a.astype(float).copy())
    m = np.zeros_like(a)
    v = np.zeros_like(a)

    best_a = a.copy()
    # we already have the full convolution from the first iteration later
    best_score = compute_score(best_a)

    n = len(a)

    for t in range(1, max_iter + 1):
        if time.time() >= deadline:
            break

        # linear schedule for the smoothing parameter
        alpha = alpha_start + (alpha_end - alpha_start) * min(1.0, t / max_iter)

        grad, conv = smoothed_gradient(a, alpha)

        # Adam moments
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)

        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)

        # optional decreasing stepsize
        lr = eta * (1.0 - t / max_iter)

        step = lr * m_hat / (np.sqrt(v_hat) + eps)
        a_new = a - step

        # occasional isotropic noise to escape plateaus
        if noise_every and (t % noise_every == 0):
            a_new += np.random.randn(n) * noise_scale

        a_new = project_to_simplex(a_new)

        cur_score = 2.0 * n * float(np.max(conv))   # exact score for this iterate
        if cur_score < best_score:
            best_score = cur_score
            best_a = a_new.copy()

        a = a_new

        if t % 5000 == 0:
            print(
                f"    Adam iter {t:,d} – cur {cur_score:.6f} – best {best_score:.6f}"
            )

    return best_a, best_score


# ----------------------------------------------------------------------
# Simple stochastic hill‑climber (mass‑transfer moves)
# ----------------------------------------------------------------------


def hill_climb(seq, deadline, max_steps=150_000):
    """
    Moves a small fraction of the mass from one coordinate to another.
    Accepts only improving moves; the step size adapts automatically.
    """
    a = np.array(seq, dtype=float)
    n = a.shape[0]

    best_a = a.copy()
    best_score = compute_score(best_a)

    step_scale = 0.1          # initial fraction of mass to move

    while max_steps > 0 and time.time() < deadline:
        i, j = np.random.randint(n, size=2)
        if i == j:
            continue

        # transfer a fraction of the mass from i to j
        delta = step_scale * best_a[i]
        if delta <= 0.0:
            continue
        cand = best_a.copy()
        cand[i] -= delta
        cand[j] += delta
        # projection keeps the vector on the simplex and removes tiny negatives
        cand = project_to_simplex(cand)

        cur_score = compute_score(cand)

        if cur_score < best_score:
            best_score = cur_score
            best_a = cand
            step_scale = min(step_scale * 1.05, 0.5)   # enlarge after success
        else:
            step_scale *= 0.97                         # shrink after failure

        max_steps -= 1

    return best_a.tolist(), best_score


# ----------------------------------------------------------------------
# Main entry point required by the evaluation harness
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative sequence that minimises `evaluate_sequence`.
    The routine respects the given time budget and prints intermediate scores.
    """
    np.random.seed(seed)
    start_time = time.time()
    # leave a little slack for the harness
    deadline = start_time + budget_s - 2.0

    best_seq = None
    best_val = float("inf")

    # ------------------------------------------------------------------
    # Optional fallback (if a previous construction is available)
    # ------------------------------------------------------------------
    try:
        fallback = [float(x) for x in height_sequence_1]   # type: ignore
        fallback_val = evaluate_sequence(fallback)
        print(f"Fallback score = {fallback_val:.6f}")
        if fallback_val < best_val:
            best_val = fallback_val
            best_seq = fallback
    except Exception:
        pass

    # ------------------------------------------------------------------
    # Parameter grid for the multi‑start Adam phase
    # ------------------------------------------------------------------
    lengths = [2048, 4096, 8192]   # more points → more freedom
    dirichlet_alphas = [0.3, 0.8, 2.0]

    for n in lengths:
        if time.time() >= deadline - 5.0:
            break
        print(f"\n=== Optimising length n = {n} ===")

        # a small collection of sensible starts
        initial_vectors = [
            arcsine_initial(n),
            triangular_initial(n),
            uniform_initial(n),
        ]

        for a_param in dirichlet_alphas:
            vec = np.random.dirichlet(np.full(n, a_param))
            initial_vectors.append(vec)

        # run a short Adam optimisation for each start
        for idx, init_vec in enumerate(initial_vectors, start=1):
            if time.time() >= deadline - 5.0:
                break
            print(f"  start {idx}/{len(initial_vectors)}")
            # allocate iterations proportional to remaining time
            remaining = deadline - time.time()
            # very rough estimate: 30 iterations per millisecond => 30 000 per sec
            iters = int(min(250_000, remaining * 30))
            seq, val = adam_optimize(
                init_a=init_vec,
                deadline=deadline,
                max_iter=iters,
                eta=0.02,
                alpha_start=1.0,
                alpha_end=2000.0,
                noise_every=1000,
                noise_scale=1e-4,
            )
            print(f"    Adam best score = {val:.6f}")
            if val < best_val:
                best_val = val
                best_seq = seq.tolist()
                print(f"    *** Global best updated to {best_val:.6f} (n={n}) ***")

    # ------------------------------------------------------------------
    # Final refinement with hill‑climbing
    # ------------------------------------------------------------------
    if best_seq is not None and time.time() < deadline - 1.0:
        print("\nRefining best sequence with hill‑climb")
        refined_seq, refined_val = hill_climb(
            best_seq, deadline=deadline, max_steps=200_000
        )
        print(f"    refined score = {refined_val:.6f}")
        if refined_val < best_val:
            best_val = refined_val
            best_seq = refined_seq

    # ------------------------------------------------------------------
    # Safeguard: trivial uniform vector if everything else failed
    # ------------------------------------------------------------------
    if best_seq is None:
        best_seq = [1.0 / 256] * 256
        best_val = evaluate_sequence(best_seq)

    final_score = evaluate_sequence(best_seq)
    print(f"\nFinished search. Final reported score = {final_score:.6f}")

    # The harness expects a plain Python list of floats.
    return best_seq
