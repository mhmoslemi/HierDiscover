def propose_candidate(seed=0, budget_s=1000, **kwargs):
    import time, numpy as np
    np.random.seed(seed)
    start = time.time()
    deadline = start + budget_s - 2

    best_seq = None
    best_score = float('inf')

    # param families
    lengths = [16384, 8192, 4096, 2048]
    exponents = [-0.9, -0.8, -0.7, -0.6, -0.55, -0.5, -0.45, -0.4, -0.3, -0.2, -0.1]
    concs = [0.2, 0.5, 1.0, 2.0, 5.0]  # Dirichlet concentration parameters

    for n in lengths:
        if time.time() >= deadline - 5:
            break
        print(f'\nOptimizing length {n}')
        
        # simple parametric families
        candidates = []

        # arcsine family with exponents
        for e in exponents:
            i = np.arange(n, dtype=float)
            prod = (i + 0.5) * (n - 0.5 - i) + 1e-6
            vec = prod**e
            vec = project_to_simplex(vec)
            candidates.append(vec)

        # uniform
        candidates.append(np.full(n, 1.0/n))

        # Dirichlet random draws
        for a in concs:
            for _ in range(2):  # 2 draws each
                if time.time() >= deadline - 5:
                    break
                vec = np.random.dirichlet(np.full(n, a))
                candidates.append(vec)

        # For each candidate compute score and keep top few
        # Maybe we keep top K=5 based on compute_score
        scored = []
        for vec in candidates:
            sc = compute_score(vec)  # fast with sum=1
            scored.append((sc, vec))
        scored.sort(key=lambda x: x[0])
        top = scored[:5]
        for rank, (sc0, vec0) in enumerate(top, start=1):
            if sc0 < best_score:
                best_score = sc0
                best_seq = vec0.tolist()
                print(f'  initial candidate rank {rank} score {sc0:.6f} -> best')
            # run ADAM on each top candidate
            if time.time() >= deadline - 5:
                break
            print(f'  ADAM on cand rank {rank} start score {sc0:.6f}')
            # allocate time for this run
            remaining = deadline - time.time()
            # set max_iter proportionally to remaining/len(top)
            # but also cap
            max_iter = int(min(200000, remaining * 30))
            if max_iter <= 0:
                continue
            a_opt, best_adam_score = adam_optimize(
                init_a=vec0,
                time_deadline=deadline,
                max_iter=max_iter,
                eta=0.02,
                alpha_start=1.0,
                alpha_end=5000.0,  # maybe larger for sharper max
                noise_every=2000,
                noise_scale=1e-4,
            )
            if best_adam_score < best_score:
                best_score = best_adam_score
                best_seq = a_opt.tolist()
                print(f'    NEW GLOBAL BEST after ADAM: {best_score:.6f}')
        # after processing all top candidates for this n, do hill climbing on current global best (if same n)

"""
Propose a low‑score non‑negative step function.

Algorithm
---------
1.  Sample a few parametric families (arcsine‑like with tunable exponent,
    uniform, Dirichlet draws and mixtures) for several sequence lengths.
2.  Keep the best few candidates (by the fast “sum‑=‑1” score) and improve
    them with a projected Adam optimiser on a smooth approximation of the
    max‑convolution.  The smoothing parameter α is increased from 1 to a
    large value (≈5 000) – a homotopy that lets the optimiser first move
    quickly and then focus on the true objective.
3.  The globally best vector found so far is finally refined by a
    simulated‑annealing style hill‑climber that moves small fractions of
    mass between positions, accepting worsening moves with a temperature
    schedule.  All iterates are projected onto the probability simplex so
    the sum stays 1.
4.  The best sequence (as a plain Python list) is returned.  The function
    respects the 1000 s time budget; intermediate progress is printed.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Basic linear‑algebra / FFT utilities
# ----------------------------------------------------------------------


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex (non‑negative, sum = 1)."""
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / v.shape[0])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, v.shape[0] + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.where(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full (linear) convolution a * b using a real FFT (fast)."""
    m = len(a) + len(b) - 1
    n_fft = next_pow2(m)
    A = np.fft.rfft(a, n=n_fft)
    B = np.fft.rfft(b, n=n_fft)
    C = A * B
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


def compute_score(a: np.ndarray) -> float:
    """
    Fast replica of `evaluate_sequence` for a vector normalised to sum=1.
    Returns 2·n·max(a*a) (the value that evaluate_sequence would return).
    """
    n = a.shape[0]
    conv = convolution_fft(a, a)
    return 2.0 * n * float(np.max(conv))


# ----------------------------------------------------------------------
# Smoothed‑max gradient (used by Adam)
# ----------------------------------------------------------------------


def smoothed_gradient(a: np.ndarray, alpha: float):
    """
    Gradient of the soft‑max approximation of max(a*a):
        f_α(a) = (1/α)·log Σ_k exp(α·(a*a)_k)

    Returns (gradient, raw convolution).
    """
    n = a.shape[0]
    # self‑convolution (length 2n‑1)
    conv = convolution_fft(a, a)

    # soft‑max weights (stable formulation)
    maxc = np.max(conv)
    exp_term = np.exp(alpha * (conv - maxc))
    w = exp_term / np.sum(exp_term)          # Σ w_k = 1, length 2n‑1

    # convolution of reversed a with the weight vector
    a_rev = a[::-1]
    conv_rev = convolution_fft(a_rev, w)    # length n+(2n‑1)-1 = 3n‑2
    start = n - 1
    rev_slice = conv_rev[start:start + n]   # needed for each component

    # diagonal contribution (w_{2i}) inside the smoothed max
    w_diag = w[::2][:n]

    grad = 2.0 * rev_slice - w_diag * a
    return grad, conv


def adam_optimize(
    init_a: np.ndarray,
    time_deadline: float,
    max_iter: int = 200_000,
    eta: float = 0.02,
    beta1: float = 0.9,
    beta2: float = 0.999,
    eps: float = 1e-8,
    alpha_start: float = 1.0,
    alpha_end: float = 5000.0,
    noise_every: int = 1000,
    noise_scale: float = 2e-4,
):
    """
    Projected Adam optimiser on the smoothed max‑convolution objective.
    Returns (best_vector, best_score) where the score is the true (unsmoothed)
    2·n·max convolution value.
    """
    a = project_to_simplex(init_a.astype(float).copy())
    m = np.zeros_like(a)
    v = np.zeros_like(a)

    best_a = a.copy()
    best_score = compute_score(a)

    n = len(a)

    for t in range(1, max_iter + 1):
        if time.time() >= time_deadline:
            break

        # homotopy on the smoothing parameter
        alpha = alpha_start + (alpha_end - alpha_start) * min(1.0, t / max_iter)

        grad, _ = smoothed_gradient(a, alpha)

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)

        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)

        step = eta * m_hat / (np.sqrt(v_hat) + eps)
        a_new = a - step

        # occasional isotropic noise to escape shallow plateaus
        if noise_every and (t % noise_every == 0):
            a_new += np.random.randn(n) * noise_scale

        a_new = project_to_simplex(a_new)

        cur_score = compute_score(a_new)
        if cur_score < best_score:
            best_score = cur_score
            best_a = a_new.copy()

        a = a_new

        # light progress output
        if t % 5_000 == 0:
            print(f"    Adam iter {t:,d} – cur {cur_score:.6f} – best {best_score:.6f}")

    return best_a, best_score


# ----------------------------------------------------------------------
# Simulated‑annealing hill‑climber (exact objective)
# ----------------------------------------------------------------------


def hill_climb_anneal(seq, time_deadline, max_steps=200_000,
                     init_temp=0.2, final_temp=1e-5):
    """
    Stochastic mass‑transfer hill‑climber with a simple temperature schedule.
    Works directly with the true objective (no smoothing).  Returns a tuple
    (best_sequence_as_list, best_score).
    """
    a = np.array(seq, dtype=float)
    a = a / np.sum(a)               # enforce sum = 1 (simple normalisation)
    n = a.shape[0]

    best_a = a.copy()
    best_score = compute_score(best_a)
    cur_a = a.copy()
    cur_score = best_score

    # step scale – fraction of source mass to move
    step_scale = 0.07

    for step in range(max_steps):
        if time.time() >= time_deadline:
            break

        # exponential temperature decay
        t_frac = step / max_steps
        temp = init_temp * (final_temp / init_temp) ** t_frac

        # propose a mass‑transfer move
        i, j = np.random.randint(n, size=2)
        if i == j:
            continue
        if cur_a[i] <= 0.0:
            continue
        delta = step_scale * cur_a[i]
        cand = cur_a.copy()
        cand[i] -= delta
        cand[j] += delta
        # safety – tiny negative values caused by rounding
        if cand[i] < 0.0:
            cand[i] = 0.0
        cand = project_to_simplex(cand)

        cand_score = compute_score(cand)

        # Metropolis acceptance
        if cand_score < cur_score or np.random.rand() < np.exp(-(cand_score - cur_score) / max(temp, 1e-12)):
            cur_a = cand
            cur_score = cand_score
            if cand_score < best_score:
                best_score = cand_score
                best_a = cand.copy()
                step_scale = min(step_scale * 1.02, 0.5)   # slightly enlarge moves after success
        else:
            step_scale = max(step_scale * 0.985, 1e-4)   # shrink a bit after rejection

        if step % 20000 == 0 and step > 0:
            print(f"    Hill‑climb step {step:,d} – cur {cur_score:.6f} – best {best_score:.6f}")

    return best_a.tolist(), best_score

# Helper to build families of initial vectors

def arcsine_like_vector(n: int, exponent: float) -> np.ndarray:
    """
    Returns a normalised vector proportional to ((i+0.5)*(n-0.5-i))**exponent.
    exponent = -0.5 reproduces the classical arcsine start.
    """
    i = np.arange(n, dtype=float)
    prod = (i + 0.5) * (n - 0.5 - i) + 1e-12
    v = prod ** exponent
    return project_to_simplex(v)


def mixture_vector(n: int, mix: float) -> np.ndarray:
    """
    Linear mixture of arcsine (exponent -0.5) and uniform:
        mix * arcsine + (1‑mix) * uniform
    """
    arcs = arcsine_like_vector(n, -0.5)
    uni = np.full(n, 1.0 / n)
    v = mix * arcs + (1.0 - mix) * uni
    return project_to_simplex(v)

        # Run Adam on each top candidate (time‑aware iteration budget)
        # ------------------------------------------------------------------
        for rank, (init_score, init_vec) in enumerate(top_candidates, start=1):
            if time.time() >= deadline - 5.0:
                break
            print(f"  Candidate {rank}/{top_k} – init score {init_score:.6f}")

            # Allocate iterations roughly proportional to the remaining time
            remaining = deadline - time.time()
            # Rough heuristic: 30 gradient steps per second is a safe lower bound
            max_iter = int(min(200_000, remaining * 30))

            # For the very large lengths we deliberately cut the iteration budget
            # to keep total wall‑clock time reasonable.
            if n >= 8192:
                max_iter = int(min(max_iter, 80_000))
            if n >= 16384:
                max_iter = int(min(max_iter, 50_000))

            if max_iter <= 0:
                continue

            a_opt, best_adam_score = adam_optimize(
                init_a=init_vec,
                time_deadline=deadline,
                max_iter=max_iter,
                eta=0.02,
                alpha_start=1.0,
                alpha_end=5000.0,
                noise_every=1500,
                noise_scale=1e-4,
            )
            print(f"    Adam best score = {best_adam_score:.6f}")

            if best_adam_score < best_val:
                best_val = best_adam_score
                best_seq = a_opt.tolist()
                print(f"    *** Global best improved to {best_val:.6f} (n={n}) ***")

        # ------------------------------------------------------------------
        # If we have a best vector of this length, give it a quick hill‑climb
        # ------------------------------------------------------------------
        if best_seq is not None and len(best_seq) == n and time.time() < deadline - 2.0:
            print("\n  Refining with hill‑climb (annealing) …")
            # allocate remaining time proportionally (but not more than 150 k steps)
            remaining = deadline - time.time()
            max_hc_steps = int(min(150_000, remaining * 30))
            refined_seq, refined_score = hill_climb_anneal(
                best_seq,
                time_deadline=deadline,
                max_steps=max_hc_steps,
                init_temp=0.2,
                final_temp=1e-5,
            )
            print(f"    Hill‑climb refined score = {refined_score:.6f}")
            if refined_score < best_val:
                best_val = refined_score
                best_seq = refined_seq
                print(f"    *** Global best updated after hill‑climb: {best_val:.6f} ***")

    # ------------------------------------------------------------------
    # Final sanity check – if nothing succeeded fall back to a tiny uniform vector.
    # ------------------------------------------------------------------
    if best_seq is None:
        best_seq = [1.0 / 256] * 256
        best_val = evaluate_sequence(best_seq)

    final_val = evaluate_sequence(best_seq)
    print(f"\nFinished search. Final reported score = {final_val:.6f}")

    # The harness expects a plain Python list of floats.
    return best_seq
