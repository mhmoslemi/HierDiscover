def max_conv_and_subgrad(a):
    n = len(a)
    # compute convolution via FFT:
    fft = np.fft.rfft(a, n=2*n-1)
    conv = np.fft.irfft(fft * fft, n=2*n-1)
    max_val = float(conv.max())
    tol = max_val * 1e-12 + 1e-14
    active = np.where(conv >= max_val - tol)[0]
    grad = np.zeros_like(a)
    for k in active:
        low = max(0, k - n + 1)
        high = min(n-1, k)
        idx = np.arange(low, high+1)
        partner = k - idx
        grad[idx] += 2.0 * a[partner]
    if active.size > 1:
        grad /= active.size
    return max_val, grad

def solve_lp(a, M):
    n = len(a)
    # Build sparse matrix C
    # offsets = np.arange(n)
    data = np.outer(a, np.ones(n))   # shape (n,n)
    C = sp.diags(data, offsets=np.arange(n), shape=(2*n-1, n), format='csr')
    # Now constraints: C @ b <= M
    # b >=0
    c = -np.ones(n)  # maximize sum b => minimize -sum b
    bounds = [(0, None)] * n
    res = linprog(c, A_ub=C, b_ub=np.full(2*n-1, M), bounds=bounds, method='highs')
    if not res.success:
        raise RuntimeError("LP failed")
    b = res.x
    return b

"""Hybrid Adam + Linear‑programming optimiser.

The routine minimises

    2·n·max(a*a) / (sum(a)**2)

for a non‑negative step function a (list of heights).  The variables are
kept on the probability simplex via a soft‑max (exponentiated‑gradient)
parameterisation and updated with an Adam optimiser.
Periodically a linear programme is solved:

    maximise    Σ b_j
    subject to  (a * b)[k] ≤ M   for all k,
                b_j ≥ 0,

where M = max(a*a) and * is the (linear) convolution.
The LP finds a direction b that can increase the total mass while keeping
the current worst‑case convolution unchanged.  Mixing the current a with the
normalised LP solution often reduces the peak value dramatically.
The best vector seen during the allocated time budget is returned.
"""

import time
import numpy as np
import scipy.sparse as sp
import scipy.optimize as opt

# ----------------------------------------------------------------------
# Helper functions (top‑level, no closures)
# ----------------------------------------------------------------------


def _max_conv_and_subgrad(a: np.ndarray) -> tuple[float, np.ndarray]:
    """
    Compute max entry of the self‑convolution a*a and a sub‑gradient
    of that entry (averaged over all active indices).

    Returns
    -------
    max_val : float
        The maximal convolution value.
    grad : ndarray
        Sub‑gradient of the maximal convolution w.r.t. a.
    """
    n = a.shape[0]
    # convolution via FFT (real‑valued)
    fft = np.fft.rfft(a, n=2 * n - 1)
    conv = np.fft.irfft(fft * fft, n=2 * n - 1)
    max_val = float(conv.max())

    # indices where the maximum is (numerically) attained
    tol = max_val * 1e-12 + 1e-14
    active = np.where(conv >= max_val - tol)[0]

    grad = np.zeros_like(a)
    for k in active:
        low = max(0, k - n + 1)
        high = min(n - 1, k)
        idx = np.arange(low, high + 1, dtype=int)
        partner = k - idx
        grad[idx] += 2.0 * a[partner]

    if active.size > 1:
        grad /= float(active.size)

    return max_val, grad


def _build_conv_matrix(a: np.ndarray) -> sp.csr_matrix:
    """
    Build the (2n‑1)×n matrix C such that C @ b = a * b (linear convolution).
    For each shift k the row of C contains the appropriate coefficients a_i.
    """
    n = a.shape[0]
    # each diagonal offset d (0 … n‑1) contains the constant a[d]
    data = np.tile(a[:, None], (1, n))          # shape (n, n)
    offsets = np.arange(n)                     # 0,1,…,n‑1
    C = sp.diags(data, offsets, shape=(2 * n - 1, n), format="csr")
    return C


def _solve_lp(a: np.ndarray, M: float) -> np.ndarray | None:
    """
    Solve the LP
        maximise Σ b_j
        subject to   a * b ≤ M   (element‑wise)
                     b ≥ 0

    Parameters
    ----------
    a : ndarray of shape (n,)
        Current coefficient vector (already normalised to sum 1).
    M : float
        Current maximal convolution value.

    Returns
    -------
    b_opt : ndarray of shape (n,)   or   None if the solver fails.
    """
    n = a.shape[0]
    C = _build_conv_matrix(a)               # (2n‑1) × n sparse matrix
    ub = np.full(2 * n - 1, M)              # right‑hand side of the inequalities
    c = -np.ones(n)                         # maximise Σ b  ⇒ minimise –Σ b
    bounds = [(0.0, None)] * n               # b_j ≥ 0

    # Using the HiGHS interior‑point/simplex solver (fast for moderate sizes)
    res = opt.linprog(c, A_ub=C, b_ub=ub, bounds=bounds, method="highs")
    if not res.success:
        return None
    return res.x

                        m.fill(0.0)
                        v.fill(0.0)
                        step = 0

        restart += 1

    # ------------------------------------------------------------------
    # 5. Return the best sequence discovered (fallback to uniform)
    # ------------------------------------------------------------------
    if best_a is None:
        best_a = np.full(n, 1.0 / n)

    return [float(v) for v in best_a.tolist()]
