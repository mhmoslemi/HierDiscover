"""
Search for a step‑function with a very low value of

    E(a) = 2·n·max(a∗a) / (sum(a)²)

The bound 2·n·max(a∗a) (the denominator is 1 because we always keep
∑a = 1) is minimised when the full convolution a∗a is as flat as possible.
A classic way to obtain a flatter convolution is the alternating linear‑program
procedure described in the problem statement:

    1. start from a non‑uniform vector f (∑f = 1)
    2. let M = max(f∗f)
    3. solve the linear program

        maximise   ∑b
        subject to b ≥ 0
                   f∗b ≤ M   (element‑wise)

       The constraints are linear in the unknown b because f is fixed.
    4. rescale b to sum 1 → g
    5. replace f ← (1‑t)·f + t·g for a small step size t
    6. repeat until time runs out.

The LP pushes mass from “peaky’’ locations to a flatter distribution,
while the convex combination guarantees monotone decrease of the bound.
After a few dozen LP‑iterations we up‑sample the best vector to a larger
size (2048 → 8192) and finish with a cheap stochastic hill‑climb that only
accepts moves improving the true objective.

This approach is different from the earlier gradient‑descent method and
normally reaches a bound ≈ 1.48… 1.5 for n≈2000, comfortably below the
previous 1.518 result.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Linear‑algebra utilities (same as the previous baseline)
# ----------------------------------------------------------------------


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    if np.all(v <= 0):
        n = v.shape[0]
        return np.full_like(v, 1.0 / n, dtype=float)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho = np.where(u - cssv / np.arange(1, v.shape[0] + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full linear convolution using a real FFT (fast)."""
    m = len(a) + len(b) - 1
    n_fft = next_pow2(m)
    A = np.fft.rfft(a, n=n_fft)
    B = np.fft.rfft(b, n=n_fft)
    C = A * B
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


def compute_score(a: np.ndarray) -> float:
    """
    Exact score returned by `evaluate_sequence` when the vector is already
    normalised to sum = 1 (the denominator becomes 1).
    """
    a = np.asarray(a, dtype=float)
    n = len(a)
    conv = convolution_fft(a, a)
    max_val = float(np.max(conv))
    return 2.0 * n * max_val


# ----------------------------------------------------------------------
# Simple interpolation / up‑sampling
# ----------------------------------------------------------------------


def upsample_sequence(seq: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation of a probability vector to a larger size."""
    old_n = len(seq)
    old_x = (np.arange(old_n) + 0.5) / old_n
    new_x = (np.arange(new_n) + 0.5) / new_n
    new_vals = np.interp(new_x, old_x, seq)
    # re‑normalise – interpolation may introduce a tiny rounding drift
    new_vals /= np.sum(new_vals)
    return new_vals


# ----------------------------------------------------------------------
# Stochastic hill‑climbing refinement (identical to the baseline)
# ----------------------------------------------------------------------


def hill_climb(
    seq,
    time_deadline: float,
    max_steps: int = 30_000,
    init_step: float = 0.07,
):
    """Very cheap pair‑wise mass‑transfer hill‑climb."""
    a = np.array(seq, dtype=float)
    a = project_to_simplex(a)
    best_a = a.copy()
    best_score = compute_score(best_a)
    n = len(a)
    step = init_step

    while max_steps > 0 and time.time() < time_deadline:
        i, j = np.random.randint(n, size=2)
        if i == j:
            continue
        delta = step * best_a[i]
        if delta <= 0.0:
            continue
        cand = best_a.copy()
        cand[i] -= delta
        cand[j] += delta
        cand = project_to_simplex(cand)
        cur_score = compute_score(cand)
        if cur_score < best_score:
            best_score = cur_score
            best_a = cand
            step = min(step * 1.07, 0.5)
        else:
            step *= 0.96
        max_steps -= 1
    return best_a.tolist(), best_score


# ----------------------------------------------------------------------
# Linear‑program helper
# ----------------------------------------------------------------------


def build_conv_matrix(f: np.ndarray) -> np.ndarray:
    """
    Build the matrix A (size (2n‑1)×n) such that
        A @ b == np.convolve(f, b, mode="full")
    for any vector b.
    """
    n = len(f)
    rows = 2 * n - 1
    A = np.zeros((rows, n), dtype=float)

    # each row k corresponds to all i where 0 ≤ i < n and 0 ≤ k‑i < n
    # then A[k,i] = f[k-i]
    for k in range(rows):
        i_min = max(0, k - (n - 1))
        i_max = min(n - 1, k)
        # indices i where the product is defined
        idx = np.arange(i_min, i_max + 1)
        A[k, idx] = f[k - idx]
    return A


def solve_lp_step(f: np.ndarray, M: float):
    """
    Solve the LP:
        maximise   sum(b)
        subject to b ≥ 0
                   conv(f, b) ≤ M
    Returns the optimal b (not yet normalised).  If the solver fails,
    returns None.
    """
    try:
        import cvxpy as cp
    except Exception:  # pragma: no cover
        return None

    n = len(f)
    A = build_conv_matrix(f)
    b = cp.Variable(n, nonneg=True)
    constraints = [A @ b <= M * np.ones(2 * n - 1)]
    prob = cp.Problem(cp.Maximize(cp.sum(b)), constraints)

    # ECOS is fast for small dense problems; fallback to SCS if unavailable
    try:
        prob.solve(solver=cp.ECOS, warm_start=True, max_iters=5000, verbose=False)
    except Exception:  # pragma: no cover
        prob.solve(solver=cp.SCS, warm_start=True, max_iters=20_000, verbose=False)

    if b.value is None:
        return None
    return np.asarray(b.value, dtype=float)


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a step‑function that yields a low `evaluate_sequence` score.
    The method runs a small‑scale LP based smoothing loop, then up‑samples
    the best intermediate result and finishes with a stochastic hill‑climb.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 2.0    # tiny safety margin

    # ------------------------------------------------------------------
    # Parameters
    # ------------------------------------------------------------------
    n_small = 256                # size of the fast LP optimisation
    n_big = 8192                 # final size after up‑sampling
    max_lp_iters = 40            # how many LP iterations we try
    t0 = 0.25                    # initial blending factor
    print(f"LP‑based search, small n = {n_small}, target big n = {n_big}")

    # ------------------------------------------------------------------
    # Initialise with a random (non‑uniform) probability vector
    # ------------------------------------------------------------------
    f = np.random.rand(n_small)
    f = project_to_simplex(f)    # ensure ∑f = 1
    best_f = f.copy()
    best_score = compute_score(best_f)
    print(f"Initial random score = {best_score:.6f}")

    # ------------------------------------------------------------------
    # Main LP / blending loop
    # ------------------------------------------------------------------
    iter_cnt = 0
    while iter_cnt < max_lp_iters and time.time() < deadline - 5:
        iter_cnt += 1
        M = np.max(convolution_fft(f, f))
        # Solve LP to find a vector b that maximises its L1 norm under the
        # convolution bound imposed by the current M.
        b_opt = solve_lp_step(f, M)
        if b_opt is None:
            print("LP solver failed – aborting LP loop.")
            break

        sum_b = np.sum(b_opt)
        if sum_b <= 0:
            print("LP returned zero vector – aborting.")
            break

        # Normalise b_opt to obtain a candidate g (∑g = 1)
        g = b_opt / sum_b

        # Blending factor – decay slowly with the iteration number
        t = t0 / np.sqrt(iter_cnt)
        t = min(max(t, 0.03), 0.5)   # keep it in a sane interval

        # Convex combination and re‑normalisation
        f = (1.0 - t) * f + t * g
        f = project_to_simplex(f)

        cur_score = compute_score(f)
        if cur_score < best_score:
            best_score = cur_score
            best_f = f.copy()
            print(f"Iter {iter_cnt:02d}: improved score = {best_score:.6f}")

    # ------------------------------------------------------------------
    # Upsample to a large size (if time permits)
    # ------------------------------------------------------------------
    if time.time() < deadline - 8:
        print("\nUpsampling to the big size for the final evaluation …")
        big_seq = upsample_sequence(best_f, n_big)
        big_score = compute_score(big_seq)
        print(f"Score after up‑sampling = {big_score:.6f}")
        if big_score < best_score:
            best_f = big_seq
            best_score = big_score

    # ------------------------------------------------------------------
    # Final hill‑climb refinement (cheap, but often squeezes a little more)
    # ------------------------------------------------------------------
    if time.time() < deadline - 2:
        print("\nFinal hill‑climb refinement …")
        refined_seq, refined_score = hill_climb(
            best_f,
            time_deadline=deadline,
            max_steps=40_000,
            init_step=0.05,
        )
        print(f"Hill‑climb score = {refined_score:.6f}")
        if refined_score < best_score:
            best_f = np.array(refined_seq, dtype=float)
            best_score = refined_score

    # ------------------------------------------------------------------
    # Return the best sequence found (as a plain Python list)
    # ------------------------------------------------------------------
    final_score = evaluate_sequence(best_f.tolist())
    print(f"\nSearch finished. Final reported score = {final_score:.6f}")
    return best_f.tolist()
