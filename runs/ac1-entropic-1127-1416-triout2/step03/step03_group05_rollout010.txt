def project_onto_simplex(v):
    # v: np.ndarray shape (n,)
    # returns w: np.ndarray sum=1, w_i >=0
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(1, n+1)
    cond = u - cssv / ind > 0
    rho = ind[cond].max()
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0)
    return w
```

    tau_start = 0.2
    tau_end = 0.001

    iteration = 0
    stagnation = 0

"""
Propose a low‑score step‑function sequence using a temperature‑annealed
projected‑gradient descent on the simplex.  The objective is

    2·n·max(conv(a,a)) / (sum(a)**2)   (sum(a)=1 ⇒ 2·n·max(conv))

We minimise a smooth approximation of the max via a softmax weighting,
gradients are obtained by an FFT‑based convolution, and the iterate is
projected back onto the probability simplex each step.  Several short
restarts with different random initialisations are performed within the
time budget; the best sequence found is returned.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Simplex projection (Duchi et al. 2008)
# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x>=0, sum x = 1}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]                     # sort decreasing
    cssv = np.cumsum(u) - 1.0
    rho = np.nonzero(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:                         # all entries will be zero
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Initialisation strategies
# ----------------------------------------------------------------------
def _initial_sequence(kind: str, n: int, rng: np.random.Generator,
                      prev_seq: np.ndarray | None) -> np.ndarray:
    """Return a feasible probability vector of length n."""
    if kind == "prev" and prev_seq is not None:
        seq = prev_seq.copy()
        if seq.shape[0] != n:
            if seq.shape[0] > n:
                seq = seq[:n]
            else:
                seq = np.concatenate([seq, np.zeros(n - seq.shape[0])])
        seq = np.maximum(seq, 0.0)
        total = seq.sum()
        if total == 0.0:
            return np.full(n, 1.0 / n)
        return seq / total

    if kind == "uniform":
        return np.full(n, 1.0 / n)

    if kind == "random_dirichlet":
        # concentration < 1 gives sparse vectors
        alpha = 0.5
        sample = rng.gamma(alpha, 1.0, size=n)
        return sample / sample.sum()

    if kind == "geometric":
        r = 0.98 + 0.02 * rng.random()
        left = r ** np.arange(n)
        right = r ** np.arange(n - 1, -1, -1)
        seq = left + right
        return seq / seq.sum()

    if kind == "sparse":
        # put mass on ~sqrt(n) random positions
        k = max(2, int(np.sqrt(n)))
        idx = rng.choice(n, size=k, replace=False)
        seq = np.zeros(n)
        seq[idx] = 1.0 / k
        return seq

    # fallback
    return np.full(n, 1.0 / n)


# ----------------------------------------------------------------------
# Core optimisation routine (temperature‑annealed projected gradient)
# ----------------------------------------------------------------------
def _run_gradient_descent(init_a: np.ndarray,
                         deadline: float,
                         restart_id: int,
                         rng: np.random.Generator) -> tuple[np.ndarray, float]:
    """
    Perform projected gradient descent on the simplex with a
    temperature‑annealed smooth approximation of the max‑convolution.
    Returns the best vector seen and its corresponding score.
    """
    a = init_a.copy()
    n = a.shape[0]

    # FFT lengths (next powers of two)
    Nconv = 1 << ((2 * n - 2).bit_length())
    Ngrad = 1 << ((3 * n - 2).bit_length())

    # parameters for the annealing schedule and step size
    tau_start = 0.20          # high temperature → smooth objective
    tau_end = 0.001           # low temperature → true max
    step0 = 0.5               # initial step size (will decay like 1/sqrt(t))

    best_a = a.copy()
    # initial score
    a_fft = np.fft.rfft(a, n=Nconv)
    conv = np.fft.irfft(a_fft * a_fft, n=Nconv)[: 2 * n - 1]
    best_score = 2.0 * n * conv.max()

    start = time.time()
    it = 0
    stagnation = 0

    while time.time() < deadline:
        it += 1

        # --------------------------------------------------------------
        # 1) convolution & current score
        # --------------------------------------------------------------
        a_fft = np.fft.rfft(a, n=Nconv)
        conv = np.fft.irfft(a_fft * a_fft, n=Nconv)[: 2 * n - 1]
        max_val = conv.max()
        cur_score = 2.0 * n * max_val

        # --------------------------------------------------------------
        # 2) keep the best solution seen so far
        # --------------------------------------------------------------
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            stagnation = 0
            print(f"Restart {restart_id:02d} iter {it:6d}: best = {best_score:.6f}")
        else:
            stagnation += 1

        # --------------------------------------------------------------
        # 3) temperature schedule
        # --------------------------------------------------------------
        elapsed = time.time() - start
        frac = min(1.0, elapsed / (deadline - start))
        tau = max(tau_end, tau_start * (tau_end / tau_start) ** frac)

        # --------------------------------------------------------------
        # 4) compute the (smooth) sub‑gradient of max(conv)
        # --------------------------------------------------------------
        if tau > 0.01:
            # smooth soft‑max weighting
            shift = max_val
            weights = np.exp((conv - shift) / tau)
            w_sum = weights.sum()
            if w_sum == 0.0:
                weights = np.full_like(conv, 1.0 / conv.size)
            else:
                weights /= w_sum
        else:
            # hard max – use the active set
            tol = max_val * 1e-12 + 1e-14
            active = np.where(conv >= max_val - tol)[0]
            weights = np.zeros_like(conv)
            if active.size > 0:
                weights[active] = 1.0 / active.size

        # gradient = 2 * (weights ⋆ a)  (cross‑correlation)
        a_rev = a[::-1]
        w_fft = np.fft.rfft(weights, n=Ngrad)
        a_rev_fft = np.fft.rfft(a_rev, n=Ngrad)
        corr_full = np.fft.irfft(w_fft * a_rev_fft, n=Ngrad)
        grad = 2.0 * corr_full[n - 1: n - 1 + n]   # slice the relevant part

        # --------------------------------------------------------------
        # 5) step & projection onto the simplex
        # --------------------------------------------------------------
        step = step0 / np.sqrt(it)
        a_next = a - step * grad
        a = _project_onto_simplex(a_next)

        # --------------------------------------------------------------
        # 6) occasional jitter to escape plateaus
        # --------------------------------------------------------------
        if stagnation > 5000:
            jitter = rng.normal(scale=0.005, size=n)
            a = _project_onto_simplex(a + jitter)
            stagnation = 0

    return best_a, best_score

# ----------------------------------------------------------------------

    """
    rng = np.random.default_rng(123456789)      # deterministic randomness
    safety = 5.0                                 # leave a small safety margin
