def subgradient_descent(a: np.ndarray, deadline: float, max_iter: int=..., eta0: float=0.5):
    a = normalize(a)
    best = a.copy()
    best_val = compute_score(a)
    for it in range(1, max_iter+1):
        if time.time()>deadline: break
        conv = convolution_fft(a,a)
        idx = int(np.argmax(conv))
        # weight vector
        w = np.zeros(len(conv))
        w[idx] = 1.0
        grad = gradient_from_weights(a,w)
        # step size
        eta = eta0/np.sqrt(it)
        a = a - eta*grad
        # ensure non-negativity, project onto simplex
        a = np.maximum(a, 0.0)
        a = normalize(a)
        sc = compute_score(a)
        if sc<best_val:
            best=sc; best_a=a.copy()
        # optionally add noise occasionally
    return best_a, best_val
```

def hardmax_descent(a, deadline, max_iter=20000, eta0=0.5, noise_every=0, noise_scale=0.0):
    a = normalize(a.copy())
    best_a = a.copy()
    best_val = compute_score(a)
    n = len(a)
    for it in range(1, max_iter+1):
        if time.time()>=deadline:
            break
        # compute conv
        conv = convolution_fft(a,a)
        idx = int(np.argmax(conv))
        w = np.zeros_like(conv)
        w[idx] = 1.0
        grad = gradient_from_weights(a, w)
        eta = eta0/np.sqrt(it)
        # multiplicative update with exponent
        a = a * np.exp(-eta * grad)
        # optionally add noise
        if noise_every and it%noise_every==0:
            a += np.random.randn(n)*noise_scale
            a = np.maximum(a, 0.0)
        a = normalize(a)
        cur_score = compute_score(a)
        if cur_score < best_val:
            best_val = cur_score
            best_a = a.copy()
    return best_a, best_val
```

def hill_climbing(a, deadline, max_steps=200000, step_scale=1e-2, min_eps=1e-6):
    a = a.copy()
    best = a.copy()
    best_val = compute_score(a)
    n = len(a)
    while time.time()<deadline and max_steps>0:
        i = rng.integers(n)
        j = rng.integers(n)
        if i==j: continue
        # epsilon drawn from uniform scaling factor
        eps = step_scale * a[i] * rng.uniform()
        if eps < min_eps: continue
        # shift
        a_new = a.copy()
        a_new[i] = max(0.0, a_new[i] - eps)
        a_new[j] = a_new[j] + eps
        # renormalize due to possible rounding errors
        # but we guarantee sum remains unchanged if no clipping; if i becomes 0 then sum may be less but still sum is same because we subtracted eps; so sum unchanged.
        if a_new[i] < 0:
            continue
        a_new = normalize(a_new) # ensure tiny numerical errors fixed; but renorm changes sum.
        cur = compute_score(a_new)
        if cur < best_val:
            best_val = cur
            best = a_new.copy()
            a = a_new
        max_steps -= 1
    return best, best_val
```

def hill_climbing(a: np.ndarray, deadline: float, rng: np.random.Generator, max_steps: int = 50000, step_factor: float = 0.02) -> Tuple[np.ndarray,float]:
    a = a.copy()
    best = a.copy()
    best_val = compute_score(a)
    n=len(a)
    steps = 0
    while time.time()<deadline and steps<max_steps:
        i = rng.integers(n)
        j = rng.integers(n)
        if i==j:
            continue
        eps = step_factor * a[i] * rng.random()
        if eps < 1e-12:
            continue
        a_new = a.copy()
        if a_new[i] < eps:
            continue
        a_new[i] -= eps
        a_new[j] += eps
        a_new = normalize(a_new)
        cur_val = compute_score(a_new)
        if cur_val < best_val:
            best_val = cur_val
            best = a_new.copy()
            a = a_new    # continue from this improved point
        steps += 1
    return best,best_val

def next_pow2(x: int) -> int:
    """Return smallest power of two >= x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Linear convolution using real FFT, zero padded to a power of two."""
    m = len(a) + len(b) - 1
    n_fft = next_pow2(m)
    A = np.fft.rfft(a, n=n_fft)
    B = np.fft.rfft(b, n=n_fft)
    C = A * B
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


def normalize(v: np.ndarray) -> np.ndarray:
    """Project non‑negative vector onto the probability simplex (sum=1)."""
    total = v.sum()
    if total <= 0.0:
        return np.full_like(v, 1.0 / len(v))
    return v / total

def compute_score(a: np.ndarray) -> float:
    """Internal proxy for evaluate_sequence (scale‑invariant)."""
    n = len(a)
    conv = convolution_fft(a, a)
    max_val = float(np.max(conv))
    s = float(np.sum(a))
    if s < 1e-12:
        return float('inf')
    return 2.0 * n * max_val / (s * s)

def arcsine_initial(n: int) -> np.ndarray:
    """Discrete arcsine density normalised to sum 1."""
    i = np.arange(n, dtype=float)
    eps = 0.25
    vals = 1.0 / np.sqrt((i + 0.5) * (n - 0.5 - i) + eps)
    return vals / vals.sum()


def generate_shape(n: int, p: float, eps: float) -> np.ndarray:
    """Symmetric power‑law shape a_i ∝ ((i+eps)*(n‑1‑i+eps))⁻ᵖ."""
    i = np.arange(n, dtype=float)
    base = ((i + eps) * (n - 1 - i + eps)) ** (-p)
    total = base.sum()
    if total == 0.0:
        return np.full(n, 1.0 / n)
    return base / total

def softmax_weights(conv: np.ndarray, lam: float) -> np.ndarray:
    """Numerically stable soft‑max over convolution entries."""
    x = lam * conv
    mx = np.max(x)
    w = np.exp(x - mx)
    w /= w.sum()
    return w


def gradient_from_weights(a: np.ndarray, w: np.ndarray) -> np.ndarray:
    """
    Gradient of Σ_k w_k·(a*a)_k w.r.t a.
    Implements   g_j = 2· Σ_m a_m · w_{m+j}.
    Computed via one FFT‑based convolution.
    """
    w_rev = w[::-1]
    conv = convolution_fft(a, w_rev)               # length = len(a)+len(w)-1 = 3·n‑2
    idx = 2 * len(a) - 2                            # position 2n‑2 in conv
    grads = 2.0 * conv[idx - np.arange(len(a))]
    return grads
```

def optimise_sequence(
    initial_a: np.ndarray,
    deadline: float,
    *,
    lam_start: float = 5.0,
    lam_end: float = 80.0,
    lam_ramp_iters: int = 4000,
    eta0: float = 0.6,
    max_iter: int = 200_000,
    noise_every: int = 0,
    noise_scale: float = 0.0,
) -> Tuple[np.ndarray, float]:
    """
    Exponentiated‑gradient descent on a smooth soft‑max approximation of
    the max‑convolution.  Returns best vector seen and its (non‑smooth)
    score.
    """
    a = normalize(initial_a.astype(float).copy())
    best_a = a.copy()
    best_sc = compute_score(a)

    n = len(a)

    for it in range(1, max_iter + 1):
        if time.time() >= deadline:
            break

        # annealing of the soft‑max temperature
        lam = lam_start + (lam_end - lam_start) * min(1.0, it / lam_ramp_iters)

        # current convolution and soft‑max weights
        conv = convolution_fft(a, a)                     # length 2n‑1
        w = softmax_weights(conv, lam)

        # gradient of the weighted sum of convolutions
        grad = gradient_from_weights(a, w)

        # step size (standard 1/√t schedule)
        eta = eta0 / np.sqrt(it)

        # multiplicative (exponentiated) update – guarantees non‑negativity
        a = a * np.exp(-eta * grad)

        # optional exploration noise
        if noise_every and (it % noise_every == 0):
            a += np.random.randn(n) * noise_scale
            a = np.maximum(a, 0.0)

        # re‑project onto the simplex
        a = normalize(a)

        cur_sc = compute_score(a)
        if cur_sc < best_sc:
            best_sc = cur_sc
            best_a = a.copy()

        # occasional progress report
        if it % 2000 == 0:
            print(f"      iter={it:,d}  λ={lam:.1f}  cur={cur_sc:.6f}  best={best_sc:.6f}")

    return best_a, best_sc

def hardmax_descent(
    a: np.ndarray,
    deadline: float,
    *,
    max_iter: int = 100_000,
    eta0: float = 0.5,
    noise_every: int = 0,
    noise_scale: float = 0.0,
) -> Tuple[np.ndarray, float]:
    """
    Exponentiated‑gradient descent where the soft‑max weight is a Dirac
    at the current maximal convolution index (the exact sub‑gradient of
    the max‑convolution objective).  Returns the best vector seen.
    """
    a = normalize(a.astype(float).copy())
    best_a = a.copy()
    best_sc = compute_score(a)

    n = len(a)

    for it in range(1, max_iter + 1):
        if time.time() >= deadline:
            break

        # compute convolution and locate the maximal entry
        conv = convolution_fft(a, a)          # length 2n‑1
        idx = int(np.argmax(conv))

        # construct one‑hot weight vector
        w = np.zeros_like(conv)
        w[idx] = 1.0

        # gradient of the selected entry
        grad = gradient_from_weights(a, w)

        # step size decay
        eta = eta0 / np.sqrt(it)

        # multiplicative update
        a = a * np.exp(-eta * grad)

        # optional noise injection
        if noise_every and (it % noise_every == 0):
            a += np.random.randn(n) * noise_scale
            a = np.maximum(a, 0.0)

        # project back onto simplex
        a = normalize(a)

        cur_sc = compute_score(a)
        if cur_sc < best_sc:
            best_sc = cur_sc
            best_a = a.copy()

        if it % 2000 == 0:
            print(f"      hard‑iter={it:,d}  cur={cur_sc:.6f}  best={best_sc:.6f}")

    return best_a, best_sc

def hill_climbing(
    a: np.ndarray,
    deadline: float,
    rng: np.random.Generator,
    *,
    max_steps: int = 100_000,
    step_factor: float = 0.02,
) -> Tuple[np.ndarray, float]:
    """
    Simple stochastic coordinate‑wise hill climbing on the simplex.
    Each step moves a random epsilon from one coordinate to another.
    """
    a = a.copy()
    best_a = a.copy()
    best_sc = compute_score(a)
    n = len(a)
    steps = 0
    while time.time() < deadline and steps < max_steps:
        i = rng.integers(n)
        j = rng.integers(n)
        if i == j:
            continue
        # Choose a proportion of a[i] to transfer
        eps = step_factor * a[i] * rng.random()
        if eps <= 0.0:
            continue
        if a[i] < eps:
            continue

        a_new = a.copy()
        a_new[i] -= eps
        a_new[j] += eps
        a_new = normalize(a_new)

        cur_sc = compute_score(a_new)
        if cur_sc < best_sc:
            best_sc = cur_sc
            best_a = a_new.copy()
            a = a_new  # continue from improved point
        steps += 1
        if steps % 20000 == 0:
            print(f"      hill step={steps:,d}  cur={cur_sc:.6f}  best={best_sc:.6f}")
    return best_a, best_sc

def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs) -> List[float]:
    """
    Search for a non‑negative step‑function with a low evaluate_sequence value.
    The routine respects the allotted runtime and always returns a feasible list.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 2.0   # safety margin

    best_seq: List[float] = []
    best_val: float = float('inf')

    # ------------------------------------------------------------------
    # 0. Optional fallback from a previously discovered construction.
    # ------------------------------------------------------------------
    try:
        fallback = [float(x) for x in height_sequence_1]  # type: ignore
        val = evaluate_sequence(fallback)
        print(f"Fallback (height_sequence_1) score = {val:.6f}")
        if val < best_val:
            best_val = val
            best_seq = fallback
    except NameError:
        pass

    # ------------------------------------------------------------------
    # 1.  Lengths to explore (powers of two – cheap FFTs).
    # ------------------------------------------------------------------
    lengths = [1024, 2048, 4096]

    for n in lengths:
        if time.time() >= deadline - 5.0:
            break
        print(f"\n=== Exploring length n = {n} ===")

        # --------------------------------------------------------------
        # 1a.  Build a modest library of candidate start vectors.
        # --------------------------------------------------------------
        candidates: List[np.ndarray] = []

        # uniform
        candidates.append(np.full(n, 1.0 / n))

        # classic arcsine
        candidates.append(arcsine_initial(n))

        # power‑law family (p, eps)
        for _ in range(5):
            p = rng.uniform(0.2, 0.8)
            eps = rng.uniform(0.001, max(2.0, 0.02 * n))
            candidates.append(generate_shape(n, p, eps))

        # beta‑like shapes
        for _ in range(3):
            alpha = rng.uniform(0.2, 5.0)
            beta = rng.uniform(0.2, 5.0)
            i = np.arange(n, dtype=float)
            base = (i + 0.5) ** (alpha - 1) * (n - 1 - i + 0.5) ** (beta - 1)
            total = base.sum()
            if total == 0.0:
                base = np.full(n, 1.0 / n)
            else:
                base = base / total
            candidates.append(base)

        # Dirichlet / exponential samples

"""
Propose a step‑function sequence with low evaluate_sequence value.

The algorithm combines three complementary optimisation phases:

1. **Soft‑max gradient descent** – a smoothed approximation of the
   max‑convolution (as in the previous baseline) gives a quick, decent
   shape.

2. **Hard‑max exponentiated‑gradient descent** – the soft‑max weight is
   replaced by a Dirac at the current maximal convolution index.
   This follows the exact sub‑gradient of the true max‑convolution
   objective and therefore refines the shape more aggressively.

3. **Stochastic hill‑climbing** – random coordinate‑wise transfers of
   mass on the probability simplex allow the search to escape shallow
   local minima.

Several parametric families (uniform, arcsine, power‑law, beta‑like,
Dirichlet) are generated for a few lengths (1024, 2048, 4096).  The best
candidates are refined by the pipeline above.  The routine respects the
supplied time budget and always returns a feasible list of non‑negative
floats.
"""

import time
from typing import List, Tuple

import numpy as np

# ----------------------------------------------------------------------
# Basic numeric utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Linear convolution via real FFT (full length, zero‑padded to a power‑of‑2)."""
    m = len(a) + len(b) - 1
    n_fft = next_pow2(m)
    A = np.fft.rfft(a, n=n_fft)
    B = np.fft.rfft(b, n=n_fft)
    C = A * B
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


def normalize(v: np.ndarray) -> np.ndarray:
    """Project a non‑negative vector onto the probability simplex (sum = 1)."""
    total = v.sum()
    if total <= 0.0:
        return np.full_like(v, 1.0 / len(v))
    return v / total


def compute_score(a: np.ndarray) -> float:
    """
    Internal proxy for evaluate_sequence (scale‑invariant):
        2·n·max(a*a) / (sum(a)²)
    """
    n = len(a)
    conv = convolution_fft(a, a)
    max_val = float(np.max(conv))
    s = float(np.sum(a))
    if s < 1e-12:
        return float("inf")
    return 2.0 * n * max_val / (s * s)


# ----------------------------------------------------------------------
# Parametric families for initial candidates
# ----------------------------------------------------------------------


def arcsine_initial(n: int) -> np.ndarray:
    """Classic arcsine density (discrete, normalised to sum = 1)."""
    i = np.arange(n, dtype=float)
    eps = 0.25
    vals = 1.0 / np.sqrt((i + 0.5) * (n - 0.5 - i) + eps)
    return vals / vals.sum()


def generate_shape(n: int, p: float, eps: float) -> np.ndarray:
    """Symmetric power‑law shape a_i ∝ ((i+ε)(n‑1‑i+ε))⁻ᵖ, normalised."""
    i = np.arange(n, dtype=float)
    base = ((i + eps) * (n - 1 - i + eps)) ** (-p)
    total = base.sum()
    if total == 0.0:
        return np.full(n, 1.0 / n)
    return base / total


# ----------------------------------------------------------------------
# Soft‑max utilities (used in the first optimisation stage)
# ----------------------------------------------------------------------


def softmax_weights(conv: np.ndarray, lam: float) -> np.ndarray:
    """Numerically stable soft‑max over the convolution entries."""
    x = lam * conv
    mx = np.max(x)
    w = np.exp(x - mx)
    w /= w.sum()
    return w


def gradient_from_weights(a: np.ndarray, w: np.ndarray) -> np.ndarray:
    """
    Gradient of Σ_k w_k·(a*a)_k w.r.t. a.
    Implements   g_j = 2· Σ_m a_m · w_{m+j},
    computed via one FFT‑based convolution.
    """
    w_rev = w[::-1]
    conv = convolution_fft(a, w_rev)               # length = len(a)+len(w)-1 = 3·n‑2
    idx = 2 * len(a) - 2                            # position 2n‑2 in conv
    grads = 2.0 * conv[idx - np.arange(len(a))]
    return grads


def optimise_sequence(
    initial_a: np.ndarray,
    deadline: float,
    *,
    lam_start: float = 5.0,
    lam_end: float = 80.0,
    lam_ramp_iters: int = 4000,
    eta0: float = 0.6,
    max_iter: int = 200_000,
    noise_every: int = 0,
    noise_scale: float = 0.0,
) -> Tuple[np.ndarray, float]:
    """
    Exponentiated‑gradient descent on a smooth soft‑max approximation of
    the max‑convolution.  Returns the best vector seen and its (non‑smooth)
    score.
    """
    a = normalize(initial_a.astype(float).copy())
    best_a = a.copy()
    best_sc = compute_score(a)

    n = len(a)

    for it in range(1, max_iter + 1):
        if time.time() >= deadline:
            break

        lam = lam_start + (lam_end - lam_start) * min(1.0, it / lam_ramp_iters)

        conv = convolution_fft(a, a)                # length 2n‑1
        w = softmax_weights(conv, lam)

        grad = gradient_from_weights(a, w)

        eta = eta0 / np.sqrt(it)
        a = a * np.exp(-eta * grad)

        if noise_every and (it % noise_every == 0):
            a += np.random.randn(n) * noise_scale
            a = np.maximum(a, 0.0)

        a = normalize(a)

        cur_sc = compute_score(a)
        if cur_sc < best_sc:
            best_sc = cur_sc
            best_a = a.copy()

        if it % 2000 == 0:
            print(f"      iter={it:,d}  λ={lam:.1f}  cur={cur_sc:.6f}  best={best_sc:.6f}")

    return best_a, best_sc


# ----------------------------------------------------------------------
# Hard‑max exponentiated‑gradient descent (exact sub‑gradient of max)
# ----------------------------------------------------------------------


def hardmax_descent(
    a: np.ndarray,
    deadline: float,
    *,
    max_iter: int = 100_000,
    eta0: float = 0.5,
    noise_every: int = 0,
    noise_scale: float = 0.0,
) -> Tuple[np.ndarray, float]:
    """
    Exponentiated‑gradient descent where the soft‑max weight is a Dirac at
    the current maximal convolution index (exact sub‑gradient of the
    max‑convolution).  Returns the best vector seen.
    """
    a = normalize(a.astype(float).copy())
    best_a = a.copy()
    best_sc = compute_score(a)

    n = len(a)

    for it in range(1, max_iter + 1):
        if time.time() >= deadline:
            break

        conv = convolution_fft(a, a)
        idx = int(np.argmax(conv))

        w = np.zeros_like(conv)
        w[idx] = 1.0

        grad = gradient_from_weights(a, w)

        eta = eta0 / np.sqrt(it)
        a = a * np.exp(-eta * grad)

        if noise_every and (it % noise_every == 0):
            a += np.random.randn(n) * noise_scale
            a = np.maximum(a, 0.0)

        a = normalize(a)

        cur_sc = compute_score(a)
        if cur_sc < best_sc:
            best_sc = cur_sc
            best_a = a.copy()

        if it % 2000 == 0:
            print(f"      hard‑iter={it:,d}  cur={cur_sc:.6f}  best={best_sc:.6f}")

    return best_a, best_sc


# ----------------------------------------------------------------------
# Simple stochastic hill climbing on the simplex
# ----------------------------------------------------------------------


def hill_climbing(
    a: np.ndarray,
    deadline: float,
    rng: np.random.Generator,
    *,
    max_steps: int = 100_000,
    step_factor: float = 0.02,
) -> Tuple[np.ndarray, float]:
    """
    Random coordinate‑wise mass transfers on the simplex.
    Each step moves ε = step_factor·a[i]·U (U∈[0,1]) from i to j.
    """
    a = a.copy()
    best_a = a.copy()
    best_sc = compute_score(a)
    n = len(a)
    steps = 0

    while time.time() < deadline and steps < max_steps:
        i = rng.integers(n)
        j = rng.integers(n)
        if i == j:
            continue

        eps = step_factor * a[i] * rng.random()
        if eps <= 0.0 or a[i] < eps:
            continue

        a_new = a.copy()
        a_new[i] -= eps
        a_new[j] += eps
        a_new = normalize(a_new)

        cur_sc = compute_score(a_new)
        if cur_sc < best_sc:
            best_sc = cur_sc
            best_a = a_new.copy()
            a = a_new  # continue from the improved point

        steps += 1
        if steps % 20000 == 0:
            print(f"      hill step={steps:,d}  cur={cur_sc:.6f}  best={best_sc:.6f}")

    return best_a, best_sc
