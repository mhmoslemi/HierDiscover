def project_onto_simplex(v):
    """Projects vector v onto the probability simplex (non-negative entries sum to 1)."""
    # v may be any real numbers.
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    rho = np.nonzero(u - cssv / np.arange(1, n+1) > 0)[0]
    if rho.size == 0:
        theta = 0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0)
    return w

"""
Non‑negative step‑function minimisation (improved version).

The optimiser uses an Adam‑style exponentiated‑gradient on a smooth log‑sum‑exp
approximation of the convolution maximum.  The schedule for the temperature
λ is linearly annealed from a small start value to a large final value, which
tightens the approximation and drives the vector toward a shape that equalises
its autocorrelation.  The updates are projected onto the probability simplex
by simple clipping and normalisation, and a symmetry enforcement
(a_i = a_{n‑1‑i}) is applied at every step because the optimal solution is
known to be symmetric.  Several parametric families provide diverse starts,
the best candidates are refined, and a final polishing stage is performed.
The routine respects the time budget and always returns a feasible sequence.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Low‑level numeric utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Return the smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """
    Linear convolution via real FFT (full length, zero‑padded to a power of two).
    """
    m = len(a) + len(b) - 1
    n_fft = next_pow2(m)
    A = np.fft.rfft(a, n=n_fft)
    B = np.fft.rfft(b, n=n_fft)
    C = A * B
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


def softmax_weights(conv: np.ndarray, lam: float) -> np.ndarray:
    """
    Numerically stable soft‑max over the convolution entries.
    The argument lam plays the role of inverse temperature.
    """
    x = lam * conv
    mx = np.max(x)
    ex = np.exp(x - mx)
    return ex / np.sum(ex)


def gradient_from_weights(a: np.ndarray, w: np.ndarray) -> np.ndarray:
    """
    Gradient of Σ_k w_k·(a*a)_k with respect to a.
    Implements g_j = 2·Σ_m a_m·w_{m+j}.
    """
    # w has length 2·n‑1, a has length n.
    w_rev = w[::-1]
    conv = convolution_fft(a, w_rev)                     # length = 3·n‑2
    idx = 2 * len(a) - 2                                 # points to position 2n‑2
    grads = 2.0 * conv[idx - np.arange(len(a))]
    return grads


def ensure_simplex(v: np.ndarray) -> np.ndarray:
    """
    Project a vector onto the probability simplex {x ≥ 0, Σ x = 1}
    by clipping negatives and renormalising.
    """
    v = np.maximum(v, 0.0)
    s = v.sum()
    if s == 0.0:
        # fallback to uniform distribution
        return np.full_like(v, 1.0 / len(v))
    return v / s


def compute_score_internal(a: np.ndarray) -> float:
    """
    Fast internal objective used for candidate ranking.
    Assumes Σ a_i = 1, so the denominator of evaluate_sequence is omitted.
    """
    n = len(a)
    conv = convolution_fft(a, a)
    return 2.0 * n * np.max(conv)


# ----------------------------------------------------------------------
# Parametric families for initial candidates
# ----------------------------------------------------------------------


def uniform_initial(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n)


def arcsine_initial(n: int) -> np.ndarray:
    i = np.arange(n, dtype=float)
    eps = 0.25
    vals = 1.0 / np.sqrt((i + 0.5) * (n - 0.5 - i) + eps)
    return vals / vals.sum()


def powerlaw_initial(n: int, p: float, eps: float) -> np.ndarray:
    i = np.arange(n, dtype=float)
    base = ((i + eps) * (n - 1 - i + eps)) ** (-p)
    if base.sum() == 0.0:
        return uniform_initial(n)
    return base / base.sum()


def beta_initial(n: int, alpha: float, beta: float) -> np.ndarray:
    i = np.arange(n, dtype=float)
    base = (i + 0.5) ** (alpha - 1) * (n - 1 - i + 0.5) ** (beta - 1)
    if base.sum() == 0.0:
        return uniform_initial(n)
    return base / base.sum()


def triangular_initial(n: int) -> np.ndarray:
    i = np.arange(n, dtype=float)
    centre = (n - 1) / 2.0
    vals = 1.0 - np.abs(i - centre) / centre
    vals = np.maximum(vals, 0.0)
    return vals / vals.sum()


def exponential_initial(n: int, rate: float = 0.01) -> np.ndarray:
    i = np.arange(n, dtype=float)
    centre = (n - 1) / 2.0
    vals = np.exp(-rate * np.abs(i - centre))
    return vals / vals.sum()


def upscale(seq: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation to a larger length, followed by simplex projection."""
    old_n = len(seq)
    old_x = np.arange(old_n, dtype=float)
    new_x = np.linspace(0, old_n - 1, new_n, dtype=float)
    new_seq = np.interp(new_x, old_x, seq)
    return ensure_simplex(new_seq)


# ----------------------------------------------------------------------
# Adam‑style optimiser on the smooth surrogate objective
# ----------------------------------------------------------------------


def adam_optimize(
    initial_a: np.ndarray,
    deadline: float,
    *,
    lam_start: float = 1.0,
    lam_end: float = 500.0,
    lam_ramp_iters: int = 30000,
    eta: float = 0.08,
    beta1: float = 0.9,
    beta2: float = 0.999,
    eps: float = 1e-8,
    max_iter: int = 200_000,
    noise_every: int = 0,
    noise_scale: float = 0.0,
    enforce_symmetry: bool = True,
    verbose: bool = False,
) -> tuple[np.ndarray, float]:
    """
    Adam optimiser for the log‑sum‑exp surrogate of the max‑convolution.
    Returns the best vector (sum = 1) and the corresponding internal score.
    """
    a = ensure_simplex(initial_a.astype(float).copy())
    n = len(a)

    # Adam state
    m = np.zeros_like(a)
    v = np.zeros_like(a)

    best_a = a.copy()
    best_score = compute_score_internal(a)

    for it in range(1, max_iter + 1):
        if time.time() >= deadline:
            break

        # --------------------------------------------------------------
        # 1.  Evaluate current point (used for best‑track)
        # --------------------------------------------------------------
        conv = convolution_fft(a, a)
        cur_score = 2.0 * n * np.max(conv)
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()

        # --------------------------------------------------------------
        # 2.  Soft‑max temperature
        # --------------------------------------------------------------
        lam = lam_start + (lam_end - lam_start) * min(1.0, it / lam_ramp_iters)

        # --------------------------------------------------------------
        # 3.  Gradient of the smooth surrogate
        # --------------------------------------------------------------
        w = softmax_weights(conv, lam)          # shape (2n‑1,)
        grad = gradient_from_weights(a, w)      # shape (n,)

        # --------------------------------------------------------------
        # 4.  Adam parameter update
        # --------------------------------------------------------------
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        step = eta / (np.sqrt(v_hat) + eps)
        a = a - step * m_hat

        # --------------------------------------------------------------
        # 5.  Optional exploration noise
        # --------------------------------------------------------------
        if noise_every and (it % noise_every == 0):
            a += np.random.randn(n) * noise_scale

        # --------------------------------------------------------------
        # 6.  Enforce symmetry (optional)
        # --------------------------------------------------------------
        if enforce_symmetry:
            a = (a + a[::-1]) * 0.5

        # --------------------------------------------------------------
        # 7.  Project back onto the simplex
        # --------------------------------------------------------------
        a = ensure_simplex(a)

        # --------------------------------------------------------------
        # 8.  Progress report
        # --------------------------------------------------------------
        if verbose and (it % 2000 == 0):
            print(f"      iter={it:,d}  λ={lam:.1f}  cur={cur_score:.6f}  best={best_score:.6f}")

    # final best score recomputed for safety
    best_score = compute_score_internal(best_a)
    return best_a, best_score


# ----------------------------------------------------------------------
# Main entry point required by the competition harness
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function with a low `evaluate_sequence`
    value.  A multi‑stage scheme is employed:

    1.  Build a small library of parametric start vectors for several lengths.
    2.  Run the Adam optimiser (soft‑max surrogate) on the most promising starts.
    3.  If time permits, upscale the current best solution and polish it.
    The routine always returns a feasible list of floats.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    # safety margin so that we do not exceed the time limit
    deadline = start_time + budget_s - 2.0

    best_seq: list[float] = []
    best_val: float = float("inf")

    # ------------------------------------------------------------------
    # 0.  Fallback copy of a previously discovered construction (if any).
    # ------------------------------------------------------------------
    try:
        fallback = [float(x) for x in height_sequence_1]  # type: ignore
        val = evaluate_sequence(fallback)
        print(f"Fallback (height_sequence_1) score = {val:.6f}")
        if val < best_val:
            best_val = val
            best_seq = fallback
    except Exception:
        pass

    # ------------------------------------------------------------------
    # 1.  Lengths to explore – we start modest and go higher if time permits.
    # ------------------------------------------------------------------
    lengths = [2048, 4096]       # 8192 may be attempted later if time allows

    for n in lengths:
        if time.time() >= deadline - 5.0:
            break
        print(f"\n=== Exploring length n = {n} ===")

        # --------------------------------------------------------------
        # 1a.  Assemble a modest library of candidate start vectors.
        # --------------------------------------------------------------
        cand_pool: list[np.ndarray] = []

        # deterministic families
        cand_pool.append(uniform_initial(n))
        cand_pool.append(arcsine_initial(n))
        cand_pool.append(triangular_initial(n))

        # power‑law shapes (randomised parameters)
        for _ in range(4):
            p = rng.uniform(0.2, 1.2)
            eps = rng.uniform(0.001, max(2.0, 0.02 * n))
            cand_pool.append(powerlaw_initial(n, p, eps))

        # beta‑like shapes
        for _ in range(3):
            alpha = rng.uniform(0.2, 5.0)
            beta = rng.uniform(0.2, 5.0)
            cand_pool.append(beta_initial(n, alpha, beta))

        # exponential decays
        for _ in range(2):
            rate = rng.uniform(0.005, 0.05)
            cand_pool.append(exponential_initial(n, rate))

        # random Dirichlet draws (lightly smooth)
        for _ in range(2):
            samp = rng.exponential(1.0, size=n)
            samp /= samp.sum()
            cand_pool.append(samp)

        # --------------------------------------------------------------
        # 1b.  Quick internal scoring, keep the most promising candidates.
        # --------------------------------------------------------------
        scored = [(compute_score_internal(c), c) for c in cand_pool]
        scored.sort(key=lambda x: x[0])
        top_candidates = [c for _, c in scored[:3]]   # keep three best starters

        # --------------------------------------------------------------
        # 1c.  Refine each retained candidate with the Adam optimiser.
        # --------------------------------------------------------------
        for idx, start_vec in enumerate(top_candidates):
            if time.time() >= deadline - 5.0:
                break
            print(f"  candidate {idx + 1}/{len(top_candidates)} for n={n}")

            # allocate iterations based on remaining time (rough 200–300 it/s)
            remaining = max(0.0, deadline - time.time())
            max_iter = int(min(180_000, max(5_000, remaining * 250)))

            refined, _ = adam_optimize(
                initial_a=start_vec,
                deadline=deadline,
                lam_start=1.0,
                lam_end=500.0,
                lam_ramp_iters=int(max_iter * 0.6),
                eta=0.08,
                max_iter=max_iter,
                noise_every=2000,
                noise_scale=2e-5,
                enforce_symmetry=True,
                verbose=False,
            )

            # final evaluation with the official (slow) evaluator
            val = evaluate_sequence(refined.tolist())
            print(f"    evaluated score = {val:.6f}")

            if val < best_val:
                best_val = val
                best_seq = refined.tolist()
                print(f"    *** New global best: {best_val:.6f} (n={n}) ***")

    # ------------------------------------------------------------------
    # 2.  Optional up‑scaling and final polishing if time permits.
    # ------------------------------------------------------------------
    if best_seq and time.time() < deadline - 5.0:
        # try to double the resolution (capped at 8192) and polish
        cur_n = len(best_seq)
        target_n = min(8192, cur_n * 2)
        if target_n > cur_n:
            print("\nUpscaling best sequence for final polishing")
            a_up = upscale(np.array(best_seq, dtype=float), target_n)

            remaining = max(0.0, deadline - time.time())
            max_iter = int(min(120_000, remaining * 250))

            a_refined, _ = adam_optimize(
                initial_a=a_up,
                deadline=deadline,
                lam_start=1.0,
                lam_end=800.0,
                lam_ramp_iters=int(max_iter * 0.7),
                eta=0.05,
                max_iter=max_iter,
                noise_every=500,
                noise_scale=1e-5,
                enforce_symmetry=True,
                verbose=False,
            )
            final_val = evaluate_sequence(a_refined.tolist())
            print(f"Final polished (up‑scaled) score = {final_val:.6f}")
            if final_val < best_val:
                best_val = final_val
                best_seq = a_refined.tolist()
                print("*** Updated global best after up‑scaled polish ***")
        else:
            # no up‑scaling possible – just polish the existing best
            print("\nFinal polishing of the global best")
            a = np.array(best_seq, dtype=float)
            remaining = max(0.0, deadline - time.time())
            max_iter = int(min(120_000, remaining * 250))

            a_refined, _ = adam_optimize(
                initial_a=a,
                deadline=deadline,
                lam_start=1.0,
                lam_end=800.0,
                lam_ramp_iters=int(max_iter * 0.7),
                eta=0.05,
                max_iter=max_iter,
                noise_every=500,
                noise_scale=1e-5,
                enforce_symmetry=True,
                verbose=False,
            )
            final_val = evaluate_sequence(a_refined.tolist())
            print(f"Final polished score = {final_val:.6f}")
            if final_val < best_val:
                best_val = final_val
                best_seq = a_refined.tolist()
                print("*** Updated global best after final polish ***")

    # ------------------------------------------------------------------
    # 3.  Guard – guarantee a valid return.
    # ------------------------------------------------------------------
    if not best_seq:
        # fallback to a simple uniform distribution of modest size
        best_seq = uniform_initial(256).tolist()
        best_val = evaluate_sequence(best_seq)

    print(f"\nSearch finished. Best reported score = {best_val:.6f}")
    return best_seq
