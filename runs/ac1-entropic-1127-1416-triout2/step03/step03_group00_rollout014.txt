"""
Propose a step‑height sequence that minimises ``evaluate_sequence``.

Algorithm
---------
1. **Initialisation** – use the previously found sequence
   ``height_sequence_1`` (if it exists) and normalise it to a
   probability vector (sum = 1).  Otherwise start from a uniform vector.

2. **Exponentiated‑gradient (mirror‑descent) phase** – minimise the
   ∞‑norm of the self‑convolution while staying on the simplex.
   The sub‑gradient of the current maximal convolution entry is
   obtained exactly (average over all active indices).  The update
   uses a multiplicative (exponentiated) step and a Euclidean projection
   (renormalisation) onto the simplex.  This phase consumes the bulk
   of the time budget.

3. **Guided simulated annealing** – starting from the best point of
   phase 1, perform a large number of Monte‑Carlo moves that transfer a
   random amount of mass from a “high‑gradient’’ index to a
   “low‑gradient’’ index.  The temperature decays slowly (geometric
   cooling).  Moves that improve the score are always accepted; worsening
   moves are accepted with the Metropolis probability.

4. The best sequence observed during the whole run is returned (still
   normalised to sum = 1 – the scaling does not affect the score).

The method stays completely on the probability simplex, therefore all
generated numbers are non‑negative and the evaluation function never
rejects them.
"""

import time
import math
import numpy as np


# ----------------------------------------------------------------------
# Helper routines
# ----------------------------------------------------------------------


def compute_score(a: np.ndarray) -> float:
    """
    Exact value of ``evaluate_sequence`` for a vector whose sum is 1.
    Returns 2·n·max(conv) where conv = a * a (linear convolution).
    """
    n = a.shape[0]
    fft = np.fft.rfft(a, n=2 * n - 1)
    conv = np.fft.irfft(fft * fft, n=2 * n - 1)
    return 2.0 * n * conv.max()


def beta_sequence(alpha: float, beta: float, n: int) -> np.ndarray:
    """
    Discretised Beta(α,β) density on n points, normalised to sum to 1.
    """
    x = (np.arange(n) + 0.5) / n
    # work in log‑space to avoid overflow for extreme α,β
    logw = (alpha - 1.0) * np.log(x) + (beta - 1.0) * np.log(1.0 - x)
    logw = np.clip(logw, -700.0, 700.0)
    w = np.exp(logw)
    if not np.isfinite(w.sum()) or w.sum() == 0.0:
        w = np.ones_like(w)
    return w / w.sum()


def subgradient_of_max_conv(a: np.ndarray, conv: np.ndarray) -> np.ndarray:
    """
    Compute a sub‑gradient of ``max(conv)`` at the point ``a``.
    The gradient is the average of the exact gradients of all active
    convolution entries (those attaining the current maximum up to a tiny
    tolerance).
    """
    n = a.shape[0]
    max_b = conv.max()
    eps = max_b * 1e-6
    active = np.where(conv >= max_b - eps)[0]

    if active.size == 0:                     # safety net
        grad = 2.0 * a
        return grad

    grad = np.zeros_like(a)
    for k in active:
        low = max(0, k - n + 1)
        high = min(n - 1, k)
        idx = np.arange(low, high + 1, dtype=int)
        partner = k - idx
        # ∂conv_k/∂a_i = 2·a_{k‑i}
        grad[idx] += 2.0 * a[partner]
    grad /= float(active.size)
    return grad


def exponential_update(a: np.ndarray, grad: np.ndarray, step: float) -> np.ndarray:
    """
    Mirror‑descent update on the probability simplex using the
    exponentiated‑gradient (multiplicative) rule.
    """
    # multiplicative factor, clipped to avoid under‑flow
    factor = np.exp(-step * grad)
    a_new = a * factor
    a_new = np.clip(a_new, 1e-16, None)      # keep strictly positive
    a_new /= a_new.sum()
    return a_new


# ----------------------------------------------------------------------
# Main optimisation routine
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 0, budget_s: int = 1000) -> list[float]:
    """
    Search for a non‑negative height sequence that minimises
    ``evaluate_sequence``.  The algorithm runs for at most ``budget_s``
    seconds and returns the best sequence discovered.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5   # tiny safety margin

    # --------------------------------------------------------------
    # 1 – problem size & initial point
    # --------------------------------------------------------------
    try:
        init = np.array(height_sequence_1, dtype=float)   # type: ignore[name-defined]
        n = init.shape[0]
        print(f"Using existing height_sequence_1 of length {n}.")
    except NameError:
        # fallback – a plain uniform start (any reasonable start works)
        n = 5146
        init = np.ones(n, dtype=float) / n
        print(f"No existing sequence – using uniform start, length {n}.")

    # normalise to a probability vector (sum == 1)
    a = init / init.sum()

    best_a = a.copy()
    best_score = compute_score(best_a)
    print(f"Initial score = {best_score:.6f}")

    # --------------------------------------------------------------
    # 2 – Exponentiated‑gradient (mirror‑descent) phase
    # --------------------------------------------------------------
    eta0 = 1.0          # base step size
    it_exp = 0
    # keep 200 s for the subsequent simulated‑annealing phase
    while time.time() < deadline - 200.0:
        # compute convolution and its maximum
        fft = np.fft.rfft(a, n=2 * n - 1)
        conv = np.fft.irfft(fft * fft, n=2 * n - 1)
        cur_score = 2.0 * n * conv.max()

        # record improvement
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            if it_exp % 50000 == 0:
                print(f"Exp‑grad iter {it_exp:6d}: new best {best_score:.6f}")

        # sub‑gradient of the current max‑convolution entry
        grad = subgradient_of_max_conv(a, conv)

        # diminishing step size
        step = eta0 / math.sqrt(it_exp + 1.0)

        # multiplicative (exponential) update and re‑normalisation
        a = exponential_update(a, grad, step)

        it_exp += 1

    print(f"Exponentiated‑gradient finished after {it_exp} iterations; "
          f"best score so far {best_score:.6f}")

    # --------------------------------------------------------------
    # 3 – Guided simulated annealing
    # --------------------------------------------------------------
    cur_a = best_a.copy()
    fft = np.fft.rfft(cur_a, n=2 * n - 1)
    conv = np.fft.irfft(fft * fft, n=2 * n - 1)
    cur_score = 2.0 * n * conv.max()

    # temperature schedule – start relatively hot, end very cold
    T_start = cur_score * 0.10          # 10 % of the current score
    T_end = cur_score * 1e-5
    it_sa = 0

        T = max(T_end, T_start * (0.9995 ** it_sa))

        # ---------------------------------------------------------
        # 3.1 – compute a gradient‑based proposal distribution
        # ---------------------------------------------------------
        grad = subgradient_of_max_conv(cur_a, conv)

        # source – biased towards indices that contribute most to the
        # current maximal convolution entry (large gradient)
        src_probs = grad / grad.sum()
        src = np.random.choice(n, p=src_probs)

        # target – biased towards indices with *small* gradient (inverse)
        inv = 1.0 / (grad + 1e-12)
        tgt_probs = inv / inv.sum()
        tgt = np.random.choice(n, p=tgt_probs)

        if src == tgt or cur_a[src] <= 0.0:
            # trivial move – skip
            it_sa += 1
            continue

        # transfer a random fraction of the source mass
        delta = cur_a[src] * np.random.rand()
        cand = cur_a.copy()
        cand[src] -= delta
        cand[tgt] += delta

        # evaluate candidate (full convolution)
        fft = np.fft.rfft(cand, n=2 * n - 1)
        conv_c = np.fft.irfft(fft * fft, n=2 * n - 1)
        cand_score = 2.0 * n * conv_c.max()

        # Metropolis acceptance
        if cand_score < cur_score:
            accept = True
        else:
            accept_prob = math.exp(-(cand_score - cur_score) / max(T, 1e-12))
            accept = np.random.rand() < accept_prob

        if accept:
            cur_a = cand
            conv = conv_c
            cur_score = cand_score
            if cand_score < best_score:
                best_score = cand_score
                best_a = cand.copy()
                print(f"SA iter {it_sa:7d}: new best {best_score:.6f}")

        # occasional diversification – replace current point by a fresh
        # random Dirichlet draw (roughly uniform on the simplex)
        if it_sa % 50000 == 0 and it_sa > 0:
            cur_a = np.random.dirichlet(np.ones(n))
            fft = np.fft.rfft(cur_a, n=2 * n - 1)
            conv = np.fft.irfft(fft * fft, n=2 * n - 1)
            cur_score = 2.0 * n * conv.max()
            print(f"SA iter {it_sa}: random restart performed")

        # periodic simple‑status output
        if it_sa % 20000 == 0 and it_sa > 0:
            print(f"SA iter {it_sa:7d}: current best {best_score:.6f}")

        it_sa += 1

    print(f"Simulated‑annealing finished after {it_sa} iterations; "
          f"final best score = {best_score:.6f}")

    # --------------------------------------------------------------
    # 4 – Return the best sequence (still normalised to sum == 1)
    # --------------------------------------------------------------
    best_a = best_a / best_a.sum()
    return [float(v) for v in best_a.tolist()]
