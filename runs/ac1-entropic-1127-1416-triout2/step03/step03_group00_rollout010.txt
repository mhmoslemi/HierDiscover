def propose_candidate(seed=0, budget_s=1000):
    import time, numpy as np, math, sys
    np.random.seed(seed)
    start = time.time()
    deadline = start + budget_s - 0.1

    # load existing best if present
    try:
        init_best = np.array(height_sequence_1, dtype=float)
        init_best = init_best / init_best.sum()
        n = init_best.size
        best_a = init_best.copy()
        best_score = compute_score(best_a)
        print(f"Loaded existing best of length {n}, score = {best_score:.6f}")
    except NameError:
        n = 5146
        best_a = np.full(n, 1.0/n)
        best_score = compute_score(best_a)
        print(f"Using uniform start length {n}, score = {best_score:.6f}")

    # store some candidate generation functions
    init_functions = []

    def init_beta():
        a = beta_sequence(np.random.uniform(0.2,4.0),
                          np.random.uniform(0.2,4.0), n)
        return a / a.sum()
    init_functions.append(init_beta)

    def init_dirichlet():
        a = np.random.gamma(shape=1.0, scale=1.0, size=n)
        return a / a.sum()
    init_functions.append(init_dirichlet)

    def init_edge():
        # linear decreasing from edges to center
        idx = np.arange(n)
        dist = np.minimum(idx, n-1-idx)
        a = dist.astype(float) + 1e-6
        return a / a.sum()
    init_functions.append(init_edge)

    def init_sparse():
        a = np.zeros(n)
        k = max(2, n // 1000)  # ~ few nonzeros
        chosen = np.random.choice(n, size=k, replace=False)
        vals = np.random.rand(k)
        a[chosen] = vals
        a /= a.sum()
        return a
    init_functions.append(init_sparse)

    # define evaluation using compute_score for speed
    def eval_score(a):
        return compute_score(a)

    # main loop for restarts
    restart = 0
    while time.time() < deadline - 5.0:  # keep 5s for final.
        # select a random init method
        init_func = np.random.choice(init_functions)
        a = init_func()
        # optionally blend with best known to keep good region
        if np.random.rand() < 0.2:
            a = 0.5 * a + 0.5 * best_a
            a /= a.sum()
        # parameters
        max_iters = 2000
        eta0 = 0.8
        alpha = 5.0  # starting smoothing
        alpha_factor = 1.001  # increase each iteration
        for it in range(max_iters):
            # compute conv via FFT length m = 2*n-1
            fft_a = np.fft.rfft(a, n=2*n-1)
            conv = np.fft.irfft(fft_a * fft_a, n=2*n-1)
            # compute softmax weights
            maxc = conv.max()
            # stability shift
            exp_vals = np.exp(alpha * (conv - maxc))
            sum_exp = exp_vals.sum()
            if not np.isfinite(sum_exp) or sum_exp == 0.0:
                # fall back to uniform weights
                p = np.full_like(conv, 1.0/conv.size)
            else:
                p = exp_vals / sum_exp
            # correlation to get gradient
            # compute correlation via FFT
            L = 1 << ((3*n-2).bit_length())
            fft_a_full = np.fft.rfft(a, n=L)
            fft_p = np.fft.rfft(p, n=L)
            corr = np.fft.irfft(fft_a_full * np.conj(fft_p), n=L)
            offset = len(p) - 1
            g = corr[offset:offset + n]

            eta = eta0 / math.sqrt(it + 1.0)
            # exponentiated gradient update
            a = a * np.exp(-eta * g)
            # renormalize

                a = np.full(n, 1.0/n)

    print(f"Search finished. Best score {best_score:.6f}, length {n}")
    # return as list of floats
    return [float(v) for v in best_a.tolist()]

"""
Propose a step‑function (non‑negative vector) that minimizes
evaluate_sequence().

Algorithm – two‐level search
--------------------------------
1.  **Multiple restarts** – we repeatedly create a new start vector by
    sampling from a handful of simple families (Beta‑densities,
    Dirichlet, edge‑concentrated, sparse).  The best vector seen so far
    is kept across restarts.

2.  **Entropy‑smoothed mirror descent** – on each restart we minimise
    the smooth approximation

        f_α(a) = (1/α)·log Σ_k exp(α·(a∗a)_k)

    of the L∞‑norm of the self‑convolution.  Its gradient is
    g_i = Σ_k p_k·a_{k‑i} where p is the soft‑max over the
    convolution entries.  An exponentiated‑gradient (mirror descent on
    the simplex) step

        a_i ← a_i·exp(−η·g_i) ,   a ← a / Σ a

    is performed with a diminishing step‑size η and a slowly increasing
    smoothing parameter α.  This keeps all iterates on the probability
    simplex and drives the distribution towards a shape with a flat
    autocorrelation.

3.  **Short simulated‑annealing refinement** – after the mirror‑descent
    phase a quick SA pass (mass‑transfer moves) is run to escape any
    remaining local minima.

The whole procedure respects the 1000 s time budget; the best vector
found is returned (normalised to sum = 1, which is the natural scaling
for evaluate_sequence()).  The implementation uses only NumPy and
FFT‑based convolutions for O(n log n) per iteration (n≈5 k).
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Helper functions (top level – no closures)
# ----------------------------------------------------------------------


def simplex_normalise(v: np.ndarray) -> np.ndarray:
    """Return v scaled to have sum = 1 (preserves non‑negativity)."""
    s = v.sum()
    if s <= 0 or not np.isfinite(s):
        # fall back to uniform distribution
        return np.full_like(v, 1.0 / v.size)
    return v / s


def compute_score(a: np.ndarray) -> float:
    """
    Exact value of evaluate_sequence() for a vector that already satisfies
    sum(a) == 1.  Formula: 2·n·max(conv(a,a)).
    """
    n = a.shape[0]
    # linear convolution length = 2*n‑1
    fft_a = np.fft.rfft(a, n=2 * n - 1)
    conv = np.fft.irfft(fft_a * fft_a, n=2 * n - 1)
    return 2.0 * n * conv.max()


def beta_sequence(alpha: float, beta: float, n: int) -> np.ndarray:
    """
    Discretised Beta(α,β) density on n points, normalised to sum = 1.
    Mid‑point rule avoids the singular endpoints.
    """
    x = (np.arange(n) + 0.5) / n
    logw = (alpha - 1.0) * np.log(x) + (beta - 1.0) * np.log(1.0 - x)
    # prevent overflow / underflow
    logw = np.clip(logw, -700.0, 700.0)
    w = np.exp(logw)
    if not np.isfinite(w.sum()) or w.sum() == 0.0:
        w = np.ones_like(w)
    return w / w.sum()


def init_dirichlet(n: int) -> np.ndarray:
    """Draw a random Dirichlet(1,…,1) vector (i.e. i.i.d. Gamma(1))."""
    a = np.random.gamma(shape=1.0, scale=1.0, size=n)
    return simplex_normalise(a)


def init_edge_concentrated(n: int) -> np.ndarray:
    """
    Linear “edge‑heavy” shape: larger values near the two ends,
    smaller in the middle.
    """
    idx = np.arange(n)
    dist_to_center = np.minimum(idx, n - 1 - idx)
    a = dist_to_center.astype(float) + 1.0  # edges get larger weight
    return simplex_normalise(a)


def init_sparse(n: int, k: int = 10) -> np.ndarray:
    """Place mass on k randomly chosen positions."""
    a = np.zeros(n, dtype=float)
    chosen = np.random.choice(n, size=min(k, n), replace=False)
    vals = np.random.rand(chosen.size) + 1e-6
    a[chosen] = vals
    return simplex_normalise(a)

# Main candidate generator
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 0, budget_s: int = 1000) -> list[float]:
    """
    Search for a non‑negative height sequence that minimises
    evaluate_sequence() within ``budget_s`` seconds.
    Returns the best sequence found (as a list of floats, sum == 1).
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.2      # a tiny safety margin

    # ------------------------------------------------------------------
    # 0. Load a previously known solution (if any) and initialise globals
    # ------------------------------------------------------------------
    try:
        # ``height_sequence_1`` is supplied by the surrounding environment.
        init_best = np.array(height_sequence_1, dtype=float)  # type: ignore[name-defined]
        init_best = simplex_normalise(init_best)
        n = init_best.size
        best_a = init_best.copy()
        best_score = compute_score(best_a)
        print(f"[Init] Loaded height_sequence_1 (n={n}), score={best_score:.6f}")
    except NameError:
        # Fallback – uniform distribution of length ≈5 k
        n = 5146
        best_a = np.full(n, 1.0 / n, dtype=float)
        best_score = compute_score(best_a)
        print(f"[Init] No previous sequence – using uniform start, n={n}, score={best_score:.6f}")

    # ------------------------------------------------------------------
    # 1. Helper to evaluate a vector quickly (uses compute_score)
    # ------------------------------------------------------------------
    def eval_fast(v: np.ndarray) -> float:
        """Fast evaluation assuming sum(v) == 1."""
        return compute_score(v)

    # ------------------------------------------------------------------
    # 2. Collection of simple initialisers (all produce sum==1)
    # ------------------------------------------------------------------
    init_functions = [
        lambda: beta_sequence(np.random.uniform(0.2, 5.0),
                              np.random.uniform(0.2, 5.0), n),
        lambda: init_dirichlet(n),
        lambda: init_edge_concentrated(n),
        lambda: init_sparse(n, k=np.random.randint(5, 30)),
    ]

    # ------------------------------------------------------------------
    # 3. Main restart loop
    # ------------------------------------------------------------------
    restart_counter = 0
    while time.time() < deadline - 8.0:       # keep a few seconds for final return
        # ----- pick a start -----
        a = np.random.choice(init_functions)()
        # occasionally blend with the currently best vector (helps keep good mass)
        if np.random.rand() < 0.2:
            a = 0.5 * a + 0.5 * best_a
            a = simplex_normalise(a)

        # ----- mirror‑descent on the log‑sum‑exp smoothed max -----
        max_iters = 2000
        eta0 = 0.8           # base step size
        alpha_start = 1.0
        alpha_end = 80.0
        for it in range(max_iters):
            # stop early if we run out of time
            if time.time() > deadline - 4.0:
                break

            # 1) self‑convolution (FFT, length 2n‑1)
            fft_a = np.fft.rfft(a, n=2 * n - 1)
            conv = np.fft.irfft(fft_a * fft_a, n=2 * n - 1)

            # 2) soft‑max weights over the convolution entries
            maxc = conv.max()
            # shift for numerical stability (maxc is subtracted)
            exp_vals = np.exp(alpha_start * (conv - maxc))
            sum_exp = exp_vals.sum()
            if sum_exp == 0.0 or not np.isfinite(sum_exp):
                p = np.full_like(conv, 1.0 / conv.size)
            else:
                p = exp_vals / sum_exp

            # 3) gradient g_i = Σ_k p_k·a_{k‑i}
            # this is the (non‑circular) cross‑correlation of a with p
            L = 1 << ((3 * n - 2).bit_length())          # next power of two ≥3n‑2
            fft_a_full = np.fft.rfft(a, n=L)
            fft_p = np.fft.rfft(p, n=L)
            corr = np.fft.irfft(fft_a_full * np.conj(fft_p), n=L)
            offset = len(p) - 1               # index where lag 0 appears
            g = corr[offset:offset + n]

            # 4) exponentiated‑gradient update (mirror descent on simplex)
            eta = eta0 / math.sqrt(it + 1.0)
            a = a * np.exp(-eta * g)
            a = simplex_normalise(a)

            # 5) linearly increase the smoothing parameter α
            alpha_start = alpha_start + (alpha_end - alpha_start) / max_iters

            # occasionally record improvement
            if (it & 31) == 0:               # every 32 iterations
                cur_score = eval_fast(a)
                if cur_score < best_score:
                    best_score = cur_score
                    best_a = a.copy()
                    print(f"[Restart {restart_counter}] it {it:4d} → "
                          f"new best {best_score:.6f}")

        # ----- short simulated‑annealing refinement on the current best -----
        cur_a = best_a.copy()
        cur_score = eval_fast(cur_a)
        # start temperature as a small fraction of current score
        T = cur_score * 0.02
        T_end = cur_score * 1e-5
        sa_iters = 0
        while time.time() < deadline - 2.0 and T > T_end:
            # simple mass‑transfer move (keeps sum exactly 1)
            src = np.random.choice(n, p=cur_a)
            tgt = np.random.randint(n - 1)
            if tgt >= src:
                tgt += 1
            if cur_a[src] <= 0.0:
                continue
            delta = cur_a[src] * np.random.rand()
            cand = cur_a.copy()
            cand[src] -= delta
            cand[tgt] += delta

            cand_score = eval_fast(cand)
            if cand_score < cur_score:
                accept = True
            else:
                accept = np.random.rand() < math.exp(-(cand_score - cur_score) / (T + 1e-12))

            if accept:
                cur_a = cand
                cur_score = cand_score
                if cand_score < best_score:
                    best_score = cand_score
                    best_a = cand.copy()
                    print(f"[SA  R{restart_counter}] iter {sa_iters:5d} → "
                          f"new best {best_score:.6f}")

            # geometric cooling
            T *= 0.9995
            sa_iters += 1

        restart_counter += 1

    # ------------------------------------------------------------------
    # 4. Return the best vector (already normalised to sum = 1)
    # ------------------------------------------------------------------
    print(f"[Done] Best score after {restart_counter} restarts: {best_score:.6f}")
    return [float(v) for v in best_a.tolist()]
