def upsample(a, new_n):
    old_n = len(a)
    x_old = np.linspace(0, 1, old_n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)

def beta_shape(n, alpha, beta, eps=1e-12):
    i = np.arange(n, dtype=float)
    left = (i + 0.5) ** (alpha - 1)
    right = (n - 0.5 - i) ** (beta - 1)
    vals = left * right
    vals = np.maximum(vals, 0)  # ensure non-negative
    total = vals.sum()
    if total == 0:
        return np.full(n, 1.0/n)

def adam_optimize(initial_a, deadline, lam_start=1.0, lam_end=200.0, lam_ramp_iters=20000,
                 alpha0=0.05, beta1=0.9, beta2=0.999, eps=1e-8, max_iter=200000, rng=None):
    a = initial_a.astype(float).copy()
    n = len(a)
    # normalize sum to sqrt(2n)
    a = a / a.sum() * np.sqrt(2.0 * n)
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    best_a = a.copy()
    best_obj = compute_score(a)  # internal fast objective equals official
    for it in range(1, max_iter+1):
        if time.time() >= deadline:
            break
        lam = lam_start + (lam_end - lam_start) * min(1.0, it / lam_ramp_iters)
        conv = convolution_fft(a, a)
        # softmax
        x = lam * conv
        mx = np.max(x)
        w = np.exp(x - mx)
        w_sum = np.sum(w)
        if w_sum == 0:
            continue
        w /= w_sum
        # gradient
        grad = gradient_from_weights(a, w)
        # Adam update
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad ** 2)
        m_hat = m / (1 - beta1**it)
        v_hat = v / (1 - beta2**it)
        step = alpha0 * m_hat / (np.sqrt(v_hat) + eps)
        a = a - step
        # keep non-neg
        a = np.maximum(a, 0.0)
        # renorm
        sum_a = a.sum()
        if sum_a == 0:
            a = np.full_like(a, 1.0 / n)
            sum_a = a.sum()
        a = a * (np.sqrt(2.0 * n) / sum_a)
        # evaluate objective after renorm: use compute_score on a
        cur_obj = compute_score(a)
        if cur_obj < best_obj:
            best_obj = cur_obj
            best_a = a.copy()
        # occasional noise
        if rng is not None and (it % 3000 == 0):
            a += rng.normal(scale=1e-6, size=n)
            a = np.maximum(a, 0.0)
            # renorm again
            a = a / a.sum() * np.sqrt(2.0 * n)
        if it % 5000 == 0:
            print(f"    ADAM it {it:,d}, lam={lam:.1f}, cur={cur_obj:.6f}, best={best_obj:.6f}")
    return best_a, best_obj

"""
Hybrid search for a low‑autocorrelation non‑negative step function.

The algorithm consists of three stages:

1. **Parametric sweep** –  Build a modest library of analytically
   parametrised shapes (uniform, arcsine, power‑law, beta‑like, exponential
   Dirichlet samples and random convex mixtures of them).  All candidates are
   normalised to unit ℓ₁‑norm and scored quickly with the official objective.

2. **Adam‑based soft‑max optimisation** –  Starting from the most promising
   candidates we run a projected Adam optimiser on the smooth
   log‑sum‑exp surrogate of the ‑max‑convolution.  The temperature λ is
   annealed from a small value to a large one (≈200–400) so that the
   surrogate gradually turns into the true max.  After each gradient step
   the vector is projected onto the simplex and then re‑scaled to the
   “canonical” sum √(2 n); for such normalisation the objective equals the
   maximum convolution value, which is cheap to evaluate.

3. **Final polishing** –  The globally best sequence is up‑sampled to a
   larger length (4096) and refined again with a longer schedule and a
   higher λ.  The routine respects the overall time budget (default 1000 s)
   and always returns a feasible list of floats.

The code is self‑contained, uses only NumPy and the supplied evaluate_sequence
function, and prints a few progress messages.
"""

import time
import numpy as np
from typing import List, Tuple

# ----------------------------------------------------------------------
# Low‑level numeric utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Linear convolution via real FFT (full length, zero‑padded to power‑of‑2)."""
    m = len(a) + len(b) - 1
    n_fft = next_pow2(m)
    A = np.fft.rfft(a, n=n_fft)
    B = np.fft.rfft(b, n=n_fft)
    C = A * B
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


def softmax_weights(conv: np.ndarray, lam: float) -> np.ndarray:
    """Numerically stable soft‑max over the convolution entries."""
    x = lam * conv
    mx = np.max(x)
    w = np.exp(x - mx)
    w /= np.sum(w)
    return w


def gradient_from_weights(a: np.ndarray, w: np.ndarray) -> np.ndarray:
    """
    Gradient of Σ_k w_k·(a*a)_k w.r.t a.
    Implements   g_j = 2· Σ_i a_i·w_{i+j}.
    Computed via one FFT‑based convolution.
    """
    w_rev = w[::-1]
    conv = convolution_fft(a, w_rev)               # length = 3·n‑2
    idx = 2 * len(a) - 2
    grads = 2.0 * conv[idx - np.arange(len(a))]
    return grads


def compute_score(a: np.ndarray) -> float:
    """
    Official objective (scale‑invariant):
        2·n·max(a*a) / (sum(a)²)
    """
    n = len(a)
    conv = convolution_fft(a, a)
    max_val = float(np.max(conv))
    s = float(np.sum(a))
    if s < 1e-12:
        return float("inf")
    return 2.0 * n * max_val / (s * s)


# ----------------------------------------------------------------------
# Parametric families for initial candidates
# ----------------------------------------------------------------------


def arcsine_shape(n: int, eps: float = 0.25) -> np.ndarray:
    """Discrete arcsine‑type density, normalised to sum = 1."""
    i = np.arange(n, dtype=float)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - 0.5 - i) + eps)
    return vals / vals.sum()


def generate_shape(n: int, p: float, eps: float) -> np.ndarray:
    """Symmetric power‑law shape a_i ∝ ((i+ε)(n‑1‑i+ε))⁻ᵖ, normalised."""
    i = np.arange(n, dtype=float)
    base = ((i + eps) * (n - 1 - i + eps)) ** (-p)
    total = base.sum()
    if total == 0.0:
        return np.full(n, 1.0 / n)
    return base / total


def beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    """Beta‑like shape a_i ∝ (i+½)^{α‑1}·(n‑½‑i)^{β‑1}, normalised."""
    i = np.arange(n, dtype=float)
    left = (i + 0.5) ** (alpha - 1)
    right = (n - 0.5 - i) ** (beta - 1)
    vals = left * right
    total = vals.sum()
    if total == 0.0:
        return np.full(n, 1.0 / n)
    return vals / total


def upsample(a: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation to a larger length, followed by normalisation."""
    old_n = len(a)
    x_old = np.linspace(0.0, 1.0, old_n)
    x_new = np.linspace(0.0, 1.0, new_n)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0.0)
    # canonical normalisation (sum = sqrt(2 n))
    a_new = a_new / a_new.sum() * np.sqrt(2.0 * new_n)
    return a_new


def generate_initial_candidates(n: int, rng: np.random.Generator) -> List[np.ndarray]:
    """Build a diverse set of starting vectors for length n."""
    candidates: List[np.ndarray] = []

    # 1) elementary families
    candidates.append(np.full(n, 1.0 / n))                     # uniform
    for eps in [0.001, 0.01, 0.05, 0.1, 0.25, 0.5]:
        candidates.append(arcsine_shape(n, eps))

    for p in [0.2, 0.4, 0.6, 0.8]:
        for eps in [0.01, 0.1, 1.0]:
            candidates.append(generate_shape(n, p, eps))

    for a in [0.2, 0.4, 0.6, 0.8]:
        for b in [0.2, 0.4, 0.6, 0.8]:
            candidates.append(beta_shape(n, a, b))

    # 2) Dirichlet / exponential samples
    for _ in range(6):
        vec = rng.exponential(1.0, size=n)
        vec = np.maximum(vec, 0.0)
        candidates.append(vec / vec.sum())

    # 3) random convex mixtures of the above base shapes
    base = candidates.copy()
    m = len(base)
    for _ in range(20):
        w = rng.dirichlet(np.ones(m))
        mix = np.zeros(n)
        for wi, bi in zip(w, base):
            mix += wi * bi
        candidates.append(mix)

    return candidates


# ----------------------------------------------------------------------
# Adam optimiser on the smooth max‑convolution surrogate
# ----------------------------------------------------------------------


def adam_optimize(
    initial_a: np.ndarray,
    deadline: float,
    *,
    lam_start: float = 1.0,
    lam_end: float = 200.0,
    lam_ramp_iters: int = 20000,
    alpha0: float = 0.05,
    beta1: float = 0.9,
    beta2: float = 0.999,
    eps: float = 1e-8,
    max_iter: int = 200_000,
    rng: np.random.Generator | None = None,
) -> Tuple[np.ndarray, float]:
    """
    Projected Adam on the log‑sum‑exp surrogate of the max‑convolution.
    Returns the best vector seen and its (non‑smooth) official score.
    """
    a = initial_a.astype(float).copy()
    n = len(a)

    # canonical normalisation: sum = sqrt(2·n)  →  objective = max(conv)
    a = a / a.sum() * np.sqrt(2.0 * n)

    m = np.zeros_like(a)
    v = np.zeros_like(a)

    best_a = a.copy()
    best_score = compute_score(a)

    for it in range(1, max_iter + 1):
        if time.time() >= deadline:
            break

        # anneal the soft‑max temperature
        lam = lam_start + (lam_end - lam_start) * min(1.0, it / lam_ramp_iters)

        # convolution and soft‑max weights
        conv = convolution_fft(a, a)
        w = softmax_weights(conv, lam)

        # gradient of the weighted sum
        grad = gradient_from_weights(a, w)

        # Adam update (step = -η·∇)
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        step = alpha0 * m_hat / (np.sqrt(v_hat) + eps)
        a = a - step

        # non‑negativity & re‑projection onto the simplex
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s == 0.0:
            a = np.full_like(a, 1.0 / n)
            s = a.sum()
        a = a * (np.sqrt(2.0 * n) / s)

        # occasional tiny exploration noise
        if rng is not None and (it % 3000 == 0):
            a += rng.normal(scale=1e-6, size=n)
            a = np.maximum(a, 0.0)
            a = a / a.sum() * np.sqrt(2.0 * n)

        # evaluate the true objective (max‑convolution)
        cur_score = compute_score(a)
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()

        # periodic progress report
        if it % 5000 == 0:
            print(f"    Adam it={it:,d} λ={lam:.1f} cur={cur_score:.6f} best={best_score:.6f}")

    return best_a, best_score


# ----------------------------------------------------------------------
# Main entry point required by the competition harness
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs) -> List[float]:
    """
    Search for a non‑negative step‑function with a low ``evaluate_sequence`` value.
    The routine respects the given time budget, explores several parametric families,
    refines the most promising candidates with a projected Adam optimiser,
    and finally polishing the global best at a larger resolution.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 2.0   # safety margin

    best_seq: List[float] = []
    best_val: float = float("inf")

    # ------------------------------------------------------------------
    # 0. Optional fallback from a previously discovered construction.
    # ------------------------------------------------------------------
    try:
        fallback = [float(x) for x in height_sequence_1]   # type: ignore
        val = evaluate_sequence(fallback)
        print(f"Fallback (height_sequence_1) score = {val:.6f}")
        if val < best_val:
            best_val = val
            best_seq = fallback
    except Exception:
        pass

    # ------------------------------------------------------------------
    # 1.  Coarse search at a moderate length (n = 2048)
    # ------------------------------------------------------------------
    n0 = 2048
    print(f"\n=== Coarse stage (n = {n0}) ===")
    candidates = generate_initial_candidates(n0, rng)

    # Quick scoring of all candidates
    scored = [(compute_score(c), c) for c in candidates]
    scored.sort(key=lambda x: x[0])
    top_candidates = [c for _, c in scored[:4]]   # keep a few best

    print(f"Best coarse candidate before refinement: {scored[0][0]:.6f}")

    # ------------------------------------------------------------------
    # 2.  Refine the elite candidates with Adam (still at n0)
    # ------------------------------------------------------------------
    for idx, cand in enumerate(top_candidates):
        if time.time() >= deadline - 5.0:
            break
        print(f"\nRefining candidate {idx + 1}/{len(top_candidates)} (n = {n0})")
        remaining = max(0.0, deadline - time.time())
        # allocate roughly 15 optimisation steps per second
        max_iter = int(min(150_000, max(2_000, remaining * 15)))
        refined, _ = adam_optimize(

        if val < best_val:
            best_val = val
            best_seq = refined.tolist()
            print(f"    *** New global best: {best_val:.6f} ***")

    # ------------------------------------------------------------------
    # 3.  Final polishing at a larger resolution (n = 4096)
    # ------------------------------------------------------------------
    if best_seq and time.time() < deadline - 5.0:
        print("\n=== Final polishing stage (n = 4096) ===")
        a_big = upsample(np.array(best_seq, dtype=float), 4096)

        remaining = max(0.0, deadline - time.time())
        max_iter = int(min(250_000, max(5_000, remaining * 15)))
        a_refined, _ = adam_optimize(
            initial_a=a_big,
            deadline=deadline,
            lam_start=1.0,
            lam_end=400.0,
            lam_ramp_iters=40000,
            alpha0=0.03,
            max_iter=max_iter,
            rng=rng,
        )
        final_val = evaluate_sequence(a_refined.tolist())
        print(f"Final polished official score = {final_val:.6f}")
        if final_val < best_val:
            best_val = final_val
            best_seq = a_refined.tolist()
            print("*** Updated global best after final polish ***")

    # ------------------------------------------------------------------
    if not best_seq:
        best_seq = [1.0] * 256
        best_val = evaluate_sequence(best_seq)

    print(f"\nSearch finished. Best reported score = {best_val:.6f}")
    return best_seq
