def subgradient_descent(a, deadline, max_iter=50000, step0=0.5):
    a = project_to_simplex(a.copy())
    best_a = a.copy()
    best_sc = compute_score(a)
    n = len(a)
    L = 2*n - 1
    for it in range(1, max_iter+1):
        if time.time() >= deadline:
            break
        conv = convolution_fft(a,a)
        max_val = conv.max()
        # Identify positions i where conv[i] close to max
        tol = max_val * 1e-12 + 1e-14
        mask = np.abs(conv - max_val) <= tol
        w = mask.astype(float)  # weights; could be normalized but not needed for subgradient
        # compute weighted gradient as above
        w_rev = w[::-1]
        conv_rev = convolution_fft(a, w_rev)
        weighted_grad = 2.0 * conv_rev[L - 1 - np.arange(n)]
        grad = weighted_grad  # since we are optimizing compute_score = 2*n*max, factor 2*n may be absorbed in step size
        # step size schedule
        eta = step0 / np.sqrt(it)
        a = a - eta * grad
        # projection
        a = project_to_simplex(a)
        cur_sc = compute_score(a)
        if cur_sc < best_sc:
            best_sc = cur_sc
            best_a = a.copy()
    return best_a
```

def randomized_local_search(a, deadline, max_iters=200000, init_temp=0.01, decay=0.9999):
    best_a = a.copy()
    best_sc = compute_score(a)
    n = len(a)
    temp = init_temp
    for it in range(max_iters):
        if time.time() >= deadline:
            break
        # propose move
        i, j = rng.choice(n, size=2, replace=False)
        # choose delta up to fraction of a[i]
        max_delta = a[i] * 0.1
        delta = rng.random() * max_delta
        # create new candidate
        cand = best_a.copy()
        cand[i] -= delta
        cand[j] += delta
        # ensure nonnegative
        if cand[i] < 0:
            continue # reject
        # project onto simplex for safety, but sum preserved already
        # cand = project_to_simplex(cand) # optional
        sc = compute_score(cand)
        if sc < best_sc or rng.random() < np.exp(-(sc - best_sc) / temp):
            best_a = cand
            best_sc = sc
        temp *= decay
        # occasional best update after many iterations ...
    return best_a

def adam_optimize_smooth(initial_a, deadline, max_iter=50000, lr0=0.2, lam_start=2.0, lam_end=80.0,
                         noise_every=2000, noise_scale=5e-5):
    a = project_to_simplex(initial_a.astype(float).copy())
    best_a = a.copy()
    best_sc = compute_score(a)
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    n = len(a)
    for it in range(1, max_iter+1):
        if time.time() >= deadline:
            break
        lam = lam_start + (lam_end - lam_start) * min(1.0, it / max_iter)
        grad = smooth_gradient(a, lam)
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad ** 2)
        m_hat = m / (1 - beta1 ** it)
        v_hat = v / (1 - beta2 ** it)
        step = lr0 * m_hat / (np.sqrt(v_hat) + eps)
        a_new = a - step
        # Add jitter
        if noise_every and (it % noise_every == 0):
            a_new += np.random.randn(n) * noise_scale
        a_new = project_to_simplex(a_new)
        cur_sc = compute_score(a_new)
        if cur_sc < best_sc:
            best_sc = cur_sc
            best_a = a_new.copy()
        a = a_new
        # occasional printing
        if it % 5000 == 0:
            print(f"    AdamSmooth iter={it} cur={cur_sc:.6f} best={best_sc:.6f}")
    return best_a, best_sc

def subgradient_descent_exact(initial_a, deadline, max_iter=50000, step0=0.1):
    a = project_to_simplex(initial_a.copy())
    best_a = a.copy()
    best_sc = compute_score(a)
    n = len(a)
    L = 2 * n - 1
    for it in range(1, max_iter+1):
        if time.time() >= deadline:
            break
        conv = convolution_fft(a, a)
        max_val = np.max(conv)
        # identify max positions within tolerance
        tol = max_val * 1e-12 + 1e-14
        mask = (np.abs(conv - max_val) <= tol).astype(float)
        # compute weighted gradient using mask
        mask_rev = mask[::-1]
        conv_rev = convolution_fft(a, mask_rev)
        weighted_grad = 2.0 * conv_rev[L - 1 - np.arange(n)]
        grad = weighted_grad  # derivative of max_b; factor 2n is absorbed into step size
        eta = step0 / np.sqrt(it)
        a = a - eta * grad
        a = project_to_simplex(a)
        cur_sc = compute_score(a)
        if cur_sc < best_sc:
            best_sc = cur_sc
            best_a = a.copy()
        if it % 5000 == 0:
            print(f"    Subgrad iter={it} cur={cur_sc:.6f} best={best_sc:.6f}")
    return best_a

def simulated_annealing(a, deadline, max_iters=200000, init_temp=0.01, decay=0.99995, min_temp=1e-6):
    cur = a.copy()
    cur_sc = compute_score(cur)
    best = cur.copy()
    best_sc = cur_sc
    n = len(cur)
    temp = init_temp
    while max_iters>0 and time.time() < deadline:
        max_iters -= 1
        # propose move
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        # transfer delta from i to j
        delta = cur[i] * 0.1 * rng.random()  # up to 10% of mass at i
        if delta <= 0:
            continue
        cand = cur.copy()
        cand[i] -= delta
        cand[j] += delta
        # ensure non-negative (i not negative)
        if cand[i] < 0:
            continue
        # sum unchanged, but may cause small floating error; reproject onto simplex to avoid drift
        cand = project_to_simplex(cand)
        cand_sc = compute_score(cand)
        if cand_sc < cur_sc or rng.random() < np.exp(-(cand_sc - cur_sc) / max(temp, 1e-12)):
            cur = cand
            cur_sc = cand_sc
            if cur_sc < best_sc:
                best = cur.copy()
                best_sc = cur_sc
        temp = max(temp * decay, min_temp)
        if max_iters % 5000 == 0:
            print(f"    SA iter={max_iters} cur={cur_sc:.6f} best={best_sc:.6f}")
    return best
```

"""
Propose a low‑discrepancy step function.

The search combines several complementary ideas:

1.  **Diverse random sampling** –  Beta‑family, inverse‑polynomial
    (generalised arcsine), mixtures with the uniform vector, simple
    triangular shapes, etc.  All candidates are normalised to Σa = 1,
    because the official objective is invariant under scaling.

2.  **Smooth Adam optimisation** –  Replace  max(conv(a,a))  by a
    temperature‑controlled soft‑max (log‑sum‑exp).  The gradient of the
    smooth surrogate is computed analytically via an FFT‑based
    convolution and an Adam update is applied.  After each step the
    vector is projected onto the probability simplex.

3.  **Exact sub‑gradient descent** –  For the true objective
    f(a)=2·n·max(conv(a,a)) (sum is fixed to 1) the sub‑gradient is
    2·n·∑_{k∈K} shift(a,k) where K are the indices attaining the
    maximum.  A simple diminishing‑step scheme with Euclidean projection
    further reduces the bound.

4.  **Simulated‑annealing fine‑tuning** –  Small random mass transfers
    between coordinates are accepted according to a cooling schedule.
    The fast surrogate 2·n·max(conv)  is used for the Metropolis test.

All phases respect the global time budget (default 1000 s).  The best
vector found is finally evaluated with the official ``evaluate_sequence``
and returned.  The implementation uses only NumPy and the standard
library, stays well below the time limit and guarantees a non‑empty,
non‑negative result.
"""

import time
import numpy as np
from typing import List, Tuple

# ----------------------------------------------------------------------
# Low‑level utilities (no closures)
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """
    Linear convolution via real FFT.
    Length = len(a)+len(b)-1.
    """
    m = len(a) + len(b) - 1
    n_fft = next_pow2(m)
    A = np.fft.rfft(a, n=n_fft)
    B = np.fft.rfft(b, n=n_fft)
    C = A * B
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


def compute_score(a: np.ndarray) -> float:
    """
    Fast surrogate for the official score.
    After projection onto the simplex Σa = 1 the denominator disappears,
    therefore the score is simply 2·n·max(conv(a,a)).
    """
    n = len(a)
    conv = convolution_fft(a, a)
    max_val = float(np.max(conv))
    return 2.0 * n * max_val


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection onto the probability simplex {x ≥ 0, Σx = 1}.
    Duchi et al., 2008.
    """
    if np.all(v <= 0):
        # uniform fallback – never happens in practice
        return np.full_like(v, 1.0 / v.size)

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho = np.where(u - cssv / np.arange(1, v.size + 1) > 0)[0]
    if len(rho) == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def arcsine_initial(n: int) -> np.ndarray:
    """Classic arcsine shape: a_i ∝ 1/√((i+½)*(n‑½‑i))."""
    i = np.arange(n, dtype=float)
    eps = 0.25
    vals = 1.0 / np.sqrt((i + 0.5) * (n - 0.5 - i) + eps)
    vals /= vals.sum()
    return vals


def generate_beta_shape(n: int, alpha: float, beta: float, eps: float) -> np.ndarray:
    """
    (i+ε)^{α‑1}·(n‑1‑i+ε)^{β‑1} on {0,…,n‑1}, normalised to Σ=1.
    """
    i = np.arange(n, dtype=float)
    left = (i + eps) ** (alpha - 1.0)
    right = (n - 1 - i + eps) ** (beta - 1.0)
    base = left * right
    total = base.sum()
    if total <= 0.0 or not np.isfinite(total):
        return np.full(n, 1.0 / n)
    return base / total


def generate_power_shape(n: int, p: float, eps: float) -> np.ndarray:
    """
    Generalised arcsine a_i ∝ ((i+½)*(n‑½‑i)+ε)^{-p}.
    p>0, eps≥0.
    """
    i = np.arange(n, dtype=float)
    denom = ((i + 0.5) * (n - 0.5 - i) + eps) ** p
    vals = 1.0 / denom
    total = vals.sum()
    if total <= 0.0 or not np.isfinite(total):
        return np.full(n, 1.0 / n)
    return vals / total


def generate_triangular_shape(n: int, peak: int) -> np.ndarray:
    """Symmetric triangular bump with its apex at ``peak``."""
    left = np.arange(peak + 1)
    right = np.arange(n - peak - 1)[::-1]
    seq = np.concatenate([left, right]).astype(float)
    seq /= seq.sum()
    return seq


def generate_mixture_shape(n: int, weight_arc: float) -> np.ndarray:
    """Mixture (weight_arc)·arcsine + (1‑weight_arc)·uniform."""
    arc = arcsine_initial(n)
    uni = np.full(n, 1.0 / n)
    mix = weight_arc * arc + (1.0 - weight_arc) * uni
    mix /= mix.sum()
    return mix

def smooth_gradient(a: np.ndarray, lam: float) -> np.ndarray:
    """
    Gradient of the soft‑max approximation of max(conv(a,a)).
    Returns ∇_a (2·n·softmax_max) where the sum constraint is handled
    by the subsequent projection onto the simplex.
    """
    n = len(a)
    conv = convolution_fft(a, a)               # length L = 2·n‑1
    max_c = np.max(conv)

    # numerically stable soft‑max
    exp_term = np.exp(lam * (conv - max_c))
    w = exp_term / np.sum(exp_term)            # weights, sum to 1

    # weighted sub‑gradient:   2·∑_k w_k shift(a,k)
    w_rev = w[::-1]
    conv_rev = convolution_fft(a, w_rev)       # length n + L - 1
    L = 2 * n - 1
    weighted_grad = 2.0 * conv_rev[L - 1 - np.arange(n)]

    # objective factor 2·n
    return 2.0 * n * weighted_grad


def adam_optimize_smooth(initial_a: np.ndarray,
                         deadline: float,
                         max_iter: int = 100_000,
                         lr0: float = 0.15,
                         lam_start: float = 2.0,
                         lam_end: float = 80.0,
                         noise_every: int = 2000,
                         noise_scale: float = 5e-5) -> Tuple[np.ndarray, float]:
    """
    Adam optimiser on the smooth surrogate.
    Returns the best vector (projected on the simplex) and its fast score.
    """
    a = project_to_simplex(initial_a.astype(float).copy())
    best_a = a.copy()
    best_sc = compute_score(a)

    beta1, beta2 = 0.9, 0.999
    eps = 1e-8
    m = np.zeros_like(a)
    v = np.zeros_like(a)

    for it in range(1, max_iter + 1):
        if time.time() >= deadline:
            break

        # linear temperature schedule
        lam = lam_start + (lam_end - lam_start) * min(1.0, it / max_iter)

        grad = smooth_gradient(a, lam)

        # Adam moments
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)

        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        step = lr0 * m_hat / (np.sqrt(v_hat) + eps)

        a_new = a - step

        # occasional jitter to avoid stagnation
        if noise_every and (it % noise_every == 0):
            a_new += np.random.randn(a.size) * noise_scale

        a_new = project_to_simplex(a_new)

        cur_sc = compute_score(a_new)
        if cur_sc < best_sc:
            best_sc = cur_sc
            best_a = a_new.copy()

        a = a_new

        if it % 5000 == 0:
            print(f"    AdamSmooth it={it:,d} cur={cur_sc:.6f} best={best_sc:.6f}")

    return best_a, best_sc


def subgradient_descent_exact(initial_a: np.ndarray,
                              deadline: float,
                              max_iter: int = 80_000,
                              step0: float = 0.05) -> np.ndarray:
    """
    Sub‑gradient descent on the exact objective  f(a)=2·n·max(conv(a,a)).
    The step size is η_t = step0 / sqrt(t).  Projection onto the simplex
    keeps the sum fixed to 1.
    """
    a = project_to_simplex(initial_a.copy())
    best_a = a.copy()
    best_sc = compute_score(a)
    n = len(a)
    L = 2 * n - 1

    for it in range(1, max_iter + 1):
        if time.time() >= deadline:
            break

        conv = convolution_fft(a, a)
        max_val = np.max(conv)

        # indices attaining the maximum (tolerance accounts for floating error)
        tol = max_val * 1e-12 + 1e-14
        mask = (np.abs(conv - max_val) <= tol).astype(float)

        # weighted gradient = 2·∑_{k∈K} shift(a,k)
        mask_rev = mask[::-1]
        conv_rev = convolution_fft(a, mask_rev)
        weighted_grad = 2.0 * conv_rev[L - 1 - np.arange(n)]

        eta = step0 / np.sqrt(it)
        a = a - eta * weighted_grad
        a = project_to_simplex(a)

        cur_sc = compute_score(a)
        if cur_sc < best_sc:
            best_sc = cur_sc
            best_a = a.copy()

        if it % 5000 == 0:
            print(f"    SubGrad it={it:,d} cur={cur_sc:.6f} best={best_sc:.6f}")

    return best_a


def simulated_annealing(a: np.ndarray,
                        rng: np.random.Generator,
                        deadline: float,
                        max_iters: int = 120_000,
                        init_temp: float = 0.02,
                        decay: float = 0.9999) -> np.ndarray:
    """
    Simple Metropolis‑style local search.
    Randomly move a small fraction of mass from one coordinate to another.
    The fast surrogate (2·n·max) is used for the acceptance test.
    """
    cur = a.copy()
    cur_sc = compute_score(cur)
    best = cur.copy()
    best_sc = cur_sc
    n = len(cur)
    temp = init_temp

    it = 0
    while it < max_iters and time.time() < deadline:
        it += 1
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue

        # move up to 20 % of the mass at i
        delta = cur[i] * 0.2 * rng.random()
        if delta <= 1e-15:
            continue
        if cur[i] - delta < 0:
            continue

        cand = cur.copy()
        cand[i] -= delta
        cand[j] += delta          # sum is preserved, non‑negativity kept

        cand_sc = compute_score(cand)

        # Metropolis acceptance
        if cand_sc < cur_sc or rng.random() < np.exp(-(cand_sc - cur_sc) / max(temp, 1e-12)):
            cur = cand
            cur_sc = cand_sc
            if cur_sc < best_sc:
                best = cur.copy()
                best_sc = cur_sc

        temp = max(temp * decay, 1e-8)

        if it % 5000 == 0:
            print(f"    SA it={it:,d} cur={cur_sc:.6f} best={best_sc:.6f}")

    return best

    return best_seq
