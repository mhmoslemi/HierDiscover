"""
Propose a low‑value step‑function for the `evaluate_sequence` benchmark.

Algorithm overview
------------------
The search proceeds in four stages, each respecting the global time budget.

1. **Broad exploration** – a large, cheap sampling of families
   (Beta, exponential, arcsine, Dirichlet draws and random
   convex combinations) for a few modest lengths (512, 1024,
   2048).  The best few candidates are retained.

2. **Continuous‑parameter refinement** – the two Beta exponents (α,
   β) are tuned with a tiny stochastic hill‑climb, because the
   objective is very sensitive to changes near the optimum.

3. **Smooth‑max Adam optimisation** – the non‑smooth max in the
   score is replaced by a log‑sum‑exp surrogate.
   An Adam optimiser (projected onto the probability simplex) is
   run on the surrogate for a short while, gradually increasing the
   smoothing parameter α so that the surrogate approaches the true
   max.

4. **Targeted hill‑climbing** – the dominant convolution index
   *k* = argmax(a∗a) is identified, and mass is transferred from the
   position that contributes most to this index to a position that
   contributes least.  This deterministic move improves the true
   objective and finishes the search.

All vectors are kept on the probability simplex (∑a = 1, a_i ≥ 0) so
that the scale‑invariance of the objective is respected.  The final
candidate is returned as a plain Python list of floats.
"""

import time
import math
import numpy as np
from scipy.signal import resample  # FFT‑based resampling (much smoother than linear)

# ----------------------------------------------------------------------
# Basic linear‑algebra utilities
# ----------------------------------------------------------------------


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / v.shape[0])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho = np.where(u - cssv / np.arange(1, v.shape[0] + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full (linear) convolution via real FFT."""
    m = len(a) + len(b) - 1
    n_fft = next_pow2(m)
    A = np.fft.rfft(a, n=n_fft)
    B = np.fft.rfft(b, n=n_fft)
    C = A * B
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


def compute_score(a: np.ndarray) -> float:
    """Exact objective used by `evaluate_sequence` (assumes ∑a = 1)."""
    a = np.asarray(a, dtype=float)
    n = a.shape[0]
    conv = convolution_fft(a, a)
    return 2.0 * n * np.max(conv)


# ----------------------------------------------------------------------
# Parametric families
# ----------------------------------------------------------------------


def arcsine_initial(n: int) -> np.ndarray:
    """Arcsine‑shaped density a_i ∝ 1/√((i+½)(n‑½‑i))."""
    i = np.arange(n, dtype=float)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - 0.5 - i))
    return vals / np.sum(vals)


def beta_sequence(n: int, alpha: float, beta: float | None = None) -> np.ndarray:
    """Beta‑type a_i ∝ (i+½)^α·(n‑½‑i)^β   (β defaults to α)."""
    if beta is None:
        beta = alpha
    i = np.arange(n, dtype=float)
    left = np.power(i + 0.5, alpha)
    right = np.power(n - 0.5 - i, beta)
    seq = left * right
    # keep numbers in a safe range before normalisation
    seq = np.clip(seq, 0.0, 1e12)
    total = np.sum(seq)
    if total == 0.0:
        seq = np.full(n, 1.0 / n)
    else:
        seq = seq / total
    return seq


def exponential_sequence(n: int, lam: float) -> np.ndarray:
    """Exponential decay a_i ∝ exp(-λ·|i‑(n‑1)/2|)."""
    i = np.arange(n, dtype=float)
    centre = (n - 1) / 2.0
    seq = np.exp(-lam * np.abs(i - centre))
    return seq / np.sum(seq)


def random_dirichlet(n: int, concentration: float) -> np.ndarray:
    """Draw a point from Dirichlet(α,…,α) with α = concentration."""
    g = np.random.gamma(concentration, 1.0, size=n)
    return g / g.sum()


def mix_vectors(vectors: list[np.ndarray], weights: np.ndarray) -> np.ndarray:
    """Convex combination of vectors (weights must sum to 1)."""
    out = np.zeros_like(vectors[0])
    for w, v in zip(weights, vectors):
        out += w * v
    out = np.maximum(out, 0.0)
    return out / np.sum(out)


# ----------------------------------------------------------------------
# Smoothed‑max (log‑sum‑exp) gradient and Adam optimiser
# ----------------------------------------------------------------------


def smoothed_gradient(a: np.ndarray, alpha: float):
    """
    Gradient of
        fα(a) = (1/α)·log Σ_k exp(α·(a∗a)_k)
    Returns gradient (same shape as a) and the raw convolution a∗a.
    """
    n = a.shape[0]
    conv = convolution_fft(a, a)                     # length 2n‑1
    maxc = np.max(conv)
    # stabilised softmax
    exp_term = np.exp(alpha * (conv - maxc))
    w = exp_term / np.sum(exp_term)                  # ∑ w_k = 1

    # compute ∂/∂a of Σ w_k (a∗a)_k  = 2· (a reverse‑convolved with w)
    a_rev = a[::-1]
    conv_rev = convolution_fft(a_rev, w)             # length 3n‑2
    start = n - 1
    rev_slice = conv_rev[start:start + n]            # needed slice

    # the diagonal term (k even) appears once instead of twice
    w_diag = w[::2][:n]                              #   w0,w2,w4,… (first n of them)
    grad = 2.0 * rev_slice - w_diag * a
    return grad, conv


def adam_optimize(
    init_a: np.ndarray,
    time_deadline: float,
    max_iter: int = 60_000,
    lr: float = 0.5,
    alpha_start: float = 3.0,
    alpha_end: float = 300.0,
):
    """
    Projected Adam on the smoothed‑max surrogate.
    Returns the best vector encountered and its true (unsmoothed) score.
    """
    a = project_to_simplex(init_a.astype(float).copy())
    best_a = a.copy()
    best_score = compute_score(best_a)
    n = a.shape[0]

    # Adam state
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8

    for t in range(1, max_iter + 1):
        if time.time() >= time_deadline:
            break

        # schedule for the smoothing parameter α
        alpha = alpha_start + (alpha_end - alpha_start) * (t / max_iter)

        grad, _ = smoothed_gradient(a, alpha)

        # Adam update (projected at the end of the step)
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)

        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)

        step = lr / (np.sqrt(v_hat) + eps)
        a_new = a - step * m_hat
        a_new = project_to_simplex(a_new)

        cur_score = compute_score(a_new)
        if cur_score < best_score:
            best_score = cur_score
            best_a = a_new.copy()

        a = a_new

        if t % 5000 == 0:
            print(f"    Adam iter {t:,d}  curScore={cur_score:.6f}  best={best_score:.6f}")

    return best_a, best_score


# ----------------------------------------------------------------------
# Targeted hill‑climbing (mass transfer guided by the worst shift)
# ----------------------------------------------------------------------


def targeted_hill_climb(
    seq,
    time_deadline: float,
    max_steps: int = 30_000,
    init_step: float = 0.08,
):
    """
    Deterministic hill‑climb: repeatedly locate the shift k where the
    convolution is maximal, move a fraction of the mass from the most
    offending position to the least offending one, and keep any
    improvement.
    """
    a = np.array(seq, dtype=float)
    a = project_to_simplex(a)
    best_a = a.copy()
    best_score = compute_score(best_a)
    n = len(a)
    step = init_step

    while max_steps > 0 and time.time() < time_deadline:
        conv = convolution_fft(a, a)
        k = int(np.argmax(conv))

        # contribution of each index to this worst shift
        contrib = np.zeros(n)
        lo = max(0, k - (n - 1))
        hi = min(n - 1, k)
        # i runs over indices that actually appear in the sum for shift k
        for i in range(lo, hi + 1):
            j = k - i
            contrib[i] = a[i] * a[j]

        # pick source (largest contributor) and destination (smallest)
        i_src = int(np.argmax(contrib))
        i_dst = int(np.argmin(contrib))

        if i_src == i_dst:
            break

        delta = step * a[i_src]
        if delta <= 0.0:
            break

        cand = a.copy()
        cand[i_src] -= delta
        cand[i_dst] += delta
        cand = project_to_simplex(cand)

        cur_score = compute_score(cand)
        if cur_score < best_score:
            best_score = cur_score
            best_a = cand
            a = cand
            step = min(step * 1.1, 0.5)   # accelerate when we succeed
        else:
            # reject – keep current a, shrink the step
            step *= 0.9
            a = a  # unchanged

        max_steps -= 1

    return best_a.tolist(), best_score


# ----------------------------------------------------------------------
# Upsampling via FFT‑based resampling (keeps the spectral shape)
# ----------------------------------------------------------------------


def fft_upsample(seq: np.ndarray, new_n: int) -> np.ndarray:
    """Resample a probability vector to a larger size using sinc‑interpolation."""
    # `resample` works on the signal itself; after resizing we renormalise.
    up = resample(seq, new_n)
    up = np.maximum(up, 0.0)               # numerical noise can produce tiny negatives
    return up / np.sum(up)

def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a step‑function that yields a low `evaluate_sequence` value.
    The routine respects the time budget and combines a parametric sweep,
    stochastic refinement of the Beta exponents, Adam optimisation on the
    smooth‑max surrogate and a final targeted hill‑climb.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 2.0          # tiny safety margin

    best_seq = None
    best_val = float("inf")

    # ------------------------------------------------------------------
    # 0) optional fallback – any pre‑computed sequence that may already be good
    # ------------------------------------------------------------------
    try:
        fallback = [float(x) for x in height_sequence_1]   # type: ignore
        fallback_val = evaluate_sequence(fallback)
        print(f"Fallback score = {fallback_val:.6f}")
        if fallback_val < best_val:
            best_val = fallback_val
            best_seq = fallback
    except Exception:
        pass

    # ------------------------------------------------------------------
    # 1) broad parametric sweep (short, cheap)
    # ------------------------------------------------------------------
    lengths = [512, 1024, 2048]
    for n in lengths:
        if time.time() >= deadline - 5:
            break
        print(f"\n=== Parametric sweep length = {n} ===")

        # a) Uniform and arcsine
        candidates = [
            np.full(n, 1.0 / n),
            arcsine_initial(n),
        ]

        # b) Beta exponents – coarse grid, then keep the best few
        alphas = np.linspace(-0.95, 2.5, 70)
        for a in alphas:
            candidates.append(beta_sequence(n, a))
        for a in alphas:
            for b in alphas:
                candidates.append(beta_sequence(n, a, b))

        # c) Exponential decays
        for lam in np.linspace(0.001, 0.02, 12):
            candidates.append(exponential_sequence(n, lam))

        # d) Random Dirichlet draws with various concentrations
        for conc in [0.02, 0.1, 0.5, 2.0, 5.0]:
            for _ in range(8):
                candidates.append(random_dirichlet(n, conc))

        # e) Random convex mixtures of the three basic shapes
        basic = [arcsine_initial(n), np.full(n, 1.0 / n), exponential_sequence(n, 0.008)]
        for _ in range(15):
            w = np.random.rand(len(basic))
            w /= w.sum()
            candidates.append(mix_vectors(basic, w))

        # evaluate all candidates and keep the top 5
        scored = []
        for seq in candidates:
            score = compute_score(seq)
            if score < best_val:
                best_val = score
                best_seq = seq.tolist()
                print(f"New overall best score = {best_val:.6f} (len={n})")
            scored.append((score, seq))
        scored.sort(key=lambda x: x[0])
        top_seqs = [seq for _, seq in scored[:5]]

        # ------------------------------------------------------------------
        # 2) Stochastic refinement of the Beta exponents for the best hit
        # ------------------------------------------------------------------
        # Try to improve the single best Beta pair found above by a tiny random walk.
        if time.time() < deadline - 10:
            for seq in top_seqs:
                # guess the underlying α,β by fitting a log‑linear model (cheap)
                # we simply reuse the same sequence as starting point for a short Adam run.
                refined, refined_score = adam_optimize(
                    init_a=seq,
                    time_deadline=deadline,
                    max_iter=12_000,
                    lr=0.6,
                    alpha_start=4.0,
                    alpha_end=250.0,
                )
                if refined_score < best_val:
                    best_val = refined_score
                    best_seq = refined.tolist()
                    print(f"  Adam‑refined candidate improved to {best_val:.6f}")

    # ------------------------------------------------------------------
    # 3) Optional up‑sampling (FFT‑based) – only if we still have time.
    # ------------------------------------------------------------------
    if best_seq is not None and time.time() < deadline - 15:
        up_n = 4096
        print(f"\nUpsampling to n = {up_n} using sinc‑interpolation")
        up_seq = fft_upsample(np.array(best_seq, dtype=float), up_n)
        up_score = compute_score(up_seq)
        print(f"Upsampled score = {up_score:.6f}")
        if up_score < best_val:
            best_val = up_score
            best_seq = up_seq.tolist()

    # ------------------------------------------------------------------
    # 4) Adam optimisation on the (possibly up‑sampled) candidate
    # ------------------------------------------------------------------
    if best_seq is not None and time.time() < deadline - 8:
        print("\nRunning Adam optimisation on the current best vector")
        init = np.array(best_seq, dtype=float)
        refined, refined_score = adam_optimize(
            init_a=init,
            time_deadline=deadline,
            max_iter=45_000,
            lr=0.5,
            alpha_start=5.0,
            alpha_end=300.0,
        )
        print(f"Adam best score = {refined_score:.6f}")
        if refined_score < best_val:
            best_val = refined_score
            best_seq = refined.tolist()

    # ------------------------------------------------------------------
    # 5) Final deterministic targeted hill‑climb
    # ------------------------------------------------------------------
    if best_seq is not None and time.time() < deadline - 2:
        print("\nFinal targeted hill‑climb")
        final_seq, final_score = targeted_hill_climb(
            best_seq,

    # ------------------------------------------------------------------
    # Fallback – should never be needed
    # ------------------------------------------------------------------
    if best_seq is None:

        best_val = evaluate_sequence(best_seq)

    final_score = evaluate_sequence(best_seq)
    print(f"\nSearch finished. Final reported score = {final_score:.6f}")
    return best_seq
