"""
Propose a low‑autocorrelation step‑function.

The algorithm consists of three phases:

1. **Coarse parametric sweep** –  For a few candidate lengths we
   generate families of symmetric Beta‑like and power‑law shapes
   (parameterised by an exponent and a smoothing ε).  The fast internal
   score (2·n·max(conv)/sum²) is computed for each shape and the best
   few are retained.

2. **Hard‑max exponentiated‑gradient refinement** –  Starting from every
   retained candidate we run a mirror‑descent style optimiser.
   The gradient is taken only from the *active* convolution indices
   (those attaining the current maximum).  This is the exact sub‑gradient
   of the nonsmooth objective and often leads to a stronger decrease
   than the soft‑max surrogate used previously.  The update is

        a ← a * exp(‑η·g) ,   a ← a / sum(a)

   with a step‑size η∝1/√t and a tiny exploration noise added occasionally.

3. **Light simulated‑annealing polish** –  After the gradient phase a
   short annealing on the log‑variables is performed to escape possible
   shallow local minima.  The temperature decays exponentially.

All work is done on the probability simplex (sum = 1, a_i ≥ 0), which
makes the objective scale‑invariant; the final list is returned to the
competition harness.  The routine respects the supplied time budget
(e.g. 1000 s) and always falls back to a trivial uniform sequence if
nothing else is found.
"""

import time
import numpy as np
from typing import List

# ----------------------------------------------------------------------
# Low‑level numeric utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Linear convolution via real FFT (full length, padded to next power of two)."""
    m = a.size + b.size - 1
    n_fft = next_pow2(m)
    A = np.fft.rfft(a, n=n_fft)
    B = np.fft.rfft(b, n=n_fft)
    C = A * B
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


def compute_score(a: np.ndarray) -> float:
    """
    Fast internal score: 2·n·max(a*a) / (sum(a)²).
    The vector a is assumed non‑negative but need not be normalised.
    """
    n = a.size
    conv = convolution_fft(a, a)
    max_val = float(np.max(conv))
    s = float(np.sum(a))
    if s < 1e-12:
        return float("inf")
    return 2.0 * n * max_val / (s * s)


def normalize(v: np.ndarray) -> np.ndarray:
    """Project a non‑negative vector onto the probability simplex."""
    total = v.sum()
    if total <= 0.0:
        return np.full_like(v, 1.0 / v.size)
    return v / total


def arcsine_initial(n: int) -> np.ndarray:
    """Classic arcsine density (discrete, normalised)."""
    i = np.arange(n, dtype=float)
    eps = 0.25
    vals = 1.0 / np.sqrt((i + 0.5) * (n - 0.5 - i) + eps)
    return vals / vals.sum()


def generate_shape(n: int, p: float, eps: float) -> np.ndarray:
    """Symmetric power‑law shape a_i ∝ ((i+ε)(n‑1‑i+ε))⁻ᵖ."""
    i = np.arange(n, dtype=float)
    base = ((i + eps) * (n - 1 - i + eps)) ** (-p)
    total = base.sum()
    if total == 0.0:
        return np.full(n, 1.0 / n)
    return base / total


def beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    """Beta‑like symmetric shape a_i ∝ (i+½)^{α‑1}·(n‑1‑i+½)^{β‑1}."""
    i = np.arange(n, dtype=float)
    base = (i + 0.5) ** (alpha - 1) * (n - 1 - i + 0.5) ** (beta - 1)
    total = base.sum()
    if total == 0.0:
        return np.full(n, 1.0 / n)
    return base / total


def softmax_weights(conv: np.ndarray, lam: float) -> np.ndarray:
    """Stable soft‑max over the convolution entries."""
    x = lam * conv
    mx = np.max(x)
    w = np.exp(x - mx)
    w /= w.sum()
    return w


def gradient_from_weights(a: np.ndarray, w: np.ndarray) -> np.ndarray:
    """
    Gradient of Σ_k w_k·(a*a)_k w.r.t a.
    Implements   g_j = 2· Σ_m a_m·w_{m+j}.
    Computed via a single FFT convolution.
    """
    # w is defined on the full convolution domain (length 2·n‑1)
    w_rev = w[::-1]
    conv = convolution_fft(a, w_rev)               # length = 3·n‑2
    # central index that aligns with shift j:
    idx = 2 * len(a) - 2
    grads = 2.0 * conv[idx - np.arange(len(a))]
    return grads


# ----------------------------------------------------------------------
# Optimisation primitives
# ----------------------------------------------------------------------


def hardmax_optimize(
    init_a: np.ndarray,
    deadline: float,
    *,
    eta0: float = 0.8,
    max_iter: int = 200_000,
    noise_every: int = 800,
    noise_scale: float = 5e-6,
) -> np.ndarray:
    """
    Mirror‑descent (exponentiated‑gradient) on the true max‑convolution.
    The gradient is computed only from the indices attaining the current
    maximum (hard max = sub‑gradient of the nonsmooth objective).
    """
    a = normalize(init_a.astype(float).copy())
    best_a = a.copy()
    best_sc = compute_score(a)
    n = a.size

    for it in range(1, max_iter + 1):
        if time.time() >= deadline:
            break

        # current convolution and its maximum
        conv = convolution_fft(a, a)
        max_val = conv.max()

        # build a weight vector that puts equal mass on all active indices
        mask = conv >= max_val - 1e-14   # tolerance for numerical ties
        w = np.zeros_like(conv)
        w[mask] = 1.0 / np.count_nonzero(mask)

        # sub‑gradient w.r.t. a
        grad = gradient_from_weights(a, w)

        # exponentiated‑gradient step
        eta = eta0 / np.sqrt(it)
        a = a * np.exp(-eta * grad)

        # occasional tiny random perturbation
        if noise_every and (it % noise_every == 0):
            a += np.random.randn(n) * noise_scale
            a = np.maximum(a, 0.0)

        # re‑project onto simplex
        a = normalize(a)

        cur_sc = compute_score(a)
        if cur_sc < best_sc:
            best_sc = cur_sc
            best_a = a.copy()

        # (optional) progress report – printed sparsely to keep overhead low
        if it % 5000 == 0:
            print(f"    hardmax iter {it:,d}  cur={cur_sc:.6f}  best={best_sc:.6f}")

    return best_a


def anneal_refine(
    init_a: np.ndarray,
    deadline: float,
    *,
    init_temp: float = 0.1,
    decay: float = 0.9995,
    max_iter: int = 100_000,
) -> np.ndarray:
    """
    Very light simulated‑annealing on log‑variables.
    Each proposal multiplies a by exp(noise) and renormalises.
    Acceptance follows the Metropolis rule.
    """
    a = normalize(init_a.astype(float).copy())
    best_a = a.copy()
    best_sc = compute_score(a)
    n = a.size
    temp = init_temp

    for it in range(1, max_iter + 1):
        if time.time() >= deadline:
            break
        # propose a multiplicative Gaussian perturbation
        noise = np.random.randn(n) * temp
        cand = a * np.exp(noise)
        cand = normalize(cand)

        sc = compute_score(cand)
        if sc < best_sc:
            best_sc = sc
            best_a = cand.copy()

        # Metropolis acceptance
        if sc < compute_score(a) or np.random.rand() < np.exp(-(sc - compute_score(a)) / max(temp, 1e-12)):
            a = cand

        # cool down
        temp *= decay

        if it % 20000 == 0:
            print(f"    anneal step {it:,d}  temp={temp:.5f}  best={best_sc:.6f}")

    return best_a

        candidates.append(arcsine_initial(n))

        # symmetric Beta families (α = β)
        for alpha in np.linspace(0.25, 1.25, 9):
            candidates.append(beta_shape(n, alpha, alpha))

        # power‑law family (p, ε)
        for p in np.linspace(0.3, 0.9, 7):
            for eps in (0.001, 0.01, 0.05, 0.2):
                candidates.append(generate_shape(n, p, eps))

        # few random Dirichlet draws
        for _ in range(4):
            samp = rng.exponential(1.0, size=n)
            candidates.append(samp / samp.sum())

        # --------------------------------------------------------------
        # 1b.  Quick internal scoring – keep the three most promising.
        # --------------------------------------------------------------
        scored = [(compute_score(c), c) for c in candidates]
        scored.sort(key=lambda x: x[0])
        top_candidates = [c for _, c in scored[:3]]

        # --------------------------------------------------------------
        # 1c.  Refine each retained candidate.
        # --------------------------------------------------------------
        for idx, cand in enumerate(top_candidates):
            if time.time() >= deadline - 5.0:
                break
            print(f"  candidate {idx + 1}/{len(top_candidates)} (n={n})")

            # allocate a modest iteration budget (≈ 10 it/s for n≈2000)
            it_budget = int(min(30_000, max(2_000, (deadline - time.time()) * 10)))

            refined = hardmax_optimize(
                init_a=cand,
                deadline=deadline,
                eta0=0.7,
                max_iter=it_budget,
                noise_every=1000,
                noise_scale=1e-5,
            )

            # short annealing polish
            refined = anneal_refine(
                refined,
                deadline,
                init_temp=0.08,
                decay=0.9997,
                max_iter=int(min(15_000, (deadline - time.time()) * 8)),
            )

            # final official evaluation
            val = evaluate_sequence(refined.tolist())
            print(f"    evaluated score = {val:.6f}")

            if val < best_val:
                best_val = val
                best_seq = refined.tolist()
                print(f"    *** New global best: {best_val:.6f} (n={n}) ***")

    # ------------------------------------------------------------------
    # 2.  Final polishing of the overall best (if time permits).
    # ------------------------------------------------------------------
    if best_seq and time.time() < deadline - 5.0:
        a = np.array(best_seq, dtype=float)
        print("\nFinal polishing of the global best")
        a = hardmax_optimize(
            a,
            deadline,
            eta0=0.5,
            max_iter=int(min(80_000, (deadline - time.time()) * 12)),
            noise_every=800,
            noise_scale=5e-6,
        )
        a = anneal_refine(
            a,
            deadline,
            init_temp=0.05,
            decay=0.9998,
            max_iter=int(min(30_000, (deadline - time.time()) * 8)),
        )
        final_val = evaluate_sequence(a.tolist())
        print(f"Final polished score = {final_val:.6f}")
        if final_val < best_val:
            best_val = final_val
            best_seq = a.tolist()
            print("*** Updated global best after final polish ***")

    # ------------------------------------------------------------------
    # 3.  Guard – always return a feasible sequence.
    # ------------------------------------------------------------------
    if not best_seq:
        best_seq = [1.0] * 256
        best_val = evaluate_sequence(best_seq)

    print(f"\nSearch finished. Best reported score = {best_val:.6f}")
    return best_seq
