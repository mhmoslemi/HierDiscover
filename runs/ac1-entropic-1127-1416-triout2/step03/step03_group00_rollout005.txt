"""
Propose a step‑height sequence that minimises
    evaluate_sequence(a) = 2·n·max(conv(a,a)) / (sum(a)^2)

The algorithm works in three phases:

1. **Coarse random search** – sample many Beta(α,β) densities (skewed
   shapes) and keep the one with the smallest exact score.

2. **Smooth‑max Adam optimisation** – minimise a smooth approximation of
   the max (soft‑max with a temperature τ) while staying on the simplex
   (non‑negative, sum = 1).  The gradient of the soft‑max of the
   convolution is obtained by a single FFT‑based correlation, and an
   Adam update with Euclidean projection ensures fast, stable convergence.

3. **Short simulated‑annealing refinement** – local mass‑transfer moves
   (preserving the sum) are accepted with a Metropolis rule.  This
   explores the neighbourhood of the Adam optimum and can escape tiny
   plateaux.

All phases respect the 1000 s budget, keep the best observed sequence,
and return a list of floats ready for `evaluate_sequence`.
"""

import time
import math
import numpy as np


# ----------------------------------------------------------------------
# Helpers
# ----------------------------------------------------------------------


def simplex_projection(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the probability simplex:
        { w | w_i >= 0 , sum_i w_i = 1 }.
    Implementation of Duchi et al. (2008).
    """
    v = np.asarray(v, dtype=float)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all entries non‑positive → uniform distribution
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1] - 1
    theta = cssv[cond][-1] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def beta_sequence(alpha: float, beta: float, n: int) -> np.ndarray:
    """
    Discretised Beta(α,β) density on n points, normalised to sum 1.
    Mid‑point rule avoids the singular endpoints.
    """
    x = (np.arange(n) + 0.5) / n
    logw = (alpha - 1.0) * np.log(x) + (beta - 1.0) * np.log(1.0 - x)
    logw = np.clip(logw, -700.0, 700.0)          # avoid overflow
    w = np.exp(logw)
    if not np.isfinite(w.sum()) or w.sum() == 0.0:
        w = np.ones_like(w)
    return w / w.sum()


def next_pow2(x: int) -> int:
    """Small utility – next power of two >= x."""
    return 1 << (x - 1).bit_length()

        best_score = 2.0 * n * conv.max()
        best_a = a0.copy()
        print(f"Initial existing score = {best_score:.6f}")

    # a quick sweep over beta parameters
    for _ in range(600):

            best_a = cand.copy()

    # --------------------------------------------------------------

    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    step_scale = 0.05                     # base step size (tuned empirically)

    tau = 0.5
    tau_mul = 1.001
    max_tau = 5000.0

    fft_len = next_pow2(3 * n)

    it = 0
    # reserve ~30 s for the final SA phase

        a_fft = np.fft.rfft(a, n=fft_len)

        weights = np.exp(exp_arg)
        weights /= weights.sum()            # soft‑max probabilities

        # ---- gradient of the smooth max (correlation) ----
        w_fft = np.fft.rfft(weights, n=fft_len)
        # correlation = irfft(a_fft * conj(w_fft))

        # ---- Adam update with Euclidean projection onto the simplex ----
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1 ** (it + 1))
        v_hat = v / (1.0 - beta2 ** (it + 1))
        step = step_scale * m_hat / (np.sqrt(v_hat) + eps)

        # ---- temperature schedule ----
        tau = min(max_tau, tau * tau_mul)
        it += 1

    print(f"Adam optimisation finished after {it} iterations; best score = {best_score:.6f}")

    # --------------------------------------------------------------
    # 3. Simulated‑annealing fine‑tuning (short, aggressive)
    # --------------------------------------------------------------
    cur_a = best_a.copy()
    a_fft = np.fft.rfft(cur_a, n=fft_len)
    conv_full = np.fft.irfft(a_fft * a_fft, n=fft_len)[: 2 * n - 1]
    cur_score = 2.0 * n * conv_full.max()

    # temperature schedule – start with a modest fraction of the current score
    T_start = cur_score * 0.02
    T_end = cur_score * 1e-5
    it_sa = 0

    while time.time() < deadline:
        # exponential cooling
        T = max(T_end, T_start * (0.9995 ** it_sa))

        # ---- propose a simple mass‑transfer move (keeps sum = 1) ----
        src = np.random.choice(n, p=cur_a)               # biased towards larger masses
        tgt = np.random.randint(n - 1)
        if tgt >= src:
            tgt += 1
        if cur_a[src] <= 0.0:

        cand[src] -= delta
        cand[tgt] += delta

        cand_fft = np.fft.rfft(cand, n=fft_len)

        cand_score = 2.0 * n * conv_cand.max()

        # ---- Metropolis acceptance ----
        if cand_score < cur_score or np.random.rand() < math.exp(-(cand_score - cur_score) / max(T, 1e-12)):

            cur_score = cand_score
            if cand_score < best_score:

    # 4. Return the best sequence (already normalised to sum 1)
    # --------------------------------------------------------------
    best_a = best_a / best_a.sum()
    return [float(v) for v in best_a.tolist()]
