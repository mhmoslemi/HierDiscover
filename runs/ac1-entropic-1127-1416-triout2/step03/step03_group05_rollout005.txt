        fraction = (deadline - time.time()) / (deadline - t_start)  # remaining fraction

"""
Propose a step‑function sequence that (hopefully) beats the current record
(~1.503) for the objective

    2·n·max(conv(a,a)) / (sum(a)**2)

The algorithm works in two phases:

1. **Adam on the simplex** – the coefficients are represented by soft‑max
   logits, guaranteeing Σaᵢ = 1.  A sub‑gradient of the maximal
   convolution entry is obtained analytically (the same routine used in the
   reference solution).  Several short Adam restarts with different
   initialisations explore the landscape.

2. **Simulated annealing refinement** – starting from the best Adam result,
   random “mass‑transfer” moves (move a small amount of weight from one
   index to another) are performed.  The move is accepted if it improves the
   score or with a Metropolis probability that slowly cools to zero.  This
   phase can escape local minima that Adam may miss.

Both phases respect the time budget (default 1000 s) and return a plain
Python list of non‑negative floats.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Convolution helpers
# ----------------------------------------------------------------------


def _fft_len(n: int) -> int:
    """Length of the zero‑padded FFT (next power of two)."""
    return 1 << ((2 * n - 2).bit_length())


def _maxconv_and_subgrad(a: np.ndarray, fft_len: int):
    """
    Compute the aperiodic convolution a*a, its maximal value, and a
    sub‑gradient of that maximal value w.r.t. a.
    Returns (max_val, grad) where grad has the same shape as a.
    """
    n = a.shape[0]
    # FFT‑based convolution (only the first 2·n‑1 entries are needed)
    fft = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(fft * fft, n=fft_len)[: 2 * n - 1]

    max_val = conv.max()
    # tolerance for numerically tied maxima
    tol = max_val * 1e-12 + 1e-14
    active = np.where(conv >= max_val - tol)[0]

    grad = np.zeros_like(a)
    for k in active:
        low = max(0, k - n + 1)
        high = min(n - 1, k)
        idx = np.arange(low, high + 1, dtype=int)
        # contribution of the partner indices (i + partner = k)
        grad[idx] += 2.0 * a[k - idx]

    if active.size > 1:
        grad /= float(active.size)          # any convex combination is ok

    return max_val, grad


def _max_conv_scalar(a: np.ndarray) -> float:
    """Return only the maximal convolution value for a (sum is assumed 1)."""
    n = a.shape[0]
    L = _fft_len(n)
    fft = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(fft * fft, n=L)[: 2 * n - 1]
    return float(conv.max())


def _score_from_a(a: np.ndarray) -> float:
    """
    Evaluate the objective 2·n·max(conv) / sum(a)².
    The sum is kept (normally =1) but the formula is used for completeness.
    """
    sum_a = a.sum()
    if sum_a < 1e-12:
        return float("inf")
    max_conv = _max_conv_scalar(a)
    n = a.shape[0]
    return 2.0 * n * max_conv / (sum_a ** 2)


# ----------------------------------------------------------------------
# Initialisation helpers
# ----------------------------------------------------------------------


def _initial_sequence(kind: str, n: int, rng: np.random.Generator,
                      prev_seq: np.ndarray | None) -> np.ndarray:
    """
    Produce a feasible probability vector of length n.
    ``kind`` can be 'prev', 'uniform', 'random_dirichlet' or 'geometric'.
    """
    if kind == "prev" and prev_seq is not None:
        seq = prev_seq.copy()
        if seq.shape[0] != n:
            if seq.shape[0] > n:
                seq = seq[:n]
            else:
                seq = np.concatenate([seq,
                                      np.zeros(n - seq.shape[0], dtype=seq.dtype)])
        seq = np.maximum(seq, 0.0)
        total = seq.sum()
        if total == 0.0:
            return np.full(n, 1.0 / n)
        return seq / total

    if kind == "uniform":
        return np.full(n, 1.0 / n)

    if kind == "random_dirichlet":
        alpha = 0.5                     # sparse Dirichlet
        sample = rng.gamma(alpha, 1.0, size=n)
        return sample / sample.sum()

    if kind == "geometric":
        r = 0.98 + 0.02 * rng.random()
        left = r ** np.arange(n)
        right = r ** np.arange(n - 1, -1, -1)
        seq = left + right
        return seq / seq.sum()

    # Fallback
    return np.full(n, 1.0 / n)


# ----------------------------------------------------------------------
# Adam optimiser (soft‑max parametrisation)
# ----------------------------------------------------------------------


def _adam_optimize(init_a: np.ndarray,
                   deadline: float,
                   restart_id: int,
                   target: float = 1.503) -> tuple[np.ndarray, float]:
    """
    Adam optimisation on the logits of a probability vector.
    Returns the best (a, score) observed before ``deadline``.
    """
    n = init_a.shape[0]
    fft_len = _fft_len(n)

    # initialise logits (soft‑max)
    eps_init = 1e-12
    b = np.log(init_a + eps_init)

    # Adam state
    m = np.zeros_like(b)
    v = np.zeros_like(b)
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8

    # evaluate start point
    max_val, _ = _maxconv_and_subgrad(init_a, fft_len)
    best_score = 2.0 * n * max_val          # sum(init_a)==1
    best_a = init_a.copy()

    stagnation = 0
    it = 0

    while time.time() < deadline:
        it += 1

        # ---- simplex projection via soft‑max ----
        b_max = np.max(b)
        a = np.exp(b - b_max)               # shift for stability
        a = a / a.sum()

        # ---- objective & sub‑gradient ----
        max_val, grad_a = _maxconv_and_subgrad(a, fft_len)
        cur_score = 2.0 * n * max_val

        # ---- keep the best ----
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            stagnation = 0
            print(f"Restart {restart_id:02d} iter {it:6d}: best = {best_score:.6f}")
        else:
            stagnation += 1

        # ---- early exit ----
        if best_score < target:
            break

        # ---- jitter to escape plateaus ----
        if stagnation > 8000:
            b += 0.01 * np.random.randn(n)
            stagnation = 0

        # ---- Adam update (logit space) ----
        grad_obj = 2.0 * n * grad_a          # d (2·n·max)/da
        avg = np.dot(a, grad_obj)            # Jacobian of soft‑max
        grad_b = a * (grad_obj - avg)        # chain rule

        m = beta1 * m + (1.0 - beta1) * grad_b
        v = beta2 * v + (1.0 - beta2) * (grad_b ** 2)

        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        lr = 0.4 / np.sqrt(it)               # slowly decaying step size
        b -= lr * m_hat / (np.sqrt(v_hat) + eps)

    return best_a, best_score


# ----------------------------------------------------------------------
# Simulated‑annealing refinement
# ----------------------------------------------------------------------


def _anneal_optimize(start_a: np.ndarray,
                     deadline: float) -> tuple[np.ndarray, float]:
    """
    Simple Metropolis‑type annealing that moves a small amount of mass
    between two randomly chosen indices.  The temperature decays linearly
    to zero at ``deadline``.
    """
    a = start_a.copy()
    n = a.shape[0]
    cur_score = _score_from_a(a)
    best_a = a.copy()
    best_score = cur_score

    rng = np.random.default_rng()
    start_time = time.time()
    it = 0

    # initial temperature proportional to the current score
    T0 = cur_score * 0.02 + 1e-6

    while time.time() < deadline:
        it += 1
        # linear cooling
        remaining = max(deadline - time.time(), 0.0)
        fraction = remaining / (deadline - start_time + 1e-12)
        T = T0 * fraction

        # pick a source index with non‑zero weight
        support = np.where(a > 1e-12)[0]
        if support.size == 0:
            break
        i = rng.choice(support)
        j = rng.integers(0, n)
        if i == j:
            continue

        # amount to transfer (up to 50 % of the source mass)
        max_move = a[i] * 0.5
        if max_move < 1e-14:
            continue
        epsilon = max_move * rng.random()

        a_new = a.copy()
        a_new[i] -= epsilon
        a_new[j] += epsilon

        new_score = _score_from_a(a_new)

        # Metropolis acceptance
        if new_score < cur_score or rng.random() < np.exp(-(new_score - cur_score) / max(T, 1e-12)):
            a = a_new
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_a = a.copy()
                print(f"Anneal iter {it:7d}: new best = {best_score:.6f}")

        # occasional aggressive jitter if stuck
        if it % 20000 == 0 and cur_score - best_score > 1e-4:
            # sprinkle a tiny random perturbation
            jitter = 0.001 * rng.standard_normal(n)
            a = np.maximum(a + jitter, 0.0)
            a /= a.sum()
            cur_score = _score_from_a(a)

    return best_a, best_score
