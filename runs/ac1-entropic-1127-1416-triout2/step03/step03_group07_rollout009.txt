"""
Propose a step‑function (non‑negative list of heights) that minimises

        2·n·max(conv(a,a)) / (sum(a)**2)

where `conv` is the full (zero‑padded) discrete convolution.
The method consists of two stages:

1. **Mirror‑descent** (the same exponentiated‑gradient scheme that was
   used previously) with several random restarts.  It quickly drives the
   objective close to the known best value.
2. **Simulated‑annealing refinement** starting from the best mirror‑descent
   candidate.  Small “mass‑transfer’’ moves on the probability simplex are
   accepted according to a temperature schedule, allowing the search to
   escape local plateaus and to improve the bound further.

Both stages respect the simplex constraint `sum(a)=1` (the objective is
scale‑invariant) and keep all entries non‑negative.  The whole search runs
under the time budget given by `budget_s` (default 1000 s) and returns the
best sequence found.
"""

import time
import math
import numpy as np


def _project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of a vector onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1] - 1
    theta = cssv[cond][-1] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def _max_and_subgradient(a: np.ndarray) -> tuple[float, np.ndarray]:
    """
    Return the current maximum entry of the autocorrelation `a*a`
    and a sub‑gradient of that entry (averaged if several indices tie).

    The gradient formula follows the reference implementation:
        ∂/∂a_i  (a*a)[k] = 2·a_{k-i}
    for all i that take part in the active maximal indices `k`.
    """
    n = a.shape[0]
    # FFT based convolution (full length 2*n-1)
    fft = np.fft.rfft(a, n=2 * n - 1)
    conv = np.fft.irfft(fft * fft, n=2 * n - 1)
    max_val = float(conv.max())

    # Indices that are (numerically) tied for the maximum
    tol = max_val * 1e-12 + 1e-14
    active = np.where(conv >= max_val - tol)[0]

    grad = np.zeros_like(a)
    for k in active:
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        idx = np.arange(lo, hi + 1, dtype=int)
        partner = k - idx
        grad[idx] += 2.0 * a[partner]

    if active.size > 1:
        grad /= float(active.size)
    return max_val, grad


def _max_conv(a: np.ndarray) -> float:
    """
    Return only the maximum entry of the autocorrelation `a*a`.
    Used inside the simulated‑annealing loop for speed.
    """
    n = a.shape[0]
    fft = np.fft.rfft(a, n=2 * n - 1)
    conv = np.fft.irfft(fft * fft, n=2 * n - 1)
    return float(conv.max())


def _mirror_descent(a_init: np.ndarray,
                    n: int,
                    deadline: float,
                    lower_bound: float = 1.0,
                    max_step: float = 5.0) -> tuple[np.ndarray, float]:
    """
    Exponentiated‑gradient (mirror‑descent) optimiser.
    Returns the best vector seen during the run and its objective value.
    """
    a = a_init.copy()
    a = np.maximum(a, 1e-30)      # avoid exact zeros
    a = a / a.sum()               # simplex projection
    g_cum = np.zeros_like(a)

    best_a = a.copy()
    max_conv, _ = _max_and_subgradient(a)
    best_score = 2.0 * n * max_conv

    it = 0
    rng = np.random.default_rng()
    while time.time() < deadline:
        max_conv, grad = _max_and_subgradient(a)
        cur_score = 2.0 * n * max_conv

        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time():.1f}] MD iter {it}: new best {best_score:.6f}")

        norm_sq = float(np.dot(grad, grad))
        if norm_sq == 0.0:
            break

        step = (cur_score - lower_bound) / (norm_sq + 1e-12)
        step = max(1e-12, min(step, max_step))

        g_cum += step * grad
        tmp = -g_cum
        tmp -= np.max(tmp)                      # stabilise the exponentials
        a = np.exp(tmp)
        a = a / a.sum()

        # occasional stochastic jitter to escape flat regions
        if it % 5000 == 0 and it > 0:
            eps = 0.001 * rng.standard_normal(n)
            a = np.maximum(a + eps, 0.0)
            a = a / a.sum()

        it += 1
    return best_a, best_score


def _simulated_annealing(a_start: np.ndarray,
                         n: int,
                         deadline: float,
                         cur_score: float) -> tuple[np.ndarray, float]:
    """
    Simple simulated‑annealing local search.
    Moves a small amount of mass from one entry to another while staying
    on the simplex.  Accepts uphill moves with a temperature that decays
    geometrically.
    """
    best_a = a_start.copy()
    best_score = cur_score
    cur_a = a_start.copy()
    cur_score = cur_score

    rng = np.random.default_rng()
    # initial temperature – 10 % of the current objective (empirical)
    T = max(cur_score * 0.1, 1e-6)
    decay = 0.99995
    it = 0

    while time.time() < deadline:
        # propose a mass transfer i -> j
        i, j = rng.integers(0, n, size=2)
        while i == j:
            j = rng.integers(0, n)

        max_delta = cur_a[i]
        if max_delta <= 0.0:
            continue

        # transfer up to 50 % of the available mass at i
        delta = rng.random() * max_delta * 0.5
        a_new = cur_a.copy()
        a_new[i] -= delta
        a_new[j] += delta

        new_score = 2.0 * n * _max_conv(a_new)

        if new_score < best_score:
            best_score = new_score
            best_a = a_new.copy()
            print(f"[{time.time():.1f}] SA iter {it}: new best {best_score:.6f}")

        # Metropolis acceptance
        if new_score < cur_score or rng.random() < math.exp(-(new_score - cur_score) / T):
            cur_a = a_new
            cur_score = new_score

        # inject a tiny random jitter every 1000 steps
        if it % 1000 == 0 and it > 0:
            noise = rng.normal(scale=0.001, size=n)
            a_noise = np.maximum(cur_a + noise, 0.0)
            a_noise = a_noise / a_noise.sum()
            cur_a = a_noise
            cur_score = 2.0 * n * _max_conv(cur_a)

        T = max(T * decay, 1e-12)   # keep a tiny temperature to avoid div‑by‑0
        it += 1

    return best_a, best_score


def propose_candidate(budget_s: int = 1000) -> list[float]:
    """
    Search for a step‑function that minimises the evaluation metric.
    Uses mirror descent with several restarts followed by simulated annealing.
    The function respects the 1000 s time budget (minus a small safety margin)
    and returns the best sequence it discovered.
    """
    # ------------------------------------------------------------------
    # 1. Determine problem size and an optional starting point.
    # ------------------------------------------------------------------
    deadline = time.time() + budget_s - 5.0   # give a few seconds guard

    try:
        # `height_sequence_1` is a global that may contain a historic best.
        init_seq = np.asarray(height_sequence_1, dtype=float)  # type: ignore[name-defined]
        n = init_seq.shape[0]
        init_seq = np.clip(init_seq, 0.0, 1000.0)
        if init_seq.sum() > 0.0:
            init_seq = init_seq / init_seq.sum()
        else:
            init_seq = None
    except Exception:
        init_seq = None
        n = 2048                     # default length when nothing is known

    best_a = None
    best_score = float("inf")

    # ------------------------------------------------------------------
    # 2. Mirror‑descent phase – a few independent restarts.
    # ------------------------------------------------------------------
    # We allocate roughly the first half of the time budget to this phase.
    md_deadline = time.time() + (deadline - time.time()) * 0.45
    restart = 0
    while time.time() < md_deadline and restart < 8:
        if init_seq is not None and restart == 0:
            a0 = init_seq.copy()
        else:
            rng = np.random.default_rng(int(time.time() * 1e6) % (2**32 - 1))
            a0 = rng.random(n)
            a0 = a0 / a0.sum()

        # give each restart a slice of the remaining MD time
        sub_deadline = time.time() + (md_deadline - time.time()) / (8 - restart)
        a_opt, score_opt = _mirror_descent(a0, n, sub_deadline)
        if score_opt < best_score:
            best_score = score_opt
            best_a = a_opt.copy()
        restart += 1

    # ------------------------------------------------------------------
    # 3. Simulated‑annealing refinement on the best candidate.
    # ------------------------------------------------------------------
    if best_a is None:
        # fall back to a uniform distribution (objective = 2)
        best_a = np.full(n, 1.0 / n)
        best_score = 2.0 * n * _max_conv(best_a)

    # Remaining time goes to the annealer.
    if time.time() < deadline:
        a_opt2, score_opt2 = _simulated_annealing(best_a, n, deadline, best_score)
        if score_opt2 < best_score:
            best_score = score_opt2
            best_a = a_opt2.copy()

    # ------------------------------------------------------------------
    # 4. Return the sequence in plain Python list form.
    # ------------------------------------------------------------------
    # Clip to the limits that `evaluate_sequence` enforces (0 … 1000)
    final_seq = [float(max(0.0, min(1000.0, v))) for v in best_a.tolist()]

    print(f"[{time.time():.1f}] Returning best score {best_score:.6f} "
          f"with length {len(final_seq)}")
    return final_seq
