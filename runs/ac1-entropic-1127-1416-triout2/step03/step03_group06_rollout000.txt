"""
Propose a candidate step function that (hopefully) beats the previous
best score ~1.521639.

Algorithm
---------
* Initialise the search with a collection of simple analytic shapes
  (ramp, uniform, exponential, polynomial, Beta‑distribution) and keep
  the best one as a starting point.
* Optimise the shape on the probability simplex using **Adam** on the
  *log‑probability* (i.e. a soft‑max parametrisation).  This yields a
  KL‑projection that respects non‑negativity without an explicit Euclidean
  projection.
* The gradient of the objective is the sub‑gradient of the maximal entry
  of the autocorrelation `a*a`.  For an active index `k` the sub‑gradient
  w.r.t. `a_i` equals `2*a_{k-i}`; we average over all active indices.
* Adam’s first and second moment estimates are updated each iteration;
  the learning‑rate decays slowly.  The current vector is recovered from
  the soft‑max of the logits `z` and normalised to sum = 1, so the
  evaluation function `evaluate_sequence` sees the same shape (any global
  scaling would be irrelevant).
* If no improvement is seen for a long stretch we inject a small random
  perturbation (a Dirichlet draw mixed in) and reset the optimiser to
  escape plateaus.
* The loop runs until the wall‑clock budget (`budget_s` = 1000 s by default)
  expires or the target score 1.503 is reached.  The best vector seen is
  returned as a plain Python list.

The method is deliberately lightweight (FFT‑based convolution each
iteration) and fits comfortably into the 1000 s time limit while often
finding a lower bound than the earlier projected sub‑gradient descent.
"""

import time
import numpy as np

def simplex_projection(v):
    """
    Euclidean projection of a vector onto the probability simplex:
    { w | w_i ≥ 0 , Σ w_i = 1 }.
    (Not used in the main optimiser, but kept for completeness.)
    """
    v = np.asarray(v, dtype=float)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1] - 1
    theta = cssv[cond][-1] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def compute_objective(a: np.ndarray) -> float:
    """
    Return the value of the objective 2·n·max(a∗a) assuming Σ a_i = 1.
    This matches the value returned by ``evaluate_sequence`` after the
    internal normalisation step.
    """
    n = a.shape[0]
    fft = np.fft.rfft(a, n=2 * n - 1)
    conv = np.fft.irfft(fft * fft, n=2 * n - 1)
    return 2.0 * n * conv.max()


def beta_sequence(alpha: float, beta: float, n: int) -> np.ndarray:
    """Discretised Beta(α,β) density on `n` mid‑points, normalised to sum = 1."""
    x = (np.arange(n) + 0.5) / n
    # work in log‑space to avoid overflow for extreme parameters
    logw = (alpha - 1.0) * np.log(x) + (beta - 1.0) * np.log(1.0 - x)
    logw = np.clip(logw, -700.0, 700.0)
    w = np.exp(logw)
    tot = w.sum()
    if tot == 0 or not np.isfinite(tot):
        w = np.ones_like(w)
        tot = w.sum()
    return w / tot


def exponential_shape(lam: float, n: int) -> np.ndarray:
    """a_i ∝ exp(‑lam·i) (lam ≥ 0), normalised to sum = 1."""
    i = np.arange(n, dtype=float)
    w = np.exp(-lam * i)
    w /= w.sum()
    return w


def polynomial_shape(p: float, n: int) -> np.ndarray:
    """a_i ∝ 1 - (i/(n‑1))**p (p ≥ 0), normalised to sum = 1."""
    i = np.arange(n, dtype=float)
    frac = i / (n - 1)
    w = 1.0 - np.power(frac, p)
    w = np.clip(w, 0.0, None)
    w /= w.sum()
    return w


def subgradient_active(a: np.ndarray, active: np.ndarray) -> np.ndarray:
    """
    Compute the averaged sub‑gradient of the maximal convolution entry
    over the supplied ``active`` index set.
    For each active index k we add 2·a_{k‑i} to the gradient component i.
    Finally we divide by the number of active indices.
    """
    n = a.shape[0]
    g = np.zeros_like(a)
    for k in active:
        low = max(0, k - n + 1)
        high = min(n - 1, k)
        idx = np.arange(low, high + 1, dtype=int)
        partner = k - idx
        g[idx] += 2.0 * a[partner]
    if active.size > 0:
        g /= float(active.size)
    return g

        g_a = subgradient_active(a, active)

        # ----- convert to gradient w.r.t. logits z (soft‑max Jacobian) -----
        dot = np.dot(a, g_a)          # scalar = ⟨a, ∂L/∂a⟩
        grad_z = a * (g_a - dot)      # ∇_z L = diag(a)·(∇_a L - ⟨a,∇_a L⟩·1)

        # ----- Adam update -----
        m = beta1 * m + (1.0 - beta1) * grad_z
        v = beta2 * v + (1.0 - beta2) * (grad_z ** 2)

        m_hat = m / (1.0 - beta1 ** (iteration + 1))
        v_hat = v / (1.0 - beta2 ** (iteration + 1))

        lr = lr0 / (1.0 + 0.0004 * iteration)   # gentle decay
        z -= lr * m_hat / (np.sqrt(v_hat) + eps_adam)

        # ----- occasional re‑centering to keep logits numerically stable -----
        if (iteration + 1) % 1000 == 0:
            z -= np.mean(z)

        # ----- stagnation handling -----
        if stagnation >= max_stagnation:
            # Blend the current iterate with a fresh Dirichlet draw
            rand = np.random.gamma(1.0, 1.0, size=n)
            rand /= rand.sum()
            blend = 0.02
            a_pert = (1.0 - blend) * a + blend * rand
            # Reset logits to correspond to the perturbed point
            z = np.log(a_pert + eps_log)
            m.fill(0.0)
            v.fill(0.0)
            stagnation = 0
            print(f"Iter {iteration:6d}: perturbation applied, current {cur_score:.6f}")

        iteration += 1

        # ----- early exit if the target is reached -----
        if best_score <= 1.503:
            print(f"Target reached at iteration {iteration}, score = {best_score:.6f}")
            break

    print(f"Finished after {iteration} iterations, best score = {best_score:.6f}")

    # --------------------------------------------------------------
    # 5. Return the best sequence as a plain Python list
    # --------------------------------------------------------------
    # Ensure the output complies with the evaluator's expectations:
    # non‑negative numbers, reasonable magnitude (the optimisation already
    # works with a normalised vector, scaling does not change the score).
    return [float(v) for v in best_seq.tolist()]
