def propose_candidate(budget_s=1000):
    deadline = time.time() + budget_s - 0.5
    # hyperparameters
    lower_bound = 1.0 # theoretical

    best_score = float('inf')
    best_seq = None

    # Determine a set of possible lengths; ensure not too big
    lengths = [2048, 4096, 6144, 8192]  # maybe include previous best length
    # If height_sequence_1 exist, get its length for possible use
    try:
        init_seq = np.asarray(height_sequence_1, dtype=float)
        lengths.append(len(init_seq))
    except Exception:
        init_seq = None

    restart = 0
    while time.time() < deadline:
        # choose length
        n = int(np.random.choice(lengths))
        # initialize a
        if init_seq is not None and restart % 5 == 0:
            # adapt init_seq to current n: either crop or pad
            if len(init_seq) > n:
                idx = np.linspace(0, len(init_seq)-1, n, dtype=int)
                a = init_seq[idx].copy()
            else:
                pad_len = n - len(init_seq)
                a = np.concatenate([init_seq, np.full(pad_len, 1e-12)])
        else:
            a = np.random.rand(n)
        a = np.maximum(a, 1e-12)
        a = a / a.sum()
        # Adam parameters
        m = np.zeros_like(a)
        v = np.zeros_like(a)
        beta1, beta2 = 0.9, 0.999
        eps = 1e-8
        lr = 0.02  # base step size
        # schedule for beta (smoothness)
        beta_start, beta_end = 5.0, 5000.0
        # Precompute FFT size
        M = 1 << ((3*n-1).bit_length())
        # a_rev placeholder
        it = 0
        while time.time() < deadline:
            # compute convolution of a with itself (full)
            fa = np.fft.rfft(a, M)
            conv_full = np.fft.irfft(fa * fa, M)[:2*n-1]

            # smoothing parameter
            t = it / 5000.0  # scale
            beta = beta_start * (beta_end / beta_start) ** min(t, 1.0)  # exponential schedule up to 5000 it
            # compute log-sum-exp and softmax
            b = beta * conv_full
            b_max = b.max()
            L = b_max + np.log(np.exp(b - b_max).sum())
            p = np.exp(b - L)  # shape 2n-1, sum(p)=1

            # compute w = correlation(p, a)
            a_rev = a[::-1]
            fp = np.fft.rfft(p, M)
            fa_rev = np.fft.rfft(a_rev, M)
            corr_full = np.fft.irfft(fp * fa_rev, M)
            w = corr_full[n-1 : n-1 + n]

            # gradient of smoothed objective
            grad = 4.0 * n * (beta * w - L)

            # Adam update
            it += 1
            m = beta1 * m + (1 - beta1) * grad
            v = beta2 * v + (1 - beta2) * (grad ** 2)
            m_hat = m / (1 - beta1 ** it)
            v_hat = v / (1 - beta2 ** it)
            step = lr * m_hat / (np.sqrt(v_hat) + eps)

            a = a - step
            # ensure non negativity before projection
            a = np.maximum(a, 0.0)
            a = _project_to_simplex(a)

            # evaluate exact objective
            max_conv = conv_full.max()
            cur_score = 2.0 * n * max_conv  # sum a=1

            if cur_score < best_score:
                best_score = cur_score
                best_seq = a.copy()
                print(f"[{time.time():.2f}] restart {restart}, it {it}: new best {best_score:.6f}")

                if best_score < 1.503:
                    print("Target reached")
                    return best_seq.tolist()

            # occasional jitter

                eps_noise = np.random.normal(scale=1e-4, size=n)
                a = np.maximum(a + eps_noise, 0.0)
                a = _project_to_simplex(a)

        restart += 1
    if best_seq is None:
        best_seq = np.full(n, 1.0 / n)
    return best_seq.tolist()

"""
Projected Adam‑gradient descent on a smoothed max objective.

The true objective is
        F(a) = 2·n·max(a*a) / (sum(a)^2)      (sum(a)=1 ⇒ F=2·n·max(a*a))

We minimise a smooth surrogate:
        L(a) = log‑sum‑exp(β·(a*a))               (β large ⇒ L≈β·max)
and optimise  G(a)=2·n·L(a)  (the denominator is constant because we keep
the vector on the probability simplex).

∇G = 4·n·β·w   where   w_i = Σ_k p_k·a_{k-i},
p = softmax(β·(a*a))   and the correlation  w = p ⋆ a
is computed by FFT (O(n log n)).  Adam is used for an adaptive step size,
and after every update the vector is projected back onto the simplex.
A continuation schedule slowly raises β, first giving a very smooth landscape
and then focusing on the true max.  Several restarts and occasional jitter
help escape local minima.  The best sequence seen is returned.

The routine runs for at most ``budget_s`` seconds and prints progress.
"""

import time
import numpy as np
from math import log, exp


def _project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of a vector onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1] - 1
    theta = cssv[cond][-1] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def _next_pow2(x: int) -> int:
    """Return the smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _logsumexp(x: np.ndarray) -> float:
    """Stable log‑sum‑exp."""
    m = np.max(x)
    return float(m + np.log(np.exp(x - m).sum()))
