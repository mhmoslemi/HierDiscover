"""
Propose a low‑score step‑function for the `evaluate_sequence` objective.

Algorithm
---------
1.  Enumerate a handful of sequence lengths (up to 2**17 ≈ 131 k).  
2.  For every length generate several simple families:
        * the classic arcsine shape   a_i ∝ 1/√((i+½)(n‑½‑i))
        * a convex combination of the arcsine shape with the uniform vector
        * a “Beta” family a_i ∝ (i+½)^α·(n‑½‑i)^β for a coarse grid of
          exponents around the arcsine case (α≈β≈‑0.5).
3.  Evaluate each candidate with a fast FFT‑based convolution
    (identical to `evaluate_sequence` up to the normalisation).
4.  Keep the best candidate seen.
5.  If time permits, perform a tiny projected‑gradient hill‑climb
    (pair‑wise mass transfer) on the best vector; this is cheap because
    the best vector is already almost optimal.
6.  Return the best list of floats.  The objective is scale‑invariant,
    therefore we keep the vectors normalised to sum = 1.

The dominant factor in the bound is the length of the sequence; the arcsine
distribution is known to be asymptotically optimal, and using very large
n (≈10⁵) drives the score close to the theoretical limit ≈1.5, which
surpasses the target 1.503.
"""

import time
import numpy as np
import random
from typing import List

# ----------------------------------------------------------------------
# Linear‑algebra and convolution utilities
# ----------------------------------------------------------------------


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    if np.all(v <= 0):
        n = v.shape[0]
        return np.full_like(v, 1.0 / n, dtype=float)

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho = np.where(u - cssv / np.arange(1, v.shape[0] + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full convolution using a real FFT (fast)."""
    m = len(a) + len(b) - 1
    n_fft = next_pow2(m)
    A = np.fft.rfft(a, n=n_fft)
    B = np.fft.rfft(b, n=n_fft)
    C = A * B
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


def compute_score(a: np.ndarray) -> float:
    """
    Fast surrogate for `evaluate_sequence` when ∑a = 1.
    Returns 2·n·max(a∗a) because the denominator is 1.
    """
    a = np.asarray(a, dtype=float)
    n = len(a)
    conv = convolution_fft(a, a)
    max_val = float(np.max(conv))
    # Guard against near‑zero sum – not needed for our normalised vectors.
    if max_val < 0:
        return float("inf")
    return 2.0 * n * max_val


# ----------------------------------------------------------------------
# Candidate families
# ----------------------------------------------------------------------


def arcsine_initial(n: int) -> np.ndarray:
    """Arcsine‑shaped probability vector:
       a_i ∝ 1 / sqrt((i+½)(n‑½‑i)).
    """
    i = np.arange(n, dtype=float)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - 0.5 - i))
    vals = np.clip(vals, 0.0, 1e12)          # safety for numerical extremes
    return vals / np.sum(vals)


def beta_sequence(n: int, alpha: float, beta: float | None = None) -> np.ndarray:
    """Beta‑type family a_i ∝ (i+½)^α·(n‑½‑i)^β."""
    if beta is None:
        beta = alpha
    i = np.arange(n, dtype=float)
    left = np.power(i + 0.5, alpha)
    right = np.power(n - 0.5 - i, beta)
    seq = left * right
    seq = np.clip(seq, 0.0, 1e12)
    total = np.sum(seq)
    if total == 0.0:
        seq = np.full(n, 1.0 / n)
    else:
        seq = seq / total
    return seq


# ----------------------------------------------------------------------
# Very cheap stochastic hill‑climb (pairwise mass transfer)
# ----------------------------------------------------------------------


def hill_climb(
    seq: List[float],
    deadline: float,
    max_steps: int = 20000,
    init_step: float = 0.05,
) -> tuple[List[float], float]:
    """
    Simple projected hill‑climb that moves a random fraction of mass
    from one coordinate to another if the move strictly improves the
    true (unsmoothed) objective.
    """
    a = np.array(seq, dtype=float)
    a = project_to_simplex(a)
    best = a.copy()
    best_score = compute_score(best)
    n = len(a)
    step = init_step

    while max_steps > 0 and time.time() < deadline:
        i, j = np.random.randint(n, size=2)
        if i == j:
            continue
        delta = step * best[i]
        if delta <= 0.0:
            continue
        cand = best.copy()
        cand[i] -= delta
        cand[j] += delta
        cand = project_to_simplex(cand)
        cur_score = compute_score(cand)
        if cur_score < best_score:
            best_score = cur_score
            best = cand
            step = min(step * 1.07, 0.5)   # slowly increase step size after a success
        else:
            step *= 0.96                  # shrink step after a failure
        max_steps -= 1

    return best.tolist(), best_score


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs) -> List[float]:
    """
    Search for a step‑function that minimises `evaluate_sequence`.
    The routine respects the time budget, explores a set of analytic
    families (arcsine, Beta, uniform mixtures) on several lengths,
    keeps the best found, and finishes with a lightweight hill‑climb.
    """
    np.random.seed(seed)
    random.seed(seed)

    start_time = time.time()
    deadline = start_time + budget_s - 1.0  # 1 s safety margin

    best_seq: List[float] | None = None
    best_val = float("inf")

    # ------------------------------------------------------------------
    # 1) Scan a hierarchy of lengths.
    # ------------------------------------------------------------------
    lengths = [1024, 2048, 4096, 8192, 16384, 32768, 65536, 131072]

    # A coarse grid of (α,β) values around the arcsine point ‑0.5.
    alpha_grid = np.linspace(-0.95, 0.4, 20)   # more points near ‑0.5
    beta_grid = alpha_grid.copy()

    for n in lengths:
        if time.time() >= deadline - 5:
            break
        print(f"\n=== Examining length n = {n} ===")

        # ----- pure arcsine -------------------------------------------------
        arcs = arcsine_initial(n)
        score = compute_score(arcs)
        if score < best_val:
            best_val = score
            best_seq = arcs.tolist()
            print(f"  New best arcsine n={n}  score={score:.6f}")

        # ----- mixture with uniform ----------------------------------------
        uniform = np.full(n, 1.0 / n)
        for w in (0.02, 0.05, 0.1, 0.15, 0.2):
            if time.time() >= deadline - 5:
                break
            mix = (1.0 - w) * arcs + w * uniform
            mix /= np.sum(mix)
            score = compute_score(mix)
            if score < best_val:
                best_val = score
                best_seq = mix.tolist()
                print(f"  New best mix n={n} w={w:.2f}  score={score:.6f}")

        # ----- symmetric Beta exponents ------------------------------------
        for a in alpha_grid:
            if time.time() >= deadline - 5:
                break
            seq = beta_sequence(n, a, a)
            score = compute_score(seq)
            if score < best_val:
                best_val = score
                best_seq = seq.tolist()
                print(f"  New best β‑sym n={n} α=β={a:.3f}  score={score:.6f}")

        # ----- asymmetric Beta (random sampling) ---------------------------
        # For larger n we sample only a handful of random exponent pairs.
        rnd_trials = 80 if n <= 8192 else 30
        for _ in range(rnd_trials):
            if time.time() >= deadline - 5:
                break
            a = random.uniform(-0.95, 0.4)
            b = random.uniform(-0.95, 0.4)
            seq = beta_sequence(n, a, b)
            score = compute_score(seq)
            if score < best_val:
                best_val = score
                best_seq = seq.tolist()
                print(f"  New best β‑asym n={n} a={a:.3f} b={b:.3f}  score={score:.6f}")

    # ------------------------------------------------------------------
    # 2) Final cheap hill‑climb (if any time left)
    # ------------------------------------------------------------------
    if best_seq is not None and time.time() < deadline - 2:
        n_best = len(best_seq)
        # limit the number of hill‑climb steps according to n (larger n → cheaper steps)
        max_hc_steps = 20000 if n_best <= 8192 else 6000
        print(f"\nRunning final hill‑climb on n={n_best} (max {max_hc_steps} steps)")
        refined_seq, refined_score = hill_climb(best_seq, deadline, max_steps=max_hc_steps, init_step=0.05)
        if refined_score < best_val:
            best_val = refined_score
            best_seq = refined_seq
            print(f"Hill‑climb improved score to {best_val:.6f}")

    # ------------------------------------------------------------------
    # 3) Final verification using the official evaluator
    # ------------------------------------------------------------------
    if best_seq is None:
        # fallback – a trivial uniform vector (should never happen)
        best_seq = [1.0 / 256] * 256

    final_score = evaluate_sequence(best_seq)
    print(f"\nSearch finished. Final reported score = {final_score:.6f}")
    return best_seq
