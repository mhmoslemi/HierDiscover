"""
Hybrid optimisation for the step‑function minimisation problem.

The search proceeds in three phases:

1. **Broad initial scan** – for several lengths (powers of two) a handful of
   parametric families (uniform, arcsine, symmetric power‑law, beta‑like,
   Dirichlet) are sampled.  The fast internal score (identical to
   ``evaluate_sequence`` apart from clipping) is used to keep only the best
   few candidates.

2. **Two‑stage refinement**
   * *Soft‑max phase* – a smooth approximation of the max‑convolution
     (soft‑max with a temperature ramp) is minimised by exponentiated‑gradient
     descent on the probability simplex.
   * *Hard‑max phase* – the true max‑convolution is targeted directly.
     An additive projected gradient descent (sub‑gradient of the maximal
     convolution entry) is applied with a diminishing step size.

   Small Gaussian noise is injected occasionally to escape shallow plateaux.

3. **Final polishing** – the globally best vector is refined once more with
   the hard‑max optimiser using a longer schedule.

All work respects the provided time budget; if time runs out a feasible
fallback is returned.  The function ``propose_candidate`` is the entry point
required by the competition harness.
"""

import time
import numpy as np
from typing import List, Tuple

# ----------------------------------------------------------------------
# Low‑level numeric utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Linear convolution via real FFT (full length, zero‑padded to power‑of‑2)."""
    m = len(a) + len(b) - 1
    n_fft = next_pow2(m)
    A = np.fft.rfft(a, n=n_fft)
    B = np.fft.rfft(b, n=n_fft)
    C = A * B
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


def compute_score(a: np.ndarray) -> float:
    """
    Official objective (scale‑invariant):
        2·n·max(a*a) / (sum(a)²)
    """
    n = len(a)
    conv = convolution_fft(a, a)
    max_val = float(np.max(conv))
    s = float(np.sum(a))
    if s < 1e-12:
        return float("inf")
    return 2.0 * n * max_val / (s * s)


def softmax_weights(conv: np.ndarray, lam: float) -> np.ndarray:
    """Numerically stable soft‑max over the convolution entries."""
    x = lam * conv
    mx = np.max(x)
    w = np.exp(x - mx)
    w /= w.sum()
    return w


def gradient_from_weights(a: np.ndarray, w: np.ndarray) -> np.ndarray:
    """
    Gradient of Σ_k w_k·(a*a)_k w.r.t. a.
    Implements   g_j = 2· Σ_m a_m · w_{m+j}.
    Computed via one FFT‑based convolution.
    """
    w_rev = w[::-1]
    conv = convolution_fft(a, w_rev)               # length = 3·n‑2
    idx = 2 * len(a) - 2                            # position 2n‑2
    grads = 2.0 * conv[idx - np.arange(len(a))]
    return grads


def normalize(v: np.ndarray) -> np.ndarray:
    """Project a non‑negative vector onto the probability simplex."""
    total = v.sum()
    if total <= 0.0:
        return np.full_like(v, 1.0 / len(v))
    return v / total


# ----------------------------------------------------------------------
# Shape generators (parametric families)
# ----------------------------------------------------------------------


def arcsine_initial(n: int) -> np.ndarray:
    """Discrete arcsine density (normalised)."""
    i = np.arange(n, dtype=float)
    eps = 0.25
    vals = 1.0 / np.sqrt((i + 0.5) * (n - 0.5 - i) + eps)
    return vals / vals.sum()


def generate_powerlaw_shape(n: int, p: float, eps: float) -> np.ndarray:
    """Symmetric power‑law shape a_i ∝ ((i+ε)(n‑1‑i+ε))⁻ᵖ."""
    i = np.arange(n, dtype=float)
    base = ((i + eps) * (n - 1 - i + eps)) ** (-p)
    total = base.sum()
    if total == 0.0:
        return np.full(n, 1.0 / n)
    return base / total


def generate_beta_like(n: int, alpha: float, beta: float) -> np.ndarray:
    """Beta‑like symmetric shape."""
    i = np.arange(n, dtype=float)
    base = (i + 0.5) ** (alpha - 1) * (n - 1 - i + 0.5) ** (beta - 1)
    total = base.sum()
    if total <= 0.0:
        return np.full(n, 1.0 / n)
    return base / total

# Optimisers

def optimise_softmax(
    initial_a: np.ndarray,
    deadline: float,
    *,
    lam_start: float = 5.0,
    lam_end: float = 200.0,
    lam_ramp_iters: int = 6000,
    eta0: float = 0.7,
    max_iter: int = 200_000,
    noise_every: int = 2000,
    noise_scale: float = 1e-5,
) -> Tuple[np.ndarray, float]:
    """
    Exponentiated‑gradient descent on a smooth soft‑max approximation
    of the maximal self‑convolution.
    Returns the best vector seen and its (non‑smooth) score.
    """
    a = normalize(initial_a.astype(float).copy())
    best_a = a.copy()
    best_sc = compute_score(a)

    n = len(a)

    for it in range(1, max_iter + 1):
        if time.time() >= deadline:
            break

        # temperature schedule
        lam = lam_start + (lam_end - lam_start) * min(1.0, it / lam_ramp_iters)

        # convolution and soft‑max weights
        conv = convolution_fft(a, a)               # length 2n‑1
        w = softmax_weights(conv, lam)

        # gradient w.r.t. a
        grad = gradient_from_weights(a, w)

        # step size – 1/√t schedule
        eta = eta0 / np.sqrt(it)

        # multiplicative update (guarantees non‑negativity)
        a = a * np.exp(-eta * grad)

        # occasional Gaussian noise
        if noise_every and (it % noise_every == 0):
            a += np.random.randn(n) * noise_scale
            a = np.maximum(a, 0.0)

        # project back onto the simplex
        a = normalize(a)

        cur_sc = compute_score(a)
        if cur_sc < best_sc:
            best_sc = cur_sc
            best_a = a.copy()

        if it % 4000 == 0:
            print(f"      softmax iter={it:,d} λ={lam:.1f} cur={cur_sc:.6f} best={best_sc:.6f}")

    return best_a, best_sc


def optimise_hard_max(
    initial_a: np.ndarray,
    deadline: float,
    *,
    eta0: float = 0.3,
    max_iter: int = 200_000,
    noise_every: int = 5000,
    noise_scale: float = 5e-6,
) -> Tuple[np.ndarray, float]:
    """
    Projected additive gradient descent on the true maximal convolution entry.
    At each step the sub‑gradient of the (unique) maximal entry is used.
    """
    a = normalize(initial_a.astype(float).copy())
    best_a = a.copy()
    best_sc = compute_score(a)

    n = len(a)

    for it in range(1, max_iter + 1):
        if time.time() >= deadline:
            break

        # full convolution and its maximal index
        conv = convolution_fft(a, a)               # length 2n‑1
        max_idx = int(np.argmax(conv))

        # sub‑gradient for the maximal entry (delta weight)
        w = np.zeros_like(conv)
        w[max_idx] = 1.0
        grad = gradient_from_weights(a, w)          # shape (n,)

        # diminishing additive step
        eta = eta0 / np.sqrt(it)
        a = a - eta * grad
        a = np.maximum(a, 0.0)

        # optional exploration noise
        if noise_every and (it % noise_every == 0):
            a += np.random.randn(n) * noise_scale
            a = np.maximum(a, 0.0)

        # re‑project onto the simplex
        a = normalize(a)

        cur_sc = compute_score(a)
        if cur_sc < best_sc:
            best_sc = cur_sc
            best_a = a.copy()

        if it % 4000 == 0:
            print(f"      hard‑max iter={it:,d} cur={cur_sc:.6f} best={best_sc:.6f}")

    return best_a, best_sc

def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs) -> List[float]:
    """
    Search for a non‑negative step‑function with a low ``evaluate_sequence`` value.
    The routine respects the time budget, explores several lengths,
    refines promising candidates with a two‑stage optimiser and returns the
    best sequence found.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 2.0   # safety margin

    best_seq: List[float] = []
    best_val: float = float("inf")

    # ------------------------------------------------------------------
    # 0. Optional fallback from a previously discovered construction.
    # ------------------------------------------------------------------
    try:
        fallback = [float(x) for x in height_sequence_1]  # type: ignore
        val = evaluate_sequence(fallback)
        print(f"Fallback (height_sequence_1) score = {val:.6f}")
        if val < best_val:
            best_val = val
            best_seq = fallback
    except Exception:
        pass

    # ------------------------------------------------------------------
    # 1. Lengths to explore (powers of two – cheap FFTs).
    # ------------------------------------------------------------------
    lengths = [1024, 2048, 4096]

    # Add a larger length if enough time is left after the initial phases.
    if deadline - time.time() > 250:
        lengths.append(8192)

    for n in lengths:
        if time.time() >= deadline - 5.0:
            break
        print(f"\n=== Exploring length n = {n} ===")

        # --------------------------------------------------------------
        # 1a. Build a modest library of candidate start vectors.
        # --------------------------------------------------------------
        candidates: List[np.ndarray] = []

        # uniform
        candidates.append(np.full(n, 1.0 / n))

        # classic arcsine
        candidates.append(arcsine_initial(n))

        # symmetric power‑law draws
        for _ in range(6):
            p = rng.uniform(0.15, 0.85)
            eps = rng.uniform(0.001, max(2.0, 0.02 * n))
            candidates.append(generate_powerlaw_shape(n, p, eps))

        # beta‑like shapes (symmetric)
        for _ in range(4):
            alpha = rng.uniform(0.2, 5.0)
            beta = rng.uniform(0.2, 5.0)
            candidates.append(generate_beta_like(n, alpha, beta))

        # Dirichlet / exponential samples
        for _ in range(3):
            samp = rng.exponential(1.0, size=n)
            samp /= samp.sum()
            candidates.append(samp)

        # Small random perturbations of the above candidates
        perturbed: List[np.ndarray] = []
        for c in candidates:
            perturbed.append(c)  # keep original
            for _ in range(2):
                noise = rng.normal(scale=0.01, size=n)
                c2 = np.maximum(c + noise, 0.0)
                if c2.sum() == 0:
                    continue
                c2 = c2 / c2.sum()
                perturbed.append(c2)
        candidates = perturbed

        # --------------------------------------------------------------
        # 1b. Quick sweep using the fast internal score, keep the best few.
        # --------------------------------------------------------------
        scored = [(compute_score(c), c) for c in candidates]
        scored.sort(key=lambda x: x[0])
        top_candidates = [c for _, c in scored[:4]]   # keep the four most promising

        # --------------------------------------------------------------
        # 1c. Refine each retained candidate.
        # --------------------------------------------------------------
        for idx, cand in enumerate(top_candidates):
            if time.time() >= deadline - 5.0:
                break
            print(f"  candidate {idx + 1}/{len(top_candidates)} (n={n})")

            # allocate a budget of iterations based on remaining time
            remaining = max(0.0, deadline - time.time())
            max_iter_soft = int(min(150_000, max(5_000, remaining * 400)))
            refined, _ = optimise_softmax(
                initial_a=cand,
                deadline=deadline,
                lam_start=5.0,
                lam_end=200.0,
                lam_ramp_iters=8000,
                eta0=0.7,
                max_iter=max_iter_soft,
                noise_every=2000,
                noise_scale=1e-5,
            )

            # Follow‑up hard‑max optimisation (focus on the true max)
            remaining = max(0.0, deadline - time.time())
            max_iter_hard = int(min(120_000, max(3_000, remaining * 300)))
            refined_hard, _ = optimise_hard_max(
                initial_a=refined,
                deadline=deadline,
                eta0=0.3,
                max_iter=max_iter_hard,
                noise_every=4000,
                noise_scale=5e-6,
            )

            # Evaluate with the official evaluator
            val = evaluate_sequence(refined_hard.tolist())
            print(f"    evaluated score = {val:.6f}")

            if val < best_val:
                best_val = val
                best_seq = refined_hard.tolist()
                print(f"    *** New global best: {best_val:.6f} (n={n}) ***")

    # ------------------------------------------------------------------
    # 2. Final polishing of the overall best sequence (if time permits).
    # ------------------------------------------------------------------
    if best_seq and time.time() < deadline - 5.0:
        a = np.array(best_seq, dtype=float)
        remaining = max(0.0, deadline - time.time())
        max_iter_final = int(min(250_000, remaining * 300))
        print("\nFinal polishing of the global best")
        a_refined, _ = optimise_hard_max(
            initial_a=a,
            deadline=deadline,
            eta0=0.25,
            max_iter=max_iter_final,
            noise_every=5000,
            noise_scale=2e-6,
        )
        final_val = evaluate_sequence(a_refined.tolist())
        print(f"Final polished score = {final_val:.6f}")
        if final_val < best_val:
            best_val = final_val
            best_seq = a_refined.tolist()
            print("*** Updated global best after final polish ***")

    # ------------------------------------------------------------------
    # 3. Guard – always return a feasible sequence.
    # ------------------------------------------------------------------
    if not best_seq:
        best_seq = [1.0] * 256
        best_val = evaluate_sequence(best_seq)

    print(f"\nSearch finished. Best reported score = {best_val:.6f}")
    return best_seq
