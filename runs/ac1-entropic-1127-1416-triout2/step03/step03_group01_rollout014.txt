def compute_obj(a):    # fast compute of evaluate sequence, assuming sum>0
    s = a.sum()
    conv = convolution_fft(a, a)
    return float(2 * len(a) * conv.max() / (s*s))

def compute_obj_and_subgrad(a):
    n = len(a)
    conv = convolution_fft(a, a)
    M = float(conv.max())
    t = int(np.argmax(conv))
    # compute gradM
    idx = np.arange(n)
    j = t - idx
    mask = (j >= 0) & (j < n)
    gradM = np.zeros_like(a)
    gradM[mask] = 2.0 * a[j[mask]]
    # gradient of F
    C = 2.0 * n
    S = a.sum()
    grad = C * (S * gradM - 2.0 * M) / (S**3)
    # compute objective value
    obj = C * M / (S**2)
    return obj, grad
```

def adam_optimize_obj(initial_a, deadline, max_iter=200000, lr0=0.1):
    a = project_to_simplex(initial_a.astype(float).copy())
    best_a = a.copy()
    best_obj = compute_obj(a)
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    b1,b2 = 0.9, 0.999
    eps = 1e-8
    for it in range(1, max_iter+1):
        if time.time() > deadline:
            break
        obj, grad = compute_obj_and_subgrad(a)
        # Adam update
        m = b1*m + (1-b1)*grad
        v = b2*v + (1-b2)*(grad**2)
        m_hat = m / (1-b1**it)
        v_hat = v / (1-b2**it)
        step = lr0 * m_hat / (np.sqrt(v_hat) + eps)
        a_new = a - step
        # projection
        a_new = project_to_simplex(a_new)
        cur_obj = compute_obj(a_new)
        if cur_obj < best_obj:
            best_obj = cur_obj
            best_a = a_new.copy()
        a = a_new
        if it % 5000 == 0:
            print(f"    AdamIter {it:6d} cur={cur_obj:.6f} best={best_obj:.6f}")
    return best_a, best_obj

def lp_refine(f, max_iter=10, t=0.2, deadline=None):
    n = len(f)
    target_sum = np.sqrt(2.0 * n)
    # ensure f is normalized to target sum
    f = f * (target_sum / f.sum())
    best_f = f.copy()
    best_M = np.max(convolution_fft(f, f))
    for it in range(max_iter):
        if deadline is not None and time.time() > deadline:
            break
        M = best_M
        g0 = solve_lp_max_sum(f, M)
        if g0 is None or g0.sum() == 0:
            break
        # scale to target sum
        factor = target_sum / g0.sum()
        g = g0 * factor
        # mix
        f_new = (1 - t) * f + t * g
        # recompute M
        M_new = np.max(convolution_fft(f_new, f_new))
        if M_new < best_M - 1e-12:
            f = f_new
            best_M = M_new
            best_f = f_new.copy()
        else:
            # try smaller t
            reduction = False
            for tt in [t/2, t/4, t/8]:
                f_new = (1 - tt) * f + tt * g
                M_new = np.max(convolution_fft(f_new, f_new))
                if M_new < best_M:
                    f = f_new
                    best_M = M_new
                    best_f = f_new.copy()
                    reduction = True
                    break
            if not reduction:
                break
    return best_f

def lp_refine(f, max_iter=10, t=0.2, deadline=None):
    n = len(f)
    target_sum = np.sqrt(2.0 * n)
    # ensure normalization
    f = f * (target_sum / f.sum())
    best_f = f.copy()
    best_M = np.max(convolution_fft(f, f))
    for it in range(max_iter):
        if deadline is not None and time.time() > deadline:
            break
        # solve LP
        g0 = solve_lp_max_sum(f, best_M)
        if g0 is None or g0.sum() == 0:
            break
        # scale to target sum
        factor = target_sum / g0.sum()
        g = g0 * factor
        # mix
        f_new = (1 - t) * f + t * g
        M_new = np.max(convolution_fft(f_new, f_new))
        if M_new < best_M - 1e-12:
            f = f_new
            best_f = f_new

                if M_new < best_M:
                    f = f_new
                    best_f = f_new
                    best_M = M_new
                    improved = True
                    break
            if not improved:
                break

"""
Propose a low‑discrepancy step function.

The search combines three ideas:

1. **LP‑based rebalancing** –  Starting from a classic arcsine shape
   (scaled to the standard sum √(2n)), we repeatedly solve a tiny linear
   program:
        max   Σ g_j
        s.t.  (f * g)_k ≤ M   for all k,
              g_j ≥ 0,
   where M = max(f * f).  The solution g is rescaled to the same total
   mass as f and mixed with f.  This operation provably reduces the
   self‑convolution maximum.

2. **Exact gradient descent on the true objective** –  After the LP
   phase we project the vector onto the probability simplex (Σ a = 1)
   and optimise the genuine objective
        F(a) = 2·n·max(a*a) / (Σa)²
   using Adam.  The gradient is obtained analytically; only a single
   FFT‑based convolution is required per iteration.

3. **Final polishing** –  The best candidate found over several
   lengths (256 … 8192) is refined once more with a long Adam run.

All helper functions are top‑level, there are no closures or lambdas,
and the algorithm respects the 1000 s time budget.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Low‑level utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """
    Real‑FFT linear convolution of two 1‑D arrays.
    Returns an array of length len(a) + len(b) - 1.
    """
    m = len(a) + len(b) - 1
    n_fft = next_pow2(m)
    A = np.fft.rfft(a, n=n_fft)
    B = np.fft.rfft(b, n=n_fft)
    C = A * B
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the probability simplex
    {x ≥ 0, Σx = 1}.  Duchi et al., 2008.
    """
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / v.size)

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho = np.where(u - cssv / np.arange(1, v.size + 1) > 0)[0]
    if len(rho) == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def arcsine_initial(n: int) -> np.ndarray:
    """
    Classical arcsine density on {0,…,n‑1}:
        a_i ∝ 1 / √((i+½)*(n‑½‑i)).
    The result is normalised to Σ a_i = 1.
    """
    i = np.arange(n, dtype=float)
    eps = 0.25
    vals = 1.0 / np.sqrt((i + 0.5) * (n - 0.5 - i) + eps)
    vals /= vals.sum()
    return vals


def generate_beta_shape(
    n: int, alpha: float, beta: float, eps: float
) -> np.ndarray:
    """
    Symmetric beta family density:
        a_i ∝ (i+ε)^{α‑1}·(n‑1‑i+ε)^{β‑1}.
    Returned vector is normalised to Σ a_i = 1.
    """
    i = np.arange(n, dtype=float)
    left = (i + eps) ** (alpha - 1.0)
    right = (n - 1 - i + eps) ** (beta - 1.0)
    base = left * right
    total = base.sum()
    if total <= 0.0 or not np.isfinite(total):
        return np.full(n, 1.0 / n)
    return base / total


# ----------------------------------------------------------------------
# Exact objective and its sub‑gradient
# ----------------------------------------------------------------------


def compute_obj_and_subgrad(a: np.ndarray):
    """
    Returns (objective value, gradient) of
        F(a) = 2·n·max(a*a) / (Σa)²
    The gradient is a sub‑gradient w.r.t. the (non‑negative) entries.
    """
    n = a.shape[0]
    conv = convolution_fft(a, a)            # length 2n‑1
    M = float(conv.max())
    t = int(np.argmax(conv))                # index where the max is attained

    # gradient of the max‑convolution term
    idx = np.arange(n)
    j = t - idx
    mask = (j >= 0) & (j < n)
    gradM = np.zeros_like(a)
    gradM[mask] = 2.0 * a[j[mask]]

    S = a.sum()
    C = 2.0 * n
    grad = C * (S * gradM - 2.0 * M) / (S ** 3)
    obj = C * M / (S ** 2)
    return obj, grad


# ----------------------------------------------------------------------
# Adam optimiser on the true objective
# ----------------------------------------------------------------------


def adam_optimize_obj(
    initial_a: np.ndarray,
    time_deadline: float,
    max_iter: int = 200_000,
    lr0: float = 0.1,
) -> tuple[np.ndarray, float]:
    """
    Adam optimiser of the exact objective.
    Returns the best vector (projected onto the simplex) and its
    objective value.
    """
    a = project_to_simplex(initial_a.astype(float).copy())
    best_a = a.copy()
    best_val = compute_obj_and_subgrad(a)[0]

    beta1, beta2 = 0.9, 0.999
    eps = 1e-8
    m = np.zeros_like(a)
    v = np.zeros_like(a)

    for it in range(1, max_iter + 1):
        if time.time() >= time_deadline:
            break

        obj, grad = compute_obj_and_subgrad(a)

        # Adam moments
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)

        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        step = lr0 * m_hat / (np.sqrt(v_hat) + eps)
        a_new = a - step
        a_new = project_to_simplex(a_new)

        cur_obj = compute_obj_and_subgrad(a_new)[0]
        if cur_obj < best_val:
            best_val = cur_obj
            best_a = a_new.copy()

        a = a_new

        if it % 5000 == 0:
            print(f"    Adam iter={it:,d} cur={cur_obj:.6f} best={best_val:.6f}")

    return best_a, best_val


# ----------------------------------------------------------------------
# Linear‑programming rebalancing (the “LP step”)
# ----------------------------------------------------------------------


def build_convolution_matrix(f: np.ndarray) -> np.ndarray:
    """
    Constructs the (2n‑1) × n matrix A such that A @ g = f * g.
    """
    n = f.shape[0]
    m = 2 * n - 1
    A = np.zeros((m, n), dtype=np.float64)
    for i in range(n):
        A[i : i + n, i] = f
    return A


def solve_lp_max_sum(f: np.ndarray, M: float):
    """
    Solve  max Σ g_j   s.t.   (f * g)_k ≤ M   ∀k ,   g ≥ 0.
    Returns the optimal g as a dense NumPy array, or None on failure.
    """
    try:
        import cvxpy as cp
    except Exception:
        return None

    A = build_convolution_matrix(f)
    n = f.shape[0]
    g = cp.Variable(n, nonneg=True)
    constraints = [A @ g <= M]
    prob = cp.Problem(cp.Maximize(cp.sum(g)), constraints)

    try:
        prob.solve(solver=cp.ECOS, warm_start=True, max_iters=2000, verbose=False)
    except Exception:
        try:
            prob.solve(solver=cp.SCS, max_iters=2000, verbose=False)
        except Exception:
            return None
    if g.value is None:
        return None
    return np.asarray(g.value, dtype=np.float64)


def lp_refine(
    f: np.ndarray,
    max_iter: int = 10,
    t: float = 0.2,
    deadline: float | None = None,
) -> np.ndarray:
    """
    Repeatedly apply the LP step to a vector f (sum = sqrt(2n)).
    Returns the improved vector (still with the same sum).
    """
    n = len(f)
    target_sum = np.sqrt(2.0 * n)
    f = f * (target_sum / f.sum())
    best_f = f.copy()
    best_M = np.max(convolution_fft(f, f))

    for it in range(max_iter):
        if deadline is not None and time.time() > deadline:
            break

        g0 = solve_lp_max_sum(f, best_M)
        if g0 is None or g0.sum() == 0.0:
            break

        # rescale g0 to the target total mass
        factor = target_sum / g0.sum()
        g = g0 * factor

        # try mixing with step size t
        f_new = (1.0 - t) * f + t * g
        M_new = np.max(convolution_fft(f_new, f_new))

        if M_new < best_M - 1e-14:
            f = f_new
            best_f = f_new
            best_M = M_new
            continue

        # if not improved, try smaller mixing coefficients
        improved = False
        for tt in (t / 2, t / 4, t / 8, t / 16):
            f_new = (1.0 - tt) * f + tt * g
            M_new = np.max(convolution_fft(f_new, f_new))
            if M_new < best_M - 1e-14:
                f = f_new
                best_f = f_new
                best_M = M_new
                improved = True
                break
        if not improved:
            break

    return best_f

# Main entry point required by the competition harness
# ----------------------------------------------------------------------


def propose_candidate(
    seed: int = 42,
    budget_s: int = 1000,
    **kwargs,
) -> list[float]:
    """
    Search for a non‑negative step sequence with a low
    evaluate_sequence() value.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 2.0  # safety margin

    best_seq: list[float] = []
    best_val: float = float("inf")

    # ------------------------------------------------------------------
    # 1️⃣  Scan several lengths (smaller lengths are cheaper to optimise)
    # ------------------------------------------------------------------
    lengths = [256, 512, 1024, 2048, 4096, 8192]

    for n in lengths:
        if time.time() >= deadline - 5.0:
            break
        print(f"\n=== Optimising length n = {n} ===")
        target_sum = np.sqrt(2.0 * n)

        # ---------- 1a. Initialise ----------
        # primary start: arcsine shape scaled to the standard sum
        f_start = arcsine_initial(n) * target_sum

        # occasionally try a completely random start (still on the same simplex)
        if rng.random() < 0.3:
            rnd = rng.random(n)
            f_start = rnd / rnd.sum() * target_sum

        # ---------- 1b. LP rebalancing ----------
        if n <= 2048:
            f_lp = lp_refine(f_start, max_iter=12, t=0.2, deadline=deadline)
        else:
            f_lp = f_start

        # ---------- 1c. Project to probability simplex ----------
        a = project_to_simplex(f_lp.copy())  # Σ a = 1

        # ---------- 1d. Adam optimisation on the true objective ----------
        remaining = deadline - time.time()
        if remaining <= 0:
            break
        max_iter = int(min(200_000, remaining * 30))
        # heuristic learning rate shrinking with n
        lr0 = 0.2 / np.sqrt(n)
        a_opt, _ = adam_optimize_obj(
            a, time_deadline=deadline, max_iter=max_iter, lr0=lr0
        )
        # ---------- 1e. Evaluate ----------
        try:
            val = evaluate_sequence(a_opt.tolist())
        except Exception:
            val = float("inf")
        print(f"  → length {n} final score = {val:.6f}")

        if val < best_val:
            best_val = val
            best_seq = a_opt.tolist()
            print(f"  *** New global best: {best_val:.6f} (n={n}) ***")

    # ------------------------------------------------------------------
    # 2️⃣  Final polishing of the overall best vector
    # ------------------------------------------------------------------
    if best_seq and time.time() < deadline - 5.0:
        a = np.array(best_seq, dtype=float)
        remaining = deadline - time.time()
        max_iter = int(min(400_000, remaining * 30))
        print("\nFinal polishing of the global best …")
        a_final, _ = adam_optimize_obj(
            a, time_deadline=deadline, max_iter=max_iter, lr0=0.02
        )
        final_val = evaluate_sequence(a_final.tolist())
        print(f"Final polished score = {final_val:.6f}")
        if final_val < best_val:
            best_val = final_val
            best_seq = a_final.tolist()
            print("*** Updated global best after final polish ***")

    # ------------------------------------------------------------------
    # 3️⃣  Guard – guarantee a return value
    # ------------------------------------------------------------------
    if not best_seq:
        best_seq = arcsine_initial(256).tolist()
        best_val = evaluate_sequence(best_seq)

    print(f"\nSearch finished. Best reported score = {best_val:.6f}")
    return best_seq
