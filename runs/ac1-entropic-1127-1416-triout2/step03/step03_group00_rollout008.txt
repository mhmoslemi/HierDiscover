"""
Optimise step‐function heights via smoothed max‐convolution gradient descent.
The objective max(a*a) is approximated by a log‑sum‑exp (smooth max) which
has a closed‑form gradient that can be computed by a single FFT‑based
convolution.  An Adam optimiser and a simplex projection keep the iterates
on the probability simplex.  Several random restarts with increasing
smoothness (β) explore the landscape; a short simulated‑annealing phase
finishes the search.
"""

import time
import math
import numpy as np

def simplex_projection(v: np.ndarray) -> np.ndarray:
    """Euclidean projection onto the simplex {x>=0, sum x = 1}."""
    v = np.asarray(v, dtype=float)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1] - 1
    theta = cssv[cond][-1] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w

def compute_score(a: np.ndarray) -> float:
    """Exact evaluation (sum(a)=1). Returns 2·n·max(a*a)."""
    n = a.shape[0]
    fft_a = np.fft.rfft(a, n=2 * n - 1)
    conv = np.fft.irfft(fft_a * fft_a, n=2 * n - 1)

    """

    # slice corresponds to indices n‑1 … n‑1+n‑1
    grad = 2.0 * conv_wr[n - 1 : n - 1 + n]
    return grad

"""
Optimise a non‑negative step‑function (probability vector) so that the
maximum entry of its self‑convolution is as small as possible.
The objective max(a*a) is approximated by a smooth log‑sum‑exp
(which equals the exact max when the temperature β→∞).  The gradient of the
smooth max has a closed form that can be evaluated by a single FFT‑based
convolution.  An Adam optimiser together with Euclidean projection onto the
simplex keeps every iterate feasible.  Several random restarts with an
increasing β explore the landscape, and a short simulated‑annealing phase
finishes the search.
"""

def simplex_projection(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of a vector onto the probability simplex."""
    v = np.asarray(v, dtype=float)
    n = v.shape[0]
    # sort descending
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1] - 1        # zero‑based index
    theta = cssv[cond][-1] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def compute_score(a: np.ndarray) -> float:
