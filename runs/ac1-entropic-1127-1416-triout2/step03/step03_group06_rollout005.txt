def propose_candidate(seed=0, budget_s=1000):
    np.random.seed(seed)
    start = time.time()
    deadline = start + budget_s - 5

    # Determine n
    try:
        n = len(height_sequence_1)
        init = np.array(height_sequence_1, dtype=float)
        init = np.maximum(init, 0.0)
        init_sum = init.sum()
        if init_sum == 0:
            init = np.arange(n, 0, -1, dtype=float)
        a = init / init_sum
    except NameError:
        n = 8000
        a = np.arange(n, 0, -1, dtype=float)
        a = a / a.sum()
    
    best_seq = a.copy()
    best_score = compute_objective(best_seq)
    print(f"Initial score: {best_score:.6f}")

    a_prev = a.copy()
    
    # define parameters
    stages = [
        (5.0, 50000),
        (20.0, 150000),
        (100.0, 300000),
        (400.0, 500000),
        (1500.0, 1000000)
    ]
    eta0 = 0.5   # step size base
    momentum = 0.5
    iteration_total = 0
    stagnation = 0
    
    for beta, max_iters in stages:
        iter_stage = 0
        while time.time() < deadline and iter_stage < max_iters:
            # compute FFT of a zero-padded to N = 3n-2
            N = 3 * n - 2
            # pad a
            a_padded = np.zeros(N, dtype=float)
            a_padded[:n] = a
            fft_a = np.fft.rfft(a_padded)
            # convolution to get full conv
            conv_full = np.fft.irfft(fft_a * fft_a, n=N)
            conv = conv_full[:2*n-1]
            maxb = conv.max()
            cur_score = 2.0 * n * maxb
            if cur_score < best_score - 1e-12:
                best_score = cur_score
                best_seq = a.copy()
                stagnation = 0
                print(f"Iter {iteration_total:7d}: new best {best_score:.6f} beta={beta}")
            else:
                stagnation += 1

            # compute softmax weights
            # for numeric stability subtract max
            maxc = maxb
            exp_beta = np.exp(beta * (conv - maxc))
            w = exp_beta / exp_beta.sum()
            # pad w
            w_padded = np.zeros(N, dtype=float)
            w_padded[:2*n-1] = w
            fft_w = np.fft.rfft(w_padded)
            # correlation via inverse FFT of fft_w * conj(fft_a)
            corr = np.fft.irfft(fft_w * np.conj(fft_a), n=N)
            # extract gradient component
            # g_j = sum_i w_i a_{i - j} = corr[n-1 - j]
            indices = np.arange(n)
            g = corr[n-1 - indices]
            grad = 4.0 * n * g

            # step size schedule
            step = eta0 / np.sqrt(iter_stage + 1)
            a_next = a - step * grad + momentum * (a - a_prev)
            # project onto simplex
            a_next = simplex_projection(a_next)

            a_prev = a
            a = a_next

            iteration_total += 1
            iter_stage += 1

            if stagnation >= 50000:
                # random perturb
                sigma = max(0.01, 0.5 / np.sqrt(iteration_total + 1))
                noise = sigma * np.random.randn(n) * a
                a = np.clip(a + noise, 0.0, None)
                a /= a.sum()
                a_prev = a.copy()
                stagnation = 0
                print(f"Iter {iteration_total:7d}: perturbation applied, score {cur_score:.6f}")

            if best_score <= 1.503:
                print(f"Target reached at iter {iteration_total}, score {best_score:.6f}")
                break
        if best_score <= 1.503 or time.time() >= deadline:
            break

    # final return
    print(f"Finished after {iteration_total} iterations, best score = {best_score:.6f}")
    return [float(v) for v in best_seq.tolist()]

"""
Propose a candidate step‑function that minimises ``evaluate_sequence``.

Algorithm
---------
1. Initialise a probability vector `a` (sum = 1) from the previously
   discovered best sequence ``height_sequence_1`` if it exists,
   otherwise from a simple decreasing ramp.
2. Optimise the *smooth* surrogate of the max‑convolution:
        Φβ(a) = 2·n·log‑sum‑exp( β·(a∗a) ) / β
   whose gradient can be written in closed form using a single
   FFT‑based correlation.
3. Run a projected (Euclidean) gradient descent with Nesterov‑style
   momentum on the simplex.  The smoothing parameter β is increased
   gradually (continuation) so that the optimisation first moves
   quickly on a smooth landscape and later refines the true
   max‑objective.
4. Occasionally apply a small random perturbation when progress stalls
   (to escape plateaux).
5. Keep the best vector ever seen (according to the exact
   ``evaluate_sequence``).  The loop respects the 1000 s budget
   and finally returns the best sequence as a plain list of floats.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities
# ----------------------------------------------------------------------


def simplex_projection(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the probability simplex:
    { w | w_i ≥ 0 , Σ w_i = 1 }.
    Implementation follows Duchi et al. (2008).
    """
    v = np.asarray(v, dtype=float)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all entries are non‑positive → uniform distribution
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1] - 1               # zero‑based index
    theta = cssv[cond][-1] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def compute_objective(a: np.ndarray) -> float:
    """
    Exact objective used by ``evaluate_sequence`` for a vector with
    Σ a_i = 1:
        2·n·max(a∗a)
    """
    n = a.shape[0]
    # linear convolution via FFT (length 2·n‑1)
    fft_a = np.fft.rfft(a, n=2 * n - 1)
    conv = np.fft.irfft(fft_a * fft_a, n=2 * n - 1)
    return 2.0 * n * conv.max()


# ----------------------------------------------------------------------
# Main search routine
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 0, budget_s: int = 1000):
    """
    Return a list of non‑negative heights (a probability vector) that
    minimises ``evaluate_sequence``.  The routine stops after
    ``budget_s`` seconds (minus a small safety margin) and returns the
    best sequence encountered.
    """
    np.random.seed(seed)
    start_time = time.time()
    # keep a few seconds for the outer driver
    deadline = start_time + budget_s - 5.0

    # --------------------------------------------------------------
    # 1. Problem size and initial vector
    # --------------------------------------------------------------
    try:
        # ``height_sequence_1`` is provided by the outer environment
        init = np.asarray(height_sequence_1, dtype=float)
        if init.ndim != 1:
            raise ValueError
        n = init.shape[0]
        init = np.maximum(init, 0.0)          # safety against negatives
        s = init.sum()
        if s == 0.0:
            raise ValueError
        a = init / s                          # normalise to a probability vector
    except Exception:
        # fallback: a simple decreasing ramp (the previous best shape)
        n = 5146
        a = np.arange(n, 0, -1, dtype=float)
        a = a / a.sum()

    # --------------------------------------------------------------
    # 2. Book‑keeping
    # --------------------------------------------------------------
    best_seq = a.copy()
    best_score = compute_objective(best_seq)
    print(f"Initial length {n}, score {best_score:.6f}")

    a_prev = a.copy()
    iteration_total = 0
    stagnation = 0

    # --------------------------------------------------------------
    # 3. Continuation schedule (beta grows, step‑size decays)
    # --------------------------------------------------------------
    schedule = [
        (5.0,   30_000),   # smooth start
        (20.0,  70_000),
        (100.0, 100_000),
        (400.0, 100_000),
        (1500.0, 200_000)  # very sharp, approximates the true max
    ]

    # base step‑size (tuned empirically)
    eta0 = 0.15
    momentum = 0.55

    # --------------------------------------------------------------
    # 4. Main optimisation loop
    # --------------------------------------------------------------
    for beta, max_iters in schedule:
        # pre‑compute constant values for the current stage
        N = 3 * n - 2                    # length needed for correlation
        half_N = N // 2 + 1              # length of rfft output
        iter_stage = 0

        while time.time() < deadline and iter_stage < max_iters:
            # ----- 4.1 convolution a∗a (full length N) -----
            a_pad = np.zeros(N, dtype=float)
            a_pad[:n] = a
            fft_a = np.fft.rfft(a_pad)
            conv_full = np.fft.irfft(fft_a * fft_a, n=N)
            conv = conv_full[: 2 * n - 1]           # linear self‑convolution
            maxb = conv.max()
            cur_score = 2.0 * n * maxb

            # ----- keep global best -----
            if cur_score < best_score - 1e-12:
                best_score = cur_score
                best_seq = a.copy()
                stagnation = 0
                print(f"Iter {iteration_total:7d}: new best {best_score:.6f} (β={beta:.1f})")
            else:
                stagnation += 1

            # ----- 4.2 smooth weights (softmax over conv entries) -----
            # numerically stable softmax
            shift = maxb                     # max(conv) = maxb
            exp_terms = np.exp(beta * (conv - shift))
            w = exp_terms / exp_terms.sum()   # shape (2n‑1,)

            # ----- 4.3 gradient of the smooth surrogate -----
            # pad w to length N
            w_pad = np.zeros(N, dtype=float)
            w_pad[: 2 * n - 1] = w
            fft_w = np.fft.rfft(w_pad)

            # correlation = conv(w, a_rev) = ifft(fft_w * conj(fft_a))
            corr = np.fft.irfft(fft_w * np.conj(fft_a), n=N)

            # extract g_j = Σ_i w_i a_{i‑j}
            # correlation index k = (n‑1) + j encodes shift j
            idx = (n - 1) + np.arange(n)
            g = corr[idx]                     # shape (n,)

            # gradient of Φβ : 4·n·g
            grad = 4.0 * n * g

            # ----- 4.4 projected (Nesterov) gradient step -----
            step = eta0 / np.sqrt(iter_stage + 1.0)
            a_next = a - step * grad + momentum * (a - a_prev)

            # Euclidean projection back onto the simplex
            a_next = simplex_projection(a_next)

            a_prev = a
            a = a_next

            iteration_total += 1
            iter_stage += 1

            # ----- 4.5 occasional random perturbation (escape plateaus) -----
            if stagnation >= 60_000:
                sigma = max(0.01, 0.4 / np.sqrt(iteration_total + 1.0))
                noise = sigma * np.random.randn(n) * a
                a = np.clip(a + noise, 0.0, None)
                a /= a.sum()
                a_prev = a.copy()
                stagnation = 0
                print(f"Iter {iteration_total:7d}: perturbation applied (β={beta:.1f})")

            # ----- early exit if target already reached -----
            if best_score <= 1.503:
                print(f"Target satisfied at iter {iteration_total}, score {best_score:.6f}")
                break

        if best_score <= 1.503 or time.time() >= deadline:
            break

    # --------------------------------------------------------------
    # 5. Return best sequence found
    # --------------------------------------------------------------
    print(f"Finished after {iteration_total} iterations, best score = {best_score:.6f}")
    return [float(v) for v in best_seq.tolist()]
