"""
Optimisation of a non‑negative step function via
smooth max‑convolution minimisation and L‑BFGS continuation.

The algorithm proceeds as follows:

* For each chosen length `n` a small library of parametric
  candidates (uniform, arcsine, power‑law, beta‑like, Dirichlet)
  is built.  The candidates are ranked by the cheap internal score
  `2·n·max(conv)`.

* The top few seeds are refined by a *continuation* scheme:
  - The non‑smooth objective `2·n·max(a*a)` is approximated by the
    smooth surrogate

          Φ_λ(a) = 2·n/λ·log( Σ exp( λ·(a*a)_k ) ),

    whose gradient can be obtained analytically by a single FFT‑based
    convolution.
  - Starting from a low temperature `λ≈5` we progressively raise `λ`
    (5 → 20 → 80 → 200 → 400) and run a short L‑BFGS‑B optimisation
    at each stage.  The solution of one stage is warm‑started for the
    next, yielding a reliable descent towards the true max‑convolution
    objective.
  - After each stage the current vector is evaluated with the official
    ``evaluate_sequence`` and the best result is retained.

* The globally best sequence found is finally polished with a longer
  continuation (λ up to 800).

All operations stay on the probability simplex (positivity + unit sum)
by representing a vector as `softmax(x)` (i.e. `a = exp(x)/sum(exp(x))`).
The routine respects the allocated wall‑clock budget (default 1000 s)
and always returns a feasible list of non‑negative floats.
"""

import time
from typing import List, Tuple

import numpy as np
from scipy.optimize import minimize

# ----------------------------------------------------------------------
# Low‑level numeric utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Linear convolution via real FFT (full length, zero‑padded to power‑of‑2)."""
    m = len(a) + len(b) - 1
    n_fft = next_pow2(m)
    A = np.fft.rfft(a, n=n_fft)
    B = np.fft.rfft(b, n=n_fft)
    C = A * B
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


def gradient_from_weights(a: np.ndarray, w: np.ndarray) -> np.ndarray:
    """
    Gradient of Σ_k w_k·(a*a)_k w.r.t. a (weights w sum to 1).
    Implements   g_j = 2· Σ_m a_m · w_{m+j}.
    """
    w_rev = w[::-1]
    conv = convolution_fft(a, w_rev)          # length = 3·n‑2
    idx = 2 * len(a) - 2                       # points to position 2n‑2
    grads = 2.0 * conv[idx - np.arange(len(a))]
    return grads


# ----------------------------------------------------------------------
# Simple parametric families for seeding
# ----------------------------------------------------------------------


def arcsine_initial(n: int) -> np.ndarray:
    """Discrete approximation of the arcsine density on {0,…,n‑1}."""
    i = np.arange(n, dtype=float)
    eps = 0.25
    vals = 1.0 / np.sqrt((i + 0.5) * (n - 0.5 - i) + eps)
    return vals / vals.sum()


def generate_shape(n: int, p: float, eps: float) -> np.ndarray:
    """Symmetric power‑law shape a_i ∝ ((i+ε)(n‑1‑i+ε))⁻ᵖ."""
    i = np.arange(n, dtype=float)
    base = ((i + eps) * (n - 1 - i + eps)) ** (-p)
    total = base.sum()
    if total == 0.0:
        return np.full(n, 1.0 / n)
    return base / total


def resize_sequence(seq: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation of a sequence to a new length (preserves positivity)."""
    old_n = len(seq)
    if old_n == new_n:
        return seq.copy()
    x_old = np.arange(old_n, dtype=float)
    x_new = np.linspace(0, old_n - 1, new_n)
    new_seq = np.interp(x_new, x_old, seq)
    new_seq = np.maximum(new_seq, 0.0)
    new_seq /= new_seq.sum()
    return new_seq


# ----------------------------------------------------------------------
# Smooth surrogate of the max‑convolution objective
# ----------------------------------------------------------------------


def smooth_obj_grad(x: np.ndarray, lam: float) -> Tuple[float, np.ndarray]:
    """
    Returns (Φ_λ(a), ∇_x Φ_λ(a)) where a = softmax(x).
    The surrogate is
        Φ_λ(a) = 2·n/λ·log Σ_k exp(λ·(a*a)_k)
    and its gradient is computed analytically using FFT.
    """
    # a on the simplex
    exp_x = np.exp(x)
    a = exp_x / exp_x.sum()
    n = len(a)

    # convolution a*a (full length 2n‑1)
    conv = convolution_fft(a, a)

    # soft‑max weights w_k = exp(λ conv_k) / Σ exp(λ conv)
    z = lam * conv
    z_max = np.max(z)
    exp_z = np.exp(z - z_max)
    w = exp_z / exp_z.sum()

    # smooth objective Φ_λ = 2·n·logsumexp(z) / λ
    log_sumexp = np.log(exp_z.sum()) + z_max   # = log Σ exp(z)
    phi = 2.0 * n * log_sumexp / lam

    # gradient w.r.t. a: 2·n·∇_a Σ w_k conv_k = 2·n·gradient_from_weights(a, w)
    g_a = 2.0 * n * gradient_from_weights(a, w)

    # chain rule for x → a (softmax): ∂Φ/∂x_j = a_j·(g_a_j – Σ_i a_i g_a_i)
    dot = np.dot(g_a, a)
    g_x = a * (g_a - dot)

    return float(phi), g_x.astype(float)


class SmoothObjective:
    """Callable wrapper required by `scipy.optimize.minimize`."""

    def __init__(self, lam: float):
        self.lam = lam

    def __call__(self, x: np.ndarray):
        val, grad = smooth_obj_grad(x, self.lam)
        return val, grad


# ----------------------------------------------------------------------
# Continuation optimiser (low‑temperature warm‑starts)
# ----------------------------------------------------------------------


def optimise_continuation(
    init_a: np.ndarray,
    deadline: float,
    *,
    lam_schedule: List[float] = (5.0, 20.0, 80.0, 200.0, 400.0),
    max_iter_per_stage: int = 200,
) -> Tuple[np.ndarray, float]:
    """
    Runs a continuation of L‑BFGS‑B optimisations at increasing temperatures.
    Returns the best vector (still on the simplex) and its *official* score.
    """
    # initialise the softmax parametrisation
    x = np.log(init_a + 1e-12)  # avoid log(0)

    best_vec = None
    best_score = float("inf")
    n = len(init_a)

    for lam in lam_schedule:
        if time.time() >= deadline:
            break

        # adapt the number of internal L‑BFGS iterations to the remaining budget
        remaining = deadline - time.time()
        # assume about 15 gradient/evaluation calls per second
        max_iter = min(max_iter_per_stage, int(max(5, remaining * 10)))
        if max_iter < 5:
            break

        obj = SmoothObjective(lam)
        res = minimize(
            obj,
            x,
            method="L-BFGS-B",
            jac=True,
            options={"maxiter": max_iter, "disp": False},
        )
        x = res.x

        a = np.exp(x)
        a = a / a.sum()

        # evaluate with the official (slow) evaluator
        try:
            score = evaluate_sequence(a.tolist())
        except Exception:  # safety net – should never happen
            score = float("inf")

        if score < best_score:
            best_score = score
            best_vec = a.copy()

        print(
            f"    λ={lam:.0f}, max_iter={max_iter}, interim score={score:.6f}"
        )

    # fall‑back: if none of the stages managed to evaluate, return the last x
    if best_vec is None:
        a = np.exp(x)
        best_vec = a / a.sum()
        best_score = evaluate_sequence(best_vec.tolist())

    return best_vec, best_score


# ----------------------------------------------------------------------
# Main entry point required by the competition harness
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs) -> List[float]:
    """
    Search for a low‑value step function within the allotted time.
    Returns the best sequence as a plain Python list of floats.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 2.0  # tiny guard‑band

    best_seq: List[float] = []
    best_val: float = float("inf")

    # ------------------------------------------------------------------
    # 0.  Optional fallback from a previously discovered construction.
    # ------------------------------------------------------------------
    try:
        fallback = [float(x) for x in height_sequence_1]  # type: ignore
        val = evaluate_sequence(fallback)
        print(f"Fallback (height_sequence_1) score = {val:.6f}")
        if val < best_val:
            best_val = val
            best_seq = fallback
    except Exception:
        pass

    # ------------------------------------------------------------------
    # 1.  Lengths to explore – powers of two (FFT‑friendly).
    # ------------------------------------------------------------------
    lengths = [1024, 2048, 4096, 8192]

    for n in lengths:
        if time.time() >= deadline - 5.0:
            break
        print(f"\n=== Exploring length n = {n} ===")

        # --------------------------------------------------------------
        # 1a.  Assemble a modest library of seeds.
        # --------------------------------------------------------------
        seeds: List[np.ndarray] = []

        # uniform
        seeds.append(np.full(n, 1.0 / n))

        # classic arcsine
        seeds.append(arcsine_initial(n))

        # random power‑law shapes
        for _ in range(5):
            p = rng.uniform(0.05, 2.0)
            eps = rng.uniform(1e-6, max(0.5, 0.02 * n))
            seeds.append(generate_shape(n, p, eps))

        # beta‑like shapes
        for _ in range(3):
            alpha = rng.uniform(0.2, 5.0)
            beta = rng.uniform(0.2, 5.0)
            i = np.arange(n, dtype=float)
            base = (i + 0.5) ** (alpha - 1) * (n - 1 - i + 0.5) ** (beta - 1)
            tot = base.sum()
            if tot == 0.0:
                base = np.full(n, 1.0 / n)
            else:
                base = base / tot
            seeds.append(base)

        # Dirichlet / exponential draws
        for _ in range(2):
            samp = rng.exponential(1.0, size=n)
            samp /= samp.sum()
            seeds.append(samp)

        # include a resized version of the current global best (if any)
        if best_seq:
            resized = resize_sequence(np.asarray(best_seq, dtype=float), n)
            seeds.append(resized)

        # --------------------------------------------------------------
        # 1b.  Quick internal ranking (fast score, no heavy evaluate_sequence).
        # --------------------------------------------------------------
        internal_scores = [(2.0 * n * np.max(convolution_fft(s, s)), s) for s in seeds]
        internal_scores.sort(key=lambda t: t[0])
        top_seeds = [s for _, s in internal_scores[:3]]   # keep three most promising

        # --------------------------------------------------------------
        # 1c.  Refine each promising seed via continuation + L‑BFGS.
        # --------------------------------------------------------------
        for idx, seed_vec in enumerate(top_seeds):
            if time.time() >= deadline - 5.0:
                break
            print(f"  candidate {idx + 1}/{len(top_seeds)} (n={n})")

            # allocate a modest budget for this candidate
            remaining = max(0.0, deadline - time.time())
            # rough estimate: ~0.05 s per L‑BFGS stage → allow up to 30 s
            cand_deadline = time.time() + min(30.0, remaining)

            refined, cand_score = optimise_continuation(
                init_a=seed_vec,
                deadline=cand_deadline,
                lam_schedule=(5.0, 20.0, 80.0, 200.0, 400.0),
                max_iter_per_stage=250,
            )

            print(f"    candidate final score = {cand_score:.6f}")

            if cand_score < best_val:
                best_val = cand_score
                best_seq = refined.tolist()
                print(f"    *** New global best: {best_val:.6f} (n={n}) ***")

    # ------------------------------------------------------------------
    # 2.  Final polishing of the overall best (if time permits).
    # ------------------------------------------------------------------
    if best_seq and time.time() < deadline - 5.0:
        print("\nFinal polishing of the global best")
        a0 = np.asarray(best_seq, dtype=float)
        remaining = max(0.0, deadline - time.time())
        polish_deadline = time.time() + min(40.0, remaining)

        polished, final_score = optimise_continuation(
            init_a=a0,
            deadline=polish_deadline,
            lam_schedule=(20.0, 80.0, 200.0, 400.0, 800.0),
            max_iter_per_stage=300,
        )
        print(f"Polished score = {final_score:.6f}")
        if final_score < best_val:
            best_val = final_score
            best_seq = polished.tolist()
            print("*** Updated global best after final polish ***")

    # ------------------------------------------------------------------
    # 3.  Safety net – always return a feasible sequence.
    # ------------------------------------------------------------------
    if not best_seq:
        best_seq = [1.0] * 256
        best_val = evaluate_sequence(best_seq)

    print(f"\nSearch finished. Best reported score = {best_val:.6f}")
    return best_seq
