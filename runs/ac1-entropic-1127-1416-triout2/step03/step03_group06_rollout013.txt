"""
Propose a step‑function (non‑negative sequence) that minimises

    2·n·max(a∗a) / (∑a)²

where n = len(a) and ∗ is the ordinary (linear) convolution.
The optimisation uses a smooth approximation of the max‑operator
(soft‑max with a gradually increasing temperature) together with an
Adam optimiser on the probability simplex (Euclidean projection after each
step).  Random perturbations are injected when progress stalls.
The routine works for the default length 5146 (or the length of the global
`height_sequence_1` if it is present) and respects a 1000 s budget.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper functions (all top‑level – no closures)
# ----------------------------------------------------------------------


def simplex_projection(v):
    """
    Euclidean projection of a vector onto the probability simplex:
    { w | w_i ≥ 0 , Σ w_i = 1 }.
    Implementation follows Duchi et al. (2008).
    """
    v = np.asarray(v, dtype=float)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all entries are non‑positive → uniform distribution
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1] - 1               # zero‑based index
    theta = cssv[cond][-1] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def next_pow2(x: int) -> int:
    """Small utility – next power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def conv_full_fft(a: np.ndarray) -> np.ndarray:
    """
    Linear convolution a∗a using FFT.
    Returns the full (2·n‑1) array.
    """
    n = a.shape[0]
    L = next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, L)
    conv = np.fft.irfft(fa * fa, L)[:2 * n - 1]
    return conv


def correlation_fft(w: np.ndarray, a: np.ndarray) -> np.ndarray:
    """
    Cross‑correlation (w ⋆ a)[s] = Σ_j w_{j+s}·a_j for s = 0,…,n‑1.
    The full correlation length is len(w)+len(a)-1; we extract the slice
    that corresponds to non‑negative shifts (the part needed for the
    gradient of the soft‑max of the convolution).
    """
    n = a.shape[0]
    L = next_pow2(len(w) + n - 1)
    fw = np.fft.rfft(w, L)
    fa = np.fft.rfft(a, L)
    corr_full = np.fft.irfft(fw * np.conj(fa), L)
    start = n - 1
    return corr_full[start:start + n]


def compute_objective(a: np.ndarray) -> float:
    """
    Exact evaluation of the target function assuming Σ a_i = 1.
    Returns 2·n·max(a∗a).
    """
    n = a.shape[0]
    conv = conv_full_fft(a)
    return 2.0 * n * conv.max()


# ----------------------------------------------------------------------
# Main search routine
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 0, budget_s: int = 1000):
    """
    Search for a non‑negative step function (list of floats) that minimises
    ``evaluate_sequence``.  A simple stochastic optimisation based on a
    smoothed max‑objective and Adam updates is performed inside the time
    budget.  The best sequence found is returned.
    """
    np.random.seed(seed)
    start_time = time.time()
    # keep a small safety margin for the outer harness
    deadline = start_time + budget_s - 5.0

    # --------------------------------------------------------------
    # 1. Problem size and initial sequence
    # --------------------------------------------------------------
    try:
        # if the previous run stored a good candidate it is available as
        # the global variable `height_sequence_1`
        a0 = np.asarray(height_sequence_1, dtype=float).copy()  # type: ignore[name-defined]
    except Exception:
        # fallback: uniform distribution – works for any length
        a0 = None

    if a0 is None:
        # use the default length that appeared in the prompt
        n = 5146
        a = np.full(n, 1.0 / n)
    else:
        n = a0.shape[0]
        # make sure we are on the probability simplex
        a0 = np.clip(a0, 0.0, None)
        s = a0.sum()
        a = a0 / s if s > 0 else np.full(n, 1.0 / n)

    # initialise bookkeeping
    best_seq = a.copy()
    best_score = compute_objective(best_seq)
    print(f"Initial score (seed) : {best_score:.6f}")

    # --------------------------------------------------------------
    # 2. Quick exploration of a few deterministic families
    #    (only while time permits)
    # --------------------------------------------------------------
    def maybe_update(candidate):
        nonlocal best_seq, best_score
        s = compute_objective(candidate)
        if s < best_score:
            best_score = s
            best_seq = candidate.copy()
            print(f"Seed improvement      : {s:.6f}")

    # a few simple parametric families
    if time.time() < deadline:
        # exponential family
        for lam in np.linspace(0.001, 10.0, 30):
            i = np.arange(n, dtype=float)
            w = np.exp(-lam * i)
            w /= w.sum()
            maybe_update(w)

    if time.time() < deadline:
        # polynomial decreasing shape 1-(i/(n‑1))^p
        for p in [0.5, 1.0, 1.5, 2.0, 3.0, 5.0]:
            i = np.arange(n, dtype=float)
            frac = i / (n - 1)
            w = 1.0 - np.power(frac, p)
            w = np.clip(w, 0.0, None)
            w /= w.sum()
            maybe_update(w)

    if time.time() < deadline:
        # coarse Beta sweep
        alphas = np.linspace(0.2, 5.0, 10)
        betas = np.linspace(0.2, 5.0, 10)
        for α in alphas:
            for β in betas:
                if time.time() >= deadline:
                    break
                x = (np.arange(n) + 0.5) / n
                logw = (α - 1.0) * np.log(x) + (β - 1.0) * np.log(1.0 - x)
                logw = np.clip(logw, -700.0, 700.0)
                w = np.exp(logw)
                tot = w.sum()
                if tot == 0 or not np.isfinite(tot):
                    continue
                w /= tot
                maybe_update(w)
            if time.time() >= deadline:
                break

    # --------------------------------------------------------------
    # 3. Adam optimisation of a smooth approximation of the max
    # --------------------------------------------------------------
    # Adam state
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8
    eta0 = 0.20               # base learning rate (effective step size)

    # soft‑max temperature (γ) – start low, increase slowly
    gamma = 5.0
    gamma_inc = 1.03
    gamma_max = 200.0

    iteration = 0
    stagnation = 0

    while time.time() < deadline:
        # ----- 1) convolution of the current vector -----
        conv = conv_full_fft(a)
        max_conv = conv.max()

        # ----- 2) smooth max via soft‑max -----
        # stable computation: shift by the maximum before exponentiating
        shift = max_conv
        exp_term = np.exp(gamma * (conv - shift))
        sum_exp = exp_term.sum()
        w = exp_term / sum_exp                     # shape (2*n-1,)

        # ----- 3) gradient of the smooth max (2·corr) -----
        grad = 2.0 * correlation_fft(w, a)         # shape (n,)

        # ----- 4) Adam update (projected onto the simplex) -----
        iteration += 1
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)

        m_hat = m / (1.0 - beta1 ** iteration)
        v_hat = v / (1.0 - beta2 ** iteration)

        step = eta0 / (np.sqrt(v_hat) + eps)
        a_next = a - step * m_hat

        # Euclidean projection back onto Σ a_i = 1, a_i ≥ 0
        a_next = simplex_projection(a_next)

        # ----- 5) evaluate & bookkeeping -----
        cur_score = compute_objective(a_next)
        if cur_score < best_score - 1e-12:
            best_score = cur_score
            best_seq = a_next.copy()
            stagnation = 0
            print(
                f"Iter {iteration:6d}  new best {best_score:.6f} "
                f"(max conv {np.max(conv_full_fft(a_next)):.6e})"
            )
        else:
            stagnation += 1

        a = a_next

        # ----- 6) temperature schedule -----
        if gamma < gamma_max:
            gamma = min(gamma_max, gamma * gamma_inc)

        # ----- 7) occasional random perturbation to escape plateaus -----
        if stagnation >= 20000:
            sigma = max(0.01, 0.5 / np.sqrt(iteration))
            noise = sigma * np.random.randn(n) * a
            a = a + noise
            a = np.clip(a, 0.0, None)
            a /= a.sum()
            m.fill(0.0)
            v.fill(0.0)
            stagnation = 0
            print(f"Iter {iteration:6d}  perturbation applied, score {cur_score:.6f}")

        # ----- 8) early exit if target is already reached -----
        if best_score <= 1.503:
            print(f"Target reached at iteration {iteration}, best score {best_score:.6f}")
            break

    # --------------------------------------------------------------
    # 9. Return the best sequence encountered
    # --------------------------------------------------------------
    print(f"Finished after {iteration} iterations, best score = {best_score:.6f}")
    # Convert to plain Python list as required by the external driver.
    return [float(v) for v in best_seq.tolist()]
