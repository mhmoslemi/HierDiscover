def propose_candidate(seed=0, budget_s=1000):
    np.random.seed(seed)
    start = time.time()
    deadline = start + budget_s - 0.5

    # choose n
    try:
        init_seq = np.array(height_sequence_1, dtype=float)
        n = init_seq.shape[0]
        print(f"Start from existing sequence length {n}")
    except NameError:
        n = 5146
        init_seq = np.ones(n) / n  # uniform start fallback

    best_a = None
    best_score = np.inf

    # function to run one optimization run from given a0
    def optimize_from(a0):
        nonlocal best_a, best_score
        a = a0.copy()
        a = a / a.sum()
        # parameters
        lam = 1.0
        lam_max = 2000.0
        lam_factor = 1.5
        max_total_iters = 20000
        it = 0
        while time.time() < deadline and lam <= lam_max and it < max_total_iters:
            # compute convolution
            fft_a = np.fft.rfft(a, n=2*n-1)
            conv = np.fft.irfft(fft_a * fft_a, n=2*n-1)
            # smooth max weights
            # log-sum-exp trick
            maxc = conv.max()
            # use lam to compute softmax
            scaled = lam * (conv - maxc)
            exp_scaled = np.exp(scaled)
            w = exp_scaled / exp_scaled.sum()
            # gradient via correlation
            corr = np.convolve(w, a[::-1], mode='full')
            grad = 2.0 * corr[n-1:n-1+n]
            # line search
            step = 1.0 / (lam + 1.0)  # basic step size; adapt
            # try several step reductions
            improved = False
            for _ in range(10):
                a_new = simplex_projection(a - step * grad)
                # evaluate exact score
                score_new = compute_score(a_new)
                if score_new < best_score:
                    best_score = score_new
                    best_a = a_new.copy()
                    print(f"New best {best_score:.6f} (lam={lam:.1f}, it={it})")
                # decide if improvement for this move
                cur_score = compute_score(a)
                if score_new <= cur_score:
                    a = a_new
                    improved = True
                    break
                else:
                    step *= 0.5
            # if no improvement after line search, still accept? maybe keep a unchanged
            it += 1
            # schedule lam increase
            if it % 200 == 0:
                lam = min(lam_max, lam * lam_factor)
        # final evaluation
        final_score = compute_score(a)
        if final_score < best_score:
            best_score = final_score
            best_a = a.copy()
            print(f"Run finished: best {best_score:.6f}")

    # initial run from existing or default
    optimize_from(init_seq)

    # additional random restarts
    while time.time() < deadline - 5.0:
        # random beta shape
        alpha = np.random.uniform(0.3, 5.0)
        beta = np.random.uniform(0.3, 5.0)
        a0 = beta_sequence(alpha, beta, n)
        optimize_from(a0)

    # fallback: if we never found a solution, just return uniform
    if best_a is None:
        best_a = np.ones(n) / n

    # return as list
    best_a = best_a / best_a.sum()
    return [float(x) for x in best_a.tolist()]

"""
Propose a step–function (non‑negative probability vector) that minimizes
evaluate_sequence().  The algorithm combines

1.  A smooth approximation of the max‑convolution (log‑sum‑exp) so that a
    differentiable objective can be used.
2.  Projected gradient descent with a simple back‑tracking line‑search on the
    probability simplex.
3.  A continuation on the smoothing parameter λ (small → smooth, large →
    close to the true max).  Several random restarts (Beta‑shaped starts) are
    performed within the time budget.

The code is completely self‑contained; it only relies on numpy and the
existing evaluate_sequence() that will be called after the candidate is
returned.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Helper functions (all top‑level, no closures)
# ----------------------------------------------------------------------


def simplex_projection(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the probability simplex:
    { w | w_i >= 0 , sum w_i = 1 }.
    Duchi et al. (2008) algorithm.
    """
    v = np.asarray(v, dtype=float)
    n = v.shape[0]

    # sort in decreasing order
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all non‑positive → uniform distribution
        return np.full_like(v, 1.0 / n)

    rho = ind[cond][-1] - 1          # zero‑based index
    theta = cssv[cond][-1] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def compute_score(a: np.ndarray) -> float:
    """
    Exact value of evaluate_sequence() for a vector already normalized to
    sum(a) == 1.  The formula reduces to 2·n·max(a * a) where * denotes
    linear convolution.
    """
    n = a.shape[0]
    fft_a = np.fft.rfft(a, n=2 * n - 1)
    conv = np.fft.irfft(fft_a * fft_a, n=2 * n - 1)
    return 2.0 * n * conv.max()


def beta_sequence(alpha: float, beta: float, n: int) -> np.ndarray:
    """
    Discretised Beta(α,β) density on n points, normalized to sum 1.
    Mid‑point rule avoids the singular endpoints.
    """
    x = (np.arange(n) + 0.5) / n
    logw = (alpha - 1.0) * np.log(x) + (beta - 1.0) * np.log(1.0 - x)
    # protect against overflow/underflow
    logw = np.clip(logw, -700.0, 700.0)
    w = np.exp(logw)
    if not np.isfinite(w.sum()) or w.sum() == 0.0:
        w = np.ones_like(w)
    return w / w.sum()


def _next_pow2(x: int) -> int:
    """Return the smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


# ----------------------------------------------------------------------
# Main optimisation routine
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 0, budget_s: int = 1000) -> list[float]:
    """
    Search for a non‑negative height sequence that minimises evaluate_sequence().
    Uses a smooth‑max objective, projected gradient descent with continuation on
    the smoothing parameter, and a few random restarts.  The routine respects the
    given time budget (seconds) and returns the best vector found (normalised to
    sum == 1).
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5          # safety margin

    # --------------------------------------------------------------
    # 1. Problem size & initial point
    # --------------------------------------------------------------
    try:
        # try to continue from a previous run
        init = np.array(height_sequence_1, dtype=float)   # type: ignore[name-defined]
        n = init.shape[0]
        print(f"Using existing height_sequence_1 of length {n}.")
    except NameError:
        # no previous solution – start from a modest Beta shape
        n = 5146
        init = beta_sequence(2.5, 2.5, n)
        print(f"No existing sequence – using a beta start, length {n}.")

    # Ensure we start on the simplex
    a0 = init / init.sum()
    best_a = a0.copy()
    best_score = compute_score(best_a)
    print(f"Initial score = {best_score:.6f}")

    # --------------------------------------------------------------
    # 2. Core optimiser (smooth‑max + projected gradient)
    # --------------------------------------------------------------
    def optimise_from(start_vec: np.ndarray) -> None:
        """
        Run a single optimisation run starting from *start_vec*.
        Updates the global best_a / best_score when improvements appear.
        """
        nonlocal best_a, best_score

        a = start_vec.copy()
        a /= a.sum()
        n_local = a.shape[0]

        # pre‑allocate FFT length for the correlation step (w * a_rev)
        L_corr = _next_pow2(3 * n_local - 2)

        # smoothing parameter λ schedule
        lam = 1.0            # start smooth
        lam_target = 5000.0  # final (quite sharp)
        lam_factor = 1.5
        lam_update_every = 300   # iterations

        # basic line‑search parameters
        max_iters = 5000
        it = 0
        cur_fft = np.fft.rfft(a, n=2 * n_local - 1)
        cur_conv = np.fft.irfft(cur_fft * cur_fft, n=2 * n_local - 1)
        cur_score = 2.0 * n_local * cur_conv.max()

        while time.time() < deadline and it < max_iters:
            # ------------------------------------------------------
            # 2.1  Build the smooth‑max weights (softmax on conv)
            # ------------------------------------------------------
            maxc = cur_conv.max()
            # stable softmax: subtract the max before exponentiation
            scaled = lam * (cur_conv - maxc)
            exp_scaled = np.exp(scaled)
            w = exp_scaled / exp_scaled.sum()          # shape (2n-1,)

            # ------------------------------------------------------
            # 2.2  Gradient of the smooth objective:
            #       grad_i = 2 * Σ_j w_{i+j} a_j   (correlation)
            # ------------------------------------------------------
            a_rev = a[::-1]
            fft_w = np.fft.rfft(w, n=L_corr)
            fft_ar = np.fft.rfft(a_rev, n=L_corr)
            corr_full = np.fft.irfft(fft_w * fft_ar, n=L_corr)
            # central slice gives the correlation for i = 0 … n-1
            grad = 2.0 * corr_full[n_local - 1:n_local - 1 + n_local]

            # ------------------------------------------------------
            # 2.3  Back‑tracking line search on the *true* objective
            # ------------------------------------------------------
            step = 2.0 / (lam + 1.0 + math.sqrt(it + 1))   # simple diminishing step
            improved = False
            for _ in range(10):

                # evaluate exact score (no smoothing!)
                fft_cand = np.fft.rfft(cand, n=2 * n_local - 1)
                conv_cand = np.fft.irfft(fft_cand * fft_cand, n=2 * n_local - 1)
                cand_score = 2.0 * n_local * conv_cand.max()

                if cand_score <= cur_score:
                    a = cand
                    cur_conv = conv_cand
                    cur_score = cand_score
                    improved = True
                    break
                else:
                    step *= 0.5

            # keep the globally best solution
            if cur_score < best_score:
                best_score = cur_score
                best_a = a.copy()
                print(f"New global best {best_score:.6f} (it {it}, λ={lam:.1f})")

            # ------------------------------------------------------
            # 2.4  λ‑schedule & iteration bookkeeping
            # ------------------------------------------------------
            it += 1
            if (it % lam_update_every) == 0 and lam < lam_target:
                lam = min(lam_target, lam * lam_factor)

            # occasional random diversification if stuck
            if not improved and (it % 1000) == 0:

                mask = np.random.rand(n_local) < 0.1
                if mask.any():
                    a[mask] = beta_sequence(
                        np.random.uniform(0.5, 5.0),
                        np.random.uniform(0.5, 5.0),
                        mask.sum(),
                    )
                    a = a / a.sum()
                    # recompute convolution for the perturbed point
                    fft_a = np.fft.rfft(a, n=2 * n_local - 1)
                    cur_conv = np.fft.irfft(fft_a * fft_a, n=2 * n_local - 1)
                    cur_score = 2.0 * n_local * cur_conv.max()

    # ------------------------------------------------------------------
    # 3. Run optimisation: start from the known sequence and then from
    #    a few random Beta initialisers.
    # ------------------------------------------------------------------
    optimise_from(best_a)                     # try the existing solution first

    # Additional stochastic restarts while time permits
    while time.time() < deadline - 5.0:
        alpha = np.random.uniform(0.3, 5.0)
        beta = np.random.uniform(0.3, 5.0)
        a_start = beta_sequence(alpha, beta, n)
        optimise_from(a_start)

    # ------------------------------------------------------------------
    # 4. Return the best vector found (already normalised to sum == 1)
    # ------------------------------------------------------------------
    best_a = best_a / best_a.sum()
    return [float(v) for v in best_a.tolist()]
