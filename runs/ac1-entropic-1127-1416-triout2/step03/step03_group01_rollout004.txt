"""
Propose a step‑function with a low value of evaluate_sequence.

The algorithm works in three layers:

1. **Exploration** –  For several lengths (2048 … 16384) we create many
   candidate probability vectors.  They come from a mixture of
   –   classic arcsine shape,
   –   symmetric beta densities,
   –   symmetric power‑law (edge‑heavy) densities,
   –   sparse Dirichlet draws.
   Each candidate is projected onto the simplex (Σa = 1) and scored by the exact
   fast surrogate  ``2 n·max(a*a)`` (identical to the official objective because the
   evaluator is scale‑invariant).

2. **Refinement** –  The best few candidates per length are polished with a
   projected Adam optimiser that minimizes a *smooth* approximation of the
   max‑convolution.  The temperature λ is annealed exponentially from a very
   smooth regime (λ≈0.5) to a sharp one (λ≈200–300).  The optimiser respects the
   simplex constraint by projecting after each update.  Random jitter is added
   periodically to escape shallow minima.

3. **Final polishing** –  The globally best vector found so far is refined
   again with a longer Adam run (larger λ) and, if time permits, a brief
   stochastic “annealing’’ perturbation phase.

All heavy work uses FFT‑based convolution, which is O(n log n) and fast even for
n=16384.  The official ``evaluate_sequence`` function is called only for a few
promising candidates, keeping the total running time comfortably below the
1000 s budget.

The returned list is guaranteed to be non‑empty, non‑negative and to respect
the evaluator’s security checks.
"""

import time
import numpy as np
from scipy.special import logsumexp
from typing import List, Tuple

# ----------------------------------------------------------------------
# Low‑level utilities (no closures)
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """
    Real‑FFT convolution of two 1‑D arrays.
    Returns the linear convolution (length = len(a)+len(b)-1).
    """
    m = len(a) + len(b) - 1
    n_fft = next_pow2(m)
    A = np.fft.rfft(a, n=n_fft)
    B = np.fft.rfft(b, n=n_fft)
    C = A * B
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


def compute_score(a: np.ndarray) -> float:
    """
    Exact surrogate for the official score.
    The vector a is assumed to sum to 1; then
        score = 2·n·max(a*a)
    equals the value returned by evaluate_sequence for the same a.
    """
    n = len(a)
    conv = convolution_fft(a, a)
    return float(2.0 * n * np.max(conv))


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection onto the probability simplex {x ≥ 0, Σx = 1}.
    Duchi et al., 2008.
    """
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / v.size)

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho = np.where(u - cssv / np.arange(1, v.size + 1) > 0)[0]
    if len(rho) == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def arcsine_initial(n: int) -> np.ndarray:
    """Classic arcsine shape: a_i ∝ 1/√((i+½)*(n‑½‑i))."""
    i = np.arange(n, dtype=float)
    eps = 0.25
    vals = 1.0 / np.sqrt((i + 0.5) * (n - 0.5 - i) + eps)
    return vals / vals.sum()


def generate_beta_shape(n: int, alpha: float, beta: float, eps: float) -> np.ndarray:
    """
    Symmetric beta‑family density on {0,…,n‑1}:
        a_i ∝ (i+ε)^{α‑1}·(n‑1‑i+ε)^{β‑1}.
    Returned vector is normalized to Σa = 1.
    """
    i = np.arange(n, dtype=float)
    left = (i + eps) ** (alpha - 1.0)
    right = (n - 1 - i + eps) ** (beta - 1.0)
    base = left * right
    total = base.sum()
    if total <= 0.0 or not np.isfinite(total):
        return np.full(n, 1.0 / n)
    return base / total


def generate_powerlaw_shape(n: int, p: float, eps: float) -> np.ndarray:
    """
    Symmetric edge‑heavy shape:
        a_i ∝ (i+ε)^{-p} + (n‑1‑i+ε)^{-p}.
    """
    i = np.arange(n, dtype=float)
    left = (i + eps) ** (-p)
    right = (n - 1 - i + eps) ** (-p)
    vec = left + right
    if not np.isfinite(vec).all():
        vec = np.maximum(vec, 0.0)
    return vec / vec.sum()


def generate_dirichlet_shape(n: int, concentration: float, rng: np.random.Generator) -> np.ndarray:
    """
    Sparse Dirichlet draw using Gamma(α,1) variables.
    """
    g = rng.gamma(shape=concentration, scale=1.0, size=n)
    if g.sum() == 0.0:
        g = np.ones(n)
    return g / g.sum()


# ----------------------------------------------------------------------
# Smooth gradient of the soft‑max objective
# ----------------------------------------------------------------------


def smooth_gradient(a: np.ndarray, lam: float) -> Tuple[np.ndarray, float]:
    """
    Gradient of the smoothed objective

        F_λ(a) = 2·n·softmax_λ(a*a) / (Σa)²

    where softmax_λ(x) = (log∑exp(λ·x))/λ.
    Returns (gradient, smooth_max_value).
    """
    n = len(a)
    conv = convolution_fft(a, a)               # length 2n‑1
    max_c = np.max(conv)

    # stable soft‑max weights
    shifted = lam * (conv - max_c)              # max shift = 0
    exp_term = np.exp(shifted)                 # already safe
    sum_exp = np.sum(exp_term)
    w = exp_term / sum_exp
    B = max_c + np.log(sum_exp) / lam          # smooth approximation of max

    # weighted sub‑gradient of a*a
    w_rev = w[::-1]
    conv_rev = convolution_fft(a, w_rev)       # length 3n‑2
    L = 2 * n - 1
    weighted_grad = 2.0 * conv_rev[L - 1 - np.arange(n)]

    s = np.sum(a)
    term1 = (2.0 * n / (s ** 2)) * weighted_grad
    term2 = -(4.0 * n * B / (s ** 3)) * np.ones_like(a)
    grad = term1 + term2
    return grad, B


# ----------------------------------------------------------------------
# Adam optimiser on the smooth surrogate
# ----------------------------------------------------------------------


def adam_optimize(initial_a: np.ndarray,
                  time_deadline: float,
                  max_iter: int = 100_000,
                  lr0: float = 0.2,
                  lam_start: float = 0.5,
                  lam_end: float = 200.0,
                  noise_every: int = 2000,
                  noise_scale: float = 5e-5) -> Tuple[np.ndarray, float]:
    """
    Projected Adam on the smooth max‑convolution surrogate.
    Returns the best vector (projected on the simplex) and its fast score.
    """
    a = project_to_simplex(initial_a.astype(float).copy())
    best_a = a.copy()
    best_sc = compute_score(a)

    beta1, beta2 = 0.9, 0.999
    eps = 1e-8
    m = np.zeros_like(a)
    v = np.zeros_like(a)

    n = len(a)

    for it in range(1, max_iter + 1):
        if time.time() >= time_deadline:
            break

        # exponential schedule for λ (more aggressive than linear)
        lam = lam_start * (lam_end / lam_start) ** (min(1.0, it / max_iter))

        grad, _ = smooth_gradient(a, lam)

        # Adam moments
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)

        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        step = lr0 * m_hat / (np.sqrt(v_hat) + eps)
        a_new = a - step

        # occasional jitter to avoid stagnation
        if noise_every and (it % noise_every == 0):
            a_new += np.random.randn(n) * noise_scale

        a_new = project_to_simplex(a_new)

        cur_sc = compute_score(a_new)
        if cur_sc < best_sc:
            best_sc = cur_sc
            best_a = a_new.copy()

        a = a_new

        # lightweight progress output
        if it % 5000 == 0:
            print(f"    Adam iter={it:,d}  cur={cur_sc:.6f}  best={best_sc:.6f}")

    return best_a, best_sc

            # only call the official (slow) evaluator if we see improvement
            if fast_val < best_val:
                official_val = evaluate_sequence(refined.tolist())
                print(f"     official score = {official_val:.6f}")
                if official_val < best_val:
                    best_val = official_val
                    best_seq = refined.tolist()
                    print(f"*** New global best = {best_val:.6f} (n={n}) ***")
            # keep the best refined shape even if the fast score is not yet beating
            if fast_val < best_val and not best_seq:
                best_seq = refined.tolist()
                best_val = fast_val

        # ------------------- 1d. quick stochastic annealing on the current best of this length ---
        if best_seq and len(best_seq) == n and time.time() < deadline - 5.0:
            a = np.array(best_seq, dtype=float)
            steps = 2000
            cur_best = a.copy()
            cur_sc = compute_score(cur_best)
            for _ in range(steps):
                if time.time() >= deadline:
                    break
                i, j = rng.integers(0, n, size=2)
                delta = (rng.random() - 0.5) * 0.02   # ±1 %
                if cur_best[i] - delta < 0 or cur_best[j] + delta < 0:
                    continue
                cand = cur_best.copy()
                cand[i] -= delta
                cand[j] += delta
                cand = project_to_simplex(cand)
                cand_sc = compute_score(cand)
                if cand_sc < cur_sc:
                    cur_sc, cur_best = cand_sc, cand
            if cur_sc < best_val:
                official_val = evaluate_sequence(cur_best.tolist())
                if official_val < best_val:
                    best_val = official_val
                    best_seq = cur_best.tolist()
                    print(f"*** Updated best after annealing = {best_val:.6f} ***")

    # --------------------------------------------------------------
    # 2.  Final polishing of the overall best vector
    # --------------------------------------------------------------
    if best_seq and time.time() < deadline - 5.0:
        a = np.array(best_seq, dtype=float)
        remaining = deadline - time.time()
        # allocate a generous number of iterations (≈40 per second)
        it_per_sec = 40_000 / (len(a) / 2048)
        max_iter = int(min(300_000, remaining * it_per_sec))
        print(f"\nFinal polishing (length={len(a)}) max_iter={max_iter}")
        a_refined, _ = adam_optimize(
            initial_a=a,
            time_deadline=deadline,
            max_iter=max_iter,
            lr0=0.15,
            lam_start=0.5,
            lam_end=300.0,
            noise_every=1000,
            noise_scale=1e-5,
        )
        final_val = evaluate_sequence(a_refined.tolist())
        print(f"Final polished official score = {final_val:.6f}")
        if final_val < best_val:
            best_val = final_val
            best_seq = a_refined.tolist()
            print("*** New global best after final polishing ***")

    # --------------------------------------------------------------
    # 3.  Guard – guarantee a return value
    # --------------------------------------------------------------
    if not best_seq:
        # safe deterministic fallback (uniform 1024‑vector)
        n0 = 1024
        best_seq = [1.0 / n0] * n0
        best_val = evaluate_sequence(best_seq)
        print("Fallback uniform used, score =", best_val)

    print(f"\nSearch finished. Best reported score = {best_val:.6f}")
    return best_seq
