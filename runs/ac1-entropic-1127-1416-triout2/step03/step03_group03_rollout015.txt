def propose_candidate(seed=..., budget_s=1000):
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 2

    best_seq=None
    best_val=inf

    # Parameter sweep
    lengths=[1024, 2048, 4096]
    exps=[0.4,0.45,0.48,0.5,0.52,0.55,0.58,0.6,0.65]
    epsilons=[0.05,0.1,0.2,0.25,0.35,0.5]

    for n in lengths:
        if time.time()>deadline - 10: break
        i = np.arange(n,float)
        for p in exps:
            for eps in epsilons:
                if time.time()>deadline - 10: break
                # compute values
                vals = ( (i+eps) * (n - eps - i) ) ** (-p)
                # avoid overflow
                if np.any(np.isinf(vals)) or np.any(np.isnan(vals)):
                    continue
                seq = vals / np.sum(vals)  # sum to 1 (simple normalization)
                score = compute_score(seq)
                if score < best_val:
                    best_val = score
                    best_seq = seq.tolist()
                    # optionally print
                    print(f"New best via param sweep: n={n}, p={p}, eps={eps} score={score:.6f}")
    # At this point we have a candidate.

    # Optimize via Adam further
    if best_seq is not None and time.time() < deadline - 5:
        print("\nRefining with projected Adam")
        # convert to ndarray
        a0 = np.array(best_seq, dtype=float)
        # allocate remaining iterations roughly proportional to time left
        remaining = deadline - time.time()
        # calibrate max_iter
        max_iter = int(min(500_000, remaining * 40))
        seq_adam, score_adam = adam_optimize(init_a=a0,
                                            time_deadline=deadline,
                                            max_iter=max_iter,
                                            eta=0.01,
                                            alpha_start=1.0,
                                            alpha_end=5000.0,
                                            noise_every=2000,
                                            noise_scale=1e-4)
        print(f"Adam refinement score: {score_adam:.6f}")
        if score_adam < best_val:
            best_val = score_adam
            best_seq = seq_adam.tolist()

    # final hill climbing
    if best_seq is not None and time.time() < deadline - 1:
        print("\nFinal hill climbing")
        refined_seq, refined_val = hill_climb_permutation(best_seq,
                                                          time_deadline=deadline,
                                                          max_steps=200_000)
        print(f"Hill climb refined score: {refined_val:.6f}")
        if refined_val < best_val:
            best_val = refined_val
            best_seq = refined_seq

    # fallback
    if best_seq is None:
        best_seq = [1.0/256]*256
        best_val = evaluate_sequence(best_seq)

    final_score = evaluate_sequence(best_seq)
    print(f"\nFinal score: {final_score:.6f}")
    return best_seq

def param_optimize(n, init_p=0.5, init_eps=0.25, time_deadline=..., max_iters=200):
    p = init_p
    eps = init_eps
    best_seq = arcsine_initial(n)  # or compute vect with current p, eps
    best_score = compute_score(best_seq)
    delta_p = 0.05
    delta_eps = 0.05
    for it in range(max_iters):
        improved = False
        for dp in [0, delta_p, -delta_p]:
            for de in [0, delta_eps, -delta_eps]:
                if dp == 0 and de == 0: continue
                p_new = p + dp
                eps_new = eps + de
                if p_new <= 0.1 or eps_new <= 0: continue

                if score < best_score:
                    best_score = score
                    best_seq = seq
                    p = p_new
                    eps = eps_new
                    improved = True
                    break
            if improved:
                break
        if not improved:
            delta_p *= 0.5
            delta_eps *= 0.5
            if delta_p < 1e-4 and delta_eps < 1e-4:

def param_search(n, time_deadline):
    best_score = float('inf')
    best_seq = None
    # Exploratory exponents and eps
    exponents = np.linspace(0.4, 0.7, 15) # 15 values
    epsilons = np.linspace(0.1, 0.8, 15) # 15 values
    for p in exponents:
        for eps in epsilons:
            if time.time() > time_deadline - 2:
                break
            i = np.arange(n, dtype=float)
            denom = (i+eps)*(n - eps - i)
            # ensure positive
            vals = np.power(denom, -p, where=denom>0, out=np.zeros_like(denom))
            # If any denominator zero, skip
            if np.any(vals==0):
                continue
            seq = vals / np.sum(vals)
            score = compute_score(seq)
            if score < best_score:
                best_score = score
                best_seq = seq
    return best_seq, best_score

def param_search(n, time_deadline):
    best_score = float('inf')
    best_seq = None
    exponents = np.linspace(0.35,0.7,15)
    epsilons = np.linspace(0.1,0.8,15)
    i = np.arange(n, dtype=float)
    for p in exponents:
        for eps in epsilons:
            if time.time() > time_deadline - 1:
                break
            denom = (i + eps) * (n - eps - i)
            # ensure >0
            vals = np.power(denom, -p)
            seq = vals / np.sum(vals)
            # remove any NaN/Inf
            if np.any(np.isnan(seq)) or np.any(np.isinf(seq)):
                continue
            score = compute_score(seq)
            if score < best_score:
                best_score = score
                best_seq = seq
    return best_seq, best_score
```

# ----------------------------------------------------------------------
# Utility functions
# ----------------------------------------------------------------------


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection onto the probability simplex (non‑negative, sum = 1)."""
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / v.shape[0])

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, v.shape[0] + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.where(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full convolution a*b via a real FFT (fast)."""
    m = len(a) + len(b) - 1
    n_fft = next_pow2(m)
    A = np.fft.rfft(a, n=n_fft)
    B = np.fft.rfft(b, n=n_fft)
    C = A * B
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


def compute_score(a: np.ndarray) -> float:
    """
    Fast replica of `evaluate_sequence` for vectors with sum = 1.
    Returns 2·n·max(a*a).
    """
    n = a.shape[0]
    conv = convolution_fft(a, a)
    return 2.0 * n * float(np.max(conv))


# ----------------------------------------------------------------------
# Smoothed‑max gradient (soft‑max approximation)
# ----------------------------------------------------------------------


def smoothed_gradient(a: np.ndarray, alpha: float):
    """
    Gradient of f_α(a) = (1/α)·log Σ_k exp(α·(a*a)_k).
    Returns (grad, raw convolution).
    """
    n = a.shape[0]
    conv = convolution_fft(a, a)

    # stable log‑sum‑exp
    maxc = np.max(conv)
    exp_term = np.exp(alpha * (conv - maxc))
    w = exp_term / np.sum(exp_term)           # length 2n‑1, Σw = 1

    a_rev = a[::-1]
    conv_rev = convolution_fft(a_rev, w)      # length 3n‑2
    start = n - 1
    rev_slice = conv_rev[start:start + n]     # size n

    # diagonal contributions (indices 2i)
    w_diag = w[::2][:n]

    grad = 2.0 * rev_slice - w_diag * a
    return grad, conv


def adam_optimize(
    init_a: np.ndarray,
    time_deadline: float,
    max_iter: int = 200_000,
    eta: float = 0.02,
    beta1: float = 0.9,
    beta2: float = 0.999,
    eps: float = 1e-8,
    alpha_start: float = 1.0,
    alpha_end: float = 5000.0,
    noise_every: int = 800,
    noise_scale: float = 2e-4,
):
    """
    Projected Adam optimiser on the smoothed max‑convolution objective.
    Returns the best vector found (ndarray) and its *true* score.
    """
    a = project_to_simplex(init_a.astype(float).copy())
    m = np.zeros_like(a)
    v = np.zeros_like(a)

    best_a = a.copy()
    best_score = compute_score(a)

    n = len(a)

    for t in range(1, max_iter + 1):
        if time.time() >= time_deadline:
            break

        # linear homotopy for α
        alpha = alpha_start + (alpha_end - alpha_start) * min(1.0, t / max_iter)

        grad, _ = smoothed_gradient(a, alpha)

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)

        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)

        step = eta * m_hat / (np.sqrt(v_hat) + eps)
        a_new = a - step

        # occasional isotropic noise
        if noise_every and (t % noise_every == 0):
            a_new += np.random.randn(n) * noise_scale

        a_new = project_to_simplex(a_new)

        cur_score = compute_score(a_new)
        if cur_score < best_score:
            best_score = cur_score
            best_a = a_new.copy()

        a = a_new

        if t % 5000 == 0:
            print(f"    Adam iter {t:,d} – cur {cur_score:.6f} – best {best_score:.6f}")

    return best_a, best_score


# ----------------------------------------------------------------------
# Pure permutation / mass‑transfer hill‑climbing
# ----------------------------------------------------------------------


def hill_climb_permutation(seq, time_deadline, max_steps=150_000):
    """
    Stochastic hill‑climb that swaps two entries or moves a small fraction
    of mass.  The true (unsmoothed) objective is used for acceptance.
    """
    a = np.array(seq, dtype=float)
    n = a.shape[0]

    best_a = a.copy()
    best_score = compute_score(best_a)

    step_scale = 0.1  # initial fraction of mass transferred in a move

    while max_steps > 0 and time.time() < time_deadline:
        move_type = np.random.rand()
        cand = best_a.copy()

        if move_type < 0.5:
            # simple swap
            i, j = np.random.randint(n, size=2)
            cand[i], cand[j] = cand[j], cand[i]
        else:
            # move a fraction from i to j
            i, j = np.random.randint(n, size=2)
            if i != j:
                delta = step_scale * best_a[i]
                if delta > 0:
                    cand[i] -= delta
                    cand[j] += delta
                    if cand[i] < 0:
                        cand[i] = 0.0

        cand = project_to_simplex(cand)

        cur_score = compute_score(cand)

        if cur_score < best_score:
            best_score = cur_score
            best_a = cand
            step_scale = min(step_scale * 1.05, 0.5)
        else:
            step_scale *= 0.97

        max_steps -= 1

    return best_a.tolist(), best_score


# ----------------------------------------------------------------------
# Parametric family scan (discretised Beta / arcsine)
# ----------------------------------------------------------------------


def parametric_scan(n: int, time_deadline: float):
    """
    Exhaustive scan over a two‑parameter family:
        a_i ∝ ((i+ε)*(n‑ε‑i))^{-p}
    Returns the best vector (ndarray) found and its score.
    """
    best_score = float("inf")
    best_seq = None

    i = np.arange(n, dtype=float)

    # reasonable ranges (adjustable if time permits)
    exponents = np.linspace(0.35, 0.70, 15)      # p
    epsilons = np.linspace(0.10, 0.80, 15)       # ε

    for p in exponents:
        for eps in epsilons:
            if time.time() >= time_deadline - 1.0:
                return best_seq, best_score
            denom = (i + eps) * (n - eps - i)
            # avoid division by zero – denom is always positive in our ranges
            vals = np.power(denom, -p)
            seq = vals / np.sum(vals)            # normalise to sum = 1
            # sanity check
            if np.any(np.isnan(seq)) or np.any(np.isinf(seq)):
                continue
            score = compute_score(seq)
            if score < best_score:
                best_score = score
                best_seq = seq.copy()
    return best_seq, best_score

# Main entry point required by the harness
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative sequence that minimises `evaluate_sequence`.
    The routine performs a cheap parametric scan, refines the best
    candidate with a projected Adam optimiser, and finally polishes it
    with a stochastic hill‑climber.
    """
    np.random.seed(seed)
    start_time = time.time()
    # keep a safety margin for the harness itself
    deadline = start_time + budget_s - 2.0

    best_seq = None
    best_val = float("inf")

        print(f"Fallback score = {fallback_val:.6f}")
        if fallback_val < best_val:
            best_val = fallback_val
            best_seq = fallback
    except NameError:
        pass

    # ------------------------------------------------------------------
    # Stage 1 – parametric family scan on several lengths
    # ------------------------------------------------------------------
    lengths = [2048, 4096, 8192]  # larger n can give a slightly better constant
    for n in lengths:
        if time.time() >= deadline - 20.0:
            break
        print(f"\n=== Parametric scan for n = {n} ===")
        seq, val = parametric_scan(n, deadline)
        if seq is not None and val < best_val:
            best_val = val
            best_seq = seq.tolist()
            print(f"    New best from scan: score = {best_val:.6f}")

    # ------------------------------------------------------------------
    # Stage 2 – Adam refinement (if time permits)
    # ------------------------------------------------------------------
    if best_seq is not None and time.time() < deadline - 10.0:
        print("\nRefining best candidate with projected Adam optimiser")
        a0 = np.array(best_seq, dtype=float)

        # allocate iterations based on remaining time
        remaining = deadline - time.time()
        max_iter = int(min(200_000, remaining * 30))   # rough budget

        seq_adam, val_adam = adam_optimize(
            init_a=a0,
            time_deadline=deadline,
            max_iter=max_iter,
            eta=0.01,
            alpha_start=1.0,
            alpha_end=8000.0,
            noise_every=2000,
            noise_scale=1e-4,
        )
        print(f"    Adam refinement score = {val_adam:.6f}")
        if val_adam < best_val:
            best_val = val_adam
            best_seq = seq_adam.tolist()

    # ------------------------------------------------------------------
    # Stage 3 – final hill‑climbing polishing
    # ------------------------------------------------------------------
    if best_seq is not None and time.time() < deadline - 2.0:
        print("\nFinal hill‑climbing refinement")
        refined_seq, refined_val = hill_climb_permutation(
            best_seq,
            time_deadline=deadline,
            max_steps=300_000,
        )
        print(f"    Hill‑climb refined score = {refined_val:.6f}")
        if refined_val < best_val:
            best_val = refined_val
            best_seq = refined_seq

    # ------------------------------------------------------------------
    # Safeguard: produce a trivial uniform vector if nothing succeeded
    # ------------------------------------------------------------------
    if best_seq is None:
        best_seq = [1.0 / 256] * 256
        best_val = evaluate_sequence(best_seq)

    final_score = evaluate_sequence(best_seq)
    print(f"\nFinished search. Final reported score = {final_score:.6f}")

    # The harness expects a plain Python list of floats.
    return best_seq
