"""
Hybrid random‑search + hard‑max exponentiated‑gradient descent.

The search proceeds in three stages:

1. **Broad sampling** –  A large collection of parametric families
   (uniform, linear / quadratic, power‑law, beta‑like, exponential,
   sparse random) is sampled for each chosen length `n`.  The fast
   internal score `2·n·max(conv)/(sum²)` (identical to the public
   evaluator up‑to clipping) is used to keep the `k` most promising
   candidates.

2. **Hard‑max mirror descent** –  Starting from each retained vector we
   perform exponentiated‑gradient updates based on the true
   sub‑gradient of the *max* convolution (the index where the convolution
   attains its maximum).  The step size follows a `η₀/√t` schedule,
   the iterate stays on the probability simplex via a multiplicative
   update and a normalisation step, and a tiny amount of Gaussian noise
   is added periodically to escape plateaus.

3. **Final polishing & evaluation** –  The current global best is
   refined once more with a longer schedule and then handed to the
   official `evaluate_sequence` function.  The routine respects the
   supplied time budget (default 1000 s) and always returns a feasible
   non‑negative list.
"""

import time
import numpy as np
from typing import List, Tuple

# ----------------------------------------------------------------------
# Low‑level numeric utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Linear convolution via real FFT, full length."""
    m = len(a) + len(b) - 1
    n_fft = next_pow2(m)
    A = np.fft.rfft(a, n=n_fft)
    B = np.fft.rfft(b, n=n_fft)
    C = A * B
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


def compute_score(a: np.ndarray) -> float:
    """
    Fast version of the official objective.
    The vector is assumed non‑negative; the function works for any shape.
    """
    n = len(a)
    conv = convolution_fft(a, a)
    max_val = float(np.max(conv))
    s = float(np.sum(a))
    if s < 1e-12:
        return float("inf")
    return 2.0 * n * max_val / (s * s)


def normalize(v: np.ndarray) -> np.ndarray:
    """Project a non‑negative vector onto the probability simplex."""
    total = v.sum()
    if total <= 0.0:
        return np.full_like(v, 1.0 / len(v))
    return v / total


# ----------------------------------------------------------------------
# Hard‑max exponentiated‑gradient descent
# ----------------------------------------------------------------------


def hardmax_optimize(
    init_a: np.ndarray,
    deadline: float,
    *,
    eta0: float = 0.8,
    max_iter: int = 200_000,
    noise_every: int = 1000,
    noise_scale: float = 1e-6,
) -> Tuple[np.ndarray, float]:
    """
    Sub‑gradient descent on the *max* of the self‑convolution.
    Returns the best vector seen and its (non‑smooth) score.
    """
    a = normalize(init_a.astype(float).copy())
    best = a.copy()
    best_sc = compute_score(a)

    n = len(a)
    it = 0
    while it < max_iter and time.time() < deadline:
        it += 1

        # 1) compute convolution and locate the active index
        conv = convolution_fft(a, a)
        k = int(np.argmax(conv))

        # 2) sub‑gradient of max(conv) w.r.t. a:
        #    g_i = 2 * a_{k-i}   if 0 ≤ k-i < n, else 0
        grad = np.zeros_like(a)
        low = max(0, k - (n - 1))
        high = min(n - 1, k)
        idx = np.arange(low, high + 1, dtype=int)
        jdx = k - idx
        grad[idx] = 2.0 * a[jdx]

        # 3) exponentiated‑gradient step
        eta = eta0 / np.sqrt(it)
        a = a * np.exp(-eta * grad)

        # occasional isotropic noise to avoid stagnation
        if noise_every and (it % noise_every == 0):
            a += np.random.randn(n) * noise_scale
            a = np.maximum(a, 0.0)

        # 4) re‑project onto the simplex
        a = normalize(a)

        cur_sc = compute_score(a)
        if cur_sc < best_sc:
            best_sc = cur_sc
            best = a.copy()

        # occasional progress report
        if it % 5000 == 0:
            print(
                f"    it={it:,d}  max_idx={k}  cur={cur_sc:.6f}  best={best_sc:.6f}"
            )
    return best, best_sc

# Candidate generation helpers

def arcsine_family(n: int, eps: float) -> np.ndarray:
    """Discrete arcsine‑like density with a small regularisation epsilon."""
    i = np.arange(n, dtype=float)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - 0.5 - i) + eps)
    return vals / vals.sum()


def powerlaw_family(n: int, p: float, eps: float) -> np.ndarray:
    """a_i ∝ ((i+ε)(n‑1‑i+ε))⁻ᵖ."""
    i = np.arange(n, dtype=float)
    base = ((i + eps) * (n - 1 - i + eps)) ** (-p)
    if base.sum() == 0.0:
        return np.full(n, 1.0 / n)
    return base / base.sum()


def beta_family(n: int, alpha: float, beta: float) -> np.ndarray:
    """Beta‑like shape a_i ∝ (i+½)^{α‑1}(n‑½‑i)^{β‑1}."""
    i = np.arange(n, dtype=float)
    base = (i + 0.5) ** (alpha - 1) * (n - 0.5 - i) ** (beta - 1)
    if base.sum() == 0.0:
        return np.full(n, 1.0 / n)
    return base / base.sum()


def exponential_family(n: int, rate: float) -> np.ndarray:
    """Geometric‑like decreasing shape a_i ∝ rate^{i} (0<rate<1)."""
    i = np.arange(n, dtype=float)
    vals = rate ** i
    return vals / vals.sum()


def linear_family(n: int, increasing: bool = True) -> np.ndarray:
    """Simple linear shape."""
    i = np.arange(1, n + 1, dtype=float)
    if not increasing:
        i = i[::-1]
    return i / i.sum()


def quadratic_family(n: int, increasing: bool = True) -> np.ndarray:
    """Quadratic (or reverse) shape."""
    i = np.arange(1, n + 1, dtype=float) ** 2
    if not increasing:
        i = i[::-1]
    return i / i.sum()


def sparse_random(n: int, k: int, rng: np.random.Generator) -> np.ndarray:
    """k‑sparse random vector (positive entries, normalised)."""
    pos = rng.choice(n, size=k, replace=False)
    vals = rng.random(k) + 1e-6
    v = np.zeros(n, dtype=float)
    v[pos] = vals
    return v / v.sum()

    """
    rng = np.random.default_rng(seed)
    start_time = time.time()

    # ------------------------------------------------------------------

            best_seq = fallback
    except NameError:

    # ------------------------------------------------------------------
    # 1.  Lengths to explore (powers of two – cheap FFTs).
    # ------------------------------------------------------------------
    lengths = [1024, 2048, 4096]

    for n in lengths:

        # --------------------------------------------------------------

        # --------------------------------------------------------------
        candidates: List[np.ndarray] = []

            rate = rng.uniform(0.3, 0.99)
