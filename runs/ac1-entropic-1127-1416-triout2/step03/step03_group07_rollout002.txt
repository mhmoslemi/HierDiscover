"""
Propose a step‑function (non‑negative vector) that minimizes

    2·n·max(a*a) / (sum(a)**2)

using a two‑phase heuristic:

1. **Fast mirror‑descent** (exponentiated‑gradient) with a Polyak step
   size.  This quickly drives the objective down from a random start or
   from the historic best (`height_sequence_1`).

2. **Simulated‑annealing refinement** that performs tiny mass‑transfer
   moves on the simplex and accepts uphill moves with a temperature
   schedule.  The refinement is able to escape the small local minima
   left by the gradient phase.

Both phases respect the constraints `a_i ≥ 0`, `sum(a)=1`.  The
objective is evaluated via FFT‑based convolution (identical to the
implementation used in the reference optimiser) which is very fast for
vectors of several thousand dimensions.

The function returns the best vector found before the time budget
expires.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities
# ----------------------------------------------------------------------


def _project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of a vector onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1, dtype=np.float64)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all entries are zero – return uniform distribution
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1] - 1
    theta = cssv[cond][-1] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def _max_and_subgradient(a: np.ndarray):
    """
    Compute the maximum entry of the (linear) autocorrelation a*a and a
    sub‑gradient (averaged over all active indices).

    Returns
    -------
    max_val : float
        The maximum value of the convolution.
    grad : np.ndarray, shape (n,)
        A sub‑gradient of the max‑convolution w.r.t. a.
    """
    n = a.shape[0]
    # convolution via FFT (linear, zero‑padded)
    fft = np.fft.rfft(a, n=2 * n - 1)
    conv = np.fft.irfft(fft * fft, n=2 * n - 1)

    max_val = float(conv.max())

    # tolerance for detecting active (max‑attaining) indices
    tol = max_val * 1e-12 + 1e-14
    active = np.where(conv >= max_val - tol)[0]

    grad = np.zeros_like(a)
    # accumulate contributions from each active convolution index
    for k in active:
        low = max(0, k - n + 1)
        high = min(n - 1, k)
        idx = np.arange(low, high + 1, dtype=int)
        partner = k - idx
        grad[idx] += 2.0 * a[partner]

    if active.size > 1:
        grad /= float(active.size)

    return max_val, grad


def _score_from_max(max_val: float, n: int) -> float:
    """Return the objective value 2·n·max_val (sum(a) assumed = 1)."""
    return 2.0 * n * max_val

# ----------------------------------------------------------------------

    evaluation function.  The routine runs for at most ``budget_s`` seconds
    and returns the best sequence found.

    Parameters
    ----------
    budget_s : int, optional
        Time budget in seconds.  Default is 1000.

    Returns
    -------
    list[float]
        The best vector (normalized to sum = 1) discovered.
    """

    # 0. Global settings and timing
    # ------------------------------------------------------------------
    deadline = time.time() + budget_s - 0.2   # tiny safety margin

    # ------------------------------------------------------------------
    # 1. Problem size and initial point
    # ------------------------------------------------------------------
    try:

    # ------------------------------------------------------------------
    restart = 0

        # initialise a point for this restart

        else:

        # ensure positivity and normalise to the simplex
        a = np.maximum(a, 1e-30)

        # cumulative (scaled) sub‑gradient used by the exponential mirror map
        g_cum = np.zeros_like(a)

        it = 0

            max_conv, grad = _max_and_subgradient(a)

                a_new[j] += delta

    else:
        best_a /= best_a_sum
