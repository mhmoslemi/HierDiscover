The code searches a probability‐vector (non‑negative numbers that sum to 1) which minimises  

\[
\text{score}=2\,n\;\max_k (a*a)_k\qquad
\bigl(\text{this is exactly the value returned by `evaluate_sequence` when the sum is 1}\bigr)
\]

The search consists of three stages:

1. **Parameter sweep** – a family of analytically‑motivated shapes is sampled.  
   * Symmetric Beta‑densities `Beta(p,p)` (the uniform case is `p=1`, the classical
     arcsine is `p=½`).  
   * Power‑law edge shapes `a_i ∝ (i+½)^{‑γ}(n‑i‑½)^{‑γ}` (the arcsine corresponds
     to `γ=½`).  
   Every candidate is projected onto the simplex and scored with a fast FFT
   based routine (`compute_score`). The best vector found so far is kept.

2. **Refinement by projected Adam** – the best vector from step 1 is taken as a
   starting point for a projected Adam optimiser on a smooth approximation of
   the max‑convolution (log‑sum‑exp).  The optimiser respects the simplex
   constraints and is stopped as soon as the time budget is exhausted.

3. **Final hill‑climbing** – a cheap stochastic hill‑climber that swaps two
   entries or transfers a small amount of mass between them is run on the
   refined vector.  Each move is accepted only if the true (non‑smoothed) score
   improves.

All work is bounded by the 1000 s budget; the routine constantly checks the
deadline and aborts early when necessary.  If nothing is found a trivial
uniform vector is returned as a safe fallback.

```python
"""
Propose a low‑score step‑function.

The search consists of:
  1) a quick sweep over two analytical families (symmetric Beta and
     power‑law edge shapes) to obtain a good initial vector,
  2) a projected Adam optimiser on a smooth (log‑sum‑exp) approximation of
     the max‑convolution,
  3) a lightweight permutation / mass‑transfer hill‑climber that evaluates the
     true objective.

All steps respect the 1000 s time budget and keep the vector on the
probability simplex (non‑negative, sum = 1).  The routine finally returns a
plain Python list of floats ready for `evaluate_sequence`.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Basic linear‑algebra / signal‑processing utilities
# ----------------------------------------------------------------------


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / v.shape[0])

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, v.shape[0] + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.where(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full (linear) convolution using a real FFT."""
    m = len(a) + len(b) - 1
    n_fft = next_pow2(m)
    A = np.fft.rfft(a, n=n_fft)
    B = np.fft.rfft(b, n=n_fft)
    C = A * B
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


def compute_score(a: np.ndarray) -> float:
    """
    Fast replica of `evaluate_sequence` when sum(a) == 1.
    Returns 2 * n * max(a * a).
    """
    n = a.shape[0]
    conv = convolution_fft(a, a)
    return 2.0 * n * float(np.max(conv))


# ----------------------------------------------------------------------
# Families of analytic start vectors
# ----------------------------------------------------------------------


def beta_distribution_vector(n: int, alpha: float, beta: float) -> np.ndarray:
    """Discretised symmetric Beta(alpha,beta) density, projected onto simplex."""
    x = (np.arange(n, dtype=float) + 0.5) / n  # mid‑point rule
    # Guard against overflow when alpha<1 or beta<1
    vals = np.power(x, alpha - 1.0) * np.power(1.0 - x, beta - 1.0)
    vals = np.nan_to_num(vals, nan=0.0, posinf=1e300, neginf=0.0)
    return project_to_simplex(vals)


def powerlaw_edge_vector(n: int, exponent: float) -> np.ndarray:
    """
    a_i ∝ ((i+½)*(n‑i‑½))^{-exponent}; arcsine corresponds to exponent = 0.5.
    """
    i = np.arange(n, dtype=float)
    eps = 1e-12
    vals = 1.0 / ((i + 0.5) * (n - 0.5 - i) + eps) ** exponent
    return project_to_simplex(vals)


# ----------------------------------------------------------------------
# Smoothed‑max gradient (log‑sum‑exp) and projected Adam optimiser
# ----------------------------------------------------------------------


def smoothed_gradient(a: np.ndarray, alpha: float):
    """
    Gradient of the soft‑max (log‑sum‑exp) surrogate of max(a*a).

    Returns (grad, raw_convolution).  `grad` has the same shape as `a`.
    """
    n = a.shape[0]
    conv = convolution_fft(a, a)                      # length 2n‑1
    maxc = np.max(conv)
    exp_term = np.exp(alpha * (conv - maxc))
    w = exp_term / np.sum(exp_term)                  # soft‑max weights (sum = 1)

    # convolution of reversed a with the weight vector
    a_rev = a[::-1]
    conv_rev = convolution_fft(a_rev, w)            # length 3n‑2
    start = n - 1
    rev_slice = conv_rev[start:start + n]           # needed part

    # diagonal contribution (elements w_{2i})
    w_diag = w[::2][:n]

    grad = 2.0 * rev_slice - w_diag * a
    return grad, conv


def adam_optimize(
    init_a: np.ndarray,
    time_deadline: float,
    max_iter: int = 200_000,
    eta: float = 0.02,
    beta1: float = 0.9,
    beta2: float = 0.999,
    eps: float = 1e-8,
    alpha_start: float = 1.0,
    alpha_end: float = 2000.0,
    noise_every: int = 800,
    noise_scale: float = 2e-4,
):
    """
    Projected Adam optimiser on the smoothed objective.
    Returns the best vector found (as ndarray) and its *true* score.
    """
    a = project_to_simplex(init_a.astype(float).copy())
    m = np.zeros_like(a)
    v = np.zeros_like(a)

    best_a = a.copy()
    best_score = compute_score(a)

    n = len(a)

    for t in range(1, max_iter + 1):
        if time.time() >= time_deadline:
            break

        # homotopy on the smoothing parameter
        alpha = alpha_start + (alpha_end - alpha_start) * min(1.0, t / max_iter)

        grad, _ = smoothed_gradient(a, alpha)

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)

        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)

        step = eta * m_hat / (np.sqrt(v_hat) + eps)
        a_new = a - step

        # occasional isotropic noise to escape shallow plateaus
        if noise_every and (t % noise_every == 0):
            a_new += np.random.randn(n) * noise_scale

        a_new = project_to_simplex(a_new)

        cur_score = compute_score(a_new)
        if cur_score < best_score:
            best_score = cur_score
            best_a = a_new.copy()

        a = a_new

    return best_a, best_score


# ----------------------------------------------------------------------
# Pure (true‑objective) hill‑climber
# ----------------------------------------------------------------------


def hill_climb_permutation(seq, time_deadline, max_steps=100_000):
    """
    Simple stochastic hill‑climber.
    Moves: either swap two entries or move a small fraction of mass from i to j.
    Acceptance is based on the *exact* score (no smoothing).
    """
    a = np.array(seq, dtype=float)
    n = a.shape[0]

    best_a = a.copy()
    best_score = compute_score(best_a)

    step_scale = 0.1  # initial transferred fraction

    while max_steps > 0 and time.time() < time_deadline:
        move_type = np.random.rand()
        cand = best_a.copy()

        if move_type < 0.5:
            i, j = np.random.randint(n, size=2)
            cand[i], cand[j] = cand[j], cand[i]
        else:
            i, j = np.random.randint(n, size=2)
            if i == j:
                continue
            delta = step_scale * best_a[i]
            if delta <= 0.0:
                continue
            cand[i] -= delta
            cand[j] += delta
            # Guard against tiny negatives
            if cand[i] < 0.0:
                cand[i] = 0.0

        cand = project_to_simplex(cand)

        cur_score = compute_score(cand)
        if cur_score < best_score:
            best_score = cur_score
            best_a = cand
            step_scale = min(step_scale * 1.05, 0.5)   # increase step size on success
        else:
            step_scale *= 0.97                         # shrink on failure

        max_steps -= 1

    return best_a.tolist(), best_score


# ----------------------------------------------------------------------
# Upscaling helper (interpolation from a coarse grid to a finer one)
# ----------------------------------------------------------------------


def upscale_sequence(seq: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation of a simplex‑vector to a larger dimension."""
    old_n = len(seq)
    # mid‑point coordinates in [0,1)
    x_old = (np.arange(old_n, dtype=float) + 0.5) / old_n
    x_new = (np.arange(new_n, dtype=float) + 0.5) / new_n
    new_seq = np.interp(x_new, x_old, seq, left=0.0, right=0.0)
    return project_to_simplex(new_seq)


# ----------------------------------------------------------------------
# Main entry point required by the evaluation harness
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step‑function that minimises `evaluate_sequence`.
    The routine works under a 1000 s budget and returns a plain Python list.
    """
    np.random.seed(seed)
    start_time = time.time()
    # leave a small safety margin for the harness
    deadline = start_time + budget_s - 2.0

    best_seq = None
    best_val = float("inf")

    # ------------------------------------------------------------------
    # Optional fallback if a previously discovered sequence exists
    # ------------------------------------------------------------------
    try:
        fallback = [float(x) for x in height_sequence_1]
        fallback_val = evaluate_sequence(fallback)
        print(f"Fallback score = {fallback_val:.6f}")
        if fallback_val < best_val:
            best_val = fallback_val
            best_seq = fallback
    except NameError:
        pass

    # ------------------------------------------------------------------
    # 1) PARAMETER SWEEP – Beta and power‑law families (coarse stage)
    # ------------------------------------------------------------------
    n_base = 2048
    print("\n=== Parameter sweep (n =", n_base, ") ===")

    # Store the best parameters for later up‑scaling (useful for reporting)
    best_params = None

    # ----- symmetric Beta(p,p) -----
    for p in np.linspace(0.30, 0.90, 31):          # 31 values from 0.30 to 0.90
        if time.time() >= deadline - 5.0:
            break
        a = beta_distribution_vector(n_base, p, p)
        sc = compute_score(a)
        if sc < best_val:
            best_val = sc
            best_seq = a.tolist()
            best_params = ("beta", p, p)
            print(f"  New best (Beta) p={p:.3f}  score={sc:.6f}")

    # ----- power‑law edge shape -----
    for exp in np.linspace(0.20, 1.00, 33):        # 33 values from 0.20 to 1.00
        if time.time() >= deadline - 5.0:
            break
        a = powerlaw_edge_vector(n_base, exp)
        sc = compute_score(a)
        if sc < best_val:
            best_val = sc
            best_seq = a.tolist()
            best_params = ("powerlaw", exp)
            print(f"  New best (Power) exp={exp:.3f}  score={sc:.6f}")

    # ----- few random Dirichlet starts (just in case) -----
    for _ in range(15):
        if time.time() >= deadline - 5.0:
            break
        a = np.random.dirichlet(np.full(n_base, 0.7))
        sc = compute_score(a)
        if sc < best_val:
            best_val = sc
            best_seq = a.tolist()
            best_params = ("dirichlet",)
            print(f"  New best (Dirichlet) score={sc:.6f}")

    # ------------------------------------------------------------------
    # 2) REFINEMENT BY PROJECTED ADAM (still on the base size)
    # ------------------------------------------------------------------
    if best_seq is not None and time.time() < deadline - 5.0:
        print("\nRefining with projected Adam (base size)")
        # we allow a fairly generous number of iterations – the optimiser
        # will stop early if the deadline is near.
        seq_arr = np.asarray(best_seq, dtype=float)
        refined_seq, refined_score = adam_optimize(
            init_a=seq_arr,
            time_deadline=deadline,
            max_iter=150_000,
            eta=0.02,
            alpha_start=1.0,
            alpha_end=3000.0,
            noise_every=1000,
            noise_scale=1e-4,
        )
        print(f"  Adam refinement score = {refined_score:.6f}")
        if refined_score < best_val:
            best_val = refined_score
            best_seq = refined_seq.tolist()

    # ------------------------------------------------------------------
    # 3) OPTIONAL UPSCALING (if time permits)
    # ------------------------------------------------------------------
    # Larger n reduces discretisation error; we try one upscale step.
    if time.time() < deadline - 30.0:
        # pick a larger length; 8192 is a good compromise
        n_large = 8192
        print("\nUpscaling to larger grid (n =", n_large, ")")
        upscaled = upscale_sequence(np.asarray(best_seq, dtype=float), n_large)

        # short Adam pass on the upscaled vector
        upscaled_ref, upscaled_score = adam_optimize(
            init_a=upscaled,
            time_deadline=deadline,
            max_iter=80_000,
            eta=0.015,
            alpha_start=1.0,
            alpha_end=2500.0,
            noise_every=1200,
            noise_scale=1e-4,
        )
        print(f"  Upscaled Adam score = {upscaled_score:.6f}")
        if upscaled_score < best_val:
            best_val = upscaled_score
            best_seq = upscaled_ref.tolist()
            # keep a reference to the larger size for the final hill‑climb
            n_base = n_large

    # ------------------------------------------------------------------
    # 4) FINAL HILL‑CLIMB ON THE CURRENT BEST VECTOR
    # ------------------------------------------------------------------
    if best_seq is not None and time.time() < deadline - 2.0:
        print("\nFinal hill‑climbing refinement")
        max_hc_steps = 120_000 if len(best_seq) <= 4096 else 60_000
        refined_seq, refined_score = hill_climb_permutation(
            best_seq,
            time_deadline=deadline,
            max_steps=max_hc_steps,
        )
        print(f"  Hill‑climb score = {refined_score:.6f}")
        if refined_score < best_val:
            best_val = refined_score
            best_seq = refined_seq

    # ------------------------------------------------------------------
    # SAFEGUARD: fall back to a uniform vector if nothing was produced
    # ------------------------------------------------------------------
    if best_seq is None:
        best_seq = [1.0 / 256] * 256
        best_val = evaluate_sequence(best_seq)

    final_score = evaluate_sequence(best_seq)
    print(f"\nSearch finished. Final reported score = {final_score:.6f}")
    if best_params is not None:
        print(f"Best parameters: {best_params}")

    # The harness expects a plain Python list of floats.
    return best_seq
```
