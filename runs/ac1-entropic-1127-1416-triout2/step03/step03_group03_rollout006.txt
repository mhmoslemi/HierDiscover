def perm_sa(a, deadline, max_steps=200000) -> best_seq, best_score:
    cur = a.copy()
    cur_score = compute_score(cur)
    best = cur.copy()
    best_score = cur_score

    n = len(a)

    # optional: compute contributions? Not needed.

    # parameters
    start_T = 0.1 # maybe choose scaling relative to cur_score
    end_T = 1e-6
    # compute exponential decay per step
    total_steps_allowed = max_steps
    # but adjust based on time
    step=0
    while step < max_steps and time.time() < deadline:
        # Compute temperature
        t = step / max_steps
        T = start_T * ((end_T/start_T) ** t)  # exponential cooling
        # propose move: swap or mass transfer
        move_type = np.random.rand()
        cand = cur.copy()
        if move_type < 0.6:
            # swap
            i, j = np.random.randint(n, size=2)
            if i != j:
                cand[i], cand[j] = cand[j], cand[i]
        else:
            # mass transfer
            i, j = np.random.randint(n, size=2)
            if i != j and cand[i] > 0:
                # move a random fraction up to some fraction of cand[i]
                frac = np.random.rand() * 0.1  # up to 10%
                delta = cand[i] * frac
                cand[i] -= delta
                cand[j] += delta
                # minimal negative guard
                if cand[i] < 0:
                    cand[i] = 0.0
        # No need to reproject: total sum remains 1 because we transferred mass; but swap also okay.
        # However due to floating errors we may want to renorm:
        # Maybe renorm every few steps.
        if step % 5000 == 0:
            # renorm to sum 1
            cand = cand / np.sum(cand)
        # compute new score
        cand_score = compute_score(cand)
        delta_score = cand_score - cur_score
        if delta_score < 0 or np.random.rand() < np.exp(-delta_score / (T + 1e-12)):
            # accept
            cur = cand
            cur_score = cand_score
            if cur_score < best_score:
                best = cur.copy()
                best_score = cur_score
        step += 1

    return best.tolist(), best_score

"""
Search for a non‑negative step‑function that minimizes
`evaluate_sequence`.  The algorithm works as follows:

1.  Sample a distribution of the required length (arcsine profile or a
    Dirichlet draw) and normalise it to a probability simplex.
2.  Randomly permute the entries – this changes the self‑convolution
    because large values are no longer aligned.
3.  Apply a simulated‑annealing / Metropolis hill‑climber that repeatedly
    swaps two positions or moves a fraction of mass from one index to
    another.  The true (unsmoothed) objective `2·n·max(a*a)` is used for
    acceptance, therefore the search directly optimises the target
    bound.
4.  The whole process is repeated for several lengths (up to 8192) and
    for several random starts until the 1000 s budget expires.
   The best sequence found is returned.

All helper functions are defined at top level, there are no closures,
and the code respects the time budget.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Basic utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full linear convolution a * b using real FFT."""
    m = len(a) + len(b) - 1
    n_fft = next_pow2(m)
    A = np.fft.rfft(a, n=n_fft)
    B = np.fft.rfft(b, n=n_fft)
    C = A * B
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the probability simplex
    (non‑negative entries that sum to 1).
    """
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / v.shape[0])

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, v.shape[0] + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.where(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def compute_score(a: np.ndarray) -> float:
    """
    Fast replica of `evaluate_sequence` under the assumption Σa = 1.
    Returns 2·n·max(a*a).
    """
    n = a.shape[0]
    conv = convolution_fft(a, a)
    return 2.0 * n * float(np.max(conv))


def arcsine_initial(n: int) -> np.ndarray:
    """
    Classical arcsine start:
        a_i ∝ 1 / sqrt( (i+0.5)*(n-0.5-i) )
    The vector is projected onto the simplex afterwards.
    """
    i = np.arange(n, dtype=float)
    eps = 0.25
    vals = 1.0 / np.sqrt((i + 0.5) * (n - 0.5 - i) + eps)
    return project_to_simplex(vals)


# ----------------------------------------------------------------------
# Simulated‑annealing permutation / mass‑transfer hill‑climber
# ----------------------------------------------------------------------


def simulated_annealing_permutation(init_a: np.ndarray, deadline: float) -> tuple[list[float], float]:
    """
    Metropolis search that repeatedly swaps two entries or moves a small
    fraction of mass.  The true objective (max convolution) is used for
    acceptance; a simple linear cooling schedule is employed.
    """
    cur = init_a.copy()
    n = cur.shape[0]
    cur_score = compute_score(cur)
    best = cur.copy()
    best_score = cur_score

    # cooling schedule parameters (relative to the current score)
    T_start = 0.05 * cur_score  # about 5 % of the score
    T_end = 1e-5 * cur_score

    # We do not know in advance how many steps we can afford,
    # so we loop until the deadline is hit.
    step = 0
    # estimate a safe upper bound to avoid infinite loops
    max_steps_est = 500_000

    while step < max_steps_est and time.time() < deadline:
        # linear interpolation between T_start and T_end
        t = step / max_steps_est
        T = (1 - t) * T_start + t * T_end

        # ----- propose a neighbour -----
        cand = cur.copy()
        if np.random.rand() < 0.6:
            # simple swap
            i, j = np.random.randint(n, size=2)
            if i != j:
                cand[i], cand[j] = cand[j], cand[i]
        else:
            # move a fraction of mass from i to j
            i, j = np.random.randint(n, size=2)
            if i != j and cand[i] > 0.0:
                # random fraction up to 20 % of the source mass
                frac = np.random.rand() * 0.2
                delta = cand[i] * frac
                cand[i] -= delta
                cand[j] += delta
                # protect against tiny negative rounding errors
                if cand[i] < 0.0:
                    cand[i] = 0.0

        # occasional renormalisation to keep Σa = 1 numerically
        if step % 2000 == 0:

            if s != 0.0:

        delta = cand_score - cur_score

        # Metropolis acceptance
        if delta < 0.0 or np.random.rand() < np.exp(-delta / (T + 1e-12)):
            cur = cand
            cur_score = cand_score
            # keep the best ever seen
            if cur_score < best_score:
                best = cur.copy()
                best_score = cur_score
                print(f"    SA step {step:,d} – new best {best_score:.6f}")

        step += 1

    return best.tolist(), best_score
