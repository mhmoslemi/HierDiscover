"""
Improved search for a low‑value step‑function.

The objective is
    E(a) = 2·n·max(a∗a) / (sum(a)²)

The algorithm is a multi‑phase exploration:

1.  *Broad parametric sweep* over several lengths.
    Families: Beta‑shapes (symmetric & asymmetric), exponential,
    arcsine and random Dirichlet draws.  Every candidate is projected
    onto the probability simplex (∑aᵢ = 1) – the scale‑invariance of the
    objective makes this a natural normalisation.

2.  *Upsampling* of the currently best vector to larger lengths
    (8192 → 16384) by linear interpolation – a finer discretisation
    often lowers the maximal convolution.

3.  *Projected sub‑gradient descent* on the true (non‑smoothed) loss.
    At each iteration the gradient of the active (max) convolution entry
    is computed analytically (∂max/∂aᵢ = 2·a_{k‑i}) and a diminishing
    step is taken, followed by a Euclidean projection onto the simplex.

4.  *Directed hill‑climbing*: the index k where the convolution attains
    its maximum is identified, the contribution vector
    wᵢ = a_{k‑i} is formed, and a small amount of mass is transferred
    from a large‑w index to a small‑w index.  This greedy move is
    accepted only if it reduces the true objective.

5.  A final cheap stochastic hill‑climb (the original implementation)
    cleans up any remaining noise.

The whole routine respects a user‑supplied time budget (default 1000 s)
and constantly prints progress.  It returns the best list of floats
found, ready for `evaluate_sequence`.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Basic linear‑algebra utilities
# ----------------------------------------------------------------------


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    if np.all(v <= 0):
        n = v.shape[0]
        return np.full_like(v, 1.0 / n, dtype=float)

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho = np.where(u - cssv / np.arange(1, v.shape[0] + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full convolution using a real FFT (fast)."""
    m = len(a) + len(b) - 1
    n_fft = next_pow2(m)
    A = np.fft.rfft(a, n=n_fft)
    B = np.fft.rfft(b, n=n_fft)
    C = A * B
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


def compute_score(a: np.ndarray) -> float:
    """Identical to `evaluate_sequence` when ∑a = 1."""
    a = np.asarray(a, dtype=float)
    n = len(a)
    conv = convolution_fft(a, a)
    max_val = float(np.max(conv))
    s = float(np.sum(a))
    if s < 1e-12:
        return float("inf")
    return 2.0 * n * max_val / (s * s)


# ----------------------------------------------------------------------
# Families of candidate step‑functions
# ----------------------------------------------------------------------


def arcsine_initial(n: int) -> np.ndarray:
    """Arcsine‑shaped start: aᵢ ∝ 1 / √((i+½)(n‑½‑i))."""
    i = np.arange(n, dtype=float)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - 0.5 - i))
    vals = np.clip(vals, 0.0, 1e12)
    return vals / np.sum(vals)


def beta_sequence(n: int, alpha: float, beta: float | None = None) -> np.ndarray:
    """Beta‑type coefficients aᵢ ∝ (i+½)^α·(n‑½‑i)^β."""
    if beta is None:
        beta = alpha
    i = np.arange(n, dtype=float)
    left = np.power(i + 0.5, alpha)
    right = np.power(n - 0.5 - i, beta)
    seq = left * right
    seq = np.clip(seq, 0.0, 1e12)
    total = np.sum(seq)
    if total == 0.0:
        seq = np.full(n, 1.0 / n)
    else:
        seq = seq / total
    return seq


def exponential_sequence(n: int, lam: float) -> np.ndarray:
    """aᵢ ∝ exp(−λ·|i−(n−1)/2|)."""
    i = np.arange(n, dtype=float)
    centre = (n - 1) / 2.0
    seq = np.exp(-lam * np.abs(i - centre))
    return seq / np.sum(seq)


def upsample_sequence(seq: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation of a probability vector to a larger size."""
    old_n = len(seq)
    old_x = (np.arange(old_n) + 0.5) / old_n
    new_x = (np.arange(new_n) + 0.5) / new_n
    new_vals = np.interp(new_x, old_x, seq)
    return new_vals / np.sum(new_vals)


# ----------------------------------------------------------------------
# Projected sub‑gradient descent on the true objective
# ----------------------------------------------------------------------


def subgradient_descent(
    a0: np.ndarray,
    deadline: float,
    max_iter: int = 50_000,
    step0: float = 0.9,
) -> tuple[list[float], float]:
    """Gradient descent on  E(a)=2·n·max(a∗a)/(∑a)²  (projected on simplex)."""
    a = project_to_simplex(a0.astype(float).copy())
    best_a = a.copy()
    best_score = compute_score(a)
    n = len(a)

    for t in range(1, max_iter + 1):
        if time.time() >= deadline:
            break

        conv = convolution_fft(a, a)
        max_val = float(np.max(conv))
        k = int(np.argmax(conv))                # index of the active max entry

        # gradient of the active convolution term: ∂c_k/∂a_i = 2·a_{k‑i} (if defined)
        idx = np.arange(n)
        partner = k - idx
        mask = (partner >= 0) & (partner < n)
        grad = np.zeros_like(a)
        grad[mask] = 2.0 * a[partner[mask]]

        s = np.sum(a)
        # derivative of E = (2n / s²)·c_k - (4n·c_k / s³)·∑a_i
        dE = (2.0 * n / (s ** 2)) * grad - (4.0 * n * max_val / (s ** 3)) * np.ones_like(a)

        # diminishing step size
        step = step0 / np.sqrt(t)
        a_new = a - step * dE
        a_new = project_to_simplex(a_new)

        cur_score = compute_score(a_new)
        if cur_score < best_score:
            best_score = cur_score
            best_a = a_new.copy()

        a = a_new

        # occasional diagnostics
        if t % 5000 == 0:
            print(f"    subgrad iter {t:,d}  curScore {cur_score:.6f}  best {best_score:.6f}")

    return best_a.tolist(), best_score


# ----------------------------------------------------------------------
# Directed hill‑climbing using the active convolution index
# ----------------------------------------------------------------------


def directed_hill_climb(
    seq,
    deadline: float,
    max_moves: int = 20_000,
    init_frac: float = 0.07,
) -> tuple[list[float], float]:
    """Move mass from indices that contribute most to the current max
    convolution towards those that contribute least."""
    a = np.array(seq, dtype=float)
    a = project_to_simplex(a)
    best_a = a.copy()
    best_score = compute_score(best_a)
    n = len(a)

    frac = init_frac

    while max_moves > 0 and time.time() < deadline:
        conv = convolution_fft(a, a)
        max_val = float(np.max(conv))
        k = int(np.argmax(conv))

        # contribution of each coordinate to the active entry:
        idx = np.arange(n)
        partner = k - idx
        mask = (partner >= 0) & (partner < n)
        contrib = np.zeros_like(a)
        contrib[mask] = a[partner[mask]]

        # pick donor (large contribution) and receiver (small contribution)
        donor = int(np.argmax(contrib))
        receiver = int(np.argmin(contrib))

        if donor == receiver or a[donor] <= 0.0:
            # fallback to a random pair
            i, j = np.random.randint(n, size=2)
            donor, receiver = i, j

        delta = frac * a[donor]          # move a fraction of the donor mass
        if delta <= 0.0:
            break

        cand = a.copy()
        cand[donor] -= delta
        cand[receiver] += delta
        cand = project_to_simplex(cand)

        cur_score = compute_score(cand)
        if cur_score < best_score:
            best_score = cur_score
            best_a = cand.copy()
            # be a little more aggressive when we succeed
            frac = min(frac * 1.07, 0.5)
        else:
            # become more conservative on failures
            frac *= 0.96

        a = cand
        max_moves -= 1

    return best_a.tolist(), best_score


# ----------------------------------------------------------------------
# Stochastic hill‑climbing (the original cheap version)
# ----------------------------------------------------------------------


def hill_climb(seq, time_deadline: float, max_steps: int = 30_000, init_step: float = 0.07):
    a = np.array(seq, dtype=float)
    a = project_to_simplex(a)
    best_a = a.copy()
    best_score = compute_score(best_a)
    n = len(a)
    step = init_step

    while max_steps > 0 and time.time() < time_deadline:
        i, j = np.random.randint(n, size=2)
        if i == j:
            continue
        delta = step * best_a[i]
        if delta <= 0.0:
            continue
        cand = best_a.copy()
        cand[i] -= delta
        cand[j] += delta
        cand = project_to_simplex(cand)
        cur_score = compute_score(cand)
        if cur_score < best_score:
            best_score = cur_score
            best_a = cand
            step = min(step * 1.07, 0.5)
        else:
            step *= 0.96
        max_steps -= 1
    return best_a.tolist(), best_score
