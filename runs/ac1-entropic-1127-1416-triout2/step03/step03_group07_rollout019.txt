def propose_candidate(budget_s: int = 1000) -> list[float]:
    """
    Simulated‑annealing + Adam optimiser for the step‑function objective
    2·n·max(a*a)/(sum(a)**2).

    The algorithm maintains a log‑probability vector p (softmax gives a),
    updates it with Adam using the exact sub‑gradient of the max‑convolution,
    and periodically restarts from the historic best or a random point.
    Random jitter is added every few thousand steps to avoid getting stuck.
    The best sequence seen before the deadline is returned.
    """
    import time, numpy as np

    deadline = time.time() + budget_s - 0.5   # safety margin

    # ------------------------------------------------------------------
    # Determine problem size (n) and obtain an initial shape.
    # ------------------------------------------------------------------
    try:
        # if previous best is provided, reuse its length as baseline
        init_seq = np.asarray(height_sequence_1, dtype=float)
        base_n = init_seq.shape[0]
    except Exception:
        init_seq = None
        base_n = 2048

    # Choose a length that balances speed / quality.
    # We'll aim for ~8000 if we have enough time left.
    # Use a simple heuristic: if we have >200s we use larger n.
    remaining = deadline - time.time()
    if remaining > 200:
        n = max(base_n, 8192)
    else:
        n = max(base_n, 4096)

    # Helper to resample a sequence to length n.
    def resample(seq, new_n):
        seq = np.asarray(seq, dtype=float)
        m = seq.shape[0]
        if m == new_n:
            return seq.copy()
        # linear interpolation on the index interval [0, 1]
        x_old = np.linspace(0.0, 1.0, m)
        x_new = np.linspace(0.0, 1.0, new_n)
        new_seq = np.interp(x_new, x_old, seq)
        new_seq[new_seq < 0] = 0.0
        s = new_seq.sum()
        if s > 0:
            new_seq /= s
        else:
            new_seq = np.full(new_n, 1.0 / new_n)
        return new_seq

    # Prepare a function that given a probability vector a returns the
    # objective (assuming sum(a)=1).
    def score_of(a):
        # a is a 1‑D ndarray, sum(a)≈1
        L = 2 * n
        # convolution via FFT (real)
        A = np.fft.rfft(a, n=2 * n - 1)
        conv = np.fft.irfft(A * A, n=2 * n - 1)
        max_val = conv.max()
        return float(L * max_val)

    # ------------------------------------------------------------------
    # Optimisation primitives
    # ------------------------------------------------------------------
    # Adam hyper‑parameters (can be tweaked inside the loop)
    beta1 = 0.9
    beta2 = 0.999
    eps_adam = 1e-8

    # Variables to keep track of the overall best.
    best_a = None
    best_score = float("inf")

    # ------------------------------------------------------------------
    # Main search – repeated restarts until the deadline.
    # ------------------------------------------------------------------
    restart = 0
    rng = np.random.default_rng()
    while time.time() < deadline:
        # initialise a: use historic best at first restart, otherwise random or
        # resampled version of init_seq.
        if restart == 0 and init_seq is not None:
            a = resample(init_seq, n)
        else:
            # random Dirichlet
            a = rng.random(n) + 1e-6
            a = a / a.sum()
        # ensure strictly positive to avoid log(0)
        a = np.maximum(a, 1e-12)

        # logits
        p = np.log(a)

        # Adam state
        m = np.zeros_like(p)
        v = np.zeros_like(p)
        t = 0

        # learning‑rate schedule – start relatively high and decay.
        lr0 = 0.5
        # we will adapt lr depending on iteration count

        it = 0
        while time.time() < deadline:
            # compute convolution and max
            A = np.fft.rfft(a, n=2 * n - 1)
            conv = np.fft.irfft(A * A, n=2 * n - 1)
            max_val = float(conv.max())
            cur_score = float(2 * n * max_val)   # sum(a)≈1

            # keep global best
            if cur_score < best_score:
                best_score = cur_score
                best_a = a.copy()
                print(f"[{time.time():.1f}] restart {restart}, iter {it}: new best = {best_score:.6f}")

                if best_score < 1.503:
                    print("Target reached – early exit.")
                    return best_a.tolist()

            # sub‑gradient of the max‑convolution w.r.t. a
            tol = max_val * 1e-12 + 1e-14
            active = np.where(conv >= max_val - tol)[0]
            grad_a = np.zeros_like(a)
            for k in active:
                lo = max(0, k - n + 1)
                hi = min(n - 1, k)
                idx = np.arange(lo, hi + 1, dtype=int)
                partner = k - idx
                grad_a[idx] += 2.0 * a[partner]
            if active.size > 1:
                grad_a /= float(active.size)

            # transform gradient to log‑it space (softmax Jacobian)
            dot = np.dot(a, grad_a)
            grad_p = a * (grad_a - dot)

            # Adam update
            t += 1
            m = beta1 * m + (1.0 - beta1) * grad_p
            v = beta2 * v + (1.0 - beta2) * (grad_p ** 2)
            m_hat = m / (1.0 - beta1 ** t)
            v_hat = v / (1.0 - beta2 ** t)

            # learning‑rate decay (simple 1/√it)
            lr = lr0 / np.sqrt(t)

            step = lr * m_hat / (np.sqrt(v_hat) + eps_adam)
            p = p - step

            # optional clipping of logits to keep numbers stable
            if np.max(p) - np.min(p) > 50:
                p = p - np.max(p) + 25

            # compute new a via softmax
            a = np.exp(p - np.max(p))
            s = a.sum()
            a = a / s

            # occasional jitter to escape plateaus
            if it % 5000 == 0 and it > 0:
                jitter = rng.normal(scale=0.001, size=n)
                a = np.maximum(a + jitter, 0.0)
                a = a / a.sum()
                # re‑initialise logits after jitter
                p = np.log(a + 1e-30)

            it += 1

        restart += 1

    # ------------------------------------------------------------------
    # Return the best sequence obtained.
    # ------------------------------------------------------------------
    if best_a is None:
        # fallback: uniform distribution
        best_a = np.full(n, 1.0 / n)
    # Ensure the returned list respects the evaluate_sequence safety checks.
    best_a = np.maximum(best_a, 0.0)
    best_a = np.minimum(best_a, 1000.0)
    # renormalise so that sum is not too small (not mandatory for evaluate_sequence)
    total = best_a.sum()
    if total < 1e-8:
        best_a = np.full_like(best_a, 1.0 / len(best_a))
    else:
        best_a = best_a / total   # keep sum≈1, does not affect the objective

    return best_a.tolist()

"""
A hybrid optimisation for the step‑function objective

The code combines three ideas:

1. A quick sweep over simple parametric families (triangular, power‑law,
   sinusoidal, the classic “square‑root” series) to obtain a decent
   starting point.
2. An Adam‑based mirror‑descent on the simplex (log‑it representation)
   using an exact sub‑gradient of the max‑convolution.
3. An optional linear‑programming refinement (cvxpy) that, given the
   current distribution f, finds a distribution g of maximal mass
   such that the convolution f∗g stays below the current bound.
   The two distributions are mixed and the process repeats.

The routine respects the 1000 s budget, logs progress and finally
returns the best sequence discovered.
"""

import time
import numpy as np

# optional import – if missing the LP phase is skipped
try:
    import cvxpy as cp
    _has_cvxpy = True
except Exception:
    _has_cvxpy = False


def _conv_max_score(a: np.ndarray) -> float:
    """Return 2·n·max(a*a) for a (assumed normalized to sum≈1)."""
    n = a.shape[0]
    A = np.fft.rfft(a, n=2 * n - 1)
    conv = np.fft.irfft(A * A, n=2 * n - 1)
    return float(2.0 * n * conv.max())


def _subgrad_max_conv(a: np.ndarray) -> np.ndarray:
    """
    Exact sub‑gradient of max(conv) w.r.t. a.
    Returns a vector of length n.
    """
    n = a.shape[0]
    A = np.fft.rfft(a, n=2 * n - 1)
    conv = np.fft.irfft(A * A, n=2 * n - 1)
    max_val = conv.max()
    tol = max_val * 1e-12 + 1e-14
    active = np.where(conv >= max_val - tol)[0]

    grad = np.zeros_like(a)
    for k in active:
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        idx = np.arange(lo, hi + 1, dtype=int)
        partner = k - idx
        grad[idx] += 2.0 * a[partner]
    if active.size > 1:
        grad /= float(active.size)
    return grad


def _softmax(x: np.ndarray) -> np.ndarray:
    """Stable soft‑max that returns a probability vector."""
    y = np.exp(x - np.max(x))
    s = y.sum()
    if s == 0.0:
        return np.full_like(x, 1.0 / x.shape[0])
    return y / s


def _parametric_candidates(n: int) -> list[np.ndarray]:
    """Generate a small collection of hand‑crafted candidates."""
    cand = []

    # uniform
    cand.append(np.full(n, 1.0 / n))

    # triangular (mountain shape)
    tri = np.arange(1, n // 2 + 1, dtype=float)
    if n % 2 == 0:
        tri = np.concatenate([tri, tri[::-1]])
    else:
        tri = np.concatenate([tri, [n // 2 + 1], tri[::-1]])
    cand.append(tri / tri.sum())

    # power‑law symmetric: a_i ∝ (i+1)^p * (n-i)^p
    for p in (-2.0, -1.5, -1.0, -0.5, 0.5, 1.0, 1.5, 2.0):
        i = np.arange(n, dtype=float) + 1.0
        a = (i ** p) * ((n - i + 1) ** p)
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s > 0:
            cand.append(a / s)

    # sine‑based families
    xs = np.linspace(0.0, np.pi, n, endpoint=False) + np.pi / (2 * n)
    sin_vals = np.sin(xs)
    for p in (0.5, 1.0, 2.0, 3.0):
        a = sin_vals ** p
        cand.append(a / a.sum())

    # the classic “square‑root” series (coefficients of (1‑z)^‑½)
    sqrt_series = np.empty(n, dtype=float)
    sqrt_series[0] = 1.0
    for i in range(1, n):
        sqrt_series[i] = sqrt_series[i - 1] * (2 * i - 1) / (2.0 * i)
    cand.append(sqrt_series / sqrt_series.sum())

    # 1/√(i)‑style, symmetric
    i = np.arange(1, n + 1, dtype=float)
    a = (1.0 / np.sqrt(i)) * (1.0 / np.sqrt(n - i + 1))
    cand.append(a / a.sum())

    return cand


def _adam_optimize(
    a0: np.ndarray,
    deadline: float,
    best_sofar: tuple[float, np.ndarray] | None = None,
) -> tuple[float, np.ndarray]:
    """
    Adam mirror‑descent on the simplex.
    Returns (best_score, best_a) found before the deadline.
    """
    n = a0.shape[0]
    # initialise logits
    p = np.log(np.maximum(a0, 1e-12))

    # Adam state
    m = np.zeros_like(p)
    v = np.zeros_like(p)
    beta1 = 0.9
    beta2 = 0.999
    eps_adam = 1e-8
    lr0 = 0.8

    best_score, best_a = (float("inf"), None)
    if best_sofar is not None:
        best_score, best_a = best_sofar

    it = 0
    while time.time() < deadline:
        a = _softmax(p)
        cur_score = _conv_max_score(a)

        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time():.1f}] Adam iter {it}: new best = {best_score:.6f}")

            if best_score < 1.503:
                # early exit
                return best_score, best_a

        # sub‑gradient of the max‑convolution
        grad_a = _subgrad_max_conv(a)

        # transform to logits gradient (soft‑max Jacobian)
        dot = np.dot(a, grad_a)
        grad_p = a * (grad_a - dot)

        # Adam update
        it += 1
        m = beta1 * m + (1.0 - beta1) * grad_p
        v = beta2 * v + (1.0 - beta2) * (grad_p ** 2)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        step = (lr0 / np.sqrt(it)) * m_hat / (np.sqrt(v_hat) + eps_adam)

        p = p - step

        # keep logits numerically well‑behaved
        if np.max(p) - np.min(p) > 50.0:
            p = p - np.max(p) + 25.0

        # occasional jitter to avoid stagnation
        if it % 5000 == 0:
            rng = np.random.default_rng()
            jitter = rng.normal(scale=0.001, size=n)
            a = _softmax(p) + jitter
            a = np.maximum(a, 0.0)
            a /= a.sum()
            p = np.log(a + 1e-30)

    return best_score, best_a


def _lp_refine(a: np.ndarray, M: float, n: int):
    """
    One LP refinement step: given current distribution a and its
    maximum convolution value M, solve
        max   sum(g)
        s.t.  a * g ≤ M   (componentwise)
              g ≥ 0
    The solution g is returned (may be all zeros if infeasible).
    """
    if not _has_cvxpy:
        return None

    g = cp.Variable(n, nonneg=True)
    a_param = cp.Parameter(n, value=a, constant=True)
    conv_ag = cp.conv(a_param, g)  # length 2n‑1
    constraints = [conv_ag <= M]
    prob = cp.Problem(cp.Maximize(cp.sum(g)), constraints)
    try:
        prob.solve(solver=cp.ECOS, warm_start=True, max_iters=2000)
    except Exception:
        return None
    if g.value is None:
        return None
    g_val = np.maximum(g.value, 0.0)
    total = g_val.sum()
    if total <= 0:

    # normalise to sum = 1
    return g_val / total

"""
Hybrid optimiser for the step‑function objective

Objective:
    2·n·max(a*a) / (sum(a)**2)   (scale‑invariant, sum(a) must be >0)

The routine runs for at most ``budget_s`` seconds and returns the best
non‑negative sequence found.  It combines three ideas:

1. **Fast parametric sweep** – a handful of analytically‑motivated families
   (uniform, triangular, power‑law, sinusoidal, the classic
   (1‑z)^‑½ series, symmetric 1/√‑law) are evaluated and the best of them
   is taken as a warm start.

2. **Adam mirror‑descent on the simplex** – we work with log‑it variables
   (soft‑max parametrisation) and use the exact sub‑gradient of the
   max‑convolution.  Adam’s adaptive step‑size together with occasional
   jitter gives a robust local optimisation.

3. **Optional LP refinement** – using ``cvxpy`` we solve a linear programme
   that, given the current distribution *f*, finds a distribution *g*
   of maximal total mass while keeping the convolution *f∗g* below the
   current bound.  Mixing *f* and *g* can lower the bound further.

All intermediate progress is printed; the final sequence respects the
security checks of ``evaluate_sequence`` (non‑negative, bounded, non‑zero
sum).
"""

import time
import numpy as np

# optional import – if missing the LP phase is simply skipped
try:
    import cvxpy as cp

    _HAS_CVXPY = True
except Exception:  # pragma: no cover
    _HAS_CVXPY = False


def _conv_max_score(a: np.ndarray) -> float:
    """Return 2·n·max(a*a) (sum(a) is assumed ≈1)."""
    n = a.shape[0]
    A = np.fft.rfft(a, n=2 * n - 1)
    conv = np.fft.irfft(A * A, n=2 * n - 1)
    return float(2.0 * n * conv.max())


def _subgrad_max_conv(a: np.ndarray) -> np.ndarray:
    """
    Exact sub‑gradient of max(conv) w.r.t. a.
    Returns a vector of length n.
    """
    n = a.shape[0]
    A = np.fft.rfft(a, n=2 * n - 1)
    conv = np.fft.irfft(A * A, n=2 * n - 1)
    max_val = conv.max()
    tol = max_val * 1e-12 + 1e-14
    active = np.where(conv >= max_val - tol)[0]

    grad = np.zeros_like(a)
    for k in active:
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        idx = np.arange(lo, hi + 1, dtype=int)
        partner = k - idx
        grad[idx] += 2.0 * a[partner]
    if active.size > 1:
        grad /= float(active.size)
    return grad


def _softmax(x: np.ndarray) -> np.ndarray:
    """Stable soft‑max -> probability vector."""
    y = np.exp(x - np.max(x))
    s = y.sum()
    if s == 0.0:
        return np.full_like(x, 1.0 / x.shape[0])
    return y / s


def _parametric_candidates(n: int) -> list[np.ndarray]:
    """Hand‑crafted families of candidate sequences (not exhaustive)."""
    cand = []

    # 1. uniform
    cand.append(np.full(n, 1.0 / n))

    # 2. symmetric triangular (mountain) shape
    t = np.arange(1, n // 2 + 1, dtype=float)
    if n % 2 == 0:
        tri = np.concatenate([t, t[::-1]])
    else:
        tri = np.concatenate([t, [n // 2 + 1], t[::-1]])
    cand.append(tri / tri.sum())

    # 3. symmetric power‑law a_i ∝ (i+1)^p·(n‑i)^p
    for p in (-2.0, -1.5, -1.0, -0.5, 0.5, 1.0, 1.5, 2.0):
        i = np.arange(n, dtype=float) + 1.0
        a = (i ** p) * ((n - i + 1) ** p)
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s > 0:
            cand.append(a / s)

    # 4. sine‑based families: a_i ∝ sin(π·(i+0.5)/n)^p
    xs = np.linspace(0.0, np.pi, n, endpoint=False) + np.pi / (2 * n)
    sin_vals = np.sin(xs)
    for p in (0.5, 1.0, 2.0, 3.0):
        a = sin_vals ** p
        cand.append(a / a.sum())

    # 5. classic (1‑z)^‑½ series  (coefficients of √(1‑z)^‑1)
    sqrt_series = np.empty(n, dtype=float)
    sqrt_series[0] = 1.0
    for i in range(1, n):
        sqrt_series[i] = sqrt_series[i - 1] * (2.0 * i - 1) / (2.0 * i)
    cand.append(sqrt_series / sqrt_series.sum())

    # 6. symmetric 1/√(i)·1/√(n‑i+1) shape
    i = np.arange(1, n + 1, dtype=float)
    a = (1.0 / np.sqrt(i)) * (1.0 / np.sqrt(n - i + 1))
    cand.append(a / a.sum())

    return cand


def _adam_optimize(
    a0: np.ndarray,
    deadline: float,
    best_sofar: tuple[float, np.ndarray] | None = None,
) -> tuple[float, np.ndarray]:
    """
    Adam mirror‑descent on the simplex, starting from ``a0``.
    Returns (best_score, best_a) found before ``deadline``.
    """
    n = a0.shape[0]
    # log‑it variables (soft‑max will give a)
    p = np.log(np.maximum(a0, 1e-12))

    # Adam state
    m = np.zeros_like(p)
    v = np.zeros_like(p)
    beta1 = 0.9
    beta2 = 0.999
    eps_adam = 1e-8
    lr0 = 0.8

    best_score, best_a = (float("inf"), None)
    if best_sofar is not None:
        best_score, best_a = best_sofar

    it = 0
    while time.time() < deadline:
        a = _softmax(p)
        cur_score = _conv_max_score(a)

        if cur_score < best_score:
            best_score, best_a = cur_score, a.copy()
            print(f"[{time.time():.1f}] Adam iter {it}: new best = {best_score:.6f}")

            if best_score < 1.503:
                return best_score, best_a

        # sub‑gradient of the max‑convolution
        grad_a = _subgrad_max_conv(a)

        # map to log‑it space (Jacobian of soft‑max)
        dot = np.dot(a, grad_a)
        grad_p = a * (grad_a - dot)

        # Adam update (with a simple 1/√t decay)
        it += 1
        m = beta1 * m + (1.0 - beta1) * grad_p
        v = beta2 * v + (1.0 - beta2) * (grad_p ** 2)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        step = (lr0 / np.sqrt(it)) * m_hat / (np.sqrt(v_hat) + eps_adam)
        p = p - step

        # keep the logits numerically well‑scaled
        if np.max(p) - np.min(p) > 50.0:
            p = p - np.max(p) + 25.0

        # jitter every few thousand iterations to escape plateaus
        if it % 5000 == 0:
            rng = np.random.default_rng()
            jitter = rng.normal(scale=0.001, size=n)
            a = _softmax(p) + jitter
            a = np.maximum(a, 0.0)
            a /= a.sum()
            p = np.log(a + 1e-30)

    return best_score, best_a


def _lp_refine(a: np.ndarray, M: float, n: int) -> np.ndarray | None:
    """
    One CVXPY refinement step:
        maximise   sum(g)
        s.t.       a ∗ g ≤ M   (component‑wise)
                   g ≥ 0
    Returns a normalised feasible vector (or ``None`` on failure).
    """
    if not _HAS_CVXPY:
        return None

    g = cp.Variable(n, nonneg=True)
    a_param = cp.Parameter(n, value=a, constant=True)
    conv_ag = cp.conv(a_param, g)          # length 2·n‑1
    prob = cp.Problem(cp.Maximize(cp.sum(g)), [conv_ag <= M])
    try:
        prob.solve(solver=cp.ECOS, warm_start=True, max_iters=2000)
    except Exception:                     # pragma: no cover
        return None
    if g.value is None:
        return None
    g_val = np.maximum(g.value, 0.0)
    total = g_val.sum()
    if total <= 0.0:
        return None
    return g_val / total   # normalise to sum = 1


def propose_candidate(budget_s: int = 1000) -> list[float]:
    """
    Search for a sequence of non‑negative heights that minimises
    2·n·max(a*a)/(sum(a)**2).

    The routine respects the time budget and returns the best sequence
    encountered.
    """
    deadline = time.time() + budget_s - 0.5     # safety margin

    # --------------------------------------------------------------
    # Problem size (n) and optional historic starting point.
    # --------------------------------------------------------------
    try:
        init_seq = np.asarray(height_sequence_1, dtype=float)  # type: ignore[name-defined]
        base_n = init_seq.shape[0]
    except Exception:                          # pragma: no cover
        init_seq = None
        base_n = 2048

    # Choose a length that balances speed and quality.
    # Values around 3000–4000 work well under the 1000 s budget.
    n = max(2048, min(base_n, 4096))

    # --------------------------------------------------------------
    # Phase 1 – cheap sweep over simple families.
    # --------------------------------------------------------------
    best_a = None
    best_score = float("inf")

    for cand in _parametric_candidates(n):
        if time.time() >= deadline:
            break
        cand = np.maximum(cand, 0.0)
        s = cand.sum()
        if s == 0.0:
            continue
        cand = cand / s
        sc = _conv_max_score(cand)
        if sc < best_score:
            best_score, best_a = sc, cand.copy()
            print(f"[{time.time():.1f}] parametric candidate: score = {sc:.6f}")

    # incorporate the historic construction if available
    if init_seq is not None:
        if init_seq.shape[0] != n:
            # linear interpolation to the new length
            x_old = np.linspace(0.0, 1.0, init_seq.shape[0])
            x_new = np.linspace(0.0, 1.0, n)
            init_seq = np.interp(x_new, x_old, init_seq)
        init_seq = np.maximum(init_seq, 0.0)
        s = init_seq.sum()
        if s > 0.0:
            init_seq = init_seq / s
            sc = _conv_max_score(init_seq)
            if sc < best_score:
                best_score, best_a = sc, init_seq.copy()
                print(f"[{time.time():.1f}] historic sequence: score = {sc:.6f}")

    # --------------------------------------------------------------
    # Phase 2 – Adam mirror‑descent (multiple restarts).
    # --------------------------------------------------------------
    remaining = deadline - time.time()
    restarts = 3
    per_restart = max(0.0, remaining / restarts)

    for r in range(restarts):
        if time.time() >= deadline:
            break
        # start either from the current best or from a fresh random point
        if best_a is not None and r == 0:
            a0 = best_a.copy()
        else:
            rng = np.random.default_rng()
            a0 = rng.random(n) + 1e-9
            a0 = a0 / a0.sum()
        sub_deadline = time.time() + per_restart
        sub_best, sub_vec = _adam_optimize(a0, sub_deadline, (best_score, best_a))
        if sub_best < best_score:
            best_score, best_a = sub_best, sub_vec.copy()

        # ----------------------------------------------------------
        # Optional LP refinement (only one per restart to keep time low)
        # ----------------------------------------------------------
        if _HAS_CVXPY and time.time() < deadline:
            # current bound M = max(conv) = best_score / (2·n)
            M = best_score / (2.0 * n)
            refined = _lp_refine(best_a, M, n)
            if refined is not None:
                mixed = 0.5 * best_a + 0.5 * refined
                mixed = mixed / mixed.sum()
                mixed_score = _conv_max_score(mixed)
                print(f"[{time.time():.1f}] LP refinement: mixed score = {mixed_score:.6f}")
                if mixed_score < best_score:
                    best_score, best_a = mixed_score, mixed.copy()

    # --------------------------------------------------------------
    # Phase 3 – opportunistic random sampling (if time permits).
    # --------------------------------------------------------------
    rng = np.random.default_rng()
    while time.time() < deadline:
        a = rng.random(n) + 1e-9
        a = a / a.sum()
        sc = _conv_max_score(a)
        if sc < best_score:

    # --------------------------------------------------------------

    best_a = np.minimum(best_a, 1000.0)         # respect evaluate_sequence limits
    total = best_a.sum()
    if total < 1e-8:
        best_a = np.full_like(best_a, 1.0 / n)
    else:
        best_a = best_a / total                  # normalise (objective is scale‑free)

    print(f"[{time.time():.1f}] finished – best score = {best_score:.6f}")
    return best_a.tolist()
