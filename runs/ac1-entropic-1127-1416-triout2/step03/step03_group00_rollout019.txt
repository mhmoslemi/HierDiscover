def propose_candidate(seed=0, budget_s=1000):
    np.random.seed(seed)
    start = time.time()
    deadline = start + budget_s - 0.5

    # Choose n from existing sequence else set default (maybe 5146 as before)
    try:
        init = np.array(height_sequence_1, dtype=float)
        n = init.shape[0]
        print("Using existing length")
    except NameError:
        n = 5146
        init = np.random.rand(n)  # random start

    # Normalize to sum 1
    a0 = init / init.sum()

    # Parameter: number of restarts we can attempt
    # We'll run log-spaced restarts within time
    best_score = np.inf
    best_a = None

    # We'll define a function to run optimization from an initial a
    def run_opt(a_init):
        nonlocal best_score, best_a

        # Convert to logits for exponentiated gradient
        # Avoid zeros: use small epsilon
        eps = 1e-12
        u = np.log(a_init + eps)  # initial logits
        # Adam parameters
        beta1 = 0.9
        beta2 = 0.999
        epsadam = 1e-8
        mt = np.zeros_like(u)
        vt = np.zeros_like(u)
        t = 0

        while time.time() < deadline:
            # compute a via softmax
            # to improve numerical stability subtract max
            u_max = np.max(u)
            exp_u = np.exp(u - u_max)
            a = exp_u / exp_u.sum()

            # compute convolution via FFT (length 2*n-1)
            fft_a = np.fft.rfft(a, n=2 * n - 1)
            conv = np.fft.irfft(fft_a * fft_a, n=2 * n - 1)
            max_b = conv.max()
            cur_score = 2.0 * n * max_b

            # record best
            if cur_score < best_score:
                best_score = cur_score
                best_a = a.copy()
                print(f"New global best {best_score:.6f}")

            # subgradient for max (if at local optimum maybe all same)
            eps_active = max_b * 1e-8
            active = np.where(conv >= max_b - eps_active)[0]  # indices

            # compute gradient w.r.t a
            g = np.zeros_like(a)
            for k in active:
                low = max(0, k - n + 1)
                high = min(n - 1, k)
                idx = np.arange(low, high + 1, dtype=int)
                partner = k - idx
                g[idx] += 2.0 * a[partner]
            # Optionally divide by active size (average)
            # We'll use average to reduce magnitude
            g = g / max(1, active.size)

            # gradient w.r.t logits (mirror)
            # g_a = derivative wrt a, now grad w.r.t u:
            dot_ag = np.dot(a, g)
            grad_u = a * (g - dot_ag)  # shape n

            # Adam update
            t += 1
            mt = beta1 * mt + (1 - beta1) * grad_u
            vt = beta2 * vt + (1 - beta2) * (grad_u ** 2)
            m_hat = mt / (1 - beta1 ** t)
            v_hat = vt / (1 - beta2 ** t)
            step = 0.1 / (np.sqrt(v_hat) + epsadam)  # base step 0.1

            u = u - step * m_hat

            # periodically add random noise to escape plateaus
            if t % 50000 == 0:
                u += np.random.normal(scale=0.01, size=n)

        # after time expired, exit

    # orchestrate multiple restarts
    n_restarts = 5
    for i in range(n_restarts):
        if time.time() >= deadline:
            break
        # Choose random start shape
        if i == 0 and 'height_sequence_1' in globals():
            a_start = height_sequence_1 / np.sum(height_sequence_1)
        else:
            # random Beta parameters or random dirichlet
            alpha = np.random.uniform(0.3, 5.0)
            beta = np.random.uniform(0.3, 5.0)
            a_start = beta_sequence(alpha, beta, n)
        print(f"Restart {i+1}/{n_restarts} starting.")
        run_opt(a_start)

    # after exhausted time, return best sequence
    if best_a is None:
        # fallback uniform
        best_a = np.full(n, 1.0 / n)
    # ensure nonnegative sum >0
    best_a = best_a / best_a.sum()
    return [float(v) for v in best_a.tolist()]

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Helper functions
# ----------------------------------------------------------------------


def beta_sequence(alpha: float, beta: float, n: int) -> np.ndarray:
    """
    Returns a discretised Beta(α,β) density on n equally‑spaced points,
    normalised to sum to one.  The midpoint rule avoids the singular
    endpoints, and the log‑space formulation prevents overflow.
    """
    x = (np.arange(n) + 0.5) / n
    logw = (alpha - 1.0) * np.log(x) + (beta - 1.0) * np.log(1.0 - x)
    logw = np.clip(logw, -700.0, 700.0)          # avoid overflow in exp
    w = np.exp(logw)
    s = w.sum()
    if not np.isfinite(s) or s == 0.0:
        w = np.ones_like(w)
        s = w.sum()
    return w / s


def _conv_self(a: np.ndarray) -> np.ndarray:
    """
    Full linear self‑convolution a*a using FFT.
    The length of the result is 2·n‑1.
    """
    n = a.shape[0]
    fft = np.fft.rfft(a, n=2 * n - 1)
    return np.fft.irfft(fft * fft, n=2 * n - 1)


def _subgradient(a: np.ndarray, conv: np.ndarray) -> np.ndarray:
    """
    Sub‑gradient of  max_k (a*a)[k]  with respect to a.
    Uses every index attaining (approximately) the maximum.
    """
    max_b = conv.max()
    eps = max_b * 1e-8                      # tolerance for “active” set
    active = np.where(conv >= max_b - eps)[0]
    n = a.shape[0]
    g = np.zeros_like(a)
    for k in active:
        low = max(0, k - n + 1)
        high = min(n - 1, k)
        idx = np.arange(low, high + 1, dtype=int)
        partner = k - idx
        g[idx] += 2.0 * a[partner]          # derivative contribution
    # Any convex combination of the active gradients is a valid sub‑gradient.
    # Averaging keeps the magnitude moderate.
    g /= float(max(1, active.size))
    return g

# Main optimisation routine
# ----------------------------------------------------------------------

    best_score = np.inf
