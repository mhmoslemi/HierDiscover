v_hat = v_t / (1 - beta2**t)
update = step_size * m_hat / (np.sqrt(v_hat) + epsilon)

"""
Gradient‑descent with smoothed max (log‑sum‑exp) + Adam + simplex projection,
followed by a short simulated‑annealing refinement.

The algorithm works directly on the probability simplex (sum = 1).  It
optimises the smooth approximation

    F_λ(a) = (1/λ)·log( Σ_k exp( λ·(a*a)_k ) )

which converges to the true objective 2·n·max(a*a) as λ→∞.
Adam provides an adaptive step size, while a projection onto the
simplex guarantees feasibility.  After the gradient phase a classic
mass‑transfer SA run fine‑tunes the best solution found.
"""

import time
import math
import numpy as np

def simplex_projection(v: np.ndarray) -> np.ndarray:
    """Project v onto the probability simplex {w≥0, sum w = 1}."""
    v = np.asarray(v, dtype=float)
    n = v.shape[0]
    # sort descending
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1] - 1
    theta = cssv[cond][-1] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w

def compute_score(a: np.ndarray) -> float:
    """Exact evaluation of evaluate_sequence when sum(a)==1."""
    n = a.shape[0]
    fft = np.fft.rfft(a, n=2 * n - 1)
    conv = np.fft.irfft(fft * fft, n=2 * n - 1)
    return 2.0 * n * conv.max()

def beta_sequence(alpha: float, beta: float, n: int) -> np.ndarray:
    """Discrete Beta(α,β) density on n points, normalised to sum 1."""
    x = (np.arange(n) + 0.5) / n
    logw = (alpha - 1.0) * np.log(x) + (beta - 1.0) * np.log(1.0 - x)
    logw = np.clip(logw, -700.0, 700.0)
    w = np.exp(logw)
    if not np.isfinite(w.sum()) or w.sum() == 0.0:
        w = np.ones_like(w)
    return w / w.sum()

        print(f"No existing sequence – using a symmetric beta start, length {n}.")

    a = init / init.sum()          # probability vector
    best_a = a.copy()
    best_score = compute_score(best_a)
    print(f"Initial score = {best_score:.6f}")

    # -----------------------------------------------------------------
    # Pre‑compute FFT length (next power of two bigger than 3n)
    # -----------------------------------------------------------------
    L = 1 << ((3 * n - 2) - 1).bit_length()   # >= 3n-2
    # -----------------------------------------------------------------

    v = np.zeros_like(a)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    lr = 0.25               # base learning rate
    t = 0

    # -----------------------------------------------------------------
    # Smoothing parameter schedule for the log‑sum‑exp
    # -----------------------------------------------------------------
    lam = 0.5               # initial λ
    lam_factor = 1.001      # multiplicative increase each iteration
    lam_max = 400.0

    # -----------------------------------------------------------------
    # Main smoothed‑max gradient descent loop
    # -----------------------------------------------------------------
    while time.time() < deadline - 30.0:   # reserve ~30 s for SA
        # forward FFT of a (for conv and for reversed version)
        Fa = np.fft.rfft(a, n=L)

        # ----- convolution a*a (full length 2n‑1) -----
        conv_full = np.fft.irfft(Fa * Fa, n=L)
        # keep only the first 2n‑1 relevant entries
        conv = conv_full[: 2 * n - 1]
        max_conv = conv.max()
        cur_score = 2.0 * n * max_conv

        # record best
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"Iter {t:6d}: new best {best_score:.6f}")

        # ----- compute softmax weights over convolution entries -----
        M = max_conv
        # stability: shift by M
        if lam < 1e-12:
            w = np.full_like(conv, 1.0 / conv.size)
        else:

            exp_arg = lam * (conv - M)

            w = np.exp(exp_arg)

            else:

        lam = min(lam * lam_factor, lam_max)

            jitter = np.random.randn(n) * 0.001
            a = simplex_projection(a + jitter)

    # -----------------------------------------------------------------

        best_a = cur_a.copy()
        print(f"SA start: improved best to {best_score:.6f}")

    T_start = cur_score * 0.01
    T_end = cur_score * 1e-5
    it_sa = 0

        # exponential cooling
        T = max(T_end, T_start * (0.9997 ** it_sa))

        # --- propose a mass‑transfer move (conserves sum) ---
        src = np.random.choice(n, p=cur_a)                 # more mass → higher prob.
        tgt = np.random.randint(n - 1)
        if tgt >= src:
            tgt += 1
        if cur_a[src] == 0.0:
            continue
        delta = cur_a[src] * np.random.rand()   # move a random fraction
        cand = cur_a.copy()
        cand[src] -= delta
        cand[tgt] += delta

        cand_score = compute_score(cand)

        # Metropolis acceptance
        if cand_score < cur_score:
            accept = True
        else:
            accept_prob = math.exp(-(cand_score - cur_score) / max(T, 1e-12))
            accept = np.random.rand() < accept_prob

        if accept:
            cur_a = cand
            cur_score = cand_score
            if cand_score < best_score:
                best_score = cand_score
                best_a = cand.copy()
                print(f"SA iter {it_sa:6d}: new best {best_score:.6f}")

        # occasional large restart to diversify

                                  np.random.uniform(0.5, 5.0), n)
            cur_score = compute_score(cur_a)

        it_sa += 1

    print(f"Simulated‑annealing finished after {it_sa} iterations; final best {best_score:.6f}")

    # -----------------------------------------------------------------
    # Return best sequence (already normalised to sum == 1)
    # -----------------------------------------------------------------
    best_a = best_a / best_a.sum()
    return [float(v) for v in best_a.tolist()]

    t = 0                # iteration counter

    # Smoothing parameter for log‑sum‑exp.
    lam = 0.5            # start small (smooth)
    lam_factor = 1.001   # multiplicative increase per iteration
    lam_max = 400.0

    # --------------------------------------------------------------
    # 3. Smoothed‑max gradient descent (Adam) – main loop
    # --------------------------------------------------------------
    while time.time() < deadline - 30.0:               # keep ~30 s for SA
        # ------- convolution a*a (full length 2n‑1) -------
        Fa = np.fft.rfft(a, n=L)
        conv_full = np.fft.irfft(Fa * Fa, n=L)
        conv = conv_full[: 2 * n - 1]                 # keep only the needed part
        max_conv = conv.max()
        cur_score = 2.0 * n * max_conv

        # Record improvement.
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"Iter {t:6d}: new best {best_score:.6f}")

        # ------- soft‑max weights over convolution entries -------
        M = max_conv
        if lam < 1e-12:
            w = np.full_like(conv, 1.0 / conv.size)
        else:
            # Use a shifted exponent for numerical stability.
            exp_arg = lam * (conv - M)
            exp_arg = np.clip(exp_arg, -700.0, 0.0)   # avoid under‑/overflow
            w = np.exp(exp_arg)
            w_sum = w.sum()
            if w_sum == 0.0:                         # pathological case
                w = np.full_like(conv, 1.0 / conv.size)
            else:
                w /= w_sum

        # ------- gradient of the smoothed objective -------
        a_rev = a[::-1]
        Fw = np.fft.rfft(w, n=L)
        Fa_rev = np.fft.rfft(a_rev, n=L)
        grad_full = np.fft.irfft(Fw * Fa_rev, n=L)
        # central n entries correspond to the required gradient.
        grad = 2.0 * grad_full[n - 1 : n - 1 + n]

        # ------- Adam update (minimise → step opposite to gradient) -------
        t += 1
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        step = lr * m_hat / (np.sqrt(v_hat) + eps)

        a = a - step
        a = simplex_projection(a)                     # stay feasible

        # ------- increase smoothing (approach true max) -------
        lam = min(lam * lam_factor, lam_max)

        # occasional tiny jitter to escape plateaus
        if t % 5000 == 0:
            jitter = np.random.randn(n) * 0.001
            a = simplex_projection(a + jitter)

    print(f"Gradient phase finished after {t} iterations; best score {best_score:.6f}")

    # --------------------------------------------------------------
    # 4. Simulated‑annealing fine‑tuning (uses the exact objective)
    # --------------------------------------------------------------
    cur_a = a.copy()
    cur_score = compute_score(cur_a)

    # If the SA starting point is already better, adopt it.
    if cur_score < best_score:
        best_score = cur_score
        best_a = cur_a.copy()
        print(f"SA start improved best to {best_score:.6f}")

    # Temperature schedule – exponential cooling.
    T_start = cur_score * 0.01
    T_end = cur_score * 1e-5
    it_sa = 0

    while time.time() < deadline:
        T = max(T_end, T_start * (0.9997 ** it_sa))

        # ---- mass‑transfer proposal (keeps sum exactly 1) ----
        src = np.random.choice(n, p=cur_a)                # bias towards larger masses
        tgt = np.random.randint(n - 1)
        if tgt >= src:
            tgt += 1
        if cur_a[src] == 0.0:
            continue
        delta = cur_a[src] * np.random.rand()            # random fraction of source mass
        cand = cur_a.copy()
        cand[src] -= delta
        cand[tgt] += delta

        cand_score = compute_score(cand)

        # Metropolis acceptance rule.
        if cand_score < cur_score:
            accept = True
        else:
            accept_prob = math.exp(-(cand_score - cur_score) / max(T, 1e-12))
            accept = np.random.rand() < accept_prob

        if accept:
            cur_a = cand
            cur_score = cand_score
            if cand_score < best_score:
                best_score = cand_score
                best_a = cand.copy()
                print(f"SA iter {it_sa:6d}: new best {best_score:.6f}")

        # occasional large restart to diversify the search space
        if it_sa % 40000 == 0 and it_sa > 0:
            cur_a = beta_sequence(np.random.uniform(0.5, 5.0),
                                  np.random.uniform(0.5, 5.0), n)
            cur_score = compute_score(cur_a)

        it_sa += 1

    print(f"Simulated‑annealing finished after {it_sa} iterations; final best {best_score:.6f}")

    # --------------------------------------------------------------
    # 5. Return the best sequence (already normalised to sum == 1)
    # --------------------------------------------------------------
    best_a = best_a / best_a.sum()
    return [float(v) for v in best_a.tolist()]
