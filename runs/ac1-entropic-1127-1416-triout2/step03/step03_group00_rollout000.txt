def conv_update(conv, a, src, tgt, delta):
    # conv: array length L
    # a: original a (numpy array)
    # returns new convolution array after moving delta from src to tgt
    L = conv.shape[0]
    n = a.shape[0]
    conv_new = conv.copy()
    # subtract source contribution
    conv_new[src:src+n] -= 2.0 * delta * a
    conv_new[2*src] += delta * delta
    # add target contribution
    conv_new[tgt:tgt+n] += 2.0 * delta * a
    conv_new[2*tgt] += delta * delta
    # adjust double-interaction term
    conv_new[src + tgt] -= 2.0 * delta * delta
    return conv_new
```

"""
propose_candidate – Simulated‑annealing with an O(n) incremental convolution update.

The evaluation function is scale‑invariant, therefore the search is performed
on the probability simplex (sum == 1, non‑negative).  Starting from either the
previously discovered sequence ``height_sequence_1`` or a Beta‑shaped
distribution, a long simulated‑annealing run is performed.  Each move
transfers a random amount of mass from a source index to a distinct target
index.  The convolution of the sequence with itself is updated incrementally
in O(n) time (no FFT per move), which allows millions of proposals within the
time budget and typically drives the bound below the previous 1.5169 value.

The temperature is linearly cooled from a fraction of the current best score
to a tiny final value.  Periodic Dirichlet restarts keep the search from
getting stuck in poor local minima.  The best sequence seen is returned as a
plain ``list[float]``.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Helper functions
# ----------------------------------------------------------------------


def beta_sequence(alpha: float, beta: float, n: int) -> np.ndarray:
    """
    Discretised Beta(α,β) density on n points, normalised to sum to 1.
    The mid‑point rule avoids the singular endpoints.
    """
    x = (np.arange(n) + 0.5) / n
    # use log‑form to stay stable for extreme parameters
    logw = (alpha - 1.0) * np.log(x) + (beta - 1.0) * np.log(1.0 - x)
    logw = np.clip(logw, -700.0, 700.0)   # overflow guard
    w = np.exp(logw)
    if not np.isfinite(w.sum()) or w.sum() == 0.0:
        w = np.ones_like(w)
    return w / w.sum()


def conv_update(conv: np.ndarray, a: np.ndarray,
                src: int, tgt: int, delta: float) -> np.ndarray:
    """
    Return the convolution a*a after moving a mass ``delta`` from ``src`` to
    ``tgt`` (both indices distinct).  ``conv`` must be the current full
    convolution (length 2*len(a)-1) of ``a`` with itself; ``a`` is the
    *pre‑move* vector.

    The update formula (derived from (a+Δ)*(a+Δ) = a*a + 2*a*Δ + Δ*Δ) yields

        conv' = conv
                - 2·Δ·(a * e_src) + Δ²·e_{2·src}
                + 2·Δ·(a * e_tgt) + Δ²·e_{2·tgt}
                - 2·Δ²·e_{src+tgt} .

    All operations are performed with simple slice assignments (O(n)).
    """
    n = a.shape[0]
    new_conv = conv.copy()

    # source removal   (subtract)
    new_conv[src:src + n] -= 2.0 * delta * a
    new_conv[2 * src] += delta * delta

    # target addition   (add)
    new_conv[tgt:tgt + n] += 2.0 * delta * a
    new_conv[2 * tgt] += delta * delta

    # interaction term between src and tgt (appears twice in the product)
    new_conv[src + tgt] -= 2.0 * delta * delta

    return new_conv


# ----------------------------------------------------------------------
# Main candidate generator
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 0, budget_s: int = 1000) -> list[float]:
    """
    Search for a non‑negative height sequence that minimises
    ``evaluate_sequence``.  The function respects the given time budget
    (seconds) and returns the best sequence found.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.1   # safety margin

    # --------------------------------------------------------------
    # 1. Problem size and initial point
    # --------------------------------------------------------------
    try:
        init = np.array(height_sequence_1, dtype=float)   # type: ignore[name-defined]
        n = init.shape[0]
        print(f"Using existing height_sequence_1 of length {n}.")
    except NameError:
        # fallback: a symmetric beta shape, length chosen as a power of two
        n = 4096
        init = beta_sequence(2.5, 2.5, n)
        print(f"No previous sequence – using a beta start, length {n}.")

    # Normalise to a probability vector (sum == 1).  Scale is irrelevant.
    a = init / init.sum()

    # Initial full convolution (via FFT – done once)
    fft_a = np.fft.rfft(a, n=2 * n - 1)
    conv = np.fft.irfft(fft_a * fft_a, n=2 * n - 1)

    cur_score = best_score = 2.0 * n * conv.max()
    best_a = a.copy()
    print(f"Initial score = {best_score:.6f}")

    # --------------------------------------------------------------
    # 2. Simulated‑annealing with incremental updates
    # --------------------------------------------------------------
    it = 0
    # Temperature schedule: start relatively high, finish near zero.
    T_start = best_score * 0.05
    T_end = best_score * 1e-6

    while time.time() < deadline:
        # Linear cooling based on remaining fraction of time
        remaining = deadline - time.time()
        fraction = max(0.0, remaining / (deadline - start_time))
        T = T_end + (T_start - T_end) * fraction

        # ------------------------------------------------------------------
        # Propose a mass‑transfer move
        # ------------------------------------------------------------------
        # source index – weighted by current mass (larger entries are more
        # likely to move and have a bigger impact)
        src = np.random.choice(n, p=a)

        # target index – uniform among the remaining positions
        tgt = np.random.randint(n - 1)
        if tgt >= src:
            tgt += 1

        if a[src] <= 0.0:
            continue

        # Choose a random fraction of the source mass.
        # Using a square‑root bias favours moderately large moves.
        delta = a[src] * (np.random.rand() ** 0.5)

        # Compute the candidate convolution via the O(n) update.
        cand_conv = conv_update(conv, a, src, tgt, delta)
        cand_score = 2.0 * n * cand_conv.max()

        # ------------------------------------------------------------------
        # Metropolis acceptance
        # ------------------------------------------------------------------
        if cand_score < cur_score:
            accept = True
        else:
            accept_prob = math.exp(-(cand_score - cur_score) / (T + 1e-12))
            accept = np.random.rand() < accept_prob

        if accept:
            # Apply the move permanently
            a[src] -= delta
            a[tgt] += delta
            # Guard against tiny negative values caused by rounding
            if a[src] < 0.0:
                a[src] = 0.0
            conv = cand_conv
            cur_score = cand_score

            if cand_score < best_score:
                best_score = cand_score
                best_a = a.copy()
                if it % 2000 == 0:
                    print(f"Iter {it:6d}: new best {best_score:.6f}")

        # ------------------------------------------------------------------
        # Periodic full restart to escape deep local minima
        # ------------------------------------------------------------------
        if it % 250000 == 0 and it > 0:
            # Draw a fresh Dirichlet (uniform) distribution
            a = np.random.dirichlet(np.ones(n))
            fft_a = np.fft.rfft(a, n=2 * n - 1)
            conv = np.fft.irfft(fft_a * fft_a, n=2 * n - 1)
            cur_score = 2.0 * n * conv.max()
            if cur_score < best_score:
                best_score = cur_score
                best_a = a.copy()
                print(f"Iter {it}: restart produced new best {best_score:.6f}")

        it += 1

    print(f"Simulated‑annealing finished after {it} iterations; "
          f"best score = {best_score:.6f}")

    # --------------------------------------------------------------
    # 3. Return best sequence (already normalised)
    # --------------------------------------------------------------
    best_a = best_a / best_a.sum()
    return [float(v) for v in best_a.tolist()]
