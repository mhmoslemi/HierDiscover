    start = time.time()

        n = 4000   # default size

    # random initial improvements using Beta sampling
    best_a = init / init.sum()
    best_score = compute_score(best_a)
    # Random search over Beta parameters
    for _ in range(200):
        if time.time() >= deadline - 60: break
        alpha = np.random.uniform(0.2, 5.0)
        beta = np.random.uniform(0.2, 5.0)
        a_candidate = beta_sequence(alpha, beta, n)
        a_candidate = a_candidate / a_candidate.sum()
        sc = compute_score(a_candidate)
        if sc < best_score:
            best_score = sc
            best_a = a_candidate.copy()
    print(f"Best after random beta init: {best_score}")

    # gradient descent with Adam
    a = best_a.copy()
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    eta = 1.0  # base lr

    iter_g = 0
    # time reserve for SA: 30 sec
    while time.time() < deadline - 30.0:
        # compute conv
        fft_a = np.fft.rfft(a, n=2*n-1)
        conv = np.fft.irfft(fft_a * fft_a, n=2*n-1)
        max_b = conv.max()
        cur_score = 2.0 * n * max_b
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            if iter_g % 1000 == 0:
                print(f"Adam iter {iter_g:5d}: new best {best_score:.6f}")

        # compute subgradient using active set
        eps_active = max_b * 1e-6
        active = np.where(conv >= max_b - eps_active)[0]
        g = np.zeros_like(a)
        for k in active:
            lo = max(0, k - n + 1)
            hi = min(n-1, k)
            idx = np.arange(lo, hi+1)
            partner = k - idx
            g[idx] += 2.0 * a[partner]
        if active.size > 0:
            g /= float(active.size)

        # Adam moments
        m = beta1 * m + (1 - beta1) * g
        v = beta2 * v + (1 - beta2) * (g * g)

        t = iter_g + 1
        m_hat = m / (1 - beta1**t)
        v_hat = v / (1 - beta2**t)

        step = eta / (np.sqrt(v_hat) + eps)

        # take gradient step
        a = simplex_projection(a - step * m_hat)

        iter_g += 1

    if compute_score(best_a) < cur_score:

        # cooling factor
        T = max(T_end, T_start * (0.999)**it_sa)

        # propose move: transfer mass between two indices
        src = np.random.choice(n, p=cur_a / cur_a.sum())
        tgt = np.random.randint(n-1)
        if tgt >= src:
            tgt += 1

        if cur_a[src] == 0:

        delta = cur_a[src] * np.random.rand()  # transfer some

        cand[src] -= delta
        cand[tgt] += delta

        cand_score = compute_score(cand)
        # acceptance
        if cand_score < cur_score or np.random.rand() < math.exp(-(cand_score - cur_score) / max(T,1e-12)):

            if cand_score < best_score:

            cur_a = new_a
            cur_score = compute_score(cur_a)

        it_sa += 1

    best_a = best_a / best_a.sum()
    return [float(v) for v in best_a.tolist()]

"""
Propose a candidate step‑function (non‑negative vector) that minimizes
``evaluate_sequence``.  The algorithm consists of three stages.

1. **Random symmetric Beta warm‑start** – 200 random Beta(α,α) shapes are
   generated, the best one (lowest value of 2·n·max(a∗a)) is kept.

2. **Projected Adam optimisation** –  the objective  max_k (a∗a)_k is
   non‑smooth; we use a sub‑gradient equal to the average of the
   gradients of the *active* entries (those whose convolution value is
   within 1e‑6 of the maximum).  An Adam‑style adaptive step is taken and
   the iterate is projected onto the probability simplex (sum = 1,
   non‑negative).  The iterate is symmetrised after every step
   (a ← (a+rev(a))/2) because the problem is invariant under reversal.

3. **Simulated‑annealing refinement** –  a simple mass‑transfer Metropolis
   walk on the simplex with an exponential cooling schedule finishes the
   search.  Every 20 000 SA steps the current point is re‑initialised with
   a fresh Beta draw to avoid stagnation.

The whole procedure respects the time budget (``budget_s`` seconds) and
returns the best vector found, normalised to sum = 1.
"""

def simplex_projection(v: np.ndarray) -> np.ndarray:
    """Project v onto the probability simplex {w | w_i >= 0, sum w_i = 1}.
    Implements the O(n) algorithm of Duchi et al. (2008)."""
    v = np.asarray(v, dtype=float)
    n = v.shape[0]

    # sort v in decreasing order
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all entries are non‑positive – return uniform distribution
        return np.full_like(v, 1.0 / n)

    rho = ind[cond][-1] - 1                 # zero‑based index of rho
    theta = cssv[cond][-1] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def compute_score(a: np.ndarray) -> float:
    """Exact value of ``evaluate_sequence`` for a vector with sum(a) == 1.
    Returns 2·n·max( a∗a )."""
    n = a.shape[0]
    L = 1 << ((2 * n - 2).bit_length())     # power‑of‑two length for FFT
    fft_a = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(fft_a * fft_a, n=L)[: 2 * n - 1]
    return 2.0 * n * conv.max()


def beta_sequence(alpha: float, beta: float, n: int) -> np.ndarray:
    """Discretised Beta(α,β) density on n points (mid‑point rule)
    and normalised to sum to 1."""
    x = (np.arange(n) + 0.5) / n
    # work in log‑space to avoid overflow
    logw = (alpha - 1.0) * np.log(x) + (beta - 1.0) * np.log(1.0 - x)
    logw = np.clip(logw, -700.0, 700.0)      # avoid under/overflow
    w = np.exp(logw)
    if not np.isfinite(w.sum()) or w.sum() == 0.0:
        w = np.ones_like(w)
    return w / w.sum()


def self_convolution(a: np.ndarray) -> np.ndarray:
    """Linear convolution a∗a (length 2·len(a)‑1) using a power‑of‑two FFT."""
    n = a.shape[0]
    L = 1 << ((2 * n - 2).bit_length())
    fft_a = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(fft_a * fft_a, n=L)[: 2 * n - 1]
    return conv


def symmetrise(v: np.ndarray) -> np.ndarray:
    """Replace v by (v + rev(v))/2 – preserves the simplex property."""
    return (v + v[::-1]) * 0.5


# ----------------------------------------------------------------------
# Main candidate generator
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 0, budget_s: int = 1000) -> list[float]:
    """
    Search for a non‑negative height sequence that minimises
    ``evaluate_sequence`` within the given time budget.

    Parameters
    ----------
    seed : int
        Random seed for reproducibility.
    budget_s : int
        Time budget in seconds.

    Returns
    -------
    list[float]
        The best sequence found (normalised to sum == 1).
    """
    np.random.seed(seed)
    start = time.time()
    deadline = start + budget_s - 0.5        # tiny safety margin

    # --------------------------------------------------------------
    # 1. Problem size & initial point
    # --------------------------------------------------------------
    try:
        init = np.array(height_sequence_1, dtype=float)   # type: ignore[name-defined]
        n = init.shape[0]
        print(f"Using existing height_sequence_1 of length {n}.")
    except NameError:
        # fallback – a moderately sized symmetric beta shape
        n = 5146
        init = beta_sequence(2.5, 2.5, n)
        print(f"No previous sequence – start with symmetric beta, n={n}.")

    a = init / init.sum()                     # start on the simplex
    a = symmetrise(a)                         # enforce symmetry from the start

    best_a = a.copy()
    best_score = compute_score(best_a)
    print(f"Initial score = {best_score:.6f}")

    # --------------------------------------------------------------
    # 2. Random symmetric Beta warm‑start
    # --------------------------------------------------------------
    for _ in range(200):
        if time.time() >= deadline - 60:       # keep ~1 min for later phases
            break
        alpha = np.random.uniform(0.2, 5.0)
        # symmetry – use the same exponent for both sides
        cand = beta_sequence(alpha, alpha, n)
        cand = symmetrise(cand) / cand.sum()
        sc = compute_score(cand)
        if sc < best_score:
            best_score = sc
            best_a = cand.copy()
            print(f"Beta warm‑start: new best {best_score:.6f}")

    # --------------------------------------------------------------
    # 3. Projected Adam optimisation on the sub‑gradient of max‑conv
    # --------------------------------------------------------------
    a = best_a.copy()
    m = np.zeros_like(a)          # first moment
    v = np.zeros_like(a)          # second moment
    beta1, beta2 = 0.9, 0.999
    eps = 1e-12
    eta0 = 1.0                    # base learning rate
    it = 0

    # reserve about 30 s for simulated‑annealing
    while time.time() < deadline - 30.0:
        # --- convolution & current score ---
        conv = self_convolution(a)
        max_b = conv.max()
        cur_score = 2.0 * n * max_b

        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            if it % 1000 == 0:
                print(f"Adam iter {it:5d}: new best {best_score:.6f}")

        # --- active‑set sub‑gradient (average over all near‑max entries) ---
        eps_active = max_b * 1e-6
        active = np.where(conv >= max_b - eps_active)[0]
        g = np.zeros_like(a)

        for k in active:
            lo = max(0, k - n + 1)
            hi = min(n - 1, k)
            idx = np.arange(lo, hi + 1, dtype=int)
            partner = k - idx
            g[idx] += 2.0 * a[partner]

        if active.size:
            g /= float(active.size)

        # --- Adam update ---
        m = beta1 * m + (1.0 - beta1) * g
        v = beta2 * v + (1.0 - beta2) * (g * g)

        it += 1
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        step = eta0 / (np.sqrt(v_hat) + eps)

        a = simplex_projection(a - step * m_hat)
        a = symmetrise(a)          # stay in the symmetric subspace
        # (symmetrisation preserves the simplex constraints)

    print(f"Adam phase finished after {it} iterations; best score {best_score:.6f}")

    # --------------------------------------------------------------
    # 4. Simulated‑annealing fine‑tuning
    # --------------------------------------------------------------
    cur_a = a.copy()
    cur_score = compute_score(cur_a)

    # ensure we keep the overall best seen so far
    if best_score < cur_score:
        cur_a = best_a.copy()
        cur_score = best_score

    # temperature schedule
    T_start = cur_score * 0.02
    T_end = cur_score * 1e-5
    it_sa = 0

    while time.time() < deadline:
        # exponential cooling
        T = max(T_end, T_start * (0.999) ** it_sa)

        # ---- propose a mass transfer ----
        src = np.random.choice(n, p=cur_a / cur_a.sum())
        tgt = np.random.randint(n - 1)
        if tgt >= src:
            tgt += 1

        if cur_a[src] <= 0.0:
            continue

        delta = cur_a[src] * np.random.rand()
        cand = cur_a.copy()
        cand[src] -= delta
        cand[tgt] += delta
        cand = symmetrise(cand)   # keep symmetry

        cand_score = compute_score(cand)

        # ---- Metropolis acceptance ----
        if cand_score < cur_score:
            accept = True
        else:
            accept_prob = math.exp(-(cand_score - cur_score) / max(T, 1e-12))
            accept = np.random.rand() < accept_prob

        if accept:
            cur_a = cand
            cur_score = cand_score
            if cand_score < best_score:
                best_score = cand_score
                best_a = cand.copy()
                print(f"SA iter {it_sa:6d}: new best {best_score:.6f}")

        # occasional large perturbation to avoid being stuck
        if it_sa % 20000 == 0 and it_sa > 0:
            a_tmp = beta_sequence(np.random.uniform(0.2, 5.0),
                                  np.random.uniform(0.2, 5.0), n)
            cur_a = symmetrise(a_tmp) / a_tmp.sum()
            cur_score = compute_score(cur_a)

        it_sa += 1

    print(f"Simulated‑annealing finished after {it_sa} iterations; final best {best_score:.6f}")

    # --------------------------------------------------------------
    # 5. Return result normalised to sum == 1
    # --------------------------------------------------------------
    best_a = best_a / best_a.sum()
    return [float(v) for v in best_a.tolist()]
