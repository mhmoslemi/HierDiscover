function adam_optimize(initial_a, deadline,...):

   n = len(a)

   best_a = a.copy()

       a = normalize(a)
       cur_score = compute_score(a)
       if cur_score < best_score:

           best_a = a.copy()
       if it % 2000 == 0:

def lp_max_sum(f, M):
    n = len(f)
    from scipy.linalg import toeplitz
    # build A matrix
    first_col = np.concatenate([f, np.zeros(n-1, dtype=float)])
    first_row = np.concatenate([f[:1], np.zeros(n-1, dtype=float)])
    A = toeplitz(first_col, first_row)  # shape (2n-1, n)
    b_ub = np.full(2*n-1, M, dtype=float)
    c = -np.ones(n, dtype=float)  # maximize sum => minimize -sum
    bounds = [(0, None)] * n
    # solve.
    from scipy.optimize import linprog
    res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
    if not res.success:
        raise RuntimeError(f"LP failed: {res.message}")
    return res.x

def iterative_lp_adam(a0, deadline, n_lp_steps=3, t=0.5):

"""
Hybrid search for a low‑max‑convolution step‑function.

The algorithm mixes three ideas:

1.  A rich set of parametric initial shapes (uniform, arcsine,
    power‑law, beta‑like, Dirichlet) on a few lengths (512‑2048).

2.  A smooth‑max (log‑sum‑exp) surrogate for the max‑convolution
    is minimised with an Adam‑style projected gradient descent that
    works on the probability simplex.

3.  After a first refinement a tiny linear‑programming sub‑step is
    performed: given the current sequence *f* we maximise ∑bᵢ subject
    to  f∗b ≤ M (M = max(f∗f)).  The optimiser of the LP is a new
    candidate *g* which is blended into *f*.  A few LP + Adam cycles
    dramatically tighten the max‑convolution.

All work is bounded by the supplied time budget (default 1000 s);
the best feasible sequence found is returned.
"""

import time
import numpy as np
from scipy.optimize import linprog
from scipy.linalg import toeplitz
from typing import List, Tuple

# ----------------------------------------------------------------------
# Low‑level numeric utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Linear convolution via real FFT (full length)."""
    m = len(a) + len(b) - 1
    n_fft = next_pow2(m)
    A = np.fft.rfft(a, n=n_fft)
    B = np.fft.rfft(b, n=n_fft)
    C = A * B
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


def compute_score(a: np.ndarray) -> float:
    """
    Official objective (scale‑invariant):
        2·n·max(a∗a) / (sum(a)²)
    a is assumed non‑negative.
    """
    n = len(a)
    conv = convolution_fft(a, a)
    max_val = float(np.max(conv))
    s = float(np.sum(a))
    if s < 1e-12:
        return float("inf")
    return 2.0 * n * max_val / (s * s)


def softmax_weights(conv: np.ndarray, lam: float) -> np.ndarray:
    """Numerically stable soft‑max over the convolution entries."""
    x = lam * conv
    mx = np.max(x)
    w = np.exp(x - mx)
    w /= w.sum()
    return w


def gradient_from_weights(a: np.ndarray, w: np.ndarray) -> np.ndarray:
    """
    Gradient of Σ_k w_k·(a∗a)_k w.r.t a.
    Implements g_j = 2· Σ_i a_i·w_{i+j}.
    """
    w_rev = w[::-1]
    conv = convolution_fft(a, w_rev)          # length = 3·n‑2
    idx = 2 * len(a) - 2                       # points to position 2n‑2
    grads = 2.0 * conv[idx - np.arange(len(a))]
    return grads


def normalize(v: np.ndarray) -> np.ndarray:
    """Project a non‑negative vector onto the probability simplex."""
    total = v.sum()
    if total <= 0.0:
        return np.full_like(v, 1.0 / len(v))
    return v / total


# ----------------------------------------------------------------------
# Parametric families for initial candidates
# ----------------------------------------------------------------------


def arcsine_initial(n: int) -> np.ndarray:
    """Discrete arcsine density (with small regulariser)."""
    i = np.arange(n, dtype=float)
    eps = 0.25
    vals = 1.0 / np.sqrt((i + 0.5) * (n - 0.5 - i) + eps)
    return normalize(vals)


def generate_shape(n: int, p: float, eps: float) -> np.ndarray:
    """
    Symmetric power‑law shape a_i ∝ ((i+ε)(n‑1‑i+ε))⁻ᵖ.
    """
    i = np.arange(n, dtype=float)
    base = ((i + eps) * (n - 1 - i + eps)) ** (-p)
    return normalize(base)


# ----------------------------------------------------------------------
# Adam optimiser on the smooth max‑convolution proxy
# ----------------------------------------------------------------------


def adam_optimize(
    init_a: np.ndarray,
    deadline: float,
    *,
    lam_start: float = 5.0,
    lam_end: float = 80.0,
    lam_ramp_iters: int = 4000,
    lr: float = 0.05,
    beta1: float = 0.9,
    beta2: float = 0.999,
    eps: float = 1e-8,
    max_iter: int = 200_000,
) -> Tuple[np.ndarray, float]:
    """
    Projected Adam on the smooth approximation
        Φ_λ(a)=log‑sum‑exp(λ·(a∗a))/λ .
    Returns the best sequence seen and its (exact) score.
    """
    a = normalize(init_a.astype(float).copy())
    best_a = a.copy()
    best_sc = compute_score(a)

    m = np.zeros_like(a)
    v = np.zeros_like(a)

    for it in range(1, max_iter + 1):
        if time.time() >= deadline:
            break

        # anneal λ
        lam = lam_start + (lam_end - lam_start) * min(1.0, it / lam_ramp_iters)

        # smoothed max and its gradient
        conv = convolution_fft(a, a)                 # length 2n‑1
        w = softmax_weights(conv, lam)
        grad = gradient_from_weights(a, w)           # gradient of weighted sum

        # Adam moment updates
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)

        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        # descent step (note the minus sign – we minimise)
        step = lr * m_hat / (np.sqrt(v_hat) + eps)
        a = a - step

        # keep feasibility
        a = np.maximum(a, 0.0)
        a = normalize(a)

        cur_sc = compute_score(a)
        if cur_sc < best_sc:
            best_sc = cur_sc
            best_a = a.copy()

        if it % 2000 == 0:
            print(f"    Adam iter={it:,d}  λ={lam:.1f}  cur={cur_sc:.6f}  best={best_sc:.6f}")

    return best_a, best_sc


# ----------------------------------------------------------------------
# Linear‑programming sub‑step (maximise L1 norm under convolution bound)
# ----------------------------------------------------------------------


def lp_max_sum(f: np.ndarray, M: float) -> np.ndarray:
    """
    Solve  max ∑ bᵢ   s.t.   b ≥ 0,  f∗b ≤ M .
    Returns the optimal b (not yet normalised).
    """
    n = len(f)

    # convolution matrix A such that (f∗b) = A @ b
    first_col = np.concatenate([f, np.zeros(n - 1, dtype=float)])
    first_row = np.concatenate([f[:1], np.zeros(n - 1, dtype=float)])
    A = toeplitz(first_col, first_row)          # shape (2n‑1, n)

    c = -np.ones(n, dtype=float)                # maximise sum → minimise −sum
    b_ub = np.full(2 * n - 1, M, dtype=float)  # upper bound for every convolution entry
    bounds = [(0.0, None)] * n

    res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method="highs")
    if not res.success:
        raise RuntimeError(f"LP failed: {res.message}")
    return res.x


def lp_adam_refine(
    start_a: np.ndarray,
    deadline: float,
    *,
    lp_steps: int = 4,
    mix_factor: float = 0.5,
) -> Tuple[np.ndarray, float]:
    """
    Alternate a few LP steps with a short Adam refinement.
    Returns the best sequence encountered.
    """
    a = start_a.copy()
    best_a = a.copy()
    best_sc = compute_score(a)

    t = mix_factor
    for step in range(lp_steps):
        if time.time() >= deadline - 2.0:
            break

        # current bound M
        M = float(np.max(convolution_fft(a, a)))

        # solve LP
        try:
            b = lp_max_sum(a, M)
        except Exception as e:
            print(f"    LP step {step + 1} failed: {e}")
            break

        if b.sum() == 0.0:
            break

        g = b / b.sum()               # normalise to the simplex
        a = (1.0 - t) * a + t * g     # convex combination, keeps sum = 1

        # a few quick Adam iterations to smooth out the new point
        sub_deadline = min(deadline, time.time() + 5.0)
        a, _ = adam_optimize(
            a,
            sub_deadline,
            lam_start=5.0,
            lam_end=30.0,
            lam_ramp_iters=2000,
            lr=0.05,
            max_iter=2000,
        )

        cur_sc = compute_score(a)
        print(f"    After LP+Adam step {step + 1}: score = {cur_sc:.6f}")
        if cur_sc < best_sc:
            best_sc = cur_sc
            best_a = a.copy()

        # slowly decrease the mixing weight
        t = max(0.05, t * 0.8)

    return best_a, best_sc

def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs) -> List[float]:
    """

    # ------------------------------------------------------------------

    except Exception:
        pass

    # ------------------------------------------------------------------
    # 1. Lengths to explore (powers of two – cheap FFTs).
    # ------------------------------------------------------------------
    lengths = [512, 1024, 2048]

    for n in lengths:
        if time.time() >= deadline - 5.0:
            break
        print(f"\n=== Exploring length n = {n} ===")

        # --------------------------------------------------------------
        # 1a. Build a modest library of candidate start vectors.
        # --------------------------------------------------------------
        candidates: List[np.ndarray] = []

        # uniform

        # power‑law family

            beta = rng.uniform(0.2, 5.0)
            i = np.arange(n, dtype=float)
            base = (i + 0.5) ** (alpha - 1) * (n - 1 - i + 0.5) ** (beta - 1)
            candidates.append(normalize(base))

        # exponential / Dirichlet samples
        for _ in range(2):
            samp = rng.exponential(1.0, size=n)
            candidates.append(normalize(samp))

        # Dirichlet (symmetric)
        for _ in range(2):
            samp = rng.dirichlet(np.ones(n))
            candidates.append(normalize(samp))

        # --------------------------------------------------------------
        # 1b. Quick sweep with the cheap internal score, keep the best few.
        # --------------------------------------------------------------
        scored = [(compute_score(c), c) for c in candidates]
        scored.sort(key=lambda x: x[0])
        top_candidates = [c for _, c in scored[:3]]
        print(f"  Selected top {len(top_candidates)} candidates after fast scoring.")

        # --------------------------------------------------------------
        # 1c. Refine each retained candidate.

        for idx, cand in enumerate(top_candidates):

            # ---- initial Adam descent (fast) ----
            sub_deadline = min(deadline, time.time() + 20.0)
            a_refined, _ = adam_optimize(
                cand,
                sub_deadline,
                lam_start=5.0,
                lam_end=80.0,
                lam_ramp_iters=3000,
                lr=0.05,
                max_iter=200000,

            # ---- LP + short Adam polishing ----
            if time.time() < deadline - 5.0:
                a_refined, _ = lp_adam_refine(a_refined, deadline, lp_steps=4, mix_factor=0.5)

            cur_score = compute_score(a_refined)
            print(f"    Internal score after refinements = {cur_score:.6f}")

            # ---- official evaluation (only for the current best) ----
            val = evaluate_sequence(a_refined.tolist())
            print(f"    Official evaluator score = {val:.6f}")

            if val < best_val:

                best_seq = a_refined.tolist()
                print(f"    *** New global best: {best_val:.6f} (n={n}) ***")

        a = np.array(best_seq, dtype=float)
        print("\nFinal global polishing")
        a, _ = adam_optimize(
            a,
            deadline,
            lam_start=5.0,
            lam_end=150.0,
            lam_ramp_iters=8000,
            lr=0.03,
            max_iter=50000,

        final_val = evaluate_sequence(a.tolist())
        print(f"Final polished official score = {final_val:.6f}")

        best_seq = [1.0] * n
        best_val = evaluate_sequence(best_seq)

    print(f"\nSearch finished. Best reported score = {best_val:.6f}")
    return best_seq
