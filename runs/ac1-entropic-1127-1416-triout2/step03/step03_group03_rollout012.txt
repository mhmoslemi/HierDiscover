"""
Propose a step‑function (non‑negative sequence) that minimizes
`evaluate_sequence`.  

The search proceeds in three stages:

1. **Smoothed‑max Adam** – several random restarts are optimized with a
   log‑sum‑exp approximation of the max‑convolution.  The smoothing
   parameter α is increased up to a very large value (≈5·10⁴) so that the
   optimization ultimately focuses on the true max.

2. **Exact sub‑gradient descent** – starting from the best Adam result we
   switch to a pure sub‑gradient method that follows the gradient of the
   *current* maximum convolution entry.  This directly reduces the
   `max(conv)` term while staying on the probability simplex.

3. **Permutation / mass‑transfer hill‑climb** – a stochastic local search
   (swap or move a fraction of mass) refines the solution using the
   true (unsmoothed) objective.

All heavy work uses FFT‑based convolutions (`O(n log n)`).  The routine
adapts the number of iterations to the remaining wall‑clock time, so it
fits into the 1000 s budget.  It returns a plain Python list of floats
that can be fed to `evaluate_sequence`.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Basic geometry utilities
# ----------------------------------------------------------------------


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """ Euclidean projection onto the probability simplex (sum = 1,
        non‑negative). """
    if np.all(v <= 0):
        # all entries non‑positive → uniform simplex point
        return np.full_like(v, 1.0 / v.shape[0])

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, v.shape[0] + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.where(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def next_pow2(x: int) -> int:
    """ Smallest power of two ≥ x. """
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """ Full convolution (linear) of two real vectors using FFT. """
    m = len(a) + len(b) - 1
    n_fft = next_pow2(m)
    A = np.fft.rfft(a, n=n_fft)
    B = np.fft.rfft(b, n=n_fft)
    C = A * B
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


def compute_score(a: np.ndarray) -> float:
    """ Fast surrogate of `evaluate_sequence` when Σa = 1.
        Returns 2·n·max(conv(a,a)).
    """
    n = a.shape[0]
    conv = convolution_fft(a, a)
    return 2.0 * n * float(np.max(conv))


def arcsine_initial(n: int) -> np.ndarray:
    """ Classical arcsine start: a_i ∝ 1/√((i+½)(n‑½‑i)). """
    i = np.arange(n, dtype=float)
    eps = 0.25
    vals = 1.0 / np.sqrt((i + 0.5) * (n - 0.5 - i) + eps)
    return project_to_simplex(vals)


# ----------------------------------------------------------------------
# Smoothed‑max Adam optimiser (log‑sum‑exp)
# ----------------------------------------------------------------------


def smoothed_gradient(a: np.ndarray, alpha: float):
    """
    Gradient of the soft‑max approximation
        f_α(a) = (1/α)·log Σ_k exp(α·(a*a)_k)

    Returns (grad, conv) where grad has the same shape as `a`.
    """
    n = a.shape[0]
    conv = convolution_fft(a, a)                     # length 2n‑1
    maxc = np.max(conv)
    exp_term = np.exp(alpha * (conv - maxc))
    w = exp_term / np.sum(exp_term)                 # Σ w = 1

    # To obtain ∂conv_k/∂a we need convolution of reversed a with w.
    a_rev = a[::-1]
    conv_rev = convolution_fft(a_rev, w)            # length 3n‑2
    start = n - 1
    rev_slice = conv_rev[start:start + n]           # needed for each component

    # diagonal weights (where i=j) appear only once in the original sum
    w_diag = w[::2][:n]

    grad = 2.0 * rev_slice - w_diag * a
    return grad, conv


def adam_optimize(
    init_a: np.ndarray,
    time_deadline: float,
    max_iter: int = 150_000,
    eta: float = 0.02,
    beta1: float = 0.9,
    beta2: float = 0.999,
    eps: float = 1e-8,
    alpha_start: float = 1.0,
    alpha_end: float = 50_000.0,
    noise_every: int = 1500,
    noise_scale: float = 1e-4,
):
    """
    Projected Adam optimiser on the smoothed max‑convolution objective.
    Returns the best vector encountered and its *exact* (unsmoothed) score.
    """
    a = project_to_simplex(init_a.astype(float).copy())
    m = np.zeros_like(a)
    v = np.zeros_like(a)

    best_a = a.copy()
    best_score = compute_score(a)

    for t in range(1, max_iter + 1):
        if time.time() >= time_deadline:
            break

        # linear schedule for the smoothing parameter
        alpha = alpha_start + (alpha_end - alpha_start) * min(1.0, t / max_iter)

        grad, _ = smoothed_gradient(a, alpha)

        # Adam moment updates
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)

        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)

        step = eta * m_hat / (np.sqrt(v_hat) + eps)
        a_new = a - step

        # occasional isotropic noise
        if noise_every and (t % noise_every == 0):
            a_new += np.random.randn(*a.shape) * noise_scale

        a_new = project_to_simplex(a_new)

        cur_score = compute_score(a_new)
        if cur_score < best_score:
            best_score = cur_score
            best_a = a_new.copy()

        a = a_new

        # light progress output
        if t % 5000 == 0:
            print(
                f"    Adam iter {t:,d} – cur {cur_score:.6f} – best {best_score:.6f}"
            )

    return best_a, best_score

def subgradient_descent(
    init_a: np.ndarray,
    time_deadline: float,
    max_iter: int = 200_000,
    eta0: float = 0.5,
):
    """
    Sub‑gradient descent that follows the gradient of the *current* maximum
    convolution entry.  The step size decays like η₀/√t.
    """
    a = project_to_simplex(init_a.astype(float).copy())
    n = a.shape[0]

    best_a = a.copy()
    best_score = compute_score(a)

    for t in range(1, max_iter + 1):
        if time.time() >= time_deadline:
            break

        conv = convolution_fft(a, a)
        max_idx = int(np.argmax(conv))

        # compute gradient of the max entry (2*a_{k‑j} where defined)
        grad = np.zeros_like(a)
        j_start = max(0, max_idx - (n - 1))
        j_end = min(max_idx, n - 1)
        if j_start <= j_end:
            js = np.arange(j_start, j_end + 1, dtype=int)
            grad[js] = 2.0 * a[max_idx - js]

        eta = eta0 / np.sqrt(t)
        a_new = a - eta * grad
        a_new = project_to_simplex(a_new)

        cur_score = compute_score(a_new)
        if cur_score < best_score:
            best_score = cur_score
            best_a = a_new.copy()

        a = a_new

        if t % 5000 == 0:
            print(
                f"    Subgrad iter {t:,d} – cur {cur_score:.6f} – best {best_score:.6f}"
            )

    return best_a, best_score

# Stochastic hill‑climb (swap / mass transfer) with simulated annealing
# ----------------------------------------------------------------------


def hill_climb(
    init_seq,
    time_deadline,
    max_steps: int = 250_000,
    start_temp: float = 1e-2,
    end_temp: float = 1e-5,
):
    """
    Local search that swaps two entries or transfers a fraction of mass.
    Accepts worse moves with a temperature that decays linearly.
    """
    a = np.array(init_seq, dtype=float)
    n = a.shape[0]

    best_a = a.copy()
    best_score = compute_score(best_a)

    step_frac = 0.1  # initial fraction of mass moved in a transfer
    for step in range(1, max_steps + 1):
        if time.time() >= time_deadline:
            break

        # linear temperature schedule
        temp = start_temp + (end_temp - start_temp) * (step / max_steps)

        cand = best_a.copy()
        if np.random.rand() < 0.5:
            # swap
            i, j = np.random.randint(n, size=2)
            cand[i], cand[j] = cand[j], cand[i]
        else:
            # transfer a fraction from i to j
            i, j = np.random.randint(n, size=2)
            if i == j:
                continue
            delta = step_frac * cand[i]
            if delta <= 0:
                continue
            cand[i] -= delta
            cand[j] += delta
            # simple guard against tiny negatives
            if cand[i] < 0:
                cand[i] = 0.0

        cand = project_to_simplex(cand)
        cur_score = compute_score(cand)

        # accept if improvement or with Metropolis probability
        if cur_score < best_score or np.random.rand() < np.exp((best_score - cur_score) / max(temp, 1e-12)):
            best_a = cand
            best_score = cur_score
            # slowly increase transfer fraction on success
            step_frac = min(step_frac * 1.03, 0.6)

        else:
            # shrink on failure
            step_frac *= 0.97

        if step % 20000 == 0:
            print(
                f"    Hill‑climb step {step:,d} – cur {cur_score:.6f} – best {best_score:.6f}"
            )

    return best_a.tolist(), best_score

# Main entry point
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 123, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative sequence that minimizes `evaluate_sequence`.
    The routine respects the time budget, runs several Adam restarts,
    follows up with sub‑gradient descent and finishes with a stochastic
    hill‑climber.
    """
    np.random.seed(seed)
    start_time = time.time()
    # reserve a couple of seconds for harness I/O
    deadline = start_time + budget_s - 2.0

    best_seq = None
    best_val = float("inf")

    # ------------------------------------------------------------------
    # Parameters
    # ------------------------------------------------------------------
    n = 8192                     # length of the main search (larger -> better asymptotics)
    init_vectors = [
        arcsine_initial(n),
        np.full(n, 1.0 / n),                 # uniform
        np.random.dirichlet(np.full(n, 0.5)),  # concentrated Dirichlet
        np.random.dirichlet(np.full(n, 1.0)),
        np.random.dirichlet(np.full(n, 2.0)),
    ]

    # ------------------------------------------------------------------
    # Phase 1 – smoothed‑max Adam on several restarts
    # ------------------------------------------------------------------
    print(f"\n=== Phase 1 – Adam optimisation (n = {n}) ===")
    for idx, init_vec in enumerate(init_vectors, start=1):
        if time.time() >= deadline - 20.0:
            break
        print(f"  Adam start {idx}/{len(init_vectors)}")
        # allocate a modest number of iterations; the remaining time will be used later
        remaining = deadline - time.time()
        iters = int(min(150_000, remaining * 30))   # heuristic: ~30 iters per second
        seq, val = adam_optimize(
            init_a=init_vec,
            time_deadline=deadline,
            max_iter=iters,
            eta=0.02,
            alpha_start=1.0,
            alpha_end=50_000.0,
            noise_every=1500,
            noise_scale=1e-4,
        )
        print(f"    Adam phase best score = {val:.6f}")
        if val < best_val:
            best_val = val
            best_seq = seq.tolist()
            print(f"    *** Global best updated to {best_val:.6f} ***")

    # ------------------------------------------------------------------
    # Phase 2 – exact sub‑gradient descent (focuses on true max)
    # ------------------------------------------------------------------
    if best_seq is not None and time.time() < deadline - 15.0:
        print("\n=== Phase 2 – Sub‑gradient descent ===")
        sub_deadline = min(deadline, time.time() + 120.0)  # give it up to 2 min
        seq, val = subgradient_descent(
            init_a=np.array(best_seq, dtype=float),
            time_deadline=sub_deadline,
            max_iter=200_000,
            eta0=0.6,
        )
        print(f"    Sub‑gradient phase best score = {val:.6f}")
        if val < best_val:
            best_val = val
            best_seq = seq.tolist()
            print(f"    *** Global best updated to {best_val:.6f} ***")

    # ------------------------------------------------------------------
    # Phase 3 – stochastic hill‑climb / simulated annealing
    # ------------------------------------------------------------------
    if best_seq is not None and time.time() < deadline - 5.0:
        print("\n=== Phase 3 – Hill‑climbing refinement ===")
        refined_seq, refined_val = hill_climb(
            best_seq,
            time_deadline=deadline,
            max_steps=250_000,
            start_temp=1e-2,
            end_temp=1e-5,
        )
        print(f"    Refined score = {refined_val:.6f}")
        if refined_val < best_val:
            best_val = refined_val
            best_seq = refined_seq

    # ------------------------------------------------------------------
    # Fallback (should never trigger)
    # ------------------------------------------------------------------
    if best_seq is None:
        # trivial uniform vector of modest size
        best_seq = [1.0 / 256] * 256
        best_val = evaluate_sequence(best_seq)

    final_score = evaluate_sequence(best_seq)
    print(f"\nFinished search. Final reported score = {final_score:.6f}")

    # The harness expects a plain Python list of floats.
    return best_seq
