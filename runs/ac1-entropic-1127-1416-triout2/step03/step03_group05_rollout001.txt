"""
Differential‑Evolution + occasional Adam refinement.

*   Uses a population of candidate step‑functions (non‑negative,
    ℓ¹‑normalised).
*   The fitness is 2·n·max(conv(a,a)) (the denominator is 1 because of the
    normalisation).
*   A simple DE scheme explores the search space; every few generations the
    worst individuals are refreshed with fresh random draws.
*   Every ~20 generations the current best is given a short Adam‑subgradient
    optimisation (the routine from the previous state‑of‑the‑art) to escape
    plateaus.
*   The algorithm stops just before the allotted time budget and returns the
    best sequence found as a plain Python list of floats.

The implementation respects the constraints of the evaluation harness,
contains only top‑level functions (no closures), and prints occasional
progress messages.
"""
import time
import numpy as np

# ----------------------------------------------------------------------
# Helper: max‑convolution and its sub‑gradient (used by the Adam step)
# ----------------------------------------------------------------------
def _maxconv_and_subgrad(a: np.ndarray, fft_len: int):
    """Return (max_value, gradient) of the aperiodic convolution a*a."""
    n = a.shape[0]
    # convolution via FFT (first 2·n‑1 entries are needed)
    fft = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(fft * fft, n=fft_len)[: 2 * n - 1]

    max_val = conv.max()
    # tolerance for numerically tied maxima
    tol = max_val * 1e-12 + 1e-14
    active = np.where(conv >= max_val - tol)[0]

    grad = np.zeros_like(a)
    for k in active:
        low = max(0, k - n + 1)
        high = min(n - 1, k)
        idx = np.arange(low, high + 1, dtype=int)
        grad[idx] += 2.0 * a[k - idx]

    if active.size > 1:
        grad /= float(active.size)          # any convex combination works
    return max_val, grad


# ----------------------------------------------------------------------
# Adam optimiser on the simplex (used for occasional refinement)
# ----------------------------------------------------------------------
def _adam_optimize(init_a: np.ndarray,
                   deadline: float,
                   restart_id: int,
                   target: float = 1.503) -> tuple[np.ndarray, float]:
    """
    Adam optimisation of 2·n·max(conv(a,a)) with the simplex constraint Σa=1.
    Returns the best vector encountered before *deadline*.
    """
    n = init_a.shape[0]
    fft_len = 1 << ((2 * n - 2).bit_length())

    # logits (soft‑max re‑parameterisation)
    eps_init = 1e-12
    b = np.log(init_a + eps_init)

    # Adam moments
    m = np.zeros_like(b)
    v = np.zeros_like(b)
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8

    # initial evaluation
    max_val, _ = _maxconv_and_subgrad(init_a, fft_len)
    best_score = 2.0 * n * max_val
    best_a = init_a.copy()

    stagnation = 0
    it = 0

    while time.time() < deadline:
        it += 1

        # probability vector from logits
        b_max = np.max(b)
        a = np.exp(b - b_max)
        a = a / a.sum()

        # convolution information
        max_val, grad_a = _maxconv_and_subgrad(a, fft_len)
        cur_score = 2.0 * n * max_val

        # keep best
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            stagnation = 0
            print(f"Adam restart {restart_id:02d} iter {it:6d}: best = {best_score:.6f}")
        else:
            stagnation += 1

        # termination criteria
        if best_score < target:
            break
        if stagnation > 8000:
            b += 0.01 * np.random.randn(n)
            stagnation = 0

        # Adam update (gradient w.r.t. logits)
        grad_obj = 2.0 * n * grad_a
        avg = np.dot(a, grad_obj)
        grad_b = a * (grad_obj - avg)

        m = beta1 * m + (1.0 - beta1) * grad_b
        v = beta2 * v + (1.0 - beta2) * (grad_b ** 2)

        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        lr = 0.4 / np.sqrt(it)
        b -= lr * m_hat / (np.sqrt(v_hat) + eps)

    return best_a, best_score


# ----------------------------------------------------------------------
# Utility to compute the maximum of aperiodic convolution (fast, no grad)
# ----------------------------------------------------------------------
def _maxconv(a: np.ndarray, fft_len: int) -> float:
    """Return max(conv(a,a)) – assumes Σa = 1."""
    n = a.shape[0]
    fa = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(fa * fa, n=fft_len)[: 2 * n - 1]
    return float(conv.max())


# ----------------------------------------------------------------------
# Initial‑sequence factory (used to build a diverse initial population)
# ----------------------------------------------------------------------
def _initial_sequence(kind: str, n: int, rng: np.random.Generator,
                      prev_seq: np.ndarray | None) -> np.ndarray:
    """Return a non‑negative ℓ¹‑normalised vector of length *n*."""
    if kind == "prev" and prev_seq is not None:
        seq = prev_seq.copy()
        if seq.shape[0] != n:
            if seq.shape[0] > n:
                seq = seq[:n]
            else:
                seq = np.concatenate([seq,
                                      np.zeros(n - seq.shape[0], dtype=seq.dtype)])
        seq = np.maximum(seq, 0.0)
        total = seq.sum()
        if total == 0.0:
            seq = np.full(n, 1.0 / n)
        else:
            seq = seq / total
        return seq

    if kind == "uniform":
        return np.full(n, 1.0 / n)

    if kind == "random_dirichlet":
        alpha = 0.5
        sample = rng.gamma(alpha, 1.0, size=n)
        return sample / sample.sum()

    if kind == "geometric":
        r = 0.985 + 0.015 * rng.random()      # r ∈ (0.985, 1.0)
        left = r ** np.arange(n)
        right = r ** np.arange(n - 1, -1, -1)
        seq = left + right
        return seq / seq.sum()

    if kind == "sine":
        i = np.arange(n)
        seq = np.sin(np.pi * (i + 0.5) / n)   # half‑sine, strictly positive
        return seq / seq.sum()

        seq = np.minimum(i + 1, n - i)       # symmetric triangular shape
        return seq / seq.sum()

    if kind == "sqrt":
        i = np.arange(n)
        seq = np.sqrt(i + 1) + np.sqrt(n - i)
        return seq / seq.sum()

    # fallback
    return np.full(n, 1.0 / n)


# ----------------------------------------------------------------------
# Main entry point required by the evaluation harness
# ----------------------------------------------------------------------
def propose_candidate(budget_s: int = 1000) -> list:
    """
    Search for a step‑function with a low value of
        2·n·max(conv(a,a)) / (sum(a)**2)

    The algorithm combines differential evolution with occasional Adam
    refinement.  It runs for roughly *budget_s* seconds and returns the best
    sequence found (as plain Python floats).
    """
    start = time.time()
    deadline = start + budget_s - 1.0            # keep a small safety margin

    # ------------------------------------------------------------------
    # Problem size – try to reuse a previously known construction if present
    # ------------------------------------------------------------------
    try:
        # `height_sequence_1` may be provided by the harness
        previous = np.asarray(height_sequence_1, dtype=float)   # type: ignore[name-defined]
        n = previous.shape[0]
    except Exception:
        previous = None
        n = 5146                                   # default used in earlier runs

    # ------------------------------------------------------------------
    # DE parameters
    # ------------------------------------------------------------------
    popsize = 30
    F = 0.8                     # differential weight
    CR = 0.9                    # crossover probability
    rng = np.random.default_rng(20231127)

    # FFT length for fast convolution (next power of two)
    fft_len = 1 << ((2 * n - 2).bit_length())

    # ------------------------------------------------------------------
    # Build initial population (diverse seeds)
    # ------------------------------------------------------------------
    init_kinds = ["prev", "uniform", "random_dirichlet", "geometric",
                  "sine", "triangle", "sqrt"]
    population = []
    scores = []

    while len(population) < popsize:
        kind = rng.choice(init_kinds)
        if kind == "prev" and previous is None:
            kind = "uniform"
        a = _initial_sequence(kind, n, rng, previous)
        # tiny jitter to avoid exact duplicates
        a += 1e-9 * rng.random(n)
        a = np.maximum(a, 0.0)
        a /= a.sum()
        population.append(a)
        scores.append(2.0 * n * _maxconv(a, fft_len))

    population = np.array(population)          # shape (popsize, n)
    scores = np.array(scores, dtype=float)

    best_idx = int(np.argmin(scores))
    best = population[best_idx].copy()
    best_score = float(scores[best_idx])
    print(f"Initial best score {best_score:.6f}")

    # ------------------------------------------------------------------
    # Main DE loop
    # ------------------------------------------------------------------
    generation = 0
    while time.time() < deadline - 0.2:
        generation += 1

        for i in range(popsize):
            # ---- mutation (DE/rand/1) ----------------------------------
            idxs = rng.choice([j for j in range(popsize) if j != i],
                              size=3, replace=False)
            a, b, c = population[idxs[0]], population[idxs[1]], population[idxs[2]]
            donor = a + F * (b - c)
            donor = np.maximum(donor, 0.0)          # enforce non‑negativity

            # ---- crossover ---------------------------------------------
            cross_mask = rng.random(n) < CR
            if not cross_mask.any():               # guarantee at least one donor entry
                cross_mask[rng.integers(0, n)] = True
            trial = np.where(cross_mask, donor, population[i])

            # ---- renormalise to the simplex -----------------------------
            s = trial.sum()
            if s == 0.0:
                trial = np.full(n, 1.0 / n)
            else:
                trial = trial / s

            # ---- evaluate ------------------------------------------------
            cur_score = 2.0 * n * _maxconv(trial, fft_len)

            # ---- selection ------------------------------------------------
            if cur_score < scores[i]:
                population[i] = trial
                scores[i] = cur_score
                if cur_score < best_score:
                    best_score = cur_score
                    best = trial.copy()
                    print(f"Gen {generation:04d} ind {i:02d} new best {best_score:.6f}")

        # ---- diversity boost every 50 generations -------------------------
        if generation % 50 == 0:
            worst_idxs = np.argsort(scores)[-5:]          # replace 5 worst individuals
            for wi in worst_idxs:
                a_new = rng.gamma(0.5, 1.0, size=n)
                a_new = a_new / a_new.sum()
                population[wi] = a_new
                scores[wi] = 2.0 * n * _maxconv(a_new, fft_len)

        # ---- occasional local Adam refinement -----------------------------
        if generation % 20 == 0 and time.time() < deadline - 5.0:
            # give the current best a few seconds of Adam optimisation
            inner_deadline = time.time() + 4.5
            refined, refined_score = _adam_optimize(best, inner_deadline,
                                                    restart_id=999,
                                                    target=best_score)
            if refined_score < best_score:
                best_score = refined_score
                best = refined.copy()
                # inject the refined individual back into the population
                replace_idx = rng.integers(popsize)
                population[replace_idx] = best.copy()
                scores[replace_idx] = best_score
                print(f"Adam refinement improved best to {best_score:.6f} (gen {generation})")

    print(f"Finished after {generation} generations – best score {best_score:.6f}")
    # Return a plain Python list (the external evaluator expects this format)
    return [float(v) for v in best.tolist()]
