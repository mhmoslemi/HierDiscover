"""
Propose a low‑value step‑function for `evaluate_sequence`.

The algorithm works in several stages within the 1000 s time budget:

1. **Coarse parametric sweep** – generate many candidates from the
   Beta family a_i ∝ (i+½)^α·(n‑½‑i)^β (including the classic arcsine
   case α=β=‑0.5) and keep the best one seen.

2. **Optional up‑sampling** – linearly interpolate the best vector to a
   larger length (8192) if time permits.

3. **Smoothed‑max Adam optimisation** – minimise a smooth
   log‑sum‑exp approximation of the max‐convolution with an Adam
   optimiser, slowly raising the smoothing parameter α up to 2000.
   The true (unsmoothed) score is monitored and the best vector is
   retained.

4. **Exact max‑subgradient descent** – after the Adam stage we switch
   to a true sub‑gradient of the max‑convolution (the gradient of the
   active convolution index).  A decreasing stepsize (η∝1/√t) is used,
   always projecting back onto the probability simplex.

5. **Final stochastic hill‑climb** – cheap pairwise mass‑transfer moves
   with adaptive step size improve the solution further.

The whole search is bounded by the deadline, and the best sequence
found is returned as a list of floats.  All vectors are kept on the
simplex (∑a_i = 1, a_i ≥ 0), which respects the scale‑invariance of the
objective.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Linear‑algebra utilities
# ----------------------------------------------------------------------


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of v onto the probability simplex.
    """
    if np.all(v <= 0):
        n = v.shape[0]
        return np.full_like(v, 1.0 / n, dtype=float)

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho = np.where(u - cssv / np.arange(1, v.shape[0] + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full convolution using a real FFT (fast)."""
    m = len(a) + len(b) - 1
    n_fft = next_pow2(m)
    A = np.fft.rfft(a, n=n_fft)
    B = np.fft.rfft(b, n=n_fft)
    C = A * B
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


def compute_score(a: np.ndarray) -> float:
    """
    Score identical to `evaluate_sequence` when ∑a = 1.
    (the function is scale‑invariant, so we normalise to a probability vector)
    """
    a = np.asarray(a, dtype=float)
    n = len(a)
    conv = convolution_fft(a, a)
    max_val = float(np.max(conv))
    s = float(np.sum(a))
    if s < 1e-12:
        return float("inf")
    return 2.0 * n * max_val / (s * s)


# ----------------------------------------------------------------------
# Families of candidate step‑functions
# ----------------------------------------------------------------------


def arcsine_initial(n: int) -> np.ndarray:
    """Arcsine‑shaped start: a_i ∝ 1 / sqrt((i+0.5)(n‑0.5‑i))."""
    i = np.arange(n, dtype=float)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - 0.5 - i))
    vals = np.clip(vals, 0.0, 1e12)
    return vals / np.sum(vals)


def beta_sequence(n: int, alpha: float, beta: float | None = None) -> np.ndarray:
    """Beta‑type coefficients a_i ∝ (i+½)^α·(n‑½‑i)^β."""
    if beta is None:
        beta = alpha
    i = np.arange(n, dtype=float)
    left = np.power(i + 0.5, alpha)
    right = np.power(n - 0.5 - i, beta)
    seq = left * right
    seq = np.clip(seq, 0.0, 1e12)
    total = np.sum(seq)
    if total == 0.0:
        seq = np.full(n, 1.0 / n)
    else:
        seq = seq / total
    return seq


def exponential_sequence(n: int, lam: float) -> np.ndarray:
    """a_i ∝ exp(-lam·|i - (n‑1)/2|)."""
    i = np.arange(n, dtype=float)
    center = (n - 1) / 2.0
    seq = np.exp(-lam * np.abs(i - center))
    return seq / np.sum(seq)


def upsample_sequence(seq: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation of a probability vector to a larger size."""
    old_n = len(seq)
    old_x = (np.arange(old_n) + 0.5) / old_n
    new_x = (np.arange(new_n) + 0.5) / new_n
    new_vals = np.interp(new_x, old_x, seq)
    return new_vals / np.sum(new_vals)


# ----------------------------------------------------------------------
# Smoothed‑max gradient (Adam optimiser)
# ----------------------------------------------------------------------


def smoothed_gradient(a: np.ndarray, alpha: float):
    """
    Gradient of the smoothed maximal convolution

        f_α(a) = (1/α)·log Σ_k exp(α·(a∗a)_k)

    Returns the gradient (shape n) and the raw convolution (length 2n‑1).
    """
    n = len(a)
    conv = convolution_fft(a, a)                     # length 2n‑1
    maxc = np.max(conv)
    # stable log‑sum‑exp
    exp_term = np.exp(alpha * (conv - maxc))
    w = exp_term / np.sum(exp_term)                 # ∑ w_k = 1

    # derivative of a∗a wrt a is a symmetric Toeplitz matrix;
    # we obtain the gradient via a second convolution.
    a_rev = a[::-1]
    conv_rev = convolution_fft(a_rev, w)            # length 3n‑2
    start = n - 1
    rev_slice = conv_rev[start:start + n]
    # the diagonal of the weight matrix appears on every second entry
    w_diag = w[::2][:n]
    grad = 2.0 * rev_slice - w_diag * a
    return grad, conv


def improve_sequence_adam(
    initial_a: np.ndarray,
    time_deadline: float,
    max_iter: int = 180_000,
    eta0: float = 0.8,
    alpha_start: float = 5.0,
    alpha_end: float = 2000.0,
    beta1: float = 0.9,
    beta2: float = 0.999,
    eps: float = 1e-8,
):
    """
    Adam optimiser on the smoothed‑max objective.
    Returns the best vector encountered and its (unsmoothed) score.
    """
    a = project_to_simplex(initial_a.astype(float).copy())
    best_a = a.copy()
    best_score = compute_score(a)
    m = np.zeros_like(a)
    v = np.zeros_like(a)

    for t in range(1, max_iter + 1):
        if time.time() >= time_deadline:
            break

        # linear schedule for the smoothing parameter
        alpha = alpha_start + (alpha_end - alpha_start) * min(1.0, t / max_iter)

        grad, _ = smoothed_gradient(a, alpha)

        # Adam updates
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)

        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)

        step = eta0 / (np.sqrt(t) + eps)
        a_new = a - step * m_hat / (np.sqrt(v_hat) + eps)

        a_new = project_to_simplex(a_new)

        cur_score = compute_score(a_new)
        if cur_score < best_score:
            best_score = cur_score
            best_a = a_new.copy()
            # occasional log
            if t % 5000 == 0:
                print(f"    Adam iter {t:,d}  new best {best_score:.6f}")

        a = a_new

    return best_a, best_score


# ----------------------------------------------------------------------
# Exact max‑subgradient descent
# ----------------------------------------------------------------------


def max_subgrad_step(a: np.ndarray, step: float) -> np.ndarray:
    """
    One sub‑gradient step for the true max‑convolution.
    The active index k* = argmax (a∗a) is identified,
    the sub‑gradient g_i = a_i + a_{k*-i} (if the latter index exists)
    is formed and a ← a − step·g is projected back onto the simplex.
    """
    n = len(a)
    conv = convolution_fft(a, a)
    k = int(np.argmax(conv))

    # build the shifted copy of a that appears in the sub‑gradient
    shift = np.zeros_like(a)
    if k < n:
        # indices 0 … k have a counterpart
        shift[: k + 1] = a[k::-1]
    else:
        # k ≥ n, only the tail overlaps
        offset = k - (n - 1)          # first index that has a counterpart
        # slice a[n-1 : offset-1 : -1] has length n‑offset
        shift[offset:] = a[n - 1 : offset - 1 : -1]

    g = a + shift                     # sub‑gradient of the active max entry
    a_new = a - step * g
    a_new = np.maximum(a_new, 0.0)    # keep non‑negative before projection
    return project_to_simplex(a_new)


def max_subgradient_descent(
    init_a: np.ndarray,
    time_deadline: float,
    max_steps: int = 250_000,
    eta0: float = 0.8,
):
    """
    Projected sub‑gradient descent on the true L∞ objective.
    Returns the best vector seen and its untuned score.
    """
    a = project_to_simplex(init_a.astype(float).copy())
    best_a = a.copy()
    best_score = compute_score(a)

    for t in range(1, max_steps + 1):
        if time.time() >= time_deadline:
            break
        step = eta0 / np.sqrt(t)            # decaying stepsize
        a = max_subgrad_step(a, step)

        cur_score = compute_score(a)
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            if t % 5000 == 0:
                print(f"    sub‑grad step {t:,d}  new best {best_score:.6f}")

    return best_a, best_score

# Cheap stochastic hill‑climb (pairwise mass transfer)
# ----------------------------------------------------------------------


def hill_climb(
    seq,
    time_deadline: float,
    max_steps: int = 120_000,
    init_step: float = 0.07,
):
    """
    Pair‑wise mass‑transfer hill‑climbing.
    Moves that lower the true (unsmoothed) score are kept.
    """
    a = np.array(seq, dtype=float)
    a = project_to_simplex(a)
    best_a = a.copy()
    best_score = compute_score(best_a)
    n = len(a)
    step = init_step

    while max_steps > 0 and time.time() < time_deadline:
        i, j = np.random.randint(n, size=2)
        if i == j:
            continue
        delta = step * best_a[i]
        if delta <= 0.0:
            continue
        cand = best_a.copy()
        cand[i] -= delta
        cand[j] += delta
        cand = project_to_simplex(cand)
        cur_score = compute_score(cand)
        if cur_score < best_score:
            best_score = cur_score
            best_a = cand
            step = min(step * 1.07, 0.5)
        else:
            step *= 0.96
        max_steps -= 1
    return best_a.tolist(), best_score
