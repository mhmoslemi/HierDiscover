"""
Simple (yet surprisingly effective) optimizer for the step‐function
objective

    2·n·max(a*a) / (sum(a)**2)

The routine mixes three ideas:

1.  Mirror‑descent on the probability simplex using an exponentiated‑
    gradient update.  The sub‑gradient of the maximum convolution entry
    is obtained via FFT (exact, O(n log n)).

2.  A momentum term on the accumulated (scaled) sub‑gradient – this
    smooths the noisy updates and often yields faster descent.

3.  A very cheap simulated‑annealing style “mass‑swap” move plus a
    small Gaussian jitter every few thousand iterations to escape local
    minima.

The optimiser runs for at most ``budget_s`` seconds, keeps the best
vector ever seen and returns it as a plain Python list.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper – sub‑gradient of the maximal convolution entry
# ----------------------------------------------------------------------
def _max_and_subgradient(a: np.ndarray):
    """
    Returns (max_conv, grad) where
        max_conv = max_k (a*a)[k]
    and ``grad`` is a sub‑gradient of that maximal entry
    (averaged if several entries are tied).
    """
    n = a.shape[0]
    # FFT‑based self‑convolution (real‑valued)
    fft = np.fft.rfft(a, n=2 * n - 1)
    conv = np.fft.irfft(fft * fft, n=2 * n - 1)

    max_val = float(conv.max())
    # tolerance for “active” indices (numerical ties)
    tol = max_val * 1e-12 + 1e-14
    active = np.where(conv >= max_val - tol)[0]

    grad = np.zeros_like(a)
    for k in active:
        low = max(0, k - n + 1)
        high = min(n - 1, k)
        idx = np.arange(low, high + 1, dtype=int)
        partner = k - idx
        grad[idx] += 2.0 * a[partner]

    if active.size > 1:
        grad /= float(active.size)
    return max_val, grad
