"""
Search for a low‑value step‑function for the objective

    E(a) = 2·n·max(a∗a) / (sum(a)²)

The algorithm consists of four stages

1. **Wide parametric sweep** – exhaustive search over Beta‑type families,
   arcsine‑shaped powers and a few simple mixtures.  The sweep is performed
   on a moderate length (2048) so thousands of candidates can be evaluated
   quickly.

2. **Upsampling** – the best candidate is linearly interpolated to a larger
   length (8192).  Upsampling usually reduces the ‑max‑value because the
   convolution becomes a finer approximation of the continuous problem.

3. **Projected sub‑gradient descent** – the true non‑smooth objective
   (the max of the convolution) is minimised by iteratively moving opposite
   to a sub‑gradient of the max.  After each step the vector is projected
   onto the probability simplex (∑a = 1, a≥0).  A diminishing step size
   (η ∝ 1/√t) together with occasional tiny Gaussian noise gives a fast
   descent while staying on‑simplex.

4. **Stochastic hill‑climb** – a cheap pair‑wise mass‑transfer hill‑climber
   finishes the search, accepting only moves that improve the true
   (unsmoothed) score.

All stages stop automatically when the allocated wall‑clock budget
(≈ 1000 s) is exhausted.  The final sequence is returned as a Python list
of floats, ready to be fed to `evaluate_sequence`.
"""

import time
import heapq
import numpy as np

# ----------------------------------------------------------------------
# Linear‑algebra helpers
# ----------------------------------------------------------------------


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    n = v.shape[0]
    if np.all(v <= 0):
        return np.full(n, 1.0 / n, dtype=float)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho = np.where(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full convolution using a real FFT (fast)."""
    m = len(a) + len(b) - 1
    n_fft = next_pow2(m)
    A = np.fft.rfft(a, n=n_fft)
    B = np.fft.rfft(b, n=n_fft)
    C = A * B
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


def compute_score(a: np.ndarray) -> float:
    """Exact score used by `evaluate_sequence` (sum is assumed = 1)."""
    a = np.asarray(a, dtype=float)
    n = a.shape[0]
    conv = convolution_fft(a, a)
    max_val = float(np.max(conv))
    # sum(a) is 1 after projection; guard against near‑zero just in case.
    s = float(np.sum(a))
    if s < 1e-12:
        return float("inf")
    return 2.0 * n * max_val / (s * s)


# ----------------------------------------------------------------------
# Families of candidate step‑functions
# ----------------------------------------------------------------------


def beta_sequence(n: int, alpha: float, beta: float | None = None) -> np.ndarray:
    """Beta‑type coefficients a_i ∝ (i+½)^α·(n‑½‑i)^β."""
    if beta is None:
        beta = alpha
    i = np.arange(n, dtype=float)
    left = np.power(i + 0.5, alpha)
    right = np.power(n - 0.5 - i, beta)
    seq = left * right
    # Clip pathological spikes – values >1e12 are irrelevant after normalisation.
    seq = np.clip(seq, 0.0, 1e12)
    total = np.sum(seq)
    if total == 0.0:
        seq = np.full(n, 1.0 / n)
    else:
        seq = seq / total
    return seq


def arcsine_sequence(n: int) -> np.ndarray:
    """Arcsine shape a_i ∝ 1 / sqrt((i+½)(n‑½‑i))."""
    i = np.arange(n, dtype=float)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - 0.5 - i))
    return vals / np.sum(vals)


def exponential_sequence(n: int, lam: float) -> np.ndarray:
    """a_i ∝ exp(-lam·|i - (n‑1)/2|)."""
    i = np.arange(n, dtype=float)
    centre = (n - 1) / 2.0
    seq = np.exp(-lam * np.abs(i - centre))
    return seq / np.sum(seq)


def power_transform(seq: np.ndarray, p: float) -> np.ndarray:
    """Raise each entry to power p and renormalise."""
    if p == 1.0:
        return seq
    raised = np.power(seq, p)
    total = np.sum(raised)
    if total == 0.0:
        return np.full_like(seq, 1.0 / len(seq))
    return raised / total


def upsample_sequence(seq: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation of a probability vector to a larger size."""
    old_n = len(seq)
    old_x = (np.arange(old_n) + 0.5) / old_n
    new_x = (np.arange(new_n) + 0.5) / new_n
    new_vals = np.interp(new_x, old_x, seq)
    return new_vals / np.sum(new_vals)


# ----------------------------------------------------------------------
# Sub‑gradient utilities for the true max‑objective
# ----------------------------------------------------------------------


def compute_subgradient(a: np.ndarray) -> np.ndarray:
    """
    Sub‑gradient of max_k (a∗a)_k.
    Uses all offsets attaining the (numerical) maximum.
    """
    n = a.shape[0]
    conv = convolution_fft(a, a)               # length 2n‑1
    max_val = conv.max()
    # tolerance – a tiny absolute part plus a relative part
    eps = max_val * 1e-12 + 1e-15
    active_idx = np.where(conv >= max_val - eps)[0]
    offsets = active_idx - (n - 1)               # true shift values
    g = np.zeros_like(a)

    for off in offsets:
        k = abs(int(off))
        if k == 0:
            g += 2.0 * a                        # derivative of Σ a_i²
        else:
            # contributions for i where i+k < n
            g[: n - k] += a[k:]                  # ∂/∂a_i term
            g[k: ]     += a[: n - k]             # ∂/∂a_{i+k} term
    if len(offsets) > 1:
        g /= len(offsets)                        # any convex combination is valid
    return g


def subgradient_descent(
    a0: np.ndarray,
    deadline: float,
    max_iters: int = 40_000,
    eta0: float = 0.25,
) -> tuple[np.ndarray, float]:
    """
    Projected sub‑gradient descent on the true max‑objective.
    Returns the best vector encountered and its (unsmoothed) score.
    """
    a = project_to_simplex(a0.astype(float).copy())
    best_a = a.copy()
    best_score = compute_score(a)
    t = 1

    while time.time() < deadline - 0.5 and t <= max_iters:
        g = compute_subgradient(a)
        eta = eta0 / np.sqrt(t)          # diminishing step
        a_new = a - eta * g

        # occasional tiny Gaussian jitter to help escape shallow plateaus
        if t % 500 == 0:
            a_new += np.random.randn(a.shape[0]) * 1e-5

        a = project_to_simplex(a_new)

        cur_score = compute_score(a)
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
        t += 1

    return best_a, best_score


# ----------------------------------------------------------------------
# Cheap stochastic hill‑climbing (pair‑wise mass transfer)
# ----------------------------------------------------------------------


def hill_climb(
    seq,
    time_deadline: float,
    max_steps: int = 30_000,
    init_step: float = 0.05,
):
    """
    Simple stochastic hill‑climb that moves a fraction of mass from one
    index to another and keeps the move only if the true objective improves.
    """
    a = np.array(seq, dtype=float)
    a = project_to_simplex(a)
    best_a = a.copy()
    best_score = compute_score(best_a)
    n = len(a)
    step = init_step

    while max_steps > 0 and time.time() < time_deadline:
        i, j = np.random.randint(n, size=2)
        if i == j:
            continue
        delta = step * best_a[i]
        if delta <= 0.0:
            continue
        cand = best_a.copy()
        cand[i] -= delta
        cand[j] += delta
        cand = project_to_simplex(cand)
        cur_score = compute_score(cand)
        if cur_score < best_score:
            best_score = cur_score
            best_a = cand
            step = min(step * 1.07, 0.5)   # accelerate when we succeed
        else:
            step *= 0.96                  # shrink when we fail
        max_steps -= 1
    return best_a.tolist(), best_score

# Main entry point
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step‑function minimising `evaluate_sequence`.
    Returns the best list of floats found before the time budget expires.
    """
    np.random.seed(seed)
    start = time.time()
    deadline = start + budget_s - 2.0          # small safety margin

    best_seq = None
    best_val = float("inf")

    # ------------------------------------------------------------------
    # Optional fallback to a previously discovered construction.
    # ------------------------------------------------------------------
    try:
        fallback = [float(x) for x in height_sequence_1]   # type: ignore
        fallback_val = evaluate_sequence(fallback)
        print(f"Fallback score = {fallback_val:.6f}")
        if fallback_val < best_val:
            best_val = fallback_val
            best_seq = fallback
    except Exception:
        pass

    # ------------------------------------------------------------------
    # Phase 1 – exhaustive parametric sweep (n = 2048)
    # ------------------------------------------------------------------
    n_search = 2048
    print("\n=== Phase 1 – parametric sweep (n = %d) ===" % n_search)

    # Helper to update global best
    def try_candidate(arr: np.ndarray, label: str):
        nonlocal best_seq, best_val
        sc = compute_score(arr)
        if sc < best_val:
            best_val = sc
            best_seq = arr.tolist()
            print(f"  New best [{label}]  score = {sc:.6f}")

    # 1) Symmetric Beta family
    alphas = np.linspace(-2.5, 2.5, 71)          # step ≈0.071
    for a in alphas:
        if time.time() >= deadline - 5:
            break
        seq = beta_sequence(n_search, a)
        try_candidate(seq, f"βsym a={a:.3f}")

    # 2) Asymmetric random Beta pairs
    for _ in range(800):
        if time.time() >= deadline - 5:
            break
        a = np.random.uniform(-2.5, 2.5)
        b = np.random.uniform(-2.5, 2.5)
        seq = beta_sequence(n_search, a, b)
        try_candidate(seq, f"βasym a={a:.3f} b={b:.3f}")

    # 3) Arcsine shape and its powered variants
    arcs = arcsine_sequence(n_search)
    try_candidate(arcs, "arcsine")
    powers = np.linspace(0.2, 3.0, 28)
    for p in powers:
        if time.time() >= deadline - 5:
            break
        seq = power_transform(arcs, p)
        try_candidate(seq, f"arcs^p={p:.3f}")

    # 4) Linear mixture of arcsine and uniform
    uniform = np.full(n_search, 1.0 / n_search)
    for w in np.linspace(0.0, 1.0, 21):
        if time.time() >= deadline - 5:
            break
        seq = (1.0 - w) * arcs + w * uniform
        seq = seq / np.sum(seq)
        try_candidate(seq, f"mix w={w:.2f}")

    # 5) Exponential decay family
    for lam in np.logspace(-4, -1, 20):
        if time.time() >= deadline - 5:
            break
        seq = exponential_sequence(n_search, lam)
        try_candidate(seq, f"exp λ={lam:.5f}")

    # ------------------------------------------------------------------
    # Phase 2 – upsample to a larger size (8192) if time permits.
    # ------------------------------------------------------------------
    large_n = 8192
    if best_seq is not None and time.time() < deadline - 10:
        print(f"\n=== Phase 2 – upsampling to n = {large_n} ===")
        best_arr = np.array(best_seq, dtype=float)
        up_seq = upsample_sequence(best_arr, large_n)
        up_score = compute_score(up_seq)
        print(f"Upsampled score = {up_score:.6f}")
        if up_score < best_val:
            best_val = up_score
            best_seq = up_seq.tolist()

    # ------------------------------------------------------------------
    # Phase 3 – sub‑gradient descent on the upsampled candidate.
    # ------------------------------------------------------------------
    if best_seq is not None and time.time() < deadline - 8:
        n_cur = len(best_seq)
        print(f"\n=== Phase 3 – sub‑gradient descent (n = {n_cur}) ===")
        init_a = np.array(best_seq, dtype=float)
        refined_a, refined_score = subgradient_descent(
            init_a,
            deadline,
            max_iters=60_000,
            eta0=0.25,
        )
        print(f"Sub‑gradient best score = {refined_score:.6f}")
        if refined_score < best_val:
            best_val = refined_score
            best_seq = refined_a.tolist()

    # ------------------------------------------------------------------
    # Phase 4 – final cheap hill‑climb refinement.
    # ------------------------------------------------------------------
    if best_seq is not None and time.time() < deadline - 2:
        print("\n=== Phase 4 – final hill‑climb refinement ===")
        refined_seq, refined_val = hill_climb(
            best_seq,
            time_deadline=deadline,
            max_steps=50_000,
            init_step=0.05,
        )
        print(f"Hill‑climb score = {refined_val:.6f}")
        if refined_val < best_val:
            best_val = refined_val
            best_seq = refined_seq

    # ------------------------------------------------------------------
    # Fallback in the extremely unlikely case that no candidate was built.
    # ------------------------------------------------------------------
    if best_seq is None:
        best_seq = [1.0] * 256
        best_val = evaluate_sequence(best_seq)

    final_score = evaluate_sequence(best_seq)
    print(f"\nSearch finished. Final reported score = {final_score:.6f}")
    return best_seq
