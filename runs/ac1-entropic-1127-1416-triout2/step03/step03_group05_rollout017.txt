"""
Search for a step‑function sequence that minimizes

    2·n·max(conv(a,a)) / (sum(a)**2)

using a smooth‑max surrogate for the max‑convolution and Adam
updates on logits (soft‑max parametrisation).  Several diverse
initialisers (uniform, Dirichlet, geometric, asymmetric geometric,
power‑law) are tried; each restart runs a limited‐time Adam
optimisation with an annealing “temperature’’ β that gradually
tightens the smooth‑max to the true max.  The best sequence
encountered before the time budget expires is returned as a plain
list of floats.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper: smooth max of aperiodic convolution and its gradient
# ----------------------------------------------------------------------


def _smooth_max_and_grad(a: np.ndarray,
                         beta: float,
                         fft_len_conv: int,
                         fft_len_grad: int) -> tuple[float, np.ndarray]:
    """
    Returns (max_val, grad_obj) where:
      - max_val = max(a * a)  (true max, not the smooth approximation)
      - grad_obj = ∂/∂a [ 2·n·max_val ]  using a smooth max with temperature β
    The returned gradient already includes the outer factor 2·n.
    """
    n = a.shape[0]

    # ---- convolution a*a -------------------------------------------------
    fa = np.fft.rfft(a, n=fft_len_conv)
    conv = np.fft.irfft(fa * fa, n=fft_len_conv)[: 2 * n - 1]

    max_val = float(conv.max())

    # ---- smooth‑max weights -----------------------------------------------
    # w_k ∝ exp((conv_k - max)/β)  ; the subtraction avoids overflow
    w = np.exp((conv - max_val) / beta)
    sum_w = w.sum()
    w_norm = w / sum_w                     # Σ w_norm = 1

    # ---- gradient of smooth‑max -------------------------------------------
    # ∂max/∂a_i = Σ_k w_norm_k * ∂conv_k/∂a_i ,   ∂conv_k/∂a_i = 2·a_{k-i}
    # The sum over k can be performed as a convolution of a_rev with w_norm.
    a_rev = a[::-1]

    fa_rev = np.fft.rfft(a_rev, n=fft_len_grad)
    fw = np.fft.rfft(w_norm, n=fft_len_grad)
    corr = np.fft.irfft(fa_rev * fw, n=fft_len_grad)[: 3 * n - 2]

    # slice the part that aligns with a_i (see analysis derivation)
    start = n - 1
    grad_temp = corr[start:start + n]      # Σ_k w_norm_k * a_{k-i}   (no factor 2)

    # full gradient of the original objective:
    #   grad_obj = 2·n * 2 * grad_temp = 4·n * grad_temp
    grad_obj = 4.0 * n * grad_temp
    return max_val, grad_obj


# ----------------------------------------------------------------------
# Helper: generate initial probability vectors (sum = 1, non‑negative)
# ----------------------------------------------------------------------


def _initial_sequence(kind: str,
                     n: int,
                     rng: np.random.Generator,
                     prev: np.ndarray | None) -> np.ndarray:
    """Return a feasible start vector (Σ a = 1)."""

    if kind == "prev" and prev is not None:
        seq = prev.astype(float, copy=True)
        if seq.shape[0] != n:
            if seq.shape[0] > n:
                seq = seq[:n]
            else:
                seq = np.concatenate([seq,
                                      np.zeros(n - seq.shape[0])])
        seq = np.maximum(seq, 0.0)
        tot = seq.sum()
        return seq / tot if tot > 0 else np.full(n, 1.0 / n)

    if kind == "uniform":
        return np.full(n, 1.0 / n)

    if kind == "random_dirichlet":
        # α < 1 gives a sparser Dirichlet
        alpha = 0.5
        sample = rng.gamma(alpha, 1.0, size=n)
        return sample / sample.sum()

    if kind == "geometric":
        # symmetric geometric decay from both ends
        r = rng.uniform(0.94, 0.99)
        left = r ** np.arange(n)
        right = r ** np.arange(n - 1, -1, -1)
        seq = left + right
        return seq / seq.sum()

    if kind == "asymmetric_geometric":
        # decay only from the left side (mass near the start)
        r = rng.uniform(0.88, 0.99)
        seq = r ** np.arange(n)
        return seq / seq.sum()

    if kind == "power_law":
        # a_i ∝ (i+1)^{-p}
        p = rng.uniform(0.3, 1.8)
        idx = np.arange(1, n + 1)
        seq = 1.0 / (idx ** p)
        return seq / seq.sum()

    # fallback
    return np.full(n, 1.0 / n)


# ----------------------------------------------------------------------
# Adam optimiser using the smooth‑max surrogate
# ----------------------------------------------------------------------


def _adam_optimize_smooth(init_a: np.ndarray,
                          deadline: float,
                          restart_id: int,
                          target: float = 1.503,
                          beta_init: float = 0.02,
                          beta_min: float = 0.001) -> tuple[np.ndarray, float]:
    """
    Adam optimisation on the logits of a probability vector.
    Returns the best vector (Σ a=1) and its objective value seen
    before `deadline`.
    """
    n = init_a.shape[0]

    # FFT lengths (next powers of two for speed)
    fft_len_conv = 1 << ((2 * n - 2).bit_length())          # for a*a
    fft_len_grad = 1 << ((3 * n - 2).bit_length())          # for a_rev * w

    # logits (add tiny epsilon to avoid log(0))
    eps = 1e-12
    b = np.log(init_a + eps)

    # Adam state
    m = np.zeros_like(b)
    v = np.zeros_like(b)
    beta1, beta2 = 0.9, 0.999
    adam_eps = 1e-8

    # evaluate the start point
    max_val, _ = _smooth_max_and_grad(init_a,
                                      beta=beta_init,
                                      fft_len_conv=fft_len_conv,
                                      fft_len_grad=fft_len_grad)
    best_score = 2.0 * n * max_val
    best_a = init_a.copy()

    stagnation = 0
    it = 0
    beta = beta_init

    while time.time() < deadline:
        it += 1

        # ---------- 1) obtain a from logits (soft‑max) ----------
        b_max = np.max(b)
        a = np.exp(b - b_max)
        a /= a.sum()                     # Σ a = 1

        # ---------- 2) smooth‑max and its gradient ----------
        max_val, grad_obj = _smooth_max_and_grad(
            a, beta, fft_len_conv, fft_len_grad)
        cur_score = 2.0 * n * max_val      # true objective (using real max)

        # ---------- 3) keep the best ----------
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            stagnation = 0
            print(f"Restart {restart_id:02d} iter {it:6d}: "
                  f"best = {best_score:.6f}  beta={beta:.5f}")
        else:
            stagnation += 1

        # ---------- 4) termination ----------
        if best_score < target:
            break
        if stagnation > 5000:
            # random jitter in log‑space to escape plateaus
            b += 0.015 * np.random.randn(n)
            stagnation = 0

        # anneal β slowly towards beta_min
        beta = max(beta_min, beta * 0.9995)

        # ---------- 5) Adam update (logit space) ----------
        # grad_obj already accounts for the outer factor 2·n
        avg = np.dot(a, grad_obj)
        grad_b = a * (grad_obj - avg)          # Jacobian of soft‑max

        m = beta1 * m + (1.0 - beta1) * grad_b
        v = beta2 * v + (1.0 - beta2) * (grad_b ** 2)

        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        # learning‑rate schedule (large initially, then decays)
        lr = 0.6 / np.sqrt(it)

        b -= lr * m_hat / (np.sqrt(v_hat) + adam_eps)

    return best_a, best_score
