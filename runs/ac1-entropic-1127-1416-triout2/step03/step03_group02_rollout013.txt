def optimise_subgradient(initial_a, deadline, eta0=0.5, max_iter=200000, noise_every=0, noise_scale=0.0):
    a = np.maximum(initial_a, 1e-30)  # avoid zeros? but zeros okay.
    a = a / np.sum(a)  # normalize to sum=1
    best_a = a.copy()
    best_score = compute_score(a)
    n = len(a)
    for it in range(1, max_iter+1):
        if time.time() >= deadline:
            break
        conv = convolution_fft(a, a)
        max_val = conv.max()
        # find indices of max (allow tolerance)
        mask = np.isclose(conv, max_val, rtol=0, atol=1e-12)
        max_inds = np.where(mask)[0]
        # gradient
        grad = np.zeros_like(a)
        for k in max_inds:
            # compute overlapping region
            i_start = max(0, k - (n - 1))
            i_end = min(k, n - 1)
            # vector of indices j = i_start .. i_end
            j = np.arange(i_start, i_end + 1)
            grad[j] += 2.0 * a[k - j]
        # step size
        eta = eta0 / np.sqrt(it)
        # multiplicative update
        a = a * np.exp(-eta * grad)
        # optional noise
        if noise_every and (it % noise_every == 0):
            a += np.random.randn(n) * noise_scale
            a = np.maximum(a, 0.0)
        # renormalize to simplex
        s = a.sum()
        if s == 0:
            a = np.full_like(a, 1.0 / n)
        else:
            a = a / s
        # evaluate internal
        cur_score = compute_score(a)
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
        # optional progress printing maybe each few thousand
        if it % 2000 == 0:
            print(f"        subgrad iter={it:,d} cur={cur_score:.6f} best={best_score:.6f}")
    return best_a, best_score
```

"""
Hybrid optimisation for the step‑function min‑max convolution problem.

The search proceeds in three phases:

1. **Exploration** –  Several sequence lengths are tried (powers of two).
   For each length a diverse library of initial shapes is generated
   (uniform, arcsine, power‑law, beta‑like, Dirichlet, random perturbations).
   The cheap internal score `compute_score` is used to keep the most promising
   candidates.

2. **Sub‑gradient descent on the true max‑convolution** –  Starting from each
   retained candidate we run a specialised exponentiated‑gradient (mirror‑descent)
   method that directly minimises the non‑smooth objective
        Φ(a)=max_k (a∗a)_k .
   The gradient of a single active offset is known analytically, which allows an
   O(n) update per iteration (the convolution needed only to locate the
   maximal offset).  The update is multiplicative, guaranteeing non‑negativity,
   and we re‑normalise onto the probability simplex after each step.
   A simple 1/√t step‑size schedule and occasional tiny Gaussian noise help
   escape shallow plateaus.

3. **Final polishing** –  The globally best sequence found is refined a
   second time with a longer sub‑gradient run and a very small learning rate,
   after which it is returned (as a plain list of floats).

The whole routine respects the supplied time budget (default 1000 s) and
always returns a feasible non‑negative sequence.
"""

import time
import numpy as np
from typing import List, Tuple

# ----------------------------------------------------------------------
# Low‑level numeric utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Linear convolution via real FFT (full length, zero‑padded to power‑of‑2)."""
    m = len(a) + len(b) - 1
    n_fft = next_pow2(m)
    A = np.fft.rfft(a, n=n_fft)
    B = np.fft.rfft(b, n=n_fft)
    C = A * B
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


def compute_score(a: np.ndarray) -> float:
    """
    Official objective (scale‑invariant):
        2·n·max(a*a) / (sum(a)²)
    a is assumed non‑negative; the function works for any vector.
    """
    n = len(a)
    conv = convolution_fft(a, a)
    max_val = float(np.max(conv))
    s = float(np.sum(a))
    if s < 1e-12:
        return float("inf")
    return 2.0 * n * max_val / (s * s)


# ----------------------------------------------------------------------
# Families of initial shapes
# ----------------------------------------------------------------------


def arcsine_initial(n: int) -> np.ndarray:
    """Discrete arcsine density (normalised to sum = 1)."""
    i = np.arange(n, dtype=float)
    eps = 0.25
    vals = 1.0 / np.sqrt((i + 0.5) * (n - 0.5 - i) + eps)
    return vals / vals.sum()


def generate_powerlaw(n: int, p: float, eps: float) -> np.ndarray:
    """Symmetric power‑law a_i ∝ ((i+ε)(n‑1‑i+ε))⁻ᵖ."""
    i = np.arange(n, dtype=float)
    base = ((i + eps) * (n - 1 - i + eps)) ** (-p)
    total = base.sum()
    if total == 0.0:
        return np.full(n, 1.0 / n)
    return base / total


def generate_beta_like(n: int, alpha: float, beta: float) -> np.ndarray:
    """Beta‑like shape a_i ∝ (i+½)^{α‑1} (n‑1‑i+½)^{β‑1}."""
    i = np.arange(n, dtype=float)
    base = (i + 0.5) ** (alpha - 1) * (n - 1 - i + 0.5) ** (beta - 1)
    total = base.sum()
    if total == 0.0:
        return np.full(n, 1.0 / n)
    return base / total


def random_dirichlet(n: int, concentration: float, rng: np.random.Generator) -> np.ndarray:
    """Sample a Dirichlet‑like vector (exponential with scale 1 then normalise)."""
    if concentration <= 0:
        # heavy‑tailed “sparse” draws
        samp = rng.exponential(1.0, size=n) ** (1.0 / max(0.1, concentration))
    else:
        # standard gamma draws
        samp = rng.gamma(concentration, 1.0, size=n)
    samp += 1e-12
    return samp / samp.sum()


def perturb(a: np.ndarray, rng: np.random.Generator, scale: float = 0.02) -> np.ndarray:
    """Add a tiny random perturbation and renormalise."""
    b = a + rng.normal(0.0, scale, size=a.shape)
    b = np.maximum(b, 0.0)
    s = b.sum()
    if s == 0:
        return np.full_like(a, 1.0 / len(a))
    return b / s


# ----------------------------------------------------------------------
# Sub‑gradient optimiser (direct minimisation of the max‑convolution)
# ----------------------------------------------------------------------


def _max_conv_gradient(a: np.ndarray, conv: np.ndarray) -> np.ndarray:
    """
    Compute a sub‑gradient of max_k (a∗a)_k.
    The gradient for a particular active offset k is:
        g_j = 2·a_{k‑j}   (when the index is valid)
    If several offsets share the maximal value, we average their gradients.
    """
    n = len(a)
    max_val = conv.max()
    # tolerance for “active” offsets – absolute because values are O(1/n)
    mask = np.isclose(conv, max_val, atol=1e-15, rtol=0.0)
    active = np.where(mask)[0]
    grad = np.zeros_like(a)

    for k in active:
        lo = max(0, k - (n - 1))
        hi = min(k, n - 1)
        idx = np.arange(lo, hi + 1, dtype=int)
        grad[idx] += 2.0 * a[k - idx]

    # average if more than one active offset
    if len(active) > 1:
        grad /= len(active)
    return grad


def optimise_subgradient(
    initial_a: np.ndarray,
    deadline: float,
    *,
    eta0: float = 0.6,
    max_iter: int = 200_000,
    noise_every: int = 0,
    noise_scale: float = 0.0,
) -> Tuple[np.ndarray, float]:
    """
    Sub‑gradient (exponentiated) descent on the true max‑convolution.
    Returns the best vector seen and its (non‑smooth) score.
    """
    a = np.maximum(initial_a, 1e-30)
    a = a / a.sum()               # work on the probability simplex
    best_a = a.copy()
    best_sc = compute_score(a)

    n = len(a)

    for it in range(1, max_iter + 1):
        if time.time() >= deadline:
            break

        # current convolution and sub‑gradient
        conv = convolution_fft(a, a)
        grad = _max_conv_gradient(a, conv)

        # step size schedule η_t = η0 / sqrt(t)
        eta = eta0 / np.sqrt(it)

        # multiplicative (exponentiated) update
        a = a * np.exp(-eta * grad)

        # optional exploration noise
        if noise_every and (it % noise_every == 0):
            a += np.random.randn(n) * noise_scale
            a = np.maximum(a, 0.0)

        # re‑project onto the simplex
        s = a.sum()
        if s == 0.0:
            a = np.full_like(a, 1.0 / n)
        else:
            a = a / s

        cur_sc = compute_score(a)
        if cur_sc < best_sc:
            best_sc = cur_sc
            best_a = a.copy()

        if it % 2000 == 0:
            print(f"    subgrad iter={it:,d} cur={cur_sc:.6f} best={best_sc:.6f}")

    return best_a, best_sc

# Main entry point required by the competition harness

def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs) -> List[float]:
    """
    Search for a non‑negative step‑function with a low ``evaluate_sequence`` value.
    The routine explores a few sequence lengths, generates a diverse set of
    initial candidates, refines them with a direct sub‑gradient method and
    finally polishes the global best.  It always returns a feasible list of
    floats before the wall‑clock budget expires.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    # keep a tiny safety margin for the final evaluation call
    deadline = start_time + budget_s - 2.0

    best_seq: List[float] = []
    best_val: float = float("inf")

    # ------------------------------------------------------------------
    # 0.  Optional fallback from a previously discovered construction.
    # ------------------------------------------------------------------
    try:
        fallback = [float(x) for x in height_sequence_1]  # type: ignore
        val = evaluate_sequence(fallback)
        print(f"Fallback (height_sequence_1) score = {val:.6f}")
        if val < best_val:
            best_val = val
            best_seq = fallback
    except NameError:
        pass

    # ------------------------------------------------------------------
    # 1.  Sequence lengths to explore (powers of two give cheap FFTs).
    # ------------------------------------------------------------------
    lengths = [1024, 2048, 4096]

    for n in lengths:
        if time.time() >= deadline - 5.0:
            break
        print(f"\n=== Exploring length n = {n} ===")

        # --------------------------------------------------------------
        # 1a.  Build a modest library of candidate start vectors.
        # --------------------------------------------------------------
        candidates: List[np.ndarray] = []

        # (i) uniform
        candidates.append(np.full(n, 1.0 / n))

        # (ii) classic arcsine
        candidates.append(arcsine_initial(n))

        # (iii) random power‑law shapes
        for _ in range(5):
            p = rng.uniform(0.1, 1.0)
            eps = rng.uniform(0.001, max(1.0, 0.02 * n))
            candidates.append(generate_powerlaw(n, p, eps))

        # (iv) beta‑like shapes
        for _ in range(3):
            alpha = rng.uniform(0.2, 5.0)
            beta = rng.uniform(0.2, 5.0)
            candidates.append(generate_beta_like(n, alpha, beta))

        # (v) Dirichlet / exponential draws (both dense and sparse)
        for conc in [0.5, 1.0, 2.0]:
            candidates.append(random_dirichlet(n, conc, rng))

        # (vi) small random perturbations of arcsine – helps to escape local minima
        for _ in range(2):
            cand = perturb(arcsine_initial(n), rng, scale=0.02)
            candidates.append(cand)

        # --------------------------------------------------------------
        # 1b.  Quick sweep with the fast internal score, keep the best few.
        # --------------------------------------------------------------
        scored = [(compute_score(c), c) for c in candidates]
        scored.sort(key=lambda x: x[0])
        top_candidates = [c for _, c in scored[:4]]   # keep the four most promising

        # --------------------------------------------------------------
        # 1c.  Refine each retained candidate with sub‑gradient descent.
        # --------------------------------------------------------------
        for idx, cand in enumerate(top_candidates):
            if time.time() >= deadline - 5.0:
                break
            print(f"  candidate {idx + 1}/{len(top_candidates)} (n={n})")

            # allocate a slice of the remaining budget
            remaining = max(0.0, deadline - time.time())
            # a very rough estimate: ~300 iterations per second for n≈2000
            max_iter = int(min(120_000, max(2_000, remaining * 300)))

            refined, _ = optimise_subgradient(
                initial_a=cand,
                deadline=deadline,
                eta0=0.6,
                max_iter=max_iter,
                noise_every=500,
                noise_scale=5e-6,
            )

            # final evaluation with the official evaluator
            val = evaluate_sequence(refined.tolist())
            print(f"    evaluated score = {val:.6f}")

            if val < best_val:
                best_val = val
                best_seq = refined.tolist()
                print(f"    *** New global best: {best_val:.6f} (n={n}) ***")

    # ------------------------------------------------------------------
    # 2.  Final polishing of the overall best sequence (if time permits).
    # ------------------------------------------------------------------
    if best_seq and time.time() < deadline - 5.0:
        a = np.array(best_seq, dtype=float)
        remaining = max(0.0, deadline - time.time())
        max_iter = int(min(200_000, remaining * 300))

        print("\nFinal polishing of the global best")
        a_refined, _ = optimise_subgradient(
            initial_a=a,
            deadline=deadline,
            eta0=0.3,                 # smaller steps for fine‑tuning
            max_iter=max_iter,
            noise_every=0,
            noise_scale=0.0,
        )
        final_val = evaluate_sequence(a_refined.tolist())
        print(f"Final polished score = {final_val:.6f}")
        if final_val < best_val:
            best_val = final_val
            best_seq = a_refined.tolist()
            print("*** Updated global best after final polish ***")

    # ------------------------------------------------------------------
    # 3.  Guard – always return a feasible sequence.
    # ------------------------------------------------------------------
    if not best_seq:
        best_seq = [1.0] * 256
        best_val = evaluate_sequence(best_seq)

    print(f"\nSearch finished. Best reported score = {best_val:.6f}")
    return best_seq
