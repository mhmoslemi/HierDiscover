            # update temperature schedule
            # temperature decays linearly or exponentially
            t_frac = it / total_iters

def _grad_weighted(a, p):
    n = a.shape[0]
    L = n + (2*n-1) - 1  # =3n-2
    L = 1<< (L-1).bit_length()  # next power of 2
    A = np.fft.rfft(a, n=L)
    P = np.fft.rfft(p, n=L)
    # correlation via conj of P
    corr = np.fft.irfft(A * np.conj(P), n=L)
    offset = (len(p) - 1)  # =2n-2
    grad = 2.0 * corr[offset: offset + n]
    return grad
```

"""
Simulated‑annealing + mirror‑descent hybrid optimiser.

The routine first tries a simple stochastic hill‑climbing (simulated annealing)
that moves a small fraction of mass between two coordinates while keeping the
vector on the probability simplex.  The objective that is minimised is

        2·n·max(conv(a,a))          (sum(a) = 1)

where `conv` is the linear convolution.  Convolution is evaluated via an
FFT for speed.  After roughly half of the time budget the best vector found
by the annealer is fed to a mirror‑descent optimiser (the same exponentiated‑
gradient scheme used in the reference implementation) to perform a fine
tuning phase.  The best sequence seen overall is returned.

All helper functions are defined at the top level, no closures are used.
"""

import time
import numpy as np


def _project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of a vector onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1] - 1
    theta = cssv[cond][-1] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def _max_and_subgradient(a: np.ndarray) -> tuple[float, np.ndarray]:
    """
    Return (max entry of a*a, a sub‑gradient for that entry).

    The gradient is the average of the sub‑gradients of all active
    (max‑attaining) convolution indices.
    """
    n = a.shape[0]
    fft = np.fft.rfft(a, n=2 * n - 1)
    conv = np.fft.irfft(fft * fft, n=2 * n - 1)
    max_val = float(conv.max())

    # tolerance for “active’’ indices
    tol = max_val * 1e-12 + 1e-14
    active = np.where(conv >= max_val - tol)[0]

    grad = np.zeros_like(a)
    for k in active:
        low = max(0, k - n + 1)
        high = min(n - 1, k)
        idx = np.arange(low, high + 1, dtype=int)
        partner = k - idx
        grad[idx] += 2.0 * a[partner]

    if active.size > 1:
        grad /= float(active.size)
    return max_val, grad


def _objective_fft(a: np.ndarray, n: int, L: int) -> float:
    """
    Fast evaluation of 2·n·max(conv(a,a)) with sum(a)=1.
    `L` is the FFT length (power‑of‑two ≥ 2·n‑1).
    """
    fft = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(fft * fft, n=L)[: 2 * n - 1]
    return 2.0 * n * float(conv.max())


def _annealing_optimize(
    a0: np.ndarray, n: int, deadline: float
) -> tuple[np.ndarray, float]:
    """
    Simple simulated‑annealing optimiser.  Keeps the vector on the simplex
    by moving a fraction of mass from one index to another.
    """
    L = 1 << ((2 * n - 1).bit_length())      # FFT size (next power of two)
    a = a0.copy()
    cur_obj = _objective_fft(a, n, L)
    best_a = a.copy()
    best_obj = cur_obj
    rng = np.random.default_rng()
    start = time.time()
    it = 0

    while time.time() < deadline:
        elapsed = time.time() - start
        prog = elapsed / (deadline - start)           # 0 … 1
        # linear schedules for temperature and step‑fraction
        T = 0.5 * (1.0 - prog) + 1e-6 * prog
        step_frac = 0.5 * (1.0 - prog) + 1e-6 * prog

        i, j = rng.integers(n, size=2)
        if i == j:
            continue

        # transfer a random fraction (≤ step_frac) of the mass at i to j
        p = step_frac * rng.random()
        delta = p * a[i]
        a_new = a.copy()
        a_new[i] -= delta
        a_new[j] += delta

        new_obj = _objective_fft(a_new, n, L)

        # Metropolis acceptance
        if new_obj < cur_obj or rng.random() < np.exp(-(new_obj - cur_obj) / (T + 1e-12)):
            a = a_new
            cur_obj = new_obj
            if cur_obj < best_obj:
                best_obj = cur_obj
                best_a = a.copy()
                print(
                    f"[{time.time():.1f}] SA new best = {best_obj:.6f} (iter {it})"
                )

        # occasional jitter to escape shallow plateaus
        if it % 5000 == 0 and it > 0:
            if rng.random() < 0.02:
                noise = rng.normal(scale=step_frac * 0.01, size=n)
                a = np.maximum(a + noise, 0.0)
                a /= a.sum()
                cur_obj = _objective_fft(a, n, L)
                if cur_obj < best_obj:
                    best_obj = cur_obj
                    best_a = a.copy()
                    print(
                        f"[{time.time():.1f}] SA jitter new best = {best_obj:.6f}"
                    )
        it += 1

    return best_a, best_obj


def _mirror_descent_optimize(
    a0: np.ndarray, n: int, deadline: float, lower_bound: float = 1.0
) -> tuple[np.ndarray, float]:
    """
    Mirror‑descent optimiser identical in spirit to the reference code but
    with a few tuned parameters (larger max step, occasional jitter).
    """
    best_a: np.ndarray | None = None
    best_score = float("inf")
    restart = 0

    while time.time() < deadline:
        rng = np.random.default_rng(
            seed=int(time.time() * 1e6) % (2**32 - 1)
        )
        if a0 is not None and restart == 0:
            a = a0.copy()
        else:
            a = rng.random(n)

        a = np.maximum(a, 1e-30)
        a /= a.sum()
        g_cum = np.zeros_like(a)

        it = 0
        while time.time() < deadline:
            max_conv, grad = _max_and_subgradient(a)
            cur_score = 2.0 * n * max_conv  # sum(a) == 1

            if cur_score < best_score:
                best_score = cur_score
                best_a = a.copy()
                print(
                    f"[{time.time():.1f}] Mirror new best = {best_score:.6f}"
                )
                if best_score < 1.503:
                    return best_a, best_score

            norm_sq = float(np.dot(grad, grad))
            if norm_sq == 0.0:
                break

            # Polyak step size (clipped)
            step = (cur_score - lower_bound) / (norm_sq + 1e-12)
            step = max(1e-12, min(step, 10.0))   # a slightly larger clip

            g_cum += step * grad

            # Exponentiated‑gradient update (mirror descent)
            tmp = -g_cum
            tmp -= np.max(tmp)  # numerical stability
            a = np.exp(tmp)
            s = a.sum()
            if s == 0.0:
                a = np.full(n, 1.0 / n)
            else:
                a /= s

            # occasional stochastic jitter
            if it % 4000 == 0 and it > 0:
                eps = 0.001 * rng.standard_normal(n)
                a = np.maximum(a + eps, 0.0)
                a /= a.sum()

            it += 1

        restart += 1

    if best_a is None:
        best_a = np.full(n, 1.0 / n)
    return best_a, best_score


def propose_candidate(budget_s: int = 1000) -> list[float]:
    """
    Search for a step‑function (non‑negative heights) that minimises
    `evaluate_sequence`.  The search runs for at most ``budget_s`` seconds
    and returns the best sequence found.
    """
    deadline = time.time() + budget_s - 0.5          # tiny safety margin

    # ------------------------------------------------------------------
    # 1. Load an initial construction (if available) and set problem size.
    # ------------------------------------------------------------------
    try:
        init = np.asarray(height_sequence_1, dtype=float)    # type: ignore[name-defined]
        n = init.shape[0]
    except Exception:

        n = 2048
        init = np.random.random(n)

    init = np.maximum(init, 0.0)
    if init.sum() == 0.0:
        init = np.random.random(n)
    init = init / init.sum()

    # ------------------------------------------------------------------
    # 2. Split the time budget: half for annealing, half for mirror descent.
    # ------------------------------------------------------------------

    half_deadline = start_time + (deadline - start_time) * 0.5

    best_a, best_obj = _annealing_optimize(init, n, half_deadline)

    # ----- 2.2 Mirror‑descent fine‑tuning --------------------------------
    if time.time() < deadline:

            best_a, n, deadline, lower_bound=1.0
        )

    # ------------------------------------------------------------------
    # 3. Return the best sequence (as plain Python list).
    # ------------------------------------------------------------------
    return [float(v) for v in best_a.tolist()]
