    # v: array, maybe negative, sum to something.
    # Projects onto the simplex {x >=0, sum = 1}
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    rho = np.where(u - cssv / np.arange(1, len(v)+1) > 0)[0][-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0)
    return w

"""
Optimise a non‑negative step‑function by Adam on a smoothed max‑convolution
objective.  The true objective is

    score = 2·n·max(conv(a,a))   with   sum(a)=1 , a_i≥0 .

The max is replaced by a soft‑max (log‑sum‑exp) with a temperature β that is
raised linearly during the run.  This yields a dense, smooth gradient that
covers many convolution entries instead of only the active ones.  The
variables are represented by soft‑max logits, a Adam update is applied,
and after each step the logits are projected back onto the simplex via the
soft‑max.  Multiple short restarts (different random or geometric starts)
are performed until the time budget expires.  The best sequence seen
according to the *exact* objective is returned.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Simple “soft‑max” conversion (no external closure)
# ----------------------------------------------------------------------
def _softmax(logits: np.ndarray) -> np.ndarray:
    """Numerically stable soft‑max, returns a probability vector (sum = 1)."""
    m = np.max(logits)
    ex = np.exp(logits - m)
    return ex / ex.sum()


# ----------------------------------------------------------------------
# Initialisation helpers (same ideas as the previous code)
# ----------------------------------------------------------------------
def _initial_sequence(kind: str, n: int, rng: np.random.Generator,
                      prev: np.ndarray | None) -> np.ndarray:
    """Return a feasible probability vector of length n."""
    if kind == "prev" and prev is not None:
        seq = prev.copy()
        if seq.shape[0] > n:
            seq = seq[:n]
        elif seq.shape[0] < n:
            seq = np.concatenate([seq, np.zeros(n - seq.shape[0])])
        seq = np.maximum(seq, 0.0)
        s = seq.sum()
        return seq / s if s > 0 else np.full(n, 1.0 / n)

    if kind == "uniform":
        return np.full(n, 1.0 / n)

    if kind == "dirichlet":
        # α < 1 gives sparse vectors, good for exploration
        alpha = 0.4
        sample = rng.gamma(alpha, 1.0, size=n)
        return sample / sample.sum()

    if kind == "geometric":
        # symmetric geometric decay from both ends
        r = 0.98 + 0.02 * rng.random()
        left = r ** np.arange(n)
        right = r ** np.arange(n - 1, -1, -1)
        seq = left + right
        return seq / seq.sum()

    # fallback
    return np.full(n, 1.0 / n)


# ----------------------------------------------------------------------
# Core optimisation routine (Adam on a smoothed max‑convolution)
# ----------------------------------------------------------------------
def _adam_optimize_smooth(init_a: np.ndarray,
                          deadline: float,
                          restart_id: int,
                          target: float = 1.5) -> tuple[np.ndarray, float]:
    """
    Adam optimiser that minimises
        2·n·max(conv(a,a))
    using a soft‑max (log‑sum‑exp) surrogate for the max.
    Returns the best vector (sum = 1) found before *deadline* and its
    exact score.
    """
    n = init_a.shape[0]

    # ---------- FFT size (single size works for both convolutions) ----------
    L = 1 << ((3 * n - 2).bit_length())   # >= 3·n‑2, enough for all convolutions

    # ---------- logits (unconstrained) ----------
    eps_init = 1e-12
    logits = np.log(init_a + eps_init)

    # ---------- Adam state ----------
    m = np.zeros_like(logits)
    v = np.zeros_like(logits)
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8

    # ---------- bookkeeping ----------
    best_a = init_a.copy()
    # exact score of the start point
    a_fft = np.fft.rfft(best_a, L)
    conv_full = np.fft.irfft(a_fft * a_fft, n=L)[:2 * n - 1]
    best_score = float(2.0 * n * conv_full.max())
    stagnation = 0
    it = 0
    start_time = time.time()
    beta_start, beta_end = 5.0, 30.0   # temperature schedule

    while time.time() < deadline:
        it += 1

        # ------------------------------------------------------------------
        # 1) obtain probability vector from logits
        # ------------------------------------------------------------------
        a = _softmax(logits)

        # ------------------------------------------------------------------
        # 2) forward convolution a*a (full, length 2n‑1)
        # ------------------------------------------------------------------
        a_fft = np.fft.rfft(a, L)
        conv = np.fft.irfft(a_fft * a_fft, n=L)[:2 * n - 1]
        cur_score = float(2.0 * n * conv.max())

        # ------------------------------------------------------------------
        # 3) keep the best exact solution seen so far
        # ------------------------------------------------------------------
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            stagnation = 0
            print(f"Restart {restart_id:02d} it {it:6d}:   best = {best_score:.6f}")
        else:
            stagnation += 1

        # ------------------------------------------------------------------
        # 4) termination / early‑stop checks
        # ------------------------------------------------------------------
        if best_score < target:
            break
        if stagnation > 5000:                     # escape flat plateaus
            logits += 0.02 * np.random.randn(n)
            stagnation = 0

        # ------------------------------------------------------------------
        # 5) temperature (beta) – raised linearly with elapsed time
        # ------------------------------------------------------------------
        elapsed = time.time() - start_time
        total = deadline - start_time
        prog = min(1.0, elapsed / total) if total > 0 else 1.0
        beta = beta_start + (beta_end - beta_start) * prog

        # ------------------------------------------------------------------
        # 6) soft‑max weights for the smoothed max
        # ------------------------------------------------------------------
        shift = conv.max()               # for numerical stability
        w = np.exp(beta * (conv - shift))
        w = w / w.sum()                  # Σ w = 1  – this is the probability vector p_k

        # ------------------------------------------------------------------
        # 7) gradient of the smoothed max (vector of length n)
        #    ∇_a ( Σ_k w_k * conv_k ) = 2 * Σ_k w_k * a_{k‑i}
        #    This is a convolution of w with the reversed a.
        # ------------------------------------------------------------------
        a_rev = a[::-1]
        w_fft = np.fft.rfft(w, L)
        a_rev_fft = np.fft.rfft(a_rev, L)
        grad_full = np.fft.irfft(w_fft * a_rev_fft, n=L)[:3 * n - 2]
        # as derived: grad_i = 2 * conv_rev[:n][::-1]
        grad_smooth = 2.0 * grad_full[:n][::-1]          # shape (n,)

        # ------------------------------------------------------------------
        # 8) gradient of the *actual* objective (2·n·max) approximated by smooth version
        # ------------------------------------------------------------------
        grad_a = 2.0 * n * grad_smooth

        # ------------------------------------------------------------------
        # 9) map to logits (soft‑max Jacobian)
        # ------------------------------------------------------------------
        avg = np.dot(a, grad_a)
        grad_logits = a * (grad_a - avg)

        # ------------------------------------------------------------------
        # 10) Adam update on logits
        # ------------------------------------------------------------------
        m = beta1 * m + (1.0 - beta1) * grad_logits
        v = beta2 * v + (1.0 - beta2) * (grad_logits ** 2)

        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        lr = 0.5 / np.sqrt(it)                     # simple diminishing step size
        logits -= lr * m_hat / (np.sqrt(v_hat) + eps)

    return best_a, best_score
