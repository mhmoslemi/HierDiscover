def _anneal_optimize(init_a, deadline, restart_id, target=1.503):
    n = init_a.shape[0]
    # FFT lengths
    fft_len1 = 1 << ((2*n - 2).bit_length())
    fft_len2 = 1 << ((3*n - 2).bit_length())
    # logits
    eps = 1e-12
    b = np.log(init_a + eps)
    # Adam moments
    m = np.zeros_like(b)
    v = np.zeros_like(b)
    beta1, beta2 = 0.9, 0.999
    eps_adam = 1e-8
    # hyperparams
    alpha = 0.5   # initial smoothing
    alpha_max = 1000.0
    # store best
    a_curr = init_a
    # compute initial real objective
    conv_full = _full_conv_fft(a_curr, fft_len1) # we will compute inside loop anyway.
    max_val = conv_full.max()
    best_score = 2.0 * n * max_val
    best_a = a_curr.copy()
    stagnation = 0
    it = 0
    while time.time() < deadline:
        it += 1

        # softmax retrieval
        b_max = np.max(b)
        a = np.exp(b - b_max)
        a /= a.sum()

        # compute convolution and smoothed gradient
        # conv and fft of a (vector)
        fft_a = np.fft.rfft(a, n=fft_len1)
        conv = np.fft.irfft(fft_a * fft_a, n=fft_len1)[: 2*n - 1]
        max_conv_val = conv.max()
        cur_score = 2.0 * n * max_conv_val

        # update best
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            stagnation = 0
            print(f"Restart {restart_id:02d} iter {it:6d}: best = {best_score:.6f}")
        else:
            stagnation += 1

        # check termination conditions
        if best_score < target:
            break
        # plateau handling
        if stagnation > 5000:
            b += 0.02 * np.random.randn(n)
            stagnation = 0

        # compute smoothed gradient
        # weights:
        # scale exponent
        scaled = alpha * (conv - max_conv_val)  # shift by max for numeric stability
        w_raw = np.exp(scaled)
        w = w_raw / w_raw.sum()
        # weighted gradient using convolution
        w_rev = w[::-1]

        fft_w = np.fft.rfft(w_rev, n=fft_len2)
        fft_a2 = np.fft.rfft(a, n=fft_len2)
        weighted_full = np.fft.irfft(fft_a2 * fft_w, n=fft_len2)
        # get gradient slice
        start = 2 * n - 2
        grad_a = 2.0 * weighted_full[start - np.arange(n)]

        # gradient of objective (includes factor 2*n)
        grad_obj = 2.0 * n * grad_a

        # transform to logits gradient
        avg = np.dot(a, grad_obj)
        grad_b = a * (grad_obj - avg)

        # Adam updates
        m = beta1 * m + (1 - beta1) * grad_b
        v = beta2 * v + (1 - beta2) * (grad_b ** 2)
        m_hat = m / (1 - beta1 ** it)
        v_hat = v / (1 - beta2 ** it)

        # learning rate schedule perhaps start 0.3 and warmup
        lr = 0.4 / np.sqrt(it)
        b -= lr * m_hat / (np.sqrt(v_hat) + eps_adam)

        # increase alpha gradually
        alpha = min(alpha_max, alpha * 1.02)

    return best_a, best_score

"""
Propose a step‑function sequence that minimizes
    2·n·max(conv(a,a)) / (sum(a)**2)

Algorithm
---------

*   The variables are kept on the probability simplex via a soft‑max
    re‑parameterisation (logits b).  The loss is homogeneous so we can fix
    Σa_i = 1.

*   Instead of differentiating the exact max‑convolution we minimise a
    *smoothed* maximum:
        m_α(a) = (1/α)·log Σ_k exp(α·conv_k(a))

    For α → ∞ we recover the true max, but for moderate α the
    objective is smooth and yields useful gradients.  α is increased
    gradually (deterministic annealing).

*   The gradient of m_α is a convex combination of the sub‑gradients of
    the individual convolution entries.  Using the convolution theorem,
    the weighted sub‑gradient can be obtained with a single FFT
    convolution:
        ∇_a m_α = 2·(w_rev * a)[2·n‑2‑j]   (j = 0…n‑1)
    where w are the soft‑max weights derived from the current
    convolution and w_rev is w reversed.

*   Adam optimisation on the logits drives the iterates.  Several short
    restarts (different random / geometric / sparse initialisations)
    are performed within the given time budget; the best sequence seen
    is returned.

*   All helper functions are top‑level, there are no closures and only
    numpy, time and the already‑imported ``evaluate_sequence`` are used.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper functions (top‑level – no closures)
# ----------------------------------------------------------------------


def _full_convolution_fft(a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Full a*a convolution (length 2·n‑1) using FFT of length *fft_len*.
    """
    fft_a = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(fft_a * fft_a, n=fft_len)
    n = a.shape[0]
    return conv[: 2 * n - 1]


def _smooth_max_and_weighted_grad(
    a: np.ndarray,
    alpha: float,
    fft_len_conv: int,
    fft_len_grad: int,
) -> tuple[float, np.ndarray, np.ndarray]:
    """
    Returns (smoothed_max, gradient_of_smoothed_max, raw_conv).

    *   ``conv``  – the full (2·n‑1) convolution a*a.
    *   ``w``     – soft‑max weights derived from conv with temperature 1/α.
    *   ``grad``  – ∇_a  ( Σ_k w_k·conv_k ) = 2·(w_rev * a)[2·n‑2‑j].
    """
    n = a.shape[0]

    # ----- convolution a*a -------------------------------------------------
    fft_a = np.fft.rfft(a, n=fft_len_conv)
    conv = np.fft.irfft(fft_a * fft_a, n=fft_len_conv)[: 2 * n - 1]

    # ----- smoothed maximum (log‑sum‑exp) ---------------------------------
    # shift by the maximal entry for numerical stability
    conv_max = conv.max()
    scaled = alpha * (conv - conv_max)
    w_raw = np.exp(scaled)
    w_sum = w_raw.sum()
    w = w_raw / w_sum                     # Σ w = 1
    # smoothed max = conv_max + log(w_sum) / α
    if alpha == 0.0:
        smooth_max = conv_max
    else:
        smooth_max = conv_max + np.log(w_sum) / alpha

    # ----- weighted gradient -----------------------------------------------
    # w_rev is w reversed; convolution w_rev * a yields the needed vector
    w_rev = w[::-1]
    fft_w = np.fft.rfft(w_rev, n=fft_len_grad)
    fft_a2 = np.fft.rfft(a, n=fft_len_grad)
    weighted_full = np.fft.irfft(fft_a2 * fft_w, n=fft_len_grad)

    # slice: g[j] = (w_rev * a)[2·n‑2‑j]   for j = 0 … n‑1
    start = 2 * n - 2
    grad = 2.0 * weighted_full[start - np.arange(n)]

    return smooth_max, grad, conv


def _initial_sequence(
    kind: str,
    n: int,
    rng: np.random.Generator,
    prev_seq: np.ndarray | None,
) -> np.ndarray:
    """
    Produce a feasible starting probability vector of length *n*.

    ``kind`` can be:
        'prev'               – use the previous best (padded / truncated)
        'uniform'            – all entries equal
        'random_dirichlet'   – Dirichlet(0.5) for sparsity
        'geometric'          – symmetric geometric decay
        'sparse_random'      – a few equally‑weighted spikes
    """
    if kind == "prev" and prev_seq is not None:
        seq = prev_seq.copy()
        if seq.shape[0] != n:
            if seq.shape[0] > n:
                seq = seq[:n]
            else:
                seq = np.concatenate([seq, np.zeros(n - seq.shape[0])])
        seq = np.maximum(seq, 0.0)
        total = seq.sum()
        if total == 0.0:
            return np.full(n, 1.0 / n)
        return seq / total

    if kind == "uniform":
        return np.full(n, 1.0 / n)

    if kind == "random_dirichlet":
        alpha = 0.5
        sample = rng.gamma(alpha, 1.0, size=n)
        return sample / sample.sum()

    if kind == "geometric":
        r = 0.98 + 0.02 * rng.random()
        left = r ** np.arange(n)
        right = r ** np.arange(n - 1, -1, -1)
        seq = left + right
        return seq / seq.sum()

    if kind == "sparse_random":
        # number of spikes ≈ sqrt(n) – typical size of a Sidon‑type set
        k = max(2, int(np.sqrt(n)))
        idx = rng.choice(n, size=k, replace=False)
        seq = np.zeros(n)
        seq[idx] = 1.0 / k
        return seq

    # fallback
    return np.full(n, 1.0 / n)


def _anneal_optimize(
    init_a: np.ndarray,
    deadline: float,
    restart_id: int,
    target: float = 1.503,
) -> tuple[np.ndarray, float]:
    """
    Adam optimisation on the logits of the probability vector.
    The loss is the *smoothed* maximum of the convolution; α (temperature)
    is increased gradually during the run.
    Returns the best feasible vector (Σa=1) and its true objective.
    """
    n = init_a.shape[0]

    # FFT lengths – powers of two for speed
    fft_len_conv = 1 << ((2 * n - 2).bit_length())          # ≥ 2·n‑1
    fft_len_grad = 1 << ((3 * n - 2).bit_length())          # ≥ 3·n‑2

    eps_logit = 1e-12
    b = np.log(init_a + eps_logit)                          # logits

    # Adam moments
    m = np.zeros_like(b)
    v = np.zeros_like(b)
    beta1, beta2 = 0.9, 0.999
    eps_adam = 1e-8

    # annealing parameters
    alpha = 0.5               # starts with a very smooth max
    alpha_max = 800.0
    alpha_factor = 1.02       # multiplicative increase each iteration

    # keep the best objective seen so far (real max, not smoothed)
    conv0 = _full_convolution_fft(init_a, fft_len_conv)
    best_score = 2.0 * n * conv0.max()
    best_a = init_a.copy()
    stagnation = 0
    it = 0

    while time.time() < deadline:
        it += 1

        # ----- obtain probability vector from logits -----------------------
        b_max = np.max(b)
        a = np.exp(b - b_max)
        a /= a.sum()

        # ----- smoothed maximum + weighted gradient -----------------------
        smooth_max, grad_smooth, conv = _smooth_max_and_weighted_grad(
            a, alpha, fft_len_conv, fft_len_grad
        )
        cur_score = 2.0 * n * conv.max()          # true objective for monitoring

        # ----- keep best -------------------------------------------------
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            stagnation = 0
            print(
                f"Restart {restart_id:02d} iter {it:6d}: best = {best_score:.6f}"
            )
        else:
            stagnation += 1

        # ----- early‑stop & plateau handling -----------------------------
        if best_score < target:
            break
        if stagnation > 8000:
            # random jitter in log‑space to escape flat regions
            b += 0.02 * np.random.randn(n)
            stagnation = 0

        # ----- gradient of the *real* objective (factor 2·n) -------------
        grad_obj = 2.0 * n * grad_smooth          # ∂/∂a (2·n·smooth_max)

        # ----- transform to logits (soft‑max Jacobian) -------------------
        avg = np.dot(a, grad_obj)
        grad_b = a * (grad_obj - avg)

        # ----- Adam update -----------------------------------------------
        m = beta1 * m + (1.0 - beta1) * grad_b
        v = beta2 * v + (1.0 - beta2) * (grad_b ** 2)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        lr = 0.4 / np.sqrt(it)                   # diminishing step‑size
        b -= lr * m_hat / (np.sqrt(v_hat) + eps_adam)

        # ----- increase smoothness ---------------------------------------
        alpha = min(alpha_max, alpha * alpha_factor)

    return best_a, best_score

def propose_candidate(budget_s: int = 1000) -> list:

    deadline = time.time() + budget_s - 5.0     # safety margin

    # ------------------------------------------------------------------

    except Exception:
        previous = None
        n = 5146                                 # historic default

    # ------------------------------------------------------------------

    # ------------------------------------------------------------------
    best_overall_a = None
    best_overall_score = float("inf")
    restart_id = 0

    # ------------------------------------------------------------------

    # ------------------------------------------------------------------
    while time.time() < deadline - 10.0:
