def subgradient_descent(initial, time_deadline, max_iter=200_000, eta0=0.1):
    a = project_to_simplex(initial)
    best_a = a.copy()
    best_score = compute_score(a)
    n = len(a)
    for t in range(1, max_iter+1):
        if time.time() >= time_deadline:
            break
        conv, max_val, max_idx = conv_and_max(a)
        grad = np.zeros_like(a)
        for k in max_idx:
            d = k - (n-1)
            if d >= 0:
                grad[d:] += a[:n-d]
            else:
                grad[:n+d] += a[-d:]
        grad *= 2.0
        eta = eta0 / np.sqrt(t)
        a = a - eta * grad
        a = project_to_simplex(a)
        cur_score = compute_score(a)
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
        # optionally add small random perturb
        if t % 5000 == 0:
            print(f"    subgrad iter {t:,d}  curScore {cur_score:.6f}  best {best_score:.6f}")
    return best_a, best_score

"""
Propose a low‑score step‑function for ``evaluate_sequence``.

The algorithm consists of several complementary stages:

1. **Rich parametric sweep** – generates many candidates from a variety of
   analytic families (Beta, exponential, Gaussian, cosine‑window, arcsine,
   uniform and random Dirichlet draws).  The best candidate seen so far is kept.

2. **Projected sub‑gradient descent on the true L∞‑objective** – the
   gradient of the maximal convolution entry is a simple sum of shifted copies
   of the current vector, so we can move opposite to this direction, project
   back onto the probability simplex and use a diminishing stepsize
   (ηₜ = η₀/√t).  This stage runs for many thousands of iterations
   (limited only by the time budget) and dramatically flattens the
   convolution.

3. **Upsampling** – if time permits the current best shape is linearly
   interpolated to a larger length (8192).  A longer vector can realise a
   convolution that is closer to the uniform average, thus often reduces
   the objective further.

4. **Fine stochastic hill‑climb** – a cheap pair‑wise mass‑transfer
   hill‑climber (identical to the one used previously) finishes the search
   by exploiting any remaining small improvements.

All vectors are kept on the simplex (∑aᵢ = 1, aᵢ ≥ 0) so the scale‑invariance
of the evaluation function is respected.  The routine respects the given
time budget (default 1000 s) and returns the best sequence found.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Basic linear‑algebra utilities
# ----------------------------------------------------------------------


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    if np.all(v <= 0):
        # If everything is non‑positive, the projection is the uniform vector.
        n = v.shape[0]
        return np.full_like(v, 1.0 / n, dtype=float)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho = np.where(u - cssv / np.arange(1, v.shape[0] + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full convolution using a real FFT (fast)."""
    m = len(a) + len(b) - 1
    n_fft = next_pow2(m)
    A = np.fft.rfft(a, n=n_fft)
    B = np.fft.rfft(b, n=n_fft)
    C = A * B
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


def compute_score(a: np.ndarray) -> float:
    """Score identical to `evaluate_sequence` when ∑a = 1."""
    a = np.asarray(a, dtype=float)
    n = len(a)
    conv = convolution_fft(a, a)
    max_val = float(np.max(conv))
    s = float(np.sum(a))
    if s < 1e-12:
        return float("inf")
    return 2.0 * n * max_val / (s * s)


# ----------------------------------------------------------------------
# Analytic families of candidate step‑functions
# ----------------------------------------------------------------------


def uniform_sequence(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=float)


def arcsine_initial(n: int) -> np.ndarray:
    """Arcsine‑shaped start: a_i ∝ 1 / sqrt((i+0.5)(n‑0.5‑i))."""
    i = np.arange(n, dtype=float)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - 0.5 - i))
    vals = np.clip(vals, 0.0, 1e12)
    return vals / np.sum(vals)


def beta_sequence(n: int, alpha: float, beta: float | None = None) -> np.ndarray:
    """Beta‑type coefficients a_i ∝ (i+½)^α·(n‑½‑i)^β."""
    if beta is None:
        beta = alpha
    i = np.arange(n, dtype=float)
    left = np.power(i + 0.5, alpha)
    right = np.power(n - 0.5 - i, beta)
    seq = left * right
    seq = np.clip(seq, 0.0, 1e12)
    total = np.sum(seq)
    if total == 0.0:
        seq = np.full(n, 1.0 / n)
    else:
        seq = seq / total
    return seq


def exponential_sequence(n: int, lam: float) -> np.ndarray:
    """a_i ∝ exp(-lam·|i - (n‑1)/2|)."""
    i = np.arange(n, dtype=float)
    centre = (n - 1) / 2.0
    seq = np.exp(-lam * np.abs(i - centre))
    seq = np.clip(seq, 0.0, 1e12)
    return seq / np.sum(seq)


def gaussian_sequence(n: int, sigma_frac: float) -> np.ndarray:
    """a_i ∝ exp(- (i‑centre)² / (2·σ²)), σ = σ_frac·n."""
    i = np.arange(n, dtype=float)
    centre = (n - 1) / 2.0
    sigma = max(sigma_frac * n, 1e-9)
    seq = np.exp(-0.5 * ((i - centre) / sigma) ** 2)
    seq = np.clip(seq, 0.0, 1e12)
    return seq / np.sum(seq)


def cosine_window_sequence(n: int, power: float) -> np.ndarray:
    """a_i ∝ sin(π·(i+½)/n)ᵖ – a smooth bell shape."""
    i = np.arange(n, dtype=float)
    base = np.sin(np.pi * (i + 0.5) / n)
    seq = np.power(base, power)
    seq = np.clip(seq, 0.0, 1e12)
    return seq / np.sum(seq)


def random_dirichlet_sequence(n: int, concentration: float) -> np.ndarray:
    """Draw a random point on the simplex with Dirichlet(α) where α = concentration."""
    if concentration <= 0:
        # fallback to uniform
        return uniform_sequence(n)
    draws = np.random.gamma(concentration, 1.0, size=n)
    seq = draws / np.sum(draws)
    return seq


def upsample_sequence(seq: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation of a probability vector to a larger size."""
    old_n = len(seq)
    old_x = (np.arange(old_n) + 0.5) / old_n
    new_x = (np.arange(new_n) + 0.5) / new_n
    new_vals = np.interp(new_x, old_x, seq)
    new_vals = np.clip(new_vals, 0.0, 1e12)
    return new_vals / np.sum(new_vals)


# ----------------------------------------------------------------------
# Sub‑gradient descent on the true L∞‑objective
# ----------------------------------------------------------------------


def max_convolution_subgradient(a: np.ndarray, max_idx: np.ndarray) -> np.ndarray:
    """
    Sub‑gradient of   max_k (a∗a)_k   evaluated at the current maximising
    indices `max_idx`.  The gradient for a single index k is 2·shift(a, k‑(n‑1)).
    """
    n = a.shape[0]
    grad = np.zeros_like(a)
    for k in max_idx:
        d = k - (n - 1)                     # signed shift amount
        if d >= 0:
            # shift right by d → contributions from a[0:n-d] go to positions d:n
            grad[d:] += a[:n - d]
        else:
            # shift left by -d → contributions from a[-d:n] go to positions 0:n+d
            nd = -d
            grad[:n - nd] += a[nd:]
    grad *= 2.0
    return grad


def subgradient_descent(
    init_a: np.ndarray,
    time_deadline: float,
    max_iter: int = 200_000,
    eta0: float = 0.8,
) -> tuple[np.ndarray, float]:
    """
    Projected sub‑gradient descent directly on   t = max_k (a∗a)_k .
    The step size follows ηₜ = η₀/√t.
    Returns the best vector (and its score) encountered.
    """
    a = project_to_simplex(init_a.astype(float).copy())
    best_a = a.copy()
    best_score = compute_score(a)
    n = len(a)

    for t in range(1, max_iter + 1):
        if time.time() >= time_deadline:
            break

        conv = convolution_fft(a, a)
        max_val = float(conv.max())
        # gather all indices attaining (or nearly attaining) the maximum
        # a small tolerance helps when the max is not exactly unique due to FP error
        tol = 1e-14
        max_idx = np.where(conv >= max_val - tol)[0]

        grad = max_convolution_subgradient(a, max_idx)
        eta = eta0 / np.sqrt(t)
        a = a - eta * grad
        a = project_to_simplex(a)

        cur_score = compute_score(a)
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()

        # occasional progress reports
        if t % 5000 == 0:
            print(f"    subgrad iter {t:,d}  curScore {cur_score:.6f}  best {best_score:.6f}")

    return best_a, best_score


# ----------------------------------------------------------------------
# Simple stochastic hill‑climbing (final polishing)
# ----------------------------------------------------------------------


def hill_climb(
    seq,
    time_deadline: float,
    max_steps: int = 50_000,
    init_step: float = 0.05,
):
    """
    Pair‑wise mass‑transfer hill‑climbing.
    Moves that lower the true (unsmoothed) score are kept.
    """
    a = np.array(seq, dtype=float)
    a = project_to_simplex(a)
    best_a = a.copy()
    best_score = compute_score(best_a)
    n = len(a)
    step = init_step

    while max_steps > 0 and time.time() < time_deadline:
        i, j = np.random.randint(n, size=2)
        if i == j:
            continue
        delta = step * best_a[i]
        if delta <= 0.0:
            continue
        cand = best_a.copy()
        cand[i] -= delta
        cand[j] += delta
        cand = project_to_simplex(cand)
        cur_score = compute_score(cand)
        if cur_score < best_score:
            best_score = cur_score
            best_a = cand
            step = min(step * 1.07, 0.5)
        else:
            step *= 0.96
        max_steps -= 1
    return best_a.tolist(), best_score

            max_steps=60_000,
            init_step=0.06,
        )
        print(f"Hill‑climb score = {refined_val:.6f}")
        if refined_val < best_val:
            best_val = refined_val
            best_seq = refined_seq

    # ------------------------------------------------------------------
    # 5.  Safety fallback (should never be needed).
    # ------------------------------------------------------------------
    if best_seq is None:
        print("No candidate generated – falling back to a naïve uniform vector.")
        best_seq = uniform_sequence(256).tolist()
        best_val = evaluate_sequence(best_seq)

    final_score = evaluate_sequence(best_seq)
    print(f"\nSearch finished. Final reported score = {final_score:.6f}")
    return best_seq
