"""
Optimisation of a non‑negative step‑function for the objective

    2·n·max(conv(a,a)) / (sum(a)^2)

The objective is scale‑invariant, therefore we keep all candidates on the
probability simplex (∑a = 1).  The search proceeds in three phases:

1. **Parametric initialisation** – we generate candidates from a symmetric
   Beta(α,α) density (the uniform case α=1, the arcsine case α=½, etc.).
   A few Dirichlet random vectors are added for diversity.

2. **Sub‑gradient descent on the true ∞‑norm** – at each iteration the
   convolution is computed (FFT), the shift(s) attaining the maximal
   coefficient are identified and the exact sub‑gradient
   (the sum of the two overlapping copies) is taken.  After a step
   proportional to 1/√t we project back onto the simplex.  This directly
   minimises the max‑convolution rather than a smoothed surrogate.

3. **Pure combinatorial refinement** – a stochastic hill‑climber that swaps
   entries or transfers a small amount of mass between two positions, always
   accepting moves that improve the true (unsmoothed) objective.

The whole routine respects the 1000 s budget, prints progress, and finally
returns a plain Python list of floats ready for `evaluate_sequence`.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Basic linear‑algebra helpers
# ----------------------------------------------------------------------


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of a vector onto the probability simplex."""
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / v.shape[0])

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, v.shape[0] + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.where(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full linear convolution via a real FFT (fast)."""
    m = len(a) + len(b) - 1
    n_fft = next_pow2(m)
    A = np.fft.rfft(a, n=n_fft)
    B = np.fft.rfft(b, n=n_fft)
    C = A * B
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


def compute_score(a: np.ndarray) -> float:
    """
    Fast replica of `evaluate_sequence` when ∑a = 1.
    Returns 2·n·max(a*a).
    """
    n = a.shape[0]
    conv = convolution_fft(a, a)
    return 2.0 * n * float(np.max(conv))


# ----------------------------------------------------------------------
# Parametric initial vectors (symmetric Beta distributions)
# ----------------------------------------------------------------------


def beta_initial(n: int, alpha: float) -> np.ndarray:
    """
    Symmetric Beta(α,α) density discretised on n points,
    then projected onto the simplex.
    """
    x = (np.arange(n, dtype=float) + 0.5) / n  # midpoint rule
    vals = np.power(x, alpha - 1.0) * np.power(1.0 - x, alpha - 1.0)
    # avoid pure zeros when α<1
    eps = 1e-12
    vals = np.maximum(vals, eps)
    vals = project_to_simplex(vals)
    return vals


# ----------------------------------------------------------------------
# Sub‑gradient optimiser on the true max‑convolution
# ----------------------------------------------------------------------


def subgradient_optimize(
    init_a: np.ndarray,
    deadline: float,
    max_iter: int = 5000,
    eta0: float = 0.5,
) -> tuple[np.ndarray, float]:
    """
    Projected sub‑gradient descent minimizing the ℓ∞‑norm of the
    self‑convolution.  Returns the best vector found and its exact score.
    """
    a = project_to_simplex(init_a.astype(float).copy())
    n = len(a)

    best_a = a.copy()
    best_score = compute_score(a)

    for t in range(1, max_iter + 1):
        if time.time() >= deadline:
            break

        conv = convolution_fft(a, a)
        max_val = float(conv.max())
        cur_score = 2.0 * n * max_val

        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()

        # indices attaining (almost) the maximum
        idxs = np.where(np.abs(conv - max_val) < 1e-12)[0]

        # accumulate sub‑gradient for all maximal shifts
        grad = np.zeros_like(a)
        for k in idxs:
            start_i = max(0, k - (n - 1))
            end_i = min(k, n - 1)
            i_range = np.arange(start_i, end_i + 1)
            grad[i_range] += 2.0 * a[k - i_range]

        # diminishing step size
        eta = eta0 / np.sqrt(t)
        a = a - eta * grad
        a = project_to_simplex(a)

        # occasional tiny perturbation to escape plateaus
        if t % 500 == 0:
            a += np.random.randn(n) * 1e-5
            a = project_to_simplex(a)

    return best_a, best_score


# ----------------------------------------------------------------------
# Simple stochastic hill‑climber (pure permutation / mass transfer)
# ----------------------------------------------------------------------


def hill_climb_permutation(
    seq,
    deadline,
    max_steps: int = 150_000,
):
    """
    Stochastic hill‑climb that only swaps two entries or moves a small
    fraction of mass.  The true (unsmoothed) objective is used for acceptance.
    """
    a = np.array(seq, dtype=float)
    n = a.shape[0]

    best_a = a.copy()
    best_score = compute_score(best_a)

    step_scale = 0.1  # fraction of current mass transferred in a move

    steps = 0
    while steps < max_steps and time.time() < deadline:
        move_type = np.random.rand()
        cand = best_a.copy()

        if move_type < 0.5:
            # simple swap
            i, j = np.random.randint(n, size=2)
            cand[i], cand[j] = cand[j], cand[i]
        else:
            # move a fraction from i to j
            i, j = np.random.randint(n, size=2)
            if i == j:
                continue
            delta = step_scale * cand[i]
            if delta <= 0:
                continue
            cand[i] -= delta
            cand[j] += delta
            # numerical safety
            if cand[i] < 0:
                cand[i] = 0.0

        cand = project_to_simplex(cand)
        cur_score = compute_score(cand)

        if cur_score < best_score:
            best_score = cur_score
            best_a = cand
            step_scale = min(step_scale * 1.05, 0.5)  # encourage larger steps
        else:
            step_scale *= 0.97  # shrink on failure

        steps += 1

    return best_a.tolist(), best_score

        # add a few Dirichlet random starts for extra diversity
        for _ in range(3):
            vec = np.random.dirichlet(np.full(n, 1.0))
            initial_vectors.append(vec)

        # also sweep the Beta shapes defined above
        for a in alphas:
            initial_vectors.append(beta_initial(n, a))

        # ------------------------------------------------------------------
        # Run a modest sub‑gradient optimisation for each start
        # ------------------------------------------------------------------
        for idx, init_vec in enumerate(initial_vectors, start=1):
            if time.time() >= deadline - 5.0:
                break
            print(f"  start {idx}/{len(initial_vectors)} – α‑shape {idx}/{len(initial_vectors)}")
            # allocate a number of iterations proportional to remaining time
            remaining = deadline - time.time()
            iters = int(min(6000, remaining * 20))  # rough heuristic
            seq, val = subgradient_optimize(
                init_a=init_vec,
                deadline=deadline,
                max_iter=iters,
                eta0=0.5,
            )
            print(f"    sub‑grad best score = {val:.6f}")
            if val < best_val:
                best_val = val
                best_seq = seq.tolist()
                print(f"    *** Global best updated to {best_val:.6f} (n={n}) ***")

    # ------------------------------------------------------------------
    # Final hill‑climbing refinement
    # ------------------------------------------------------------------
    if best_seq is not None and time.time() < deadline - 1.0:
        print("\nRefining best sequence with permutation hill‑climb")
        refined_seq, refined_val = hill_climb_permutation(
            best_seq,
            deadline=deadline,
            max_steps=150_000,
        )
        print(f"    refined score = {refined_val:.6f}")
        if refined_val < best_val:
            best_val = refined_val
            best_seq = refined_seq

    # ------------------------------------------------------------------
    # Safeguard: if nothing was produced, fall back to a trivial uniform vector.
    # ------------------------------------------------------------------
    if best_seq is None:
        best_seq = [1.0 / 256] * 256
        best_val = evaluate_sequence(best_seq)

    final_score = evaluate_sequence(best_seq)
    print(f"\nFinished search. Final reported score = {final_score:.6f}")

    # The harness expects a plain Python list of floats.
    return best_seq
