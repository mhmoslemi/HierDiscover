    import time, numpy as np
    deadline = time.time() + budget_s - 0.1
    best_seq = None
    best_score = np.inf
    restarts = 0
    # possible lengths list:
    lengths = [int(n0*1.0), int(n0*1.5), int(n0*2), 10000, 15000, 20000]
    # We'll iterate over lengths while time remains.
    for n in lengths:
        if time.time() >= deadline: break
        # initialize u: either from existing best (if shape matches length) or random.
        # If we have best_seq and its length equals n, we can use its log values?
        # We'll convert best_seq to u: u = log(best_seq) (with small epsilon).
        if best_seq is not None and len(best_seq) == n:

        m = np.zeros_like(u)

            # gradually increase alpha
            if t % 2000 == 0:
                alpha = min(alpha * 1.5, 2000.0)

            m_conv = conv_full.max()

"""Adam‑softmax optimiser for the max‑convolution objective.

The routine parametrises the step heights by a softmax vector `a = softmax(u)` so
that the non‑negativity and unit‑sum constraints are satisfied automatically.
A smoothed version of `max(conv)` (log‑sum‑exp) provides a differentiable
approximation; its gradient is obtained via FFT‑based cross‑correlation.
Adam is used for adaptive steps.  Several restarts and a slowly increasing
temperature `alpha` are employed; the best sequence found (as measured by
`evaluate_sequence`) is returned.
"""

import time
import numpy as np

def _next_pow2(x: int) -> int:
    """Return the smallest power of two >= x."""
    return 1 << (x - 1).bit_length()

def _softmax(u: np.ndarray) -> np.ndarray:
    """Numerically stable softmax."""
    u_shift = u - np.max(u)
    e = np.exp(u_shift)
    s = e.sum()
    if s == 0.0:
        return np.full_like(u, 1.0 / u.shape[0])
    return e / s

def _conv_self_fft(a: np.ndarray, L: int) -> np.ndarray:
    """Return the linear convolution a*a of length 2*len(a)-1 using FFT size L."""
    fft_a = np.fft.rfft(a, L)
    conv = np.fft.irfft(fft_a * fft_a, L)
    n = a.shape[0]
    return conv[: 2 * n - 1]

def _gradient_via_corr(a: np.ndarray, w: np.ndarray, L: int) -> np.ndarray:
    """Return the gradient of Σ w_k (a*a)_k w.r.t a, i.e. 2·(a ⋆ w)."""
    # a : length n
    # w : length 2n-1
    fft_a = np.fft.rfft(a, L)
    fft_w = np.fft.rfft(w, L)
    # correlation a ⋆ w  (shift w forward)
    corr = np.fft.irfft(fft_a * np.conj(fft_w), L)
    n = a.shape[0]
    # first n entries correspond to shifts 0..n-1
    grad = 2.0 * corr[:n]
    return grad

def propose_candidate(budget_s: int = 1000) -> list[float]:
    """

    ``2·n·max(a*a) / (sum(a)**2)`` within the given time budget.
    Returns a list of floats (the heights).
    """
    deadline = time.time() + budget_s - 0.2   # small safety margin
    best_seq = None
    best_score = float("inf")
    target = 1.503

    # Determine a base length.  Use the historic solution if present,
    # otherwise fall back to a moderately large value.
    try:
        init_seq = np.asarray(height_sequence_1, dtype=float)  # type: ignore[name-defined]
        base_n = init_seq.shape[0]
        init_seq = np.maximum(init_seq, 0.0)
        if init_seq.sum() > 0:
            init_seq = init_seq / init_seq.sum()
        else:
            init_seq = None
    except Exception:
        init_seq = None
        base_n = 0

    # A small collection of lengths to try – larger sizes give a smaller
    # theoretical lower bound (≈1) but are slower.
    lengths = []
    if base_n > 0:
        lengths.append(base_n)
    # Add a few larger sizes (powers of two are convenient for the FFT)
    for mul in (1, 2, 4):
        n = max(4000, base_n * mul) if base_n > 0 else 4000 * mul
        # keep n reasonable (≤ 25000) to stay within the time budget
        n = min(n, 25000)
        if n not in lengths:
            lengths.append(int(n))

    restart = 0
    # Main restart loop
    while time.time() < deadline and restart < 20:
        for n in lengths:
            if time.time() >= deadline:
                break
            # initialise the optimisation variable u (log‑probabilities)
            if init_seq is not None and init_seq.shape[0] == n and restart == 0:
                # start from the historic solution
                eps = 1e-12
                u = np.log(np.maximum(init_seq, eps))
            else:

                    seed=int(time.time() * 1e6) % (2**32 - 1)

                # Gaussian initialisation gives a reasonable spread
                u = rng.normal(loc=0.0, scale=1.0, size=n)

            # Adam parameters
            m = np.zeros_like(u)
            v = np.zeros_like(u)
            beta1 = 0.9
            beta2 = 0.999
            eps_adam = 1e-8
            lr0 = 0.05
            t = 0

            # and the cross‑correlation (n + 2n‑2 = 3n‑2).
            L = _next_pow2(3 * n)

                a = _softmax(u)

                # occasional jitter to escape shallow plateaus

                    jitter = 0.001 * np.random.randn(n)
                    u += jitter

        restart += 1

    # If we never found a feasible point, fall back to a uniform distribution.
    if best_seq is None:
        # default length 4000
        n = 4000
        best_seq = np.full(n, 1.0 / n)

    # Return as plain Python list.
    return [float(v) for v in best_seq.tolist()]

    """Return the smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _softmax(u: np.ndarray) -> np.ndarray:
    """Numerically stable softmax."""
    u_shift = u - np.max(u)
    e = np.exp(u_shift)
    s = e.sum()
    if s == 0.0:
        return np.full_like(u, 1.0 / u.shape[0])
    return e / s


def _conv_self_fft(a: np.ndarray, L: int) -> np.ndarray:
    """Linear self‑convolution a*a (length 2·len(a)-1) using FFT size L."""
    fft_a = np.fft.rfft(a, L)
    conv = np.fft.irfft(fft_a * fft_a, L)
    n = a.shape[0]
    return conv[: 2 * n - 1]


def _gradient_via_corr(a: np.ndarray, w: np.ndarray, L: int) -> np.ndarray:
    """
    Gradient of Σ w_k·(a*a)_k w.r.t. a.
    For a given weight vector `w` (length 2·n‑1) the gradient is
    2·(a ⋆ w) where `⋆` denotes the (zero‑padded) cross‑correlation.
    """
    fft_a = np.fft.rfft(a, L)
    fft_w = np.fft.rfft(w, L)
    # cross‑correlation a ⋆ w (shifts w forward)
    corr = np.fft.irfft(fft_a * np.conj(fft_w), L)
    n = a.shape[0]
    # first n entries correspond to shifts 0,…,n‑1
    return 2.0 * corr[:n]


def propose_candidate(budget_s: int = 1000) -> list[float]:
    """
    Search for a non‑negative step function that minimises

        2·n·max(a*a) / (sum(a)**2)

    within the given time budget (seconds).  Returns the best
    height sequence found as a plain Python list.
    """
    deadline = time.time() + budget_s - 0.2      # small safety margin
    best_seq = None
    best_score = float("inf")
    target = 1.503                                 # early‑exit target

    # ------------------------------------------------------------
    # 1. Determine a starting length and (if available) a historic seq.
    # ------------------------------------------------------------
    try:
        init_seq = np.asarray(height_sequence_1, dtype=float)   # type: ignore[name-defined]
        base_n = init_seq.shape[0]
        init_seq = np.maximum(init_seq, 0.0)
        if init_seq.sum() > 0:
            init_seq = init_seq / init_seq.sum()
        else:
            init_seq = None
    except Exception:
        init_seq = None
        base_n = 0

    # Various lengths – larger `n` lowers the theoretical bound (≈1) but costs more.
    lengths = []
    if base_n > 0:
        lengths.append(base_n)
    for mul in (1, 2, 4):
        n = max(4000, base_n * mul) if base_n > 0 else 4000 * mul
        n = min(int(n), 25000)           # keep FFT sizes reasonable
        if n not in lengths:
            lengths.append(int(n))

    restart = 0
    # ------------------------------------------------------------
    # 2. Main optimisation loop (multiple restarts, multiple lengths)
    # ------------------------------------------------------------
    while time.time() < deadline and restart < 20:
        for n in lengths:
            if time.time() >= deadline:
                break

            # ----- initialise Adam state -----
            if init_seq is not None and init_seq.shape[0] == n and restart == 0:
                eps = 1e-12
                u = np.log(np.maximum(init_seq, eps))
            else:
                rng = np.random.default_rng(
                    seed=int(time.time() * 1e6) % (2**32 - 1)
                )
                u = rng.normal(loc=0.0, scale=1.0, size=n)

            m = np.zeros_like(u)
            v = np.zeros_like(u)
            beta1, beta2 = 0.9, 0.999
            eps_adam = 1e-8
            lr0 = 0.05
            t = 0

            # FFT size sufficient for both self‑convolution (2n‑1)
            # and cross‑correlation (n + 2n‑2 = 3n‑2)
            L = _next_pow2(3 * n)

            # smoothing parameter for the soft‑max approximation
            alpha = 4.0
            alpha_max = 2000.0

            # ----------------------------------------------------
            # 3. Inner optimisation loop for this (restart, n)
            # ----------------------------------------------------
            while time.time() < deadline:
                t += 1

                # ---- a = softmax(u) (satisfies constraints) ----
                a = _softmax(u)

                # ---- self‑convolution a*a ----
                conv = _conv_self_fft(a, L)            # length 2n‑1
                max_conv = conv.max()
                cur_score = 2.0 * n * max_conv

                # ---- keep global best ----
                if cur_score < best_score:
                    best_score = cur_score
                    best_seq = a.copy()
                    print(
                        f"[{time.time():.1f}] restart {restart}, n {n}, iter {t}: "
                        f"new best = {best_score:.6f}"
                    )
                    if best_score < target:
                        return best_seq.tolist()

                # ---- gradually sharpen the smooth max ----
                if t % 2000 == 0:
                    alpha = min(alpha * 1.5, alpha_max)

                # ---- soft‑max weights w (sum‑to‑one) ----
                m_conv = max_conv
                w = np.exp(alpha * (conv - m_conv))
                w_sum = w.sum()
                dw = w / w_sum if w_sum != 0.0 else np.full_like(w, 1.0 / w.size)

                # ---- gradient wrt a (via FFT‑based correlation) ----
                grad_a = _gradient_via_corr(a, dw, L)   # shape (n,)

                # ---- back‑propagation through the softmax ----
                dot = np.dot(grad_a, a)
                grad_u = a * (grad_a - dot)

                # ---- Adam update ----
                m = beta1 * m + (1 - beta1) * grad_u
                v = beta2 * v + (1 - beta2) * (grad_u ** 2)
                m_hat = m / (1 - beta1 ** t)
                v_hat = v / (1 - beta2 ** t)
                step = lr0 / (np.sqrt(v_hat) + eps_adam)
                u -= step * m_hat

                # ---- occasional jitter to escape plateaus ----
                if t % 5000 == 0:
                    jitter = 0.001 * np.random.randn(n)
                    u += jitter
        restart += 1

    # ------------------------------------------------------------
    # 4. Fallback – should never happen but keeps the signature safe
    # ------------------------------------------------------------
    if best_seq is None:
        n = 4000
        best_seq = np.full(n, 1.0 / n)

    return [float(v) for v in best_seq.tolist()]
