from scipy.optimize import linprog

def solve_lp_max_sum(a, M):
   n = len(a)
   A = build_convolution_matrix(a)
   b = M * np.ones(2*n-1)
   c = -np.ones(n)  # maximize sum(b)
   bounds = [(0, None)]*n
   res = linprog(c, A_ub=A, b_ub=b, bounds=bounds, method='highs')
   if not res.success:
       print("LP not solved:", res.message)
       return None
   return res.x

"""
Propose a low‑discrepancy step function.

The search combines three ideas that complement the previous surrogate‑Adam
approach:

1. **Explicit LP improvement** –  For a fixed length `n` we repeatedly solve a
   linear programme that maximises the total mass of a new vector `b`
   under the constraint that the convolution `a * b` does not exceed the current
   maximum self‑convolution value `M = max(a*a)`.  After solving we scale `b`
   to the original ℓ¹‑norm, mix it with the current vector `a` and repeat.
   This is the classic “alternating linear‑program” method used in the
   literature and directly reduces the true objective `max(a*a)`.

2. **Smooth Adam polishing** –  Starting from the best LP‑refined vector we run
   the Adam optimiser on a smooth (soft‑max) surrogate of the objective
   exactly as in the previous state‑of‑the‑art code.  The Adam stage finishes
   the optimisation and yields a further reduction of the maximum
   convolution.

3. **Multi‑scale exploration** –  The whole pipeline is executed for several
   lengths (`256 … 2048`).  For each length the best candidate seen by the
   official `evaluate_sequence` is kept.

All helper functions are top‑level, there are no closures or lambdas, and the
implementation respects the 1000 s time budget.
"""

import time
import math
from typing import List, Tuple

import numpy as np
from scipy.optimize import linprog


# ----------------------------------------------------------------------
# Low‑level utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Linear convolution via real FFT."""
    m = len(a) + len(b) - 1
    n_fft = next_pow2(m)
    A = np.fft.rfft(a, n=n_fft)
    B = np.fft.rfft(b, n=n_fft)
    C = A * B
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


def max_convolution_and_index(a: np.ndarray) -> Tuple[float, int]:
    """Return (max value, index) of the self‑convolution a*a."""
    conv = convolution_fft(a, a)
    idx = int(np.argmax(conv))
    return float(conv[idx]), idx


def project_to_simplex(v: np.ndarray, target_sum: float = 1.0) -> np.ndarray:
    """
    Euclidean projection of a vector onto the probability simplex
    {x ≥ 0, Σx = target_sum}.
    Duchi et al., 2008.
    """
    if target_sum <= 0:
        raise ValueError("target_sum must be positive")
    v_shift = v - np.mean(v) + target_sum / v.size
    # The algorithm works for an arbitrary target_sum by scaling
    u = np.sort(v_shift)[::-1]
    cssv = np.cumsum(u) - target_sum
    rho = np.where(u - cssv / np.arange(1, v.size + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v_shift - theta, 0.0)
    return w


def arcsine_initial(n: int) -> np.ndarray:
    """
    Classical arcsine shape:
        a_i ∝ 1 / √((i+½)*(n‑½‑i)).
    Returns a vector with unit ℓ¹‑norm.
    """
    i = np.arange(n, dtype=float)
    eps = 0.25
    vals = 1.0 / np.sqrt((i + 0.5) * (n - 0.5 - i) + eps)
    vals /= vals.sum()
    return vals


def compute_score(a: np.ndarray) -> float:
    """Fast surrogate: 2·n·max(a*a) (sum(a) = 1)."""
    n = len(a)
    conv = convolution_fft(a, a)
    max_val = float(np.max(conv))
    return 2.0 * n * max_val


# ----------------------------------------------------------------------
# Adam optimiser on the smooth surrogate (identical to the previous code)
# ----------------------------------------------------------------------


def smooth_gradient(a: np.ndarray, lam: float) -> Tuple[np.ndarray, float]:
    """
    Gradient of the smooth max‑convolution objective for temperature λ.
    Returns (grad, smooth_value).
    """
    n = len(a)
    conv = convolution_fft(a, a)          # length 2n‑1
    max_c = np.max(conv)

    # soft‑max weights
    exp_term = np.exp(lam * (conv - max_c))
    sum_exp = np.sum(exp_term)
    w = exp_term / sum_exp
    B = max_c + np.log(sum_exp) / lam      # smooth approximation of max

    # weighted sub‑gradient (2 * a * reversed(weight))
    w_rev = w[::-1]
    conv_rev = convolution_fft(a, w_rev)  # length 3n‑2
    L = 2 * n - 1
    weighted_grad = 2.0 * conv_rev[L - 1 - np.arange(n)]

    s = np.sum(a)
    term1 = (2.0 * n / (s ** 2)) * weighted_grad
    term2 = -(4.0 * n * B / (s ** 3)) * np.ones_like(a)
    grad = term1 + term2
    return grad, B


def adam_optimize(initial_a: np.ndarray,
                  time_deadline: float,
                  max_iter: int = 100_000,
                  lr0: float = 0.2,
                  lam_start: float = 5.0,
                  lam_end: float = 30.0,
                  noise_every: int = 2000,
                  noise_scale: float = 5e-5) -> Tuple[np.ndarray, float]:
    """
    Adam optimiser on the smooth surrogate.
    Returns the best vector (still on the simplex) and its fast surrogate score.
    """
    a = project_to_simplex(initial_a.astype(float).copy())
    best_a = a.copy()
    best_sc = compute_score(a)

    beta1, beta2 = 0.9, 0.999
    eps = 1e-8
    m = np.zeros_like(a)
    v = np.zeros_like(a)

    for it in range(1, max_iter + 1):
        if time.time() >= time_deadline:
            break

        lam = lam_start + (lam_end - lam_start) * min(1.0, it / max_iter)
        grad, _ = smooth_gradient(a, lam)

        # Adam moment updates
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)

        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        step = lr0 * m_hat / (np.sqrt(v_hat) + eps)
        a_new = a - step

        if noise_every and (it % noise_every == 0):
            a_new += np.random.randn(len(a)) * noise_scale

        a_new = project_to_simplex(a_new)

        cur_sc = compute_score(a_new)
        if cur_sc < best_sc:
            best_sc = cur_sc
            best_a = a_new.copy()

        a = a_new

    return best_a, best_sc


# ----------------------------------------------------------------------
# Linear‑program improvement step
# ----------------------------------------------------------------------


def build_convolution_matrix(a: np.ndarray) -> np.ndarray:
    """
    Build the (2n‑1) × n matrix A such that A @ b = a * b (linear convolution).
    Row i corresponds to index i of the convolution.
    """
    n = len(a)
    rows = 2 * n - 1
    A = np.empty((rows, n), dtype=np.float64)

    for i in range(rows):
        lo = max(0, i - n + 1)
        hi = min(i, n - 1)
        ks = np.arange(lo, hi + 1, dtype=int)
        A[i, ks] = a[i - ks]
        # elements outside the valid range stay zero (already allocated)
    return A


def lp_step_max_sum(a: np.ndarray, M: float) -> np.ndarray | None:
    """
    Solve   max  Σ b_j   s.t.   a * b ≤ M   and   b ≥ 0.
    Returns the optimal b (may be None if the LP fails).
    """
    n = len(a)
    A = build_convolution_matrix(a)                 # shape (2n‑1, n)
    ub = M * np.ones(2 * n - 1, dtype=np.float64)
    c = -np.ones(n, dtype=np.float64)               # maximise Σb → minimise -Σb
    bounds = [(0, None)] * n

    try:
        res = linprog(c, A_ub=A, b_ub=ub, bounds=bounds,
                      method='highs',
                      options={'presolve': True})
    except Exception as e:
        print(f"    LP failed with exception: {e}")
        return None

    if not res.success:
        print(f"    LP not solved: {res.message}")
        return None
    return res.x

def propose_candidate(seed: int = 42,
                     budget_s: int = 1000,
                     **kwargs) -> List[float]:
    """
    Search for a non‑negative step sequence with a low ``evaluate_sequence``
    value.  The algorithm runs a short LP‑based improvement loop followed by a
    smooth‑Adam polish for several candidate lengths.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 2.0   # small safety margin

    best_seq: List[float] = []
    best_val: float = float("inf")

    # ------------------------------------------------------------------
    # Optional fallback from a previously discovered sequence
    # ------------------------------------------------------------------
    try:
        fallback = [float(x) for x in height_sequence_1]   # type: ignore
        val = evaluate_sequence(fallback)
        print(f"Fallback (height_sequence_1) score = {val:.6f}")
        if val < best_val:
            best_val = val
            best_seq = fallback
    except Exception:
        pass

    # ------------------------------------------------------------------
    # Lengths to explore – larger lengths give better asymptotics but cost
    # more time; we stop when the remaining budget is too small.
    # ------------------------------------------------------------------
    lengths = [256, 512, 1024, 2048]

    for n in lengths:
        if time.time() >= deadline - 5.0:
            break

        print(f"\n=== Exploring length n = {n} ===")
        target_sum = math.sqrt(2.0 * n)

        # ---- initialise with the classic arcsine shape, normalised to target_sum
        a = arcsine_initial(n) * (target_sum / arcsine_initial(n).sum())
        # sanity check
        a = np.maximum(a, 0.0)

        # ---- evaluate the initial candidate
        val = evaluate_sequence(a.tolist())
        print(f"  Initial candidate score = {val:.6f}")
        if val < best_val:
            best_val = val
            best_seq = a.tolist()

        # ---- main LP improvement loop
        M, _ = max_convolution_and_index(a)
        lp_improved = False
        for it in range(1, 21):          # at most 20 LP rounds per length
            if time.time() >= deadline - 5.0:
                break

            b = lp_step_max_sum(a, M)
            if b is None:
                break

            sum_b = b.sum()
            if sum_b <= 0:
                break

            # scale b to preserve the ℓ¹‑norm (= target_sum)
            b_scaled = b * (target_sum / sum_b)

            # convex combination (t = 0.5 works well in practice)
            t = 0.5
            a_new = (1.0 - t) * a + t * b_scaled

            # re‑normalise exactly to target_sum (tiny round‑off may appear)
            a_new = a_new * (target_sum / a_new.sum())
            a_new = np.maximum(a_new, 0.0)

            M_new, _ = max_convolution_and_index(a_new)
            if M_new < M - 1e-12:               # strict improvement
                a = a_new
                M = M_new
                lp_improved = True
                # evaluate officially
                val = evaluate_sequence(a.tolist())
                print(f"    LP iter {it:2d}: maxconv={M:.6f}  score={val:.6f}")
                if val < best_val:
                    best_val = val
                    best_seq = a.tolist()
                    print(f"    *** New global best after LP: {best_val:.6f} ***")
            else:
                # no improvement – stop early
                print(f"    LP iter {it:2d}: no further improvement (M stays {M:.6f})")
                break

        # ------------------------------------------------------------------
        # If we made any progress, run a final Adam polishing on the
        # normalised vector (sum = 1).  Adam works on the simplex, so we
        # divide by target_sum, polish, and then restore the scale.
        # ------------------------------------------------------------------
        if lp_improved and time.time() < deadline - 5.0:
            a_norm = a / target_sum               # now Σ a_norm = 1
            remaining = deadline - time.time()
            max_iter = int(min(250_000, remaining * 30))
            print(f"  Polishing with Adam (max_iter={max_iter}) …")
            a_opt, _ = adam_optimize(
                initial_a=a_norm,
                time_deadline=deadline,
                max_iter=max_iter,
                lr0=0.15,
                lam_start=5.0,
                lam_end=30.0,
                noise_every=1200,
                noise_scale=3e-5,
            )
            a_opt_scaled = a_opt * target_sum
            val = evaluate_sequence(a_opt_scaled.tolist())
            print(f"  Polished score = {val:.6f}")
            if val < best_val:
                best_val = val
                best_seq = a_opt_scaled.tolist()
                print("*** Updated global best after Adam polish ***")

    # ------------------------------------------------------------------
    # Guard – guarantee a non‑empty return (fallback to a simple uniform)
    # ------------------------------------------------------------------
    if not best_seq:
        print("No valid candidate found – falling back to uniform 256‑vector.")
        best_seq = [1.0] * 256
        best_val = evaluate_sequence(best_seq)

    print(f"\nFinished search. Best reported score = {best_val:.6f}")
    return best_seq
