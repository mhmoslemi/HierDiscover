def propose_candidate(seed=123, budget_s=1000):
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 2

    best_seq = None
    best_val = float('inf')

    # fallback if height_sequence_1 exists
    ...
    
    lengths = [1024, 2048, 4096, 8192] (but limit to ensure time)
    for n in lengths:
        if time.time() > deadline - 20: break
        
        # generate candidates
        candidates = []
        # uniform
        candidates.append(np.full(n, 1/n))
        # arcsine
        candidates.append(arcsine_initial(n))
        # semi-circle shape
        i = np.arange(n, dtype=float)
        semicircle = np.sqrt(i * (n-1-i))
        candidates.append(semicircle / semicircle.sum())
        # triangular shape
        tri = np.minimum(i, n-1-i)
        candidates.append(tri / tri.sum())
        # cosine shape
        cos = 0.5 * (1.0 + np.cos(np.pi * (2.0*i/(n-1) - 1.0)))  # from 0 to pi
        candidates.append(cos / cos.sum())
        # gaussian
        for sigma in [0.15, 0.2, 0.25]:
            x = (i - (n-1)/2) / ((n-1)/2)
            gauss = np.exp(-0.5 * (x / sigma)**2)
            candidates.append(gauss / gauss.sum())
        # exponential decay from ends
        for decay in [0.01,0.02,0.03]:
            expo = np.exp(-decay * np.abs(i - (n-1)/2))
            candidates.append(expo / expo.sum())
        # power law
        for p in rng.uniform(-0.9, 0.9, size=5):
            eps = 0.01
            base = ((i + eps) * (n-1-i + eps)) ** (-p)
            candidates.append(base / base.sum())
        # beta shape samples
        for _ in range(5):
            alpha = rng.uniform(0.5, 5.0)
            beta = rng.uniform(0.5, 5.0)
            base = (i + 0.5) ** (alpha-1) * (n-1-i + 0.5) ** (beta-1)
            if base.sum() > 0:
                candidates.append(base / base.sum())
        # Dirichlet random
        for _ in range(2):
            samp = rng.exponential(1.0, size=n)
            samp /= samp.sum()
            candidates.append(samp)

        # evaluate fast scores
        scored = [(compute_score(c), c) for c in candidates]
        scored.sort(key=lambda x: x[0])
        # Keep top 5
        top = [c for _,c in scored[:5]]

        # For each top candidate run optimizer
        for idx, cand in enumerate(top):
            if time.time() > deadline - 5: break
            print(f"Length {n} candidate {idx+1}/{len(top)} initial score {scored[idx][0]:.6f}")

            # Determine iterations based on remaining time
            remaining = deadline - time.time()
            # allocate small proportion e.g., 0.5 sec per iteration? Let's schedule fixed max iter e.g., 20000.
            max_iter = int(min(20000, remaining * 30)) # 30 iter per sec approx.

"""
A fast multi‑family parametric search + high‑temperature exponentiated‑gradient
refinement for the step‑function minimisation problem.

The algorithm works as follows:

1. **Candidate generation** –  For each chosen length (powers of two up to
   8192) a large library of simple analytic shapes is built:
   uniform, arcsine, semicircle, triangular, cosine, Gaussian,
   exponential decay, symmetric power‑law (including negative exponents)
   and Beta‑like families (α,β∈[0.5,5]).  All candidates are normalised to
   unit L1‑norm.

2. **Cheap screening** –  Using a fast FFT‑based score (the exact objective
   without clamping) we keep the few (≤5) most promising candidates for a
   given length.

3. **Refinement** –  Each retained candidate is fed to a soft‑max
   exponentiated‑gradient optimiser.  The temperature λ is annealed from
   5 up to a large value (200 – 300) so that the smooth approximation
   converges to the true max‑convolution.  The optimiser runs for as many
   iterations as the remaining time permits (typically tens of thousands
   of steps) and keeps the best vector seen.

4. **Global polishing** –  The overall best sequence is refined once more
   with a longer schedule and an even higher final λ.

All updates stay on the probability simplex (non‑negative + sum = 1) via
multiplicative updates followed by a projection.  The routine always
returns a feasible sequence before the 1000 s budget expires.
"""

import time
import numpy as np
from typing import List, Tuple

# ----------------------------------------------------------------------
# Low‑level numeric utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two >= x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Linear convolution via real FFT (full length, zero‑padded to power‑of‑2)."""
    m = len(a) + len(b) - 1
    n_fft = next_pow2(m)
    A = np.fft.rfft(a, n=n_fft)
    B = np.fft.rfft(b, n=n_fft)
    C = A * B
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


def compute_score(a: np.ndarray) -> float:
    """
    Official objective (scale‑invariant):
        2·n·max(a*a) / (sum(a)²)
    a is assumed to be non‑negative; the function works for any vector.
    """
    n = len(a)
    conv = convolution_fft(a, a)
    max_val = float(np.max(conv))
    s = float(np.sum(a))
    if s < 1e-12:
        return float("inf")
    return 2.0 * n * max_val / (s * s)


def normalize(v: np.ndarray) -> np.ndarray:
    """Project a non‑negative vector onto the probability simplex."""
    total = v.sum()
    if total <= 0.0:
        return np.full_like(v, 1.0 / len(v))
    return v / total


# ----------------------------------------------------------------------
# Shape families (all return vectors that sum to 1)
# ----------------------------------------------------------------------


def uniform_shape(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n)


def arcsine_initial(n: int) -> np.ndarray:
    """Discrete arcsine density (regularised to avoid infinities)."""
    i = np.arange(n, dtype=float)
    eps = 0.25
    vals = 1.0 / np.sqrt((i + 0.5) * (n - 0.5 - i) + eps)
    return vals / vals.sum()


def semicircle_shape(n: int) -> np.ndarray:
    """f(x) ∝ sqrt(x(1‑x)) on a discrete grid."""
    i = np.arange(n, dtype=float)
    vals = np.sqrt(i * (n - 1 - i))
    # avoid the trivial zero vector when n is tiny
    if vals.sum() == 0.0:
        return uniform_shape(n)
    return vals / vals.sum()


def triangular_shape(n: int) -> np.ndarray:
    """A symmetric triangle that is zero at the ends."""
    i = np.arange(n, dtype=float)
    vals = np.minimum(i, n - 1 - i)
    return vals / vals.sum()


def cosine_shape(n: int) -> np.ndarray:
    """Cosine bump: ½(1+cos(π(2i/(n‑1)−1))) , zero at the ends."""
    i = np.arange(n, dtype=float)
    vals = 0.5 * (1.0 + np.cos(np.pi * (2.0 * i / (n - 1) - 1.0)))
    return vals / vals.sum()


def gaussian_shape(n: int, sigma: float = 0.2) -> np.ndarray:
    """Truncated Gaussian centred in the middle of the interval."""
    i = np.arange(n, dtype=float)
    x = (i - (n - 1) / 2) / ((n - 1) / 2)
    vals = np.exp(-0.5 * (x / sigma) ** 2)
    return vals / vals.sum()


def exponential_shape(n: int, decay: float = 0.02) -> np.ndarray:
    """Symmetric exponential decay from the centre."""
    i = np.arange(n, dtype=float)
    vals = np.exp(-decay * np.abs(i - (n - 1) / 2))
    return vals / vals.sum()


def powerlaw_shape(n: int, p: float, eps: float = 0.001) -> np.ndarray:
    """Symmetric power‑law: a_i ∝ ((i+ε)(n‑1‑i+ε))^{-p} (p may be negative)."""
    i = np.arange(n, dtype=float)
    base = ((i + eps) * (n - 1 - i + eps)) ** (-p)
    if base.sum() == 0.0:
        return uniform_shape(n)
    return base / base.sum()


def beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    """Beta‑like density on a discrete grid."""
    i = np.arange(n, dtype=float)
    base = (i + 0.5) ** (alpha - 1) * (n - 1 - i + 0.5) ** (beta - 1)
    total = base.sum()
    if total == 0.0:
        return uniform_shape(n)
    return base / total


# ----------------------------------------------------------------------
# Soft‑max utilities used by the optimiser
# ----------------------------------------------------------------------


def softmax_weights(conv: np.ndarray, lam: float) -> np.ndarray:
    """Stable soft‑max over convolution entries."""
    x = lam * conv
    mx = np.max(x)
    w = np.exp(x - mx)
    w /= w.sum()
    return w


def gradient_from_weights(a: np.ndarray, w: np.ndarray) -> np.ndarray:
    """
    Gradient of Σ_k w_k·(a*a)_k w.r.t a.
    Implements   g_j = 2· Σ_m a_m · w_{m+j}.
    Computed via one FFT‑based convolution.
    """
    w_rev = w[::-1]
    conv = convolution_fft(a, w_rev)                 # length = 3·n‑2
    idx = 2 * len(a) - 2
    grads = 2.0 * conv[idx - np.arange(len(a))]
    return grads


# ----------------------------------------------------------------------
# Optimiser (soft‑max exponentiated gradient)
# ----------------------------------------------------------------------


def optimise_sequence(
    initial_a: np.ndarray,
    deadline: float,
    *,
    lam_start: float = 5.0,
    lam_end: float = 200.0,
    lam_ramp_iters: int = 6000,
    eta0: float = 0.6,
    max_iter: int = 200_000,
    noise_every: int = 0,
    noise_scale: float = 0.0,
) -> Tuple[np.ndarray, float]:
    """
    Exponentiated‑gradient descent on a smooth approximation of the max‑convolution.
    Returns the best vector seen and its (non‑smooth) score.
    """
    a = normalize(initial_a.astype(float).copy())
    best_a = a.copy()
    best_sc = compute_score(a)

    n = len(a)

    for it in range(1, max_iter + 1):
        if time.time() >= deadline:
            break

        # temperature schedule
        lam = lam_start + (lam_end - lam_start) * min(1.0, it / lam_ramp_iters)

        # current convolution and soft‑max weights
        conv = convolution_fft(a, a)                     # length 2n‑1
        w = softmax_weights(conv, lam)

        # gradient of the smooth surrogate
        grad = gradient_from_weights(a, w)

        # learning‑rate schedule (1/√t)
        eta = eta0 / np.sqrt(it)

        # multiplicative update (mirrored‑descent on the simplex)
        a = a * np.exp(-eta * grad)

        # occasional exploration noise
        if noise_every and (it % noise_every == 0):
            a += np.random.randn(n) * noise_scale
            a = np.maximum(a, 0.0)

        # project back onto the simplex
        a = normalize(a)

        cur_sc = compute_score(a)
        if cur_sc < best_sc:
            best_sc = cur_sc
            best_a = a.copy()

        if it % 2000 == 0:
            print(f"      iter={it:,d} λ={lam:.1f} cur={cur_sc:.6f} best={best_sc:.6f}")

    return best_a, best_sc

            # Allocate iteration budget proportional to remaining time.
            remaining = max(0.0, deadline - time.time())
            # Heuristic: ~30 optimiser steps per second.
            max_iter = int(min(150_000, max(2_000, remaining * 30)))

            refined, _ = optimise_sequence(
                initial_a=cand,
                deadline=deadline,
                lam_start=5.0,
                lam_end=250.0,
                lam_ramp_iters=8000,
                eta0=0.6,
                max_iter=max_iter,
                noise_every=2000,
                noise_scale=5e-6,
            )

            # Evaluate with the official (slower) evaluator.
            try:
                val = evaluate_sequence(refined.tolist())
            except Exception:
                val = float("inf")
            print(f"    evaluated score = {val:.6f}")

            if val < best_val:
                best_val = val
                best_seq = refined.tolist()
                print(f"    *** New global best: {best_val:.6f} (n={n}) ***")

    # ------------------------------------------------------------------
    # 2. Final polishing of the overall best sequence (if time permits).
    # ------------------------------------------------------------------
    if best_seq and time.time() < deadline - 5.0:
        a = np.array(best_seq, dtype=float)
        remaining = max(0.0, deadline - time.time())
        max_iter = int(min(120_000, remaining * 30))

        print("\nFinal polishing of the global best")
        a_refined, _ = optimise_sequence(
            initial_a=a,
            deadline=deadline,
            lam_start=5.0,
            lam_end=350.0,
            lam_ramp_iters=12_000,
            eta0=0.5,
            max_iter=max_iter,
            noise_every=800,
            noise_scale=2e-6,
        )
        final_val = evaluate_sequence(a_refined.tolist())
        print(f"Final polished score = {final_val:.6f}")
        if final_val < best_val:
            best_val = final_val
            best_seq = a_refined.tolist()
            print("*** Updated global best after final polish ***")

    # ------------------------------------------------------------------
    # 3. Guard – always return a feasible sequence.
    # ------------------------------------------------------------------
    if not best_seq:
        best_seq = [1.0] * 256
        best_val = evaluate_sequence(best_seq)

    print(f"\nSearch finished. Best reported score = {best_val:.6f}")
    return best_seq
