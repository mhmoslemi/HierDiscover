        # Mix

        # Project to simplex (should already sum 1)
        current = project_to_simplex(current)  # keep nonnegative sum=1
        # Evaluate
        cur_score = compute_score_fast(current)
        if cur_score < best_score:
            best_score = cur_score
            best = current.copy()
            print(f" LP iter {iter_idx} best score {best_score:.6f}")
        # Optionally reduce t gradually
        t = max(t * 0.95, 0.01)
    return best, best_score

"""
Step‑function optimizer for the “max‑convolution / sum²” objective.

The algorithm combines three ideas:

1.  Random (or deterministic) starts on the probability simplex.
2.  A small number of linear‑programming (LP) steps.
    – For the current vector f we compute M = max(f∗f).
    – Keeping M fixed we maximise 𝟙ᵀb subject to the linear constraints
          (f ∗ b)[k] ≤ M   for all k,
      which pushes the mass of b into indices where f is small.
    – The new candidate g = b (renormalised) is mixed with f:
          f ← (1‑t)·f + t·g   (t starts large and slowly shrinks).
    This “LP‑step” is exactly the iteration described in the problem statement
      but implemented with SciPy’s high‑performance HiGHS solver.
3.  A cheap stochastic hill‑climber that swaps or transfers a tiny fraction
    of mass while always evaluating the true (non‑smoothed) score.

All vectors are kept on the probability simplex, so the objective reduces to

        score(f) = 2·n·max(f∗f)               (∑f = 1)

and scaling of the final list does not change the reported value.
The routine respects the 1000 s time budget, prints intermediate scores,
and finally returns the best list of floats found.
"""

import time
import numpy as np
from scipy.optimize import linprog

# ----------------------------------------------------------------------
# Basic linear‑algebra helpers
# ----------------------------------------------------------------------


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of a vector onto the probability simplex."""
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / v.shape[0])

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, v.shape[0] + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.where(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full convolution a * b using a real FFT (fast)."""
    m = len(a) + len(b) - 1
    n_fft = next_pow2(m)
    A = np.fft.rfft(a, n=n_fft)
    B = np.fft.rfft(b, n=n_fft)
    C = A * B
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


def compute_score_fast(a: np.ndarray) -> float:
    """
    Fast evaluation when ∑a = 1.
    Returns 2·n·max(a∗a).
    """
    n = a.shape[0]
    conv = convolution_fft(a, a)
    max_val = float(np.max(conv))
    return 2.0 * n * max_val


# ----------------------------------------------------------------------
# Initialisation helpers
# ----------------------------------------------------------------------


def arcsine_initial(n: int) -> np.ndarray:
    """Arcsine start: a_i ∝ 1/√((i+½)(n‑½‑i))."""
    i = np.arange(n, dtype=float)
    eps = 0.25
    vals = 1.0 / np.sqrt((i + 0.5) * (n - 0.5 - i) + eps)
    return project_to_simplex(vals)


def dirichlet_initial(n: int, conc: float) -> np.ndarray:
    """Draw a random point from Dirichlet(α,…,α)."""
    vec = np.random.dirichlet(np.full(n, conc))
    return vec


def powerlaw_initial(n: int, p: float) -> np.ndarray:
    """a_i ∝ (i+1)^{-p} (then projected onto the simplex)."""
    i = np.arange(1, n + 1, dtype=float)
    vals = i ** (-p)
    return project_to_simplex(vals)


# ----------------------------------------------------------------------
# Linear‑programming based improvement step
# ----------------------------------------------------------------------


def lp_improve(f: np.ndarray,
               t: float,
               max_lp_iters: int,
               time_deadline: float) -> tuple[np.ndarray, float]:
    """
    Repeatedly solves the LP
        maximise Σ b_j
        subject to   (f ∗ b)_k ≤ M   for all k,
                     b ≥ 0,
    where M = max(f∗f).
    The solution b is normalised to sum 1 and mixed with f:
        f ← (1‑t)·f + t·b .
    The mixing parameter t is decreased geometrically.
    Returns the best vector found and its fast score.
    """
    n = f.shape[0]
    cur = f.copy()
    best = cur.copy()
    best_score = compute_score_fast(best)

    # pre‑allocate index arrays for the Toeplitz constraint matrix
    i = np.arange(n, dtype=int)
    j = np.arange(n, dtype=int)
    rows = (j[:, None] + i[None, :]).ravel()          # shape (n*n,)
    cols = np.repeat(j, n)                            # shape (n*n,)

    for it in range(max_lp_iters):
        if time.time() >= time_deadline:
            break

        # current max‑convolution (M)
        conv = convolution_fft(cur, cur)
        M = float(np.max(conv))

        # Build the (2n‑1)×n constraint matrix A·b ≤ M.
        # For a fixed row i (offset k) only entries where 0≤k−j<n appear.
        data = np.tile(cur, n)                         # copies of f_i
        A = scipy.sparse.csr_matrix((data, (rows, cols)),
                                    shape=(2 * n - 1, n))

        # Linear program: maximise 1ᵀb   →   minimise –1ᵀb
        c = -np.ones(n, dtype=float)
        b_ub = np.full(2 * n - 1, M, dtype=float)
        bounds = [(0.0, None)] * n

        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds,
                      method='highs', options={'presolve': True})

        if not res.success:
            # If the LP fails (should be rare) we abort the LP phase.
            break

        b = res.x
        sum_b = np.sum(b)
        if sum_b <= 0:
            break

        # Normalise b to the simplex and mix.
        g = b / sum_b
        cur = (1.0 - t) * cur + t * g
        # Numerical drift → re‑project onto the simplex.
        cur = project_to_simplex(cur)

        cur_score = compute_score_fast(cur)
        if cur_score < best_score:
            best_score = cur_score
            best = cur.copy()
            print(f"    LP iter {it+1:3d} – best score {best_score:.6f}")

        # shrink the mixing factor slowly
        t = max(t * 0.95, 0.01)

    return best, best_score


# ----------------------------------------------------------------------
# Simple stochastic hill‑climber (true objective)
# ----------------------------------------------------------------------


def hill_climb(seq,
               time_deadline,
               max_steps: int = 120_000) -> tuple[list[float], float]:
    """
    Stochastic hill‑climber that swaps two entries or moves a fraction
    of mass from one index to another.  The true non‑smoothed score
    (evaluate_sequence) is used for acceptance.
    """
    a = np.array(seq, dtype=float)
    n = a.shape[0]

    best_a = a.copy()
    best_score = compute_score_fast(best_a)   # fast proxy (sum=1)

    step_scale = 0.1  # start moving 10 % of the source mass

    while max_steps > 0 and time.time() < time_deadline:
        cand = best_a.copy()
        if np.random.rand() < 0.5:
            # swap two entries
            i, j = np.random.randint(n, size=2)
            cand[i], cand[j] = cand[j], cand[i]
        else:
            # move a fraction from i to j
            i, j = np.random.randint(n, size=2)
            if i == j:
                continue
            delta = step_scale * cand[i]
            if delta <= 0:
                continue
            cand[i] -= delta
            cand[j] += delta
            # keep numerical negativity harmless
            if cand[i] < 0:
                cand[i] = 0.0

        cand = project_to_simplex(cand)
        cur_score = compute_score_fast(cand)

        if cur_score < best_score:
            best_score = cur_score
            best_a = cand
            step_scale = min(step_scale * 1.07, 0.5)   # slowly enlarge steps
        else:
            step_scale *= 0.97                         # shrink on failure

        max_steps -= 1

    return best_a.tolist(), best_score


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
