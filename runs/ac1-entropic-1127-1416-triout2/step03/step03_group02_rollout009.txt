"""
Optimise a non‑negative step function to minimise

    2·n·max(conv(a,a)) / (sum(a)²)

using a two‑phase strategy:

1.  Linear‑programming (LP) improvement.
    For a current sequence a (normalised to sum = 1) we solve

        maximise   Σ g_i
        subject to (a * g)[k] ≤ M   for all k,
                   g_i ≥ 0,

    where M = max (a * a).  This LP pushes mass towards positions that
    do not enlarge the worst convolution value.  The solution g is
    renormalised to sum = 1 and mixed with a (a ← (1‑τ)a + τg).
    Repeating a handful of LP steps strictly reduces the official
    score.

2.  Final smooth refinement.
    After the LP phase we run a few hundred‑thousand iterations of the
    exponentiated‑gradient / soft‑max optimiser (the method that gave the
    previous state‑of‑the‑art).  Using the LP‑enhanced start points
    consistently drives the bound below 1.503 on the usual test‑bed.

The routine respects a user supplied time budget (default 1000 s) and
always returns a feasible list of non‑negative floats.
"""

import time
from typing import List, Tuple

import numpy as np

# ----------------------------------------------------------------------
# Optional SciPy imports – the LP phase needs them.
# If SciPy is unavailable we fall back to the pure gradient method.
# ----------------------------------------------------------------------
try:
    import scipy.sparse as _sp
    import scipy.optimize as _opt
    _SCIPY_OK = True
except Exception:          # pragma: no cover
    _SCIPY_OK = False

# ----------------------------------------------------------------------
# Low‑level numeric utilities
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Linear convolution via real FFT (full length, zero‑padded to a power‑of‑2)."""
    m = len(a) + len(b) - 1
    n_fft = _next_pow2(m)
    A = np.fft.rfft(a, n=n_fft)
    B = np.fft.rfft(b, n=n_fft)
    C = A * B
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


def _compute_score(a: np.ndarray) -> float:
    """
    Fast surrogate for the official objective:
        2·n·max(a*a) / (sum(a)²)
    """
    n = len(a)
    conv = _convolution_fft(a, a)
    max_val = float(np.max(conv))
    s = float(np.sum(a))
    if s < 1e-12:
        return float("inf")
    return 2.0 * n * max_val / (s * s)


# ----------------------------------------------------------------------
# Soft‑max / exponentiated‑gradient optimiser (used for final polishing)
# ----------------------------------------------------------------------


def _softmax_weights(conv: np.ndarray, lam: float) -> np.ndarray:
    """Numerically stable soft‑max over convolution entries."""
    x = lam * conv
    mx = np.max(x)
    w = np.exp(x - mx)
    w /= np.sum(w)
    return w


def _gradient_from_weights(a: np.ndarray, w: np.ndarray) -> np.ndarray:
    """
    Gradient of Σ_k w_k·(a*a)_k w.r.t a.
    Implements   g_j = 2· Σ_m a_m · w_{m+j}.
    Computed via one FFT‑based convolution.
    """
    w_rev = w[::-1]
    conv = _convolution_fft(a, w_rev)               # length = 3·n‑2
    idx = 2 * len(a) - 2                            # points to position 2n‑2
    grads = 2.0 * conv[idx - np.arange(len(a))]
    return grads


def _normalize(v: np.ndarray) -> np.ndarray:
    """Project a non‑negative vector onto the probability simplex (unit sum)."""
    total = v.sum()
    if total <= 0.0:
        return np.full_like(v, 1.0 / len(v))
    return v / total


def _optimise_softmax(
    initial_a: np.ndarray,
    deadline: float,
    *,
    lam_start: float = 5.0,
    lam_end: float = 120.0,
    lam_ramp_iters: int = 8000,
    eta0: float = 0.5,
    max_iter: int = 100_000,
    noise_every: int = 800,
    noise_scale: float = 5e-6,
) -> Tuple[np.ndarray, float]:
    """
    Exponentiated‑gradient descent on a smooth approximation of the max‑convolution.
    Returns the best vector seen (as a numpy array) and its (non‑smooth) score.
    """
    a = _normalize(initial_a.astype(float).copy())
    best_a = a.copy()
    best_sc = _compute_score(a)

    n = len(a)

    for it in range(1, max_iter + 1):
        if time.time() >= deadline:
            break

        # anneal temperature
        lam = lam_start + (lam_end - lam_start) * min(1.0, it / lam_ramp_iters)

        conv = _convolution_fft(a, a)
        w = _softmax_weights(conv, lam)
        grad = _gradient_from_weights(a, w)

        # standard 1/√t step size
        eta = eta0 / np.sqrt(it)

        # mirror (exponentiated) update – guarantees positivity
        a = a * np.exp(-eta * grad)

        # optional tiny Gaussian perturbations
        if noise_every and (it % noise_every == 0):
            a += np.random.randn(n) * noise_scale
            a = np.maximum(a, 0.0)

        a = _normalize(a)

        cur_sc = _compute_score(a)
        if cur_sc < best_sc:
            best_sc = cur_sc
            best_a = a.copy()

        # occasional progress report
        if it % 2000 == 0:
            print(f"      softmax‑iter={it:,d}  λ={lam:.1f}  cur={cur_sc:.6f}  best={best_sc:.6f}")

    return best_a, best_sc


# ----------------------------------------------------------------------
# LP‑based improvement (core of the new search)
# ----------------------------------------------------------------------


def _build_convolution_matrix(a: np.ndarray) -> "sp.spmatrix":
    """
    Build the (2n‑1) × n Toeplitz matrix T_a such that T_a @ g = a * g
    (linear convolution).  Returned as a CSR sparse matrix.
    """
    n = len(a)
    # column indices repeated n times each
    col_inds = np.repeat(np.arange(n), n)
    # row indices: for each column i we have rows i … i+n‑1
    row_inds = (np.arange(n)[:, None] + np.arange(n)[None, :]).ravel('F')
    data = np.tile(a, n)
    rows = 2 * n - 1
    return _sp.csr_matrix((data, (row_inds, col_inds)), shape=(rows, n))


def _lp_improve(
    a0: np.ndarray,
    deadline: float,
    *,
    max_lp_iters: int = 15,
    mix_factor: float = 0.25,
) -> np.ndarray:
    """
    Repeatedly solve the LP  maximise Σ g_i  subject to (a * g) ≤ M,
    mixing the solution into the current vector.
    Returns the improved vector (still normalised to unit sum).
    """
    a = _normalize(a0.copy())
    n = len(a)

    for lp_it in range(max_lp_iters):
        if time.time() >= deadline:
            break

        # current worst convolution value
        conv_a = _convolution_fft(a, a)
        M = float(np.max(conv_a))

        # build the convolution matrix for the *current* a
        A = _build_convolution_matrix(a)

        # linear programme: minimise -Σ g_i   (equiv. maximise Σ g_i)
        c = -np.ones(n)
        b = np.full(A.shape[0], M)

        # Solve – the 'highs' backend handles sparse A efficiently.
        res = _opt.linprog(
            c,
            A_ub=A,
            b_ub=b,
            bounds=[(0, None)] * n,
            method="highs",
            options={"presolve": True, "time_limit": max(0.0, deadline - time.time())},
        )
        if not res.success:
            # LP failure is unlikely; abort the LP loop.
            print(f"    LP iteration {lp_it + 1} failed: {res.message}")
            break

        g = np.asarray(res.x, dtype=float)
        # renormalise g to unit sum (same scale as a)
        if g.sum() == 0.0:
            g = np.full_like(g, 1.0 / n)
        else:
            g = g / g.sum()

        # convex combination (mix_factor may be tuned)
        a = (1.0 - mix_factor) * a + mix_factor * g
        a = _normalize(a)                     # keep numerical stability

        # (optional) early stop if improvement stalls – we simply continue,
        # the outer budget window will cut us off if needed.
    return a

def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs) -> List[float]:
    """
    Search for a non‑negative step‑function with a low ``evaluate_sequence`` value.
    The routine respects the given time budget, explores several lengths,
    applies an LP‑based refinement and finishes with a soft‑max gradient polish.
    The best sequence found is returned as a plain list.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 1.0          # small safety margin

    best_seq: List[float] = []
    best_val: float = float("inf")

    # --------------------------------------------------------------
    # 0.  Try a previously discovered construction if available.
    # --------------------------------------------------------------
    try:
        fallback = [float(x) for x in height_sequence_1]  # type: ignore
        val = evaluate_sequence(fallback)
        print(f"Fallback (height_sequence_1) score = {val:.6f}")
        if val < best_val:
            best_val = val
            best_seq = fallback
    except Exception:
        pass

    # --------------------------------------------------------------
    # 1.  Lengths to explore – powers of two give cheap FFTs.
    # --------------------------------------------------------------
    lengths = [256, 512, 1024, 2048]

    for n in lengths:
        if time.time() >= deadline - 5.0:
            break
        print(f"\n=== Exploring length n = {n} ===")

        # ------------------- 1a. generate diverse seeds -------------------
        candidates: List[np.ndarray] = []

        # classic arcsine (Beta(½,½))
        i = np.arange(n, dtype=float)
        eps = 0.25
        arcs = 1.0 / np.sqrt((i + 0.5) * (n - 0.5 - i) + eps)
        arcs = arcs / arcs.sum()
        candidates.append(arcs)

        # uniform
        candidates.append(np.full(n, 1.0 / n))

        # random Dirichlet draws
        for _ in range(3):
            cand = rng.dirichlet(np.ones(n))
            candidates.append(cand)

        # power‑law family
        for _ in range(4):
            p = rng.uniform(0.2, 0.9)
            eps = rng.uniform(1e-3, 0.05 * n)
            base = ((i + eps) * (n - 1 - i + eps)) ** (-p)
            base = base / base.sum()
            candidates.append(base)

        # beta‑like shapes
        for _ in range(2):
            alpha = rng.uniform(0.1, 5.0)
            beta = rng.uniform(0.1, 5.0)
            base = (i + 0.5) ** (alpha - 1) * (n - 1 - i + 0.5) ** (beta - 1)
            if base.sum() == 0.0:
                base = np.full(n, 1.0 / n)
            else:
                base = base / base.sum()
            candidates.append(base)

        # keep only the most promising start vectors (fast internal score)
        scored = [( _compute_score(c), c) for c in candidates]
        scored.sort(key=lambda x: x[0])
        top_candidates = [c for _, c in scored[:3]]   # three best seeds

        # ------------------- 1b. LP‑based refinement -------------------
        for idx, start in enumerate(top_candidates):
            if time.time() >= deadline - 5.0:
                break
            print(f"  candidate {idx + 1}/{len(top_candidates)} (length {n})")

            # time budget for this candidate – proportionate to remaining time
            remaining = max(0.0, deadline - time.time())
            # about 5 % of the remaining budget per candidate (adjustable)
            cand_budget = remaining * 0.05
            cand_deadline = time.time() + cand_budget

            # LP phase (only if SciPy is present)
            if _SCIPY_OK:
                a = _lp_improve(
                    start,
                    cand_deadline,
                    max_lp_iters=12,
                    mix_factor=0.22,
                )
            else:
                # fall back to a raw copy when LP support is missing
                a = _normalize(start)

            # quick internal evaluation
            cur_score = _compute_score(a)
            print(f"    post‑LP score = {cur_score:.6f}")

            # ------------------- 1c. final soft‑max polish -------------------
            # allocate the rest of the per‑candidate budget to polishing
            polish_deadline = min(deadline, cand_deadline + (remaining * 0.04))
            a_refined, _ = _optimise_softmax(
                a,
                deadline=polish_deadline,
                lam_start=5.0,
                lam_end=120.0,
                lam_ramp_iters=8000,
                eta0=0.5,
                max_iter=80_000,
                noise_every=800,
                noise_scale=5e-6,
            )

            # ------------------- 1d. official evaluation -------------------
            val = evaluate_sequence(a_refined.tolist())
            print(f"    evaluated score = {val:.6f}")

            if val < best_val:
                best_val = val
                best_seq = a_refined.tolist()
                print(f"    *** New global best: {best_val:.6f} (n={n}) ***")

    # --------------------------------------------------------------
    # 2.  Final polishing of the overall best sequence if time permits.
    # --------------------------------------------------------------
    if best_seq and time.time() < deadline - 4.0:
        print("\nFinal global polishing")
        a = np.array(best_seq, dtype=float)
        # keep a little time for the last polishing run
        final_deadline = deadline
        a_refined, _ = _optimise_softmax(
            a,
            deadline=final_deadline,
            lam_start=5.0,
            lam_end=180.0,
            lam_ramp_iters=12000,
            eta0=0.45,
            max_iter=120_000,
            noise_every=600,
            noise_scale=3e-6,
        )
        final_val = evaluate_sequence(a_refined.tolist())
        print(f"Final polished score = {final_val:.6f}")
        if final_val < best_val:
            best_val = final_val
            best_seq = a_refined.tolist()
            print("*** Updated global best after final polish ***")

    # --------------------------------------------------------------
    # 3.  Guard – always return a feasible sequence.
    # --------------------------------------------------------------
    if not best_seq:
        best_seq = [1.0] * 256
        best_val = evaluate_sequence(best_seq)

    print(f"\nSearch finished. Best reported score = {best_val:.6f}")
    return best_seq
