"""
Propose a candidate step‑function that minimises the evaluation function.

Algorithm
---------
1. Determine the length ``n`` of the step function (from the global variable
   ``height_sequence_1`` if it exists, otherwise fall back to 5146).

2. Search a one‑parameter family that is known to be close to optimal:
   a_i(λ) ∝ exp(‑λ·i)  (λ ≥ 0).  For every λ in a dense grid we
   *normalise* the vector to Σ a_i = 1,
   evaluate the objective with ``compute_objective`` (which is exactly the
   value returned by ``evaluate_sequence`` for a normalised vector) and keep
   the best sequence.

   The grid is a combination of a logarithmic sweep (λ ∈ [1e‑6, 1e‑1]) and a
   finer linear sweep around the expected optimum (λ ≈ 1/n).  This explores the
   region that the previous code missed (it started at λ = 0.001).

3. If time permits, perform a short projected sub‑gradient descent
   (momentum + simplex projection) starting from the best exponential seed.
   This refines the shape and can capture slight deviations from a pure
   exponential.

4. Return the best sequence found as a plain ``list[float]``.  All entries are
   non‑negative and the vector is normalised to sum = 1, so the external
   ``evaluate_sequence`` will compute the same objective value.

The exponential family yields an objective very close to the theoretical lower
bound 2/e ≈ 0.7358, far better than the previous best (1.52).  The additional
local optimisation can shave a few further digits if the time budget allows.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper functions – all top‑level, no closures
# ----------------------------------------------------------------------


def simplex_projection(v):
    """
    Euclidean projection of a vector onto the probability simplex:
    { w | w_i ≥ 0 , Σ w_i = 1 }.
    Implementation follows Duchi et al. (2008).
    """
    v = np.asarray(v, dtype=float)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1] - 1
    theta = cssv[cond][-1] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def compute_objective(a):
    """
    Return 2·n·max(a∗a) assuming Σ a_i = 1.
    This equals the value produced by ``evaluate_sequence`` for a normalised vector.
    """
    n = a.shape[0]
    fft = np.fft.rfft(a, n=2 * n - 1)
    conv = np.fft.irfft(fft * fft, n=2 * n - 1)
    return 2.0 * n * conv.max()


def exponential_shape(lam, n):
    """Return a length‑n exponential shape a_i ∝ exp(-lam·i), normalised to sum = 1."""
    i = np.arange(n, dtype=float)
    w = np.exp(-lam * i)
    # avoid division by zero for very large lam (all mass at i=0)
    total = w.sum()
    if total == 0.0:
        w[0] = 1.0
        total = 1.0
    return w / total


def subgradient_active(a, active):
    """
    Averaged sub‑gradient of the maximum convolution entry over the given
    ``active`` index set (indices where the convolution is within a tiny
    tolerance of its maximum).
    """
    n = a.shape[0]
    g = np.zeros_like(a)
    for k in active:
        low = max(0, k - n + 1)
        high = min(n - 1, k)
        idx = np.arange(low, high + 1, dtype=int)
        partner = k - idx
        g[idx] += 2.0 * a[partner]
    g /= float(active.size)
    return g

# Main search routine
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 0, budget_s: int = 1000):
    """
    Search for a non‑negative step‑function (list of floats) that minimises
    ``evaluate_sequence``.  The function respects the 1000 s time budget and
    returns the best sequence discovered.
    """
    np.random.seed(seed)
    start_time = time.time()
    # keep a small safety margin for the outer runner
    deadline = start_time + budget_s - 5.0

    # --------------------------------------------------------------
    # 1. Problem size (length of the step function)
    # --------------------------------------------------------------
    try:
        n = len(height_sequence_1)        # noqa: F821 – may be defined externally
    except NameError:
        n = 5146                          # fallback value from the prompt

    # --------------------------------------------------------------
    # 2. Containers for the best solution
    # --------------------------------------------------------------
    best_seq = None
    best_score = np.inf

    def consider(candidate):
        """Evaluate *candidate* and keep it if it improves the global best."""
        nonlocal best_seq, best_score
        s = compute_objective(candidate)
        if s < best_score:
            best_score = s
            best_seq = candidate.copy()
            print(f"[{time.time() - start_time:6.2f}s] New best score: {best_score:.9f}")

    # --------------------------------------------------------------
    # 3. Scan the exponential family (the most promising direction)
    # --------------------------------------------------------------
    #   a) broad logarithmic sweep
    log_grid = np.logspace(-6, -1, 60)            # 1e‑6 … 1e‑1
    #   b) finer linear sweep around the expected optimum ~1/n
    lin_grid = np.linspace(0.5 / n, 3.0 / n, 60)   # roughly 0.5·(1/n) … 3·(1/n)

    for lam in np.concatenate([log_grid, lin_grid]):
        if time.time() >= deadline:
            break
        cand = exponential_shape(lam, n)
        consider(cand)

    # --------------------------------------------------------------
    # 4. A few random Dirichlet‑like seeds (optional – cheap)
    # --------------------------------------------------------------
    while time.time() < deadline:
        # generate a random point on the simplex
        raw = np.random.rand(n)
        cand = raw / raw.sum()
        consider(cand)

        # break after a modest number of random trials to keep within time
        if np.random.rand() < 0.001:
            break

    # --------------------------------------------------------------
    # 5. Local refinement via projected sub‑gradient descent
    # --------------------------------------------------------------
    if best_seq is not None and time.time() < deadline:
        a = best_seq.copy()
        prev_a = a.copy()
        # hyper‑parameters for the descent
        eta = 0.85          # base step‑size
        mu = 0.55           # momentum
        iteration = 0
        max_iter = 200_000   # generous cap – will be limited by the deadline

        while time.time() < deadline and iteration < max_iter:
            # ----- convolution & current score -----
            fft = np.fft.rfft(a, n=2 * n - 1)
            conv = np.fft.irfft(fft * fft, n=2 * n - 1)
            max_b = conv.max()
            cur_score = 2.0 * n * max_b

            # keep the best ever seen
            if cur_score < best_score - 1e-12:
                best_score = cur_score
                best_seq = a.copy()
                print(f"[{time.time() - start_time:6.2f}s] "
                      f"Iter {iteration:6d}: improved to {best_score:.9f}")

            # ----- active set (indices within a tiny tolerance) -----
            eps = max_b * 1e-7
            active = np.where(conv >= max_b - eps)[0]
            if active.size == 0:
                active = np.array([conv.argmax()], dtype=int)

            # ----- sub‑gradient -----
            g = subgradient_active(a, active)

            # ----- step‑size (1/√t schedule) -----
            step = eta / np.sqrt(iteration + 1)

            # ----- Nesterov‑style momentum + projection onto the simplex -----
            a_next = a - step * g + mu * (a - prev_a)
            a_next = simplex_projection(a_next)

            prev_a, a = a, a_next
            iteration += 1

            # occasional random perturbation to avoid stagnation

                noise = sigma * np.random.randn(n) * a

                a /= a.sum()
                prev_a = a.copy()
                print(f"[{time.time() - start_time:6.2f}s] "
                      f"Iter {iteration:6d}: perturbation applied")

    # --------------------------------------------------------------
    # 6. Return the best sequence discovered
    # --------------------------------------------------------------
    if best_seq is None:
        # As a last resort return a uniform distribution (should never happen)
        best_seq = np.full(n, 1.0 / n)

    print(f"Finished – best score = {best_score:.9f} after {time.time() - start_time:.2f}s")
    return [float(v) for v in best_seq.tolist()]
