"""
A fast, simple “mix‑and‑match” optimiser for the step‑function problem.

The idea is a three‑stage pipeline

1.  Several short Adam runs with a richer set of starting points
    (uniform, geometric, symmetric geometric, power‑law, random Dirichlet
    and the previous best if it exists).  The Adam optimiser works on
    logits of a soft‑max vector so the simplex constraint Σa_i = 1 is
    satisfied automatically.

2.  The best candidate from stage 1 is then refined by a tiny simulated‑annealing
    hill‑climber that moves a small amount of mass from one index to another.
    The moves are accepted if they improve the true objective
    `2·n·max(conv(a,a))/(sum(a)**2)`, otherwise they may be accepted with a
    temperature that decays to zero.

3.  The final best sequence (as a plain list of floats) is returned.

The optimiser respects the time budget (default 1000 s) and prints a few
progress messages.
"""

import time
import numpy as np

# --------------------------------------------------------------
# Helper: fast evaluation of the objective (no security checks)
# --------------------------------------------------------------
def _objective(a: np.ndarray) -> float:
    """Return 2·n·max(conv(a,a)) / (sum(a)**2).  a must be non‑negative."""
    n = a.shape[0]
    # convolution via FFT – n may not be a power of two
    fft_len = 1 << ((2 * n - 2).bit_length())
    fa = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(fa * fa, n=fft_len)[: 2 * n - 1]
    max_b = conv.max()
    s = a.sum()
    if s < 1e-12:                # should never happen
        return np.inf
    return float(2.0 * n * max_b / (s * s))


# --------------------------------------------------------------
# Gradient of the max‑convolution (sub‑gradient)
# --------------------------------------------------------------
def _maxconv_and_subgrad(a: np.ndarray, fft_len: int):
    """
    Compute max(conv(a,a)) and a sub‑gradient of the max w.r.t. a.
    Returns (max_val, grad) where grad has the same shape as a.
    """
    n = a.shape[0]
    fa = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(fa * fa, n=fft_len)[: 2 * n - 1]

    max_val = conv.max()
    # tolerance to catch ties
    tol = max_val * 1e-12 + 1e-14
    active = np.where(conv >= max_val - tol)[0]

    grad = np.zeros_like(a)
    for k in active:
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        idx = np.arange(lo, hi + 1, dtype=int)
        grad[idx] += 2.0 * a[k - idx]

    if active.size > 1:
        grad /= float(active.size)   # any convex combination works
    return max_val, grad


# --------------------------------------------------------------
# Initialisation strategies
# --------------------------------------------------------------
def _initial_sequence(kind: str, n: int, rng: np.random.Generator,
                      prev: np.ndarray | None) -> np.ndarray:
    """
    Produce a feasible start point (non‑negative, Σ=1) for the optimiser.
    ``kind`` can be:
        'prev'            – previous best (adjust length if necessary)
        'uniform'         – constant vector
        'geometric'       – left‑decaying geometric series
        'sym_geometric'   – geometric decay from both ends
        'power'           – (i+1)^‑p + (n‑i)^‑p  (p sampled)
        'random_dirichlet' – Dirichlet(0.5) (sparse)
    """
    if kind == "prev" and prev is not None:
        seq = prev.copy()
        if seq.shape[0] != n:
            if seq.shape[0] > n:
                seq = seq[:n]
            else:
                seq = np.concatenate([seq,
                                      np.zeros(n - seq.shape[0], dtype=seq.dtype)])
        seq = np.maximum(seq, 0.0)
        tot = seq.sum()
        if tot == 0.0:
            return np.full(n, 1.0 / n)
        return seq / tot

    if kind == "uniform":
        return np.full(n, 1.0 / n)

    if kind == "geometric":
        # r close to 1 gives a gentle decay
        r = 0.98 + 0.02 * rng.random()
        vec = r ** np.arange(n)
        return vec / vec.sum()

    if kind == "sym_geometric":
        # both ends heavy, centre light
        r = 0.98 + 0.02 * rng.random()
        left = r ** np.arange(n)
        right = r ** np.arange(n - 1, -1, -1)
        vec = left + right
        return vec / vec.sum()

    if kind == "power":
        p = 0.3 + 1.7 * rng.random()          # p ∈ (0.3, 2.0)
        idx = np.arange(n)
        left = (idx + 1) ** (-p)
        right = (n - idx) ** (-p)
        vec = left + right
        return vec / vec.sum()

    if kind == "random_dirichlet":
        # α = 0.5 gives sparse vectors
        alpha = 0.5
        sample = rng.gamma(alpha, 1.0, size=n)
        return sample / sample.sum()

    # fallback
    return np.full(n, 1.0 / n)


# --------------------------------------------------------------
# Adam optimiser (logits → soft‑max)
# --------------------------------------------------------------
def _adam_optimize(init_a: np.ndarray,
                   deadline: float,
                   restart_id: int,
                   target: float = 1.503) -> tuple[np.ndarray, float]:
    """
    Adam optimiser on the log‑softmax of the probability vector.
    Returns (best_a, best_score) seen before the deadline.
    """
    n = init_a.shape[0]
    fft_len = 1 << ((2 * n - 2).bit_length())

    # logits, avoid log(0)
    eps = 1e-12
    b = np.log(init_a + eps)

    # Adam moments
    m = np.zeros_like(b)
    v = np.zeros_like(b)
    beta1, beta2 = 0.9, 0.999
    adam_eps = 1e-8

    # evaluate start point
    max_val, _ = _maxconv_and_subgrad(init_a, fft_len)
    best_score = float(2 * n * max_val / (init_a.sum() ** 2))
    best_a = init_a.copy()

    stagnation = 0
    it = 0

    while time.time() < deadline:
        it += 1

        # ---------------- logits → probabilities (soft‑max) ----------------
        b_max = np.max(b)
        a = np.exp(b - b_max)
        a = a / a.sum()

        # ---------------- convolution, max and sub‑gradient ----------------
        max_val, grad_a = _maxconv_and_subgrad(a, fft_len)
        cur_score = float(2 * n * max_val / (a.sum() ** 2))

        # ---------------- keep the best ------------------------------------
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            stagnation = 0
            print(f"[Restart {restart_id:02d}] iter {it:6d} – new best = {best_score:.6f}")
        else:
            stagnation += 1

        # ---------------- early stop --------------------------------------
        if best_score < target:
            break

        # ---------------- escape plateaus ---------------------------------
        if stagnation > 12000:
            # random jitter in log‑space
            b += 0.02 * np.random.randn(n)
            stagnation = 0

        # ---------------- Adam update --------------------------------------
        # gradient of original objective (Σa = 1 ⇒ factor 2·n)
        grad_obj = 2.0 * n * grad_a / (a.sum() ** 2)   # sum(a)=1 after soft‑max
        avg = np.dot(a, grad_obj)
        grad_b = a * (grad_obj - avg)                 # Jacobian of soft‑max

        m = beta1 * m + (1.0 - beta1) * grad_b
        v = beta2 * v + (1.0 - beta2) * (grad_b ** 2)

        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        # a slowly decreasing learning‑rate (more aggressive than the original)
        lr = 0.8 / np.sqrt(it)

        b -= lr * m_hat / (np.sqrt(v_hat) + adam_eps)

    return best_a, best_score


# --------------------------------------------------------------
# Local refinement – tiny simulated‑annealing hill climb
# --------------------------------------------------------------
def _local_refinement(a0: np.ndarray,
                      deadline: float,
                      start_temp: float = 1e-3) -> tuple[np.ndarray, float]:
    """
    Starting from a0, repeatedly move a small amount of mass from a random
    source index to a random destination index.  Accept moves that improve
    the objective; otherwise accept with a temperature that decays
    linearly to zero at the deadline.
    """
    a = a0.copy()
    best_a = a.copy()
    best_score = _objective(a)

    n = a.shape[0]
    rng = np.random.default_rng(987654)

    while time.time() < deadline:
        # linear temperature schedule
        remaining = deadline - time.time()
        total = deadline - start_temp  # dummy, just to keep >0
        T = max(0.0, remaining / (deadline - time.time() + 1e-12))

        # choose source (must have some mass) and destination
        src = rng.integers(0, n)
        if a[src] < 1e-12:
            continue
        dst = rng.integers(0, n)
        if src == dst:
            continue

        # move a fraction of source mass (step size slowly decays)
        step = 0.02 * a[src]                # 2 % of the source mass
        a_new = a.copy()
        a_new[src] -= step
        a_new[dst] += step

        # evaluate
        sc = _objective(a_new)

        if sc < best_score or rng.random() < np.exp(-(sc - best_score) / (T + 1e-12)):
            a = a_new
            if sc < best_score:
                best_score = sc
                best_a = a_new.copy()
                print(f"   SA refinement – new best = {best_score:.6f}")

    return best_a, best_score
