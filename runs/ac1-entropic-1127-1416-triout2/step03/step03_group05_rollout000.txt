```
def _project_simplex(v):
    # v is 1-d numpy array
    # Use algorithm from Duchi et al.
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(len(v)) + 1
    cond = u - cssv / ind > 0
    rho = ind[cond].max()
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0)
    return w
```

Works for sum to 1 (non-negative). Ensure that v could be small negative; algorithm works.

"""
Minimal‑max autocorrelation via Entropic Mirror Descent.

The objective 2·n·max(conv(a,a))/(sum(a)**2) is invariant under scaling,
so we fix Σa_i = 1 and minimise 2·n·max(conv(a,a)).
The algorithm is a projected mirror‑descent on the probability simplex
using the exponentiated‑gradient (entropy) update:

    a ← a ∘ exp(‑η·∇obj)      (∘ = element‑wise product)
    a ← a / Σa                (renormalise)

∇obj = 2·n·∂max(conv)/∂a.  The sub‑gradient of the max‑convolution entry
is computed by an FFT (O(n log n)).  Several restarts with different
initialisations (previous best, uniform, Dirichlet‑sparse, geometric) are
run within the time budget.  Stagnation triggers a small random jitter
followed by a simplex projection.  The best sequence found is returned.

The method differs from the previous Adam‑based approach and is designed
to explore the simplex more aggressively, improving the bound below the
target 1.503.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Simplex utilities
# ----------------------------------------------------------------------


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """ Euclidean projection of vector v onto the probability simplex. """
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all entries are zero – return uniform
        return np.full(n, 1.0 / n)
    rho = ind[cond].max()
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Convolution + sub‑gradient of the maximal entry
# ----------------------------------------------------------------------


def _maxconv_and_subgrad(a: np.ndarray, fft_len: int):
    """
    Return (max_val, grad) where max_val = max(conv(a,a))
    and grad = ∂max_val / ∂a (a sub‑gradient).
    """
    n = a.shape[0]
    # aperiodic convolution via FFT
    fft = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(fft * fft, n=fft_len)[: 2 * n - 1]

    max_val = conv.max()
    # tolerance for ties (numerical)
    tol = max_val * 1e-12 + 1e-14
    active = np.where(conv >= max_val - tol)[0]

    grad = np.zeros_like(a)
    for k in active:
        low = max(0, k - n + 1)
        high = min(n - 1, k)
        idx = np.arange(low, high + 1, dtype=int)
        grad[idx] += 2.0 * a[k - idx]

    if active.size > 1:
        grad /= float(active.size)
    return max_val, grad


# ----------------------------------------------------------------------
# Initial‑sequence factories (no closures)
# ----------------------------------------------------------------------


def _initial_sequence(kind: str, n: int, rng: np.random.Generator,
                      prev: np.ndarray | None) -> np.ndarray:
    """Return a feasible probability vector of length n."""
    if kind == "prev" and prev is not None:
        seq = prev.copy()
        if seq.shape[0] > n:
            seq = seq[:n]
        elif seq.shape[0] < n:
            seq = np.concatenate([seq, np.zeros(n - seq.shape[0])])
        seq = np.maximum(seq, 0.0)
        total = seq.sum()
        if total == 0.0:
            return np.full(n, 1.0 / n)
        return seq / total

    if kind == "uniform":
        return np.full(n, 1.0 / n)

    if kind == "dirichlet":
        # sparse Dirichlet (α < 1)
        alpha = 0.3
        sample = rng.gamma(alpha, 1.0, size=n)
        return sample / sample.sum()

    if kind == "geometric":
        r = 0.95 + 0.05 * rng.random()  # r ∈ (0.95,1.0)
        left = r ** np.arange(n)
        right = r ** np.arange(n - 1, -1, -1)
        seq = left + right
        return seq / seq.sum()

    # fallback
    return np.full(n, 1.0 / n)


# ----------------------------------------------------------------------
# Mirror‑descent optimiser (entropy mirror map)
# ----------------------------------------------------------------------


def _mirror_descent(initial_a: np.ndarray,
                    deadline: float,
                    restart_id: int,
                    n: int,
                    fft_len: int,
                    rng: np.random.Generator,
                    target: float = 1.503) -> tuple[np.ndarray, float]:
    """
    Entropic mirror descent on the simplex.
    Returns the best vector seen before `deadline` together with its score.
    """
    a = initial_a.copy()
    best_a = a.copy()
    # initial evaluation
    max_val, _ = _maxconv_and_subgrad(a, fft_len)
    best_score = 2.0 * n * max_val

    it = 0
    stagnation = 0
    eta0 = 0.5               # initial learning‑rate factor

    while time.time() < deadline:
        it += 1
        max_val, grad = _maxconv_and_subgrad(a, fft_len)
        cur_score = 2.0 * n * max_val

        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            stagnation = 0
            print(f"Restart {restart_id:02d} iter {it:6d}: best = {best_score:.6f}")
        else:
            stagnation += 1

        # stop early if we already hit the target
        if best_score < target:
            break

        # gradient of the objective (note the factor 2·n)
        grad_obj = 2.0 * n * grad

        # learning rate decays like 1/√t
        eta = eta0 / np.sqrt(it)

        # entropy mirror‑map update: a ← a ∘ exp(‑η·grad)
        update = -eta * grad_obj
        # clip to avoid overflow/underflow in the exponent
        update = np.clip(update, -20.0, 20.0)
        a = a * np.exp(update)

        # renormalise to stay on the simplex
        s = a.sum()
        if s == 0.0:                # numerical safety
            a = np.full(n, 1.0 / n)
        else:
            a = a / s

        # occasional jitter to escape flat plateaux
        if stagnation > 2000:
            jitter = rng.normal(scale=0.01, size=n)
            a = np.maximum(a + jitter, 0.0)
            a = _project_simplex(a)
            stagnation = 0

    return best_a, best_score
