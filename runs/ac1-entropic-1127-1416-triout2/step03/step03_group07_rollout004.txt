"""
Propose a step‑function (non‑negative list) that minimizes

        2·n·max(a*a) / (sum(a)**2)

using a projected Adam‑style sub‑gradient descent on the
max‑convolution.  The method runs many short random restarts,
tries a few different lengths (including the historic best if
available), and adds occasional jitter to escape plateaus.
Only the best sequence found before the time budget expires is
returned.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper: max of the a*a convolution and a (sub)gradient of that max.
# ----------------------------------------------------------------------
def _max_and_subgradient(a: np.ndarray) -> tuple[float, np.ndarray]:
    """
    Return the maximum entry of the a*a convolution and a sub‑gradient
    of that entry w.r.t. a.

    The gradient for a given convolution index k is
        ∂/∂a_i (a*a)[k] = 2·a_{k-i}
    for all i that contribute to the index.
    If several indices attain the maximum, their gradients are averaged.
    """
    n = a.shape[0]
    # convolution via FFT (zero‑pad to length 2*n-1)
    fft = np.fft.rfft(a, n=2 * n - 1)
    conv = np.fft.irfft(fft * fft, n=2 * n - 1)
    max_val = float(conv.max())

    # active set (indices tied for the maximum, tolerant to numerical noise)
    tol = max_val * 1e-12 + 1e-14
    active = np.where(conv >= max_val - tol)[0]

    grad = np.zeros_like(a)
    for k in active:
        low = max(0, k - n + 1)          # smallest i with j = k-i in range
        high = min(n - 1, k)             # largest i
        idx = np.arange(low, high + 1, dtype=int)
        partner = k - idx                # corresponding j = k-i
        grad[idx] += 2.0 * a[partner]

    if active.size > 1:
        grad /= float(active.size)

    return max_val, grad


# ----------------------------------------------------------------------
# Main routine
# ----------------------------------------------------------------------
def propose_candidate(budget_s: int = 1000) -> list[float]:
    """
    Search for a non‑negative sequence that minimizes the evaluation
    function.  The routine respects the given time budget (seconds) and
    returns the best sequence found so far.
    """
    # ------------------------------------------------------------------
    # 1. time budget
    # ------------------------------------------------------------------
    deadline = time.time() + budget_s - 2.0   # safety margin

    # ------------------------------------------------------------------
    # 2. candidate lengths to explore
    # ------------------------------------------------------------------
    lengths = [1024, 2048, 4096, 8192, 16384]
    # add the historic length if available
    if 'height_sequence_1' in globals():
        try:
            prev_len = len(height_sequence_1)
            lengths.append(prev_len)
        except Exception:
            pass
    lengths = list(set(lengths))               # deduplicate

    # ------------------------------------------------------------------
    # 3. optimisation parameters
    # ------------------------------------------------------------------
    target_score = 1.503                      # we stop early if we beat this
    best_score = float("inf")
    best_a = None
    restart = 0

    # Adam hyper‑parameters (common for all restarts)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    max_abs_step = 0.5                         # clip overly large steps

    # ------------------------------------------------------------------
    # 4. main optimisation loop (random restarts)
    # ------------------------------------------------------------------
    while time.time() < deadline:
        # pick a length at random (bias toward larger lengths for final phase)
        n = int(np.random.choice(lengths))
        rng = np.random.default_rng()

        # ----- initialise a -------------------------------------------------
        if restart == 0 and 'height_sequence_1' in globals():
            # use the historic sequence, interpolated if lengths differ
            old = np.asarray(height_sequence_1, dtype=float)
            if old.shape[0] != n:
                x_old = np.linspace(0.0, 1.0, old.shape[0])
                x_new = np.linspace(0.0, 1.0, n)
                a = np.interp(x_new, x_old, old)
            else:
                a = old.copy()
        else:
            a = rng.random(n)

        # ensure positivity and project onto the simplex (sum = 1)
        a = np.maximum(a, 1e-12)
        a /= a.sum()

        # Adam moment estimates
        m = np.zeros_like(a)
        v = np.zeros_like(a)
        t_iter = 0
        no_improve = 0
        best_local = float("inf")

        # ----- inner optimisation loop --------------------------------------
        while time.time() < deadline:
            # --- convolution max and its sub‑gradient -----------------------
            max_conv, grad = _max_and_subgradient(a)

            # the objective (since sum(a)==1) is 2*n*max_conv
            cur_score = 2.0 * n * max_conv

            # keep globally best solution
            if cur_score < best_score:
                best_score = cur_score
                best_a = a.copy()
                print(f"[{time.time():.1f}] restart {restart}, n={n}, "
                      f"iter {t_iter}, new best = {best_score:.6f}")

                if best_score < target_score:
                    print("Target reached – returning early.")
                    return [float(v) for v in best_a.tolist()]

            # loss improvement tracking
            if cur_score < best_local - 1e-9:
                best_local = cur_score
                no_improve = 0
            else:
                no_improve += 1
                if no_improve > 5000:
                    # stagnation – break this restart
                    break

            # --- Adam step --------------------------------------------------
            t_iter += 1
            # gradient of the *objective* (factor 2*n) – scaling absorbed by lr
            g_obj = 2.0 * n * grad

            m = beta1 * m + (1.0 - beta1) * g_obj
            v = beta2 * v + (1.0 - beta2) * (g_obj * g_obj)

            m_hat = m / (1.0 - beta1 ** t_iter)
            v_hat = v / (1.0 - beta2 ** t_iter)

            # base learning rate – we scale it roughly with 1/√n
            lr = 0.5 / np.sqrt(n)
            step = lr * m_hat / (np.sqrt(v_hat) + eps)

            # clamp the step size to avoid exploding updates
            max_step = max_abs_step
            if np.max(np.abs(step)) > max_step:
                step = step * (max_step / np.max(np.abs(step)))

            # gradient descent (move opposite to the sub‑gradient)
            a = a - step

            # --- projection onto the probability simplex --------------------
            a = np.maximum(a, 0.0)
            s = a.sum()
            if s == 0.0:
                a = np.full(n, 1.0 / n)
            else:
                a = a / s

            # --- occasional random jitter to escape plateaus ---------------
            if t_iter % 4000 == 0:
                eps_noise = 0.001 * rng.standard_normal(n)
                a = np.maximum(a + eps_noise, 0.0)
                a = a / a.sum()

        restart += 1

    # ------------------------------------------------------------------
    # 5. Return the best sequence found (fallback to uniform if needed)
    # ------------------------------------------------------------------
    if best_a is None:
        # use a simple uniform distribution of the last length tried
        best_a = np.full(n, 1.0 / n)

    return [float(v) for v in best_a.tolist()]
