Implementation of subgradient descent:

```
def subgradient_descent(a, time_deadline, max_iter=200_000, step0=0.1, decay=0.999):
    a = a.copy()
    best_a = a.copy()
    best_score = compute_score(a)
    n = len(a)
    for t in range(1, max_iter+1):
        if time.time() >= time_deadline:
            break
        conv = convolution_fft(a, a)   # length 2n-1
        M = np.max(conv)
        # indicator of max indices (within tolerance)
        inds = np.where(conv >= M - 1e-12)[0]  # list
        # create mask
        mask = np.zeros_like(conv)
        mask[inds] = 1.0
        # gradient = 2 * conv_rev of reversed a with mask
        a_rev = a[::-1]
        # conv_rev length 3n-2
        conv_rev = convolution_fft(a_rev, mask)
        start = n - 1
        grad = 2.0 * conv_rev[start:start + n]
        # Step size schedule
        step = step0 / np.sqrt(t)   # or step0 * decay**t
        a_new = a - step * grad
        a_new = project_to_simplex(a_new)  # keep non-negative sum=1
        cur_score = compute_score(a_new)
        if cur_score < best_score:
            best_score = cur_score
            best_a = a_new.copy()
        a = a_new
    return best_a, best_score
```

"""
Search for a step‑function with a very small evaluation score.

The core idea is to start from the *discrete arcsine* distribution.
For a continuous arcsine density on [0,1] the self‑convolution is exactly
uniform, which yields the theoretical lower bound of the objective
(≈ 1).  By taking
    a_i ∝ 1/√((i+½)(n‑½‑i))
and normalising by the sum (instead of Euclidean projection) we obtain a
discrete analogue whose convolution is already almost flat, giving scores
close to 1 for n in the few‑thousand range.

To tighten the result we perform a few iterations of a simple
sub‑gradient descent that directly minimises the quantity
    M(a) = max_k (a∗a)[k]
and finally polish the best vector with a cheap stochastic hill‑climber.
All operations respect the 1000 s budget and the returned sequence consists
only of non‑negative floats.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Basic convolution utilities (FFT based, O(n log n))
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full linear convolution using a real FFT."""
    m = len(a) + len(b) - 1
    n_fft = next_pow2(m)
    A = np.fft.rfft(a, n=n_fft)
    B = np.fft.rfft(b, n=n_fft)
    C = A * B
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


# ----------------------------------------------------------------------
# Scoring helpers (fast replica of evaluate_sequence when sum(a)=1)
# ----------------------------------------------------------------------


def compute_score(a: np.ndarray) -> float:
    """2·n·max(a*a) for a vector that already sums to 1."""
    n = a.shape[0]
    conv = convolution_fft(a, a)
    return float(2.0 * n * np.max(conv))


# ----------------------------------------------------------------------
# Simple Euclidean projection onto the probability simplex
# ----------------------------------------------------------------------


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Return the Euclidean projection of v onto {x ≥ 0, sum x = 1}."""
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / v.shape[0])

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, v.shape[0] + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.where(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Generators for parametric families (arcsine and power variants)
# ----------------------------------------------------------------------


def arcsine_distribution(n: int, eps: float = 1e-12) -> np.ndarray:
    """
    Discrete arcsine distribution on {0,…,n‑1}.
    a_i ∝ 1 / sqrt( (i+½)(n‑½‑i) + eps ).
    The vector is *strictly* normalised by its L1‑norm (no projection).
    """
    i = np.arange(n, dtype=float)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - 0.5 - i) + eps)
    return vals / np.sum(vals)


def power_distribution(n: int, rho: float, eps: float = 1e-12) -> np.ndarray:
    """
    Generalised power law: a_i ∝ ((i+½)(n‑½‑i)+eps)^(-rho),
    then normalised to sum 1.
    """
    i = np.arange(n, dtype=float)
    vals = ((i + 0.5) * (n - 0.5 - i) + eps) ** (-rho)
    return vals / np.sum(vals)


# ----------------------------------------------------------------------
# Sub‑gradient descent on the true max‑convolution objective
# ----------------------------------------------------------------------


def subgradient_descent(
    a: np.ndarray,
    deadline: float,
    max_iter: int = 200_000,
    step0: float = 0.3,
    tolerance: float = 1e-12,
) -> tuple[np.ndarray, float]:
    """
    Projected sub‑gradient method that directly minimises
        M(a) = max_k (a∗a)[k].

    The (sub‑)gradient of the active constraints is
        g_i = 2 * Σ_{k∈K} a_{k-i},
    where K = {k : (a∗a)[k] = M(a)}.
    """
    n = a.shape[0]
    best_a = a.copy()
    best_val = compute_score(a)

    a_rev = a[::-1]                 # constant for the whole loop
    for t in range(1, max_iter + 1):
        if time.time() >= deadline:
            break

        conv = convolution_fft(a, a)
        M = np.max(conv)

        # indices attaining the maximum (allow tiny tolerance)
        active = np.where(conv >= M - tolerance)[0]
        mask = np.zeros_like(conv)
        mask[active] = 1.0

        # gradient = 2 * (a_rev * mask) restricted back to length n
        conv_rev = convolution_fft(a_rev, mask)          # length = 3n‑2
        start = n - 1
        grad = 2.0 * conv_rev[start:start + n]

        # simple diminishing step size
        step = step0 / np.sqrt(t)
        a_new = a - step * grad
        a_new = project_to_simplex(a_new)

        cur_val = compute_score(a_new)
        if cur_val < best_val:
            best_val = cur_val
            best_a = a_new.copy()

        a = a_new

        # occasional progress report (lightweight)
        if t % 20000 == 0:
            print(f"    Sub‑grad iter {t:,d} – cur {cur_val:.6f} – best {best_val:.6f}")

    return best_a, best_val


# ----------------------------------------------------------------------
# Simple stochastic hill‑climber (final polishing)
# ----------------------------------------------------------------------


def hill_climb(seq, deadline, max_steps=150_000):
    """Swap / small mass‑transfer hill‑climbing using the true objective."""
    a = np.array(seq, dtype=float)
    n = a.shape[0]
    best_a = a.copy()
    best_score = compute_score(best_a)

    step_scale = 0.1

    while max_steps > 0 and time.time() < deadline:
        cand = best_a.copy()
        if np.random.rand() < 0.5:
            i, j = np.random.randint(n, size=2)
            cand[i], cand[j] = cand[j], cand[i]
        else:
            i, j = np.random.randint(n, size=2)
            if i != j:
                delta = step_scale * best_a[i]
                cand[i] -= delta
                cand[j] += delta
        cand = project_to_simplex(cand)
        cur = compute_score(cand)
        if cur < best_score:
            best_score = cur
            best_a = cand
            step_scale = min(step_scale * 1.04, 0.5)
        else:
            step_scale *= 0.97
        max_steps -= 1

    return best_a.tolist(), best_score

# ----------------------------------------------------------------------
