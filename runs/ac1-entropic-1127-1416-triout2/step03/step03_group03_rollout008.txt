"""
Propose a low‑score step function.

The core idea is a *random sparse subset* search.  For a chosen length n
we pick a random set of k indices (k ≈ p·n with 0.2 ≤ p ≤ 0.9) and assign
equal mass 1/k to those positions.  The convolution of such a vector is
the auto‑correlation of a binary indicator scaled by 1/k², so the
objective 2 n·max(a∗a) ≈ 2 n/k²·max(N_k) where N_k is the number of ordered
pairs with a given sum.  For random subsets N_k is close to its average
k²/(2 n), giving a constant factor ≈1.2–1.4, i.e. a score well below the
previous best 1.518.  The search runs many random draws (fast FFT‑based
evaluation) and keeps the best vector found.  After the stochastic phase
a short projected sub‑gradient refinement and a permutation/ mass‑transfer
hill‑climber polish the result.

The routine respects the 1000 s budget, prints progress and finally
returns a plain Python list of non‑negative floats ready for the
`evaluate_sequence` harness.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Simple helper utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full convolution a * b using a real FFT (fast)."""
    m = len(a) + len(b) - 1
    n_fft = next_pow2(m)
    A = np.fft.rfft(a, n=n_fft)
    B = np.fft.rfft(b, n=n_fft)
    C = A * B
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


def compute_score(a: np.ndarray) -> float:
    """
    Fast replica of `evaluate_sequence` when ∑a = 1.
    Returns 2·n·max(a*a)  (exactly the value returned by
    evaluate_sequence for a normalized vector).
    """
    n = a.shape[0]
    conv = convolution_fft(a, a)
    return 2.0 * n * float(np.max(conv))


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the probability simplex
    (non‑negative entries that sum to 1).
    """
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / v.shape[0])

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, v.shape[0] + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.where(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


# ----------------------------------------------------------------------
# Simple hill‑climber (identical to the one used previously but with a
# larger step budget)
# ----------------------------------------------------------------------


def hill_climb_permutation(seq, time_deadline, max_steps=300_000):
    """
    Stochastic hill‑climb that swaps entries or transfers a fraction of mass.
    The true (unsmoothed) objective is used for acceptance.
    """
    a = np.array(seq, dtype=float)
    n = a.shape[0]

    best_a = a.copy()
    best_score = compute_score(best_a)

    step_scale = 0.1  # fraction of mass moved in a transfer

    while max_steps > 0 and time.time() < time_deadline:
        move_type = np.random.rand()
        cand = best_a.copy()

        if move_type < 0.5:
            # simple swap
            i, j = np.random.randint(n, size=2)
            cand[i], cand[j] = cand[j], cand[i]
        else:
            # move a fraction from i to j
            i, j = np.random.randint(n, size=2)
            if i == j:
                continue
            delta = step_scale * best_a[i]
            if delta <= 0:
                continue
            cand[i] -= delta
            cand[j] += delta
            if cand[i] < 0:
                cand[i] = 0.0

        cand = project_to_simplex(cand)
        cur_score = compute_score(cand)

        if cur_score < best_score:
            best_score = cur_score
            best_a = cand
            step_scale = min(step_scale * 1.05, 0.5)  # success → increase step
        else:
            step_scale *= 0.97                     # failure → decrease step

        max_steps -= 1

    return best_a.tolist(), best_score


# ----------------------------------------------------------------------
# Projected sub‑gradient optimisation of the exact max‑convolution
# ----------------------------------------------------------------------


def subgrad_optimize(init_a: np.ndarray,
                     time_deadline: float,
                     max_iter: int = 200_000,
                     step0: float = 0.07) -> tuple[np.ndarray, float]:
    """
    Simple projected sub‑gradient descent on the true objective
    2 n·max(a∗a).  The sub‑gradient at the current maximum shift k
    is g[i] = 2·a[k‑i] (if the index is valid, otherwise 0).
    """
    a = init_a.astype(float).copy()
    best_a = a.copy()
    best_score = compute_score(a)
    n = len(a)

    for t in range(1, max_iter + 1):
        if time.time() >= time_deadline:
            break

        conv = convolution_fft(a, a)
        k = int(np.argmax(conv))

        # sub‑gradient of the max‑convolution
        idx = np.arange(n)
        j = k - idx
        mask = (j >= 0) & (j < n)
        g = np.zeros_like(a)
        g[mask] = 2.0 * a[j[mask]]

        # diminishing step size
        step = step0 / np.sqrt(t)

        a_new = a - step * g
        a_new = project_to_simplex(a_new)

        cur_score = compute_score(a_new)
        if cur_score < best_score:
            best_score = cur_score
            best_a = a_new.copy()

        a = a_new

        # occasional tiny jitter to help escape flat plateaus
        if t % 1000 == 0:
            a += np.random.randn(n) * (step0 * 0.005 / np.sqrt(t))
            a = project_to_simplex(a)

    return best_a, best_score
