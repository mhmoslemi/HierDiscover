    a = a / a.sum()

def resample_to_length(seq, n):
    old_n = len(seq)
    x_old = np.linspace(0, 1, old_n)
    x_new = np.linspace(0, 1, n)
    new_seq = np.interp(x_new, x_old, seq)
    new_seq = np.maximum(new_seq, 0)
    s = new_seq.sum()
    if s == 0:
        new_seq = np.full(n, 1.0 / n)
    else:
        new_seq /= s

"""
propose_candidate
=================

A fast simplex‑projected exponentiated‑gradient optimiser that directly
minimises the *true* max‑convolution (instead of a soft‑max surrogate).
The algorithm works in three stages:

1. **Recovery of a fallback** – if the historical solution
   ``height_sequence_1`` is present it is evaluated first.

2. **Exploratory search** – for a handful of lengths (1024 … 8192)
   a few deterministic (arcsine, uniform) and random Dirichlet starts
   are refined with the EG optimiser.  The best vector found over all
   lengths is kept.

3. **Final polishing** – the globally best vector is run through a long
   EG optimisation with a slowly decaying learning‑rate.

All updates stay on the probability simplex (sum = 1, non‑negative);
the evaluator is called only for the current best vector, so the
runtime budget is spent mostly on cheap surrogate calls.
"""

import time
import numpy as np
from typing import List, Tuple

# ----------------------------------------------------------------------
# Low‑level utilities (no closures)
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """
    Linear convolution using the real FFT.  Returns an array of length
    len(a) + len(b) - 1.
    """
    m = len(a) + len(b) - 1
    n_fft = next_pow2(m)
    A = np.fft.rfft(a, n=n_fft)
    B = np.fft.rfft(b, n=n_fft)
    C = A * B
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


def compute_score(a: np.ndarray) -> float:
    """
    Fast surrogate for the official score when Σa = 1:
        score = 2·n·max(a*a)
    """
    n = a.shape[0]
    conv = convolution_fft(a, a)
    max_val = float(np.max(conv))
    return 2.0 * n * max_val


def arcsine_initial(n: int) -> np.ndarray:
    """
    Classic arcsine shape:
        a_i ∝ 1/√((i+½)*(n‑½‑i))
    Normalised to Σa = 1.
    """
    i = np.arange(n, dtype=float)
    eps = 0.25
    vals = 1.0 / np.sqrt((i + 0.5) * (n - 0.5 - i) + eps)
    vals /= vals.sum()
    return vals


def resample_to_length(seq: np.ndarray, n: int) -> np.ndarray:
    """
    Linear interpolation of a probability vector onto a new length.
    The result is re‑normalised to sum to one.
    """
    old_n = seq.shape[0]
    if old_n == n:
        return seq.copy()
    x_old = np.linspace(0.0, 1.0, old_n)
    x_new = np.linspace(0.0, 1.0, n)
    new_seq = np.interp(x_new, x_old, seq)
    new_seq = np.maximum(new_seq, 0.0)
    s = new_seq.sum()
    if s == 0.0 or not np.isfinite(s):
        new_seq = np.full(n, 1.0 / n)
    else:
        new_seq /= s
    return new_seq


# ----------------------------------------------------------------------
# Exponentiated‑gradient optimiser on the true max‑convolution
# ----------------------------------------------------------------------


def max_conv_and_grad(a: np.ndarray) -> Tuple[float, np.ndarray, int]:
    """
    Returns (max_value, gradient_of_max, index_of_max) for the self‑convolution
    of a.  The gradient is the exact sub‑gradient of the max entry.
    """
    n = a.shape[0]
    conv = convolution_fft(a, a)
    k = int(np.argmax(conv))
    max_val = float(conv[k])

    # gradient:  ∂/∂a_i Σ_j a_j a_{k-j} = 2·a_{k-i}   (if the index exists)
    i = np.arange(n)
    j = k - i
    mask = (j >= 0) & (j < n)
    grad = np.zeros_like(a)
    grad[mask] = 2.0 * a[j[mask]]
    return max_val, grad, k


def eg_optimize(initial_a: np.ndarray,
                time_deadline: float,
                max_iter: int = 200_000,
                eta0: float = 1.0,
                rng: np.random.Generator = None) -> Tuple[np.ndarray, float]:
    """
    Projected exponentiated‑gradient descent on the simplex.
    The objective (for Σa = 1) is 2·n·max(a*a).  Returns the best vector
    seen and its surrogate score.
    """
    if rng is None:
        rng = np.random.default_rng()
    a = np.maximum(initial_a.astype(float), 0.0)
    s = a.sum()
    if s == 0.0 or not np.isfinite(s):
        a = np.full_like(a, 1.0 / a.size)
    else:
        a /= s

    best_a = a.copy()
    best_sc = compute_score(a)
    n = a.shape[0]

    for it in range(1, max_iter + 1):
        if time.time() >= time_deadline:
            break

        # --- sub‑gradient of the true max ---
        _, grad, _ = max_conv_and_grad(a)

        # --- learning‑rate schedule (1/√t) ---
        eta = eta0 / np.sqrt(it)

        # multiplicative EG update, then re‑normalise to the simplex
        a = a * np.exp(-eta * grad)
        # numerical guard
        a = np.maximum(a, 0.0)
        total = a.sum()
        if total == 0.0 or not np.isfinite(total):
            a = np.full_like(a, 1.0 / n)
        else:
            a /= total

        # occasional injected noise to escape plateaus
        if it % 5000 == 0:
            noise = rng.normal(scale=5e-6, size=n)
            a = a + noise
            a = np.maximum(a, 0.0)
            a /= a.sum()

        cur_sc = compute_score(a)
        if cur_sc < best_sc:
            best_sc = cur_sc
            best_a = a.copy()

        # a very light progress report (optional)
        if it % 30_000 == 0:
            print(f"       EG iter={it:,d}  cur={cur_sc:.6f}  best={best_sc:.6f}")

    return best_a, best_sc

def propose_candidate(seed: int = 42,
                     budget_s: int = 1000,
                     **kwargs) -> List[float]:
    """
    Search for a non‑negative step sequence with a low ``evaluate_sequence``
    value.  The routine mixes a quick fallback, a multi‑scale exploration
    and a final polishing phase, all driven by a simplex‑projected
    exponentiated‑gradient optimiser that works on the true max‑convolution.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 2.0          # safety margin

    best_seq: List[float] = []
    best_val: float = float("inf")

    # --------------------------------------------------------------
    # 0.  Fallback from the historic solution (if present)
    # --------------------------------------------------------------
    try:
        fallback = [float(x) for x in height_sequence_1]          # type: ignore
        val = evaluate_sequence(fallback)
        print(f"Fallback (height_sequence_1) score = {val:.6f}")
        if val < best_val:
            best_val = val
            best_seq = fallback
    except Exception:
        pass

    # --------------------------------------------------------------
    # 1.  Multi‑scale exploration
    # --------------------------------------------------------------
    # Lengths that comfortably fit into the time budget.
    lengths = [1024, 2048, 4096, 8192]

    for n in lengths:
        if time.time() >= deadline - 5.0:
            break
        print(f"\n=== Exploring length {n} ===")

        # ---- 1a. deterministic starters ---------------------------------
        starters: List[np.ndarray] = [
            arcsine_initial(n),
            np.full(n, 1.0 / n)                     # uniform
        ]

        # ---- 1b. resample best known shape (if any) --------------------
        if best_seq:
            a_res = resample_to_length(np.array(best_seq, dtype=float), n)
            starters.append(a_res)

        # ---- 1c. a few random Dirichlet draws ---------------------------
        for _ in range(4):
            rand_vec = rng.random(n)
            rand_vec /= rand_vec.sum()
            starters.append(rand_vec)

        # ---- 1d. optimisation of each starter -------------------------
        for idx, init in enumerate(starters, 1):
            if time.time() >= deadline - 5.0:
                break

            remaining = deadline - time.time()
            # allocate a modest slice of the remaining time to each start
            max_iter = int(min(150_000, remaining * 30))
            print(f"  Starter {idx}/{len(starters)} – max_iter={max_iter}")

            a_opt, fast_sc = eg_optimize(
                initial_a=init,
                time_deadline=deadline,
                max_iter=max_iter,
                eta0=1.0,
                rng=rng,
            )
            # official evaluation (slow) – only for the current best candidate
            val = evaluate_sequence(a_opt.tolist())
            print(f"    → official score = {val:.6f}")
            if val < best_val:
                best_val = val
                best_seq = a_opt.tolist()
                print(f"    *** New global best: {best_val:.6f} (n={n}) ***")

    # 2.  Final polishing of the overall best vector
    # --------------------------------------------------------------
    if best_seq and time.time() < deadline - 5.0:
        a = np.array(best_seq, dtype=float)
        remaining = deadline - time.time()
        max_iter = int(min(600_000, remaining * 40))
        print(f"\nFinal polishing – length {a.shape[0]}, max_iter={max_iter}")
        a_refined, _ = eg_optimize(
            initial_a=a,
            time_deadline=deadline,
            max_iter=max_iter,
            eta0=0.8,
            rng=rng,
        )
        final_val = evaluate_sequence(a_refined.tolist())
        print(f"Final polished score = {final_val:.6f}")
        if final_val < best_val:
            best_val = final_val
            best_seq = a_refined.tolist()
            print("*** Updated global best after final polish ***")

    # --------------------------------------------------------------
    # 3.  Safety net – guaranteed return
    # --------------------------------------------------------------
    if not best_seq:
        # Very crude fallback (uniform 256‑vector)
        best_seq = [1.0 / 256] * 256
        best_val = evaluate_sequence(best_seq)
        print(f"Safety fallback used, score = {best_val:.6f}")

    print(f"\nSearch completed. Best reported score = {best_val:.6f}")
    return best_seq
