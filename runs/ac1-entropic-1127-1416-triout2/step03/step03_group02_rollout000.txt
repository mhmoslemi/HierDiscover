"""
Hybrid up‑sampling + soft‑max exponentiated‑gradient optimisation.

* start from a small, well‑behaved family (uniform, arcsine, power‑law,
  beta‑like, Dirichlet).
* keep the few best candidates (fast FFT‑based score) and run a two‑phase
  exponentiated‑gradient descent that gradually raises the soft‑max
  temperature λ.  The update is multiplicative, guarantees non‑negativity
  and the simplex constraint (∑a = 1) is enforced by normalisation.
* after a length n is processed, the current best vector is “upsampled”
  (each entry is duplicated) to length 2·n and refined again – this
  exploits the fact that a finer discretisation can lower the maximal
  auto‑convolution.
* a final polishing stage with a very large λ (≈ 400) cleans the
  remaining irregularities.
* the routine respects the supplied time budget (default 1000 s) and
  always returns a feasible list; intermediate progress is printed.
"""

import time
import numpy as np
from typing import List, Tuple

# ----------------------------------------------------------------------
# Low‑level numeric utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Linear convolution via real FFT (full length, zero‑padded to a power‑of‑2)."""
    m = len(a) + len(b) - 1
    n_fft = next_pow2(m)
    A = np.fft.rfft(a, n=n_fft)
    B = np.fft.rfft(b, n=n_fft)
    C = A * B
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


def compute_score(a: np.ndarray) -> float:
    """
    “Official” objective (scale‑invariant):
        2·n·max(a*a) / (sum(a)²)
    """
    n = len(a)
    conv = convolution_fft(a, a)          # length = 2·n‑1
    max_val = float(np.max(conv))
    s = float(np.sum(a))
    if s < 1e-12:
        return float("inf")
    return 2.0 * n * max_val / (s * s)


def arcsine_initial(n: int) -> np.ndarray:
    """Discrete arcsine density (mid‑point rule)."""
    i = np.arange(n, dtype=float)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - 0.5 - i) + 1e-12)
    return vals / vals.sum()


def generate_shape(n: int, p: float, eps: float) -> np.ndarray:
    """
    Symmetric power‑law shape a_i ∝ ((i+ε)(n‑1‑i+ε))⁻ᵖ.
    """
    i = np.arange(n, dtype=float)
    base = ((i + eps) * (n - 1 - i + eps)) ** (-p)
    total = base.sum()
    if total == 0.0:
        return np.full(n, 1.0 / n)
    return base / total


def softmax_weights(conv: np.ndarray, lam: float) -> np.ndarray:
    """
    Numerically stable soft‑max of the convolution entries.
    Returns a probability vector of the same length as conv.
    """
    x = lam * conv
    mx = np.max(x)
    w = np.exp(x - mx)
    w_sum = w.sum()
    if w_sum == 0.0:
        return np.full_like(w, 1.0 / w.size)
    w /= w_sum
    return w


def gradient_from_weights(a: np.ndarray, w: np.ndarray) -> np.ndarray:
    """
    Gradient of Σ_k w_k·(a*a)_k with respect to a.
    Implements   g_j = 2· Σ_m a_m·w_{m+j}.
    Computed via a single FFT‑based convolution.
    """
    # reverse w because convolution implements Σ_i a_i·w_{j-i}
    w_rev = w[::-1]
    conv = convolution_fft(a, w_rev)          # length = 3·n‑2
    idx = 2 * len(a) - 2                       # point that aligns with zero shift
    grads = 2.0 * conv[idx - np.arange(len(a))]
    return grads


def normalize(v: np.ndarray) -> np.ndarray:
    """Project a non‑negative vector onto the probability simplex (sum = 1)."""
    total = v.sum()
    if total <= 0.0:
        return np.full_like(v, 1.0 / len(v))
    return v / total


# ----------------------------------------------------------------------
# Core optimiser (exponentiated‑gradient on a soft‑max approximation)
# ----------------------------------------------------------------------


def optimise_sequence(
    initial_a: np.ndarray,
    deadline: float,
    *,
    lam_start: float = 5.0,
    lam_end: float = 120.0,
    lam_ramp_iters: int = 4000,
    eta0: float = 0.6,
    max_iter: int = 200_000,
    noise_every: int = 0,
    noise_scale: float = 0.0,
) -> Tuple[np.ndarray, float]:
    """
    Runs a multiplicative (exponentiated) gradient descent on the
    smooth approximation of the max‑convolution objective.

    Parameters
    ----------
    initial_a : np.ndarray
        Starting vector (will be normalised to the simplex).
    deadline : float
        Absolute time (as returned by ``time.time()``) after which the
        optimisation must stop.
    lam_start, lam_end : float
        Temperature of the soft‑max; it is increased linearly from
        ``lam_start`` to ``lam_end`` over ``lam_ramp_iters`` steps.
    eta0 : float
        Initial step‑size; the schedule is η_t = η₀ / √t.
    max_iter : int
        Hard limit on the number of iterations.
    noise_every, noise_scale : int, float
        Optional Gaussian perturbation to escape plateaus.

    Returns
    -------
    best_a : np.ndarray
        The best vector seen (normalised to sum=1).
    best_score : float
        Its (non‑smooth) score as computed by ``compute_score``.
    """
    a = normalize(initial_a.astype(float).copy())
    best_a = a.copy()
    best_score = compute_score(a)

    n = len(a)

    for it in range(1, max_iter + 1):
        if time.time() >= deadline:
            break

        # ---- temperature schedule ----
        if it <= lam_ramp_iters:
            lam = lam_start + (lam_end - lam_start) * (it / lam_ramp_iters)
        else:
            lam = lam_end

        # ---- current convolution & soft‑max weights ----
        conv = convolution_fft(a, a)          # length = 2·n‑1
        w = softmax_weights(conv, lam)

        # ---- gradient of the soft‑max weighted sum ----
        grad = gradient_from_weights(a, w)

        # ---- step size (√t decay) ----
        eta = eta0 / np.sqrt(it)

        # ---- exponentiated update (keeps non‑negativity) ----
        a = a * np.exp(-eta * grad)

        # ---- optional exploration noise ----
        if noise_every and (it % noise_every == 0):
            a += np.random.randn(n) * noise_scale
            a = np.maximum(a, 0.0)

        # ---- re‑project onto the simplex ----
        a = normalize(a)

        # ---- keep the best observed vector ----
        cur_score = compute_score(a)
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()

        # ---- occasional progress report ----
        if it % 2000 == 0:
            print(
                f"        iter={it:,d}  λ={lam:.1f}  cur={cur_score:.6f}  best={best_score:.6f}"
            )

    return best_a, best_score


# ----------------------------------------------------------------------
# Main entry point required by the competition harness
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs) -> List[float]:
    """
    Search for a non‑negative step‑function with a low ``evaluate_sequence``
    value.  The routine respects the given time budget, explores several
    lengths, refines promising candidates with the optimiser above and
    finally returns the best sequence found.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    # keep a small safety margin for the final ``return``
    deadline = start_time + budget_s - 2.0

    best_seq: List[float] = []
    best_val: float = float("inf")

    # ------------------------------------------------------------------
    # 0.  Fallback to a previously discovered construction (if available)
    # ------------------------------------------------------------------
    try:
        fallback = [float(x) for x in height_sequence_1]  # type: ignore
        val = evaluate_sequence(fallback)
        print(f"Fallback (height_sequence_1) score = {val:.6f}")
        if val < best_val:
            best_val, best_seq = val, fallback
    except Exception:
        pass

    # ------------------------------------------------------------------
    # 1. Exploration over increasing lengths (powers of two)
    # ------------------------------------------------------------------
    lengths = [512, 1024, 2048, 4096]

    for n in lengths:
        if time.time() >= deadline - 5.0:
            break
        print(f"\n=== Exploring length n = {n} ===")

        # --------------------------------------------------------------
        # 1a. Build a modest library of start vectors.
        # --------------------------------------------------------------
        candidates: List[np.ndarray] = []

        # uniform
        candidates.append(np.full(n, 1.0 / n))

        # arcsine
        candidates.append(arcsine_initial(n))

        # power‑law family
        for _ in range(5):
            p = rng.uniform(0.2, 0.9)
            eps = rng.uniform(0.001, max(2.0, 0.02 * n))
            candidates.append(generate_shape(n, p, eps))

        # beta‑like shapes
        for _ in range(3):
            alpha = rng.uniform(0.2, 5.0)
            beta = rng.uniform(0.2, 5.0)
            i = np.arange(n, dtype=float)
            base = (i + 0.5) ** (alpha - 1) * (n - 1 - i + 0.5) ** (beta - 1)
            total = base.sum()
            if total == 0.0:
                base = np.full(n, 1.0 / n)
            else:
                base = base / total
            candidates.append(base)

        # Dirichlet / exponential samples
        for _ in range(2):
            samp = rng.exponential(1.0, size=n)
            samp /= samp.sum()
            candidates.append(samp)

        # --------------------------------------------------------------
        # 1b. Fast ranking using the FFT‑based score, keep the 3 best.
        # --------------------------------------------------------------
        scored = [(compute_score(c), c) for c in candidates]
        scored.sort(key=lambda x: x[0])
        top_candidates = [c for _, c in scored[:3]]

        # --------------------------------------------------------------
        # 1c. Refine each retained candidate.
        # --------------------------------------------------------------
        for idx, cand in enumerate(top_candidates):
            if time.time() >= deadline - 5.0:
                break
            print(f"  candidate {idx + 1}/{len(top_candidates)} (length {n})")

            # allocate iterations according to remaining time
            remaining = max(0.0, deadline - time.time())
            # about 12–15 iterations per second for these sizes
            max_iter = int(min(30_000, max(2_000, remaining * 12)))

            refined, _ = optimise_sequence(
                initial_a=cand,
                deadline=deadline,
                lam_start=5.0,
                lam_end=180.0,
                lam_ramp_iters=int(max_iter * 0.6),
                eta0=0.6,
                max_iter=max_iter,
                noise_every=800,
                noise_scale=8e-6,
            )

            # official evaluation (slow but exact)
            val = evaluate_sequence(refined.tolist())
            print(f"    evaluated score = {val:.6f}")

            if val < best_val:
                best_val, best_seq = val, refined.tolist()
                print(f"    *** New global best: {best_val:.6f} (n={n}) ***")

        # --------------------------------------------------------------
        # 1d. Upsample the current best of this length (if any) to 2·n.
        # --------------------------------------------------------------
        if best_seq and len(best_seq) == n and time.time() < deadline - 5.0:
            print(f"\nUpsampling best sequence of length {n} to {2 * n}")
            a = np.array(best_seq, dtype=float)

            # simple duplication (each entry appears twice)
            a_up = np.repeat(a, 2)
            a_up = normalize(a_up + rng.normal(scale=1e-7, size=a_up.shape))

            # refine the upsampled vector
            remaining = max(0.0, deadline - time.time())
            max_iter = int(min(35_000, max(3_000, remaining * 12)))

            refined_up, _ = optimise_sequence(
                initial_a=a_up,
                deadline=deadline,
                lam_start=5.0,
                lam_end=250.0,
                lam_ramp_iters=int(max_iter * 0.6),
                eta0=0.55,
                max_iter=max_iter,
                noise_every=1000,
                noise_scale=5e-6,
            )
            val_up = evaluate_sequence(refined_up.tolist())
            print(f"    upsampled length {2 * n} score = {val_up:.6f}")
            if val_up < best_val:
                best_val, best_seq = val_up, refined_up.tolist()
                print(f"    *** New global best after up‑sampling: {best_val:.6f} ***")

    # ------------------------------------------------------------------
    # 2. Final polishing of the overall best sequence (if time permits).
    # ------------------------------------------------------------------
    if best_seq and time.time() < deadline - 5.0:
        a = np.array(best_seq, dtype=float)
        remaining = max(0.0, deadline - time.time())
        max_iter = int(min(80_000, max(5_000, remaining * 12)))

        print("\nFinal polishing of the global best")
        polished, _ = optimise_sequence(
            initial_a=a,
            deadline=deadline,
            lam_start=5.0,
            lam_end=400.0,
            lam_ramp_iters=int(max_iter * 0.7),
            eta0=0.45,
            max_iter=max_iter,
            noise_every=1200,
            noise_scale=3e-6,
        )
        final_val = evaluate_sequence(polished.tolist())
        print(f"Final polished score = {final_val:.6f}")
        if final_val < best_val:
            best_val, best_seq = final_val, polished.tolist()
            print("*** Updated global best after final polishing ***")

    # ------------------------------------------------------------------
    # 3. Guarantee a feasible return.
    # ------------------------------------------------------------------
    if not best_seq:
        best_seq = [1.0] * 256
        best_val = evaluate_sequence(best_seq)

    print(f"\nSearch finished. Best reported score = {best_val:.6f}")
    return best_seq
