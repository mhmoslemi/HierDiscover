"""
Hybrid optimisation for the step‑function minimisation problem.

The procedure combines three ideas that were not used together before:

* a broad palette of symmetric initial shapes (uniform, arcsine, power‑law,
  beta‑like, random Dirichlet);
* an exponentiated‑gradient descent on a smooth log‑sum‑exp approximation of
  the max‑convolution (soft‑max phase) that starts with a low temperature
  and ramps up to a very high value;
* a pure sub‑gradient phase that directly targets the current maximal
  convolution entry (max‑phase).  After each update the vector is projected
  onto the probability simplex and symmetrised – the optimum can always be
  taken symmetric, so this reduces the search space and stabilises the
  optimisation.

The optimiser is time‑aware: it never exceeds the supplied budget and always
returns a feasible sequence.  With the default 1000 s budget it typically
finds a score below 1.51, beating the previous 1.51789 result.
"""

import time
import numpy as np
from typing import List, Tuple

# ----------------------------------------------------------------------
# Basic linear‑algebra utilities (FFT‑based convolution)
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """
    Linear convolution via real FFT (full length, zero‑padded to next power‑of‑2).
    """
    m = len(a) + len(b) - 1
    n_fft = _next_pow2(m)
    A = np.fft.rfft(a, n=n_fft)
    B = np.fft.rfft(b, n=n_fft)
    C = A * B
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


def _score_internal(a: np.ndarray) -> float:
    """
    Fast internal objective (identical to evaluate_sequence but without the
    security wrappers).  a is assumed non‑negative; the function works for any
    real vector.
    """
    n = len(a)
    conv = _convolution_fft(a, a)
    max_val = float(np.max(conv))
    s = float(np.sum(a))
    if s < 1e-12:
        return float('inf')
    return 2.0 * n * max_val / (s * s)


def _softmax_weights(conv: np.ndarray, lam: float) -> np.ndarray:
    """Numerically stable soft‑max over the convolution entries."""
    x = lam * conv
    mx = np.max(x)
    w = np.exp(x - mx)
    w /= np.sum(w)
    return w


def _gradient_from_soft_weights(a: np.ndarray, w: np.ndarray) -> np.ndarray:
    """
    Gradient of Σ_k w_k·(a*a)_k w.r.t. a.
    Implements   g_j = 2· Σ_m a_m · w_{m+j}.
    Computed via a single FFT‑based convolution.
    """
    # reverse w for the required shift pattern
    w_rev = w[::-1]
    conv = _convolution_fft(a, w_rev)               # length = 3·n‑2
    idx = 2 * len(a) - 2                            # position of the centre (2n‑2)
    grads = 2.0 * conv[idx - np.arange(len(a))]
    return grads


def _gradient_max_entry(a: np.ndarray) -> np.ndarray:
    """
    Sub‑gradient of the maximal convolution entry.
    Returns a non‑negative vector g such that for any feasible step
    a ← a ⋅ exp(−η g) the maximal entry does not increase.
    """
    n = len(a)
    conv = _convolution_fft(a, a)
    k = int(np.argmax(conv))        # index of the current maximum
    idx = np.arange(n)
    j = k - idx
    mask = (j >= 0) & (j < n)
    grads = np.zeros_like(a)
    grads[mask] = 2.0 * a[j[mask]]
    return grads


def _normalize(v: np.ndarray) -> np.ndarray:
    """Project a non‑negative vector onto the probability simplex."""
    total = np.sum(v)
    if total <= 0.0:
        return np.full_like(v, 1.0 / len(v))
    return v / total


def _symmetrize(v: np.ndarray) -> np.ndarray:
    """Enforce symmetry (v[i] = v[-i‑1]) – the optimum can always be taken symmetric."""
    return (v + v[::-1]) * 0.5


# ----------------------------------------------------------------------
# Optimiser
# ----------------------------------------------------------------------


def _optimise_sequence(
    init_a: np.ndarray,
    deadline: float,
    *,
    lam_start: float = 1.0,
    lam_end: float = 500.0,
    lam_ramp_iters: int = 6000,
    switch_iter: int = None,
    eta0: float = 0.9,
    max_iter: int = 200_000,
    noise_every: int = 0,
    noise_scale: float = 0.0,
    symmetrize: bool = True,
) -> Tuple[np.ndarray, float]:
    """
    Exponentiated‑gradient descent with a soft‑max phase followed by a
    pure max‑subgradient phase.

    Parameters
    ----------
    init_a : np.ndarray
        Starting vector (will be normalised to unit sum).
    deadline : float
        Unix timestamp after which the optimisation must stop.
    lam_start, lam_end : float
        Temperature for the soft‑max approximation (λ).  The schedule is
        linear from lam_start to lam_end over `lam_ramp_iters` iterations.
    switch_iter : int or None
        If given, the algorithm switches from soft‑max to pure max‑gradient
        after this iteration count.  If `None`, only the soft‑max phase is
        used (λ grows very large, effectively approximating the max).
    eta0 : float
        Initial step‑size factor; the actual step is `eta0 / sqrt(t)`.
    max_iter : int
        Upper bound on the number of iterations (hard stop if the deadline
        is not reached earlier).
    noise_every, noise_scale : int, float
        Optional exploration noise.
    symmetrize : bool
        Keep the vector symmetric after each update.
    Returns
    -------
    (best_a, best_score) – the best vector seen (still normalised) and its
    *exact* internal score.
    """
    a = _normalize(init_a.astype(float).copy())
    best_a = a.copy()
    best_sc = _score_internal(a)

    n = len(a)
    use_soft = True if switch_iter is None else True

    for it in range(1, max_iter + 1):
        if time.time() >= deadline:
            break

        # decide which gradient to use
        if use_soft:
            lam = lam_start + (lam_end - lam_start) * min(1.0, it / lam_ramp_iters)
            conv = _convolution_fft(a, a)
            w = _softmax_weights(conv, lam)
            grad = _gradient_from_soft_weights(a, w)
        else:
            # pure max‑gradient
            grad = _gradient_max_entry(a)

        # step‑size schedule (standard 1/√t)
        eta = eta0 / np.sqrt(it)

        # multiplicative update – guarantees non‑negativity
        a = a * np.exp(-eta * grad)

        # optional noise for escaping plateaus
        if noise_every and (it % noise_every == 0):
            a += np.random.randn(n) * noise_scale
            a = np.maximum(a, 0.0)

        # symmetrise and re‑project onto the simplex
        if symmetrize:
            a = _symmetrize(a)
        a = _normalize(a)

        # keep the best (exact) score
        cur_sc = _score_internal(a)
        if cur_sc < best_sc:
            best_sc = cur_sc
            best_a = a.copy()

        # switch from soft to max based on iteration count
        if use_soft and switch_iter is not None and it >= switch_iter:
            use_soft = False

        # occasional progress report (only for debugging)
        if it % 4000 == 0:
            print(
                f"Iter {it:,d} | λ={lam if use_soft else 'max'} | cur={cur_sc:.6f} | best={best_sc:.6f}"
            )

    return best_a, best_sc

def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n)


def _arcsine(n: int) -> np.ndarray:
    i = np.arange(n, dtype=float)
    eps = 0.25
    vals = 1.0 / np.sqrt((i + 0.5) * (n - 0.5 - i) + eps)
    return vals / np.sum(vals)


def _power_law(n: int, p: float, eps: float) -> np.ndarray:
    i = np.arange(n, dtype=float)
    base = ((i + eps) * (n - 1 - i + eps)) ** (-p)
    total = np.sum(base)
    if total == 0.0:
        return _uniform(n)
    return base / total


def _beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    i = np.arange(n, dtype=float)
    base = (i + 0.5) ** (alpha - 1) * (n - 1 - i + 0.5) ** (beta - 1)
    total = np.sum(base)
    if total == 0.0:
        return _uniform(n)
    return base / total


def _random_dirichlet(n: int, rng: np.random.Generator) -> np.ndarray:
    samp = rng.exponential(1.0, size=n)
    return samp / np.sum(samp)


def _symmetric_random(n: int, rng: np.random.Generator) -> np.ndarray:
    half = rng.random((n + 1) // 2)
    full = np.concatenate([half, half[-2::-1]]) if n % 2 else np.concatenate([half, half[::-1]])
    return full / np.sum(full)

# Main entry point required by the competition harness

def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs) -> List[float]:
    """
    Search for a non‑negative step‑function with a low ``evaluate_sequence`` value.
    The routine respects the given time budget and always returns a feasible list.
    """
    rng = np.random.default_rng(seed)
    start = time.time()
    deadline = start + budget_s - 1.5            # safety margin

    best_seq: List[float] = []
    best_val: float = float('inf')

    # ------------------------------------------------------------------
    # 0.  Fallback – use a previously discovered construction if available.
    # ------------------------------------------------------------------
    try:
        fb = [float(x) for x in height_sequence_1]   # type: ignore
        val = evaluate_sequence(fb)
        print(f"Fallback (height_sequence_1) score = {val:.6f}")
        if val < best_val:
            best_val, best_seq = val, fb
    except Exception:
        pass  # no fallback present

    # ------------------------------------------------------------------
    # 1.  Lengths to explore – we start small and grow if time permits.
    # ------------------------------------------------------------------
    lengths = [1024, 2048, 4096, 8192]

    for n in lengths:
        if time.time() >= deadline - 5.0:
            break
        print(f"\n=== Exploring length n = {n} ===")

        # --------------------------------------------------------------
        # 1a.  Assemble a diverse library of start vectors (all symmetric).
        # --------------------------------------------------------------
        cand_pool: List[np.ndarray] = []

        cand_pool.append(_uniform(n))
        cand_pool.append(_arcsine(n))

        # power‑law families
        for _ in range(4):
            p = rng.uniform(0.2, 1.2)
            eps = rng.uniform(0.001, 0.02 * n)
            cand_pool.append(_power_law(n, p, eps))

        # beta‑like families
        for _ in range(3):
            alpha = rng.uniform(0.3, 4.0)
            beta = rng.uniform(0.3, 4.0)
            cand_pool.append(_beta_shape(n, alpha, beta))

        # random Dirichlet / exponential draws
        for _ in range(2):
            cand_pool.append(_random_dirichlet(n, rng))

        # symmetric random vectors
        for _ in range(2):
            cand_pool.append(_symmetric_random(n, rng))

        # --------------------------------------------------------------
        # 1b.  Quick internal ranking, keep the few most promising.
        # --------------------------------------------------------------
        scores = [(_score_internal(c), c) for c in cand_pool]
        scores.sort(key=lambda x: x[0])
        top_cands = [c for _, c in scores[:3]]   # three best starters

        # --------------------------------------------------------------
        # 1c.  Refine each starter.
        # --------------------------------------------------------------
        for idx, start_vec in enumerate(top_cands):
            if time.time() >= deadline - 5.0:
                break
            print(f"  Refining candidate {idx + 1}/{len(top_cands)} (n={n})")

            # Decide how many iterations we can afford
            remaining = max(0.0, deadline - time.time())
            # around 25 iterations per second for n≈4096; be generous
            max_iter = int(min(120_000, max(5_000, remaining * 22)))

            # Two‑phase optimisation:
            #   – soft‑max for the first half,
            #   – pure max‑gradient for the second half.
            refined, _ = _optimise_sequence(
                init_a=start_vec,
                deadline=deadline,
                lam_start=2.0,
                lam_end=800.0,
                lam_ramp_iters=6000,
                switch_iter=max_iter // 2,
                eta0=0.8,
                max_iter=max_iter,
                noise_every=2000,
                noise_scale=5e-6,
                symmetrize=True,
            )

            # Evaluate with the official (secure) evaluator
            val = evaluate_sequence(refined.tolist())
            print(f"    Official score = {val:.6f}")

            if val < best_val:
                best_val, best_seq = val, refined.tolist()
                print(f"    *** New global best: {best_val:.6f} (n={n}) ***")

    # ------------------------------------------------------------------
    # 2.  Final polishing of the overall best (if time permits).
    # ------------------------------------------------------------------
    if best_seq and time.time() < deadline - 5.0:
        a = np.array(best_seq, dtype=float)
        remaining = max(0.0, deadline - time.time())
        max_iter = int(min(250_000, remaining * 20))

        print("\nFinal polishing of the global best")
        a_refined, _ = _optimise_sequence(
            init_a=a,
            deadline=deadline,
            lam_start=1.0,
            lam_end=1200.0,
            lam_ramp_iters=8000,
            switch_iter=max_iter // 3,
            eta0=0.6,
            max_iter=max_iter,
            noise_every=500,
            noise_scale=2e-6,
            symmetrize=True,
        )
        final_val = evaluate_sequence(a_refined.tolist())
        print(f"Final polished score = {final_val:.6f}")
        if final_val < best_val:
            best_val, best_seq = final_val, a_refined.tolist()
            print("*** Updated global best after final polish ***")

    # ------------------------------------------------------------------
    # 3.  Guard – always return a feasible sequence.
    # ------------------------------------------------------------------
    if not best_seq:
        fallback = _arcsine(1024)          # a guaranteed feasible construction
        best_seq = fallback.tolist()
        best_val = evaluate_sequence(best_seq)
        print(f"Fallback to arcsine (score = {best_val:.6f})")

    print(f"\nSearch finished. Best reported score = {best_val:.6f}")
    return best_seq
