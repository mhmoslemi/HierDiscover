"""
Improved search for a non‑negative step‑function that minimizes

    E(a) = 2·n·max(a∗a) / (sum(a)²)

The algorithm works in several stages (all within the given time budget).

1. **Parametric sweep** – many candidates from a Beta‑type family,
   an arcsine shape, uniform mixtures and simple exponentials are
   evaluated for a few moderate lengths (1024‑4096).  The best vector
   (normalised to sum = 1) is kept.

2. **Projected sub‑gradient descent on the true max‑objective** – the
   convex function `max(a∗a)` is minimised by moving opposite to a
   sub‑gradient that can be computed via a single convolution.  The
   iterate stays on the probability simplex.

3. **Alternating linear‑programming step** – for the current vector `a`
   we solve a linear programme that finds a non‑negative vector `g`
   maximising `∑ g` subject to the convolution bound
   `(a * g) ≤ max(a∗a)`.  After normalising `g` we line‑search a few
   mixing coefficients `θ ∈ (0,1)` and replace `a` by the best
   convex combination.  This is the block‑coordinate idea from the
   reference solution, but here it is performed a few times with a
   proper line‑search.

4. **Upsampling** – the best vector is linearly interpolated to a
   larger size (8192) which usually lowers the bound because the
   discretisation becomes finer.

5. **Final stochastic hill‑climb** – cheap pair‑wise mass‑transfer
   moves are applied, accepting only improvements of the true (unsmoothed)
   score.

All steps respect the time budget (`budget_s`).  The routine returns a
list of floats ready for `evaluate_sequence`.
"""

def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex (sum = 1, v_i ≥ 0)."""
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / v.shape[0])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho = np.where(u - cssv / np.arange(1, v.shape[0] + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full convolution using real FFT (fast)."""
    m = len(a) + len(b) - 1
    n_fft = next_pow2(m)
    A = np.fft.rfft(a, n=n_fft)
    B = np.fft.rfft(b, n=n_fft)
    C = A * B
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


def compute_score(a: np.ndarray) -> float:
    """Exact objective E(a) used by `evaluate_sequence` (up to scale invariance)."""
    a = np.asarray(a, dtype=float)
    n = a.shape[0]
    conv = convolution_fft(a, a)
    max_val = float(np.max(conv))
    s = float(np.sum(a))
    if s < 1e-12:
        return float("inf")
    return 2.0 * n * max_val / (s * s)

def arcsine_initial(n: int) -> np.ndarray:
    """Arcsine‑shaped start: a_i ∝ 1 / sqrt((i+0.5)(n‑0.5‑i))."""
    i = np.arange(n, dtype=float)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - 0.5 - i))
    return vals / np.sum(vals)


def beta_sequence(n: int, alpha: float, beta: float | None = None) -> np.ndarray:
    """Beta‑type coefficients a_i ∝ (i+½)^α·(n‑½‑i)^β."""
    if beta is None:
        beta = alpha
    i = np.arange(n, dtype=float)
    left = np.power(i + 0.5, alpha)
    right = np.power(n - 0.5 - i, beta)
    seq = left * right
    seq = np.clip(seq, 0.0, 1e12)
    total = np.sum(seq)
    if total == 0.0:
        seq = np.full(n, 1.0 / n)
    else:
        seq = seq / total
    return seq


def exponential_sequence(n: int, lam: float) -> np.ndarray:
    """a_i ∝ exp(-lam·|i - (n‑1)/2|)."""
    i = np.arange(n, dtype=float)
    center = (n - 1) / 2.0
    seq = np.exp(-lam * np.abs(i - center))
    return seq / np.sum(seq)


def upsample_sequence(seq: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation of a probability vector to a larger size."""
    old_n = len(seq)
    old_x = (np.arange(old_n) + 0.5) / old_n
    new_x = (np.arange(new_n) + 0.5) / new_n
    new_vals = np.interp(new_x, old_x, seq)
    return new_vals / np.sum(new_vals)


# --------------------------------------------------------------
# Projected sub‑gradient descent on max(a∗a)
# --------------------------------------------------------------

def subgradient_descent(a0: np.ndarray,
                        deadline: float,
                        max_iter: int = 15000,
                        base_eta: float = 0.12,
                        tol: float = 1e-12) -> tuple[list[float], float]:
    """
    Sub‑gradient descent on the convex function f(a)=max(a∗a)
    while staying on the probability simplex.
    Returns the best vector seen (as a list) and its score.
    """
    a = project_to_simplex(a0.copy())
    best_a = a.copy()
    best_score = compute_score(a)
    n = len(a)

    a_rev = a[::-1]                     # will be updated each iteration
    eps = 1e-14

    for t in range(1, max_iter + 1):
        if time.time() >= deadline:
            break

        # full self‑convolution and its maximum
        conv = convolution_fft(a, a)
        M = float(np.max(conv))

        # binary mask of active (max‑tight) indices
        active = (conv >= M - tol).astype(float)

        # sub‑gradient: g = 2 * Aᵀ·active, where A is the Toeplitz matrix of a.
        # Using convolution: g = 2 * (active * a_rev)[n-1 : n-1+n]
        conv_active = convolution_fft(active, a[::-1])
        subgrad = 2.0 * conv_active[n - 1:n - 1 + n]

        # step size schedule (inverse‑sqrt)
        eta = base_eta / math.sqrt(t)

        a_candidate = a - eta * subgrad
        a_candidate = project_to_simplex(a_candidate)

        cur_score = compute_score(a_candidate)
        if cur_score < best_score - 1e-12:
            best_score = cur_score
            best_a = a_candidate.copy()

        a = a_candidate

        # occasional progress report
        if t % 3000 == 0:
            print(f"    sub‑grad iter {t:,d}  curScore {cur_score:.6f}  best {best_score:.6f}")

    return best_a.tolist(), best_score


# --------------------------------------------------------------
# Alternating linear‑programming refinement
# --------------------------------------------------------------

def alternating_lp(a0: np.ndarray,
                   deadline: float,
                   outer_iters: int = 7,
                   theta_candidates: tuple[float, ...] = (0.2, 0.4, 0.6, 0.8)) -> tuple[list[float], float]:
    """
    Block‑coordinate step: given a, solve a linear programme for g
    with (a * g) ≤ max(a∗a) and maximise ∑ g.
    Afterwards perform a small line‑search over mixing coefficients.
    Repeats a few times.
    Returns the best vector found and its score.
    """
    a = a0.copy()
    n = len(a)
    best_score = compute_score(a)
    best_a = a.copy()

    for outer in range(outer_iters):
        if time.time() >= deadline:
            break

        # current bound M
        conv = convolution_fft(a, a)
        M = float(np.max(conv))

        # Build the Toeplitz matrix A (size (2n‑1) × n) with rows k,
        # columns i, entry a_{k-i} (zero when out of range).
        # Using sparse diagonal representation.
        diagonals = [a[d] * np.ones(n - d) for d in range(n)]
        offsets = list(range(n))
        A = sp.diags(diagonals, offsets, shape=(2 * n - 1, n), format='csr')

        # Linear programme: maximise sum(g)  <=> minimise -sum(g)
        c = -np.ones(n)
        bounds = [(0, None)] * n
        b_ub = np.full(2 * n - 1, M)

        res = opt.linprog(c,
                          A_ub=A,
                          b_ub=b_ub,
                          bounds=bounds,
                          method='highs',
                          options={'presolve': True})
        if res.status != 0:
            # LP failed – abort the alternating phase.
            print("    LP failed (status {}), aborting LP phase".format(res.status))
            break

        g = res.x
        sum_g = np.sum(g)
        if sum_g == 0.0:
            break
        g_norm = g / sum_g  # now ∑ g_norm = 1

        # try a few mixing coefficients θ and keep the best
        best_local_a = a
        best_local_score = best_score
        for theta in theta_candidates:
            cand = (1.0 - theta) * a + theta * g_norm
            cand = project_to_simplex(cand)   # numerical safety
            cand_score = compute_score(cand)
            if cand_score < best_local_score:
                best_local_score = cand_score
                best_local_a = cand

        # if no improvement, stop early
        if best_local_score >= best_score - 1e-12:
            print(f"    LP outer iter {outer+1}: no improvement, stopping")
            break

        # accept improvement
        a = best_local_a
        best_score = best_local_score
        best_a = a.copy()
        print(f"    LP outer iter {outer+1}: new score {best_score:.6f}")

    return best_a.tolist(), best_score


# --------------------------------------------------------------
# Simple stochastic hill‑climbing (pairwise mass transfer)
# --------------------------------------------------------------

def hill_climb(seq,
               time_deadline: float,
               max_steps: int = 40000,
               init_step: float = 0.07) -> tuple[list[float], float]:
    """
    Cheap pair‑wise mass‑transfer hill‑climb.
    Only moves that improve the true (unsmoothed) score are kept.
    """
    a = np.array(seq, dtype=float)
    a = project_to_simplex(a)
    best_a = a.copy()
    best_score = compute_score(best_a)
    n = len(a)
    step = init_step

    while max_steps > 0 and time.time() < time_deadline:
        i, j = np.random.randint(n, size=2)
        if i == j:
            continue
        delta = step * best_a[i]
        if delta <= 0.0:
            continue
        cand = best_a.copy()
        cand[i] -= delta
        cand[j] += delta
        cand = project_to_simplex(cand)
        cur_score = compute_score(cand)
        if cur_score < best_score:
            best_score = cur_score
            best_a = cand
            step = min(step * 1.07, 0.5)
        else:
            step *= 0.96
        max_steps -= 1

    return best_a.tolist(), best_score


# --------------------------------------------------------------
# Main entry point
# --------------------------------------------------------------

def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a step‑function that yields a low `evaluate_sequence`
    value.  The routine respects the time budget, combines a parametric
    sweep, projected sub‑gradient descent, an alternating LP phase,
    optional up‑sampling and a final hill‑climb.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 2.0   # tiny safety margin

    best_seq = None
    best_val = float("inf")

    # ----------------------------------------------------------
    # Optional fallback to a previously discovered construction.
    # ----------------------------------------------------------
    try:
        fallback = [float(x) for x in height_sequence_1]   # may be undefined
        fallback_val = evaluate_sequence(fallback)
        print(f"Fallback score = {fallback_val:.6f}")
        if fallback_val < best_val:
            best_val = fallback_val
            best_seq = fallback
    except Exception:
        pass

    # ----------------------------------------------------------
    # Phase 1 – broad parametric sweep (small to medium n)
    # ----------------------------------------------------------
    lengths = [1024, 2048, 4096]
    for n in lengths:
        if time.time() >= deadline - 5:
            break
        print(f"\n=== Exploring length n = {n} ===")

        # ----- symmetric Beta family -----
        alphas = np.linspace(-0.95, 2.5, 120)
        for a in alphas:
            if time.time() >= deadline - 5:
                break
            seq = beta_sequence(n, a)
            score = compute_score(seq)
            if score < best_val:
                best_val = score
                best_seq = seq.tolist()
                print(f"New best (sym β) n={n} α={a:.3f}  score={score:.6f}")

        # ----- arcsine / uniform mixture -----
        arcs = arcsine_initial(n)
        uniform = np.full(n, 1.0 / n)
        for w in np.linspace(0.0, 1.0, 23):
            if time.time() >= deadline - 5:
                break
            seq = (1.0 - w) * arcs + w * uniform
            seq = seq / np.sum(seq)
            score = compute_score(seq)
            if score < best_val:
                best_val = score
                best_seq = seq.tolist()
                print(f"New best (mix) n={n} w={w:.2f}  score={score:.6f}")

        # ----- simple exponential decay -----
        for lam in np.linspace(0.001, 0.025, 16):
            if time.time() >= deadline - 5:
                break
            seq = exponential_sequence(n, lam)
            score = compute_score(seq)
            if score < best_val:
                best_val = score
                best_seq = seq.tolist()
                print(f"New best (exp) n={n} λ={lam:.4f}  score={score:.6f}")

        # ----- random asymmetric Beta pairs -----
        for _ in range(200):
            if time.time() >= deadline - 5:
                break
            a = np.random.uniform(-0.95, 2.5)
            b = np.random.uniform(-0.95, 2.5)
            seq = beta_sequence(n, a, b)
            score = compute_score(seq)
            if score < best_val:
                best_val = score
                best_seq = seq.tolist()
                print(f"New best (asym) n={n} a={a:.3f} b={b:.3f}  score={score:.6f}")

    # ----------------------------------------------------------
    # Phase 2 – projected sub‑gradient descent on the max‑convolution
    # ----------------------------------------------------------
    if best_seq is not None and time.time() < deadline - 8:
        print("\nStarting projected sub‑gradient descent")
        init_a = np.array(best_seq, dtype=float)
        refined_seq, refined_score = subgradient_descent(
            init_a,
            time_deadline=deadline,
            max_iter=15000,
            base_eta=0.12
        )
        if refined_score < best_val:
            best_val = refined_score
            best_seq = refined_seq
            print(f"Sub‑gradient descent best score = {refined_score:.6f}")

    # ----------------------------------------------------------
    # Phase 3 – alternating LP refinement (few outer iterations)
    # ----------------------------------------------------------
    if best_seq is not None and time.time() < deadline - 6:
        print("\nRunning alternating LP refinement")
        a_arr = np.array(best_seq, dtype=float)
        refined_seq, refined_score = alternating_lp(
            a_arr,
            time_deadline=deadline,
            outer_iters=7,
            theta_candidates=(0.2, 0.4, 0.6, 0.8)
        )
        if refined_score < best_val:
            best_val = refined_score
            best_seq = refined_seq
            print(f"LP refinement best score = {refined_score:.6f}")

    # ----------------------------------------------------------
    # Phase 4 – up‑sampling to a larger size (if time permits)
    # ----------------------------------------------------------
    large_n = 8192
    if best_seq is not None and time.time() < deadline - 5:
        print(f"\nUpsampling best candidate to n = {large_n}")
        up_seq = upsample_sequence(np.array(best_seq, dtype=float), large_n)
        up_score = compute_score(up_seq)
        print(f"Upsampled score = {up_score:.6f}")
        if up_score < best_val:
            best_val = up_score
            best_seq = up_seq.tolist()

    # ----------------------------------------------------------
    # Phase 5 – final cheap stochastic hill‑climb
    # ----------------------------------------------------------
    if best_seq is not None and time.time() < deadline - 2:
        print("\nFinal hill‑climb refinement")
        refined_seq, refined_val = hill_climb(
            best_seq,
            time_deadline=deadline,
            max_steps=40000,
            init_step=0.05,
        )
        print(f"Hill‑climb score = {refined_val:.6f}")
        if refined_val < best_val:
            best_val = refined_val
            best_seq = refined_seq

    # ----------------------------------------------------------
    # Fallback in the (unlikely) case no candidate was built.
    # ----------------------------------------------------------
    if best_seq is None:
        best_seq = [1.0] * 256
        best_val = evaluate_sequence(best_seq)

    final_score = evaluate_sequence(best_seq)
    print(f"\nSearch finished. Final reported score = {final_score:.6f}")
    return best_seq
