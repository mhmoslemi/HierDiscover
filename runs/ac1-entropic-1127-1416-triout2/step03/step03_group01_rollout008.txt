"""
Propose a low‑discrepancy step function.

Algorithm
---------
1. **Exploratory sampling** –  For a chosen length `n` we generate many
   shapes from a very flexible two‑parameter beta family, plus the classic
   arcsine shape.  The best candidate (according to a very fast surrogate
   ``2·n·max(a*a)``) is kept.

2. **Iterative LP refinement** –  Starting from that candidate we run the
   classic extremal‑set procedure described in the statement.  For the
   current vector `f` we compute `M = max(f*f)`.  The indices where the
   self‑convolution is close to `M` form an *active set*.  We solve the
   linear program

        max   Σ g_j
        s.t.  (f * g)_k ≤ M   for k in the active set
              g_j ≥ 0

   (a small LP – only a few dozen constraints).  The optimal `g` is
   rescaled to a probability vector and blended with `f`:
        f ← (1‑t)·f + t·g ,   t decreasing.

   This step provably never increases the objective and quickly drives
   the bound down.

3. **Sub‑gradient Adam polishing** –  After the LP phase we fine‑tune the
   vector using an Adam optimiser on the *exact* objective
        J(a) = 2·n·max(a*a)   (the sum of `a` is kept equal to 1).
   Because the sum is fixed, the gradient of `max(a*a)` is simply the
   (convex) sub‑gradient obtained from the active indices of the
   convolution.  Adam’s adaptive step‑size and projection onto the
   probability simplex make the optimisation stable.

4. **Multiple lengths** –  The whole pipeline is executed for a few
   lengths (2048, 4096, …).  The best solution (according to the official
   ``evaluate_sequence``) is returned.

The code respects the 1000 s budget, prints progress, and never returns
an empty list.  All helper functions are top‑level as required.
"""

import time
import numpy as np
from typing import List, Tuple

# ----------------------------------------------------------------------
# Low‑level utilities (no closures)
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """
    Real‑FFT convolution of two 1‑D arrays.
    Returns the linear convolution (length = len(a)+len(b)-1).
    """
    m = len(a) + len(b) - 1
    n_fft = next_pow2(m)
    A = np.fft.rfft(a, n=n_fft)
    B = np.fft.rfft(b, n=n_fft)
    C = A * B
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection onto the probability simplex {x ≥ 0, Σx = 1}.
    Duchi et al., 2008.
    """
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / v.size)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho = np.where(u - cssv / np.arange(1, v.size + 1) > 0)[0]
    if len(rho) == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def compute_score(a: np.ndarray) -> float:
    """
    Very fast surrogate for the official score.
    The vector ``a`` is assumed to be normalized to sum = 1,
    therefore the objective reduces to 2·n·max(a*a).
    """
    n = len(a)
    conv = convolution_fft(a, a)
    max_val = float(np.max(conv))
    return 2.0 * n * max_val


def arcsine_initial(n: int) -> np.ndarray:
    """Classic arcsine shape: a_i ∝ 1/√((i+½)*(n‑½‑i))."""
    i = np.arange(n, dtype=float)
    eps = 0.25
    vals = 1.0 / np.sqrt((i + 0.5) * (n - 0.5 - i) + eps)
    vals /= vals.sum()
    return vals


def generate_beta_shape(n: int, alpha: float, beta: float, eps: float) -> np.ndarray:
    """
    Symmetric beta‑family density on {0,…,n‑1}:
        a_i ∝ (i+ε)^{α‑1}·(n‑1‑i+ε)^{β‑1}.
    Returned vector is normalized to Σa = 1.
    """
    i = np.arange(n, dtype=float)
    left = (i + eps) ** (alpha - 1.0)
    right = (n - 1 - i + eps) ** (beta - 1.0)
    base = left * right
    total = base.sum()
    if total <= 0.0 or not np.isfinite(total):
        return np.full(n, 1.0 / n)
    return base / total

# Sub‑gradient utilities (exact max, not smoothed)
# ----------------------------------------------------------------------


def subgradient_and_max(a: np.ndarray, eps: float = 1e-12) -> Tuple[np.ndarray, float]:
    """
    Returns the sub‑gradient of max(a*a) and the maximal value.
    The sub‑gradient is the sum of the gradients of all active convolution
    entries (those within ``eps`` of the maximum).
    """
    n = len(a)
    conv = convolution_fft(a, a)        # length = 2n‑1
    max_val = float(np.max(conv))
    # active set (binary mask)
    mask = conv >= max_val - eps * max_val
    w = mask.astype(float)                        # shape (2n‑1,)
    # gradient of Σ_k w_k (a*a)_k  w.r.t. a  = 2 * (a * w_rev) evaluated properly
    L = 2 * n - 1
    w_rev = w[::-1]
    conv_rev = convolution_fft(a, w_rev)           # length = 3n‑2
    grad = 2.0 * conv_rev[L - 1 - np.arange(n)]    # shape (n,)
    return grad, max_val

                          eps_subgrad: float = 1e-12) -> Tuple[np.ndarray, float]:
    """
    Adam optimiser for the exact objective  J(a)=2·n·max(a*a)
    (the sum of ``a`` is kept equal to 1 by Euclidean projection onto
    the probability simplex after each step).
    Returns the best vector seen and its fast surrogate score.
    """
    a = project_to_simplex(initial_a.astype(float).copy())
    best_a = a.copy()
    best_sc = compute_score(a)

    beta1, beta2 = 0.9, 0.999
    eps = 1e-8
    m = np.zeros_like(a)
    v = np.zeros_like(a)

    for it in range(1, max_iter + 1):
        if time.time() >= time_deadline:
            break

        grad, _ = subgradient_and_max(a, eps=eps_subgrad)

        # Adam moments
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)

        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        step = lr0 * m_hat / (np.sqrt(v_hat) + eps)
        a_new = a - step
        a_new = project_to_simplex(a_new)

        cur_sc = compute_score(a_new)
        if cur_sc < best_sc:
            best_sc = cur_sc
            best_a = a_new.copy()

        a = a_new

        if it % 5000 == 0:
            print(f"    Adam‑subgrad iter={it:,d} cur={cur_sc:.6f} best={best_sc:.6f}")

    return best_a, best_sc


# ----------------------------------------------------------------------
# Linear‑programming refinement (active‑set method)
# ----------------------------------------------------------------------


def lp_refine(f: np.ndarray,
              active_tol: float,
              t: float,
              max_lp_iters: int,
              time_deadline: float) -> Tuple[np.ndarray, float]:
    """
    One pass of the active‑set LP refinement described in the statement.
    Returns the updated vector and its fast surrogate score.
    """
    from scipy.optimize import linprog

    n = len(f)
    conv = convolution_fft(f, f)
    M = float(np.max(conv))

    # pick active indices (those within ``active_tol`` relative of the max)
    active_mask = conv >= M - active_tol * M
    active_idxs = np.where(active_mask)[0]
    # keep at most 30 most active positions to keep the LP tiny
    if len(active_idxs) > 30:
        # sort decreasing by conv value and take first 30
        order = np.argsort(-conv[active_idxs])
        active_idxs = active_idxs[order[:30]]

    m = len(active_idxs)
    if m == 0:
        # nothing to improve
        return f, compute_score(f)

    # build constraint matrix A (m × n)  :  (f * g)_k ≤ M  for k in active set
    A = np.zeros((m, n), dtype=float)
    for row, k in enumerate(active_idxs):
        i_start = max(0, k - (n - 1))
        i_end = min(k, n - 1)
        i_vals = np.arange(i_start, i_end + 1)
        j_vals = k - i_vals                     # 0 ≤ j < n
        A[row, j_vals] = f[i_vals]

    # Linear program: maximise Σ g_j  ⇔ minimise −Σ g_j
    c = -np.ones(n, dtype=float)
    bounds = [(0.0, None)] * n
    b = np.full(m, M, dtype=float)

    res = linprog(c,
                  A_ub=A,
                  b_ub=b,
                  bounds=bounds,
                  method='highs',
                  options={'presolve': True, 'time_limit': max(0.0, time_deadline - time.time())})

    if not res.success:
        # If LP fails we abort the refinement for this length.
        return f, compute_score(f)

    g0 = res.x
    sum_g0 = float(g0.sum())
    if sum_g0 <= 0.0:
        return f, compute_score(f)

    # Normalise g0 to a probability vector
    g = g0 / sum_g0

    # Convex blend
    f_new = (1.0 - t) * f + t * g
    f_new = project_to_simplex(f_new)

    return f_new, compute_score(f_new)


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42,
                     budget_s: int = 1000,
                     **kwargs) -> List[float]:
    """
    Search for a non‑negative step sequence that minimizes
    ``evaluate_sequence``.  The search proceeds as follows:

    * exploratory sampling of beta‑family shapes,
    * iterative LP refinement using the active‑set method,
    * final polishing with an Adam optimiser on the exact max‑convolution.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    # leave a small safety margin for the harness
    deadline = start_time + budget_s - 2.0

    best_seq: List[float] = []
    best_val: float = float("inf")

    # ------------------------------------------------------------------
    # optional fallback (if a previous run produced ``height_sequence_1``)
    # ------------------------------------------------------------------
    try:
        fallback = [float(x) for x in height_sequence_1]  # type: ignore
        val = evaluate_sequence(fallback)
        print(f"Fallback score = {val:.6f}")
        if val < best_val:
            best_val = val
            best_seq = fallback
    except Exception:
        pass

    # ------------------------------------------------------------------
    # 1.  Choose lengths to explore (powers of two give fast FFTs)
    # ------------------------------------------------------------------
    lengths = [2048, 4096, 8192]                     # larger lengths give a better bound
    for n in lengths:
        if time.time() >= deadline - 5.0:
            break
        print(f"\n=== Length n = {n} ===")

        # ------------------------------------------------------------------
        # 1a.  Random beta sampling + deterministic starters
        # ------------------------------------------------------------------
        local_best = None
        local_score = float("inf")

        starters = [arcsine_initial(n), np.full(n, 1.0 / n)]
        for st in starters:
            sc = compute_score(st)
            if sc < local_score:
                local_score = sc
                local_best = st.copy()

        # many random beta draws (fast surrogate)
        max_samples = 1200
        for _ in range(max_samples):
            if time.time() >= deadline - 5.0:
                break
            alpha = rng.uniform(0.1, 2.5)
            beta = rng.uniform(0.1, 2.5)
            eps = rng.uniform(0.001, max(1.0, 0.02 * n))
            cand = generate_beta_shape(n, alpha, beta, eps)
            sc = compute_score(cand)
            if sc < local_score:
                local_score = sc
                local_best = cand

        # ------------------------------------------------------------------
        # 2.  LP refinement loop
        # ------------------------------------------------------------------
        if local_best is None:
            continue

        f = local_best
        best_f = f.copy()
        best_f_score = compute_score(f)

        lp_iter = 0
        t = 0.25                     # initial blending factor
        while time.time() < deadline - 5.0 and lp_iter < 40:
            remaining = deadline - time.time()
            f, sc = lp_refine(f,
                              active_tol=1e-9,
                              t=t,
                              max_lp_iters=1,
                              time_deadline=deadline)

            if sc < best_f_score:
                best_f_score = sc
                best_f = f.copy()

            # evaluate with the official evaluator every few LP steps
            if lp_iter % 5 == 0:
                val = evaluate_sequence(f.tolist())
                print(f"  LP step {lp_iter:02d} – surrogate={sc:.6f} – official={val:.6f}")
                if val < best_val:
                    best_val = val
                    best_seq = f.tolist()
                    print(f"  *** New global best (LP) = {best_val:.6f} ***")

            # slowly decay the mixing factor
            t = max(0.02, t * 0.92)
            lp_iter += 1

        # ------------------------------------------------------------------
        # 3.  Adam sub‑gradient polishing on the best LP result
        # ------------------------------------------------------------------
        if time.time() < deadline - 5.0:
            remaining = deadline - time.time()
            max_iter = int(min(200_000, remaining * 30))
            print(f"\n  Adam‑subgrad polishing (max_iter={max_iter}) …")
            polished_a, _ = adam_subgrad_optimize(
                initial_a=best_f,
                time_deadline=deadline,
                max_iter=max_iter,
                lr0=0.15,
                eps_subgrad=1e-12,
            )
            val = evaluate_sequence(polished_a.tolist())
            print(f"  Polished score = {val:.6f}")
            if val < best_val:
                best_val = val
                best_seq = polished_a.tolist()
                print(f"  *** New global best (polish) = {best_val:.6f} ***")

    # ------------------------------------------------------------------
    # 4.  Guard – guarantee a return value
    # ------------------------------------------------------------------
    if not best_seq:
        # fallback to a safe uniform vector of moderate size
        fallback = np.full(1024, 1.0 / 1024)
        best_seq = fallback.tolist()
        best_val = evaluate_sequence(best_seq)

    print(f"\nFinished search. Best reported score = {best_val:.6f}")
    return best_seq
