"""
Soft‑max mirror‑descent optimiser for
    2·n·max(a*a)/(sum(a)**2)

The routine keeps the sequence on the probability simplex (sum = 1) and
updates it by an exponentiated‑gradient step.  Instead of using a
sub‑gradient of the raw maximum, we smooth the max with a
soft‑max (log‑sum‑exp) using a temperature β that is increased from a
moderate start value up to a large final value.  The gradient of the
soft‑max is a weighted sum of the individual convolution‑gradients,
which can be obtained efficiently with one FFT‑based cross‑correlation.

Several independent restarts (the historic best sequence is used if
available) are performed until the time budget expires.  The best
sequence seen is returned as a plain Python list.
"""

import time
import numpy as np
from scipy.optimize import linprog  # optional, not used here but kept for safety

# ----------------------------------------------------------------------
# Helper utilities
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Return the smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _conv_self(a: np.ndarray) -> np.ndarray:
    """Convolution a*a (full length 2·n‑1) using FFT."""
    n = a.shape[0]
    fft_a = np.fft.rfft(a, n=2 * n - 1)
    conv = np.fft.irfft(fft_a * fft_a, n=2 * n - 1)
    # Numerical noise may produce tiny negatives – clip for safety.
    np.maximum(conv, 0, out=conv)
    return conv


def _grad_softmax_conv(a: np.ndarray, conv: np.ndarray, beta: float) -> np.ndarray:
    """
    Compute the gradient of the soft‑max (temperature β) of the
    self‑convolution entries.

    The gradient is
        ∑_k w_k ∂/∂a conv_k(a)  with  w_k ∝ exp(β·conv_k)
    and ∂conv_k/∂a_i = 2·a_{k‑i} (if the index is valid).

    The sum is performed efficiently via a cross‑correlation (FFT).
    """
    # ----- soft‑max weights ------------------------------------------------
    # subtract the maximum to avoid overflow; all weights ≤ 1
    max_c = conv.max()
    w = np.exp(beta * (conv - max_c))
    w_sum = w.sum()
    if w_sum == 0.0:                     # pathological – fall back to pure max
        w = (conv == max_c).astype(float)
        w_sum = w.sum()
    w /= w_sum

    # ----- cross‑correlation  w_rev * a  ---------------------------------
    #   grad_i = 2 * Σ_k w_k * a_{k-i}
    # which is the correlation of w with a evaluated at lag = -i.
    n = a.shape[0]
    w_rev = w[::-1]                      # reverse the weight vector
    L = _next_pow2((2 * n - 1) + n - 1)  # length ≥ 3·n‑2
    fft_wr = np.fft.rfft(w_rev, n=L)
    fft_a = np.fft.rfft(a, n=L)
    corr_full = np.fft.irfft(fft_wr * fft_a, n=L)
    # relevant slice = indices n‑1 … 2·n‑2 (inclusive) → length n
    grad = 2.0 * corr_full[n - 1 : 2 * n - 1]
    return grad


# ----------------------------------------------------------------------
# Main optimisation entry point
# ----------------------------------------------------------------------


def propose_candidate(budget_s: int = 1000) -> list[float]:
    """
    Search for a non‑negative step function that minimises
        2·n·max(a*a)/(sum(a)**2)

    The function runs for at most ``budget_s`` seconds and returns the best
    sequence encountered.
    """
    # ------------------------------------------------------------------
    # 1.  Problem size, possible warm‑start
    # ------------------------------------------------------------------
    deadline = time.time() + budget_s - 5.0        # safety margin
    try:
        init = np.asarray(height_sequence_1, dtype=float)   # type: ignore[name-defined]
        n = init.shape[0]
        init = np.maximum(init, 0.0)
        s = init.sum()
        if s > 0:
            init = init / s                     # normalise to a probability vector
        else:
            init = None
    except Exception:
        init = None
        n = 4096                                 # a reasonable default length

    # ------------------------------------------------------------------
    # 2.  Optimiser hyper‑parameters
    # ------------------------------------------------------------------
    lower_bound = 1.0                # theoretical lower bound for the objective
    max_step = 5.0                   # polyak step size clipping
    best_a = None
    best_score = float("inf")
    restart = 0

    # temperature schedule for the soft‑max
    beta_start = 10.0
    beta_factor = 1.002
    beta_max = 15000.0

    while time.time() < deadline:
        rng = np.random.default_rng(int(time.time() * 1e6) % (2**32 - 1))

        # ------------------------------------------------------------------
        # 3.  Initial point for this restart
        # ------------------------------------------------------------------
        if init is not None and restart == 0:
            a = init.copy()
        else:
            a = rng.random(n)
        a = np.maximum(a, 1e-30)          # avoid exact zeros
        a = a / a.sum()                   # project onto the simplex

        # cumulative (mirrored) gradient – the exponentiated‑gradient variable
        g_cum = np.zeros_like(a)

        it = 0

        # ------------------------------------------------------------------

            max_conv = float(conv.max())
            cur_score = 2.0 * n * max_conv      # because sum(a) == 1

                g_cum = np.zeros_like(a)          # reset the mirror variable

            it += 1

        restart += 1

    # ------------------------------------------------------------------
    # 5.  Return the best sequence discovered (fallback to uniform)
    # ------------------------------------------------------------------
    if best_a is None:
        best_a = np.full(n, 1.0 / n)

    return [float(v) for v in best_a.tolist()]
