def _adam_optimize_symmetric(init_a, deadline, restart_id, target=1.503):
    n = len(init_a)
    fft_len = 1 << ((2*n - 2).bit_length())
    # Initialize logits:
    eps = 1e-12
    b = np.log(init_a + eps)
    # Adam states
    m = np.zeros_like(b)
    v = np.zeros_like(b)
    beta1=0.9
    beta2=0.999
    eps_adam = 1e-8

    # compute start objective
    max_val, _ = _maxconv_and_subgrad(init_a, fft_len)
    best_score = 2.0*n*max_val
    best_a = init_a.copy()
    stagnation=0
    it=0

    while time.time() < deadline:
        it += 1
        # compute a as softmax
        maxb = np.max(b)
        a = np.exp(b - maxb)
        a = a / a.sum()
        # symmetrize a
        a = symmetrize(a)
        a = a / a.sum()  # ensure normalized (should be unchanged)
        # compute conv and grad
        max_val, grad_a = _maxconv_and_subgrad(a, fft_len)
        cur_score = 2.0*n*max_val

        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            stagnation=0
            print(f"Restart {restart_id:02d} iter {it:6d}: best = {best_score:.6f}")
        else:
            stagnation+=1

        # termination:
        if best_score < target:
            break
        if stagnation > 5000:
            # jitter in logits
            b += jitter_std * np.random.randn(n)
            stagnation=0

        # project gradient to symmetric
        grad_a = symmetrize(grad_a)

        grad_obj = 2.0*n * grad_a
        avg = np.dot(a, grad_obj)
        grad_b = a * (grad_obj - avg)
        # symmetrize grad_b just in case
        grad_b = symmetrize(grad_b)

        # moments
        m = beta1*m + (1-beta1)*grad_b
        v = beta2*v + (1-beta2)*(grad_b**2)

        # bias corrections
        m_hat = m / (1 - beta1**it)
        v_hat = v / (1 - beta2**it)

        lr = 0.5 / np.sqrt(it)  # a bit higher
        b -= lr * m_hat / (np.sqrt(v_hat) + eps_adam)

        # symmetrize logits
        b = symmetrize(b)

    return best_a, best_score

assistantfinal**Solution Overview**

The objective  

\[
\text{score}(a)=\frac{2\,n\;\max(a*a)}{(\sum a)^2}
\]

is homogeneous – scaling all entries does not change the value.  
Therefore we can restrict ourselves to **probability vectors** (∑a = 1) and
focus on minimising  

\[
M(a)=\max (a*a) .
\]

A few facts are useful  

* The problem is symmetric: if we reverse a vector, the convolution does not
  change.  Averaging a vector with its reversal never increases the maximum,
  therefore an optimal solution can be taken **symmetric**.  
* The convolution is smooth almost everywhere; its sub‑gradient at the
  maximal index can be computed analytically.  This enables a fast
  gradient‑based optimiser.
* Good initialisations are highly non‑uniform – most mass should be placed
  near the two ends, while the centre stays small.  Simple parametric families
  (geometric decay, geometric + flat plateau, two‑parameter edge‑plateau
  shapes) already contain very strong candidates.

The algorithm consists of three stages

1. **Parametric exploration** – random samples from a few simple families
   (geometric, geometric + plateau, edge‑plateau).  
   For each sample the objective is evaluated (FFT based) and the best
   candidate is stored.
2. **Projected Adam optimisation** – starting from the best candidate found
   in stage 1 we run an Adam optimiser on the logits of a soft‑max
   representation.  After each update the vector is **symmetrised** and the
   gradient is projected onto the symmetric sub‑space.  A small jitter is
   injected when progress stalls.
3. **Fine‑tuning** – if time permits we run a longer Adam optimisation from
   the current best solution.

All steps respect the time budget (default 1000 s).  The final sequence is
returned as a plain Python list of floats; it can be fed directly into
`evaluate_sequence`.

```python
"""
Propose a step‑function sequence that minimises

    2·n·max(conv(a,a)) / (sum(a)**2)

The method consists of:

1. Random sampling of simple symmetric families (geometric decay, geometric
   decay with a flat centre, two‑parameter edge‑plateau) to obtain a good
   starting point.
2. Projected Adam optimisation on the soft‑max logits, with an explicit
   symmetrisation step after each update (the optimum can be taken symmetric
   without loss of optimality).
3. A final longer Adam run if time permits.

All helper functions are top‑level – there are no closures or lambdas.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Utility functions
# ----------------------------------------------------------------------


def _symmetrize(v: np.ndarray) -> np.ndarray:
    """Return (v + v[::-1]) / 2 – works for even and odd lengths."""
    return (v + v[::-1]) * 0.5


def _maxconv_and_subgrad(a: np.ndarray, fft_len: int):
    """
    Compute aperiodic convolution a*a, its maximal entry, and a sub‑gradient
    of that maximal entry w.r.t. a.

    Returns
    -------
    max_val : float
        The largest value of a*a.
    grad : ndarray, shape a.shape
        A sub‑gradient (zero where the index is not active).  For a convex
        max‑function any convex combination of the active gradients is valid.
    """
    n = a.shape[0]
    # convolution via FFT (only first 2·n‑1 entries are needed)
    fft = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(fft * fft, n=fft_len)[: 2 * n - 1]

    max_val = conv.max()
    # tolerance for numerically tied maxima
    tol = max_val * 1e-12 + 1e-14
    active = np.where(conv >= max_val - tol)[0]

    grad = np.zeros_like(a)
    for k in active:
        low = max(0, k - n + 1)
        high = min(n - 1, k)
        idx = np.arange(low, high + 1, dtype=int)
        # each pair (i, k-i) contributes 2*a[k-i] to the gradient of a[i]
        grad[idx] += 2.0 * a[k - idx]

    if active.size > 1:
        grad /= float(active.size)          # any convex combination is ok

    return max_val, grad


# ----------------------------------------------------------------------
# Parametric families (all symmetric by construction)
# ----------------------------------------------------------------------


def _geom_shape(n: int, r: float) -> np.ndarray:
    """Geometric decay from the edges: a[i] = r**min(i, n‑1‑i)."""
    d = np.minimum(np.arange(n), n - 1 - np.arange(n))
    a = np.exp(d * np.log(r))            # r**d but more stable for large d
    a /= a.sum()
    return a


def _geom_plateau_shape(n: int, r: float, cutoff: int) -> np.ndarray:
    """
    Geometric decay until distance `cutoff`, then a flat interior.

    Parameters
    ----------
    cutoff : int
        Number of points (starting from the edge) that follow the geometric
        decay.  The interior (length n‑2*cutoff) receives the constant value
        r**cutoff.
    """
    d = np.minimum(np.arange(n), n - 1 - np.arange(n))
    a = np.where(d <= cutoff,
                 np.exp(d * np.log(r)),
                 np.exp(cutoff * np.log(r)))
    a /= a.sum()
    return a


def _edge_plateau_shape(n: int, L: int, r: float, gamma: float) -> np.ndarray:
    """
    Two‑parameter family:
      * first L points on each side follow a geometric series with ratio r,
      * the middle part (if any) receives a constant weight gamma times the
        geometric scaling factor.

    The whole vector is scaled to sum to one.
    """
    if L == 0:                     # pure uniform distribution
        return np.full(n, 1.0 / n)

    # geometric vector for one side
    idx = np.arange(L)
    edge_vals = np.exp(idx * np.log(r))          # r**idx
    S_edge = edge_vals.sum()

    # denominator of the scaling factor
    interior_len = n - 2 * L
    denom = 2.0 * S_edge + interior_len * gamma
    if denom == 0.0:
        # degenerate case – fall back to uniform
        return np.full(n, 1.0 / n)

    scale = 1.0 / denom
    a = np.empty(n, dtype=np.float64)

    # left edge
    a[:L] = scale * edge_vals
    # right edge (mirrored)
    a[-L:] = scale * edge_vals[::-1]

    # interior (if any)
    if interior_len > 0:
        a[L:n - L] = scale * gamma

    return a


# ----------------------------------------------------------------------
# Projected Adam optimiser (logits + explicit symmetrisation)
# ----------------------------------------------------------------------


def _adam_optimize_symmetric(init_a: np.ndarray,
                             deadline: float,
                             restart_id: int,
                             target: float = 1.503) -> tuple[np.ndarray, float]:
    """
    Adam optimiser on the soft‑max logits of a probability vector.
    Symmetry is enforced explicitly after each update, and the gradient is
    projected onto the symmetric sub‑space.

    Returns
    -------
    best_a : ndarray
        The best probability vector found before `deadline`.
    best_score : float
        Corresponding value of 2·n·max(conv(a,a)).
    """
    n = init_a.shape[0]
    fft_len = 1 << ((2 * n - 2).bit_length())   # power‑of‑two for FFT speed

    eps_init = 1e-12
    b = np.log(init_a + eps_init)               # logits

    m = np.zeros_like(b)
    v = np.zeros_like(b)
    beta1, beta2 = 0.9, 0.999
    eps_adam = 1e-8

    # initial score
    max_val, _ = _maxconv_and_subgrad(init_a, fft_len)
    best_score = 2.0 * n * max_val
    best_a = init_a.copy()

    stagnation = 0
    it = 0

    while time.time() < deadline:
        it += 1

        # ---------- soft‑max ----------
        b_max = np.max(b)
        a = np.exp(b - b_max)
        a /= a.sum()

        # enforce symmetry
        a = _symmetrize(a)
        a /= a.sum()          # tiny numerical deviation may appear

        # ---------- objective & sub‑gradient ----------
        max_val, grad_a = _maxconv_and_subgrad(a, fft_len)
        cur_score = 2.0 * n * max_val

        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            stagnation = 0
            print(f"Restart {restart_id:02d} iter {it:6d}:   best = {best_score:.6f}")
        else:
            stagnation += 1

        # ---------- early stopping ----------
        if best_score < target:
            break
        if stagnation > 3000:
            # random jitter in log‑space to escape plateaus
            b += 0.02 * np.random.randn(n)
            stagnation = 0

        # ---------- gradient in log‑space ----------
        grad_a = _symmetrize(grad_a)                     # stay in symmetric sub‑space
        grad_obj = 2.0 * n * grad_a                      # d(2 n max) / d a

        avg = np.dot(a, grad_obj)                        # soft‑max Jacobian term
        grad_b = a * (grad_obj - avg)                    # ∂obj / ∂logits
        grad_b = _symmetrize(grad_b)                     # keep symmetry

        # ---------- Adam update ----------
        m = beta1 * m + (1.0 - beta1) * grad_b
        v = beta2 * v + (1.0 - beta2) * (grad_b ** 2)

        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        lr = 0.4 / np.sqrt(it)                          # decreasing learning rate
        b -= lr * m_hat / (np.sqrt(v_hat) + eps_adam)

        # enforce symmetry on logits as well
        b = _symmetrize(b)

    return best_a, best_score

# Main entry point
# ----------------------------------------------------------------------

    for _ in range(max_trials):

            break

        # Randomly pick a family
        family = rng.integers(0, 4)
        if family == 0:  # pure geometric

    # ------------------------------------------------------------------

    # ------------------------------------------------------------------

        best_a = np.full(n, 1.0 / n)

    # Return plain Python list (the evaluation harness expects this)
    return [float(v) for v in best_a.tolist()]
```
