```
def subgradient(a):
    conv = convolution_fft(a, a)
    M = conv.max()
    max_indices = np.where(conv >= M - 1e-12)[0]  # within tolerance
    # pick uniformly random among them
    k0 = rng.choice(max_indices)
    n = len(a)
    low = max(0, k0-(n-1))
    high = min(n-1, k0)
    idx = np.arange(low, high+1, dtype=int)
    grad_M = np.zeros_like(a)
    grad_M[idx] = 2 * a[k0 - idx]
    # compute full gradient
    s = a.sum()
    C = 2.0 * n / (s**2)
    grad = C * grad_M - (4.0 * n * M / (s**3)) * np.ones_like(a)
    return grad, M, s
```

The goal is to minimise  

\[
F(a)=\frac{2\,n\;\max(a*a)}{(\sum_i a_i)^2},
\qquad a_i\ge 0,
\]

where `a*a` is the linear self‑convolution of the step heights.  
Because the denominator and the convolution are both homogeneous of degree 2,
the value of `F` does **not** depend on an overall scale – we can therefore
normalise the vector to the probability simplex  
\(\sum_i a_i = 1\).  After normalisation the problem becomes

*minimise* \(M=\max(a*a)\) *subject to* \(a\ge 0,\;\sum a = 1\).

The code below solves the problem by **projected Adam optimisation** of the
*exact* (nonsmoothed) objective.  The gradient of the max‑convolution is a
sub‑gradient that can be written in closed form; we compute it directly
instead of using a soft‑max surrogate.  The optimisation loop consists of

1. **Sampling a pool of promising initial shapes**  
   – classic arcsine, uniform, a two‑parameter beta family and sparse Dirichlet‑like draws.
2. **Running Adam with the exact sub‑gradient** on the best candidate of each
   length (256 … 8192).  The step is projected onto the simplex after every
   update.
3. **Final polishing** of the globally best vector with a longer Adam run.

All operations are `O(n log n)` thanks to an FFT‑based convolution, so even
tens of thousands of iterations comfortably fit into the 1000 s budget.

```python
"""
propose_candidate – find a non‑negative step sequence with a low
evaluate_sequence score.

Algorithm
---------
1.  Sample several initial shapes (arcsine, uniform, beta‑family,
    sparse Dirichlet) for a range of lengths.
2.  For each length keep the best sampled shape and improve it with
    **projected Adam** that uses the *exact* sub‑gradient of the official
    objective (no soft‑max surrogate).
3.  Keep the globally best vector; run a final, longer Adam polish.
4.  Return the best sequence found (always non‑empty, non‑negative).

All helper functions are top‑level, there are no closures.
"""

import time
import numpy as np
from typing import List, Tuple

# ----------------------------------------------------------------------
# Low‑level utilities (no closures)
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """
    Linear convolution of two 1‑D arrays using real FFT.
    Returns an array of length len(a)+len(b)-1.
    """
    m = len(a) + len(b) - 1
    n_fft = next_pow2(m)
    A = np.fft.rfft(a, n=n_fft)
    B = np.fft.rfft(b, n=n_fft)
    C = A * B
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the probability simplex
    {x ≥ 0, Σx = 1}.  Duchi et al. (2008).
    """
    if np.all(v <= 0):
        # all entries non‑positive → uniform vector
        return np.full_like(v, 1.0 / v.size)

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho = np.where(u - cssv / np.arange(1, v.size + 1) > 0)[0]
    if len(rho) == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def compute_obj(a: np.ndarray) -> float:
    """
    Exact value of the official objective (without the safety clipping
    performed inside evaluate_sequence).  The vector a is assumed
    non‑negative; the sum is forced to be >0.
    """
    n = len(a)
    conv = convolution_fft(a, a)
    M = float(conv.max())
    s = float(a.sum())
    if s < 1e-12:
        return np.inf
    return 2.0 * n * M / (s * s)


def arcsine_initial(n: int) -> np.ndarray:
    """
    Classic arcsine shape:
        a_i ∝ 1 / sqrt((i+½)*(n‑½‑i)).
    Returned vector is normalised to Σa = 1.
    """
    i = np.arange(n, dtype=float)
    eps = 0.25
    vals = 1.0 / np.sqrt((i + 0.5) * (n - 0.5 - i) + eps)
    vals /= vals.sum()
    return vals


def generate_beta_shape(n: int, alpha: float, beta: float, eps: float) -> np.ndarray:
    """
    Symmetric beta‑family density on {0,…,n‑1}:
        a_i ∝ (i+ε)^{α‑1}·(n‑1‑i+ε)^{β‑1}.
    Normalised to Σa = 1.
    """
    i = np.arange(n, dtype=float)
    left = (i + eps) ** (alpha - 1.0)
    right = (n - 1 - i + eps) ** (beta - 1.0)
    base = left * right
    total = base.sum()
    if total <= 0.0 or not np.isfinite(total):
        return np.full(n, 1.0 / n)
    return base / total


# ----------------------------------------------------------------------
# Exact sub‑gradient of the objective
# ----------------------------------------------------------------------


def subgradient_exact(a: np.ndarray, rng: np.random.Generator) -> Tuple[np.ndarray, float]:
    """
    Compute a sub‑gradient of the official objective at a.
    Returns (grad, obj_value) where obj_value = 2·n·max(a*a)/(sum(a)^2).

    The sub‑gradient of max(a*a) is obtained from a random index among
    the (potentially several) maximisers.
    """
    n = len(a)
    conv = convolution_fft(a, a)               # length = 2n‑1
    M = float(conv.max())
    # indices where the maximum is attained (within a tiny tolerance)
    max_idx = np.where(conv >= M - 1e-12)[0]
    # pick one of them uniformly at random to obtain a valid sub‑gradient
    k0 = int(rng.choice(max_idx))

    # gradient of the selected convolution entry:
    #   d[(a*a)_k0]/da_i = 2 * a_{k0-i}  if 0 ≤ k0-i < n, else 0
    grad_M = np.zeros_like(a)
    low = max(0, k0 - (n - 1))
    high = min(n - 1, k0)
    idx = np.arange(low, high + 1, dtype=int)
    grad_M[idx] = 2.0 * a[k0 - idx]

    s = float(a.sum())
    C = 2.0 * n / (s * s)

    # Full gradient of the objective:
    grad = C * grad_M - (4.0 * n * M / (s ** 3)) * np.ones_like(a)

    obj = C * M
    return grad, obj


# ----------------------------------------------------------------------
# Projected Adam optimiser that uses the exact sub‑gradient
# ----------------------------------------------------------------------


def adam_exact_optimize(initial_a: np.ndarray,
                        time_deadline: float,
                        max_iter: int = 100_000,
                        lr0: float = 0.4,
                        beta1: float = 0.9,
                        beta2: float = 0.999,
                        eps: float = 1e-8,
                        noise_every: int = 0,
                        noise_scale: float = 0.0,
                        rng: np.random.Generator = None) -> Tuple[np.ndarray, float]:
    """
    Adam optimisation on the *exact* objective.
    The iterates are projected onto the simplex after each step.
    Returns the best vector seen and its (fast) objective value.
    """
    if rng is None:
        rng = np.random.default_rng()

    a = project_to_simplex(initial_a.astype(float).copy())
    best_a = a.copy()
    best_obj = compute_obj(a)

    m = np.zeros_like(a)
    v = np.zeros_like(a)

    for it in range(1, max_iter + 1):
        if time.time() >= time_deadline:
            break

        grad, cur_obj = subgradient_exact(a, rng)

        # Adam moment updates
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)

        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        step = lr0 * m_hat / (np.sqrt(v_hat) + eps)
        a_new = a - step

        if noise_every and (it % noise_every == 0):
            a_new += rng.normal(scale=noise_scale, size=a.shape)

        a_new = project_to_simplex(a_new)

        cur_obj = compute_obj(a_new)
        if cur_obj < best_obj:
            best_obj = cur_obj
            best_a = a_new.copy()

        a = a_new

        if it % 5000 == 0:
            print(f"    Adam iter={it:,d}  cur={cur_obj:.6f}  best={best_obj:.6f}")

    return best_a, best_obj

def propose_candidate(seed: int = 42,
                     budget_s: int = 1000,
                     **kwargs) -> List[float]:
    """
    Search for a non‑negative step sequence with a low evaluate_sequence
    value.  The routine explores several lengths, samples many
    initial shapes and runs a projected Adam optimiser that uses the
    exact sub‑gradient of the official objective.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    # keep a small safety margin for the harness
    deadline = start_time + budget_s - 2.0

    best_seq: List[float] = []
    best_val: float = float("inf")

    # --------------------------------------------------------------
    # 0.  Fallback from a previously computed construction (if any)
    # --------------------------------------------------------------
    try:
        fallback = [float(x) for x in height_sequence_1]  # type: ignore
        val = evaluate_sequence(fallback)
        print(f"Fallback (height_sequence_1) score = {val:.6f}")
        if val < best_val:
            best_val = val
            best_seq = fallback
    except Exception:
        pass

    # --------------------------------------------------------------
    # 1.  Exploration over several lengths
    # --------------------------------------------------------------
    lengths = [256, 512, 1024, 2048, 4096, 8192]

    for n in lengths:
        if time.time() >= deadline - 5.0:
            break
        print(f"\n=== Exploring length n = {n} ===")

        # ---- 1a.  Deterministic starters ---------------------------------
        local_best_a = None
        local_best_obj = float("inf")
        starters = [arcsine_initial(n), np.full(n, 1.0 / n)]
        for st in starters:
            obj = compute_obj(st)
            if obj < local_best_obj:
                local_best_obj = obj
                local_best_a = st.copy()

        # ---- 1b.  Random beta‑family sampling ----------------------------
        max_beta_samples = 300
        for _ in range(max_beta_samples):
            if time.time() >= deadline - 5.0:
                break
            alpha = rng.uniform(0.1, 3.0)
            beta = rng.uniform(0.1, 3.0)
            eps = rng.uniform(0.001, max(1.0, 0.02 * n))
            cand = generate_beta_shape(n, alpha, beta, eps)
            obj = compute_obj(cand)
            if obj < local_best_obj:
                local_best_obj = obj
                local_best_a = cand

        # ---- 1c.  Sparse Dirichlet‑like draws -----------------------------
        for _ in range(100):
            if time.time() >= deadline - 5.0:
                break
            conc = rng.uniform(0.01, 0.5)            # small concentration → sparse vectors
            raw = rng.gamma(conc, 1.0, size=n)
            cand = raw / raw.sum()
            obj = compute_obj(cand)
            if obj < local_best_obj:
                local_best_obj = obj
                local_best_a = cand

        # --------------------------------------------------------------
        # 2.  Refine the locally best candidate with Adam (exact gradient)
        # --------------------------------------------------------------
        if local_best_a is not None and time.time() < deadline - 5.0:
            remaining = deadline - time.time()
            max_iter = int(min(200_000, remaining * 30))
            print(f"  Refining (Adam exact) – max_iter={max_iter}")
            refined_a, _ = adam_exact_optimize(
                initial_a=local_best_a,
                time_deadline=deadline,
                max_iter=max_iter,
                lr0=0.5,
                beta1=0.9,
                beta2=0.999,
                eps=1e-8,
                noise_every=2500,
                noise_scale=5e-5,
                rng=rng,
            )
            # official evaluation (expensive but only once per length)
            val = evaluate_sequence(refined_a.tolist())
            print(f"  → refined official score = {val:.6f}")
            if val < best_val:
                best_val = val
                best_seq = refined_a.tolist()
                print(f"  *** New global best: {best_val:.6f} (n={n}) ***")
        else:
            # If for some reason we have no refined candidate, just test the raw best
            if local_best_a is not None:
                val = evaluate_sequence(local_best_a.tolist())
                if val < best_val:
                    best_val = val
                    best_seq = local_best_a.tolist()
                    print(f"  *** New global best (raw) ***")

    # --------------------------------------------------------------
    # 3.  Final polishing of the overall best vector
    # --------------------------------------------------------------
    if best_seq and time.time() < deadline - 5.0:
        a = np.array(best_seq, dtype=float)
        remaining = deadline - time.time()
        max_iter = int(min(400_000, remaining * 30))
        print(f"\nFinal polishing (Adam exact) – max_iter={max_iter}")
        polished_a, _ = adam_exact_optimize(
            initial_a=a,
            time_deadline=deadline,
            max_iter=max_iter,
            lr0=0.35,
            beta1=0.9,
            beta2=0.999,
            eps=1e-8,
            noise_every=1500,
            noise_scale=3e-5,
            rng=rng,
        )
        final_val = evaluate_sequence(polished_a.tolist())
        print(f"Final polished official score = {final_val:.6f}")
        if final_val < best_val:
            best_val = final_val
            best_seq = polished_a.tolist()
            print("*** Updated global best after final polish ***")

    # --------------------------------------------------------------
    # 4.  Guard – guarantee a return value
    # --------------------------------------------------------------
    if not best_seq:
        best_seq = [1.0] * 256
        best_val = evaluate_sequence(best_seq)

    print(f"\nFinished search. Best reported score = {best_val:.6f}")
    return best_seq
```
