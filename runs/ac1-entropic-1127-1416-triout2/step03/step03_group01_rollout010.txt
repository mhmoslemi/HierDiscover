"""
Propose a low‑discrepancy step‑function.

The search consists of three phases:

1.  **Exploration** – for several lengths we generate a few
    deterministic (arcsine, uniform) and random initial shapes.
    The three best (according to the cheap surrogate
    ``2·n·max(a*a)``) are kept.

2.  **Smooth‑to‑hard Adam optimisation** – each kept candidate is
    refined by Adam.  A temperature λ drives a soft‑max
    approximation of the max‑convolution; λ grows exponentially
    from a moderate start (≈5) to a very large value (≈600–2000)
    so that the optimisation finally sees the true max.
    After each Adam step the vector is projected onto the simplex
    (Σa = 1, a ≥ 0) and, optionally, symmetrised
    ``a ← (a + a[::-1]) / 2``.  The optimizer runs as long as the
    time budget permits.

3.  **Final polishing** – the globally best vector from phase 2
    undergoes a longer Adam run with an even larger λ‑end.
    The resulting vector is evaluated with the official
    ``evaluate_sequence`` function; if it improves the best score
    it becomes the final answer.

All helper utilities are top‑level, there are no closures and the
code respects the 1000 s time limit.
"""

import time
import numpy as np
from typing import List, Tuple

# ----------------------------------------------------------------------
# Low‑level utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """
    Linear convolution of two real 1‑D arrays using the real FFT.
    Returns an array of length len(a)+len(b)-1.
    """
    m = len(a) + len(b) - 1
    n_fft = next_pow2(m)
    A = np.fft.rfft(a, n=n_fft)
    B = np.fft.rfft(b, n=n_fft)
    C = A * B
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the probability simplex
    {x ≥ 0, Σx = 1}.  Duchi et al., 2008.
    """
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / v.size)

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho = np.where(u - cssv / np.arange(1, v.size + 1) > 0)[0]
    if len(rho) == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def compute_score(a: np.ndarray) -> float:
    """
    Fast surrogate for the official score.
    The vector is assumed to be normalised to Σa = 1,
    therefore the denominator of the official formula equals 1
    and the score reduces to ``2·n·max(a*a)``.
    """
    n = len(a)
    conv = convolution_fft(a, a)
    return 2.0 * n * float(np.max(conv))


def arcsine_initial(n: int) -> np.ndarray:
    """Classic arcsine shape a_i ∝ 1/√((i+½)*(n‑½‑i))."""
    i = np.arange(n, dtype=float)
    eps = 0.25
    vals = 1.0 / np.sqrt((i + 0.5) * (n - 0.5 - i) + eps)
    vals /= vals.sum()
    return vals


def generate_beta_shape(n: int, alpha: float, beta: float, eps: float) -> np.ndarray:
    """
    Symmetric beta‑family density on {0,…,n‑1}:
        a_i ∝ (i+ε)^{α‑1}·(n‑1‑i+ε)^{β‑1}.
    Returned vector is normalised to Σa = 1.
    """
    i = np.arange(n, dtype=float)
    left = (i + eps) ** (alpha - 1.0)
    right = (n - 1 - i + eps) ** (beta - 1.0)
    base = left * right
    total = base.sum()
    if total <= 0.0 or not np.isfinite(total):
        return np.full(n, 1.0 / n)
    return base / total


# ----------------------------------------------------------------------
# Smooth gradient (soft‑max) of the true objective
# ----------------------------------------------------------------------


def smooth_gradient(a: np.ndarray, lam: float) -> Tuple[np.ndarray, float]:
    """
    Gradient of the objective
        f(a) = 2·n·max(a*a) / (Σa)²
    where the max is replaced by a soft‑max with temperature λ.
    Returns (gradient, smooth‑max value).
    """
    n = len(a)
    conv = convolution_fft(a, a)               # length 2n‑1
    max_c = np.max(conv)

    # soft‑max weights (numerically stable)
    exp_term = np.exp(lam * (conv - max_c))
    sum_exp = np.sum(exp_term)
    w = exp_term / sum_exp
    B = max_c + np.log(sum_exp) / lam          # smooth approximation of the max

    # weighted sub‑gradient of the convolution
    w_rev = w[::-1]
    conv_rev = convolution_fft(a, w_rev)       # length 3n‑2
    L = conv.shape[0]                          # = 2n‑1
    weighted_grad = 2.0 * conv_rev[L - 1 - np.arange(n)]

    s = np.sum(a)
    term1 = (2.0 * n / (s ** 2)) * weighted_grad
    term2 = -(4.0 * n * B / (s ** 3)) * np.ones_like(a)
    grad = term1 + term2
    return grad, B


# ----------------------------------------------------------------------
# Adam optimiser – smooth → hard max, optional symmetry
# ----------------------------------------------------------------------


def adam_optimize(
    initial_a: np.ndarray,
    time_deadline: float,
    max_iter: int = 100_000,
    lr0: float = 0.2,
    lam_start: float = 5.0,
    lam_end: float = 800.0,
    noise_every: int = 0,
    noise_scale: float = 0.0,
    enforce_sym: bool = False,
    lam_switch: float = None,
) -> Tuple[np.ndarray, float]:
    """
    Projected Adam optimiser on the (soft‑max) surrogate.
    *lam* grows exponentially from ``lam_start`` to ``lam_end``.
    When ``lam`` exceeds ``lam_switch`` a hard‑max (sub‑gradient)
    is used instead of the soft‑max to obtain the true gradient.
    """
    a = project_to_simplex(initial_a.astype(float).copy())
    best_a = a.copy()
    best_sc = compute_score(a)

    beta1, beta2 = 0.9, 0.999
    eps = 1e-8
    m = np.zeros_like(a)
    v = np.zeros_like(a)

    n = len(a)
    if lam_switch is None:
        lam_switch = 0.9 * lam_end  # switch to hard max near the end

    for it in range(1, max_iter + 1):
        if time.time() >= time_deadline:
            break

        # exponential schedule for λ
        progress = min(1.0, it / max_iter)
        lam = lam_start * (lam_end / lam_start) ** progress

        # compute gradient
        if lam < lam_switch:
            grad, _ = smooth_gradient(a, lam)
        else:
            # hard‑max sub‑gradient
            conv = convolution_fft(a, a)
            max_c = np.max(conv)
            k_max = int(np.argmax(conv))
            idx = k_max - np.arange(n)
            mask = (idx >= 0) & (idx < n)
            grad_m = np.zeros_like(a)
            grad_m[mask] = 2.0 * a[idx[mask]]               # ∂M/∂a
            s = np.sum(a)
            grad = (2.0 * n / (s ** 2)) * grad_m - (4.0 * n * max_c / (s ** 3)) * np.ones_like(a)

        # Adam moments
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)

        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        step = lr0 * m_hat / (np.sqrt(v_hat) + eps)
        a_new = a - step

        # optional isotropic noise (helps to escape plateaus)
        if noise_every and (it % noise_every == 0):
            a_new += np.random.randn(n) * noise_scale

        # symmetry enforcement (optional)
        if enforce_sym:
            a_new = (a_new + a_new[::-1]) / 2.0

        # projection onto the simplex Σa = 1 , a ≥ 0
        a_new = project_to_simplex(a_new)

        cur_sc = compute_score(a_new)
        if cur_sc < best_sc:
            best_sc = cur_sc
            best_a = a_new.copy()

        a = a_new

        # occasional progress report
        if it % 5000 == 0:
            print(f"    Adam iter={it:,d}  cur={cur_sc:.6f}  best={best_sc:.6f}  λ={lam:.1f}")

    return best_a, best_sc
