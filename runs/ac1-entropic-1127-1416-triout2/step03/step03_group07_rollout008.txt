            score = evaluate_sequence(seq)
            if score < best_score:
                best_score=score; best_seq=seq

    # Parameter families: define functions that generate a of length n.

    def gen_gaussian(c):
        # center at (n-1)/2
        i = np.arange(n)
        m = (n-1)/2
        a = np.exp(-c * ((i - m)**2))
        a_sum = a.sum()
        return a / a_sum

    def gen_triangular():
        i = np.arange(n)
        a = (i+1)*(n-i)
        a = a.astype(float)
        a /= a.sum()
        return a

    def gen_power(alpha):
        i = np.arange(n)
        a = ((i+1)*(n-i))**alpha
        a /= a.sum()
        return a

    def gen_edge_heavy(beta):
        i = np.arange(n)
        a = ((i+1)*(n-i))**(-beta) + 1e-12  # avoid zero
        a /= a.sum()
        return a

    def gen_sine():
        i = np.arange(n)
        a = np.sin(np.pi*(i+0.5)/n)**2   # sine squared shape
        a /= a.sum()
        return a

    families = [
        (gen_gaussian, {'c': [0.0005, 0.001, 0.002, 0.005, 0.01, 0.02, 0.05]}),

        (gen_power, {'alpha': [0.2, 0.5, 1, 2, 3]}),
        (gen_edge_heavy, {'beta': [0.2, 0.5, 1, 2, 3]}),
        (gen_sine, {})
    ]

    # Evaluate each combination
    for func, params in families:
        if not params: # no param grid
            a = func()
            score = evaluate_sequence(a.tolist())

def propose_candidate(budget_s=1000):
    import time, numpy as np, itertools, random

    start = time.time()
    deadline = start + budget_s - 0.5

    # Determine length n: try using length of historic sequence if exists else default.
    try:
        init_seq = np.asarray(height_sequence_1, dtype=float)
        n = int(init_seq.shape[0])
        init_seq = np.maximum(init_seq, 0.0)
        s = init_seq.sum()
        if s > 0:
            init_seq = init_seq / s
        else:
            init_seq = None
    except Exception:
        init_seq = None
        n = 4096   # default

    # Helper to compute score quickly
    def fast_score(a):
        # Input a as np array summing to 1
        L = 2 * n - 1
        fft = np.fft.rfft(a, n=L)
        conv = np.fft.irfft(fft * fft, n=L)
        return 2.0 * n * conv.max()

    # Helper to create random probability vector
    def random_prob():
        x = np.random.random(n)
        x = np.maximum(x, 1e-30)  # avoid zero for stability
        return x / x.sum()

    # Initialize population
    pop_size = 60
    elite_frac = 0.15
    elite_size = max(1, int(pop_size * elite_frac))
    population = []
    # Add historic best if available
    if init_seq is not None:
        population.append(init_seq.copy())
    # Add some deterministic shapes
    # Triangular
    tri = np.arange(1, n+1) * np.arange(n, 0, -1)
    tri = tri.astype(float)
    tri /= tri.sum()
    population.append(tri)
    # Edge heavy: p=-0.5
    p = -0.5
    idx = np.arange(n)
    edge = ((idx+1)*(n-idx))**p
    edge = edge.astype(float)
    edge = edge / edge.sum()
    population.append(edge)
    # Cosine shape (inverted)
    cos = 0.5 * (1 + np.cos(np.pi * idx / (n-1)))
    cos = cos / cos.sum()
    population.append(cos)
    # Sine squared shape
    sine = np.sin(np.pi * (idx + 0.5) / n)**2
    sine = sine / sine.sum()
    population.append(sine)

    # Fill rest with random
    while len(population) < pop_size:
        population.append(random_prob())

    # Evaluate scores
    scores = np.empty(len(population))
    for i, a in enumerate(population):
        scores[i] = fast_score(a)

    best_idx = np.argmin(scores)
    best_a = population[best_idx].copy()
    best_score = scores[best_idx]
    print(f"Initial best score: {best_score:.6f}")

    # Evolution loop
    gen = 0
    sigma = 0.1   # initial mutation strength
    # Adaptive sigma schedule: decays exponentially
    while time.time() < deadline:
        gen += 1
        # Select elites
        elite_inds = np.argsort(scores)[:elite_size]
        elites = [population[i] for i in elite_inds]
        # Create new population
        new_pop = []
        # Keep elites
        new_pop.extend(elites)
        # Generate offspring until pop size reached
        while len(new_pop) < pop_size:
            # random choose two parents from elites
            a1, a2 = np.random.choice(elites, size=2, replace=False)
            # convex combination
            alpha = np.random.rand()
            child = alpha * a1 + (1 - alpha) * a2
            # mutation
            if sigma > 0:
                noise = sigma * np.random.randn(n)
                child = child + noise * child    # relative noise
                child = np.maximum(child, 0.0)
            # renormalize
            s = child.sum()
            if s == 0:
                child = np.full(n, 1.0 / n)
            else:
                child = child / s
            new_pop.append(child)

        # Evaluate new population
        new_scores = np.empty(pop_size)
        for i, a in enumerate(new_pop):
            new_scores[i] = fast_score(a)

        # Update best
        cur_best_idx = np.argmin(new_scores)
        cur_best_score = new_scores[cur_best_idx]
        if cur_best_score < best_score:
            best_score = cur_best_score
            best_a = new_pop[cur_best_idx].copy()
            print(f"[{time.time():.1f}] Generation {gen}: new best = {best_score:.6f}")
            # early exit if target reached
            if best_score < 1.503:
                break

        population = new_pop
        scores = new_scores

        # adapt sigma: maybe reduce every few generations
        if gen % 20 == 0:
            sigma *= 0.9
            sigma = max(sigma, 1e-4)

        # also optionally inject random individuals occasionally
        if gen % 50 == 0:
            # replace worst half with random
            worst_inds = np.argsort(scores)[- (pop_size // 2):]
            for idx in worst_inds:
                population[idx] = random_prob()
                scores[idx] = fast_score(population[idx])

    # Final verification using evaluate_sequence
    try:
        final_score = evaluate_sequence(best_a.tolist())
        print(f"Final score via evaluate_sequence: {final_score}")
        if final_score < best_score:
            best_score = final_score
            # best_a already correct
    except Exception:
        pass

    # Ensure sum atleast 0.01? Actually any sum>0 required; we have sum=1.
    # Return as list of floats
    return [float(v) for v in best_a.tolist()]

"""
Evolutionary+Mirror‑Descent optimiser for

    2·n·max(a*a) / (sum(a)**2)

The algorithm works in three stages:

1. **Initial population** – a mix of random vectors, simple analytic
   shapes (triangular, cosine, edge‑heavy power laws) and the historic
   best sequence (if available).

2. **Evolutionary loop** – a small population (≈60) is evolved with
   selection, convex recombination and Gaussian mutation.  After each
   generation all candidates are re‑normalised onto the probability
   simplex and their objective is evaluated quickly via an FFT‑based
   convolution.  The best individual is kept and the mutation strength
   is slowly reduced.

3. **Local refinement** – the current best vector is polished with a
   mirror‑descent (exponentiated‑gradient) update that uses the exact
   sub‑gradient of the ‑∞‑norm of the auto‑convolution.  A Polyak
   step‑size (clipped) is employed; this often squeezes a few extra
   percent out of the solution.

The routine runs for at most ``budget_s`` seconds (default 1000) and
returns the best sequence found as a plain list of ``float``.  All
values are non‑negative and automatically normalised to unit sum, so
the final call to ``evaluate_sequence`` yields the true score.
"""

import numpy as np
import itertools

def _max_and_subgradient(a: np.ndarray) -> tuple[float, np.ndarray]:
    """
    Return (max entry of a*a, a sub‑gradient of that entry).

    The convolution is computed with FFT for speed.
    The sub‑gradient is the average of the exact gradients of all
    active (max‑attaining) convolution entries.
    """
    n = a.shape[0]
    L = 2 * n - 1
    fft = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(fft * fft, n=L)
    max_val = float(conv.max())

    # indices that are (almost) tied for the maximum
    tol = max_val * 1e-12 + 1e-14
    active = np.where(conv >= max_val - tol)[0]

    grad = np.zeros_like(a)
    for k in active:
        low = max(0, k - n + 1)
        high = min(n - 1, k)
        idx = np.arange(low, high + 1, dtype=int)
        partner = k - idx
        grad[idx] += 2.0 * a[partner]

    if active.size > 1:
        grad /= float(active.size)
    return max_val, grad


def _fast_score(a: np.ndarray) -> float:
    """
    Fast evaluation of the objective 2·n·max(conv) for a vector that
    already satisfies sum(a)=1 and a≥0.
    """
    n = a.shape[0]
    L = 2 * n - 1
    fft = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(fft * fft, n=L)
    return 2.0 * n * float(conv.max())


def _random_prob(n: int) -> np.ndarray:
    """Draw a random point on the (n‑dim) probability simplex."""
    x = np.random.random(n)
    x = np.maximum(x, 1e-30)      # avoid exact zeros
    return x / x.sum()
