"""
Propose a low‑discrepancy step‑function.

The search works in three phases:

1. **Exploratory sampling** – for several lengths we draw many
   candidates from a flexible beta family (α,β,ε) together with a few
   deterministic starters (uniform, arcsine).  Each candidate is scored
   with the *exact* objective
        J(a) = 2·n·max(a*a) / (∑a)² .
   The best candidate per length is kept.

2. **Projected Adam optimisation** – the best candidate per length is
   refined by a projected Adam optimiser.  First a soft‑max
   (log‑sum‑exp) surrogate of the max is used (λ grows from a small
   value to ≈ 50) to obtain a smooth landscape; after a configurable
   fraction of the budget the optimiser switches to the true sub‑gradient
   of the max (no smoothing).  After every step the iterate is projected
   onto the probability simplex (∑a = 1, a ≥ 0).  The best iterate (according
   to the exact objective) is retained.

3. **Final polishing** – the globally best vector is polished again with
   the same Adam scheme, this time allocating almost the whole remaining
   time and ending with a long exact‑gradient phase.

All heavy work (convolutions, gradients, projections) is implemented
with NumPy/FFT and runs comfortably within the 1000 s budget.  The
returned sequence is guaranteed to be non‑negative, non‑empty and to
respect the evaluator’s clamping limits.
"""

import time
import numpy as np
from typing import List, Tuple

# ----------------------------------------------------------------------
# Low‑level utilities (no closures)
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """
    Linear convolution of two real 1‑D arrays using the real FFT.
    Returns an array of length len(a)+len(b)-1.
    """
    m = len(a) + len(b) - 1
    n_fft = next_pow2(m)
    A = np.fft.rfft(a, n=n_fft)
    B = np.fft.rfft(b, n=n_fft)
    C = A * B
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the probability simplex
    {x | x_i ≥ 0, Σ x_i = 1}.
    Duchi et al., 2008.
    """
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / v.size)

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho = np.where(u - cssv / np.arange(1, v.size + 1) > 0)[0]
    if len(rho) == 0:          # should not happen, but guard anyway
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def objective_exact(a: np.ndarray) -> float:
    """
    Exact value of the official evaluation function (up to the evaluator’s
    clamping, which is irrelevant for the ranges we generate).
    The vector a is assumed to be non‑negative.
    """
    n = a.shape[0]
    conv = convolution_fft(a, a)
    max_val = np.max(conv)
    s = np.sum(a)
    if s < 1e-12:          # safety – avoid division by zero
        return np.inf
    return 2.0 * n * max_val / (s * s)


def arcsine_initial(n: int) -> np.ndarray:
    """Classic arcsine shape: a_i ∝ 1/√((i+½)*(n‑½‑i))."""
    i = np.arange(n, dtype=float)
    eps = 0.25
    vals = 1.0 / np.sqrt((i + 0.5) * (n - 0.5 - i) + eps)
    vals /= vals.sum()
    return vals


def generate_beta_shape(n: int, alpha: float, beta: float, eps: float) -> np.ndarray:
    """
    Symmetric beta‑family density on {0,…,n‑1}:
        a_i ∝ (i+ε)^{α‑1}·(n‑1‑i+ε)^{β‑1}.
    Returns a normalized (ℓ₁) vector.
    """
    i = np.arange(n, dtype=float)
    left = (i + eps) ** (alpha - 1.0)
    right = (n - 1 - i + eps) ** (beta - 1.0)
    base = left * right
    total = base.sum()
    if total <= 0.0 or not np.isfinite(total):
        # fall back to uniform if something goes wrong
        return np.full(n, 1.0 / n)
    return base / total


# ----------------------------------------------------------------------
# Gradient utilities
# ----------------------------------------------------------------------


def smooth_gradient(a: np.ndarray, lam: float) -> Tuple[np.ndarray, float]:
    """
    Gradient of the smoothed objective where the max is replaced by the
    log‑sum‑exp with temperature λ.
    Returns (gradient, smoothed_max).
    """
    n = a.shape[0]
    conv = convolution_fft(a, a)                 # length 2n‑1
    max_c = np.max(conv)

    # stable soft‑max weights
    exp_term = np.exp(lam * (conv - max_c))
    sum_exp = np.sum(exp_term)
    w = exp_term / sum_exp                         # weights ∈ Δ_{2n‑1}
    B = max_c + np.log(sum_exp) / lam              # smooth max

    # weighted gradient of the convolution:
    #   ∂ Σ_k w_k (a*a)_k / ∂a = 2 * (a * w_rev)
    w_rev = w[::-1]
    conv_rev = convolution_fft(a, w_rev)           # length n + (2n‑1) - 1 = 3n‑2
    L = 2 * n - 1
    weighted_grad = 2.0 * conv_rev[L - 1 - np.arange(n)]

    s = np.sum(a)
    term1 = (2.0 * n / (s * s)) * weighted_grad
    term2 = -(4.0 * n * B / (s ** 3)) * np.ones_like(a)
    grad = term1 + term2
    return grad, B


def exact_gradient(a: np.ndarray) -> Tuple[np.ndarray, float]:
    """
    Sub‑gradient of the true objective (max instead of soft‑max).
    Returns (gradient, max_convolution).
    The formula follows from
        ∂max_k (a*a)_k / ∂a_i = a_i + a_{k*‑i}   (if the latter index is valid)
    where k* is an index attaining the maximum.
    """
    n = a.shape[0]
    conv = convolution_fft(a, a)
    max_idx = int(np.argmax(conv))
    max_val = conv[max_idx]

    # start with the always‑present term a_i
    grad_num = a.copy()                # contribution a_i
    # add the partner term a_{k*‑i} where it exists
    idx = np.arange(n)
    partner = max_idx - idx
    mask = (partner >= 0) & (partner < n)
    grad_num[mask] += a[partner[mask]]

    s = np.sum(a)
    term1 = (2.0 * n / (s * s)) * grad_num
    term2 = -(4.0 * n * max_val / (s ** 3)) * np.ones_like(a)
    grad = term1 + term2
    return grad, max_val


# ----------------------------------------------------------------------
# Projected Adam optimiser (smooth → exact switch)
# ----------------------------------------------------------------------


def adam_optimize(initial_a: np.ndarray,
                  time_deadline: float,
                  max_iter: int = 150_000,
                  lr0: float = 0.2,
                  lam_start: float = 0.5,
                  lam_end: float = 50.0,
                  exact_start_frac: float = 0.7,
                  noise_every: int = 1500,
                  noise_scale: float = 5e-5) -> Tuple[np.ndarray, float]:
    """
    Projected Adam optimiser.
    * smooth phase* uses a log‑sum‑exp surrogate (λ grows from lam_start to lam_end);
    after `exact_start_frac * max_iter` iterations it switches to the true
    sub‑gradient of the max.
    Returns the best iterate (projected onto the simplex) and its exact objective.
    """
    a = project_to_simplex(initial_a.astype(float).copy())
    best_a = a.copy()
    best_val = objective_exact(a)

    beta1, beta2 = 0.9, 0.999
    eps = 1e-8
    m = np.zeros_like(a)
    v = np.zeros_like(a)

    n = len(a)
    exact_start = int(max_iter * exact_start_frac)

    for it in range(1, max_iter + 1):
        if time.time() >= time_deadline:
            break

        # ----------- choose gradient ----------
        if it <= exact_start:
            # smooth stage
            lam = lam_start + (lam_end - lam_start) * (it / exact_start)
            grad, _ = smooth_gradient(a, lam)
        else:
            # exact stage
            grad, _ = exact_gradient(a)

        # ---------- Adam moments ----------
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)

        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        step = lr0 * m_hat / (np.sqrt(v_hat) + eps)
        a_new = a - step

        # optional noise to escape plateaus
        if noise_every and (it % noise_every == 0):
            a_new += np.random.randn(n) * noise_scale

        # simplex projection (keeps sum = 1)
        a_new = project_to_simplex(a_new)

        cur_val = objective_exact(a_new)
        if cur_val < best_val:
            best_val = cur_val
            best_a = a_new.copy()

        a = a_new

        # occasional progress report
        if it % 10_000 == 0:
            print(f"    Adam iter={it:,d}  cur={cur_val:.6f}  best={best_val:.6f}")

    return best_a, best_val

    except Exception:

    # ------------------------------------------------------------------
    # 1. Exploration over several lengths
    # ------------------------------------------------------------------
    lengths = [512, 1024, 2048, 4096, 8192]   # larger lengths allow finer shapes

    for n in lengths:
        if time.time() >= deadline - 5.0:
            break
        print(f"\n=== Exploring length n = {n} ===")

        # ---- 1a. Deterministic starters ---------------------------------
        starters = [np.full(n, 1.0 / n), arcsine_initial(n)]
        local_best_a = None
        local_best_score = float("inf")
        for st in starters:
            sc = objective_exact(st)
            if sc < local_best_score:
                local_best_score = sc
                local_best_a = st.copy()

        # ---- 1b. Random beta‑family sampling ----------------------------
        # number of samples adapts to remaining time

            cand = generate_beta_shape(n, alpha, beta, eps)
            sc = objective_exact(cand)
            if sc < local_best_score:

                best_seq = refined_a.tolist()
                print(f"  *** New global best: {best_val:.6f} (n={n}) ***")
        else:

                if val < best_val:

                    best_seq = local_best_a.tolist()
                    print(f"  *** New global best (raw) ***")

    # 3. Final polishing of the overall best vector
    # ------------------------------------------------------------------
    if best_seq and time.time() < deadline - 5.0:

        best_seq = [1.0] * 256
        best_val = evaluate_sequence(best_seq)

    return best_seq
