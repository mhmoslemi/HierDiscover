import numpy as np, time, scipy.optimize

def build_conv_matrix(a):
    n = len(a)
    rows = 2*n-1
    A = np.zeros((rows, n), dtype=np.float64)
    for i in range(n):
        A[i:i+n, i] = a[i]
    return A

def lp_step(a, M, t):
    n = len(a)
    A = build_conv_matrix(a)
    b = np.full(2*n-1, M + 1e-12, dtype=np.float64)  # slack
    c = -np.ones(n)   # maximize sum
    bounds = [(0, None)] * n
    res = scipy.optimize.linprog(c, A_ub=A, b_ub=b, bounds=bounds, method='highs')
    if not res.success:
        return a, False
    x = res.x
    sumx = np.sum(x)
    if sumx <= 0:
        return a, False
    g = x / sumx  # normalized to sum=1
    # mix
    new_a = (1 - t) * a + t * g
    # reproject onto simplex (just ensure sum=1)
    new_a = np.maximum(new_a, 0)
    new_a = new_a / new_a.sum()
    return new_a, True

"""
Hybrid LP + soft‑max optimisation for the step‑function minimisation problem.

The idea is to alternate two complementary optimisation phases:

1. **LP “mass‑inflation” step** (inspired by the original iterative LP scheme):
   For the current distribution `a` (∑a = 1) we compute
       M = max_k (a * a)[k] .
   We then solve the linear program
       maximise    ∑_j g_j
       subject to  (a * g)[k] ≤ M   (k = 0 … 2n‑2)
                   g_j ≥ 0 .
   The solution `g` can carry more total weight than `a` while respecting the same
   peak autocorrelation `M`.  After normalising `g` to unit sum we mix it with the
   current vector:   a ← (1‑t)·a + t·g .  This strictly reduces the
   objective `2·n·max(a*a) / (∑a)²` and moves the distribution towards a
   better shape.

2. **Soft‑max exponentiated‑gradient refinement**:
   Using the smooth approximation
       Φ_λ(a) = (2·n/∑a²) * (1/λ)·log ∑_k exp(λ·(a*a)_k)
   we run a few hundred exponentiated‑gradient steps (the code from the
   previous best solution).  This polishes the LP‑output and can escape
   shallow plateaus.

Both phases respect the non‑negativity and simplex constraints, require only
basic NumPy/SciPy primitives and fit comfortably into the 1000 s budget.
The routine returns the best feasible sequence discovered.
"""

import time
import math
from typing import List, Tuple

import numpy as np

# try to import the LP solver; if unavailable we fall back to the pure
# gradient method (the code still works, just without the LP boost).
try:
    from scipy.optimize import linprog
except Exception:  # pragma: no cover
    linprog = None

# ----------------------------------------------------------------------
# Low‑level numeric utilities
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Linear convolution via real FFT (full length, zero‑padded to a power‑of‑2)."""
    m = len(a) + len(b) - 1
    n_fft = _next_pow2(m)
    A = np.fft.rfft(a, n=n_fft)
    B = np.fft.rfft(b, n=n_fft)
    C = A * B
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


def compute_score(a: np.ndarray) -> float:
    """
    Official objective (scale‑invariant):
        2·n·max(a*a) / (sum(a)²)
    The vector `a` need not be normalised – the formula is scale‑invariant.
    """
    n = len(a)
    conv = _convolution_fft(a, a)
    max_val = float(np.max(conv))
    s = float(np.sum(a))
    if s < 1e-12:
        return float("inf")
    return 2.0 * n * max_val / (s * s)


def arcsine_initial(n: int) -> np.ndarray:
    """Discrete arcsine density, normalised to sum = 1."""
    i = np.arange(n, dtype=float)
    eps = 0.25
    vals = 1.0 / np.sqrt((i + 0.5) * (n - 0.5 - i) + eps)
    return vals / vals.sum()


def generate_shape(n: int, p: float, eps: float) -> np.ndarray:
    """
    Symmetric power‑law shape:
        a_i ∝ ((i+ε)(n‑1‑i+ε))⁻ᵖ .
    Returned vector is normalised to unit sum.
    """
    i = np.arange(n, dtype=float)
    base = ((i + eps) * (n - 1 - i + eps)) ** (-p)
    total = base.sum()
    if total == 0.0:
        return np.full(n, 1.0 / n)
    return base / total


# ----------------------------------------------------------------------
# Soft‑max gradient descent (the “polish” phase)
# ----------------------------------------------------------------------


def _softmax_weights(conv: np.ndarray, lam: float) -> np.ndarray:
    """Numerically stable soft‑max over convolution entries."""
    x = lam * conv
    mx = np.max(x)
    w = np.exp(x - mx)
    w /= w.sum()
    return w


def _gradient_from_weights(a: np.ndarray, w: np.ndarray) -> np.ndarray:
    """
    Gradient of Σ_k w_k·(a*a)_k w.r.t. a.
    Implements   g_j = 2· Σ_m a_m · w_{m+j}.
    Computed via one FFT‑based convolution.
    """
    w_rev = w[::-1]
    conv = _convolution_fft(a, w_rev)          # length = 3·n‑2
    idx = 2 * len(a) - 2                       # position of the central element
    grads = 2.0 * conv[idx - np.arange(len(a))]
    return grads


def _normalize_simplex(v: np.ndarray) -> np.ndarray:
    """Project a non‑negative vector onto the probability simplex."""
    total = v.sum()
    if total <= 0.0:
        return np.full_like(v, 1.0 / len(v))
    return v / total


def _softmax_optimize(
    init_a: np.ndarray,
    deadline: float,
    *,
    lam_start: float = 5.0,
    lam_end: float = 80.0,
    lam_ramp_iters: int = 4000,
    eta0: float = 0.6,
    max_iter: int = 200_000,
    noise_every: int = 0,
    noise_scale: float = 0.0,
) -> Tuple[np.ndarray, float]:
    """
    Exponentiated‑gradient descent on the smooth max‑convolution objective.
    Returns the best vector seen and its (non‑smooth) score.
    """
    a = _normalize_simplex(init_a.astype(float).copy())
    best_a = a.copy()
    best_sc = compute_score(a)
    n = len(a)

    for it in range(1, max_iter + 1):
        if time.time() >= deadline:
            break

        # anneal the soft‑max temperature
        lam = lam_start + (lam_end - lam_start) * min(1.0, it / lam_ramp_iters)

        conv = _convolution_fft(a, a)
        w = _softmax_weights(conv, lam)

        grad = _gradient_from_weights(a, w)

        eta = eta0 / math.sqrt(it)

        a = a * np.exp(-eta * grad)

        if noise_every and (it % noise_every == 0):
            a += np.random.randn(n) * noise_scale
            a = np.maximum(a, 0.0)

        a = _normalize_simplex(a)

        cur_sc = compute_score(a)
        if cur_sc < best_sc:
            best_sc = cur_sc
            best_a = a.copy()

        # occasional progress report (quiet after finish)
        if it % 2000 == 0:
            print(
                f"      iter={it:,d}  λ={lam:.1f}  cur={cur_sc:.6f}"
                f"  best={best_sc:.6f}"
            )

    return best_a, best_sc


# ----------------------------------------------------------------------
# Linear‑programming “mass‑inflation” step
# ----------------------------------------------------------------------


def _conv_matrix(a: np.ndarray) -> np.ndarray:
    """
    Build the (2n‑1) × n matrix *A* such that (a * x) = A @ x.
    Row *k* contains the coefficients a_i for all i where x_{k‑i}
    participates in the convolution.
    """
    n = len(a)
    rows = 2 * n - 1
    A = np.zeros((rows, n), dtype=np.float64)
    for i in range(n):
        A[i : i + n, i] = a[i]
    return A


def _lp_step(a: np.ndarray, M: float, mix: float) -> Tuple[np.ndarray, bool]:
    """
    Perform a single LP step:
      maximise    sum(g)
      subject to    (a * g) <= M   (component‑wise)
                    g >= 0

    The solution `g` is normalised to unit sum and mixed with `a`
    using the mixing coefficient `mix` (0 < mix ≤ 1).

    Returns the new vector and a flag indicating whether the LP succeeded.
    """
    if linprog is None:  # pragma: no cover
        return a, False

    n = len(a)
    A = _conv_matrix(a)
    # Slight slack prevents infeasibility caused by rounding errors.
    b = np.full(A.shape[0], M + 1e-12, dtype=np.float64)

    c = -np.ones(n, dtype=np.float64)          # maximise sum(g)  ⇒  minimise -sum(g)
    bounds = [(0.0, None)] * n

    # The “highs” method is the default and works well for dense problems.
    res = linprog(c, A_ub=A, b_ub=b, bounds=bounds, method="highs")
    if not res.success:
        return a, False

    g_raw = res.x
    total = g_raw.sum()
    if total <= 0.0:
        return a, False

    g = g_raw / total                # normalise to unit sum
    new_a = (1.0 - mix) * a + mix * g
    new_a = np.maximum(new_a, 0.0)
    new_a = _normalize_simplex(new_a)
    return new_a, True
