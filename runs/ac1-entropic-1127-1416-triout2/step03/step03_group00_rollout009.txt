def propose_candidate(seed=0, budget_s=1000):

    import modules.

    np.random.seed(seed)
    start = time.time()
    deadline = start + budget_s - 0.5

    # Determine n
    try:
        init = np.array(height_sequence_1, dtype=float)
        n = init.shape[0]
        print(f"Using existing sequence length {n}")
    except NameError:
        n = 5000 # or 5146 default.
        init = beta_sequence(2.5, 2.5, n)
        print(f"No existing sequence; using beta start length {n}")

    # Normalise to sum 1
    a = init / np.sum(init)

    # Enforce symmetry:
    a = (a + a[::-1]) / 2.0
    a = a / np.sum(a)

    # record best
    best_a = a.copy()
    best_score = compute_score(best_a)

    # Phase 1: smooth Adam optimization
    # hyperparameters
    beta1 = 0.9
    beta2 = 0.999
    eps_adam = 1e-8
    lr0 = 0.2 # base step size
    max_iters = 200000
    # we will maybe limit by deadline
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    iter = 0

    while time.time() < deadline - 200 and iter < max_iters:
        # compute convolution and max
        # compute full conv via FFT
        fft_a = np.fft.rfft(a, n=2*n-1)
        conv = np.fft.irfft(fft_a * fft_a, n=2*n-1)
        max_b = conv.max()
        score = 2.0 * n * max_b
        # update best
        if score < best_score:
            best_score = score
            best_a = a.copy()
            if iter % 5000 == 0:
                print(f"Adam iter {iter}: new best score {best_score:.6f}")

        # smoothing epsilon based on max_b, ensure not zero
        epsilon = max(max_b * 0.02, 1e-12)  # 2% smoothing

        # compute weights w = exp(conv/epsilon)
        # to avoid overflow, shift by max
        shift = conv.max()
        w = np.exp((conv - shift) / epsilon)
        w_sum = w.sum()
        if w_sum == 0 or not np.isfinite(w_sum):
            # fallback to uniform
            w = np.ones_like(conv) / conv.size
        else:
            w = w / w_sum

        # compute gradient g = 2 * correlation of w with a
        # pad a to length 2n-1
        a_pad = np.concatenate([a, np.zeros(n-1, dtype=float)])
        # compute FFT of w and a_pad
        # choose length L = next_pow2(len(w) + len(a_pad) - 1) maybe for speed
        L = 1 << ((2 * n - 1 + n - 1 - 1).bit_length())  # compute next power of two >= 3n-2
        # But to simplify use same length as conv length maybe 2n-1 for correlation using conj
        # cross-correlation via FFT: corr = ifft(fft(w) * conj(fft(a_pad)))
        W = np.fft.rfft(w, n=L)
        A = np.fft.rfft(a_pad, n=L)
        corr_full = np.fft.irfft(W * np.conj(A), n=L)
        # corr indices for lag 0..n-1 correspond to slice (n-1):(n-1+n)
        start_idx = n - 1
        g = 2.0 * corr_full[start_idx:start_idx + n]

        # Adam update
        iter += 1
        lr = lr0 * np.sqrt(1 - beta2**iter) / (1 - beta1**iter)
        m = beta1 * m + (1 - beta1) * g
        v = beta2 * v + (1 - beta2) * (g * g)

        # gradient descent step
        a = a - step
        # project onto simplex
        a = simplex_projection(a)
        # enforce symmetry
        a = (a + a[::-1]) * 0.5

    # after loop, log progress

    cur_a = best_a.copy()

    conv_cur = np.fft.irfft(fft_cur * fft_cur, n=2*n-1)
    cur_score = 2.0 * n * conv_cur.max()
    it_sa = 0
    # temperature schedule
    T_start = cur_score * 0.02
    T_end = cur_score * 1e-6
    # using exponential decay
    while time.time() < deadline:

        # propose move: choose src with probability proportional to a (mass)
        src = np.random.choice(n, p=cur_a / cur_a.sum())
        # propose target as another index
        tgt = np.random.randint(n-1)
        if tgt >= src:

        # ensure both nonzero
        if cur_a[src] == 0:

        # propose delta fraction [0, cur_a[src]]
        delta = cur_a[src] * np.random.rand()
        cand = cur_a.copy()
        cand[src] -= delta
        cand[tgt] += delta
        # enforce symmetry: also move symmetric counterpart (src_sym = n-1-src, tgt_sym)
        src_sym = n-1-src
        tgt_sym = n-1-tgt
        # apply same delta to symmetric positions

        if src_sym != src:

            cand[src_sym] -= delta
            cand[tgt_sym] += delta

        # project onto simplex (should sum remain 1)
        # (mass moved equally maintain sum)
        # compute score for candidate
        fft_c = np.fft.rfft(cand, n=2*n-1)
        conv_c = np.fft.irfft(fft_c * fft_c, n=2*n-1)
        cand_score = 2.0 * n * conv_c.max()
        # acceptance
        if cand_score < cur_score:
            accept = True
        else:
            accept_prob = math.exp(-(cand_score - cur_score) / max(T, 1e-12))
            accept = np.random.rand() < accept_prob
        if accept:
            cur_a = cand
            cur_score = cand_score
            if cand_score < best_score:
                best_score = cand_score
                best_a = cand.copy()
                print(f"SA iter {it_sa}: new best {best_score:.6f}")
        # occasional random restart
        if it_sa % 100000 == 0 and it_sa > 0:
            # reinitialize from beta distribution
            tmp = beta_sequence(np.random.uniform(0.5,5.0),
                                np.random.uniform(0.5,5.0), n)
            tmp = tmp / tmp.sum()
            tmp = (tmp + tmp[::-1]) * 0.5

            cur_score = 2.0 * n * conv_tmp.max()
        it_sa += 1

    print(f"Simulated annealing finished after {it_sa} iterations; best score {best_score:.6f}")

    best_a = best_a / best_a.sum()
    return [float(v) for v in best_a.tolist()]

"""
Hybrid optimizer for the step‑function bound.

The search works on the probability simplex (non‑negative entries that sum to 1)
and forces a_i = a_{N‑1‑i} (symmetry) which reduces the effective
dimensionality.  Two phases are performed:

1.  A smooth approximation of the “max” in the objective
    (log‑sum‑exp) is minimised with Adam.  The gradient of the smoothed
    objective is 2 · correlation(w, a) where w are the softmax‑weights of
    the convolution a∗a.  After each Adam step the vector is projected onto
    the simplex and symmetrised.

2.  A simulated‑annealing fine‑tuning that moves a random fraction of mass
    between two positions while keeping the sequence symmetric.
    Periodic random restarts from a Beta‑distribution diversify the search.

Both phases keep the best sequence encountered and return it when the
time budget expires.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Helper functions
# ----------------------------------------------------------------------


def simplex_projection(v: np.ndarray) -> np.ndarray:
    """Euclidean projection onto the probability simplex {x≥0, sum(x)=1}."""
    v = np.asarray(v, dtype=float)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all non‑positive – return the uniform distribution
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1] - 1               # zero‑based index
    theta = cssv[cond][-1] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def compute_score(a: np.ndarray) -> float:
    """
    Exact value of ``evaluate_sequence`` for a vector normalised to sum(a)==1.
    Returns 2·N·max(a∗a) where N = len(a).
    """
    N = a.shape[0]
    fft_a = np.fft.rfft(a, n=2 * N - 1)
    conv = np.fft.irfft(fft_a * fft_a, n=2 * N - 1)
    return 2.0 * N * conv.max()


def beta_sequence(alpha: float, beta: float, N: int) -> np.ndarray:
    """
    Discretised Beta(α,β) density on N points, normalised to sum = 1.
    Mid‑point rule avoids singularities at the endpoints.
    """
    x = (np.arange(N, dtype=float) + 0.5) / N
    logw = (alpha - 1.0) * np.log(x) + (beta - 1.0) * np.log(1.0 - x)
    # Clip to avoid overflow in the exponentiation
    logw = np.clip(logw, -700.0, 700.0)
    w = np.exp(logw)
    if not np.isfinite(w.sum()) or w.sum() == 0.0:
        w = np.ones_like(w)
    w /= w.sum()
    return w


# ----------------------------------------------------------------------
# Main candidate generator
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 0, budget_s: int = 1000) -> list[float]:
    """
    Search for a non‑negative height sequence that minimises
    ``evaluate_sequence``.  Up to ``budget_s`` seconds are used; the best
    sequence found before the deadline is returned.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5          # safety margin

    # --------------------------------------------------------------
    # 0.   Choose problem size and an initial point
    # --------------------------------------------------------------
    try:
        init = np.array(height_sequence_1, dtype=float)   # type: ignore[name-defined]
        N = init.shape[0]
        print(f"Using existing height_sequence_1 (N={N}).")
    except NameError:
        # fall back to a reasonably large length
        N = 5146
        init = beta_sequence(2.5, 2.5, N)
        print(f"No existing sequence – using a beta start (N={N}).")

    # normalise to a probability vector (sum == 1)
    a = init / init.sum()

    # enforce symmetry a[i] = a[N-1-i]
    a = (a + a[::-1]) * 0.5
    a = a / a.sum()

    # keep the best solution seen so far
    best_a = a.copy()
    best_score = compute_score(best_a)
    print(f"Initial score = {best_score:.6f}")

    # --------------------------------------------------------------
    # 0b.  Try a few random Beta starts and keep the best of them
    # --------------------------------------------------------------
    for _ in range(8):
        tmp = beta_sequence(np.random.uniform(0.5, 5.0),
                            np.random.uniform(0.5, 5.0), N)
        tmp = (tmp + tmp[::-1]) * 0.5
        tmp /= tmp.sum()
        sc = compute_score(tmp)
        if sc < best_score:
            best_score = sc
            best_a = tmp.copy()
            a = tmp.copy()
            print(f"Random beta restart gave better score {best_score:.6f}")

    # --------------------------------------------------------------
    # 1.   Smooth Adam optimisation of the log‑sum‑exp surrogate
    # --------------------------------------------------------------
    beta1, beta2 = 0.9, 0.999
    eps_adam = 1e-8
    lr0 = 0.5                     # base learning rate
    m = np.zeros_like(a)         # first moment
    v = np.zeros_like(a)         # second moment

    # length for the correlation FFT (next power of two ≥ 3N‑2)
    L = 1
    while L < (3 * N - 2):
        L <<= 1

    it_adam = 0
    while time.time() < deadline - 200:          # reserve ~200 s for SA
        # ----- convolution and max -----
        fft_a = np.fft.rfft(a, n=2 * N - 1)
        conv = np.fft.irfft(fft_a * fft_a, n=2 * N - 1)
        max_b = conv.max()
        cur_score = 2.0 * N * max_b

        # ----- record improvement -----
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            if it_adam % 2000 == 0:
                print(f"Adam iter {it_adam:5d}: new best {best_score:.6f}")

        # ----- smooth weights (softmax over convolution entries) -----
        eps = max(max_b * 0.02, 1e-12)                  # 2 % smoothing
        shift = conv.max()
        w = np.exp((conv - shift) / eps)               # safe, max becomes 1
        w_sum = w.sum()
        if not np.isfinite(w_sum) or w_sum == 0.0:
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        # ----- gradient of the smooth surrogate -----
        # a is length N, we need it zero‑padded to length 2N‑1
        a_pad = np.concatenate([a, np.zeros(N - 1, dtype=float)])

        # correlation(w, a) via FFT (linear correlation, lags 0..N‑1)
        W = np.fft.rfft(w, n=L)
        A = np.fft.rfft(a_pad, n=L)
        corr = np.fft.irfft(W * np.conj(A), n=L)      # length L, first (2N‑1+N‑1) are linear corr.
        grad = 2.0 * corr[:N]                         # g_j = 2 * Σ_i w_{i+j} a_i

        # ----- Adam update -----
        it_adam += 1
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad * grad)
        lr = lr0 * math.sqrt(1 - beta2 ** it_adam) / (1 - beta1 ** it_adam)
        step = lr * m / (np.sqrt(v) + eps_adam)

        a = a - step
        # re‑impose symmetry and feasibility
        a = (a + a[::-1]) * 0.5
        a = simplex_projection(a)

    print(f"Adam phase finished after {it_adam} iterations; best score so far {best_score:.6f}")

    # --------------------------------------------------------------
    # 2.   Simulated‑annealing refinement (still respecting symmetry)
    # --------------------------------------------------------------
    cur_a = best_a.copy()
    cur_score = compute_score(cur_a)

    it_sa = 0
    # exponential cooling – start with a modest fraction of the current score
    T_start = cur_score * 0.02
    T_end   = cur_score * 1e-6

    while time.time() < deadline:
        # temperature schedule
        T = max(T_end, T_start * (0.9998 ** it_sa))

        # ----- propose a mass‑transfer move -----
        src = np.random.choice(N, p=cur_a / cur_a.sum())
        tgt = np.random.randint(N - 1)
        if tgt >= src:
            tgt += 1
        if cur_a[src] <= 0.0:
            continue
        delta = cur_a[src] * np.random.rand()
        cand = cur_a.copy()
        cand[src] -= delta
        cand[tgt] += delta

        # enforce symmetry (averaging with the reversed copy)
        cand = (cand + cand[::-1]) * 0.5

        # ----- evaluate candidate -----
        cand_score = compute_score(cand)

        # ----- Metropolis acceptance -----
        if cand_score < cur_score:
            accept = True
        else:
            accept_prob = math.exp(-(cand_score - cur_score) / max(T, 1e-12))
            accept = np.random.rand() < accept_prob

        if accept:
            cur_a = cand
            cur_score = cand_score
            if cand_score < best_score:
                best_score = cand_score
                best_a = cand.copy()
                print(f"SA iter {it_sa:6d}: new best {best_score:.6f}")

        # occasional large random restart to avoid stagnation
        if it_sa % 75000 == 0 and it_sa > 0:
            tmp = beta_sequence(np.random.uniform(0.5, 5.0),
                                np.random.uniform(0.5, 5.0), N)
            tmp = (tmp + tmp[::-1]) * 0.5
            tmp /= tmp.sum()
            cur_a = tmp
            cur_score = compute_score(cur_a)

        it_sa += 1

    print(f"Simulated‑annealing finished after {it_sa} iterations; final best score = {best_score:.6f}")

    # --------------------------------------------------------------
    # 3.   Return the best sequence (already normalised to sum = 1)
    # --------------------------------------------------------------
    best_a = best_a / best_a.sum()
    return [float(v) for v in best_a.tolist()]
