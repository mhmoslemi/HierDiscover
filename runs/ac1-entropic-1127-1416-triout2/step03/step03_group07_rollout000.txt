```
def propose_candidate(budget_s=1000):
    deadline = time.time() + budget_s - 5
    best_a, best_score = None, inf
    # define possible n choices
    n_choices = [1024, 2048, 4096, 5120, 6144] # prune combos.
    restart = 0
    while time.time() < deadline:
        # Choose n randomly from choices
        n = rng.choice(n_choices)
        # Initial vector:
        if best_a is not None and restart == 0 and len(best_a)==n:
            a0 = np.array(best_a)
        else:
            # random initialization on simplex via Dirichlet
            a0 = rng.dirichlet(np.ones(n))
        # Convert to logits for softmax: z = log(a0) maybe plus noise
        with np.errstate(divide='ignore'):
            z = np.log(a0 + 1e-30)
        # Adam state
        m = np.zeros_like(z)
        v = np.zeros_like(z)
        beta1=0.9; beta2=0.999; eps=1e-8
        t=0
        while time.time() < deadline:
            # Compute a = softmax(z)
            a = np.exp(z - np.max(z))
            a /= a.sum()
            # Compute max_conv and subgradient
            max_conv, grad_a = _max_and_subgradient(a)
            score = 2.0 * n * max_conv
            if score < best_score:
                best_score = score
                best_a = a.copy()
                print(f"{time.time():.1f} restart {restart} iter {t}: new best {best_score:.6f}")
                if best_score < 1.503:
                    return best_a.tolist()
            # Compute gradient w.r.t z
            # grad_wrt_z = a * (grad_a - np.dot(grad_a, a))
            grad_z = a * (grad_a - np.dot(grad_a, a))
            t += 1
            # Adam update
            m = beta1 * m + (1 - beta1) * grad_z
            v = beta2 * v + (1 - beta2) * (grad_z**2)
            m_hat = m / (1 - beta1**t)
            v_hat = v / (1 - beta2**t)
            step_size = 0.05 / (np.sqrt(v_hat) + eps)  # adapt
            z -= step_size * m_hat
            # Periodically add jitter
            if t % 500 == 0:
                jitter = 0.001 * rng.standard_normal(n)
                z += jitter
        restart += 1
    if best_a is None:
        best_a = np.full(n, 1.0/n)
    return best_a.tolist()
```

"""
Propose a step‑function sequence that (approximately) minimizes

    2·n·max(a*a) / (sum(a)**2)

The implementation uses a *soft‑max* parametrisation of the non‑negative
vector a (so a_i = exp(z_i)/∑exp(z)), and runs an ADAM optimiser on the
parameters z.  The gradient of the objective is the sub‑gradient of the
maximum entry of the self‑convolution a*a, which we obtain with a fast
FFT based routine.  Several independent restarts with different problem
sizes n are performed until the time budget expires.  The current best
solution is printed regularly and returned as a plain Python list.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper: sub‑gradient of max(a*a)
# ----------------------------------------------------------------------
def _max_and_subgradient(a: np.ndarray) -> tuple[float, np.ndarray]:
    """
    Compute
        max_val = max_k (a*a)[k]
    and a sub‑gradient of the entry achieving the maximum.
    The gradient is the average over all active indices (within tolerance).
    """
    n = a.shape[0]
    # convolution via FFT (real‑to‑real)
    fft = np.fft.rfft(a, n=2 * n - 1)
    conv = np.fft.irfft(fft * fft, n=2 * n - 1)

    max_val = float(conv.max())

    # indices where the maximum is attained (within a relative/absolute tolerance)
    tol = max_val * 1e-12 + 1e-14
    active = np.where(conv >= max_val - tol)[0]

    grad = np.zeros_like(a)
    for k in active:
        low = max(0, k - n + 1)
        high = min(n - 1, k)
        idx = np.arange(low, high + 1, dtype=int)
        partner = k - idx
        # contribution 2*a[partner] for each i (including i=j when k even)
        grad[idx] += 2.0 * a[partner]

    if active.size > 1:
        grad /= float(active.size)

    return max_val, grad


# ----------------------------------------------------------------------
# Main optimisation routine
# ----------------------------------------------------------------------
def propose_candidate(budget_s: int = 1000) -> list[float]:
    """
    Search for a sequence that minimises the evaluation function.
    The routine respects the given time budget (seconds) and returns the
    best sequence found as a plain list of floats.
    """
    # ------------------------------------------------------------------
    # 1. Time budget & common objects
    # ------------------------------------------------------------------
    deadline = time.time() + budget_s - 5.0   # safety margin for cleanup
    rng = np.random.default_rng()

    # ------------------------------------------------------------------
    # 2. Global best tracking
    # ------------------------------------------------------------------
    best_a = None
    best_score = float("inf")

    # ------------------------------------------------------------------
    # 3. Determine the list of candidate lengths.
    #    We include the length of the historic solution (if available)
    #    and a handful of other sizes.
    # ------------------------------------------------------------------
    n_candidates = [1024, 2048, 4096, 5120, 6144, 8192]
    try:
        historic = np.asarray(height_sequence_1, dtype=float)  # type: ignore[name-defined]
        historic = np.maximum(historic, 0.0)
        if historic.sum() > 0:
            historic /= historic.sum()
            n_candidates.append(int(historic.size))
    except Exception:
        pass

    # ------------------------------------------------------------------
    # 4. Optimisation loop – many independent restarts
    # ------------------------------------------------------------------
    restart = 0
    while time.time() < deadline:
        restart += 1

        # ----------------------------------------------------------------
        # 4.1 Choose problem size and initialise parameters
        # ----------------------------------------------------------------
        n = int(rng.choice(n_candidates))
        if restart == 1 and best_a is not None and len(best_a) == n:
            a0 = np.asarray(best_a, dtype=float).copy()
        elif 'historic' in locals() and historic.size == n:
            a0 = historic.copy()
        else:
            # Dirichlet(1) = uniform on the simplex
            a0 = rng.dirichlet(np.ones(n))

        # Soft‑max parametrisation: z such that a = softmax(z)
        with np.errstate(divide="ignore"):
            z = np.log(a0 + 1e-30)   # avoid log(0)

        # ADAM state
        m = np.zeros_like(z)
        v = np.zeros_like(z)
        beta1 = 0.9
        beta2 = 0.999
        eps = 1e-8
        lr = 0.05                 # base learning rate (global)
        it = 0

        # ----------------------------------------------------------------
        # 4.2 Inner optimisation on this restart
        # ----------------------------------------------------------------
        while time.time() < deadline:
            # Compute current a from logits
            exp_z = np.exp(z - np.max(z))      # stabilise exponentials
            a = exp_z / exp_z.sum()

            # Gradient of the objective w.r.t. a (sub‑gradient of max conv)
            max_conv, grad_a = _max_and_subgradient(a)

            # Objective for the simplex (sum(a)=1) is 2·n·max_conv
            cur_score = 2.0 * n * max_conv

            # Record global best
            if cur_score < best_score:
                best_score = cur_score
                best_a = a.copy()
                print(
                    f"[{time.time():.1f}] restart {restart}, it {it}: "
                    f"new best = {best_score:.6f} (n={n})"
                )
                if best_score < 1.503:
                    # Target reached – finish early
                    return best_a.tolist()

            # ------------------------------------------------------------
            # 4.3 Convert gradient to the z‑space (soft‑max Jacobian)
            # ------------------------------------------------------------
            # dL/dz_i = a_i * ( dL/da_i - sum_j a_j * dL/da_j )
            grad_z = a * (grad_a - np.dot(grad_a, a))

            # ------------------------------------------------------------
            # 4.4 ADAM update
            # ------------------------------------------------------------
            it += 1
            m = beta1 * m + (1.0 - beta1) * grad_z
            v = beta2 * v + (1.0 - beta2) * (grad_z ** 2)
            m_hat = m / (1.0 - beta1 ** it)
            v_hat = v / (1.0 - beta2 ** it)
            step = lr * m_hat / (np.sqrt(v_hat) + eps)
            z -= step

            # ------------------------------------------------------------
            # 4.5 Occasional stochastic jitter to escape plateaus
            # ------------------------------------------------------------
            if it % 500 == 0:
                jitter = 0.001 * rng.standard_normal(n)
                z += jitter

            # ------------------------------------------------------------
            # 4.6 Early termination of this restart if progress stalls
            # ------------------------------------------------------------
            if it % 5000 == 0:
                # Rough stall detection: if the last 5k iterations did not
                # improve the local best by >1e-6, break out to a fresh restart.
                recent_score = 2.0 * n * _max_and_subgradient(a)[0]
                if abs(recent_score - cur_score) < 1e-6:
                    break

        # end of inner while
    # end of outer while

    # ------------------------------------------------------------------
    # 5. Return the best sequence found (uniform fallback if nothing)
    # ------------------------------------------------------------------
    if best_a is None:
        # fallback: uniform distribution of length 2048
        n = 2048
        best_a = np.full(n, 1.0 / n)

    return [float(v) for v in best_a.tolist()]
