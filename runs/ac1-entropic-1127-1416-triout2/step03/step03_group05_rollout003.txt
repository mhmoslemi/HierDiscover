def _smooth_adam_optimize(init_a, deadline, restart_id, target=1.503):
    n = init_a.shape[0]
    # precompute lengths for FFTs
    fft_len_conv = 1 << ((2*n - 2).bit_length())
    fft_len_grad = 1 << ((3*n - 2).bit_length())
    # init logits
    b = np.log(init_a + 1e-12)
    m = np.zeros_like(b)
    v = np.zeros_like(b)
    beta = 0.5
    beta_max = 100.0
    beta_step = 0.01
    best_a = init_a.copy()
    # compute initial score
    conv = np.fft.irfft(np.fft.rfft(init_a, fft_len_conv) * np.fft.rfft(init_a, fft_len_conv), n=fft_len_conv)[:2*n-1]
    max_conv = conv.max()
    best_score = 2.0 * n * max_conv
    stagnation = 0
    it = 0
    beta = beta # start linear

    while time.time() < deadline:
        it += 1

        # update beta
        if beta < beta_max:
            beta = min(beta_max, beta + beta_step)

        # a from logits
        b_max = np.max(b)
        a = np.exp(b - b_max)
        a = a / a.sum()

        # compute conv and max_conv via FFT
        fft_a = np.fft.rfft(a, n=fft_len_conv)
        conv = np.fft.irfft(fft_a * fft_a, n=fft_len_conv)[:2*n-1]
        max_conv = conv.max()

        # compute probabilities for smoothed max
        # w = np.exp(beta*(conv - max_conv)) -> produce at least 1.0 for max.
        # subtract max to avoid overflow.
        w = np.exp(beta * (conv - max_conv))
        sum_w = w.sum()
        probs = w / sum_w

        # compute gradient wrt a (smooth)
        a_rev = a[::-1]
        fft_rev = np.fft.rfft(a_rev, n=fft_len_grad)
        fft_probs = np.fft.rfft(probs, n=fft_len_grad)
        conv_rev = np.fft.irfft(fft_rev * fft_probs, n=fft_len_grad)[: (n + (2*n-1) -1) ]  # length 3n-2
        # slice
        start = n - 1
        grad_a = 2.0 * conv_rev[start:start + n]   # gradient of smooth max

        # gradient of full objective (scale by 2*n)
        grad_obj = 2.0 * n * grad_a

        # transform to logits gradient (softmax Jacobian)
        avg = np.dot(a, grad_obj)
        grad_b = a * (grad_obj - avg)

        # Adam update
        beta1, beta2 = 0.9, 0.999
        eps = 1e-8
        m = beta1 * m + (1.0 - beta1) * grad_b
        v = beta2 * v + (1.0 - beta2) * (grad_b ** 2)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        lr = 0.5 / np.sqrt(it)  # similar

        b -= lr * m_hat / (np.sqrt(v_hat) + eps)

        # compute current true score
        cur_score = 2.0 * n * max_conv  # sum a approx 1

        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            stagnation = 0
            print(f"Restart {restart_id:02d} iter {it:6d}: best = {best_score:.6f} beta={beta:.2f}")
        else:
            stagnation += 1

        if best_score < target:
            break

        if stagnation > 5000:
            # jitter
            b += 0.01 * np.random.randn(n)

"""
Smooth‐max Adam optimizer for the aperiodic auto‑convolution bound.

The objective is

    2·n·max(conv(a,a)) / (sum(a)^2) ,

with a_i ≥ 0.  Because the formula is scale‑invariant we restrict
∑a_i = 1 and optimise a over the probability simplex.
The maximum of the convolution is replaced by a smooth log‑sum‑exp
approximation (temperature β).  During optimisation β is increased
linearly from 0.5 up to 100, turning the smooth objective gradually
into the true max‑objective.  The optimisation is performed on the
soft‑max logits of a using Adam with a diminishing learning‑rate.
Several short restarts with different initialisations (previous best,
uniform, Dirichlet, geometric, triangular) are performed until the
time budget (default 1000 s) is exhausted.  The best sequence seen
is returned as a plain Python list of floats.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper functions – all are top‑level, no closures
# ----------------------------------------------------------------------


def _full_convolution(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Aperiodic convolution a * a (full length 2·n−1) using FFT."""
    fft_a = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(fft_a * fft_a, n=fft_len)[: 2 * a.shape[0] - 1]
    return conv


def _smooth_gradient(a: np.ndarray,
                     probs: np.ndarray,
                     fft_len_grad: int) -> np.ndarray:
    """
    Gradient of  Σ_k probs_k·conv_k(a)  w.r.t. a.
    It equals 2·(a_rev ⊛ probs)[n−1 : n−1+n] where ⊛ is aperiodic convolution.
    """
    n = a.shape[0]
    a_rev = a[::-1]
    fft_rev = np.fft.rfft(a_rev, n=fft_len_grad)
    fft_probs = np.fft.rfft(probs, n=fft_len_grad)
    conv_full = np.fft.irfft(fft_rev * fft_probs,
                             n=fft_len_grad)[: n + (2 * n - 1) - 1]   # length 3·n−2
    start = n - 1
    grad = 2.0 * conv_full[start:start + n]   # factor 2 from the derivative of a·a
    return grad


def _initial_sequence(kind: str,
                      n: int,
                      rng: np.random.Generator,
                      prev_seq: np.ndarray | None) -> np.ndarray:
    """Return a feasible probability vector (Σa=1, a_i≥0) of length n."""
    if kind == "prev" and prev_seq is not None:
        seq = prev_seq.copy()
        if seq.shape[0] > n:
            seq = seq[:n]
        elif seq.shape[0] < n:
            seq = np.concatenate([seq,
                                  np.zeros(n - seq.shape[0], dtype=seq.dtype)])
        seq = np.maximum(seq, 0.0)
        total = seq.sum()
        if total == 0.0:
            return np.full(n, 1.0 / n)
        return seq / total

    if kind == "uniform":
        return np.full(n, 1.0 / n)

    if kind == "random_dirichlet":
        # α < 1 yields sparse vectors
        alpha = 0.5
        sample = rng.gamma(alpha, 1.0, size=n)
        return sample / sample.sum()

    if kind == "geometric":
        r = 0.98 + 0.02 * rng.random()
        left = r ** np.arange(n)
        right = r ** np.arange(n - 1, -1, -1)
        seq = left + right
        return seq / seq.sum()

    if kind == "triangular":
        # symmetric triangle: 1,2,...,mid,...,2,1
        half = np.arange(1, n // 2 + 1)
        if n % 2 == 0:
            seq = np.concatenate([half, half[::-1]])
        else:
            seq = np.concatenate([half, [half[-1] + 1], half[::-1]])
        return seq.astype(float) / seq.sum()

    # fallback
    return np.full(n, 1.0 / n)


def _smooth_adam_optimize(init_a: np.ndarray,
                          deadline: float,
                          restart_id: int,
                          target: float = 1.503) -> tuple[np.ndarray, float]:
    """
    Adam optimisation on the soft‑max logits of a probability vector.
    Uses a temperature β that is increased linearly from 0.5 to 100,
    turning the smooth log‑sum‑exp of the convolution into the true
    max‑convolution.
    Returns the best sequence (Σa=1) and its exact objective value.
    """
    n = init_a.shape[0]

    # FFT lengths (next power of two for speed)
    fft_len_conv = 1 << ((2 * n - 2).bit_length())
    fft_len_grad = 1 << ((3 * n - 2).bit_length())

    # initialise logits (soft‑max parametrisation)
    eps_init = 1e-12
    logits = np.log(init_a + eps_init)

    # Adam moments
    m = np.zeros_like(logits)
    v = np.zeros_like(logits)
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8

    # temperature schedule
    beta_min = 0.5
    beta_max = 100.0
    beta = beta_min
    beta_step = (beta_max - beta_min) / 10000.0   # reach β≈100 after ~10 k iters

    # initial score (exact max‑convolution)
    conv0 = _full_convolution(init_a, fft_len_conv)
    best_score = 2.0 * n * conv0.max()
    best_a = init_a.copy()

    stagnation = 0
    it = 0

    while time.time() < deadline:
        it += 1

        # --------------------------------------------------------------
        # 1) update temperature β
        # --------------------------------------------------------------
        if beta < beta_max:
            beta = min(beta_max, beta + beta_step)

        # --------------------------------------------------------------
        # 2) obtain probability vector from logits
        # --------------------------------------------------------------
        log_max = np.max(logits)
        a = np.exp(logits - log_max)
        a = a / a.sum()                     # Σa = 1

        # --------------------------------------------------------------
        # 3) full aperiodic convolution a*a
        # --------------------------------------------------------------
        fft_a = np.fft.rfft(a, n=fft_len_conv)
        conv = np.fft.irfft(fft_a * fft_a,
                            n=fft_len_conv)[: 2 * n - 1]

        max_conv = conv.max()

        # --------------------------------------------------------------
        # 4) smooth max via log‑sum‑exp, obtain probability weights
        # --------------------------------------------------------------
        # w_k = exp(β·(conv_k - max_conv))  – numerically stable
        w = np.exp(beta * (conv - max_conv))
        sum_w = w.sum()
        probs = w / sum_w                     # Σ probs = 1

        # --------------------------------------------------------------
        # 5) gradient of smooth max (w.r.t. a)
        # --------------------------------------------------------------
        grad_a = _smooth_gradient(a, probs, fft_len_grad)   # shape (n,)

        # --------------------------------------------------------------
        # 6) gradient w.r.t. logits (soft‑max Jacobian)
        # --------------------------------------------------------------
        grad_obj = 2.0 * n * grad_a          # because exact obj = 2·n·max_conv
        avg = np.dot(a, grad_obj)
        grad_logits = a * (grad_obj - avg)

        # --------------------------------------------------------------
        # 7) Adam update on logits
        # --------------------------------------------------------------
        m = beta1 * m + (1.0 - beta1) * grad_logits
        v = beta2 * v + (1.0 - beta2) * (grad_logits ** 2)

        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        # diminishing learning‑rate
        lr = 0.5 / np.sqrt(it)
        logits -= lr * m_hat / (np.sqrt(v_hat) + eps)

        # --------------------------------------------------------------
        # 8) keep best exact solution seen
        # --------------------------------------------------------------
        cur_score = 2.0 * n * max_conv          # exact objective (sum≈1)
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            stagnation = 0
            print(f"Restart {restart_id:02d} iter {it:6d}: "
                  f"best = {best_score:.6f}  β={beta:.2f}")
        else:
            stagnation += 1

        # --------------------------------------------------------------
        # 9) simple termination / jitter logic
        # --------------------------------------------------------------
        if best_score < target:
            break
        if stagnation > 5000:
            # random jitter in log‑space to escape plateaus
            logits += 0.01 * np.random.randn(n)
            stagnation = 0

    return best_a, best_score

    """

    deadline = time.time() + budget_s - 3.0   # small safety margin

    # ------------------------------------------------------------------
    # Problem size – try to reuse the length of a previously found solution
    # ------------------------------------------------------------------
    try:

    except Exception:

    best_overall_a = None

    restart_id = 0

    strategies = ["prev", "uniform", "random_dirichlet", "geometric", "triangular"]

    # ------------------------------------------------------------------
    # Multiple short Adam restarts – each gets a slice of the remaining time
    # ------------------------------------------------------------------
    while time.time() < deadline - 8.0:          # reserve a few seconds for cleanup
