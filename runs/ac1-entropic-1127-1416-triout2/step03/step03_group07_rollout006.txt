def gradient_smooth(a, gamma):
    n = len(a)
    # compute convolution b = a*a (full length 2n-1)
    b = np.convolve(a, a)
    # compute softmax weights over b
    maxb = b.max()
    # to avoid overflow, subtract maxb
    exp = np.exp(gamma*(b - maxb))
    w = exp / exp.sum()  # shape 2n-1
    # reverse w
    w_rev = w[::-1]
    # compute conv = np.convolve(w_rev, a) -> length (2n-1)+n-1 = 3n-2
    conv = np.convolve(w_rev, a)
    # slice

    # conv size = len(w_rev)+n-1 = (2n-1)+(n-1)=3n-2, so index valid.

    return g

Gradient‑descent optimiser for 2·n·max(a*a)/(sum(a)**2).

The algorithm treats the L‑∞ objective as a smooth log‑sum‑exp
approximation:
    f_γ(a) = (1/γ)·log Σ_k exp(γ·(a*a)[k]).

convolution, which we evaluate with FFTs.  An Adam‑style update is

parameter γ is increased gradually so that the optimisation first
sees a softened landscape and finally concentrates on the true
maximum.  Several random restarts are performed until the time budget
expires; the best sequence encountered is returned.

The code is deliberately different from the previously supplied
mirror‑descent implementation and explores a smoother optimisation

import time

# Helper functions

def _project_to_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the probability simplex.
    Returns a vector w such that w >= 0 and sum(w) == 1.
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # numeric safety – return the uniform distribution
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1] - 1
    theta = cssv[cond][-1] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def _fft_convolve(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Convolution of two real vectors using FFT (returns the linear convolve)."""
    n = len(x) + len(y) - 1
    N = 1 << ((n - 1).bit_length())          # next power of two
    X = np.fft.rfft(x, N)
    Y = np.fft.rfft(y, N)
    return np.fft.irfft(X * Y, N)[:n]


def _smooth_obj_and_grad(a: np.ndarray, gamma: float):
    """
    Returns (max_entry_of_a*a, gradient_of smoothed L∞ objective).

    The smoothed objective is
        f_γ(a) = (1/gamma) * log Σ_k exp(gamma * (a*a)[k])
    Its gradient can be written as
        g_i = 2 * Σ_k softmax_k * a_{k-i},
    where softmax_k = exp(gamma*(b_k-max(b))) / Σ_j exp(gamma*(b_j-max(b)))
    and b = a*a (full convolution).  The sum is realised via a single
    convolution w * rev(a).
    """
    n = a.shape[0]

    # ---------- 1) a * a (full convolution) ----------
    L = 2 * n - 1
    N1 = 1 << ((L - 1).bit_length())        # power‑of‑two for FFT
    A = np.fft.rfft(a, N1)
    conv_full = np.fft.irfft(A * A, N1)[:L]   # length 2n‑1

    max_b = conv_full.max()

    # ---------- 2) soft‑max weights ----------
    # shift by max_b for numerical stability
    exp_w = np.exp(gamma * (conv_full - max_b))
    w = exp_w / exp_w.sum()                 # shape (2n‑1,)

    # ---------- 3) gradient ----------
    # g_i = 2 * Σ_k w_k a_{k-i}  (indices outside [0,n-1] are ignored)
    # This is a convolution of w with the reversed a.
    a_rev = a[::-1]
    grad_full = _fft_convolve(w, a_rev)     # length = (2n-1) + n-1 = 3n-2
    start = n - 1
    grad = 2.0 * grad_full[start:start + n]  # slice of length n

    return max_b, grad
