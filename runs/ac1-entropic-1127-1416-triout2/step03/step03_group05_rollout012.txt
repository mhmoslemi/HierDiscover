def _adam_optimize_smooth(init_a, deadline, restart_id):
    n = len(init_a)
    L = 1 << ((3*n-2).bit_length())
    eps = 1e-12
    b = np.log(init_a + eps)    # logits

    m = np.zeros_like(b)
    v = np.zeros_like(b)
    beta1, beta2, epsilon_adam = 0.9, 0.999, 1e-8

    best_score = float('inf')
    best_a = None

    it = 0

    while time.time() < deadline:

        it += 1

        # compute a
        bmax = np.max(b)
        a = np.exp(b - bmax)
        a = a / a.sum()

        # compute conv a*a
        fft_len = 1 << ((2*n-2).bit_length())
        # Actually we need L for grad conv due to w conv length.
        # We'll compute conv via rfft of length L as well; we can zero-pad further.
        # Use rfft with L, compute conv = irfft(rfft(a, L)**2)[:2*n-1].

        a_fft = np.fft.rfft(a, n=L)
        conv_full = np.fft.irfft(a_fft * a_fft, n=L)[:2*n-1]

        max_val = conv_full.max()
        # Determine lambda schedule based on iteration
        # Example: lambda = min(200, 0.5 * it) truncated? Or logistic.

        # Use a piecewise schedule: early phases lambda small (e.g., 5) for smooth gradient; after >2000 it grows.

        if it < 2000:
            lam = 5.0
        elif it < 8000:
            lam = 10.0
        elif it < 20000:
            lam = 20.0
        else:
            lam = 50.0

        # Compute smoothed objective
        # compute w: softmax(lambda * conv_full)
        temp = lam * conv_full
        tmax = temp.max()
        exp_vals = np.exp(temp - tmax)
        sum_exp = exp_vals.sum()
        w = exp_vals / sum_exp   # shape 2n-1

        # Weighted expectation J = np.dot(w, conv_full)
        J = np.dot(w, conv_full)
        obj = 2.0 * n * J

        # compute gradient wrt a
        # E_dm = 2 * (conv_a_w) where conv_a_w = convolution of a with w (full)
        # We'll compute via FFT: w_pad = zeros(L); w_pad[:2*n-1] = w
        w_pad = np.zeros(L)
        w_pad[:2*n-1] = w
        # Compute conv_a_w via irfft(a_fft * np.fft.rfft(w_pad, n=L))
        w_fft = np.fft.rfft(w_pad, n=L)
        conv_a_w_full = np.fft.irfft(a_fft * w_fft, n=L)[:3*n-2]   # length 3n-2

        # Extract conv1, conv2 as before
        # conv2 = conv_a_w_full[0:n]
        conv2 = conv_a_w_full[0:n]
        # conv1 = conv_a_w_full[n-1:2*n-1][::-1]
        conv1 = conv_a_w_full[n-1:2*n-1][::-1]
        E_dm = conv1 + conv2   # length n
        E_dm = 2.0 * E_dm

        # Compute p = w * conv_full
        p = w * conv_full
        # compute conv_a_p_full similarly:
        p_pad = np.zeros(L)
        p_pad[:2*n-1] = p
        p_fft = np.fft.rfft(p_pad, n=L)
        conv_a_p_full = np.fft.irfft(a_fft * p_fft, n=L)[:3*n-2]

        conv2p = conv_a_p_full[0:n]
        conv1p = conv_a_p_full[n-1:2*n-1][::-1]
        E_m_dm = conv1p + conv2p
        E_m_dm = 2.0 * E_m_dm

        # Cov = E_m_dm - J * E_dm
        Cov = E_m_dm - J * E_dm

        # gradient of objective w.r.t a:
        grad_a = 2.0 * n * (E_dm + lam * Cov)  # same as earlier

        # Convert to logit gradient
        avg = np.dot(a, grad_a)
        grad_b = a * (grad_a - avg)

        # Adam updates
        m = beta1 * m + (1.0 - beta1) * grad_b
        v = beta2 * v + (1.0 - beta2) * (grad_b ** 2)

        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        lr = 0.4 / np.sqrt(it)

        b -= lr * m_hat / (np.sqrt(v_hat) + epsilon_adam)

        # keep best
        if obj < best_score:
            best_score = obj
            best_a = a.copy()
            print(f"Restart {restart_id:02d} iter {it:6d} best = {best_score:.6f}")

        # optional early break if best < target
        if best_score < 1.5:  # slight better than target
            break

    return best_a, best_score

"""
search for a low‑score step‑function sequence.

The algorithm minimises the exact objective
    2·n·max(conv(a,a))
while enforcing a_i ≥ 0 and Σa_i = 1 by optimising the
log‑softmax (logits) of a probability vector.

Instead of the non‑smooth max we use a *smooth* approximation:
    w_k = softmax(λ·conv_k)  (λ is a temperature that grows during the run)
and minimise the expected value  Σ w_k·conv_k.
The gradient of this smooth objective can be written in closed form
using only a few FFT‑based convolutions, which makes each iteration
≈ O(n·log n).

A continuation scheme (small λ → large λ) together with Adam updates
and occasional random jitter helps to escape plateaus.
Multiple short restarts with different initialisations are performed
until the time budget (≈ 1000 s) runs out.  The best sequence found
is returned as a plain Python list – exactly what the evaluation
harness expects.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Return the smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _initial_sequence(kind: str, n: int, rng: np.random.Generator,
                      prev: np.ndarray | None) -> np.ndarray:
    """Generate a feasible start vector (sum = 1, non‑negative)."""
    if kind == "prev" and prev is not None:
        seq = prev.copy()
        if seq.shape[0] != n:
            if seq.shape[0] > n:
                seq = seq[:n]
            else:
                seq = np.concatenate([seq,
                                      np.zeros(n - seq.shape[0],
                                               dtype=seq.dtype)])
        seq = np.maximum(seq, 0.0)
        total = seq.sum()
        if total == 0.0:
            return np.full(n, 1.0 / n)
        return seq / total

    if kind == "uniform":
        return np.full(n, 1.0 / n)

    if kind == "random_dirichlet":
        # α < 1 yields sparse vectors – good for exploration
        alpha = 0.5
        sample = rng.gamma(alpha, 1.0, size=n)
        return sample / sample.sum()

    if kind == "geometric":
        r = 0.98 + 0.02 * rng.random()
        left = r ** np.arange(n)
        right = r ** np.arange(n - 1, -1, -1)
        seq = left + right
        return seq / seq.sum()

    # fallback
    return np.full(n, 1.0 / n)


# ----------------------------------------------------------------------
# Adam optimiser with a smooth‑max objective
# ----------------------------------------------------------------------


def _adam_optimize_smooth(init_a: np.ndarray,
                          deadline: float,
                          restart_id: int,
                          target: float = 1.503) -> tuple[np.ndarray, float]:
    """
    Adam optimisation of the soft‑max‑parameterised sequence.

    Returns the best sequence (as a probability vector, Σa=1) and the
    corresponding exact objective value 2·n·max(conv(a,a)).
    """
    n = init_a.shape[0]
    # length needed for convolutions a↔w (a length n, w length 2n‑1)
    L = _next_pow2(3 * n - 2)

    # initialise logits (soft‑max parameters)
    eps_init = 1e-12
    b = np.log(init_a + eps_init)

    # Adam state
    m = np.zeros_like(b)
    v = np.zeros_like(b)
    beta1, beta2 = 0.9, 0.999
    eps_adam = 1e-8

    # schedule for the smooth‑max temperature λ
    lam_start = 1.0          # very smooth at the beginning
    lam_end = 200.0          # close to the true max at the end
    lam_ramp_iters = 20000   # after this λ will be at its maximum

    best_score = float('inf')
    best_a = None
    stagnation = 0
    it = 0

    while time.time() < deadline:
        it += 1

        # --------------------------------------------------------------
        # 1) obtain the probability vector from the logits
        # --------------------------------------------------------------
        b_max = np.max(b)
        a = np.exp(b - b_max)
        a /= a.sum()                     # Σa = 1, a_i ≥ 0

        # --------------------------------------------------------------
        # 2) a * a convolution (full), needed for the exact objective
        # --------------------------------------------------------------
        a_fft = np.fft.rfft(a, n=L)
        conv_full = np.fft.irfft(a_fft * a_fft, n=L)[: 2 * n - 1]
        max_conv = conv_full.max()
        cur_score = 2.0 * n * max_conv

        # --------------------------------------------------------------
        # 3) smooth‑max parameters
        # --------------------------------------------------------------
        lam = lam_start + (lam_end - lam_start) * min(1.0, it / lam_ramp_iters)

        temp = lam * conv_full
        tmax = temp.max()
        exp_vals = np.exp(temp - tmax)          # shift for stability
        sum_exp = exp_vals.sum()
        w = exp_vals / sum_exp                  # soft‑max weights, length 2n‑1

        # expectation of the convolution under w
        J = np.dot(w, conv_full)                # scalar

        # --------------------------------------------------------------
        # 4) gradients w.r.t. a (using the analytical formula)
        # --------------------------------------------------------------
        # (a) compute w‑convolution term: E[ d conv ]
        w_pad = np.zeros(L)
        w_pad[: 2 * n - 1] = w
        w_fft = np.fft.rfft(w_pad, n=L)
        conv_a_w_full = np.fft.irfft(a_fft * w_fft, n=L)[: 3 * n - 2]

        # split into the two cross‑correlation parts
        conv2 = conv_a_w_full[:n]                                 # Σ_j a_j w_{i‑j}
        conv1 = conv_a_w_full[n - 1 : 2 * n - 1][::-1]            # Σ_j a_j w_{i+j}
        E_dm = 2.0 * (conv1 + conv2)                               # shape (n,)

        # (b) compute weighted‑by‑conv term: E[ conv·d conv ]
        p = w * conv_full                                          # pointwise product, length 2n‑1
        p_pad = np.zeros(L)
        p_pad[: 2 * n - 1] = p
        p_fft = np.fft.rfft(p_pad, n=L)
        conv_a_p_full = np.fft.irfft(a_fft * p_fft, n=L)[: 3 * n - 2]

        conv2p = conv_a_p_full[:n]
        conv1p = conv_a_p_full[n - 1 : 2 * n - 1][::-1]
        E_m_dm = 2.0 * (conv1p + conv2p)                           # shape (n,)

        # (c) covariance term
        Cov = E_m_dm - J * E_dm

        # (d) full gradient w.r.t. a
        grad_a = 2.0 * n * (E_dm + lam * Cov)

        # --------------------------------------------------------------
        # 5) back‑propagate through the soft‑max (logits) map
        # --------------------------------------------------------------
        avg = np.dot(a, grad_a)
        grad_b = a * (grad_a - avg)

        # --------------------------------------------------------------
        # 6) Adam update on the logits
        # --------------------------------------------------------------
        m = beta1 * m + (1.0 - beta1) * grad_b
        v = beta2 * v + (1.0 - beta2) * (grad_b ** 2)

        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        lr = 0.4 / np.sqrt(it)          # diminishing learning rate
        b -= lr * m_hat / (np.sqrt(v_hat) + eps_adam)

        # --------------------------------------------------------------
        # 7) bookkeeping
        # --------------------------------------------------------------
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            stagnation = 0
            print(f"Restart {restart_id:02d} iter {it:6d}: best = {best_score:.6f}")
        else:
            stagnation += 1
            if stagnation > 6000:
                # escape a possible plateau
                b += 0.01 * np.random.randn(n)
                stagnation = 0

        if best_score < target:
            break

    # fall back to the start point if nothing improved
    if best_a is None:
        best_a = init_a.copy()
        best_score = 2.0 * n * np.max(np.fft.irfft(np.fft.rfft(best_a, n=L) ** 2,
                                                    n=L)[: 2 * n - 1])

    return best_a, best_score

    """
    Search for a step‑function sequence with a low evaluate_sequence score.
    The function runs for at most ``budget_s`` seconds and returns the
    best sequence (list of floats) discovered.
    """
    rng = np.random.default_rng(123456)           # reproducible randomness
    deadline = time.time() + budget_s - 5.0       # leave a small safety margin

    # ------------------------------------------------------------------
    # Determine the problem size (use the previous best if available)
    # ------------------------------------------------------------------
    try:
        previous = np.asarray(height_sequence_1, dtype=float)   # type: ignore[name-defined]
        n = previous.shape[0]
    except Exception:
        previous = None
        n = 5146                                   # default length used earlier

    # ------------------------------------------------------------------
    # Global bookkeeping
    # ------------------------------------------------------------------
    best_overall_a = None
    best_overall_score = float('inf')
    restart_id = 0

    init_strategies = ["prev", "uniform", "random_dirichlet", "geometric"]

    # ------------------------------------------------------------------
    # Multiple short Adam restarts – each gets a slice of the remaining time
    # ------------------------------------------------------------------
    while time.time() < deadline - 5.0:
        kind = init_strategies[restart_id % len(init_strategies)]
        if kind == "prev" and previous is None:
            kind = "uniform"

        a0 = _initial_sequence(kind, n, rng, previous)

        # Allocate at most 30 s or half of the remaining budget to this restart
        time_left = deadline - time.time()
        this_deadline = time.time() + min(30.0, 0.5 * time_left)

        a_opt, score_opt = _adam_optimize_smooth(a0, this_deadline, restart_id)

        if score_opt < best_overall_score:
            best_overall_score = score_opt
            best_overall_a = a_opt.copy()
            print(f"Global best after restart {restart_id:02d}: {best_overall_score:.6f}")

        if best_overall_score < 1.503:
            print(f"Target reached (score={best_overall_score:.6f}) after {restart_id+1} restarts.")
            break

        restart_id += 1

    # ------------------------------------------------------------------
    # Fallback – guarantees a valid return value
    # ------------------------------------------------------------------
    if best_overall_a is None:
        best_overall_a = np.full(n, 1.0 / n)

    # Convert to a plain Python list (the evaluator expects this)
    return [float(v) for v in best_overall_a.tolist()]
