"""
Propose a low‑score step‑function.

The new search fixes a bug in the original smoothed‑max gradient
(computed as ∇(log‑sum‑exp(convolution))) and runs a longer projected‑Adam
optimisation on several random starts for moderate sequence lengths.
The optimisation uses a homotopy on the soft‑max temperature, a proper
gradient (no spurious subtraction) and a final stochastic mass‑transfer
hill‑climber.  With a reasonable time budget (≤1000 s) this approach
typically drives the evaluated score well below the previous 1.518 …
"""

import time
import numpy as np


# ----------------------------------------------------------------------
# Basic utilities
# ----------------------------------------------------------------------


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """ Euclidean projection of a vector onto the probability simplex. """
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / v.shape[0])

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, v.shape[0] + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.where(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def next_pow2(x: int) -> int:
    """ Smallest power of two ≥ x. """
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """ Full convolution a * b using a real FFT (fast). """
    m = len(a) + len(b) - 1
    n_fft = next_pow2(m)
    A = np.fft.rfft(a, n=n_fft)
    B = np.fft.rfft(b, n=n_fft)
    C = A * B
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


def compute_score(a: np.ndarray) -> float:
    """Fast replica of `evaluate_sequence` for a vector with ∑a = 1."""
    n = a.shape[0]
    conv = convolution_fft(a, a)
    return 2.0 * n * float(np.max(conv))


# ----------------------------------------------------------------------
# Smoothed max (log‑sum‑exp) gradient – the fixed version
# ----------------------------------------------------------------------


def smoothed_gradient(a: np.ndarray, alpha: float):
    """
    Gradient of the soft‑max (log‑sum‑exp) approximation of max(convolution):
        f_α(a) = (1/α)·log Σ_k exp(α·c_k) ,   c = a*a

    Returns (grad, conv) where:
      grad – ∂f_α/∂a (same shape as a)
      conv – the full convolution a*a (for diagnostics)
    """
    n = a.shape[0]

    # self‑convolution
    conv = convolution_fft(a, a)                    # length 2n‑1

    # soft‑max weights (stable)
    maxc = np.max(conv)
    exp_term = np.exp(alpha * (conv - maxc))
    w = exp_term / np.sum(exp_term)                # Σ w_i = 1

    # Gradient:  ∇f = Σ_i w_i ∇c_i = 2 * (w * rev(a))_{i+n‑1}
    a_rev = a[::-1]
    conv2 = convolution_fft(w, a_rev)              # length (2n‑1)+n‑1 = 3n‑2
    start = n - 1
    grad = 2.0 * conv2[start:start + n]

    return grad, conv


# ----------------------------------------------------------------------
# Projected Adam optimiser on the smoothed objective
# ----------------------------------------------------------------------


def adam_optimize(
    init_a: np.ndarray,
    time_deadline: float,
    max_iter: int = 200_000,
    eta: float = 0.02,
    beta1: float = 0.9,
    beta2: float = 0.999,
    eps: float = 1e-8,
    alpha_start: float = 1.0,
    alpha_end: float = 5000.0,
    noise_every: int = 1000,
    noise_scale: float = 1e-4,
):
    """
    Projected Adam optimiser on the smoothed max‑convolution objective.
    Returns the best vector (as ndarray) encountered and its *true* score.
    """
    a = project_to_simplex(init_a.astype(float).copy())
    m = np.zeros_like(a)
    v = np.zeros_like(a)

    best_a = a.copy()
    best_score = compute_score(a)

    n = len(a)

    for t in range(1, max_iter + 1):
        if time.time() >= time_deadline:
            break

        # homotopy on the smoothing parameter
        alpha = alpha_start + (alpha_end - alpha_start) * min(1.0, t / max_iter)

        grad, _ = smoothed_gradient(a, alpha)

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)

        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)

        step = eta * m_hat / (np.sqrt(v_hat) + eps)
        a_new = a - step

        # occasional isotropic noise to escape plateaus
        if noise_every and (t % noise_every == 0):
            a_new += np.random.randn(n) * noise_scale

        a_new = project_to_simplex(a_new)

        cur_score = compute_score(a_new)
        if cur_score < best_score:
            best_score = cur_score
            best_a = a_new.copy()

        a = a_new

        if t % 5000 == 0:
            print(
                f"    Adam iter {t:,d} – α={alpha:.1f} – cur {cur_score:.6f} – best {best_score:.6f}"
            )

    return best_a, best_score


# ----------------------------------------------------------------------
# Pure permutation / mass‑transfer hill climbing
# ----------------------------------------------------------------------


def hill_climb_permutation(seq, time_deadline, max_steps=200_000):
    """
    Stochastic hill‑climb that swaps entries or transfers a fraction of mass.
    Operates directly on the *true* (unsmoothed) objective.
    """
    a = np.array(seq, dtype=float)
    n = a.shape[0]

    best_a = a.copy()
    best_score = compute_score(best_a)

    step_scale = 0.15  # start with a modest fraction of mass

    while max_steps > 0 and time.time() < time_deadline:
        move_type = np.random.rand()
        cand = best_a.copy()

        if move_type < 0.5:
            # swap two positions
            i, j = np.random.randint(n, size=2)
            cand[i], cand[j] = cand[j], cand[i]
        else:
            # move a fraction from i to j
            i, j = np.random.randint(n, size=2)
            if i == j:
                continue
            delta = step_scale * best_a[i]
            if delta <= 0.0:
                continue
            cand[i] -= delta
            cand[j] += delta
            # clamp tiny negative values caused by round‑off
            if cand[i] < 0.0:
                cand[i] = 0.0

        cand = project_to_simplex(cand)

        cur_score = compute_score(cand)

        if cur_score < best_score:
            best_score = cur_score
            best_a = cand
            step_scale = min(step_scale * 1.07, 0.5)   # enlarge step on success
        else:
            step_scale *= 0.97                         # shrink on failure

        max_steps -= 1

    return best_a.tolist(), best_score

def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step‑function that minimises `evaluate_sequence`.
    The routine:
      * tries several sequence lengths (256 … 2048),
      * launches a few random + analytical starts,
      * runs a corrected projected‑Adam optimiser with a soft‑max homotopy,
      * finishes with a stochastic hill‑climber.
    The best vector found (as a plain Python list) is returned.
    """
    np.random.seed(seed)
    start_time = time.time()
    # leave a small safety margin for the harness
    deadline = start_time + budget_s - 2.0

    best_seq = None
    best_val = float("inf")

    # ------------------------------------------------------------------
    # Optional fallback from a historic good solution (if present).
    # ------------------------------------------------------------------
    try:
        fallback = [float(x) for x in height_sequence_1]  # type: ignore
        fallback_val = evaluate_sequence(fallback)
        print(f"Fallback score = {fallback_val:.6f}")
        if fallback_val < best_val:
            best_val = fallback_val
            best_seq = fallback
    except Exception:
        pass

    # ------------------------------------------------------------------
    # Parameter grid for the Adam phase.
    # ------------------------------------------------------------------
    lengths = [256, 512, 1024, 2048]      # moderate sizes – cheap FFTs
    dirichlet_alphas = [0.2, 0.5, 1.0, 2.0]

    for n in lengths:
        if time.time() >= deadline - 5.0:
            break
        print(f"\n=== Optimising length n = {n} ===")

        # Prepare a list of diverse initial points
        initial_vectors = [
            arcsine_initial(n),
            np.full(n, 1.0 / n),                      # uniform
        ]

        for a in dirichlet_alphas:
            vec = np.random.dirichlet(np.full(n, a))
            initial_vectors.append(vec)

        # Run Adam optimisation for each start
        for idx, init_vec in enumerate(initial_vectors, start=1):
            if time.time() >= deadline - 5.0:
                break
            print(f"  start {idx}/{len(initial_vectors)}")
            remaining = deadline - time.time()
            # allocate a generous number of iterations while respecting time
            # (one iteration for n=2048 costs ~ a few ms)
            iters = int(min(300_000, remaining * 30))

            seq, val = adam_optimize(
                init_a=init_vec,
                time_deadline=deadline,
                max_iter=iters,
                eta=0.02,
                alpha_start=1.0,
                alpha_end=5000.0,
                noise_every=1500,
                noise_scale=5e-5,
            )
            print(f"    Adam best score = {val:.6f}")
            if val < best_val:
                best_val = val
                best_seq = seq.tolist()
                print(f"    *** Global best updated to {best_val:.6f} (n={n}) ***")

    # ------------------------------------------------------------------
    # Final hill‑climb refinement of the overall best candidate.
    # ------------------------------------------------------------------
    if best_seq is not None and time.time() < deadline - 1.0:
        print("\nRefining best sequence with permutation hill‑climb")
        refined_seq, refined_val = hill_climb_permutation(
            best_seq, time_deadline=deadline, max_steps=250_000
        )
        print(f"    refined score = {refined_val:.6f}")
        if refined_val < best_val:
            best_val = refined_val
            best_seq = refined_seq

    # ------------------------------------------------------------------
    # Safeguard: if nothing was produced, fall back to a simple arcsine.
    # ------------------------------------------------------------------
    if best_seq is None:
        print("No candidate produced – falling back to arcsine(256).")
        best_seq = arcsine_initial(256).tolist()
        best_val = evaluate_sequence(best_seq)

    final_score = evaluate_sequence(best_seq)
    print(f"\nFinished search. Final reported score = {final_score:.6f}")

    # The harness expects a plain Python list of floats.
    return best_seq
