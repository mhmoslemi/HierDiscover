"""
Hybrid optimisation for the step‑function‐minimisation problem.

1. Initialise several simple shapes (arcsine, uniform, Dirichlet,
   small noisy perturbations) on a fixed length (default 2048) and
   normalise them to the probability simplex.

2. *Smooth phase* – exponentiated‑gradient descent on a
   temperature‑annealed log‑sum‑exp approximation of the
   max‑convolution.  The temperature λ is increased slowly up to a
   large value (≈500) so that the optimisation first explores the
   landscape and then focuses on the true maximum.

3. *Max‑focused phase* – a plain sub‑gradient descent on the exact
   objective `max_k (a*a)_k`.  The sub‑gradient for the active index
   `k*` is `g_i = 2·a_{k*‑i}` (zero when the index is out of range).
   An exponentiated‑gradient update with a decaying step size
   (η_t = η₀/√t) keeps the variables non‑negative; after each step
   the vector is projected onto the simplex.

4. The two phases are run sequentially for each start vector;
   the best result (according to the official `evaluate_sequence`
   function) is kept.  If time permits a final polishing run of the
   max‑focused phase is performed on the current global best.

The algorithm respects the time budget, prints progress and always
returns a feasible non‑negative sequence.
"""

import time
import numpy as np
from typing import List, Tuple

# ----------------------------------------------------------------------
# Low‑level numeric utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Return the smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Linear convolution via real FFT (full length, zero‑padded to a power‑of‑2)."""
    m = len(a) + len(b) - 1
    n_fft = next_pow2(m)
    A = np.fft.rfft(a, n=n_fft)
    B = np.fft.rfft(b, n=n_fft)
    C = A * B
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


def compute_score(a: np.ndarray) -> float:
    """
    Internal surrogate of the official objective:
        2·n·max(a*a) / (sum(a)²)   (n = len(a))
    """
    n = len(a)
    conv = convolution_fft(a, a)
    max_val = float(np.max(conv))
    s = float(np.sum(a))
    if s < 1e-12:
        return float("inf")
    return 2.0 * n * max_val / (s * s)


def arcsine_initial(n: int) -> np.ndarray:
    """Discrete arcsine density (unnormalised)."""
    i = np.arange(n, dtype=float)
    eps = 0.25
    vals = 1.0 / np.sqrt((i + 0.5) * (n - 0.5 - i) + eps)
    return vals / np.sum(vals)


def generate_shape(n: int, p: float, eps: float) -> np.ndarray:
    """Symmetric power‑law shape a_i ∝ ((i+ε)(n‑1‑i+ε))⁻ᵖ."""
    i = np.arange(n, dtype=float)
    base = ((i + eps) * (n - 1 - i + eps)) ** (-p)
    total = base.sum()
    if total == 0.0:
        return np.full(n, 1.0 / n)
    return base / total


def softmax_weights(conv: np.ndarray, lam: float) -> np.ndarray:
    """Numerically stable soft‑max over the convolution entries."""
    x = lam * conv
    mx = np.max(x)
    w = np.exp(x - mx)
    w /= w.sum()
    return w


def gradient_from_weights(a: np.ndarray, w: np.ndarray) -> np.ndarray:
    """
    Gradient of Σ_k w_k·(a*a)_k w.r.t a.
    Implements g_j = 2· Σ_m a_m · w_{m+j}.
    """
    w_rev = w[::-1]
    conv = convolution_fft(a, w_rev)          # length = 3·n‑2
    idx = 2 * len(a) - 2                       # centre of the above convolution
    grads = 2.0 * conv[idx - np.arange(len(a))]
    return grads


def project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x≥0, sum(x)=1}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all entries are zero or negative – fallback to uniform vector
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


def max_gradient(a: np.ndarray, k: int) -> np.ndarray:
    """
    Sub‑gradient of the max‑convolution term (a*a)_k.
    g_i = 2·a_{k-i} (if 0≤k-i<n) else 0.
    """
    n = len(a)
    indices = np.arange(n)
    j = k - indices
    mask = (j >= 0) & (j < n)
    g = np.zeros_like(a)
    g[mask] = 2.0 * a[j[mask]]
    return g


# ----------------------------------------------------------------------
# Optimisers
# ----------------------------------------------------------------------


def optimise_softmax(
    initial_a: np.ndarray,
    deadline: float,
    *,
    lam_start: float = 5.0,
    lam_end: float = 500.0,
    lam_ramp_iters: int = 10_000,
    eta0: float = 0.6,
    max_iter: int = 500_000,
    noise_every: int = 0,
    noise_scale: float = 0.0,
) -> Tuple[np.ndarray, float]:
    """
    Exponentiated‑gradient descent on the smooth approximation
    Φ_λ(a) = (2·n / Σ a_i²)·(1/λ·log Σ_k exp(λ·(a*a)_k)).
    Returns the best vector seen and its (non‑smooth) internal score.
    """
    a = project_simplex(initial_a.astype(float).copy())
    best_a = a.copy()
    best_sc = compute_score(a)
    n = len(a)

    for it in range(1, max_iter + 1):
        if time.time() >= deadline:
            break

        lam = lam_start + (lam_end - lam_start) * min(1.0, it / lam_ramp_iters)

        conv = convolution_fft(a, a)          # length 2n‑1
        w = softmax_weights(conv, lam)

        grad = gradient_from_weights(a, w)    # same size as a

        eta = eta0 / np.sqrt(it)
        a = a * np.exp(-eta * grad)

        if noise_every and (it % noise_every == 0):
            a += np.random.randn(n) * noise_scale
            a = np.maximum(a, 0.0)

        a = project_simplex(a)

        cur_sc = compute_score(a)
        if cur_sc < best_sc:
            best_sc = cur_sc
            best_a = a.copy()

        if it % 5000 == 0:
            print(f"      softmax iter={it:,d} λ={lam:.1f} cur={cur_sc:.6f} best={best_sc:.6f}")

    return best_a, best_sc


def optimise_max_subgrad(
    initial_a: np.ndarray,
    deadline: float,
    *,
    eta0: float = 0.5,
    max_iter: int = 400_000,
    noise_every: int = 0,
    noise_scale: float = 0.0,
) -> Tuple[np.ndarray, float]:
    """
    Sub‑gradient descent on the exact objective max_k (a*a)_k.
    Uses multiplicative updates and simplex projection.
    """
    a = project_simplex(initial_a.astype(float).copy())
    best_a = a.copy()
    best_sc = compute_score(a)

            a = np.maximum(a, 0.0)

        a = project_simplex(a)

        cur_sc = compute_score(a)

        if it % 5000 == 0:
            print(f"      max‑subgrad iter={it:,d} cur={cur_sc:.6f} best={best_sc:.6f}")

    return best_a, best_sc

            initial_a=a_smooth,
            deadline=deadline,
            eta0=0.5,
            max_iter=400_000,
            noise_every=800,
            noise_scale=2e-6,
        )

        seq = a_refined.tolist()
        val = evaluate_sequence(seq)
        print(f"    evaluated score = {val:.6f}")

        if val < best_val:
            best_val = val
            best_seq = seq
            print(f"    *** New global best: {best_val:.6f} ***")

    # ------------------------------------------------------------------
    # 4. Final polishing (if time permits).
    # ------------------------------------------------------------------
    if best_seq and time.time() < deadline - 5.0:
        print("\nFinal polishing of the overall best sequence")
        a_start = np.array(best_seq, dtype=float)

        a_final, _ = optimise_max_subgrad(
            initial_a=a_start,
            deadline=deadline,
            eta0=0.3,
            max_iter=600_000,
            noise_every=500,
            noise_scale=1e-6,
        )
        final_val = evaluate_sequence(a_final.tolist())
        print(f"Final polished score = {final_val:.6f}")

        if final_val < best_val:
            best_val = final_val
            best_seq = a_final.tolist()
            print("*** Updated global best after final polish ***")

    # ------------------------------------------------------------------
    # 5. Guard – always return a feasible sequence.
    # ------------------------------------------------------------------
    if not best_seq:
        best_seq = [1.0] * 256
        best_val = evaluate_sequence(best_seq)

    print(f"\nSearch finished. Best reported score = {best_val:.6f}")
    return best_seq
