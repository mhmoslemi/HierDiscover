"""
A smoothed Adam optimiser for the step‑function problem.

The objective
    2·n·max(conv(a,a)) / (sum(a)^2)

is invariant to scaling, so the search is performed on the simplex
Σa_i = 1.  The max is approximated by a soft‑max (log‑sum‑exp) with a
temperature that is increased during optimisation.  This yields a
smooth gradient that can be propagated through the convolution using
FFT‑based circular convolutions (the aperiodic convolution is obtained
by appropriate slicing).  Adam updates are applied to the soft‑max
logits; occasional random jitter helps escape plateaus.  Several
re‑starts with different initialisations are performed within the time
budget and the best sequence ever seen is returned.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper functions – all top‑level (no closures)
# ----------------------------------------------------------------------


def _fft_len_for_conv(length: int) -> int:
    """Return the smallest power‑of‑two length that can hold a convolution
    of size ``length`` (used for FFT‑based convolutions)."""
    return 1 << ((length - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray, fft_len: int) -> np.ndarray:
    """Aperiodic convolution of ``x`` and ``y`` using FFT, truncated to the
    valid length (len(x)+len(y)-1)."""
    fx = np.fft.rfft(x, n=fft_len)
    fy = np.fft.rfft(y, n=fft_len)
    conv_full = np.fft.irfft(fx * fy, n=fft_len)
    return conv_full[: x.shape[0] + y.shape[0] - 1]


def _softmax_logits(logits: np.ndarray) -> np.ndarray:
    """Stable soft‑max returning a probability vector (sums to 1)."""
    m = np.max(logits)
    exp_shift = np.exp(logits - m)
    return exp_shift / exp_shift.sum()


def _initial_sequence(kind: str, n: int, rng: np.random.Generator,
                      prev: np.ndarray | None) -> np.ndarray:
    """Generate a feasible start vector of length ``n``."""
    if kind == "prev" and prev is not None:
        seq = prev.copy()
        if seq.shape[0] > n:
            seq = seq[:n]
        elif seq.shape[0] < n:
            seq = np.concatenate([seq, np.zeros(n - seq.shape[0])])
        seq = np.maximum(seq, 0.0)
        total = seq.sum()
        if total == 0.0:
            return np.full(n, 1.0 / n)
        return seq / total

    if kind == "uniform":
        return np.full(n, 1.0 / n)

    if kind == "random_dirichlet":
        # concentration < 1 gives sparse vectors
        alpha = 0.5
        sample = rng.gamma(alpha, 1.0, size=n)
        return sample / sample.sum()

    if kind == "geometric":
        r = 0.98 + 0.02 * rng.random()
        left = r ** np.arange(n)
        right = r ** np.arange(n - 1, -1, -1)
        seq = left + right
        return seq / seq.sum()

    # fallback
    return np.full(n, 1.0 / n)


def _adam_optimize_smooth(init_a: np.ndarray,
                          deadline: float,
                          restart_id: int,
                          target: float = 1.503) -> tuple[np.ndarray, float]:
    """
    Adam optimiser that minimizes the *smoothed* version of the objective.
    The smoothing is a soft‑max over the convolution values with a temperature
    (parameter ``alpha``) that grows with the iteration count.
    Returns the best vector (sum = 1) and its true (non‑smoothed) score.
    """
    n = init_a.shape[0]

    # FFT length that can accommodate a * (a_rev) * p (length 3n‑2)
    fft_len = _fft_len_for_conv(3 * n - 2)

    # initialise logits (soft‑max parametrisation)
    eps_init = 1e-12
    logits = np.log(init_a + eps_init)

    # Adam state
    m = np.zeros_like(logits)
    v = np.zeros_like(logits)
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8

    # evaluate the start point (true objective)
    a = _softmax_logits(logits)
    conv = _conv_fft(a, a, fft_len)[: 2 * n - 1]
    best_score = 2.0 * n * conv.max()
    best_a = a.copy()

    stagnation = 0
    it = 0

    # smoothing schedule parameters
    alpha_start = 0.5
    alpha_max = 500.0          # large enough to make the soft‑max sharp
    alpha_rate = 0.0005        # linear increase per iteration

    while time.time() < deadline:
        it += 1

        # --------------------------------------------------------------
        # 1) obtain the simplex‑constrained vector
        # --------------------------------------------------------------
        a = _softmax_logits(logits)

        # --------------------------------------------------------------
        # 2) plain convolution and true score (for bookkeeping)
        # --------------------------------------------------------------
        conv = _conv_fft(a, a, fft_len)[: 2 * n - 1]
        cur_max = conv.max()
        cur_score = 2.0 * n * cur_max

        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            stagnation = 0
            print(f"Restart {restart_id:02d} iter {it:6d}: best = {best_score:.6f}")
        else:
            stagnation += 1

        # early stop if we already beat the target
        if best_score < target:
            break
        # jitter when stuck for too long
        if stagnation > 8000:
            logits += 0.01 * np.random.randn(n)
            stagnation = 0

        # --------------------------------------------------------------
        # 3) smoothing parameter (soft‑max temperature)
        # --------------------------------------------------------------
        alpha = min(alpha_max, alpha_start + alpha_rate * it)

        # soft‑max weights over the convolution entries (stable)
        shift = cur_max
        # exponentiation safely: subtract max to avoid overflow
        exp_arg = alpha * (conv - shift)
        w_unnorm = np.exp(exp_arg)
        w = w_unnorm / w_unnorm.sum()      # length 2n‑1, sums to 1

        # --------------------------------------------------------------
        # 4) gradient of the *smoothed* objective w.r.t. a
        #    (derived via convolution of a_rev with w)
        # --------------------------------------------------------------
        # a_rev = a reversed
        a_rev = a[::-1]
        # weighted convolution a_rev * w
        conv_rev_full = _conv_fft(a_rev, w, fft_len)[: 3 * n - 2]
        # slice that corresponds to indices n‑1 … n‑1+n‑1
        g_slice = conv_rev_full[n - 1:n - 1 + n]      # shape (n,)
        # gradient of the true objective (2·n·max) w.r.t. a:
        #   ∂/∂a  2 n Σ_k w_k b_k  = 4 n * (a_rev * w)_{slice}
        grad_a = 4.0 * n * g_slice

        # --------------------------------------------------------------
        # 5) transform gradient to logit space (soft‑max Jacobian)
        # --------------------------------------------------------------
        avg = np.dot(a, grad_a)
        grad_logits = a * (grad_a - avg)

        # --------------------------------------------------------------
        # 6) Adam update on logits
        # --------------------------------------------------------------
        m = beta1 * m + (1.0 - beta1) * grad_logits
        v = beta2 * v + (1.0 - beta2) * (grad_logits ** 2)

        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        # diminishing learning‑rate (helps late‑stage fine‑tuning)
        lr = 0.4 / np.sqrt(it)

        logits -= lr * m_hat / (np.sqrt(v_hat) + eps)

    return best_a, best_score

# ----------------------------------------------------------------------


def propose_candidate(budget_s: int = 1000) -> list:
    """
    Search for a low‑score step‑function sequence within ``budget_s`` seconds.
    Returns a list of non‑negative floats.  The algorithm consists of many
    short Adam restarts, each employing a smooth approximation of the
    max‑convolution objective with an increasing temperature.
    """
    rng = np.random.default_rng(12345)          # deterministic randomness
    deadline = time.time() + budget_s - 5.0     # safety margin

    # --------------------------------------------------------------
    # Problem size and possible previous best
    # --------------------------------------------------------------
    try:
        # ``height_sequence_1`` may be defined by the host environment
        previous = np.asarray(height_sequence_1, dtype=float)   # type: ignore[name-defined]
        n = previous.shape[0]
    except Exception:      # pragma: no cover
        previous = None
        n = 5146                      # default length used in earlier experiments

    # --------------------------------------------------------------
    # Global bookkeeping
    # --------------------------------------------------------------
    best_overall_a = None
    best_overall_score = float("inf")
    restart_id = 0

    # initialise strategies (cycling through them)
    strategies = ["prev", "uniform", "random_dirichlet", "geometric"]

    # --------------------------------------------------------------
    # Repeated short Adam runs – each gets a slice of the remaining budget
    # --------------------------------------------------------------
    while time.time() < deadline - 10.0:          # keep a few seconds for cleanup
        kind = strategies[restart_id % len(strategies)]

        if kind == "prev" and previous is None:
            kind = "uniform"

        a0 = _initial_sequence(kind, n, rng, previous)

        # allocate at most 20 s or a fraction of the remaining time
        time_left = deadline - time.time()
        this_deadline = time.time() + min(20.0, 0.4 * time_left)

        a_opt, score_opt = _adam_optimize_smooth(a0, this_deadline, restart_id)

        if score_opt < best_overall_score:
            best_overall_score = score_opt
            best_overall_a = a_opt.copy()
            print(f"Global best after restart {restart_id:02d}: {best_overall_score:.6f}")

        # early exit if we have already passed the target bound
        if best_overall_score < 1.503:
            print(f"Target reached (score={best_overall_score:.6f}) after "
                  f"{restart_id + 1} restarts.")
            break

        restart_id += 1

    # --------------------------------------------------------------
    # Simple local refinement (optional jitter‑based hill‑climb)
    # --------------------------------------------------------------
    if best_overall_a is not None:
        # a few random perturbations to possibly improve the final result
        a = best_overall_a.copy()
        for _ in range(2000):
            # pick two indices and move a tiny amount epsilon from i to j
            i, j = rng.integers(0, n, size=2)
            if i == j:
                continue
            eps = 1e-5 * rng.random()
            if a[i] < eps:
                continue
            a[i] -= eps
            a[j] += eps
            # keep the vector on the simplex (numerical drift)
            a = np.maximum(a, 0.0)
            a_sum = a.sum()
            if a_sum == 0:
                continue
            a /= a_sum
            # evaluate
            conv = np.convolve(a, a)[: 2 * n - 1]
            cur_score = 2.0 * n * conv.max()
            if cur_score < best_overall_score:
                best_overall_score = cur_score
                best_overall_a = a.copy()
                print(f"Refine: improved global best to {best_overall_score:.6f}")

    # fallback – never return an empty sequence
    if best_overall_a is None:
        best_overall_a = np.full(n, 1.0 / n)

    # convert to a plain Python list for the external evaluator
    return [float(v) for v in best_overall_a.tolist()]
