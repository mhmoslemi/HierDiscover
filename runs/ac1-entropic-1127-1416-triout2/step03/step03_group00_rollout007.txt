"""
Propose a step‑function (probability vector) that minimises
evaluate_sequence().  The algorithm consists of three stages:

1.  Initialise a “U‑shaped” arcsine distribution on the simplex.
2.  Convex optimisation of the smoothed objective
    f(a)=log‑sum‑exp(α·(a*a)) using Adam (mirrored gradients) and a
    Euclidean projection onto the probability simplex after each step.
    The smoothing parameter α is increased gradually so that the
    optimisation first follows a smooth landscape and finally concentrates
    on the true maximum of the convolution.
3.  A short simulated‑annealing (mass‑transfer) refinement that
    explores the neighbourhood of the best point found by Adam.

All heavy work (convolutions and gradients) is performed with FFTs,
so even for n≈5000 the inner loop runs in a few microseconds, allowing
many thousands of iterations within the 1000 s budget.
"""

import time
import math
import numpy as np


# ----------------------------------------------------------------------
# Helper functions
# ----------------------------------------------------------------------


def simplex_projection(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the probability simplex:
    { w | w_i >= 0 , sum w_i = 1 }.
    Implements the algorithm of Duchi et al. (2008).
    """
    v = np.asarray(v, dtype=float)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1] - 1          # zero‑based
    theta = cssv[cond][-1] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def beta_sequence(alpha: float, beta: float, n: int) -> np.ndarray:
    """
    Discretised Beta(α,β) density on n points, normalised to sum to 1.
    Uses the midpoint rule.
    """
    x = (np.arange(n, dtype=float) + 0.5) / n
    # work in log‑space to avoid overflow for extreme parameters
    logw = (alpha - 1.0) * np.log(x) + (beta - 1.0) * np.log(1.0 - x)
    logw = np.clip(logw, -700.0, 700.0)
    w = np.exp(logw)
    if not np.isfinite(w.sum()) or w.sum() == 0.0:
        w = np.ones_like(w)
    return w / w.sum()


def next_pow2(x: int) -> int:
    """Return the smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def conv_fft(a: np.ndarray, L: int) -> np.ndarray:
    """Linear convolution a*a using FFT of length L (real‑valued)."""
    fft_a = np.fft.rfft(a, n=L)
    full = np.fft.irfft(fft_a * fft_a, n=L)
    return full[: 2 * a.shape[0] - 1]    # truncate to the true length


def grad_softmax_conv(a: np.ndarray,
                     conv: np.ndarray,
                     alpha: float,
                     L_grad: int) -> np.ndarray:
    """
    Gradient of  φ(a)= log‑sum‑exp(α·(a*a)) .
    Returns ∇φ(a) (size = len(a)).
    """
    # softmax weights (numerically stable)
    max_c = conv.max()
    w = np.exp(alpha * (conv - max_c))
    q = w / w.sum()                     # length 2n‑1, sum(q)=1

    # g = 2 * (a ⋆ q)  where ⋆ denotes cross‑correlation.
    # Compute via convolution with the reversed weight vector.
    q_rev = q[::-1]
    fft_a = np.fft.rfft(a, n=L_grad)
    fft_q = np.fft.rfft(q_rev, n=L_grad)
    corr_full = np.fft.irfft(fft_a * fft_q, n=L_grad)

    m = 2 * n - 1
    # The wanted entries are centred around index m‑1
    centre = m - 1
    g = 2.0 * corr_full[centre - np.arange(n)]
    return g


# ----------------------------------------------------------------------
# Main candidate generator
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 0, budget_s: int = 1000) -> list[float]:
    """
    Search for a non‑negative height sequence with minimal evaluate_sequence().
    The routine respects a time budget (default 1000 s).  It returns the best
    sequence (as a list of floats) found before the deadline.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5  # small safety margin

    # --------------------------------------------------------------
    # 1. Problem size and initial point
    # --------------------------------------------------------------
    try:
        init = np.array(height_sequence_1, dtype=float)     # type: ignore[name-defined]
        n = init.shape[0]
        print(f"Using existing sequence of length {n}.")
    except NameError:
        n = 5146
        print(f"No existing sequence – using n = {n} with arcsine start.")

    # arcsine‑like start: w_i ∝ 1/√(x·(1‑x)), x = (i+0.5)/n
    i = np.arange(n, dtype=float)
    eps = 1e-12
    a = 1.0 / np.sqrt((i + 0.5) * (n - 0.5 - i) + eps)
    a = a / a.sum()                     # now a lies on the simplex

    # keep the best solution seen so far
    conv = conv_fft(a, next_pow2(2 * n - 1))
    best_score = 2.0 * n * conv.max()
    best_a = a.copy()
    print(f"Initial score = {best_score:.6f}")

    # --------------------------------------------------------------
    # 2. Adam optimisation of a smoothed max‑convolution objective
    # --------------------------------------------------------------
    # Adam hyper‑parameters
    beta1, beta2 = 0.9, 0.999
    eps_adam = 1e-8
    lr = 0.2                     # base learning rate

    # smoothing parameter (log‑sum‑exp temperature)
    alpha = 5.0
    alpha_mul = 1.03            # geometric increase each iteration
    max_alpha = 200.0

    # pre‑compute FFT sizes (powers of two for speed)
    L_conv = next_pow2(2 * n - 1)          # for a*a
    L_grad = next_pow2(3 * n - 2)          # for gradient convolution

    m = np.zeros_like(a)   # first moment
    v = np.zeros_like(a)   # second moment
    it = 0

    # run Adam while leaving ~30 s for the final SA phase
    while time.time() < deadline - 30.0:
        # ---- convolution a*a ------------------------------------------------
        fft_a = np.fft.rfft(a, n=L_conv)
        conv_full = np.fft.irfft(fft_a * fft_a, n=L_conv)[: 2 * n - 1]

        cur_score = 2.0 * n * conv_full.max()
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            if it % 500 == 0:
                print(f"Adam iter {it:6d}: new best {best_score:.6f}")

        # ---- gradient of smoothed max ----------------------------------------
        g = grad_softmax_conv(a, conv_full, alpha, L_grad)

        # ---- Adam update ------------------------------------------------------
        m = beta1 * m + (1.0 - beta1) * g
        v = beta2 * v + (1.0 - beta2) * (g * g)
        t = it + 1
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        step = lr / (np.sqrt(v_hat) + eps_adam)

        a = a - step * m_hat
        a = simplex_projection(a)    # keep a on the probability simplex

        # increase smoothing (makes the objective sharper)
        if alpha < max_alpha:
            alpha = min(max_alpha, alpha * alpha_mul)

        it += 1

    print(f"Adam phase finished after {it} iterations; best score = {best_score:.6f}")

    # --------------------------------------------------------------
    # 3. Simulated‑annealing fine‑tuning (mass‑transfer moves)
    # --------------------------------------------------------------
    cur_a = best_a.copy()
    # evaluate current score
    fft_cur = np.fft.rfft(cur_a, n=L_conv)
    conv_cur = np.fft.irfft(fft_cur * fft_cur, n=L_conv)[: 2 * n - 1]
    cur_score = 2.0 * n * conv_cur.max()

    # SA schedule
    T_start = cur_score * 0.02
    T_end = cur_score * 1e-6
    decay = 0.9999
    T = T_start
    it_sa = 0

    while time.time() < deadline:
        T = max(T_end, T * decay)

        # ---- propose a mass‑transfer move (keeps sum = 1) -------------------
        src = np.random.choice(n, p=cur_a)          # favour larger entries
        tgt = np.random.randint(n - 1)
        if tgt >= src:
            tgt += 1

        if cur_a[src] == 0.0:
            continue

        delta = cur_a[src] * np.random.rand()       # random fraction of source mass
        cand = cur_a.copy()
        cand[src] -= delta
        cand[tgt] += delta

        # ---- evaluate the candidate -----------------------------------------
        fft_c = np.fft.rfft(cand, n=L_conv)
        conv_c = np.fft.irfft(fft_c * fft_c, n=L_conv)[: 2 * n - 1]
        cand_score = 2.0 * n * conv_c.max()

        # ---- Metropolis acceptance -------------------------------------------
        if cand_score < cur_score:
            accept = True
        else:
            prob = math.exp(-(cand_score - cur_score) / max(T, 1e-12))
            accept = np.random.rand() < prob

        if accept:
            cur_a = cand
            cur_score = cand_score
            if cand_score < best_score:
                best_score = cand_score
                best_a = cand.copy()
                print(f"SA iter {it_sa:6d}: new best {best_score:.6f}")

        # occasional large random restart to escape stagnation
        if it_sa % 50000 == 0 and it_sa > 0:
            tmp = beta_sequence(
                np.random.uniform(0.5, 4.0),
                np.random.uniform(0.5, 4.0),
                n,
            )
            cur_a = tmp
            # recompute its score (cheap – use the same helper)
            fft_tmp = np.fft.rfft(cur_a, n=L_conv)
            conv_tmp = np.fft.irfft(fft_tmp * fft_tmp, n=L_conv)[: 2 * n - 1]
            cur_score = 2.0 * n * conv_tmp.max()

        it_sa += 1

    print(f"SA finished after {it_sa} iterations; final best score = {best_score:.6f}")

    # --------------------------------------------------------------
    # 4. Return the best sequence (still on the simplex)
    # --------------------------------------------------------------
    best_a = best_a / best_a.sum()
    return [float(v) for v in best_a.tolist()]
