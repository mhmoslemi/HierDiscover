def propose_candidate(seed=42, budget_s=1000, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 2.0

    best_seq = None
    best_score = float('inf')

    # try multiple lengths
    lengths = [4096, 8192, 16384, 32768]
    for n in lengths:
        if time.time() >= deadline - 5.0:
            break
        # arcsine start
        a = arcsine_initial(n)  # already on simplex, sum=1
        # optionally jitter small amount
        # local gradient descent
        # simple subgradient descent on max
        max_iters = int(min(2000, (deadline - time.time()) * 5))  # approx steps
        step0 = 0.5
        for t in range(1, max_iters+1):
            if time.time() >= deadline - 1.0:
                break
            # compute convolution
            conv = convolution_fft(a, a)
            k_max = int(np.argmax(conv))
            # gradient for max term
            idx = k_max - np.arange(n)
            mask = (idx >= 0) & (idx < n)
            grad = np.zeros_like(a)
            grad[mask] = 2.0 * a[idx[mask]]
            # step size diminishing
            step = step0 / np.sqrt(t)
            a_new = a - step * grad
            # ensure nonnegative and project to simplex
            a_new = project_to_simplex(np.maximum(a_new, 0.0))
            a = a_new
        # evaluate
        seq = a.tolist()
        val = evaluate_sequence(seq)
        print(f"Length {n}, score {val}")
        if val < best_score:
            best_score = val
            best_seq = seq

    # fallback to arcsine of 2048 if nothing
    if best_seq is None:
        best_seq = arcsine_initial(2048).tolist()
        best_score = evaluate_sequence(best_seq)

    print(f"Final best score: {best_score}")
    return best_seq

"""
Propose a low‑score step‑function.

The algorithm is deliberately simple yet powerful:
1.  Build a family of “arcsine‑shaped’’ probability vectors of
    increasing length (the arcsine law is known to flatten the
    self‑convolution).
2.  For each length, optionally apply a few steps of projected
    sub‑gradient descent on the exact max‑convolution objective.
3.  Evaluate the true objective (``evaluate_sequence``) and keep the
    best vector found.  The search stops as soon as the target score
    1.503 is reached or the time budget is exhausted.

The arcsine initialization already yields a convolution that is
nearly uniform; with a modest number of gradient steps we can push
the maximum a little lower, and with sufficiently large length the
score drops below the target without any heavy optimisation machinery.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Utility functions
# ----------------------------------------------------------------------


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """ Euclidean projection of a vector onto the probability simplex. """
    if np.all(v <= 0):
        # all non‑positive, return uniform distribution
        return np.full_like(v, 1.0 / v.shape[0])

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, v.shape[0] + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.where(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def next_pow2(x: int) -> int:
    """ Smallest power of two ≥ x. """
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """ Full convolution using real FFT (fast for large vectors). """
    m = len(a) + len(b) - 1
    n_fft = next_pow2(m)
    A = np.fft.rfft(a, n=n_fft)
    B = np.fft.rfft(b, n=n_fft)
    C = A * B
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


def arcsine_initial(n: int) -> np.ndarray:
    """
    Classical arcsine start:
        a_i ∝ 1 / sqrt( (i+0.5)·(n-0.5-i) )
    The vector is projected onto the simplex (sum = 1, non‑negative).
    """
    i = np.arange(n, dtype=float)
    # tiny epsilon prevents division by zero at the boundaries
    eps = 1e-12
    vals = 1.0 / np.sqrt((i + 0.5) * (n - 0.5 - i) + eps)
    return project_to_simplex(vals)


def subgradient_descent(a: np.ndarray,
                        deadline: float,
                        max_iter: int = 500) -> np.ndarray:
    """
    Simple projected sub‑gradient descent on the true max‑convolution.
    The step size decreases like η / sqrt(t).  The routine stops
    early if the time budget expires.
    """
    n = a.shape[0]
    eta0 = 0.5  # initial step magnitude
    t = 0
    while t < max_iter and time.time() < deadline:
        t += 1
        # current full convolution
        conv = convolution_fft(a, a)
        k_max = int(np.argmax(conv))            # index of the current maximum

        # gradient of the active term c_{k_max} = Σ a_i a_{k_max-i}
        idx = k_max - np.arange(n)
        mask = (idx >= 0) & (idx < n)
        grad = np.zeros_like(a)
        grad[mask] = 2.0 * a[idx[mask]]

        # diminishing step size
        eta = eta0 / np.sqrt(t)
        a = a - eta * grad

        # keep feasibility (non‑negative) and project back onto simplex
        a = project_to_simplex(np.maximum(a, 0.0))

    return a


# ----------------------------------------------------------------------
# Main entry point required by the evaluation harness
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative sequence that minimises `evaluate_sequence`.
    The routine tries a few increasingly large arcsine‑shaped vectors,
    optionally refines them with a cheap projected sub‑gradient loop,
    and returns the best one found before the time budget expires.
    """
    np.random.seed(seed)
    start_time = time.time()
    # keep a tiny safety margin for the harness
    deadline = start_time + budget_s - 2.0

    best_seq = None
    best_score = float("inf")
    target_score = 1.503  # we stop as soon as we beat this

    # candidate lengths – larger vectors give a lower theoretical bound
    candidate_lengths = [8192, 16384, 32768, 65536]

    for n in candidate_lengths:
        if time.time() >= deadline - 2.0:
            break

        print(f"\n=== Trying length n = {n} ===")
        # 1. initialise with the arcsine law (already normalised to sum=1)
        a = arcsine_initial(n)

        # 2. a very small projected sub‑gradient refinement
        #    use more iterations for the smaller sizes only
        remaining = deadline - time.time()
        if remaining > 5.0:
            max_iters = 1200 if n <= 16384 else 200
            a = subgradient_descent(a, deadline, max_iter=max_iters)

        # 3. evaluate the true objective
        seq = a.tolist()
        try:
            score = evaluate_sequence(seq)
        except Exception as e:
            # any unexpected failure – treat as infinitely bad
            print(f"Evaluation error: {e}")
            score = float("inf")

        print(f"Length {n}: score = {score:.6f}")

        if score < best_score:
            best_score = score
            best_seq = seq
            print(f"*** New global best ***")

        if best_score <= target_score:
            print("Target reached – terminating early.")
            break

    # ------------------------------------------------------------------
    # Fallback in the very unlikely case nothing was produced
    # ------------------------------------------------------------------
    if best_seq is None:
        print("Fallback: using a tiny uniform vector.")
        best_seq = [1.0 / 256] * 256
        best_score = evaluate_sequence(best_seq)

    final_score = evaluate_sequence(best_seq)
    print(f"\nFinished search. Final reported score = {final_score:.6f}")

    # Return a plain Python list as required by the harness
    return best_seq
