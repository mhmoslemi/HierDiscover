"""
Mirror‑descent with a smooth (soft‑max) surrogate for the max‑convolution.
The algorithm works on the probability simplex (sum(a)=1) and uses
exponentiated‑gradient updates.  Instead of taking a sub‑gradient only at
the active indices (where the convolution attains its maximum), all
convolution entries are weighted by a soft‑max with a temperature that
is slowly increased.  This yields a richer descent direction and helps
to flatten the autocorrelation more quickly.

The routine performs several independent restarts for a few different
lengths (including the historic length, if available) and keeps the best
sequence seen.  It stops early if the target bound 1.503 is reached.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities
# ----------------------------------------------------------------------


def _project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of a vector onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1] - 1
    theta = cssv[cond][-1] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def _next_pow2(x: int) -> int:
    """Smallest power‑of‑two ≥ x."""
    return 1 << (x - 1).bit_length()


def _conv_self_fft(a: np.ndarray) -> np.ndarray:
    """Linear convolution a*a (full length 2·n‑1) using FFT."""
    n = a.shape[0]
    L = _next_pow2(2 * n - 1)
    A = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(A * A, n=L)
    return conv[: 2 * n - 1]  # truncate to the exact linear length


def _correlation_fft(w: np.ndarray, a: np.ndarray) -> np.ndarray:
    """
    Linear correlation (w * a_rev) at non‑negative shifts.
    Returns an array g of length n where
        g[i] = Σ_j w[i+j] * a[j]   (0 ≤ i < n)
    """
    n = a.shape[0]
    L = _next_pow2(3 * n - 2)               # length for w (2n‑1) + a (n) – 1
    W = np.fft.rfft(w, n=L)
    A_rev = np.fft.rfft(a[::-1], n=L)
    corr = np.fft.irfft(W * A_rev, n=L)
    start = n - 1                            # index where shift 0 begins
    return 2.0 * corr[start : start + n]     # factor 2 as ∂(a*a)/∂a_i = 2·a_{k‑i}


def _softmax_weights(conv: np.ndarray, lam: float) -> np.ndarray:
    """
    Soft‑max weighting of the convolution values.
    The largest entry gets weight 1, all others receive exp(lam·(c‑c_max)).
    The result is normalized to sum 1.
    """
    c_max = conv.max()
    # work in the shifted domain to avoid overflow
    exp_arg = lam * (conv - c_max)          # max becomes 0
    # clip to avoid underflow in exp
    exp_arg = np.clip(exp_arg, -700, 0)
    w = np.exp(exp_arg)
    s = w.sum()
    if s == 0.0:
        # extreme underflow – fall back to uniform weighting
        return np.full_like(w, 1.0 / w.size)
    return w / s
