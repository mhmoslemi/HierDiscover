"""
Improved search for step-function heights minimizing `evaluate_sequence`.

Algorithm:
- Initialise symmetric probability vectors with the classic arcsine shape (near‑optimal).
- Run a projected Adam optimiser on a log‑sum‑exp smooth approximation of the
  max‑convolution, enforcing symmetry after each step.
- The smoothing parameter α is increased gradually on a log‑scale.
- The Adam learning‑rate is decayed as 1/√t and momentum is used.
- Keep the best vector seen (true objective evaluated with the exact max‑convolution).
- Finally polish the best candidate with a deterministic sub‑gradient descent on the
  true non‑smooth objective, still respecting the simplex and symmetry.
- All work respects the given time budget; progress and intermediate scores are printed.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Basic utilities
# ----------------------------------------------------------------------


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection onto the probability simplex (non‑negative, sum = 1)."""
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / v.shape[0])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, v.shape[0] + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.where(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def symmetrize(v: np.ndarray) -> np.ndarray:
    """Force symmetry v[i] == v[-1-i] by averaging with its reverse."""
    return (v + v[::-1]) * 0.5


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full convolution using a real FFT (fast)."""
    m = len(a) + len(b) - 1
    n_fft = next_pow2(m)
    A = np.fft.rfft(a, n=n_fft)
    B = np.fft.rfft(b, n=n_fft)
    C = A * B
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


def compute_score(a: np.ndarray) -> float:
    """
    Fast replica of `evaluate_sequence` when ∑a = 1.
    Returns 2·n·max(a*a).
    """
    n = a.shape[0]
    conv = convolution_fft(a, a)
    max_val = float(np.max(conv))
    return 2.0 * n * max_val


def arcsine_initial(n: int) -> np.ndarray:
    """Classical arcsine start a_i ∝ 1 / sqrt((i+0.5)*(n-0.5-i))."""
    i = np.arange(n, dtype=float)
    eps = 0.25
    vals = 1.0 / np.sqrt((i + 0.5) * (n - 0.5 - i) + eps)
    return project_to_simplex(vals)


# ----------------------------------------------------------------------
# Smoothed‑max gradient (used inside Adam)
# ----------------------------------------------------------------------


def smoothed_gradient(a: np.ndarray, alpha: float):
    """
    Gradient of the soft‑max approximation of max(a*a):

        f_α(a) = (1/α)·log Σ_k exp(α·(a*a)_k)

    Returns the gradient vector (same shape as a) and the raw convolution.
    """
    n = a.shape[0]
    conv = convolution_fft(a, a)

    # soft‑max weights (stable formulation)
    maxc = np.max(conv)
    exp_term = np.exp(alpha * (conv - maxc))
    w = exp_term / np.sum(exp_term)          # Σ w_k = 1, length 2n‑1

    # Convolution of reversed a with the weight vector
    a_rev = a[::-1]
    conv_rev = convolution_fft(a_rev, w)    # length 3n‑2
    start = n - 1
    rev_slice = conv_rev[start:start + n]   # needed for each component

    # Diagonal contribution (indices 2i)
    w_diag = w[::2][:n]

    grad = 2.0 * rev_slice - w_diag * a
    return grad, conv


# ----------------------------------------------------------------------
# Projected Adam optimiser (symmetry‑aware)
# ----------------------------------------------------------------------


def adam_optimize_sym(init_a: np.ndarray,
                      time_deadline: float,
                      max_iter: int = 300_000,
                      eta0: float = 0.05,
                      beta1: float = 0.9,
                      beta2: float = 0.999,
                      eps: float = 1e-8,
                      alpha_start: float = 0.5,
                      alpha_end: float = 4000.0):
    """
    Projected Adam optimiser on the smoothed max‑convolution objective.

    The iterate is kept symmetric and projected onto the probability simplex.
    Returns the best vector (ndarray) found and its exact (unsmoothed) score.
    """
    a = project_to_simplex(init_a.astype(float).copy())
    a = symmetrize(a)

    m = np.zeros_like(a)
    v = np.zeros_like(a)

    best_a = a.copy()
    best_score = compute_score(a)

    n = len(a)

    for t in range(1, max_iter + 1):
        if time.time() >= time_deadline:
            break

        # –‑ Smoothing schedule (log‑linear) –‑
        # α grows from alpha_start to alpha_end following a log‑scale.
        progress = min(1.0, t / max_iter)
        # logarithmic interpolation
        log_a0 = np.log(alpha_start + 1e-12)
        log_a1 = np.log(alpha_end + 1e-12)
        alpha = np.exp(log_a0 + (log_a1 - log_a0) * progress)

        grad, _ = smoothed_gradient(a, alpha)

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)

        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)

        # step‑size decays like 1/√t
        eta = eta0 / np.sqrt(t)

        step = eta * m_hat / (np.sqrt(v_hat) + eps)
        a_new = a - step

        # enforce feasibility
        a_new = np.maximum(a_new, 0.0)
        a_new = symmetrize(a_new)
        a_new = project_to_simplex(a_new)

        cur_score = compute_score(a_new)
        if cur_score < best_score:
            best_score = cur_score
            best_a = a_new.copy()
            # occasional verbose output
            # print(f"      Adam iter {t:,d} – new best {best_score:.6f}")

        a = a_new

        # light progress print every few thousand iterations
        if t % 10_000 == 0:
            print(f"    Adam iter {t:,d} – cur {cur_score:.6f} – best {best_score:.6f}")

    return best_a, best_score


# ----------------------------------------------------------------------
# Sub‑gradient descent on the true (non‑smoothed) max‑convolution
# ----------------------------------------------------------------------


def refine_subgradient(seq, time_deadline, max_steps=200_000):
    """
    Deterministic sub‑gradient descent that directly minimizes
    max(a*a) while keeping a on the probability simplex and symmetric.
    """
    a = np.array(seq, dtype=float)
    n = a.shape[0]

    best_a = a.copy()
    best_score = compute_score(best_a)

    step0 = 0.4  # initial step magnitude

    for t in range(1, max_steps + 1):
        if time.time() >= time_deadline:
            break

        conv = convolution_fft(a, a)
        k_max = int(np.argmax(conv))

        # Build the sub‑gradient of the max‑convolution at k_max
        grad = np.zeros_like(a)
        lo = max(0, k_max - (n - 1))
        hi = min(n - 1, k_max)
        j = np.arange(lo, hi + 1)               # indices of a that appear
        i = k_max - j                           # corresponding positions in grad
        grad[i] = 2.0 * a[j]

        # Zero‑mean the direction so that the simplex sum stays unchanged
        direction = -grad + np.mean(grad)

        # Decreasing step size ~ 1/√t
        eta = step0 / np.sqrt(t)
        a_new = a + eta * direction

        # Feasibility restoration
        a_new = np.maximum(a_new, 0.0)
        a_new = symmetrize(a_new)
        a_new = project_to_simplex(a_new)

        cur_score = compute_score(a_new)
        if cur_score < best_score:
            best_score = cur_score
            best_a = a_new.copy()
            # print(f"      refine step {t:,d} – new best {best_score:.6f}")

        a = a_new

    return best_a.tolist(), best_score

        grad = np.zeros_like(a)
        lo = max(0, k_max - (n - 1))
        hi = min(n - 1, k_max)
        j = np.arange(lo, hi + 1)       # indices of a that appear
        i = k_max - j                   # positions in grad
        grad[i] = 2.0 * a[j]

        # zero‑mean direction to keep the sum invariant
        direction = -grad + np.mean(grad)

        eta = step0 / np.sqrt(t)
        a_new = a + eta * direction

        a_new = np.maximum(a_new, 0.0)
        a_new = symmetrize(a_new)
        a_new = project_to_simplex(a_new)

        cur_score = compute_score(a_new)
        if cur_score < best_score:
            best_score = cur_score
            best_a = a_new.copy()

        a = a_new

    return best_a.tolist(), best_score

# Main entry point required by the harness
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative sequence that minimises `evaluate_sequence`.

    The routine performs:
      * several symmetric arcsine starts for multiple lengths,
      * a symmetry‑aware projected Adam optimiser on a smoothed max‑convolution,
      * a final deterministic sub‑gradient polish.
    It prints intermediate scores and respects the time budget.
    """
    np.random.seed(seed)
    start_time = time.time()
    # reserve a small safety margin for the harness
    deadline = start_time + budget_s - 2.0

    best_seq = None
    best_val = float("inf")

    # ------------------------------------------------------------------
    # Optional fallback if a previously discovered candidate exists.
    # ------------------------------------------------------------------
    try:
        fallback = [float(x) for x in height_sequence_1]
        fallback_val = evaluate_sequence(fallback)
        print(f"Fallback score = {fallback_val:.6f}")
        if fallback_val < best_val:
            best_val = fallback_val
            best_seq = fallback
    except NameError:
        pass

    # ------------------------------------------------------------------
    # Multi‑length optimisation (symmetry‑aware Adam)
    # ------------------------------------------------------------------
    lengths = [2048, 4096, 8192, 16384]  # larger n improves the theoretical bound
    for n in lengths:
        if time.time() >= deadline - 15.0:
            break
        print(f"\n=== Optimising length n = {n} ===")

        # symmetric starts; arcsine is already close to optimal
        starts = [
            arcsine_initial(n),
            np.full(n, 1.0 / n),  # uniform (also symmetric)
        ]
        # add a couple of random Dirichlet draws for diversity
        for _ in range(2):
            vec = np.random.dirichlet(np.full(n, 0.7))
            starts.append(vec)

        for idx, init_vec in enumerate(starts, start=1):
            if time.time() >= deadline - 10.0:
                break
            print(f"  start {idx}/{len(starts)}")
            # Allocate iterations based on remaining time and problem size
            remaining = deadline - time.time()
            # Rough heuristic: larger n → fewer iterations
            iters = int(min(300_000, remaining * (2000.0 / n) * 6000.0))
            seq, val = adam_optimize_sym(
                init_a=init_vec,
                time_deadline=deadline,
                max_iter=iters,
                eta0=0.08,
                alpha_start=0.5,
                alpha_end=4000.0,
            )
            print(f"    Adam best score = {val:.6f}")
            if val < best_val:
                best_val = val
                best_seq = seq.tolist()
                print(f"    *** Global best updated to {best_val:.6f} (n={n}) ***")

    # ------------------------------------------------------------------
    # Final deterministic refinement (sub‑gradient on true objective)
    # ------------------------------------------------------------------
    if best_seq is not None and time.time() < deadline - 2.0:
        print("\nRefining best sequence with deterministic sub‑gradient descent")
        refined_seq, refined_val = refine_subgradient(
            best_seq,
            time_deadline=deadline,
            max_steps=250_000,
        )
        print(f"    refined score = {refined_val:.6f}")
        if refined_val < best_val:
            best_val = refined_val
            best_seq = refined_seq

    # ------------------------------------------------------------------
    # Safeguard: if nothing was produced, fall back to a simple uniform vector.
    # ------------------------------------------------------------------
    if best_seq is None:
        best_seq = [1.0 / 256] * 256
        best_val = evaluate_sequence(best_seq)

    final_score = evaluate_sequence(best_seq)
    print(f"\nFinished search. Final reported score = {final_score:.6f}")

    # The harness expects a plain Python list of floats.
    return best_seq
