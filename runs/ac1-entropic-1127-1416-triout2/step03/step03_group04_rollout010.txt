"""
Solution for minimizing E(a) = 2·n·max(a*a) / (∑a)².

The algorithm consists of three phases:

1. Broad parametric sweep (Beta, exponential, arcsine, mixtures) over a few
   problem sizes (512–4096). The three best candidates are kept.

2. For each kept candidate a short sub‑gradient descent on the exact
   max‑convolution is performed.  The sub‑gradient of the max is a simple
   closed‑form which allows cheap O(n) updates.  A decreasing step size
   (ηₜ = η₀ / √t) together with Euclidean projection onto the probability
   simplex keeps the iterate feasible.

3. A final stochastic pair‑wise mass‑transfer hill‑climb finishes the search.
   If time permits the best vector is linearly up‑sampled to a larger length
   (8192) before the hill‑climb – this often reduces the discretisation error.

All phases respect the global time budget (default 1000 s).  The function
`propose_candidate` returns the best sequence found; the competition harness
will feed it to `evaluate_sequence`.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Linear algebra utilities
# ----------------------------------------------------------------------


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex (∑v = 1, v ≥ 0)."""
    # Implementation follows the algorithm of Wang & Carreira‑Perpiñán (2013)
    if np.all(v <= 0):
        # Degenerate case – return the uniform distribution.
        return np.full_like(v, 1.0 / v.shape[0], dtype=float)

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho = np.where(u - cssv / np.arange(1, v.shape[0] + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def next_pow2(x: int) -> int:
    """Return the smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full (linear) convolution using a real FFT."""
    m = len(a) + len(b) - 1
    n_fft = next_pow2(m)
    A = np.fft.rfft(a, n=n_fft)
    B = np.fft.rfft(b, n=n_fft)
    C = A * B
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


def compute_score(a: np.ndarray) -> float:
    """
    Compute the objective 2·n·max(a*a) / (∑a)².
    The input is assumed to be a probability vector (∑a = 1).  The function
    mirrors `evaluate_sequence` for speed.
    """
    a = np.asarray(a, dtype=float)
    n = a.shape[0]
    conv = convolution_fft(a, a)
    max_val = float(np.max(conv))
    s = float(np.sum(a))
    if s < 1e-12:
        return float("inf")
    return 2.0 * n * max_val / (s * s)


# ----------------------------------------------------------------------
# Families of candidate step‑functions
# ----------------------------------------------------------------------


def arcsine_initial(n: int) -> np.ndarray:
    """Arcsine‑type shape: a_i ∝ 1 / √((i+½)(n‑½‑i))."""
    i = np.arange(n, dtype=float)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - 0.5 - i))
    return vals / np.sum(vals)


def beta_sequence(n: int, alpha: float, beta: float | None = None) -> np.ndarray:
    """
    Symmetric or asymmetric Beta‑type shape:
        a_i ∝ (i+½)^α · (n‑½‑i)^β .
    The parameters may be negative (edges become heavier) or positive.
    """
    if beta is None:
        beta = alpha
    i = np.arange(n, dtype=float)
    left = np.power(i + 0.5, alpha)
    right = np.power(n - 0.5 - i, beta)
    seq = left * right
    # Guard against overflow / underflow.
    seq = np.clip(seq, 0.0, 1e12)
    total = np.sum(seq)
    if total == 0.0:
        seq = np.full(n, 1.0 / n)
    else:
        seq = seq / total
    return seq


def exponential_sequence(n: int, lam: float) -> np.ndarray:
    """Exponential decay centred in the middle: a_i ∝ exp(-lam·|i-(n‑1)/2|)."""
    i = np.arange(n, dtype=float)
    centre = (n - 1) / 2.0
    seq = np.exp(-lam * np.abs(i - centre))
    return seq / np.sum(seq)


def upsample_sequence(seq: np.ndarray, new_n: int) -> np.ndarray:
    """
    Linear interpolation of a probability vector to a larger length.
    The result is renormalised to sum to one.
    """
    old_n = len(seq)
    old_x = (np.arange(old_n) + 0.5) / old_n
    new_x = (np.arange(new_n) + 0.5) / new_n
    new_vals = np.interp(new_x, old_x, seq)
    return new_vals / np.sum(new_vals)


# ----------------------------------------------------------------------
# Sub‑gradient descent on the exact max‑convolution
# ----------------------------------------------------------------------


def subgradient_descent(
    initial: np.ndarray,
    deadline: float,
    max_iter: int = 30_000,
    eta0: float = 0.4,
) -> tuple[list[float], float]:
    """
    Minimise the exact objective using a simple sub‑gradient of the max
    convolution.  The update rule is projected onto the simplex after each step.
    """
    a = project_to_simplex(initial.astype(float).copy())
    n = a.shape[0]
    best_a = a.copy()
    best_score = compute_score(a)

    t = 1
    while t <= max_iter and time.time() < deadline:
        # Full convolution and its maximum.
        conv = convolution_fft(a, a)
        # Index of a (any) maximum – a deterministic choice.
        k = int(np.argmax(conv))

        # Gradient of max(conv) with respect to a (sub‑gradient).
        ind = np.arange(n)
        j = k - ind
        mask = (j >= 0) & (j < n)
        g = np.zeros_like(a)

        # General case: contribution 2 * a_{k-i}
        g[mask] = 2.0 * a[j[mask]]
        # Diagonal correction where i == k-i (i.e. k even and i == k/2)
        diag = mask & (j == ind)
        if np.any(diag):
            g[diag] = a[ind[diag]]

        # Step size schedule ηₜ = η₀ / √t
        eta = eta0 / np.sqrt(t)
        a = a - eta * g
        a = project_to_simplex(a)

        cur_score = compute_score(a)
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()

        t += 1

        # Very occasional tiny random perturbation to escape shallow plateaus.
        if t % 5000 == 0 and np.random.rand() < 0.2:
            a = a + 1e-5 * np.random.randn(n)
            a = project_to_simplex(a)

    return best_a.tolist(), best_score


# ----------------------------------------------------------------------
# Stochastic hill‑climbing refinement
# ----------------------------------------------------------------------


def hill_climb(
    seq,
    time_deadline: float,
    max_steps: int = 60_000,
    init_step: float = 0.07,
):
    """
    Pair‑wise mass‑transfer hill‑climb.
    At each step a random source index i donates a fraction of its current
    mass to a random target j.  The move is kept only if the true objective
    improves.  The step size is adapted heuristically.
    """
    a = np.array(seq, dtype=float)
    a = project_to_simplex(a)
    best_a = a.copy()
    best_score = compute_score(best_a)
    n = len(a)
    step = init_step

    while max_steps > 0 and time.time() < time_deadline:
        i, j = np.random.randint(n, size=2)
        if i == j:
            continue
        delta = step * best_a[i]
        if delta <= 0.0:
            continue
        cand = best_a.copy()
        cand[i] -= delta
        cand[j] += delta
        cand = project_to_simplex(cand)
        cur_score = compute_score(cand)
        if cur_score < best_score:
            best_score = cur_score
            best_a = cand
            step = min(step * 1.08, 0.5)
        else:
            step *= 0.96
        max_steps -= 1
    return best_a.tolist(), best_score

"""
Solution for minimizing

    E(a) = 2·n·max(a*a) / (∑a)² ,

where `a` is a non‑negative step‑function (list of floats).

The algorithm works in four stages, all respecting the global time budget
(default 1000 s):

1. **Broad parametric sweep** – families based on Beta‑type, arcsine,
   exponential, and mixtures are evaluated for a few lengths
   (512‑4096).  The five most promising candidates are retained.

2. **Sub‑gradient descent** – for each retained candidate a cheap
   sub‑gradient of the exact max‑convolution is taken (the derivative of
   `max_k (a*a)_k` is known analytically).  After each step the vector is
   projected onto the probability simplex.  A decreasing step size
   `ηₜ = η₀/√t` is used.

3. **Upsampling (optional)** – the best vector is linearly interpolated to
   a larger size (8192).  This often reduces the discretisation error.

4. **Final hill‑climb** – a stochastic pair‑wise mass‑transfer hill‑climb
   refines the solution using the true (non‑smoothed) objective.

The function `propose_candidate` is the required entry‑point; it returns
the best sequence found, which the competition harness will feed to
`evaluate_sequence`.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Linear‑algebra utilities
# ----------------------------------------------------------------------


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex (∑v = 1, v ≥ 0)."""
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / v.shape[0], dtype=float)

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho = np.where(u - cssv / np.arange(1, v.shape[0] + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full linear convolution using a real FFT (fast for large vectors)."""
    m = len(a) + len(b) - 1
    n_fft = next_pow2(m)
    A = np.fft.rfft(a, n=n_fft)
    B = np.fft.rfft(b, n=n_fft)
    C = A * B
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


def compute_score(a: np.ndarray) -> float:
    """
    Compute the objective 2·n·max(a*a) / (∑a)².
    The input is assumed to be a probability vector (∑a = 1, a ≥ 0).
    """
    a = np.asarray(a, dtype=float)
    n = a.shape[0]
    conv = convolution_fft(a, a)
    max_val = float(np.max(conv))
    s = float(np.sum(a))
    if s < 1e-12:
        return float("inf")
    return 2.0 * n * max_val / (s * s)


# ----------------------------------------------------------------------
# Families of candidate step‑functions
# ----------------------------------------------------------------------


def arcsine_initial(n: int) -> np.ndarray:
    """Arcsine‑type shape a_i ∝ 1 / √((i+½)(n‑½‑i))."""
    i = np.arange(n, dtype=float)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - 0.5 - i))
    return vals / np.sum(vals)


def beta_sequence(n: int, alpha: float, beta: float | None = None) -> np.ndarray:
    """
    Beta‑type shape a_i ∝ (i+½)^α · (n‑½‑i)^β .
    Parameters may be negative (heavier edges) or positive.
    """
    if beta is None:
        beta = alpha
    i = np.arange(n, dtype=float)
    left = np.power(i + 0.5, alpha)
    right = np.power(n - 0.5 - i, beta)
    seq = left * right
    seq = np.clip(seq, 0.0, 1e12)
    total = np.sum(seq)
    if total == 0.0:
        seq = np.full(n, 1.0 / n)
    else:
        seq = seq / total
    return seq


def exponential_sequence(n: int, lam: float) -> np.ndarray:
    """A symmetric exponential decay centred in the middle."""
    i = np.arange(n, dtype=float)
    centre = (n - 1) / 2.0
    seq = np.exp(-lam * np.abs(i - centre))
    return seq / np.sum(seq)


def upsample_sequence(seq: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation of a probability vector to a larger size."""
    old_n = len(seq)
    old_x = (np.arange(old_n) + 0.5) / old_n
    new_x = (np.arange(new_n) + 0.5) / new_n
    new_vals = np.interp(new_x, old_x, seq)
    return new_vals / np.sum(new_vals)

# Sub‑gradient descent on the exact max‑convolution
# ----------------------------------------------------------------------


def subgradient_descent(
    initial: np.ndarray,
    deadline: float,
    max_iter: int = 30_000,
    eta0: float = 0.4,
) -> tuple[list[float], float]:
    """
    Perform sub‑gradient descent on  E(a)  using the exact sub‑gradient of
    max(a*a).  After each update the vector is projected onto the simplex.
    """
    a = project_to_simplex(initial.astype(float).copy())
    n = a.shape[0]
    best_a = a.copy()
    best_score = compute_score(a)

    t = 1
    while t <= max_iter and time.time() < deadline:
        conv = convolution_fft(a, a)
        k = int(np.argmax(conv))

        # Sub‑gradient of max(conv) w.r.t. a
        ind = np.arange(n)
        j = k - ind
        mask = (j >= 0) & (j < n)
        g = np.zeros_like(a)
        g[mask] = 2.0 * a[j[mask]]
        diag = mask & (j == ind)
        if np.any(diag):
            g[diag] = a[ind[diag]]

        eta = eta0 / np.sqrt(t)
        a = a - eta * g

        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()

        t += 1

        # Occasional tiny random jitter to escape shallow plateaus
        if t % 5000 == 0 and np.random.rand() < 0.2:

            a = project_to_simplex(a)

    return best_a.tolist(), best_score

    n = len(a)

    print(f"\nSearch finished. Final reported score = {final_score:.6f}")
    return best_seq
