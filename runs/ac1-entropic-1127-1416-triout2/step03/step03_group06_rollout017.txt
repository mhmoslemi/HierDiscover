"""
Propose a step‑function that minimizes ``evaluate_sequence``.

Algorithm
---------
1. Determine the length ``n`` (use the global *height_sequence_1* if it exists,
   otherwise fall back to 5146).
2. Build a small set of diverse seed vectors (linear ramp, uniform,
   exponential, polynomial, Beta‑distribution, random Dirichlet).
   Keep the best (lowest objective) among them.
3. Starting from the best seed, run a long projected Adam optimisation on
   a *smooth* max–objective:
        F(a) = (2·n/α)·log(∑_k exp(α·(a*a)[k])),
   whose gradient is
        ∇F_i = 4·n·∑_k w_k·a_{k-i},
   where w is the soft‑max of the convolution entries.
   The smoothness parameter α is increased from 200 to 5000 during the run.
4. After each Adam step project the iterate onto the probability simplex
   (Duchi et al. 2008) to enforce non‑negativity and unit sum.
5. Keep the best vector ever seen (according to the *exact* objective
   ``2·n·max(a*a)``).  If the algorithm makes no improvement for a long time,
   inject a small random perturbation to escape a plateau.
6. Stop when the time budget (≈ 1000 s) is exhausted or the target score
   1.503 is reached.  Return the best sequence as a plain ``list[float]``.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Utility functions – all are top‑level (no closures)
# ----------------------------------------------------------------------


def simplex_projection(v):
    """Euclidean projection of v onto the probability simplex."""
    v = np.asarray(v, dtype=float)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1] - 1
    theta = cssv[cond][-1] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def next_pow_two(x: int) -> int:
    """Small helper: next power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def compute_objective(a: np.ndarray) -> float:
    """Exact objective: 2·n·max(a*a)   (assumes Σa = 1)."""
    n = a.shape[0]
    L = next_pow_two(2 * n - 1)
    fft_a = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(fft_a * fft_a, n=L)[: 2 * n - 1]
    return float(2.0 * n * conv.max())


def compute_gradient(a: np.ndarray, alpha: float):
    """
    Gradient of the smoothed max objective
        F(a) = (2·n/α)·log Σ_k exp(α·(a*a)[k]).
    Returns (grad, max_conv) where grad has shape (n,).
    """
    n = a.shape[0]

    # ---- convolution a*a -------------------------------------------------
    L1 = next_pow_two(2 * n - 1)
    fft_a = np.fft.rfft(a, n=L1)
    conv = np.fft.irfft(fft_a * fft_a, n=L1)[: 2 * n - 1]

    max_c = conv.max()

    # ---- soft‑max weights -----------------------------------------------
    # exp(α·(c - max_c)) is safe (max entry => 1, others ≤ 1)
    w = np.exp(alpha * (conv - max_c))
    w_sum = w.sum()
    if w_sum == 0.0:                # should never happen, but be safe
        w = np.full_like(w, 1.0 / w.size)
    else:
        w /= w_sum

    # ---- gradient = 4·n· (w * a_rev)  (convolution) -----------------------
    a_rev = a[::-1]
    L2 = next_pow_two((2 * n - 1) + n - 1)          # length for w * a_rev
    fft_w = np.fft.rfft(w, n=L2)
    fft_rev = np.fft.rfft(a_rev, n=L2)
    conv_full = np.fft.irfft(fft_w * fft_rev, n=L2)
    conv_full = conv_full[: (3 * n - 2)]            # keep the needed part

    # slice indices n‑1 … n‑1+n‑1 give the required gradient entries
    grad = 4.0 * n * conv_full[(n - 1) : (n - 1) + n]
    return grad, max_c


def beta_sequence(alpha: float, beta: float, n: int) -> np.ndarray:
    """Discretised Beta(α,β) density on the mid‑points, normalised to sum 1."""
    x = (np.arange(n, dtype=float) + 0.5) / n
    logw = (alpha - 1.0) * np.log(x) + (beta - 1.0) * np.log(1.0 - x)
    logw = np.clip(logw, -700.0, 700.0)        # avoid overflow in exp
    w = np.exp(logw)
    if not np.isfinite(w).all() or w.sum() == 0.0:
        w = np.ones_like(w)
    w /= w.sum()
    return w


def exponential_shape(lam: float, n: int) -> np.ndarray:
    """a_i ∝ exp(-lam·i), normalised to sum 1."""
    i = np.arange(n, dtype=float)
    w = np.exp(-lam * i)
    w /= w.sum()
    return w


def polynomial_shape(p: float, n: int) -> np.ndarray:
    """a_i ∝ 1 - (i/(n‑1))**p, clipped to non‑negative, normalised."""
    i = np.arange(n, dtype=float)
    frac = i / (n - 1)
    w = 1.0 - np.power(frac, p)
    w = np.clip(w, 0.0, None)
    w /= w.sum()
    return w

def propose_candidate(seed: int = 0, budget_s: int = 1000):
    """Return a list of non‑negative heights that (hopefully) minimises the
    external ``evaluate_sequence``.  Runs for at most ``budget_s`` seconds."""
    np.random.seed(seed)
    start_time = time.time()
    # keep a few seconds margin for the outer harness
    deadline = start_time + budget_s - 5.0

    # --------------------------------------------------------------
    # 1. Length of the step function
    # --------------------------------------------------------------
    try:
        n = len(height_sequence_1)        # type: ignore[name-defined]
    except Exception:
        n = 5146                          # fallback used in the prompt

    # --------------------------------------------------------------
    # 2. Storage for the best solution found so far
    # --------------------------------------------------------------
    best_seq = None                       # numpy array, normalised to sum 1
    best_score = np.inf

    # --------------------------------------------------------------
    # 3. Helper to update the global best
    # --------------------------------------------------------------
    def update_best(candidate: np.ndarray):
        nonlocal best_seq, best_score
        sc = compute_objective(candidate)
        if sc < best_score:
            best_score = sc
            best_seq = candidate.copy()
            print(f"[seed] new best {sc:.6f}")

    # --------------------------------------------------------------
    # 4. Generate an initial collection of diverse seeds
    # --------------------------------------------------------------
    # linear decreasing ramp (the analytical construction used previously)
    ramp = np.arange(n, 0, -1, dtype=float)
    ramp = 2.0 * ramp / (n * (n + 1))
    ramp /= ramp.sum()
    update_best(ramp)

    # uniform vector
    uniform = np.full(n, 1.0 / n)
    update_best(uniform)

    # exponential family
    for lam in np.linspace(0.001, 5.0, 20):
        a_exp = exponential_shape(lam, n)
        update_best(a_exp)

    # polynomial family
    for p in [0.5, 1.0, 1.5, 2.0, 3.0]:
        a_poly = polynomial_shape(p, n)
        update_best(a_poly)

    # systematic Beta sweep (coarse)
    alphas = np.linspace(0.2, 5.0, 12)
    betas = np.linspace(0.2, 5.0, 12)
    for α in alphas:
        for β in betas:
            if time.time() >= deadline:
                break
            a_beta = beta_sequence(α, β, n)
            update_best(a_beta)
        if time.time() >= deadline:
            break

    # random Dirichlet‑like and random Beta seeds
    for _ in range(200):
        if time.time() >= deadline:
            break
        # Dirichlet‑like
        conc = np.random.uniform(0.1, 3.0, size=n)
        a_dir = np.random.gamma(conc, 1.0)
        a_dir /= a_dir.sum()
        update_best(a_dir)

        # random Beta
        α = np.random.uniform(0.05, 5.0)
        β = np.random.uniform(0.05, 5.0)
        a_beta = beta_sequence(α, β, n)
        update_best(a_beta)

    # --------------------------------------------------------------
    # 5. Adam optimisation of the smoothed max‑objective
    # --------------------------------------------------------------
    if best_seq is None:                # very unlikely, but be safe
        best_seq = uniform.copy()
        best_score = compute_objective(best_seq)

    a = best_seq.copy()                 # current iterate
    m = np.zeros_like(a)                # first moment estimate
    v = np.zeros_like(a)                # second moment estimate
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    base_lr = 0.05
    alpha_start = 200.0
    alpha_end = 5000.0
    max_iter = 500_000
    stagnation = 0
    iteration = 0

    while time.time() < deadline and iteration < max_iter:
        iteration += 1

        # linear schedule for the smoothing parameter α
        alpha = alpha_start + (alpha_end - alpha_start) * min(1.0,
                                                             iteration / 250_000)

        # ---- gradient of the smoothed max ---------------------------------
        grad, max_conv = compute_gradient(a, alpha)

        # ---- Adam update --------------------------------------------------
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1 ** iteration)
        v_hat = v / (1.0 - beta2 ** iteration)
        step = base_lr / (np.sqrt(v_hat) + eps)
        a_next = a - step * m_hat
        a_next = simplex_projection(a_next)

        # ---- exact objective (the one evaluated by evaluate_sequence) -----
        cur_score = compute_objective(a_next)

        if cur_score < best_score - 1e-12:
            best_score = cur_score
            best_seq = a_next.copy()
            stagnation = 0
            print(f"Iter {iteration:7d}: new best {best_score:.6f}")
        else:
            stagnation += 1

        # ---- occasional random perturbation to leave plateaus -------------
        if stagnation >= 50_000:
            sigma = max(0.01, 0.5 / np.sqrt(iteration))
            noise = sigma * np.random.randn(n) * a_next
            a_next = a_next + noise
            a_next = np.clip(a_next, 0.0, None)
            a_next /= a_next.sum()
            a_next = simplex_projection(a_next)
            stagnation = 0
            print(f"Iter {iteration:7d}: perturbation applied, "
                  f"current best {best_score:.6f}")

        a = a_next

        # early exit if target already reached
        if best_score <= 1.503:
            print(f"Target reached at iteration {iteration}, score = {best_score:.6f}")
            break

    # --------------------------------------------------------------
    # 6. Return the best sequence discovered
    # --------------------------------------------------------------
    print(f"Finished after {iteration} iterations, best score = {best_score:.6f}")
    return [float(v) for v in best_seq.tolist()]
