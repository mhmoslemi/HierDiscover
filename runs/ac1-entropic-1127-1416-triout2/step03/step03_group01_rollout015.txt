"""
Propose a low‑discrepancy step‑function via a hybrid search:

1. **Exploration** – For a few lengths (1024, 2048, 4096) we generate
   symmetric Beta‑family shapes (a_i ∝ (i+ε)^{α‑1}(n‑1‑i+ε)^{α‑1})
   together with the classic arcsine starter.  The best candidate per
   length is kept using the fast surrogate ``compute_score`` (exact for
   a sum‑one vector).

2. **Exact sub‑gradient descent** –  The objective after normalising the
   sum to 1 is ``f(a)=2·n·max(a*a)``.  Its sub‑gradient at a point is
   ``2·a_{k−i}`` averaged over all indices *k* attaining the current
   maximum of the self‑convolution.  We perform a projected sub‑gradient
   descent with a diminishing step size, occasional Gaussian noise and
   projection onto the probability simplex.  The best vector found is
   retained.

3. **Final polishing** –  The best vector from the previous stage is
   refined with the smooth Adam optimiser used in the reference code
   (log‑sum‑exp surrogate of the max).  This gives a fine‑grained
   improvement.

All helper functions are top‑level; no closures or file I/O are used.
The routine respects a 1000 s budget and always returns a non‑empty,
non‑negative sequence.
"""

import time
import numpy as np
from typing import List, Tuple

# ----------------------------------------------------------------------
# Low‑level utilities (no closures)
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """
    Real‑FFT convolution of two 1‑D arrays.
    Returns the linear convolution (length = len(a)+len(b)-1).
    """
    m = len(a) + len(b) - 1
    n_fft = next_pow2(m)
    A = np.fft.rfft(a, n=n_fft)
    B = np.fft.rfft(b, n=n_fft)
    C = A * B
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


def compute_score(a: np.ndarray) -> float:
    """
    Fast exact surrogate for ``evaluate_sequence`` when the vector
    is normalised to sum = 1.  Returns 2·n·max(a*a).
    """
    n = len(a)
    conv = convolution_fft(a, a)
    max_val = float(np.max(conv))
    return 2.0 * n * max_val


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the probability simplex
    {x ≥ 0, Σx = 1}.  Implements the algorithm of Duchi et al. (2008).
    """
    if np.all(v <= 0):
        # uniform fallback – never happens in practice
        return np.full_like(v, 1.0 / v.size)

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho = np.where(u - cssv / np.arange(1, v.size + 1) > 0)[0]
    if len(rho) == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def arcsine_initial(n: int) -> np.ndarray:
    """Classic arcsine shape: a_i ∝ 1/√((i+½)*(n‑½‑i))."""
    i = np.arange(n, dtype=float)
    eps = 0.25
    vals = 1.0 / np.sqrt((i + 0.5) * (n - 0.5 - i) + eps)
    vals /= vals.sum()
    return vals


def generate_beta_shape(
    n: int, alpha: float, beta: float, eps: float
) -> np.ndarray:
    """
    Symmetric beta‑family density on {0,…,n‑1}:
        a_i ∝ (i+ε)^{α‑1}·(n‑1‑i+ε)^{β‑1}.
    The result is normalised to Σa = 1.
    """
    i = np.arange(n, dtype=float)
    left = (i + eps) ** (alpha - 1.0)
    right = (n - 1 - i + eps) ** (beta - 1.0)
    base = left * right
    total = base.sum()
    if total <= 0.0 or not np.isfinite(total):
        # fallback to uniform
        return np.full(n, 1.0 / n)
    return base / total

# ----------------------------------------------------------------------


def smooth_gradient(a: np.ndarray, lam: float) -> Tuple[np.ndarray, float]:
    """
    Gradient of the soft‑max surrogate of max(a*a) for temperature λ.
    Returns (grad, smooth objective value).
    """
    n = len(a)
    conv = convolution_fft(a, a)               # length 2n‑1
    max_c = np.max(conv)

    exp_term = np.exp(lam * (conv - max_c))
    sum_exp = np.sum(exp_term)
    w = exp_term / sum_exp                     # soft‑max weights
    B = max_c + np.log(sum_exp) / lam          # smooth max value

    # Weighted sub‑gradients via convolution with reversed weights
    w_rev = w[::-1]
    conv_rev = convolution_fft(a, w_rev)       # length 3n‑2
    L = 2 * n - 1
    weighted_grad = 2.0 * conv_rev[L - 1 - np.arange(n)]

    s = np.sum(a)
    term1 = (2.0 * n / (s ** 2)) * weighted_grad
    term2 = -(4.0 * n * B / (s ** 3)) * np.ones_like(a)
    grad = term1 + term2
    return grad, B


def adam_optimize(
    initial_a: np.ndarray,
    time_deadline: float,
    max_iter: int = 100_000,
    lr0: float = 0.2,
    lam_start: float = 5.0,
    lam_end: float = 30.0,
    noise_every: int = 2000,
    noise_scale: float = 5e-5,
) -> Tuple[np.ndarray, float]:
    """
    Adam optimiser on the smooth surrogate.  Returns the best vector
    (projected onto the simplex) and its fast score.
    """
    a = project_to_simplex(initial_a.astype(float).copy())
    best_a = a.copy()
    best_sc = compute_score(a)

    beta1, beta2 = 0.9, 0.999
    eps = 1e-8
    m = np.zeros_like(a)
    v = np.zeros_like(a)

    for it in range(1, max_iter + 1):
        if time.time() >= time_deadline:
            break

        lam = lam_start + (lam_end - lam_start) * min(1.0, it / max_iter)
        grad, _ = smooth_gradient(a, lam)

        # Adam moments
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)

        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        step = lr0 * m_hat / (np.sqrt(v_hat) + eps)
        a_new = a - step

        if noise_every and (it % noise_every == 0):
            a_new += np.random.randn(len(a)) * noise_scale

        a_new = project_to_simplex(a_new)

        cur_sc = compute_score(a_new)
        if cur_sc < best_sc:
            best_sc = cur_sc
            best_a = a_new.copy()

        a = a_new

        if it % 5000 == 0:
            print(f"    Adam iter={it:,d} cur={cur_sc:.6f} best={best_sc:.6f}")

    return best_a, best_sc


# ----------------------------------------------------------------------
# Exact sub‑gradient descent on the true max‑convolution objective
# ----------------------------------------------------------------------


def subgradient_descent(
    initial_a: np.ndarray,
    time_deadline: float,
    max_iter: int = 200_000,
    step0: float = 0.3,
    noise_every: int = 1000,
    noise_scale: float = 1e-4,
) -> Tuple[np.ndarray, float]:
    """
    Projected sub‑gradient descent on the exact objective
    f(a)=2·n·max(a*a) (sum of a is 1).  Returns (best_a, best_score).
    """
    a = project_to_simplex(initial_a.astype(float).copy())
    n = len(a)
    best_a = a.copy()
    best_sc = compute_score(a)

    cur_sc = best_sc

    for it in range(1, max_iter + 1):
        if time.time() >= time_deadline:
            break

        # current convolution and its maximum
        conv = convolution_fft(a, a)
        max_val = np.max(conv)
        max_idxs = np.where(conv == max_val)[0]

        # sub‑gradient: average over all active indices
        grad = np.zeros_like(a)
        for k in max_idxs:
            i_start = max(0, k - (n - 1))
            i_end = min(k, n - 1)
            i_range = np.arange(i_start, i_end + 1)
            grad[i_range] += a[k - i_range]

        grad = 2.0 * grad / len(max_idxs)

        # diminishing step size
        eta = step0 / np.sqrt(it)

        a_candidate = a - eta * grad
        a_candidate = project_to_simplex(a_candidate)

        # occasional random perturbation to escape flat regions
        if noise_every and (it % noise_every == 0):
            a_candidate += np.random.randn(n) * noise_scale
            a_candidate = project_to_simplex(a_candidate)

        cand_sc = compute_score(a_candidate)

        # keep global best
        if cand_sc < best_sc:
            best_sc = cand_sc
            best_a = a_candidate.copy()
            print(f"    Subgrad iter={it:,d} best={best_sc:.6f}")

        # accept only if we made progress (simple greedy scheme)
        if cand_sc <= cur_sc:
            a = a_candidate
            cur_sc = cand_sc
        # otherwise keep the old a (no back‑tracking for simplicity)

    return best_a, best_sc


# ----------------------------------------------------------------------
# Main entry point required by the competition harness
# ----------------------------------------------------------------------


def propose_candidate(
    seed: int = 42, budget_s: int = 1000, **kwargs
) -> List[float]:
    """
    Search for a non‑negative step sequence with a low
    ``evaluate_sequence`` value.  The method consists of
    (i) exploration of Beta‑family shapes,
    (ii) exact sub‑gradient descent,
    (iii) final Adam polishing.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 2.0  # tiny safety margin

    best_seq: List[float] = []
    best_val: float = float("inf")

    # ------------------------------------------------------------------
    # 0. Optional fallback from a previous run (if the variable exists)
    # ------------------------------------------------------------------
    try:
        fallback = np.array([float(x) for x in height_sequence_1], dtype=float)  # type: ignore
        fallback = np.maximum(fallback, 0.0)
        fallback = fallback / fallback.sum()
        val = evaluate_sequence(fallback.tolist())
        print(f"Fallback (height_sequence_1) score = {val:.6f}")
        if val < best_val:
            best_val = val
            best_seq = fallback.tolist()
    except Exception:
        pass

    # ------------------------------------------------------------------
    # 1. Exploration over a few lengths
    # ------------------------------------------------------------------
    lengths = [1024, 2048, 4096]

    for n in lengths:
        if time.time() >= deadline - 5.0:
            break
        print(f"\n=== Exploring length n = {n} ===")

        # start from the classic arcsine shape
        a_local = arcsine_initial(n)
        sc_local = compute_score(a_local)
        best_local_a = a_local.copy()
        best_local_sc = sc_local

        # ------------------------------------------------------------------
        # 1b. Random Beta-family sampling (fast surrogate)
        # ------------------------------------------------------------------
        max_samples = 800
        for _ in range(max_samples):
            if time.time() >= deadline - 5.0:
                break
            # Symmetric shape – use the same α for both sides
            alpha = rng.uniform(0.05, 1.5)
            beta = alpha
            eps = rng.uniform(0.001, 0.01 * n)
            cand = generate_beta_shape(n, alpha, beta, eps)
            sc = compute_score(cand)
            if sc < best_local_sc:
                best_local_sc = sc
                best_local_a = cand.copy()

        # ------------------------------------------------------------------
        # 2. Exact sub‑gradient refinement
        # ------------------------------------------------------------------
        if time.time() < deadline - 5.0:
            remaining = deadline - time.time()
            # allow roughly 30 iters per second for this phase
            max_iter = int(min(200_000, remaining * 30))
            print(f"  Sub‑gradient refinement (max_iter={max_iter}) …")
            refined_a, refined_sc = subgradient_descent(
                best_local_a,
                time_deadline=deadline,
                max_iter=max_iter,
                step0=0.3,
                noise_every=1000,
                noise_scale=1e-4,
            )
            if refined_sc < best_local_sc:
                best_local_sc, best_local_a = refined_sc, refined_a.copy()
                print(f"  → refined sub‑grad score = {refined_sc:.6f}")

        # ------------------------------------------------------------------
        # 3. Evaluate with the official (slow) evaluator once
        # ------------------------------------------------------------------
        if time.time() < deadline - 5.0:
            val = evaluate_sequence(best_local_a.tolist())
            print(f"  → official score = {val:.6f}")
            if val < best_val:
                best_val = val
                best_seq = best_local_a.tolist()
                print(f"  *** New global best: {best_val:.6f} (n={n}) ***")

    # ------------------------------------------------------------------
    # 4. Final polishing of the overall best vector with Adam
    # ------------------------------------------------------------------
    if best_seq and time.time() < deadline - 5.0:
        a = np.array(best_seq, dtype=float)
        remaining = deadline - time.time()
        max_iter = int(min(300_000, remaining * 30))
        print(f"\nFinal polishing with Adam (max_iter={max_iter}) …")
        a_refined, _ = adam_optimize(
            initial_a=a,
            time_deadline=deadline,
            max_iter=max_iter,
            lr0=0.15,
            lam_start=5.0,
            lam_end=50.0,
            noise_every=800,
            noise_scale=2e-5,
        )
        final_val = evaluate_sequence(a_refined.tolist())
        print(f"Polished final score = {final_val:.6f}")
        if final_val < best_val:
            best_val = final_val
            best_seq = a_refined.tolist()
            print("*** Updated global best after final polish ***")

    # ------------------------------------------------------------------
    # 5. Guard – guarantee a return value
    # ------------------------------------------------------------------
    if not best_seq:
        best_seq = [1.0] * 256
        best_val = evaluate_sequence(best_seq)

    print(f"\nFinished search. Best reported score = {best_val:.6f}")
    return best_seq
