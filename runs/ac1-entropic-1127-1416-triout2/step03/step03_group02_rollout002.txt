def project_simplex(v):
    # returns projection of v onto the probability simplex {x>=0, sum=1}
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.where(u + (1 - cssv) / np.arange(1, n+1) > 0)[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1)
    w = np.maximum(v - theta, 0)
    return w

def softmax_optimize(initial_a, deadline, max_iter=50000, lam_start=5., lam_end=400., lam_ramp_iters=10000, eta0=0.6, noise_every=2000, noise_scale=1e-5):
    a = project_simplex(initial_a.astype(float).copy())
    best_a = a.copy()
    best_sc = compute_score(a)

    n = len(a)

    for it in range(1, max_iter+1):
        if time.time() >= deadline:
            break
        lam = lam_start + (lam_end - lam_start) * min(1.0, it/lam_ramp_iters)
        conv = convolution_fft(a, a)
        w = softmax_weights(conv, lam)
        grad = gradient_from_weights(a, w)

"""
Hybrid optimisation for the step‑function minimisation problem.

The search consists of three stages:

1. **Candidate generation** – for several lengths a small library of
   parametric families (uniform, arcsine, power‑law, beta‑like,
   exponential) and random convex mixtures is sampled.  The best few
   (according to a fast internal objective) are retained.

2. **Soft‑max exponential‑gradient descent** – each retained candidate
   is refined by minimising a smooth log‑sum‑exp approximation of the
   max‑convolution.  The temperature λ is annealed from a modest value
   up to several hundred, driving the update toward the true
   non‑smooth objective.

3. **Exact sub‑gradient refinement** – after the soft‑max phase a
   pure sub‑gradient descent on the true max‑convolution is performed.
   The gradient is the sum of the shifted vectors for all lags attaining
   the current maximum; a Euclidean projection onto the probability
   simplex guarantees feasibility.

A final polishing (soft‑max + sub‑gradient) is applied to the best
overall sequence if time permits.  All operations stay on the
probability simplex (non‑negative, unit sum) and respect the 1000 s
budget.
"""

import time
import numpy as np
from typing import List, Tuple

# ----------------------------------------------------------------------
# Low‑level numeric utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Linear convolution via real FFT (full length, zero‑padded to power‑of‑2)."""
    m = len(a) + len(b) - 1
    n_fft = next_pow2(m)
    A = np.fft.rfft(a, n=n_fft)
    B = np.fft.rfft(b, n=n_fft)
    C = A * B
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


def compute_score(a: np.ndarray) -> float:
    """
    Official objective (scale‑invariant):
        2·n·max(a*a) / (sum(a)²)
    a must be non‑negative.  The function works for any vector.
    """
    n = len(a)
    conv = convolution_fft(a, a)
    max_val = float(np.max(conv))
    s = float(np.sum(a))
    if s < 1e-12:
        return float("inf")
    return 2.0 * n * max_val / (s * s)


# ----------------------------------------------------------------------
# Parametric families for initial candidates
# ----------------------------------------------------------------------


def arcsine_initial(n: int) -> np.ndarray:
    """Discrete arcsine density, normalised to unit sum."""
    i = np.arange(n, dtype=float)
    eps = 0.25  # prevents division by zero at the endpoints
    vals = 1.0 / np.sqrt((i + 0.5) * (n - 0.5 - i) + eps)
    return vals / vals.sum()


def power_law_shape(n: int, p: float, eps: float) -> np.ndarray:
    """Symmetric power‑law a_i ∝ ((i+ε)(n‑1‑i+ε))⁻ᵖ."""
    i = np.arange(n, dtype=float)
    base = ((i + eps) * (n - 1 - i + eps)) ** (-p)
    s = base.sum()
    if s == 0.0:
        return np.full(n, 1.0 / n)
    return base / s


def beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    """Beta‑like density on a discrete grid."""
    i = np.arange(n, dtype=float)
    base = (i + 0.5) ** (alpha - 1) * (n - 1 - i + 0.5) ** (beta - 1)
    s = base.sum()
    if s == 0.0:
        return np.full(n, 1.0 / n)
    return base / s


def exponential_dirichlet(n: int, rng: np.random.Generator) -> np.ndarray:
    """Sample from an exponential distribution and normalise."""
    samp = rng.exponential(1.0, size=n)
    return samp / samp.sum()


def random_convex_combination(
    base_vectors: List[np.ndarray], rng: np.random.Generator
) -> np.ndarray:
    """Take a random convex combination of a subset of base vectors."""
    k = rng.integers(2, min(5, len(base_vectors) + 1))
    idx = rng.choice(len(base_vectors), size=k, replace=False)
    coeffs = rng.random(k)
    coeffs /= coeffs.sum()
    comb = sum(c * base_vectors[i] for c, i in zip(coeffs, idx))
    # re‑normalise to avoid tiny numerical drift
    return comb / comb.sum()


# ----------------------------------------------------------------------
# Simplex projection (Euclidean)
# ----------------------------------------------------------------------


def project_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of v onto the probability simplex:
        {x | x_i ≥ 0, Σ x_i = 1}.
    """
    n = v.shape[0]
    # sort v in descending order
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho_arr = u + (1.0 - cssv) / np.arange(1, n + 1)
    rho = np.where(rho_arr > 0)[0]
    if rho.size == 0:
        # all entries non‑positive → uniform simplex point
        return np.full_like(v, 1.0 / n)
    rho = rho[-1]
    theta = (cssv[rho] - 1.0) / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Gradient utilities for the smooth and the exact objective
# ----------------------------------------------------------------------


def softmax_weights(conv: np.ndarray, lam: float) -> np.ndarray:
    """Soft‑max over convolution entries (numerically stable)."""
    x = lam * conv
    mx = np.max(x)
    w = np.exp(x - mx)
    w /= w.sum()
    return w


def gradient_from_weights(a: np.ndarray, w: np.ndarray) -> np.ndarray:
    """
    Gradient of Σ_k w_k·(a*a)_k w.r.t. a.
    Implements   g = 2· a * w_rev,
    where w_rev is w reversed.  The result is a vector of length |a|.
    """
    w_rev = w[::-1]
    conv = convolution_fft(a, w_rev)  # length = 3·n‑2
    idx = 2 * len(a) - 2
    grads = 2.0 * conv[idx - np.arange(len(a))]
    return grads


# ----------------------------------------------------------------------
# Optimisation primitives
# ----------------------------------------------------------------------


def softmax_exponential_gradient(
    initial_a: np.ndarray,
    deadline: float,
    *,
    lam_start: float = 5.0,
    lam_end: float = 400.0,
    lam_ramp_iters: int = 8000,
    eta0: float = 0.6,
    max_iter: int = 200_000,
    noise_every: int = 2000,
    noise_scale: float = 1e-5,
) -> Tuple[np.ndarray, float]:
    """
    Exponentiated‑gradient descent on the log‑sum‑exp surrogate of the
    max‑convolution.  Returns the best vector seen and its exact score.
    """
    a = project_simplex(initial_a.astype(float).copy())
    best_a = a.copy()
    best_sc = compute_score(a)

    n = len(a)

    for it in range(1, max_iter + 1):
        if time.time() >= deadline:
            break

        # anneal temperature λ
        lam = lam_start + (lam_end - lam_start) * min(1.0, it / lam_ramp_iters)

        # current convolution and soft‑max weights
        conv = convolution_fft(a, a)
        w = softmax_weights(conv, lam)

        # gradient of the weighted sum
        grad = gradient_from_weights(a, w)

        # multiplicative update
        eta = eta0 / np.sqrt(it)
        a = a * np.exp(-eta * grad)

        # optional isotropic noise
        if noise_every and (it % noise_every == 0):
            a += np.random.randn(n) * noise_scale
            a = np.maximum(a, 0.0)

        # re‑project onto the simplex
        a = project_simplex(a)

        cur_sc = compute_score(a)
        if cur_sc < best_sc:
            best_sc = cur_sc
            best_a = a.copy()

        if it % 5000 == 0:
            print(
                f"    softmax iter={it:,d}  λ={lam:.1f}  cur={cur_sc:.6f}  best={best_sc:.6f}"
            )

    return best_a, best_sc


def subgradient_descent(
    initial_a: np.ndarray,
    deadline: float,
    *,
    max_iter: int = 200_000,
    eta0: float = 0.5,
    rel_tol: float = 1e-12,
) -> Tuple[np.ndarray, float]:
    """
    Projected sub‑gradient descent on the true objective
    (max‑convolution).  Uses the exact sub‑gradient obtained from the
    lag(s) attaining the current maximum.  Returns the best vector seen.
    """
    a = project_simplex(initial_a.astype(float).copy())
    best_a = a.copy()
    best_sc = compute_score(a)

    n = len(a)

    for it in range(1, max_iter + 1):
        if time.time() >= deadline:
            break

        conv = convolution_fft(a, a)
        max_val = conv.max()

        # mask of indices where the convolution is (almost) maximal
        mask = np.abs(conv - max_val) <= rel_tol * max_val
        w = mask.astype(float)

        grad = gradient_from_weights(a, w)

        eta = eta0 / np.sqrt(it)
        a = a - eta * grad
        a = project_simplex(a)

        cur_sc = compute_score(a)
        if cur_sc < best_sc:
            best_sc = cur_sc
            best_a = a.copy()

        if it % 5000 == 0:
            print(
                f"    subgrad iter={it:,d}  cur={cur_sc:.6f}  best={best_sc:.6f}"
            )

    return best_a, best_sc


def simulated_annealing(
    initial_a: np.ndarray,
    deadline: float,
    *,
    max_iter: int = 100_000,
    start_temp: float = 0.02,
    end_temp: float = 0.0005,
) -> Tuple[np.ndarray, float]:
    """
    Simple simulated‑annealing local search that perturbs two coordinates
    while preserving the simplex constraint.  Used as a final polishing
    step when time permits.
    """
    rng = np.random.default_rng()
    cur_a = project_simplex(initial_a.astype(float).copy())
    cur_sc = compute_score(cur_a)
    best_a = cur_a.copy()
    best_sc = cur_sc
    n = len(cur_a)

    for it in range(1, max_iter + 1):
        if time.time() >= deadline:
            break

        temp = start_temp + (end_temp - start_temp) * (it / max_iter)

        i, j = rng.choice(n, size=2, replace=False)
        delta = rng.normal(scale=0.01)

        # propose a move that keeps the sum unchanged
        new_a = cur_a.copy()
        new_a[i] = max(new_a[i] + delta, 0.0)
        # adjust j to keep total mass = 1
        diff = new_a[i] - cur_a[i]
        new_a[j] = max(new_a[j] - diff, 0.0)

        # re‑project onto simplex to correct any tiny drift
        new_a = project_simplex(new_a)

        new_sc = compute_score(new_a)

        if new_sc < cur_sc or rng.random() < np.exp(-(new_sc - cur_sc) / max(temp, 1e-12)):
            cur_a, cur_sc = new_a, new_sc
            if cur_sc < best_sc:
                best_sc, best_a = cur_sc, cur_a.copy()

        if it % 5000 == 0:
            print(
                f"    SA iter={it:,d}  cur={cur_sc:.6f}  best={best_sc:.6f}"
            )

    return best_a, best_sc

def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs) -> List[float]:
    """
    Search for a non‑negative step‑function with a low ``evaluate_sequence``
    value.  The routine respects the given time budget, explores several
    lengths and families, refines each candidate with a two‑phase
    optimisation (soft‑max → sub‑gradient), and finally performs a short
    simulated‑annealing polish.  The best feasible sequence found is
    returned as a plain Python list.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 2.0  # safety margin

    best_seq: List[float] = []
    best_val: float = float("inf")

    # ------------------------------------------------------------------
    # 0. Optional fallback from a previously discovered construction.
    # ------------------------------------------------------------------
    try:
        fallback = [float(x) for x in height_sequence_1]  # type: ignore
        val = evaluate_sequence(fallback)
        print(f"Fallback (height_sequence_1) score = {val:.6f}")
        if val < best_val:
            best_val = val
            best_seq = fallback
    except NameError:
        pass

    # ------------------------------------------------------------------
    # 1. Exploration over several lengths.
    # ------------------------------------------------------------------
    lengths = [1024, 2048, 4096, 8192]

    # keep a small library of base vectors to allow convex combinations later
    base_library: List[np.ndarray] = []

    for n in lengths:
        if time.time() >= deadline - 5.0:
            break
        print(f"\n=== Exploring length n = {n} ===")

        # --------------------------------------------------------------
        # 1a. Build a pool of initial candidates.
        # --------------------------------------------------------------
        candidates: List[np.ndarray] = []

        # uniform
        candidates.append(np.full(n, 1.0 / n))

        # arcsine
        candidates.append(arcsine_initial(n))

        # power‑law family
        for _ in range(6):
            p = rng.uniform(0.2, 0.9)
            eps = rng.uniform(0.001, max(2.0, 0.02 * n))
            candidates.append(power_law_shape(n, p, eps))

        # beta‑like shapes
        for _ in range(4):
            alpha = rng.uniform(0.2, 5.0)
            beta = rng.uniform(0.2, 5.0)
            candidates.append(beta_shape(n, alpha, beta))

        # exponential (Dirichlet) draws
        for _ in range(3):
            candidates.append(exponential_dirichlet(n, rng))

        # random convex mixtures of previously stored bases (if any)
        if base_library:
            for _ in range(2):
                candidates.append(random_convex_combination(base_library, rng))

        # store the newly built vectors as potential bases for later up‑sampling
        base_library.extend(candidates)

        # --------------------------------------------------------------
        # 1b. Quick sweep with the fast internal score, keep the best few.
        # --------------------------------------------------------------
        scored: List[Tuple[float, np.ndarray]] = [(compute_score(c), c) for c in candidates]
        scored.sort(key=lambda x: x[0])
        top_candidates = [c for _, c in scored[:4]]  # keep up to four promising vectors

        # --------------------------------------------------------------
        # 1c. Refine each retained candidate.
        # --------------------------------------------------------------
        for idx, cand in enumerate(top_candidates):
            if time.time() >= deadline - 5.0:
                break
            print(f"  candidate {idx + 1}/{len(top_candidates)} (len={n})")

            # allocate iteration budget based on remaining time
            remaining = max(0.0, deadline - time.time())
            # empirical 30‑40 inner iterations per second for n≈4096
            max_soft_iter = int(min(80_000, max(2_000, remaining * 35)))
            refined_soft, _ = softmax_exponential_gradient(
                initial_a=cand,
                deadline=deadline,
                lam_start=5.0,
                lam_end=400.0,
                lam_ramp_iters=8000,
                eta0=0.6,
                max_iter=max_soft_iter,
                noise_every=2000,
                noise_scale=1e-5,
            )

            # sub‑gradient fine‑tuning
            remaining = max(0.0, deadline - time.time())
            max_sub_iter = int(min(120_000, max(2_000, remaining * 35)))
            refined_sub, _ = subgradient_descent(
                initial_a=refined_soft,
                deadline=deadline,
                max_iter=max_sub_iter,
                eta0=0.5,
            )

            # final official evaluation
            seq = refined_sub.tolist()
            val = evaluate_sequence(seq)
            print(f"    evaluated score = {val:.6f}")

            if val < best_val:
                best_val = val
                best_seq = seq
                print(f"    *** New global best: {best_val:.6f} (n={n}) ***")

    # ------------------------------------------------------------------
    # 2. Final polishing of the overall best sequence (if time permits).
    # ------------------------------------------------------------------
    if best_seq and time.time() < deadline - 5.0:
        print("\nFinal polishing of the global best")
        a = np.array(best_seq, dtype=float)

        remaining = max(0.0, deadline - time.time())
        max_soft = int(min(120_000, max(5_000, remaining * 30)))
        a_soft, _ = softmax_exponential_gradient(
            initial_a=a,
            deadline=deadline,
            lam_start=5.0,
            lam_end=600.0,
            lam_ramp_iters=12_000,
            eta0=0.5,
            max_iter=max_soft,
            noise_every=4000,
            noise_scale=5e-6,
        )

        remaining = max(0.0, deadline - time.time())
        max_sub = int(min(180_000, max(5_000, remaining * 30)))
        a_sub, _ = subgradient_descent(
            initial_a=a_soft,
            deadline=deadline,
            max_iter=max_sub,
            eta0=0.4,
        )

        # optional simulated‑annealing polish (quick)
        remaining = max(0.0, deadline - time.time())
        if remaining > 2.0:

                initial_a=a_sub,

                start_temp=0.015,
                end_temp=0.0003,
            )
        else:

        final_val = evaluate_sequence(a_final.tolist())
        print(f"Final polished score = {final_val:.6f}")
        if final_val < best_val:
            best_val = final_val
            best_seq = a_final.tolist()
            print("*** Updated global best after final polish ***")

    # ------------------------------------------------------------------
    # 3. Guard – always return a feasible sequence.
    # ------------------------------------------------------------------
    if not best_seq:

        best_seq = [1.0] * 256
        best_val = evaluate_sequence(best_seq)

    print(f"\nSearch finished. Best reported score = {best_val:.6f}")
    return best_seq
