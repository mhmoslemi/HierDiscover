"""
Propose a candidate step‑function using Adam‑style mirror descent.

The algorithm works on the simplex of non‑negative coefficients (sum = 1)
by maintaining log‑weights (logits) `x`.  The current coefficients are
`a = softmax(x)`.  At each iteration we evaluate

    obj(a) = 2·n·max(conv(a,a))

where `conv(a,a)` is the full (linear) self‑convolution.  The sub‑gradient
of `max(conv(a,a))` with respect to `a` is obtained by scanning all active
maximal convolution entries; the gradient of `obj` is simply `2·n` times
that sub‑gradient.  By the chain rule the gradient w.r.t. the logits is

    g_x = a * (g_a – ⟨a,g_a⟩),

where `g_a` is the gradient of the objective w.r.t. `a`.  We update the
logits with Adam (first‑ and second‑moment estimates) and periodically add
small random jitter to escape plateaus.  Several random restarts with
different problem sizes are performed until the time budget expires.
Whenever a new global best objective is found it is printed.  If the target
value 1.503 is reached the routine terminates early.

The returned list contains the best coefficient vector found (any global
scale is irrelevant because the objective is homogeneous).
"""

import time
import numpy as np
import math
import random

def _max_and_subgradient(a: np.ndarray) -> tuple[float, np.ndarray]:
    """
    Compute the maximum entry of the full self‑convolution a*a and a
    sub‑gradient of that maximum with respect to a.

    Parameters
    ----------
    a : np.ndarray
        Non‑negative vector of length n (assumed to sum to 1).

    Returns
    -------
    max_val : float
        The maximal convolution value.
    grad : np.ndarray
        Sub‑gradient of `max_val` w.r.t. `a`.  It has the same shape as `a`.
    """
    n = a.shape[0]
    # convolution via FFT (real‑to‑real)
    fft = np.fft.rfft(a, n=2 * n - 1)
    conv = np.fft.irfft(fft * fft, n=2 * n - 1)

    max_val = float(conv.max())

    # tolerance to decide which entries are active
    tol = max_val * 1e-12 + 1e-14
    active = np.where(conv >= max_val - tol)[0]

    grad = np.zeros_like(a)
    for k in active:
        low = max(0, k - n + 1)
        high = min(n - 1, k)
        idx = np.arange(low, high + 1, dtype=int)
        partner = k - idx
        grad[idx] += 2.0 * a[partner]

    if active.size > 1:
        grad /= float(active.size)

    return max_val, grad


def _softmax(x: np.ndarray) -> np.ndarray:
    """Numerically stable softmax, returns a probability vector that sums to 1."""
    mx = np.max(x)
    y = np.exp(x - mx)
    s = y.sum()
    if s == 0.0:
        return np.full_like(x, 1.0 / x.shape[0])
    return y / s
