def pgd_optimize(init_a, deadline, max_iter=100000, step0=0.5, line_search_factor=0.5, min_step=1e-8):

    a = init_a.copy()
    best_a = a.copy()
    best_score = compute_score(a)

    for it in range(1, max_iter+1):
        if time.time() >= deadline:
            break
        grad = subgradient_max(a)  # shape (n,)
        # direction: -grad
        # take step via backtracking
        step = step0
        while step > min_step:
            a_new = a - step * grad
            # project onto simplex
            a_new = project_to_simplex(a_new)
            cur_score = compute_score(a_new)
            if cur_score < best_score:
                best_score = cur_score
                best_a = a_new.copy()
                a = a_new
                break  # accept step
            else:
                step *= line_search_factor
        else:
            # step too small, maybe do random perturbation
            a = a + np.random.randn(len(a)) * (step0*0.01)
            a = project_to_simplex(a)

        # optionally adjust step0 adaptively: if accept good step, maybe increase step0, else decrease.
        if step < step0:
            step0 *= 0.9
        else:
            step0 *= 1.05

        if it % 2000 == 0:
            print(f"Iter {it} best score {best_score:.6f}")

    return best_a, best_score

"""
Propose a low‑score step‑function.

The search combines several ideas that go beyond the reference implementation:

* **Several problem sizes** – larger vectors (up to 32768 entries) give a
  theoretical lower bound close to 1, so we allocate the available time to
  the biggest size the budget permits.

* **Multiple random starts** – arcsine, uniform and Dirichlet draws with
  different concentration parameters produce a wide variety of shapes.

* **Two complementary smooth optimisers**
    - A projected Adam optimiser on the soft‑max approximation of the
      maximum of the self‑convolution (α ramps from 1 to 5·10⁴).
    - A gradient descent on the *flatness* of the convolution
      ( minimise ∥a*a‑μ∥₂², μ = 1/(2n‑1) ).  This drives the whole
      convolution curve towards a constant and therefore reduces its peak.

* **Exact sub‑gradient descent** – after the smooth phases we switch to
  the true sub‑gradient of max(a*a).  A simple back‑tracking line‑search
  with projection onto the probability simplex gives a reliable final
  polishing step.

* **Stochastic hill‑climbing** – a cheap permutation / mass‑transfer
  hill‑climber finishes the search, exploiting any remaining tiny
  improvement opportunities.

All heavy work (convolutions) is performed with an FFT‑based routine
(O(n·log n) per evaluation).  The routine respects the 1000 s budget,
prints intermediate scores and finally returns the best sequence
found (a plain ``list[float]``) which can be fed directly to
``evaluate_sequence``.
"""

import time
import numpy as np


# ----------------------------------------------------------------------
# Basic linear‑algebra / signal‑processing utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full linear convolution using a real FFT (fast)."""
    m = len(a) + len(b) - 1
    n_fft = next_pow2(m)
    A = np.fft.rfft(a, n=n_fft)
    B = np.fft.rfft(b, n=n_fft)
    C = A * B
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection onto the probability simplex (non‑negative,
    sum = 1)."""
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / v.shape[0])

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, v.shape[0] + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.where(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def compute_score(a: np.ndarray) -> float:
    """
    Fast replica of ``evaluate_sequence`` when ∑a = 1.
    Returns 2·n·max(a*a).
    """
    n = a.shape[0]
    conv = convolution_fft(a, a)
    max_val = float(np.max(conv))
    return 2.0 * n * max_val


# ----------------------------------------------------------------------
# Initialisation helpers
# ----------------------------------------------------------------------


def arcsine_initial(n: int) -> np.ndarray:
    """Arcsine‑type start: a_i ∝ 1/√((i+½)(n−½−i))."""
    i = np.arange(n, dtype=float)
    eps = 0.25
    vals = 1.0 / np.sqrt((i + 0.5) * (n - 0.5 - i) + eps)
    return project_to_simplex(vals)


def dirichlet_initial(n: int, concentration: float) -> np.ndarray:
    """Draw a Dirichlet vector with given concentration."""
    return np.random.dirichlet(np.full(n, concentration))


# ----------------------------------------------------------------------
# Smoothed‑max optimiser (Adam)
# ----------------------------------------------------------------------


def smoothed_gradient(a: np.ndarray, alpha: float):
    """
    Gradient of the soft‑max approximation of max(a*a):
        f_α(a) = (1/α)·log Σ_k exp(α·(a*a)_k)

    Returns (grad, conv) where conv = a*a.
    """
    n = a.shape[0]
    conv = convolution_fft(a, a)                     # length 2n‑1

    # stable soft‑max weights
    maxc = np.max(conv)
    exp_term = np.exp(alpha * (conv - maxc))
    w = exp_term / np.sum(exp_term)                 # Σ w = 1

    # convolution of reversed a with the weight vector
    a_rev = a[::-1]
    conv_rev = convolution_fft(a_rev, w)            # length 3n‑2
    start = n - 1
    rev_slice = conv_rev[start:start + n]           # needed for each component

    # diagonal correction (weights where i=j)
    w_diag = w[::2][:n]

    grad = 2.0 * rev_slice - w_diag * a
    return grad, conv


def adam_optimize(
    init_a: np.ndarray,
    time_deadline: float,
    max_iter: int = 200_000,
    eta: float = 0.02,
    beta1: float = 0.9,
    beta2: float = 0.999,
    eps: float = 1e-8,
    alpha_start: float = 1.0,
    alpha_end: float = 50_000.0,
    noise_every: int = 2000,
    noise_scale: float = 1e-4,
):
    """
    Projected Adam optimiser on the smoothed max objective.
    Returns the best point (ndarray) and its true score.
    """
    a = project_to_simplex(init_a.astype(float).copy())
    m = np.zeros_like(a)
    v = np.zeros_like(a)

    best_a = a.copy()
    best_score = compute_score(a)

    n = len(a)
    for t in range(1, max_iter + 1):
        if time.time() >= time_deadline:
            break

        # homotopy on α
        alpha = alpha_start + (alpha_end - alpha_start) * min(1.0, t / max_iter)

        grad, _ = smoothed_gradient(a, alpha)

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)

        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)

        step = eta * m_hat / (np.sqrt(v_hat) + eps)
        a_new = a - step

        # occasional isotropic noise
        if noise_every and (t % noise_every == 0):
            a_new += np.random.randn(n) * noise_scale

        a_new = project_to_simplex(a_new)

        cur_score = compute_score(a_new)
        if cur_score < best_score:
            best_score = cur_score
            best_a = a_new.copy()

        a = a_new

        if t % 5000 == 0:
            print(f"    Adam iter {t:,d} – cur {cur_score:.6f} – best {best_score:.6f}")

    return best_a, best_score


# ----------------------------------------------------------------------
# Flatness optimiser (minimise ∥a*a‑μ∥₂²)
# ----------------------------------------------------------------------


def flatness_gradient(a: np.ndarray):
    """
    Gradient of ½·∥a*a - μ∥₂² where μ = 1/(2n-1).
    The factor ½ is omitted (absorbed in step size).
    """
    n = a.shape[0]
    conv = convolution_fft(a, a)                       # length 2n‑1
    mu = 1.0 / (2 * n - 1)
    diff = conv - mu                                    # shape (2n‑1,)

    a_rev = a[::-1]
    # derivative of the L2 term: 2 * (∂conv/∂a)ᵀ·diff = 4 * (a_rev * diff)
    conv_rev = convolution_fft(a_rev, diff)            # length 3n‑2
    start = n - 1
    grad = 4.0 * conv_rev[start:start + n]
    return grad


def flatness_optimize(
    init_a: np.ndarray,
    time_deadline: float,
    max_iter: int = 100_000,
    step0: float = 0.5,
    line_factor: float = 0.7,
    min_step: float = 1e-8,
):
    """
    Simple projected gradient descent on the flatness objective.
    Returns the best vector and its true max‑convolution score.
    """
    a = project_to_simplex(init_a.copy())
    best_a = a.copy()
    best_score = compute_score(a)

    n = len(a)
    step = step0
    it = 0
    while it < max_iter and time.time() < time_deadline:
        it += 1
        # gradient of the flatness objective (aims to flatten the whole convolution)
        grad = flatness_gradient(a)

        # back‑tracking line search
        while step > min_step:
            a_new = a - step * grad
            a_new = project_to_simplex(a_new)
            cur_score = compute_score(a_new)
            if cur_score < best_score:
                best_score = cur_score
                best_a = a_new.copy()
                a = a_new
                # slightly increase step after a successful move
                step = min(step / line_factor, step0)
                break
            else:
                step *= line_factor
        else:
            # step vanished – inject a tiny random jitter to escape
            a = project_to_simplex(a + np.random.randn(n) * 1e-4)

        if it % 2000 == 0:
            print(f"    Flatness iter {it:,d} – best {best_score:.6f}")

    return best_a, best_score


# ----------------------------------------------------------------------
# Exact sub‑gradient descent on the max‑convolution
# ----------------------------------------------------------------------


def subgradient_max(a: np.ndarray) -> np.ndarray:
    """
    Sub‑gradient of max(a*a) with respect to a (sum‑to‑1, non‑negative).
    The active set consists of all indices where the convolution attains its
    maximum (within a tiny tolerance).
    Returns a vector of shape (n,).
    """
    n = a.shape[0]
    conv = convolution_fft(a, a)                      # length 2n‑1
    max_val = np.max(conv)
    # treat entries within 1e‑12 relative of the max as active
    tol = max(1e-12, max_val * 1e-12)
    active = np.where(conv >= max_val - tol)[0]

    # weight equally over the active set
    w = np.zeros_like(conv)
    if active.size > 0:
        w[active] = 1.0 / active.size

    a_rev = a[::-1]
    conv_rev = convolution_fft(a_rev, w)              # length 3n‑2
    start = n - 1
    grad = 2.0 * conv_rev[start:start + n]
    return grad


def subgradient_optimize(
    init_a: np.ndarray,
    time_deadline: float,
    max_iter: int = 150_000,
    step0: float = 1.0,
    line_factor: float = 0.7,
    min_step: float = 1e-8,
):
    """
    Projected sub‑gradient descent on the true max‑convolution objective.
    Returns the best vector and its true score.
    """
    a = project_to_simplex(init_a.copy())
    best_a = a.copy()
    best_score = compute_score(a)

    n = len(a)
    step = step0
    it = 0
    while it < max_iter and time.time() < time_deadline:
        it += 1
        grad = subgradient_max(a)

        # back‑tracking line search (monotone descent)
        while step > min_step:
            a_new = a - step * grad
            a_new = project_to_simplex(a_new)
            cur_score = compute_score(a_new)
            if cur_score < best_score:
                best_score = cur_score
                best_a = a_new.copy()
                a = a_new
                step = min(step / line_factor, step0)   # grow after success
                break
            else:
                step *= line_factor
        else:
            # if step vanished, jitter a little
            a = project_to_simplex(a + np.random.randn(n) * 1e-4)

        if it % 2000 == 0:
            print(f"    Sub‑grad iter {it:,d} – best {best_score:.6f}")

    return best_a, best_score

def hill_climb_permutation(seq, time_deadline, max_steps=200_000):
    """
    Stochastic hill‑climber that swaps two entries or transfers a small
    fraction of mass.  The true (unsmoothed) objective is used for acceptance.
    """
    a = np.array(seq, dtype=float)
    n = a.shape[0]

    best_a = a.copy()
    best_score = compute_score(best_a)

    step_scale = 0.1

    while max_steps > 0 and time.time() < time_deadline:
        move_type = np.random.rand()
        cand = best_a.copy()

        if move_type < 0.5:
            # simple swap
            i, j = np.random.randint(n, size=2)
            cand[i], cand[j] = cand[j], cand[i]
        else:
            # move a fraction from i to j
            i, j = np.random.randint(n, size=2)
            if i != j and cand[i] > 0:
                delta = step_scale * cand[i]
                cand[i] -= delta
                cand[j] += delta

        cand = project_to_simplex(cand)
        cur_score = compute_score(cand)

        if cur_score < best_score:
            best_score = cur_score
            best_a = cand
            step_scale = min(step_scale * 1.05, 0.5)
        else:
            step_scale *= 0.97

        max_steps -= 1

    return best_a.tolist(), best_score
