def refine_sequence(a, deadline):
    current_a = a.copy()
    current_score = compute_score(current_a)
    best_a = current_a.copy()
    best_score = current_score
    # parameters
    init_temp = 1.0
    final_temp = 1e-4
    total_steps = 200000
    step = 0
    while time.time() < deadline and step < total_steps:
        # recompute conv and max index every few iterations to avoid extra cost? but we need conv each step anyway.
        conv = convolution_fft(current_a, current_a)
        max_val = conv.max()
        max_idx = int(conv.argmax())
        # compute weight vector for contributions at max_idx
        # weight_i = a[max_idx - i] if in range else 0
        # Also compute contributions: contrib_i = weight_i
        # Choose i high contrib, j low contrib randomly
        # sample i with probability proportional to current_a[i] * weight_i
        weights = np.zeros_like(current_a)
        # valid range
        low = max_idx - (len(current_a) - 1)
        high = min(max_idx, len(current_a) - 1)
        # indices i in [low, high]
        i_vals = np.arange(low, high+1)
        # for these, partner = max_idx - i
        partner = max_idx - i_vals
        weights[i_vals] = current_a[i_vals] * current_a[partner]
        # random move: sample i with probability proportional to weights (if all zero fallback uniform)
        prob = weights / (weights.sum() + 1e-20)
        if prob.sum() == 0:
            i = np.random.randint(len(current_a))
        else:
            i = np.random.choice(len(current_a), p=prob)
        # choose j uniformly
        j = np.random.randint(len(current_a))
        # ensure i != j
        if i==j:
            continue
        # delta = eps * current_a[i]; eps dynamic based on step count
        eps = 0.05 * (1 - step / total_steps) + 0.001  # decreasing
        delta = eps * current_a[i]
        if delta > 0:
            cand = current_a.copy()
            cand[i] -= delta
            cand[j] += delta
            # ensure nonnegative due to rounding
            if cand[i] < 0:
                cand[i] = 0.0
            # project? maybe normalizing not needed because mass conserved
            # but projection onto simplex ensures sum=1 exactly.
            cand = project_to_simplex(cand)
            cand_score = compute_score(cand)
            # temperature schedule linear
            temp = init_temp + (final_temp - init_temp) * (step / total_steps)
            if cand_score < current_score or np.random.rand() < np.exp(-(cand_score - current_score)/temp):
                current_a = cand
                current_score = cand_score
                if cand_score < best_score:
                    best_score = cand_score
                    best_a = cand.copy()
        step+=1
        # maybe occasional recompute conv for next iteration
    return best_a, best_score

def propose_candidate(seed=42, budget_s=1000):
    np.random.seed(seed)
    start = time.time()
    deadline = start + budget_s - 2.0

    n = 8192
    a = arcsine_initial(n)
    score = compute_score(a)
    print(f"Initial score {score}")

    # optionally perform Adam optimization for some iterations
    iter_adam = 300_000
    a_opt, score_opt = adam_optimize(a, deadline, max_iter=iter_adam, eta=0.01, alpha_start=1.0, alpha_end=5000.0, noise_every=2000, noise_scale=5e-4)
    # note: adam_optimize already prints progress
    # Use result as start for hillclimb
    if time.time() < deadline - 10:
        refined, refined_score = refine_sequence(a_opt, deadline)
        if refined_score < score_opt:
            a_opt, score_opt = refined, refined_score

    # if time remains, try random restarts
    while time.time() < deadline - 30:
        # random Dirichlet
        a_rand = np.random.dirichlet(np.ones(n))
        # hillclimb on random start for limited steps
        r_seq, r_score = refine_sequence(a_rand, deadline)
        if r_score < score_opt:
            a_opt, score_opt = r_seq, r_score
            print(f"New best from random restart {r_score}")

    final_seq = a_opt.tolist()
    final_score = evaluate_sequence(final_seq)
    print(f"Final score = {final_score}")
    return final_seq

"""
Propose a low‑score step‑function.

Algorithm
---------
1. Choose a moderately large length (default 8192).  
2. Generate a small family of smooth, symmetric candidates:
   * the classic arcsine shape (Beta(½,½));
   * a one‑parameter Beta family a_i ∝ ((i+½)(n‑½‑i))^{α‑1} for several α∈[0.2,1.2];
   * a few random Dirichlet draws for diversity.
   All candidates are projected onto the probability simplex (∑a_i = 1).

3. Pick the candidate with the smallest true score
   (score = 2·n·max(a∗a) – the scaling of evaluate_sequence cancels).

4. Refine the current best vector with a *targeted* stochastic hill‑climber.
   The hill‑climber always looks at the active convolution index
   m = argmax(a∗a) and preferentially moves a small amount of mass from
   indices that contribute heavily to this peak (large a_i·a_{m‑i})
   towards random “low‑contribution’’ indices.
   Acceptance follows a simple simulated‑annealing rule,
   allowing occasional uphill moves early on.

5. If time permits, repeat step 4 from a few random restarts.

All heavy work uses FFT‑based convolutions (O(n·log n) per evaluation).
The routine respects the 1000 s budget and returns the best sequence it
found as a plain list of floats.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Basic numeric utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full (linear) convolution a * b via a real FFT."""
    m = len(a) + len(b) - 1
    n_fft = next_pow2(m)
    A = np.fft.rfft(a, n=n_fft)
    B = np.fft.rfft(b, n=n_fft)
    C = A * B
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of v onto the probability simplex:
    non‑negative entries that sum to 1.
    """
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / v.shape[0])

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, v.shape[0] + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.where(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def compute_score(a: np.ndarray) -> float:
    """
    Fast replica of `evaluate_sequence` for a normalized vector (∑a = 1).
    Returns 2·n·max(a∗a).
    """
    n = a.shape[0]
    conv = convolution_fft(a, a)
    return 2.0 * n * float(np.max(conv))


# ----------------------------------------------------------------------
# Candidate generators
# ----------------------------------------------------------------------


def arcsine_initial(n: int) -> np.ndarray:
    """
    Classical arcsine start:
    a_i ∝ 1 / sqrt( (i+½)(n‑½‑i) ).
    Afterwards projected to the simplex.
    """
    i = np.arange(n, dtype=float)
    eps = 1e-12
    vals = 1.0 / np.sqrt((i + 0.5) * (n - 0.5 - i) + eps)
    return project_to_simplex(vals)


def beta_initial(n: int, alpha: float) -> np.ndarray:
    """
    Symmetric Beta(α,α) shape on a discrete grid.
    a_i ∝ ((i+½)(n‑½‑i))^{α‑1}.
    For α=0.5 this is the arcsine law.
    """
    i = np.arange(n, dtype=float)
    # exponent may be negative; guard against overflow with a tiny epsilon
    exp = alpha - 1.0
    vals = ((i + 0.5) * (n - 0.5 - i) + 1e-12) ** exp
    # Clip tiny negative values caused by floating‑point errors
    vals = np.maximum(vals, 0.0)
    return project_to_simplex(vals)


# ----------------------------------------------------------------------
# Targeted hill‑climbing (simulated annealing)
# ----------------------------------------------------------------------


def targeted_hillclimb(a0: np.ndarray,
                       deadline: float,
                       max_steps: int = 200_000,
                       init_temp: float = 0.02,
                       final_temp: float = 1e-5,
                       start_scale: float = 0.12,
                       end_scale: float = 0.001) -> tuple[list[float], float]:
    """
    Stochastic hill‑climber that looks at the active convolution peak.
    Moves a small fraction of mass from indices that contribute heavily
    to the current peak to random low‑contribution indices.
    Uses a simple annealing schedule and projects back onto the simplex.
    Returns the best vector (as a list) and its true score.
    """
    a = a0.copy()
    n = a.shape[0]
    cur_score = compute_score(a)
    best_a = a.copy()
    best_score = cur_score

    step = 0
    while step < max_steps and time.time() < deadline:
        # current convolution and its maximal index
        conv = convolution_fft(a, a)
        max_val = float(np.max(conv))
        max_idx = int(np.argmax(conv))

        # contribution weight: w_i = a_i * a_{max_idx-i}
        partner = max_idx - np.arange(n)
        mask = (partner >= 0) & (partner < n)
        weight = np.zeros_like(a)
        weight[mask] = a[mask] * a[partner[mask]]

        # sampling distribution for the donor index (biased towards high weight)
        w_sum = weight.sum()
        if w_sum > 0:
            prob = weight / w_sum
            i = np.random.choice(n, p=prob)
        else:
            i = np.random.randint(n)

        # random receiver (avoid i)
        j = np.random.randint(n - 1)
        if j >= i:
            j += 1

        # step‑size schedule (linear decay from start_scale to end_scale)
        scale = start_scale + (end_scale - start_scale) * (step / max_steps)
        delta = scale * a[i]                     # amount of mass to move
        if delta <= 0.0:
            step += 1
            continue

        cand = a.copy()
        cand[i] -= delta
        cand[j] += delta

        # numerical safety before projection
        cand = np.clip(cand, 0.0, None)
        cand = project_to_simplex(cand)

        cand_score = compute_score(cand)

        # annealing temperature (linear)
        temp = init_temp + (final_temp - init_temp) * (step / max_steps)

        # accept if better, or with Metropolis probability
        if cand_score < cur_score or np.random.rand() < np.exp(-(cand_score - cur_score) / max(temp, 1e-12)):
            a = cand
            cur_score = cand_score
            if cand_score < best_score:
                best_score = cand_score
                best_a = cand.copy()

        step += 1

        # occasional progress output (light I/O)
        if step % 5000 == 0:
            print(f"    hill‑climb step {step:,d} – cur {cur_score:.6f} – best {best_score:.6f}")

    return best_a.tolist(), best_score


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative sequence that minimises `evaluate_sequence`.
    The routine performs:
        * generation of a small parametric family (arcsine + Beta(α,α));
        * evaluation of all members;
        * a targeted annealing hill‑climber on the best member;
        * a few extra random restarts if time permits.
    Returns the best sequence as a plain Python list of floats.
    """
    np.random.seed(seed)
    start_time = time.time()
    # keep a tiny safety margin for the harness
    deadline = start_time + budget_s - 2.0

    # ------------------------------------------------------------------
    # Parameters
    # ------------------------------------------------------------------
    n = 8192                     # length of the step‑function
    alphas = [0.35, 0.45, 0.5, 0.55, 0.65, 0.8]  # Beta‑family exponents
    max_dirichlet_draws = 3      # number of random Dirichlet starts
    max_hill_steps = 250_000     # steps for the main hill‑climb

    best_seq = None
    best_score = float("inf")

    # ------------------------------------------------------------------
    # 1) parametric candidates (arcsine + Beta family)
    # ------------------------------------------------------------------
    candidates = []

    # arcsine (exact β=0.5)
    candidates.append(arcsine_initial(n))

    # Beta family
    for a in alphas:
        candidates.append(beta_initial(n, a))

    # a few Dirichlet draws (for extra diversity)
    for _ in range(max_dirichlet_draws):
        candidates.append(np.random.dirichlet(np.ones(n)))

    print(f"\nGenerated {len(candidates)} initial candidates (n={n}).")

    for idx, vec in enumerate(candidates, start=1):
        if time.time() >= deadline - 5.0:
            break
        # ensure feasible (project) – most generators already satisfy it
        vec = project_to_simplex(vec)
        score = compute_score(vec)
        print(f"  candidate {idx}/{len(candidates)} – score {score:.6f}")
        if score < best_score:
            best_score = score
            best_seq = vec.tolist()
            print(f"    *** new global best (parametric) = {best_score:.6f}")

    # ------------------------------------------------------------------
    # 2) refined hill‑climb on the current best vector
    # ------------------------------------------------------------------
    if best_seq is not None and time.time() < deadline - 5.0:
        print("\nRefining best candidate with targeted hill‑climb...")
        refined_seq, refined_score = targeted_hillclimb(
            np.array(best_seq, dtype=float),
            deadline,
            max_steps=max_hill_steps,
            init_temp=0.015,
            final_temp=5e-6,
            start_scale=0.15,
            end_scale=0.0005,
        )
        print(f"    refined score = {refined_score:.6f}")
        if refined_score < best_score:
            best_score = refined_score
            best_seq = refined_seq

    # ------------------------------------------------------------------
    # 3) occasional extra random restarts (if time permits)
    # ------------------------------------------------------------------
    extra_restart = 0
    while time.time() < deadline - 5.0:
        extra_restart += 1
        print(f"\nExtra random restart #{extra_restart}")
        a0 = np.random.dirichlet(np.ones(n))
        # short hill‑climb (fewer steps) to explore a different basin
        steps = 80_000
        seq, sc = targeted_hillclimb(a0, deadline, max_steps=steps,
                                      init_temp=0.02, final_temp=1e-5,
                                      start_scale=0.12, end_scale=0.0008)
        print(f"  restart #{extra_restart} best score = {sc:.6f}")
        if sc < best_score:
            best_score = sc
            best_seq = seq
            print(f"    *** new global best from restart = {best_score:.6f}")

    # ------------------------------------------------------------------
    # 4) fallback (should never be needed)
    # ------------------------------------------------------------------
    if best_seq is None:
        best_seq = [1.0 / 256] * 256
        best_score = evaluate_sequence(best_seq)

    final_score = evaluate_sequence(best_seq)
    print(f"\nFinished search. Final reported score = {final_score:.6f}")
    return best_seq
