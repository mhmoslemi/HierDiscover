"""
Propose a low‑score step‑function.

Algorithm
---------
1. **Parameter sweep** – For several modest lengths (256…4096) we
   generate families of “arcsine‑like’’ vectors
        a_i ∝ 1 / ((i+δ)*(n‑δ‑i))^p
   with a grid of exponents `p` and offsets `δ`.  After projection onto the
   probability simplex the vector sums to 1, so the true objective is simply
   `2·n·max(conv(a,a))`.  The sweep is cheap (FFT‑based convolution) and
   explores many shapes that the previous optimiser might miss.

2. **Local refinement** – The best candidate from the sweep is tightened by a
   fast hill‑climber that repeatedly moves a small fraction of mass between
   two positions (or swaps two entries) and keeps the best improvement.
   The hill‑climber works directly with the exact objective (no smoothing).

3. **Fallback** – If nothing is found a uniform vector of length 256 is used.

All heavy work respects the 1000 s time budget.  The routine returns a plain
Python list of floats ready for `evaluate_sequence`.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Basic utilities (same as in the previous solution)
# ----------------------------------------------------------------------


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of a vector onto the probability simplex."""
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / v.shape[0])

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, v.shape[0] + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.where(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full convolution a * b using a real FFT (fast)."""
    m = len(a) + len(b) - 1
    n_fft = next_pow2(m)
    A = np.fft.rfft(a, n=n_fft)
    B = np.fft.rfft(b, n=n_fft)
    C = A * B
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


def compute_score(a: np.ndarray) -> float:
    """
    Fast replica of `evaluate_sequence` when ∑a = 1.
    Returns 2·n·max(a*a).
    """
    n = a.shape[0]
    conv = convolution_fft(a, a)
    max_val = float(np.max(conv))
    return 2.0 * n * max_val


# ----------------------------------------------------------------------
# Families of candidate shapes
# ----------------------------------------------------------------------


def arcsine_initial(n: int) -> np.ndarray:
    """Classic arcsine start (δ=0.5, p=0.5)."""
    i = np.arange(n, dtype=float)
    eps = 0.25
    vals = 1.0 / np.sqrt((i + 0.5) * (n - 0.5 - i) + eps)
    return project_to_simplex(vals)


def generate_arcsine_like(n: int, p: float, delta: float, eps: float = 1e-12) -> np.ndarray:
    """
    a_i ∝ 1 / ((i+δ)*(n‑δ‑i))^p   (δ > 0, p > 0)
    The vector is projected onto the simplex so that it sums to 1.
    """
    i = np.arange(n, dtype=float)
    denom = (i + delta) * (n - delta - i) + eps
    vals = np.power(denom, -p)
    return project_to_simplex(vals)


# ----------------------------------------------------------------------
# Simple hill‑climber (mass‑transfer moves)
# ----------------------------------------------------------------------


def hill_climb(seq: np.ndarray, deadline: float, max_steps: int = 120_000) -> np.ndarray:
    """
    Stochastic hill‑climber that moves a small fraction of the mass.
    Uses the true (non‑smoothed) objective.
    """
    a = seq.copy()
    n = a.shape[0]
    best_score = compute_score(a)
    step_scale = 0.12  # start with ≈12 % of the donor mass

    while max_steps > 0 and time.time() < deadline:
        i, j = np.random.randint(n, size=2)
        if i == j:
            continue

        cand = a.copy()
        # move a fraction from i to j
        delta = step_scale * cand[i]
        if delta <= 0:
            continue
        cand[i] -= delta
        cand[j] += delta
        # re‑project (keeps sum = 1 and non‑negativity)
        cand = project_to_simplex(cand)

        cur_score = compute_score(cand)
        if cur_score < best_score:
            best_score = cur_score
            a = cand
            step_scale = min(step_scale * 1.06, 0.5)   # increase a bit on success
        else:
            step_scale *= 0.97  # shrink on failure
        max_steps -= 1

    return a

# ----------------------------------------------------------------------
