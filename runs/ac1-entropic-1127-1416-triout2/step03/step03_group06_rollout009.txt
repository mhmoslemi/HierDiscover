def gradient_from_softmax(a, beta=10.0):
    n = a.shape[0]
    # compute conv
    fft_a = np.fft.rfft(a, n=2*n-1)
    conv = np.fft.irfft(fft_a * fft_a, n=2*n-1)
    max_conv = conv.max()
    # softmax weights
    exps = np.exp(beta * (conv - max_conv))
    w = exps / exps.sum()
    # compute gradient: g = 2 * sum_k w_k * a_{k - i}
    # compute via convolution of a_rev and w
    a_rev = a[::-1]
    L = len(a_rev) + len(w) - 1
    fft_rev = np.fft.rfft(a_rev, n=L)
    fft_w = np.fft.rfft(w, n=L)
    conv_rev = np.fft.irfft(fft_rev * fft_w, n=L)
    grad = 2.0 * conv_rev[n-1:n-1+n]  # slice length n
    return grad, max_conv

def local_optimize(a0, max_iters=2000, beta_start=5.0, beta_end=30.0, alpha=0.05, deadline=None):
    a = a0.copy()
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    n = a.shape[0]
    for t in range(1, max_iters+1):
        if deadline is not None and time.time() >= deadline:
            break
        beta = beta_start + (beta_end - beta_start) * (t-1)/(max_iters-1)
        grad, max_conv = gradient_from_softmax(a, beta=beta)
        # scale grad by 2*n (objective gradient)
        grad_scaled = 2.0 * n * grad
        m = beta1 * m + (1-beta1) * grad_scaled
        v = beta2 * v + (1-beta2) * (grad_scaled**2)
        m_hat = m / (1 - beta1**t)

        # optional early break if max_conv small?
    return a

"""
Genetic search + Adam‑style soft‑max gradient refinement.

The evaluation metric (after normalising the vector to sum‑1) is
    f(a) = 2·n·max_k (a∗a)_k .
We minimise f on the probability simplex.
The algorithm works in three phases:

1.  Initialise a diverse population (deterministic ramps, exponentials,
    polynomials, Beta‑distributions and random Dirichlet draws).  All
    candidates are normalised to sum 1 and ranked by the objective.

2.  Run a steady‑state genetic loop:
      – keep the `POP_SIZE` best individuals,
      – create `POP_SIZE` children by convex combinations of two random
        elite parents,
      – mutate them with multiplicative Gaussian noise,
      – project back onto the simplex.
   The loop runs until the time budget is exhausted (or the target
   score ≤ 1.503 is reached).  The current global best is printed
   regularly.

3.  After the GA finishes, a short Adam‑optimisation with a smooth
   soft‑max approximation of the max‑convolution is performed on the
   best individual to obtain a fine‑grained local improvement.

All operations are vectorised, convolution is performed by FFTs,
and the simplex projection follows Duchi‑et al. (2008).  The function
`propose_candidate` returns the best sequence found as a plain
`list[float]`.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# 1.  Simplex projection (Euclidean)
# ----------------------------------------------------------------------
def simplex_projection(v: np.ndarray) -> np.ndarray:
    """Project a vector onto the probability simplex."""
    v = np.asarray(v, dtype=float)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1] - 1           # zero‑based
    theta = cssv[cond][-1] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w

# ----------------------------------------------------------------------
# 2.  Objective (assumes Σ a = 1)
# ----------------------------------------------------------------------
def compute_objective(a: np.ndarray) -> float:
    """Return 2·n·max(a∗a) for a probability vector a."""
    n = a.shape[0]
    fft = np.fft.rfft(a, n=2 * n - 1)
    conv = np.fft.irfft(fft * fft, n=2 * n - 1)
    return 2.0 * n * conv.max()

# ----------------------------------------------------------------------
# 3.  Seed generators
# ----------------------------------------------------------------------
def beta_sequence(alpha: float, beta: float, n: int) -> np.ndarray:
    """Discretised Beta(α,β) density on the mid‑points, normalised."""
    x = (np.arange(n, dtype=float) + 0.5) / n
    logw = (alpha - 1.0) * np.log(x) + (beta - 1.0) * np.log(1.0 - x)
    logw = np.clip(logw, -700.0, 700.0)
    w = np.exp(logw)
    total = w.sum()
    if total == 0 or not np.isfinite(total):
        w = np.ones_like(w)
        total = w.sum()
    return w / total


def exponential_shape(lam: float, n: int) -> np.ndarray:
    """a_i ∝ exp(-lam·i), normalised."""
    i = np.arange(n, dtype=float)
    w = np.exp(-lam * i)
    w /= w.sum()
    return w


def polynomial_shape(p: float, n: int) -> np.ndarray:
    """a_i ∝ 1 - (i/(n‑1))**p, normalised."""
    i = np.arange(n, dtype=float)
    frac = i / (n - 1)
    w = 1.0 - np.power(frac, p)
    w = np.clip(w, 0.0, None)
    w /= w.sum()
    return w

# ----------------------------------------------------------------------
# 4.  Soft‑max gradient of the max‑convolution
# ----------------------------------------------------------------------
def softmax_conv_gradient(a: np.ndarray, beta: float = 10.0):
    """
    Compute a smooth approximation of the sub‑gradient of
    max_k (a∗a)_k using a soft‑max with temperature 1/β.
    Returns the gradient (shape n) and the current max convolution value.
    """
    n = a.shape[0]
    # full convolution a*a
    fft_a = np.fft.rfft(a, n=2 * n - 1)
    conv = np.fft.irfft(fft_a * fft_a, n=2 * n - 1)
    max_conv = conv.max()

    # soft‑max weights (numerically safe)
    exps = np.exp(beta * (conv - max_conv))
    w = exps / exps.sum()                     # length 2n‑1, sum to 1

    # gradient = 2 * sum_k w_k * a_{k‑i}
    a_rev = a[::-1]
    L = len(a_rev) + len(w) - 1               # = 3n‑2
    fft_rev = np.fft.rfft(a_rev, n=L)
    fft_w   = np.fft.rfft(w, n=L)
    conv_rev = np.fft.irfft(fft_rev * fft_w, n=L)
    # slice gives the required n‑vector
    grad = 2.0 * conv_rev[n - 1:n - 1 + n]
    return grad, max_conv

# ----------------------------------------------------------------------
# 5.  Adam optimiser on the simplex (using the smooth gradient)
# ----------------------------------------------------------------------
def adam_optimize(
    a0: np.ndarray,
    max_iters: int,
    beta_start: float,
    beta_end: float,
    alpha: float,
    deadline: float,
):
    """
    Adam optimiser (projected onto the simplex) using the
    soft‑max gradient.  The temperature β is linearly increased
    from `beta_start` to `beta_end`.
    """
    a = a0.copy()
    n = a.shape[0]
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-12

    for t in range(1, max_iters + 1):
        if time.time() >= deadline:
            break

        # schedule for β
        beta = beta_start + (beta_end - beta_start) * (t - 1) / (max_iters - 1)

        grad, _ = softmax_conv_gradient(a, beta=beta)
        # scale by the outer factor 2·n (objective = 2·n·max)
        grad_scaled = 2.0 * n * grad

        m = beta1 * m + (1.0 - beta1) * grad_scaled
        v = beta2 * v + (1.0 - beta2) * (grad_scaled ** 2)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)

        step = -alpha * m_hat / (np.sqrt(v_hat) + eps)
        a = a + step
        a = np.clip(a, 0.0, None)

        s = a.sum()
        if s == 0.0:
            a = np.full(n, 1.0 / n)
        else:
            a /= s

    return a

# 6.  Genetic algorithm core
# ----------------------------------------------------------------------

    Search for a non‑negative step function minimising ``evaluate_sequence``.
    Returns the best sequence discovered before the time budget expires.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 5.0      # safety margin

    # --------------------------------------------------------------
    # Problem size
    # --------------------------------------------------------------
    try:

    best_seq = None

    def consider(candidate: np.ndarray):
        nonlocal best_seq, best_score
        score = compute_objective(candidate)
        if score < best_score:

        return (score, candidate)

    # --------------------------------------------------------------

    # --------------------------------------------------------------
    POP_SIZE = 30
    population = []   # list of (score, vector)

    # deterministic families

    a_ramp = np.arange(n, 0, -1, dtype=float)
    a_ramp = a_ramp / a_ramp.sum()

    for lam in np.linspace(0.001, 5.0, 10):
        population.append(consider(exponential_shape(lam, n)))

    for p in [0.5, 1.0, 1.5, 2.0, 3.0]:

    # keep the best POP_SIZE individuals

        # update global best if improved
