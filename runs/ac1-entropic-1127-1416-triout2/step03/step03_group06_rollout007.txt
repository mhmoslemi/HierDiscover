"""
Propose a candidate step‑function that (hopefully) improves the
known upper bound 1.521639.  The algorithm proceeds in three stages:

1. **Diverse seeding** – generate many analytically‑defined
   non‑negative sequences (uniform, exponential, polynomial,
   power‑law, reciprocal, Beta‑distribution, random Dirichlet,
   and mixtures) and keep the best few (the “top‑K” list).

2. **Local refinement** – from the best seed run a projected
   sub‑gradient descent on the simplex with Nesterov‑style momentum,
   a diminishing step‑size, and occasional random perturbations to
   escape plateaus.  The descent is stopped as soon as the time
   budget is exhausted or a target score ≤ 1.503 is reached.

3. **Result** – return the best sequence (normalised to sum 1) as a
   plain ``list[float]``.  All numbers are non‑negative, the sum is
   positive and the sequence length matches the reference
   ``height_sequence_1`` if it exists, otherwise a fallback length
   of 5146.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Basic geometric utilities
# ----------------------------------------------------------------------


def simplex_projection(v):
    """
    Euclidean projection of a vector onto the probability simplex:
    { w | w_i ≥ 0 , Σ w_i = 1 }.
    """
    v = np.asarray(v, dtype=float)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1] - 1
    theta = cssv[cond][-1] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def compute_objective(a):
    """
    Objective for a normalized vector (Σ a_i = 1):
    2·n·max(a * a)  where * denotes linear convolution.
    """
    n = a.shape[0]
    fft = np.fft.rfft(a, n=2 * n - 1)
    conv = np.fft.irfft(fft * fft, n=2 * n - 1)
    return float(2.0 * n * conv.max())


def subgradient_active(a, active):
    """
    Averaged sub‑gradient of the max‑convolution entry over the
    indices in ``active``.
    """
    n = a.shape[0]
    g = np.zeros_like(a)
    for k in active:
        low = max(0, k - n + 1)
        high = min(n - 1, k)
        idx = np.arange(low, high + 1, dtype=int)
        partner = k - idx
        g[idx] += 2.0 * a[partner]
    g /= float(active.size)
    return g


# ----------------------------------------------------------------------
# Families of candidate sequences
# ----------------------------------------------------------------------


def beta_sequence(alpha, beta, n):
    """Discretised Beta(α,β) density on ``n`` mid‑points, normalised to sum 1."""
    x = (np.arange(n) + 0.5) / n
    logw = (alpha - 1.0) * np.log(x) + (beta - 1.0) * np.log(1.0 - x)
    # protect against overflow/under‑flow
    logw = np.clip(logw, -700.0, 700.0)
    w = np.exp(logw)
    total = w.sum()
    if total <= 0.0 or not np.isfinite(total):
        w = np.ones_like(w)
        total = w.sum()
    return w / total


def exponential_shape(lam, n):
    """a_i ∝ exp(-lam·i)  (lam ≥ 0), normalised to sum 1."""
    i = np.arange(n, dtype=float)
    w = np.exp(-lam * i)
    w /= w.sum()
    return w


def polynomial_shape(p, n):
    """a_i ∝ 1 - (i/(n‑1))**p  (p ≥ 0), normalised."""
    i = np.arange(n, dtype=float)
    frac = i / (n - 1)
    w = 1.0 - np.power(frac, p)
    w = np.clip(w, 0.0, None)
    w /= w.sum()
    return w


def edge_power_shape(p, n, eps=1e-12):
    """
    U‑shaped distribution: a_i ∝ ((i+1)*(n-i))^{-p}.
    Larger p concentrates more mass at the two ends.
    """
    i = np.arange(n, dtype=float) + 1.0          # i+1
    prod = i * (n - i + 1.0)                     # (i+1)*(n-i)
    # use log to avoid overflow for large p
    logw = -p * np.log(prod + eps)
    w = np.exp(np.clip(logw, -700.0, 700.0))
    w /= w.sum()
    return w


def edge_reciprocal_shape(n, eps=1e-12):
    """Special case p=1 of edge_power_shape."""
    return edge_power_shape(1.0, n, eps)


# ----------------------------------------------------------------------
# Main optimisation routine
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 0, budget_s: int = 1000):
    """
    Search for a step‑function (list of non‑negative heights) that
    minimises ``evaluate_sequence``.  The routine respects the 1000 s
    time budget and returns the best sequence discovered.
    """
    np.random.seed(seed)
    start_time = time.time()
    # a few seconds safety margin for the external driver
    deadline = start_time + budget_s - 5.0

    # --------------------------------------------------------------
    # 1. Problem size
    # --------------------------------------------------------------
    hs = globals().get("height_sequence_1")
    if isinstance(hs, (list, np.ndarray)):
        n = len(hs)
    else:
        n = 5146                     # fallback length from the prompt

    # --------------------------------------------------------------
    # 2. Book‑keeping utilities
    # --------------------------------------------------------------
    best_seq = None
    best_score = np.inf
    TOP_K = 8
    top = []                         # list of (score, seq)

    def consider(candidate):
        """Normalise *candidate*, evaluate it, and possibly update the top‑K."""
        nonlocal best_seq, best_score, top
        cand = np.asarray(candidate, dtype=float)
        cand = np.maximum(cand, 0.0)
        s = cand.sum()
        if s < 1e-12:                 # avoid division by zero
            return
        cand /= s                      # enforce Σ a_i = 1
        score = compute_objective(cand)
        if score < best_score:
            best_score = score
            best_seq = cand.copy()
            print(f"New global best (seed) : {best_score:.6f}")
        # maintain the sorted top‑K list
        if len(top) < TOP_K:
            top.append((score, cand.copy()))
            top.sort(key=lambda x: x[0])
        else:
            if score < top[-1][0]:
                top[-1] = (score, cand.copy())
                top.sort(key=lambda x: x[0])

    # --------------------------------------------------------------
    # 3. Seed families
    # --------------------------------------------------------------
    # a) reference sequence if it exists
    if isinstance(hs, (list, np.ndarray)):
        seq_ref = np.maximum(np.asarray(hs, dtype=float), 0.0)
        if seq_ref.sum() > 0:
            consider(seq_ref)

    # b) uniform
    consider(np.full(n, 1.0 / n))

    # c) linear decreasing ramp (heavy on the left)
    ramp = np.arange(n, 0, -1, dtype=float)
    consider(ramp)

    # d) exponential family (lam in log‑space)
    for lam in np.logspace(-3, np.log10(5.0), num=20):
        consider(exponential_shape(lam, n))

    # e) polynomial family (p in [0,5])
    for p in np.linspace(0.0, 5.0, num=11):
        consider(polynomial_shape(p, n))

    # f) edge‑power shapes (U‑shaped)
    for p in [0.2, 0.5, 1.0, 1.5, 2.0, 3.0]:
        consider(edge_power_shape(p, n))

    # g) edge‑reciprocal (p = 1)
    consider(edge_reciprocal_shape(n))

    # h) Beta distribution sweep – include very small alphas/betas
    beta_vals = [0.01, 0.02, 0.05, 0.1, 0.2, 0.5, 1.0, 2.0, 5.0]
    for α in beta_vals:
        for β in beta_vals:
            if time.time() >= deadline:
                break
            consider(beta_sequence(α, β, n))
        if time.time() >= deadline:
            break

    # i) Random Dirichlet‑like vectors (concentration < 1 encourages spikes)
    for _ in range(150):
        if time.time() >= deadline:
            break
        # a random concentration per coordinate in (0.05, 1.0)
        conc = np.random.uniform(0.05, 1.0, size=n)
        a_dir = np.random.gamma(conc, 1.0)
        consider(a_dir)

    # j) Random Beta parameters (continuous sampling)
    for _ in range(100):
        if time.time() >= deadline:
            break
        α = np.random.uniform(0.01, 5.0)
        β = np.random.uniform(0.01, 5.0)
        consider(beta_sequence(α, β, n))

    # --------------------------------------------------------------
    # 4. Convex combinations of the current top candidates
    # --------------------------------------------------------------
    for _ in range(200):
        if time.time() >= deadline:
            break
        if len(top) < 2:
            break
        i, j = np.random.choice(len(top), 2, replace=False)
        w = np.random.rand()
        mix = w * top[i][1] + (1.0 - w) * top[j][1]
        consider(mix)

    # --------------------------------------------------------------
    # 5. Local refinement (projected sub‑gradient descent)
    # --------------------------------------------------------------
    def local_descent(start_vec):
        """Run sub‑gradient descent starting from *start_vec*."""
        a = start_vec.copy()
        prev_a = a.copy()
        best_local = a.copy()
        best_local_score = compute_objective(a)
        eta = 0.85          # base step‑size coefficient
        mu = 0.55           # momentum coefficient
        stagnation = 0
        iteration = 0
        max_iter = 800000   # hard cap; time limit will usually stop earlier

        while time.time() < deadline and iteration < max_iter:
            # --- convolution and current score ---
            fft = np.fft.rfft(a, n=2 * n - 1)
            conv = np.fft.irfft(fft * fft, n=2 * n - 1)
            max_b = conv.max()
            cur_score = 2.0 * n * max_b

            # --- record improvement ---
            if cur_score < best_local_score - 1e-12:
                best_local_score = cur_score
                best_local = a.copy()
                print(f"Iter {iteration:6d}: new local best {best_local_score:.6f}")
                stagnation = 0
            else:
                stagnation += 1

            # --- active set (indices where conv ≈ max) ---
            eps = max_b * 1e-7
            active = np.where(conv >= max_b - eps)[0]
            if active.size == 0:
                active = np.array([conv.argmax()], dtype=int)

            # --- sub‑gradient ---
            g = subgradient_active(a, active)

            # --- step‑size (1/√t schedule) ---
            step = eta / np.sqrt(iteration + 1)

            # --- try a few step‑size factors and pick the best ---
            best_candidate = None
            best_candidate_score = np.inf
            for factor in (1.0, 0.5, 0.2):
                a_candidate = a - step * factor * g + mu * (a - prev_a)
                a_candidate = simplex_projection(a_candidate)
                sc = compute_objective(a_candidate)
                if sc < best_candidate_score:
                    best_candidate_score = sc
                    best_candidate = a_candidate

            # update
            prev_a, a = a, best_candidate
            iteration += 1

            # --- occasional random perturbation to escape plateaus ---
            if stagnation >= 25000:
                sigma = max(0.01, 0.5 / np.sqrt(iteration + 1))
                noise = sigma * np.random.randn(n) * a
                a = a + noise
                a = np.clip(a, 0.0, None)
                a /= a.sum()
                prev_a = a.copy()
                stagnation = 0
                print(f"Iter {iteration:6d}: perturbation, score {cur_score:.6f}")

            # early stop if we already beat the target
            if best_local_score <= 1.503:
                break

        return best_local, best_local_score

    # Run descent from the best seed discovered so far
    if best_seq is None:
        # fallback – uniform distribution
        best_seq = np.full(n, 1.0 / n)

    print(f"Starting local descent from seed score {best_score:.6f}")
    refined_seq, refined_score = local_descent(best_seq)

    # possibly improve further from the second‑best seed if time permits
    if time.time() < deadline and len(top) > 1:
        sec_seq = top[1][1]
        print(f"Trying second‑best seed (score {top[1][0]:.6f})")
        seq2, score2 = local_descent(sec_seq)
        if score2 < refined_score:
            refined_seq, refined_score = seq2, score2

    # --------------------------------------------------------------
    # 6. Return the best sequence discovered
    # --------------------------------------------------------------
    print(f"Finished search; best score = {refined_score:.6f}")
    return [float(v) for v in refined_seq.tolist()]
