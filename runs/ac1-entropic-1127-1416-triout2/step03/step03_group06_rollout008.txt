"""
Propose a candidate step‑function that improves the current best bound.
The algorithm combines a rich set of deterministic and random seeds,
takes the best few of them and refines each by a soft‑max‑smoothed
gradient descent on the probability simplex (Adam + Euclidean projection).
A gradual increase of the soft‑max temperature focuses the descent onto the
true max‑convolution entry, while a small learning‑rate schedule and occasional
random jitter help escape plateaus.
"""

import time
import numpy as np

# ------------------------------------------------------------
# Simplex projection (Euclidean) – Duchi et al. (2008)
# ------------------------------------------------------------
def simplex_projection(v):
    v = np.asarray(v, dtype=float)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all non‑positive → uniform distribution
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1] - 1                     # zero‑based index
    theta = cssv[cond][-1] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w

# ------------------------------------------------------------
# Objective for sequences that already sum to 1:
#   f(a) = 2·n·max(a*a)
# ------------------------------------------------------------
def compute_objective(a):
    n = a.shape[0]
    # auto‑convolution via FFT (length 2·n‑1)
    fft = np.fft.rfft(a, n=2 * n - 1)
    conv = np.fft.irfft(fft * fft, n=2 * n - 1)
    return 2.0 * n * conv.max()

# ------------------------------------------------------------
# Helper: full auto‑convolution (length 2·n‑1)
# ------------------------------------------------------------
def full_convolution(a):
    n = a.shape[0]
    fft = np.fft.rfft(a, n=2 * n - 1)
    conv = np.fft.irfft(fft * fft, n=2 * n - 1)
    return conv

# ------------------------------------------------------------
# Soft‑max weights that approximate the indicator of the
# maximal convolution entry (γ controls the sharpness)
# ------------------------------------------------------------
def softmax_weights(conv, gamma):
    max_c = conv.max()
    # shift by the maximum to avoid overflow
    exp_term = np.exp(gamma * (conv - max_c))
    s = exp_term.sum()
    if s == 0 or not np.isfinite(s):
        # fall back to uniform weights
        return np.full_like(conv, 1.0 / conv.size)
    return exp_term / s

# ------------------------------------------------------------
# Gradient of the soft‑max‑averaged convolution:
#   g = 2 * (a_rev * w)[n‑1 : n‑1+n]
# where * denotes (full) convolution.
# ------------------------------------------------------------
def gradient_from_weights(a, w):
    n = a.shape[0]
    a_rev = a[::-1]
    # convolution via FFT (both vectors length 2·n‑1)
    conv = np.fft.irfft(np.fft.rfft(a_rev, n=2 * n - 1) *
                        np.fft.rfft(w, n=2 * n - 1),
                        n=2 * n - 1)
    # central slice has length n
    return 2.0 * conv[n - 1:n - 1 + n]

# ------------------------------------------------------------
# Families for the initial pool of seeds
# ------------------------------------------------------------
def beta_sequence(alpha, beta, n):
    """Discretised Beta(α,β) density on the mid‑points, normalised to sum 1."""
    x = (np.arange(n) + 0.5) / n
    # work in log‑space to avoid overflow
    logw = (alpha - 1.0) * np.log(x) + (beta - 1.0) * np.log(1.0 - x)
    logw = np.clip(logw, -700.0, 700.0)
    w = np.exp(logw)
    total = w.sum()
    if total == 0 or not np.isfinite(total):
        w = np.ones_like(w)
        total = w.sum()
    return w / total


def exponential_shape(lam, n):
    """a_i ∝ exp(‑lam·i) , λ≥0."""
    i = np.arange(n, dtype=float)
    w = np.exp(-lam * i)
    w /= w.sum()
    return w


def polynomial_shape(p, n):
    """a_i ∝ 1 − (i/(n‑1))^p , p≥0."""
    i = np.arange(n, dtype=float)
    frac = i / (n - 1)
    w = 1.0 - np.power(frac, p)
    w = np.clip(w, 0.0, None)
    w /= w.sum()
    return w


# ------------------------------------------------------------
# Main search routine
# ------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: int = 1000):
    """
    Return a list of non‑negative heights (a Python list of floats)
    that minimises ``evaluate_sequence`` under the time budget.
    """
    np.random.seed(seed)
    start_time = time.time()
    # keep a small safety margin
    deadline = start_time + budget_s - 5.0

    # --------------------------------------------------------
    # 1. Length of the step function (fallback to 5146)
    # --------------------------------------------------------
    try:
        n = len(height_sequence_1)          # type: ignore[name-defined]
    except NameError:
        n = 5146

    # --------------------------------------------------------
    # 2. Helpers to keep the globally best solution
    # --------------------------------------------------------
    best_seq = None
    best_score = np.inf
    TOP_K = 5
    top = []                       # list of (score, seq)

    def consider(candidate):
        """Evaluate *candidate* (assumed already normalised) and update the
        global best and the top‑K list."""
        nonlocal best_seq, best_score, top
        s = compute_objective(candidate)
        if s < best_score:
            best_score = s
            best_seq = candidate.copy()
            print(f"[init] new global best  {s:.6f}")
        if len(top) < TOP_K:
            top.append((s, candidate.copy()))
            top.sort(key=lambda x: x[0])
        else:
            if s < top[-1][0]:
                top[-1] = (s, candidate.copy())
                top.sort(key=lambda x: x[0])

    # --------------------------------------------------------
    # 3. Deterministic seed families
    # --------------------------------------------------------
    # Uniform
    uniform = np.full(n, 1.0 / n)
    consider(uniform)

    # Linear decreasing ramp (the analytic construction used before)
    ramp = np.arange(n, 0, -1, dtype=float)
    ramp = ramp / ramp.sum()
    consider(ramp)

    # Exponential family (λ from 0.001 to 10, 25 values)
    for lam in np.logspace(-3, 1, num=25):
        if time.time() >= deadline:

        cand = exponential_shape(lam, n)
        consider(cand)

    # Polynomial family (various powers)
    for p in [0.5, 1.0, 1.5, 2.0, 3.0, 5.0, 8.0]:
        if time.time() >= deadline:

    # --------------------------------------------------------

    # --------------------------------------------------------
    alphas = np.linspace(0.2, 5.0, 10)

    for α in alphas:

                break

    # --------------------------------------------------------

    # --------------------------------------------------------
    for _ in range(200):

        # Dirichlet‑like (Gamma) draw

        a = np.random.gamma(conc, 1.0)
        if a.sum() == 0:

        a /= a.sum()
        consider(a)

        # Random Beta parameters
        α = np.random.uniform(0.05, 5.0)
        β = np.random.uniform(0.05, 5.0)
        cand = beta_sequence(α, β, n)
        consider(cand)

    # --------------------------------------------------------

    try:

        if previous.sum() != 0:

            consider(previous)
    except NameError:
        pass

    # --------------------------------------------------------
    # 7. Refine the TOP_K seeds with a soft‑max‑smoothed Adam descent
    # --------------------------------------------------------
    # Adam hyper‑parameters
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8
    base_lr = 0.25          # initial learning‑rate (will be divided by √t)

    # soft‑max schedule
    gamma_start = 1.0
    gamma_max = 30.0
    gamma_mul = 1.001       # slowly increase γ each iteration

    # iterate over the best seeds (most promising first)
    for rank, (seed_score, seed_seq) in enumerate(top):
        if time.time() >= deadline:
            break
        print(f"\n[refine] starting refinement from seed #{rank+1} "
              f"(score {seed_score:.6f})")
        a = seed_seq.copy()            # already sums to 1
        m = np.zeros_like(a)           # 1st moment
        v = np.zeros_like(a)           # 2nd moment
        t_iter = 0
        gamma = gamma_start
        stagnation = 0

        while time.time() < deadline:
            # ----------------------------------------------------
            # compute convolution and current objective
            # ----------------------------------------------------
            conv = full_convolution(a)
            max_conv = conv.max()
            cur_score = 2.0 * n * max_conv

            # keep the best ever seen
            if cur_score < best_score - 1e-12:
                best_score = cur_score
                best_seq = a.copy()
                print(f"[improve] new global best {best_score:.6f}")
                stagnation = 0
            else:
                stagnation += 1

            # ----------------------------------------------------
            # soft‑max weights → smooth sub‑gradient
            # ----------------------------------------------------
            w = softmax_weights(conv, gamma)

            # ----------------------------------------------------
            # gradient (soft‑max averaged)
            # ----------------------------------------------------
            grad = gradient_from_weights(a, w)

            # ----------------------------------------------------
            # Adam update + projection onto the simplex
            # ----------------------------------------------------
            t_iter += 1
            m = beta1 * m + (1.0 - beta1) * grad
            v = beta2 * v + (1.0 - beta2) * (grad * grad)

            a = np.clip(a, 0.0, None)
            a = simplex_projection(a)

            # ----------------------------------------------------
            # temperature schedule for the soft‑max
            # ----------------------------------------------------
            if gamma < gamma_max:

            # ----------------------------------------------------

            # ----------------------------------------------------

                jitter = sigma * np.random.randn(n) * a
                a = a + jitter
                a = np.clip(a, 0.0, None)
                a = simplex_projection(a)
                stagnation = 0
                print(f"[jitter] applied at iter {t_iter:,}")

            # ----------------------------------------------------
            # early exit if we have already reached the target
            # ----------------------------------------------------
            if best_score <= 1.503:

                break

        # end while (time exhausted or target reached)
        if best_score <= 1.503:
            break

    # ------------------------------------------------------------
    # 8. Return the best sequence discovered (as a plain list)
    # ------------------------------------------------------------
    print(f"\n[final] finished search – best score = {best_score:.6f}")
    # safety guard – ensure a valid non‑empty sequence is returned
    if best_seq is None:
        # fallback: uniform distribution (should never happen)
        best_seq = np.full(n, 1.0 / n)

    return [float(v) for v in best_seq.tolist()]
