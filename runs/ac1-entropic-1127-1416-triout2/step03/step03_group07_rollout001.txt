def propose_candidate(budget_s=1000):
    deadline = time.time() + budget_s - 5.0
    # choose n
    n = 8000
    # Initialize best
    best_u = None
    best_score = float('inf')
    best_a = None
    # get init from height_sequence_1 if available, else random
    try:
        init_seq = np.asarray(height_sequence_1, dtype=float) # potentially length maybe 5146
        if init_seq.shape[0] >= n:
            init_seq = init_seq[:n]
        else:
            # pad with zeros
            init_seq = np.pad(init_seq, (0, n - init_seq.shape[0]), 'constant')
        # compute logits u = log(init_seq) (set small floor)
        eps = 1e-12
        init_seq = np.maximum(init_seq, eps)
        init_u = np.log(init_seq)
    except Exception:
        init_u = None
    # we will run multiple restarts
    restart = 0
    while time.time() < deadline:
        # Create new u initialization
        if restart == 0 and init_u is not None:
            u = init_u.copy()
        else:
            # random normal initialization
            rng = np.random.default_rng(seed=int(time.time()*1e6) % 2**32)
            u = rng.normal(loc=0.0, scale=1.0, size=n)
        # Adam variables
        m = np.zeros_like(u)
        v = np.zeros_like(u)
        beta1 = 0.9
        beta2 = 0.999
        eps = 1e-8
        lr = 0.1   # initial learning rate
        t_iter = 0
        while time.time() < deadline:
            # softmax a
            u_shift = u - np.max(u)  # for numerical stability
            exp_u = np.exp(u_shift)
            a = exp_u / exp_u.sum()
            # compute max conv and gradient
            max_conv, grad_max = _max_and_subgradient(a)  # grad_max is subgradient of max conv
            obj = 2.0 * n * max_conv
            # check best
            if obj < best_score:
                best_score = obj
                best_a = a.copy()
                best_u = u.copy()
                print(f"[{time.time():.1f}] restart {restart}, iter {t_iter}: new best = {best_score:.6f}")
                if best_score < 1.503:
                    print("Target reached.")
                    return [float(v) for v in best_a.tolist()]
            # compute gradient w.r.t a
            grad_a = 2.0 * n * grad_max  # shape (n,)
            # gradient w.r.t logits u
            dot = np.dot(grad_a, a)
            grad_u = a * (grad_a - dot)
            # Adam update
            t_iter += 1
            m = beta1 * m + (1 - beta1) * grad_u
            v = beta2 * v + (1 - beta2) * (grad_u ** 2)
            m_hat = m / (1 - beta1 ** t_iter)
            v_hat = v / (1 - beta2 ** t_iter)
            # learning rate schedule: decay with sqrt(t)
            lr_t = lr / np.sqrt(t_iter)
            u -= lr_t * m_hat / (np.sqrt(v_hat) + eps)
            # optional jitter after some iterations
            if t_iter % 5000 == 0:
                # add small gaussian noise
                noise = 0.01 * rng.standard_normal(n)
                u += noise
            # stop condition within restart (if gradient small)
            if np.linalg.norm(grad_u) < 1e-12:
                break
        restart += 1
    # after time out, return best
    if best_a is None:
        # fallback uniform
        a = np.full(n, 1.0 / n)
        best_a = a
    return [float(v) for v in best_a.tolist()]

"""
Propose a step‑function (non‑negative sequence) that minimizes

        2·n·max(a*a) / (sum(a)**2)

using a projected Adam optimizer on the simplex (softmax
parameterisation).  The optimizer works directly on the
soft‑max logits (unconstrained) and uses the exact sub‑gradient of
the max‑convolution (computed via an FFT).  Several random restarts,
learning‑rate decay and occasional Gaussian jitter are employed to
escape plateaus.  The current best sequence is kept and returned
when the time budget expires (or as soon as the target value <1.503
is reached).
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper: max of self‑convolution and a sub‑gradient for the active set.
# ----------------------------------------------------------------------
def _max_and_subgradient(a: np.ndarray) -> tuple[float, np.ndarray]:
    """
    Returns (max_entry, subgradient) of the discrete convolution a*a.

    The subgradient is the average over all indices where the maximum
    is attained (as in the reference implementation).
    """
    n = a.shape[0]
    # convolution via FFT (length 2n‑1)
    fft = np.fft.rfft(a, n=2 * n - 1)
    conv = np.fft.irfft(fft * fft, n=2 * n - 1)

    max_val = float(conv.max())

    # collect (approximately) active indices
    tol = max_val * 1e-12 + 1e-14
    active = np.where(conv >= max_val - tol)[0]

    grad = np.zeros_like(a)
    for k in active:
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        idx = np.arange(lo, hi + 1, dtype=int)
        partner = k - idx
        grad[idx] += 2.0 * a[partner]

    if active.size > 1:
        grad /= float(active.size)

    return max_val, grad

# ----------------------------------------------------------------------

        2·n·max(a*a) / (sum(a)**2)

    best sequence found.  It uses a soft‑max + Adam optimisation scheme
    with multiple random restarts.

    def _initial_logits():

        # Try to reuse the historic best, otherwise use a smooth sine shape.
        try:
            seq = np.asarray(height_sequence_1, dtype=float)  # type: ignore[name-defined]
            seq = np.maximum(seq, 1e-12)
            # If dimensions differ, either truncate or pad with a small constant.
            if seq.shape[0] > n:
                seq = seq[:n]
            elif seq.shape[0] < n:

                seq = np.concatenate([seq, pad])

        except Exception:

            x = np.arange(n, dtype=float)

            lr_t = lr0 / np.sqrt(t_iter)

            # --------------------------------------------------------------

            # --------------------------------------------------------------

                break

    # ------------------------------------------------------------------

    # finite, not too large)
    best_a = np.maximum(best_a, 0.0)
    best_a = np.minimum(best_a, 1000.0)

    return [float(v) for v in best_a.tolist()]
