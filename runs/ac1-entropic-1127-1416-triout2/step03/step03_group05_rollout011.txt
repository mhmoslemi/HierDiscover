Fast softmax‑augmented Adam optimizer for the max‑convolution objective.

The algorithm optimises a probability distribution a (Σa=1, a_i≥0) that
minimises 2·n·max(conv(a,a)).  Instead of using a sub‑gradient of the
hard maximum we introduce a temperature‑controlled softmax over all
convolution entries.  The gradient of the smoothed objective is obtained
analytically via an additional convolution, which together with an FFT
implementation yields an O(n log n) per‑iteration cost.  A continuation
scheme raises the softmax temperature during optimisation, allowing the
optimizer to first explore a smooth landscape and then focus on the true
maximum.  Multiple short restarts with diverse initialisations (previous
solution, uniform, Dirichlet, geometric) are performed inside the given
time budget.  The best sequence found is returned as a plain list of
floats.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper functions – no closures
# ----------------------------------------------------------------------


def _initial_sequence(kind: str, n: int, rng: np.random.Generator,
                      prev: np.ndarray | None) -> np.ndarray:
    """
    Produce an initial feasible probability vector of length ``n``.
    ``kind`` can be 'prev', 'uniform', 'random_dirichlet', 'geometric'.
    """
    if kind == "prev" and prev is not None:
        seq = prev.copy()
        if seq.shape[0] != n:
            if seq.shape[0] > n:
                seq = seq[:n]
            else:
                seq = np.concatenate([seq, np.zeros(n - seq.shape[0], dtype=seq.dtype)])
        seq = np.maximum(seq, 0.0)
        tot = seq.sum()
        if tot == 0.0:
            return np.full(n, 1.0 / n)
        return seq / tot

    if kind == "uniform":
        return np.full(n, 1.0 / n)

    if kind == "random_dirichlet":
        # concentration < 1 tends to produce sparse vectors
        alpha = 0.4
        sample = rng.gamma(alpha, 1.0, size=n)
        return sample / sample.sum()

    if kind == "geometric":
        # symmetric geometric decay from both ends, random rate near 1
        r = 0.98 + 0.02 * rng.random()
        left = r ** np.arange(n)
        right = r ** np.arange(n - 1, -1, -1)
        seq = left + right
        return seq / seq.sum()

    # fallback
    return np.full(n, 1.0 / n)


def _softmax_max_and_grad(a: np.ndarray, alpha: float):
    """
    For a probability vector ``a`` (sum = 1) compute

        * the aperiodic convolution c = a * a,
        * the soft‑max weights w_k = exp(alpha * c_k) / Σ exp(alpha * c_j),
        * the gradient of the weighted average Σ w_k c_k with respect to ``a``.
    Returns (c, max_val, grad_a) where ``grad_a`` has the same shape as ``a``.
    """
    n = a.shape[0]
    # Length needed for the three‑term convolution (a with w_rev)
    fft_len = 1 << ((3 * n - 2).bit_length())

    # FFT of a
    fft_a = np.fft.rfft(a, n=fft_len)

    # a * a   (length 2n‑1)
    conv_full = np.fft.irfft(fft_a * fft_a, n=fft_len)
    conv = conv_full[: 2 * n - 1]

    # maximum entry of the convolution (used for shifting)
    max_val = conv.max()

    # soft‑max weights (stable)
    shifted = alpha * (conv - max_val)
    # Clip to avoid overflow in exp
    # (numpy exp works fine with large negative numbers)
    exp_shifted = np.exp(shifted)
    w = exp_shifted / exp_shifted.sum()

    # reverse w for the gradient convolution
    w_rev = w[::-1]

    # FFT of w_rev (same length)
    fft_w_rev = np.fft.rfft(w_rev, n=fft_len)

    # a * w_rev  (length 3n‑2)
    grad_full = np.fft.irfft(fft_a * fft_w_rev, n=fft_len)
    grad_full = grad_full[: 3 * n - 2]

    # pick the n entries that correspond to the gradient of Σ w_k c_k
    # see derivation in the analysis section
    idx = 2 * n - 2 - np.arange(n)
    grad_a = 2.0 * grad_full[idx]

    return conv, max_val, grad_a


def _adam_optimize_softmax(init_a: np.ndarray,
                           deadline: float,
                           restart_id: int,
                           target_score: float = 1.503) -> tuple[np.ndarray, float]:
    """
    Adam optimisation on the soft‑max‑smoothed objective.
    Returns the best probability vector (sum = 1) and the best true score
    (2·n·max(conv)).
    """
    n = init_a.shape[0]
    # logits for the soft‑max parametrisation
    eps = 1e-12
    logits = np.log(init_a + eps)

    # Adam state
    m = np.zeros_like(logits)
    v = np.zeros_like(logits)
    beta1, beta2 = 0.9, 0.999
    eps_adam = 1e-8

    # temperature schedule for the soft‑max
    alpha = 0.5          # start small (smooth landscape)
    alpha_inc = 0.001    # linear increase per iteration
    alpha_max = 200.0    # never go beyond this

    best_score = np.inf
    best_a = init_a.copy()
    stagnation = 0
    it = 0

    while time.time() < deadline:
        it += 1

        # ------------------- probability vector from logits -------------------
        lmax = np.max(logits)
        a = np.exp(logits - lmax)
        a /= a.sum()

        # ------------------- compute convolution, soft‑max gradient ----------
        conv, max_conv, grad_a = _softmax_max_and_grad(a, alpha)

        # true objective (max‑convolution version)
        cur_score = 2.0 * n * max_conv

        # keep the best
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            stagnation = 0
            print(f"Restart {restart_id:02d} iter {it:6d}: best = {best_score:.6f}")
        else:
            stagnation += 1

        # early exit if we already passed the target
        if best_score < target_score:
            break

        # ------------------- stagnation handling ----------------------------
        if stagnation > 5000:
            # add a small random jitter in log‑space to escape plateaus
            logits += 0.01 * np.random.randn(n)
            stagnation = 0

        # ------------------- Adam update ----------------------------------
        # gradient of the smoothed objective: 2·n·grad_a
        grad_obj = 2.0 * n * grad_a

        # transform to logits (Jacobian of softmax)
        avg = np.dot(a, grad_obj)
        grad_logits = a * (grad_obj - avg)

        # moments update
        m = beta1 * m + (1.0 - beta1) * grad_logits
        v = beta2 * v + (1.0 - beta2) * (grad_logits ** 2)

        # bias‑corrected estimates
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        # learning‑rate schedule (diminishing)
        lr = 0.4 / np.sqrt(it)

        logits -= lr * m_hat / (np.sqrt(v_hat) + eps_adam)

        # ------------------- temperature schedule -------------------------
        alpha = min(alpha_max, alpha + alpha_inc)

    return best_a, best_score
