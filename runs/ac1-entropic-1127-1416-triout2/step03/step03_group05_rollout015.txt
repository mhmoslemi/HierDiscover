def _adam_smooth_optimize(init_a, deadline, restart_id, target=1.503):
    n = init_a.shape[0]
    # prepare fft lengths
    conv_len = 2*n - 1
    fft_len1 = 1 << ((conv_len - 1).bit_length())
    grad_len = 3*n - 2
    fft_len2 = 1 << ((grad_len - 1).bit_length())
    eps = 1e-12
    # logits
    b = np.log(init_a + eps)
    # Adam moments
    m = np.zeros_like(b)
    v = np.zeros_like(b)
    beta1 = 0.9
    beta2 = 0.999
    adam_eps = 1e-8
    best_score = float('inf')
    best_a = init_a.copy()
    stagnation = 0
    it = 0
    alpha = 5.0  # start smoothing parameter
    while time.time() < deadline:
        it += 1
        # softmax
        b_max = np.max(b)
        a = np.exp(b - b_max)
        a = a / a.sum()
        # compute conv a*a
        A_fft = np.fft.rfft(a, n=fft_len1)
        conv = np.fft.irfft(A_fft * A_fft, n=fft_len1)[:conv_len]
        max_val = conv.max()
        score = 2.0 * n * max_val
        if score < best_score:
            best_score = score
            best_a = a.copy()
            stagnation = 0
            print(f"Restart {restart_id:02d} iter {it:6d}:   best = {best_score:.6f}")
        else:
            stagnation += 1
        if best_score < target:
            break
        # compute weight for smooth max
        diff = conv - max_val   # <= 0
        exp_vals = np.exp(alpha * diff)
        w = exp_vals / exp_vals.sum()
        # gradient via convolution w * a
        W_fft = np.fft.rfft(w, n=fft_len2)
        A2_fft = np.fft.rfft(a, n=fft_len2)
        grad_full = np.fft.irfft(W_fft * A2_fft, n=fft_len2)[:grad_len]
        grad_a = 2.0 * grad_full[:n]   # derivative of conv under weights
        # gradient of objective (2n factor)
        grad_obj = 2.0 * n * grad_a
        # transform to logits gradient
        avg = np.dot(a, grad_obj)
        grad_b = a * (grad_obj - avg)
        # Adam update
        m = beta1 * m + (1 - beta1) * grad_b
        v = beta2 * v + (1 - beta2) * (grad_b ** 2)
        m_hat = m / (1 - beta1 ** it)
        v_hat = v / (1 - beta2 ** it)
        lr = 0.4 / np.sqrt(it)
        b -= lr * m_hat / (np.sqrt(v_hat) + adam_eps)
        # increase alpha slowly to sharpen max
        if it % 500 == 0:
            alpha = min(alpha * 1.2, 200.0)
        # jitter escape
        if stagnation > 5000:
            b += 0.02 * np.random.randn(n)
            stagnation = 0
    return best_a, best_score

"""
Propose a candidate step‑function sequence that minimizes
    2·n·max(conv(a,a))/(sum(a)**2)

The search uses a smoothed‑max Adam optimizer.  A sequence a is
parameterised by logits b (softmax) so that Σa = 1.  At each iteration
the aperiodic convolution a∗a is computed by FFT, a smooth max is
approximated by a soft‑max with a temperature parameter α, and the
corresponding gradient (which can be written as a convolution of the
soft‑max weights with a) is used to update b with Adam.  α is increased
gradually so the optimisation first sees a smooth landscape and later
focuses on the true maximum.  Several restarts with different
initialisations (previous solution, uniform, geometric, random
Dirichlet) are performed within the time budget.  The best sequence
found is returned as a plain Python list.
"""

import time
import numpy as np

# ------------------------------------------------------------
# Helper utilities
# ------------------------------------------------------------

def _next_pow2(x: int) -> int:
    """Return the smallest power of two >= x."""
    return 1 << ((x - 1).bit_length())


def _full_convolve(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """
    Full aperiodic convolution of a and b using FFT.
    Length = len(a) + len(b) - 1.
    """
    n = a.shape[0] + b.shape[0] - 1
    L = _next_pow2(n)
    A = np.fft.rfft(a, n=L)
    B = np.fft.rfft(b, n=L)
    c = np.fft.irfft(A * B, n=L)
    return c[:n]


def _softmax_logits(logits: np.ndarray) -> np.ndarray:
    """Stable soft‑max that returns a probability vector summing to 1."""
    m = np.max(logits)
    exp = np.exp(logits - m)
    return exp / exp.sum()


def _initial_sequence(kind: str, n: int, rng: np.random.Generator,
                      prev_seq: np.ndarray | None) -> np.ndarray:
    """
    Produce a feasible start probability vector of length n.
    Supported kinds: 'prev', 'uniform', 'random_dirichlet', 'geometric'.
    Result is non‑negative and sums to 1.
    """
    if kind == "prev" and prev_seq is not None:
        seq = prev_seq.astype(float, copy=True)
        if seq.shape[0] > n:
            seq = seq[:n]
        elif seq.shape[0] < n:
            seq = np.concatenate([seq, np.zeros(n - seq.shape[0])])
        seq = np.maximum(seq, 0.0)
        total = seq.sum()
        if total == 0.0:
            return np.full(n, 1.0 / n)
        return seq / total

    if kind == "uniform":
        return np.full(n, 1.0 / n)

    if kind == "random_dirichlet":
        # Dirichlet with concentration < 1 encourages sparsity.
        alpha = 0.5
        sample = rng.gamma(alpha, 1.0, size=n)
        return sample / sample.sum()

    if kind == "geometric":
        # Symmetric geometric decay from both ends.
        r = 0.98 + 0.02 * rng.random()
        left = r ** np.arange(n)
        right = left[::-1]
        seq = left + right
        seq = np.maximum(seq, 0.0)
        seq_sum = seq.sum()
        if seq_sum == 0.0:
            return np.full(n, 1.0 / n)
        return seq / seq_sum

    # fallback
    return np.full(n, 1.0 / n)


def _adam_smooth_optimize(init_a: np.ndarray,
                          deadline: float,
                          restart_id: int,
                          target_score: float = 1.503) -> tuple[np.ndarray, float]:
    """
    Adam optimisation on the logits of a probability vector using a
    smoothed max (soft‑max) of the auto‑convolution.
    Returns the best probability vector found and its score.
    """
    n = init_a.shape[0]
    conv_len = 2 * n - 1
    grad_len = 3 * n - 2
    fft_len_conv = _next_pow2(conv_len)
    fft_len_grad = _next_pow2(grad_len)

    # Initialise logits; small epsilon avoids log(0).
    eps_init = 1e-12
    b = np.log(init_a + eps_init)

    # Adam moments.
    m = np.zeros_like(b)
    v = np.zeros_like(b)
    beta1 = 0.9
    beta2 = 0.999
    adam_eps = 1e-8

    # Smoothing parameter α (starts small, grows gradually).
    alpha = 5.0
    alpha_growth = 1.15  # multiplicative increase every few iterations.

    # bookkeeping
    best_score = float('inf')
    best_a = init_a.copy()
    stagnation = 0
    it = 0

    while time.time() < deadline:
        it += 1

        # 1) probability vector from logits
        a = _softmax_logits(b)

        # 2) convolution a*a (full)
        A_fft = np.fft.rfft(a, n=fft_len_conv)
        conv_full = np.fft.irfft(A_fft * A_fft, n=fft_len_conv)[:conv_len]
        max_val = conv_full.max()
        cur_score = 2.0 * n * max_val

        # keep best
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            stagnation = 0
            print(f"Restart {restart_id:02d} iter {it:6d}: best = {best_score:.6f}")
        else:
            stagnation += 1

        # early exit if target reached
        if best_score < target_score:
            break

        # 3) smooth‑max weights (soft‑max over the convolution entries)
        diff = conv_full - max_val   # ≤ 0
        # To keep numbers moderate we use the current α.
        w_raw = np.exp(alpha * diff)   # largest entry becomes 1.
        w = w_raw / w_raw.sum()        # now Σ w = 1

        # 4) gradient of the smoothed objective.
        #   grad_a = 2 * (w * a) convolution, i.e. 2 * (w ∗ a)[:n]
        W_fft = np.fft.rfft(w, n=fft_len_grad)
        A2_fft = np.fft.rfft(a, n=fft_len_grad)
        grad_full = np.fft.irfft(W_fft * A2_fft, n=fft_len_grad)[:grad_len]
        grad_a = 2.0 * grad_full[:n]               # shape (n,)

        # 5) chain rule through the soft‑max (logits space)
        grad_obj = 2.0 * n * grad_a                # derivative of 2n·smooth_max
        avg = np.dot(a, grad_obj)
        grad_b = a * (grad_obj - avg)              # Jacobian of soft‑max

        # 6) Adam update on logits
        m = beta1 * m + (1.0 - beta1) * grad_b
        v = beta2 * v + (1.0 - beta2) * (grad_b ** 2)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        lr = 0.4 / np.sqrt(it)                     # diminishing learning‑rate
        b -= lr * m_hat / (np.sqrt(v_hat) + adam_eps)

        # 7) schedule for the smoothing parameter
        if it % 400 == 0:
            alpha = min(alpha * alpha_growth, 200.0)

        # 8) occasional random jitter to escape flat plateaus
        if stagnation > 5000:
            b += 0.02 * np.random.randn(n)
            stagnation = 0

    return best_a, best_score

    if kind == "prev" and prev_seq is not None:
        seq = prev_seq.astype(float, copy=True)
        if seq.shape[0] > n:
            seq = seq[:n]
        elif seq.shape[0] < n:
            seq = np.concatenate([seq, np.zeros(n - seq.shape[0])])
        seq = np.maximum(seq, 0.0)
        total = seq.sum()
        if total == 0.0:
            return np.full(n, 1.0 / n)
        return seq / total

    if kind == "uniform":
        return np.full(n, 1.0 / n)

    if kind == "random_dirichlet":
        # Concentration < 1 encourages sparsity.
        alpha = 0.5
        sample = rng.gamma(alpha, 1.0, size=n)
        return sample / sample.sum()

    if kind == "geometric":
        # Symmetric geometric decay from both ends.
        r = 0.98 + 0.02 * rng.random()
        left = r ** np.arange(n)
        right = left[::-1]
        seq = left + right
        seq = np.maximum(seq, 0.0)
        s = seq.sum()
        return seq / s if s > 0 else np.full(n, 1.0 / n)

    # fallback
    return np.full(n, 1.0 / n)


def _adam_smooth_optimize(init_a: np.ndarray,
                          deadline: float,
                          restart_id: int,
                          target_score: float = 1.503) -> tuple[np.ndarray, float]:
    """
    Adam optimisation on the logits of a probability vector using a
    smoothed maximum (soft‑max) of the auto‑convolution.
    Returns the best probability vector found and its score.
    """
    n = init_a.shape[0]
    conv_len = 2 * n - 1
    grad_len = 3 * n - 2
    fft_len = _next_pow2(grad_len)          # large enough for both convolutions

    # initialise logits (log‑softmax)
    eps_init = 1e-12
    b = np.log(init_a + eps_init)

    # Adam state
    m = np.zeros_like(b)
    v = np.zeros_like(b)
    beta1 = 0.9
    beta2 = 0.999
    adam_eps = 1e-8

    # smoothing temperature α (starts small, grows gradually)
    alpha = 5.0
    alpha_growth = 1.15

    best_score = float('inf')
    best_a = init_a.copy()
    stagnation = 0
    it = 0

    while time.time() < deadline:
        it += 1

        # 1) probability vector from logits
        a = _softmax_logits(b)

        # 2) convolution a*a (full) via a single FFT of length fft_len
        A_fft = np.fft.rfft(a, n=fft_len)
        conv_full = np.fft.irfft(A_fft * A_fft, n=fft_len)[:conv_len]
        max_val = conv_full.max()
        cur_score = 2.0 * n * max_val

        # keep the best solution seen so far
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            stagnation = 0
            print(f"Restart {restart_id:02d} iter {it:6d}: best = {best_score:.6f}")
        else:
            stagnation += 1

        # stop early if target already reached
        if best_score < target_score:
            break

        # 3) smooth‑max weights (soft‑max over convolution entries)
        diff = conv_full - max_val                     # ≤ 0
        w_raw = np.exp(alpha * diff)                  # max entry becomes 1
        w = w_raw / w_raw.sum()                       # Σ w = 1

        # 4) gradient of the smooth objective.
        #    grad_a = 2 * (w ∗ a)[:n]   (∗ = full convolution)
        W_fft = np.fft.rfft(w, n=fft_len)
        grad_full = np.fft.irfft(W_fft * A_fft, n=fft_len)[:grad_len]
        grad_a = 2.0 * grad_full[:n]

        # 5) chain rule through the soft‑max (logits space)
        grad_obj = 2.0 * n * grad_a                     # derivative of 2·n·smooth_max
        avg = np.dot(a, grad_obj)
        grad_b = a * (grad_obj - avg)                  # Jacobian of soft‑max

        # 6) Adam update on logits
        m = beta1 * m + (1.0 - beta1) * grad_b
        v = beta2 * v + (1.0 - beta2) * (grad_b ** 2)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        lr = 0.4 / np.sqrt(it)                         # diminishing learning‑rate
        b -= lr * m_hat / (np.sqrt(v_hat) + adam_eps)

        # 7) gradually increase the smoothing temperature
        if it % 400 == 0:
            alpha = min(alpha * alpha_growth, 200.0)

        # 8) random jitter to escape possible plateaus
        if stagnation > 5000:
            b += 0.02 * np.random.randn(n)
            stagnation = 0

    return best_a, best_score

def propose_candidate(budget_s: int = 1000) -> list:
    """
    Search for a low‑score step‑function sequence.
    The routine respects the time budget (seconds) and returns the best
    sequence (as a plain Python list of floats) found so far.
    """
    rng = np.random.default_rng(123456)          # deterministic randomness
    # small safety margin to avoid overrunning the allotted time
    deadline = time.time() + budget_s - 5.0

    # Determine problem size and possibly a previous solution.
    try:
        previous = np.asarray(height_sequence_1, dtype=float)   # type: ignore[name-defined]
        n = int(previous.shape[0])
    except Exception:
        previous = None
        n = 5146   # default length used in earlier attempts

    # Global bookkeeping
    best_overall_a = None
    best_overall_score = float('inf')
    restart_counter = 0

    # Cycle through a few different initialisation strategies.
    init_strategies = ["prev", "uniform", "geometric", "random_dirichlet"]

    while time.time() < deadline - 10.0:  # keep a few seconds for finalisation
        kind = init_strategies[restart_counter % len(init_strategies)]

        # If the requested kind needs a previous sequence that we don't have,
        # fall back to the uniform distribution.
        if kind == "prev" and previous is None:
            kind = "uniform"

        # Build a feasible starting probability vector (sum = 1).
        a0 = _initial_sequence(kind, n, rng, previous)

        # Allocate a slice of the remaining budget to this restart.
        remaining = deadline - time.time()
        this_deadline = time.time() + min(40.0, 0.5 * remaining)

        a_opt, score_opt = _adam_smooth_optimize(a0, this_deadline,
                                                 restart_counter,
                                                 target_score=1.503)

        if score_opt < best_overall_score:
            best_overall_score = score_opt
            best_overall_a = a_opt.copy()
            print(f"Global best after restart {restart_counter:02d}: {best_overall_score:.6f}")

        # Stop early if we have already beaten the target.
        if best_overall_score < 1.503:
            print(f"Target reached (score={best_overall_score:.6f}) after {restart_counter+1} restarts.")
            break

        restart_counter += 1

    # Fallback – ensure a valid sequence is always returned.
    if best_overall_a is None:
        best_overall_a = np.full(n, 1.0 / n)

    # Convert to a plain Python list of floats for the evaluator.
    return [float(v) for v in best_overall_a.tolist()]
