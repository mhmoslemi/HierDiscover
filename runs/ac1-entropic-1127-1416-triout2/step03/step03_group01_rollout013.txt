def optimize_by_frank_wolfe(n, init_f, deadline):
    f = init_f.copy()
    best_f = f.copy()
    best_val = compute_score(f) # using 2*n*max(conv)
    iteration = 0
    t = 0.2
    while time.time() < deadline - 0.5:
        iteration += 1
        # compute M
        conv_f = convolution_fft(f, f)
        M = np.max(conv_f)
        # Build matrix
        A = build_conv_matrix(f)
        # Solve LP
        c = -np.ones(n)
        b_ub = np.full(2*n-1, M)
        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=(0, None), method='highs', options={'presolve': False})
        if not res.success:
            print(f"LP failed at iter {iteration}, status: {res.message}")
            break
        g0 = res.x
        sum_g0 = np.sum(g0)
        if sum_g0 <= 0:
            break
        # Scale to sum=1
        lambda_scale = 1.0 / sum_g0
        g = lambda_scale * g0
        # combine
        f_new = (1 - t) * f + t * g
        # renormalize for numerical errors
        f_new = f_new / f_new.sum()
        # compute new objective
        val = compute_score(f_new)
        if val < best_val:
            best_val = val
            best_f = f_new.copy()
            print(f"Iter {iteration}: improved to {best_val:.6f}, M={M:.6f}, t={t}")
        f = f_new
        # optional adapt t
        if iteration % 20 == 0:
            # Reduce t slowly
            t = max(0.02, t * 0.9)
    return best_f, best_val

"""
Propose a step‑function via a Frank–Wolfe style alternating linear program.

The algorithm works on a fixed length ``n`` (default 2048) and keeps the
coefficients on the probability simplex (∑a_i = 1, a_i ≥ 0).  For a current
function ``f`` we compute the worst autocorrelation value

    M = max_k (f * f)[k]    (linear convolution)

and then solve the linear program

    maximise   ∑ g_i
    subject to 0 ≤ g_i
                (f * g)[k] ≤ M   for all k = 0,…,2n‑2

The LP finds a “partner’’ vector ``g`` that can carry as much mass as
possible without violating the current worst‑case bound ``M``.  The
solution ``g₀`` is scaled to unit sum (λ = 1/∑g₀) which reduces the
feasible bound to ``λ·M``.  Finally we move from ``f`` towards the scaled
partner by a line search over the mixing coefficient α∈[0,1] and keep the
best mixture.  This step is guaranteed to strictly decrease the objective

    ϕ(f) = 2·n·max_k (f*f)[k] .

The loop repeats while time is left (≈ 1000 s).  The best vector seen so
far is returned as a plain Python list; it is directly comparable to the
official ``evaluate_sequence`` function because the sum is normalised to 1.
If SciPy is not available the routine falls back to the previously
provided Adam optimiser.
"""

import time
from typing import List, Tuple

import numpy as np

# ----------------------------------------------------------------------
# Optional SciPy imports – used for the linear program
# ----------------------------------------------------------------------
try:                                 # pragma: no‑cover
    from scipy.optimize import linprog

except Exception:                    # pragma: no‑cover
    SCIPY_AVAILABLE = False

def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """
    Linear convolution of two 1‑D real vectors using the FFT.
    Returned length is len(a)+len(b)-1.
    """
    m = len(a) + len(b) - 1
    n_fft = next_pow2(m)
    A = np.fft.rfft(a, n=n_fft)
    B = np.fft.rfft(b, n=n_fft)
    C = A * B
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


def compute_score(a: np.ndarray) -> float:
    """
    Exact official objective for a simplex‑normalised vector:
        2·n·max(a*a)  .
    """
    n = len(a)
    conv = convolution_fft(a, a)
    return 2.0 * n * float(np.max(conv))


def arcsine_initial(n: int) -> np.ndarray:
    """Classic arcsine shape, normalised to unit sum."""
    i = np.arange(n, dtype=float)
    eps = 0.25
    vals = 1.0 / np.sqrt((i + 0.5) * (n - 0.5 - i) + eps)
    vals /= vals.sum()
    return vals


def uniform_initial(n: int) -> np.ndarray:
    """Uniform distribution on the simplex."""
    return np.full(n, 1.0 / n)


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection onto the probability simplex {x ≥ 0, Σx = 1}.
    Duchi et al., 2008.
    """
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / v.size)

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho = np.where(u - cssv / np.arange(1, v.size + 1) > 0)[0]
    if len(rho) == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def build_conv_matrix(f: np.ndarray) -> np.ndarray:
    """
    Build the dense matrix A (shape (2n‑1, n)) that satisfies
        (f * g)[k] = (A @ g)[k] .
    Column ``j`` contains the vector ``f`` shifted down by ``j`` rows,
    i.e. A[k, j] = f[k‑j] when 0 ≤ k‑j < n, otherwise 0.
    The construction uses a Toeplitz matrix.
    """
    n = f.shape[0]
    # First column: f followed by (n‑1) zeros.
    c = np.concatenate([f, np.zeros(n - 1, dtype=f.dtype)])
    # First row: f[0] followed by zeros.
    r = np.zeros(n, dtype=f.dtype)
    r[0] = f[0]
    A = scipy.linalg.toeplitz(c, r)   # shape (2n‑1, n)
    return A

# Frank–Wolfe linear‑programming optimiser
# ----------------------------------------------------------------------


def frank_wolfe_optimize(init_f: np.ndarray,
                         deadline: float,
                         max_iter: int = 500) -> Tuple[np.ndarray, float]:
    """
    Minimises ϕ(f)=2·n·max(f*f) on the simplex by repeatedly solving
    the linear programme described in the module docstring.
    """
    f = init_f.copy()                     # always kept normalised
    n = len(f)
    best_f = f.copy()
    best_val = compute_score(f)

    # candidates for the mixing coefficient (including 0 → keep current)
    alphas = np.linspace(0.0, 1.0, 11)

    it = 0
    while time.time() < deadline - 0.5 and it < max_iter:
        it += 1

        # current worst autocorrelation
        conv_f = convolution_fft(f, f)
        M = float(np.max(conv_f))

        # ----- build the LP ----- #
        A = build_conv_matrix(f)          # dense (2n‑1) × n matrix
        c = -np.ones(n)                   # maximise sum(g) <=> minimise -sum(g)
        b_ub = np.full(2 * n - 1, M)      # all constraints share the same bound

        # solve the LP
        res = linprog(c, A_ub=A, b_ub=b_ub,
                      bounds=(0, None),
                      method='highs',
                      options={'presolve': False, 'time_limit': max(deadline - time.time(), 0.0)})
        if not res.success:
            # In the very rare case of failure we abort the loop.
            print(f"  [iter {it}] LP failed: {res.message}")
            break

        g0 = res.x
        sum_g0 = float(np.sum(g0))
        if sum_g0 <= 0.0:
            break

        # Scale to keep the simplex constraint.
        scale = 1.0 / sum_g0
        g = scale * g0                     # Σg = 1

        # ----- line‑search over the mixing coefficient ----- #
        best_local_f = f
        best_local_val = best_val
        best_alpha = 0.0

        for α in alphas[1:]:               # skip α=0 (already handled)
            cand = (1.0 - α) * f + α * g
            # Numerical drift may break the simplex; re‑project.
            cand = cand / np.sum(cand)
            val = compute_score(cand)
            if val < best_local_val:
                best_local_val = val
                best_local_f = cand
                best_alpha = α

        # update the current iterate
        f = best_local_f
        cur_val = best_local_val

        if cur_val < best_val:
            best_val = cur_val
            best_f = f.copy()
            print(f"  [iter {it:3d}] improved → score={best_val:.6f}  α={best_alpha:.2f}")

        # occasional random jitter to escape tiny plateaus
        if it % 30 == 0:
            jitter = (np.random.rand(n) - 0.5) * 1e-5
            f = project_to_simplex(f + jitter)

    return best_f, best_val

def smooth_gradient(a: np.ndarray, lam: float) -> Tuple[np.ndarray, float]:
    """Gradient of the soft‑max surrogate (kept for fallback)."""
    n = len(a)
    conv = convolution_fft(a, a)               # length 2n‑1
    max_c = np.max(conv)

    # stable soft‑max
    exp_term = np.exp(lam * (conv - max_c))
    sum_exp = np.sum(exp_term)
    w = exp_term / sum_exp                     # weights, sum to 1
    # smooth approximation of the max
    B = max_c + np.log(sum_exp) / lam

    # weighted sum of sub‑gradients:
    w_rev = w[::-1]
    conv_rev = convolution_fft(a, w_rev)       # length 3n‑2
    L = 2 * n - 1
    weighted_grad = 2.0 * conv_rev[L - 1 - np.arange(n)]

    s = np.sum(a)
    term1 = (2.0 * n / (s ** 2)) * weighted_grad
    term2 = -(4.0 * n * B / (s ** 3)) * np.ones_like(a)
    grad = term1 + term2
    return grad, B


def adam_optimize(initial_a: np.ndarray,
                  time_deadline: float,
                  max_iter: int = 150_000,
                  lr0: float = 0.15,
                  lam_start: float = 5.0,
                  lam_end: float = 30.0,
                  noise_every: int = 1000,
                  noise_scale: float = 3e-5) -> Tuple[np.ndarray, float]:
    """Simple Adam optimiser on the smooth surrogate (fallback)."""
    a = project_to_simplex(initial_a.astype(float).copy())
    best_a = a.copy()
    best_sc = compute_score(a)

    beta1, beta2 = 0.9, 0.999
    eps = 1e-8
    m = np.zeros_like(a)
    v = np.zeros_like(a)

    n = len(a)

    for it in range(1, max_iter + 1):
        if time.time() >= time_deadline:
            break

        lam = lam_start + (lam_end - lam_start) * min(1.0, it / max_iter)

        grad, _ = smooth_gradient(a, lam)

        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)

        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        step = lr0 * m_hat / (np.sqrt(v_hat) + eps)
        a_new = a - step

        if noise_every and (it % noise_every == 0):
            a_new += np.random.randn(n) * noise_scale

        a_new = project_to_simplex(a_new)

        cur_sc = compute_score(a_new)
        if cur_sc < best_sc:
            best_sc = cur_sc
            best_a = a_new.copy()

        a = a_new

        if it % 5000 == 0:
            print(f"    Adam iter={it:,d}  cur={cur_sc:.6f}  best={best_sc:.6f}")

    return best_a, best_sc

    start_time = time.time()
    deadline = start_time + budget_s - 2.0   # safety margin

    # ------------------------------------------------------------------
    # 0.  Try to reuse a previously found construction, if present.
    # ------------------------------------------------------------------
    best_seq: List[float] = []
    best_val: float = float("inf")
    try:
        fallback = [float(x) for x in height_sequence_1]   # type: ignore
        val = evaluate_sequence(fallback)
        print(f"Fallback (height_sequence_1) score = {val:.6f}")
        if val < best_val:
            best_val = val
            best_seq = fallback
    except Exception:
        pass

    # ------------------------------------------------------------------
    # 1.  Choose a length.  We stick with the length of the fallback if it
    #     exists, otherwise we pick a moderate size (2048) that works well
    #     with the LP based optimiser.
    # ------------------------------------------------------------------
    n = len(best_seq) if best_seq else 2048
    print(f"\nOptimising sequences of length n = {n}")

    # ------------------------------------------------------------------
    # 2.  Initial candidates (arcsine, uniform, possibly the fallback)
    # ------------------------------------------------------------------
    candidates = [arcsine_initial(n), uniform_initial(n)]
    if best_seq:
        candidates.append(np.array(best_seq, dtype=float) / np.sum(best_seq))
    # pick the best among the cheap candidates
    best_init = None
    best_score = float("inf")
    for cand in candidates:
        sc = compute_score(cand)
        if sc < best_score:
            best_score = sc
            best_init = cand.copy()
    if best_init is None:                # pragma: no‑cover  (should never happen)
        best_init = arcsine_initial(n)

    # ------------------------------------------------------------------
    # 3.  Main optimisation loop
    # ------------------------------------------------------------------
    if SCIPY_AVAILABLE:
        # Use the Frank–Wolfe LP scheme
        print("Running Frank–Wolfe LP optimiser (SciPy available)…")
        final_f, final_val = frank_wolfe_optimize(best_init, deadline)
        final_val = compute_score(final_f)   # recompute for safety
        print(f"LP optimiser finished – best score = {final_val:.6f}")
    else:
        # SciPy not present – fall back to the Adam optimiser from earlier
        print("SciPy not available – falling back to Adam optimiser.")
        final_f, final_val = adam_optimize(best_init, deadline)

    # ------------------------------------------------------------------
    # 4.  Final verification with the official evaluator
    # ------------------------------------------------------------------
    final_seq = final_f.tolist()
    try:
        official_val = evaluate_sequence(final_seq)
        print(f"Official evaluator reports score = {official_val:.6f}")
        if official_val < best_val:
            best_val = official_val
            best_seq = final_seq
    except Exception as e:               # pragma: no‑cover
        print(f"Error during official evaluation: {e}")

    # ------------------------------------------------------------------
    # 5.  Guard – guarantee a non‑empty return value
    # ------------------------------------------------------------------
    if not best_seq:
        # Very crude safe fallback (uniform 256‑vector)
        best_seq = [1.0 / 256.0] * 256
        best_val = evaluate_sequence(best_seq)

    print(f"\nFinished search. Best reported score = {best_val:.6f}")
    return best_seq
