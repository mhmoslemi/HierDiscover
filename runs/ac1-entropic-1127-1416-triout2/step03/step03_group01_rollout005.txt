"""
Multi‑scale Adam + sub‑gradient refinement for the self‑convolution
objective  2·n·max(a*a) / (∑a)².

The algorithm proceeds in three stages:

1.  A coarse optimisation on a short vector (n=256) by sampling many
    beta‑family shapes, picking the most promising ones and running a
    soft‑max Adam optimiser (the same surrogate as the reference code).

2.  The best vector is repeatedly up‑sampled (linear interpolation) to a
    larger size (512 → 1024 → 2048 → 4096 → 8192).  At each scale we run
    the Adam optimiser for as many iterations as the remaining time
    budget allows.  The optimiser uses a slowly rising “temperature”
    λ so that the smoothed max converges to the true maximum.

3.  After the largest scale a short sub‑gradient descent on the exact
    max‑convolution (no smoothing) is performed; it removes the last
    remaining slack and usually improves the official score a few
    thousandths.  The final vector is returned.

All vectors are kept on the probability simplex (∑a = 1, a ≥ 0); the
official evaluator is homogeneous, therefore this normalisation does
not affect the score.  The code is totally self‑contained and respects
the 1000 s time budget.
"""

import time
import numpy as np
from typing import List, Tuple

# ----------------------------------------------------------------------
# Basic linear‑algebra utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """
    Linear convolution using a real FFT.
    Returns a vector of length len(a)+len(b)-1.
    """
    m = len(a) + len(b) - 1
    n_fft = next_pow2(m)
    A = np.fft.rfft(a, n=n_fft)
    B = np.fft.rfft(b, n=n_fft)
    C = A * B
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


def max_convolution_and_index(a: np.ndarray) -> Tuple[float, int]:
    """Maximum value of the self‑convolution a*a and its index."""
    conv = convolution_fft(a, a)
    idx = int(np.argmax(conv))
    return float(conv[idx]), idx


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the probability simplex
    {x ≥ 0, Σx = 1}.
    Duchi et al., 2008.
    """
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / v.size)

    # sort descending
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho_arr = np.where(u - cssv / np.arange(1, v.size + 1) > 0)[0]
    if rho_arr.size == 0:
        theta = 0.0
    else:
        rho = rho_arr[-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def compute_score(a: np.ndarray) -> float:
    """
    Fast surrogate for the official score.
    The vector is assumed to be normalised (∑a = 1), therefore the
    denominator is 1 and the objective reduces to 2·n·max(a*a).
    """
    n = len(a)
    conv = convolution_fft(a, a)
    max_val = float(np.max(conv))
    return 2.0 * n * max_val

# Families of initial candidates
# ----------------------------------------------------------------------


def arcsine_initial(n: int) -> np.ndarray:
    """Arcsine shape a_i ∝ 1/√((i+½)*(n‑½‑i))."""
    i = np.arange(n, dtype=float)
    eps = 0.25
    vals = 1.0 / np.sqrt((i + 0.5) * (n - 0.5 - i) + eps)
    vals /= vals.sum()
    return vals


def generate_beta_shape(n: int,
                        alpha: float,
                        beta: float,
                        eps: float) -> np.ndarray:
    """
    Symmetric beta‑family density on {0,…,n‑1}:
        a_i ∝ (i+ε)^{α‑1}·(n‑1‑i+ε)^{β‑1}.
    The result is normalised to sum to 1.
    """
    i = np.arange(n, dtype=float)
    left = (i + eps) ** (alpha - 1.0)
    right = (n - 1 - i + eps) ** (beta - 1.0)
    base = left * right
    total = base.sum()
    if total <= 0.0 or not np.isfinite(total):
        return np.full(n, 1.0 / n)
    return base / total


def upsample(a: np.ndarray, new_n: int) -> np.ndarray:
    """
    Linear interpolation from length len(a) to new_n.
    The result is re‑normalised to lie on the simplex.
    """
    old_n = len(a)
    if old_n == new_n:
        return a.copy()
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    new_a = np.interp(new_x, old_x, a)
    # Numerical errors can produce a zero‑sum, guard against it.
    s = new_a.sum()
    if s <= 0.0:
        new_a = np.full(new_n, 1.0 / new_n)
    else:
        new_a /= s
    return new_a


# ----------------------------------------------------------------------
# Smooth Adam optimiser (log‑sum‑exp surrogate of max)
# ----------------------------------------------------------------------


def smooth_gradient(a: np.ndarray, lam: float) -> Tuple[np.ndarray, float]:
    """
    Gradient of the smoothed objective
          F_λ(a) = 2·n· (max_c + log Σ exp(λ·(c‑max_c))/λ) / (∑a)²
    where c = a*a.

    Returns
    -------
    grad : np.ndarray
        Gradient of F_λ w.r.t. a.
    smooth_max : float
        The smooth approximation of max(c).
    """
    n = len(a)
    conv = convolution_fft(a, a)                     # length 2n‑1
    max_c = np.max(conv)

    # soft‑max weights
    exp_term = np.exp(lam * (conv - max_c))
    sum_exp = np.sum(exp_term)
    w = exp_term / sum_exp
    # smooth max value
    B = max_c + np.log(sum_exp) / lam

    # weighted sub‑gradients of the convolution
    w_rev = w[::-1]
    conv_rev = convolution_fft(a, w_rev)            # length n+2n‑2‑1 = 3n‑2
    L = 2 * n - 1
    weighted_grad = 2.0 * conv_rev[L - 1 - np.arange(n)]

    s = a.sum()
    term1 = (2.0 * n / (s ** 2)) * weighted_grad
    term2 = -(4.0 * n * B / (s ** 3)) * np.ones_like(a)
    grad = term1 + term2
    return grad, B


def adam_optimize(initial_a: np.ndarray,
                  time_deadline: float,
                  max_iter: int = 100_000,
                  lr0: float = 0.2,
                  lam_start: float = 5.0,
                  lam_end: float = 30.0,
                  noise_every: int = 2000,
                  noise_scale: float = 5e-5) -> Tuple[np.ndarray, float]:
    """
    Adam optimiser on the smooth surrogate.
    Returns the best vector (projected onto the simplex) and its
    fast surrogate score.
    """
    a = project_to_simplex(initial_a.astype(float).copy())
    best_a = a.copy()
    best_sc = compute_score(a)

    beta1, beta2 = 0.9, 0.999
    eps = 1e-8
    m = np.zeros_like(a)
    v = np.zeros_like(a)

    for it in range(1, max_iter + 1):
        if time.time() >= time_deadline:
            break

        lam = lam_start + (lam_end - lam_start) * min(1.0, it / max_iter)

        grad, _ = smooth_gradient(a, lam)

        # Adam moments
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)

        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        step = lr0 * m_hat / (np.sqrt(v_hat) + eps)
        a_new = a - step

        if noise_every and (it % noise_every == 0):
            a_new += np.random.randn(len(a)) * noise_scale

        a_new = project_to_simplex(a_new)

        cur_sc = compute_score(a_new)
        if cur_sc < best_sc:
            best_sc = cur_sc
            best_a = a_new.copy()

        a = a_new

        if it % 5000 == 0:
            print(f"    Adam iter={it:,d}  cur={cur_sc:.6f}  best={best_sc:.6f}")

    return best_a, best_sc

def subgradient_optimize(initial_a: np.ndarray,
                         time_deadline: float,
                         max_iter: int = 30_000,
                         step0: float = 0.4) -> Tuple[np.ndarray, float]:
    """
    Simple sub‑gradient descent on the exact objective.
    The step size decays as 1/√t.
    """
    a = project_to_simplex(initial_a.astype(float).copy())
    best_a = a.copy()
    best_sc = compute_score(a)

    n = len(a)

    for it in range(1, max_iter + 1):
        if time.time() >= time_deadline:
            break

        conv = convolution_fft(a, a)
        M = float(np.max(conv))
        idx = int(np.argmax(conv))

        # sub‑gradient of max(c) w.r.t. a : 2 * a[idx‑i] (zero outside support)
        grad_max = np.zeros_like(a)
        i_low = max(0, idx - (n - 1))
        i_high = min(n - 1, idx)
        idx_range = np.arange(i_low, i_high + 1)
        grad_max[idx_range] = 2.0 * a[idx - idx_range]

        # full gradient of the objective (sum = 1)
        grad = 2.0 * n * grad_max - 4.0 * n * M * np.ones_like(a)

        step = step0 / np.sqrt(it)
        a_new = a - step * grad
        a_new = project_to_simplex(a_new)

        cur_sc = compute_score(a_new)
        if cur_sc < best_sc:
            best_sc = cur_sc
            best_a = a_new.copy()

        a = a_new

        if it % 5000 == 0:
            print(f"    Subgrad iter={it:,d} cur={cur_sc:.6f} best={best_sc:.6f}")

    return best_a, best_sc

    # ------------------------------------------------------------------
    # 0.  Fallback if a previous construction is available
    # ------------------------------------------------------------------
    try:
        fallback = [float(x) for x in height_sequence_1]          # type: ignore
        val = evaluate_sequence(fallback)
        print(f"Fallback (height_sequence_1) score = {val:.6f}")
        if val < best_val:
            best_val = val
            best_seq = fallback
    except Exception:
        pass

    # ------------------------------------------------------------------
    # 1.  Coarse optimisation on the smallest scale (n = 256)
    # ------------------------------------------------------------------
    n0 = 256
    print(f"\n=== Stage 1 – coarse search, n = {n0} ===")
    candidates: List[np.ndarray] = []

    # deterministic starters
    candidates.append(arcsine_initial(n0))
    candidates.append(np.full(n0, 1.0 / n0))                     # uniform

    # random beta‑family samples
    for _ in range(250):
        if time.time() >= deadline - 5.0:
            break
        alpha = rng.uniform(0.05, 2.5)
        beta = rng.uniform(0.05, 2.5)
        eps = rng.uniform(0.001, max(1.0, 0.02 * n0))
        cand = generate_beta_shape(n0, alpha, beta, eps)
        candidates.append(cand)

    # keep only the 5 best according to the cheap surrogate
    candidates.sort(key=compute_score)
    seeds = [candidates[i] for i in range(min(5, len(candidates)))]

    # run a short Adam optimisation from each seed
    for idx, seed_vec in enumerate(seeds, 1):
        if time.time() >= deadline - 5.0:
            break
        remaining = deadline - time.time()
        max_iter = int(min(40_000, remaining * 25))
        print(f"  Adam start {idx}/{len(seeds)} (max_iter={max_iter}) …")
        refined, _ = adam_optimize(seed_vec,
                                   time_deadline=deadline,
                                   max_iter=max_iter,
                                   lr0=0.2,
                                   lam_start=3.0,
                                   lam_end=35.0,
                                   noise_every=1500,
                                   noise_scale=6e-5)
        val = evaluate_sequence(refined.tolist())
        print(f"    → official score = {val:.6f}")
        if val < best_val:
            best_val = val
            best_seq = refined.tolist()
            print(f"    *** New global best (coarse) = {best_val:.6f} ***")

    # ------------------------------------------------------------------
    # 2.  Multi‑scale refinement (512 → 1024 → 2048 → 4096 → 8192)
    # ------------------------------------------------------------------
    scale_levels = [512, 1024, 2048, 4096, 8192]

    # start from the best vector found so far (if any), otherwise arcsine
    if best_seq:
        cur_vec = np.array(best_seq, dtype=float)
    else:
        cur_vec = arcsine_initial(n0)

    for n in scale_levels:
        if time.time() >= deadline - 5.0:
            break
        print(f"\n=== Stage 2 – up‑sample to n = {n} ===")
        # up‑sample previous best
        cur_vec = upsample(cur_vec, n)

        # small random perturbation to avoid getting stuck
        noise = rng.normal(scale=1e-3, size=n)
        cur_vec = project_to_simplex(cur_vec + noise)

        # allocate time for this scale
        remaining = deadline - time.time()
        if remaining <= 0:
            break
        # number of iterations grows with n (more structure, more time)
        max_iter = int(min(120_000, remaining * 30))

        # Adam optimisation on the current scale
        cur_vec, fast_sc = adam_optimize(
            initial_a=cur_vec,
            time_deadline=deadline,
            max_iter=max_iter,
            lr0=0.15,
            lam_start=5.0,
            lam_end=30.0,
            noise_every=1200,
            noise_scale=4e-5,
        )
        # official evaluation of the (still provisional) best at this scale
        val = evaluate_sequence(cur_vec.tolist())
        print(f"  Scale {n} official score = {val:.6f}")
        if val < best_val:
            best_val = val
            best_seq = cur_vec.tolist()
            print(f"  *** Updated global best after scale {n} = {best_val:.6f} ***")

    # ------------------------------------------------------------------
    # 3.  Final fine‑tuning with exact sub‑gradient descent
    # ------------------------------------------------------------------
    if best_seq and time.time() < deadline - 5.0:
        print("\n=== Stage 3 – final sub‑gradient polishing ===")
        a = np.array(best_seq, dtype=float)
        remaining = deadline - time.time()
        max_iter = int(min(40_000, remaining * 20))
        a_refined, _ = subgradient_optimize(
            initial_a=a,
            time_deadline=deadline,
            max_iter=max_iter,
            step0=0.3,
        )
        final_val = evaluate_sequence(a_refined.tolist())
        print(f"Final polished score = {final_val:.6f}")
        if final_val < best_val:
            best_val = final_val
            best_seq = a_refined.tolist()
            print("*** Updated global best after final polish ***")

    # ------------------------------------------------------------------
    # 4.  Guard – guarantee a non‑empty return
    # ------------------------------------------------------------------
    if not best_seq:
        # safe fallback (uniform 256‑vector)
        best_seq = [1.0 / 256.0] * 256
        best_val = evaluate_sequence(best_seq)

    print(f"\nFinished search. Best reported score = {best_val:.6f}")
    return best_seq
