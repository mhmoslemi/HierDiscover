"""
Propose a candidate step‑function height vector that minimises
evaluate_sequence().

Algorithm
---------
1.  Initialise from the previously discovered best sequence (if any) or
    from a random Beta distribution.  The vector is kept on the
    probability simplex (non‑negative, sum == 1).

2.  Continuous optimisation using a *smoothed* max objective.
    For a vector a we write b = a∗a (linear convolution, length 2n‑1) and
    minimise
        f_λ(a) = (1/λ)·log Σ_k exp(λ·b_k)          (λ→∞ → max(b))
    The gradient of f_λ is a weighted sum of the gradients of the
    individual convolution entries.  Using the soft‑max weights w,
    the gradient can be written as a convolution of w with the reversed
    vector a, which we compute efficiently with FFTs.
    Updates are performed by *exponentiated gradient descent*,
    i.e. a ← a·exp(−η·∇f) followed by a renormalisation onto the
    simplex.  The step size η decays as 1/√t, while λ is increased
    slowly (continuation method) so that the optimisation first moves
    in a smooth landscape and later focuses on the true max‑objective.

3.  When the time budget for the smooth phase is exhausted a short
    simulated‑annealing (SA) phase fine‑tunes the best vector found so
    far.  SA performs inexpensive “mass‑transfer” moves that keep the
    sum exactly 1 and accepts moves with a Metropolis rule that slowly
    cools.

The whole procedure runs inside the given time budget (default 1000 s)
and returns the best sequence encountered.  All numbers are non‑negative
and the caller’s ``evaluate_sequence`` will apply the required
normalisation internally.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Return the smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def full_convolution(a: np.ndarray, nfft: int) -> np.ndarray:
    """
    Linear convolution a∗a using FFT of length nfft (≥ 2·len(a)−1).
    Returns the full vector of length 2·len(a)−1.
    """
    f = np.fft.rfft(a, n=nfft)
    conv = np.fft.irfft(f * f, n=nfft)
    return conv[: 2 * a.shape[0] - 1]


def softmax_weights(conv: np.ndarray, lam: float) -> np.ndarray:
    """
    Compute soft‑max over the convolution entries.
    The result sums to 1 and is numerically stable.
    """
    mx = conv.max()
    exps = np.exp(lam * (conv - mx))
    # avoid division by zero if lam is zero
    total = exps.sum()
    if total == 0.0:
        # fall back to uniform weights
        return np.full_like(conv, 1.0 / conv.size)
    return exps / total


def gradient_of_softmax(a: np.ndarray, weights: np.ndarray,
                        L: int) -> np.ndarray:
    """
    Given a (length n) and soft‑max weights w (length 2n‑1),
    compute ∇f = 2·(w * a_rev) restricted to the first n entries.
    The convolution is performed via FFT of length L (≥ 3·n‑2).
    """
    n = a.shape[0]
    a_rev = a[::-1]                     # reversed a
    # FFT of w and a_rev
    fw = np.fft.rfft(weights, n=L)
    fa = np.fft.rfft(a_rev, n=L)
    conv = np.fft.irfft(fw * fa, n=L)   # linear convolution, length L
    start = n - 1                       # position where length‑n slice begins
    grad = 2.0 * conv[start:start + n]
    return grad


def simplex_normalise(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection onto the probability simplex.
    Implements the algorithm of Duchi et al. (2008).
    """
    v = np.asarray(v, dtype=float)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1] - 1
    theta = cssv[cond][-1] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def compute_score(a: np.ndarray) -> float:
    """
    Exact value of evaluate_sequence for a vector with sum(a)==1.
    (score = 2·n·max(a∗a))
    """
    n = a.shape[0]
    conv = full_convolution(a, nfft=next_pow2(2 * n - 1))
    return 2.0 * n * conv.max()


# ----------------------------------------------------------------------
# Main candidate generator
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 0, budget_s: int = 1000) -> list[float]:
    """
    Search for a non‑negative height sequence that minimises
    evaluate_sequence().  The routine spends most of the time on a
    smoothed‑max gradient descent and finishes with a brief simulated‑
    annealing refinement.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5          # safety margin

    # ------------------------------------------------------------------
    # 0. Problem size and initial vector
    # ------------------------------------------------------------------
    try:
        init = np.array(height_sequence_1, dtype=float)   # type: ignore[name-defined]
        n = init.shape[0]
        print(f"Using existing height_sequence_1 (length {n}).")
    except NameError:
        # no previous solution – start from a random Beta shape
        n = 6000
        alpha, beta = np.random.uniform(0.5, 5.0, size=2)
        x = (np.arange(n) + 0.5) / n
        logw = (alpha - 1.0) * np.log(x) + (beta - 1.0) * np.log(1.0 - x)
        init = np.exp(np.clip(logw, -700, 700))
        init = np.maximum(init, 0.0)
        print(f"No previous sequence – using a Beta({alpha:.2f},{beta:.2f}) start (length {n}).")

    # Normalise to a probability vector (sum == 1)
    a = init / init.sum()
    a = simplex_normalise(a)          # just to be safe

    best_a = a.copy()
    best_score = compute_score(best_a)
    print(f"Initial score = {best_score:.6f}")

    # Pre‑allocate FFT length for gradient
    L_grad = next_pow2(3 * n - 2)      # enough for w (2n‑1) * a_rev (n)

    # ------------------------------------------------------------------
    # 1. Smoothed‑max gradient descent (exponentiated updates)
    # ------------------------------------------------------------------
    it = 0
    lam = 0.1                         # start smoothing parameter
    lam_max = 12.0                    # final λ (approximates true max)
    eta0 = 0.5                        # base step size for EG
    # keep enough time for the SA phase (≈30 s)
    while time.time() < deadline - 30.0:
        # --- compute the full convolution a∗a ---
        conv = full_convolution(a, nfft=next_pow2(2 * n - 1))

        # current objective value (exact)
        cur_score = 2.0 * n * conv.max()
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"Iter {it:6d}: new best {best_score:.6f}")

        # --- soft‑max over convolution entries ---
        w = softmax_weights(conv, lam)

        # --- gradient of the smoothed objective ---
        g = gradient_of_softmax(a, w, L_grad)

        # --- exponentiated‑gradient update (keeps positivity) ---
        eta = eta0 / math.sqrt(it + 1.0)
        a = a * np.exp(-eta * g)

        # --- renormalise onto the simplex ---
        s = a.sum()
        if s <= 0.0 or not np.isfinite(s):
            # reinitialise if we ever hit a degenerate point
            a = np.full(n, 1.0 / n)
        else:
            a = a / s

        # --- continuation for λ (slowly increase) ---
        lam = min(lam_max, lam * 1.001 + 0.0005)

        it += 1

    print(f"Gradient phase finished after {it} iterations; best score {best_score:.6f}")

    # ------------------------------------------------------------------
    # 2. Simulated‑annealing fine‑tuning (≈30 s)
    # ------------------------------------------------------------------
    cur_a = best_a.copy()
    cur_score = compute_score(cur_a)

    # temperature schedule
    T_start = cur_score * 0.02
    T_end = cur_score * 1e-6
    it_sa = 0
    while time.time() < deadline:
        # exponential cooling
        T = max(T_end, T_start * (0.9985 ** it_sa))

        # --- propose a small mass‑transfer move ---
        src = np.random.choice(n, p=cur_a)          # source weighted by mass
        if cur_a[src] <= 0.0:
            continue
        tgt = np.random.randint(n - 1)
        if tgt >= src:
            tgt += 1

        delta = cur_a[src] * np.random.rand()       # fraction of src mass
        cand = cur_a.copy()
        cand[src] -= delta
        cand[tgt] += delta

        # evaluate candidate (fast via FFT)
        conv_c = full_convolution(cand, nfft=next_pow2(2 * n - 1))
        cand_score = 2.0 * n * conv_c.max()

        # Metropolis acceptance
        if cand_score < cur_score:
            accept = True
        else:
            accept = np.random.rand() < math.exp(-(cand_score - cur_score) / max(T, 1e-14))

        if accept:
            cur_a = cand
            cur_score = cand_score
            if cand_score < best_score:
                best_score = cand_score
                best_a = cand.copy()
                print(f"  SA iter {it_sa:7d}: new best {best_score:.6f}")

        # occasional perturbation to avoid stagnation
        if it_sa % 80000 == 0 and it_sa > 0:
            # replace current point with a fresh random Beta draw
            a_tmp = beta_sequence(np.random.uniform(0.5, 4.0),
                                  np.random.uniform(0.5, 4.0), n)
            cur_a = a_tmp / a_tmp.sum()
            cur_score = compute_score(cur_a)

        it_sa += 1

    print(f"Simulated‑annealing finished after {it_sa} iterations; final best {best_score:.6f}")

    # ------------------------------------------------------------------
    # 3. Return the best sequence (will be re‑normalised by the judge)
    # ------------------------------------------------------------------
    best_a = best_a / best_a.sum()
    return [float(v) for v in best_a.tolist()]

# ----------------------------------------------------------------------
# Helper used when the code falls back to a Beta start (not required
# by the optimiser but kept for completeness).
# ----------------------------------------------------------------------
def beta_sequence(alpha: float, beta: float, n: int) -> np.ndarray:
    """
    Discretised Beta(α,β) density on n points, normalised to sum to 1.
    Mid‑point rule avoids the singular endpoints.
    """
    x = (np.arange(n) + 0.5) / n
    logw = (alpha - 1.0) * np.log(x) + (beta - 1.0) * np.log(1.0 - x)
    logw = np.clip(logw, -700.0, 700.0)          # avoid overflow
    w = np.exp(logw)
    if not np.isfinite(w.sum()) or w.sum() == 0.0:
        w = np.ones_like(w)
    return w / w.sum()
