"""
Proposes a step‑function (non‑negative list) that minimizes the
`evaluate_sequence` metric.

Algorithm
---------
1. **Seed generation** – A wide variety of analytically‑defined shapes
   (uniform, linear ramp, exponential, power‑law, cosine, triangular,
   discretised Beta‑distributions) and random Dirichlet vectors are
   created and normalised to sum 1.  The best few seeds are kept.

2. **Local optimisation** – Starting from the best seed we repeat the
   following until the time budget expires:

   a) *Linear‑programming step* – The convolution `c = a*a` is computed.
      All indices whose value is within a tiny relative tolerance of the
      current maximum form the *active set*.  For each active index `k`
      the first‑order expansion

          c_k(a+Δ) ≈ c_k(a) + ∇c_k(a)·Δ

      is used, where ∇c_k(a)_i = 2·a_{k‑i} (zero if the index is out of
      range).  An LP is built:

          maximise   t
          subject to ∇c_k·Δ + t ≤ max_c – c_k   (for every active k)
                     Σ Δ_i = 0                (preserve the sum)
                     Δ_i ≥ –a_i               (keep non‑negativity)
                     t   ≥ 0

      The LP variables are the displacement `Δ` and the guaranteed
      improvement `t`.  If the LP succeeds and the *true* objective after
      applying the step is lower, the step is accepted.

      iterations a random perturbation (Gaussian noise proportional to the
      current vector) or a convex combination of the current best seeds is
      applied.

def simplex_projection(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of a vector onto the probability simplex."""
    v = np.asarray(v, dtype=float)
    n = v.size
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1] - 1
    theta = cssv[cond][-1] / (rho + 1)
    return np.maximum(v - theta, 0.0)


def compute_objective(a: np.ndarray) -> float:
    """Return 2·n·max(a∗a) assuming Σ a = 1."""
    n = a.shape[0]
    fft = np.fft.rfft(a, n=2 * n - 1)
    conv = np.fft.irfft(fft * fft, n=2 * n - 1)
    return 2.0 * n * conv.max()


def conv_and_max(a: np.ndarray):
    """Return (full convolution array, its maximum)."""
    n = a.shape[0]
    fft = np.fft.rfft(a, n=2 * n - 1)
    conv = np.fft.irfft(fft * fft, n=2 * n - 1)
    return conv, conv.max()


def subgradient_active(a: np.ndarray, active: np.ndarray) -> np.ndarray:
    """Averaged sub‑gradient of the maximum convolution entry."""
    n = a.shape[0]
    g = np.zeros_like(a)
    for k in active:
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        idx = np.arange(lo, hi + 1, dtype=int)
        partner = k - idx
        g[idx] += 2.0 * a[partner]
    if active.size > 0:
        g /= float(active.size)
    return g

def beta_sequence(alpha: float, beta: float, n: int) -> np.ndarray:
    """Discretised Beta(α,β) density on the mid‑points, normalised to sum 1."""
    x = (np.arange(n) + 0.5) / n
    # work in log‑space to avoid overflow/underflow
    logw = (alpha - 1.0) * np.log(x) + (beta - 1.0) * np.log(1.0 - x)
    logw = np.clip(logw, -700.0, 700.0)
    w = np.exp(logw)
    if not np.isfinite(w).all() or w.sum() == 0.0:
        w = np.ones_like(w)
    w /= w.sum()
    return w


def exponential_shape(lam: float, n: int) -> np.ndarray:
    i = np.arange(n, dtype=float)
    w = np.exp(-lam * i)
    w /= w.sum()
    return w


def polynomial_shape(p: float, n: int) -> np.ndarray:
    i = np.arange(n, dtype=float)
    frac = i / (n - 1)
    w = 1.0 - np.power(frac, p)
    w = np.clip(w, 0.0, None)
    w /= w.sum()
    return w


def cosine_shape(n: int) -> np.ndarray:
    """a_i ∝ 1 - cos(π·i/(n‑1))  (smooth hump)."""
    i = np.arange(n, dtype=float)
    w = 1.0 - np.cos(math.pi * i / (n - 1))
    w /= w.sum()
    return w


def triangular_shape(n: int) -> np.ndarray:
    """a_i ∝ (i+1)*(n-i) – a symmetric triangular hump."""
    i = np.arange(n, dtype=float)
    w = (i + 1) * (n - i)
    w /= w.sum()
    return w

    # 6. Initialise optimisation from the best seed found so far
    # --------------------------------------------------------------
    if best_seq is None:
        best_seq = np.full(n, 1.0 / n)
        best_score = compute_objective(best_seq)

    a = best_seq.copy()
    prev_a = a.copy()
    eta = 0.90          # initial step size factor
    mu = 0.55           # momentum coefficient
    iteration = 0
    stagnation = 0
    max_iter = 500000   # safety cap

    # Helper for LP construction (only used when scipy is available)
    def attempt_lp_step(a_vec: np.ndarray, conv: np.ndarray,
                       max_b: float, active_idx: np.ndarray):
        """Try to solve the LP that guarantees a decrease of the max entry."""
        if not _HAS_LP:
            return None, None  # LP not available

        m = active_idx.size
        if m == 0 or m > 30:            # too many constraints → skip LP
            return None, None

        # Build A_ub and b_ub
        A_ub = np.zeros((m, n + 1), dtype=np.float64)
        b_ub = np.empty(m, dtype=np.float64)

        for row, k in enumerate(active_idx):
            lo = max(0, k - n + 1)
            hi = min(n - 1, k)
            idx = np.arange(lo, hi + 1, dtype=int)
            partner = k - idx
            A_ub[row, idx] = 2.0 * a_vec[partner]      # gradient part
            A_ub[row, -1] = 1.0                         # coefficient of t
            b_ub[row] = max_b - conv[k]                 # RHS (≤0 for the true max)

        # Equality: sum(Δ) = 0 (preserve total mass)
        A_eq = np.zeros((1, n + 1), dtype=np.float64)
        A_eq[0, :n] = 1.0
        b_eq = np.array([0.0], dtype=np.float64)

        # Bounds for Δ (Δ_i ≥ –a_i) and for t (t ≥ 0)
        bounds = [(-a_vec[i], None) for i in range(n)] + [(0.0, None)]

        # Objective: maximise t  → minimise -t
        c = np.zeros(n + 1, dtype=np.float64)
        c[-1] = -1.0

        try:
            res = linprog(c, A_ub=A_ub, b_ub=b_ub,
                          A_eq=A_eq, b_eq=b_eq,
                          bounds=bounds,
                          method='highs',
                          options={'presolve': True, 'time_limit': 5.0})
        except Exception:  # pragma: no cover
            return None, None

        if not res.success:
            return None, None
        t_val = res.x[-1]
        if t_val <= 0.0:
            return None, None
        delta = res.x[:n]
        return delta, t_val

    # --------------------------------------------------------------
    # 7. Main optimisation loop
    # --------------------------------------------------------------
    while time.time() < deadline and iteration < max_iter:
        conv, max_b = conv_and_max(a)
        cur_score = 2.0 * n * max_b

        # record global best if improved
        if cur_score < best_score - 1e-12:
            best_score = cur_score
            best_seq = a.copy()
            print(f"[iter {iteration:6d}] New best: {best_score:.6f}")
            stagnation = 0
        else:
            stagnation += 1

        # ---------- active set ----------
        # indices whose value is within a small relative tolerance of the max
        eps = max_b * 1e-6
        active = np.where(conv >= max_b - eps)[0]
        # keep at most 30 entries to keep the LP small
        if active.size > 30:
            # take the 30 largest values
            idx_sorted = np.argsort(-conv)[:30]
            active = idx_sorted

        # ---------- try LP step ----------
        delta, t_val = attempt_lp_step(a, conv, max_b, active)
        if delta is not None:
            a_candidate = a + delta
            # numerical safety
            a_candidate = np.clip(a_candidate, 0.0, None)
            s = a_candidate.sum()
            if s > 0.0:
                a_candidate /= s
            # evaluate true objective
            cand_score = compute_objective(a_candidate)
            if cand_score < cur_score - 1e-12:
                # accept
                a = a_candidate
                prev_a = a.copy()
                iteration += 1
                continue  # go to next iteration directly

        # ---------- fallback: projected sub‑gradient descent ----------
        g = subgradient_active(a, active)
        step = eta / math.sqrt(iteration + 1.0)
        a_next = a - step * g + mu * (a - prev_a)
        a_next = simplex_projection(a_next)
        prev_a, a = a, a_next

        # occasional random perturbation when stuck
        if stagnation >= 25000:
            sigma = max(0.005, 0.5 / math.sqrt(iteration + 1.0))
            noise = sigma * np.random.randn(n) * a
            a = a + noise
            a = np.clip(a, 0.0, None)
            a /= a.sum()
            prev_a = a.copy()
            stagnation = 0
            print(f"[iter {iteration:6d}] Perturbation applied.")

        # mix two top candidates now and then (helps escape plateaus)
        if iteration % 12000 == 0 and len(top) >= 2:
            i, j = np.random.choice(len(top), 2, replace=False)
            w = np.random.rand()
            mixed = w * top[i][1] + (1.0 - w) * top[j][1]
            mixed /= mixed.sum()
            a = mixed
            prev_a = a.copy()
            print(f"[iter {iteration:6d}] Mixed top candidates.")

        iteration += 1

        # early exit if target reached
        if best_score <= 1.503:
            print(f"Target reached at iteration {iteration}, score = {best_score:.6f}")
            break

    # --------------------------------------------------------------
    # 8. Return the best sequence found (plain Python list)
    # --------------------------------------------------------------
    print(f"Finished after {iteration} iterations, best score = {best_score:.6f}")
    return [float(v) for v in best_seq.tolist()]
