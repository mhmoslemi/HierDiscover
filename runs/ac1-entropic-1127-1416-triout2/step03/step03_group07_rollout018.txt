"""
Propose a step‑function (non‑negative list) that minimizes

        2·n·max(a*a) / (sum(a)**2)

The implementation first tries a small convex‑optimization problem
(using SciPy’s SLSQP) that directly minimises the peak of the
self‑convolution under the simplex constraint.  If SciPy is unavailable
or the optimisation does not finish in time, a simple stochastic search
fallback (Dirichlet sampling) is used.  The whole routine respects the
global time‑budget (default 1000 s) and always returns a valid sequence.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper for the convex optimisation (SciPy) -------------------------------------------------
# ----------------------------------------------------------------------
def _solve_with_slsqp(n: int, deadline: float):
    """
    Solve min_t  t  subject to
        sum(a) = 1,
        a_i >= 0,
        (a * a)[k] <= t   for all k = 0 .. 2*n-2 .
    Returns a tuple (a_opt, t_opt, success) where *a_opt* is a numpy array,
    *t_opt* is the optimal maximal convolution value and *success* is a bool.
    """
    try:
        from scipy.optimize import minimize
    except Exception:                     # SciPy not present
        return None, None, False

    # --------------------------------------------------------------
    # 1. variables:  a_0 .. a_{n-1}  and the scalar t (last entry)
    # --------------------------------------------------------------
    x0 = np.concatenate([np.full(n, 1.0 / n), np.array([1.0 / n])])

    # --------------------------------------------------------------
    # 2. bounds (a_i >= 0, t >= 0)
    # --------------------------------------------------------------
    bounds = [(0.0, None)] * n + [(0.0, None)]

    # --------------------------------------------------------------
    # 3. constraints
    # --------------------------------------------------------------
    # equality: sum(a) = 1
    def eq_sum(x):
        return np.sum(x[:n]) - 1.0

    # inequality:   t - conv_k(x[:n]) >= 0   for every k
    def ineq_conv(x):
        a = x[:n]
        t = x[-1]
        conv = np.convolve(a, a)                     # length 2*n-1
        return t - conv                               # vector of length 2*n-1

    constraints = [
        {"type": "eq", "fun": eq_sum},
        {"type": "ineq", "fun": ineq_conv},
    ]

    # --------------------------------------------------------------
    # 4. objective – we only minimise the scalar t (last component)
    # --------------------------------------------------------------
    def objective(x):
        return x[-1]

    # --------------------------------------------------------------
    # 5. run the optimiser – respect the remaining time budget
    # --------------------------------------------------------------
    # Set a generous iteration limit but break early if we run out of time.
    max_iter = 3000

    # SciPy does not accept a direct time‑limit, so we use a callback.
    def callback(xk):
        # if we are past the deadline, raise to abort the optimiser
        if time.time() > deadline:
            raise RuntimeError("time budget exhausted")

    try:
        res = minimize(
            objective,
            x0,
            method="SLSQP",
            bounds=bounds,
            constraints=constraints,
            options={"maxiter": max_iter, "ftol": 1e-12, "disp": False},
            callback=callback,
        )
    except RuntimeError:
        # time budget exhausted inside callback
        return None, None, False

    if not res.success:
        # optimisation finished but did not converge – treat as failure
        return None, None, False

    a_opt = res.x[:n]
    t_opt = float(res.x[-1])

    # Numerical noise may produce tiny negative entries; clamp and renorm.
    a_opt = np.maximum(a_opt, 0.0)
    s = a_opt.sum()
    if s == 0:
        a_opt = np.full(n, 1.0 / n)
    else:
        a_opt /= s

    return a_opt, t_opt, True

# Simple stochastic fallback (Dirichlet sampling)
# ----------------------------------------------------------------------
def _fallback_random_search(n: int, deadline: float):
    """
    Very cheap stochastic search: repeatedly draw Dirichlet samples
    and keep the best (lowest evaluated score).  The search stops
    when the deadline is reached.
    """
    rng = np.random.default_rng()
    best_a = np.full(n, 1.0 / n)
    best_score = float("inf")
    it = 0
    while time.time() < deadline:
        a = rng.dirichlet(np.ones(n))
        # quick cheap evaluation without the security checks of evaluate_sequence
        # (the checks are harmless, we can call the function directly)

        if score < best_score:
            best_score = score
            best_a = a.copy()
            print(f"[{time.time():.1f}] fallback iter {it}: new best = {best_score:.6f}")
        it += 1
    return best_a

    else:

        best_a = np.full(n, 1.0 / n)
        best_score = evaluate_sequence(best_a.tolist())

    # ------------------------------------------------------------------
    # 3. Stochastic fallback (Dirichlet sampling) – continue until time up

        cand_score = evaluate_sequence(cand.tolist())
        if cand_score < best_score:
            best_score = cand_score
            best_a = cand.copy()
            print(f"[{time.time():.1f}] new overall best = {best_score:.6f}")

        # early exit if we already beat the target
        if best_score < 1.503:

    # ------------------------------------------------------------------

    # ------------------------------------------------------------------
    # Ensure the output respects the required non‑negative, finite format.
    best_a = np.maximum(best_a, 0.0)
    total = best_a.sum()
    if total == 0:
        best_a = np.full_like(best_a, 1.0 / best_a.size)
    else:
        best_a = best_a / total

    return [float(v) for v in best_a.tolist()]
