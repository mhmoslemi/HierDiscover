"""
Propose a step‑function with a very low evaluation score.

Algorithm
---------
1. **Normalization** – For a chosen length `n` we keep the sum of the
   coefficients equal to `sqrt(2·n)`.  Under this scaling the
   `evaluate_sequence` value is exactly the peak of the self‑convolution,
   i.e. `max_k (a*a)_k`.  Minimising this peak is the true objective.

2. **Iterative LP improvement** –  
   *Start* from a smooth “arcsine’’ distribution (or a random Dirichlet).
   For the current vector `f` let `M = max (f*f)`.  
   Solve the linear program  

        maximise   Σ g_i
        subject to  (f * g)_k ≤ M   for all k = 0 … 2n‑2
                    g_i ≥ 0

   (`f * g` is linear in `g`).  The solution `g` has the largest possible
   ℓ₁‑norm while still respecting the current peak `M`.  After scaling
   `g` to the required sum we move a fraction `t` towards it:

        f ← (1‑t)·f + t·g ,   t = 0.5

   The new peak `M' = max (f*f)` is recomputed; if it strictly improves we
   keep the update, otherwise we either try the full step `t = 1` (take `g`)
   or stop the iteration.  Repeating this process provably never increases
   the peak and quickly drives it down.

3. **Local hill‑climbing** –  A cheap stochastic mass‑transfer optimiser
   (swap or move a fraction of mass between two positions) is applied to
   the best vector found by the LP loop.  It uses the true (unsmoothed)
   objective and stops when the time budget is exhausted.

4. **Multiple lengths & restarts** –  The whole procedure is executed for a
   few different values of `n` (1024, 2048, 4096) and with a couple of
   random starts.  The best sequence overall is returned.

The implementation relies only on NumPy, SciPy (`linprog` and `toeplitz`)
and pure Python.  All work respects the 1000 s budget; intermediate
progress is printed for diagnostics.
"""

import time
import numpy as np
from scipy.optimize import linprog
from scipy.linalg import toeplitz

# ----------------------------------------------------------------------
# Basic utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full convolution of two real vectors using FFT (fast)."""
    m = len(a) + len(b) - 1
    n_fft = next_pow2(m)
    A = np.fft.rfft(a, n=n_fft)
    B = np.fft.rfft(b, n=n_fft)
    C = A * B
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


def max_self_convolution(a: np.ndarray) -> float:
    """Return max_k (a*a)_k."""
    conv = convolution_fft(a, a)
    return float(np.max(conv))


def arcsine_initial(n: int) -> np.ndarray:
    """
    Classical arcsine start: a_i ∝ 1 / sqrt( (i+0.5)·(n-0.5-i) ).
    The vector is projected onto the probability simplex (sum = 1).
    """
    i = np.arange(n, dtype=float)
    eps = 0.25
    vals = 1.0 / np.sqrt((i + 0.5) * (n - 0.5 - i) + eps)
    # projection onto simplex (Euclidean)
    u = np.sort(vals)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = np.where(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
    else:
        theta = 0.0
    proj = np.maximum(vals - theta, 0.0)
    # renormalise to sum = 1 (numerical safety)
    proj /= proj.sum()
    return proj


def build_conv_matrix(f: np.ndarray) -> np.ndarray:
    """
    Build the matrix A (size (2n‑1) × n) such that A @ g = f * g
    (full convolution).  Row k corresponds to index k of the convolution.
    """
    n = len(f)
    # first column: f followed by n‑1 zeros
    c = np.concatenate([f, np.zeros(n - 1, dtype=f.dtype)])
    # first row: f[0] then zeros
    r = np.concatenate(([f[0]], np.zeros(n - 1, dtype=f.dtype)))
    A = toeplitz(c, r)                     # shape (2n‑1, n)
    return A


# ----------------------------------------------------------------------
# Linear‑programming based improvement step
# ----------------------------------------------------------------------


def lp_improve(f: np.ndarray, M: float, target_sum: float) -> np.ndarray | None:
    """
    Solve the LP that maximises Σ g_i under (f * g) ≤ M.
    Returns the scaled optimal g (sum == target_sum) or None if LP fails.
    """
    n = len(f)
    A = build_conv_matrix(f)               # (2n‑1) × n
    b = np.full(2 * n - 1, M, dtype=float)

    # Objective: maximise Σ g_i  → minimise –Σ g_i
    c = -np.ones(n, dtype=float)

    bounds = [(0.0, None)] * n

    # Using the high‑performance HiGHS interior‑point method
    res = linprog(
        c,
        A_ub=A,
        b_ub=b,
        bounds=bounds,
        method="highs",
        options={"presolve": True},
    )
    if not res.success:
        return None

    g = res.x
    s = g.sum()
    if s <= 0.0:
        return None
    scale = target_sum / s
    g *= scale
    return g


# ----------------------------------------------------------------------
# Simple stochastic hill‑climber (true objective)
# ----------------------------------------------------------------------


def hill_climb(seq: list[float], deadline: float, max_steps: int = 100_000) -> tuple[list[float], float]:
    """
    Random local search that moves a small fraction of mass between two positions.
    The true `max_self_convolution` is used for acceptance.
    """
    a = np.array(seq, dtype=float)
    n = a.shape[0]
    target_sum = np.sqrt(2.0 * n)

    # keep exact sum after every move (to avoid drift)
    def renorm(v: np.ndarray) -> np.ndarray:
        return v * (target_sum / v.sum())

    best = a.copy()
    best_score = max_self_convolution(best)
    step_scale = 0.1  # relative fraction of mass transferred

    while max_steps > 0 and time.time() < deadline:
        max_steps -= 1
        i, j = np.random.randint(n, size=2)
        if i == j:
            continue

        cand = best.copy()
        delta = step_scale * cand[i]
        if delta <= 0:
            continue
        cand[i] -= delta
        cand[j] += delta
        # avoid tiny negatives caused by rounding
        if cand[i] < 0:
            cand[i] = 0.0
        cand = renorm(cand)

        cur_score = max_self_convolution(cand)

        if cur_score < best_score:
            best_score = cur_score
            best = cand
            step_scale = min(step_scale * 1.05, 0.5)
        else:
            step_scale *= 0.97

    return best.tolist(), best_score


# ----------------------------------------------------------------------
# Core optimisation loop for a fixed length
# ----------------------------------------------------------------------


def optimise_length(
    n: int,
    deadline: float,
    max_lp_iters: int = 30,
    lp_step_fraction: float = 0.5,
) -> tuple[list[float], float]:
    """
    Run the LP‑based iterative improvement for a given length `n`.
    Returns the best sequence found (as a list) and its peak‑convolution value.
    """
    target_sum = np.sqrt(2.0 * n)

    # ------------------------------------------------------------------
    # Try a few different starts (arcsine, uniform, random Dirichlet)
    # ------------------------------------------------------------------
    rng = np.random.default_rng(12345 + n)  # deterministic per length
    starts = [
        arcsine_initial(n) * target_sum,
        np.full(n, target_sum / n),
    ]
    # two random Dirichlet draws with different concentrations
    for a in (0.5, 2.0):
        draws = rng.dirichlet(np.full(n, a))
        starts.append(draws * target_sum)

    best_vec = None
    best_val = np.inf

    for start_idx, f in enumerate(starts, 1):
        if time.time() >= deadline - 5.0:
            break

        print(f"  Length {n}, start {start_idx}/{len(starts)}")
        M = max_self_convolution(f)

        for it in range(1, max_lp_iters + 1):
            if time.time() >= deadline - 2.0:
                break

            g = lp_improve(f, M, target_sum)
            if g is None:
                break

            # convex combination (t = 0.5)
            f_new = (1.0 - lp_step_fraction) * f + lp_step_fraction * g
            M_new = max_self_convolution(f_new)

            if M_new < M - 1e-12:
                f, M = f_new, M_new
                # print modest progress
                if it % 5 == 0:
                    print(f"    LP iter {it:02d}: improved M = {M:.6f}")
                continue

            # try a full step to g
            M_g = max_self_convolution(g)
            if M_g < M - 1e-12:
                f, M = g, M_g
                if it % 5 == 0:
                    print(f"    LP iter {it:02d}: took full g, M = {M:.6f}")
                continue

            # no improvement – stop the LP loop
            break

        # after LP loop, apply a light hill‑climb for polishing
        if time.time() < deadline - 2.0:
            # generous step budget (depends on remaining time)
            remaining = deadline - time.time()
            steps = int(min(200_000, remaining * 30))
            refined, refined_val = hill_climb(f.tolist(), deadline, max_steps=steps)
            f = np.array(refined, dtype=float)
            M = refined_val

        if M < best_val:
            best_val = M
            best_vec = f.tolist()
            print(f"    *** New best for n={n}: M = {best_val:.6f}")

    return best_vec, best_val

# Main entry point required by the evaluation harness
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative sequence that minimises `evaluate_sequence`.
    Uses a linear‑programming based descent followed by a stochastic hill‑climb.
    """
    np.random.seed(seed)
    start_time = time.time()
    # leave a small safety margin for the harness
    deadline = start_time + budget_s - 2.0

    best_seq = None
    best_score = np.inf

    # ------------------------------------------------------------------
    # Lengths to try – larger n gives better asymptotic constants,
    # but we keep them modest to stay within memory / time limits.
    # ------------------------------------------------------------------
    lengths = [1024, 2048, 4096]

    for n in lengths:
        if time.time() >= deadline - 5.0:
            break
        print(f"\n=== Optimising length n = {n} ===")
        seq, val = optimise_length(n, deadline)
        if seq is not None and val < best_score:
            best_score = val
            best_seq = seq
            print(f"  >>> Global best updated: M = {best_score:.6f} (n={n})")

    # ------------------------------------------------------------------
    # Final safeguard – if nothing was found, return a trivial uniform vector.
    # ------------------------------------------------------------------
    if best_seq is None:
        n = 256
        best_seq = [1.0 / n] * n
        best_score = evaluate_sequence(best_seq)

    # Verify with the official evaluator (may include the clipping logic)
    final_score = evaluate_sequence(best_seq)
    print(f"\nFinished search. Final reported score = {final_score:.6f}")

    # The harness expects a plain Python list of floats.
    return best_seq
