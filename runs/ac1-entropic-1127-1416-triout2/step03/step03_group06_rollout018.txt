"""
Propose a step‑function (non‑negative vector) that minimizes
evaluate_sequence().

Algorithm
---------
1. Determine the length *n* (fallback 5146).  The vector is always
   normalised to sum = 1 (the objective is scale‑invariant).

2. Build a small pool of diverse seeds:
   * the historic best `height_sequence_1` (if present)
   * uniform, linear ramp, exponential and a few random Dirichlet draws.
   The pool is filtered by the cheap objective
   ``2 * n * max(convolve(a,a))`` (i.e. the value returned by
   evaluate_sequence for a sum‑1 vector).

3. Pick the best seed and run a long *smooth‑max* optimisation:
   * The true objective `max(conv)` is approximated by the smooth
     function  fβ(a)= (1/β)·log Σk exp(β·convk) .
   * For a given β the gradient is
          ∇fβ(a) = 2·cross‑correlation( w , a )
     where w_k = exp(β·convk)/Σj exp(β·convj) .
   * An Adam optimiser (first/second‑moment adaptivity) is used.
   * After each step the iterate is projected onto the probability simplex
     (non‑negative, unit sum) with Duchi’s Euclidean projection.
   * β is increased slowly (β←min(β+Δβ, βmax)) so that the optimisation
     starts with a very smooth landscape and finishes with an accurate
     max‑approximation.
   * If no improvement is seen for a while a small random perturbation
     (Gaussian noise proportional to the current vector) is added to
     escape plateaus.

4. The best vector seen during the whole run (including the initial seeds)
   is returned as a plain ``list[float]``.  The routine respects the
   1000 s budget (a 5 s safety margin is kept).

The method differs from the previous sub‑gradient approach by using a
smooth approximation of the max‑operator together with Adam’s adaptive
step‑size, which helps to get past flat regions and typically yields a
lower final bound.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities (top‑level, no closures)
# ----------------------------------------------------------------------


def simplex_projection(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    v = np.asarray(v, dtype=float)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1] - 1
    theta = cssv[cond][-1] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def compute_objective(a: np.ndarray) -> float:
    """
    Fast evaluation of 2·n·max(a*a) under the convention Σa_i = 1.
    This is exactly the value returned by ``evaluate_sequence`` for a sum‑1
    vector (the denominator cancels out).
    """
    n = a.shape[0]
    L = 2 * n - 1
    fft_a = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(fft_a * fft_a, n=L)
    return 2.0 * n * conv.max()


def generate_seeds(n: int, best_global: np.ndarray = None) -> list[np.ndarray]:
    """
    Produce a modest collection of diverse seed vectors and return them.
    All seeds are normalised to sum = 1.
    """
    seeds = []

    # 1. historic best if supplied
    if best_global is not None:
        seeds.append(best_global.copy())

    # 2. deterministic families
    # uniform
    seeds.append(np.full(n, 1.0 / n))

    # linear decreasing ramp (large at the left, zero at the right)
    ramp = np.arange(n, 0, -1, dtype=float)
    ramp = ramp / ramp.sum()
    seeds.append(ramp)

    # exponential families
    for lam in np.linspace(0.001, 5.0, 12):
        i = np.arange(n, dtype=float)
        w = np.exp(-lam * i)
        w /= w.sum()
        seeds.append(w)

    # polynomial families
    for p in [0.5, 1.0, 1.5, 2.0, 3.0]:
        i = np.arange(n, dtype=float)
        frac = i / (n - 1)
        w = 1.0 - np.power(frac, p)
        w = np.clip(w, 0.0, None)
        w /= w.sum()
        seeds.append(w)

    # 3. random Dirichlet draws
    rng = np.random.default_rng()
    for _ in range(40):
        conc = rng.uniform(0.1, 3.0, size=n)
        w = rng.gamma(conc, 1.0)
        w /= w.sum()
        seeds.append(w)

    # 4. random mixtures of the first few seeds (helps to explore neighbourhoods)
    base = seeds[:5]
    for _ in range(20):
        i, j = rng.choice(len(base), 2, replace=False)
        alpha = rng.random()
        w = alpha * base[i] + (1.0 - alpha) * base[j]
        w /= w.sum()
        seeds.append(w)

    return seeds


# ----------------------------------------------------------------------
# Main optimisation routine
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 0, budget_s: int = 1000) -> list[float]:
    """
    Search for a non‑negative step function that minimises ``evaluate_sequence``.
    The function respects a 1000 s budget and returns the best vector found.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 5.0   # 5 s safety margin

    # --------------------------------------------------------------
    # 1. Problem size
    # --------------------------------------------------------------
    try:
        n = len(height_sequence_1)   # type: ignore[name-defined]
    except NameError:
        n = 5146                     # fallback length

    # --------------------------------------------------------------
    # 2. Initialise pool of seeds and keep the best few
    # --------------------------------------------------------------
    # Use the historic best if present; otherwise start from uniform.
    historic = None
    try:
        historic = np.array([float(x) for x in height_sequence_1], dtype=float)
        historic = historic / historic.sum()
    except NameError:
        pass

    raw_seeds = generate_seeds(n, historic)
    # evaluate cheaply and keep the top 5
    scored = [(compute_objective(s), s) for s in raw_seeds]
    scored.sort(key=lambda x: x[0])
    TOP_K = 5
    top_seeds = [s for _, s in scored[:TOP_K]]
    best_score = scored[0][0]
    best_seq = scored[0][1].copy()
    print(f"Initial best (seed) score = {best_score:.6f}")

    # --------------------------------------------------------------
    # 3. Smooth‑max Adam optimisation (run on the best seed)
    # --------------------------------------------------------------
    a = best_seq.copy()            # current iterate, Σa = 1
    m = np.zeros_like(a)           # first moment (Adam)
    v = np.zeros_like(a)           # second moment (Adam)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-12
    lr0 = 0.5                      # base learning rate (will be scaled by Adam)
    beta_smooth = 0.5              # starting smoothness parameter β
    beta_max = 1500.0              # final β (roughly the point where log‑sum‑exp ≈ max)
    beta_inc = 0.5                 # increment per iteration (will be capped)

    iteration = 0
    stagnation = 0
    L = 2 * n - 1

    while time.time() < deadline:
        iteration += 1

        # ----- current objective -----
        fft_a = np.fft.rfft(a, n=L)
        conv = np.fft.irfft(fft_a * fft_a, n=L)
        cur_max = conv.max()
        cur_score = 2.0 * n * cur_max

        if cur_score < best_score - 1e-12:
            best_score = cur_score
            best_seq = a.copy()
            stagnation = 0
            print(f"Iter {iteration:6d}: new best {best_score:.6f}")
        else:
            stagnation += 1

        # ----- smooth max weights -----
        # avoid overflow – subtract current max before exponentiation
        max_c = cur_max
        beta_smooth = min(beta_max, beta_smooth + beta_inc)
        exp_term = np.exp(beta_smooth * (conv - max_c))
        w = exp_term / exp_term.sum()            # w length = L, sums to 1

        # ----- gradient of the smooth objective -----
        fft_w = np.fft.rfft(w, n=L)
        corr = np.fft.irfft(fft_a * np.conj(fft_w), n=L)   # cross‑correlation (full length)
        grad = 2.0 * corr[:n]                               # only first n entries are needed

        # ----- Adam update -----
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1 ** iteration)
        v_hat = v / (1.0 - beta2 ** iteration)
        step = lr0 / (np.sqrt(v_hat) + eps)

        a_candidate = a - step * m_hat
        a_candidate = simplex_projection(a_candidate)

        a = a_candidate

        # ----- occasional random perturbation to escape plateaus -----
        if stagnation >= 25000:
            # magnitude decays with time, but never becomes zero
            sigma = max(0.005, 0.5 / np.sqrt(iteration + 1))
            noise = sigma * np.random.randn(n) * a
            a = a + noise
            a = np.clip(a, 0.0, None)
            a = a / a.sum()
            stagnation = 0
            print(f"Iter {iteration:6d}: perturbation applied, score {cur_score:.6f}")

        # early exit if target already reached
        if best_score <= 1.503:
            print(f"Target reached at iteration {iteration}, score={best_score:.6f}")
            break

    # --------------------------------------------------------------
    # 4. Return the best sequence discovered (as plain Python list)
    # --------------------------------------------------------------
    print(f"Finished after {iteration} iterations, best score = {best_score:.6f}")
    return [float(v) for v in best_seq.tolist()]
