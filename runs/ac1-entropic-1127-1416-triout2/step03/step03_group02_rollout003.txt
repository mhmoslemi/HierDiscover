def adam_optimize(initial_a, deadline, lam_start=5.0, lam_end=200.0,
                 lam_ramp_iters=5000, 
                 alpha0=0.05, max_iter=200_000,
                 noise_every=1000, noise_scale=1e-6):
    # convert initial_a to probabilities (normalize)
    p = np.array(initial_a, dtype=float)
    p = p / p.sum()
    # param z = log(p) at start (avoid -inf)
    eps = 1e-12
    z = np.log(p + eps)

    m = np.zeros_like(z)
    v = np.zeros_like(z)
    beta1 = 0.9
    beta2 = 0.999
    eps_adam = 1e-8

    best_p = p.copy()
    best_sc = compute_score(p)

    for t in range(1, max_iter+1):
        if time.time() >= deadline:
            break

        lam = lam_start + (lam_end - lam_start) * min(1.0, t / lam_ramp_iters)

        # compute p via softmax
        p = np.exp(z - np.max(z))
        p = p / p.sum()

        # compute conv and weights
        conv = convolution_fft(p, p)
        w = softmax_weights(conv, lam)

        # compute gradient wrt p
        grad_p = gradient_from_weights(p, w)   # shape n
        # also compute gradient of objective: since we approximated max via softmax, objective approximate 2n * (1/lam) log sum exp(lam conv)
        # But we just use grad_p as derived from smoothmax. Actually gradient wrt p of smoothmax = 2 * conv(p, w_rev). That's grad_p.
        # So grad_p = d/dp of smoothmax.

        # compute gradient wrt z
        avg = np.dot(p, grad_p)
        grad_z = p * (grad_p - avg)   # shape n

        # Adam update
        m = beta1 * m + (1 - beta1) * grad_z
        v = beta2 * v + (1 - beta2) * (grad_z**2)
        m_hat = m / (1 - beta1**t)
        v_hat = v / (1 - beta2**t)
        step = alpha0 / (np.sqrt(v_hat) + eps_adam)

        # gradient descent (we want minimization) so we subtract step * m_hat
        z -= step * m_hat

        # optional noise injection
        if noise_every and (t % noise_every == 0):
            z += np.random.randn(*z.shape) * noise_scale

        # evaluate current true score
        p_cur = np.exp(z - np.max(z))
        p_cur = p_cur / p_cur.sum()
        cur_sc = compute_score(p_cur)
        if cur_sc < best_sc:
            best_sc = cur_sc
            best_p = p_cur.copy()
            # optionally print progress
            if t % 2000 == 0:
                print(f"    iter={t:,d} lam={lam:.1f} cur={cur_sc:.6f} best={best_sc:.6f}")

    return best_p, best_sc
```

def local_search(p: np.ndarray, deadline: float, rng: np.random.Generator, step_scale: float = 0.01) -> np.ndarray:
    """
    Simple hill climbing: repeatedly move a small amount of probability mass
    from a random index with large value to another random index, accepting
    only if the true objective improves. Stops when time runs out.
    """
    n = len(p)
    best = p.copy()
    best_sc = compute_score(p)
    while time.time() < deadline - 0.01:
        # choose source and target indices
        src = rng.integers(0, n)

        new = new / new.sum()
        sc = compute_score(new)
        if sc < best_sc:
            best_sc = sc
            best = new
    return best

"""
Hybrid Adam‑soft‑max optimiser with a final local‐search polish.

The objective to minimise is

    2·n·max_k (p∗p)_k   ,   p_i ≥ 0 ,   Σ p_i = 1

because evaluate_sequence() is scale‑invariant.
We therefore optimise a probability vector p on the simplex.

* A smooth approximation of the max‑operator is the log‑sum‑exp
  with a temperature λ (soft‑max).  Its gradient can be written as a
  single convolution, which we compute efficiently with real FFTs.

* The optimisation is performed in the unconstrained log‑softmax
  space   z_i  (p = softmax(z)).  An Adam update is applied to z,
  guaranteeing p stays non‑negative and normalised.

* Multiple lengths (1024, 2048, 4096, 8192) and several parametric
  initialisations (uniform, arcsine, power‑law, beta‑like, random Dirichlet)
  are tried.  The best few candidates are refined with a longer schedule
  and a higher final λ.

* After the Adam phase a cheap hill‑climbing local search moves a tiny
  amount of probability mass from indices that dominate the worst
  autocorrelation peak to random other indices, accepting only
  improvements.  This often reduces the true max‑convolution a bit
  further.

The routine respects a 1000 s budget, always returns a feasible sequence,
and prints progress information.
"""

import time
import math
from typing import List, Tuple
import numpy as np

# ----------------------------------------------------------------------
# Basic numerical utilities (FFT based convolution)
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """
    Linear convolution via real FFT (full length, zero‑padded to a power of two).
    """
    m = len(a) + len(b) - 1
    n_fft = next_pow2(m)
    A = np.fft.rfft(a, n=n_fft)
    B = np.fft.rfft(b, n=n_fft)
    C = A * B
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


def compute_score(p: np.ndarray) -> float:
    """
    Official objective (scale‑invariant):
        2·n·max(p∗p)   (p already normalised to sum 1)
    """
    n = len(p)
    conv = convolution_fft(p, p)
    return 2.0 * n * float(np.max(conv))


# ----------------------------------------------------------------------
# Initial families (all returned already normalised)
# ----------------------------------------------------------------------


def arcsine_initial(n: int) -> np.ndarray:
    """Discrete arcsine density, normalised to sum = 1."""
    i = np.arange(n, dtype=float)
    eps = 0.25
    vals = 1.0 / np.sqrt((i + 0.5) * (n - 0.5 - i) + eps)
    return vals / vals.sum()


def generate_shape(n: int, p: float, eps: float) -> np.ndarray:
    """Symmetric power‑law shape a_i ∝ ((i+ε)(n‑1‑i+ε))^(−p)."""
    i = np.arange(n, dtype=float)
    base = ((i + eps) * (n - 1 - i + eps)) ** (-p)
    s = base.sum()
    if s == 0.0:
        return np.full(n, 1.0 / n)
    return base / s


# ----------------------------------------------------------------------
# Soft‑max utilities for the smooth max
# ----------------------------------------------------------------------


def softmax_weights(conv: np.ndarray, lam: float) -> np.ndarray:
    """
    Stable soft‑max over the convolution entries.
    Returns a probability vector of the same length as `conv`.
    """
    x = lam * conv
    mx = np.max(x)
    w = np.exp(x - mx)
    w /= w.sum()
    return w


def gradient_from_weights(p: np.ndarray, w: np.ndarray) -> np.ndarray:
    """
    Gradient of Σ_k w_k·(p∗p)_k with respect to p.
    Implements   g_j = 2· Σ_m p_m · w_{m+j}.
    Computed via one FFT‑based convolution.
    """
    # w has length 2·n‑1, we need the reversed version for convolution
    w_rev = w[::-1]
    conv = convolution_fft(p, w_rev)               # length = n + (2n‑1) – 1 = 3n‑1
    idx = 2 * len(p) - 2                            # index of element corresponding to shift 0
    grads = 2.0 * conv[idx - np.arange(len(p))]
    return grads


# ----------------------------------------------------------------------
# Adam optimiser on the soft‑max surrogate
# ----------------------------------------------------------------------


def adam_optimize(
    init_p: np.ndarray,
    deadline: float,
    *,
    lam_start: float = 5.0,
    lam_end: float = 200.0,
    lam_ramp_iters: int = 5000,
    alpha0: float = 0.05,
    max_iter: int = 200_000,
    noise_every: int = 0,
    noise_scale: float = 0.0,
) -> Tuple[np.ndarray, float]:
    """
    Adam optimisation of the smooth max (log‑sum‑exp) of the self‑convolution.
    Returns the best probability vector seen and its true (non‑smooth) score.
    """
    # initialise in the log‑softmax space
    p = init_p.astype(float).copy()
    p /= p.sum()
    eps = 1e-12
    z = np.log(p + eps)

    m = np.zeros_like(z)
    v = np.zeros_like(z)
    beta1 = 0.9
    beta2 = 0.999
    eps_adam = 1e-8

    best_p = p.copy()
    best_sc = compute_score(p)

    for it in range(1, max_iter + 1):
        if time.time() >= deadline:
            break

        # ----- temperature schedule -----
        lam = lam_start + (lam_end - lam_start) * min(1.0, it / lam_ramp_iters)

        # ----- current probability vector (soft‑max) -----
        z_max = np.max(z)
        p = np.exp(z - z_max)
        p /= p.sum()

        # ----- smooth max and its gradient -----
        conv = convolution_fft(p, p)                # length 2n‑1
        w = softmax_weights(conv, lam)              # soft‑max weights
        grad_p = gradient_from_weights(p, w)        # ∂/∂p  (smooth max)

        # ----- chain rule to obtain ∂/∂z -----
        avg = np.dot(p, grad_p)
        grad_z = p * (grad_p - avg)

        # ----- Adam update (minimisation) -----
        m = beta1 * m + (1.0 - beta1) * grad_z
        v = beta2 * v + (1.0 - beta2) * (grad_z ** 2)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        step = alpha0 / (np.sqrt(v_hat) + eps_adam)

        z -= step * m_hat

        # ----- occasional exploration noise -----
        if noise_every and (it % noise_every == 0):
            z += np.random.randn(*z.shape) * noise_scale

        # ----- evaluate true objective periodically -----
        if it % 2000 == 0 or it == max_iter:
            # recompute p for the true score
            p_cur = np.exp(z - np.max(z))
            p_cur /= p_cur.sum()
            cur_sc = compute_score(p_cur)
            if cur_sc < best_sc:
                best_sc = cur_sc
                best_p = p_cur.copy()
                print(f"        iter={it:,d} λ={lam:.1f} cur={cur_sc:.6f} best={best_sc:.6f}")

    return best_p, best_sc

# Simple hill‑climbing local improvement

def local_search(p: np.ndarray, deadline: float, rng: np.random.Generator,
                 step_scale: float = 0.01, max_trials: int = 200_000) -> np.ndarray:
    """
    Randomly move a small amount of mass from a heavy index to a random
    other index, accepting only improvements.  Stops when the time budget
    is exhausted or after `max_trials` attempts.
    """
    n = len(p)
    best = p.copy()
    best_sc = compute_score(best)

    trials = 0
    while time.time() < deadline - 0.01 and trials < max_trials:
        trials += 1
        src = rng.integers(0, n)
        if best[src] < 1e-12:
            continue
        tgt = rng.integers(0, n)
        if src == tgt:
            continue

        # propose moving a fraction of src's mass
        delta = step_scale * best[src] * (0.5 + rng.random())
        new = best.copy()
        new[src] = max(0.0, new[src] - delta)
        new[tgt] = new[tgt] + delta
        new /= new.sum()                      # re‑normalise (sum stays 1)

        sc = compute_score(new)
        if sc < best_sc:
            best_sc = sc
            best = new
    return best

        a_refined, _ = adam_optimize(
            init_p=a,
            deadline=deadline,
            lam_start=10.0,
            lam_end=800.0,
            lam_ramp_iters=15000,
            alpha0=0.03,
            max_iter=max_iter,
            noise_every=800,
            noise_scale=5e-6,
        )

        # light local search to escape the last plateau
        a_final = local_search(a_refined, deadline, rng, step_scale=0.008)

        final_val = evaluate_sequence(a_final.tolist())
        print(f"Final polished evaluate_sequence score = {final_val:.6f}")
        if final_val < best_val:
            best_val = final_val
            best_seq = a_final.tolist()
            print("*** Updated global best after final polish ***")

    # ------------------------------------------------------------------
    # 3. Safety fallback
    # ------------------------------------------------------------------
    if not best_seq:
        best_seq = [1.0] * 256
        best_val = evaluate_sequence(best_seq)

    print(f"\nSearch finished. Best reported score = {best_val:.6f}")
    return best_seq
