def anneal(seq, deadline, max_iter=200000):
    a = np.array(seq, float)
    a = project_to_simplex(a)
    best_a = a.copy()
    best_score = compute_score(a)
    n = len(a)
    t0 = 1.0
    t_end = 0.001
    start = time.time()
    for it in range(1, max_iter+1):
        if time.time() > deadline:
            break
        # temperature schedule
        T = t0 * (1 - (it / max_iter)) ** 3
        # random pair
        i, j = np.random.randint(n, size=2)
        if i==j: continue
        # propose move
        delta = np.random.rand() * T * a[i]  # fraction of a[i]
        if delta == 0: continue
        cand = a.copy()
        cand[i] -= delta
        cand[j] += delta
        # projection onto simplex cheap maybe not needed if delta small not exceed bound? Ensure non-neg.
        cand = project_to_simplex(cand)   # ensures sum=1, nonneg
        score = compute_score(cand)
        if score < best_score:
            best_score = score
            best_a = cand.copy()
            a = cand
        else:
            # Metropolis acceptance
            if np.random.rand() < np.exp(-(score - best_score)/max(T,1e-9)):
                a = cand
    return best_a.tolist(), best_score
```

def simulated_annealing(seq, deadline, max_iter):
    a = np.array(seq, float)
    # ensure simplex:
    a = a / a.sum()
    best_a = a.copy()
    best_score = compute_score(a)
    current_score = best_score
    n = len(a)
    # Precompute cumulative distribution for weighted index selection
    for it in range(1, max_iter+1):
        if time.time() > deadline: break
        # schedule temperature
        frac = it / max_iter
        T = max(0.1 * (1 - frac)**2, 1e-6)  # decreasing
        # choose source index weighted by a
        src = np.random.choice(n, p=a)
        dst = np.random.randint(n)
        if src == dst:
            continue
        max_transfer = a[src]
        if max_transfer <= 0:
            continue
        delta = max_transfer * T * np.random.rand()
        if delta <= 0:
            continue
        cand = a.copy()
        cand[src] -= delta
        cand[dst] += delta
        # ensure new vector sum =1
        # if rounding leads to small negative due to float errors, fix
        cand[src] = max(cand[src], 0.0)
        # renormalize? Not needed because sum preserved if we move exact delta.
        score = compute_score(cand)
        # accept if improvement
        if score < current_score:
            a = cand
            current_score = score
            if score < best_score:
                best_score = score
                best_a = cand.copy()
        else:
            # metropolis: accept with probability exp(-(score-current_score)/T_scale)
            # set scale factor maybe based on current_score magnitude
            Tscale = T * 0.01  # tune
            prob = np.exp(-(score - current_score) / (Tscale + 1e-12))
            if np.random.rand() < prob:
                a = cand
                current_score = score
    return best_a.tolist(), best_score
```

def adam_optimize(init_a, deadline, max_iter=100000, alpha_start=5.0, alpha_end=500.0):
    a = init_a.copy()
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1=0.9
    beta2=0.999
    eps=1e-8
    n = len(a)
    for t in range(1, max_iter+1):
        if time.time()>deadline: break
        alpha = alpha_start + (alpha_end - alpha_start) * (t/max_iter)
        # compute grad of smoothed max
        grad, conv = smoothed_gradient(a, alpha)
        # Adam update
        m = beta1*m + (1-beta1)*grad
        v = beta2*v + (1-beta2)*(grad*grad)
        m_hat = m / (1 - beta1**t)
        v_hat = v / (1 - beta2**t)
        step = 0.01 / (np.sqrt(v_hat) + eps)  # learning rate scaling
        a_new = a - step * m_hat
        a_new = project_to_simplex(a_new)
        cur_score = compute_score(a_new)
        if cur_score < best_score:
            best_score = cur_score
            best_a = a_new.copy()
        a = a_new
    return best_a, best_score

def adam_optimize(initial_a, deadline, max_iter=50000, base_lr=0.5):
    a = project_to_simplex(initial_a.copy())
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1=0.9
    beta2=0.999
    eps=1e-8
    best_a = a.copy()
    best_score = compute_score(a)
    for t in range(1, max_iter+1):
        if time.time()>deadline:
            break
        # schedule alpha (smoothness) gradually increasing
        alpha = 5.0 + (400.0 - 5.0) * min(1.0, t / max_iter)
        grad, _ = smoothed_gradient(a, alpha)
        # Adam update
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad*grad)
        m_hat = m / (1 - beta1**t)
        v_hat = v / (1 - beta2**t)
        # learning rate schedule: decay over time
        lr = base_lr / (np.sqrt(t))
        step = lr * m_hat / (np.sqrt(v_hat) + eps)
        a = a - step
        # projection
        a = project_to_simplex(a)
        cur_score = compute_score(a)
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
    return best_a.tolist(), best_score

def simulated_annealing(seq, deadline, max_iter=200_000):
    a = np.array(seq, dtype=float)
    a = a / np.sum(a)
    best_a = a.copy()
    best_score = compute_score(a)
    cur_score = best_score
    n = len(a)
    for it in range(1, max_iter+1):
        if time.time() > deadline:
            break
        # Temperature schedule: exponential decay
        frac = it / max_iter
        T = max(0.2 * (1 - frac)**2, 1e-6)  # example
        # source index weighted by a (high probability for larger mass)
        src = np.random.choice(n, p=a)
        dst = np.random.randint(n)
        if src == dst:
            continue
        # limit transfer size
        max_transfer = a[src]
        if max_transfer <= 0:
            continue
        delta = max_transfer * T * np.random.rand()
        if delta == 0:
            continue
        cand = a.copy()
        cand[src] -= delta
        cand[dst] += delta
        # enforce non-negativity due to floating errors
        if cand[src] < -1e-12:
            continue
        # score
        score = compute_score(cand)
        # accept if improvement
        if score < cur_score:
            a = cand
            cur_score = score
            if score < best_score:
                best_score = score
                best_a = cand.copy()
        else:
            # accept with probability exp(-(score - cur_score)/ (T * scale))
            scale = 0.01  # small scaling
            prob = np.exp(-(score - cur_score) / (scale * T + 1e-12))
            if np.random.rand() < prob:
                a = cand
                cur_score = score
    return best_a.tolist(), best_score
```

"""
A multi‑phase optimizer for the step‑function bound

    E(a) = 2·n·max(a*a) / (sum(a)^2)

The algorithm consists of

1. **A linear‑programming refinement** – keeping the ℓ¹‑norm fixed
   (∑a = √(2n)) we repeatedly solve a tiny LP:
        maximise   ∑g_j
        s.t.       (a * g)_k ≤ M   (for all k)
                   g_j ≥ 0
   where M = max(a*a).  The LP returns a vector g that is then
   scaled back to the required ℓ¹‑norm and mixed with the current a.
   Even a handful of such steps noticeably reduces the maximal
   convolution value.

2. **Upsampling** – the improved small‑n candidate is interpolated
   to a larger length (default 8192).  The constant in the asymptotic
   bound is almost unchanged, but the finer discretisation often
   yields a small extra drop.

3. **Projected Adam optimisation** on a smooth log‑sum‑exp
   surrogate of the max.  The gradient is computed analytically via
   FFT‑based convolutions and the iterates are projected onto the
   probability simplex.

4. **Stochastic annealing / hill‑climbing** – a cheap pairwise mass‑
   transfer random walk finishes the search, accepting occasional
   uphill moves to escape shallow local minima.

All steps respect the time budget (default 1000 s) and only use
standard scientific Python libraries (numpy, scipy).  The returned
list of non‑negative floats is ready for ``evaluate_sequence``.
"""

import time
import numpy as np
import scipy.optimize

# ----------------------------------------------------------------------
# Basic linear‑algebra utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full convolution using real FFT (fast)."""
    m = len(a) + len(b) - 1
    n_fft = next_pow2(m)
    A = np.fft.rfft(a, n=n_fft)
    B = np.fft.rfft(b, n=n_fft)
    C = A * B
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex (sum=1, v≥0)."""
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / v.shape[0])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho = np.where(u - cssv / np.arange(1, v.shape[0] + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def compute_score(a: np.ndarray) -> float:
    """
    Exact objective used by ``evaluate_sequence`` when ∑a = 1.
    Returns 2·n·max(conv)  (the denominator is 1).
    """
    a = np.asarray(a, dtype=float)
    n = a.shape[0]
    conv = convolution_fft(a, a)
    return float(2.0 * n * np.max(conv))


# ----------------------------------------------------------------------
# Families of candidate step‑functions
# ----------------------------------------------------------------------


def arcsine_initial(n: int) -> np.ndarray:
    """Arcsine shape a_i ∝ 1/√((i+½)(n‑½‑i))."""
    i = np.arange(n, dtype=float)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - 0.5 - i))
    return vals / np.sum(vals)


def beta_sequence(n: int, alpha: float, beta: float | None = None) -> np.ndarray:
    """Beta‑type a_i ∝ (i+½)^α · (n‑½‑i)^β."""
    if beta is None:
        beta = alpha
    i = np.arange(n, dtype=float)
    left = np.power(i + 0.5, alpha)
    right = np.power(n - 0.5 - i, beta)
    seq = left * right
    seq = np.clip(seq, 0.0, 1e12)
    total = np.sum(seq)
    if total == 0.0:
        seq = np.full(n, 1.0 / n)
    else:
        seq = seq / total
    return seq


def exponential_sequence(n: int, lam: float) -> np.ndarray:
    """a_i ∝ exp(−lam·|i−(n−1)/2|)."""
    i = np.arange(n, dtype=float)
    center = (n - 1) / 2.0
    seq = np.exp(-lam * np.abs(i - center))
    return seq / np.sum(seq)


def sine_sequence(n: int) -> np.ndarray:
    """Half‑sine shape a_i ∝ sin(π·(i+½)/n)."""
    i = np.arange(n, dtype=float) + 0.5
    seq = np.sin(np.pi * i / n)
    return seq / np.sum(seq)


def upsample_sequence(seq: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation of a probability vector to a larger size."""
    old_n = len(seq)
    old_x = (np.arange(old_n) + 0.5) / old_n
    new_x = (np.arange(new_n) + 0.5) / new_n
    new_vals = np.interp(new_x, old_x, seq)
    return new_vals / np.sum(new_vals)


# ----------------------------------------------------------------------
# Smoothed‑max gradient (log‑sum‑exp surrogate)
# ----------------------------------------------------------------------


def smoothed_gradient(a: np.ndarray, alpha: float):
    """
    Gradient of  f_α(a) = (1/α)·log Σ_k exp(α·(a∗a)_k) .
    Returns (gradient, raw convolution a∗a).
    """
    n = a.shape[0]
    conv = convolution_fft(a, a)                 # length 2n‑1
    maxc = np.max(conv)
    exp_term = np.exp(alpha * (conv - maxc))
    w = exp_term / np.sum(exp_term)              # softmax weights, Σ w = 1

    # correlation a ⋆ w (equivalent to gradient of Σ w·conv)
    w_rev = w[::-1]
    corr = convolution_fft(a, w_rev)            # length 3n‑2
    # slice indices n‑1 … 2n‑2 (inclusive) and reverse to obtain
    # Σ_j a_j·w_{j+k} for k = 0…n‑1
    grad_part = corr[n - 1 : 2 * n - 1][::-1]
    grad = 2.0 * grad_part
    return grad, conv


# ----------------------------------------------------------------------
# Linear programming refinement (core of the new strategy)
# ----------------------------------------------------------------------


def solve_lp_max_sum(f: np.ndarray, M: float) -> np.ndarray:
    """
    Solve
        maximise   Σ g_j
        s.t.       (f * g)_k ≤ M   for all k
                   g_j ≥ 0
    Using SciPy's high‑performance 'highs' LP solver.
    """
    n = f.shape[0]
    num_cons = 2 * n - 1
    # Build the constraint matrix A (num_cons × n)
    A = np.zeros((num_cons, n), dtype=float)
    for k in range(num_cons):
        lo = max(0, k - (n - 1))
        hi = min(k, n - 1)
        idx = np.arange(lo, hi + 1, dtype=int)
        # coefficient for g[l] is f[k‑l]
        A[k, lo:hi + 1] = f[k - idx]
    b = np.full(num_cons, M, dtype=float)
    c = -np.ones(n, dtype=float)                 # minimise –Σ g  → maximise Σ g
    bounds = [(0, None)] * n
    res = scipy.optimize.linprog(c, A_ub=A, b_ub=b, bounds=bounds,
                                 method='highs', options={'disp': False})
    if not res.success:
        raise RuntimeError(f"LP failed: {res.message}")
    return res.x


def lp_refine(initial_f: np.ndarray, M0: float, deadline: float,
              max_iters: int = 12, t0: float = 0.2) -> tuple[np.ndarray, float]:
    """
    Perform a few LP‑based mixing steps.
    ``initial_f`` is assumed to have ℓ¹‑norm = sqrt(2·n).
    Returns the best vector (still ℓ¹‑norm = sqrt(2·n)) and its max‑conv value.
    """
    n = initial_f.shape[0]
    target_sum = np.sqrt(2.0 * n)
    f = initial_f.copy()
    best_f = f.copy()
    best_M = M0
    t = t0
    for it in range(max_iters):
        if time.time() >= deadline:
            break
        try:
            g = solve_lp_max_sum(f, best_M)
        except Exception as e:
            print(f"[LP] iteration {it} failed: {e}")
            break
        S = np.sum(g)
        if S <= 0:
            print("[LP] zero‑sum solution – terminating.")
            break
        scaling = target_sum / S
        g_scaled = g * scaling
        # convex combination
        f = (1.0 - t) * f + t * g_scaled
        # keep exact ℓ¹‑norm (numerical drift)
        f = f * (target_sum / np.sum(f))
        cur_M = np.max(convolution_fft(f, f))
        if cur_M < best_M:
            best_M = cur_M
            best_f = f.copy()
            # gradually reduce the mixing step
            t = max(t * 0.85, 0.02)
        else:
            # if not improving, shrink t more aggressively
            t = max(t * 0.6, 0.02)
    return best_f, best_M

        # gradually increase the smoothing parameter
        alpha = 5.0 + (400.0 - 5.0) * min(1.0, t / max_iter)
        grad, _ = smoothed_gradient(a, alpha)

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        lr = base_lr / np.sqrt(t)                # decaying learning rate
        step = lr * m_hat / (np.sqrt(v_hat) + eps)
        a = a - step
        a = project_to_simplex(a)

        cur_score = compute_score(a)
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()

        # occasional status output (very cheap)
        if t % 20000 == 0:
            print(f"[Adam] iter {t:,d}  cur={cur_score:.6f}  best={best_score:.6f}")

    return best_a.tolist(), best_score


# ----------------------------------------------------------------------
# Stochastic annealing / hill‑climbing
# ----------------------------------------------------------------------


def annealing_hill_climb(seq, deadline, max_iter: int = 250_000):
    """
    Pair‑wise mass‑transfer annealing.  The move
        a_i ← a_i – δ ,  a_j ← a_j + δ
    preserves the ℓ¹‑norm.  A simple temperature schedule allows
    occasional uphill steps.
    """
    a = np.array(seq, dtype=float)
    a = a / np.sum(a)                # guarantee exact probability simplex
    n = a.shape[0]
    best_a = a.copy()
    best_score = compute_score(a)
    cur_score = best_score

    for it in range(1, max_iter + 1):
        if time.time() >= deadline:
            break
        # linear cooling (fast at the start, slow at the end)
        frac = it / max_iter
        T = max(0.2 * (1.0 - frac) ** 2, 1e-6)

        # pick a source proportionally to its mass – encourages moves from large entries
        src = np.random.choice(n, p=a)
        dst = np.random.randint(n)
        if src == dst:
            continue
        max_transfer = a[src]
        if max_transfer <= 0.0:
            continue
        delta = max_transfer * T * np.random.rand()
        if delta <= 0.0:
            continue
        cand = a.copy()
        cand[src] -= delta
        cand[dst] += delta
        # floating‑point safe projection (tiny negatives are clipped)
        cand = np.maximum(cand, 0.0)
        cand = cand / np.sum(cand)

        cand_score = compute_score(cand)
        if cand_score < cur_score:
            a = cand
            cur_score = cand_score
            if cand_score < best_score:
                best_score = cand_score
                best_a = cand.copy()
        else:
            # Metropolis acceptance
            prob = np.exp(-(cand_score - cur_score) / (0.01 * T + 1e-12))
            if np.random.rand() < prob:
                a = cand
                cur_score = cand_score

        # tiny progress report
        if it % 50000 == 0:
            print(f"[Anneal] iter {it:,d}  cur={cur_score:.6f}  best={best_score:.6f}")

    return best_a.tolist(), best_score


# ----------------------------------------------------------------------
# Main entry point for the competition harness
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a step‑function that yields a low ``evaluate_sequence`` value.
    The method combines a short LP refinement, up‑sampling,
    projected Adam optimisation, and a final annealing hill‑climb.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 2.0   # tiny safety margin

    best_seq = None
    best_val = float("inf")

    # ------------------------------------------------------------------
    # Optional fallback (if a hard‑coded good sequence is available)
    # ------------------------------------------------------------------
    try:
        fallback = [float(x) for x in height_sequence_1]   # type: ignore[name-defined]
        fallback_val = evaluate_sequence(fallback)
        print(f"[Fallback] score = {fallback_val:.6f}")
        if fallback_val < best_val:
            best_val = fallback_val
            best_seq = fallback
    except Exception:
        pass

    # ------------------------------------------------------------------
    # Phase 1 – small‑n LP‑based refinement.
    # ------------------------------------------------------------------
    small_n = 1024
    sum_target = np.sqrt(2.0 * small_n)

    # start from an arcsine shape (good generic seed)
    f0 = arcsine_initial(small_n) * sum_target
    M0 = np.max(convolution_fft(f0, f0))
    print(f"[LP] initial max‑conv (n={small_n}) = {M0:.6e}")

    if time.time() < deadline - 5.0:
        f_best, M_best = lp_refine(f0, M0, deadline,
                                   max_iters=12, t0=0.2)
        # normalise back to probability simplex for later phases
        cand = f_best / np.sum(f_best)
        cand_score = compute_score(cand)
        print(f"[LP] refined score (sum=1) = {cand_score:.6f}")
        if cand_score < best_val:
            best_val = cand_score
            best_seq = cand.tolist()

    # ------------------------------------------------------------------
    # Phase 2 – broaden the search with a few parametric families
    # (fast, cheap, and often yields a slightly better seed).
    # ------------------------------------------------------------------
    lengths = [512, 1024, 2048]
    for n in lengths:
        if time.time() >= deadline - 5:
            break
        print(f"\n[Parametric sweep] n = {n}")

        # arcsine
        seq = arcsine_initial(n)
        seq = seq / np.sum(seq)
        score = compute_score(seq)
        if score < best_val:
            best_val = score
            best_seq = seq.tolist()
            print(f"   new best arcsine n={n}  score={score:.6f}")

        # half‑sine
        seq = sine_sequence(n)
        seq = seq / np.sum(seq)
        score = compute_score(seq)
        if score < best_val:
            best_val = score
            best_seq = seq.tolist()
            print(f"   new best sine n={n}  score={score:.6f}")

        # Beta exponents (symmetric)
        alphas = np.linspace(-0.9, 1.5, 20)
        for a in alphas:
            seq = beta_sequence(n, a)
            score = compute_score(seq)
            if score < best_val:
                best_val = score
                best_seq = seq.tolist()
                print(f"   new best β sym a={a:.2f} n={n}  score={score:.6f}")

        # Random asymmetric Beta pairs
        for _ in range(80):
            a = np.random.uniform(-0.9, 1.5)
            b = np.random.uniform(-0.9, 1.5)
            seq = beta_sequence(n, a, b)
            score = compute_score(seq)
            if score < best_val:
                best_val = score
                best_seq = seq.tolist()
                print(f"   new best β asym a={a:.2f} b={b:.2f} n={n}  score={score:.6f}")

    # ------------------------------------------------------------------
    # Phase 3 – upsample to a larger grid (better asymptotic constant).
    # ------------------------------------------------------------------
    large_n = 8192
    if best_seq is not None and time.time() < deadline - 10:
        print(f"\n[Upsampling] to n = {large_n}")
        up_seq = upsample_sequence(np.array(best_seq, dtype=float), large_n)
        up_score = compute_score(up_seq)
        print(f"   upsampled score = {up_score:.6f}")
        if up_score < best_val:
            best_val = up_score
            best_seq = up_seq.tolist()

    # ------------------------------------------------------------------
    # Phase 4 – short projected Adam optimisation on the smoothed max.
    # ------------------------------------------------------------------
    if best_seq is not None and time.time() < deadline - 6:
        print("\n[Adam] starting projected Adam optimisation")
        init_a = np.array(best_seq, dtype=float)
        refined_seq, refined_score = adam_optimize(
            initial_a=init_a,
            deadline=deadline,
            max_iter=120_000,
            base_lr=0.5,
        )
        print(f"   Adam best score = {refined_score:.6f}")
        if refined_score < best_val:
            best_val = refined_score
            best_seq = refined_seq

    # ------------------------------------------------------------------
    # Phase 5 – final stochastic annealing / hill‑climb.
    # ------------------------------------------------------------------
    if best_seq is not None and time.time() < deadline - 2:
        print("\n[Annealing] final stochastic refinement")
        refined_seq, refined_score = annealing_hill_climb(
            best_seq,
            deadline=deadline,
            max_iter=250_000,
        )
        print(f"   Annealing best score = {refined_score:.6f}")
        if refined_score < best_val:
            best_val = refined_score
            best_seq = refined_seq

    # ------------------------------------------------------------------
    # Safety net – ensure we always return something.
    # ------------------------------------------------------------------
    if best_seq is None:
        print("[Safety] falling back to uniform vector")
        best_seq = [1.0] * 256
        best_val = evaluate_sequence(best_seq)

    final_score = evaluate_sequence([float(x) for x in best_seq])
    print(f"\nSearch finished. Final reported score = {final_score:.6f}")
    return best_seq
