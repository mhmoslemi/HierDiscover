def propose_candidate(budget_s=1000):
    import numpy as np, time, math, random

    deadline = time.time() + budget_s - 0.5
    n = None
    # use previous if available
    try:
        prev = np.asarray(height_sequence_1, dtype=float)
        n = prev.shape[0]
        a = np.maximum(prev, 0.0)
        s = a.sum()
        if s == 0:
            a = np.full_like(prev, 1.0/prev.shape[0])
        else:
            a = a / s
    except Exception:
        n = 5000   # maybe 5000 or 5146 default
        a = np.full(n, 1.0/n)  # uniform

    # ensure shape length n
    # ensure sum 1
    a = a / a.sum()
    # compute initial convolution and score
    def compute_conv(a):
        # use FFT length as next power of two >= 2n-1
        fft_len = 1 << ((2 * n - 2).bit_length())
        fa = np.fft.rfft(a, n=fft_len)
        conv = np.fft.irfft(fa * fa, n=fft_len)[:2*n-1]
        return conv

    B = compute_conv(a)
    cur_max = B.max()
    cur_score = 2 * n * cur_max  # sum=1
    best_a = a.copy()
    best_score = cur_score

    # Simulated annealing parameters
    init_temp = (best_score - 1.0) * 0.5  # half of gap to lower bound? ensure >0
    if init_temp <= 0:
        init_temp = 0.01 * best_score
    temp = init_temp
    start_time = time.time()
    # controls step size
    # We'll linearly decrease max delta fraction from 0.5 to 0.001 over time
    while time.time() < deadline:
        # compute remaining fraction
        elapsed = time.time() - start_time
        frac = elapsed / (deadline - start_time)
        # temperature schedule exponential cooling
        temp = init_temp * math.exp(-5 * frac)  # decays ~ factor e^-5 over budget
        # step size fraction schedule
        step_frac = 0.5 * (1 - frac) + 0.001  # from 0.5 to 0.001 gradually
        # generate move
        i = np.random.randint(0, n)
        # choose j different from i maybe
        j = np.random.randint(0, n-1)
        if j >= i:
            j += 1
        # ensure we have mass to move from i
        if a[i] < 1e-12:
            continue
        delta = a[i] * step_frac * np.random.rand()
        if delta <= 0:
            continue
        # create new a vector
        a_new = a.copy()
        a_new[i] -= delta
        a_new[j] += delta
        # conv and score
        B_new = compute_conv(a_new)
        new_score = 2 * n * B_new.max()
        # decide acceptance
        if new_score < cur_score or np.random.rand() < math.exp(-(new_score - cur_score) / (temp + 1e-12)):
            # accept
            a = a_new
            cur_score = new_score
            # update best
            if new_score < best_score:
                best_score = new_score
                best_a = a.copy()
                print(f"Time {elapsed:.1f}s: new best = {best_score:.6f}")
        # optionally also do occasional None moves to diversify etc.
    # after loop, return best_a
    return best_a.tolist()

"""
Propose a step‑function sequence that minimises
    2·n·max(conv(a,a)) / (sum(a)**2)

The algorithm works in two phases:

1. **Simulated‑annealing (SA) on the simplex** – starting from the
   previously known best sequence (or a uniform one) the mass is
   repeatedly transferred between two positions.  The convolution is
   updated incrementally (O(n) per move) and a Metropolis rule with a
   decreasing temperature accepts improving and occasional worsening
   moves.  The best vector seen during SA is kept.

2. **Adam‑style sub‑gradient optimisation** – the SA result is fed to a
   short Adam optimisation that works on the soft‑max logits of the
   probability vector and uses a sub‑gradient of the max‑convolution.
   This refines the solution found by SA.

Both phases respect the supplied time budget and print occasional
progress messages.  The final sequence is returned as a plain list of
floats (non‑negative and summing to 1, so the evaluator sees the true
objective value).
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Helper for the Adam optimisation (identical to the previous solution)
# ----------------------------------------------------------------------


def _maxconv_and_subgrad(a: np.ndarray, fft_len: int):
    """Full aperiodic convolution a*a and a sub‑gradient of its maximum."""
    n = a.shape[0]
    fft = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(fft * fft, n=fft_len)[: 2 * n - 1]

    max_val = conv.max()
    tol = max_val * 1e-12 + 1e-14
    active = np.where(conv >= max_val - tol)[0]

    grad = np.zeros_like(a)
    for k in active:
        low = max(0, k - n + 1)
        high = min(n - 1, k)
        idx = np.arange(low, high + 1, dtype=int)
        grad[idx] += 2.0 * a[k - idx]

    if active.size > 1:
        grad /= float(active.size)
    return max_val, grad


def _adam_optimize(init_a: np.ndarray,
                   deadline: float,
                   restart_id: int,
                   target: float = 1.5) -> tuple[np.ndarray, float]:
    """Adam on the soft‑max logits of a probability vector."""
    n = init_a.shape[0]
    fft_len = 1 << ((2 * n - 2).bit_length())

    # logits (log‑softmax)
    eps_init = 1e-12
    b = np.log(init_a + eps_init)

    m = np.zeros_like(b)
    v = np.zeros_like(b)
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8

    # initial score
    max_val, _ = _maxconv_and_subgrad(init_a, fft_len)
    best_score = 2.0 * n * max_val
    best_a = init_a.copy()

    stagnation = 0
    it = 0

    while time.time() < deadline:
        it += 1

        # probability vector from logits
        b_max = np.max(b)
        a = np.exp(b - b_max)
        a = a / a.sum()

        # max‑convolution and sub‑gradient
        max_val, grad_a = _maxconv_and_subgrad(a, fft_len)
        cur_score = 2.0 * n * max_val

        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            stagnation = 0
            print(f"Adam restart {restart_id:02d} iter {it:6d}: best = {best_score:.6f}")
        else:
            stagnation += 1

        # stop early if a target is reached
        if best_score < target:
            break
        if stagnation > 10000:
            b += 0.02 * np.random.randn(n)   # escape plateau
            stagnation = 0

        # Adam update in log‑space
        grad_obj = 2.0 * n * grad_a
        avg = np.dot(a, grad_obj)
        grad_b = a * (grad_obj - avg)

        m = beta1 * m + (1.0 - beta1) * grad_b
        v = beta2 * v + (1.0 - beta2) * (grad_b ** 2)

        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        lr = 0.4 / math.sqrt(it)
        b -= lr * m_hat / (np.sqrt(v_hat) + eps)

    return best_a, best_score

    """Search for a low‑score step‑function and return it."""
    rng = np.random.default_rng(12345)
    deadline = time.time() + budget_s - 0.5   # tiny safety margin

    # ------------------------------------------------------------------
    # initialise a (probability vector) and its convolution B
    # ------------------------------------------------------------------
    a = _initial_sequence_from_global()
    n = a.shape[0]

    # initial full convolution via FFT (used only for the first SA step)
    fft_len = 1 << ((2 * n - 2).bit_length())
    a_fft = np.fft.rfft(a, n=fft_len)
    B = np.fft.irfft(a_fft * a_fft, n=fft_len)[: 2 * n - 1]

    cur_score = 2.0 * n * B.max()
    best_a = a.copy()
    best_score = cur_score
    print(f"Initial length {n}, score = {cur_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 1 – simulated annealing
    # ------------------------------------------------------------------
    start_time = time.time()
    sa_deadline = start_time + 0.6 * (deadline - start_time)   # 60 % of the budget

    # temperature (a fraction of the current gap to the theoretical lower bound)
    init_temp = max((cur_score - 1.0) * 0.5, 0.01 * cur_score)

    while time.time() < sa_deadline:
        elapsed = time.time() - start_time
        frac = elapsed / (sa_deadline - start_time)   # 0 → 1

        # exponential cooling
        temp = init_temp * math.exp(-5.0 * frac)

        # move size shrinks from ~0.5 to ~0.001
        max_move_frac = 0.5 * (1.0 - frac) + 0.001

        # pick source index with probability proportional to its mass
        i = rng.choice(n, p=a)
        # pick a different target uniformly
        j = rng.integers(0, n - 1)
        if j >= i:
            j += 1

        if a[i] < 1e-16:
            continue

        # amount to transfer
        delta = a[i] * max_move_frac * rng.random()
        if delta <= 0.0:
            continue

        # incremental convolution update (copy then modify)
        B_new = B.copy()
        B_new[i:i + n] -= 2.0 * delta * a
        B_new[j:j + n] += 2.0 * delta * a
        B_new[2 * i] += delta * delta
        B_new[2 * j] += delta * delta
        B_new[i + j] -= 2.0 * delta * delta

        new_max = B_new.max()
        new_score = 2.0 * n * new_max

        # Metropolis acceptance
        if new_score < cur_score or rng.random() < math.exp(-(new_score - cur_score) / (temp + 1e-12)):
            # accept move
            a[i] -= delta
            a[j] += delta
            B = B_new
            cur_score = new_score

            if new_score < best_score:
                best_score = new_score
                best_a = a.copy()
                print(f"SA time {elapsed:5.1f}s – new best = {best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 2 – Adam refinement (uses the sub‑gradient of the max‑conv)
    # ------------------------------------------------------------------
    while time.time() < deadline - 5.0:
        remaining = deadline - time.time()
        # each Adam run gets at most 30 s or half of what is left
        this_deadline = time.time() + min(30.0, 0.5 * remaining)
        a_opt, score_opt = _adam_optimize(best_a.copy(),
                                          this_deadline,
                                          restart_id=0,
                                          target=best_score)
        if score_opt < best_score:
            best_score = score_opt
            best_a = a_opt.copy()
            print(f"Adam refinement – new best = {best_score:.6f}")

        # if time is low, break
        if time.time() + 5.0 >= deadline:
            break

    # final normalisation (just in case of tiny drift)
    best_a = np.maximum(best_a, 0.0)
    if best_a.sum() == 0.0:
        best_a = np.full_like(best_a, 1.0 / best_a.shape[0])
    else:
        best_a = best_a / best_a.sum()

    print(f"Finished – best score = {best_score:.6f}")
    return best_a.tolist()
