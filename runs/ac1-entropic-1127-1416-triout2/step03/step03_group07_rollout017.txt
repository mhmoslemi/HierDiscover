"""
Hybrid Adam+softmax‐max optimizer for the step‑function problem.
Starting from a simplex‑projected vector (optionally the historic best),
the algorithm repeatedly
* computes the self‑convolution via FFT,
* approximates the max by a soft‑max with a decreasing temperature τ,
* obtains the gradient of the smooth objective,
* updates the logits with Adam,
* projects back onto the probability simplex via the soft‑max,
* keeps the best sequence seen.
Multiple restarts with different lengths are performed until the
time budget expires.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper functions
# ----------------------------------------------------------------------
def _real_full_convolve(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full convolution of two real vectors using FFT (real‑valued)."""
    L = x.shape[0] + y.shape[0] - 1
    # next power of two for efficient FFT
    M = 1 << (L - 1).bit_length()
    X = np.fft.rfft(x, n=M)
    Y = np.fft.rfft(y, n=M)
    conv = np.fft.irfft(X * Y, n=M)[:L]
    return conv


def _self_convolve(a: np.ndarray) -> np.ndarray:
    """Return a*a (full mode) using FFT."""
    # length 2*len(a)-1
    n = a.shape[0]
    m = 2 * n - 1
    M = 1 << (m - 1).bit_length()
    A = np.fft.rfft(a, n=M)
    conv = np.fft.irfft(A * A, n=M)[:m]
    return conv


def _softmax(x: np.ndarray) -> np.ndarray:
    """Numerically stable soft‑max."""
    mx = np.max(x)

    n = a.shape[0]
    conv = _self_convolve(a)
    return 2.0 * n * np.max(conv)

def _optimise_fixed_length(
    n: int,
    deadline: float,
    init_a: np.ndarray | None = None,
) -> tuple[np.ndarray, float]:
    """
    Adam optimisation of the smooth relaxed max for a given length ``n``.
    Returns the best sequence (probability vector) found and its exact score.
    """
    rng = np.random.default_rng()

    # initialise logits x (soft‑max will give a)
    if init_a is not None:
        a0 = np.clip(init_a, 0.0, None)
        if a0.sum() == 0.0:
            a0 = np.full(n, 1.0 / n)
        else:
            a0 = a0 / a0.sum()
        x = np.log(a0 + 1e-12)
    else:
        x = rng.normal(size=n)

    # Adam parameters
    m = np.zeros_like(x)
    v = np.zeros_like(x)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    lr = 0.3

    # temperature schedule for the soft‑max approximation of the max
    tau = 20.0
    tau_min = 0.5
    tau_decay = 0.9995

    best_local_a = None
    best_local_score = float("inf")
    it = 0

    while time.time() < deadline:
        # ---------- soft‑max to get a on the simplex ----------
        mx = np.max(x)
        a = np.exp(x - mx)
        a_sum = a.sum()
        if a_sum == 0.0:
            a = np.full(n, 1.0 / n)
        else:
            a /= a_sum

        # ---------- self‑convolution ----------
        conv = _self_convolve(a)                     # length 2n‑1

        # ---------- smooth max (soft‑max) ----------
        # w_i = softmax(τ·conv_i)
        z = tau * conv
        z_max = np.max(z)
        exp_z = np.exp(z - z_max)
        w = exp_z / np.sum(exp_z)                  # size 2n‑1, sums to 1

        # ---------- gradient w.r.t. a ----------
        a_rev = a[::-1]                             # reversed a
        grad_full = _real_full_convolve(w, a_rev)   # length 3n‑2
        grad_a = 2.0 * grad_full[n - 1 : n - 1 + n]  # slice size n

        # chain rule to logits
        mu = np.dot(grad_a, a)
        grad_x = a * (grad_a - mu)

        # ---------- Adam update ----------
        it += 1
        m = beta1 * m + (1.0 - beta1) * grad_x
        v = beta2 * v + (1.0 - beta2) * (grad_x ** 2)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        step = lr * m_hat / (np.sqrt(v_hat) + eps)
        x -= step

        # keep logits in a reasonable range to avoid overflow
        np.clip(x, -50.0, 50.0, out=x)

        # ---------- keep the best exact score ----------
        if (it % 100) == 0:
            cur_score = _objective(a)
            if cur_score < best_local_score:
                best_local_score = cur_score
                best_local_a = a.copy()
                print(
                    f"[{time.time():.1f}] n={n} it={it} new best={best_local_score:.6f}"
                )
                # early stop when we have already crossed the target
                if best_local_score < 1.503:
                    break

        # ---------- temperature decay ----------
        tau = max(tau_min, tau * tau_decay)

        # occasional stochastic jitter (helps escape plateaus)
        if (it % 5000) == 0:

            a = np.maximum(a + eps_vec, 0.0)
            a /= a.sum()
            # bring the logits back to the new a
            x = np.log(a + 1e-12)

    # final exact evaluation of the last best candidate
    if best_local_a is None:
        best_local_a = a
        best_local_score = _objective(a)

    return best_local_a, best_local_score

"""
Hybrid Adam optimiser with a soft‑max approximation of the max.

The code repeatedly runs an Adam optimiser for a probability vector
`a` (obtained as a soft‑max of free logits).  The objective is the
self‑convolution `a*a`.  The non‑smooth max is replaced by a soft‑max
with a temperature τ that is slowly decreased, turning the problem
into a smooth one that can be tackled with gradient methods.
The true objective `2·n·max(conv)` is evaluated periodically and the
best sequence seen across several restarts (with different lengths)
is returned.
"""

import time
import numpy as np


# ----------------------------------------------------------------------
# Low‑level convolution utilities (real FFT, full mode)
# ----------------------------------------------------------------------
def _real_full_convolve(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full convolution of two real vectors using FFT."""
    L = x.shape[0] + y.shape[0] - 1
    M = 1 << ((L - 1).bit_length())          # next power of two ≥ L
    X = np.fft.rfft(x, n=M)
    Y = np.fft.rfft(y, n=M)
    conv = np.fft.irfft(X * Y, n=M)[:L]
    return conv


def _self_convolve(a: np.ndarray) -> np.ndarray:
    """Return the full convolution a*a (real FFT, length 2·n‑1)."""
    n = a.shape[0]
    m = 2 * n - 1
    M = 1 << ((m - 1).bit_length())
    A = np.fft.rfft(a, n=M)
    conv = np.fft.irfft(A * A, n=M)[:m]
    return conv


# ----------------------------------------------------------------------
# Exact evaluation of the required objective (sum(a) is 1)
# ----------------------------------------------------------------------
def _objective(a: np.ndarray) -> float:
    """2·n·max(a*a) – assuming sum(a)==1."""
    n = a.shape[0]
    conv = _self_convolve(a)
    return 2.0 * n * np.max(conv)

# Optimiser for a fixed length
# ----------------------------------------------------------------------

    n: int,

            a0 = np.full(n, 1.0 / n)
        else:
            a0 = a0 / a0.sum()
        x = np.log(a0 + 1e-12)
    else:
        x = rng.normal(size=n)

    # Adam state
    m = np.zeros_like(x)
    v = np.zeros_like(x)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    lr = 0.3

    # temperature for the soft‑max approximation of max
    tau = 20.0
    tau_min = 0.5
    tau_decay = 0.9995

    best_local_a = None
    best_local_score = float("inf")
    it = 0

    while time.time() < deadline:
        # ---- build a on the simplex (soft‑max) ----
        mx = np.max(x)
        a = np.exp(x - mx)
        s = a.sum()
        if s == 0.0:
            a = np.full(n, 1.0 / n)
        else:
            a /= s

        # ---- self‑convolution ----
        conv = _self_convolve(a)                     # length 2·n‑1

        # ---- smooth max via soft‑max (temperature τ) ----
        z = tau * conv
        z_max = np.max(z)
        exp_z = np.exp(z - z_max)
        w = exp_z / np.sum(exp_z)                  # normalised, length 2·n‑1

        # ---- gradient w.r.t. a (using convolution theorem) ----
        a_rev = a[::-1]                             # reversed a
        grad_full = _real_full_convolve(w, a_rev)   # length 3·n‑2
        grad_a = 2.0 * grad_full[n - 1 : n - 1 + n]   # slice of size n

        # ---- chain rule to logits ----
        mu = np.dot(grad_a, a)
        grad_x = a * (grad_a - mu)

        # ---- Adam update ----
        it += 1
        m = beta1 * m + (1.0 - beta1) * grad_x
        v = beta2 * v + (1.0 - beta2) * (grad_x ** 2)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        step = lr * m_hat / (np.sqrt(v_hat) + eps)
        x -= step

        # keep logits bounded (prevents overflow in the exponentials)
        np.clip(x, -50.0, 50.0, out=x)

        # ---- record the best exact score seen so far ----
        if (it % 100) == 0:
            cur_score = _objective(a)
            if cur_score < best_local_score:
                best_local_score = cur_score
                best_local_a = a.copy()
                print(f"[{time.time():.1f}] n={n} it={it} new best={best_local_score:.6f}")
                if best_local_score < 1.503:      # target already reached
                    break

        # ---- temperature schedule (smooth → sharp) ----
        tau = max(tau_min, tau * tau_decay)

        # ---- occasional jitter to shake out of plateaus ----
        if (it % 5000) == 0:
            eps_vec = 0.001 * rng.standard_normal(n)
            a = np.maximum(a + eps_vec, 0.0)
            a /= a.sum()
            x = np.log(a + 1e-12)

    # if no improvement happened, fall back to the last a we computed
    if best_local_a is None:
        best_local_a = a
        best_local_score = _objective(a)

    return best_local_a, best_local_score


# ----------------------------------------------------------------------
# Public entry point
# ----------------------------------------------------------------------
def propose_candidate(budget_s: int = 1000) -> list[float]:
    """
    Search for a non‑negative step function that minimises
        2·n·max(a*a)/(sum(a)**2)

    The routine performs several Adam runs (different lengths, random
    restarts) until the time budget expires and returns the best
    sequence discovered.
    """
    deadline = time.time() + budget_s - 5.0          # small safety margin

    # ------------------------------------------------------------------
    # historic construction – use it if available
    # ------------------------------------------------------------------
    try:
        historic = np.asarray(height_sequence_1, dtype=float)   # type: ignore[name-defined]
        historic = np.clip(historic, 0.0, None)
        if historic.sum() > 0.0:
            historic = historic / historic.sum()
        else:
            historic = None
    except Exception:
        historic = None

    rng = np.random.default_rng()
    best_seq: np.ndarray | None = None
    best_score = float("inf")

    # lengths that we will try (larger n → a lower theoretical lower bound)
    lengths = [2048, 4096, 6144, 8192, 10240]

    # --------------------------------------------------------------
    # outer loop – keep launching optimisation runs until time runs out
    # --------------------------------------------------------------
    while time.time() < deadline:
        # give each run at most ~80 s (or the remaining time)
        run_deadline = min(deadline, time.time() + 80.0)

        n = int(rng.choice(lengths))

        # attempt to initialise from an already good solution of the same size
        init = None
        if historic is not None and historic.shape[0] == n:
            init = historic
        elif best_seq is not None and best_seq.shape[0] == n:
            init = best_seq

        seq, score = _optimise_fixed_length(n, run_deadline, init)

        if score < best_score:
            best_score = score
            best_seq = seq
            print(f"[{time.time():.1f}] GLOBAL new best = {best_score:.6f} (n={n})")
            if best_score < 1.503:          # we have beaten the target
                break

    # fallback – very unlikely we end up here without a sequence
    if best_seq is None:
        n = 2048
        best_seq = np.full(n, 1.0 / n)

    return [float(v) for v in best_seq.tolist()]
