def subgradient_descent(a, deadline, max_iters, step0=1.0):
    best_a = a.copy()
    best_val = compute_score(a)
    t = 1
    while time.time() < deadline and t <= max_iters:
        conv = convolution_fft(a, a)
        k_star = np.argmax(conv)
        # Subgradient of max: gradient = 2 * shift of a
        n = len(a)
        idx = np.arange(n)
        valid = (k_star - idx >= 0) & (k_star - idx < n)
        grad = np.zeros_like(a)
        grad[valid] = 2 * a[k_star - idx[valid]]
        # step size diminishing
        step = step0 / np.sqrt(t)
        a_new = a - step * grad
        a_new = project_to_simplex(a_new)
        cur_val = compute_score(a_new)
        if cur_val < best_val:
            best_val = cur_val
            best_a = a_new.copy()
        # maybe accept if better, else maybe keep anyway? we could still move.
        a = a_new
        t+=1

def solve_lp_for_a(a, S):
    n = len(a)
    # Build A_ub
    from scipy.sparse import csr_matrix
    row_ind = []
    col_ind = []
    data = []
    # iterate over i
    for i, ai in enumerate(a):
        rows = np.arange(i, i + n, dtype=int)
        cols = np.arange(n, dtype=int)
        row_ind.append(rows)
        col_ind.append(cols)
        data.append(np.full(n, ai))
    row_ind = np.concatenate(row_ind)
    col_ind = np.concatenate(col_ind)
    data = np.concatenate(data)
    # M column entries
    M_rows = np.arange(2*n-1, dtype=int)
    M_cols = np.full(2*n-1, n, dtype=int)
    M_data = np.full(2*n-1, -1.0)
    row_ind = np.concatenate([row_ind, M_rows])
    col_ind = np.concatenate([col_ind, M_cols])
    data = np.concatenate([data, M_data])
    A_ub = csr_matrix((data, (row_ind, col_ind)), shape=(2*n-1, n+1))
    # RHS zeros
    b_ub = np.zeros(2*n-1)
    # equality
    A_eq = np.zeros((1, n+1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([S])
    # objective c
    c = np.zeros(n+1)
    c[n] = 1.0   # minimize M
    # bounds
    bounds = [(0, None)]*n + [(0, None)]
    from scipy.optimize import linprog
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                  bounds=bounds, method='highs', options={'presolve': True})
    if not res.success:
        raise RuntimeError("LP failed")
    b_opt = res.x[:n]
    M_opt = res.x[n]
    return b_opt, M_opt

"""
Improved search for a low‑E step‑function.

The algorithm works in three stages:

1. **Initial shape** – start from the classic arcsine distribution
   (a_i ∝ 1/√((i+½)(n‑½‑i))) on a chosen length n.

2. **Iterative LP‑based refinement** – for the current a solve a small
   linear programme that finds the best partner b minimizing
   max(a * b) under the same ℓ₁‑norm (∑a = ∑b = √(2n)).
   The self‑convolution of the mixture
   a(t) = (1‑t)a + t b can be written as a quadratic function of t,
   so a 1‑D golden‑section search yields the optimal mixing weight.
   The update is performed with the exact optimal t (instead of a tiny
   fixed step), which drives the objective down much faster.

3. **Cheap stochastic hill‑climb** – a final pair‑wise mass‑transfer
   hill‑climb on the true non‑smoothed objective polishes the result.

All vectors are kept on the simplex (∑a = √(2n)), because the
objective is homogeneous.  The code respects the 1000 s time budget,
falls back to a previously known construction if anything fails,
and returns the best sequence found.
"""

import time
import numpy as np
from scipy.sparse import diags, csr_matrix, hstack
from scipy.optimize import linprog

# ----------------------------------------------------------------------
# Basic utilities (simplex projection, convolution, score)
# ----------------------------------------------------------------------


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / v.shape[0], dtype=float)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho = np.where(u - cssv / np.arange(1, v.shape[0] + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full convolution using real FFT (fast)."""
    m = len(a) + len(b) - 1
    n_fft = next_pow2(m)
    A = np.fft.rfft(a, n=n_fft)
    B = np.fft.rfft(b, n=n_fft)
    C = A * B
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


def compute_score(a: np.ndarray) -> float:
    """Exact score used by evaluate_sequence (no scaling needed)."""
    a = np.asarray(a, dtype=float)
    n = len(a)
    conv = convolution_fft(a, a)
    max_val = float(np.max(conv))
    s = float(np.sum(a))
    if s < 1e-12:
        return float("inf")
    return 2.0 * n * max_val / (s * s)


# ----------------------------------------------------------------------
# Simple families for a quick start
# ----------------------------------------------------------------------


def arcsine_initial(n: int) -> np.ndarray:
    """Arcsine‑shaped distribution, normalised to sum = 1."""
    i = np.arange(n, dtype=float)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - 0.5 - i))
    return vals / np.sum(vals)


def upsample_sequence(seq: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation of a probability vector to a larger size."""
    old_n = len(seq)
    old_x = (np.arange(old_n) + 0.5) / old_n
    new_x = (np.arange(new_n) + 0.5) / new_n
    new_vals = np.interp(new_x, old_x, seq)
    return new_vals / np.sum(new_vals)


# ----------------------------------------------------------------------
# Linear‑program sub‑routine (find best partner b for fixed a)
# ----------------------------------------------------------------------


def build_lp_matrix(a: np.ndarray) -> csr_matrix:
    """
    Build the inequality matrix for the LP:
        (a * b)_k - M <= 0   for all k = 0..2n‑2
    The matrix has shape (2n‑1, n+1) where the last column represents -M.
    """
    n = len(a)
    # Toeplitz‑like matrix for the convolution a*b (only the a‑part)
    # Each diagonal offset i (i >= 0) contains the constant a[i].
    A = diags(a, offsets=np.arange(n), shape=(2 * n - 1, n), format="csr")
    # Column for -M
    M_col = csr_matrix(
        (-np.ones(2 * n - 1), (np.arange(2 * n - 1), np.zeros(2 * n - 1))),
        shape=(2 * n - 1, 1),
    )
    A_ub = hstack([A, M_col], format="csr")
    return A_ub


def solve_lp_for_a(a: np.ndarray, S: float):
    """
    Solve
        minimise   M
        subject to a * b <= M   (pointwise)
                   sum(b) = S
                   b >= 0 , M >= 0
    Returns (b_opt, M_opt).
    """
    n = len(a)
    A_ub = build_lp_matrix(a)
    b_ub = np.zeros(2 * n - 1)

    # Equality: sum(b) = S
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([S])

    # Objective: minimise M
    c = np.zeros(n + 1)
    c[-1] = 1.0

    # Bounds
    bounds = [(0, None)] * n + [(0, None)]

    res = linprog(
        c,
        A_ub=A_ub,
        b_ub=b_ub,
        A_eq=A_eq,
        b_eq=b_eq,
        bounds=bounds,
        method="highs",
        options={"presolve": True},
    )
    if not res.success:
        raise RuntimeError(f"LP failed: {res.message}")
    b_opt = res.x[:n]
    M_opt = res.x[-1]
    return b_opt, M_opt


# ----------------------------------------------------------------------
# 1‑D line search for the optimal mixing weight t
# ----------------------------------------------------------------------


def optimal_t(conv_aa: np.ndarray, conv_ab: np.ndarray, conv_bb: np.ndarray):
    """Golden‑section search on [0,1] for the t that minimises max convolution."""
    phi = (np.sqrt(5) - 1) / 2  # 0.618...

    def f(t):
        # quadratic combination of the three convolutions
        comb = (1 - t) ** 2 * conv_aa + 2 * t * (1 - t) * conv_ab + t ** 2 * conv_bb
        return np.max(comb)

    lo, hi = 0.0, 1.0
    f_lo, f_hi = f(lo), f(hi)  # just to initialise (unused later)

    for _ in range(30):  # enough for double precision
        t1 = lo + (1 - phi) * (hi - lo)
        t2 = lo + phi * (hi - lo)
        f1 = f(t1)
        f2 = f(t2)
        if f1 < f2:
            hi = t2
        else:
            lo = t1
    t_opt = (lo + hi) / 2.0
    return t_opt, f(t_opt)


# ----------------------------------------------------------------------
# Stochastic hill‑climb (final polishing)
# ----------------------------------------------------------------------


def hill_climb(seq, time_deadline: float, max_steps: int = 30000, init_step: float = 0.07):
    """
    Pair‑wise mass‑transfer hill‑climbing.
    Accepts only moves that improve the true objective.
    """
    a = np.array(seq, dtype=float)
    a = project_to_simplex(a) * np.sqrt(2.0 * len(a))
    best_a = a.copy()
    best_score = compute_score(best_a)
    n = len(a)
    step = init_step

    while max_steps > 0 and time.time() < time_deadline:
        i, j = np.random.randint(n, size=2)
        if i == j:
            continue
        delta = step * best_a[i]
        if delta <= 0.0:
            continue
        cand = best_a.copy()
        cand[i] -= delta
        cand[j] += delta
        cand = project_to_simplex(cand) * np.sqrt(2.0 * n)
        cur_score = compute_score(cand)
        if cur_score < best_score:
            best_score = cur_score
            best_a = cand
            step = min(step * 1.07, 0.5)
        else:
            step *= 0.96
        max_steps -= 1
    return best_a.tolist(), best_score

# Main optimisation routine
# ----------------------------------------------------------------------


def improve_by_lp(initial_a: np.ndarray, deadline: float, max_iter: int = 12):
    """
    Repeatedly solve the LP for the current a, then blend towards the
    LP‑optimal partner using the exact optimal mixing weight.
    """
    a = initial_a.copy()
    n = len(a)
    S = np.sqrt(2.0 * n)
    best_a = a.copy()
    best_score = compute_score(best_a)

    for it in range(max_iter):
        if time.time() >= deadline - 2.0:
            break
        # 1) solve the LP for the current a
        try:
            b, M_opt = solve_lp_for_a(a, S)
        except Exception as e:
            print(f"LP failed at iteration {it}: {e}")
            break

        # 2) compute the three convolutions needed for the line search
        conv_aa = convolution_fft(a, a)
        conv_ab = convolution_fft(a, b)
        conv_bb = convolution_fft(b, b)

        # 3) optimal mixing weight
        t_opt, new_obj = optimal_t(conv_aa, conv_ab, conv_bb)

        # 4) update a
        a = (1.0 - t_opt) * a + t_opt * b
        # renormalise exactly (numerical drift)
        a *= S / np.sum(a)

        cur_score = compute_score(a)
        if cur_score < best_score - 1e-12:
            best_score = cur_score
            best_a = a.copy()
            print(
                f" LP‑iter {it:2d}: score {cur_score:.6f}, t={t_opt:.4f}, max‑conv={new_obj:.6f}"
            )
        # early stop if mixing makes almost no progress
        if t_opt < 1e-5:
            break

    return best_a, best_score
