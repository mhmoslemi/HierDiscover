"""
Propose a step‑function that minimises
    2·n·max(a*a) / (Σa)²

The search consists of three stages:

1. **Sampling** –  generate many random points on the probability simplex
   (Dirichlet draws, beta‑family shapes, sparse uniform samples) for a
   chosen length *n*.  Keep the few with the smallest surrogate score
   (the surrogate is the official objective with Σa=1, i.e. 2·n·max).

2. **Exact sub‑gradient optimisation** –  starting from each of the
   promising candidates we run an Adam optimiser that uses the true
   sub‑gradient of the *max* of the linear convolution
   (the active lag is the one attaining the maximum).  After each update
   the iterate is projected back onto the probability simplex.  A small
   amount of isotropic noise is injected from time to time to escape
   plateaus.

3. **Polishing** –  the globally best vector is refined again with the
   same Adam scheme for a longer run.

All calculations are performed with an FFT based convolution
(`convolution_fft`) which is fast even for several thousand coefficients.
The routine respects the 1000 s overall budget and returns a list that
will be accepted by ``evaluate_sequence`` (non‑negative, non‑empty).

The method differs from the previous implementation by using the *exact*
sub‑gradient of the max‑convolution rather than a soft‑max surrogate,
and by focussing the optimisation effort on a single length (2048) that
offers a good trade‑off between search speed and theoretical bound.
"""

import time
import numpy as np
from typing import List, Tuple

# ----------------------------------------------------------------------
# Low‑level utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Linear convolution via real FFT."""
    m = len(a) + len(b) - 1
    n_fft = next_pow2(m)
    A = np.fft.rfft(a, n=n_fft)
    B = np.fft.rfft(b, n=n_fft)
    C = A * B
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection onto the simplex {x ≥ 0, Σx = 1}."""
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / v.size)

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho = np.where(u - cssv / np.arange(1, v.size + 1) > 0)[0]
    if len(rho) == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def compute_score(a: np.ndarray) -> float:
    """Fast surrogate of the official score (Σa = 1 → denominator = 1)."""
    n = a.shape[0]
    conv = convolution_fft(a, a)
    max_val = float(np.max(conv))
    return 2.0 * n * max_val


def random_beta_shape(
    n: int, rng: np.random.Generator, alpha: float, beta: float, eps: float
) -> np.ndarray:
    """Symmetric beta‑family density, normalised to Σ=1."""
    i = np.arange(n, dtype=float)
    left = (i + eps) ** (alpha - 1.0)
    right = (n - 1 - i + eps) ** (beta - 1.0)
    base = left * right
    total = base.sum()
    if total <= 0.0 or not np.isfinite(total):
        return np.full(n, 1.0 / n)
    return base / total


def max_conv_and_grad(a: np.ndarray) -> Tuple[np.ndarray, float]:
    """
    Compute the exact sub‑gradient of max(a*a) and the corresponding max value.
    The active lag is the index attaining the maximum.
    """
    n = a.shape[0]
    conv = convolution_fft(a, a)
    max_val = float(conv.max())
    k = int(np.argmax(conv))               # active lag (0 … 2n‑2)

    # gradient w.r.t. a_i is 2·a_{k-i} whenever the index is inside the array
    idx = np.arange(n)
    j = k - idx
    mask = (j >= 0) & (j < n)
    grad = np.zeros_like(a)
    grad[mask] = 2.0 * a[j[mask]]
    return grad, max_val


def adam_optimize_exact(
    init_a: np.ndarray,
    deadline: float,
    max_iter: int = 100_000,
    lr0: float = 0.2,
    noise_every: int = 2000,
    noise_scale: float = 5e-5,
) -> Tuple[np.ndarray, float]:
    """
    Adam optimiser that minimises the true max‑convolution.
    The iterate is kept on the probability simplex (Σa = 1, a_i ≥ 0).
    Returns the best vector found and its surrogate score.
    """
    a = project_to_simplex(init_a.astype(float).copy())
    best_a = a.copy()
    best_sc = compute_score(a)

    beta1, beta2 = 0.9, 0.999
    eps = 1e-8
    m = np.zeros_like(a)
    v = np.zeros_like(a)

    n = a.shape[0]

    for it in range(1, max_iter + 1):
        if time.time() >= deadline:
            break

        grad, max_val = max_conv_and_grad(a)
        cur_sc = 2.0 * n * max_val            # surrogate score (same as evaluate when Σa=1)

        if cur_sc < best_sc:
            best_sc = cur_sc
            best_a = a.copy()

        # Adam moments
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)

        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        step = lr0 * m_hat / (np.sqrt(v_hat) + eps)
        a_new = a - step

        if noise_every and (it % noise_every == 0):
            a_new += np.random.randn(n) * noise_scale

        a_new = project_to_simplex(a_new)

        a = a_new

        if it % 5000 == 0:
            print(
                f"    Adam it={it:,d}  cur={cur_sc:.6f}  best={best_sc:.6f}"
            )

    return best_a, best_sc

def propose_candidate(
    seed: int = 42,
    budget_s: int = 1000,
    **kwargs,
) -> List[float]:
    """
    Search for a non‑negative step function with a low ``evaluate_sequence``
    value.  The method samples many initial points on the simplex,
    refines the most promising ones with an Adam optimiser that uses the
    exact sub‑gradient of the max‑convolution, and finally polishes the
    overall best vector.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 2.0  # safety margin

    best_seq: List[float] = []
    best_val: float = float("inf")

    # --------------------------------------------------------------
    # 0.  Optional fallback from previous run (if the variable exists)
    # --------------------------------------------------------------
    try:
        fallback = [float(x) for x in height_sequence_1]  # type: ignore
        val = evaluate_sequence(fallback)
        print(f"Fallback (height_sequence_1) score = {val:.6f}")
        if val < best_val:
            best_val = val
            best_seq = fallback
    except Exception:
        pass

    # --------------------------------------------------------------
    # 1.  Choose a single length that gives a good trade‑off.
    # --------------------------------------------------------------
    n = 2048  # the length that gave the previous best result
    print(f"\n=== Optimisation for length n = {n} ===")

    # ---- 1a. generate a pool of initial vectors --------------------
    pool: List[np.ndarray] = []

    # deterministic starters
    pool.append(np.full(n, 1.0 / n))          # uniform
    # a few beta‑family samples with varied parameters
    for _ in range(15):
        alpha = rng.uniform(0.1, 3.0)
        beta = rng.uniform(0.1, 3.0)
        eps = rng.uniform(0.001, max(1.0, 0.02 * n))
        pool.append(random_beta_shape(n, rng, alpha, beta, eps))

    # Dirichlet draws – random points on the simplex
    for _ in range(20):
        a = rng.dirichlet(np.ones(n))
        pool.append(a)

    # sparse uniform samples (few non‑zero entries)
    for _ in range(5):
        m = rng.integers(20, n // 4)          # number of support points
        idx = rng.choice(n, size=m, replace=False)
        a = np.zeros(n, dtype=float)
        a[idx] = 1.0 / m
        pool.append(a)

    # ---- 1b. keep the best few candidates -------------------------
    scores = [(compute_score(a), a) for a in pool]
    scores.sort(key=lambda x: x[0])
    top_k = 5
    best_candidates = [a for _, a in scores[:top_k]]
    print(f"  Best initial surrogate score = {scores[0][0]:.6f}")

    # --------------------------------------------------------------
    # 2.  Refine each top candidate with the exact‑gradient Adam optimiser
    # --------------------------------------------------------------
    for idx, cand in enumerate(best_candidates, 1):
        if time.time() >= deadline - 5.0:
            break
        remaining = deadline - time.time()
        max_iter = int(min(120_000, remaining * 25))   # allocate generous budget
        print(f"\nRefining candidate #{idx} (max_iter={max_iter}) …")
        refined_a, refined_sc = adam_optimize_exact(
            init_a=cand,
            deadline=deadline,
            max_iter=max_iter,
            lr0=0.2,
            noise_every=1500,
            noise_scale=5e-5,
        )
        val = evaluate_sequence(refined_a.tolist())
        print(f"  → refined official score = {val:.6f}")
        if val < best_val:
            best_val = val
            best_seq = refined_a.tolist()
            print(f"  *** New global best: {best_val:.6f} ***")

    # --------------------------------------------------------------
    # 3.  Final polishing of the overall best vector
    # --------------------------------------------------------------
    if best_seq and time.time() < deadline - 5.0:
        a = np.array(best_seq, dtype=float)
        remaining = deadline - time.time()
        max_iter = int(min(250_000, remaining * 30))
        print(f"\nFinal polishing (Adam, max_iter={max_iter}) …")
        polished_a, _ = adam_optimize_exact(
            init_a=a,
            deadline=deadline,
            max_iter=max_iter,
            lr0=0.15,
            noise_every=1000,
            noise_scale=3e-5,
        )
        final_val = evaluate_sequence(polished_a.tolist())
        print(f"Final polished official score = {final_val:.6f}")
        if final_val < best_val:
            best_val = final_val
            best_seq = polished_a.tolist()
            print("*** Updated global best after final polish ***")

    # --------------------------------------------------------------
    # 4.  Guard – always return a valid (non‑empty) sequence
    # --------------------------------------------------------------
    if not best_seq:
        best_seq = [1.0] * 256
        best_val = evaluate_sequence(best_seq)

    print(f"\nFinished search. Best reported official score = {best_val:.6f}")
    return best_seq
