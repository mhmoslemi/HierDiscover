def expg_descent(a0, deadline, max_iter=200000, eta0=0.2, alpha_max=400):
    a = np.maximum(a0, 1e-12)
    a = a / np.sum(a)
    best_a = a.copy()
    best_score = compute_score(a)
    t=0
    while time.time() < deadline and t < max_iter:
        t+=1
        # schedule alpha linearly up to alpha_max over iterations
        alpha = min(alpha_max, 5 + (alpha_max-5)*t/max_iter)
        grad, conv = smoothed_gradient(a, alpha)
        # compute step size perhaps decaying
        eta = eta0 / np.sqrt(t)
        # exponentiated update
        # clip gradient for stability
        delta = -eta * grad
        # to avoid overflow, clip delta values
        delta = np.clip(delta, -20, 20)
        a = a * np.exp(delta)
        # renormalize
        a = a / np.sum(a)
        # optionally add small random noise every so often
        if t % 5000 == 0:
            # small jitter if improve
            a = a + np.random.randn(len(a))*1e-5
            a = np.maximum(a, 0)
            a = a / np.sum(a)
        cur_score = compute_score(a)
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
    return best_a.tolist(), best_score
```

"""
Improved search for a low‑value step‑function.

Main ideas
-----------
* Work directly on a relatively large length (default 16384).  The lower
  theoretical bound 2 n·max(a∗a) → 1 is approached only by improving the
  *shape* of the sequence, not by increasing n, but a larger n gives the
  optimizer more freedom.
* Start from a variety of simple families (arcsine, beta, exponential,
  random Dirichlet) and keep the best one.
* Perform a long exponentiated‑gradient descent on the *smooth* log‑sum‑exp
  approximation of the max‑convolution (the same smoothed objective used
  previously, but with an exponentiated‑gradient update and a more
  aggressive schedule for the smoothness parameter α).  The update is
  ``a ← a·exp(−η ∇f)`` which automatically respects the non‑negativity and
  simplex constraints.
* Finish with a cheap stochastic hill‑climb that transfers a small amount of
  mass between random coordinates, accepting only improving moves.
  The hill‑climb runs for as many iterations as time permits.
* All heavy work (convolutions) is performed by FFT, giving ~O(n log n)
  per iteration, allowing several hundred thousand iterations within the
  1000 s budget.

The function `propose_candidate` is the entry point required by the
competition harness.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Basic linear‑algebra utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full linear convolution via a real FFT (fast)."""
    m = len(a) + len(b) - 1
    n_fft = next_pow2(m)
    A = np.fft.rfft(a, n=n_fft)
    B = np.fft.rfft(b, n=n_fft)
    C = A * B
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


def compute_score(a: np.ndarray) -> float:
    """Score identical to `evaluate_sequence` when Σa = 1."""
    a = np.asarray(a, dtype=float)
    n = len(a)
    conv = convolution_fft(a, a)
    max_val = float(np.max(conv))
    s = float(np.sum(a))
    if s < 1e-12:
        return float("inf")
    return 2.0 * n * max_val / (s * s)


# ----------------------------------------------------------------------
# Families of candidate step‑functions
# ----------------------------------------------------------------------


def arcsine_initial(n: int) -> np.ndarray:
    """Arcsine‑shaped start: a_i ∝ 1 / sqrt((i+0.5)(n‑0.5‑i))."""
    i = np.arange(n, dtype=float)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - 0.5 - i))
    return vals / np.sum(vals)


def beta_sequence(n: int, alpha: float, beta: float | None = None) -> np.ndarray:
    """Beta‑type coefficients a_i ∝ (i+½)^α·(n‑½‑i)^β."""
    if beta is None:
        beta = alpha
    i = np.arange(n, dtype=float)
    left = np.power(i + 0.5, alpha)
    right = np.power(n - 0.5 - i, beta)
    seq = left * right
    seq = np.clip(seq, 0.0, 1e12)
    total = np.sum(seq)
    if total == 0.0:
        seq = np.full(n, 1.0 / n)
    else:
        seq = seq / total
    return seq


def exponential_sequence(n: int, lam: float) -> np.ndarray:
    """a_i ∝ exp(-lam·|i - (n‑1)/2|)."""
    i = np.arange(n, dtype=float)
    center = (n - 1) / 2.0
    seq = np.exp(-lam * np.abs(i - center))
    return seq / np.sum(seq)


def random_dirichlet(n: int, concentration: float = 0.5) -> np.ndarray:
    """Sample a Dirichlet vector with the given concentration (smaller → spikier)."""
    g = np.random.gamma(concentration, 1.0, size=n)
    seq = g / np.sum(g)
    return seq


# ----------------------------------------------------------------------
# Smoothed‑max utilities (log‑sum‑exp approximation)
# ----------------------------------------------------------------------


def smoothed_gradient(a: np.ndarray, alpha: float):
    """
    Gradient of the smooth max
        f_α(a) = (1/α)·log Σ_k exp(α·(a∗a)_k)
    Returns the gradient (same shape as a) and the raw convolution a∗a.
    """
    n = len(a)
    conv = convolution_fft(a, a)                     # length 2n‑1
    maxc = np.max(conv)
    # shift to avoid overflow
    exp_term = np.exp(alpha * (conv - maxc))
    w = exp_term / np.sum(exp_term)                  # soft‑max weights, Σ w = 1
    a_rev = a[::-1]
    conv_rev = convolution_fft(a_rev, w)            # length 3n‑2
    start = n - 1
    rev_slice = conv_rev[start:start + n]            # sum_{k} w_k a_{k-i}
    # diagonal entries of the convolution correspond to i=j → k=2i
    w_diag = w[::2][:n]                              # w at even indices (0,2,…)
    grad = 2.0 * rev_slice - w_diag * a
    return grad, conv


# ----------------------------------------------------------------------
# Exponentiated‑gradient descent on the smoothed objective
# ----------------------------------------------------------------------


def expg_descent(
    init_a: np.ndarray,
    time_deadline: float,
    max_iter: int = 300_000,
    eta0: float = 0.6,
    alpha_start: float = 5.0,
    alpha_end: float = 400.0,
    jitter_every: int = 5_000,
    jitter_scale: float = 2e-5,
):
    """
    Projected (simplex‑preserving) exponentiated gradient descent on the
    smoothed max.  Returns the best vector (as a list) and its *true*
    (unsmoothed) score.
    """
    a = np.maximum(init_a.astype(float).copy(), 1e-14)
    a = a / np.sum(a)               # ensure we are on the simplex
    best_a = a.copy()
    best_score = compute_score(best_a)
    n = len(a)

    for t in range(1, max_iter + 1):
        if time.time() >= time_deadline:
            break

        # linear schedule for the smoothness parameter
        alpha = alpha_start + (alpha_end - alpha_start) * min(1.0, t / max_iter)

        grad, _ = smoothed_gradient(a, alpha)

        # step size decays like 1/√t (common in stochastic settings)
        eta = eta0 / np.sqrt(t)

        # exponentiated‑gradient update (stay on simplex automatically)
        delta = -eta * grad
        # clip to avoid numerical overflow/underflow
        delta = np.clip(delta, -20, 20)
        a = a * np.exp(delta)
        a = a / np.sum(a)

        # occasional tiny jitter to escape shallow plateaus
        if jitter_every and (t % jitter_every == 0):
            a = a + np.random.randn(n) * jitter_scale
            a = np.maximum(a, 0.0)
            a = a / np.sum(a)

        cur_score = compute_score(a)
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()

        if t % 10_000 == 0:
            print(
                f"    expg iter {t:,d}  curScore {cur_score:.6f}  best {best_score:.6f}"
            )

    return best_a.tolist(), best_score


# ----------------------------------------------------------------------
# Stochastic hill‑climbing refinement (pair‑wise mass transfer)
# ----------------------------------------------------------------------


def hill_climb(
    seq,
    time_deadline: float,
    max_steps: int = 200_000,
    init_step: float = 0.07,
):
    """
    Simple pair‑wise mass‑transfer hill‑climb.  Moves that lower the true
    score are kept; the step size adapts based on recent success.
    """
    a = np.array(seq, dtype=float)
    a = a / np.sum(a)
    best_a = a.copy()
    best_score = compute_score(best_a)
    n = len(a)
    step = init_step

    while max_steps > 0 and time.time() < time_deadline:
        i, j = np.random.randint(n, size=2)
        if i == j:
            continue
        # amount transferred proportional to the source mass
        delta = step * best_a[i]
        if delta <= 0.0:
            continue
        cand = best_a.copy()
        cand[i] -= delta
        cand[j] += delta
        cand = np.maximum(cand, 0.0)
        cand = cand / np.sum(cand)

        cur_score = compute_score(cand)
        if cur_score < best_score:
            best_score = cur_score
            best_a = cand
            step = min(step * 1.07, 0.5)   # be more aggressive after a success
        else:
            step *= 0.96                    # shrink after a failure
        max_steps -= 1

    return best_a.tolist(), best_score
