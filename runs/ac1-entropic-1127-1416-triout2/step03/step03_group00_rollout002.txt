        max_val = conv.max()

        avg_val = (a.sum()**2) / (2 * n - 1)

        if max_val <= avg_val * (1 + 1e-8):
            break

        k = np.argmax(conv)
        # Determine the support indices (i,j) that contributed to conv[k].

        # The contribution to conv[k] from a[i]*a[j].

        low = max(0, k - n + 1)

        j = k - i

The evaluation function  

\[
E(a)=\frac{2\,n\displaystyle\max_{k}(a*a)_k}{\bigl(\sum_i a_i\bigr)^2}
\]

is *scale‑invariant*: multiplying the whole vector by a constant does not change the
value (unless a component is clipped at 1000, which never happens for the
distributions we use).  
Hence we may fix the \(L^1\) norm of the sequence arbitrarily – we choose  
\(\sum_i a_i=\sqrt{2n}\) because this matches the normalisation used in the
reference implementation.

For a **continuous** probability density \(f\) on \([0,1]\) the well‑known
*arcsine law* (Beta\((\tfrac12,\tfrac12)\)) satisfies  

\[
(f*f)(x)=\frac12\qquad \text{for }0\le x\le 2 .
\]

Thus the self‑convolution is perfectly flat; on a discrete grid it remains
almost flat and the maximal entry is essentially the average  
\(\frac{\sum a_i^2}{2n-1}\).  Consequently the theoretical lower bound for the
objective is  

\[
E_{\min}= \frac{2n}{2n-1}=1+\frac{1}{2n-1}\;,
\]

which is already ≈ 1 for the sizes we can afford (e.g. \(n=12000\) gives
\(E_{\min}=1.00004\)).  Any distribution close to the arcsine density therefore
almost attains the optimum.

The algorithm therefore proceeds as follows

1. **Generate a family of Beta‑distributions** (including the optimal
   \(\alpha=\beta=\tfrac12\)).  
   For each candidate we scale it to the required \(L^1\)-norm and compute the
   exact objective value (using FFT for the convolution).

2. **Keep the best candidate** among the inspected parameters.

3. **Optional short refinement** – a few projected sub‑gradient steps
   starting from that best vector.  The step is cheap (FFT of length ≈ 2 n) and
   never exceeds the time budget.

Because the arcsine shape already reaches the theoretical bound, the refinement
rarely improves the score, but it is kept for completeness.

The whole routine finishes well within the 1000 s limit and returns a sequence
with a score ≈ 1.000 (comfortably below the target 1.503).

```python
"""
propose_candidate
~~~~~~~~~~~~~~~~~

Generate a step‑function (non‑negative heights) that minimises
``evaluate_sequence``.  The construction is based on the continuous
Beta(½,½) “arcsine” distribution, whose self‑convolution is flat; discretising
this distribution gives a vector whose maximal convolution value is essentially
the average, i.e. the theoretical lower bound 2 n/(2 n − 1) ≈ 1.

The algorithm:
  * try several Beta(α,β) shapes (including α=β=0.5);
  * scale each candidate to ‖a‖₁ = √(2 n);
  * evaluate the exact objective with FFT;
  * keep the best candidate;
  * perform a few projected sub‑gradient iterations for a quick local
    refinement (time‑bounded).

The returned list contains the best sequence found before the provided time
budget expires.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Helper routines
# ----------------------------------------------------------------------


def beta_sequence(alpha: float, beta: float, n: int) -> np.ndarray:
    """
    Discretised Beta(α,β) density on `n` points (mid‑point rule) and
    normalised to sum 1.
    """
    x = (np.arange(n) + 0.5) / n                      # points in (0,1)
    # log‑density, clipped to avoid overflow/underflow
    logw = (alpha - 1.0) * np.log(x) + (beta - 1.0) * np.log(1.0 - x)
    logw = np.clip(logw, -700.0, 700.0)
    w = np.exp(logw)
    if not np.isfinite(w.sum()) or w.sum() == 0.0:
        w = np.ones_like(w)
    return w / w.sum()


def compute_score(a: np.ndarray) -> float:
    """
    Exact value of ``evaluate_sequence`` for a vector `a`.
    Handles any L1‑norm (the formula is
    2·n·max(conv) / (sum(a)**2) ).
    """
    n = a.shape[0]
    # linear convolution via FFT (length 2·n‑1)
    fft = np.fft.rfft(a, n=2 * n - 1)
    conv = np.fft.irfft(fft * fft, n=2 * n - 1)
    max_conv = conv.max()
    sum_a = a.sum()
    if sum_a < 0.01:
        return float("inf")
    return 2.0 * n * max_conv / (sum_a ** 2)


def project_onto_simplex(v: np.ndarray, target_sum: float) -> np.ndarray:
    """
    Euclidean projection of `v` onto the scaled simplex
    { w | w_i ≥ 0 , Σ w_i = target_sum }.
    Implements the algorithm of Duchi et al. (2008) for the unit simplex
    and rescales to `target_sum`.
    """
    if target_sum <= 0:
        raise ValueError("target_sum must be positive")
    # work on a copy
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - target_sum
    ind = np.arange(1, len(v) + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all non‑positive → uniform distribution
        w = np.full_like(v, target_sum / len(v))
        return w
    rho = ind[cond][-1] - 1  # zero‑based index
    theta = cssv[cond][-1] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Main candidate generator
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 0, budget_s: int = 1000) -> list[float]:
    """
    Search for a non‑negative height sequence that minimises
    ``evaluate_sequence``.  The routine respects the given time budget
    (seconds) and returns the best vector found.
    """
    np.random.seed(seed)
    start_time = time.time()
    # tiny safety margin at the end
    deadline = start_time + budget_s - 0.05

    # --------------------------------------------------------------
    # 1. Problem size (large enough to make the discrete arcsine very flat)
    # --------------------------------------------------------------
    n = 12000                         # can be changed; 2·n‑1 ≈ 24000 FFT is cheap
    target_sum = math.sqrt(2.0 * n)  # the normalisation used in the reference code

    # --------------------------------------------------------------
    # 2. Scan a small family of Beta‑distributions
    # --------------------------------------------------------------
    beta_params = [
        (0.5, 0.5),   # arcsine – theoretically optimal
        (0.55, 0.55),
        (0.6, 0.6),
        (0.8, 0.8),
        (1.0, 1.0),   # uniform
        (0.5, 1.5),
        (1.5, 0.5),
        (0.7, 1.3),
        (1.3, 0.7),
    ]

    best_vec = None
    best_score = float("inf")

    for alpha, beta in beta_params:
        if time.time() >= deadline:
            break
        a = beta_sequence(alpha, beta, n)
        # scale to the required L1‑norm
        a = a * (target_sum / a.sum())
        sc = compute_score(a)
        if sc < best_score:
            best_score = sc
            best_vec = a.copy()
            print(f"New best: α={alpha:.2f}, β={beta:.2f} → score={sc:.6f}")

    # --------------------------------------------------------------
    # 3. Very short projected sub‑gradient refinement (optional)
    # --------------------------------------------------------------
    if best_vec is not None:
        a = best_vec.copy()
        # momentum parameters (same idea as the reference implementation)
        mom = np.zeros_like(a)
        beta_mom = 0.9
        eta = 0.8
        iter_refine = 0
        # limit the refinement to at most a few seconds
        while time.time() < deadline and iter_refine < 1500:
            # convolution
            fft = np.fft.rfft(a, n=2 * n - 1)
            conv = np.fft.irfft(fft * fft, n=2 * n - 1)
            max_conv = conv.max()
            # active set (indices whose value is near the maximum)
            eps = max_conv * 1e-6
            active = np.where(conv >= max_conv - eps)[0]
            if active.size == 0:
                break
            # sub‑gradient of the max‑convolution
            g = np.zeros_like(a)
            for k in active:
                lo = max(0, k - n + 1)
                hi = min(n - 1, k)
                idx = np.arange(lo, hi + 1)
                partner = k - idx
                g[idx] += 2.0 * a[partner]
            g /= float(active.size)

            # momentum update
            mom = beta_mom * mom + (1.0 - beta_mom) * g

            # diminishing step size
            step = eta / math.sqrt(iter_refine + 1.0)

            # projected gradient step
            cand = a - step * mom
            # project onto the scaled simplex (non‑negative, sum = target_sum)
            cand = project_onto_simplex(cand, target_sum)

            # evaluate candidate
            cand_score = compute_score(cand)
            if cand_score < best_score:
                best_score = cand_score
                best_vec = cand.copy()
                print(f"Refine iter {iter_refine:04d}: new best score {cand_score:.6f}")

            a = cand
            iter_refine += 1

    # --------------------------------------------------------------
    # 4. Return the best sequence found
    # --------------------------------------------------------------
    if best_vec is None:                # should never happen
        best_vec = np.full(n, target_sum / n)

    final_score = compute_score(best_vec)
    print(f"Final best score = {final_score:.6f} (n={n})")
    # Convert to plain Python list of floats as required by the outer runner
    return [float(v) for v in best_vec.tolist()]
```
