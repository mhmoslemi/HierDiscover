"""
Propose a low‑score step‑function for the convolution‑max objective.

Algorithm
---------

1. **Broad parametric sweep** – generate many candidate probability vectors
   of moderate size (1024–4096) using simple families:
   * symmetric Beta‑type  a_i ∝ (i+½)^α·(n‑½‑i)^β
   * arcsine shape mixed with uniform
   * exponential decay
   * random Dirichlet draws (to inject diversity)

   Keep the best vector seen.

2. **Upsample** – linearly interpolate the current best to a larger length
   (8192).  Larger n usually lowers the bound because the discrete
   approximation becomes finer.

3. **Exponentiated‑gradient (EG) descent on the true max‑convolution**.
   The objective (with ‖a‖₁ = 1) is `f(a)=max_k (a∗a)_k`.  Its sub‑gradient
   at the active index `k*` is
        g_i = 2·a_{k*‑i}   (if 0≤k*‑i<n, else 0).

   We perform an EG step
        a_i ← a_i·exp(−η·g_i)    (η = η₀/√t),
   then symmetrise (`a←(a+a[::-1])/2`) and renormalise to the simplex.
   The loop runs until the time budget is exhausted, constantly updating the
   globally best vector.

4. **Cheap stochastic hill‑climb** – a final pair‑wise mass‑transfer
   refinement (as in the previous baseline) to polish any remaining
   imperfections.

All vectors are kept on the probability simplex, ensuring the scale‑invariant
objective is respected.  The routine respects a 1000 s wall‑clock budget and
returns the best sequence found.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Basic linear‑algebra utilities
# ----------------------------------------------------------------------


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    if np.all(v <= 0):
        n = v.shape[0]
        return np.full_like(v, 1.0 / n, dtype=float)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho = np.where(u - cssv / np.arange(1, v.shape[0] + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full convolution using a real FFT (fast)."""
    m = len(a) + len(b) - 1
    n_fft = next_pow2(m)
    A = np.fft.rfft(a, n=n_fft)
    B = np.fft.rfft(b, n=n_fft)
    C = A * B
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


def compute_score(a: np.ndarray) -> float:
    """
    Score identical to `evaluate_sequence` when ∑a = 1.
    Returns 2·n·max(a∗a) because the denominator is 1.
    """
    a = np.asarray(a, dtype=float)
    n = len(a)
    conv = convolution_fft(a, a)
    max_val = float(np.max(conv))
    if max_val < 0:  # should never happen (a ≥ 0)
        max_val = 0.0
    return 2.0 * n * max_val


# ----------------------------------------------------------------------
# Families of candidate step‑functions
# ----------------------------------------------------------------------


def arcsine_initial(n: int) -> np.ndarray:
    """Arcsine‑shaped start: a_i ∝ 1 / sqrt((i+0.5)(n‑0.5‑i))."""
    i = np.arange(n, dtype=float)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - 0.5 - i))
    vals = np.clip(vals, 0.0, 1e12)
    return vals / np.sum(vals)


def beta_sequence(n: int, alpha: float, beta: float | None = None) -> np.ndarray:
    """Beta‑type coefficients a_i ∝ (i+½)^α·(n‑½‑i)^β."""
    if beta is None:
        beta = alpha
    i = np.arange(n, dtype=float)
    left = np.power(i + 0.5, alpha)
    right = np.power(n - 0.5 - i, beta)
    seq = left * right
    seq = np.clip(seq, 0.0, 1e12)
    total = np.sum(seq)
    if total == 0.0:
        seq = np.full(n, 1.0 / n)
    else:
        seq = seq / total
    return seq


def exponential_sequence(n: int, lam: float) -> np.ndarray:
    """a_i ∝ exp(-lam·|i - (n‑1)/2|)."""
    i = np.arange(n, dtype=float)
    center = (n - 1) / 2.0
    seq = np.exp(-lam * np.abs(i - center))
    return seq / np.sum(seq)


def upsample_sequence(seq: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation of a probability vector to a larger size."""
    old_n = len(seq)
    old_x = (np.arange(old_n) + 0.5) / old_n
    new_x = (np.arange(new_n) + 0.5) / new_n
    new_vals = np.interp(new_x, old_x, seq)
    new_vals = np.clip(new_vals, 0.0, None)
    new_vals /= np.sum(new_vals)
    return new_vals


# ----------------------------------------------------------------------
# Stochastic hill‑climbing refinement (same as baseline)
# ----------------------------------------------------------------------


def hill_climb(seq, time_deadline: float, max_steps: int = 30000, init_step: float = 0.07):
    """
    Pair‑wise mass transfer hill‑climbing.
    Moves that lower the true (unsmoothed) score are kept.
    """
    a = np.array(seq, dtype=float)
    a = project_to_simplex(a)
    best_a = a.copy()
    best_score = compute_score(best_a)
    n = len(a)
    step = init_step

    while max_steps > 0 and time.time() < time_deadline:
        i, j = np.random.randint(n, size=2)
        if i == j:
            continue
        delta = step * best_a[i]
        if delta <= 0.0:
            continue
        cand = best_a.copy()
        cand[i] -= delta
        cand[j] += delta
        cand = project_to_simplex(cand)
        cur_score = compute_score(cand)
        if cur_score < best_score:
            best_score = cur_score
            best_a = cand
            step = min(step * 1.07, 0.5)
        else:
            step *= 0.96
        max_steps -= 1
    return best_a.tolist(), best_score

def eg_descent(
    a: np.ndarray,
    time_deadline: float,
    eta0: float = 0.4,
    max_iter: int = 200_000,
    noise_every: int = 5000,
    noise_scale: float = 1e-5,
):

    Sub‑gradient descent on max(a∗a) using an exponentiated‑gradient step.
    The vector `a` is assumed to lie on the probability simplex.
    Returns the best vector encountered and its unsmoothed score.
    """
    n = a.shape[0]
    best_a = a.copy()
    best_score = compute_score(best_a)

    t = 1
    while t <= max_iter and time.time() < time_deadline:
        # convolution and index of the maximal entry
        conv = convolution_fft(a, a)
        k_star = int(np.argmax(conv))

        # gradient for the active index (sub‑gradient of the max)
        idx = np.arange(n)
        rev_idx = k_star - idx
        mask = (rev_idx >= 0) & (rev_idx < n)
        grad = np.zeros_like(a)
        grad[mask] = 2.0 * a[rev_idx[mask]]

        # step size schedule (≈ η₀/√t)
        eta = eta0 / np.sqrt(t)

        # EG update (still non‑negative)
        a = a * np.exp(-eta * grad)

        # optional tiny isotropic noise to escape plateaus
        if noise_every and (t % noise_every == 0):
            a += np.random.randn(n) * noise_scale
            a = np.clip(a, 0.0, None)

        # enforce symmetry (the problem is symmetric)
        a = (a + a[::-1]) / 2.0

        # renormalise to the simplex
        s = a.sum()
        if s == 0.0:
            a = np.full_like(a, 1.0 / n)
        else:
            a = a / s

        cur_score = compute_score(a)
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            # occasional log
            if t % 5000 == 0:
                print(f"    EG iter {t:,d}  new best score {best_score:.6f}")

        t += 1

    return best_a.tolist(), best_score

def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a step‑function that yields a low `evaluate_sequence`
    value.  The routine respects the time budget, combining a parametric sweep,
    up‑sampling, an exponentiated‑gradient refinement and a final hill‑climb.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 3.0  # safety margin

    best_seq = None
    best_val = float("inf")

    # ------------------------------------------------------------------
    # Optional fallback (previously discovered construction)
    # ------------------------------------------------------------------
    try:
        fallback = [float(x) for x in globals()["height_sequence_1"]]
        fallback_val = evaluate_sequence(fallback)
        print(f"Fallback score = {fallback_val:.6f}")
        if fallback_val < best_val:
            best_val = fallback_val
            best_seq = fallback
    except Exception:
        pass

    # ------------------------------------------------------------------
    # Phase 1 – broad parametric sweep on moderate sizes
    # ------------------------------------------------------------------
    lengths = [1024, 2048, 4096]
    for n in lengths:
        if time.time() >= deadline - 10:
            break
        print(f"\n=== Parametric sweep, length = {n} ===")

        # ----- symmetric Beta family -----
        alphas = np.linspace(-0.95, 3.0, 140)
        for α in alphas:
            if time.time() >= deadline - 10:
                break
            seq = beta_sequence(n, α)
            score = compute_score(seq)
            if score < best_val:
                best_val = score
                best_seq = seq.tolist()
                print(f"New best (sym β) n={n} α={α:.3f}  score={score:.6f}")

        # ----- mix arcsine / uniform -----
        arcs = arcsine_initial(n)
        uniform = np.full(n, 1.0 / n)
        for w in np.linspace(0.0, 1.0, 31):
            if time.time() >= deadline - 10:
                break
            seq = (1.0 - w) * arcs + w * uniform
            seq = seq / np.sum(seq)
            score = compute_score(seq)
            if score < best_val:
                best_val = score
                best_seq = seq.tolist()
                print(f"New best (mix) n={n} w={w:.2f}  score={score:.6f}")

        # ----- exponential decay -----
        for lam in np.linspace(0.001, 0.02, 20):
            if time.time() >= deadline - 10:
                break
            seq = exponential_sequence(n, lam)
            score = compute_score(seq)
            if score < best_val:
                best_val = score
                best_seq = seq.tolist()
                print(f"New best (exp) n={n} λ={lam:.4f}  score={score:.6f}")

        # ----- random Dirichlet draws (diversity) -----
        for _ in range(200):
            if time.time() >= deadline - 10:
                break
            concentration = np.random.uniform(0.1, 2.0)
            seq = np.random.dirichlet(np.full(n, concentration))
            score = compute_score(seq)
            if score < best_val:
                best_val = score
                best_seq = seq.tolist()
                print(f"New best (dir) n={n} conc={concentration:.2f}  score={score:.6f}")

    # ------------------------------------------------------------------
    # Phase 2 – up‑sample to a larger size (if time permits)
    # ------------------------------------------------------------------
    target_n = 8192
    if best_seq is not None and time.time() < deadline - 8:
        print(f"\nUpsampling best candidate to length {target_n}")
        best_arr = np.array(best_seq, dtype=float)
        up_seq = upsample_sequence(best_arr, target_n)
        up_score = compute_score(up_seq)
        print(f"Upsampled score = {up_score:.6f}")
        if up_score < best_val:
            best_val = up_score
            best_seq = up_seq.tolist()

    # ------------------------------------------------------------------
    # Phase 3 – exponentiated‑gradient refinement on the true objective
    # ------------------------------------------------------------------
    if best_seq is not None and time.time() < deadline - 6:
        print("\nRunning EG descent on max‑convolution")
        init_a = np.array(best_seq, dtype=float)
        # enforce symmetry before descent
        init_a = (init_a + init_a[::-1]) / 2.0
        init_a = init_a / init_a.sum()
        refined_seq, refined_score = eg_descent(
            init_a,
            time_deadline=deadline,
            eta0=0.4,
            max_iter=200_000,
            noise_every=5000,
            noise_scale=1e-5,
        )
        print(f"EG descent best score = {refined_score:.6f}")
        if refined_score < best_val:
            best_val = refined_score
            best_seq = refined_seq

    # ------------------------------------------------------------------
    # Phase 4 – final cheap stochastic hill‑climb refinement
    # ------------------------------------------------------------------
    if best_seq is not None and time.time() < deadline - 2:
        print("\nFinal hill‑climb refinement")
        refined_seq, refined_val = hill_climb(
            best_seq,
            time_deadline=deadline,
            max_steps=50000,
            init_step=0.05,
        )
        print(f"Hill‑climb score = {refined_val:.6f}")
        if refined_val < best_val:
            best_val = refined_val
            best_seq = refined_seq

    # ------------------------------------------------------------------
    # Safety fallback (should never be needed)
    # ------------------------------------------------------------------
    if best_seq is None:
        best_seq = [1.0] * 256
        best_val = evaluate_sequence(best_seq)

    final_score = evaluate_sequence(best_seq)
    print(f"\nSearch finished. Final reported score = {final_score:.6f}")
    return best_seq
