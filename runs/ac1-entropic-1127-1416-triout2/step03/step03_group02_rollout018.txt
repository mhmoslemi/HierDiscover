Search for a low‑score step‑function using a two‑phase
soft‑max → hard‑max (sub‑gradient) optimisation.
Phase‑1 runs a long exponentiated‑gradient descent with a
log‑sum‑exp temperature that is linearly ramped from a low
value to a very large one (λ≈2000) so that the smooth
approximation gradually turns into the true max.
Phase‑2 switches to a pure hard‑max sub‑gradient descent for
fine‑grained flattening of the convolution.  Several
parameterised families (arcsine, power‑law, beta‑like,
random Dirichlet) are seeded; the best few are refined and
the overall best sequence is returned.  The routine respects
the given time budget and always returns a feasible
non‑negative list.
"""
import time
import numpy as np
from typing import List, Tuple

def next_pow2(x: int) -> int:
    """Return the smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()

def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Linear convolution using real FFT (full length)."""
    m = len(a) + len(b) - 1
    n = next_pow2(m)
    A = np.fft.rfft(a, n=n)
    B = np.fft.rfft(b, n=n)
    C = A * B
    conv = np.fft.irfft(C, n=n)[:m]
    return conv

def compute_score(a: np.ndarray) -> float:
    """Official objective (scale‑invariant)."""
    n = len(a)
    conv = convolution_fft(a, a)
    max_val = float(np.max(conv))
    s = float(np.sum(a))
    if s < 1e-12:
        return float("inf")
    return 2.0 * n * max_val / (s * s)

def normalize(v: np.ndarray) -> np.ndarray:
    """Project non‑negative vector onto the probability simplex."""
    total = v.sum()
    if total <= 0.0:
        return np.full_like(v, 1.0 / len(v))
    return v / total

def arcsine_initial(n: int) -> np.ndarray:
    """Discrete arcsine‑like density (with a small regulariser)."""
    i = np.arange(n, dtype=float)
    eps = 0.25
    vals = 1.0 / np.sqrt((i + 0.5) * (n - 0.5 - i) + eps)
    return normalize(vals)

def generate_shape(n: int, p: float, eps: float) -> np.ndarray:
    """Symmetric power‑law shape a_i ∝ ((i+ε)(n‑1‑i+ε))⁻ᵖ."""
    i = np.arange(n, dtype=float)
    base = ((i + eps) * (n - 1 - i + eps)) ** (-p)
    return normalize(base)

def beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    """Beta‑type shape on the integer grid."""
    i = np.arange(n, dtype=float)
    base = (i + 0.5) ** (alpha - 1) * (n - 1 - i + 0.5) ** (beta - 1)
    if base.sum() == 0.0:
        base = np.full(n, 1.0 / n)
    else:
        base = base / base.sum()
    return base

def softmax_weights(conv: np.ndarray, lam: float) -> np.ndarray:
    """Stable soft‑max over convolution entries."""
    x = lam * conv
    mx = np.max(x)
    w = np.exp(x - mx)
    w /= w.sum()
    return w

def gradient_from_weights(a: np.ndarray, w: np.ndarray) -> np.ndarray:
    """
    Gradient of Σ_k w_k·(a*a)_k w.r.t. a.
    Implements   g = 2·(a * w_rev)
    where w_rev is the reversal of the weight vector.
    """
    w_rev = w[::-1]
    conv = convolution_fft(a, w_rev)       # length = 3·n‑2
    idx = 2 * len(a) - 2                    # position of lag 2n‑2 in the full conv
    grads = 2.0 * conv[idx - np.arange(len(a))]
    return grads

def optimise_softmax(
    init: np.ndarray,
    deadline: float,
    lam_start: float = 2.0,
    lam_end: float = 2000.0,
    lam_ramp_iters: int = 50000,
    eta0: float = 0.8,
    max_iter: int = 300_000,
    noise_every: int = 0,
    noise_scale: float = 0.0,
) -> Tuple[np.ndarray, float]:
    """
    Exponentiated‑gradient descent on the log‑sum‑exp
    smooth approximation of the max‑convolution.
    The temperature λ is linearly increased from lam_start
    to lam_end during the first ``lam_ramp_iters`` steps.
    """
    a = normalize(init.astype(float).copy())
    best_a = a.copy()
    best_sc = compute_score(a)

    n = len(a)
    for it in range(1, max_iter + 1):
        if time.time() >= deadline:
            break

        # temperature schedule
        if it <= lam_ramp_iters:
            lam = lam_start + (lam_end - lam_start) * it / lam_ramp_iters
        else:
            lam = lam_end

        conv = convolution_fft(a, a)
        w = softmax_weights(conv, lam)
        grad = gradient_from_weights(a, w)

        # step size (inverse‑sqrt decay)
        eta = eta0 / np.sqrt(it)

        # exponentiated update (guarantees non‑negativity)
        a = a * np.exp(-eta * grad)

        # optional exploration noise
        if noise_every and (it % noise_every == 0):
            a += np.random.randn(n) * noise_scale
            a = np.maximum(a, 0.0)

        # project to simplex
        a = normalize(a)

        cur = compute_score(a)
        if cur < best_sc:
            best_sc = cur
            best_a = a.copy()

        # occasional report
        if it % 5000 == 0:
            print(f"      softmax iter={it:,d} λ={lam:.0f} cur={cur:.6f} best={best_sc:.6f}")

    return best_a, best_sc

def hardmax_refine(
    init: np.ndarray,
    deadline: float,
    eta0: float = 0.3,
    max_iter: int = 200_000,
) -> Tuple[np.ndarray, float]:
    """
    Sub‑gradient descent on the true max‑convolution.
    In each iteration the index of the current maximum
    is identified and the corresponding (hard) weight vector
    is used to compute the gradient.
    """
    a = normalize(init.astype(float).copy())
    best_a = a.copy()
    best_sc = compute_score(a)

    n = len(a)
    for it in range(1, max_iter + 1):
        if time.time() >= deadline:
            break

        conv = convolution_fft(a, a)
        k = int(np.argmax(conv))          # index of the current peak
        w = np.zeros_like(conv)
        w[k] = 1.0

        grad = gradient_from_weights(a, w)

        # step‑size schedule
        eta = eta0 / np.sqrt(it)

        # exponentiated update
        a = a * np.exp(-eta * grad)

        # keep numbers strictly positive before projection
        a = np.maximum(a, 0.0)

        a = normalize(a)

        cur = compute_score(a)
        if cur < best_sc:
            best_sc = cur
            best_a = a.copy()

        if it % 5000 == 0:
            print(f"      hardmax  iter={it:,d} cur={cur:.6f} best={best_sc:.6f}")

    return best_a, best_sc

    """
    Exponentiated‑gradient descent on the log‑sum‑exp approximation
    of the max‑convolution.  λ is linearly increased from lam_start
    to lam_end during the first ``lam_ramp_iters`` iterations.
    """
    a = normalize(init.astype(float).copy())
    best_a = a.copy()
    best_sc = compute_score(a)

    n = len(a)

    for it in range(1, max_iter + 1):
        if time.time() >= deadline:
            break

        # temperature schedule
        if it <= lam_ramp_iters:
            lam = lam_start + (lam_end - lam_start) * it / lam_ramp_iters
        else:
            lam = lam_end

        conv = convolution_fft(a, a)
        w = softmax_weights(conv, lam)
        grad = gradient_from_weights(a, w)

        eta = eta0 / np.sqrt(it)

        a = a * np.exp(-eta * grad)

        if noise_every and (it % noise_every == 0):
            a += np.random.randn(n) * noise_scale
            a = np.maximum(a, 0.0)

        a = normalize(a)

        cur_sc = compute_score(a)
        if cur_sc < best_sc:
            best_sc = cur_sc
            best_a = a.copy()

        if it % 5000 == 0:
            print(f"      softmax iter={it:,d} λ={lam:.0f} cur={cur_sc:.6f} best={best_sc:.6f}")

    return best_a, best_sc


def hardmax_refine(
    init: np.ndarray,
    deadline: float,
    *,
    eta0: float = 0.3,
    max_iter: int = 200_000,
) -> Tuple[np.ndarray, float]:
    """
    Sub‑gradient descent on the *exact* max‑convolution.
    In each iteration the index of the current maximal entry is taken,
    a hard weight vector is built, and the corresponding gradient is used.
    """
    a = normalize(init.astype(float).copy())
    best_a = a.copy()
    best_sc = compute_score(a)

    n = len(a)

    for it in range(1, max_iter + 1):
        if time.time() >= deadline:
            break

        conv = convolution_fft(a, a)
        k = int(np.argmax(conv))

        w = np.zeros_like(conv)
        w[k] = 1.0
        grad = gradient_from_weights(a, w)

        eta = eta0 / np.sqrt(it)

        a = a * np.exp(-eta * grad)
        a = np.maximum(a, 0.0)
        a = normalize(a)

        cur_sc = compute_score(a)
        if cur_sc < best_sc:
            best_sc = cur_sc
            best_a = a.copy()

        if it % 5000 == 0:
            print(f"      hardmax iter={it:,d} cur={cur_sc:.6f} best={best_sc:.6f}")

    return best_a, best_sc

# Main entry point required by the competition harness

def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs) -> List[float]:
    """
    Search for a non‑negative step‑function with a low ``evaluate_sequence``
    value.  The routine respects the given time budget, explores several
    lengths, refines the most promising seeds and finally returns the best
    sequence found.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 2.0   # small safety margin

    best_seq: List[float] = []
    best_val: float = float("inf")

    # ------------------------------------------------------------------
    # 0. Optional fallback from a previously discovered construction.
    # ------------------------------------------------------------------
    try:
        fallback = [float(x) for x in height_sequence_1]  # type: ignore
        val = evaluate_sequence(fallback)
        print(f"Fallback (height_sequence_1) score = {val:.6f}")
        if val < best_val:
            best_val = val
            best_seq = fallback
    except Exception:
        pass

    # ------------------------------------------------------------------
    # 1. Lengths to explore (powers of two give cheap FFTs).
    # ------------------------------------------------------------------
    lengths = [1024, 2048, 4096]

    for n in lengths:
        if time.time() >= deadline - 5.0:
            break
        print(f"\n=== Exploring length n = {n} ===")

        # --------------------------------------------------------------
        # 1a. Build a modest library of seed vectors.
        # --------------------------------------------------------------
        seeds: List[np.ndarray] = []

        # uniform
        seeds.append(np.full(n, 1.0 / n))

        # classic arcsine
        seeds.append(arcsine_initial(n))

        # power‑law family
        for _ in range(4):
            p = rng.uniform(0.05, 0.45)
            eps = rng.uniform(0.0, max(2.0, 0.02 * n))
            seeds.append(generate_shape(n, p, eps))

        # beta‑like family
        for _ in range(3):
            alpha = rng.uniform(0.2, 5.0)
            beta = rng.uniform(0.2, 5.0)
            seeds.append(beta_shape(n, alpha, beta))

        # random Dirichlet / exponential samples
        for _ in range(3):
            samp = rng.exponential(1.0, size=n)
            seeds.append(normalize(samp))

        # --------------------------------------------------------------
        # 1b. Quick sweep, keep the three most promising seeds.
        # --------------------------------------------------------------
        scored = [(compute_score(s), s) for s in seeds]
        scored.sort()                           # sort by the first tuple element
        top_seeds = [s for _, s in scored[:3]]

        # --------------------------------------------------------------
        # 1c. Refine each retained seed.
        # --------------------------------------------------------------
        for idx, seed_vec in enumerate(top_seeds):
            if time.time() >= deadline - 5.0:
                break
            print(f"  ++ candidate {idx + 1}/{len(top_seeds)} (n={n})")

            # ----- phase‑1 : soft‑max gradient descent -----------------
            refined_soft, _ = optimise_softmax(
                init=seed_vec,
                deadline=deadline,
                lam_start=2.0,
                lam_end=2000.0,
                lam_ramp_iters=40000,
                eta0=0.8,
                max_iter=120_000,
                noise_every=2000,
                noise_scale=5e-6,
            )

            # ----- phase‑2 : hard‑max sub‑gradient fine‑tuning ---------
            refined_hard, _ = hardmax_refine(
                init=refined_soft,
                deadline=deadline,
                eta0=0.3,
                max_iter=120_000,
            )

            # ----- official evaluation --------------------------------
            try:
                val = evaluate_sequence(refined_hard.tolist())
                print(f"    evaluated score = {val:.6f}")
                if val < best_val:
                    best_val = val
                    best_seq = refined_hard.tolist()
                    print(f"    *** New global best: {best_val:.6f} (n={n}) ***")
            except Exception as exc:
                print("    evaluation failed:", exc)

    # ------------------------------------------------------------------
    # 2. Final polishing of the overall best (time permitting).
    # ------------------------------------------------------------------
    if best_seq and time.time() < deadline - 5.0:
        a = np.array(best_seq, dtype=float)
        remaining = deadline - time.time()
        max_iter = int(min(200_000, remaining * 15))

        print("\nFinal polishing of the global best")
        a_refined, _ = optimise_softmax(
            init=a,
            deadline=deadline,
            lam_start=2.0,
            lam_end=3000.0,
            lam_ramp_iters=60000,
            eta0=0.5,
            max_iter=max_iter,
            noise_every=800,
            noise_scale=2e-6,
        )
        final_val = evaluate_sequence(a_refined.tolist())
        print(f"Final polished score = {final_val:.6f}")
        if final_val < best_val:
            best_val = final_val
            best_seq = a_refined.tolist()
            print("*** Updated global best after final polish ***")

    # ------------------------------------------------------------------
    # 3. Guard – always return a feasible sequence.
    # ------------------------------------------------------------------
    if not best_seq:
        best_seq = [1.0] * 256
        best_val = evaluate_sequence(best_seq)

    print(f"\nSearch finished. Best reported score = {best_val:.6f}")
    return best_seq
