"""
Propose a low‑score step function.

The program implements a two‑phase optimiser that works inside the 1000 s
budget:

1. **Projected sub‑gradient descent** on the true (non‑smoothed) objective.
   For a current vector a (∑a=1) we compute the full self‑convolution,
   locate the index k where the convolution attains its maximum and
   construct a sub‑gradient of that entry.  After a small gradient step we
   project back onto the probability simplex.  A few thousand cheap
   iterations are enough to drive the objective well below the uniform
   value 2.0.

2. **Fine‑grained hill‑climbing** – stochastic mass‑transfer moves (swap,
   shift a fraction of probability from one index to another, split or
   merge small packets).  The true objective (via `evaluate_sequence`) is
   used for acceptance, allowing the optimiser to escape the tiny plateaus
   left by the sub‑gradient phase.

The optimiser is run from several random restarts (uniform, Dirichlet
samples with different concentration parameters and the classical arcsine
shape).  Different lengths (1024, 2048, 4096) are tried; the best sequence
found over all runs is returned.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities (no closures – top‑level required by the harness)
# ----------------------------------------------------------------------


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of a vector onto the probability simplex."""
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / v.shape[0])

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, v.shape[0] + 1, dtype=float)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.where(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full linear convolution using a real FFT."""
    m = len(a) + len(b) - 1
    n_fft = next_pow2(m)
    A = np.fft.rfft(a, n=n_fft)
    B = np.fft.rfft(b, n=n_fft)
    C = A * B
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


def compute_score_simplex(a: np.ndarray) -> float:
    """
    Fast replica of `evaluate_sequence` for vectors that already sum to 1
    (the common case during optimisation).  Returns 2·n·max(a*a).
    """
    n = a.shape[0]
    conv = convolution_fft(a, a)
    return 2.0 * n * float(np.max(conv))


def subgradient_of_max_conv(a: np.ndarray, max_idx: int) -> np.ndarray:
    """
    Sub‑gradient of the convolution entry that currently attains the maximum.
    For shift s = max_idx - (n-1) the gradient is:
        g_i = a_{i-s} + a_{i+s}   (indices out of range are ignored)
    If s == 0 this reduces to g = 2·a.
    """
    n = a.shape[0]
    shift = max_idx - (n - 1)
    if shift == 0:
        return 2.0 * a

    g = np.zeros_like(a)
    if shift > 0:
        # i-s is valid for i >= shift
        g[shift:] += a[:-shift]
        # i+s is valid for i < n-shift
        g[: n - shift] += a[shift:]
    else:  # shift < 0
        s = -shift
        g[: n - s] += a[s:]
        g[s:] += a[: n - s]
    return g


def random_dirichlet(n: int, alpha: float) -> np.ndarray:
    """Draw a Dirichlet vector of length n with concentration α."""
    return np.random.dirichlet(np.full(n, alpha))


def arcsine_initial(n: int) -> np.ndarray:
    """Classical arcsine start: a_i ∝ 1 / sqrt((i+½)(n‑½‑i))."""
    i = np.arange(n, dtype=float)
    eps = 1e-6
    vals = 1.0 / np.sqrt((i + 0.5) * (n - 0.5 - i) + eps)
    return project_to_simplex(vals)


# ----------------------------------------------------------------------
# Optimisation phases
# ----------------------------------------------------------------------


def projected_subgradient_opt(
    init_a: np.ndarray,
    deadline: float,
    max_iter: int = 100_000,
    eta0: float = 0.5,
):
    """
    Projected sub‑gradient descent on the true objective (max convolution).
    The vector is always kept on the simplex, therefore the denominator in
    `evaluate_sequence` is 1 and the score is simply 2·n·max(conv).
    """
    a = project_to_simplex(init_a.astype(float).copy())
    best_a = a.copy()
    best_score = compute_score_simplex(a)

    n = len(a)

    for t in range(1, max_iter + 1):
        if time.time() >= deadline:
            break

        # full convolution
        conv = convolution_fft(a, a)
        max_idx = int(np.argmax(conv))
        grad = subgradient_of_max_conv(a, max_idx)

        # simple diminishing step size
        eta = eta0 / np.sqrt(t)
        a = a - eta * grad
        a = project_to_simplex(a)

        cur_score = compute_score_simplex(a)
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            # optional progress message
            if t % 5_000 == 0:
                print(f"    Sub‑grad iter {t:,d} – cur {cur_score:.6f} – best {best_score:.6f}")

    return best_a, best_score


def hill_climb(
    start_seq,
    deadline: float,
    max_steps: int = 200_000,
):
    """
    Stochastic hill‑climber that repeatedly moves a small fraction of mass
    between two random positions (or swaps two entries).  The true
    `evaluate_sequence` score is used for acceptance.
    """
    a = np.array(start_seq, dtype=float)
    n = a.shape[0]

    best_a = a.copy()
    best_score = evaluate_sequence(best_a.tolist())

    step_frac = 0.05  # initial fraction of mass moved in a step

    while max_steps > 0 and time.time() < deadline:
        # pick move type
        if np.random.rand() < 0.5:
            # swap two positions – does not change the sum
            i, j = np.random.randint(n, size=2)
            cand = best_a.copy()
            cand[i], cand[j] = cand[j], cand[i]
        else:
            # move a small fraction from i to j
            i, j = np.random.randint(n, size=2)
            if i == j:
                continue
            delta = step_frac * best_a[i]
            if delta <= 0.0:
                continue
            cand = best_a.copy()
            cand[i] -= delta
            cand[j] += delta
            # safeguard against tiny negatives caused by rounding
            if cand[i] < 0.0:
                cand[i] = 0.0

        # ensure feasibility
        cand = project_to_simplex(cand)

        cur_score = evaluate_sequence(cand.tolist())
        if cur_score < best_score:
            best_score = cur_score
            best_a = cand
            step_frac = min(step_frac * 1.03, 0.4)  # grow slowly when successful
        else:
            step_frac *= 0.97  # shrink when rejected

        max_steps -= 1

    return best_a.tolist(), best_score
