"""
Propose a sequence of non‑negative heights that minimizes ``evaluate_sequence``.
Algorithm (≈1000 s budget)

1. Initialise a probability vector a (sum == 1).  Use the previously known
   ``height_sequence_1`` if it exists, otherwise start from a decreasing
   exponential distribution.
2. **Smoothed gradient descent** (≈10–15 % of the budget):
   minimise the log‑sum‑exp surrogate of   max(a*a)  on the simplex using
   Adam and Euclidean projection.  The gradient is computed by a single
   FFT‑based cross‑correlation, so each iteration is cheap.
3. **Fast simulated‑annealing** (the remaining time):
   – keep the current convolution `conv = a*a` exactly (updated incrementally
     after each mass‑transfer move);
   – propose moves that transfer a random fraction of the mass from a source
     index to a target index;
   – accept with a Metropolis rule using a slowly cooling temperature;
   – every 200 k moves recompute the convolution from scratch to avoid
     accumulated floating‑point drift and optionally perform a random
     restart.
4. Return the best vector seen (renormalised to sum == 1).

The implementation avoids any file I/O, uses only NumPy and the provided
``evaluate_sequence`` function, and prints progress information.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities
# ----------------------------------------------------------------------


def simplex_projection(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the probability simplex {
    w | w_i ≥ 0 , Σ w_i = 1 }.
    Duchi et al. (2008) algorithm.
    """
    v = np.asarray(v, dtype=float)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1] - 1          # zero‑based
    theta = cssv[cond][-1] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def compute_score_from_conv(conv: np.ndarray, n: int) -> float:
    """Score = 2·n·max(conv)  (sum of a is 1)."""
    return 2.0 * n * conv.max()


def fft_convolve(a: np.ndarray, b: np.ndarray, out_len: int) -> np.ndarray:
    """Linear convolution via FFT, result truncated to length out_len."""
    L = 1 << ((len(a) + len(b) - 1).bit_length())
    fa = np.fft.rfft(a, n=L)
    fb = np.fft.rfft(b, n=L)
    conv = np.fft.irfft(fa * fb, n=L)
    return conv[:out_len]


def softmax(x: np.ndarray, temperature: float) -> np.ndarray:
    """Numerically stable softmax with temperature > 0."""
    if temperature <= 0:
        temperature = 1e-12
    shift = x.max()
    ex = np.exp((x - shift) / temperature)
    s = ex.sum()
    if s == 0:
        return np.full_like(x, 1.0 / x.size)
    return ex / s

# Main optimiser
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 0, budget_s: int = 1000) -> list[float]:
    """
    Search for a non‑negative height sequence that minimises
    ``evaluate_sequence``.  The function runs for at most ``budget_s`` seconds
    and returns the best vector found (as a plain Python list of floats).
    """
    np.random.seed(seed)
    start = time.time()
    deadline = start + budget_s - 0.5        # safety margin

    # --------------------------------------------------------------
    # 1. Initialisation
    # --------------------------------------------------------------
    try:
        # use the sequence left over from a previous run if present
        init = np.array(height_sequence_1, dtype=float)    # type: ignore[name-defined]
        n = init.shape[0]
        print(f"Using existing height_sequence_1 (n={n}).")
    except NameError:
        # fallback – decreasing exponential shape, length ≈5000
        n = 5146
        lam = 0.0015                # controls how fast the tail decays
        idx = np.arange(n, dtype=float)
        init = np.exp(-lam * idx)
        print(f"No existing sequence – using exponential start (n={n}).")

    a = init / init.sum()          # normalised probability vector
    best_a = a.copy()
    best_score = math.inf

    # --------------------------------------------------------------
    # Helper: recompute the full self‑convolution (exact)
    # --------------------------------------------------------------
    def recompute_conv(vec: np.ndarray) -> np.ndarray:
        """Return a* a (linear convolution, length 2n‑1)."""
        L = 1 << ((2 * n - 1).bit_length())
        fa = np.fft.rfft(vec, n=L)
        conv = np.fft.irfft(fa * fa, n=L)[: 2 * n - 1]
        return conv

    # initial convolution and score
    conv = recompute_conv(a)
    cur_score = compute_score_from_conv(conv, n)
    best_score = cur_score
    best_a = a.copy()
    print(f"Initial score = {best_score:.6f}")

    # --------------------------------------------------------------
    # 2. Smoothed gradient descent (log‑sum‑exp surrogate)
    #    – run while we still have ≈15 % of the time left for SA
    # --------------------------------------------------------------
    # Adam hyper‑parameters
    lr = 0.5
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8
    m = np.zeros_like(a)
    v = np.zeros_like(a)

    # temperature for the soft‑max approximation
    temp_factor_start = 0.3           # start with 30 % of max(conv)
    temp_factor_decay = 0.9999        # geometric decay per iteration
    T = conv.max() * temp_factor_start

    max_iter_grad = 50000            # hard cap; we also stop early on time
    it = 0
    while time.time() < deadline - 30.0 and it < max_iter_grad:
        # recompute convolution (needed for both max and for the softmax)
        conv = recompute_conv(a)
        max_c = conv.max()
        # update temperature (never let it become too small)
        T = max(max_c * 1e-8, T * temp_factor_decay)

        # soft‑max weights on the convolution entries
        w = softmax(conv, T)

        # --------------------------------------------------------
        # Gradient of the smoothed objective:
        #   g = 2 * (a  ⋆  w)   where ⋆ denotes cross‑correlation
        #   we compute it via FFT:  g = 2 * corr(a, w)
        # --------------------------------------------------------
        L_corr = 1 << ((3 * n - 2).bit_length())
        fa = np.fft.rfft(a, n=L_corr)
        fw = np.fft.rfft(w, n=L_corr)
        corr_full = np.fft.irfft(fa * np.conj(fw), n=L_corr)
        offset = len(w) - 1                      # index of lag‑0
        grad = 2.0 * corr_full[offset: offset + n]

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)
        m_hat = m / (1.0 - beta1 ** (it + 1))
        v_hat = v / (1.0 - beta2 ** (it + 1))
        step = lr * m_hat / (np.sqrt(v_hat) + eps)

        a = a - step
        # projection onto the simplex (ensures non‑negativity & sum=1)
        a = simplex_projection(a)

        # evaluate current score (cheap, no extra FFT)
        cur_score = compute_score_from_conv(conv, n)
        if cur_score < best_score:
            best_score, best_a = cur_score, a.copy()
            if it % 500 == 0:
                print(f"Grad iter {it:5d}: new best {best_score:.6f}")

        it += 1

    print(f"Gradient phase finished after {it} iterations; best score = {best_score:.6f}")

    # --------------------------------------------------------------
    # 3. Fast simulated annealing with incremental convolution updates
    # --------------------------------------------------------------
    cur_a = a.copy()
    conv = recompute_conv(cur_a)          # ensure consistency after GD
    cur_score = compute_score_from_conv(conv, n)

    # temperature schedule for SA
    T_start = cur_score * 0.05            # start relatively low (already near optimum)
    T_end = cur_score * 1e-6
    cooling = 0.99997                     # exponential cooling factor

    it_sa = 0
    # enter SA loop
    while time.time() < deadline:
        # current temperature
        T = max(T_end, T_start * (cooling ** it_sa))

        # ------------------------------------------------------------------
        # Propose a mass‑transfer move (keeps Σ a_i = 1)
        # ------------------------------------------------------------------
        # source – pick proportionally to current mass (more mass = higher chance)
        src = np.random.choice(n, p=cur_a / cur_a.sum())
        # target – random distinct index
        tgt = np.random.randint(n - 1)
        if tgt >= src:
            tgt += 1

        if cur_a[src] <= 0.0:
            continue

        # transfer a random fraction (bias towards small fractions)
        frac = np.random.rand() ** 2
        delta = cur_a[src] * frac
        if delta == 0.0:
            continue

        # ------------------------------------------------------------------
        # Incremental update of the convolution (exact)
        # ------------------------------------------------------------------
        # Remember original a for the update (cur_a is still unchanged)
        a_orig = cur_a

        # Apply the linear part: 2·δ·(a_{k‑tgt} – a_{k‑src})
        conv[tgt: tgt + n] += 2.0 * delta * a_orig
        conv[src: src + n] -= 2.0 * delta * a_orig

        # Quadratic δ² contributions
        conv[2 * src] += delta * delta
        conv[2 * tgt] += delta * delta
        conv[src + tgt] -= 2.0 * delta * delta

        # New score after the tentative move
        new_score = compute_score_from_conv(conv, n)

        # Metropolis acceptance
        accept = False
        if new_score < cur_score:
            accept = True
        else:
            prob = math.exp(-(new_score - cur_score) / max(T, 1e-12))
            if np.random.rand() < prob:
                accept = True

        if accept:
            # Commit the move
            cur_a[src] -= delta
            cur_a[tgt] += delta
            cur_score = new_score
            # Check for improvement
            if new_score < best_score:
                best_score, best_a = new_score, cur_a.copy()
                print(f"SA iter {it_sa:7d}: new best {best_score:.6f}")
        else:
            # Revert convolution (a unchanged)
            conv[tgt: tgt + n] -= 2.0 * delta * a_orig
            conv[src: src + n] += 2.0 * delta * a_orig
            conv[2 * src] -= delta * delta
            conv[2 * tgt] -= delta * delta
            conv[src + tgt] += 2.0 * delta * delta
            # cur_a stays the same

        # Periodically recompute the convolution from scratch to avoid drift
        if it_sa % 200_000 == 0 and it_sa > 0:
            conv = recompute_conv(cur_a)
            cur_score = compute_score_from_conv(conv, n)

        # Every 300 k iterations try a random restart (helps escape plateaus)
        if it_sa % 300_000 == 0 and it_sa > 0:
            cur_a = np.random.dirichlet(np.ones(n))
            conv = recompute_conv(cur_a)
            cur_score = compute_score_from_conv(conv, n)

        it_sa += 1

    print(f"Simulated annealing finished after {it_sa} iterations; final best score = {best_score:.6f}")

    # --------------------------------------------------------------
    # 4. Return the best sequence (already normalised)
    # --------------------------------------------------------------
    best_a = best_a / best_a.sum()
    return [float(v) for v in best_a.tolist()]
