"""
Propose a height sequence that minimises ``evaluate_sequence``.

The algorithm has three stages:

1. **Smoothed exponentiated gradient descent** –  
   The nonsmooth objective `max_k (a*a)_k` is approximated by a
   log‑sum‑exp (soft‑max).  The resulting smooth gradient is
   computed by FFTs and an exponentiated‑gradient update keeps the
   iterate on the probability simplex.

2. **Guided simulated annealing** –  
   After the fast descent we refine the best point.  In each move we
   identify the convolution index that currently attains the maximum,
   pick a pair of indices that contributes most to this peak and
   transfer a random fraction of mass from the “worst’’ index to a
   random target.  A Metropolis acceptance rule with an exponential
   cooling schedule is used.

3. **Return the best observed sequence** – it is normalised to sum = 1
   (the evaluation function is scale‑invariant).

The whole search respects the given time budget (default 1000 s) and
uses only NumPy, no external I/O.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities
# ----------------------------------------------------------------------


def simplex_projection(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the probability simplex:
    {w | w_i ≥ 0 , Σ w_i = 1}.
    Duchi et al., 2008.
    """
    v = np.asarray(v, dtype=float)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1] - 1
    theta = cssv[cond][-1] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def beta_sequence(alpha: float, beta: float, n: int) -> np.ndarray:
    """
    Discretised Beta(α,β) density on ``n`` points, normalised to sum = 1.
    """
    x = (np.arange(n) + 0.5) / n
    logw = (alpha - 1.0) * np.log(x) + (beta - 1.0) * np.log(1.0 - x)
    logw = np.clip(logw, -700.0, 700.0)          # avoid overflow
    w = np.exp(logw)
    if not np.isfinite(w.sum()) or w.sum() == 0.0:
        w = np.ones_like(w)
    return w / w.sum()


def compute_score(a: np.ndarray) -> float:
    """
    Exact value of ``evaluate_sequence`` for a vector that already satisfies
    Σ a_i = 1.  The formula reduces to 2·n·max(conv(a,a)).
    """
    n = a.shape[0]
    fft = np.fft.rfft(a, n=2 * n - 1)
    conv = np.fft.irfft(fft * fft, n=2 * n - 1)
    return 2.0 * n * conv.max()

    deadline = start_time + budget_s - 0.5          # safety margin

    # --------------------------------------------------------------
    # 0. Choose problem size and initialise a feasible point
    # --------------------------------------------------------------
    try:
        init = np.array(height_sequence_1, dtype=float)   # type: ignore[name-defined]
        n = init.shape[0]
        print(f"Using existing height_sequence_1 of length {n}.")
    except NameError:
        n = 5146
        init = beta_sequence(2.5, 2.5, n)
        print(f"No previous sequence – using a beta start, length {n}.")

    a = init / init.sum()               # enforce Σ a_i = 1
    best_a = a.copy()
    best_score = compute_score(best_a)
    print(f"Initial score = {best_score:.6f}")

    # --------------------------------------------------------------
    # 1. Smoothed exponentiated gradient descent (soft‑max surrogate)
    # --------------------------------------------------------------
    # FFT length (next power of two ≥ 3·n) – enough for all convolutions
    L = 1 << int(np.ceil(np.log2(3 * n)))
    alpha = 1.0          # soft‑max temperature (small → smooth)
    eta = 0.5            # learning‑rate for exponentiated update
    iter_gd = 0

    # reserve ~30 s for the final annealing phase
    while time.time() < deadline - 30.0:
        # ----- convolution a*a  (full length 2n‑1) -----
        a_fft = np.fft.rfft(a, n=L)
        conv_full = np.fft.irfft(a_fft * a_fft, n=L)
        conv = conv_full[: 2 * n - 1]

        # ----- soft‑max weights -----
        max_c = conv.max()
        # exponentiate a shifted version to avoid overflow
        shift = alpha * (conv - max_c)          # ≤ 0
        # further shift to keep numbers moderate
        shift -= shift.max()
        w = np.exp(shift)
        w_sum = w.sum()
        w_norm = w / w_sum                      # ∑ w_k = 1  (softmax)

        # ----- gradient of the smooth objective -----
        # grad_i = 2 * Σ_k w_k * a_{k-i}  (where the index is valid)
        # This is a correlation of w_norm with a; we compute it via FFT.
        w_rev = w_norm[::-1]                    # reverse for correlation
        w_rev_fft = np.fft.rfft(w_rev, n=L)
        grad_full = np.fft.irfft(w_rev_fft * a_fft, n=L)
        grad = 2.0 * grad_full[n - 1 : n - 1 + n]   # valid part, length n

        # ----- exponentiated gradient step (keeps a on the simplex) -----
        # small clipping prevents overflow in the exponential
        log_update = -eta * grad
        log_update = np.clip(log_update, -50.0, 50.0)
        a = a * np.exp(log_update)
        a_sum = a.sum()
        if a_sum == 0.0:                 # numerical safety – fall back to uniform
            a = np.full(n, 1.0 / n)
        else:
            a /= a_sum

        # ----- bookkeeping -----
        cur_score = compute_score(a)
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"GD iter {iter_gd:6d}: new best {best_score:.6f}")

        # ----- schedules for the hyper‑parameters -----
        iter_gd += 1
        eta = max(0.02, eta * 0.9995)          # slowly decay learning rate
        if iter_gd % 500 == 0:
            alpha = min(2000.0, alpha * 1.02)  # make the surrogate sharper

    print(f"Gradient phase finished after {iter_gd} iterations; "
          f"best score so far {best_score:.6f}")

    # --------------------------------------------------------------
    # 2. Guided simulated‑annealing fine‑tuning
    # --------------------------------------------------------------
    cur_a = best_a.copy()
    cur_score = best_score
    # temperature schedule (geometric cooling)
    T_start = cur_score * 0.02
    T_end = cur_score * 1e-5
    it_sa = 0

    while time.time() < deadline:
        # recompute convolution for the current point
        a_fft = np.fft.rfft(cur_a, n=L)
        conv_full = np.fft.irfft(a_fft * a_fft, n=L)
        conv = conv_full[: 2 * n - 1]
        k_max = int(conv.argmax())                     # most offending shift

        # ----- choose a source index that contributes most to the peak -----
        lo = max(0, k_max - n + 1)
        hi = min(n - 1, k_max)
        idxs = np.arange(lo, hi + 1, dtype=int)
        partners = k_max - idxs
        contributions = cur_a[idxs] * cur_a[partners]
        src = idxs[np.argmax(contributions)]

        # ----- pick a (different) target index -----
        tgt = np.random.randint(n - 1)
        if tgt >= src:
            tgt += 1

        # ----- propose a mass‑transfer move -----
        if cur_a[src] <= 0.0:
            continue
        delta = cur_a[src] * np.random.rand() * 0.5   # up to 50 % of src
        cand = cur_a.copy()
        cand[src] -= delta
        cand[tgt] += delta

        # ----- evaluate candidate -----
        cand_score = compute_score(cand)

        # ----- Metropolis acceptance -----
        if cand_score < cur_score:
            accept = True
        else:
            T = max(T_end, T_start * (0.9995 ** it_sa))
            accept_prob = math.exp(-(cand_score - cur_score) / max(T, 1e-12))
            accept = np.random.rand() < accept_prob

        if accept:
            cur_a = cand
            cur_score = cand_score
            if cand_score < best_score:
                best_score = cand_score
                best_a = cand.copy()
                print(f"SA iter {it_sa:7d}: new best {best_score:.6f}")

        # ----- occasional random restart to avoid stagnation -----
        if it_sa % 50000 == 0 and it_sa > 0:
            tmp = beta_sequence(
                np.random.uniform(0.5, 5.0),
                np.random.uniform(0.5, 5.0),
                n,
            )
            cur_a = 0.9 * cur_a + 0.1 * tmp
            cur_a /= cur_a.sum()
            cur_score = compute_score(cur_a)

        it_sa += 1

    print(f"Simulated annealing finished after {it_sa} iterations; "
          f"final best score = {best_score:.6f}")

    # --------------------------------------------------------------
    # 3. Return the best sequence (already normalised to Σ = 1)
    # --------------------------------------------------------------
    best_a = best_a / best_a.sum()
    return [float(v) for v in best_a.tolist()]
