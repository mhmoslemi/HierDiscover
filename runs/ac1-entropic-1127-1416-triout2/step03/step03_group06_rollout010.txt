    w_exp = np.exp((conv - max_c) / eps)
    w = w_exp / w_exp.sum()
    # correlation
    corr = correlation_w_a(w, a)
    # diag correction
    idx = 2 * np.arange(n)
    w_diag = w[idx]
    grad = 2.0 * corr - w_diag * a
    # gradient of objective O = 2*n * max_c => factor 2*n
    grad = 2.0 * n * grad
    return grad, max_c

"""
Optimisation of a non‑negative step function for the evaluation function.

The algorithm builds a diverse set of seed vectors (uniform, ramps,
exponentials, Beta‑distributions, Dirichlet samples and the known
`height_sequence_1`).  From the best few seeds it launches a
projected gradient descent on the probability simplex that minimises a
smooth approximation of the maximum convolution entry.
The smooth max is realised by a log‑sum‑exp (temperature ε) whose
gradient can be expressed analytically via a correlation of the weight
vector with the current sequence.
A simple additive step with a decreasing step‑size is used, followed
by a Euclidean projection onto the simplex (Duchi et al. 2008).  If the
search stagnates a small random perturbation is applied to escape
plateaux.  The best sequence found before the time budget expires is
returned as a plain ``list[float]``.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Helper functions (top level – no closures)
# ----------------------------------------------------------------------


def simplex_projection(v):
    """
    Euclidean projection of a vector onto the probability simplex:
    { w | w_i ≥ 0 , Σ w_i = 1 }.
    Implementation follows Duchi et al. (2008).
    """
    v = np.asarray(v, dtype=float)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1] - 1
    theta = cssv[cond][-1] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def compute_objective(a):
    """
    Return 2·n·max(a∗a) assuming Σ a_i = 1.
    This matches the value returned by ``evaluate_sequence`` after the
    normalisation step.
    """
    n = a.shape[0]
    fft_a = np.fft.rfft(a, n=2 * n - 1)
    conv = np.fft.irfft(fft_a * fft_a, n=2 * n - 1)
    return 2.0 * n * conv.max()


def beta_sequence(alpha, beta, n):
    """Discretised Beta(α,β) density on ``n`` mid‑points, normalised to sum 1."""
    x = (np.arange(n) + 0.5) / n
    logw = (alpha - 1.0) * np.log(x) + (beta - 1.0) * np.log(1.0 - x)
    logw = np.clip(logw, -700.0, 700.0)          # avoid overflow in exp
    w = np.exp(logw)
    total = w.sum()
    if total == 0 or not np.isfinite(total):
        w = np.ones_like(w)
        total = w.sum()
    return w / total


def exponential_shape(lam, n):
    """a_i ∝ exp(‑lam·i)  (lam ≥ 0), normalised to sum 1."""
    i = np.arange(n, dtype=float)
    w = np.exp(-lam * i)
    w /= w.sum()
    return w


def polynomial_shape(p, n):
    """a_i ∝ 1 - (i/(n‑1))**p  (p ≥ 0), clipped to non‑negative, normalised."""
    i = np.arange(n, dtype=float)
    frac = i / (n - 1)
    w = 1.0 - np.power(frac, p)
    w = np.clip(w, 0.0, None)
    w /= w.sum()
    return w


def correlation_w_a(w, a):
    """
    Compute the correlation   corr[i] = Σ_k w[k]·a[k‑i]   for i=0..n‑1.
    Implemented via FFT (zero‑padded convolution with the reversed a).
    """
    n = a.shape[0]
    a_rev = a[::-1]
    # length needed for a full linear convolution of w (len 2n‑1) and a_rev (len n)
    target_len = len(w) + n - 1                     # = 3·n‑2
    n_fft = 1 << (target_len - 1).bit_length()     # next power of two
    fft_w = np.fft.rfft(w, n=n_fft)
    fft_a_rev = np.fft.rfft(a_rev, n=n_fft)
    conv = np.fft.irfft(fft_w * fft_a_rev, n=n_fft)
    start = n - 1
    return conv[start:start + n]                    # length n


def gradient_smooth_max(a, eps_factor):
    """
    Smooth approximation of the gradient of 2·n·max(a∗a)
    using a log‑sum‑exp with temperature ε = eps_factor·max_c.
    Returns (gradient, current_max_convolution).
    """
    n = a.shape[0]
    # full auto‑convolution
    fft_a = np.fft.rfft(a, n=2 * n - 1)
    conv = np.fft.irfft(fft_a * fft_a, n=2 * n - 1)
    max_c = conv.max()
    # temperature
    eps = max_c * eps_factor
    if eps < 1e-15:
        eps = max_c * 1e-3
    # smooth weights – shifted by max to keep numbers ≤ 1
    w_exp = np.exp((conv - max_c) / eps)
    w = w_exp / w_exp.sum()
    # correlation needed for the gradient
    corr = correlation_w_a(w, a)          # length n
    # diagonal correction (indices 0,2,4,…,2n‑2)
    diag_idx = 2 * np.arange(n, dtype=int)
    w_diag = w[diag_idx]
    grad = 2.0 * corr - w_diag * a          # ∂M/∂a (M = smooth max)
    grad = 2.0 * n * grad                  # ∂(2·n·M)/∂a
    return grad, max_c


# ----------------------------------------------------------------------
# Main search routine
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 0, budget_s: int = 1000):
    """
    Search for a non‑negative step‑function (list of floats) that minimises
    ``evaluate_sequence``.  The routine respects a ~1000 s wall‑clock limit
    and returns the best sequence discovered.
    """
    np.random.seed(seed)
    start_time = time.time()
    # keep a safety margin so the outer harness does not cut us off
    deadline = start_time + budget_s - 5.0

    # --------------------------------------------------------------
    # 1. Determine problem size (length of the step function)
    # --------------------------------------------------------------
    try:
        n = len(height_sequence_1)          # type: ignore[name-defined]
    except NameError:
        n = 5146                            # fallback used in earlier runs

    # --------------------------------------------------------------
    # 2. Helpers to keep the global best and a small top‑K list
    # --------------------------------------------------------------
    best_seq = None
    best_score = np.inf
    TOP_K = 5
    top = []                               # list of (score, seq)

    def consider(candidate):
        """Evaluate *candidate* and update global / top‑K if it improves."""
        nonlocal best_seq, best_score, top
        s = compute_objective(candidate)
        if s < best_score:
            best_score = s
            best_seq = candidate.copy()
            print(f"Initial global best: {s:.6f}")
        if len(top) < TOP_K:
            top.append((s, candidate.copy()))
            top.sort(key=lambda x: x[0])
        else:
            if s < top[-1][0]:
                top[-1] = (s, candidate.copy())
                top.sort(key=lambda x: x[0])

    # --------------------------------------------------------------
    # 3. Generate deterministic and random seed families
    # --------------------------------------------------------------

    # uniform
    consider(np.full(n, 1.0 / n))

    # linear decreasing ramp
    ramp = np.arange(n, 0, -1, dtype=float)
    ramp = ramp / ramp.sum()
    consider(ramp)

    # exponential family
    for lam in [0.01, 0.1, 0.5, 1.0, 2.0, 5.0]:
        consider(exponential_shape(lam, n))

    # polynomial family
    for p in [0.5, 1.0, 2.0, 3.0, 5.0]:
        consider(polynomial_shape(p, n))

    # coarse Beta sweep
    alphas = np.linspace(0.2, 5.0, 6)
    betas = np.linspace(0.2, 5.0, 6)
    for α in alphas:
        for β in betas:
            if time.time() >= deadline:
                break
            consider(beta_sequence(α, β, n))
        if time.time() >= deadline:
            break

    # random Dirichlet‑like samples
    for _ in range(40):
        if time.time() >= deadline:
            break
        conc = np.random.uniform(0.1, 3.0, size=n)
        a_dir = np.random.gamma(conc, 1.0)
        a_dir /= a_dir.sum()
        consider(a_dir)

    # include the previously known construction if available
    try:
        consider(np.array(height_sequence_1, dtype=float) / np.sum(height_sequence_1))
    except NameError:
        pass

    # --------------------------------------------------------------
    # 4. Initialise the optimisation from the best seed found
    # --------------------------------------------------------------
    a = best_seq.copy()                     # current iterate (sum == 1)
    best_score = compute_objective(a)
    best_seq = a.copy()
    print(f"Start optimisation, initial score = {best_score:.6f}")

    # --------------------------------------------------------------
    # 5. Smooth‑max projected gradient descent (with occasional perturb.)
    # --------------------------------------------------------------
    eps_factor = 0.20          # initial temperature as fraction of max_c
    eps_decay = 0.9995         # geometric decay each iteration
    eps_min = 1e-5

    base_step = 0.2            # additive step size before schedule
    iteration = 0
    stagnation = 0
    max_iter = 10_000_000      # hard upper bound (unlikely to be reached)

    while time.time() < deadline and iteration < max_iter:
        # gradient of the smoothed objective and the exact max‑convolution value
        grad, max_c = gradient_smooth_max(a, eps_factor)
        cur_score = 2.0 * n * max_c

        # keep the best ever seen
        if cur_score < best_score - 1e-12:
            best_score = cur_score
            best_seq = a.copy()
            print(f"Iter {iteration:6d}: new best {best_score:.6f}")
            stagnation = 0
        else:
            stagnation += 1

        # step‑size schedule (1/√t)
        step = base_step / math.sqrt(iteration + 1)

        # additive update + projection onto the simplex
        a_next = a - step * grad
        a = simplex_projection(a_next)

        # temperature schedule
        eps_factor = max(eps_factor * eps_decay, eps_min)

        iteration += 1

        # occasional random perturbation to escape plateaus
        if stagnation >= 60_000:
            sigma = max(0.01, 0.5 / math.sqrt(iteration + 1))
            noise = sigma * np.random.randn(n) * a
            a = a + noise
            a = np.clip(a, 0.0, None)
            a /= a.sum()
            stagnation = 0
            print(f"Iter {iteration:6d}: perturbation applied, score {cur_score:.6f}")

        # early exit if target reached
        if best_score <= 1.503:
            print(f"Target achieved at iteration {iteration}, score = {best_score:.6f}")
            break

    # --------------------------------------------------------------
    # 6. Return the best sequence discovered (as plain Python list)
    # --------------------------------------------------------------
    print(f"Finished after {iteration} iterations, best score = {best_score:.6f}")
    return [float(v) for v in best_seq.tolist()]
