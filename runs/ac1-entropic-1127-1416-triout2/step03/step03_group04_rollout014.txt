def mix_sequences(seqs, weights):
    """Combine sequences (lists/arrays) according to weights and normalize."""
    combined = np.zeros_like(seqs[0])
    for seq,w in zip(seqs, weights):
        combined += w * np.asarray(seq)
    # ensure nonneg
    combined = np.clip(combined, 0, None)
    total = combined.sum()
    if total == 0:
        combined = np.full_like(combined, 1.0/len(combined))
    else:
        combined = combined / total
    return combined

def lp_refine(seq, deadline, max_iters=5, t=0.2):
    try:
        from scipy.optimize import linprog
    except Exception as e:
        print("scipy not available; skipping LP refinement")
        return seq
    a = np.array(seq, dtype=float)
    # Ensure sum = 1
    a = a / a.sum()
    n = len(a)
    # Pre compute full target sum sqrt(2n) for scaling
    # Not needed; we keep scale 1.
    best_a = a.copy()
    best_score = compute_score(a)
    it = 0
    while time.time() < deadline - 1 and it < max_iters:
        it += 1
        # compute max convolution
        conv = convolution_fft(a, a)
        M = np.max(conv)
        # Build A matrix for constraints
        # Use dense for simplicity
        rows = 2*n - 1
        A = np.zeros((rows, n), dtype=float)
        for k in range(rows):
            i_min = max(0, k-(n-1))
            i_max = min(k, n-1)
            # j: i_min..i_max
            # Corresponding f[i] = a[i]; index j = k - i
            # For each j in i_min..i_max:
            # A[k, j] = a[k-j]
            # So we set A[k, i_range] = a[i_range'] maybe simple vectorized
            # We'll loop
            for j in range(i_min, i_max+1):
                i = k - j
                A[k, j] = a[i]
        # Solve LP: maximize sum(b) -> minimize -sum(b)
        c = -np.ones(n)
        b_ub = np.full(rows, M)
        bounds = [(0, None)] * n
        # Use dense method 'highs'
        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
        if not res.success:
            print(f"LP iteration {it} failed: {res.message}")
            break
        b = res.x
        # Scale b to have sum=1 (like a)
        if b.sum() == 0:
            b_norm = np.full_like(b, 1.0/n)
        else:
            b_norm = b / b.sum()
        # Update a as convex combination
        a = (1 - t) * a + t * b_norm
        # Ensure numeric positivity
        a = np.clip(a, 0, None)
        a = a / a.sum()
        cur_score = compute_score(a)
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"LP refine iter {it} improved score {cur_score:.6f}")
        else:
            # maybe reduce t
            t *= 0.8
    return best_a.tolist()

def improve_sequence_smoothed2(initial_a, time_deadline, max_iter=20000, eta0=0.5, alpha_start=5., alpha_end=300., noise_every=500, noise_scale=1e-5):
    a = project_to_simplex(initial_a.copy())
    best_a = a.copy()
    best_score = compute_score(a)
    n = len(a)
    for t in range(1, max_iter+1):
        if time.time() >= time_deadline:
            break
        # compute smoothing parameter
        alpha = alpha_start + (alpha_end - alpha_start) * min(1.0, t / max_iter)
        # compute grad smoothed
        grad, conv = smoothed_gradient(a, alpha)
        # optional scale step per iteration
        eta = eta0 / np.sqrt(t)
        a_new = a - eta * grad
        # noise
        if noise_every and (t % noise_every == 0):
            a_new += np.random.randn(n) * noise_scale
        a_new = project_to_simplex(a_new)
        cur_score = compute_score(a_new)
        if cur_score < best_score:
            best_score = cur_score
            best_a = a_new.copy()
            # optional print
        a = a_new
        if t % 2000 == 0:
            print(f" GD iter {t} cur {cur_score:.6f} best {best_score:.6f}")
    return best_a, best_score

"""
Search for a non‑negative step function that minimises

    E(a) = 2·n·max(a∗a) / (∑a)²

The algorithm works in several stages (all respecting a 1000 s budget).

1. **Simple parametric sweep** – arcsine, uniform, beta‑type families,
   exponential decay, and a dense random mixture of a few components.
   Every candidate is evaluated with a fast FFT‑based convolution and the
   best one is kept.

2. **Optional linear‑program refinement** – if SciPy is available we solve
   a small LP that maximises the ℓ₁‑norm of a new vector `b` while keeping
   the convolution bound `f∗b ≤ M`.  The solution is mixed with the current
   best vector (convex combination) and iterated a few times.

3. **Upsampling** – the best candidate is linearly interpolated to a larger
   length (default 8192).  A finer discretisation usually lowers the bound.

4. **Projected gradient descent on a smoothed max** – we replace the hard
   max by a log‑sum‑exp (temperature α) and perform projected sub‑gradient
   descent on the simplex.  The gradient of the smoothed objective is
   computed analytically (no ad‑hoc “‑ w_diag * a” term).

5. **Final cheap stochastic hill‑climb** – pair‑wise mass transfers that
   improve the true unsmoothed objective are accepted.  This acts as a
   polishing step.

The whole search stops before the budget expires and returns the best
sequence found (as a plain list of floats).  The returned sequence is
normalised to sum = 1, which is scale‑invariant for `evaluate_sequence`.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Linear‑algebra utilities
# ----------------------------------------------------------------------


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    if np.all(v <= 0):
        n = v.shape[0]
        return np.full_like(v, 1.0 / n, dtype=float)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho = np.where(u - cssv / np.arange(1, v.shape[0] + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full convolution using a real FFT (fast)."""
    m = len(a) + len(b) - 1
    n_fft = next_pow2(m)
    A = np.fft.rfft(a, n=n_fft)
    B = np.fft.rfft(b, n=n_fft)
    C = A * B
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


def compute_score(a: np.ndarray) -> float:
    """Score identical to `evaluate_sequence` when ∑a = 1."""
    a = np.asarray(a, dtype=float)
    n = len(a)
    conv = convolution_fft(a, a)
    max_val = float(np.max(conv))
    s = float(np.sum(a))
    if s < 1e-12:
        return float("inf")
    return 2.0 * n * max_val / (s * s)


# ----------------------------------------------------------------------
# Families of candidate step‑functions
# ----------------------------------------------------------------------


def arcsine_initial(n: int) -> np.ndarray:
    """Arcsine‑shaped start: a_i ∝ 1/√((i+½)(n‑½‑i))."""
    i = np.arange(n, dtype=float)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - 0.5 - i))
    return vals / np.sum(vals)


def beta_sequence(n: int, alpha: float, beta: float | None = None) -> np.ndarray:
    """Beta‑type coefficients a_i ∝ (i+½)^α·(n‑½‑i)^β."""
    if beta is None:
        beta = alpha
    i = np.arange(n, dtype=float)
    left = np.power(i + 0.5, alpha)
    right = np.power(n - 0.5 - i, beta)
    seq = left * right
    seq = np.clip(seq, 0.0, 1e12)
    total = np.sum(seq)
    if total == 0.0:
        seq = np.full(n, 1.0 / n)
    else:
        seq = seq / total
    return seq


def exponential_sequence(n: int, lam: float) -> np.ndarray:
    """a_i ∝ exp(-lam·|i - (n‑1)/2|)."""
    i = np.arange(n, dtype=float)
    center = (n - 1) / 2.0
    seq = np.exp(-lam * np.abs(i - center))
    return seq / np.sum(seq)


def upsample_sequence(seq: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation of a probability vector to a larger size."""
    old_n = len(seq)
    old_x = (np.arange(old_n) + 0.5) / old_n
    new_x = (np.arange(new_n) + 0.5) / new_n
    new_vals = np.interp(new_x, old_x, seq)
    return new_vals / np.sum(new_vals)


# ----------------------------------------------------------------------
# Smoothed‑max gradient utilities
# ----------------------------------------------------------------------


def smoothed_gradient(a: np.ndarray, alpha: float):
    """
    Gradient of the smoothed maximum
        f_α(a) = (1/α)·log Σ_k exp(α·(a∗a)_k)

    Returns the gradient (same shape as a) and the raw self‑convolution.
    """
    conv = convolution_fft(a, a)                     # length 2n‑1
    maxc = np.max(conv)
    exp_term = np.exp(alpha * (conv - maxc))
    w = exp_term / np.sum(exp_term)                  # soft‑max weights, Σw = 1
    a_rev = a[::-1]                                 # reversed a
    conv_rev = convolution_fft(w, a_rev)            # length 3n‑2
    start = len(a) - 1
    grad = 2.0 * conv_rev[start:start + len(a)]      # ∂/∂a of the smooth max
    return grad, conv


def improve_sequence_smoothed(
    a0: np.ndarray,
    deadline: float,
    max_iter: int = 12000,
    eta0: float = 0.5,
    alpha_start: float = 5.0,
    alpha_end: float = 250.0,
    noise_every: int = 800,
    noise_scale: float = 1e-5,
):
    """
    Projected sub‑gradient descent on the log‑sum‑exp surrogate.
    Keeps the best (true) score encountered.
    """
    a = project_to_simplex(a0.copy())
    best_a = a.copy()
    best_score = compute_score(a)

    n = len(a)
    for t in range(1, max_iter + 1):
        if time.time() >= deadline:
            break

        # temperature schedule (smoothness of the max)
        alpha = alpha_start + (alpha_end - alpha_start) * min(1.0, t / max_iter)

        grad, _ = smoothed_gradient(a, alpha)

        # diminishing step size
        eta = eta0 / np.sqrt(t)
        a_next = a - eta * grad

        # occasional isotropic noise to escape tiny plateaus
        if noise_every and (t % noise_every == 0):
            a_next += np.random.randn(n) * noise_scale

        a_next = project_to_simplex(a_next)

        cur_score = compute_score(a_next)
        if cur_score < best_score:
            best_score = cur_score
            best_a = a_next.copy()

        a = a_next

        if t % 2000 == 0:
            print(f"    GD iter {t:,d} cur {cur_score:.6f} best {best_score:.6f}")

    return best_a, best_score


# ----------------------------------------------------------------------
# Stochastic hill‑climbing refinement
# ----------------------------------------------------------------------


def hill_climb(seq, time_deadline: float, max_steps: int = 50000, init_step: float = 0.07):
    """
    Pair‑wise mass‑transfer hill‑climb.
    Only moves that lower the *true* objective are accepted.
    """
    a = np.array(seq, dtype=float)
    a = project_to_simplex(a)
    best_a = a.copy()
    best_score = compute_score(best_a)
    n = len(a)
    step = init_step

    while max_steps > 0 and time.time() < time_deadline:
        i, j = np.random.randint(n, size=2)
        if i == j:
            continue
        delta = step * best_a[i]
        if delta <= 0.0:
            continue
        cand = best_a.copy()
        cand[i] -= delta
        cand[j] += delta
        cand = project_to_simplex(cand)
        cur_score = compute_score(cand)
        if cur_score < best_score:
            best_score = cur_score
            best_a = cand
            step = min(step * 1.07, 0.5)
        else:
            step *= 0.96
        max_steps -= 1
    return best_a.tolist(), best_score


# ----------------------------------------------------------------------
# Optional linear‑program refinement (requires SciPy)
# ----------------------------------------------------------------------


def lp_refine(seq, deadline, max_iters: int = 6, t: float = 0.2):
    """
    One‑step linear‑program improvement described in the problem statement.
    Solves:

        max   Σ b_j
        s.t.  (a * b)_k ≤ M   ∀k
              b_j ≥ 0

    where M = max(a∗a).  The solution is mixed with the current vector.
    """
    try:
        from scipy.optimize import linprog
    except Exception:
        print("SciPy not available – skipping LP refinement.")
        return seq

    a = np.array(seq, dtype=float)
    a = a / a.sum()
    n = len(a)

    best_a = a.copy()
    best_score = compute_score(best_a)

    while time.time() < deadline - 1 and max_iters > 0:
        max_iters -= 1
        conv = convolution_fft(a, a)
        M = np.max(conv)

        # Build the constraint matrix A (size (2n‑1)×n)
        rows = 2 * n - 1
        A = np.zeros((rows, n), dtype=float)
        for k in range(rows):
            i_min = max(0, k - (n - 1))
            i_max = min(k, n - 1)
            # j runs from i_min … i_max, with a_{k‑j} as the coefficient
            js = np.arange(i_min, i_max + 1, dtype=int)
            As = a[k - js]
            A[k, js] = As

        # Linear program: maximise sum(b)  ⇔ minimise -sum(b)
        c = -np.ones(n)
        b_ub = np.full(rows, M, dtype=float)
        bounds = [(0, None)] * n

        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method="highs")
        if not res.success:
            print(f"LP failed: {res.message}")
            break

        b = res.x
        if b.sum() == 0:
            b_norm = np.full_like(b, 1.0 / n)
        else:
            b_norm = b / b.sum()

        # convex combination
        a = (1 - t) * a + t * b_norm
        a = np.clip(a, 0, None)
        a = a / a.sum()

        cur_score = compute_score(a)
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"LP improvement – new score {cur_score:.6f}")
        else:
            # if no improvement, shrink the mixing factor
            t *= 0.8

    return best_a.tolist()
