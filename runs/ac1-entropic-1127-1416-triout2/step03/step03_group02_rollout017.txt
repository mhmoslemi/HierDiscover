def lp_improve(a, M, t=0.1, max_iters=20):
    n = len(a)
    for _ in range(max_iters):
        # compute M
        conv = convolution_fft(a, a)
        M = conv.max()
        # Build constraint matrix
        # Row k: for each j in [0,n-1], coefficient = a[k-j] if 0 <= k-j < n else 0
        # Equivalent to Toeplitz matrix with first column = a padded with zeros then a[1:]? Let's just compute via loops:
        A = np.zeros((2*n-1, n))
        for k in range(2*n-1):
            # i ranges maxi = max(0, k-n+1) to min(k, n-1)
            i_start = max(0, k - (n - 1))
            i_end = min(k, n - 1)
            # then a coefficient for b_j = a_i where j = k - i
            # For each i in [i_start, i_end], j = k-i.
            for i in range(i_start, i_end+1):
                j = k - i
                A[k, j] = a[i]
        # Solve LP: maximize sum(b) i.e., minimize -ones@b
        c = -np.ones(n)   # minimize - sum => maximize sum
        bounds = [(0, None)] * n
        res = linprog(c, A_ub=A, b_ub=np.full(2*n-1, M), bounds=bounds, method='highs')
        if not res.success:
            break
        b = res.x
        # Scale to same sum
        b = b / b.sum()   # sum=1
        # Update a via mixture
        a = (1 - t) * a + t * b
    return a
```

"""
Advanced search for a low‑score step‑function.

The programme combines three complementary ideas:

1.  A fast *grid search* over a few simple parametric families
    (arcsine, a product‑power law, beta‑like, random Dirichlet).
    The internal objective (2·n·max(conv)/sum²) is evaluated with an
    FFT‑based convolution, which is O(n log n).

2.  A *linear‑programming improvement* (inspired by the paper
    referenced in the prompt).  For a fixed sequence a we compute

          M = max (a*a)

    and then solve

          maximise   1ᵀb
          subject to  a ∗ b ≤ M   (element‑wise)
                      b ≥ 0

    This LP is tiny (≈2 n constraints, n variables).  The optimum b
    has a larger total mass than a while respecting the same convolution
    bound; after normalising b to unit sum we mix it with a:

          a ← (1‑t)·a + t·b ,   0<t≤1

    Repeating a few LP steps strictly decreases the maximal convolution
    and pushes the distribution towards the (almost flat) optimum.

3.  A *sub‑gradient / exponentiated‑gradient refinement* on the true
    maximum (i.e. the λ→∞ limit of the soft‑max used previously).
    At each iteration the weight vector w is an indicator of the
    current maximal convolution indices; the gradient
    g = 2·conv(a, w_rev) gives a sub‑gradient of max(a*a).  An
    exponentiated update (a ← a·exp(‑η·g)) together with a projection
    onto the probability simplex maintains feasibility and quickly
    reduces the L∞‑norm of the convolution.

The search proceeds on a modest size (n≈512) where the LP is cheap,
upsamples the best intermediate shape to the target length (n=2048)
and finishes with a longer exponentiated‑gradient run.  The final
candidate is evaluated with the official ``evaluate_sequence`` routine.
All steps respect the supplied time budget (default 1000 s).
"""

import time
import numpy as np
from typing import List, Tuple
from scipy.optimize import linprog

# ----------------------------------------------------------------------
# Basic numeric utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full linear convolution using a real FFT (zero‑padded to a power‑of‑2)."""
    m = len(a) + len(b) - 1
    n_fft = next_pow2(m)
    A = np.fft.rfft(a, n=n_fft)
    B = np.fft.rfft(b, n=n_fft)
    C = A * B
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


def compute_score(a: np.ndarray) -> float:
    """
    Official objective (scale‑invariant):
        2·n·max(a*a) / (sum(a)²)
    ``a`` may contain any non‑negative numbers; the function works for any vector.
    """
    n = len(a)
    conv = convolution_fft(a, a)
    max_val = float(np.max(conv))
    s = float(np.sum(a))
    if s < 1e-12:
        return float("inf")
    return 2.0 * n * max_val / (s * s)


def normalize(v: np.ndarray) -> np.ndarray:
    """Project a non‑negative vector onto the probability simplex."""
    total = np.sum(v)
    if total <= 0.0:
        return np.full_like(v, 1.0 / len(v))
    return v / total

# Families of starting points
# ----------------------------------------------------------------------


def arcsine_initial(n: int) -> np.ndarray:
    """Classic discrete arcsine density (symmetrised, normalised)."""
    i = np.arange(n, dtype=float)
    eps = 0.25  # small regularisation to avoid division by zero
    vals = 1.0 / np.sqrt((i + 0.5) * (n - 0.5 - i) + eps)
    return normalize(vals)


def generate_product_shape(n: int, p: float, eps: float) -> np.ndarray:
    """
    Symmetric power‑law shape a_i ∝ ((i+ε)*(n‑1‑i+ε))⁻ᵖ.
    ``p`` may be any positive number; ``eps`` stabilises the ends.
    """
    i = np.arange(n, dtype=float)
    base = ((i + eps) * (n - 1 - i + eps)) ** (-p)
    return normalize(base)


def generate_beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    """Beta‑like density: (i+½)^(α‑1)·(n‑1‑i+½)^(β‑1)."""
    i = np.arange(n, dtype=float)
    base = (i + 0.5) ** (alpha - 1) * (n - 1 - i + 0.5) ** (beta - 1)
    if np.all(base == 0):
        base = np.full(n, 1.0)
    return normalize(base)


def dirichlet_sample(n: int, rng: np.random.Generator) -> np.ndarray:
    """Sample a random point on the simplex via i.i.d. exponentials."""
    samp = rng.exponential(1.0, size=n)
    return normalize(samp)


# ----------------------------------------------------------------------
# Gradient utilities (soft‑max / sub‑gradient)
# ----------------------------------------------------------------------


def softmax_weights(conv: np.ndarray, lam: float) -> np.ndarray:
    """Soft‑max over the convolution entries (numerically stable)."""
    x = lam * conv
    mx = np.max(x)
    w = np.exp(x - mx)
    w /= np.sum(w)
    return w


def gradient_from_weights(a: np.ndarray, w: np.ndarray) -> np.ndarray:
    """
    Gradient of Σ_k w_k·(a*a)_k w.r.t. a.
    Implements   g_j = 2· Σ_m a_m · w_{m+j}
    using a single FFT‑based convolution.
    """
    # w is defined on the full convolution support (length 2n‑1)
    w_rev = w[::-1]
    conv = convolution_fft(a, w_rev)               # length = 3·n‑2
    idx = 2 * len(a) - 2                            # position of 2n‑2 in the full conv
    grads = 2.0 * conv[idx - np.arange(len(a))]
    return grads


def subgradient_optimize(
    a0: np.ndarray,
    deadline: float,
    *,
    eta0: float = 0.8,
    max_iter: int = 60_000,
) -> Tuple[np.ndarray, float]:
    """
    Exponentiated‑gradient descent on the *exact* max‑convolution
    (λ→∞ soft‑max).  The weight vector is an indicator of the current
    maximal indices, yielding a genuine sub‑gradient.
    """
    a = normalize(a0.astype(float).copy())
    best = a.copy()
    best_sc = compute_score(a)
    n = len(a)

    for it in range(1, max_iter + 1):
        if time.time() >= deadline:
            break

        conv = convolution_fft(a, a)
        max_val = conv.max()
        # indicator weight (normalized)
        mask = np.isclose(conv, max_val, atol=1e-14)
        w = np.zeros_like(conv)
        w[mask] = 1.0 / np.count_nonzero(mask)

        grad = gradient_from_weights(a, w)
        eta = eta0 / np.sqrt(it)
        a = a * np.exp(-eta * grad)   # multiplicative update → stays non‑negative
        a = normalize(a)

        sc = compute_score(a)
        if sc < best_sc:
            best_sc = sc
            best = a.copy()

        # occasional progress report
        if it % 5000 == 0:
            print(f"    subgrad iter={it:,d}  cur={sc:.6f}  best={best_sc:.6f}")

    return best, best_sc


# ----------------------------------------------------------------------
# Linear‑programming improvement (stage 2 of the original paper)
# ----------------------------------------------------------------------


def build_constraint_matrix(a: np.ndarray) -> np.ndarray:
    """
    Construct the matrix A (shape (2n‑1, n)) encoding the linear constraints
    (a ∗ b)_k ≤ M for all k, i.e.

          Σ_i a_i · b_{k-i}   ≤   M   .

    The entry A[k, j] = a_{k‑j} when the index is in range, otherwise 0.
    """
    n = len(a)
    rows = 2 * n - 1
    A = np.zeros((rows, n), dtype=float)

    for k in range(rows):
        i_start = max(0, k - (n - 1))
        i_end = min(k, n - 1)
        # i runs over indices of a that contribute to position k
        for i in range(i_start, i_end + 1):
            j = k - i
            A[k, j] = a[i]
    return A


def lp_step(a: np.ndarray, t: float) -> np.ndarray:
    """
    Single LP‑based improvement step.

    * M = max(a*a)
    * maximise Σ b_j  subject to  a∗b ≤ M ,  b ≥ 0
    * renormalise b to unit sum and mix with a:
          a ← (1‑t)·a + t·b
    """
    n = len(a)
    conv = convolution_fft(a, a)
    M = conv.max()
    A = build_constraint_matrix(a)

    # Linear programme: minimise -Σ b_j   (i.e. maximise Σ b_j)
    c = -np.ones(n, dtype=float)
    bounds = [(0, None)] * n
    b_ub = np.full(2 * n - 1, M, dtype=float)

    res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method="highs")
    if not res.success:
        # fallback – no improvement possible
        return a

    b = res.x
    if b.sum() == 0.0:
        return a
    b = normalize(b)

    new_a = (1.0 - t) * a + t * b
    new_a = normalize(new_a)
    return new_a


def lp_improve(a: np.ndarray, deadline: float, *, steps: int = 5, t0: float = 0.3) -> np.ndarray:
    """
    Apply a few LP improvement steps with a decaying mixing factor.
    """
    a_cur = a.copy()
    for s in range(steps):
        if time.time() >= deadline - 5.0:
            break
        # slowly reduce the mixing weight as we get closer to optimum
        t = t0 * (0.9 ** s)
        a_cur = lp_step(a_cur, t)
    return a_cur


# ----------------------------------------------------------------------
# Large‑scale refinement (soft‑max based, as in the previous baseline)
# ----------------------------------------------------------------------


def optimise_sequence(
    initial_a: np.ndarray,
    deadline: float,
    *,
    lam_start: float = 5.0,
    lam_end: float = 200.0,
    lam_ramp_iters: int = 10_000,
    eta0: float = 0.5,
    max_iter: int = 200_000,
    noise_every: int = 0,
    noise_scale: float = 0.0,
) -> Tuple[np.ndarray, float]:
    """
    Exponentiated‑gradient descent on a smoothed version of the max‑convolution.
    The smoother temperature λ is increased linearly from ``lam_start`` to
    ``lam_end`` over ``lam_ramp_iters`` iterations.
    """
    a = normalize(initial_a.astype(float).copy())
    best_a = a.copy()
    best_sc = compute_score(a)
    n = len(a)

    for it in range(1, max_iter + 1):
        if time.time() >= deadline:
            break

        lam = lam_start + (lam_end - lam_start) * min(1.0, it / lam_ramp_iters)

        conv = convolution_fft(a, a)
        w = softmax_weights(conv, lam)

        grad = gradient_from_weights(a, w)

        eta = eta0 / np.sqrt(it)
        a = a * np.exp(-eta * grad)

        if noise_every and (it % noise_every == 0):
            a += np.random.randn(n) * noise_scale
            a = np.maximum(a, 0.0)

        a = normalize(a)

        cur_sc = compute_score(a)
        if cur_sc < best_sc:
            best_sc = cur_sc
            best_a = a.copy()

        if it % 5000 == 0:
            print(f"      iter={it:,d}  λ={lam:.1f}  cur={cur_sc:.6f}  best={best_sc:.6f}")

    return best_a, best_sc


# ----------------------------------------------------------------------
# Main entry point for the competition harness
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs) -> List[float]:
    """
    Search for a non‑negative step‑function with a low `evaluate_sequence`
    value.  The routine respects the given time budget, explores several
    parametric families, applies a small LP‑based improvement, up‑samples
    to length 2048, and finishes with a longer exponentiated‑gradient run.
    The best feasible sequence found is returned as a list of floats.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 2.0   # safety margin

    best_seq: List[float] = []
    best_val: float = float("inf")

    # ------------------------------------------------------------------
    # 0. Optional fallback from a previously discovered construction.
    # ------------------------------------------------------------------
    try:
        fallback = [float(x) for x in height_sequence_1]  # type: ignore
        val = evaluate_sequence(fallback)
        print(f"Fallback (height_sequence_1) score = {val:.6f}")
        if val < best_val:
            best_val = val
            best_seq = fallback
    except Exception:
        pass

    # ------------------------------------------------------------------
    # 1.  Search on a modest size where the LP is cheap.
    # ------------------------------------------------------------------
    n_small = 512
    print(f"\n=== Small‑scale exploration (n={n_small}) ===")

    candidates_small: List[np.ndarray] = []

    # a) classic families
    candidates_small.append(arcsine_initial(n_small))
    candidates_small.append(np.full(n_small, 1.0 / n_small))

    # b) product‑power‑law shapes (systematic grid)
    for p in np.linspace(0.05, 0.49, 9):
        for eps in [0.0, 0.1, 0.5, 1.0, 2.0]:
            candidates_small.append(generate_product_shape(n_small, p, eps))

    # c) beta‑like shapes
    for _ in range(6):
        alpha = rng.uniform(0.2, 2.5)
        beta = rng.uniform(0.2, 2.5)
        candidates_small.append(generate_beta_shape(n_small, alpha, beta))

    # d) random Dirichlet samples
    for _ in range(6):
        candidates_small.append(dirichlet_sample(n_small, rng))

    # ------------------------------------------------------------------
    # 1b. Quick sweep, keep the most promising few.
    # ------------------------------------------------------------------
    scored_small = [(compute_score(c), c) for c in candidates_small]
    scored_small.sort(key=lambda x: x[0])
    top_small = [c for _, c in scored_small[:5]]
    print(f"  kept {len(top_small)} top candidates (small n).")

    # ------------------------------------------------------------------
    # 1c. Refine each retained candidate with sub‑gradient + LP.
    # ------------------------------------------------------------------
    for idx, cand in enumerate(top_small):
        if time.time() >= deadline - 10.0:
            break
        print(f"\n  small candidate {idx + 1}/{len(top_small)}")
        # sub‑gradient (exact max) refinement
        sub_ref, sub_sc = subgradient_optimize(cand, deadline, eta0=0.9, max_iter=30_000)
        print(f"    after sub‑grad: score = {sub_sc:.6f}")

        # LP improvement (a few aggressive steps)
        lp_ref = lp_improve(sub_ref, deadline, steps=6, t0=0.35)
        lp_sc = compute_score(lp_ref)
        print(f"    after LP step: score = {lp_sc:.6f}")

        # keep the better of the two
        cand_best = lp_ref if lp_sc < sub_sc else sub_ref
        cand_sc = min(lp_sc, sub_sc)

        if cand_sc < best_val:
            best_val = cand_sc
            best_seq = cand_best.tolist()
            print(f"    *** New global best (small) score = {best_val:.6f} ***")

    # ------------------------------------------------------------------
    # 2.  Upsample to the target length and perform a long refined search.
    # ------------------------------------------------------------------
    n_big = 2048
    factor = n_big // n_small
    if not best_seq:
        # fallback to uniform if nothing survived the small stage
        best_seq = [1.0 / n_small] * n_small

    a_big = np.repeat(np.array(best_seq, dtype=float), factor)
    a_big = normalize(a_big)   # ensure exact simplex

    print(f"\n=== Upsampled to n={n_big}, starting large‑scale refinement ===")
    remaining = max(0.0, deadline - time.time())
    max_iter_big = int(min(180_000, remaining * 20))   # generous iteration budget

    refined_big, big_sc = optimise_sequence(
        initial_a=a_big,
        deadline=deadline,
        lam_start=5.0,
        lam_end=250.0,
        lam_ramp_iters=12_000,
        eta0=0.5,
        max_iter=max_iter_big,
        noise_every=800,
        noise_scale=5e-6,
    )
    final_val = evaluate_sequence(refined_big.tolist())
    print(f"\nLarge‑scale polished score = {final_val:.6f}")

    if final_val < best_val:
        best_val = final_val
        best_seq = refined_big.tolist()
        print("*** Updated global best after large polish ***")

    # ------------------------------------------------------------------
    # 3. Optional tiny random local tweaks (if time remains).
    # ------------------------------------------------------------------
    while time.time() < deadline - 5.0:
        arr = np.array(best_seq, dtype=float)
        i, j = rng.choice(len(arr), size=2, replace=False)
        # move a tiny random amount from i to j
        delta = rng.uniform(0.0, min(arr[i], 0.005))
        if delta == 0.0:
            continue
        arr[i] -= delta
        arr[j] += delta
        arr = normalize(arr)
        val = evaluate_sequence(arr.tolist())
        if val < best_val:
            best_val = val
            best_seq = arr.tolist()
            print(f"Random tweak improved score to {best_val:.6f}")

    # ------------------------------------------------------------------
    # 4. Guard – always return a feasible sequence.
    # ------------------------------------------------------------------
    if not best_seq:
        best_seq = [1.0] * 256
        best_val = evaluate_sequence(best_seq)

    print(f"\nSearch finished. Best reported score = {best_val:.6f}")
    return best_seq
