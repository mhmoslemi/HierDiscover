def expand_to_sequence(x, n):
    m = len(x)
    block_len = n // m
    extra = n % m
    seq = []
    for i in range(m):
        current_len = block_len + (1 if i < extra else 0)
        # distribute mass x[i] equally across its entries
        val = x[i] / current_len
        seq.extend([val]*current_len)
    return seq[:n]  # trim if any rounding issues

def simplex_projection(v):
    """Euclidean projection of a vector onto the probability simplex."""
    v = np.asarray(v, dtype=float)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1] - 1
    theta = cssv[cond][-1] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def compute_objective(a):
    """
    Fast surrogate for ``evaluate_sequence`` when Σa = 1.
    Returns 2·n·max(a∗a).
    """
    n = a.shape[0]
    # linear convolution via FFT (length 2·n‑1)
    fft = np.fft.rfft(a, n=2 * n - 1)
    conv = np.fft.irfft(fft * fft, n=2 * n - 1)
    return float(2.0 * n * conv.max())


def decreasing_power(p, n):
    """
    a_i ∝ (1 - i/(n-1))**p  (p > 0).  Returned vector is normalised to sum 1.
    """
    i = np.arange(n, dtype=float)
    # avoid division by zero when n==1 (not the case here)
    w = np.power(1.0 - i / (n - 1), p)
    w = np.clip(w, 0.0, None)
    total = w.sum()
    if total == 0 or not np.isfinite(total):
        w = np.full_like(w, 1.0 / n)
        total = w.sum()
    w /= total
    return w


def random_dirichlet(n):
    """Generate a random point on the simplex via a Gamma/Dirichlet draw."""
    x = np.random.gamma(1.0, 1.0, size=n)
    s = x.sum()
    if s == 0 or not np.isfinite(s):
        return np.full(n, 1.0 / n)
    return x / s
