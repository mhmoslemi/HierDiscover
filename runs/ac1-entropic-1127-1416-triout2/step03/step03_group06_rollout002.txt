def adam_descent(a0, max_iters):
    a = a0.copy()
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    lr = 0.5  # base learning rate
    best_local_seq = a.copy()
    best_local_score = compute_objective(a)

    for t in range(1, max_iters+1):
        # compute conv & max
        fft = np.fft.rfft(a, n=2 * n - 1)
        conv = np.fft.irfft(fft * fft, n=2 * n - 1)
        max_b = conv.max()
        cur_score = 2.0 * n * max_b
        if cur_score < best_local_score:
            best_local_score = cur_score
            best_local_seq = a.copy()
            # also update global best via consider
        # active set
        eps_act = max_b * 1e-6
        active = np.where(conv >= max_b - eps_act)[0]
        if active.size == 0:
            active = np.array([conv.argmax()], dtype=int)
        g = subgradient_active(a, active)  # averaged subgradient

        # Adam update (minimization)
        m = beta1 * m + (1 - beta1) * g
        v = beta2 * v + (1 - beta2) * (g * g)
        m_hat = m / (1 - beta1 ** t)
        v_hat = v / (1 - beta2 ** t)
        step = lr * m_hat / (np.sqrt(v_hat) + eps)

        a_new = a - step
        # project onto simplex
        a_new = simplex_projection(a_new)
        a = a_new

        # occasional reset or perturb
        if t % 20000 == 0:
            # some random noise to escape
            sigma = max(0.01, 0.5 / np.sqrt(t))
            noise = sigma * np.random.randn(n) * a
            a = np.clip(a + noise, 0, None)
            a = a / a.sum()
            m.fill(0); v.fill(0)
        if time.time() >= deadline:
            break

    return best_local_seq, best_local_score

"""

minimises the black‑box ``evaluate_sequence``.

Algorithm
---------
1. Determine the problem size ``n`` (taken from the global ``height_sequence_1``
   if it exists, otherwise a fallback of 5 000).
2. Build a large, diverse pool of seed sequences:
   * uniform, linear ramp, exponential decay, power‑law decay,
   * discretised Beta(α,β) densities on a fine grid (α∈[0.02,0.5],
     β∈[1.0,5.0]),
   * random Dirichlet vectors with varying concentration.
   All seeds are normalised to have ℓ₁‑norm 1.
3. Keep the ``TOP_K`` best seeds (smallest objective value) and remember the
   overall best sequence seen so far.
4. For every seed in the top list run a limited **Adam‑style projected
   gradient descent** on the probability simplex:
   * at each iteration compute the full autocorrelation ``conv = a*a`` via FFT,
   * identify the (almost) active set – indices whose value is within
     ``1e‑6 * max(conv)`` of the current maximum,
   * compute the sub‑gradient of the maximum entry (averaged over the active
     set) – this is exactly the gradient used in the reference solution,
   * update the point with Adam (first‑ and second‑moment estimates) and
     project back onto the simplex,
   * every 20 000 iterations add a small random perturbation to escape plateaus.
5. Throughout the search keep the globally best sequence (by calling our
   ``compute_objective`` which is identical to ``evaluate_sequence`` after the
   normalisation step) and stop as soon as the 1000 s budget is exhausted.
6. Return the best sequence found as a plain ``list[float]``.
"""

def simplex_projection(v):
    """
    Euclidean projection of a vector onto the probability simplex:
    { w | w_i ≥ 0 , Σ w_i = 1 }.
    Implementation follows Duchi et al. (2008).
    """
    v = np.asarray(v, dtype=float)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):                 # all non‑positive → uniform
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1] - 1              # zero‑based index
    theta = cssv[cond][-1] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def compute_objective(a):
    """
    Return the value of ``evaluate_sequence`` for a vector that already
    satisfies Σ a_i = 1.  The function therefore boils down to
        2·n·max(a * a)
    where * denotes the linear (full) convolution.
    """
    n = a.shape[0]
    # FFT based convolution (length 2·n‑1)
    fft = np.fft.rfft(a, n=2 * n - 1)
    conv = np.fft.irfft(fft * fft, n=2 * n - 1)
    return float(2.0 * n * conv.max())


def subgradient_active(a, active):
    """
    Averaged sub‑gradient of the maximum convolution entry over the
    supplied ``active`` index set.
    """
    n = a.shape[0]
    g = np.zeros_like(a)
    for k in active:
        low = max(0, k - n + 1)
        high = min(n - 1, k)
        idx = np.arange(low, high + 1, dtype=int)
        partner = k - idx
        g[idx] += 2.0 * a[partner]
    g /= float(active.size)
    return g


def beta_sequence(alpha, beta, n):
    """
    Discretised Beta(α,β) density on ``n`` centre points,
    normalised to ℓ₁‑norm 1.  Handles extreme α/β by clipping the log‑density.
    """
    x = (np.arange(n, dtype=float) + 0.5) / n
    # log‑density, stable for small parameters
    logw = (alpha - 1.0) * np.log(x) + (beta - 1.0) * np.log(1.0 - x)
    logw = np.clip(logw, -700.0, 700.0)          # avoid overflow in exp
    w = np.exp(logw)
    total = w.sum()
    if total == 0 or not np.isfinite(total):
        w = np.ones_like(w)
        total = w.sum()
    return w / total


def exponential_shape(lam, n):
    """a_i ∝ exp(‑lam·i)  (lam ≥ 0), normalised to ℓ₁‑norm 1."""
    i = np.arange(n, dtype=float)
    w = np.exp(-lam * i)
    w /= w.sum()
    return w


def power_law_shape(p, n):
    """
    a_i ∝ (i+1)^{-p}   (p ≥ 0), normalised to ℓ₁‑norm 1.
    For p=0 we obtain the uniform distribution.
    """
    i = np.arange(n, dtype=float) + 1.0
    w = i ** (-p)
    w /= w.sum()
    return w


def linear_ramp(n):
    """Strictly decreasing linear ramp a_i ∝ (n‑i)."""
    a = np.arange(n, 0, -1, dtype=float)
    a /= a.sum()
    return a


# ----------------------------------------------------------------------
# Main search routine
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 0, budget_s: int = 1000):
    """
    Search for a non‑negative step‑function (list of floats) that minimises
    ``evaluate_sequence``.  The function respects the 1000 s time budget and
    returns the best sequence discovered.
    """
    np.random.seed(seed)
    start_time = time.time()
    # safety margin for the outer runner
    deadline = start_time + budget_s - 5.0

    # ------------------------------------------------------------------
    # 1. Problem dimension
    # ------------------------------------------------------------------
    try:
        n = len(height_sequence_1)   # provided by the outer environment
    except NameError:
        n = 5000                     # fallback if the variable is missing

    # ------------------------------------------------------------------
    # 2. Global bookkeeping
    # ------------------------------------------------------------------
    best_seq = None          # numpy array, ℓ₁‑normalised
    best_score = np.inf
    TOP_K = 10               # number of seeds kept for the local search
    top = []                 # list of tuples (score, seq)

    def consider(candidate):
        """Normalise *candidate*, evaluate it and possibly update the globals."""
        nonlocal best_seq, best_score, top
        a = np.asarray(candidate, dtype=float)
        s = a.sum()
        if s == 0:
            return
        a = a / s
        score = compute_objective(a)
        if score < best_score:
            best_score = score
            best_seq = a.copy()
            print(f"[{time.time() - start_time:5.1f}s] Global best   {score:.6f}")
        # maintain a sorted top‑K list (smallest scores first)
        if len(top) < TOP_K:
            top.append((score, a.copy()))
            top.sort(key=lambda x: x[0])
        else:
            if score < top[-1][0]:
                top[-1] = (score, a.copy())
                top.sort(key=lambda x: x[0])

    # ------------------------------------------------------------------
    # 3. Deterministic seed families
    # ------------------------------------------------------------------
    consider(np.full(n, 1.0 / n))           # uniform
    consider(linear_ramp(n))                # linear decreasing ramp

    # Exponential decay (λ on a log‑scale)
    for lam in np.logspace(-3, 0.5, 25):
        consider(exponential_shape(lam, n))

    # Power‑law decay (p on a linear scale)
    for p in np.linspace(0.0, 3.0, 31):
        consider(power_law_shape(p, n))

    # ------------------------------------------------------------------
    # 4. Systematic Beta‑distribution sweep (fine grid)
    # ------------------------------------------------------------------
    alphas = np.linspace(0.02, 0.5, 15)      # very small α → strong decrease
    betas = np.linspace(1.0, 5.0, 15)
    for α in alphas:
        for β in betas:
            if time.time() >= deadline:
                break
            consider(beta_sequence(α, β, n))
        if time.time() >= deadline:
            break

    # ------------------------------------------------------------------
    # 5. Randomised seeds (Dirichlet‑like)
    # ------------------------------------------------------------------
    for _ in range(300):
        if time.time() >= deadline:
            break
        # randomly drawn concentration parameters (0.1 – 3.0)
        conc = np.random.uniform(0.1, 3.0, size=n)
        a_dir = np.random.gamma(conc, 1.0)
        consider(a_dir)

    # ------------------------------------------------------------------
    # 6. Incorporate the historic best (if available)
    # ------------------------------------------------------------------
    try:
        hist = np.array(height_sequence_1, dtype=float)
        consider(hist)
    except NameError:
        pass

    # ------------------------------------------------------------------
    # 7. Local optimisation – Adam projected gradient descent
    # ------------------------------------------------------------------
    # Adam hyper‑parameters (chosen conservatively for stability)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    base_lr = 0.4          # base learning rate; the effective step shrinks
                           # as the moments adapt

    def adam_descent(start_vec, max_iters, label):
        """Run Adam on the simplex starting from ``start_vec``."""
        a = start_vec.copy()
        m = np.zeros_like(a)
        v = np.zeros_like(a)
        best_local = a.copy()
        best_local_score = compute_objective(a)
        last_improve_iter = 0

        for t in range(1, max_iters + 1):
            # stop if we have exhausted the time budget
            if time.time() >= deadline:
                break

            # --- convolution & current objective ---
            fft = np.fft.rfft(a, n=2 * n - 1)
            conv = np.fft.irfft(fft * fft, n=2 * n - 1)
            max_b = conv.max()
            cur_score = 2.0 * n * max_b

            # keep the best ever seen (global bookkeeping)
            if cur_score < best_local_score - 1e-12:
                best_local_score = cur_score
                best_local = a.copy()
                consider(a)                     # may update global best

                last_improve_iter = t

            # --- active set (within 1e‑6 relative tolerance) ---
            eps_act = max_b * 1e-6
            active = np.where(conv >= max_b - eps_act)[0]
            if active.size == 0:
                active = np.array([conv.argmax()], dtype=int)

            # --- sub‑gradient of the max convolution entry ---
            g = subgradient_active(a, active)   # averaged sub‑gradient

            # --- Adam update (minimisation) ---
            m = beta1 * m + (1.0 - beta1) * g
            v = beta2 * v + (1.0 - beta2) * (g * g)
            m_hat = m / (1.0 - beta1 ** t)
            v_hat = v / (1.0 - beta2 ** t)
            step = base_lr * m_hat / (np.sqrt(v_hat) + eps)

            a_new = a - step
            a = simplex_projection(a_new)

            # --- occasional random perturbation to escape plateaus ---
            if t - last_improve_iter >= 20000:
                sigma = max(0.01, 0.5 / np.sqrt(t))
                noise = sigma * np.random.randn(n) * a
                a = np.clip(a + noise, 0.0, None)
                a /= a.sum()
                m.fill(0.0)
                v.fill(0.0)
                last_improve_iter = t
                print(f"[{time.time() - start_time:5.1f}s] {label} "
                      f"perturbation at iter {t}")

        return best_local, best_local_score

    # ------------------------------------------------------------------
    # 8. Run the local optimiser on the best seeds
    # ------------------------------------------------------------------
    # Allocate the remaining time roughly equally among the retained seeds.
    remaining_time = deadline - time.time()
    if remaining_time > 20:
        # we give each seed at most 150 000 iterations – this is well below
        # the time needed for a single 150 k iteration loop on our hardware.
        max_iters_per_seed = 150_000
        for idx, (sc, seq) in enumerate(top):
            if time.time() >= deadline:
                break
            label = f"seed#{idx+1}"
            print(f"[{time.time() - start_time:5.1f}s] Starting {label} "
                  f"with score {sc:.6f}")
            adam_descent(seq, max_iters_per_seed, label)

    # ------------------------------------------------------------------
    # 9. Return result
    # ------------------------------------------------------------------
    print(f"[{time.time() - start_time:5.1f}s] Finished – best score "
          f"{best_score:.6f}")
    # Convert the NumPy array back to a plain Python list
    return [float(v) for v in best_seq.tolist()]
