"""
Propose a step‑height sequence that (approximately) minimizes
evaluate_sequence().

Algorithm overview
------------------
1. **Initialisation** – start from the previously found best `height_sequence_1`
   (if present) or a symmetric Beta(2.5,2.5) distribution.
2. **Exponentiated‑gradient (EG) descent** – minimise a smooth
   approximation of the ℓ∞‑norm of the auto‑convolution.
   The smoothing parameter is slowly increased; the update
   a ← a·exp(−η·∇) keeps the vector strictly positive,
   after which we renormalise to a probability simplex.
3. **Active‑set linear‑programming refinement** – at the current iterate
   identify the (almost) active convolution indices, build the linear
   constraints   (f⋆g)_k ≤ max(f⋆f)   for those k and solve a tiny LP
   (random objective, non‑negativity and a harmless sum‑bound) to obtain
   an extreme feasible direction g.  The new iterate is a convex
   combination f←(1−τ)f+τg (τ≈0.2).
4. **Simulated‑annealing fine‑tuning** – short stochastic local search
   that moves a random fraction of mass between two coordinates.
   The temperature is exponential; occasional full random restarts keep
   the walk from stagnating.
All phases respect the time budget (default 1000 s); any remaining time
is spent in the final annealing stage.

The code uses only NumPy, SciPy and the supplied `evaluate_sequence`
function.  No file or network access is performed.
"""

import time
import math
import numpy as np
import scipy.optimize as opt
import scipy.sparse as sp

# ----------------------------------------------------------------------
# Helper utilities
# ----------------------------------------------------------------------


def beta_sequence(alpha: float, beta: float, n: int) -> np.ndarray:
    """Discretised Beta(α,β) density on n points (mid‑point rule)."""
    x = (np.arange(n) + 0.5) / n
    logw = (alpha - 1.0) * np.log(x) + (beta - 1.0) * np.log(1.0 - x)
    # protect against overflow / underflow
    logw = np.clip(logw, -700.0, 700.0)
    w = np.exp(logw)
    if not np.isfinite(w.sum()) or w.sum() == 0.0:
        w = np.ones_like(w)
    return w / w.sum()


def conv_self(a: np.ndarray) -> np.ndarray:
    """Linear auto‑convolution a⋆a using FFT (length 2·n−1)."""
    n = a.shape[0]
    fft_a = np.fft.rfft(a, n=2 * n - 1)
    conv = np.fft.irfft(fft_a * fft_a, n=2 * n - 1)
    return conv


def smoothmax_gradient(a: np.ndarray, smoothing: float) -> np.ndarray:
    """
    Return the gradient of the smoothed max (log‑sum‑exp) of the
    auto‑convolution of `a`.  The gradient is computed via cross‑correlation.
    """
    n = a.shape[0]
    # auto‑convolution
    conv = conv_self(a)

    # smoothed max via log‑sum‑exp
    z = smoothing * conv
    z_max = np.max(z)
    exp_z = np.exp(z - z_max)          # stability
    w = exp_z / np.sum(exp_z)          # soft‑max weights, sum to 1

    # cross‑correlation w ⋆ a  (g_i = 2·∑_k w_k·a_{k-i})
    L = (2 * n - 1) + n - 1            # length needed for full correlation
    fft_w = np.fft.rfft(w, n=L)
    fft_a_rev = np.fft.rfft(a[::-1], n=L)
    full_corr = np.fft.irfft(fft_w * fft_a_rev, n=L)
    grad = 2.0 * full_corr[n - 1: n - 1 + n]
    return grad


def simplex_normalise(v: np.ndarray) -> np.ndarray:
    """Renormalise a non‑negative vector to have sum exactly 1."""
    v = np.maximum(v, 1e-15)          # avoid exact zeros → numerical stability
    return v / v.sum()


def score_of(a: np.ndarray) -> float:
    """Exact evaluate_sequence() value when sum(a) == 1 (i.e. 2·n·max(conv))."""
    n = a.shape[0]
    conv = conv_self(a)
    return 2.0 * n * np.max(conv)


# ----------------------------------------------------------------------
# Main search routine
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 0, budget_s: int = 1000) -> list[float]:
    """
    Search for a non‑negative height sequence that minimises
    evaluate_sequence().  The function runs for at most `budget_s` seconds
    and returns the best sequence found.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5          # tiny safety margin

    # --------------------------------------------------------------
    # 0. Problem size & initial point
    # --------------------------------------------------------------
    try:
        init = np.array(height_sequence_1, dtype=float)   # type: ignore[name-defined]
        n = init.shape[0]
        print(f"Using existing height_sequence_1 (n={n}).")
    except NameError:
        n = 5146
        init = beta_sequence(2.5, 2.5, n)
        print(f"No existing sequence – start from Beta(2.5,2.5), n={n}.")

    a = simplex_normalise(init)          # current iterate (sum == 1)
    best_a = a.copy()
    best_score = score_of(best_a)
    print(f"Initial score = {best_score:.6f}")

    # --------------------------------------------------------------
    # 1. Exponentiated‑gradient descent with a growing smoothness
    # --------------------------------------------------------------
    eg_deadline = deadline - 0.30 * budget_s   # reserve ~30% of time for later phases
    smoothing = 5.0          # start with a gentle smooth max
    eta = 0.25               # base step size for EG
    it_eg = 0
    while time.time() < eg_deadline:
        cur_score = score_of(a)
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"EG iter {it_eg:6d}: new best {best_score:.6f}")

        # gradient of the smoothed objective
        grad = smoothmax_gradient(a, smoothing)

        # EG multiplicative update, followed by renormalisation
        a = a * np.exp(-eta * grad)
        a = simplex_normalise(a)

        # slowly tighten the approximation of the true max
        smoothing = min(500.0, smoothing * 1.012)

        # optional step‑size decay (simple schedule)
        eta = max(0.01, eta * 0.9999)

        it_eg += 1

    print(f"EG phase finished after {it_eg} iterations; best score = {best_score:.6f}")

    # --------------------------------------------------------------
    # 2. Active‑set linear‑programming refinement
    # --------------------------------------------------------------
    lp_deadline = deadline - 0.15 * budget_s   # keep ~15% for final SA
    it_lp = 0
    max_active = 150          # cap size of the active set (speed / memory)

    while time.time() < lp_deadline:
        conv = conv_self(a)
        M = np.max(conv)

        # active indices (within a tiny relative tolerance)
        eps = M * 1e-4
        active = np.where(conv >= M - eps)[0]
        if active.size == 0:
            active = np.array([conv.argmax()], dtype=int)
        # If too many, keep a random subset
        if active.size > max_active:
            active = np.random.choice(active, size=max_active, replace=False)

        # Build sparse constraint matrix A_ub · g ≤ M
        rows, cols, data = [], [], []
        for row_id, k in enumerate(active):
            i_start = max(0, k - n + 1)
            i_end = min(k, n - 1)
            i = np.arange(i_start, i_end + 1)
            j = k - i                     # partner index of g
            rows.extend([row_id] * len(i))
            cols.extend(j.tolist())
            data.extend(a[i].tolist())

        # additional harmless sum‑bound to keep the LP bounded
        sum_bound = 5.0
        sum_row = len(active)
        rows.extend([sum_row] * n)
        cols.extend(list(range(n)))
        data.extend([1.0] * n)

        A_ub = sp.csr_matrix((data, (rows, cols)),
                             shape=(len(active) + 1, n))
        b_ub = np.concatenate([np.full(len(active), M), np.array([sum_bound])])

        # Random linear objective – picking an extreme feasible point
        c = -np.random.rand(n)   # minimisation of -c·g  == maximisation of c·g

        lp_res = opt.linprog(c,
                             A_ub=A_ub,
                             b_ub=b_ub,
                             bounds=[(0, None)] * n,
                             method='highs',
                             options={'presolve': True})

        if not lp_res.success:
            # If the solver fails (rare), break out of the LP loop.
            print(f"LP iteration {it_lp} failed: {lp_res.message}")
            break

        g = lp_res.x
        sum_g = g.sum()
        if sum_g == 0.0:
            # Degenerate solution – abort LP phase.
            break
        g = g / sum_g                      # normalise to sum == 1

        # Convex combination with a modest mixing parameter
        tau = 0.20
        a = (1.0 - tau) * a + tau * g
        a = simplex_normalise(a)

        cur_score = score_of(a)
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"LP iter {it_lp:4d}: new best {best_score:.6f}")

        it_lp += 1

    print(f"LP phase finished after {it_lp} iterations; best score = {best_score:.6f}")

    # --------------------------------------------------------------
    # 3. Simulated‑annealing fine‑tuning (remaining time)
    # --------------------------------------------------------------
    cur_a = a.copy()
    cur_score = score_of(cur_a)

    # temperature schedule – start with a small fraction of the current score
    T_start = cur_score * 0.01
    T_end = cur_score * 1e-6
    it_sa = 0

    while time.time() < deadline:
        # geometric cooling
        T = max(T_end, T_start * (0.9995 ** it_sa))

        # ---- propose a mass‑transfer move ----
        src = np.random.choice(n, p=cur_a / cur_a.sum())
        tgt = np.random.randint(n - 1)
        if tgt >= src:
            tgt += 1
        if cur_a[src] == 0.0:
            continue
        delta = cur_a[src] * np.random.rand()          # up to 100 % of src mass
        cand = cur_a.copy()
        cand[src] -= delta
        cand[tgt] += delta

        cand_score = score_of(cand)

        # Metropolis acceptance
        if cand_score < cur_score:
            accept = True
        else:
            accept_prob = math.exp(-(cand_score - cur_score) / max(T, 1e-12))
            accept = np.random.rand() < accept_prob

        if accept:
            cur_a = cand
            cur_score = cand_score
            if cand_score < best_score:
                best_score = cand_score
                best_a = cand.copy()
                print(f"SA iter {it_sa:6d}: new best {best_score:.6f}")

        # occasional full random restart to avoid getting stuck
        if it_sa % 50000 == 0 and it_sa > 0:
            cur_a = beta_sequence(np.random.uniform(0.5, 5.0),
                                  np.random.uniform(0.5, 5.0), n)
            cur_a = simplex_normalise(cur_a)
            cur_score = score_of(cur_a)

        it_sa += 1

    print(f"SA finished after {it_sa} iterations; final best score = {best_score:.6f}")

    # --------------------------------------------------------------
    # 4. Return the best sequence (still normalised to sum == 1)
    # --------------------------------------------------------------
    best_a = simplex_normalise(best_a)
    return [float(v) for v in best_a.tolist()]
