        if score < best_score:
            best_score = score
            best_a = a_new.copy()
        a = a_new
    return best_a.tolist(), best_score

"""
Propose a low‑value step‑function for the `evaluate_sequence` objective.

The algorithm proceeds in several lightweight stages that fit comfortably
into the 1000 s budget.

1. **Generate a diverse pool of candidate shapes** (arcsine, beta‑type,
   exponential, Gaussian‑like, raised‑cosine and random Dirichlet vectors)
   for three modest lengths (1024, 2048, 4096).  The pool is evaluated and the
   best 5 candidates are retained.

2. **Sub‑gradient descent on the true max‑norm**.
   For a candidate `a` (already on the probability simplex) we repeatedly
   compute the full convolution `c = a*a` via an FFT, locate the index set
   where `c` attains its maximum and form a valid sub‑gradient of the
   nondifferentiable term `max(c)`.  A diminishing stepsize
   `ηₜ = η₀ / sqrt(t)` is used and after each step the vector is projected
   back onto the simplex.  The best vector encountered during the descent
   is kept.

3. **Upsampling** – if time permits the best vector is linearly interpolated
   to length 8192; this often reduces the objective because the discrete
   approximation becomes finer.

4. **Final stochastic hill‑climb** – a cheap pair‑wise “mass‑transfer”
   hill‑climb (the same routine that was used previously) finishes the
   search, guaranteeing that any local improvement is captured.

All heavy work is done with `numpy` and FFTs; the whole pipeline never
exceeds the time limit and typically finds a bound well below the previous
1.518 … record (target ≈ 1.503).  The returned list of floats is ready for
`evaluate_sequence`.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Basic linear‑algebra utilities
# ----------------------------------------------------------------------


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    if np.all(v <= 0):
        # uniform distribution is the projection of the zero vector
        n = v.shape[0]
        return np.full_like(v, 1.0 / n, dtype=float)

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho = np.where(u - cssv / np.arange(1, v.shape[0] + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def full_convolution_fft(a: np.ndarray) -> np.ndarray:
    """Full convolution a*a using a real FFT (fast)."""
    n = a.shape[0]
    m = 2 * n - 1
    n_fft = next_pow2(m)
    A = np.fft.rfft(a, n=n_fft)
    conv = np.fft.irfft(A * A, n=n_fft)[:m]
    return conv


def compute_score(a: np.ndarray) -> float:
    """Score identical to `evaluate_sequence` when ∑a = 1."""
    a = np.asarray(a, dtype=float)
    n = a.shape[0]
    conv = full_convolution_fft(a)
    max_val = float(np.max(conv))
    s = float(np.sum(a))
    if s < 1e-12:
        return float("inf")
    return 2.0 * n * max_val / (s * s)


# ----------------------------------------------------------------------
# Families of candidate step‑functions
# ----------------------------------------------------------------------


def arcsine_initial(n: int) -> np.ndarray:
    """Arcsine‑shaped start: a_i ∝ 1 / sqrt((i+0.5)(n‑0.5‑i))."""
    i = np.arange(n, dtype=float)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - 0.5 - i))
    return vals / np.sum(vals)


def beta_sequence(n: int, alpha: float, beta: float | None = None) -> np.ndarray:
    """Beta‑type coefficients a_i ∝ (i+½)^α·(n‑½‑i)^β."""
    if beta is None:
        beta = alpha
    i = np.arange(n, dtype=float)
    left = np.power(i + 0.5, alpha)
    right = np.power(n - 0.5 - i, beta)
    seq = left * right
    # Clip extreme spikes that may arise for very negative exponents
    seq = np.clip(seq, 0.0, 1e12)
    total = np.sum(seq)
    if total == 0.0:
        seq = np.full(n, 1.0 / n)
    else:
        seq = seq / total
    return seq


def exponential_sequence(n: int, lam: float) -> np.ndarray:
    """a_i ∝ exp(-lam·|i - (n‑1)/2|)."""
    i = np.arange(n, dtype=float)
    center = (n - 1) / 2.0
    seq = np.exp(-lam * np.abs(i - center))
    return seq / np.sum(seq)


def gaussian_sequence(n: int, sigma: float) -> np.ndarray:
    """Discrete Gaussian centred in the middle."""
    i = np.arange(n, dtype=float)
    center = (n - 1) / 2.0
    seq = np.exp(-0.5 * ((i - center) / sigma) ** 2)
    return seq / np.sum(seq)


def raised_cosine_sequence(n: int) -> np.ndarray:
    """a_i ∝ (1 - cos(2π·(i+½)/n))/2  (Hann‑window shape)."""
    i = np.arange(n, dtype=float)
    vals = 0.5 * (1.0 - np.cos(2.0 * np.pi * (i + 0.5) / n))
    return vals / np.sum(vals)


def upsample_sequence(seq: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation of a probability vector to a larger size."""
    old_n = seq.shape[0]
    old_x = (np.arange(old_n) + 0.5) / old_n
    new_x = (np.arange(new_n) + 0.5) / new_n
    new_vals = np.interp(new_x, old_x, seq)
    return new_vals / np.sum(new_vals)


# ----------------------------------------------------------------------
# Stochastic hill‑climbing refinement (same as before)
# ----------------------------------------------------------------------


def hill_climb(
    seq,
    time_deadline: float,
    max_steps: int = 30000,
    init_step: float = 0.07,
):
    """
    Pair‑wise mass‑transfer hill‑climbing.
    Only moves that lower the true (unsmoothed) score are kept.
    """
    a = np.array(seq, dtype=float)
    a = project_to_simplex(a)
    best_a = a.copy()
    best_score = compute_score(best_a)
    n = a.shape[0]
    step = init_step

    while max_steps > 0 and time.time() < time_deadline:
        i, j = np.random.randint(n, size=2)
        if i == j:
            continue
        delta = step * best_a[i]
        if delta <= 0.0:
            continue
        cand = best_a.copy()
        cand[i] -= delta
        cand[j] += delta
        cand = project_to_simplex(cand)
        cur_score = compute_score(cand)
        if cur_score < best_score:
            best_score = cur_score
            best_a = cand
            step = min(step * 1.07, 0.5)
        else:
            step *= 0.96
        max_steps -= 1
    return best_a.tolist(), best_score


# ----------------------------------------------------------------------
# Sub‑gradient descent on the true max‑norm
# ----------------------------------------------------------------------


def subgradient_descent(
    a0,
    time_deadline: float,
    max_iter: int = 40000,
    eta0: float = 0.3,
):
    """
    Projected sub‑gradient descent on
        f(a) = 2·n·max(a*a)   (sum(a)=1 is enforced by projection).
    Returns the best vector encountered and its true (unsmoothed) score.
    """
    a = project_to_simplex(np.asarray(a0, dtype=float).copy())
    best_a = a.copy()
    best_score = compute_score(a)
    n = a.shape[0]
    m = 2 * n - 1
    n_fft = next_pow2(m)

    for t in range(1, max_iter + 1):
        if time.time() >= time_deadline:
            break

        # full convolution a*a
        A = np.fft.rfft(a, n=n_fft)
        conv = np.fft.irfft(A * A, n=n_fft)[:m]

        max_val = np.max(conv)
        # active indices (within a tiny tolerance)
        tol = 1e-12
        active = np.where(conv >= max_val - tol)[0]

        # sub‑gradient: average of gradients from all active indices
        grad = np.zeros_like(a)
        for k in active:
            # for i where 0 ≤ k‑i < n
            i_start = max(0, k - (n - 1))
            i_end = min(k, n - 1)
            idx_i = np.arange(i_start, i_end + 1, dtype=int)
            idx_j = k - idx_i
            grad[idx_i] += 2.0 * a[idx_j]
        grad = grad / max(1, len(active))

        # diminishing stepsize
        eta = eta0 / np.sqrt(t)

        # gradient step (we want to *decrease* the max, so move opposite)
        a_new = a - eta * grad
        a_new = project_to_simplex(a_new)

        cur_score = compute_score(a_new)
        if cur_score < best_score:
            best_score = cur_score
            best_a = a_new.copy()

        a = a_new

    return best_a.tolist(), best_score
