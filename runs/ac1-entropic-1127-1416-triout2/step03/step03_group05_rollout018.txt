def _project_simplex(v):
    # v can contain negative; returns projection onto simplex {x >=0, sum=1}
    # algorithm from Duchi et al.
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(len(v)) + 1
    cond = u - cssv / ind > 0
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0)
    return w
```

def propose_candidate(budget_s=1000):
    import time, numpy as np

    rng = np.random.default_rng(12345)
    deadline = time.time() + budget_s - 5

    try:
        prev = np.asarray(height_sequence_1, dtype=float)
        n = prev.shape[0]
    except Exception:
        prev = None
        n = 5146

    best_seq = None
    best_score = float('inf')
    restart = 0

    # define init strategies functions
    def init_sequence(kind):
        if kind == "prev" and prev is not None:
            seq = prev.copy()
            if seq.shape[0] != n:
                if seq.shape[0] > n:
                    seq = seq[:n]
                else:
                    seq = np.concatenate([seq, np.zeros(n - len(seq))])
            seq = np.maximum(seq, 0.0)
            if seq.sum() == 0:
                seq = np.full(n, 1.0/n)
            else:
                seq = seq / seq.sum()
            return seq
        if kind == "uniform":
            return np.full(n, 1.0/n)
        if kind == "random_dirichlet":
            alpha = 0.5
            sample = rng.gamma(alpha, 1.0, size=n)
            return sample / sample.sum()
        if kind == "geometric":
            # choose alpha in range e.g., 2..6
            a_param = rng.uniform(1.0, 6.0)
            r = np.exp(-a_param / n)
            # compute c
            denom = 1.0 - r**n
            c = (1.0 - r) / denom
            seq = c * (r ** np.arange(n))
            return seq / seq.sum()
        if kind == "half_sine":
            x = np.arange(1, n+1)
            seq = np.sin(np.pi * x / (n+1))
            seq = np.maximum(seq, 0.0)
            return seq / seq.sum()
        raise ValueError

    init_kinds = ["prev","uniform","random_dirichlet","geometric","half_sine"]
    # Use while time remains:
    while time.time() < deadline - 5:
        kind = init_kinds[restart % len(init_kinds)]
        a = init_sequence(kind)
        a_opt, score_opt = _optimize_with_smoothing(a, deadline, restart)
        if score_opt < best_score:
            best_score = score_opt
            best_seq = a_opt.copy()
            print(f"New global best after restart {restart}: {best_score:.6f}")
        if best_score < 1.503:
            print(f"Target reached with score {best_score:.6f}")
            break
        restart += 1

    if best_seq is None:
        best_seq = np.full(n, 1.0/n)

    return [float(v) for v in best_seq.tolist()]

"""
A smooth‑max gradient descent for the step‑function problem.

The objective for a non‑negative sequence `a` (∑a = 1) is
    score(a) = 2·n·max_k (a * a)[k] .
The max can be approximated by a smooth log‑sum‑exp:
    Fβ(a) = (1/β)·log Σ_k exp(β·(a*a)[k])   (β>0).

∂Fβ/∂a = 2·Σ_k w_k·a_{k‑i} with
    w_k = exp(β·b_k) / Σ_j exp(β·b_j) ,   b = a*a .
The weighted sum Σ_k w_k·a_{k‑i} can be computed by a single FFT‑based
convolution of `a` with the reversed weight vector `w_rev`.  After each
gradient step the vector is projected onto the probability simplex
(non‑negative, sum = 1).

The optimiser runs many short restarts with different initialisations
(uniform, Dirichlet, geometric, half‑sine, previous best).  A simple
learning‑rate schedule and an increasing β turn the smooth approximation
into the true max, while occasional random jitter helps escape plateaus.
The best sequence seen before the time budget expires is returned.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Simplex projection (Duchi et al. 2008)
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Project vector v onto the probability simplex {x≥0, Σx=1}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all entries are zero – return uniform
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# FFT‑based convolution utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _conv_full(a: np.ndarray, b: np.ndarray, fft_len: int) -> np.ndarray:
    """Full convolution of a and b using a pre‑computed FFT length."""
    fa = np.fft.rfft(a, n=fft_len)
    fb = np.fft.rfft(b, n=fft_len)
    conv = np.fft.irfft(fa * fb, n=fft_len)
    return conv[: a.shape[0] + b.shape[0] - 1]


# ----------------------------------------------------------------------
# Smooth‑max gradient (vectorised, FFT based)
# ----------------------------------------------------------------------
def _smooth_max_grad(a: np.ndarray, beta: float,
                    conv_len: int, fft_len_a: int,
                    fft_len_full: int) -> tuple[np.ndarray, float]:
    """
    Compute the gradient of the smooth max (log‑sum‑exp) of the aperiodic
    convolution a*a, and also return the true max‑convolution value.
    Returns (grad, max_val).
    """
    # ----- a*a ---------------------------------------------------------
    A_fft = np.fft.rfft(a, n=fft_len_a)
    conv = np.fft.irfft(A_fft * A_fft, n=fft_len_a)[:conv_len]
    max_val = conv.max()

    # ----- smooth weights ------------------------------------------------
    # w_k ∝ exp(beta * (b_k - max_val))  (numerical stability)
    exp_w = np.exp(beta * (conv - max_val))
    w = exp_w / exp_w.sum()                 # shape (conv_len,)

    # ----- gradient = 2 * Σ_k w_k * a_{k-i} ---------------------------
    w_rev = w[::-1]                         # reverse for the required sum
    # convolution a * w_rev  (full length = 3n-2)
    A_full = np.fft.rfft(a, n=fft_len_full)
    W_full = np.fft.rfft(w_rev, n=fft_len_full)
    full_conv = np.fft.irfft(A_full * W_full, n=fft_len_full)
    # desired slice: indices n-1 … 2n-2 (inclusive), then reversed
    n = a.shape[0]
    slice_ = full_conv[n - 1: 2 * n - 1][::-1]   # length n
    grad = 2.0 * slice_
    return grad, max_val


# ----------------------------------------------------------------------
# Optimisation driver – smooth‑max gradient descent with restarts
# ----------------------------------------------------------------------
def _optimize_smooth(a0: np.ndarray,
                     deadline: float,
                     restart_id: int,
                     target: float = 1.503) -> tuple[np.ndarray, float]:
    """
    Smooth‑max gradient descent starting from a0.
    Returns the best sequence seen before *deadline* and its true score.
    """
    a = a0.copy()
    n = a.shape[0]

    # FFT lengths (fixed for the whole optimisation)
    conv_len = 2 * n - 1
    fft_len_a = _next_pow2(conv_len)          # for a*a
    fft_len_full = _next_pow2(3 * n - 2)       # for a * w_rev

    best_a = a.copy()
    # initial true score
    _, max0 = _smooth_max_grad(a, beta=1.0,
                               conv_len=conv_len,
                               fft_len_a=fft_len_a,
                               fft_len_full=fft_len_full)
    best_score = 2.0 * n * max0

    it = 0
    stagnation = 0
    # learning‑rate schedule
    eta0 = 0.12
    # smoothing parameter β schedule (starts low, ends high)
    beta = 0.5
    beta_target = 20.0
    beta_inc = (beta_target - beta) / 30000.0   # increase slowly

    while time.time() < deadline:
        it += 1

        # increase smoothing gradually
        if beta < beta_target:
            beta = min(beta_target, beta + beta_inc)

        # gradient of smooth max and true maximum convolution value
        grad, cur_max = _smooth_max_grad(a, beta,
                                         conv_len,
                                         fft_len_a,
                                         fft_len_full)
        cur_score = 2.0 * n * cur_max

        # keep best ever seen

            best_score = cur_score
            best_a = a.copy()
            stagnation = 0
            if it % 500 == 0:
                print(f"Restart {restart_id:02d} it {it:6d}: best = {best_score:.6f}")
        else:
            stagnation += 1

        # --- termination criteria ---------------------------------------
        if best_score < target:
            break
        if stagnation > 2000:
            # random jitter to escape plateaus
            a += 1e-3 * np.random.default_rng().normal(size=n)
            a = _project_simplex(a)
            stagnation = 0

        # --- gradient step (projected) ---------------------------------
        eta = eta0 / np.sqrt(it)
        a_new = a - eta * grad
        a = _project_simplex(a_new)

    return best_a, best_score


# ----------------------------------------------------------------------
# Initialisation helpers
# ----------------------------------------------------------------------
def _initial_sequence(kind: str, n: int, rng: np.random.Generator,
                      prev_seq: np.ndarray | None) -> np.ndarray:
    """Generate a feasible probability vector of length n."""
    if kind == "prev" and prev_seq is not None:
        seq = prev_seq.copy()
        if seq.shape[0] != n:
            if seq.shape[0] > n:
                seq = seq[:n]
            else:
                seq = np.concatenate([seq,
                                      np.zeros(n - seq.shape[0],
                                               dtype=seq.dtype)])
        seq = np.maximum(seq, 0.0)
        if seq.sum() == 0.0:
            return np.full(n, 1.0 / n)
        return seq / seq.sum()

    if kind == "uniform":
        return np.full(n, 1.0 / n)

    if kind == "random_dirichlet":
        # concentration α<1 yields sparse vectors
        alpha = 0.5
        sample = rng.gamma(alpha, 1.0, size=n)
        return sample / sample.sum()

    if kind == "geometric":
        # pick a random “decay” parameter α∈[1,6]  →  r = exp(-α/n)
        alpha = rng.uniform(1.0, 6.0)
        r = np.exp(-alpha / n)
        denom = 1.0 - r ** n
        c = (1.0 - r) / denom
        seq = c * (r ** np.arange(n))
        return seq / seq.sum()

    if kind == "half_sine":
        # a_i ∝ sin(π·(i+1)/(n+1))   (positive, peaks in the middle)
        idx = np.arange(1, n + 1)
        seq = np.sin(np.pi * idx / (n + 1))
        return seq / seq.sum()

    # fallback
    return np.full(n, 1.0 / n)

        n = previous.shape[0]

        n = 5146                                   # default length from earlier runs

    best_overall_a = None
    best_overall_score = float('inf')
    restart_id = 0

    # strategy list (order matters for deterministic cycling)
    strategies = ["prev", "uniform", "random_dirichlet", "geometric", "half_sine"]

    while time.time() < deadline - 2.0:
        kind = strategies[restart_id % len(strategies)]
        # avoid using "prev" when we have no previous solution
        if kind == "prev" and previous is None:
            kind = "uniform"

        a0 = _initial_sequence(kind, n, rng, previous)

        a_opt, score_opt = _optimize_smooth(a0, this_deadline, restart_id)

        if score_opt < best_overall_score:
            best_overall_score = score_opt
            best_overall_a = a_opt.copy()
            print(f"Global best after restart {restart_id:02d}: {best_overall_score:.6f}")

        if best_overall_score < 1.503:
            print(f"Target achieved (score={best_overall_score:.6f}) after {restart_id+1} restarts.")
            break

        restart_id += 1

    # ------------------------------------------------------------------
    # Fallback guarantee – never return an empty sequence
    # ------------------------------------------------------------------
    if best_overall_a is None:
        best_overall_a = np.full(n, 1.0 / n)

    # Return as plain Python list (the evaluator expects this)
    return [float(v) for v in best_overall_a.tolist()]
