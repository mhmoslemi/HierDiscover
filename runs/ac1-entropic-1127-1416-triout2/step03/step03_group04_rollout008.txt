"""
Optimiser for the step‑function bound

    E(a) = 2·n·max(a∗a) / (∑a)²

The algorithm works in four phases:

1. **Initial shape** – start from the continuous‑arcsine distribution,
   discretised on a modest size (n≈256).  This shape is known to give a flat
   auto‑convolution and is a good starting point.

2. **Linear‑programming refinement** – given the current vector `f` we solve
   the LP

        maximise    ∑ g_i
        subject to  f * g ≤ M          (component‑wise)
                    g_i ≥ 0

   where `M = max(f∗f)`.  The solution `g` concentrates mass where `f` is
   smallest; a small convex combination `f←(1‑t)f + t ĝ` (with `ĝ` renormalised
   to the probability simplex) strictly reduces the maximal convolution.
   The LP is solved with CVXPY; a few dozen iterations are cheap for
   n≈256.

3. **Upsampling** – if time permits the best vector is linearly interpolated
   to a large size (n=8192).  Upsampling does not change the objective
   (scale‑invariance) but reduces discretisation error.

4. **Final stochastic hill‑climb** – a cheap pair‑wise mass‑transfer hill‑climb
   on the true (unsmoothed) objective finishes the search.

The whole search respects the 1000‑second budget and returns a list of
non‑negative floats ready for ``evaluate_sequence``.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Basic utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full linear convolution using a real FFT."""
    m = len(a) + len(b) - 1
    n_fft = next_pow2(m)
    A = np.fft.rfft(a, n=n_fft)
    B = np.fft.rfft(b, n=n_fft)
    C = A * B
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


def compute_score(a: np.ndarray) -> float:
    """Exact score (identical to evaluate_sequence when ∑a = 1)."""
    a = np.asarray(a, dtype=float)
    n = len(a)
    conv = convolution_fft(a, a)
    max_val = float(np.max(conv))
    s = float(np.sum(a))
    if s < 1e-12:
        return float("inf")
    return 2.0 * n * max_val / (s * s)


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    if np.all(v <= 0):
        n = v.shape[0]
        return np.full_like(v, 1.0 / n, dtype=float)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho = np.where(u - cssv / np.arange(1, v.shape[0] + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def arcsine_initial(n: int) -> np.ndarray:
    """Discretised arcsine density a_i ∝ 1/√((i+½)(n‑½‑i))."""
    i = np.arange(n, dtype=float)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - 0.5 - i))
    return vals / np.sum(vals)


def upsample_sequence(seq: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation of a probability vector to a larger size."""
    old_n = len(seq)
    old_x = (np.arange(old_n) + 0.5) / old_n
    new_x = (np.arange(new_n) + 0.5) / new_n
    new_vals = np.interp(new_x, old_x, seq)
    new_vals = np.maximum(new_vals, 0.0)
    return new_vals / np.sum(new_vals)


# ----------------------------------------------------------------------
# Linear‑programming refinement utilities
# ----------------------------------------------------------------------


def build_conv_matrix(a: np.ndarray) -> np.ndarray:
    """
    Build the (2n‑1)×n matrix A such that (A @ g)[k] = (a * g)[k].
    Entry A[k,i] = a[i] if 0 ≤ k‑i < n else 0.
    """
    n = len(a)
    rows = np.arange(2 * n - 1)[:, None]          # shape (2n‑1,1)
    cols = np.arange(n)[None, :]                  # shape (1,n)
    mask = (rows - cols >= 0) & (rows - cols < n)
    A = np.where(mask, a[cols], 0.0)
    return A


def solve_lp_for_g(a: np.ndarray, M: float):
    """
    Solve
        maximise    ∑ g_i
        subject to  a * g ≤ M  (component‑wise)
                    g_i ≥ 0
    Returns the optimal g as a NumPy array (or None on failure).
    """
    try:
        import cvxpy as cp
    except Exception:
        return None

    A = build_conv_matrix(a)
    n = len(a)
    g = cp.Variable(n, nonneg=True)
    constraints = [A @ g <= M]
    prob = cp.Problem(cp.Maximize(cp.sum(g)), constraints)
    # ECOS is a fast LP solver; we use a modest iteration limit.
    prob.solve(solver=cp.ECOS, warm_start=True, max_iters=5000, verbose=False)

    if prob.status not in ("optimal", "optimal_inaccurate"):
        return None
    g_opt = np.asarray(g.value).ravel()
    # Numerical cleanup
    g_opt[g_opt < 0] = 0.0
    return g_opt


def convex_mix(a: np.ndarray, g: np.ndarray, t: float) -> np.ndarray:
    """
    Return the convex combination (1‑t)·a + t·ĝ,
    where ĝ is g renormalised to the simplex.
    The result is projected back onto the simplex to keep feasibility.
    """
    if np.sum(g) == 0:
        return a.copy()
    g_hat = g / np.sum(g)
    mixed = (1.0 - t) * a + t * g_hat
    return project_to_simplex(mixed)


# ----------------------------------------------------------------------
# Stochastic hill‑climbing refinement
# ----------------------------------------------------------------------


def hill_climb(
    seq,
    time_deadline: float,
    max_steps: int = 30000,
    init_step: float = 0.07,
):
    """
    Cheap pair‑wise mass transfer hill‑climb.
    Only moves that strictly improve the true objective are accepted.
    """
    a = np.array(seq, dtype=float)
    a = project_to_simplex(a)
    best_a = a.copy()
    best_score = compute_score(best_a)
    n = len(a)
    step = init_step

    while max_steps > 0 and time.time() < time_deadline:
        i, j = np.random.randint(n, size=2)
        if i == j:
            continue
        delta = step * best_a[i]
        if delta <= 0.0:
            continue
        cand = best_a.copy()
        cand[i] -= delta
        cand[j] += delta
        cand = project_to_simplex(cand)
        cur_score = compute_score(cand)
        if cur_score < best_score:
            best_score = cur_score
            best_a = cand
            step = min(step * 1.07, 0.5)
        else:
            step *= 0.96
        max_steps -= 1
    return best_a.tolist(), best_score


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step‑function that yields a low
    ``evaluate_sequence`` value.  The routine respects the time budget
    (≈1000 s) and uses an LP‑based refinement followed by up‑sampling and
    a final hill‑climb.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 2.0   # safety margin

    # --------------------------------------------------------------
    # Phase 0 – optional fallback (if a known good construction exists)
    # --------------------------------------------------------------
    best_seq = None
    best_val = float("inf")
    try:
        fallback = [float(x) for x in height_sequence_1]  # global may exist
        fallback_val = evaluate_sequence(fallback)
        print(f"Fallback score = {fallback_val:.6f}")
        if fallback_val < best_val:
            best_val = fallback_val
            best_seq = fallback
    except Exception:
        pass

    # --------------------------------------------------------------
    # Phase 1 – initialise with the arcsine shape (moderate length)
    # --------------------------------------------------------------
    n_small = 256
    a = arcsine_initial(n_small)
    a = project_to_simplex(a)          # ensure exact simplex
    best_a = a.copy()
    best_score = compute_score(best_a)
    print(f"Initial score (n={n_small}) = {best_score:.6f}")

    # --------------------------------------------------------------
    # Phase 2 – LP‑based refinement (repeat until time runs out)
    # --------------------------------------------------------------
    t_mix = 0.4          # initial mixing weight
    iter_lp = 0
    while time.time() < deadline - 10:
        # 2a – current bound
        conv = convolution_fft(best_a, best_a)
        M = float(np.max(conv))

        # 2b – solve the LP for a companion vector g
        g_opt = solve_lp_for_g(best_a, M)
        if g_opt is None:
            print("LP could not be solved – exiting LP loop.")
            break

        # 2c – convex mix and project back to the simplex
        new_a = convex_mix(best_a, g_opt, t_mix)
        new_score = compute_score(new_a)

        if new_score + 1e-12 < best_score:      # accept improvement
            best_a = new_a
            best_score = new_score
            print(f"LP iteration {iter_lp:02d}: improved score = {best_score:.6f}")
            # gradually shrink the mixing step to fine‑tune
            t_mix = max(0.05, t_mix * 0.93)
        else:
            # if no improvement, reduce the mixing weight more aggressively
            t_mix *= 0.85
            if t_mix < 0.02:
                # mixing has become ineffective – stop the LP phase
                break

        iter_lp += 1
        if iter_lp >= 50:
            # safeguard against excessive iteration count
            break

    # --------------------------------------------------------------
    # Phase 3 – upsample to a large resolution (if time permits)
    # --------------------------------------------------------------
    if time.time() < deadline - 5:
        large_n = 8192
        print(f"\nUpsampling best vector to n = {large_n}")
        up_seq = upsample_sequence(best_a, large_n)
        up_score = compute_score(up_seq)
        print(f"Score after up‑sampling = {up_score:.6f}")
        if up_score < best_score:
            best_a = up_seq
            best_score = up_score

    # --------------------------------------------------------------
    # Phase 4 – final stochastic hill‑climb (cheap but effective)
    # --------------------------------------------------------------
    if time.time() < deadline - 2:
        print("\nRunning final hill‑climb refinement")
        refined_seq, refined_val = hill_climb(
            best_a.tolist(),
            time_deadline=deadline,
            max_steps=60000,
            init_step=0.05,
        )
        print(f"Hill‑climb score = {refined_val:.6f}")
        if refined_val < best_score:
            best_a = np.array(refined_seq, dtype=float)
            best_score = refined_val

    # --------------------------------------------------------------
    # Safety fallback – should never happen, but guarantees a return
    # --------------------------------------------------------------
    if best_a is None:
        best_a = np.ones(128) / 128.0
        best_score = compute_score(best_a)

    final_score = evaluate_sequence(best_a.tolist())
    print(f"\nSearch finished. Final reported score = {final_score:.6f}")

    return best_a.tolist()
