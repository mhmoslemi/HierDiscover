"""
Search for a non‑negative step‑function that minimises `evaluate_sequence`.

Algorithm
---------

1. **Parameter sweep** –  For several lengths `n` we evaluate the
   one‑parameter family  

        a_i ∝ ((i+½)*(n‑½‑i))^{-p}   (p≈0.30…0.80)

   (the continuous arcsine/Beta shape).  The vectors are simply
   normalised to sum 1 (no projection shift).  The best vector from
   this grid is kept.

2. **Projected Adam** –  The best candidate is refined with a
   projected Adam optimiser on a smooth log‑sum‑exp approximation of
   the max‑convolution.  The smoothing parameter is increased
   homotopically from 1 to a large value, the iterate is projected
   onto the probability simplex after each step.

3. **Stochastic hill‑climbing** –  Finally a cheap mass‑transfer /
   swap hill‑climber is run for the remaining time.  Each move is
   evaluated with the true (unsmoothed) objective and accepted only
   if it improves the score.

All phases respect the 1000 s wall‑clock budget; the function returns
a plain Python list of floats ready for `evaluate_sequence`.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Basic numerical helpers
# ----------------------------------------------------------------------


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of a vector onto the probability simplex."""
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / v.shape[0])

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, v.shape[0] + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.where(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full convolution using a real FFT (fast)."""
    m = len(a) + len(b) - 1
    n_fft = next_pow2(m)
    A = np.fft.rfft(a, n=n_fft)
    B = np.fft.rfft(b, n=n_fft)
    C = A * B
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


def compute_score(a: np.ndarray) -> float:
    """
    Fast replica of `evaluate_sequence` for vectors with sum = 1.
    Returns 2·n·max(a*a).
    """
    n = a.shape[0]
    conv = convolution_fft(a, a)
    return 2.0 * n * float(np.max(conv))


# ----------------------------------------------------------------------
# Smoothed max gradient (Adam optimiser)
# ----------------------------------------------------------------------


def smoothed_gradient(a: np.ndarray, alpha: float):
    """
    Gradient of the soft‑max approximation of max(a*a):
        f_α(a) = (1/α)·log Σ_k exp(α·(a*a)_k)

    Returns the gradient (same shape as a) and the raw convolution.
    """
    n = a.shape[0]
    conv = convolution_fft(a, a)                       # length 2n‑1
    maxc = np.max(conv)
    exp_term = np.exp(alpha * (conv - maxc))
    w = exp_term / np.sum(exp_term)                    # soft‑max weights, sum = 1

    # convolve reversed a with the weight vector
    a_rev = a[::-1]
    conv_rev = convolution_fft(a_rev, w)               # length 3n‑2
    start = n - 1
    rev_slice = conv_rev[start:start + n]              # needed for each component
    w_diag = w[::2][:n]                                # weights of the diagonal entries

    grad = 2.0 * rev_slice - w_diag * a
    return grad, conv


def adam_optimize(
    init_a: np.ndarray,
    time_deadline: float,
    max_iter: int = 200_000,
    eta: float = 0.02,
    beta1: float = 0.9,
    beta2: float = 0.999,
    eps: float = 1e-8,
    alpha_start: float = 1.0,
    alpha_end: float = 2000.0,
    noise_every: int = 800,
    noise_scale: float = 2e-4,
):
    """
    Projected Adam optimiser on the smoothed objective.
    Returns the best vector (ndarray) found and its true (unsmoothed) score.
    """
    a = project_to_simplex(init_a.astype(float).copy())
    m = np.zeros_like(a)
    v = np.zeros_like(a)

    best_a = a.copy()
    best_score = compute_score(a)

    n = len(a)

    for t in range(1, max_iter + 1):
        if time.time() >= time_deadline:
            break

        # homotopy on the smoothing parameter
        alpha = alpha_start + (alpha_end - alpha_start) * min(1.0, t / max_iter)

        grad, _ = smoothed_gradient(a, alpha)

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)

        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)

        step = eta * m_hat / (np.sqrt(v_hat) + eps)
        a_new = a - step

        # occasional isotropic noise to escape shallow plateaus
        if noise_every and (t % noise_every == 0):
            a_new += np.random.randn(n) * noise_scale

        a_new = project_to_simplex(a_new)

        cur_score = compute_score(a_new)
        if cur_score < best_score:
            best_score = cur_score
            best_a = a_new.copy()

        a = a_new

        if t % 5000 == 0:
            print(f"    Adam iter {t:,d} – cur {cur_score:.6f} – best {best_score:.6f}")

    return best_a, best_score


# ----------------------------------------------------------------------
# Simple hill‑climber (mass‑transfer / swap)
# ----------------------------------------------------------------------


def hill_climb_permutation(seq, time_deadline, max_steps=100_000):
    """
    Stochastic hill‑climb that swaps two entries or moves a small fraction of mass.
    The true unsmoothed objective is used for acceptance.
    """
    a = np.array(seq, dtype=float)
    n = a.shape[0]

    best_a = a.copy()
    best_score = compute_score(best_a)

    step_scale = 0.1  # initial fraction of mass transferred in a move

    while max_steps > 0 and time.time() < time_deadline:
        move_type = np.random.rand()
        cand = best_a.copy()

        if move_type < 0.5:
            # simple swap of two distinct indices
            i, j = np.random.choice(n, 2, replace=False)
            cand[i], cand[j] = cand[j], cand[i]
        else:
            # move a fraction of mass from i to j (i != j)
            i, j = np.random.choice(n, 2, replace=False)
            delta = step_scale * best_a[i]
            if delta > 0.0:
                cand[i] -= delta
                cand[j] += delta
                # guard against tiny negative round‑off
                if cand[i] < 0.0:
                    cand[i] = 0.0

        # re‑project onto the simplex to keep feasibility
        cand = project_to_simplex(cand)

        cur_score = compute_score(cand)

        if cur_score < best_score:
            best_score = cur_score
            best_a = cand
            step_scale = min(step_scale * 1.05, 0.5)   # slowly enlarge step
        else:
            step_scale *= 0.97                         # shrink on failure

        max_steps -= 1

    return best_a.tolist(), best_score


# ----------------------------------------------------------------------
# Main entry point required by the evaluation harness
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative sequence that minimises `evaluate_sequence`.
    The routine respects the 1000 s time budget, prints progress
    and finally returns a plain Python list of floats.
    """
    np.random.seed(seed)
    start_time = time.time()
    # keep a tiny safety margin for the harness
    deadline = start_time + budget_s - 2.0

    best_seq = None
    best_score = float("inf")

    # ------------------------------------------------------------------
    # Optional fallback if a previously discovered candidate exists.
    # ------------------------------------------------------------------
    try:
        fallback = [float(x) for x in height_sequence_1]
        fallback_val = evaluate_sequence(fallback)
        print(f"Fallback score = {fallback_val:.6f}")
        if fallback_val < best_score:
            best_score = fallback_val
            best_seq = fallback
    except NameError:
        pass

    # ------------------------------------------------------------------
    # 1. Grid search over a one‑parameter family of Beta‑like shapes.
    # ------------------------------------------------------------------
    print("\n--- Grid search over shape parameter p ---")
    lengths = [1024, 2048, 4096, 8192]          # reasonable sizes for the budget
    p_vals = np.linspace(0.30, 0.80, 26)       # step ≈0.02

    for n in lengths:
        if time.time() >= deadline - 5.0:
            break
        i = np.arange(n, dtype=float)
        base = (i + 0.5) * (n - 0.5 - i) + 1e-12   # product (i+½)(n‑½‑i)
        for p in p_vals:
            w = base ** (-p)                      # raw (un‑normalised) weights
            a = w / w.sum()                       # normalise to sum = 1
            score = compute_score(a)
            if score < best_score:
                best_score = score
                best_seq = a.copy()
                best_n = n
                best_p = p
                print(f"Grid new best → n={n}, p={p:.3f}, score={score:.6f}")

    # ------------------------------------------------------------------
    # 2. Refine the current best with projected Adam optimisation.
    # ------------------------------------------------------------------
    if best_seq is not None and time.time() < deadline - 5.0:
        print("\n--- Adam refinement ---")
        remaining = deadline - time.time()
        # Rough estimate: ~30 Adam iterations per second on this hardware
        max_iter = int(min(600_000, remaining * 30))
        seq_adam, val_adam = adam_optimize(
            init_a=best_seq,
            time_deadline=deadline,
            max_iter=max_iter,
            eta=0.03,
            alpha_start=1.0,
            alpha_end=8000.0,
            noise_every=2000,
            noise_scale=1e-4,
        )
        print(f"Adam finished – score {val_adam:.6f}")
        if val_adam < best_score:
            best_score = val_adam
            best_seq = np.array(seq_adam, copy=True)

    # ------------------------------------------------------------------
    # 3. Final stochastic hill‑climb for the last slice of the budget.
    # ------------------------------------------------------------------
    if best_seq is not None and time.time() < deadline - 1.0:
        print("\n--- Final hill‑climb ---")
        remaining = deadline - time.time()
        # Rough guess: a few thousand hill‑climb steps per second
        max_steps = int(min(150_000, remaining * 3000))
        refined_seq, refined_score = hill_climb_permutation(
            best_seq.tolist(),
            time_deadline=deadline,
            max_steps=max_steps,
        )
        print(f"Hill‑climb result – score {refined_score:.6f}")
        if refined_score < best_score:
            best_score = refined_score
            best_seq = refined_seq

    # ------------------------------------------------------------------
    # Safeguard: if nothing succeeded, fall back to a simple uniform vector.
    # ------------------------------------------------------------------
    if best_seq is None:
        best_seq = [1.0 / 256] * 256
        best_score = evaluate_sequence(best_seq)

    final_score = evaluate_sequence(best_seq)
    print(f"\nFinished search – final reported score = {final_score:.6f}")
    return best_seq
