"""
Propose candidate step heights using a smooth‑max Adam optimizer
followed by a short simulated‑annealing fine‑tuning stage.
The optimizer works in the soft‑max (simplex) parametrisation and
enforces symmetry a_i = a_{n‑1‑i} which is known to be optimal for
the autocorrelation‑maximisation problem.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Return the smallest power of two >= x."""
    return 1 << (x - 1).bit_length()


def simplex_softmax(x: np.ndarray) -> np.ndarray:
    """Stable soft‑max that maps R^n to the probability simplex."""
    mx = np.max(x)
    e = np.exp(x - mx)
    s = e.sum()
    return e / s


def compute_conv(a: np.ndarray, L1: int) -> np.ndarray:
    """
    Linear convolution a * a (full length 2·n‑1) using an FFT of length L1.
    """
    # real‑to‑real FFT for speed
    A = np.fft.rfft(a, n=L1)
    conv_full = np.fft.irfft(A * A, n=L1)
    # keep only the first 2·n‑1 entries (the linear part)
    n = a.shape[0]
    return conv_full[: 2 * n - 1]


def gradient_corr(a: np.ndarray, w: np.ndarray, L2: int) -> np.ndarray:
    """
    Gradient of  (1/α)·log‑sum‑exp(α·conv)  w.r.t. a.
    Returns a vector of length n (the derivative w.r.t. a_i).
    Implements the correlation g = 2·corr(a_rev, w)  where a_rev = a[::-1].
    """
    n = a.shape[0]
    a_rev = a[::-1]
    # convolution via FFT (real input → real output)
    A_rev = np.fft.rfft(a_rev, n=L2)
    W = np.fft.rfft(w, n=L2)
    corr_full = np.fft.irfft(A_rev * W, n=L2)
    # extract the segment that corresponds to shifts 0 … n‑1
    start = n - 1
    g = 2.0 * corr_full[start : start + n]
    return g


def compute_score(a: np.ndarray) -> float:
    """
    Exact value of ``evaluate_sequence`` for a probability vector
    (i.e. sum(a) == 1).  The formula reduces to   2·n·max(a*a) .
    """
    n = a.shape[0]
    conv = compute_conv(a, _next_pow2(2 * n - 1))
    return float(2.0 * n * conv.max())


def beta_sequence(alpha: float, beta: float, n: int) -> np.ndarray:
    """
    Discretised Beta(α,β) density on n points, normalised to sum to 1.
    Mid‑point rule avoids the singular endpoints.
    """
    x = (np.arange(n) + 0.5) / n
    logw = (alpha - 1.0) * np.log(x) + (beta - 1.0) * np.log(1.0 - x)
    logw = np.clip(logw, -700.0, 700.0)
    w = np.exp(logw)
    if not np.isfinite(w.sum()) or w.sum() == 0.0:
        w = np.ones_like(w)
    return w / w.sum()


# ----------------------------------------------------------------------
# Optimisation core
# ----------------------------------------------------------------------


def adam_optimize(
    init_a: np.ndarray,
    n: int,
    L1: int,
    L2: int,
    end_time: float,
    best_global_score: float,
    best_global_a: np.ndarray,
) -> tuple[np.ndarray, float]:
    """
    Adam optimiser on the smooth‐max surrogate.
    Returns the best vector (probability simplex) seen before ``end_time``.
    """
    # initialise log‑softmax parameters
    x = np.log(init_a + 1e-12)
    # enforce symmetry from the start
    x = (x + x[::-1]) * 0.5

    m = np.zeros_like(x)
    v = np.zeros_like(x)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-12
    lr = 0.3                      # base learning rate – tuned empirically
    t = 1

    best_a = best_global_a.copy()
    best_score = best_global_score

    start = time.time()
    # schedule for the smooth‑max temperature
    alpha_start = 4.0
    alpha_end = 200.0

    while time.time() < end_time:
        # ----- simplex point -------------------------------------------------
        a = simplex_softmax(x)
        # enforce symmetry (keeps the sum unchanged)
        a = (a + a[::-1]) * 0.5

        # ----- convolution & objective ---------------------------------------
        conv = compute_conv(a, L1)          # length 2n‑1
        cur_max = conv.max()
        cur_score = 2.0 * n * cur_max

        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            if t % 1000 == 0:
                print(f"Adam iter {t:6d}: new best {best_score:.6f}")

        # ----- smooth‑max weight vector ---------------------------------------
        elapsed = (time.time() - start) / (end_time - start)
        alpha = alpha_start + (alpha_end - alpha_start) * elapsed
        # stable soft‑max over the convolution values
        exps = np.exp(alpha * (conv - cur_max))
        w = exps / exps.sum()

        # ----- gradient of the surrogate --------------------------------------
        g = gradient_corr(a, w, L2)          # shape (n,)

        # ----- back‑prop through the simplex (soft‑max) -----------------------
        avg = np.dot(a, g)                   # ⟨a,g⟩
        grad_x = a * (g - avg)               # ∇_x L

        # ----- Adam update ----------------------------------------------------
        m = beta1 * m + (1.0 - beta1) * grad_x
        v = beta2 * v + (1.0 - beta2) * (grad_x * grad_x)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        x = x - lr * m_hat / (np.sqrt(v_hat) + eps)

        # keep symmetry of the parameters as well
        x = (x + x[::-1]) * 0.5

        t += 1

    return best_a, best_score


def simulated_annealing_refine(
    start_a: np.ndarray,
    n: int,
    end_time: float,
    best_score: float,
    best_a: np.ndarray,
) -> tuple[np.ndarray, float]:
    """
    Very light simulated‑annealing phase that moves mass between two
    coordinates while staying on the simplex.
    """
    cur_a = start_a.copy()
    cur_score = compute_score(cur_a)

    # exponential cooling schedule
    T_start = cur_score * 0.01
    T_end = cur_score * 1e-5

    it = 0

"""
Propose candidate step heights using a smooth‑max Adam optimizer
followed by a short simulated‑annealing fine‑tuning stage.

The optimizer works in the soft‑max (simplex) parametrisation and
enforces symmetry a_i = a_{n‑1‑i}, which is known to be optimal for
the autocorrelation‑maximisation problem.  The combination of a smooth
surrogate for the max‑operator and Adam’s adaptive steps gives a
substantial improvement over the previous sub‑gradient + SA pipeline.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two not smaller than ``x``."""
    return 1 << (x - 1).bit_length()


def simplex_softmax(x: np.ndarray) -> np.ndarray:
    """Stable soft‑max mapping ℝⁿ → {v≥0, Σv=1}."""
    mx = np.max(x)
    e = np.exp(x - mx)
    return e / e.sum()


def compute_conv(a: np.ndarray, L1: int) -> np.ndarray:
    """
    Linear convolution a * a (full length 2·n‑1) via an FFT of length L1.
    """
    A = np.fft.rfft(a, n=L1)
    conv_full = np.fft.irfft(A * A, n=L1)
    n = a.shape[0]
    return conv_full[: 2 * n - 1]


def gradient_corr(a: np.ndarray, w: np.ndarray, L2: int) -> np.ndarray:
    """
    Gradient of the smooth‑max surrogate w.r.t. ``a``.
    Implements g = 2·corr(a_rev, w) where a_rev = a[::-1].
    """
    n = a.shape[0]
    a_rev = a[::-1]
    A_rev = np.fft.rfft(a_rev, n=L2)
    W = np.fft.rfft(w, n=L2)
    corr_full = np.fft.irfft(A_rev * W, n=L2)
    start = n - 1
    return 2.0 * corr_full[start : start + n]


def compute_score(a: np.ndarray) -> float:
    """
    Exact value of ``evaluate_sequence`` for a probability vector
    (i.e. Σa = 1).  The formula reduces to 2·n·max(a*a).
    """
    n = a.shape[0]
    L1 = _next_pow2(2 * n - 1)
    conv = compute_conv(a, L1)
    return float(2.0 * n * conv.max())


def beta_sequence(alpha: float, beta: float, n: int) -> np.ndarray:
    """
    Discretised Beta(α,β) density on ``n`` points, normalised to sum to 1.
    The midpoint rule avoids the singular end‑points.
    """
    x = (np.arange(n) + 0.5) / n
    logw = (alpha - 1.0) * np.log(x) + (beta - 1.0) * np.log(1.0 - x)
    logw = np.clip(logw, -700.0, 700.0)          # avoid overflow
    w = np.exp(logw)
    if not np.isfinite(w.sum()) or w.sum() == 0.0:
        w = np.ones_like(w)
    return w / w.sum()


# ----------------------------------------------------------------------
# Optimisation core
# ----------------------------------------------------------------------


def adam_optimize(
    init_a: np.ndarray,
    n: int,
    L1: int,
    L2: int,
    end_time: float,
    best_global_score: float,
    best_global_a: np.ndarray,
) -> tuple[np.ndarray, float]:
    """
    Adam optimisation on the smooth‑max surrogate.
    Returns the best probability vector encountered before ``end_time``.
    """
    # initialise log‑softmax parameters
    x = np.log(init_a + 1e-12)
    # enforce symmetry from the outset
    x = (x + x[::-1]) * 0.5

    m = np.zeros_like(x)
    v = np.zeros_like(x)
    beta1, beta2 = 0.9, 0.999
    eps = 1e-12
    lr = 0.3                     # learning rate (empirically good)
    t = 1

    best_a = best_global_a.copy()
    best_score = best_global_score

    start = time.time()
    alpha_start, alpha_end = 4.0, 200.0    # smooth‑max temperature schedule

    while time.time() < end_time:
        # ----- current simplex point ----------------------------------------
        a = simplex_softmax(x)
        a = (a + a[::-1]) * 0.5                 # keep symmetry

        # ----- convolution & true objective ---------------------------------
        conv = compute_conv(a, L1)
        cur_max = conv.max()
        cur_score = 2.0 * n * cur_max

        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            if t % 1000 == 0:
                print(f"Adam iter {t:6d}: new best {best_score:.6f}")

        # ----- smooth‑max weight vector --------------------------------------
        elapsed = (time.time() - start) / (end_time - start)
        alpha = alpha_start + (alpha_end - alpha_start) * elapsed
        # stable softmax over the convolution values
        exps = np.exp(alpha * (conv - cur_max))
        w = exps / exps.sum()

        # ----- gradient of the surrogate ------------------------------------
        g = gradient_corr(a, w, L2)            # shape (n,)

        # ----- back‑prop through soft‑max ------------------------------------
        avg = np.dot(a, g)                     # ⟨a,g⟩
        grad_x = a * (g - avg)                 # ∇_x L

        # ----- Adam update ---------------------------------------------------
        m = beta1 * m + (1.0 - beta1) * grad_x
        v = beta2 * v + (1.0 - beta2) * (grad_x * grad_x)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        x = x - lr * m_hat / (np.sqrt(v_hat) + eps)

        # keep parameters symmetric as well
        x = (x + x[::-1]) * 0.5

        t += 1

    return best_a, best_score
