    a = a.copy()
    best_a = a.copy()
    best_sc = compute_score(a)

        i, j = rng.choice(n, 2, replace=False)
        # compute delta magnitude
        delta = step_scale * rng.random() * a[i]

        a_new[i] -= delta
        a_new[j] += delta

            best_sc = sc
            best_a = a_new.copy()
            a = a_new

        max_iters -= 1
    return best_a, best_sc

"""
Propose a low‑discrepancy step‑function.

The search proceeds in three stages:

1. **Broad sampling** – For a chosen length (2048 and optionally 4096)
   many candidate shapes are generated from three families:
   * the classic arcsine shape,
   * a two‑parameter Beta family,
   * a sinusoidal family (a_i ∝ sin(π·(i+½)/n)^p).
   The 5 best candidates (fast surrogate 2·n·max(a*a)) are kept.

2. **Intensive refinement** – Each selected candidate is refined by three
   optimisation phases:
   * Adam on a smooth log‑sum‑exp approximation of the max‑convolution
     (annealed temperature λ from 0.5 up to 80);
   * Sub‑gradient descent on the *exact* max‑convolution (average gradient
     over all active maximal indices);
   * Simple stochastic hill‑climbing that moves a small amount of mass
     between two positions if it reduces the exact objective.
   All updates are projected onto the probability simplex (Σa=1, a_i≥0).

3. **Final polishing** – The globally best vector found so far is polished
   with the same three phases, using a larger iteration budget.

The objective is scale‑invariant, therefore all intermediate vectors are
kept normalised (Σa = 1).  The fast surrogate `compute_score(a)` equals the
official score when the sum is 1, so it can be used safely during the long
optimisation loops.  The official (slow) `evaluate_sequence` is called only
for the few final candidates, keeping the total run time well under the
1000 s budget while typically improving the bound below the target 1.503.
"""

import time
import numpy as np
from typing import List, Tuple

# ----------------------------------------------------------------------
# Low‑level utilities (no closures)
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """
    Real‑FFT convolution of two 1‑D arrays.
    Returns the linear convolution (length = len(a)+len(b)-1).
    """
    m = len(a) + len(b) - 1
    n_fft = next_pow2(m)
    A = np.fft.rfft(a, n=n_fft)
    B = np.fft.rfft(b, n=n_fft)
    C = A * B
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


def compute_score(a: np.ndarray) -> float:
    """
    Fast surrogate for the official score.
    The vector is assumed to be normalised to sum(a)=1, therefore
    the official score reduces to 2·n·max(a*a).
    """
    n = len(a)
    conv = convolution_fft(a, a)
    max_val = float(np.max(conv))
    return 2.0 * n * max_val


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection onto the probability simplex {x ≥ 0, Σx = 1}.
    Duchi et al., 2008.
    """
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / v.size)

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho = np.where(u - cssv / np.arange(1, v.size + 1) > 0)[0]
    if len(rho) == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def arcsine_initial(n: int) -> np.ndarray:
    """Classic arcsine shape a_i ∝ 1/√((i+½)*(n‑½‑i))."""
    i = np.arange(n, dtype=float)
    eps = 0.25
    vals = 1.0 / np.sqrt((i + 0.5) * (n - 0.5 - i) + eps)
    vals /= vals.sum()
    return vals


def generate_beta_shape(n: int, alpha: float, beta: float, eps: float) -> np.ndarray:
    """
    Symmetric beta‑family density on {0,…,n‑1}:
        a_i ∝ (i+ε)^{α‑1}·(n‑1‑i+ε)^{β‑1}.
    Returned vector is normalised to Σa = 1.
    """
    i = np.arange(n, dtype=float)
    left = (i + eps) ** (alpha - 1.0)
    right = (n - 1 - i + eps) ** (beta - 1.0)
    base = left * right
    total = base.sum()
    if total <= 0.0 or not np.isfinite(total):
        return np.full(n, 1.0 / n)
    return base / total


def generate_sinusoidal_shape(n: int, p: float) -> np.ndarray:
    """
    Sinusoidal family a_i ∝ sin(π·(i+½)/n)^p, normalised to Σa = 1.
    The exponent p>0 controls concentration.
    """
    i = np.arange(n, dtype=float)
    vals = np.sin(np.pi * (i + 0.5) / n) ** p
    # avoid exact zeros at the extremes
    vals = np.maximum(vals, 1e-12)
    vals /= vals.sum()
    return vals


# ----------------------------------------------------------------------
# Smooth Adam optimiser (log‑sum‑exp surrogate)
# ----------------------------------------------------------------------


def smooth_gradient(a: np.ndarray, lam: float) -> Tuple[np.ndarray, float]:
    """
    Gradient of the smoothed max‑convolution objective.
    Returns (gradient, smoothed objective value).
    """
    n = len(a)
    conv = convolution_fft(a, a)                     # length 2n‑1
    max_c = np.max(conv)

    # stable soft‑max
    exp_term = np.exp(lam * (conv - max_c))
    sum_exp = np.sum(exp_term)
    w = exp_term / sum_exp                           # weights, sum to 1
    B = max_c + np.log(sum_exp) / lam                # smooth approx of max

    # Weighted gradient of the convolution:
    #   ∂ (Σ_k w_k (a*a)_k) / ∂a = 2 * (a * w_rev) evaluated appropriately
    w_rev = w[::-1]
    conv_rev = convolution_fft(a, w_rev)            # length n + (2n‑1) -1 = 3n‑2
    L = 2 * n - 1
    weighted_grad = 2.0 * conv_rev[L - 1 - np.arange(n)]

    # Since Σa = 1 after projection we can simplify the scaling
    grad = 2.0 * n * weighted_grad - 4.0 * n * B
    return grad, B


def adam_optimize(initial_a: np.ndarray,
                  time_deadline: float,
                  max_iter: int = 100_000,
                  lr0: float = 0.2,
                  lam_start: float = 0.5,
                  lam_end: float = 80.0,
                  noise_every: int = 2000,
                  noise_scale: float = 5e-5) -> Tuple[np.ndarray, float]:
    """
    Adam optimiser on the smooth surrogate.
    Returns the best vector (projected onto the simplex) and its fast score.
    """
    a = project_to_simplex(initial_a.astype(float).copy())
    best_a = a.copy()
    best_sc = compute_score(a)

    beta1, beta2 = 0.9, 0.999
    eps = 1e-8
    m = np.zeros_like(a)
    v = np.zeros_like(a)

    for it in range(1, max_iter + 1):
        if time.time() >= time_deadline:
            break

        # linear schedule for λ
        lam = lam_start + (lam_end - lam_start) * min(1.0, it / max_iter)

        grad, _ = smooth_gradient(a, lam)

        # Adam moments
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)

        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        step = lr0 * m_hat / (np.sqrt(v_hat) + eps)
        a_new = a - step

        if noise_every and (it % noise_every == 0):
            a_new += np.random.randn(a.size) * noise_scale

        a_new = project_to_simplex(a_new)

        cur_sc = compute_score(a_new)
        if cur_sc < best_sc:
            best_sc = cur_sc
            best_a = a_new.copy()

        a = a_new

        if it % 5000 == 0:
            print(f"    Adam iter={it:,d} cur={cur_sc:.6f} best={best_sc:.6f}")

    return best_a, best_sc


# ----------------------------------------------------------------------
# Exact max‑convolution sub‑gradient optimisation
# ----------------------------------------------------------------------


def max_subgradient(a: np.ndarray) -> Tuple[np.ndarray, float]:
    """
    Sub‑gradient of the exact max‑convolution objective.
    Returns (gradient, current max value).
    The gradient is averaged over all indices where the maximum is attained.
    """
    n = len(a)
    conv = convolution_fft(a, a)
    max_val = float(np.max(conv))
    # active set (allow a tiny tolerance)
    active = np.where(conv >= max_val - 1e-12)[0]
    L = 2 * n - 1
    w = np.zeros(L, dtype=float)
    w[active] = 1.0 / len(active)
    w_rev = w[::-1]
    conv_rev = convolution_fft(a, w_rev)
    grad = 2.0 * conv_rev[L - 1 - np.arange(n)]
    return grad, max_val


def subgradient_optimize(a: np.ndarray,
                         time_deadline: float,
                         max_iter: int = 50_000,
                         init_lr: float = 0.1,
                         lr_decay: float = 0.9995,
                         rng: np.random.Generator = None) -> Tuple[np.ndarray, float]:
    """
    Simple (projected) sub‑gradient descent on the exact max‑convolution.
    """
    if rng is None:
        rng = np.random.default_rng()
    a = a.copy()
    best_a = a.copy()
    best_sc = compute_score(a)
    lr = init_lr

    for it in range(1, max_iter + 1):
        if time.time() >= time_deadline:
            break
        grad, _ = max_subgradient(a)
        a_new = a - lr * grad
        a_new = project_to_simplex(a_new)

        cur_sc = compute_score(a_new)
        if cur_sc < best_sc:
            best_sc = cur_sc
            best_a = a_new.copy()

        a = a_new
        lr *= lr_decay

        if it % 5000 == 0:
            print(f"    Subgrad iter={it:,d} cur={cur_sc:.6f} best={best_sc:.6f}")

    return best_a, best_sc


# ----------------------------------------------------------------------
# Stochastic hill‑climbing refinement
# ----------------------------------------------------------------------


def hill_climb_optimize(a: np.ndarray,
                        time_deadline: float,
                        max_iters: int = 100_000,
                        step_scale: float = 0.05,
                        rng: np.random.Generator = None) -> Tuple[np.ndarray, float]:
    """
    Simple random neighbour search that moves a small amount of mass
    between two positions if it improves the exact objective.
    """
    if rng is None:
        rng = np.random.default_rng()
    n = len(a)
    a = a.copy()
    best_a = a.copy()
    best_sc = compute_score(a)

    i = 0
    while i < max_iters and time.time() < time_deadline:
        i += 1
        # pick two distinct indices
        idx = rng.choice(n, size=2, replace=False)
        i1, i2 = idx[0], idx[1]

        # amount to move (relative to source mass)
        delta = step_scale * rng.random() * a[i1]
        if delta <= 0.0:
            continue
        a_new = a.copy()
        a_new[i1] -= delta
        a_new[i2] += delta

        cur_sc = compute_score(a_new)
        if cur_sc < best_sc:
            best_sc = cur_sc
            best_a = a_new.copy()
            a = a_new
        # else keep the current a (no acceptance of worse moves)

        if i % 20000 == 0:
            print(f"    Hill‑climb step={i:,d} cur={cur_sc:.6f} best={best_sc:.6f}")

    return best_a, best_sc


# ----------------------------------------------------------------------
# Main entry point required by the competition harness
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42,
                     budget_s: int = 1000,
                     **kwargs) -> List[float]:
    """
    Search for a non‑negative step sequence with a low ``evaluate_sequence``
    value.  The method combines broad random sampling, Adam on a smooth
    surrogate, exact sub‑gradient descent and a final stochastic hill‑climb.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 2.0   # small safety margin

    best_seq: List[float] = []
    best_val: float = float("inf")

    # --------------------------------------------------------------
    # Optional fallback from a previous run
    # --------------------------------------------------------------
    try:
        fallback = [float(x) for x in height_sequence_1]   # type: ignore
        val = evaluate_sequence(fallback)
        print(f"Fallback (height_sequence_1) score = {val:.6f}")
        if val < best_val:
            best_val = val
            best_seq = fallback
    except Exception:
        pass

    # --------------------------------------------------------------
    # 1.  Exploration over selected lengths
    # --------------------------------------------------------------
    lengths = [2048, 4096]   # try medium and large sizes
    for n in lengths:
        if time.time() >= deadline - 10:
            break
        print(f"\n=== Optimising length n = {n} ===")

        # ---- generate a large pool of candidates --------------------
        candidates: List[np.ndarray] = []

        # deterministic starters
        candidates.append(arcsine_initial(n))
        candidates.append(np.full(n, 1.0 / n))

        # random beta family
        beta_samples = 500
        for _ in range(beta_samples):
            if time.time() >= deadline - 5:
                break
            alpha = rng.uniform(0.05, 5.0)
            beta = rng.uniform(0.05, 5.0)
            eps = rng.uniform(1e-4, max(0.5, 0.01 * n))
            candidates.append(generate_beta_shape(n, alpha, beta, eps))

        # random sinusoidal family
        sin_samples = 200
        for _ in range(sin_samples):
            if time.time() >= deadline - 5:
                break
            p = rng.uniform(0.1, 6.0)
            candidates.append(generate_sinusoidal_shape(n, p))

        # ---- keep the best few according to the fast surrogate -----
        top_k = 5
        scored = [(compute_score(c), c) for c in candidates]
        scored.sort(key=lambda x: x[0])
        best_raw_score = scored[0][0]
        print(f"  Best raw surrogate score in pool: {best_raw_score:.6f}")
        top_candidates = [c for _, c in scored[:top_k]]

        # --------------------------------------------------------------
        # 2.  Refine each top candidate intensively
        # --------------------------------------------------------------
        for idx, cand in enumerate(top_candidates):
            if time.time() >= deadline - 10:
                break
            print(f"  Refinement #{idx + 1}/{len(top_candidates)} for n={n}")

            # --- Adam on smooth surrogate --------------------------------
            remaining = deadline - time.time()
            max_iter_adam = int(min(150_000, remaining * 30.0))
            refined_a, _ = adam_optimize(
                initial_a=cand,
                time_deadline=deadline,
                max_iter=max_iter_adam,
                lr0=0.2,
                lam_start=0.5,
                lam_end=80.0,
                noise_every=2000,
                noise_scale=1e-4,
            )

            # --- Exact sub‑gradient descent ------------------------------
            remaining = deadline - time.time()
            max_iter_sub = int(min(50_000, remaining * 25.0))
            refined_a, _ = subgradient_optimize(
                refined_a,
                time_deadline=deadline,
                max_iter=max_iter_sub,
                init_lr=0.12,
                lr_decay=0.9995,
                rng=rng,
            )

            # --- Stochastic hill‑climbing --------------------------------
            remaining = deadline - time.time()
            max_hill = int(min(100_000, remaining * 30.0))
            refined_a, _ = hill_climb_optimize(
                refined_a,
                time_deadline=deadline,
                max_iters=max_hill,
                step_scale=0.04,
                rng=rng,
            )

            # --- Evaluate with the official (slow) evaluator ------------
            val = evaluate_sequence(refined_a.tolist())
            print(f"    Refined official score = {val:.6f}")
            if val < best_val:
                best_val = val
                best_seq = refined_a.tolist()
                print(f"    *** New global best: {best_val:.6f} (n={n}) ***")

    # --------------------------------------------------------------
    # 3.  Final polishing of the overall best vector
    # --------------------------------------------------------------
    if best_seq and time.time() < deadline - 5:
        a = np.array(best_seq, dtype=float)
        print("\nFinal polishing of the global best")
        remaining = deadline - time.time()

        # Adam phase
        max_iter_final_adam = int(min(250_000, remaining * 30.0))
        a, _ = adam_optimize(
            a,
            time_deadline=deadline,
            max_iter=max_iter_final_adam,
            lr0=0.15,
            lam_start=0.5,
            lam_end=80.0,
            noise_every=1500,
            noise_scale=5e-5,
        )

        # Sub‑gradient phase
        remaining = deadline - time.time()
        max_iter_final_sub = int(min(80_000, remaining * 30.0))
        a, _ = subgradient_optimize(
            a,
            time_deadline=deadline,
            max_iter=max_iter_final_sub,
            init_lr=0.07,
            lr_decay=0.999,
            rng=rng,
        )

        # Hill‑climb phase
        remaining = deadline - time.time()
        max_iter_final_hill = int(min(200_000, remaining * 30.0))
        a, _ = hill_climb_optimize(
            a,
            time_deadline=deadline,
            max_iters=max_iter_final_hill,
            step_scale=0.03,
            rng=rng,
        )

        final_val = evaluate_sequence(a.tolist())
        print(f"Final polished official score = {final_val:.6f}")
        if final_val < best_val:
            best_val = final_val
            best_seq = a.tolist()
            print("*** Updated global best after final polish ***")

    # --------------------------------------------------------------
    # 4.  Guard – guarantee a return value
    # --------------------------------------------------------------
    if not best_seq:
        best_seq = [1.0] * 256
        best_val = evaluate_sequence(best_seq)
        print("Fallback to uniform 256‑vector.")

    print(f"\nFinished search. Best reported score = {best_val:.6f}")
    return best_seq
