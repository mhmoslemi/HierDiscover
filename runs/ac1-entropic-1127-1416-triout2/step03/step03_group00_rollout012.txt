```
def gradient_smooth(a, beta):
    n = a.shape[0]
    M = 2*n - 1
    # compute convolution c = a * a (linear)
    Lc = int(2**np.ceil(np.log2(M)))
    a_fft = np.fft.rfft(a, n=Lc)
    conv_full = np.fft.irfft(a_fft * a_fft, n=Lc)[:M]    # length M
    max_c = conv_full.max()
    # compute weights
    # compute stable exponent
    exp_arg = beta * (conv_full - max_c)
    w = np.exp(exp_arg)
    w_sum = w.sum()
    if w_sum == 0:
        w = np.full_like(w, 1.0 / M)
    else:
        w /= w_sum
    
    # gradient via convolution of w with reversed a
    rev_a = a[::-1]
    Lg = int(2**np.ceil(np.log2(M + n - 1)))
    w_fft = np.fft.rfft(w, n=Lg)
    rev_a_fft = np.fft.rfft(rev_a, n=Lg)
    conv_full2 = np.fft.irfft(w_fft * rev_a_fft, n=Lg)  # length Lg
    # slice gradient: indices n-1 to n-1 + n
    grad = 2.0 * conv_full2[n-1 : n-1 + n]
    return grad, max_c, conv_full
```

```
def optimize_within(a, run_budget):
    start = time.time()
    deadline = start + run_budget
    n = a.shape[0]
    # hyperparameters
    beta_start = 0.5
    beta_end = 15.0
    max_iter = 20000
    eta = 0.5
    best_a = a.copy()
    best_score = evaluate_sequence(best_a.tolist())
    for iter in range(max_iter):
        if time.time() > deadline:
            break
        beta = beta_start + (beta_end - beta_start) * (iter / max_iter)
        grad, max_c, _ = gradient_smooth(a, beta)
        # Step size schedule: decaying
        step = eta / np.sqrt(iter + 1)
        a = simplex_projection(a - step * grad)
        # occasionally compute exact score
        if iter % 200 == 0:
            score = evaluate_sequence(a.tolist())
            if score < best_score:
                best_score = score
                best_a = a.copy()
                print(f"Iter {iter} local best {best_score:.6f}")
    # after gradient, do SA for remainder time
    cur_a = best_a.copy()
    cur_score = best_score
    # simple SA with mass-transfer moves
    T_start = cur_score * 0.01
    T_end = cur_score * 1e-5
    it_sa = 0
    while time.time() < deadline:
        T = max(T_end, T_start * (0.9995 ** it_sa))
        # propose move
        src = np.random.choice(n, p=cur_a)
        tgt = np.random.randint(n-1)
        if tgt >= src:
            tgt += 1
        if cur_a[src] == 0:
            continue
        delta = cur_a[src] * np.random.rand()
        cand = cur_a.copy()
        cand[src] -= delta
        cand[tgt] += delta
        cand_score = evaluate_sequence(cand.tolist())
        if cand_score < cur_score or np.random.rand() < np.exp(-(cand_score - cur_score)/ max(T, 1e-12)):
            cur_a = cand
            cur_score = cand_score
            if cand_score < best_score:
                best_score = cand_score
                best_a = cand.copy()
                print(f"SA iter {it_sa} new best {best_score:.6f}")
        it_sa += 1
    return best_a, best_score
```

"""
Propose a step‑function (non‑negative heights) that lowers the evaluation
function

    2·n·max(a*a) / (sum(a)^2)

by directly optimising the shape of a probability vector a (sum(a)=1).
The algorithm consists of:

1.  A fast projected gradient descent on a smooth approximation of the
    ℓ∞‑norm of the auto‑convolution (log‑sum‑exp with a schedule for the
    smoothing parameter β).  The gradient is obtained analytically and
    evaluated with FFTs; each iterate is projected onto the probability
    simplex.

2.  A short simulated‑annealing (SA) refinement that performs mass‑transfer
    moves while keeping the sum exactly one.  The SA phase uses the same
    FFT‑based max‑convolution routine to keep the cost low.

Multiple independent restarts with different lengths and random Beta‑shaped
initialisations are performed until the time budget expires.  The best
sequence found is returned as a plain ``list[float]``.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities
# ----------------------------------------------------------------------


def simplex_projection(v: np.ndarray) -> np.ndarray:
    """ Euclidean projection of v onto the probability simplex {w≥0, Σw=1}. """
    v = np.asarray(v, dtype=float)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):                     # all non‑positive → uniform
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1] - 1                  # zero‑based index
    theta = cssv[cond][-1] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def beta_sequence(alpha: float, beta: float, n: int) -> np.ndarray:
    """ Discrete Beta(α,β) density on n points (mid‑point rule, normalised). """
    x = (np.arange(n) + 0.5) / n
    logw = (alpha - 1.0) * np.log(x) + (beta - 1.0) * np.log(1.0 - x)
    # Prevent overflow of the exponentials
    logw = np.clip(logw, -700.0, 700.0)
    w = np.exp(logw)
    if not np.isfinite(w.sum()) or w.sum() == 0.0:
        w = np.ones_like(w)
    return w / w.sum()


def next_pow2(x: int) -> int:
    """ Smallest power of two ≥ x (useful for FFT padding). """
    return 1 << (x - 1).bit_length()


def max_convolution(a: np.ndarray) -> float:
    """ Return max_k (a*a)_k for a probability vector a (linear convolution). """
    n = a.shape[0]
    m = 2 * n - 1
    L = next_pow2(m)
    af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(af * af, n=L)[:m]
    return float(conv.max())


def gradient_smooth(a: np.ndarray, beta: float):
    """
    Compute gradient of the smoothed max (log‑sum‑exp) objective.
    Returns (grad, max_c, conv) where conv = a*a (linear convolution).
    """
    n = a.shape[0]
    m = 2 * n - 1                     # length of the linear convolution
    # ----- convolution a*a -------------------------------------------------
    Lc = next_pow2(m)
    af = np.fft.rfft(a, n=Lc)
    conv = np.fft.irfft(af * af, n=Lc)[:m]          # shape (m,)
    max_c = conv.max()

    # ----- soft‑max weights -----------------------------------------------
    # w_i ∝ exp(beta * (conv_i - max_c))  (all ≤ 1, stable)
    exp_arg = beta * (conv - max_c)                # ≤ 0
    # Clip to avoid under‑flow
    exp_arg = np.clip(exp_arg, -700.0, 0.0)
    w = np.exp(exp_arg)
    w_sum = w.sum()
    if w_sum == 0.0:
        w = np.full_like(w, 1.0 / m)
    else:
        w /= w_sum

    # ----- gradient = 2·(w ⋆ a) -------------------------------------------
    # correlation can be realised as a convolution of w with the reversed a
    rev_a = a[::-1]
    Lg = next_pow2(m + n - 1)          # length for linear convolution w * rev_a
    wf = np.fft.rfft(w, n=Lg)
    rev_af = np.fft.rfft(rev_a, n=Lg)
    conv2 = np.fft.irfft(wf * rev_af, n=Lg)        # length Lg
    # slice the part that corresponds to indices 0 … n‑1
    grad = 2.0 * conv2[n - 1 : n - 1 + n]
    return grad, max_c, conv

# Core optimisation of a single start point
# ----------------------------------------------------------------------


def _optimize_one(start_a: np.ndarray, time_budget: float):
    """
    Perform a gradient‑descent phase followed by a short SA phase.
    Returns (best_a, best_score).
    """
    start_time = time.time()
    deadline = start_time + time_budget

    n = start_a.shape[0]

    # ---------- Phase 1 – smoothed gradient descent -----------------------
    a = start_a.copy()
    best_a = a.copy()
    # exact score for the current a (since sum(a)=1)
    best_score = 2.0 * n * max_convolution(a)

    max_iter = 30000                     # generous upper bound
    eta0 = 0.5                           # base step size
    beta_start, beta_end = 0.5, 20.0    # smoothing schedule

    for it in range(max_iter):
        if time.time() >= deadline - 5.0:      # reserve a few seconds
            break
        # linearly increase β
        beta = beta_start + (beta_end - beta_start) * (it / max_iter)

        grad, max_c, _ = gradient_smooth(a, beta)

        # diminishing step size (√t rule)
        step = eta0 / math.sqrt(it + 1.0)
        a = simplex_projection(a - step * grad)

        # occasional exact evaluation (every 200 iterations)
        if (it % 200) == 0:
            cur_score = 2.0 * n * max_convolution(a)
            if cur_score < best_score:
                best_score = cur_score
                best_a = a.copy()
                print(f"GD iter {it:5d}: new best {best_score:.6f}")

    # ---------- Phase 2 – simulated annealing refinement ------------------
    cur_a = best_a.copy()
    cur_score = best_score

    # simple exponential cooling
    T_start = cur_score * 0.02
    T_end = cur_score * 1e-5
    it_sa = 0

    while time.time() < deadline:
        # temperature according to iteration count
        T = max(T_end, T_start * (0.9995 ** it_sa))

        # ----- propose a mass‑transfer move (keeps ℓ1‑norm = 1) -----
        src = np.random.choice(n, p=cur_a)          # bias towards larger entries
        tgt = np.random.randint(n - 1)
        if tgt >= src:
            tgt += 1
        if cur_a[src] <= 0.0:
            continue
        delta = cur_a[src] * np.random.rand()       # up to 100 % of source mass
        cand = cur_a.copy()
        cand[src] -= delta
        cand[tgt] += delta

        cand_score = 2.0 * n * max_convolution(cand)

        # Metropolis acceptance
        if cand_score < cur_score:
            accept = True
        else:
            accept = np.random.rand() < math.exp(-(cand_score - cur_score) / max(T, 1e-12))

        if accept:
            cur_a = cand
            cur_score = cand_score
            if cand_score < best_score:
                best_score = cand_score
                best_a = cand.copy()
                print(f"SA iter {it_sa:6d}: new best {best_score:.6f}")

        # occasional large diversification
        if it_sa % 50000 == 0 and it_sa > 0:
            # fresh random Beta draw
            tmp = beta_sequence(np.random.uniform(0.5, 5.0),
                                np.random.uniform(0.5, 5.0), n)
            cur_a = tmp / tmp.sum()
            cur_score = 2.0 * n * max_convolution(cur_a)

        it_sa += 1

    return best_a, best_score

    gradient‑descent phase followed by a brief simulated‑annealing refinement.
    """

    start_time = time.time()
    deadline = start_time + budget_s - 0.2   # tiny safety margin

    best_seq = None
    best_score = float("inf")

    # possible problem sizes (chosen to keep FFTs cheap yet give a good bound)
    candidate_lengths = [1024, 2048, 4096, 6144]

    # If a previously‑found sequence exists, add its length to the pool
    try:

            # keep a normalised copy for possible reuse
            existing = existing / existing.sum()
            existing_available = True
        else:
            existing_available = False
    except NameError:
        existing_available = False

    # ------------------------------------------------------------------
    # Main restart loop – run as many independent optimisations as time allows
    # ------------------------------------------------------------------
    while time.time() < deadline:
        # pick a length (biased toward larger problems)
        n = np.random.choice(candidate_lengths)

        # initialise -------------------------------------------------------
        if existing_available and np.random.rand() < 0.2:
            # reuse the previous solution, possibly interpolating to a new size
            if existing.size == n:
                a0 = existing.copy()
            else:

                a0 = np.interp(x_new, x_old, existing)
        else:

            alpha = np.random.uniform(0.3, 5.0)
