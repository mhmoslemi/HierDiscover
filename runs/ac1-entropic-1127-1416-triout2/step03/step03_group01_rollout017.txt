"""
Search algorithm for low max‑convolution step functions.

The routine first draws many random parametrised shapes (beta‑family,
power‑law and exponential) for several lengths, keeping the best
according to a fast surrogate (2·n·max(conv) with the sum normalised to 1).
Each top candidate is then refined by a simple sub‑gradient descent on the
exact non‑smooth objective (max‑convolution) with projection onto the
probability simplex and an optional monotonicity projection.
A final short Adam polishing (soft‑max surrogate) is applied to the
overall best vector.  The method stays well within the 1000 s budget.
"""
import time
import numpy as np
from typing import List, Tuple

def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()

def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Linear convolution using real FFT."""
    m = len(a) + len(b) - 1
    n_fft = next_pow2(m)
    A = np.fft.rfft(a, n=n_fft)
    B = np.fft.rfft(b, n=n_fft)
    C = A * B
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv

def compute_score(a: np.ndarray) -> float:
    """Fast surrogate – assumes a sums to 1."""
    n = len(a)
    conv = convolution_fft(a, a)
    return 2.0 * n * float(np.max(conv))

def arcsine_initial(n: int) -> np.ndarray:
    """Classic arcsine shape (normalised)."""
    i = np.arange(n, dtype=float)
    eps = 0.25
    vals = 1.0 / np.sqrt((i + 0.5) * (n - 0.5 - i) + eps)
    vals /= vals.sum()
    return vals

def generate_beta_shape(n: int, alpha: float, beta: float, eps: float) -> np.ndarray:
    """Symmetric beta‑family density (normalised)."""
    i = np.arange(n, dtype=float)
    left = (i + eps) ** (alpha - 1.0)
    right = (n - 1 - i + eps) ** (beta - 1.0)
    base = left * right
    total = base.sum()
    if total <= 0.0 or not np.isfinite(total):
        return np.full(n, 1.0 / n)
    return base / total

def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection onto the probability simplex."""
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / v.size)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho = np.where(u - cssv / np.arange(1, v.size + 1) > 0)[0]
    if len(rho) == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w

def adam_optimize(initial_a: np.ndarray,
                  time_deadline: float,
                  max_iter: int = 100_000,
                  lr0: float = 0.2,
                  lam_start: float = 5.0,
                  lam_end: float = 30.0,
                  noise_every: int = 2000,
                  noise_scale: float = 5e-5) -> Tuple[np.ndarray, float]:
    """Adam optimiser on the soft‑max surrogate (identical to the previous baseline)."""
    a = project_to_simplex(initial_a.astype(float).copy())
    best_a = a.copy()
    best_sc = compute_score(a)

    def smooth_gradient(a_vec: np.ndarray, lam: float) -> Tuple[np.ndarray, float]:
        n = len(a_vec)
        conv = convolution_fft(a_vec, a_vec)
        max_c = np.max(conv)
        exp_term = np.exp(lam * (conv - max_c))
        sum_exp = np.sum(exp_term)
        w = exp_term / sum_exp
        B = max_c + np.log(sum_exp) / lam

        w_rev = w[::-1]
        conv_rev = convolution_fft(a_vec, w_rev)
        L = 2 * n - 1
        weighted_grad = 2.0 * conv_rev[L - 1 - np.arange(n)]
        s = np.sum(a_vec)
        term1 = (2.0 * n / (s ** 2)) * weighted_grad
        term2 = -(4.0 * n * B / (s ** 3)) * np.ones_like(a_vec)
        grad = term1 + term2
        return grad, B

    beta1, beta2 = 0.9, 0.999
    eps = 1e-8
    m = np.zeros_like(a)
    v = np.zeros_like(a)

    for it in range(1, max_iter + 1):
        if time.time() >= time_deadline:
            break
        lam = lam_start + (lam_end - lam_start) * min(1.0, it / max_iter)
        grad, _ = smooth_gradient(a, lam)
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        step = lr0 * m_hat / (np.sqrt(v_hat) + eps)
        a_new = a - step
        if noise_every and (it % noise_every == 0):
            a_new += np.random.randn(len(a)) * noise_scale
        a_new = project_to_simplex(a_new)
        cur_sc = compute_score(a_new)
        if cur_sc < best_sc:
            best_sc = cur_sc
            best_a = a_new.copy()
        a = a_new
        if it % 5000 == 0:
            print(f"    Adam iter={it:,d}  cur={cur_sc:.6f}  best={best_sc:.6f}")

    return best_a, best_sc

def subgradient_descent(initial_a: np.ndarray,
                        time_deadline: float,
                        max_iter: int = 200_000,
                        step0: float = 0.5,
                        enforce_monotone: bool = True) -> Tuple[np.ndarray, float]:
    """
    Simple sub‑gradient descent on the true max‑convolution objective.
    The vector is kept on the probability simplex (Σa = 1, a_i ≥ 0).
    """
    a = initial_a.astype(float).copy()
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s <= 0:
        a = np.full_like(a, 1.0 / a.size)
    else:
        a = a / s

    best_a = a.copy()
    best_sc = compute_score(a)

    n = len(a)
    rng = np.random.default_rng(0)

    for it in range(1, max_iter + 1):
        if time.time() >= time_deadline:
            break
        # compute convolution and locate the maximal index
        conv = convolution_fft(a, a)
        k_star = int(np.argmax(conv))

        # sub‑gradient of max(conv) w.r.t. a
        i = np.arange(n)
        j = k_star - i
        mask = (j >= 0) & (j < n)
        grad = np.zeros_like(a)
        grad[mask] = 2.0 * a[j[mask]]

        # diminishing step size
        step = step0 / np.sqrt(it)

        a = a - step * grad
        a = np.maximum(a, 0.0)

        if enforce_monotone:
            # enforce non‑increasing shape
            a = np.maximum.accumulate(a[::-1])[::-1]

        total = a.sum()
        if total <= 0:
            a = np.full_like(a, 1.0 / n)
        else:
            a = a / total

        # occasional evaluation
        if it % 200 == 0:
            sc = compute_score(a)
            if sc < best_sc:
                best_sc = sc
                best_a = a.copy()
    return best_a, best_sc

def propose_candidate(seed: int = 42,
                     budget_s: int = 1000,
                     **kwargs) -> List[float]:
    """
    Main entry point.  Returns a list of non‑negative heights with low
    ``evaluate_sequence`` value.  The routine respects the 1000 s time
    budget, uses a mixture of random parametric sampling, sub‑gradient
    refinement and a final Adam polish.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 2.0   # safety margin

    best_seq: List[float] = []
    best_val: float = float("inf")

    # optional fallback from a previously‑found construction
    try:
        fallback = [float(x) for x in height_sequence_1]  # type: ignore
        val = evaluate_sequence(fallback)
        print(f"Fallback (height_sequence_1) score = {val:.6f}")
        if val < best_val:
            best_val = val
            best_seq = fallback
    except Exception:
        pass

    lengths = [256, 512, 1024, 2048, 4096]

    for n in lengths:
        if time.time() >= deadline - 5.0:
            break
        print(f"\n=== Exploring length n = {n} ===")
        # --- 0. deterministic starters ---------------------------------
        best_a = None
        best_score = float("inf")
        starters = [np.full(n, 1.0 / n), arcsine_initial(n)]
        for st in starters:
            sc = compute_score(st)
            if sc < best_score:
                best_score = sc
                best_a = st.copy()

        # --- 1. random parametric sampling ---------------------------
        max_samples = 700
        for _ in range(max_samples):
            if time.time() >= deadline - 5.0:
                break
            shape_type = rng.integers(0, 3)   # 0 beta, 1 power‑law, 2 exponential
            if shape_type == 0:
                # Beta family – wide log‑uniform ranges
                alpha = 10 ** rng.uniform(-2, 1)   # 0.01 … 10
                beta = 10 ** rng.uniform(-2, 1)
                eps = rng.uniform(0.0, max(1.0, 0.02 * n))
                cand = generate_beta_shape(n, alpha, beta, eps)
            elif shape_type == 1:
                # Power‑law a_i ∝ (i+δ)^{-p}
                p = rng.uniform(0.1, 5.0)
                delta = rng.uniform(0.0, 5.0)
                i = np.arange(n, dtype=float)
                vals = (i + delta) ** (-p)
                cand = vals / vals.sum()
            else:
                # Exponential decay a_i ∝ exp(-γ i)
                gamma = rng.uniform(0.0005, 0.02)
                i = np.arange(n, dtype=float)
                vals = np.exp(-gamma * i)
                cand = vals / vals.sum()
            sc = compute_score(cand)
            if sc < best_score:
                best_score = sc
                best_a = cand.copy()

        # --- 2. refinement with sub‑gradient descent -----------------
        if best_a is not None and time.time() < deadline - 5.0:
            remaining = deadline - time.time()
            # allocate a fraction of remaining time to this length
            max_iter = int(min(150_000, remaining * 30))
            print(f"  Sub‑gradient refinement: max_iter={max_iter}")
            refined_a, refined_sc = subgradient_descent(best_a,
                                                       time_deadline=deadline,
                                                       max_iter=max_iter,
                                                       step0=0.5,
                                                       enforce_monotone=False)
            # official evaluation
            val = evaluate_sequence(refined_a.tolist())
            print(f"  Refined official score = {val:.6f}")
            if val < best_val:
                best_val = val
                best_seq = refined_a.tolist()
                print(f"  *** New global best: {best_val:.6f} (n={n}) ***")
        else:
            # if we could not refine, just test the raw best
            if best_a is not None:
                val = evaluate_sequence(best_a.tolist())
                print(f"  Raw candidate score = {val:.6f}")
                if val < best_val:
                    best_val = val
                    best_seq = best_a.tolist()
                    print(f"  *** New global best (raw) ***")

    # --- 3. Final polishing with Adam (soft‑max surrogate) --------
    if best_seq and time.time() < deadline - 5.0:
        a = np.array(best_seq, dtype=float)
        remaining = deadline - time.time()
        max_iter = int(min(250_000, remaining * 30))
        print(f"\nFinal polishing (Adam) – max_iter={max_iter}")
        a_refined, _ = adam_optimize(initial_a=a,
                                      time_deadline=deadline,
                                      max_iter=max_iter,
                                      lr0=0.15,
                                      lam_start=5.0,
                                      lam_end=30.0,
                                      noise_every=1000,
                                      noise_scale=3e-5)
        final_val = evaluate_sequence(a_refined.tolist())
        print(f"Final polished official score = {final_val:.6f}")
        if final_val < best_val:
            best_val = final_val
            best_seq = a_refined.tolist()
            print("*** Updated global best after final polish ***")

    # --- 4. Guard against empty result -----------------------------
    if not best_seq:
        best_seq = [1.0] * 256
        best_val = evaluate_sequence(best_seq)

    print(f"\nFinished search. Best reported score = {best_val:.6f}")
    return best_seq

"""
Search for a step‑function with a low max‑convolution value.

The algorithm proceeds in four stages:

1. **Parameterised random sampling** – for several lengths (256 … 4096)
   a large number of candidate shapes are drawn from three flexible families:
   * a beta‑density a_i ∝ (i+ε)^{α‑1}(n‑1‑i+ε)^{β‑1}
   * a power‑law a_i ∝ (i+δ)^{‑p}
   * an exponential a_i ∝ exp(‑γ i)

   The parameters are sampled on a log‑uniform scale (α,β∈[0.01,10],
   p∈[0.1,5], γ∈[5e‑4,2·10⁻²]), normalising each vector to Σa=1.
   The fastest surrogate for the official objective,
   `2·n·max(a*a)`, is used to keep the best candidate for each length.

2. **Sub‑gradient refinement** – the best candidate of a length is refined
   by a simple sub‑gradient descent on the exact non‑smooth objective.
   For a vector `a` the gradient of `max(conv(a,a))` is obtained from the
   shift attaining the maximum; after each step the vector is projected onto
   the probability simplex (Σa=1, a_i≥0).  Optional monotonicity enforcement
   can be turned on but is disabled by default.

3. **Final polishing** – the overall best vector is given a short Adam
   optimisation on a smooth “soft‑max” surrogate (identical to the baseline
   code).  The Adam phase refines the shape without leaving the feasible
   set.

4. **Safety guard** – if nothing is found a uniform 256‑vector is returned.

All helper functions are top‑level (no closures), the routine respects the
1000 s time budget, and the returned list is guaranteed to be non‑negative
and non‑empty, so `evaluate_sequence` never yields `inf`.
"""

import time
import numpy as np
from typing import List, Tuple

# ----------------------------------------------------------------------
# Low‑level utilities (no closures)
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Linear convolution via real FFT (full length)."""
    m = len(a) + len(b) - 1
    n_fft = next_pow2(m)
    A = np.fft.rfft(a, n=n_fft)
    B = np.fft.rfft(b, n=n_fft)
    C = A * B
    conv = np.fft.irfft(C, n=n_fft)[:m]
    return conv


def compute_score(a: np.ndarray) -> float:
    """
    Fast surrogate for the official score.
    Assumes Σa = 1, therefore the denominator cancels and
    the function reduces to 2·n·max(a*a).
    """
    n = len(a)
    conv = convolution_fft(a, a)
    return 2.0 * n * float(np.max(conv))


def arcsine_initial(n: int) -> np.ndarray:
    """Classic arcsine shape (normalised)."""
    i = np.arange(n, dtype=float)
    eps = 0.25
    vals = 1.0 / np.sqrt((i + 0.5) * (n - 0.5 - i) + eps)
    vals /= vals.sum()
    return vals


def generate_beta_shape(n: int, alpha: float, beta: float, eps: float) -> np.ndarray:
    """Beta‑family density on {0,…,n‑1}, normalised to Σ=1."""
    i = np.arange(n, dtype=float)
    left = (i + eps) ** (alpha - 1.0)
    right = (n - 1 - i + eps) ** (beta - 1.0)
    base = left * right
    total = base.sum()
    if total <= 0.0 or not np.isfinite(total):
        return np.full(n, 1.0 / n)
    return base / total


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection onto the probability simplex {x≥0, Σx=1}."""
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / v.size)

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho = np.where(u - cssv / np.arange(1, v.size + 1) > 0)[0]
    if len(rho) == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Adam optimiser (soft‑max surrogate) – identical to the reference baseline
# ----------------------------------------------------------------------


def adam_optimize(initial_a: np.ndarray,
                  time_deadline: float,
                  max_iter: int = 100_000,
                  lr0: float = 0.2,
                  lam_start: float = 5.0,
                  lam_end: float = 30.0,
                  noise_every: int = 2000,
                  noise_scale: float = 5e-5) -> Tuple[np.ndarray, float]:
    """Adam optimiser on the smooth surrogate."""
    a = project_to_simplex(initial_a.astype(float).copy())
    best_a = a.copy()
    best_sc = compute_score(a)

    def smooth_gradient(a_vec: np.ndarray, lam: float) -> Tuple[np.ndarray, float]:
        n = len(a_vec)
        conv = convolution_fft(a_vec, a_vec)
        max_c = np.max(conv)
        exp_term = np.exp(lam * (conv - max_c))
        sum_exp = np.sum(exp_term)
        w = exp_term / sum_exp
        B = max_c + np.log(sum_exp) / lam

        w_rev = w[::-1]
        conv_rev = convolution_fft(a_vec, w_rev)
        L = 2 * n - 1
        weighted_grad = 2.0 * conv_rev[L - 1 - np.arange(n)]

        s = np.sum(a_vec)
        term1 = (2.0 * n / (s ** 2)) * weighted_grad
        term2 = -(4.0 * n * B / (s ** 3)) * np.ones_like(a_vec)
        grad = term1 + term2
        return grad, B

    beta1, beta2 = 0.9, 0.999
    eps = 1e-8
    m = np.zeros_like(a)
    v = np.zeros_like(a)

    for it in range(1, max_iter + 1):
        if time.time() >= time_deadline:
            break
        lam = lam_start + (lam_end - lam_start) * min(1.0, it / max_iter)
        grad, _ = smooth_gradient(a, lam)

        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)

        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        step = lr0 * m_hat / (np.sqrt(v_hat) + eps)
        a_new = a - step

        if noise_every and (it % noise_every == 0):
            a_new += np.random.randn(len(a)) * noise_scale

        a_new = project_to_simplex(a_new)
        cur_sc = compute_score(a_new)

        if cur_sc < best_sc:
            best_sc = cur_sc
            best_a = a_new.copy()
        a = a_new

        if it % 5000 == 0:
            print(f"    Adam iter={it:,d}  cur={cur_sc:.6f}  best={best_sc:.6f}")

    return best_a, best_sc


# ----------------------------------------------------------------------
# Simple sub‑gradient descent on the exact max‑convolution objective
# ----------------------------------------------------------------------


def subgradient_descent(initial_a: np.ndarray,
                        time_deadline: float,
                        max_iter: int = 150_000,
                        step0: float = 0.5,
                        enforce_monotone: bool = False) -> Tuple[np.ndarray, float]:
    """
    Sub‑gradient descent on  max(conv(a,a)).
    The iterate is kept on the probability simplex (Σa = 1, a_i ≥ 0).
    If `enforce_monotone` is True a non‑increasing shape is imposed after
    each step via an isotonic‑regression‑like projection.
    """
    a = np.maximum(initial_a.astype(float).copy(), 0.0)
    s = a.sum()
    if s <= 0:
        a = np.full_like(a, 1.0 / a.size)
    else:
        a /= s

    best_a = a.copy()
    best_sc = compute_score(a)
    n = len(a)

    for it in range(1, max_iter + 1):
        if time.time() >= time_deadline:
            break

        # compute self‑convolution and the index attaining the max
        conv = convolution_fft(a, a)
        k_star = int(np.argmax(conv))

        # sub‑gradient of max(conv) w.r.t. a
        i = np.arange(n)
        j = k_star - i
        mask = (j >= 0) & (j < n)
        grad = np.zeros_like(a)
        grad[mask] = 2.0 * a[j[mask]]

        # diminishing step size
        step = step0 / np.sqrt(it)
        a -= step * grad
        a = np.maximum(a, 0.0)

        if enforce_monotone:
            # project onto the cone of non‑increasing sequences
            a = np.maximum.accumulate(a[::-1])[::-1]

        total = a.sum()
        if total <= 0:
            a = np.full_like(a, 1.0 / n)
        else:
            a /= total

        # periodic evaluation to keep track of the best iterate
        if it % 200 == 0:
            sc = compute_score(a)
            if sc < best_sc:
                best_sc = sc
                best_a = a.copy()

    return best_a, best_sc
