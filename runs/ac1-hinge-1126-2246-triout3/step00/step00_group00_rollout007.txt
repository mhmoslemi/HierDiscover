def propose_candidate(seed=0, budget_s=1000):
    import numpy as np
    import time
    import scipy.optimize

    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5

    best_score = float('inf')
    best_sequence = None

    # candidate lengths
    lengths = [2,3,4,5,6,7,8,9,10,12,15,20,25,30,40,50,70,100]
    for n in lengths:
        if time.time() > deadline:
            break

        # DE bounds
        bounds = [(0.0, 5.0) for _ in range(n)]

        # function to evaluate objective for DE
        def de_obj(x):
            if np.any(x < 0):
                return 1e6
            total = np.sum(x)
            if total < 1e-12:
                return 1e6
            a = x / total  # normalize sum to 1
            # Convert to Python list for evaluate_sequence
            return evaluate_sequence(a.tolist())

        # run DE with modest budget
        # we set maxiter based on remaining time estimate
        # compute remaining budget
        remaining = max(0.0, deadline - time.time())
        maxiter = 200
        # If not enough time, lower maxiter
        if remaining < 5:
            maxiter = 20

        result = scipy.optimize.differential_evolution(
            de_obj,
            bounds,
            maxiter=maxiter,
            popsize=5,
            polish=False,
            seed=seed,
            updating='deferred',
            disp=False,
        )
        seq = result.x
        if np.sum(seq) == 0:
            continue
        seq_norm = seq / np.sum(seq)
        score = evaluate_sequence(seq_norm.tolist())
        if score < best_score:
            best_score = score
            best_sequence = seq_norm.tolist()
            print(f"n={n}, DE best score={score:.6f}")

        # local refinement via simple stochastic hill climbing
        current = seq_norm.copy()
        cur_score = score
        # Determine step size proportionally
        step = 0.05
        for _ in range(2000):
            if time.time() > deadline:
                break
            # pick two indices
            i, j = np.random.choice(n, 2, replace=False)
            # amount to transfer
            delta = step * np.random.randn()
            # propose new values
            new = current.copy()
            new[i] = max(0.0, new[i] + delta)
            new[j] = max(0.0, new[j] - delta)
            # renormalize to sum=1
            if new.sum() == 0:
                continue
            new = new / new.sum()
            new_score = evaluate_sequence(new.tolist())
            if new_score < cur_score:
                current = new
                cur_score = new_score
                if cur_score < best_score:
                    best_score = cur_score
                    best_sequence = current.tolist()
                    print(f"n={n}, refined score={cur_score:.6f}")
            # optionally adjust step size adaptively (optional)
    # Return best sequence found, ensure non-negative and finite
    if best_sequence is None:
        # fallback: uniform sequence of length 10
        n=10
        best_sequence = [1.0/n]*n
    return best_sequence

def propose_candidate(seed=42, budget_s=1000):
    """Generate a sequence of non‑negative heights that minimizes the evaluation function.
    
    Strategy: For a series of moderate lengths (2 … 100) we run a cheap
    differential‑evolution optimisation that searches directly on the simplex
    (vectors are normalised to sum = 1 before calling ``evaluate_sequence``).
    The best sequence seen across all lengths is returned.  A short stochastic
    hill‑climbing stage further refines each candidate.
    """
    import numpy as np, time, scipy.optimize as opt
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5
    best_score = float('inf')
    best_seq = None

    # candidate lengths (short to medium, higher lengths usually give lower bound)
    lengths = [2,3,4,5,6,7,8,9,10,12,15,20,25,30,40,50,70,100]

    for n in lengths:
        # break if time exhausted
        if time.time() > deadline:
            break

        # define evaluation for differential evolution: works on raw vector x,
        # we normalise it to sum = 1 (scaling does not affect the objective).
        def de_obj(x, n=n):
            # clip negatives (should not happen with bounds)
            if np.any(x < 0):
                return 1e6
            total = np.sum(x)
            if total < 1e-12:
                return 1e6
            a = x / total
            # evaluate
            return evaluate_sequence(a.tolist())

        # simple bounds – numbers larger than 5 are unnecessary after normalisation
        bounds = [(0.0, 5.0) for _ in range(n)]

        # adaptively reduce maxiter if we are low on time
        remaining = max(0.0, deadline - time.time())
        maxiter = 200
        if remaining < 30:
            maxiter = max(10, int(remaining / (n * 0.001)))  # crude

        # run DE
        result = opt.differential_evolution(
            de_obj,
            bounds,
            maxiter=maxiter,
            popsize=5,
            polish=False,
            seed=seed,
            updating='deferred',
            disp=False,
        )
        if not result.success:
            continue
        vec = result.x
        if np.sum(vec) == 0:
            continue
        seq = vec / np.sum(vec)          # normalise to sum = 1
        score = evaluate_sequence(seq.tolist())
        if score < best_score:
            best_score = score
            best_seq = seq.tolist()
            print(f"Length {n}: DE found score {score:.6f}")

        # Simple stochastic refinement (hill climbing)
        cur = seq.copy()
        cur_score = score
        step = 0.05
        # limit number of refinements per length
        for _ in range(2000):
            if time.time() > deadline:
                break
            i, j = np.random.choice(n, 2, replace=False)
            delta = step * np.random.randn()
            new = cur.copy()
            new[i] = max(0.0, new[i] + delta)
            new[j] = max(0.0, new[j] - delta)
            if new.sum() == 0:
                continue
            new = new / new.sum()
            new_score = evaluate_sequence(new.tolist())
            if new_score < cur_score:
                cur = new
                cur_score = new_score
                if new_score < best_score:
                    best_score = new_score
                    best_seq = new.tolist()
                    print(f"Length {n}: refined score {new_score:.6f}")
            # optional: slowly shrink step size
            step *= 0.9999

    # fallback if nothing found
    if best_seq is None:
        n = 10
        best_seq = [1.0 / n] * n
    return best_seq

"""
Propose a step‑function (list of non‑negative heights) that minimizes the
evaluation function defined in the problem statement.

Strategy
--------
* Search over a modest set of lengths (2 … 100).  For each length we run a
  cheap global optimiser – `scipy.optimize.differential_evolution` – that
  works directly on the simplex (the vector is normalised to sum = 1 before
  calling ``evaluate_sequence``; scaling does not affect the objective).

* After DE finishes a short stochastic hill‑climbing refinement is performed
  to exploit any remaining improvement.

* The best sequence seen over all lengths is returned.  The routine respects
  the supplied time budget (default 1000 s) and prints progress information.
"""

import time
import numpy as np
import scipy.optimize as opt


def propose_candidate(seed: int = 42, budget_s: int = 1000) -> list[float]:
    """
    Return a list of non‑negative heights (the step function) with a low
    `evaluate_sequence` score.

    Parameters
    ----------
    seed : int
        Random seed for reproducibility.
    budget_s : int
        Total time budget in seconds.  The function stops a little early
        to avoid being killed.

    Returns
    -------
    list[float]
        The best sequence found within the time limit.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5  # leave a small safety margin

    best_score = float("inf")
    best_seq: list[float] | None = None

    # Lengths to explore – short to medium sizes give the best chance of
    # finding a value below the target 1.503.
    lengths = [
        2, 3, 4, 5, 6, 7, 8, 9, 10,
        12, 15, 20, 25, 30, 40, 50, 70, 100,
    ]

    for n in lengths:
        if time.time() > deadline:
            break

        # ------------------------------------------------------------
        # Differential Evolution on the simplex (sum = 1)
        # ------------------------------------------------------------
        def de_obj(x: np.ndarray, n=n) -> float:
            """Objective for DE – normalise the vector and call evaluate_sequence."""
            # Bounds already guarantee non‑negativity, but keep a guard.
            if np.any(x < 0):
                return 1e6
            total = np.sum(x)
            if total < 1e-12:
                return 1e6
            a = x / total          # enforce ∑a = 1  (scaling does not change the score)
            return evaluate_sequence(a.tolist())

        # The values themselves are irrelevant after normalisation; a generous
        # upper bound (5.0) keeps the search space small.
        bounds = [(0.0, 5.0) for _ in range(n)]

        # Adapt the max number of generations to the remaining time.
        remaining = max(0.0, deadline - time.time())
        maxiter = 200
        if remaining < 30:
            maxiter = max(10, int(remaining / (n * 0.001)))  # very rough safeguard

        result = opt.differential_evolution(
            de_obj,
            bounds,
            maxiter=maxiter,
            popsize=5,
            polish=False,
            seed=seed,
            updating="deferred",
            disp=False,
        )

        if not result.success:
            continue

        cand = result.x
        if np.sum(cand) == 0:
            continue
        cand_norm = cand / np.sum(cand)
        cand_score = evaluate_sequence(cand_norm.tolist())

        if cand_score < best_score:
            best_score = cand_score
            best_seq = cand_norm.tolist()
            print(f"[n={n}] DE best score = {cand_score:.6f}")

        # ------------------------------------------------------------
        # Simple stochastic refinement (hill‑climbing)
        # ------------------------------------------------------------
        cur = cand_norm.copy()
        cur_score = cand_score
        step = 0.05  # magnitude of mass transferred between two positions

        for _ in range(2000):
            if time.time() > deadline:
                break

            i, j = np.random.choice(n, 2, replace=False)
            delta = step * np.random.randn()   # may be positive or negative

            new = cur.copy()
            new[i] = max(0.0, new[i] + delta)
            new[j] = max(0.0, new[j] - delta)

            if new.sum() == 0:
                continue
            new = new / new.sum()
            new_score = evaluate_sequence(new.tolist())

            # accept only improvements (simple greedy descent)
            if new_score < cur_score:
                cur = new
                cur_score = new_score
                if new_score < best_score:
                    best_score = new_score
                    best_seq = new.tolist()
                    print(f"[n={n}] refined score = {new_score:.6f}")

            # slowly shrink the proposal step – helps the tail end of the search
            step *= 0.9999

    # --------------------------------------------------------------------
    # Fallback – should never be needed but guarantees a valid return.
    # --------------------------------------------------------------------
    if best_seq is None:
        n = 10
        best_seq = [1.0 / n] * n
        best_score = evaluate_sequence(best_seq)

    print(f"Final best score = {best_score:.6f} with length {len(best_seq)}")
    return best_seq
