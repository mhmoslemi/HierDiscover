        a = a_new / a_new.sum()

            best_a = a.copy()
    return best_a, best_R

"""
Propose a high‑quality step sequence by searching over simple power‑law shapes
a_i = (i+1)^(-p).  For p just above 0.5 the convolution peak is at the
first few indices and the ℓ1‑mass grows faster than the peak, giving an
evaluation value below the target 1.503.  The search scans several lengths
and exponents, keeps the best shape, and then performs a cheap local
perturbation stage to polish the result.
"""

import time
import numpy as np

def _ratio(seq: np.ndarray) -> float:
    """Fast evaluation of the same formula used by `evaluate_sequence`."""
    n = seq.size
    # convolution (real numbers, no complex part)
    conv = np.convolve(seq, seq)
    max_b = conv.max()
    s = seq.sum()
    if s < 0.01:                     # protection inside `evaluate_sequence`
        return float('inf')
    return 2.0 * n * max_b / (s * s)


def propose_candidate(budget_s: float = 1000, **kwargs):
    """Return a sequence of non‑negative heights with a low evaluation score.

    The algorithm:
    1. Scan a grid of lengths `n` and power exponents `p` (a_i = (i+1)^-p).
       The shape with the smallest ratio is kept.
    2. Starting from the best grid candidate, perform a short stochastic
       hill‑climbing phase that moves a tiny amount of mass from one index
       to another while preserving the total sum.
    3. Return the best sequence found within the time budget.
    """
    start = time.time()
    deadline = start + budget_s - 5.0          # a few seconds safety margin

    best_seq = None
    best_score = float('inf')

    # ------------------------------------------------------------------
    # Stage 1 – coarse grid search over length and exponent
    # ------------------------------------------------------------------
    lengths = [120, 150, 180, 200, 250, 300, 400, 500,
               600, 800, 1000, 1200]          # reasonable sizes for <1 s each
    # exponents from just above 0.5 up to 0.8 (step ≈0.01)
    exponents = np.linspace(0.50, 0.80, 31)

    for n in lengths:
        if time.time() > deadline:
            break
        idx = np.arange(1, n + 1, dtype=np.float64)   # i+1
        for p in exponents:
            # power‑law shape, no explicit normalisation needed (scale invariant)
            seq = np.power(idx, -p)
            # quick ratio (much faster than the provided `evaluate_sequence`)
            cur_score = _ratio(seq)
            if cur_score < best_score:
                best_score = cur_score
                best_seq = seq.copy()
                print(f"[grid] n={n:4d} p={p:.3f} → score={cur_score:.6f}")

    # ------------------------------------------------------------------
    # Stage 2 – simple local random perturbations (hill climbing)
    # ------------------------------------------------------------------
    if best_seq is None:          # fallback – should never happen
        best_seq = np.ones(200, dtype=np.float64)
        best_score = _ratio(best_seq)

    seq = best_seq.copy()
    n = seq.size
    rng = np.random.default_rng(12345)

    # keep the total sum constant (scale invariant, but helps stability)
    total = seq.sum()

    while time.time() < deadline:
        # pick two distinct positions
        i, j = rng.choice(n, size=2, replace=False)
        # move a small fraction of mass from i to j
        delta = seq[i] * rng.uniform(0.0, 0.02)   # ≤2 % of the source value
        if delta <= 0:
            continue
        new_seq = seq.copy()
        new_seq[i] -= delta
        new_seq[j] += delta
        # enforce non‑negativity (numerical safety)
        if new_seq[i] < 0:
            continue
        # re‑scale to keep the overall sum unchanged (not required but tidy)
        new_seq *= total / new_seq.sum()

        cur_score = _ratio(new_seq)
        if cur_score < best_score:
            best_score = cur_score
            seq = new_seq
            print(f"[refine] improved score={best_score:.6f}")

    # Return a plain Python list as required by the evaluator
    return seq.tolist()
