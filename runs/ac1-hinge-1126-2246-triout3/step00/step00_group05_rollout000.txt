def solve_lp(f_sequence, rhs):

    c = -np.ones(n) # maximize sum(b) => minimize -sum(b)
    # Build constraint matrix for conv <= rhs: each row index k
    A = np.zeros((2*n-1, n))
    for k in range(2*n-1):
        # A[k,i] = f_sequence[i] where j = k - i in [0,n-1] => i from max(0, k-(n-1)) to min(n-1, k)

"""
Propose a low‑value step‑function sequence.

The algorithm is a compact version of the classic “LP‑based balancing”
procedure:

1. Pick a moderate length `n` (the search explores several lengths).
2. Initialise a random non‑negative vector `f` and normalise it so that
   Σ f = √(2 n).  The evaluation metric is invariant under scaling,
   therefore this normalisation does not affect the score.
3. Repeatedly solve the linear program

        maximise   Σ g_j
        subject to  (f * g)[k] ≤ M  for all k
                    g_j ≥ 0

   where `M = max (f * f)`.  The optimal `g` gives the direction that
   increases the total mass while keeping the convolution bound.
4. Form a convex combination `f ← (1‑t)·f + t·g` (after re‑normalising `g`).
   The mixing factor `t` is adapted: it grows when we see improvement
   and shrinks otherwise.
5. Keep the best vector seen across all lengths and restarts.

If the LP fails (or after a long period without improvement) the algorithm
applies a small random perturbation to escape possible local minima.
The whole search stops before the allotted budget (default 1000 s) and
returns the best sequence found.

The code uses only NumPy and SciPy (`linprog` with the “highs” solver)
and does not rely on any file or network access.
"""

import time
import math
import numpy as np
from scipy.optimize import linprog

# ------------------------------------------------------------
# Helper: solve the LP   maximise Σ g  s.t. conv(f,g) ≤ M, g ≥ 0
# ------------------------------------------------------------
def solve_convolution_lp(f_seq, max_conv, time_limit=5.0):
    """
    Solve the linear program:
        maximise   sum(g)
        subject to conv(f,g) ≤ max_conv
                   g ≥ 0

    Parameters
    ----------
    f_seq : list or 1‑d np.ndarray
        Current sequence (non‑negative).
    max_conv : float
        Upper bound for every entry of the convolution f*g.
    time_limit : float, optional
        Maximum time (seconds) allowed for the LP solver.

    Returns
    -------
    np.ndarray or None
        Optimal vector `g` (as a dense array) or None if the
        solver did not succeed.
    """
    f = np.asarray(f_seq, dtype=float)
    n = f.shape[0]

    # Objective: maximise sum(g)  → minimise -sum(g)
    c = -np.ones(n, dtype=float)

    # Build the convolution constraints matrix A (size (2n‑1) × n)
    # Row k corresponds to (f * g)[k] = Σ_i f[i] * g[k-i]
    rows = 2 * n - 1
    A = np.zeros((rows, n), dtype=float)
    for k in range(rows):
        i_min = max(0, k - (n - 1))
        i_max = min(n - 1, k)
        # For each i, the variable g[j] where j = k-i gets coefficient f[i]
        # Vectorised assignment:
        i_vals = np.arange(i_min, i_max + 1, dtype=int)
        j_vals = k - i_vals
        A[k, j_vals] = f[i_vals]

    # Upper bounds for each convolution entry
    b = np.full(rows, max_conv, dtype=float)

    # Bounds for variables g_j ≥ 0
    bounds = [(0, None)] * n

    # Solve with HiGHS (fast interior‑point/simplex hybrid)
    opts = {"max_iter": 10_000, "time_limit": time_limit, "presolve": True}
    res = linprog(c, A_ub=A, b_ub=b, bounds=bounds,
                  method='highs', options=opts)

    if res.success:
        return np.asarray(res.x, dtype=float)
    return None

    # Parameters controlling the outer loop (different lengths & restarts)
    # ------------------------------------------------------------------
    length_pool = [128, 256, 512, 1024]          # candidate sizes (small → fast)
    best_seq = None
    best_score = float('inf')

    # ------------------------------------------------------------------
    # Main search over the chosen lengths
    # ------------------------------------------------------------------
    for n in length_pool:
        if time.time() >= deadline:
            break

        # ----- initialise a random sequence --------------------------------
        f = np.random.rand(n)                     # values in (0,1)
        target_sum = math.sqrt(2.0 * n)
        f *= target_sum / f.sum()                  # normalise Σf = √(2n)

        # ----- evaluate initial point --------------------------------------
        try:
            cur_score = evaluate_sequence(f.tolist())
        except Exception:
            cur_score = float('inf')
        if cur_score < best_score:
            best_score, best_seq = cur_score, f.tolist()
            print(f"[Init] n={n} score={best_score:.6f}")

        # ----- inner optimisation loop ------------------------------------
        t = 0.20               # initial mixing factor
        no_improve_steps = 0   # counter for stagnation detection
        max_no_improve = 80    # after this many stagnant steps we restart

        while time.time() < deadline:
            # Compute current convolution bound M = max(f*f)
            M = np.max(np.convolve(f, f))

            # Solve the LP to obtain a direction g
            g = solve_convolution_lp(f, M, time_limit=2.0)

            if g is None:
                # LP failed – apply a tiny random perturbation
                idx = np.random.randint(n)
                f[idx] = max(0.0, f[idx] + np.random.randn() * 0.05)
                # Re‑normalise to preserve Σf = √(2n)
                f *= target_sum / f.sum()
            else:
                # Normalise g to the same Σ = √(2n)
                g *= target_sum / g.sum()

                # Convex combination (mixing)
                f = (1.0 - t) * f + t * g

                # Occasionally shrink t if progress stalls
                # (updated later after score check)

            # ----- evaluate the new candidate --------------------------------
            try:
                cur_score = evaluate_sequence(f.tolist())
            except Exception:
                # Should not happen because we keep entries finite,
                # but guard against unexpected numerical issues.
                cur_score = float('inf')

            # ----- bookkeeping ------------------------------------------------
            if cur_score < best_score:
                best_score, best_seq = cur_score, f.tolist()
                print(f"[Improve] n={n} score={best_score:.6f} t={t:.3f}")
                # On improvement, we can be a little bolder with t
                t = min(0.5, t * 1.02)
                no_improve_steps = 0
            else:
                no_improve_steps += 1
                # If we have not improved for a while, reduce the step size
                if no_improve_steps >= max_no_improve:
                    # Random restart for this length
                    print(f"[Restart] n={n} after {no_improve_steps} stalls")
                    f = np.random.rand(n)
                    f *= target_sum / f.sum()
                    t = 0.20
                    no_improve_steps = 0
                else:
                    # Gradually decay t to fine‑tune
                    t = max(0.01, t * 0.985)

            # Safety break to avoid endless tight loops in pathological cases
            if time.time() + 0.05 >= deadline:
                break

    # ------------------------------------------------------------------
    # If for some reason we never found a feasible sequence, fall back
    # to a simple constant sequence (which yields a score of exactly 2).
    # ------------------------------------------------------------------
    if best_seq is None:
        n = 256
        const = [1.0] * n
        best_seq = const

    print(f"[Result] final best score = {best_score:.6f} (length={len(best_seq)})")
    return best_seq
