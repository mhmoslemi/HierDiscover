def propose_candidate(seed=123, budget_s=1000):
    import time, numpy as np
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5
    best_seq = None
    best_score = float('inf')
    # Choose n maybe 2000 to get better approximation
    n = 2000
    target_sum = np.sqrt(2 * n)
    # hyperparameters
    for _ in range(200):   # random restarts with different parameters
        # random alpha in (0.45,0.55)
        alpha = np.random.uniform(0.45, 0.55)
        delta = np.random.exponential(scale=1.0)  # positive
        # generate base sequence
        idx = np.arange(n, dtype=float)
        seq = (idx + delta) ** (-alpha)
        # normalize to sum target_sum
        seq = seq * (target_sum / seq.sum())
        # projected gradient descent
        lr = 0.2
        t = 0
        while time.time() < deadline and t < 2000:
            conv = np.convolve(seq, seq)
            max_val = conv.max()
            # maybe compute subgradient from indices near max
            eps = max_val * 1e-8
            ks = np.where(conv >= max_val - eps)[0]
            # gradient
            grad = np.zeros_like(seq)
            for k in ks:
                i_start = max(0, k - (n - 1))
                i_end = min(k, n - 1)
                # vector of indexes
                i_range = np.arange(i_start, i_end + 1)
                j = k - i_range
                grad[i_range] += 2.0 * seq[j]
            # Normalize gradient magnitude
            grad_norm = np.linalg.norm(grad)
            if grad_norm == 0:
                break
            step = lr / (1 + t * 1e-3)  # decreasing step
            seq = seq - step * grad / grad_norm
            seq = np.maximum(seq, 0.0)
            # renorm
            seq_sum = seq.sum()
            if seq_sum == 0:
                break
            seq = seq * (target_sum / seq_sum)
            # evaluate
            score = evaluate_sequence(seq.tolist())
            if score < best_score:
                best_score = score
                best_seq = seq.tolist()
                print(f'New best {best_score:.6f} alpha {alpha:.4f} delta {delta:.4f} iter {t}')
            t += 1
        # end while
    return best_seq if best_seq is not None else [target_sum]

"""
search_for_sequence.py

Search for a non‑negative step‑function (a list of heights) that minimizes the
evaluation function

    evaluate_sequence(seq) = 2 * n * max(conv(seq,seq)) / (sum(seq)**2)

where `conv` is the usual discrete convolution.  The algorithm starts from a
smooth “inverse‑square‑root’’ profile – a_i ≈ 1/√(i+δ) – which is known to give a
self‑convolution peak close to π/2 ≈ 1.57 (much better than the uniform
sequence that gives 2.0).  Randomly sampled exponents α and offsets δ are tried,
and the best candidate is refined by a simple projected sub‑gradient descent
that directly reduces the maximal convolution entry.  The whole search runs
until the supplied time budget expires and returns the best list found.
"""

import time
import numpy as np

# The evaluation routine is supplied by the judge; we just call it.
# from the external environment: evaluate_sequence(seq: list[float]) -> float


def _build_profile(n: int, alpha: float, delta: float) -> np.ndarray:
    """
    Produce a decreasing profile a_i = (i + delta) ** (-alpha) and scale it so
    that the L1‑norm equals sqrt(2*n).  The result is a numpy array of length n.
    """
    idx = np.arange(n, dtype=float)
    profile = (idx + delta) ** (-alpha)
    # Normalise to the required sum sqrt(2*n)
    target_sum = np.sqrt(2.0 * n)
    profile *= target_sum / profile.sum()
    return profile


def _subgradient_step(seq: np.ndarray, step: float) -> np.ndarray:
    """
    Perform one projected sub‑gradient step aimed at reducing
    max_k (conv(seq,seq))_k.
    """
    n = seq.shape[0]
    conv = np.convolve(seq, seq)
    max_val = conv.max()
    # indices where the convolution attains (or is extremely close to) the maximum
    eps = max_val * 1e-12
    ks = np.where(conv >= max_val - eps)[0]

    # compute sub‑gradient:  g_i = 2 * sum_{k∈K} seq_{k-i} (where defined)
    grad = np.zeros_like(seq)
    for k in ks:
        i_start = max(0, k - (n - 1))
        i_end = min(k, n - 1)
        i_range = np.arange(i_start, i_end + 1, dtype=int)
        j = k - i_range
        grad[i_range] += 2.0 * seq[j]

    # Normalise the step direction to avoid overly large jumps
    norm = np.linalg.norm(grad)
    if norm == 0.0:
        return seq
    direction = grad / norm
    # take a step and project back onto the feasible set (non‑negative,
    # fixed L1‑norm)
    new_seq = seq - step * direction
    new_seq = np.maximum(new_seq, 0.0)
    s = new_seq.sum()
    if s > 0.0:
        new_seq *= (np.sqrt(2.0 * n) / s)
    else:
        # very unlikely, but fall back to the previous sequence
        new_seq = seq.copy()
    return new_seq


def propose_candidate(seed: int = 42, budget_s: int = 1000, **_) -> list[float]:
    """
    Main entry point used by the grader.
    Returns a list of non‑negative floats that (hopefully) yields a low score.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5  # leave a tiny safety margin

    # Length of the construction – a few thousand works well in practice
    n = 2000
    target_sum = np.sqrt(2.0 * n)

    best_seq = None
    best_score = float("inf")

    # ------------------------------------------------------------------
    # 1) Random exploration of the (alpha, delta) family
    # ------------------------------------------------------------------
    while time.time() < deadline - 5.0:  # keep some time for refinement
        # Sample parameters
        alpha = np.random.uniform(0.45, 0.55)          # exponent around 0.5
        delta = np.random.exponential(scale=1.0)     # positive shift

        seq = _build_profile(n, alpha, delta)

        # Quick evaluation (the evaluation function already normalises)
        try:
            score = evaluate_sequence(seq.tolist())
        except Exception:
            continue

        if score < best_score:
            best_score = score
            best_seq = seq.copy()
            print(
                f"[{time.time():.1f}] New best from family: "
                f"score={best_score:.6f}, alpha={alpha:.4f}, delta={delta:.4f}"
            )
        # Very cheap – try many candidates
        if best_score <= 1.55:       # early exit if we already beat the target ball‑park
            break

    # ------------------------------------------------------------------
    # 2) Local refinement via projected sub‑gradient descent
    # ------------------------------------------------------------------
    if best_seq is not None:
        # Simple schedule for the step size
        step = 0.5
        it = 0
        while time.time() < deadline and it < 5000:
            # Reduce step size slowly
            cur_step = step / (1 + 0.0005 * it)
            new_seq = _subgradient_step(best_seq, cur_step)

            # Evaluate the new candidate
            try:
                new_score = evaluate_sequence(new_seq.tolist())
            except Exception:
                break  # something went terribly wrong – abort refinement

            if new_score < best_score:
                best_score = new_score
                best_seq = new_seq.copy()
                print(
                    f"[{time.time():.1f}] Improvement via sub‑grad: "
                    f"score={best_score:.6f}, iter={it}"
                )
                # keep the step size a little larger after a success
                step = max(step, cur_step * 1.2)

            it += 1

    # ------------------------------------------------------------------
    # 3) Very cheap stochastic hill‑climbing as a last boost
    # ------------------------------------------------------------------
    if best_seq is not None:
        while time.time() < deadline:
            # perturb a random subset of positions
            pert = best_seq.copy()
            idx = np.random.randint(0, n, size=max(1, n // 100))
            # Gaussian perturbation proportional to the current values
            noise = 0.05 * pert[idx] * np.random.randn(idx.size)
            pert[idx] = np.maximum(pert[idx] + noise, 0.0)
            # re‑normalise
            total = pert.sum()
            if total > 0.0:
                pert *= (target_sum / total)
            else:
                continue

            try:
                score = evaluate_sequence(pert.tolist())
            except Exception:
                continue

            if score < best_score:
                best_score = score
                best_seq = pert.copy()
                print(
                    f"[{time.time():.1f}] Random hill‑climb improvement: "
                    f"score={best_score:.6f}"
                )
                # if we got really close to the desired bound we can stop early
                if best_score < 1.51:
                    break

    # If nothing succeeded (should never happen) fall back to a uniform vector
    if best_seq is None:
        best_seq = np.full(n, target_sum / n, dtype=float)

    return best_seq.tolist()
