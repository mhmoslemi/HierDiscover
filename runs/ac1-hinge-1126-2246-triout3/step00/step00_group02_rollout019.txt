def max_convolution(seq):
    n = len(seq)
    size = 1 << ((2*n-1).bit_length())
    fft_a = np.fft.rfft(seq, n=size)
    conv = np.fft.irfft(fft_a * fft_a, n=size)[:2*n-1]
    return np.max(conv)
```

"""
Propose a candidate sequence that minimizes `evaluate_sequence`.

The algorithm works directly on the objective `M = max(conv(a, a))`
under the natural normalisation `sum(a) = sqrt(2*n)`.
For a fixed length `n` the evaluation returned by `evaluate_sequence`
is exactly `M`, because `2*n*max_b/(sum(a)**2) = max_b`.

We start from a randomised uniform vector and apply a simple
projected gradient descent on the peak of the auto‑convolution.
If `k*` is the index where the current peak occurs,
the (sub‑)gradient of `M` at `a` is
`g_i = 2*a_{k* - i}` (zero when the index is out of range).
A descent direction is `d = -g + λ·1` where `λ` makes the step
sum‑preserving.  After each step we clip negative entries,
renormalise the sum, and keep the best vector seen.
Occasional random jitter prevents stagnation.
The whole optimisation runs until the time budget expires and
returns the best sequence discovered.

The implementation uses an FFT based convolution (`np.fft.rfft`)
to obtain `max_b` in `O(n log n)` time, which allows thousands of
iterations within the 1000 s limit.
"""

import time
import numpy as np

# -------------------------------------------------------------
# Helper: fast linear convolution (real FFT) and max value
def max_convolution(a: np.ndarray) -> tuple[float, int]:
    """
    Return (max_value, argmax_index) of the linear convolution a*a.
    Uses a power‑of‑two FFT size for speed.
    """
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())          # next power of two
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    k = int(np.argmax(conv))
    return float(conv[k]), k


# -------------------------------------------------------------
def optimise_one_length(n: int, deadline: float) -> tuple[np.ndarray, float]:
    """
    Optimise a sequence of length `n` until the given deadline.
    Returns the best sequence (as a numpy array) and its max convolution value.
    """
    target_sum = np.sqrt(2.0 * n)          # required L1 norm for the true score
    avg = target_sum / n

    # start from a noisy uniform vector
    a = np.full(n, avg, dtype=np.float64)
    a += np.random.randn(n) * avg * 0.05   # small random perturbation
    a = np.clip(a, 0.0, None)
    a *= target_sum / a.sum()              # re‑normalise exactly

    best_a = a.copy()
    best_M, _ = max_convolution(best_a)

    # optimisation parameters
    iter_no_improve = 0
    max_no_improve = 3000                # restart after this many non‑improving steps
    iteration = 0
    start = time.time()

    while time.time() < deadline:
        # current convolution and its peak
        M, k_peak = max_convolution(a)

        # sub‑gradient of the peak (only indices that influence the peak)
        idx = k_peak - np.arange(n)                # candidate partner indices
        valid = (idx >= 0) & (idx < n)
        g = np.zeros_like(a)
        g[valid] = 2.0 * a[idx[valid]]

        # make the step sum‑preserving: d = -g + λ·1 with λ = mean(g)
        lam = g.mean()
        d = -g + lam

        # step size (decreases with iterations)
        step = avg * 0.2 / np.sqrt(iteration + 1.0)

        a_new = a + step * d
        a_new = np.clip(a_new, 0.0, None)
        # renormalise to keep the required L1 norm
        total = a_new.sum()
        if total > 0:
            a_new *= target_sum / total
        else:
            # fallback – should never happen, but keep a feasible point
            a_new = np.full_like(a, avg)

        # evaluate the new point
        M_new, _ = max_convolution(a_new)

        # accept only improvements (simple hill‑climb)
        if M_new < M:
            a = a_new
            M = M_new
            iter_no_improve = 0
        else:
            iter_no_improve += 1
            # occasional small random jitter to escape tiny plateaus
            if iter_no_improve % 500 == 0:
                jitter = np.random.randn(n) * avg * 0.01
                a = np.clip(a + jitter, 0.0, None)
                a *= target_sum / a.sum()

        # keep the best ever seen
        if M < best_M:
            best_M = M
            best_a = a.copy()
            # early exit if we beat the target bound
            if best_M <= 1.503:
                break

        # restart if we have been stuck too long
        if iter_no_improve >= max_no_improve:
            a = np.abs(np.random.randn(n))
            a *= target_sum / a.sum()
            iter_no_improve = 0

        iteration += 1
        # safety break if we really run out of time
        if time.time() >= deadline:
            break

    return best_a, best_M


# -------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: float = 1000, **kwargs) -> list[float]:
    """
    Entry point required by the evaluation harness.
    Tries a few different lengths (chosen randomly) and keeps the best
    sequence found before the time budget runs out.
    """
    np.random.seed(seed)
    overall_deadline = time.time() + budget_s - 5.0   # reserve a few seconds

    best_seq = None
    best_score = float("inf")

    # try a handful of different lengths (larger lengths give more freedom)
    while time.time() < overall_deadline:
        # length chosen between 2000 and 8000 (adjustable)
        n = int(np.random.randint(2000, 8001))
        # leave a small slice of the remaining time for this run
        per_run_deadline = time.time() + min(30.0, overall_deadline - time.time())
        seq, score = optimise_one_length(n, per_run_deadline)

        # score returned by `optimise_one_length` is exactly max_b,
        # which equals the value of `evaluate_sequence` for a correctly normalised vector.
        if score < best_score:
            best_score = score
            best_seq = seq.copy()
            print(f"New global best: score={best_score:.6f} (n={n})")

    # final safety: if nothing found, fall back to a uniform sequence
    if best_seq is None:
        n = 4000
        uniform = [np.sqrt(2 * n) / n] * n
        return uniform

    # Convert to plain Python list (as required)
    return best_seq.tolist()
