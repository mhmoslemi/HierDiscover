import numpy as np
import time
from scipy.optimize import linprog

def convolve_max(seq):
    return np.max(np.convolve(seq, seq))

def normalize(seq):
    s = np.sum(seq)
    if s == 0:
        return seq
    factor = np.sqrt(2 * len(seq)) / s
    return seq * factor

def solve_lp(f_seq, M):
    n = len(f_seq)
    # Build constraints a_ub * b <= b_ub
    # each k: sum_j f_i * b_{k-i} <= M
    rows = []
    rhs = []
    for k in range(2*n - 1):
        row = np.zeros(n)
        # indices i s.t. 0 <= i < n and 0 <= k-i < n
        i_min = max(0, k - (n-1))
        i_max = min(n-1, k)
        for i in range(i_min, i_max+1):
            j = k - i
            row[j] = f_seq[i]
        rows.append(row)
        rhs.append(M)
    # nonnegativity: -I * b <= 0
    rows = np.vstack([rows, -np.eye(n)])
    rhs = np.concatenate([rhs, np.zeros(n)])
    c = -np.ones(n)  # maximize sum(b)
    res = linprog(c, A_ub=rows, b_ub=rhs, method='highs', options={'presolve': True, 'time_limit': 2.0})
    if res.success:
        return res.x
    else:
        return None

def propose_candidate(seed=123, budget_s=1000.0):
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5   # small margin

    # Choose length n based on time budget, moderate size
    n = 800  # may adjust larger if time permits
    # Initial shape: geometric with alpha ~ 1.2
    alpha = 1.2
    r = np.exp(-alpha / n)
    seq = np.array([r**i for i in range(n)], dtype=float)
    seq = normalize(seq)

    best_seq = seq.copy()
    best_score = evaluate_sequence(best_seq.tolist())

    # Main loop
    iteration = 0
    while time.time() < deadline:
        iteration += 1
        # Compute current max convolution
        conv = np.convolve(seq, seq)
        M = np.max(conv)

        # Solve LP to get a direction with larger sum (but sum is fixed, we only need g)
        g = solve_lp(seq, M)
        if g is not None:
            # Normalize g to same sum
            g = normalize(g)
            # Try a few mixing coefficients
            for alpha_mix in [0.1, 0.3, 0.5, 0.7, 0.9]:
                cand = (1 - alpha_mix) * seq + alpha_mix * g
                cand = normalize(cand)  # keep sum fixed (should already be)
                score = evaluate_sequence(cand.tolist())
                if score < best_score:
                    best_score = score
                    best_seq = cand.copy()
                    seq = cand  # accept as current
                    break
            else:
                # No improvement, fall back to small random perturbation
                idx = np.random.randint(n)
                perturb = np.random.randn() * 0.02 * seq[idx]
                seq[idx] = max(0.0, seq[idx] + perturb)
                seq = normalize(seq)
        else:
            # random perturbation
            idx = np.random.randint(n)
            seq[idx] = max(0.0, seq[idx] + np.random.randn() * 0.01)
            seq = normalize(seq)

        # Evaluate current seq
        curr_score = evaluate_sequence(seq.tolist())
        if curr_score < best_score:
            best_score = curr_score
            best_seq = seq.copy()

        # Occasionally log progress
        if iteration % 10 == 0:
            print(f"Iter {iteration}: best={best_score:.6f}, n={n}")

        # Occasionally adapt n (increase length) if improvement stalls
        if iteration % 200 == 0 and n < 2000:
            # try to enlarge by appending zeros and re-normalize
            new_n = n + 200
            seq = np.concatenate([seq, np.zeros(200)])
            seq = normalize(seq)
            n = new_n

    print(f"Finished after {iteration} iterations, best score={best_score:.6f}")
    return best_seq.tolist()

def solve_lp(f_seq, M):
    n = len(f_seq)
    # Build A_ub matrix (2n-1) conv constraints + n nonneg constraints
    rows = []
    rhs = []
    for k in range(2 * n - 1):
        row = np.zeros(n)
        i_min = max(0, k - (n - 1))
        i_max = min(n - 1, k)
        # indices for i
        i_vals = np.arange(i_min, i_max + 1)
        j_vals = k - i_vals
        row[j_vals] = f_seq[i_vals]
        rows.append(row)
        rhs.append(M)
    rows = np.vstack([rows, -np.eye(n)])
    rhs = np.concatenate([rhs, np.zeros(n)])
    c = -np.ones(n)  # maximize sum => minimize -sum
    res = linprog(c, A_ub=rows, b_ub=rhs, method='highs', options={'presolve': True, 'time_limit': 2.0})
    if res.success:
        return res.x
    return None

"""
Hybrid search for a low‐value step sequence.

The algorithm mixes three cheap heuristics inside a time budget:

1. **LP direction** – solve a linear program that maximises the ℓ1‑norm of a
   candidate `g` under the constraint `conv(f, g) ≤ M` (where `M` is the current
   maximal self‑convolution of `f`).  The new sequence is taken as a convex
   combination of `f` and `g`; a small line‑search over several mixing factors
   yields a monotone improvement of the evaluation score.

2. **Projected sub‑gradient step** – the current maximal convolution occurs at
   one or a few indices `k`.  Its sub‑gradient w.r.t. the coefficients is
   `∂/∂a_i conv_k = 2 a_{k-i}` (or `a_i` when `i==k-i`).  Moving a short step
   opposite this gradient and re‑projecting back onto the simplex
   (non‑negative and fixed ℓ1‑norm) reduces `max(conv)`.

3. **Random perturbations** – occasional tiny random changes keep the search
   from stagnating.

All sequences are normalised so that `sum(a) = sqrt(2*n)` (the scale used by the
evaluation function).  The best sequence seen before the timeout is returned.
"""

import time
import numpy as np
from scipy.optimize import linprog

# ----------------------------------------------------------------------
# Helper utilities
# ----------------------------------------------------------------------


def _normalise(seq: np.ndarray) -> np.ndarray:
    """Scale a non‑negative vector so that its L1‑norm equals sqrt(2·len)."""
    n = len(seq)
    total = np.sum(seq)
    if total == 0:
        return seq
    factor = np.sqrt(2.0 * n) / total
    return seq * factor


def _max_convolution(seq: np.ndarray) -> float:
    """Return max_k (seq * seq)_k."""
    return float(np.max(np.convolve(seq, seq)))


def _solve_lp(f_seq: np.ndarray, M: float):
    """
    Solve   maximise   sum(b)
    s.t.    conv(f_seq, b) ≤ M  (for every shift)
            b ≥ 0
    Returns the optimal b as a dense NumPy array, or None on failure.
    """
    n = len(f_seq)
    # Build the linear constraints conv(f, b) ≤ M
    A_rows = []
    b_ub = []

    for k in range(2 * n - 1):
        row = np.zeros(n)
        i_min = max(0, k - (n - 1))
        i_max = min(n - 1, k)
        i_vals = np.arange(i_min, i_max + 1, dtype=int)
        j_vals = k - i_vals
        # row[j] = f[i] for all valid i,j with i+j = k
        row[j_vals] = f_seq[i_vals]
        A_rows.append(row)
        b_ub.append(M)

    # Non‑negativity constraints  -b ≤ 0
    A_rows.append(-np.eye(n))
    b_ub.append(np.zeros(n))

    A_ub = np.vstack(A_rows)
    b_ub = np.concatenate(b_ub)

    # maximise sum(b)  → minimise -sum(b)
    c = -np.ones(n)

    # Use the HiGHS interior‑point method (fast for moderate sizes)
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, method="highs")
    if res.success:
        return res.x
    return None


def _gradient_step(seq: np.ndarray, step_frac: float = 0.15) -> np.ndarray:
    """
    One projected sub‑gradient descent step on max(conv(seq, seq)).
    The step size is a fraction (step_frac) of the normalised gradient.
    """
    n = len(seq)
    conv = np.convolve(seq, seq)
    M = np.max(conv)

    # indices whose convolution value is within 1e‑12 of the maximum
    active = np.where(conv >= M - 1e-12 * M)[0]

    grad = np.zeros_like(seq)
    for k in active:
        i_min = max(0, k - (n - 1))
        i_max = min(n - 1, k)
        i_vals = np.arange(i_min, i_max + 1, dtype=int)
        j_vals = k - i_vals
        # derivative of a_i * a_j w.r.t a_i is a_j (two symmetric terms give 2*a_j)
        grad[i_vals] += seq[j_vals] * 2.0
        # the diagonal term a_i*a_i has been counted twice; correct it
        diag = (k % 2 == 0) and (k // 2 >= i_min) and (k // 2 <= i_max)
        if diag:
            idx = k // 2
            grad[idx] -= seq[idx]  # remove the extra count, leaving 2*a_i

    if np.allclose(grad, 0):
        return seq

    direction = -grad
    norm = np.linalg.norm(direction)
    if norm == 0:
        return seq
    step = step_frac * direction / norm
    new_seq = seq + step
    new_seq = np.maximum(new_seq, 0.0)
    new_seq = _normalise(new_seq)
    return new_seq


def _random_perturb(seq: np.ndarray, scale: float = 0.02) -> np.ndarray:
    """Small random perturbation of a single coefficient."""
    n = len(seq)
    idx = np.random.randint(n)
    delta = np.random.randn() * scale * seq[idx]
    seq[idx] = max(0.0, seq[idx] + delta)
    return _normalise(seq)

def propose_candidate(seed: int = 123, budget_s: float = 1000.0, **kwargs) -> list:
    """
    Return a list of non‑negative heights that (hopefully) gives a low score
    for ``evaluate_sequence``.  The function runs for at most ``budget_s``
    seconds, using a hybrid of linear‑programming, projected sub‑gradient,
    and stochastic perturbations.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5   # tiny safety margin

    # ------------------------------------------------------------------
    # Initialise a starting sequence.
    # Prefer the previously discovered construction if it exists, otherwise
    # use a smooth exponential profile.
    # ------------------------------------------------------------------
    try:
        # height_sequence_1 is provided by the surrounding environment
        init = np.array(height_sequence_1, dtype=float)
        n = len(init)
        seq = _normalise(init)
    except Exception:
        # No previous construction – fall back to a geometric sequence.
        n = 800                     # moderate size (fast LP solves)
        alpha = 1.2                 # a decent starting decay
        r = np.exp(-alpha / n)
        seq = np.array([r ** i for i in range(n)], dtype=float)
        seq = _normalise(seq)

    best_seq = seq.copy()
    best_score = evaluate_sequence(best_seq.tolist())

    iteration = 0
    step_frac = 0.2              # initial gradient step size

    # ------------------------------------------------------------------
    # Main improvement loop.
    # ------------------------------------------------------------------
    while time.time() < deadline:
        iteration += 1

        # Occasionally shrink the gradient step to fine‑tune later on
        if iteration % 200 == 0:
            step_frac = max(0.01, step_frac * 0.8)

        # ------------------------------------------------------------------
        # 1) LP direction + line search (most powerful move)
        # ------------------------------------------------------------------
        if np.random.rand() < 0.55:
            M = _max_convolution(seq)
            g = _solve_lp(seq, M)
            if g is not None:
                g = _normalise(g)

                # try a set of mixing coefficients and keep the best one
                mix_factors = [0.0, 0.05, 0.10, 0.20, 0.30, 0.45, 0.60, 0.80, 1.0]
                improved = False
                for t in mix_factors:
                    cand = (1.0 - t) * seq + t * g
                    cand = _normalise(cand)       # keeps the ℓ1‑norm fixed
                    score = evaluate_sequence(cand.tolist())
                    if score < best_score:
                        best_score = score
                        best_seq = cand.copy()
                        seq = cand
                        improved = True
                        break
                if not improved:
                    # no mixing improved the score – fall back to a tiny random move
                    seq = _random_perturb(seq, scale=0.01)

            else:
                # LP failed – do a random tweak
                seq = _random_perturb(seq, scale=0.015)

        # ------------------------------------------------------------------
        # 2) Projected sub‑gradient step (cheap, many iterations)
        # ------------------------------------------------------------------
        elif np.random.rand() < 0.90:
            cand = _gradient_step(seq, step_frac=step_frac)
            score = evaluate_sequence(cand.tolist())
            if score < best_score:
                best_score = score
                best_seq = cand.copy()
                seq = cand
            else:
                # keep the current sequence, but still try a tiny perturbation
                seq = _random_perturb(seq, scale=0.005)

        # ------------------------------------------------------------------
        # 3) Pure random perturbation (exploration)
        # ------------------------------------------------------------------
        else:
            seq = _random_perturb(seq, scale=0.02)
            score = evaluate_sequence(seq.tolist())
            if score < best_score:
                best_score = score
                best_seq = seq.copy()

        # ------------------------------------------------------------------
        # Logging – printed every few hundred iterations.
        # ------------------------------------------------------------------
        if iteration % 250 == 0:
            elapsed = time.time() + budget_s - deadline
            print(f"Iter {iteration:5d} | best score = {best_score:.6f} | n = {n}")

    print(f"Search finished after {iteration} iterations. Best score = {best_score:.6f}")
    return best_seq.tolist()
