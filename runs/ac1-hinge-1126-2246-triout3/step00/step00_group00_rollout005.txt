"""
Linear ramp generator for the step‑function inequality problem.

The evaluation function is
    eval = 2 * n * max(conv(a, a)) / (sum(a) ** 2)

For a sequence a_i = C * i (i = 0 … n‑1) with C = 1000/(n‑1) we have
    max(conv) = C² * n (n‑1)(n‑2) / 6
    sum(a)      = C * n (n‑1) / 2

Hence    eval = 4/3  (≈ 1.3333) for any n ≥ 2, which is well below the target
1.503.  The algorithm simply builds this monotone increasing ramp.
If the global variable `height_sequence_1` exists we reuse its length,
otherwise a default length of 4000 is used.

The function prints a short log (length and attained score) and returns the
sequence as a plain Python list of floats.
"""

import time
import numpy as np

def propose_candidate(seed: int = 42, budget_s: float = 1000, **kwargs) -> list[float]:
    """
    Generate a linear increasing sequence in the interval [0, 1000].
    The sequence length is taken from the existing `height_sequence_1`
    if available, otherwise a default of 4000 is used.

    Parameters
    ----------
    seed : int
        Unused – kept for compatibility with the evaluation harness.
    budget_s : float
        Unused – the construction is instantaneous.
    **kwargs
        Ignored.

    Returns
    -------
    list[float]
        The candidate sequence.
    """
    # Determine the length to use.
    try:
        n = len(height_sequence_1)          # type: ignore[name-defined]
    except Exception:
        n = 4000                            # a reasonable size if no global is present

    if n < 2:
        # degenerate case – fall back to a tiny valid sequence
        return [1.0]

    # Scale factor so that the last element equals 1000.
    scale = 1000.0 / (n - 1)

    # Build the linear ramp a_i = scale * i, i = 0 .. n-1.
    seq = [float(scale * i) for i in range(n)]

    # Optional sanity check (not required for correctness).

    except Exception as e:
