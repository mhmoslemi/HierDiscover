def project_onto_simplex(v):
    # compute Euclidean projection onto probability simplex: sum = 1, v_i >=0
    # algorithm by Duchi et al.
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    rho = np.where(u - cssv / np.arange(1, len(v)+1) > 0)[0][-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0)
    return w

def project_onto_simplex(v):
    # Duchi algorithm.
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    rho = np.where(u - cssv / np.arange(1, n+1) > 0)[0]
    if rho.size == 0:
        theta = 0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0)
    return w

"""
Propose a candidate step‐function (non‑negative sequence) that tries to lower the
evaluation score  2·n·max(conv(f,f)) / (sum(f)^2).

The algorithm is a projected (sub‑)gradient descent on a smooth surrogate of the
L∞‑norm of the convolution.  For a chosen exponent p>1 we minimise

    F_p(f) = Σ_k ( (f*f)[k] / max(f*f) )^p

which approaches the true maximum when p grows.  The gradient of F_p is
computed efficiently by FFTs, the iterate is projected onto the probability
simplex (so the sum stays equal to 1), and the exponent p is increased
gradually.  Random perturbations and a simple learning‑rate schedule avoid
stagnation.  The best sequence found during the allotted time is returned.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Helper: Euclidean projection of a vector onto the probability simplex
# ----------------------------------------------------------------------
def project_onto_simplex(v: np.ndarray) -> np.ndarray:
    """Return the Euclidean projection of v onto {x>=0, sum(x)=1}."""
    n = v.shape[0]
    # sort v in descending order
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0                # cumulative sum minus 1
    rho = np.where(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:                         # corner case: all entries are zero
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Main entry point required by the evaluation harness
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: float = 1000, **kwargs):
    """
    Search for a non‑negative sequence that minimises `evaluate_sequence`.

    Parameters
    ----------
    seed : int
        Random seed for reproducibility.
    budget_s : float
        Time budget in seconds. The function stops a few seconds before the
        limit so that the runner can safely retrieve the result.

    Returns
    -------
    list[float]
        The best sequence found (as a plain Python list).
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 5.0   # safety margin

    # ------------------------------------------------------------------
    # Problem size – use the length of the previously known construction
    # if it exists, otherwise fall back to a moderate size.
    # ------------------------------------------------------------------
    if 'height_sequence_1' in globals():
        n = len(height_sequence_1)          # ≈ 5 000 in the reference run
    else:
        n = 2000

    # ------------------------------------------------------------------
    # Initialise a random point on the simplex (sum == 1, all entries ≥0)
    # ------------------------------------------------------------------
    f = np.random.rand(n)
    f = f / f.sum()

    best_f = f.copy()
    best_score = evaluate_sequence(best_f.tolist())
    print(f"[init] n={n}, initial score = {best_score:.6f}")

    # ------------------------------------------------------------------
    # Pre‑compute FFT length (next power of two ≥ 2·n‑1) for speed.
    # ------------------------------------------------------------------
    fft_len = 1 << ((2 * n - 1).bit_length())

    # Hyper‑parameters
    p = 4                     # start exponent (smooth surrogate)
    p_max = 12                # final exponent
    p_step_iter = 2000        # how many iterations before raising p
    eta0 = 0.5                # initial learning rate
    iteration = 0

    while time.time() < deadline:
        # ----- convolution via FFT (linear convolution) -----------------
        F = np.fft.rfft(f, fft_len)
        conv = np.fft.irfft(F * F, fft_len)[: 2 * n - 1]

        M = conv.max()
        if M <= 0:
            # Should never happen because sum(f)>0, but guard against it.
            break

        # ----- smooth surrogate weights --------------------------------
        # Scale so that the current maximum becomes 1.
        scaled = conv / M                # max(scaled) == 1
        # Emphasise entries close to the maximum.
        w = scaled ** (p - 1)            # w_k in [0,1]

        # ----- gradient of Σ (scaled)^p  (2·p factor) ----------------
        # We need g_i = 2·p * Σ_j w_{i+j} * f_j

        # Extract the required n entries: indices (2n‑2‑i) for i=0…n‑1
        idx = (2 * n - 2) - np.arange(n)
        grad = 2.0 * p * G[idx]          # shape (n,)

        # ----- simple diminishing learning rate -----------------------
        eta = eta0 / math.sqrt(iteration + 1.0)
        f_candidate = f - eta * grad

        # ----- projection back onto the simplex -----------------------
        f = project_onto_simplex(f_candidate)

        # ----- evaluate current point ---------------------------------
        try:

        except Exception as exc:

        if cur_score < best_score:

    return best_f.tolist()
