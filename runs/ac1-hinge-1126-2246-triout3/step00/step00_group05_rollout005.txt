def propose_candidate(seed=0, budget_s=1000):
    """
    Search for a non‑negative step‑function (list of heights) that minimizes
    ``evaluate_sequence``.
    Strategy:
    1.  Determine a working size `n` (the length of the provided
        ``height_sequence_1`` if present, otherwise a default of 600).
    2.  Initialise a set of candidate families that are known to give low
        autocorrelation: power‑law, exponential, and linear decays.
        For each family a small grid of parameters is evaluated and the best
        candidate is kept.
    3.  Starting from the best candidate run a simple simulated‑annealing /
        hill‑climbing loop.  In each iteration a small amount of mass is
        transferred from one index to another; the move is accepted if it
        improves the current objective, otherwise it may be accepted with a
        temperature‑dependent probability.
    4.  The loop stops when the time budget is exhausted; the best sequence
        found is returned.
    The algorithm respects the non‑negativity and scale‑invariance of the
    objective, normalises every candidate to unit L1‑norm and lets the
    evaluator perform the final clipping.
    """
    import time, numpy as np
    
    # ------------------------------------------------------------
    # Helpers
    # ------------------------------------------------------------
    def normalize(v):
        """Scale vector to have L1 norm = 1 (avoid all‑zero case)."""
        s = np.sum(v)
        if s == 0:
            # fallback uniform
            return np.full_like(v, 1.0 / v.size)
        return v / s
    
    # ------------------------------------------------------------
    # Initialisation
    # ------------------------------------------------------------
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.2   # a tiny safety margin

    try:
        n = len(height_sequence_1)          # type: ignore[name-defined]

        n = 600                               # reasonable default
    
    # Make sure we have at least a modest size (the problem works for any n≥1)
    n = max(100, n)
    print(f"Search will use n = {n}")
    
    # Start with the uniform vector (baseline)
    best_seq = np.full(n, 1.0 / n)
    best_score = evaluate_sequence(best_seq.tolist())
    print(f"Uniform baseline score: {best_score:.6f}")
    
    # Optionally evaluate the provided reference sequence
    if 'height_sequence_1' in globals():
        ref = np.array(height_sequence_1, dtype=float)

        ref_score = evaluate_sequence(ref.tolist())
        print(f"Reference sequence score: {ref_score:.6f}")
        if ref_score < best_score:

    # 1) Grid search over a few parametric families
    # ------------------------------------------------------------

    for alpha in alphas:
        seq = (np.arange(1, n + 1, dtype=float) ** (-alpha))
        seq = normalize(seq)
        score = evaluate_sequence(seq.tolist())
        if score < best_score:
            best_score, best_seq = score, seq.copy()
            print(f"Power‑law alpha={alpha:.3f} -> new best {score:.6f}")
    
    # exponential decay: a_i ∝ r**i
    rs = np.linspace(0.90, 0.999, 12)   # very gentle to steep decay
    for r in rs:
        seq = (r ** np.arange(n, dtype=float))
        seq = normalize(seq)
        score = evaluate_sequence(seq.tolist())
        if score < best_score:
            best_score, best_seq = score, seq.copy()
            print(f"Exponential r={r:.3f} -> new best {score:.6f}")
    
    # linear decay: values from 1 down to p (inclusive)
    ps = np.linspace(0.0, 0.9, 19)      # p = low end as fraction of the high end
    for p in ps:

        if new_score < curr_score:

    print(f"Finished after {iteration} iterations – best score {best_score:.6f}")
    # Return as plain Python list, as required by the evaluator
    return best_seq.tolist()

"""
Propose a step‑function (list of non‑negative heights) that minimizes the
evaluation function `evaluate_sequence`.

Algorithm
---------
1. **Determine the length** – use the length of the provided reference
   sequence `height_sequence_1` if it exists, otherwise fall back to a
   modest default (≈600).  The problem is scale‑invariant, therefore we
   normalise every candidate to unit L1‑norm.

2. **Parametric grid search** – evaluate a few simple families that are
   known to give low autocorrelation:
   * power‑law decay  a_i ∝ (i+1)^(-α)      (α in [0, 2.5])
   * exponential decay a_i ∝ r^i            (r in [0.90, 0.999])
   * linear decay    a_i decreasing from 1 to p (p∈[0, 0.9])

   The best vector from this sweep becomes the starting point.

3. **Stochastic refinement** – a lightweight simulated‑annealing /
   hill–climbing loop.
   * At each iteration a small amount of mass is transferred from a
     randomly chosen index *i* to another index *j* (preserving the total
     sum).
   * The new vector is evaluated.  If it improves the current score the
     move is always accepted; otherwise it may be accepted with a
     temperature‑dependent probability that linearly cools to zero.
   * The best‑ever vector is remembered.

The loop stops when the allotted time (`budget_s`) is exhausted.  The
final best sequence (as a plain Python list) is returned.

All helper functions are defined at the top level – there are no closures.
"""

import time
import numpy as np

def _normalize(v: np.ndarray) -> np.ndarray:
    """Return a copy of *v* scaled to have L1‑norm equal to 1.
    If *v* is the zero vector a uniform distribution is returned.
    """
    s = np.sum(v)
    if s == 0.0:
        return np.full_like(v, 1.0 / v.size, dtype=float)
    return v / s

    # ------------------------------------------------------------

    # ------------------------------------------------------------

    # Reference sequence, if it exists
    if 'height_sequence_1' in globals():

        ref = _normalize(ref)

        print(f"[reference] reference sequence score = {ref_score:.6f}")

            print("[update] adopted reference as current best")

    # ------------------------------------------------------------
