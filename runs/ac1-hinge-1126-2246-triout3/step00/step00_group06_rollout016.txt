def compute_full_conv(a):
    n = a.shape[0]
    size = 1 << ((2*n - 1).bit_length())
    fft_a = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fft_a * fft_a, n=size)[:2*n-1]
    # avoid tiny negative values due to numerical error
    conv[conv < 0] = 0.0
    return conv

Then computing contributions:

def contributions_to_peaks(a, peaks):
    n = a.shape[0]
    contrib = np.zeros(n)
    for k in peaks:
        start = max(0, k - (n-1))
        end = min(k, n-1)
        i = np.arange(start, end+1)
        contrib[i] += a[k - i]
    return contrib

def propose_candidate(seed=42, budget_s=1000):
    """Searches for a non‑negative sequence minimizing evaluate_sequence.
    Uses a simple projected gradient/hill‑climbing scheme that repeatedly moves
    mass from positions contributing most to the current maximal convolution value
    to positions contributing least, with occasional random perturbations.
    The algorithm runs until the time budget expires and returns the best sequence
    found as a list of Python floats.
    """
    import numpy as np, time, random
    np.random.seed(seed)
    # length of sequence – use the same length as the known construction if available
    try:
        n = len(height_sequence_1)
    except NameError:
        n = 1500  # fallback
    # initialise with uniform distribution (sum = 1)
    a = np.full(n, 1.0 / n, dtype=np.float64)
    # keep best
    best_a = a.copy()
    # compute initial score
    conv = compute_full_conv(a)
    best_score = 2.0 * n * np.max(conv)   # sum == 1
    best_score = float(best_score)
    # timing
    deadline = time.time() + budget_s - 0.5   # reserve a little margin
    iteration = 0
    no_improve = 0
    print(f"Start optimisation: n={n}, initial score={best_score:.6f}")

    while time.time() < deadline:
        # recompute convolution for current a
        conv = compute_full_conv(a)
        max_val = np.max(conv)
        # indices where conv is maximal (within tolerance)
        tol = 1e-12
        peak_idxs = np.where(np.abs(conv - max_val) <= tol)[0]
        # compute contributions of each position to the peaks
        contrib = contributions_to_peaks(a, peak_idxs)
        # choose source (large contribution) and destination (small contribution)
        src = np.argmax(contrib)
        dst = np.argmin(contrib)
        if a[src] <= 1e-12:
            # nothing to move, random perturbation
            src_candidates = np.where(a > 1e-12)[0]
            if src_candidates.size == 0:
                break
            src = int(np.random.choice(src_candidates))
        # step size – a decreasing fraction of the source mass
        frac = 0.2 * (1.0 - iteration / (budget_s*100))  # heuristic decay
        if frac < 0.001:
            frac = 0.001
        eps = frac * a[src]
        # occasional larger jumps
        if np.random.rand() < 0.01:
            eps = min(a[src], eps * 10.0)
        # perform move
        a_new = a.copy()
        a_new[src] -= eps
        a_new[dst] += eps
        # ensure non‑negative
        if a_new[src] < 0:
            # adjust epsilon to avoid negativity
            eps = a[src]
            a_new[src] = 0.0
            a_new[dst] += eps

        # compute new score (fast: only need max of convolution)
        conv_new = compute_full_conv(a_new)
        score_new = 2.0 * n * np.max(conv_new)

        if score_new + 1e-12 < best_score:
            best_score = float(score_new)
            best_a = a_new.copy()
            a = a_new   # keep moving from the improved state
            no_improve = 0
            if iteration % 100 == 0:
                print(f"Iter {iteration:6d} – improved score: {best_score:.6f}")
        else:
            # accept with a small probability to escape plateaus
            temperature = max(0.001, (deadline - time.time()) / (budget_s))
            if np.random.rand() < np.exp(-(score_new - best_score) / (temperature * best_score + 1e-12)):
                a = a_new
            no_improve += 1

        # occasional random shake‑up to avoid getting stuck
        if no_improve >= 5000:
            # random perturbation of a few entries
            for _ in range(5):
                i, j = np.random.randint(0, n, size=2)
                if i == j:
                    continue
                delta = 0.05 * a[i] * (np.random.rand() - 0.5)
                a[i] = max(0.0, a[i] - delta)
                a[j] = max(0.0, a[j] + delta)
            # renormalise to keep sum exactly 1
            total = np.sum(a)
            if total > 0:
                a /= total
            no_improve = 0

        iteration += 1
        # periodic logging
        if iteration % 1000 == 0:
            elapsed = time.time() + budget_s - deadline
            print(f"[{elapsed:.1f}s] iteration {iteration}, best score {best_score:.6f}")

    # final normalisation (important for evaluate_sequence)
    best_a /= np.sum(best_a)
    print(f"Finished after {iteration} iterations, best score {best_score:.6f}")
    # return Python list
    return [float(x) for x in best_a.tolist()]

"""
Hill‑climbing search for a non‑negative step sequence minimizing
evaluate_sequence.  The algorithm repeatedly moves a small amount of mass
from positions that contribute most to the current maximal convolution
value to positions that contribute least, normalising after each move.
Random perturbations are injected when progress stalls.  The search runs
until the allotted time expires and returns the best sequence discovered.
"""
import time
import numpy as np

def full_convolution(a, size):
    """Return the linear convolution a*a (full length) using FFT."""
    af = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(af * af, n=size)
    return conv[:2 * len(a) - 1]

def compute_contributions(a, peak_idxs):
    """Compute for each index i the total weight a_{k-i} over all peak indices k."""
    n = len(a)
    contrib = np.zeros(n, dtype=np.float64)
    for k in peak_idxs:
        start = max(0, k - (n - 1))
        end = min(k, n - 1)
        i = np.arange(start, end + 1)
        contrib[i] += a[k - i]
    return contrib

        conv = full_convolution(a, fft_size)
        max_val = np.max(conv)

        peak_idxs = np.where(np.abs(conv - max_val) < 1e-12)[0]
