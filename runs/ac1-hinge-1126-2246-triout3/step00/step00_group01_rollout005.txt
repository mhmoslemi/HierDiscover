def propose_candidate(seed=42, budget_s=1000):
    import numpy as np, time, math
    np.random.seed(seed)
    n = len(height_sequence_1)
    # start from random Dirichlet
    a = np.random.rand(n)
    a = a / a.sum()
    # keep best
    best_a = a.copy()
    best_score = evaluate_sequence(best_a.tolist())
    deadline = time.time() + budget_s - 5
    # precompute FFT size
    n2 = 1
    while n2 < 2*n: n2 <<= 1
    #function to compute conv via fft
    def conv_fft(x):
        fx = np.fft.rfft(x, n=n2)
        conv_full = np.fft.irfft(fx*fx, n=n2)
        return conv_full[:2*n-1]
    it = 0
    while time.time() < deadline:
        # compute conv
        conv = conv_fft(a)
        max_idx = int(conv.argmax())
        # compute partner
        i_start = max(0, max_idx - (n-1))
        i_end = min(n-1, max_idx)
        partner = np.zeros(n)
        if i_start <= i_end:
            partner[i_start:i_end+1] = a[i_start:i_end+1][::-1]
        # choose indices
        # high weight where partner large
        high_idx = np.argmax(partner)  # will be some index with largest partner
        # low weight where partner small
        low_idx = np.argmin(partner)   # may be edges (partner zero). To avoid zero, choose lowest but with positive a
        # ensure high_idx != low_idx && a[high_idx] > eps

            a[low_idx] += eps
            # enforce non-negative

            # renormalize to keep sum 1 (tiny drift)

        else:
            # random perturb

"""
Propose a candidate step‑function (non‑negative sequence) that improves the
evaluation score 2·n·max(conv(a,a))/sum(a)².

The method is a simple projected sub‑gradient descent on the quantity
max_k (a*a)[k].  At each iteration we:

1. Compute the convolution a*a by FFT (O(n log n)).
2. Locate the index k_max where the convolution attains its maximum.
3. The sub‑gradient of the max‑function w.r.t. a_i is
       g_i = 2·a_{k_max-i}   (if k_max-i lies in the array, otherwise 0).
   This tells us how a_i contributes to the current peak.
4. Take a small step opposite to the gradient, clip negatives to zero and
   renormalise the vector so that it stays a probability distribution
   (sum = 1).  Renormalisation is a cheap projection onto the simplex.
5. Periodically evaluate the true objective with ``evaluate_sequence`` and
   keep the best sequence seen so far.

The algorithm starts from a random Dirichlet point (to break the perfect
symmetry of the uniform sequence) and runs until the allocated time budget
(~1000 s) is exhausted.  Simple random perturbations are injected every few
thousand steps to avoid stagnation.

The returned list is the best sequence discovered during the run.
"""

import time
import numpy as np

def _project_onto_simplex(v):
    """Project a vector onto the probability simplex (sum=1, v>=0)."""
    # Implementation of the O(n log n) algorithm by
    # Wang & Carreira‑Perpiñán (2013)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # All entries are zero – return uniform distribution
        return np.full(n, 1.0 / n, dtype=v.dtype)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def propose_candidate(seed: int = 42, budget_s: float = 1000.0, **kwargs):
    """
    Search for a sequence that minimises `evaluate_sequence`.

    Parameters
    ----------
    seed : int
        Random seed for reproducibility.
    budget_s : float
        Time budget in seconds (the function must return before the budget expires).

    Returns
    -------
    list[float]
        The best sequence found (list of non‑negative floats).
    """
    np.random.seed(seed)

    # length of the problem – we reuse the length of the provided construction
    n = len(height_sequence_1)

    # ----- FFT helper --------------------------------------------------------
    # Choose a power‑of‑two length for efficient convolutions.
    n_fft = 1
    while n_fft < 2 * n:
        n_fft <<= 1

    def _conv_fft(x):
        """Return the linear convolution of x with itself (length 2·n‑1) using FFT."""
        fx = np.fft.rfft(x, n=n_fft)
        conv_full = np.fft.irfft(fx * fx, n=n_fft)
        return conv_full[: 2 * n - 1]

    # ----- initialise a random probability vector ---------------------------
    a = np.random.rand(n)
    a /= a.sum()                     # now sum(a) == 1
    best_a = a.copy()
    best_score = evaluate_sequence(best_a.tolist())

    start_time = time.time()
    deadline = start_time + budget_s - 5.0   # a small safety margin
    it = 0
    # learning‑rate schedule – start relatively aggressive and decay slowly
    initial_lr = 0.2

    while time.time() < deadline:
        # 1) current convolution and location of the maximal entry
        conv = _conv_fft(a)
        k_max = int(conv.argmax())

        # 2) sub‑gradient of the max‑function
        #    g_i = 2 * a_{k_max - i}  (if the partner index is inside the array)
        i_start = max(0, k_max - (n - 1))
        i_end = min(n - 1, k_max)
        g = np.zeros_like(a)
        if i_start <= i_end:
            idx = np.arange(i_start, i_end + 1, dtype=int)
            partner = k_max - idx                     # partner indices
            g[idx] = 2.0 * a[partner]

        # 3) step size (linear decay, never smaller than a tiny epsilon)
        lr = max(initial_lr * (1.0 - it / 200_000.0), 1e-6)

        # 4) gradient step, clip negatives and renormalise onto the simplex
        a = a - lr * g
        a = np.clip(a, 0.0, None)          # enforce non‑negativity
        if a.sum() == 0.0:                 # extremely unlikely, but guard against it
            a = np.full(n, 1.0 / n)
        else:
            a = _project_onto_simplex(a)   # quick projection (sum = 1)

        it += 1

        # ----- occasional evaluation & bookkeeping --------------------------
        if it % 200 == 0:
            # true objective (uses the same clipping rules as the competition)
            cur_score = evaluate_sequence(a.tolist())
            if cur_score < best_score:
                best_score = cur_score
                best_a = a.copy()
                print(f"[{it}]  New best score: {best_score:.6f}")

        # ----- inject a small random jitter every few thousand steps --------
        if it % 5000 == 0:
            jitter = np.random.randn(n) * 1e-4
            a = a + jitter
            a = np.clip(a, 0.0, None)
            a = _project_onto_simplex(a)

    # final reporting
    final_score = evaluate_sequence(best_a.tolist())
    print(f"Finished after {it} iterations. Best score = {final_score:.6f}")
    return best_a.tolist()
