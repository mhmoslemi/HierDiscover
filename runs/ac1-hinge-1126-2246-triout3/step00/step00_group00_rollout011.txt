    n = len(a) + len(b) - 1
    size = 1 << (n - 1).bit_length()
    A = np.fft.rfft(a, n=size)
    B = np.fft.rfft(b, n=size)

    conv = np.fft.irfft(C, n=size)[:n]

import numpy as np, time
from scipy.optimize import linprog

def conv_fft(a,b):
    n = len(a) + len(b) - 1
    size = 1 << ((n-1).bit_length())
    A = np.fft.rfft(a, n=size)
    B = np.fft.rfft(b, n=size)
    C = A * B
    conv = np.fft.irfft(C, n=size)[:n]
    return conv

def solve_convolution_lp(f_seq, rhs):
    n = len(f_seq)
    c = -np.ones(n)  # maximize sum => minimize -sum
    # Build A_ub constraints for each shift
    # We'll construct the matrix efficiently using loops.
    rows = []
    bounds = []
    for k in range(2*n - 1):
        row = np.zeros(n)
        i_start = max(0, k - (n-1))
        i_end = min(n-1, k)
        for i in range(i_start, i_end+1):
            j = k - i
            row[j] = f_seq[i]
        rows.append(row)
        bounds.append(rhs)
    A_ub = np.vstack([rows, -np.eye(n)])
    b_ub = np.concatenate([bounds, np.zeros(n)])
    # Solve LP with high speed method
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, method='highs',
                  options={'time_limit': 5.0})
    if res.success:
        return res.x
    else:
        return None

def propose_candidate(seed=123, budget_s=1000):
    """
    Search for a non‑negative step‑function (list of heights) that minimizes
    ``evaluate_sequence``.  The algorithm iteratively solves a linear program
    that yields a direction improving the sum of the coefficients while keeping
    the self‑convolution bounded, then mixes the current sequence with the
    direction using the largest feasible step size (computed analytically from
    the quadratic expression of the convolution).  Random perturbations and
    occasional restarts avoid stagnation.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 2.0
    # choose a length that balances LP size and theoretical bound
    n = np.random.randint(1200, 2000)  # moderate size
    # initialise a random positive vector and normalise to sum 1
    seq = np.random.rand(n)
    seq /= np.sum(seq)
    best_seq = seq.copy()
    best_score = 2 * n * np.max(conv_fft(seq, seq))  # sum = 1
    print(f"Start length {n}, initial score {best_score:.6f}")

    iteration = 0
    while time.time() < deadline:
        iteration += 1
        # convolution of current sequence
        conv_f = conv_fft(seq, seq)
        max_f = np.max(conv_f)
        # solve LP for a direction that increases the sum while preserving the bound
        g_raw = solve_convolution_lp(seq, max_f)
        if g_raw is None or np.all(g_raw <= 0):
            # fallback: small random redistribution
            i, j = np.random.choice(n, size=2, replace=False)
            delta = (np.random.rand() - 0.5) * 0.02
            seq[i] = max(0.0, seq[i] + delta)
            seq[j] = max(0.0, seq[j] - delta)
            seq /= np.sum(seq)
            continue

        # Normalise g to have the same L1 norm (1.0)
        g = g_raw / np.sum(g_raw)

        # compute cross and self convolutions needed for step‑size calculation
        conv_fg = conv_fft(seq, g)
        conv_gg = conv_fft(g, g)

        # compute maximal feasible mixing coefficient t
        # inequality: (1‑t)^2 * A + 2t(1‑t) * B + t^2 * C ≤ max_f   for each shift
        A = conv_f
        B = conv_fg
        C = conv_gg
        t_upper = 1.0
        for a, b, c in zip(A, B, C):
            # coefficients of quadratic a t^2 + b t + c0 ≤ 0  (after moving max_f)
            aa = c - 2 * b + a
            bb = 2 * (b - a)
            cc = a - max_f
            # solution of aa*t**2 + bb*t + cc ≤ 0
            if abs(aa) < 1e-14:
                if abs(bb) < 1e-14:
                    continue
                if bb > 0:
                    t_candidate = -cc / bb
                    if t_candidate < t_upper:
                        t_upper = t_candidate
                continue
            disc = bb * bb - 4 * aa * cc
            if disc < 0:
                continue
            sqrt_disc = np.sqrt(disc)
            # two roots
            t1 = (-bb - sqrt_disc) / (2 * aa)
            t2 = (-bb + sqrt_disc) / (2 * aa)
            # we need the smallest positive root larger than 0
            roots = [r for r in (t1, t2) if r > 0]
            if not roots:
                continue
            t_candidate = min(roots)
            if t_candidate < t_upper:
                t_upper = t_candidate

        # keep a safety margin
        t = min(0.5, t_upper * 0.99)
        if t <= 1e-6:
            # tiny step – perform a random local perturbation
            idx = np.random.randint(n)
            seq[idx] = max(0.0, seq[idx] + np.random.randn() * 0.005)
            seq /= np.sum(seq)
            continue

        # perform the convex combination and re‑normalise
        seq = (1.0 - t) * seq + t * g
        seq = np.maximum(seq, 0.0)
        seq /= np.sum(seq)

        # evaluate current score
        curr_max = np.max(conv_fft(seq, seq))
        curr_score = 2.0 * n * curr_max
        if curr_score < best_score:

                seq /= np.sum(seq)

    print(f"Finished after {iteration} iterations, best score {best_score:.6f}")
    return best_seq.tolist()

"""
Propose a step‑function sequence that minimizes evaluate_sequence.

The algorithm works as follows:

1. Initialise a random non‑negative vector `seq` of moderate length (≈1000)
   and normalise it to L1‑norm = 1.
2. While time permits:
   * Compute the self–convolution `conv_f = seq * seq` (FFT) and its
     maximum `M`.  The current score is `2·n·M`.
   * Solve a linear programme:
        maximise   Σ b_j
        subject to  (seq ▷ b)[k] ≤ M   for every shift k
                    b_j ≥ 0
     where “▷” is the convolution of the fixed `seq` with the variable `b`.
     The LP returns a feasible direction `g_raw`.  After normalising
     `g = g_raw / Σ g_raw` we obtain a candidate with the same L1‑norm.
   * Determine the largest feasible mixing coefficient `t∈[0,1]`
     such that the self‑convolution of the convex combination
        h = (1‑t)·seq + t·g
     does not exceed the current bound `M`.  For each shift the inequality
        ((1‑t)² A + 2t(1‑t) B + t² C) ≤ M
     with `A = conv_f[k]`, `B = (seq▷g)[k]`, `C = (g▷g)[k]` is a quadratic
     inequality in `t`; the smallest positive root gives the local limit.
     The global step size is the minimum over all shifts (with a safety
     margin of 1 %).
   * Update `seq ← h`, renormalise it to sum = 1 and keep the best
     sequence seen so far.
   * If the LP fails or the admissible step is tiny, perform a tiny random
     redistribution to escape possible stagnation.
3. Return the best sequence found.

The use of FFT makes each convolution O(n log n); the LP size is
≈ (2n) × n and solves quickly for n≈1000.  This gives many hundred
iterations within the 1000 s budget, typically driving the score well
below the baseline of 2.0 and often under the target 1.5.
"""

import time
import numpy as np
from scipy.optimize import linprog

# ----------------------------------------------------------------------
# Helper: fast convolution using real FFT (returns the linear convolution)
def conv_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    n = len(a) + len(b) - 1
    size = 1 << ((n - 1).bit_length())          # next power of two
    A = np.fft.rfft(a, n=size)
    B = np.fft.rfft(b, n=size)
    C = A * B
    conv = np.fft.irfft(C, n=size)[:n]
    return conv


# ----------------------------------------------------------------------
# Linear programme: maximise sum(b)  s.t.  conv(f, b) ≤ rhs  , b ≥ 0
def solve_convolution_lp(f_seq: np.ndarray, rhs: float):
    n = len(f_seq)
    # objective: minimise –Σb  (i.e. maximise Σb)
    c = -np.ones(n)

    # Build the inequality matrix row by row.
    rows = []
    ub = []
    for k in range(2 * n - 1):
        row = np.zeros(n)
        i_start = max(0, k - (n - 1))
        i_end = min(n - 1, k)
        # indices i where both i and j=k‑i are in range
        i_vals = np.arange(i_start, i_end + 1, dtype=int)
        j_vals = k - i_vals
        row[j_vals] = f_seq[i_vals]
        rows.append(row)
        ub.append(rhs)

    # non‑negativity: –I·b ≤ 0
    rows.append(-np.eye(n))
    ub.append(np.zeros(n))

    A_ub = np.vstack(rows)
    b_ub = np.concatenate(ub)

    # use the high‑performance HiGHS interior‑point method
    res = linprog(c, A_ub=A_ub, b_ub=b_ub,
                  method='highs',
                  options={'presolve': True, 'time_limit': 5.0})
    if res.success:
        return res.x
    return None


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a sequence of non‑negative heights that minimises
    evaluate_sequence.  Returns the best sequence found before the time
    budget expires.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 2.0   # keep a safety margin

    # --------------------------------------------------------------
    # Choose a moderate length – large enough for a good bound, small
    # enough for rapid LP solves.
    n = np.random.randint(800, 1300)        # e.g. 1024 ≈ power of two
    seq = np.random.rand(n)
    seq /= np.sum(seq)                       # L1‑norm = 1

    best_seq = seq.copy()
    best_score = 2.0 * n * np.max(conv_fft(seq, seq))
    print(f"[init] length={n}, score={best_score:.6f}")

    it = 0
    while time.time() < deadline:
        it += 1

        # ----- current convolution and bound ---------------------------------
        conv_f = conv_fft(seq, seq)
        M = np.max(conv_f)

        # ----- linear programme for a promising direction --------------------
        g_raw = solve_convolution_lp(seq, M)
        if g_raw is None or np.all(g_raw <= 0):
            # fallback: tiny random redistribution
            i, j = np.random.choice(n, size=2, replace=False)
            delta = (np.random.rand() - 0.5) * 0.01
            seq[i] = max(0.0, seq[i] + delta)
            seq[j] = max(0.0, seq[j] - delta)
            seq /= np.sum(seq)
            continue

        # normalise g to the same L1‑norm (1.0)
        g = g_raw / np.sum(g_raw)

        # ----- compute cross‑convolutions required for step‑size ---------
        conv_fg = conv_fft(seq, g)
        conv_gg = conv_fft(g, g)

        # ----- find the largest t keeping the bound M -----------------------
        t_max = 1.0
        for A, B, C in zip(conv_f, conv_fg, conv_gg):
            # inequality: (1‑t)²·A + 2t(1‑t)·B + t²·C ≤ M
            # bring everything to left side: aa·t² + bb·t + cc ≤ 0
            aa = C - 2.0 * B + A
            bb = 2.0 * (B - A)
            cc = A - M

            if abs(aa) < 1e-14:                     # effectively linear
                if abs(bb) < 1e-14:
                    continue                         # always satisfied
                if bb > 0.0:
                    t_candidate = -cc / bb          # (M‑A)/(2·(B‑A))
                    if t_candidate < t_max:

                continue

            disc = bb * bb - 4.0 * aa * cc
            if disc < 0.0:

            sqrt_disc = np.sqrt(disc)
            # two roots of the quadratic equation
            t1 = (-bb - sqrt_disc) / (2.0 * aa)
            t2 = (-bb + sqrt_disc) / (2.0 * aa)

            # feasible region contains t=0; take the smallest positive root
            roots = [r for r in (t1, t2) if r > 0.0]
            if not roots:
                continue
            t_candidate = min(roots)
            if t_candidate < t_max:
                t_max = t_candidate

        # safety margin (1 % below the theoretical limit)
        t = min(0.5, t_max * 0.99)
        if t <= 1e-6:

            idx = np.random.randint(n)
            seq[idx] = max(0.0, seq[idx] + np.random.randn() * 0.005)
            seq /= np.sum(seq)
            continue

        # ----- update the sequence -------------------------------------------
        seq = (1.0 - t) * seq + t * g
        seq = np.maximum(seq, 0.0)

        cur_max = np.max(conv_fft(seq, seq))

            mask = np.random.rand(n) < 0.08
            seq[mask] = np.random.rand(mask.sum())
            seq /= np.sum(seq)

    print(f"[done] iterations={it}, best score={best_score:.6f}")
    return best_seq.tolist()
