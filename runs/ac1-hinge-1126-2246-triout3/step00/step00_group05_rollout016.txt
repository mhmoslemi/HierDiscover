"""
Propose a sequence of non‑negative heights that minimises

    2 * n * max(conv(a, a)) / (sum(a) ** 2)

where ``conv`` is the linear convolution.  The implementation uses a
smooth‑max (log‑sum‑exp) surrogate for the infinity norm and performs
gradient descent on that surrogate while keeping the total sum of the
sequence fixed (the objective is scale‑invariant).  The gradient is
computed efficiently with two FFT‑based convolutions.  Simple
simulated‑annealing style acceptance is added to escape shallow local
minima and occasional random restarts keep the search from stagnating.
The routine runs for the allotted time budget (default 1000 s) and
returns the best sequence found.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper functions – all defined at the module level (no closures)
# ----------------------------------------------------------------------


def _smoothed_weights(conv: np.ndarray, beta: float) -> np.ndarray:
    """
    Return a vector w_k = exp(beta * (conv_k - max(conv)))  (0 ≤ w_k ≤ 1).
    The weight is 1 at the maximal convolution entry and decays
    exponentially elsewhere.
    """
    # max value is taken from the supplied array (no extra copy)
    m = conv.max()
    # all entries are ≤ m, so the exponent is ≤ 0 → no overflow.
    return np.exp(beta * (conv - m))


def _gradient_of_smoothed_max(seq: np.ndarray, beta: float) -> tuple[np.ndarray, float]:
    """
    Compute the gradient of the smoothed‑max of the self‑convolution
    with respect to ``seq`` and return (gradient, current_max).

    The gradient of  (1/β) log Σ exp(β * conv)   is
        2 * Σ_k w_k * a_{k-i}
    where w_k are the smoothed weights from `_smoothed_weights`.
    """
    # convolution a * a  (full length 2n-1)
    conv = np.convolve(seq, seq)
    # smoothed weights centred at the current maximum
    w = _smoothed_weights(conv, beta)

    # gradient_i = 2 * Σ_j w_{i+j} * a_j
    # This is exactly a convolution of w with the reversed sequence.
    grad_full = np.convolve(w, seq[::-1])
    n = len(seq)
    grad = 2.0 * grad_full[n - 1 : n - 1 + n]   # slice of length n
    return grad, conv.max()


def _evaluate(seq: np.ndarray) -> float:

    return evaluate_sequence(seq.tolist())
