```
def convolve_fft(a):
    n = len(a)
    size = 1 << ((2*n-1).bit_length()) # next power of 2
    A = np.fft.rfft(a, n=size)
    B = A * A
    conv = np.fft.irfft(B, n=size)
    return conv[:2*n-1]
```

```
def fft_convolve(a):
    n = a.shape[0]
    size = 1 << ((2*n -1).bit_length())
    A = np.fft.rfft(a, size)
    conv = np.fft.irfft(A * A, size)
    return conv[:2*n-1]
```

```
def local_search(a, time_deadline):
    n = len(a)
    target_sum = np.sqrt(2*n)
    best_local = a.copy()
    best_score = evaluate_sequence(a.tolist())
    iteration = 0
    while time.time() < time_deadline:
        # compute convolution via FFT
        conv = fft_convolve(a)
        M = conv.max()
        k_max = int(conv.argmax())
        # compute contributions
        # compute partner indices vectorized for all i
        i = np.arange(n)
        j = k_max - i
        mask = (j >= 0) & (j < n)
        contrib = np.zeros(n)
        contrib[mask] = a[i[mask]] * a[j[mask]]
        # find i_max where contrib max
        i_max = int(np.argmax(contrib))
        j_max = int(k_max - i_max)
        # if j_max out of range (should not happen)
        if not (0 <= j_max < n):
            # skip, small perturbation
            idx = np.random.randint(n)
            a[idx] += np.random.randn() * 0.001
            a = np.maximum(a, 0.0)
            a = a * (target_sum / a.sum())
            continue
        # choose delta
        # Scale delta to fraction of smaller of a[i_max], a[j_max]
        max_frac = 0.5
        min_val = min(a[i_max], a[j_max])
        if min_val <= 0:
            delta = 0.0
        else:
            delta = min_val * np.random.uniform(0.05, max_frac)
        if delta <= 1e-12:
            # random perturb
            idx = np.random.randint(n)
            a[idx] = max(0.0, a[idx] + np.random.randn() * 0.001)
            a = a * (target_sum / a.sum())
            continue
        # find low-evaluation sum L
        # Use list of sums with low conv values
        low_candidates = np.where(conv < M * 0.7)[0]
        if low_candidates.size == 0:
            L = np.random.randint(2*n-1)
        else:
            L = int(np.random.choice(low_candidates))
        # pick p such that q = L - p in range
        attempts = 0
        while True:
            p = np.random.randint(n)
            q = L - p
            if 0 <= q < n:
                break
            attempts += 1
            if attempts > 10:
                p = np.random.randint(n)
                q = np.random.randint(n)
                break
        # apply move
        a_new = a.copy()
        a_new[i_max] -= delta
        a_new[j_max] -= delta
        a_new[p] += delta
        a_new[q] += delta
        # ensure nonnegative (errors may cause tiny negative)
        a_new = np.maximum(a_new, 0.0)
        # re-normalize sum to target (due to rounding)
        total = a_new.sum()
        if total == 0:
            continue
        a_new *= (target_sum / total)
        # compute evaluate
        score_new = evaluate_sequence(a_new.tolist())
        if score_new < best_score:
            best_score = score_new
            best_local = a_new.copy()
            a = a_new
            # print improvement
            print(f"Iter {iteration}, new best: {best_score:.6f}")
        else:
            # maybe accept w/ prob
            # temperature schedule
            T = max(0.001, (time_deadline - time.time()) / (budget_s))
            # Not essential; we may reject
            # Also maybe revert w/out acceptance
            # but we can still keep a for further attempts
            # we could sometimes accept
            if np.random.rand() < 0.001:
                a = a_new
        iteration += 1
    return best_local, best_score
```

    Generate a monotone linear step function.
    ------------------------------------------------------------
    Theory
    ------------------------------------------------------------
    Let `a_i ≥ 0` (i = 0…n‑1) and Σ a_i = √(2 n).  The evaluation

        E(a) = 2 n·max_k Σ_i a_i·a_{k‑i} / (Σ a_i)² = max_k Σ_i a_i·a_{k‑i}

    Hence `E(a) ≈ 1.333…`, which is well below the current bound 2.0

    ------------------------------------------------------------
    Implementation
    ------------------------------------------------------------
