def project_to_simplex(v):
    # ensure sum(v) = 1 and v >= 0.
    # Implementation: sort descending, compute cumulative sum, find rho, compute theta.
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    rho = np.where(u - cssv/(np.arange(len(v))+1) > 0)[0][-1]
    theta = cssv[rho] / (rho+1)
    w = np.maximum(v - theta, 0)
    return w

"""
Search for a non‑negative step function (height sequence) that minimises
the provided ``evaluate_sequence`` objective.

The algorithm works in two stages:

1. **Heuristic initialisation** – generate several candidate sequences
   using simple analytic windows (uniform, Hann, arcsine, exponential)
   for a few moderate lengths (50‑500).  Each candidate is normalised to
   have unit sum (the objective is scale‑invariant) and evaluated with
   ``evaluate_sequence``.  The best candidate so far is kept.

2. **Projected gradient descent** – starting from the best candidate,
   iteratively reduce the maximal convolution coefficient.
   The gradient of a smooth soft‑max approximation of the max‑convolution
   is computed analytically:
        ∂max/∂a_i ≈ 2·∑_k w_k·a_{k‑i},
   where w_k are soft‑max weights of the convolution values.
   After a gradient step the vector is projected onto the probability
   simplex (non‑negative, sum = 1).  The learning rate decays slowly,
   the temperature of the soft‑max grows, and occasional random
   perturbations are injected to avoid stagnation.

The whole search respects the supplied time budget (default 1000 s) and
returns the best sequence discovered.  All numbers are non‑negative,
and the routine never performs I/O or uses closures.
"""
import time
import numpy as np


def _project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Project a non‑negative vector onto the simplex {x ≥ 0, sum(x)=1}."""
    # Implementation of the algorithm from
    #   Wang & Carreira‑Perpiñán, “Projection onto the probability simplex”
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(len(v)) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # fallback – return uniform vector
        return np.ones_like(v) / len(v)
    rho = ind[cond][-1] - 1  # zero‑based index
    theta = cssv[cond][-1] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def _softmax_weights(b: np.ndarray, lam: float) -> np.ndarray:
    """Soft‑max weighting of the convolution array b (lam>0)."""
    # shift by max to avoid overflow
    mb = np.max(b)
    ex = np.exp(lam * (b - mb))
    return ex / np.sum(ex)


def _gradient_step(a: np.ndarray, lam: float, lr: float) -> np.ndarray:
    """One projected gradient step aiming to reduce max(a*a)."""
    n = a.shape[0]
    conv = np.convolve(a, a)
    w = _softmax_weights(conv, lam)                     # shape (2n‑1)
    # grad = 2 * (a_rev * w)  (cross‑correlation)
    grad_full = np.convolve(a[::-1], w)                # length 3n‑2
    grad = 2.0 * grad_full[n - 1: 2 * n - 1]            # keep n elements
    a_new = a - lr * grad
    a_new = np.maximum(a_new, 0.0)
    a_new = _project_to_simplex(a_new)
    return a_new, conv


def _gen_uniform(n: int) -> np.ndarray:
    a = np.ones(n, dtype=float)
    return a / a.sum()


def _gen_hann(n: int) -> np.ndarray:
    # Hann window = 0.5·(1‑cos(2πi/(n‑1)))
    i = np.arange(n, dtype=float)
    a = 0.5 * (1.0 - np.cos(2.0 * np.pi * i / (n - 1)))
    return a / a.sum()


def _gen_arcsine(n: int) -> np.ndarray:
    # Discrete approximation of pdf 1/(π√(x(1‑x))) on (0,1)
    i = np.arange(n, dtype=float)
    x = (i + 0.5) / n                     # avoids 0 and 1
    a = 1.0 / np.sqrt(x * (1.0 - x))
    a[np.isinf(a)] = 0.0
    return a / a.sum()


def _gen_exponential(n: int, alpha: float) -> np.ndarray:
    centre = (n - 1) / 2.0
    i = np.arange(n, dtype=float)
    a = np.exp(-alpha * np.abs(i - centre))
    return a / a.sum()


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """Return a list of non‑negative heights that (hopefully) minimises
    ``evaluate_sequence`` within the time budget."""
    np.random.seed(seed)
    deadline = time.time() + budget_s - 5  # a small safety margin

    best_seq = None
    best_score = float('inf')

    # -----------------------------------------------------------------
    # Phase 1 – collect a few hand‑crafted starting points
    # -----------------------------------------------------------------
    lengths = [80, 120, 160, 200, 260, 340]      # modest sizes – cheap to eval
    generators = [
        _gen_uniform,
        _gen_hann,
        _gen_arcsine,
        lambda n: _gen_exponential(n, alpha=0.02),
        lambda n: _gen_exponential(n, alpha=0.05),
    ]

    for n in lengths:
        for gen in generators:
            a = gen(n).astype(float)
            # convert to plain list for evaluation
            seq = a.tolist()
            try:
                sc = evaluate_sequence(seq)
            except Exception:
                sc = float('inf')
            if sc < best_score:
                best_score = sc
                best_seq = a.copy()
                print(f"[Init] length={n} gen={gen.__name__} score={sc:.6f}")
                if best_score <= 1.503:
                    print("[Init] target reached")
                    return best_seq.tolist()
        # also try a random Dirichlet start
        a = np.random.rand(n)
        a /= a.sum()
        seq = a.tolist()
        try:
            sc = evaluate_sequence(seq)
        except Exception:
            sc = float('inf')
        if sc < best_score:
            best_score = sc
            best_seq = a.copy()
            print(f"[Init] length={n} random score={sc:.6f}")

    if best_seq is None:
        # fallback – uniform length 10
        best_seq = _gen_uniform(10)
        best_score = evaluate_sequence(best_seq.tolist())
        print("[Fallback] using tiny uniform sequence")

    # -----------------------------------------------------------------
    # Phase 2 – projected gradient descent from the best seed
    # -----------------------------------------------------------------
    a = best_seq.astype(float)          # current iterate, sum==1
    n = a.shape[0]
    iteration = 0
    lam = 5.0        # initial soft‑max temperature
    lr = 0.05        # initial learning rate

    while time.time() < deadline:
        # one gradient update
        a, conv = _gradient_step(a, lam, lr)

        # evaluate (fast, because sum(a)==1)
        max_b = conv.max()
        score = 2.0 * n * max_b        # exactly evaluate_sequence for sum=1

        if score < best_score:
            best_score = score
            best_seq = a.copy()
            print(f"[Iter {iteration}] new best score={score:.6f} (n={n})")
            if best_score <= 1.503:
                print("[Iter] target achieved – exiting early")
                break

        # schedule adjustments
        iteration += 1
        # slowly increase temperature (focus more on the true max)
        lam = min(30.0, lam + 0.005)          # up to about 30
        # decay learning rate
        lr = max(0.002, lr * 0.9995)

        # occasional random jitter to escape tiny plateaus
        if iteration % 300 == 0:
            jitter = np.random.randn(n) * 0.001
            a = np.maximum(a + jitter, 0.0)
            a = _project_to_simplex(a)

        # if we have stagnated for a while, try a fresh random start
        if iteration % 2000 == 0:
            a = np.random.rand(n)
            a = _project_to_simplex(a)

    # Return the best sequence found (as plain Python list)
    return best_seq.tolist()
