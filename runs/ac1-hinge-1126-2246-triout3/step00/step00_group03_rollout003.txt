def fast_eval(seq):
    arr = np.clip(np.asarray(seq, dtype=float), 0.0, 1000.0)
    sum_a = arr.sum()
    if sum_a < 0.01:
        return float('inf')
    n = arr.shape[0]
    # using FFT for conv
    size = int(2**np.ceil(np.log2(2*n-1)))
    fft_a = np.fft.rfft(arr, n=size)
    conv = np.fft.irfft(fft_a * fft_a, n=size)[:2*n-1]
    max_b = conv.max()
    return float(2.0 * n * max_b / (sum_a * sum_a))

def flatten_move(a, max_idx, step, target_sum):
    n = len(a)
    # Determine pairs (i,j) for max_idx
    i_start = max(0, max_idx - (n-1))
    i_end = min(n-1, max_idx)
    # choose pair with largest product a[i] * a[j]
    best_prod = -1
    best_pair = (i_start, max_idx - i_start)  # default
    for i in range(i_start, i_end+1):
        j = max_idx - i
        prod = a[i] * a[j]
        if prod > best_prod:
            best_prod = prod
            best_pair = (i, j)
    i, j = best_pair
    eps = step * min(a[i], a[j])
    if eps <= 0:
        return a
    a_new = a.copy()
    a_new[i] -= eps
    a_new[j] -= eps
    # find low indices
    low_idx = np.argpartition(a_new, 2)[:2]  # indices of two smallest values
    # If low_idx includes i or j (unlikely since we decreased them), it's fine; we can exclude them
    for k in low_idx:
        a_new[k] += eps / 2  # distribute equally?
    # Ensure nonnegative
    a_new = np.clip(a_new, 0.0, 1000.0)
    # renorm sum
    sum_new = a_new.sum()
    if sum_new > 0:
        a_new = a_new * (target_sum / sum_new)

"""
Simple simulated‑annealing optimizer for the step‑function bound.

The algorithm works with a fixed length `n`.  For a given sequence `a`
( non‑negative, clipped to [0,1000] ) the evaluation metric reduces,
after normalising the ℓ¹‑norm to √(2 n), to the maximal entry of the
autoconvolution `conv = a * a`.  The goal is therefore to minimise
`max(conv)`.

We start from a uniform (flat) sequence, then repeatedly:

* compute the convolution maximum and its index,
* move a small amount of mass from the pair of entries that
  contribute to that maximal index (taken as the “central” pair)
  to the two currently smallest entries – this flattens the
  convolution,
* accept the move if it improves the objective or with a
  temperature‑dependent probability (simulated annealing).

The temperature is exponentially cooled.  Random “mass‑transfer”
moves are interleaved to provide additional exploration.  The
procedure is run for several candidate lengths (400, 800, 1200,
2000) until the global time budget expires, keeping the best
sequence found.  The returned list can be directly fed to
`evaluate_sequence`.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Helper functions (all top‑level, no closures)
# ----------------------------------------------------------------------


def _conv_max_and_index(seq: np.ndarray):
    """
    Compute the maximal entry of the autoconvolution of `seq`
    using a fast FFT and return both the maximum value and its index.
    """
    arr = np.clip(seq, 0.0, 1000.0)
    n = arr.shape[0]
    # length needed for full convolution
    size = 1 << ((2 * n - 1).bit_length())
    # real FFT is sufficient because the sequence is real
    fft_a = np.fft.rfft(arr, n=size)
    conv = np.fft.irfft(fft_a * fft_a, n=size)[: 2 * n - 1]
    max_idx = int(np.argmax(conv))
    max_val = float(conv[max_idx])
    return max_val, max_idx


def _renorm(seq: np.ndarray, target_sum: float):
    """
    Rescale `seq` so that its ℓ¹‑norm becomes exactly `target_sum`.
    Sequence is also clipped to the allowed range [0,1000].
    """
    seq = np.clip(seq, 0.0, 1000.0)
    s = seq.sum()
    if s == 0.0:
        # fallback to a uniform distribution
        seq = np.full_like(seq, target_sum / seq.shape[0])
    else:
        seq = seq * (target_sum / s)
    return seq


def _central_pair_move(seq: np.ndarray, max_idx: int,
                       step: float, target_sum: float):
    """
    Take mass from the pair of indices that (approximately) generate
    `max_idx` (chosen as the centre of the pair) and give it to the
    two smallest entries.  This flattens the convolution peak.
    """
    n = seq.shape[0]
    # pick the “central’’ pair (i , j) with i+j = max_idx
    i = max_idx // 2
    j = max_idx - i
    # make sure the pair is inside the bounds, otherwise fall back
    if i < 0 or i >= n or j < 0 or j >= n:
        # choose a random admissible pair for robustness
        i_start = max(0, max_idx - (n - 1))
        i_end = min(n - 1, max_idx)
        i = np.random.randint(i_start, i_end + 1)
        j = max_idx - i

    # amount of mass to relocate (proportional to the smaller entry)
    eps = step * min(seq[i], seq[j])
    if eps <= 0.0:
        return seq  # nothing to do

    new_seq = seq.copy()
    new_seq[i] -= eps
    new_seq[j] -= eps

    # add the removed mass to the two currently smallest components
    low_idx = np.argpartition(new_seq, 2)[:2]
    # distribute the total removed mass (2*eps) equally
    new_seq[low_idx[0]] += eps
    new_seq[low_idx[1]] += eps

    # keep the sequence feasible
    new_seq = _renorm(new_seq, target_sum)
    return new_seq


def _random_mass_transfer(seq: np.ndarray, target_sum: float,
                          max_frac: float = 0.05):
    """
    Transfer a random fraction (up to `max_frac`) of mass from one
    index to another, preserving the total ℓ¹‑norm.
    """
    n = seq.shape[0]
    i, j = np.random.randint(n, size=2)
    while i == j:
        j = np.random.randint(n)

    # take at most `max_frac` of the source mass
    eps = np.random.rand() * max_frac * seq[i]
    if eps <= 0.0:
        return seq

    new_seq = seq.copy()
    new_seq[i] -= eps
    new_seq[j] += eps
    new_seq = _renorm(new_seq, target_sum)
    return new_seq


# ----------------------------------------------------------------------
# Main candidate generator
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs):
    """
    Search for a step‑function sequence with a small convolution peak.

    Parameters
    ----------
    seed : int
        Random seed for reproducibility.
    budget_s : int
        Approximate time budget in seconds (the caller will kill the
        process after 1000 s; we stop a few seconds early).

    Returns
    -------
    list[float]
        The best sequence found within the time limit.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 5.0   # keep a small safety margin

    # ------------------------------------------------------------------
    # Settings
    # ------------------------------------------------------------------
    # lengths we will explore – increasing them later gives more room to
    # flatten the convolution.
    lengths = [400, 800, 1200, 2000]
    # SA parameters (same for all lengths)
    init_temp = 1.0          # initial temperature
    min_temp = 1e-6
    decay = 0.99995          # exponential cooling per iteration
    move_step = 0.02         # fraction of the smallest involved entry to move
    random_move_prob = 0.25  # how often we perform a purely random move

    best_score = float('inf')
    best_seq = None

    # ------------------------------------------------------------------
    # Main optimisation loop over the different lengths
    # ------------------------------------------------------------------
    for n in lengths:
        if time.time() >= deadline:
            break

        target_sum = math.sqrt(2.0 * n)

        # Initial sequence: uniform (flat) distribution, summed to target_sum
        a = np.full(n, target_sum / n, dtype=float)

        # Small random perturbation to break symmetry early on
        a += np.random.randn(n) * (target_sum / n) * 0.01
        a = _renorm(a, target_sum)

        # Evaluate the current sequence (max of the convolution)
        cur_max, cur_idx = _conv_max_and_index(a)
        cur_score = cur_max            # because sum == √(2n) → eval == max
        temperature = init_temp

        iter_counter = 0
        while time.time() < deadline:
            iter_counter += 1

            # Keep the globally best sequence seen so far
            if cur_score < best_score:
                best_score = cur_score
                best_seq = a.copy()
                print(f"[{time.time() - start_time:6.1f}s] "
                      f"New global best: {best_score:.6f} (n={n}, iter={iter_counter})")

            # Decide which move to try
            if np.random.rand() < random_move_prob:
                a_new = _random_mass_transfer(a, target_sum)
            else:
                a_new = _central_pair_move(a, cur_idx, move_step, target_sum)

            # Evaluate the neighbour
            new_max, new_idx = _conv_max_and_index(a_new)
            new_score = new_max

            # Acceptance test (simulated annealing)
            accept = False
            if new_score < cur_score:
                accept = True
            else:
                # probability to accept a worse move
                prob = math.exp(-(new_score - cur_score) / max(temperature, 1e-12))
                if np.random.rand() < prob:
                    accept = True

            if accept:
                a = a_new
                cur_score = new_score
                cur_idx = new_idx

            # Cool the temperature
            temperature = max(min_temp, temperature * decay)

            # Optional early stop for this length if we are not improving
            # (not essential; kept simple).

    # ------------------------------------------------------------------
    # If everything failed, fall back to a simple uniform sequence.
    # ------------------------------------------------------------------
    if best_seq is None:
        n = 100
        target_sum = math.sqrt(2.0 * n)
        best_seq = np.full(n, target_sum / n, dtype=float)

    # Return a plain Python list (the evaluator expects a list, not ndarray)
    return list(best_seq)
