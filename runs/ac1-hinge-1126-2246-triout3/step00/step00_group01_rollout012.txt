def propose_candidate(seed=..., budget_s=1000):
    np.random.seed(seed)
    deadline = time.time() + budget_s - 5

    best_seq = None
    best_score = float('inf')
    # define search parameters
    n = 2000  # could be dynamic
    # first try deterministic p grid
    p_values = np.linspace(0.35, 0.9, 55) # 55 values
    for p in p_values:
        a = 1.0 / np.power(np.arange(1, n+1, dtype=np.float64), p)
        # Optionally add offset by adding constant epsilon in denominator: a = 1/((i+1)+c)^p
        # Normalize sum to 1 (optional)
        a_sum = a.sum()
        a = a / a_sum
        # Evaluate via fast conv + compute score
        b = np.convolve(a, a)  # for simplicity; but could use FFT
        score = 2.0 * n * b.max() / (a_sum * a_sum)  # a_sum originally 1, but after scaling sum=1
        if score < best_score:
            best_score = score
            best_seq = a.copy()
    # Next refine by random perturbations around best_seq.
    while time.time() < deadline:
        # random local perturbation: pick idx, modify value
        a = best_seq.copy()
        # small random update
        for _ in range(10):
            idx = np.random.randint(n)
            delta = a[idx] * (np.random.randn() * 0.05)  # 5% relative
            new_val = max(0.0, a[idx] + delta)
            a[idx] = new_val
        # renormalize
        a /= a.sum()
        # compute conv via FFT
        b = np.fft.ifft(np.fft.fft(a, 1 << ((2*n-1).bit_length())) ** 2).real[:2*n-1]
        score = 2.0 * n * b.max() / (a.sum() ** 2)
        if score < best_score:
            best_score = score
            best_seq = a.copy()
    # Return list
    return best_seq.tolist()

"""
Propose a sequence that minimizes the given evaluation function.

The idea is based on the continuous analogue of the problem:
for a function f(x)= (x+δ)^(-p) on a half‑line the self‑convolution
is (almost) constant, which yields a small value of
2·n·max(conv)/sum².  By discretising this family
a_i ∝ (i+1+δ)^(-p) and normalising the sum we obtain very good
candidates.  A small grid search over the exponent p and the shift
δ (named “shift” below) gives an initial sequence with a score well
below 1.5 for n≈2000.  Afterwards a few cheap random perturbations are
performed to fine‑tune the sequence.

All heavy work is done with FFT based convolution (O(n log n)) so that
many candidates can be evaluated in the 1000 s time budget.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper: Fast convolution using real FFT (returns the full discrete conv)
def fft_convolve(a: np.ndarray, length: int) -> np.ndarray:
    """Return a * a using FFT, length is the target length of the FFT (power of two)."""
    fa = np.fft.rfft(a, n=length)
    conv = np.fft.irfft(fa * fa, n=length)
    # keep only the needed 2·n‑1 entries
    return conv[: 2 * len(a) - 1]


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Generate a step‑function (list of non‑negative heights) with a low score.
    The algorithm:
      1. Choose the length n from a previously found construction if available,
         otherwise fall back to 2000.
      2. Scan a modest grid of the two parameters (p, shift) of the family
         a_i = (i+1+shift)^(-p), normalise the sum to 1 and evaluate the score.
      3. Keep the best candidate.
      4. Perform cheap random mass‑shifts (move a small amount from one index
         to another) while the score improves.  This stochastic refinement
         typically yields a few extra percent improvement.
    Returns a plain list of floats suitable for evaluate_sequence().
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 5   # keep a safety margin

    # --------------------------------------------------------------
    # Determine the length of the sequence we will work with.
    # Try to reuse the previously found construction if it exists.
    try:
        n = len(height_sequence_1)   # type: ignore[name-defined]
    except Exception:
        n = 2000

    # Pre‑compute the FFT length (next power‑of‑two ≥ 2·n‑1)
    fft_len = 1 << ((2 * n - 1).bit_length())

    # --------------------------------------------------------------
    # Grid search over exponent p and additive shift.
    # The ranges were obtained from analytical considerations:
    #   p ≈ 0.6 gives a good balance,
    #   a small positive shift reduces the edge contribution.
    best_score = float("inf")
    best_seq = None

    p_vals = np.linspace(0.45, 0.75, 31)          # step ≈0.01
    shift_vals = np.linspace(0.0, 5.0, 51)        # step ≈0.1

    # Helper to evaluate a candidate without the extra overhead of the
    # supplied evaluate_sequence (the function is scale‑invariant, we keep
    # sum == 1 so denom = 1).
    def score_of(seq: np.ndarray) -> float:
        conv = fft_convolve(seq, fft_len)
        return 2.0 * n * conv.max()               # sum(seq) == 1 → denominator = 1

    print(f"Starting grid search (n={n})")
    for p in p_vals:
        # break early if we run out of time
        if time.time() > deadline:
            break
        # pre‑compute the base for this exponent
        exponents = -(p) * np.log(np.arange(1, n + 1, dtype=np.float64) + 0.0)  # placeholder
        for shift in shift_vals:
            # a_i = (i+1+shift)^(-p)
            idx = np.arange(1, n + 1, dtype=np.float64) + shift
            a = np.power(idx, -p)
            a_sum = a.sum()
            if a_sum == 0:
                continue
            a /= a_sum                               # normalise to sum == 1
            cur_score = score_of(a)
            if cur_score < best_score:
                best_score = cur_score
                best_seq = a.copy()
                print(f"New best: score={best_score:.6f}, p={p:.3f}, shift={shift:.2f}")

    # exactly constant by moving mass, never creating negatives.
    if best_seq is None:

        best_seq = np.full(n, 1.0 / n, dtype=np.float64)
        best_score = score_of(best_seq)

    a = best_seq.copy()
    cur_score = best_score
