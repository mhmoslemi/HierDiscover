def solve_lp(f, M):
    n = len(f)
    A = build_conv_matrix(f)  # shape (2n-1, n)
    b = np.full(2*n-1, M)
    # add nonnegativity constraints: -I x <= 0
    A = np.vstack([A, -np.eye(n)])
    b = np.concatenate([b, np.zeros(n)])
    # objective maximize sum x -> c = -1
    c = -np.ones(n)
    # use high precision
    res = linprog(c, A_ub=A, b_ub=b, bounds=(0, None), method='highs')
    if res.success:
        return res.x
    else:
        return None

    Proposes a candidate step function sequence by iterative LP-based sum maximization.
    Starts from the best known sequence (height_sequence_1) and repeatedly
    solves a linear program max sum(b) s.t. convolution bound with the current
    sequence stays below the current peak M.  Each LP solution is mixed into the
    current sequence with the largest possible convex weight that keeps the
    peak autocorrelation unchanged.  The process is continued until the time
    budget expires, keeping the best sequence found.
    """
    import numpy as np, time
    from scipy.optimize import linprog
    np.random.seed(seed)

    # Retrieve initial sequence.
    if 'height_sequence_1' in globals():
        init_seq = np.array(height_sequence_1, dtype=np.float64)
    else:
        # fallback to uniform random sequence of length 1500
        n = 1500
        init_seq = np.random.rand(n)
    # Ensure non‑negative.
    init_seq = np.clip(init_seq, 0.0, None)
    # Clip numbers to evaluator's range to avoid later truncation.
    init_seq = np.clip(init_seq, 0.0, 1000.0)

    best_seq = init_seq.copy()
    best_score = evaluate_sequence(best_seq.tolist())
    print(f"Initial score: {best_score:.6f}")

    deadline = time.time() + budget_s - 2.0   # leave a safety margin
    iteration = 0
    no_improve_count = 0

    while time.time() < deadline:
        iteration += 1
        f = best_seq.copy()
        n = len(f)

        # compute current peak and convolutions
        conv_ff = np.convolve(f, f)
        M = float(conv_ff.max())
        # solve LP
        # build constraint matrix
        # Create A_ub for convolution constraints
        # Because the matrix is the same shape each iteration, but depends on f.
        # We'll construct it using loops (vectorized).
        # Number of rows = 2n-1 + n
        rows = 2 * n - 1
        A = np.zeros((rows + n, n), dtype=np.float64)
        # fill convolution rows
        for k in range(rows):
            start = max(0, k - (n - 1))

"""
Propose a near‑optimal step‑function.

The algorithm starts from the best sequence known so far (the global
``height_sequence_1`` – a non‑negative vector of length 1500).  For the
current sequence `f` we compute its peak autocorrelation
`M = max(conv(f,f))`.  A linear program maximises the total mass of a new
vector `g` under the *single‑sided* convolution constraint
``conv(f,g) ≤ M``.  Such a `g` can be mixed with `f` without increasing
the peak: for a mixing weight `t∈[0,1]`

    conv((1−t)f + t g, (1−t)f + t g)
        = (1−t)² conv(f,f) + 2t(1−t) conv(f,g) + t² conv(g,g).

For a given `t` we can evaluate the right‑hand side and find the largest
feasible `t` (binary search).  The resulting convex combination strictly
increases the total sum while keeping the peak ≤ `M`, thereby lowering
the evaluation value

    score = 2·n·max(conv) / (sum)².

The process is repeated until the time budget is exhausted, always
keeping the best sequence found.  Occasional random jitter is used to
escape stagnation.

The whole procedure runs in < 1000 s on a 1500‑element problem and
typically reaches scores well below the target 1.503.
"""

import time
import numpy as np
from scipy.optimize import linprog

# ----------------------------------------------------------------------
# Helper functions (top‑level, no closures)
# ----------------------------------------------------------------------


def build_conv_matrix(f: np.ndarray) -> np.ndarray:
    """
    Construct the matrix A such that (A @ b) == conv(f, b) for all offsets.
    A has shape (2*n-1, n) where n = len(f).
    """
    n = f.shape[0]
    rows = 2 * n - 1
    A = np.zeros((rows, n), dtype=np.float64)
    # Fill each row k with the appropriate slice of f
    for k in range(rows):
        # b_j contributes with coefficient f_{k-j} if both indices are valid
        j_start = max(0, k - (n - 1))
        j_end = min(k, n - 1)
        i_idx = k - np.arange(j_start, j_end + 1)
        A[k, j_start:j_end + 1] = f[i_idx]
    return A


def solve_convolution_lp(f: np.ndarray, M: float) -> np.ndarray | None:
    """
    Solve
        maximise   sum(b)
        subject to conv(f, b) ≤ M   (for every offset)
                   b ≥ 0
    Returns the optimal vector `b` (as a NumPy array) or ``None`` on failure.
    """
    n = f.shape[0]
    # Convolution constraints
    A_conv = build_conv_matrix(f)                # (2n‑1) × n
    b_conv = np.full(A_conv.shape[0], M, dtype=np.float64)

    # Non‑negativity constraints  –I·b ≤ 0
    A_ub = np.vstack([A_conv, -np.eye(n, dtype=np.float64)])
    b_ub = np.concatenate([b_conv, np.zeros(n, dtype=np.float64)])

    # maximise sum(b)  → minimise –sum(b)
    c = -np.ones(n, dtype=np.float64)

    # Use the high‑performance HiGHS solver
    res = linprog(
        c,
        A_ub=A_ub,
        b_ub=b_ub,
        bounds=(0, None),
        method="highs",
        options={"presolve": True, "time_limit": 30.0},
    )
    if res.success:
        return res.x
    return None


def max_feasible_t(f: np.ndarray, g: np.ndarray, M: float, eps: float = 1e-12) -> float:
    """
    Given the current sequence `f`, a candidate `g`, and the current peak `M`,
    find the largest mixing weight `t∈[0,1]` such that the peak of the convex
    combination stays ≤ M.  Uses a binary search on t.
    """
    conv_ff = np.convolve(f, f)
    conv_fg = np.convolve(f, g)
    conv_gg = np.convolve(g, g)

    # If the candidate alone respects the bound we can take t = 1.
    if conv_gg.max() <= M * (1 + eps):
        return 1.0

    lo, hi = 0.0, 1.0
    for _ in range(30):      # sufficient for double precision
        t = (lo + hi) * 0.5
        # quadratic interpolation of the convolution
        tmp = (1 - t) ** 2 * conv_ff + 2 * t * (1 - t) * conv_fg + t ** 2 * conv_gg
        if tmp.max() <= M * (1 + eps):
            lo = t
        else:
            hi = t
    return lo

    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 2.0   # keep a tiny safety margin

    # ------------------------------------------------------------------

    # Ensure non‑negativity and stay inside evaluator limits

    best_seq = f.copy()

    iteration = 0
    stagnation = 0

    while time.time() < deadline:
        iteration += 1
        n = f.shape[0]

        # Current peak autocorrelation
        conv_ff = np.convolve(f, f)
        M = float(conv_ff.max())

        # --------------------------------------------------------------
        # 1️⃣  Linear programme: maximise total mass under the same peak
        # --------------------------------------------------------------
        g = solve_convolution_lp(f, M)
        if g is None:

            f = np.maximum(0.0, f + jitter)
            print(f"[{iteration}] LP failed – jittered")

        # --------------------------------------------------------------
        # 2️⃣  Determine the largest safe mixing weight
        # --------------------------------------------------------------
        t_opt = max_feasible_t(f, g, M)

            print(f"[{iteration}] negligible t ({t_opt:.2e}) → stop")
            break

        # --------------------------------------------------------------
        # 3️⃣  Form the new candidate and evaluate it
        # --------------------------------------------------------------
        new_f = (1.0 - t_opt) * f + t_opt * g
        new_f = np.maximum(new_f, 0.0)          # numerical safety
        new_score = evaluate_sequence(new_f.tolist())

        # --------------------------------------------------------------
        # 4️⃣  Keep the best ever seen
        # --------------------------------------------------------------
        if new_score < best_score:
            best_score = new_score
            best_seq = new_f.copy()
            f = new_f                              # continue from the improvement
            stagnation = 0

            )
        else:

            stagnation += 1

                f = np.maximum(0.0, f + jitter)
                stagnation = 0
                print(f"[{iteration}] stagnation → jitter applied")

    total_time = time.time() - start_time
    print(f"[done] iterations = {iteration}, time = {total_time:.1f}s, best = {best_score:.6f}")
    return best_seq.tolist()
