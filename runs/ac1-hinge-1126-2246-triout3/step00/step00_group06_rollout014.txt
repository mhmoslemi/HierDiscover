"""
Propose a candidate sequence of non‑negative heights that (hopefully) beats the
current bound of 2.0 for the evaluation function.

The algorithm works in three stages:
1.  A cheap parametric sweep (exponential and power‑law families) to obtain a
    decent starting point.
2.  A short random‑restart phase that keeps the best of many simple random
    draws.
3.  A simulated‑annealing / hill‑climbing refinement that repeatedly moves a
    small amount of mass from one index to another (or redistributes a small
    random subset) while always preserving the total sum.  Each move is kept
    if it improves the score or, with a temperature‑dependent probability,
    if it is slightly worse – this helps to escape local minima.

The evaluation function is scale‑invariant, but we keep the total sum close to
√(2 n) (the normalisation used by the reference code) to stay safely inside
the internal clipping limits.

The routine respects the 1000 s time budget, continuously reports progress,
and finally returns the best sequence found.
"""

import time
import numpy as np
import random
from math import exp, sqrt, log

# ----------------------------------------------------------------------
# Helper functions (no closures, no lambdas as required)
# ----------------------------------------------------------------------


def normalize_to_sum(seq: np.ndarray, target_sum: float) -> np.ndarray:
    """Scale a non‑negative vector so that its L1‑sum equals target_sum."""
    s = seq.sum()
    if s == 0:
        # avoid division by zero – return a uniform tiny vector
        return np.full_like(seq, target_sum / len(seq))
    return seq * (target_sum / s)


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex (sum=1, v>=0)."""
    # Implements the algorithm of Wang & Carreira‑Perpiñán (2013)
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # numerical safety – return uniform distribution
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1] - 1
    theta = cssv[cond][-1] / (rho + 1)
    w = np.maximum(v - theta, 0)
    return w


def exponential_family(n: int, c: float) -> np.ndarray:
    """
    a_i ∝ exp(-c*i/n)  (i = 0 … n‑1).
    The parameter c controls the decay; c=0 gives the uniform sequence.
    """
    i = np.arange(n, dtype=float)
    arr = np.exp(-c * i / n)
    return arr


def power_law_family(n: int, p: float) -> np.ndarray:
    """
    a_i ∝ (i+1)^(-p)  (i = 0 … n‑1).
    p=0 gives the uniform sequence; larger p puts more mass at the left.
    """
    i = np.arange(1, n + 1, dtype=float)
    arr = i ** (-p)
    return arr


def random_dirichlet_sequence(n: int, rng: np.random.Generator) -> np.ndarray:
    """
    Draw a random point on the simplex using Dirichlet(1,…,1) (i.e. uniform
    over the simplex) and scale to the target sum later.
    """
    vec = rng.dirichlet(np.ones(n))
    return vec


def hill_climb(
    start_seq: np.ndarray,
    target_sum: float,
    deadline: float,
    rng: np.random.Generator,
) -> np.ndarray:
    """
    Simple simulated‑annealing / hill‑climbing optimizer.
    Moves consist of either a tiny transfer between two positions or a full
    redistribution of a small random subset.  The total mass is kept constant.
    """
    n = len(start_seq)
    cur = start_seq.copy()
    cur_score = evaluate_sequence(cur.tolist())
    best = cur.copy()
    best_score = cur_score

    # annealing schedule parameters
    start_temp = 0.04
    end_temp = 1e-5

    it = 0
    while time.time() < deadline:
        # linear temperature decay
        t_frac = (deadline - time.time()) / (deadline - (deadline - (deadline - time.time())))
        temperature = start_temp * (1.0 - t_frac) + end_temp * t_frac

        # ------- decide which move to perform -------
        if rng.random() < 0.6:
            # ----- tiny mass transfer between two coordinates -----
            i, j = rng.choice(n, size=2, replace=False)
            if cur[i] == 0:
                continue
            # move at most 5 % of the donor amount
            delta = rng.random() * 0.05 * cur[i]
            new = cur.copy()
            new[i] -= delta
            new[j] += delta
        else:
            # ----- redistribute a random subset (size 8‑12) -----
            subset_size = int(rng.integers(6, 13))
            idx = rng.choice(n, size=subset_size, replace=False)
            total = cur[idx].sum()
            # draw a fresh Dirichlet vector and keep the total mass
            new_weights = rng.dirichlet(np.ones(subset_size)) * total
            new = cur.copy()
            new[idx] = new_weights

        # keep non‑negativity explicitly (tiny numerical issues)
        new = np.maximum(new, 0.0)

        # normalise to keep the total sum exactly target_sum
        # (instead of rescaling the whole vector we only need a tiny correction)
        diff = new.sum() - target_sum
        if abs(diff) > 1e-12:
            # distribute the small discrepancy uniformly (preserves non‑negativity
            # because diff is tiny)
            new -= diff / n

        # evaluate
        try:
            new_score = evaluate_sequence(new.tolist())
        except Exception:
            # any numerical mishap – reject this move
            continue

        # acceptance test
        improve = new_score < cur_score
        if improve or rng.random() < exp(-(new_score - cur_score) / (temperature + 1e-12)):
            cur = new
            cur_score = new_score
            if new_score < best_score:
                best = new.copy()
                best_score = new_score
                print(
                    f"[{time.time():.1f}] New best score: {best_score:.6f} (iter {it})"
                )
        it += 1

    return best
