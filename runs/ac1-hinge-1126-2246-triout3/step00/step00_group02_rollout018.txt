def proj_simplex(v):
    """Project vector v onto the probability simplex (non-negative entries sum to 1)."""
    # Standard algorithm: sort v descending, compute cumulative sum, find rho, compute theta = (cumsum - 1)/rho, output max(v - theta,0).
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    rho = np.nonzero(u - cssv / np.arange(1, n+1) > 0)[0][-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0)
    return w

import numpy as np, time

def proj_simplex(v):
    """Project onto probability simplex."""
    n = v.shape[0]
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    rho = np.nonzero(u - cssv / np.arange(1, n+1) > 0)[0][-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0)
    return w

def gradient_step(a, step_factor):
    """Perform one gradient descent step on sum of squares of convolution."""
    conv = np.convolve(a, a)
    grad = 4.0 * np.correlate(conv, a, mode='valid')
    # step size scaled to maximum absolute gradient component
    max_abs = np.max(np.abs(grad)) + 1e-12
    step = step_factor / max_abs
    a_new = proj_simplex(a - step * grad)
    return a_new, conv

"""
search for a non‑negative step function that minimises
evaluate_sequence(sequence).

The algorithm performs many random restarts.  Each restart runs a
gradient‑descent style optimisation that reduces the L2‑norm of the
self‑convolution (∑(a*a)^2).  Flattening the convolution also lowers its
maximum, which directly improves the evaluation metric.
After each gradient step the vector is projected onto the probability
simplex (non‑negative, sum = 1) – this respects the scale‑invariance of
the objective.  The best sequence seen within the allotted time is
returned.
"""

import time
import numpy as np


def _project_to_simplex(v):
    """Project a vector onto the probability simplex (non‑negative, sum = 1)."""
    n = v.shape[0]
    if n == 0:
        return v
    # sort v in descending order
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho = np.nonzero(u - cssv / np.arange(1, n + 1) > 0)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def _gradient_step(a, conv, step_factor):
    """
    One gradient‑descent step on J(a)=∑_k (a*a)[k]².
    Returns the updated a (projected onto the simplex).
    """
    # grad J = 4 * correlate(conv, a)  (valid mode gives length n)
    grad = 4.0 * np.correlate(conv, a, mode='valid')
    max_abs = np.max(np.abs(grad)) + 1e-12
    step = step_factor / max_abs
    a_new = _project_to_simplex(a - step * grad)
    return a_new

    while time.time() < deadline:

        n = np.random.randint(120, 350)   # length of this restart

        a = np.random.rand(n)
