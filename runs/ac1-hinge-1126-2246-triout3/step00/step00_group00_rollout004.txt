def solve_lp_convolution(f_seq, M):
    n = len(f_seq)
    c = -np.ones(n)  # maximize sum
    # Build A_ub and b_ub for conv constraints.
    # For each k in 0..2n-2:
    A = np.zeros((2*n-1, n))
    for k in range(2*n-1):
        # i index from max(0, k-(n-1)) to min(n-1, k)
        # For each i, j = k-i
        # coefficient for b_j is f_i
        # Equivalent: row[k][j] = f_i for each i where j = k-i.
        # So we can fill by iterating i.
        # Loop i: for i in range(n):
        #   j = k - i
        #   if 0 <= j < n: A[k][j] = f_seq[i]
        # Do this.
        for i in range(n):
            j = k - i
            if 0 <= j < n:
                A[k, j] = f_seq[i]
    # To also enforce b >= 0, we provide bounds.
    bounds = [(0, None) for _ in range(n)]
    # Use hiGHS linear programming with max_iter maybe limited.
    result = linprog(c, A_ub=A, b_ub=np.full(2*n-1, M), bounds=bounds, method='highs')
    if result.success:
        return result.x
    else:
        return None

```python
"""Propose a candidate sequence for minimize evaluate_sequence.

Algorithm:
- Use a mixture of linear programming (LP) steps and simulated annealing.
- LP step: given current sequence f (sum ≠0) solve max sum(g) s.t. conv(f,g) ≤ M where M= max(conv(f,f)).
  The solution g tends to spread mass while respecting current max convolution bound.
- Then find best convex combination α∈[0,1] between f and normed g that gives lowest score.
- SA step: small random zero‑mean perturbations of the current sequence; accept
  improvements or occasional worse moves with a temperature schedule.
- The outer loop alternates these steps while respecting the overall time budget.
- Several random initial sequences (including the previously known best
  `height_sequence_1`) are tried; the best sequence found is returned.

The implementation works with moderate sequence lengths (≤800) for the LP
to keep memory usage reasonable; for larger lengths the algorithm relies
solely on the simulated‑annealing perturbations.
"""

import time
import numpy as np
from scipy.optimize import linprog

def solve_convolution_lp(f_seq, bound):
    """Return a non‑negative vector g that maximises sum(g) subject to
    (f_seq * g)[k] ≤ bound for all k.  Returns None on failure.

    The constraints are linear because each entry of the convolution is a
    linear function of the unknown vector g.
    """
    n = len(f_seq)
    # Objective: maximize sum(g) -> minimise -sum(g)
    c = -np.ones(n)
    # Build convolution constraints A_ub @ g ≤ bound
    rows = []
    for k in range(2 * n - 1):
        row = np.zeros(n)
        # indices i where f[i] contributes to convolution at k
        # j = k - i must be in [0, n)
        i_min = max(0, k - (n - 1))
        i_max = min(n - 1, k)
        # For i in [i_min, i_max], j = k - i
        # The coefficient of g[j] is f[i]
        # equivalent to placing f[i] at column j
        # Fill the row
        for i in range(i_min, i_max + 1):
            j = k - i
            row[j] = f_seq[i]
        rows.append(row)
    A_ub = np.vstack(rows)
    b_ub = np.full(2 * n - 1, bound, dtype=float)

    # Non‑negativity bounds are provided via bounds argument
    bounds = [(0, None) for _ in range(n)]
    try:
        result = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds,
                         method='highs', options={'presolve': True})
    except Exception:
        return None
    if result.success:
        # Clip tiny negatives that can appear due to numerical issues
        g = np.maximum(result.x, 0.0)
        return g.tolist()
    return None

def best_mixture_score(cur_seq, g_seq, sum_sq):
    """Given two sequences of the same length, return (best_alpha, best_score)
    where best_alpha∈[0,1] minimises the evaluate_sequence of the convex
    combination (1-α)·cur_seq + α·g_seq.  sum_sq is (∑ cur_seq)^2, which
    stays constant because the mixture keeps the total sum unchanged.
    """
    n = len(cur_seq)
    cur_arr = np.asarray(cur_seq)
    g_arr = np.asarray(g_seq)
    conv_f = np.convolve(cur_arr, cur_arr)
    conv_g = np.convolve(g_arr, g_arr)
    conv_fg = np.convolve(cur_arr, g_arr)
    # Search coarse grid then refine
    best_alpha = 0.0
    best_val = 2 * n * np.max(conv_f) / sum_sq  # current score
    # initial coarse search
    alphas = np.linspace(0.0, 1.0, 31)
    for a in alphas:
        mixed = (1 - a) ** 2 * conv_f + 2 * a * (1 - a) * conv_fg + a ** 2 * conv_g
        max_b = np.max(mixed)
        val = 2 * n * max_b / sum_sq
        if val < best_val:
            best_val = val
            best_alpha = a
    # refine around best_alpha if not at edges
    if 0.0 < best_alpha < 1.0:
        a0 = max(0.0, best_alpha - 0.05)
        a1 = min(1.0, best_alpha + 0.05)
        alphas = np.linspace(a0, a1, 31)
        for a in alphas:
            mixed = (1 - a) ** 2 * conv_f + 2 * a * (1 - a) * conv_fg + a ** 2 * conv_g
            max_b = np.max(mixed)
            val = 2 * n * max_b / sum_sq
            if val < best_val:
                best_val = val
                best_alpha = a
    # also try a full move to g if it improves
    if best_alpha == 0.0:
        # maybe direct g is better
        val_g = 2 * n * np.max(conv_g) / sum_sq
        if val_g < best_val:
            best_val = val_g
            best_alpha = 1.0
    return best_alpha, best_val

def random_seq(length, rng):
    """Generate a random non‑negative sequence of a given length.
    The sequence is scaled so that its ℓ₁‑norm equals √(2 · length)."""
    seq = rng.random(length).astype(float)
    # avoid all zeros
    if seq.sum() == 0:
        seq[0] = 1.0
    scale = np.sqrt(2 * length) / seq.sum()
    seq *= scale
    return seq.tolist()

def simulated_annealing(seq, deadline):
    """Perform a simple annealing on a fixed‑size sequence until deadline.
    Returns the best sequence found and its score."""
    cur = np.array(seq, dtype=float)
    cur_score = evaluate_sequence(cur.tolist())
    best = cur.copy()
    best_score = cur_score
    n = len(cur)
    # temperature schedule: exponential decay from 0.5 to 1e-4
    T0 = 0.5
    Tf = 1e-4
    start = time.time()
    it = 0
    while time.time() < deadline:
        it += 1
        # Linear progress fraction
        frac = (time.time() - start) / (deadline - start + 1e-12)
        T = T0 * (Tf / T0) ** frac
        # Propose a zero‑mean perturbation
        delta = np.random.randn(n) * T
        cand = cur + delta
        cand = np.maximum(cand, 0.0)
        # If the sum became extremely small, bring it back to a safe scale
        if cand.sum() < 0.01:
            cand = np.maximum(cand, 1e-6)
            cand *= np.sqrt(2 * n) / cand.sum()
        cand_score = evaluate_sequence(cand.tolist())
        # Acceptance rule
        if cand_score < cur_score or np.random.rand() < np.exp(-(cand_score - cur_score) / max(T, 1e-12)):
            cur = cand
            cur_score = cand_score
            if cand_score < best_score:
                best = cand.copy()
                best_score = cand_score
                print(f" SA iter {it}: new best {best_score:.6f}")
    return best.tolist(), best_score

def propose_candidate(seed=0, budget_s=1000, **kwargs):
    """

    # -----------------------------------------------------------------
    candidates = []

        pass

    best_score = float('inf')

            if sc < best_score:

        except Exception:
            continue

    # -----------------------------------------------------------------
    # 2. Main optimisation loop
    # -----------------------------------------------------------------
    cur_seq = best_seq.copy()
    cur_score = best_score
    iteration = 0

    while time.time() < deadline:
        iteration += 1
        n = len(cur_seq)
        # -------------------------------------------------------------
        # 2a. LP direction – only for lengths that keep the constraint
        #      matrix reasonably small (≤800)
        # -------------------------------------------------------------
        if n <= 800:

            g = solve_convolution_lp(cur_seq, M)

                sum_cur = np.sum(cur_seq)

                    # Find best convex combination (alpha) between cur_seq and g

                        # Build the mixed sequence

                        cur_score = mixed_score
                        if cur_score < best_score:
                            best_score, best_seq = cur_score, cur_seq.copy()
                            print(f"Iter {iteration}: LP‑mix improved to {best_score:.6f}")
                        continue  # go to next iteration
        # -------------------------------------------------------------

            if cur_score < best_score:
                best_score, best_seq = cur_score, cur_seq.copy()
                print(f"Iter {iteration}: SA improved to {best_score:.6f}")

        # -------------------------------------------------------------
        # 2c. Occasional random restart to escape stagnation

        if np.random.rand() < 0.02:

            cur_seq = random_seq(L, rng)
            cur_score = evaluate_sequence(cur_seq)
            print(f"Iter {iteration}: random restart len={L} score={cur_score:.6f}")

        # -------------------------------------------------------------
        # 2d. Prune very small entries to keep the vector dense

        arr = np.array(cur_seq)

            else:
                arr *= np.sqrt(2 * len(arr)) / arr.sum()
            cur_seq = arr.tolist()
            cur_score = evaluate_sequence(cur_seq)
            if cur_score < best_score:
                best_score, best_seq = cur_score, cur_seq.copy()
                print(f"Iter {iteration}: pruning improved to {best_score:.6f}")

    # -----------------------------------------------------------------
    # 3. Fallback if no sequence was produced

    if best_seq is None or len(best_seq) == 0:

"""
Propose a candidate sequence for the step‑function inequality.

The optimizer mixes two ideas:

1. **LP direction** – exactly the move used in the reference code:
   given the current sequence `f` we solve a linear programme that
   maximises the ℓ₁‑norm of a new sequence `g` under the constraint
   `conv(f,g) ≤ M` where `M = max(conv(f,f))`.  After normalising `g`
   to the same total sum as `f`, we search the optimal convex combination
   `h = (1‑α)·f + α·g` (α∈[0,1]) that yields the smallest score.
   The convolution of the mixture can be evaluated analytically from the
   three pre‑computed convolutions `conv(f,f)`, `conv(g,g)` and `conv(f,g)`,
   which makes the line‑search cheap.

2. **Simulated‑annealing refinement** – a simple zero‑mean random
   perturbation of the current vector.  The perturbation magnitude follows
   an exponential temperature schedule; moves that improve the score are
   always accepted, while worsening moves are accepted with probability
   `exp(-(Δscore)/T)`.  This helps to escape local minima and to explore
   the space when the LP becomes too expensive (large `n`).

The outer loop alternates the LP step (only for `n ≤ 800` to keep the
constraint matrix modest) and a short SA burst, occasionally performing a
random restart.  Several diverse initial sequences are tried, including
the previously known best construction (`height_sequence_1`).  The best
sequence discovered before the time budget expires is returned.

The algorithm works entirely with NumPy and SciPy and respects the
1000‑second budget.
"""

import time
import numpy as np
from scipy.optimize import linprog

# ----------------------------------------------------------------------
# Helper: linear programme that maximises sum(g) under a convolution bound
# ----------------------------------------------------------------------
def solve_convolution_lp(f_seq, bound):
    """Return a non‑negative vector g that maximises sum(g) subject to
    (f_seq * g)[k] ≤ bound for all k.  Returns None on failure.

    The constraints are linear because each entry of the convolution is a
    linear function of the unknown vector g.
    """
    n = len(f_seq)
    # Objective: maximise sum(g)  → minimise -sum(g)
    c = -np.ones(n)

    rows = []
    for k in range(2 * n - 1):
        row = np.zeros(n)
        i_min = max(0, k - (n - 1))
        i_max = min(n - 1, k)
        for i in range(i_min, i_max + 1):
            j = k - i
            row[j] = f_seq[i]
        rows.append(row)

    A_ub = np.vstack(rows)
    b_ub = np.full(2 * n - 1, bound, dtype=float)
    bounds = [(0, None) for _ in range(n)]

    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds,
                      method='highs', options={'presolve': True})
    except Exception:
        return None

    if res.success:
        g = np.maximum(res.x, 0.0)          # clip tiny negatives
        return g.tolist()
    return None


# ----------------------------------------------------------------------
# Helper: best convex combination of two sequences
# ----------------------------------------------------------------------
def best_mixture_score(cur_seq, g_seq, sum_sq):
    """Given two sequences of equal length, find α∈[0,1] that minimises
    the score of (1‑α)·cur_seq + α·g_seq.  Returns (α, score)."""
    n = len(cur_seq)
    cur = np.asarray(cur_seq)
    g   = np.asarray(g_seq)

    conv_f  = np.convolve(cur, cur)
    conv_g  = np.convolve(g, g)
    conv_fg = np.convolve(cur, g)

    # current score (α = 0)
    best_alpha = 0.0
    best_val   = 2 * n * np.max(conv_f) / sum_sq

    # coarse grid
    for a in np.linspace(0.0, 1.0, 31):
        mixed = (1 - a) ** 2 * conv_f + 2 * a * (1 - a) * conv_fg + a ** 2 * conv_g
        val = 2 * n * np.max(mixed) / sum_sq
        if val < best_val:
            best_val, best_alpha = val, a

    # refinement around the best point (if interior)
    if 0.0 < best_alpha < 1.0:
        lo, hi = max(0.0, best_alpha - 0.05), min(1.0, best_alpha + 0.05)
        for a in np.linspace(lo, hi, 31):
            mixed = (1 - a) ** 2 * conv_f + 2 * a * (1 - a) * conv_fg + a ** 2 * conv_g
            val = 2 * n * np.max(mixed) / sum_sq
            if val < best_val:
                best_val, best_alpha = val, a

    # try the pure g move
    if best_alpha == 0.0:
        val_g = 2 * n * np.max(conv_g) / sum_sq
        if val_g < best_val:
            best_val, best_alpha = val_g, 1.0

    return best_alpha, best_val

# Helper: generate a random start sequence
# ----------------------------------------------------------------------
def random_seq(length, rng):
    """Random non‑negative vector with ℓ₁‑norm = √(2·length)."""
    seq = rng.random(length).astype(float)
    if seq.sum() == 0:
        seq[0] = 1.0
    scale = np.sqrt(2 * length) / seq.sum()
    seq *= scale
    return seq.tolist()

# Helper: very light simulated annealing on a fixed sequence
# ----------------------------------------------------------------------
def simulated_annealing(seq, deadline):
    """Anneal `seq` until `deadline`.  Returns (best_seq, best_score)."""
    cur = np.array(seq, dtype=float)
    cur_score = evaluate_sequence(cur.tolist())
    best = cur.copy()
    best_score = cur_score
    n = len(cur)

    T0, Tf = 0.5, 1e-4   # temperature schedule
    start = time.time()
    it = 0
    while time.time() < deadline:
        it += 1
        frac = (time.time() - start) / (deadline - start + 1e-12)
        T = T0 * (Tf / T0) ** frac

        delta = np.random.randn(n) * T
        cand = cur + delta
        cand = np.maximum(cand, 0.0)

        # avoid degenerate zero‑sum vectors
        if cand.sum() < 0.01:
            cand = np.maximum(cand, 1e-6)
            cand *= np.sqrt(2 * n) / cand.sum()

        cand_score = evaluate_sequence(cand.tolist())

        # Metropolis acceptance
        if cand_score < cur_score or np.random.rand() < np.exp(-(cand_score - cur_score) / max(T, 1e-12)):
            cur, cur_score = cand, cand_score
            if cand_score < best_score:
                best, best_score = cand.copy(), cand_score
                print(f" SA iter {it}: new best {best_score:.6f}")
    return best.tolist(), best_score
