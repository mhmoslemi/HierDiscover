"""
Propose a candidate step‑function heights sequence.

The algorithm is a simple simulated‑annealing / coordinate‑descent
procedure that repeatedly moves a small amount of mass from indices that
contribute most to the current maximal convolution value to indices that
contribute least.  This directly targets the quantity

        2 * n * max(conv(a,a)) / (sum(a)**2)

which the evaluator minimises.
The method works with any starting sequence (the global `height_sequence_1`
is used if present) and respects the 1000‑second budget.
"""

import time
import math
import random
import numpy as np

# ----------------------------------------------------------------------
# Helper functions (all defined at top level – no closures or lambdas)
# ----------------------------------------------------------------------


def _initial_sequence():
    """
    Produce an initial non‑negative sequence.
    If a global `height_sequence_1` exists it is used (normalised),
    otherwise a uniform sequence of the appropriate length is created.
    The sum is normalised to sqrt(2*n) – the same normalisation used in
    the reference implementation.
    """
    if "height_sequence_1" in globals():
        base = np.array(height_sequence_1, dtype=float)
        base = np.maximum(base, 0.0)          # safety, remove possible negatives
        n = int(len(base))
    else:
        n = 5000                               # a reasonable default if nothing is given
        base = np.ones(n, dtype=float)

    # Normalise the sum to sqrt(2*n) (scale does not affect the objective
    # unless values hit the clipping limits, which we avoid).
    target_sum = math.sqrt(2.0 * n)
    if base.sum() == 0:
        base[:] = 1.0
    base = base / base.sum() * target_sum
    return base


def _contribution_weights(seq: np.ndarray, k_max: int) -> np.ndarray:
    """
    For the current maximal convolution index k_max compute a weight
    for each position i that quantifies how strongly a_i participates
    in that maximal term.
    The weight is a_i + a_{k_max-i} (when the partner index exists);
    otherwise it is just a_i.
    """
    n = seq.shape[0]
    idx = np.arange(n)
    partner = k_max - idx
    valid = (partner >= 0) & (partner < n)

    w = np.empty(n, dtype=seq.dtype)
    # both i and its partner are inside the sequence
    w[valid] = seq[valid] + seq[partner[valid]]
    # partner is out of range → only the single term remains
    w[~valid] = seq[~valid]
    return w


def _move_mass(seq: np.ndarray,
               src: int,
               dst: int,
               delta: float,
               target_sum: float) -> np.ndarray:
    """
    Return a copy of `seq` where `delta` mass is taken from `src`
    and added to `dst`.  The whole vector is re‑normalised so that its
    L1‑sum equals `target_sum`.
    """
    new_seq = seq.copy()
    new_seq[src] -= delta
    new_seq[dst] += delta
    # Re‑normalise – this keeps the objective scale‑invariant
    new_seq *= target_sum / new_seq.sum()
    return new_seq


# ----------------------------------------------------------------------
# Main entry point required by the competition harness
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: float = 1000.0, **kwargs):
    """
    Search for a sequence of non‑negative heights that minimises
    `evaluate_sequence`.  The routine runs for at most ``budget_s`` seconds
    (a safety margin of a few seconds is kept) and returns the best
    sequence seen so far.

    Parameters
    ----------
    seed : int
        Random seed for reproducibility.
    budget_s : float
        Overall time budget in seconds (the harness will stop the process
        after this time).

    Returns
    -------
    list[float]
        The best sequence found.
    """
    # ------------------------------------------------------------------
    # Setup
    # ------------------------------------------------------------------
    random.seed(seed)
    np.random.seed(seed)

    seq = _initial_sequence()
    n = seq.shape[0]
    target_sum = math.sqrt(2.0 * n)

    # Current and best scores
    curr_score = evaluate_sequence(seq.tolist())
    best_seq = seq.copy()
    best_score = curr_score

    print(f"[init] n={n}, initial score={curr_score:.6f}")

    # Annealing / step‑size schedule parameters
    step_factor = 0.5          # maximum fraction of source entry that can be moved
    temperature = 0.05         # controls acceptance of uphill moves
    min_step = 1e-5            # never go below this absolute step size
    max_no_improve = 2000      # after this many non‑improving steps we restart

    no_improve_counter = 0
    iteration = 0

    deadline = time.time() + budget_s - 5.0   # leave a small safety margin

    # ------------------------------------------------------------------
    # Main optimisation loop
    # ------------------------------------------------------------------
    while time.time() < deadline:
        # --------------------------------------------------------------
        # 1) Identify the convolution index that attains the current maximum
        # --------------------------------------------------------------
        conv = np.convolve(seq, seq)                # O(n log n) via FFT internally
        k_max = int(np.argmax(conv))

        # --------------------------------------------------------------
        # 2) Compute contribution weights for each position
        # --------------------------------------------------------------
        w = _contribution_weights(seq, k_max)

        # --------------------------------------------------------------
        # 3) Pick a source (high weight) and a destination (low weight)
        # --------------------------------------------------------------
        # Take the top/bottom 5 % of indices as candidate pools
        pool_size = max(1, n // 20)

        sorted_idx = np.argsort(w)                 # ascending
        low_pool = sorted_idx[:pool_size]          # least contributors
        high_pool = sorted_idx[-pool_size:]        # strongest contributors

        src = int(np.random.choice(high_pool))
        dst = int(np.random.choice(low_pool))

        # --------------------------------------------------------------
        # 4) Propose a transfer of mass from src → dst
        # --------------------------------------------------------------
        max_transfer = seq[src] * step_factor
        if max_transfer <= min_step:
            # If the source is essentially zero, skip this iteration
            continue
        delta = max_transfer * np.random.rand()    # random amount ≤ max_transfer

        new_seq = _move_mass(seq, src, dst, delta, target_sum)

        # --------------------------------------------------------------
        # 5) Evaluate the new candidate
        # --------------------------------------------------------------
        try:
            new_score = evaluate_sequence(new_seq.tolist())
        except Exception:
            # Should never happen – just be defensive
            continue

        # --------------------------------------------------------------
        # 6) Acceptance rule (simulated annealing)
        # --------------------------------------------------------------
        accept = False
        if new_score < curr_score:
            accept = True
        else:
            # uphill move accepted with probability exp(-(Δ)/T)
            prob = math.exp(-(new_score - curr_score) / temperature)
            if np.random.rand() < prob:
                accept = True

        if accept:
            seq = new_seq
            curr_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_seq = new_seq.copy()
                no_improve_counter = 0
                print(f"[iter {iteration:06d}] new best: {best_score:.6f}")
            else:
                no_improve_counter += 1
        else:
            no_improve_counter += 1

        # --------------------------------------------------------------
        # 7) Schedule updates
        # --------------------------------------------------------------
        iteration += 1
        # Slowly reduce step size and temperature – typical annealing schedule
        if iteration % 500 == 0:
            step_factor = max(step_factor * 0.9, 0.01)
            temperature = max(temperature * 0.95, 1e-6)

        # --------------------------------------------------------------
        # 8) Random restart if we have been stuck for a while
        # --------------------------------------------------------------
        if no_improve_counter >= max_no_improve:
            # Re‑initialise with a fresh random vector
            seq = np.random.rand(n)
            seq *= target_sum / seq.sum()
            curr_score = evaluate_sequence(seq.tolist())
            no_improve_counter = 0
            print(f"[restart] iteration {iteration}, score reset to {curr_score:.6f}")

    # ------------------------------------------------------------------
    # Return the best sequence discovered
    # ------------------------------------------------------------------
    print(f"[finished] best score = {best_score:.6f} after {iteration} iterations")
    return best_seq.tolist()
