def compute_score(seq):
    a = np.array(seq, dtype=np.float64)
    sum_a = np.sum(a)
    if sum_a < 0.01:
        return np.inf
    # Compute convolution via FFT
    n = len(a)
    size = 1 << ((2*n-1).bit_length())
    A = np.fft.rfft(a, size)
    b = np.fft.irfft(A*A, size)[:2*n-1]
    max_b = np.max(b)
    return 2 * n * max_b / (sum_a**2)

"""
Propose a candidate sequence that minimizes the evaluation function.

The algorithm works in three stages:

1.  **Parameterized initialisation** – several families of monotone
    sequences are generated (pure exponential, stretched exponential,
    and a simple power‑law).  The length of the sequence is chosen to be
    several thousand (fast enough for FFT‑based convolutions).

2.  **Targeted hill‑climbing** – for a given starting sequence the
    convolution is computed with an FFT.  The shift ``k_max`` at which the
    convolution attains its maximum is located.  A move is then proposed
    that transfers a small amount of mass from the pair of indices
    contributing most to this peak to two random locations.  Occasionally
    a plain random swap or a small mass transfer between two random
    positions is tried.  Moves that improve (or very slightly worsen) the
    objective are accepted (simulated‑annealing style) and the best
    sequence seen so far is remembered.

3.  **Multi‑restart search** – the hill‑climber is run for a short
    budget on each of the initial families.  The overall best sequence
    is returned.

All heavy work (convolution and score computation) uses an FFT based
implementation which runs in O(N log N).  The code respects the
time‑budget (default 1000 s) and never writes to the file system.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Helper functions (all top‑level, no closures)
# ----------------------------------------------------------------------


def _fft_convolve(a: np.ndarray) -> np.ndarray:
    """Return a * a (full convolution) using real FFTs."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())          # next power of two
    A = np.fft.rfft(a, size)
    conv = np.fft.irfft(A * A, size)[: 2 * n - 1]
    return conv


def _score(seq: np.ndarray) -> float:
    """Exact score used by ``evaluate_sequence`` (without its clipping)."""
    n = seq.shape[0]
    s = np.sum(seq)
    if s < 0.01:
        return np.inf
    conv = _fft_convolve(seq)
    max_b = np.max(conv)
    return float(2.0 * n * max_b / (s * s))


def _make_exponential(n: int, c: float) -> np.ndarray:
    """Geometric decay a_i = r**i, r = 1 - c/n (c>0)."""
    eps = c / n
    r = 1.0 - eps
    return r ** np.arange(n)


def _make_stretched_exponential(n: int, gamma: float, p: float) -> np.ndarray:
    """
    a_i = exp( -gamma * (i / n)**p ),  i=0..n-1.
    ``gamma`` controls overall decay, ``p`` the stretch (p=1 yields simple exp).
    """
    x = np.arange(n, dtype=np.float64) / n
    return np.exp(-gamma * (x ** p))


def _make_power_law(n: int, alpha: float) -> np.ndarray:
    """a_i = (i+1)**(-alpha)  (monotonically decreasing)."""
    return (np.arange(1, n + 1, dtype=np.float64) ** (-alpha))


def _random_swap_move(seq: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """Swap two random positions."""
    i, j = rng.integers(0, seq.size, size=2)
    if i == j:
        return None
    new = seq.copy()
    new[i], new[j] = new[j], new[i]
    return new


def _random_transfer_move(seq: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """Transfer a small fraction from one entry to another."""
    i, j = rng.integers(0, seq.size, size=2)
    if i == j or seq[i] <= 0.0:
        return None
    # up‑to 5 % of the source is moved
    delta = rng.random() * 0.05 * seq[i]
    new = seq.copy()
    new[i] -= delta
    new[j] += delta
    return new


def _targeted_move(seq: np.ndarray, conv: np.ndarray,
                   rng: np.random.Generator) -> np.ndarray:
    """
    Reduce the current peak of the convolution:
    * pick a pair (i, j) with i+j = k_max (k_max is where conv is maximal)
    * move a small amount of mass away from both i and j to two random locations.
    """
    n = seq.shape[0]
    k_max = int(np.argmax(conv))
    lo = max(0, k_max - (n - 1))
    hi = min(k_max, n - 1)

    # contribution of each admissible i to the peak
    idx = np.arange(lo, hi + 1, dtype=int)
    jdx = k_max - idx
    contrib = seq[idx] * seq[jdx]

    # choose i with probability proportional to its contribution
    total = np.sum(contrib)
    if total == 0.0:
        return None
    probs = contrib / total
    i = rng.choice(idx, p=probs)
    j = k_max - i

    # amount to move (up‑to 10 % of the smaller of the two)
    max_move = min(seq[i], seq[j])
    if max_move <= 0.0:
        return None
    delta = rng.random() * 0.10 * max_move
    if delta <= 0.0:
        return None

    # two random distinct destinations (different from i, j)
    dest = rng.choice(np.setdiff1d(np.arange(n), [i, j]), size=2, replace=False)
    l, m = dest[0], dest[1]

    new = seq.copy()
    new[i] -= delta
    new[j] -= delta
    new[l] += delta
    new[m] += delta
    return new


def _hill_climb(start_seq: np.ndarray,
                budget_s: float,
                rng: np.random.Generator) -> (np.ndarray, float):
    """
    Simple simulated‑annealing hill climbing.
    Returns the best sequence found within the supplied time budget.
    """
    deadline = time.time() + budget_s
    cur = start_seq.copy()
    cur_score = _score(cur)
    best = cur.copy()
    best_score = cur_score

    # simple exponential temperature schedule
    T0 = 0.1
    Tmin = 1e-5
    decay = 0.9995

    T = T0
    it = 0
    while time.time() < deadline:
        # decide which kind of move to propose
        move_type = rng.random()
        if move_type < 0.35:
            cand = _random_swap_move(cur, rng)
        elif move_type < 0.70:
            cand = _random_transfer_move(cur, rng)
        else:
            # need current convolution for the targeted move
            conv = _fft_convolve(cur)
            cand = _targeted_move(cur, conv, rng)

        if cand is None:
            continue

        cand_score = _score(cand)

        # accept if better, otherwise with a small temperature probability
        accept = False
        if cand_score < cur_score:
            accept = True
        else:
            # Metropolis criterion
            if T > Tmin and rng.random() < math.exp((cur_score - cand_score) / T):
                accept = True

        if accept:
            cur = cand
            cur_score = cand_score
            if cur_score < best_score:
                best, best_score = cur.copy(), cur_score
                print(f"    New local best: {best_score:.6f}")

        # temperature decay
        T = max(Tmin, T * decay)
        it += 1
        if it % 500 == 0:
            # occasional progress report
            print(f"Iter {it:6d} – best={best_score:.6f} – T={T:.2e}")

    return best, best_score

# Main entry point required by the evaluation harness
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a sequence of non‑negative heights that yields a low
    ``evaluate_sequence`` value.  The function runs for at most
    ``budget_s`` seconds and returns the best sequence found.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 5.0   # small safety margin

    # --- choose a working length -------------------------------------------------
    # Use the length of the provided construction if it exists, otherwise a
    # moderate size that keeps the FFT cheap.
    if 'height_sequence_1' in globals():
        n = len(height_sequence_1)
    else:
        n = 4000

    print(f"Target length = {n}")

    # --- generate a few diversified starting points -------------------------------
    starters = []

    # 1) pure exponential with several c‑values
    for c in (0.9, 1.2, 1.5, 2.0):
        starters.append(_make_exponential(n, c))

    # 2) stretched exponentials (different powers)
    for gamma in (2.0, 4.0, 6.0):
        for p in (1.2, 1.5, 2.0):
            starters.append(_make_stretched_exponential(n, gamma, p))

    # 3) simple power‑law (alpha>1 gives finite sum)
    for alpha in (0.5, 0.8, 1.0):
        starters.append(_make_power_law(n, alpha))

    # 4) random perturbation of the SOTA sequence if it exists
    if 'height_sequence_1' in globals():
        base = np.array(height_sequence_1, dtype=np.float64)
        if base.shape[0] != n:
            # truncate / pad to required length
            if base.shape[0] > n:
                base = base[:n]
            else:
                base = np.pad(base, (0, n - base.shape[0]), constant_values=0.0)
        starters.append(base + rng.normal(scale=0.02, size=n).clip(min=0.0))

    # --- allocate time for each restart -----------------------------------------
    per_restart = max(2.0, (deadline - time.time()) / len(starters))

    best_overall_seq = None
    best_overall_score = float('inf')

    for idx, start in enumerate(starters):
        time_left = deadline - time.time()
        if time_left < 5.0:
            break
        budget = min(per_restart, time_left - 1.0)   # keep a buffer
        print(f"\n=== Restart {idx+1}/{len(starters)} – budget {budget:.1f}s ===")

        # Ensure non‑negative and clip to the allowed range
        seq = np.clip(start, 0.0, 1000.0)

        # Normalise to a reasonable scale (optional – scaling does not affect the score)
        seq_sum = np.sum(seq)
        if seq_sum < 0.01:
            seq += 0.01 / n

        # Run the local optimiser
        opt_seq, opt_score = _hill_climb(seq, budget, rng)

        print(f"  Restart {idx+1} result: score = {opt_score:.6f}")

        if opt_score < best_overall_score:
            best_overall_score = opt_score
            best_overall_seq = opt_seq.copy()
            print(f"*** New global best after restart {idx+1}: {best_overall_score:.6f}")

    # Final safeguard – if nothing was found fall back to a simple exponential
    if best_overall_seq is None:
        best_overall_seq = _make_exponential(n, 1.3)

    # Return as a plain Python list (the external evaluator expects a list)
    print(f"\nFinal best score: {best_overall_score:.6f}")
    return best_overall_seq.tolist()
