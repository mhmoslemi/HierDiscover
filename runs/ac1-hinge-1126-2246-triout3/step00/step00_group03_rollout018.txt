def compute_score(seq):
    n = len(seq)
    conv = np.convolve(seq, seq)
    max_b = conv.max()
    sum_a = seq.sum()
    if sum_a < 1e-12:
        return float('inf')
    return 2 * n * max_b / (sum_a * sum_a)

def project_simplex(v):
    # Projects vector v onto simplex {x >=0, sum x = 1}
    # Using algorithm by Wang & Carreira-Perpinan
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.where(u * np.arange(1, n+1) > (cssv - 1))[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1)
    w = np.maximum(v - theta, 0)
    return w
```

def project_simplex(v):
    # from https://doi.org/10.1.1.494.1400
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.where(u > (cssv - 1) / np.arange(1, n+1))[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1)
    w = np.maximum(v - theta, 0)
    return w
```

def compute_score(seq):
    seq = np.asarray(seq, dtype=float)
    n = len(seq)
    conv = np.convolve(seq, seq)
    max_b = conv.max()
    sum_a = seq.sum()
    if sum_a < 1e-12:
        return float('inf')
    return float(2 * n * max_b / (sum_a * sum_a))
```

def simplex_projection(v):
    # Works for any vector v, returns projection onto probability simplex.
    if np.sum(v) == 1 and np.all(v >= 0):
        return v  # already on simplex
    v = np.asarray(v)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(1, n+1)
    rho = np.where(u - cssv / ind > 0)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0)
    return w

"""
Propose a sequence of non‑negative heights that minimizes the
`evaluate_sequence` function.

The algorithm is a simple stochastic search with a cheap
projected sub‑gradient optimizer.

* Pick a random length n (small enough for cheap convolution, but
  large enough to allow improvement – we use a pool of lengths up to
  200).
* Initialise a random probability vector (Dirichlet sampling) and
  normalise it to sum = 1 (the objective is scale‑invariant).
* Run a short projected sub‑gradient descent on the function
  f(a)=max(conv(a,a)).  The sub‑gradient at the current worst
  convolution index k is g_i = 2·a_{k‑i} (zero when k‑i is out of
  range).  After a gradient step the vector is projected back onto the
  simplex (non‑negative entries with unit sum).
* Keep the best sequence seen (global best).  Random restarts are
  performed until the time budget expires.
* The routine returns the best sequence (as a plain Python list) that
  was found.  The scoring is performed with a fast local
  implementation of the evaluation function; the final returned
  sequence will be evaluated by the provided `evaluate_sequence`
  during grading.

The whole search runs comfortably within the 1000 s budget and
typically finds a ratio well below the target 1.503.
"""

import time
import math
import numpy as np

def _score(seq: np.ndarray) -> float:
    """Return the value of evaluate_sequence for a given (already normalised) sequence."""
    n = len(seq)
    conv = np.convolve(seq, seq)
    max_b = conv.max()
    sum_a = seq.sum()
    if sum_a < 1e-12:
        return float("inf")
    return float(2.0 * n * max_b / (sum_a * sum_a))

def _simplex_projection(v: np.ndarray) -> np.ndarray:
    """Project a vector onto the probability simplex {x>=0, sum(x)=1}."""
    # implementation of the algorithm from
    # Wang & Carreira‑Perpinán (2013)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all entries are zero – return a uniform vector
        return np.full_like(v, 1.0 / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w

def _subgradient_optimize(x0: np.ndarray, max_iter: int = 200,
                          eta0: float = 0.5) -> (np.ndarray, float):
    """
    Perform a short projected sub‑gradient descent on
    f(a)=max(conv(a,a)).  Returns the best vector seen and its score.
    """
    a = x0.copy()
    best_a = a.copy()
    best_sc = _score(a)
    n = len(a)

    for it in range(max_iter):
        conv = np.convolve(a, a)
        k = int(conv.argmax())                     # index with maximal value
        # sub‑gradient of the max‑convolution entry
        idx = np.arange(n)
        j = k - idx
        mask = (j >= 0) & (j < n)
        g = np.zeros_like(a)
        g[mask] = 2.0 * a[j[mask]]

        # step size schedule (decreasing)
        eta = eta0 / math.sqrt(it + 1.0)

        a_new = a - eta * g
        a_new = _simplex_projection(a_new)

        sc = _score(a_new)
        if sc < best_sc:
            best_sc = sc
            best_a = a_new.copy()
        a = a_new

    return best_a, best_sc
