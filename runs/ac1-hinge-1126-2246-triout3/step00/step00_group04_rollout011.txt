def gradient_step(a, eta):
    n = len(a)
    S = sum(a)
    conv = np.convolve(a, a)
    M = conv.max()
    k0 = conv.argmax()
    grad = np.zeros_like(a)
    # compute gradient of M w.r.t a_i: 2 * a_{k0 - i}
    for i in range(n):
        j = k0 - i
        if 0 <= j < n:
            grad[i] = 2 * a[j]
        else:
            grad[i] = 0
    # gradient of F = 2n M / S^2
    grad_F = (2 * n / (S**2)) * grad - (4 * n * M / (S**3)) * np.ones_like(a)
    # update opposite direction to minimize: a_new = a - eta * grad_F
    a_new = a - eta * grad_F
    # clip to non-negative:
    a_new = np.maximum(a_new, 0)
    # if sum becomes zero or too small, renorm:
    sum_new = a_new.sum()
    if sum_new < 1e-8:
        a_new = a # revert
    else:
        # renorm to original sum? can't because objective invariance? scaling doesn't affect ratio: we can keep same sum.
        # Actually we can keep sum as is because ratio invariant under scaling. But to keep sum ~1 for numeric stability, we can renorm to sum=1.
        a_new = a_new / sum_new
    return a_new

def propose_candidate(seed=123, budget_s=1000, **kwargs):
    import numpy as np, time, random, math
    np.random.seed(seed)
    deadline = time.time() + budget_s - 5
    best_seq = None
    best_score = float('inf')
    iteration = 0
    while time.time() < deadline:
        # choose length
        n = np.random.randint(3, 30)  # moderate
        # Dirichlet sample with random concentration
        alpha = np.random.rand(n) * 3 + 0.1   # between 0.1 and 3.1
        a = np.random.gamma(alpha, 1)
        if a.sum() == 0:
            continue
        a = a / a.sum()  # normalize sum=1
        # get initial score
        score = evaluate_sequence(a.tolist())
        if score < best_score:
            best_score = score
            best_seq = a.tolist()
            print(f"New best score {best_score:.6f} at iteration {iteration}, n={n}")
        # local hill climbing for a few steps
        for _ in range(50):
            # propose neighbor: transfer epsilon from index i to j
            i, j = np.random.randint(n, size=2)
            if i == j:
                continue
            epsilon = np.random.rand() * a[i]  # up to full value at i
            a_new = a.copy()
            a_new[i] -= epsilon
            a_new[j] += epsilon
            # ensure positivity, maybe small epsilon
            if a_new[i] < 0:
                continue
            # Normalize sum to 1 (actually sum unchanged)
            # Evaluate
            new_score = evaluate_sequence(a_new.tolist())
            if new_score < score:
                a = a_new
                score = new_score
                if new_score < best_score:
                    best_score = new_score
                    best_seq = a.tolist()
                    print(f"Iter {iteration}: improved score {best_score:.6f}")
        iteration += 1
    # final return best_seq; maybe ensure not too many zeros
    return best_seq

"""
Propose a step‑function (non‑negative list of heights) that minimises

    2 * n * max(convolve(a, a)) / (sum(a) ** 2)

The algorithm is a simple stochastic optimiser:

* Randomly choose a length `n` (3 … 30) and draw a Dirichlet vector.
* The vector is normalised to sum = 1 (the objective is scale‑invariant).
* A short hill‑climbing phase repeatedly moves a small amount of mass from one
  index to another; improvements are always kept, occasional worsening moves
  are accepted with a tiny probability (simulated‑annealing style) to escape
  tiny local minima.
* Occasionally the current vector is sorted in descending order – this
  strongly reduces the central convolution peak because large entries are
  paired with small ones.
* The best sequence seen before the time budget expires is returned.

The primitive operations are cheap (convolution of at most 30 elements), so
many thousands of candidates can be evaluated within the 1000 s budget.
"""

import time
import numpy as np

def _rand_dirichlet(n: int) -> np.ndarray:
    """Draw a random point on the simplex of dimension n."""
    # random concentration parameters give a wide variety of shapes
    alpha = np.random.rand(n) * 3.0 + 0.1   # each α ∈ (0.1 , 3.1)
    sample = np.random.gamma(alpha, 1.0)
    total = sample.sum()
    if total == 0.0:
        return np.full(n, 1.0 / n)
    return sample / total


def _hill_climb(a: np.ndarray, max_inner: int = 40) -> tuple[np.ndarray, float]:
    """
    Perform a short local search starting from ``a``.
    Returns the (possibly) improved vector and its score.
    """
    cur = a.copy()
    cur_score = evaluate_sequence(cur.tolist())

    # step size as a fraction of the donor entry; it shrinks a little
    # during the inner loop to act like a simple annealing schedule.
    base_step = 0.3

    for it in range(max_inner):
        n = cur.size
        # choose two distinct indices
        i, j = np.random.choice(n, size=2, replace=False)

        # amount to move from i to j (up to a fraction of the donor)
        eps = (np.random.rand() - 0.5) * base_step * cur[i]
        # keep both entries non‑negative
        if cur[i] - eps < 0.0:
            eps = cur[i]                      # move everything out of i
        elif cur[j] + eps < 0.0:
            eps = -cur[j]                     # move everything into j

        if eps == 0.0:
            continue

        cand = cur.copy()
        cand[i] -= eps
        cand[j] += eps

        cand_score = evaluate_sequence(cand.tolist())

        # accept if better, otherwise with a tiny probability
        if cand_score < cur_score or np.random.rand() < 0.01:
            cur, cur_score = cand, cand_score

        # slowly reduce the step magnitude
        base_step *= 0.995

    return cur, cur_score


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a sequence of non‑negative heights that lowers the evaluation
    function.  The function runs for at most ``budget_s`` seconds (leaving a
    small safety margin) and returns the best sequence found.
    """
    np.random.seed(seed)

    deadline = time.time() + budget_s - 5.0   # a few seconds safety margin
    best_seq = None
    best_score = float("inf")
    iteration = 0

    # main outer loop – many independent restarts
    while time.time() < deadline:
        # 1. choose a length (small lengths give cheap convolutions)
        n = np.random.randint(3, 31)  # 3 … 30

        # 2. draw a random point on the simplex
        a = _rand_dirichlet(n)

        # 3. occasionally enforce a descending order (helps break symmetry)
        if np.random.rand() < 0.3:
            a = np.sort(a)[::-1]

        # 4. evaluate the raw candidate
        cur_score = evaluate_sequence(a.tolist())
        if cur_score < best_score:
            best_score = cur_score
            best_seq = a.tolist()
            print(f"[{iteration:5d}] New best (raw)  score={best_score:.6f}  n={n}")

        # 5. local improvement phase
        a, cur_score = _hill_climb(a, max_inner=80)

        if cur_score < best_score:
            best_score = cur_score
            best_seq = a.tolist()
            print(f"[{iteration:5d}] New best (refined) score={best_score:.6f}  n={n}")

        # 6. occasionally try a pure random perturbation of the best so far
        if best_seq is not None and np.random.rand() < 0.05:
            # take the current best, add a tiny Gaussian jitter and re‑normalise
            b = np.array(best_seq, copy=True)
            jitter = np.random.randn(b.size) * 0.02
            b = np.maximum(b + jitter, 0.0)
            if b.sum() > 1e-8:
                b = b / b.sum()
                b_score = evaluate_sequence(b.tolist())
                if b_score < best_score:
                    best_score, best_seq = b_score, b.tolist()
                    print(f"[{iteration:5d}] Jitter improved score={best_score:.6f}")

        iteration += 1

    # safety net – if nothing was found (unlikely) return a trivial sequence
    if best_seq is None:
        best_seq = [1.0]

    return best_seq
