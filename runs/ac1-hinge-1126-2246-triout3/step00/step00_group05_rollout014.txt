def _eval_fast(seq: np.ndarray) -> float:
    """Fast evaluation using the same formula as evaluate_sequence."""
    # The official evaluate_sequence clamps values, but our candidates are far
    # from the limits, so we can safely compute the raw value.
    n = len(seq)
    conv = np.convolve(seq, seq)
    max_conv = np.max(conv)
    sum_seq = np.sum(seq)
    # protect against division by zero (should never happen for our candidates)
    if sum_seq < 1e-12:
        return float("inf")
    return 2.0 * n * max_conv / (sum_seq ** 2)


def _smooth_max_gradient(seq: np.ndarray, lam: float = 20.0) -> np.ndarray:
    """
    Compute the gradient of the smooth approximation
        softmax_λ(conv) = (1/λ)·log(∑exp(λ·conv)),
    where conv = seq * seq (self‑convolution).
    The gradient w.r.t. seq is
        2 * (probabilities ⋆ reversed(seq)),
    where probabilities are the softmax weights over the convolution entries.
    """
    n = len(seq)
    conv = np.convolve(seq, seq)               # length 2·n‑1
    max_val = np.max(conv)                     # for numerical stability
    # softmax probabilities (weights that sum to 1)
    exp_terms = np.exp(lam * (conv - max_val))
    probs = exp_terms / np.sum(exp_terms)      # shape (2·n‑1,)

    # convolution of probs with reversed seq gives the required sum:
    #   grad_i = 2 * Σ_k probs_k * seq_{k-i}
    rev_seq = seq[::-1]
    # The convolution length is (2·n‑1) + n - 1 = 3·n‑2; we need the central n values.
    full = np.convolve(probs, rev_seq)        # length 3·n‑2
    start = n - 1
    grad = 2.0 * full[start:start + n]
    return grad


def _scale_to_target_sum(seq: np.ndarray, target_sum: float) -> np.ndarray:
    """Scale the whole vector so that its L1‑norm equals target_sum."""
    cur = np.sum(seq)
    if cur == 0:
        return seq
    factor = target_sum / cur
    return seq * factor

            step_size = min(step_size * 1.01, 1.0)

        curr_seq = new_seq

    print(f"Finished search – best score = {best_score:.6f}")
    # Return a plain Python list as required by the evaluator
    return best_seq.tolist()
