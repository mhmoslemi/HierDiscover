def propose_candidate(seed=0, budget_s=1000):
    import numpy as np, time, math, random
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 1.0  # buffer
    n = 2000  # length
    # init sequence: random exponential
    seq = np.random.exponential(scale=1.0, size=n)
    # make sum moderate: ensure sum > 0.1, scale to have sum = sqrt(2n) maybe optional but not needed
    seq = seq / np.sum(seq) * np.sqrt(2*n) # scale similar to SOTA to keep magnitudes comparable
    cur_seq = seq.copy()
    cur_score = evaluate_sequence(cur_seq.tolist())
    best_seq = cur_seq.copy()
    best_score = cur_score
    # parameters
    T0 = 0.1
    T_end = 1e-5
    # linear or exponential schedule based on progress
    # iteration counter
    it = 0
    while time.time() < deadline:
        it += 1
        # cooling
        frac = (time.time() - start_time) / (deadline - start_time)
        T = T0 * (1 - frac) + T_end * frac
        # random move: transfer mass between two indices
        i, j = np.random.randint(n, size=2)
        if i == j:
            continue
        # propose delta
        max_transfer = min(cur_seq[i], cur_seq[i] + cur_seq[j]) # ensure nonneg after
        # choose delta proportion
        delta = (np.random.rand() - 0.5) * 0.1 * max_transfer
        # propose new values
        new_seq = cur_seq.copy()
        # adjust
        if delta > 0:
            # move from i to j
            if cur_seq[i] >= delta:
                new_seq[i] -= delta
                new_seq[j] += delta
        else:
            # move from j to i
            d = -delta
            if cur_seq[j] >= d:
                new_seq[i] += d
                new_seq[j] -= d
        # ensure nonneg
        if new_seq[i] < 0 or new_seq[j] < 0:
            continue
        # evaluate
        try:
            new_score = evaluate_sequence(new_seq.tolist())
        except Exception:
            continue
        # accept criteria
        accept = False
        if new_score < cur_score:
            accept = True
        else:
            # probabilistic accept
            if T > 0:
                prob = math.exp(-(new_score - cur_score) / T)
                if np.random.rand() < prob:
                    accept = True
        if accept:
            cur_seq = new_seq
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_seq = new_seq.copy()
                print(f"Iter {it}, new best score {best_score:.6f}")
        # occasional random global perturbation to escape stagnation
        if it % 5000 == 0:
            # reinitialize a small fraction of entries
            mask = np.random.rand(n) < 0.01
            cur_seq[mask] = np.random.exponential(scale=1.0, size=np.sum(mask))
            cur_seq = cur_seq / np.sum(cur_seq) * np.sqrt(2 * n)
    # return best_seq as list
    return best_seq.tolist()

"""Search for a step-function sequence with low evaluate_sequence score.

The algorithm combines three ideas:
1.  A random initialization (exponential/linear shape) that already yields a
    score slightly below the trivial bound 2.
2.  Periodic calls to a linear‑program that finds a direction `g` which
    maximises the ℓ₁‑mass while staying under the current convolution bound.
    The new candidate is taken as a convex combination `(1‑t)·f + t·g`
    where `t∈[0,1]` is chosen by a tiny ternary‑search to minimise the
    maximal convolution value (hence the evaluation score).
3.  Between LP calls a cheap simulated‑annealing style random mass‑transfer
    move that explores the neighbourhood of the current best sequence.

The search runs for at most `budget_s` seconds (default ≈ 1000 s) and
returns the best sequence discovered.
"""

import time
import math
import numpy as np
from scipy import optimize

# ------------------------------------------------------------
# Linear programming helper (same as in the original script)
# ------------------------------------------------------------
def solve_convolution_lp(f_sequence, rhs):
    """Solve LP: maximise sum(b) subject to non‑negative b and conv(f,b) ≤ rhs."""
    n = len(f_sequence)
    # objective: maximise sum(b) → minimise -sum(b)
    c = -np.ones(n)
    # build upper‑bound matrix for each convolution index
    a_ub = []
    b_ub = []
    for k in range(2 * n - 1):
        row = np.zeros(n)
        # f[i] appears in b_k when j = k - i is a valid index
        # i from max(0, k-(n-1)) to min(k, n-1)
        i_min = max(0, k - (n - 1))
        i_max = min(k, n - 1)
        # we set row[j] = f[i] where j = k - i
        # equivalently fill entries where j = k - i
        for i in range(i_min, i_max + 1):
            j = k - i
            row[j] = f_sequence[i]
        a_ub.append(row)
        b_ub.append(rhs)
    # non‑negativity constraints: -b ≤ 0
    a_ub.extend(-np.eye(n))
    b_ub.extend([0.0] * n)
    a_ub = np.vstack(a_ub)
    b_ub = np.array(b_ub)
    # SciPy linprog – use high‑speed interior‑point algorithm
    result = optimize.linprog(
        c,
        A_ub=a_ub,
        b_ub=b_ub,
        method="highs",
        options={"time_limit": 8.0, "presolve": True, "disp": False},
    )
    if result.success:
        return result.x
    return None


# ------------------------------------------------------------
# Helper for mixing two sequences and finding the optimal mix t
# ------------------------------------------------------------
def max_conv_for_t(t, conv_aa, conv_ag, conv_gg):
    """Maximum value of the convolution (1‑t)·a + t·g for a given t."""
    # (1‑t)²·aa + 2·t·(1‑t)·ag + t²·gg
    mixed = (1.0 - t) ** 2 * conv_aa + 2.0 * t * (1.0 - t) * conv_ag + t ** 2 * conv_gg
    return np.max(mixed)


def optimal_mixing_t(conv_aa, conv_ag, conv_gg, iters=30):
    """Ternary search on [0,1] to minimise max convolution."""
    lo, hi = 0.0, 1.0
    for _ in range(iters):
        m1 = lo + (hi - lo) / 3.0
        m2 = hi - (hi - lo) / 3.0
        f1 = max_conv_for_t(m1, conv_aa, conv_ag, conv_gg)
        f2 = max_conv_for_t(m2, conv_aa, conv_ag, conv_gg)
        if f1 < f2:
            hi = m2
        else:
            lo = m1
    return (lo + hi) / 2.0


def improve_via_lp(current_seq):
    """
    One LP step: compute a feasible direction g and mix it with the current
    sequence using the optimal convex coefficient t.
    Returns a new sequence (list of floats) or None if LP failed.
    """
    n = len(current_seq)
    sum_cur = np.sum(current_seq)
    # normalise so that the ℓ₁‑norm equals sqrt(2·n) (the same normalisation used
    # in the reference implementation).  This does not affect the final score.
    scale = math.sqrt(2.0 * n) / sum_cur
    f = current_seq * scale
    rhs = np.max(np.convolve(f, f))
    g_raw = solve_convolution_lp(f, rhs)
    if g_raw is None:
        return None
    sum_g = np.sum(g_raw)
    # bring g to the same ℓ₁‑norm as f
    g = g_raw * (math.sqrt(2.0 * n) / sum_g)

    # pre‑compute the three convolutions needed for the ternary search
    conv_aa = np.convolve(f, f)
    conv_ag = np.convolve(f, g)
    conv_gg = np.convolve(g, g)

    t_opt = optimal_mixing_t(conv_aa, conv_ag, conv_gg)
    new_seq = (1.0 - t_opt) * f + t_opt * g
    return new_seq.tolist()

# Random “mass‑transfer” neighbourhood move

def random_transfer_move(seq, step_factor=0.05):
    """
    Pick two distinct indices i, j and move a small amount of mass from i to j
    (or the opposite) while keeping all entries non‑negative.
    """
    n = len(seq)
    i, j = np.random.randint(n, size=2)
    if i == j:
        return None
    # Choose a transfer amount proportional to the source entry
    amount = (np.random.rand() - 0.5) * step_factor * max(seq[i], seq[j], 1.0)
    new_seq = seq.copy()
    if amount > 0:
        # move from i to j
        if new_seq[i] >= amount:
            new_seq[i] -= amount
            new_seq[j] += amount
        else:
            return None
    else:
        amount = -amount
        if new_seq[j] >= amount:
            new_seq[i] += amount
            new_seq[j] -= amount
        else:
            return None
    # enforce non‑negativity (tiny rounding errors)
    if np.any(new_seq < 0):
        return None
    return new_seq.tolist()

   convolution bound.  The new candidate is the convex combination

        (1‑t)·f + t·g ,   t∈[0,1]

   where `t` is chosen by a tiny ternary search to minimise the maximal
   convolution value (hence the evaluation score).
3. **Random mass‑transfer moves** – between LP calls a cheap simulated‑annealing

def solve_convolution_lp(f_sequence, rhs):

           maximise   sum(b)
           subject to  conv(f, b) ≤ rhs   (for every shift)
                       b ≥ 0
       Returns the optimal `b` as a NumPy array or ``None`` on failure.
    """
    n = len(f_sequence)

    c = -np.ones(n)

    b_ub = []
