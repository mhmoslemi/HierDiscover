"""
Simple simulated-annealing style optimizer for the step‑function evaluation problem.

The objective to minimise is

    f(a) = 2 * n * max(conv(a, a)) / (sum(a) ** 2)

where `conv` is the linear convolution of the non‑negative vector `a`.
Because the expression is scale‑invariant, we keep the ℓ₁‑norm fixed to
sqrt(2*n) (the normalisation used in the reference paper).  The algorithm
repeatedly identifies the shift `k` that currently attains the maximal
convolution value and moves a small amount of mass away from the pair of
indices `(i, j = k-i)` that contributes the largest product `a_i * a_j`.
The moved mass is redistributed to two random positions.  A temperature
schedule (simulated‑annealing) occasionally accepts uphill moves, helping
the search escape local plateaus.

The routine works for any length `n`.  A modest size (≈2000) gives a good
trade‑off between solution quality and speed, but the code automatically
adjusts the length if a previously found construction (`height_sequence_1`)
is available.

The function `propose_candidate` returns the best sequence found before the
deadline.  All intermediate progress is printed for debugging.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities
# ----------------------------------------------------------------------


def normalise_to_target(seq: np.ndarray, target_sum: float) -> np.ndarray:
    """Scale a non‑negative vector so that its ℓ₁‑norm equals `target_sum`."""
    s = np.sum(seq)
    if s == 0:
        return seq
    return seq * (target_sum / s)


def random_initial_sequence(n: int, target_sum: float, rng: np.random.Generator) -> np.ndarray:
    """Create a random non‑negative start vector of length `n`."""
    # start from a Dirichlet‑like distribution (uniform on the simplex)
    x = rng.random(n)
    x = normalise_to_target(x, target_sum)
    return x


def move_mass(seq: np.ndarray, k_max: int, rng: np.random.Generator,
              step_frac: float) -> np.ndarray:
    """
    Perform one local move that tries to lower the peak at shift `k_max`.

    - Find the pair (i, j) with i + j = k_max that has the largest product
      a_i * a_j.
    - Transfer a small amount `delta` from that pair to two other random
      locations (keeping the total sum unchanged).
    """
    n = seq.shape[0]
    # range of i that participates in the convolution for shift k_max
    i_start = max(0, k_max - (n - 1))
    i_end = min(n - 1, k_max)
    i_vals = np.arange(i_start, i_end + 1)
    j_vals = k_max - i_vals

    # products for all contributing pairs
    products = seq[i_vals] * seq[j_vals]
    # index of the pair with the largest product
    arg = int(np.argmax(products))
    i = int(i_vals[arg])
    j = int(j_vals[arg])

    # amount of mass to move – a fraction of the smaller entry of the pair
    if i == j:
        # diagonal term a_i * a_i : move from a_i only
        max_transfer = seq[i]
        if max_transfer == 0:
            return seq
        delta = step_frac * max_transfer
        # choose a random receiver (different from i)
        recv = rng.choice([idx for idx in range(n) if idx != i])
        new_seq = seq.copy()
        new_seq[i] -= delta
        new_seq[recv] += delta
        return new_seq
    else:
        max_transfer = min(seq[i], seq[j])
        if max_transfer == 0:
            return seq
        delta = step_frac * max_transfer
        # pick two distinct receivers that are not i or j
        candidates = [idx for idx in range(n) if idx not in (i, j)]
        if len(candidates) < 2:
            # fall back to a simple random perturbation
            new_seq = seq.copy()
            new_seq[i] = max(0.0, seq[i] - delta)
            new_seq[j] = max(0.0, seq[j] - delta)
            new_seq[i] += delta  # give back to i – no net change
            return new_seq
        recv = rng.choice(candidates, size=2, replace=False)
        l1, l2 = int(recv[0]), int(recv[1])
        new_seq = seq.copy()
        new_seq[i] -= delta
        new_seq[j] -= delta
        new_seq[l1] += delta
        new_seq[l2] += delta
        return new_seq

    last_improve = start_time

    iteration = 0
    while time.time() < deadline:
        iteration += 1

        # --------------------------------------------------------------
        # 1) compute the convolution and locate the current peak
        # --------------------------------------------------------------
        conv = np.convolve(cur_seq, cur_seq)
        M = float(conv.max())
        k_max = int(conv.argmax())

        # --------------------------------------------------------------
        # 2) propose a new vector by moving mass away from the worst pair
        # --------------------------------------------------------------
        new_seq = move_mass(cur_seq, k_max, rng, step_frac)

        # --------------------------------------------------------------
        # 3) evaluate the candidate
        # --------------------------------------------------------------
        try:
            new_score = evaluate_sequence(new_seq.tolist())
        except Exception:
            new_score = np.inf

        # --------------------------------------------------------------
        # 4) acceptance rule (simulated annealing)
        # --------------------------------------------------------------
        accept = False
        if new_score < cur_score:
            accept = True
        else:
            # probability of uphill move
            prob = np.exp((cur_score - new_score) / max(temperature, 1e-12))
            if rng.random() < prob:
                accept = True

        if accept:
            cur_seq = new_seq
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_seq = new_seq.copy()
                last_improve = time.time()
                print(f"[{time.time() - start_time:6.1f}s] new best: {best_score:.6f}")

        # --------------------------------------------------------------
        # 5) temperature schedule & occasional perturbations
        # --------------------------------------------------------------
        temperature = max(0.01, temperature * 0.9999)

        # random global jitter every few hundred iterations
        if iteration % 300 == 0:
            jitter = 0.001 * rng.standard_normal(n)
            cur_seq = np.maximum(0.0, cur_seq + jitter)
            cur_seq = normalise_to_target(cur_seq, target_sum)

        # if we have stagnated for a while, re‑initialise partially
        if time.time() - last_improve > 30.0:
            # keep the best part of the current solution and inject fresh noise
            cur_seq = best_seq * 0.8 + 0.2 * rng.random(n) * target_sum
            cur_seq = normalise_to_target(cur_seq, target_sum)
            cur_score = evaluate_sequence(cur_seq.tolist())
            last_improve = time.time()
            print(f"[{time.time() - start_time:6.1f}s] re‑initialised after stagnation")

        # slowly decay the step size as we go
        if iteration % 500 == 0:
            step_frac = max(0.001, step_frac * 0.95)

    print(f"[finished] best score = {best_score:.6f} after {iteration} iterations")
    return best_seq.tolist()
