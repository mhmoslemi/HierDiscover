"""
A fast, derivative‑free search for a step‑function with a small
evaluation score.

The algorithm works with a moderate length (500‑800) sequence.
It starts from a simple geometrically decreasing sequence (which already
beats the uniform 2.0 bound) and then repeatedly performs two kinds of
updates:

1. **Linear‑programming direction** – solves an LP that finds a
   non‑negative vector `g` with a larger total mass while keeping the
   cross‑convolution `f * g` below the current maximum `M = max(f*f)`.
   The current sequence `f` is then mixed with `g` using a short line
   search on the mixing weight `t`.  This step reduces `max(f*f)` while
   increasing the sum of the coefficients, which lowers the score.

2. **Random local perturbations** – when the LP fails or does not give an
   improvement, a tiny amount of mass is transferred between two random
   positions (projected onto the non‑negative orthant).  This helps to
   escape plateaus.

The search is time‑bounded (default 1000 s).  The best sequence found
within the budget is returned.  The implementation uses only NumPy and
SciPy (HiGHS solver) and logs progress to stdout.
"""

import time
import numpy as np
from scipy.optimize import linprog

# ----------------------------------------------------------------------
# Helper utilities
# ----------------------------------------------------------------------


def _normalize(seq):
    """Return a copy of seq scaled to have sum = 1.0 (scale invariant)."""
    s = float(np.sum(seq))
    if s == 0.0:
        return np.ones_like(seq) / len(seq)
    return [x / s for x in seq]


def _max_convolution(seq):
    """Maximum value of the self‑convolution of seq."""
    b = np.convolve(seq, seq)
    return float(np.max(b))


def _solve_lp(f_seq, rhs):
    """
    Solve the LP:
        maximize sum(b)
        s.t.   conv(f_seq, b) <= rhs
               b >= 0
    Returns the optimal b as a list (or None on failure).
    """
    n = len(f_seq)
    # maximise sum(b)  → minimise -sum(b)
    c = -np.ones(n)

    # Build A_ub * b <= b_ub for each convolution index.
    # a_ub[k, j] = f_seq[i] where i + j == k and 0<=i,j<n
    a_ub_rows = []
    b_ub = []

    for k in range(2 * n - 1):
        row = np.zeros(n)
        # i runs over all valid positions
        i_min = max(0, k - (n - 1))
        i_max = min(n - 1, k)
        # fill the row efficiently
        for i in range(i_min, i_max + 1):
            j = k - i
            row[j] = f_seq[i]
        a_ub_rows.append(row)
        b_ub.append(rhs)

    # non‑negativity constraints: -I * b <= 0
    a_ub_rows.append(-np.eye(n))
    b_ub.append(np.zeros(n))

    A_ub = np.vstack(a_ub_rows)
    b_ub = np.hstack(b_ub)

    # HiGHS is fast and respects the time limit.
    res = linprog(
        c,
        A_ub=A_ub,
        b_ub=b_ub,
        method="highs",
        options={"time_limit": 5.0, "disp": False},
    )
    if not res.success:
        return None
    return res.x.tolist()


def _lp_direction(seq):
    """
    Compute a better direction using the LP described in the
    reference.  Returns a new sequence (mixed) or None if the LP fails.
    """
    n = len(seq)
    # Normalise to sum = 1 (scale does not affect the LP constraints
    # because all constraints are homogeneous).
    f = _normalize(seq)

    # Current bound M = max convolution of f with itself.
    M = _max_convolution(f)

    # Solve the LP for a partner vector g0.
    g0 = _solve_lp(f, M)
    if g0 is None:
        return None

    # Scale g0 so that it has the same total mass as the original seq.
    sum_f = np.sum(seq)
    sum_g0 = np.sum(g0)
    if sum_g0 == 0.0:
        return None
    scale = sum_f / sum_g0
    g = [x * scale for x in g0]

    # --- line search on the mixing weight t ---------------------------------
    # Try a few different mixing ratios and keep the best improvement.
    best_mix = seq
    best_score = evaluate_sequence(seq)

    # candidate t values (including 0 → keep the old sequence)
    t_vals = [0.0, 0.01, 0.02, 0.05, 0.1, 0.2, 0.5]
    for t in t_vals[1:]:
        mixed = [(1.0 - t) * a + t * b for a, b in zip(seq, g)]
        try:
            sc = evaluate_sequence(mixed)
        except Exception:
            continue
        if sc < best_score:
            best_score = sc
            best_mix = mixed

    return best_mix


def _random_perturb(seq, sigma=0.01):
    """
    Transfer a small random amount of mass between two positions while
    keeping the total sum unchanged and all entries non‑negative.
    """
    n = len(seq)
    if n < 2:
        return seq[:]

    i, j = np.random.choice(n, 2, replace=False)

    # Size of the transfer – proportional to the typical magnitude.
    delta = sigma * np.random.randn()
    # Make sure we do not create negative entries.
    max_decrease = min(delta, seq[i])
    max_increase = -min(delta, -seq[j])  # prevent j from going negative

    # Restrict delta to the feasible interval.
    delta = np.clip(delta, -seq[i], seq[j])

    new_seq = seq[:]
    new_seq[i] -= delta
    new_seq[j] += delta
    # Numerical safety – clip tiny negatives to zero.
    new_seq[i] = max(0.0, new_seq[i])
    new_seq[j] = max(0.0, new_seq[j])
    return new_seq

    """
    Search for a step‑function (list of non‑negative heights) that minimizes
    ``evaluate_sequence``.  The routine runs for approximately ``budget_s``
    seconds and returns the best sequence found.

    Parameters
    ----------
    seed : int
        Random seed for reproducibility.
    budget_s : float
        Time budget in seconds.

    Returns
    -------
    list[float]
        The best sequence discovered.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 2.0  # a small safety margin

    # ------------------------------------------------------------------
    # Keep the globally best result across all restarts.
    # ------------------------------------------------------------------
    global_best_seq = None
    global_best_score = float("inf")

    # ------------------------------------------------------------------
    # Main optimisation loop – repeatedly start from a fresh randomisation.
    # ------------------------------------------------------------------
    while time.time() < deadline:
        # Choose a problem size (larger sizes give better theoretical bound
        # but are slower to optimise).  We stay between 500 and 800.
        N = np.random.randint(500, 801)

        # Initialise with a simple geometrically decreasing sequence.
        # Parameter r is chosen so that the decay length ≈ N/2.
        decay = max(0.001, 2.0 / N)                     # ε ≈ 2/N
        r = 1.0 - decay
        init = [(1.0 - r) * (r ** i) for i in range(N)]
        # Normalise to unit sum (scale invariance).
        s = float(np.sum(init))
        init = [x / s for x in init]

        # Keep a copy of the current working sequence.
        cur_seq = init
        try:
            cur_score = evaluate_sequence(cur_seq)
        except Exception:
            cur_score = float("inf")

        # If this random start is already better than the global best,
        # record it.
        if cur_score < global_best_score:
            global_best_score = cur_score
            global_best_seq = cur_seq[:]
            print(
                f"[{time.time() - start_time:.1f}s] New global best "
                f"score={global_best_score:.6f} (N={N})"
            )

        # --------------------------------------------------------------
        # Local optimisation for this start: alternate LP directions and
        # random perturbations.
        # --------------------------------------------------------------
        inner_deadline = time.time() + min(30.0, deadline - time.time())
        while time.time() < inner_deadline:
            # Try the LP‑based direction.
            new_seq = _lp_direction(cur_seq)

            if new_seq is None:
                # LP failed – fall back to a cheap random move.
                new_seq = _random_perturb(cur_seq, sigma=0.005)

            # Evaluate candidate.
            try:
                new_score = evaluate_sequence(new_seq)
            except Exception:
                new_score = float("inf")

            # Accept if we have an improvement.
            if new_score < cur_score:
                cur_seq = new_seq
                cur_score = new_score
                # Update the global best if needed.
                if cur_score < global_best_score:
                    global_best_score = cur_score
                    global_best_seq = cur_seq[:]
                    print(
                        f"[{time.time() - start_time:.1f}s] "
                        f"Improved global best: {global_best_score:.6f} "
                        f"(N={N})"
                    )
                continue

            # If not improved, try a small random perturbation to escape.
            pert = _random_perturb(cur_seq, sigma=0.01)
            try:
                pert_score = evaluate_sequence(pert)
            except Exception:
                pert_score = float("inf")

            if pert_score < cur_score:
                cur_seq = pert
                cur_score = pert_score
                if cur_score < global_best_score:
                    global_best_score = cur_score
                    global_best_seq = cur_seq[:]
                    print(
                        f"[{time.time() - start_time:.1f}s] "
                        f"Improved global best by perturb: "
                        f"{global_best_score:.6f} (N={N})"
                    )
            # else: keep current sequence and continue the loop.

        # End of inner optimisation for this restart.

    # ------------------------------------------------------------------
    # Return the best sequence discovered.
    # ------------------------------------------------------------------
    if global_best_seq is None:
        # As a fallback, return a simple uniform sequence.
        N = 500
        global_best_seq = [1.0 / N] * N
        print("Fallback to uniform sequence.")
    else:
        print(
            f"Finished search. Best score = {global_best_score:.6f} "
            f"with length {len(global_best_seq)}."
        )

    return global_best_seq
