def propose_candidate(seed=0, budget_s=1000, **kwargs):
    """
    Search for a sequence of non‑negative heights that lowers the
    `evaluate_sequence` score. The method exploits the scale‐invariance
    of the objective and constructs monotone increasing power‑law
    sequences a_i = C * (i/(n‑1))**p, with the maximum clipped at 1000.
    It scans a range of exponents p, keeps the best (lowest) score,
    optionally refines around the best value, and returns the final list.
    """
    import time, numpy as np
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5

    try:
        n = len(height_sequence_1)
    except NameError:
        # fallback: use a moderate length that fits in time
        n = 3500

    # helper: evaluate a candidate made from exponent p
    def make_seq(p):
        idx = np.arange(n, dtype=float)
        # avoid division by zero for n=1
        if n == 1:
            return [0.0]
        # scale to max 1000
        seq = 1000.0 * (idx / (n - 1)) ** p
        return seq.tolist()

    best_seq = None
    best_score = float('inf')
    best_p = None

    # coarse scan
    for p in np.linspace(0.1, 12.0, 120):
        seq = make_seq(p)
        score = evaluate_sequence(seq)
        if score < best_score:
            best_score = score
            best_seq = seq
            best_p = p

    print(f"Coarse scan best p={best_p:.4f}, score={best_score:.6f}")

    # refine in a neighbourhood of best p
    if best_p is not None:
        low = max(0.0, best_p - 1.0)
        high = best_p + 1.0
        # perform finer grid
        for p in np.linspace(low, high, 200):
            if time.time() > deadline:
                break
            seq = make_seq(p)
            score = evaluate_sequence(seq)
            if score < best_score:
                best_score = score
                best_seq = seq
                best_p = p

    print(f"Final best p={best_p:.6f}, score={best_score:.6f}")

    # final random local search around best shape (optional)
    # use small Gaussian perturbations preserving non‑negativity
    seq = np.array(best_seq, dtype=float)
    # ensure we have a work copy
    curr_seq = seq.copy()
    curr_score = best_score

    # local search parameters
    sigma = 5.0   # typical magnitude of perturbation
    while time.time() < deadline:
        # pick two indices i<j and transfer a small amount
        i, j = np.random.randint(n, size=2)
        if i == j:
            continue
        delta = np.random.randn() * sigma
        # propose new values
        new_seq = curr_seq.copy()
        new_seq[i] = max(0.0, new_seq[i] + delta)
        new_seq[j] = max(0.0, new_seq[j] - delta)  # keep total sum roughly constant
        # evaluate after clipping to 0-1000 (evaluate_sequence does that)
        score = evaluate_sequence(new_seq.tolist())
        # accept if improvement
        if score < curr_score:
            curr_seq = new_seq
            curr_score = score
            if score < best_score:
                best_score = score
                best_seq = new_seq.tolist()
                # reduce sigma as we improve
                sigma = max(0.1, sigma * 0.99)
        else:
            # occasionally accept worse move (simulated annealing)
            if np.random.rand() < 0.001:
                curr_seq = new_seq
                curr_score = score

    print(f"After local search best score={best_score:.6f}")
    return best_seq

"""
propose_candidate : search for a step‑function (non‑negative list) that minimises
evaluate_sequence.  The objective is scale‑invariant, therefore we may fix the
largest entry to 1000 (the hard bound in evaluate_sequence) and only vary the
shape.  For a monotone increasing power‑law shape

    a_i = 1000 * (i/(n‑1))**p   (i = 0,…,n‑1)

the value of evaluate_sequence depends essentially on the exponent p but not
on the absolute scale.  By scanning p we can locate the exponent that gives the
smallest possible score for the given length n (taken from the existing
height_sequence_1 if present).  After a coarse grid we perform a finer scan
around the best exponent.  The best sequence found is returned.

No external files are accessed and the implementation obeys the required
no‑closure, top‑level‑function rule.
"""
import time
import numpy as np

def make_sequence(p: float, n: int) -> list[float]:
    """
    Returns a monotone increasing power‑law sequence of length `n` whose maximum
    value is 1000 (the clipping bound).  The shape is given by
        a[i] = 1000 * (i/(n‑1))**p
    with a[0] = 0.  The function works for n≥2; for n==1 it returns [0.0].
    """
    if n <= 1:
        return [0.0]
    # index array 0 … n‑1 as float
    idx = np.arange(n, dtype=float)
    # compute the power‑law, scaling the last entry to exactly 1000
    seq = 1000.0 * (idx / (n - 1)) ** p
    return seq.tolist()


def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a sequence that gives a low value of `evaluate_sequence`.
    The method:
        1. Determine the length `n` (use the length of the global
           `height_sequence_1` if it exists, otherwise fall back to 3500).
        2. Scan a wide range of exponents `p` for the power‑law shape
           defined by `make_sequence`.
        3. Keep the best (lowest) score; then perform a finer scan around that
           exponent.
        4. Return the best sequence found.
    The whole routine respects the 1000‑second time budget.
    """
    np.random.seed(seed)

    # -------------------------------------------------------------
    # 1. Determine the length of the candidate sequence
    # -------------------------------------------------------------
    try:
        n = len(height_sequence_1)  # type: ignore[name-defined]
    except Exception:
        n = 3500   # a safe default length if the global is unavailable

    deadline = time.time() + budget_s - 0.5  # keep a small safety margin

    # -------------------------------------------------------------
    # 2. Coarse scan over a wide range of exponents
    # -------------------------------------------------------------
    best_score = float('inf')
    best_seq = None
    best_p = None

    # Use a reasonably dense grid (≈120 points) from p=0.1 up to p=12.
    coarse_ps = np.linspace(0.1, 12.0, 120)
    for p in coarse_ps:
        if time.time() > deadline:
            break
        seq = make_sequence(p, n)
        score = evaluate_sequence(seq)   # evaluate_sequence is supplied by the runner
        if score < best_score:
            best_score = score
            best_seq = seq
            best_p = p

    print(f"[coarse] best p = {best_p:.4f}, score = {best_score:.6f}")

    # -------------------------------------------------------------
    # 3. Refine around the best exponent found
    # -------------------------------------------------------------
    if best_p is not None:
        # Search a neighbourhood of width ±1.0 around the current best p
        low = max(0.0, best_p - 1.0)
        high = best_p + 1.0
        fine_ps = np.linspace(low, high, 200)   # finer resolution
        for p in fine_ps:
            if time.time() > deadline:
                break
            seq = make_sequence(p, n)
            score = evaluate_sequence(seq)
            if score < best_score:
                best_score = score
                best_seq = seq
                best_p = p

    print(f"[refine] best p = {best_p:.6f}, score = {best_score:.6f}")

    # -------------------------------------------------------------
    # 4. (Optional) very short local random perturbation phase
    #    – only if time permits.  This can occasionally improve the
    #    result further without breaking the monotonic shape.
    # -------------------------------------------------------------
    if best_seq is not None and time.time() < deadline:
        cur_seq = np.array(best_seq, dtype=float)
        cur_score = best_score
        # Standard deviation for the additive noise; start modest.
        sigma = 5.0
        while time.time() < deadline:
            # Pick two different indices and move a tiny amount from one to the other.
            i, j = np.random.randint(0, n, size=2)
            if i == j:
                continue
            delta = np.random.randn() * sigma
            new_seq = cur_seq.copy()
            new_seq[i] = max(0.0, new_seq[i] + delta)
            new_seq[j] = max(0.0, new_seq[j] - delta)   # keep total roughly constant
            new_score = evaluate_sequence(new_seq.tolist())
            # Accept improvement immediately.
            if new_score < cur_score:
                cur_seq = new_seq
                cur_score = new_score
                if new_score < best_score:
                    best_score = new_score
                    best_seq = new_seq.tolist()
                    # tighten the perturbation scale as we get better.
                    sigma = max(0.1, sigma * 0.99)
            else:
                # Very occasional uphill moves (simulated annealing style).
                if np.random.rand() < 0.001:
                    cur_seq = new_seq
                    cur_score = new_score

    print(f"[final] best score = {best_score:.6f}")
    # `best_seq` is guaranteed to be a list of floats (non‑negative, ≤1000).
    return best_seq
