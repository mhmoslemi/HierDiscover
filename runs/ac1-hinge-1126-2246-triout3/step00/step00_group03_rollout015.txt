def convolve_fft(a,b):
    size = len(a)+len(b)-1
    N = 1 << (size-1).bit_length()
    A = np.fft.rfft(a, n=N)
    B = np.fft.rfft(b, n=N)
    C = A * B
    conv = np.fft.irfft(C, n=N)[:size]
    return conv

def simplex_projection(v):
    # returns projection onto {x>=0, sum=1}
    # Use O(n log n)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(len(v)) + 1
    cond = u - cssv / ind > 0
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0)
    return w

def convolve_fft(a,b):
    size = len(a)+len(b)-1
    N = 1 << (size-1).bit_length()
    A = np.fft.rfft(a, N)
    B = np.fft.rfft(b, N)
    C = A * B
    conv = np.fft.irfft(C, N)[:size]
    return conv

def solve_lp(f, M):
    n = len(f)

"""
Propose a candidate sequence that (hopefully) improves the upper bound.
The algorithm follows the published iterative scheme but with several
enhancements:

1.  The sequence is kept on the simplex (non‑negative, fixed ℓ₁‑norm
    √(2·n)).  This makes the evaluation function equal to the maximum
    entry of the self‑convolution, i.e.  ``evaluate_sequence(seq) == max(f*f)``.
2.  In each iteration we solve a linear program that finds a direction
    ``g`` with the same ℓ₁‑norm as ``f`` and satisfying ``f*g ≤ M`` where
    ``M`` is the current maximum convolution value.
3.  Instead of mixing with a tiny fixed step (the reference code used ``t=0.01``)
    we compute the *optimal* mixing coefficient ``t∈[0,1]`` by a cheap ternary
    search that minimises the new maximum convolution of the convex
    combination ``f' = (1‑t)·f + t·g``.
4.  The convolution operations are performed with FFT (``O(n log n)``) for
    speed.  Linear‑program constraints are built explicitly; for modest sizes
    (n≈500…2000) this is cheap.
5.  The procedure is repeated until the time budget (≈1000 s) expires,
    keeping the best sequence seen.  Random perturbations are applied
    occasionally to escape plateaus.

The returned list is a non‑negative step function that respects
the security checks of ``evaluate_sequence``.
"""

import time
import math
import numpy as np
from scipy.optimize import linprog

# ----------------------------------------------------------------------
# Helper functions
# ----------------------------------------------------------------------


def convolve_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """
    Compute the discrete convolution of two real vectors using FFT.
    Returns the convolution of length len(a)+len(b)-1.
    """
    size = len(a) + len(b) - 1
    N = 1 << (size - 1).bit_length()          # next power of two
    A = np.fft.rfft(a, N)
    B = np.fft.rfft(b, N)
    C = A * B
    conv = np.fft.irfft(C, N)[:size]
    return conv


def solve_convolution_lp(f: np.ndarray, M: float):
    """
    Solve the linear program

        maximise   sum(b)
        subject to  (f * b)_k ≤ M   for all k
                    b ≥ 0

    ``f`` and ``b`` are length‑n non‑negative vectors,
    ``M`` is the current maximal convolution value.
    Returns the optimal ``b`` as a NumPy array, or ``None`` on failure.
    """
    n = len(f)
    # Build the Toeplitz‑style matrix for the convolution constraints.
    # Row k, column j contains f[k‑j] when 0 ≤ k‑j < n, else 0.
    rows = np.zeros((2 * n - 1, n), dtype=np.float64)
    for k in range(2 * n - 1):
        j_start = max(0, k - (n - 1))
        j_end = min(k, n - 1)
        # j runs over the valid indices; the partner index is i = k - j
        rows[k, j_start:j_end + 1] = f[k - np.arange(j_start, j_end + 1)]

    # Non‑negativity constraints  –b ≤ 0
    A_ub = np.vstack([rows, -np.eye(n, dtype=np.float64)])
    b_ub = np.hstack([np.full(2 * n - 1, M), np.zeros(n, dtype=np.float64)])

    # maximise sum(b)  → minimise -sum(b)
    c = -np.ones(n, dtype=np.float64)

    # Use the HiGHS interior‑point solver (fast for moderate sizes)
    res = linprog(
        c,
        A_ub=A_ub,
        b_ub=b_ub,
        method="highs",
        options={"presolve": True, "time_limit": 2.0},
    )
    if res.success:
        # Numerical safety: clip tiny negatives to zero
        b_opt = np.maximum(res.x, 0.0)
        return b_opt
    return None


def optimal_mixing_t(conv_ff: np.ndarray,
                     conv_gg: np.ndarray,
                     conv_fg: np.ndarray) -> float:
    """
    For the convex combination f' = (1‑t)·f + t·g,
    the self‑convolution is

        C(t) = (1‑t)²·conv_ff + t²·conv_gg + 2·t·(1‑t)·conv_fg

    The function returns the t∈[0,1] that minimises max(C(t))
    using a few steps of ternary search (convex in t).
    """
    lo, hi = 0.0, 1.0
    for _ in range(30):
        m1 = lo + (hi - lo) / 3.0
        m2 = hi - (hi - lo) / 3.0
        c1 = (1 - m1) ** 2 * conv_ff + m1 ** 2 * conv_gg + 2 * m1 * (1 - m1) * conv_fg
        c2 = (1 - m2) ** 2 * conv_ff + m2 ** 2 * conv_gg + 2 * m2 * (1 - m2) * conv_fg
        if c1.max() < c2.max():
            hi = m2
        else:
            lo = m1
    return (lo + hi) / 2.0


def random_dirichlet(n: int, total: float) -> np.ndarray:
    """Draw a random point on the simplex with prescribed ℓ₁‑norm."""
    x = np.random.gamma(1.0, 1.0, size=n)
    x = x / x.sum() * total
    return x

# ----------------------------------------------------------------------

    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 2.0          # small safety margin

    best_seq = None
    best_score = np.inf

    # Choose a few problem sizes; larger n gives a better theoretical bound
    # but also higher per‑iteration cost.  We try several sizes while time permits.

        best_score = M
        best_seq = f.copy()

        iteration = 0
        no_improve_cnt = 0

        while time.time() < deadline:
            # ------------------------------------------------------------------
            # 1) Linear programme to obtain a feasible direction g
            # ------------------------------------------------------------------
            g_raw = solve_convolution_lp(f, M)
            if g_raw is None:
                # If the LP fails, perform a tiny random perturbation and continue
                idx = np.random.randint(n)
                f[idx] = max(0.0, f[idx] + np.random.randn() * 0.01)
                f *= sqrt_2n / f.sum()
                conv_ff = convolve_fft(f, f)
                M = conv_ff.max()
                continue

            # Rescale g so that it has the same ℓ₁‑norm as f ( = √(2n) )
            sum_g = g_raw.sum()
            if sum_g == 0.0:
                # Degenerate case – fall back to a random simplex point
                g = random_dirichlet(n, sqrt_2n)
            else:
                g = g_raw * (sqrt_2n / sum_g)

            # ------------------------------------------------------------------
            # 2) Compute the three relevant convolutions (FFT‑based)
            # ------------------------------------------------------------------
            conv_gg = convolve_fft(g, g)
            conv_fg = convolve_fft(f, g)

            # ------------------------------------------------------------------
            # 3) Find the optimal mixing coefficient t ∈ [0, 1]
            # ------------------------------------------------------------------
            t_opt = optimal_mixing_t(conv_ff, conv_gg, conv_fg)

            # ------------------------------------------------------------------
            # 4) Update the current sequence
            # ------------------------------------------------------------------
            f = (1.0 - t_opt) * f + t_opt * g
            conv_ff = convolve_fft(f, f)
            M_new = conv_ff.max()

            # ------------------------------------------------------------------
            # 5) Keep the best sequence found so far
            # ------------------------------------------------------------------
            if M_new + 1e-12 < best_score:
                best_score = M_new
                best_seq = f.copy()
                no_improve_cnt = 0
                print(
                    f"[n={n}] iter={iteration:4d}  new best M={best_score:.6f}  t={t_opt:.4f}"
                )
            else:
                no_improve_cnt += 1

            # ------------------------------------------------------------------
            # 6) Stagnation handling – occasional random restart
            # ------------------------------------------------------------------
            if no_improve_cnt >= 150:
                # Small random perturbation keeps the search alive
                f = random_dirichlet(n, sqrt_2n)
                conv_ff = convolve_fft(f, f)
                M_new = conv_ff.max()
                no_improve_cnt = 0

            M = M_new
            iteration += 1

            # Bail out early if we have already reached a very good bound
            if best_score < 1.51:
                # This is close to the target 1.503; continue a little longer
                pass

    # Return the best sequence as a plain Python list (evaluate_sequence will
    # perform the final checks when the platform scores the submission).
    return best_seq.tolist()
