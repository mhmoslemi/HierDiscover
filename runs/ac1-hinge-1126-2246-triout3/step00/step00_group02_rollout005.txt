def propose_candidate(seed=0, budget_s=1000, **kwargs):
    '''
    Simple algorithm that returns a uniform step function.

    The function creates a sequence where all heights are equal,
    which analytically yields the minimal possible value of the
    evaluation function (2). It also performs a quick random
    local search in the remaining time in case floating‑point
    artefacts allow an even smaller value.
    '''
    import numpy as np, time, random
    # Start with uniform sequence
    n = kwargs.get('length', 1000)
    # ensure sum > 0
    uniform_val = 1.0
    best_seq = [uniform_val] * n
    best_score = evaluate_sequence(best_seq)

    deadline = time.time() + budget_s - 0.5
    # random perturbation hill climbing
    while time.time() < deadline:
        # propose variation by scaling a random subset
        seq = best_seq.copy()
        i = random.randrange(n)
        # add small gaussian noise
        seq[i] = max(0.0, seq[i] + random.gauss(0, 0.5))
        # enforce max 1000
        if seq[i] > 1000.0: seq[i] = 1000.0
        score = evaluate_sequence(seq)
        if score < best_score:
            best_seq, best_score = seq, score

    print(f'Best score found: {best_score}')
    return best_seq

"""
Simple but reliable search for a step‑function that minimizes
evaluate_sequence.

The mathematics of the objective
    2 * n * max(f*f) / (sum(f)^2)
guarantees that a perfectly uniform, non‑negative sequence attains the
global optimum value 2.  Nevertheless we perform a cheap stochastic
search (random perturbations + a tiny linear‑programming step) in case
floating‑point artefacts allow a value < 2.  The algorithm always falls
back to the uniform sequence, so it never returns an invalid (inf)
score.
"""

import time
import random
import numpy as np
from scipy import optimize

# ----------------------------------------------------------------------
# Helper – solve the LP used in the reference method
# ----------------------------------------------------------------------
def solve_convolution_lp(f_seq, rhs):
    """
    Solve:
        max  Σ b_i
        s.t. conv(f_seq, b) ≤ rhs   (all 2n‑1 shifts)
             b_i ≥ 0
    Returns the optimal b as a NumPy array or None on failure.
    """
    n = len(f_seq)
    # Objective: maximize sum(b)  →  minimize -1⋅b
    c = -np.ones(n)

    # Build the convolution constraints (2n‑1 rows)
    a_ub = []
    b_ub = []
    for k in range(2 * n - 1):
        row = np.zeros(n)
        # row[j] = f_seq[i] for i+j == k  ⇔  j = k‑i
        i_min = max(0, k - (n - 1))
        i_max = min(n - 1, k)
        for i in range(i_min, i_max + 1):
            j = k - i
            row[j] = f_seq[i]
        a_ub.append(row)
        b_ub.append(rhs)

    # Non‑negativity constraints  -b ≤ 0
    a_ub.extend(-np.eye(n))
    b_ub.extend([0.0] * n)

    a_ub = np.vstack(a_ub)
    b_ub = np.array(b_ub, dtype=float)

    # Simple linprog call – we limit the time per call to keep the whole
    # search responsive.
    try:
        res = optimize.linprog(
            c,
            A_ub=a_ub,
            b_ub=b_ub,
            bounds=(0, None),
            method="highs",
            options={"time_limit": 5.0},
        )
    except Exception:  # pragma: no cover – safety net
        return None

    if res.success:
        return res.x
    return None


# ----------------------------------------------------------------------
# Helper – generate a direction that (usually) improves the objective
# ----------------------------------------------------------------------
def get_good_direction_to_move_into(seq):
    """
    Implements the LP “direction” from the state‑of‑the‑art description.
    Returns a new sequence (list of floats) that is a small move toward the
    LP solution, or None if the LP fails.
    """
    n = len(seq)
    s = float(np.sum(seq))
    if s < 1e-9:               # degenerate – shouldn't happen
        return None

    # Normalise to the theoretical scaling √(2n) used in the paper.
    norm = np.sqrt(2.0 * n) / s
    f_norm = [x * norm for x in seq]

    # Upper bound for the convolution of the normalised sequence.
    rhs = float(np.max(np.convolve(f_norm, f_norm)))

    g = solve_convolution_lp(f_norm, rhs)
    if g is None:
        return None

    # Rescale g back to the original scale.
    g_sum = float(np.sum(g))
    if g_sum < 1e-9:
        return None
    g_norm = [x * (np.sqrt(2.0 * n) / g_sum) for x in g]

    # Small convex combination – step size t is deliberately tiny.
    t = 0.02
    new_seq = [(1 - t) * a + t * b for a, b in zip(seq, g_norm)]
    return new_seq


# ----------------------------------------------------------------------
# Main entry point expected by the evaluator
# ----------------------------------------------------------------------
def propose_candidate(seed=42, budget_s=1000, **kwargs):
    """
    Return a list of non‑negative floats (the step heights) that (hopefully)
    yields a low value of ``evaluate_sequence``.

    The routine works as follows:
      1. Start from a uniform sequence (known optimum = 2.0).
      2. For the remaining time perform a light‑weight stochastic search:
         * Occasionally invoke the LP direction from the reference algorithm.
         * Otherwise apply a tiny random perturbation to a random element.
      3. Keep the best sequence seen and return it at the deadline.
    """
    random.seed(seed)
    np.random.seed(seed)

    # ------------------------------------------------------------------
    # 1️⃣  Initial solution – a perfectly uniform vector.
    # ------------------------------------------------------------------
    # Use a length that is comfortably within the allowed budget.
    n = kwargs.get("length", 1000)
    # Uniform height 1.0 gives sum = n and max convolution = n,
    # which evaluates exactly to 2.0 (the theoretical optimum).
    best_seq = [1.0] * n
    best_score = evaluate_sequence(best_seq)

    # ------------------------------------------------------------------
    # 2️⃣  Time‑bounded stochastic improvement loop.
    # ------------------------------------------------------------------
    deadline = time.time() + max(0.0, budget_s - 0.5)   # keep a tiny safety margin
    iteration = 0

    while time.time() < deadline:
        iteration += 1

        # Occasionally ask the LP for a principled direction.
        if random.random() < 0.3:
            candidate = get_good_direction_to_move_into(best_seq)
            if candidate is None:                     # fallback to random tweak
                candidate = best_seq[:]
        else:
            # Simple random perturbation of a single coordinate.
            candidate = best_seq[:]
            idx = random.randrange(n)
            # add a small Gaussian noise, keep within [0,1000]
            noise = random.gauss(0.0, 0.5)
            candidate[idx] = max(0.0, min(1000.0, candidate[idx] + noise))

        # Evaluate – any exception means an invalid candidate, just skip.
        try:
            score = evaluate_sequence(candidate)
        except Exception:  # pragma: no cover
            continue

        # Keep the improvement.
        if score < best_score:
            best_score = score
            best_seq = candidate
            print(f"[{iteration:05d}] New best score: {best_score:.6f}")

        # Occasionally try a completely fresh random uniform candidate;
        # useful if the current length is sub‑optimal.
        if iteration % 2000 == 0:
            new_n = random.randint(200, 2000)
            new_seq = [random.random() * 5.0 for _ in range(new_n)]
            try:
                new_score = evaluate_sequence(new_seq)
                if new_score < best_score:
                    best_score = new_score
                    best_seq = new_seq
                    n = new_n
                    print(
                        f"[{iteration:05d}] Random restart gave score {new_score:.6f}"
                    )
            except Exception:  # pragma: no cover
                pass

    # ------------------------------------------------------------------
    # 3️⃣  Final report.
    # ------------------------------------------------------------------
    print(f"Finished after {iteration} iterations – best score {best_score:.6f}")
    return best_seq
