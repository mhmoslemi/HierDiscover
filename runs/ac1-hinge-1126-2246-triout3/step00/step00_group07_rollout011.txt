```python
def conv_max(seq):
    n=len(seq)
    # next pow of two for speed maybe.
    size = 1 << (2*n-1).bit_length()
    fft_seq = np.fft.rfft(seq, n=size)
    # auto-convolution via elementwise multiplication
    conv = np.fft.irfft(fft_seq * fft_seq, n=size)
    # conv length size, but we need first 2*n-1 entries
    return np.max(conv[:2*n-1])
```

"""
Simple stochastic hill-climbing optimizer for the step‑function bound.
The algorithm repeatedly:
  * draws a random length n and a random initial non‑negative distribution,
  * computes the autocorrelation (self‑convolution) via FFT,
  * identifies the index k where the convolution attains its maximum,
  * moves a small amount of probability mass away from the pair of indices that
    contribute to that maximum,
  * accepts the move if it improves the objective (or with a Metropolis rule).
The best sequence found before the time budget expires is returned.
"""

import numpy as np
import time

def _objective(a):
    """Return the value of evaluate_sequence for a (sum‑normalised)."""
    a = np.asarray(a, dtype=np.float64)
    n = a.shape[0]
    s = np.sum(a)
    if s < 0.01:
        return np.inf
    # FFT convolution a*a
    size = 1 << ((2 * n - 1).bit_length())
    A = np.fft.rfft(a, n=size)
    b = np.fft.irfft(A * A, n=size)
    max_b = np.max(b[:2 * n - 1])
    return 2.0 * n * max_b / (s * s)

def _make_initial(n):
    """Create a random starting distribution of length n, normalised to 1."""
    # start from uniform and add a small random perturbation
    a = np.full(n, 1.0 / n, dtype=np.float64)
    a += 0.02 * np.random.randn(n)          # noise amplitude
    a = np.clip(a, 0.0, None)
    total = a.sum()
    if total < 1e-12:
        a = np.full(n, 1.0 / n)
        total = a.sum()
    a /= total
    return a

def _hill_climb(a, deadline):
    """Perform a limited stochastic hill‑climb on a, return best found."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    best_seq = a.copy()
    best_score = _objective(a)
    cur = a.copy()
    temperature = 0.5
    it = 0
    while time.time() < deadline and it < 5000:
        # current convolution and worst index
        A = np.fft.rfft(cur, n=size)
        conv = np.fft.irfft(A * A, n=size)
        conv = conv[:2 * n - 1]
        max_b = conv.max()
        k = int(conv.argmax())
        cur_score = 2.0 * n * max_b / (np.sum(cur) ** 2)   # sum is 1
        # keep best
        if cur_score < best_score:
            best_score = cur_score
            best_seq = cur.copy()
        # determine the set of i that participate in the peak k
        i_start = max(0, k - (n - 1))
        i_end = min(n - 1, k)
        active = []
        for i in range(i_start, i_end + 1):
            j = k - i
            if cur[i] > 0 and cur[j] > 0:
                active.append(i)
        if not active:
            # fall back to a random move
            source = np.random.randint(n)
            dest = np.random.randint(n - 1)
            if dest >= source:
                dest += 1
        else:
            i = np.random.choice(active)
            j = k - i
            # choose the larger of the two as the “source” of mass
            if cur[i] >= cur[j]:
                source = i
                partner = j
            else:
                source = j
                partner = i
            # pick a destination that is not the source (could be partner as well)
            dest = np.random.randint(n)
            while dest == source:
                dest = np.random.randint(n)

        # propose moving a small fraction of mass from source to dest
        max_delta = min(cur[source], 0.1 * cur[source])
        if max_delta <= 0:
            it += 1
            continue
        delta = np.random.uniform(0.0, max_delta)
        proposal = cur.copy()
        proposal[source] -= delta
        proposal[dest] += delta

        prop_score = _objective(proposal)

        # Metropolis acceptance
        if prop_score < cur_score or np.random.rand() < np.exp((cur_score - prop_score) / temperature):
            cur = proposal

        # temperature schedule
        it += 1
        if it % 1000 == 0:
            temperature *= 0.90
            # occasional progress print

"""
Simple stochastic hill‑climbing optimizer for the step‑function bound.

The algorithm repeatedly:
  * draws a random length n and a random non‑negative starting vector,
  * evaluates its self‑convolution (via FFT) and finds the index k where the
    convolution attains its maximum,
  * moves a small amount of probability mass away from the pair of indices
    that contribute to that maximum,
  * accepts the move if it improves the objective (or with a Metropolis rule).

All sequences are kept normalised to have sum 1, therefore the objective
2·n·max(b)/(sum²) is invariant under scaling.  Several independent hill‑climbs
are launched within the time budget; the best sequence found is returned.
"""

import numpy as np
import time


def _objective(a: np.ndarray) -> float:
    """Return the value of evaluate_sequence for a (sum‑normalised)."""
    a = np.asarray(a, dtype=np.float64)
    n = a.shape[0]
    s = np.sum(a)
    if s < 0.01:
        return np.inf
    # convolution a*a using a power‑of‑two FFT length
    size = 1 << ((2 * n - 1).bit_length())
    A = np.fft.rfft(a, n=size)
    b = np.fft.irfft(A * A, n=size)
    max_b = np.max(b[: 2 * n - 1])
    return 2.0 * n * max_b / (s * s)


def _make_initial(n: int) -> np.ndarray:
    """Create a random starting distribution of length n, normalised to 1."""
    a = np.full(n, 1.0 / n, dtype=np.float64)
    a += 0.02 * np.random.randn(n)          # small random perturbation
    a = np.clip(a, 0.0, None)
    total = a.sum()
    if total < 1e-12:
        a = np.full(n, 1.0 / n)
        total = a.sum()
    a /= total
    return a


def _hill_climb(start: np.ndarray, deadline: float):
    """
    Perform a short stochastic hill‑climb starting from *start*.
    Returns the best sequence discovered and its score.
    """
    n = start.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    best_seq = start.copy()
    best_score = _objective(start)

    cur = start.copy()
    temperature = 0.5
    it = 0

    while time.time() < deadline and it < 5000:
        # convolution and the worst index
        A = np.fft.rfft(cur, n=size)
        conv = np.fft.irfft(A * A, n=size)
        conv = conv[: 2 * n - 1]
        max_b = conv.max()
        k = int(conv.argmax())
        cur_score = 2.0 * n * max_b / (np.sum(cur) ** 2)   # sum is 1

        # keep the best seen so far
        if cur_score < best_score:
            best_score = cur_score
            best_seq = cur.copy()

        # indices that actually contribute to the peak at k
        i_start = max(0, k - (n - 1))
        i_end = min(n - 1, k)
        active = []
        for i in range(i_start, i_end + 1):
            j = k - i
            if cur[i] > 0.0 and cur[j] > 0.0:
                active.append(i)

        if active:
            i = np.random.choice(active)
            j = k - i
            # take mass from the larger of the two partners
            source = i if cur[i] >= cur[j] else j
        else:
            source = np.random.randint(n)

        # pick a random destination distinct from the source
        dest = np.random.randint(n)
        while dest == source:
            dest = np.random.randint(n)

        # move a modest fraction of the source mass
        max_delta = min(cur[source], 0.1 * cur[source])
        if max_delta <= 0.0:
            it += 1
            continue
        delta = np.random.uniform(0.0, max_delta)

        proposal = cur.copy()
        proposal[source] -= delta
        proposal[dest] += delta

        prop_score = _objective(proposal)

        # Metropolis acceptance
        if prop_score < cur_score or np.random.rand() < np.exp((cur_score - prop_score) / temperature):
            cur = proposal

        # temperature decay and occasional progress report
        it += 1
        if it % 1000 == 0:
            temperature = max(temperature * 0.9, 1e-4)
            print(f"[hill‑climb] iter {it}, cur={cur_score:.6f}, best={best_score:.6f}, n={n}")

    return best_seq, best_score


def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs):
    """
    Search for a step‑function giving a low evaluation score.

    Several short hill‑climbs are run on random initial vectors of
    varying length (300‒1200).  The best sequence found before the time
    budget expires is returned.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5  # small safety margin
    global_best = None
    global_best_score = np.inf
    attempt = 0

    while time.time() < deadline:
        n = np.random.randint(300, 1200)          # a reasonable size range
        start = _make_initial(n)

        # each hill‑climb gets at most 20 s (or whatever remains)
        sub_deadline = time.time() + min(20.0, deadline - time.time())
        seq, score = _hill_climb(start, sub_deadline)

        if score < global_best_score:
            global_best_score = score
            global_best = seq
            print(f"[attempt {attempt}] New global best: score={global_best_score:.6f}, n={n}")

        attempt += 1

    # fallback – uniform distribution of moderate size
    if global_best is None:
        n = 500
        global_best = np.full(n, 1.0 / n)

    # final normalisation (evaluate_sequence will clamp anyway)
    total = np.sum(global_best)
    if total > 0:
        global_best = (global_best / total).tolist()
    else:
        global_best = global_best.tolist()

    return global_best
