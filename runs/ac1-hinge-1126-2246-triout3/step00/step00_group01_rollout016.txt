def solve_convolution_lp(f, M):
    n = len(f)
    rows = []
    cols = []
    data = []
    for k in range(2*n -1):
        # indices i such that 0 <= i < n, 0 <= k-i < n
        i_min = max(0, k-(n-1))
        i_max = min(k, n-1)
        for i in range(i_min, i_max+1):
            j = k - i
            rows.append(k)
            cols.append(j)
            data.append(f[i])
    # Nonnegativity: -I x <= 0
    A_ub = scipy.sparse.csr_matrix((data, (rows, cols)), shape=(2*n-1, n))
    # add -I constraints
    A_ub = scipy.sparse.vstack([A_ub, -scipy.sparse.identity(n)])
    b_ub = np.concatenate([np.full(2*n-1, M), np.zeros(n)])

    result = linprog(c=np.full(n, -1.0), A_ub=A_ub, b_ub=b_ub, bounds=(0, None), method='highs')
    if result.success:
        return result.x
    else:
        return None

def propose_candidate(seed=42, budget_s=1000, **kwargs):
    import numpy as np, time, random
    from scipy.optimize import linprog

    np.random.seed(seed)
    random.seed(seed)

    deadline = time.time() + budget_s - 5

    # Choose sequence length
    n = 500
    target_sum = np.sqrt(2 * n)

    # Initialize sequence: combination of uniform and geometric
    # create start seq as uniform:
    seq = np.full(n, target_sum / n)

    best_seq = seq.copy()
    best_M = np.max(np.convolve(seq, seq))

    def lp_step(f, M):
        """Solve LP to get g maximizing sum while conv(f,g) <= M."""
        n = len(f)
        # Build matrix A_ub of size (2n-1 + n) x n
        rows = []
        cols = []
        data = []

        for k in range(2*n - 1):
            i_min = max(0, k - (n-1))
            i_max = min(k, n-1)
            for i in range(i_min, i_max + 1):
                j = k - i
                rows.append(k)
                cols.append(j)
                data.append(f[i])
        # Convert to dense array for speed (if n small)
        A = np.zeros((2*n - 1 + n, n))
        for r, c, v in zip(rows, cols, data):
            A[r, c] = v
        # add non-negativity rows as -I x <= 0
        A[-n:] = -np.eye(n)
        b = np.concatenate([np.full(2*n - 1, M), np.zeros(n)])

        cvec = -np.ones(n)  # maximize sum -> minimize -sum
        res = linprog(cvec, A_ub=A, b_ub=b, method='highs')
        if res.success:
            return res.x
        else:
            return None

    # Main loop
    iter_cnt = 0
    while time.time() < deadline:
        iter_cnt += 1

        # Normalize current seq to target sum
        sum_seq = np.sum(best_seq)
        if sum_seq == 0:
            best_seq = np.full(n, target_sum / n)
            sum_seq = target_sum

        f_norm = best_seq * (target_sum / sum_seq)

        g = lp_step(f_norm, M)
        if g is None:
            # random perturbation
            i = np.random.randint(n)
            delta = np.random.randn() * 0.01 * target_sum / n

        g_scaled = g * (target_sum / sum_g)

        # compute new M
        M_new = np.max(np.convolve(g_scaled, g_scaled))

        # accept if improvement

            best_seq = g_scaled.copy()

            M_mix = np.max(np.convolve(mix, mix))

            best_seq[i] = max(0.0, best_seq[i] + delta)

"""
Propose a better step‑function sequence using an iterative LP‐based improvement.

The evaluation function for a sequence `a` is  

    score = 2·n·max_k (a*a)[k] / (sum_i a_i)^2 .

If we scale the sequence so that  Σ a_i = √(2 n) the denominator becomes 2 n
and the score equals the maximal convolution value `M = max_k (a*a)[k]`.
Hence the problem reduces to:

    minimise   M
    s.t.       Σ a_i = √(2 n) ,   a_i ≥ 0 ,
               (a*f)[k] ≤ M  for all k   (where f = a).

For a given bound `M` the linear programme

    maximise   Σ b_i
    s.t.       (a*b)[k] ≤ M   ∀k ,   b_i ≥ 0

finds a sequence `b` with the largest possible sum while keeping the
convolution under the same bound `M`.  Scaling `b` back to the required
norm reduces the bound, because the convolution scales quadratically.
Repeatedly solving this LP and re‑normalising produces a monotone
decrease of the maximal convolution value.

The implementation below
* starts from a uniform sequence (sum = √(2 n))
* solves the LP, rescales the solution and accepts it whenever the
  maximal convolution decreases,
* if the LP does not improve the bound it tries a few convex mixes of
  the current sequence and the LP solution,
* falls back to small random perturbations to escape possible stalls.

The loop stops when the allotted time is exhausted (≈ budget_s
seconds) and the best sequence found is returned.
"""

import time
import numpy as np
from scipy.optimize import linprog


def solve_convolution_lp(f_seq: np.ndarray, bound: float) -> np.ndarray | None:
    """
    Solve the linear programme:
        maximise   Σ b_i
        subject to (f_seq * b)[k] ≤ bound   for all k,
                   b_i ≥ 0.

    Returns the optimal `b` (as a NumPy array) or ``None`` if the LP
    fails.
    """
    n = f_seq.shape[0]

    # ----- Build the inequality matrix A_ub·b ≤ b_ub -----
    # one row per convolution index (2*n-1 rows) plus n rows for non‑negativity.
    rows = []
    cols = []
    data = []

    # convolution constraints
    for k in range(2 * n - 1):
        i_min = max(0, k - (n - 1))
        i_max = min(k, n - 1)
        for i in range(i_min, i_max + 1):
            j = k - i               # b_j participates with coefficient f_i
            rows.append(k)
            cols.append(j)
            data.append(f_seq[i])

    # non‑negativity constraints –LP expects them as inequalities
    #   –I·b ≤ 0   (i.e. b_i ≥ 0)
    for i in range(n):
        rows.append(2 * n - 1 + i)
        cols.append(i)
        data.append(-1.0)

    A_ub = np.zeros((2 * n - 1 + n, n), dtype=float)
    A_ub[rows, cols] = data

    b_ub = np.concatenate([np.full(2 * n - 1, bound), np.zeros(n, dtype=float)])

    # maximise Σb_i  → minimise −Σb_i
    c = -np.ones(n, dtype=float)

    # use the high‑performance HiGHS solver
    res = linprog(
        c,
        A_ub=A_ub,
        b_ub=b_ub,
        bounds=(0, None),            # explicitly enforce non‑negativity
        method="highs",
        options={"presolve": True},
    )
    if res.success:
        return res.x
    return None


def max_convolution(seq: np.ndarray) -> float:
    """Return max_k (seq * seq)[k] using FFT for speed."""
    n = seq.shape[0]
    m = 1 << (2 * n - 1).bit_length()   # next power of two
    fft_seq = np.fft.rfft(seq, m)
    conv = np.fft.irfft(fft_seq * fft_seq, m)[: 2 * n - 1]
    return float(conv.max())


def propose_candidate(seed: int = 42, budget_s: float = 1000, **kwargs) -> list[float]:
    """
    Search for a sequence of non‑negative heights that minimises the
    provided `evaluate_sequence` function.  The algorithm uses the LP
    improvement scheme described above and runs until the supplied
    time budget expires.

    Returns the best sequence found (as a plain list of floats).
    """
    np.random.seed(seed)

    # -----------------------------------------------------------------
    # Parameters – length is chosen to keep LPs cheap while still
    # giving a non‑trivial bound (the theoretical optimum tends to 1 as n
    # grows, so n=500 is a good trade‑off).
    # -----------------------------------------------------------------
    n = 500
    target_sum = np.sqrt(2.0 * n)          # after normalisation the score equals max conv
    deadline = time.time() + budget_s - 5  # leave a small safety margin

    best_seq = cur.copy()

    while time.time() < deadline:
        iteration += 1

        # -------------------------------------------------------------
        # 1. Normalise the current candidate to the target sum.
        #    (the LP expects this normalisation)
        # -------------------------------------------------------------
        cur_sum = cur.sum()

            cur = np.full(n, target_sum / n, dtype=float)
            cur_sum = cur.sum()
        cur_norm = cur * (target_sum / cur_sum)

        # current bound M
        M = max_convolution(cur_norm)

        # 2. Solve the LP that maximises the sum while respecting the
        #    same bound M.
        # -------------------------------------------------------------

                best_seq = mixed.copy()
                cur = mixed.copy()

                break

        # 6. Stagnation handling – small random perturbation.
        # -------------------------------------------------------------

        delta = (np.random.rand() - 0.5) * 0.02 * target_sum / n
