def solve_convex_for_n(n, time_budget):
    import cvxpy as cp, numpy as np, time
    start = time.time()
    # variable
    x = cp.Variable(n)
    t = cp.Variable(nonneg=True)
    constraints = [cp.sum(cp.exp(x)) <= 1]
    # build convolution constraints
    for k in range(2 * n - 1):
        i_start = max(0, k - (n - 1))
        i_end = min(n - 1, k)
        expr_terms = [cp.exp(x[i] + x[k - i]) for i in range(i_start, i_end + 1)]
        constraints.append(cp.sum(expr_terms) <= t)
    prob = cp.Problem(cp.Minimize(t), constraints)
    try:
        # use SCS solver; set max_iters based on time_budget
        max_iters = int(2000 * (time_budget / 30.0))  # heuristic
        prob.solve(solver=cp.SCS, max_iters=max_iters, verbose=False)
    except Exception as e:
        print(f"Solver exception: {e}")
        return None, None
    if prob.status not in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE]:
        return None, None
    a = np.exp(x.value)  # array of shape n
    t_val = t.value
    return a, t_val

def gradient_refine(seq, max_iters, step):
    a = np.array(seq, dtype=float)
    n = len(a)
    for it in range(max_iters):
        conv = np.convolve(a, a)
        M = np.max(conv)
        # index of maximum conv
        max_inds = np.where(conv >= M - 1e-12)[0] # tolerance
        # compute gradient g_i = 2 * sum_{k in max_inds} a_{k - i} if within bounds
        g = np.zeros_like(a)
        for k in max_inds:
            i_start = max(0, k - (n - 1))
            i_end = min(n - 1, k)
            # for all i in range i_start..i_end, add 2*a_{k-i}
            # vectorized:
            i_range = np.arange(i_start, i_end + 1)
            g[i_range] += 2 * a[k - i_range]
        S = np.sum(a)
        # compute direction delta = 2*M/S - g_i
        delta = 2 * M / S - g
        # step size schedule
        a = a + step * delta
        # enforce non-negative
        a = np.maximum(0, a)
        # renormalize to maintain sum (any scale) else compute evaluation after adjustments
        # keep sum fixed to previous S to avoid scaling impact
        current_sum = np.sum(a)
        if current_sum > 0:
            a = a * (S / current_sum)
        # optional annealing: reduce step
        step *= 0.995
    return a.tolist()

"""
Propose a step‐function that minimizes

    2·n·max(conv(a,a)) / (sum(a)**2)

The algorithm works in three phases:

1. **Convex relaxation** – using CVXPY with the exponential cone we solve

        min   t
        s.t.  Σ exp(x_i) ≤ 1,
              exp(x_i) ≥ ε   (ensures a non‑zero vector),
              Σ_{i+j=k} exp(x_i+x_j) ≤ t   for all k.

   The decision variables are x_i = log(a_i) so that a_i = exp(x_i) ≥ 0.
   The problem is convex; the optimum tends to make the convolution
   values (almost) equal, i.e. “flat”.  For a set of modest lengths
   (≈150–250) we solve this problem with the SCS solver.

2. **Projected gradient refinement** – starting from the convex solution
   we improve the objective with a simple projected gradient step that
   directly follows the sub‑gradient of the original evaluation function.
   The step keeps the vector non‑negative and renormalises its ℓ₁‑norm,
   which does not affect the evaluation because the metric is scale‑invariant.

3. **Random hill‑climbing fallback** – if time permits, we continue to
   perturb the best vector found so far (small Gaussian noise and occasional
   shuffling of a short segment).  Every improvement is kept.

All operations respect the 1000 s time budget, avoid I/O, and use only
standard scientific Python libraries.
"""

import time
import numpy as np
import random

# ----------------------------------------------------------------------
# Helper: convex optimisation with exponential cone (CVXPY)
# ----------------------------------------------------------------------
def solve_convex_for_n(n, time_budget):
    """
    Solve the convex relaxation for a sequence of length n.
    Returns (a, t) where a = exp(x) and t = max_k Σ a_i a_j for i+j=k.
    If the problem cannot be solved within the allocated time, (None, None) is returned.
    """
    import cvxpy as cp

    # small positive lower bound to avoid the trivial zero solution
    eps = 1e-4
    log_eps = np.log(eps)

    # decision variables
    x = cp.Variable(n)               # x_i = log a_i
    t = cp.Variable(nonneg=True)     # upper bound on all convolution entries

    # constraints
    constraints = [
        cp.sum(cp.exp(x)) <= 1,      # normalise the total mass (not exact)
        x >= log_eps                 # a_i = exp(x_i) ≥ eps  → guarantees sum > 0.01
    ]

    # convolution constraints: for each possible sum index k
    for k in range(2 * n - 1):
        i_start = max(0, k - (n - 1))
        i_end   = min(n - 1, k)
        expr = cp.sum([cp.exp(x[i] + x[k - i]) for i in range(i_start, i_end + 1)])
        constraints.append(expr <= t)

    # minimise the uniform upper bound t
    prob = cp.Problem(cp.Minimize(t), constraints)

    # a crude time‑budget → number of SCS iterations
    # 2000 iterations ≈ 0.5 s for n≈150, scale proportionally
    max_iters = max(500, int(2000 * time_budget / 10.0))

    try:
        prob.solve(solver=cp.SCS, max_iters=max_iters, verbose=False)
    except Exception as e:
        print("  CVX solve exception:", e)
        return None, None

    if prob.status not in (cp.OPTIMAL, cp.OPTIMAL_INACCURATE):
        return None, None

    a = np.exp(x.value)   # the actual coefficients (non‑negative)
    t_val = t.value
    return a, t_val


# ----------------------------------------------------------------------
# Helper: one step of projected gradient descent on the original objective
# ----------------------------------------------------------------------
def gradient_refine(seq, max_iters=2000, step=0.1):
    """
    Perform a simple projected gradient descent on the original metric.
    seq – list/array of non‑negative coefficients.
    Returns a refined sequence (list of floats).
    """
    a = np.array(seq, dtype=float)
    n = a.size

    for it in range(max_iters):
        conv = np.convolve(a, a)
        M = np.max(conv)

        # indices where the convolution attains its maximum (tolerance for numeric noise)
        max_inds = np.where(conv >= M - 1e-12)[0]

        # compute sub‑gradient of max(conv) w.r.t. each a_i:
        #   g_i = Σ_{k∈K} 2 * a_{k-i}   (zero when index out of range)
        g = np.zeros_like(a)
        for k in max_inds:
            i0 = max(0, k - (n - 1))
            i1 = min(n - 1, k)
            i_range = np.arange(i0, i1 + 1)
            g[i_range] += 2.0 * a[k - i_range]

        S = np.sum(a)                     # ℓ₁‑norm (scale‑invariant)
        if S == 0.0:                      # protect against division by zero
            break

        # sub‑gradient of the full objective 2n·M / S²
        delta = 2.0 * M / S - g           # direction that decreases the objective

        # simple gradient step with decreasing step size
        a += step * delta
        a = np.maximum(a, 0.0)           # enforce non‑negativity

        # re‑normalise the ℓ₁‑norm (any scale works – it does not change the score)
        cur_sum = np.sum(a)
        if cur_sum > 0:
            a *= (S / cur_sum)

        step *= 0.995                    # slowly decay the step size

    return a.tolist()

            score = evaluate_sequence(a.tolist())
        except Exception as e:
            print("    → Evaluation error:", e)
            continue

        print(f"    initial score = {score:.6f}")
        if score < best_score:
            best_score = score
            best_seq = a.tolist()

        # --------------------------------------------------------------
        # Phase 2 – gradient refinement
        # --------------------------------------------------------------
        refined = gradient_refine(a, max_iters=2000, step=0.1)
        try:
            refined_score = evaluate_sequence(refined)
        except Exception:
            refined_score = float('inf')
        print(f"    after gradient refinement = {refined_score:.6f}")

        if refined_score < best_score:
            best_score = refined_score
            best_seq = refined

    # --------------------------------------------------------------
    # Phase 3 – random hill‑climbing fallback (if time permits)
    # --------------------------------------------------------------
    if best_seq is None:
        # start from a simple random vector if everything else failed
        n0 = 200
        best_seq = list(np.random.rand(n0))
        best_score = evaluate_sequence(best_seq)
        print(f"[!] Fallback start score = {best_score:.6f}")

    while time.time() < deadline:
        # small random perturbation
        cur = best_seq.copy()
        idx = np.random.randint(len(cur))
        cur[idx] = max(0.0, cur[idx] + np.random.randn() * 0.05)

        # occasionally shuffle a short subsequence to explore combinatorial moves
        if np.random.rand() < 0.05:
            start = np.random.randint(len(cur))
            end = min(len(cur), start + np.random.randint(1, 10))
            segment = cur[start:end]
            np.random.shuffle(segment)
            cur[start:end] = segment

        try:
            cur_score = evaluate_sequence(cur)
        except Exception:
            continue

        if cur_score < best_score:
            best_score = cur_score
            best_seq = cur
            print(f"[+] New best score {best_score:.6f}")

        # stop early if we have already crossed the target threshold
        if best_score < 1.503:
            break

    print(f"[+] Returning best sequence (score {best_score:.6f})")
    return best_seq
