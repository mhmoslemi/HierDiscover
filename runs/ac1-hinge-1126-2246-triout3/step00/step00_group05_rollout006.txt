def simplex_projection(v, S):
    # project onto {x >=0, sum x = S}
    n = len(v)
    # sort v descending
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - S
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0)
    return w
```

```
def simplex_projection(v, s):
    # v: np.ndarray; s: scalar (sum target)
    # Implementation from Duchi et al.
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.where(u - cssv / np.arange(1, n+1) > 0)[0]
    if rho.size == 0:
        # all zero
        theta = 0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0)

"""
gradient_descent_flatten.py

A simple gradient‐descent based optimizer that tries to make the
auto‑convolution of a non‑negative step function as flat as possible.
For a sequence of length *n* we keep the L1‑norm fixed to
√(2 n).  With this normalization the evaluation function reduces to the
maximum entry of the convolution, so minimizing the peak of the
convolution is equivalent to minimizing the score.

The algorithm:
    1. Randomly choose a length n (between 800 and 4000) and initialise a
       non‑negative vector with the required L1‑norm.
    2. Repeatedly:
          * compute the full convolution a∗a
          * compute the error to the desired flat level
            (target = S²/(2n‑1) where S = √(2 n))
          * take the gradient of the squared error
            grad = 4 · corr(error, a)   (valid mode)
          * take a step in the opposite direction and project back onto
            the simplex {x ≥ 0, Σx = S}.
          * occasionally add a small random jitter to escape shallow
            local minima.
          * keep the best sequence seen so far.
    3. The outer loop repeats the whole optimisation with different
       random restarts until the allotted time runs out.

The projection onto the simplex is done with the O(n log n) algorithm
of Duchi et al.  All operations are vectorised with NumPy and the
runtime easily fits into the 1000 s budget.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper: projection of a vector onto the simplex {x >= 0, sum(x)=S}
def simplex_projection(v: np.ndarray, S: float) -> np.ndarray:
    """Project v onto the simplex of non‑negative vectors with sum S."""
    if S < 0:
        raise ValueError("S must be non‑negative")
    n = v.shape[0]
    # sort v in descending order
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - S
    ind = np.arange(1, n + 1)          # 1‑based indexing
    # find the right rho
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all entries zero after projection
        theta = cssv[-1] / n
    else:
        rho = np.where(cond)[0][-1]    # last true index
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Core optimisation routine for a single length n
def optimise_length(n: int,
                    budget_s: float,
                    deadline: float,
                    best_seq: np.ndarray,
                    best_score: float) -> tuple[np.ndarray, float]:
    """
    Run a gradient‑descent optimisation for a fixed length n.
    Returns the best sequence found inside this run and its score.
    """
    S = np.sqrt(2.0 * n)                     # required L1‑norm
    # random start (exponential gives a Dirichlet‑like distribution)
    a = np.random.exponential(scale=1.0, size=n)
    a = a / a.sum() * S                      # enforce the sum

    # parameters for the inner loop
    max_iters = 5000
    step0 = 0.5                              # base step size
    target_val = S * S / (2 * n - 1)         # flat target for each entry

    for it in range(max_iters):
        # stop if we run out of the global time budget
        if time.time() >= deadline:
            break

        # convolution and current score (max entry)
        conv = np.convolve(a, a)
        cur_score = conv.max()               # identical to evaluate_sequence
        if cur_score < best_score:
            best_score = cur_score
            best_seq = a.copy()
            print(f"[{time.time():.1f}] New global best = {best_score:.6f}"
                  f" (n={n}, it={it})")

        # early exit if the convolution is already nearly flat
        err = conv - target_val
        err_norm = np.linalg.norm(err)
        if err_norm < 1e-4:
            # already flat enough; keep the best we have for this length
            break

        # gradient of the squared error (vectorised via correlation)
        # grad_i = 4 * sum_j err_{i+j} * a_j
        grad = 4.0 * np.correlate(err, a, mode='valid')

        # adaptive step: scale inversely with gradient magnitude
        grad_norm = np.linalg.norm(grad) + 1e-12
        step = step0 / (grad_norm * (1.0 + it * 0.001))

        # gradient step + projection onto the simplex
        a = simplex_projection(a - step * grad, S)

        # occasional random jitter to escape plateaus
        if (it + 1) % 250 == 0 and np.random.rand() < 0.3:
            jitter = np.random.randn(n) * step0 * 0.05
            a = simplex_projection(a + jitter, S)

    return best_seq, best_score


# ----------------------------------------------------------------------
