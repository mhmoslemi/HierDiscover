def propose_candidate(seed=123, budget_s=1000):
    np.random.seed(seed)
    start = height_sequence_1 if available else constant.
    # Ensure length at least something. Use length 7521.
    a = np.array(start, dtype=float)
    # Run hill climbing until time budget.
    best_a = a.copy()
    best_score = evaluate_sequence(a.tolist())
    start_time = time.time()
    while time.time() - start_time < budget_s - 5:  # leave a second margin
        # compute convolution
        b = np.convolve(a, a)
        max_b = b.max()
        # find indices where b near max
        k_max = np.argmax(b)
        # gradient contributions
        grad = np.zeros_like(a)
        # For each i compute a[k_max - i] if valid
        for i in range(len(a)):
            j = k_max - i
            if 0 <= j < len(a):
                grad[i] = a[j]
        # choose i to decrease and j to increase
        i_candidates = np.where(a > 1e-8)[0]
        if len(i_candidates) == 0:
            break
        i = i_candidates[np.argmax(grad[i_candidates])]
        # choose j where contribution minimal
        j = np.argmin(grad)  # could be zero
        # ensure not same index
        if i == j:
            # random perturb
            i = np.random.randint(len(a))
            j = np.random.randint(len(a))
        # decide delta
        delta = min(a[i], 0.01 * a[i] + 1e-5)  # small fraction
        # propose new a
        a_new = a.copy()
        a_new[i] -= delta
        a_new[j] += delta
        # compute score
        score = evaluate_sequence(a_new.tolist())
        if score < best_score:
            best_score = score
            best_a = a_new.copy()
            a = a_new
        else:
            # maybe accept with probability if improving other measure?
            # we can also try random walk: sometimes accept worse to escape
            if np.random.rand() < 0.001:  # rare acceptance
                a = a_new
    # after loop return best_a
    return best_a.tolist()

The algorithm works in three stages:

1. **Coarse random search** – sample many geometric decay rates
   a_i = r**i (r close to 1) and keep the best one.  The shape is
   scaled to the canonical sum √(2 n) (the same scaling used by the
   reference method) because the evaluation is scale‑invariant.

2. **Gradient‐descent style refinement** – the current sequence is
   improved by moving a small amount of mass away from the indices that
   contribute the most to the current maximal convolution value.
   For the index k where the convolution attains its maximum we have
   ∂(a*f)_k / ∂a_i = a_{k‑i}; moving a_i in the opposite direction
   reduces the peak.  The step size is chosen so that no entry becomes
   negative, then the whole vector is re‑scaled to keep the sum equal to
   √(2 n).

3. **Escapes & random perturbations** – if a gradient step does not
   improve the score we perform a tiny random mass transfer between two
   random positions or add a small Gaussian noise.  These moves help
   escape plateaus.

The whole loop runs until the time budget (default 1000 s) expires.
The best sequence found is returned as a plain Python list.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper functions
# ----------------------------------------------------------------------


def geometric_sequence(n, r, target_sum):
    """Return a vector a_i = r**i scaled so that sum(a)=target_sum."""
    a = np.power(r, np.arange(n, dtype=float))
    # avoid all‑zero vector when r is tiny
    if a.sum() == 0:
        a = np.ones(n, dtype=float)
    a *= target_sum / a.sum()
    return a


def max_convolution_index(seq):
    """Return (max_value, list_of_indices) of convolution seq*seq."""
    conv = np.convolve(seq, seq)
    max_val = conv.max()
    idxs = np.where(conv >= max_val - 1e-12)[0]
    return max_val, idxs, conv


def gradient_for_max(seq, k):
    """
    Sub‑gradient of the max‑convolution term b_k = Σ a_i a_{k‑i}
    w.r.t. the vector a.  Returns a vector of the same shape.
    """
    n = len(seq)
    i = np.arange(n)
    j = k - i
    mask = (j >= 0) & (j < n)
    g = np.zeros_like(seq)
    g[mask] = seq[j[mask]]
    return g


def random_transfer(seq, eps=1e-4):
    """Transfer a tiny amount of mass between two random positions."""
    n = len(seq)
    i = np.random.randint(n)
    # ensure we take from a non‑zero entry
    if seq[i] < eps:
        return seq
    j = np.random.randint(n)
    delta = min(seq[i] * 0.01, eps * np.mean(seq))
    new_seq = seq.copy()
    new_seq[i] -= delta
    new_seq[j] += delta
    return new_seq


def perturb(seq, scale=0.02):
    """Add small Gaussian noise and re‑normalise."""
    n = len(seq)
    noise = np.random.randn(n) * scale * np.mean(seq)
    new_seq = np.maximum(seq + noise, 0.0)
    if new_seq.sum() == 0:
        new_seq = np.ones_like(seq)
    return new_seq


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------


def propose_candidate(seed=42, budget_s=1000, **kwargs):
    """
    Return a list of non‑negative heights (the step function) that
    attempts to minimise evaluate_sequence within the given time budget.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5   # a small safety margin

    # ------------------------------------------------------------------
    # Determine the problem size
    # ------------------------------------------------------------------
    if "height_sequence_1" in globals():
        n = len(height_sequence_1)
    else:
        # fallback length – the reference solution uses 7521
        n = 7521

    target_sum = np.sqrt(2.0 * n)          # same normalisation as the SOTA code

    # ------------------------------------------------------------------
    # 1) Coarse random search over geometric decays
    # ------------------------------------------------------------------
    best_seq = geometric_sequence(n, r=0.9, target_sum=target_sum)
    best_score = evaluate_sequence(best_seq.tolist())
    print(f"[init] length={n}, initial score={best_score:.6f}")

    # sample many r values; we focus on r very close to 1 because the
    # theory predicts the optimum there.
    for _ in range(150):
        # sample a parameter X = 2n*(1-r) uniformly in a useful interval
        X = np.random.uniform(0.5, 8.0)          # empirical range
        r = 1.0 - X / (2.0 * n)
        if r <= 0.0:
            continue
        cand = geometric_sequence(n, r, target_sum)
        score = evaluate_sequence(cand.tolist())
        if score < best_score:
            best_score = score
            best_seq = cand
            print(f"[rand] new best (r={r:.6f}) score={best_score:.6f}")

    # ------------------------------------------------------------------
    # 2) Gradient‑descent like refinement
    # ------------------------------------------------------------------
    seq = best_seq.copy()

    while time.time() < deadline:
        # compute convolution and the current maximal index
        max_val, max_idxs, conv = max_convolution_index(seq)
        k = np.random.choice(max_idxs)            # pick one active index
        grad = gradient_for_max(seq, k)

        # Determine the maximal admissible step that keeps all entries ≥ 0
        pos = grad > 1e-14
        if np.any(pos):
            max_step = np.min(seq[pos] / grad[pos])
            # we take a conservative fraction of that step
            step = max_step * np.random.uniform(0.3, 0.7)
        else:
            step = 0.0

        # Build a candidate by moving opposite to the gradient
        cand = seq - step * grad
        cand = np.maximum(cand, 0.0)               # numerical safety
        # re‑scale to keep the canonical sum
        if cand.sum() == 0:
            # avoid division by zero – re‑initialise randomly
            cand = np.random.rand(n)
        cand *= target_sum / cand.sum()

        cand_score = evaluate_sequence(cand.tolist())

        if cand_score < best_score:
            best_score = cand_score
            best_seq = cand.copy()
            print(f"[grad] improved score={best_score:.6f}  time={time.time()-start_time:.1f}s")
        # accept the candidate if it improves the *current* sequence,
        # otherwise keep the old one (simple hill‑climb)
        if cand_score < evaluate_sequence(seq.tolist()):
            seq = cand
        else:
            # occasional random escape moves
            if np.random.rand() < 0.02:
                seq = perturb(seq, scale=0.05)

        # also try a tiny random transfer now and then
        if np.random.rand() < 0.05:
            transfer_seq = random_transfer(seq)
            transfer_seq *= target_sum / transfer_seq.sum()
            tr_score = evaluate_sequence(transfer_seq.tolist())
            if tr_score < best_score:
                best_score = tr_score
                best_seq = transfer_seq.copy()
                print(f"[rand‑trans] new best score={best_score:.6f}")
            if tr_score < evaluate_sequence(seq.tolist()):
                seq = transfer_seq

    # ------------------------------------------------------------------
    # Return the best sequence found (as plain Python list)
    # ------------------------------------------------------------------
    print(f"[final] best score={best_score:.6f} after {time.time()-start_time:.1f}s")
    return best_seq.tolist()
