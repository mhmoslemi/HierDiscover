def project_simplex(v, s=1.0):
    """Project vector v onto the simplex {x >= 0, sum x = s}."""
    n = len(v)
    # shift to ensure sum > s? Not needed.
    # Algorithm:
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.where(u - cssv / np.arange(1, n+1) > 0)[0]
    if len(rho) == 0:
        # all nonpositive, return uniform distribution
        return np.full(n, s / n)
    rho = rho[-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0)
    return w

def propose_candidate(seed=0, budget_s=1000, **kwargs):
    np.random.seed(seed)
    import time, numpy as np
    deadline = time.time() + budget_s - 5
    n = 1500
    # Initialize a: random Dirichlet with concentration 0.5 maybe.
    # We'll generate random positive numbers and normalize.
    a = np.random.rand(n)
    a = a / a.sum()
    # optionally apply smoothing? Not needed.
    best_score = evaluate_sequence(a.tolist())
    best_seq = a.copy()
    print(f"Initial evaluate: {best_score}")
    iteration = 0
    # step schedule
    base_eta = 10.0  # maybe too large? We'll adapt.
    while time.time() < deadline:
        iteration += 1
        # compute conv
        fft_len = 2 * n - 1
        A = np.fft.rfft(a, fft_len)
        b = np.fft.irfft(A * A, fft_len)
        # find peak index
        k_star = int(np.argmax(b))
        max_b = b[k_star]
        # compute gradient vector
        # Compute reversed a
        # Since g_j = 2*a[k_star - j] if 0 <= k_star - j < n else 0
        low = max(0, k_star - (n - 1))
        high = min(n - 1, k_star)
        # indices of j: low..high
        idx = k_star - np.arange(low, high + 1)
        g = np.zeros_like(a)
        g[low:high+1] = 2 * a[idx]
        # update step size
        # Use diminishing step: eta = base_eta / np.sqrt(iteration)
        eta = base_eta / np.sqrt(iteration)
        # multiplicative update
        a_new = a * np.exp(-eta * g)
        # normalize
        a_new_sum = a_new.sum()
        if a_new_sum == 0:  # all zeros due to huge eta, fallback
            a_new = np.random.rand(n)
            a_new_sum = a_new.sum()
        a_new = a_new / a_new_sum
        # occasional random perturb to avoid stagnation
        if iteration % 5000 == 0:
            # add small Gaussian noise to a fraction
            mask = np.random.rand(n) < 0.1
            noise = np.random.randn(mask.sum()) * 0.01
            a_new[mask] += noise
            # clip negative
            a_new = np.maximum(a_new, 0)
            if np.sum(a_new) == 0:
                a_new = np.random.rand(n)
            a_new = a_new / a_new.sum()
        a = a_new
        # Evaluate current
        score = evaluate_sequence(a.tolist())
        if score < best_score:
            best_score = score
            best_seq = a.copy()
            print(f"Iter {iteration}: score {best_score:.6f}")
            # Early exit if hitting target
            if best_score < 1.503:
                break
    print(f"Finished: best {best_score}")
    return best_seq.tolist()

"""
Propose a candidate step‑function (list of non‑negative heights).

The algorithm builds a probability vector `a` of length 1500 (sum ≈ 1) and
optimises it to minimise the quantity

    score = 2 * n * max(conv(a, a))

which is exactly the value returned by ``evaluate_sequence`` when the sum of
the coefficients equals 1.  The objective is therefore to flatten the
autocorrelation (the discrete convolution of the sequence with itself).

Approach
--------
* Initialise `a` with a random positive vector (or the global
  ``height_sequence_1`` if it is available) and normalise it to sum = 1.
* Repeatedly:
    – Compute the full convolution via an FFT (O(n log n)).
    – Identify a small set of indices where the convolution attains its
      largest values (the “active” constraints).
    – Form a sub‑gradient of the max‑convolution by averaging the exact
      gradients of the active constraints.  For a particular index `k`

        ∂(a*a)_k / ∂a_j = 2 * a_{k‑j}   (if the index exists, otherwise 0).

    – Perform a mirror‑descent / exponentiated‑gradient step:

        a ← a * exp(‑η * g)      (element‑wise)

      and renormalise to keep the sum equal to 1.
    – Occasionally inject small random perturbations to escape
      plateaux.

* Keep the best vector seen so far (lowest score) and, after the main
  optimisation loop, run a short hill‑climbing phase that swaps small
  amounts of mass between random positions.
* The loop is bounded by the supplied time budget (default ≈ 1000 s).

The method is completely self‑contained, uses only NumPy and the
provided ``evaluate_sequence`` for the final check, and respects the
non‑negativity and sum‑> 0.01 constraints.
"""

import time
import numpy as np

def _project_to_simplex(v, s=1.0):
    """Project a vector onto the probability simplex {x ≥ 0, sum x = s}."""
    if s < 0:
        raise ValueError("Simplex radius must be non‑negative")
    n = v.shape[0]
    # Sort v in descending order
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    ind = np.arange(n) + 1
    # Find the largest rho where u_i > cssv_i / i
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # All entries will be zero; return uniform point
        return np.full(n, s / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0)
    # Numerical safety: renormalise exactly
    w_sum = w.sum()
    if w_sum == 0:
        w = np.full(n, s / n)
    else:
        w = w * (s / w_sum)
    return w
