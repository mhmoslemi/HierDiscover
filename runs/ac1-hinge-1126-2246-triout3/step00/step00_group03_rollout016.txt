def solve_lp_for_sum(f_seq, bound):
   # compute constraints
   n = len(f_seq)
   c = -np.ones(n)  # maximize sum => minimize -sum
   # Build inequality A_ub * b <= b_ub
   # For each k in 0..2n-2:
   rows = []
   b_vals = []
   for k in range(2*n-1):
       row = np.zeros(n)
       # For j in range(n):
       low = max(0, k - (n-1))
       high = min(n-1, k)
       # For each j in [low, high], i = k - j
       # row[j] = f_seq[i]
       # let indices j be in [low, high]; set row[j] = f_seq[k-j]
       # vectorised:
       # row[low:high+1] = f_seq[k - np.arange(low, high+1)]
       idx = np.arange(low, high+1)
       row[idx] = f_seq[k - idx]
       rows.append(row)
       b_vals.append(bound)
   # Add non-negativity constraints: -I * b <= 0
   rows.append(-np.eye(n))
   b_vals.append(np.zeros(n))
   A_ub = np.vstack(rows)
   b_ub = np.concatenate(b_vals)
   result = linprog(c, A_ub=A_ub, b_ub=b_ub, method='highs', options={'time_limit': 1.0})
   if result.success:
        return result.x
   else:
        return None

"""
Proposes a step‑function height sequence by iteratively
* normalising the current sequence to sum = sqrt(2·n) (so the
  evaluation equals the maximal convolution value),
* solving a linear programme that finds another non‑negative
  sequence with the same convolution bound but maximal L1‑sum,
* mixing the two sequences (convex combination) which never
  increases the convolution bound,
* adapting the mixing step size and occasionally injecting random
  perturbations or restarting from a random seed.
The loop runs until the allotted time expires and returns the
best sequence (lowest value of `evaluate_sequence`) discovered.
"""

def solve_convolution_lp(f_seq, rhs):
    """
    Solve: maximise sum(b)  subject to  conv(f_seq, b) <= rhs,  b >= 0.
    Returns the optimal b (numpy array) or None on failure.
    """
    n = len(f_seq)
    # objective: minimise -sum(b)  <=> maximise sum(b)
    c = -np.ones(n)

    # build the convolution matrix for the inequalities conv(f,b) <= rhs
    rows = []
    bounds = []

    for k in range(2 * n - 1):
        # indices j with 0 <= k-j < n
        low = max(0, k - (n - 1))
        high = min(n - 1, k)

        row = np.zeros(n)
        row[j] = f_seq[k - j]
        rows.append(row)
        bounds.append(rhs)

    # non‑negativity constraints: -b <= 0
    rows.append(-np.eye(n))
    bounds.append(np.zeros(n))

    b_ub = np.concatenate(bounds)

    # use HiGHS (default) – fast for typical sizes we use
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, method='highs')
    if res.success:
        return res.x
    return None

"""
Proposes a step‑function height sequence that minimises the supplied
`evaluate_sequence` function.

The algorithm follows the idea from the reference implementation
(Linear‑Programming based “direction” updates) but adds several
enhancements:

* Randomised choice of the sequence length (200 – 800) – short enough
  to keep each LP cheap while still allowing a rich search space.
* Adaptive mixing coefficient `t` – it is increased after a successful
  improvement and decreased after a failure.
* Periodic random perturbations and occasional full‑vector jitter to
  escape plateaus.
* Multiple independent restarts (fresh random initialisations) until
  the time budget expires.
* A small “fallback” to the previously discovered construction
  (`height_sequence_1`) if no better sequence is found.

Each iteration:

1. Normalise the current sequence to have L1‑norm `sqrt(2·n)`; with this
   normalisation the evaluation value equals the maximal
   convolution entry `M = max(f * f)`.
2. Compute `M` and solve the LP

       maximise   Σ b_j
       subject to  conv(f, b) ≤ M ,   b ≥ 0

   which yields a new feasible sequence `g` with the same convolution
   bound but larger L1‑norm.
3. Normalise `g` to the same L1‑norm, then take a convex combination
   `new = (1‑t)·f + t·g`.  Because the convolution operator is bilinear,
   this combination never exceeds the bound `M`, and for a suitable
   `t` it typically reduces the maximal entry.
4. Accept the new sequence if the evaluation improves; otherwise shrink
   `t` and optionally apply a small random jitter.

The loop runs until the allocated wall‑clock time is exhausted and the
best sequence (lowest evaluation) encountered is returned.
"""

def solve_convolution_lp(f_seq, rhs):

        maximise   Σ b_j

    rows = []

    bounds.append(np.zeros(n))

    b_ub = np.concatenate(bounds)
