"""
Propose a sequence of non‑negative heights that minimizes the evaluation function.

The function performs a simple stochastic search:
* The trivial 2‑element optimum (ratio 2 : 1) gives a score of 16/9 ≈ 1.777.
* Random probability vectors are sampled (Dirichlet distribution) for
  lengths 2‑12.  Each sample is locally refined by a few gradient‑descent
  steps that try to reduce the maximal convolution value.
* The best sequence found during the allotted time is returned.

The algorithm is lightweight, uses only NumPy and the already‑imported
`evaluate_sequence`, and runs comfortably within the 1000 s budget.
"""

import time
import random
import numpy as np

# ----------------------------------------------------------------------
def _convolution_max(seq: np.ndarray) -> float:
    """Return max of the full convolution seq * seq."""
    return np.max(np.convolve(seq, seq))


def _normalize_to_one(seq: np.ndarray) -> np.ndarray:
    """Scale a non‑negative vector so that its sum equals 1 (if sum>0)."""
    s = seq.sum()
    if s < 1e-12:
        # fallback – shouldn't happen because we keep sum>0.01 globally
        return np.full_like(seq, 1.0 / len(seq))
    return seq / s


def _local_descent(seq: np.ndarray,
                   max_iters: int = 12,
                   step0: float = 0.025) -> np.ndarray:
    """
    Perform a few gradient‑like updates aiming to shrink the current
    maximal convolution entry.

    At each iteration we locate the index k where the convolution attains
    its maximum, compute the (sub)gradient of that entry (2*a_{k-i}) and
    move a little in the opposite direction.  After the move we clip to
    non‑negative values and renormalise to keep the sum equal to 1.
    """
    a = seq.copy()
    n = len(a)
    for it in range(max_iters):
        # current convolution and its maximising shift
        conv = np.convolve(a, a)
        k = int(np.argmax(conv))

        # sub‑gradient of b_k = Σ_i a_i a_{k-i}
        grad = np.zeros(n)
        for i in range(n):
            j = k - i
            if 0 <= j < n:
                grad[i] = 2.0 * a[j]            # 2*a_{k-i}
        # step size decays with iteration number
        step = step0 / (1.0 + it)
        a_new = a - step * grad
        a_new = np.maximum(a_new, 0.0)
        a = _normalize_to_one(a_new)

    return a


def _random_dirichlet(n: int, alpha: float) -> np.ndarray:
    """Generate a random probability vector of length n (Dirichlet)."""
    # numpy's dirichlet already returns a sum‑to‑1 vector.
    return np.random.dirichlet([alpha] * n)


# ----------------------------------------------------------------------
def propose_candidate(budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a good step‑function sequence.

    Parameters
    ----------
    budget_s : int
        Time budget in seconds (default 1000).  The function stops a few
        seconds before the deadline to avoid premature termination.

    Returns
    -------
    list[float]
        The best sequence discovered (list of non‑negative floats).
    """
    start_time = time.time()
    deadline = start_time + budget_s - 3.0   # safety margin

    # ------------------------------------------------------------------
    # 1.  Known analytic optimum for n = 2  (ratio 2 : 1)
    best_seq = [2.0, 1.0]                     # any scaling works
    best_score = evaluate_sequence(best_seq)
    print(f"[Init] start score (n=2 optimum) = {best_score:.6f}")

    # ------------------------------------------------------------------
    # 2.  Stochastic search over small lengths
    #    The search space is tiny (n ≤ 12) so we can afford many trials.
    trial = 0
    while time.time() < deadline:
        trial += 1
        # random length – at least 2, at most 12 (larger lengths give
        # diminishing returns for this simple search)
        n = random.randint(2, 12)

        # draw a Dirichlet sample; the concentration parameter α controls
        # how spiky the vector is.  We randomise α to explore both uniform
        # and highly non‑uniform regions.
        alpha = random.expovariate(1.0) + 0.05   # α in (0.05, …)
        vec = _random_dirichlet(n, alpha)

        # local refinement (few gradient steps)
        vec = _local_descent(vec, max_iters=15, step0=0.03)

        # convert to plain Python list for the evaluator
        cand = vec.tolist()
        try:
            score = evaluate_sequence(cand)
        except Exception:
            # defensive – evaluate_sequence is robust, but guard anyway
            continue

        if score < best_score:
            best_score = score
            best_seq = cand[:]
            print(f"[{trial:06d}] New best: {best_score:.6f}  (n={n})")
        # occasional random perturbation to escape plateaus
        if trial % 2500 == 0:
            # re‑seed a little (helps reproducibility across runs)
            random.seed(random.randint(0, 1 << 30))

    # ------------------------------------------------------------------
    # 3.  Final report
    elapsed = time.time() - start_time
    print(f"Search finished after {elapsed:.1f}s, best score = {best_score:.6f}")
    print(f"Best sequence length = {len(best_seq)}")
    return best_seq
