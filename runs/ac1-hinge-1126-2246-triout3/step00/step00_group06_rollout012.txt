def propose_candidate(seed=42, budget_s=1000, n=1500, **kwargs):
   np.random.seed(seed)
   start = time.time()
   deadline = start + budget_s - 5

   # initialize sequence
   idx = np.arange(n) + 1.0
   seq = 1.0 / np.sqrt(idx)
   seq = seq / seq.sum()  # sum=1

   # (Optional) random perturbation to break symmetry?
   # first few operations can be random jitter
   seq += np.random.randn(n) * 1e-6
   seq = np.maximum(seq, 0.0)
   seq = seq / seq.sum()

   best_seq = seq.copy()
   best_score = evaluate_sequence(seq.tolist())
   curr_seq = seq
   curr_score = best_score

   # parameters
   total_time = budget_s - 5
   init_temp = 0.1
   final_temp = 1e-6

   iteration = 0

   while time.time() < deadline:
        elapsed = time.time() - start
        # linear schedule for temperature
        T = init_temp * (final_temp / init_temp) ** (elapsed / total_time)

        # choose mutation type
        if np.random.rand() < 0.2:
            # Dirichlet redistribution
            m = np.random.randint(2, 6)
            idxs = np.random.choice(n, m, replace=False)
            total_mass = curr_seq[idxs].sum()
            # draw new proportions
            new_vals = np.random.dirichlet(np.ones(m)) * total_mass
            new_seq = curr_seq.copy()
            new_seq[idxs] = new_vals
        else:
            # transfer mass between two indices
            i = np.random.randint(n)
            j = np.random.randint(n-1)
            if j >= i:
                j = j + 1
            # suggest delta
            max_delta = curr_seq[i] * 0.5
            delta = (np.random.rand() * 2 - 1) * max_delta * (T / init_temp)
            # clamp to keep non-negative
            if delta > curr_seq[i]:
                delta = curr_seq[i]
            if delta < -curr_seq[j]:
                delta = -curr_seq[j]
            new_seq = curr_seq.copy()
            new_seq[i] -= delta
            new_seq[j] += delta

        # evaluate new candidate
        try:
            new_score = evaluate_sequence(new_seq.tolist())
        except Exception:
            continue

        # Acceptance criteria: Simulated annealing
        if new_score < curr_score or np.random.rand() < np.exp(-(new_score - curr_score) / max(T, 1e-12)):
            curr_seq = new_seq
            curr_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_seq = new_seq.copy()
                print(f"Iter {iteration}: new best score {best_score:.6f}")

        # occasional renormalization to avoid drift
        if iteration % 1000 == 0 and iteration > 0:
            # renormalize to sum 1
            s = best_seq.sum()
            if s > 0:
                best_seq = best_seq / s
                curr_seq = curr_seq / s

        iteration += 1

   # final return as list
   return best_seq.tolist()

"""
Propose a high‑quality step‑function for the given evaluation metric.

Approach
--------
* Initialise the sequence with the discretised density   
      f(x) ∝ 1/√x   (the known minimiser of the L∞‑norm of the
      self‑convolution in the continuous setting).  After normalisation
      the list has sum 1, which is scale‑invariant for the score.
* Repeatedly improve the sequence with a Simulated‑Annealing loop.
  Three kinds of moves are used:
    – *Max‑reduction move*: pick the index `k` where the convolution is
      maximal and transfer a small amount of mass from the pair
      (i, j) with i+j=k that contributes most to that peak towards the
      currently smallest entry.  This directly lowers the peak.
    – *Dirichlet reshuffle*: choose a few random positions,
      keep their total mass and redraw a Dirichlet distribution on them.
      Provides larger, global perturbations.
    – *Two‑coordinate transfer*: move a random amount of mass from one
      coordinate to another, useful for fine exploration.
* Acceptance follows a temperature schedule (SA): always accept
  improvements, otherwise accept with probability `exp(-Δ/ T)`.
* The best sequence encountered is returned.  The whole loop respects
  the 1000 s budget.

The algorithm works entirely with `numpy` and does not rely on external
files or network access.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities

def _score(seq: np.ndarray) -> float:
    """Fast evaluation of the objective (identical to evaluate_sequence)."""
    n = seq.size
    # convolution of the non‑negative sequence with itself
    conv = np.convolve(seq, seq)
    max_b = conv.max()
    sum_a = seq.sum()
    if sum_a < 0.01:
        return np.inf
    return float(2.0 * n * max_b / (sum_a * sum_a))


def _initial_sequence(n: int) -> np.ndarray:
    """Discretised version of f(x) = 1/(2√x) on [0,1], normalised to sum 1."""
    # avoid division by zero for the first element
    idx = np.arange(1, n + 1, dtype=np.float64)
    seq = 1.0 / np.sqrt(idx)                # ∝ 1/√i
    seq /= seq.sum()                        # normalise to sum 1
    return seq


def _max_reduction_move(seq: np.ndarray, eps_factor: float) -> np.ndarray:
    """
    Reduce the current maximal convolution value.
    Find the pair (i,j) that contributes the most to the peak and move a
    small amount of mass from both towards the smallest coordinate.
    """
    n = seq.size
    conv = np.convolve(seq, seq)
    k = int(conv.argmax())                  # position of the peak
    # all pairs (i, j) with i+j = k
    i_vals = np.arange(n)
    j_vals = k - i_vals
    mask = (j_vals >= 0) & (j_vals < n)
    i_vals = i_vals[mask]
    j_vals = j_vals[mask]

    # product for each pair (i, j)
    prod = seq[i_vals] * seq[j_vals]
    # pick the pair with the largest contribution
    best = int(prod.argmax())
    i = int(i_vals[best])
    j = int(j_vals[best])

    # amount to move – a small fraction of the smaller entry
    eps = min(seq[i], seq[j]) * eps_factor
    if eps <= 0.0:
        return seq.copy()

    # destination: the (currently) smallest coordinate (not i or j)
    r = int(np.argmin(seq))
    if r == i or r == j:
        # pick the second smallest if needed
        candidates = np.argsort(seq)
        for cand in candidates:
            if cand != i and cand != j:
                r = int(cand)
                break

    new_seq = seq.copy()
    new_seq[i] -= eps
    new_seq[j] -= eps
    new_seq[r] += 2.0 * eps               # keep the total sum unchanged
    # numerical safety – clip tiny negatives that may appear due to rounding
    np.maximum(new_seq, 0.0, out=new_seq)
    # renormalise to avoid drift (preserves scale‑invariance)
    new_seq /= new_seq.sum()
    return new_seq


def _dirichlet_move(seq: np.ndarray) -> np.ndarray:
    """Redistribute mass among a small random subset of indices."""
    n = seq.size
    m = np.random.randint(2, 7)               # size of the subset (2‑6)
    idx = np.random.choice(n, m, replace=False)
    total = seq[idx].sum()
    if total == 0.0:
        return seq.copy()
    # new proportions – Dirichlet(1,…,1) → uniform on the simplex
    new_vals = np.random.dirichlet(np.ones(m)) * total
    new_seq = seq.copy()
    new_seq[idx] = new_vals
    return new_seq


def _transfer_move(seq: np.ndarray, scale: float) -> np.ndarray:
    """Move a random amount of mass from one coordinate to another."""
    n = seq.size
    i = np.random.randint(n)
    j = np.random.randint(n - 1)
    if j >= i:
        j += 1                         # guarantee i != j
    # allowed transfer is limited by the source amount
    max_delta = seq[i] * 0.5
    delta = (np.random.rand() * 2.0 - 1.0) * max_delta * scale
    # clip to keep both entries non‑negative
    delta = np.clip(delta, -seq[j], seq[i])
    new_seq = seq.copy()
    new_seq[i] -= delta
    new_seq[j] += delta
    # renormalise (tiny drift may appear)
    new_seq /= new_seq.sum()
    return new_seq

# Main entry point required by the evaluation harness
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: int = 1000, *, n: int = 1500,
                     **kwargs) -> list[float]:
    """
    Return a sequence of non‑negative heights (length `n`) that minimises
    the provided evaluation function.  The function runs a simulated‑annealing
    search for up to `budget_s` seconds and returns the best sequence found.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 5.0      # a little safety margin

    # ----- initialise with the near‑optimal continuous solution -----
    current = _initial_sequence(n)
    best_seq = current.copy()
    best_score = _score(best_seq)
    cur_score = best_score

    # ----- SA schedule -----
    total_time = budget_s - 5.0
    T_start = 1.0e-1
    T_end = 1.0e-6

    iteration = 0
    while time.time() < deadline:
        elapsed = time.time() - start_time
        # exponential cooling
        T = T_start * (T_end / T_start) ** (elapsed / total_time)

        # choose a move type
        r = np.random.rand()
        if r < 0.40:
            # aggressive reduction of the current peak
            eps_fac = 0.05 * (T / T_start)    # shrink as temperature cools
            cand = _max_reduction_move(current, eps_fac)
        elif r < 0.70:
            # a medium‑scale Dirichlet reshuffle
            cand = _dirichlet_move(current)
        else:
            # small two‑coordinate transfer
            cand = _transfer_move(current, scale=T / T_start)

        cand_score = _score(cand)

        # SA acceptance rule
        if cand_score < cur_score:
            accept = True
        else:
            prob = np.exp(-(cand_score - cur_score) / max(T, 1e-12))
            accept = np.random.rand() < prob

        if accept:
            current = cand
            cur_score = cand_score
            if cand_score < best_score:
                best_seq = cand.copy()
                best_score = cand_score
                print(f"Iter {iteration:8d}: new best = {best_score:.6f}")

        # occasional clean‑up (renormalise to avoid accumulated drift)
        if iteration % 2000 == 0 and iteration > 0:
            current /= current.sum()
            best_seq /= best_seq.sum()
            cur_score = _score(current)

        iteration += 1

    # Return a plain Python list as required by the external harness
    return best_seq.tolist()
