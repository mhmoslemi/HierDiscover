def propose_candidate(seed=0, budget_s=1000):
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5

    n = 1500  # choose appropriate length; may adapt.
    # start from uniform or some random.
    a = np.full(n, 1.0/n)

    # compute objective
    best_a = a.copy()
    best_score = evaluate_sequence(best_a.tolist())
    print('initial score', best_score)

    # Precompute indices pair contributions for each k? Might be heavy.

    step = 0.1  # fraction of typical amplitude (to be tuned)
    iteration = 0
    while time.time() < deadline:
        conv = np.convolve(a, a)
        max_k = int(np.argmax(conv))
        max_val = conv[max_k]

        if delta <= 0:

        k = np.random.randint(n)
        # make a copy and update
        a_new = a.copy()
        a_new[i] -= delta
        a_new[j] -= delta

        # ensure nonnegativity
        if a_new[i] < 0 or a_new[j] < 0:

            best_a = a_new.copy()

        if score <= np.inf and score <= (best_score + 1e-12):

def project_simplex(v):
    # ensure sum(v) = 1, v_i >=0
    if v.sum() == 1 and np.all(v>=0):
        return v
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(n) + 1
    rho = np.where(u - cssv / ind > 0)[0][-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0)
    return w
```

"""
propose_candidate
-----------------

A stochastic “max‑convolution flattening” search.

The evaluation function is

    F(a) = 2·n·max(conv(a,a)) / (sum(a)^2) ,

where `conv` is the ordinary (full) convolution of a non‑negative sequence
`a`.  The value is invariant under scaling of the whole vector, therefore we
keep the L1‑norm equal to one during the search.

The algorithm works as follows

1.  Try several sequence lengths (small to medium, up to a few thousand).
2.  Initialise a candidate with a random Dirichlet (exponential) draw.
3.  Repeatedly locate the index `k` where the convolution is maximal.
4.  Among the pairs *(i , j = k‑i)* contributing to that maximal entry,
    pick the pair with the largest product `a_i·a_j`.  Move a small
    fraction of the mass from both `a_i` and `a_j` to a third random index.
    This reduces the dominant convolution entry while preserving the sum.
5.  Occasionally perform a tiny random jitter to escape plateaus.
6.  Keep the best sequence seen so far (lowest value of
    ``evaluate_sequence``).  The loop stops when the time budget expires.

The method is completely self‑contained, uses only ``numpy`` and the provided
``evaluate_sequence`` function, and finishes well within the 1000 s limit.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper: project a vector onto the probability simplex (sum = 1, non‑neg.)
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto {x | x_i >= 0, sum(x)=1}."""
    n = v.shape[0]
    # sort v in descending order
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(n) + 1
    # find the rho index
    rho = np.where(u - cssv / ind > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Core improvement step – try to lower the current maximal convolution entry
# ----------------------------------------------------------------------
def _reduce_max_convolution(a: np.ndarray, step_factor: float) -> np.ndarray:
    """
    One deterministic perturbation aimed at decreasing the current maximal
    convolution value.

    Parameters
    ----------
    a : np.ndarray
        Current probability vector (sum = 1, all entries >= 0).
    step_factor : float
        Relative amount of mass to move from the most damaging pair.

    Returns
    -------
    np.ndarray
        Modified vector (still sums to 1, non‑negative).
    """
    n = a.shape[0]
    conv = np.convolve(a, a)
    k = int(np.argmax(conv))

    # Find the pair (i, j) with j = k-i that gives the biggest product a_i*a_j
    best_i = -1
    best_prod = -1.0
    for i in range(n):
        j = k - i
        if 0 <= j < n:
            prod = a[i] * a[j]
            if prod > best_prod:
                best_prod = prod
                best_i = i
                best_j = j

    # If the convolution peak is caused only by a single element, we cannot
    # move mass from both sides – fall back to a small random jitter.
    if best_i == -1 or best_prod <= 0.0:
        return a

    i, j = best_i, best_j

    # Amount of probability mass we are allowed to take from i and j
    delta = min(a[i], a[j]) * step_factor
    if delta <= 1e-15:
        return a

    # Destination index (different from i and j)
    m = np.random.randint(n)
    while m == i or m == j:
        m = np.random.randint(n)

    # Perform the transfer
    a_new = a.copy()
    a_new[i] -= delta
    a_new[j] -= delta
    a_new[m] += 2.0 * delta
    # Numerical noise may cause a tiny negative; clip and re‑project.
    a_new = np.maximum(a_new, 0.0)
    a_new = _project_simplex(a_new)
    return a_new


# ----------------------------------------------------------------------
# Main entry point required by the grader
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: float = 1000, **kwargs):
    """
    Search for a non‑negative step function (list of heights) that yields a
    low value of ``evaluate_sequence``.  The routine respects the given time
    budget and returns the best sequence found.
    """
    np.random.seed(seed)
    deadline = time.time() + float(budget_s) - 0.1   # leave a tiny safety margin

    best_seq = None
    best_score = float('inf')

    # Lengths to explore – from very small up to a few thousand.
    lengths = [16, 32, 64, 128, 256, 512, 1024, 2048]

    for n in lengths:
        if time.time() >= deadline:
            break

        # ---- random start -------------------------------------------------
        a = np.random.exponential(scale=1.0, size=n)
        a = _project_simplex(a)                     # forces sum = 1
        cur_score = evaluate_sequence(a.tolist())
        if cur_score < best_score:
            best_score = cur_score
            best_seq = a.copy()
            print(f"[Init] n={n:4d} score={cur_score:.6f}")

        # ---- optimisation loop -------------------------------------------
        step_factor = 0.25          # initial fraction of mass to move
        it = 0
        # Number of inner iterations is limited by the remaining time.
        while time.time() < deadline and it < 50000:
            # main deterministic move
            a = _reduce_max_convolution(a, step_factor)

            # occasional tiny random jitter (helps escape plateaus)
            if np.random.rand() < 0.03:
                idx = np.random.randint(n)
                eps = (np.random.rand() - 0.5) * step_factor * a[idx]
                a[idx] = max(a[idx] + eps, 0.0)
                a = _project_simplex(a)

            # evaluate occasionally (every few steps) – cheap enough
            if it % 5 == 0:
                cur_score = evaluate_sequence(a.tolist())
                if cur_score < best_score:
                    best_score = cur_score
                    best_seq = a.copy()
                    print(f"[Update] n={n:4d} it={it:5d} score={cur_score:.6f}")
                # simple cooling schedule for the step size
                if it % 500 == 0 and it > 0:
                    step_factor = max(step_factor * 0.9, 1e-4)

            it += 1

        # end of this length – move on to the next one

    # If nothing succeeded (should not happen) fall back to a simple 2‑point
    # sequence which gives the theoretical upper bound 2.0.
    if best_seq is None:
        best_seq = np.array([0.5, 0.5])

    # Return a plain Python list as required by the grader.
    return best_seq.tolist()
