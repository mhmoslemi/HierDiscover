"""
Propose a step‑function (list of non‑negative heights) that minimizes the
evaluation metric

    2 * n * max(conv(a, a)) / (sum(a)^2)

The idea is to keep the convolution as flat as possible by using a sparse
set of indices whose pairwise sums collide only a few times.
For a set S ⊂ {0,…,n‑1} let a_i = 1/|S| for i∈S and 0 otherwise.
Then

    max(conv(a,a)) = (max #ordered pairs with the same sum) / |S|^2

so minimizing the bound reduces to building a large set with a small
maximum collision count.

We greedily construct such a set for different allowed collision limits
(max_c = 2,3,4,5) and keep the best sequence found within the time budget.
The construction works as follows:

* Choose a random ordering of the indices 0…n‑1.
* Scan the ordering, adding an index i to the set if, for every already
  chosen j, the ordered‑pair count for the sum i+j would stay ≤ max_c.
  The self‑pair i+i contributes only 1 to the count.
* The procedure is repeated many times with different random orders;
  the largest feasible set encountered is kept.

The resulting sequence is simply the indicator of that set scaled to
unit sum (weight = 1/|S|).  Because scaling cancels in the evaluation,
any non‑zero scaling works, but unit sum avoids the “sum too small”
guard in evaluate_sequence().

The algorithm is lightweight (O(n·|S|) per attempt) and can explore
thousands of random attempts within the 1000 s budget, giving a good
chance of reaching a set size >200 with max_c = 4, which yields a bound
well below the target 1.503.
"""

import time
import math
import numpy as np
from scipy import optimize

# ----------------------------------------------------------------------
def _greedy_set(n: int, max_c: int, rng: np.random.Generator,
                target_k: int | None = None) -> tuple[list[int], dict[int, int]]:
    """
    Greedy construction of a set S ⊂ {0,…,n‑1} such that the ordered
    pair‑sum count never exceeds ``max_c``.
    Returns (S, count_dict) where ``count_dict[s]`` is the number of
    ordered pairs (i,j)∈S×S with i+j = s.
    """
    perm = rng.permutation(n)
    S = []
    cnt = {}                     # sum → ordered‑pair count
    for i in perm:
        # early abort if we already reached the requested size
        if target_k is not None and len(S) >= target_k:
            break

        # check the self‑pair i+i (contributes +1)
        s_self = i + i
        if cnt.get(s_self, 0) + 1 > max_c:
            continue

        # check all pairs with already selected indices (contribute +2)
        feasible = True
        for j in S:
            s = i + j
            if cnt.get(s, 0) + 2 > max_c:
                feasible = False
                break
        if not feasible:
            continue

        # add i to the set and update the counters
        for j in S:
            s = i + j
            cnt[s] = cnt.get(s, 0) + 2
        cnt[s_self] = cnt.get(s_self, 0) + 1
        S.append(i)
    return S, cnt


def _evaluate_from_set(n: int, S: list[int], max_c_observed: int) -> float:
    """
    Compute the evaluation value for the uniform‑weight indicator of set S.
    """
    k = len(S)
    if k == 0:
        return float('inf')
    # each ordered pair contributes (1/k)^2, max contribution = max_c/k^2
    return 2.0 * n * max_c_observed / (k * k)


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for the best sequence within the given time budget.
    Returns a list of floats (the heights).
    """
    # ------------------------------------------------------------------
    # Global data – the reference sequence is available as height_sequence_1.
    # Use its length as the problem size.
    if 'height_sequence_1' in globals():
        n = len(height_sequence_1)
    else:
        # Fallback: a moderately large size if the reference is missing.
        n = 5000

    rng = np.random.default_rng(seed)
    deadline = time.time() + budget_s - 0.5   # leave a small safety margin

    best_score = float('inf')
    best_seq : list[float] | None = None

    # Try several collision limits.  Larger limits allow larger sets.
    for max_c in (2, 3, 4, 5):
        # Upper bound on achievable set size for this max_c (theoretical).
        max_k = int(math.sqrt(max_c * (2 * n - 1)))
        # Aim for a high fraction of this bound.
        target_k = int(0.92 * max_k)

        print(f"[{time.strftime('%H:%M:%S')}] max_c={max_c}, "
              f"theoretical max_k={max_k}, target_k≈{target_k}")

        attempts = 0
        # Keep trying until time runs out or we already beat the target bound.
        while time.time() < deadline:
            attempts += 1
            S, cnt = _greedy_set(n, max_c, rng, target_k=None)

            if not S:
                continue

            # Observed maximum collision count (should be ≤ max_c).
            observed_c = max(cnt.values()) if cnt else 0
            if observed_c > max_c:
                # Should never happen because of the check, but be safe.
                continue

            score = _evaluate_from_set(n, S, observed_c)

            # Build the actual sequence (uniform weight on the set).
            weight = 1.0 / len(S)
            seq = [0.0] * n

                seq[idx] = weight

            if true_score < best_score:

            if len(S) >= target_k and best_score < 2.0:
                # No need to waste more time on smaller attempts.
                break

        # If we already meet the target, skip remaining max_c values.
        if best_score < 1.503:
            break

    # ------------------------------------------------------------------
    # Fallback: if for some reason no set was built, return a uniform vector.
    if best_seq is None:
        print("Fallback to uniform sequence.")
        best_seq = [1.0] * n

    return best_seq
