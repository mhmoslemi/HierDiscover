        # start with random positive entries
        a = np.random.rand(n)
        # optional: ensure sum not too small
        if np.sum(a) < 0.1:
            a += 0.1
        # maybe multiply by random scaling to spread values within [0,1]
        # Then gradient descent loop
        num_steps = np.random.randint(50, 200)
        lr = np.random.uniform(0.001, 0.1)
        for step in range(num_steps):
            # compute conv
            c = np.convolve(a, a)
            max_idx = np.argmax(c)
            max_c = c[max_idx]
            sum_a = a.sum()
            # gradient
            shift = max_idx - np.arange(len(a))
            v = np.zeros_like(a)
            mask = (shift >= 0) & (shift < len(a))
            v[mask] = a[shift[mask]]
            grad = 4 * len(a) * (sum_a * v - max_c) / (sum_a**3)
            # adapt learning rate maybe decays
            a = a - lr * grad
            # enforce nonnegativity
            a = np.maximum(a, 0)
            # renormalize to avoid drift (optional)
            sum_a_new = a.sum()
            if sum_a_new > 1e-12:
                a = a * (sum_a / sum_a_new)
            # optionally add small random perturbation to escape flat region
            if step % 10 == 0:
                # small jitter
                jitter = np.random.randn(len(a)) * 0.001
                a = np.maximum(a + jitter, 0)
                # renorm
                a = a * (sum_a / a.sum())
            # compute score
            # maybe compute every few steps
        # after steps, compute final score
        cur_score = evaluate_sequence(a.tolist())
        if cur_score < best_score:
            best_score = cur_score
            best_seq = a.tolist()
            print(f"New best {best_score:.6f} at n={n}")

    return best_seq

def gradient_step(a, lr):
    n = len(a)
    conv = np.convolve(a, a)
    max_idx = int(np.argmax(conv))
    max_c = conv[max_idx]
    sum_a = a.sum()
    # compute v array
    i = np.arange(n)
    j = max_idx - i
    mask = (j >= 0) & (j < n)
    v = np.zeros_like(a)
    v[mask] = a[j[mask]]
    # grad of F as derived
    grad = 4 * n * (sum_a * v - max_c) / (sum_a**3)

"""
search_convolution.py
---------------------

*Goal*: Find a non‑negative step‑function (a list of heights) that minimises

        evaluate_sequence(seq) = 2·n·max(conv(seq,seq)) / (sum(seq)**2)

where *n* is the length of the sequence.  The score of the trivial uniform
sequence is exactly ``2`` for any length, so any improvement must flatten the
auto‑convolution – i.e. make all convolution values as close to the average as
possible.

*Algorithm* (high‑level)

1. **LP sub‑step** –  For the current sequence `f` we normalise it so that
   ``sum(f) = sqrt(2·n)`` (scaling does not affect the final score).  With this
   normalised vector we solve a linear programme that maximises the L1‑mass of
   a new vector ``g`` under the convolution bound

        (f * g)[k] ≤ M   for every shift k,
        g_i ≥ 0,

   where ``M`` is the current maximum of ``f*f``.  The LP is linear because
   ``(f*g)[k]`` is linear in the unknowns `g_i`.

2. **Mixing / line‑search** –  The vectors `f` and the LP‑solution `g`
   (both normalised to the same L1‑norm) are combined:

        h(t) = (1‑t)·f + t·g ,      t ∈ [0,1].

   Several candidate values of ``t`` are tried (including a few relatively
   large steps) and the one giving the smallest evaluation score is kept.
   This simple line‑search replaces the fixed step size used in the reference
   implementation and lets the optimiser take much bigger safe steps when
   possible.

3. **Exploration** –  If the LP fails (it can happen for pathological
   vectors) a tiny random perturbation is applied to stay away from dead‑ends.
   Occasionally a completely fresh random start is performed to avoid being
   trapped in a bad local basin.

4. **Loop** –  The procedure repeats until the allotted time is exhausted.
   The best sequence ever seen is remembered and finally returned.

The whole search runs entirely with NumPy / SciPy and respects the
``evaluate_sequence`` security checks (non‑negative, reasonable magnitude,
etc.).  No file or network access is performed.
"""

import time
import numpy as np
from scipy.optimize import linprog

# ----------------------------------------------------------------------
# Helper: solve the linear programme  max sum(g)  s.t.  (f * g)[k] ≤ M , g ≥ 0
# ----------------------------------------------------------------------
def solve_convolution_lp(f_norm: np.ndarray, bound: float):
    """
    Solve the LP for a fixed *f_norm* (already normalised to have
    sum = sqrt(2·n)).  ``bound`` is the current maximum of the convolution
    ``f_norm * f_norm``.
    Returns the optimal vector *g* or ``None`` if the solver fails.
    """
    n = f_norm.shape[0]

    # Build the matrix A where A[k, j] = f_norm[k‑j]  (or 0 if out of range)
    rows = np.arange(2 * n - 1)[:, None]          # shape (2n‑1,1)
    cols = np.arange(n)                           # shape (1,n)
    diff = rows - cols                             # (2n‑1, n)
    mask = (diff >= 0) & (diff < n)
    A = np.zeros((2 * n - 1, n), dtype=float)
    A[mask] = f_norm[diff[mask]]

    # RHS is the same bound for every convolution inequality
    b = np.full(2 * n - 1, bound, dtype=float)

    # Objective: maximise sum(g)  <=> minimise -sum(g)
    c = -np.ones(n, dtype=float)

    # Variables are constrained to be non‑negative via ``bounds``
    bounds = [(0, None)] * n

    # Use the high‑performance HiGHS solver; give it a modest time limit
    res = linprog(c, A_ub=A, b_ub=b, bounds=bounds,
                  method='highs', options={'time_limit': 5.0, 'disp': False})

    if res.success:
        return res.x
    return None


# ----------------------------------------------------------------------
# Build a new candidate by LP + line‑search
# ----------------------------------------------------------------------
def lp_step_and_mix(curr_seq: np.ndarray):
    """
    Perform one optimisation step:
      * solve the LP for the current sequence,
      * try several mixing factors t and keep the best mixed vector.
    Returns the best mixed vector and its score (or (None, inf) on failure).
    """
    n = curr_seq.shape[0]
    sum_curr = curr_seq.sum()
    if sum_curr < 1e-12:
        return None, np.inf

    # Normalise so that sum = sqrt(2·n)  (scaling does not affect the true score)
    scale = np.sqrt(2.0 * n) / sum_curr
    f_norm = curr_seq * scale

    # Current convolution bound M
    conv_f = np.convolve(f_norm, f_norm)
    bound = conv_f.max()

    # Solve the LP
    g_raw = solve_convolution_lp(f_norm, bound)
    if g_raw is None:
        return None, np.inf

    # Re‑normalise g to the same L1‑norm as f_norm
    sum_g_raw = g_raw.sum()
    if sum_g_raw < 1e-12:
        return None, np.inf
    g_norm = g_raw * (np.sqrt(2.0 * n) / sum_g_raw)

    # ------------------------------------------------------------------
    # Line‑search over mixing coefficients
    # ------------------------------------------------------------------
    best_local_seq = None
    best_local_score = np.inf
    best_t = 0.0

    # Candidate t values – include a few aggressive steps
    t_candidates = [0.0, 0.02, 0.05, 0.1, 0.2, 0.5]

    for t in t_candidates:
        mixed = (1.0 - t) * curr_seq + t * g_norm
        # ``evaluate_sequence`` expects a plain Python list
        try:
            score = evaluate_sequence(mixed.tolist())
        except Exception:
            # Should not happen – guard against unexpected NaNs etc.
            score = np.inf
        if score < best_local_score:
            best_local_score = score
            best_local_seq = mixed
            best_t = t

    return best_local_seq, best_local_score

    # --------------------------------------------------------------
    # Starting point – use the known construction if available,
    # otherwise fall back to a short uniform vector.
    # --------------------------------------------------------------
    if 'height_sequence_1' in globals():
        best_seq = np.array(height_sequence_1, dtype=float)
    else:
        best_seq = np.ones(200, dtype=float)

    best_score = evaluate_sequence(best_seq.tolist())
    print(f'Initial score = {best_score:.6f} (length = {len(best_seq)})')

    # Main optimisation loop
    while time.time() < deadline:
        # Occasionally start from a fresh random vector to increase diversity
        if np.random.rand() < 0.12:
            n = np.random.randint(300, 1200)          # moderate size – fast LPs
            # Ensure a positive total mass
            cur_seq = np.random.rand(n) + 0.01
        else:
            cur_seq = best_seq.copy()

        # Perform a small number of inner optimisation steps
        inner_steps = np.random.randint(5, 15)
        for _ in range(inner_steps):
            new_seq, new_score = lp_step_and_mix(cur_seq)

            # If the LP failed we inject a tiny random tweak and retry
            if new_seq is None:
                idx = np.random.randint(len(cur_seq))
                cur_seq[idx] = max(0.0, cur_seq[idx] + np.random.randn() * 0.01)
                continue

            # Accept the locally best mixed sequence
            cur_seq = new_seq

            # Update the global best if we improved
            if new_score < best_score:
                best_score = new_score
                best_seq = cur_seq.copy()
                print(f'New global best = {best_score:.6f} '
                      f'(len={len(best_seq)}) at t‑mix step')

        # Tiny global jitter every few iterations to help escape stagnation
        if np.random.rand() < 0.05:
            cur_seq += np.random.randn(*cur_seq.shape) * 0.003
            cur_seq = np.clip(cur_seq, 0.0, None)

    print(f'Finished – best score found: {best_score:.6f} (length = {len(best_seq)})')
    # ``evaluate_sequence`` requires a plain list, not a NumPy array
    return best_seq.tolist()
