"""
Fast heuristic for minimizing the evaluation function.

The search uses three ingredients:

1.  A good analytic start – the discrete analogue of the
    continuous density a(x)=1/√x (β≈‑0.5).  For a given length n
    we build a sequence a_i = (i+γ)^β and normalise it to sum to 1.
    This already yields a score ≈π/2≈1.57.

2.  A simple sub‑gradient step that reduces the current maximal
    convolution value.  If k* is the index where the full
    convolution a*a attains its maximum, the sub‑gradient of
    max(b) with respect to a is
        g_i = 2·a_{k*‑i}   (0≤k*‑i<n) else 0.
    Moving a small amount of mass from indices with large g
    to indices with small g flattens the convolution and lowers the
    score.

3.  Random “re‑starts’’ and occasional random mass transfers keep
    the search from getting stuck in a local minimum.

The algorithm works for any length (the default is the length of the
provided `height_sequence_1`).  It respects the time budget (default
1000 s) and always returns the best sequence found.
"""

import time
import numpy as np
import random

# ----------------------------------------------------------------------
# Helper functions (no closures, no lambdas – required by the rules)
# ----------------------------------------------------------------------


def evaluate_fast(seq: np.ndarray) -> float:
    """Fast replica of evaluate_sequence for a NumPy 1‑D array."""
    # the official evaluator protects against NaN/Inf and clamps, we do the same
    seq = np.where(np.isnan(seq) | np.isinf(seq), 0.0, seq)
    seq = np.clip(seq, 0.0, 1000.0)
    n = seq.size
    # convolution (full) – NumPy uses an efficient FFT for long vectors
    conv = np.convolve(seq, seq)
    max_b = np.max(conv)
    sum_a = np.sum(seq)
    if sum_a < 0.01:
        return float("inf")
    return float(2.0 * n * max_b / (sum_a * sum_a))


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of vector v onto the probability simplex:
        w_i ≥ 0 , Σ w_i = 1
    Implements the algorithm of Duchi et al. (2008).
    """
    if np.all(v >= 0) and abs(v.sum() - 1.0) < 1e-12:
        return v.copy()
    n = v.size
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # fallback – uniform distribution
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


def power_law_sequence(n: int, beta: float, gamma: float) -> np.ndarray:
    """Generate a non‑negative sequence a_i = (i+γ)^β and normalise to sum 1."""
    # avoid zero division for very negative betas
    idx = np.arange(n, dtype=np.float64) + gamma
    seq = np.power(idx, beta, where=idx > 0, out=np.zeros_like(idx))
    total = np.sum(seq)
    if total == 0.0:
        # fall back to uniform
        return np.full(n, 1.0 / n)
    return seq / total


def subgradient_move(seq: np.ndarray, step: float) -> np.ndarray:
    """
    Perform a single sub‑gradient move that reduces the maximal
    convolution value.
    """
    n = seq.size
    conv = np.convolve(seq, seq)
    k_max = int(np.argmax(conv))

    # build the sub‑gradient g_i = 2 * a_{k_max-i} (if index valid)
    g = np.zeros_like(seq)

    if k_max < n:
        # indices 0 … k_max have partners
        i = np.arange(k_max + 1)
        g[i] = 2.0 * seq[k_max - i]
    else:
        # indices k_max-n+1 … n-1 have partners
        start = k_max - (n - 1)
        i = np.arange(start, n)
        g[i] = 2.0 * seq[k_max - i]

    # move a little amount of mass from large‑g to small‑g positions
    i_high = int(np.argmax(g))
    i_low = int(np.argmin(g))

    # amount to transfer – a fraction of the source entry
    delta = step * seq[i_high]
    if delta <= 0.0:
        return seq  # nothing to do

    new_seq = seq.copy()
    new_seq[i_high] -= delta
    new_seq[i_low] += delta

    # preserve feasibility
    new_seq = np.maximum(new_seq, 0.0)
    s = new_seq.sum()
    if s == 0.0:
        return seq
    new_seq /= s
    return new_seq


def random_mass_transfer(seq: np.ndarray, scale: float) -> np.ndarray:
    """
    Randomly pick two indices and transfer a small amount of mass.
    Scale controls the typical size of the transfer (relative to the
    source entry).
    """
    n = seq.size
    i, j = np.random.choice(n, 2, replace=False)
    if seq[i] == 0.0:
        return seq
    delta = scale * np.random.rand() * seq[i]
    new_seq = seq.copy()
    new_seq[i] -= delta
    new_seq[j] += delta
    new_seq = np.maximum(new_seq, 0.0)
    s = new_seq.sum()
    if s == 0.0:
        return seq
    new_seq /= s
    return new_seq

    returns the best sequence discovered.
    """

    random.seed(seed)
    np.random.seed(seed)

    start_time = time.time()
    deadline = start_time + budget_s - 5.0      # small safety margin

        n = 3536

    # ------------------------------------------------------------------
    beta0 = -0.5
    gamma0 = 0.5
    cur_seq = power_law_sequence(n, beta0, gamma0)
    cur_score = evaluate_fast(cur_seq)

    # Keep the best sequence ever seen
    best_seq = cur_seq.copy()
    best_score = cur_score

    # Occasionally also keep the very first (possibly uniform) sequence
    uniform_seq = np.full(n, 1.0 / n)
    uniform_score = evaluate_fast(uniform_seq)
    if uniform_score < best_score:
        best_seq = uniform_seq.copy()
        best_score = uniform_score

    # ------------------------------------------------------------------
    # Parameters for the sub‑gradient / random walk
    # ------------------------------------------------------------------
    step = 0.02                # initial fraction of source mass to move

            # if we improve, gently increase the step size
