def propose_candidate(seed=0, budget_s=1000):
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5
    best_seq = None
    best_score = float('inf')
    iteration = 0

    while time.time() < deadline:
        # pick random length n
        n = np.random.randint(200, 2000)
        # random initial p from Dirichlet
        p = np.random.rand(n)
        p = p / p.sum()
        # small perturb to break symmetry
        # local search iterations
        local_iter = 0
        # schedule for learning rate:
        base_eta = 0.5
        max_local_iters = 2000
        while time.time() < deadline and local_iter < max_local_iters:
            # compute conv and max
            conv = np.convolve(p, p)
            max_val = conv.max()
            # maybe evaluate early
            score = evaluate_sequence(p.tolist())
            if score < best_score:
                best_score = score
                best_seq = p.tolist()
                print(f"New best {best_score:.6f} at n={n} iter={iteration}")
            # gradient ok?
            # find indices where conv close to max
            eps = 1e-12
            max_indices = np.where(np.abs(conv - max_val) < eps)[0]
            # pick first index
            k = max_indices[0]
            # compute gradient
            g = np.zeros(n)
            # due to symmetrical contributions, compute for each i where 0 <= k-i < n: 
            # g[i] = 2 * p[k-i]
            low = max(0, k - (n-1))
            high = min(n-1, k)
            # i ranges low..high
            # j = k - i
            i_range = np.arange(low, high+1)
            j_range = k - i_range
            g[i_range] = 2 * p[j_range]
            # learning rate decreasing
            eta = base_eta / (1 + local_iter * 0.001)
            # multiplicative update
            p = p * np.exp(-eta * g)
            # add small random noise to break symmetry
            p += 0.001 * np.random.rand(n)
            # project onto simplex: ensure sum=1
            p = p / p.sum()
            # optional: enforce non-negativity (already)
            local_iter += 1
        iteration += 1
    # after loop, return best_seq as list
    return best_seq

def simplex_projection(v):
    # ensures non-negative and sum to 1.
    # from https://arxiv.org/pdf/1101.6081.pdf
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0)
    return w
```

"""
Propose a step‑function (list of non‑negative heights) that minimizes the
`evaluate_sequence` objective.

Algorithm
---------

1. **Deterministic seeds** – For several lengths `n` we construct a few
   analytically motivated probability vectors:
   * Uniform distribution (baseline).
   * Harmonic “U‑shape’’  p_i ∝ 1/(i+1) + 1/(n‑i) .
   * Arcsine‑type      p_i ∝ 1/√((i+0.5)*(n‑i‑0.5)) .
   These are normalised to sum = 1 (the objective is invariant under global
   scaling).  Their scores are evaluated and the best candidate is stored.

2. **Random restarts + sub‑gradient descent** – While time permits we
   repeatedly
   * pick a random length `n` (200 … 2000),
   * initialise a random probability vector,
   * perform a projected sub‑gradient descent on the objective
     `max_k (p*p)[k]` (the ℓ∞‑norm of the convolution).  
     For the current maximal index `k*` the sub‑gradient is
     `g_i = 2·p_{k*‑i}` (zero when the index is out of range).  
     We take a diminishing step `α = α0/√(t+1)` and project the result onto
     the probability simplex (non‑negative entries that sum to 1).  A tiny
     random jitter is added each iteration to avoid stagnation.
   * After each iteration the score `2·n·max(p*p)` is computed; the globally
     best vector is kept.

3. **Early stopping** – If a sequence reaches a score ≤ 1.51 (the target
   1.503 ± few ·10⁻³) we stop immediately.

The whole procedure respects the 1000 s budget and returns the best
sequence found.  All numbers are non‑negative and well within the clipping
limits of `evaluate_sequence`.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper: projection onto the probability simplex (sum = 1, all ≥ 0)
def simplex_projection(v):
    """Project a vector v onto the simplex {x ≥ 0, Σx = 1}."""
    v = np.asarray(v, dtype=float)
    n = v.shape[0]
    # sort descending
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    # find the largest rho where u_i - cssv_i / i > 0
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all entries are zero – return uniform
        return np.full_like(v, 1.0 / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w

# ----------------------------------------------------------------------
def max_convolution(p):
    """Return max_k (p * p)[k] for a non‑negative vector p (sum(p) = 1)."""
    conv = np.convolve(p, p)
    return conv.max()

# ----------------------------------------------------------------------
def subgradient_descent(p, n, best_score, deadline,
                        max_iters=3000,
                        alpha0=0.5,
                        jitter=1e-4):
    """
    Perform a few iterations of projected sub‑gradient descent on the
    ℓ∞‑norm of the convolution of p with itself.
    Returns (best_local_p, best_local_score).
    """
    cur_score = 2.0 * n * max_convolution(p)
    best_local_p = p.copy()
    best_local_score = cur_score

    it = 0
    while time.time() < deadline and it < max_iters:
        # current convolution and its maximal index
        conv = np.convolve(p, p)
        max_val = conv.max()
        k = int(np.argmax(conv))

        # sub‑gradient of the max‑convolution entry
        g = np.zeros_like(p)
        low = max(0, k - (n - 1))
        high = min(n - 1, k)
        i_range = np.arange(low, high + 1, dtype=int)
        j_range = k - i_range
        g[i_range] = 2.0 * p[j_range]

        # diminishing step size
        alpha = alpha0 / np.sqrt(it + 1.0)

        # gradient step + projection onto simplex
        p_new = simplex_projection(p - alpha * g)

        # tiny random jitter to escape flat regions
        p_new += jitter * np.random.rand(n)
        p_new /= p_new.sum()

        # evaluate (fast) score using the known formula
        new_score = 2.0 * n * max_convolution(p_new)

        # accept the new point if it improved the objective; otherwise keep it
        # (the projection already moves us in the descent direction, so we
        #  normally stay close to the optimum even without a strict acceptance test)
        p = p_new
        cur_score = new_score

        if new_score < best_local_score:
            best_local_score = new_score
            best_local_p = p_new.copy()

        # early exit if we already beat the target
        if best_local_score <= 1.51:
            break

        it += 1

    return best_local_p, best_local_score

# ----------------------------------------------------------------------
def propose_candidate(seed=42, budget_s=1000, **kwargs):
    """
    Search for a sequence of non‑negative heights that minimises
    `evaluate_sequence`.  Returns the best sequence found before the time
    budget expires.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5   # small safety margin

    best_seq = None
    best_score = float("inf")

    # ------------------------------------------------------------------
    # 1) deterministic analytic seeds (uniform, harmonic, arcsine)
    # ------------------------------------------------------------------
    lengths = [200, 400, 800, 1200, 1600, 2000]
    for n in lengths:
        if time.time() >= deadline:
            break

        # uniform
        p_uni = np.full(n, 1.0 / n)
        score_uni = 2.0 * n * max_convolution(p_uni)
        if score_uni < best_score:
            best_score = score_uni
            best_seq = p_uni.tolist()
            print(f"[seed] uniform n={n} score={best_score:.6f}")

        # harmonic U‑shape: p_i ∝ 1/(i+1) + 1/(n-i)
        i = np.arange(n, dtype=float)
        w_harm = 1.0 / (i + 1.0) + 1.0 / (n - i)
        p_harm = w_harm / w_harm.sum()
        score_harm = 2.0 * n * max_convolution(p_harm)
        if score_harm < best_score:
            best_score = score_harm
            best_seq = p_harm.tolist()
            print(f"[seed] harmonic n={n} score={best_score:.6f}")

        # arcsine‑type shape: p_i ∝ 1/√((i+0.5)*(n-0.5-i))
        x = (i + 0.5) / n            # scaled to (0,1)
        w_arc = 1.0 / np.sqrt(x * (1.0 - x))
        p_arc = w_arc / w_arc.sum()
        score_arc = 2.0 * n * max_convolution(p_arc)
        if score_arc < best_score:
            best_score = score_arc
            best_seq = p_arc.tolist()
            print(f"[seed] arcsine n={n} score={best_score:.6f}")

        # early stop if we already hit the desired bound
        if best_score <= 1.51:
            break

    # ------------------------------------------------------------------
    # 2) random restarts with sub‑gradient descent
    # ------------------------------------------------------------------
    while time.time() < deadline:
        # pick a random length (favor medium‑large sizes)
        n = np.random.randint(200, 2001)

        # random initial probability vector (Dirichlet-like)
        p = np.random.rand(n)
        p = p / p.sum()

        # a quick evaluation to see if this restart is promising
        cur_score = 2.0 * n * max_convolution(p)
        if cur_score < best_score:
            best_score = cur_score
            best_seq = p.tolist()
            print(f"[restart] initial improve n={n} score={best_score:.6f}")

        # sub‑gradient descent on this start point
        p_opt, score_opt = subgradient_descent(p, n, best_score, deadline)

        if score_opt < best_score:
            best_score = score_opt
            best_seq = p_opt.tolist()
            print(f"[update] new global best n={n} score={best_score:.6f}")

        # stop early if target met
        if best_score <= 1.51:
            break

    # ------------------------------------------------------------------
    # 3) safety fallback (uniform of moderate size) – should never be used
    # ------------------------------------------------------------------
    if best_seq is None:
        n = 500
        best_seq = [1.0 / n] * n
        print("[fallback] using uniform fallback")

    # final sanity check – ensure list of floats is returned
    return list(map(float, best_seq))
