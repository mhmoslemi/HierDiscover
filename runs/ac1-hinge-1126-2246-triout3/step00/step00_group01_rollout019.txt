def max_convolution(seq):
    n = len(seq)

    f = np.fft.rfft(seq, L)
    conv = np.fft.irfft(f * f, L)[:2*n-1]
    return np.max(conv)

def propose_candidate(seed=0, budget_s=1000):
    import numpy as np, time, math
    np.random.seed(seed)
    n = len(height_sequence_1)  # existing length
    target_sum = math.sqrt(2*n)
    # start with uniform
    seq = np.full(n, target_sum / n, dtype=np.float64)
    # compute initial values
    sum_sq = target_sum**2
    # Precompute conv length for FFT
    L = 1 << ((2*n - 1).bit_length())
    # function to compute max convolution
    def max_conv(s):
        fft_s = np.fft.rfft(s, L)
        conv = np.fft.irfft(fft_s * fft_s, L)[:2*n-1]
        return conv.max(), conv
    max_b, conv = max_conv(seq)
    best_seq = seq.copy()
    best_score = 2 * n * max_b / sum_sq
    best_m = max_b
    start = time.time()
    deadline = start + budget_s - 2
    it = 0
    while time.time() < deadline:
        # locate index with max conv
        k = int(np.argmax(conv))
        # compute weight vector w for each index i
        # w[i] = seq[k - i] if 0 <= k - i < n else 0
        # Use vectorized
        idx = np.arange(n)
        j = k - idx
        w = np.where((j >= 0) & (j < n), seq[j], 0.0)
        # choose i_max where w is largest (and seq[i] > 0)
        # to ensure we can take from that position
        candidates = np.where(seq > 1e-12)[0]
        if len(candidates) == 0:
            break
        i_max = candidates[np.argmax(w[candidates])]
        # choose i_min where w is smallest (prefer zero) and also not i_max
        # we can just pick random index with small w
        i_min = np.random.choice(np.where(w == w.min())[0])
        if i_max == i_min:
            # random other index
            i_min = np.random.choice(np.delete(np.arange(n), i_max))
        # compute delta
        delta = seq[i_max] * np.random.rand() * 0.5  # up to half of source
        if delta <= 0:
            continue
        # apply move
        seq[i_max] -= delta
        seq[i_min] += delta
        # compute new conv and max
        max_b_new, conv_new = max_conv(seq)
        # if improvement
        if max_b_new < max_b - 1e-12:
            max_b = max_b_new
            conv = conv_new
            score = 2 * n * max_b / sum_sq
            if score < best_score:
                best_score = score
                best_seq = seq.copy()
                print(f"Iter {it}: new best score {best_score:.6f}, max_b {max_b:.6f}")
        else:
            # revert move
            seq[i_max] += delta
            seq[i_min] -= delta
        it += 1
    return best_seq.tolist()

from scipy import sparse

def build_conv_matrix(f):
    n = len(f)
    rows = 2*n - 1
    data = []
    rows_idx = []
    cols_idx = []
    for k in range(rows):
        # i from max(0, k-(n-1)) to min(n-1, k)
        i_start = max(0, k - (n - 1))
        i_end = min(n - 1, k)
        # iterate i
        for i in range(i_start, i_end + 1):
            j = k - i
            # coefficient is f[i]
            rows_idx.append(k)
            cols_idx.append(j)
            data.append(f[i])
    A = sparse.csr_matrix((data, (rows_idx, cols_idx)), shape=(rows, n), dtype=np.float64)
    return A

"""
Propose a sequence that minimizes `evaluate_sequence`.

The algorithm starts from the uniform step‑function (which gives a score of 2),
then iteratively improves it using a linear program exactly as described in the
paper excerpt.  For a current sequence `f` we compute the current maximal
convolution value `M = max(f*f)`.  Keeping `M` fixed we solve

    maximise   sum(g)
    subject to conv(f, g) ≤ M ,   g ≥ 0

which gives a new candidate `g`.  After normalising `g` so that
`sum(g) = sqrt(2·n)` we line‑search a mixing parameter `t∈[0,1]` and replace

    f ← (1−t)·f + t·g

whenever the maximal convolution decreases.  The convolution, the maximal
value and the linear program are all evaluated with fast FFT‐based code, so
many iterations fit comfortably into the 1000 s budget.

The routine stops when no improvement is possible or when the time limit is
reached and returns the best sequence found.
"""

import time
import math
import numpy as np
from scipy.optimize import linprog
from scipy.linalg import toeplitz

# ----------------------------------------------------------------------
# Helper functions (top level – required by the rules)
# ----------------------------------------------------------------------


def _conv_fft(a: np.ndarray, b: np.ndarray, L: int) -> np.ndarray:
    """
    Linear convolution of two real vectors using FFT.
    Returns the first (2·len(a)−1) entries.
    """
    fa = np.fft.rfft(a, L)
    fb = np.fft.rfft(b, L)
    conv = np.fft.irfft(fa * fb, L)
    return conv[: 2 * len(a) - 1]


def _conv_self_fft(a: np.ndarray, L: int) -> np.ndarray:
    """
    Convolution of a vector with itself (shortcut).
    """
    fa = np.fft.rfft(a, L)
    conv = np.fft.irfft(fa * fa, L)
    return conv[: 2 * len(a) - 1]


def _build_conv_matrix(f: np.ndarray) -> np.ndarray:
    """
    Build the dense matrix A (shape (2n−1, n)) that represents the linear
    constraints  conv(f, g) ≤ M.
    Row k contains the coefficients f[i] for all i,j with i+j = k,
    i.e. A[k, j] = f[k‑j] when 0 ≤ k‑j < n, otherwise 0.
    """
    n = f.shape[0]
    # first column: f[0], f[1], …, f[n‑1], 0, …, 0   (length 2n‑1)
    col = np.concatenate([f, np.zeros(n - 1)])
    # first row: f[0] followed by zeros (length n)
    row = np.zeros(n)
    row[0] = f[0]
    return toeplitz(col, row)


# ----------------------------------------------------------------------
# Main routine required by the competition framework
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 123, budget_s: int = 1000) -> list[float]:
    """
    Return a sequence of non‑negative heights that (hopefully) improves on the
    current best bound.  The routine runs a short LP‑based optimisation loop
    (≤ 50 iterations) and respects the time budget.
    """
    np.random.seed(seed)

    # ------------------------------------------------------------------
    # Problem size and normalisation
    # ------------------------------------------------------------------
    n = len(height_sequence_1)                 # use the provided length
    target_sum = math.sqrt(2.0 * n)            # the normalisation used in the paper
    factor = 2.0 * n / (target_sum ** 2)       # constant factor; equals 1 for our normalisation

    # start from the uniform sequence (sum = target_sum)
    f = np.full(n, target_sum / n, dtype=np.float64)

    # FFT length – next power of two for speed
    L = 1 << ((2 * n - 1).bit_length())

    # initial data
    conv_f_f = _conv_self_fft(f, L)
    max_b = conv_f_f.max()
    best_score = factor * max_b
    best_seq = f.copy()
    print(f"[init] score={best_score:.6f}   max_b={max_b:.6f}")

    deadline = time.time() + budget_s - 5   # keep a small safety margin
    iteration = 0
    max_iterations = 50                     # safety cap

    while time.time() < deadline and iteration < max_iterations:
        # ------------------------------------------------------------------
        # 1) Build LP constraints for the current f and solve for g
        # ------------------------------------------------------------------
        A = _build_conv_matrix(f)                      # shape (2n‑1, n)
        b_rhs = np.full(2 * n - 1, max_b, dtype=np.float64)

        # linprog wants a minimisation problem; we maximise sum(g) → minimise -sum(g)
        c = -np.ones(n, dtype=np.float64)

        # bounds enforce g ≥ 0
        res = linprog(
            c,
            A_ub=A,
            b_ub=b_rhs,
            bounds=[(0.0, None)] * n,
            method="highs",
            options={"presolve": True, "time_limit": 10.0},
        )

        if not res.success:
            print(f"[iter {iteration}] LP failed – stopping")
            break

        g = res.x
        sum_g = g.sum()
        if sum_g <= 1e-12:
            print(f"[iter {iteration}] LP returned zero sum – stopping")
            break

        # ------------------------------------------------------------------
        # 2) Normalise g to the same L1‑norm as f (target_sum)
        # ------------------------------------------------------------------
        g = g * (target_sum / sum_g)

        # ------------------------------------------------------------------
        # 3) Pre‑compute the three convolutions needed for mixing
        # ------------------------------------------------------------------
        conv_f_g = _conv_fft(f, g, L)
        conv_g_g = _conv_self_fft(g, L)

        # ------------------------------------------------------------------
        # 4) Line search over the mixing parameter t ∈ [0,1]
        #    (grid 0.0, 0.05, …, 1.0).  For each t we can compute the mixed
        #    convolution analytically from the three pre‑computed ones.
        # ------------------------------------------------------------------
        t_grid = np.linspace(0.0, 1.0, 21)[1:-1]   # skip 0 and 1 for now
        best_t = 0.0
        best_M = max_b

        for t in t_grid:
            w0 = (1.0 - t) ** 2
            w1 = 2.0 * t * (1.0 - t)
            w2 = t ** 2
            conv_mix = w0 * conv_f_f + w1 * conv_f_g + w2 * conv_g_g
            M_t = conv_mix.max()
            if M_t < best_M:
                best_M = M_t
                best_t = t

        # also test the extreme point t = 1 (replace f by g)
        M_one = conv_g_g.max()
        if M_one < best_M:
            best_M = M_one
            best_t = 1.0

        # ------------------------------------------------------------------
        # 5) Accept the improvement if the maximal convolution decreased
        # ------------------------------------------------------------------
        if best_M < max_b - 1e-12:
            # update the current sequence and its self‑convolution
            if best_t == 1.0:
                f = g.copy()
                conv_f_f = conv_g_g.copy()
            else:
                f = (1.0 - best_t) * f + best_t * g
                w0 = (1.0 - best_t) ** 2
                w1 = 2.0 * best_t * (1.0 - best_t)
                w2 = best_t ** 2
                conv_f_f = w0 * conv_f_f + w1 * conv_f_g + w2 * conv_g_g

            max_b = best_M
            cur_score = factor * max_b
            if cur_score < best_score:
                best_score = cur_score
                best_seq = f.copy()
                print(
                    f"[iter {iteration}] improved: score={best_score:.6f}, "
                    f"max_b={max_b:.6f}, t={best_t:.3f}"
                )
        else:
            # no improvement – stop the loop
            print(f"[iter {iteration}] no improvement (max_b={max_b:.6f}); terminating")
            break

        iteration += 1

    # ------------------------------------------------------------------
    # Return the best sequence found (as a plain Python list)
    # ------------------------------------------------------------------
    return best_seq.tolist()
