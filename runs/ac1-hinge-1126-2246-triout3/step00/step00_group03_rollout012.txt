import numpy as np, time

def fast_evaluate(seq):
    a = np.asarray(seq, dtype=np.float64)
    n = a.shape[0]
    sum_a = a.sum()
    if sum_a < 0.01:
        return np.inf
    # compute conv via FFT
    L = 1 << ((2*n - 1).bit_length())
    # rfft
    fa = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(fa * fa, n=L)[:2*n-1]
    max_b = conv.max()
    return 2.0 * n * max_b / (sum_a * sum_a)
```

def fast_evaluate(seq, n=None):
    a = np.asarray(seq, dtype=np.float64)
    if n is None:
        n = a.shape[0]
    sum_a = a.sum()
    if sum_a < 0.01:
        return np.inf
    L = 1 << ((2*n - 1).bit_length())
    fa = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(fa * fa, n=L)[:2*n-1]
    max_b = conv.max()
    return 2.0 * n * max_b / (sum_a * sum_a)
```

Fast heuristic search for a low‑score step‑function.

The evaluation metric is   2 * n * max(conv(a,a)) / (sum(a)^2)
and is invariant under scaling of the whole vector.  The goal is therefore
to minimise the maximal self‑convolution for a fixed ℓ₁‑norm.

A simple analytic family that beats the uniform construction is a
power‑law sequence a_i = (i+1)^(-p).  For p≈0.57 the theoretical bound
(ignoring ordering effects) is already below the target 1.503.
The search therefore

* tries a few exponents p,
* shuffles the generated values arbitrarily (large entries are spread
  over the whole range, which reduces the peak of the convolution),
* evaluates candidates with a fast FFT based convolution,
* keeps the best candidate seen,
* performs a tiny local‑swap hill‑climb on promising candidates.

The whole routine respects the 1000 s budget and returns a plain
list of floats that can be fed to the provided `evaluate_sequence`
function.
"""

import time
import numpy as np

def _fast_score(seq: np.ndarray, n: int, fft_len: int) -> float:
    """Return the evaluation score using an FFT based convolution."""
    s = seq.sum()
    if s < 0.01:                     # protect against division by zero
        return np.inf
    # linear convolution via zero‑padded FFT
    fa = np.fft.rfft(seq, n=fft_len)
    conv = np.fft.irfft(fa * fa, n=fft_len)[: 2 * n - 1]
    return 2.0 * n * conv.max() / (s * s)
