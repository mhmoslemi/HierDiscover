            return seq
    except Exception:
        # No global variable, or it is unusable – fall back to a uniform one.
        pass

    # Choose a moderate length (e.g., 2000) – any length works.
    n = 2000
    # Use a value of 1.0; the exact scaling does not affect the ratio,
    # because the objective is homogeneous of degree 0.
    uniform_value = 1.0
    return [uniform_value] * n
