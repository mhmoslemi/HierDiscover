"""
Propose a candidate step‑function sequence.

The algorithm starts from a simple exponentially decaying sequence
a_i = exp(-i / n) (where n is the length of the sequence).  This shape
already improves the evaluation score to about 1.84 for n≈1500.  Then a
local hill‑climbing search is performed: at each iteration the
convolution entry with maximal value is identified and a small amount
of mass is moved from the index that contributes most to this maximum
to a randomly chosen other index.  The convolution is updated
incrementally, which makes each iteration O(n) instead of the O(n²)
cost of a full recomputation.  Moves that improve the score are kept;
otherwise they are discarded.  The search runs until the allocated
budget (default 1000 s) is exhausted and the best sequence found is
returned.

The procedure respects all constraints of `evaluate_sequence`:
entries stay non‑negative and far below the internal clipping bound
(1000).  Because the evaluation is homogeneous with respect to scaling,
no explicit normalisation is required.
"""
import time
import numpy as np

def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a sequence that minimises `evaluate_sequence`.

    Parameters
    ----------
    seed : int, optional
        Random seed for reproducibility.
    budget_s : int, optional
        Time budget in seconds.
    **kwargs : dict
        Ignored; present for compatibility with the testing harness.

    Returns
    -------
    list[float]
        The best sequence found within the time budget.
    """
    # ------------------------------------------------------------
    # 0. Setup
    # ------------------------------------------------------------
    rng = np.random.default_rng(seed)
    n = 1500
    # use the length of the existing construction if it exists
    if "height_sequence_1" in globals():
        try:
            n = len(height_sequence_1)
        except Exception:
            pass

    # ------------------------------------------------------------
    # 1. Initial candidate – exponential decay (c = 1)
    # ------------------------------------------------------------
    r = np.exp(-1.0 / n)               # r = exp(-c/n) with c = 1
    a = r ** np.arange(n, dtype=float)  # a_i = r^i
    sum_a = a.sum()                    # sum is > 0, scaling invariant
    # full convolution (linear, not circular)
    b = np.convolve(a, a)              # length = 2*n-1
    # evaluate the current score (same formula as evaluate_sequence)
    best_score = 2.0 * n * b.max() / (sum_a * sum_a)
    best_seq = a.copy()
    # keep a copy of the convolution for incremental updates
    # ------------------------------------------------------------
    # 2. Local optimisation loop
    # ------------------------------------------------------------
    start_time = time.time()
    deadline = start_time + budget_s - 5.0   # keep a small safety margin
    # fraction of the selected entry to move in each step
    step_fraction = 0.05
    iteration = 0
    # for occasional progress reports
    report_interval = 5000

    while time.time() < deadline:
        iteration += 1

        # --------------------------------------------------------------------
        # a) locate the convolution entry with maximal value
        # --------------------------------------------------------------------
        k_max = int(np.argmax(b))

        # indices that actually contribute to this entry (i + j = k_max)
        i_start = max(0, k_max - (n - 1))
        i_end = min(k_max, n - 1)
        idx_range = np.arange(i_start, i_end + 1, dtype=int)

        # contributions a_i * a_{k_max-i}
        contrib = a[idx_range] * a[k_max - idx_range]

        # index with the largest contribution (the “culprit”)
        i_culprit = idx_range[np.argmax(contrib)]

        # --------------------------------------------------------------------
        # b) pick a destination index (different from the culprit)
        # --------------------------------------------------------------------
        j_target = rng.integers(0, n)
        if j_target == i_culprit:
            # choose another index
            j_target = (j_target + 1) % n
            if j_target == i_culprit:
                continue

        # --------------------------------------------------------------------
        # c) decide how much mass to move
        # --------------------------------------------------------------------
        # move at most `step_fraction` of the culprit’s mass,
        # but never more than the whole amount
        delta = min(step_fraction * a[i_culprit], a[i_culprit] * 0.9)
        if delta <= 0.0:
            continue

        # --------------------------------------------------------------------
        # d) store old state (required for possible rejection)
        # --------------------------------------------------------------------
        a_old = a.copy()
        b_old = b.copy()

        # --------------------------------------------------------------------
        # e) apply the mass transfer (maintain total sum)
        # --------------------------------------------------------------------
        a[i_culprit] = a_old[i_culprit] - delta
        a[j_target] = a_old[j_target] + delta

        # --------------------------------------------------------------------
        # f) incremental update of the convolution (O(n))
        # --------------------------------------------------------------------
        # contributions of the decreasing entry
        b[i_culprit : i_culprit + n] -= 2.0 * delta * a_old
        # contributions of the increasing entry
        b[j_target : j_target + n] += 2.0 * delta * a_old

        # the quadratic corrections (ΔΔ term)
        b[2 * i_culprit] += delta * delta
        b[2 * j_target]   += delta * delta
        b[i_culprit + j_target] -= 2.0 * delta * delta

        # numerical tidying – remove possible tiny negative values caused
        # by floating‑point round‑off.
        b = np.maximum(b, 0.0)

        # --------------------------------------------------------------------
        # g) compute the new score
        # --------------------------------------------------------------------
        new_score = 2.0 * n * b.max() / (sum_a * sum_a)

        # --------------------------------------------------------------------
        # h) accept or reject the move
        # --------------------------------------------------------------------
        if new_score < best_score:
            # keep the move
            best_score = new_score
            best_seq = a.copy()
        else:
            # reject – revert to the saved state
            a = a_old
            b = b_old

        # occasional progress output
        if iteration % report_interval == 0:
            elapsed = time.time() - start_time
            print(f"Iter {iteration:>8d} | time {elapsed:5.1f}s | best score {best_score:.6f}")

    # ------------------------------------------------------------
    # 3. Return the best sequence discovered
    # ------------------------------------------------------------
    print(f"Finished after {iteration} iterations – best score = {best_score:.6f}")
    return best_seq.tolist()
