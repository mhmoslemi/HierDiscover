def project_simplex(v):
    # Bring vector onto simplex (non-negative, sum=1)
    # Standard algorithm: sort descending, find rho, compute theta.
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(len(v)) + 1
    rho = np.where(u - cssv / ind > 0)[0][-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0)
    return w

"""
Propose a step‑function (list of non‑negative heights) that minimises
`evaluate_sequence`.  

The evaluation function can be written as  

    score = 2 * n * max_k ( (w * w)[k] )          (with sum(w)=1)

where `w` is the normalised coefficient vector and `*` is the (linear)
convolution.  Hence we have to find a probability distribution `w`
whose self‑convolution is as flat as possible.  The smallest possible
value of `max_k (w*w)[k]` is `1/(2n‑1)` (the average of the
convolution), giving a theoretical lower bound `2n/(2n‑1)≈1`.  Uniform
coefficients give `max = 1/n` and a score of `2`, so we must move away
from the uniform vector.

The algorithm below is a simple projected sub‑gradient method:

* start from a random point on the probability simplex,
* repeatedly compute the convolution, locate the index `k_max` that
  attains the maximum,
* the sub‑gradient of the max‑convolution at the current point is
  `g_i = 2 * w_{k_max-i}` (zero when the index is out of range),
* take a step in the opposite direction and project back onto the
  simplex,
* keep the best vector seen overall (including several random restarts).

All operations are O(n log n) thanks to an FFT based convolution,
so thousands of iterations easily fit into the 1000‑second budget.
The final vector is returned as a plain list of floats (any scaling
is irrelevant because the score is scale‑invariant).
"""

import time
import numpy as np

# -------------------------------------------------------------
# Helper functions (must be defined at top‑level, no closures)
# -------------------------------------------------------------

def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Project a vector onto the simplex {x >= 0, sum(x) = 1}."""
    # Implements the algorithm from:
    #   Wang & Carreira‑Perpiñán, "Projection onto the probability simplex"
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all entries are zero – return the uniform distribution
        return np.full_like(v, 1.0 / n)
    rho = np.nonzero(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def _max_convolution(v: np.ndarray) -> float:
    """
    Return max_k (v * v)[k] using an FFT based full convolution.
    The function works for any length n and uses a power‑of‑two padding
    to get good performance.
    """
    n = v.shape[0]
    # smallest power of two ≥ 2*n
    size = 1 << ((2 * n - 1).bit_length())
    # real FFT – saves roughly half the work
    fv = np.fft.rfft(v, n=size)
    conv = np.fft.irfft(fv * fv, n=size)[: 2 * n - 1]
    return float(np.max(conv))

# Main search routine
# -------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: int = 1000, **_) -> list[float]:
    """
    Search for a coefficient sequence that gives a low value of
    `evaluate_sequence`.  The routine uses a multi‑restart projected
    sub‑gradient descent, respects the time limit, and finally returns
    the best sequence found.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 5.0   # small safety margin

    # Use the length of the provided construction if it exists,
    # otherwise fall back to a reasonably large size.
    n = int(globals().get("height_sequence_1", [0] * 7_152).__len__())
    if n < 2:                         # extremely unlikely, but be safe
        n = 7_152

    best_w = None
    best_score = float("inf")
    restart = 0

    # Parameters for the sub‑gradient method
    eta0 = 0.8            # initial step size (will be scaled down)
    max_iter_per_restart = 3000

    # -----------------------------------------------------------------
    # Main loop – multiple random restarts until time runs out
    # -----------------------------------------------------------------
    while time.time() < deadline:
        # ----- initialise a random point on the simplex ----------------
        if restart % 5 == 0:
            # every 5th restart start from the uniform vector (baseline)
            w = np.full(n, 1.0 / n)
        else:
            w = np.random.rand(n)
            w /= w.sum()

        # ----- sub‑gradient descent ------------------------------------
        for it in range(max_iter_per_restart):
            # compute full convolution and locate the peak
            conv = None
            # FFT based convolution (fast)
            size = 1 << ((2 * n - 1).bit_length())
            fv = np.fft.rfft(w, n=size)
            conv = np.fft.irfft(fv * fv, n=size)[: 2 * n - 1]
            k_max = int(np.argmax(conv))
            max_val = float(conv[k_max])

            # current score (scale‑invariant, sum(w)=1)
            cur_score = 2.0 * n * max_val
            if cur_score < best_score:
                best_score = cur_score
                best_w = w.copy()
                print(
                    f"[{time.time():.1f}] restart {restart}, it {it}, "
                    f"new best = {best_score:.6f}"
                )

            # ---- sub‑gradient of the max‑convolution -------------------
            g = np.zeros_like(w)
            # indices i for which (k_max - i) lies inside [0, n-1]
            i_start = max(0, k_max - (n - 1))
            i_end = min(n - 1, k_max)
            i_idx = np.arange(i_start, i_end + 1, dtype=int)
            j_idx = k_max - i_idx
            g[i_idx] = 2.0 * w[j_idx]

            # ---- update ------------------------------------------------
            step = eta0 / np.sqrt(it + 1)           # diminishing step size
            w_new = w - step * g
            w = _project_simplex(w_new)

            # ---- early exit if we are out of time --------------------
            if time.time() >= deadline:
                break

        restart += 1

    # -----------------------------------------------------------------
    # Fallback – if something went wrong return the uniform vector
    # -----------------------------------------------------------------
    if best_w is None:
        best_w = np.full(n, 1.0 / n)

    # The evaluator is scale‑invariant, but we keep the values modest.
    # Return a plain Python list as required by the runner.
    return best_w.tolist()
