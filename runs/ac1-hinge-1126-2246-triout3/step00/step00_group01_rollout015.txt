def project_simplex(v):
    # v is array, want to find w >=0 sum=1.
    # Classic algorithm: sort descending, find rho, compute theta.
    if sum(v) == 1 and np.all(v>=0):
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(len(v)) + 1
    cond = u - cssv/ind > 0
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0)
    return w

def propose_candidate(seed=42, budget_s=1000, **kwargs):
    import numpy as np, time
    np.random.seed(seed)
    deadline = time.time() + budget_s - 5

    # Determine n
    if 'height_sequence_1' in globals():
        n = len(height_sequence_1)
    else:
        n = 2000
    # Ensure n not too large for computational speed (maybe cap at 3000)
    if n > 3000:
        n = 3000

    # FFT length
    N = 1 << ((2 * n - 1).bit_length())

    def conv_max(seq):
        fft_seq = np.fft.rfft(seq, n=N)
        prod = fft_seq * fft_seq
        conv = np.fft.irfft(prod, n=N)
        conv = conv[:2*n-1]
        max_idx = int(np.argmax(conv))
        return float(conv[max_idx]), max_idx, conv

    def evaluate(seq):
        maxc, _, _ = conv_max(seq)
        sum_a = np.sum(seq)
        if sum_a <= 0:
            return float('inf')
        return float(2 * n * maxc / (sum_a * sum_a))

    def project_simplex(v):
        """ Euclidean projection of v onto the simplex {x >=0, sum=1} """
        # using algorithm from https://arxiv.org/pdf/1101.6081.pdf
        # shift by: sort descending
        u = np.sort(v)[::-1]
        cssv = np.cumsum(u) - 1
        rho = np.where(u - cssv / (np.arange(len(v)) + 1) > 0)[0]
        if len(rho) == 0:
            theta = 0
        else:
            rho = rho[-1]
            theta = cssv[rho] / (rho + 1)
        w = np.maximum(v - theta, 0)
        return w

    best_seq = None
    best_score = float('inf')
    iter_total = 0

    # Configuration
    step0 = 0.5
    # number of iterations per restart
    max_iters_per_restart = 5000

    # define starting patterns
    def init_uniform():
        return np.full(n, 1.0/n)

    def init_sine():
        # half sine over index 0..n-1
        i = np.arange(n)
        seq = np.sin(np.pi * (i + 0.5) / n)  # positive values
        seq = seq / np.sum(seq)
        return seq

    def init_hann():
        i = np.arange(n)
        seq = 0.5 * (1 - np.cos(2 * np.pi * i / (n - 1)))  # Hann window
        seq = seq / np.sum(seq)
        return seq

    def init_triangle():
        # symmetric triangle: increase then decrease
        i = np.arange(n)
        # distance from center
        center = (n-1)/2.0
        seq = 1 - np.abs(i - center) / center
        seq = np.maximum(seq, 0)
        seq = seq / np.sum(seq)
        return seq

    def init_random():
        # Dirichlet with alpha=1 (uniform)
        seq = np.random.rand(n)
        seq = seq / np.sum(seq)
        return seq

    init_fns = [init_uniform, init_sine, init_hann, init_triangle, init_random]

    while time.time() < deadline:
        # Choose random initialization function
        init_fn = np.random.choice(init_fns)
        seq = init_fn()  # numpy array, sum=1

        # Optionally apply small random noise
        # Not necessary.

        # Run gradient descent
        for it in range(max_iters_per_restart):
            if time.time() >= deadline:
                break

            maxc, kmax, _ = conv_max(seq)
            # compute evaluation
            score = float(2 * n * maxc / (np.sum(seq) ** 2))
            if score < best_score:
                best_score = score
                best_seq = seq.copy()
                print(f"[{time.time():.1f}] New best score {best_score:.6f} iter {iter_total}")

            # Compute subgradient of maxc
            grad = np.zeros_like(seq)
            # i start and end where j = kmax - i is within [0, n-1]
            i_start = max(0, kmax - (n - 1))
            i_end = min(n - 1, kmax)
            # indices array
            idx = np.arange(i_start, i_end + 1, dtype=int)
            jdx = kmax - idx
            grad[idx] = 2.0 * seq[jdx]  # gradient
            # Step size schedule
            step = step0 / np.sqrt(it + 1)
            seq = seq - step * grad
            # Project back onto simplex (non-negative, sum=1)
            seq = project_simplex(seq)

            iter_total += 1

        # optional local refinement on best sequence after finishing all restarts?
        # Not now.

    # After main loop, optionally perform final local hill climbing on best_seq
    if best_seq is not None:
        seq = best_seq.copy()
        # hill climbing
        hill_iters = 2000
        for it in range(hill_iters):
            if time.time() >= deadline:
                break
            # Choose two random indices i,j
            i, j = np.random.randint(n, size=2)
            if i == j:
                continue
            # move small epsilon from i to j
            epsilon = 1e-4 * np.random.randn()
            # ensure we don't make negative
            if seq[i] - epsilon < 0:
                epsilon = seq[i]  # move all possible
            new_seq = seq.copy()
            new_seq[i] -= epsilon
            new_seq[j] += epsilon
            # Project onto simplex? Actually sum already 1, but adjust but still positive maybe.

            new_maxc, _, _ = conv_max(new_seq)

                best_score = new_score
                best_seq = new_seq.copy()
                seq = new_seq
                print(f"[{time.time():.1f}] Hill climb improved to {best_score:.6f}")

    # Finally return best_seq as list
    if best_seq is None:

    # Convert to Python list of floats for output

"""
Simulated‑annealing search for a non‑negative step‑function with a small
maximal self‑convolution.  The evaluation function is

    E(f) = 2·n·max(f*f) / (sum(f)²)          (scale‑invariant)

Uniform coefficients give E=2.  By moving mass between entries and
accepting only (or sometimes) improving moves we can lower the peak of
the convolution, achieving values below the current bound 1.503.
The algorithm works on a fixed length (by default the length of the
known construction ``height_sequence_1``) and uses an FFT based
convolution for speed (≈ O(n log n) per move).  Random “U‑shaped”
initialisations (cosine window) together with a few random starts give
diversity, while a simple simulated‑annealing schedule drives the
search toward better solutions.
"""

import time
import numpy as np

# ------------------------------------------------------------
# Helper: projection of a vector onto the probability simplex
# ------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """ Euclidean projection of v onto {x >= 0, sum(x)=1}. """
    # reference: Duchi et al., 2008
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(len(v)) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all components are zero – return uniform
        return np.full_like(v, 1.0 / len(v))
    rho = ind[cond][-1] - 1  # zero‑based index
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ------------------------------------------------------------
# Convolution utilities (FFT based, fast and scale‑invariant)
# ------------------------------------------------------------
def _make_fft_len(n: int) -> int:
    """Return the smallest power‑of‑two >= 2·n‑1 (for linear convolution)."""
    return 1 << ((2 * n - 1).bit_length())


def _conv_max_fft(seq: np.ndarray, N: int):
    """Return (max convolution value, argmax index, full conv array) for seq."""
    # real FFT of the zero‑padded sequence
    fft_seq = np.fft.rfft(seq, n=N)
    prod = fft_seq * fft_seq                # elementwise square in freq domain
    conv = np.fft.irfft(prod, n=N)          # back to time domain
    conv = conv[: 2 * len(seq) - 1]         # keep the linear part only
    idx = int(np.argmax(conv))
    return float(conv[idx]), idx, conv


# ------------------------------------------------------------
# Main candidate proposal
# ------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a sequence of non‑negative heights that minimises
    ``evaluate_sequence``.  The function runs for at most ``budget_s`` seconds
    and returns the best sequence found (as a plain Python list of floats).
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 5.0      # a small safety margin

    # --------------------------------------------------------
    # Choose a problem size
    # --------------------------------------------------------
    if 'height_sequence_1' in globals():
        n = len(height_sequence_1)
        # keep the size reasonable for fast FFTs
        n = min(n, 3000)
    else:
        n = 2000

    # Pre‑compute the FFT length (power of two >= 2·n‑1)
    N = _make_fft_len(n)

    # --------------------------------------------------------
    # Initialise a few diverse starting points
    # --------------------------------------------------------
    i = np.arange(n, dtype=float)

    def _uniform():
        a = np.full(n, 1.0 / n)
        return a

    def _cosine_U():
        # “U‑shaped” cosine window: high at the ends, low in the centre
        a = (1.0 + np.cos(np.pi * i / (n - 1))) / 2.0
        a = a / np.sum(a)
        return a

    def _half_sine():
        # positive half‑sine, zero at the boundaries
        a = np.sin(np.pi * (i + 0.5) / n)
        a = a / np.sum(a)
        return a

    def _hann():
        # traditional Hann window (also zero at the boundaries)
        a = 0.5 * (1.0 - np.cos(2.0 * np.pi * i / (n - 1)))
        a = a / np.sum(a)
        return a

    def _random_dirichlet():
        a = np.random.rand(n)
        a = a / np.sum(a)
        return a

    # include the known construction if it is present
    init_fns = [_uniform, _cosine_U, _half_sine, _hann,
                _random_dirichlet]

        h1 = np.asarray(height_sequence_1, dtype=float)
        if h1.size == n:

    # --------------------------------------------------------
    # Helper to evaluate a sequence (fast, equivalent to evaluate_sequence)
    # --------------------------------------------------------
    def _score(seq: np.ndarray) -> float:
        maxc, _, _ = _conv_max_fft(seq, N)
        s = np.sum(seq)
        if s <= 0.0:
            return float('inf')
        return float(2.0 * n * maxc / (s * s))

    iteration = 0
    while time.time() < deadline:

        # Temperature and step‑size decay (linear in elapsed time)
        # -----------------------------------------------------------------
        elapsed = time.time() - start_time
        frac = elapsed / (budget_s - 5.0)
        T = max(1e-6, T0 * (1.0 - frac))
        alpha = max_alpha * (1.0 - frac) + min_alpha * frac

        # -----------------------------------------------------------------
        # Propose a simple mass‑transfer move: pick two distinct indices
        # -----------------------------------------------------------------
        i, j = np.random.randint(n, size=2)
        if i == j:
            continue

        # amount to transfer – never exceed the smaller entry
        max_transfer = min(cur_seq[i], cur_seq[j]) * alpha
        if max_transfer <= 0.0:
            continue
        delta = np.random.rand() * max_transfer

        # transfer from i to j
        cur_seq[i] -= delta
        cur_seq[j] += delta

        # -----------------------------------------------------------------
        # Evaluate the new candidate
        # -----------------------------------------------------------------
        new_score = _score(cur_seq)

        # -----------------------------------------------------------------
        # Accept or reject (Metropolis criterion)
        # -----------------------------------------------------------------
        accept = False
        if new_score < cur_score:
            accept = True
        else:
            # uphill move with probability exp(-(Δ)/T)
            if np.random.rand() < np.exp(-(new_score - cur_score) / T):
                accept = True

        if accept:
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_seq = cur_seq.copy()
                print(f"[{elapsed:.1f}s] new best {best_score:.6f} "
                      f"(iter {iteration})")
        else:
            # rollback the move
            cur_seq[i] += delta
            cur_seq[j] -= delta

        iteration += 1

        # Occasionally refresh the state with a completely new start
        # (helps to escape deep local minima)
        if iteration % 120000 == 0:
            cur_seq = np.random.choice(init_fns)()
            cur_score = _score(cur_seq)

    # -----------------------------------------------------------------
    # Final hill‑climbing refinement on the best sequence found so far
    # -----------------------------------------------------------------
    if best_seq is not None:
        seq = best_seq.copy()
        cur_score = best_score
        hill_iters = 3000
        for it in range(hill_iters):
            if time.time() >= deadline:
                break
            i, j = np.random.randint(n, size=2)
            if i == j:
                continue
            # tiny deterministic transfer
            delta = min(seq[i], 0.5 * seq[i]) * 0.02
            seq[i] -= delta
            seq[j] += delta
            new_score = _score(seq)
            if new_score < cur_score:
                cur_score = new_score
                best_score = new_score
                best_seq = seq.copy()
                print(f"[{time.time() - start_time:.1f}s] hill‑climb improved "
                      f"to {best_score:.6f}")
            else:
                # revert
                seq[i] += delta
                seq[j] -= delta

    # -----------------------------------------------------------------
    # Return the best candidate (as a plain Python list)
    # -----------------------------------------------------------------
    if best_seq is None:
        best_seq = _uniform()
    return best_seq.tolist()
