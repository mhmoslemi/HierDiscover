def propose_candidate(seed=42, budget_s=1000, **kwargs):
    """
    Random search for binary step function minimizing evaluate_sequence.
    Generates random 0/1 sequences of varying lengths, keeps the best found.
    Uses simple hill‑climbing by swapping bits to further reduce the max convolution.
    """
    import numpy as np, time, random
    np.random.seed(seed)
    random.seed(seed)
    deadline = time.time() + max(budget_s - 0.5, 0)  # keep small slack
    
    best_seq = None
    best_score = float('inf')
    # sizes to try; start from moderate size and increase if time permits
    sizes = [1024, 2048, 4096, 8192]
    # convert to list to iterate through sizes cyclically based on remaining time
    size_idx = 0
    
    # Goal: stop if we achieve score below target (e.g., 1.4)
    target = 1.4
    
    # Helper to evaluate
    def eval_seq(arr):
        return evaluate_sequence(arr.tolist())
    
    # Main loop
    while time.time() < deadline:
        # Choose length
        N = sizes[size_idx % len(sizes)]
        size_idx += 1
        # Choose density ~0.5 but random around it
        density = 0.5 + 0.1 * (np.random.rand() - 0.5)  # between 0.45 and 0.55
        # Generate random binary
        seq = (np.random.rand(N) < density).astype(float)
        # Ensure non‑empty
        if seq.sum() < 0.01:
            seq[0] = 1.0
        
        score = eval_seq(seq)
        if score < best_score:
            best_score = score
            best_seq = seq.copy()
            print(f"[{time.time():.2f}] New best: {best_score:.6f} (N={N}, density={density:.3f})")
            if best_score <= target:
                # Early stop – we already beat the target.
                break
        
        # Optional: try a cheap local optimisation on the best sequence found
        # Do a few random swaps if we have enough time
        if best_seq is not None and time.time() < deadline - 0.2:
            # do up to 30 swaps per iteration
            for _ in range(30):
                # pick a position with 1 and a position with 0 to keep sum unchanged
                ones = np.where(best_seq > 0.5)[0]
                zeros = np.where(best_seq < 0.5)[0]
                if len(ones) == 0 or len(zeros) == 0:
                    break
                i = np.random.choice(ones)
                j = np.random.choice(zeros)
                # swap
                best_seq[i], best_seq[j] = 0.0, 1.0
                new_score = eval_seq(best_seq)
                if new_score < best_score:
                    best_score = new_score
                    print(f"[{time.time():.2f}] Swap improved: {best_score:.6f}")
                else:
                    # revert
                    best_seq[i], best_seq[j] = 1.0, 0.0
                if best_score <= target:
                    break
    
    # If nothing was found (should not happen) fall back to a simple sequence
    if best_seq is None:
        best_seq = np.ones(10, dtype=float)
    
    # Return as a plain Python list of floats
    return best_seq.tolist()

# -*- coding: utf-8 -*-
"""
Random binary search for a low‑max‑convolution step function.

The algorithm repeatedly draws 0/1 sequences of moderate length (≈10³–10⁴)
with density ≈½, evaluates the target functional
`evaluate_sequence`, and keeps the best sequence seen.
Because the functional is invariant under a global scaling, binary
values are optimal for a given support.  A simple time‑bounded loop
stops either when the allotted budget expires or when a score below a
chosen target (≈1.4) is reached.

The implementation uses only top‑level helpers (no closures) and
relies on NumPy for fast convolution inside `evaluate_sequence`.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities (top‑level – no closures)
# ----------------------------------------------------------------------


def _binary_sequence(length: int, density: float, rng: np.random.RandomState) -> np.ndarray:
    """
    Return a random binary (0/1) sequence of a given length.
    ``density`` is the probability of a 1; values are returned as ``float``.
    """
    return (rng.rand(length) < density).astype(float)


def _score_sequence(seq: np.ndarray) -> float:
    """
    Evaluate a step‑function sequence using the provided ``evaluate_sequence``.
    ``seq`` may be a NumPy array; it is converted to a plain list before
    calling the external evaluator.
    """
    # The external function is assumed to be available in the global namespace.
    return evaluate_sequence(seq.tolist())


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: float = 1000, **kwargs):
    """
    Search for a non‑negative step‑function that minimises
    ``evaluate_sequence``.  The function runs for at most ``budget_s`` seconds
    and returns a Python list of floats (the best sequence found).

    Strategy:
      * Randomly generate binary sequences of several lengths.
      * Keep the best one (lowest score).
      * Stop early if a score ≤ 1.4 is achieved.
    """
    # Initialise RNG for reproducibility
    rng = np.random.RandomState(seed)

    # Reserve a small safety margin for finalisation
    deadline = time.time() + max(budget_s - 0.5, 0.0)

    # Lengths that we will try.  They grow geometrically, giving finer
    # behaviour for larger N while still being cheap to evaluate.
    candidate_lengths = [1024, 2048, 4096, 8192]

    best_seq = None          # NumPy array holding the current best
    best_score = float("inf")
    target_score = 1.40      # early‑stop threshold (well below the current 2.0)

    attempts = 0
    # ------------------------------------------------------------------
    # Main random search loop
    # ------------------------------------------------------------------
    while time.time() < deadline:
        attempts += 1

        # Pick a length (cycle through the list to keep variety)
        length = candidate_lengths[attempts % len(candidate_lengths)]

        # Random density in [0.45, 0.55] – keeps the sum away from zero
        density = 0.45 + 0.10 * rng.rand()

        # Generate a fresh binary candidate
        seq = _binary_sequence(length, density, rng)

        # Guard against the pathological case of an all‑zero vector
        if seq.sum() < 0.01:
            seq[0] = 1.0

        # Evaluate
        try:
            score = _score_sequence(seq)
        except Exception:
            # Any unexpected error is ignored – the candidate is discarded
            continue

        # Keep the best one
        if score < best_score:
            best_score = score
            best_seq = seq.copy()
            print(f"[{time.time():.2f}] New best: {best_score:.6f} "
                  f"(len={length}, density={density:.3f}, attempts={attempts})")
            # Early exit if we are already sufficiently good
            if best_score <= target_score:
                break

    # ------------------------------------------------------------------
    # Fallback – should never happen, but ensures a non‑empty return.
    # ------------------------------------------------------------------
    if best_seq is None or best_seq.sum() < 0.01:
        # Very small deterministic fallback (guaranteed to be valid)
        best_seq = np.ones(10, dtype=float)

    # Return a plain Python list as required by the evaluation harness
    return best_seq.tolist()
