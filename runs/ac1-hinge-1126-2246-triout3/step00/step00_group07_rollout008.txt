"""
Propose a candidate step function by searching for a sparse support set that spreads
the self‑convolution as uniformly as possible.

Idea
----
For a non‑negative sequence a with Σa = 1 the evaluation is

    E(a) = 2·n·max_k (a*a)_k .

Thus we want to minimise the maximum entry of the auto‑convolution.
If the support of a is a random subset of the indices, the number of ordered
pairs (i,j) that contribute to a given sum k is roughly the same for all k,
≈ m²/(2·n) where m = |support|.  With equal weights w = 1/m each contribution
has size w², so the expected maximum is about 1/(2·n) and the expected score
≈ 1, well below the target 1.503.

The algorithm therefore repeatedly:
* picks a random subset of indices of size m ≈ α·n (α∈[0.3,0.8]),
* assigns equal weight 1/m on this support,
* evaluates the sequence,
* keeps the best one found.

A short hill‑climbing phase (random swaps of support elements) is performed
after the sampling loop to fine‑tune the support.  The whole search stops
before the allotted time budget (default 1000 s) and returns the best
sequence discovered.

The method needs only calls to ``evaluate_sequence`` and does not rely on any
external files or network access.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities
# ----------------------------------------------------------------------
def build_sequence(n: int, support: np.ndarray, weight: float) -> list[float]:
    """Return a list of length n with the given weight on the support indices."""
    seq = np.zeros(n, dtype=float)
    seq[support] = weight
    return seq.tolist()


def random_support(rng: np.random.Generator, n: int, m: int) -> np.ndarray:
    """Draw m distinct indices from range(n) using rng."""
    return rng.choice(n, size=m, replace=False)


def hill_climb(seq: list[float],
               support: np.ndarray,
               weight: float,
               deadline: float,
               best_score: float) -> tuple[list[float], float]:
    """
    Simple hill‑climbing: repeatedly try to swap a support index with a
    non‑support index.  Keep the change if the score improves.
    """
    n = len(seq)
    support_set = set(support.tolist())
    seq_arr = np.array(seq, dtype=float)

    # pre‑allocate an array for fast temporary modifications
    while time.time() < deadline:
        # pick one element to remove and one to add
        out_idx = rng.choice(list(support_set))
        # pick a candidate not currently in the support
        cand_idx = rng.integers(0, n)
        while cand_idx in support_set:
            cand_idx = rng.integers(0, n)

        # perform the swap
        seq_arr[out_idx] = 0.0
        seq_arr[cand_idx] = weight

        # evaluate
        new_score = evaluate_sequence(seq_arr.tolist())
        if new_score < best_score:
            best_score = new_score
            # accept the swap permanently
            support_set.remove(out_idx)
            support_set.add(cand_idx)
        else:
            # revert the change
            seq_arr[cand_idx] = 0.0
            seq_arr[out_idx] = weight

    # return the final sequence and its score
    return seq_arr.tolist(), best_score

    The function runs for up to ``budget_s`` seconds and returns the best
    sequence found.  All entries are non‑negative and the sum is forced to 1.
    """
    global height_sequence_1  # may be defined in the surrounding environment
    rng = np.random.default_rng(seed)

    # Determine the length of the sequence we are allowed to use.
    try:
        n = len(height_sequence_1)          # use the length of the known construction
    except Exception:
        n = 5079                            # fallback length given in the description

    deadline = time.time() + budget_s - 5.0   # keep a small safety margin

    best_seq = None
    best_score = float('inf')
    iteration = 0

    # ------------------------------------------------------------------
    # Random‑subset sampling phase
    # ------------------------------------------------------------------
    while time.time() < deadline:
        # Choose a random density α ∈ [0.3, 0.8] and the corresponding support size.
        alpha = rng.uniform(0.3, 0.8)

        supp = random_support(rng, n, m)

        # Evaluate the candidate.

        except Exception:                     # any unexpected error → treat as ∞

        # Keep the best one.

            best_seq = seq

    print(f"Final best score: {best_score:.6f}")
    return best_seq
