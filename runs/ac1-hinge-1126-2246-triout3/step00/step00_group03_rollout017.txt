def propose_candidate(seed=0, budget_s=1000):
    np.random.seed(seed)
    deadline = time.time() + budget_s - 5
    # length
    n = len(height_sequence_1) if 'height_sequence_1' in globals() else 1000
    sum_target = np.sqrt(2 * n)
    # initial sequence uniform
    a = np.full(n, sum_target / n)
    cur_score = evaluate_sequence(a.tolist())
    best_score = cur_score
    best_seq = a.copy()
    # parameters
    step = sum_target * 0.1  # initial step size maybe 0.1 of average
    # temperature for SA
    T = 0.1
    iter = 0
    while time.time() < deadline:
        # compute convolution using FFT
        size = 1 << ((2*n - 1).bit_length())
        # compute FFT of a
        A = np.fft.rfft(a, n=size)
        conv = np.fft.irfft(A * A, n=size)[:2*n-1]
        M = conv.max()
        kmax = int(np.argmax(conv))
        # gradient
        g = np.zeros_like(a)
        # fill g by vectorized indexing:
        idx = np.arange(n)
        j = kmax - idx
        mask = (j >= 0) & (j < n)
        g[mask] = 2 * a[j[mask]]
        # propose new a
        candidate = a - step * g
        # clip
        candidate = np.maximum(candidate, 0.0)

        if total > 0:
            candidate = candidate * (sum_target / total)
        else:
            # if candidate all zero, skip
            candidate = a.copy()
        new_score = evaluate_sequence(candidate.tolist())
        # acceptance
        if new_score < cur_score or np.random.rand() < np.exp(-(new_score - cur_score) / max(T, 1e-12)):
            a = candidate
            cur_score = new_score
            if new_score < best_score:

                best_seq = a.copy()

def propose_candidate(seed=0, budget_s=1000, **kwargs):
    import time, numpy as np
    np.random.seed(seed)
    deadline = time.time() + budget_s - 5

    # choose length
    if 'height_sequence_1' in globals():
        n = len(height_sequence_1)
    else:
        n = 1000
    # target sum for scaling
    sum_target = np.sqrt(2.0 * n)

    # initial exponential shape with c=1.0
    c = 1.0
    delta = c / n
    r = 1.0 - delta
    # generate a_i = r**i
    a = np.power(r, np.arange(n, dtype=float))
    # scale to target sum
    a = a * (sum_target / a.sum())

    # compute evaluation using convolution
    size = 1 << ((2*n - 1).bit_length())
    def compute_score(arr):
        A = np.fft.rfft(arr, n=size)
        conv = np.fft.irfft(A * A, n=size)[:2*n-1]
        M = conv.max()
        S = arr.sum()
        return 2.0 * n * M / (S * S), M, conv

    best_score, best_M, _ = compute_score(a)
    best_seq = a.copy()
    print(f"Initial score {best_score:.6f}")

    # hill climbing
    step_factor = 1.2  # max scaling factor
    iteration = 0
    while time.time() < deadline:
        # pick random pair (i, j) symmetric about centre
        i = np.random.randint(n // 2)
        j = n - 1 - i
        # ensure a_i < a_j
        if a[i] >= a[j]:
            continue
        # propose scaling factor t > 1, random in [1, step_factor]
        t = 1.0 + (step_factor - 1.0) * np.random.rand()
        # new values preserving product a[i]*a[j]
        p = a[i] * a[j]
        a_i_new = a[i] * t
        a_j_new = p / a_i_new
        # ensure non-negative (always true)
        # temporary array copy
        a_tmp = a.copy()
        a_tmp[i] = a_i_new
        a_tmp[j] = a_j_new

        # compute new score
        new_score, new_M, _ = compute_score(a_tmp)

        # accept if better
        if new_score < best_score:
            a = a_tmp
            best_score = new_score
            best_seq = a.copy()
            # maybe print progress
            print(f"Iter {iteration}: improved to {best_score:.6f}")
        # else maybe accept with probability for simulated annealing
        elif np.random.rand() < 0.001:
            a = a_tmp
        iteration += 1

        # adapt step_factor slowly reduce
        if iteration % 5000 == 0:
            step_factor = max(1.01, step_factor * 0.9)

    # final scaling to keep values within 0-1000 as per security clamping
    best_seq = np.clip(best_seq, 0.0, 1000.0)
    # optionally rescale to sum_target to avoid tiny sum
    total = best_seq.sum()
    if total < 0.01:
        best_seq = best_seq + (sum_target - total) / n
    else:
        best_seq = best_seq * (sum_target / total)

    return best_seq.tolist()

"""
Propose a sequence that lowers the evaluation function using a smooth‑max
gradient descent on the self‑convolution.

The evaluation is
    E(a) = 2 * n * max(a * a) / (sum(a) ** 2)

which is scale‑invariant.  We minimise the numerator (the maximal convolution
value) while keeping the denominator large.  Directly differentiating the
max is not possible, so we replace it by a smooth approximation

    smooth_max(x) = (1/β) * log Σ exp(β * x_i) ,

with a modest β (≈5).  The gradient of smooth_max with respect to the vector
a can be written using convolutions, which we compute efficiently with FFTs.
A projected (non‑negative) gradient step is taken repeatedly until the time
budget (≈1000 s) expires.  The best sequence seen during the run is returned.

The algorithm works for any length; we use the length of the existing
`height_sequence_1` when it is present, otherwise a default of 1000.
"""
import time
import numpy as np

def _fft_convolve(x: np.ndarray, y: np.ndarray, size: int) -> np.ndarray:
    """Linear convolution of x and y using FFT of given size."""
    X = np.fft.rfft(x, n=size)
    Y = np.fft.rfft(y, n=size)
    conv = np.fft.irfft(X * Y, n=size)
    return conv[: x.size + y.size - 1]


def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs):
    """
    Search for a sequence with a low value of evaluate_sequence().
    The function runs for at most `budget_s` seconds and returns the best
    sequence found (as a plain Python list of floats).
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 5   # a little safety margin

    # ------------------------------------------------------------------
    # 1. Choose a problem size
    # ------------------------------------------------------------------
    if "height_sequence_1" in globals():
        n = len(height_sequence_1)
    else:
        n = 1000                     # fallback length

    # target sum is irrelevant for the objective, but we keep a reasonable
    # magnitude to avoid numerical extremes
    target_sum = np.sqrt(2.0 * n)

    # ------------------------------------------------------------------
    # 2. Initialise a promising shape.
    #    Exponential decay a_i = exp(-c * i / n) gives a good starting value.
    # ------------------------------------------------------------------
    c_init = 1.0                     # parameter giving E ≈ 1.84 for large n
    idx = np.arange(n, dtype=float)
    a = np.exp(-c_init * idx / n)    # values in (e^{-c},1]
    # scale to a moderate total mass (scale does not affect the objective)
    a *= target_sum / a.sum()

    # pre‑compute FFT size – we need length up to (n) + (2n‑1) – 1 = 3n‑2
    fft_size = 1 << ((3 * n - 2).bit_length())

    # smoothing parameter for the soft‑max
    beta = 5.0

    # bookkeeping
    best_seq = a.copy()
    # evaluate the true objective (using the exact max)
    conv_full = _fft_convolve(a, a, fft_size)[: 2 * n - 1]
    best_score = 2.0 * n * conv_full.max() / (a.sum() ** 2)
    print(f"[init] n={n}  score={best_score:.6f}")

    # gradient‑descent hyper‑parameters
    lr = 0.1 * a.mean()          # initial step size
    decay = 0.9995               # slowly shrink the step
    iteration = 0

    while time.time() < deadline:
        # ------------------------------------------------------------------
        # 3. Compute smooth max and its gradient
        # ------------------------------------------------------------------
        # a*a (linear convolution)
        conv = _fft_convolve(a, a, fft_size)[: 2 * n - 1]

        # smooth approximation of the maximum
        # avoid overflow in exp by clipping
        scaled = beta * conv
        np.clip(scaled, a_min=None, a_max=700, out=scaled)   # safe upper bound
        exp_scaled = np.exp(scaled)
        sum_exp = exp_scaled.sum()
        smooth_max = np.log(sum_exp) / beta

        # weights for the gradient of the smooth max
        w = exp_scaled / sum_exp                     # length 2n‑1, non‑negative

        # gradient of smooth_max wrt a: 2 * (w * a)  (convolution)
        grad_smooth = 2.0 * _fft_convolve(w, a, fft_size)[:n]

        # ------------------------------------------------------------------
        # 4. Full objective gradient (chain rule)
        #    O(a) = 2 n * smooth_max / (sum(a)^2)
        # ------------------------------------------------------------------
        S = a.sum()
        coeff1 = (2.0 * n) / (S * S)
        coeff2 = (4.0 * n * smooth_max) / (S * S * S)
        grad = coeff1 * grad_smooth - coeff2 * np.ones_like(a)

        # ------------------------------------------------------------------
        # 5. Gradient step with projection onto the non‑negative orthant
        # ------------------------------------------------------------------
        a_new = a - lr * grad
        np.maximum(a_new, 0.0, out=a_new)            # enforce a_i >= 0

        # keep values bounded to avoid pathological growth (the evaluator
        # clamps to 1000 anyway, but staying well below that helps stability)
        np.clip(a_new, 0.0, 1000.0, out=a_new)

        # ------------------------------------------------------------------
        # 6. Evaluate the true (non‑smoothed) objective and keep the best
        # ------------------------------------------------------------------
        conv_new = _fft_convolve(a_new, a_new, fft_size)[: 2 * n - 1]
        score_new = 2.0 * n * conv_new.max() / (a_new.sum() ** 2)

        if score_new < best_score:
            best_score = score_new
            best_seq = a_new.copy()
            print(f"[{iteration:6d}] improved score = {best_score:.6f}")

        # accept the step if it does not explode; otherwise keep the old a
        if not np.any(np.isnan(a_new)):
            a = a_new

        # ------------------------------------------------------------------
        # 7. Simple learning‑rate schedule
        # ------------------------------------------------------------------
        lr *= decay
        iteration += 1

        # occasional random jitter to escape shallow plateaux
        if iteration % 5000 == 0:
            jitter = (np.random.rand(n) - 0.5) * lr * 0.5
            a += jitter
            np.maximum(a, 0.0, out=a)

    # ----------------------------------------------------------------------
    # 8. Final post‑processing: ensure a reasonable sum and clip bounds
    # ----------------------------------------------------------------------
    total = best_seq.sum()
    if total < 1e-6:                         # guard against an all‑zero vector
        best_seq = np.full_like(best_seq, target_sum / n)
    else:
        # rescale to the same total mass (purely cosmetic)
        best_seq = best_seq * (target_sum / total)

    # final clipping to satisfy the security checks of evaluate_sequence()
    best_seq = np.clip(best_seq, 0.0, 1000.0)

    print(f"[final] best score = {best_score:.6f}  length = {len(best_seq)}")
    return best_seq.tolist()
