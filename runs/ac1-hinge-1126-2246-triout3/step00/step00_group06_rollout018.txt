"""
Simple evolutionary search for a step‑function that minimises the
evaluation metric defined by ``evaluate_sequence``.

The algorithm works with a single candidate sequence (no explicit population)
and repeatedly applies two kinds of moves:

1. **Gradient step** – computes the convolution, finds the index/indices
   where the maximum occurs and moves the sequence a little in the negative
   gradient direction of those entries.  Afterwards the vector is projected
   back onto the probability simplex (non‑negative and sum = 1).

2. **Random mass‑transfer** – picks two positions i≠j, moves a small random
   amount of mass from i to j and re‑normalises.

Both moves keep the sequence non‑negative and preserve the L1‑norm, so the
objective value is invariant under scaling.  The best sequence seen during
the allotted time budget is returned.

The routine respects the time limit given by ``budget_s`` (default 1000 s)
and prints occasional progress updates.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities
# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray) -> np.ndarray:
    """Project a non‑negative vector onto the simplex (sum = 1)."""
    v = np.maximum(v, 0.0)
    total = v.sum()
    if total > 0.0:
        return v / total
    # fallback: uniform vector (should never happen in practice)
    return np.full_like(v, 1.0 / v.size)


def _random_simplex(n: int, rng: np.random.Generator) -> np.ndarray:
    """Draw a random point on the simplex using exponential variables."""
    x = rng.exponential(size=n)
    return x / x.sum()


def _gradient_step(seq: np.ndarray, step_factor: float = 0.1) -> np.ndarray:
    """
    Perform a single projected gradient step aimed at reducing the
    maximal convolution entry.  ``step_factor`` controls the step size.
    """
    n = seq.size
    conv = np.convolve(seq, seq)
    max_val = conv.max()
    # indices where the maximum is attained (usually a single index)
    max_idx = np.flatnonzero(conv == max_val)

    # gradient of the max‑convolution entry w.r.t. the sequence
    grad = np.zeros_like(seq)
    for k in max_idx:
        i_start = max(0, k - (n - 1))
        i_end = min(n - 1, k)
        i_range = np.arange(i_start, i_end + 1, dtype=int)
        # derivative contribution a_{k-i}
        grad[i_range] += seq[k - i_range]

    # Normalise the gradient direction to keep updates moderate
    norm = np.linalg.norm(grad) + 1e-12
    step = step_factor * (max_val / norm)

    new_seq = seq - step * grad
    new_seq = _project_onto_simplex(new_seq)
    return new_seq


def _random_transfer(seq: np.ndarray, rng: np.random.Generator,
                     max_fraction: float = 0.25) -> np.ndarray:
    """
    Move a random fraction of mass from one index to another.
    ``max_fraction`` bounds the proportion of the source entry that can be moved.
    """
    n = seq.size
    i, j = rng.integers(0, n, size=2)
    while i == j:
        j = rng.integers(0, n)

    # how much of a[i] to transfer (up to max_fraction of its current value)
    amount = seq[i] * rng.random() * max_fraction
    if amount <= 0.0:
        return seq  # nothing to do

    new_seq = seq.copy()
    new_seq[i] -= amount
    new_seq[j] += amount
    return _project_onto_simplex(new_seq)

# Main entry point required by the competition framework
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Return a sequence of non‑negative heights that (hopefully) improves upon
    the current best known bound.  The function runs for at most ``budget_s``
    seconds and returns the best sequence discovered.
    """
    rng = np.random.default_rng(seed)
    deadline = time.time() + budget_s - 0.5   # a small safety margin

    # ------------------------------------------------------------------
    # initialise – try to use the provided construction if available
    # ------------------------------------------------------------------
    if 'height_sequence_1' in globals() and isinstance(globals()['height_sequence_1'],
                                                       (list, np.ndarray)):
        base = np.asarray(globals()['height_sequence_1'], dtype=float)
        n = base.size
        # make sure the start point is feasible (non‑negative, sum > 0)
        if n == 0:
            n = 1500
            current = np.full(n, 1.0 / n)
        else:
            if (base < 0).any():
                base = np.abs(base)
            current = _project_onto_simplex(base)
    else:
        # fallback: uniform distribution of length 1500
        n = 1500
        current = np.full(n, 1.0 / n)

    best_seq = current.copy()
    try:
        best_score = evaluate_sequence(best_seq.tolist())
    except Exception:   # defensive – should not happen
        best_score = float('inf')
    print(f"[init] length={n}, initial score={best_score:.6f}")

    iteration = 0
    last_report = time.time()

    # ------------------------------------------------------------------
    # main search loop
    # ------------------------------------------------------------------
    while time.time() < deadline:
        iteration += 1

        # decide which move to use (70% gradient, 30% random transfer)
        if rng.random() < 0.7:

            candidate = _random_simplex(n, rng)

            score = evaluate_sequence(candidate.tolist())
        except Exception:

            last_report = time.time()

    print(f"[finished] total iterations={iteration}, best score={best_score:.6f}")
    # Return a plain Python list as required by the evaluator
    return best_seq.tolist()
