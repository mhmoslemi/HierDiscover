"""
Propose a candidate sequence for the step‑function problem.

For non‑negative entries the evaluation function simplifies to

    score = 2 * n * Σ a_i² / (Σ a_i)² .

By Cauchy–Schwarz, Σ a_i² ≥ (Σ a_i)² / n with equality iff all a_i are equal.
Hence the theoretical lower bound of the score is exactly **2**, attained by any
constant (uniform) sequence, irrespective of its length or scaling.

The function below returns such a uniform sequence (length 1500 to match the
statement) and performs a tiny random‑search loop only as a safeguard – it
cannot improve on the analytical optimum, but it guarantees the code respects
the allotted time budget.
"""

import time
import numpy as np

def propose_candidate(seed: int = 12345, budget_s: float = 1000, **kwargs):
    """
    Return a non‑negative height sequence that reaches the proven optimal
    value 2 for the given evaluation function.

    Parameters
    ----------
    seed : int, optional
        Random seed for the (optional) stochastic perturbation.
    budget_s : float, optional
        Total time allowed (seconds).  The algorithm needs only a tiny
        fraction of this budget because the optimum is known analytically.
    **kwargs
        Ignored – present for compatibility with the evaluation harness.

    Returns
    -------
    list[float]
        The candidate sequence.
    """
    np.random.seed(seed)

    # ------------------------------------------------------------------
    # 1.  Build the analytically optimal uniform sequence.
    # ------------------------------------------------------------------
    n = 1500                     # length requested by the problem statement
    uniform_value = 1.0          # any positive constant works
    best_seq = [uniform_value] * n
    best_score = evaluate_sequence(best_seq)
    print(f"[init] uniform sequence (n={n}) → score = {best_score:.12f}")

    # ------------------------------------------------------------------
    # 2.  Tiny stochastic search (mostly decorative – cannot beat the optimum).
    # ------------------------------------------------------------------
    deadline = time.time() + budget_s - 0.5   # keep a safety margin
    while time.time() < deadline:
        # Randomly pick a few coordinates and add zero‑mean noise.
        cand = np.array(best_seq, dtype=float)
        idx = np.random.choice(n, size=min(10, n), replace=False)
        cand[idx] += np.random.normal(scale=0.01, size=idx.shape)

        np.clip(cand, 0.0, 1000.0, out=cand)

        # Guard against the sum‑too‑small check.
        if cand.sum() < 0.01:
            continue

        # Evaluate the candidate.
        try:
            cand_score = evaluate_sequence(cand.tolist())
        except Exception:
            continue

        # Keep the best we have seen.
        if cand_score < best_score:
            best_score = cand_score
            best_seq = cand.tolist()
            print(f"[improve] new best score = {best_score:.12f}")

        # The analytical optimum is already reached; break early.
        if best_score <= 2.0000001:
            break

    print(f"[final] returning sequence of length {len(best_seq)} with score {best_score:.12f}")
