from scipy.optimize import linprog
from scipy import sparse

def solve_convolution_lp(f_seq, rhs):
    n = len(f_seq)
    # Build sparse matrix A_ub shape (2n-1 + n, n)
    rows = []
    cols = []
    data = []
    # convolution constraints
    for k in range(2*n-1):
        i_start = max(0, k-(n-1))
        i_end = min(n-1, k)
        for i in range(i_start, i_end+1):
            j = k - i
            rows.append(k)
            cols.append(j)
            data.append(f_seq[i])
    # nonnegativity constraints: -g <= 0 (or g >= 0)
    # We'll use bounds argument instead, simpler.
    A_ub = sparse.csr_matrix((data, (rows, cols)), shape=(2*n-1, n))
    b_ub = np.full(2*n-1, rhs)
    # Use variable bounds for nonnegativity
    bounds = [(0, None)] * n
    c = -np.ones(n)  # maximize sum -> minimize -sum
    result = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs', options={'time_limit': 10})
    if result.success:
        return result.x
    else:
        return None

def max_convolution(seq):
    n = len(seq)
    fft_len = 1 << (2 * n - 1).bit_length()
    fa = np.fft.rfft(seq, n=fft_len)
    conv = np.fft.irfft(fa * fa, n=fft_len)[:2*n-1]
    return conv.max()
```

"""
Propose a sequence of non‑negative heights that minimizes the given
evaluation function.

The algorithm is an improved version of the published line‑search /
LP approach:

1. Keep the sequence *normalized* – its ℓ₁‑norm is always
   sqrt(2·n).  With this normalisation the evaluation score equals the
   maximum entry of the (linear) autocorrelation a∗a, so minimising the
   score is equivalent to lowering that maximum.

2. At each iteration a linear programme is solved:
      maximise   Σ g_i
      subject to  (f ∗ g)[k] ≤ M   for a selected set of shifts k
                  g_i ≥ 0
   where M = max(f ∗ f) and f is the current (normalised) sequence.
   Only the *active* shifts – those where the convolution already attains
   the current maximum – are enforced, which keeps the LP small.
   If the resulting g violates any omitted constraint it is added to the
   active set (cut‑plane step).

3. The new sequence is taken as a convex combination
      f_new = (1−t)·f + t·g
   with the largest possible step‑size t that keeps the autocorrelation
   ≤ M (binary search on t).  Afterwards the result is re‑normalised to
   keep the ℓ₁‑norm fixed; this strictly decreases the evaluation score.

4. If the LP fails or the step is too tiny, a tiny random perturbation
   is applied to escape possible stalls.

The loop runs until the time budget is exhausted, keeping the best
sequence found.  The implementation uses a sparse matrix for the LP
constraints (SciPy ‘highs’ solver) and NumPy’s FFT for fast convolutions.
"""

import time
import numpy as np
from scipy.optimize import linprog
from scipy import sparse

# ----------------------------------------------------------------------
# Helper functions ------------------------------------------------------
# ----------------------------------------------------------------------


def _target_sum(n: int) -> float:
    """Desired ℓ₁‑norm for a sequence of length n."""
    return np.sqrt(2.0 * n)


def _normalize(seq: np.ndarray) -> np.ndarray:
    """Scale a non‑negative sequence to have the target ℓ₁‑norm."""
    n = seq.shape[0]
    s = seq.sum()
    if s == 0:
        # avoid division by zero – give a uniform tiny vector
        seq = np.full_like(seq, 1e-9)
        s = seq.sum()
    return seq * (_target_sum(n) / s)


def _full_convolution(seq: np.ndarray) -> np.ndarray:
    """Efficient linear convolution (full length) using FFT."""
    n = seq.shape[0]
    # length for FFT: next power of two ≥ 2n‑1
    fft_len = 1 << ((2 * n - 1).bit_length())
    fft_a = np.fft.rfft(seq, n=fft_len)
    conv = np.fft.irfft(fft_a * fft_a, n=fft_len)[: 2 * n - 1]
    # Numerical noise can produce tiny negative values
    return np.maximum(conv, 0.0)


def _solve_lp_active(f: np.ndarray, M: float, active_idx: np.ndarray) -> np.ndarray | None:
    """
    Solve the LP:
        maximise   Σ g_i
        subject to (f ∗ g)[k] ≤ M for all k in active_idx,
                   g_i ≥ 0.
    Returns the optimal g as a 1‑D NumPy array, or None on failure.
    """
    n = f.shape[0]
    # Build sparse coefficient matrix for the active constraints
    rows = []
    cols = []
    data = []
    for row, k in enumerate(active_idx):
        i_start = max(0, k - (n - 1))
        i_end = min(n - 1, k)
        # for i in [i_start, i_end] the term f[i] multiplies g[k-i]
        for i in range(i_start, i_end + 1):
            j = k - i
            rows.append(row)
            cols.append(j)
            data.append(f[i])
    if not rows:  # no active constraints (should not happen)
        return None
    A_ub = sparse.csr_matrix((data, (rows, cols)), shape=(len(active_idx), n))
    b_ub = np.full(len(active_idx), M, dtype=float)

    # maximise Σ g_i  ⇔ minimise −Σ g_i
    c = -np.ones(n, dtype=float)

    # Bounds enforce non‑negativity
    bounds = [(0.0, None)] * n

    # Use the modern HiGHS interior‑point solver; it accepts sparse matrices.
    res = linprog(
        c,
        A_ub=A_ub,
        b_ub=b_ub,
        bounds=bounds,
        method="highs",
        options={"time_limit": 5.0, "disp": False},
    )
    if res.success:
        return res.x
    return None


def _binary_search_step(f, g, conv_f, conv_fg, conv_gg, M, tol=1e-10):
    """
    Find the largest t∈[0,1] such that max( (1−t)²·conv_f
                                         + 2t(1−t)·conv_fg
                                         + t²·conv_gg ) ≤ M.
    """
    lo, hi = 0.0, 1.0
    for _ in range(30):  # enough for double precision
        mid = (lo + hi) * 0.5
        # convex combination of the three convolutions
        mixed = (1.0 - mid) ** 2 * conv_f + 2.0 * mid * (1.0 - mid) * conv_fg + mid ** 2 * conv_gg
        if mixed.max() <= M * (1.0 + 1e-12):
            lo = mid
        else:
            hi = mid
        if hi - lo < tol:
            break
    return lo


def _perturb(seq: np.ndarray, scale: float = 0.01) -> np.ndarray:
    """Apply a tiny random change to a randomly chosen entry."""
    new = seq.copy()
    i = np.random.randint(seq.size)
    new[i] = max(0.0, new[i] + np.random.randn() * scale)
    # Re‑normalise to keep the ℓ₁‑norm fixed
    return _normalize(new)


# ----------------------------------------------------------------------
# Main candidate generator -----------------------------------------------
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Return a sequence of non‑negative heights that (hopefully) has a lower
    evaluation score than the current best (≈ 2.0).

    The routine runs for roughly ``budget_s`` seconds, performing an
    accelerated LP‑based iterative optimisation described above.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5  # tiny safety margin

    # ------------------------------------------------------------------
    # 1) Initialise ----------------------------------------------------
    # ------------------------------------------------------------------
    try:
        # The competition supplies a global variable ``height_sequence_1``.
        base = np.array(height_sequence_1, dtype=float)
    except NameError:
        # Fallback: a simple uniform start of length 2000
        base = np.ones(2000, dtype=float)

    base = np.maximum(base, 0.0)               # enforce non‑negativity
    n = base.shape[0]
    f = _normalize(base)                       # normalised current sequence

    # Keep the best sequence seen so far
    best_seq = f.copy()
    best_score = evaluate_sequence(best_seq.tolist())
    print(f"[init] n={n}, initial score={best_score:.6f}")

    iteration = 0
    no_improve_cnt = 0

    # ------------------------------------------------------------------
    # 2) Main optimisation loop ----------------------------------------
    # ------------------------------------------------------------------
    while time.time() < deadline and iteration < 5000:
        iteration += 1

        # Current autocorrelation and its maximum
        conv_f = _full_convolution(f)
        M = conv_f.max()

        # --------------------------------------------------------------
        # 2a) Build the set of active constraints (indices where the bound is tight)
        # --------------------------------------------------------------
        eps_active = max(1e-12, 1e-8 * M)          # tolerance for “active”
        active_idx = np.nonzero(conv_f >= M - eps_active)[0]

        # --------------------------------------------------------------
        # 2b) Solve LP to obtain a direction g that enlarges Σg while respecting the bound
        # --------------------------------------------------------------
        g = _solve_lp_active(f, M, active_idx)

        # If LP failed (e.g. time‑out), fall back to a tiny random perturbation
        if g is None:
            f = _perturb(f)
            continue

        # --------------------------------------------------------------
        # 2c) Verify that the candidate respects *all* convolution constraints.
        #      If some were violated, add them to the active set and resolve
        #      (a simple one‑step cut‑plane strategy).
        # --------------------------------------------------------------
        conv_fg = _full_convolution(np.convolve(f, g, mode="full")[: 2 * n - 1])  # actually f∗g
        # The previous line is overly complicated; simpler:
        conv_fg = np.convolve(f, g, mode="full")
        if conv_fg.max() > M * (1.0 + 1e-9):
            # add the most violating index and retry the LP (once per iteration)
            viol = np.argmax(conv_fg)
            active_idx = np.unique(np.concatenate([active_idx, [viol]]))
            g = _solve_lp_active(f, M, active_idx)
            if g is None:
                f = _perturb(f)
                continue
            conv_fg = np.convolve(f, g, mode="full")

        # --------------------------------------------------------------
        # 2d) Determine maximal admissible mixing step t (binary search)
        # --------------------------------------------------------------
        conv_gg = _full_convolution(g)
        t = _binary_search_step(f, g, conv_f, conv_fg, conv_gg, M)

        # If the admissible step is extremely small, we try a random jitter
        if t < 1e-6:
            f = _perturb(f)
            continue

        # --------------------------------------------------------------
        # 2e) Form the new sequence and re‑normalise
        # --------------------------------------------------------------
        f_new = (1.0 - t) * f + t * g
        f_new = _normalize(f_new)

        # --------------------------------------------------------------
        # 2f) Evaluate – keep the best one seen
        # --------------------------------------------------------------
        score = evaluate_sequence(f_new.tolist())
        if score < best_score:
            best_score = score
            best_seq = f_new.copy()
            no_improve_cnt = 0
            print(f"[iter {iteration:4d}] improved score = {best_score:.6f}  (t={t:.4f})")
        else:
            no_improve_cnt += 1

        # --------------------------------------------------------------
        # 2g) Prepare for next iteration
        # --------------------------------------------------------------
        f = f_new

        # Simple restart logic – if we have stagnated for a while, re‑initialise
        if no_improve_cnt >= 200:
            # generate a fresh monotone start (random exponential decay)
            rnd = np.random.rand(n) ** 4
            rnd = np.sort(rnd)[::-1]               # decreasing
            f = _normalize(rnd)
            no_improve_cnt = 0
            print(f"[iter {iteration:4d}] restart due to stagnation")

    # ------------------------------------------------------------------
    # 3) Return the best sequence found
    # ------------------------------------------------------------------
    print(f"[done] best score = {best_score:.6f} after {iteration} iterations")
    return best_seq.tolist()
