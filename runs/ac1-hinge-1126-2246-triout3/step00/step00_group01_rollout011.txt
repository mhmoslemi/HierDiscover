def zigzag(arr):
    n=len(arr)
    res=np.empty(n)
    left=0
    right=n-1
    i=0
    while left<=right:
        if i%2==0:
            res[left]=arr[i]
            left+=1
        else:
            res[right]=arr[i]
            right-=1
        i+=1
    return res
```

def compute_score(seq):
    a = np.clip(seq, 0.0, 1000.0)
    n=len(a)
    sum_a = np.sum(a)
    if sum_a < 0.01:
        return float('inf')
    # FFT convolution
    m = int(2**np.ceil(np.log2(2*n-1)))
    fa = np.fft.rfft(a, m)
    conv = np.fft.irfft(fa*fa, m)[:2*n-1]
    max_b = np.max(conv)
    return float(2 * n * max_b / (sum_a**2))
```

import numpy as np
import time
import random

def compute_score(seq):
    a = np.clip(seq, 0.0, 1000.0)
    n = len(a)
    sum_a = np.sum(a)
    if sum_a < 0.01:
        return float('inf')
    # convolution via FFT
    m = int(2**np.ceil(np.log2(2*n-1)))
    fa = np.fft.rfft(a, m)
    conv = np.fft.irfft(fa*fa, m)[:2*n-1]
    max_b = np.max(conv)
    return float(2 * n * max_b / (sum_a**2))

"""
Propose a sequence of non‑negative heights that reduces the evaluation score.

The algorithm performs a stochastic search on sequences built from a simple
geometric decay model

    a_i = r**i , 0 <= i < n ,   (r < 1)

so that the largest element is 1 and all products a_i * a_j are bounded by 1.
For a given length `n` and decay factor `r` the sum of the entries is
approximately 1/(1‑r) while the maximum entry of the full convolution can be
made small by randomising the order of the values.  The search explores many
pairs (n, r), shuffles the generated values, and performs a short hill‑climbing
phase (random swaps / multiplicative perturbations) to further lower the peak
of the convolution.  The best sequence found before the time budget expires is
returned.

The score is computed with the same logic as `evaluate_sequence` but using an
FFT‑based convolution for speed.
"""

import time
import numpy as np
import random

# ----------------------------------------------------------------------
# Helper: fast evaluation that mirrors evaluate_sequence (without the safety checks)
def _fast_score(seq):
    """Return the evaluation score for a numeric sequence (list or 1‑D ndarray)."""
    a = np.clip(seq, 0.0, 1000.0)               # same clipping as evaluate_sequence
    n = a.size
    sum_a = a.sum()
    if sum_a < 0.01:                           # avoid division by zero
        return float('inf')
    # convolution by FFT (length rounded up to next power of two)
    m = int(2 ** np.ceil(np.log2(2 * n - 1)))
    fa = np.fft.rfft(a, m)
    conv = np.fft.irfft(fa * fa, m)[: 2 * n - 1]
    max_b = conv.max()
    return float(2.0 * n * max_b / (sum_a ** 2))

# ----------------------------------------------------------------------
def _anti_pair_permutation(values):
    """
    Produce a permutation that interleaves the sorted values from the two ends:
    largest, second‑largest, third‑largest, … placed alternately at the left
    and right side of the array.  This is only a cheap heuristic; the main
    optimisation comes from random shuffling later on.
    """
    sorted_vals = np.sort(values)[::-1]   # descending
    n = sorted_vals.size
    out = np.empty(n, dtype=sorted_vals.dtype)
    left, right = 0, n - 1
    for i, v in enumerate(sorted_vals):
        if i % 2 == 0:
            out[left] = v
            left += 1
        else:
            out[right] = v
            right -= 1
    return out

        n = np.random.randint(1500, 3000)  # moderate size (fits comfortably in memory)

            # propose a neighbour
            cand = cur_seq.copy()
            if np.random.rand() < 0.6:
                # swap two positions
                i, j = np.random.randint(n, size=2)
                cand[i], cand[j] = cand[j], cand[i]
            else:
                # multiplicative jitter on a single entry
                i = np.random.randint(n)
                factor = np.random.uniform(0.85, 1.15)
                cand[i] = max(0.0, cand[i] * factor)

            new_score = _fast_score(cand)

            # accept if better, otherwise with a Metropolis probability
            if new_score < cur_score or np.random.rand() < np.exp((cur_score - new_score) / max(temperature, 1e-12)):
                cur_seq = cand
                cur_score = new_score
                temperature *= 0.9995           # slow cooling
                if new_score < best_score:
                    best_seq = cand.tolist()
                    best_score = new_score
                    print(f"[improve] it={it}, n={n}, eps={eps:.5e}, score={new_score:.6f}")

            # occasional diversification: reshuffle completely
            if it % 400 == 0 and np.random.rand() < 0.3:
                np.random.shuffle(cur_seq)

        # after the inner loop we loop back to generate a fresh (n, r) pair

    # ------------------------------------------------------------------
    # final safety: if nothing succeeded, fall back to a simple uniform sequence
    if best_seq is None:
        best_seq = [1.0] * 10
        best_score = _fast_score(best_seq)
        print("[fallback] using uniform sequence, score=", best_score)

    print("Search finished – best score obtained:", best_score)
    return best_seq
