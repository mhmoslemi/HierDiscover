def project_simplex(v, s=1):
    # v: numpy array
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.where(u - cssv / np.arange(1,n+1) > 0)[0]
    if len(rho) == 0:
        theta = 0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0)
    return w

def full_convolution(a):
    n = a.shape[0]
    size = 2*n - 1
    N = 1 << ((size - 1).bit_length())   # power of two >= size
    # rfft returns N/2+1 real frequency bins
    fft_a = np.fft.rfft(a, n=N)
    conv = np.fft.irfft(fft_a * fft_a, n=N)[:size]
    return conv

def project_simplex(v, s=1.0):
    """Project onto the probability simplex {x >= 0, sum x = s}."""
    if np.sum(v) == s and np.all(v >= 0):
        return v
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.where(u - cssv / np.arange(1, n+1) > 0)[0]
    if len(rho) == 0:
        theta = 0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0)
    return w

Now conv via FFT:

def full_convolution(a):
    n = a.shape[0]
    size = 2 * n - 1
    N = 1 << ((size - 1).bit_length())
    fft_a = np.fft.rfft(a, n=N)
    conv = np.fft.irfft(fft_a * fft_a, n=N)[:size]
    return conv

def subgrad_move(a, eta):
    n = a.shape[0]
    conv = full_convolution(a)
    k = int(np.argmax(conv))
    grad = np.zeros_like(a)
    i_min = max(0, k - n + 1)
    i_max = min(n - 1, k)
    # indices i where i_min <= i <= i_max
    # compute a_{k-i}
    i_range = np.arange(i_min, i_max + 1)
    grad[i_range] = 2 * a[k - i_range]
    # take step
    v = a - eta * grad
    # projection onto simplex
    a_new = project_simplex(v)
    return a_new, conv[k]

"""
Sequence Optimiser for the self‑convolution bound.

We start from the analytically motivated shape a_i ∝ 1/√(i+1),
which makes the discrete convolution of the sequence almost flat
(and yields the theoretical bound ≈π/2 ≈ 1.57 for the evaluation
function).  From this seed we run a simple projected sub‑gradient
descent: at each step the index k where the convolution attains its
maximum is found, the gradient of that entry (∂(a*a)_k/∂a_i = 2 a_{k‑i})
is computed and the sequence is moved a little opposite to the gradient.
After the step we project back onto the probability simplex
(∑a_i = 1, a_i ≥ 0).  Random mass‑transfer perturbations and occasional
re‑initialisations are added to avoid stagnation.  The best sequence
found before the time budget expires is returned.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper functions (all top‑level, no closures)
# ----------------------------------------------------------------------


def project_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Project vector v onto the simplex {x ≥ 0, sum x = s}."""
    if s < 0:
        raise ValueError("Simplex radius must be non‑negative")
    n = v.shape[0]
    # Fast projection (see https://arxiv.org/abs/1101.6081)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def full_convolution(a: np.ndarray) -> np.ndarray:
    """Return the full convolution a * a using FFT (real inputs)."""
    n = a.shape[0]
    size = 2 * n - 1
    # next power of two for efficient FFT
    N = 1 << ((size - 1).bit_length())
    fft_a = np.fft.rfft(a, n=N)
    conv = np.fft.irfft(fft_a * fft_a, n=N)[:size]
    return conv


def subgrad_step(a: np.ndarray, eta: float) -> np.ndarray:
    """
    Perform one projected sub‑gradient step that tries to lower the
    current maximum convolution entry.
    """
    n = a.shape[0]
    conv = full_convolution(a)
    k = int(np.argmax(conv))                     # index of the worst entry
    # gradient of (a*a)_k w.r.t. a_i is 2 * a_{k-i} whenever the index is valid
    i_min = max(0, k - n + 1)
    i_max = min(n - 1, k)
    idx = np.arange(i_min, i_max + 1, dtype=int)
    grad = np.zeros_like(a)
    grad[idx] = 2.0 * a[k - idx]                 # vectorised assignment
    # gradient descent step followed by projection onto the simplex
    v = a - eta * grad
    a_new = project_simplex(v, s=1.0)
    return a_new


def evaluate_fast(a: np.ndarray) -> float:
    """
    Fast evaluation of the objective for a sequence whose entries sum to 1.
    Since sum(a)=1, evaluate_sequence = 2 * n * max(conv).
    """
    n = a.shape[0]
    max_b = full_convolution(a).max()
    return 2.0 * n * max_b


# ----------------------------------------------------------------------
# Main entry point required by the competition framework
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Return a list of non‑negative heights that (hopefully) improves on the
    current bound.  The algorithm:
      1. Initialise a_i ∝ 1/√(i+1) (the analytic near‑optimal shape).
      2. Project onto the simplex (∑a_i = 1).
      3. Run a time‑limited projected sub‑gradient descent with occasional
         random perturbations.
      4. Keep the best sequence seen and return it.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 5.0   # leave a small safety margin

    # ------------------------------------------------------------------
    # Choose a length.  Prefer the length of the previously discovered
    # construction if it exists, otherwise fall back to a modest size.
    # ------------------------------------------------------------------
    if 'height_sequence_1' in globals():
        n = len(height_sequence_1)
    else:
        n = 2000                     # reasonable default

    # ------------------------------------------------------------------

    # ------------------------------------------------------------------
    a = 1.0 / np.sqrt(np.arange(1, n + 1, dtype=float))
    a = a / a.sum()                 # normalise to sum = 1 (the simplex)

    # ------------------------------------------------------------------
    # Keep the best candidate found so far
    # ------------------------------------------------------------------
    best_a = a.copy()
    best_score = evaluate_fast(best_a)

    # also compare with the uniform distribution and the historic best
    uniform = np.full(n, 1.0 / n)
    uniform_score = evaluate_fast(uniform)
    if uniform_score < best_score:
        best_a, best_score = uniform.copy(), uniform_score

    if 'height_sequence_1' in globals():

    # ------------------------------------------------------------------

    cur_score = best_score

    return best_a.tolist()
