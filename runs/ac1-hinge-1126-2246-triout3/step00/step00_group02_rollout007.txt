"""
Search for a low‑value step‑function using the classic “LP‑improve” scheme.

The idea (originating from the reference code) is:
    * Start with a simple non‑negative sequence (uniform ones).
    * Let M = max( f * f )  (the worst autocorrelation value).
    * Solve a linear program:
          maximise   Σ b_j
          subject to   (f * b)_k ≤ M   for all k,
                     b_j ≥ 0.
      The LP finds a new sequence `b` that has a larger ℓ₁‑norm
      while never increasing the current maximal convolution value M.
    * Normalise the new sequence to the standard total weight √(2n)
      (this does not change the evaluation score because the score is
       homogeneous of degree 0).
    * Replace the current sequence by a convex combination of the old and
      the new one.  A relatively large mixing factor (here 0.5) gives fast
      descent while still keeping the LP feasible.
    * Repeat until time runs out or no further improvement is observed.

The algorithm is run for a few different lengths (200–500) and a few
random restarts; the best sequence found overall is returned.
"""

import time
import numpy as np
from scipy.optimize import linprog

# ----------------------------------------------------------------------
# Helper: build the linear‑program constraints for a given f and bound M
# ----------------------------------------------------------------------
def _build_lp_constraints(f, M):
    """
    Returns (A_ub, b_ub) for the LP:
        maximise  Σ b_j
        s.t.      (f * b)_k ≤ M    (k = 0 … 2n‑2)
                  b_j ≥ 0
    """
    n = len(f)
    # convolution constraints
    a_rows = []
    b_rows = []
    for k in range(2 * n - 1):
        row = np.zeros(n)
        # (f * b)_k = Σ_i f_i * b_{k-i}
        # coefficient of b_j is f_{k-j} if that index is valid
        # we iterate i (the index of f)
        for i in range(n):
            j = k - i
            if 0 <= j < n:
                row[j] = f[i]
        a_rows.append(row)
        b_rows.append(M)
    # non‑negativity: –b ≤ 0
    a_rows.append(-np.eye(n))
    b_rows.append(np.zeros(n))
    A_ub = np.vstack(a_rows)
    b_ub = np.hstack(b_rows)
    return A_ub, b_ub


# ----------------------------------------------------------------------
# Solve the LP (maximise sum(b) under the convolution bound)
# ----------------------------------------------------------------------
def _solve_lp(f, M, max_seconds=5.0):
    """
    Solve the LP described above.
    Returns the optimal b (as a NumPy array) or None on failure / timeout.
    """
    A_ub, b_ub = _build_lp_constraints(f, M)

    # objective: maximise Σ b_j  → minimise –Σ b_j
    c = -np.ones(len(f))

    # use the HiGHS solver (fast and robust)
    # we impose a modest time limit via the `options` argument.
    options = {"max_iter": 10_000, "time_limit": max_seconds}
    result = linprog(
        c,
        A_ub=A_ub,
        b_ub=b_ub,
        bounds=(0, None),
        method="highs",
        options=options,
    )
    if result.success:
        return result.x
    return None


# ----------------------------------------------------------------------
# Core search routine
# ----------------------------------------------------------------------
def propose_candidate(seed=42, budget_s=1000):
    """
    Produce a sequence that (hopefully) beats the 1.503 target.
    The routine runs several independent attempts with different lengths
    and random seeds, each using the LP‑improvement loop described above.
    The best sequence found before the time budget expires is returned.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 5  # keep a small safety margin

    # keep the globally best result
    best_seq = None
    best_score = float("inf")

    # try a few different problem sizes (shorter sequences are faster)
    lengths = [200, 300, 400, 500]

    for n in lengths:
        if time.time() >= deadline:
            break

        print(f"\n--- Starting optimisation for n = {n} ---")

        # --- initialise: uniform sequence (sum = sqrt(2n)) ---
        cur = np.ones(n)
        cur = cur * (np.sqrt(2 * n) / cur.sum())   # normalise sum

        # evaluate once
        cur_score = evaluate_sequence(cur.tolist())
        print(f"Initial score (n={n}): {cur_score:.6f}")

        # keep the best for this length
        best_local_seq = cur.copy()
        best_local_score = cur_score

        # parameters for the inner loop
        t_mix = 0.5          # mixing factor (new = (1‑t)old + t*g)
        no_improve = 0      # counts successive iterations without progress
        max_no_improve = 30 # give up after this many stagnant steps

        while time.time() < deadline:
            # current bound M
            conv = np.convolve(cur, cur)
            M = conv.max()

            # solve the LP
            g = _solve_lp(cur, M, max_seconds=2.0)
            if g is None:
                # LP failure – inject a tiny random perturbation and continue
                idx = np.random.randint(n)
                cur[idx] = max(0.0, cur[idx] + np.random.randn() * 1e-3)
                continue

            # normalise the candidate to the standard ℓ₁‑norm
            sum_g = g.sum()
            if sum_g < 1e-12:
                break
            g = g * (np.sqrt(2 * n) / sum_g)

            # convex combination (aggressive mixing)
            new_seq = (1.0 - t_mix) * cur + t_mix * g

            # evaluate
            new_score = evaluate_sequence(new_seq.tolist())

            # keep the best ever seen
            if new_score < best_local_score:
                best_local_score = new_score
                best_local_seq = new_seq.copy()
                no_improve = 0
                print(f"  Improved score: {new_score:.6f} (iter)")
            else:
                no_improve += 1

            # accept the new sequence for the next iteration
            cur = new_seq

            # stop early if we already beat the target
            if best_local_score < 1.503:
                break

            # give up if we have been stagnant for a while
            if no_improve >= max_no_improve:
                break

        # ------------------------------------------------------------------
        # After finishing this length, compare with the global best
        # ------------------------------------------------------------------
        print(
            f"Finished n={n}: best local score = {best_local_score:.6f}"
        )
        if best_local_score < best_score:
            best_score = best_local_score
            best_seq = best_local_seq.copy()
            print(f"*** New global best score: {best_score:.6f}")

        # stop completely if target already reached
        if best_score < 1.503:
            break

    # ------------------------------------------------------------------
    # Return the best sequence we have discovered (as a plain list)
    # ------------------------------------------------------------------
    if best_seq is None:
        # very unlikely – fall back to a simple uniform sequence
        best_seq = [1.0] * 200
    return best_seq.tolist()
