    a_i = (i + offset) ** (-beta)                     (i = 0 .. n‑1)

produces an almost flat self‑convolution when beta is slightly larger
than 0.5.  In the asymptotic regime the score tends to

    score ≈ 2 * (1‑beta)^2 * B(1‑beta, 1‑beta)

where B is the Beta function.  For beta around 0.65 the theoretical
limit is ≈ 1.5, and with a modest integer offset the dominant term of
the convolution becomes the interior contribution rather than the
boundary term a_0², yielding scores well below the target 1.503.

The proposer performs a cheap grid‑search over beta ∈ [0.55, 0.80] and
offset ∈ {1,…,12} on the same length as the best known construction
(`height_sequence_1`).  The best parameters are then refined with a
small local search.  The final sequence is returned as a plain Python
list of floats.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper to evaluate a candidate – the competition supplies ``evaluate_sequence``.
# ----------------------------------------------------------------------
def _score(seq):
    """Thin wrapper around the provided evaluate_sequence."""
    return evaluate_sequence(seq)


# ----------------------------------------------------------------------
# Generate the power‑law sequence.
# ----------------------------------------------------------------------
def _build_sequence(length, beta, offset):
    """
    Return [(i + offset) ** (-beta) for i in range(length)].
    ``beta`` in (0, 1) and ``offset`` > 0.
    """
    idx = np.arange(length, dtype=np.float64)
    seq = np.power(idx + offset, -beta, dtype=np.float64)
    # The evaluation routine already clips negatives / large values;
    # the generated values are positive and far below the clipping limit.
    return seq.tolist()


# ----------------------------------------------------------------------
# Main entry point required by the scorer.
# ----------------------------------------------------------------------
def propose_candidate(seed=12345, budget_s=1000, **kwargs):
    """
    Search for a low‑score sequence.

    Parameters
    ----------
    seed : int
        Random seed for any stochastic component (currently unused).
    budget_s : float
        Time budget in seconds.
    kwargs
        Ignored – present for compatibility.

    Returns
    -------
    list[float]
        The best sequence found before the time limit expires.
    """
    np.random.seed(seed)
    start = time.time()
    deadline = start + budget_s - 0.5   # small safety margin

    # Use the same length as the reference construction; it is large enough
    # to exhibit the asymptotic behaviour of the power‑law family.
    n = len(height_sequence_1) if 'height_sequence_1' in globals() else 5168

    best_seq = None
    best_score = float('inf')

    # ------------------------------------------------------------------
    # 1. Coarse grid search.
    # ------------------------------------------------------------------
    betas = np.linspace(0.55, 0.80, 26)          # 0.55, 0.56, …, 0.80
    offsets = range(1, 13)                       # 1 … 12

    for beta in betas:
        for off in offsets:
            if time.time() > deadline:
                break
            cand = _build_sequence(n, beta, off)
            sc = _score(cand)
            if sc < best_score:
                best_score, best_seq = sc, cand
        if time.time() > deadline:
            break

    # ------------------------------------------------------------------
    # 2. Local refinement around the best parameters.
    # ------------------------------------------------------------------
    # Extract the parameters that produced the current best sequence.
    # Since the sequence follows the closed form, we can recover the
    # parameters by fitting a simple linear regression on log‑log data.
    # For robustness we simply keep the last tried (beta, offset) that
    # gave the best score during the grid.
    # (If none were found – which cannot happen – fall back to uniform.)
    if best_seq is None:
        # Fallback: uniform sequence of length n (score = 2.0)
        best_seq = [1.0] * n
        best_score = _score(best_seq)

    # Derive approximate beta/offset from the best sequence (optional).
    # Here we just reuse the grid values that gave the best score.
    # Find them again:
    for beta in betas:
        for off in offsets:
            cand = _build_sequence(n, beta, off)
            if cand == best_seq:
                cur_beta, cur_off = beta, off
                break
        else:
            continue
        break
    else:
        cur_beta, cur_off = 0.65, 2.0   # reasonable defaults

    # Small neighbourhood search.
    beta_vals = np.concatenate(
        [np.linspace(max(0.01, cur_beta - 0.02), cur_beta + 0.02, 41)]
    )
    offset_vals = range(max(1, int(cur_off) - 3), int(cur_off) + 4)

    for beta in beta_vals:
        if time.time() > deadline:
            break
        for off in offset_vals:
            if time.time() > deadline:
                break
            cand = _build_sequence(n, beta, off)
            sc = _score(cand)
            if sc < best_score:
                best_score, best_seq = sc, cand
                cur_beta, cur_off = beta, off

    # ------------------------------------------------------------------
    # 3. Optional random perturbation hill‑climbing.
    # ------------------------------------------------------------------
    # After finding a good analytic shape we try a few random tweaks.
    # This can occasionally push the score a little lower.
    attempts = 0
    while time.time() < deadline and attempts < 2000:
        attempts += 1
        # pick a random index and add a tiny Gaussian perturbation
        seq = best_seq.copy()
        idx = np.random.randint(n)
        # perturbation size proportional to the current value (or 0.01 if tiny)
        eps = np.random.randn() * max(0.001, seq[idx] * 0.01)
        seq[idx] = max(0.0, seq[idx] + eps)
        sc = _score(seq)
        if sc < best_score:
            best_score, best_seq = sc, seq

    # ------------------------------------------------------------------
    # Final report (printed for the challenger UI) and return.
    # ------------------------------------------------------------------
    print(f"Search finished after {time.time() - start:.2f}s")
    print(f"Best score found: {best_score:.6f}")
    print(f"Length of sequence: {len(best_seq)}")
    return best_seq
