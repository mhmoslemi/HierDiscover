def propose_candidate(seed=42, budget_s=1000, **kwargs):
    """
    Random search + greedy hill‑climbing for a low‑autocorrelation binary step function.
    The objective is   2·n·max(conv(a,a)) / (sum(a))².
    The algorithm repeatedly:
        * draws a random length n (≈500‑8000) and a random density,
        * builds a binary vector of that length,
        * evaluates its score using an FFT‑based convolution,
        * performs a short greedy swap‑based hill‑climb,
        * remembers the best vector seen.
    The search stops a few seconds before the time limit and returns the best vector.
    """
    import numpy as np, time, random, math

    # fast score based on FFT (identical to evaluate_sequence up to rounding)
    def fast_score(seq):
        a = np.asarray(seq, dtype=np.float64)
        n = a.shape[0]
        sum_a = a.sum()
        if sum_a < 1e-12:
            return math.inf
        size = 1 << ((2 * n - 1).bit_length())
        A = np.fft.rfft(a, size)
        conv = np.fft.irfft(A * A, size)[:2 * n - 1]
        max_b = conv.max()
        return float(2 * n * max_b / (sum_a * sum_a))

    np.random.seed(seed)
    random.seed(seed + 12345)

    deadline = time.time() + budget_s - 5.0   # spare a few seconds for clean‑up
    best_seq = None
    best_val = math.inf

    # if a previous construction exists, seed the search with it
    try:
        if 'height_sequence_1' in globals():
            seq0 = [float(x) for x in height_sequence_1]
            val0 = fast_score(seq0)
            best_seq, best_val = seq0, val0
            print(f"seeded from height_sequence_1: score={best_val:.6f}")
    except Exception:
        pass

    # main random‑restart loop
    while time.time() < deadline:
        # pick a length and a target number of ones
        n = np.random.randint(800, 5000)          # moderate size – fast FFT and enough room for randomisation
        density = np.random.uniform(0.2, 0.8)      # try a variety of densities
        m = max(1, int(round(density * n)))      # number of ones

        seq = np.zeros(n, dtype=np.float64)
        ones_idx = np.random.choice(n, m, replace=False)
        seq[ones_idx] = 1.0

        cur_val = fast_score(seq)
        if cur_val < best_val:
            best_val, best_seq = cur_val, seq.tolist()

def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Randomised search + tiny greedy hill‑climbing for a low‑autocorrelation
    step function.

    The evaluation that the platform uses is
        2·n·max(conv(a,a)) / (sum(a))² .
    For a binary (0/1) vector the sum equals the number of 1‑entries.
    A random placement of the 1‑entries makes the convolution roughly flat,
    giving a score close to 1 – well below the target 1.503.

    Algorithm
    ----------
    1. Initialise the RNG.
    2. If a previous construction ``height_sequence_1`` exists, use it as the
       first incumbent.
    3. Repeatedly:
       * Pick a random length n (≈ 800 … 5000) and a random density ρ∈[0.2,0.8].
       * Build a binary vector of length n with exactly m=⌈ρ·n⌉ ones.
       * Evaluate the vector with a fast FFT‑based convolution (identical to
         ``evaluate_sequence`` up to rounding).
       * If it improves the best score, remember it.
       * Perform a short greedy hill‑climb: repeatedly swap a 1 with a 0.
         Keep the swap only if the score improves.  Stop after a few
         successful swaps or after ~0.4 s.
    4. Stop a few seconds before the deadline and return the best vector found.

    The search runs entirely within the given time budget and never uses
    filesystem or network I/O.
    """
    import math
    import time
    import random

    import numpy as np

    # ----------------------------------------------------------------------
    # Helper: fast evaluation (identical to ``evaluate_sequence`` apart from
    # rounding) using an FFT‑based convolution.
    def fast_score(seq):
        a = np.asarray(seq, dtype=np.float64)
        n = a.shape[0]
        sum_a = a.sum()
        if sum_a < 1e-12:
            return math.inf
        # next power of two ≥ 2·n‑1  (FFT size)
        size = 1 << ((2 * n - 1).bit_length())
        A = np.fft.rfft(a, size)
        conv = np.fft.irfft(A * A, size)[: 2 * n - 1]
        max_b = conv.max()
        return float(2 * n * max_b / (sum_a * sum_a))

    # ----------------------------------------------------------------------
    # Initialise RNGs
    np.random.seed(seed)
    random.seed(seed + 12345)

    # Safety margin – finish a few seconds early.
    deadline = time.time() + budget_s - 5.0

    best_seq = None
    best_val = math.inf

    # If the platform supplied a previous construction, seed the search with it.
    try:
        if "height_sequence_1" in globals():
            cand = [float(x) for x in height_sequence_1]
            val = fast_score(cand)
            best_seq, best_val = cand, val
            print(f"[seed] height_sequence_1 score ≈ {best_val:.6f}")
    except Exception:
        pass

    # ----------------------------------------------------------------------
    # Main random‑restart loop
    while time.time() < deadline:
        # 1) pick length and density
        n = np.random.randint(800, 5000)               # moderate size – fast FFT
        density = np.random.uniform(0.2, 0.8)           # aim for varied sparsity
        m = max(1, int(round(density * n)))            # number of ones (≥1)

        # 2) binary vector with exactly m ones
        seq = np.zeros(n, dtype=np.float64)
        ones_idx = np.random.choice(n, m, replace=False)
        seq[ones_idx] = 1.0

        cur_val = fast_score(seq)
        if cur_val < best_val:
            best_val, best_seq = cur_val, seq.tolist()
            print(f"[new global] random init n={n} m={m} score={best_val:.6f}")

        # --------------------------------------------------------------
        # 3) tiny greedy hill‑climb (swap‑based). Stop early if we already
        #    beat the target (1.503) to save time.
        if best_val < 1.4:
            # we are already comfortably below the target – abort early
            break

        # sets of indices for fast membership testing
        ones = set(ones_idx.tolist())
        zeros = set(range(n)) - ones

        # limit local work to ≈0.4 s or 200 successful swaps, whichever comes first
        local_deadline = time.time() + 0.4
        successful_swaps = 0
        while time.time() < local_deadline and successful_swaps < 200:
            i_off = random.choice(tuple(ones))
            i_on = random.choice(tuple(zeros))

            # propose swap
            seq[i_off] = 0.0
            seq[i_on] = 1.0
            new_val = fast_score(seq)

            if new_val < cur_val:
                # accept the improvement
                cur_val = new_val
                successful_swaps += 1
                ones.remove(i_off)
                ones.add(i_on)
                zeros.remove(i_on)
                zeros.add(i_off)

                if cur_val < best_val:
                    best_val, best_seq = cur_val, seq.tolist()
                    print(f"[new global] local improve n={n} score={best_val:.6f}")
                    if best_val < 1.4:
                        break
            else:
                # reject – revert
                seq[i_off] = 1.0
                seq[i_on] = 0.0

        # end of one random restart
    # ----------------------------------------------------------------------
    # Fallback – a trivial admissible sequence
    if best_seq is None:
        best_seq = [1.0]
        best_val = fast_score(best_seq)

    print(f"[final] best length={len(best_seq)}  score≈{best_val:.6f}")
    return best_seq
