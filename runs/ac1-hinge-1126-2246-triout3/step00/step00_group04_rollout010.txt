def reduce_peak(seq):
    c = np.convolve(seq, seq)
    k = np.argmax(c)
    # find pairs
    pairs = [(i, k-i) for i in range(len(seq)) if 0 <= k-i < len(seq)]
    # compute products
    prods = [seq[i]*seq[j] for i,j in pairs]
    # pick a pair with highest product
    idx = np.argmax(prods)
    i,j = pairs[idx]
    # compute delta
    delta = 0.01 * max(seq[i], seq[j])  # small
    # reduce both
    seq[i] = max(0.0, seq[i] - delta)
    seq[j] = max(0.0, seq[j] - delta)
    # allocate mass to random other indices
    allocate = delta * 2
    # choose random index not i or j
    k_indices = np.random.choice([idx for idx in range(len(seq)) if idx not in (i,j)], size=1)
    seq[k_indices[0]] += allocate
    return seq
```

def mix_sequences_best_t(f, g):

    n = len(arr)
    conv = np.convolve(arr, arr)
    max_idx = int(np.argmax(conv))
    # compute contributions per i
    # For each i where j= max_idx - i, in range [0,n)
    # product = arr[i] * arr[j]
    # we want to reduce entries with high a_i maybe.

    candidates = []
    for i in range(max_idx+1):
        j = max_idx - i
        if i < n and j < n:
            prod = arr[i] * arr[j]
            candidates.append((prod, i, j))
    # sort descending by prod
    candidates.sort(reverse=True)
    prod_i, i, j = candidates[0]
    # We will reduce both i and j by factor.
    # Compute delta = min(arr[i], arr[j]) * 0.1
    delta_i = arr[i] * 0.05

    low_idx = np.argmin(arr)
    # copy arr to new
    new_arr = arr.copy()
    new_arr[i] = max(0.0, arr[i] - delta_i)
    new_arr[j] = max(0.0, arr[j] - delta_j)
    new_arr[low_idx] = new_arr[low_idx] + delta_i + delta_j

def local_step(arr, reduction=0.05):
    n = len(arr)
    conv = np.convolve(arr, arr)
    k_max = int(np.argmax(conv))
    # find contributions list
    idxs = []
    for i in range(n):
        j = k_max - i
        if 0 <= j < n:
            idxs.append((i, j, arr[i]*arr[j]))
    # sort by product descending
    idxs.sort(key=lambda x: x[2], reverse=True)
    # pick top pair
    i, j, prod = idxs[0]
    # compute delta to move from both i and j
    delta_i = arr[i] * reduction
    delta_j = arr[j] * reduction
    # ensure not exceed current values
    delta_i = min(delta_i, arr[i])
    delta_j = min(delta_j, arr[j])
    total_delta = delta_i + delta_j
    if total_delta <= 0:
        return arr.copy()
    # find low target indices (where arr is minimal)
    # choose maybe 5 smallest indices
    low_indices = np.argsort(arr)[:5]
    # distribute total_delta uniformly among those low indices
    new_arr = arr.copy()
    new_arr[i] -= delta_i
    new_arr[j] -= delta_j
    # distribute equally
    add_per = total_delta / len(low_indices)
    new_arr[low_indices] += add_per
    # clip
    np.clip(new_arr, 0.0, 1000.0, out=new_arr)
    return new_arr

def balance_step(arr, reduction=0.05, frac=0.05):
    n = len(arr)
    sorted_idx = np.argsort(arr)
    n_frac = max(1, int(frac * n))
    low_idx = sorted_idx[:n_frac]

    add_per = total_removed / n_frac

"""
Propose a step‐function sequence that minimizes the supplied
evaluate_sequence() metric.

The algorithm is a fast, gradient‑free local search:

1.  Start from two seeds – the known construction ``height_sequence_1``
    (if it exists) and a random vector of the same length.
2.  Keep the best of the seeds.
3.  Repeatedly improve the current vector with three cheap moves:
      •  *gradient step* – a simple line‑search on the sub‑gradient of the
         maximal convolution value,
      •  *balance step* – move a small fraction of the mass from the
         largest entries to the smallest ones,
      •  *random jitter* – Gaussian perturbation of a few coordinates.
4.  Accept a move only if it strictly improves the objective.
5.  The loop runs until the time budget (≈ 1000 s) expires.

The code uses only NumPy and the already imported ``evaluate_sequence``.
All arithmetic is performed with NumPy arrays for speed; a tiny wrapper
``fast_evaluate`` reproduces the behaviour of ``evaluate_sequence`` without
the security overhead.
"""

import time
import numpy as np

# --------------------------------------------------------------
# Helper – fast replica of evaluate_sequence (no security checks)
def fast_evaluate(seq):
    """
    Compute the same value as evaluate_sequence but without the
    input‑validation overhead.  The function clamps the values to the
    interval [0, 1000] exactly as evaluate_sequence does.
    """
    arr = np.clip(seq, 0.0, 1000.0)
    s = np.sum(arr)
    if s < 0.01:                     # protect against division by zero
        return float('inf')
    conv = np.convolve(arr, arr)     # linear convolution (real → real)
    max_b = np.max(conv)
    n = arr.shape[0]
    return float(2.0 * n * max_b / (s * s))


# --------------------------------------------------------------
# Move 1 – gradient descent on the max‑convolution sub‑gradient
def _grad_step(arr, lr):
    """One gradient‑descent step with a fixed learning rate."""
    n = arr.shape[0]
    conv = np.convolve(arr, arr)
    k = int(np.argmax(conv))
    M = conv[k]
    S = np.sum(arr)

    # sub‑gradient of max(conv) w.r.t. a_i  (zero outside the support of k)
    g = np.zeros_like(arr)
    idx = k - np.arange(n)
    mask = (idx >= 0) & (idx < n)
    g[mask] = 2.0 * arr[idx[mask]]

    # ∂O/∂a = (2 n / S^2) * (g - 2 M / S * a)
    coeff = (2.0 * n) / (S * S)
    grad = coeff * (g - (2.0 * M / S) * arr)

    # take a step, keep non‑negative, re‑normalise the L1‑sum to stay
    # on the same scale (scale does not affect the objective).
    new = arr - lr * grad
    new = np.clip(new, 0.0, 1000.0)
    new_sum = np.sum(new)
    if new_sum > 0.0:
        new *= (S / new_sum)
    return new


def gradient_step(arr):
    """
    Try a handful of learning rates and keep the best improvement.
    Returns a new array (may be identical to the input if no improvement).
    """
    best_arr = arr
    best_val = fast_evaluate(arr)
    # different learning rates – the first successful one is usually enough
    for lr in (1.0, 0.5, 0.2, 0.1, 0.05, 0.02, 0.01):
        cand = _grad_step(arr, lr)
        val = fast_evaluate(cand)
        if val < best_val:
            best_val = val
            best_arr = cand
    return best_arr


# --------------------------------------------------------------
# Move 2 – move mass from the largest entries to the smallest ones
def balance_step(arr, frac=0.05, reduction=0.03):
    """
    Transfer a small fraction of the mass of the largest elements to the
    smallest elements.  ``frac`` controls how many entries are taken from
    each side (as a fraction of the length) and ``reduction`` is the
    relative amount taken from the high side.
    """
    n = arr.shape[0]
    # number of indices that participate in the transfer
    k = max(1, int(frac * n))

    # indices of the smallest / largest values
    order = np.argsort(arr)
    low = order[:k]
    high = order[-k:]

    # amount taken from each high entry
    delta = arr[high] * reduction
    total_removed = np.sum(delta)

    new = arr.copy()
    new[high] -= delta
    if k > 0:
        add = total_removed / k
        new[low] += add

    np.clip(new, 0.0, 1000.0, out=new)
    return new


# --------------------------------------------------------------
# Move 3 – small random jitter on a handful of coordinates
def jitter_step(arr, sigma=0.06, n_change=3):
    """Add Gaussian noise to a few randomly chosen indices."""
    new = arr.copy()
    n = arr.shape[0]
    for _ in range(n_change):
        i = np.random.randint(n)
        # noise scaled to the current magnitude (avoids excessive steps)
        scale = max(0.1, new[i])
        new[i] = max(0.0, new[i] + np.random.randn() * sigma * scale)
    np.clip(new, 0.0, 1000.0, out=new)
    return new

        cur = rand.copy()
    else:
        cur = base.copy()

    best_seq = cur.copy()
    best_score = fast_evaluate(best_seq)

    it = 0
    # ------------------------------------------------------------------
    # 2) main optimisation loop
    while time.time() < deadline:
        it += 1

        r = np.random.rand()
        if r < 0.55:               # most often try a gradient move
            cand = gradient_step(cur)
        elif r < 0.80:             # balance high / low values
            cand = balance_step(cur, frac=0.07, reduction=0.04)
        else:                       # occasional random jitter
            cand = jitter_step(cur, sigma=0.08, n_change=4)

        cand_score = fast_evaluate(cand)

        # accept only improvements (strictly better)
        if cand_score < best_score:
            best_score = cand_score
            best_seq = cand.copy()
            cur = cand.copy()
            print(f"Iter {it:6d}: new best = {best_score:.6f}")

        # keep moving from the current best even if a trial fails
        elif cand_score < fast_evaluate(cur):
            cur = cand.copy()

        # every few hundred iterations try a fresh random perturbation
        if it % 300 == 0:
            cur = best_seq.copy()
            # inject a mild random shake to escape possible plateaus
            cur = jitter_step(cur, sigma=0.12, n_change=6)

    print(f"Search finished after {it} iterations – best score {best_score:.6f}")
    # evaluate_sequence will be called by the grader; we only need to return a list
    return best_seq.tolist()
```
