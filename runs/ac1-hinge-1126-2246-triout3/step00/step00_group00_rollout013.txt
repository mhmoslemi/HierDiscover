def random_perturb(seq):
    n = len(seq)
    i, j = np.random.choice(n, size=2, replace=False)
    delta = np.random.rand() * 0.01 * seq[i]  # small proportion of i value
    if seq[i] >= delta:
        seq[i] -= delta
        seq[j] += delta
    else:
        # move all from i to j
        seq[j] += seq[i]
        seq[i] = 0
    return seq

def solve_convolution_lp(f_seq, M):
    n = len(f_seq)
    c = -np.ones(n)
    A_ub = []
    b_ub = []
    for k in range(2*n - 1):
        row = np.zeros(n)
        for i in range(n):
            j = k - i
            if 0 <= j < n:
                row[j] = f_seq[i]
        A_ub.append(row)
        b_ub.append(M)
    A_ub = np.array(A_ub)
    b_ub = np.array(b_ub)
    bounds = [(0, None)] * n
    # Solve with high speed method
    result = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs', options={'presolve': True, 'time_limit': 5})
    if result.success:
        return result.x
    return None
```

"""
search.py – Minimise the evaluation function for a non‑negative step function.

The algorithm works as follows:

1.  Initialise a random candidate (or reuse the provided SOTA sequence).
    The sequence is normalised so that its ℓ₁‑norm equals √(2·n); this
    does not affect the objective because the evaluation is scale‑invariant.

2.  Repeatedly improve the candidate.
    * Given the current sequence `f`, compute its maximal self‑convolution
      `M = max(f*f)`.
    * Solve the linear programme

          maximise   Σ b_i
          subject to  (f * b)_k ≤ M   for all k = 0 … 2n‑2
                      b_i ≥ 0

      The LP returns a direction `g` that never exceeds the current
      bound `M`.  After the LP we rescale `g` to the same ℓ₁‑norm as `f`.

    * Mix `f` and `g` with a set of trial mixing coefficients `t∈[0,1]`
      (e.g. 0.05, 0.1, 0.2, 0.5, 0.8).  The mixed sequence
      `h = (1‑t)·f + t·g` is evaluated and the best improving mixture is
      kept.

    * If the LP fails, or no mixing improves the score, a tiny random
      perturbation (mass transfer between two coordinates) is applied so
      that the search does not stall.

3.  The whole process is repeated with several random restarts until the
    allotted time (≈ 1000 s) expires.  The best sequence seen overall is
    returned.

The heavy work is the LP (∼ O(n²) constraints).  For speed the length is
kept moderate (200 – 1500) which is sufficient – the theoretical lower
bound of the objective approaches 1 as *n* grows, and values well below
the naïve uniform bound 2 are reachable.

All helper functions are top‑level, contain no closures, and print
periodic progress messages.
"""

import time
import numpy as np
from scipy.optimize import linprog

# ----------------------------------------------------------------------
# Linear programme: maximise sum(b) with convolution bound.
# ----------------------------------------------------------------------
def solve_convolution_lp_max(f_seq, M, time_limit=5.0):
    """
    Solve
        maximise   Σ b_i
        s.t.       (f * b)_k ≤ M   for all 0 ≤ k < 2·n‑1
                   b_i ≥ 0

    Returns the optimal b as a NumPy array, or None if the solver fails.
    """
    n = len(f_seq)
    # objective: minimise -sum(b)  === maximise sum(b)
    c = -np.ones(n)

    # Build the inequality matrix A_ub * b ≤ b_ub
    # each row k corresponds to Σ_i f_i * b_{k‑i} ≤ M
    rows = np.zeros((2 * n - 1, n), dtype=float)
    for k in range(2 * n - 1):
        i_start = max(0, k - (n - 1))
        i_end = min(n - 1, k)
        i = np.arange(i_start, i_end + 1, dtype=int)
        j = k - i
        rows[k, j] = f_seq[i]
    A_ub = rows
    b_ub = np.full(2 * n - 1, M, dtype=float)

    bounds = [(0.0, None) for _ in range(n)]

    # Use HiGHS (default) – fast and robust.
    opts = {"presolve": True, "time_limit": time_limit}
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds,
                  method="highs", options=opts)

    if res.success:
        return res.x
    return None


# ----------------------------------------------------------------------
# Small random perturbation – keeps total sum unchanged.
# ----------------------------------------------------------------------
def random_perturb(arr, magnitude=0.02):
    """Transfer a small random amount of mass between two coordinates."""
    n = len(arr)
    i, j = np.random.choice(n, size=2, replace=False)
    delta = magnitude * np.abs(arr[i]) * np.random.rand()
    if arr[i] < delta:   # avoid negative
        delta = arr[i]
    arr[i] -= delta
    arr[j] += delta
    return arr


# ----------------------------------------------------------------------
# Core optimisation loop for a single starting sequence.
# ----------------------------------------------------------------------
def improve_sequence(start_seq, deadline):
    """
    Perform alternating LP+mixing steps starting from `start_seq`
    until `deadline` is reached.  Returns the best sequence found and
    its score.
    """
    seq = np.asarray(start_seq, dtype=float)
    n = len(seq)
    target_sum = np.sqrt(2.0 * n)

    # Normalise to the target ℓ₁‑norm (scale invariant)
    seq *= target_sum / np.sum(seq)

    best_seq = seq.copy()
    best_score = evaluate_sequence(best_seq.tolist())

    # mixing coefficients tried each iteration
    mix_factors = [0.05, 0.1, 0.2, 0.5, 0.8]

    it = 0
    while time.time() < deadline - 1.0:
        it += 1
        # current bound
        conv_self = np.convolve(seq, seq)
        M = np.max(conv_self)

        # --------------------------------------------------------------
        # LP direction
        # --------------------------------------------------------------
        g = solve_convolution_lp_max(seq, M, time_limit=4.0)
        if g is None:
            # fallback: tiny random walk
            seq = random_perturb(seq, magnitude=0.01)
            continue

        # Rescale g to the same ℓ₁‑norm as seq
        sum_g = np.sum(g)
        if sum_g <= 0:
            seq = random_perturb(seq, magnitude=0.01)
            continue
        g = g * (target_sum / sum_g)

        # --------------------------------------------------------------
        # Try several convex combinations of f and g
        # --------------------------------------------------------------
        improved = False
        for t in mix_factors:
            cand = (1.0 - t) * seq + t * g
            # ensure non‑negative (numerical noise)
            cand = np.maximum(cand, 0.0)
            cand_score = evaluate_sequence(cand.tolist())
            if cand_score < best_score:
                best_score = cand_score
                best_seq = cand.copy()
                seq = cand   # continue from the improvement
                improved = True
                print(f"[{time.time():.1f}] iter {it:4d} | t={t:.2f} "
                      f"→ new best {best_score:.6f}")
                break   # accept first improvement

        if not improved:
            # No mixing improved the objective – try a small random tweak.
            seq = random_perturb(seq, magnitude=0.01)

    return best_seq.tolist(), best_score


# ----------------------------------------------------------------------
# Main entry point expected by the grader.
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Return a list of non‑negative heights that (hopefully) yields a low
    value of `evaluate_sequence`.  The routine runs for at most
    `budget_s` seconds and performs a few random restarts, each of which
    executes the LP‑based improvement loop described above.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 2.0   # safety margin

    # keep the overall best across restarts
    overall_best_seq = None
    overall_best_score = float("inf")

    restart = 0
    while time.time() < deadline - 5.0:
        restart += 1
        # ----- initialise a starting sequence ------------------------
        if 'height_sequence_1' in globals() and np.random.rand() < 0.5:
            init = np.asarray(height_sequence_1, dtype=float)
        else:
            n = np.random.randint(200, 1200)      # moderate size for speed
            init = np.random.random(n)

        # run the optimiser on this start
        seq, sc = improve_sequence(init, deadline)

        if sc < overall_best_score:
            overall_best_score = sc
            overall_best_seq = seq
            print(f"[{time.time():.1f}] restart {restart} → "
                  f"new overall best {overall_best_score:.6f} (len={len(seq)})")

        # stop early if we already hit a very good bound
        if overall_best_score < 1.55:   # a comfortable cushion above target
            break

    if overall_best_seq is None:
        # safety fallback – uniform sequence (never infinite)
        n = 500
        uniform = np.full(n, np.sqrt(2 * n) / n)
        overall_best_seq = uniform.tolist()
        overall_best_score = evaluate_sequence(overall_best_seq)

    print(f"Finished. Best score = {overall_best_score:.6f}, length = {len(overall_best_seq)}")
    return overall_best_seq
