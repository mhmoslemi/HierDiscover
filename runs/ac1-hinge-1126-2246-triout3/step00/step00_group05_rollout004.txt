"""
Propose a step function sequence with a low value of the given
evaluation metric.

The algorithm is a simple stochastic gradient descent /
simulated‑annealing hybrid:

* Initialise a random non‑negative vector a of a chosen length n.
  The vector is normalised to have unit sum (the metric is scale‑
  invariant, so any normalisation works).

* In each iteration we look at the current convolution c = a∗a,
  find the index k where c is maximal and compute the gradient of
  that max‑convolution term w.r.t. a:
        ∂c_k/∂a_i = 2·a_{k‑i}   (if the index k‑i exists, else 0).

  A direction d that reduces the maximal convolution while keeping the
  total sum unchanged is obtained by
        d = –grad + mean(grad)·1   (so Σ d_i = 0).

* A candidate a' = a + α·d is built, negative components are clipped to
  zero and the vector is renormalised to unit sum.  If the candidate
  improves the score it is accepted and the step size α is slowly
  increased; otherwise α is decreased.

* To avoid getting stuck in symmetric points (e.g. the uniform vector)
  a small random perturbation is injected whenever the gradient direction
  becomes too small or after a configurable number of non‑improving
  steps.  Occasionally the whole vector is re‑initialised (restart).

The loop runs until the allocated time (≈ budget_s seconds) expires.
The best sequence seen so far is returned.
"""

import time
import numpy as np

def _gradient_direction(a: np.ndarray, k_max: int) -> np.ndarray:
    """
    Compute the gradient of the convolution term at index k_max
    w.r.t. the vector a and return a sum‑preserving descent direction.
    """
    n = a.shape[0]
    # contribution only when the partner index is inside the array
    i = np.arange(n)
    j = k_max - i
    mask = (j >= 0) & (j < n)

    grad = np.zeros_like(a)
    # derivative of c_k = Σ a_i a_{k-i}  → 2·a_{k-i}
    grad[mask] = 2.0 * a[j[mask]]

    # make the direction have zero sum (preserve total mass)
    mu = grad.mean()
    direction = -grad + mu
    return direction


def _make_candidate(a: np.ndarray, direction: np.ndarray, step: float) -> np.ndarray:
    """
    Apply a step in the given direction, enforce non‑negativity and
    re‑normalise to unit sum.
    """
    cand = a + step * direction
    # clip negatives (the evaluation routine also clips, but we keep it tidy)
    cand = np.where(cand > 0.0, cand, 0.0)

    total = cand.sum()
    if total <= 1e-12:                     # degenerate – re‑initialise
        cand = np.random.rand(a.shape[0])
        total = cand.sum()
    cand /= total
    return cand


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """Return a list of non‑negative heights that (hopefully) beats the target."""
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5   # safety margin

    # ----------------------------------------------------------------------
    # Choose a problem size.  Longer vectors give a lower theoretical bound,
    # but are slower.  A moderate size (≈ 2000–3000) works well within the budget.
    # ----------------------------------------------------------------------
    length_options = [800, 1200, 1800, 2500, 3000, 3500]
    n = int(np.random.choice(length_options))

    # Initial random vector (Dirichlet/Uniform)
    a = np.random.rand(n)
    a /= a.sum()                     # normalise to unit sum

    # Current score and best‑ever record
    curr_score = evaluate_sequence(a.tolist())
    best_seq = a.copy()
    best_score = curr_score

    # Adaptive step size and restart control
    step = 0.05
    min_step = 1e-6
    max_step = 0.5
    no_improve = 0
    restart_after = 5000            # iterations without improvement → restart

    iteration = 0
    while time.time() < deadline:
        iteration += 1

        # ------------------------------------------------------------------
        # Compute convolution and the index of the maximal entry
        # ------------------------------------------------------------------
        conv = np.convolve(a, a)
        k_max = int(np.argmax(conv))

        # ------------------------------------------------------------------
        # Gradient direction that tries to reduce the maximal convolution
        # ------------------------------------------------------------------
        direction = _gradient_direction(a, k_max)

        # If the direction is (almost) zero we are at a symmetric point;
        # inject a tiny random jitter to break symmetry.
        if np.linalg.norm(direction) < 1e-12:
            direction = np.random.randn(n) * 0.01
            direction -= direction.mean()          # keep zero sum

        # ------------------------------------------------------------------
        # Propose a new candidate by moving a bit along the direction
        # ------------------------------------------------------------------
        cand = _make_candidate(a, direction, step)
        cand_score = evaluate_sequence(cand.tolist())

        # ------------------------------------------------------------------
        # Acceptance logic (simple greedy with adaptive step size)
        # ------------------------------------------------------------------
        if cand_score < curr_score:
            # accept improvement
            a = cand
            curr_score = cand_score
            step = min(step * 1.08, max_step)
            no_improve = 0
        else:
            # reject – shrink the step
            step = max(step * 0.55, min_step)
            no_improve += 1

        # ------------------------------------------------------------------
        # Record the best sequence ever seen
        # ------------------------------------------------------------------
        if curr_score < best_score:
            best_score = curr_score
            best_seq = a.copy()
            print(
                f"[{time.time() - start_time:6.1f}s] New best: {best_score:.6f} "
                f"(len={n}, step={step:.4g})"
            )

        # ------------------------------------------------------------------
        # Random restart if we have been stuck for a long time
        # ------------------------------------------------------------------
        if no_improve >= restart_after:
            a = np.random.rand(n)
            a /= a.sum()
            curr_score = evaluate_sequence(a.tolist())
            step = 0.05
            no_improve = 0
            print(
                f"[{time.time() - start_time:6.1f}s] Restarted with fresh random vector."
            )

        # occasional cheap random perturbation to help escape shallow plateaus
        if iteration % 500 == 0 and np.random.rand() < 0.2:
            # move a small amount of mass from a random index to another
            i, j = np.random.choice(n, size=2, replace=False)
            delta = min(a[i], a[i] * 0.2)      # at most 20 % of the donor mass
            if delta > 0:
                a[i] -= delta
                a[j] += delta
                # renormalise (sum unchanged, but guard against round‑off)
                a /= a.sum()
                curr_score = evaluate_sequence(a.tolist())

    # ----------------------------------------------------------------------
    # Return the best sequence found (as a plain Python list)
    # ----------------------------------------------------------------------
    print(f"Search finished – best score: {best_score:.6f}")
    return best_seq.tolist()
