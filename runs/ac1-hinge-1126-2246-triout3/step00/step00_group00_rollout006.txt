"""
Propose a step‑function (list of non‑negative heights) that gives a very
small value of the provided `evaluate_sequence` function.

Idea
----
For a geometric progression  

    a_i = (c / n) * (1 - c/n) ** i            (i = 0 … n‑1)

the total sum is  

    S = 1 - (1 - c/n) ** n  ≃ 1 – e^{-c},

while the largest entry of the self‑convolution occurs at the centre
(k = n‑1) and is approximately  

    M ≃ n * (c/n)² * (1 - c/n)^{n‑1}  ≃ c² / n * e^{-c} .

Hence the evaluation metric

    E = 2·n·M / S² ≃ 2·c²·e^{-c} / (1‑e^{-c})²

can be made arbitrarily small by choosing a moderate constant `c`
(e.g. 5 … 20) and a sufficiently large length `n`.  The construction is
deterministic, cheap to compute and respects all security checks
(only non‑negative floats, each ≤ 1000).

The algorithm searches a few values of `n` and `c` (within the allowed
time budget) and returns the sequence with the lowest score found.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper – build a geometric sequence of length n with parameter c (>0)
# ----------------------------------------------------------------------
def geometric_sequence(n: int, c: float) -> list[float]:
    """
    Return a list of length `n` with entries
        a_i = (c/n) * (1 - c/n) ** i .
    The sequence is non‑negative and all entries are ≤ 1000.
    """
    # Ensure the ratio r stays in (0, 1)
    r = 1.0 - c / n
    if r <= 0.0:
        # If c >= n we cannot build a proper geometric decay,
        # fall back to a tiny constant value.
        r = 0.0
    a0 = c / n
    # Use float64 to keep the tiny powers accurate.
    exponents = np.arange(n, dtype=np.float64)
    seq = a0 * np.power(r, exponents)
    # Convert to plain Python floats (the evaluator will do it again)
    return seq.tolist()

def propose_candidate(seed: int = 42, budget_s: float = 1000, **kwargs):
    """
    Generate a candidate step‑function sequence.
    The search explores a few geometric families (different lengths and
    decay constants) within the allotted time and returns the best
    sequence found according to `evaluate_sequence`.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5   # leave a safety margin

    best_seq = None
    best_score = float("inf")

    # ------------------------------------------------------------------
    # 1. Systematic sweep over a few lengths and decay constants.
    # ------------------------------------------------------------------
    lengths = [800, 1500, 2500, 3536, 5000, 8000, 10000]
    # Decay constants (c) – a moderate range that yields good scores.
    c_values = np.linspace(1.0, 25.0, 25)   # 1 … 25 in 1‑step increments

    for n in lengths:
        if time.time() > deadline:
            break
        for c in c_values:
            if c >= n:                 # keep r = 1 - c/n positive
                continue
            seq = geometric_sequence(n, c)
            # Quick sanity check – the evaluator would reject an empty list
            # or a sum that is too tiny; we avoid those cases.
            if not seq:
                continue
            try:
                score = evaluate_sequence(seq)   # provided by the platform
            except Exception:   # any unexpected problem – skip this candidate
                continue
            if score < best_score:
                best_score = score
                best_seq = seq
                print(f"New best: n={n}, c={c:.2f}, score={score:.6f}")

    # ------------------------------------------------------------------
    # 2. Random fine‑tuning around the current best (if time permits).
    # ------------------------------------------------------------------
    # Use a simple mutation: add a tiny normally‑distributed noise to a

    while time.time() < deadline:
