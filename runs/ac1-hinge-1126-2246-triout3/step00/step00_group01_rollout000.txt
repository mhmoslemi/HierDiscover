import numpy as np
import time

def fast_evaluate(seq):
    """Fast version of evaluate_sequence: returns 2*n*max(conv)/sum^2."""
    seq = np.asarray(seq, dtype=float)
    n = seq.shape[0]
    sum_seq = seq.sum()
    if sum_seq < 0.01:
        return np.inf
    conv = np.convolve(seq, seq)
    max_b = conv.max()
    return float(2.0 * n * max_b / (sum_seq * sum_seq))

def random_binary_sequence(rng, n, density):
    """Return binary sequence of length n with given density (fraction of ones)."""
    m = max(1, int(round(density * n)))
    # Ensure m <= n
    m = min(m, n)
    seq = np.zeros(n, dtype=float)
    if m == n:
        seq[:] = 1.0
    else:
        idx = rng.choice(n, size=m, replace=False)
        seq[idx] = 1.0
    return seq

The evaluation function is
    score = 2 * n * max(conv(a, a)) / (sum(a)^2)

For a binary sequence (0/1) with m ones,
    score = 2 * n * max_pairs / m^2,
where max_pairs is the maximal number of ordered pairs (i, j) of
ones with i + j = k.  By choosing a random support of density ≈½,
the additive energy is spread almost uniformly, giving scores very close

---------

    the sum is too small.
    """
    seq = np.asarray(seq, dtype=float)
    n = seq.shape[0]
    total = seq.sum()
    if total < 0.01:
        return np.inf
    conv = np.convolve(seq, seq)
    max_b = conv.max()
    return float(2.0 * n * max_b / (total * total))
