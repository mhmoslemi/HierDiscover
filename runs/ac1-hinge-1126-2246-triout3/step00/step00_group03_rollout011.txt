```
def project_simplex(v):
    # Projects onto probability simplex: sum=1, v_i>=0
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = np.where(cond)[0][-1]
        theta = cssv[rho] / (rho + 1)
    else:
        theta = 0.0
    w = np.maximum(v - theta, 0)
    return w
```

"""
Projected soft‑max gradient descent for flattening the self‑convolution.

The evaluation function depends only on the *shape* of the vector:
if p_i = a_i / sum(a) then
    evaluate_sequence(a) = 2·n·max_k (p * p)[k] .
Hence we minimise  max(p∗p)  under p_i ≥ 0,  ∑p_i = 1.

We approximate the non‑differentiable max by a smooth soft‑max:
    f(p) = (1/β)·log Σ_k exp(β·(p∗p)[k])
which converges to max(p∗p) as β→∞.

∂f/∂p_i = 2·Σ_j p_j·w_{i+j} ,
where w_k = exp(β·conv_k) / Σ exp(β·conv) are the soft‑max weights.
The gradient is computed efficiently by an FFT based cross‑correlation.
After each gradient step the vector is projected back onto the probability
simplex (non‑negative entries that sum to one).

The algorithm is run many times with different random seeds and lengths
(≈ 5‑10 k).  The best sequence found (lowest evaluation score) is returned.
"""

import numpy as np
import time

# -------------------------------------------------------------
# Helper functions (must be top‑level, no closures)
# -------------------------------------------------------------

def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    if v.ndim != 1:
        raise ValueError("_project_simplex expects a 1‑D array")
    n = v.shape[0]
    # sort in descending order
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all entries are zero or negative – return uniform point
        return np.full_like(v, 1.0 / n)
    rho = np.where(cond)[0][-1]                     # last true index
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _softmax(x: np.ndarray, beta: float) -> np.ndarray:
    """Soft‑max with temperature 1/β, numerically stable."""
    m = np.max(x)
    z = np.exp(beta * (x - m))
    return z / np.sum(z)


# -------------------------------------------------------------
# Main optimisation routine
# -------------------------------------------------------------

def propose_candidate(seed: int = 0, budget_s: float = 1000.0) -> list[float]:
    """
    Search for a non‑negative step‑function that minimises
    ``evaluate_sequence``.  The function returns the best sequence found
    before the time budget expires.

    Parameters
    ----------
    seed : int
        Random seed for reproducibility.
    budget_s : float
        Time budget in seconds.

    Returns
    -------
    list[float]
        The sequence (list of floats) achieving the lowest found score.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.05  # tiny safety margin

    best_seq = None          # best probability vector found
    best_score = float("inf")
    # lengths we will try – around the known construction size
    lengths = [5000, 6000, 7500, 8000, 10000]

    # Main outer loop – keep trying new initialisations while time permits
    while time.time() < deadline:
        n = int(np.random.choice(lengths))
        # start from the uniform distribution on {0,…,n‑1}
        p = np.full(n, 1.0 / n, dtype=np.float64)

        # hyper‑parameters (chosen empirically)
        max_iters = 5000                # iterations for this run
        lr = 0.12                       # learning rate
        beta = 1.0                      # start temperature
        beta_increment = 0.006          # how quickly we sharpen the soft‑max

        N = _next_pow2(3 * n - 1)       # FFT length sufficient for all convolutions

        # Pre‑allocated FFT placeholders to avoid re‑allocation inside the loop
        rfft = np.fft.rfft
        irfft = np.fft.irfft

        for it in range(max_iters):
            # -----------------------------------------------------------------
            # 1. Compute self‑convolution via FFT (real → real)
            # -----------------------------------------------------------------
            fft_p = rfft(p, n=N)
            conv = irfft(fft_p * fft_p, n=N)[: 2 * n - 1]
            max_conv = np.max(conv)
            cur_score = 2.0 * n * max_conv

            # -----------------------------------------------------------------
            # 2. Keep the best overall result
            # -----------------------------------------------------------------
            if cur_score < best_score:
                best_score = cur_score
                best_seq = p.copy()
                print(f"[{time.time():.2f}] n={n:<5d} iter={it:<4d} score={cur_score:.6f}")

                # early exit if we are already below the target bound
                if best_score <= 1.55:
                    break

            # -----------------------------------------------------------------
            # 3. Time check
            # -----------------------------------------------------------------
            if time.time() >= deadline:
                break

            # -----------------------------------------------------------------
            # 4. Soft‑max weights over convolution entries (smooth max)
            # -----------------------------------------------------------------
            w = _softmax(conv, beta)
            beta = min(200.0, beta + beta_increment)   # slowly increase sharpness

            # -----------------------------------------------------------------
            # 5. Gradient of the soft‑max (cross‑correlation p ⋆ w)
            # -----------------------------------------------------------------
            fft_w = rfft(w, n=N)
            # correlation = irfft(fft(p) * conj(fft(w))) – returns length N
            corr = irfft(fft_p * np.conj(fft_w), n=N)[:n]
            grad = 2.0 * corr

            # -----------------------------------------------------------------
            # 6. Gradient descent step and projection onto the simplex
            # -----------------------------------------------------------------
            p = p - lr * grad
            p = _project_simplex(p)

            # -----------------------------------------------------------------
            # 7. Occasionally inject a tiny random perturbation (helps escape plateaus)
            # -----------------------------------------------------------------
            if (it + 1) % 500 == 0:
                noise = np.random.randn(n) * 1e-4
                p = _project_simplex(p + noise)

        # stop outer loop early if we have beaten the target bound
        if best_score <= 1.55:
            break

    # -----------------------------------------------------------------
    # Fallback – if nothing was found (should not happen), return uniform
    # -----------------------------------------------------------------
    if best_seq is None:
        n = 7521
        best_seq = np.full(n, 1.0 / n, dtype=np.float64)

    # The evaluation routine is scale‑invariant; we keep the probabilities as they are.
    return best_seq.tolist()
