"""
Efficient construction of a step function with a low evaluation score.

The idea is to place several *identical uniform blocks* far enough apart
so that their self‑convolutions never interfere.  A uniform block of
length L with entries 1/L has a convolution max‑value of 1/L.  If we put
K such blocks with a gap of length L between consecutive blocks, the
overall length becomes (2·K‑1)·L, the total sum is K and the maximum
convolution stays 1/L.  The evaluation function then yields

    2·N·max_b / sum^2 = 2·(2K‑1)L·(1/L) / K^2 = (4K‑2)/K^2

which is ≤ 1.5 for K≥2 and drops quickly as K grows.  With K=3 and
L=500 the score is 10/9 ≈ 1.111, well below the target 1.503.

The `propose_candidate` function builds this construction (K=3,
L=500) and returns the corresponding list of heights.
"""

import time
import numpy as np

def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Build a step function consisting of three identical uniform blocks,
    each of length `block_len`, separated by zero gaps of the same length.
    This construction achieves an evaluation score of 10/9 ≈ 1.111.

    Parameters
    ----------
    seed : int, optional
        Unused – kept for compatibility with the framework.
    budget_s : int, optional
        Unused – the construction is instantaneous.
    **kwargs : dict
        Ignored.

    Returns
    -------
    list[float]
        The sequence of heights.
    """
    # Parameters: choose a moderate block size.
    block_len = 500          # length of each uniform block
    num_blocks = 3           # K in the analysis (≥2 for a score ≤ 1.5)

    # Uniform value inside a block so that each block sums to 1.
    uniform_value = 1.0 / block_len

    sequence = []
    for i in range(num_blocks):
        # Append the uniform block.
        sequence.extend([uniform_value] * block_len)

        # Add a zero gap after the block, except after the last one.
        if i < num_blocks - 1:
            sequence.extend([0.0] * block_len)

    # Optional: report the expected evaluation (useful for debugging).
    try:

        eval_score = evaluate_sequence(sequence)  # type: ignore
        print(f"Constructed sequence length = {len(sequence)}")
        print(f"Expected evaluation score ≈ {eval_score:.6f}")
    except Exception:
        # If evaluation is not available during generation, we simply skip it.
        pass

    return sequence
