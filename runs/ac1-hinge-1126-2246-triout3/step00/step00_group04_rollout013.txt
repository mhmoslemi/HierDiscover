"""
Propose a candidate sequence that makes the self‑convolution as flat as possible.
The core idea is to start from a near‑optimal analytic shape
    a_i ≈ 1/√(i+1)                                 (i = 0 … n‑1)
which comes from the continuous identity
    ∫_0^x dt /√t √(x‑t) = π
and already yields an evaluation ≈ π/2 ≈ 1.57 (much better than the
uniform step function with value 2).  From this seed we perform a cheap
local “balancing” optimisation that repeatedly moves a little mass from the
larger of a pair (i, k‑i) that contributes to the current maximal convolution
value to the smaller one.  This strictly reduces the maximal entry of the
convolution while keeping the total sum unchanged.  Occasionally a small
random perturbation or the LP‑based direction from the original SOTA code
is tried to escape plateaus.

The whole routine runs for the allotted time budget (default 950 s, leaving a
few seconds margin for the caller) and returns the best sequence found.
All numbers are forced non‑negative, and the algorithm works for moderate
lengths (n ≈ 300–1500) so that each evaluation (a single np.convolve) is fast.
"""

import time
import numpy as np
from scipy import optimize

# ----------------------------------------------------------------------
# Helper: linear programme used by the original SOTA routine (kept for reuse)
def solve_convolution_lp(f_sequence, rhs):
    """Maximise sum(b) subject to conv(f,b) ≤ rhs and b ≥ 0.
    Returns the optimal b as a 1‑D numpy array or None on failure."""
    n = len(f_sequence)
    c = -np.ones(n)                # maximise sum → minimise negative sum
    a_ub = []
    b_ub = []
    for k in range(2 * n - 1):
        row = np.zeros(n)
        for i in range(n):
            j = k - i
            if 0 <= j < n:
                row[j] = f_sequence[i]
        a_ub.append(row)
        b_ub.append(rhs)
    # non‑negativity constraints  –b ≤ 0
    a_ub_nonneg = -np.eye(n)
    b_ub_nonneg = np.zeros(n)

    a_ub = np.vstack([a_ub, a_ub_nonneg])
    b_ub = np.hstack([b_ub, b_ub_nonneg])

    result = optimize.linprog(
        c,
        A_ub=a_ub,
        b_ub=b_ub,
        method="highs",
        options={"time_limit": 5.0, "disp": False},
    )
    if result.success:
        return result.x
    return None


def get_good_direction_to_move_into(sequence):
    """Return a new sequence obtained by the LP step used in the reference code."""
    n = len(sequence)
    sum_sequence = np.sum(sequence)
    # normalise to the “canonical” L1 norm √(2n) – the scaling does not affect the final score
    normalized = [x * np.sqrt(2 * n) / sum_sequence for x in sequence]
    rhs = np.max(np.convolve(normalized, normalized))
    g = solve_convolution_lp(normalized, rhs)
    if g is None:
        return None
    sum_g = np.sum(g)
    normalized_g = [x * np.sqrt(2 * n) / sum_g for x in g]
    # a relatively aggressive step size (larger than the original 0.01)
    t = 0.05
    new_seq = [(1 - t) * x + t * y for x, y in zip(sequence, normalized_g)]
    return new_seq

# Analytic initial shapes ------------------------------------------------
def power_law_seq(n, exponent=0.5, offset=0.0):
    """a_i = 1/(i+offset)^{exponent}  (offset avoids division by zero)."""
    idx = np.arange(n, dtype=float)
    return 1.0 / np.power(idx + offset + 1.0, exponent)


def exponential_seq(n, lam=0.001):
    """a_i = exp(-lam * i)"""
    idx = np.arange(n, dtype=float)
    return np.exp(-lam * idx)


def linear_decay_seq(n):
    """a_i linearly decreasing from 1 to a tiny positive value."""
    return np.linspace(1.0, 0.01, n)


def uniform_seq(n):
    """Plain uniform sequence (baseline)."""
    return np.full(n, 1.0)


# ----------------------------------------------------------------------
# Local “balancing” move -------------------------------------------------
def balance_move(seq, eps_factor=0.03):
    """
    Identify the shift k where the convolution is maximal.
    Pick a random pair (i, j=k-i) inside the overlap and move a small amount
    of mass from the larger entry toward the smaller one.
    The total sum stays unchanged, the move never creates negatives.
    """
    n = len(seq)
    conv = np.convolve(seq, seq)
    k = int(np.argmax(conv))

    # admissible indices i with 0 ≤ i < n and 0 ≤ j = k-i < n
    low = max(0, k - (n - 1))
    high = min(n - 1, k)
    if low > high:
        return seq  # should not happen

    i = np.random.randint(low, high + 1)
    j = k - i
    if i == j:
        return seq  # moving mass between identical positions changes nothing

    a_i, a_j = seq[i], seq[j]
    if a_i <= a_j:
        # swap so that a_i is the larger one
        i, j = j, i
        a_i, a_j = a_j, a_i

    diff = a_i - a_j
    if diff <= 0:
        return seq

    # amount moved – a small fraction of the difference (or absolute cap)
    delta = min(diff * eps_factor, diff * 0.5, 0.1 * a_i + 1e-8)
    if delta <= 0:
        return seq

    new_seq = seq.copy()
    new_seq[i] -= delta
    new_seq[j] += delta
    # clip to stay inside the evaluator’s limits (0 … 1000)
    new_seq[i] = max(0.0, new_seq[i])
    new_seq[j] = max(0.0, new_seq[j])
    return new_seq


def random_perturb(seq, scale=0.02):
    """Add small Gaussian noise to a random subset of the entries."""
    n = len(seq)
    new_seq = seq.copy()
    # pick ~10 % of indices
    mask = np.random.rand(n) < 0.10
    noise = np.random.randn(n) * scale
    new_seq[mask] += noise[mask]
    new_seq[new_seq < 0] = 0.0
    return new_seq

        except Exception:

        if new_score < best_score:

    return best_seq
