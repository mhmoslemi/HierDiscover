def next_pow2(x): return 1 << (x-1).bit_length()

def convolve_fft(a, b):
    n = len(a) + len(b) - 1
    nfft = 1 << ((n - 1).bit_length())
    A = np.fft.rfft(a, nfft)
    B = np.fft.rfft(b, nfft)
    C = A * B
    c = np.fft.irfft(C, nfft)[:n]
    return c

def correlation_valid(x, y):
    # x length N, y length M
    # return conv(x, y[::-1]) valid part length N-M+1
    conv_full = convolve_fft(x, y[::-1])
    start = len(y) - 1
    end = start + (len(x) - len(y) + 1)
    return conv_full[start:end]

def project_simplex(v, s=1.0):
    # v is numpy array
    n = v.shape[0]
    # sort descending
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    ind = np.arange(1, n+1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all zero? Return zeros
        return np.zeros_like(v)
    rho = ind[cond][-1]  # last true
    theta = cssv[rho-1] / rho
    w = np.maximum(v - theta, 0)
    return w

"""
Gradient‑descent on the variance of the self‑convolution.

The evaluation function is homogeneous:
    2·n·max(conv) / (sum(a))².
If we keep the vector a on the probability simplex (sum = 1, a_i ≥ 0)
the denominator equals 1 and the score reduces to 2·n·max(conv).
The algorithm repeatedly:

    1. computes b = a * a   (self‑convolution),
    2. computes the gradient of   Σ (b_k – avg)²   w.r.t. a,
    3. takes a small step against that gradient,
    4. projects back onto the simplex (keeps a_i ≥ 0 and Σ a_i = 1).

Several random restarts – some starting from the known SOTA
sequence `height_sequence_1`, others from Dirichlet‑type random vectors –
are performed until the time budget expires.  The best sequence found
is returned as a plain Python list.
"""

import time
import random
import numpy as np

def _next_pow2(x: int) -> int:
    """Return the smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _convolve_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Linear convolution via FFT (real‑valued, returns length len(a)+len(b)-1)."""
    n = len(a) + len(b) - 1
    nfft = _next_pow2(n)
    A = np.fft.rfft(a, n=nfft)
    B = np.fft.rfft(b, n=nfft)
    C = A * B
    c = np.fft.irfft(C, n=nfft)[:n]
    return c


def _correlation_valid(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """
    Valid cross‑correlation of x with y (length = len(x) - len(y) + 1).
    Equivalent to np.correlate(x, y, mode='valid') but uses FFT.
    """
    conv_full = _convolve_fft(x, y[::-1])
    start = len(y) - 1
    end = start + (len(x) - len(y) + 1)
    return conv_full[start:end]


def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """
    Euclidean projection of a vector onto the simplex { w | w_i ≥ 0, Σ w_i = s }.
    Implements the algorithm of Duchi et al. (2008).
    """
    if s < 0:
        raise ValueError("Simplex radius must be non‑negative")

    n = v.shape[0]
    if n == 0:
        return v

    u = np.sort(v)[::-1]                # sorted descending
    cssv = np.cumsum(u) - s
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0

    if not np.any(cond):
        # All entries are zero after projection
        return np.zeros_like(v)

    rho = ind[cond][-1]                 # last true index (1‑based)
    theta = cssv[rho - 1] / rho
    w = np.maximum(v - theta, 0)
    return w

    def _score(a: np.ndarray) -> float:
        n = a.shape[0]
        b = _convolve_fft(a, a)
        return 2.0 * n * np.max(b)  # sum(a) == 1

    # a few random lengths

    # Outer loop: different random restarts
    while time.time() < deadline:

            a0 /= np.sum(a0)

        step = 0.6                 # initial step size
