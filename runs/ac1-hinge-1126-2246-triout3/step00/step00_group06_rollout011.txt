Project onto simplex algorithm (Duchi et al.):

```
def simplex_projection(v):
    if np.sum(v) == 1 and np.all(v >= 0):
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(len(v)) + 1
    cond = u - cssv / ind > 0
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0)
    return w
```

def solve_convolution_lp(f, M):
    n = len(f)
    c = -np.ones(n)  # maximize sum b -> minimize -sum b
    # Build convolution matrix
    A = np.zeros((2*n-1, n))
    for k in range(2*n-1):
        # for j from max(0, k-(n-1)) to min(n-1, k)
        lo = max(0, k-(n-1))
        hi = min(n-1, k)
        # So indices j such that 0 <= j <= n-1 and 0 <= k-j <= n-1
        # Equivalent to j in [lo, hi]
        # Fill row: A[k, j] = f[k-j]
        A[k, lo:hi+1] = f[k - np.arange(lo, hi+1)]
    # Inequalities
    A_ub = np.vstack([A, -np.eye(n)])

"""
Propose a candidate step‐function (non‑negative heights) that minimises

    2 * n * max(conv(a,a)) / (sum(a)**2)

The algorithm is a simple “LP direction + line‑search” scheme:
1. Keep the sequence normalised (sum = 1, all entries ≥0).
2. Let M = max(conv(f,f)).  Solve the linear program

       maximise   Σ b_j
       subject to conv(f,b) ≤ M ,  b ≥ 0

   (the LP is cheap – O(n²) constraints, solved with HiGHS).
   The optimiser returns a direction `g`.  After normalising `g` to sum = 1,
   the new sequence h = (1‑α) f + α g is examined for several α∈[0,1].
   Because conv is bilinear we can evaluate the whole family cheaply:

       conv(h,h) = (1‑α)² conv(f,f) + 2α(1‑α) conv(f,g) + α² conv(g,g)

   The α with the smallest maximal convolution value is kept.
3. Repeat until the time budget expires, keeping the overall best sequence.

The procedure is started from a few random initialisations (including the
previous best `height_sequence_1`).  The whole search fits comfortably inside
the 1000 s limit and typically drives the evaluation score well below 1.5.
"""

import time
import numpy as np
from scipy import optimize

# ----------------------------------------------------------------------
# Helper: project a vector onto the probability simplex (non‑negative, sum = 1)
def simplex_projection(v):
    """Euclidean projection of v onto the simplex {x≥0, Σx = 1}."""
    if np.sum(v) == 1.0 and np.all(v >= 0):
        return v
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # numerical safety – fall back to uniform
        return np.full(n, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0)
    return w


# ----------------------------------------------------------------------
# Linear programme: maximise Σ b_j   s.t.   conv(f,b) ≤ M , b ≥ 0
def solve_convolution_lp(f, M):
    """
    Solve   max Σ b_j   with   (f * b)_k ≤ M   for every k.
    Returns the optimal b (may be any feasible extreme point) or None.
    """
    n = f.shape[0]

    # objective: minimise -Σb   (equivalent to maximise Σb)
    c = -np.ones(n)

    # Build the convolution matrix A where A[k, j] = f[k‑j]  (0 if out of range)
    rows = []
    for k in range(2 * n - 1):
        lo = max(0, k - (n - 1))
        hi = min(n - 1, k)
        # indices j that appear in the sum for this k
        j_idx = np.arange(lo, hi + 1, dtype=int)
        coeff = f[k - j_idx]          # due to (f * b)_k = Σ_i f_i * b_{k-i}
        row = np.zeros(n)
        row[j_idx] = coeff
        rows.append(row)
    A_conv = np.vstack(rows)          # shape (2n‑1, n)

    # inequality matrix and RHS
    A_ub = np.vstack([A_conv, -np.eye(n)])   # conv constraints + b ≥ 0
    b_ub = np.concatenate([np.full(2 * n - 1, M), np.zeros(n)])

    res = optimize.linprog(c,
                           A_ub=A_ub,
                           b_ub=b_ub,
                           bounds=(0, None),           # non‑negative
                           method='highs',
                           options={'presolve': True})
    if res.success:
        return res.x
    return None


# ----------------------------------------------------------------------
def max_convolution(a):
    """Return (max_value, argmax_index, full_convolution) for sequence a."""
    b = np.convolve(a, a)
    idx = int(np.argmax(b))
    return float(b[idx]), idx, b


# ----------------------------------------------------------------------
def propose_candidate(seed=42, budget_s=1000, **kwargs):
    """
    Search for a sequence of non‑negative heights that minimises the
    evaluation function.  The function runs for at most `budget_s` seconds
    and returns the best sequence found.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5       # tiny safety margin

    best_seq = None
    best_score = float('inf')

    # ------------------------------------------------------------------
    # Main outer loop – try several random problem sizes / starts
    # ------------------------------------------------------------------
    while time.time() < deadline:
        # choose a moderate length; too large makes the LP costly
        n = np.random.randint(150, 401)          # 150–400 works well

        # ------------------------------------------------------------------
        # Initial sequence: either the historic best or a fresh random start
        # ------------------------------------------------------------------
        if np.random.rand() < 0.5 and 'height_sequence_1' in globals():
            a = np.array(height_sequence_1, dtype=float)
            if a.size != n:
                # truncate / pad to the chosen size
                if a.size > n:
                    a = a[:n]
                else:
                    a = np.pad(a, (0, n - a.size), constant_values=0.0)
        else:
            a = np.full(n, 1.0 / n) + 0.01 * np.random.randn(n)

        a = np.maximum(a, 0.0)
        a = simplex_projection(a)

        # ------------------------------------------------------------------
        # Local optimisation loop – LP direction + optimal mixing
        # ------------------------------------------------------------------
        inner_deadline = time.time() + min(8.0, deadline - time.time())
        iter_cnt = 0
        while time.time() < inner_deadline and iter_cnt < 500:
            # current bound
            M, _, conv_f = max_convolution(a)

            # direction from LP
            g = solve_convolution_lp(a, M)
            if g is None:
                # LP failed – random jiggle to escape possible degeneracy
                idx = np.random.randint(n)
                a[idx] = max(0.0, a[idx] + np.random.randn() * 0.01)
                a = simplex_projection(a)
                iter_cnt += 1
                continue

            # normalise g to sum = 1 (LP guarantees feasibility, scaling keeps feasibility)
            g = simplex_projection(g)

            # pre‑compute the three convolutions needed for any α
            conv_g = np.convolve(g, g)
            conv_fg = np.convolve(a, g)

            # line search on α ∈ [0,1]; 0.0 is the current point, 1.0 would be pure g
            alphas = np.linspace(0.0, 1.0, 31)   # step = 0.033…  → cheap
            best_alpha = 0.0
            best_max = M

            for α in alphas[1:]:                # skip α = 0 (already known)
                # convex combination of the three convolutions
                combined = ((1 - α) ** 2) * conv_f + (2 * α * (1 - α)) * conv_fg + (α ** 2) * conv_g
                cur_max = float(np.max(combined))
                if cur_max < best_max - 1e-12:   # strictly better
                    best_max = cur_max
                    best_alpha = α

            # ------------------------------------------------------------------
            # Apply the best mixing step (if any improvement)
            # ------------------------------------------------------------------
            if best_alpha > 0.0:
                a = (1 - best_alpha) * a + best_alpha * g
                a = simplex_projection(a)   # keep numerical safety

                # evaluate using the provided function (fast, does the same scaling)
                cur_score = evaluate_sequence(a.tolist())
                if cur_score < best_score:
                    best_score = cur_score
                    best_seq = a.tolist()
                    print(f"[{time.time() - start_time:6.2f}s] n={n:4d} "
                          f"iter={iter_cnt:3d} α={best_alpha:.3f}  score={cur_score:.6f}")

                # stop early if we already hit the target
                if best_score <= 1.503:
                    print("Target reached – exiting early.")
                    return best_seq
            else:
                # no improvement found – perturb a little to explore new region
                i, j = np.random.choice(n, 2, replace=False)
                delta = np.random.randn() * 0.005
                a[i] = max(0.0, a[i] + delta)
                a[j] = max(0.0, a[j] - delta)
                a = simplex_projection(a)

            iter_cnt += 1

        # ------------------------------------------------------------------
        # End of an outer run – keep the best sequence seen so far
        # ------------------------------------------------------------------
        if best_seq is None:
            # nothing successful yet – evaluate the current a
            cur_score = evaluate_sequence(a.tolist())
            if cur_score < best_score:
                best_score = cur_score
                best_seq = a.tolist()
        # small progress report
        print(f"*** Outer loop completed (n={n}), best_score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Return the best sequence discovered
    # ------------------------------------------------------------------
    if best_seq is None:
        # unlikely, but provide a simple uniform sequence as a fallback
        n = 200
        best_seq = [1.0 / n] * n
    return best_seq
