"""
Step‑function optimizer.

The algorithm keeps the ℓ1‑norm of the sequence fixed to
    S = sqrt(2·n)      ( n = length of the sequence )
so that the evaluation score returned by ``evaluate_sequence`` is
exactly the maximum value of the self‑convolution:
    score = max_k (f * f)[k].

Starting from an initial non‑negative sequence we repeatedly:
1. Compute the current bound M = max(conv(f,f)).
2. Solve a linear program
       maximise   Σ g_i
       subject to conv(f,g) ≤ M  (entry‑wise) ,  g_i ≥ 0 .
   The LP finds a direction g whose ℓ1‑norm can be increased without
   raising the convolution bound.
3. Rescale g to have the same ℓ1‑norm S as f.
4. Mix f and g with a few candidate step sizes t ∈ {0.2,0.1,0.05,0.02,0.01}
   and keep the mixture that yields the largest decrease of the score.
5. If no improvement is found a small random perturbation is applied.

The process is repeated for several random restarts (including the
provided ``height_sequence_1`` if it exists) until the time budget is
almost exhausted.  The best sequence found is returned.
"""

import time
import random
import numpy as np
from scipy.optimize import linprog

# ----------------------------------------------------------------------
# Helper: solve the LP  maximise Σ g_i  s.t. conv(f,g) ≤ M,  g ≥ 0
# ----------------------------------------------------------------------
def solve_convolution_lp(f_seq: np.ndarray, M: float):
    """
    Solve the linear program:
        maximise   sum(g)
        subject to conv(f, g) <= M   (componentwise)
                   g >= 0
    Returns the optimal g as a NumPy array or None if the solver fails.
    """
    n = f_seq.shape[0]
    rows = 2 * n - 1

    # Build the constraint matrix A such that (A @ g)_k = (conv(f,g))[k]
    A = np.zeros((rows, n), dtype=np.float64)
    for k in range(rows):
        lo = max(0, k - (n - 1))
        hi = min(k, n - 1)
        j = np.arange(lo, hi + 1, dtype=np.int64)
        i = k - j
        A[k, j] = f_seq[i]

    b = np.full(rows, M, dtype=np.float64)
    c = -np.ones(n, dtype=np.float64)          # maximise sum(g)  -> minimise -sum(g)
    bounds = [(0, None)] * n                   # g_i >= 0

    # HiGHS is fast and reliable for these medium‑size LPs
    res = linprog(c, A_ub=A, b_ub=b, bounds=bounds, method='highs')
    if res.success:
        return res.x
    return None

            n = 300                     # smaller, faster LPs
        else:
            n = random.randint(200, 1000)

        # --- initialise a feasible sequence (uniform distribution) ------------
        S = np.sqrt(2.0 * n)            # target ℓ1‑norm
        f = np.full(n, S / n, dtype=np.float64)

        # tiny random jitter to break symmetry
        f += np.random.randn(n) * 0.001 * S / n
        f = np.clip(f, 0.0, None)
        f *= S / f.sum()                # re‑normalise exactly to S

        cur_score = eval_seq(f)
        if cur_score < best_score:
            best_score = cur_score
            best_seq = f.copy()
            print(f"[Restart {rs}] initial score = {cur_score:.6f} (len={n})")

        # --------------------------------------------------------------
        # Main optimisation loop for this restart
        # --------------------------------------------------------------
        iter_no = 0
        while time.time() < deadline and iter_no < 300:
            n = f.shape[0]
            # Current convolution bound
            conv_f = np.convolve(f, f)
            M = conv_f.max()

            # Solve the LP to obtain a promising direction g
            g_raw = solve_convolution_lp(f, M)
            if g_raw is None:
                # LP failure – fall back to a tiny random tweak
                idx = random.randrange(n)
                f[idx] = max(0.0, f[idx] + np.random.randn() * 0.005 * S / n)
                f *= S / f.sum()
                iter_no += 1
                continue

            # Rescale g so that it has the same ℓ1‑norm S as f
            sum_g_raw = g_raw.sum()
            if sum_g_raw <= 0:
                # Degenerate solution – perturb and continue
                idx = random.randrange(n)
                f[idx] = max(0.0, f[idx] + np.random.randn() * 0.005 * S / n)
                f *= S / f.sum()
                iter_no += 1
                continue
            g = g_raw * (S / sum_g_raw)

            # Try a handful of mixing parameters and keep the best improvement
            candidate_f = None
            candidate_score = cur_score
            for t in (0.2, 0.1, 0.05, 0.02, 0.01):
                new_f = (1.0 - t) * f + t * g
                # No need to renormalise: both operands already sum to S
                new_score = eval_seq(new_f)
                if new_score < candidate_score:
                    candidate_score = new_score
                    candidate_f = new_f

            if candidate_f is not None:
                f = candidate_f
                cur_score = candidate_score
                if cur_score < best_score:
                    best_score = cur_score
                    best_seq = f.copy()
                    print(f"[Iter {iter_no:04d}] New best = {best_score:.6f} (len={n})")
            else:
                # No improvement – inject a small random perturbation
                idx = random.randrange(n)
                f[idx] = max(0.0, f[idx] + np.random.randn() * 0.005 * S / n)
                f *= S / f.sum()

            iter_no += 1

        print(f"[Restart {rs}] finished with best score {cur_score:.6f}")

    # --------------------------------------------------------------
    # Fallback in the extremely unlikely case no sequence was found
    # --------------------------------------------------------------
    if best_seq is None:
        n = 400
        S = np.sqrt(2.0 * n)
        best_seq = np.full(n, S / n, dtype=np.float64)
        best_score = eval_seq(best_seq)

    print(f"*** Final best score = {best_score:.6f} (length = {len(best_seq)})")
    return best_seq.tolist()
