def propose_candidate(seed=123, budget_s=1000, **kwargs):
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5
    
    n = len(height_sequence_1) if 'height_sequence_1' in globals() else 5168
    target_sum = np.sqrt(2 * n)
    
    alphas = np.linspace(0.5, 0.65, 31)
    deltas = [0.0, 0.5, 1.0, 2.0]
    best_seq = None
    best_score = np.inf
    
    for alpha in alphas:
        for delta in deltas:
            idx = np.arange(1, n+1, dtype=np.float64) + delta
            seq = idx ** (-alpha)
            seq = seq * (target_sum / seq.sum())
            # Clip to [0,1000] automatically in evaluate_sequence, but we may also clip early.
            score = evaluate_sequence(seq.tolist())
            if score < best_score:
                best_score = score
                best_seq = seq.tolist()
                print(f"New best: alpha={alpha:.4f}, delta={delta:.2f}, score={score:.6f}")
            if time.time() >= deadline:
                return best_seq
    # Optional local refinement: gradient descent
    seq = np.array(best_seq)
    while time.time() < deadline:
        # pick random index and perturb
        i = np.random.randint(len(seq))
        delta = np.random.randn() * 0.01 * seq[i]  # relative perturb
        new_seq = seq.copy()
        new_seq[i] = max(0.0, new_seq[i] + delta)
        # renormalize to keep sum correct
        new_seq = new_seq * (target_sum / new_seq.sum())
        score = evaluate_sequence(new_seq.tolist())
        if score < best_score:
            best_score = score
            seq = new_seq
            best_seq = new_seq.tolist()
            print(f"Improved by perturbation: score={score:.6f}")
    return best_seq

    """
    Search for a low‑evaluation step‑function by using a parametrised
    power‑law family a_i = (i+δ)^{-α}.  The sequence is normalised so that

    function) and the best sequence seen during the time budget is

    finetune the result.
    """

    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5   # keep a tiny margin

    # length of the construction – use the length of the known
    # height_sequence_1 if it exists, otherwise fall back to 5168

    except NameError:

    alphas = np.linspace(0.5, 0.65, 31)   # 0.500, 0.505, …, 0.650

    best_score = np.inf
    best_seq = None

    # 1.  Coarse grid search over the power‑law family.
    # ------------------------------------------------------------------

        for delta in deltas:

            idx = np.arange(1, n + 1, dtype=np.float64) + delta

            # execution environment.

import numpy as np

def propose_candidate(seed: int = 42, budget_s: float = 1000, **kwargs):
    """
    Search for a low‑evaluation step‑function.

    The method builds candidates from the one‑parameter family

        a_i = (i + δ)^{-α}    ( i = 1 … n )

    which is known analytically to give a flat self‑convolution for
    α≈½.  By scanning α in a small interval around 0.5 and a few
    offsets δ we obtain a very good initial sequence.  Afterwards a tiny
    random walk with global renormalisation refines the best candidate.

    The sequence is finally normalised so that Σ a_i = √(2·n), the
    normalisation used by ``evaluate_sequence``; consequently the
    returned score equals the maximal convolution value.
    """

    # basic set‑up
    # ------------------------------------------------------------------

    # ------------------------------------------------------------------
