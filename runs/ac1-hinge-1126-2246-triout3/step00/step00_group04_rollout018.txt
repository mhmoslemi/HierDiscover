def propose_candidate(seed=42, budget_s=1000, **kwargs):
    import time, numpy as np
    np.random.seed(seed)
    deadline = time.time() + budget_s - 5 # leaving some margin
    best_seq = None
    best_score = float('inf')
    # Choose multiple lengths to test
    lengths = [1000, 2000, 4000, 5000, 6000] maybe limited to ~6000
    for n in lengths:
        # Consider grid of c values
        for c in np.linspace(0.5, 6.0, 56):  # step 0.1
            if time.time() > deadline:
                break
            r = 1 - c / n
            if r <= 0:
                continue
            # Build sequence as r**np.arange(n)
            seq = (r ** np.arange(n)).tolist()
            score = evaluate_sequence(seq)
            if score < best_score:
                best_score = score
                best_seq = seq.copy()
                print(f'New best: n={n}, c={c:.3f}, r={r:.6f}, score={score:.6f}')

def project_simplex(v):
    # Ensure non-negative and sum to 1
    v = np.maximum(v, 0)
    s = np.sum(v)
    if s == 0:
        # reset uniform
        v = np.ones_like(v) / len(v)
    else:
        v = v / s
    return v

"""Optimization of a non‑negative step function.

The goal is to minimise
    2 * n * max(conv(a, a)) / (sum(a)**2)

where `conv` is the full discrete convolution.
The routine implements a simple stochastic hill‑climbing where mass is
re‑distributed from the pair of indices that contribute most to the current
maximum convolution entry.  The algorithm is deliberately lightweight so it
always returns a valid sequence before the time budget expires.

The search is performed for several problem sizes (n = 200 … 1200).  For each
size a random Dirichlet start point is generated, then a few thousand
local‑move iterations are tried.  The best sequence encountered over all
trials is returned.
"""

import time
import numpy as np

def _project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Project a non‑negative vector onto the probability simplex (sum = 1)."""
    v = np.maximum(v, 0.0)
    s = v.sum()
    if s == 0.0:
        # fallback to uniform distribution
        return np.ones_like(v) / v.shape[0]
    return v / s


def _improve(a: np.ndarray, max_iters: int, deadline: float) -> (np.ndarray, float):
    """
    Perform a short stochastic improvement run on vector `a`.
    Returns the best vector found and its score.
    """
    n = a.shape[0]
    best_vec = a.copy()
    best_score = evaluate_sequence(best_vec.tolist())
    step = 0.05                     # size of each mass transfer
    for _ in range(max_iters):
        if time.time() > deadline:
            break

        # full convolution and location of the current worst (largest) entry
        conv = np.convolve(a, a)
        k = int(np.argmax(conv))

        # all index pairs (i, j) that contribute to this shift
        i_arr = np.arange(n)
        j_arr = k - i_arr
        mask = (j_arr >= 0) & (j_arr < n)
        i_idx = i_arr[mask]
        j_idx = j_arr[mask]

        # product a_i * a_j for each pair – we want the largest one
        prod = a[i_idx] * a[j_idx]
        if prod.size == 0:
            break
        max_pair = int(np.argmax(prod))
        i_max = i_idx[max_pair]
        j_max = j_idx[max_pair]

        # choose a random destination index different from i_max/j_max
        l = np.random.randint(n)
        while l == i_max or l == j_max:
            l = np.random.randint(n)

        # amount of probability mass to move
        delta = min(a[i_max], a[j_max]) * step
        if delta <= 0:
            # nothing to move – give a small random nudge and continue
            a = _project_to_simplex(a + np.random.randn(n) * 1e-4)
            continue

        # remove mass from the two indices that form the worst product
        a[i_max] -= delta
        a[j_max] -= delta
        # give all that mass to the random destination
        a[l] += 2.0 * delta

        # keep the vector feasible
        a = _project_to_simplex(a)

        # evaluate the new candidate
        score = evaluate_sequence(a.tolist())
        if score < best_score:
            best_score = score
            best_vec = a.copy()
            print(f"   improved score = {best_score:.6f} (n={n}, k={k})")
        # occasional random perturbation to escape plateaus
        if np.random.rand() < 0.001:
            a = _project_to_simplex(a + np.random.randn(n) * 1e-3)

    return best_vec, best_score


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a step‑function sequence with a low evaluation score.

    Parameters
    ----------
    seed : int
        Random seed for reproducibility.
    budget_s : int
        Approximate time budget (seconds). The function will stop a little
        before this limit and always return a valid sequence.

    Returns
    -------
    list[float]
        The best sequence discovered within the time budget.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5            # a small safety margin

    best_overall_vec = None
    best_overall_score = float("inf")

    # try a few different lengths – larger n gives more flexibility
    lengths = [200, 400, 800, 1200]
    for n in lengths:
        if time.time() > deadline:
            break
        # start from a random point on the simplex
        a = np.random.dirichlet(np.ones(n))
        # number of local moves – scale inversely with n to keep overall time low
        max_iters = max(2000, 200000 // n)
        vec, score = _improve(a, max_iters, deadline)

        if score < best_overall_score:
            best_overall_score = score
            best_overall_vec = vec.copy()
            print(f"New best overall: score={best_overall_score:.6f}, n={n}")

    # fallback: uniform sequence (guaranteed to be feasible)
    if best_overall_vec is None:
        n = 500
        best_overall_vec = np.full(n, 1.0 / n)

    # final sanity check – ensure the sequence complies with the evaluator
    final_score = evaluate_sequence(best_overall_vec.tolist())
    if final_score > best_overall_score:
        # extremely unlikely, but keep the best we have
        best_overall_vec = best_overall_vec / best_overall_vec.sum()
    return best_overall_vec.tolist()
