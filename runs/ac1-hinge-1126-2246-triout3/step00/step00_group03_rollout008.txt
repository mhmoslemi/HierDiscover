    conv = np.convolve(w,w)
    best_M = conv.max()

    iteration = 0

    while time.time() < deadline and iteration < 200:
        iteration += 1
        # Build LP constraints matrix
        # Using dense arrays maybe fine for n up to 2000
        rows = []
        cols = []
        data = []
        # A_ub rows for each k
        for k in range(2*n - 1):
            # indices j where 0 <= k - j < n => j in [max(0, k-(n-1)), min(n-1, k)]
            j_min = max(0, k - (n - 1))
            j_max = min(n - 1, k)
            # For each j in this range, coefficient is w[k - j]
            for j in range(j_min, j_max + 1):
                rows.append(k)
                cols.append(j)
                data.append(w[k - j])
            # coefficient for M variable (index n)
            rows.append(k)
            cols.append(n)
            data.append(-1.0)
        # Convert to CSR
        A_ub = csr_matrix((data, (rows, cols)), shape=(2*n - 1, n + 1), dtype=np.float64)
        b_ub = np.zeros(2*n - 1)

        # Equality constraint: sum bi = 1
        A_eq = np.zeros((1, n+1), dtype=np.float64)
        A_eq[0, :n] = 1.0
        b_eq = np.array([1.0])

        # objective: minimize M
        c = np.zeros(n+1)
        c[-1] = 1.0

        # bounds: bi >= 0, M >= 0
        bounds = [(0, None)] * n + [(0, None)]

        # Solve LP
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq, bounds=bounds, method='highs')
        if not res.success:
            print(f'LP failed at iteration {iteration}, message: {res.message}')
            break

        b_new = res.x[:n]
        M_new = res.x[-1]

        # Compute actual max conv of b_new with itself (maybe < M_new)
        actual_conv = np.convolve(b_new, b_new)
        actual_M = actual_conv.max()

        if actual_M < best_M - 1e-12:

Propose a step‑function sequence with an improved evaluation score.

The algorithm works as follows:

1.  Choose a moderately sized length (the code tries a few different
    lengths, each ≤ 2000).  Large lengths give a finer search space but
    increase the cost of each linear program; the chosen sizes give a
    good trade‑off for the 1000 s time budget.

2.  Initialise a probability distribution `w` (non‑negative, sum = 1).
    Several random starts are tried.

3.  Repeatedly solve a linear programme that, for the current distribution
    `w`, finds a new distribution `b` with the *minimum possible* value of
    `M = max_k (w * b)[k]` under the constraints `b ≥ 0` and `sum(b)=1`.

    The LP is:
        minimise   M
        s.t.      Σ_i w_i b_{k‑i} – M ≤ 0      for all k = 0 … 2n‑2
                 Σ_i b_i = 1
                 b_i ≥ 0 ,   M ≥ 0

    The constraint matrix is a Toeplitz matrix generated from the current
    `w`; it is built with NumPy (dense) which is fast for n ≈ 2000.
    The LP is solved with SciPy’s HiGHS interior‑point method.

4.  If the new distribution reduces the true maximum convolution
    `max_k (b * b)[k]` (the quantity actually used by `evaluate_sequence`),
    it becomes the current best and the iteration continues with `w ← b`.
    The process stops when no further improvement is observed or the time
    budget expires.

5.  After the optimisation the best distribution found is scaled (by a
    constant factor – the objective of `evaluate_sequence` is scale‑invariant)
    to avoid extremely tiny numbers and returned as a plain Python list.

The whole routine is self‑contained, uses only NumPy and SciPy and respects
the 1000 s overall budget.
"""

import time
import numpy as np
from scipy.optimize import linprog
from scipy.linalg import toeplitz


def _max_convolution(v: np.ndarray) -> float:
    """Return max(conv(v, v)).  v is assumed to sum to 1."""
    c = np.convolve(v, v)
    return float(c.max())


def _build_lp_matrices(w: np.ndarray):
    """
    Build A_ub, b_ub, A_eq, b_eq for the LP that minimises the
    supremum of conv(w, b) subject to b ≥ 0 and sum(b)=1.
    """
    n = w.size
    # Toeplitz matrix for the convolution constraints:
    # row k, column j contains w[k-j] when 0 ≤ k-j < n, otherwise 0.
    first_col = np.concatenate([w, np.zeros(n - 1, dtype=w.dtype)])
    first_row = np.zeros(n, dtype=w.dtype)      # forces zero for i<j
    A = toeplitz(first_col, first_row)          # shape (2n-1, n)

    # Append the column for the variable M (coefficient -1 in each row)
    A_ub = np.hstack([A, -np.ones((2 * n - 1, 1), dtype=w.dtype)])
    b_ub = np.zeros(2 * n - 1, dtype=w.dtype)

    # Equality: sum(b_i) = 1, M is free (coeff 0)
    A_eq = np.hstack([np.ones((1, n), dtype=w.dtype), np.zeros((1, 1), dtype=w.dtype)])
    b_eq = np.array([1.0], dtype=w.dtype)

    return A_ub, b_ub, A_eq, b_eq


def _solve_min_max_conv(w: np.ndarray):
    """
    Solve the LP described in the docstring for a given w.
    Returns (b, M) where b is the new distribution and M the optimal
    bound on max(conv(w, b)).
    """
    n = w.size
    c = np.zeros(n + 1, dtype=np.float64)
    c[-1] = 1.0                                 # minimise M

    A_ub, b_ub, A_eq, b_eq = _build_lp_matrices(w)

    # bounds: each b_i ≥ 0, M ≥ 0
    bounds = [(0.0, None)] * n + [(0.0, None)]

    # HiGHS is the fastest interior‑point solver in SciPy
    res = linprog(
        c,
        A_ub=A_ub,
        b_ub=b_ub,
        A_eq=A_eq,
        b_eq=b_eq,
        bounds=bounds,
        method="highs",
        options={"presolve": True, "time_limit": 8.0},
    )
    if not res.success:
        raise RuntimeError(f"LP failed: {res.message}")

    b_opt = res.x[:n]
    M_opt = res.x[-1]
    return b_opt, float(M_opt)


def _optimize_length(n, deadline, rng):
    """
    Perform the alternating LP optimisation for a fixed length n.
    Returns the best vector found (as a NumPy array) together with its
    max‑convolution value.
    """
    # start from a random probability vector
    w = rng.random(n)
    w /= w.sum()

    best_w = w.copy()
    best_M = _max_convolution(w)

    iteration = 0
    while time.time() < deadline:
        iteration += 1
        try:
            b, M_lp = _solve_min_max_conv(w)
        except Exception as e:
            print(f"[n={n}] LP error at iter {iteration}: {e}")
            break

        # true max‑convolution of the candidate (scale‑invariant measure)
        cur_M = _max_convolution(b)

        if cur_M + 1e-12 < best_M:          # improvement found
            best_M = cur_M
            best_w = b.copy()
            print(
                f"[n={n}] iter {iteration:3d}: M improved to {best_M:.6e} "
                f"(LP bound {M_lp:.6e})"
            )
            # continue optimisation from the improved point
            w = b
        else:
            # no improvement – stop this length
            print(f"[n={n}] iter {iteration}: no improvement (M={cur_M:.6e})")
            break

    return best_w, best_M


def propose_candidate(seed: int = 42, budget_s: float = 1000.0, **kwargs):
    """
    Search for a non‑negative step‑function sequence that minimises the
    provided `evaluate_sequence` function.

    The return value is a list of floats (the heights).  The algorithm
    uses a series of linear programmes to iteratively decrease the
    maximal autocorrelation of the sequence, which directly lowers the
    evaluation metric.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 2.0  # keep a safety margin

    # Try a few different lengths; keep the best result.
    # Length of the previous best construction is ~7521, but a shorter
    # length makes each LP cheap while still allowing a good bound.
    candidate_lengths = [500, 1000, 1500, 2000]

    # If the global height_sequence_1 is available we can also use its
    # length as an extra candidate.
    try:
        candidate_lengths.append(len(height_sequence_1))
    except Exception:
        pass

    best_seq = None
    best_score = float("inf")
    rng = np.random.default_rng(seed)

    for n in candidate_lengths:
        if time.time() >= deadline:
            break
        # Allocate a fraction of the remaining time to this length
        remaining = deadline - time.time()
        # give each length at most 15 % of the remaining budget (but at least 5 s)
        sub_budget = max(5.0, remaining * 0.15)

        sub_deadline = time.time() + sub_budget
        print(f"\n=== Optimising length n={n} (budget {sub_budget:.1f}s) ===")
        try:
            w_opt, max_conv = _optimize_length(n, sub_deadline, rng)
        except Exception as e:
            print(f"Failed optimisation for n={n}: {e}")
            continue

        # The evaluation score (scale‑invariant) for a probability vector:
        score = 2.0 * n * max_conv
        print(f"Result for n={n}: score≈{score:.6f}  (max conv {max_conv:.6e})")

        if score < best_score:
            best_score = score
            # Scale the vector to avoid extremely tiny numbers (scale cancels out)
            scale = 100.0
            best_seq = (w_opt * scale).tolist()
            print(f"*** New overall best score {best_score:.6f} ***")

    # Fallback: if nothing succeeded, return a simple uniform sequence.
    if best_seq is None:
        n = 500
        best_seq = [1.0] * n
        print("Fallback to uniform sequence.")

    # Ensure the sum is comfortably above the 0.01 guard.
    s = sum(best_seq)
    if s < 0.02:
        factor = 0.02 / s
        best_seq = [x * factor for x in best_seq]

    return best_seq
