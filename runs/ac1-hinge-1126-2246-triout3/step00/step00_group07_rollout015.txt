def subgradient_step(a, lr):
    n = len(a)
    b = np.convolve(a, a)
    max_b = b.max()
    eps = 1e-12  # compute active set.
    active = np.where(b >= max_b - 1e-12)[0]
    # Compute gradient.
    grad = np.zeros(n)
    for k in active:
        # for i in range(n): partner index j = k - i
        # if 0 <= j < n: grad[i] += 2*a[j]
        # Equivalent: grad[i] = sum_{k in active} 2*a[k-i] where k-i in range.
        # We can vectorize: shift a by k for each k.
        # But simple loop O(|active|*n) which is small because active likely few (maybe 1 or a few).
        # We'll loop.
        for i in range(n):
            j = k - i
            if 0 <= j < n:
                grad[i] += 2 * a[j]
    # Possibly divide by number of active to average.
    grad /= len(active)
    a_new = a - lr * grad
    # Projection onto nonnegative then re-normalize.
    a_new = np.maximum(a_new, 0.0)
    s = a_new.sum()
    if s == 0:
        # reinitialize random vector
        a_new = np.random.random(n)
        a_new = a_new / a_new.sum()
    else:
        a_new = a_new / s
    return a_new

"""
Hill‑climbing search for a step‑function that lowers the evaluation metric.

The algorithm works with a fixed length `n`.  It keeps the sequence `a`
non‑negative and normalised (sum = 1).  The convolution `b = a * a`
(and consequently its maximal entry) is updated incrementally when a
small amount of mass is transferred from one position to another.
Only moves that lower the objective
    score = 2 * n * max(b) / (sum(a) ** 2)      (sum(a) = 1)
are accepted.  The step size shrinks as the time budget is consumed.
Occasional random restarts avoid getting stuck in poor local minima.

The returned sequence is a plain Python list of floats, ready for
`evaluate_sequence`.
"""

import time
import numpy as np


def propose_candidate(seed=42, budget_s=1000, **kwargs):
    """
    Search for a sequence of non‑negative heights that minimises
    `evaluate_sequence`.  The function runs for at most ``budget_s`` seconds
    and returns the best sequence found.

    Parameters
    ----------
    seed : int
        Random seed for reproducibility.
    budget_s : float
        Time budget in seconds.

    Returns
    -------
    list[float]
        The candidate height sequence.
    """
    np.random.seed(seed)
    # safety margin for the outer framework
    deadline = time.time() + budget_s - 5.0

    # ------------------------------------------------------------
    # Helper: compute the score from *already* known convolution `b`.
    # The sum of `a` is kept equal to 1, therefore the denominator is 1.
    # ------------------------------------------------------------
    def score_from_conv(conv, n):
        return float(2.0 * n * conv.max())

    # ------------------------------------------------------------
    # Core hill‑climbing loop (single length, several restarts)
    # ------------------------------------------------------------
    best_a = None
    best_score = float("inf")

    # length of the sequence – a compromise between quality and speed
    n = 2500

    # how many restarts are allowed inside the total budget
    # (the exact number is adaptive – we simply break when time runs out)
    while time.time() < deadline:
        # ----- initialise a random Dirichlet vector (more variance than uniform)
        alpha = 0.5                         # concentration parameter
        a = np.random.gamma(alpha, 1.0, size=n)
        a /= a.sum()                        # normalise to sum = 1

        # initial full convolution and score
        b = np.convolve(a, a)                # length = 2*n-1
        cur_score = score_from_conv(b, n)

        # store the best seen for this restart
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[init] score={cur_score:.6f}")

        # --- hill‑climbing iterations for this restart -----------------
        # counters for adaptive behaviour
        no_improve = 0
        total_iters = 0
        while time.time() < deadline:
            total_iters += 1

            # linear schedule for the maximal transferred fraction
            # start with large steps, shrink to ~0.01 near the end
            elapsed = time.time() - (deadline - budget_s + 5.0)
            progress = min(1.0, elapsed / (budget_s - 5.0))
            max_frac = 0.5 * (1.0 - progress) + 0.01   # between 0.51 → 0.01

            # pick source (large) and destination (small) indices
            i_src = int(np.argmax(a))          # index with largest mass
            i_dst = int(np.argmin(a))          # index with smallest mass
            if i_src == i_dst:
                # Should not happen, but guard against it
                i_dst = (i_dst + 1) % n

            # amount to move (a fraction of the source value)
            delta = a[i_src] * (np.random.rand() * max_frac)
            if delta <= 0.0:
                continue

            # --------------------------------------------------------
            # Incrementally update the convolution:
            #   b' = b - 2*delta*shift_i + 2*delta*shift_j
            #   plus the quadratic corrections at indices i+j , 2i , 2j
            # --------------------------------------------------------
            # keep a reference to the old `a` (used for the slice updates)
            a_old = a  # no copy – we will modify a only after acceptance

            # slice updates (vectorised, O(n))
            b[i_src : i_src + n] -= 2.0 * delta * a_old
            b[i_dst : i_dst + n] += 2.0 * delta * a_old

            # quadratic (order‑delta²) corrections
            idx_sum = i_src + i_dst
            b[idx_sum] -= 2.0 * delta * delta
            b[2 * i_src] += delta * delta
            b[2 * i_dst] += delta * delta

            # --------------------------------------------------------
            # Evaluate the new objective
            # --------------------------------------------------------
            new_score = score_from_conv(b, n)

            if new_score < cur_score - 1e-12:          # accept only improvements
                # commit the move
                a[i_src] -= delta
                a[i_dst] += delta
                cur_score = new_score
                no_improve = 0

                # record globally best solution
                if new_score < best_score:
                    best_score = new_score
                    best_a = a.copy()
                    print(f"[improve] iter={total_iters} score={new_score:.6f}")

            else:
                # reject – revert the convolution changes (a unchanged)
                b[i_src : i_src + n] += 2.0 * delta * a_old
                b[i_dst : i_dst + n] -= 2.0 * delta * a_old
                b[idx_sum] += 2.0 * delta * delta
                b[2 * i_src] -= delta * delta
                b[2 * i_dst] -= delta * delta
                no_improve += 1

            # if we have gone a long time without improvement,
            # perform a cheap random perturbation to try to escape
            if no_improve > 5000:
                # a small random shuffle of a subset of entries
                perm = np.random.permutation(n)
                a = a[perm]
                b = np.convolve(a, a)
                cur_score = score_from_conv(b, n)
                no_improve = 0
                print(f"[shuffle] iter={total_iters} new_score={cur_score:.6f}")

            # safety break if the inner loop runs too long
            if total_iters >= 200_000:
                break

        # end of one restart – continue while there is time left

    # ------------------------------------------------------------
    # Final sanity check using the provided `evaluate_sequence`
    # ------------------------------------------------------------
    if best_a is None:
        # fallback – a simple uniform sequence (should never happen)
        best_a = np.full(n, 1.0 / n)

    # Clip to admissible range (evaluate_sequence caps at 0…1000 anyway)
    best_a = np.clip(best_a, 0.0, 1000.0)
    final_score = evaluate_sequence(best_a.tolist())
    print(f"[final] length={n} score={final_score:.6f}")

    return best_a.tolist()
