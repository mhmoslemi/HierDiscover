"""
Simple stochastic optimizer for the step‑function evaluation problem.

The objective (scale‑invariant) is to minimise
    2 * n * max(conv(a,a)) / (sum(a)**2)
where ``conv`` is the full linear convolution.
For a fixed length ``n`` we can keep the sequence normalised to
sum = 1, so the score is simply ``2*n*max(conv)``.

The algorithm works in three stages:

1. **Parametric sweeps** – try a few analytically motivated families
   (power laws, exponentials, two‑region piecewise constants) and keep
   the best sequence found.

2. **Local refinement** – starting from the best candidate, repeatedly
   identify the convolution index with the current maximum, move a small
   amount of mass from the index that contributes most to that peak to the
   smallest entry, re‑normalise and accept the change only if the score
   improves.  This is a simple projected sub‑gradient descent on the
   ``max``‑convolution functional.

3. **Return** the best sequence discovered before the time budget expires.

All heavy work (convolution and max‑value) is performed with an FFT
(`np.fft.rfft`) which makes each iteration very cheap even for several
thousand entries.  The code respects the 1000 s wall‑clock limit,
prints occasional progress messages and finally returns a plain Python
list of non‑negative heights.
"""

import time
import random
import math
import numpy as np

def _score(seq: np.ndarray) -> float:
    """Return the evaluation score for a normalised sequence (sum == 1)."""
    n = seq.shape[0]
    # length of the FFT (next power of two for speed)
    m = 1 << ((2 * n - 1).bit_length())
    ft = np.fft.rfft(seq, m)
    conv = np.fft.irfft(ft * ft, m)[: 2 * n - 1]
    max_conv = np.max(conv)
    # sum(seq) == 1 → the denominator of evaluate_sequence is 1
    return float(2.0 * n * max_conv)


def propose_candidate(budget_s: float = 1000, seed: int = 42):
    """
    Search for a low‑score height sequence.

    Parameters
    ----------
    budget_s : float
        Wall‑clock time budget (seconds).  The function stops a few seconds
        before the limit to avoid a forced termination.
    seed : int
        Random seed for reproducibility.

    Returns
    -------
    list[float]
        The best sequence found, ready to be passed to ``evaluate_sequence``.
    """
    np.random.seed(seed)
    random.seed(seed)

    # ------------------------------------------------------------------
    # 1️⃣  Problem size and baseline
    # ------------------------------------------------------------------
    n = 3518                         # length used in the reference run
    a = np.full(n, 1.0 / n, dtype=np.float64)   # uniform start (sum == 1)
    best_seq = a.copy()
    best_score = _score(best_seq)
    print(f"[init] uniform score = {best_score:.6f}")

    deadline = time.time() + budget_s - 5.0   # safety margin

    # ------------------------------------------------------------------
    # 2️⃣  Parametric family sweeps
    # ------------------------------------------------------------------

    # ----- power law: a_i ∝ (i+1)^α  (α ∈ [-3,0]) --------------------
    for alpha in np.linspace(-3.0, 0.0, 31):
        if time.time() >= deadline:
            break
        idx = np.arange(1, n + 1, dtype=np.float64)
        seq = idx ** alpha
        total = seq.sum()
        if total == 0.0:
            continue
        seq /= total
        sc = _score(seq)
        if sc < best_score:
            best_score = sc
            best_seq = seq.copy()
            print(f"[power] α={alpha:5.2f} → score {sc:.6f}")

    # ----- exponential decay: a_i ∝ exp(-β·i)  (β ∈ [0,0.01]) ----------
    for beta in np.linspace(0.0, 0.010, 21):
        if time.time() >= deadline:
            break
        idx = np.arange(n, dtype=np.float64)
        seq = np.exp(-beta * idx)
        seq /= seq.sum()
        sc = _score(seq)
        if sc < best_score:
            best_score = sc
            best_seq = seq.copy()
            print(f"[exp   ] β={beta:6.4f} → score {sc:.6f}")

    # ----- two‑region piecewise constant --------------------------------
    # first region (length = cut) gets weight w, the rest gets (1‑w)
    for cut_frac in np.linspace(0.1, 0.9, 9):
        if time.time() >= deadline:
            break
        cut = int(cut_frac * n)
        if cut == 0 or cut == n:
            continue
        for w in np.linspace(0.5, 0.95, 10):
            if time.time() >= deadline:
                break
            first_val = w / cut
            second_val = (1.0 - w) / (n - cut)
            seq = np.full(n, second_val, dtype=np.float64)
            seq[:cut] = first_val
            sc = _score(seq)
            if sc < best_score:
                best_score = sc
                best_seq = seq.copy()
                print(
                    f"[piece] cut={cut_frac:.2f}, w={w:.2f} → score {sc:.6f}"
                )

    # ------------------------------------------------------------------
    # 3️⃣  Local refinement – projected sub‑gradient descent
    # ------------------------------------------------------------------
    cur_seq = best_seq.copy()
    step_factor = 0.02               # relative amount moved each step
    it = 0
    while time.time() < deadline:
        # convolution and the index attaining the current maximum
        m = 1 << ((2 * n - 1).bit_length())
        ft = np.fft.rfft(cur_seq, m)
        conv = np.fft.irfft(ft * ft, m)[: 2 * n - 1]
        k_max = int(np.argmax(conv))

        # contributions of each position to this peak (c_i = a_i * a_{k-i})
        idx = np.arange(n)
        partner = k_max - idx
        mask = (partner >= 0) & (partner < n)
        contrib = np.zeros_like(cur_seq)
        contrib[mask] = cur_seq[idx[mask]] * cur_seq[partner[mask]]

        # pick a source (large contribution) and a target (small height)
        src = int(np.argmax(contrib))
        tgt = int(np.argmin(cur_seq))

        if src == tgt:
            # fall back to a random move if everything is equal
            src = random.randrange(n)
            tgt = random.randrange(n)
            while tgt == src:
                tgt = random.randrange(n)

        delta = cur_seq[src] * step_factor
        if delta <= 0.0:
            break

        # propose the move
        cur_seq[src] -= delta
        cur_seq[tgt] += delta

        # re‑normalise to keep sum == 1 (numerical drift)
        cur_seq /= cur_seq.sum()

        new_score = _score(cur_seq)

        if new_score < best_score:
            best_score = new_score
            best_seq = cur_seq.copy()
            print(f"[iter {it:05d}] improved → {new_score:.6f}")
        else:
            # reject the move (simple greedy acceptance)
            cur_seq[tgt] -= delta
            cur_seq[src] += delta
            # slowly shrink the step size if we're stuck
            step_factor *= 0.995

        it += 1

    print(f"[final] best score = {best_score:.6f}")
    # ``evaluate_sequence`` will clamp negatives/large values anyway,
    # but we keep everything non‑negative and normalised.
    return best_seq.tolist()
