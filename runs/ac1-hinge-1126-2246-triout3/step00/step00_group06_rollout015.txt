def propose_candidate(seed=..., budget_s=1000):

    # decide n possibly from height_sequence_1 else 1500

    while time.time() < deadline:
        # Randomly choose generation method
        method = np.random.choice(['exp', 'power', 'dirichlet', 'perturb', 'mix'])
        if method == 'exp':
            # sample r from uniform on [0.5, 0.99] maybe log uniform
            # compute a_i = r**i

            # maybe also random offset
        elif method == 'power':

            raw = 1.0 / np.power(np.arange(1, n+1), p)
        elif method == 'dirichlet':
            alpha = np.random.uniform(0.1, 5.0)  # concentration
            raw = np.random.dirichlet(np.full(n, alpha))
        elif method == 'perturb':
            # start from best_seq and perturb
            seq = best_seq.copy()
            i = np.random.randint(n)
            j = np.random.randint(n)
            delta = seq[i] * np.random.uniform(0.001, 0.1)
            seq[i] = max(0.0, seq[i] - delta)
            seq[j] = seq[j] + delta
            raw = seq
        else: # mix: combine uniform and best
            # maybe average uniform and best
            t = np.random.rand()
            raw = (1 - t) * best_seq + t * np.random.random(n)

        raw = np.maximum(raw, 0)
        if raw.sum() == 0:

        raw = raw / raw.sum()
        # evaluate
        score = evaluate_sequence(raw.tolist())
        if score < best_score:
            best_score = score
            best_seq = raw
            print(f"New best {score:.6f} at time {time.time():.2f}")

        # optional local search on best_seq after certain intervals
        # maybe small hill climbing with few steps
        # we can embed inside loop
    return best_seq.tolist()

def simplex_projection(v):
    """Projects v onto simplex sum(v)=1, v>=0."""
    # code from Duchi: sort descending
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u + (1 - cssv) / np.arange(1, n+1) > 0)[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1)
    w = np.maximum(v - theta, 0)
    return w

"""
Propose a sequence of non‑negative heights that minimizes the evaluation
function.  The core idea is to use a *discrete power‑law* shape
a_i ∝ (i+α)^{-p} (with p≈0.5) which mimics the continuous kernel
f(x)=1/√x whose self‑convolution is constant (the optimal flat
autocorrelation).  A small randomised search over the exponent p,
the offset α and over a few alternative families (exponential decay,
Dirichlet draws) is performed.  The best candidate found is then
refined by a simple hill‑climbing move that transfers a tiny amount of
mass from one index to another while keeping the L1‑norm equal to 1.
The whole search runs until the allotted time (≈1000 s) expires and
returns the best sequence discovered.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities
# ----------------------------------------------------------------------
def _project_to_simplex(v):
    """
    Projects a real vector onto the probability simplex:
        {x | x_i >= 0, sum x_i = 1}
    (standard O(n log n) algorithm by Duchi et al.)
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u + (1.0 - cssv) / np.arange(1, n + 1) > 0)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def _make_powerlaw(n, p, alpha):
    """Return a normalized power‑law vector a_i ∝ (i+α)^{-p}."""
    idx = np.arange(1, n + 1) + alpha   # shift by α (α≥0)
    raw = 1.0 / np.power(idx, p)
    raw_sum = raw.sum()
    if raw_sum == 0:
        return np.full(n, 1.0 / n)
    return raw / raw_sum


def _make_exponential(n, r):
    """Return a normalized exponential vector a_i ∝ r**i (0<r<1)."""
    raw = np.power(r, np.arange(n))
    raw_sum = raw.sum()
    if raw_sum == 0:
        return np.full(n, 1.0 / n)
    return raw / raw_sum


def _make_dirichlet(n, concentration):
    """Draw a Dirichlet sample with equal concentration parameter."""
    raw = np.random.dirichlet(np.full(n, concentration))
    return raw


def _local_transfer(seq, frac):
    """
    Move a small fraction (frac) of mass from a random index i to a
    random index j (i≠j).  The result is automatically renormalised.
    """
    n = seq.shape[0]
    i = np.random.randint(n)
    j = np.random.randint(n - 1)
    if j >= i:
        j += 1
    delta = seq[i] * frac
    if delta <= 0:
        return seq
    new_seq = seq.copy()
    new_seq[i] -= delta
    new_seq[j] += delta
    # because we only transferred mass the sum is still 1,
    # but we clip possible tiny negatives caused by rounding.
    new_seq[i] = max(new_seq[i], 0.0)
    # re‑project onto the simplex to guard against numerical drift
    return _project_to_simplex(new_seq)


# ----------------------------------------------------------------------
# Main search routine
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """Return a list of non‑negative heights with a low evaluation score."""
    np.random.seed(seed)
    deadline = time.time() + budget_s - 5.0   # a small safety margin

    # ------------------------------------------------------------------
    # Determine the length of the sequence (use the previous construction if
    # available, otherwise default to 1500)
    # ------------------------------------------------------------------
    global height_sequence_1
    if 'height_sequence_1' in globals() and isinstance(height_sequence_1, (list, np.ndarray)):
        n = len(height_sequence_1)
        start_seq = np.array(height_sequence_1, dtype=float)
        start_seq = np.maximum(start_seq, 0.0)
        start_seq = start_seq / start_seq.sum()          # normalise to sum = 1
    else:
        n = 1500
        start_seq = np.full(n, 1.0 / n)                  # uniform fallback

    # initialise the best candidate
    best_seq = start_seq.copy()
    best_score = evaluate_sequence(best_seq.tolist())
    print(f"[init] length={n}, start score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Randomised exploration of parametrised families
    # ------------------------------------------------------------------
    while time.time() < deadline:
        # randomly pick a generation method
        r = np.random.rand()
        if r < 0.35:
            # power‑law family (p≈0.5 is promising)
            p = np.random.uniform(0.30, 0.70)                 # exponent
            alpha = np.random.exponential(scale=2.0)         # shift, often close to 0
            cand = _make_powerlaw(n, p, alpha)
        elif r < 0.65:
            # exponential decay
            log_r = np.random.uniform(np.log(0.5), np.log(0.99))
            r_val = np.exp(log_r)
            cand = _make_exponential(n, r_val)
        else:
            # Dirichlet draw with random concentration
            conc = np.random.uniform(0.05, 5.0)
            cand = _make_dirichlet(n, conc)

        # Occasionally blend with a little uniform component (helps escape
        # pathological spikes); the blend factor is kept tiny.
        if np.random.rand() < 0.15:
            lam = np.random.rand() ** 2        # bias towards very small λ
            cand = (1.0 - lam) * cand + lam * (1.0 / n)

        # Normalise (safety)
        cand = np.maximum(cand, 0.0)
        cand_sum = cand.sum()
        if cand_sum == 0:
            continue
        cand = cand / cand_sum

        # evaluate
        score = evaluate_sequence(cand.tolist())
        if score < best_score:
            best_score = score
            best_seq = cand.copy()
            print(f"[improve] new best {best_score:.6f} (method {r:.2f})")

        # --------------------------------------------------------------
        # Local hill‑climbing on the current best solution (occasional)
        # --------------------------------------------------------------
        if np.random.rand() < 0.08:
            # the step size decays as we get closer to the deadline
            remaining = deadline - time.time()
            frac = 0.05 * (remaining / budget_s)   # up to 5 % early, smaller later
            candidate = _local_transfer(best_seq, frac)
            cand_score = evaluate_sequence(candidate.tolist())
            if cand_score < best_score:
                best_score = cand_score
                best_seq = candidate.copy()
                print(f"[local] improved to {best_score:.6f}")

    # ------------------------------------------------------------------
    # Final sanity check and return
    # ------------------------------------------------------------------
    print(f"[final] best score obtained: {best_score:.6f}")
    # make sure we return a plain Python list of floats
    return best_seq.tolist()
