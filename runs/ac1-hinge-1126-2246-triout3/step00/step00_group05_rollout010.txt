def greedy_sidon_set(N, size_target):
    # returns list of integers < N of length size_target (or less if impossible)
    set_vals = []
    sum_set = set()
    for candidate in range(N):
        # compute sums with existing
        conflict = False
        for x in set_vals:
            if candidate + x in sum_set:
                conflict = True
                break
        if conflict:
            continue
        # check self sum
        if candidate + candidate in sum_set:
            continue
        # no conflict, add
        # add new sums
        for x in set_vals:
            sum_set.add(candidate + x)
        sum_set.add(candidate + candidate)
        set_vals.append(candidate)
        if len(set_vals) >= size_target:
            break
    return set_vals

"""
Propose a sequence of non‑negative heights that beats the uniform bound (score = 2).

The construction is based on the following idea:

* After normalising the total height to √(2 n) the objective becomes simply
  max(conv(a,a)).  Hence we have to make the *largest* autocorrelation value
  as small as possible.

* Split the indices into two groups.
    – “large’’ positions (a few of them) receive a relatively big value w.
    – All remaining positions receive a tiny uniform value ε.

* The large positions are chosen to form a **Sidon set** (all unordered
  sums i+j are distinct).  With this property every index of the convolution
  contains at most one product of two large entries, i.e. at most
  2·w² for i≠j and w² for i=j.

* By placing the large positions among the smallest indices we keep the
  number of small‑small products at those indices small (the number of
  pairs that sum to k grows only ~k+1 near the left edge).  Consequently
  the three possible contributors to the maximum are
        ε²·n          – centre of the small‑only block,
        2·w²          – a large–large pair,
        ε·w·m         – large–small overlap.
  We search for parameters (m, w) that make all three below the current
  best score.

The algorithm proceeds as follows:

1. Determine n from the provided ``height_sequence_1`` and the target sum
   S = √(2 n).
2. Build a maximal Sidon set (greedy construction) and keep its sorted
   prefix – this guarantees distinct sums for any chosen prefix length m.
3. Repeatedly sample a pair (m,w):
       – m  : number of large entries (1 … |Sidon|),
       – w  : large height (chosen below sqrt(best_score/2) to keep 2·w²
                under the current best bound).
   Compute the required ε from the sum constraint.
   If ε≥0 and the cheap estimate
        max(ε²·n, 2·w², ε·w·m)  <  current best,
   build the full sequence, normalise it to the exact sum S (scaling does
   not change the objective) and evaluate it.
   Keep the best sequence found.

4. When time runs out, return the best sequence.

The search uses only elementary NumPy operations, a greedy Sidon set
construction (O(n·√n)) and a fast loop of random candidates, well within
the 1000 s budget.
"""

import time
import random
import numpy as np

# Helper: greedy construction of a Sidon set (all unordered sums distinct)
def greedy_sidon_set(N: int, target_size: int):
    """
    Returns a list of at most `target_size` integers in [0,N) such that
    all unordered sums i+j are distinct.  The algorithm scans the
    numbers from 0 upward and inserts a number iff it does not create a
    duplicate sum with the already chosen ones.
    """
    chosen = []
    sums = set()          # stores i+j for all i,j already chosen (i≤j)
    for cand in range(N):
        # test whether any new sum would clash
        clash = False
        # i=j case
        if (cand + cand) in sums:
            continue
        for a in chosen:
            if (cand + a) in sums:
                clash = True
                break
        if clash:
            continue
        # no clash – add cand
        for a in chosen:
            sums.add(cand + a)
        sums.add(cand + cand)
        chosen.append(cand)
        if len(chosen) >= target_size:
            break
    return chosen


# ------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a step‑function height vector with a low value of
    `evaluate_sequence`.  The function returns the best sequence found
    before the time limit expires.
    """
    random.seed(seed)
    np.random.seed(seed)

    # --------------------------------------------------------
    # problem size and normalisation
    n = len(height_sequence_1)           # length of the known construction
    S = np.sqrt(2.0 * n)                 # required total sum after normalisation

    # start from the uniform sequence (score = 2.0) – it is a valid baseline
    uniform_val = S / n
    best_seq = [float(uniform_val)] * n
    best_score = evaluate_sequence(best_seq)
    print(f"Initial uniform score = {best_score:.6f}")

    # --------------------------------------------------------
    # build a large Sidon set (maximal possible size ≈ √n)
    max_m = int(np.sqrt(n))              # theoretical upper bound for a Sidon set
    sidon_full = greedy_sidon_set(n, max_m)
    sidon_full.sort()
    max_possible = len(sidon_full)
    print(f"Sidon set size = {max_possible} (target length = {n})")

    # --------------------------------------------------------
    deadline = time.time() + budget_s - 5.0   # leave a few seconds margin

    # main random search
    while time.time() < deadline:
        # pick a number of large entries
        m = random.randint(1, max_possible)

        # admissible large height: we must keep 2·w² < best_score
        # (otherwise the large–large contribution can already beat the best)
        w_upper = np.sqrt(best_score / 2.0) - 1e-9
        if w_upper <= 0.0:
            continue
        # random w in a reasonable interval
        w = random.uniform(0.05, w_upper)

        # compute the small value ε that satisfies the sum constraint
        eps = (S - m * w) / (n - m)
        if eps < 0.0:
            continue

        # cheap upper bound on the eventual max convolution
        #   ε²·n          – centre of the small‑only block
        #   2·w²          – a large–large pair (worst case)
        #   ε·w·m         – many large–small contributions at one index
        cheap = max(eps * eps * n, 2.0 * w * w, eps * w * m)
        if cheap >= best_score:
            continue

        # build the candidate sequence
        seq = np.full(n, eps, dtype=np.float64)
        seq[sidon_full[:m]] = w

        # renormalise to the exact required sum (scaling does not change the score)
        total = seq.sum()
        if total != 0.0:
            seq *= (S / total)

        # evaluate – any exception (e.g. numerical) is ignored
        try:
            score = evaluate_sequence(seq.tolist())
        except Exception:          # pragma: no cover
            continue

        if score < best_score:
            best_score = score
            best_seq = seq.tolist()
            print(f"New best {best_score:.6f} (m={m}, w={w:.4f}, ε={eps:.5f})")

    # --------------------------------------------------------
    # final sanity check – ensure the returned sequence satisfies the
    # constraints (non‑negative, correct length).  The evaluator already
    # performs these checks, but we keep a cheap guard.
    best_seq = [float(max(0.0, x)) for x in best_seq]
    if len(best_seq) != n:
        # fallback to the uniform sequence if something went wrong
        best_seq = [float(uniform_val)] * n

    print(f"Returning sequence with score {best_score:.6f}")
    return best_seq
