def max_convolution(a):
    # compute a * a convolution via FFT
    n = len(a)
    # next power of two length for speed maybe
    N = 1 << (2*n-1).bit_length()
    # compute FFT
    A = np.fft.rfft(a, n=N)
    b = np.fft.irfft(A * A, n=N)
    # b length N, but actual conv length is 2*n-1 (first 2*n-1 entries)
    b = b[:2*n-1]
    return b.max()

def propose_candidate(budget_s=1000, seed=0):
    import numpy as np, time, math, sys
    np.random.seed(seed)
    deadline = time.time() + budget_s - 2

    best_seq = None
    best_score = float('inf')
    target_score = 1.503
    iteration = 0

    def conv_max(a):
        n = a.shape[0]
        N = 1 << ((2 * n - 1).bit_length())
        A = np.fft.rfft(a, n=N)
        b = np.fft.irfft(A * A, n=N)[:2*n-1]
        return b

    while time.time() < deadline:
        # choose n randomly between 800 and 2000
        n = np.random.randint(800, 2000)
        S = math.sqrt(2.0 * n)
        # random init: draw from exponential to give varied distribution
        a = np.random.exponential(scale=1.0, size=n)
        a = a * (S / a.sum())
        # maybe apply a few smoothing steps to avoid extremes?
        # but keep as is.

        # compute initial score
        b = conv_max(a)
        M = b.max()
        score = M # since sum fixed
        if score < best_score:
            best_score = score
            best_seq = a.tolist()
            print(f'Initial seq length {n}, score {score:.6f}')
        # local search
        step_no_improve = 0
        max_no_improve = 2000
        while time.time() < deadline and step_no_improve < max_no_improve:
            iteration += 1
            # compute conv and max
            b = conv_max(a)
            M = b.max()
            k = int(np.argmax(b)) # particular index
            # compute product contributions for this k
            i_start = max(0, k - (n - 1))
            i_end = min(n - 1, k)
            idx = np.arange(i_start, i_end + 1)
            jdx = k - idx
            prod = a[idx] * a[jdx]
            # choose src as max product
            src = idx[np.argmax(prod)]
            # choose dest as index with minimum a (global) not equal src
            dest = np.argmin(a)
            if dest == src:
                # pick second minimal
                sorted_idx = np.argsort(a)
                dest = sorted_idx[1] if sorted_idx[0] == src else sorted_idx[0]
            # compute epsilon: up to fraction of src value
            max_eps = a[src] * 0.2  # allow up to 20%
            if max_eps <= 0:
                # src is zero, pick random move
                src = np.random.randint(n)
                dest = np.random.randint(n)
                while dest == src:
                    dest = np.random.randint(n)
                max_eps = a[src] * 0.1 if a[src] > 0 else 0.0
            eps = max_eps * np.random.rand()
            if eps <= 0:
                step_no_improve += 1
                continue
            a_new = a.copy()
            a_new[src] -= eps
            a_new[dest] += eps
            # numerical stability: ensure no negative tiny
            a_new = np.clip(a_new, 0.0, None)
            # sum unchanged (approx)
            # recalc conv
            b_new = conv_max(a_new)
            M_new = b_new.max()
            if M_new < M - 1e-12:  # strict improvement
                a = a_new
                score = M_new
                step_no_improve = 0
                if M_new < best_score:
                    best_score = M_new
                    best_seq = a.tolist()
                    print(f'Iter {iteration}, new best {best_score:.6f}, len {n}')
                    if best_score < target_score:
                        print('Target reached')
                        return best_seq
            else:

                # occasional random jitter to escape plateau
                if step_no_improve % 100 == 0:

                    a = a_jitter
        # after local search, restart
    # after budget
    print(f'Best score after search: {best_score:.6f}')

"""
Local-search heuristic for minimizing the self‑convolution maximum.
The evaluation function (after normalising the sum to sqrt(2·n))
is simply  max_k (a * a)_k .  This routine builds a random
non‑negative sequence, then repeatedly:
  * computes the convolution via an FFT,
  * finds the index where the maximum occurs,
  * identifies the pair (i,j) that contributes the largest product
    to that maximum,
  * moves a small amount of mass from the “worst’’ index to the
    “best’’ (lowest‑product) index,
  * accepts the move only if the maximum decreases.
If no improvement is seen for a while the algorithm perturbs the
sequence randomly and restarts with a new length.  The whole search
stops after the allotted time (default 1000 s) and returns the best
sequence found so far.
The code uses only NumPy (no external LP/solver) and therefore runs
comfortably within the 2‑CPU limit.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Helper: convolution via FFT (returns full linear convolution array)
def _full_convolution(a: np.ndarray) -> np.ndarray:
    """Linear convolution of a with itself using an FFT (real input)."""
    n = a.shape[0]
    N = 1 << ((2 * n - 1).bit_length())      # next power of two ≥ 2n‑1
    A = np.fft.rfft(a, n=N)
    b = np.fft.irfft(A * A, n=N)[: 2 * n - 1]
    return b


# ----------------------------------------------------------------------
def propose_candidate(budget_s: float = 1000, seed: int = 0):
    """
    Search for a sequence of non‑negative heights that minimises
    ``evaluate_sequence``.  The function runs for at most ``budget_s``
    seconds and returns a plain Python list of floats.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 2.0   # leave a couple of seconds margin

    best_seq = None
    best_score = float("inf")
    TARGET = 1.503          # target value mentioned in the statement

    # ------------------------------------------------------------------
    def _score_from_conv(conv: np.ndarray) -> float:
        """Score equals the maximal convolution entry (sum already normalised)."""
        return float(conv.max())

    # ------------------------------------------------------------------
    while time.time() < deadline:
        # ---------- pick a fresh problem size ----------
        n = np.random.randint(800, 2000)          # moderate size, cheap FFTs
        S = math.sqrt(2.0 * n)                    # required L1‑norm

        # ---------- random initial vector (exponential gives a spread) ----------
        a = np.random.exponential(scale=1.0, size=n)
        a = a * (S / a.sum())                     # enforce exact sum

        # ---------- evaluate initial candidate ----------
        conv = _full_convolution(a)
        cur_score = _score_from_conv(conv)

        if cur_score < best_score:
            best_score = cur_score
            best_seq = a.tolist()
            print(f"[init] n={n}  score={cur_score:.6f}")

        # ---------- local improvement loop ----------
        no_improve = 0          # how many successive attempts without gain
        MAX_NO_IMPROVE = 2000   # after this we restart with a new random start

        while time.time() < deadline and no_improve < MAX_NO_IMPROVE:
            # compute convolution and locate the current maximum
            conv = _full_convolution(a)
            M = conv.max()
            k = int(conv.argmax())   # index where the maximum occurs

            # contributions of each index to the active maximum
            i_start = max(0, k - (n - 1))
            i_end = min(n - 1, k)
            idx = np.arange(i_start, i_end + 1)
            jdx = k - idx
            prod = a[idx] * a[jdx]                # product a_i * a_{k-i}

            # source = index with the largest product, destination = smallest product
            src = int(idx[np.argmax(prod)])
            dst = int(idx[np.argmin(prod)])

            # if src and dst coincide (unlikely when many equal) pick a global minimum
            if src == dst:
                dst = int(np.argmin(a))

            # amount to move: a fraction of the source entry (up to 20 %)
            max_eps = a[src] * 0.20
            if max_eps <= 0.0:
                # nothing to move – break out and restart
                no_improve += 1
                continue
            eps = max_eps * np.random.rand()      # randomised step size

            # propose the move (sum stays exactly S)
            a_new = a.copy()
            a_new[src] -= eps
            a_new[dst] += eps
            # numerical safety – clip tiny negatives that may appear because of rounding
            if a_new[src] < 0.0:
                a_new[src] = 0.0

            # evaluate new maximum
            conv_new = _full_convolution(a_new)
            M_new = conv_new.max()

            if M_new < M - 1e-12:                 # accept only genuine improvement
                a = a_new
                cur_score = float(M_new)
                no_improve = 0
                if cur_score < best_score:
                    best_score = cur_score
                    best_seq = a.tolist()
                    print(f"[impr] iter={int(time.time())}  n={n}  score={cur_score:.6f}")
                    if best_score < TARGET:
                        print(f"Target reached ({best_score:.6f}).")
                        return best_seq
            else:
                # no improvement – count it
                no_improve += 1
                # occasional random jitter to escape flat plateaux
                if no_improve % 100 == 0:
                    # small Gaussian perturbation on a tiny random subset
                    jitter_idx = np.random.choice(n, size=min(5, n), replace=False)
                    jitter = np.random.randn(jitter_idx.size) * a[jitter_idx] * 0.05
                    a[jitter_idx] = np.clip(a[jitter_idx] + jitter, 0.0, None)
                    # re‑normalise the sum (tiny deviation may accumulate)
                    a = a * (S / a.sum())

        # end of inner improvement loop – either time ran out or stagnation
        # a fresh random start will be attempted automatically by the outer loop

    # ------------------------------------------------------------------
    # final reporting
    print(f"Search finished. Best score found: {best_score:.6f}")
    if best_seq is None:
        # fallback – a tiny valid sequence
        best_seq = [0.0, 0.0, 0.0]
    return best_seq
