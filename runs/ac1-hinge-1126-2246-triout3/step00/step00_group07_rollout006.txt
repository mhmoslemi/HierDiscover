"""
Propose a candidate sequence for the step‑function optimization problem.

The algorithm works in two phases:

1. **Analytic initialization** – generate a family of sequences of the form  
   a_i = (i + offset) ** (-p) (with p≈0.5).  These sequences are known to make
   the discrete auto‑convolution almost constant (the continuous analogue
   f(x)=1/√x satisfies f∗f≡π).  We scan a small grid of (offset, p) values,
   keep the one that yields the smallest evaluation score.

2. **Greedy equalisation** – starting from the best analytic candidate we repeatedly
   locate the convolution index `k_max` where the current bound is tight.
   The contribution of each coefficient a_i to this maximum is
   `c_i = a_i * a_{k_max-i}` (zero when the pair is out of range).  We move a
   small amount of mass from the index with the largest contribution to an
   index with the smallest contribution.  This deterministic step flattens
   the convolution profile and monotonically improves the score.  Random
   jitter is added occasionally to escape possible plateaus.

The whole search runs until the provided time budget is almost exhausted.
The returned sequence is a plain Python list of floats that satisfies all
security checks of `evaluate_sequence`.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper functions
# ----------------------------------------------------------------------


def make_power_sequence(n: int, power: float = 0.5, offset: float = 1.0) -> np.ndarray:
    """
    Return an n‑element non‑negative sequence a_i = (i + offset) ** (-power).

    Parameters
    ----------
    n : int
        Length of the sequence.
    power : float, optional
        Exponent (default 0.5, i.e. 1/√).
    offset : float, optional
        Positive shift to avoid singularities (default 1.0 gives a_0 = 1).

    Returns
    -------
    np.ndarray
        Array of shape (n,) containing the generated coefficients.
    """
    idx = np.arange(n, dtype=np.float64)
    return (idx + offset) ** (-power)


def greedy_flatten(a: np.ndarray, max_iter: int, deadline: float) -> np.ndarray:
    """
    Perform a deterministic greedy redistribution that reduces
    max(conv(a,a)) while keeping the L1‑norm unchanged.

    Parameters
    ----------
    a : np.ndarray
        Starting sequence (will be modified in‑place).
    max_iter : int
        Upper bound on the number of redistribution steps.
    deadline : float
        Unix time after which the routine must stop.

    Returns
    -------
    np.ndarray
        The best sequence seen during the run (a copy, not a view).
    """
    n = a.shape[0]
    best_seq = a.copy()
    best_score = evaluate_sequence(best_seq.tolist())
    print(f"[greedy] initial score = {best_score:.6f}")

    for it in range(max_iter):
        if time.time() >= deadline:
            break

        # convolution and the position of the maximal entry
        conv = np.convolve(a, a)
        k_max = int(np.argmax(conv))

        # contribution of each index to that maximal entry
        idx = np.arange(n, dtype=np.int64)
        j = k_max - idx
        mask = (j >= 0) & (j < n)
        contrib = np.zeros_like(a)
        contrib[mask] = a[mask] * a[j[mask]]

        # indices with largest / smallest contributions
        i_high = int(np.argmax(contrib))
        i_low = int(np.argmin(contrib))

        # move a tiny fraction from the high‑impact index to the low‑impact one
        delta = min(a[i_high] * 0.02, 0.001)   # keep steps modest
        if delta <= 0.0:
            break

        a[i_high] -= delta
        a[i_low] += delta

        # occasional random jitter to avoid stagnation
        if it % 100 == 0:
            rnd_idx = np.random.randint(n)
            a[rnd_idx] = max(0.0, a[rnd_idx] + np.random.randn() * 0.0005)

        # compute the new score
        cur_score = evaluate_sequence(a.tolist())
        if cur_score < best_score:
            best_score = cur_score
            best_seq = a.copy()
            print(f"[greedy] iter {it:4d} – improved score = {best_score:.6f}")

    print(f"[greedy] finished – best score = {best_score:.6f}")
    return best_seq

def propose_candidate(seed: int = 0, budget_s: float = 1000.0, **kwargs):
    """
    Generate a step‑function height sequence that (hopefully) improves upon
    the current best upper bound (≈2.0).  The routine respects the time
    budget and returns the best sequence discovered.

    Parameters
    ----------
    seed : int, optional
        Seed for the internal RNG (default 0 → deterministic).
    budget_s : float, optional
        Total allowed time in seconds (default 1000 s).
    **kwargs
        Ignored – present only to match the expected signature.

    Returns
    -------
    list[float]
        The sequence of non‑negative heights.
    """
    np.random.seed(seed)
    # leave a small safety margin for the caller
    deadline = time.time() + budget_s - 5.0

    # ------------------------------------------------------------------
    # 1. Analytic grid search
    # ------------------------------------------------------------------
    n = 5200                     # length that works well with the FFT based conv
    power = 0.5                  # start from the theoretical exponent
    offsets = np.linspace(1.0, 4.0, 7)      # try a few offsets: 1, 1.5, …, 4
    best_a = None
    best_score = np.inf

    print("[search] starting simple grid search...")
    for off in offsets:
        a = make_power_sequence(n, power=power, offset=off)
        # tiny normalisation – not required but keeps numbers away from extremes
        scale = np.sum(a)
        if scale < 1e-12:
            continue
        a = a / scale * np.sqrt(2 * n)   # mimic the classical normalisation
        score = evaluate_sequence(a.tolist())
        print(f"  offset={off:.2f} → score={score:.6f}")
        if score < best_score:
            best_score = score
            best_a = a.copy()

    # fallback – if something went wrong use the classic 1/√(i+1) shape

        best_a = make_power_sequence(n, power=0.5, offset=1.0)

    # ------------------------------------------------------------------
    # 2. Greedy flattening / local optimisation
    # ------------------------------------------------------------------
    max_iter = 2000
    optimized_a = greedy_flatten(best_a, max_iter, deadline)

    # final sanity check – evaluate once more (may have changed due to time limit)
    final_score = evaluate_sequence(optimized_a.tolist())
    print(f"[final] sequence length = {len(optimized_a)} – final score = {final_score:.6f}")

    return optimized_a.tolist()
