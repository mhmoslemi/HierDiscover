"""
Subgradient and LP based optimizer for the step‑function evaluation.
The algorithm performs many short random restarts.
For each restart a random Dirichlet vector (sum = 1) is refined by:
- a stochastic sub‑gradient step that reduces the peak of the self‑convolution,
- occasional linear‑programming “direction” updates (as in the reference SOTA)
The best sequence found within the time budget is returned.
"""

import time
import numpy as np
from scipy import optimize

def convolve_fft(x):
    """Return the linear convolution x*x using a real FFT (length 2*len(x)-1)."""
    n = x.shape[0]
    size = 1 << ((2 * n - 1).bit_length())  # next power of two
    # real FFT of the vector
    fx = np.fft.rfft(x, n=size)
    # pointwise product corresponds to convolution of x with itself
    conv = np.fft.irfft(fx * fx, n=size)
    return conv[: 2 * n - 1]

def evaluate_local(seq):
    """Fast evaluation (same formula as evaluate_sequence)."""
    s = seq.sum()
    if s < 1e-12:
        return float("inf")
    n = len(seq)
    conv = convolve_fft(seq)
    m = conv.max()
    return 2.0 * n * m / (s * s)

def subgradient(seq):
    """
    Compute a sub‑gradient of the functional max_k (seq*seq)_k.
    Returns (g, max_value, conv) where ``g`` is the gradient vector.
    """
    n = len(seq)
    conv = convolve_fft(seq)
    max_val = conv.max()
    # indices where the maximum is attained (within a tiny tolerance)
    tol = max_val * 1e-12 + 1e-14
    active = np.where(conv >= max_val - tol)[0]

    grad = np.zeros_like(seq)
    # For each active convolution index k we add the contribution
    #   ∂/∂a_i  sum_j a_j a_{k‑j} = 2 a_{k‑i}   (when the index is valid)
    for k in active:
        i_start = max(0, k - (n - 1))
        i_end = min(n - 1, k)
        i_range = np.arange(i_start, i_end + 1, dtype=int)
        j_range = k - i_range
        grad[i_range] += 2.0 * seq[j_range]
    return grad, max_val, conv

def solve_convolution_lp(f_seq, rhs):
    """
    Linear programme: maximise sum(b) subject to (f * b)_k ≤ rhs and b ≥ 0.
    Returns the optimal vector ``b`` (numpy ndarray) or ``None`` on failure.
    """
    n = len(f_seq)
    # maximise sum(b)  → minimise -sum(b)
    c = -np.ones(n)

    # Build matrix for the convolution constraints.
    # Each row corresponds to a fixed k (0 … 2n‑2) and the entry for variable j is f[i]
    # with i = k‑j, provided i is in the range [0, n‑1].
    rows = []
    rhs_vec = []
    for k in range(2 * n - 1):
        row = np.zeros(n)
        # i runs over the indices of f that can appear in the kth convolution term
        i_min = max(0, k - (n - 1))
        i_max = min(n - 1, k)
        # j = k - i
        for i in range(i_min, i_max + 1):
            j = k - i
            row[j] = f_seq[i]
        rows.append(row)
        rhs_vec.append(rhs)

    # Non‑negativity constraints: -I·b ≤ 0
    rows.append(-np.eye(n))
    rhs_vec.append(np.zeros(n))

    A_ub = np.vstack(rows)
    b_ub = np.hstack(rhs_vec)

    # Use the interior‑point method; enforce a modest time limit per call.
    res = optimize.linprog(
        c,
        A_ub=A_ub,
        b_ub=b_ub,
        method="highs",
        options={"time_limit": 8.0}
    )
    if res.success:
        return res.x
    return None

"""
Sub‑gradient + occasional LP optimiser for the step‑function evaluation.

The goal is to minimise
        2·n·max( a * a ) / ( sum a )²
for a non‑negative sequence a.
The routine performs many short random restarts.
Each restart starts from a Dirichlet‑distributed vector (sum = 1) and is refined by

* a stochastic sub‑gradient step that reduces the peak of the self‑convolution,
* occasional linear‑programming direction updates (the same idea used in the
  reference SOTA code).

The best sequence found before the time budget expires is returned.
"""

import time
import numpy as np
from scipy import optimize

# ----------------------------------------------------------------------
# Helper routines (top‑level – no closures)
# ----------------------------------------------------------------------


def convolve_fft(x):
    """Linear convolution x*x using a real FFT (returns length 2·len(x)‑1)."""
    n = x.shape[0]
    size = 1 << ((2 * n - 1).bit_length())          # next power of two
    fx = np.fft.rfft(x, n=size)
    conv = np.fft.irfft(fx * fx, n=size)            # inverse transform
    return conv[: 2 * n - 1]


def evaluate_local(seq):
    """Fast replica of the supplied ``evaluate_sequence`` (scale‑invariant)."""
    s = seq.sum()
    if s < 1e-12:
        return float("inf")
    n = len(seq)
    conv = convolve_fft(seq)
    m = conv.max()
    return 2.0 * n * m / (s * s)


def subgradient(seq):
    """
    Compute a sub‑gradient of the functional ``max_k (seq*seq)_k``.
    Returns (grad, max_value, conv) where ``grad`` is the gradient vector.
    """
    n = len(seq)
    conv = convolve_fft(seq)
    max_val = conv.max()

    # indices attaining the maximal value (tiny tolerance for numerical noise)
    tol = max_val * 1e-12 + 1e-14
    active = np.where(conv >= max_val - tol)[0]

    grad = np.zeros_like(seq)
    for k in active:
        i_start = max(0, k - (n - 1))
        i_end = min(n - 1, k)
        i_range = np.arange(i_start, i_end + 1, dtype=int)
        j_range = k - i_range
        grad[i_range] += 2.0 * seq[j_range]          # ∂/∂a_i of the k‑th term
    return grad, max_val, conv


def solve_convolution_lp(f_seq, rhs):
    """
    Linear programme:
        maximise   sum(b)
        subject to  (f_seq * b)_k ≤ rhs   for all k,
                    b ≥ 0.

    Returns the optimal vector ``b`` (numpy array) or ``None`` on failure.
    """
    n = len(f_seq)
    # linear objective (maximise sum → minimise -sum)
    c = -np.ones(n)

    rows = []
    rhs_vec = []

    # convolution constraints
    for k in range(2 * n - 1):
        row = np.zeros(n)
        i_min = max(0, k - (n - 1))
        i_max = min(n - 1, k)
        for i in range(i_min, i_max + 1):
            j = k - i
            row[j] = f_seq[i]
        rows.append(row)
        rhs_vec.append(rhs)

    # non‑negativity: -I·b ≤ 0
    rows.append(-np.eye(n))
    rhs_vec.append(np.zeros(n))

    A_ub = np.vstack(rows)
    b_ub = np.hstack(rhs_vec)

    # “highs” is the default modern solver in SciPy.
    try:
        res = optimize.linprog(
            c, A_ub=A_ub, b_ub=b_ub,
            method="highs",
            options={"time_limit": 8.0}
        )
    except Exception:          # fallback – older SciPy versions may not support time_limit
        res = optimize.linprog(c, A_ub=A_ub, b_ub=b_ub, method="highs")

    if res.success:
        return res.x
    return None

def propose_candidate(seed=42, budget_s=1000, **kwargs):
    """
    Search for a sequence that minimises
        2·n·max(a*a) / (sum a)²

    The function respects the supplied time budget and returns the best
    sequence found (as a plain list of floats).
    """
    rng = np.random.default_rng(seed)
    deadline = time.time() + max(0.0, budget_s - 5.0)   # leave a tiny safety margin

    # ------------------------------------------------------------------
    # lengths to explore – always keep the length of the known construction
    # ------------------------------------------------------------------
    lengths = [256, 512, 1024, 2048, 4096]
    try:
        init_len = len(height_sequence_1)   # supplied globally by the platform
        if init_len not in lengths:
            lengths.append(init_len)
    except Exception:
        pass

    # probabilities for picking a length (larger sizes get a slightly larger weight)
    length_weights = np.array(lengths, dtype=float)
    length_weights = length_weights / length_weights.sum()

    best_seq = None
    best_score = float("inf")

    def maybe_update(candidate):
        """Update the global best if the candidate improves the score."""
        nonlocal best_seq, best_score
        sc = evaluate_local(candidate)
        if sc < best_score:
            best_score = sc
            best_seq = candidate.copy()

        return sc

    restart = 0
    while time.time() < deadline:
        # --------------------------------------------------------------
        # pick a problem size and initialise a random Dirichlet vector
        # --------------------------------------------------------------
        n = rng.choice(lengths, p=length_weights)
        alpha = 0.3 + rng.random() * 0.9                     # Dirichlet concentration
        seq = rng.dirichlet(np.full(n, alpha)).astype(float)  # sum == 1

        # --------------------------------------------------------------
        # inner optimisation loop (sub‑gradient + occasional LP)
        # --------------------------------------------------------------
        max_iters = 3000
        step0 = 0.5
        for it in range(max_iters):

            # ----- stochastic sub‑gradient step -----
            grad, max_val, _ = subgradient(seq)
            grad_norm = np.linalg.norm(grad) + 1e-20
            step = step0 / np.sqrt(it + 1)
            noise = rng.normal(scale=step * 0.2, size=n)      # small random kick
            seq_new = seq - step * (grad / grad_norm) + noise
            seq_new = np.maximum(seq_new, 0.0)                # enforce non‑negativity

                g = solve_convolution_lp(seq, max_val)
                if g is not None:
                    g_sum = g.sum()
                    if g_sum > 0:

                    # try a few mixing coefficients
                    for t in (0.1, 0.25, 0.5):

                        cand = np.maximum(cand, 0.0)
                        cand_sum = cand.sum()
                        if cand_sum > 0:

                        cand_score = evaluate_local(cand)
                        if cand_score < best_score:
                            seq = cand
                            maybe_update(seq)
                            break

        # final check after the inner loop
        maybe_update(seq)

        if restart % 5 == 0:

    # ------------------------------------------------------------------

    # ------------------------------------------------------------------
    if best_seq is None:
        try:

        except Exception:
            n = 128
        best_seq = np.ones(n) / n

    # Return a plain Python list as required by the evaluation harness
    return [float(v) for v in best_seq]
