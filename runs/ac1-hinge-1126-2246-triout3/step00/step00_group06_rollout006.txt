def propose_candidate(seed=0, budget_s=1000):
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5

    # Starting sequences: include height_sequence_1, uniform, random
    candidates = []
    # unify to list of floats
    # Let's create uniform sequence length 1500 of 1.0
    uniform = [1.0] * 1500
    candidates.append(uniform)
    if 'height_sequence_1' in globals():
        candidates.append(list(height_sequence_1))
    
    # also random length between 200 and 2000
    candidates.append([np.random.random() for _ in range(np.random.randint(200, 2000))])

    # For each candidate, we will run local search
    best_seq = None
    best_val = float('inf')
    for init_seq in candidates:
        seq = init_seq[:]
        val = evaluate_sequence(seq)
        if val < best_val:
            best_val = val
            best_seq = seq[:]
        # hill climbing loop
        while time.time() < deadline:
            # choose move type
            move_type = np.random.choice(['transfer','zero','add_noise','lp'])
            if move_type == 'transfer':
                # pick i, j distinct
                i,j = np.random.randint(len(seq)), np.random.randint(len(seq))
                if i == j:
                    continue
                delta = np.random.rand() * seq[i]  # up to whole value
                if delta <= 0:
                    continue
                seq[i] -= delta
                seq[j] += delta
            elif move_type == 'zero':
                # zero out a random index and spread its mass uniformly to others
                i = np.random.randint(len(seq))
                mass = seq[i]
                if mass == 0:
                    continue
                seq[i] = 0.0
                # distribute among random subset of size m
                m = max(1, int(0.01 * len(seq)))  # 1% of entries
                idxs = np.random.choice(len(seq), size=m, replace=False)
                add = mass / m
                for idx in idxs:
                    seq[idx] += add
            elif move_type == 'add_noise':
                # add small gaussian noise to each entry
                noise = np.random.randn(len(seq)) * 0.01
                seq = [max(0.0, x + n) for x,n in zip(seq, noise)]
            elif move_type == 'lp':
                # use get_good_direction_to_move_into
                new_seq = get_good_direction_to_move_into(seq)
                if new_seq is not None:
                    seq = new_seq

            val_new = evaluate_sequence(seq)
            if val_new < best_val:
                best_val = val_new
                best_seq = seq[:]
                print(f'New best {best_val:.6f}')
            # Accept move if lower, else maybe revert or accept with probability
            # Use simple hill climbing: keep if better else revert
            if val_new <= val:
                val = val_new
            else:
                # with small prob revert
                if np.random.rand() < 0.01:
                    # keep worse
                    val = val_new
                else:
                    # revert to previous
                    # need to store prev seq
                    pass

def gradient_descent_step(seq, eta=0.1):
    seq_arr = np.array(seq)
    conv = np.convolve(seq_arr, seq_arr)
    max_idx = np.argmax(conv)
    # gradient: each element i derivative of conv[max_idx] w.r.t a_i
    # d/d a_i of sum_{j} a_j * a_{max_idx - j} = a_{max_idx - i}
    grad = np.zeros_like(seq_arr)
    for i in range(len(seq_arr)):
        j = max_idx - i
        if 0 <= j < len(seq_arr):
            if i == j:
                grad[i] = 2 * seq_arr[i]
            else:
                grad[i] = seq_arr[j]
    # step opposite gradient
    new_seq = seq_arr - eta * grad
    new_seq[new_seq < 0] = 0
    # renorm sum to previous sum to avoid scaling changes.
    sum_old = seq_arr.sum()
    sum_new = new_seq.sum()
    if sum_new == 0:
        return seq
    new_seq = new_seq * (sum_old / sum_new)
    return new_seq.tolist()
```

"""
Propose a step‐function (list of non‑negative heights) that minimizes

    evaluate_sequence(seq) = 2 * n * max(conv(seq,seq)) / (sum(seq)^2)

The algorithm is a hybrid stochastic local search:

* Start from a few reasonable seeds (uniform, the provided SOTA sequence,
  a random sequence).
* Repeatedly apply one of several cheap perturbations:
    - random Gaussian noise,
    - transfer a fraction of mass from one index to another,
    - zero out one index and redistribute its mass,
    - a gradient step that reduces the currently maximal convolution entry,
    - a linear‑programming direction (the same LP used in the reference SOTA).

* Accept a move if it improves the current value, otherwise keep the old
  sequence (occasionally accept a worse move to escape local minima).

* Keep the best sequence seen so far and return it when the time budget
  (≈1000 s) expires.

All helper functions are top‑level, no closures or lambdas are used.
"""

import time
import numpy as np
from scipy import optimize

linprog = optimize.linprog


def solve_convolution_lp(f_sequence, rhs):
    """Linear program:
       maximize   Σ b_j
       subject to conv(f, b) ≤ rhs  (component‑wise) and b ≥ 0.
    """
    n = len(f_sequence)
    # maximise Σ b  → minimise -Σ b
    c = -np.ones(n)

    # Construct A_ub, b_ub for each convolution index k
    a_ub = []
    b_ub = []
    for k in range(2 * n - 1):
        row = np.zeros(n)
        for i in range(n):
            j = k - i
            if 0 <= j < n:
                row[j] = f_sequence[i]
        a_ub.append(row)
        b_ub.append(rhs)

    # Non‑negativity constraints  -b ≤ 0
    a_ub = np.vstack([a_ub, -np.eye(n)])
    b_ub = np.hstack([b_ub, np.zeros(n)])

    # use the modern HiGHS solver (fast) and a small time limit per call
    res = linprog(
        c,
        A_ub=a_ub,
        b_ub=b_ub,
        method="highs",
        options={"time_limit": 5.0, "presolve": True},
    )
    if res.success:
        return np.maximum(0.0, res.x)   # clip tiny negatives that can appear
    return None


def get_good_direction_to_move_into(sequence):
    """
    One LP step as described in the SOTA paper.
    Returns a new sequence that is a convex combination of the current one
    and a direction g that maximises Σ b_j under the current convolution bound.
    """
    n = len(sequence)
    seq = np.array(sequence, dtype=float)
    sum_seq = seq.sum()
    if sum_seq < 1e-12:
        return None

    # Normalise to the standard L1 = sqrt(2n) (the normalisation used in the paper)
    seq_norm = seq * np.sqrt(2 * n) / sum_seq
    rhs = np.max(np.convolve(seq_norm, seq_norm))

    g = solve_convolution_lp(seq_norm, rhs)
    if g is None:
        return None

    # Re‑normalise g to the same L1
    sum_g = g.sum()
    if sum_g < 1e-12:
        return None
    g_norm = g * np.sqrt(2 * n) / sum_g

    # Small convex step (t can be tuned; 0.01 works well)
    t = 0.01
    new_seq = (1 - t) * seq + t * g_norm
    return new_seq.tolist()


def gradient_descent_step(sequence, eta=0.05):
    """
    Gradient step that reduces the currently maximal convolution entry.
    Gradient of conv[k*] (k* = argmax conv) w.r.t a_i is a_{k*-i} (2*a_i on the diagonal).
    """
    a = np.array(sequence, dtype=float)
    n = len(a)

    conv = np.convolve(a, a)
    k_star = int(np.argmax(conv))

    grad = np.zeros_like(a)
    for i in range(n):
        j = k_star - i
        if 0 <= j < n:
            if i == j:
                grad[i] = 2.0 * a[i]
            else:
                grad[i] = a[j]

    # move opposite to the gradient
    a_new = a - eta * grad
    a_new[a_new < 0.0] = 0.0

    # keep sum away from zero (evaluate_sequence would return inf otherwise)
    if a_new.sum() < 0.01:
        a_new += (0.01 - a_new.sum()) / n

    return a_new.tolist()


def transfer_step(sequence, scale=0.1):
    """Move a random fraction of mass from index i to index j."""
    n = len(sequence)
    if n < 2:
        return sequence[:]

    i, j = np.random.choice(n, size=2, replace=False)
    amount = sequence[i] * scale * np.random.rand()
    amount = min(amount, sequence[i])
    new_seq = sequence[:]
    new_seq[i] -= amount
    new_seq[j] += amount
    return new_seq


def zero_step(sequence, fraction=0.05):
    """
    Zero a random entry and redistribute its mass uniformly over a random
    subset (size ≈ fraction * n) of the remaining positions.
    """
    n = len(sequence)
    if n == 0:
        return sequence[:]

    i = np.random.randint(n)
    mass = sequence[i]
    if mass == 0.0:
        return sequence[:]

    new_seq = sequence[:]
    new_seq[i] = 0.0

    m = max(1, int(fraction * n))
    idxs = np.random.choice(n, size=m, replace=False)
    add = mass / m
    for idx in idxs:
        new_seq[idx] += add
    return new_seq


def noise_step(sequence, sigma=0.02):
    """Add small Gaussian noise and clip to non‑negative."""
    a = np.array(sequence, dtype=float)
    a += np.random.randn(len(a)) * sigma
    a[a < 0.0] = 0.0
    # guard against the sum becoming too tiny
    if a.sum() < 0.01:
        a += (0.01 - a.sum()) / len(a)
    return a.tolist()
