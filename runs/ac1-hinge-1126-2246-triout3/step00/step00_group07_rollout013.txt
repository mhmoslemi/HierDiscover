def solve_convolution_lp(f_seq, rhs):
    n = len(f_seq)
    c = -np.ones(n)  # maximize sum => minimize -sum.
    # Build constraints for each k: sum_i f_i * g_{k-i} <= rhs.
    # Represent variables as g_j.
    A_ub = []
    b_ub = []
    for k in range(2*n - 1):
        row = np.zeros(n)
        # iterate over i where j = k - i in range
        i_min = max(0, k - (n - 1))
        i_max = min(k, n - 1)
        # For i in [i_min,i_max]:
        # row[j] += f_i where j = k - i
        # Equivalent: for each i, j = k-i
        # We'll fill directly:
        for i in range(i_min, i_max+1):
            j = k - i
            row[j] += f_seq[i]
        A_ub.append(row)
        b_ub.append(rhs)
    # non-negativity: -I * g <= 0
    A_ub = np.vstack([A_ub, -np.eye(n)])
    b_ub = np.concatenate([b_ub, np.zeros(n)])

    res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=(0, None), method='highs', options={'time_limit': 10})
    if res.success:
        return res.x
    else:
        return None

"""
Propose a better step‑function sequence.

The evaluation metric is
    score = 2·n·max(conv(a, a)) / (sum(a) ** 2)

which is invariant under a global scaling of the vector a.
Thus we can keep the L1‑norm of a fixed (conveniently sqrt(2·n)).
The algorithm repeatedly:

1. Normalises the current vector to have sum √(2n).
2. Solves a linear programme that finds a non‑negative direction g
   with the largest possible sum while respecting the convolution bound
   conv(f, g) ≤ M, where M is the current maximal convolution value.
   (Only the linear constraints in the variables g are needed – f is fixed.)
3. Mixes the current vector f with the normalised g using several step
   sizes t ∈ {0.02,0.05,0.1,0.2,0.4}.  The mixed vector still has the same
   L1‑norm, so the score is unchanged if the maximal convolution does not
   increase.  We keep the first mixture that improves the score.
4. If the LP fails or no improvement is found, a small random perturbation
   of a random entry is performed (and kept if it improves the score).

The LP is solved with SciPy's HiGHS backend; for safety the size of the LP
is limited to n ≤ 800 (larger vectors are treated only with random
perturbations).  The whole loop runs until the time budget is exhausted.
The best sequence found is returned.
"""

import time
import math
import numpy as np
from scipy import optimize

# --------------------------------------------------------------
# Linear programming: maximise sum(g) subject to conv(f, g) ≤ M
# --------------------------------------------------------------
def solve_convolution_lp(f_seq: np.ndarray, rhs: float):
    """
    Solve
        maximise   Σ g_j
        subject to conv(f_seq, g) ≤ rhs   (for all indices)
                   g_j ≥ 0

    Returns the optimal g as a 1‑D ndarray or None on failure.
    """
    n = len(f_seq)
    # objective: minimise -Σ g_j  (so maximising Σ g_j)
    c = -np.ones(n)

    # Build A_ub * g ≤ b_ub for the convolution constraints.
    # For each k = 0 .. 2n‑2 we have Σ_i f_i * g_{k-i} ≤ rhs.
    rows = []
    b = []
    for k in range(2 * n - 1):
        row = np.zeros(n)
        i_start = max(0, k - (n - 1))
        i_end = min(k, n - 1)
        i_vals = np.arange(i_start, i_end + 1)
        j_vals = k - i_vals
        # each j receives the contribution f_i
        row[j_vals] = f_seq[i_vals]
        rows.append(row)
        b.append(rhs)

    # non‑negativity: -g_j ≤ 0
    A_ub = np.vstack([rows, -np.eye(n)])
    b_ub = np.concatenate([b, np.zeros(n)])

    # HiGHS is fast and respects a time limit.
    try:
        res = optimize.linprog(
            c,
            A_ub=A_ub,
            b_ub=b_ub,
            bounds=(0, None),
            method="highs",
            options={"time_limit": 5.0},
        )
    except Exception:  # pragma: no cover
        return None

    if res.success:
        return res.x
    return None


# --------------------------------------------------------------
# Helper: normalise a vector to have L1‑norm = sqrt(2·len)
# --------------------------------------------------------------
def _normalise_to_sqrt2n(v: np.ndarray) -> np.ndarray:
    n = len(v)
    total = np.sum(v)
    if total <= 0:
        return v
    scale = math.sqrt(2.0 * n) / total
    return v * scale

                print(f"[iter {iteration}] random tweak improved score={best_score:.6f}")
            else:
                # occasionally accept a worsening move to escape plateaus
                if rng.random() < 0.001:
                    cur = new_vec

        # ------------------------------------------------------
        # 6) Occasionally restart with a fresh random vector (to avoid
        #    getting stuck in a deep local minimum).
        # ------------------------------------------------------
        if iteration % 50 == 0:
            n_restart = rng.integers(400, 801)
            cur = rng.random(n_restart)
            cur = np.maximum(cur, 0.0)
            cur = _normalise_to_sqrt2n(cur)  # give it the right norm

    # ----------------------------------------------------------
    # Return the best sequence we have seen (as a plain list)
    # ----------------------------------------------------------
    print(f"[done] best score={best_score:.6f} length={len(best_seq)}")
    return best_seq.tolist()
```
