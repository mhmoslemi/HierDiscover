def propose_candidate(budget_s=1000):
    import numpy as np, time, math

    deadline = time.time() + budget_s - 5

    # Choose length N maybe 2048 or try multiple lengths
    N = 2048
    # initialize shape arcsine
    i = np.arange(N)
    a = 1.0 / np.sqrt((i + 0.5) * (N - 0.5 - i))
    a = a / a.sum()

    best_seq = a.copy()
    best_score = evaluate_sequence(best_seq.tolist())

    # parameters
    eta = 0.5  # learning rate
    eps = 1e-6
    target_average = 1.0 / (2 * N - 1)

    # precompute FFT length for conv of length 2N-1
    L1 = 1 << int(np.ceil(np.log2(2 * N - 1)))
    # conv a_rev length maybe ~2N-1+N-1 = 3N-2; find L2
    L2 = 1 << int(np.ceil(np.log2(3 * N - 2)))

    while time.time() < deadline:
        # compute convolution a*a via FFT
        A = np.fft.rfft(a, n=L1)
        B = np.fft.irfft(A * A, n=L1)[:2 * N - 1]
        # compute objective: max(B)/... also compute target scores
        curr_score = evaluate_sequence(a.tolist())
        if curr_score < best_score:
            best_score = curr_score
            best_seq = a.copy()
            print(f"New best {best_score:.6f} at time {time.time()}")

        # flattening objective
        diff = B - target_average  # shape 2N-1
        # compute gradient: g = 4 * conv(diff, a_rev) slice length N
        a_rev = a[::-1]
        # compute via FFT
        D = np.fft.rfft(diff, n=L2)
        R = np.fft.rfft(a_rev, n=L2)
        conv = np.fft.irfft(D * R, n=L2)
        # conv length L2; the relevant slice for indices N-1 ... N-1+N-1?
        start = N - 1
        g = 4.0 * conv[start:start+N]  # gradient

        # gradient descent step
        a = a - eta * g

        # enforce nonnegativity
        a = np.maximum(a, 0.0)

        # renormalize sum to 1
        s = a.sum()
        if s < 1e-12:
            # reinitialize
            a = np.ones(N) / N
            s = 1.0
        a = a / s

        # adapt learning rate maybe decay
        eta *= 0.9999

    return best_seq.tolist()

    # function to compute convolution via FFT
    def conv_fft(x, y):
        n = len(x) + len(y) - 1
        nfft = 1 << int(np.ceil(np.log2(n)))
        X = np.fft.rfft(x, n=nfft)
        Y = np.fft.rfft(y, n=nfft)
        return np.fft.irfft(X * Y, n=nfft)[:n]

    best_seq = None
    best_score = float('inf')
    # try several lengths
    for N in [512, 1024, 2048, 4096]:
        if time.time() > deadline:
            break
        # initial a: random close to uniform + small perturb.
        a = np.ones(N) / N
        # add small random noise
        a += 0.01 * np.random.rand(N)
        a = np.maximum(a, 0)
        a = a / a.sum()
        target_avg = 1.0 / (2 * N - 1)
        # precompute FFT lengths
        nfft_conv = 1 << int(np.ceil(np.log2(2 * N - 1)))
        nfft_grad = 1 << int(np.ceil(np.log2(3 * N - 2)))

        eta = 0.5
        # cooling schedule
        while time.time() < deadline:
            # compute B = a convolve a
            A = np.fft.rfft(a, n=nfft_conv)
            B_full = np.fft.irfft(A * A, n=nfft_conv)[:2 * N - 1]

            # evaluate periodically
            if np.random.rand() < 0.01:  # ~1% of iter
                score = evaluate_sequence(a.tolist())
                if score < best_score:
                    best_score = score
                    best_seq = a.copy()
                    print(f"N={N}, new best {best_score:.6f} at time {time.time():.2f}")

            # compute diff to target flat
            diff = B_full - target_avg

            # compute gradient = 4 * conv(diff, a_rev)
            a_rev = a[::-1]
            D = np.fft.rfft(diff, n=nfft_grad)
            R = np.fft.rfft(a_rev, n=nfft_grad)

"""
Propose a step‑function (non‑negative list of heights) that seeks to
minimise the evaluation metric

    2·n·max(conv(a,a)) / (sum(a)**2)

where `conv` is the linear convolution.  The metric is scale‑invariant,
so we normalise the sequence to have sum 1 at every iteration.

The implementation uses a simple projected sub‑gradient method
targeting the infinity‑norm of the auto‑convolution.
At each iteration we locate the index `k*` where the convolution attains
its maximum and move the current vector `a` in the opposite direction of
the sub‑gradient of that active constraint:

    ∂(conv_k*)(a) / ∂a_i = 2·a_{k*‑i}   (out‑of‑range indices are 0)

The step is projected onto the probability simplex (non‑negative,
renormalised) and an adaptive stepsize is employed.
Occasional random jitter helps to escape plateaus.
The best sequence seen during the search is returned.

The algorithm works for any length; it starts from the existing
`height_sequence_1` (if present) or from a random Dirichlet vector.
All heavy work (convolutions and sub‑gradient calculations) is done
with FFTs, making the method fast enough to explore many thousands of
iterations within the 1000 s budget.
"""

import time
import numpy as np

def _conv_fft(x: np.ndarray, y: np.ndarray, nfft: int) -> np.ndarray:
    """Linear convolution via FFT (real inputs)."""
    X = np.fft.rfft(x, n=nfft)
    Y = np.fft.rfft(y, n=nfft)
    return np.fft.irfft(X * Y, n=nfft)[: len(x) + len(y) - 1]


def propose_candidate(seed: int = 0, budget_s: int = 1000) -> list[float]:
    """
    Search for a good height sequence within `budget_s` seconds.

    Parameters
    ----------
    seed : int
        Random seed for reproducibility.
    budget_s : int
        Time budget (seconds).  The function returns the best sequence found.

    Returns
    -------
    list[float]
        A non‑negative sequence (list of floats) that minimises
        `evaluate_sequence`.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 5  # keep a small safety margin

    # ------------------------------------------------------------------
    # 1. initialise the sequence
    # ------------------------------------------------------------------
    if "height_sequence_1" in globals():
        a = np.array(height_sequence_1, dtype=float)
        N = a.size
        # add a tiny random perturbation to avoid starting exactly at a fixed point
        a += 0.01 * np.random.randn(N)
        a = np.maximum(a, 0.0)
        if a.sum() == 0.0:
            a = np.ones(N) / N
        else:
            a = a / a.sum()
    else:
        N = 2048
        a = np.random.rand(N)
        a = a / a.sum()

    # keep the best candidate encountered
    best_seq = a.copy()
    best_score = evaluate_sequence(best_seq.tolist())
    print(f"[init] length={N}, initial score={best_score:.6f}")

    # ------------------------------------------------------------------
    # 2. pre‑compute FFT lengths (powers of two) – these stay constant
    # ------------------------------------------------------------------
    # length of the full convolution a*a
    nconv_len = 2 * N - 1
    nfft_conv = 1 << int(np.ceil(np.log2(nconv_len)))

    # length needed for gradient convolution: mask (len nconv_len) * a_rev (len N)
    ngrad_len = nconv_len + N - 1          # = 3*N - 2
    nfft_grad = 1 << int(np.ceil(np.log2(ngrad_len)))

    # ------------------------------------------------------------------
    # 3. optimisation loop (projected sub‑gradient on max(B))
    # ------------------------------------------------------------------
    eta = 0.5                 # initial stepsize
    min_eta = 1e-6
    patience = 0              # counter for consecutive failed steps

    while time.time() < deadline:
        # ----- current convolution -------------------------------------------------
        A = np.fft.rfft(a, n=nfft_conv)
        B = np.fft.irfft(A * A, n=nfft_conv)[:nconv_len]   # linear convolution a*a

        Bmax = B.max()
        k_star = int(np.argmax(B))   # index where the max is attained

        # ----- occasionally evaluate the true objective ---------------------------
        if np.random.rand() < 0.02:   # about 2 % of the iterations
            cur_score = evaluate_sequence(a.tolist())
            if cur_score < best_score:
                best_score = cur_score
                best_seq = a.copy()
                print(f"[update] new best score={best_score:.6f} (t={time.time():.2f})")

        # ----- sub‑gradient of the active constraint (single active index) ---------
        mask = np.zeros_like(B)
        mask[k_star] = 1.0
        a_rev = a[::-1]

        M = np.fft.rfft(mask, n=nfft_grad)
        R = np.fft.rfft(a_rev, n=nfft_grad)
        grad_full = np.fft.irfft(M * R, n=nfft_grad)   # length = ngrad_len
        # relevant slice corresponds to indices (N‑1 … N‑1+N‑1)
        start = N - 1
        grad = 2.0 * grad_full[start:start + N]        # ∂B_k*/∂a

        # ----- tentative update ---------------------------------------------------
        a_candidate = a - eta * grad
        a_candidate = np.maximum(a_candidate, 0.0)
        s = a_candidate.sum()
        if s == 0.0:                     # degenerate case – restart from uniform
            a_candidate = np.ones(N) / N
        else:
            a_candidate /= s

        # ----- evaluate the effect on the max convolution -------------------------
        A_c = np.fft.rfft(a_candidate, n=nfft_conv)
        B_c = np.fft.irfft(A_c * A_c, n=nfft_conv)[:nconv_len]
        Bmax_c = B_c.max()

        # ----- accept / reject step ------------------------------------------------
        if Bmax_c < Bmax - 1e-14:        # genuine descent
            a = a_candidate
            eta = min(eta * 1.02, 1.0)   # slowly increase stepsize
            patience = 0
        else:
            eta = max(eta * 0.5, min_eta)
            patience += 1
            if patience > 30:
                # inject a small random jitter to try to leave a plateau
                jitter = 0.001 * np.random.randn(N)
                a = np.maximum(a + jitter, 0.0)
                a /= a.sum()
                patience = 0
                eta = max(eta, 0.01)

    # ------------------------------------------------------------------
    # 4. Return the best sequence discovered
    # ------------------------------------------------------------------
    print(f"[final] best score found = {best_score:.6f}")
    return best_seq.tolist()
