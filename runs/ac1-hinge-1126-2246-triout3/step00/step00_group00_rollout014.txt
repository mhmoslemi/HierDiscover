def propose_candidate(seed=42, budget_s=1000, **kwargs):
    import numpy as np, time
    np.random.seed(seed)
    deadline = time.time() + budget_s - 5

    best_seq = None
    best_score = float('inf')
    # try multiple seeds for start shapes
    for attempt in range(5):
        # choose random length between 300 and 1000
        n = np.random.randint(300, 800) # moderate
        # choose geometry parameter
        c = 0.5 + np.random.rand() * 2.0   # from 0.5 to 2.5
        epsilon = c / n
        r = max(0.0, 1.0 - epsilon)
        a = np.array([r**i for i in range(n)], dtype=float)
        # add small random noise
        a += 0.01 * np.random.rand(n)
        # normalize sum to 1
        a /= a.sum()
        # gradient descent
        beta = 2.0
        lr = 0.1
        iter = 0
        while time.time() < deadline and iter < 500:
            grad, smooth_obj = compute_gradient(a, beta)
            a -= lr * grad
            # clip negatives
            a = np.maximum(a, 0)
            # renormalize sum to 1
            s = a.sum()
            if s == 0:
                a = np.ones_like(a) / n
                s = 1.0
            else:
                a /= s
            # occasionally evaluate real score
            if iter % 20 == 0:
                cur_score = evaluate_sequence(a.tolist())
                if cur_score < best_score:
                    best_score = cur_score
                    best_seq = a.copy()
                # adapt beta and lr
                if cur_score < best_score * 0.9 and beta < 10:
                    beta *= 1.2
                if beta > 10:
                    beta = 10
                if lr > 1e-4:
                    lr *= 0.99
            iter += 1

    # After gradient descent, try LP improvement on best_seq
    if best_seq is not None:
        seq = best_seq.tolist()
        curr_seq = seq.copy()
        best_score = evaluate_sequence(seq)
        # LP loop
        while time.time() < deadline:
            new_seq = get_good_direction_to_move_into(curr_seq)
            if new_seq is None:
                # random perturb
                idx = np.random.randint(len(curr_seq))
                curr_seq[idx] = max(0, curr_seq[idx] + np.random.randn() * 0.01)

                curr_seq = new_seq
            cur_score = evaluate_sequence(curr_seq)
            if cur_score < best_score:
                best_score = cur_score
                best_seq = np.array(curr_seq)
    # Return best sequence as list
    if best_seq is None:
        # fallback: uniform sequence length 100
        n = 100
        best_seq = np.ones(n) / n
    return best_seq.tolist()
