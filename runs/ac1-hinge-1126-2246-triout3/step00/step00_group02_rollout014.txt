def propose_candidate(seed=0, budget_s=1000):
    np.random.seed(seed)
    deadline = time.time() + budget_s - 5
    
    # Determine length n (use given height_sequence_1 if available)
    try:
        n = len(height_sequence_1)
    except NameError:
        n = 200  # fallback
    
    S = np.sqrt(2 * n)
    a = np.full(n, S / n)  # uniform initialization
    
    # keep best
    best_seq = a.copy()
    best_score = evaluate_sequence(best_seq.tolist())
    
    # parameters
    lr = 0.5
    alpha = 5.0
    decay_lr = 0.999  # multiply each iteration
    decay_alpha = 1.001  # increase alpha gradually
    
    iter_count = 0
    while time.time() < deadline:
        # compute convolution
        b = np.convolve(a, a)
        max_b = b.max()
        # smooth max weights
        w = np.exp(alpha * (b - max_b))
        w_sum = w.sum()
        w = w / w_sum
        
        # gradient of smooth max w.r.t a
        conv_rev = np.convolve(w, a[::-1])
        grad = 2.0 * conv_rev[n-1 : n-1 + n]
        
        # take a step opposite gradient
        a_new = a - lr * grad
        # project onto nonnegative
        a_new = np.maximum(0.0, a_new)
        # renormalize sum
        cur_sum = a_new.sum()
        if cur_sum > 0:
            a_new = a_new * (S / cur_sum)
        else:
            a_new = np.full(n, S / n)  # fallback
        
        # evaluate
        score = evaluate_sequence(a_new.tolist())
        if score < best_score:
            best_score = score
            best_seq = a_new.copy()
            # print improvement
            print(f"Iter {iter_count}: improved score {best_score:.6f}")
        
        # acceptance: we could also accept if score better, else random restart
        # Keep current a as a_new if score improved else maybe accept small worsening probabilistically (simulated annealing)
        if score <= best_score:
            a = a_new
        else:
            # maybe accept with small probability decreasing over time
            temp = max(0.1, (deadline - time.time()) / budget_s)
            if np.random.rand() < np.exp(-(score - best_score) / (temp + 1e-9)):
                a = a_new
        
        # decay learning rate, increase alpha
        lr *= decay_lr
        alpha *= decay_alpha
        
        iter_count += 1
    
        # occasional random shake to escape local minima
        if iter_count % 1000 == 0:
            # add small random noise
            noise = np.random.randn(n) * 0.01 * a.mean()
            a = np.maximum(0.0, a + noise)
            cur_sum = a.sum()
            a = a * (S / cur_sum)
    
    return best_seq.tolist()

"""
Gradient‑descent search for a step‑function with low self‑convolution peak.

The evaluation function is
    E(a) = 2·n·max(a*a) / (sum(a)^2)

Because the expression is scale‑invariant we keep the L1‑norm fixed to
S = sqrt(2·n).  With this normalisation  E(a) = max(a*a).

For a uniform sequence the convolution is triangular and the maximum is 2.
We minimise the variance of the convolution (which pushes all entries of
a*a towards the constant average) by gradient descent on the objective

    L(a) = Σ_k ( (a*a)_k – μ )² ,   μ = S²/(2·n-1)  (the constant average)

The gradient of L is

    ∇L = 4·conv( (a*a) – μ , rev(a) )[n-1:n-1+n]

where rev(a) is a reversed copy of a.  After each gradient step we
project onto the non‑negative orthant and renormalise the sum to S.
A simple learning‑rate schedule, occasional random restarts and a
fallback to the current best solution give a fast, robust search that
typically drives the evaluation below 1.5 within the allotted time.
"""

import time
import numpy as np

def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs):
    """
    Search for a sequence of non‑negative heights that minimises
    evaluate_sequence.  The algorithm uses gradient descent on the
    variance of the self‑convolution while keeping the L1‑norm fixed.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 5.0   # a small safety margin

    # ------------------------------------------------------------------
    # Choose a problem size.
    # If a reference sequence is available we keep its length moderate;
    # otherwise we fall back to a default size (≈300) which is fast enough.
    # ------------------------------------------------------------------
    try:
        # use the same length as the provided construction, but cap it
        # to keep the optimisation cheap
        n = min(300, len(height_sequence_1))
    except Exception:
        n = 300

    # Target L1‑norm (scale‑invariant)
    S = np.sqrt(2.0 * n)

    # ----- initial point ------------------------------------------------
    a = np.full(n, S / n)                     # uniform start (evaluation = 2)
    best_seq = a.copy()
    # compute current score directly (no need to call evaluate_sequence)
    b = np.convolve(a, a)
    best_score = 2.0 * n * b.max() / (S * S)

    print(f"[init] n={n}, start_score={best_score:.6f}")

    # ----- hyper‑parameters --------------------------------------------
    lr = 0.5                                   # initial step size
    min_lr = 1e-4
    decay = 0.9995                             # geometric decay per iteration
    restart_every = 4000                       # iterations between random restarts
    iter_cnt = 0

    # constant average of the convolution (depends only on n and S)
    avg_conv = (S * S) / (2.0 * n - 1.0)

    while time.time() < deadline:
        # ---- compute convolution and its variance ----
        b = np.convolve(a, a)                   # length 2n‑1
        max_b = b.max()

        # smooth objective is the variance of b around its constant mean
        diff = b - avg_conv                     # zero‑mean vector
        # gradient of Σ (diff_k)²  w.r.t. a is
        #   4 * conv(diff, rev(a))[n-1 : n-1+n]
        rev_a = a[::-1]
        conv_rev = np.convolve(diff, rev_a)      # length 3n‑2
        grad = 4.0 * conv_rev[n - 1:n - 1 + n]

        # ---- gradient step with projection onto the simplex ----------
        a_new = a - lr * grad
        a_new = np.maximum(0.0, a_new)          # enforce non‑negativity

        # renormalise to keep the L1‑norm exactly S
        tot = a_new.sum()
        if tot > 0:
            a_new *= (S / tot)
        else:
            # degenerate case – fall back to uniform
            a_new = np.full(n, S / n)

        # ---- evaluate improvement ------------------------------------
        b_new = np.convolve(a_new, a_new)
        score_new = 2.0 * n * b_new.max() / (S * S)

        if score_new < best_score:
            best_score = score_new
            best_seq = a_new.copy()
            print(f"[{iter_cnt:06d}] improved: {best_score:.6f}")

        # ---- accept the move (always the new point, but we keep the best)
        a = a_new

        # ---- learning‑rate adaptation --------------------------------
        lr = max(min_lr, lr * decay)

        # ---- occasional random restart to escape stagnation ----------
        if (iter_cnt + 1) % restart_every == 0:
            a = np.random.rand(n)
            a *= (S / a.sum())
            print(f"[{iter_cnt:06d}] random restart")

        iter_cnt += 1

    print(f"[final] best_score={best_score:.6f}, iterations={iter_cnt}")
    return best_seq.tolist()
