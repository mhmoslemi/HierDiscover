def project_to_simplex(v):
    # algorithm for Euclidean projection onto simplex.
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1

def simplex_projection(v):
    n = len(v)
    sorted_v = np.sort(v)[::-1]
    cssv = np.cumsum(sorted_v) - 1
    rho = np.nonzero(sorted_v - cssv / np.arange(1, n+1) > 0)[0]
    if len(rho) == 0:
        # all nonpositive, return uniform
        return np.full(n, 1.0/n)
    rho = rho[-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0)
    return w

def simplex_projection(v):
    # Ensures sum(v) = 1, v >=0, minimal Euclidean distance.
    n = v.shape[0]
    # sort descending
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    rho = np.where(u - cssv / np.arange(1, n+1) > 0)[0]
    if rho.size == 0:
        # all values are zero or negative => return uniform
        return np.full(n, 1.0 / n)
    rho = rho[-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0)
    # due to numerical issues, maybe adjust sum to 1
    w_sum = w.sum()
    if w_sum > 0:
        w = w / w_sum
    else:
        w = np.full(n, 1.0 / n)
    return w

def convolve_fft(x, y):
    n = len(x) + len(y) - 1

"""
Propose a candidate step‑function height sequence that minimises

    2 * n * max(conv(a, a)) / (sum(a)**2)

with non‑negative entries.  The objective is equivalent (after fixing
∑a_i = 1) to minimising 2·n·max_k (a*a)[k], i.e. flattening the
self‑convolution of the probability vector a.
The algorithm performs many short optimisation runs:

* each run chooses a random length n (≈ 300–1200) and a random Dirichlet
  start vector a (∑a = 1);
* it then repeatedly
  – computes the full convolution a*a,
  – builds a “soft‑max’’ weight that concentrates on the largest
    convolution entries,
  – computes the gradient of the soft‑max w.r.t. a (a cross‑correlation
    of a with the weights),
  – takes a step in the opposite direction and projects back onto the
    probability simplex;
* the learning‑rate decays and the soft‑max temperature τ is increased
  during the run, which gradually focuses the optimisation on the true
  peak of the convolution;
* every few iterations the true evaluation is measured (using the
  provided evaluate_sequence) and the best sequence seen overall is
  retained.

All arithmetic is done with NumPy and convolutions are performed with
FFT‑based real convolutions for speed.  The routine respects the 1000 s
budget and returns the best sequence discovered.
"""

import time
import numpy as np

# ------------------------------------------------------------
# Helper: projection of a vector onto the probability simplex
# ------------------------------------------------------------
def simplex_projection(v: np.ndarray) -> np.ndarray:
    """Project v onto the simplex {x >= 0, sum(x) = 1} (Euclidean projection)."""
    n = v.shape[0]
    # sort v in descending order
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho = np.where(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:  # all non‑positive, return uniform distribution
        return np.full(n, 1.0 / n, dtype=v.dtype)
    rho = rho[-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    # numeric cleanup: renormalise to guarantee exact sum 1
    s = w.sum()
    if s > 0:
        w /= s
    else:
        w = np.full(n, 1.0 / n, dtype=v.dtype)
    return w


# ------------------------------------------------------------
# Helper: real FFT based convolution (full mode)
# ------------------------------------------------------------
def convolve_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Return the full convolution of real vectors x and y using FFT."""
    n = len(x) + len(y) - 1
    nfft = 1 << (n - 1).bit_length()          # next power of two
    fx = np.fft.rfft(x, n=nfft)
    fy = np.fft.rfft(y, n=nfft)
    conv = np.fft.irfft(fx * fy, nfft)[:n]
    # numerical noise may produce tiny negatives – clip to zero
    conv[conv < 0] = 0.0
    return conv


# ------------------------------------------------------------
# Core optimisation for a single length n
# ------------------------------------------------------------
def minimise_max_convolution(n: int,
                            time_deadline: float,
                            base_lr: float = 0.8,
                            lr_decay: float = 0.9993,
                            initial_tau: float = 5.0,
                            tau_factor: float = 1.8,
                            steps_per_phase: int = 250) -> (np.ndarray, float):
    """
    Perform a short optimisation for a given length n.
    Returns the best vector (sum = 1) found and its evaluation score.
    """
    # start from a Dirichlet (uniform) random point on the simplex
    a = np.random.dirichlet(np.ones(n)).astype(np.float64)

    best_local_a = a.copy()
    # evaluate the starting point
    conv = convolve_fft(a, a)
    best_local_score = 2.0 * n * np.max(conv)

    # Phase schedule – gradually increase the soft‑max temperature τ
    tau = initial_tau
    lr = base_lr

    # total number of phases limited by the remaining time
    while time.time() < time_deadline:
        for _ in range(steps_per_phase):
            if time.time() >= time_deadline:
                break

            # full self‑convolution
            conv = convolve_fft(a, a)                 # length 2n‑1
            max_conv = np.max(conv)

            # soft‑max weights focusing on the largest entries
            # (shifted for numerical stability)
            shifted = conv - max_conv
            weights = np.exp(tau * shifted)           # non‑negative, sum ≈ 1

            # gradient of the smooth “max’’ (p = 1 in the derivation)
            # g_i = 2 * Σ_k weights_k * a_{k-i}
            conv_w_rev = convolve_fft(a, weights[::-1])   # length 3n‑2
            g = 2.0 * conv_w_rev[2 * n - 2 - np.arange(n)]

            # gradient step and simplex projection
            a = a - lr * g
            a = simplex_projection(a)

            # learning‑rate decay
            lr *= lr_decay

        # evaluate after each phase
        conv = convolve_fft(a, a)
        cur_score = 2.0 * n * np.max(conv)
        if cur_score < best_local_score:
            best_local_score = cur_score
            best_local_a = a.copy()
        # increase τ to sharpen the focus on the true maximum
        tau *= tau_factor

    return best_local_a, best_local_score


# ------------------------------------------------------------
# Main entry point required by the judge
# ------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a step‑function height sequence with a low evaluation value.
    The function respects the 1000 second budget and returns the best
    sequence discovered.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5   # leave a little margin

    best_score = np.inf
    best_sequence = None
    iteration = 0

    # ------------------------------------------------------------------
    # Main search loop – many independent short optimisation runs
    # ------------------------------------------------------------------
    while time.time() < deadline:
        # pick a random length (moderate size keeps each run cheap)
        n = np.random.randint(300, 1200)

        # run the optimiser for this length; give it a fraction of the budget
        # (the inner function stops automatically when the global deadline is hit)
        a_opt, score_opt = minimise_max_convolution(
            n=n,
            time_deadline=deadline,
            base_lr=0.8,
            lr_decay=0.9993,
            initial_tau=5.0,
            tau_factor=1.8,
            steps_per_phase=250,
        )

        # Use the official evaluator to obtain the exact score (handles clipping etc.)
        try:
            eval_score = evaluate_sequence(a_opt.tolist())
        except Exception:
            # Should never happen – fall back to our own cheap estimate
            eval_score = 2.0 * n * np.max(convolve_fft(a_opt, a_opt))

        if eval_score < best_score:
            best_score = eval_score
            best_sequence = a_opt.copy()
            print(
                f"[{iteration:04d}] New best – n={n}, score={best_score:.6f}"
            )
        iteration += 1

    if best_sequence is None:
        # fallback – a simple uniform vector (should never be needed)
        best_sequence = np.full(10, 1.0 / 10.0, dtype=np.float64)

    print(f"Search finished. Best score = {best_score:.6f}")
    return best_sequence.tolist()
