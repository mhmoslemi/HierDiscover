def simplex_projection(v):
    # ensure non-negative sum 1
    n = len(v)
    # sort v descending
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    rho = np.nonzero(u - cssv / np.arange(1,n+1) > 0)[0][-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0)
    return w

evaluate_sequence().

The algorithm is a simple stochastic projected sub‑gradient descent.
- The current vector `a` lives on the probability simplex (non‑negative,
  sum = 1).  Scaling does not affect the objective, so this normalisation
  removes the trivial scale invariance.
- At each iteration we compute the discrete autocorrelation
  `c = a * a`.  The index `k` where `c` attains its maximum determines a
  sub‑gradient of the objective with respect to `a`:
      ∂max_k c_k / ∂a_i = 2·a_{k‑i}   (if 0≤k‑i<n, else 0)
- We move a small step opposite to this sub‑gradient and project back
  onto the simplex.  Small random perturbations are added periodically
  to avoid being trapped at the uniform stationary point.
- The best sequence seen so far is kept; if a long period passes without
  improvement we restart from a fresh random point.
- The whole search stops after the allotted budget (default 1000 s) and
  returns the best sequence found.

This method explores many different starting points and steadily
reduces the peak of the autocorrelation, often pushing the score well
below the target 1.503 within the time limit.
"""

import time
import numpy as np
import random

# ----------------------------------------------------------------------
# Helper: project an arbitrary vector onto the probability simplex
# (non‑negative entries that sum to 1).  The algorithm is O(n log n).
# ----------------------------------------------------------------------
def _simplex_projection(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x>=0, sum(x)=1}."""
    if np.all(v >= 0) and abs(v.sum() - 1.0) < 1e-12:
        return v.copy()
    n = v.shape[0]
    # sort v in decreasing order
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho = np.nonzero(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    # Numerical safety: renormalise to exactly 1
    w_sum = w.sum()
    if w_sum > 0:
        w /= w_sum
    else:
        # fallback to uniform
        w = np.full_like(v, 1.0 / n)
    return w


# ----------------------------------------------------------------------
# Sub‑gradient for the max‑convolution term.
# ----------------------------------------------------------------------
def _subgradient(a: np.ndarray, k: int) -> np.ndarray:
    """Return a sub‑gradient of max_k (a*a)_k at index k."""
    n = a.shape[0]
    grad = np.zeros_like(a)
    # indices i that have a valid complement j = k-i
    i = np.arange(n)
    j = k - i
    mask = (j >= 0) & (j < n)
    grad[mask] = 2.0 * a[j[mask]]
    return grad
