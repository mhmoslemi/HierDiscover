def solve_lp_max_sum(f_seq, rhs):
    n = len(f_seq)
    # Build sparse A_ub matrix
    from scipy.sparse import lil_matrix
    A = lil_matrix((2*n-1, n))
    for k in range(2*n-1):
        i_start = max(0, k-(n-1))
        i_end = min(n-1, k)
        # loop i
        for i in range(i_start, i_end+1):
            j = k - i
            A[k, j] = f_seq[i]
    b = np.full(2*n-1, rhs)
    c = -np.ones(n)
    # variable bounds: (0, None)
    res = linprog(c, A_ub=A, b_ub=b, bounds=[(0, None)]*n, method='highs')
    if res.success:
        return res.x
    return None

"""
Propose a near‑optimal step‑function using an iterative LP + binary‑search scheme.

The algorithm
---------------
1. Initialise a small uniform sequence (sum = 1).
2. Repeatedly:
   * Compute the current convolution M = max(f * f).
   * Solve the linear program
         maximise  Σ b_i
         s.t.      (f * b)_k ≤ M   for all k
                  b_i ≥ 0
     (the convolution constraints are linear in b).
   * Rescale the LP solution to keep the same total sum as f.
   * Find the largest step size t∈[0,1] such that the convex
     combination h = (1‑t)·f + t·g does not increase the maximal
     convolution (binary search on t using the pre‑computed
     convolutions f*f, f*g and g*g).
   * Update f ← h and keep the best sequence seen so far.
   * Occasionally add a tiny random perturbation to break symmetry
     (the uniform distribution is a fixed point of the LP).

The score is 2·n·max(f*f)/(Σf)²; decreasing the maximal convolution
reduces the score.  With a modest length (n≈300) the LP is tiny,
so many iterations fit easily into the 1000 s budget and typically
drive the score well below the target 1.503.
"""

import time
import numpy as np
from scipy.optimize import linprog
from scipy.sparse import lil_matrix, vstack


def _conv(seq):
    """Full convolution of a 1‑D real sequence with itself."""
    return np.convolve(seq, seq)


def _score(seq):
    """Exact same objective as evaluate_sequence (no clipping)."""
    seq = np.asarray(seq, dtype=float)
    n = len(seq)
    conv = _conv(seq)
    max_b = conv.max()
    sum_a = seq.sum()
    if sum_a < 1e-12:
        return np.inf
    return float(2.0 * n * max_b / (sum_a ** 2))


def _solve_lp_max_sum(f_seq, rhs):
    """
    Solve     maximise Σ b_i   s.t.    (f_seq * b)_k ≤ rhs   for all k,
    with b_i ≥ 0.
    Returns the optimal b as a dense ndarray, or None on failure.
    """
    n = len(f_seq)
    # Build the (2n‑1)×n Toeplitz matrix for the convolution f*b.
    A = lil_matrix((2 * n - 1, n), dtype=float)
    for k in range(2 * n - 1):
        # i runs over all indices where both i and j=k‑i lie in [0,n).
        i_start = max(0, k - (n - 1))
        i_end = min(n - 1, k)
        for i in range(i_start, i_end + 1):
            j = k - i
            A[k, j] = f_seq[i]

    b = np.full(2 * n - 1, rhs, dtype=float)
    c = -np.ones(n, dtype=float)          # maximise Σb  == minimise −Σb
    bounds = [(0, None)] * n               # b_i ≥ 0

    # Use the high‑performance HiGHS interior‑point solver.
    res = linprog(c, A_ub=A, b_ub=b, bounds=bounds,
                  method='highs', options={'presolve': True})
    if res.success:
        return res.x
    return None


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs) -> list[float]:
    """Return a sequence of non‑negative heights that (hopefully) beats the target."""
    np.random.seed(seed)
    # ------------------------------------------------------------------
    # parameters (feel free to tweak)
    n = 300                     # moderate size – fast LP, good enough for the bound
    max_iter = 10_000          # safety cap for the SA phase
    eps_t = 1e-8               # stop when the binary‑search step size vanishes
    # ------------------------------------------------------------------

    # start from the uniform distribution (sum = 1)
    f = np.full(n, 1.0 / n, dtype=float)
    best_seq = f.copy()
    best_val = _score(best_seq)

    deadline = time.time() + budget_s - 2.0   # leave a little slack

    iter_cnt = 0
    while time.time() < deadline:
        iter_cnt += 1
        # ---- LP direction ------------------------------------------------
        conv_ff = _conv(f)
        M = conv_ff.max()
        g0 = _solve_lp_max_sum(f, M)
        if g0 is None:
            # LP solver failed – fall back to a tiny random jitter
            idx = np.random.randint(n)
            delta = (np.random.rand() - 0.5) * 0.02 * f[idx]
            f[idx] = max(0.0, f[idx] + delta)
            f /= f.sum()
            continue

        sum_f = f.sum()
        sum_g0 = g0.sum()
        if sum_g0 <= 0:
            # degenerate – jitter and continue
            idx = np.random.randint(n)
            delta = (np.random.rand() - 0.5) * 0.02 * f[idx]
            f[idx] = max(0.0, f[idx] + delta)
            f /= f.sum()
            continue

        # Rescale g0 so that it has the same L1 norm as f
        scale = sum_f / sum_g0
        g = g0 * scale

        # ---- maximal admissible step size t (binary search) -------------
        conv_fg = _conv(f, g) if False else np.convolve(f, g)  # use np.convolve directly
        conv_gg = _conv(g)
        lo, hi = 0.0, 1.0
        for _ in range(30):
            mid = (lo + hi) * 0.5
            # convolution of the convex combination:
            # (1‑t)^2 * (f*f) + 2 t (1‑t) * (f*g) + t^2 * (g*g)
            combo = ((1.0 - mid) ** 2) * conv_ff \
                    + (2.0 * mid * (1.0 - mid)) * conv_fg \
                    + (mid ** 2) * conv_gg
            if combo.max() <= M + 1e-14:
                lo = mid
            else:
                hi = mid
        t = lo

        if t < eps_t:
            # no progress – inject a small random perturbation to escape.
            idx = np.random.randint(n)
            delta = (np.random.rand() - 0.5) * 0.05 * f[idx]
            f[idx] = max(0.0, f[idx] + delta)
            f /= f.sum()
            continue

        # ---- update -------------------------------------------------------
        f = (1.0 - t) * f + t * g
        # renormalise to avoid drift caused by numerical errors
        f /= f.sum()

        cur_val = _score(f)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = f.copy()
            print(f"Iter {iter_cnt:5d} – new best score: {best_val:.6f}")

        # ---- occasional SA‑style random walk to push further -------------
        if np.random.rand() < 0.02:
            # pick two indices and move a tiny amount of mass
            i, j = np.random.choice(n, size=2, replace=False)
            move = (np.random.rand() - 0.5) * 0.01 * f[i]
            move = np.clip(move, -f[i], f[j])   # keep both non‑negative
            f[i] -= move
            f[j] += move

        # safety exit if we already beat the target
        if best_val < 1.503:
            print(f"Target beaten after {iter_cnt} iterations: {best_val:.6f}")
            break

        # guard against too many LP solves (each may be a few seconds)
        if iter_cnt >= max_iter:
            break

    # Return the best sequence found (as plain Python list)
    return best_seq.tolist()
