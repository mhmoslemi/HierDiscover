def project_simplex(v, s):
   # v is np.array.
   # s >0
   n = len(v)
   u = np.sort(v)[::-1]
   cssv = np.cumsum(u) - s
   ind = np.arange(1,n+1)
   cond = u - cssv / ind > 0
   rho = ind[cond][-1]  # last True
   theta = cssv[rho-1] / rho
   w = np.maximum(v - theta, 0)
   return w

import numpy as np
import time
import math

def project_simplex(v, s):
    """Project vector v onto the simplex {x >= 0, sum x = s}."""
    # Known algorithm:
    if s < 0:
        raise ValueError("Sum must be non-negative")
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho_arr = np.where(u - cssv / np.arange(1, n+1) > 0)[0]
    if len(rho_arr) == 0:
        # All entries will be zero
        theta = cssv[-1] / n
    else:
        rho = rho_arr[-1] + 1  # +1 because indices start 0
        theta = cssv[rho-1] / rho
    w = np.maximum(v - theta, 0)
    return w

import numpy as np
import time
import math

def _project_simplex(v, s):
    """Project vector v onto the simplex {x>=0, sum x = s}."""
    if s < 0:
        raise ValueError("Simplex sum must be non‑negative")
    n = v.shape[0]
    # sort v in descending order
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    ind = np.arange(1, n + 1)
    # Find appropriate rho
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = cssv[-1] / n
    else:
        rho = np.nonzero(cond)[0][-1] + 1   # +1 because rho is 1‑based
        theta = cssv[rho - 1] / rho
    w = np.maximum(v - theta, 0)
    return w

    """
    Gradient‑descent / random‑perturbation search for a non‑negative
    step‑function that minimises `evaluate_sequence`.  The algorithm:
      * starts from the supplied `height_sequence_1` if available, else from a
        random feasible point;
      * keeps the L1‑norm fixed to √(2·n) (the problem is scale invariant);
      * repeatedly reduces the largest convolution value by moving mass away
        from the entries that participate in the current maximum;
      * occasionally performs a small random mass‑transfer to escape local
        minima and occasionally restarts from a fresh random point;
      * projects every tentative vector onto the simplex {x≥0, ∑x = √(2·n)}.
    The best sequence seen during the allotted time is returned.
    """
    np.random.seed(seed)

    #   Initial sequence (fallback to random if the global is missing)
    # ------------------------------------------------------------------ #

    n = init_seq.shape[0]
    target_sum = math.sqrt(2.0 * n)

    # Normalise to the simplex (L1 = sqrt(2n))

import numpy as np
import time
import math

# --------------------------------------------------------------
# Helper: projection of a vector onto the simplex
#   { x_i >= 0 , sum_i x_i = s }.
# --------------------------------------------------------------
def _project_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Euclidean projection onto the probability simplex scaled by ``s``."""
    if s < 0:
        raise ValueError("Simplex sum must be non‑negative")
    n = v.shape[0]
    # sort v decreasingly
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    ind = np.arange(1, n + 1)
    # find the largest index fulfilling the positivity condition
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all entries will be zero – unlikely because we keep a positive sum
        theta = cssv[-1] / n
    else:
        rho = np.nonzero(cond)[0][-1]      # zero‑based index
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w

def propose_candidate(seed: int = 42, budget_s: float = 1000, **kwargs):
    """
    Gradient‑descent / random‑perturbation search for a non‑negative
    sequence that minimises ``evaluate_sequence``.

    The algorithm works as follows
    1. start from the supplied global ``height_sequence_1`` (if present) –
       otherwise from a random feasible vector;
    2. normalise the L1‑norm to √(2·n) (the objective is scale‑invariant);
    3. repeatedly reduce the current **largest** convolution value
       by moving mass away from the entries that contribute to it;
    4. occasionally perform a small random mass‑transfer to escape local
       minima and, after a long stagnation, restart from a fresh random
       point;
    5. after each tentative move project back onto the simplex
       (non‑negative, correct L1‑norm).

    The best sequence seen before the time budget expires is returned.
    """
    np.random.seed(seed)

    # ------------------------------------------------------------------
    # 1) initialise a feasible vector
    # ------------------------------------------------------------------
    try:
        # ``height_sequence_1`` is provided by the surrounding framework
        init = np.array(height_sequence_1, dtype=np.float64)
    except NameError:
        # fall back to a random start if the global is missing
        init = np.abs(np.random.randn(1000)).astype(np.float64)

    n = init.shape[0]
    target_sum = math.sqrt(2.0 * n)          # the canonical L1‑norm

    # project the initial vector onto the simplex (enforces sum = target_sum)
    a = _project_simplex(init, target_sum)

    # store the best solution found so far
    b = np.convolve(a, a)
    best_val = 2.0 * n * np.max(b) / (np.sum(a) ** 2)   # mathematically = np.max(b)
    best_seq = a.copy()
    print(f"[init] n={n}, objective={best_val:.6f}")

    # ------------------------------------------------------------------
    # 2) hyper‑parameters (they evolve with time)
    # ------------------------------------------------------------------
    start_time = time.time()
    deadline = start_time + budget_s - 0.2          # tiny safety margin

    lr_start = 0.8      # learning rate at the beginning
    lr_end   = 0.001    # learning rate at the end

    rand_start = 0.15   # probability of a pure random move initially
    rand_end   = 0.005  # probability at the end

    noise_start = 0.05  # scale of random mass transfer (fraction of source)
    noise_end   = 0.001

    patience = 500      # iterations without improvement before a restart
    last_improve = 0

    it = 0
    while time.time() < deadline:
        it += 1
        # ------------------------------------------------------------------
        # fraction of the total budget already consumed (0 … 1)
        # ------------------------------------------------------------------
        elapsed = time.time() - start_time
        frac = min(1.0, elapsed / budget_s)

        # linearly interpolate the schedules
        lr = lr_start * (1.0 - frac) + lr_end * frac
        rand_prob = rand_start * (1.0 - frac) + rand_end * frac
        noise_scale = noise_start * (1.0 - frac) + noise_end * frac

        # ------------------------------------------------------------------
        # 3) propose a new candidate (gradient step or random transfer)
        # ------------------------------------------------------------------
        if np.random.rand() < rand_prob:
            # -------- random mass transfer ---------------------------------
            i, j = np.random.choice(n, 2, replace=False)
            # move a small fraction of the mass from i to j
            delta = noise_scale * a[i] * np.random.rand()
            cand = a.copy()
            cand[i] -= delta
            cand[j] += delta
        else:
            # -------- deterministic descent on the current maximal conv ---
            conv = np.convolve(a, a)                 # full auto‑convolution
            k_max = int(np.argmax(conv))             # index of the current peak

            # gradient of the maximal entry:
            #   ∂b_kmax/∂a_i = 2 * a_{k_max - i}  (if the partner index is inside)
            grad = np.zeros_like(a)
            i_low = max(0, k_max - (n - 1))
            i_high = min(k_max, n - 1)
            # only the range [i_low, i_high] participates in the sum
            idx = np.arange(i_low, i_high + 1)
            partner = k_max - idx
            grad[idx] = 2.0 * a[partner]
            cand = a - lr * grad

        # ------------------------------------------------------------------
        # 4) enforce feasibility (non‑negative & correct L1‑norm)
        # ------------------------------------------------------------------
        cand = np.clip(cand, 0.0, None)
        cand = _project_simplex(cand, target_sum)

        # ------------------------------------------------------------------
        # 5) cheap objective evaluation (max of the auto‑convolution)
        # ------------------------------------------------------------------
        conv = np.convolve(cand, cand)
        cur_max = np.max(conv)          # because the sum is already fixed
        cur_val = cur_max                # equals the full evaluate_sequence score

        if cur_val < best_val:
            best_val = cur_val
            best_seq = cand.copy()
            last_improve = it
            print(f"[iter {it:5d}] new best max={best_val:.6f}")

        # ------------------------------------------------------------------

            a = _project_simplex(a, target_sum)
            last_improve = it
            print(f"[iter {it:5d}] restart after {patience} idle iterations")
            continue

        a = cand

    # Final validation with the official evaluator
    # ----------------------------------------------------------------------
    final_seq = best_seq.tolist()
    try:

        print(f"[done] best official score = {final_score:.6f}")
    except Exception as exc:
        print(f"[error] evaluation failed: {exc}")
