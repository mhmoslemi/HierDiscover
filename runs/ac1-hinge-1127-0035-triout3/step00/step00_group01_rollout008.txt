    seq = np.array(seq, dtype=float)

def project_simplex(v):
    # Duchi algorithm
    v = np.asarray(v)
    n = v.shape[0]
    # sort descending
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0)
    return w

def generate_seq(shape, n, param):
    # returns array of length n normalized to sum=1
    if shape == 'geometric':
        r = param
        seq = r**np.arange(n)
    elif shape == 'power':
        p = param
        seq = 1.0/(np.arange(1, n+1)**p)

        seq = (i+1)*(n-i)

    # Normalize
    if np.sum(seq) == 0:

def random_move(seq, step):
    n = len(seq)
    i, j = np.random.randint(n, size=2)
    if i == j:
        return seq
    delta = (np.random.rand() - 0.5) * step * seq[i]
    new_seq = seq.copy()
    new_seq[i] -= delta
    new_seq[j] += delta
    new_seq = project_simplex(new_seq)
    return new_seq

def targeted_move(seq, conv, kstar, step):
    n=len(seq)
    # compute indices that contribute to kstar
    i_indices = np.arange(n)
    j_indices = kstar - i_indices
    valid = (j_indices>=0) & (j_indices<n)
    i_valid = i_indices[valid]
    j_valid = j_indices[valid]
    # compute contributions
    contrib = seq[i_valid] * seq[j_valid]
    if contrib.size == 0:
        # fallback to random move
        return random_move(seq, step)
    # select index with max contribution
    idx = np.argmax(contrib)
    i = i_valid[idx]
    j = j_valid[idx]
    # Transfer small epsilon from both i,j to another random index k
    epsilon = step * min(seq[i], seq[j])  # step controls magnitude
    if epsilon < 1e-12:
        # fallback
        return random_move(seq, step)
    k = np.random.randint(n)
    # ensure k distinct
    while k in (i, j):
        k = np.random.randint(n)
    new_seq = seq.copy()
    new_seq[i] = max(0, new_seq[i] - epsilon)
    new_seq[j] = max(0, new_seq[j] - epsilon)
    new_seq[k] += 2*epsilon
    new_seq = project_simplex(new_seq)
    return new_seq

"""
Propose a good step‑function for the given scoring routine.

The algorithm consists of three stages:

1. **Initial candidates** – we generate a handful of parametric families
   (geometric, power‑law, sinusoidal, symmetric geometric, etc.) for a few
   different lengths.  Each candidate is normalised to a simplex
   (non‑negative with unit sum) and its exact score
   `2 * n * max(conv(a,a)) / (sum(a)**2)` is evaluated using an FFT‑based
   convolution.

2. **Local optimisation** – starting from the best candidate we perform an
   iterative stochastic improvement.  In each iteration we either apply a
   simple random mass transfer between two positions or a *targeted* move
   that reduces the contribution to the current maximal convolution index.
   After the move we project the vector back onto the probability simplex.
   If the new score is lower we accept the move; otherwise we keep the old
   configuration.  When no improvement occurs for a while we re‑initialise
   with a fresh random vector (different length) to avoid stagnation.

3. **Return** – after the allotted time (≈ 990 s) the best sequence found is
   returned as a plain Python list of floats.  All entries are non‑negative,
   ≤ 1000 and already normalised (scaling does not affect the score).

The implementation avoids any closures, only uses NumPy and the standard
library, and respects the 1‑CPU restriction.
"""

# Helper functions (all top‑level, no closures)

def _project_simplex(v):
    """Project a vector onto the probability simplex (non‑negative, sum = 1)."""
    v = np.asarray(v, dtype=float)
    n = v.shape[0]
    # sort descending
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all zero – return uniform vector
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1] - 1  # last true index (zero‑based)
    theta = cssv[cond][-1] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def _conv_max(seq):
    """Return max_k (seq * seq)[k] using a real FFT (fast for size up to ~2000)."""
    n = len(seq)
    # next power of two for zero‑padding (full convolution length = 2n‑1)
    size = 1 << ((2 * n - 1).bit_length())
    fft_seq = np.fft.rfft(seq, size)
    prod = fft_seq * fft_seq
    conv = np.fft.irfft(prod, size)[: 2 * n - 1]
    # Numerical noise may produce tiny negative values – clamp.
    return float(np.max(conv))


def _full_conv(seq):
    """Return the full linear convolution seq * seq (real FFT)."""
    n = len(seq)
    size = 1 << ((2 * n - 1).bit_length())
    fft_seq = np.fft.rfft(seq, size)
    prod = fft_seq * fft_seq
    conv = np.fft.irfft(prod, size)[: 2 * n - 1]
    return conv

    if S < 0.01:
        return float("inf"), None, None
    conv = _full_conv(seq)
    max_b = float(np.max(conv))
    k_star = int(np.argmax(conv))
    n = len(seq)
    score = 2.0 * n * max_b / (S * S)
    return score, conv, k_star


def _random_move(seq, step):
    """Transfer a random amount of mass between two positions (maintains sum)."""
    n = len(seq)
    i, j = np.random.randint(n, size=2)
    if i == j:
        return seq.copy()
    # amount proportional to current value at i
    delta = (np.random.rand() - 0.5) * step * seq[i]
    new_seq = seq.copy()
    new_seq[i] -= delta
    new_seq[j] += delta
    return _project_simplex(new_seq)


def _targeted_move(seq, conv, k_star, step):
    """
    Reduce the contribution to the current maximal convolution index.
    Pick the pair (i,j) with i+j=k_star that yields the largest product
    a_i * a_j and move a tiny amount of mass from both of them to a random
    third position.
    """
    n = len(seq)
    i_all = np.arange(n)
    j_all = k_star - i_all
    valid = (j_all >= 0) & (j_all < n)
    i_valid = i_all[valid]
    j_valid = j_all[valid]

    if i_valid.size == 0:
        return _random_move(seq, step)

    contrib = seq[i_valid] * seq[j_valid]
    idx = int(np.argmax(contrib))
    i = int(i_valid[idx])
    j = int(j_valid[idx])

    # move a small epsilon from both i and j to a random third index
    eps = step * min(seq[i], seq[j])
    if eps <= 1e-12:
        return _random_move(seq, step)

    k = np.random.randint(n)
    while k == i or k == j:
        k = np.random.randint(n)

    new_seq = seq.copy()
    new_seq[i] = max(0.0, new_seq[i] - eps)
    new_seq[j] = max(0.0, new_seq[j] - eps)
    new_seq[k] += 2.0 * eps
    return _project_simplex(new_seq)


def _generate_sequence(shape, n, param=None):
    """Create a normalised candidate sequence for a given parametric family."""
    if shape == "geometric":
        r = param if param is not None else 0.95
        seq = r ** np.arange(n, dtype=float)
    elif shape == "sym_geometric":
        # symmetric sum of two geometric tails
        r = param if param is not None else 0.95
        left = r ** np.arange(n, dtype=float)
        right = left[::-1]
        seq = left + right
    elif shape == "power":
        p = param if param is not None else 1.0
        idx = np.arange(1, n + 1, dtype=float)
        seq = 1.0 / (idx ** p)
    elif shape == "sine":
        idx = np.arange(1, n + 1, dtype=float)
        seq = np.sin(np.pi * idx / (n + 1))
    elif shape == "quadratic":
        idx = np.arange(n, dtype=float)
        seq = (idx + 1) * (n - idx)
    elif shape == "random":
        seq = np.random.rand(n)
    else:
        seq = np.random.rand(n)

    # Avoid the degenerate all‑zero case
    if seq.sum() == 0.0:
        seq = np.ones_like(seq)
    seq = _project_simplex(seq)  # normalise to sum = 1
    return seq

def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a sequence that minimises the provided evaluation function.

    Parameters
    ----------
    seed : int
        Random seed for reproducibility.
    budget_s : int
        Time budget in seconds (the function will stop a few seconds early).

    Returns
    -------
    list[float]
        The best sequence of non‑negative heights found within the time limit.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 10.0  # leave a small safety margin

    # ------------------------------------------------------------------
    # 0.  Keep the globally supplied SOTA sequence (if any) as a baseline.
    # ------------------------------------------------------------------
    best_seq = None
    best_score = float("inf")

    if "height_sequence_1" in globals():
        seq0 = np.asarray(height_sequence_1, dtype=float)
        seq0 = _project_simplex(seq0)  # normalise – scaling does not affect the score
        sc0, _, _ = _score_and_conv(seq0)
        best_seq = seq0.copy()
        best_score = sc0
        print(f"Baseline SOTA score: {best_score:.6f} (length {len(seq0)})")

    # ------------------------------------------------------------------
    # 1.  Exhaustive sweep over a few parametric families (quick, cheap).
    # ------------------------------------------------------------------
    shapes = ["geometric", "sym_geometric", "power", "sine", "quadratic"]
    lengths = [200, 400, 600]
    for n in lengths:
        for shape in shapes:
            # choose a few parameters for each shape
            if shape == "geometric" or shape == "sym_geometric":
                # r close to 1 – we need a fine grid
                rs = np.linspace(0.90, 0.9999, 25)
                for r in rs:
                    seq = _generate_sequence(shape, n, r)
                    sc, _, _ = _score_and_conv(seq)
                    if sc < best_score:
                        best_score = sc
                        best_seq = seq.copy()
                        print(f"New best {shape} (n={n}, r={r:.4f}): {best_score:.6f}")
            elif shape == "power":
                ps = np.linspace(0.1, 2.0, 20)
                for p in ps:
                    seq = _generate_sequence(shape, n, p)
                    sc, _, _ = _score_and_conv(seq)
                    if sc < best_score:
                        best_score = sc
                        best_seq = seq.copy()
                        print(f"New best {shape} (n={n}, p={p:.2f}): {best_score:.6f}")
            else:
                # shapes without extra parameter
                seq = _generate_sequence(shape, n)
                sc, _, _ = _score_and_conv(seq)
                if sc < best_score:
                    best_score = sc
                    best_seq = seq.copy()
                    print(f"New best {shape} (n={n}): {best_score:.6f}")

    # ------------------------------------------------------------------
    # 2.  Stochastic local optimisation starting from the best candidate.
    # ------------------------------------------------------------------
    if best_seq is None:
        # fallback – random start
        n = np.random.randint(150, 801)
        best_seq = _project_simplex(np.random.rand(n))
        best_score, _, _ = _score_and_conv(best_seq)

    cur_seq = best_seq.copy()
    cur_score, cur_conv, cur_kstar = _score_and_conv(cur_seq)

    it = 0
    stagnant = 0
    while time.time() < deadline:
        # Step size slowly decays; keep it positive.
        step = max(0.001, 0.02 * np.exp(-it / 8000.0))

        # Choose move type
        if np.random.rand() < 0.55:
            cand_seq = _random_move(cur_seq, step)
        else:
            cand_seq = _targeted_move(cur_seq, cur_conv, cur_kstar, step)

        cand_score, cand_conv, cand_k = _score_and_conv(cand_seq)

        if cand_score < cur_score:
            cur_seq, cur_score, cur_conv, cur_kstar = (
                cand_seq,
                cand_score,
                cand_conv,
                cand_k,
            )
            stagnant = 0
            if cand_score < best_score:
                best_score = cand_score
                best_seq = cand_seq.copy()
                print(
                    f"Iter {it:7d} – new overall best: {best_score:.6f} (len={len(best_seq)})"
                )
        else:
            stagnant += 1
            # occasional uphill move (tiny probability) to escape flat regions
            if np.random.rand() < 0.0005:
                cur_seq, cur_score, cur_conv, cur_kstar = (
                    cand_seq,
                    cand_score,
                    cand_conv,
                    cand_k,
                )
                stagnant = 0

        # If we have been stuck for a while, restart with a fresh random vector.
        if stagnant > 3000:
            n = np.random.randint(150, 801)
            cur_seq = _project_simplex(np.random.rand(n))
            cur_score, cur_conv, cur_kstar = _score_and_conv(cur_seq)
            stagnant = 0

        it += 1

    print(f"Search finished after {it} iterations. Best score: {best_score:.6f}")
    # The scoring function is invariant under scaling, but we keep the values modest.
    final_seq = [float(x) for x in best_seq]
    return final_seq
