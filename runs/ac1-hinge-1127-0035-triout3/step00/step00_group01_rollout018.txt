    while time.time() < deadline:
        # compute conv and find max index
        conv = np.convolve(a, a)
        k_max = int(np.argmax(conv))
        # gradient

def propose_candidate(seed=0, budget_s=1000, **kwargs):
    import numpy as np, time
    from scipy.optimize import linprog

    np.random.seed(seed)

    # if height_sequence_1 provided, set n to its length else default 200
    n = len(height_sequence_1) if 'height_sequence_1' in globals() else 200
    # to reduce runtime, optionally limit n to e.g., 200 if length >500
    n = min(n, 300)

    # initialize sequence to uniform normalized sum sqrt(2*n)
    cur = np.full(n, np.sqrt(2 * n) / n)

    best_seq = cur.copy()
    best_score = evaluate_sequence(best_seq.tolist())
    print(f"Initial score: {best_score:.6f}")

    deadline = time.time() + budget_s - 1.0

    # Precompute set of t candidates
    t_candidates = [0.01, 0.05, 0.1, 0.2, 0.3]
    # main loop
    iteration = 0
    while time.time() < deadline:
        # compute M = max convolution
        conv = np.convolve(cur, cur)
        M = np.max(conv)
        # solve LP to get g
        g = solve_convolution_lp(cur, M)
        if g is None:
            # random perturbation if LP fails
            idx = np.random.randint(n)
            cur[idx] = max(0, cur[idx] + np.random.randn() * 0.01 * np.sqrt(2*n)/n)
            cur = np.clip(cur, 0, 1000.0)
            cur = cur * (np.sqrt(2*n) / np.sum(cur))
            continue
        # normalize g to have same sum
        sum_g = np.sum(g)
        g_norm = g * np.sqrt(2 * n) / sum_g

        # try several t and pick best improvement
        best_local = None
        best_local_score = best_score
        for t in t_candidates:
            new_seq = (1 - t) * cur + t * g_norm
            # ensure within bounds (non-negative), clip
            new_seq = np.clip(new_seq, 0, 1000.0)
            # normalization not needed as both already sum sqrt(2*n)
            score = evaluate_sequence(new_seq.tolist())
            if score < best_local_score:
                best_local_score = score
                best_local = new_seq
        if best_local is not None:
            cur = best_local
            if best_local_score < best_score:
                best_score = best_local_score
                best_seq = cur.copy()
                print(f"Iter {iteration}: new best {best_score:.6f}")
        else:
            # no improvement, small random perturb
            cur = cur + 0.001 * (np.random.rand(n) - 0.5)
            cur = np.clip(cur, 0, 1000.0)
            cur = cur * (np.sqrt(2*n) / np.sum(cur))

        iteration += 1
        # occasional restart: if no improvement for many iterations, randomize slight
        if iteration % 200 == 0:
            cur = best_seq + 0.01 * np.random.randn(n)
            cur = np.clip(cur, 0, 1000.0)
            cur = cur * (np.sqrt(2*n) / np.sum(cur))
    # after loop, return best_seq as list of floats
    return best_seq.tolist()

from scipy.optimize import linprog

# ----------------------------------------------------------------------
# Helper: project a vector onto the simplex Σx_i = sqrt(2n), x_i ≥ 0
def _renorm_to_sqrt2n(v, n):
    """Scale v to have ℓ₁‑norm sqrt(2*n) and clip to allowed bounds."""
    # protect against zero norm – fall back to uniform
    norm = np.sum(v)
    if norm <= 0:
        v = np.full_like(v, np.sqrt(2.0 * n) / v.shape[0])
    else:
        v = v * (np.sqrt(2.0 * n) / norm)
    # clamp to the bounds used in evaluate_sequence
    v = np.clip(v, 0.0, 1000.0)
    return v


# ----------------------------------------------------------------------
def solve_convolution_lp(f, M):
    """
    Solve the linear programme:
        maximise   Σ b_j
        subject to conv_k(f,b) ≤ M   for all k=0..2n-2
                    b_j ≥ 0
    The objective is linear because conv_k(f,b) = Σ_i f_i b_{k-i}.

    Returns the optimal b as a 1‑d numpy array, or None on failure.
    """
    n = f.shape[0]
    # Objective: maximise sum(b)  → minimise -sum(b)
    c = -np.ones(n)

    # Build inequality matrix A_ub·b ≤ b_ub for the convolution constraints.
    # There are (2n‑1) constraints, one per output index of full convolution.
    rows = []
    rhs = []
    for k in range(2 * n - 1):
        row = np.zeros(n)
        # conv_k(f,b) = Σ_i f_i * b_{k-i}, where index outside [0,n) is ignored.
        for i in range(n):
            j = k - i
            if 0 <= j < n:
                row[j] += f[i]
        rows.append(row)
        rhs.append(M)

    A_ub = np.vstack(rows)
    b_ub = np.asarray(rhs)

    # Variable bounds b_j ≥ 0 (no explicit upper bound needed)
    bounds = [(0, None)] * n

    # Use the HiGHS interior‑point method; it handles moderate dense problems well.
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method="highs")
    if res.success:
        return res.x
    return None


# ----------------------------------------------------------------------
def propose_candidate(seed=42, budget_s=1000, **kwargs):
    """
    Search for a sequence that minimises `evaluate_sequence`.

    The routine runs a projected‑subgradient descent where the descent
    direction is obtained from the linear programme described above.
    Multiple step sizes are tried each iteration and the best candidate
    is kept.  Random perturbations are injected periodically to escape
    plateaus.  The function respects the global `height_sequence_1` (if
    present) as a possible starting point, but also works with a fresh
    uniform initialisation.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 1.0  # spare a second for clean exit

    # ------------------------------------------------------------------
    # Choose a reasonable length.  Using the full 4160‑element list would
    # make each LP very large; a moderate size (≤300) works well within
    # the time budget and still yields an excellent bound.
    try:
        n_full = len(height_sequence_1)  # may be defined globally
    except Exception:
        n_full = 200
    n = min(300, max(100, n_full))  # keep n between 100 and 300

    # Initial sequence: uniform, normalised to sum = sqrt(2n)
    cur = np.full(n, np.sqrt(2.0 * n) / n)

    # Keep the best ever seen
    best_seq = cur.copy()
    best_score = evaluate_sequence(best_seq.tolist())
    print(f"[init] length={n}, score={best_score:.6f}")

    # Pre‑define a small set of step‑size candidates for the convex combination
    t_candidates = [0.01, 0.05, 0.1, 0.2, 0.3, 0.5]

    iteration = 0
    no_improve = 0

    while time.time() < deadline:
        # ---- current maximal convolution entry
        conv = np.convolve(cur, cur)
        M = np.max(conv)

        # ---- linear programme to find a better direction
        g_raw = solve_convolution_lp(cur, M)

        if g_raw is None:
            # LP failed (numerical issue); apply a tiny random jitter
            jitter = 0.02 * (np.random.rand(n) - 0.5) * np.sqrt(2.0 * n) / n
            cur = np.clip(cur + jitter, 0.0, 1000.0)
            cur = _renorm_to_sqrt2n(cur, n)
            continue

        # ---- normalise the LP solution to the same ℓ₁‑norm as `cur`
        g = _renorm_to_sqrt2n(g_raw, n)

        # ---- try several convex combinations and keep the best one
        improved = False
        for t in t_candidates:
            cand = (1.0 - t) * cur + t * g
            cand = _renorm_to_sqrt2n(cand, n)   # keep the invariant sum = √(2n)

            # evaluate – the function already clips negative/large values
            try:
                score = evaluate_sequence(cand.tolist())
            except Exception:
                continue

            if score < best_score - 1e-12:          # strict improvement
                best_score = score
                best_seq = cand.copy()
                cur = cand.copy()                    # continue from the improvement
                improved = True
                print(f"[iter {iteration}] new best = {best_score:.6f} (t={t})")
                break   # accept the first improving t (larger t may overshoot)

        if not improved:
            # No step gave improvement – inject a small random perturbation
            jitter = 0.005 * (np.random.rand(n) - 0.5) * np.sqrt(2.0 * n) / n
            cur = np.clip(cur + jitter, 0.0, 1000.0)
            cur = _renorm_to_sqrt2n(cur, n)
            no_improve += 1
        else:
            no_improve = 0

        # Periodically re‑randomise a little if we have been stuck for a while
        if no_improve >= 150:
            cur = best_seq + 0.02 * np.random.randn(n)
            cur = _renorm_to_sqrt2n(cur, n)
            no_improve = 0

        iteration += 1

    # Return the best sequence found (as a plain Python list)
    return best_seq.tolist()
