def propose_candidate(seed=0, budget_s=1000):
    import numpy as np, time, math
    start = time.time()
    deadline = start + budget_s - 0.1
    np.random.seed(seed)

    if 'height_sequence_1' in globals():
        n = len(height_sequence_1)
    else:
        n = 3000  # fallback

    # initialize best from uniform
    best_a = np.full(n, 1.0/n, dtype=np.float64)
    best_score = evaluate_sequence(best_a.tolist())

    print(f"Start with uniform: score={best_score}")

    # 1) random param sweep
    param_deadline = start + 0.1 * budget_s  # 10% time for param sweep
    while time.time() < param_deadline:
        # sample random powers
        p = np.random.uniform(-4, 4)  # single exponent
        a = np.exp(p * np.log(np.arange(1, n+1, dtype=np.float64)))
        # optionally add second exponent
        if np.random.rand() < 0.5:
            q = np.random.uniform(-4, 4)
            a = a * np.exp(q * np.log(np.arange(n, 0, -1, dtype=np.float64)))  # (n-i) term
        a_sum = a.sum()
        if a_sum == 0:
            continue
        a = a / a_sum  # normalize sum to 1
        score = evaluate_sequence(a.tolist())
        if score < best_score:
            best_score = score
            best_a = a.copy()
            print(f"New param shape score={best_score:.6f}, p={p:.3f}, q={q if 'q' in locals() else None}")
    # Now hill climbing from best_a
    cur_a = best_a.copy()
    cur_score = best_score

    # annealing parameters
    T0 = 1.0
    T_end = 1e-4
    max_step_factor = 0.2
    min_step_factor = 1e-4

    it = 0
    while time.time() < deadline:
        elapsed = time.time() - start
        progress = elapsed / budget_s
        # temperature schedule
        T = T0 * (1 - progress) + T_end
        step_factor = max_step_factor * (1 - progress) + min_step_factor

        # propose move
        i = np.random.randint(n)
        j = np.random.randint(n-1)
        if j >= i:
            j += 1
        delta = min(cur_a[i], step_factor * cur_a[i])
        if delta <= 0:
            continue
        a_new = cur_a.copy()
        a_new[i] -= delta
        a_new[j] += delta
        # ensure sum unchanged (tiny floating error may appear)
        # no need to renorm

        new_score = evaluate_sequence(a_new.tolist())
        # acceptance
        if new_score < cur_score or np.random.rand() < math.exp(-(new_score - cur_score) / (T + 1e-12)):
            cur_a = a_new
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_a = cur_a.copy()
                print(f"Iter {it}: new best score={best_score:.6f}")
        it += 1

        # occasional large random perturbation to avoid stagnation
        if it % 5000 == 0:
            # perturb a small fraction 1%
            idxs = np.random.choice(n, size=int(0.01*n), replace=False)
            perturb = np.random.randn(len(idxs)) * 0.01 * cur_a[idxs]
            cur_a[idxs] = np.maximum(0.0, cur_a[idxs] + perturb)
            # renormalize sum to 1
            cur_a = cur_a / cur_a.sum()
            cur_score = evaluate_sequence(cur_a.tolist())
            if cur_score < best_score:
                best_score = cur_score
                best_a = cur_a.copy()
                print(f"Iter {it}: after perturb best={best_score:.6f}")

    print(f"Finished: best_score={best_score:.6f}")
    return best_a.tolist()

"""
Propose a step‐function (non‑negative sequence) that minimises

    2 * n * max(conv(a,a)) / (sum(a)**2)

The algorithm works in three stages:

1. **Parameter sweep** –  simple one‑ or two‑parameter families
   (power laws) are sampled quickly to obtain a decent starting shape.

2. **Targeted simulated annealing** –  we keep the current sequence
   ``a`` (sum≈1).  At each iteration the convolution is inspected,
   the index of the current maximum is identified and a small amount
   of mass is moved from an index that contributes to this maximum
   to an index that does not.  A temperature schedule allows occasional
   uphill moves to escape local minima.  The best sequence seen so far
   is stored.

3. **Occasional random perturbations** –  to avoid stagnation a small
   random jitter is applied from time to time.

The whole search is time‑bounded (default 1000 s).  The function returns
the best sequence found.  It uses only NumPy, the provided
``evaluate_sequence`` and standard library modules.
"""
import time
import math
import numpy as np


def _initial_power_shape(n: int, rng: np.random.Generator) -> np.ndarray:
    """Generate a shape a_i ∝ (i+1)^p * (n-i)^q (one of p,q may be zero)."""
    i = np.arange(1, n + 1, dtype=np.float64)          # 1 … n
    rev = np.arange(n, 0, -1, dtype=np.float64)       # n … 1

    # random exponents – keep them modest to avoid overflow
    p = rng.uniform(-4.0, 4.0)
    q = rng.uniform(-4.0, 4.0) if rng.random() < 0.5 else 0.0

    # a = exp(p*log(i)) * exp(q*log(rev))
    a = np.exp(p * np.log(i))
    if q != 0.0:
        a *= np.exp(q * np.log(rev))

    # sanity – discard pathological exponent choices
    if not np.all(np.isfinite(a)):
        return None
    s = a.sum()
    if s == 0.0:
        return None
    return a / s


def propose_candidate(seed: int = 0, budget_s: int = 1000) -> list[float]:
    """Return the best sequence found within ``budget_s`` seconds."""
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.1   # tiny safety margin

    # -------------------------------------------------------------
    # 1. Choose a problem size (use the length of the supplied seed if any)
    # -------------------------------------------------------------
    if "height_sequence_1" in globals():
        n = len(height_sequence_1)
    else:
        n = 3000                     # fallback size (still large enough)

    # -------------------------------------------------------------
    # 2. Initialise with a uniform sequence (sum = 1)
    # -------------------------------------------------------------
    a = np.full(n, 1.0 / n, dtype=np.float64)
    best_a = a.copy()
    best_score = evaluate_sequence(best_a.tolist())

    print(f"[init] n={n}, uniform score={best_score:.6f}")

    # -------------------------------------------------------------
    # 3. Quick parametric sweep (≈10 % of the time budget)
    # -------------------------------------------------------------
    sweep_deadline = start_time + 0.1 * budget_s
    while time.time() < sweep_deadline:
        cand = _initial_power_shape(n, rng)
        if cand is None:
            continue
        sc = evaluate_sequence(cand.tolist())
        if sc < best_score:
            best_score = sc
            best_a = cand.copy()
            a = cand.copy()                     # start the next phase from it
            print(f"[sweep] improved to {best_score:.6f}")

    cur_a = best_a.copy()
    cur_score = best_score

    # -------------------------------------------------------------
    # 4. Simulated‑annealing / hill‑climbing phase
    # -------------------------------------------------------------
    # Temperature schedule
    T0 = 1.0
    T_end = 1e-4
    # Step‑size schedule (fraction of the source entry that is moved)
    max_step = 0.20
    min_step = 1e-4

    iteration = 0
    # Pre‑allocate an index array for speed
    idx_array = np.arange(n, dtype=np.int64)

    while time.time() < deadline:
        # progress‑dependent hyper‑parameters
        elapsed = time.time() - start_time
        progress = elapsed / budget_s
        T = T0 * (1.0 - progress) + T_end
        step_factor = max_step * (1.0 - progress) + min_step

        # ---------------------------------------------------------
        # a) decide on a move – 70 % targeted, 30 % random
        # ---------------------------------------------------------
        if rng.random() < 0.7:
            # compute the current convolution once (needed for the max‑index)
            conv_cur = np.convolve(cur_a, cur_a)
            k_max = int(conv_cur.argmax())

            # indices that actually contribute to b[k_max]
            lo = max(0, k_max - (n - 1))
            hi = min(n - 1, k_max)
            contrib_mask = (idx_array >= lo) & (idx_array <= hi)

            contrib_idx = idx_array[contrib_mask]
            if contrib_idx.size == 0:          # should never happen
                i = rng.integers(n)
            else:
                probs = cur_a[contrib_mask]
                total = probs.sum()
                if total == 0.0:
                    i = rng.choice(contrib_idx)
                else:
                    probs = probs / total
                    i = rng.choice(contrib_idx, p=probs)

            # pick a destination that does NOT contribute to the current max
            non_mask = ~contrib_mask
            non_idx = idx_array[non_mask]
            if non_idx.size == 0:
                # corner case (k_max at an extreme) – just pick anywhere else
                j = rng.integers(n)
                while j == i:
                    j = rng.integers(n)
            else:
                j = rng.choice(non_idx)
        else:
            # simple random move – source biased towards larger entries
            probs = cur_a / cur_a.sum()
            i = rng.choice(idx_array, p=probs)
            j = rng.integers(n)
            while j == i:
                j = rng.integers(n)

        # ---------------------------------------------------------
        # b) propose the transfer
        # ---------------------------------------------------------
        delta = min(cur_a[i], step_factor * cur_a[i])
        if delta <= 0.0:
            continue

        a_new = cur_a.copy()
        a_new[i] -= delta
        a_new[j] += delta

        # renormalise to keep the sum exactly 1 (tiny floating‑point drift)
        a_new /= a_new.sum()

        # ---------------------------------------------------------
        # c) evaluate the candidate
        # ---------------------------------------------------------
        new_score = evaluate_sequence(a_new.tolist())

        # ---------------------------------------------------------
        # d) Metropolis acceptance
        # ---------------------------------------------------------
        accept = False
        if new_score < cur_score:
            accept = True
        else:
            # uphill move with probability exp(-(Δ)/T)
            prob = math.exp(-(new_score - cur_score) / (T + 1e-12))
            if rng.random() < prob:
                accept = True

        if accept:
            cur_a = a_new
            cur_score = new_score
            # record global best
            if new_score < best_score:
                best_score = new_score
                best_a = a_new.copy()
                print(f"[iter {iteration:07d}] new best = {best_score:.6f}")

        # ---------------------------------------------------------
        # e) occasional diagnostics / perturbations
        # ---------------------------------------------------------
        if iteration % 2000 == 0:
            print(f"[iter {iteration:07d}] cur={cur_score:.6f}  best={best_score:.6f}")

        if iteration % 20000 == 0 and iteration > 0:
            # a tiny stochastic jitter to revive stagnation
            mask = rng.random(n) < 0.01                # ~1 % of positions
            jitter = rng.normal(0.0, 0.005, size=n)     # small Gaussian noise
            cur_a[mask] += jitter[mask]
            cur_a = np.clip(cur_a, 0.0, None)
            cur_a /= cur_a.sum()
            cur_score = evaluate_sequence(cur_a.tolist())
            if cur_score < best_score:
                best_score = cur_score
                best_a = cur_a.copy()
                print(f"[jitter] improved best to {best_score:.6f}")

        iteration += 1

    print(f"Finished after {iteration} iterations – best score = {best_score:.6f}")
    return best_a.tolist()
