def compute_score(seq):
    a = np.asarray(seq, dtype=np.float64)
    n = len(a)
    # compute convolution via FFT
    size = 1 << ((2*n-1).bit_length())
    fa = np.fft.rfft(a, size)
    conv = np.fft.irfft(fa*fa, size)[:2*n-1]
    max_b = np.max(conv)
    sum_a = np.sum(a)
    if sum_a < 0.01:
        return float('inf')
    return float(2 * n * max_b / (sum_a ** 2))
```

"""
Propose a step‑function (non‑negative sequence) with a low
evaluate_sequence() score.

The algorithm works in three stages:

1. **Parametric warm‑start** –  a family of power‑law sequences
   a[i] = (i + 1 + d)^{-p}  (p≈0.5, d≥0) is generated for several
   (p,d) pairs on the target length (default 7156).  The best of these
   candidates is taken as the initial solution.

2. **Fast score evaluation** –  the objective
   2·n·max(conv(a,a)) / (sum(a)^2) is computed with an FFT based
   convolution, which is much faster than the supplied
   evaluate_sequence() (the latter is used only for the final answer).

3. **Stochastic hill‑climbing** –  the current best vector is mutated
   repeatedly:
        * a small random subset of indices is perturbed multiplicatively
          (log‑normal steps) to keep positivity,
        * the whole vector is re‑normalised to keep sum(a)=1,
        * the new score is evaluated,
        * the move is kept if it improves the score (or with a very small
          probability to escape plateaux).
   The loop runs until the allotted time (~1000 s) is exhausted.

The final sequence is returned as a plain Python list of floats.
"""

import time
import math
import random
import numpy as np

# ----------------------------------------------------------------------
# Helper: fast computation of the evaluation function (identical to
# evaluate_sequence, but without the safety‑checks and using FFT).
def fast_score(seq: np.ndarray) -> float:
    """Return the same value as evaluate_sequence for a non‑negative array."""
    a = np.asarray(seq, dtype=np.float64)
    n = a.shape[0]
    # size for FFT – next power of two ≥ 2*n‑1
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    max_b = np.max(conv)
    sum_a = np.sum(a)
    if sum_a < 0.01:
        return float("inf")
    return float(2.0 * n * max_b / (sum_a * sum_a))


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 123, budget_s: int = 1000, **kwargs):
    """
    Search for a sequence with a low evaluate_sequence() value.

    The function respects the time budget (≈budget_s seconds) and returns
    the best sequence found as a list of floats.
    """
    np.random.seed(seed)
    random.seed(seed)

    # ------------------------------------------------------------------
    # 1. Warm start – scan a small grid of (p,d) for power‑law shapes.
    n = 7156                     # target length (can be changed)
    best_seq = None
    best_val = float("inf")

    # grid: p around 0.5 (the exponent that gives the continuous optimum)
    # d shifts the singularity away from the left endpoint.
    p_grid = np.linspace(0.44, 0.56, 13)   # step ≈0.01
    d_grid = np.arange(0.0, 6.0, 1.0)     # 0,1,…,5

    print("=== Grid search for power‑law warm‑starts ===")
    for p in p_grid:
        for d in d_grid:
            idx = np.arange(n, dtype=np.float64)
            # a[i] = (i+1+d)^{-p}
            a = (idx + 1.0 + d) ** (-p)
            a_sum = a.sum()
            if a_sum == 0:
                continue
            a /= a_sum                     # normalise L1‑norm to 1
            val = fast_score(a)
            if val < best_val:
                best_val = val
                best_seq = a.copy()
                print(f"grid p={p:.3f}, d={d:.1f} → score={val:.6f}")

    if best_seq is None:                     # fallback: pure uniform
        best_seq = np.ones(n, dtype=np.float64) / n
        best_val = fast_score(best_seq)

    # ------------------------------------------------------------------
    # 2. Stochastic hill‑climbing (simulated annealing style)
    seq = best_seq.copy()
    cur_val = best_val
    start_time = time.time()
    deadline = start_time + budget_s - 2.0   # small safety margin

    it = 0
    # gradually shrink the perturbation magnitude
    while time.time() < deadline:
        it += 1
        # relative scale of the log‑normal perturbation; start with 5 %
        sigma = 0.08 * (0.9995 ** it)       # exponential decay
        # number of coordinates to touch – 1 % of length, at least 1
        n_mut = max(1, int(0.01 * n))

        # choose indices without replacement
        idxs = np.random.choice(n, size=n_mut, replace=False)

        # multiplicative log‑normal update (keeps positivity)
        proposal = seq.copy()
        log_factor = np.random.randn(n_mut) * sigma
        proposal[idxs] *= np.exp(log_factor)

        # enforce non‑negativity (exp never gives negative)
        # renormalise to keep sum == 1 (scaling invariance of the score)
        s = proposal.sum()
        if s == 0:
            continue
        proposal /= s

        new_val = fast_score(proposal)

        # accept if better, otherwise with a tiny probability
        if new_val < cur_val or np.random.rand() < 0.0005:
            seq = proposal
            cur_val = new_val
            if new_val < best_val:
                best_val = new_val
                best_seq = proposal.copy()
                print(
                    f"Iter {it:6d}: new best score = {best_val:.6f} "
                    f"(sigma={sigma:.5f})"
                )

    # ------------------------------------------------------------------
    # Final sanity check with the official evaluate_sequence (if it exists)
    try:
        # evaluate_sequence is assumed to be in the global namespace
        official = evaluate_sequence(best_seq.tolist())
        print(f"Official evaluate_sequence score = {official:.6f}")
    except Exception as e:
        print("Could not call evaluate_sequence:", e)

    print(f"Search finished after {it} iterations, best score = {best_val:.6f}")
    return best_seq.tolist()
