"""
Propose a step‑function sequence whose evaluation score is < 1.503.

The algorithm starts from the classic binomial (Dyck) sequence
    a_i = C(2i,i)/4**i  (i = 0,…,n‑1)
which satisfies a*a = 1 (constant) in the infinite limit.
After normalising the ℓ1‑norm to sqrt(2·n) the initial score is
≈ π/2 ≈ 1.571.  We then iteratively improve this value by the
standard LP‑based "direction‑finding" step:

    maximise Σ b_j   s.t.   (f * b)[k] ≤ M   (∀k) ,   b ≥ 0

where f is the current sequence, M = max(f*f).  The LP yields a
vector b whose ℓ1‑norm is typically larger than that of f.
We rescale b to the same ℓ1‑norm as f (so the sum is unchanged) and
mix it with f using a convex combination.  Because b has a smaller
self‑convolution than f, the resulting mix reduces the maximum
autocorrelation M.  Repeating this step quickly drives the score
below the target.

The implementation works for a modest length (n≈600) which makes each
LP cheap (≈10⁵ variables) and fits comfortably in the 1000 s budget.
Random perturbations are used as a fallback if the LP fails or gets
stuck.
"""

import time
import math
import random

import numpy as np
from scipy.optimize import linprog
from scipy.sparse import csr_matrix


def binomial_sequence(n: int) -> list[float]:
    """Return a_i = C(2i,i)/4**i for i=0..n-1 using the recurrence
    a_{i+1}=a_i * (2*i+1)/(2*(i+1)).
    """
    seq = [1.0]
    for i in range(1, n):
        seq.append(seq[-1] * (2 * i - 1) / (2 * i))
    return seq


def max_convolution(seq: np.ndarray) -> float:
    """Return max_k (seq * seq)[k]."""
    conv = np.convolve(seq, seq)
    return float(conv.max())


def solve_lp_for_direction(f: np.ndarray, M: float,
                           rows: np.ndarray, cols: np.ndarray) -> np.ndarray | None:
    """Linear programme:
       maximise Σ b_j   s.t.   (f * b)[k] ≤ M   for all k,
                              b_j ≥ 0.
       The constraint matrix is pre‑computed (rows, cols) – only the
       data (the coefficients f_i) changes each call.
       Returns the optimal b or None on failure.
    """
    n = f.size
    # data = f_i repeated n times (once per column j)
    data = np.repeat(f, n)
    A = csr_matrix((data, (rows, cols)), shape=(2 * n - 1, n))
    b = np.full(2 * n - 1, M, dtype=np.float64)

    c = -np.ones(n, dtype=np.float64)  # maximise sum => minimise -sum
    bounds = [(0, None)] * n

    # HiGHS is the default nowadays and works well with sparse matrices
    result = linprog(c, A_ub=A, b_ub=b, bounds=bounds, method='highs')
    if result.success:
        return result.x
    return None


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """Return a list of non‑negative heights (step function) with a low
    evaluation score.  The function runs for at most ``budget_s`` seconds.
    """
    random.seed(seed)
    np.random.seed(seed)

    # ------------------------------------------------------------------
    # 1.  Problem size and normalisation
    # ------------------------------------------------------------------
    n = 600                     # length of the sequence (tunable)
    S = math.sqrt(2.0 * n)      # required ℓ1‑norm (makes score = max conv)

    # 2.  Initial sequence: binomial coefficients (Dyck/path) → flat autocorr.
    f = np.asarray(binomial_sequence(n), dtype=np.float64)
    f *= (S / f.sum())          # normalise ℓ1‑norm to S

    # Pre‑compute row / column indices for the convolution matrix (once).
    idx_i = np.arange(n, dtype=np.int64)
    idx_j = np.arange(n, dtype=np.int64)
    row_inds = (idx_i[:, None] + idx_j[None, :]).ravel()   # i + j
    col_inds = np.tile(idx_j, n)                           # j repeated for each i

    # Keep track of the best sequence found so far.
    best_f = f.copy()
    best_M = max_convolution(best_f)
    best_score = best_M               # because sum = S, evaluation = M

    print(f"[init] n={n}, initial M={best_M:.6f}")

    start_time = time.time()
    deadline = start_time + budget_s - 5.0   # safety margin

    iteration = 0
    no_improve_counter = 0

    # ------------------------------------------------------------------
    # 3.  Main optimisation loop
    # ------------------------------------------------------------------
    while time.time() < deadline:
        iteration += 1

        # current maximum autocorrelation
        M = max_convolution(f)

        # solve the LP to obtain a direction that can increase the ℓ1‑norm
        g = solve_lp_for_direction(f, M, row_inds, col_inds)

        if g is None:
            # LP failure – random jitter to escape possible degeneracy
            idx = np.random.randint(n)
            f[idx] = max(0.0, f[idx] + np.random.randn() * 0.01)
            continue

        sum_g = g.sum()
        if sum_g <= 0.0:
            # degenerate solution – perturb a random entry
            idx = np.random.randint(n)
            f[idx] = max(0.0, f[idx] + np.random.randn() * 0.01)
            continue

        # rescale g so that its ℓ1‑norm equals S (same as f)
        scale = S / sum_g
        g_scaled = g * scale

        # Try a few mixing coefficients and keep the one that reduces M most.
        improved = False
        for t in (0.3, 0.2, 0.1, 0.4, 0.5):
            f_candidate = (1.0 - t) * f + t * g_scaled
            M_candidate = max_convolution(f_candidate)

            # Accept if we obtain a strictly smaller maximum autocorrelation.
            if M_candidate + 1e-8 < best_M:
                f = f_candidate
                best_f = f_candidate.copy()
                best_M = M_candidate
                best_score = M_candidate
                improved = True
                no_improve_counter = 0
                print(f"[iter {iteration:04d}] t={t:.2f} → M={best_M:.6f}")
                break   # stop scanning t‑values once improvement found

        if not improved:
            no_improve_counter += 1
            # occasional random jitter to avoid stagnation
            if no_improve_counter % 20 == 0:
                idx = np.random.randint(n)
                f[idx] = max(0.0, f[idx] + np.random.randn() * 0.02)

        # Very early exit if we already beat the target comfortably.
        if best_M < 1.503 and time.time() - start_time > 30:
            # we have a good solution; keep refining a little longer.
            pass

    # ------------------------------------------------------------------
    # 4.  Return the best sequence found.
    # ------------------------------------------------------------------
    final_score = evaluate_sequence(best_f.tolist())
    print(f"[done] best M={best_M:.6f}, evaluation={final_score:.6f}, "
          f"iterations={iteration}")

    # Convert to plain Python list as required by the surrounding harness.
    return best_f.tolist()
