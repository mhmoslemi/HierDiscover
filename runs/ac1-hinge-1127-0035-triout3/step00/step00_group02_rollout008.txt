    n = len(a)
    b = np.convolve(a, a)
    t = 1.0/(2*n - 1)
    r = b - t
    # compute gradient
    # use convolve with reversed r
    conv = np.convolve(a, r[::-1])
    # conv length: n + (2*n-1) -1 = 3*n -2
    # slice conv[n-1 : 2*n-1] (size n)

    # step: update
    a_new = a - step * g
    a_new[a_new < 0] = 0.0
    # renorm

        a_new = np.full_like(a, 1.0/len(a))
    else:

    return a_new

def propose_candidate(seed=0, budget_s=1000):
    import time, numpy as np
    rng = np.random.default_rng(seed)
    deadline = time.time() + budget_s - 0.5
    best_seq = None
    best_score = float('inf')
    def normalize(a):
        s = a.sum()
        if s <= 0:
            # uniform fallback
            n = len(a)
            return np.full(n, 1.0/n)
        return a / s
    def score_of(a):
        n = len(a)
        b = np.convolve(a, a)
        maxb = b.max()
        s = a.sum()
        if s < 0.01:
            return float('inf')
        return float(2.0 * n * maxb / (s * s))
    # list of candidate lengths
    lengths = [300, 600, 1200, 1800]
    for n in lengths:
        # generate some start points
        for start_type in range(3):
            if time.time() >= deadline:
                break
            if start_type == 0:
                a = np.full(n, 1.0/n)  # uniform
            elif start_type == 1:
                # random Dirichlet
                a = rng.gamma(1.0, 1.0, size=n)
                a = normalize(a)
            else:
                # try to upscale previous best if exists
                if best_seq is not None:
                    # interpolate best_seq to new length
                    old = np.array(best_seq)
                    x_old = np.linspace(0, 1, len(old))
                    x_new = np.linspace(0, 1, n)
                    a = np.interp(x_new, x_old, old)
                    a = normalize(a)
                else:
                    a = np.full(n, 1.0/n)
            cur = a.copy()
            cur_score = score_of(cur)
            if cur_score < best_score:
                best_score, best_seq = cur_score, cur.tolist()
                print(f"Initial best: {best_score:.6f} len={n} type={start_type}")
            step = 0.5
            # iteration loop
            it = 0
            while time.time() < deadline:
                # gradient descent step
                ncur = len(cur)
                # compute convolution and gradient
                b = np.convolve(cur, cur)
                t = 1.0/(2*ncur - 1)
                r = b - t
                # gradient via convolution
                conv = np.convolve(cur, r[::-1])
                g = 4.0 * conv[ncur-1:2*ncur-1][::-1]
                # line search
                eta = step / (1.0 + it * 0.001)
                new = cur - eta * g
                new = np.maximum(new, 0.0)
                new = normalize(new)
                new_score = score_of(new)
                # accept if improvement or with small probability
                if new_score < cur_score or rng.random() < 0.001:
                    cur = new
                    cur_score = new_score
                # record best
                if new_score < best_score:
                    best_score = new_score
                    best_seq = new.tolist()
                    print(f"New best {best_score:.6f} after {it} it len={ncur}")
                # occasional random noise
                if it % 200 == 0:
                    # perturb a few entries
                    for _ in range(3):
                        idx = rng.integers(ncur)
                        cur[idx] = max(0.0, cur[idx] + rng.normal(scale=0.01))
                    cur = normalize(cur)
                it += 1
                if it > 20000:
                    break
    print(f"Final best score {best_score}")
    return best_seq

"""
Propose a step‑function sequence that minimizes the evaluation function.

The algorithm works as follows:

1.  A set of initial sequences (uniform, random Dirichlet and an
    up‑scaled version of the current best) is generated for several
    lengths (from a few hundred up to ~2000 entries).

2.  For each start we run a projected sub‑gradient descent that tries to
    flatten the auto‑convolution.  Let a be the current non‑negative
    vector (∑a = 1).  Its convolution b = a*a has length 2·n−1.
    The target flat level is t = 1/(2·n−1) (the average of b).  Define
    r = b−t.  The gradient of the squared deviation
    L = ‖b−t·1‖₂² w.r.t. a is

        g = 4·conv(a, r[::-1])[n‑1 : 2·n‑1][::-1]

    (a fast O(n log n) convolution).  A step a←a−η·g is taken,
    negative entries are clipped to zero and the vector is re‑normalised
    to unit ℓ₁‑norm.

3.  The step size η is slowly decreased; every few hundred iterations a
    small random perturbation or a brief smoothing (convolution with a
    three‑point kernel) is applied to escape shallow local minima.

4.  The best sequence (lowest score) found across all runs before the
    deadline is returned.

The score used internally is exactly the one returned by
``evaluate_sequence`` after fixing the ℓ₁‑norm to 1:
    score = 2·n·max(b) / (∑a)² = 2·n·max(b)
because ∑a = 1.  This avoids calling the (slower) security‑checked
function at every iteration.

The routine respects the 1000 s time limit and works with a single CPU.
"""

import time
import numpy as np

def _normalize(vec: np.ndarray) -> np.ndarray:
    """Project onto the simplex {x≥0, sum(x)=1}."""
    s = vec.sum()
    if s <= 0.0:
        # fallback to uniform distribution
        n = vec.shape[0]
        return np.full(n, 1.0 / n, dtype=float)
    return vec / s


def _score(vec: np.ndarray) -> float:
    """Exact objective value for a vector with sum(vec)=1."""
    n = vec.shape[0]
    b = np.convolve(vec, vec)
    maxb = b.max()
    # the external evaluator also returns 2·n·maxb / (sum^2)
    # here sum == 1, so we can omit the denominator
    return float(2.0 * n * maxb)


def _gradient_flatten(vec: np.ndarray) -> np.ndarray:
    """
    Gradient of L = ||a*a - t||_2^2 where t = 1/(2n-1)·1.
    The formula reduces to:
        g = 4 * conv(a, (a*a - t)[::-1])[n-1 : 2n-1][::-1]
    """
    n = vec.shape[0]
    conv_full = np.convolve(vec, vec)                 # length 2n‑1
    t = 1.0 / (2 * n - 1)
    r = conv_full - t                                 # deviation vector
    # convolution with reversed r
    tmp = np.convolve(vec, r[::-1])                  # length 3n‑2
    # extract the slice that corresponds to g_i = Σ_j a_j r_{i+j}
    g = 4.0 * tmp[n - 1 : 2 * n - 1][::-1]
    return g


def _smooth(vec: np.ndarray) -> np.ndarray:
    """Mild smoothing with a 3‑point kernel that preserves the sum."""
    kernel = np.array([0.25, 0.5, 0.25], dtype=float)
    smoothed = np.convolve(vec, kernel, mode='same')
    return _normalize(smoothed)


def propose_candidate(seed: int = 0, budget_s: float = 1000, **kwargs):
    """
    Search for a sequence of non‑negative heights that yields a low
    evaluation score.

    Parameters
    ----------
    seed : int
        Random seed for reproducibility.
    budget_s : float
        Time budget in seconds (the function will stop before this
        limit and return the best sequence found).

    Returns
    -------
    list[float]
        The best sequence discovered within the time budget.
    """
    rng = np.random.default_rng(seed)
    deadline = time.time() + budget_s - 0.5      # leave a tiny margin

    best_seq = None
    best_val = float('inf')

    # ------------------------------------------------------------------
    # Helper to update the global best and print progress
    # ------------------------------------------------------------------
    def _maybe_update(candidate):
        nonlocal best_seq, best_val
        val = _score(candidate)
        if val < best_val:
            best_val = val
            best_seq = candidate.tolist()
            print(f"[{time.time():.2f}] New global best: {best_val:.6f} "
                  f"(len={candidate.shape[0]})")
        return val

    # ------------------------------------------------------------------
    # Prepare a list of target lengths; longer lengths give a lower
    # theoretical bound (≈1) but are slower to optimise.
    # ------------------------------------------------------------------
    target_lengths = [300, 600, 1200, 1800]

    # a short routine that runs optimisation from a given start vector
    def _optimize_from(start_vec):
        a = _normalize(start_vec.astype(float))
        cur_val = _score(a)

        # keep a copy of the best local vector (may be better than global)
        local_best = a.copy()
        local_best_val = cur_val

        # initial step size
        step = 0.5
        it = 0
        while time.time() < deadline:
            # --- gradient descent step ---------------------------------
            grad = _gradient_flatten(a)
            # decreasing step size (simple schedule)
            eta = step / (1.0 + 0.0005 * it)

            new_a = a - eta * grad
            new_a = np.maximum(new_a, 0.0)
            new_a = _normalize(new_a)

            new_val = _score(new_a)

            # accept if improvement; occasionally accept a worse step to
            # escape plateaus (tiny probability)
            if new_val < cur_val or rng.random() < 0.001:
                a = new_a
                cur_val = new_val

            # update local best
            if cur_val < local_best_val:
                local_best_val = cur_val
                local_best = a.copy()
                # also reflect into the global best
                _maybe_update(local_best)

            # --- occasional stochastic perturbations --------------------
            if it % 250 == 0:
                # random jitter on a few coordinates
                for _ in range(3):
                    idx = rng.integers(a.shape[0])
                    a[idx] = max(0.0, a[idx] + rng.normal(scale=0.01))
                a = _normalize(a)
                cur_val = _score(a)

                # occasional smoothing to damp high‑frequency artefacts
                if rng.random() < 0.3:
                    a = _smooth(a)
                    cur_val = _score(a)

            it += 1
            if it > 25000:          # safety cap per start
                break

        # final attempt to push the global best
        _maybe_update(local_best)

    # ------------------------------------------------------------------
    # 1) Try the previously discovered sequence if it exists.
    # ------------------------------------------------------------------
    try:
        if 'height_sequence_1' in globals():
            h1 = np.array(height_sequence_1, dtype=float)
            print("Starting from height_sequence_1 (len={})".format(h1.shape[0]))
            _optimize_from(h1)
    except Exception as exc:
        print("Failed to use height_sequence_1:", exc)

    # ------------------------------------------------------------------
    # 2) Systematically explore several lengths with different starts.
    # ------------------------------------------------------------------
    for n in target_lengths:
        if time.time() >= deadline:
            break
        print(f"=== Optimisation for length {n} ===")

        # a) uniform start
        uniform = np.full(n, 1.0 / n, dtype=float)
        _optimize_from(uniform)

        # b) random Dirichlet (Gamma) start
        dirichlet = rng.gamma(shape=1.0, scale=1.0, size=n)
        _optimize_from(dirichlet)

        # c) up‑sample the current best (if any) to the new length
        if best_seq is not None:
            old = np.array(best_seq, dtype=float)
            x_old = np.linspace(0.0, 1.0, old.shape[0])
            x_new = np.linspace(0.0, 1.0, n)
            upsamp = np.interp(x_new, x_old, old)
            _optimize_from(upsamp)

    # ------------------------------------------------------------------
    # Fallback: if nothing was found (should not happen), return a simple
    # uniform sequence of length 300.
    # ------------------------------------------------------------------
    if best_seq is None:
        print("Fallback to uniform sequence of length 300.")
        best_seq = [1.0 / 300.0] * 300

    print(f"Finished. Best score = {best_val:.6f}")
    return best_seq
