def gradient_step(seq, step=0.01): 
    n = len(seq)
    conv = np.convolve(seq, seq)
    k_max = int(np.argmax(conv))
    # compute gradient g
    g = np.zeros(n)
    for i in range(n):
        j = k_max - i
        if 0 <= j < n:
            if i != j:
                g[i] = 2 * seq[j]
            else:
                g[i] = seq[i]  # single term
        # else g remains zero
    total = np.sum(g) * step
    # Apply step: seq - step*g + (total/n) to keep sum constant.
    new_seq = np.array(seq) - step * g
    # compute amount removed
    removed = step * np.sum(g)
    new_seq += removed / n
    # Clip negative to zero (ensuring sum>0)
    new_seq = np.maximum(new_seq, 0.0)
    return new_seq.tolist()

def random_move(seq, sigma=0.01):
    new_seq = seq.copy()
    n = len(seq)
    # pick two indices
    i = np.random.randint(n)
    j = np.random.randint(n)
    if i == j: return new_seq
    # amount to transfer
    amount = np.random.randn() * sigma
    # Ensure nonnegative
    if amount > 0:
        amount = min(amount, new_seq[i])  # don't take more than i has
        new_seq[i] -= amount
        new_seq[j] += amount
    else:
        # negative amount means move from j to i
        amount = -amount
        amount = min(amount, new_seq[j])
        new_seq[j] -= amount
        new_seq[i] += amount
    return new_seq

"""
search_algorithm.py

A self‑contained candidate generator for the step‑function problem.
The algorithm combines a cheap parametric sweep (exponential,
power‑law, sinusoidal, Hann window, quadratic, etc.) with a few
simple local optimisation heuristics (random swaps, Gaussian
perturbations and a sub‑gradient step that reduces the peak of the
self‑convolution).  All candidates are evaluated with the provided
``evaluate_sequence`` function and the best one found before the time
budget expires is returned.

The implementation respects the required API:
    def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs) -> list[float]

and uses only standard scientific libraries (numpy, time, random).
"""

import time
import random
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities
# ----------------------------------------------------------------------


def arrange_alternating(arr):
    """Spread large values apart: place the sorted values alternately
    at the leftmost and rightmost free positions."""
    vals = np.sort(arr)[::-1]                     # descending
    n = len(vals)
    out = np.empty(n, dtype=vals.dtype)
    l, r = 0, n - 1
    toggle = True
    for v in vals:
        if toggle:
            out[l] = v
            l += 1
        else:
            out[r] = v
            r -= 1
        toggle = not toggle
    return out


def gradient_step(seq, step=0.01):
    """
    One sub‑gradient step for the objective
          f(a) = max(conv(a,a))
    (the factor 2·n / sum(a)^2 is omitted because it is scale‑invariant).

    The step moves the sequence away from the indices that realise the
    current maximum of the convolution and then re‑normalises the
    vector to keep the L1‑norm constant.
    """
    a = np.array(seq, dtype=float)
    n = len(a)

    # compute convolution and the index attaining the maximal value
    conv = np.convolve(a, a)
    k_max = int(np.argmax(conv))

    # sub‑gradient of the max‑convolution w.r.t. a
    g = np.zeros_like(a)
    for i in range(n):
        j = k_max - i
        if 0 <= j < n:
            if i != j:
                g[i] = 2.0 * a[j]
            else:
                g[i] = a[i]               # diagonal term appears only once
    # take a step and project back onto the simplex (non‑negative,
    # fixed L1‑norm)
    a_new = a - step * g
    a_new = np.maximum(a_new, 0.0)
    total = a_new.sum()
    if total < 1e-12:               # avoid division by zero – keep old vector
        return seq
    a_new /= total                  # normalise to L1 = 1 (scale does not matter)
    return a_new.tolist()


def random_swap(seq):
    """Return a copy of ``seq`` where two positions have been swapped."""
    a = np.array(seq, copy=True)
    n = len(a)
    i, j = np.random.randint(n, size=2)
    a[i], a[j] = a[j], a[i]
    return a.tolist()


def random_perturb(seq, scale=0.02):
    """Add a small Gaussian perturbation to a random element."""
    a = np.array(seq, copy=True)
    n = len(a)
    idx = np.random.randint(n)
    delta = np.random.randn() * scale * np.mean(a)
    a[idx] = max(0.0, a[idx] + delta)
    # renormalise to keep the L1‑norm comparable (scale does not affect the score)
    total = a.sum()
    if total > 0:
        a /= total
    return a.tolist()


# ----------------------------------------------------------------------
# Main candidate generator
# ----------------------------------------------------------------------


def propose_candidate(seed=0, budget_s=1000, **kwargs):
    """
    Search for a sequence of non‑negative heights that minimises
    ``evaluate_sequence``.  The function respects the time budget,
    prints periodic progress messages and finally returns the best
    sequence discovered.
    """
    np.random.seed(seed)
    random.seed(seed)

    deadline = time.time() + budget_s - 0.5          # small safety margin

    # ---- initialise with the known construction if it exists ----------
    try:
        best_sequence = list(height_sequence_1)      # provided globally
        best_score = evaluate_sequence(best_sequence)
    except Exception:
        best_sequence = None
        best_score = float("inf")

    # ------------------------------------------------------------------
    # 1) Coarse parametric sweep (different analytic shapes)
    # ------------------------------------------------------------------
    def try_candidate(arr, descr):
        """Helper that evaluates *arr* and updates the global best."""
        nonlocal best_sequence, best_score
        try:
            sc = evaluate_sequence(arr.tolist())
        except Exception:
            return
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_sequence = arr.tolist()
            print(f"  new best {best_score:.6f}  len={len(arr)}  {descr}")

    lengths = [120, 200, 300, 500, 800, 1200, 1800]   # moderate sizes
    alphas = np.logspace(-3, 0.5, num=12)              # for exponentials
    powers = np.linspace(0.1, 4.0, num=12)             # for power‑law

    for n in lengths:
        # ---------- uniform (baseline) ----------
        try_candidate(np.ones(n), "uniform")

        # ---------- exponential decay ----------
        for alpha in alphas:
            arr = np.exp(-alpha * np.arange(n))
            try_candidate(arr, f"exp(alpha={alpha:.3g})")
            # also test the "alternating" ordering
            try_candidate(arrange_alternating(arr), f"exp(alpha={alpha:.3g}) + alt")

        # ---------- power‑law ----------
        idx = np.arange(1, n + 1)
        for p in powers:
            arr = idx.astype(float) ** (-p)
            try_candidate(arr, f"power(p={p:.2f})")
            try_candidate(arrange_alternating(arr), f"power(p={p:.2f}) + alt")

        # ---------- half‑sine (smooth rise) ----------
        arr = np.sin(np.pi * (np.arange(n) + 0.5) / (2.0 * n))
        try_candidate(arr, "half‑sine")
        try_candidate(arrange_alternating(arr), "half‑sine + alt")

        # ---------- Hann window ----------
        if n > 1:
            arr = 0.5 * (1.0 - np.cos(2.0 * np.pi * np.arange(n) / (n - 1)))
            try_candidate(arr, "Hann")
            try_candidate(arrange_alternating(arr), "Hann + alt")

        # ---------- quadratic (parabolic) ----------
        arr = np.arange(n, dtype=float) * (n - np.arange(n))
        try_candidate(arr, "quadratic")
        try_candidate(arrange_alternating(arr), "quadratic + alt")

        # ---------- U‑shaped (large at the ends) ----------
        i = np.arange(n, dtype=float)
        arr = 1.0 / (i + 1.0) + 1.0 / (n - i)
        try_candidate(arr, "U‑shape")
        try_candidate(arrange_alternating(arr), "U‑shape + alt")

    # If nothing beat the initial construction (unlikely) fall back to
    # a random normalised vector.
    if best_sequence is None:
        n = 200
        best_sequence = np.random.random(n).tolist()
        best_score = evaluate_sequence(best_sequence)

    print(f"=== sweep finished, best score = {best_score:.6f} ===")

    # ------------------------------------------------------------------
    # 2) Local improvement phase (random swaps, perturbations,
    #    and occasional sub‑gradient steps)
    # ------------------------------------------------------------------
    curr_seq = np.array(best_sequence, dtype=float)
    curr_score = best_score

    # small utility to keep the best record
    def maybe_update(seq, sc):
        nonlocal best_sequence, best_score, curr_seq, curr_score
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_sequence = seq[:]
            print(f"*** global improvement {best_score:.6f} ({len(seq)} elems) ***")
        if sc < curr_score:
            curr_seq = np.array(seq, copy=True)
            curr_score = sc

    while time.time() < deadline:
        # 70 % random swap, 30 % Gaussian perturbation
        if np.random.rand() < 0.7:
            cand = random_swap(curr_seq.tolist())
        else:
            cand = random_perturb(curr_seq.tolist(), scale=0.02)

        try:
            cand_score = evaluate_sequence(cand)
        except Exception:
            continue
        maybe_update(cand, cand_score)

        # occasional sub‑gradient step (every ~20 iterations)
        if np.random.rand() < 0.05:
            grad_seq = gradient_step(curr_seq.tolist(), step=0.01)
            try:
                grad_score = evaluate_sequence(grad_seq)
            except Exception:
                continue
            maybe_update(grad_seq, grad_score)

        # occasional random restart (tiny probability)
        if np.random.rand() < 0.001:
            n = random.randint(80, 1500)
            rnd = np.random.random(n)
            rnd /= rnd.sum()          # normalise – scale does not matter
            cand_score = evaluate_sequence(rnd.tolist())
            maybe_update(rnd.tolist(), cand_score)

    print(f"=== search complete, final best score = {best_score:.6f} ===")
    # ``best_sequence`` is already a plain Python list of floats
    return best_sequence
