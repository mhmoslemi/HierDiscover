            # normalize sum to 1
            s = a0.sum()
            if s > 0:

                a = a0
    best_a = a.copy()
    best_score = evaluate_sequence(best_a.tolist())
    # print baseline
    print(f"Initial score: {best_score:.6f} with n={n}")

    # parameters
    epsilon = 1e-12
    step_factor = 0.2  # can adjust
    # schedule for step size shrink after each iteration?
    iteration = 0
    while time.time() < deadline:
        iteration += 1
        conv = np.convolve(a, a)
        M = conv.max()
        # compute active set indices:
        # use tolerance relative to M.
        # If M is zero (should not), break.
        if M == 0:
            break
        active = np.where(conv >= M - epsilon)[0]

        # compute gradient contributions
        g = np.zeros_like(a)
        # vectorized loop
        n_val = n
        for k in active:
            i_start = k - (n_val - 1)
            if i_start < 0:
                i_start = 0
            i_end = k
            if i_end > n_val - 1:
                i_end = n_val - 1
            # slice and reversed slice addition
            # using view
            slice_a = a[i_start:i_end+1]
            g[i_start:i_end+1] += slice_a[::-1]
        # direction d = mean(g) - g (sum zero)
        mean_g = g.mean()
        d = mean_g - g
        # step size: adapt based on norm of d
        max_abs = np.max(np.abs(d))
        if max_abs == 0:
            # gradient zero => possible optimum; break
            break
        # step size schedule: decreasing with iteration
        # Use step_factor * (1 / sqrt(iteration))
        eta = step_factor / np.sqrt(iteration) / max_abs
        # update a
        a_new = a + eta * d
        # enforce nonnegativity
        np.maximum(a_new, 0, out=a_new)
        # renormalize sum to 1 (scale invariance)
        sum_a = a_new.sum()
        if sum_a == 0:
            # fallback: reinitialize uniform
            a_new = np.full(n, 1.0/n)
        else:
            a_new *= (1.0 / sum_a)  # sum becomes 1

        a = a_new

        # evaluate
        try:
            score = evaluate_sequence(a.tolist())
        except Exception as e:
            # fallback: continue
            continue

        if score < best_score:

            a /= a.sum()
    # end while
    print(f"Finished after {iteration} iterations, best score = {best_score:.6f}")

def propose_candidate(seed=12345, budget_s=1000, **kwargs):

    np.random.seed(seed)

    def compute_score(seq):

        M = conv.max()
        S = seq.sum()
        if S < 0.01:
            return float('inf')
        return 2.0 * n * M / (S * S)
    # initialize best
    best_seq = None
    best_score = float('inf')
    # start with random sequence
    a = np.random.rand(n)
    a = a / a.sum()
    # few restarts possible
    iter_count = 0
    no_improve_steps = 0
    while time.time() < deadline:
        # compute convolution and active set
        conv = np.convolve(a, a)
        M = conv.max()
        # compute score
        score = 2.0 * n * M / (a.sum() ** 2) # sum = 1 anyway
        if score < best_score:
            best_score = score
            best_seq = a.copy()

            no_improve_steps = 0
        else:
            no_improve_steps += 1
        # gradient step
        active = np.where(conv >= M - 1e-12)[0]
        # compute gradient contributions
        g = np.zeros_like(a)
        for k in active:
            i_start = k - (n - 1)
            if i_start < 0:
                i_start = 0
            i_end = k
            if i_end > n - 1:
                i_end = n - 1

The evaluation metric is   2·n·max(conv(a,a)) / (sum(a)^2) ,  which is scale invariant.
The goal is to reshape a non‑negative vector a (length n) so that its self‑convolution
becomes as flat as possible, i.e. the maximum entry is close to the average.
Starting from a random (non‑uniform) vector we repeatedly
  * compute the convolution and locate the active (maximal) indices,
  * compute the sub‑gradient of the maximal convolution entry,
  * move a in the direction that equalises these gradients while preserving the
    total sum (projection onto the simplex),
  * occasionally add jitter or restart if progress stalls.

The algorithm is inexpensive (O(n) per iteration) and runs for the whole
time budget, returning the best sequence seen.

        if cur_val < best_val:
