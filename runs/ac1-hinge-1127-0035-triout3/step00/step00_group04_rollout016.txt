import numpy as np
import time

def compute_score(seq):
    """Return the objective 2*n*max(conv)/sum^2 for a non‑negative sequence."""
    n = len(seq)
    if n == 0:
        return np.inf
    conv = np.convolve(seq, seq)
    max_b = np.max(conv)
    s = np.sum(seq)
    if s < 1e-12:
        return np.inf
    return 2.0 * n * max_b / (s * s)

def exponential_sequence(n, c):
    """Return a normalized decreasing exponential a_i = r**i where r = exp(-c/n)."""
    r = np.exp(-c / n)
    seq = r ** np.arange(n, dtype=float)
    seq /= np.sum(seq)
    return seq

def power_sequence(n, alpha):
    """Return a normalized decreasing power law a_i = (n-i)**alpha."""
    idx = np.arange(n, dtype=float)
    seq = (n - idx) ** alpha
    seq /= np.sum(seq)
    return seq

    while time.time() < deadline:

        elapsed = time.time() - (deadline - total_time)
        t_frac = elapsed / total_time

        # compute the current max‑convolution index

        k_max = int(np.argmax(conv))

        # find a pair (i,j) with i+j=k_max and both entries positive

"""
Search for a non‑negative step‑function that minimises

    2·n·max(conv(f,f)) / (sum f)^2

The algorithm:
1.  Produce a family of exponentially decaying sequences a_i = r**i,
    where r = exp(-c/n).  For each length n and decay‑parameter c
    the analytical expression of the convolution shows that this
    family is already close to optimal (objective ≈ 1.8).  We scan a
    small grid of (n,c) to obtain the best raw candidate.
2.  Starting from that candidate we run a cheap simulated‑annealing
    hill‑climber.  In each iteration we locate the convolution index
    that attains the current maximum, pick a pair (i,j) belonging to
    that diagonal and transfer a tiny amount of mass from i and j
    to two random positions.  After the perturbation we project back
    onto the probability simplex (non‑negative, sum = 1).  The move
    is accepted if the objective improves, otherwise with a
    Metropolis probability that gradually cools.  The loop runs
    until the allotted time expires.
The returned list of floats is ready for ``evaluate_sequence``.
"""

def compute_score(seq):
    """Return the objective 2·n·max(conv)/sum² for a non‑negative sequence."""

    max_b = np.max(conv)
    s = np.sum(seq)

    Moves a small amount of mass away from the diagonal that currently
    yields the maximal convolution value.
