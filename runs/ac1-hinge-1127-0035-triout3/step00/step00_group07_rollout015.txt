def project_simplex(v):
    # v is 1D np array
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0)
    return w
```

"""
Propose a step‑function (list of non‑negative heights) that minimises
evaluate_sequence.  The optimiser combines a simple simulated‑annealing
scheme with a gradient step that pushes mass away from the convolution
indices currently attaining the maximum value.  The search is performed
for a variety of lengths; the best sequence found before the time budget
expires is returned.
"""

import time
import numpy as np

# --------------------------------------------------------------
# Helper: projection onto the probability simplex (sum = 1, non‑neg.)
# --------------------------------------------------------------
def _project_onto_simplex(v):
    """Project a 1‑D numpy array onto the simplex {x>=0, sum(x)=1}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(n) + 1
    # Find rho = max{i | u_i - cssv_i / i > 0}
    rho = ind[u - cssv / ind > 0][-1]
    theta = cssv[rho - 1] / rho
    w = np.maximum(v - theta, 0.0)
    return w

    * For each length perform a short simulated‑annealing run:
        - random “mass‑transfer’’ moves keep the total sum = 1,
        - a gradient step that pushes probability mass away from
          the convolution entries that currently attain the maximum.
    * The current best sequence (lowest score) is continually updated.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 2  # a tiny safety margin

    best_seq = None
    best_score = float("inf")

    # lengths to explore – short runs on many lengths, longer runs on the largest
    lengths = [12, 20, 30, 40, 60, 80, 120, 180, 240, 300]

    # main loop over lengths
    for n in lengths:
        # perform a few restarts for each length
        for restart in range(5):
            # start from the uniform distribution and add a tiny perturbation
            a = np.full(n, 1.0 / n, dtype=float)
            a += 0.001 * np.random.randn(n)
            a = np.maximum(a, 0.0)
            a = _project_onto_simplex(a)          # guarantees sum == 1

            # initial score
            conv = np.convolve(a, a)
            cur_score = 2.0 * n * conv.max() / (a.sum() ** 2)

            # simulated‑annealing / gradient loop
            temperature = 1.0          # initial temperature for Metropolis step
            step_frac = 0.05           # size of random mass‑transfer moves
            grad_step = 0.02           # size of gradient‑descent moves

            it = 0
            while time.time() < deadline:
                it += 1

                # ----- occasional random perturbation (to escape plateaus) -----
                if np.random.rand() < 0.02:                     # 2 % chance
                    i, j = np.random.choice(n, 2, replace=False)
                    # move a random fraction of a[i] to a[j]
                    delta = step_frac * a[i] * np.random.rand()
                    delta = min(delta, a[i])
                    a[i] -= delta
                    a[j] += delta
                    # re‑normalise (the move preserves the sum but clipping may break it)
                    a = _project_onto_simplex(a)

                else:
                    # ----- gradient step that reduces the current maximum -----
                    conv = np.convolve(a, a)
                    max_val = conv.max()
                    # indices where the convolution is (almost) maximal
                    tol = max_val * 1e-12
                    mask = (conv >= max_val - tol).astype(float)

                    # gradient of the max‑convolution (sub‑gradient)
                    # g_i = 2 * sum_k mask_k * a_{k-i}
                    rev_a = a[::-1]
                    # convolution of mask with reversed a gives the needed sums
                    tmp = np.convolve(mask, rev_a)
                    grad = 2.0 * tmp[n - 1 : n - 1 + n]       # length n

                    # take a small step opposite to the gradient
                    a_new = a - grad_step * grad
                    a_new = np.maximum(a_new, 0.0)
                    a_new = _project_onto_simplex(a_new)

                    # replace only if the move improves (or by Metropolis rule)
                    conv_new = np.convolve(a_new, a_new)
                    new_score = 2.0 * n * conv_new.max() / (a_new.sum() ** 2)

                    accept = False
                    if new_score < cur_score:
                        accept = True
                    else:
                        # Metropolis acceptance with current temperature
                        prob = np.exp(-(new_score - cur_score) / max(temperature, 1e-12))
                        if np.random.rand() < prob:
                            accept = True

                    if accept:
                        a = a_new
                        cur_score = new_score
                        # record the globally best sequence
                        if cur_score < best_score:
                            best_score = cur_score
                            best_seq = a.tolist()
                            print(f"[{time.strftime('%H:%M:%S')}] "
                                  f"New best: {best_score:.6f} (n={n}, it={it})")

                # temperature and step‑size schedules
                temperature *= 0.99996
                if it % 500 == 0:
                    step_frac *= 0.95
                    grad_step *= 0.97

                # early exit if we already beat the target
                if best_score < 1.503:
                    print("Target reached – exiting early.")
                    return best_seq

    # Fallback – if nothing was found (should never happen) return a simple uniform list
    if best_seq is None:
        best_seq = [1.0 / 10] * 10
    return best_seq
