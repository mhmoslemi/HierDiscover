def solve_convolution_lp(f_seq, rhs):
    n = len(f_seq)
    c = -np.ones(n)  # maximize sum -> minimize -sum

    # Build A_ub, each row corresponds to k offset constraints: sum_i f_i * b_{k-i} <= rhs.
    # row_j = f_i for i where j = k - i.
    A = []
    b = []
    for k in range(2*n-1):
        row = np.zeros(n)
        # i loops
        start_i = max(0, k-(n-1))
        end_i = min(n-1, k)
        for i in range(start_i, end_i+1):
            j = k - i
            row[j] = f_seq[i]
        A.append(row)
        b.append(rhs)
    # Nonnegativity constraints: -b <= 0
    A_nonneg = -np.eye(n)
    b_nonneg = np.zeros(n)
    A = np.vstack([A, A_nonneg])
    b = np.concatenate([b, b_nonneg])
    result = linprog(c, A_ub=A, b_ub=b, method='highs')
    if result.success:
        return result.x
    else:
        return None

def max_convolution(x):
    n=len(x)
    # Use real FFT for speed
    # pad to length next power of 2 >= 2n-1
    size=1
    while size < 2*n-1:
        size <<= 1
    fx=np.fft.rfft(x, n=size)
    conv=np.fft.irfft(fx*fx, n=size)
    # conv length size, only first 2n-1 entries relevant
    return np.max(conv[:2*n-1])

def propose_candidate(seed=42, budget_s=1000, **kwargs):

    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.1
    n = kwargs.get('length', 2000)  # maybe they provide length?
    # we can decide length approx 2000 to target <1.5.
    # initialize uniform probabilities sum 1
    p = np.full(n, 1.0/n)
    best_score = evaluate_sequence((p*np.sqrt(2*n)).tolist())
    best_p = p.copy()
    # parameters
    t = 0.1
    no_improve_iters = 0
    while time.time() < deadline:
        # compute convolution max
        conv = max_convolution_fft(p)  # returns array of conv
        M = conv.max()
        # find indices with minimal p
        i_min = np.argmin(p)
        # create spike g
        g = np.zeros_like(p)
        g[i_min] = 1.0  # spike with total mass 1
        # convex combination
        new_p = (1 - t) * p + t * g
        # renorm to sum=1 (still sum=1 because both p and g sum =1)
        # compute new score using evaluate_sequence
        new_seq = (new_p * np.sqrt(2*n)).tolist()
        new_score = evaluate_sequence(new_seq)
        if new_score < best_score:
            best_score = new_score
            best_p = new_p.copy()
            p = new_p
            no_improve_iters = 0

            t *= 0.5

                p = p / p.sum()

def max_convolution_fft(p):
    n = len(p)
    size = 1
    while size < 2*n - 1:
        size <<= 1
    fft = np.fft.rfft(p, n=size)
    conv = np.fft.irfft(fft*fft, n=size)
    return conv[:2*n-1]

"""
Propose a candidate step‑function sequence.

The objective 2 * n * max(conv(a, a)) / (sum(a)**2) is scale‑invariant.
If we work with a probability vector p (sum(p) = 1) the score becomes
    score(p) = 2 * n * max(conv(p, p))

The algorithm keeps a probability vector and iteratively reduces the
maximum of its self‑convolution:

1. Start from a random probability distribution.
2. Compute the full convolution by FFT and locate the indices where the
   maximum is attained (the “active” set).
3. For each active offset k the sub‑gradient of max(conv) w.r.t. p_i is
   2 * p_{k‑i} (zero if the index is out of range).  Summing over all
   active k gives a cheap sub‑gradient g.
4. Transfer a fraction of the mass from the index with the largest
   gradient (most harmful for the current max) to the index with the
   smallest gradient (least harmful).  The step size is adapted
   automatically – it is increased after a successful move and reduced
   otherwise.
5. After each move recompute the score using the fast FFT convolution.
   Keep the best distribution seen so far.
6. When the improvement stalls a small random perturbation is applied
   to escape possible local minima.
7. Finally the best probability vector is scaled back to a sum of
   sqrt(2 n) (the normalization used by evaluate_sequence) and returned.

The method uses only O(n log n) work per iteration, works for n of a few
thousand, and typically drives the score well below the uniform value
2.0 within the allotted time.
"""

import numpy as np
import time

def _max_convolution_fft(p: np.ndarray) -> np.ndarray:
    """Return the full convolution p * p using a real FFT (fast)."""
    n = p.shape[0]
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    # real FFT of the zero‑padded vector
    fp = np.fft.rfft(p, n=size)
    conv = np.fft.irfft(fp * fp, n=size)
    # keep only the first 2n‑1 entries (the rest are zero‑padding artefacts)
    conv = conv[: 2 * n - 1]
    # tiny negative values caused by rounding are set to zero
    np.maximum(conv, 0, out=conv)
    return conv


def _subgradient(p: np.ndarray, conv: np.ndarray, max_val: float, tol: float = 1e-12) -> np.ndarray:
    """
    Compute a sub‑gradient of max(conv(p, p)) at the current p.
    Only offsets where conv == max_val (up to tol) are considered.
    """
    n = p.shape[0]
    # active offsets (where the maximum is attained)
    active = np.where(np.abs(conv - max_val) <= max(tol, max_val * 1e-12))[0]
    grad = np.zeros_like(p)
    # For each active offset k, the derivative w.r.t. p_i is 2 * p_{k-i}
    for k in active:
        i_min = max(0, k - (n - 1))
        i_max = min(n - 1, k)
        # vectorised contribution for this k
        # j = k - i  => p_j = p[k - i]
        indices_i = np.arange(i_min, i_max + 1, dtype=np.intp)
        indices_j = k - indices_i
        grad[indices_i] += 2.0 * p[indices_j]
    return grad


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs) -> list[float]:
    """Return a step‑function (list of non‑negative heights) with a low score."""
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5   # a little safety margin

    # length of the sequence – a few thousand give a good trade‑off
    n = kwargs.get("length", 2000)
    # start from a random probability vector
    p = np.random.rand(n)
    p /= p.sum()

    # best solution bookkeeping
    best_p = p.copy()
    # score = 2 * n * max(conv(p, p))  (p is a probability vector)
    conv = _max_convolution_fft(p)
    best_score = 2.0 * n * conv.max()

    # step size for mass transfer (fraction of source weight)
    step_frac = 0.2
    # counters for stagnation handling
    no_improve = 0

    while time.time() < deadline:
        conv = _max_convolution_fft(p)
        max_val = conv.max()
        score = 2.0 * n * max_val

        if score < best_score:
            best_score = score
            best_p = p.copy()
            no_improve = 0
        else:
            no_improve += 1

        # compute a sub‑gradient of the max‑convolution
        grad = _subgradient(p, conv, max_val)

        # source = index that contributes most to the current max,
        # destination = index that contributes least.
        src = int(np.argmax(grad))
        dst = int(np.argmin(grad))

        # move a fraction of the source mass to the destination
        delta = step_frac * p[src]
        if delta <= 0:
            # nothing to move – random perturbation
            idx = np.random.randint(n)
            p[idx] = max(0.0, p[idx] + np.random.randn() * 0.005)
            p = np.clip(p, 0, None)
            if p.sum() == 0:
                p = np.full(n, 1.0 / n)
            else:
                p /= p.sum()
            step_frac = 0.2
            continue

        new_p = p.copy()
        new_p[src] -= delta
        new_p[dst] += delta

        # keep everything non‑negative (numerical safety)
        if new_p[src] < -1e-12:
            # reject this move
            step_frac *= 0.5
            continue
        new_p = np.clip(new_p, 0, None)

        # recompute the max‑convolution for the trial vector
        new_conv = _max_convolution_fft(new_p)
        new_score = 2.0 * n * new_conv.max()

        if new_score < score:
            # successful improvement – keep the move and increase step size a bit
            p = new_p
            step_frac = min(step_frac * 1.05, 0.5)
            no_improve = 0
        else:
            # move was not beneficial – shrink the step size
            step_frac *= 0.5
            if step_frac < 1e-4:
                # perform a small random perturbation to escape flat regions
                idx = np.random.randint(n)
                p[idx] = max(0.0, p[idx] + np.random.randn() * 0.01)
                p = np.clip(p, 0, None)
                if p.sum() == 0:
                    p = np.full(n, 1.0 / n)
                else:
                    p /= p.sum()
                step_frac = 0.2

        # occasional full‑vector perturbation if we are stuck for a while
        if no_improve > 200:
            p += np.random.randn(n) * 0.005
            p = np.clip(p, 0, None)
            if p.sum() == 0:
                p = np.full(n, 1.0 / n)
            else:
                p /= p.sum()
            step_frac = 0.2
            no_improve = 0

    # Scale the best probability vector back to the normalization used by
    # evaluate_sequence: sum = sqrt(2 * n)
    scale = np.sqrt(2.0 * n)
    final_seq = (best_p * scale).tolist()
    # The evaluator will cap values to [0, 1000] and enforce non‑negativity,
    # but the construction already respects those limits.
    return final_seq
