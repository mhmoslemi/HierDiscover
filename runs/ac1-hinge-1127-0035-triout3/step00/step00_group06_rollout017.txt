"""
Optimization of the step‑function evaluation.

The objective to minimise is
    evaluate = 2 * n * max_k (a * a)[k] / (sum a)^2
which is scale invariant.  Hence we only need to search for a good
*shape* of the non‑negative vector `a`.  The algorithm explores several
parameterised families (exponential, power‑law, Gaussian, two‑phase
exponential) together with random Dirichlet vectors.  For each candidate
the cheap `evaluate_sequence` function is called; the best sequence seen
so far is kept.  After the global sweep a simple stochastic local
search (multiplicative perturbations of a small random subset of entries)
is performed to try to refine the current best candidate.

The whole search stops after the allocated time budget (default 1000 s)
and returns the best sequence found.
"""

import time
import numpy as np

def _generate_exponential(n: int, c: float) -> list[float]:
    """a_i = (1 - c/n)**i   (c > 0)."""
    eps = c / n
    r = 1.0 - eps
    # np.arange is faster than a Python loop
    return list(r ** np.arange(n, dtype=float))


def _generate_powerlaw(n: int, alpha: float) -> list[float]:
    """a_i = (i+1)^(-alpha)."""
    idx = np.arange(1, n + 1, dtype=float)
    return list(idx ** (-alpha))


def _generate_gaussian(n: int, sigma: float) -> list[float]:
    """a_i = exp( - (i/sigma)^2 )."""
    idx = np.arange(n, dtype=float)
    return list(np.exp(- (idx / sigma) ** 2))


def _generate_double_exponential(
    n: int, k_frac: float, a: float, b: float
) -> list[float]:
    """
    Two‑phase exponential:
        a_i = exp(-a*i)          for i < K
        a_i = exp(-b*i)          for i >= K
    where K = int(k_frac * n).
    """
    K = int(k_frac * n)
    idx = np.arange(n, dtype=float)
    seq = np.empty(n, dtype=float)
    seq[:K] = np.exp(-a * idx[:K])
    seq[K:] = np.exp(-b * idx[K:])
    return list(seq)


def _random_dirichlet(n: int, rng: np.random.Generator) -> list[float]:
    """Sample a random non‑negative vector (Dirichlet with alpha=1)."""
    return list(rng.random(n))


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a sequence of non‑negative heights that gives a low value
    of `evaluate_sequence`.  The function runs for at most `budget_s`
    seconds and returns the best sequence found.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5   # a small safety margin

    # ------------------------------------------------------------------
    # 1.   Initialise best candidate (use the provided construction if
    #      it exists, otherwise start from a random vector).
    # ------------------------------------------------------------------
    if "height_sequence_1" in globals():
        best_seq = list(height_sequence_1)          # type: ignore
        try:
            best_score = evaluate_sequence(best_seq)
        except Exception:
            best_score = np.inf
    else:
        # fallback: a short random vector
        best_seq = _random_dirichlet(50, np.random.default_rng())
        best_score = evaluate_sequence(best_seq)

    print(f"Initial score: {best_score:.6f} (len={len(best_seq)})")

    # ------------------------------------------------------------------
    # 2.   Global exploration of several parametric families.
    # ------------------------------------------------------------------
    lengths = [200, 500, 1000, 1500, 2000]          # a few different n
    rng = np.random.default_rng(seed + 1)

    # helper to evaluate and possibly update the incumbent
    def _try(seq):
        nonlocal best_seq, best_score
        if time.time() > deadline:
            return False
        try:
            score = evaluate_sequence(seq)
        except Exception:
            return True                     # continue search
        if score < best_score:
            best_score = score
            best_seq = seq.copy()
            print(f"  New best: {best_score:.6f} (len={len(seq)})")
        return True

    # --- 2‑phase sweep -------------------------------------------------
    for n in lengths:
        # exponential family (c roughly the useful range)
        for c in np.linspace(0.2, 4.0, 20):
            if not _try(_generate_exponential(n, c)):
                break

        # power‑law family
        for alpha in np.linspace(0.2, 3.0, 20):
            if not _try(_generate_powerlaw(n, alpha)):
                break

        # Gaussian family (sigma around the length)
        for sigma in np.linspace(0.8 * n, 1.2 * n, 12):
            if not _try(_generate_gaussian(n, sigma)):
                break

        # two‑phase exponential (coarse grid)
        for k_frac in (0.3, 0.5, 0.7):
            for a_param in (0.001, 0.005, 0.010):
                for b_param in (0.0005, 0.001, 0.002):
                    if not _try(_generate_double_exponential(n, k_frac, a_param, b_param)):
                        break

        # a few random Dirichlet vectors
        for _ in range(5):
            if not _try(_random_dirichlet(n, rng)):
                break

        if time.time() > deadline:
            break

    # ------------------------------------------------------------------
    # 3.   Stochastic local refinement.
    # ------------------------------------------------------------------
    curr_seq = best_seq.copy()
    n = len(curr_seq)
    # pre‑compute the number of indices mutated each step (5 % of n, at least 1)
    perturb_size = max(1, n // 20)

    while time.time() < deadline:
        new_seq = curr_seq.copy()
        # pick a random subset of positions and apply a small multiplicative noise
        idxs = np.random.choice(n, size=perturb_size, replace=False)
        noise = np.exp(np.random.randn(perturb_size) * 0.05)   # ≈5 % log‑normal noise
        for i, factor in zip(idxs, noise):
            new_seq[i] *= factor

        # evaluate the perturbed candidate
        try:
            new_score = evaluate_sequence(new_seq)
        except Exception:
            continue

        # keep globally best solution
        if new_score < best_score:
            best_score = new_score
            best_seq = new_seq.copy()
            print(f"  Improved by perturbation: {best_score:.6f}")

        # simple hill climbing acceptance (always accept improvements,
        # occasionally accept a worse move to escape plateaus)
        try:
            cur_score = evaluate_sequence(curr_seq)
        except Exception:
            cur_score = np.inf
        if new_score < cur_score:
            curr_seq = new_seq
        elif np.random.rand() < 0.01:          # occasional random walk
            curr_seq = new_seq

    print(f"Finished. Best score = {best_score:.6f} (len={len(best_seq)})")
    return best_seq
