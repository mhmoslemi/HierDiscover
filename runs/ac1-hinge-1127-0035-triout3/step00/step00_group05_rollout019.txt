def simplex_projection(v):
    # v is ndarray of any shape
    # Returns projection onto the probability simplex {x>=0, sum=1}
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(len(v)) + 1
    cond = u - cssv / ind > 0
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0)
    return w

    import numpy as np, time
    np.random.seed(seed)
    deadline = time.time() + budget_s - 5
    # Choose length n
    n = 2000
    # Initial a: sample from Beta(0.5,0.5) to get U-shaped distribution.
    a = np.random.beta(0.5, 0.5, size=n)
    a = a / a.sum()
    best_a = a.copy()
    best_score = 2 * n * np.max(np.convolve(a, a))
    # Ensure best_score is correct (maybe compute via evaluate_sequence later)
    iter_count = 0
    while time.time() < deadline:

        # Use np.fft to compute linear convolution
        fa = np.fft.rfft(a, n*2-1)
        b = np.fft.irfft(fa * fa, n*2-1)
        # b length 2n-1
        max_val = b.max()
        # early termination if already near target
        if best_score <= 1.503:
            break
        # identify near-max indices
        thresh = max_val * 0.995
        idxs = np.where(b >= thresh)[0]
        # compute gradient
        grad = np.zeros_like(a)
        for k in idxs:
            # For i from max(0, k-(n-1)) to min(n-1, k):
            i_start = max(0, k - (n-1))
            i_end = min(n-1, k)
            # vector of i indices
            # For each i, j = k-i
            # grad[i] += 2*a[j]

            # compute slice of a_j
            # j range = k - i_start ... k - i_end (reverse)
            # Actually easier: loop i (since range length maybe up to n).
            for i in range(i_start, i_end+1):
                j = k - i
                grad[i] += 2.0 * a[j]
        # step size schedule
        step = 0.05 * (1.0 - iter_count / 5000.0)  # decreasing
        a_new = a - step * grad
        # clip
        a_new = np.maximum(a_new, 0.0)
        # project onto simplex
        a_new = simplex_projection(a_new)
        a = a_new
        # compute new max
        fa = np.fft.rfft(a, n*2-1)
        b = np.fft.irfft(fa * fa, n*2-1)
        max_val = b.max()
        score = 2 * n * max_val  # sum is 1
        if score < best_score:
            best_score = score
            best_a = a.copy()
        iter_count += 1
        # occasional random jitter to escape
        if iter_count % 1000 == 0:
            # add small Gaussian noise

import cvxpy as cp
def solve_gp(n):
    a = cp.Variable(n, pos=True)
    M = cp.Variable(pos=True)
    constraints = []
    # using convolution: For each k in range(2*n-1):
    for k in range(2*n-1):
        terms = []
        i_min = max(0, k-(n-1))
        i_max = min(n-1, k)
        # For each i in that range:
        for i in range(i_min, i_max+1):
            j = k - i
            terms.append(cp.multiply(a[i], a[j]))
        # sum of monomials:
        constraints.append(cp.sum(terms) <= M)
    # objective:
    prob = cp.Problem(cp.Minimize(M), constraints)
    prob.solve(gp=True, solver=cp.ECOS, warm_start=True)
    # retrieve a values.
    a_vals = a.value
    # Normalize to sum 1:
    a_vals = np.maximum(a_vals, 0)
    a_vals = a_vals / a_vals.sum()
    return a_vals

def propose_candidate(seed=123, budget_s=1000, **kwargs):
    import numpy as np, time, cvxpy as cp
    np.random.seed(seed)
    deadline = time.time() + budget_s - 5

    # Choose length
    n = 400  # moderate size to keep solver tractable
    # Build GP
    a = cp.Variable(n, pos=True)
    M = cp.Variable(pos=True)

    constraints = []
    for k in range(2*n - 1):
        i_min = max(0, k - (n - 1))
        i_max = min(n - 1, k)
        terms = [cp.multiply(a[i], a[k - i]) for i in range(i_min, i_max + 1)]
        constraints.append(cp.sum(terms) <= M)

    prob = cp.Problem(cp.Minimize(M), constraints)
    # Use ECOS solver with reasonable max_iters
    prob.solve(gp=True, solver=cp.ECOS, verbose=False, max_iters=5000)

    if a.value is None:
        # fallback to uniform
        seq = np.ones(n) / n
    else:
        seq = np.array(a.value, dtype=float)
        # scale to sum 1 (any positive scaling leaves the ratio unchanged)
        total = seq.sum()
        if total <= 0:
            seq = np.ones(n) / n
        else:
            seq = seq / total

    # Optionally clamp to [0,1000] just to satisfy evaluate_sequence constraints
    seq = np.clip(seq, 0.0, 1000.0)

    # Return as list
    return seq.tolist()

def propose_candidate(seed=0, budget_s=1000):
    import numpy as np, time
    np.random.seed(seed)
    deadline = time.time() + budget_s - 5

    n = 2000  # length
    # start with uniform distribution
    a = np.full(n, 1.0 / n)
    # compute initial score
    def compute_score(vec):
        # sum is 1
        fa = np.fft.rfft(vec, n*2-1)
        b = np.fft.irfft(fa * fa, n*2-1)
        return float(2 * n * np.max(b.real))

    best_a = a.copy()
    best_score = compute_score(a)

    # initial temperature
    temp0 = 0.1
    iteration = 0
    while time.time() < deadline:
        # temperature schedule
        T = temp0 * (1 - iteration / 500000)  # linear decay
        # choose donor idx (high mass) maybe random weighted by a
        i = np.random.randint(n)
        # donor should have some mass to give
        if a[i] < 1e-8:
            continue
        # choose receiver j, preferably far from i
        j = np.random.randint(n)
        if i == j:
            continue
        # propose delta
        delta = a[i] * 0.1  # move 10% of donor mass
        # ensure not too small
        if delta < 1e-8:
            continue
        a_new = a.copy()
        a_new[i] -= delta
        a_new[j] += delta
        # compute score
        new_score = compute_score(a_new)
        # accept if improvement or with probability
        if new_score < best_score:
            best_score = new_score
            best_a = a_new.copy()
            a = a_new
        else:
            # metropolis acceptance
            if T > 0 and np.random.rand() < np.exp(-(new_score - best_score) / T):
                a = a_new
        iteration += 1
        # occasional restart with random small perturbation
        if iteration % 10000 == 0:
            # small random noise
            noise = np.random.randn(n) * 1e-4
            a = np.maximum(a + noise, 0)
            a = a / a.sum()
            # recompute best
            cur_score = compute_score(a)
            if cur_score < best_score:
                best_score = cur_score
                best_a = a.copy()
        # stop if reached target
        if best_score <= 1.5:
            break
    return best_a.tolist()

"""
Propose a candidate step‑function sequence.

The algorithm is a simple simulated‑annealing / hill‑climbing search
that works directly on the evaluation metric returned by
`evaluate_sequence`.  Starting from the uniform distribution (which
gives a value of 2.0) we repeatedly move a small amount of mass from a
random index to another index, keeping the total L¹‑norm equal to 1.
The move is accepted if it improves the score, otherwise it may be
accepted with a temperature‑controlled probability to escape local
minima.  The convolution needed for the score is computed by an FFT,
making each iteration cheap (O(n log n)).  The search runs until the
time budget (≈ 1000 s) is exhausted or a score ≤ 1.5 is found.

The returned list contains only non‑negative numbers, sums to 1 and
conforms to all security checks performed inside `evaluate_sequence`.
"""

import time
import numpy as np

def _conv_score(vec: np.ndarray) -> float:
    """
    Compute 2 * n * max(conv(vec, vec)) for a vector that already sums to 1.
    The convolution is performed with FFT for speed.
    """
    n = vec.shape[0]
    # length of linear convolution
    L = 2 * n - 1
    # FFT of zero‑padded vector
    fft_vec = np.fft.rfft(vec, n=L)
    conv = np.fft.irfft(fft_vec * fft_vec, n=L)
    max_val = float(conv.real.max())
    return 2.0 * n * max_val


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a sequence of non‑negative heights that yields a low
    evaluation score.  The routine returns a Python list of floats.
    """
    np.random.seed(seed)
    # ------------------------------------------------------------------
    # Parameters (tuned empirically for the given time budget)
    # ------------------------------------------------------------------
    n = 2000                     # length of the step function
    max_iter = 10_000_000        # safety cap for the loop
    temp_start = 0.08            # initial temperature for simulated annealing
    transfer_frac = 0.12         # fraction of the donor mass moved each step
    report_every = 5000          # how often we print progress
    # ------------------------------------------------------------------

    # Start from a uniform distribution (sum == 1)
    a = np.full(n, 1.0 / n, dtype=float)

    # current best solution
    best_vec = a.copy()
    best_score = _conv_score(best_vec)

    # time management
    deadline = time.time() + budget_s - 5  # a few seconds margin
    start_time = time.time()
    iteration = 0

    # pre‑compute indices used for “edge” and “center” regions
    edge_idxs = np.r_[np.arange(int(0.05 * n)), np.arange(int(0.95 * n), n)]
    centre_idxs = np.arange(int(0.3 * n), int(0.7 * n))

    while iteration < max_iter and time.time() < deadline:
        # linear temperature schedule
        T = temp_start * max(0.0, 1.0 - iteration / max_iter)

        # pick a donor (preferably where the mass is larger)
        donor = np.random.choice(centre_idxs) if np.random.rand() < 0.7 else np.random.randint(n)
        if a[donor] <= 1e-12:
            # nothing to give – skip this iteration
            iteration += 1
            continue

        # pick a receiver (preferably near the edges)
        if np.random.rand() < 0.7:
            receiver = np.random.choice(edge_idxs)
        else:
            receiver = np.random.randint(n)

        if donor == receiver:
            iteration += 1
            continue

        # amount to transfer (a small fraction of the donor’s mass)
        delta = a[donor] * transfer_frac
        if delta < 1e-12:          # numerically too small
            iteration += 1
            continue

        # propose the new vector (mass is conserved)
        proposal = a.copy()
        proposal[donor] -= delta
        proposal[receiver] += delta

        # compute the new score
        prop_score = _conv_score(proposal)

        # Acceptance rule
        improved = prop_score < best_score
        if improved or (T > 0 and np.random.rand() < np.exp(-(prop_score - best_score) / T)):
            a = proposal
            if improved:
                best_score = prop_score
                best_vec = proposal.copy()

        # occasional small random perturbation to keep the search diverse
        if iteration % (report_every * 10) == 0:
            noise = (np.random.randn(n) * 1e-5)
            a = np.maximum(a + noise, 0.0)
            a /= a.sum()                     # renormalise
            cur_score = _conv_score(a)
            if cur_score < best_score:
                best_score = cur_score
                best_vec = a.copy()

        # progress report
        if iteration % report_every == 0:
            elapsed = time.time() - start_time
            print(
                f"Iter {iteration:8d} | score {best_score: .6f} "
                f"| elapsed {elapsed:.1f}s | temp {T:.4f}"
            )
            # early exit if we already beat the target
            if best_score <= 1.503:
                break

        iteration += 1

    # Final sanity: ensure non‑negative and sum‑to‑one (the security checks
    # inside `evaluate_sequence` will also enforce these properties)
    best_vec = np.maximum(best_vec, 0.0)
    total = best_vec.sum()
    if total == 0.0:                     # fallback – should never happen
        best_vec = np.full(n, 1.0 / n)
    else:
        best_vec /= total

    return best_vec.tolist()
