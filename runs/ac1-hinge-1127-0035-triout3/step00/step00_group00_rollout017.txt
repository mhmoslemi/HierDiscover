def project_simplex(v):
    # project v onto simplex {x >= 0, sum x = 1}
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    rho = np.where(u - cssv / np.arange(1, n+1) > 0)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0)
    return w
```

def project_simplex(v):
    """Project vector v onto probability simplex (sum=1, v>=0)."""
    if np.sum(v) == 1 and np.all(v >= 0):
        return v
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    rho = np.where(u - cssv / np.arange(1, n+1) > 0)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0)
    return w

"""
Propose a candidate sequence of non‑negative heights that reduces the
evaluation metric   2 * n * max(conv(a, a)) / (sum(a)**2).

The algorithm starts from a uniform distribution (which gives a score of 2)
and performs a projected gradient descent on a surrogate loss that encourages
the convolution to become flat.  The surrogate loss is the variance of the
convolution entries; minimizing it tends to lower the maximum convolution value.
After each gradient step the vector is projected onto the probability simplex
(non‑negative with sum = 1) so the scale of the sequence stays constant and the
evaluation remains invariant to overall scaling.

A simple adaptive step size and occasional random perturbations are used to
escape shallow plateaus.  The routine tries several sequence lengths and
keeps the best sequence found before the time budget expires.
"""

import time
import numpy as np

def _project_simplex(v):
    """Project v onto the simplex {x >= 0, sum(x) = 1}."""
    n = v.shape[0]
    # sort in descending order
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho = np.where(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        # all entries are zero – return uniform vector
        return np.full_like(v, 1.0 / n)
    rho = rho[-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def _gradient_variance(a, conv_a):
    """
    Gradient of the variance of the convolution of a with itself.
    The variance loss is L = sum_k ( (conv_a[k] - mean)^2 ).
    Its gradient w.r.t. a is
        g_j = 2 * sum_i a_i * (conv_a[i + j] - mean)
    for j = 0 .. n-1.
    """
    n = a.shape[0]
    mean = conv_a.mean()
    diff = conv_a - mean                     # length 2n‑1
    rev_a = a[::-1]                         # reversed a
    # convolution rev_a * diff gives all needed dot‑products
    # the slice `n-1 : n-1+n` corresponds to the j‑th gradient component
    conv_rev = np.convolve(rev_a, diff)     # length 3n‑2
    grad = 2.0 * conv_rev[n - 1:n - 1 + n]   # shape (n,)
    return grad


def propose_candidate(seed=42, budget_s=1000):
    """
    Search for a sequence of non‑negative numbers that yields a low evaluation
    score.  The function runs for up to ``budget_s`` seconds and returns the
    best sequence discovered.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 2.0   # a little safety margin

    best_seq = None
    best_score = float('inf')

    # try a few different lengths (small enough for fast convolution,
    # large enough to give the optimizer room to improve)
    lengths = [50, 100, 200, 400, 800]

    for n in lengths:
        if time.time() >= deadline:
            break

        # start from the uniform distribution (sum = 1)
        a = np.full(n, 1.0 / n)

        # add a tiny random perturbation to break symmetry
        for _ in range(5):
            i, j = np.random.randint(0, n, size=2)
            delta = np.random.rand() * a[i] * 0.5
            a[i] -= delta
            a[j] += delta
        a = _project_simplex(a)

        # adaptive step size
        step = 0.1

        # main optimisation loop
        while time.time() < deadline:
            conv_aa = np.convolve(a, a)               # length 2n‑1
            max_conv = conv_aa.max()
            score = 2.0 * n * max_conv                # sum(a)=1 → denominator =1
            if score < best_score:
                best_score = score
                best_seq = a.copy()
                print(f"[{time.time():.1f}] New best score {best_score:.6f} (n={n})")

            # compute gradient of the variance surrogate loss
            grad = _gradient_variance(a, conv_aa)

            # gradient step followed by projection onto the simplex
            a_new = a - step * grad
            a_new = _project_simplex(a_new)

            # evaluate the new point
            conv_new = np.convolve(a_new, a_new)
            new_score = 2.0 * n * conv_new.max()

            # accept the step if it improves the actual evaluation
            if new_score < score:

                step = min(step * 1.05, 1.0)   # gently increase step size
            else:

                if step < 1e-6:

                    delta = np.random.rand() * a[i] * 0.1
                    a[i] -= delta
                    a[j] += delta
                    a = _project_simplex(a)
                    step = 0.1

    # fallback – a single positive entry (always valid)
    if best_seq is None:
        best_seq = np.array([1.0])

    return best_seq.tolist()
