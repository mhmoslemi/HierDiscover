def propose_candidate(seed=0, budget_s=1000, **kwargs):
    import time, numpy as np
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5

    best_seq = None
    best_score = np.inf

    def evaluate(seq):
        return evaluate_sequence(seq)

    # try multiple random restarts
    while time.time() < deadline:
        # choose length n
        n = np.random.randint(200, 800)  # moderate
        # initialize distribution: random dirichlet
        a = np.random.dirichlet(np.ones(n))
        # gradient descent steps
        iter_num = 0
        # compute score for initial
        score = evaluate(a.tolist())
        if score < best_score:
            best_score = score
            best_seq = a.tolist()
            print(f"New best init score {best_score:.6f} len {n}")

        # use simple loop until time or iteration limit
        # we can adapt number of steps based on remaining time
        max_iter = 2000
        while time.time() < deadline and iter_num < max_iter:
            # compute convolution
            conv = np.convolve(a, a)
            k_max = int(np.argmax(conv))
            max_val = conv[k_max]
            # if max_val is too close to average (i.e., judged as good), may break early
            # compute gradient
            grad = np.zeros_like(a)
            for i in range(n):
                j = k_max - i
                if 0 <= j < n:
                    if i == j:
                        grad[i] = a[i]
                    else:
                        grad[i] = 2.0 * a[j]
            # step size schedule
            lr = 0.02 / (1 + iter_num * 0.001)  # decreasing
            a_new = a - lr * grad
            # clip negative
            a_new = np.maximum(a_new, 0.0)
            # project onto simplex to keep sum=1
            if a_new.sum() == 0:
                a_new = np.random.dirichlet(np.ones(n))
            else:
                # simplex projection
                u = np.sort(a_new)[::-1]
                cssv = np.cumsum(u) - 1
                rho = np.where(u - cssv / np.arange(1, n+1) > 0)[0][-1]
                theta = cssv[rho] / (rho + 1.0)
                a_new = np.maximum(a_new - theta, 0)
            a = a_new
            # evaluate new score
            new_score = evaluate(a.tolist())
            if new_score < best_score:
                best_score = new_score
                best_seq = a.tolist()
                print(f"Iter {iter_num}: improved to {best_score:.6f} len {n}")

"""
search for a step‑function (non‑negative sequence) that minimizes
evaluate_sequence().  The algorithm works as follows:

1.  Initialise many random probability vectors (sum = 1) using a Dirichlet
    distribution of moderate size (200‑800 elements).  Scaling does not
    affect the objective, therefore fixing the sum to 1 is convenient.

2.  For each candidate run a short projected sub‑gradient descent.
    At every iteration we locate the index k where the discrete convolution
    a*a attains its maximum.  The gradient of that entry w.r.t. a is
    easily obtained (∂/∂a_i Σ a_i a_{k-i} = a_{k-i}+a_{k-i} = 2·a_{k-i}
    except when i = k-i).  We step in the opposite direction, clip to
    non‑negative values and project back onto the probability simplex.

3.  Every few iterations we evaluate the candidate with evaluate_sequence()
    and keep the best sequence seen so far.

4.  After the descent we try a few cheap re‑orderings (zig‑zag interleaving
    of largest and smallest coefficients, random shuffles) – these often
    reduce the convolution peak because they decorrelate large values that
    would otherwise multiply each other.

5.  The best sequence found (including possibly the supplied
    ``height_sequence_1``) is returned.  The routine respects the 1000 s
    budget and prints progress.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# helper utilities – must be top‑level (no closures)
# ----------------------------------------------------------------------
def _project_simplex(v):
    """Euclidean projection of a vector onto the probability simplex."""
    v = np.asarray(v, dtype=float)
    n = v.shape[0]
    # sort v in descending order
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all entries non‑positive – return uniform distribution
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1] - 1
    theta = cssv[cond][-1] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def _zigzag_order(v):
    """
    Interleave the sorted values: largest, smallest, 2nd largest,
    2nd smallest, …  This often spreads large coefficients apart.
    """
    w = sorted(v, reverse=True)
    l, r = 0, len(w) - 1
    out = []
    take_large = True
    while l <= r:
        if take_large:
            out.append(w[l])
            l += 1
        else:
            out.append(w[r])
            r -= 1
        take_large = not take_large
    return out

    # ------------------------------------------------------------------

    # ------------------------------------------------------------------

            best_val = val

        except Exception:
            pass

    # ------------------------------------------------------------------
    # 2) main stochastic search loop
    # ------------------------------------------------------------------
    while time.time() < deadline:
        attempts += 1
        # pick a problem size – small enough for many iterations,
        # large enough to give a decent bound.
        n = np.random.randint(200, 800)

        # start from a random point on the simplex
        a = np.random.dirichlet(np.ones(n))
