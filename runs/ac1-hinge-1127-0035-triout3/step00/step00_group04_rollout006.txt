"""
propose_candidate
-----------------
Construct a step function whose self‑convolution is almost flat.
The idea is to use the discrete analogue of the *arcsine distribution*:
    a_i = n / sqrt((i+0.5) * (n‑i‑0.5))   ,  i = 0 … n‑1

For this shape the convolution a∗a is (up to discretisation) constant,
so the evaluation function
    2·n·max(a∗a) / (sum(a))²
approaches the theoretical limit 2n/(2n‑1) ≈ 1 for large n.
The routine creates such sequences for a few different lengths,
keeps the best one, and then performs cheap local “mass‑transfer”
perturbations while time permits.

The returned list contains only non‑negative floats, satisfies every
security check in ``evaluate_sequence`` and is ready to be scored.
"""

import math
import random
import time
import numpy as np


def _arcsine_sequence(n: int) -> list[float]:
    """Return a discrete arcsine‑shaped sequence of length n."""
    seq = [0.0] * n
    for i in range(n):
        # denominator = sqrt((i+0.5)*(n-i-0.5))
        denom = math.sqrt((i + 0.5) * (n - i - 0.5))
        # denom > 0 for all i in a finite n
        seq[i] = n / denom
    return seq


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a sequence with a low value of evaluate_sequence().
    The search works in two phases:
      1. try several arcsine‑shaped sequences of different lengths;
      2. refine the current best by random small mass‑transfer moves.
    The function stops before the given budget (seconds) and returns
    the best sequence found.
    """
    # ---- basic setup -------------------------------------------------
    random.seed(seed)
    np.random.seed(seed)
    start = time.time()
    deadline = start + budget_s - 0.5           # tiny safety margin

    # keep the best sequence found so far
    best_seq = None
    best_score = float("inf")

    # ---- 1️⃣ try deterministic arcsine candidates --------------------
    # start from the length of the known construction if it exists
    candidate_lengths = []
    if "height_sequence_1" in globals():
        try:
            candidate_lengths.append(len(height_sequence_1))
        except Exception:
            pass

    # add a few hand‑picked lengths and a handful of random ones
    candidate_lengths.extend([500, 1000, 2000, 3000, 4000, 5000])
    while len(candidate_lengths) < 15:
        candidate_lengths.append(random.randint(500, 5000))

    # evaluate each candidate (stop early if time runs out)
    for n in candidate_lengths:
        if time.time() >= deadline:
            break
        seq = _arcsine_sequence(n)
        try:
            score = evaluate_sequence(seq)
        except Exception:
            # any unexpected error – treat as infinite
            score = float("inf")
        print(f"[init] n={n:<5d}  score={score:.6f}")
        if score < best_score:
            best_score = score
            best_seq = seq

    # fallback: if for whatever reason no candidate succeeded,
    # use a simple uniform vector (worst case is 2.0, still valid)
    if best_seq is None:
        n = 1000
        best_seq = [1.0] * n
        best_score = evaluate_sequence(best_seq)
        print("[fallback] using uniform sequence")

    # ---- 2️⃣ local refinement ----------------------------------------
    n = len(best_seq)

    # helper to perform a tiny mass‑transfer between two positions
    def perturb(seq):
        i = random.randrange(n)
        j = random.randrange(n)
        while j == i:
            j = random.randrange(n)
        # move up to 5 % of a_i to a_j (keeping sum unchanged)
        delta = seq[i] * random.random() * 0.05
        if delta <= 0.0:
            return seq  # nothing to do
        new_seq = seq.copy()
        new_seq[i] = max(0.0, new_seq[i] - delta)
        new_seq[j] = new_seq[j] + delta
        return new_seq

    iteration = 0
    # main optimisation loop – stop before deadline
    while time.time() < deadline:
        iteration += 1
        # occasional full‑noise move to escape very flat plateaus
        if random.random() < 0.01:
            noise = np.random.randn(n) * 0.01
            cand = [max(0.0, best_seq[k] + noise[k]) for k in range(n)]
        else:
            cand = perturb(best_seq)

        try:
            cand_score = evaluate_sequence(cand)
        except Exception:
            continue  # ignore malformed candidates

        if cand_score < best_score:
            best_score = cand_score
            best_seq = cand
            print(f"[improve] iter={iteration:<6d}  score={cand_score:.6f}")

        # every few hundred iterations we recompute n (it never changes)
        if iteration % 500 == 0:
            n = len(best_seq)

    print(f"[final] best score={best_score:.6f}  length={len(best_seq)}")
    return best_seq
