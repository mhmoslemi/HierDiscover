"""
Hybrid LP‑gradient stochastic search for a low‑autocorrelation (step) sequence.

The evaluation metric is   F(a)=2·n·max(conv(a,a))/sum(a)^2 .
For a fixed sum the objective equals the maximal convolution coefficient,
so we want to flatten the autocorrelation of a non‑negative sequence.

The algorithm combines three complementary moves:

1. **LP improvement** – given the current sequence `a` we normalise it to
   sum = sqrt(2n) and solve the linear program
        maximise   sum(b)
        subject   conv(a,b) ≤ M    (M = current max‑convolution)
                    b ≥ 0 .
   The LP uses only the *active* constraints (indices where the
   convolution attains its maximum) which keeps the problem small.
   The new sequence is a small mixture of `a` and the LP solution.

2. **Projected sub‑gradient step** – the gradient of the maximum
   convolution entry (when it is unique) is `g_i = 2·a_{k*‑i}` where
   `k*` is the index of the current maximum.  Updating
        a ← a – η·g
   and projecting back onto the probability simplex (non‑negative,
   sum = 1) reduces the peak autocorrelation.

3. **Random mass exchange** – a pair of positions exchange a random
   fraction of mass, providing exploration and helping to escape
   plateaus.

All moves keep the sequence non‑negative and renormalise it to sum = 1,
which leaves the evaluation invariant (the metric is scale‑free).
The best sequence seen during the allotted time is returned.
"""

import time
import numpy as np
from scipy.optimize import linprog


def _fast_evaluate(seq: np.ndarray) -> float:
    """Fast version of evaluate_sequence (no security checks)."""
    n = len(seq)
    conv = np.convolve(seq, seq)
    max_b = conv.max()
    s = seq.sum()
    if s < 1e-12:
        return float("inf")
    return float(2.0 * n * max_b / (s * s))


def _solve_convolution_lp(f_seq: np.ndarray, rhs: float,
                          active_idx: np.ndarray) -> np.ndarray | None:
    """
    Solve   maximise sum(b)  subject to  conv(f_seq, b) ≤ rhs  on the
    active indices only, with b ≥ 0.
    Returns the optimal b as a dense 1‑D array, or None if LP fails.
    """
    n = len(f_seq)
    # objective: maximise sum(b)  → minimise -sum(b)
    c = -np.ones(n)

    # build only the rows for the active convolution constraints
    k_list = active_idx.tolist()
    m = len(k_list) + n                     # total constraints (+ non‑neg)
    A = np.zeros((m, n))
    b = np.zeros(m)

    # convolution constraints
    row = 0
    for k in k_list:
        i_start = max(0, k - (n - 1))
        i_end = min(n - 1, k)
        # fill the row: for i in [i_start, i_end] set A[row, j] = f_seq[i]
        # where j = k - i
        i_vals = np.arange(i_start, i_end + 1, dtype=int)
        j_vals = k - i_vals
        A[row, j_vals] = f_seq[i_vals]
        b[row] = rhs
        row += 1

    # non‑negativity constraints  –I·b ≤ 0
    A[row:, :] = -np.eye(n)
    # corresponding RHS entries are already zero

    # solve with HiGHS (fast, accepts dense matrices)
    res = linprog(
        c,
        A_ub=A,
        b_ub=b,
        method="highs",
        options={"time_limit": 5.0},  # seconds per LP
    )
    if res.success:
        return res.x
    return None


def _lp_step(seq: np.ndarray) -> np.ndarray | None:
    """
    Perform one LP‑based improvement step.
    Returns a new sequence (still summing to 1) or None on failure.
    """
    n = len(seq)
    total = seq.sum()
    if total == 0.0:
        return None

    # normalise to the usual sum = sqrt(2n)
    scale = np.sqrt(2.0 * n) / total
    f_norm = seq * scale

    # current maximal convolution value (M)
    conv_f = np.convolve(f_norm, f_norm)
    M = conv_f.max()

    # indices where the bound is active (within a tiny tolerance)
    tol = 1e-12
    active = np.where(conv_f >= M - tol)[0]

    # solve LP
    g = _solve_convolution_lp(f_norm, M, active)
    if g is None:
        return None

    # normalise the LP solution back to the original total sum
    g_sum = g.sum()
    if g_sum == 0.0:
        return None
    g_norm = g * (np.sqrt(2.0 * n) / g_sum)

    # small mixing (t can be tuned)
    t = 0.01
    new_seq = (1.0 - t) * seq + t * g_norm
    new_seq = np.maximum(new_seq, 0.0)

    # renormalise to sum = 1 (scale‑invariant objective)
    new_seq /= new_seq.sum()
    return new_seq

            conv = np.convolve(seq, seq)
            k_max = int(conv.argmax())
            # gradient of the maximal entry: g_i = 2 * a_{k_max - i}
            grad = np.zeros_like(seq)
            i_start = max(0, k_max - (n - 1))
            i_end = min(n - 1, k_max)
            idx = np.arange(i_start, i_end + 1, dtype=int)
            grad[idx] = 2.0 * seq[k_max - idx]

            # simple decreasing step size schedule
            eta = grad_eta / (1.0 + iter_cnt * 0.0005)

            cand = seq - eta * grad
            cand = np.maximum(cand, 0.0)

            # optional gentle symmetrisation (helps stability)
            cand = (cand + cand[::-1]) * 0.5

            # renormalise to keep the sum = 1 (objective is scale‑free)
            cand /= cand.sum()
            seq = cand
        # 3) Random mass‑exchange perturbation (exploration)
        else:
            i, j = np.random.choice(n, size=2, replace=False)
            frac = np.random.rand() * 0.03   # up to 3 % of the source mass
            delta = seq[i] * frac
            cand = seq.copy()
            cand[i] -= delta
            cand[j] += delta
            cand = np.maximum(cand, 0.0)
            cand /= cand.sum()
            seq = cand

        # ------------------------------------------------------------------
        # evaluate the new candidate
        # ------------------------------------------------------------------
        cur_score = _fast_evaluate(seq)

        # keep the best ever seen
        if cur_score < best_score:
            best_score = cur_score
            best_seq = seq.copy()

        # a very light acceptance rule for random perturbations:
        # if the move made things worse we occasionally revert to the previous best
        if cur_score > best_score and np.random.rand() > 0.02:
            # revert to the global best with small probability of staying
            seq = best_seq.copy()

        # occasional progress report
        if iter_cnt % 250 == 0:
            elapsed = time.time() - start_time
            print(
                f"Iter {iter_cnt:5d} | time {elapsed:5.1f}s | "
                f"best {best_score:.6f}"
            )

    # ----------------------------------------------------------------------
    # final housekeeping
    # ----------------------------------------------------------------------
    # Ensure every entry respects the limits enforced by evaluate_sequence
    best_seq = np.clip(best_seq, 0.0, 1000.0)
    # A sum smaller than 0.01 is rejected – we already normalise to 1
    print(
        f"Search finished after {time.time() - start_time:.1f}s – "
        f"best score {best_score:.6f}"
    )
    return best_seq.tolist()
