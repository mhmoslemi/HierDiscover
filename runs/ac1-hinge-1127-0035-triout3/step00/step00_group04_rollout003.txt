def generate_sequence(n, p, max_val=1000.0):
    # generate decreasing shape: values = max_val * ((n-1-i)/(n-1))**p
    # For i from 0..n-1.
    if n==1:
        return [max_val]
    t = np.arange(n, dtype=float)
    seq = max_val * ((n - 1 - t) / (n - 1)) ** p
    # Ensure last element maybe zero.
    return seq.tolist()

"""
search for a step function (non‑negative heights) minimising
    2 * n * max(conv(a,a)) / (sum(a)^2)

The search builds simple power‑law decreasing sequences:
    a_i = C * (1 - i/(n-1))**p

with C ≤ 1000 (the limit enforced by evaluate_sequence). For a given
pair (n, p) the value of the evaluation function can be expressed
analytically, but we simply query evaluate_sequence.  The algorithm
enumerates a modest grid of lengths (100 … 5000) and exponents
(p = 1 … 10), keeps the best sequence found and finally performs a
few tiny random perturbations to possibly improve it further.
The whole routine respects the given time budget.
"""

import time
import numpy as np

def generate_power_sequence(n: int, p: float, max_val: float = 1000.0) -> list[float]:
    """
    Return a decreasing power‑law sequence of length n.
    The first element equals max_val, the last element equals 0.
    """
    if n <= 1:
        return [max_val]
    # factor runs from 1 (i=0) down to 0 (i=n-1)
    idx = np.arange(n, dtype=float)
    factor = (n - 1 - idx) / (n - 1)
    seq = max_val * np.power(factor, p)
    # tiny safeguard against negative rounding errors
    seq = np.maximum(0.0, seq)
    return seq.tolist()


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a sequence with a low evaluation value.
    The function respects the supplied time budget and returns the
    best sequence discovered.
    """

    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5  # keep a small safety margin

    # candidate parameters – lengths and powers
    lengths = list(range(200, 5200, 200))          # 200, 400, …, 5000
    powers   = list(range(1, 11))                 # p = 1 … 10

    # shuffle the grid so that we explore different points early
    grid = [(n, p) for n in lengths for p in powers]
    np.random.shuffle(grid)

    best_score = float('inf')
    best_seq = None

    # -----------------------------------------------------------------
    # 1) exhaustive scan of the (n, p) grid
    # -----------------------------------------------------------------
    for n, p in grid:
        if time.time() >= deadline:
            break

        seq = generate_power_sequence(n, p, max_val=1000.0)
        try:
            score = evaluate_sequence(seq)
        except Exception:
            continue

        # log progress (resize to a few decimal places)
        print(f"tested n={n:4d}, p={p:2d} → score={score:.6f}")

        if score < best_score:
            best_score = score
            best_seq = seq
            print(f"  ** new best **  n={n}, p={p}, score={best_score:.6f}")

    # -----------------------------------------------------------------
    # 2) if time remains, apply cheap random perturbations around the
    #    best candidate found so far
    # -----------------------------------------------------------------
    if best_seq is not None:
        best_arr = np.array(best_seq, dtype=float)
        n_best = len(best_arr)

        while time.time() < deadline:
            # create a tiny Gaussian perturbation (≤1 % of the current max)
            noise = np.random.randn(n_best) * 0.01 * np.max(best_arr)
            pert = best_arr + noise
            pert = np.clip(pert, 0.0, 1000.0)   # keep within allowed range

            try:
                s = evaluate_sequence(pert.tolist())
            except Exception:
                break

            if s < best_score:
                best_score = s
                best_seq = pert.tolist()
                best_arr = pert
                print(f"  ++ perturbation improved score to {best_score:.6f}")

            # stop after a few successful improvements or when the
            # improvement becomes negligible
            if s >= best_score * 0.9999:
                break

    # -----------------------------------------------------------------
    # 3) fallback – should never happen but guarantees a return value
    # -----------------------------------------------------------------
    if best_seq is None:
        # uniform sequence of length 1000 (its value is exactly 2.0)
        best_seq = [1.0] * 1000
        best_score = evaluate_sequence(best_seq)
        print("fallback to uniform sequence, score=", best_score)

    print(f"final best score: {best_score:.6f} on a sequence of length {len(best_seq)}")
    return best_seq

import time
import numpy as np


def generate_power_sequence(n: int, p: float, max_val: float = 1000.0) -> list[float]:
    """
    Produce a decreasing power‑law sequence of length `n`.

    a_0 = max_val, a_{n-1} = 0,  a_i = max_val * ((n‑1‑i)/(n‑1))**p .
    All entries are non‑negative and never exceed `max_val`.
    """
    if n <= 1:
        return [max_val]

    idx = np.arange(n, dtype=float)
    factor = (n - 1 - idx) / (n - 1)           # runs from 1 down to 0
    seq = max_val * np.power(factor, p)
    seq = np.maximum(0.0, seq)                # protect against tiny negatives
    return seq.tolist()


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a low‑scoring height sequence within the given time budget.
    Returns the best sequence discovered (as a plain Python list of floats).
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5   # small safety margin

    # -----------------------------------------------------------------
    # 1) enumerate a grid of (length, exponent) pairs
    # -----------------------------------------------------------------
    lengths = list(range(200, 5200, 200))    # 200, 400, …, 5000
    powers   = list(range(1, 11))           # p = 1 … 10

    grid = [(n, p) for n in lengths for p in powers]
    np.random.shuffle(grid)                 # explore in random order

    best_score = float('inf')
    best_seq = None

    for n, p in grid:
        if time.time() >= deadline:
            break

        seq = generate_power_sequence(n, p, max_val=1000.0)

        try:
            score = evaluate_sequence(seq)
        except Exception:
            continue

        # brief progress report
        print(f"tested n={n:4d}, p={p:2d} → score={score:.6f}")

        if score < best_score:
            best_score = score
            best_seq = seq
            print(f"  ** new best **  n={n}, p={p}, score={best_score:.6f}")

    # -----------------------------------------------------------------
    # 2) cheap random perturbations around the current best (if time permits)
    # -----------------------------------------------------------------
    if best_seq is not None:
        best_arr = np.array(best_seq, dtype=float)
        n_best = len(best_arr)

        while time.time() < deadline:
            # add a small Gaussian jitter (≤1 % of current maximal entry)
            noise = np.random.randn(n_best) * 0.01 * np.max(best_arr)
            pert = best_arr + noise
            pert = np.clip(pert, 0.0, 1000.0)   # respect the problem bounds

            try:
                s = evaluate_sequence(pert.tolist())
            except Exception:
                break

            if s < best_score:
                best_score = s
                best_seq = pert.tolist()
                best_arr = pert
                print(f"  ++ perturbation improved score to {best_score:.6f}")

            # stop early if no measurable improvement
            if s >= best_score * 0.9999:
                break

    # -----------------------------------------------------------------
    # 3) fallback (should never be needed)
    # -----------------------------------------------------------------
    if best_seq is None:
        best_seq = [1.0] * 1000
        best_score = evaluate_sequence(best_seq)
        print("fallback to uniform sequence, score =", best_score)

    print(f"final best score: {best_score:.6f} on a sequence of length {len(best_seq)}")
    return best_seq
