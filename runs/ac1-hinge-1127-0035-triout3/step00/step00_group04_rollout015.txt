def propose_candidate(seed=0, budget_s=950):
    np.random.seed(seed)
    deadline = time.time() + budget_s
    n = 3000   # choose?
    a = np.full(n, 1.0) # uniform
    best = a.copy()
    best_score = evaluate_sequence(best.tolist())
    iteration = 0
    while time.time() < deadline:
        # Compute convolution
        conv = np.convolve(a, a)
        max_idx = int(np.argmax(conv))
        max_b = conv[max_idx]
        # compute gradient
        g = np.zeros_like(a)
        # for i in range(n):
        #   j = max_idx - i
        #   if 0 <= j < n:
        #       g[i] = 2 * a[j]
        # vectorized:
        j_idx = max_idx - np.arange(n)
        mask = (j_idx >= 0) & (j_idx < n)
        g[mask] = 2 * a[j_idx[mask]]
        sum_a = a.sum()
        # gradient of objective f:
        # df = (2 * n / sum_a**2) * g - (4 * n * max_b / sum_a**3) * np.ones_like(a)
        # but we can just take descent direction d = g - lambda*1 where lambda = something.
        # compute factor:
        factor1 = 2 * n / (sum_a ** 2)
        factor2 = 4 * n * max_b / (sum_a ** 3)
        grad = factor1 * g - factor2 * np.ones_like(a)
        # step size schedule:
        eta = 0.01 / (1 + iteration * 0.001) # adjust
        a_new = a - eta * grad
        # clip negative and impose upper bound
        a_new = np.clip(a_new, 0, 1000) # maybe you want max 1? but fine
        # optionally rescale to keep sum not too low
        # ensure sum > 0
        if a_new.sum() < 1e-6:
            a_new = np.full_like(a_new, 1e-3)
        # assign for next iteration
        a = a_new
        # evaluate
        score = evaluate_sequence(a.tolist())
        if score < best_score:
            best_score = score
            best = a.copy()
        iteration += 1
        # optional early break if score ~ lower bound
        if best_score < 1.01: break

def solve_convolution_lp(f_seq, rhs):
    n = len(f_seq)
    # Form A_ub for constraints conv(f, b) <= rhs
    rows = []
    cols = []
    data = []
    for k in range(2*n - 1):
        for i in range(max(0, k - (n-1)), min(n-1, k)+1):
            j = k - i
            rows.append(k)
            cols.append(j)
            data.append(f_seq[i])  # coefficient f_i for variable b_j
    A_ub = sparse.csr_matrix((data, (rows, cols)), shape=(2*n - 1, n))
    # Bounds b >= 0 will be set via bounds argument
    c = -np.ones(n)  # maximize sum -> minimize negative sum
    result = linprog(c, A_ub=A_ub, b_ub=np.full(2*n-1, rhs), bounds=(0, None), method='highs')
    if result.success:
        return result.x
    return None

    a *= target_sum / a.sum()
    return a

def solve_convolution_lp(f, bound):
    n = len(f)
    # Build sparse convolution matrix
    rows = []
    cols = []
    data = []
    for k in range(2*n - 1):
        i_start = max(0, k - (n - 1))
        i_end = min(n - 1, k)
        # for j = k - i
        for i in range(i_start, i_end + 1):
            j = k - i
            rows.append(k)
            cols.append(j)
            data.append(f[i])
    A = sparse.csr_matrix((data, (rows, cols)), shape=(2*n - 1, n))
    c = -np.ones(n)   # maximize sum => minimize -sum
    bounds = [(0, None)] * n
    res = optimize.linprog(c, A_ub=A, b_ub=np.full(2*n - 1, bound), bounds=bounds, method='highs')
    if res.success:
        return res.x
    return None

        contrib = a * a[::-1]               # a[i] * a[n-1-i]

                        best_seq = a.copy()
        iter_count += 1
        # gentle temperature decay
        temperature *= 0.9999

"""
Fast construction of a low‑value step‑function sequence.

The code builds the classic inverse‑square‑root (binomial) sequence,
which gives a bound of ≈π/2 ≈ 1.571.  From there a cheap stochastic
hill‑climb moves mass from indices that contribute most to the central
convolution peak toward the ends of the array.  Every few hundred
iterations a tiny linear‑programming refinement (the same LP used in
the reference implementation) is performed to enlarge the ℓ₁‑norm
while keeping the current peak unchanged.  The algorithm keeps the
best sequence seen during the allotted time and finally returns it as
a plain Python list.
"""

import time
import numpy as np
from scipy.optimize import linprog
from scipy import sparse


def build_binomial_sequence(n: int) -> np.ndarray:
    """
    Return the length‑n sequence a_i = C(2i,i)/4**i (up to a common scale).
    The coefficients satisfy a * a = constant and lead to a bound ≈π/2.
    The result is scaled so that sum(a) = sqrt(2n) (the normalisation used
    by the scoring function).
    """
    a = np.empty(n, dtype=np.float64)
    a[0] = 1.0
    # recurrence a_{i} = a_{i-1} * (2i-1) / (2i)
    for i in range(1, n):
        a[i] = a[i - 1] * (2.0 * i - 1.0) / (2.0 * i)
    target_sum = np.sqrt(2.0 * n)
    a *= target_sum / a.sum()
    return a


def solve_convolution_lp(f: np.ndarray, bound: float) -> np.ndarray | None:
    """
    Solve the LP:
        maximise   sum(b_i)
        subject to  (f * b)_k <= bound   for all k
                    b_i >= 0

    Returns the optimal b (as a dense numpy array) or None if the LP
    fails.  The matrix of constraints is built as a sparse Toeplitz‑like
    matrix, which works well with the HiGHS solver.
    """
    n = f.shape[0]
    # Build the (2n‑1) × n constraint matrix A where A[k, j] = f[k‑j] if 0≤k‑j<n else 0
    rows, cols, data = [], [], []
    for k in range(2 * n - 1):
        i_start = max(0, k - (n - 1))
        i_end = min(n - 1, k)
        # i ranges over indices of f that appear in (f*b)_k
        for i in range(i_start, i_end + 1):
            j = k - i
            rows.append(k)
            cols.append(j)
            data.append(f[i])
    A = sparse.csr_matrix((data, (rows, cols)), shape=(2 * n - 1, n))

    # maximise sum(b)  → minimise -sum(b)
    c = -np.ones(n, dtype=np.float64)
    bounds = [(0, None)] * n
    res = linprog(
        c,
        A_ub=A,
        b_ub=np.full(2 * n - 1, bound, dtype=np.float64),
        bounds=bounds,
        method="highs",
        options={"presolve": True},
    )
    if res.success:
        return res.x
    return None

    """

    evaluation function used in the competition.

    Steps

    1. Initialise the sequence with the analytic binomial construction.
    2. Repeatedly perform cheap stochastic moves: transfer a small
       fraction of mass from a “high‑contribution’’ index (large

       improves the score, otherwise keep the old sequence.
    3. Every few hundred iterations run a tiny LP‑based refinement that
       finds a direction with a larger ℓ₁‑norm while preserving the

                        best_seq = a.copy()

        # temperature decay (very gentle)
        temperature *= 0.9999
        it += 1
