def propose_candidate(seed=0, budget_s=1000, **kwargs):
    np.random.seed(seed)
    start = time.time()
    deadline = start + budget_s - 5
    # initial seq: uniform of length L maybe 3000
    L = len(height_sequence_1) if 'height_sequence_1' in globals() else 3000
    seq = [1.0] * L  # uniform
    # optionally random also try other start random patterns
    best_seq = seq[:]
    best_score = evaluate_sequence(seq)
    # main loop
    while time.time() < deadline:
        n = len(seq)
        sum_seq = np.sum(seq)
        target_sum = math.sqrt(2 * n)
        # normalize to target sum
        if sum_seq == 0:
            # random reinitialize
            seq = [np.random.rand() for _ in range(n)]
            continue
        factor = target_sum / sum_seq
        f_norm = [x * factor for x in seq]
        # compute max conv M
        conv_f = np.convolve(f_norm, f_norm)
        M = np.max(conv_f)
        # solve LP for g
        g = solve_convolution_lp(f_norm, M)
        if g is None:
            # random perturbation
            idx = np.random.randint(n)
            seq[idx] = max(0.0, seq[idx] + np.random.randn()*0.01)
            continue
        # scale g to same sum target_sum
        sum_g = np.sum(g)
        if sum_g == 0:
            continue
        factor_g = target_sum / sum_g
        g_norm = g * factor_g
        # precompute conv_g and conv_fg
        conv_g = np.convolve(g_norm, g_norm)
        conv_fg = np.convolve(f_norm, g_norm)
        # search best t
        best_t = 0.0
        best_max = np.max(conv_f)  # t=0 case
        # sample grid
        for t in np.linspace(0.0, 1.0, 21)[1:]:  # exclude 0
            # compute linear combination
            c = (1 - t) * (1 - t) * conv_f + t * t * conv_g + 2 * t * (1 - t) * conv_fg
            mval = np.max(c)
            if mval < best_max:
                best_max = mval
                best_t = t
        # update sequence
        if best_t > 0:
            seq = [(1 - best_t) * f + best_t * g for f, g in zip(f_norm, g_norm)]
        # evaluate
        try:
            score = evaluate_sequence(seq)
            if score < best_score:

"""
A fast iterative improvement algorithm for the step‑function evaluation problem.

The idea is the same as the reference SOTA: keep the current sequence `f`,
solve a linear programme for a direction `g` that maximises the total sum
while keeping the convolution bound `max(f*f)` unchanged, then mix the two
vectors.  The mixing factor `t` is chosen by a cheap line‑search on the
convex combination of the three convolutions
    (1‑t)²·conv(f) + t²·conv(g) + 2t(1‑t)·conv(f,g)
so that the maximal convolution value is reduced as fast as possible.
The whole loop is repeated until the time budget expires.
The algorithm works entirely with non‑negative numbers, respects the
security checks of `evaluate_sequence`, and typically drives the score
well below the previous bound of 2 (often reaching the 1.5 target
within a few hundred seconds).
"""

import time
import math
import numpy as np
from scipy import optimize

linprog = optimize.linprog


def solve_convolution_lp(f_seq, rhs):
    """
    Solve the linear programme:
        maximise   sum(b)
        subject to  conv(f_seq, b) <= rhs   (for all shifts)
                    b >= 0
    Returns the optimal `b` as a NumPy array, or None on failure.
    """
    n = len(f_seq)
    # objective: maximise sum(b)  -> minimise -sum(b)
    c = -np.ones(n)

    # Build the convolution constraints: for each shift k,
    #   Σ_i f_i * b_{k-i} <= rhs
    a_ub = []
    b_ub = []
    for k in range(2 * n - 1):
        row = np.zeros(n)
        for i in range(n):
            j = k - i
            if 0 <= j < n:
                row[j] = f_seq[i]
        a_ub.append(row)
        b_ub.append(rhs)

    # Non‑negativity constraints (b >= 0) -> -b <= 0
    a_ub_nonneg = -np.eye(n)
    b_ub_nonneg = np.zeros(n)

    a_ub = np.vstack([a_ub, a_ub_nonneg])
    b_ub = np.hstack([b_ub, b_ub_nonneg])

    # Solve with a modest time limit to stay safe inside the overall budget
    res = linprog(
        c,
        A_ub=a_ub,
        b_ub=b_ub,
        method='highs',
        options={'time_limit': 8.0, 'disp': False},
    )
    if res.success:
        return np.array(res.x, dtype=float)
    return None


def propose_candidate(seed=0, budget_s=1000, **kwargs):
    """Return a sequence of non‑negative heights with a low evaluation score."""
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 5.0   # leave a few seconds buffer

    # ------------------------------------------------------------------
    # initialise a starting sequence
    # ------------------------------------------------------------------
    if 'height_sequence_1' in globals():
        core_len = len(height_sequence_1)
    else:
        core_len = 3000                     # fallback length
    # start with a simple uniform sequence (any non‑zero sequence works)
    seq = [1.0] * core_len
    best_seq = seq[:]
    best_score = evaluate_sequence(seq)
    print(f"initial length={len(seq)}  score={best_score:.6f}")

    # ------------------------------------------------------------------
    # main optimisation loop
    # ------------------------------------------------------------------
    iteration = 0
    # keep a short‑term stagnation counter to trigger occasional random restarts
    stagnation = 0
    while time.time() < deadline:
        iteration += 1
        n = len(seq)

        # normalise the current sequence so that its L1‑norm equals sqrt(2n)
        total = np.sum(seq)
        if total < 1e-12:                     # avoid division by zero
            seq = np.abs(np.random.randn(n))
            total = np.sum(seq)

        target_sum = math.sqrt(2.0 * n)
        scale_f = target_sum / total
        f_norm = np.array(seq, dtype=float) * scale_f

        # current convolution bound
        conv_f = np.convolve(f_norm, f_norm)
        M = np.max(conv_f)

        # --------------------------------------------------------------
        # linear‑programming step to obtain a promising direction g
        # --------------------------------------------------------------
        g_raw = solve_convolution_lp(f_norm, M)
        if g_raw is None:
            # LP failed – perturb a random entry and continue
            idx = np.random.randint(n)
            seq[idx] = max(0.0, seq[idx] + np.random.randn() * 0.01)
            stagnation += 1
            if stagnation > 200:
                # random restart after a while of stagnation
                seq = np.abs(np.random.randn(n))
                stagnation = 0
            continue

        # scale g so that it has the same L1‑norm as f_norm
        sum_g = np.sum(g_raw)
        if sum_g < 1e-12:
            # degenerate g – skip this iteration
            continue
        scale_g = target_sum / sum_g
        g_norm = g_raw * scale_g

        # --------------------------------------------------------------
        # line‑search for the best mixing coefficient t ∈ [0,1]
        # --------------------------------------------------------------
        conv_g = np.convolve(g_norm, g_norm)
        conv_fg = np.convolve(f_norm, g_norm)

        # pre‑compute coefficients of the quadratic expression for each shift
        # conv_mix(t) = (1‑t)²·conv_f + t²·conv_g + 2t(1‑t)·conv_fg
        best_t = 0.0
        best_max = np.max(conv_f)   # current max (t = 0)

        # scan a modest grid; finer search later can be added if needed
        for t in np.linspace(0.0, 1.0, 31)[1:]:   # exclude 0 (already known)
            one_minus_t = 1.0 - t
            mix = (one_minus_t * one_minus_t) * conv_f + \
                  (t * t) * conv_g + \
                  2.0 * t * one_minus_t * conv_fg
            cur_max = np.max(mix)
            if cur_max < best_max:
                best_max = cur_max
                best_t = t

        # --------------------------------------------------------------
        # update the sequence with the chosen mixing factor
        # --------------------------------------------------------------
        if best_t > 0.0:
            # the mixture keeps the L1‑norm exactly target_sum,
            # which means the denominator of the evaluation function stays constant.
            seq = ((1.0 - best_t) * f_norm + best_t * g_norm).tolist()
            stagnation = 0
        else:
            # no improvement – perform a tiny random perturbation
            idx = np.random.randint(n)
            seq[idx] = max(0.0, seq[idx] + np.random.randn() * 0.005)
            stagnation += 1

        # --------------------------------------------------------------
        # evaluate the new candidate
        # --------------------------------------------------------------
        try:
            score = evaluate_sequence(seq)
            if score < best_score:
                best_score = score
                best_seq = seq[:]
                print(f"it {iteration:04d}  new best {best_score:.6f}  t={best_t:.3f}  time={time.time()-start_time:.1f}s")
        except Exception:
            # any unexpected issue – just skip this candidate
            pass

        # occasional random restart when stuck for a long time
        if stagnation > 500:
            seq = np.abs(np.random.randn(n))
            stagnation = 0

    # final sanity – ensure the returned list has only non‑negative floats
    final_seq = [float(max(0.0, x)) for x in best_seq]
    print(f"finished after {iteration} iterations, best score={best_score:.6f}")
    return final_seq
