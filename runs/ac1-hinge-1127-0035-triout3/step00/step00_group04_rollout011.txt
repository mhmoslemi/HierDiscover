def project_simplex(v, z=1):
    """Return projection of v onto simplex {x>=0, sum(x)=z}."""
    # reference from Duchi et al.
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - z
    rho = np.where(u - cssv / np.arange(1, n+1) > 0)[0][-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0)
    return w

def simplex_projection(v, z=1):
    """Projection of v onto simplex {x>=0, sum x = z}."""
    # Requires numpy as np
    n = len(v)
    # sort v descending
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - z
    ind = np.arange(1, n+1)
    # find rho = max{i | u_i - cssv_i / i > 0}
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all non-positive, return zeros (edge case)
        return np.zeros_like(v)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0)
    return w

def project_to_simplex(v, z=1.0):
    """Project v onto the simplex {x >=0, sum(x)=z} using the O(n log n) algorithm."""
    n = v.shape[0]
    # sort v descending
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - z
    ind = np.arange(1, n+1)
    # find rho
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all zero case
        return np.zeros_like(v)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0)
    return w

"""
Propose a sequence of non‑negative heights that lowers the value of
`evaluate_sequence`.  The method is a simple projected gradient descent on
the smoothed (soft‑max) version of the max‑convolution objective.

Key ideas
---------
* The evaluation is invariant under scaling, therefore we fix the
  ℓ¹–norm of the sequence to the canonical value `sqrt(2·n)`.
* Let b = a ⋆ a be the auto‑convolution.  The objective is
  `max(b)`.  We replace the hard max by a soft‑max with a temperature
  parameter `α`.  Its gradient w.r.t. `a` is
      ∇ = 2·corr(a, w)   where   wₖ ∝ exp(α·bₖ)
  (`corr` is the 1‑D cross‑correlation, i.e. `np.correlate`).
* After a gradient step we project back onto the simplex
  `{x ≥ 0, Σx = sqrt(2·n)}`.
* The algorithm runs many short inner loops, evaluates the true score
  regularly, keeps the best sequence seen and restarts from a random
  point from time to time.  When a score better than the target
  (≈1.5) is found the search stops early.

The whole search respects the 1000 s time limit, uses only one CPU
and no external I/O.
"""

import time
import numpy as np

# -------------------------------------------------------------
# Helper: projection onto the probability simplex (with arbitrary sum)
def _project_to_simplex(v: np.ndarray, target_sum: float) -> np.ndarray:
    """
    Projects `v` onto the simplex {x ≥ 0, Σx = target_sum}.
    Implements the O(n log n) algorithm of Duchi et al.
    """
    if target_sum < 0:
        raise ValueError("target_sum must be non‑negative")
    n = v.shape[0]
    # sort in descending order
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - target_sum
    ind = np.arange(1, n + 1)          # 1‑based indices
    # find the last index where the condition holds
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all entries are zero – return the zero vector
        return np.zeros_like(v)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# -------------------------------------------------------------
# Fast computation of the score (same as evaluate_sequence for valid inputs)
def _score(a: np.ndarray) -> float:
    """Return 2·n·max(conv(a,a)) / (sum(a)²).  Assumes a ≥ 0."""
    n = a.shape[0]
    b = np.convolve(a, a)
    return float(2.0 * n * np.max(b) / (np.sum(a) ** 2))

        vec = np.maximum(vec, 0.0)
        total = np.sum(vec)
        if total == 0.0:

            total = target_sum
        vec = vec * (target_sum / total)
