def propose_candidate(seed=42, budget_s=1000, **kwargs):
    """
    Simulated‑annealing search for a non‑negative step‑function that minimizes the
    evaluation function defined in ``evaluate_sequence``.
    The algorithm works on a moderate length (few hundred) sequence,
    starts from the uniform distribution (which gives a score of 2) and
    repeatedly moves a small amount of mass from one coordinate to another.
    Moves that decrease the maximum convolution value are always kept,
    otherwise they are accepted with a probability that depends on a temperature
    that slowly cools down.  The best sequence seen so far is returned.
    """
    import numpy as np, time, random, math

    # ----------------------------------------------------------------------
    # helpers
    # ----------------------------------------------------------------------
    def max_convolution(arr):
        """Return max value of the full convolution of arr with itself."""
        return np.convolve(arr, arr).max()

    def score_from_max(arr, max_val=None):
        """Compute the exact score that evaluate_sequence would return."""
        if max_val is None:
            max_val = max_convolution(arr)
        n = arr.size
        s = arr.sum()
        # protect against division by zero – should never happen
        if s < 1e-12:
            return float('inf')
        return float(2.0 * n * max_val / (s * s))

    # ----------------------------------------------------------------------
    # search parameters
    # ----------------------------------------------------------------------
    np.random.seed(seed)
    random.seed(seed)

    deadline = time.time() + budget_s - 0.2   # a tiny safety margin

    # try a couple of different lengths – larger n gives a lower theoretical
    # optimum but each step is more expensive.
    candidate_lengths = [200, 400, 800]   # can be extended if time permits
    best_seq = None
    best_score = float('inf')
    iter_counter = 0

    # ----------------------------------------------------------------------
    # main search loop – each iteration launches a short simulated‑annealing run
    # ----------------------------------------------------------------------
    while time.time() < deadline:
        # pick a length (biased towards larger sizes because they have more room)
        n = int(random.choice(candidate_lengths))

        # ---------- initialise a uniform distribution with a tiny perturbation ----------
        a = np.full(n, 1.0 / n, dtype=np.float64)
        a += np.random.normal(scale=1e-4, size=n)    # break symmetry
        a = np.clip(a, 0.0, None)
        a /= a.sum()                     # keep L1‑norm = 1 (the scale does not matter)

        cur_max = max_convolution(a)
        cur_score = score_from_max(a, cur_max)

        # keep the best we have seen inside this run
        best_local_seq = a.copy()
        best_local_score = cur_score

        # simulated‑annealing parameters
        temperature = cur_score * 0.5     # start at a fraction of the current score
        # run until either time runs out or we perform a reasonable number of steps
        # (the outer while loop will stop everything when the overall deadline hits)
        while time.time() < deadline:
            # --------------------------------------------------------------
            # propose a move: move a small fraction of mass from i to j
            # --------------------------------------------------------------
            i = np.random.randint(n)
            # ensure j != i
            j = np.random.randint(n - 1)
            if j >= i:
                j += 1

            if a[i] <= 0.0:
                continue

            # choose a fraction of the source mass to relocate
            frac = np.random.rand() * 0.2   # at most 20 % of the source entry
            delta = a[i] * frac
            if delta <= 0.0:
                continue

            new_a = a.copy()
            new_a[i] -= delta
            new_a[j] += delta

            # compute the new max‑convolution value (fast enough for n ≈ 800)
            new_max = np.convolve(new_a, new_a).max()
            new_score = score_from_max(new_a, new_max)

            # --------------------------------------------------------------
            # Metropolis acceptance test
            # --------------------------------------------------------------
            accept = False
            if new_score < cur_score:
                accept = True
            else:
                # probabilistic acceptance of uphill moves
                prob = math.exp(-(new_score - cur_score) / max(temperature, 1e-12))
                if np.random.rand() < prob:
                    accept = True

            if accept:
                a = new_a
                cur_score = new_score
                cur_max = new_max
                # keep the best local solution
                if cur_score < best_local_score:
                    best_local_score = cur_score
                    best_local_seq = a.copy()

            # inject a random jitter to escape plateaus
            # --------------------------------------------------------------
