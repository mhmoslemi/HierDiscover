def fast_evaluate(seq):
    s = seq.sum()
    if s < 0.01:
        return float('inf')
    conv = np.convolve(seq, seq)
    max_b = conv.max()
    return float(2 * len(seq) * max_b / (s * s))

"""
Hill‑climbing optimizer for the step‑function evaluation.

The evaluation function is
    eval(a) = 2·n·max(a*a) / (sum(a)^2)          (n = len(a))

For a fixed sum the task is to minimise the maximum entry of the
autocorrelation (the convolution of the sequence with itself).  The
objective is convex, therefore a simple stochastic hill‑climber that
preserves the total sum works well in practice.

The algorithm:
  * Choose a moderate length (default 200) and initialise a random
    positive sequence whose entries sum to 1.
  * Repeatedly pick two positions i≠j and transfer a small amount
    δ from a_j to a_i (so the sum stays exactly 1).  The transfer size
    is proportional to the current values, controlled by a step factor
    that slowly adapts.
  * Accept the move only when it strictly reduces the evaluation value.
    The best sequence seen so far is kept.
  * If no improvement is observed for a while the step size is reduced
    and the current sequence is randomly re‑initialised – this gives
    occasional restarts that help escape local minima.
  * The loop runs until the allotted time (≈ 990 s) expires and then
    returns the best sequence found.

Because the evaluation function only depends on the convolution and the
sum, we compute it locally with NumPy (avoiding the heavy argument checks
performed by ``evaluate_sequence``) and finally return a plain Python
list, which ``evaluate_sequence`` will process exactly the same way.
"""

import time
import numpy as np

def _fast_evaluate(seq: np.ndarray) -> float:
    """
    Fast version of evaluate_sequence for internal use.
    It assumes a non‑negative NumPy array (no NaNs, no infinities).
    """
    s = seq.sum()
    if s < 0.01:               # matches the guard in evaluate_sequence
        return float("inf")
    conv = np.convolve(seq, seq)
    max_b = conv.max()
    return float(2.0 * len(seq) * max_b / (s * s))


def propose_candidate(seed: int = 42,
                     budget_s: int = 1000,
                     length: int = 200) -> list[float]:
    """
    Return a sequence of non‑negative heights that (hopefully) yields an
    evaluation score lower than the current best (≈ 2.0).  The method is a
    time‑bounded stochastic hill‑climber that keeps the total sum of the
    sequence equal to 1, because the evaluation is scale‑invariant.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 1.0   # leave a small safety margin

    n = length                     # length of the candidate vector
    # ------------------------------------------------------------------
    # initialise a random positive vector with unit sum
    seq = np.random.rand(n)
    seq /= seq.sum()

    best_seq = seq.copy()
    best_score = _fast_evaluate(best_seq)

    # hill‑climbing parameters
    step_factor = 0.3               # relative magnitude of each transfer
    no_imp_counter = 0
    RESTART_AFTER = 8000           # iterations without improvement → restart
    STEP_DECAY = 0.95               # shrink step when stuck
    MIN_STEP = 0.02

    # ------------------------------------------------------------------
    while time.time() < deadline:
        # pick two distinct indices
        i, j = np.random.choice(n, 2, replace=False)

        # propose a transfer δ from j to i
        max_val = max(seq[i], seq[j])
        delta = (np.random.rand() * 2.0 - 1.0) * step_factor * max_val

        # ensure non‑negativity
        if seq[i] + delta < 0.0 or seq[j] - delta < 0.0:
            continue

        cand = seq.copy()
        cand[i] += delta
        cand[j] -= delta

        # quick evaluation (no re‑normalisation needed – sum is unchanged)
        cand_score = _fast_evaluate(cand)

        # accept strictly improving moves
        if cand_score < best_score:
            best_score = cand_score
            best_seq = cand.copy()
            seq = cand
            no_imp_counter = 0
            # slightly enlarge step when we are making progress
            step_factor = min(step_factor * 1.05, 1.0)
            print(f"[{time.time():.1f}s] New best score: {best_score:.6f}")
        else:
            no_imp_counter += 1
            # occasional random move to inject diversity
            if np.random.rand() < 0.0005:
                seq = cand

        # if we have been idle for a while, decay the step size and maybe restart
        if no_imp_counter % RESTART_AFTER == 0:
            step_factor = max(step_factor * STEP_DECAY, MIN_STEP)
            # hard restart to explore a new region of the space
            seq = np.random.rand(n)
            seq /= seq.sum()
            no_imp_counter = 0

    # convert to plain list for the external evaluator
    return best_seq.tolist()
