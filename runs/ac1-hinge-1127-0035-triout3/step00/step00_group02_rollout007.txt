    """
    Search for a non‑negative step function that minimizes
    evaluate_sequence.  The algorithm starts from a uniform
    sequence (which is known to be optimal) and then performs
    a short stochastic local search combined with an LP‑based
    direction step.  The best sequence found before the timeout
    is returned.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.1

    # Determine base length
    n = None
    if 'height_sequence_1' in globals():
        n = len(height_sequence_1)
    else:
        n = 3005   # fallback length from the statement

    # Initial uniform sequence
    init_val = 1.0
    best_seq = [init_val] * n
    best_score = evaluate_sequence(best_seq)
    print(f"Initial uniform sequence of length {n}, score={best_score}")

    # Helper functions
    def local_mutation(seq, sigma=0.05):
        # pick an index at random and add Gaussian noise, keep non‑negative
        new_seq = seq.copy()
        idx = np.random.randint(len(seq))
        new_seq[idx] = max(0.0, new_seq[idx] + np.random.randn() * sigma)
        return new_seq

    def lp_step(seq):
        # return a new sequence after LP improvement, or None
        n_local = len(seq)
        sum_seq = np.sum(seq)
        if sum_seq < 0.01:
            return None
        # normalise to sqrt(2n) as required by the LP routine
        norm_factor = np.sqrt(2 * n_local) / sum_seq
        f_norm = [x * norm_factor for x in seq]
        rhs = np.max(np.convolve(f_norm, f_norm))
        g = solve_convolution_lp(f_norm, rhs)
        if g is None:
            return None
        sum_g = np.sum(g)
        # normalise g back to same total sum as seq
        g_norm = [x * np.sqrt(2 * n_local) / sum_g for x in g]
        t = 0.01
        return [(1 - t) * a + t * b for a, b in zip(seq, g_norm)]

    # Main search loop
    curr_seq = best_seq.copy()
    iteration = 0
    while time.time() < deadline:
        iteration += 1
        # Random local move
        if np.random.rand() < 0.7:
            cand = local_mutation(curr_seq, sigma=0.1)
        else:
            cand = lp_step(curr_seq)
            if cand is None:
                cand = local_mutation(curr_seq, sigma=0.2)

        try:
            score = evaluate_sequence(cand)
        except Exception:
            continue

        # lower evaluation is better
        if score < best_score:
            best_score = score
            best_seq = cand.copy()
            curr_seq = cand.copy()
            print(f"Iter {iteration}: new best score {best_score:.12f}")
        else:
            # accept with small probability to escape plateaus
            if np.random.rand() < 0.01:
                curr_seq = cand.copy()

    print(f"Search finished after {iteration} iterations, best score {best_score}")

def solve_convolution_lp(f_sequence, rhs):
    """Solve LP to maximize sum(b) subject to conv(f,b) <= rhs and b >= 0."""
    n = len(f_sequence)
    # objective: maximise sum(b) -> minimise -sum(b)
    c = -np.ones(n)
    # Build convolution constraints for each shift k
    a_ub = []
    b_ub = []
    for k in range(2 * n - 1):
        row = np.zeros(n)
        # for each i where j = k-i is in range
        for i in range(n):
            j = k - i
            if 0 <= j < n:
                row[j] = f_sequence[i]
        a_ub.append(row)
        b_ub.append(rhs)
    # Add non‑negativity constraints: -b_i <= 0
    a_ub = np.vstack([a_ub, -np.eye(n)])
    b_ub = np.hstack([b_ub, np.zeros(n)])
    # Solve the LP.  We keep the time limit very low to avoid stalls.
    result = optimize.linprog(c, A_ub=a_ub, b_ub=b_ub,
                             method='highs',
                             options={'time_limit': 5.0, 'disp': False})
    if result.success:
        return result.x
    return None

def normalize_to_sqrt_2n(seq):
    """Scale seq so that its ℓ₁‑norm equals sqrt(2*len(seq))."""
    n = len(seq)
    total = np.sum(seq)
    if total == 0:
        return seq
    factor = np.sqrt(2 * n) / total
    return [x * factor for x in seq]

def lp_improve(seq, t=0.01):
    """Return a new sequence after one LP‑based improvement step.
    If the LP fails None is returned.
    """

    rhs = np.max(np.convolve(f_norm, f_norm))

"""
Search for a non‑negative step‑function that minimizes ``evaluate_sequence``.

The algorithm is deliberately simple because the objective
``2 * n * max(conv(a,a)) / (sum(a)**2)`` is scale‑invariant and has a
theoretical lower bound of 2, attained by the uniform distribution.
We therefore start from a uniform vector (the known optimum) and then
run a short stochastic local search combined with an LP‑based direction
step.  The best sequence found before the time‑out is returned.
"""

import time
import numpy as np
from scipy import optimize

# ----------------------------------------------------------------------
# Helper functions (all defined at top level – no closures)
# ----------------------------------------------------------------------


def solve_convolution_lp(f_sequence, rhs):
    """
    Solve the linear program

        maximise   sum(b)
        subject to conv(f, b) <= rhs   (for every shift)
                  b >= 0

    Parameters
    ----------
    f_sequence : list[float]   – normalised sequence (ℓ₁‑norm = sqrt(2n))
    rhs        : float          – current convolution bound (max entry)

    Returns
    -------
    np.ndarray or None
        The optimal b vector, or None if the LP fails.
    """
    n = len(f_sequence)
    # maximise sum(b)  → minimise -sum(b)
    c = -np.ones(n)

    # Build the convolution constraints.
    # For each shift k we have   sum_i f_i * b_{k-i} <= rhs .
    a_ub = []
    b_ub = []
    for k in range(2 * n - 1):
        row = np.zeros(n)
        for i in range(n):
            j = k - i
            if 0 <= j < n:
                row[j] = f_sequence[i]
        a_ub.append(row)
        b_ub.append(rhs)

    # Non‑negativity: -b_i <= 0
    a_ub = np.vstack([a_ub, -np.eye(n)])
    b_ub = np.hstack([b_ub, np.zeros(n)])

    # Solve with HiGHS; the time_limit option prevents pathological runs.
    result = optimize.linprog(
        c,
        A_ub=a_ub,
        b_ub=b_ub,
        method="highs",
        options={"time_limit": 5.0, "disp": False},
    )
    if result.success:
        return result.x
    return None


def normalize_to_sqrt_2n(seq):
    """
    Scale a sequence so that its ℓ₁‑norm equals √(2·len(seq)).
    If the sum is zero the original sequence is returned unchanged.
    """
    n = len(seq)
    total = np.sum(seq)
    if total == 0:
        return seq
    factor = np.sqrt(2 * n) / total
    return [x * factor for x in seq]


def lp_improve(seq, step=0.01):
    """
    Perform one LP‑based improvement step.

    The current sequence is normalised, the LP is solved to obtain a new
    vector `g` with a larger ℓ₁‑norm while keeping the convolution bound
    unchanged, and finally a convex combination with a small step size is
    returned.  If the LP fails, None is returned.
    """
    n = len(seq)
    f_norm = normalize_to_sqrt_2n(seq)
    rhs = np.max(np.convolve(f_norm, f_norm))
    g = solve_convolution_lp(f_norm, rhs)
    if g is None:
        return None
    # Bring g back to the same ℓ₁‑norm as the original sequence
    g_norm = normalize_to_sqrt_2n(g.tolist())
    # Convex combination
    new_seq = [(1.0 - step) * a + step * b for a, b in zip(seq, g_norm)]
    return new_seq


def mutate(seq, sigma=0.05):
    """
    Return a copy of `seq` where a single random coefficient is perturbed
    by Gaussian noise (standard deviation `sigma`).  The result is clipped
    to be non‑negative.
    """
    new_seq = seq.copy()
    idx = np.random.randint(len(seq))
    new_seq[idx] = max(0.0, new_seq[idx] + np.random.randn() * sigma)
    return new_seq
