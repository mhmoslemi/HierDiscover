```
def proj_simplex(v):
    # ensure sum=1, values >=0
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(1, n+1)
    cond = u - cssv / ind > 0
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0)
    return w
```

```
def proj_simplex(v):
    # Project onto simplex {x >=0, sum x = 1}
    n = len(v)
    if np.sum(v) == 1 and np.all(v >= 0):

    u = np.sort(v)[::-1]

    if len(rho) == 0:
        theta = 0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0)
    return w
```

assistantfinal**Solution Explanation**

The score to be minimised is  

```
score = 2·n·max(conv(a,a)) / (sum(a)²)
```

where  

* `a` – a non‑negative list (the “step heights”),
* `n = len(a)`,
* `conv(a,a)` – the full convolution of `a` with itself.

The score is *scale invariant* – multiplying every element by the same positive
constant does not change the value – therefore we may *fix the L¹ norm* of the
vector, e.g. `sum(a)=1`.  With that normalisation the score reduces to

```
score = 2·n·max(conv(a,a))               (1)
```

Our task becomes: **find a probability vector `a` (non‑negative, sum = 1) that
makes the maximum entry of its self‑convolution as small as possible**.

The problem is non‑convex because the convolution is quadratic.
A direct global optimiser is not feasible, but a simple *smooth‑approximation
gradient descent* works very well:

* approximate the `max` with a soft‑max (`log‑sum‑exp`) – the larger the
  temperature `α` the closer the approximation,
* compute the gradient of the smooth objective,
* take a step, then **project back onto the probability simplex**
  (non‑negative and sum = 1).

The gradient of the soft‑max of the convolution can be written as a
convolution itself, which can be evaluated efficiently.  
The whole inner loop needs only `O(n²)` elementary operations; for
`n≈200` this is a few 10⁴ floating‑point operations and we can afford many
thousands of iterations within the given time budget.

To avoid being trapped in a local minimum we

* start from several different random initialisations,
* occasionally add a tiny random perturbation,
* gradually increase the soft‑max temperature `α` and slowly decrease the
  learning rate.

The best vector seen during the search is returned.  The code below follows
exactly this plan and conforms to the required interface.

```python
"""
Propose a candidate step‑height sequence.

The algorithm fixes the L¹ norm of the sequence to 1 (so the evaluation
function reduces to 2·n·max(conv)).  It then minimises a smooth approximation
of the maximum convolution value by projected gradient descent on the
probability simplex.  Multiple random restarts, a gradually increasing
soft‑max temperature and occasional random perturbations are used to escape
local minima.  The best sequence found before the time limit is returned.
"""

import time
import numpy as np

# --------------------------------------------------------------
# Helper: projection onto the probability simplex {x ≥ 0, sum(x)=1}
def _proj_simplex(v: np.ndarray) -> np.ndarray:
    """Project vector v onto the simplex."""
    if np.all(v >= 0) and np.isclose(v.sum(), 1.0):
        return v
    n = v.shape[0]
    u = np.sort(v)[::-1]                # sort descending
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all entries are zero – return uniform vector
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1]                 # last index where condition holds
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


# --------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a step‑function giving a low evaluation score.

    Parameters
    ----------
    seed : int
        Random seed for reproducibility.
    budget_s : int
        Time budget in seconds (the function must return before it expires).

    Returns
    -------
    list[float]
        The best sequence found (all entries non‑negative, sum ≥ 0.01).
    """
    np.random.seed(seed)
    # leave a small safety margin for the calling harness
    deadline = time.time() + budget_s - 0.5

    best_seq = None
    best_score = float("inf")

    # ------------------------------------------------------------------
    # main outer loop – several random restarts until time runs out
    while time.time() < deadline:
        # problem size – a few hundred entries are cheap yet expressive
        n = np.random.randint(180, 301)          # 180 ≤ n ≤ 300, even/odd OK

        # ---- initial guess -------------------------------------------------
        # start from a random point on the simplex
        a = np.random.rand(n)
        a /= a.sum()

        # algorithmic parameters
        lr = 0.20                 # initial learning rate
        alpha = 1.0               # soft‑max temperature (low → smooth)
        max_inner = 800          # max iterations for this restart

        # ------------------------------------------------------------------
        for it in range(max_inner):
            # stop early if time is gone
            if time.time() >= deadline:
                break

            # full convolution and the exact score (sum(a)==1)
            conv = np.convolve(a, a)                 # length 2·n‑1
            max_conv = conv.max()
            score = 2.0 * n * max_conv                # (1)

            # keep the best vector seen so far
            if score < best_score:
                best_score = score
                best_seq = a.copy()
                # small progress report
                print(f"[{time.time():.1f}] n={n} it={it}  best={best_score:.6f}")

                # if we already beat the target we can return immediately
                if best_score < 1.503:
                    return best_seq.tolist()

            # ----------------------------------------------------------------
            # smooth‑max (log‑sum‑exp) and its gradient
            #   softmax weights w_k = exp(alpha·(conv_k‑max_conv)) / Σ exp(...)
            #   gradient g = 2·(w * a)  (convolution of w with reversed a)
            exp_vals = np.exp(alpha * (conv - max_conv))
            w = exp_vals / exp_vals.sum()

            # gradient via a single convolution (no Python loops)
            rev_a = a[::-1]                       # reversed a
            tmp = np.convolve(w, rev_a)           # length 3·n‑2
            # grad[i] = 2 * Σ_j w[i+j] * a[j]   →   slice the proper part
            grad = 2.0 * tmp[n - 1 : n - 1 + n]

            # gradient step and projection onto simplex
            a = a - lr * grad
            a = _proj_simplex(a)

            # occasional tiny random perturbation to escape plateaus
            if it % 63 == 0 and it > 0:
                a += np.random.randn(n) * 1e-4
                a = _proj_simplex(a)

            # schedule: slowly increase smoothness, shrink step size
            if it % 100 == 0 and it > 0:
                alpha = min(alpha * 1.07, 30.0)   # temperature rises
                lr *= 0.93                       # learning rate decays

        # end of inner loop – try a fresh random start
        # ------------------------------------------------------------------

    # ----------------------------------------------------------------------
    # safety net – if nothing was found (should not happen) return a uniform
    if best_seq is None:
        best_seq = np.full(1, 1.0)   # trivial feasible vector

    return best_seq.tolist()
```
