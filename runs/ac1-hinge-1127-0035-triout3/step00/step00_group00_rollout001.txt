function project_simplex(v): returns w.

Pseudo:

```
def project_simplex(v):
    # ensures sum(w) = 1 and w_i >= 0
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0)
    return w
```

def optimize_local(a, deadline):
    n = len(a)
    max_iters = 5000
    step0 = 0.5
    alpha0 = 5.0
    alpha1 = 30.0
    best_local_val = float('inf')
    best_a = a.copy()
    for it in range(max_iters):
        if time.time() > deadline: break
        alpha = alpha0 + (alpha1 - alpha0) * it / max_iters
        conv = np.convolve(a, a)
        max_conv = conv.max()
        # compute smoothed objective
        # avoid overflow
        expo = np.exp(alpha * (conv - conv.max()))
        p = expo / expo.sum()
        grad = 2.0 * np.correlate(p, a, mode='valid')
        step = step0 / np.sqrt(it+1)
        a_new = project_simplex(a - step * grad)
        # evaluate conv of new
        conv_new = np.convolve(a_new, a_new)
        max_conv_new = conv_new.max()
        # If new max smaller, accept
        if max_conv_new < max_conv:
            a = a_new
            max_conv = max_conv_new
        # Keep track of best
        obj_val = 2.0 * len(a) * max_conv   # evaluate with sum=1
        if obj_val < best_local_val:
            best_local_val = obj_val
            best_a = a.copy()
    # compute final evaluation using evaluate_sequence for accurate result
    final_seq = best_a.tolist()
    try:
        final_val = evaluate_sequence(final_seq)
    except Exception:
        final_val = float('inf')
    return final_seq, final_val

def project_simplex(v):
    # Implementation by sorting O(n log n)
    v = np.asarray(v, dtype=np.float64)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(len(v)) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all v <= 0, return zero vector with sum 1 by setting first element to 1?
        w = np.zeros_like(v)
        w[0] = 1.0
        return w
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0)
    return w
```

def local_optimize(a_initial, deadline):
    import numpy as np, time
    a = np.array(a_initial, dtype=np.float64)
    n = len(a)
    best_a = a.copy()
    conv = np.convolve(a, a)
    best_obj = 2.0 * n * conv.max()
    max_iters = 2000
    step0 = 0.5
    alpha0 = 5.0
    alpha1 = 30.0
    for it in range(max_iters):
        if time.time() > deadline:
            break
        alpha = alpha0 + (alpha1 - alpha0) * (it / max_iters)
        conv = np.convolve(a, a)
        max_val = conv.max()
        # smoothed objective (log-sum-exp)
        # stable compute
        shift = conv.max()
        expo = np.exp(alpha * (conv - shift))
        p = expo / expo.sum()
        grad = 2.0 * np.correlate(p, a, mode='valid')
        step = step0 / np.sqrt(it + 1.0)
        a_new = project_simplex(a - step * grad)
        # Evaluate new max
        conv_new = np.convolve(a_new, a_new)
        max_new = conv_new.max()
        # accept if improvement else occasional random acceptance
        if max_new < max_val or np.random.rand() < 0.001:  # small chance to escape
            a = a_new
            max_val = max_new
        # track best
        obj = 2.0 * n * max_val
        if obj < best_obj:
            best_obj = obj
            best_a = a.copy()
    # Return best_a and its evaluate_sequence value
    seq = best_a.tolist()
    try:
        val = evaluate_sequence(seq)
    except Exception:
        # fallback to compute ourselves
        conv = np.convolve(best_a, best_a)
        val = 2.0 * n * conv.max()
    return seq, float(val)

        best_seq = [1.0 / n] * n

"""
Propose a step‑function sequence that (approximately) minimizes

    2 * n * max(conv(a, a)) / (sum(a)^2)

The objective is scale‑invariant, so we fix sum(a)=1 and minimise the
maximum entry of the self‑convolution.  The code performs many
independent restarts, each of which runs a projected gradient descent on
a smooth surrogate of the max‑function (log‑sum‑exp).  The gradient is
computed analytically via a correlation and the iterate is projected back
onto the probability simplex.  Random restarts, occasional perturbations
of the current best solution and a small chance to accept non‑improving
steps help to escape local minima.  The best sequence found before the
time budget expires is returned.
"""

import time
import numpy as np

def _project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex {x>=0, sum(x)=1}."""
    # reference: https://arxiv.org/abs/1309.1541
    n = v.size
    # sort v in descending order
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all entries are zero or negative → put all mass on the first coordinate
        w = np.zeros_like(v)
        w[0] = 1.0
        return w
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


def _local_optimize(a0: np.ndarray, deadline: float) -> tuple[list[float], float]:
    """
    Projected (smoothed) gradient descent on a single start vector `a0`.
    Returns the best sequence found and its true evaluation (using
    evaluate_sequence).
    """
    a = a0.astype(np.float64, copy=True)
    n = a.shape[0]

    # hyper‑parameters
    max_iters = 1500                # enough for convergence on moderate n
    step0 = 0.5                     # initial step size
    alpha_start = 5.0               # start of smoothing parameter
    alpha_end = 30.0                # final smoothing (closer to true max)

    best_a = a.copy()
    conv = np.convolve(a, a)
    best_val = 2.0 * n * conv.max()   # true objective (sum(a)=1)

    for it in range(max_iters):
        if time.time() > deadline:
            break

        # smoothing schedule
        alpha = alpha_start + (alpha_end - alpha_start) * it / max_iters

        # convolution and smoothing weights
        conv = np.convolve(a, a)
        max_conv = conv.max()
        # stable log‑sum‑exp computation
        shift = max_conv
        exp_vals = np.exp(alpha * (conv - shift))
        p = exp_vals / exp_vals.sum()          # softmax over convolution entries

        # gradient of the smoothed max: 2 * correlation(p, a)
        grad = 2.0 * np.correlate(p, a, mode='valid')

        # diminishing step size
        step = step0 / np.sqrt(it + 1.0)

        # projected gradient step
        a_candidate = _project_to_simplex(a - step * grad)

        # evaluate candidate (use the true max for a cheap acceptance test)
        conv_candidate = np.convolve(a_candidate, a_candidate)
        max_candidate = conv_candidate.max()

        # accept if we improved the true max, otherwise keep the old point
        # a tiny random acceptance probability helps to escape plateaus
        if (max_candidate < max_conv) or (np.random.rand() < 0.001):
            a = a_candidate
            max_conv = max_candidate

        cur_val = 2.0 * n * max_conv
        if cur_val < best_val:
            best_val = cur_val
            best_a = a.copy()

    # final evaluation with the official function (handles clipping, etc.)
    seq = best_a.tolist()
    try:
        true_val = float(evaluate_sequence(seq))
    except Exception:                     # safety fallback
        conv = np.convolve(best_a, best_a)
        true_val = 2.0 * n * conv.max()
    return seq, true_val


def propose_candidate(seed: int = 42, budget_s: float = 1000, **kwargs) -> list[float]:
    """
    Search for a sequence of non‑negative heights that minimises
    `evaluate_sequence`.  The routine performs many random restarts,
    each followed by a fast projected gradient descent on a smoothed
    L∞ objective.  The best sequence discovered before the time budget
    expires is returned.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 5.0   # keep a small safety margin

    best_seq = None
    best_score = float('inf')
    iteration = 0

    # If a reference solution is available, start from it
    if 'height_sequence_1' in globals():

        except Exception:
            print("[init] Failed to evaluate height_sequence_1 – ignored.")

    while time.time() < deadline:
        # pick a problem size – moderate lengths keep the per‑iteration cost low
        n = np.random.randint(300, 1200)

        # random starting point on the simplex
        a0 = np.random.rand(n)
        a0 /= a0.sum()

        # with small probability, perturb the current best solution (if same length)
        if best_seq is not None and len(best_seq) == n and np.random.rand() < 0.15:

            a0 += 0.01 * np.random.randn(n)      # small Gaussian noise

        if score < best_score:

    return best_seq
