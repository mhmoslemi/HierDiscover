def gradient_step(seq, step_size):
    n = len(seq)
    conv = np.convolve(seq, seq)
    k_max = conv.argmax()
    # gradient vector
    grad = np.zeros(n)
    for i in range(n):
        j = k_max - i
        if 0 <= j < n:
            grad[i] = 2 * seq[j]
    # We aim to move opposite to gradient to reduce M
    # Compute step: seq_new = seq - eta * (grad - mean(grad))
    # Subtract mean to keep sum unchanged (in first order)
    mean_grad = grad.mean()
    delta = - step_size * (grad - mean_grad)
    seq_new = seq + delta
    # enforce nonnegative
    seq_new = np.maximum(seq_new, 0)
    # renorm sum to original sum (scale factor)
    s = seq_new.sum()
    if s < 1e-9:
        # reinitialize random small positive
        seq_new = np.random.rand(n)
        s = seq_new.sum()
    seq_new = seq_new * (seq.sum() / s)
    return seq_new

"""
Propose candidate sequence for the step function optimization.

Algorithm:
1. Generate a handful of deterministic candidate families (geometric,
   linear, cosine) for several lengths.  Each candidate is normalized to
   the required ℓ₁‑norm (sqrt(2·n)).  The best of these serves as a
   starting point.
2. Perform a stochastic local search (simulated annealing) that preserves
   the ℓ₁‑norm while keeping all entries non‑negative.  At each iteration we
   tweak two random coordinates, optionally accept the move if it improves
   the evaluation or with a temperature‑dependent probability.
3. Every few hundred iterations we invoke the linear‑programming “good
   direction” step described in the reference implementation (maximise the
   sum of a vector g under the convolution bound).  We then mix the current
   sequence with g using a line‑search over the mixing coefficient t.
4. The best sequence seen during the whole run is returned.

The whole routine respects the 1000 s time budget.
"""

import numpy as np
import time
from scipy.optimize import linprog

def _normalize(seq):
    """Scale seq so that its ℓ₁‑norm equals sqrt(2·len(seq))."""
    seq = np.asarray(seq, dtype=float)
    total = seq.sum()
    if total == 0:
        seq = np.ones_like(seq)
        total = seq.sum()
    factor = np.sqrt(2.0 * len(seq)) / total
    return (seq * factor).tolist()

def _evaluate(seq):
    """Wrapper around the provided evaluate_sequence (assumed global)."""
    return evaluate_sequence(seq)

def _geometric_seq(n, a):
    """Geometric decreasing sequence a_i = r**i with r = exp(-a/n)."""
    r = np.exp(-a / n)
    seq = r ** np.arange(n, dtype=float)
    return _normalize(seq)

def _linear_seq(n):
    """Linearly decreasing sequence (n, n‑1, …, 1)."""
    seq = np.arange(n, 0, -1, dtype=float)
    return _normalize(seq)

def _cosine_seq(n):
    """Half‑period cosine shape, all entries positive."""
    seq = np.sin(np.pi * (np.arange(n, dtype=float) + 0.5) / n)
    return _normalize(seq)

def solve_convolution_lp(f_sequence, rhs):
    """
    Solve the LP: maximise sum(b) s.t. conv(f, b) ≤ rhs, b ≥ 0.
    Returns the optimal b as a numpy array, or None on failure.
    """
    n = len(f_sequence)
    c = -np.ones(n)                # maximise → minimise –sum
    a_ub = []
    b_ub = []
    # Build constraints for each convolution index
    for k in range(2 * n - 1):
        row = np.zeros(n)
        for i in range(n):
            j = k - i
            if 0 <= j < n:
                row[j] = f_sequence[i]
        a_ub.append(row)
        b_ub.append(rhs)
    # Non‑negativity constraints: –b ≤ 0
    a_ub = np.vstack([a_ub, -np.eye(n)])
    b_ub = np.hstack([b_ub, np.zeros(n)])
    # Use HiGHS interior point solver; give a modest time limit.
    res = linprog(c, A_ub=a_ub, b_ub=b_ub, method='highs', options={'time_limit': 5})
    if res.success:
        return res.x
    return None

def get_good_direction_to_move_into(sequence):
    """Return a new sequence obtained by a convex combination with the LP direction."""
    n = len(sequence)
    seq = np.asarray(sequence, dtype=float)
    total = seq.sum()
    if total == 0:
        return None
    # Normalise to the canonical ℓ₁‑norm
    normalized = (seq * (np.sqrt(2.0 * n) / total)).tolist()
    rhs = np.max(np.convolve(normalized, normalized))
    g = solve_convolution_lp(normalized, rhs)
    if g is None:
        return None
    sum_g = g.sum()
    if sum_g == 0:
        return None
    normalized_g = (g * (np.sqrt(2.0 * n) / sum_g)).tolist()
    # Try a small set of mixing coefficients and keep the best.
    best_seq = None
    best_score = np.inf
    for t in (0.02, 0.05, 0.1, 0.2):
        mixed = [(1.0 - t) * a + t * b for a, b in zip(sequence, normalized_g)]
        sc = _evaluate(mixed)
        if sc < best_score:
            best_score = sc
            best_seq = mixed
    return best_seq

def propose_candidate(seed=0, budget_s=1000, **kwargs):
    """
    Produce a step‑function height vector that (hopefully) improves on the current
    bound of 2.0.  The routine first tests a few analytical families, then runs a
    simulated‑annealing style local search augmented with the LP‑based direction
    described in the problem statement.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 5   # keep a safety margin
    best_seq = None
    best_score = np.inf

    # ------------------------------------------------------------------ #
    # 1.  Hand‑crafted families (geometric, linear, cosine) for several   #
    #     lengths.  The best of these becomes the initial point.          #
    # ------------------------------------------------------------------ #
    lengths = [500, 800, 1200, 1600, 2000]
    for n in lengths:
        # geometric family – sweep a around the theoretically interesting region
        for a in np.linspace(0.8, 3.0, 12):
            seq = _geometric_seq(n, a)
            sc = _evaluate(seq)
            if sc < best_score:
                best_score, best_seq = sc, seq
                print(f"Geom n={n:4d} a={a:4.2f} → {sc:.6f}")
        # linear decreasing
        seq = _linear_seq(n)
        sc = _evaluate(seq)
        if sc < best_score:
            best_score, best_seq = sc, seq
            print(f"Linear n={n:4d} → {sc:.6f}")
        # cosine shape
        seq = _cosine_seq(n)
        sc = _evaluate(seq)
        if sc < best_score:
            best_score, best_seq = sc, seq
            print(f"Cosine n={n:4d} → {sc:.6f}")

    # If we already beat the target we can stop early.
    if best_score < 1.503:
        print(f"Early success: score={best_score:.6f}")
        return best_seq

    # ------------------------------------------------------------------ #
    # 2.  Stochastic local search (simulated annealing).                 #
    # ------------------------------------------------------------------ #
    curr_seq = list(best_seq)        # copy
    curr_score = best_score
    n = len(curr_seq)
    # annealing schedule
    temperature = 1.0
    decay = 0.9995
    it = 0
    last_improve = time.time()

    while time.time() < deadline:
        it += 1
        # --- random two‑coordinate move that preserves the ℓ₁‑norm ------------
        i, j = np.random.randint(n, size=2)
        if i == j:
            continue
        # propose a change delta on coordinate i; compensate on j
        delta = np.random.randn() * temperature
        new_seq = curr_seq.copy()
        # Apply delta to i and compensate on j
        new_i = max(0.0, new_seq[i] + delta)
        diff = new_i - new_seq[i]                  # actual change after clipping
        new_j = max(0.0, new_seq[j] - diff)
        if new_j <= 0.0:
            # reject moves that would make j negative after compensation
            continue
        new_seq[i] = new_i
        new_seq[j] = new_j

        # Rescale to keep the ℓ₁‑norm exactly the same (numerical drift)
        s = sum(new_seq)
        if s == 0:
            continue
        target_sum = sum(curr_seq)
        scale = target_sum / s
        if scale != 1.0:
            new_seq = [x * scale for x in new_seq]

        new_score = _evaluate(new_seq)

        # Metropolis acceptance
        accept = False
        if new_score < curr_score:
            accept = True
        else:
            prob = np.exp((curr_score - new_score) / max(temperature, 1e-8))
            if np.random.rand() < prob:
                accept = True

        if accept:
            curr_seq, curr_score = new_seq, new_score
            if curr_score < best_score:
                best_score, best_seq = curr_score, list(curr_seq)
                print(f"Iter {it:7d} – new best {best_score:.6f}")
                last_improve = time.time()

        # --- occasional LP‑based improvement -------------------------------
        if it % 200 == 0:
            direction = get_good_direction_to_move_into(curr_seq)
            if direction is not None:
                dir_score = _evaluate(direction)
                if dir_score < curr_score:
                    curr_seq, curr_score = direction, dir_score
                    if curr_score < best_score:
                        best_score, best_seq = curr_score, list(curr_seq)
                        print(f"LP step {it:7d} – new best {best_score:.6f}")

        # --- temperature decay ------------------------------------------------
        temperature *= decay

        # If we have been stuck for a long time, restart from the global best.
        if time.time() - last_improve > 30.0:

            jitter = [max(0.0, x + np.random.randn() * 0.01) for x in best_seq]
            jitter = _normalize(jitter)
            curr_seq, curr_score = jitter, _evaluate(jitter)
            last_improve = time.time()

    print(f"Finished – best score {best_score:.6f}")
    return best_seq

"""
Propose candidate sequence for the step‑function optimisation problem.

Goal
----
Return a non‑negative list `a_0,…,a_{n-1}` that minimises

    2·n·max(a*a) / (sum(a)^2)

where `a*a` denotes the discrete convolution of the list with itself.
The function `evaluate_sequence` (already available in the environment) computes
this value with safety checks.

Approach
--------
1. **Deterministic families** – geometric, linear and cosine shapes are built
   for several lengths (≈ 500‑2000).  Each candidate is normalised so that
   `sum = sqrt(2·n)`.  The best of these serves as a starting point.

2. **Stochastic local search** – a simulated‑annealing style hill‑climber
   repeatedly tweaks two random coordinates while preserving the ℓ₁‑norm.
   Moves that improve the score are always accepted; otherwise they are
   accepted with a temperature‑dependent probability.

3. **LP‑based direction** – every few hundred iterations we invoke the
   linear‑programming step described in the statement: maximise the sum of a
   vector `g` under the convolution bound `conv(f,g) ≤ max(conv(f,f))`.  The
   current sequence `f` is mixed with `g` (a line‑search over the mixing weight
   `t`) and the best mix is kept.

4. The algorithm respects a 1000 s budget; the best sequence ever seen is
   returned.

The whole routine runs on a single CPU and uses only `numpy` and
`scipy.optimize.linprog`.
"""

import time
import numpy as np
from scipy.optimize import linprog

# --------------------------------------------------------------------------- #
# Helper utilities ----------------------------------------------------------- #
# --------------------------------------------------------------------------- #

def _normalize(seq):
    """Scale `seq` so that its ℓ₁‑norm equals sqrt(2·len(seq))."""
    seq = np.asarray(seq, dtype=float)
    total = seq.sum()
    if total == 0:
        seq = np.ones_like(seq)
        total = seq.sum()
    factor = np.sqrt(2.0 * len(seq)) / total
    return (seq * factor).tolist()


def _evaluate(seq):
    """Thin wrapper around the provided `evaluate_sequence`."""
    return evaluate_sequence(seq)


def _geometric_seq(n, a):
    """a_i = r**i,   r = exp(-a/n).  Afterwards normalised."""
    r = np.exp(-a / n)
    seq = r ** np.arange(n, dtype=float)
    return _normalize(seq)


def _linear_seq(n):
    """Monotonically decreasing arithmetic sequence (n, n-1, …, 1)."""
    seq = np.arange(n, 0, -1, dtype=float)
    return _normalize(seq)


def _cosine_seq(n):
    """Half‑period cosine shape, all entries > 0."""
    seq = np.sin(np.pi * (np.arange(n, dtype=float) + 0.5) / n)
    return _normalize(seq)


# --------------------------------------------------------------------------- #
# Linear‑programming direction (the “LP step” from the statement) ------------ #
# --------------------------------------------------------------------------- #

def solve_convolution_lp(f_sequence, rhs):
    """
    Solve: maximise sum(b)   s.t.   conv(f, b) ≤ rhs  and  b ≥ 0.
    Returns the optimal b as a NumPy array, or None on failure.
    """
    n = len(f_sequence)
    c = -np.ones(n)                     # maximise → minimise –sum(b)
    a_ub = []
    b_ub = []

    # build one inequality for each entry of the convolution f*b
    for k in range(2 * n - 1):
        row = np.zeros(n)
        for i in range(n):
            j = k - i
            if 0 <= j < n:
                row[j] = f_sequence[i]
        a_ub.append(row)
        b_ub.append(rhs)

    # non‑negativity constraints  -b ≤ 0
    a_ub = np.vstack([a_ub, -np.eye(n)])
    b_ub = np.hstack([b_ub, np.zeros(n)])

    # HiGHS is the default modern solver; give it a modest time limit.
    res = linprog(c, A_ub=a_ub, b_ub=b_ub, method="highs",
                  options={"time_limit": 5})
    if res.success:
        return res.x
    return None


def get_good_direction_to_move_into(sequence):
    """
    Return a new sequence obtained by mixing the current one with an LP‑generated
    direction `g`.  The mix weight `t` is chosen from a short grid to minimise
    the evaluation function.
    """
    n = len(sequence)
    seq = np.asarray(sequence, dtype=float)

    # normalise to the canonical ℓ₁‑norm required by the LP formulation
    total = seq.sum()
    if total == 0:
        return None
    normalized = (seq * (np.sqrt(2.0 * n) / total)).tolist()

    rhs = np.max(np.convolve(normalized, normalized))
    g = solve_convolution_lp(normalized, rhs)
    if g is None:
        return None

    sum_g = g.sum()
    if sum_g == 0:
        return None
    normalized_g = (g * (np.sqrt(2.0 * n) / sum_g)).tolist()

    # small line‑search over mixing coefficients
    best_mix = None
    best_score = np.inf
    for t in (0.02, 0.05, 0.1, 0.2):
        mixed = [(1.0 - t) * a + t * b for a, b in zip(sequence, normalized_g)]
        sc = _evaluate(mixed)
        if sc < best_score:
            best_score, best_mix = sc, mixed
    return best_mix

# Main entry point ---------------------------------------------------------- #
# --------------------------------------------------------------------------- #

def propose_candidate(seed=0, budget_s=1000, **kwargs):
    """
    Produce a step‑function height vector that (hopefully) improves on the
    baseline bound of 2.0.  The routine first tests a few analytical families,
    then runs a simulated‑annealing style local search augmented with the
    LP‑based direction described above.  The best sequence seen is returned.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 5      # small safety margin
    best_seq = None
    best_score = np.inf

    # ------------------------------------------------------------------- #
    # 1)  Hand‑crafted families (geometric, linear, cosine)                #
    # ------------------------------------------------------------------- #
    lengths = [500, 800, 1200, 1600, 2000]
    for n in lengths:
        # geometric sweep – the parameter a≈1‑2 is known to be interesting
        for a in np.linspace(0.8, 3.0, 12):
            seq = _geometric_seq(n, a)
            sc = _evaluate(seq)
            if sc < best_score:
                best_score, best_seq = sc, seq
                print(f"Geometric n={n:4d} a={a:4.2f} → {sc:.6f}")
        # linear decreasing
        seq = _linear_seq(n)
        sc = _evaluate(seq)
        if sc < best_score:
            best_score, best_seq = sc, seq
            print(f"Linear    n={n:4d}    → {sc:.6f}")
        # cosine shape
        seq = _cosine_seq(n)
        sc = _evaluate(seq)
        if sc < best_score:
            best_score, best_seq = sc, seq
            print(f"Cosine    n={n:4d}    → {sc:.6f}")

    # Early exit if we already beat the target
    if best_score < 1.503:
        print(f"Early success – score {best_score:.6f}")
        return best_seq

    # ------------------------------------------------------------------- #
    # 2)  Stochastic local search (simulated annealing)                   #
    # ------------------------------------------------------------------- #
    curr_seq = list(best_seq)                # mutable copy
    curr_score = best_score
    n = len(curr_seq)

    temperature = 1.0
    decay = 0.9995
    it = 0
    last_improve = time.time()

    while time.time() < deadline:
        it += 1

        # --------------------------------------------------------------- #
        # Random two‑coordinate move preserving the ℓ₁‑norm                #
        # --------------------------------------------------------------- #
        i, j = np.random.randint(n, size=2)
        if i == j:
            continue
        delta = np.random.randn() * temperature
        new_seq = curr_seq.copy()

        new_i = max(0.0, new_seq[i] + delta)
        diff = new_i - new_seq[i]                     # actual change after clipping
        new_j = max(0.0, new_seq[j] - diff)
        if new_j <= 0.0:
            continue                                   # reject move that would make j negative

        new_seq[i] = new_i
        new_seq[j] = new_j

        # Renormalise to keep the exact ℓ₁‑norm (numerical drift may happen)
        cur_sum = sum(curr_seq)
        new_sum = sum(new_seq)
        if new_sum == 0:
            continue
        if abs(new_sum - cur_sum) > 1e-12:
            scale = cur_sum / new_sum
            new_seq = [x * scale for x in new_seq]

        new_score = _evaluate(new_seq)

        # Metropolis acceptance criterion
        accept = False
        if new_score < curr_score:
            accept = True
        else:
            prob = np.exp((curr_score - new_score) / max(temperature, 1e-8))
            if np.random.rand() < prob:
                accept = True

        if accept:
            curr_seq, curr_score = new_seq, new_score
            if curr_score < best_score:
                best_score, best_seq = curr_score, list(curr_seq)
                print(f"Iter {it:7d} – new best {best_score:.6f}")
                last_improve = time.time()

        # --------------------------------------------------------------- #
        # Every few hundred iterations invoke the LP “good direction”.   #
        # --------------------------------------------------------------- #
        if it % 200 == 0:
            mixed = get_good_direction_to_move_into(curr_seq)
            if mixed is not None:
                mixed_score = _evaluate(mixed)
                if mixed_score < curr_score:
                    curr_seq, curr_score = mixed, mixed_score
                    if curr_score < best_score:
                        best_score, best_seq = curr_score, list(curr_seq)
                        print(f"LP step {it:7d} – new best {best_score:.6f}")

        # --------------------------------------------------------------- #
        # Temperature decay                                                #
        # --------------------------------------------------------------- #
        temperature *= decay

        # --------------------------------------------------------------- #
        # Restart with a slight jitter if we have been idle for a while   #
        # --------------------------------------------------------------- #
        if time.time() - last_improve > 30.0:
            jitter = [max(0.0, x + np.random.randn() * 0.01) for x in best_seq]
            jitter = _normalize(jitter)
            curr_seq, curr_score = jitter, _evaluate(jitter)
            last_improve = time.time()

    print(f"Finished – best score {best_score:.6f}")
    return best_seq
