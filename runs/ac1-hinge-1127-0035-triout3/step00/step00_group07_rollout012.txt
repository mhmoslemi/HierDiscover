import numpy as np
import time

def propose_candidate(seed=42, budget_s=1000, **kwargs):
    """
    Simple deterministic exploration of power-law step sequences.

    Constructs candidate sequences a_i = (i+1)^p for a range of powers p and lengths n,
    scales them to respect the 0‑1000 bound, evaluates each with `evaluate_sequence`,
    and returns the sequence achieving the smallest score.
    """
    # we won't use seed? we can set np.random.seed for any random perturbation
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5

    best_seq = None
    best_score = float('inf')

    # try a few exponents and lengths
    exponents = [1.0, 1.2, 1.4, 1.6, 1.8, 2.0]
    # lengths up to 500 (ensuring max after scaling <=1000)
    min_len = 5
    max_len = 500

    for p in exponents:
        # choose a set of lengths spaced geometrically
        # use powers of 2 and some intermediate values
        lengths = list(range(min_len, max_len + 1, 5))
        for n in lengths:
            # generate raw sequence (i from 1..n)
            raw = np.arange(1, n + 1, dtype=float) ** p
            max_raw = raw.max()
            # scaling to keep values within 0‑1000
            scale = 1.0
            if max_raw > 1000.0:
                scale = 1000.0 / max_raw
                raw = raw * scale
            seq = raw.tolist()
            try:
                score = evaluate_sequence(seq)
                if score < best_score:
                    best_score = score
                    best_seq = seq
            except Exception:
                pass

            if time.time() >= deadline:
                return best_seq if best_seq is not None else []

    # As a fallback, return a simple quadratic sequence of modest length
    if best_seq is None:
        n = 30
        seq = (np.arange(1, n+1) ** 2).tolist()
        best_seq = seq
    return best_seq

"""
Simple power‑law construction for the step‑function heights.

The evaluation score is
    R = 2 * n * max(a*a) / (sum(a)**2).

For a sequence a_i = (i+1)^p (i = 0 … n‑1) the maximal convolution
value is attained near the centre and, for sufficiently large n,
R approaches the theoretical limit

    R(p) = 2·B(p+1,p+1)·(p+1)^2 ,

where B is the Beta‑function.  As p grows this limit shrinks
(e.g. p=2 → R≈0.6, p=3 → R≈0.23).  By scaling the whole vector we can
keep every entry ≤ 1000 without changing the score.

The function explores a few exponents p and lengths n, rescales each
candidate to satisfy the 0‑1000 bound, evaluates it with the already
provided ``evaluate_sequence`` and keeps the best (lowest) score.
A tiny hill‑climbing phase is performed while time permits.
"""

import time
import numpy as np

def propose_candidate(seed: int = 42, budget_s: float = 1000, **kwargs):
    """Return a sequence of non‑negative heights that yields a small score."""
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5   # leave a tiny margin

    best_seq = None
    best_score = float('inf')

    # ------------------------------------------------------------------
    # 1) Coarse grid search over exponents and lengths.
    # ------------------------------------------------------------------
    exponents = [1.0, 1.2, 1.4, 1.6, 1.8, 2.0, 2.5, 3.0]
    # lengths to try – we go up to a few hundred; larger lengths give almost
    # the asymptotic value but also cost more time.
    lengths = list(range(10, 501, 10))

    for p in exponents:
        for n in lengths:
            # raw power‑law values: (i+1)^p
            i = np.arange(1, n + 1, dtype=np.float64)
            raw = np.power(i, p)

            # Scale uniformly so the maximum does not exceed 1000.
            max_raw = raw.max()
            if max_raw > 1000.0:
                raw = raw * (1000.0 / max_raw)

            seq = raw.tolist()
            try:
                score = evaluate_sequence(seq)
            except Exception:
                # safety – any unexpected failure is ignored
                continue

            if score < best_score:
                best_score = score
                best_seq = seq

            if time.time() >= deadline:
                return best_seq if best_seq is not None else []

    # ------------------------------------------------------------------
    # 2) Small stochastic refinement (hill‑climbing) around the best found.
    # ------------------------------------------------------------------
    if best_seq is None:
        # fallback – a modest quadratic sequence (guaranteed to be valid)
        n = 30
        best_seq = (np.arange(1, n + 1, dtype=np.float64) ** 2).tolist()
        best_score = evaluate_sequence(best_seq)

    # Turn the best sequence into a NumPy array for fast updates.
    cur = np.array(best_seq, dtype=np.float64)

    while time.time() < deadline:
        # pick a random index and propose a tiny perturbation
        idx = np.random.randint(cur.size)
        delta = np.random.randn() * 0.02 * cur[idx]  # 2 % relative change
        new_val = max(0.0, cur[idx] + delta)

        # keep the whole vector within the 0‑1000 bound
        new_seq = cur.copy()
        new_seq[idx] = new_val
        new_seq = np.clip(new_seq, 0.0, 1000.0)

        try:
            new_score = evaluate_sequence(new_seq.tolist())
        except Exception:
            continue

        if new_score < best_score:
            best_score = new_score
            cur = new_seq
            best_seq = cur.tolist()

    return best_seq
