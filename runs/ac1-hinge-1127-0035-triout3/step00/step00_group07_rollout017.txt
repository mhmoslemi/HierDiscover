def proj_simplex(v, s):
    """Project v onto the simplex {x >=0, sum x = s}."""
    n = len(v)
    # Sort v decreasing
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    # Find rho: max i where u_i - cssv_i / (i+1) > 0
    rho = np.where(u - cssv / np.arange(1, n+1) > 0)[0]
    if len(rho) == 0:
        theta = 0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0)
    return w

"""
Propose a sequence of non‑negative heights that (approximately) minimises

    2 * n * max(conv(a, a)) / (sum(a)**2)

The algorithm combines three ideas:

1. **Start from a known construction** (the global `height_sequence_1`
   if it exists, otherwise a uniform sequence).

2. **Projected sub‑gradient descent on the ℓ∞‑norm of the
   self‑convolution**.
   For a fixed sequence `a` the active constraints are the indices where
   the convolution attains its maximum.  The sub‑gradient of the
   ℓ∞‑norm at `a` is
        g_i = 2 * Σ_{k∈Active} a_{k-i},
   which can be computed efficiently with a single correlation.
   After a gradient step the vector is projected back onto the simplex
   `{x≥0, Σx = √(2n)}` (the objective is homogeneous, so the exact
   normalization is irrelevant).

3. **Occasional random perturbations** and restart when progress stalls
   to escape local minima.

The routine runs for the allotted time (default 900 s, leaving a few
seconds margin) and returns the best sequence found.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Helper: Euclidean projection onto the probability simplex (scaled)
def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """
    Project vector v onto the set {x ≥ 0, Σ x = s}.
    Implements the O(n log n) algorithm of Duchi et al.
    """
    if s < 0:
        raise ValueError("Simplex radius must be non‑negative")
    n = v.shape[0]
    # sort v in descending order
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    # find the smallest rho such that u[rho] - cssv[rho]/(rho+1) <= 0
    rho_array = u - cssv / np.arange(1, n + 1)
    rho = np.where(rho_array > 0)[0]
    if len(rho) == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Gradient of the ℓ∞‑norm of the convolution (sub‑gradient)
def _subgradient_max_conv(a: np.ndarray, conv: np.ndarray) -> np.ndarray:
    """
    Compute a sub‑gradient of max(conv(a,a)) w.r.t. a.
    `conv` must be the full linear convolution of `a` with itself.
    The gradient is 2 * correlation(active_weights, a),
    where `active_weights` is 1 on the indices attaining the maximum.
    """
    n = a.shape[0]
    max_val = conv.max()
    # tolerance to gather all active indices (numerical safety)
    eps = max_val * 1e-10 + 1e-12
    active = np.where(conv >= max_val - eps)[0]
    # build weight vector; equal weight for every active entry
    w = np.zeros_like(conv, dtype=float)
    w[active] = 1.0 / max(1, active.size)      # avoid division by zero
    # correlation = conv(w, a) with reversed ordering
    # np.correlate returns Σ_k w_k * a_{k+shift}
    corr = np.correlate(w, a, mode='full')
    offset = len(conv) - 1                     # position of zero shift
    # gradient component i corresponds to shift = -i
    g = 2.0 * corr[offset - np.arange(n)]
    return g

    print(f"[done] best score found: {best_score:.6f} after {iteration} iterations")
    return best_a.tolist()
