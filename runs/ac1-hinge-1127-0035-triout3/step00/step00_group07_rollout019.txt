def propose_candidate(seed=..., budget_s=1000, **kwargs):

    np.random.seed(seed)

    deadline = time.time() + budget_s - 10

    best_seq = None; best_score = np.inf.

    # search over p values.

    n_small = 500  # for quick eval
    # compute scale factor to keep max <= 1000
    for p in np.linspace(0, 3, 31):
        seq = [(i+1)**p for i in range(n_small)]
        # normalize to sum 1 (optional not needed because scaling invariance)
        # For safety scale to max <= 1000
        seq = [float(x) for x in seq]
        max_val = max(seq)
        if max_val > 0:
            scale = 1000.0 / max_val
        else:
            scale = 1.0
        seq = [x*scale for x in seq]
        score = evaluate_sequence(seq)
        if score < best_score:
            best_score = score
            best_p = p

    # refine around best_p
    for p in np.linspace(best_p-0.5, best_p+0.5, 21):
        ... evaluate

    # final large sequence
    n_large = 5661  # from comment, but we can choose same.
    seq = [(i+1)**best_p for i in range(n_large)]
    max_val = max(seq)
    scale = 1000.0 / max_val
    seq = [float(x*scale) for x in seq]

    final_score = evaluate_sequence(seq)
    print("Best p:", best_p, "Score:", final_score)

"""
Catalan based sequence with LP refinement.

The infinite series a_i = C(2i,i)/4^i (the coefficients of (1‑x)^‑½)
has the remarkable property that the full convolution a * a is identically
1.  Truncating to a finite length n gives a non‑negative sequence whose
convolution is 1 for the first n entries and < 1 afterwards, so the
evaluation score is π/2 ≈ 1.571.  To improve on this we start from the
truncated Catalan sequence and repeatedly solve a small linear programme:
given the current sequence f we maximise the total mass of a new sequence g
subject to the linear constraints  (f * g)[k] ≤ M  (where M is the current
maximum convolution value).  Mixing a little of g into f increases the sum
of the coefficients while keeping the convolution bound essentially the
same, which strictly lowers the evaluation score
    2·n·max(f*f) / (sum(f)²)
because the denominator grows and the numerator does not.

The routine runs until the time budget is exhausted (or a few hundred
iterations) and returns the best sequence found.
"""

import time
import numpy as np
from scipy.optimize import linprog


def _catalan_coeffs(n: int) -> np.ndarray:
    """Return the first n Catalan‑type coefficients a_i = C(2i,i)/4**i."""
    a = np.empty(n, dtype=float)
    a[0] = 1.0
    for i in range(1, n):
        # recurrence a_i = a_{i‑1} * (1 - 0.5/i)
        a[i] = a[i - 1] * (1.0 - 0.5 / i)
    return a


def _build_conv_constraints(f: np.ndarray, M: float):
    """
    Build the linear‑programming constraints for
        (f * g)[k] ≤ M  for k = 0 … n‑1
    (the first n convolutions are the active ones for the Catalan seed).
    Returns A_ub, b_ub suitable for scipy.linprog.
    """
    n = f.shape[0]
    rows = []
    # only the first n convolutions are tight for the Catalan seed
    for k in range(n):
        row = np.zeros(n, dtype=float)
        # i runs over indices where both f[i] and g[k‑i] exist
        i_start = max(0, k - (n - 1))
        i_end = min(k, n - 1)
        # fill the row: contribution of g[j] where j = k‑i
        for i in range(i_start, i_end + 1):
            j = k - i
            row[j] = f[i]
        rows.append(row)
    A_ub = np.vstack(rows)
    b_ub = np.full(n, M, dtype=float)
    return A_ub, b_ub


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """Return a height sequence with an evaluation score below the baseline.

    The function runs for at most ``budget_s`` seconds and returns the best
    sequence discovered.  The returned list contains non‑negative floats and
    respects the evaluate_sequence security checks.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 5  # a little safety margin

    # ------------------------------------------------------------------
    # 1.  Initialise with the truncated Catalan series (length ≈ 500)
    # ------------------------------------------------------------------
    n = 500                     # modest size – fast LP solves
    f = _catalan_coeffs(n)     # start vector
    best_seq = f.copy()
    best_score = evaluate_sequence(best_seq.tolist())
    print(f"[init] n={n}, initial score={best_score:.6f}")

    # ------------------------------------------------------------------
    # 2.  Main refinement loop
    # ------------------------------------------------------------------
    iteration = 0
    while time.time() < deadline and iteration < 200:
        iteration += 1

        # current convolution bound
        conv_f = np.convolve(f, f)
        M = float(conv_f.max())

        # build LP: maximise sum(g)  s.t. (f * g)[k] ≤ M   (k=0..n‑1)
        A_ub, b_ub = _build_conv_constraints(f, M)

        # objective: maximise sum(g)  → minimise -1ᵀg
        c = -np.ones(n, dtype=float)

        # solve the LP (highs is fast and reliable for medium‑size problems)
        res = linprog(
            c,
            A_ub=A_ub,
            b_ub=b_ub,
            bounds=[(0, None)] * n,
            method="highs",
            options={"presolve": True, "disp": False, "time_limit": 5},
        )

        if not res.success:
            # fallback: a tiny random perturbation
            idx = np.random.randint(n)
            f_new = f.copy()
            f_new[idx] = max(0.0, f_new[idx] * (1.0 + np.random.uniform(-0.1, 0.1)))
            f = f_new
            continue

        g = res.x
        # mix a fraction of the new direction into f
        t = 0.1  # step size – small enough for stability
        f = (1.0 - t) * f + t * g

        # evaluate the new candidate
        cur_score = evaluate_sequence(f.tolist())
        if cur_score < best_score:
            best_score = cur_score
            best_seq = f.copy()
            print(f"[iter {iteration:3d}] improved score={best_score:.6f}")

    # ------------------------------------------------------------------
    # 3.  Return the best sequence found (as a plain Python list)
    # ------------------------------------------------------------------
    print(f"[final] best score after {iteration} iterations: {best_score:.6f}")
    return best_seq.tolist()
