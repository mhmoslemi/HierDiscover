def propose_candidate(seed=42, budget_s=995, **kwargs):
    """
    Generate a decreasing power‑law step function.
    The sequence is parametrised by an exponent p:
        a_i = 1000 * ((N-1-i)/(N-1))**p
    We sweep p in a coarse grid, keep the best exponent,
    then perform a finer local search. The best sequence
    (lowest evaluate_sequence score) is returned.
    """
    import numpy as np, time, math
    # length
    try:
        N = len(height_sequence_1)
    except NameError:
        N = 1000
    max_value = 1000.0

    best_seq = None
    best_score = float('inf')
    best_p = None

    start = time.time()
    deadline = start + budget_s - 5.0

    np.random.seed(seed)

    # Coarse grid
    coarse_ps = np.linspace(0.5, 9.0, 180)   # around 0.5..9
    for p in coarse_ps:
        # Early exit if out of time
        if time.time() > deadline:
            break
        # build sequence
        # using vectorised computation for speed
        idx = np.arange(N, dtype=float)
        seq = max_value * ((N - 1 - idx) / (N - 1)) ** p
        # convert to list for evaluate_sequence (expects list)
        seq_list = seq.tolist()
        score = evaluate_sequence(seq_list)
        if score < best_score:
            best_score = score
            best_seq = seq_list
            best_p = p

    # Refine around best p if we have a candidate
    if best_p is not None:
        # refine within +-0.5 of best p, 200 steps
        lo = max(0.0, best_p - 0.6)
        hi = best_p + 0.6
        fine_ps = np.linspace(lo, hi, 250)
        for p in fine_ps:
            if time.time() > deadline:
                break
            idx = np.arange(N, dtype=float)
            seq = max_value * ((N - 1 - idx) / (N - 1)) ** p
            seq_list = seq.tolist()
            score = evaluate_sequence(seq_list)
            if score < best_score:
                best_score = score
                best_seq = seq_list
                best_p = p

    # As a final safety, if nothing was found (shouldn't happen) fallback to uniform
    if best_seq is None:
        best_seq = [max_value] * N
        best_score = evaluate_sequence(best_seq)

    # Logging
    print(f"Proposed exponent p = {best_p:.5f}")
    print(f"Length = {len(best_seq)}")
    print(f"Best score = {best_score:.6f}")

    return best_seq

import time
import numpy as np

def propose_candidate(seed=42, budget_s=995, **kwargs):
    """
    Generate a decreasing power‑law step function and search for the exponent
    that minimises ``evaluate_sequence``.

    The sequence is defined by a single parameter `p`:
        a_i = 1000 * ((N-1-i)/(N-1))**p   for i = 0 … N-1
    where `N` is the length of the reference construction `height_sequence_1`
    (or 1000 if the reference is not available).

    The algorithm works in two phases:
      1. A coarse sweep over a wide range of exponents.
      2. A fine sweep around the best exponent found in step 1.

    The sequence with the lowest evaluation score is returned.
    """
    # ----------------------------------------------------------------------
    # 1. Determine the length of the sequence we will work with.
    # ----------------------------------------------------------------------
    try:
        N = len(height_sequence_1)          # use the length of the previous best
    except NameError:
        N = 1000                               # fallback – any positive length works

    MAX_VAL = 1000.0                           # hard cap enforced by evaluate_sequence

    # ----------------------------------------------------------------------
    # 2. Search state.
    # ----------------------------------------------------------------------
    best_seq = None
    best_score = float('inf')
    best_p = None

    start = time.time()
    deadline = start + budget_s - 5.0   # leave a few seconds for clean‑up / printing

    np.random.seed(seed)

    # ----------------------------------------------------------------------
    # 3. Coarse grid search (fast, exploratory).
    # ----------------------------------------------------------------------
    coarse_ps = np.linspace(0.5, 9.0, 180)   # 0.5 … 9.0 with ≈0.05 step
    idx = np.arange(N, dtype=float)        # pre‑compute indices once

    for p in coarse_ps:
        if time.time() > deadline:
            break

        # a_i = MAX_VAL * ((N-1-i)/(N-1))**p   (monotonically decreasing)
        seq = MAX_VAL * ((N - 1 - idx) / (N - 1)) ** p
        seq_list = seq.tolist()        # evaluate_sequence expects a list

        score = evaluate_sequence(seq_list)
        if score < best_score:
            best_score = score
            best_seq = seq_list
            best_p = p

    # ----------------------------------------------------------------------
    # 4. Fine‑grained local search around the best exponent.
    # ----------------------------------------------------------------------
    if best_p is not None:
        lo = max(0.0, best_p - 0.6)
        hi = best_p + 0.6
        fine_ps = np.linspace(lo, hi, 250)   # ≈0.0048 step in the neighbourhood

        for p in fine_ps:
            if time.time() > deadline:
                break

            seq = MAX_VAL * ((N - 1 - idx) / (N - 1)) ** p
            seq_list = seq.tolist()
            score = evaluate_sequence(seq_list)
            if score < best_score:
                best_score = score
                best_seq = seq_list
                best_p = p

    # ----------------------------------------------------------------------
    # 5. Safety net – should the search somehow fail.
    # ----------------------------------------------------------------------
    if best_seq is None:
        best_seq = [MAX_VAL] * N
        best_score = evaluate_sequence(best_seq)

    # ----------------------------------------------------------------------
    # 6. Report the result (print is captured by the runner).
    # ----------------------------------------------------------------------
    print(f"Chosen exponent p = {best_p:.6f}")
    print(f"Sequence length = {len(best_seq)}")
    print(f"Best evaluation score = {best_score:.6f}")

    return best_seq
