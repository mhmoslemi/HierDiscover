def propose_candidate(seed=12345, budget_s=1000, **kwargs):
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5
    # start from height_sequence_1 or random
    if np.random.rand() < 0.7:
        seq = np.array(height_sequence_1, dtype=float)
    else:
        n = np.random.randint(500, 2000)
        seq = np.random.rand(n)
        seq = seq / np.sum(seq) * np.sqrt(2 * n) # optional scaling? evaluate is scale invariant but could be left as is.
    best_seq = seq.copy()
    best_score = evaluate_sequence(seq.tolist())
    # iterate
    step = 0.01  # step size for gradient descent
    while time.time() < deadline:
        # compute conv
        conv = np.convolve(seq, seq)
        M = np.max(conv)
        # indices achieving max (list)
        max_idxs = np.where(conv == M)[0]
        # gradient: each a_i contributes to each max conv index.
        grad = np.zeros_like(seq)
        for k in max_idxs:
            # contributions where i in range
            low = max(0, k - (len(seq) - 1))
            high = min(len(seq) - 1, k)
            # For i from low..high: a_i interacts with a_{k-i}
            # add gradient 2 * a_{k-i}
            grad[low:high+1] += 2 * seq[k - np.arange(low, high+1)]
        # we want to reduce max, so move opposite direction of gradient.
        # also maintain nonnegativity and normalization.
        # compute new seq = seq - eta * grad (project)
        eta = step / (np.linalg.norm(grad) + 1e-12)
        new_seq = seq - eta * grad
        # enforce nonnegativity:
        new_seq = np.maximum(new_seq, 0)
        # renormalize to sum=1 (or keep scaling)
        sum_new = np.sum(new_seq)
        if sum_new < 1e-12:
            # restart random
            new_seq = np.random.rand(seq.shape[0])
            sum_new = np.sum(new_seq)
        new_seq /= sum_new
        # compute score
        score = evaluate_sequence(new_seq.tolist())
        if score < best_score:
            best_score = score
            best_seq = new_seq.copy()
            # optionally increase step
        else:
            # maybe adapt step size or revert.
            # fallback: small random perturbation.
            if np.random.rand() < 0.1:
                idx = np.random.randint(len(seq))
                new_seq[idx] += np.random.randn() * step
                new_seq = np.maximum(new_seq, 0)
                new_seq /= np.sum(new_seq)
        seq = new_seq
    return best_seq.tolist()

"""
A fast search for a step‑function that minimises

    2 * n * max(conv(a,a)) / (sum(a)**2)

The sequence is approximated by a small number of constant blocks.
The block heights are optimised globally with SciPy's differential
evolution and then refined by a tiny random hill‑climbing loop.
Convolution is performed with an FFT (O(n log n)) so thousands of
objective evaluations fit easily into the 1000 s budget.
"""

import time
import numpy as np
from scipy.optimize import differential_evolution

# ----------------------------------------------------------------------
# Helper: fast computation of the evaluation metric (identical to
# evaluate_sequence after the security‑checks).
def _metric(seq: np.ndarray) -> float:
    """Return the value 2*n*max(conv(seq,seq))/sum(seq)^2."""
    # Clip as evaluate_sequence does
    a = np.clip(seq, 0.0, 1000.0)
    total = a.sum()
    if total < 0.01:                # avoid division by near‑zero
        return np.inf

    n = a.size
    # linear convolution via FFT (length >= 2*n-1)
    m = int(2 ** np.ceil(np.log2(2 * n - 1)))
    fa = np.fft.rfft(a, n=m)
    conv = np.fft.irfft(fa * fa, n=m)[: 2 * n - 1]
    max_b = np.max(conv)
    return float(2.0 * n * max_b / (total * total))


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 12345,
                     budget_s: float = 1000.0,
                     **kwargs) -> list[float]:
    """
    Produce a list of non‑negative heights whose evaluation score is
    (hopefully) lower than the current best of about 2.0.

    The algorithm:
      1.  Take the length of the reference sequence ``height_sequence_1``
          (if it exists) otherwise fall back to 1000.
      2.  Approximate the full sequence by K flat blocks (K≈20).
      3.  Optimise the K block values with differential evolution.
          The objective is the metric defined above.
      4.  Run a short random hill‑climbing phase until the time budget
          is exhausted to polish the solution.
      5.  Return the best sequence found as a plain Python list.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5    # keep a tiny safety margin

    # ------------------------------------------------------------------
    # 1. length of the problem
    try:
        n = len(height_sequence_1)           # type: ignore[name-defined]
    except Exception:
        n = 1000                              # a reasonable fallback

    # ------------------------------------------------------------------
    # 2. block partition (K blocks, each roughly n/K long)
    K = 20                                    # number of blocks (adjustable)
    if K > n:
        K = n
    base_len = n // K
    remainder = n % K
    block_lengths = np.full(K, base_len, dtype=int)
    # distribute the remainder to the first blocks (asymmetry is harmless)
    block_lengths[:remainder] += 1

    # ------------------------------------------------------------------
    # 3. optimisation – we treat each block height as a free variable
    def _build_sequence(params: np.ndarray) -> np.ndarray:
        """Expand a vector of K block heights into a full length‑n sequence."""
        seq = np.empty(n, dtype=np.float64)
        start = 0
        for p, L in zip(params, block_lengths):
            seq[start:start + L] = p
            start += L
        return seq

    # objective for DE (lower is better)
    def _obj(params: np.ndarray) -> float:
        seq = _build_sequence(params)
        return _metric(seq)

    # Bounds – block heights are forced positive and limited to 5.0
    # (5 is far below the 1000 cap inside ``evaluate_sequence``)
    bounds = [(1e-5, 5.0) for _ in range(K)]

    # Differential Evolution configuration
    max_iter = 200                               # generous but still cheap
    result = differential_evolution(
        func=_obj,
        bounds=bounds,
        strategy='best1bin',
        maxiter=max_iter,
        popsize=15,
        tol=1e-6,
        mutation=(0.5, 1.0),
        recombination=0.7,
        seed=seed,
        polish=True,
        init='latinhypercube',
        disp=False,
    )

    best_params = result.x
    best_seq = _build_sequence(best_params)
    best_score = _obj(best_params)
    print(f"[DE] best score after global search: {best_score:.6f}")

    # ------------------------------------------------------------------
    # 4. Local random refinement (hill climbing) while time permits
    while time.time() < deadline:

        # enforce bounds
        cand = np.clip(cand, 1e-5, 5.0)
        cand_score = _obj(cand)
        if cand_score < best_score:
            best_score = cand_score
            best_params = cand
            best_seq = _build_sequence(best_params)
            print(f"[HC] improved score: {best_score:.6f}")

        # occasional random restart to escape possible stagnation
        if np.random.rand() < 0.01:
            cand = np.random.uniform(1e-5, 5.0, size=K)
            cand_score = _obj(cand)
            if cand_score < best_score:
                best_score = cand_score
                best_params = cand
                best_seq = _build_sequence(best_params)
                print(f"[RC] new random best: {best_score:.6f}")

    print(f"[FINAL] best evaluation score: {best_score:.6f}")
