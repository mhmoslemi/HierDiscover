"""
Propose a better step‐function height sequence.

The algorithm is a refined version of the SOTA “LP direction + small mixing”
procedure.

1.  Start from a simple uniform sequence (any non‑negative sequence works).
2.  Repeatedly
    a) compute the current maximum convolution value M = max(a*a);
    b) solve a linear program that finds a non‑negative vector g with maximal
       ℓ1‑norm under the linear constraints   a * g ≤ M   (convolution).
    c) rescale g so that its total sum equals the sum of the current sequence,
       and if necessary shrink it so that its *self* convolution never exceeds M
       (this guarantees that mixing will not increase M);
    d) test a few mixing ratios   a_new = (1‑t)·a + t·g   (t ∈ {0.01,…,0.3})
       and keep the best improvement in the evaluation function.
    e) if the LP failed or no improvement was found, apply a tiny random
       perturbation and re‑normalise the sum.

The mixing step reduces the centre peak of the convolution while keeping the
overall bound M non‑increasing.  By repeatedly applying the step we drive the
value 2·n·max(conv)/(sum²) down from the trivial uniform bound 2 towards the
target ≈1.5.

The whole search runs for the allocated time budget (default 1000 s) and
returns the best sequence discovered.
"""

import time
import numpy as np
from scipy.optimize import linprog

# ------------------------------------------------------------
# Linear programme: maximise sum(b) subject to a * b ≤ rhs,
#                     b ≥ 0.
# ------------------------------------------------------------
def solve_convolution_lp(a_seq, rhs):
    """
    Solve  max 1ᵀb   s.t.   conv(a_seq, b) ≤ rhs, b ≥ 0.
    Returns the optimal b as a NumPy array, or None on failure.
    """
    n = len(a_seq)
    # objective: maximise sum(b) → minimise -sum(b)
    c = -np.ones(n)

    # Build the inequality matrix for the convolution constraints.
    # There are (2n‑1) constraints for each possible sum index.
    rows = []
    for k in range(2 * n - 1):
        row = np.zeros(n)
        i_start = max(0, k - (n - 1))
        i_end = min(n - 1, k)
        # i runs over indices of a_seq that contribute to (a*b)[k]
        for i in range(i_start, i_end + 1):
            j = k - i                     # j = k‑i is the index in b
            row[j] = a_seq[i]
        rows.append(row)

    # Non‑negativity constraints b ≥ 0  →  -I·b ≤ 0
    rows.append(-np.eye(n))
    A_ub = np.vstack(rows)
    b_ub = np.concatenate([np.full(2 * n - 1, rhs), np.zeros(n)])

    # Solve with the HiGHS interior point method (fast, robust)
    res = linprog(
        c,
        A_ub=A_ub,
        b_ub=b_ub,
        bounds=[(0, None)] * n,
        method="highs",
        options={"disp": False},
    )
    if res.success:
        # Clip tiny negatives that can appear due to numerical noise
        return np.maximum(res.x, 0.0)
    return None


# ------------------------------------------------------------
# Main candidate generator
# ------------------------------------------------------------
def propose_candidate(seed=12345, budget_s=1000, n=500):
    """
    Return a list of non‑negative heights that (hopefully) improves on the
    uniform bound 2.0.  The routine uses the LP‑based direction finder described
    above and runs until `budget_s` seconds have elapsed.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 5.0   # keep a safety margin

    # ------------------------------------------------------------------
    # initialise: uniform sequence (any non‑negative sequence works)
    # ------------------------------------------------------------------
    seq = np.ones(n, dtype=float)           # sum = n
    best_seq = seq.copy()
    best_score = float('inf')
    try:
        best_score = evaluate_sequence(best_seq.tolist())
    except Exception:
        best_score = float('inf')
    print(f"[init] score={best_score:.6f}, n={n}")

    # candidate mixing ratios to try each iteration
    t_candidates = [0.01, 0.03, 0.07, 0.15, 0.30]

    # ------------------------------------------------------------------
    # iterative improvement loop
    # ------------------------------------------------------------------
    while time.time() < deadline:
        # current maximum of the self‑convolution
        conv = np.convolve(seq, seq)
        M = np.max(conv)

        # --------------------------------------------------------------
        # 1) obtain a promising direction by solving the LP
        # --------------------------------------------------------------
        g = solve_convolution_lp(seq, M)

        if g is None:
            # LP failed – apply a tiny random perturbation and continue
            idx = np.random.randint(n)
            seq[idx] = max(0.0, seq[idx] + np.random.randn() * 0.01)
            # re‑normalise the sum to keep the scale unchanged
            total = np.sum(seq)
            if total > 0:
                seq = seq * (np.sum(best_seq) / total)
            continue

        # --------------------------------------------------------------
        # 2) scale g to keep the total sum unchanged
        # --------------------------------------------------------------
        sum_seq = np.sum(seq)
        sum_g = np.sum(g)
        if sum_g == 0.0:
            continue
        g = g * (sum_seq / sum_g)

        # --------------------------------------------------------------
        # 3) ensure g*g never exceeds the current bound M
        #    (otherwise mixing could increase the maximum)
        # --------------------------------------------------------------
        max_gg = np.max(np.convolve(g, g))
        if max_gg > M:
            factor = np.sqrt(M / max_gg)
            g = g * factor

        # --------------------------------------------------------------
        # 4) try several mixing weights and keep the best improvement
        # --------------------------------------------------------------
        improved = False
        for t in t_candidates:
            cand = (1.0 - t) * seq + t * g
            # numerical safety
            cand = np.maximum(cand, 0.0)
            try:

            except Exception:

            if score < best_score - 1e-12:
                best_score = score
                best_seq = cand.copy()
                seq = cand.copy()
                improved = True
                print(

                )

        if not improved:
            # no improvement – perform a gentle random jiggle to escape
            idx = np.random.randint(n)
            seq[idx] = max(0.0, seq[idx] + np.random.randn() * 0.01)
            # keep the total sum constant
            total = np.sum(seq)
            if total > 0:
                seq = seq * (sum_seq / total)

    # final sanity check – make sure we return a plain Python list
    return best_seq.tolist()
