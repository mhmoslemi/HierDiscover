"""
Propose a sequence of non‑negative heights that minimizes the given evaluation
function.

The algorithm is a simple projected gradient descent on the L2‑deviation of the
self‑convolution from its mean (which is the optimal constant value).  The
gradient of 
    L(a) = Σ_k ( (a*a)_k – mean )²
is   ∇L = 4 * ( ( (a*a)_k – mean )  ⋆  a )      (valid cross‑correlation)
which can be evaluated efficiently with `scipy.signal.correlate`.  

A warm‑start based on the coefficients of the power series of
√(1/(1‑z)) = Σ_{i≥0} C_i z^i (C₀=1, C_{i+1}=C_i·(2i+1)/(i+2)) gives a shape that
already yields a fairly flat convolution.  The sequence is symmetrised,
normalised to the standard sum √(2 n) (the evaluator is scale‑invariant) and
then refined by the gradient steps.  Non‑negativity and the sum constraint are
enforced after each step.  Simple random jitter and occasional restarts avoid
getting stuck in local minima.

The routine runs until the supplied time budget expires (≈ 1000 s) and returns
the best sequence found.
"""

import time
import math

import numpy as np
from scipy.signal import correlate
from scipy import optimize  # only to keep the import environment similar to the starter code


def _initial_sequence(n: int) -> np.ndarray:
    """
    Create a symmetric sequence whose left half follows the coefficients
    C_i = comb(2i,i) / 4**i = Π_{k=0}^{i-1} (2k+1)/(k+2).
    The resulting shape is large at the two ends and decreases toward the centre.
    The sequence is then normalised to have sum sqrt(2*n).
    """
    # compute the decreasing coefficients C_i
    coeff = np.empty(n, dtype=np.float64)
    coeff[0] = 1.0
    for i in range(1, n):
        # C_i = C_{i-1} * (2(i-1)+1) / (i+1)
        coeff[i] = coeff[i - 1] * (2 * i - 1) / (i + 1)

    # make it symmetric: a[i] = coeff[min(i, n-1-i)]
    idx = np.arange(n)
    a = coeff[np.minimum(idx, n - 1 - idx)]

    # normalise the sum to sqrt(2*n) (the evaluation is scale‑invariant)
    target_sum = math.sqrt(2.0 * n)
    a = a / a.sum() * target_sum
    return a

    # ----- optimisation hyper‑parameters ---------------------------------
    step = 0.6 * math.sqrt(2.0 * n)   # initial step magnitude

        b = np.convolve(a, a)                 # length: 2*n-1
