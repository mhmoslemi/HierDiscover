"""
Propose a step‑function sequence with a low evaluation value.

The algorithm works on a moderate length (≈200) to keep the linear‐program
small.  It starts from a geometrically decreasing sequence (which already
has a value < 2) and then repeatedly solves a linear program that finds a
direction `g` with a larger ℓ¹‑norm while keeping the convolution bound
`‖f*g‖∞` no larger than the current bound `M = ‖f*f‖∞`.  A line‑search over
mixing weights `t` is performed and the best improvement is kept.  This
procedure raises the sum of the coefficients without increasing the maximal
convolution value, which drives the metric

    2·n·max(conv) / (sum(a)²)

down.  Random perturbations are added occasionally to escape stagnation.
The whole search respects the 1000 s budget and returns the best sequence
found.
"""

import time
import numpy as np
from scipy.optimize import linprog

# ----------------------------------------------------------------------
# Helper: build the constraint matrix for the LP  conv(f, b) ≤ M
def _build_A_ub(f):
    n = len(f)
    rows = []
    # Convolution constraints: for each k = 0 … 2n-2
    for k in range(2 * n - 1):
        row = np.zeros(n)
        i_start = max(0, k - (n - 1))
        i_end = min(n - 1, k)
        # i runs over the support where j = k-i is valid
        i_vals = np.arange(i_start, i_end + 1, dtype=int)
        j_vals = k - i_vals
        row[j_vals] = f[i_vals]          # contribution of b_j multiplied by f_i
        rows.append(row)
    A_ub = np.vstack(rows)                # shape (2n‑1, n)
    # Non‑negativity of b
    A_ub = np.vstack([A_ub, -np.eye(n)])  # add rows for –b ≤ 0
    return A_ub


# ----------------------------------------------------------------------
# Solve the LP:  maximize sum(b)  s.t.  conv(f, b) ≤ M , b ≥ 0
def _solve_lp(f_norm, M):
    n = len(f_norm)
    c = -np.ones(n)                       # maximise sum(b)  → minimise –sum(b)
    A_ub = _build_A_ub(f_norm)
    b_ub = np.hstack([np.full(2 * n - 1, M), np.zeros(n)])
    # Using the HiGHS solver (fast and reliable for modest sizes)
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, method='highs')
    if res.success:
        return res.x
    return None


# ----------------------------------------------------------------------
# One optimisation step based on the LP described in the statement.
# Returns a list (new sequence) or None if the LP failed.
def _lp_improvement_step(seq, t_candidates):
    n = len(seq)
    total = np.sum(seq)
    if total < 1e-9:            # degenerate, should never happen
        return None

    # Normalise to the standard sum = sqrt(2n) used by the original method
    scale = np.sqrt(2.0 * n) / total
    f_norm = seq * scale

    # Current bound M = max(conv(f_norm, f_norm))
    cur_conv = np.convolve(f_norm, f_norm)
    M = np.max(cur_conv)

    # Solve the linear programme
    g = _solve_lp(f_norm, M)
    if g is None:
        return None

    # Bring g back to the same normalisation as f_norm
    sum_g = np.sum(g)
    if sum_g < 1e-9:
        return None
    g_norm = g * (np.sqrt(2.0 * n) / sum_g)

    # Try several mixing factors and keep the best
    best_local = None
    best_score = np.inf
    for t in t_candidates:
        new_f = (1.0 - t) * f_norm + t * g_norm
        # Undo the temporary normalisation (so the sum stays constant)
        new_seq = new_f / scale
        try:
            sc = evaluate_sequence(new_seq.tolist())
        except Exception:
            continue
        if sc < best_score:
            best_score = sc
            best_local = new_seq
    if best_local is None:
        return None
    return best_local


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a sequence with a low value of `evaluate_sequence`.
    Returns the best sequence found before the time budget expires.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 5.0   # a little safety margin

    # --------------------------------------------------------------
    # 1) initialise – try a few geometric families and keep the best start
    n = 200                                   # moderate size for the LP
    alphas = [0.8, 1.0, 1.2, 1.5]            # decay parameters to try
    best_seq = None
    best_score = np.inf

    for a in alphas:
        r = np.exp(-a / n)
        seq = np.array([r ** i for i in range(n)], dtype=float)
        score = evaluate_sequence(seq.tolist())
        if score < best_score:
            best_score = score
            best_seq = seq.copy()
    # if something went wrong fall back to a uniform start
    if best_seq is None:
        best_seq = np.ones(n, dtype=float)
        best_score = evaluate_sequence(best_seq.tolist())

    cur_seq = best_seq.copy()
    cur_score = best_score
    print(f"Initial length={n}, score={best_score:.6f}")

    # --------------------------------------------------------------
    # 2) main loop – LP improvement + occasional stochastic perturbation
    iter_no = 0
    t_vals = [0.01, 0.05, 0.1, 0.2, 0.4, 0.6]   # mixing weights to test
    while time.time() < deadline:
        # Try an LP‑based improvement step
        new_seq = _lp_improvement_step(cur_seq, t_vals)

        # If the LP step failed (or gave no improvement), fall back to a tiny random move
        if new_seq is None:
            # random local redistribution that keeps the ℓ¹‑norm unchanged
            i, j = np.random.randint(n, size=2)
            if i == j:
                continue
            delta = min(cur_seq[i], cur_seq[j]) * 0.1
            if delta <= 0:
                continue
            r = np.random.randint(n)
            cand = cur_seq.copy()
            cand[i] -= delta
            cand[j] -= delta
            cand[r] += 2.0 * delta
            try:
                new_score = evaluate_sequence(cand.tolist())
            except Exception:
                continue
        else:
            # Evaluate the candidate produced by the LP step
            try:
                new_score = evaluate_sequence(new_seq.tolist())
            except Exception:
                new_seq = None
                continue

        # Accept the candidate if it improves the global best
        if new_seq is not None and new_score < best_score:
            best_score = new_score
            best_seq = new_seq.copy()
            cur_seq = new_seq.copy()
            cur_score = new_score
            print(f"Iter {iter_no:04d}: NEW BEST = {best_score:.6f}")
        else:
            # otherwise keep the current sequence (or replace it with the new one if it is better)
            if new_seq is not None and new_score < cur_score:
                cur_seq = new_seq.copy()
                cur_score = new_score
        iter_no += 1

        # occasional aggressive random perturbation to escape plateaus
        if iter_no % 100 == 0 and np.random.rand() < 0.3:
            idx = np.random.randint(n)
            cur_seq[idx] = max(0.0, cur_seq[idx] + np.random.randn() * 0.05)

    print(f"Search finished after {iter_no} iterations. Best score = {best_score:.6f}")
    return best_seq.tolist()
