Pseudo-code for projecting onto simplex:

def project_onto_simplex(v):
    # v: numpy array
    n = len(v)
    # sort descending
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(n) + 1
    # find rho
    rho = np.where(u - cssv / ind > 0)[0][-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0)
    return w

"""
Propose a low‐value step sequence.

The evaluation metric is
    2·n·max(conv(a,a)) / (sum(a))**2

which is invariant under scaling of the vector.
Hence we fix sum(a)=1 and minimise the peak of the auto‑convolution.
The objective (maximum of several quadratic forms) is non‑convex, so we use a
combination of:

* a simple sub‑gradient descent on the active convolution entry,
* projection onto the probability simplex (a_i ≥ 0, Σa_i = 1),
* occasional random jitter to escape plateaus,
* several independent restarts with different lengths.

The algorithm runs for the allotted time (default 1000 s) and returns the
best sequence found.
"""

import time
import numpy as np
from scipy import optimize

# ----------------------------------------------------------------------
# Helper: project a vector onto the probability simplex Σx_i = 1, x_i ≥ 0
def _project_onto_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the simplex."""
    n = v.shape[0]
    # sort v in decreasing order
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(1, n + 1)
    # find the rho index
    rho = np.where(u - cssv / ind > 0)[0]
    if rho.size == 0:                     # all entries are <= 0
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Gradient of the active convolution entry.
def _conv_gradient(a: np.ndarray, k_max: int) -> np.ndarray:
    """
    Gradient of b_k = (a * a)[k] at the index k = k_max.
    For an ordered convolution the derivative w.r.t. a_i is:
        ∂b_k/∂a_i = a_{k-i}          if i != k-i
        ∂b_k/∂a_i = 2 * a_i          if i == k-i
    """
    n = a.shape[0]
    i = np.arange(n)
    j = k_max - i
    mask = (j >= 0) & (j < n)
    g = np.zeros_like(a)
    # first contribution a_{k-i}
    g[mask] = a[j[mask]]
    # double the diagonal terms (i == j)
    diag = mask & (i == j)
    g[diag] += a[i[diag]]          # now they are 2*a_i
    return g


# ----------------------------------------------------------------------
# One sub‑gradient descent run with a fixed length.
def _run_subgradient(initial_a: np.ndarray, time_limit: float):
    """
    Perform sub‑gradient descent on `initial_a` for at most `time_limit`
    seconds.  Returns the best vector found and its score.
    """
    start = time.time()
    a = initial_a.copy()
    n = a.shape[0]

    # keep the best result seen so far
    best_a = a.copy()
    # since sum(a)=1 we can compute the score directly:
    #   score = 2 * n * max(conv(a,a))
    b = np.convolve(a, a)
    best_score = 2.0 * n * np.max(b)
    iteration = 0

    while time.time() - start < time_limit:
        # convolution and index of the current maximum
        b = np.convolve(a, a)
        k_max = int(np.argmax(b))

        # gradient of the active entry
        g = _conv_gradient(a, k_max)

        # diminishing step size
        eta = 1.0 / np.sqrt(iteration + 1)

        # gradient step (move opposite to the gradient)
        a_candidate = a - eta * g

        # occasional jitter to break symmetry / plateaus
        if iteration % 200 == 0:
            jitter = 0.001 * np.random.randn(n)
            a_candidate += jitter

        # keep the vector on the simplex
        a = _project_onto_simplex(a_candidate)

        # evaluate
        b = np.convolve(a, a)
        cur_score = 2.0 * n * np.max(b)

        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time():.1f}] n={n} iter={iteration:5d}  score={best_score:.6f}")

        iteration += 1

    return best_a, best_score


# ----------------------------------------------------------------------
# Optional LP refinement (the previous SOTA routine).  It can be called a
# few times on the current best vector to squeeze a little more value.
def _solve_convolution_lp(f_seq, rhs):
    """
    Solve the linear program:
        maximise   Σ b_j
        subject to conv(f_seq, b) ≤ rhs,  b ≥ 0
    Returns the optimal b vector (numpy array) or None on failure.
    """
    n = len(f_seq)
    c = -np.ones(n)                     # maximise sum → minimise -sum
    # Build constraints for each convolution index
    A_ub = []
    b_ub = []
    for k in range(2 * n - 1):
        row = np.zeros(n)
        for i in range(n):
            j = k - i
            if 0 <= j < n:
                row[j] = f_seq[i]
        A_ub.append(row)
        b_ub.append(rhs)
    # non‑negativity constraints  -b ≤ 0
    A_ub.extend(-np.eye(n))
    b_ub.extend([0.0] * n)

    res = optimize.linprog(c, A_ub=np.vstack(A_ub), b_ub=np.array(b_ub), method='highs')
    if res.success:
        return np.maximum(res.x, 0.0)
    return None


def _lp_improve(seq, t=0.02):
    """A tiny LP‑based move from the SOTA paper, applied to `seq`."""
    n = len(seq)
    s = np.sum(seq)
    if s < 1e-12:
        return seq
    # Normalise to the convention Σ=√(2n) used in the paper
    norm_factor = np.sqrt(2.0 * n) / s
    f_norm = seq * norm_factor
    rhs = np.max(np.convolve(f_norm, f_norm))
    g = _solve_convolution_lp(f_norm, rhs)
    if g is None:
        return seq
    # renormalise g to the original sum
    g = g * (np.sqrt(2.0 * n) / np.sum(g))
    # blend
    new_seq = (1.0 - t) * seq + t * g
    # project back onto the simplex (sum=1) – the evaluation routine
    # is scale‑invariant, but we keep sum=1 for numerical stability.
    new_seq = _project_onto_simplex(new_seq)
    return new_seq


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: float = 1000, **kwargs):
    """
    Search for a step‑function height vector with a low evaluation score.

    Parameters
    ----------
    seed : int
        Random seed for reproducibility.
    budget_s : float
        Time budget in seconds (default 1000).  The function tries to
        return the best sequence found before the budget expires.

    Returns
    -------
    list[float]
        The best sequence of non‑negative heights found.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 5.0   # small safety margin

    # --------------------------------------------------------------
    # configuration: different lengths we will try (all are modest enough
    # to finish convolution many times within the budget)
    lengths = [512, 1024, 2048, 3072] if budget_s > 30 else [256, 512, 1024]

    global_best_seq = None
    global_best_score = float('inf')

    # --------------------------------------------------------------
    # Main loop over restarts / different lengths
    for n in lengths:
        # allocate a fraction of the remaining time for this length
        now = time.time()
        remaining = deadline - now
        if remaining <= 5.0:
            break
        # cap each run to at most 30 s (or less if time is tight)
        run_time = min(30.0, remaining * 0.6)

        print(f"--- start length {n}, allotted {run_time:.1f}s ---")

        # initialise a random start (or reuse the known construction)
        if np.random.rand() < 0.3 and 'height_sequence_1' in globals():
            start = np.array(height_sequence_1, dtype=float)
            if start.shape[0] != n:
                # cut or pad with tiny noise to get the right dimension
                if start.shape[0] > n:
                    start = start[:n]
                else:
                    extra = np.random.rand(n - start.shape[0]) * 1e-3
                    start = np.concatenate([start, extra])
        else:
            start = np.random.rand(n)

        # normalise to sum‑1 (the evaluation is scale invariant)
        start = _project_onto_simplex(start)

        # run the sub‑gradient optimisation
        best_seq, best_score = _run_subgradient(start, run_time)

        # optional cheap LP polishing (a couple of steps)
        for _ in range(3):
            best_seq = _lp_improve(best_seq, t=0.01)
            score = 2.0 * n * np.max(np.convolve(best_seq, best_seq))
            if score < best_score:
                best_score = score

        print(f"Finished length {n}: best score {best_score:.6f}")

        if best_score < global_best_score:
            global_best_score = best_score
            global_best_seq = best_seq.copy()

        # stop early if we already beat the target
        if global_best_score < 1.55:
            break

    # --------------------------------------------------------------
    # Final sanity check – if nothing was found, fall back to a uniform
    # sequence (which gives a score of exactly 2).
    if global_best_seq is None:
        n = 512
        global_best_seq = np.full(n, 1.0 / n)

    # convert to plain Python list (the evaluation routine expects this)
    result = [float(x) for x in global_best_seq]
    print(f"FINAL RESULT: length={len(result)} score≈{global_best_score:.6f}")
    return result
