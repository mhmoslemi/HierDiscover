def propose_candidate(...):
    n = 2000
    sum_target = math.sqrt(2 * n)
    # start with uniform plus small random noise
    seq = np.full(n, sum_target / n) + np.random.normal(scale=0.01 * sum_target / n, size=n)
    seq = np.maximum(seq, 0)
    seq *= sum_target / np.sum(seq)  # renorm
    best_seq = seq.copy()
    best_score = evaluate_sequence(seq.tolist())
    step = sum_target / n * 0.5   # step magnitude
    temperature = 1.0
    start_time = time.time()
    deadline = start_time + budget_s - 5

    while time.time() < deadline:
        # compute convolution (numpy convolve)
        conv = np.convolve(seq, seq)
        max_idx = int(np.argmax(conv))
        # compute gradient
        grad = np.zeros_like(seq)
        # compute contributions where k_max - i in valid range
        valid_i = np.arange(n)
        j = max_idx - valid_i
        mask = (j>=0) & (j < n)
        grad[mask] = 2 * seq[j[mask]]
        # propose new seq: move opposite gradient
        seq_new = seq - step * grad
        # add small random noise
        seq_new += np.random.normal(scale=step*0.1, size=n)  # noise
        seq_new = np.maximum(seq_new, 0)
        # renormalize sum
        seq_new_sum = seq_new.sum()
        if seq_new_sum == 0:
            continue
        seq_new *= sum_target / seq_new_sum
        # evaluate
        new_score = evaluate_sequence(seq_new.tolist())
        # accept if better or with probability

def improve(seq, steps):
    n = len(seq)
    sum_target = np.sqrt(2 * n)
    for step in range(steps):
        conv = np.convolve(seq, seq)
        max_val = conv.max()
        k = conv.argmax()
        # compute contributions products
        i_arr = np.arange(n)
        j_arr = k - i_arr
        mask = (j_arr >=0) & (j_arr < n)
        # contributions
        contrib = np.zeros(n)
        contrib[mask] = seq[i_arr[mask]] * seq[j_arr[mask]]
        i_max = np.argmax(contrib)   # index with largest product
        j_max = k - i_max
        # choose donors: we will move a fraction from i_max and j_max to other positions
        delta_i = seq[i_max] * 0.1   # 10% from i_max
        delta_j = seq[j_max] * 0.1
        # ensure not exceed
        delta_i = min(delta_i, seq[i_max])
        delta_j = min(delta_j, seq[j_max])
        # choose two recipient indices far from i_max and j_max (random)
        avail = np.arange(n)
        avail = avail[(avail != i_max) & (avail != j_max)]
        r1, r2 = np.random.choice(avail, 2, replace=False)
        seq[i_max] -= delta_i
        seq[j_max] -= delta_j
        seq[r1] += delta_i
        seq[r2] += delta_j
        # clip negatives
        seq = np.maximum(seq, 0.0)
        # renorm
        seq_sum = seq.sum()
        if seq_sum == 0:
            continue
        seq *= sum_target / seq_sum
    return seq

    is returned.
    """
    import numpy as np, time, math, random

    np.random.seed(seed)
    random.seed(seed)

    deadline = time.time() + budget_s - 5

    best_seq = None
    best_score = float('inf')

    # possible lengths to explore (set of lengths up to a few thousand)
    lengths = [600, 800, 1000, 1200, 1500, 2000]

    # Include the length of the known sequence if available
    try:
        known_len = len(height_sequence_1)
        if known_len not in lengths:
            lengths.append(known_len)
    except Exception:
        pass

    # helper to compute the evaluation directly (avoid overhead of evaluate_sequence)
    def fast_eval(seq):
        n = len(seq)
        s = seq.sum()
        if s < 0.01:
            return float('inf')
        conv = np.convolve(seq, seq)
        return float(2.0 * n * conv.max() / (s * s))

    while time.time() < deadline:
        # pick a length
        n = int(np.random.choice(lengths))
        target_sum = math.sqrt(2.0 * n)

        # initialise sequence: either the known one (if same length) or random exponential
        if 'height_sequence_1' in globals() and len(height_sequence_1) == n and np.random.rand() < 0.3:
            seq = np.array(height_sequence_1, dtype=float)
            # ensure it has the right sum (renormalise)
            seq = np.maximum(seq, 0.0)
            seq *= target_sum / seq.sum()
        else:
            # random positive numbers
            seq = np.random.exponential(scale=1.0, size=n)
            seq = np.maximum(seq, 0.0)
            seq *= target_sum / seq.sum()

        current_score = fast_eval(seq)
        # keep if already best
        if current_score < best_score:
            best_score = current_score
            best_seq = seq.copy()
            print(f"Initial best {best_score:.6f} (n={n})")

        # local improvement loop
        # parameters controlling step size
        step_frac = 0.05   # fraction of the value moved each iteration
        max_no_improve = 500   # stop early if no improvement for a while
        no_improve = 0

        while time.time() < deadline:
            # compute convolution and the index of the maximal entry
            conv = np.convolve(seq, seq)
            max_idx = int(conv.argmax())

            # compute the contribution of each index to this peak
            i_arr = np.arange(n)
            j_arr = max_idx - i_arr
            mask = (j_arr >= 0) & (j_arr < n)
            # product contributions
            contrib = np.zeros(n)
            contrib[mask] = seq[i_arr[mask]] * seq[j_arr[mask]]

            # pick the index that participates in the largest product
            i_star = int(contrib.argmax())
            j_star = max_idx - i_star

            # amount to move away (a small fraction of each)
            delta_i = seq[i_star] * step_frac
            delta_j = seq[j_star] * step_frac

            # guard against removing too much
            delta_i = min(delta_i, seq[i_star])
            delta_j = min(delta_j, seq[j_star])

            # choose two distinct recipient positions different from the donors
            candidates = np.setdiff1d(np.arange(n), np.array([i_star, j_star]))
            if candidates.size < 2:
                break
            r1, r2 = np.random.choice(candidates, size=2, replace=False)

            # perform the redistribution
            seq[i_star] -= delta_i
            seq[j_star] -= delta_j
            seq[r1] += delta_i
            seq[r2] += delta_j

            # maintain non‑negativity (tiny negative values can appear due to rounding)
            seq = np.maximum(seq, 0.0)

            # renormalise to keep the L1‑norm exactly sqrt(2*n)
            total = seq.sum()
            if total == 0:
                # re‑initialise if we somehow killed all mass
                seq = np.random.exponential(scale=1.0, size=n)
                seq *= target_sum / seq.sum()
                continue
            seq *= target_sum / total

            # evaluate
            new_score = fast_eval(seq)

            if new_score < current_score - 1e-8:
                current_score = new_score
                no_improve = 0
                if new_score < best_score:
                    best_score = new_score
                    best_seq = seq.copy()
                    print(f"New global best {best_score:.6f} (n={n})")
            else:
                no_improve += 1
                # occasionally accept a worsening move to escape plateaus
                if np.random.rand() < 0.001:
                    current_score = new_score
                if no_improve > max_no_improve:
                    break

            # slowly decay the step size so we fine‑tune near the end
            step_frac *= 0.9999
            if step_frac < 1e-5:
                step_frac = 1e-5

    # return the best sequence found as a plain Python list
    if best_seq is None:
        # as a last resort fallback to a simple uniform sequence
        n = 1000
        best_seq = np.full(n, math.sqrt(2.0 * n) / n)
    return best_seq.tolist()

"""
gradient_anneal_search.py
~~~~~~~~~~~~~~~~~~~~~~~~~

A simple stochastic local‑search that attempts to minimise the evaluation
function

    2 * n * max(conv(a, a)) / (sum(a)**2)

for a non‑negative step‑function a.  The algorithm works as follows:

* Choose a length n (several values are tried) and initialise a candidate
  sequence either uniformly, as a sorted exponential (monotone) or completely
  random.  The sequence is scaled so that its L1‑norm equals sqrt(2*n) –
  this makes the evaluation independent of an overall scale.

* Repeatedly locate the shift k where the full convolution a*a attains its
  maximum.  The product a[i] * a[k‑i] that contributes most to this peak is
  identified.  A small fraction of mass is taken away from the two indices
  involved and redistributed to two other random positions.  Afterwards the
  whole vector is re‑normalised to keep the L1‑norm fixed.

* The move is kept if the score improves; a tiny probability of accepting a
  worse move provides an annealing effect that helps escape plateaus.

* The best sequence found over the whole time budget is returned.

The method needs no external files or network access and stays within the
allowed CPU usage.  It works entirely with numpy which provides fast C‑level
convolution, allowing thousands of iterations within the 1000 s budget.
"""

import time
import math
import random
import numpy as np


def _fast_evaluate(seq: np.ndarray) -> float:
    """Return the same value that evaluate_sequence would give (no clipping)."""
    n = len(seq)
    s = seq.sum()
    if s < 0.01:
        return float("inf")
    conv = np.convolve(seq, seq)
    return float(2.0 * n * conv.max() / (s * s))


def _initial_sequence(n: int, target_sum: float) -> np.ndarray:
    """
    Produce a starting vector of length n with L1‑norm target_sum.
    Three strategies are used at random:
        - uniform,
        - sorted decreasing exponential,
        - plain exponential.
    """
    mode = random.random()
    if mode < 0.33:                      # uniform
        seq = np.full(n, target_sum / n, dtype=float)
    elif mode < 0.66:                    # sorted decreasing exponential
        seq = np.random.exponential(scale=1.0, size=n)
        seq = np.sort(seq)[::-1]        # descending
    else:                                # random exponential
        seq = np.random.exponential(scale=1.0, size=n)
    # enforce non‑negativity and renormalise
    seq = np.maximum(seq, 0.0)
    seq *= target_sum / seq.sum()
    return seq


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a step‑function (list of non‑negative heights) that minimizes
    ``evaluate_sequence``.  The function runs for at most ``budget_s`` seconds
    and returns the best sequence discovered.
    """
    random.seed(seed)
    np.random.seed(seed)

    # time bookkeeping
    deadline = time.time() + budget_s - 5   # leave a safety margin

    best_seq = None
    best_score = float("inf")

    # a handful of lengths to explore – longer vectors give more freedom
    lengths = [600, 800, 1000, 1200, 1500, 2000]

    # include the length of the known construction if it exists
    try:
        known_len = len(height_sequence_1)          # noqa: F821 (global may exist)
        if known_len not in lengths:
            lengths.append(known_len)
    except Exception:
        pass

    while time.time() < deadline:
        # pick a length and the associated target L1‑norm
        n = int(random.choice(lengths))
        target_sum = math.sqrt(2.0 * n)

        # initialise a candidate sequence
        if "height_sequence_1" in globals() and len(height_sequence_1) == n and random.random() < 0.3:
            seq = np.array(height_sequence_1, dtype=float)   # noqa: F821
            seq = np.maximum(seq, 0.0)
            seq *= target_sum / seq.sum()
        else:
            seq = _initial_sequence(n, target_sum)

        score = _fast_evaluate(seq)
        if score < best_score:
            best_score = score
            best_seq = seq.copy()
            print(f"[Init] new best {best_score:.6f} with n={n}")

        # local improvement loop
        step_frac = 0.07            # percentage of a value moved each step
        no_improve = 0
        max_no_improve = 800       # stop local search if stagnant
        while time.time() < deadline:
            conv = np.convolve(seq, seq)
            k_max = int(conv.argmax())

            # indices that take part in the maximal convolution entry
            i_arr = np.arange(n)

            # contribution of each i to that entry (a_i * a_{k-i})
            contrib = np.zeros(n, dtype=float)
            contrib[mask] = seq[i_arr[mask]] * seq[j_arr[mask]]

            i_star = int(contrib.argmax())
            j_star = k_max - i_star

            # amount of mass to move away (fraction of current value)
            delta_i = seq[i_star] * step_frac
            delta_j = seq[j_star] * step_frac
            delta_i = min(delta_i, seq[i_star])

            seq[r1] += delta_i

            step_frac = max(step_frac * 0.9999, 1e-5)

    if best_seq is None:
        n = 1000
        best_seq = np.full(n, math.sqrt(2.0 * n) / n)

    # Convert to plain Python list for the external evaluator
    return best_seq.tolist()
