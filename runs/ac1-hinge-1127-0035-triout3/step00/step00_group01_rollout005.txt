```python
def solve_convolution_lp(f_norm, rhs):
    n = len(f_norm)
    c = -np.ones(n)  # maximize sum(g)
    # Build a_ub: for each k from 0 to 2n-2, conv constraint: sum_i f_i * g_{k-i} <= rhs
    # Represent g variables; each constraint row length n.
    a_ub = np.zeros((2*n-1, n))
    for k in range(2*n-1):
        for i in range(n):
            j = k - i
            if 0 <= j < n:
                a_ub[k, j] += f_norm[i]
    # Nonnegativity: g >= 0 => -I * g <= 0
    a_ub = np.vstack([a_ub, -np.eye(n)])
    b_ub = np.hstack([np.full(2*n-1, rhs), np.zeros(n)])
    # Solve using HiGHS
    res = linprog(c, A_ub=a_ub, b_ub=b_ub, bounds=(0, None), method='highs')
    if res.success:
        return res.x
    return None
```

def solve_convolution_lp(f, M):
    n = len(f)
    # Build A matrix
    rows = 2*n -1
    A = np.zeros((rows, n))
    for j in range(n):
        A[j:j+n, j] = f
    # Upper bounds:
    b = np.full(rows, M, dtype=float)
    # Use bounds directly rather than extra ab_ub for non-negatives:
    bounds = [(0, None) for _ in range(n)]
    # Use hiGHS method for speed
    res = linprog(c=-np.ones(n), A_ub=A, b_ub=b, bounds=bounds, method='highs')
    if res.success:
        return res.x
    else:
        return None
