def subgradient_optimize(n, iters, step0=0.1):
    a = np.full(n, 1.0/n)  # sum=1
    best_a = a.copy()
    best_val = evaluate_sequence(a)

    for t in range(1, iters+1):
        conv = np.convolve(a, a)
        max_b = conv.max()
        # find index of max (choose first)
        ks = np.where(conv == max_b)[0]
        k = ks[0]
        # compute gradient g_i = a_{k-i} (not including factor 2)
        g = np.zeros_like(a)
        for i in range(n):
            j = k - i
            if 0 <= j < n:
                g[i] = a[j]
        # direction d = -g + np.mean(g)
        d = -g + np.mean(g)
        step = step0 / np.sqrt(t)
        a_new = a + step * d
        # project onto non-negative and renorm to sum=1
        a_new = np.maximum(0, a_new)
        s = a_new.sum()
        if s > 0:
            a_new = a_new / s
        else:
            a_new = np.full(n, 1.0/n)
        a = a_new
        val = evaluate_sequence(a)
        if val < best_val:
            best_val = val
            best_a = a.copy()
    return best_a, best_val

def optimize_sequence(n, total_iters, time_deadline):
    a = np.full(n, 1.0/n)  # uniform
    best_seq = a.copy()
    best_val = evaluate_sequence(a)
    beta = 0.1
    step = 0.5 # maybe.

    iteration = 0
    while time.time() < time_deadline and iteration < total_iters:
        # compute conv and evaluation each iteration maybe less frequently
        conv = np.convolve(a, a)
        max_b = conv.max()
        # compute softmax weights
        # use scaling to avoid overflow:
        scaled = beta * conv
        max_scaled = scaled.max()
        exp_scaled = np.exp(scaled - max_scaled)
        p = exp_scaled / exp_scaled.sum()
        # compute gradient
        grad = 2 * np.correlate(p, a, mode='valid')
        # direction with zero sum
        d = -grad + grad.mean()
        # step decreasing
        step_t = step / np.sqrt(iteration+1)
        a_new = a + step_t * d
        # enforce nonnegativity
        a_new = np.maximum(0, a_new)
        sum_a = a_new.sum()
        if sum_a > 0:
            a_new = a_new / sum_a
        else:
            a_new = np.full(n, 1.0/n)
        a = a_new
        # compute evaluate periodically (maybe each iter)
        val = evaluate_sequence(a)
        if val < best_val:
            best_val = val
            best_seq = a.copy()
        # schedule: increase beta gradually
        if (iteration+1) % 200 == 0:
            beta *= 1.5
            if beta > 50:
                beta = 50
        iteration += 1

    return best_seq, best_val

def smooth_optimize(a, beta, step):
    conv = np.convolve(a, a)
    scaled = beta * conv
    max_scaled = scaled.max()
    exp_scaled = np.exp(scaled - max_scaled)
    p = exp_scaled / exp_scaled.sum()
    # gradient
    grad = 2.0 * np.correlate(p, a, mode='valid')
    d = -grad + grad.mean()
    a += step * d
    a = np.maximum(a, 0)
    total = a.sum()
    if total > 0:
        a /= total
    else:
        a = np.full_like(a, 1.0/len(a))
    return a

def grad_optimize(a, deadline):
    best = a.copy()
    best_score = evaluate_sequence(best.tolist())
    iteration = 0
    step0 = 1.0
    beta = 0.2
    while time.time() < deadline:
        conv = np.convolve(a, a)
        # compute p for softmax
        # Use scaling to avoid overflow
        scaled = beta * conv
        max_scaled = np.max(scaled)
        exp_scaled = np.exp(scaled - max_scaled)
        p = exp_scaled / exp_scaled.sum()
        # gradient
        grad = 2.0 * np.correlate(p, a, mode='valid')
        # direction with zero sum
        direction = -grad + grad.mean()
        # step size
        step = step0 / np.sqrt(iteration + 1)
        # update
        a = a + step * direction
        # clip and renorm
        a = np.maximum(0, a)
        sum_a = a.sum()
        if sum_a > 0:
            a = a / sum_a
        else:
            a = np.full_like(a, 1.0/len(a))
        # optional evaluate periodically
        if iteration % 10 == 0:
            score = evaluate_sequence(a.tolist())
            if score < best_score:
                best_score = score

"""
search for a step‑function (non‑negative sequence) that minimises

    2 * n * max(conv(a, a)) / (sum(a) ** 2)

The code combines three complementary heuristics within the allowed
time budget:

1. **Spectral seeding** – generate a zero‑mean signal whose DFT has
   constant magnitude (random phases).  After shifting to make the
   signal non‑negative and normalising, the resulting sequence is a
   good start because its autocorrelation (convolution with itself)
   is already close to flat.

2. **Smoothed gradient descent** – minimise a smooth approximation of
   the max‑convolution (soft‑max).  The gradient is obtained by
   correlating the soft‑max weights with the current sequence; the
   update is projected onto the simplex (non‑negative, sum = 1).

3. **Greedy local refinement** – repeatedly move a small amount of
   mass from a randomly chosen index to another one, keeping the
   sum unchanged.  Improvements are kept; occasional random
   perturbations help escape plateaus.

All three stages are run for a handful of different lengths (up to a
few thousand entries).  The best sequence found before the overall
deadline is returned.
"""

import time
import numpy as np

# -------------------------------------------------------------
# Helper utilities
# -------------------------------------------------------------
def _project_to_simplex(x: np.ndarray) -> np.ndarray:
    """Project a vector onto the probability simplex (non‑negative, sum = 1)."""
    # based on Duchi et al. (2008)
    if x.sum() == 0:
        return np.full_like(x, 1.0 / len(x))
    u = np.sort(x)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(len(x)) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = cssv[-1] / len(x)
    else:
        rho = ind[cond][-1]
        theta = cssv[cond][-1] / rho
    w = np.maximum(x - theta, 0)
    # avoid division by zero
    s = w.sum()
    if s == 0:
        w = np.full_like(w, 1.0 / len(w))
    else:
        w = w / s
    return w


# -------------------------------------------------------------
# 1. Spectral (random‑phase) seeding
# -------------------------------------------------------------
def spectral_sequence(n: int) -> list[float]:
    """
    Produce a length‑n non‑negative sequence using a random‑phase
    spectrum.  The DC component is set to zero, the remaining frequencies
    have unit magnitude and conjugate symmetry, guaranteeing a real‑valued
    time signal with zero mean.  After a shift to make all entries
    non‑negative the sequence is normalised.
    """
    # random phases for frequencies 1 .. n//2 (excluding DC)
    phases = np.random.rand(n // 2 - 1) * 2 * np.pi
    A = np.zeros(n, dtype=np.complex128)

    # DC term = 0 (forces zero mean)
    A[0] = 0.0

    # positive frequencies
    pos = np.arange(1, n // 2)
    A[pos] = np.exp(1j * phases)

    # negative frequencies – complex conjugates
    A[-pos] = np.conj(A[pos])

    # Nyquist frequency for even n must be real
    if n % 2 == 0:
        A[n // 2] = np.random.choice([-1.0, 1.0])

    # inverse FFT, keep the real part
    a = np.fft.ifft(A).real

    # shift to become non‑negative and normalise
    shift = -a.min() + 1e-12
    a += shift
    a /= a.sum()
    return a.tolist()


# -------------------------------------------------------------
# 2. Smoothed gradient descent on the soft‑max of the convolution
# -------------------------------------------------------------
def grad_optimize(initial_seq: list[float], deadline: float) -> tuple[list[float], float]:
    """
    Perform projected gradient descent on the soft‑max of the convolution.
    The objective is   log Σ exp(β * (a * a)),
    whose gradient can be computed efficiently via correlation.
    """
    a = np.array(initial_seq, dtype=np.float64)
    n = len(a)
    best_seq = a.copy()
    best_score = evaluate_sequence(best_seq.tolist())

    iteration = 0
    step0 = 1.0
    beta = 0.2

    while time.time() < deadline:
        conv = np.convolve(a, a)                     # full linear convolution (2n‑1)
        # soft‑max weights
        scaled = beta * conv
        max_scaled = scaled.max()
        weights = np.exp(scaled - max_scaled)
        weights /= weights.sum()

        # gradient of the soft‑max objective w.r.t. a
        grad = 2.0 * np.correlate(weights, a, mode='valid')
        # make the update keep Σ a_i constant (project onto hyperplane)
        direction = -grad + grad.mean()
        step = step0 / np.sqrt(iteration + 1.0)

        a = a + step * direction
        a = np.maximum(a, 0.0)               # non‑negative
        a = _project_to_simplex(a)           # enforce Σ a_i = 1

        if iteration % 10 == 0:
            sc = evaluate_sequence(a.tolist())
            if sc < best_score:
                best_score = sc
                best_seq = a.copy()
        # anneal β slowly – larger β focuses more on the true max
        if iteration % 200 == 0 and iteration > 0:
            beta = min(beta * 1.25, 100.0)

        iteration += 1
        if iteration > 5000:          # safety guard
            break

    return best_seq.tolist(), best_score


# -------------------------------------------------------------
# 3. Greedy local refinement (mass transfer)
# -------------------------------------------------------------
def local_refine(seq: list[float], deadline: float) -> tuple[list[float], float]:
    """
    Simple hill‑climbing: move a small amount of weight from one index to
    another.  Accept only improvements (occasionally accept a worse move
    to escape shallow local minima).  The sum of the sequence stays equal
    to one.
    """
    a = np.array(seq, dtype=np.float64)
    n = len(a)
    best_seq = a.copy()
    best_score = evaluate_sequence(best_seq.tolist())

    while time.time() < deadline:
        i = np.random.randint(n)
        if a[i] <= 0.0:
            continue
        j = np.random.randint(n)
        if i == j:
            continue

        # transfer a small random fraction of a[i] to a[j]
        delta = a[i] * np.random.rand() * 0.07
        a_candidate = a.copy()
        a_candidate[i] -= delta
        a_candidate[j] += delta

        sc = evaluate_sequence(a_candidate.tolist())
        if sc < best_score:
            best_score = sc
            best_seq = a_candidate.copy()
            a = a_candidate
        else:
            # tiny probability to accept a worsening move (simulated‑annealing style)
            if np.random.rand() < 0.001:
                a = a_candidate

        # occasional random perturbation to avoid stagnation
        if np.random.rand() < 0.005:
            idx = np.random.choice(n, size=2, replace=False)
            a[idx] = np.random.rand(2)
            a = _project_to_simplex(a)

    return best_seq.tolist(), best_score
