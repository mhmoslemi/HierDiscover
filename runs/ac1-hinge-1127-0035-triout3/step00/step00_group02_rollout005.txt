def simplex_projection(v):
    # v is a 1D numpy array
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(n) + 1
    rho = np.where(u - cssv / ind > 0)[0][-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0)
    return w

"""
Propose a candidate step‑function (list of non‑negative heights) that minimizes
evaluate_sequence(seq) = 2·n·max(conv(seq,seq))/(sum(seq)**2).

Approach
========
1. **Random sampling** – Generate many Dirichlet (simplex) vectors of various
   lengths and keep the best one seen.
2. **Smooth‑max gradient descent** – Starting from a promising vector, minimise
   a smooth approximation of the maximal convolution entry:
        fβ(a) = (1/β)·log( Σ_k exp(β·c_k) ),  c = conv(a,a)
   The gradient of the inner term is
        ∇c_k = 2·shift_k(a)  (only where indices overlap).
   A simple projected gradient step (onto the probability simplex) reduces the
   smooth‑max and, as β grows, concentrates the optimisation on the true
   infinity‑norm.
3. **Local pairwise hill‑climbing** – Small random mass transfers between two
   coordinates are accepted if they improve the score (simulated‑annealing style).

The whole procedure runs until the allotted time (≈ 1000 s) expires, keeping and
returning the best sequence found.  The sequence is returned as a plain Python
list of floats; the evaluation is scale‑invariant, so we normalise vectors to
sum = 1 for numerical stability.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper functions (no closures, no lambdas)
# ----------------------------------------------------------------------


def _project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex {x≥0, sum x = 1}."""
    if np.all(v <= 0):
        # corner case: all non‑positive -> uniform distribution
        return np.full_like(v, 1.0 / v.size)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(v.size) + 1
    # Find the largest rho where u_i - cssv_i / i > 0
    rho = np.where(u - cssv / ind > 0)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def _smooth_max_weights(conv: np.ndarray, beta: float) -> np.ndarray:
    """
    Compute soft‑max weights that approximate the max of `conv`.
    w_k ∝ exp(beta * (conv_k - max(conv))).
    """
    max_c = np.max(conv)
    # subtracting max keeps numbers well‑scaled
    exp_vals = np.exp(beta * (conv - max_c))
    return exp_vals / np.sum(exp_vals)


def _gradient_step(a: np.ndarray, beta: float, step: float) -> np.ndarray:
    """
    One projected gradient step for the smooth‑max objective.
    a is assumed to lie on the simplex (sum=1, a_i≥0).
    """
    n = a.size
    conv = np.convolve(a, a)
    w = _smooth_max_weights(conv, beta)

    grad = np.zeros_like(a)
    # For each convolution index k, add its contribution to the gradient.
    # d(conv_k)/d a_i = 2 * a_{k-i} whenever the index is valid.
    for k, wk in enumerate(w):
        i_start = max(0, k - (n - 1))
        i_end = min(n - 1, k)
        i_vals = np.arange(i_start, i_end + 1)
        j_vals = k - i_vals
        grad[i_vals] += 2.0 * wk * a[j_vals]

    # take a step opposite to the gradient and re‑project onto the simplex
    a_new = a - step * grad
    a_new = np.maximum(a_new, 0.0)
    a_new = _project_to_simplex(a_new)
    return a_new


def _local_smooth_optimize(a: np.ndarray,
                           max_iter: int,
                           init_step: float = 0.5,
                           init_beta: float = 5.0) -> tuple[np.ndarray, float]:
    """
    Apply several gradient steps with a gradually increasing β.
    Returns the best vector seen and its evaluation score.
    """
    best_a = a.copy()
    best_score = evaluate_sequence(best_a.tolist())
    step = init_step
    beta = init_beta

    for it in range(max_iter):
        a = _gradient_step(a, beta, step)

        # slowly tighten the approximation to the true max
        beta = min(200.0, beta * 1.02)
        # slowly shrink the step size
        step *= 0.997

        cur_score = evaluate_sequence(a.tolist())
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
    return best_a, best_score


def _pairwise_hillclimb(a: np.ndarray,
                        time_deadline: float,
                        init_step_frac: float = 0.1) -> tuple[np.ndarray, float]:
    """
    Small random mass transfers between two coordinates.
    Accepts moves that improve the score (occasionally accepts worse moves
    according to a simple temperature schedule).
    """
    best_a = a.copy()
    best_score = evaluate_sequence(best_a.tolist())
    n = a.size
    rng = np.random.default_rng()

    it = 0
    while time.time() < time_deadline and it < 20000:
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue

        # propose to move a fraction of the donor to the receiver
        donor = i if a[i] > a[j] else j
        receiver = j if donor == i else i
        max_transfer = a[donor]
        if max_transfer == 0.0:
            it += 1
            continue
        delta = init_step_frac * max_transfer * (rng.random() - 0.5)  # can be negative
        if delta > max_transfer:
            delta = max_transfer
        if delta < -a[receiver]:
            delta = -a[receiver]

        new_a = a.copy()
        new_a[donor] -= delta
        new_a[receiver] += delta

        # keep on the simplex (numerical drift)
        new_a = _project_to_simplex(new_a)

        new_score = evaluate_sequence(new_a.tolist())
        # simple simulated‑annealing acceptance
        if new_score < best_score or rng.random() < 0.001:
            a = new_a
            if new_score < best_score:
                best_score = new_score
                best_a = new_a.copy()
        it += 1

    return best_a, best_score

def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a step‑function that minimises `evaluate_sequence`.

    Parameters
    ----------
    seed : int
        Random seed for reproducibility.
    budget_s : int
        Approximate time budget in seconds (the function returns before the
        budget expires).

    Returns
    -------
    list[float]
        The best sequence found.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 1.0  # leave a small safety margin

    best_seq = None
    best_score = float("inf")

    # ------------------------------------------------------------------
    # 1. Use any known construction as a starting point
    # ------------------------------------------------------------------
    if "height_sequence_1" in globals():
        seq0 = np.asarray(height_sequence_1, dtype=float)
        if seq0.size > 0:
            seq0 = _project_to_simplex(seq0)
            sc = evaluate_sequence(seq0.tolist())
            if sc < best_score:
                best_score = sc
                best_seq = seq0.copy()
                print(f"[init] height_sequence_1 → score {best_score:.6f}")

    # ------------------------------------------------------------------
    # 2. Random sampling on several lengths
    # ------------------------------------------------------------------
    lengths = [2, 3, 4, 5, 6, 7, 8, 9, 10,
               12, 15, 20, 30, 40, 50, 75, 100, 150, 200]
    while time.time() < deadline:
        n = int(rng.choice(lengths))
        # Dirichlet (uniform) points on the simplex
        a = rng.random(n)
        a = a / np.sum(a)
        sc = evaluate_sequence(a.tolist())
        if sc < best_score:
            best_score = sc
            best_seq = a.copy()
            print(f"[sample] n={n} → new best {best_score:.6f}")

        # ----------------------------------------------------------------
        # 3. Smooth‑max gradient descent from this random start
        # ----------------------------------------------------------------
        # Determine a safe number of iterations based on remaining time
        remaining = deadline - time.time()
        if remaining <= 0:
            break
        max_iters = min(500, int(remaining * 300))  # a few hundred is cheap
        a_opt, sc_opt = _local_smooth_optimize(a,
                                                max_iter=max_iters,
                                                init_step=0.5,
                                                init_beta=5.0)
        if sc_opt < best_score:
            best_score = sc_opt
            best_seq = a_opt.copy()
            print(f"[opt] n={n}, iters={max_iters} → better {best_score:.6f}")

    # ------------------------------------------------------------------
    # 4. Final fine‑tuning with pairwise hill‑climbing on the current best
    # ------------------------------------------------------------------
    if best_seq is not None:
        a = best_seq.copy()
        a, sc = _pairwise_hillclimb(a, time_deadline=deadline)
        if sc < best_score:
            best_score = sc
            best_seq = a.copy()
            print(f"[hill] final refinement → {best_score:.6f}")

    # ------------------------------------------------------------------
    # 5. Fallback (should never happen)
    # ------------------------------------------------------------------
    if best_seq is None:
        best_seq = np.ones(10, dtype=float)
        best_seq = _project_to_simplex(best_seq)
        best_score = evaluate_sequence(best_seq.tolist())
        print("[fallback] using uniform sequence; score =", best_score)

    print(f"[final] best score {best_score:.6f} on length {len(best_seq)}")
    return best_seq.tolist()
