def gradient_step(a, lr=0.01):
    n = len(a)
    size = next_pow2(2*n-1)
    fft_a = np.fft.fft(a, size)
    conv = np.fft.ifft(fft_a * fft_a).real[:2*n-1]
    max_idx = np.argmax(conv)
    M = conv[max_idx]
    S = a.sum()
    # compute gradient of M
    grad_M = np.zeros_like(a)
    # For each i where j = max_idx - i in [0,n-1], add a[j]
    for i in range(len(a)):
        j = max_idx - i
        if 0 <= j < n:
            grad_M[i] = a[j]
    # gradient of obj w.r.t a_i = (2*n/S**2)*grad_M[i] - (4*n*M/S**3)
    coeff = 2*n / (S**2)
    coeff2 = 4*n * M / (S**3)
    grad = coeff * grad_M - coeff2
    # update
    a_new = a - lr * grad
    a_new[a_new < 0] = 0
    return a_new

"""
Simple stochastic search for a step‑function that reduces the evaluation metric.

The algorithm works in two phases.
* Phase‑1 – random sampling.  Many random sequences of different
  lengths are generated (uniform or exponential distribution, optional
  sparsification).  The best sequence seen so far is kept.
* Phase‑2 – local refinement.  Starting from the best sequence, small
  random perturbations (single‑coordinate moves or pairwise mass transfer)
  are applied; a move is kept when it improves the metric.  The step size
  is slowly reduced and occasional random restarts keep the search from
  getting stuck.

The search runs until the time budget (``budget_s``) is exhausted and
returns the best sequence discovered.
"""

import time
import numpy as np

def _random_sequence(length, rng):
    """Generate a random non‑negative sequence.
    With 50 % probability a uniform distribution is used, otherwise an
    exponential distribution.  About half of the entries are randomly set
    to zero to obtain sparse candidates.
    """
    if rng.rand() < 0.5:
        seq = rng.rand(length)   # uniform [0,1)
    else:
        seq = rng.exponential(scale=1.0, size=length)   # exponential
    # sparsify – turn off about 30 % of the entries
    mask = rng.rand(length) < 0.7
    seq = seq * mask
    # ensure the sequence is not all zeros
    if np.all(seq == 0):
        seq[rng.randint(0, length)] = 1.0
    return seq.tolist()

def _local_step(seq, rng, step):
    """Perform a tiny stochastic modification of ``seq``.
    Returns a new list and the index that was changed (for logging only).
    Two kinds of moves are used:
      * single‑coordinate additive noise,
      * transfer of a small amount of mass from one coordinate to another.
    """
    n = len(seq)
    new_seq = seq.copy()
    if rng.rand() < 0.6:               # additive noise
        i = rng.randint(0, n)
        delta = rng.randn() * step
        new_val = max(0.0, new_seq[i] + delta)
        new_seq[i] = new_val
        return new_seq, (i,)
    else:                               # transfer move
        i, j = rng.choice(n, size=2, replace=False)
        # take a fraction of a_i and add it to a_j
        eps = rng.rand() * step * new_seq[i]
        new_seq[i] = max(0.0, new_seq[i] - eps)
        new_seq[j] = new_seq[j] + eps
        return new_seq, (i, j)

def propose_candidate(seed=42, budget_s=1000, **kwargs):
    """Return a sequence that (hopefully) yields a low value of
    ``evaluate_sequence``.

    The search proceeds as explained in the module docstring.
    ``seed`` determines the random generator state; ``budget_s`` is the
    wall‑clock time budget in seconds.
    """
    rng = np.random.RandomState(seed)
    deadline = time.time() + budget_s - 0.5          # reserve a little margin

    # ------------------------------------------------------------------
    # 1)  Random sampling phase
    # ------------------------------------------------------------------
    best_seq = None
    best_score = float('inf')

    # Use a variety of lengths; keep them moderate so that each evaluation
    # is cheap.  The maximal length is limited by the budget.
    base_len = 300
    if 'height_sequence_1' in globals():
        base_len = max(200, len(height_sequence_1) // 4)

    length_choices = [int(base_len * f) for f in (0.5, 0.75, 1.0, 1.25, 1.5, 2.0)
                      if int(base_len * f) >= 20 and int(base_len * f) <= 4000]

    # At most a few thousand random candidates – this part is very cheap.
    while time.time() < deadline and best_score > 1.55:
        n = int(rng.choice(length_choices))
        cand = _random_sequence(n, rng)
        try:
            score = evaluate_sequence(cand)
        except Exception:
            continue
        if score < best_score:
            best_score = score
            best_seq = cand
            print(f"[sample] new best score {best_score:.6f}  len={n}")

    # If nothing was found (should not happen) fall back to a small constant.
    if best_seq is None:
        best_seq = [1.0] * 10
        best_score = evaluate_sequence(best_seq)

    # ------------------------------------------------------------------
    # 2)  Local refinement phase
    # ------------------------------------------------------------------
    seq = np.array(best_seq, dtype=float)
    step = max(seq.mean() * 0.2, 0.05)  # initial magnitude of perturbations

    # Main optimisation loop – simple stochastic hill climbing.
    while time.time() < deadline:
        # propose a random move
        new_seq, changed = _local_step(seq.tolist(), rng, step)
        try:
            new_score = evaluate_sequence(new_seq)
        except Exception:
            continue

        # Accept the move only if it improves the objective.
        if new_score < best_score:
            best_score = new_score
            best_seq = new_seq
            seq = np.array(new_seq, dtype=float)
            # occasionally shrink the step size – successful moves tend to
            # indicate that we are approaching a basin of attraction.
            step = max(step * 0.95, 0.001)
            print(f"[refine] improved to {best_score:.6f}  len={len(seq)}  step={step:.5f}")
            # Early exit if we reach the announced target.
            if best_score < 1.52:
                break
        else:
            # With a small probability we still accept a non‑improving move.
            if rng.rand() < 0.005:
                seq = np.array(new_seq, dtype=float)

        # Occasionally perform a random restart to avoid getting stuck.
        if rng.rand() < 0.001:
            n = int(rng.choice(length_choices))
            seq = np.array(_random_sequence(n, rng), dtype=float)
            step = max(seq.mean() * 0.2, 0.05)

    # ``evaluate_sequence`` silently clamps values to [0,1000]; the
    # optimisation never produces huge entries, but a final pass through
    # the clamping logic keeps us safe.
    final_seq = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    print(f"[final] best score {best_score:.6f}  length {len(final_seq)}")
    return final_seq

"""
Simple constructive & local‑search optimizer for the step‑function metric.

The idea is to place a relatively small set of “heavy’’ entries whose pairwise
sums are all distinct (a Sidon set).  With distinct sums the autocorrelation
peak contributed by the heavy entries is just their own square (≈1).  All
remaining positions are filled with a tiny value `δ`.  The tiny entries raise
the total L1‑norm (the denominator) while contributing only a very small amount
to the maximal convolution value (the numerator).  By tuning `δ` we can drive
the evaluation score well below the constant‑sequence value 2, aiming for the
target ≈1.5.

The algorithm proceeds in two phases:

1. **Construction phase** – for a handful of candidate lengths we
   *build a random Sidon set* (greedy, using a shuffled order) and try a
   logarithmic grid of `δ` values.  The best sequence found in this phase is
   kept.

2. **Refinement phase** – a tiny stochastic hill‑climber (single‑coordinate
   additive noise or a mass‑transfer move) is applied to the current best
   sequence until the time budget is exhausted.  Moves are accepted only if
   they improve the score; a very small probability of uphill moves
   helps escape shallow plateaus.

The whole search runs for at most ``budget_s`` seconds and returns the best
sequence discovered.
"""

import time
import numpy as np
from math import sqrt

def _greedy_sidon(n: int, rng: np.random.RandomState) -> list[int]:
    """
    Build a Sidon set (all i+j are distinct) inside {0,…,n‑1}
    using a greedy algorithm on a random permutation of the indices.
    Returns the list of selected positions.
    """
    used_sums = set()
    positions = []
    order = rng.permutation(n)
    for i in order:
        # self‑sum must be free
        if i + i in used_sums:
            continue
        conflict = False
        for p in positions:
            if i + p in used_sums:
                conflict = True
                break
        if conflict:
            continue
        # accept i – mark all new sums
        for p in positions:
            used_sums.add(i + p)
        used_sums.add(i + i)
        positions.append(i)
    return sorted(positions)


def _candidate_from_sidon(n: int, rng: np.random.RandomState,
                         target_score: float = 1.503) -> tuple[list[float], float]:
    """
    For a given length ``n`` build a Sidon set of heavy entries (weight 1)
    and try a range of tiny weights ``δ`` for the remaining positions.
    Returns the best sequence and its score.
    """
    heavy = _greedy_sidon(n, rng)
    m = len(heavy)
    if m == 0:                     # fallback – shouldn't happen
        seq = rng.rand(n).tolist()
        return seq, evaluate_sequence(seq)

    best_seq = None
    best_val = float('inf')

    # try deltas ranging from 1e‑4 up to 1e‑1 (log‑space)
    for delta in np.logspace(-4, -1, num=30):
        seq = np.full(n, delta, dtype=float)
        seq[heavy] = 1.0
        try:
            val = evaluate_sequence(seq.tolist())
        except Exception:
            continue
        if val < best_val:
            best_val = val
            best_seq = seq.tolist()
            # early exit if we already beat the target comfortably
            if best_val < target_score * 0.95:
                break
    return best_seq, best_val


def _local_step(seq: list[float], rng: np.random.RandomState,
                step: float) -> list[float]:
    """
    Produce a slight random perturbation of ``seq``.
    Two move types are used:
      * additive Gaussian noise on a single coordinate,
      * transfer of a random fraction of mass from one index to another.
    The resulting sequence is clipped to the non‑negative orthant.
    """
    a = np.array(seq, dtype=float)
    n = a.size
    if rng.rand() < 0.6:                       # additive noise
        i = rng.randint(0, n)
        a[i] = max(0.0, a[i] + rng.randn() * step)
    else:                                      # mass‑transfer move
        i, j = rng.choice(n, size=2, replace=False)
        eps = rng.rand() * step * a[i]         # take a fraction of a[i]
        a[i] = max(0.0, a[i] - eps)
        a[j] = a[j] + eps
    return a.tolist()


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a sequence that yields a low value of ``evaluate_sequence``.
    The function respects the ``budget_s`` wall‑clock limit (seconds) and
    returns the best sequence found within that time.
    """
    rng = np.random.RandomState(seed)
    deadline = time.time() + budget_s - 0.5   # spare a bit for cleanup

    # ------------------------------------------------------------------
    # 1) Construction phase: sweep a few lengths and delta values.
    # ------------------------------------------------------------------
    best_seq = None
    best_score = float('inf')

    # Choose a range of lengths (moderate size keeps convolution cheap)
    length_candidates = [600, 900, 1200, 1500, 1800, 2100, 2500, 3000]
    for n in length_candidates:
        if time.time() >= deadline:
            break
        seq, val = _candidate_from_sidon(n, rng, target_score=1.503)
        if seq is None:
            continue
        if val < best_score:
            best_score = val
            best_seq = seq
            print(f"[construction] n={n}  score={best_score:.6f}")

        # a tiny random tweak can already improve the best found
        # (helps break ties between different δ values)
        noisy = _local_step(seq, rng, step=0.01)
        try:
            v2 = evaluate_sequence(noisy)
            if v2 < best_score:
                best_score, best_seq = v2, noisy
                print(f"[construction‑noise] n={n}  score={best_score:.6f}")
        except Exception:
            pass

    # If nothing succeeded, fall back to a short random run
    if best_seq is None:
        best_seq = rng.rand(10).tolist()
        best_score = evaluate_sequence(best_seq)

    # ------------------------------------------------------------------
    # 2) Local refinement (tiny stochastic hill‑climbing)
    # ------------------------------------------------------------------
    cur_seq = best_seq[:]
    step_size = max(np.mean(cur_seq) * 0.2, 1e-3)

    while time.time() < deadline:
        # propose a random move
        cand = _local_step(cur_seq, rng, step=step_size)
        try:
            cand_score = evaluate_sequence(cand)
        except Exception:
            continue

        # accept only improving moves (very small uphill chance)
        if cand_score < best_score:
            best_score = cand_score
            best_seq = cand
            cur_seq = cand
            # shrink step size slowly – successful moves indicate we are
            # approaching a basin of attraction.
            step_size = max(step_size * 0.96, 1e-4)
            print(f"[refine] score={best_score:.6f}  step={step_size:.5f}")
            # stop early if we already beat the announced target comfortably
            if best_score < 1.50:
                break
        else:
            if rng.rand() < 0.003:        # occasional uphill move
                cur_seq = cand

        # occasional random restart to avoid stagnation
        if rng.rand() < 0.001:
            n_restart = rng.choice(length_candidates)
            cur_seq = rng.rand(n_restart).tolist()
            step_size = max(np.mean(cur_seq) * 0.2, 1e-3)

    # Final safety: clamp to the range accepted by ``evaluate_sequence``.
    final_seq = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    print(f"[final] best score {best_score:.6f}  length {len(final_seq)}")
    return final_seq
