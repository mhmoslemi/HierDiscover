def proj_simplex(v):
    # v is vector, return w s.t. w >=0, sum=1 closest in Euclidean norm.
    # algorithm: sort v descending, compute rho etc.
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    rho = np.where(u - cssv / (np.arange(1, len(v)+1)) > 0)[0][-1]
    theta = cssv[rho] / (rho+1)
    w = np.maximum(v - theta, 0)
    return w

    # maybe random start to get diversity
    if np.random.rand() < 0.5:
        # load height_sequence_1, convert to w (normalize to sum 1)
        seq = height_sequence_1
        seq = np.array(seq, dtype=float)
        if len(seq) < n: # pad or truncate

        else:

        sum_seq = np.sum(seq)
        # normalize to sum sqrt(2n): w = seq / sum_seq
        w = seq / sum_seq
        w = w / np.sum(w)  # should sum to 1

    best_w = w.copy()
    best_val = evaluate_sequence((best_w * np.sqrt(2 * n)).tolist())
    iteration = 0

    while time.time() < deadline:
        iteration += 1
        # compute convolution
        conv = compute_convolution(w)  # length 2n-1
        # find max
        max_val = conv.max()
        # active indices within tolerance
        tol = 1e-12
        active = np.where(conv >= max_val - tol)[0]
        # compute subgrad
        subgrad = np.zeros_like(w)
        for k in active:
            i_start = max(0, k - (n - 1))
            i_end = min(n - 1, k)
            # generate indices
            # i range
            idx = np.arange(i_start, i_end + 1)
            subgrad[idx] += w[k - idx]
        # gradient descent step
        # step size schedule: decreasing
        eta = 0.5 / (iteration ** 0.5)  # tune
        w_new = w - eta * subgrad
        # project onto simplex

        if np.random.rand() < 0.05:
            noise = np.random.randn(n) * 0.01

        # Evaluate using evaluate_sequence with scaling sqrt(2n)
        seq = (w_new * np.sqrt(2 * n)).tolist()
        cur_val = evaluate_sequence(seq)

        if iteration % 1000 == 0:

                # random restart
                w = np.random.dirichlet(np.ones(n))
                continue
        w = w_new

    # final best sequence: convert to heights
    best_seq = (best_w * np.sqrt(2 * n)).tolist()
    # ensure sequence non-negative float
    best_seq = [float(x) for x in best_seq]

    Uses a simple projected (or multiplicative) gradient descent on the
    probability simplex, aiming to flatten the distribution of the sum of
    two independent draws from the sequence.  The method is fast (FFT‑based
    convolution) and runs until the time budget is exhausted.
    """

    w = w / np.sum(w)

    best_w = w.copy()
    best_score = 2.0 * n * np.max(np.convolve(w, w))  # initial evaluation
    iteration = 0

    # Helper: project onto simplex (if needed)
    def project_simplex(v):
        """Euclidean projection onto the probability simplex."""
        u = np.sort(v)[::-1]
        cssv = np.cumsum(u) - 1

        else:

    def self_convolution(x):

        fx = np.fft.rfft(x, size)

        conv = self_convolution(w)

def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step‑function that minimises
    ``evaluate_sequence`` (i.e. minimises the peak of the aperiodic
    auto‑convolution).

    The algorithm works on the *probability simplex*.
    Let `w` be a vector with non‑negative entries and sum 1.
    Setting the actual heights as ``a = w * sqrt(2·n)`` guarantees that the
    L1‑norm is ``sqrt(2·n)`` – the same normalisation used in the reference
    implementation – and the evaluation value equals ``2·n·max(conv_w)``.
    Consequently minimising the maximal convolution entry of `w` is exactly
    the required optimisation.

    A simple projected multiplicative‑gradient descent is performed:

    * compute the self‑convolution of `w` (FFT‑based, O(n log n));
    * find the index/indices where the convolution attains its maximum;
    * the sub‑gradient of the max‑function is the (cross‑)correlation of `w`
      with the indicator of the active set;
    * update `w` with a decreasing step size using the exponential update
      (always stays non‑negative) and renormalise to keep it on the simplex;
    * keep the best vector seen during the run.

    The method is fast, works for the large sequences used in the competition
    (the current best construction has length ~ 8000) and respects the
    1000 s time budget.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 5.0          # a small safety margin

    # ----------------------------------------------------------------------
    # Sequence length – use the length of the existing construction if
    # available, otherwise fall back to a moderate size.
    # ----------------------------------------------------------------------
    try:
        n = len(height_sequence_1)                 # type: ignore[name-defined]
    except Exception:
        n = 2000

    # normalising factor sqrt(2·n) (the total height after scaling)
    norm = np.sqrt(2.0 * n)

    # ----------------------------------------------------------------------
    # Initialise the probability vector w (sum = 1).
    # ----------------------------------------------------------------------
    if "height_sequence_1" in globals():

    else:

    # Helper utilities

        u = np.sort(v)[::-1]

            theta = cssv[rho] / (rho + 1)
        return np.maximum(v - theta, 0.0)

        size = 1 << ((2 * len(x) - 1).bit_length())   # next power of two
        fx = np.fft.rfft(x, size)
        conv = np.fft.irfft(fx * fx, size)[: 2 * len(x) - 1]

    best_w = w.copy()

    conv0 = self_convolution(w)

        iteration += 1

        conv = self_convolution(w)
        max_val = conv.max()

        if cur_score < best_score:
            best_score = cur_score
            best_w = w.copy()
            print(f"Iter {iteration:5d} – new best {best_score:.6f}")

        # --------------------------------------------------------------

        # --------------------------------------------------------------

        for k in active:
