from scipy.optimize import linprog

def solve_lp_max_sum(f, M):
    n = len(f)
    # Build a_ub matrix
    rows = []
    for k in range(2*n-1):
        row = np.zeros(n)
        # i from max(0, k-(n-1)) to min(n-1, k)
        i_min = max(0, k-(n-1))
        i_max = min(n-1, k)
        # for each i, j = k-i
        # coef for variable b_j is f_i
        # So row[j] += f_i
        # Equivalent: for j in values:
        # row[j] = f_{k-j}
        # Use vectorized: for j in range(n):
        # idx = k - j
        # if 0 <= idx < n: row[j] = f[idx]
        for j in range(n):
            i = k - j
            if 0 <= i < n:
                row[j] = f[i]
        rows.append(row)
    a_ub = np.vstack(rows)
    b_ub = np.full(2*n-1, M)
    # add nonnegativity constraints
    a_ub = np.vstack([a_ub, -np.eye(n)])
    b_ub = np.concatenate([b_ub, np.zeros(n)])
    c = -np.ones(n)  # maximize sum => minimize -sum
    res = linprog(c, A_ub=a_ub, b_ub=b_ub, method='highs', options={'time_limit':5})
    if res.success:
        return np.maximum(res.x, 0.0)  # ensure nonneg
    return None
```

def improve_sequence(f, max_iters, time_deadline):
    n = len(f)
    S = np.sqrt(2*n)
    # Normalize f to sum S
    f = f * (S / np.sum(f))
    best_seq = f.copy()
    best_M = np.max(np.convolve(f, f))
    while time.time() < time_deadline:
        M = np.max(np.convolve(f, f))
        if M < best_M - 1e-9:
            best_M = M
            best_seq = f.copy()
        # Solve LP
        g0 = solve_lp_max_sum(f, M)
        if g0 is None:
            # perturb?
            idx = np.random.randint(n)
            f[idx] = max(0, f[idx] + np.random.randn()*0.01)
            continue
        Sg0 = np.sum(g0)
        if Sg0 <= 0:
            continue
        # scale g to sum S
        g = g0 * (S / Sg0)
        # try mixing with several t values
        improved = False
        for t in [0.5, 0.3, 0.2, 0.1, 0.05, 0.02, 0.01]:
            f_candidate = (1 - t) * f + t * g
            M_candidate = np.max(np.convolve(f_candidate, f_candidate))
            if M_candidate < M - 1e-12:
                f = f_candidate
                improved = True
                break
        if not improved:
            # fallback smaller t random?
            idx = np.random.randint(n)
            f[idx] = max(0, f[idx] + np.random.randn()*0.01)
    return best_seq

"""
Propose a step‑function (non‑negative sequence) that minimizes
evaluate_sequence().  

The algorithm is an iterative improvement inspired by the published
LP‑based scheme:

1.  Choose a sequence length n (default 300) and initialise a non‑negative
    vector f with the required normalisation  Σf = √(2n).  The initial
    vector is taken either from the global ``height_sequence_1`` (if
    present) or from a small random perturbation of a uniform vector.

2.  While time permits:
        *  Compute M = max(conv(f, f)).
        *  Solve the linear program
               maximise   Σb
               subject to conv(f, b) ≤ M   (componentwise)
                         b ≥ 0
           The feasible set contains many extreme points; the solver
           usually returns a “spike’’ vector (all mass in a single
           position).  The solution is called g₀.
        *  Rescale g₀ to the fixed sum S = √(2n):
               g = g₀ * (S / Σg₀)
        *  Try a convex combination  f′ = (1‑t)·f + t·g  with a list of
           decreasing step sizes t = [0.5,0.3,0.2,0.1,0.05,0.02,0.01,0.005].
           The first t that yields a strictly smaller max‑convolution
           (i.e. max(conv(f′,f′)) < M) is accepted.
        *  If none of the t‑values improve M, a tiny random perturbation
           of a single coordinate is applied and the vector is
           re‑normalised to the same sum.

   Any improvement (a lower M – which equals the evaluation score because
   the sum is fixed) is recorded; the best sequence seen so far is
   returned.

2 LP solves per iteration are cheap for n≈300, so many iterations can be
performed within the 1000 s budget, driving the score below the trivial
upper bound 2.0 and typically reaching the target ≈1.5.

The function respects all restrictions (no filesystem/IO, single CPU,
explicit `propose_candidate` definition, no closures, no lambda
expressions).
"""

import time
import numpy as np
from scipy import optimize

def _max_convolution(seq: np.ndarray) -> float:
    """Return max of the full self‑convolution of a non‑negative vector."""
    conv = np.convolve(seq, seq)
    return float(conv.max())

def _solve_lp_max_sum(f: np.ndarray, M: float):
    """
    Solve   max Σb   s.t.   conv(f, b) ≤ M   (componentwise),   b ≥ 0.
    Returns the optimal b as a NumPy array, or None if the LP fails.
    """
    n = f.size
    # Build matrix A_ub for constraints conv(f,b) ≤ M.
    rows = []
    for k in range(2 * n - 1):
        # row[j] = f[k-j] if 0 ≤ k-j < n else 0
        row = np.zeros(n)
        # indices j such that i = k-j is in [0,n)
        # equivalent to j ∈ [max(0, k-(n-1)), min(n-1, k)]
        j_min = max(0, k - (n - 1))
        j_max = min(n - 1, k)
        # fill the slice
        idx = k - np.arange(j_min, j_max + 1)
        row[j_min:j_max + 1] = f[idx]
        rows.append(row)

    A = np.vstack(rows)                     # shape (2n‑1, n)
    b = np.full(2 * n - 1, M, dtype=float) # RHS for convolution bounds

    # non‑negativity: -I · b ≤ 0
    A = np.vstack([A, -np.eye(n)])
    b = np.concatenate([b, np.zeros(n)])

    c = -np.ones(n)                         # minimise –Σb  ⇒ maximise Σb
    # a modest per‑LP time limit; the problems are small.
    res = optimize.linprog(c, A_ub=A, b_ub=b, method='highs',
                           options={'time_limit': 2.0})
    if res.success:
        # Numerical noise may produce tiny negatives – clip to zero.
        return np.maximum(res.x, 0.0)
    return None
