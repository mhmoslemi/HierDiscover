def propose_candidate(budget_s=900, seed=0, **kwargs):
    np.random.seed(seed)
    deadline = time.time() + budget_s
    n = 400  # maybe a bit larger e.g., 800
    # initial sequence: random positive values (e.g., uniform)
    seq = np.random.rand(n)
    seq = seq / np.sum(seq)  # normalize sum=1 (objective scale-invariant)
    best_seq = seq.copy()
    best_score = evaluate_sequence(seq.tolist())
    temperature = 1.0
    temp_decay = 0.99995
    step_fraction = 0.05  # fraction of source value to transfer
    while time.time() < deadline:
        # compute conv and M
        conv = np.convolve(seq, seq)
        M = conv.max()
        idx = conv.argmax()
        # compute contributions for each i
        partners = idx - np.arange(n)
        # mask for valid partners
        valid = (partners >= 0) & (partners < n)
        contrib = np.zeros(n)
        contrib[valid] = seq * seq[partners[valid]]
        # choose i_max with high contribution, i_min with low contribution
        i_max = np.argmax(contrib)
        # for i_min choose randomly among low contributions (including zeros)
        low_indices = np.where(contrib <= np.percentile(contrib, 10))[0]
        if low_indices.size == 0:
            i_min = np.random.randint(n)
        else:
            i_min = np.random.choice(low_indices)
        # transfer mass
        delta = step_fraction * seq[i_max]  # ensure not exceed
        if delta > 0:
            new_seq = seq.copy()
            new_seq[i_max] -= delta
            new_seq[i_min] += delta
            # keep non-negative
            # evaluate
            cur_score = evaluate_sequence(new_seq.tolist())
            if cur_score < best_score:
                best_score = cur_score
                best_seq = new_seq.copy()
                seq = new_seq
            else:
                # Simulated annealing probability
                if np.exp(-(cur_score - best_score) / temperature) > np.random.rand():
                    seq = new_seq
        # temperature schedule
        temperature *= temp_decay
        # occasionally random perturbation to escape stagnation
        if np.random.rand() < 0.001:
            # random reposition of a small amount
            i = np.random.randint(n)
            delta = step_fraction * seq[i]
            j = np.random.randint(n)
            if i != j:
                new_seq = seq.copy()
                new_seq[i] -= delta
                new_seq[j] += delta
                cur_score = evaluate_sequence(new_seq.tolist())
                if cur_score < best_score:
                    best_score = cur_score
                    best_seq = new_seq.copy()
                    seq = new_seq

"""
Propose a new height sequence for the step‑function optimisation problem.

The algorithm works in two phases.
* Phase 1 – random sampling: many random Dirichlet‑distributed vectors are
  generated (the sum is fixed – scaling does not affect the objective).  The
  best sequence seen so far is kept.
* Phase 2 – guided local search: starting from the best sample we repeatedly
  reduce the maximal convolution value.  At each iteration we locate the
  index `k` where the convolution `c = f*f` attains its maximum.  The term
  `f[i] * f[k‑i]` (when the index is inside the array) is the contribution
  of the pair `(i, k‑i)` to that maximal value.  We move a small amount of
  mass from a pair with the largest contribution to a pair with a small
  contribution, thereby flattening the convolution.  Simple simulated‑annealing
  logic is used to occasionally accept worsening moves and to escape local
  minima.  The total sum of the sequence is preserved (by a renormalisation)
  so that the objective is invariant under scaling.

The whole search runs for the allotted time (default 950 s).  The function
returns the best sequence discovered.
"""

import time
import numpy as np

# evaluate_sequence is provided by the evaluation harness.


def _random_dirichlet(n: int) -> np.ndarray:
    """
    Produce a random non‑negative vector of length n with sum equal to 1.
    Using a Dirichlet(1,…,1) distribution (i.e. normalised exponentials).
    """
    vec = np.random.exponential(1.0, size=n)
    vec /= vec.sum()
    return vec


def _convolution_max(seq: np.ndarray) -> tuple[float, int, np.ndarray]:
    """
    Compute the linear convolution of seq with itself.
    Returns (max_value, argmax_index, full_convolution_array).
    """
    conv = np.convolve(seq, seq)
    idx = int(np.argmax(conv))
    return float(conv[idx]), idx, conv


def _transfer_mass(seq: np.ndarray, idx_max: int, step_frac: float) -> np.ndarray:
    """
    Perform a single mass‑transfer move:
    * locate the pair (i, j) that contributes most to the maximal convolution
      value (i + j = idx_max)
    * locate a pair that contributes least (or zero)
    * move a fraction `step_frac` of the source entry to the target entry
      while preserving the total sum.
    The function returns the new sequence (may be identical if no move is
    possible).
    """
    n = seq.size
    partners = idx_max - np.arange(n)                     # partner index for each i
    valid = (partners >= 0) & (partners < n)
    contrib = np.zeros(n, dtype=seq.dtype)
    # contribution of each i to the maximal convolution entry
    contrib[valid] = seq[valid] * seq[partners[valid]]

    # source – largest contribution
    i_src = int(np.argmax(contrib))
    # target – smallest contribution (prefer zero‑contribution indices)
    # we take the 10 % smallest values to have a set of candidates.
    thresh = np.percentile(contrib, 10)
    candidates = np.where(contrib <= thresh)[0]
    if candidates.size == 0:
        i_tgt = int(np.argmin(contrib))
    else:
        i_tgt = int(np.random.choice(candidates))

    if i_src == i_tgt or seq[i_src] == 0.0:
        # fall back to a tiny random perturbation
        i_src = int(np.random.randint(n))
        i_tgt = int(np.random.randint(n))
        while i_tgt == i_src:
            i_tgt = int(np.random.randint(n))

    delta = step_frac * seq[i_src]
    new_seq = seq.copy()
    new_seq[i_src] -= delta
    new_seq[i_tgt] += delta
    # Numerical safety – clip tiny negatives that may appear due to rounding
    new_seq[new_seq < 0] = 0.0
    # Renormalise to keep the sum exactly 1 (objective is scale‑invariant)
    total = new_seq.sum()
    if total == 0.0:                     # pathological – restart from random
        new_seq = _random_dirichlet(n)
    else:
        new_seq /= total
    return new_seq


def propose_candidate(seed: int = 42,
                     budget_s: float = 950,
                     *,
                     length: int = 500,
                     **kwargs) -> list[float]:
    """
    Search for a sequence with a low evaluation score.

    Parameters
    ----------
    seed : int
        Random seed for reproducibility.
    budget_s : float
        Time budget in seconds (the harness adds a small safety margin).
    length : int
        Length of the sequence used during optimisation.  The final sequence
        returned has the same length.

    Returns
    -------
    List[float]
        The best sequence found within the time budget.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 5.0      # keep a small safety margin

    # ------------------------------------------------------------------
    # Phase 1 – random sampling
    # ------------------------------------------------------------------
    best_seq = None
    best_score = float('inf')
    n = length

    # a few quick random trials to obtain a decent starting point
    sample_trials = 3000
    for _ in range(sample_trials):
        cand = _random_dirichlet(n)
        try:
            score = evaluate_sequence(cand.tolist())
        except Exception:
            continue
        if score < best_score:
            best_score = score
            best_seq = cand
    if best_seq is None:                     # very unlikely, but be safe
        best_seq = _random_dirichlet(n)
        best_score = evaluate_sequence(best_seq.tolist())

    print(f"[init] best score after random sampling: {best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 2 – guided local optimisation
    # ------------------------------------------------------------------
    cur_seq = best_seq.copy()
    step_fraction = 0.05          # how much of the source mass is moved each step
    temperature = 1.0
    temp_decay = 0.9999

    # statistics for adaptive logging
    last_log = time.time()
    it = 0

    while time.time() < deadline:
        it += 1
        max_val, idx_max, _ = _convolution_max(cur_seq)

        # propose a new sequence by moving mass away from the strongest pair
        new_seq = _transfer_mass(cur_seq, idx_max, step_fraction)

        # evaluate
        try:
            new_score = evaluate_sequence(new_seq.tolist())
        except Exception:
            # any numerical problem – skip this candidate
            continue

        # accept if better, otherwise accept with Metropolis probability
        if new_score < best_score:
            best_score = new_score
            best_seq = new_seq.copy()
            cur_seq = new_seq
        else:
            # simulated‑annealing acceptance
            prob = np.exp(-(new_score - best_score) / max(1e-9, temperature))
            if prob > np.random.rand():
                cur_seq = new_seq

        # annealing schedule
        temperature *= temp_decay
        # slowly shrink the step size to fine‑tune near the end
        if it % 2000 == 0:
            step_fraction = max(0.001, step_fraction * 0.9)

        # periodic logging (roughly every 20 s)
        if time.time() - last_log > 20.0:
            elapsed = time.time() - (deadline - budget_s)
            print(f"[{elapsed:6.1f}s] iter {it:6d} best={best_score:.6f} "
                  f"cur={new_score:.6f} step={step_fraction:.4f}")
            last_log = time.time()

        # occasional restart if we appear stuck (no improvement for a while)
        if it % 10000 == 0 and best_score > 1.6:
            cur_seq = _random_dirichlet(n)
            step_fraction = 0.05
            temperature = 1.0
            print("[restart] re‑initialising with a fresh random point")

    print(f"[final] best score obtained: {best_score:.6f} after {it} iterations")
    # return a plain Python list as required by the harness
    return best_seq.tolist()
