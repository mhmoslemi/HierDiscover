        else:

            delta = (np.random.rand() - 0.5) * 0.1 * max(cand)

"""
Local search that flattens the self‑convolution of a non‑negative step function.

The algorithm starts from the provided construction (``height_sequence_1``) or,
if unavailable, from a random initialization.  It repeatedly:

1. Evaluates the full convolution and finds the index ``k_max`` where the
   convolution attains its maximum.
2. Forms a list of the paired contributions ``a[i] * a[k_max-i]``.
3. Moves a small amount of mass from a pair that contributes heavily to a
   pair that contributes only little, keeping the total sum unchanged.
   This reduces the peak of the convolution while preserving the
   denominator of the evaluation function.
4. Occasionally shuffles the whole sequence or performs a tiny Gaussian
   perturbation to escape local plateaus.
5. Accepts a new sequence if it improves the objective or with a tiny
   probability (simulated‑annealing style).  The best sequence seen so far
   is stored and returned at the end of the allotted budget.

The score being minimised is exactly the one used by ``evaluate_sequence``:
    2 * n * max(conv) / (sum(a)^2)

All numbers are kept in the allowed range [0, 1000] and non‑negative.
The routine is deliberately simple, fast and runs within the 1000 s budget.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities
# ----------------------------------------------------------------------


def compute_score(seq):
    """
    Compute the objective 2*n*max(conv(seq,seq))/sum(seq)^2 with the same
    clamping rules as ``evaluate_sequence``.
    """
    # protect against invalid values (the search never creates them,
    # but we stay consistent with the evaluator)
    seq = np.maximum(0.0, np.minimum(1000.0, np.asarray(seq, dtype=float)))
    n = len(seq)
    if n == 0:
        return np.inf
    conv = np.convolve(seq, seq)
    max_b = np.max(conv)
    sum_a = np.sum(seq)
    if sum_a < 0.01:
        return np.inf
    return float(2.0 * n * max_b / (sum_a ** 2))


def improve_convolution(seq, epsilon_factor=0.02):
    """
    One local‑move step aiming at lowering the peak of the self‑convolution.

    The step:
        * finds the index k_max where conv(seq,seq) is maximal,
        * selects a “high‑contribution” pair (i, j) with i+j = k_max,
        * selects a “low‑contribution” pair,
        * transfers a small amount of mass from the high index to the low one.
    The total sum stays unchanged.
    """
    n = len(seq)
    if n == 0:
        return seq

    # full convolution – used to locate the maximal bin
    conv = np.convolve(seq, seq)
    k_max = int(conv.argmax())

    # collect contributions a[i] * a[k_max-i] for all i that have a partner
    contributions = []
    for i in range(n):
        j = k_max - i
        if 0 <= j < n:
            contributions.append((seq[i] * seq[j], i, j))

    if not contributions:
        return seq

    # sort by contribution size
    contributions.sort(key=lambda x: x[0], reverse=True)

    # split into a small “high” set and a small “low” set
    top_k = max(1, len(contributions) // 10)          # 10 % of the pairs
    high_set = contributions[:top_k]
    low_set = contributions[-top_k:]

    # randomly pick a pair from each set
    _, i_h, j_h = high_set[np.random.randint(len(high_set))]
    _, i_l, j_l = low_set[np.random.randint(len(low_set))]

    # decide which side of each pair will give or receive mass
    src = i_h if seq[i_h] >= seq[j_h] else j_h
    dst = i_l if seq[i_l] <= seq[j_l] else j_l

    # magnitude of the transfer (1–2 % of the source value)
    eps = epsilon_factor * seq[src]
    eps = min(eps, seq[src])               # never subtract more than we have
    if eps < 1e-12:
        return seq

    new_seq = np.array(seq, copy=True)
    new_seq[src] = max(0.0, new_seq[src] - eps)
    new_seq[dst] = min(1000.0, new_seq[dst] + eps)
    return new_seq.tolist()


def random_perturb(seq, scale=0.05):
    """Add a tiny Gaussian noise to a random entry."""
    if not seq:
        return seq
    i = np.random.randint(len(seq))
    delta = (np.random.rand() - 0.5) * 2.0 * scale * max(seq)
    new_seq = np.array(seq, copy=True)
    new_seq[i] = max(0.0, min(1000.0, new_seq[i] + delta))
    return new_seq.tolist()


# ----------------------------------------------------------------------
# Main entry point required by the evaluation framework
# ----------------------------------------------------------------------


def propose_candidate(seed=0, budget_s=1000, **kwargs):
    """
    Return a list of non‑negative heights that (hopefully) yields an
    evaluation score below the current 2.0 bound.

    The search is a simple stochastic hill‑climber that repeatedly
    flattens the convolution peak (see ``improve_convolution``) and
    occasionally shuffles or perturbs the current sequence.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.2   # small safety margin

    # --------------------------------------------------------------
    # initialise the sequence
    # --------------------------------------------------------------
    if 'height_sequence_1' in globals():
        # start from the known construction but add a slight random mask
        base = np.asarray(height_sequence_1, dtype=float)
        mask = 0.5 + np.random.rand(len(base))  # random factor in [0.5,1.5)
        seq = (base * mask).tolist()
    else:
        # fallback: a random sequence of moderate length
        length = np.random.randint(500, 1500)
        seq = (np.random.rand(length) * 20.0).tolist()  # small values to begin

    # enforce the limits that ``evaluate_sequence`` applies
    seq = [max(0.0, min(1000.0, float(x))) for x in seq]

    best_seq = seq.copy()
    best_score = compute_score(best_seq)
    cur_seq = seq.copy()
    cur_score = best_score

    # --------------------------------------------------------------
    # search loop
    # --------------------------------------------------------------
    iteration = 0
    last_improve_time = time.time()
    while time.time() < deadline:
        iteration += 1

        # occasionally restart from the best observed sequence with a fresh perturb
        if iteration % 5000 == 0 and time.time() - last_improve_time > 30:
            cur_seq = random_perturb(best_seq, scale=0.5)

        # decide which elementary move to apply
        if np.random.rand() < 0.15:
            # pure random shuffle – can dramatically change the convolution pattern
            cand_seq = cur_seq.copy()
            np.random.shuffle(cand_seq)
        elif np.random.rand() < 0.7:
            # targeted move that tries to lower the maximal convolution bin
            cand_seq = improve_convolution(cur_seq, epsilon_factor=0.02)
        else:
            # small Gaussian perturbation for fine tuning
            cand_seq = random_perturb(cur_seq, scale=0.02)

        cand_score = compute_score(cand_seq)

        # keep the absolute best sequence seen so far
        if cand_score < best_score:
            best_score = cand_score
            best_seq = cand_seq.copy()
            last_improve_time = time.time()
            print(f"[{time.time():.1f}] Iter {iteration:6d} – new best score: {best_score:.6f}")

        # a simple Metropolis‑style acceptance for the current walk
        if cand_score <= cur_score or np.random.rand() < 0.01:
            cur_seq = cand_seq
            cur_score = cand_score

        # occasional aggressive random restart if we are stuck
        if iteration % 2000 == 0 and np.random.rand() < 0.05:
            cur_seq = np.random.rand(len(cur_seq)) * 20.0
            cur_seq = [float(x) for x in cur_seq]

    # --------------------------------------------------------------
    # final reporting
    # --------------------------------------------------------------
    print(f"Search finished after {iteration} iterations.")
    print(f"Best score obtained: {best_score:.6f} (length {len(best_seq)})")
    return best_seq
