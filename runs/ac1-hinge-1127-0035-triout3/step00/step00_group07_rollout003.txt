    np.random.seed(seed)
    deadline = time.time() + budget_s - 5

    # Choose length n; if height_sequence_1 exist use its length else default 2000
    try:
        n = len(height_sequence_1)
    except NameError:
        n = 2000

    # Initialize a random Dirichlet vector
    a = np.random.dirichlet(np.ones(n))
    # Ensure positivity
    # Normalization already sum=1

    # Adam parameters
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    lr = 0.1
    alpha = 1.0
    alpha_max = 20.0
    alpha_inc = (alpha_max - alpha) / 5000  # approximate

    best_seq = a.copy()
    best_score = evaluate_sequence(best_seq.tolist())

    iteration = 0
    while time.time() < deadline:
        iteration += 1
        # convolution
        b = np.convolve(a, a)
        # smooth max weights
        # increase alpha gradually
        if alpha < alpha_max:
            alpha = min(alpha_max, alpha + alpha_inc)

        max_b = np.max(b)
        # Want smooth max approximated by softmax
        # Use exp with alpha scaling to approximate max
        # compute w as softmax of alpha*b
        # shift for numeric stability
        b_shift = b - max_b
        exp_vals = np.exp(alpha * b_shift)
        w = exp_vals / np.sum(exp_vals)

        # gradient via convolution
        # gradient_i = 2 * sum_k w_k * a_{k-i}
        grad = 2 * np.convolve(w, a[::-1], mode='full')[n-1:n-1+n]

        # Adam update
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad ** 2)
        m_hat = m / (1 - beta1 ** iteration)
        v_hat = v / (1 - beta2 ** iteration)
        a = a - lr * m_hat / (np.sqrt(v_hat) + eps)

        # projection onto non-negative simplex
        a = np.maximum(a, 0)
        s = a.sum()
        if s == 0 or np.isnan(s):
            # reinitialize
            a = np.random.dirichlet(np.ones(n))
            m = np.zeros_like(a)
            v = np.zeros_like(a)
            continue
        a = a / s

        # occasional random noise to avoid stagnation
        if iteration % 500 == 0:
            noise = np.random.randn(n) * 0.01
            a += noise
            a = np.maximum(a, 0)
            a /= a.sum()

        # evaluate occasionally
        if iteration % 10 == 0:
            score = evaluate_sequence(a.tolist())
            if score < best_score:
                best_score = score
                best_seq = a.copy()
                print(f"Iter {iteration}: new best {best_score:.6f}")

    print(f"Finished after {iteration} iterations. Best score {best_score:.6f}")
    return best_seq.tolist()

"""
Propose a sequence that minimizes
    2 * n * max(conv(a, a)) / (sum(a) ** 2)

The algorithm is a simple first‑order method that directly reduces the
maximum of the self‑convolution.  The idea is:

*   The objective depends only on the shape of the vector a (it is
    homogeneous of degree 0), therefore we keep the vector on the
    probability simplex (non‑negative, sum = 1).

*   We minimise a smooth upper bound on the ℓ∞‑norm of the convolution,
    namely the “log‑sum‑exp’’ (soft‑max).  Its gradient can be written
    as a convolution, so each iteration only needs two O(n²) convolutions
    (the convolution itself and the gradient).

*   An Adam optimiser is used together with projection onto the simplex.
    The soft‑max temperature α is increased slowly – at the beginning the
    algorithm follows a smooth landscape, later it concentrates on the
    actual maximum.

*   Random restarts and occasional Gaussian noise help to escape
    plateaus.

The code respects the 1000 s budget, prints progress and finally returns
the best sequence found.
"""

import time
import numpy as np

def _softmax_weights(b, alpha):
    """Return soft‑max weights w that sum to 1 (shape len(b))."""
    # shift by the maximum for numerical stability
    b_max = np.max(b)
    exp_vals = np.exp(alpha * (b - b_max))
    # all exp_vals are ≤ 1, so overflow is impossible
    w = exp_vals / np.sum(exp_vals)
    return w


def _gradient_from_weights(w, a):
    """
    Compute ∇_a ( Σ_k w_k * (a * a)_k )  = 2 * conv(w, a[::-1]).
    The returned gradient has the same length as a.
    """
    n = a.shape[0]
    conv_full = np.convolve(w, a[::-1], mode='full')
    # The relevant slice is from n‑1 (inclusive) to n‑1+n (exclusive)
    grad = 2.0 * conv_full[n - 1 : n - 1 + n]
    return grad
