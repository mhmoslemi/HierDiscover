def golden_section_minimize(func, a, b, tol=1e-5, max_iter=30):
    phi = (np.sqrt(5) - 1) / 2  # 0.618...
    c = b - phi*(b-a)
    d = a + phi*(b-a)
    fc = func(c)
    fd = func(d)
    for i in range(max_iter):
        if b - a < tol:
            break
        if fc < fd:
            b = d
            d = c
            fd = fc
            c = b - phi*(b-a)
            fc = func(c)
        else:
            a = c
            c = d
            fc = fd
            d = a + phi*(b-a)
            fd = func(d)
    # return center point
    if fc < fd:
        return c, fc
    else:
        return d, fd
```

def simplex_projection(v):
    # v is 1D numpy array
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0)
    return w

    random.seed(seed)
    np.random.seed(seed)
    deadline = time.time() + budget_s - 2.0
    best_seq = None
    best_score = float('inf')
    # candidate lengths

    for n in lengths:
        if time.time() >= deadline:
            break
        # search for best geometric r

        a = lo
        b = hi
        c = b - phi * (b - a)
        d = a + phi * (b - a)
        # evaluate function
        def eval_r(r):
            seq = geometric_sequence(n, r)
            return evaluate_sequence(seq)

                c = d

def geometric_sequence(n, r):
    # produce a list of length n, nonnegative, sum=1
    # compute powers r^i using np.exp(i * np.log(r))
    i = np.arange(n, dtype=np.float64)
    a = np.power(r, i)
    total = a.sum()
    if total <= 0:
        # fallback to uniform
        a = np.ones(n, dtype=np.float64) / n
    else:
        a = a / total
    return a.tolist()

"""
Propose a low‑evaluation step function.

The objective of `evaluate_sequence` is   f(a)=2·n·max(a*a)/(sum a)^2 .
For a fixed length `n` we minimise the maximal self‑convolution value.
The algorithm works in two phases:

1.  **LP‑driven improvement** – starting from a random non‑negative
    sequence we repeatedly solve a linear programme:
        maximise   Σ b_j
        subject to  (a * b)[k] ≤ M  for all k,
                    b_j ≥ 0,
    where `M` is the current maximal convolution of `a`.
    The solution `b` has a larger ℓ₁‑norm while keeping the
    convolution bound.  Mixing the old sequence with `b` (step size `t`)
    strictly decreases the bound `M`.  Repeating this many hundred/thousand
    times drives the score far below the trivial value 2.0.

2.  **Simple greedy smoothing** – after the LP phase we perform a few
    cheap gradient‑style moves that transfer a little mass from indices
    that contribute most to the current maximal convolution towards indices
    that contribute the least.  This typically squeezes the score a little
    further.

Both phases respect the non‑negativity, sum‑positivity and the 1000‑cap
required by `evaluate_sequence`.  The whole routine stops well before the
global 1000 s budget expires and returns the best sequence found.
"""

import time
import math
import random
import numpy as np
from scipy import optimize

# ----------------------------------------------------------------------
# Linear programme: maximise sum(b)  subject to  (a * b)[k] ≤ M , b ≥ 0
# ----------------------------------------------------------------------
def solve_convolution_lp(a_seq, rhs):
    """
    Solve the LP:
        maximise   Σ b_j
        subject to (a_seq * b)[k] ≤ rhs  for all k,
                   b_j ≥ 0.
    Returns the optimal b as a NumPy array or None on failure.
    """
    n = len(a_seq)
    # objective: minimise -Σ b_j  (linprog does minimisation)
    c = -np.ones(n)

    # build the inequality matrix for each convolution index k
    rows = []
    for k in range(2 * n - 1):
        row = np.zeros(n)
        # a_i contributes to position k with b_{k-i}
        for i in range(n):
            j = k - i
            if 0 <= j < n:
                row[j] = a_seq[i]
        rows.append(row)

    # non‑negativity constraints  b_j ≥ 0  →  -b_j ≤ 0
    rows.append(-np.eye(n))
    # RHS vector
    b_vec = np.concatenate([np.full(2 * n - 1, rhs), np.zeros(n)])

    # solve with the high‑performance HiGHS solver
    res = optimize.linprog(c, A_ub=np.vstack(rows), b_ub=b_vec,
                          method='highs', options={'disp': False})
    if res.success:
        return res.x
    return None


# ----------------------------------------------------------------------
# LP driven update step
# ----------------------------------------------------------------------
def lp_update(seq, step):
    """
    Perform one LP‑based update:
        - normalise `seq` so that its ℓ₁‑norm equals √(2·n)
        - compute current maximal convolution `M`
        - solve the LP to obtain a better direction `g`
        - mix the old and new vectors with mixing factor `step`
    Returns the new sequence (as a list) and the new maximal convolution.
    """
    n = len(seq)
    seq = np.asarray(seq, dtype=float)

    # Normalise to Σ a_i = sqrt(2n) (the scale used by the LP)
    scale = math.sqrt(2.0 * n) / seq.sum()
    a_norm = seq * scale

    # current bound M
    conv = np.convolve(a_norm, a_norm)
    M = np.max(conv)

    # solve LP to maximise the ℓ₁‑norm while respecting the bound
    g = solve_convolution_lp(a_norm, M)
    if g is None:
        return None, M

    # normalise g to the same ℓ₁‑norm as a_norm
    g = np.asarray(g, dtype=float)
    g = g * (math.sqrt(2.0 * n) / g.sum())

    # convex combination (step ∈ (0,1))
    new = (1.0 - step) * a_norm + step * g

    # No further normalisation – the ℓ₁‑norm stays √(2n)
    return new.tolist(), M


# ----------------------------------------------------------------------
# Greedy smoothing step (uses the max‑convolution index)
# ----------------------------------------------------------------------
def greedy_smooth(seq, eps_factor=0.05):
    """
    Transfer a small fraction of mass from the element that contributes most
    to the current maximal convolution to the element that contributes the
    least.  The move is accepted only if the score improves.
    Returns the possibly updated sequence.
    """
    n = len(seq)
    a = np.asarray(seq, dtype=float)

    conv = np.convolve(a, a)
    max_val = conv.max()
    max_idxs = np.flatnonzero(conv == max_val)
    k = int(np.random.choice(max_idxs))

    # gradient of conv[k] w.r.t. a_i is 2·a_{k-i} (zero if out of range)
    i = np.arange(n)
    j = k - i
    mask = (j >= 0) & (j < n)
    grad = np.zeros_like(a)
    grad[mask] = 2.0 * a[j[mask]]

    # choose donor (largest contribution) and receiver (smallest)
    donor = int(np.argmax(grad * (a > 1e-12)))   # weighted by existence of mass
    receiver = int(np.argmin(grad))

    if donor == receiver:
        return seq  # nothing to do

    eps = eps_factor * a[donor]          # move a modest fraction
    if eps <= 0.0:
        return seq

    # perform the transfer
    a[donor] -= eps
    a[receiver] += eps

    # re‑normalise to avoid drifting numerical sum
    total = a.sum()
    if total <= 0.0:
        return seq
    a /= total

    # accept only if the score improves
    new_score = evaluate_sequence(a.tolist())
    old_score = evaluate_sequence(seq)
    if new_score < old_score:
        return a.tolist()
    # otherwise revert
    a[receiver] -= eps
    a[donor] += eps
    a /= a.sum()
    return seq


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed=42, budget_s=1000, **kwargs):
    """
    Returns a list of non‑negative heights with a low evaluation score.
    The routine mixes an LP‑based global search with a cheap greedy
    refinement.  It respects the time budget and logs progress.
    """
    random.seed(seed)
    np.random.seed(seed)

    deadline = time.time() + budget_s - 5.0   # leave a few seconds margin

    # ------------------------------------------------------------------
    # 1. Initial random / geometric seeds
    # ------------------------------------------------------------------
    n = 300                     # moderate size – fast LP, good theoretical bound
    seq = np.random.dirichlet(np.ones(n)).tolist()   # random start, Σ=1

    # bring the sequence to the normalisation used by the LP (Σ=√(2n))
    scale = math.sqrt(2.0 * n) / sum(seq)
    seq = (np.asarray(seq) * scale).tolist()

    best_seq = seq[:]
    best_score = evaluate_sequence(best_seq)
    print(f"Initial score (n={n}): {best_score:.6f}")

    # ------------------------------------------------------------------
    # 2. Repeated LP‑driven improvement
    # ------------------------------------------------------------------
    step = 0.1          # start with a relatively aggressive mixing
    iteration = 0
    while time.time() < deadline and iteration < 5000:
        upd, M = lp_update(seq, step)
        if upd is None:
            break
        seq = upd
        cur_score = evaluate_sequence(seq)
        if cur_score < best_score:
            best_score = cur_score
            best_seq = seq[:]
            print(f"LP iter {iteration:4d}: score={best_score:.6f}  (M={M:.6f})")
        # slowly shrink the step size to stabilise the search
        step = max(0.01, step * 0.9995)
        iteration += 1

    # ------------------------------------------------------------------
    # 3. Light greedy smoothing to polish the result
    # ------------------------------------------------------------------
    smooth_iters = 2000
    for i in range(smooth_iters):
        if time.time() >= deadline:
            break
        new_seq = greedy_smooth(seq, eps_factor=0.03)
        if new_seq is not seq:          # an improvement was made
            seq = new_seq
            cur_score = evaluate_sequence(seq)
            if cur_score < best_score:
                best_score = cur_score
                best_seq = seq[:]
                print(f"Smooth iter {i:4d}: score={best_score:.6f}")

    print(f"Finished – best score obtained: {best_score:.6f}")
    # Convert to plain Python list of floats before returning
    return [float(x) for x in best_seq]
