def propose_candidate(seed=123, budget_s=1000):

    deadline = time.time() + budget_s - 5
    # Starting sequence
    if hasattr(globals(), 'height_sequence_1') and height_sequence_1:
        seq = np.array(height_sequence_1, dtype=float)
    else:
        n = 8000
        seq = np.full(n, np.sqrt(2*n)/n)
    # Ensure non-negative and sum normalized
    seq = np.maximum(seq, 0)
    sum_seq = np.sum(seq)
    target_sum = np.sqrt(2*len(seq))
    if sum_seq == 0:

    else:
        seq = seq * (target_sum / sum_seq)
    best_seq = seq.copy()
    best_score = evaluate_sequence(seq.tolist())
    iteration = 0
    while time.time() < deadline:
        iteration += 1
        # compute convolution
        conv = np.convolve(seq, seq)
        k_max = np.argmax(conv)
        max_val = conv[k_max]
        # Compute contributions
        # partner indices = k_max - np.arange(n)
        n = len(seq)
        idx = np.arange(n)
        partner = k_max - idx
        valid = (partner >= 0) & (partner < n)
        contributions = np.zeros(n)
        contributions[valid] = seq[idx[valid]] * seq[partner[valid]]
        # choose source (largest contribution) and target (smallest)
        # exclude source with zero contribution (but may be okay)
        if np.all(contributions == 0):
            # random perturbation
            i = np.random.randint(n)
            delta = (np.random.rand() - 0.5) * 0.01  # small random
            seq[i] = max(seq[i] + delta, 0.0)
            # renormalize
            seq = seq * (target_sum / np.sum(seq))
            continue
        i_max = np.argmax(contributions)
        # choose target index with minimal contribution but not same as i_max
        i_min = np.argmin(contributions)
        if i_max == i_min:
            # pick random target
            i_min = (i_max + np.random.randint(1, n)) % n
        # compute delta
        # step fraction decays over time
        fraction = 0.5 * (1.0 - iteration / (deadline - time.time() + iteration))  # unrealistic
        # easier: fraction = max(0.5 * (0.99 ** iteration), 0.001)
        fraction = max(0.5 * (0.99 ** (iteration // 10)), 0.001)
        delta = seq[i_max] * fraction
        if delta <= 0:
            delta = 0.0001
        # transfer delta from i_max to i_min
        seq[i_max] -= delta
        seq[i_min] += delta
        # ensure nonneg ( i_max may become negative due to rounding)
        if seq[i_max] < 0:
            seq[i_min] += seq[i_max]  # add back negative part
            seq[i_max] = 0.0
        # sum unchanged - but might slightly deviate due to rounding - re-normalize to target_sum
        current_sum = np.sum(seq)
        if abs(current_sum - target_sum) > 1e-12:
            seq = seq * (target_sum / current_sum)
        # evaluate
        score = evaluate_sequence(seq.tolist())
        if score < best_score:
            best_score = score
            best_seq = seq.copy()
            # print progress
            if iteration % 100 == 0:
                print(f"Iter {iteration}, score {score}")
        # occasional random perturbation to escape
        if iteration % 5000 == 0:

            ids = np.random.choice(n, size=10, replace=False)
            # random redistribution among these
            total = np.sum(seq[ids])
            # assign random Dirichlet distribution
            rand = np.random.rand(len(ids))
            rand = rand / rand.sum()
            seq[ids] = total * rand
            # renorm sum
            seq = seq * (target_sum / np.sum(seq))

    return best_seq.tolist()

import numpy as np
import time
import math
import random

def compute_objective(seq, n):
    # compute 2*n*max(conv(seq,seq))/sum(seq)^2
    # Conv using numpy's FFT for speed
    # Use direct np.convolve (fast enough for n~8000)
    conv = np.convolve(seq, seq)
    max_b = conv.max()
    sum_a = seq.sum()
    if sum_a <= 0:
        return float('inf')
    return 2.0 * n * max_b / (sum_a * sum_a)

def subgradient_move(seq, n, step):
    """
    Perform a deterministic mass‑transfer move aimed at decreasing
    the current maximal convolution value.
    Returns a new numpy array (copy) with the same L1 norm as seq.
    """
    # copy
    new = seq.copy()
    # compute convolution and locate the index with largest value
    conv = np.convolve(new, new)
    kmax = conv.argmax()
    # contributions of each entry to this particular convolution index
    idx = np.arange(n)
    partner = kmax - idx
    mask = (partner >= 0) & (partner < n)
    contrib = np.zeros(n)
    contrib[mask] = new[idx[mask]] * new[partner[mask]]
    # source is element with largest contribution
    src = int(contrib.argmax())
    # target is element with smallest contribution (different from src)
    # If all contributions equal we just pick a random distinct index
    if contrib.max() == contrib.min():
        tgt = (src + 1) % n
    else:
        # get sorted indices, choose the smallest different from src
        sorted_idx = np.argsort(contrib)
        tgt = int(sorted_idx[0])
        if tgt == src:
            tgt = int(sorted_idx[1]) if n > 1 else src
    # amount to move – a fraction of the source value
    delta = new[src] * step
    if delta <= 0:
        return new
    new[src] -= delta
    new[tgt] += delta
    # guard against tiny negative due to rounding
    if new[src] < 0:
        new[tgt] += new[src]   # add the negative remainder back
        new[src] = 0.0
    return new

def random_move(seq, n, step):

"""
Simulated‑annealing + sub‑gradient search for a step sequence that minimises

    2·n·max(f*f) / (∑f)²

The algorithm repeatedly proposes two kinds of moves:

* **Sub‑gradient move** – locate the index k where the convolution f*f attains
  its maximum, compute each element's contribution to that index and transfer a
  fraction of mass from the largest contributor to the smallest one.  This
  directly attacks the current peak of the autocorrelation.

* **Random mass‑transfer move** – pick two distinct positions i,j at random and
  move a fraction of the mass from i to j.  This provides exploration and helps
  escape local minima.

A simple temperature schedule is used to occasionally accept uphill moves.
The best sequence seen during the run is remembered and returned.
All operations stay inside the allowed range (non‑negative, ≤1000) and the L¹
norm is kept exactly √(2·n) so that the evaluation function reduces to the
maximum convolution value.
"""

import math
import time
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities – they are defined at top level (no closures)
# ----------------------------------------------------------------------


def compute_objective(seq: np.ndarray, n: int) -> float:
    """
    Returns 2·n·max(conv(seq,seq)) / (sum(seq)²).
    The sequence is assumed to be non‑negative and its sum > 0.
    """
    conv = np.convolve(seq, seq)
    max_b = conv.max()
    s = seq.sum()
    if s <= 0.0:
        return float("inf")
    return 2.0 * n * max_b / (s * s)


def subgradient_move(seq: np.ndarray, n: int, frac: float) -> np.ndarray:
    """
    Deterministic move aimed at decreasing the current maximal convolution value.
    * Locate the index k where conv(seq,seq) is maximal.
    * Compute for every i the product seq[i]·seq[k‑i] (zero if the partner is out of range).
    * Transfer a fraction `frac` of the mass from the index with the largest
      contribution to the index with the smallest contribution.
    The returned array has the same L1 norm as `seq` (up to rounding errors).
    """
    new = seq.copy()
    conv = np.convolve(new, new)
    kmax = int(conv.argmax())

    idx = np.arange(n)
    partner = kmax - idx
    mask = (partner >= 0) & (partner < n)

    contrib = np.zeros(n, dtype=new.dtype)
    contrib[mask] = new[idx[mask]] * new[partner[mask]]

    src = int(contrib.argmax())

    if contrib.max() == contrib.min():
        # completely flat – pick any different index
        tgt = (src + 1) % n
    else:
        order = np.argsort(contrib)
        tgt = int(order[0])
        if tgt == src:
            tgt = int(order[1]) if n > 1 else src

    delta = new[src] * frac
    if delta <= 0.0:
        return new

    new[src] -= delta
    new[tgt] += delta

    # protect against tiny negative values caused by rounding
    if new[src] < 0.0:
        new[tgt] += new[src]
        new[src] = 0.0
    return new


def random_move(seq: np.ndarray, n: int, frac: float) -> np.ndarray:
    """
    Random mass‑transfer move.
    Choose i ≠ j uniformly at random and move a fraction `frac` of the mass
    from i to j.
    """
    i = np.random.randint(n)
    j = np.random.randint(n - 1)
    if j >= i:
        j += 1

    new = seq.copy()
    delta = new[i] * frac
    if delta <= 0.0:
        return new

    new[i] -= delta
    new[j] += delta

    if new[i] < 0.0:           # safety – should not happen
        new[j] += new[i]
        new[i] = 0.0
    return new


def step_fraction(it: int, start: float = 0.3, decay: float = 0.999) -> float:
    """
    Geometrically decreasing fraction of mass that is moved at iteration `it`.
    The value never goes below 1e‑4.
    """
    f = start * (decay ** it)
    return max(f, 1e-4)


# ----------------------------------------------------------------------
# Main entry point required by the evaluation harness
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a sequence of non‑negative heights that minimises the
    evaluation function.  The method combines a sub‑gradient move that
    reduces the current peak of the autocorrelation with a random
    mass‑transfer move for exploration, using a simple simulated‑annealing
    acceptance rule.
    The best sequence found before the time budget expires is returned.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 1.0  # small safety margin

    # --------------------------------------------------------------
    # Initialise – reuse the known construction if it exists
    # --------------------------------------------------------------
    if "height_sequence_1" in globals() and isinstance(
        globals()["height_sequence_1"], (list, tuple)
    ):
        init = np.array(globals()["height_sequence_1"], dtype=float)
    else:
        # fallback: a modest uniform vector
        init_len = 5000
        init = np.full(init_len, 1.0, dtype=float)

    n = len(init)
    target_sum = math.sqrt(2.0 * n)

    # enforce non‑negativity and normalise the L¹ norm
    init = np.maximum(init, 0.0)
    s = init.sum()
    if s == 0.0:
        init[:] = target_sum / n
    else:
        init = init * (target_sum / s)

    cur_seq = init.copy()
    cur_score = compute_objective(cur_seq, n)

    best_seq = cur_seq.copy()
    best_score = cur_score

    iteration = 0
    T0 = 0.2  # initial temperature for uphill acceptance

    # --------------------------------------------------------------
    # Main optimisation loop
    # --------------------------------------------------------------
    while time.time() < deadline:
        iteration += 1
        frac = step_fraction(iteration)

        # Choose which move to try
        if np.random.rand() < 0.65:
            cand = subgradient_move(cur_seq, n, frac)
        else:
            cand = random_move(cur_seq, n, frac)

        # Re‑normalise to keep the exact L¹ norm (removes accumulated drift)
        total = cand.sum()
        if total != 0.0:
            cand = cand * (target_sum / total)

        cand_score = compute_objective(cand, n)

        # Acceptance rule
        if cand_score < cur_score:
            cur_seq, cur_score = cand, cand_score
        else:
            # temperature decays roughly exponentially with the iteration count
            T = T0 * (0.9995 ** (iteration // 50))
            accept_prob = math.exp(-(cand_score - cur_score) / max(T, 1e-12))
            if np.random.rand() < accept_prob:
                cur_seq, cur_score = cand, cand_score
            # else: reject – keep the previous state

        # Track the overall best solution
        if cur_score < best_score:
            best_score = cur_score
            best_seq = cur_seq.copy()
            if iteration % 500 == 0:
                print(
                    f"Iter {iteration:6d} – new best objective {best_score:.6f}"
                )

        # Occasionally perform a stronger random perturbation to escape plateaus
        if iteration % 20000 == 0:
            # Randomly re‑draw a tiny fraction of the entries
            k = min(30, n)
            idx = np.random.choice(n, size=k, replace=False)
            mass = best_seq[idx].sum()
            rand = np.random.rand(k)
            rand = rand / rand.sum() * mass
            best_seq[idx] = rand
            best_seq = best_seq * (target_sum / best_seq.sum())
            cur_seq = best_seq.copy()
            cur_score = compute_objective(cur_seq, n)

    # --------------------------------------------------------------
    # Return the best sequence discovered
    # --------------------------------------------------------------
    print(
        f"Finished after {iteration} iterations. Best objective: {best_score:.6f}"
    )
    return best_seq.tolist()
