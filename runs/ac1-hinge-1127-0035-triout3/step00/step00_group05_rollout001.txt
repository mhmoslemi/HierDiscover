def full_convolve(x,y):
    L = len(x)+len(y)-1
    N = 1 << (L-1).bit_length()  # next pow2
    X = np.fft.rfft(x, n=N)
    Y = np.fft.rfft(y, n=N)
    Z = X * Y
    conv = np.fft.irfft(Z, n=N)[:L]
    return conv

    start = time.time()

    # We'll attempt to run repeated attempts for different n and track best.

    best_seq = None

    # List of lengths to try: maybe [256, 512,1024,2048,4096,7866] where 7866 is current best. We'll attempt them.

    lengths = [256, 512, 1024, 2048, 4096, 6000, 8000]
    # Randomly shuffle to cover.

    for n in lengths:
        if time.time() > deadline:
            break
        # initialize a uniform distribution scaled.
        a = np.full(n, np.sqrt(2 * n) / n)
        target_sum = np.sqrt(2 * n)

        # parameters for gradient descent
        beta = 0.5  # start low
        eta = 0.2
        max_iter = 2000
        # gradually increase beta after some steps.
        for t in range(max_iter):
            if time.time() > deadline:
                break
            # compute b via fft conv
            b = full_convolve(a, a)
            max_b = b.max()
            # compute softmax weights
            # subtract max to avoid overflow
            w = np.exp(beta * (b - max_b))
            w /= w.sum()
            # calculate gradient
            a_rev = a[::-1]
            conv2 = full_convolve(w, a_rev)
            G = 2.0 * conv2[n-1:n-1+n]
            # update via exponentiated gradient
            a = a * np.exp(-eta * G)
            # renormalize sum
            sum_a = a.sum()
            # to avoid zero sum
            if sum_a == 0:
                a = np.full(n, target_sum / n)
            else:
                a = a * (target_sum / sum_a)

            # evaluate occasionally
            if t % 10 == 0:
                seq = a.tolist()
                score = evaluate_sequence(seq)
                if score < best_score:
                    best_score = score
                    best_seq = seq.copy()
                    print(f"New best score {best_score:.6f} at n={n}, iter={t}")
            # slowly adjust beta
            if t % 100 == 0 and beta < 10:
                beta *= 1.5
        # after loop maybe final evaluate
        final_score = evaluate_sequence(a.tolist())
        if final_score < best_score:
            best_score = final_score
            best_seq = a.tolist()

    # if time left attempt random perturbations from best_seq
    while time.time() < deadline:
        # random local search

"""
Simple gradient‑based optimizer for the self‑convolution bound.

The evaluation function is
    f(a) = 2·n·max(a∗a) / (sum(a))² .
If we keep sum(a)=√(2n) the objective reduces to simply
    max(a∗a) .

We therefore minimise the ℓ∞‑norm of the full convolution while
keeping the coefficients non‑negative and their sum fixed.
The algorithm works as follows:

1. Choose a length n (several lengths are tried).
2. Initialise a non‑negative vector a with the required sum.
3. Repeatedly:
   • Compute the full convolution b = a∗a.
   • Form a soft‑max weighting w_k ∝ exp(β·(b_k‑max(b))) .
   • The gradient of 𝔼_w[b] w.r.t. a is
         g = 2·(w ∗ a_rev)[n‑1 : n‑1+n]      (a_rev = a reversed)
     which is obtained by another FFT‑based convolution.
   • Update a by an exponentiated‑gradient step
         a ← a·exp(−η·g)          (keeps a ≥ 0)
   • Enforce symmetry a_i = a_{n‑1‑i} (helps flatten the autocorrelation)
   • Renormalise a so that sum(a)=√(2n).

β is increased slowly (soft‑max → hard‑max) and η is slowly reduced.
Every few iterations the true ``evaluate_sequence`` is called; the best
sequence seen so far is kept.  When the time budget is nearly exhausted the
search stops and the best sequence is returned.

The code uses only NumPy and the provided ``evaluate_sequence`` function.
"""

import time
import numpy as np

def _full_convolve(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Return the full (linear) convolution of two real vectors using FFT."""
    L = len(x) + len(y) - 1
    N = 1 << (L - 1).bit_length()          # next power of two
    X = np.fft.rfft(x, n=N)
    Y = np.fft.rfft(y, n=N)
    Z = X * Y
    conv = np.fft.irfft(Z, n=N)[:L]
    return conv

    # Light random local‑search near the current best if time permits.
    # ------------------------------------------------------------------
    while time.time() < deadline:
        if best_seq is None:
            break
        seq = np.array(best_seq, dtype=np.float64)
        n = len(seq)
        target_sum = np.sqrt(2.0 * n)

        # Apply a few small random perturbations
        for _ in range(3):
            idx = np.random.randint(n)
            # 5 % relative Gaussian noise, keeping values non‑negative
            delta = np.random.randn() * seq[idx] * 0.05
            seq[idx] = max(0.0, seq[idx] + delta)

        # Re‑impose symmetry and renormalise
        seq = (seq + seq[::-1]) * 0.5
        s = seq.sum()
        if s > 0.0:
            seq *= target_sum / s
        else:
            seq = np.full(n, target_sum / n, dtype=np.float64)

        try:
            score = evaluate_sequence(seq.tolist())
        except Exception:
            score = float("inf")
        if score < best_score:
            best_score = score
            best_seq = seq.tolist()
            print(
                f"[{time.time() - start_time:6.2f}s] "
                f"Tweak improved best to {best_score:.6f}"
            )

    # Return the best sequence discovered (as a plain Python list)
    return best_seq
