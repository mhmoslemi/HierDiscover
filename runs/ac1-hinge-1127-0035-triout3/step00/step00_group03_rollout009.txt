"""
Simple construction for a low‑autocorrelation step function.

The evaluation function is
    2 * n * max(conv(a, a)) / (sum(a)**2)

If we normalize sum(a) = 1, the score is just 2 * n * max(conv(a,a)).
For a sequence of the form

    a_i ∝ (1 - i/(n-1))^p   (0 ≤ i < n)

the convolution peak occurs at shift (n‑1) and its value is

    max(conv) = (p+1)^2 * B(p+1, p+1) / n

where B is the beta‑function.  The resulting score is therefore

    score = 2 * (p+1)^2 * B(p+1, p+1)

which is independent of n and drops quickly as p grows
(e.g. p=1 → 1.33, p=2 → 0.60, p=3 → 0.23, p=4 → 0.08 …).

The algorithm simply tries a few small integer exponents and returns the
shape with the smallest score.  The construction is deterministic,
fast and respects all security checks of ``evaluate_sequence``.
"""

import numpy as np
import time

def propose_candidate(seed: int = 42,
                     budget_s: float = 1000,
                     length: int = 7411) -> list[float]:
    """
    Return a non‑negative step‑function (list of heights) that obtains a
    very low value of ``evaluate_sequence``.

    The function tries power‑law decreasing shapes
        a_i = (p+1) * (1 - i/(n-1))**p
    and picks the exponent p (1 ≤ p ≤ 8) giving the smallest score.
    The sequence is normalised to have sum = 1 (any scaling would give the
    same score because the evaluation function is homogeneous).
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5   # a tiny safety margin

    # length of the step function – must be at least 2
    n = max(2, int(length))

    # pre‑compute the index vector
    idx = np.arange(n, dtype=float)

    best_seq = None
    best_score = np.inf

    # try a handful of exponents; higher p gives smaller scores
    for p in range(1, 9):
        # shape a_i ∝ (1 - i/(n-1))**p, normalised to sum = 1
        if n == 1:
            raw = np.array([1.0])
        else:
            # (p+1) factor makes the integral of the continuous analogue equal 1
            raw = (p + 1) * (1.0 - idx / (n - 1.0)) ** p

        total = raw.sum()
        if total == 0:
            continue
        seq = raw / total

        # ``evaluate_sequence`` is cheap; we can call it for every candidate
        try:
            score = evaluate_sequence(seq.tolist())
        except Exception:
            continue

        # keep the best one seen so far
        if score < best_score:
            best_score = score
            best_seq = seq.tolist()
            print(f"p={p:2d} → score={score:.6f}")

        # stop early if we run out of time
        if time.time() > deadline:
            break

    # If for any reason something went wrong (should not happen), fall back
    # to a uniform sequence.
    if best_seq is None:
        best_seq = [1.0 / n] * n
        best_score = evaluate_sequence(best_seq)
        print("Fallback to uniform sequence:", best_score)

    print(f"Final choice: length={n}, score={best_score:.6f}")
    return best_seq
