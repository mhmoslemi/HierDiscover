def hillclimb(a, time_deadline):
    best = a.copy()
    best_score = evaluate_sequence(a.tolist())
    while time.time() < time_deadline:
        # Randomly pick i,j distinct
        i, j = np.random.choice(len(a), 2, replace=False)
        # Determine amount delta: small fraction of a[i]
        if a[i] == 0: continue
        delta = a[i] * np.random.rand() * 0.1   # up to 10% of a_i
        # Move delta from i to j
        a2 = a.copy()
        a2[i] -= delta
        a2[j] += delta
        # Evaluate
        score = evaluate_sequence(a2.tolist())
        if score < best_score:
            best_score = score
            best = a2.copy()
            a = a2
        else:
            # accept with probability e^{- (score-best_score)/T}
            # Simulated annealing
            T = max(0.01, (deadline - time.time())/deadline) # decreasing temperature
            if np.random.rand() < np.exp(-(score-best_score)/T):
                a = a2
        # Occasionally random perturbation to escape local minima
        if np.random.rand() < 0.001:
            a = np.maximum(a + np.random.randn(len(a))*0.001,0)
            a = simplex_projection(a)

"""
Propose a step‑function (list of non‑negative heights) that minimises

    2 * n * max( a * a ) / (sum(a)**2)

where “*” is the discrete convolution.
The algorithm works in three stages:

1.  Initialise a candidate of a given length (taken from the existing
    `height_sequence_1` if present, otherwise a default size).
2.  Perform a cheap grid search over a two‑parameter family of symmetric
    “U‑shaped’’ sequences.
    a_i ∝ 1 + β·( (i/(n‑1))**p + (1‑i/(n‑1))**p )
    – β increases the mass near the edges, p controls how sharply the
    edges are preferred.  All candidates are normalised to sum‑to‑1.
3.  Refine the best candidate with a simple stochastic hill‑climbing /
    simulated‑annealing loop that transfers a small amount of mass between
    two random positions and projects back onto the probability simplex.

The search runs until the allocated time budget (default 1000 s) is almost
exhausted, printing intermediate improvements.  The returned list is the
best sequence found.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
def _simplex_projection(v: np.ndarray) -> np.ndarray:
    """
    Project a vector `v` onto the probability simplex:
        { x | x_i >= 0, sum_i x_i = 1 }.
    Implements the O(n log n) algorithm of
    Wang & Carreira‑Perpiñán (2013).
    """
    if v.ndim != 1:
        raise ValueError("simplex projection expects a 1‑D array")
    # sort in descending order
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(len(v)) + 1
    # find rho = max { j | u_j - cssv_j / j > 0 }
    cond = u - cssv / ind > 0
    if not np.any(cond):
        rho = 0
    else:
        rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs) -> list:
    """
    Search for a sequence of non‑negative heights with a low evaluation
    score.  The function respects the 1000 s time limit and returns the
    best sequence discovered.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5      # a safety margin

    # ------------------------------------------------------------------
    # Choose a working length (use the existing construction if available)
    try:
        n = len(height_sequence_1)                # type: ignore[name-defined]
        # keep it reasonable for the search (too large would be slow)
        n = max(500, min(n, 5000))
    except Exception:
        n = 2000

    # ------------------------------------------------------------------
    # Baseline: uniform distribution (known to give score 2.0)
    uniform = np.full(n, 1.0 / n)
    best_seq = uniform.tolist()
    best_score = evaluate_sequence(best_seq)
    print(f"[init] uniform score = {best_score:.6f} (n = {n})")

    # ------------------------------------------------------------------
    # 1. Grid search over a symmetric “U‑shaped’’ family
    p_vals = np.linspace(0.2, 5.0, 25)        # shape exponent
    beta_vals = np.linspace(0.0, 4.0, 25)     # weight of the edge‑preferring term

    idx = np.arange(n)
    d = idx / (n - 1.0)

    for p in p_vals:
        # pre‑compute shape component for this exponent
        shape = np.power(d, p) + np.power(1.0 - d, p)   # symmetric, high at edges
        for beta in beta_vals:
            if time.time() >= deadline:
                break
            raw = np.ones(n) + beta * shape
            raw = np.where(raw > 0.0, raw, 0.0)          # enforce non‑negativity
            total = raw.sum()
            if total == 0.0:
                continue
            seq = raw / total
            score = evaluate_sequence(seq.tolist())
            if score < best_score:
                best_score = score
                best_seq = seq.tolist()
                print(f"[grid] new best {best_score:.6f}  p={p:.2f}  β={beta:.2f}")

        if time.time() >= deadline:
            break

    # ------------------------------------------------------------------
    # 2. Stochastic hill‑climbing refinement (simulated annealing style)
    curr_seq = np.array(best_seq, dtype=float)

    # helper to evaluate and possibly update the global best
    def _maybe_update(candidate: np.ndarray):
        nonlocal best_seq, best_score
        cand_score = evaluate_sequence(candidate.tolist())
        if cand_score < best_score:
            best_score = cand_score
            best_seq = candidate.tolist()
            print(f"[refine] improved to {best_score:.6f}")

    # Main refinement loop
    while time.time() < deadline:
        # pick two distinct positions
        i, j = np.random.choice(n, 2, replace=False)

        # amount to transfer: a fraction of the source entry
        eps = curr_seq[i] * np.random.rand() * 0.2   # up to 20 % of a_i
        if eps == 0.0:
            continue

        trial = curr_seq.copy()
        trial[i] -= eps
        trial[j] += eps

        # project back onto the simplex (keeps sum = 1, non‑negative)
        trial = _simplex_projection(trial)

        # evaluate and accept according to a simple annealing schedule
        new_score = evaluate_sequence(trial.tolist())
        delta = new_score - best_score

        if delta < 0:
            # strict improvement
            curr_seq = trial
            _maybe_update(trial)
        else:
            # occasional uphill moves to escape plateaux
            # temperature decays linearly to a tiny floor
            T = max(0.001,
                    (deadline - time.time()) / (deadline - start_time))
            if np.random.rand() < np.exp(-delta / T):
                curr_seq = trial

    # ------------------------------------------------------------------
    print(f"[final] best score = {best_score:.6f}  (length = {len(best_seq)})")
    return best_seq
