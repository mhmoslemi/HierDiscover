def simplex_projection(v, s):
    """Projects vector v onto the simplex {x >= 0, sum x = s}."""
    # from https://doi.org/10.1016/j.jstatsoft.2009.05.025
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    ind = np.arange(1, n+1)
    cond = u - cssv / ind > 0
    # Find rho = max i where cond
    rho = np.where(cond)[0][-1]  # last index
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0)
    return w

def simplex_projection(v, s):
    """Projects vector v onto the simplex {x >= 0, sum x = s}."""
    # similar to algorithm by Wang & Carreira-Perpiñán (2013)
    n = len(v)
    # Sort v in descending order
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    # find rho
    rho = np.where(u - cssv / (np.arange(1, n + 1)) > 0)[0]
    if len(rho) == 0:
        theta = 0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0)
    return w

def conv_max_and_index(a):
    n = a.shape[0]
    L = 1 << (int(np.ceil(np.log2(2 * n - 1))))
    A = np.fft.rfft(a, L)
    s = np.fft.irfft(A * A, L)
    s = s[:2 * n - 1]
    k = int(np.argmax(s))
    max_val = float(s[k])
    return max_val, k

def convolve_seq(a,b):
    n = len(a) + len(b) - 1
    L = 1 << (int(np.ceil(np.log2(n))))
    A = np.fft.rfft(a, L)
    B = np.fft.rfft(b, L)
    c = np.fft.irfft(A * B, L)
    return c[:n]

"""
Propose a sequence of non‑negative heights that (hopefully) improves the
evaluation metric:

    eval = 2 * n * max(conv(a, a)) / (sum(a) ** 2)

The algorithm works as follows:

1.  Use a moderate length (≈ 1000) to keep the linear programme tractable.
2.  Initialise with a uniform sequence (which already yields a value of 2.0)
    and repeatedly improve it.
3.  In each iteration:
    * Solve a linear programme that finds a direction `g` with a larger
      L1‑sum while keeping the convolution `conv(a, g)` bounded by the
      current maximum `M = max(conv(a, a))`.
    * Normalise `g` so that it has the same L1‑sum as `a`.
    * Mix `a` and `g` with a mixing coefficient `t ∈ [0, 1]`.  The optimal
      `t` is obtained by a tiny line‑search that evaluates the resulting
      convolution analytically (using the identity
      conv((1‑t)a + t g, (1‑t)a + t g) = (1‑t)² conv(a,a) +
      2 t (1‑t) conv(a,g) + t² conv(g,g)).
    * If the new maximum is smaller, accept the update; otherwise perform a
      small random perturbation and continue.
4.  Keep the best sequence seen during the search and return it when the
    time budget expires.

The linear programme is solved with `scipy.optimize.linprog` (high‑speed
interior‑point method) using a sparse constraint matrix to keep memory
usage modest.  Convolutions are performed with FFTs for speed.
"""

import time
import numpy as np
from scipy import optimize
from scipy import sparse

c_linprog = optimize.linprog
np_rand = np.random.rand


def _simplex_projection(v: np.ndarray, s: float) -> np.ndarray:
    """Project `v` onto the simplex {x ≥ 0, sum(x) = s}."""
    if s < 0:
        raise ValueError("Simplex sum must be non‑negative")
    n = v.shape[0]
    # sort descending
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.where(cond)[0][-1]  # last True index
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def _conv_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Convolution via FFT (real inputs)."""
    n = a.shape[0] + b.shape[0] - 1
    L = 1 << int(np.ceil(np.log2(n)))
    A = np.fft.rfft(a, L)
    B = np.fft.rfft(b, L)
    c = np.fft.irfft(A * B, L)
    return c[:n]


def _solve_convolution_lp(f: np.ndarray, M: float) -> np.ndarray | None:
    """
    Solve the LP:
        maximise   sum(b)
        subject to  conv(f, b) ≤ M  (for every shift)
                    b ≥ 0
    Returns the optimal `b` (as a dense ndarray) or None if infeasible.
    """
    n = f.shape[0]

    # ----- build sparse A_ub -----
    # first (2n-1) rows correspond to conv(f, b) ≤ M
    row = []
    col = []
    data = []

    for k in range(2 * n - 1):
        i_start = max(0, k - (n - 1))
        i_end = min(n - 1, k)
        # j runs from i_start .. i_end
        # coefficient for b_j is f_{k-j}
        j_vals = np.arange(i_start, i_end + 1, dtype=np.int32)
        i_vals = k - j_vals
        coeffs = f[i_vals]

        row.extend([k] * len(j_vals))
        col.extend(j_vals.tolist())
        data.extend(coeffs.tolist())

    # non‑negativity constraints: -b_j ≤ 0
    rows_nonneg = np.arange(2 * n - 1, 2 * n - 1 + n, dtype=np.int32)
    cols_nonneg = np.arange(n, dtype=np.int32)
    data_nonneg = -np.ones(n, dtype=np.float64)

    row.extend(rows_nonneg.tolist())
    col.extend(cols_nonneg.tolist())
    data.extend(data_nonneg.tolist())

    A_ub = sparse.csr_matrix((data, (row, col)), shape=(2 * n - 1 + n, n))
    b_ub = np.concatenate([np.full(2 * n - 1, M, dtype=np.float64), np.zeros(n)])

    # maximise sum(b)  → minimise -sum(b)
    c = -np.ones(n, dtype=np.float64)

    # use the ‘highs’ method (fast and accepts sparse matrices)
    try:
        res = c_linprog(c, A_ub=A_ub, b_ub=b_ub, method="highs", options={"presolve": True})
    except Exception:  # pragma: no cover
        return None

    if res.success:
        return np.maximum(res.x, 0.0)  # safeguard tiny negatives from the solver
    return None


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a height sequence with a low evaluation value.
    The function runs for at most `budget_s` seconds and returns the best
    sequence found as a plain list of Python floats.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 5.0  # leave a tiny margin

    # ------------------------------------------------------------------
    # 0. Choose a problem size.
    # ------------------------------------------------------------------
    # Use a size that keeps the LP manageable but still large enough to
    # allow non‑trivial improvements.
    n = 1000
    # If a reference sequence is available, use its length as an upper bound.
    if "height_sequence_1" in globals():
        n_ref = len(height_sequence_1)
        # keep n reasonable (≤ 2000) to stay within memory limits.
        n = min(2000, max(500, n_ref // 2))

    S = np.sqrt(2.0 * n)          # target L1‑sum (scale‑invariant)

    # ------------------------------------------------------------------
    # 1. Initial sequence – start from uniform (already gives 2.0).
    # ------------------------------------------------------------------
    a = np.full(n, S / n, dtype=np.float64)   # uniform start
    conv_a = _conv_fft(a, a)
    M = float(conv_a.max())                    # current max(conv(a,a))

    best_seq = a.copy()
    best_M = M
    best_val = 2.0  # uniform gives exactly 2.0

    # ------------------------------------------------------------------
    # 2. Main optimisation loop
    # ------------------------------------------------------------------
    iteration = 0
    while time.time() < deadline:
        iteration += 1

        # ----- 2a. Linear‑programming direction -----
        g_raw = _solve_convolution_lp(a, M)
        if g_raw is None:
            # fallback: small random shake and continue
            idx = np.random.randint(n)
            a[idx] = max(0.0, a[idx] + np.random.randn() * 0.01)
            a = _simplex_projection(a, S)
            conv_a = _conv_fft(a, a)
            M = float(conv_a.max())
            continue

        # normalise g to the same L1‑sum as a
        sum_g = g_raw.sum()
        if sum_g == 0:
            break
        g = g_raw * (S / sum_g)

        # ----- 2b. Convolutions needed for mixing -----
        conv_f = conv_a                     # a * a (already computed)
        conv_g = _conv_fft(g, g)            # g * g
        conv_fg = _conv_fft(a, g)           # a * g

        # ----- 2c. Tiny line‑search for the best mixing coefficient t -----
        # Evaluate a small grid; the optimum is usually near 0 or 1.
        t_candidates = np.linspace(0.0, 1.0, 11)  # 0,0.1,…,1
        best_t = 1.0
        best_M_new = M

        for t in t_candidates:
            # convex combination: a_new = (1‑t) a + t g
            # conv_new = (1‑t)² conv_f + 2 t (1‑t) conv_fg + t² conv_g
            coeff = (1.0 - t)
            conv_new = (coeff * coeff) * conv_f + (2.0 * t * coeff) * conv_fg + (t * t) * conv_g
            M_new = float(conv_new.max())
            if M_new < best_M_new - 1e-12:
                best_M_new = M_new
                best_t = t

        # ----- 2d. Accept improvement or perturb -----
        if best_t < 1.0 - 1e-12:  # we found a better mixture
            a = (1.0 - best_t) * a + best_t * g
            conv_a = _conv_fft(a, a)
            M = best_M_new
            if M < best_M:
                best_M = M
                best_seq = a.copy()
                best_val = 2.0 * n * M / (S * S)   # == M after normalisation
                print(f"[Iter {iteration}] Improved: eval={best_val:.6f}, t={best_t:.3f}")
        else:
            # No improvement – apply a modest random perturbation to escape.
            i1, i2 = np.random.choice(n, 2, replace=False)
            delta = (np_rand() - 0.5) * 0.02 * a[i1]   # 2 % relative move
            a[i1] = max(0.0, a[i1] + delta)
            a[i2] = max(0.0, a[i2] - delta)  # keep total sum unchanged (approximately)
            a = _simplex_projection(a, S)
            conv_a = _conv_fft(a, a)
            M = float(conv_a.max())

        # safety break if we already beat the target comfortably
        if best_val < 1.55:
            break

    # ------------------------------------------------------------------
    # 3. Return the best sequence found (as a plain Python list)
    # ------------------------------------------------------------------
    print(f"Final best evaluation value: {best_val:.6f}")
    return best_seq.tolist()
