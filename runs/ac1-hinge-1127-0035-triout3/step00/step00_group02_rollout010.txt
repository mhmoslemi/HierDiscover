Define a function project_simplex(v):

```
def project_simplex(v):
    # onto {x >=0, sum x =1}
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(len(v)) + 1
    cond = u - cssv / ind > 0
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0)
    return w
```

    np.random.seed(seed)

    # Determine length

        init_seq = np.array(height_sequence_1, dtype=float)
        n = len(init_seq)
    except NameError:
        n = 2000
        init_seq = np.random.rand(n)
    # Normalize to sum=1
    p = np.clip(init_seq, 0, None)
    if p.sum() < 1e-9:
        p = np.random.rand(n)
    p = p / p.sum()
    best_p = p.copy()
    best_eval = evaluate_sequence(p.tolist())
    print(f"Initial eval: {best_eval}")
    # parameters
    lr = 0.2
    stagnation = 0
    while time.time() < deadline:
        # compute conv via FFT
        L = 1 << ((2*n - 1).bit_length())
        P = np.fft.rfft(p, L)
        conv = np.fft.irfft(P * P, L)[:2*n-1]
        max_idx = np.argmax(conv)
        max_val = conv[max_idx]
        eval_val = 2 * n * max_val / (p.sum()**2)  # sum =1
        if eval_val < best_eval:
            best_eval = eval_val
            best_p = p.copy()
            print(f"New best: {best_eval:.6f} at time {time.time():.2f}")
            stagnation = 0
        else:
            stagnation += 1
        # gradient
        g = np.zeros_like(p)
        shift = max_idx - np.arange(n)
        mask = (shift >= 0) & (shift < n)
        g[mask] = 2 * p[shift[mask]]
        # gradient step towards reducing max
        p_new = p - lr * g
        # project onto simplex
        p_new = project_simplex(p_new)
        # evaluate quickly approximated
        P_new = np.fft.rfft(p_new, L)
        conv_new = np.fft.irfft(P_new * P_new, L)[:2*n-1]
        eval_new = 2 * n * conv_new.max()
        if eval_new < eval_val:
            p = p_new
            lr = min(lr * 1.02, 1.0)
        else:
            lr = max(lr * 0.5, 1e-5)
            # random perturbation to escape
            if stagnation > 200:
                # random restart
                p = np.random.rand(n)
                p = p / p.sum()
                stagnation = 0
    # return best sequence as list
    return best_p.tolist()

def conv_fft(p, L=None):
    n = len(p)
    if L is None:
        L = 1 << ((2*n - 1).bit_length())
    P = np.fft.rfft(p, L)
    conv = np.fft.irfft(P * P, L)[:2*n - 1]
    return conv

def gradient_flatten(p, target, L):
    n = len(p)
    # Compute conv of p with p
    P = np.fft.rfft(p, L)
    conv = np.fft.irfft(P * P, L)[:2*n - 1]
    r = conv - target
    # Compute conv of r with reversed p
    p_rev = p[::-1]
    R = np.fft.rfft(r, L)
    P_rev = np.fft.rfft(p_rev, L)
    grad_full = np.fft.irfft(R * P_rev, L)
    # Extract relevant segment
    start = n - 1
    grad = 4.0 * grad_full[start:start + n]  # shape n
    return conv, r, grad

The goal is to minimise

fix Σ a_i = 1 and minimise the peak of the aperiodic self‑convolution.

where r = conv(a,a) – target and a_rev is a reversed copy of a.
All convolutions are performed with FFTs for speed (O(n log n)).
After each gradient step we project back onto the simplex.
We perform a simple adaptive step‑size scheme and occasional random
re‑starts if progress stalls.  The best sequence seen during the run

The routine works for the provided `height_sequence_1` (length 3005) but

# Helper: projection of a vector onto the simplex {x>=0, sum(x)=1}
def _project_simplex(v):
    """Projection of v onto the probability simplex."""
    # based on the algorithm of Duchi et al. (2008)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(n) + 1
    # Find the last index where u_i > cssv_i / i
    rho = ind[u - cssv / ind > 0][-1]
    theta = cssv[rho - 1] / rho
    w = np.maximum(v - theta, 0.0)
    return w

def propose_candidate(seed=42, budget_s=1000, **kwargs):
    """Search for a low‑value step‑function sequence.

    Returns:
        list[float] – a sequence of non‑negative heights.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 5   # a safety margin

    # ------------------------------------------------------------------
    # Determine the working length and an initial point
    try:
        init_seq = np.array(height_sequence_1, dtype=float)  # may be defined globally
        n = len(init_seq)
    except Exception:
        n = 2000                     # fallback size
        init_seq = np.random.rand(n)

    # start from the given sequence (clipped, normalised) or from random
    a = np.clip(init_seq, 0.0, None)
    if a.sum() < 1e-12:              # degenerate – fall back to random
        a = np.random.rand(n)
    a = a / a.sum()                  # put on the simplex

    # best solution seen so far
    best_a = a.copy()
    best_score = evaluate_sequence(best_a.tolist())
    print(f"[init] n={n}, score={best_score:.6f}")

    # target constant value of the convolution (average over 2n‑1 entries)
    target = 1.0 / (2 * n - 1)

    # length for FFTs: must accommodate convolution of length 3n‑2 (r * a_rev)
    L = 1 << ((3 * n - 2).bit_length())

    # optimisation parameters
    lr = 0.2                        # initial learning rate
    it = 0
    stagnation = 0                  # counts iterations without improvement

    # ------------------------------------------------------------------
    while time.time() < deadline:
        # ----- convolution a * a (FFT)
        A = np.fft.rfft(a, L)
        conv = np.fft.irfft(A * A, L)[:2 * n - 1]

        max_val = conv.max()
        cur_score = 2.0 * n * max_val   # because Σa = 1

        # keep the best ever seen
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[iter {it:05d}] new best = {best_score:.6f}")
            stagnation = 0
        else:
            stagnation += 1

        # ----- gradient of L₂ loss  (flattening towards constant target)
        r = conv - target                     # residual vector, length 2n‑1
        a_rev = a[::-1]

        R = np.fft.rfft(r, L)
        A_rev = np.fft.rfft(a_rev, L)
        grad_full = np.fft.irfft(R * A_rev, L)          # length L
        # relevant slice corresponds to indices n‑1 … n‑1+n‑1
        grad = 4.0 * grad_full[n - 1 : n - 1 + n]      # shape (n,)

        # ----- take a gradient step and re‑project onto the simplex
        a_next = a - lr * grad
        a_next = _project_simplex(a_next)

        # quick evaluation of the candidate
        A_next = np.fft.rfft(a_next, L)
        conv_next = np.fft.irfft(A_next * A_next, L)[:2 * n - 1]
        next_score = 2.0 * n * conv_next.max()

        # ----- simple back‑tracking line search on the learning rate
        if next_score < cur_score:
            a = a_next
            lr = min(lr * 1.05, 5.0)   # slowly increase step size when successful
        else:
            lr *= 0.7
            if lr < 1e-5:
                # re‑initialise around the current best to escape a possible bad region
                a = best_a + 0.01 * np.random.randn(n)
                a = np.clip(a, 0.0, None)
                a = a / a.sum()
                lr = 0.2
                stagnation = 0

        # occasional hard restart if we have been stuck for a long time
        if stagnation > 3000:
            a = np.random.rand(n)
            a = a / a.sum()
            lr = 0.2
            stagnation = 0

        it += 1

    # ------------------------------------------------------------------
    # Return the best sequence discovered (as plain Python list)
    print(f"[final] best score = {best_score:.6f} after {it} iterations")
    return best_a.tolist()
