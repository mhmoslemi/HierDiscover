def simplex_projection(v):
    """Project onto simplex sum_i x_i = 1, x_i >=0."""
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(1,n+1)
    cond = u - cssv / ind > 0
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0)
    return w

def simplex_proj(v):
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(1, len(v)+1)
    cond = u - cssv / ind > 0
    if not np.any(cond):

        return np.zeros_like(v)
    rho = ind[cond][-1]

    w = np.maximum(v - theta, 0)

```python
def propose_candidate(seed=42, budget_s=1000):
    np.random.seed(seed)
    deadline = time.time() + budget_s - 5.0  # safety
    best_seq = None
    best_val = float('inf')
    # choose lengths to try
    lengths = [800, 1600, 3000]  # adjust based on remaining time
    for n in lengths:
        if time.time() > deadline:
            break
        print(f"Starting optimization for n={n}")
        # init a uniform + small noise
        a = np.full(n, 1.0/n)
        a += 0.001 * np.random.rand(n)  # add noise
        # project onto simplex (sum=1)
        a = simplex_projection(a)
        # precompute sizes
        L = 2*n - 1
        N = 1 << ((2*L - 1).bit_length())  # power of two >= 2L-1 for conv
        beta = 5.0
        eta0 = 0.5
        it = 0
        last_improve = 0
        # keep current max
        # compute initial b
        while time.time() < deadline:
            # compute self convolution
            A_fft = np.fft.rfft(a, N)
            b_full = np.fft.irfft(A_fft * A_fft, N)[:L]
            max_b = b_full.max()

            if score < best_val:
                best_val = score
                best_seq = a.copy()
                print(f"Iter {it}, n={n}, new best {best_val:.6f}")

                # early stop if below target
                if best_val < 1.503:

            # softmax weights
            w = np.exp(beta * (b_full - max_b))
            w_sum = w.sum()
            if w_sum == 0:

            idx = np.arange(n)
```

```python
def simplex_projection(v):
    if v.ndim != 1:
        raise ValueError
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(1, n+1)
    # Find rho = max {j | u_j - cssv_j / j > 0}
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # All are zero, return uniform?
        w = np.full(n, 1.0/n)
        return w
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0)
    return w
```

"""
Propose a step‑function (non‑negative list) that minimizes

    2 * n * max(conv(a,a)) / (sum(a)**2)

using a smooth‑approximation of the ∞‑norm and projected (mirror) gradient
descent on the simplex.  The algorithm works for any length n; we use a
large n (≈3000) so that the theoretical lower bound 2 n/(2 n‑1) ≈ 1 is
approachable.  The method does not rely on the LP used in the reference
implementation and therefore explores a different region of the search space.

The core steps are:

1. Initialise a with a small random perturbation of the uniform distribution
   and project onto the probability simplex (sum = 1, a_i ≥ 0).

2. Repeatedly:
   * Compute the self‑convolution a∗a via FFT.
   * Form a smooth approximation of the max by a softmax:
         w_k ∝ exp(β·(b_k‑max(b))).
   * Compute the gradient of the soft‑max objective:
         g = 2·conv(a, rev(w))   (linear‑time via FFT).
   * Perform a multiplicative (mirror) update a ← a·exp(‑η·g) and project
     back onto the simplex.
   * Slowly increase the softmax temperature β and decrease the stepsize η.

3. Keep the best sequence seen (according to the exact
   `evaluate_sequence` formula) and return it when the time budget expires.

The loop stops early if the target bound 1.503 is reached.
"""

import time
import numpy as np

# ------------------------------------------------------------
# Helper utilities
# ------------------------------------------------------------

def simplex_projection(v):
    """Project a vector onto the probability simplex {x ≥ 0, sum x = 1}."""
    if v.ndim != 1:
        raise ValueError("simplex_projection expects a 1‑D array")
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all entries are zero – return the uniform distribution
        return np.full(n, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


def convolve_self_fft(a, N):
    """
    Linear convolution a*a using FFT.
    Returns only the first 2*len(a)-1 entries (the full linear conv).
    """
    A = np.fft.rfft(a, N)
    b_full = np.fft.irfft(A * A, N)[: 2 * len(a) - 1]
    return b_full


def gradient_via_fft(a, w, N):
    """
    Compute g_i = 2 * Σ_j a_j * w_{i+j}   (i = 0..n‑1)
    by a single FFT‑based convolution.
    w has length L = 2*n‑1 and is assumed real.
    """
    n = len(a)
    L = 2 * n - 1
    # FFT of a (zero‑padded to length N) – already computed elsewhere in many cases,
    # but we recompute here for clarity.
    A = np.fft.rfft(a, N)
    w_rev = w[::-1]                       # reverse w
    W_rev = np.fft.rfft(w_rev, N)
    conv = np.fft.irfft(A * W_rev, N)     # linear convolution length ≤ N
    # Extract the needed slice: g_i = 2 * conv[L‑1‑i]
    idx = L - 1 - np.arange(n)
    g = 2.0 * conv[idx]
    return g


# ------------------------------------------------------------
# Main candidate generator
# ------------------------------------------------------------

def propose_candidate(seed=42, budget_s=1000, **kwargs):
    """
    Search for a sequence of non‑negative heights (step function) with a
    low evaluation score.  The function respects the time limit (budget_s)
    and returns the best sequence found.
    """
    np.random.seed(seed)
    # leave a small safety margin for the caller
    deadline = time.time() + budget_s - 5.0

    # --------------------------------------------------------
    # parameters (feel free to tune)
    # --------------------------------------------------------
    n = 3000                       # length of the sequence we optimise
    L = 2 * n - 1
    # FFT size: next power of two ≥ 2*L‑1 to obtain linear convolution
    N = 1 << ((2 * L - 1).bit_length())

    beta = 5.0                     # soft‑max temperature (starts modest)
    beta_max = 80.0                # upper bound for β
    eta_0 = 0.5                    # initial (mirror) stepsize
    max_iter = 200_000             # hard limit – will be cut by time budget

    # --------------------------------------------------------
    # initialise a on the simplex (uniform + tiny noise)
    # --------------------------------------------------------
    a = np.full(n, 1.0 / n, dtype=np.float64)
    a += 0.001 * np.random.rand(n)            # small random perturbation
    a = simplex_projection(a)

    best_seq = a.copy()
    # compute exact score once (using the provided evaluate_sequence)
    # The exact function rescales clamped values, but because we stay well
    # within the admissible range the simple formula 2*n*max(b) is sufficient.
    b = convolve_self_fft(a, N)
    best_score = 2.0 * n * b.max()
    print(f"[init] n={n}, initial score={best_score:.6f}")

    it = 0
    last_improvement = 0

    # --------------------------------------------------------
    # main optimisation loop
    # --------------------------------------------------------
    while time.time() < deadline and it < max_iter:
        # 1) self‑convolution (FFT)
        b = convolve_self_fft(a, N)
        max_b = b.max()
        score = 2.0 * n * max_b

        # keep the best ever seen
        if score < best_score:
            best_score = score
            best_seq = a.copy()
            last_improvement = it
            print(f"[{it:06d}] new best: {best_score:.6f}")

            if best_score < 1.503:            # target already reached
                break

        # 2) soft‑max weights concentrating on large convolution entries
        #    (shifted by max_b for numerical stability)
        w = np.exp(beta * (b - max_b))
        w_sum = w.sum()
        if w_sum == 0.0:
            w = np.full_like(w, 1.0 / w.shape[0])
        else:
            w /= w_sum

        # 3) gradient of the smooth max (via FFT)
        g = gradient_via_fft(a, w, N)

        # 4) mirror (multiplicative) step and projection onto simplex
        eta = eta_0 / np.sqrt(it + 1.0)
        a = a * np.exp(-eta * g)
        a = simplex_projection(a)

        # 5) schedule for β (increase, i.e. sharpen the soft‑max)
        beta = min(beta_max, beta * 1.001)

        # 6) occasional random jitter if we stagnate for a while
        if it - last_improvement > 3000 and (it % 500 == 0):
            jitter = 0.0005 * np.random.randn(n)
            a = np.maximum(a + jitter, 0.0)
            a = simplex_projection(a)
            last_improvement = it  # reset stagnation counter

        it += 1

    # final safety – make sure we return a list of Python floats
    final_sequence = best_seq.tolist()
    print(f"[final] iterations={it}, best score={best_score:.6f}, length={n}")
    return final_sequence
