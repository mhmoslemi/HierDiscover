```
def solve_convolution_lp(f_seq, rhs):
    n = len(f_seq)
    c = -np.ones(n)  # maximize sum
    # Build A_ub and b_ub
    rows = []
    for k in range(2*n-1):
        # Row of length n: for each j, coefficient = f_seq[k-j] if valid
        row = np.zeros(n)
        for j in range(n):
            i = k - j
            if 0 <= i < n:
                row[j] = f_seq[i]
        rows.append(row)
    A_ub = np.vstack([rows, -np.eye(n)])  # includes -b_j <= 0 i.e., b >= 0
    b_ub = np.concatenate([np.full(2*n - 1, rhs), np.zeros(n)])
    # Solve
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, method='highs')
    if res.success:
        return res.x
    else:
        return None
```

```python
import numpy as np, time, math
from scipy.optimize import linprog

def normalize(seq):
    n = len(seq)
    target = math.sqrt(2.0 * n)
    total = np.sum(seq)
    if total == 0:
        return np.full_like(seq, target / n)
    return seq * (target / total)

def init_exponential(n, c=1.0):
    t = c / n
    q = 1.0 - t
    idx = np.arange(n)
    seq = q ** idx
    seq = normalize(seq.astype(float))
    return seq

def solve_lp(f_seq, rhs):
    n = len(f_seq)
    c = -np.ones(n)
    # Build constraints: For each shift k, sum_i f_i * b_{k-i} <= rhs
    rows = []
    for k in range(2 * n - 1):
        row = np.zeros(n)
        # i from max(0, k-(n-1)) to min(k, n-1)
        i_min = max(0, k - (n - 1))
        i_max = min(k, n - 1)
        # then j = k - i
        # So fill row[j] = f_i
        # vectorized:
        i_vals = np.arange(i_min, i_max + 1)
        j_vals = k - i_vals
        row[j_vals] = f_seq[i_vals]
        rows.append(row)
    A_ub = np.vstack([rows, -np.eye(n)])
    b_ub = np.concatenate([np.full(2 * n - 1, rhs), np.zeros(n)])
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, method='highs', options={'time_limit': 5.0})
    if res.success:
        return res.x
    return None

def local_search(seq, budget):

        # find shift(s) achieving max
        kmax = np.argmax(conv)
        # compute products for that shift
        i_min = max(0, kmax - (n - 1))
        i_max = min(kmax, n - 1)
        i_vals = np.arange(i_min, i_max + 1)
        j_vals = kmax - i_vals
        products = seq[i_vals] * seq[j_vals]
        # pick donor pair with largest product

        i_donor = i_vals[idx]
        j_donor = j_vals[idx]

        # compute delta as small fraction

        new_seq[i_donor] -= delta

            best = new_seq.copy()
```

"""
Propose a step‑function (non‑negative heights) that minimizes
the evaluation function

    2 * n * max(conv(a, a)) / (sum(a)**2)

The algorithm mixes three ideas:

1. **Smart initialization** – an exponential decay a_i = q**i
   (with q ≈ 1 – c/n) gives a good start (value ≈ 1.8 < 2).

2. **LP direction** – for the current sequence f,
   solve a linear program that maximises the total sum of a
   new sequence g while keeping the convolution bound
   conv(f, g) ≤ M (M = max(conv(f, f))).
   Mixing f and g reduces the bound.

3. **Line‑search & occasional random perturbations** – after the LP
   we try several mixing ratios t and keep the best improvement.
   If the LP fails we apply a tiny random Gaussian perturbation.

Multiple lengths and several initial decay parameters are tried
within the allotted time budget; the best sequence found is returned.
"""

import time
import math
import numpy as np
from scipy.optimize import linprog

# ----------------------------------------------------------------------
# Helper utilities
# ----------------------------------------------------------------------


def normalize_to_target(seq: np.ndarray) -> np.ndarray:
    """
    Scale ``seq`` so that its sum equals sqrt(2 * n) where n = len(seq).
    The scaling does not change the evaluation value because it is
    homogeneous of degree 0.
    """
    n = seq.shape[0]
    target = math.sqrt(2.0 * n)
    total = np.sum(seq)
    if total == 0:
        # degenerate case – give a tiny uniform vector
        return np.full_like(seq, target / n, dtype=float)
    return seq * (target / total)


def exponential_init(n: int, c: float = 1.0) -> np.ndarray:
    """
    Return an exponential decay vector of length n:
        a_i = (1 - c/n) ** i      (i = 0 … n‑1)
    Afterwards the vector is normalised to the required sum.
    """
    t = c / n
    q = 1.0 - t
    idx = np.arange(n, dtype=float)
    seq = q ** idx
    return normalize_to_target(seq)


def solve_convolution_lp(f_seq: np.ndarray, rhs: float):
    """
    Solve the linear program
        maximise   sum(g)
        subject to conv(f, g) <= rhs    (for every shift)
                   g >= 0
    Returns the optimal g as a 1‑d NumPy array or ``None`` on failure.
    """
    n = f_seq.shape[0]
    # objective: maximise sum(g)  → minimise -sum(g)
    c = -np.ones(n, dtype=float)

    # Build the constraint matrix A_ub * g <= b_ub
    # One row for each possible shift (2*n-1 rows)
    rows = []
    for k in range(2 * n - 1):
        # indices i that contribute to shift k
        i_min = max(0, k - (n - 1))
        i_max = min(k, n - 1)
        i_vals = np.arange(i_min, i_max + 1, dtype=int)
        j_vals = k - i_vals                 # j = k - i
        row = np.zeros(n, dtype=float)
        row[j_vals] = f_seq[i_vals]         # coefficient = f_i
        rows.append(row)

    # non‑negativity constraints: -g <= 0
    A_ub = np.vstack([np.asarray(rows), -np.eye(n)])
    b_ub = np.concatenate([np.full(2 * n - 1, rhs, dtype=float), np.zeros(n, dtype=float)])

    # Solve – use the HiGHS solver (fast and reliable).  A modest time limit
    # protects us from pathological runs.
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub,
                      method='highs',
                      options={'time_limit': 5.0})
        if res.success:
            return res.x
    except Exception:
        pass
    return None


def random_perturb(seq: np.ndarray) -> np.ndarray:
    """
    Apply a tiny Gaussian perturbation to every entry and clip to ≥0.
    The sum is not preserved – the caller will renormalise afterwards.
    """
    noise = np.random.randn(seq.size) * 0.01 * np.mean(seq)
    new_seq = seq + noise
    new_seq = np.maximum(new_seq, 0.0)
    return new_seq


def line_search_mix(f: np.ndarray, g: np.ndarray, cur_score: float):
    """
    Try a few mixing ratios (1‑t)*f + t*g, re‑normalise each candidate,
    evaluate it and return the best (seq, score, improved_flag).
    """
    best_seq = f
    best_score = cur_score
    improved = False
    # candidate mixing factors – a little exploration around the small
    # step used in the reference implementation.
    for t in (0.02, 0.05, 0.1, 0.2, 0.3):
        cand = (1.0 - t) * f + t * g
        cand = normalize_to_target(cand)
        try:
            cand_score = evaluate_sequence(cand.tolist())
        except Exception:
            continue
        if cand_score < best_score - 1e-12:
            best_seq = cand
            best_score = cand_score
            improved = True
            break
    return best_seq, best_score, improved

# Main entry point
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a sequence of non‑negative heights that yields a low
    evaluation value.  The function respects the global time budget and
    returns the best sequence discovered (list of floats).
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5   # a tiny safety margin

    best_overall_seq = None
    best_overall_score = float('inf')

    # ------------------------------------------------------------------
    # Lengths to explore – a mix of modest sizes (fast LP) and the
    # existing construction length if it is reasonable.
    # ------------------------------------------------------------------
    lengths = [200, 300, 400, 500, 800]
    if 'height_sequence_1' in globals():
        # reuse the known construction as a fallback baseline
        lengths.append(len(height_sequence_1))

    for n in lengths:
        if time.time() >= deadline:
            break

        # ------------------------------------------------------------------
        # Try a few exponential decay parameters as starting points.
        # ------------------------------------------------------------------
        for c_param in (0.5, 1.0, 2.0):
            if time.time() >= deadline:
                break

            # initialise
            curr = exponential_init(n, c=c_param)
            try:
                curr_score = evaluate_sequence(curr.tolist())
            except Exception:
                continue

            if curr_score < best_overall_score:
                best_overall_score = curr_score
                best_overall_seq = curr.tolist()
                print(f"[init] n={n}, c={c_param:.2f} → score={curr_score:.6f}")

            # ------------------------------------------------------------------
            # Main optimisation loop for this (n, c) pair.
            # ------------------------------------------------------------------
            iter_without_improve = 0
            while time.time() < deadline:
                # current bound M
                conv_full = np.convolve(curr, curr)
                M = conv_full.max()

                # Direction via LP
                g = solve_convolution_lp(curr, M)

                if g is None:
                    # LP failed – fall back to a small random step
                    curr = random_perturb(curr)
                    curr = normalize_to_target(curr)
                    continue

                # normalise the LP solution to the same sum as ``curr``
                g = normalize_to_target(g)

                # try mixing several t‑values
                cand, cand_score, improved = line_search_mix(curr, g, curr_score)

                if improved:
                    curr = cand
                    curr_score = cand_score
                    iter_without_improve = 0
                    if curr_score < best_overall_score:
                        best_overall_score = curr_score
                        best_overall_seq = curr.tolist()
                        print(f"[improve] n={n}, c={c_param:.2f}, score={curr_score:.6f}")
                else:
                    # no progress – inject a random perturbation to escape
                    curr = random_perturb(curr)
                    curr = normalize_to_target(curr)
                    iter_without_improve += 1

                # simple early‑stop for this start if we stagnate
                if iter_without_improve >= 30:
                    break

    # ------------------------------------------------------------------
    # As a final safety net, if we never improved beyond the initial
    # (uniform) construction, fall back to that.
    # ------------------------------------------------------------------
    if best_overall_seq is None:
        # uniform fallback (always valid)
        n = 300
        best_overall_seq = [1.0] * n
    return best_overall_seq
