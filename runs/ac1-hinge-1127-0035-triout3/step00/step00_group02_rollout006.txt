def propose_candidate(seed=123, budget_s=1000):
    import numpy as np, time, random
    np.random.seed(seed)
    random.seed(seed)
    deadline = time.time() + budget_s - 2.0
    n = 2000
    
    # Initialize sequence
    seq = np.random.dirichlet(np.ones(n) * 0.5)  # sum 1
    best_seq = seq.copy()
    best_score = evaluate_sequence(best_seq.tolist())
    iter_count = 0
    # Hyperparameters
    step_fraction = 0.05   # max fraction of source weight to move
    temp_start = 0.01
    temp_end = 0.0001
    total_iters_est = 200000
    # while not time out
    while time.time() < deadline:
        iter_count += 1
        # compute gradient for max convolution positions
        b = np.convolve(seq, seq)
        M = b.max()
        max_positions = np.where(b == M)[0]
        # gradient
        g = np.zeros_like(seq)
        n = len(seq)
        for k in max_positions:
            # compute i range
            i_start = max(0, k - (n-1))
            i_end = min(n-1, k)
            # slice arrays
            i_range = np.arange(i_start, i_end+1)
            j_range = k - i_range
            # contributions
            # For i != j
            mask = (i_range != j_range)
            g[i_range[mask]] += seq[j_range[mask]]
            # For i == j
            diag = (i_range == j_range)
            g[i_range[diag]] += 2 * seq[i_range[diag]]
        # pick indices
        i_high = np.argmax(g)
        i_low = np.argmin(g)
        # move amount
        # We want to move from high to low proportionally to gradient difference
        delta = step_fraction * seq[i_high]  # may be zero if seq[i_high]==0
        if delta > 0:
            # Ensure not to violate nonnegativity
            new_seq = seq.copy()
            new_seq[i_high] -= delta
            new_seq[i_low] += delta
        else:
            # random tweak
            i = np.random.randint(n)
            j = np.random.randint(n)
            if i != j and seq[i] > 0:
                delta = step_fraction * seq[i]
                new_seq = seq.copy()
                new_seq[i] -= delta
                new_seq[j] += delta
            else:
                continue
        # evaluate
        try:
            score = evaluate_sequence(new_seq.tolist())
        except Exception as e:
            continue
        # Acceptance
        # simple hill climbing: accept if better
        if score < best_score:
            best_score = score
            best_seq = new_seq.copy()
            seq = new_seq.copy()
        else:
            # Simulated annealing acceptance
            # compute temperature schedule based on iteration
            # T decreases linearly
            t = max(temp_end, temp_start * (1 - iter_count / total_iters_est))
            if np.random.rand() < np.exp(-(score - best_score) / t):
                seq = new_seq.copy()
        # optionally random restart if stagnates
        if iter_count % 5000 == 0:
            # evaluation of current seq may be same as best for long time; maybe random perturbation
            if np.random.rand() < 0.3:
                # random restart with same sum
                seq = np.random.dirichlet(np.ones(n) * 0.5)
            else:
                # small random perturbation to escape flat region
                idx = np.random.randint(n)
                delta = (np.random.rand() - 0.5) * 0.1
                seq[idx] = max(0.0, seq[idx] + delta)
                # renormalize to keep sum 1
                sum_seq = seq.sum()
                if sum_seq > 0:
                    seq = seq / sum_seq
        # occasional print
        if iter_count % 1000 == 0:
            print(f"Iter {iter_count}, best_score {best_score:.6f}")
    # final: convert to list and return
    return best_seq.tolist()

def propose_candidate(seed=123, budget_s=1000):
    """Hill-climbing / simulated annealing search for a non‑negative sequence
    that minimizes the evaluation function.  Starting from either the known
    construction ``height_sequence_1`` or a random Dirichlet vector, each
    iteration computes the convolution, finds the index k where the convolution
    attains its maximal value and moves a small amount of mass from the
    coefficient which contributes most to this peak to the coefficient that
    contributes least.  Occasionally random perturbations and a simple
    simulated‑annealing acceptance criterion are used to avoid stagnation.
    The best sequence seen before the time limit is returned.
    """
    import numpy as np, time, random
    # set rng
    rng = np.random.default_rng(seed)
    random.seed(seed)

    deadline = time.time() + budget_s - 0.5  # keep margin

    # Choose length
    n = 2000   # could be tuned

    # Initialise
    try:
        seq = np.array(list(height_sequence_1), dtype=float)
        # Ensure positive length, if too long cut or pad
        if len(seq) != n:
            # adjust length by truncating or padding with small values
            if len(seq) > n:
                seq = seq[:n]
            else:
                pad = np.full(n - len(seq), 0.0)
                seq = np.concatenate([seq, pad])
    except NameError:
        seq = rng.dirichlet(np.ones(n) * 0.5)

    # Normalise sum to 1 (optional but convenient)
    total = seq.sum()
    if total > 0:
        seq = seq / total

    best_seq = seq.copy()
    best_score = evaluate_sequence(best_seq.tolist())

    # hyper‑parameters
    step_frac = 0.05   # fraction of source coefficient moved each iteration
    temp_start = 0.01
    temp_end = 1e-4
    estimated_iters = 100000   # used for linear temperature schedule

    iter_no = 0
    no_improve_cnt = 0

    while time.time() < deadline:
        iter_no += 1

        # convolution and max index(es)
        conv = np.convolve(seq, seq)
        M = conv.max()
        max_pos = np.where(conv == M)[0]

        # gradient for the maximal positions
        grad = np.zeros_like(seq)
        nlen = len(seq)
        for k in max_pos:
            i_start = max(0, k - (nlen - 1))
            i_end = min(nlen - 1, k)
            i_idx = np.arange(i_start, i_end + 1, dtype=int)
            j_idx = k - i_idx
            # diagonal entries where i == j
            diag_mask = (i_idx == j_idx)
            # non‑diagonal
            nd_mask = ~diag_mask
            grad[i_idx[nd_mask]] += seq[j_idx[nd_mask]]
            grad[i_idx[diag_mask]] += 2.0 * seq[i_idx[diag_mask]]

        # choose source and target indices
        src = int(np.argmax(grad))
        tgt = int(np.argmin(grad))

        # safety: avoid moving from a zero coefficient
        if seq[src] <= 1e-12:
            # fall back to random perturbation
            i = rng.integers(nlen)
            j = rng.integers(nlen)
            if i != j and seq[i] > 1e-12:
                src, tgt = i, j
            else:
                # small random jitter
                seq = seq + (rng.random(nlen) - 0.5) * 1e-4
                seq = np.clip(seq, 0.0, None)
                seq = seq / seq.sum()
                continue

        delta = step_frac * seq[src]
        if delta <= 0:
            continue

        candidate = seq.copy()
        candidate[src] -= delta
        candidate[tgt] += delta

        # evaluate
        try:
            cand_score = evaluate_sequence(candidate.tolist())
        except Exception:
            continue

        # acceptance rule
        if cand_score < best_score:
            best_score = cand_score
            best_seq = candidate.copy()
            seq = candidate.copy()
            no_improve_cnt = 0
        else:
            # simulated annealing
            temperature = max(temp_end,

            # probability of accepting a worse move
            if np.random.rand() < np.exp(-(cand_score - best_score) / temperature):

    # final normalisation (not required for the score but keeps the output tidy)

    return best_seq.tolist()

def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Hill‑climbing / simulated‑annealing search for a non‑negative step‑function
    that minimises ``evaluate_sequence``.

    The algorithm works as follows

    1. Initialise a sequence:
        * if the global variable ``height_sequence_1`` is present, use it (it
          already has a good length);
        * otherwise start from a random Dirichlet vector of length ``n``.
    2. Repeatedly
        * compute the self‑convolution ``b = a * a`` and locate the index/indices
          where ``b`` attains its maximum ``M``;
        * build a simple gradient: for every maximal index ``k`` the contribution
          of coefficient ``a_i`` to ``M`` is ``a_{k‑i}`` (twice if ``i = k‑i``);
          the coefficient with the largest gradient contributes most to the peak,
          the one with the smallest gradient contributes least;
        * move a small fraction of mass from the “high‑gradient’’ index to the
          “low‑gradient’’ index (the sum of the sequence stays unchanged);
        * accept the move if it improves the score; otherwise accept it with a
          temperature‑controlled probability (simulated annealing) and otherwise
          keep the current sequence.
        * occasional random perturbations are applied when progress stalls.
    3. Keep the best sequence seen before the time budget expires and return it.

    The method never produces negative entries and the sum stays equal to 1,
    therefore the security checks inside ``evaluate_sequence`` are satisfied.
    """
    import time
    import random
    import numpy as np

    # ----------------------------------------------------------------------
    # Helper utilities
    # ----------------------------------------------------------------------
    def _initial_sequence():
        """Return an initial sequence and its length."""
        try:
            # Use the previously discovered construction if it exists.
            seq0 = np.array(list(height_sequence_1), dtype=float)
            n0 = seq0.size
            # Normalise to unit sum (scale does not affect the score).
            if seq0.sum() > 0:
                seq0 = seq0 / seq0.sum()
            # Clip any accidental negative values.
            seq0 = np.clip(seq0, 0.0, None)
            return seq0
        except Exception:
            # No previous construction – start from a random Dirichlet vector.
            n0 = 2000
            rng = np.random.default_rng(seed)
            seq0 = rng.dirichlet(np.ones(n0) * 0.5)
            return seq0

    def _gradient_for_max_positions(a, max_pos):
        """
        Compute a simple sub‑gradient of the maximal convolution value.
        For a given maximal shift ``k`` the derivative of ``b_k`` w.r.t. ``a_i``
        is ``a_{k-i}`` (twice if ``i = k-i``).  The gradient summed over all
        maximal positions tells which coefficients influence the peak most.
        """
        n = a.shape[0]
        g = np.zeros_like(a)
        for k in max_pos:
            i_start = max(0, k - (n - 1))
            i_end = min(n - 1, k)
            i = np.arange(i_start, i_end + 1, dtype=int)
            j = k - i
            # diagonal (i == j) contributes 2*a_i, off‑diagonal contributes a_j
            diag = (i == j)
            off = ~diag
            g[i[off]] += a[j[off]]
            g[i[diag]] += 2.0 * a[i[diag]]
        return g

    # ----------------------------------------------------------------------
    # Main search
    # ----------------------------------------------------------------------
    rng = np.random.default_rng(seed)
    random.seed(seed)

    deadline = time.time() + budget_s - 0.5  # keep a small safety margin

    a = _initial_sequence()
    n = a.shape[0]

    # Evaluate the starting point.
    best_seq = a.copy()
    best_score = evaluate_sequence(best_seq.tolist())

    # Hyper‑parameters
    step_frac = 0.05                # fraction of a[src] moved each iteration
    temp_start = 0.02               # initial temperature for SA
    temp_end = 1e-4                 # final temperature
    estimated_iters = 200_000       # used for linear cooling schedule
    no_improve = 0                  # counter for stagnation detection

    it = 0
    while time.time() < deadline:
        it += 1

        # ------------------------------------------------------------------
        # 1) convolution and maximal positions
        # ------------------------------------------------------------------
        conv = np.convolve(a, a)
        M = conv.max()
        max_pos = np.where(conv == M)[0]

        # ------------------------------------------------------------------
        # 2) gradient for the maximal positions
        # ------------------------------------------------------------------
        grad = _gradient_for_max_positions(a, max_pos)

        # Pick source (largest gradient) and target (smallest gradient).
        src = int(np.argmax(grad))
        tgt = int(np.argmin(grad))

        # Guard against degenerate sources (zero mass).
        if a[src] <= 1e-14:
            # fall back to a tiny random move
            src = int(rng.integers(n))
            tgt = int(rng.integers(n))
            if src == tgt or a[src] <= 1e-14:
                # give up this iteration
                continue

        delta = step_frac * a[src]
        if delta <= 0:
            continue

        cand = a.copy()
        cand[src] -= delta
        cand[tgt] += delta

        # ------------------------------------------------------------------
        # 3) evaluate candidate
        # ------------------------------------------------------------------
        try:
            cand_score = evaluate_sequence(cand.tolist())
        except Exception:
            # safety – ignore malformed candidates
            continue

        # ------------------------------------------------------------------
        # 4) acceptance rule
        # ------------------------------------------------------------------
        if cand_score < best_score:
            best_score = cand_score
            best_seq = cand.copy()
            a = cand.copy()
            no_improve = 0
        else:
            # simulated‑annealing acceptance
            temperature = max(
                temp_end,
                temp_start * (1.0 - it / estimated_iters)
            )
            prob = np.exp(-(cand_score - best_score) / temperature)
            if rng.random() < prob:
                a = cand.copy()
            no_improve += 1

        # ------------------------------------------------------------------
        # 5) occasional random perturbation to avoid being stuck
        # ------------------------------------------------------------------
        if no_improve > 3000:
            # small jitter on a random subset
            mask = rng.random(n) < 0.02
            if mask.any():
                jitter = (rng.random(mask.sum()) - 0.5) * step_frac
                cand2 = a.copy()
                cand2[mask] += jitter
                cand2 = np.clip(cand2, 0.0, None)
                # re‑normalise to keep the sum exactly 1 (scale‑invariant)
                s = cand2.sum()
                if s > 0:
                    cand2 = cand2 / s
                a = cand2
            no_improve = 0

        # ------------------------------------------------------------------
        # 6) periodic progress report
        # ------------------------------------------------------------------
        if it % 2000 == 0:
            print(f"[{it:6d}] best_score = {best_score:.8f}")

    # ----------------------------------------------------------------------
    # Return the best sequence found (as a plain Python list)
    # ----------------------------------------------------------------------
    # Normalise once more for neatness – the score is scale‑invariant.
    if best_seq.sum() > 0:
        best_seq = best_seq / best_seq.sum()
    return best_seq.tolist()
