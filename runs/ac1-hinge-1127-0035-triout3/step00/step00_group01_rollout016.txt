def project_simplex(v):
    """Project v onto probability simplex {x >=0, sum x = 1}."""
    # Sort v descending
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(len(v)) + 1
    # find rho: largest i where u_i - cssv_i / i > 0
    rho = np.nonzero(u - cssv / ind > 0)[0][-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0)
    return w

"""
propose_candidate
-----------------

Search for a non‑negative step‑function (a list of heights) that minimises

    2 * n * max(conv(a,a)) / (sum(a)**2)

where ``conv`` is the discrete convolution implemented in ``evaluate_sequence``.
The algorithm works in three stages:

1. **Parametric grid search** –  try sequences of the form
       a_i = (i + 1 + β·n)^{-p}
   with exponents p close to 0.5 (the analytically optimal 1/√i law)
   and a shift β∈[0,0.9] that moves the singularity away from the
   first element.  The best candidate from the grid is kept.

2. **Projected gradient descent** – starting from the best grid point,
   repeatedly move the probability vector in the direction that most
   reduces the current maximal convolution entry.  The sub‑gradient for a
   maximal index k is   g_i = 2 * a_{k-i} (if the index is in range).
   After each step we project back onto the probability simplex
   (non‑negative entries that sum to 1).  The step size is scaled so that
   no coordinate changes by more than a small constant.

3. **Return the best sequence found** – the sequence is returned as a plain
   Python list of floats (scaling does not matter, the evaluation function
   is homogeneous).

The whole search respects the supplied time budget (default 1000 s) and
will always output a valid sequence even if the time runs out early.
"""

import time
import numpy as np

def _project_simplex(v):
    """
    Project a vector onto the probability simplex:
        {x | x_i >= 0,  sum_i x_i = 1}
    Uses the O(n log n) algorithm of Duchi et al.
    """
    if v.ndim != 1:
        raise ValueError("simplex projection expects a 1‑D array")
    n = v.shape[0]
    # sort v in descending order
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    # find the rho index
    rho = np.where(u - cssv / ind > 0)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def _conv_fft(a, nfft):
    """
    Return the linear convolution a * a using a real FFT.
    The output length is 2*len(a)-1 (truncated to that length).
    """
    fa = np.fft.rfft(a, n=nfft)
    conv_full = np.fft.irfft(fa * fa, n=nfft)
    return conv_full[: 2 * len(a) - 1]
