"""
Propose a step‑function sequence with a low evaluation score.

The core idea is analytic: for a probability vector
    a_i ∝ (i+1)^(-p)   (i = 0,…,n‑1)
the self‑convolution becomes flatter as the exponent p grows.
The asymptotic value of the evaluation function is

    E(p) = 2 (1‑p)^2 * Γ(1‑p)^2 / Γ(2‑2p)

which attains its minimum for p≈0.55 giving E≈1.45 < 1.503.
We therefore scan a small grid of exponents, keep the best power‑law
vector, and finish with a few cheap gradient‑descent steps that
explicitly reduce the current maximal convolution coefficient.
All operations stay inside the allowed time budget (≤ 1000 s) and
respect the non‑negative, bounded‑by‑1000 constraints of
`evaluate_sequence`.

The returned list can be fed directly to `evaluate_sequence`.
"""

import time
import numpy as np

def _fast_score(seq: np.ndarray) -> float:
    """Compute the evaluation function using FFT (much faster than np.convolve)."""
    n = len(seq)
    size = 1 << ((2 * n - 1).bit_length())          # next power of two
    fft_seq = np.fft.rfft(seq, n=size)
    conv = np.fft.irfft(fft_seq * fft_seq, n=size)
    max_b = conv[: 2 * n - 1].max()
    sum_a = seq.sum()
    # the denominator is never zero because we keep a positive sum
    return 2.0 * n * max_b / (sum_a * sum_a)


def _gradient_step(seq: np.ndarray, lr: float) -> tuple[np.ndarray, float]:
    """
    One gradient descent step that tries to lower the maximal convolution entry.

    Returns the new sequence (projected onto the simplex) and the new maximal
    convolution value.
    """
    n = len(seq)
    size = 1 << ((2 * n - 1).bit_length())
    fft_seq = np.fft.rfft(seq, n=size)
    conv = np.fft.irfft(fft_seq * fft_seq, n=size)
    conv = conv[: 2 * n - 1]
    max_val = conv.max()
    k = int(conv.argmax())                     # index of the maximal entry

    # gradient of the k‑th convolution coefficient: ∂b_k/∂a_j = 2 a_{k‑j}
    grad = np.zeros_like(seq)
    i_start = max(0, k - (n - 1))
    i_end = min(k, n - 1)                       # inclusive
    idx = np.arange(i_start, i_end + 1, dtype=int)
    rev = k - idx
    grad[i_start : i_end + 1] = 2.0 * seq[rev]

    new_seq = seq - lr * grad
    new_seq = np.maximum(new_seq, 0.0)          # enforce non‑negativity
    total = new_seq.sum()
    if total == 0.0:                            # fallback to uniform if we annihilated everything
        new_seq = np.ones_like(seq) / n
    else:
        new_seq /= total                        # project onto the simplex (sum = 1)

    # compute the new max after the update (again via FFT)
    fft_new = np.fft.rfft(new_seq, n=size)
    conv_new = np.fft.irfft(fft_new * fft_new, n=size)
    max_new = conv_new[: 2 * n - 1].max()
    return new_seq, max_new
