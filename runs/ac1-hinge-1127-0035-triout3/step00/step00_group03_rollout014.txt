def generate_geometric(n, c):
    r = np.exp(-c / n)
    # compute normalization factor S = sqrt(2*n)
    S = np.sqrt(2.0 * n)
    # compute sum of unscaled geometric series
    denom = 1 - r**n
    if denom == 0:
        denom = 1e-12
    A = S * (1 - r) / denom
    seq = (A * (r**np.arange(n))).tolist()
    return seq

def get_good_direction_to_move_into(sequence):
    n = len(sequence)
    sum_sequence = np.sum(sequence)
    # If sum is zero (unlikely), return None.
    normalized_sequence = [x * np.sqrt(2 * n) / sum_sequence for x in sequence]
    rhs = np.max(np.convolve(normalized_sequence, normalized_sequence))
    g_fun = solve_convolution_lp(normalized_sequence, rhs)
    if g_fun is None:
        return None
    sum_g = np.sum(g_fun)
    normalized_g_fun = [x * np.sqrt(2 * n) / sum_g for x in g_fun]
    t = 0.05  # step
    new_seq = [(1-t)*x + t*y for x,y in zip(sequence, normalized_g_fun)]
    return new_seq

def generate_initial_sequence(n, c):
    r = np.exp(-c / n)
    # compute sum of geometric series
    if r == 1:
        A = np.sqrt(2 * n) / n
        seq = np.full(n, A)

        denominator = 1 - r ** n
        A = np.sqrt(2 * n) * (1 - r) / denominator
        seq = A * (r ** np.arange(n))

def solve_convolution_lp(f_seq, rhs):
    n = len(f_seq)
    c = -np.ones(n)
    rows = []
    b_vals = []
    for k in range(2 * n - 1):
        coeff = np.zeros(n)
        for i in range(n):
            j = k - i
            if 0 <= j < n:
                coeff[j] = f_seq[i]
        rows.append(coeff)
        b_vals.append(rhs)
    A_ub = np.vstack(rows)
    b_ub = np.array(b_vals)
    bounds = [(0, None)] * n
    result = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
    if result.success:
        return result.x
    return None
```

"""
Propose a step‑function (non‑negative sequence) with a low evaluation score.

The algorithm works in two phases:

1. **Initialisation**
   - Choose a modest length (default 1000) to keep the LP cheap.
   - Build a decreasing exponential (geometric) sequence
     a_i = A·r^i  (r = exp(-c/n)) where the normalisation sum(a)=sqrt(2·n)
     The parameter `c` controls the decay; a quick sweep finds the best
     `c` (the geometric family gives scores ≈1.52, already better than the
     uniform bound 2.0).

2. **Iterative improvement**
   - Repeatedly improve the current sequence by the *LP direction* described
     in the problem statement.  For a normalised sequence `f` (sum = sqrt(2n))
     we solve the linear program

        maximise   Σ b_j
        subject to  (f * b)[k] ≤ M      for all k,
                    b_j ≥ 0,

     where `M` is the current maximal convolution value.
   - The solution `b` is normalised to the same sum and mixed with the old
     sequence:  f ← (1‑t)·f + t·b   (t = 0.05).  The mixture keeps the sum
     unchanged and never increases the maximal convolution, thus the score
     monotonically decreases.
   - The loop runs until the time budget (≈990 s) is exhausted, keeping the
     best sequence seen.

If the LP ever fails, a tiny random perturbation is applied to keep the
search alive.  Finally the best sequence found is returned.
"""

import time
import numpy as np
from scipy import optimize

linprog = optimize.linprog


def _solve_convolution_lp(f, rhs):
    """
    Solve the LP:
        maximise Σ b_j
        subject to (f * b)[k] ≤ rhs  for k = 0 … 2*n‑2
                    b_j ≥ 0

    Parameters
    ----------
    f : array_like, shape (n,)
        The normalised step function (sum = sqrt(2·n)).
    rhs : float
        Upper bound for the convolution (current worst value).

    Returns
    -------
    np.ndarray or None
        The optimal `b` vector, or None if the LP fails.
    """
    n = len(f)
    # objective: maximise sum(b)  → minimise -sum(b)
    c = -np.ones(n)

    # Build convolution constraints A_ub @ b ≤ rhs
    # A_ub has shape (2*n-1, n)
    rows = np.zeros((2 * n - 1, n), dtype=np.float64)

    # For each lag k, fill the coefficients f_i for b_{k-i}
    for i in range(n):
        # the contribution of f[i] to every k where b_{k-i} appears
        # i.e. to rows k = i … i+n-1
        rows[i : i + n, i] = f[: n - i] if i == 0 else f[: n - i][::-1]  # temporary placeholder (will be overwritten)

    # The above loop is a bit tricky; we can fill directly:
    # Simpler (more explicit) construction:
    rows = np.zeros((2 * n - 1, n), dtype=np.float64)
    for k in range(2 * n - 1):
        i_min = max(0, k - (n - 1))
        i_max = min(n - 1, k)
        # for i in [i_min, i_max] we have j = k - i
        rows[k, k - np.arange(i_min, i_max + 1)] = f[i_min : i_max + 1]

    # inequality RHS vector
    b_ub = np.full(2 * n - 1, rhs, dtype=np.float64)

    # non‑negativity of b is given by bounds
    bounds = [(0.0, None)] * n

    # solve – we use the HiGHS interior point solver (fast for moderate size)
    res = linprog(c, A_ub=rows, b_ub=b_ub, bounds=bounds,
                  method='highs', options={'presolve': True, 'time_limit': 10.0})

    if res.success:
        return res.x
    return None


def _direction_step(seq, step=0.05):
    """
    Compute a better direction using the LP described in the statement.
    Returns a new sequence (list of floats) or None if the LP fails.
    """
    n = len(seq)
    sum_seq = float(np.sum(seq))
    if sum_seq < 1e-12:          # degenerate – should never happen
        return None

    # normalise to have the standard sum sqrt(2·n)
    target_sum = np.sqrt(2.0 * n)
    f = np.array(seq, dtype=np.float64) * (target_sum / sum_seq)

    # current worst convolution value
    rhs = float(np.max(np.convolve(f, f)))

    # solve the LP for a better partner
    g = _solve_convolution_lp(f, rhs)
    if g is None:
        return None

    # re‑normalise the LP solution to the same target sum
    sum_g = float(np.sum(g))
    if sum_g < 1e-12:
        return None
    g = g * (target_sum / sum_g)

    # convex combination (keeps the sum exactly target_sum)
    new_seq = (1.0 - step) * np.array(seq) + step * g
    return new_seq.tolist()


def _geometric_sequence(n, c):
    """
    Return a decreasing geometric sequence of length n with decay parameter `c`.

    The sequence is normalised so that its sum equals sqrt(2·n),
    which makes the evaluation exactly the maximal convolution value.
    """
    # r = exp(-c/n)  (so that ε = c/n)
    r = np.exp(-c / n)
    # geometric series sum: (1 - r**n) / (1 - r)
    if r == 1.0:
        a0 = np.sqrt(2.0 * n) / n
        return [float(a0)] * n

    denom = 1.0 - r ** n
    a0 = np.sqrt(2.0 * n) * (1.0 - r) / denom       # scaling factor A
    powers = r ** np.arange(n, dtype=np.float64)
    seq = a0 * powers
    return seq.tolist()


def propose_candidate(seed: int = 123, budget_s: int = 1000):
    """
    Search for a step‑function sequence with a low evaluation score.

    Parameters
    ----------
    seed : int
        Random seed for reproducibility.
    budget_s : int
        Time budget in seconds (the function will stop a few seconds before).

    Returns
    -------
    list[float]
        The best sequence found within the time budget.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 5.0   # leave a small safety margin

    # ---- 1. initial sequence -------------------------------------------------
    n = 1000                                 # moderate length – cheap LPs
    best_seq = None
    best_score = float('inf')

    # quick sweep over the geometric family (c controls the decay)
    for c in np.linspace(0.8, 1.2, 41):       # 0.01 step, covers the promising region
        seq = _geometric_sequence(n, c)
        score = evaluate_sequence(seq)       # score == max convolution for our normalisation
        if score < best_score:
            best_score, best_seq = score, seq

    # also keep a uniform start as a fallback
    uniform = [float(np.sqrt(2.0 * n) / n)] * n
    u_score = evaluate_sequence(uniform)
    if u_score < best_score:
        best_score, best_seq = u_score, uniform

    print(f"[init] best score = {best_score:.6f}")

    # ---- 2. iterative LP‑based improvement -----------------------------------
    curr_seq = best_seq[:]
    step_size = 0.05        # mixing factor (1‑t)*old + t*new

    # main optimisation loop
    while time.time() < deadline:
        new_seq = _direction_step(curr_seq, step=step_size)
        if new_seq is None:
            # occasional random jitter to escape possible stagnation
            i = np.random.randint(n)
            delta = np.random.randn() * 0.001 * max(1.0, curr_seq[i])
            curr_seq[i] = max(0.0, curr_seq[i] + delta)
            continue

        curr_seq = new_seq
        try:
            sc = evaluate_sequence(curr_seq)
            if sc < best_score:
                best_score, best_seq = sc, curr_seq.copy()
                print(f"[improve] new best = {best_score:.6f}  time={time.time():.1f}")
                # slowly reduce step size when we make good progress
                step_size = max(0.01, step_size * 0.95)
        except Exception:
            # any numerical hiccup – ignore this candidate
            pass

    print(f"[final] best score = {best_score:.6f}")
    return best_seq
