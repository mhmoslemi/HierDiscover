def fast_eval(a):
    n = len(a)
    # compute conv via FFT
    size = 1
    while size < 2*n-1:
        size <<= 1
    A = np.fft.rfft(a, size)
    conv = np.fft.irfft(A * A, size)[:2*n-1]
    max_b = np.max(conv)
    sum_a = np.sum(a)
    if sum_a < 0.01:
        return np.inf
    return 2 * n * max_b / (sum_a**2)

"""
propose_candidate
-----------------

Search for a step‐function (non‑negative list of heights) that minimises the
`evaluate_sequence` score.

The algorithm is intentionally different from the LP‑based approach that was
used before:

1.  Randomly initialise a sequence of moderate length (100‑2000).  Three
    families of start points are used:
        * uniform (all entries equal),
        * pure random,
        * exponential decay a_i = exp(-c·i/n)  (c is a random constant).

    The sequence is normalised so that its sum equals 1 – this does not change
    the objective (the evaluation function is homogeneous of degree 0).

2.  Simulated annealing (SA) is run on the normalised sequence.
    A move transfers a random fraction of the mass from one index *i*
    to a different index *j*.  The new sequence is accepted
    unconditionally if the score improves, otherwise with probability
    `exp((old‑score‑new‑score)/temp)`.  The temperature decays slowly.

    The SA loop is cheap because the convolution needed for the score is
    evaluated by an FFT based routine (`fast_eval`) which runs in O(n log n).

3.  After SA, a short gradient‑descent phase tries to reduce the maximal
    convolution entry directly.  For the current peak at index `k_max`
    the gradient of the peak with respect to the sequence is
    `g_i = 2·a_{k_max‑i}` (zero outside the admissible range).  A small step
    opposite to that gradient is taken and the vector is re‑normalised to
    keep the sum equal to 1.

4.  The best sequence found over many independent restarts (bounded by the
    remaining time budget) is returned.  If nothing better than the
    supplied `height_sequence_1` is discovered, the fallback is a uniform
    sequence.

All operations use only NumPy and the standard library, respect the time
budget, and avoid any file or network I/O.  The returned list contains only
non‑negative floats, ready to be handed to `evaluate_sequence`.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper: fast evaluation of the objective using FFT (O(n log n))
def fast_eval(seq: np.ndarray) -> float:
    """Return the evaluation score for `seq` (no security checks)."""
    n = len(seq)
    sum_a = np.sum(seq)
    if sum_a < 0.01:
        return float("inf")
    # length of convolution = 2*n-1, choose next power of two for FFT
    size = 1 << ((2 * n - 1).bit_length())
    A = np.fft.rfft(seq, size)
    conv = np.fft.irfft(A * A, size)[: 2 * n - 1]
    max_b = np.max(conv)
    return float(2 * n * max_b / (sum_a ** 2))


# ----------------------------------------------------------------------
# Simulated annealing on a fixed length sequence
def anneal(initial_seq: np.ndarray, rng: np.random.Generator,
           max_iter: int, temp0: float, decay: float):
    """Perform SA on `initial_seq` (already normalised to sum‑1)."""
    a = initial_seq.copy()
    score = fast_eval(a)
    best_seq = a.copy()
    best_score = score
    temp = temp0
    n = len(a)

    for _ in range(max_iter):
        # pick two distinct indices
        i = rng.integers(0, n)
        j = rng.integers(0, n - 1)
        if j >= i:
            j += 1

        # move a random fraction of mass from i to j
        delta = rng.random() * a[i] * 0.5  # at most 50 % of a[i]
        if delta == 0.0:
            continue
        new_a = a.copy()
        new_a[i] -= delta
        new_a[j] += delta

        new_score = fast_eval(new_a)

        # accept if better, otherwise with Metropolis probability
        if new_score < score or rng.random() < np.exp((score - new_score) / temp):
            a = new_a
            score = new_score
            if new_score < best_score:
                best_score = new_score
                best_seq = new_a.copy()

        # temperature schedule
        temp = max(temp * decay, 1e-12)

    return best_seq, best_score


# ----------------------------------------------------------------------
# Gradient‑descent phase that targets the current maximal convolution entry
def gradient_refine(seq: np.ndarray, steps: int, eta0: float):
    """Few gradient steps aimed at reducing the maximal convolution value."""
    a = seq.copy()
    n = len(a)
    for step in range(steps):
        conv = np.convolve(a, a)
        kmax = int(np.argmax(conv))

        # gradient of the peak w.r.t. a is 2·a_{kmax‑i} on the overlapping window
        lo = max(0, kmax - (n - 1))
        hi = min(n - 1, kmax)
        idx = np.arange(lo, hi + 1, dtype=int)
        grad = np.zeros_like(a)
        grad[idx] = 2.0 * a[kmax - idx]

        # simple projected gradient (keep non‑negative, renormalise to sum‑1)
        eta = eta0 / (step + 1.0)   # decreasing step size
        a -= eta * grad
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s < 1e-12:
            break
        a /= s
    return a
