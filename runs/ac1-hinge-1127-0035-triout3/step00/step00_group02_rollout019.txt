"""
Optimising the evaluation bound by constructing a sparse set of indices.
The objective 2·n·max(conv)/sum² is minimised when the convolution maximum
is kept small while the ℓ¹‑norm (the sum) is made large.
If the sequence is binary (0/1) the convolution entry at k equals the
number of ordered pairs (i,j) in the support with i+j=k.
Thus, for a support S we have

    max(conv) = max_k |{ i∈S : k−i∈S }|   (ordered pair count)

and

    evaluation = 2·n·max_cnt / |S|² .

Hence we look for a large set S ⊂ {0,…,n−1} such that every sum
appears only a few times (low ordered‑pair count).  This is a
“B_h‑set” problem.  A greedy random construction is fast and works well
for the target bound ≈1.5 on n=3005.

The algorithm repeatedly builds a candidate set S with a random
ordering of the indices and a limit `max_allowed` on the ordered‑pair
count per sum.  After each construction we compute a cheap estimate of
the evaluation; only promising candidates are handed to the provided
`evaluate_sequence` function.  The best sequence found before the time
budget expires is returned.
"""

import time
import numpy as np


def _greedy_bset(n, max_allowed, rng):
    """
    Greedy construction of a set S ⊆ [0,n) such that for every sum s
    the ordered pair count does not exceed `max_allowed`.

    Returns:
        S (list[int]) – the selected indices,
        cnt (np.ndarray) – array of ordered‑pair counts per sum.
    """
    order = rng.permutation(n)
    cnt = np.zeros(2 * n - 1, dtype=np.int16)   # ordered‑pair counters
    S = []
    for x in order:
        # self‑pair would add 1 to cnt[2*x]
        if cnt[2 * x] + 1 > max_allowed:
            continue
        # check all existing elements
        ok = True
        for y in S:
            s = x + y
            if cnt[s] + 2 > max_allowed:   # (x,y) and (y,x)
                ok = False
                break
        if not ok:
            continue
        # add x to the set
        for y in S:
            cnt[x + y] += 2
        cnt[2 * x] += 1
        S.append(int(x))
    return S, cnt


def propose_candidate(seed=42, budget_s=1000, **kwargs):
    """
    Search for a step‑function sequence that lowers the evaluation metric.
    The search builds random B_h‑sets (bounded sum‑multiplicity sets) and
    keeps the best binary sequence found within the time budget.
    """
    rng = np.random.default_rng(seed)
    deadline = time.time() + budget_s - 5   # a small safety margin

    # target length – the previous construction used 3005, we keep it
    n = 3005

    # best solution so far (fallback: uniform sequence)
    best_seq = [float(np.sqrt(2 * n)) / n] * n
    best_score = evaluate_sequence(best_seq)

    # try several different limits on the ordered‑pair count
    allowed_values = [3, 4, 5, 6, 7, 8]

    while time.time() < deadline:
        max_allowed = int(rng.choice(allowed_values))
        S, cnt = _greedy_bset(n, max_allowed, rng)

        if not S:                     # should never happen, but be safe
            continue

        size = len(S)
        max_cnt = int(cnt.max())

        # cheap bound: eval ≤ 2·n·max_cnt / size²
        est = (2.0 * n * max_cnt) / (size * size)
        if est >= best_score:         # not promising
            continue

        # build the binary candidate
        seq = [0.0] * n
        for i in S:
            seq[i] = 1.0

        # now ask the real evaluator (it also applies caps etc.)
        try:
            score = evaluate_sequence(seq)
        except Exception:
            continue

        if score < best_score:
            best_score = score
            best_seq = seq
            print(
                f"[{time.time():.1f}] New best = {score:.6f} "
                f"(|S|={size}, max_cnt={max_cnt}, allowed={max_allowed})"
            )
            # stop early if we already beat the target
            if best_score <= 1.503:
                break

    return best_seq
