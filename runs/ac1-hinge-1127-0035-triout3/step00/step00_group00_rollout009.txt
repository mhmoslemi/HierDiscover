def simplex_projection(v):
    # v is np array
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0)
    return w

def compute_conv(a):
    n = a.shape[0]
    m = 2 * n - 1
    nfft = 1 << ((m - 1).bit_length())
    a_fft = np.fft.rfft(a, n=nfft)
    conv = np.fft.irfft(a_fft * a_fft, n=nfft)[:m]
    conv = np.real(conv) # already real
    return conv

def propose_candidate(seed=42, budget_s=1000, **kwargs):
    np.random.seed(seed)
    deadline=time.time()+budget_s-0.5

    try: n=len(height_sequence_1)
    except NameError: n=5000

    # Initialize a
    if 'height_sequence_1' in globals():
        a=np.array(height_sequence_1, dtype=float)
        # ensure nonneg and set sum=1
    else:
        a = np.random.dirichlet(np.ones(n))
    # Normalize sum and project:
    a = a / a.sum()

    best_seq = a.copy()
    best_score = evaluate_sequence(best_seq.tolist())
    print(f"Initial score {best_score}")

    step0 = 0.5   # initial step size maybe 0.5
    iter=0
    no_improve=0

    while time.time()<deadline:
        iter+=1
        # compute conv
        conv = compute_conv(a)
        max_idx = int(np.argmax(conv))
        max_val = conv[max_idx]

        # compute gradient
        grad = np.zeros_like(a)
        # Use vectorized approach
        idx = np.arange(n)
        j = max_idx - idx
        mask = (j >= 0) & (j < n)
        grad[mask] = a[j[mask]]
        # gradient step
        eta = step0 / np.sqrt(iter)
        a_new = a - eta * grad
        # project onto simplex
        a = project_to_simplex(a_new)
        # optionally, small noise
        if iter % 200 == 0:
            noise = np.random.randn(n) * (0.01 / np.sqrt(iter))
            a = project_to_simplex(a + noise)

        # evaluate
        score = evaluate_sequence(a.tolist())
        if score < best_score:
            best_score = score
            best_seq = a.copy()
            no_improve=0
            print(f"Iter {iter}: improved score {best_score}")
        else:
            no_improve+=1

        # restart if stuck
        if no_improve > 2000:
            # random restart
            a = np.random.dirichlet(np.ones(n))
            a = project_to_simplex(a)
            no_improve=0
            print(f"Iter {iter}: restart due to stagnation")
    print(f"Finished: best score {best_score}")
    return best_seq.tolist()

"""
Subgradient optimizer for the step‑function evaluation problem.

The goal is to minimise

    E(f) = 2·n·max(conv(f,f)) / (sum(f)²)

where ``conv`` is the linear (non‑circular) convolution.
The objective is scale‑invariant, therefore we keep the sequence on the
probability simplex (non‑negative and sum = 1) and minimise
    2·n·max(conv(p,p))

using a simple projected sub‑gradient method.  At each iteration we:

1. Compute the convolution with an FFT (O(n log n)).
2. Find the index *k* where the convolution attains its maximum.
3. The sub‑gradient of ``max(conv(p,p))`` at that index is a shifted copy
   of the current vector:  g[i] = p[k‑i]  (if the index is in range).
4. Take a step opposite to the gradient and project back onto the simplex.
5. Keep the best vector seen during the whole run.

Random restarts and occasional jitter help to escape plateaux.
The routine works for any length; if a global ``height_sequence_1`` is
available it is used as a starting point, otherwise a random Dirichlet
vector is generated.  The function runs for at most ``budget_s`` seconds
and returns the best sequence found.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper functions (top‑level, no closures)
# ----------------------------------------------------------------------


def simplex_projection(v: np.ndarray) -> np.ndarray:
    """
    Project a vector onto the probability simplex {x >= 0, sum(x) = 1}.
    Implements the O(n) algorithm of Duchi et al. (2008).
    """
    if np.all(v >= 0) and np.isclose(v.sum(), 1.0):
        # already on the simplex
        return v.copy()
    n = v.shape[0]
    u = np.sort(v)[::-1]                 # descending order
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)            # 1‑based indices
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all entries will be zero after projection – return a basis vector
        w = np.zeros_like(v)
        w[0] = 1.0
        return w
    rho = np.where(cond)[0][-1]          # last true index
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    # numerical safety: renormalise to exactly 1
    w_sum = w.sum()
    if w_sum == 0:
        w[0] = 1.0
    else:
        w /= w_sum
    return w


def linear_convolution(a: np.ndarray) -> np.ndarray:
    """
    Compute the linear convolution of a with itself using FFT.
    Returns the real part of the convolution of length 2·len(a)‑1.
    """
    n = a.shape[0]
    m = 2 * n - 1
    # power‑of‑two length for the FFT (speed + numerical stability)
    nfft = 1 << ((m - 1).bit_length())
    af = np.fft.rfft(a, n=nfft)
    conv = np.fft.irfft(af * af, n=nfft)[:m]
    # eliminate tiny negative values caused by rounding
    return np.maximum(conv, 0.0)


def gradient_at_max(conv: np.ndarray, a: np.ndarray) -> np.ndarray:
    """
    Sub‑gradient of max(conv(a,a)) w.r.t. ``a``.
    The gradient is a shifted copy of ``a`` determined by the index where the
    convolution attains its maximum.
    """
    n = a.shape[0]
    k = int(np.argmax(conv))            # index of the active constraint
    idx = np.arange(n, dtype=int)
    j = k - idx
    mask = (j >= 0) & (j < n)
    grad = np.zeros_like(a)
    grad[mask] = a[j[mask]]
    return grad


# ----------------------------------------------------------------------
# Main candidate proposer
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a step‑function (non‑negative vector) that minimises the
    evaluation function defined by ``evaluate_sequence``.
    The function runs for at most ``budget_s`` seconds and returns the best
    sequence found as a plain Python list.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5      # a tiny safety margin

    # ------------------------------------------------------------------
    # Determine the problem dimension
    # ------------------------------------------------------------------
    try:
        # ``height_sequence_1`` may be defined in the global scope.
        n = len(height_sequence_1)
        print(f"Using length of provided sequence: n = {n}")
    except Exception:
        n = 2000                # fallback length – balances speed and quality
        print(f"No global sequence found – using n = {n}")

    # ------------------------------------------------------------------
    # Initialise a feasible point on the simplex
    # ------------------------------------------------------------------
    if 'height_sequence_1' in globals():
        # start from the supplied construction, but re‑normalise to sum = 1
        a = np.array(height_sequence_1, dtype=float)
        a = np.maximum(a, 0.0)               # just in case
        if a.sum() == 0.0:
            a = np.random.rand(n)
        a = a / a.sum()
    else:
        # Dirichlet(0.5) – a bit spikier than the uniform Dirichlet
        a = np.random.gamma(shape=0.5, scale=1.0, size=n)
        a = a / a.sum()

    # best solution seen so far
    conv = linear_convolution(a)
    best_score = 2.0 * n * conv.max()
    best_seq = a.copy()
    print(f"Initial score: {best_score:.6f}")

    # algorithmic parameters
    step0 = 0.6                 # initial step size (will decay)
    iter_cnt = 0
    last_improve_iter = 0
    stagnation_limit = 5000    # iterations without improvement before a restart

    while time.time() < deadline:
        iter_cnt += 1

        # --------------------------------------------------------------
        # 1) compute the convolution and the sub‑gradient at the max
        # --------------------------------------------------------------
        conv = linear_convolution(a)
        grad = gradient_at_max(conv, a)

        # --------------------------------------------------------------
        # 2) take a gradient step and project back onto the simplex
        # --------------------------------------------------------------
        eta = step0 / np.sqrt(iter_cnt + 1.0)   # simple 1/√t schedule
        a_candidate = a - eta * grad
        a = simplex_projection(a_candidate)

        # --------------------------------------------------------------
        # 3) occasional random jitter to escape flat regions
        # --------------------------------------------------------------
        if iter_cnt % 2000 == 0:
            jitter = np.random.randn(n) * (0.02 / np.sqrt(iter_cnt))
            a = simplex_projection(a + jitter)

        # --------------------------------------------------------------
        # 4) evaluate the new point (fast analytic form)
        # --------------------------------------------------------------
        conv = linear_convolution(a)
        cur_score = 2.0 * n * conv.max()   # since sum(a)=1, denominator = 1

        if cur_score < best_score:
            best_score = cur_score
            best_seq = a.copy()
            last_improve_iter = iter_cnt
            print(f"[{iter_cnt:6d}] improved score: {best_score:.6f}")

        # --------------------------------------------------------------
        # 5) restart if we seem stuck for a long time
        # --------------------------------------------------------------
        if iter_cnt - last_improve_iter > stagnation_limit:
            # fresh random start (Dirichlet with a moderate concentration)
            a = np.random.gamma(shape=0.5, scale=1.0, size=n)
            a = a / a.sum()
            last_improve_iter = iter_cnt
            print(f"[{iter_cnt:6d}] restart due to stagnation")

    # ------------------------------------------------------------------
    # Return the best sequence found (as plain Python list)
    # ------------------------------------------------------------------
    print(f"Finished after {iter_cnt} iterations – best score: {best_score:.6f}")
    return best_seq.tolist()
