"""
Propose a step‑function (non‑negative sequence) whose self‑convolution is
almost flat.  The continuous arcsine law has the property that the sum of two
independent arcsine‑distributed variables is uniform on the interval; after a
discrete approximation this yields a very small maximum convolution value.
The algorithm

1. builds the discrete arcsine density on the index set {0,…,n‑1},
   a_i ∝ 1/√((i+1)*(n-i)).
2. normalises the vector to sum to 1 (the evaluation function is
   scale‑invariant, any normalisation works).
3. performs a few cheap projected‑gradient steps that explicitly try to
   decrease the current maximal convolution entry while staying on the
   probability simplex.
4. returns the best candidate found.

The method runs in O(n log n) time per iteration (FFT‑based convolution) and
comfortably finishes well before the 1000‑second budget.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities
# ----------------------------------------------------------------------
def project_onto_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex {x>=0, sum(x)=1}."""
    if np.all(v >= 0) and np.isclose(v.sum(), 1.0):
        return v.copy()
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # numerical safety – fall back to uniform distribution
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / float(rho)
    w = np.maximum(v - theta, 0.0)
    return w


def convolve_self_fft(a: np.ndarray) -> np.ndarray:
    """Return the linear convolution a*a using a power‑of‑two FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())   # next power of two
    A = np.fft.rfft(a, size)
    B = A * A
    conv = np.fft.irfft(B, size)[: 2 * n - 1]
    return conv


def max_convolution_gradient(a: np.ndarray, idx_max: int) -> np.ndarray:
    """
    Gradient of the maximal convolution entry B_idx = (a*a)[idx_max]
    with respect to a.  For an ordered convolution
        B_i = Σ_j a_j a_{i-j},
    ∂B_i/∂a_t = 2 * a_{i-t} (if the index is inside the array, else 0).
    """
    n = a.shape[0]
    g = np.zeros_like(a)
    low = max(0, idx_max - (n - 1))
    high = min(idx_max, n - 1)
    if low <= high:
        t_range = np.arange(low, high + 1)
        g[low : high + 1] = 2.0 * a[idx_max - t_range]
    return g


# ----------------------------------------------------------------------
# Main search routine
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs) -> list:
    """
    Generate a sequence that (almost) minimises the objective.

    Parameters
    ----------
    seed : int
        Random seed for the optional perturbation phase.
    budget_s : int
        Time budget in seconds; the function stops well before the limit.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 5.0   # leave a couple of seconds margin

    # ------------------------------------------------------------------
    # 1. Determine the length of the problem (use the known construction
    #    if it exists, otherwise fall back to a reasonable default).
    # ------------------------------------------------------------------
    try:
        n = len(height_sequence_1)          # global supplied by the harness
    except Exception:
        n = 7156                            # nominal length taken from the prompt

    # ------------------------------------------------------------------
    # 2. Initialise with the discrete arcsine law.
    #    a_i ∝ 1 / sqrt( (i+1)*(n-i) )
    # ------------------------------------------------------------------
    idx = np.arange(n, dtype=np.float64)
    raw = 1.0 / np.sqrt((idx + 1.0) * (n - idx))
    a = raw / raw.sum()                     # now Σ a_i = 1
    best_a = a.copy()
    best_max = np.max(convolve_self_fft(a))

    # ------------------------------------------------------------------
    # 3. Optional short projected‑gradient refinement.
    #    We minimise max_i (a*a)[i] while staying on the simplex.
    # ------------------------------------------------------------------
    step = 0.5                               # initial step size
    it = 0
    while time.time() < deadline and it < 5000:
        it += 1
        conv = convolve_self_fft(a)
        idx_max = int(np.argmax(conv))
        cur_max = conv[idx_max]

        # keep the best encountered candidate
        if cur_max < best_max:
            best_max = cur_max
            best_a = a.copy()

        # compute gradient that reduces the maximal entry
        g = max_convolution_gradient(a, idx_max)

        # simple line‑search / step‑size adaptation
        a_new = a - step * g
        a_new = project_onto_simplex(a_new)

        # evaluate the new point (cheap evaluation using the same conv)
        new_conv = convolve_self_fft(a_new)
        new_max = new_conv[int(np.argmax(new_conv))]

        # accept if we made progress, otherwise shrink the step
        if new_max < cur_max:
            a = a_new
            step = min(step * 1.05, 1.0)       # slowly increase step when successful
        else:
            step *= 0.5                        # back‑off on failure

        # occasional tiny random perturbation to escape possible plateaus
        if it % 200 == 0 and np.random.rand() < 0.15:
            perturb = np.random.randn(n) * 1e-4
            a_pert = a + perturb
            a_pert = project_onto_simplex(a_pert)
            a = a_pert

    # ------------------------------------------------------------------
    # 4. Return the best sequence found (as a plain Python list).
    # ------------------------------------------------------------------
    return best_a.tolist()
