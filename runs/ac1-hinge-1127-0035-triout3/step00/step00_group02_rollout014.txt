def propose_candidate(budget_s=1000):
    import time, numpy as np, random, math

    deadline = time.time() + budget_s - 0.1
    best_seq = None
    best_score = float('inf')
    # We'll try several restarts
    # define function score
    def score_seq(seq):
        s = np.sum(seq)
        if s < 0.01:
            return float('inf')
        conv = np.convolve(seq, seq)
        maxb = np.max(conv)
        return float(2 * len(seq) * maxb / (s*s))

    # initial generator
    def random_seq():
        n = random.randint(150, 600)   # moderate size
        # choose density d between 0.2 and 0.8
        density = random.uniform(0.2, 0.8)
        m = max(1, int(density * n))
        positions = random.sample(range(n), m)
        weights = np.random.rand(m)
        weights = weights / weights.sum()
        seq = np.zeros(n)
        seq[positions] = weights
        return seq.tolist()

    # optionally include height_sequence_1 as candidate
    global height_sequence_1
    if height_sequence_1 is not None:
        candidates = [list(height_sequence_1)]
    else:
        candidates = []
    # produce multiple random candidates
    for _ in range(5):
        candidates.append(random_seq())

    for seq in candidates:
        s = score_seq(seq)
        if s < best_score:
            best_score = s
            best_seq = seq

    # main SA loop
    cur_seq = np.array(best_seq, dtype=float)
    cur_score = best_score

    # initial temperature
    T0 = 0.5  # initial acceptance threshold on relative improvement
    # schedule: exponential decay
    start_time = time.time()
    # evaluate iteration count
    iteration = 0

    while time.time() < deadline:
        iteration += 1
        n = len(cur_seq)
        # propose move
        # pick two indices i, j distinct
        i, j = random.sample(range(n), 2)
        # determine amount to transfer: based on current a_i
        a_i = cur_seq[i]
        a_j = cur_seq[j]
        if a_i == 0:
            continue
        # sample epsilon between 0 and min(a_i, 1e-2) maybe
        eps = a_i * random.uniform(0.01, 0.1)  # move up to 10% of a_i
        # create new sequence
        new_seq = cur_seq.copy()
        transfer = min(eps, a_i)
        new_seq[i] -= transfer
        new_seq[j] += transfer
        # ensure nonnegativity
        # compute score
        new_score = score_seq(new_seq.tolist())
        # acceptance
        if new_score < cur_score:
            cur_seq = new_seq
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_seq = new_seq.tolist()
                print(f"Iter {iteration}: new best {best_score:.6f}")
        else:
            # temperature decreasing
            elapsed = time.time() - start_time
            T = T0 * (1 - (elapsed / (deadline - start_time)))  # linear cooling
            prob = math.exp(-(new_score - cur_score) / (T + 1e-12))
            if random.random() < prob:
                cur_seq = new_seq
                cur_score = new_score

        # Occasionally (every 10000 iter) attempt a "large" perturbation restart
        if iteration % 5000 == 0:
            # sometimes replace with new random sequence
            if random.random() < 0.2:
                cur_seq = np.array(random_seq())
                cur_score = score_seq(cur_seq.tolist())

    # Return best_seq as list of floats
    return best_seq

        g = solve_convolution_lp(normalized_seq.tolist(), rhs)
        if g is None:
            # random perturbation
            idx = np.random.randint(n)
            cur_seq[idx] = max(0, cur_seq[idx] + np.random.randn() * 0.01)
            continue
        sum_g = np.sum(g)
        normalized_g = g * np.sqrt(2 * n) / sum_g
        # try several t values
        for t in [0.01,0.05,0.1,0.2,0.4]:
            new_seq = (1 - t) * cur_seq + t * normalized_g
            sc = score_seq(new_seq.tolist())
            if sc < best_score:
                best_score = sc
                best_seq = new_seq.tolist()
                cur_seq = new_seq  # accept
                break
        else:
            # maybe random small move
            i,j = random.sample(range(n),2)
            eps = cur_seq[i] * 0.05
            new_seq = cur_seq.copy()
            eps = min(eps, cur_seq[i])
            new_seq[i] -= eps
            new_seq[j] += eps
            cur_seq = new_seq
    return best_seq, best_score

"""
search for a step‐function (non‑negative sequence) that minimises
evaluate_sequence(seq) = 2 * n * max(conv(seq,seq)) / (sum(seq)**2).

The algorithm works in three phases:

1.  Initialise a sequence (either a truncated known solution or a random
    Dirichlet vector) and normalise it so that sum(seq)=sqrt(2*n).  For a
    normalised sequence the evaluation reduces to the maximum convolution
    value, i.e. score = max(conv(seq,seq)).  The target is to push this
    maximum below the trivial uniform bound 2.0.

2.  Repeatedly improve the sequence using the “LP‑direction” idea from the
    state‑of‑the‑art code:
        – given the current normalised sequence f, compute its current bound
          M = max(conv(f,f)).
        – solve a linear program that finds a non‑negative vector g with the
          largest possible sum under the linear constraints
                conv(f,g) <= M .
        – scale g to the same normalisation (sum = sqrt(2*n)) and mix it with
          f:   f' = (1‑t)·f + t·g .  Small values of t guarantee that the new
          maximum convolution does not increase; occasional larger t’s help
          convergence.
        – keep the best sequence seen.

3.  If the LP fails or makes no progress, perform cheap random perturbations
    (move a small amount of mass between two coordinates) and re‑normalise.
    These moves allow the optimiser to escape plateaus.

The whole search runs for the allocated time budget (default 1000 s) and
returns the best sequence discovered.
"""

import time
import random
import math
import numpy as np
from scipy.optimize import linprog

# ----------------------------------------------------------------------
# Helper functions (all top‑level – no closures)
# ----------------------------------------------------------------------


def _score(seq):
    """Return the raw value of evaluate_sequence for a list of floats."""
    # evaluate_sequence is assumed to be already imported in the environment
    return evaluate_sequence(seq)


def _normalise(seq):
    """Scale a non‑negative sequence so that its sum equals sqrt(2*n)."""
    n = len(seq)
    target = math.sqrt(2.0 * n)
    s = np.sum(seq)
    if s == 0:
        # avoid division by zero – return a tiny uniform vector
        return np.full(n, target / n)
    return seq * (target / s)


def _conv_max(seq):
    """Maximum entry of the full convolution of seq with itself."""
    return np.max(np.convolve(seq, seq))


def _solve_lp(f_seq, rhs):
    """
    Solve the LP:
        maximise   sum(g_i)
        subject to conv(f,g)[k] <= rhs  (k = 0 .. 2n‑2)
                  g_i >= 0
    Returns the optimal g as a NumPy array or None on failure.
    """
    n = len(f_seq)
    # objective: maximise sum(g)  => minimise -sum(g)
    c = -np.ones(n)

    # Build the inequality matrix for the convolution constraints.
    # Row k corresponds to sum_i f_i * g_{k-i}  (indices out of range ignored)
    A = np.zeros((2 * n - 1, n))
    for k in range(2 * n - 1):
        # only i where 0 <= k-i < n contribute
        i_low = max(0, k - (n - 1))
        i_high = min(n - 1, k)
        # for these i, the coefficient of g_{k-i} is f_i
        for i in range(i_low, i_high + 1):
            j = k - i
            A[k, j] = f_seq[i]

    b = np.full(2 * n - 1, rhs, dtype=float)

    # Use the HiGHS interior‑point solver (fast for dense problems)
    try:
        res = linprog(c, A_ub=A, b_ub=b, bounds=[(0, None)] * n,
                      method='highs', options={'disp': False})
    except Exception:          # safety net – if linprog crashes
        return None

    if res.success:
        return res.x
    return None


def _lp_improve(seq, max_iter, time_deadline):
    """
    Perform the LP‑based improvement loop on a normalised sequence.
    Returns the best sequence (NumPy array) and its score.
    """
    n = len(seq)
    best_seq = seq.copy()
    best_score = _conv_max(best_seq)          # because sum = sqrt(2n)
    cur_seq = best_seq.copy()

    # mixing parameters that will be tried each iteration
    t_vals = (0.02, 0.05, 0.10, 0.20)

    it = 0
    while it < max_iter and time.time() < time_deadline:
        it += 1

        # current bound M
        M = _conv_max(cur_seq)

        # solve LP for a direction with larger total mass
        g = _solve_lp(cur_seq.tolist(), M)
        if g is None:
            # LP failed – fall back to a cheap random move
            i, j = random.sample(range(n), 2)
            delta = cur_seq[i] * 0.07
            delta = min(delta, cur_seq[i])
            new_seq = cur_seq.copy()
            new_seq[i] -= delta
            new_seq[j] += delta
            new_seq = _normalise(new_seq)
            cur_seq = new_seq
            continue

        # normalise the LP solution to the same sum as cur_seq
        g_norm = _normalise(g)

        improved = False
        for t in t_vals:
            cand = (1.0 - t) * cur_seq + t * g_norm
            # cand is already normalised (convex combination of two normalised vectors)
            cand_score = _conv_max(cand)

            if cand_score < best_score - 1e-12:
                best_score = cand_score
                best_seq = cand.copy()
                cur_seq = cand.copy()
                improved = True
                print(f"Iter {it:5d}   t={t:4.2f}   new best max={best_score:.8f}")
                break

        if not improved:
            # occasional random nudges to escape flat regions
            if random.random() < 0.03:
                i, j = random.sample(range(n), 2)
                delta = cur_seq[i] * random.uniform(0.02, 0.10)
                delta = min(delta, cur_seq[i])
                new_seq = cur_seq.copy()
                new_seq[i] -= delta
                new_seq[j] += delta
                cur_seq = _normalise(new_seq)

    return best_seq, best_score


def _random_initial(n):
    """Create a random Dirichlet vector of length n, normalised to sqrt(2n)."""
    x = np.random.rand(n)
    # ensure non‑zero sum
    if np.allclose(x, 0):

    return _normalise(x)

    # ------------------------------------------------------------------
    it = 0
    while time.time() < deadline:
        it += 1
        # propose a small random transfer of mass
        i, j = random.sample(range(n), 2)
        eps = best_seq[i] * random.uniform(0.02, 0.12)
        eps = min(eps, best_seq[i])          # keep non‑negative
        candidate = best_seq.copy()
        candidate[i] -= eps
        candidate[j] += eps
        candidate = _normalise(candidate)

        cand_score = _conv_max(candidate)
        if cand_score < best_score - 1e-12:
            best_score = cand_score
            best_seq = candidate
            print(f"Hill‑climb {it:6d} – improvement, new max={best_score:.8f}")

    # Convert the final NumPy array to a plain Python list for the caller.
    final_seq = best_seq.tolist()
    final_eval = _score(final_seq)
    print(f"Finished – final evaluation = {final_eval:.8f}")
    return final_seq
