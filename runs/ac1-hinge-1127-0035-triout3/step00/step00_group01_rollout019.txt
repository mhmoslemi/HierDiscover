def fast_metric(seq):
    n = len(seq)
    total = seq.sum()
    if total < 1e-8: return np.inf
    # compute conv using FFT
    size = 1 << ((2*n-1).bit_length())
    a = np.zeros(size, dtype=np.float64)
    a[:n] = seq
    fft_a = np.fft.rfft(a)
    conv = np.fft.irfft(fft_a*fft_a)[:2*n-1]
    max_val = conv.max()
    return 2.0*n*max_val/(total*total)

"""
Search for a non‑negative step‑function that minimizes
    2 * n * max(conv(a,a)) / (sum(a)**2)

The algorithm starts from the analytically motivated
a_i = 1/sqrt(i+1) (which gives a ratio ≈π/2≈1.57) and then
performs a fast simulated‑annealing / hill‑climbing loop.
At each iteration the convolution is obtained with an FFT, the
position of the current maximum is identified and a small amount
of mass is transferred from an index that contributes heavily to that
maximum to an index that contributes little.  Occasionally a random
pairwise mass transfer is tried to escape local plateaus.
The loop runs until the time budget expires and the best sequence
found is returned.
"""

import numpy as np
import time
import random
import math

def _fast_metric(seq: np.ndarray) -> float:
    """Fast evaluation of the objective using FFT convolution."""
    n = seq.shape[0]
    total = seq.sum()
    if total < 1e-8:
        return float('inf')
    # length for FFT: next power of two >= 2*n-1
    size = 1 << ((2 * n - 1).bit_length())
    a = np.zeros(size, dtype=np.float64)
    a[:n] = seq
    fa = np.fft.rfft(a)
    conv = np.fft.irfft(fa * fa)[: 2 * n - 1]
    maxb = conv.max()
    return 2.0 * n * maxb / (total * total)

def _convolution(seq: np.ndarray) -> np.ndarray:
    """Full linear convolution using FFT (real output)."""
    n = seq.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    a = np.zeros(size, dtype=np.float64)
    a[:n] = seq
    fa = np.fft.rfft(a)
    conv = np.fft.irfft(fa * fa)[: 2 * n - 1]
    return conv

def _local_flatten_move(seq: np.ndarray, sigma: float) -> np.ndarray:
    """Transfer a small amount of mass from a high‑contributing index
    to a low‑contributing one with respect to the current maximal shift."""
    n = seq.shape[0]
    conv = _convolution(seq)
    kmax = int(np.argmax(conv))
    # partner index for each i: j = kmax - i
    idx = np.arange(n)
    partner = kmax - idx
    valid = (partner >= 0) & (partner < n)
    contrib = np.zeros_like(seq)
    contrib[valid] = seq[valid] * seq[partner[valid]]
    # pick source (large contribution) and destination (small contribution)
    i_src = int(np.argmax(contrib))
    # destination candidates: indices with low contribution
    low_thresh = np.percentile(contrib, 20)  # keep 20‑percentile as low
    low_candidates = np.where(contrib <= low_thresh)[0]
    if low_candidates.size == 0:
        i_dst = int(np.argmin(contrib))
    else:
        i_dst = int(random.choice(low_candidates.tolist()))
    if i_src == i_dst:
        # no useful move – fall back to a tiny random perturbation
        return seq.copy()
    # amount to move – a fraction of the source amount, capped by sigma
    max_transfer = min(seq[i_src], sigma)
    if max_transfer <= 0.0:
        return seq.copy()
    delta = max_transfer * random.random()
    new_seq = seq.copy()
    new_seq[i_src] -= delta
    new_seq[i_dst] += delta
    # guard against negative due to numerical issues
    new_seq = np.clip(new_seq, 0.0, None)
    return new_seq

def _random_transfer_move(seq: np.ndarray, sigma: float) -> np.ndarray:
    """Transfer a random amount of mass between two randomly chosen positions."""
    n = seq.shape[0]
    i, j = random.sample(range(n), 2)
    if sigma <= 0.0:
        return seq.copy()
    delta = sigma * (random.random() - 0.5)  # can be positive or negative
    new_seq = seq.copy()
    if delta > 0:
        # move from i to j
        delta = min(delta, new_seq[i])
        new_seq[i] -= delta
        new_seq[j] += delta
    else:
        delta = -delta
        # move from j to i
        delta = min(delta, new_seq[j])
        new_seq[j] -= delta
        new_seq[i] += delta
    new_seq = np.clip(new_seq, 0.0, None)
    return new_seq

"""
Search for a non‑negative step function that minimizes

    2 * n * max(conv(a, a)) / (sum(a) ** 2)

The algorithm starts from the analytical shape a_i = 1 / sqrt(i+1)
(which gives a ratio ≈ π/2 ≈ 1.57) and then performs a fast
simulated‑annealing / hill‑climbing loop.  In each iteration the
full convolution is obtained with an FFT, the index of the current
maximum is identified, and a small amount of mass is transferred from
an index that contributes heavily to that maximum to an index that
contributes little.  Random pairwise transfers are also tried to
escape plateaus.  The loop runs until the allotted time expires and the
best sequence found is returned.
"""

import numpy as np
import time
import random
import math


def _fast_metric(seq: np.ndarray) -> float:
    """Fast version of evaluate_sequence using FFT convolution."""
    n = seq.shape[0]
    total = seq.sum()
    if total < 1e-8:
        return float("inf")
    # length for FFT: next power of two >= 2*n-1
    size = 1 << ((2 * n - 1).bit_length())
    a = np.zeros(size, dtype=np.float64)
    a[:n] = seq
    fa = np.fft.rfft(a)
    conv = np.fft.irfft(fa * fa)[: 2 * n - 1]
    maxb = conv.max()
    return 2.0 * n * maxb / (total * total)


def _convolution(seq: np.ndarray) -> np.ndarray:
    """Linear convolution of seq with itself using FFT (real output)."""
    n = seq.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    a = np.zeros(size, dtype=np.float64)
    a[:n] = seq
    fa = np.fft.rfft(a)
    conv = np.fft.irfft(fa * fa)[: 2 * n - 1]
    return conv


def _local_flatten_move(seq: np.ndarray, sigma: float) -> np.ndarray:
    """
    Transfer a small amount of mass from an index that contributes most
    to the current maximal convolution entry to an index that contributes
    little.  The amount transferred is limited by `sigma`.
    """
    n = seq.shape[0]
    conv = _convolution(seq)
    kmax = int(np.argmax(conv))

    idx = np.arange(n)
    partner = kmax - idx
    valid = (partner >= 0) & (partner < n)

    contrib = np.zeros_like(seq)
    contrib[valid] = seq[valid] * seq[partner[valid]]

    # source – index with the largest contribution
    i_src = int(np.argmax(contrib))
    # destination – an index with a low contribution (bottom 20 %)
    low_thr = np.percentile(contrib, 20)
    low_cand = np.where(contrib <= low_thr)[0]
    if low_cand.size == 0:
        i_dst = int(np.argmin(contrib))
    else:
        i_dst = int(random.choice(low_cand.tolist()))

    if i_src == i_dst:
        # nothing useful to move
        return seq.copy()

    max_transfer = min(seq[i_src], sigma)
    if max_transfer <= 0.0:
        return seq.copy()
    delta = max_transfer * random.random()

    new_seq = seq.copy()
    new_seq[i_src] -= delta
    new_seq[i_dst] += delta
    new_seq = np.clip(new_seq, 0.0, None)
    return new_seq


def _random_transfer_move(seq: np.ndarray, sigma: float) -> np.ndarray:
    """
    Transfer a random amount of mass between two randomly chosen positions.
    """
    n = seq.shape[0]
    i, j = random.sample(range(n), 2)
    if sigma <= 0.0:
        return seq.copy()
    delta = sigma * (random.random() - 0.5)   # can be positive or negative
    new_seq = seq.copy()
    if delta > 0:
        # move from i to j
        delta = min(delta, new_seq[i])
        new_seq[i] -= delta
        new_seq[j] += delta
    else:
        delta = -delta
        # move from j to i
        delta = min(delta, new_seq[j])
        new_seq[j] -= delta
        new_seq[i] += delta
    new_seq = np.clip(new_seq, 0.0, None)
    return new_seq


def propose_candidate(seed: int = 42, budget_s: float = 1000.0, **kwargs):
    """
    Return a list of non‑negative heights that (hopefully) beats the
    current bound of 2.0.  The search runs for at most `budget_s` seconds.
    """
    np.random.seed(seed)
    random.seed(seed)
    deadline = time.time() + budget_s - 0.5   # safety margin

    # ----- initial sequence (analytic 1/√i) ---------------------------------
    n = 500                       # moderate length – fast enough for many iterations
    init = 1.0 / np.sqrt(np.arange(1, n + 1, dtype=np.float64))
    current = init.copy()

    # ----- bookkeeping -------------------------------------------------------
    best_seq = current.copy()
    best_score = evaluate_sequence(best_seq.tolist())
    sigma = max(0.01, current.mean() * 0.2)   # typical transfer size
    it = 0
    last_improve = time.time()

    # ----- main optimisation loop -------------------------------------------
    while time.time() < deadline:
        it += 1

        # slowly shrink the transfer size
        if it % 5000 == 0:
            sigma *= 0.95
            sigma = max(sigma, 1e-4)

        # pick a move type
        if random.random() < 0.7:
            cand = _local_flatten_move(current, sigma)
        else:
            cand = _random_transfer_move(current, sigma)

        # cheap metric for acceptance decision
        cand_metric = _fast_metric(cand)
        cur_metric = _fast_metric(current)

        # Metropolis acceptance (temperature decays with remaining time)
        if cand_metric < cur_metric:
            accept = True
        else:
            remaining = deadline - time.time()
            temperature = max(0.001, remaining / budget_s)
            prob = math.exp(-(cand_metric - cur_metric) / temperature)
            accept = random.random() < prob

        if accept:
            current = cand

        # full evaluation only when we seem to have improved globally
        if cand_metric < best_score:
            true_score = evaluate_sequence(cand.tolist())
            if true_score < best_score:
                best_score = true_score
                best_seq = cand.copy()
                last_improve = time.time()
                sigma = max(sigma, best_seq.mean() * 0.1)

        # occasional restart if stuck for a while
        if time.time() - last_improve > 30.0:
            current = 1.0 / np.sqrt(np.arange(1, n + 1, dtype=np.float64))
            sigma = max(0.01, current.mean() * 0.2)
            last_improve = time.time()

        # progress report every so often
        if it % 10000 == 0:
            print(f"Iter {it:>7d} | best_score = {best_score:.6f} | sigma = {sigma:.6f}")

    print(f"Search finished after {it} iterations. Best score = {best_score:.6f}")
    return best_seq.tolist()
