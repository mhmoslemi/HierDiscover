    start = time.time()
    np.random.seed(seed)

    # Determine length n
    if 'height_sequence_1' in globals():
        n = len(height_sequence_1)
    else:
        n = np.random.randint(5000, 10000)
    # Build decreasing linear sequence
    raw = np.arange(n, 0, -1, dtype=float)  # n, n-1, ..., 1
    # Scale to stay within [0, 1000)
    max_allowed = 999.0  # a bit below 1000 to avoid clipping
    scale = max_allowed / raw.max()
    seq = (raw * scale).tolist()

    # Compute a quick initial score for logging
    try:
        init_score = evaluate_sequence(seq)
    except Exception as e:
        init_score = float('inf')
    print(f'Linear decreasing candidate length={n}, scale={scale:.6f}, initial score={init_score:.6f}')

    # Optionally perform a tiny local refinement (few random swaps) within the time budget.
    best_seq = seq
    best_score = init_score
    seq_arr = np.array(seq, dtype=float)

    # Simple hill climbing: pick two indices and transfer a tiny amount.
    # Use very small step because shape already near-optimal.
    while time.time() - start < budget_s - 0.5:
        # propose a random perturbation
        i = np.random.randint(n)
        j = np.random.randint(n)
        if i == j:
            continue
        # amount proportional to average value
        avg = seq_arr.mean()
        # random fraction up to 1% of avg
        delta = avg * (np.random.rand() - 0.5) * 0.02  # between -1% and 1% of avg
        # ensure non‑negative
        if seq_arr[i] - delta < 0:
            continue
        # apply
        new_seq = seq_arr.copy()
        new_seq[i] -= delta
        new_seq[j] += delta
        # optional re‑normalize to keep max <= max_allowed (unlikely needed)
        if new_seq.max() > max_allowed:
            new_seq = new_seq * (max_allowed / new_seq.max())
        try:
            score = evaluate_sequence(new_seq.tolist())
        except Exception:
            continue
        if score < best_score:
            best_score = score
            best_seq = new_seq.tolist()
            seq_arr = new_seq
            print(f'Improved score: {best_score:.6f}')
        # accept non‑improving move with small probability to escape local flatness
        else:
            # small temperature decreasing over time
            elapsed = time.time() - start
            temp = max(0.01, 0.5 * (1 - elapsed / budget_s))
            if np.random.rand() < np.exp(-(score - best_score) / (temp + 1e-12)):
                seq_arr = new_seq

    print('Final best score:', best_score)
    return best_seq

        n = np.random.randint(5000, 10000)

    # ------------------------------------------------------------------
    # 2. Build the decreasing linear vector (raw integers)
    # ------------------------------------------------------------------
    raw = np.arange(n, 0, -1, dtype=float)       # [n, n‑1, …, 1]

    # ------------------------------------------------------------------
    # 3. Scale to respect the evaluator’s upper bound (1000)
    # ------------------------------------------------------------------
    max_allowed = 999.0                         # stay safely below 1000
    scale = max_allowed / raw.max()              # raw.max() == n
    seq = (raw * scale).tolist()                # final sequence (list of floats)

    # ------------------------------------------------------------------
    # 4. Initial evaluation (just for logging)
    # ------------------------------------------------------------------
    try:
        best_score = evaluate_sequence(seq)
    except Exception:  # defensive – should never happen
        best_score = float('inf')
    best_seq = seq.copy()
    print(f"[INIT] length={n}, scale={scale:.5f}, score={best_score:.6f}")

    # ------------------------------------------------------------------
    # 5. Light‑weight local search (random mass transfers) –
    #    this often uncovers tiny improvements without breaking the
    #    favourable structure.
    # ------------------------------------------------------------------
    seq_arr = np.array(seq, dtype=float)

    # parameters for the random walk
    max_iters_without_improve = 1000
    iters_without_improve = 0

    while time.time() - start_time < budget_s - 0.5:
        # pick two distinct indices
        i, j = np.random.randint(n, size=2)
        if i == j:
            continue

        # propose a tiny transfer of mass (up to ≈1 % of the average entry)
        avg = seq_arr.mean()
        delta = avg * (np.random.rand() - 0.5) * 0.02   # ±1 % of avg

        # keep non‑negativity
        if seq_arr[i] - delta < 0:
            continue

        new_seq = seq_arr.copy()
        new_seq[i] -= delta
        new_seq[j] += delta

        # safety: ensure we never exceed the hard cap (unlikely)
        if new_seq.max() > max_allowed:
            new_seq *= max_allowed / new_seq.max()

        try:
            cur_score = evaluate_sequence(new_seq.tolist())
        except Exception:
            continue

        # accept improvement unconditionally
        if cur_score < best_score:
            best_score = cur_score
            best_seq = new_seq.tolist()
            seq_arr = new_seq
            iters_without_improve = 0
            print(f"[UPDATE] better score={best_score:.6f}")
        else:
            # occasional uphill move (simulated‑annealing style) to escape flat zones
            iters_without_improve += 1
            if iters_without_improve > max_iters_without_improve:
                # decay temperature as time passes
                elapsed = time.time() - start_time
                temperature = max(0.001, 0.1 * (1.0 - elapsed / budget_s))
                prob = np.exp(-(cur_score - best_score) / (temperature + 1e-12))
                if np.random.rand() < prob:
                    seq_arr = new_seq

    print(f"[FINISH] best score={best_score:.6f} (length={n})")
    return best_seq
