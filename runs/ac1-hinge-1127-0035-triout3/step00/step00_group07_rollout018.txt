def compute_max_convolution(a):
    n = len(a)
    L = 1
    while L < 2*n:
        L <<= 1
    fft_a = np.fft.rfft(a, n=L)
    fft_b = fft_a * fft_a
    conv = np.fft.irfft(fft_b, n=L)
    conv = conv[:2*n-1]
    # clip negative due to numeric errors:
    conv = np.maximum(conv, 0.0)
    return conv.max()

"""Simulated annealing search for low-evaluation step‑function sequence.

The algorithm starts from a (possibly random) non‑negative vector,
normalises it to the usual `sqrt(2 n)` ℓ₁‑norm and then performs a
simulated‑annealing walk that transfers a small amount of mass between
indices.  The acceptance rule uses the true objective
`2·n·max(a*a)/(sum a)²`.  A directed move targeting the index that
contributes most to the current maximal convolution is used most of
the time, and a cheap linear‑programming step is invoked occasionally
to obtain a strong descent direction.  The search stops after the
provided time budget and returns the best sequence found.
"""

import time
import numpy as np
from scipy import optimize

linprog = optimize.linprog

def _next_pow2(x: int) -> int:
    """Small helper: next power of two ≥ x."""
    p = 1
    while p < x:
        p <<= 1
    return p

def _conv_max(a: np.ndarray, fft_len: int) -> float:
    """Return max of linear convolution a*a using FFT of length fft_len."""
    # real FFT because a is real
    fft_a = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(fft_a * fft_a, n=fft_len)[: 2 * len(a) - 1]
    # eliminate tiny negative artefacts
    conv = np.maximum(conv, 0.0)
    return float(conv.max())

def _score(a: np.ndarray, fft_len: int) -> float:
    """Exact objective value for a (no clipping needed if a∈[0,1000])."""
    n = len(a)
    s = a.sum()
    if s < 1e-12:
        return np.inf
    m = _conv_max(a, fft_len)
    return 2.0 * n * m / (s * s)

def solve_convolution_lp(f_seq, rhs):
    """LP: maximise Σb_j  subject to (f * b)_k ≤ rhs  and b≥0."""
    n = len(f_seq)
    c = -np.ones(n)          # maximise → minimise -Σb
    a_ub = []
    b_ub = []
    # build convolution constraints
    for k in range(2 * n - 1):
        row = np.zeros(n)
        for i in range(n):
            j = k - i
            if 0 <= j < n:
                row[j] = f_seq[i]
        a_ub.append(row)
        b_ub.append(rhs)
    # non‑negativity constraints b ≥ 0  ⇒  -b ≤ 0
    a_ub_nonneg = -np.eye(n)
    b_ub_nonneg = np.zeros(n)
    a_ub = np.vstack([a_ub, a_ub_nonneg])
    b_ub = np.hstack([b_ub, b_ub_nonneg])
    # a tiny time limit keeps the overall search responsive
    res = linprog(c, A_ub=a_ub, b_ub=b_ub, options={'time_limit': 8.0, 'disp': False})
    if res.success:
        return res.x
    return None

invoked to obtain a strong descent direction.  The walk stops after the

import time

from scipy import optimize

linprog = optimize.linprog


def _next_pow2(x: int) -> int:
    """Return the smallest power of two ≥ x."""
    p = 1
    while p < x:
        p <<= 1
    return p


def _conv_max(a: np.ndarray, fft_len: int) -> float:
    """Maximum entry of the linear convolution a*a using an FFT of length fft_len."""
    fft_a = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(fft_a * fft_a, n=fft_len)[: 2 * len(a) - 1]
    conv = np.maximum(conv, 0.0)                     # delete tiny negatives
    return float(conv.max())


def _score(a: np.ndarray, fft_len: int) -> float:
    """Exact objective value: 2·n·max(a*a)/(sum a)²."""
    n = len(a)
    s = a.sum()
    if s < 1e-12:
        return np.inf
    m = _conv_max(a, fft_len)
    return 2.0 * n * m / (s * s)


def solve_convolution_lp(f_seq, rhs):
    """Linear programme: maximise Σb_j  subject to (f*b)_k ≤ rhs and b≥0."""
    n = len(f_seq)
    c = -np.ones(n)                                 # maximise → minimise -Σb
    a_ub = []
    b_ub = []
    for k in range(2 * n - 1):
        row = np.zeros(n)
        for i in range(n):
            j = k - i
            if 0 <= j < n:
                row[j] = f_seq[i]

        b_ub.append(rhs)
    # non‑negativity  –b ≤ 0

    b_ub_nonneg = np.zeros(n)
    a_ub = np.vstack([a_ub, a_ub_nonneg])
    b_ub = np.hstack([b_ub, b_ub_nonneg])

    res = linprog(c, A_ub=a_ub, b_ub=b_ub, options={'time_limit': 8.0, 'disp': False})

    return None
