def propose_candidate(seed=0, budget_s=1000, **kwargs):
    """
    Gradient‑descent (mirror descent) search for a probability vector whose
    self‑convolution is as flat as possible.
    The algorithm works on a probability vector (sum = 1).  The final
    sequence is scaled to the standard normalisation √(2 n) before being
    returned, because the evaluation function is scale‑invariant.
    """
    import numpy as np, time

    np.random.seed(seed)

    # Choose a length: use the length of the known construction if it exists,
    # otherwise fall back to a moderate size.
    n = len(globals().get('height_sequence_1', [])) or 2000

    # start from a slightly perturbed uniform distribution
    c = np.full(n, 1.0 / n) + 0.001 * np.random.rand(n)
    c = c / c.sum()

    best_seq = c.copy()
    best_score = evaluate_sequence(c.tolist())
    print(f"initial score {best_score:.6f}")

    # parameters
    L = 1 << ((2 * n - 1).bit_length())      # power‑of‑two FFT length
    lam = 0.5                                 # soft‑max temperature  (λ)
    eta = 0.1                                 # mirror‑descent step size
    it = 0

    deadline = time.time() + budget_s - 5
    while time.time() < deadline:
        # ----- convolution of the current distribution with itself -----
        c_fft = np.fft.rfft(c, n=L)
        conv = np.fft.irfft(c_fft * c_fft, n=L)[:2 * n - 1]   # p = c*c

        # ----- smooth approximation of the max (soft‑max) -----
        max_val = conv.max()
        # use the max to avoid overflow; the exponent is always ≤ 0
        w = np.exp(lam * (conv - max_val))
        w /= w.sum()                                          # w is a probability on the shifts

        # ----- gradient of the smooth objective (mirror‑descent) -----
        w_fft = np.fft.rfft(w, n=L)
        rev_c_fft = np.fft.rfft(c[::-1], n=L)
        grad_full = np.fft.irfft(w_fft * rev_c_fft, n=L)
        grad = 2.0 * grad_full[n - 1:n - 1 + n]                # length‑n gradient

        # ----- mirror‑descent update on the probability simplex -----
        c = c * np.exp(-eta * grad)        # element‑wise
        s = c.sum()
        if s == 0 or not np.isfinite(s):
            c = np.full(n, 1.0 / n)
        else:
            c /= s

        # ----- evaluate and keep the best seen solution -----
        score = evaluate_sequence(c.tolist())
        if score < best_score:
            best_score = score
            best_seq = c.copy()
            print(f"iter {it:6d}   improved score = {best_score:.6f}")

        # ----- schedule for the hyper‑parameters -----
        lam = min(200.0, lam * 1.002)     # make the soft‑max sharper over time
        eta = max(1e-5, eta * 0.999)      # slowly shrink the step size

        # occasional re‑mix with uniform to escape possible stagnation
        if it % 5000 == 0 and it > 0:
            c = 0.9 * c + 0.1 * (1.0 / n)
            c /= c.sum()

        it += 1

    # Scale the final vector to the conventional normalisation √(2 n)
    scale = np.sqrt(2.0 * n) / best_seq.sum()
    best_seq = (best_seq * scale).tolist()
    print(f"finished after {it} iterations, best score = {best_score:.6f}")
    return best_seq

"""
Propose a step‑function sequence that lowers the evaluation score.

The evaluation is   2·n·max(conv(a,a)) / (sum(a)**2) .
Because the score is scale‑invariant we may work with a probability vector
c = a / sum(a) (∑c = 1).  Then the score is simply 2·n·max_k (c*c)_k.
The task becomes: find a non‑negative vector on the simplex whose
self‑convolution is as flat as possible (i.e. minimise the ℓ∞‑norm of the
convolution).

We perform a mirror‑descent on the simplex using a smooth approximation of
the max: the soft‑max (log‑sum‑exp).  For a temperature λ>0

    Φ_λ(c) = log(∑_k exp( λ·(c*c)_k )),

∇Φ_λ(c) can be written in closed form and is computed by one FFT‑based
convolution.  Updating with c ← c·exp(−η·∇Φ_λ(c)) keeps the iterate in the
simplex; after normalisation we evaluate the true score and keep the best
vector seen.

The algorithm starts from a slightly perturbed uniform distribution,
gradually sharpens λ (making the surrogate closer to the true max) and
slowly reduces the step size η.  Occasionally the iterate is mixed a little
with the uniform distribution to avoid stagnation.  The whole search runs
until the time budget (≈ 1000 s) expires and returns the best sequence
found, scaled to the conventional normalisation √(2 n) (the scaling does
not affect the score).

The method needs only NumPy and the provided `evaluate_sequence` function.
"""

import time
import numpy as np

def _full_convolution(a: np.ndarray, L: int) -> np.ndarray:
    """Return the full linear convolution a*a using FFT of length L."""
    af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(af * af, n=L)[: 2 * a.shape[0] - 1]
    return conv


def _softmax_weights(p: np.ndarray, lam: float) -> np.ndarray:
    """Soft‑max (log‑sum‑exp) weights for the vector p with temperature lam."""
    # p is non‑negative; shift by its maximum for numerical stability.
    m = p.max()
    w = np.exp(lam * (p - m))
    w /= w.sum()
    return w


def _gradient(c: np.ndarray, w: np.ndarray, Lg: int) -> np.ndarray:
    """
    Gradient of the soft‑max surrogate with respect to the probability vector c.
    Computes  grad_i = 2 * Σ_k w_k * c_{k-i}.
    """
    n = c.shape[0]
    # FFT of the weights (length Lg) and of the reversed probability vector.
    wf = np.fft.rfft(w, n=Lg)
    rev_cf = np.fft.rfft(c[::-1], n=Lg)
    # Linear convolution (zero‑padded) of w with reversed c.
    conv_full = np.fft.irfft(wf * rev_cf, n=Lg)
    # Slice the part that corresponds to indices 0 … n‑1.
    grad = 2.0 * conv_full[n - 1 : n - 1 + n]
    return grad


def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs):
    """
    Search for a step‑function (list of non‑negative heights) that minimises
    `evaluate_sequence`.  The algorithm uses mirror‑descent on the simplex with
    a soft‑max approximation of the max‑convolution and returns the best
    sequence discovered before the time budget runs out.
    """
    np.random.seed(seed)

    # ----------------------------------------------------------------------
    # Problem size
    # ----------------------------------------------------------------------
    # Use the length of the known construction if it exists, otherwise a
    # moderate size that allows many iterations.
    n = len(globals().get("height_sequence_1", []))
    if n == 0:
        n = 800  # fallback size; still large enough to get a good bound

    # ----------------------------------------------------------------------
    # Initial probability vector (slightly noisy uniform)
    # ----------------------------------------------------------------------
    c = np.full(n, 1.0 / n) + 0.001 * np.random.rand(n)
    c /= c.sum()

    # Keep the best (lowest) score seen so far.
    best_seq = c.copy()
    best_score = evaluate_sequence(c.tolist())
    print(f"[init] n={n}, initial score={best_score:.6f}")

    # Pre‑compute FFT lengths.
    Lc = 1 << ((2 * n - 1).bit_length())          # for c*c   (≤ 2n‑1)
    Lg = 1 << ((3 * n - 2).bit_length())          # for w * rev(c) (≤ 3n‑2)

    # Hyper‑parameters.
    lam = 0.5          # soft‑max temperature (will be increased)
    eta = 0.08         # mirror‑descent step size (will be slowly decayed)
    it = 0

    deadline = time.time() + budget_s - 4   # a few seconds margin

    while time.time() < deadline:
        # --------------------------------------------------------------
        # 1) Compute the self‑convolution p = c * c
        # --------------------------------------------------------------
        p = _full_convolution(c, Lc)

        # --------------------------------------------------------------
        # 2) Soft‑max weights that approximate the max over p
        # --------------------------------------------------------------
        w = _softmax_weights(p, lam)

        # --------------------------------------------------------------
        # 3) Gradient of the smooth surrogate
        # --------------------------------------------------------------
        grad = _gradient(c, w, Lg)

        # --------------------------------------------------------------
        # 4) Mirror‑descent update on the simplex
        # --------------------------------------------------------------
        c = c * np.exp(-eta * grad)   # element‑wise
        s = c.sum()
        if s == 0 or not np.isfinite(s):
            c = np.full(n, 1.0 / n)
        else:
            c /= s

        # --------------------------------------------------------------
        # 5) Evaluate the true objective and keep the best solution
        # --------------------------------------------------------------
        score = evaluate_sequence(c.tolist())
        if score < best_score:
            best_score = score
            best_seq = c.copy()
            print(f"[iter {it:6d}] improved score = {best_score:.6f}")

        # --------------------------------------------------------------
        # 6) Simple schedules for λ and η
        # --------------------------------------------------------------
        lam = min(200.0, lam * 1.002)          # sharpen the soft‑max
        eta = max(1e-5, eta * 0.999)           # slowly shrink step size

        # Occasionally re‑mix a bit of uniform mass to escape plateaus.
        if it % 4000 == 0 and it > 0:
            c = 0.9 * c + 0.1 * (1.0 / n)
            c /= c.sum()

        it += 1

    # ----------------------------------------------------------------------
    # 7) Rescale to the conventional norm √(2 n) (scale does not affect score)
    # ----------------------------------------------------------------------
    scale = np.sqrt(2.0 * n) / best_seq.sum()
    final_seq = (best_seq * scale).tolist()
    print(f"[done] iterations={it}, best score={best_score:.6f}")
    return final_seq
