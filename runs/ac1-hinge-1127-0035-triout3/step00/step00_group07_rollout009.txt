def neighbor(seq):

    i = np.random.randint(n)
    j = np.random.randint(n)
    if i == j: return seq copy unchanged
    # Transfer epsilon from i to j

    eps = np.random.rand() * min(seq[i], 0.1) # up to 0.1
    new = seq.copy()
    new[i] -= eps
    new[j] += eps
    return new

def propose_candidate(seed=42, budget_s=1000):
    np.random.seed(seed)
    deadline = time.time() + budget_s - 5
    best_seq = None
    best_score = float('inf')
    while time.time() < deadline:
        # choose n random within [200, 800] maybe.
        n = np.random.randint(200, 800)
        # initialize random Dirichlet sequence
        raw = np.random.exponential(scale=1.0, size=n)
        seq = raw / raw.sum()
        # ensure sum >0.01 (always as normalized)
        seq = seq.tolist()
        curr_seq = seq.copy()
        curr_score = evaluate_sequence(curr_seq)
        # set annealing schedule
        max_iter = 5000
        start_time = time.time()
        for it in range(max_iter):
            if time.time() > deadline:
                break
            # temperature schedule
            t = 1.0 - it / max_iter # linearly decrease
            # neighbors
            # pick i with probability proportional to a_i maybe for better moves
            i = np.random.choice(len(curr_seq), p=curr_seq/np.sum(curr_seq))
            j = np.random.randint(len(curr_seq))
            if i == j:
                continue
            eps = np.random.rand() * min(curr_seq[i], 0.1)
            new_seq = curr_seq.copy()
            new_seq[i] -= eps
            new_seq[j] += eps
            new_score = evaluate_sequence(new_seq)

                curr_score = new_score

                    best_score = new_score
                    best_seq = new_seq.copy()
                    print(f"New best {best_score:.6f} at n={n} iter={it}")
        # maybe also try simple scaling of values to reduce zeros?

import numpy as np, time, math

def subgradient_optimize(init_seq, max_iters, deadline):
    n = len(init_seq)
    a = np.array(init_seq, dtype=float)
    # normalize
    if a.sum() == 0:
        a = np.full_like(a, 1.0 / n)
    a /= a.sum()
    best_a = a.copy()
    best_score = 2 * n * np.max(np.convolve(a, a)) / (a.sum()**2)
    for t in range(1, max_iters+1):
        if time.time() > deadline:
            break
        # compute convolution
        b = np.convolve(a, a)
        max_idx = np.argmax(b)
        # compute gradient g
        g = np.zeros_like(a)
        # compute slice
        start = max(0, max_idx - (n-1))
        end = min(n-1, max_idx)
        i_range = np.arange(start, end+1)
        g[i_range] = 2 * a[max_idx - i_range]
        # project to sum zero
        g_mean = g.mean()
        d = -g + g_mean
        # step size schedule
        step = 0.5 / np.sqrt(t)  # step parameter base maybe 0.5
        a_new = a + step * d
        # enforce nonnegative
        a_new = np.maximum(a_new, 0)
        # renorm
        sum_new = a_new.sum()
        if sum_new == 0:
            # reinitialize
            a_new = np.random.rand(n)
            a_new /= a_new.sum()
        else:
            a_new /= sum_new
        a = a_new
        # evaluate
        score = 2 * n * np.max(np.convolve(a, a)) / (a.sum()**2)
        if score < best_score:
            best_score = score
            best_a = a.copy()
    return best_a, best_score

"""
Simple gradient‑based minimisation of the self‑convolution peak.

The evaluation score for a non‑negative sequence a with sum S is
    score = 2·n·max_k (a*a)[k] / S²   (n = len(a))

Because the score is invariant under scaling of a, we normalise
the vector to sum‑one at every step.  The optimisation objective is
the (smoothed) maximum of the convolution.  Using a soft‑max
approximation

    f(a) = (1/γ)·log( Σ_k exp(γ·b_k) ),    b = a*a

gives a smooth gradient that can be written in closed form.
For each iteration we

1. compute the convolution b,
2. form the soft‑max weights p_k,
3. obtain the gradient ∇f via one extra convolution,
4. project the gradient onto the simplex tangent (to keep Σa constant),
5. take a small step, clip negatives, renormalise.

The algorithm is run many times with random restarts and varying
lengths (200 – 800).  The best sequence seen before the time budget
expires is returned.
"""

import time
import numpy as np

def _fast_score(a: np.ndarray) -> float:
    """Return the score 2·n·max(b)/sum(a)² for a (sum should be >0)."""
    n = a.shape[0]
    max_b = np.max(np.convolve(a, a))
    s = a.sum()
    return float(2.0 * n * max_b / (s * s))


def _softmax_gradient(a: np.ndarray, gamma: float) -> np.ndarray:
    """
    Compute the gradient of the soft‑max approximation of max(conv(a,a))
    with respect to a, under the current sum‑one normalisation.

    Returns a vector of length n.
    """
    n = a.shape[0]
    b = np.convolve(a, a)                     # length 2n‑1
    max_b = b.max()
    # soft‑max weights, shifted to avoid overflow
    weights = np.exp(gamma * (b - max_b))
    weights /= weights.sum()                  # p_k, sum to 1

    # gradient = 2 * Σ_k p_k * ∂b_k/∂a_i ;  ∂b_k/∂a_i = 2*a_{k‑i} (when index valid)
    # Use convolution: g_i = 2 * Σ_j a_j * p_{i+j}
    # This is a convolution of a with the reversed weight vector.
    pr = weights[::-1]                        # reversed p
    conv = np.convolve(a, pr)                 # length n + (2n‑1) - 1 = 3n‑2
    start = 2 * n - 2                         # index that yields i+j term
    g = 2.0 * conv[start - np.arange(n)]
    return g


def _project_to_simplex_tangent(v: np.ndarray) -> np.ndarray:
    """Project vector v onto the hyperplane {x : Σx = 0}."""
    return v - v.mean()


def _optimize_one_restart(n: int, max_iters: int, deadline: float) -> tuple:
    """
    Run the gradient descent for a fixed length n.
    Returns (best_sequence, best_score) found during this run.
    """
    # random start (Dirichlet(1) -> uniform over the simplex)
    a = np.random.rand(n)
    a /= a.sum()

    best_a = a.copy()
    best_score = _fast_score(a)

    # start with a mild smoothing, ramp up to a sharper approximation
    gamma = 1.0
    gamma_inc = (30.0 - 1.0) / max_iters  # final gamma ≈30

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        # 1) current convolution and its max
        b = np.convolve(a, a)
        cur_max = b.max()

        # 2) gradient of the smoothed max
        g = _softmax_gradient(a, gamma)

        # 3) make it tangent to the simplex
        d = -_project_to_simplex_tangent(g)

        # 4) simple diminishing step size
        step = 0.6 / np.sqrt(it)

        a_new = a + step * d
        a_new = np.maximum(a_new, 0.0)          # enforce non‑negativity
        s = a_new.sum()
        if s == 0.0:                           # very unlikely, but guard against it
            a_new = np.random.rand(n)
            a_new /= a_new.sum()
            s = 1.0
        a_new /= s                              # renormalise to sum‑one

        # 5) evaluate
        new_score = _fast_score(a_new)

        # Accept the move if it improves the true max‑score; otherwise keep current a
        if new_score < best_score:
            best_score = new_score
            best_a = a_new.copy()
        if new_score <= _fast_score(a) or np.random.rand() < 0.001:
            a = a_new

        # increase smoothing parameter
        gamma = min(gamma + gamma_inc, 30.0)

        # occasional cheap progress report (optional)
        if it % 1000 == 0:
            print(f"   n={n} it={it} best_score={best_score:.6f} gamma={gamma:.2f}")

    return best_a, best_score


def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs) -> list:
    """
    Search for a non‑negative step‑function that minimises
    evaluate_sequence.  The function respects the 1000 s time budget
    and returns the best sequence discovered.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 5.0  # keep a safety margin

    best_seq = None
    best_score = float("inf")

    # main loop: random restarts until time runs out
    while time.time() < deadline:
        # Choose a length roughly in the range that gives a good trade‑off
        n = np.random.randint(200, 800)

        # Number of inner iterations is chosen adaptively based on remaining time
        # (at most a few thousand steps per restart)
        remaining = deadline - time.time()
        max_iters = int(min(5000, max(1000, remaining * 5)))  # crude heuristic

        seq, score = _optimize_one_restart(n, max_iters, deadline)

        if score < best_score:
            best_score = score
            best_seq = seq.tolist()
            print(f"New global best: score={best_score:.6f} (n={n})")

        # Early exit if we have already beaten the target
        if best_score < 1.503:
            break

    # sanity fallback – a uniform vector always yields score 2.0
    if best_seq is None:
        best_seq = [1.0]

    print(f"Finished search – best score found: {best_score:.6f}")
    return best_seq
