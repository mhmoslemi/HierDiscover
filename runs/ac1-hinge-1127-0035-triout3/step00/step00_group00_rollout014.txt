    size = 1 << ((2*n - 1).bit_length())

def hill_climb(a, score, max_iter):
    n = len(a)
    size = 1 << ((2*n-1).bit_length())
    for it in range(max_iter):
        # pick random pair
        i, j = np.random.randint(n, size=2)
        if i == j: continue
        # define delta as fraction of a[i]
        delta = np.random.rand() * 0.1 * a[i]  # up to 10% of a[i]
        if delta <=0: continue

        a_new[i] -= delta
        a_new[j] += delta

        # compute new score
        fa = np.fft.rfft(a_new, n=size)
        conv = np.fft.irfft(fa * fa, n=size)[:2*n-1]

        new_score = 2.0 * n * maxb / (np.sum(a_new) **2)

"""Search for a low-evaluation step function using parametric families and hill‑climbing.
The algorithm samples many candidate shapes (edge‑heavy, power, exponential, etc.),
evaluates them via a fast FFT‑based convolution, keeps the best,
and refines promising candidates with a simple mass‑transfer local search.
"""

import time
import random
import math
import numpy as np

def _fast_evaluation(arr):
    """Return the evaluation value of `arr` using FFT (no scaling)."""
    n = len(arr)
    if n == 0:
        return float('inf')
    # FFT size to cover full convolution
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(arr, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[:2 * n - 1]
    maxb = np.max(conv)
    s = np.sum(arr)
    if s < 1e-12:
        return float('inf')
    return 2.0 * n * maxb / (s * s)

def _hill_climb(seq, cur_score, steps=300):
    """
    Simple local search: repeatedly move a small amount of mass from one index to another.
    Accept only moves that improve the score.
    """
    n = len(seq)
    size = 1 << ((2 * n - 1).bit_length())
    a = seq.copy()
    best = cur_score
    for _ in range(steps):
        i, j = np.random.randint(n, size=2)
        if i == j:
            continue
        # move a fraction of the source entry
        delta = np.random.rand() * 0.04 * a[i]   # up to 4 % of a[i]
        if delta <= 0:
            continue
        a_new = a.copy()
        a_new[i] -= delta
        a_new[j] += delta
        # recompute score
        fa = np.fft.rfft(a_new, n=size)
        conv = np.fft.irfft(fa * fa, n=size)[:2 * n - 1]
        maxb = np.max(conv)
        new_score = 2.0 * n * maxb / (np.sum(a_new) ** 2)
        if new_score < best:
            a = a_new
            best = new_score
    return a, best

    # ----------------------------------------------------------------------

    best_score = float('inf')

    try:
        seq0 = list(height_sequence_1)               # provided by the framework
        s0 = evaluate_sequence(seq0)
        if s0 < best_score:
            best_seq = seq0
            best_score = s0
            print(f"Initial construction score {best_score:.6f}")
    except Exception:
        pass

    # ----- 2.  A tiny 2‑element improvement (x,2x) beats the constant case -----
    seq_small = [1.0, 2.0]                           # sum = 3 → evaluation = 16/9 ≈ 1.777…
    s_small = evaluate_sequence(seq_small)
    if s_small < best_score:
        best_seq = seq_small
        best_score = s_small
        print(f"2‑element starter score {best_score:.6f}")

    # ----------------------------------------------------------------------
    # Helper data for the stochastic search
    # ----------------------------------------------------------------------
    lengths = [2, 3, 5, 10, 20, 50, 100, 200, 500, 1000, 2000, 4000]
    shape_kinds = ['edge', 'power', 'exp']
    # ----------------------------------------------------------------------
    while time.time() < deadline:
        # pick a random length and a random parametric shape
        n = random.choice(lengths)
        kind = random.choice(shape_kinds)
        p = random.uniform(-3.0, 3.0)               # shape exponent

        idx = np.arange(n, dtype=np.float64)

        if kind == 'edge':
            # values are large near the ends, small near the centre.
            # (i+½)*(n‑i‑½) is the distance product; raising it to –p makes the ends heavy when p>0.
            base = (idx + 0.5) * (n - idx - 0.5) + 1e-8
            a = base ** (-p)
        elif kind == 'power':
            # simple power law a_i ∝ (i+1)^p
            a = (idx + 1) ** p
        else:  # 'exp'
            a = np.exp(-p * idx)

        # numerical safety – force non‑negativity and re‑normalise.
        a = np.maximum(a, 0.0)
        if a.sum() == 0.0:
            a = np.ones_like(a)
        a /= a.sum()                # L1‑normalisation (scale does not matter for the objective)

        # fast evaluation of the candidate
        score = _fast_evaluation(a)
        if score < best_score:
            best_score = score
            best_seq = a.tolist()
            print(f"New best {best_score:.6f} – n={n}, kind={kind}, p={p:.3f}")

        # If the candidate looks promising, run a short hill‑climb.
        # The threshold (1.9) is arbitrary – it prevents spending time on clearly bad points.
        if score < 1.9 and score < best_score * 1.02:
            a_refined, ref_score = _hill_climb(a, score, steps=600)
            if ref_score < best_score:
                best_score = ref_score
                best_seq = a_refined.tolist()
                print(f"Refined best {best_score:.6f} – n={n}")

    # ----------------------------------------------------------------------
    # Fallback – should never happen, but guarantees a valid return value.
    # ----------------------------------------------------------------------
    if best_seq is None:
        best_seq = [1.0] * 10

    return best_seq
