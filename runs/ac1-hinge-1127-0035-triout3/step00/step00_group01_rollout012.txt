"""
Proposes a candidate step-function sequence using a hybrid of
linear‑programming based improvement (the SOTA direction) and a
simple stochastic gradient descent / random perturbation search.
The algorithm runs for the allocated time budget, keeps the best
sequence seen and returns it.
"""
import time
import numpy as np
from scipy import optimize

linprog = optimize.linprog

def solve_convolution_lp(f_seq, rhs):
    """Linear program: maximize sum(g) s.t. conv(f_seq, g) <= rhs, g>=0."""
    n = len(f_seq)
    # coefficients for objective (max sum => minimize -sum)
    c = -np.ones(n)
    # Build upper bound constraints for each convolution index
    rows = []
    for k in range(2 * n - 1):
        row = np.zeros(n)
        # g_j appears in conv at index k with coefficient f_{k-j}
        # j runs over 0..n-1, and we need k-j in [0,n-1]
        # Equivalent: i = k - j -> j = k - i
        # Loop i
        i_min = max(0, k - (n - 1))
        i_max = min(n - 1, k)
        # i runs from i_min to i_max inclusive, then j = k - i
        # Set row[j] = f_i
        for i in range(i_min, i_max + 1):
            j = k - i
            row[j] = f_seq[i]
        rows.append(row)
    # Non‑negativity constraints g >= 0 are encoded as -I * g <= 0
    a_ub = np.vstack([rows, -np.eye(n)])
    b_ub = np.concatenate([np.full(len(rows), rhs), np.zeros(n)])
    # Solve with HiGHS (fast and reliable)
    res = linprog(c, A_ub=a_ub, b_ub=b_ub, method='highs', options={'time_limit': 5.0})
    if res.success:
        return np.maximum(res.x, 0.0)
    return None

def get_direction(f):
    """Return a new sequence that is a small convex combination of f and a
    better direction g found by the LP.  If LP fails None is returned."""
    n = len(f)
    s = np.sum(f)
    if s < 1e-12:
        return None
    # Normalise to the scaling used by the original SOTA code
    f_norm = f * np.sqrt(2 * n) / s
    rhs = np.max(np.convolve(f_norm, f_norm))
    g = solve_convolution_lp(f_norm, rhs)
    if g is None:
        return None
    # Rescale g back to the same normalisation as f
    g_sum = np.sum(g)
    if g_sum < 1e-12:
        return None
    g_norm = g * np.sqrt(2 * n) / g_sum
    # Choose a step size that actually improves the objective –
    # we try a small set of candidates and pick the best.
    best_seq = None
    best_score = np.inf
    for t in (0.01, 0.05, 0.1, 0.2, 0.3, 0.5):
        cand = (1 - t) * f + t * g_norm
        score = evaluate_sequence(cand.tolist())
        if score < best_score:
            best_score = score
            best_seq = cand
    return best_seq if best_seq is not None else None

def random_perturb(seq, rng, sigma=0.02):
    """Add a small zero‑mean Gaussian perturbation, re‑normalise."""
    n = len(seq)
    delta = rng.normal(scale=sigma, size=n)
    new = np.maximum(seq + delta, 0.0)
    if new.sum() == 0:
        # fallback to uniform
        new = np.full(n, 1.0 / n)
    else:
        new = new / new.sum()
    return new

def search_length(n, rng, deadline):
    """Run the hybrid optimisation for a fixed length."""
    # start from a slightly perturbed uniform distribution
    cur = np.full(n, 1.0 / n)
    cur = random_perturb(cur, rng, sigma=0.01)
    cur_score = evaluate_sequence(cur.tolist())
    best_seq = cur.copy()
    best_score = cur_score
    iteration = 0
    while time.time() < deadline:
        iteration += 1
        # first try the LP based direction
        cand = get_direction(cur)
        if cand is not None:
            cand_score = evaluate_sequence(cand.tolist())
            if cand_score < cur_score:
                cur, cur_score = cand, cand_score
        else:
            # fallback: random perturbation
            cand = random_perturb(cur, rng, sigma=0.02)
            cand_score = evaluate_sequence(cand.tolist())
            if cand_score < cur_score:
                cur, cur_score = cand, cand_score
        # keep global best
        if cur_score < best_score:
            best_score = cur_score
            best_seq = cur.copy()
            print(f"Length {n}: new best score {best_score:.6f} after {iteration} iterations")
        # occasional larger random jumps to avoid stagnation
        if iteration % 50 == 0:
            jump = random_perturb(best_seq, rng, sigma=0.1)
            jump_score = evaluate_sequence(jump.tolist())
            if jump_score < best_score:
                best_seq = jump
                best_score = jump_score
                cur = jump
                cur_score = jump_score
                print(f"Length {n}: large jump gave new best {best_score:.6f}")
    return best_seq, best_score

def propose_candidate(seed=42, budget_s=1000, **kwargs):
    """
    Search for a non‑negative height sequence that minimises
    ``evaluate_sequence``.  The routine runs for ``budget_s`` seconds,
    repeatedly optimising sequences of a few different lengths using
    a combination of LP‑based steps (the SOTA direction) and stochastic
    perturbations.  The best sequence found is returned as a plain
    Python list.
    """
    rng = np.random.default_rng(seed)
    start = time.time()
    # we will try a handful of lengths; the total budget is split evenly.
    lengths = [200, 400, 800, 1600]
    time_per_len = budget_s * 0.22  # a little less than a quarter per length
    best_overall_seq = None
    best_overall_score = np.inf

    for n in lengths:
        remaining = budget_s - (time.time() - start)
        if remaining < 5:
            break
        this_deadline = time.time() + min(time_per_len, remaining - 5)
        print(f"Starting optimisation for length {n} (deadline {this_deadline - start:.1f}s)")
        seq, score = search_length(n, rng, this_deadline)
        if score < best_overall_score:
            best_overall_score = score
            best_overall_seq = seq
            print(f"*** Global best updated: score {best_overall_score:.6f} with length {n}")
        # If we already beat the target we can stop early
        if best_overall_score < 1.503:
            break

    # Fallback: if nothing was found (should not happen) use a uniform sequence
    if best_overall_seq is None:
        n = 200
        best_overall_seq = np.full(n, 1.0 / n)
        best_overall_score = evaluate_sequence(best_overall_seq.tolist())
        print("Fallback uniform sequence used.")

    print(f"Finished search. Best score {best_overall_score:.6f} with length {len(best_overall_seq)}")
    # convert to plain list of floats as required by the caller
    return best_overall_seq.tolist()

"""
Propose a step‑function (non‑negative sequence) that minimizes
``evaluate_sequence``.  The routine works inside the given time budget
(≈ 1000 s) and combines three cheap optimisation ideas:

1. **Uniform start** – a flat sequence (the SOTA baseline).
2. **Simple hill‑climbing** – repeatedly move a small amount of mass
   from one position to another while keeping the total sum fixed.
3. **Differential Evolution** – a small population stochastic optimiser
   (SciPy) that refines the best candidate found by the hill‑climber.

Both inner loops respect the non‑negativity constraints and normalise
the vector before evaluation (the objective is scale‑invariant, so the
normalisation does not change the score).  The algorithm tries several
lengths (200, 400, 800) and keeps the overall best sequence.  If a
candidate reaches the target score (≈ 1.503) the search stops early.
All helper functions are top‑level; no closures, lambda’s or I/O are
used.
"""
import time
import numpy as np
from scipy import optimize

# ----------------------------------------------------------------------
# Helper utilities
# ----------------------------------------------------------------------


def _normalise(v: np.ndarray) -> np.ndarray:
    """Return a non‑negative vector with sum = 1 (scale invariant)."""
    total = v.sum()
    if total <= 0.0:
        # fallback to a uniform distribution
        return np.full_like(v, 1.0 / v.size)
    w = v / total
    # Clamp tiny negatives caused by rounding (should not happen)
    w = np.maximum(w, 0.0)
    # Renormalise once more to be safe
    return w / w.sum()


def _score(seq: np.ndarray) -> float:
    """Thin wrapper around the provided evaluate_sequence."""
    # evaluate_sequence expects a plain list of Python floats
    return evaluate_sequence(seq.tolist())


# ----------------------------------------------------------------------
# Hill‑climbing on the simplex
# ----------------------------------------------------------------------


def _hill_climb(initial: np.ndarray,
                budget_s: float,
                target: float,
                seed: int = 0) -> (np.ndarray, float):
    """
    Simple stochastic hill‑climbing.  ``initial`` must already be normalised.
    Returns the best sequence found and its score.
    """
    rng = np.random.default_rng(seed)
    n = initial.size
    cur = initial.copy()
    cur_score = _score(cur)

    best_seq = cur.copy()
    best_score = cur_score

    start = time.time()
    iter_cnt = 0
    # parameters that work well for the modest lengths we try
    max_iter = 200_000
    while time.time() - start < budget_s and iter_cnt < max_iter:
        iter_cnt += 1

        # pick two distinct positions and move a small amount of mass
        i, j = rng.choice(n, size=2, replace=False)
        # amount we are allowed to take from i (keep it non‑negative)
        max_take = cur[i]
        if max_take <= 0.0:
            continue
        delta = rng.uniform(0.0, min(max_take, 0.01))
        if delta == 0.0:
            continue

        cand = cur.copy()
        cand[i] -= delta
        cand[j] += delta

        # re‑normalise (numerical drift) and evaluate
        cand = _normalise(cand)
        cand_score = _score(cand)

        # accept only improving moves (strictly downhill)
        if cand_score < cur_score:
            cur, cur_score = cand, cand_score
            if cand_score < best_score:
                best_seq, best_score = cand.copy(), cand_score
                print(f"  hill‑climb iter {iter_cnt:6d} – new best {best_score:.6f}")

                # stop early if we have already beaten the target
                if best_score < target:
                    break
    return best_seq, best_score


# ----------------------------------------------------------------------
# Differential Evolution (global optimiser)
# ----------------------------------------------------------------------


def _de_optimize(n: int,
                 budget_s: float,
                 target: float,
                 seed: int = 0) -> (np.ndarray, float):
    """
    Run SciPy's differential evolution on a normalised simplex.
    The objective normalises the candidate before calling evaluate_sequence.
    """
    rng = np.random.default_rng(seed)
    # Bounds are simply non‑negative – we keep them wide and let the
    # normalisation handle the scale.
    bounds = [(0.0, 1.0) for _ in range(n)]

    # Estimate how many evaluations we can afford.
    # One DE generation uses popsize * (n+1) evaluations.
    # We aim for roughly 10 000 evaluations which is well within the time budget.
    max_evals = 12_000
    max_iter = max(1, max_evals // (15 * (n + 1)))  # default popsize=15

    def _obj(x):
        # Normalise to sum = 1 before evaluation (scale‑invariant)
        w = _normalise(np.array(x, dtype=float))
        return _score(w)

    result = optimize.differential_evolution(
        _obj,
        bounds,
        strategy="best1bin",
        maxiter=max_iter,
        popsize=15,
        tol=1e-6,
        mutation=(0.5, 1.0),
        recombination=0.7,
        seed=seed,
        polish=True,
        init="latinhypercube",
        updating="deferred",
        workers=1,
    )

    best_vec = _normalise(np.array(result.x, dtype=float))
    best_val = _score(best_vec)
    return best_vec, best_val


# ----------------------------------------------------------------------
# Main entry point required by the grader
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a sequence of non‑negative heights that minimises the
    provided ``evaluate_sequence`` function.  The routine is limited
    to ``budget_s`` seconds; if the time runs out the best sequence found
    so far is returned.

    The search explores a few sequence lengths (200, 400, 800) and,
    for each length, runs a cheap hill‑climber followed by a short
    differential‑evolution refinement.  The overall best sequence is
    reported.
    """
    # Target score that, once reached, stops the whole search early.
    TARGET_SCORE = 1.503

    rng = np.random.default_rng(seed)
    overall_best_seq = None
    overall_best_score = np.inf

    start_time = time.time()
    # we allocate roughly equal slices of the total budget to each length
    lengths = [200, 400, 800]
    time_per_len = (budget_s - 1.0) / len(lengths)  # leave a second spare

    for n in lengths:
        if time.time() - start_time >= budget_s - 0.5:
            break  # no time left for another length

        print(f"\n=== Optimising length {n} ===")
        # ------------------------------------------------------------------
        # 1) start from the uniform vector (the SOTA baseline)
        # ------------------------------------------------------------------
        uniform = np.full(n, 1.0 / n)
        # add a tiny random perturbation to break symmetry
        uniform = uniform + rng.normal(scale=0.001, size=n)
        uniform = _normalise(uniform)

        # ------------------------------------------------------------------
        # 2) local hill‑climbing
        # ------------------------------------------------------------------
        hc_budget = time_per_len * 0.6
        hc_seq, hc_score = _hill_climb(uniform, hc_budget, TARGET_SCORE,
                                       seed=rng.integers(1_000_000))
        print(f"  hill‑climb best score for n={n}: {hc_score:.6f}")

        # ------------------------------------------------------------------
        # 3) differential evolution refinement (global search)
        # ------------------------------------------------------------------
        de_budget = time_per_len * 0.35
        # make sure we have a little time left for the next length
        de_seq, de_score = _de_optimize(n, de_budget, TARGET_SCORE,
                                        seed=rng.integers(1_000_000))
        print(f"  DE best score for n={n}: {de_score:.6f}")

        # pick the best of the two local candidates
        if de_score < hc_score:
            cand_seq, cand_score = de_seq, de_score
        else:
            cand_seq, cand_score = hc_seq, hc_score

        # keep the global best
        if cand_score < overall_best_score:
            overall_best_score = cand_score
            overall_best_seq = cand_seq.copy()
            print(f"*** New global best: score {overall_best_score:.6f} (len={n})")

        # stop early if we already beat the target
        if overall_best_score < TARGET_SCORE:
            print("Target reached – stopping early.")
            break

    # ----------------------------------------------------------------------
    # Fallback (should never be needed)
    # ----------------------------------------------------------------------
    if overall_best_seq is None:
        n = 200
        overall_best_seq = np.full(n, 1.0 / n)
        overall_best_score = _score(overall_best_seq)
        print("Fallback to uniform sequence.")

    print(f"\nFinished search – best score {overall_best_score:.6f} "
          f"with length {len(overall_best_seq)}")
    return overall_best_seq.tolist()
