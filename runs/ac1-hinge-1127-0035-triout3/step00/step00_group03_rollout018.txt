def gradient_step(seq, step):
    n=len(seq)
    b=np.convolve(seq, seq)
    M=b.max()
    active = np.where(np.abs(b-M) < 1e-12)[0]  # indices
    grad=np.zeros(n)
    for k in active:
        for i in range(n):
            j=k-i
            if 0<=j<n:
                grad[i] += 2*seq[j]
    # apply step
    new_seq = seq - step*grad
    # clip nonnegative
    new_seq = np.maximum(new_seq,0)
    # renormalize to target sum
    target_sum = np.sqrt(2*len(seq))
    s = new_seq.sum()
    if s>0:
        new_seq *= target_sum/s
    else:
        # fallback: uniform
        new_seq = np.full(n, target_sum/n)
    # return new_seq
```

def proj_simplex(v, z=1):
    # ensures sum = z, v >=0.
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - z
    ind = np.arange(len(v)) + 1
    cond = u - cssv / ind > 0
    rho = ind[cond].max()
    theta = cssv[cond].max() / rho
    w = np.maximum(v - theta, 0)
    return w
```

    best_score = evaluate_sequence(best_a.tolist())  # Should be 2

    # Keep current candidate
    curr_a = best_a.copy()
    curr_score = best_score

    T_start = 1.0
    T_end = 1e-4
    start_time = time.time()
    deadline = start_time + budget_s - 5.0

    # Helper functions
    def normalize(a):
        s = a.sum()
        if s <= 0:
            # return uniform
            return np.full_like(a, target_sum / len(a))
        a = a * (target_sum / s)
        # Clip near-zero negative due to rounding
        a = np.maximum(a, 0.0)
        return a

    def perturb(a):
        # Choose two indices i,j and shift delta
        i, j = np.random.randint(len(a), size=2)
        if i == j:
            return a

        delta = (np.random.randn() * 0.1) * a[i]  # modify scaling
        # ensure not exceed a[i] (can't go negative)
        delta = max(-a[i], delta)  # keep a[i] >=0 after removal
        a[i] -= delta
        a[j] += delta
        # optional small noise addition to random other indices
        return a

    while time.time() < deadline:

"""
Hybrid LP‑guided search for a step‑function with a low
2 * n * max(conv(a,a)) / (sum(a)**2) score.

The algorithm starts from a uniform sequence (which yields a score of 2)
and repeatedly performs two kinds of moves:

1. LP move   – solve a linear program that maximises the total mass of a
   new candidate while keeping its convolution with the current sequence
   bounded by the current maximum.  The new candidate is mixed with the
   current one using an adaptive step size `t`.

2. Random perturbation – occasionally a small amount of mass is transferred
   between two random positions.  This prevents the iterates from getting
   stuck at a local optimum.

Both moves are accepted if they improve the evaluation score; otherwise
they are discarded.  The best sequence seen during the allotted time is
returned.

The implementation avoids closures, uses only top‑level functions and
relies on `numpy` and `scipy.optimize.linprog` (highs solver) for the LP.
"""

import time
import math
import random
import numpy as np
from scipy import optimize

# ----------------------------------------------------------------------
# Helper utilities
# ----------------------------------------------------------------------


def _target_sum(n: int) -> float:
    """Desired L1‑norm after normalisation (invariant for the objective)."""
    return math.sqrt(2.0 * n)


def _project_onto_simplex(v: np.ndarray, z: float) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x>=0, sum(x)=z}."""
    # algorithm from
    #   Wang, Weiran & Carreira‑Perpiñán, Miguel Á. (2013). "Projection onto the probability simplex"
    #   (https://arxiv.org/abs/1309.1541)
    if z < 0:
        raise ValueError("simplex sum must be non‑negative")
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - z
    ind = np.arange(len(v)) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all entries zero – return uniform
        return np.full_like(v, z / len(v))
    rho = ind[cond].max()
    theta = cssv[cond].max() / rho
    w = np.maximum(v - theta, 0.0)
    return w


def _make_uniform(n: int) -> np.ndarray:
    """Uniform non‑negative sequence with the required L1‑norm."""
    s = _target_sum(n)
    return np.full(n, s / n)


def _make_random(n: int, rng: np.random.Generator) -> np.ndarray:
    """Random Dirichlet‑distributed vector (all entries >=0, sum = target)."""
    # use exponential draws -> Dirichlet
    x = rng.exponential(scale=1.0, size=n)
    s = _target_sum(n)
    return x * (s / x.sum())


def _convolve(a: np.ndarray) -> np.ndarray:
    """Linear convolution a*a (full length)."""
    return np.convolve(a, a)


def _max_convolution(a: np.ndarray) -> float:
    """Maximum entry of the self‑convolution."""
    return float(np.max(_convolve(a)))


def _evaluate(a: np.ndarray) -> float:
    """Thin wrapper around the provided evaluate_sequence."""
    # evaluate_sequence expects a Python list of floats
    return evaluate_sequence(a.tolist())


# ----------------------------------------------------------------------
# Linear‑programming move
# ----------------------------------------------------------------------


def _solve_convolution_lp(f: np.ndarray, rhs: float) -> np.ndarray | None:
    """
    Solve:
        maximise   sum(b)
        subject to conv(f,b) <= rhs   (for every shift)
                    b >= 0
    Returns the optimal b (numpy array) or None on failure / timeout.
    """
    n = f.shape[0]

    # linear objective: maximise sum(b)  -> minimise -sum(b)
    c = -np.ones(n)

    # inequality matrix A_ub * b <= b_ub
    rows = []
    ub = []

    # convolution constraints
    for k in range(2 * n - 1):
        row = np.zeros(n)
        # (f * b)_k = sum_i f[i] * b[k-i]   with 0 <= k-i < n
        i_min = max(0, k - (n - 1))
        i_max = min(k, n - 1)
        # fill the appropriate columns
        for i in range(i_min, i_max + 1):
            j = k - i
            row[j] = f[i]
        rows.append(row)
        ub.append(rhs)

    # non‑negativity constraints  -b <= 0
    rows.append(-np.eye(n))
    ub.append(np.zeros(n))

    A_ub = np.vstack(rows)
    b_ub = np.hstack(ub)

    # Highs solver is fast and respects a time limit
    try:
        res = optimize.linprog(
            c,
            A_ub=A_ub,
            b_ub=b_ub,
            method="highs",
            options={"time_limit": 5.0, "disp": False},
        )
    except Exception:  # pragma: no cover – safety net
        return None

    if res.success:
        # the LP returns a vector that already satisfies sum(b) == target_sum
        # (because of the convolution bound).  We still project to the simplex
        # to guard against tiny numerical violations.
        b = _project_onto_simplex(res.x, _target_sum(n))
        return b
    return None


def _lp_step(seq: np.ndarray, step: float) -> np.ndarray | None:
    """
    Perform one LP‑guided step.
    Returns the new sequence (already normalised) or None if the LP failed.
    """
    n = seq.shape[0]
    # current bound
    cur_max = _max_convolution(seq)

    # solve LP to get a feasible direction with maximum total mass
    g = _solve_convolution_lp(seq, cur_max)
    if g is None:
        return None

    # convex combination
    new_seq = (1.0 - step) * seq + step * g
    # projection enforces exact L1‑norm and non‑negativity
    new_seq = _project_onto_simplex(new_seq, _target_sum(n))
    return new_seq


# ----------------------------------------------------------------------
# Random perturbation move
# ----------------------------------------------------------------------


def _perturb(seq: np.ndarray, rng: np.random.Generator, magnitude: float = 0.02) -> np.ndarray:
    """
    Transfer a small amount of mass between two random positions.
    `magnitude` is the maximal fraction of the source entry that can be moved.
    The result is projected back onto the simplex.
    """
    n = seq.shape[0]
    i, j = rng.integers(0, n, size=2)
    if i == j:
        return seq
    # amount to move (cannot exceed source value)
    delta = magnitude * rng.random() * seq[i]
    new_seq = seq.copy()
    new_seq[i] -= delta
    new_seq[j] += delta
    # tiny numerical drift – re‑project to be safe
    return _project_onto_simplex(new_seq, _target_sum(n))


# ----------------------------------------------------------------------
# Main entry point required by the platform
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Search for a sequence of non‑negative heights that minimises the
    provided evaluation function.

    Parameters
    ----------
    seed : int
        Random seed for reproducibility.
    budget_s : int
        Time budget in seconds for the whole search.
    kwargs : dict
        Ignored (kept for compatibility).

    Returns
    -------
    list[float]
        The best sequence found (as a plain Python list).
    """
    rng = np.random.default_rng(seed)

    # --------------------------------------------------------------
    # Choose a problem size.  A size around a few hundred gives a
    # good trade‑off between expressiveness and LP solving cost.
    # --------------------------------------------------------------
    n = 300                       # feel free to adjust if needed
    target = _target_sum(n)

    # Initialise with a uniform vector – score = 2.0
    current = _make_uniform(n)
    best_seq = current.copy()
    best_score = _evaluate(best_seq)

    # ------------------------------------------------------------------
    # Adaptive schedule for the LP mixing coefficient `t`.
    # Start aggressive, then taper off.
    # ------------------------------------------------------------------
    t_initial = 0.30
    t_final = 0.005

    start_time = time.time()
    deadline = start_time + budget_s - 5.0   # a small safety margin

    iteration = 0
    last_improve_iter = 0
    max_no_improve = 150                     # after this many iterations we re‑seed

    while time.time() < deadline:
        iteration += 1

        # Linear‑programming move
        # Interpolate step size between initial and final value
        t = t_initial + (t_final - t_initial) * (iteration / 5000.0)
        t = max(t_final, min(t_initial, t))

        new_seq = _lp_step(current, t)
        if new_seq is None:                 # LP failed – fall back to a perturbation
            new_seq = _perturb(current, rng)

        new_score = _evaluate(new_seq)

        # Accept if the score improves
        if new_score < best_score:
            best_score = new_score
            best_seq = new_seq.copy()
            current = new_seq.copy()
            last_improve_iter = iteration
            print(f"[{time.time() - start_time:6.1f}s] New best: {best_score:.8f}")

        else:
            # Occasionally accept a worse move to escape plateaus
            if rng.random() < 0.003:
                current = new_seq.copy()

        # If we have gone a long time without improvement, re‑initialise
        if iteration - last_improve_iter > max_no_improve:
            current = _make_random(n, rng)
            last_improve_iter = iteration
            print(f"[{time.time() - start_time:6.1f}s] Restarted after stagnation")

        # Small random perturbations (always cheap) to enrich exploration
        if rng.random() < 0.05:
            pert = _perturb(current, rng, magnitude=0.01)
            pert_score = _evaluate(pert)
            if pert_score < best_score:
                best_score = pert_score
                best_seq = pert.copy()
                current = pert.copy()
                last_improve_iter = iteration
                print(f"[{time.time() - start_time:6.1f}s] Improved via perturbation: {best_score:.8f}")

    # Final sanity projection – guarantees all constraints are satisfied
    best_seq = _project_onto_simplex(np.array(best_seq, dtype=float), target)
    return best_seq.tolist()
