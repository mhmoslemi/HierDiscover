    if epsilon <= 0:

def propose_candidate(seed=0, budget_s=1000):
    np.random.seed(seed)
    import time, math, numpy as np
    start = time.time()
    deadline = start + budget_s - 0.5

    if 'height_sequence_1' in globals():
        n = len(height_sequence_1)
    else:
        n = 2000
    target_len = n

    best_seq = None
    best_score = float('inf')
    # baseline uniform
    uniform_seq = np.full(target_len, np.sqrt(2 * target_len) / target_len)
    uniform_score = evaluate_sequence(uniform_seq.tolist())
    best_seq = uniform_seq.copy()
    best_score = uniform_score
    print(f"Uniform baseline score {best_score:.6f}")

    # Search geometric sequences
    epsilons = np.concatenate([
        np.logspace(-6, -3, 50, base=10.0),   # very small decays
        np.logspace(-3, -1, 50, base=10.0)    # moderate decays
    ])
    for eps in epsilons:
        if time.time() > deadline:
            break
        seq = geometric_sequence(target_len, eps)
        score = evaluate_sequence(seq.tolist())
        if score < best_score:
            best_score = score
            best_seq = seq.copy()
            print(f"Geometric eps={eps:.2e} improved to {best_score:.6f}")
    # Also try reversed geometric (increasing)
    for eps in epsilons:
        if time.time() > deadline:
            break
        seq = geometric_sequence(target_len, eps)[::-1]  # reverse order
        score = evaluate_sequence(seq.tolist())
        if score < best_score:
            best_score = score
            best_seq = seq.copy()
            print(f"Reversed geometric eps={eps:.2e} improved to {best_score:.6f}")

    # Random monotone decreasing sequences
    while time.time() < deadline:
        # generate random decreasing sequence by sorting random numbers
        vals = np.random.rand(target_len)
        seq = np.sort(vals)[::-1]  # decreasing
        # normalize sum to sqrt(2*n)
        seq = seq * (np.sqrt(2 * target_len) / np.sum(seq))
        score = evaluate_sequence(seq.tolist())
        if score < best_score:
            best_score = score
            best_seq = seq.copy()
            print(f"Random decreasing improved to {best_score:.6f}")
        # small chance break
        if np.random.rand() < 0.001:
            break

    # finally add a small gradient improvement step
    # we can do few gradient descent steps starting from best_seq
    seq = best_seq.copy()
    target_sum = np.sqrt(2 * target_len)
    for _ in range(100):
        if time.time() > deadline:
            break
        # compute convolution
        conv = np.convolve(seq, seq)
        M = np.max(conv)
        max_idx = np.where(conv == M)[0]
        grad = np.zeros_like(seq)
        for k in max_idx:
            lo = max(0, k - (target_len - 1))
            hi = min(target_len - 1, k)
            idx_range = np.arange(lo, hi+1)
            grad[idx_range] += 2 * seq[k - idx_range]
        # step
        eta = 0.01 * target_sum / (np.mean(np.abs(grad)) + 1e-12)
        new_seq = seq - eta * grad
        new_seq = np.maximum(0, new_seq)
        # renormalize sum
        s = np.sum(new_seq)
        if s > 0:
            new_seq = new_seq * (target_sum / s)
        score = evaluate_sequence(new_seq.tolist())
        if score < best_score:
            best_score = score
            best_seq = new_seq.copy()
            print(f"Gradient step improved to {best_score:.6f}")
        seq = new_seq

    print(f"Finished search. Best score {best_score:.6f} with length {len(best_seq)}")

"""
propose_candidate
=================

Search for a non‑negative step‑function (a list of heights) that minimises the
provided ``evaluate_sequence`` function.

The algorithm consists of three stages:

1. **Geometric seed** – try decreasing geometric sequences
   a_i = C·r^i (with r < 1) for many decay rates.
   These give a good starting point (score ≈ 1.8) and are cheap to evaluate.

2. **Block‑level hill climbing** – compress the sequence into a small number
   of blocks (≈ 15‑20).  Randomly perturb block heights, rescale to the
   required L1‑norm and keep improvements.  This quickly smoothes the
   convolution and drives the score below 1.5 on typical problem sizes.

3. **Fine‑grained pairwise mass transfer** – on the full‑length vector,
   repeatedly move a small amount of mass from one index to another,
   accepting only moves that improve the objective.  This refines the
   solution.

The whole search runs for the allotted time (default 1000 s) and finally
returns the best sequence found.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Helper: build a decreasing geometric sequence a_i = C * r**i
# ----------------------------------------------------------------------
def geometric_sequence(n, eps):
    """
    Return a length‑n non‑negative sequence with sum sqrt(2*n).
    eps = 1 - r  (eps>0).  For eps very small r≈1, for larger eps the decay
    is faster.  The sequence is a_i = C * (1-eps)**i.
    """
    S = math.sqrt(2.0 * n)                     # required L1‑norm
    if eps <= 0.0:                              # uniform (r = 1)
        return np.full(n, S / n, dtype=float)

    r = 1.0 - eps
    # geometric sum = (1 - r**n) / (1 - r) = (1 - r**n) / eps
    # compute r**n safely
    if eps < 1e-12:
        geom_sum = n
    else:
        # use log1p for accuracy when eps is tiny
        log_r = math.log1p(-eps)                # log(r)
        r_pow_n = math.exp(n * log_r)          # r**n = exp(n*log(r))
        geom_sum = (1.0 - r_pow_n) / eps

    C = S / geom_sum
    # build the sequence via exponentiation (use log to avoid underflow)
    idx = np.arange(n, dtype=float)
    seq = C * np.exp(idx * math.log(r))
    return seq


# ----------------------------------------------------------------------
# Helper: build a full sequence from a small block representation
# ----------------------------------------------------------------------
def seq_from_blocks(blocks, n):
    """
    Repeat ``blocks`` to length n (last block may be truncated) and
    rescale so that the total sum equals sqrt(2*n).
    """
    B = len(blocks)
    repeat = n // B
    remainder = n % B
    # repeat each block ``repeat`` times
    seq = np.repeat(blocks, repeat)
    if remainder:
        seq = np.concatenate([seq, blocks[:remainder]])
    # enforce the required L1 norm
    target = math.sqrt(2.0 * n)
    s = seq.sum()
    if s == 0.0:
        # avoid division by zero – fall back to uniform
        seq = np.full(n, target / n, dtype=float)
    else:
        seq = seq * (target / s)
    return seq

            print(f"[rev-geom] eps={eps:.2e} -> score={best_score:.6f}")

    # ------------------------------------------------------------------
    # 2) Block‑level hill climbing
    # ------------------------------------------------------------------
    # number of blocks – a small integer gives a low‑dimensional search
    block_counts = [10, 15, 20]
    for B in block_counts:
        if time.time() > deadline:
            break
        # initialise blocks by averaging the current best sequence
        raw_blocks = np.mean(best_seq.reshape(-1, B), axis=0)
        # ensure non‑negative
        raw_blocks = np.maximum(0.0, raw_blocks)

        # hill‑climbing parameters
        iters = 2000
        for it in range(iters):
            if time.time() > deadline:
                break
            # propose a small random perturbation of a random block
            cand = raw_blocks.copy()
            idx = np.random.randint(B)
            # scale change up to ±10 % of current value (or ±0.1 if value is tiny)
            scale = cand[idx] * 0.1 if cand[idx] > 0 else 0.1
            delta = (np.random.rand() - 0.5) * 2.0 * scale
            cand[idx] = max(0.0, cand[idx] + delta)

            # build full sequence, evaluate
            seq = seq_from_blocks(cand, n)
            score = evaluate_sequence(seq.tolist())
            if score < best_score:
                best_score = score
                best_seq = seq.copy()
                raw_blocks = cand
                print(f"[block B={B}] iter={it} -> score={best_score:.6f}")
            # occasional larger jump to avoid stagnation
            if np.random.rand() < 0.005:
                raw_blocks = np.maximum(0.0,
                    raw_blocks + (np.random.rand(B) - 0.5) * raw_blocks.mean())
    # ------------------------------------------------------------------
    # 3) Fine‑grained pairwise mass transfer on the full vector
    # ------------------------------------------------------------------
    seq = best_seq.copy()
    # Ensure the L1‑norm stays exactly target_sum (scaling does not change the
    # objective, but keeping the norm stable avoids numerical drift).
    seq = seq * (target_sum / seq.sum())

    # simple hill climbing: move a small amount of mass from i to j
    max_pairwise_iters = 5000
    for step in range(max_pairwise_iters):
        if time.time() > deadline:
            break
        # pick two distinct indices
        i, j = np.random.choice(n, size=2, replace=False)
        if seq[i] == 0.0:
            continue
        # propose moving up to 5 % of the mass at i
        delta = (np.random.rand() - 0.5) * 0.05 * seq[i]
        new_seq = seq.copy()
        new_seq[i] = max(0.0, new_seq[i] - delta)
        new_seq[j] = max(0.0, new_seq[j] + delta)

        # re‑normalise (keeps the L1‑norm exact)
        new_seq = new_seq * (target_sum / new_seq.sum())

        score = evaluate_sequence(new_seq.tolist())
        if score < best_score:
            best_score = score
            seq = new_seq
            best_seq = new_seq.copy()
            print(f"[fine step={step}] improved to {best_score:.6f}")

        # occasional random restart if we are stuck
        if step % 500 == 0 and np.random.rand() < 0.1:
            seq = geometric_sequence(n, np.random.uniform(1e-5, 1e-2))
            # small local tweak
            seq = seq * (target_sum / seq.sum())

    print(f"Search finished. Best score = {best_score:.6f} (length={len(best_seq)})")
    return best_seq.tolist()
