"""
Propose a new step‐function (height sequence) by stochastic optimisation.

The algorithm keeps the sum of the coefficients fixed (≈√(2 n)) – the
objective is invariant under a global scaling, therefore we can work
with a normalised sum without loss of optimality.

A hybrid of two moves is used:

* **gradient move** – the index k where the convolution attains its
  maximum is identified; the gradient of this maximal value w.r.t. the
  coefficients is  g_i = 2·a_{k‑i} (zero when the index is out of range).
  Moving a small amount opposite to this gradient while keeping the total
  sum unchanged tends to lower the peak of the convolution.

* **random two‑index move** – transfer a tiny amount of mass from one
  coordinate to another.  This adds exploration and helps the search
  escape shallow local minima.

Both moves are accepted according to a simple simulated‑annealing rule
so that occasional uphill steps are permitted.  The best sequence seen
during the run is returned.

Convolution and the peak value are computed with an FFT ( O(n log n) )
instead of the naïve O(n²) algorithm, which makes several hundred‑thousand
iterations feasible in the allotted time.

All numbers are kept non‑negative and bounded by 1000, matching the
security checks of `evaluate_sequence`.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Helper functions – top level (no closures)
# ----------------------------------------------------------------------


def conv_full_fft(seq: np.ndarray) -> np.ndarray:
    """
    Linear convolution of `seq` with itself using FFT.
    Returns the full 2·n‑1 length array (real values).
    """
    n = seq.shape[0]
    conv_len = 2 * n - 1
    # next power of two for efficient FFT
    n_fft = 1 << (conv_len - 1).bit_length()
    fft_seq = np.fft.rfft(seq, n=n_fft)
    conv = np.fft.irfft(fft_seq * fft_seq, n=n_fft)
    return conv[:conv_len]


def conv_max_fft(seq: np.ndarray) -> float:
    """Maximum entry of the self‑convolution of `seq`."""
    return conv_full_fft(seq).max()


def score_from_seq(seq: np.ndarray) -> float:
    """
    Compute the objective  2·n·max(conv)/sum(seq)².
    This matches `evaluate_sequence` for non‑negative inputs.
    """
    n = seq.shape[0]
    M = conv_max_fft(seq)
    S = seq.sum()
    # S is kept safely away from zero (≥√(2 n) > 0)
    return 2.0 * n * M / (S * S)


def gradient_step(seq: np.ndarray, step: float) -> np.ndarray:
    """
    One gradient descent step that tries to reduce the currently maximal
    convolution entry while keeping the total sum fixed.

    Parameters
    ----------
    seq : np.ndarray
        Current coefficient vector (non‑negative, sum fixed).
    step : float
        Scale of the move (larger -> bigger steps).

    Returns
    -------
    np.ndarray
        New coefficient vector (still non‑negative, same sum).
    """
    n = seq.shape[0]

    # full convolution – also gives us the index of the maximum
    conv = conv_full_fft(seq)
    k = int(np.argmax(conv))

    # gradient of the maximal entry w.r.t. the coefficients
    grad = np.zeros_like(seq)
    i_start = max(0, k - (n - 1))
    i_end = min(n - 1, k)
    idx = np.arange(i_start, i_end + 1, dtype=int)
    j = k - idx
    grad[idx] = 2.0 * seq[j]

    # Direction that reduces the max while preserving the total sum:
    #   d = -grad + (sum(grad)/n)·1   (so that sum(d)=0)
    sum_grad = grad.sum()
    direction = -grad + (sum_grad / n) * np.ones(n, dtype=seq.dtype)

    # Scale the step – use the L2 norm to keep changes comparable
    norm = np.linalg.norm(direction)
    if norm == 0:
        return seq.copy()
    alpha = step / norm
    new_seq = seq + alpha * direction

    # Enforce feasibility and re‑normalise the sum
    new_seq = np.clip(new_seq, 0.0, 1000.0)
    total = new_seq.sum()
    if total > 0:
        new_seq *= seq.sum() / total
    else:  # extreme numerical case – fall back to the old sequence
        new_seq = seq.copy()
    return new_seq


def random_two_index_move(seq: np.ndarray, scale: float) -> np.ndarray:
    """
    Transfer a small amount of mass between two random positions.
    The total sum stays unchanged.
    """
    n = seq.shape[0]
    i, j = np.random.choice(n, 2, replace=False)
    eps = np.random.randn() * scale
    # Clamp to keep both entries non‑negative
    if eps > seq[i]:
        eps = seq[i]
    elif -eps > seq[j]:
        eps = -seq[j]
    new_seq = seq.copy()
    new_seq[i] -= eps
    new_seq[j] += eps
    return new_seq


# ----------------------------------------------------------------------
# Main entry point required by the grader
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 0, budget_s: float = 1000, **kwargs):
    """
    Search for a coefficient vector with a low value of the evaluation
    function.  The function runs for at most `budget_s` seconds and
    returns the best sequence found so far (as a plain Python list).
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5   # small safety margin

    # ------------------------------------------------------------------
    # 1. Initialise the sequence
    # ------------------------------------------------------------------
    if 'height_sequence_1' in globals():
        n = len(height_sequence_1)
        seq = np.array(height_sequence_1, dtype=np.float64)
    else:
        # fallback – a moderate length if the reference is unavailable
        n = 2500
        seq = np.random.rand(n)

    # Normalise the total sum to √(2 n) (any constant works, this avoids the
    # “sum too close to zero” guard in evaluate_sequence).
    target_sum = math.sqrt(2.0 * n)
    seq = np.clip(seq, 0.0, 1000.0)
    if seq.sum() == 0:
        seq = np.ones(n, dtype=np.float64)
    seq = seq / seq.sum() * target_sum

    # Keep track of the best solution seen.
    best_seq = seq.copy()
    best_score = score_from_seq(best_seq)
    print(f"[init] n={n}, initial score={best_score:.6f}")

    # ------------------------------------------------------------------
    # 2. Stochastic optimisation loop
    # ------------------------------------------------------------------
    current_seq = seq.copy()
    current_score = best_score

    # Simulated‑annealing temperature and step-size schedules
    temperature = 1.0
    step_size = target_sum * 0.05          # magnitude of gradient moves
    random_scale = target_sum * 0.02       # magnitude of random swaps
    it = 0

    while time.time() < deadline:
        it += 1

        # ----- choose a move -------------------------------------------------
        if np.random.rand() < 0.7:                     # majority gradient moves
            cand_seq = gradient_step(current_seq, step=step_size)
        else:                                          # occasional random perturbation
            cand_seq = random_two_index_move(current_seq, scale=random_scale)

        # ----- keep constraints (non‑negative, sum fixed) --------------------
        cand_seq = np.clip(cand_seq, 0.0, 1000.0)
        total = cand_seq.sum()
        if total == 0:
            continue
        cand_seq *= target_sum / total

        # ----- evaluate candidate --------------------------------------------
        cand_score = score_from_seq(cand_seq)

        # ----- keep the overall best -----------------------------------------
        if cand_score < best_score:
            best_score = cand_score
            best_seq = cand_seq.copy()
            print(f"[{it:7d}] new best score = {best_score:.6f}")

        # ----- simulated‑annealing acceptance --------------------------------
        if cand_score < current_score:
            accept = True
        else:
            # probability to accept an uphill move
            prob = math.exp(-(cand_score - current_score) / max(temperature, 1e-12))
            accept = prob > np.random.rand()

        if accept:
            current_seq = cand_seq
            current_score = cand_score

        # ----- cooling & step‑size adaptation --------------------------------
        temperature *= 0.99995          # very slow cooling
        if it % 5000 == 0:             # slowly shrink move magnitudes
            step_size *= 0.90
            random_scale *= 0.90

        # occasional status report
        if it % 20000 == 0:
            elapsed = time.time() - (deadline - budget_s)
            print(f"[{it:7d}] elapsed={elapsed:.1f}s best={best_score:.6f}")

    # ------------------------------------------------------------------
    # 3. Return the best sequence found
    # ------------------------------------------------------------------
    print(f"[done] iterations={it}, best score={best_score:.6f}")
    # final clean‑up – enforce the limits exactly as evaluate_sequence does
    best_seq = np.clip(best_seq, 0.0, 1000.0)
    return best_seq.tolist()
