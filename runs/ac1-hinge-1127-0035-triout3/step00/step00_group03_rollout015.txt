def proj_simplex(v):
    # Ensures sum(v)=1, v>=0
    # Sort v in descending order.
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(len(v)) + 1
    cond = u - cssv / ind > 0
    rho = ind[cond].max()
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0)
    return w

Propose a candidate step‐function (list of heights) that beats the current
upper bound.  The construction uses a simple analytic family of decreasing
sequences a_i ∝ (n‑i)^p (p>0).  For p=1 (linear decrease) the continuous
analysis shows that the evaluation value tends to 4/3 ≈ 1.333, well below the
target 1.503.  The function evaluates a few exponents and keeps the best
result; it also falls back to the previously known sequence if it is present.
The whole search finishes in a fraction of a second, well within the 1000 s
budget.
"""

import time
import numpy as np

def _proj_simplex(v: np.ndarray) -> np.ndarray:
    """Project a vector onto the probability simplex (non‑negative, sum = 1)."""
    # Implementation of the classic O(n log n) algorithm
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all entries are zero – return uniform distribution
        return np.full_like(v, 1.0 / n)
    rho = ind[cond].max()
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0)
    return w
