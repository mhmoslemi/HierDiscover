"""
Algorithm: Soft‑max projected gradient descent on the convolution max.
---------------------------------------------------------------

We minimise the evaluation
    2 * n * max(conv(a, a)) / (sum(a)**2)

which (after the normalisation sum(a)=sqrt(2n) used in the original
paper) is exactly the maximum entry of the self‑convolution.
The function is homogeneous, therefore we keep the sum of the
coefficients fixed to sqrt(2*n) and only move inside the non‑negative
simplex.

For a vector `a` let `c = np.convolve(a, a)`.  The entry `c[k]` is a
quadratic form in `a`.  The (sub)gradient of the smooth approximation
```
F(a) = (1/lam) * log( sum_k exp(lam * c[k]) )
```
is
    ∇F = 2 * (a ⊛ w)                     (⊛ = cross‑correlation)
with the weight vector
    w_k = exp(lam * (c[k] - max(c))) / sum(exp(...))

`w` puts almost all weight on the current maximal convolution entry,
so moving a tiny step opposite to `∇F` reduces the true maximum.
The algorithm repeatedly:
  * builds the soft‑max weights,
  * computes the gradient by a single correlation (C code, fast),
  * takes a step in the opposite direction,
  * projects back onto the non‑negative simplex (clip & renormalise).

Multiple random restarts with different lengths are performed until the
time budget is exhausted.  The best sequence found is returned.

The code uses only NumPy and the already‑imported `evaluate_sequence`
function; no file I/O or external solvers are required.
"""

import time
import math
import numpy as np

def _softmax_weights(conv, lam):
    """Return a probability vector w concentrating on the largest entry of conv."""
    max_c = conv.max()
    # shift by the max to avoid overflow; the maximal entry gets weight 1 before normalisation
    shifted = lam * (conv - max_c)
    w = np.exp(shifted)
    w_sum = w.sum()
    if w_sum == 0.0:          # numerical safety – should not happen
        w = np.ones_like(conv) / conv.size
    else:
        w /= w_sum
    return w

def _gradient(a, w):
    """
    Compute grad_i = 2 * Σ_j a_j * w_{i+j}.
    This is a cross‑correlation of `a` and `w`; implemented with NumPy's
    fast C routine.
    """
    n = a.size
    m = w.size                     # = 2*n-1
    # np.correlate returns length n+m-1, the entry for lag ℓ (ℓ≥0) is at index m-1+ℓ
    corr = np.correlate(a, w, mode='full')
    start = m - 1                  # index corresponding to lag 0
    return 2.0 * corr[start:start + n]

def _project_to_simplex(v, target_sum):
    """
    Project non‑negative vector `v` onto the simplex {x ≥ 0, sum(x)=target_sum}.
    Simple scaling works because we never move below zero by a large amount.
    """
    v = np.maximum(v, 0.0)
    s = v.sum()
    if s <= 0.0:
        # re‑initialise uniformly if we lost all mass
        v[:] = target_sum / v.size
    else:
        v *= target_sum / s
    return v

def propose_candidate(seed=42, budget_s=1000, **kwargs):
    """
    Search for a step‑function height sequence with a low evaluation score.

    Parameters
    ----------
    seed : int, optional
        Random seed for reproducibility.
    budget_s : float, optional
        Seconds available for the search (default 1000).

    Returns
    -------
    list[float]
        The best sequence found within the time budget.
    """
    rng = np.random.default_rng(seed)
    deadline = time.time() + budget_s - 0.5   # leave a tiny safety margin

    # keep the best ever sequence
    best_seq = None
    best_score = float('inf')

    # try the provided SOTA sequence if it exists
    try:
        candidate = list(height_sequence_1)
        score = evaluate_sequence(candidate)
        if score < best_score:
            best_score = score
            best_seq = candidate
            print(f"Initial height_sequence_1 score: {best_score:.6f}")
    except Exception:
        pass

    restart = 0
    while time.time() < deadline:
        # ------- pick a problem size ---------------------------------
        # 200‑600 gives a good trade‑off between speed and quality
        n = int(rng.integers(200, 601))

        # ------- random start on the simplex -------------------------
        a = rng.random(n)
        a = _project_to_simplex(a, math.sqrt(2.0 * n))

        # ------- optimisation parameters -----------------------------
        lam = 1.0                     # start with a mild soft‑max
        max_inner = 2000             # inner iterations per restart
        step_factor = 0.5 * math.sqrt(2.0 * n)

        last_improve = 0
        for it in range(max_inner):
            if time.time() >= deadline:
                break

            # compute convolution and its soft‑max weights
            conv = np.convolve(a, a)
            w = _softmax_weights(conv, lam)

            # gradient of the smooth max surrogate
            grad = _gradient(a, w)

            # adaptive step (norm‑based)
            eta = step_factor / (np.linalg.norm(grad) + 1e-12)
            a_new = a - eta * grad
            a_new = _project_to_simplex(a_new, math.sqrt(2.0 * n))

            # evaluate the new point
            try:
                score = evaluate_sequence(a_new.tolist())
            except Exception:
                # any numerical issue – ignore this candidate
                continue

            # keep the global best
            if score < best_score:
                best_score = score
                best_seq = a_new.tolist()
                print(f"[{time.time():.1f}s] New best {best_score:.6f} "
                      f"(n={n}, restart={restart}, it={it})")
                last_improve = it

            # accept the new point if it improves (or is very close)
            # otherwise stay at the old point – this provides a simple
            # descent with occasional back‑tracking
            if score <= best_score * 1.001:
                a = a_new

            # slowly increase the soft‑max sharpness – every 250 steps
            if (it + 1) % 250 == 0 and lam < 15.0:
                lam *= 1.5

            # random jitter if we have been stuck for many steps
            if it - last_improve > 500:
                jitter = rng.normal(scale=0.01, size=n)
                a = _project_to_simplex(np.maximum(a + jitter, 0.0),
                                        math.sqrt(2.0 * n))
                last_improve = it

        restart += 1

    # fallback – if nothing succeeded, return the SOTA sequence or a tiny uniform vector
    if best_seq is None:
        try:
            best_seq = list(height_sequence_1)
        except Exception:
            best_seq = [1.0]

    return best_seq
