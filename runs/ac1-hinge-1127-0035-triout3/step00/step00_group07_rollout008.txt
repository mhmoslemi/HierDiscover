"""Fast construction of a low‑score step‑function.

The evaluation function  

    2·n·max(conv(a,a)) / ( sum(a) )²  

is homogeneous of degree 0: multiplying the whole sequence by a constant does
not change the score.  By choosing a monotone power‑law shape  

    a_i = ((i+1)/N)^p · 1000      (0 ≤ i < N)

the largest entry is exactly 1000, therefore the security clipping in
`evaluate_sequence` does not alter the sequence.  For a fixed exponent p the
asymptotic value of the score (large N) is  

    2·(p+1)²·B(p+1,p+1),

where B is the beta‐function.  This value rapidly decreases with p; e.g.
p=6 gives a score ≈ 8·10⁻³.  The algorithm simply tries a few values of p
and a few lengths N, evaluates each candidate with NumPy (much faster than
calling the provided `evaluate_sequence`), and returns the best one found
within the allotted time.
"""

import time
import numpy as np

def _score(seq: list[float]) -> float:
    """Return the raw score of a sequence (the same quantity returned by
    evaluate_sequence, but calculated locally for speed)."""
    arr = np.asarray(seq, dtype=np.float64)
    conv = np.convolve(arr, arr)
    max_b = conv.max()
    sum_a = arr.sum()
    # the checks in evaluate_sequence guarantee sum_a > 0.01,
    # but we guard against a degenerate case here.
    if sum_a < 1e-12:
        return float('inf')
    return float(2.0 * arr.size * max_b / (sum_a * sum_a))

def propose_candidate(seed: int = 42,
                     budget_s: float = 1000,
                     **kwargs) -> list[float]:
    """
    Generate a step‑function with a very low evaluation score.

    The function explores a few exponent/length combinations of the form
    a_i = ((i+1)/N)^p * 1000, respects the 0 ≤ a_i ≤ 1000 bound, and returns
    the best sequence found before the time budget expires.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.1  # leave a tiny safety margin

    # Parameter ranges – small enough to keep the search cheap.
    exponents = [1, 2, 3, 4, 5, 6, 7, 8]       # p in the description
    lengths   = [500, 1000, 2000, 5000]       # N (the number of steps)

    best_seq = None
    best_score = float('inf')

    for p in exponents:
        for N in lengths:
            if time.time() > deadline:
                break

            # a_i = ((i+1)/N)^p * 1000   guarantees max(a) == 1000 (clipping safe)
            # using float arithmetic directly avoids huge intermediate numbers.
            seq = [((i + 1) / N) ** p * 1000.0 for i in range(N)]

            sc = _score(seq)

            if sc < best_score:
                best_score = sc
                best_seq = seq
                print(
                    f'New best → p={p:2d}, N={N:5d}, score={best_score:.6f}'
                )
        # break outer loop early if time ran out
        if time.time() > deadline:
            break

    # Fallback: uniform sequence (should never be needed)
    if best_seq is None:
        N = 1000
        best_seq = [1.0] * N
        best_score = _score(best_seq)
        print(f'Fallback uniform sequence, score={best_score:.6f}')

    # Optional sanity check with the official evaluator (very cheap for one call)
    try:
        from __main__ import evaluate_sequence  # will be present in the runner
        official_score = evaluate_sequence(best_seq)
        print(f'Official evaluator score = {official_score:.6f}')
    except Exception:
        pass

    return best_seq
