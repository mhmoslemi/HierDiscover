def project_onto_simplex(v):
    # from Duchi et al.
    # ensure nonnegative and sum=1
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    rho = np.nonzero(u - cssv / np.arange(1, n+1) > 0)[0][-1]
    theta = cssv[rho] / (rho+1)
    w = np.maximum(v - theta, 0)
    return w

def project_simplex(v):
    n = len(v)
    # sort v in descending order
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    rho = np.where(u - cssv / np.arange(1, n+1) > 0)[0]
    if len(rho) == 0:
        theta = 0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0)
    return w

assistantfinal**Solution Overview**

The evaluation function is  

\[
F(a)=\frac{2\,n\;\max\limits_k (a*a)_k}{\big(\sum_i a_i\big)^2},
\]

where `a*a` is the discrete convolution of the non‑negative vector `a`.  
Because `F(λa)=F(a)` the scale of `a` does not matter; we may fix
\(\sum_i a_i = 1\) and simply try to **reduce the maximal convolution entry**.

The program implements a *simulated‑annealing* search that works directly on the
probability vector `p` (i.e. `a` after normalisation).  The main ideas are:

* **Targeted moves** – the current peak of the convolution is at index `j`.
  The contribution of a pair `(i , j‑i)` to this peak equals
  `p[i]·p[j‑i]`.  We pick the pair with the largest product and move a
  small amount of mass from the larger of the two indices to a random
  destination.  This directly attacks the dominant terms of the peak.

* **Fast convolution update** – moving a mass `δ` from `src` to `dst`
  changes the convolution by a known sparse formula, so we update the
  convolution array in *O(n)* time instead of recomputing it from scratch.
  Every 10 000 iterations we recompute it completely to avoid drift.

* **Simulated annealing** – a temperature `T` is slowly decreased.
  A move that increases the score is accepted with probability
  `exp(-(Δscore)/T)`.  This allows the search to escape shallow local minima.

* **Multiple random restarts** – after a long period without improvement the
  current vector is slightly re‑randomised, providing fresh directions.

The algorithm runs until the allotted time expires (the budget is 1000 s,
the function leaves a 0.1 s safety margin).  The best sequence found is
returned as a plain Python list; `evaluate_sequence` will clip it to the
required range automatically.

```python
"""
Simulated‑annealing optimiser for the step‑function bound.

The search works on a probability vector `p` (sum = 1).  At each iteration
the index `j` where the convolution `c = p * p` attains its maximum is
identified.  Among all pairs (i, j‑i) contributing to this peak the pair
with the largest product p[i]*p[j‑i] is selected, and a small amount of
mass `δ` is transferred from the larger component of the pair to a random
position.  The convolution is updated using an O(n) incremental formula,
while a Metropolis acceptance rule with a slowly cooling temperature
allows occasional uphill moves.  The best sequence encountered during the
run is returned.
"""

import time
import math
import numpy as np


def _conv_increment(p, conv, src, dst, delta):
    """
    Return a new convolution array after moving `delta` mass from `src`
    to `dst`.  The update uses the exact identity

        c' = c + 2·δ·(shift(p, dst) – shift(p, src))
              + δ²·(e_{2·dst} + e_{2·src} – 2·e_{src+dst})

    where `shift(p, s)[k] = p[k‑s]` (zero outside the support) and `e_k`
    denotes the Kronecker delta.
    """
    n = p.shape[0]
    conv_new = conv.copy()

    # 2·δ·(shift(p, dst) – shift(p, src))
    # shift by dst : add to slice [dst : dst+n]
    conv_new[dst:dst + n] += 2.0 * delta * p
    # shift by src : subtract from slice [src : src+n]
    conv_new[src:src + n] -= 2.0 * delta * p

    # quadratic correction δ² terms
    d2 = delta * delta
    conv_new[2 * dst] += d2
    conv_new[2 * src] += d2
    conv_new[src + dst] -= 2.0 * d2

    return conv_new


def propose_candidate(seed: int = 42, budget_s: float = 1000, **kwargs):
    """
    Simulated‑annealing optimisation for the evaluation function.

    Parameters
    ----------
    seed : int
        Random seed for reproducibility.
    budget_s : float
        Total time budget (seconds).  The function stops a few
        seconds before the deadline to guarantee a return.

    Returns
    -------
    list[float]
        The best step‑function heights found.
    """
    np.random.seed(seed)
    # safety margin – we stop a bit before the deadline
    deadline = time.time() + budget_s - 0.2

    # --------------------------------------------------------------
    # problem size – we keep the length used by the current best solution
    # --------------------------------------------------------------
    n = 4160                     # length of the sequence
    conv_len = 2 * n - 1

    # --------------------------------------------------------------
    # helpers
    # --------------------------------------------------------------
    def score_from_conv(conv_arr):
        """2·n·max(conv) – the exact value returned by evaluate_sequence."""
        return float(2.0 * n * np.max(conv_arr))

    # --------------------------------------------------------------
    # initial state (uniform + tiny noise)
    # --------------------------------------------------------------
    p = np.full(n, 1.0 / n, dtype=np.float64)
    p += 0.001 * np.random.rand(n)      # break symmetry
    p /= p.sum()                        # normalise exactly

    conv = np.convolve(p, p)            # exact convolution (size 2n‑1)
    curr_score = score_from_conv(conv)

    best_p = p.copy()
    best_score = curr_score

    print(f"[init] score = {curr_score:.6f}")

    # annealing schedule
    T = 0.02                # initial temperature
    decay = 0.99996         # exponential cooling factor

    it = 0
    last_improve_it = 0

    # --------------------------------------------------------------
    # main loop
    # --------------------------------------------------------------
    while time.time() < deadline:
        # ------------------------------------------------------------------
        # locate the current convolution peak and the pair with largest product
        # ------------------------------------------------------------------
        j_max = int(np.argmax(conv))

        lo = max(0, j_max - (n - 1))
        hi = min(j_max, n - 1)
        idx = np.arange(lo, hi + 1, dtype=np.int64)
        partner = j_max - idx
        products = p[idx] * p[partner]
        max_pair_pos = np.argmax(products)
        i_max = idx[max_pair_pos]
        k_max = partner[max_pair_pos]

        # choose source – the larger entry of the pair
        src = i_max if p[i_max] >= p[k_max] else k_max

        # pick a destination different from src
        dst = np.random.randint(n - 1)
        if dst >= src:
            dst += 1   # ensure dst != src

        # amount to move – a small random fraction of src's mass
        max_transfer = p[src]
        if max_transfer <= 1e-12:
            # src practically empty – pick a different src
            continue
        delta = max_transfer * np.random.uniform(0.001, 0.015)

        # --------------------------------------------------------------
        # propose the move (incremental convolution update)
        # --------------------------------------------------------------
        conv_candidate = _conv_increment(p, conv, src, dst, delta)

        cand_score = score_from_conv(conv_candidate)

        # --------------------------------------------------------------
        # Metropolis acceptance test
        # --------------------------------------------------------------
        accept = False
        if cand_score < curr_score:
            accept = True
        else:
            prob = math.exp(-(cand_score - curr_score) / max(T, 1e-12))
            if np.random.rand() < prob:
                accept = True

        if accept:
            # commit the move
            p[src] -= delta
            p[dst] += delta
            conv = conv_candidate
            curr_score = cand_score

            # update best record
            if cand_score < best_score:
                best_score = cand_score
                best_p = p.copy()
                last_improve_it = it
                print(f"[it {it}] NEW BEST = {best_score:.6f}")

        # --------------------------------------------------------------
        # periodic maintenance
        # --------------------------------------------------------------
        it += 1
        T *= decay               # cool down

        # every 10 000 iterations recompute the convolution from scratch
        if (it & 0x3FFF) == 0:   # 16384 ≈ 10 k
            conv = np.convolve(p, p)
            curr_score = score_from_conv(conv)

        # if we have not improved for a long stretch, inject a small random shake
        if it - last_improve_it > 300000:
            # shake 1 % of the mass randomly
            shake = np.random.rand(n)
            shake *= 0.01 / shake.sum()
            p = p * 0.99 + shake
            p /= p.sum()
            conv = np.convolve(p, p)
            curr_score = score_from_conv(conv)
            last_improve_it = it
            print(f"[it {it}] shake – score = {curr_score:.6f}")

    # final result --------------------------------------------------------
    print(f"[finished] best score = {best_score:.6f} after {it} iterations")
    return best_p.tolist()
```
