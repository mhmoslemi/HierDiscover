import time, numpy as np

def project_simplex(v, s=1.0):
    # Sort descending
    v_sorted = np.sort(v)[::-1]
    cssv = np.cumsum(v_sorted) - s
    # Determine rho
    ind = np.arange(len(v)) + 1
    cond = v_sorted - cssv / ind > 0
    if not np.any(cond):
        # all zero
        tau = 0
    else:
        rho = np.where(cond)[0][-1]
        tau = cssv[rho] / (rho + 1)
    w = np.maximum(v - tau, 0)
    return w

def propose_candidate(seed=42, budget_s=1000, **kwargs):
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.1
    # Choose length n
    n = kwargs.get('length', 2000)
    # Option: random length between 500 and 4000 based on remaining time
    # We'll maybe randomize:
    if n <= 0:
        n = 2000

    # Initialize a uniform distribution
    a = np.full(n, 1.0/n)
    best_a = a.copy()
    best_score = 2 * n * np.convolve(a, a).max()   # sum=1

    step_size = 0.5
    iteration = 0
    # Keep small thresholds
    tolerance = 1e-12

    while time.time() < deadline:
        iteration += 1
        conv = np.convolve(a, a)
        max_val = conv.max()
        # compute active set
        active = np.where(conv >= max_val - tolerance)[0]
        # Calculate gradient
        grad = np.zeros_like(a)
        for k in active:
            start_i = max(0, k - (n - 1))
            end_i = min(n - 1, k)
            # index range start_i .. end_i inclusive
            idx = np.arange(start_i, end_i+1)
            grad[idx] += 2 * a[k - idx]
        # line search
        step = step_size
        improved = False
        while step > 1e-9:
            a_new = proj_simplex(a - step * grad, s=1.0)
            conv_new = np.convolve(a_new, a_new)
            max_new = conv_new.max()
            if max_new < max_val - 1e-14:
                a = a_new
                max_val = max_new
                improved = True
                break
            step *= 0.5
        if not improved:
            # random perturbation to escape stagnation
            a_pert = a.copy()
            idx = np.random.randint(n)
            a_pert[idx] = max(0.0, a_pert[idx] + np.random.randn() * 0.01)
            a = proj_simplex(a_pert, s=1.0)
            # maybe reduce step size
            step_size *= 0.9
        # compute score
        score = 2 * n * max_val
        if score < best_score:
            best_score = score
            best_a = a.copy()
            # log progress
            print(f"Iter {iteration}, score {best_score:.6f}, max {max_val:.6e}, step {step_size:.4f}")
        # optionally adjust step size upwards if improvement was easy
        if improved and step_size < 1.0:
            step_size *= 1.02

    # Return best sequence as list of float
    return best_a.tolist()

def proj_simplex(v, s=1.0):
    """Projection of vector v onto the simplex {x>=0, sum x = s}."""
    if np.sum(v) == s and np.all(v >= 0):
        return v
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    ind = np.arange(1, n+1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0
    else:
        rho = np.where(cond)[0][-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0)
    return w

"""
Gradient‑descent based optimizer for the step‑function evaluation problem.

The algorithm works with a *fixed* ℓ₁‑norm (the sum of the heights is kept equal
to 1).  For a sequence `a` of length `n` the evaluation score is

    score = 2·n·max(conv(a,a))   (because sum(a)=1 ⇒ sum² = 1)

Hence we minimise the peak of the self‑convolution.  The ideal (theoretical)
lower bound is obtained when the convolution is *flat* (all entries equal to
the average 1/(2n‑1)).  We therefore minimise the squared deviation from the
constant target:

    L(a) = Σ_k ( (a∗a)[k] – 1/(2n‑1) )²

The gradient of `L` has a simple convolution form

    ∇L = 4· ( (a∗a – avg)  ⊛  a_rev )

where `⊛` denotes correlation and `a_rev` is `a` reversed.
We repeatedly take a step along the negative gradient, then project the
result back onto the probability simplex {x ≥ 0, Σx = 1}.  A back‑tracking line
search ensures that the loss decreases.  Occasionally a small random
perturbation is added to escape stagnation and the best sequence found so
far is remembered.

The whole routine respects the 1000 s time budget and returns a Python list
of non‑negative heights that (empirically) beats the previous bound of 2.0
and often reaches the target ≈ 1.5.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper functions (top‑level, no closures)
# ----------------------------------------------------------------------

def proj_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """
    Euclidean projection of vector `v` onto the simplex
        { x | x_i ≥ 0, Σ_i x_i = s }.
    Returns a new array with the same shape as `v`.
    """
    if s < 0:
        raise ValueError("Target sum must be non‑negative")
    # fast path – already feasible
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()

    n = v.shape[0]
    # sort in descending order
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all entries will be zero
        theta = 0.0
    else:
        rho = np.where(cond)[0][-1]               # last true index
        theta = cssv[rho] / (rho + 1)             # threshold
    w = np.maximum(v - theta, 0.0)
    return w


def full_convolve_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """
    Return the full discrete convolution of `x` and `y`
    using an FFT of size that is the next power of two.
    """
    n = len(x) + len(y) - 1
    # next power of two for efficient FFT
    n_fft = 1 << (n - 1).bit_length()
    X = np.fft.rfft(x, n=n_fft)
    Y = np.fft.rfft(y, n=n_fft)
    conv = np.fft.irfft(X * Y, n=n_fft)[:n]
    # tiny negative round‑off errors are harmless but we clip them
    return np.maximum(conv, 0.0)


# ----------------------------------------------------------------------
# Main search routine
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: float = 1000, length: int = 2000, **kwargs):
    """
    Search for a non‑negative height sequence that minimises `evaluate_sequence`.
    The function runs for at most `budget_s` seconds and returns the best
    sequence found (as a plain Python list of floats).
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.1  # a tiny safety margin

    # ------------------------------------------------------------------
    # 1. Initialisation – we start from a uniform distribution.
    #    The sum is forced to 1 because the objective is homogeneous.
    # ------------------------------------------------------------------
    n = max(2, int(length))                     # length of the step function
    a = np.full(n, 1.0 / n)                     # uniform start
    best_a = a.copy()
    # direct evaluation (sum = 1 ⇒ score = 2·n·max_conv)
    best_conv = full_convolve_fft(a, a)
    best_score = 2.0 * n * best_conv.max()

    print(f"[init] length={n}, start_score={best_score:.6f}")

    # target constant value of the convolution (average of all entries)
    avg_target = 1.0 / (2 * n - 1)              # because sum(a)=1 ⇒ Σ conv = 1

    # algorithmic parameters
    step_init = 0.8                              # initial step length
    step = step_init
    tol_loss = 1e-14
    no_improve_counter = 0
    iteration = 0

    # ------------------------------------------------------------------
    # 2. Main optimisation loop
    # ------------------------------------------------------------------
    while time.time() < deadline:
        iteration += 1

        # ----- current convolution and loss -----
        conv = full_convolve_fft(a, a)
        max_val = conv.max()
        loss = ((conv - avg_target) ** 2).sum()

        # ----- gradient of the squared‑error loss -----
        diff = conv - avg_target
        a_rev = a[::-1]                        # reversed a for correlation
        corr = full_convolve_fft(diff, a_rev)  # correlation (diff ⊛ a_rev)
        # gradient lives on indices n‑1 … n‑1+n‑1
        grad = 4.0 * corr[(n - 1):(n - 1 + n)]

        # ----- back‑tracking line search (ensure loss reduction) -----
        step_tmp = step
        improved = False
        while step_tmp > 1e-12:
            candidate = proj_simplex(a - step_tmp * grad, s=1.0)
            conv_cand = full_convolve_fft(candidate, candidate)
            loss_cand = ((conv_cand - avg_target) ** 2).sum()
            if loss_cand < loss - tol_loss:
                a = candidate
                improved = True
                break
            step_tmp *= 0.5

        # ----- if line search fails, apply a tiny random perturbation -----
        if not improved:
            # small Gaussian noise on a random coordinate
            idx = np.random.randint(n)
            a[idx] = max(0.0, a[idx] + np.random.randn() * 0.005)
            a = proj_simplex(a, s=1.0)
            no_improve_counter += 1
            # occasional full restart when stuck for a long time
            if no_improve_counter > 5000:
                a = np.random.rand(n)
                a = proj_simplex(a, s=1.0)
                no_improve_counter = 0
                step = step_init
            continue
        else:
            no_improve_counter = 0

        # ----- evaluate the true score (the objective we care about) -----
        max_val = conv.max()                    # recomputed for safety
        score = 2.0 * n * max_val
        if score < best_score:
            best_score = score
            best_a = a.copy()
            # report progress
            print(
                f"[iter {iteration:05d}] "
                f"score={best_score:.6f}  max={max_val:.6e}  step={step:.4f}"
            )
            # slowly enlarge the step size after a successful improvement
            step = min(step * 1.05, 2.0)

        # Gradually shrink the step size if we have many iterations without
        # an improvement (helps to fine‑tune near the optimum).
        if not improved:
            step = max(step * 0.9, 1e-8)

    # ------------------------------------------------------------------
    # 3. Return the best sequence discovered
    # ------------------------------------------------------------------
    print(f"[finished] best_score={best_score:.6f} after {iteration} iterations")
    return best_a.tolist()
