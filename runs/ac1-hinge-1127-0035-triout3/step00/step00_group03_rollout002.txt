import time

def compute_max_conv(p, fft_size):
    """Return (max value, argmax) of autocorrelation p*p using FFT."""
    fp = np.fft.rfft(p, fft_size)
    conv = np.fft.irfft(fp * fp, fft_size)
    conv = conv[:2*len(p)-1]
    max_idx = int(conv.argmax())
    return float(conv[max_idx]), max_idx

def evaluate_p(p, fft_size):
    n = len(p)
    max_val, _ = compute_max_conv(p, fft_size)
    return 2.0 * n * max_val

    The algorithm keeps the sum of the coefficients equal to 1
    (any scaling does not affect the score) and repeatedly
    transfers a small amount of weight from one index to another.
    Moves are biased towards indices that contribute most to the
    current maximal autocorrelation term.
    The best sequence seen before the timeout is returned.
    """

    try:
        # height_sequence_1 is provided by the environment.

    fft_size = 1 << ((2 * n - 1).bit_length())

    # Compute the score of the current candidate
    cur_max, cur_idx = compute_max_conv(p, fft_size)
    cur_score = 2.0 * n * cur_max
    best_score = cur_score

    # Parameters for the stochastic search

    max_step = 0.02                   # maximal proportion of the source weight to move
    T0 = 1e-3                         # initial temperature for occasional uphill moves
    it = 0
    last_report = start_time

        it += 1
        # -----------------------------------------------------------------
        # pick a move
        # -----------------------------------------------------------------
        if np.random.rand() < prob_target:
            # Targeted move – try to lower the current maximum autocorrelation term
            k = cur_idx
            i_min = max(0, k - (n - 1))
            i_max = min(n - 1, k)

            i_range = np.arange(i_min, i_max + 1, dtype=np.int64)
            j_range = k - i_range
            contrib = p[i_range] * p[j_range]
            # pick the index with the largest product a_i a_j
            i_star = i_range[np.argmax(contrib)]
            # amount to transfer – never more than max_step of the source weight
            eps = min(p[i_star] * max_step, p[i_star] * (0.01 + 0.01 * np.random.rand()))
            if eps <= 0:

                i_star = np.random.randint(n)
                eps = p[i_star] * max_step
            # destination index – uniform random, ensure it differs from the source
            j_star = np.random.randint(n - 1)
            if j_star >= i_star:
                j_star += 1
            new_p = p.copy()
            new_p[i_star] -= eps
            new_p[j_star] += eps
        else:

            eps = p[i_src] * max_step * np.random.rand()

        # Numerical safety – the source weight may become slightly negative because of round‑off

        new_p = np.maximum(new_p, 0.0)
        s = new_p.sum()

            new_p = np.full(n, 1.0 / n, dtype=np.float64)

        new_score = 2.0 * n * new_max

        if new_score < cur_score:

    # but we keep the vector non‑negative and renormalised.
    result = best_p.tolist()
    # Ensure the list length matches the original length if that information is required.
    return result

"""
Simulated‑annealing / greedy optimisation for the step‑function evaluation.

The score of a list `a` is

    score = 2 * n * max( a * a ) / (sum(a)**2)

where `a*a` is the discrete (full) convolution of `a` with itself.
Because a uniform scaling of the vector cancels out, we may keep
the sum of the coefficients equal to 1.  The algorithm therefore works
with a probability vector `p` (non‑negative, sum = 1) and tries to
flatten the autocorrelation `p * p`.  At each iteration a small amount
of mass is moved from an index that contributes most to the current
maximum autocorrelation term to an index that contributes the least.
The move is accepted greedily; a tiny amount of simulated annealing
allows occasional uphill steps to escape plateaus.

The best sequence seen before the time budget expires is returned.
"""

import time
import math
import numpy as np

def _max_convolution(p: np.ndarray, fft_size: int):
    """
    Return the maximum value of the autocorrelation p*p and its index.
    The convolution is computed by FFT for O(n log n) time.
    """
    # forward FFT of size `fft_size` (real → complex)
    fp = np.fft.rfft(p, fft_size)
    # element‑wise product gives the transform of the autocorrelation
    conv = np.fft.irfft(fp * fp, fft_size)
    # we only need the first 2*n‑1 entries
    conv = conv[: 2 * len(p) - 1]
    max_idx = int(conv.argmax())
    return float(conv[max_idx]), max_idx


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step function that minimises ``evaluate_sequence``.
    The function runs for at most ``budget_s`` seconds and returns the best
    sequence (list of floats) found.

    Parameters
    ----------
    seed : int, optional
        Random seed – the same seed always produces the same trajectory.
    budget_s : int, optional
        Time budget in seconds.

    Returns
    -------
    list[float]
        The best coefficient list discovered.
    """
    # -------------------------------------------------------------
    # Basic configuration
    # -------------------------------------------------------------
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 2.0          # a little safety margin
    # length of the sequence – use the known SOTA length if available,
    # otherwise fall back to a moderate size.
    try:
        n = len(height_sequence_1)                 # type: ignore[name‑defined]
    except Exception:
        n = 2000

    # initialise with a uniform distribution (sum = 1)
    p = np.full(n, 1.0 / n, dtype=np.float64)

    # pre‑compute a power‑of‑two FFT length (fast for np.fft)
    fft_size = 1 << ((2 * n - 1).bit_length())

    # current evaluation of the uniform start point
    cur_max, cur_idx = _max_convolution(p, fft_size)
    cur_score = 2.0 * n * cur_max
    best_score = cur_score
    best_p = p.copy()

    # -------------------------------------------------------------
    # Hyper‑parameters for the stochastic search
    # -------------------------------------------------------------
    prob_target = 0.75            # chance to use a “targeted” move
    max_step = 0.02               # maximal proportion of the source weight to move
    T0 = 1e-3                     # initial temperature for uphill acceptance
    last_report = start_time
    it = 0

    # -------------------------------------------------------------
    # Main optimisation loop
    # -------------------------------------------------------------
    while time.time() < deadline:
        it += 1

        # ---------------------- pick a move ----------------------
        if np.random.rand() < prob_target:
            # ----- Targeted move -----
            k = cur_idx                               # shift where the autocorrelation peaks
            # product_i = p[i] * p[k-i] (zero if the partner index is out of range)
            i_arr = np.arange(n, dtype=np.int64)
            partner = k - i_arr
            valid = (partner >= 0) & (partner < n)
            product = np.zeros_like(p)
            product[valid] = p[i_arr[valid]] * p[partner[valid]]

            # source – index that contributes most to the peak
            i_src = int(product.argmax())
            # destination – index that contributes least (preferably zero)
            zero_idxs = np.flatnonzero(product == 0.0)
            if zero_idxs.size > 0:
                i_dst = int(np.random.choice(zero_idxs))
                # make sure we really move somewhere
                if i_dst == i_src:
                    i_dst = (i_dst + 1) % n
            else:
                i_dst = int(product.argmin())
                if i_dst == i_src:
                    i_dst = (i_dst + 1) % n

            # amount of mass to transfer
            eps = min(p[i_src] * max_step,

        else:

            i_src = np.random.randint(n)
            i_dst = np.random.randint(n - 1)
            if i_dst >= i_src:

        new_p[i_src] -= eps
        new_p[i_dst] += eps

        # numerical safety – clip negatives and renormalise to sum=1
        new_p = np.maximum(new_p, 0.0)
        s = new_p.sum()
        if s == 0.0:                                 # extremely unlikely

        if new_score < cur_score:

                    print(f"Iter {it:07d} – time {elapsed:6.1f}s – best score {best_score:.6f}")
                    last_report = now

    # -------------------------------------------------------------
    # Return the best sequence (as a plain Python list)
    # -------------------------------------------------------------
    return best_p.tolist()
