def _uniform_sequence(length: int, value: float = 1.0) -> list[float]:
    """Return a list of `length` copies of `value`."""
    return [float(value) for _ in range(length)]

def _perturb(seq: list[float], scale: float = 0.05) -> list[float]:
    """
    Produce a new sequence by adding a small random delta to a random
    element (while keeping the element non‑negative).  The total sum
    changes only slightly – this is fine because the objective is
    scale‑invariant.
    """
    new_seq = seq.copy()
    i = random.randint(0, len(seq) - 1)
    delta = random.gauss(0.0, scale)
    new_val = max(0.0, new_seq[i] + delta)
    new_seq[i] = new_val
    return new_seq
