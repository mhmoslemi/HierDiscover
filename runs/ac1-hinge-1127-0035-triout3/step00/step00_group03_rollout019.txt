def propose_candidate(seed=123, budget_s=1000, **kwargs):
    import numpy as np, time
    np.random.seed(seed)
    deadline = time.time() + budget_s - 1
    n = 500
    sum_target = np.sqrt(2 * n)
    best_seq = None
    best_score = np.inf

    # Precompute index array
    i = np.arange(n)
    # FFT size:
    size = 1 << (int(np.ceil(np.log2(2*n-1))))
    # Grid search
    p_vals = np.linspace(-2, 2, 41)
    t_vals = np.linspace(0.1, 10, 100)  # covering
    for p in p_vals:
        for t in t_vals:
            if time.time() > deadline:
                break
            r = np.exp(-t / n)
            a = (i+1) ** p * (r ** i)
            # clip tiny values maybe
            # Normalize sum
            total = np.sum(a)
            if total == 0:
                continue
            a = a * (sum_target / total)
            # Compute convolution via FFT
            A = np.fft.rfft(a, n=size)
            conv = np.fft.irfft(A * A, n=size)[:2*n-1]
            maxc = conv.max()
            score = maxc  # equals evaluate after normalization
            if score < best_score:
                best_score = score
                best_seq = a.copy()
                print(f"New best eval {best_score:.6f} p={p:.2f} t={t:.2f}")
    # Local refinement
    if best_seq is not None:
        a = best_seq.copy()
        # gradient descent on smooth max
        beta = 10.0
        lr = 0.1
        for it in range(5000):
            if time.time() > deadline:
                break
            # Compute conv
            A = np.fft.rfft(a, n=size)
            conv = np.fft.irfft(A * A, n=size)[:2*n-1]
            # log-sum-exp weights
            maxc = conv.max()
            # compute smoothed weights
            exp_vals = np.exp(beta * (conv - maxc))
            w = exp_vals / exp_vals.sum()
            # compute gradient
            # grad_i = 2 * sum_{j=0}^{n-1} w[i+j] * a[j]
            # we can compute via convolution: w (length 2n-1) convolve a (length n) giving length 3n-2.
            # But simpler: use loop
            grad = np.zeros_like(a)
            for i_idx in range(n):
                grad[i_idx] = 2.0 * np.dot(w[i_idx:i_idx+n], a)
            # Update
            a -= lr * grad
            a[a < 0] = 0.0
            # renormalize
            total = np.sum(a)
            if total == 0:
                a = np.ones(n) * (sum_target / n)
                total = sum_target
            a *= (sum_target / total)
            # optionally decay lr, beta
            if it % 100 == 0:
                # re-evaluate
                A = np.fft.rfft(a, n=size)

                    best_score = score
                    best_seq = a.copy()
                    print(f"Iter {it}: improved eval {best_score:.6f}")
                # adjust lr
                lr *= 0.99
                beta = min(beta * 1.01, 200.0)

    # Return as list of floats
    if best_seq is None:
        # fallback uniform
        best_seq = np.ones(n) * (sum_target / n)
    return best_seq.tolist()

"""
propose_candidate
-----------------

Generate a non‑negative step‑function (list of heights) that minimizes the
evaluation function

    2·n·max(conv(sequence,sequence)) / (sum(sequence)**2)

where ``conv`` is the discrete convolution.  The function normalises the
sequence to the standard sum `sqrt(2·n)` (the evaluation is scale‑invariant)
and then performs a two‑phase search:

1. **Coarse parametric search** – the sequence is restricted to the form  

       a_i = (i+1) ** p  *  r**i            ( i = 0 … n‑1 )

   with `r = exp(-t/n)`.  A grid over the parameters `p` and `t` is explored
   (using FFT‑based convolution for speed).  The best candidate from this
   grid is kept.

2. **Smooth‑max gradient descent** – starting from the best candidate,
   the smooth approximation of the maximum

       S(beta) = (1/beta)·log( Σ exp(beta·conv_k) )

   is minimised by projected gradient steps.  The gradient of `S(beta)`
   w.r.t. the coefficients can be written as a convolution, so each step
   costs only O(n log n).  After every step the vector is projected onto the
   simplex (non‑negative, fixed L1‑norm) and the best sequence seen so far
   is recorded.

The whole routine respects a user supplied time budget (default 1000 s)
and prints progress messages.  The returned list is ready to be passed to
`evaluate_sequence`.
"""

import time
import numpy as np

def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """Return a list of heights that (hopefully) beats the current bound."""
    np.random.seed(seed)
    deadline = time.time() + budget_s - 2          # safety margin

    # ------------------------------------------------------------------
    # Parameters of the search
    # ------------------------------------------------------------------
    n = 800                                        # length of the step function
    target_sum = np.sqrt(2.0 * n)                  # normalisation used in the paper
    # FFT size large enough for any convolution we need (2n‑1 + n‑1)
    fft_len = 1 << (int(np.ceil(np.log2(3 * n))))

    # --------------------------------------------------------------
    # 1) Coarse parametric search
    # --------------------------------------------------------------
    best_seq = None
    best_score = np.inf

    idx = np.arange(n)                             # indices 0 … n‑1
    p_vals = np.linspace(-2.0, 2.0, 41)            # shape exponent
    t_vals = np.linspace(0.1, 10.0, 100)           # decay strength (t ≈ n·(1‑r))

    for p in p_vals:
        # early exit if time is up
        if time.time() > deadline:
            break
        # pre‑compute (i+1)^p once for this p
        base = (idx + 1.0) ** p
        for t in t_vals:
            # ----- build the candidate -------------------------------------------------
            r = np.exp(-t / n)                     # geometric factor, 0 < r < 1
            a = base * (r ** idx)                 # a_i = (i+1)^p · r^i
            total = a.sum()
            if total == 0.0:
                continue
            a *= target_sum / total                # normalise sum to sqrt(2n)

            # ----- convolution (FFT) ----------------------------------------------------
            A = np.fft.rfft(a, n=fft_len)
            conv = np.fft.irfft(A * A, n=fft_len)[: 2 * n - 1]
            max_conv = conv.max()                  # because sum is fixed, eval == max_conv

            # ----- keep the best --------------------------------------------------------
            if max_conv < best_score:
                best_score = max_conv
                best_seq = a.copy()
                print(f"[grid] new best score {best_score:.6f} (p={p:.2f}, t={t:.2f})")

    # --------------------------------------------------------------
    # 2) Smooth‑max gradient descent from the best grid point
    # --------------------------------------------------------------
    if best_seq is None:                            # fallback (should never happen)
        best_seq = np.full(n, target_sum / n)

    a = best_seq.copy()
    beta = 10.0                                     # temperature for smooth max
    lr = 0.05                                        # learning rate
    it = 0

    # pre‑allocate buffers for FFTs used in the gradient computation
    def grad_step(x, beta_val):
        """Return gradient of smooth‑max objective wrt x."""
        # convolution x*x
        X = np.fft.rfft(x, n=fft_len)
        conv = np.fft.irfft(X * X, n=fft_len)[: 2 * n - 1]

        # smooth max weights
        mx = conv.max()
        w = np.exp(beta_val * (conv - mx))
        w /= w.sum()                               # normalised weights, sum == 1

        # gradient = 2 * (w * x)  (cross‑correlation)
        # compute via convolution: grad_i = 2 * Σ w_{i+j} x_j
        w_full = w
        x_rev = x[::-1]                            # reversed x
        W = np.fft.rfft(w_full, n=fft_len)
        Xr = np.fft.rfft(x_rev, n=fft_len)
        tmp = np.fft.irfft(W * Xr, n=fft_len)[: (2 * n - 1) + n - 1]
        # extract the needed slice [n‑1 … n‑1+n‑1]
        grad = 2.0 * tmp[n - 1 : n - 1 + n]
        return grad

    while time.time() < deadline and it < 4000:
        # ----- gradient step -------------------------------------------------
        g = grad_step(a, beta)
        a -= lr * g
        a[a < 0.0] = 0.0                            # enforce non‑negativity

        # renormalise to the fixed L1‑norm
        s = a.sum()
        if s == 0.0:
            a = np.full(n, target_sum / n)
            s = target_sum
        a *= target_sum / s

        # evaluate current candidate (exact evaluation)
        X = np.fft.rfft(a, n=fft_len)
        conv = np.fft.irfft(X * X, n=fft_len)[: 2 * n - 1]
        cur_score = conv.max()
        if cur_score < best_score:
            best_score = cur_score
            best_seq = a.copy()
            print(f"[gd] it {it:04d} – new best {best_score:.6f}")

        # schedule: slowly increase beta (sharper max) and decay lr
        if it % 200 == 0 and it > 0:
            beta = min(beta * 1.1, 200.0)
            lr *= 0.95

        # early stop if we already beat the target
        if best_score <= 1.503:
            print(f"Target reached (score={best_score:.6f}) at iteration {it}")
            break

        it += 1

    # final sanity check using the provided evaluate_sequence (optional)
    # The function may clip values to [0,1000]; our numbers are far below that.
    # We simply return the best sequence found.
    return best_seq.tolist()
