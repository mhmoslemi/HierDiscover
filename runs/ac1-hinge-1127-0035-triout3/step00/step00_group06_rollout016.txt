"""
Propose a sequence of non‑negative heights that minimizes the given evaluation
function.  The algorithm is a simple implementation of the iterative LP
procedure described in the statement:

*  Start from a random (or the provided) sequence and normalise its sum to
   √(2 n) (the evaluation then equals the maximum of the self‑convolution).
*  Repeatedly solve a linear programme that, for the current sequence `f`,
   finds a non‑negative sequence `g₀` with the largest possible total weight
   while keeping the mixed convolution `f * g₀` bounded by the current maximum
   `M = max(f*f)`.
*  Rescale `g₀` to the same normalisation as `f` (call it `g`) and try a few
   convex combinations `f←(1‑t)·f + t·g`.  The combination is accepted only if
   the new maximum of `f*f` does not increase; this guarantees a monotone
   decrease of the evaluation.
*  If the LP is infeasible or no improving step is found, a small random
   perturbation is applied to escape possible stagnation.
*  The whole process is repeated for several random restarts until the time
   budget (≈ 1000 s) is exhausted.  The best sequence seen is returned.

The code uses `scipy.optimize.linprog` with the HiGHS solver (fast for the
moderate sizes used, n≈400‑800).  All operations are vectorised, and the
linear‑programming matrix is built efficiently using slice assignments.
"""

import time
import numpy as np
from scipy import optimize

# ----------------------------------------------------------------------
# Linear programming sub‑routine: maximise sum(g) under  f*g ≤ M  and g≥0
# ----------------------------------------------------------------------
def solve_convolution_lp(f_vec: np.ndarray, bound: float):
    """Solve:
          maximise   Σ g_i
          subject to conv(f, g)_k ≤ bound  for all k
                     g_i ≥ 0
       Returns the optimal g (as a 1‑D np.ndarray) or None if infeasible.
    """
    n = f_vec.shape[0]
    # Objective: minimise -Σ g_i   (linprog minimises)
    c = -np.ones(n)

    # Build the convolution constraints:   A @ g ≤ bound
    # For each k (0 … 2n‑2) the coefficient of g_j is f_{k‑j} if 0≤k‑j<n else 0.
    A = np.zeros((2 * n - 1, n), dtype=np.float64)
    # Each column j contributes the vector f shifted by +j rows.
    for j in range(n):
        A[j:j + n, j] = f_vec  # slice assignment, fast and memory‑efficient

    # Append the non‑negativity constraints g ≥ 0  →  –g ≤ 0
    A_ub = np.vstack([A, -np.eye(n)])
    b_ub = np.concatenate([np.full(2 * n - 1, bound, dtype=np.float64), np.zeros(n)])

    # HiGHS is the default, fast for these moderate sizes.
    res = optimize.linprog(c, A_ub=A_ub, b_ub=b_ub, method="highs")
    if res.success:
        return res.x
    return None

# ----------------------------------------------------------------------
def propose_candidate(budget_s: int = 1000, seed: int = 0) -> list[float]:

    Return a list of non‑negative heights that (hopefully) beats the target

    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 2.0   # a small safety margin

    best_seq = None
    best_score = np.inf

    # ------------------------------------------------------------------
    # Helper: normalise a vector so that its L¹‑norm equals √(2 n)
    # ------------------------------------------------------------------
    def normalise(vec):
        n = vec.shape[0]
        total = np.sum(vec)
        if total <= 0:
            # avoid degenerate case – fall back to uniform
            return np.full(n, np.sqrt(2 * n) / n)
        return vec * (np.sqrt(2 * n) / total)

    # ------------------------------------------------------------------
    # Main outer loop: several random restarts (different lengths)
    # ------------------------------------------------------------------
    while time.time() < deadline:
        # Choose a problem size (larger sizes give a lower theoretical bound)
        n = np.random.randint(300, 801)          # 300 ≤ n ≤ 800
        # Initialise either from the provided construction or at random
        if np.random.rand() < 0.3 and "height_sequence_1" in globals():

            seq0 = np.array(height_sequence_1, dtype=np.float64)

            f = normalise(seq0)
        else:
            # Random positive start (Dirichlet‑like)

        cur_score = M                     # because sum is normalised
        if cur_score < best_score:

        # ----------------------------------------------------------------

        # ----------------------------------------------------------------
        # Step sizes tried (larger values allow faster movement when safe)
        t_candidates = [0.5, 0.2, 0.1, 0.05, 0.02, 0.01, 0.005]
        no_improve_iters = 0
        max_inner = 200                # hard cap per restart
        for it in range(max_inner):
            if time.time() >= deadline:

            # Solve the LP to obtain a direction g₀
            g0 = solve_convolution_lp(f, M)
            if g0 is None:

                idx = np.random.randint(n)
                f[idx] = max(0.0, f[idx] + np.random.randn() * 0.01 * f[idx])

                continue

            improved = False
            for t in t_candidates:

                # Re‑normalise to keep the sum exactly √(2 n)
                f_new = normalise(f_new)

                if M_new < M - 1e-12:          # strict improvement

                    break

                idx = np.random.randint(n)
                f[idx] = max(0.0, f[idx] + np.random.randn() * 0.005 * f[idx])
                f = normalise(f)
                M = np.convolve(f, f).max()

            # Record the best sequence seen so far
            cur_score = M
            if cur_score < best_score:
                best_score = cur_score
                best_seq = f.tolist()
                print(f"[{time.time():.1f}] New best score {best_score:.6f} (n={n})")

    # If nothing was found (unlikely) fall back to a simple uniform sequence

        print("Fallback to uniform sequence.")

    return best_seq
