        seq = [(i+1) ** (-p) for i in range(n)]

            best_score, best_seq = score, seq

    current_score = best_score

        i, j = np.random.randint(n, size=2)
        if i == j: continue

        new_a = a.copy()

"""
propose_candidate

Search for a sequence of non‑negative heights that minimizes the evaluation
function

    2 * n * max(conv(a, a)) / (sum(a) ** 2)

by exploiting the analytical insight that a power‑law shape
    a[i] = (i + 1) ** (-p)
produces a near‑constant self‑convolution for p ≈ 0.5 and reduces the
objective for p a little larger (≈0.58–0.60).  

The routine enumerates a few sequence lengths, sweeps a coarse grid of the
exponent p, then refines around the best parameters.  A short stochastic
local‑search phase (random mass transfers between two coordinates) is used
to polish the best candidate found before the time budget expires.

All evaluations are performed with a lightweight direct implementation of the
score (identical to ``evaluate_sequence`` for the admissible ranges used
here), avoiding the extra security checks and keeping the search fast.
"""

import time
import numpy as np

def _score(seq):
    """
    Compute the score 2 * n * max(conv(seq, seq)) / (sum(seq) ** 2)
    This is equivalent to evaluate_sequence for non‑negative, sub‑1000
    inputs, but skips the input validation for speed.
    """
    arr = np.asarray(seq, dtype=np.float64)
    # convolution (full length)
    conv = np.convolve(arr, arr)
    max_conv = np.max(conv)
    s = np.sum(arr)
    # protect against division by zero (should never happen here)
    if s < 1e-12:
        return float('inf')
    return float(2.0 * len(arr) * max_conv / (s * s))


def _make_powerlaw(n, p):
    """Return a list a[i] = (i+1) ** (-p) for i = 0 .. n-1."""
    # using Python list comprehension is fast enough for the size we use
    return [(i + 1) ** (-p) for i in range(n)]


def propose_candidate(seed=0, budget_s=1000, **kwargs):
    """
    Search for a low‑score step‑function.
    Parameters
    ----------
    seed : int
        Random seed for reproducibility.
    budget_s : float
        Time budget in seconds.
    **kwargs : dict
        Optional keyword ``lengths`` – a list of lengths to try.
    Returns
    -------
    list[float]
        The best sequence found within the time budget.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5   # small safety margin

    # ------------------------------------------------------------------
    # 1. Grid search over lengths and exponents
    # ------------------------------------------------------------------
    length_candidates = kwargs.get(
        "lengths",
        [30, 50, 100, 200, 400, 800, 1500, 3005]
    )
    best_seq = None
    best_score = float('inf')
    best_len = None
    best_p = None

    for n in length_candidates:
        # coarse sweep of exponent p
        for p in np.linspace(0.50, 0.78, 31):   # step ≈0.009
            seq = _make_powerlaw(n, p)
            sc = _score(seq)
            if sc < best_score:
                best_score, best_seq, best_len, best_p = sc, seq, n, p

        # finer sweep around the current best p for this length
        if best_len == n:
            lo = max(0.5, best_p - 0.03)
            hi = min(0.78, best_p + 0.03)
            for p in np.linspace(lo, hi, 41):  # step ≈0.0015
                seq = _make_powerlaw(n, p)
                sc = _score(seq)
                if sc < best_score:
                    best_score, best_seq, best_len, best_p = sc, seq, n, p

        if time.time() > deadline:
            break

    # ------------------------------------------------------------------
    # 2. Simple stochastic refinement (mass‑transfer hill climbing)
    # ------------------------------------------------------------------
    if best_seq is None:
        # fallback – uniform sequence (should never happen)
        best_seq = [1.0] * 10
        best_score = _score(best_seq)

    a = np.array(best_seq, dtype=np.float64)
    n = len(a)
    cur_score = best_score
    # step size is a fraction of the current maximal entry
    step = 0.02 * max(a) if max(a) > 0 else 0.01

    while time.time() < deadline:
        # pick two distinct indices and transfer a small amount of mass
        i, j = np.random.randint(n, size=2)
        if i == j:
            continue
        delta = (np.random.rand() - 0.5) * step
        new_a = a.copy()
        new_a[i] = max(0.0, new_a[i] + delta)
        new_a[j] = max(0.0, new_a[j] - delta)

        # evaluate the perturbed sequence
        new_score = _score(new_a.tolist())
        if new_score < cur_score:
            a, cur_score = new_a, new_score
            if new_score < best_score:
                best_score, best_seq, best_len, best_p = new_score, new_a.tolist(), n, None

        # slowly anneal the step size
        step *= 0.99995
        if step < 1e-8:
            step = 0.02 * max(a)  # reset if it collapses

    # ------------------------------------------------------------------
    # 3. Report and return the best sequence discovered
    # ------------------------------------------------------------------
    print(f"Best score   : {best_score:.6f}")
    print(f"Best length  : {best_len}")
    if best_p is not None:
        print(f"Best exponent: {best_p:.6f}")
    else:
        print("Best exponent: (refined by hill‑climbing)")

    # Ensure the sequence is a plain Python list of floats
    return [float(x) for x in best_seq]
