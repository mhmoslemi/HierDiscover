def solve_convolution_lp(f_seq, rhs):
    n = len(f_seq)
    # objective: maximize sum(b) => minimize -sum(b)
    c = -np.ones(n)
    # Build A_ub constraints for each convolution index
    rows = []
    b = []
    for k in range(2 * n - 1):
        row = np.zeros(n)
        # i ranges from max(0, k-(n-1)) to min(n-1, k)
        i_start = max(0, k - (n - 1))
        i_end = min(n - 1, k)
        # for each i, j = k - i
        # increment row[j] by f_seq[i]
        row[k - np.arange(i_start, i_end + 1)] = f_seq[i_start:i_end + 1]
        rows.append(row)
        b.append(rhs)
    # non‑negativity constraints: -b <= 0   =>   rows for -I
    A_ub = np.vstack([rows, -np.eye(n)])
    b_ub = np.hstack([b, np.zeros(n)])
    # Solve
    result = linprog(c, A_ub=A_ub, b_ub=b_ub, method='highs')
    if result.success:
        return result.x
    return None

"""
Propose a candidate step‑function (list of non‑negative heights) that minimizes
the provided ``evaluate_sequence`` function.

The search runs in three phases inside the 1000 s budget:

1. **Coarse grid search** – generate simple power‑law sequences
   a_i = (i+1)^(-α) for several exponents α.  The best of these
   (together with a uniform sequence and, if available, the previous
   best construction ``height_sequence_1``) is taken as a starting point.

2. **Linear‑programming driven improvement** – following the idea from the
   reference code, we solve a small LP that finds a direction ``g`` with a
   larger ℓ¹‑norm while keeping the auto‑convolution bounded by the current
   maximum.  A few mixing fractions *t* are tried and the best improvement
   is kept.  This step is repeated until time runs out.

3. **Final random walk** – a cheap simulated‑annealing style perturbation
   explores the remaining neighbourhood and may escape tiny local minima.

All operations respect the security checks inside ``evaluate_sequence``.
The function returns the best sequence found before the deadline.
"""

import time
import numpy as np
from scipy.optimize import linprog

# ----------------------------------------------------------------------
# Helper functions
# ----------------------------------------------------------------------


def generate_power_sequence(alpha: float, n: int) -> list[float]:
    """Return a non‑negative sequence a_i = (i+1)^(-alpha) for i=0,…,n‑1."""
    idx = np.arange(1, n + 1, dtype=float)
    return (idx ** (-alpha)).tolist()


def solve_convolution_lp(f_seq: list[float], rhs: float):
    """
    Solve the linear programme:
        maximise   Σ b_j
        subject to conv(f_seq, b) ≤ rhs  (for every convolution index)
                  b ≥ 0

    Returns the optimal ``b`` as a NumPy array or ``None`` if the solver fails.
    """
    n = len(f_seq)
    # maximise sum(b)  → minimise -sum(b)
    c = -np.ones(n)

    # Build A_ub and b_ub for the convolution inequalities.
    a_rows = []
    b_vals = []
    for k in range(2 * n - 1):
        row = np.zeros(n)
        i_min = max(0, k - (n - 1))
        i_max = min(n - 1, k)
        for i in range(i_min, i_max + 1):
            j = k - i
            row[j] = f_seq[i]
        a_rows.append(row)
        b_vals.append(rhs)

    # Non‑negativity: -b ≤ 0
    a_ub = np.vstack([np.array(a_rows), -np.eye(n)])
    b_ub = np.hstack([np.array(b_vals), np.zeros(n)])

    # Solve with the high‑performance HiGHS backend.
    res = linprog(c, A_ub=a_ub, b_ub=b_ub, method="highs")
    if res.success:
        return res.x
    return None

# ----------------------------------------------------------------------

    """

    # ------------------------------------------------------------------

    # but small enough for fast LP solves.
    n = 800

        seq = generate_power_sequence(alpha, n)
        score = evaluate_sequence(seq)

    except NameError:

    # make a mutable copy for the optimisation loop
    curr_seq = np.array(best_seq, dtype=float)

    # ------------------------------------------------------------------
    # Phase 2 – LP‑based improvement loop
    # ------------------------------------------------------------------
    while time.time() < deadline:
        n = len(curr_seq)
        total = float(np.sum(curr_seq))
        if total < 1e-12:  # degenerate case – shouldn't happen
            break

        # normalise to the convention used by the LP (ℓ¹‑norm = sqrt(2n))
        norm_factor = np.sqrt(2.0 * n) / total

        sum_g = float(np.sum(g))
