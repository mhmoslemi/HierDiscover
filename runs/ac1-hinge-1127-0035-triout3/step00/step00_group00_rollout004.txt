def solve_convolution_lp(f_seq, rhs, n):
    # f_seq is numpy array length n
    c = -np.ones(n)  # maximize sum => minimize -sum
    # a_ub rows: for each k in 0..2n-2: conv(f,b)[k] <= rhs
    # Each row: coefficients b_j = f_i where i + j = k
    rows = []
    ub = []
    for k in range(2*n - 1):
        row = np.zeros(n)
        i_min = max(0, k-(n-1))
        i_max = min(n-1, k)
        # i from i_min to i_max
        # indices j = k - i
        # set row[j] = f_seq[i]
        # we can vectorize but simple loops fine.
        for i in range(i_min, i_max+1):
            j = k - i
            row[j] = f_seq[i]
        rows.append(row)
        ub.append(rhs)
    # add nonneg constraints: -I * b <= 0
    rows.append(-np.eye(n))
    ub.append(np.zeros(n))
    # add upper bound constraints b <= 1000: I * b <= 1000
    rows.append(np.eye(n))
    ub.append(np.full(n, 1000.0))

    b_ub = np.concatenate(ub)
    # solve
    result = opt.linprog(c, A_ub=A_ub, b_ub=b_ub, method='highs', options={'presolve': True, 'disp': False})
    if result.success:
        return result.x
    else:
        return None

"""
Propose a step‑function (list of non‑negative heights) with a low
evaluate_sequence score.

The algorithm is an iterative improvement that alternates two phases:

1. **Normalization** – the current sequence is scaled to the standard
   sum `sqrt(2*n)`.  With this scaling the evaluation reduces to
   `max(conv(a,a))`.

2. **Linear‑programming direction** – for the normalized sequence `f`
   we solve a linear program

        maximise   Σ b_j
        subject to  (f * b)[k] ≤ M   for all k
                    0 ≤ b_j ≤ 1000

   where `M = max( f * f )`.  The LP finds a non‑negative direction `b`
   that can increase the total mass while keeping the current worst
   convolution value `M` as a bound.

3. **Blending** – `f` and the normalised LP solution `g` are mixed
   with several trial mixing coefficients `t ∈ {0.1,0.3,0.5,0.7,0.9}`.
   The mixed sequence with the smallest evaluation (after clipping to
   `[0,1000]`) is kept.  If none improves the score a tiny random
   perturbation is applied.

The process is repeated for a few different start‑up shapes
(geometric decay, uniform, random) and for a few problem sizes
(`n = 200, 400, 800`).  The best sequence found within the time budget
(≈ 1000 s) is returned.

All heavy work (convolution, LP) is done with NumPy / SciPy and the
search stays well inside a single CPU core.
"""

import time
import numpy as np
from scipy import optimize

# ----------------------------------------------------------------------
# Helper utilities (exactly the same logic as evaluate_sequence)
def _eval(seq: np.ndarray) -> float:
    """Fast replica of evaluate_sequence (no security checks)."""
    a = np.clip(seq, 0.0, 1000.0)
    total = a.sum()
    if total < 0.01:
        return np.inf
    conv = np.convolve(a, a)
    return float(2.0 * a.size * conv.max() / (total * total))

# ----------------------------------------------------------------------
def _solve_lp(f: np.ndarray, bound: float) -> np.ndarray | None:
    """
    Solve   maximise Σ b_j   s.t.   (f * b)[k] ≤ bound   and   0 ≤ b_j ≤ 1000.
    Returns the optimal b (numpy array) or None on failure.
    """
    n = f.size
    # objective: minimise -Σb  (equivalent to maximise Σb)
    c = -np.ones(n)

    # build the convolution constraints  (f * b)[k] ≤ bound
    rows = []
    rhs = []
    for k in range(2 * n - 1):
        row = np.zeros(n)
        i_min = max(0, k - (n - 1))
        i_max = min(n - 1, k)
        # i runs over indices of f that contribute to (f*b)[k]
        # then j = k - i is the index of b
        for i in range(i_min, i_max + 1):
            j = k - i
            row[j] = f[i]
        rows.append(row)
        rhs.append(bound)

    #  b ≥ 0  ->  -b ≤ 0
    rows.append(-np.eye(n))
    rhs.append(np.zeros(n))

    #  b ≤ 1000  ->  b ≤ 1000
    rows.append(np.eye(n))
    rhs.append(np.full(n, 1000.0))

    A_ub = np.vstack(rows)
    b_ub = np.concatenate(rhs)

    # use the modern HiGHS solver – it is fast and robust
    res = optimize.linprog(
        c,
        A_ub=A_ub,
        b_ub=b_ub,
        method="highs",
        # quiet options
        options={"presolve": True, "dual_feasible": True, "time_limit": 5.0},
    )
    if res.success:
        return res.x
    return None

# ----------------------------------------------------------------------

    """
