import numpy as np
import time

def propose_candidate(seed: int = 0, budget_s: float = 1000, **kwargs):
    """Heuristic search for a sequence that minimises evaluate_sequence.

    The algorithm starts from the discrete analogue of the continuous function
    f(x) = 1/sqrt(x) (i.e. a_i ∝ 1/√(i+1)).  This choice makes the
    self‑convolution almost flat, giving a value ≈π/2 ≈ 1.57, well below the
    uniform benchmark of 2.  From this starting point a simple simulated‑annealing
    procedure moves a tiny amount of mass from indices that contribute most to
    the current maximal convolution towards indices that contribute least.
    The sum of the coefficients is kept equal to one (the scale does not affect
    the objective because the evaluation function is homogeneous).  The best
    sequence found before the time budget expires is returned.
    """
    np.random.seed(seed)
    start = time.time()
    deadline = start + budget_s - 0.2   # keep a little margin

    # --- parameters -------------------------------------------------
    n = 1000                # length of the step‑function
    # Initial shape: a_i ∝ 1/√(i+1)
    i = np.arange(n, dtype=float)
    a = 1.0 / np.sqrt(i + 1.0)
    a = a / a.sum()         # normalise so that sum(a) == 1
    # Evaluate the starting point
    best_seq = a.copy()
    best_score = evaluate_sequence(best_seq.tolist())
    curr_seq = a.copy()
    curr_score = best_score

    # Simulated‑annealing parameters
    temperature = 0.2
    temp_decay = 0.99995   # exponential cooling
    min_temp = 1e-6

    # Pre‑compute indexes for faster random choice
    indices = np.arange(n)

    # Main loop -------------------------------------------------------
    while time.time() < deadline:
        # pick a donor index proportionally to current mass (more mass → more likely)
        donor = np.random.choice(indices, p=curr_seq / curr_seq.sum())
        # pick a recipient index uniformly (different from donor)
        recipient = donor
        while recipient == donor:
            recipient = np.random.randint(n)

        # amount to move: up to 5 % of the donor mass, randomised
        max_move = curr_seq[donor] * 0.05
        if max_move <= 0:
            continue
        delta = np.random.rand() * max_move

        # propose new sequence
        proposal = curr_seq.copy()
        proposal[donor] -= delta
        proposal[recipient] += delta

        # evaluate
        try:
            score = evaluate_sequence(proposal.tolist())
        except Exception:
            continue  # safety – should not happen

        # Metropolis acceptance
        accept = False
        if score < curr_score:
            accept = True
        else:
            # probability to accept a worse move
            if temperature > min_temp:
                prob = np.exp(-(score - curr_score) / temperature)
                if np.random.rand() < prob:
                    accept = True

        if accept:
            curr_seq = proposal
            curr_score = score
            # keep the best ever seen
            if score < best_score:
                best_score = score
                best_seq = proposal.copy()
                # occasional progress report
                print(f"[{time.time() - start:6.1f}s] New best: {best_score:.6f}")

        # cool down
        temperature = max(min_temp, temperature * temp_decay)

    # Return the best sequence as a plain list of Python floats
    return best_seq.tolist()

"""
heuristic_step_function.py

A heuristic search for a non‑negative step‑function that minimises

    2 * n * max(conv(a, a)) / (sum(a) ** 2)

where ``conv`` is the full linear convolution (``np.convolve``).  Theory shows that
the uniform sequence gives a value of 2, while the continuous function
f(x)=1/(2√x) (discretised as a_i ∝ 1/√(i+1)) makes the self‑convolution almost
flat and leads to a value ≈π/2≈1.57.  Starting from this “square‑root” shape we
apply a cheap simulated‑annealing / hill‑climbing procedure that repeatedly
moves a small amount of mass from indices that contribute most to the current
maximal convolution towards indices that contribute least.  The sum of the
coefficients is kept equal to 1 (the scale does not affect the objective) and
all coefficients stay non‑negative.  The best sequence found before the time
budget expires is returned.

The algorithm runs for at most ``budget_s`` seconds (default 1000) and prints
intermediate improvements.
"""

import time
import numpy as np

def propose_candidate(seed: int = 0, budget_s: float = 1000, **kwargs):
    """
    Return a list of non‑negative heights that (heuristically) yields a small
    value of ``evaluate_sequence``.

    Parameters
    ----------
    seed : int, optional
        Random seed for reproducibility.
    budget_s : float, optional
        Number of seconds allocated for the search.  The function stops a few
        seconds before the limit to avoid a forced termination.
    **kwargs
        Ignored – present only to match the expected signature.

    Returns
    -------
    list[float]
        The best sequence discovered.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5   # keep a small safety margin

    # ------------------------------------------------------------
    # 1)  Choose a length.  Too large makes each evaluation expensive,
    #     too small limits the achievable bound.  1000 works well.
    # ------------------------------------------------------------
    n = 1000

    # ------------------------------------------------------------
    # 2)  Initialise with the (discrete) square‑root shape:
    #     a_i ∝ 1 / sqrt(i+1).  This gives a value ≈π/2.
    # ------------------------------------------------------------
    i = np.arange(n, dtype=float)
    a = 1.0 / np.sqrt(i + 1.0)
    a /= a.sum()                     # normalise so that sum(a) == 1

    # Keep track of the best sequence found.
    best_seq = a.copy()
    try:
        best_score = evaluate_sequence(best_seq.tolist())
    except Exception:               # safety – should never happen
        best_score = float('inf')
    curr_seq = a.copy()
    curr_score = best_score

    print(f"[{0:.1f}s] start – length={n}, initial score={best_score:.6f}")

    # ------------------------------------------------------------
    # 3)  Simple simulated‑annealing / hill‑climbing loop.
    #     At each step we look at the index k where the convolution is
    #     maximal, compute how much each a_i participates in that term,
    #     and move a tiny amount of mass from the worst (largest)
    #     contributor to the best (smallest) contributor.
    # ------------------------------------------------------------
    temperature = 0.2
    temp_decay = 0.99996
    min_temp = 1e-7
    iteration = 0

    while time.time() < deadline:
        iteration += 1

        # ----- convolution and the index of the current maximum ----------
        conv = np.convolve(curr_seq, curr_seq)
        k_max = int(np.argmax(conv))

        # ----- contribution of each coefficient to conv[k_max] ----------
        #   conv[k_max] = Σ a_i * a_{k_max-i}
        #   derivative w.r.t. a_i is 2 * a_{k_max-i} (if the partner index is valid)
        contrib = np.zeros_like(curr_seq)
        # partner index j = k_max - i
        j = k_max - np.arange(n)
        mask = (j >= 0) & (j < n)
        contrib[mask] = 2.0 * curr_seq[j[mask]]

        # ----- choose donor (largest contribution) and receiver (smallest) ----------
        donor = int(np.argmax(contrib))
        receiver = int(np.argmin(contrib))

        if donor == receiver:
            # nothing to move – fall back to a tiny random perturbation
            donor = int(np.random.randint(n))
            receiver = int(np.random.randint(n))
            while receiver == donor:
                receiver = int(np.random.randint(n))

        # ----- how much mass to transfer?  A few percent of the donor is enough ----------
        max_transfer = min(curr_seq[donor] * 0.08, 0.02)   # 0.02 is an absolute cap
        if max_transfer <= 0:
            continue
        delta = np.random.rand() * max_transfer

        proposal = curr_seq.copy()
        proposal[donor] -= delta
        proposal[receiver] += delta

        # (non‑negativity is guaranteed by the construction)
        # No need to renormalise – the total sum stays exactly 1.

        # ----- evaluate the proposal ------------------------------------
        try:
            new_score = evaluate_sequence(proposal.tolist())
        except Exception:
            # any unexpected failure – reject this move
            continue

        # ----- acceptance test -------------------------------------------
        accept = False
        if new_score < curr_score:
            accept = True
        else:
            # probabilistic acceptance of a worse move (simulated annealing)
            if temperature > min_temp:
                prob = np.exp(-(new_score - curr_score) / temperature)
                if np.random.rand() < prob:
                    accept = True

    # ------------------------------------------------------------
    print(f"Finished after {time.time() - start_time:.1f}s – best score {best_score:.6f}")
