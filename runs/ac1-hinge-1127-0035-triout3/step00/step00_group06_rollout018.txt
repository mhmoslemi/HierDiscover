def flatten_step(seq, step=0.01):
    n = len(seq)
    conv = np.convolve(seq, seq)
    k = np.argmax(conv)
    # contributions = [seq[i] * seq[k-i] if within bounds else 0]
    contrib = np.zeros(n)
    for i in range(n):
        j = k - i
        if 0 <= j < n:
            contrib[i] = seq[i] * seq[j]
    # find i_max where contrib max, i_min where contrib min (exclude i where both indices same maybe)
    i_max = np.argmax(contrib)
    i_min = np.argmin(contrib)  # may be zero contributions
    # make delta as small fraction of seq[i_max] and seq[j_max]
    j_max = k - i_max
    j_min = k - i_min
    # ensure indices in range
    if not (0 <= j_max < n) or not (0 <= j_min < n):
        return seq
    delta = step * min(seq[i_max], seq[j_max])
    # ensure delta >0
    if delta <= 0:
        return seq
    new_seq = seq.copy()
    new_seq[i_max] = max(0.0, new_seq[i_max] - delta)
    new_seq[j_max] = max(0.0, new_seq[j_max] - delta)
    new_seq[i_min] += delta
    new_seq[j_min] += delta
    return new_seq

"""
Simple stochastic optimizer for the “max‑auto‑convolution” problem.

The objective is 2·max(conv(a,a)) / (sum(a))**2.
All numbers must be non‑negative; scaling does not change the score.

The search starts from a cheap analytical construction
(two‑level “high‑low” shape, which already beats the uniform 2.0 bound)
and then performs many small random perturbations.
Occasionally a fresh random descending vector is tried to escape local minima.
The best sequence found before the time budget expires is returned.
"""
import time
import numpy as np


def _two_level_sequence(n, alpha=2.0 / 3.0):
    """
    Two‑level construction with high block of length α·n and low block
    of height α (the analytically optimal balance for this family).
    """
    high_len = int(round(alpha * n))
    low_len = n - high_len
    low_val = alpha                     # low height = α·high (high height = 1)
    seq = np.empty(n, dtype=float)
    seq[:high_len] = 1.0
    seq[high_len:] = low_val
    return seq


def _random_descending_sequence(n):
    """
    Random descending sequence – draws uniform numbers, sorts them descending.
    This creates many small values and a few larger ones, a shape that tends
    to flatten the convolution.
    """
    vals = np.random.rand(n)
    vals.sort()
    return vals[::-1]                   # descending


def _random_exponential_sequence(n):
    """
    Geometric decay a_i = r**i with r close to 1.
    """
    r = 0.90 + 0.0999 * np.random.rand()      # r ∈ (0.9, 0.9999)
    return np.power(r, np.arange(n, dtype=float))


def _perturb(seq, step_scale=0.05):
    """
    Produce a nearby candidate by scaling a few entries.
    The sequence is re‑sorted descending after the perturbation,
    because a monotone decreasing shape is beneficial for flattening.
    """
    cand = seq.copy()
    n = len(cand)

    # pick a handful of indices to modify
    for _ in range(np.random.randint(1, 4)):
        i = np.random.randint(n)
        # multiplicative log‑normal perturbation
        factor = np.exp(step_scale * np.random.randn())
        cand[i] *= factor
        if cand[i] < 0.0:
            cand[i] = 0.0

    # keep the ordering monotone decreasing
    cand.sort()
    return cand[::-1]


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a step‑function (list of non‑negative heights) that minimises
    evaluate_sequence.  The function respects the supplied time budget and
    returns the best sequence discovered.
    """
    np.random.seed(seed)

    # length of the construction – use the existing global if present
    try:
        n = len(height_sequence_1)          # type: ignore[name-defined]
    except Exception:
        n = 2000

    # start from the analytically good two‑level shape
    best_seq = _two_level_sequence(n)
    best_score = evaluate_sequence(best_seq.tolist())

    # also keep a tiny copy of the current working sequence
    cur_seq = best_seq.copy()

    deadline = time.time() + max(0.0, budget_s - 5.0)   # a small safety margin

    while time.time() < deadline:
        r = np.random.rand()

        # Choose a generation strategy
        if r < 0.25:
            # a fresh random descending vector (restart)
            cand = _random_descending_sequence(n)
        elif r < 0.45:
            # geometric decay
            cand = _random_exponential_sequence(n)
        elif r < 0.60:
            # a small perturbation of the current best
            cand = _perturb(best_seq, step_scale=0.03)
        else:
            # perturb the current working sequence (explorative walk)
            cand = _perturb(cur_seq, step_scale=0.07)

        # evaluate – guard against any unexpected exception
        try:
            score = evaluate_sequence(cand.tolist())
        except Exception:

        # occasionally accept a better candidate as the new base for walks
        if score < evaluate_sequence(cur_seq.tolist()):
            cur_seq = cand.copy()

        # very rare full reseed to diversify search
        if np.random.rand() < 0.005:
            cur_seq = _random_descending_sequence(n)

    # return as plain Python list
    return best_seq.tolist()
