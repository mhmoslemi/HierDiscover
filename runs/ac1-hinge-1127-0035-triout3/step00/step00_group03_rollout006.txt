"""
Propose a step‑function (non‑negative list of heights) with a low
evaluation score.

The evaluation is 2 n·max(conv(a,a)) / (sum(a)²).  For a uniform
distribution the score is exactly 2.  The theoretical lower bound is
≈1 (the average value of the convolution), so a reduction from 2 to
≈1.5 is reachable.

The algorithm uses a simple projected gradient descent on a smooth
approximation of the “max” of the convolution.  The loss is
    L(a) = (1/α)·log Σ_k exp(α·(a*a)_k)     (soft‑max of the
    convolution values).  Its gradient is
    ∇L_i = 2· Σ_j a_j·σ_{i+j}
where σ is the soft‑max weight vector.  After each gradient step the
vector is projected onto the simplex (non‑negative and sum = 1).

Multiple random restarts of different lengths are performed until the
time budget expires; the best sequence discovered is returned.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper: project a non‑negative vector onto the simplex (sum = 1)
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Return the Euclidean projection of v onto the probability simplex."""
    if v.sum() == 0:
        # any uniform vector works
        return np.full_like(v, 1.0 / v.size)
    # standard O(n log n) simplex projection
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(len(v)) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # numerical safety – fall back to uniform
        return np.full_like(v, 1.0 / v.size)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    # guard against rounding‑error loss of mass
    if w.sum() == 0:
        w = np.full_like(v, 1.0 / v.size)
    else:
        w /= w.sum()
    return w


# ----------------------------------------------------------------------
def _softmax_gradient_step(a: np.ndarray, alpha: float, lr: float) -> np.ndarray:
    """
    One descent step on the soft‑max loss.
    a   – current sequence (already normalised to sum‑1)
    alpha – smoothing parameter (larger → closer to true max)
    lr  – learning rate
    Returns the updated (and re‑projected) sequence.
    """
    # convolution a*a (linear, not circular)
    b = np.convolve(a, a)                     # length = 2·n‑1
    # soft‑max weights: σ_k = exp(α·b_k) / Σ exp(α·b)
    expb = np.exp(alpha * b)
    sigma = expb / expb.sum()                 # still length 2·n‑1

    # gradient g_i = 2· Σ_j a_j·σ_{i+j}
    # compute efficiently via a convolution with the reversed sigma
    sigma_rev = sigma[::-1]                   # length 2·n‑1
    full = np.convolve(a, sigma_rev)          # length = n + (2·n‑1) -1 = 3·n‑2
    L = sigma.size
    grad = 2.0 * full[L - 1 - np.arange(a.size)]

    # ascent direction is +grad (since L grows with σ); we need descent
    a_new = a - lr * grad

    # keep feasible: non‑negative and sum = 1
    a_new = np.maximum(a_new, 0.0)
    a_new = _project_simplex(a_new)
    return a_new
