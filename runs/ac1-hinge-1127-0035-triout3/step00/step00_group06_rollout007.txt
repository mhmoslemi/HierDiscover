    u = np.sort(v)[::-1]

    return w

def propose_candidate(seed=0, budget_s=1000):
    import numpy as np, time
    np.random.seed(seed)
    n = len(height_sequence_1) if 'height_sequence_1' in globals() else 2000
    # Precompute FFT length
    fft_len = 1 << ((2*n - 1).bit_length())
    def compute_score_and_conv(p):
        fp = np.fft.rfft(p, n=fft_len)
        conv = np.fft.irfft(fp * fp, n=fft_len)[:2*n-1]
        max_val = conv.max()
        score = 2.0 * n * max_val
        return score, conv
    best_score = float('inf')
    best_p = None
    start = time.time()
    deadline = start + budget_s - 0.1
    # parameters
    init_step = 0.01  # step magnitude for gradient
    # Scheduling of temperature for SA acceptance
    init_temp = 0.01
    # restarts loop
    while time.time() < deadline:
        # Start a new random p
        p = np.random.dirichlet(np.ones(n))
        # compute initial score
        curr_score, curr_conv = compute_score_and_conv(p)
        # local search
        # we will limit inner iterations by time
        inner_start = time.time()
        while time.time() < deadline:
            # Choose between gradient step (30%) or random move (70%)
            if np.random.rand() < 0.3:
                # gradient descent step
                # find shift with max conv
                k_max = int(np.argmax(curr_conv))
                # gradient vector
                idx = k_max - np.arange(n)
                valid = (idx >= 0) & (idx < n)
                g = np.zeros(n)
                g[valid] = 2.0 * p[idx[valid]]
                # step size
                step = init_step * (1.0 - (time.time() - start) / budget_s)  # decreasing
                p_new = p - step * g
                # ensure nonnegative, project onto simplex
                p_new = np.maximum(p_new, 0.0)
                total = p_new.sum()
                if total > 0:
                    p_new = p_new / total
                else:
                    p_new = np.random.dirichlet(np.ones(n))
                new_score, new_conv = compute_score_and_conv(p_new)
            else:
                # random move
                i, j = np.random.randint(n, size=2)
                if i == j:
                    continue
                # move fraction from i to j
                fraction = np.random.rand() * 0.02  # up to 2% of p[i]
                amount = fraction * p[i]
                p_new = p.copy()
                p_new[i] -= amount
                p_new[j] += amount
                # due to floating, ensure nonneg
                # if p_new[i] <0 set to 0 and adjust sum
                if p_new[i] < 0:
                    p_new[i] = 0.0
                # normalize to sum=1 (should stay 1 but due to clamping adjust):
                total = p_new.sum()
                p_new = p_new / total
                new_score, new_conv = compute_score_and_conv(p_new)
            # acceptance
            # simple greedy acceptance with occasional random acceptance
            T = max(1e-4, init_temp * (deadline - time.time()) / budget_s)
            if new_score < curr_score or np.random.rand() < np.exp((curr_score - new_score) / T):
                p = p_new
                curr_score = new_score
                curr_conv = new_conv
                # record best
                if curr_score < best_score:
                    best_score = curr_score
                    best_p = p.copy()
                    print(f"[{time.time()-start:.1f}s] new best score: {best_score:.6f}")
            # occasional small random perturb to escape
            if np.random.rand() < 0.001:
                p = np.random.dirichlet(np.ones(n))
                curr_score, curr_conv = compute_score_and_conv(p)
        # after inner loop finishes due to time, break to outer
    # after search
    if best_p is None:
        # fallback: uniform
        best_p = np.full(n, 1.0 / n)
        best_score = 2.0 * n * (1.0 / n)  # 2
    # Scale to typical magnitude (sqrt(2n))
    scale = np.sqrt(2.0 * n) / best_p.sum()  # best_p.sum() ==1
    final_seq = (best_p * scale).tolist()
    print(f"Finished search. best score {best_score:.6f} (target 1.503).")
    return final_seq

    if 'height_sequence_1' in globals():
        n = len(height_sequence_1)          # typically 3306
    else:
        n = 2000

    # Raw arithmetic progression 1,2,…,n
    raw = np.arange(1, n + 1, dtype=np.float64)

    # Scale so that the sum equals sqrt(2 n) (the normalisation used in the
    # original formulation).  The scaling factor is:
    #   scale = sqrt(2 n) / Σ_{i=1}^n i   = sqrt(2 n) / (n (n+1)/2)
    sum_raw = raw.sum()                     # n·(n+1)/2
    scale = math.sqrt(2.0 * n) / sum_raw

    # Final sequence
    sequence = (raw * scale).tolist()

    # Optional: log the achieved score for verification.
    try:
        score = evaluate_sequence(sequence)
        print(f"[propose_candidate] length={n}, score={score:.6f}")
    except Exception as exc:
        print("[propose_candidate] evaluation failed:", exc)

    return sequence
