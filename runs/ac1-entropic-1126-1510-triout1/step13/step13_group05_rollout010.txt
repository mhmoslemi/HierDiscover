def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Euclidean projection of ``v`` onto the simplex {x≥0, Σx=s}."""
    if s < 0:
        raise ValueError("Simplex radius must be non‑negative")
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.where(u - cssv / (np.arange(1, len(v) + 1)) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    return np.maximum(v - theta, 0.0)


def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Full linear self‑convolution a*a using a real FFT (length 2·n‑1)."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1) - 1).bit_length()   # next power of two
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[: 2 * n - 1]


def _objective(a: np.ndarray) -> float:
    """
    Exact objective ``2·n·max(conv(a,a))`` for a vector whose entries sum to 1.
    """
    n = a.shape[0]
    return float(2.0 * n * _fft_self_convolve(a).max())

def _distribution_uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _distribution_arcsine(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    vals /= vals.sum()
    return vals


def _distribution_sine(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64) + 0.5
    w = np.sin(np.pi * i / n)
    w /= w.sum()
    return w


def _distribution_cosine(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64) + 0.5
    w = 1.0 - np.cos(np.pi * i / n)
    w /= w.sum()
    return w


def _distribution_quadratic(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    x = (2.0 * i) / (n - 1) - 1.0
    w = 1.0 - x * x
    w = np.clip(w, 0.0, None)
    w /= w.sum()
    return w


def _distribution_beta_sym(n: int, p: float) -> np.ndarray:
    """Symmetric Beta(p,p) on a regular grid."""
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    w = np.exp((p - 1) * np.log(x) + (p - 1) * np.log(1.0 - x))
    if w.sum() == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _distribution_beta_asym(n: int, p: float, q: float) -> np.ndarray:
    """Beta(p,q) – not forced to be symmetric."""
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    w = np.exp((p - 1) * np.log(x) + (q - 1) * np.log(1.0 - x))
    if w.sum() == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _distribution_edge_weighted(n: int, p: float) -> np.ndarray:
    """Edge‑weighted family: w_i ∝ ((i+½)*(n‑i‑½))^{‑p}."""
    i = np.arange(n, dtype=np.float64) + 0.5
    j = (n - np.arange(n, dtype=np.float64) - 0.5)
    w = (i * j) ** (-p)
    w = np.clip(w, 0.0, None)
    total = w.sum()
    if total > 0:
        w /= total
    else:
        w = np.full(n, 1.0 / n)
    return w


def _distribution_power_sum(n: int, p: float) -> np.ndarray:
    """w_i ∝ (i+1)^{‑p} + (n‑i)^{‑p} – another edge‑heavy family."""
    i = np.arange(1, n + 1, dtype=np.float64)
    w = i ** (-p) + (n + 1 - i) ** (-p)
    w = np.clip(w, 0.0, None)
    w /= w.sum()
    return w


def _distribution_random_sparse(
    n: int, rng: np.random.Generator, density: float = 0.01
) -> np.ndarray:
    k = max(2, int(n * density))
    idx = rng.choice(n, size=k, replace=False)
    vals = np.zeros(n, dtype=np.float64)
    vals[idx] = rng.random(k)
    if vals.sum() == 0.0:
        vals[0] = 1.0
    vals /= vals.sum()
    return vals


def _mix_with_uniform(base: np.ndarray, lam: float) -> np.ndarray:
    """Convex combination λ·uniform + (1‑λ)·base."""
    n = base.shape[0]
    uniform = np.full(n, 1.0 / n, dtype=np.float64)
    return lam * uniform + (1.0 - lam) * base

# Sub‑gradient of the max‑convolution ------------------------------------
def _max_convolution_subgradient(a: np.ndarray) -> np.ndarray:
    """
    Sub‑gradient of  max_k Σ_i a_i a_{k-i}.
    Returns a vector of the same shape as ``a``.
    """
    conv = _fft_self_convolve(a)
    max_val = conv.max()
    tol = 1e-12 * max_val
    active = np.where(conv >= max_val - tol)[0]
    n = a.shape[0]
    grad = np.zeros_like(a)
    for k in active:
        lo = max(0, k - (n - 1))
        hi = min(n - 1, k)
        idx = np.arange(lo, hi + 1)
        grad[idx] += 2.0 * a[k - idx]
    if len(active) > 0:
        grad /= len(active)          # average over the active set
    return grad

def _exp_gradient_descent(
    a0: np.ndarray,
    time_limit: float,
    best_seq: np.ndarray,
    best_score: float,
    eta0: float = 0.7,
    report_every: int = 400,
    max_iter: int = 200_000,
) -> Tuple[np.ndarray, float]:
    """
    Exponentiated‑gradient descent on the exact max‑convolution objective.
    """
    a = a0.copy()
    n = a.shape[0]
    t = 0
    while time.time() < time_limit and t < max_iter:
        t += 1
        grad = _max_convolution_subgradient(a)
        eta = eta0 / math.sqrt(t)
        a = a * np.exp(-eta * grad)
        s = a.sum()
        if s > 0:
            a = a / s
        else:                                   # safety fallback
            a = np.full(n, 1.0 / n, dtype=np.float64)

        if t % report_every == 0:
            cur = _objective(a)
            if cur < best_score - 1e-14:
                best_score = cur
                best_seq = a.copy()
                print(f"[{time.time():.1f}] exp‑gd it={t} new global best {best_score:.6f}")
    return best_seq, best_score

def _conv_constraint_matrix(a: np.ndarray) -> "scipy.sparse.csr_matrix":  # type: ignore
    """Return the (2n‑1)×n matrix that maps b↦conv(a,b). Sparse."""
    n = a.shape[0]
    L = 2 * n - 1
    # each diagonal offset j (0≤j<n) contains a (length n)
    data = [a] * n
    offsets = np.arange(n)
    return diags(data, offsets, shape=(L, n), format='csr')


def _lp_step(a: np.ndarray, M: float) -> Tuple[bool, np.ndarray]:
    """
    Solve   maximise Σ b_j   subject to   conv(a,b) ≤ M,  b≥0.
    Returns (success, solution b).  Uses SciPy's HiGHS if available.
    """
    if linprog is None:
        return False, a.copy()                 # LP not available
    n = a.shape[0]
    L = 2 * n - 1
    A = _conv_constraint_matrix(a)
    ub = np.full(L, M)
    c = -np.ones(n)                           # maximise Σb  ⇔ minimise -Σb
    bounds = [(0, None)] * n
    try:
        res = linprog(
            c,
            A_ub=A,
            b_ub=ub,
            bounds=bounds,
            method='highs',
            options={"presolve": True},
        )
    except Exception:                         # pragma: no cover
        return False, a.copy()
    if not res.success:
        return False, a.copy()
    return True, res.x


def _lp_refine(
    a0: np.ndarray,
    time_limit: float,
    best_seq: np.ndarray,
    best_score: float,
    max_steps: int = 8,
) -> Tuple[np.ndarray, float]:
    """
    Apply a few LP‑based refinement steps to ``a0``.
    Each step replaces a by a blend 0.5·a + 0.5·b_norm where b_norm
    is the normalised LP solution.
    """
    a = a0.copy()
    n = a.shape[0]
    for step in range(max_steps):
        if time.time() >= time_limit:
            break
        M = _fft_self_convolve(a).max()
        ok, b = _lp_step(a, M)
        if not ok:
            break
        sum_b = b.sum()
        if sum_b <= 1.0 + 1e-12:               # no improvement possible
            break
        b_norm = b / sum_b
        # blend and renormalise (the blend already sums to 1)
        a = 0.5 * a + 0.5 * b_norm
        a /= a.sum()
        cur = _objective(a)
        if cur < best_score - 1e-12:
            best_score = cur
            best_seq = a.copy()
            print(f"[{time.time():.1f}] LP step {step+1} new best {best_score:.6f}")
    return best_seq, best_score

# Greedy pairwise polishing (cheap hill‑climbing) ----------------------
def _pairwise_swap(
    a0: np.ndarray,
    time_limit: float,
    best_seq: np.ndarray,
    best_score: float,
    max_trials: int | None = None,
) -> Tuple[np.ndarray, float]:
    a = a0.copy()
    n = a.shape[0]
    trials = 0
    while time.time() < time_limit:
        if max_trials is not None and trials >= max_trials:
            break
        trials += 1

        src = np.where(a > 1e-14)[0]
        if src.size == 0:
            break
        i = np.random.choice(src)
        j = np.random.randint(n - 1)
        if j >= i:
            j += 1

        delta = a[i] * np.random.rand() * 0.45   # up to 45 % of source mass
        if delta < 1e-15:
            continue

        new_a = a.copy()
        new_a[i] -= delta
        new_a[j] += delta
        cur = _objective(new_a)
        if cur < best_score - 1e-14:
            a = new_a
            best_score = cur
            best_seq = a.copy()
            print(
                f"[{time.time():.1f}] swap i={i}→j={j} δ={delta:.6g} new best {best_score:.6f}"
            )
    return best_seq, best_score

    start = time.time()
    deadline = start + budget_s - 2.0                # safety margin

    overall_best_seq: np.ndarray | None = None
    overall_best_score = float("inf")
    top_candidates: List[Tuple[float, np.ndarray]] = []   # (score, seq)

    # ------------------------------------------------------------------
    # 1.  Generate a rich set of seed vectors (various lengths)
    # ------------------------------------------------------------------
    lengths = [1500, 2500, 3500, 4500]               # moderate – enough resolution
    p_grid = np.concatenate(
        [
            np.logspace(-2, -0.3, 10),   # 0.01 … 0.5
            np.linspace(0.5, 2.0, 12),
            np.linspace(2.0, 5.0, 8),
        ]
    )
    lam_grid = np.linspace(0.0, 1.0, 5)               # mixing with uniform

    for n in lengths:
        if time.time() >= deadline:
            break
        print(f"[{time.time():.1f}] scanning length {n}")

        # deterministic families -------------------------------------------------
        for base_fn in (
            _distribution_uniform,
            _distribution_arcsine,
            _distribution_sine,
            _distribution_cosine,
            _distribution_quadratic,
        ):
            cand = base_fn(n)
            sc = _objective(cand)
            top_candidates.append((sc, cand.copy()))
            if sc < overall_best_score:
                overall_best_score = sc
                overall_best_seq = cand.copy()
                print(f"[{time.time():.1f}] new global best len={n} score={sc:.6f}")

        # parametric families -------------------------------------------------
        for p in p_grid:
            if time.time() >= deadline:
                break

            edge_seq = _distribution_edge_weighted(n, p)
            beta_seq = _distribution_beta_sym(n, p)
            pow_seq = _distribution_power_sum(n, p)

            for base in (edge_seq, beta_seq, pow_seq):
                for lam in lam_grid:
                    if time.time() >= deadline:
                        break
                    cand = _mix_with_uniform(base, lam)
                    sc = _objective(cand)
                    top_candidates.append((sc, cand.copy()))
                    if sc < overall_best_score:
                        overall_best_score = sc
                        overall_best_seq = cand.copy()
                        print(
                            f"[{time.time():.1f}] new global best len={n} p={p:.3g} lam={lam:.2f} score={sc:.6f}"
                        )

        # asymmetric beta (extra flexibility) ----------------------------------
        for p in np.linspace(0.3, 3.0, 6):
            for q in np.linspace(0.3, 3.0, 6):
                if time.time() >= deadline:
                    break
                beta_asym = _distribution_beta_asym(n, p, q)
                for lam in lam_grid:
                    cand = _mix_with_uniform(beta_asym, lam)
                    sc = _objective(cand)
                    top_candidates.append((sc, cand.copy()))
                    if sc < overall_best_score:
                        overall_best_score = sc
                        overall_best_seq = cand.copy()
                        print(
                            f"[{time.time():.1f}] new global best asym‑beta n={n} p={p:.2f} q={q:.2f} lam={lam:.2f} score={sc:.6f}"
                        )

        # random Dirichlet seeds -------------------------------------------------
        for _ in range(5):
            if time.time() >= deadline:
                break
            alpha = rng.uniform(0.01, 2.0)
            a = rng.dirichlet(np.full(n, alpha))
            sc = _objective(a)
            top_candidates.append((sc, a.copy()))
            if sc < overall_best_score:
                overall_best_score = sc
                overall_best_seq = a.copy()
                print(f"[{time.time():.1f}] Dirichlet len={n} score={sc:.6f}")

        # sparse random seeds ----------------------------------------------------
        for _ in range(5):
            if time.time() >= deadline:
                break
            a = _distribution_random_sparse(n, rng, density=0.008)
            sc = _objective(a)
            top_candidates.append((sc, a.copy()))
            if sc < overall_best_score:
                overall_best_score = sc
                overall_best_seq = a.copy()
                print(f"[{time.time():.1f}] sparse len={n} score={sc:.6f}")

        # keep only the most promising ≈30 candidates overall
        top_candidates.sort(key=lambda x: x[0])
        top_candidates = top_candidates[:30]

    # ------------------------------------------------------------------
    # 2.  Incorporate any previously discovered construction (if present)
    # ------------------------------------------------------------------
    try:
        prev = np.asarray(height_sequence_1, dtype=np.float64)  # type: ignore[name-defined]
        if prev.ndim == 1 and prev.size > 0:
            prev = np.maximum(prev, 0.0)
            total = prev.sum()
            if total == 0.0:
                prev = _distribution_uniform(prev.size)
            else:
                prev = prev / total
            sc = _objective(prev)
            if sc < overall_best_score:
                overall_best_score = sc
                overall_best_seq = prev.copy()
                print(f"[{time.time():.1f}] previous construction score={sc:.6f}")
            top_candidates.append((sc, prev.copy()))
    except Exception:  # pragma: no cover
        pass

    # ------------------------------------------------------------------
    # 3.  Fallback in case nothing was generated
    # ------------------------------------------------------------------
    if overall_best_seq is None:
        n = 4000
        overall_best_seq = _distribution_uniform(n)
        overall_best_score = _objective(overall_best_seq)
        print(
            f"[{time.time():.1f}] fallback uniform length={n} score={overall_best_score:.6f}"
        )

    # ------------------------------------------------------------------
    # 4.  Refine the best seeds with (a) exponential GD, (b) LP steps
    # ------------------------------------------------------------------
    remaining = deadline - time.time()
    if remaining > 30.0 and top_candidates:
        # take the six most promising seeds
        seeds = [seq for _, seq in sorted(top_candidates, key=lambda x: x[0])[:6]]
        per_seed = remaining / max(1, len(seeds))
        print(f"[{time.time():.1f}] start refinement on {len(seeds)} seeds")
        for idx, seed_seq in enumerate(seeds):
            if time.time() >= deadline:
                break
            sg_deadline = time.time() + per_seed * 0.6   # GD gets ~60 %
            a_opt, sc_opt = _exp_gradient_descent(
                seed_seq,
                sg_deadline,
                best_seq=overall_best_seq,
                best_score=overall_best_score,
                eta0=0.7,
                report_every=250,
                max_iter=250_000,
            )
            if sc_opt < overall_best_score - 1e-12:
                overall_best_score = sc_opt
                overall_best_seq = a_opt.copy()
                print(
                    f"[{time.time():.1f}] after exp‑gd seed {idx} new global best {overall_best_score:.6f}"
                )
            # LP refinement – ≈40 % of the per‑seed budget
            lp_deadline = time.time() + per_seed * 0.4
            a_opt, sc_opt = _lp_refine(
                a_opt,
                lp_deadline,
                best_seq=overall_best_seq,
                best_score=overall_best_score,
                max_steps=6,
            )
            if sc_opt < overall_best_score - 1e-12:
                overall_best_score = sc_opt
                overall_best_seq = a_opt.copy()
                print(
                    f"[{time.time():.1f}] after LP seed {idx} new global best {overall_best_score:.6f}"
                )

    # ------------------------------------------------------------------
    # 5.  Final cheap polishing
    # ------------------------------------------------------------------
    if time.time() < deadline:
        print(f"[{time.time():.1f}] final greedy polishing")
        a_opt, sc_opt = _pairwise_swap(
            overall_best_seq.copy(),
            deadline,
            best_seq=overall_best_seq,
            best_score=overall_best_score,
        )
        if sc_opt < overall_best_score - 1e-12:
            overall_best_score = sc_opt
            overall_best_seq = a_opt.copy()
            print(f"[{time.time():.1f}] after final swaps new best {overall_best_score:.6f}")

    print(
        f"Final best score (2·n·max(conv)) : {overall_best_score:.6f}  (length={len(overall_best_seq)})"
    )
    # ensure plain list of Python floats for the caller
    return overall_best_seq.tolist()
