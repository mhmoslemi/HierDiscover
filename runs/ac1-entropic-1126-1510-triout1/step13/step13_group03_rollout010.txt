def _gradient_smooth(a, w):
    n = a.shape[0]
    # w length 2n-1
    w_rev = w[::-1]
    size = 1 << ((len(w_rev) + n - 1 - 1).bit_length())
    A = np.fft.rfft(a, n=size)
    W = np.fft.rfft(w_rev, n=size)
    conv = np.fft.irfft(A * W, n=size)
    # conv length size, but relevant part conv[:len(w_rev)+n-1]
    full = conv[: len(w_rev) + n - 1]
    grad = 2.0 * full[2*n-2: 2*n-2 + n]  # slice start = 2n-2
    return grad

def _smooth_grad_descent(
    a0, time_limit, best_seq, best_score,
    alpha_start=0.5, alpha_end=1e-4, num_anneal=20,
    lr=0.1, beta1=0.9, beta2=0.999,
    eval_every=50
):
    a = a0.copy()
    n = a.shape[0]
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    t = 0
    # Determine total iterations based on time limit maybe 2000?
    # We'll adapt schedule: each epoch will lower alpha.
    # Compute number of epochs = num_anneal.
    # We'll set iterations per epoch maybe 200 (or adapt to time).
    epoch_iters = 200
    total_iters_possible = int((time_limit - time.time()) / 0.001) # rough?
    # Instead just run while time remaining and schedule alpha decreasing linearly.
    while time.time() < time_limit:
        # Compute current alpha based on schedule (exponential)
        alpha = max(alpha_end, alpha_start * (0.5 ** (t // epoch_iters)))  # each epoch halve alpha.
        # compute conv and smooth weights
        conv = _fft_self_convolve(a)
        # compute softmax weights
        maxc = conv.max()
        # stable softmax
        shifted = conv - maxc
        # If alpha is near zero, treat w as indicator
        if alpha < 1e-12:
            # w = zeros except max indices, maybe multiple if ties.
            w = np.zeros_like(conv)
            w[conv == maxc] = 1.0 / np.sum(conv == maxc)
        else:
            exp_shift = np.exp(shifted / alpha)
            sumexp = exp_shift.sum()
            w = exp_shift / sumexp

        # compute gradient
        grad = _gradient_smooth(a, w)

        # Adam update
        t += 1
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad * grad)
        m_hat = m / (1 - beta1 ** t)
        v_hat = v / (1 - beta2 ** t)
        step = lr * m_hat / (np.sqrt(v_hat) + 1e-8)

        # gradient descent move (a ascend? we minimize, so subtract step)
        a = a - step

        # Project onto simplex
        a = _project_onto_simplex(a, s=1.0)

        # Evaluate exact objective occasionally
        if t % eval_every == 0:
            cur = 2.0 * n * conv.max()
            if cur < best_score - 1e-12:
                best_score = cur
                best_seq = a.copy()
                print(f"[{time.time():.1f}] smooth descent it={t} new best {best_score:.6f} alpha={alpha:.2e}")

        # Stop early if time left < small margin?
        if time.time() > time_limit - 0.1:
            break
    return best_seq, best_score

def _smooth_descent(a0, time_limit, best_seq, best_score,
                    alpha_start=0.05, alpha_end=1e-5, decay_iters=200,
                    lr=0.2, beta1=0.9, beta2=0.999,
                    eval_every=50):
    a = a0.copy()
    n = a.shape[0]
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    t = 0
    alpha = alpha_start
    while time.time() < time_limit:
        t += 1
        conv, w, grad = _smooth_step(a, alpha)
        # Adam update
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad * grad)
        m_hat = m / (1 - beta1 ** t)
        v_hat = v / (1 - beta2 ** t)
        step = lr * m_hat / (np.sqrt(v_hat) + 1e-8)
        a = a - step
        a = _symmetrize(a)
        a = _project_onto_simplex(a, s=1.0)
        # schedule alpha
        if t % decay_iters == 0:
            alpha = max(alpha_end, alpha * 0.5)
        # evaluate exact objective occasionally
        if t % eval_every == 0:
            cur = 2.0 * n * conv.max() # conv is from earlier compute
            if cur < best_score - 1e-12:
                best_score = cur
                best_seq = a.copy()
                print(f"[{time.time():.1f}] smooth descent it={t} new best {best_score:.6f} alpha={alpha:.2e}")
    return best_seq, best_score

"""
Propose a step‑function (non‑negative sequence) that minimizes
evaluate_sequence().

Algorithm
---------
1. Scan a rich collection of symmetric analytic families (uniform,
   arcsine, sine, cosine, quadratic, edge‑weighted, exponential‑edge,
   random Dirichlet, sparse) on several lengths.  Keep the best few
   candidates.

2. Refine the top candidates with a *smooth* surrogate for the
   ∞‑norm objective:
        F_α(a) = 2·n·(max(conv(a,a)) + α·log Σ_i e^{(conv_i−max)/α})
   The gradient of the smooth part is obtained analytically via FFT.
   An Adam‐style update is applied, after each step the vector is
   symmetrised and projected onto the probability simplex.

3. Polish the result with the exact exponentiated‑gradient sub‑gradient
   descent used in the previous state‑of‑the‑art, followed by a greedy
   pairwise mass‑transfer hill‑climber.

All work respects a user‑provided time budget (default 1000 s).  The
returned list is a plain Python list; evaluate_sequence will
re‑normalise it automatically.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Simplex projection (Euclidean)
def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Project v onto {x ≥ 0, Σx = s}."""
    if s < 0:
        raise ValueError("Simplex radius must be non‑negative")
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.where(u - cssv / (np.arange(1, len(v) + 1)) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    return np.maximum(v - theta, 0.0)


# ----------------------------------------------------------------------
# FFT‑based full self‑convolution (real)
def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    n = a.shape[0]
    size = 1 << ((2 * n - 1) - 1).bit_length()
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[: 2 * n - 1]


# ----------------------------------------------------------------------
# Exact objective (sum a = 1)
def _objective(a: np.ndarray) -> float:
    n = a.shape[0]
    b = _fft_self_convolve(a)
    return float(2.0 * n * b.max())


# ----------------------------------------------------------------------
# Analytic families (all returned already normalised to sum‑1)
def _distribution_uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _distribution_arcsine(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    vals /= vals.sum()
    return vals


def _distribution_sine(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64) + 0.5
    w = np.sin(np.pi * i / n)
    w /= w.sum()
    return w


def _distribution_cosine(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64) + 0.5
    w = 1.0 - np.cos(np.pi * i / n)
    w /= w.sum()
    return w


def _distribution_quadratic(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    x = (2.0 * i) / (n - 1) - 1.0
    w = 1.0 - x * x
    w = np.clip(w, 0.0, None)
    w /= w.sum()
    return w


def _distribution_edge_weighted(n: int, p: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64) + 0.5
    j = (n - np.arange(n, dtype=np.float64) - 0.5)
    w = (i * j) ** (-p)
    total = w.sum()
    if total == 0.0:
        return _distribution_uniform(n)
    w /= total
    return w


def _distribution_exponential_edge(n: int, c: float) -> np.ndarray:
    """w_i ∝ exp(-c·i) + exp(-c·(n‑1‑i))   (c>0)."""
    i = np.arange(n, dtype=np.float64)
    w = np.exp(-c * i) + np.exp(-c * (n - 1 - i))
    w /= w.sum()
    return w


def _distribution_random_sparse(
    n: int, rng: np.random.Generator, density: float = 0.01
) -> np.ndarray:
    k = max(2, int(n * density))
    idx = rng.choice(n, size=k, replace=False)
    vals = np.zeros(n, dtype=np.float64)
    vals[idx] = rng.random(k)
    s = vals.sum()
    if s == 0.0:
        vals[0] = 1.0
        s = 1.0
    vals /= s
    return vals


def _mix_with_uniform(base: np.ndarray, lam: float) -> np.ndarray:
    """λ·uniform + (1‑λ)·base."""
    n = base.shape[0]
    uniform = np.full(n, 1.0 / n, dtype=np.float64)
    return lam * uniform + (1.0 - lam) * base


# ----------------------------------------------------------------------
# Symmetrisation (enforces a_i = a_{n-1-i})
def _symmetrize(a: np.ndarray) -> np.ndarray:
    return (a + a[::-1]) / 2.0


# ----------------------------------------------------------------------
# Gradient of the smoothed objective (uses FFT)
def _gradient_smooth(a: np.ndarray, w: np.ndarray) -> np.ndarray:
    """
    Compute ∇_a  Σ_i w_i (a*a)_i   where w_i ≥0 , Σ_i w_i =1.
    Returns a vector of length n.
    """
    n = a.shape[0]
    w_rev = w[::-1]
    size = 1 << ((len(w_rev) + n - 1) - 1).bit_length()
    A = np.fft.rfft(a, n=size)
    W = np.fft.rfft(w_rev, n=size)
    conv = np.fft.irfft(A * W, n=size)
    full = conv[: len(w_rev) + n - 1]               # length 3n‑2
    start = 2 * n - 2
    grad = 2.0 * full[start : start + n]           # slice of length n
    return grad


# ----------------------------------------------------------------------
# Smooth‑objective descent (Adam + simplex + symmetry)
def _smooth_descent(
    a0: np.ndarray,
    time_limit: float,
    best_seq: np.ndarray,
    best_score: float,
    alpha_start: float = 0.05,
    alpha_end: float = 1e-5,
    decay_iters: int = 200,
    lr: float = 0.3,
    beta1: float = 0.9,
    beta2: float = 0.999,
    eval_every: int = 100,
) -> tuple[np.ndarray, float]:
    """Adam‑style descent on the log‑sum‑exp surrogate of the ∞‑norm."""
    a = a0.copy()
    n = a.shape[0]
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    t = 0
    alpha = alpha_start

    while time.time() < time_limit:
        t += 1

        # ----- convolution & smooth weights -----
        conv = _fft_self_convolve(a)
        maxc = conv.max()
        if alpha < 1e-12:
            w = np.zeros_like(conv)
            w[conv == maxc] = 1.0 / np.sum(conv == maxc)
        else:
            shifted = conv - maxc                # for numerical stability
            exp_shift = np.exp(shifted / alpha)
            sumexp = exp_shift.sum()
            w = exp_shift / sumexp

        # ----- gradient -----
        grad = _gradient_smooth(a, w)

        # ----- Adam update -----
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad * grad)
        m_hat = m / (1 - beta1 ** t)
        v_hat = v / (1 - beta2 ** t)
        step = lr * m_hat / (np.sqrt(v_hat) + 1e-8)

        a = a - step
        a = _symmetrize(a)
        a = _project_onto_simplex(a, s=1.0)

        # ----- temperature schedule -----
        if t % decay_iters == 0:
            alpha = max(alpha_end, alpha * 0.5)

        # ----- occasional exact evaluation -----
        if t % eval_every == 0:
            cur = 2.0 * n * maxc
            if cur < best_score - 1e-12:
                best_score = cur
                best_seq = a.copy()
                print(
                    f"[{time.time():.1f}] smooth it={t} new best {best_score:.6f} α={alpha:.2e}"
                )
    return best_seq, best_score

# Exact sub‑gradient descent (exponentiated gradient) – unchanged
def _subgradient_descent_exact(
    a0: np.ndarray,
    time_limit: float,
    best_seq: np.ndarray,
    best_score: float,
    lr0: float = 0.9,
    eval_every: int = 100,
) -> tuple[np.ndarray, float]:
    a = a0.copy()
    n = a.shape[0]
    t = 0
    while time.time() < time_limit:
        t += 1
        conv = _fft_self_convolve(a)
        max_val = conv.max()
        max_idxs = np.where(conv == max_val)[0]

        # exact sub‑gradient (sum over all maximal indices)
        grad = np.zeros_like(a)
        for idx in max_idxs:
            i_start = max(0, idx - (n - 1))
            i_end = min(n - 1, idx)
            i = np.arange(i_start, i_end + 1, dtype=np.int64)
            j = idx - i
            grad[i] += 2.0 * a[j]

        eta = lr0 / math.sqrt(t)
        a *= np.exp(-eta * grad)                 # EG update (keeps positivity)
        s = a.sum()
        if s <= 0.0:
            a = np.full(n, 1.0 / n, dtype=np.float64)
        else:
            a /= s

        if t % eval_every == 0:
            cur = 2.0 * n * max_val               # sum(a)=1 ⇒ exact objective
            if cur < best_score - 1e-14:
                best_score = cur
                best_seq = a.copy()
                print(
                    f"[{time.time():.1f}] exact subgrad it={t} new global best {best_score:.6f}"
                )
    return best_seq, best_score


# ----------------------------------------------------------------------
# Greedy pairwise mass transfer
def _pairwise_swap(
    a0: np.ndarray,
    time_limit: float,
    best_seq: np.ndarray,
    best_score: float,
    max_trials: int | None = None,
) -> tuple[np.ndarray, float]:
    a = a0.copy()
    n = a.shape[0]
    trials = 0
    while time.time() < time_limit:
        if max_trials is not None and trials >= max_trials:
            break
        trials += 1

        src_candidates = np.where(a > 1e-14)[0]
        if src_candidates.size == 0:
            break
        i = np.random.choice(src_candidates)
        j = np.random.randint(n - 1)
        if j >= i:
            j += 1

        delta = a[i] * np.random.rand() * 0.45   # up to 45 % of source mass
        if delta < 1e-15:
            continue

        new_a = a.copy()
        new_a[i] -= delta
        new_a[j] += delta

        cur = _objective(new_a)
        if cur < best_score - 1e-14:
            a = new_a
            best_score = cur
            best_seq = a.copy()
            print(
                f"[{time.time():.1f}] swap i={i}→j={j} δ={delta:.6g} new best {best_score:.6f}"
            )
    return best_seq, best_score
