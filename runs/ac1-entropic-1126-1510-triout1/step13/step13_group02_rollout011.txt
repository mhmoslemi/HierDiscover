def _mirror_descent(a, deadline, best_seq, best_score, eta0=0.9, eval_interval=100):
    t=0
    while time.time()<deadline:
        t+=1
        grad,_ = _grad_of_max_conv(a)
        eta = eta0/(t**0.5)
        a = a * np.exp(-eta * grad)
        # normalize
        sum_a = a.sum()
        if sum_a == 0:
            a = np.full_like(a, 1.0/a.shape[0])
        else:
            a /= sum_a
        if t % eval_interval == 0:
            sc = _objective(a)
            if sc < best_score - 1e-14:
                best_score = sc
                best_seq = a.copy()
                print(f"[{time.time():.1f}] GD iter {t} new best {best_score:.6f}")
    return best_seq, best_score

def _targeted_annealing(a, deadline, best_seq, best_score, rng):
    n = a.shape[0]
    start_time = time.time()
    total_time = deadline - start_time
    while time.time() < deadline:
        # compute conv and max index
        conv = _fft_self_convolve(a)
        max_val = conv.max()
        k = int(np.argmax(conv))
        # find i_max, j_max
        start = max(0, k - (n - 1))
        end = min(n - 1, k)
        i_range = np.arange(start, end+1)
        j_range = k - i_range
        contribs = a[i_range] * a[j_range]
        idx_max = int(np.argmax(contribs))
        i0 = i_range[idx_max]
        j0 = j_range[idx_max]
        # if i0 == j0 use the same index
        # choose source index to reduce (choose the larger a among i0 and j0)
        src = i0 if a[i0] >= a[j0] else j0
        # pick destination index random not src
        dest_candidates = np.arange(n)
        # choose dest not equal src
        dest = rng.choice(dest_candidates[dest_candidates != src])
        # delta fraction of a[src]
        delta = a[src] * rng.random() * 0.5 # maybe up to 50%
        if delta < 1e-16:
            continue
        new_a = a.copy()
        new_a[src] -= delta
        new_a[dest] += delta
        # ensure none negative
        if new_a[src] < 0:
            new_a[src] = 0.0
        cur = _objective(new_a)
        # simulate annealing temperature
        elapsed = time.time() - start_time
        temperature = max(0.0, 1.0 - elapsed/total_time) * best_score * 0.01
        accept = False
        if cur < best_score - 1e-14:
            accept = True
        elif temperature > 0.0:
            prob = math.exp(-(cur - best_score)/temperature)
            if rng.random() < prob:
                accept = True
        if accept:
            a = new_a
            if cur < best_score - 1e-14:
                best_score = cur
                best_seq = a.copy()
                print(f"[{time.time():.1f}] targeted anneal src={src}->{dest} δ={delta:.3e} new best {best_score:.6f}")
    return best_seq, best_score

def _distribution_beta(n, s):
    i = np.arange(n, dtype=np.float64) + 0.5
    x = i / n
    # to avoid overflow, clip x in (0,1)
    eps = 1e-12
    x = np.clip(x, eps, 1 - eps)
    w = (x**(s-1)) * ((1 - x)**(s-1))
    w_sum = w.sum()
    if w_sum == 0:
        w = np.full_like(w, 1.0/n)
    else:
        w /= w_sum
    return w

"""
step_function_search_improved.py

A 1000‑second heuristic for the step‑function problem
``evaluate_sequence``.  The method consists of:

1. **Rich seeding** – generate many probability vectors from several
   analytic families (uniform, arcsine, Hann, edge‑weighted, beta,
   generalized‑normal, Dirichlet, sparse).  All vectors are normalised
   to sum = 1 and the best few are retained.

2. **Fast exponentiated‑gradient polishing** – each retained seed is
   polished with mirror‑descent on the exact objective
   ``2·n·max(conv(a,a))`` (the sum normalisation makes the denominator
   disappear).  The update is ``a ← a·exp(‑η·grad)`` followed by ℓ₁‑normalisation,
   where ``grad`` is the sub‑gradient of the currently maximal convolution
   entry.  A decaying step‑size η≈η₀/√t is used.

3. **Long gradient descent on the current best** – a generous share of the budget
   refines the best candidate found so far.

4. **Upsampling** – the best vector is linearly interpolated to twice the length
   (kept ≤ 64 000) and polished again.

5. **Two‑stage annealing** – first a cheap random pairwise‑mass‑transfer
   annealing, then a *targeted* annealing that explicitly attacks the
   dominant convolution term (the largest product ``a_i·a_j`` contributing to
   the maximal entry).  Moves that improve the objective are always kept,
   deteriorating moves are accepted with a temperature‑dependent probability.

The whole pipeline respects the 1000 s budget (2 s safety margin).  The
returned list can be fed directly to ``evaluate_sequence``.
"""

import time

# ----------------------------------------------------------------------
# Simplex utilities
# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Euclidean projection onto the simplex {x ≥ 0, Σx = s}."""
    if s < 0:
        raise ValueError("simplex radius must be non‑negative")
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.where(u - cssv / (np.arange(1, len(v) + 1)) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Convolution helpers (FFT based, linear convolution)
# ----------------------------------------------------------------------
def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Full linear self‑convolution a*a using a real FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1) - 1).bit_length()  # next power of two
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[: 2 * n - 1]


# ----------------------------------------------------------------------
# Exact objective (scale‑invariant, sum=1 ⇒ denominator = 1)
# ----------------------------------------------------------------------
def _objective(a: np.ndarray) -> float:
    """Exact value of ``evaluate_sequence`` for a probability vector (∑a = 1)."""
    n = a.shape[0]
    b = _fft_self_convolve(a)
    max_b = b.max()
    return float(2.0 * n * max_b)


# ----------------------------------------------------------------------
# Gradient of the maximal convolution entry (sub‑gradient)
# ----------------------------------------------------------------------
def _grad_of_max_conv(a: np.ndarray) -> tuple[np.ndarray, int]:
    """
    Compute the sub‑gradient of the currently maximal convolution entry.
    Returns (grad, k) where k is the index attaining the maximum.
    """
    n = a.shape[0]
    conv = _fft_self_convolve(a)
    k = int(np.argmax(conv))

    grad = np.zeros_like(a)
    if k < n:
        grad[: k + 1] = 2.0 * a[k::-1]
    else:
        start = k - (n - 1)
        grad[start:] = 2.0 * a[start:][::-1]
    return grad, k


# ----------------------------------------------------------------------
# Analytic families (all return a probability vector, sum=1)
# ----------------------------------------------------------------------
def _distribution_uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _distribution_arcsine(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    vals /= vals.sum()
    return vals


def _distribution_hann(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    w = 0.5 - 0.5 * np.cos(2.0 * np.pi * i / (n - 1))
    w_sum = w.sum()
    if w_sum == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w_sum
    return w


def _distribution_edge_weighted(n: int, p: float) -> np.ndarray:
    """Weight ∝ (i·j)^{‑p} where i,j are distances to the two edges."""
    i = np.arange(n, dtype=np.float64) + 0.5
    j = (n - np.arange(n, dtype=np.float64) - 0.5)
    w = (i * j) ** (-p)
    w_sum = w.sum()
    if w_sum == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w_sum
    return w


def _distribution_generalized_normal(n: int, q: float) -> np.ndarray:
    """Density ∝ exp(‑|x‑½|^{q}) on [0,1]."""
    i = np.arange(n, dtype=np.float64) + 0.5
    x = i / n - 0.5
    w = np.exp(- (np.abs(x) / 0.5) ** q)
    w_sum = w.sum()
    if w_sum == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w_sum
    return w


def _distribution_beta(n: int, s: float) -> np.ndarray:
    """Symmetric Beta( s , s ) distribution on the interior of [0,1]."""
    i = np.arange(n, dtype=np.float64) + 0.5
    x = i / n
    eps = 1e-12
    x = np.clip(x, eps, 1.0 - eps)
    w = (x ** (s - 1.0)) * ((1.0 - x) ** (s - 1.0))
    w_sum = w.sum()
    if w_sum == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w_sum
    return w


def _distribution_random_sparse(
    n: int, rng: np.random.Generator, density: float = 0.01
) -> np.ndarray:
    """Sparse random vector with given density."""
    k = max(2, int(n * density))
    idx = rng.choice(n, size=k, replace=False)
    vals = np.zeros(n, dtype=np.float64)
    vals[idx] = rng.random(k)
    if vals.sum() == 0.0:
        vals[0] = 1.0
    vals /= vals.sum()
    return vals


def _mix_with_uniform(base: np.ndarray, lam: float) -> np.ndarray:
    """Convex combination with the uniform distribution."""
    n = base.shape[0]
    uniform = np.full(n, 1.0 / n, dtype=np.float64)
    return lam * uniform + (1.0 - lam) * base


# ----------------------------------------------------------------------
# Exponentiated‑gradient (mirror) descent on the exact objective
# ----------------------------------------------------------------------
def _mirror_descent(
    a: np.ndarray,
    deadline: float,
    best_seq: np.ndarray,
    best_score: float,
    eta0: float = 0.9,
    eval_every: int = 60,
) -> tuple[np.ndarray, float]:
    """Mirror‑descent with entropy regulariser on the exact objective."""
    n = a.shape[0]
    t = 0
    while time.time() < deadline:
        t += 1
        grad, _ = _grad_of_max_conv(a)
        eta = eta0 / math.sqrt(t)
        a = a * np.exp(-eta * grad)          # stays non‑negative
        s = a.sum()
        if s == 0.0:
            a = np.full(n, 1.0 / n, dtype=np.float64)
        else:
            a = a / s

        if t % eval_every == 0:
            cur = _objective(a)
            if cur < best_score - 1e-14:
                best_score = cur
                best_seq = a.copy()
                print(f"[{time.time():.1f}] GD iter={t} new global best {best_score:.6f}")
    return best_seq, best_score

def _pairwise_annealing(

) -> tuple[np.ndarray, float]:
    """Randomly transfer a fraction of mass between two bins."""
    n = a.shape[0]
    start_time = time.time()
    total_time = deadline - start_time

    while time.time() < deadline:
        src_candidates = np.where(a > 1e-12)[0]
        if src_candidates.size == 0:
            break
        i = rng.choice(src_candidates)

        j = rng.integers(n - 1)
        if j >= i:
            j += 1

        delta = a[i] * rng.random() * 0.4
        if delta < 1e-16:
            continue

        new_a = a.copy()
        new_a[i] -= delta
        new_a[j] += delta

        cur = _objective(new_a)

        elapsed = time.time() - start_time
        temperature = max(0.0, 1.0 - elapsed / total_time) * best_score * 0.01

        accept = False
        if cur < best_score - 1e-14:

            if cur < best_score - 1e-14:
                best_score = cur
                best_seq = a.copy()
                print(
                    f"[{time.time():.1f}] pairwise anneal i={i}→j={j} δ={delta:.3e} new best {best_score:.6f}"
                )
    return best_seq, best_score

# Targeted annealing – attacks the dominant convolution contribution

def _targeted_annealing(

    deadline: float,
    best_seq: np.ndarray,
    best_score: float,
    rng: np.random.Generator,
) -> tuple[np.ndarray, float]:
    """Move mass away from the pair (i,j) that yields the largest a_i·a_j part of max(conv)."""
    n = a.shape[0]
    start_time = time.time()
    total_time = deadline - start_time

    while time.time() < deadline:
        # Full convolution and dominant index
        conv = _fft_self_convolve(a)
        max_val = conv.max()
        k = int(np.argmax(conv))

        # indices that contribute to this entry
        start = max(0, k - (n - 1))
        end = min(n - 1, k)
        i_range = np.arange(start, end + 1, dtype=np.int64)
        j_range = k - i_range
        contribs = a[i_range] * a[j_range]
        idx_max = int(np.argmax(contribs))
        i0 = i_range[idx_max]
        j0 = j_range[idx_max]

        # Pick the larger of the two as source
        src = i0 if a[i0] >= a[j0] else j0

        if delta < 1e-16:
            continue

        new_a = a.copy()
        new_a[src] -= delta
        new_a[dest] += delta

        cur = _objective(new_a)

        # Temperature schedule (linear cooling)
        elapsed = time.time() - start_time
        temperature = max(0.0, 1.0 - elapsed / total_time) * best_score * 0.01

        accept = False
        if cur < best_score - 1e-14:
            accept = True
        elif temperature > 0.0:
            prob = math.exp(-(cur - best_score) / temperature)
            if rng.random() < prob:
                accept = True

        if accept:
            a = new_a
            if cur < best_score - 1e-14:
                best_score = cur
                best_seq = a.copy()
                print(
                    f"[{time.time():.1f}] targeted anneal src={src}→dest={dest} δ={delta:.3e} new best {best_score:.6f}"
                )
    return best_seq, best_score


# ----------------------------------------------------------------------
# Upsampling (linear interpolation while preserving the simplex)
# ----------------------------------------------------------------------
def _upsample(a: np.ndarray, new_n: int) -> np.ndarray:
    """Interpolate a probability vector to a larger length."""
    old_n = a.shape[0]
    if new_n == old_n:
        return a.copy()
    x_old = (np.arange(old_n, dtype=np.float64) + 0.5) / old_n
    x_new = (np.arange(new_n, dtype=np.float64) + 0.5) / new_n
    a_new = np.interp(x_new, x_old, a, left=0.0, right=0.0)
    a_new = np.maximum(a_new, 0.0)
    s = a_new.sum()
    if s == 0.0:
        a_new[:] = 1.0 / new_n
    else:
        a_new /= s
    return a_new

    try:
        prev = np.asarray(height_sequence_1, dtype=np.float64)  # type: ignore[name-defined]
        if prev.ndim == 1:
            prev = prev / prev.sum()
            sc = _objective(prev)
            if sc < overall_best_score:
                overall_best_score = sc
                overall_best_seq = prev.copy()
                print(f"[{time.time():.1f}] previous best score={sc:.6f}")
            cand_pool.append((sc, prev.copy()))
    except Exception:
        pass

    # ------------------------------------------------------------------
    # 3.  Ensure a non‑empty result
    # ------------------------------------------------------------------
    if overall_best_seq is None:
        n = 3000
        overall_best_seq = _distribution_uniform(n)
        overall_best_score = _objective(overall_best_seq)
        print(f"[{time.time():.1f}] fallback uniform score={overall_best_score:.6f}")

    # ------------------------------------------------------------------
    # 4.  Short gradient‑descent polishing on *all* candidates
    # ------------------------------------------------------------------
    remaining = deadline - time.time()
    if remaining > 20.0 and cand_pool:
        per_candidate = max(0.2, remaining * 0.3 / max(len(cand_pool), 1))
        print(f"[{time.time():.1f}] quick GD on {len(cand_pool)} seeds (~{per_candidate:.2f}s each)")
        for idx, (score, a0) in enumerate(cand_pool):
            if time.time() >= deadline:
                break
            sub_dead = time.time() + per_candidate
            a_opt, sc_opt = _mirror_descent(
                a0.copy(),
                sub_dead,
                best_seq=overall_best_seq,
                best_score=overall_best_score,
                eta0=1.0,
                eval_every=30,
            )
            if sc_opt < overall_best_score - 1e-12:
                overall_best_score = sc_opt
                overall_best_seq = a_opt.copy()
                print(f"[{time.time():.1f}] after quick GD seed {idx} new best {overall_best_score:.6f}")

    # ------------------------------------------------------------------
    # 5.  Longer gradient descent on the current global best
    # ------------------------------------------------------------------
    if time.time() < deadline:
        sub_dead = time.time() + (deadline - time.time()) * 0.35
        a_opt, sc_opt = _mirror_descent(
            overall_best_seq.copy(),
            sub_dead,
            best_seq=overall_best_seq,
            best_score=overall_best_score,
            eta0=0.8,
            eval_every=60,
        )
        if sc_opt < overall_best_score - 1e-12:
            overall_best_score = sc_opt
            overall_best_seq = a_opt.copy()
            print(f"[{time.time():.1f}] long GD improved best to {overall_best_score:.6f}")

    # ------------------------------------------------------------------
    # 6.  Upsample the best vector (if time permits) and polish again
    # ------------------------------------------------------------------
    if time.time() < deadline:
        cur_n = overall_best_seq.shape[0]
        target_n = cur_n * 2
        if target_n <= 64000:                     # keep FFT size modest
            print(f"[{time.time():.1f}] upsampling {cur_n} → {target_n}")
            up = _upsample(overall_best_seq, target_n)
            up_sc = _objective(up)
            if up_sc < overall_best_score:
                overall_best_seq = up
                overall_best_score = up_sc
                print(f"[{time.time():.1f}] upsampled new best {overall_best_score:.6f}")

            # short GD on the upsampled vector
            sub_dead = time.time() + (deadline - time.time()) * 0.25
            a_opt, sc_opt = _mirror_descent(
                overall_best_seq.copy(),
                sub_dead,
                best_seq=overall_best_seq,
                best_score=overall_best_score,
                eta0=0.7,
                eval_every=50,
            )
            if sc_opt < overall_best_score - 1e-12:
                overall_best_seq = a_opt.copy()
                overall_best_score = sc_opt
                print(f"[{time.time():.1f}] GD after upsample new best {overall_best_score:.6f}")

    # ------------------------------------------------------------------
    # 7.  Pairwise random annealing
    # ------------------------------------------------------------------
    if time.time() < deadline:
        print(f"[{time.time():.1f}] pairwise random annealing phase")
        overall_best_seq, overall_best_score = _pairwise_annealing(
            overall_best_seq.copy(),
            deadline,
            best_seq=overall_best_seq,
            best_score=overall_best_score,
            rng=rng,
        )

    # ------------------------------------------------------------------
    # 8.  Targeted annealing – focus on the dominant convolution term
    # ------------------------------------------------------------------
    if time.time() < deadline:
        print(f"[{time.time():.1f}] targeted annealing phase")
        overall_best_seq, overall_best_score = _targeted_annealing(
            overall_best_seq.copy(),
            deadline,
            best_seq=overall_best_seq,
            best_score=overall_best_score,
            rng=rng,
        )

    # ------------------------------------------------------------------
    print(
        f"Final best score (objective=2·n·max(conv)): {overall_best_score:.6f} "
        f"(length={len(overall_best_seq)})"
    )
    # --------------------------------------------------------------
    # Return plain Python list (evaluate_sequence expects a list)
    # --------------------------------------------------------------
    return overall_best_seq.tolist()
