"""
Propose a low‑score step function.

The search consists of three phases

1. **Broad seeding** – generate many analytic families (uniform, arcsine,
   symmetric / asymmetric Beta, edge‑weighted, sine‑weighted, mixtures,
   Dirichlet, sparse random) for several lengths (2k‑15k).  Keep the
   ~80 best (lowest exact objective) candidates.

2. **Accelerated exponentiated‑gradient descent** – starting from each of the
   best seeds run an Adam‑style multiplicative mirror‑descent on the
   *soft‑max* surrogate of ``max(conv(a,a))``.  The temperature τ is raised
   slowly so that the surrogate converges to the true max.  The iteration
   runs until the allocated time expires or no improvement is seen for a
   while.  The best vector among all runs is retained.

3. **Fine‑grained local search** – a deterministic pairwise‑mass‑transfer
   refinement followed by a cheap simulated‑annealing walk removes any
   remaining tiny bumps.

All operations respect the global time budget (default 1000 s).  The
returned list is a plain Python list; ``evaluate_sequence`` will normalise
it internally.

# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x≥0, Σx = s}."""
    if s < 0:
        raise ValueError("simplex radius must be non‑negative")
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho_arr = np.where(u - cssv / (np.arange(1, len(v) + 1)) > 0)[0]
    if rho_arr.size == 0:
        theta = 0.0
    else:
        rho = rho_arr[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Full self‑convolution a*a using a real FFT (returns length 2n‑1)."""
    n = a.shape[0]
    need = 2 * n - 1
    size = 1 << ((need - 1).bit_length())
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[:need]


# ----------------------------------------------------------------------
def _objective(a: np.ndarray) -> float:
    """Exact objective (2·n·max(conv(a,a))) for a vector with Σa = 1."""
    n = a.shape[0]
    b = _fft_self_convolve(a)
    return float(2.0 * n * b.max())

# Analytic families ----------------------------------------------------
def _distribution_uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _distribution_arcsine(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    vals /= vals.sum()
    return vals


def _distribution_beta_sym(n: int, p: float) -> np.ndarray:
    """Symmetric beta (Beta(p,p)) on the discrete grid."""
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    w = np.exp((p - 1) * np.log(x) + (p - 1) * np.log(1.0 - x))
    if w.sum() == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _distribution_beta_asym(n: int, a: float, b: float) -> np.ndarray:
    """Asymmetric beta (Beta(a,b)) on the discrete grid."""
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    w = np.exp((a - 1) * np.log(x) + (b - 1) * np.log(1.0 - x))
    if w.sum() == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _distribution_edge_weighted(n: int, p: float) -> np.ndarray:
    """U‑shaped family: w_i ∝ ((i+0.5)*(n‑i‑0.5))^(−p)."""
    i = np.arange(n, dtype=np.float64) + 0.5
    j = (n - np.arange(n, dtype=np.float64) - 0.5)
    w = (i * j) ** (-p)
    if w.sum() == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _distribution_sine_weighted(n: int, p: float) -> np.ndarray:
    """Weight proportional to sin(π (i+½)/n)^p."""
    theta = np.pi * (np.arange(n, dtype=np.float64) + 0.5) / n
    w = np.sin(theta) ** p
    if w.sum() == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _distribution_random_sparse(
    n: int, rng: np.random.Generator, density: float = 0.01
) -> np.ndarray:
    """Sparse random vector with a given density of non‑zeros."""
    k = max(2, int(n * density))
    idx = rng.choice(n, size=k, replace=False)
    vals = np.zeros(n, dtype=np.float64)
    vals[idx] = rng.random(k)
    if vals.sum() == 0.0:

    """Convex combination lam·U + (1‑lam)·base."""
    n = base.shape[0]
    uniform = np.full(n, 1.0 / n, dtype=np.float64)

    b1 = _distribution_beta_sym(n, p1)

# Soft‑max helpers ------------------------------------------------------
def _softmax_weights(conv: np.ndarray, tau: float) -> np.ndarray:
    """Soft‑max of τ·conv (stable). Returns a probability vector."""
    max_c = conv.max()
    w_raw = np.exp(tau * (conv - max_c))
    total = w_raw.sum()
    if total == 0.0:
        # fall‑back to uniform
        return np.full_like(conv, 1.0 / conv.size)
    return w_raw / total


def _weighted_gradient(a: np.ndarray, w: np.ndarray) -> np.ndarray:
    """
    Gradient of   L(a)= Σ_i w_i·(a*a)[i]   w.r.t a.
    Closed‑form:   ∇L = 2 * (a_rev ⊛ w)   (restricted to first n entries).
    """
    n = a.shape[0]
    a_rev = a[::-1]                     # length n
    need = len(w) + n - 1                # = 3·n‑2
    size = 1 << ((need - 1).bit_length())
    A = np.fft.rfft(a_rev, n=size)
    W = np.fft.rfft(w, n=size)
    conv = np.fft.irfft(A * W, n=size)
    conv = conv[:need]                  # length need
    grad = 2.0 * conv[n - 1 : n - 1 + n]   # slice of length n
    return grad

# Accelerated exponentiated‑gradient descent (Adam‑style) -------------
def _adam_exponential_descent(
    a: np.ndarray,
    deadline: float,
    best_seq: np.ndarray,
    best_score: float,
    eta0: float = 0.6,
    tau0: float = 1.0,
    tau_mult: float = 1.002,
    eval_interval: int = 150,
    patience: int = 3000,
    beta1: float = 0.9,
    beta2: float = 0.999,
    eps: float = 1e-12,
) -> tuple[np.ndarray, float]:
    """Adam‑style multiplicative mirror descent on the soft‑max surrogate."""
    n = a.shape[0]
    m = np.zeros_like(a)          # first moment
    v = np.zeros_like(a)          # second moment
    t = 0
    no_improve = 0
    rng = np.random.default_rng()
    tau = tau0

    while time.time() < deadline and t < 300_000:
        t += 1

        conv = _fft_self_convolve(a)
        w = _softmax_weights(conv, tau)
        grad = _weighted_gradient(a, w)

        # Adam updates on the *gradient* (not on log‑probabilities)
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)

        # bias‑correction
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)

        # multiplicative step (exponentiated)
        step = eta0 * m_hat / (np.sqrt(v_hat) + eps)

        # clip the step magnitude to avoid numerical blow‑up
        max_abs = np.max(np.abs(step))
        if max_abs > 0.7:
            step = step * (0.7 / max_abs)

        a = a * np.exp(-step)

        # renormalise onto the simplex (keeps sum = 1)
        s = a.sum()
        if s == 0.0 or np.isnan(s):
            a = np.full(n, 1.0 / n, dtype=np.float64)
        else:
            a = a / s

        # increase temperature – sharpens the soft‑max
        tau *= tau_mult

        # occasional exact evaluation
        if t % eval_interval == 0:
            cur = _objective(a)
            if cur < best_score - 1e-14:
                best_score = cur
                best_seq = a.copy()
                no_improve = 0
                print(f"[{time.time():.1f}] Adam descent it={t} new best {best_score:.6f}")
            else:
                no_improve += 1

        # restart / jitter if stuck
        if no_improve >= patience:
            jitter = rng.exponential(scale=0.002, size=n)
            a = a + jitter
            a = np.maximum(a, 0.0)
            a = a / a.sum()
            no_improve = 0

    return best_seq, best_score

# Pairwise swap local refinement -----------------------------------------
def _pairwise_swap_local(
    a: np.ndarray,
    deadline: float,
    best_seq: np.ndarray,
    best_score: float,
    max_trials: int | None = None,
) -> tuple[np.ndarray, float]:
    """Randomly move a small chunk of mass; keep it if the objective improves."""
    n = a.shape[0]
    rng = np.random.default_rng()
    trials = 0
    while time.time() < deadline:
        if max_trials is not None and trials >= max_trials:
            break
        trials += 1

        # pick a source index with non‑negligible mass
        src_nonzero = np.where(a > 1e-14)[0]
        if src_nonzero.size == 0:
            break
        i = rng.choice(src_nonzero)

        # pick a distinct destination
        j = rng.integers(n - 1)
        if j >= i:
            j += 1

        # transfer a random fraction of the source mass
        delta = a[i] * rng.random() * 0.35
        if delta < 1e-16:
            continue

        new_a = a.copy()
        new_a[i] -= delta
        new_a[j] += delta

        cur = _objective(new_a)
        if cur < best_score - 1e-14:
            a = new_a
            best_score = cur
            best_seq = a.copy()
            print(
                f"[{time.time():.1f}] swap i={i}→j={j} δ={delta:.3e} new best {best_score:.6f}"
            )
    return best_seq, best_score

            best_score = cur
            best_seq = a.copy()
            print(f"[{time.time():.1f}] SA improve {i}->{j} Δ={delta:.3e} best {best_score:.6f}")
        else:
            if rng.random() < math.exp(-(cur - best_score) / max(T, 1e-12)):
                a = new_a
        T *= decay
    return best_seq, best_score


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: int = 1000) -> list[float]:
    """
    Search for a non‑negative step‑function that minimises ``evaluate_sequence``.
    Returns a plain Python list; the routine respects the supplied time budget.
    """
    rng = np.random.default_rng(seed)
    start = time.time()
    deadline = start + budget_s - 4.0               # small safety margin

    overall_best_seq = None
    overall_best_score = float("inf")
    top_candidates: list[tuple[float, np.ndarray]] = []

    # ------------------------------------------------------------------
    # 1.  Generate a rich pool of seeds
    lengths = [2000, 3000, 5000, 8000, 12000]       # moderate sizes, still fast FFT
    # parameter grids for p (Beta / edge / sine families)
    p_grid = np.concatenate(
        [
            np.logspace(-2, -1, 8),   # 0.01 … 0.1
            np.logspace(-1, 0, 10),   # 0.1 … 1
            np.logspace(0, 0.5, 8),   # 1 … ~3.2
        ]
    )
    lam_grid = np.linspace(0.0, 1.0, 7)             # mixing with uniform

    for n in lengths:
        if time.time() >= deadline:
            break
        print(f"[{time.time():.1f}] generating seeds, length={n}")

        # deterministic families
        for cand in (_distribution_uniform(n), _distribution_arcsine(n)):
            sc = _objective(cand)
            top_candidates.append((sc, cand.copy()))
            if sc < overall_best_score:
                overall_best_score, overall_best_seq = sc, cand.copy()
                print(f"[{time.time():.1f}] new global best (det) len={n} score={sc:.6f}")

        # parametrised families + mixing
        for p in p_grid:
            if time.time() >= deadline:
                break

            beta = _distribution_beta_sym(n, p)
            edge = _distribution_edge_weighted(n, p)
            sine = _distribution_sine_weighted(n, p)

            for base in (beta, edge, sine):
                for lam in lam_grid:
                    if time.time() >= deadline:
                        break
                    cand = _mix_with_uniform(base, lam)
                    sc = _objective(cand)
                    top_candidates.append((sc, cand.copy()))
                    if sc < overall_best_score:
                        overall_best_score, overall_best_seq = sc, cand.copy()
                        print(
                            f"[{time.time():.1f}] best (mix) len={n} p={p:.3g} lam={lam:.2f} score={sc:.6f}"
                        )

        # mixture of two symmetric betas (extra flexibility)
        for _ in range(6):
            if time.time() >= deadline:
                break
            p1 = 10 ** rng.uniform(-2, 0.5)
            p2 = 10 ** rng.uniform(-2, 0.5)
            w1 = rng.random()
            cand = _mixture_two_beta(n, p1, p2, w1)
            sc = _objective(cand)
            top_candidates.append((sc, cand.copy()))
            if sc < overall_best_score:
                overall_best_score, overall_best_seq = sc, cand.copy()
                print(
                    f"[{time.time():.1f}] mix‑beta len={n} p1={p1:.3g} p2={p2:.3g} w1={w1:.2f} score={sc:.6f}"
                )

        # asymmetric beta
        for _ in range(5):
            if time.time() >= deadline:
                break
            a_param = 10 ** rng.uniform(-2, 0.5)
            b_param = 10 ** rng.uniform(-2, 0.5)
            cand = _distribution_beta_asym(n, a_param, b_param)
            sc = _objective(cand)
            top_candidates.append((sc, cand.copy()))
            if sc < overall_best_score:
                overall_best_score, overall_best_seq = sc, cand.copy()
                print(
                    f"[{time.time():.1f}] asym‑beta len={n} a={a_param:.3g} b={b_param:.3g} score={sc:.6f}"
                )

        # Dirichlet random vectors (varying concentration)
        for _ in range(6):
            if time.time() >= deadline:
                break
            alpha = rng.uniform(0.01, 1.5)
            cand = rng.dirichlet(np.full(n, alpha))
            sc = _objective(cand)
            top_candidates.append((sc, cand.copy()))
            if sc < overall_best_score:
                overall_best_score, overall_best_seq = sc, cand.copy()
                print(
                    f"[{time.time():.1f}] Dirichlet len={n} α={alpha:.3f} score={sc:.6f}"
                )

        # Sparse random seeds
        for _ in range(6):
            if time.time() >= deadline:
                break
            cand = _distribution_random_sparse(n, rng, density=0.005)
            sc = _objective(cand)
            top_candidates.append((sc, cand.copy()))
            if sc < overall_best_score:
                overall_best_score, overall_best_seq = sc, cand.copy()
                print(
                    f"[{time.time():.1f}] sparse len={n} score={sc:.6f}"
                )

        # keep only a modest number of the best candidates so far
        top_candidates.sort(key=lambda x: x[0])
        top_candidates = top_candidates[:80]

    # ------------------------------------------------------------------
    # 2.  Seed from previously discovered construction, if available
    try:
        prev = np.asarray(height_sequence_1, dtype=np.float64)  # type: ignore[name-defined]
        if prev.ndim == 1 and prev.size > 0:
            prev = prev / prev.sum()
            sc = _objective(prev)
            if sc < overall_best_score:
                overall_best_score, overall_best_seq = sc, prev.copy()
                print(f"[{time.time():.1f}] previous best score={sc:.6f}")
            top_candidates.append((sc, prev.copy()))
    except Exception:
        pass

    # ------------------------------------------------------------------
    # 3.  Fallback in the unlikely case we have no candidate
    if overall_best_seq is None:
        n = 3000
        overall_best_seq = _distribution_uniform(n)
        overall_best_score = _objective(overall_best_seq)
        print(f"[{time.time():.1f}] fallback uniform score={overall_best_score:.6f}")

    # ------------------------------------------------------------------
    # 4.  Accelerated exponential descent on the most promising seeds
    remaining = deadline - time.time()
    if remaining > 30.0 and top_candidates:
        # pick the K best seeds for intensive optimisation
        K = min(8, len(top_candidates))
        seeds = [seq for _, seq in sorted(top_candidates, key=lambda x: x[0])[:K]]
        per_seed = remaining / max(1, len(seeds))
        print(f"[{time.time():.1f}] starting Adam descent on {len(seeds)} seeds")
        for idx, seed_seq in enumerate(seeds):
            if time.time() >= deadline:
                break
            sub_deadline = time.time() + per_seed
            a_opt, sc_opt = _adam_exponential_descent(
                seed_seq.copy(),
                sub_deadline,
                best_seq=overall_best_seq,
                best_score=overall_best_score,
                eta0=0.6,
                tau0=1.0,
                tau_mult=1.002,
                eval_interval=120,
                patience=2600,
                beta1=0.9,
                beta2=0.999,
            )
            if sc_opt < overall_best_score - 1e-14:
                overall_best_score, overall_best_seq = sc_opt, a_opt.copy()
                print(
                    f"[{time.time():.1f}] after descent seed {idx} new global best {overall_best_score:.6f}"
                )

    # ------------------------------------------------------------------
    # 5.  Deterministic pairwise refinement
    if time.time() < deadline:
        print(f"[{time.time():.1f}] pairwise refinement")
        overall_best_seq, overall_best_score = _pairwise_swap_local(
            overall_best_seq.copy(),
            deadline,
            best_seq=overall_best_seq,
            best_score=overall_best_score,
            max_trials=250_000,
        )

    # ------------------------------------------------------------------
    # 6.  Final stochastic local search (cheap SA)
    if time.time() < deadline:
        print(f"[{time.time():.1f}] final simulated‑annealing walk")
        overall_best_seq, overall_best_score = _random_local_search(
            overall_best_seq.copy(),
            deadline,
            best_seq=overall_best_seq,
            best_score=overall_best_score,
            init_temp=0.008,
            decay=0.9997,
        )

    # ------------------------------------------------------------------
    print(
        f"Final best score: {overall_best_score:.6f} (length={len(overall_best_seq)})"
    )
    # Return plain Python list – evaluate_sequence will handle final checks.
    return overall_best_seq.tolist()
