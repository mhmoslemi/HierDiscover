"""
Improved search for a step‑function that minimises ``evaluate_sequence``.

Strategy
--------
1. **Rich family of analytical seeds** – generate many candidates by
   varying a few parameters:
   * symmetric edge‑weighted power law   w_i ∝ ((i+½)(n‑i‑½))^(−p)
   * asymmetric power law               w_i ∝ (i+½)^α (n‑i‑½)^β
   * symmetric exponential taper        w_i ∝ e^{−γ i}+e^{−γ (n‑1‑i)}
   * cosine‑edge weighting              w_i ∝ |cos(π i/(n‑1))|^p
   * uniform, random Dirichlet, sparse vectors.
   For each shape a fine grid of the shape parameters and a fine grid of a
   mixing factor λ with the uniform distribution are examined.  The best
   few seeds (≈10) are retained.

2. **Projected sub‑gradient descent** – each promising seed is refined by
   minimising the exact objective
        f(a)=2·n·max(conv(a,a))      (the sum of a is kept equal to 1)
   using a classic sub‑gradient step and Euclidean projection onto the
   probability simplex.  The step size η decays as η0/√t and a tiny
   jitter is injected when progress stalls.

3. **Greedy pairwise mass moves** – a cheap local search moves a small
   amount of mass between two indices if the objective improves.

4. **Very cheap simulated‑annealing walk** – finally a short SA walk
   occasionally accepts worsening moves to escape tiny basins.

All stages respect the overall 1000 s budget.  The function returns a plain
Python list; ``evaluate_sequence`` will later apply the required clipping
and normalisation.
"""

import time
import math
import numpy as np
from scipy.special import gamma

# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Euclidean projection of a vector onto the simplex {x≥0, Σx=s}."""
    if s < 0:
        raise ValueError("simplex radius must be non‑negative")
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho_arr = np.where(u - cssv / (np.arange(1, len(v) + 1)) > 0)[0]
    if rho_arr.size == 0:
        theta = 0.0
    else:
        rho = rho_arr[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Full self‑convolution a*a using a real FFT (returns length 2n‑1)."""
    n = a.shape[0]
    need = 2 * n - 1
    size = 1 << ((need - 1).bit_length())
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[:need]


# ----------------------------------------------------------------------
def _objective(a: np.ndarray) -> float:
    """Exact objective 2·n·max(conv(a,a)) for a probability vector (sum=1)."""
    n = a.shape[0]
    b = _fft_self_convolve(a)
    return 2.0 * n * b.max()


# ----------------------------------------------------------------------
# Analytic families ----------------------------------------------------
def _distribution_uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _distribution_edge_weighted(n: int, p: float) -> np.ndarray:
    """w_i ∝ ((i+½)*(n‑i‑½))^(−p).  p≥0 – the larger p, the more weight at the edges."""
    i = np.arange(n, dtype=np.float64) + 0.5
    j = n - np.arange(n, dtype=np.float64) - 0.5
    w = (i * j) ** (-p)
    if w.sum() == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _distribution_power_asym(n: int, a: float, b: float) -> np.ndarray:
    """w_i ∝ (i+½)^a · (n‑i‑½)^b  (a and b may be negative)."""
    i = np.arange(n, dtype=np.float64) + 0.5
    j = n - np.arange(n, dtype=np.float64) - 0.5
    w = (i ** a) * (j ** b)
    if w.sum() == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _distribution_exponential(n: int, gamma_: float) -> np.ndarray:
    """Symmetric exponential taper w_i ∝ exp(−γ·i)+exp(−γ·(n−1−i))."""
    i = np.arange(n, dtype=np.float64)
    w = np.exp(-gamma_ * i) + np.exp(-gamma_ * (n - 1 - i))
    w_sum = w.sum()
    if w_sum == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w_sum
    return w


def _distribution_cosine_edge(n: int, p: float) -> np.ndarray:
    """w_i ∝ |cos(π·i/(n‑1))|^p  – heavy edge mass for larger p."""
    theta = np.pi * np.arange(n, dtype=np.float64) / (n - 1)
    w = np.abs(np.cos(theta)) ** p
    if w.sum() == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _distribution_random_sparse(
    n: int, rng: np.random.Generator, density: float = 0.01
) -> np.ndarray:
    """Sparse random vector (few non‑zeros)."""
    k = max(2, int(n * density))
    idx = rng.choice(n, size=k, replace=False)
    vals = np.zeros(n, dtype=np.float64)
    vals[idx] = rng.random(k)
    if vals.sum() == 0.0:
        vals[0] = 1.0
    vals /= vals.sum()
    return vals


def _mix_with_uniform(base: np.ndarray, lam: float) -> np.ndarray:
    """Linear blend lam·U + (1‑lam)·base, U is the uniform distribution."""
    n = base.shape[0]
    uniform = np.full(n, 1.0 / n, dtype=np.float64)
    return lam * uniform + (1.0 - lam) * base


# ----------------------------------------------------------------------
def _proj_grad_descent(
    a: np.ndarray,
    deadline: float,
    best_seq: np.ndarray,
    best_score: float,
    eta0: float = 0.6,
    jitter_interval: int = 1500,
) -> tuple[np.ndarray, float]:
    """
    Sub‑gradient descent on 2·n·max(conv(a,a)) while staying on the probability simplex.
    """
    n = a.shape[0]
    t = 0
    no_improve = 0
    rng = np.random.default_rng()
    cur_score = _objective(a)

    while time.time() < deadline and t < 120_000:
        t += 1

        # convolution and maximal index(es)
        conv = _fft_self_convolve(a)
        max_val = conv.max()
        cur_score = 2.0 * n * max_val

        if cur_score < best_score - 1e-14:
            best_score = cur_score
            best_seq = a.copy()
            no_improve = 0
            print(f"[{time.time():.1f}] proj‑grad it={t} new best {best_score:.6f}")
        else:
            no_improve += 1

        # sub‑gradient of the maximal entry (sum if several maxima)
        ks = np.where(conv == max_val)[0]
        grad = np.zeros_like(a)
        rev_a = a[::-1]

        for k in ks:
            offset = n - 1 - k
            if offset >= 0:
                # forward slice
                i_start = 0
                i_end = n - offset
                grad[i_start:i_end] += 2.0 * rev_a[offset:offset + i_end - i_start]
            else:
                # tail slice
                off = -offset
                i_start = off
                i_end = n
                grad[i_start:i_end] += 2.0 * rev_a[0:i_end - i_start]

        eta = eta0 / math.sqrt(t)
        a_new = a - eta * grad
        a_new = _project_onto_simplex(a_new, s=1.0)

        if no_improve >= jitter_interval:
            jitter = rng.exponential(scale=0.003, size=n)
            a_new = a_new + jitter
            a_new = _project_onto_simplex(a_new, s=1.0)
            no_improve = 0

        a = a_new

    return best_seq, best_score


# ----------------------------------------------------------------------
def _pairwise_swap_local(
    a: np.ndarray,
    deadline: float,
    best_seq: np.ndarray,
    best_score: float,
    max_trials: int | None = None,
) -> tuple[np.ndarray, float]:
    """Move a tiny fraction of mass from i to j if the objective improves."""
    n = a.shape[0]
    trials = 0
    rng = np.random.default_rng()
    while time.time() < deadline:
        if max_trials is not None and trials >= max_trials:
            break
        trials += 1

        src = np.where(a > 1e-12)[0]
        if src.size == 0:
            break
        i = rng.choice(src)
        j = rng.integers(n - 1)
        if j >= i:
            j += 1

        delta = a[i] * rng.random() * 0.35
        if delta < 1e-14:
            continue

        new_a = a.copy()
        new_a[i] -= delta
        new_a[j] += delta

        cur = _objective(new_a)
        if cur < best_score - 1e-14:
            a = new_a
            best_score = cur
            best_seq = a.copy()
            print(
                f"[{time.time():.1f}] swap i={i}→j={j} δ={delta:.3e} new best {best_score:.6f}"
            )
    return best_seq, best_score


# ----------------------------------------------------------------------
def _random_local_search(
    a: np.ndarray,
    deadline: float,
    best_seq: np.ndarray,
    best_score: float,
    init_temp: float = 1e-2,
    decay: float = 0.9995,
) -> tuple[np.ndarray, float]:
    """Very cheap simulated‑annealing walk."""
    n = a.shape[0]
    rng = np.random.default_rng()
    T = init_temp
    while time.time() < deadline:
        i = rng.integers(n)
        j = rng.integers(n - 1)
        if j >= i:
            j += 1
        delta = a[i] * rng.random() * 0.3
        if delta < 1e-14:
            continue
        new_a = a.copy()
        new_a[i] -= delta
        new_a[j] += delta
        cur = _objective(new_a)
        if cur < best_score - 1e-14:
            a = new_a
            best_score = cur
            best_seq = a.copy()
            print(
                f"[{time.time():.1f}] SA improve {i}->{j} Δ={delta:.3e} best {best_score:.6f}"
            )
        else:
            if rng.random() < math.exp(-(cur - best_score) / max(T, 1e-12)):
                a = new_a
        T *= decay
    return best_seq, best_score


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: int = 1000) -> list[float]:
    """
    Return a list of non‑negative heights (the step function) that minimises
    ``evaluate_sequence``.  The routine runs for at most *budget_s* seconds.
    """
    rng = np.random.default_rng(seed)
    start = time.time()
    deadline = start + budget_s - 4.0      # safety margin

    overall_best_seq = None
    overall_best_score = float("inf")
    all_candidates: list[tuple[float, np.ndarray]] = []

    # ------------------------------------------------------------------
    # 1. Exhaustive / random generation of analytic families
    lengths = [2500, 3500, 5000, 8000, 12000]
    # dense grid for edge‑weighted exponent
    edge_ps = np.linspace(0.05, 1.00, 20)          # more than the original coarse list
    # asymmetric power exponents
    asym_exps = np.linspace(-1.0, 1.0, 15)
    # exponential taper
    gammas = np.linspace(0.0004, 0.008, 12)
    # cosine edge exponent
    cos_ps = np.linspace(0.5, 3.0, 12)
    # mixing factor with uniform
    mix_lams = np.linspace(0.0, 1.0, 11)

    for n in lengths:
        if time.time() >= deadline:
            break
        print(f"[{time.time():.1f}] ------------------------------------------------")
        print(f"[{time.time():.1f}] exploring length {n}")

        # --- uniform --------------------------------------------------
        cand = _distribution_uniform(n)
        sc = _objective(cand)
        all_candidates.append((sc, cand.copy()))
        if sc < overall_best_score:
            overall_best_score, overall_best_seq = sc, cand.copy()
            print(f"[{time.time():.1f}]  new global best (uniform) len={n} score={sc:.6f}")

        # --- edge weighted (symmetric) --------------------------------
        for p in edge_ps:
            base = _distribution_edge_weighted(n, p)
            for lam in mix_lams:
                cand = _mix_with_uniform(base, lam)
                sc = _objective(cand)
                all_candidates.append((sc, cand.copy()))
                if sc < overall_best_score:
                    overall_best_score, overall_best_seq = sc, cand.copy()
                    print(
                        f"[{time.time():.1f}]  new best edge p={p:.3f} lam={lam:.2f} len={n} score={sc:.6f}"
                    )

        # --- asymmetric power law --------------------------------------
        for a_exp in asym_exps:
            for b_exp in asym_exps:
                base = _distribution_power_asym(n, a_exp, b_exp)
                for lam in mix_lams:
                    cand = _mix_with_uniform(base, lam)
                    sc = _objective(cand)
                    all_candidates.append((sc, cand.copy()))
                    if sc < overall_best_score:
                        overall_best_score, overall_best_seq = sc, cand.copy()
                        print(
                            f"[{time.time():.1f}]  new best asym a={a_exp:.2f} b={b_exp:.2f} lam={lam:.2f} len={n} score={sc:.6f}"
                        )

        # --- exponential taper -----------------------------------------
        for gamma_ in gammas:
            base = _distribution_exponential(n, gamma_)
            for lam in mix_lams:
                cand = _mix_with_uniform(base, lam)
                sc = _objective(cand)
                all_candidates.append((sc, cand.copy()))
                if sc < overall_best_score:
                    overall_best_score, overall_best_seq = sc, cand.copy()
                    print(
                        f"[{time.time():.1f}]  new best exp γ={gamma_:.5f} lam={lam:.2f} len={n} score={sc:.6f}"
                    )

        # --- cosine‑edge ------------------------------------------------
        for p in cos_ps:
            base = _distribution_cosine_edge(n, p)
            for lam in mix_lams:
                cand = _mix_with_uniform(base, lam)
                sc = _objective(cand)
                all_candidates.append((sc, cand.copy()))
                if sc < overall_best_score:
                    overall_best_score, overall_best_seq = sc, cand.copy()
                    print(
                        f"[{time.time():.1f}]  new best cos p={p:.2f} lam={lam:.2f} len={n} score={sc:.6f}"
                    )

        # --- random Dirichlet -------------------------------------------
        for _ in range(6):
            alpha = rng.uniform(0.01, 2.0)
            cand = rng.dirichlet(np.full(n, alpha))
            sc = _objective(cand)
            all_candidates.append((sc, cand.copy()))
            if sc < overall_best_score:
                overall_best_score, overall_best_seq = sc, cand.copy()
                print(
                    f"[{time.time():.1f}]  new best Dirichlet α={alpha:.3g} len={n} score={sc:.6f}"
                )

        # --- sparse random vectors --------------------------------------
        for _ in range(4):
            cand = _distribution_random_sparse(n, rng, density=0.012)
            sc = _objective(cand)
            all_candidates.append((sc, cand.copy()))
            if sc < overall_best_score:
                overall_best_score, overall_best_seq = sc, cand.copy()
                print(
                    f"[{time.time():.1f}]  new best sparse len={n} score={sc:.6f}"
                )

    # ------------------------------------------------------------------
    # Keep a limited set of the very best seeds
    all_candidates.sort(key=lambda x: x[0])
    top_seeds = [seq for _, seq in all_candidates[:12]]
    print(f"[{time.time():.1f}] kept {len(top_seeds)} top seeds for refinement")

    # ------------------------------------------------------------------
    # 2. Projected sub‑gradient descent on the most promising seeds
    remaining = deadline - time.time()
    if remaining > 30.0 and top_seeds:
        per_seed = max(5.0, remaining / len(top_seeds))   # allocate at least a few seconds per seed
        print(f"[{time.time():.1f}] starting sub‑gradient refinement (≈{per_seed:.1f}s per seed)")
        for idx, seed_seq in enumerate(top_seeds):
            if time.time() >= deadline:
                break
            sub_deadline = time.time() + per_seed
            a_opt, sc_opt = _proj_grad_descent(
                seed_seq.copy(),
                sub_deadline,
                best_seq=overall_best_seq,
                best_score=overall_best_score,
                eta0=0.6,
                jitter_interval=2000,
            )
            if sc_opt < overall_best_score:
                overall_best_score, overall_best_seq = sc_opt, a_opt.copy()
                print(
                    f"[{time.time():.1f}]  after proj‑grad seed {idx} new global best {overall_best_score:.6f}"
                )
            # early exit if we have already passed the target bound
            if overall_best_score <= 1.503:
                break

    # ------------------------------------------------------------------
    # 3. Greedy pairwise refinement
    if time.time() < deadline:
        print(f"[{time.time():.1f}] starting greedy pairwise refinement")
        overall_best_seq, overall_best_score = _pairwise_swap_local(
            overall_best_seq.copy(),
            deadline,
            best_seq=overall_best_seq,
            best_score=overall_best_score,
            max_trials=300_000,
        )

    # ------------------------------------------------------------------
    # 4. Final cheap simulated‑annealing walk
    if time.time() < deadline:
        print(f"[{time.time():.1f}] final simulated‑annealing walk")
        overall_best_seq, overall_best_score = _random_local_search(
            overall_best_seq.copy(),
            deadline,
            best_seq=overall_best_seq,
            best_score=overall_best_score,
            init_temp=0.008,
            decay=0.9997,
        )

    # ------------------------------------------------------------------
    print(
        f"[{time.time():.1f}] DONE – final best score {overall_best_score:.6f} (length={len(overall_best_seq)})"
    )
    # Return plain Python list; the caller will invoke ``evaluate_sequence``.
    return overall_best_seq.tolist()
