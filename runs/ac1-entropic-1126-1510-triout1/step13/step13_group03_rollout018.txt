"""
Improved search for a non‑negative step function that minimizes
``evaluate_sequence``.  The algorithm consists of three stages:

1️⃣  Enumerate a rich collection of low‑dimensional analytic families
    (uniform, arcsine, beta‑shaped, edge‑weighted, exponential‑edge,
    cosine, sine, quadratic) on several lengths.  For each family a
    fine grid of parameters (including mixtures with the uniform
    distribution) is tried and the best few candidates are kept.

2️⃣  Refine the most promising seeds with a *smooth* gradient descent.
    The max‑convolution term is approximated by a softmax
    (temperature τ) so that the gradient is a weighted sum of all
    convolution entries.  The exact weighted gradient is evaluated by
    FFT‑based cross‑correlation and sliding‑window sums, then an
    exponentiated‑gradient (EG) step is performed and the vector is
    projected back onto the probability simplex.

    Temperature is annealed from τ≈2·10⁻² to τ≈10⁻³, which lets the
    optimiser escape shallow plateaux and later focus on the true L∞
    objective.

3️⃣  Polish the result with a greedy pairwise mass‑transfer hill‑climber.
    Small random moves are kept only if they strictly improve the exact
    objective.

All stages respect the overall time budget (default 1000 s).  The
function returns the best sequence found as a plain Python list – the
calling environment will re‑normalise it inside ``evaluate_sequence``.
"""

def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x ≥ 0, Σx = s}."""
    if s < 0:
        raise ValueError("Simplex radius must be non‑negative")
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.where(u - cssv / (np.arange(1, len(v) + 1)) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    return np.maximum(v - theta, 0.0)


def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Full linear self‑convolution a*a using a real FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1) - 1).bit_length()   # next power of two
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[: 2 * n - 1]


def _fft_cross_correlate_valid(w: np.ndarray, a: np.ndarray) -> np.ndarray:
    """
    Linear cross‑correlation (valid part)  (w ⋆ a)[j] = Σ_i w[i+j]·a[i],
    returning an array of length len(a).  Implemented with FFT for speed.
    """
    n = a.shape[0]
    m = w.shape[0]                # = 2·n‑1
    size = 1 << ((m + n - 1) - 1).bit_length()
    W = np.fft.rfft(w, n=size)
    A = np.fft.rfft(a, n=size)
    corr = np.fft.irfft(W * np.conj(A), n=size)
    return corr[:n]               # valid region


def _fft_sliding_window_sum(w: np.ndarray, win: int) -> np.ndarray:
    """
    For each j compute Σ_{i=j}^{j+win‑1} w[i] (window length = win).
    Returns an array of length win (i.e. len(a)).  Implemented via FFT.
    """
    m = w.shape[0]
    size = 1 << ((m + win - 1) - 1).bit_length()
    W = np.fft.rfft(w, n=size)
    ones = np.ones(win, dtype=np.float64)
    O = np.fft.rfft(ones, n=size)
    conv = np.fft.irfft(W * O, n=size)
    # the full linear convolution has length m+win‑1; the fully‑covered
    # window sums sit at indices win‑1 … win‑1+win‑1
    start = win - 1
    return conv[start:start + win]

def _distribution_uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _distribution_arcsine(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    vals /= vals.sum()
    return vals


def _distribution_beta_edge(n: int, alpha: float) -> np.ndarray:
    """Symmetric beta‑type density  w_i ∝ (i·(n‑i))^{α‑1}  (α>0)."""
    i = np.arange(n, dtype=np.float64) + 0.5
    vals = (i * (n - i)) ** (alpha - 1.0)
    vals = np.clip(vals, 0.0, None)
    s = vals.sum()
    if s == 0.0:
        return _distribution_uniform(n)
    vals /= s
    return vals


def _distribution_edge_weighted(n: int, p: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64) + 0.5
    j = (n - np.arange(n, dtype=np.float64) - 0.5)
    w = (i * j) ** (-p)
    w_sum = w.sum()
    if w_sum == 0.0:
        return _distribution_uniform(n)
    w /= w_sum
    return w


def _distribution_exponential_edge(n: int, c: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    w = np.exp(-c * i) + np.exp(-c * (n - 1 - i))
    w /= w.sum()
    return w


def _distribution_sine(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64) + 0.5
    w = np.sin(np.pi * i / n)
    w /= w.sum()
    return w


def _distribution_cosine(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64) + 0.5
    w = 1.0 - np.cos(np.pi * i / n)
    w /= w.sum()
    return w


def _distribution_quadratic(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    x = (2.0 * i) / (n - 1) - 1.0
    w = 1.0 - x * x
    w = np.clip(w, 0.0, None)
    w /= w.sum()
    return w


def _distribution_random_sparse(
    n: int, rng: np.random.Generator, density: float = 0.01
) -> np.ndarray:
    """Sparse Dirichlet‑like seed with a prescribed density of non‑zeros."""
    k = max(2, int(n * density))
    idx = rng.choice(n, size=k, replace=False)
    vals = np.zeros(n, dtype=np.float64)
    vals[idx] = rng.random(k)
    s = vals.sum()
    if s == 0.0:
        vals[0] = 1.0
        s = 1.0
    vals /= s
    return vals


def _mix_with_uniform(base: np.ndarray, lam: float) -> np.ndarray:
    """Convex combination λ·uniform + (1‑λ)·base."""
    n = base.shape[0]
    uniform = np.full(n, 1.0 / n, dtype=np.float64)
    return lam * uniform + (1.0 - lam) * base


# ----------------------------------------------------------------------
# Exact objective (sum = 1 ⇒ 2·n·max(conv))
# ----------------------------------------------------------------------
def _objective(a: np.ndarray) -> float:
    n = a.shape[0]
    b = _fft_self_convolve(a)
    return float(2.0 * n * b.max())


# ----------------------------------------------------------------------
# Smooth (soft‑max) gradient descent
# ----------------------------------------------------------------------
def _smooth_descent(
    a0: np.ndarray,
    time_limit: float,
    best_seq: np.ndarray,
    best_score: float,
    tau_start: float = 0.02,
    tau_end: float = 0.001,
    lr0: float = 0.6,
    eval_every: int = 200,
) -> tuple[np.ndarray, float]:
    """
    Exponentiated‑gradient descent on a smoothed version of the L∞ objective.
    The soft‑max temperature τ is linearly annealed from ``tau_start`` to
    ``tau_end``.  The (exact) sub‑gradient of the smoothed objective is
    computed via FFT‑based cross‑correlation.
    """
    a = a0.copy()
    n = a.shape[0]
    start = time.time()
    it = 0
    # pre‑allocate the FFT of the all‑ones window (re‑used each iteration)
    ones = np.ones(n, dtype=np.float64)

    while time.time() < time_limit:
        it += 1
        conv = _fft_self_convolve(a)
        max_val = conv.max()
        cur_score = 2.0 * n * max_val
        if cur_score < best_score - 1e-14:
            best_score = cur_score
            best_seq = a.copy()
            print(
                f"[{time.time():.1f}] smooth it={it} new global best {best_score:.6f}"
            )

        # linear annealing of τ
        frac = (time.time() - start) / (time_limit - start)
        tau = tau_start * (1.0 - frac) + tau_end * frac

        # soft‑max weights (stable version)
        maxc = conv.max()
        shifted = conv - maxc
        exp_vals = np.exp(shifted / tau)          # all ≤ 1
        w = exp_vals / exp_vals.sum()             # length 2n‑1, sums to 1

        # gradient of the smoothed objective
        term1 = _fft_cross_correlate_valid(w, a)   # Σ w[i+j]·a[i]
        win_sum = _fft_sliding_window_sum(w, n)   # Σ_{k=j}^{j+n‑1} w[k]
        grad = term1 + a * win_sum

        # EG update (step‑size ∝ 1/√t)
        eta = lr0 / math.sqrt(it)
        a *= np.exp(-eta * grad)
        s = a.sum()
        if s <= 0.0:
            a = np.full(n, 1.0 / n, dtype=np.float64)
        else:
            a /= s

        if it % eval_every == 0:
            # quick (approx.) check already performed above
            pass

    return best_seq, best_score


# ----------------------------------------------------------------------
# Exact sub‑gradient descent (same as original, kept for final polishing)
# ----------------------------------------------------------------------
def _subgradient_descent_exact(
    a0: np.ndarray,
    time_limit: float,
    best_seq: np.ndarray,
    best_score: float,
    lr0: float = 0.9,
    eval_every: int = 100,
) -> tuple[np.ndarray, float]:
    """Exponentiated‑gradient descent on the true L∞ objective."""
    a = a0.copy()
    n = a.shape[0]
    t = 0
    while time.time() < time_limit:
        t += 1
        conv = _fft_self_convolve(a)
        max_val = conv.max()
        max_idxs = np.where(conv == max_val)[0]

        grad = np.zeros_like(a)
        for idx in max_idxs:
            i_start = max(0, idx - (n - 1))
            i_end = min(n - 1, idx)
            i = np.arange(i_start, i_end + 1, dtype=np.int64)
            j = idx - i
            grad[i] += 2.0 * a[j]

        eta = lr0 / math.sqrt(t)
        a *= np.exp(-eta * grad)
        s = a.sum()
        if s <= 0.0:
            a = np.full(n, 1.0 / n, dtype=np.float64)
        else:
            a /= s

        if t % eval_every == 0:
            cur = 2.0 * n * max_val
            if cur < best_score - 1e-14:
                best_score = cur
                best_seq = a.copy()
                print(
                    f"[{time.time():.1f}] exact subgrad it={t} new global best {best_score:.6f}"
                )
    return best_seq, best_score


# ----------------------------------------------------------------------
# Greedy pairwise hill‑climber (same as original)
# ----------------------------------------------------------------------
def _pairwise_swap(
    a0: np.ndarray,
    time_limit: float,
    best_seq: np.ndarray,
    best_score: float,
    max_trials: int | None = None,
) -> tuple[np.ndarray, float]:
    a = a0.copy()
    n = a.shape[0]
    trials = 0
    while time.time() < time_limit:
        if max_trials is not None and trials >= max_trials:
            break
        trials += 1

        src = np.where(a > 1e-14)[0]
        if src.size == 0:
            break
        i = np.random.choice(src)
        j = np.random.randint(n - 1)
        if j >= i:
            j += 1

        delta = a[i] * np.random.rand() * 0.45   # up to 45 % of source mass
        if delta < 1e-15:
            continue

        new_a = a.copy()
        new_a[i] -= delta
        new_a[j] += delta

        cur = _objective(new_a)
        if cur < best_score - 1e-14:
            a = new_a
            best_score = cur
            best_seq = a.copy()
            print(
                f"[{time.time():.1f}] swap i={i}→j={j} δ={delta:.6g} new best {best_score:.6f}"
            )
    return best_seq, best_score


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: int = 1000) -> list[float]:
    """
    Return a step‑function (plain Python list) that minimises
    ``evaluate_sequence`` within the given time budget.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 2.0          # safety margin

    overall_best_seq = None
    overall_best_score = float("inf")
    top_candidates: list[tuple[float, np.ndarray]] = []   # (score, seq)

    # ------------------------------------------------------------------
    # 1. Scan analytic families on several lengths
    # ------------------------------------------------------------------
    lengths = [1000, 2000, 3000, 4000, 6000, 8000, 12000]

    # parameter grids
    p_grid = np.concatenate(
        [
            np.logspace(-3, -0.5, 15),      # tiny to moderate edge‑weight
            np.logspace(-0.5, 0.0, 10),     # up to 1
            np.linspace(0.1, 3.0, 12),      # broader
        ]
    )
    alpha_grid = np.linspace(0.05, 0.50, 10)   # beta‑type exponent α
    lam_grid = np.linspace(0.0, 1.0, 21)      # finer mixing with uniform
    c_grid = np.logspace(-3, 0, 12)          # exponential‑edge decay

    for n in lengths:
        if time.time() >= deadline:
            break
        print(f"[{time.time():.1f}] scanning length {n}")

        # deterministic families
        for base_fn in (
            _distribution_uniform,
            _distribution_arcsine,
            _distribution_sine,
            _distribution_cosine,
            _distribution_quadratic,
        ):
            cand = base_fn(n)
            sc = _objective(cand)
            top_candidates.append((sc, cand.copy()))
            if sc < overall_best_score:
                overall_best_score, overall_best_seq = sc, cand.copy()
                print(f"[{time.time():.1f}] new global best len={n} score={sc:.6f}")

        # beta‑type (α) families
        for alpha in alpha_grid:
            if time.time() >= deadline:
                break
            beta = _distribution_beta_edge(n, float(alpha))
            for lam in lam_grid:
                if time.time() >= deadline:
                    break
                cand = _mix_with_uniform(beta, lam)
                sc = _objective(cand)
                top_candidates.append((sc, cand.copy()))
                if sc < overall_best_score:
                    overall_best_score, overall_best_seq = sc, cand.copy()
                    print(f"[{time.time():.1f}] len={n} β α={alpha:.3g} lam={lam:.2f} score={sc:.6f}")

        # edge‑weighted families (p) with uniform mix
        for p in p_grid:
            if time.time() >= deadline:
                break
            edge = _distribution_edge_weighted(n, float(p))
            for lam in lam_grid:
                if time.time() >= deadline:
                    break
                cand = _mix_with_uniform(edge, lam)
                sc = _objective(cand)
                top_candidates.append((sc, cand.copy()))
                if sc < overall_best_score:
                    overall_best_score, overall_best_seq = sc, cand.copy()
                    print(f"[{time.time():.1f}] len={n} edge p={p:.4g} lam={lam:.2f} score={sc:.6f}")

        # exponential‑edge families
        for c in c_grid:
            if time.time() >= deadline:
                break
            exp_edge = _distribution_exponential_edge(n, float(c))
            for lam in lam_grid:
                if time.time() >= deadline:
                    break
                cand = _mix_with_uniform(exp_edge, lam)
                sc = _objective(cand)
                top_candidates.append((sc, cand.copy()))
                if sc < overall_best_score:
                    overall_best_score, overall_best_seq = sc, cand.copy()
                    print(f"[{time.time():.1f}] len={n} exp c={c:.4g} lam={lam:.2f} score={sc:.6f}")

        # random Dirichlet seeds
        for _ in range(8):
            if time.time() >= deadline:
                break
            alpha = rng.uniform(0.01, 2.0)
            a = rng.dirichlet(np.full(n, alpha))
            sc = _objective(a)
            top_candidates.append((sc, a.copy()))
            if sc < overall_best_score:
                overall_best_score, overall_best_seq = sc, a.copy()
                print(f"[{time.time():.1f}] Dirichlet len={n} score={sc:.6f}")

        # sparse random seeds
        for _ in range(8):
            if time.time() >= deadline:
                break
            a = _distribution_random_sparse(n, rng, density=0.007)
            sc = _objective(a)
            top_candidates.append((sc, a.copy()))
            if sc < overall_best_score:
                overall_best_score, overall_best_seq = sc, a.copy()
                print(f"[{time.time():.1f}] sparse len={n} score={sc:.6f}")

        # keep only the most promising seeds (memory / time bound)
        top_candidates.sort(key=lambda x: x[0])
        top_candidates = top_candidates[:60]

    # ------------------------------------------------------------------
    # 2. Incorporate a previous construction if available
    # ------------------------------------------------------------------
    try:
        prev = np.asarray(height_sequence_1, dtype=np.float64)  # type: ignore[name-defined]
        if prev.ndim == 1 and prev.size > 0:
            prev = np.maximum(prev, 0.0)
            if prev.sum() == 0.0:
                prev = _distribution_uniform(prev.size)
            else:
                prev = prev / prev.sum()
            sc = _objective(prev)
            if sc < overall_best_score:
                overall_best_score, overall_best_seq = sc, prev.copy()
                print(f"[{time.time():.1f}] previous construction score={sc:.6f}")
            top_candidates.append((sc, prev.copy()))
    except Exception:
        pass

    # ------------------------------------------------------------------
    # 3. Fallback (should never trigger)
    # ------------------------------------------------------------------
    if overall_best_seq is None:
        n = 2000
        overall_best_seq = _distribution_uniform(n)
        overall_best_score = _objective(overall_best_seq)
        print(
            f"[{time.time():.1f}] fallback uniform length={n} score={overall_best_score:.6f}"
        )

    # ------------------------------------------------------------------
    # 4. Smooth (soft‑max) gradient descent on the best seeds
    # ------------------------------------------------------------------
    remaining = deadline - time.time()
    if remaining > 30.0 and top_candidates:
        # keep the five best seeds for deeper optimisation
        seeds = [seq for _, seq in sorted(top_candidates, key=lambda x: x[0])[:5]]
        per_seed = remaining / max(1, len(seeds))
        print(f"[{time.time():.1f}] start smooth descent on {len(seeds)} seeds")
        for idx, seed_seq in enumerate(seeds):
            if time.time() >= deadline:
                break
            sg_deadline = time.time() + per_seed * 0.6   # 60 % of allocated time
            a_opt, sc_opt = _smooth_descent(
                seed_seq,
                sg_deadline,
                best_seq=overall_best_seq,
                best_score=overall_best_score,
                tau_start=0.02,
                tau_end=0.001,
                lr0=0.6,
                eval_every=200,
            )
            if sc_opt < overall_best_score - 1e-12:
                overall_best_score, overall_best_seq = sc_opt, a_opt.copy()
                print(
                    f"[{time.time():.1f}] after smooth seed {idx} new global best {overall_best_score:.6f}"
                )
            # small follow‑up with exact sub‑gradient to polish
            sg_deadline = time.time() + per_seed * 0.4
            a_opt, sc_opt = _subgradient_descent_exact(
                overall_best_seq,
                sg_deadline,
                best_seq=overall_best_seq,
                best_score=overall_best_score,
                lr0=0.9,
                eval_every=500,
            )
            if sc_opt < overall_best_score - 1e-12:
                overall_best_score, overall_best_seq = sc_opt, a_opt.copy()
                print(
                    f"[{time.time():.1f}] after exact polish seed {idx} new best {overall_best_score:.6f}"
                )

    # ------------------------------------------------------------------
    # 5. Final greedy pairwise polishing
    # ------------------------------------------------------------------
    if time.time() < deadline:
        print(f"[{time.time():.1f}] final greedy swap polishing")
        a_opt, sc_opt = _pairwise_swap(
            overall_best_seq.copy(),
            deadline,
            best_seq=overall_best_seq,
            best_score=overall_best_score,
        )
        if sc_opt < overall_best_score - 1e-12:
            overall_best_score, overall_best_seq = sc_opt, a_opt.copy()
            print(f"[{time.time():.1f}] after final swaps new best {overall_best_score:.6f}")

    # ------------------------------------------------------------------
    print(
        f"Final best objective (2·n·max(conv)): {overall_best_score:.6f} (length={len(overall_best_seq)})"
    )
    # Return a plain Python list – evaluate_sequence will re‑normalise internally.
    return overall_best_seq.tolist()
