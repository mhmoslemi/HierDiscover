def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Euclidean projection of *v* onto the simplex {x ≥ 0, sum(x)=s}."""
    if s < 0:
        raise ValueError("simplex radius must be non‑negative")
    n = v.shape[0]
    # Fast projection (see Duchi et al., 2008)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.where(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Full self‑convolution a*a using a real FFT."""
    n = a.shape[0]
    need = 2 * n - 1
    size = 1 << ((need - 1).bit_length())          # next power of two
    A = rfft(a, n=size)
    conv = irfft(A * A, n=size)
    return conv[:need]


# ----------------------------------------------------------------------
def _objective(a: np.ndarray) -> float:
    """Exact objective 2·n·max(conv(a,a)) when sum(a)=1."""
    n = a.shape[0]
    conv = _fft_self_convolve(a)
    return 2.0 * n * conv.max()


# ----------------------------------------------------------------------
# Analytic families ----------------------------------------------------
def _distribution_uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _distribution_arcsine(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    vals /= vals.sum()
    return vals


def _distribution_edge_weighted(n: int, p: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64) + 0.5
    j = (n - np.arange(n, dtype=np.float64) - 0.5)
    w = (i * j) ** (-p)
    s = w.sum()
    if s == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= s
    return w


def _distribution_edge_inversepow(n: int, p: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64) + 0.5
    j = (n - np.arange(n, dtype=np.float64) - 0.5)
    w = i ** (-p) + j ** (-p)
    s = w.sum()
    if s == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= s
    return w


def _distribution_sine_weighted(n: int, p: float) -> np.ndarray:
    theta = np.pi * (np.arange(n, dtype=np.float64) + 0.5) / n
    w = np.sin(theta) ** p
    s = w.sum()
    if s == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= s
    return w


def _distribution_quadratic(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    half = (n - 1) / 2.0
    d = np.abs(i - half) + 1.0
    w = 1.0 / (d ** 2)
    w /= w.sum()
    return w


def _distribution_exponential_decay(n: int, rate: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    half = (n - 1) / 2.0
    w = np.exp(-rate * np.abs(i - half))
    w /= w.sum()
    return w


def _distribution_dirichlet(n: int, rng: np.random.Generator) -> np.ndarray:
    alpha = rng.uniform(0.01, 2.0)
    a = rng.dirichlet(np.full(n, alpha))
    return a.astype(np.float64)


def _distribution_random_sparse(
    n: int, rng: np.random.Generator, density: float = 0.004
) -> np.ndarray:
    k = max(2, int(n * density))
    idx = rng.choice(n, size=k, replace=False)
    vals = np.zeros(n, dtype=np.float64)
    vals[idx] = rng.random(k)
    if vals.sum() == 0.0:
        vals[0] = 1.0
    vals /= vals.sum()
    return vals


def _mix_with_uniform(base: np.ndarray, lam: float) -> np.ndarray:
    """Convex blend lam·U + (1‑lam)·base."""
    n = base.shape[0]
    uniform = np.full(n, 1.0 / n, dtype=np.float64)
    return lam * uniform + (1.0 - lam) * base


# ----------------------------------------------------------------------
# Gradient helpers -----------------------------------------------------
def _active_gradient(a: np.ndarray, act: np.ndarray) -> np.ndarray:
    """Sub‑gradient from the set of active convolution indices."""
    n = a.shape[0]
    grad = np.zeros_like(a)
    ind = np.arange(n, dtype=np.int32)
    for idx in act:
        offset = idx - ind
        mask = (offset >= 0) & (offset < n)
        grad[mask] += 2.0 * a[offset[mask]]
    return grad


# ----------------------------------------------------------------------
# Mirror‑descent (exponentiated‑gradient) -------------------------------
def _mirror_descent(
    a: np.ndarray,
    deadline: float,
    best_seq: np.ndarray,
    best_score: float,
    lr0: float = 0.7,
    active_frac: float = 0.03,
    max_active: int = 20,
    beta: float = 0.9,
    eval_interval: int = 200,
    patience: int = 3000,
    average: bool = False,
) -> tuple[np.ndarray, float]:
    """
    Exponentiated‑gradient descent with momentum.
    If *average* is True the gradient from the active set is averaged
    (instead of summed) before the mirror step.
    """
    n = a.shape[0]
    t = 0
    no_improve = 0
    momentum = np.zeros_like(a)
    rng = np.random.default_rng()
    ind_grid = np.arange(n, dtype=np.int32)

    while time.time() < deadline and t < 500_000:
        t += 1
        conv = _fft_self_convolve(a)
        max_c = conv.max()
        thresh = max_c * (1.0 - active_frac)

        act = np.where(conv >= thresh)[0]
        if act.size > max_active:
            top_idx = np.argpartition(conv[act], -max_active)[-max_active:]
            act = act[top_idx]

        grad = _active_gradient(a, act)
        if average:
            grad = grad / max(1, act.size)

        momentum = beta * momentum + (1.0 - beta) * grad
        eta = lr0 / math.sqrt(t)
        a = a * np.exp(-eta * momentum)
        a = _project_onto_simplex(a, s=1.0)

        if t % eval_interval == 0:
            cur = _objective(a)
            if cur < best_score - 1e-14:
                best_score, best_seq = cur, a.copy()
                no_improve = 0
                print(f"[{time.time():.1f}] it={t:,} mirror new best {best_score:.6f}")
            else:
                no_improve += 1

        if no_improve >= patience:
            jitter = rng.exponential(scale=0.006, size=n)
            a = np.maximum(a + jitter, 0.0)
            a = _project_onto_simplex(a, s=1.0)
            no_improve = 0

    return best_seq, best_score


# ----------------------------------------------------------------------
# Adam‑style adaptive gradient descent ----------------------------------
def _adam_descent(
    a: np.ndarray,
    deadline: float,
    best_seq: np.ndarray,
    best_score: float,
    lr0: float = 0.02,
    active_frac: float = 0.04,
    max_active: int = 25,
    beta1: float = 0.9,
    beta2: float = 0.999,
    eps: float = 1e-8,
    eval_interval: int = 250,
    patience: int = 3000,
) -> tuple[np.ndarray, float]:
    """
    Adam optimiser on the same sub‑gradient (active‑set) objective.
    """
    n = a.shape[0]
    t = 0
    no_improve = 0
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    rng = np.random.default_rng()

    while time.time() < deadline and t < 500_000:
        t += 1
        conv = _fft_self_convolve(a)
        max_c = conv.max()
        thresh = max_c * (1.0 - active_frac)
        act = np.where(conv >= thresh)[0]
        if act.size > max_active:
            top_idx = np.argpartition(conv[act], -max_active)[-max_active:]
            act = act[top_idx]

        grad = _active_gradient(a, act) / max(1, act.size)   # average

        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)

        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)

        lr = lr0 / math.sqrt(t)
        a = a - lr * m_hat / (np.sqrt(v_hat) + eps)
        a = _project_onto_simplex(a, s=1.0)

        if t % eval_interval == 0:
            cur = _objective(a)
            if cur < best_score - 1e-14:
                best_score, best_seq = cur, a.copy()
                no_improve = 0
                print(f"[{time.time():.1f}] it={t:,} adam  new best {best_score:.6f}")
            else:
                no_improve += 1

        if no_improve >= patience:
            jitter = rng.exponential(scale=0.006, size=n)
            a = np.maximum(a + jitter, 0.0)
            a = _project_onto_simplex(a, s=1.0)
            no_improve = 0

    return best_seq, best_score


# ----------------------------------------------------------------------
# Greedy pairwise refinement -------------------------------------------
def _pairwise_swap_local(
    a: np.ndarray,
    deadline: float,
    best_seq: np.ndarray,
    best_score: float,
    max_trials: int | None = None,
) -> tuple[np.ndarray, float]:
    """Move a random amount of mass between two positions if objective improves."""
    n = a.shape[0]
    trials = 0
    rng = np.random.default_rng()
    while time.time() < deadline:
        if max_trials is not None and trials >= max_trials:
            break
        trials += 1

        src = np.where(a > 1e-12)[0]
        if src.size == 0:
            break
        i = rng.choice(src)
        j = rng.integers(n - 1)
        if j >= i:
            j += 1

        delta = a[i] * rng.random() * 0.35
        if delta < 1e-14:
            continue

        new_a = a.copy()
        new_a[i] -= delta
        new_a[j] += delta
        cur = _objective(new_a)

        if cur < best_score - 1e-14:
            a = new_a
            best_score = cur
            best_seq = a.copy()
            print(f"[{time.time():.1f}] swap {i}->{j} Δ={delta:.3e} new best {best_score:.6f}")

    return best_seq, best_score


# ----------------------------------------------------------------------
# Simulated‑annealing style local search -------------------------------
def _random_local_search(
    a: np.ndarray,
    deadline: float,
    best_seq: np.ndarray,
    best_score: float,
    init_temp: float = 0.008,
    decay: float = 0.9997,
) -> tuple[np.ndarray, float]:
    """Very cheap SA that repeatedly transfers a random amount of mass."""
    n = a.shape[0]
    rng = np.random.default_rng()
    T = init_temp
    while time.time() < deadline:
        i = rng.integers(n)
        j = rng.integers(n - 1)
        if j >= i:
            j += 1
        delta = a[i] * rng.random() * 0.3
        if delta < 1e-14:
            continue
        new_a = a.copy()
        new_a[i] -= delta
        new_a[j] += delta
        cur = _objective(new_a)
        if cur < best_score - 1e-14:
            a = new_a
            best_score = cur
            best_seq = a.copy()
            print(
                f"[{time.time():.1f}] SA improve {i}->{j} Δ={delta:.3e} best {best_score:.6f}"
            )
        else:
            if rng.random() < math.exp(-(cur - best_score) / max(T, 1e-12)):
                a = new_a
        T *= decay
    return best_seq, best_score

def propose_candidate(seed: int = 42, budget_s: int = 1000) -> list[float]:
    """
    Search for a non‑negative step‑function that minimises ``evaluate_sequence``.
    Returns a plain Python list of heights (sum≈1, all ≥0).
    """
    rng = np.random.default_rng(seed)
    start = time.time()
    deadline = start + budget_s - 2.0  # 2 s safety margin

    overall_best_seq: np.ndarray | None = None
    overall_best_score = float("inf")
    top_candidates: list[tuple[float, np.ndarray]] = []

    # ------------------------------------------------------------------
    # 1. Diversified seed pool
    lengths = [2000, 3000, 4000, 6000, 8000, 12000, 16000]
    p_grid = np.concatenate(
        [
            np.logspace(-3, -0.5, 8, base=10),  # 0.001 … 0.316
            np.logspace(-0.5, 0.0, 8, base=10),  # 0.316 … 1
            np.logspace(0.0, 0.8, 6, base=10),   # 1 … 6.3
        ]
    )
    lam_grid = np.linspace(0.0, 0.6, 7)
    rate_grid = np.linspace(0.001, 0.02, 6)

    for n in lengths:
        if time.time() >= deadline:
            break
        print(f"[{time.time():.1f}] scanning length {n}")

        # deterministic families
        for cand in (
            _distribution_uniform(n),
            _distribution_arcsine(n),
            _distribution_quadratic(n),
            _distribution_tent(n),
        ):
            sc = _objective(cand)
            top_candidates.append((sc, cand.copy()))
            if sc < overall_best_score:
                overall_best_score, overall_best_seq = sc, cand.copy()
                print(f"[{time.time():.1f}] new global best len={n} score={sc:.6f}")

        # parametrised families + mixing with uniform
        for p in p_grid:
            if time.time() >= deadline:
                break
            edge = _distribution_edge_weighted(n, p)
            edge2 = _distribution_edge_inversepow(n, p)
            sine = _distribution_sine_weighted(n, p)
            for base in (edge, edge2, sine):
                for lam in lam_grid:
                    if time.time() >= deadline:
                        break
                    cand = _mix_with_uniform(base, lam)
                    sc = _objective(cand)
                    top_candidates.append((sc, cand.copy()))
                    if sc < overall_best_score:
                        overall_best_score, overall_best_seq = sc, cand.copy()
                        print(
                            f"[{time.time():.1f}] new global best len={n} p={p:.3g} lam={lam:.2f} score={sc:.6f}"
                        )

        # exponential‑decay family
        for rate in rate_grid:
            if time.time() >= deadline:
                break
            cand = _distribution_exponential_decay(n, rate)
            sc = _objective(cand)
            top_candidates.append((sc, cand.copy()))
            if sc < overall_best_score:
                overall_best_score, overall_best_seq = sc, cand.copy()
                print(
                    f"[{time.time():.1f}] new global best exp‑decay n={n} rate={rate:.4f} score={sc:.6f}"
                )

        # random Dirichlet seeds
        for _ in range(6):
            if time.time() >= deadline:
                break
            a = _distribution_dirichlet(n, rng)
            sc = _objective(a)
            top_candidates.append((sc, a.copy()))
            if sc < overall_best_score:
                overall_best_score, overall_best_seq = sc, a.copy()
                print(
                    f"[{time.time():.1f}] new global best Dirichlet len={n} score={sc:.6f}"
                )

        # very sparse random seeds
        for _ in range(6):
            if time.time() >= deadline:
                break
            a = _distribution_random_sparse(n, rng, density=0.004)
            sc = _objective(a)
            top_candidates.append((sc, a.copy()))
            if sc < overall_best_score:
                overall_best_score, overall_best_seq = sc, a.copy()
                print(
                    f"[{time.time():.1f}] new global best sparse len={n} score={sc:.6f}"
                )

        # keep only the strongest seeds overall (size limited)
        top_candidates.sort(key=lambda x: x[0])
        top_candidates = top_candidates[:30]

    # ------------------------------------------------------------------
    # 2. Pull in a previously discovered construction if available
    try:
        prev = np.asarray(height_sequence_1, dtype=np.float64)  # type: ignore[name-defined]
        if prev.ndim == 1 and prev.size > 0:
            prev = prev / prev.sum()
            sc = _objective(prev)
            if sc < overall_best_score:
                overall_best_score, overall_best_seq = sc, prev.copy()
                print(f"[{time.time():.1f}] previous best score={sc:.6f}")
            top_candidates.append((sc, prev.copy()))
    except Exception:
        pass

    # ------------------------------------------------------------------
    # Guard against missing seed
    if overall_best_seq is None:
        n = 3000
        overall_best_seq = _distribution_uniform(n)
        overall_best_score = _objective(overall_best_seq)
        print(f"[{time.time():.1f}] fallback uniform score={overall_best_score:.6f}")

    # ------------------------------------------------------------------
    # 3. Select a few elite seeds for intensive optimisation
    top_candidates.sort(key=lambda x: x[0])
    elite = [cand.copy() for _, cand in top_candidates[:5]]

    # ------------------------------------------------------------------
    # 4. Run independent gradient‑descent trajectories on the elite seeds
    if time.time() < deadline:
        remaining = deadline - time.time()
        # allocate roughly equal time per elite seed, keep a small buffer for polishing
        per_seed = max(1.0, (remaining - 30.0) / len(elite))
        print(f"[{time.time():.1f}] start intensive optimisation ({len(elite)} seeds)")

        for idx, seed_vec in enumerate(elite):
            sub_dead = time.time() + per_seed
            # Alternate between mirror and Adam
            if idx % 2 == 0:
                # mirror descent (average=True sometimes helps)
                seq, score = _mirror_descent(
                    seed_vec.copy(),
                    sub_dead,
                    best_seq=seed_vec,
                    best_score=_objective(seed_vec),
                    lr0=0.8,
                    active_frac=0.025,
                    max_active=20,
                    beta=0.9,
                    eval_interval=200,
                    patience=3000,
                    average=False,
                )
            else:
                seq, score = _adam_descent(
                    seed_vec.copy(),
                    sub_dead,
                    best_seq=seed_vec,
                    best_score=_objective(seed_vec),
                    lr0=0.02,
                    active_frac=0.04,
                    max_active=25,
                    beta1=0.9,
                    beta2=0.999,
                    eps=1e-8,
                    eval_interval=250,
                    patience=3000,
                )
            if score < overall_best_score:
                overall_best_score, overall_best_seq = score, seq.copy()
                print(f"[{time.time():.1f}] elite {idx} improved global best to {score:.6f}")

    # ------------------------------------------------------------------
    # 5. Greedy pairwise refinement
    if time.time() < deadline:
        print(f"[{time.time():.1f}] greedy pairwise refinement")
        overall_best_seq, overall_best_score = _pairwise_swap_local(
            overall_best_seq.copy(),
            deadline,
            best_seq=overall_best_seq,
            best_score=overall_best_score,
            max_trials=300_000,
        )

    # ------------------------------------------------------------------
    # 6. Final simulated‑annealing polishing
    if time.time() < deadline:
        print(f"[{time.time():.1f}] final simulated‑annealing search")
        overall_best_seq, overall_best_score = _random_local_search(
            overall_best_seq.copy(),
            deadline,
            best_seq=overall_best_seq,
            best_score=overall_best_score,
            init_temp=0.01,
            decay=0.9995,
        )

    # ------------------------------------------------------------------
    print(
        f"Final best exact score ≈ {overall_best_score:.6f} (length={len(overall_best_seq)})"
    )
    # The evaluator will clip and renormalise; we already keep sum≈1.
    return overall_best_seq.tolist()
