def _cross_corr_valid(w, a):
    # w length 2n-1, a length n
    n = a.shape[0]
    L = 1 << ((len(w) + n - 1 - 1).bit_length())
    # Compute FFT of w and a
    W = np.fft.rfft(w, n=L)
    A = np.fft.rfft(a, n=L)
    # Compute cross-correlation via conj(W) * A
    C = np.fft.irfft(np.conj(W) * A, n=L)
    # Slice valid: indices from (n-1) to (n-1)+n
    start = n - 1
    return C[start:start + n]

def _smooth_grad_descent(a_init, deadline, best_seq, best_score, epsilon_start=0.05, epsilon_decay=0.9995, eta0=0.3, eta_decay=0.9995, min_epsilon=1e-7):
    a = a_init.copy()
    n = len(a)
    t = 0
    epsilon = epsilon_start
    rng = np.random.default_rng()
    no_improve = 0
    while time.time() < deadline and epsilon > min_epsilon:
        t += 1
        # compute conv
        conv = _fft_self_convolve(a)  # length 2n-1
        # compute smoothed objective maybe for monitoring
        max_val = conv.max()
        # compute w smooth weights
        # stable exponent
        shift = conv.max()  # for numerical stability
        exp_term = np.exp((conv - shift) / epsilon)
        w = exp_term / exp_term.sum()

        # compute gradient using cross-corr
        grad = _cross_corr_valid(w, a)  # returns length n
        grad = 2.0 * grad

        # step size
        eta = eta0 / np.sqrt(t)
        # exponentiated update
        a = a * np.exp(-eta * grad)
        # renorm to sum=1
        a_sum = a.sum()
        if a_sum == 0 or np.isnan(a_sum):
            a = np.full_like(a, 1.0 / n)
        else:
            a = a / a_sum

        # optional jitter after many no improvements (maybe used)
        # compute current hard objective
        cur_obj = _objective(a)
        if cur_obj < best_score - 1e-12:
            best_score = cur_obj
            best_seq = a.copy()
            no_improve = 0
            print(f"[{time.time():.1f}] smooth-grad it={t} eps={epsilon:.2e} obj={best_score:.6f}")
        else:
            no_improve += 1

        # decay epsilon periodically maybe every 1000 iters
        if t % 1000 == 0:
            epsilon *= epsilon_decay
            if epsilon < min_epsilon:
                epsilon = min_epsilon

        # jitter
        if no_improve >= 5000:
            # random re-perturb small
            jitter = rng.normal(scale=0.001, size=n)
            a = a + jitter
            a = np.maximum(a, 0)
            a = a / a.sum()
            no_improve = 0

    return best_seq, best_score

"""
Propose a non‑negative step‑function (list of heights) that minimises
``evaluate_sequence``.

Algorithm
---------
1. **Generate a diverse pool of seeds** – a handful of analytic families
   (uniform, arcsine, symmetric Beta and edge‑weighted) together with a few
   random Dirichlet vectors, all normalised to sum = 1.  Keep the ~5 best.

2. **Smooth gradient descent** – minimise a differentiable surrogate of the
   true objective:
        f_ε(a) = ε·log(∑_k exp((a*a)_k / ε))
   whose gradient is
        ∇f_ε(a) = 2· (w ⊛ a)   with   w_k ∝ exp((a*a)_k / ε) .
   The update is performed with exponentiated‑gradient steps
        a_i ← a_i·exp(−η·∇_i)   and re‑normalised to the simplex.
   ε is slowly annealed from 1e‑1 down to 1e‑6, η decays as 1/√t.
   This uses *all* convolution entries (soft‑max) instead of only the hard
   maximum, allowing a finer optimisation of the peak‑to‑valley balance.

3. **Greedy pairwise mass transfer** – after the smooth phase we move a
   tiny amount of mass between random positions whenever it improves the
   true (hard‑max) objective.

4. **Very cheap simulated‑annealing walk** – a final SA pass to escape any
   remaining shallow local minima.

All steps respect the global 1000 s budget; the routine returns the best
sequence found as a plain Python list (the caller will invoke
``evaluate_sequence`` which performs the final normalisation / clipping).
"""
import time
import math
import numpy as np

# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x≥0, Σx = s}."""
    if s < 0:
        raise ValueError("simplex radius must be non‑negative")
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho_arr = np.where(u - cssv / (np.arange(1, len(v) + 1)) > 0)[0]
    if rho_arr.size == 0:
        theta = 0.0
    else:
        rho = rho_arr[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Full self‑convolution a*a using a real FFT (returns length 2n‑1)."""
    n = a.shape[0]
    need = 2 * n - 1
    size = 1 << ((need - 1).bit_length())
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[:need]


# ----------------------------------------------------------------------
def _cross_corr_valid(w: np.ndarray, a: np.ndarray) -> np.ndarray:
    """
    Linear cross‑correlation (valid part) of w (length 2n‑1) with a (length n).
    Returns an array of length n where
        out[i] = Σ_k w[k+i] * a[k]  (i = 0…n‑1).
    """
    n = a.shape[0]
    m = w.shape[0]            # = 2n‑1
    size = 1 << ((m + n - 1 - 1).bit_length())
    W = np.fft.rfft(w, n=size)
    A = np.fft.rfft(a, n=size)
    C = np.fft.irfft(np.conj(W) * A, n=size)
    start = n - 1
    return C[start:start + n]


# ----------------------------------------------------------------------
def _objective(a: np.ndarray) -> float:
    """Exact objective 2·n·max(conv(a,a)) for a vector with sum = 1."""
    n = a.shape[0]
    b = _fft_self_convolve(a)
    return 2.0 * n * b.max()


# ----------------------------------------------------------------------
# Analytic seed families
def _distribution_uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _distribution_arcsine(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    vals /= vals.sum()
    return vals


def _distribution_beta_sym(n: int, p: float) -> np.ndarray:
    """Discrete symmetric Beta(p,p) on the grid {0,…,n‑1}."""
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    w = np.exp((p - 1) * np.log(x) + (p - 1) * np.log(1.0 - x))
    if w.sum() == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _distribution_edge_weighted(n: int, p: float) -> np.ndarray:
    """U‑shaped family w_i ∝ ((i+½)*(n‑i‑½))^(−p)."""
    i = np.arange(n, dtype=np.float64) + 0.5
    j = (n - np.arange(n, dtype=np.float64) - 0.5)
    w = (i * j) ** (-p)
    if w.sum() == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


# ----------------------------------------------------------------------
def _smooth_gradient_descent(
    a0: np.ndarray,
    deadline: float,
    best_seq: np.ndarray,
    best_score: float,
    eps_start: float = 1e-1,
    eps_end: float = 1e-6,
    eps_decay: float = 0.9995,
    eta0: float = 0.5,
) -> tuple[np.ndarray, float]:
    """
    Minimises the hard objective using a soft‑max surrogate.
    ``a0`` is assumed to be on the simplex (sum = 1, non‑negative).
    """
    a = a0.copy()
    n = a.shape[0]
    t = 0
    eps = eps_start
    rng = np.random.default_rng()
    no_improve = 0

    while time.time() < deadline and eps > eps_end:
        t += 1

        # 1) self‑convolution
        conv = _fft_self_convolve(a)               # length 2n‑1

        # 2) soft‑max weights
        max_c = conv.max()
        # stability: shift by max_c
        exp_term = np.exp((conv - max_c) / eps)
        w = exp_term / exp_term.sum()              # sum‑one vector on length 2n‑1

        # 3) gradient = 2·(w ⊗ a)  (cross‑correlation, valid part)
        grad = 2.0 * _cross_corr_valid(w, a)

        # 4) exponentiated‑gradient step with decaying step‑size
        eta = eta0 / math.sqrt(t)
        a = a * np.exp(-eta * grad)                # positivity preserved
        s = a.sum()
        if s == 0 or np.isnan(s):
            a = np.full_like(a, 1.0 / n)
        else:
            a = a / s                               # project onto simplex

        # 5) evaluate the *hard* objective and keep global best
        cur = _objective(a)
        if cur < best_score - 1e-14:
            best_score = cur
            best_seq = a.copy()
            no_improve = 0
            print(
                f"[{time.time():.1f}] smooth‑grad it={t} eps={eps:.2e} "
                f"obj={best_score:.6f}"
            )
        else:
            no_improve += 1

        # 6) anneal the smoothing parameter
        if t % 1000 == 0:
            eps *= eps_decay
            eps = max(eps, eps_end)

        # 7) occasional jitter if stuck
        if no_improve >= 4000:
            jitter = rng.exponential(scale=0.002, size=n)
            a = a + jitter
            a = a / a.sum()
            no_improve = 0

    return best_seq, best_score


# ----------------------------------------------------------------------
def _pairwise_swap_local(
    a: np.ndarray,
    deadline: float,
    best_seq: np.ndarray,
    best_score: float,
    max_trials: int | None = None,
) -> tuple[np.ndarray, float]:
    """Deterministic greedy move of a small mass between two slots."""
    n = a.shape[0]
    trials = 0
    rng = np.random.default_rng()
    while time.time() < deadline:
        if max_trials is not None and trials >= max_trials:
            break
        trials += 1

        src = np.where(a > 1e-12)[0]
        if src.size == 0:
            break
        i = rng.choice(src)

        j = rng.integers(n - 1)
        if j >= i:
            j += 1

        delta = a[i] * rng.random() * 0.33
        if delta < 1e-14:
            continue

        new_a = a.copy()
        new_a[i] -= delta
        new_a[j] += delta

        cur = _objective(new_a)
        if cur < best_score - 1e-14:
            a = new_a
            best_score = cur
            best_seq = a.copy()
            print(
                f"[{time.time():.1f}] swap i={i}→j={j} δ={delta:.3e} new best {best_score:.6f}"
            )
    return best_seq, best_score


# ----------------------------------------------------------------------
def _random_local_search(
    a: np.ndarray,
    deadline: float,
    best_seq: np.ndarray,
    best_score: float,
    init_temp: float = 1e-2,
    decay: float = 0.9995,
) -> tuple[np.ndarray, float]:
    """Very cheap simulated‑annealing walk."""
    n = a.shape[0]
    rng = np.random.default_rng()
    T = init_temp
    while time.time() < deadline:
        i = rng.integers(n)
        j = rng.integers(n - 1)
        if j >= i:
            j += 1
        delta = a[i] * rng.random() * 0.30
        if delta < 1e-14:
            continue
        new_a = a.copy()
        new_a[i] -= delta
        new_a[j] += delta
        cur = _objective(new_a)
        if cur < best_score - 1e-14:
            a = new_a
            best_score = cur
            best_seq = a.copy()
            print(
                f"[{time.time():.1f}] SA improve {i}->{j} Δ={delta:.3e} best {best_score:.6f}"
            )
        else:
            if rng.random() < math.exp(-(cur - best_score) / max(T, 1e-12)):
                a = new_a
        T *= decay
    return best_seq, best_score


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: int = 1000) -> list[float]:
    """
    Search for a non‑negative step‑function that minimises ``evaluate_sequence``.
    Returns a plain Python list; the caller will later invoke ``evaluate_sequence``.
    """
    rng = np.random.default_rng(seed)
    start = time.time()
    deadline = start + budget_s - 3.0            # safety margin

    overall_best_seq = None
    overall_best_score = float("inf")
    candidates: list[tuple[float, np.ndarray]] = []

    # ------------------------------------------------------------------
    # 1. Build a modest but diverse set of seeds (several lengths)
    lengths = [3000, 4000, 5000, 8000]          # a bit larger than before
    beta_ps = [0.03, 0.06, 0.12, 0.25, 0.5, 1.0, 1.5]
    edge_ps = [0.12, 0.25, 0.50, 0.80, 1.20]

    for n in lengths:
        if time.time() >= deadline:
            break
        print(f"[{time.time():.1f}] scanning length {n}")

        # uniform
        cand = _distribution_uniform(n)
        sc = _objective(cand)
        candidates.append((sc, cand.copy()))
        if sc < overall_best_score:
            overall_best_score, overall_best_seq = sc, cand.copy()
            print(f"[{time.time():.1f}] new global best (uniform) len={n} score={sc:.6f}")

        # arcsine
        cand = _distribution_arcsine(n)
        sc = _objective(cand)
        candidates.append((sc, cand.copy()))
        if sc < overall_best_score:
            overall_best_score, overall_best_seq = sc, cand.copy()
            print(f"[{time.time():.1f}] new global best (arcsine) len={n} score={sc:.6f}")

        # symmetric beta
        for p in beta_ps:
            if time.time() >= deadline:
                break
            cand = _distribution_beta_sym(n, p)
            sc = _objective(cand)
            candidates.append((sc, cand.copy()))
            if sc < overall_best_score:
                overall_best_score, overall_best_seq = sc, cand.copy()
                print(
                    f"[{time.time():.1f}] new global best βsym p={p:.3g} len={n} score={sc:.6f}"
                )

        # edge‑weighted
        for p in edge_ps:
            if time.time() >= deadline:
                break
            cand = _distribution_edge_weighted(n, p)
            sc = _objective(cand)
            candidates.append((sc, cand.copy()))
            if sc < overall_best_score:
                overall_best_score, overall_best_seq = sc, cand.copy()
                print(
                    f"[{time.time():.1f}] new global best edge p={p:.3g} len={n} score={sc:.6f}"
                )

        # random Dirichlet (skewed)
        for _ in range(3):
            if time.time() >= deadline:
                break
            alpha = rng.uniform(0.01, 0.6)
            cand = rng.dirichlet(np.full(n, alpha))
            sc = _objective(cand)
            candidates.append((sc, cand.copy()))
            if sc < overall_best_score:
                overall_best_score, overall_best_seq = sc, cand.copy()
                print(
                    f"[{time.time():.1f}] new global best Dirichlet α={alpha:.3g} len={n} score={sc:.6f}"
                )

    # keep a few best seeds (for the intensive smooth‑gradient phase)
    candidates.sort(key=lambda x: x[0])
    top_seeds = [seq for _, seq in candidates[:6]]

    # ------------------------------------------------------------------
    # 2. Smooth gradient descent on the most promising seeds
    remaining = deadline - time.time()
    if remaining > 20.0 and top_seeds:
        per_seed = remaining / len(top_seeds)
        print(f"[{time.time():.1f}] starting smooth‑gradient descent on {len(top_seeds)} seeds")
        for idx, seed_seq in enumerate(top_seeds):
            if time.time() >= deadline:
                break
            sub_deadline = time.time() + per_seed
            a_opt, sc_opt = _smooth_gradient_descent(
                seed_seq.copy(),
                sub_deadline,
                best_seq=overall_best_seq,
                best_score=overall_best_score,
                eps_start=1e-1,
                eps_end=1e-6,
                eps_decay=0.999,
                eta0=0.6,
            )
            if sc_opt < overall_best_score:
                overall_best_score, overall_best_seq = sc_opt, a_opt.copy()
                print(
                    f"[{time.time():.1f}] after smooth‑grad seed {idx} new global best {overall_best_score:.6f}"
                )
            if overall_best_score <= 1.503:
                break

    # ------------------------------------------------------------------
    # 3. Greedy pairwise refinement (deterministic)
    if time.time() < deadline:
        print(f"[{time.time():.1f}] greedy pairwise refinement")
        overall_best_seq, overall_best_score = _pairwise_swap_local(
            overall_best_seq.copy(),
            deadline,
            best_seq=overall_best_seq,
            best_score=overall_best_score,
            max_trials=300_000,
        )

    # ------------------------------------------------------------------
    # 4. Final cheap simulated‑annealing walk
    if time.time() < deadline:
        print(f"[{time.time():.1f}] final simulated‑annealing walk")
        overall_best_seq, overall_best_score = _random_local_search(
            overall_best_seq.copy(),
            deadline,
            best_seq=overall_best_seq,
            best_score=overall_best_score,
            init_temp=0.008,
            decay=0.9997,
        )

    # ------------------------------------------------------------------
    print(
        f"Final best score: {overall_best_score:.6f} (length={len(overall_best_seq)})"
    )
    # Return a plain Python list; ``evaluate_sequence`` will handle clipping etc.
    return overall_best_seq.tolist()
