def _smooth_gradient(a, lam):
    conv = _fft_self_convolve(a)
    max_conv = conv.max()
    # Compute weights p = exp(lam*(conv - max_conv))
    # subtract max for numerical stability.
    exps = np.exp(lam * (conv - max_conv))
    sum_exps = exps.sum()
    p = exps / sum_exps  # normalized
    # gradient = 2 * convolution of p with reversed a at offset
    ar = a[::-1]
    # Compute convolution via FFT: p * ar
    # Use np.convolve
    c = np.convolve(p, ar)
    offset = a.shape[0] - 1
    grad = 2 * c[offset: offset + a.shape[0]]
    return grad

step_function_search_v2.py

A 1000‑second heuristic for the “minimise 2·n·max(conv(a,a))/(∑a)²” problem.
The approach combines

All heavy work respects the deadline (a 2‑second safety margin).  The

# ----------------------------------------------------------------------
# Simplex utilities
# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x≥0, Σx=s}."""
    if s < 0:
        raise ValueError("simplex radius must be non‑negative")
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.where(u - cssv / (np.arange(1, len(v) + 1)) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# FFT based convolution helpers
# ----------------------------------------------------------------------
def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Full linear self‑convolution a*a using a real FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 2) - 1).bit_length()  # next power‑of‑two
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[: 2 * n - 1]


def _fft_convolve(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution of x and y (both real) via FFT."""
    n = x.shape[0]
    m = y.shape[0]
    size = 1 << ((n + m - 1) - 1).bit_length()
    X = np.fft.rfft(x, n=size)
    Y = np.fft.rfft(y, n=size)
    conv = np.fft.irfft(X * Y, n=size)
    return conv[: n + m - 1]


# ----------------------------------------------------------------------
# Exact objective (sum=1 ⇒ denominator = 1)
# ----------------------------------------------------------------------
def _objective(a: np.ndarray) -> float:
    """Exact value of ``evaluate_sequence`` for a probability vector (∑a=1)."""
    n = a.shape[0]
    b = _fft_self_convolve(a)
    max_b = b.max()
    return float(2.0 * n * max_b)


# ----------------------------------------------------------------------
# Sub‑gradient of the active convolution entry
# ----------------------------------------------------------------------
def _grad_of_max_conv(a: np.ndarray) -> tuple[np.ndarray, int]:
    """Gradient of the currently maximal convolution entry."""
    n = a.shape[0]
    conv = _fft_self_convolve(a)
    k = int(np.argmax(conv))
    grad = np.zeros_like(a)
    if k < n:
        grad[: k + 1] = 2.0 * a[k::-1]
    else:
        start = k - (n - 1)
        grad[start:] = 2.0 * a[start:][::-1]
    return grad, k

# Analytic families (all return a probability vector, sum=1)

def _distribution_uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _distribution_arcsine(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    vals /= vals.sum()
    return vals


def _distribution_hann(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    w = 0.5 - 0.5 * np.cos(2.0 * np.pi * i / (n - 1))
    w_sum = w.sum()
    if w_sum == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w_sum
    return w


def _distribution_blackman(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a0 = 0.42
    a1 = 0.5
    a2 = 0.08
    w = (a0
         - a1 * np.cos(2.0 * np.pi * i / (n - 1))
         + a2 * np.cos(4.0 * np.pi * i / (n - 1)))
    w_sum = w.sum()
    if w_sum == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w_sum
    return w


def _distribution_kaiser(n: int, beta: float = 8.0) -> np.ndarray:
    """Kaiser window with shape parameter beta."""
    from scipy.special import i0  # SciPy is available in the environment
    i = np.arange(n, dtype=np.float64)
    alpha = (n - 1) / 2.0
    arg = beta * np.sqrt(1.0 - ((i - alpha) / alpha) ** 2)
    w = i0(arg) / i0(beta)
    w_sum = w.sum()
    if w_sum == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w_sum
    return w


def _distribution_edge_weighted(n: int, p: float) -> np.ndarray:
    """Weight ∝ (i·j)^{‑p} where i,j are distances to the two edges."""
    i = np.arange(n, dtype=np.float64) + 0.5
    j = (n - np.arange(n, dtype=np.float64) - 0.5)
    w = (i * j) ** (-p)
    w_sum = w.sum()
    if w_sum == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w_sum
    return w


def _distribution_generalized_normal(n: int, q: float) -> np.ndarray:
    """Density ∝ exp(‑|x‑½|^{q}) on [0,1]."""
    i = np.arange(n, dtype=np.float64) + 0.5
    x = i / n - 0.5
    w = np.exp(- (np.abs(x) / 0.5) ** q)
    w_sum = w.sum()
    if w_sum == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w_sum
    return w


def _distribution_beta(n: int, alpha: float, beta: float) -> np.ndarray:
    """Beta(α,β) density on the unit interval, discretised."""
    i = np.arange(n, dtype=np.float64) + 0.5
    x = i / n
    # use logs for stability when α or β < 1
    logdens = (alpha - 1) * np.log(x) + (beta - 1) * np.log(1 - x)
    dens = np.exp(logdens - np.max(logdens))  # shift to avoid underflow
    dens_sum = dens.sum()
    if dens_sum == 0.0:
        dens = np.full_like(dens, 1.0 / n)
    else:
        dens /= dens_sum
    return dens


def _distribution_random_sparse(
    n: int, rng: np.random.Generator, density: float = 0.02
) -> np.ndarray:
    """Sparse random vector with a given density."""
    k = max(2, int(n * density))
    idx = rng.choice(n, size=k, replace=False)
    vals = np.zeros(n, dtype=np.float64)
    vals[idx] = rng.random(k)
    if vals.sum() == 0.0:
        vals[0] = 1.0
    vals /= vals.sum()
    return vals


def _mix_with_uniform(base: np.ndarray, lam: float) -> np.ndarray:
    """Convex combination with the uniform distribution."""
    n = base.shape[0]
    uniform = np.full(n, 1.0 / n, dtype=np.float64)
    return lam * uniform + (1.0 - lam) * base


# ----------------------------------------------------------------------
# Smooth‑max surrogate (log‑sum‑exp) utilities
# ----------------------------------------------------------------------
def _smooth_gradient(a: np.ndarray, lam: float = 50.0) -> np.ndarray:
    """
    Gradient of the smooth maximum  (1/λ)·log Σexp(λ·conv) .
    Returns a vector of the same shape as a.
    """
    conv = _fft_self_convolve(a)
    max_conv = conv.max()
    # stable exponentiation
    exps = np.exp(lam * (conv - max_conv))
    p = exps / exps.sum()               # a probability distribution over k
    # grad = 2·(p * a) convolved with reversed a
    ar = a[::-1]
    c = _fft_convolve(p, ar)
    offset = a.shape[0] - 1
    grad = 2.0 * c[offset: offset + a.shape[0]]
    return grad

def _exp_gradient_descent(
    a: np.ndarray,
    deadline: float,
    best_seq: np.ndarray,
    best_score: float,
    eta0: float = 0.9,
    eval_every: int = 50,
) -> tuple[np.ndarray, float]:
    """
    Exponentiated‑gradient (mirror descent) on the exact objective.
    Update rule: a ← a·exp(‑η·grad) then ℓ₁‑normalise.
    """
    t = 0
    while time.time() < deadline:
        t += 1
        grad, _ = _grad_of_max_conv(a)
        eta = eta0 / math.sqrt(t)
        a = a * np.exp(-eta * grad)
        a = _project_onto_simplex(a)               # renorm to sum‑1
        if t % eval_every == 0:
            cur = _objective(a)
            if cur < best_score - 1e-14:
                best_score = cur
                best_seq = a.copy()
                print(f"[{time.time():.1f}] EG it={t} new best {best_score:.6f}")
    return best_seq, best_score


def _proj_subgradient_descent(
    a: np.ndarray,
    deadline: float,
    best_seq: np.ndarray,
    best_score: float,
    eta0: float = 1.0,
    eval_every: int = 50,
) -> tuple[np.ndarray, float]:

    Projected sub‑gradient descent on the exact objective.

        t += 1
        grad, _ = _grad_of_max_conv(a)

        a = _project_onto_simplex(a)
        if t % eval_every == 0:

            if cur < best_score - 1e-14:

                best_seq = a.copy()
                print(f"[{time.time():.1f}] PGD it={t} new best {best_score:.6f}")

def _smooth_descent(
    a: np.ndarray,
    deadline: float,
    best_seq: np.ndarray,
    best_score: float,
    lam: float = 50.0,
    eta0: float = 0.5,
    eval_every: int = 50,
) -> tuple[np.ndarray, float]:
    """
    Gradient descent on the smooth‑max surrogate (log‑sum‑exp).
    """
    t = 0
    while time.time() < deadline:
        t += 1
        grad = _smooth_gradient(a, lam=lam)
        eta = eta0 / math.sqrt(t)
        a = a - eta * grad
        a = _project_onto_simplex(a)
        if t % eval_every == 0:
            cur = _objective(a)
            if cur < best_score - 1e-14:

                best_seq = a.copy()
                print(f"[{time.time():.1f}] SG it={t} new best {best_score:.6f}")
    return best_seq, best_score

# Simulated‑annealing style pairwise mass transfers
# ----------------------------------------------------------------------
def _pairwise_annealing(
    a: np.ndarray,
    deadline: float,
    best_seq: np.ndarray,
    best_score: float,
    rng: np.random.Generator,
) -> tuple[np.ndarray, float]:
    """Randomly transfer a fraction of mass between two bins."""
    n = a.shape[0]
    start = time.time()
    total = deadline - start
    while time.time() < deadline:
        src = np.where(a > 1e-12)[0]
        if src.size == 0:
            break
        i = rng.choice(src)
        j = rng.integers(n - 1)
        if j >= i:
            j += 1
        delta = a[i] * rng.random() * 0.4
        if delta < 1e-16:
            continue
        new_a = a.copy()
        new_a[i] -= delta
        new_a[j] += delta
        cur = _objective(new_a)
        elapsed = time.time() - start
        temperature = max(0.0, 1.0 - elapsed / total) * best_score * 0.01
        accept = False
        if cur < best_score - 1e-14:
            accept = True
        elif temperature > 0.0:
            prob = math.exp(-(cur - best_score) / temperature)
            if rng.random() < prob:
                accept = True
        if accept:
            a = new_a
            if cur < best_score - 1e-14:
                best_score = cur
                best_seq = a.copy()
                print(
                    f"[{time.time():.1f}] anneal i={i}→j={j} δ={delta:.3e} new best {best_score:.6f}"
                )
    return best_seq, best_score


# ----------------------------------------------------------------------
# Upsampling (linear interpolation while preserving the simplex)
# ----------------------------------------------------------------------
def _upsample(a: np.ndarray, new_n: int) -> np.ndarray:
    """Interpolate a probability vector to a larger length."""
    old_n = a.shape[0]
    if new_n == old_n:
        return a.copy()
    x_old = (np.arange(old_n, dtype=np.float64) + 0.5) / old_n
    x_new = (np.arange(new_n, dtype=np.float64) + 0.5) / new_n
    a_new = np.interp(x_new, x_old, a, left=0.0, right=0.0)
    a_new = np.maximum(a_new, 0.0)
    s = a_new.sum()
    if s == 0.0:
        a_new[:] = 1.0 / new_n
    else:
        a_new /= s
    return a_new

            best_seq=overall_best_seq,
            best_score=overall_best_score,
            lam=80.0,
            eta0=0.6,
            eval_every=40,
        )
        if sc_opt2 < overall_best_score - 1e-12:
            overall_best_score = sc_opt2
            overall_best_seq = a_opt2.copy()
            print(f"[{time.time():.1f}] after smooth GD best {overall_best_score:.6f}")

    # ------------------------------------------------------------------
    # 6.  Upsample the best vector (if time permits) and polish again
    # ------------------------------------------------------------------
    if time.time() < deadline:
        cur_n = overall_best_seq.shape[0]
        target_n = cur_n * 2
        if target_n <= 64000:
            print(f"[{time.time():.1f}] upsampling {cur_n} → {target_n}")
            up = _upsample(overall_best_seq, target_n)
            up_sc = _objective(up)
            if up_sc < overall_best_score:
                overall_best_seq = up
                overall_best_score = up_sc
                print(f"   upsample improved best to {overall_best_score:.6f}")

            # brief projected‑gradient polishing on upsampled
            sub_dead = time.time() + (deadline - time.time()) * 0.20
            a_opt, sc_opt = _proj_subgradient_descent(
                overall_best_seq.copy(),
                sub_dead,
                best_seq=overall_best_seq,
                best_score=overall_best_score,
                eta0=0.7,
                eval_every=40,
            )
            if sc_opt < overall_best_score - 1e-12:
                overall_best_seq = a_opt.copy()
                overall_best_score = sc_opt
                print(f"   after upsample PGD best {overall_best_score:.6f}")

    # ------------------------------------------------------------------
    # 7.  Final simulated‑annealing polishing (pairwise mass transfers)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        print(f"[{time.time():.1f}] final annealing phase")
        a_opt, sc_opt = _pairwise_annealing(
            overall_best_seq.copy(),
            deadline,
            best_seq=overall_best_seq,
            best_score=overall_best_score,
            rng=rng,
        )
        if sc_opt < overall_best_score - 1e-12:
            overall_best_seq = a_opt.copy()
            overall_best_score = sc_opt
            print(f"   after annealing best {overall_best_score:.6f}")

    # ------------------------------------------------------------------
    print(
        f"Final best score (2·n·max(conv)) = {overall_best_score:.6f} "
        f"(length={len(overall_best_seq)})"
    )
    # Return as plain Python list – the external evaluator expects a list.
    return overall_best_seq.tolist()
