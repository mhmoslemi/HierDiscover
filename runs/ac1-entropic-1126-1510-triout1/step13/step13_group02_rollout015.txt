def smooth_gradient(a, beta):
    conv = _fft_self_convolve(a)
    maxc = conv.max()
    w = np.exp(beta * (conv - maxc))
    w = w / w.sum()
    a_rev = a[::-1]
    # convolution of w and a_rev
    # Use FFT:
    n = a.shape[0]
    size = 1 << ((len(w) + n - 1).bit_length())
    W = np.fft.rfft(w, n=size)
    A_rev = np.fft.rfft(a_rev, n=size)
    grad_full = np.fft.irfft(W * A_rev, n=size)
    # grad_i = 2 * grad_full[n-1 + i]
    grad = 2.0 * grad_full[n-1 : n-1 + n]  # slice length n
    # ensure nonnegative? gradient can be negative if w has small values?
    return grad

def _smooth_gradient_descent(a, deadline, best_seq, best_score, beta_start=5.0, beta_end=200.0, eta0=0.8, eval_every=200):
    t=0
    beta = beta_start
    total_iterations = 0
    while time.time() < deadline:
        # update beta gradually
        elapsed = time.time() - start_time_of_this_func
        total_time_alloc = deadline - start_time_of_this_func
        progress = elapsed/total_time_alloc
        beta = beta_start + progress * (beta_end - beta_start)  # linear increase

        grad = _smooth_gradient(a, beta)  # returns vector

        t += 1
        eta = eta0 / math.sqrt(t)
        # exponentiated gradient update
        a = a * np.exp(-eta * grad)
        a_sum = a.sum()
        if a_sum == 0:
            a = np.full_like(a, 1.0/a.shape[0])
        else:
            a = a / a_sum

        if t % eval_every == 0:
            cur = _objective(a)
            if cur < best_score - 1e-14:
                best_score = cur
                best_seq = a.copy()
                print(f"[{time.time():.1f}] smooth GD it={t}, beta={beta:.2f} new best {best_score:.6f}")
    return best_seq, best_score

def _smooth_gradient(a, beta):
    conv = _fft_self_convolve(a)
    maxc = conv.max()
    # compute w = exp(beta*(conv - maxc))
    # may use np.exp(beta*(conv - maxc)) directly.
    w = np.exp(beta * (conv - maxc))
    w_sum = w.sum()
    if w_sum == 0.0:
        w = np.full_like(w, 1.0/len(w))
    else:
        w /= w_sum
    # compute convolution of w with a reversed
    a_rev = a[::-1]
    n = len(a)
    size = 1 << ((len(w) + n - 1).bit_length())
    W = np.fft.rfft(w, n=size)
    A_rev = np.fft.rfft(a_rev, n=size)
    grad_full = np.fft.irfft(W * A_rev, n=size)
    # grad_i = 2 * grad_full[n-1 + i]
    start = n - 1
    grad = 2.0 * grad_full[start:start+n]
    return grad

"""
step_function_search_v2.py

A 1000‑second optimiser for the “step‑function” objective:
    2·n·max(conv(a,a)) / (∑a)²

Improvements over the previous version
-------------------------------------

1. **Richer seed families** – besides the original families we add
   exponential‑decay, beta‑distribution and a finer sweep of the
   edge‑weighted power‐law.  All seeds are normalised to sum = 1.

2. **Smoothed gradient descent** – we replace the pure sub‑gradient
   (max‑only) update by a *soft‑max* (log‑sum‑exp) smoothing of the
   convolution entries.  The smoothing parameter β is increased from
   5 to 200 during the run (continuation), which helps escape flat
   plateaus.

3. **Symmetrisation** – after each 10‑th iteration the current vector
   is symmetrised (a ← (a + a[::-1])/2) and kept if it improves the
   objective.  The optimum is known to be symmetric, so this reduces
   the search space.

4. **Longer focussed optimisation** – after a quick polish of many
   seeds we devote a large fraction of the remaining budget to a
   prolonged smoothed‑gradient run on the current global best, then
   up‑sample (linear interpolation) and polish again.

5. **Final directed annealing** – a pairwise mass‑transfer neighbourhood
   search (simulated annealing) is applied to the best candidate.

The code respects the 1000 s budget (2 s safety margin) and returns
a plain ``list[float]`` ready for ``evaluate_sequence``.  All
operations stay on the probability simplex (∑a = 1, aᵢ ≥ 0) so the
scale‑invariance of the objective is preserved.
"""

import time

# ----------------------------------------------------------------------
# Simplex utilities
# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Euclidean projection onto the simplex {x ≥ 0, Σx = s}."""
    if s < 0:
        raise ValueError("simplex radius must be non‑negative")
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.where(u - cssv / (np.arange(1, len(v) + 1)) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Convolution helpers (FFT based, linear convolution)
# ----------------------------------------------------------------------
def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Full linear self‑convolution a*a using a real FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1) - 1).bit_length()
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[: 2 * n - 1]


# ----------------------------------------------------------------------
# Exact objective (scale‑invariant for sum=1)
# ----------------------------------------------------------------------
def _objective(a: np.ndarray) -> float:
    """Exact value of ``evaluate_sequence`` for a probability vector."""
    n = a.shape[0]
    b = _fft_self_convolve(a)
    max_b = b.max()
    return float(2.0 * n * max_b)


# ----------------------------------------------------------------------
# Analytic families (all return a probability vector, sum=1)
# ----------------------------------------------------------------------
def _distribution_uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _distribution_arcsine(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    vals /= vals.sum()
    return vals


def _distribution_hann(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    w = 0.5 - 0.5 * np.cos(2.0 * np.pi * i / (n - 1))
    w_sum = w.sum()
    if w_sum == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w_sum
    return w


def _distribution_edge_weighted(n: int, p: float) -> np.ndarray:
    """Weight ∝ (i·j)^{‑p} where i,j are distances to the two edges."""
    i = np.arange(n, dtype=np.float64) + 0.5
    j = (n - np.arange(n, dtype=np.float64) - 0.5)
    w = (i * j) ** (-p)
    w_sum = w.sum()
    if w_sum == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w_sum
    return w


def _distribution_generalized_normal(n: int, q: float) -> np.ndarray:
    """Density ∝ exp(‑|x‑½|^{q}) on [0,1]."""
    i = np.arange(n, dtype=np.float64) + 0.5
    x = i / n - 0.5
    w = np.exp(- (np.abs(x) / 0.5) ** q)
    w_sum = w.sum()
    if w_sum == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w_sum
    return w


def _distribution_exponential_decay(n: int, c: float) -> np.ndarray:
    """Weight ∝ c**i + c**(n‑1‑i) for 0<c<1."""
    i = np.arange(n, dtype=np.float64)
    w = (c ** i) + (c ** (n - 1 - i))
    w_sum = w.sum()
    if w_sum == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w_sum
    return w


def _distribution_beta(n: int, alpha: float, beta: float) -> np.ndarray:
    """Discrete Beta(α,β) density on {0,…,n‑1}."""
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    # avoid log(0) by clipping
    eps = np.finfo(float).eps
    x = np.clip(x, eps, 1 - eps)
    w = np.exp((alpha - 1) * np.log(x) + (beta - 1) * np.log(1 - x))
    w_sum = w.sum()
    if w_sum == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w_sum
    return w


def _distribution_random_sparse(
    n: int, rng: np.random.Generator, density: float = 0.01
) -> np.ndarray:
    """Sparse random vector with a given density."""
    k = max(2, int(n * density))
    idx = rng.choice(n, size=k, replace=False)
    vals = np.zeros(n, dtype=np.float64)
    vals[idx] = rng.random(k)
    if vals.sum() == 0.0:
        vals[0] = 1.0
    vals /= vals.sum()
    return vals


def _mix_with_uniform(base: np.ndarray, lam: float) -> np.ndarray:
    """Convex combination with the uniform distribution."""
    n = base.shape[0]
    uniform = np.full(n, 1.0 / n, dtype=np.float64)
    return lam * uniform + (1.0 - lam) * base


# ----------------------------------------------------------------------
# Smoothed gradient (log‑sum‑exp of convolution entries)
# ----------------------------------------------------------------------
def _smooth_gradient(a: np.ndarray, beta: float) -> np.ndarray:
    """
    Gradient of the smooth approximation
        f_β(a) = (log Σ_k exp(β·conv_k))/β .
    The gradient is 2·(w * a_rev)_{n‑1…n‑1+n‑1},
    where w_k ∝ exp(β·conv_k) and * denotes convolution.
    """
    n = a.shape[0]
    # 1) conv_k
    conv = _fft_self_convolve(a)
    maxc = conv.max()
    # 2) soft‑max weights
    w = np.exp(beta * (conv - maxc))
    w_sum = w.sum()
    if w_sum == 0.0:
        w = np.full_like(w, 1.0 / w.shape[0])
    else:
        w /= w_sum
    # 3) convolution w * a_rev
    a_rev = a[::-1]
    size = 1 << ((len(w) + n - 1).bit_length())
    W = np.fft.rfft(w, n=size)
    A_rev = np.fft.rfft(a_rev, n=size)
    grad_full = np.fft.irfft(W * A_rev, n=size)
    start = n - 1
    grad = 2.0 * grad_full[start:start + n]   # length n
    return grad


# ----------------------------------------------------------------------
# Exponentiated‑gradient descent with smoothing & symmetrisation
# ----------------------------------------------------------------------
def _smooth_exponentiated_gradient(
    a: np.ndarray,
    deadline: float,
    best_seq: np.ndarray,
    best_score: float,
    beta_start: float = 5.0,
    beta_end: float = 200.0,
    eta0: float = 0.8,
    eval_every: int = 100,
) -> tuple[np.ndarray, float]:
    """
    Mirror‑descent on the smoothed objective.
    Uses a continuation on β and occasional symmetrisation.
    """
    n = a.shape[0]
    t = 0
    start_local = time.time()
    while time.time() < deadline:
        t += 1
        # linear interpolation of β
        elapsed = time.time() - start_local
        total = deadline - start_local
        progress = elapsed / total if total > 0 else 1.0
        beta = beta_start + progress * (beta_end - beta_start)

        grad = _smooth_gradient(a, beta)
        eta = eta0 / math.sqrt(t)
        a = a * np.exp(-eta * grad)   # stays non‑negative
        s = a.sum()
        if s == 0.0:
            a = np.full(n, 1.0 / n, dtype=np.float64)
        else:
            a = a / s

        # symmetrise every few iterations
        if t % 10 == 0:
            a_sym = (a + a[::-1]) * 0.5
            a_sym /= a_sym.sum()
            # keep if it improves the true objective
            cur_sym = _objective(a_sym)
            cur = _objective(a)
            if cur_sym < cur:
                a = a_sym
                # do not recompute cur now – will be done later

        if t % eval_every == 0:
            cur = _objective(a)
            if cur < best_score - 1e-14:
                best_score = cur
                best_seq = a.copy()
                print(f"[{time.time():.1f}] smooth GD it={t} β={beta:.2f} new best {best_score:.6f}")
    return best_seq, best_score


# ----------------------------------------------------------------------
# Sub‑gradient descent (fallback, same style as before)
# ----------------------------------------------------------------------
def _grad_of_max_conv(a: np.ndarray) -> tuple[np.ndarray, int]:
    """Compute sub‑gradient of the presently maximal convolution entry."""
    n = a.shape[0]
    conv = _fft_self_convolve(a)
    k = int(np.argmax(conv))
    grad = np.zeros_like(a)
    if k < n:
        grad[: k + 1] = 2.0 * a[k::-1]
    else:
        start = k - (n - 1)
        grad[start:] = 2.0 * a[start:][::-1]
    return grad, k


def _plain_exponentiated_gradient(
    a: np.ndarray,
    deadline: float,
    best_seq: np.ndarray,
    best_score: float,
    eta0: float = 0.9,
    eval_every: int = 60,
) -> tuple[np.ndarray, float]:
    """Mirror‑descent on the exact max‑convolution objective."""
    n = a.shape[0]
    t = 0
    while time.time() < deadline:
        t += 1
        grad, _ = _grad_of_max_conv(a)
        eta = eta0 / math.sqrt(t)
        a = a * np.exp(-eta * grad)
        s = a.sum()
        if s == 0.0:
            a = np.full(n, 1.0 / n, dtype=np.float64)
        else:
            a = a / s
        if t % eval_every == 0:
            cur = _objective(a)
            if cur < best_score - 1e-14:
                best_score = cur
                best_seq = a.copy()
                print(f"[{time.time():.1f}] plain GD it={t} new best {best_score:.6f}")
    return best_seq, best_score


# ----------------------------------------------------------------------
# Simulated‑annealing style pairwise mass transfers
# ----------------------------------------------------------------------
def _pairwise_annealing(
    a: np.ndarray,
    deadline: float,
    best_seq: np.ndarray,
    best_score: float,
    rng: np.random.Generator,
) -> tuple[np.ndarray, float]:
    """
    Randomly transfer a fraction of mass between two bins.
    Accept moves that improve the objective; otherwise accept with
    a temperature that decays linearly to zero.
    """
    n = a.shape[0]
    start_time = time.time()
    total_time = deadline - start_time
    while time.time() < deadline:
        src = rng.choice(np.where(a > 1e-12)[0])
        # pick a destination distinct from src
        dst = rng.integers(n - 1)
        if dst >= src:
            dst += 1
        # amount to move (up to 40 % of source mass)
        delta = a[src] * rng.random() * 0.4
        if delta < 1e-18:
            continue
        new_a = a.copy()
        new_a[src] -= delta
        new_a[dst] += delta

        cur = _objective(new_a)
        elapsed = time.time() - start_time
        temperature = max(0.0, 1.0 - elapsed / total_time) * best_score * 0.01
        accept = False
        if cur < best_score - 1e-14:
            accept = True
        elif temperature > 0.0:
            prob = math.exp(-(cur - best_score) / temperature)
            if rng.random() < prob:
                accept = True
        if accept:
            a = new_a
            if cur < best_score - 1e-14:
                best_score = cur
                best_seq = a.copy()
                print(
                    f"[{time.time():.1f}] anneal src={src}→dst={dst} Δ={delta:.3e} new best {best_score:.6f}"
                )
    return best_seq, best_score


# ----------------------------------------------------------------------
# Upsampling (linear interpolation while preserving the simplex)
# ----------------------------------------------------------------------
def _upsample(a: np.ndarray, new_n: int) -> np.ndarray:
    """Interpolate a probability vector to a larger length."""
    old_n = a.shape[0]
    if new_n == old_n:
        return a.copy()
    x_old = (np.arange(old_n, dtype=np.float64) + 0.5) / old_n
    x_new = (np.arange(new_n, dtype=np.float64) + 0.5) / new_n
    a_new = np.interp(x_new, x_old, a, left=0.0, right=0.0)
    a_new = np.maximum(a_new, 0.0)
    s = a_new.sum()
    if s == 0.0:
        a_new[:] = 1.0 / new_n
    else:
        a_new /= s
    return a_new

# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: int = 1000) -> list[float]:
    """
    Search for a non‑negative step function that minimises
    ``evaluate_sequence``.  The routine respects the 1000 s budget.
    """
    rng = np.random.default_rng(seed)
    start = time.time()
    deadline = start + budget_s - 2.0                # safety margin

    overall_best_seq: np.ndarray | None = None
    overall_best_score = float("inf")
    cand_pool: list[tuple[float, np.ndarray]] = []

    # ------------------------------------------------------------------
    # 1.  Assemble a diverse pool of seed vectors (moderate lengths)
    # ------------------------------------------------------------------
    lengths = [2000, 4000, 8000]  # keep within FFT‑friendly range
    for n in lengths:
        if time.time() >= deadline:
            break
        print(f"[{time.time():.1f}] generating seeds for length={n}")

        # deterministic families
        for base in (_distribution_uniform(n),
                     _distribution_arcsine(n),
                     _distribution_hann(n)):
            sc = _objective(base)
            cand_pool.append((sc, base.copy()))
            if sc < overall_best_score:
                overall_best_score = sc
                overall_best_seq = base.copy()
                print(f"[{time.time():.1f}] new global best (det) score={sc:.6f}")

        # edge‑weighted with a dense grid of p
        p_vals = np.concatenate((
            np.logspace(-3.0, -0.5, 12, base=10),   # 0.001 → 0.316
            np.logspace(-0.5, 0.0, 12, base=10),    # 0.316 → 1
            np.linspace(0.1, 2.0, 12),               # linear refinement
            np.logspace(0.0, 0.7, 8, base=10)       # 1 → 5
        ))
        lam_vals = np.linspace(0.0, 1.0, 9)
        for p in p_vals:
            if time.time() >= deadline:
                break
            base = _distribution_edge_weighted(n, float(p))
            for lam in lam_vals:
                if time.time() >= deadline:
                    break
                cand = _mix_with_uniform(base, float(lam))
                sc = _objective(cand)
                cand_pool.append((sc, cand.copy()))
                if sc < overall_best_score:
                    overall_best_score = sc
                    overall_best_seq = cand.copy()
                    print(f"[{time.time():.1f}] edge p={p:.3g} λ={lam:.2f} score={sc:.6f}")

        # exponential‑decay family
        c_vals = np.logspace(-3.0, -0.1, 10)  # c in (0,1)
        for c in c_vals:
            if time.time() >= deadline:
                break
            cand = _distribution_exponential_decay(n, float(c))
            sc = _objective(cand)
            cand_pool.append((sc, cand.copy()))
            if sc < overall_best_score:
                overall_best_score = sc
                overall_best_seq = cand.copy()
                print(f"[{time.time():.1f}] exp‑decay c={c:.3g} score={sc:.6f}")

        # beta distribution family (symmetric α=β)
        alpha_vals = np.logspace(-1.0, 1.0, 12)  # 0.1 … 10
        for a in alpha_vals:
            if time.time() >= deadline:
                break
            cand = _distribution_beta(n, float(a), float(a))
            sc = _objective(cand)
            cand_pool.append((sc, cand.copy()))
            if sc < overall_best_score:
                overall_best_score = sc
                overall_best_seq = cand.copy()
                print(f"[{time.time():.1f}] beta α=β={a:.3g} score={sc:.6f}")

        # generalized normal family
        for q in np.linspace(0.3, 4.0, 14):
            if time.time() >= deadline:
                break
            cand = _distribution_generalized_normal(n, float(q))
            sc = _objective(cand)
            cand_pool.append((sc, cand.copy()))
            if sc < overall_best_score:
                overall_best_score = sc
                overall_best_seq = cand.copy()
                print(f"[{time.time():.1f}] gn q={q:.2f} score={sc:.6f}")

        # random Dirichlet seeds (log‑uniform concentration)
        for _ in range(12):
            if time.time() >= deadline:
                break
            log_alpha = rng.uniform(-2.0, 1.0)      # α ∈ [0.01, 10]
            alpha = 10 ** log_alpha
            cand = rng.dirichlet(np.full(n, alpha))
            sc = _objective(cand)
            cand_pool.append((sc, cand.copy()))
            if sc < overall_best_score:
                overall_best_score = sc
                overall_best_seq = cand.copy()
                print(f"[{time.time():.1f}] Dirichlet α={alpha:.3g} score={sc:.6f}")

        # sparse random seeds
        for _ in range(6):
            if time.time() >= deadline:
                break
            cand = _distribution_random_sparse(n, rng, density=0.02)
            sc = _objective(cand)
            cand_pool.append((sc, cand.copy()))
            if sc < overall_best_score:
                overall_best_score = sc
                overall_best_seq = cand.copy()
                print(f"[{time.time():.1f}] sparse score={sc:.6f}")

        # keep only the top 30 seeds overall (memory bound)
        cand_pool.sort(key=lambda x: x[0])
        cand_pool = cand_pool[:30]

    # ------------------------------------------------------------------
    # 2.  If a previous construction is available, seed it
    # ------------------------------------------------------------------
    try:
        prev = np.asarray(height_sequence_1, dtype=np.float64)  # type: ignore[name-defined]
        if prev.ndim == 1:
            prev = prev / prev.sum()
            sc = _objective(prev)
            if sc < overall_best_score:
                overall_best_score = sc
                overall_best_seq = prev.copy()
                print(f"[{time.time():.1f}] previous best score={sc:.6f}")
            cand_pool.append((sc, prev.copy()))
    except Exception:
        pass

    # ------------------------------------------------------------------
    # 3.  Guarantee a non‑empty result
    # ------------------------------------------------------------------
    if overall_best_seq is None:
        n = 4000
        overall_best_seq = _distribution_uniform(n)
        overall_best_score = _objective(overall_best_seq)
        print(f"[{time.time():.1f}] fallback uniform score={overall_best_score:.6f}")

    # ------------------------------------------------------------------
    # 4.  Quick smoothed‑gradient polishing on all retained seeds
    # ------------------------------------------------------------------
    remaining = deadline - time.time()
    if remaining > 30.0 and cand_pool:
        per_candidate = max(0.2, remaining * 0.25 / max(len(cand_pool), 1))
        print(f"[{time.time():.1f}] quick smooth GD on {len(cand_pool)} seeds "
              f"(≈{per_candidate:.1f}s each)")
        for idx, (score, a0) in enumerate(cand_pool):
            if time.time() >= deadline:
                break
            sub_dead = time.time() + per_candidate
            a_opt, sc_opt = _smooth_exponentiated_gradient(
                a0.copy(),
                sub_dead,
                best_seq=overall_best_seq,
                best_score=overall_best_score,
                beta_start=5.0,
                beta_end=80.0,
                eta0=1.0,
                eval_every=50,
            )
            if sc_opt < overall_best_score - 1e-12:
                overall_best_score = sc_opt
                overall_best_seq = a_opt.copy()
                print(f"[{time.time():.1f}] after quick GD seed {idx} new best {overall_best_score:.6f}")

    # ------------------------------------------------------------------
    # 5.  Longer smoothed‑gradient descent on the current global best
    # ------------------------------------------------------------------
    if time.time() < deadline:
        sub_dead = time.time() + (deadline - time.time()) * 0.35
        a_opt, sc_opt = _smooth_exponentiated_gradient(
            overall_best_seq.copy(),
            sub_dead,
            best_seq=overall_best_seq,
            best_score=overall_best_score,
            beta_start=5.0,
            beta_end=200.0,
            eta0=0.7,
            eval_every=80,
        )
        if sc_opt < overall_best_score - 1e-12:
            overall_best_score = sc_opt
            overall_best_seq = a_opt.copy()
            print(f"[{time.time():.1f}] long smooth GD improved best to {overall_best_score:.6f}")

    # ------------------------------------------------------------------
    # 6.  Upsample the best vector (if time permits) and polish again
    # ------------------------------------------------------------------
    while time.time() < deadline:
        cur_n = overall_best_seq.shape[0]
        target_n = cur_n * 2
        if target_n > 64000:   # keep FFT size modest
            break
        print(f"[{time.time():.1f}] upsampling {cur_n} → {target_n}")
        up = _upsample(overall_best_seq, target_n)
        up_sc = _objective(up)
        if up_sc < overall_best_score:
            overall_best_seq = up
            overall_best_score = up_sc
            print(f"[{time.time():.1f}] upsampled new best {overall_best_score:.6f}")

        # short smooth GD after upsample
        sub_dead = time.time() + (deadline - time.time()) * 0.2
        a_opt, sc_opt = _smooth_exponentiated_gradient(
            overall_best_seq.copy(),
            sub_dead,
            best_seq=overall_best_seq,
            best_score=overall_best_score,
            beta_start=10.0,
            beta_end=150.0,
            eta0=0.6,
            eval_every=60,
        )
        if sc_opt < overall_best_score - 1e-12:
            overall_best_seq = a_opt.copy()
            overall_best_score = sc_opt
            print(f"[{time.time():.1f}] after upsample GD new best {overall_best_score:.6f}")
        # continue the loop: try another doubling if time permits

    # ------------------------------------------------------------------
    # 7.  Final annealing phase
    # ------------------------------------------------------------------
    if time.time() < deadline:
        print(f"[{time.time():.1f}] final annealing phase")
        a_opt, sc_opt = _pairwise_annealing(
            overall_best_seq.copy(),
            deadline,
            best_seq=overall_best_seq,
            best_score=overall_best_score,
            rng=rng,
        )
        if sc_opt < overall_best_score - 1e-12:
            overall_best_seq = a_opt.copy()
            overall_best_score = sc_opt
            print(f"[{time.time():.1f}] after annealing new best {overall_best_score:.6f}")

    # ------------------------------------------------------------------
    print(
        f"Final best score (objective=2·n·max(conv)) = {overall_best_score:.6f} "
        f"(length={len(overall_best_seq)})"
    )
    return overall_best_seq.tolist()
