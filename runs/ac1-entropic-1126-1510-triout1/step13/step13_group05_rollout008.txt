def _cross_correlation(w, a):
    n = a.shape[0]
    a_rev = a[::-1]
    size = 1 << ((len(w) + n - 1) - 1).bit_length()
    W = np.fft.rfft(w, n=size)
    A_rev = np.fft.rfft(a_rev, n=size)
    conv = np.fft.irfft(W * A_rev, n=size)
    # conv length = len(w) + n -1
    slice_start = n - 1
    return conv[slice_start:slice_start + n]

"""
Propose a step‑function (non‑negative list) that minimises

    2·n·max(conv(seq,seq)) / (sum(seq)^2)

The routine consists of three stages.

1️⃣  **Broad sampling** – deterministic analytic families (uniform,
    arcsine, sine, cosine, quadratic, symmetric beta, edge‑weighted,
    power‑sum) on a handful of lengths, plus Dirichlet and sparse random
    seeds.  All candidates are normalised to sum = 1 and the 10 best
    (lowest exact objective) are retained.

2️⃣  **Smooth‑gradient refinement** – each retained seed is optimised with
    a large‑scale mirror‑descent on a *soft‑max* surrogate of the
    max‑convolution.  For a vector `a` we compute the full self‑convolution
    `c = a*a`, turn it into a probability distribution
    `w_i ∝ exp(α·c_i)`, and take the exact gradient of `Σ_i w_i·c_i`:
    `∇ = 2·cross_corr(w,a)`.  An exponentiated‑gradient update
    `a ← a·exp(−η·∇)` keeps the iterates non‑negative; after every step
    we project back onto the simplex (`Σa=1`).  The temperature `α`
    is annealed from a small value (≈2) to a larger one (≈30) so that the
    algorithm first moves on a smooth landscape and later focuses on the
    true maximum.  Step‑sizes decay as `η₀/√t`.  The best iterate seen
    across all seeds is stored.

3️⃣  **Greedy polishing** – a cheap hill‑climber randomly transfers a
    fraction of mass between two positions and keeps the change only if
    the exact objective strictly improves.

The whole process respects the supplied time budget (default 1000 s)
and finally returns a plain Python list of floats.  The sequence is
already normalised, so `evaluate_sequence` only needs to apply its
clipping/safety checks.

# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Euclidean projection of ``v`` onto the simplex {x≥0, Σx=s}."""
    if s < 0:
        raise ValueError("Simplex radius must be non‑negative")
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.where(u - cssv / (np.arange(1, len(v) + 1)) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    return np.maximum(v - theta, 0.0)


# ----------------------------------------------------------------------
def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Full linear self‑convolution ``a*a`` using a real FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1) - 1).bit_length()          # next power of two
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[: 2 * n - 1]


# ----------------------------------------------------------------------
def _objective(a: np.ndarray) -> float:
    """Exact objective 2·n·max(conv(a,a)) for a vector with Σa=1."""
    n = a.shape[0]
    b = _fft_self_convolve(a)
    return float(2.0 * n * b.max())

# Analytic families ----------------------------------------------------
def _distribution_uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _distribution_arcsine(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    vals /= vals.sum()
    return vals


def _distribution_sine(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64) + 0.5
    w = np.sin(np.pi * i / n)
    w /= w.sum()
    return w


def _distribution_cosine(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64) + 0.5
    w = 1.0 - np.cos(np.pi * i / n)
    w /= w.sum()
    return w


def _distribution_quadratic(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    x = (2.0 * i) / (n - 1) - 1.0
    w = 1.0 - x * x
    w = np.clip(w, 0.0, None)
    w /= w.sum()
    return w


def _distribution_beta_sym(n: int, p: float) -> np.ndarray:
    """Symmetric Beta(p,p) on a regular grid."""
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    w = np.exp((p - 1) * np.log(x) + (p - 1) * np.log(1.0 - x))
    if w.sum() == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _distribution_edge_weighted(n: int, p: float) -> np.ndarray:
    """
    Edge‑weighted family: w_i ∝ ((i+½)*(n‑i‑½))^{‑p}.
    """
    i = np.arange(n, dtype=np.float64) + 0.5
    j = (n - np.arange(n, dtype=np.float64) - 0.5)
    w = (i * j) ** (-p)
    w = np.clip(w, 0.0, None)
    total = w.sum()
    if total > 0:
        w /= total
    else:
        w = np.full(n, 1.0 / n)
    return w


def _distribution_power_sum(n: int, p: float) -> np.ndarray:
    """w_i ∝ (i+1)^{‑p} + (n‑i)^{‑p}."""
    i = np.arange(1, n + 1, dtype=np.float64)
    w = i ** (-p) + (n + 1 - i) ** (-p)
    w = np.clip(w, 0.0, None)
    w /= w.sum()
    return w


def _distribution_random_sparse(
    n: int, rng: np.random.Generator, density: float = 0.01
) -> np.ndarray:
    """Sparse random vector with a prescribed density of non‑zeros."""
    k = max(2, int(n * density))
    idx = rng.choice(n, size=k, replace=False)
    vals = np.zeros(n, dtype=np.float64)
    vals[idx] = rng.random(k)
    if vals.sum() == 0.0:
        vals[0] = 1.0
    vals /= vals.sum()
    return vals


def _mix_with_uniform(base: np.ndarray, lam: float) -> np.ndarray:
    """Convex combination λ·uniform + (1‑λ)·base."""
    n = base.shape[0]
    uniform = np.full(n, 1.0 / n, dtype=np.float64)
    return lam * uniform + (1.0 - lam) * base


# ----------------------------------------------------------------------
def _softmax_weights(conv: np.ndarray, alpha: float) -> np.ndarray:
    """Soft‑max of a convolution vector: w_i ∝ exp(α·c_i)."""
    # shift by the max to avoid overflow
    shift = conv.max()
    exp_part = np.exp(alpha * (conv - shift))
    total = exp_part.sum()
    if total == 0.0:
        # fallback to uniform – should not happen
        return np.full_like(conv, 1.0 / conv.size)
    return exp_part / total


# ----------------------------------------------------------------------
def _cross_corr_with_weights(w: np.ndarray, a: np.ndarray) -> np.ndarray:
    """
    Compute the vector g_j = Σ_i w_i * a_{i-j}   (linear cross‑correlation).
    Returned vector has length n = len(a) and corresponds to the gradient
    of Σ_i w_i·(a*a)_i.
    """
    n = a.shape[0]
    a_rev = a[::-1]
    m = w.shape[0]                     # 2·n‑1
    size = 1 << ((m + n - 1) - 1).bit_length()
    W = np.fft.rfft(w, n=size)
    A_rev = np.fft.rfft(a_rev, n=size)
    conv = np.fft.irfft(W * A_rev, n=size)
    # linear convolution length = m + n - 1.  The slice [n‑1 : n‑1+n]
    # contains exactly Σ_i w_i·a_{i‑j}.
    start = n - 1
    return conv[start:start + n]

def _smooth_gradient_descent(
    a0: np.ndarray,
    deadline: float,
    best_seq: np.ndarray,
    best_score: float,
    eta0: float = 0.25,
    alpha_start: float = 2.0,
    alpha_end: float = 30.0,
    max_iter: int = 200_000,
    report_every: int = 500,
) -> tuple[np.ndarray, float]:
    """
    Mirror‑descent on the soft‑max surrogate of max‑convolution.
    Uses an exponentiated‑gradient update `a ← a·exp(−η·∇)` and projects
    onto the simplex after each step.
    """
    a = a0.copy()
    n = a.shape[0]
    t = 0
    while time.time() < deadline and t < max_iter:
        t += 1
        # 1) current convolution
        conv = _fft_self_convolve(a)

        # 2) soft‑max weights
        alpha = alpha_start + (alpha_end - alpha_start) * (t / max_iter)
        w = _softmax_weights(conv, alpha)

        # 3) gradient of the surrogate (2·cross‑correlation)
        grad = 2.0 * _cross_corr_with_weights(w, a)

        # 4) multiplicative (exponentiated) update + simplex projection
        eta = eta0 / math.sqrt(t)
        a = a * np.exp(-eta * grad)
        a_sum = a.sum()
        if a_sum > 0:
            a = a / a_sum
        else:
            a = np.full(n, 1.0 / n, dtype=np.float64)

        # 5) occasional exact evaluation
        if t % report_every == 0:
            cur = _objective(a)
            if cur < best_score - 1e-14:
                best_score = cur
                best_seq = a.copy()
                print(f"[{time.time():.1f}] smooth‑gd it={t} new best {best_score:.6f}")

    return best_seq, best_score

def _pairwise_swap(
    a0: np.ndarray,
    deadline: float,
    best_seq: np.ndarray,
    best_score: float,
    max_trials: int | None = None,
) -> tuple[np.ndarray, float]:
    """
    Randomly move up to ~45 % of mass from one coordinate to another.
    Keep a move only if the exact objective strictly improves.
    """
    a = a0.copy()
    n = a.shape[0]
    trials = 0
    while time.time() < deadline:

        src = np.where(a > 1e-14)[0]
        if src.size == 0:

        i = np.random.choice(src)

        if j >= i:

        new_a[i] -= delta

            best_score = cur
            best_seq = a.copy()
            print(

            )
