    # softmax weights:
    max_conv = conv.max()  # maybe use stable: subtract max before exp

"""
Propose a non‑negative step function that minimizes ``evaluate_sequence``.

The algorithm consists of three stages:

1. **Rich seeding** – generate many analytic families (arcsine, Beta,
   edge‑weighted, sine‑weighted) and random Dirichlet / sparse vectors for a
   set of lengths (up to 12 000).  Keep the ten best seeds.

2. **Smoothed projected gradient descent** – minimise a smooth approximation
   of the true objective  ``2·n·max(conv(a,a))`` while keeping ``a`` on the
   probability simplex.  The smooth objective is the log‑sum‑exp of the
   convolution values with a temperature parameter ``α``.  ``α`` is increased
   gradually (1 → 2 → 5 → 10 → 20 → 50 → 100 → 200) so that the optimisation
   first sees a smooth landscape and later focuses on the true maximum.
   Gradients are obtained analytically via FFT‐based cross‑correlation.
   After each step the vector is projected onto the simplex.  Jitter is
   injected when progress stalls.

3. **Local refinement** – a deterministic pairwise mass‑transfer sweep
   followed by a cheap simulated‑annealing walk explores the neighbourhood
   of the best solution found so far.

If time permits the best solution is up‑sampled to a larger length and the
gradient‑descent phase is repeated, because larger sequences tend to have a
lower bound.  The final vector is returned as a plain ``list[float]``; the
caller will perform the clipping and final normalisation.

The total wall‑clock budget is respected (default 1000 s).  All helper
functions are top‑level and contain no closures.
"""

import time

# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x >= 0, Σx = s}."""
    if s < 0:
        raise ValueError("simplex radius must be non‑negative")
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.where(u - cssv / (np.arange(1, len(v) + 1)) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    return np.maximum(v - theta, 0.0)


# ----------------------------------------------------------------------
def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Full (length 2·n‑1) self‑convolution a*a using a real FFT."""
    n = a.shape[0]
    need = 2 * n - 1
    size = 1 << ((need - 1).bit_length())
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[:need]


# ----------------------------------------------------------------------
def _objective(a: np.ndarray) -> float:
    """Exact objective 2·n·max(conv(a,a)) for a vector with Σa = 1."""
    n = a.shape[0]
    conv = _fft_self_convolve(a)
    return 2.0 * n * conv.max()

# Analytic families ----------------------------------------------------
def _distribution_uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _distribution_arcsine(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    vals /= vals.sum()
    return vals


def _distribution_beta_sym(n: int, p: float) -> np.ndarray:
    """Beta(p,p) on the discrete grid."""
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    w = np.exp((p - 1) * np.log(x) + (p - 1) * np.log(1.0 - x))
    if w.sum() == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _distribution_beta_asym(n: int, a: float, b: float) -> np.ndarray:
    """Beta(a,b) on the discrete grid."""
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    w = np.exp((a - 1) * np.log(x) + (b - 1) * np.log(1.0 - x))
    if w.sum() == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _distribution_edge_weighted(n: int, p: float) -> np.ndarray:
    """U‑shaped family w_i ∝ ((i+½)*(n‑i‑½))^(−p)."""
    i = np.arange(n, dtype=np.float64) + 0.5
    j = (n - np.arange(n, dtype=np.float64) - 0.5)
    w = (i * j) ** (-p)
    if w.sum() == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _distribution_sine_weighted(n: int, p: float) -> np.ndarray:
    """Weight ∝ sin(π (i+½)/n)^p."""
    theta = np.pi * (np.arange(n, dtype=np.float64) + 0.5) / n
    w = np.sin(theta) ** p
    if w.sum() == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _distribution_random_sparse(n: int, rng: np.random.Generator,
                               density: float = 0.01) -> np.ndarray:
    """Sparse random vector (few non‑zeros)."""
    k = max(2, int(n * density))
    idx = rng.choice(n, size=k, replace=False)
    vals = np.zeros(n, dtype=np.float64)
    vals[idx] = rng.random(k)
    if vals.sum() == 0.0:
        vals[0] = 1.0
    vals /= vals.sum()
    return vals


def _mix_with_uniform(base: np.ndarray, lam: float) -> np.ndarray:
    """Convex combination lam·U + (1‑lam)·base, U uniform."""
    n = base.shape[0]
    uniform = np.full(n, 1.0 / n, dtype=np.float64)
    return lam * uniform + (1.0 - lam) * base

# Smoothing utilities ---------------------------------------------------
def _smooth_objective_and_grad(a: np.ndarray, alpha: float):
    """
    Return a smooth approximation of max(conv(a,a)) and its gradient.
    The smooth max is  max + log‑sum‑exp / α.
    Gradient = 2 * cross‑correlation( w , a ) where
    w_k ∝ exp(α·conv_k).
    """
    n = a.shape[0]

    # 1) full self‑convolution
    conv = _fft_self_convolve(a)                     # length 2n‑1

    # 2) log‑sum‑exp smoothing
    max_conv = conv.max()
    shifted = conv - max_conv                       # avoid overflow
    exp_shifted = np.exp(alpha * shifted)
    sum_exp = exp_shifted.sum()
    smooth_max = max_conv + np.log(sum_exp) / alpha

    # 3) weights for the gradient
    w = exp_shifted / sum_exp                       # length 2n‑1, sum to 1

    # 4) gradient = 2 * (w * a_rev) sliced to length n
    a_rev = a[::-1]
    need = len(w) + len(a_rev) - 1                 # (2n‑1) + n – 1 = 3n‑2
    size = 1 << ((need - 1).bit_length())
    W = np.fft.rfft(w, n=size)
    A_rev = np.fft.rfft(a_rev, n=size)
    grad_full = np.fft.irfft(W * A_rev, n=size)    # length = need
    grad = 2.0 * grad_full[n - 1: n - 1 + n]       # central n entries
    return smooth_max, grad


# ----------------------------------------------------------------------
def _smooth_proj_grad_descent(a: np.ndarray,
                              deadline: float,
                              best_seq: np.ndarray,
                              best_score: float,
                              eta0: float = 0.3,
                              jitter_interval: int = 2500):
    """
    Projected gradient descent on the smoothed objective.
    ``a`` must be on the simplex (Σa = 1).  Returns the best vector seen.
    """
    n = a.shape[0]
    t = 0
    no_improve = 0
    rng = np.random.default_rng()
    # schedule of (iteration, α) – α grows slowly as optimisation proceeds
    schedule = [(0, 0.5), (2000, 2.0), (4000, 5.0), (6000, 10.0),
                (8000, 20.0), (12000, 50.0), (16000, 100.0), (20000, 200.0)]

    cur_alpha = schedule[0][1]
    sched_idx = 0

    while time.time() < deadline and t < 50000:
        t += 1

        # possibly increase smoothing parameter
        if sched_idx + 1 < len(schedule) and t >= schedule[sched_idx + 1][0]:
            sched_idx += 1
            cur_alpha = schedule[sched_idx][1]

        smooth_val, grad = _smooth_objective_and_grad(a, cur_alpha)
        cur_obj = 2.0 * n * smooth_val

        if cur_obj < best_score - 1e-14:
            best_score = cur_obj
            best_seq = a.copy()
            no_improve = 0
            print(f"[{time.time():.1f}] it={t:5d} α={cur_alpha:6.1f} "
                  f"new best {best_score:.6f}")
        else:
            no_improve += 1

        # step size decays like 1/√t
        eta = eta0 / math.sqrt(t)
        a_new = a - eta * grad
        a_new = _project_onto_simplex(a_new, s=1.0)

        # occasional jitter to escape plateaus
        if no_improve >= jitter_interval:
            jitter = rng.exponential(scale=0.003, size=n)
            a_new = a_new + jitter
            a_new = _project_onto_simplex(a_new, s=1.0)
            no_improve = 0

        a = a_new

    return best_seq, best_score

def _pairwise_swap_local(a: np.ndarray,
                         deadline: float,
                         best_seq: np.ndarray,
                         best_score: float,
                         max_trials: int | None = None):
    """Greedy deterministic neighbour search – move a small amount of mass."""
    n = a.shape[0]
    trials = 0
    rng = np.random.default_rng()
    while time.time() < deadline:
        if max_trials is not None and trials >= max_trials:
            break
        trials += 1

        src = np.where(a > 1e-12)[0]
        if src.size == 0:
            break
        i = rng.choice(src)
        j = rng.integers(n - 1)
        if j >= i:
            j += 1

        delta = a[i] * rng.random() * 0.35
        if delta < 1e-14:
            continue

        new_a = a.copy()
        new_a[i] -= delta
        new_a[j] += delta

        cur = _objective(new_a)
        if cur < best_score - 1e-14:
            a = new_a
            best_score = cur
            best_seq = a.copy()
            print(f"[{time.time():.1f}] swap {i}->{j} Δ={delta:.3e} "
                  f"new best {best_score:.6f}")
    return best_seq, best_score


# ----------------------------------------------------------------------
def _random_local_search(a: np.ndarray,
                         deadline: float,
                         best_seq: np.ndarray,
                         best_score: float,
                         init_temp: float = 1e-2,
                         decay: float = 0.9995):
    """Cheap simulated‑annealing walk."""
    n = a.shape[0]
    rng = np.random.default_rng()
    T = init_temp
    while time.time() < deadline:
        i = rng.integers(n)
        j = rng.integers(n - 1)
        if j >= i:
            j += 1
        delta = a[i] * rng.random() * 0.3
        if delta < 1e-14:
            continue
        new_a = a.copy()
        new_a[i] -= delta
        new_a[j] += delta
        cur = _objective(new_a)
        if cur < best_score - 1e-14:
            a = new_a
            best_score = cur
            best_seq = a.copy()
            print(f"[{time.time():.1f}] SA improve {i}->{j} Δ={delta:.3e} "
                  f"best {best_score:.6f}")
        else:
            if rng.random() < math.exp(-(cur - best_score) / max(T, 1e-12)):
                a = new_a
        T *= decay
    return best_seq, best_score


# ----------------------------------------------------------------------
def _interpolate_distribution(seq: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation of a distribution to a new length."""
    n = len(seq)
    if n == 0:
        return np.zeros(new_n, dtype=np.float64)
    old_x = (np.arange(n, dtype=np.float64) + 0.5) / n
    new_x = (np.arange(new_n, dtype=np.float64) + 0.5) / new_n
    return np.interp(new_x, old_x, seq, left=0.0, right=0.0)

        for p in [0.05, 0.1, 0.2, 0.4, 0.7, 1.0, 1.5, 2.0]:
            base = _distribution_beta_sym(n, p)
            for lam in [0.0, 0.25, 0.5, 0.75, 1.0]:
                cand = _mix_with_uniform(base, lam)
                all_candidates.append((_objective(cand), cand.copy()))

        # edge‑weighted family – explore a broader range of exponents
        for p in [0.1, 0.3, 0.5, 0.8, 1.0, 2.0, 3.0, 5.0]:
            base = _distribution_edge_weighted(n, p)
            for lam in [0.0, 0.5, 1.0]:
                cand = _mix_with_uniform(base, lam)
                all_candidates.append((_objective(cand), cand.copy()))

        # sine‑weighted family
        for p in [1, 2, 3, 4, 5]:
            base = _distribution_sine_weighted(n, p)
            for lam in [0.0, 0.5, 1.0]:
                cand = _mix_with_uniform(base, lam)
                all_candidates.append((_objective(cand), cand.copy()))

        # asymmetric beta – few random draws
        for _ in range(3):
            a_param = 10 ** rng.uniform(-2, 0)   # 0.01 … 1
            b_param = 10 ** rng.uniform(-2, 0)
            cand = _distribution_beta_asym(n, a_param, b_param)
            all_candidates.append((_objective(cand), cand.copy()))

        # random Dirichlet vectors
        for _ in range(3):
            alpha = rng.uniform(0.01, 2.0)
            cand = rng.dirichlet(np.full(n, alpha))
            all_candidates.append((_objective(cand), cand.copy()))

        # sparse random vectors
        for _ in range(2):
            cand = _distribution_random_sparse(n, rng, density=0.008)
            all_candidates.append((_objective(cand), cand.copy()))

    # ------------------------------------------------------------------
    # 2.  Keep the very best seeds (top 10)
    all_candidates.sort(key=lambda x: x[0])
    top_seeds = [seq for _, seq in all_candidates[:10]]
    best_score, best_seq = all_candidates[0][0], all_candidates[0][1].copy()
    print(f"[{time.time():.1f}] initial best score {best_score:.6f} "
          f"(len={len(best_seq)})")

    # ------------------------------------------------------------------
    # 3.  Import any externally supplied construction, if present
    try:
        prev = np.asarray(height_sequence_1, dtype=np.float64)  # type: ignore[name-defined]
        if prev.ndim == 1 and prev.size > 0:
            prev = prev / prev.sum()
            sc = _objective(prev)
            if sc < best_score:
                best_score, best_seq = sc, prev.copy()
                print(f"[{time.time():.1f}] imported previous best {sc:.6f}")
            all_candidates.append((sc, prev.copy()))
            top_seeds.append(prev.copy())
    except Exception:
        pass

    # ------------------------------------------------------------------
    # 4.  Smoothed projected gradient descent on the most promising seeds
    remaining = deadline - time.time()
    if remaining > 20.0 and top_seeds:
        per_seed = remaining / len(top_seeds)
        print(f"[{time.time():.1f}] starting smoothed gradient descent on "
              f"{len(top_seeds)} seeds (≈{per_seed:.1f}s each)")
        for idx, seed_vec in enumerate(top_seeds):
            if time.time() >= deadline:
                break
            sub_deadline = time.time() + per_seed
            # ensure the vector lives on the simplex
            seed_vec = seed_vec / seed_vec.sum()
            opt_seq, opt_score = _smooth_proj_grad_descent(
                seed_vec.copy(),
                sub_deadline,
                best_seq=best_seq,
                best_score=best_score,
                eta0=0.3,
                jitter_interval=2500,
            )
            if opt_score < best_score:
                best_score, best_seq = opt_score, opt_seq.copy()
                print(f"[{time.time():.1f}] after seed {idx} new best {best_score:.6f}")
            if best_score <= 1.503:
                break

    # ------------------------------------------------------------------
    # 5.  Optional up‑sampling to larger lengths (if time permits)
    while time.time() < deadline:
        cur_n = len(best_seq)
        if cur_n >= 20000:
            break
        new_n = min(20000, int(cur_n * 1.5))
        if new_n == cur_n:
            break
        print(f"[{time.time():.1f}] up‑sampling from {cur_n} → {new_n}")
        up_seq = _interpolate_distribution(best_seq, new_n)
        up_seq = up_seq / up_seq.sum()
        sub_deadline = time.time() + min(30.0, deadline - time.time())
        up_seq, up_score = _smooth_proj_grad_descent(
            up_seq,
            sub_deadline,
            best_seq=best_seq,
            best_score=best_score,
            eta0=0.2,
            jitter_interval=2000,
        )
        if up_score < best_score:
            best_score, best_seq = up_score, up_seq.copy()
            print(f"[{time.time():.1f}] after up‑sampling new best {best_score:.6f}")

    # ------------------------------------------------------------------
    # 6.  Deterministic pairwise refinement
    if time.time() < deadline:
        print(f"[{time.time():.1f}] deterministic pairwise refinement")
        best_seq, best_score = _pairwise_swap_local(
            best_seq.copy(),
            deadline,
            best_seq,
            best_score,
            max_trials=250_000,
        )

    # ------------------------------------------------------------------
    # 7.  Final cheap simulated‑annealing walk
    if time.time() < deadline:
        print(f"[{time.time():.1f}] final simulated‑annealing walk")
        best_seq, best_score = _random_local_search(
            best_seq.copy(),
            deadline,
            best_seq,
            best_score,
            init_temp=0.008,
            decay=0.9997,
        )

    # ------------------------------------------------------------------
    print(f"Final best score: {best_score:.6f} (length={len(best_seq)})")
    # Return a plain Python list – ``evaluate_sequence`` will apply its own clipping.
    return best_seq.tolist()
