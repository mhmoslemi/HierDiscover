"""
step_function_search_v2.py

A 1000 s “smart‐search” for a non‑negative step function that minimises

    2·n·max(conv(a,a)) / (∑a)²

The algorithm follows three stages:

1. **Rich seeding** – generate a very diverse pool of probability vectors
   (uniform, arcsine, Hann, triangle, edge‑weighted with both
   positive and negative exponents, symmetric Beta, sin‑power and random
   Dirichlet).  All vectors are normalised to sum = 1.
   The 40 best seeds are kept.

2. **Accelerated mirror descent** – each seed is polished with an
   exponentiated‑gradient method that uses the *full* active set of
   maximal convolution entries (instead of a single argmax).  A simple
   Adam‑style momentum on the sub‑gradient is added and the iterate is
   forced to stay symmetric.  Short runs are performed on many seeds,
   then a longer run on the current global best.

3. **Upsampling & annealing** – the best vector is linearly up‑sampled
   (twice, three times …) while staying on the simplex, each time
   followed by a brief mirror‑descent polish.  Finally a pairwise
   mass‑transfer simulated‑annealing phase is applied to escape any
   remaining local optimum.

All work respects the 1000 s budget (2 s safety margin).  The function
``propose_candidate`` returns a plain ``list[float]`` ready for
``evaluate_sequence``.
"""

# ----------------------------------------------------------------------
# Simplex utilities
# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Euclidean projection of a vector onto the simplex {x≥0, Σx=s}."""
    if s < 0:
        raise ValueError("simplex radius must be non‑negative")
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.where(u - cssv / (np.arange(1, len(v) + 1)) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Convolution helpers (FFT based, linear convolution)
# ----------------------------------------------------------------------
def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Full linear self‑convolution a*a using a real FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1) - 1).bit_length()  # next power‑of‑two
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[: 2 * n - 1]


# ----------------------------------------------------------------------
# Exact objective (scale‑invariant, sum=1 ⇒ denominator = 1)
# ----------------------------------------------------------------------
def _objective(a: np.ndarray) -> float:
    """Return the exact value of ``evaluate_sequence`` for a probability
    vector a (Σa = 1)."""
    n = a.shape[0]
    b = _fft_self_convolve(a)
    max_b = b.max()
    if max_b < 0:
        max_b = 0.0
    return float(2.0 * n * max_b)


# ----------------------------------------------------------------------
# Analytic families (all return a probability vector, sum=1)
# ----------------------------------------------------------------------
def _distribution_uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _distribution_arcsine(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    vals /= vals.sum()
    return vals


def _distribution_hann(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    w = 0.5 - 0.5 * np.cos(2.0 * np.pi * i / (n - 1))
    w_sum = w.sum()
    if w_sum == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w_sum
    return w


def _distribution_triangle(n: int) -> np.ndarray:
    """Linear rise to centre then fall (symmetric tent)."""
    i = np.arange(n, dtype=np.float64) + 0.5
    w = np.minimum(i, n - i)
    w /= w.sum()
    return w


def _distribution_edge_weighted(n: int, p: float) -> np.ndarray:
    """Weight ∝ (i·j)^{‑p} where i,j are distances to the two edges."""
    i = np.arange(n, dtype=np.float64) + 0.5
    j = (n - np.arange(n, dtype=np.float64) - 0.5)
    w = (i * j) ** (-p)
    w_sum = w.sum()
    if w_sum == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w_sum
    return w


def _distribution_generalized_normal(n: int, q: float) -> np.ndarray:
    """Density ∝ exp(‑|x‑½|^{q}) on [0,1]."""
    i = np.arange(n, dtype=np.float64) + 0.5
    x = i / n - 0.5
    w = np.exp(- (np.abs(x) / 0.5) ** q)
    w_sum = w.sum()
    if w_sum == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w_sum
    return w


def _distribution_beta_symmetric(n: int, alpha: float) -> np.ndarray:
    """Symmetric Beta(α,α) density, discretised on {i+½}/n."""
    i = np.arange(n, dtype=np.float64) + 0.5
    x = i / n
    # avoid 0^neg and 1^neg; clamp to epsilon
    eps = 1e-12
    x = np.clip(x, eps, 1 - eps)
    w = (x ** (alpha - 1)) * ((1 - x) ** (alpha - 1))
    w_sum = w.sum()
    if w_sum == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w_sum
    return w


def _distribution_sin_power(n: int, power: float) -> np.ndarray:
    """Weight ∝ sin(π·x)^{power} on (0,1).  Zero at the ends."""
    i = np.arange(n, dtype=np.float64) + 0.5
    x = i / n
    w = np.sin(np.pi * x) ** power
    w_sum = w.sum()
    if w_sum == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w_sum
    return w


def _distribution_random_sparse(
    n: int, rng: np.random.Generator, density: float = 0.01
) -> np.ndarray:
    """Sparse random vector with a given density."""
    k = max(2, int(n * density))
    idx = rng.choice(n, size=k, replace=False)
    vals = np.zeros(n, dtype=np.float64)
    vals[idx] = rng.random(k)
    if vals.sum() == 0.0:
        vals[0] = 1.0
    vals /= vals.sum()
    return vals


def _mix_with_uniform(base: np.ndarray, lam: float) -> np.ndarray:
    """Convex combination with the uniform distribution."""
    n = base.shape[0]
    uniform = np.full(n, 1.0 / n, dtype=np.float64)
    return lam * uniform + (1.0 - lam) * base


# ----------------------------------------------------------------------
# Gradient of the *active* maximal convolution entry (sub‑gradient)
# ----------------------------------------------------------------------
def _grad_of_active_max_conv(a: np.ndarray, eps: float = 1e-12) -> np.ndarray:
    """
    Compute the sub‑gradient of the maximal convolution entry.
    If several entries are (almost) maximal, their gradients are summed.
    Returns a gradient vector of the same shape as ``a``.
    """
    n = a.shape[0]
    conv = _fft_self_convolve(a)
    max_val = conv.max()
    # active indices within a tiny absolute tolerance
    active = np.where(conv >= max_val - eps)[0]

    grad = np.zeros_like(a)
    for k in active:
        if k < n:
            # overlap segment a[0..k] with reversed a[0..k]
            seg = a[k::-1]               # length k+1
            grad[: k + 1] += 2.0 * seg
        else:
            start = k - (n - 1)          # first index that participates
            seg = a[start:][::-1]        # length 2n‑1‑k
            grad[start:] += 2.0 * seg
    return grad


# ----------------------------------------------------------------------
# Exponentiated‑gradient descent with simple Adam‑style momentum
# ----------------------------------------------------------------------
def _adam_expgd(
    a: np.ndarray,
    deadline: float,
    best_seq: np.ndarray,
    best_score: float,
    eta0: float = 0.8,
    beta1: float = 0.9,
    beta2: float = 0.999,
    eps: float = 1e-8,
    eval_every: int = 70,
) -> tuple[np.ndarray, float]:
    """
    Mirror‑descent (exponentiated gradient) with Adam‑style momentum
    on the exact objective.  The update is

        a ← a · exp(‑η·m̂ / (√v̂+ϵ))

    where m̂, v̂ are bias‑corrected first/second moments of the sub‑gradient.
    After each step the vector is symmetrised and projected onto the simplex.
    """
    n = a.shape[0]
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    t = 0
    while time.time() < deadline:
        t += 1
        grad = _grad_of_active_max_conv(a)
        # Adam moments
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)

        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)

        # step size schedule (1/√t) multiplies the whole factor
        eta = eta0 / math.sqrt(t)

        # exponentiated update, keep positivity
        a = a * np.exp(-eta * m_hat / (np.sqrt(v_hat) + eps))

        # force symmetry (a_i = a_{n-1-i})
        a = (a + a[::-1]) * 0.5

        # project onto the simplex Σa = 1
        s = a.sum()
        if s == 0.0:
            a = np.full(n, 1.0 / n, dtype=np.float64)
        else:
            a = a / s

        if t % eval_every == 0:
            cur = _objective(a)
            if cur < best_score - 1e-14:
                best_score = cur
                best_seq = a.copy()
                print(
                    f"[{time.time():.1f}] Adam‑GD it={t} new global best {best_score:.6f}"
                )
    return best_seq, best_score


# ----------------------------------------------------------------------
# Simulated‑annealing style pairwise mass transfers
# ----------------------------------------------------------------------
def _pairwise_annealing(
    a: np.ndarray,
    deadline: float,
    best_seq: np.ndarray,
    best_score: float,
    rng: np.random.Generator,
) -> tuple[np.ndarray, float]:
    """
    Randomly transfer a fraction of mass between two bins.
    Improvements are always accepted; deteriorations may be accepted
    according to a linear cooling schedule.
    """
    n = a.shape[0]
    start_time = time.time()
    total_time = max(deadline - start_time, 1e-6)
    while time.time() < deadline:
        # pick a source with non‑negligible mass
        src_candidates = np.where(a > 1e-12)[0]
        if src_candidates.size == 0:
            break
        i = rng.choice(src_candidates)

        # pick a distinct target uniformly
        j = rng.integers(n - 1)
        if j >= i:
            j += 1

        # move a random fraction of the source mass
        delta = a[i] * rng.random() * 0.4
        if delta < 1e-16:
            continue

        new_a = a.copy()
        new_a[i] -= delta
        new_a[j] += delta

        # maintain symmetry (helps objective) – mirror the move
        new_a[i] = (new_a[i] + new_a[n - 1 - i]) * 0.5
        new_a[n - 1 - i] = new_a[i]
        new_a[j] = (new_a[j] + new_a[n - 1 - j]) * 0.5
        new_a[n - 1 - j] = new_a[j]

        cur = _objective(new_a)

        elapsed = time.time() - start_time
        temperature = max(0.0, 1.0 - elapsed / total_time) * best_score * 0.01

        accept = False
        if cur < best_score - 1e-14:
            accept = True
        elif temperature > 0.0:
            prob = math.exp(-(cur - best_score) / temperature)
            if rng.random() < prob:
                accept = True

        if accept:
            a = new_a
            if cur < best_score - 1e-14:
                best_score = cur
                best_seq = a.copy()
                print(
                    f"[{time.time():.1f}] anneal i={i}→j={j} δ={delta:.3e} new best {best_score:.6f}"
                )
    return best_seq, best_score


# ----------------------------------------------------------------------
# Upsampling (linear interpolation while preserving the simplex)
# ----------------------------------------------------------------------
def _upsample(a: np.ndarray, new_n: int) -> np.ndarray:
    """Interpolate a probability vector to a larger length."""
    old_n = a.shape[0]
    if new_n == old_n:
        return a.copy()
    x_old = (np.arange(old_n, dtype=np.float64) + 0.5) / old_n
    x_new = (np.arange(new_n, dtype=np.float64) + 0.5) / new_n
    a_new = np.interp(x_new, x_old, a, left=0.0, right=0.0)
    a_new = np.maximum(a_new, 0.0)
    s = a_new.sum()
    if s == 0.0:
        a_new[:] = 1.0 / new_n
    else:
        a_new /= s
    # enforce symmetry
    a_new = (a_new + a_new[::-1]) * 0.5
    return a_new

            overall_best_seq = a_opt.copy()
            overall_best_score = sc_opt
            print(f"[{time.time():.1f}] GD after upsample new best {overall_best_score:.6f}")

    # ------------------------------------------------------------------
    # 7. Final simulated‑annealing polishing
    # ------------------------------------------------------------------
    if time.time() < deadline:
        print(f"[{time.time():.1f}] final annealing phase")
        a_opt, sc_opt = _pairwise_annealing(
            overall_best_seq.copy(),
            deadline,
            best_seq=overall_best_seq,
            best_score=overall_best_score,
            rng=rng,
        )
        if sc_opt < overall_best_score - 1e-12:
            overall_best_seq = a_opt.copy()
            overall_best_score = sc_opt
            print(f"[{time.time():.1f}] after annealing new best {overall_best_score:.6f}")

    # ------------------------------------------------------------------
    print(
        f"Final best score (objective=2·n·max(conv)): {overall_best_score:.6f} "
        f"(length={len(overall_best_seq)})"
    )
    # Return as plain Python list – evaluate_sequence expects a list
    return overall_best_seq.tolist()
