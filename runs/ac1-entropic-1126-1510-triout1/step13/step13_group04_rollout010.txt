def _smooth_gradient_descent(a0, deadline, best_seq, best_score):
    t=0
    a = a0.copy()
    alpha = alpha_start
    step_size = 0.5  # initial learning rate
    while time.time() < deadline:
        # compute conv
        conv = _fft_self_convolve(a)
        max_conv = conv.max()
        # compute smooth weights
        # shift conv to avoid overflow
        shift = conv.max()
        exp_vals = np.exp(alpha * (conv - shift))
        w = exp_vals / exp_vals.sum()
        # gradient
        grad = 2 * np.correlate(w, a, mode='valid')
        # apply Adam update
        m = beta1*m + (1-beta1)*grad
        v = beta2*v + (1-beta2)*(grad**2)
        m_hat = m/(1-beta1**t)
        v_hat = v/(1-beta2**t)
        a = a - step_size * m_hat / (np.sqrt(v_hat) + epsilon)
        # project onto simplex
        a = _project_onto_simplex(a, s=1.0)
        # maybe increase alpha gradually
        if t % inc_interval == 0:
            alpha = min(alpha_end, alpha * 1.2) ...
        # evaluate objective occasionally
        cur = _objective(a)
        if cur < best_score:
            best_score = cur
            best_seq = a.copy()
            print(f"[...] new best ...")
        t+=1
    return best_seq, best_score

def _smooth_exp_gradient_descent(a, deadline, best_seq, best_score,
                                alpha_start=10.0, alpha_end=300.0,
                                eta0=0.8, eval_every=80,
                                alpha_inc_factor=1.05, alpha_update_interval=400):
    t = 0
    alpha = alpha_start
    n = a.shape[0]
    while time.time() < deadline:
        t += 1
        # Compute convolution
        conv = _fft_self_convolve(a)
        # Compute softmax weights
        maxc = conv.max()
        # If alpha too large can cause overflow, shift
        w = np.exp(alpha * (conv - maxc))
        w_sum = w.sum()
        if w_sum == 0.0:
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum
        # Gradient (weighted sum of active constraints)
        # gradient_i = 2 * sum_k w_k * a_{k-i}
        # we compute via correlation
        grad = 2.0 * np.correlate(w, a, mode='valid')
        # Exponentiated update
        eta = eta0 / math.sqrt(t)
        a = a * np.exp(-eta * grad)
        # Project to simplex
        s = a.sum()
        if s == 0.0:
            a = np.full(n, 1.0 / n)
        else:
            a = a / s
        # optionally increase alpha
        if t % alpha_update_interval == 0:
            alpha = min(alpha * alpha_inc_factor, alpha_end)
        # evaluate periodically
        if t % eval_every == 0:
            cur = _objective(a)
            if cur < best_score - 1e-14:
                best_score = cur
                best_seq = a.copy()
                print(f"[{time.time():.1f}] SM EG it={t} new global best {best_score:.6f}")
    return best_seq, best_score

"""
Propose a non‑negative step function that (almost) minimizes
``evaluate_sequence``.

Algorithm
---------

1️⃣ Load a previously found good sequence (global ``height_sequence_1``) if it
   exists, otherwise start from a uniform distribution.

2️⃣ Iteratively double the length of the current best vector (linear interpolation
   that preserves the simplex).  After each up‑sampling we run a **smooth
   exponentiated‑gradient descent**:

   * The max‑convolution ‖a∗a‖∞ is approximated by a soft‑max
     `w_k ∝ exp(α·(conv_k−max(conv)))`.  The gradient of the weighted sum
     `∑_k w_k conv_k` is `g_i = 2·∑_k w_k·a_{k−i}`.
   * The gradient is obtained efficiently by one extra FFT
     (correlation of the weight vector with the current distribution).
   * Update is multiplicative (mirror descent on the simplex):
     `a ← a·exp(−η·g)`, followed by Euclidean projection onto the simplex.
   * α is slowly increased (soft‑max becomes sharper) while the step size
     η decays as 1/√t.

   The loop stops when the overall time budget (1000 s) is exhausted.

3️⃣ When no more up‑samplings fit into the remaining time, a final run of the
   smooth descent is performed on the current best vector.

The whole procedure respects the 1000 s budget (2 s safety buffer) and returns a
plain ``list[float]`` ready for ``evaluate_sequence``.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Basic utilities
# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x≥0, Σx=s}."""
    if s < 0:
        raise ValueError("simplex radius must be non‑negative")
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.where(u - cssv / (np.arange(1, len(v) + 1)) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Full linear self‑convolution a*a using a real FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1) - 1).bit_length()
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[: 2 * n - 1]


def _objective(a: np.ndarray) -> float:
    """Exact value of ``evaluate_sequence`` (scale‑invariant)."""
    n = a.shape[0]
    b = _fft_self_convolve(a)
    max_b = b.max()
    sum_a = a.sum()
    if sum_a < 0.01:
        return float("inf")
    return float(2.0 * n * max_b / (sum_a * sum_a))


def _upsample(a: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation of a probability vector to a larger length."""
    old_n = a.shape[0]
    if new_n == old_n:
        return a.copy()
    x_old = (np.arange(old_n, dtype=np.float64) + 0.5) / old_n
    x_new = (np.arange(new_n, dtype=np.float64) + 0.5) / new_n
    a_new = np.interp(x_new, x_old, a, left=0.0, right=0.0)
    a_new = np.maximum(a_new, 0.0)
    if a_new.sum() == 0.0:
        a_new[:] = 1.0 / new_n
    else:
        a_new /= a_new.sum()
    return a_new


# ----------------------------------------------------------------------
# Gradient of weighted convolution (used by the smoothed descent)
# ----------------------------------------------------------------------
def _grad_weighted_conv(w: np.ndarray, a: np.ndarray) -> np.ndarray:
    """
    Compute  g_i = 2·∑_k w_k·a_{k−i}
    using a single FFT (correlation of w with a).
    """
    n = a.shape[0]
    a_rev = a[::-1]
    size = 1 << ((len(w) + n - 1).bit_length())
    W = np.fft.rfft(w, n=size)
    A_rev = np.fft.rfft(a_rev, n=size)
    conv = np.fft.irfft(W * A_rev, n=size)
    start = n - 1
    grad = 2.0 * conv[start:start + n]
    return grad


# ----------------------------------------------------------------------
# Smoothed exponentiated‑gradient descent (core refinement)
# ----------------------------------------------------------------------
def _smooth_exp_gradient_descent(
    a: np.ndarray,
    deadline: float,
    best_seq: np.ndarray,
    best_score: float,
    alpha_start: float = 5.0,
    alpha_end: float = 300.0,
    alpha_inc_factor: float = 1.07,
    alpha_update_interval: int = 400,
    eta0: float = 0.8,
    eval_every: int = 80,
) -> tuple[np.ndarray, float]:
    """
    Minimise the smoothed version of max(conv) using multiplicative updates.
    The soft‑max temperature (α) is increased slowly so that the algorithm
    concentrates on the true maximum as optimisation proceeds.
    """
    t = 0
    alpha = alpha_start
    n = a.shape[0]

    while time.time() < deadline:
        t += 1

        # self‑convolution (needed for the soft‑max)
        conv = _fft_self_convolve(a)

        # soft‑max weights (shifted by the max for numerical safety)
        mx = conv.max()
        w = np.exp(alpha * (conv - mx))
        w_sum = w.sum()
        if w_sum == 0.0:
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        # gradient of the weighted sum of convolution entries
        grad = _grad_weighted_conv(w, a)

        # multiplicative (exponentiated) update
        eta = eta0 / math.sqrt(t)
        a = a * np.exp(-eta * grad)

        # projection back onto the simplex (keeps sum = 1, non‑negative)
        s = a.sum()
        if s <= 0.0:
            a = np.full(n, 1.0 / n, dtype=np.float64)
        else:
            a = a / s

        # gradually sharpen the soft‑max
        if t % alpha_update_interval == 0:
            alpha = min(alpha * alpha_inc_factor, alpha_end)

        # occasional exact evaluation
        if t % eval_every == 0:
            cur = _objective(a)
            if cur < best_score - 1e-14:
                best_score = cur
                best_seq = a.copy()
                print(f"[{time.time():.1f}] SM‑EG it={t} new best {best_score:.6f}")

    return best_seq, best_score


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------

    """

    """
    rng = np.random.default_rng(seed)

    deadline = start + budget_s - 2.0   # safety margin

    # ------------------------------------------------------------------
    # 0.  Initialise from a previously discovered construction if available.
    # ------------------------------------------------------------------
    try:
        init = np.asarray(height_sequence_1, dtype=np.float64)   # type: ignore[name-defined]
        if init.ndim != 1 or init.size == 0:
            raise ValueError
        init = np.maximum(init, 0.0)
        s = init.sum()

    except Exception:
