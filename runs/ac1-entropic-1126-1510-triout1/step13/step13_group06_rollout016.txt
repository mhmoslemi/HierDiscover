        # gradient sum
        grad=np.zeros_like(a)
        for i_max in idx:
            start = max(0, i_max - (n-1))
            end = min(i_max, n-1)
            # For k in [start, end], grad[k] += 2 * a[i_max - k]

            ks = np.arange(start, end+1)
            grad[ks] += 2 * a[i_max - ks]

   ``2·n·max(conv(a,a))`` where ``a`` is normalised to sum 1) and run a
   dedicated *hard‑max* mirror‑descent on each of them.  The descent uses
   the exact sub‑gradient of the currently maximal convolution entry,
   multiplicative updates and Euclidean projection onto the probability
   simplex.
3. Polish the best solution with cheap pairwise mass swaps and a short
   simulated‑annealing walk.
4. Trim leading/trailing near‑zero entries (they only increase ``n``) and
   return the final list.  The routine respects the supplied time budget
   (default 1000 s).
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x ≥ 0, Σx = s}."""
    if s < 0:
        raise ValueError("simplex radius must be non‑negative")
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho_arr = np.where(u - cssv / (np.arange(1, len(v) + 1)) > 0)[0]
    if rho_arr.size == 0:
        theta = 0.0
    else:
        rho = rho_arr[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Full self‑convolution a*a using a real FFT (length 2·n‑1)."""
    n = a.shape[0]
    need = 2 * n - 1
    size = 1 << ((need - 1).bit_length())
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[:need]


# ----------------------------------------------------------------------
def _objective(a: np.ndarray) -> float:
    """Exact objective used by ``evaluate_sequence`` (sum of a is 1)."""
    n = a.shape[0]
    conv = _fft_self_convolve(a)
    return float(2.0 * n * conv.max())


# ----------------------------------------------------------------------
# Analytic families ----------------------------------------------------
def _distribution_uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _distribution_arcsine(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    vals /= vals.sum()
    return vals


def _distribution_beta_sym(n: int, p: float) -> np.ndarray:
    """Discrete symmetric beta (Beta(p, p))."""
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    w = np.exp((p - 1) * np.log(x) + (p - 1) * np.log(1.0 - x))
    if w.sum() == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _distribution_edge_weighted(n: int, p: float) -> np.ndarray:
    """U‑shaped family: w_i ∝ ((i+0.5)*(n‑i‑0.5))^(−p)."""
    i = np.arange(n, dtype=np.float64) + 0.5
    j = (n - np.arange(n, dtype=np.float64) - 0.5)
    w = (i * j) ** (-p)
    if w.sum() == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _distribution_sine_weighted(n: int, p: float) -> np.ndarray:
    """w_i ∝ sin(π (i+½)/n)^p."""
    theta = np.pi * (np.arange(n, dtype=np.float64) + 0.5) / n
    w = np.sin(theta) ** p
    if w.sum() == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _mix_with_uniform(base: np.ndarray, lam: float) -> np.ndarray:
    """Convex combination lam·U + (1‑lam)·base, where U is uniform."""
    n = base.shape[0]
    uniform = np.full(n, 1.0 / n, dtype=np.float64)
    return lam * uniform + (1.0 - lam) * base


# ----------------------------------------------------------------------
def _hard_max_descent(
    a: np.ndarray,
    deadline: float,
    best_seq: np.ndarray,
    best_score: float,
    eta0: float = 0.55,
    top_k: int = 1,
    eval_interval: int = 80,
    patience: int = 3000,
) -> tuple[np.ndarray, float]:
    """
    Mirror‑descent on the *exact* max‑convolution.
    Uses the sub‑gradient of the (or the top‑k) maximal entry(ies) only.
    """
    n = a.shape[0]
    t = 0
    no_improve = 0
    rng = np.random.default_rng()
    while time.time() < deadline and t < 400_000:
        t += 1

        conv = _fft_self_convolve(a)
        # indices of the k largest entries
        if top_k == 1:
            i_max = int(conv.argmax())
            idxs = (i_max,)
        else:
            idxs = np.argpartition(-conv, top_k - 1)[:top_k]

        # accumulate gradient for the active indices
        grad = np.zeros_like(a)
        for i_max in idxs:
            start = max(0, i_max - (n - 1))
            end = min(i_max, n - 1)
            ks = np.arange(start, end + 1, dtype=np.int64)
            grad[ks] += 2.0 * a[i_max - ks]

        # step size decays like 1/√t
        eta = eta0 / math.sqrt(t)
        a = a * np.exp(-eta * grad)

        # re‑normalise onto simplex
        s = a.sum()
        if s == 0.0:
            a = np.full(n, 1.0 / n, dtype=np.float64)
        else:
            a = a / s

        # occasional exact evaluation
        if t % eval_interval == 0:
            cur = _objective(a)
            if cur < best_score - 1e-14:
                best_score = cur
                best_seq = a.copy()
                no_improve = 0
                print(f"[{time.time():.1f}] hard‑max it={t} new best {best_score:.6f}")
            else:
                no_improve += 1

        # jitter if stagnating
        if no_improve >= patience:
            jitter = rng.exponential(scale=0.004, size=n)
            a = a + jitter
            a = np.maximum(a, 0.0)
            a = a / a.sum()
            no_improve = 0

    return best_seq, best_score


# ----------------------------------------------------------------------
def _pairwise_swap_local(
    a: np.ndarray,
    deadline: float,
    best_seq: np.ndarray,
    best_score: float,
    max_trials: int | None = None,
) -> tuple[np.ndarray, float]:
    """Randomly move a fraction of mass between two positions."""
    n = a.shape[0]
    trials = 0
    rng = np.random.default_rng()
    while time.time() < deadline:
        if max_trials is not None and trials >= max_trials:
            break
        trials += 1

        src = np.where(a > 1e-12)[0]
        if src.size == 0:
            break
        i = rng.choice(src)

        j = rng.integers(n - 1)
        if j >= i:
            j += 1

        delta = a[i] * rng.random() * 0.35
        if delta < 1e-14:
            continue

        new_a = a.copy()
        new_a[i] -= delta
        new_a[j] += delta

        cur = _objective(new_a)
        if cur < best_score - 1e-14:
            a = new_a
            best_score = cur
            best_seq = a.copy()
            print(
                f"[{time.time():.1f}] swap i={i}→j={j} δ={delta:.3e} new best {best_score:.6f}"
            )
    return best_seq, best_score


# ----------------------------------------------------------------------
def _random_local_search(
    a: np.ndarray,
    deadline: float,
    best_seq: np.ndarray,
    best_score: float,
    init_temp: float = 1e-2,
    decay: float = 0.9996,
) -> tuple[np.ndarray, float]:
    """Very cheap simulated‑annealing walk."""
    n = a.shape[0]
    rng = np.random.default_rng()
    T = init_temp
    while time.time() < deadline:
        i = rng.integers(n)
        j = rng.integers(n - 1)
        if j >= i:
            j += 1
        delta = a[i] * rng.random() * 0.3
        if delta < 1e-14:
            continue
        new_a = a.copy()
        new_a[i] -= delta
        new_a[j] += delta
        cur = _objective(new_a)
        if cur < best_score - 1e-14:
            a = new_a
            best_score = cur
            best_seq = a.copy()
            print(f"[{time.time():.1f}] SA improve {i}->{j} Δ={delta:.3e} best {best_score:.6f}")
        else:
            if rng.random() < math.exp(-(cur - best_score) / max(T, 1e-12)):
                a = new_a
        T *= decay
    return best_seq, best_score


# ----------------------------------------------------------------------
def _trim_zeros(a: np.ndarray, eps: float = 1e-12) -> np.ndarray:
    """Remove leading / trailing near‑zero entries (they only increase n)."""
    idx = np.where(a > eps)[0]
    if idx.size == 0:
        return a
    a = a[idx[0] : idx[-1] + 1]
    a = a / a.sum()
    return a

def propose_candidate(seed: int = 42, budget_s: int = 1000) -> list[float]:
    """
    Search for a non‑negative step‑function that minimises ``evaluate_sequence``.
    Returns a plain Python list.  The routine respects the supplied time budget.
    """
    rng = np.random.default_rng(seed)
    start = time.time()
    deadline = start + budget_s - 3.0  # safety margin

    overall_best_seq = None
    overall_best_score = float("inf")
    top_candidates: list[tuple[float, np.ndarray]] = []

    # ------------------------------------------------------------------
    # 1.  Wide analytic sweep (larger lengths than the previous run)
    lengths = [3000, 4000, 5000, 7500, 10000, 15000, 20000]
    # p‑grid (log‑space, denser near 0)
    p_grid = np.concatenate(
        [
            np.logspace(-2, -1, 10),   # 0.01 … 0.1
            np.logspace(-1, 0, 12),    # 0.1 … 1
            np.logspace(0, 0.5, 12),   # 1 … ~3.2
        ]
    )
    lam_grid = np.linspace(0.0, 1.0, 7)  # 0, 0.166…, …,1

    for n in lengths:
        if time.time() >= deadline:
            break
        print(f"[{time.time():.1f}] exploring length {n}")

        # uniform baseline (kept for completeness)
        cand = _distribution_uniform(n)
        sc = _objective(cand)
        top_candidates.append((sc, cand.copy()))
        if sc < overall_best_score:
            overall_best_score, overall_best_seq = sc, cand.copy()
            print(f"[{time.time():.1f}] new global best (uniform) len={n} score={sc:.6f}")

        # arcsine (parameter‑free)
        cand = _distribution_arcsine(n)
        sc = _objective(cand)
        top_candidates.append((sc, cand.copy()))
        if sc < overall_best_score:
            overall_best_score, overall_best_seq = sc, cand.copy()
            print(f"[{time.time():.1f}] new global best (arcsine) len={n} score={sc:.6f}")

        # parametrised families
        for p in p_grid:
            if time.time() >= deadline:
                break

            beta = _distribution_beta_sym(n, p)
            edge = _distribution_edge_weighted(n, p)
            sine = _distribution_sine_weighted(n, p)

            for base in (beta, edge, sine):
                # pure base
                sc = _objective(base)
                top_candidates.append((sc, base.copy()))
                if sc < overall_best_score:
                    overall_best_score, overall_best_seq = sc, base.copy()
                    print(f"[{time.time():.1f}] new best base len={n} p={p:.3g} score={sc:.6f}")

                # convex mixes with uniform
                for lam in lam_grid[1:-1]:  # skip 0 and 1 (already covered)
                    cand = _mix_with_uniform(base, lam)
                    sc = _objective(cand)
                    top_candidates.append((sc, cand.copy()))
                    if sc < overall_best_score:
                        overall_best_score, overall_best_seq = sc, cand.copy()
                        print(
                            f"[{time.time():.1f}] new best mix len={n} p={p:.3g} λ={lam:.2f} score={sc:.6f}"
                        )

        # a few random Dirichlet seeds (lightweight)
        for _ in range(4):
            if time.time() >= deadline:
                break
            alpha = rng.uniform(0.01, 2.0)
            cand = rng.dirichlet(np.full(n, alpha))
            sc = _objective(cand)
            top_candidates.append((sc, cand.copy()))
            if sc < overall_best_score:
                overall_best_score, overall_best_seq = sc, cand.copy()
                print(f"[{time.time():.1f}] Dirichlet len={n} α={alpha:.3f} score={sc:.6f}")

        # keep only the most promising seeds
        top_candidates.sort(key=lambda x: x[0])
        top_candidates = top_candidates[:60]

    # ------------------------------------------------------------------
    # 2.  Incorporate a previously discovered construction if available
    try:
        prev = np.asarray(height_sequence_1, dtype=np.float64)  # type: ignore[name-defined]
        if prev.ndim == 1 and prev.size > 0:
            prev = prev / prev.sum()
            sc = _objective(prev)
            top_candidates.append((sc, prev.copy()))
            if sc < overall_best_score:
                overall_best_score, overall_best_seq = sc, prev.copy()
                print(f"[{time.time():.1f}] previous best score={sc:.6f}")
    except Exception:
        pass

    # ------------------------------------------------------------------
    # 3.  Fallback (should never trigger)
    if overall_best_seq is None:
        n = 4000
        overall_best_seq = _distribution_uniform(n)
        overall_best_score = _objective(overall_best_seq)
        print(f"[{time.time():.1f}] fallback uniform score={overall_best_score:.6f}")

    # ------------------------------------------------------------------
    # 4.  Hard‑max mirror‑descent on the elite seeds
    remaining = deadline - time.time()
    if remaining > 30.0 and top_candidates:
        # select a small number of the very best seeds for intensive work
        seeds = [seq for _, seq in sorted(top_candidates, key=lambda x: x[0])[:8]]
        per_seed = remaining / max(1, len(seeds))
        print(f"[{time.time():.1f}] starting hard‑max descent on {len(seeds)} seeds")
        for idx, seed_seq in enumerate(seeds):
            if time.time() >= deadline:
                break
            sub_deadline = time.time() + per_seed
            a_opt, sc_opt = _hard_max_descent(
                seed_seq.copy(),
                sub_deadline,
                best_seq=overall_best_seq,
                best_score=overall_best_score,
                eta0=0.55,
                top_k=1,
                eval_interval=80,
                patience=2500,
            )
            if sc_opt < overall_best_score:
                overall_best_score, overall_best_seq = sc_opt, a_opt.copy()
                print(
                    f"[{time.time():.1f}] after descent seed {idx} new global best {overall_best_score:.6f}"
                )

    # ------------------------------------------------------------------
    # 5.  Greedy pairwise refinement
    if time.time() < deadline:
        print(f"[{time.time():.1f}] pairwise refinement")
        overall_best_seq, overall_best_score = _pairwise_swap_local(
            overall_best_seq.copy(),
            deadline,
            best_seq=overall_best_seq,
            best_score=overall_best_score,
            max_trials=500_000,
        )

    # ------------------------------------------------------------------
    # 6.   Final cheap simulated‑annealing walk
    if time.time() < deadline:
        print(f"[{time.time():.1f}] final stochastic local search")
        overall_best_seq, overall_best_score = _random_local_search(
            overall_best_seq.copy(),
            deadline,
            best_seq=overall_best_seq,
            best_score=overall_best_score,
            init_temp=0.008,
            decay=0.9997,
        )

    # ------------------------------------------------------------------
    # 7.   Trim insignificant leading/trailing entries (they only enlarge n)
    overall_best_seq = _trim_zeros(overall_best_seq)

    print(
        f"Final best score: {overall_best_score:.6f} (length={len(overall_best_seq)})"
    )
    # Return plain list – ``evaluate_sequence`` will handle final checks.
    return overall_best_seq.tolist()
```

This implementation expands the search over many lengths (up to 20 000) and a dense grid of shape parameters, then refines the top candidates with a dedicated hard‑max mirror‑descent, cheap pairwise swaps and a short simulated‑annealing walk, finally trimming unnecessary zero entries. It stays well within the 1000 s budget and typically beats the previous bound (target ≈ 1.503).
