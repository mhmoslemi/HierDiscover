def _grad_smooth(a, lambda_):
    n = a.shape[0]
    conv = _fft_self_convolve(a) # length 2n-1

    # w = np.exp(lambda * (conv - max_conv))
    max_val = conv.max()
    w = np.exp(lambda_ * (conv - max_val))
    # compute correlation w ⋆ a
    size = 1 << ((len(w) + n - 1).bit_length())
    # compute FFT of w and a (reverse a)
    W = np.fft.rfft(w, n=size)
    A_rev = np.fft.rfft(a[::-1], n=size)
    corr_full = np.fft.irfft(W * A_rev, n=size)[: len(w) + n - 1]
    # correlation indices: shift from -(len(a)-1) to (len(w)-1)
    # We need values for shift k = 0..n-1, which correspond to index (len(a)-1 + k)
    start = n - 1
    grad = 2 * corr_full[start : start + n]
    return grad

"""
search for a step‑function (non‑negative sequence) that minimises
``evaluate_sequence`` by a two‑phase approach:

1️⃣  Generate a diverse pool of candidates from analytic families
    (uniform, edge‑weighted, beta, power‑sine, mixtures, random Dirichlet).

2️⃣  Refine the most promising candidates with a smooth‑max mirror–
    descent: we replace the non‑smooth max‑norm by a soft‑max
    (log‑sum‑exp) and perform an exponentiated‑gradient update.
    The soft‑max parameter λ is slowly increased so that the method
    initially moves quickly on a smooth landscape and finally behaves
    like a sub‑gradient method on the true L∞ objective.

3️⃣  Finish with a cheap greedy pairwise mass‑transfer hill‑climber.

The whole routine respects the provided time budget (default 1000 s)
and returns the best plain Python list it has found.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x ≥ 0, Σx = s}."""
    if s < 0:
        raise ValueError("Simplex radius must be non‑negative")
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.where(u - cssv / (np.arange(1, len(v) + 1)) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    return np.maximum(v - theta, 0.0)


# ----------------------------------------------------------------------
def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Full linear self‑convolution a*a using a real FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1) - 1).bit_length()   # next power of two
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[: 2 * n - 1]


# ----------------------------------------------------------------------
def _objective(a: np.ndarray) -> float:
    """
    Exact objective 2·n·max(conv(a,a)) for a vector with Σa = 1.
    (The safety‑clipping performed in ``evaluate_sequence`` is omitted,
    because we keep the vector non‑negative and normalised to sum 1.)
    """
    n = a.shape[0]
    b = _fft_self_convolve(a)
    return float(2.0 * n * b.max())


# ----------------------------------------------------------------------
# Analytic families ----------------------------------------------------
def _distribution_uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _distribution_edge_weighted(n: int, p: float) -> np.ndarray:
    """w_i ∝ (i+½)^{-p} · (n‑i‑½)^{-p}   (p>0)."""
    i = np.arange(n, dtype=np.float64) + 0.5
    j = (n - np.arange(n, dtype=np.float64) - 0.5)
    w = (i * j) ** (-p)
    w_sum = w.sum()
    if w_sum == 0.0:
        return _distribution_uniform(n)
    w /= w_sum
    return w


def _distribution_power_sine(n: int, alpha: float) -> np.ndarray:
    """w_i ∝ sin(π·(i+½)/n)^{α}, α>0."""
    i = np.arange(n, dtype=np.float64) + 0.5
    w = np.sin(np.pi * i / n) ** alpha
    w_sum = w.sum()
    if w_sum == 0.0:
        return _distribution_uniform(n)
    w /= w_sum
    return w


def _distribution_beta(n: int, alpha: float, beta: float) -> np.ndarray:
    """Beta(α,β) density evaluated at mid‑points."""
    i = np.arange(n, dtype=np.float64) + 0.5
    x = i / n                     # points in (0,1)
    # keep parameters >0 to avoid NaNs
    if alpha <= 0 or beta <= 0:
        return _distribution_uniform(n)
    w = (x ** (alpha - 1.0)) * ((1.0 - x) ** (beta - 1.0))
    w_sum = w.sum()
    if w_sum == 0.0:
        return _distribution_uniform(n)
    w /= w_sum
    return w


def _distribution_random_dirichlet(n: int, rng: np.random.Generator) -> np.ndarray:
    """Random Dirichlet with a random concentration parameter."""
    alpha = rng.uniform(0.01, 2.0)
    a = rng.dirichlet(np.full(n, alpha))
    return a.astype(np.float64)


def _mix_with_uniform(base: np.ndarray, lam: float) -> np.ndarray:
    """Convex combination λ·uniform + (1‑λ)·base."""
    n = base.shape[0]
    uniform = np.full(n, 1.0 / n, dtype=np.float64)
    return lam * uniform + (1.0 - lam) * base


# ----------------------------------------------------------------------
# Smoothed‑max gradient (soft‑max of conv) ---------------------------
def _softmax_gradient(a: np.ndarray, lam: float) -> np.ndarray:
    """
    Gradient of the smooth approximation
        g(a) = (1/lam)·log Σ_k exp(lam·conv_k(a))
    where conv = a * a (linear self‑convolution).
    Returns the (unnormalised) gradient ∇g.
    """
    n = a.shape[0]
    conv = _fft_self_convolve(a)
    max_val = conv.max()
    # Soft‑max weights (numerically stable)
    w = np.exp(lam * (conv - max_val))
    w_sum = w.sum()
    if w_sum == 0.0:
        w_sum = 1.0

    # correlation w ⋆ a  (equivalent to Σ_k w_k·2·a_{k‑j})
    size = 1 << ((len(w) + n - 1).bit_length())
    W = np.fft.rfft(w, n=size)
    A_rev = np.fft.rfft(a[::-1], n=size)
    corr_full = np.fft.irfft(W * A_rev, n=size)[: len(w) + n - 1]
    start = n - 1
    grad = 2.0 * corr_full[start:start + n] / w_sum
    return grad


# ----------------------------------------------------------------------
# Mirror‑descent using the smooth‑max ----------------------------------
def _smooth_exponential_descent(
    a0: np.ndarray,
    deadline: float,
    best_seq: np.ndarray,
    best_score: float,
    lam_start: float = 2.0,
    lam_max: float = 80.0,
    eta0: float = 0.8,
    eval_every: int = 200,
) -> tuple[np.ndarray, float]:
    """
    Exponentiated‑gradient descent on the smooth log‑sum‑exp surrogate.
    The temperature λ is slowly increased so that the optimisation
    gradually approaches the true L∞ objective.
    """
    a = a0.copy()
    n = a.shape[0]
    t = 0
    lam = lam_start

    while time.time() < deadline:
        t += 1

        # ------------ gradient of the smooth surrogate -----------------
        grad = _softmax_gradient(a, lam)

        # ------------ EG update + simplex projection ------------------
        eta = eta0 / math.sqrt(t)
        a *= np.exp(-eta * grad)
        s = a.sum()
        if s <= 0.0:
            a = np.full(n, 1.0 / n, dtype=np.float64)
        else:
            a /= s

        # ------------ anneal λ ---------------------------------------
        lam = min(lam_max, lam * 1.001)

        # ------------ occasional exact evaluation ---------------------
        if t % eval_every == 0:
            cur = _objective(a)           # exact objective (2·n·max conv)
            if cur < best_score - 1e-14:
                best_score = cur
                best_seq = a.copy()
                print(
                    f"[{time.time():.1f}] smooth it={t} λ={lam:.2f} new best {best_score:.6f}"
                )
    return best_seq, best_score


# ----------------------------------------------------------------------
# Greedy pairwise mass transfer ----------------------------------------
def _pairwise_swap(
    a0: np.ndarray,
    deadline: float,
    best_seq: np.ndarray,
    best_score: float,
    max_trials: int | None = None,
) -> tuple[np.ndarray, float]:
    """
    Randomly move a fraction of mass from one index to another.
    The move is accepted only if it strictly improves the exact objective.
    """
    a = a0.copy()
    n = a.shape[0]
    trials = 0
    while time.time() < deadline:
        if max_trials is not None and trials >= max_trials:
            break
        trials += 1

        src = np.where(a > 1e-14)[0]
        if src.size == 0:
            break
        i = np.random.choice(src)
        j = np.random.randint(n - 1)
        if j >= i:
            j += 1

        delta = a[i] * np.random.rand() * 0.45   # up to 45 % of source mass
        if delta < 1e-15:
            continue

        new_a = a.copy()
        new_a[i] -= delta
        new_a[j] += delta

        cur = _objective(new_a)
        if cur < best_score - 1e-14:
            a = new_a
            best_score = cur
            best_seq = a.copy()
            print(
                f"[{time.time():.1f}] swap i={i}→j={j} δ={delta:.6g} new best {best_score:.6f}"
            )
    return best_seq, best_score

            )
            if sc_opt < overall_best_score - 1e-12:
                overall_best_score, overall_best_seq = sc_opt, a_opt.copy()
                print(
                    f"[{time.time():.1f}] after smooth seed {idx} new global best {overall_best_score:.6f}"
                )

    # ------------------------------------------------------------------
    # 5.  Final greedy pairwise polishing (hill‑climbing)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        print(f"[{time.time():.1f}] final greedy swap polishing")
        a_opt, sc_opt = _pairwise_swap(
            overall_best_seq.copy(),
            deadline,
            best_seq=overall_best_seq,
            best_score=overall_best_score,
        )
        if sc_opt < overall_best_score - 1e-12:
            overall_best_score, overall_best_seq = sc_opt, a_opt.copy()
            print(f"[{time.time():.1f}] after final swaps new best {overall_best_score:.6f}")

    # ------------------------------------------------------------------
    print(
        f"Final best objective (2·n·max(conv)) = {overall_best_score:.6f} (length={len(overall_best_seq)})"
    )
    # Return a plain Python list; evaluate_sequence will clip/check internally.
    return overall_best_seq.tolist()
