import numpy as np
import scipy.sparse as sp
from scipy.optimize import linprog
import time
import math

def propose_candidate(seed=12345, budget_s=1000):
    rng = np.random.default_rng(seed)
    start = time.time()
    deadline = start + budget_s - 2.0

    # helper functions defined earlier

    # generate seeds
    lengths = [1500, 2000, 2500]   # maybe also 3000 if time
    candidate_seqs = []
    best_score = float('inf')
    best_seq = None

    # define some families
    for n in lengths:
        if time.time() >= deadline:
            break
        # uniform
        uniform = np.full(n, 1.0/n)
        sc = _objective(uniform)
        candidate_seqs.append((sc, uniform.copy()))
        if sc < best_score:
            best_score, best_seq = sc, uniform.copy()
            print(f"New best uniform n={n} score={sc}")

        # arcsine
        arcsine = _distribution_arcsine(n)
        sc = _objective(arcsine)
        candidate_seqs.append((sc, arcsine.copy()))
        if sc < best_score:
            best_score, best_seq = sc, arcsine.copy()
            print(f"New best arcsine n={n} score={sc}")

        # symmetric beta p values
        for p in [0.2, 0.5, 1.0, 1.5, 2.0]:
            if time.time() >= deadline:
                break
            beta = _distribution_beta_sym(n, p)
            sc = _objective(beta)
            candidate_seqs.append((sc, beta.copy()))
            if sc < best_score:
                best_score, best_seq = sc, beta.copy()
                print(f"New best beta_sym p={p} n={n} score={sc}")

        # edge weighted
        for p in [0.5, 1.0, 2.0]:
            if time.time() >= deadline:
                break
            edge = _distribution_edge_weighted(n, p)
            sc = _objective(edge)
            candidate_seqs.append((sc, edge.copy()))
            if sc < best_score:
                best_score, best_seq = sc, edge.copy()
                print(f"New best edge p={p} n={n} score={sc}")

        # random Dirichlet
        for _ in range(3):
            if time.time() >= deadline:
                break
            alpha = rng.uniform(0.05, 1.5)
            dir_vec = rng.dirichlet(np.full(n, alpha))
            sc = _objective(dir_vec)
            candidate_seqs.append((sc, dir_vec.copy()))
            if sc < best_score:
                best_score, best_seq = sc, dir_vec.copy()
                print(f"New best Dirichlet alpha={alpha:.3g} n={n} score={sc}")

    # keep top seeds for further refinement
    candidate_seqs.sort(key=lambda x: x[0])
    top_seqs = [seq for _, seq in candidate_seqs[:5]]

    # main alternating LP refinement
    for idx, seq in enumerate(top_seqs):
        if time.time() >= deadline:
            break
        print(f"--- Alternating LP refinement on seed {idx} (len={len(seq)}) ---")
        seq_opt, score_opt = _alternating_lp_opt(seq, deadline, max_iter=12, t_mix=0.5)
        if score_opt < best_score:
            best_score, best_seq = score_opt, seq_opt.copy()
            print(f"New global best after LP seed {idx} score={best_score}")

    # final fine tuning with projected gradient descent (optional)
    if time.time() < deadline:
        remain = deadline - time.time()
        # allocate some seconds per call
        sub_deadline = time.time() + min(30.0, remain)
        best_seq, _ = _proj_grad_descent(best_seq.copy(), sub_deadline, best_seq.copy(), best_score)
        # Evaluate after gradient descent
        best_score = _objective(best_seq)

    print(f"Final best score {best_score} length {len(best_seq)}")
    return best_seq.tolist()

"""search for a step‑function that minimises ``evaluate_sequence``.

The approach is a two‑stage optimisation.

Stage 1 – cheap analytic seeds
    • generate uniform, arcsine, symmetric‑Beta, edge‑weighted and a few
      Dirichlet vectors for several lengths (≈1.5k–2.5k);
    • keep the ~5 best (lowest 2·n·max(conv)) as starting points.

Stage 2 – alternating LP + local refinement
    • For a seed ``f`` (∑f=1) compute M = max(f∗f).
    • Build the convolution matrix C₍f₎ (size (2n−1)×n) such that
      C₍f₎·g = f∗g.  (C₍f₎ is very sparse – a collection of n diagonals.)
    • Solve the linear programme
          maximise   1ᵀg
          subject to C₍f₎·g ≤ M·1 ,   g ≥ 0 .
      The solution pushes as much mass as possible without increasing the
      current autocorrelation bound.
    • Normalise ``g`` to unit sum and replace f by a convex mixture
          f ← (1‑t)·f + t·g   (t≈0.5, sum stays 1).
    • Iterate the LP‑mix step a dozen times (or until the time budget is
      exhausted).  The product 2·n·max(f∗f) only decreases.

Stage 3 – final fine‑tuning
    • A short projected sub‑gradient descent (exact sub‑gradient of the
      maximal convolution entry) cleans up remaining local artefacts.

All heavy work respects the global ``budget_s`` (default 1000 s).  The
returned list contains non‑negative numbers with sum≈1; the caller’s
``evaluate_sequence`` will apply the final clipping and scaling."""
import time
import math
import numpy as np
import scipy.sparse as sp
from scipy.optimize import linprog

# ----------------------------------------------------------------------
# FFT based self‑convolution (full length 2n‑1)
def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    n = a.shape[0]
    need = 2 * n - 1
    size = 1 << ((need - 1).bit_length())
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[:need]

# ----------------------------------------------------------------------
# Objective for a probability vector (sum = 1)
def _objective(a: np.ndarray) -> float:
    n = a.shape[0]
    b = _fft_self_convolve(a)
    return 2.0 * n * b.max()

# ----------------------------------------------------------------------
# Analytic families (all return normalised vectors, i.e. sum = 1)
def _distribution_uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)

def _distribution_arcsine(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    vals /= vals.sum()
    return vals

def _distribution_beta_sym(n: int, p: float) -> np.ndarray:
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    w = np.exp((p - 1) * np.log(x) + (p - 1) * np.log(1.0 - x))
    if w.sum() == 0.0:
        return np.full(n, 1.0 / n, dtype=np.float64)
    w /= w.sum()
    return w

def _distribution_edge_weighted(n: int, p: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64) + 0.5
    j = (n - np.arange(n, dtype=np.float64) - 0.5)
    w = (i * j) ** (-p)
    if w.sum() == 0.0:
        return np.full(n, 1.0 / n, dtype=np.float64)
    w /= w.sum()
    return w

# ----------------------------------------------------------------------
# Build the (2n‑1)×n convolution matrix C₍f₎  (C₍f₎·g = f∗g)
def _conv_constraint_matrix(f: np.ndarray) -> sp.csc_matrix:
    n = f.shape[0]
    # each diagonal (negative offset) contains the same coefficient f_i
    diags = [np.full(n, f_i, dtype=np.float64) for f_i in f]
    offsets = -np.arange(n, dtype=np.int64)    # 0, -1, -2, …, -(n‑1)
    return sp.diags(diags, offsets, shape=(2 * n - 1, n), format="csc")

# ----------------------------------------------------------------------
# Alternating LP optimisation (block‑coordinate descent)
def _alternating_lp_opt(
    init_seq: np.ndarray,
    deadline: float,
    max_iter: int = 12,
    mix_weight: float = 0.5,
) -> tuple[np.ndarray, float]:
    """
    Starting from ``init_seq`` (sum = 1) run a few LP‑mix iterations.
    Returns the best sequence encountered and its objective value.
    """
    f = init_seq.copy()
    best_seq = f.copy()
    best_score = _objective(f)

    n = f.shape[0]
    it = 0
    while time.time() < deadline and it < max_iter:
        it += 1
        # 1) current maximum autocorrelation
        conv = _fft_self_convolve(f)
        M = conv.max()

        # 2) build linear constraints C_f·g ≤ M
        A = _conv_constraint_matrix(f)
        b = np.full(A.shape[0], M, dtype=np.float64)

        # 3) solve LP « maximise 1ᵀg »
        res = linprog(
            c=-np.ones(n, dtype=np.float64),      # maximise sum(g)
            A_ub=A,
            b_ub=b,
            bounds=(0, None),
            method="highs",
            options={"presolve": True},
        )
        if not res.success:
            # LP failed – abort the alternating loop
            print(f"[{time.time():.1f}] LP iteration {it} failed, stop.")
            break

        g = res.x
        s = g.sum()
        if s == 0.0:
            break
        g_norm = g / s               # enforce Σg = 1

        # 4) convex combination (mix)
        new_f = (1.0 - mix_weight) * f + mix_weight * g_norm

        # 5) evaluate
        new_score = _objective(new_f)
        if new_score < best_score - 1e-14:
            best_score = new_score
            best_seq = new_f.copy()
            print(
                f"[{time.time():.1f}] LP‑mix it {it} new best {best_score:.6f}"
            )

        # prepare next iteration
        f = new_f

    return best_seq, best_score

# ----------------------------------------------------------------------
# Euclidean projection onto the simplex {x≥0, Σx = 1}
def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Euclidean projection of ``v`` onto the simplex of radius ``s``."""
    if s < 0:
        raise ValueError("simplex radius must be non‑negative")
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho_arr = np.where(u - cssv / (np.arange(1, len(v) + 1)) > 0)[0]
    if rho_arr.size == 0:
        theta = 0.0
    else:
        rho = rho_arr[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Projected sub‑gradient descent (fine‑tuning)
def _proj_grad_descent(
    a: np.ndarray,
    deadline: float,
    best_seq: np.ndarray,
    best_score: float,
    eta0: float = 0.6,
    jitter_interval: int = 2500,
) -> tuple[np.ndarray, float]:
    """Minimise 2·n·max(conv(a,a)) with exact sub‑gradient and simplex projection."""
    n = a.shape[0]
    t = 0
    no_improve = 0
    rng = np.random.default_rng()
    cur_score = _objective(a)

    while time.time() < deadline and t < 200_000:
        t += 1
        conv = _fft_self_convolve(a)
        max_val = conv.max()
        cur_score = 2.0 * n * max_val

        # update global best
        if cur_score < best_score - 1e-14:
            best_score = cur_score
            best_seq = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        # sub‑gradient of the maximal entry (sum if multiple maxima)
        ks = np.where(conv == max_val)[0]
        grad = np.zeros_like(a)
        rev_a = a[::-1]
        for k in ks:
            offset = n - 1 - k
            if offset >= 0:
                i_start = 0
                i_end = n - offset
                grad[i_start:i_end] += 2.0 * rev_a[offset:offset + i_end - i_start]
            else:
                off = -offset
                i_start = off
                i_end = n
                grad[i_start:i_end] += 2.0 * rev_a[0:i_end - i_start]

        # step with diminishing learning rate and projection
        eta = eta0 / math.sqrt(t)
        a_new = a - eta * grad
        a_new = _project_onto_simplex(a_new, s=1.0)

        # jitter if we are stuck
        if no_improve >= jitter_interval:
            jitter = rng.exponential(scale=0.003, size=n)
            a_new = a_new + jitter
            a_new = _project_onto_simplex(a_new, s=1.0)
            no_improve = 0

        a = a_new

    return best_seq, best_score
