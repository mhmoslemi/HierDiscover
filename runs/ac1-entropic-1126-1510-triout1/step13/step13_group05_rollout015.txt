def _max_subgrad_fft(a):
    n = len(a)
    conv = _fft_self_convolve(a)
    max_val = conv.max()
    tol = 1e-12 * max_val
    mask = (conv >= max_val - tol).astype(np.float64)  # length 2n-1
    # Compute correlation via FFT
    rev_mask = mask[::-1]
    # Use real FFT of appropriate size
    size = 1 << ((n + (2*n-1) - 1).bit_length())
    A = np.fft.rfft(a, n=size)
    M = np.fft.rfft(rev_mask, n=size)
    corr = np.fft.irfft(A * M, n=size)
    # corr length = size
    # slice needed indices: we need elements at index len(mask)-1 - np.arange(n) = (2n-2) - np.arange(n)
    offset = len(mask) - 1
    grad = 2.0 * corr[offset - np.arange(n)]
    # Normalize if mask sum>0:
    sumw = mask.sum()
    if sumw > 0:
        grad /= sumw
    return grad

def _simulated_annealing(a0, time_limit, best_seq, best_score, init_temp=0.01, final_temp=1e-6):

    a = a0.copy()
    n = a.shape[0]
    cur_score = _objective(a)
    best_local = cur_score
    best_local_seq = a.copy()
    T = init_temp
    # temperature schedule: exponential decay based on remaining time?
    # We'll compute slowdown factor: ratio = final_temp/init_temp, steps = maybe 10000, let T = init_temp * (ratio) ** (elapsed/total)
    start = time.time()
    while time.time() < time_limit:
        # random move
        i = np.random.choice(np.where(a > 1e-12)[0])
        j = np.random.randint(n)
        if j == i: continue
        delta = a[i] * np.random.rand() * 0.3  # up to 30% of mass
        if delta < 1e-15: continue
        a_new = a.copy()
        a_new[i] -= delta
        a_new[j] += delta
        new_score = _objective(a_new)
        diff = new_score - cur_score
        # compute temperature based on elapsed proportion
        elapsed = time.time() - start
        total = time_limit - start
        if total <= 0:
            T = final_temp
        else:
            T = init_temp * ((final_temp / init_temp) ** (elapsed / total))
        if diff < 0 or np.random.rand() < np.exp(-diff / T):
            a = a_new
            cur_score = new_score
            if cur_score < best_score - 1e-12:
                best_score = cur_score
                best_seq = a.copy()
                print(f"[{time.time():.1f}] SA new global best {best_score:.6f}")
            if cur_score < best_local:
                best_local = cur_score
                best_local_seq = a.copy()
        # optionally check temperature schedule termination
    return best_seq, best_score

"""
propose_candidate – global optimiser for the step‑function problem.

The routine works in three phases inside the allotted wall‑time:

1. **Broad family sweep** – generate many analytic / random candidates on a
   handful of lengths (≈1 k–20 k).  The candidates are normalised to sum = 1
   and scored by the exact quantity `2·n·max(conv)`.  The ~30 best are kept.

2. **Exponential‑gradient refinement** – each retained seed is processed with
   mirror‑descent on the *exact* sub‑gradient of the max‑convolution.
   The sub‑gradient is obtained by a single FFT‑based correlation, so millions
   of updates can be performed within the time‑budget.

3. **Escaping local minima** – a fast simulated‑annealing hill‑climber
   (mass moves between coordinates, Metropolis acceptance) runs from the
   current global best, followed by a strict greedy pair‑swap polish.

All helpers are defined at top level, no closures or lambdas are used.
The returned list is already non‑negative; scaling does not affect the
objective, therefore we keep the vector normalised for numerical stability.
"""

# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Euclidean projection of ``v`` onto the simplex {x≥0, Σx=s}."""
    if s < 0:
        raise ValueError("Simplex radius must be non‑negative")
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.where(u - cssv / (np.arange(1, len(v) + 1)) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    return np.maximum(v - theta, 0.0)


# ----------------------------------------------------------------------
def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Full (linear) self‑convolution a*a using an FFT of power‑of‑2 length."""
    n = a.shape[0]
    size = 1 << ((2 * n - 2).bit_length())   # next power of two ≥2·n‑1
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[: 2 * n - 1]


# ----------------------------------------------------------------------
def _objective(a: np.ndarray) -> float:
    """Exact objective 2·n·max(conv) for a probability vector Σa=1."""
    n = a.shape[0]
    b = _fft_self_convolve(a)
    return float(2.0 * n * b.max())


# ----------------------------------------------------------------------
# ----- analytic families -------------------------------------------------
def _distribution_uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _distribution_arcsine(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    vals /= vals.sum()
    return vals


def _distribution_sine(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64) + 0.5
    w = np.sin(np.pi * i / n)
    w /= w.sum()
    return w


def _distribution_cosine(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64) + 0.5
    w = 1.0 - np.cos(np.pi * i / n)
    w /= w.sum()
    return w


def _distribution_quadratic(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    x = (2.0 * i) / (n - 1) - 1.0
    w = 1.0 - x * x
    w = np.clip(w, 0.0, None)
    w /= w.sum()
    return w


def _distribution_beta_sym(n: int, p: float) -> np.ndarray:
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    w = np.exp((p - 1) * np.log(x) + (p - 1) * np.log(1.0 - x))
    if w.sum() == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _distribution_edge_weighted(n: int, p: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64) + 0.5
    j = (n - np.arange(n, dtype=np.float64) - 0.5)
    w = (i * j) ** (-p)
    w = np.clip(w, 0.0, None)
    total = w.sum()
    if total > 0:
        w /= total
    else:
        w = np.full(n, 1.0 / n)
    return w


def _distribution_power_sum(n: int, p: float) -> np.ndarray:
    i = np.arange(1, n + 1, dtype=np.float64)
    w = i ** (-p) + (n + 1 - i) ** (-p)
    w = np.clip(w, 0.0, None)
    w /= w.sum()
    return w


def _distribution_exponential(n: int, lam: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    w = np.exp(-lam * i) + np.exp(-lam * (n - 1 - i))
    w = np.clip(w, 0.0, None)
    w /= w.sum()
    return w


def _distribution_two_blocks(n: int, frac: float) -> np.ndarray:
    """Two equal‑mass blocks placed at the two ends, length = frac·n each."""
    blk = max(1, int(frac * n))
    w = np.zeros(n, dtype=np.float64)
    w[:blk] = 0.5 / blk
    w[-blk:] = 0.5 / blk
    return w


def _mix_with_uniform(base: np.ndarray, lam: float) -> np.ndarray:
    """Convex combination λ·uniform + (1‑λ)·base."""
    n = base.shape[0]
    uniform = np.full(n, 1.0 / n, dtype=np.float64)
    return lam * uniform + (1.0 - lam) * base


# ----------------------------------------------------------------------
# ----- sub‑gradient of max‑convolution (FFT‑based) -----------------------
def _max_subgradient_fft(a: np.ndarray) -> np.ndarray:
    """
    Sub‑gradient of max_k Σ_i a_i a_{k‑i}.
    Implemented via correlation a ★ active‑mask, where the mask marks all
    numerically maximal convolution entries.
    """
    n = a.shape[0]
    conv = _fft_self_convolve(a)
    max_val = conv.max()
    tol = 1e-12 * max_val
    active = (conv >= max_val - tol).astype(np.float64)      # length 2·n‑1
    cnt = active.sum()
    if cnt == 0:
        return np.zeros_like(a)

    # correlation a ★ active  (non‑circular)
    rev = active[::-1]
    size = 1 << ((3 * n - 2 - 1).bit_length())   # ≥ n+(2n‑1)-1 = 3n‑2
    A = np.fft.rfft(a, n=size)
    R = np.fft.rfft(rev, n=size)
    corr = np.fft.irfft(A * R, n=size)           # linear correlation
    offset = (2 * n - 2)                         # index of active mask centre
    grad = 2.0 * corr[offset - np.arange(n)] / cnt
    return grad


# ----------------------------------------------------------------------
# ----- exponentiated‑gradient descent (mirror descent) -------------------
def _exp_gradient_descent_fft(a0: np.ndarray,
                               time_limit: float,
                               best_seq: np.ndarray,
                               best_score: float,
                               eta0: float = 0.8,
                               max_iter: int = 500_000,
                               report_every: int = 2000) -> tuple[np.ndarray, float]:
    """Mirror‑descent on exact max‑convolution objective using the FFT sub‑grad."""
    a = a0.copy()
    n = a.shape[0]
    t = 0
    while time.time() < time_limit and t < max_iter:
        t += 1
        grad = _max_subgradient_fft(a)
        eta = eta0 / math.sqrt(t)
        a = a * np.exp(-eta * grad)
        # projection onto the simplex (normalisation)
        a_sum = a.sum()
        if a_sum > 0:
            a = a / a_sum
        else:
            a = np.full(n, 1.0 / n, dtype=np.float64)

        if t % report_every == 0:
            cur = _objective(a)
            if cur < best_score - 1e-14:
                best_score = cur
                best_seq = a.copy()
                print(f"[{time.time():.1f}] exp‑gd it={t} new global best {best_score:.6f}")
    return best_seq, best_score


# ----------------------------------------------------------------------
# ----- simulated‑annealing hill‑climber ---------------------------------
def _simulated_annealing(a0: np.ndarray,
                         time_limit: float,
                         best_seq: np.ndarray,
                         best_score: float,
                         init_temp: float = 0.02,
                         final_temp: float = 1e-6) -> tuple[np.ndarray, float]:
    """
    Metropolis hill‑climbing: move a fraction of mass between two positions.
    Accepts worsening moves with probability exp(-Δ/T). Temperature decays
    geometrically from ``init_temp`` to ``final_temp`` over the allotted time.
    """
    a = a0.copy()
    n = a.shape[0]
    cur_score = _objective(a)

    start = time.time()
    while time.time() < time_limit:
        # linear temperature schedule
        elapsed = time.time() - start
        total = time_limit - start
        if total <= 0:
            T = final_temp
        else:
            T = init_temp * ((final_temp / init_temp) ** (elapsed / total))

        # select source with non‑negligible mass
        src_candidates = np.where(a > 1e-14)[0]
        if src_candidates.size == 0:
            break
        i = np.random.choice(src_candidates)
        j = np.random.randint(n - 1)
        if j >= i:
            j += 1

        # move at most 40 % of the source mass
        delta = a[i] * np.random.rand() * 0.4
        if delta < 1e-15:
            continue

        a_new = a.copy()
        a_new[i] -= delta
        a_new[j] += delta

        new_score = _objective(a_new)
        diff = new_score - cur_score

        if diff < 0 or np.random.rand() < math.exp(-diff / max(T, 1e-16)):
            a = a_new
            cur_score = new_score
            if cur_score < best_score - 1e-14:
                best_score = cur_score
                best_seq = a.copy()
                print(f"[{time.time():.1f}] SA new global best {best_score:.6f}")
    return best_seq, best_score


# ----------------------------------------------------------------------
# ----- greedy pairwise polishing ----------------------------------------
def _pairwise_swap(a0: np.ndarray,
                   time_limit: float,
                   best_seq: np.ndarray,
                   best_score: float,
                   max_trials: int | None = None) -> tuple[np.ndarray, float]:
    """Strictly improving random mass transfers."""
    a = a0.copy()
    n = a.shape[0]
    trials = 0
    while time.time() < time_limit:
        if max_trials is not None and trials >= max_trials:
            break
        trials += 1

        src = np.where(a > 1e-14)[0]
        if src.size == 0:
            break
        i = np.random.choice(src)
        j = np.random.randint(n - 1)
        if j >= i:
            j += 1

        delta = a[i] * np.random.rand() * 0.45
        if delta < 1e-15:
            continue

        new_a = a.copy()
        new_a[i] -= delta
        new_a[j] += delta

        cur = _objective(new_a)
        if cur < best_score - 1e-14:
            a = new_a
            best_score = cur
            best_seq = a.copy()
            print(f"[{time.time():.1f}] swap i={i}→j={j} δ={delta:.3g} new best {best_score:.6f}")
    return best_seq, best_score


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: int = 1000) -> list[float]:
    """
    Search for a non‑negative step‑function minimising ``evaluate_sequence``.
    The routine respects the supplied time budget and returns a plain list.
    """
    rng = np.random.default_rng(seed)
    start = time.time()
    deadline = start + budget_s - 2.0          # safety margin

    overall_best_seq: np.ndarray | None = None
    overall_best_score = float("inf")
    top_candidates: list[tuple[float, np.ndarray]] = []   # (score, seq)

    # ------------------------------------------------------------------
    # 1.  Scan a diverse pool of families on a handful of lengths.
    # ------------------------------------------------------------------
    lengths = [1024, 2048, 4096, 8192, 12000, 20000]
    # parameter grids
    p_grid = np.concatenate(
        [
            np.logspace(-2, 0.5, 12),       # 0.01 … 3.16
            np.linspace(0.5, 5.0, 12),      # 0.5 … 5
            np.linspace(5.0, 10.0, 8),      # 5 … 10
        ]
    )
    lam_grid = np.linspace(0.0, 1.0, 5)       # mix with uniform
    expo_grid = np.logspace(-2, 1, 10)       # λ for exponential family
    block_fracs = np.linspace(0.02, 0.30, 8)  # length of the edge blocks

    for n in lengths:
        if time.time() >= deadline:
            break
        print(f"[{time.time():.1f}] scanning length {n}")

        # deterministic analytic families
        for base_fn in (
            _distribution_uniform,
            _distribution_arcsine,
            _distribution_sine,
            _distribution_cosine,
            _distribution_quadratic,
        ):
            cand = base_fn(n)
            sc = _objective(cand)
            top_candidates.append((sc, cand.copy()))
            if sc < overall_best_score:
                overall_best_score = sc
                overall_best_seq = cand.copy()
                print(f"[{time.time():.1f}] new global best len={n} score={sc:.6f}")

        # parametric families – edge weighted, beta, power‑sum
        for p in p_grid:
            if time.time() >= deadline:
                break

            edge = _distribution_edge_weighted(n, p)
            beta = _distribution_beta_sym(n, p)
            power = _distribution_power_sum(n, p)

            for base in (edge, beta, power):
                sc = _objective(base)
                top_candidates.append((sc, base.copy()))
                if sc < overall_best_score:
                    overall_best_score = sc
                    overall_best_seq = base.copy()
                    print(f"[{time.time():.1f}] len={n} p={p:.3g} score={sc:.6f}")

                for lam in lam_grid:
                    if time.time() >= deadline:
                        break
                    mixed = _mix_with_uniform(base, lam)
                    sc = _objective(mixed)
                    top_candidates.append((sc, mixed.copy()))
                    if sc < overall_best_score:
                        overall_best_score = sc
                        overall_best_seq = mixed.copy()
                        print(f"[{time.time():.1f}] len={n} p={p:.3g} lam={lam:.2f} score={sc:.6f}")

        # exponential family
        for lam in expo_grid:
            if time.time() >= deadline:
                break
            exp_seq = _distribution_exponential(n, lam)
            sc = _objective(exp_seq)
            top_candidates.append((sc, exp_seq.copy()))
            if sc < overall_best_score:
                overall_best_score = sc
                overall_best_seq = exp_seq.copy()
                print(f"[{time.time():.1f}] exponential lam={lam:.3g} score={sc:.6f}")

        # two‑block edge family
        for frac in block_fracs:
            if time.time() >= deadline:
                break
            tb = _distribution_two_blocks(n, frac)
            sc = _objective(tb)
            top_candidates.append((sc, tb.copy()))
            if sc < overall_best_score:
                overall_best_score = sc
                overall_best_seq = tb.copy()
                print(f"[{time.time():.1f}] two‑block frac={frac:.3f} score={sc:.6f}")

        # random Dirichlet seeds
        for _ in range(6):
            if time.time() >= deadline:
                break
            alpha = rng.uniform(0.01, 2.0)
            a = rng.dirichlet(np.full(n, alpha))
            sc = _objective(a)
            top_candidates.append((sc, a.copy()))
            if sc < overall_best_score:
                overall_best_score = sc
                overall_best_seq = a.copy()
                print(f"[{time.time():.1f}] Dirichlet α={alpha:.3g} score={sc:.6f}")

        # sparse random seeds
        for _ in range(4):
            if time.time() >= deadline:
                break
            a = _distribution_random_sparse(n, rng, density=0.004)
            sc = _objective(a)
            top_candidates.append((sc, a.copy()))
            if sc < overall_best_score:
                overall_best_score = sc
                overall_best_seq = a.copy()
                print(f"[{time.time():.1f}] sparse score={sc:.6f}")

        # keep only the most promising ~30 candidates overall
        top_candidates.sort(key=lambda x: x[0])
        top_candidates = top_candidates[:30]

    # ------------------------------------------------------------------
    # 2.  Incorporate a previously discovered construction, if present.
    # ------------------------------------------------------------------
    try:
        prev = np.asarray(height_sequence_1, dtype=np.float64)  # type: ignore[name-defined]
        if prev.ndim == 1 and prev.size > 0:
            prev = np.maximum(prev, 0.0)
            total = prev.sum()
            if total == 0.0:
                prev = _distribution_uniform(prev.size)
            else:
                prev = prev / total
            sc = _objective(prev)
            top_candidates.append((sc, prev.copy()))
            if sc < overall_best_score:
                overall_best_score = sc
                overall_best_seq = prev.copy()
                print(f"[{time.time():.1f}] previous construction score={sc:.6f}")
    except Exception:
        pass

    # ------------------------------------------------------------------
    # 3.  Fallback (should never be needed).
    # ------------------------------------------------------------------
    if overall_best_seq is None:
        n = 5000
        overall_best_seq = _distribution_uniform(n)
        overall_best_score = _objective(overall_best_seq)
        print(f"[{time.time():.1f}] fallback uniform len={n} score={overall_best_score:.6f}")

    # ------------------------------------------------------------------
    # 4.  Refine the surviving seeds with exact mirror‑descent.
    # ------------------------------------------------------------------
    remaining = deadline - time.time()
    if remaining > 40.0 and top_candidates:
        # pick the five best seeds for deep optimisation
        seeds = [seq for _, seq in sorted(top_candidates, key=lambda x: x[0])[:5]]
        per_seed = remaining / max(1, len(seeds))
        print(f"[{time.time():.1f}] start exact exp‑gd on {len(seeds)} seeds")
        for idx, seed_seq in enumerate(seeds):
            if time.time() >= deadline:
                break
            sg_deadline = time.time() + per_seed
            a_opt, sc_opt = _exp_gradient_descent_fft(
                seed_seq,
                sg_deadline,
                best_seq=overall_best_seq,
                best_score=overall_best_score,
                eta0=0.9,
                report_every=1500,
            )
            if sc_opt < overall_best_score - 1e-12:
                overall_best_score = sc_opt
                overall_best_seq = a_opt.copy()
                print(f"[{time.time():.1f}] after exp‑gd seed {idx} new global best {overall_best_score:.6f}")

    # ------------------------------------------------------------------
    # 5.  Simulated‑annealing escape from potential local minima.
    # ------------------------------------------------------------------
    if time.time() < deadline:
        print(f"[{time.time():.1f}] simulated annealing phase")
        overall_best_seq, overall_best_score = _simulated_annealing(
            overall_best_seq.copy(),
            deadline,
            best_seq=overall_best_seq,
            best_score=overall_best_score,
        )

    # ------------------------------------------------------------------
    # 6.  Final greedy polishing.
    # ------------------------------------------------------------------
    if time.time() < deadline:
        print(f"[{time.time():.1f}] final greedy polishing")
        overall_best_seq, overall_best_score = _pairwise_swap(
            overall_best_seq.copy(),
            deadline,
            best_seq=overall_best_seq,
            best_score=overall_best_score,
        )

    print(
        f"Final best score (2·n·max(conv)): {overall_best_score:.6f} (length={len(overall_best_seq)})"
    )
    # Return as plain list – evaluate_sequence will handle clipping / scaling.
    return overall_best_seq.tolist()
