"""
step_function_search_v2.py

A refined 1000‑second search for a non‑negative step function that
minimises the objective used by ``evaluate_sequence``:

    2·n·max(conv(a,a)) / (∑a)²

The algorithm extends the previous version with three main ideas:

1. **Richer seed families** – we also sample symmetric Beta(α,α)
   distributions (including very U‑shaped α<1) and use a finer grid for
   the edge‑weighted family.  All seeds are normalised to a probability
   vector.

2. **Smoothed exponential‑gradient descent** – instead of the raw
   sub‑gradient of the maximal convolution entry we minimise a
   smooth approximation
       O_λ(a) = (1/λ)·log‑sum‑exp(λ·conv(a,a))
   whose gradient can be computed with two FFT‑based convolutions.
   A multiplicative (mirror‑descent) update keeps the iterates on the
   simplex automatically.  λ is increased gradually so the algorithm
   first smooths the landscape and then focuses on the true maximum.

3. **Longer polishing & adaptive schedule** – after a short GD pass on
   many diverse seeds we devote most of the remaining budget to the
   smoothed GD on the current best, optionally up‑sample the vector,
   polish it again and finish with a pairwise annealing stage.

All work respects the 1000 s budget (2 s safety margin).  The function
``propose_candidate`` returns a plain ``list[float]`` ready for
``evaluate_sequence``.

# ----------------------------------------------------------------------
# Simplex projection (Euclidean)
# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Project v onto the simplex {x ≥ 0, Σx = s}."""
    if s < 0:
        raise ValueError("simplex radius must be non‑negative")
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.where(u - cssv / (np.arange(1, len(v) + 1)) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# FFT‑based linear convolution (real inputs)
# ----------------------------------------------------------------------
def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Full linear self‑convolution a*a using a real FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1) - 1).bit_length()   # next power of two
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[: 2 * n - 1]


def _fft_convolve(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution (real) of x and y via FFT."""
    n = len(x) + len(y) - 1
    size = 1 << ((n) - 1).bit_length()
    X = np.fft.rfft(x, n=size)
    Y = np.fft.rfft(y, n=size)
    conv = np.fft.irfft(X * Y, n=size)
    return conv[:n]


# ----------------------------------------------------------------------
# Exact objective (scale‑invariant, sum = 1 ⇒ denominator = 1)
# ----------------------------------------------------------------------
def _objective(a: np.ndarray) -> float:
    """Return 2·n·max(conv(a,a)) for a probability vector a."""
    n = a.shape[0]
    b = _fft_self_convolve(a)
    return float(2.0 * n * b.max())

def _distribution_uniform(n: int) -> np.ndarray:

def _distribution_arcsine(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    vals /= vals.sum()
    return vals


def _distribution_hann(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    w = 0.5 - 0.5 * np.cos(2.0 * np.pi * i / (n - 1))
    w_sum = w.sum()
    if w_sum == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w_sum
    return w


def _distribution_edge_weighted(n: int, p: float) -> np.ndarray:
    """Weight ∝ (i·j)^{‑p}, i,j distances from the two edges."""
    i = np.arange(n, dtype=np.float64) + 0.5
    j = (n - np.arange(n, dtype=np.float64) - 0.5)
    w = (i * j) ** (-p)
    w_sum = w.sum()
    if w_sum == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w_sum
    return w


def _distribution_generalized_normal(n: int, q: float) -> np.ndarray:
    """Density ∝ exp(‑|x‑½|^{q}) on [0,1]."""
    i = np.arange(n, dtype=np.float64) + 0.5
    x = i / n - 0.5
    w = np.exp(- (np.abs(x) / 0.5) ** q)
    w_sum = w.sum()
    if w_sum == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:

def _distribution_beta(n: int, alpha: float) -> np.ndarray:
    """Symmetric Beta(α,α) on (0,1) with centre at 0.5."""
    i = np.arange(n, dtype=np.float64) + 0.5
    x = i / n
    # Unnormalised density x^{α-1}(1‑x)^{α-1}
    w = (x * (1.0 - x)) ** (alpha - 1.0)
    w_sum = w.sum()
    if w_sum == 0.0 or np.isnan(w_sum):
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w_sum
    return w


def _distribution_random_sparse(
    n: int, rng: np.random.Generator, density: float = 0.01
) -> np.ndarray:
    """Sparse random vector with a given density."""
    k = max(2, int(n * density))
    idx = rng.choice(n, size=k, replace=False)
    vals = np.zeros(n, dtype=np.float64)
    vals[idx] = rng.random(k)
    if vals.sum() == 0.0:
        vals[0] = 1.0
    vals /= vals.sum()
    return vals


def _mix_with_uniform(base: np.ndarray, lam: float) -> np.ndarray:
    """Convex combination with the uniform distribution."""
    n = base.shape[0]
    uniform = np.full(n, 1.0 / n, dtype=np.float64)
    return lam * uniform + (1.0 - lam) * base


# ----------------------------------------------------------------------
# Sub‑gradient of the maximal convolution entry (used by the fast GD)
# ----------------------------------------------------------------------
def _grad_of_max_conv(a: np.ndarray) -> tuple[np.ndarray, int]:
    """Return (grad, k) where grad = ∇ (a*a)_k for the active index k."""
    n = a.shape[0]
    conv = _fft_self_convolve(a)
    k = int(np.argmax(conv))
    grad = np.zeros_like(a)
    if k < n:
        # overlap length = k+1
        grad[: k + 1] = 2.0 * a[k::-1]
    else:
        start = k - (n - 1)
        grad[start:] = 2.0 * a[start:][::-1]
    return grad, k


# ----------------------------------------------------------------------
# Exponentiated‑gradient descent on the exact (non‑smooth) objective
# ----------------------------------------------------------------------
def _exp_gradient_descent(
    a: np.ndarray,
    deadline: float,
    best_seq: np.ndarray,
    best_score: float,
    eta0: float = 0.9,
    eval_every: int = 60,
) -> tuple[np.ndarray, float]:
    """
    Mirror‑descent on the exact objective using the sub‑gradient of the
    currently maximal convolution entry.
    """
    n = a.shape[0]
    t = 0
    while time.time() < deadline:
        t += 1
        grad, _ = _grad_of_max_conv(a)
        eta = eta0 / math.sqrt(t)
        a = a * np.exp(-eta * grad)          # stay non‑negative
        s = a.sum()
        if s == 0.0:
            a = np.full(n, 1.0 / n, dtype=np.float64)
        else:
            a /= s

        if t % eval_every == 0:
            cur = _objective(a)
            if cur < best_score - 1e-14:
                best_score = cur
                best_seq = a.copy()
                print(f"[{time.time():.1f}] sub‑GD it={t} new best {best_score:.6f}")
    return best_seq, best_score


# ----------------------------------------------------------------------
# Smoothed exponential‑gradient descent (soft‑max of convolution)
# ----------------------------------------------------------------------
def _smooth_exp_gradient_descent(
    a: np.ndarray,
    deadline: float,
    best_seq: np.ndarray,
    best_score: float,
    eta0: float = 0.7,
    lam0: float = 2.0,
    lam_mult: float = 1.001,
    eval_every: int = 40,
) -> tuple[np.ndarray, float]:
    """
    Minimise the smooth approximation
        O_λ(a) = (1/λ)·log‑sum‑exp(λ·(a*a))
    using a multiplicative mirror‑descent update and a slowly increasing λ.
    """
    n = a.shape[0]
    t = 0
    lam = lam0
    rev_a = a[::-1]                     # will be updated each iteration

    while time.time() < deadline:
        t += 1

        # self‑convolution
        conv = _fft_self_convolve(a)
        max_c = conv.max()

        # soft‑max weights (stable)
        exp_shift = lam * (conv - max_c)
        w = np.exp(exp_shift)
        w_sum = w.sum()
        if w_sum == 0.0:
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        # gradient = 2·corr(w, a) (correlation = conv of w with a_rev)
        rev_a = a[::-1]
        grad_corr = _fft_convolve(w, rev_a)                # length 3n‑2
        grad = 2.0 * grad_corr[n - 1 : n - 1 + n]          # slice length n

        # mirror‑descent step (multiplicative)
        eta = eta0 / math.sqrt(t)
        a = a * np.exp(-eta * grad)
        s = a.sum()
        if s == 0.0:
            a = np.full(n, 1.0 / n, dtype=np.float64)
        else:
            a /= s

        # periodic evaluation
        if t % eval_every == 0:
            cur = _objective(a)
            if cur < best_score - 1e-14:
                best_score = cur
                best_seq = a.copy()
                print(f"[{time.time():.1f}] smooth‑GD it={t} λ={lam:.2f} new best {best_score:.6f}")

        # increase λ (more focus on the true max)
        lam = min(lam * lam_mult, 500.0)

    return best_seq, best_score

def _pairwise_annealing(

            break
        i = rng.choice(src_candidates)
        j = rng.integers(n - 1)
        if j >= i:
            j += 1

        delta = a[i] * rng.random() * 0.4
        if delta < 1e-16:
            continue

        new_a = a.copy()
        new_a[i] -= delta
        new_a[j] += delta

        cur = _objective(new_a)
        elapsed = time.time() - start_time
        temperature = max(0.0, 1.0 - elapsed / total_time) * best_score * 0.01

        accept = False
        if cur < best_score - 1e-14:
            accept = True
        elif temperature > 0.0:
            prob = math.exp(-(cur - best_score) / temperature)
            if rng.random() < prob:
                accept = True

        if accept:
            a = new_a
            if cur < best_score - 1e-14:
                best_score = cur
                best_seq = a.copy()
                print(
                    f"[{time.time():.1f}] anneal i={i}→j={j} δ={delta:.3e} new best {best_score:.6f}"
                )
    return best_seq, best_score


# ----------------------------------------------------------------------
# Upsampling (linear interpolation while preserving the simplex)
# ----------------------------------------------------------------------
def _upsample(a: np.ndarray, new_n: int) -> np.ndarray:
    """Interpolate a probability vector to a larger length."""
    old_n = a.shape[0]
    if new_n == old_n:
        return a.copy()
    x_old = (np.arange(old_n, dtype=np.float64) + 0.5) / old_n
    x_new = (np.arange(new_n, dtype=np.float64) + 0.5) / new_n
    a_new = np.interp(x_new, x_old, a, left=0.0, right=0.0)
    a_new = np.maximum(a_new, 0.0)
    s = a_new.sum()
    if s == 0.0:
        a_new[:] = 1.0 / new_n
    else:
        a_new /= s
    return a_new


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: int = 1000) -> list[float]:
    """
    Search for a non‑negative step function that minimises the evaluation
    metric.  The routine respects the 1000 s budget (2 s safety margin).
    """
    rng = np.random.default_rng(seed)
    start = time.time()
    deadline = start + budget_s - 2.0                # safety margin

    overall_best_seq: np.ndarray | None = None
    overall_best_score = float("inf")
    cand_pool: list[tuple[float, np.ndarray]] = []

    # ------------------------------------------------------------------
    # 1.  Assemble a diverse pool of seed vectors (moderate lengths)
    # ------------------------------------------------------------------
    lengths = [2000, 4000, 8000, 16000]               # longer lengths later
    for n in lengths:
        if time.time() >= deadline:
            break
        print(f"[{time.time():.1f}] generating seeds, length={n}")

        # deterministic families
        for base in (
            _distribution_uniform(n),
            _distribution_arcsine(n),
            _distribution_hann(n),
        ):
            sc = _objective(base)
            cand_pool.append((sc, base.copy()))
            if sc < overall_best_score:
                overall_best_score = sc
                overall_best_seq = base.copy()
                print(f"[{time.time():.1f}] new global best (det) score={sc:.6f}")

        # edge‑weighted with a finer grid of p
        p_vals = np.concatenate((
            np.logspace(-3.0, -0.5, 12, base=10),
            np.logspace(-0.5, 0.0, 12, base=10),
            np.linspace(0.1, 2.0, 12),
            np.logspace(0.0, 0.7, 8, base=10),
        ))
        lam_vals = np.linspace(0.0, 1.0, 9)
        for p in p_vals:
            if time.time() >= deadline:
                break
            base = _distribution_edge_weighted(n, float(p))
            for lam in lam_vals:
                if time.time() >= deadline:
                    break
                cand = _mix_with_uniform(base, float(lam))
                sc = _objective(cand)
                cand_pool.append((sc, cand.copy()))
                if sc < overall_best_score:
                    overall_best_score = sc
                    overall_best_seq = cand.copy()
                    print(
                        f"[{time.time():.1f}] new best edge p={p:.3g} λ={lam:.2f} score={sc:.6f}"
                    )

        # generalized normal family
        for q in np.linspace(0.3, 4.0, 14):
            if time.time() >= deadline:
                break
            cand = _distribution_generalized_normal(n, float(q))
            sc = _objective(cand)
            cand_pool.append((sc, cand.copy()))
            if sc < overall_best_score:
                overall_best_score = sc
                overall_best_seq = cand.copy()
                print(f"[{time.time():.1f}] new best gn q={q:.2f} score={sc:.6f}")

        # symmetric Beta family (new!)
        for alpha in np.concatenate((
            np.logspace(-1.5, -0.2, 10, base=10),   # 0.03 … 0.63 (U‑shaped)
            np.linspace(0.4, 3.0, 12),              # near‑uniform to bell‑shaped
            np.logspace(0.0, 0.5, 8, base=10),     # 1 … 3.16
        )):
            if time.time() >= deadline:
                break
            cand = _distribution_beta(n, float(alpha))
            sc = _objective(cand)
            cand_pool.append((sc, cand.copy()))
            if sc < overall_best_score:
                overall_best_score = sc
                overall_best_seq = cand.copy()
                print(f"[{time.time():.1f}] new best beta α={alpha:.3g} score={sc:.6f}")

        # random Dirichlet seeds (log‑uniform concentration)
        for _ in range(12):
            if time.time() >= deadline:
                break
            log_alpha = rng.uniform(-2.0, 1.0)      # α ∈ [0.01, 10]
            alpha = 10 ** log_alpha
            cand = rng.dirichlet(np.full(n, alpha))
            sc = _objective(cand)
            cand_pool.append((sc, cand.copy()))
            if sc < overall_best_score:
                overall_best_score = sc
                overall_best_seq = cand.copy()
                print(f"[{time.time():.1f}] new best Dirichlet α={alpha:.3g} score={sc:.6f}")

        # sparse random seeds
        for _ in range(6):
            if time.time() >= deadline:
                break
            cand = _distribution_random_sparse(n, rng, density=0.02)
            sc = _objective(cand)
            cand_pool.append((sc, cand.copy()))
            if sc < overall_best_score:
                overall_best_score = sc
                overall_best_seq = cand.copy()
                print(f"[{time.time():.1f}] new best sparse score={sc:.6f}")

        # keep only the top‑30 seeds (memory bound)
        cand_pool.sort(key=lambda x: x[0])
        cand_pool = cand_pool[:30]

    # ------------------------------------------------------------------
    # 2.  Incorporate a previously found construction if present
    # ------------------------------------------------------------------
    try:
        prev = np.asarray(height_sequence_1, dtype=np.float64)  # type: ignore[name-defined]
        if prev.ndim == 1:
            prev = prev / prev.sum()
            sc = _objective(prev)
            if sc < overall_best_score:
                overall_best_score = sc
                overall_best_seq = prev.copy()
                print(f"[{time.time():.1f}] previous best score={sc:.6f}")
            cand_pool.append((sc, prev.copy()))
    except Exception:
        pass

    # ------------------------------------------------------------------
    # 3.  Guard against an empty result
    # ------------------------------------------------------------------
    if overall_best_seq is None:
        n = 4000
        overall_best_seq = _distribution_uniform(n)
        overall_best_score = _objective(overall_best_seq)
        print(f"[{time.time():.1f}] fallback uniform score={overall_best_score:.6f}")

    # ------------------------------------------------------------------
    # 4.  Short sub‑gradient GD on all seeds (quick polishing)
    # ------------------------------------------------------------------
    remaining = deadline - time.time()
    if remaining > 20.0 and cand_pool:
        per_candidate = max(0.2, remaining * 0.3 / max(len(cand_pool), 1))
        print(f"[{time.time():.1f}] quick sub‑GD on {len(cand_pool)} seeds (~{per_candidate:.2f}s each)")
        for idx, (score, a0) in enumerate(cand_pool):
            if time.time() >= deadline:
                break
            sub_dead = time.time() + per_candidate
            a_opt, sc_opt = _exp_gradient_descent(
                a0.copy(),
                sub_dead,
                best_seq=overall_best_seq,
                best_score=overall_best_score,
                eta0=1.0,
                eval_every=30,
            )
            if sc_opt < overall_best_score - 1e-12:
                overall_best_score = sc_opt
                overall_best_seq = a_opt.copy()
                print(f"[{time.time():.1f}] after quick sub‑GD seed {idx} new best {overall_best_score:.6f}")

    # ------------------------------------------------------------------
    # 5.  Long smoothed GD on the current global best
    # ------------------------------------------------------------------
    if time.time() < deadline:
        sub_dead = time.time() + (deadline - time.time()) * 0.35
        a_opt, sc_opt = _smooth_exp_gradient_descent(
            overall_best_seq.copy(),
            sub_dead,
            best_seq=overall_best_seq,
            best_score=overall_best_score,
            eta0=0.8,
            lam0=2.0,
            lam_mult=1.001,
            eval_every=50,
        )
        if sc_opt < overall_best_score - 1e-12:
            overall_best_score = sc_opt
            overall_best_seq = a_opt.copy()
            print(f"[{time.time():.1f}] smooth‑GD improved best to {overall_best_score:.6f}")

    # ------------------------------------------------------------------
    # 6.  Upsample the best vector (if time permits) and polish again
    # ------------------------------------------------------------------
    if time.time() < deadline:
        cur_n = overall_best_seq.shape[0]
        target_n = cur_n * 2
        if target_n <= 64000:                     # keep FFT size modest
            print(f"[{time.time():.1f}] upsampling {cur_n} → {target_n}")
            up = _upsample(overall_best_seq, target_n)
            up_sc = _objective(up)
            if up_sc < overall_best_score:
                overall_best_seq = up
                overall_best_score = up_sc
                print(f"[{time.time():.1f}] upsampled new best {overall_best_score:.6f}")

            # short smooth GD on the upsampled vector
            sub_dead = time.time() + (deadline - time.time()) * 0.25
            a_opt, sc_opt = _smooth_exp_gradient_descent(
                overall_best_seq.copy(),
                sub_dead,
                best_seq=overall_best_seq,
                best_score=overall_best_score,
                eta0=0.7,
                lam0=5.0,
                lam_mult=1.0015,
                eval_every=40,
            )
            if sc_opt < overall_best_score - 1e-12:
                overall_best_seq = a_opt.copy()
                overall_best_score = sc_opt
                print(f"[{time.time():.1f}] smooth‑GD after upsample new best {overall_best_score:.6f}")

    # ------------------------------------------------------------------
    # 7.  Final annealing phase
    # ------------------------------------------------------------------
    if time.time() < deadline:
        print(f"[{time.time():.1f}] final annealing phase")
        a_opt, sc_opt = _pairwise_annealing(
            overall_best_seq.copy(),
            deadline,
            best_seq=overall_best_seq,
            best_score=overall_best_score,
            rng=rng,
        )
        if sc_opt < overall_best_score - 1e-12:
            overall_best_seq = a_opt.copy()
            overall_best_score = sc_opt
            print(f"[{time.time():.1f}] after annealing new best {overall_best_score:.6f}")

    # ------------------------------------------------------------------
    print(
        f"Final best score (objective=2·n·max(conv)): {overall_best_score:.6f} "
        f"(length={len(overall_best_seq)})"
    )
    # Return plain Python list – evaluate_sequence expects a list.
    return overall_best_seq.tolist()
