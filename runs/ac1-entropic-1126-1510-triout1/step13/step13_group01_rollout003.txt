# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x ≥ 0, ∑x = s}."""
    if s < 0:
        raise ValueError("simplex radius must be non‑negative")
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho_arr = np.where(u - cssv / (np.arange(1, len(v) + 1)) > 0)[0]
    if rho_arr.size == 0:
        theta = 0.0
    else:
        rho = rho_arr[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Full self‑convolution a*a using a real FFT (returns length 2n‑1)."""
    n = a.shape[0]
    need = 2 * n - 1
    size = 1 << ((need - 1).bit_length())
    A = rfft(a, n=size)
    conv = irfft(A * A, n=size)
    return conv[:need]


# ----------------------------------------------------------------------
def _objective(a: np.ndarray) -> float:
    """Exact objective assuming ∑a = 1: 2·n·max(conv(a,a))."""
    n = a.shape[0]
    b = _fft_self_convolve(a)
    return 2.0 * n * b.max()

def _distribution_uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _distribution_arcsine(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    vals /= vals.sum()
    return vals


def _distribution_tent(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    half = (n - 1) / 2.0
    w = np.where(i <= half, i + 1, n - i)
    w /= w.sum()
    return w


def _distribution_quadratic(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    half = (n - 1) / 2.0
    d = np.abs(i - half) + 1.0
    w = 1.0 / (d ** 2)
    w /= w.sum()
    return w


def _distribution_edge_weighted(n: int, p: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64) + 0.5
    j = (n - np.arange(n, dtype=np.float64) - 0.5)
    w = (i * j) ** (-p)
    w /= w.sum()
    return w


def _distribution_edge_inversepow(n: int, p: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64) + 0.5
    j = (n - np.arange(n, dtype=np.float64) - 0.5)
    w = i ** (-p) + j ** (-p)
    w /= w.sum()
    return w


def _distribution_sine_weighted(n: int, p: float) -> np.ndarray:
    theta = np.pi * (np.arange(n, dtype=np.float64) + 0.5) / n
    w = np.sin(theta) ** p
    w /= w.sum()
    return w


def _distribution_beta_sym(n: int, p: float) -> np.ndarray:
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    w = np.exp((p - 1) * np.log(x) + (p - 1) * np.log(1.0 - x))
    s = w.sum()
    if s == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= s
    return w


def _distribution_exponential_decay(n: int, rate: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    half = (n - 1) / 2.0
    w = np.exp(-rate * np.abs(i - half))
    w /= w.sum()
    return w


def _distribution_random_sparse(
    n: int, rng: np.random.Generator, density: float = 0.006
) -> np.ndarray:
    k = max(2, int(n * density))
    idx = rng.choice(n, size=k, replace=False)
    vals = np.zeros(n, dtype=np.float64)
    vals[idx] = rng.random(k)
    if vals.sum() == 0.0:
        vals[0] = 1.0
    vals /= vals.sum()
    return vals


def _distribution_dirichlet(n: int, rng: np.random.Generator) -> np.ndarray:
    alpha = rng.uniform(0.01, 2.0)
    a = rng.dirichlet(np.full(n, alpha))
    return a.astype(np.float64)

#   Helper to build convex mixtures of a few families
# ----------------------------------------------------------------------
def _mix_families(
    base_vectors: list[np.ndarray], rng: np.random.Generator
) -> np.ndarray:
    """
    Random convex combination of the supplied vectors.
    The mixing coefficients are drawn from a Dirichlet distribution.
    The result is projected back onto the simplex (∑=1).
    """
    k = len(base_vectors)
    if k == 0:
        raise ValueError("no base vectors")
    alphas = rng.dirichlet(np.ones(k))
    combined = np.zeros_like(base_vectors[0])
    for w, a in zip(base_vectors, alphas):
        combined += a * w
    combined = np.maximum(combined, 0.0)
    s = combined.sum()
    if s == 0.0:
        combined = np.full_like(combined, 1.0 / combined.size)
    else:
        combined /= s
    return combined

def _adam_descent(
    a: np.ndarray,
    deadline: float,
    best_seq: np.ndarray,
    best_score: float,
    lr0: float = 0.8,
    active_frac: float = 0.03,
    max_active: int = 25,
    beta1: float = 0.9,
    beta2: float = 0.999,
    eps: float = 1e-8,
    eval_interval: int = 200,
    patience: int = 4000,
) -> tuple[np.ndarray, float]:
    """
    Exponentiated‑gradient descent with Adam moments.
    ``a`` is assumed to lie on the simplex.
    """
    n = a.shape[0]
    t = 0
    no_improve = 0
    m = np.zeros_like(a)      # first moment
    v = np.zeros_like(a)      # second moment
    rng = np.random.default_rng()
    ind_grid = np.arange(n, dtype=np.int32)

    while time.time() < deadline and t < 800_000:
        t += 1

        conv = _fft_self_convolve(a)
        max_c = conv.max()
        thresh = max_c * (1.0 - active_frac)

        act = np.where(conv >= thresh)[0]
        if act.size > max_active:
            # keep only the ``max_active`` largest values
            top_idx = np.argpartition(conv[act], -max_active)[-max_active:]
            act = act[top_idx]

        # sub‑gradient from the active set
        grad = np.zeros_like(a)
        for idx in act:
            offset = idx - ind_grid
            mask = (offset >= 0) & (offset < n)
            grad[mask] += 2.0 * a[offset[mask]]

        # Adam moment updates
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)

        # bias‑correction
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)

        # decaying learning rate
        eta = lr0 / math.sqrt(t)

        # mirror (exponential) update
        a = a * np.exp(-eta * m_hat / (np.sqrt(v_hat) + eps))

        # projection onto the simplex
        a = _project_onto_simplex(a, s=1.0)

        # periodic exact evaluation
        if t % eval_interval == 0:
            cur = _objective(a)
            if cur < best_score - 1e-14:
                best_score = cur
                best_seq = a.copy()
                no_improve = 0
                print(f"[{time.time():.1f}] it={t:,} new global best {best_score:.6f}")
            else:
                no_improve += 1

        # occasional jitter to escape plateaus
        if no_improve >= patience:
            jitter = rng.exponential(scale=0.006, size=n)
            a = a + jitter
            a = np.maximum(a, 0.0)
            a = _project_onto_simplex(a, s=1.0)
            no_improve = 0

    return best_seq, best_score

def _pairwise_swap_local(
    a: np.ndarray,
    deadline: float,
    best_seq: np.ndarray,
    best_score: float,
    max_trials: int | None = None,
) -> tuple[np.ndarray, float]:
    """Move a random amount of mass between two positions if the objective improves."""
    n = a.shape[0]
    trials = 0
    rng = np.random.default_rng()
    while time.time() < deadline:
        if max_trials is not None and trials >= max_trials:
            break
        trials += 1

        src = np.where(a > 1e-12)[0]
        if src.size == 0:
            break
        i = rng.choice(src)
        j = rng.integers(n - 1)
        if j >= i:
            j += 1

        delta = a[i] * rng.random() * 0.35
        if delta < 1e-14:
            continue

        new_a = a.copy()
        new_a[i] -= delta
        new_a[j] += delta
        cur = _objective(new_a)

        if cur < best_score - 1e-14:
            a = new_a
            best_score = cur
            best_seq = a.copy()
            print(f"[{time.time():.1f}] swap {i}->{j} Δ={delta:.3e} new best {best_score:.6f}")

    return best_seq, best_score


# ----------------------------------------------------------------------
#   Cheap simulated‑annealing style local search
# ----------------------------------------------------------------------
def _random_local_search(
    a: np.ndarray,
    deadline: float,
    best_seq: np.ndarray,
    best_score: float,
    init_temp: float = 0.008,
    decay: float = 0.9997,
) -> tuple[np.ndarray, float]:
    """Repeated random mass transfers with Metropolis acceptance."""
    n = a.shape[0]
    rng = np.random.default_rng()
    T = init_temp
    while time.time() < deadline:
        i = rng.integers(n)
        j = rng.integers(n - 1)
        if j >= i:
            j += 1
        delta = a[i] * rng.random() * 0.3
        if delta < 1e-14:
            continue
        new_a = a.copy()
        new_a[i] -= delta
        new_a[j] += delta
        cur = _objective(new_a)
        if cur < best_score - 1e-14:
            a = new_a
            best_score = cur
            best_seq = a.copy()
            print(f"[{time.time():.1f}] SA improve {i}->{j} Δ={delta:.3e} best {best_score:.6f}")
        else:
            # accept with probability exp(-(Δ)/T)
            if rng.random() < math.exp(-(cur - best_score) / max(T, 1e-12)):
                a = new_a
        T *= decay
    return best_seq, best_score

    # 5.  Greedy pairwise refinement

        print(f"[{time.time():.1f}] greedy pairwise refinement")
        overall_best_seq, overall_best_score = _pairwise_swap_local(
            overall_best_seq.copy(),
            deadline,
            best_seq=overall_best_seq,
            best_score=overall_best_score,
            max_trials=300_000,
        )

    # ------------------------------------------------------------------
    # 6.  Final cheap simulated‑annealing style search
    if time.time() < deadline:
        print(f"[{time.time():.1f}] final simulated‑annealing search")
        overall_best_seq, overall_best_score = _random_local_search(
            overall_best_seq.copy(),
            deadline,
            best_seq=overall_best_seq,
            best_score=overall_best_score,
            init_temp=0.01,
            decay=0.9995,
        )

    # ------------------------------------------------------------------
    print(
        f"Final best score (exact) ≈ {overall_best_score:.6f} (length={len(overall_best_seq)})"
    )
    # ``evaluate_sequence`` will apply its own clipping and normalisation,
    # so we simply return the raw simplex vector.
    return overall_best_seq.tolist()
