def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x≥0, Σx = s}."""
    if s < 0:
        raise ValueError("simplex radius must be non‑negative")
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()

    u = np.sort(v)[::-1]

    rho_arr = np.where(u - cssv / (np.arange(1, len(v) + 1)) > 0)[0]
    if rho_arr.size == 0:
        theta = 0.0
    else:
        rho = rho_arr[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _fft_convolve(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full convolution of real vectors x and y using an FFT (real output)."""
    n = x.shape[0] + y.shape[0] - 1
    size = 1 << ((n - 1).bit_length())          # next power of two
    X = np.fft.rfft(x, n=size)
    Y = np.fft.rfft(y, n=size)
    conv = np.fft.irfft(X * Y, n=size)
    return conv[:n]


def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Full self‑convolution a*a (length 2·n‑1)."""
    n = a.shape[0]
    need = 2 * n - 1
    size = 1 << ((need - 1).bit_length())
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[:need]


# ----------------------------------------------------------------------
def _objective(a: np.ndarray) -> float:
    """
    Exact objective  2·n·max(conv(a,a))   for a vector with Σa = 1.
    a is assumed to be non‑negative; the function does **not** re‑normalise.
    """
    n = a.shape[0]
    conv = _fft_self_convolve(a)
    return 2.0 * n * conv.max()


# ----------------------------------------------------------------------
# Analytic families ----------------------------------------------------
def _distribution_uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _distribution_edge_weighted(n: int, p: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64) + 0.5
    j = (n - np.arange(n, dtype=np.float64) - 0.5)
    w = (i * j) ** (-p)
    if w.sum() == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _distribution_sine_weighted(n: int, p: float) -> np.ndarray:
    theta = np.pi * (np.arange(n, dtype=np.float64) + 0.5) / n
    w = np.sin(theta) ** p
    if w.sum() == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _distribution_beta_sym(n: int, p: float) -> np.ndarray:
    """Discrete symmetric Beta(p,p)."""
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    w = np.exp((p - 1) * np.log(x) + (p - 1) * np.log(1.0 - x))
    if w.sum() == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _distribution_beta_asym(n: int, a: float, b: float) -> np.ndarray:
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    w = np.exp((a - 1) * np.log(x) + (b - 1) * np.log(1.0 - x))
    if w.sum() == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _distribution_random_sparse(n: int, rng: np.random.Generator,
                               density: float = 0.01) -> np.ndarray:
    k = max(2, int(n * density))
    idx = rng.choice(n, size=k, replace=False)
    vals = np.zeros(n, dtype=np.float64)
    vals[idx] = rng.random(k)
    if vals.sum() == 0.0:
        vals[0] = 1.0
    vals /= vals.sum()
    return vals


def _mix_with_uniform(base: np.ndarray, lam: float) -> np.ndarray:
    """Convex combination lam·U + (1‑lam)·base (U is uniform)."""
    n = base.shape[0]
    uniform = np.full(n, 1.0 / n, dtype=np.float64)
    return lam * uniform + (1.0 - lam) * base


# ----------------------------------------------------------------------
# Smoothed gradient (soft‑max over the convolution) --------------------
def _softmax_gradient(a: np.ndarray, kappa: float) -> np.ndarray:
    """Gradient of  κ⁻¹·log‑sum‑exp(κ·conv(a,a))  w.r.t. a."""
    n = a.shape[0]
    conv = _fft_self_convolve(a)
    max_val = conv.max()
    # soft‑max weights (stable)
    exps = np.exp(kappa * (conv - max_val))
    w = exps / exps.sum()                     # w has length 2·n‑1, Σw = 1

    # gradient_i = 2·∑_j a_j·w_{i+j}
    # Compute via convolution of reversed a with w and take the middle n entries.
    rev_a = a[::-1]
    grad_full = _fft_convolve(rev_a, w)       # length 3·n‑2
    start = n - 1
    grad = 2.0 * grad_full[start:start + n]
    return grad


# ----------------------------------------------------------------------
# Adam optimiser -------------------------------------------------------
def _adam_optimize(a0: np.ndarray,
                   sub_deadline: float,
                   best_seq: np.ndarray,
                   best_score: float,
                   lr: float = 0.15,
                   beta1: float = 0.9,
                   beta2: float = 0.999,
                   eps: float = 1e-8,
                   kappa_start: float = 8.0,
                   kappa_end: float = 80.0,
                   max_iters: int = 30_000) -> tuple[np.ndarray, float]:
    """
    Adam optimisation of the smoothed objective.
    a0 is assumed to lie on the simplex (Σa = 1, a≥0).
    """
    a = a0.copy()
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    n = a.shape[0]
    t = 0
    # linear schedule for κ
    while time.time() < sub_deadline and t < max_iters:
        t += 1
        kappa = kappa_start + (kappa_end - kappa_start) * (t / max_iters)

        grad = _softmax_gradient(a, kappa)

        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)

        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)

        a_new = a - lr * m_hat / (np.sqrt(v_hat) + eps)
        a_new = _project_onto_simplex(a_new, s=1.0)

        cur_score = _objective(a_new)

        if cur_score < best_score - 1e-14:
            best_score = cur_score
            best_seq = a_new.copy()
            # occasional progress print
            if t % 2000 == 0:
                print(f"[{time.time():.1f}] Adam it={t} new best {best_score:.6f}")

        a = a_new
    return best_seq, best_score


# ----------------------------------------------------------------------
# Deterministic pairwise refinement ------------------------------------
def _pairwise_swap_local(a: np.ndarray,
                         deadline: float,
                         best_seq: np.ndarray,
                         best_score: float,
                         max_trials: int | None = None) -> tuple[np.ndarray, float]:
    """Move a small amount of mass between two positions if it strictly improves."""
    n = a.shape[0]
    rng = np.random.default_rng()
    trials = 0
    while time.time() < deadline:
        if max_trials is not None and trials >= max_trials:
            break
        trials += 1

        # pick a source with non‑negligible mass
        src = np.where(a > 1e-12)[0]
        if src.size == 0:
            break
        i = rng.choice(src)

        # pick a distinct destination
        j = rng.integers(n - 1)
        if j >= i:
            j += 1

        # move a random fraction of a[i] (up to 35 %)
        delta = a[i] * rng.random() * 0.35
        if delta < 1e-14:
            continue

        new_a = a.copy()
        new_a[i] -= delta
        new_a[j] += delta

        cur = _objective(new_a)
        if cur < best_score - 1e-14:
            a = new_a
            best_seq = a.copy()
            best_score = cur
            print(f"[{time.time():.1f}] swap {i}->{j} Δ={delta:.3e} new best {best_score:.6f}")

    return best_seq, best_score


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: int = 1000) -> list[float]:
    """
    Search for a non‑negative step‑function that minimises
    ``evaluate_sequence``.  Returns a plain Python list.
    """
    rng = np.random.default_rng(seed)
    start = time.time()
    deadline = start + budget_s - 3.0       # small safety margin

    overall_best_seq = None
    overall_best_score = float("inf")
    all_candidates: list[tuple[float, np.ndarray]] = []

    # ------------------------------------------------------------------
    # 1. Generate a rich pool of seeds
    lengths = [2000, 3000, 4000, 5000, 8000, 12000, 16000]
    edge_ps = np.geomspace(0.05, 3.0, num=12)   # fine grid
    sine_ps = [0.5, 1.0, 1.5, 2.0, 3.0, 4.0]
    beta_ps = [0.1, 0.2, 0.4, 0.7, 1.0, 1.5, 2.0]
    mix_lams = [0.0, 0.2, 0.5, 0.8, 1.0]

    for n in lengths:
        if time.time() >= deadline:
            break
        print(f"[{time.time():.1f}] scanning length {n}")

        # Uniform
        cand = _distribution_uniform(n)
        sc = _objective(cand)
        all_candidates.append((sc, cand.copy()))
        if sc < overall_best_score:
            overall_best_score, overall_best_seq = sc, cand.copy()
            print(f"[{time.time():.1f}] new global best (uniform) len={n} score={sc:.6f}")

        # Edge‑weighted family
        for p in edge_ps:
            if time.time() >= deadline:
                break
            base = _distribution_edge_weighted(n, float(p))
            for lam in mix_lams:
                cand = _mix_with_uniform(base, lam)
                sc = _objective(cand)
                all_candidates.append((sc, cand.copy()))
                if sc < overall_best_score:
                    overall_best_score, overall_best_seq = sc, cand.copy()
                    print(f"[{time.time():.1f}] new global best edge p={p:.3g} lam={lam:.2f} len={n} score={sc:.6f}")

        # Sine‑weighted family
        for p in sine_ps:
            if time.time() >= deadline:
                break
            base = _distribution_sine_weighted(n, float(p))
            for lam in mix_lams:
                cand = _mix_with_uniform(base, lam)
                sc = _objective(cand)
                all_candidates.append((sc, cand.copy()))
                if sc < overall_best_score:
                    overall_best_score, overall_best_seq = sc, cand.copy()
                    print(f"[{time.time():.1f}] new global best sine p={p} lam={lam:.2f} len={n} score={sc:.6f}")

        # Symmetric Beta family
        for p in beta_ps:
            if time.time() >= deadline:
                break
            base = _distribution_beta_sym(n, float(p))
            for lam in mix_lams:
                cand = _mix_with_uniform(base, lam)
                sc = _objective(cand)
                all_candidates.append((sc, cand.copy()))
                if sc < overall_best_score:
                    overall_best_score, overall_best_seq = sc, cand.copy()
                    print(f"[{time.time():.1f}] new global best βsym p={p:.2g} lam={lam:.2f} len={n} score={sc:.6f}")

        # Asymmetric Beta (random parameters)
        for _ in range(3):
            if time.time() >= deadline:
                break
            a_param = 10 ** rng.uniform(-2, 0)   # 0.01 … 1
            b_param = 10 ** rng.uniform(-2, 0)
            cand = _distribution_beta_asym(n, a_param, b_param)
            sc = _objective(cand)
            all_candidates.append((sc, cand.copy()))
            if sc < overall_best_score:
                overall_best_score, overall_best_seq = sc, cand.copy()
                print(f"[{time.time():.1f}] new global best asym‑beta a={a_param:.3g} b={b_param:.3g} len={n} score={sc:.6f}")

        # Random Dirichlet vectors (varied concentration)
        for _ in range(6):
            if time.time() >= deadline:
                break
            alpha = 10 ** rng.uniform(-2, 0.5)   # roughly 0.01 … 3
            cand = rng.dirichlet(np.full(n, alpha))
            sc = _objective(cand)
            all_candidates.append((sc, cand.copy()))
            if sc < overall_best_score:
                overall_best_score, overall_best_seq = sc, cand.copy()
                print(f"[{time.time():.1f}] new global best Dirichlet α={alpha:.3g} len={n} score={sc:.6f}")

        # Sparse random vectors
        for _ in range(2):
            if time.time() >= deadline:
                break
            cand = _distribution_random_sparse(n, rng, density=0.008)
            sc = _objective(cand)
            all_candidates.append((sc, cand.copy()))
            if sc < overall_best_score:
                overall_best_score, overall_best_seq = sc, cand.copy()
                print(f"[{time.time():.1f}] new global best sparse len={n} score={sc:.6f}")

    # Keep only the very best seeds (10 by default)
    all_candidates.sort(key=lambda x: x[0])
    top_seeds = [seq for _, seq in all_candidates[:10]]

    # ------------------------------------------------------------------
    # 2. Refine the top seeds with Adam + soft‑max smoothing
    remaining = deadline - time.time()
    if remaining > 30.0 and top_seeds:
        per_seed = remaining / len(top_seeds)
        print(f"[{time.time():.1f}] starting Adam refinement on {len(top_seeds)} seeds")
        for idx, seed_seq in enumerate(top_seeds):
            if time.time() >= deadline:
                break
            sub_deadline = time.time() + per_seed
            a_opt, sc_opt = _adam_optimize(seed_seq.copy(),
                                            sub_deadline,
                                            best_seq=overall_best_seq,
                                            best_score=overall_best_score,
                                            lr=0.12,
                                            kappa_start=8.0,
                                            kappa_end=80.0,
                                            max_iters=25_000)
            if sc_opt < overall_best_score:
                overall_best_score, overall_best_seq = sc_opt, a_opt.copy()
                print(f"[{time.time():.1f}] after Adam seed {idx} new global best {overall_best_score:.6f}")

            if overall_best_score <= 1.503:
                print("[early exit] target reached")
                break

    # ------------------------------------------------------------------
    # 3. Final cheap deterministic pairwise refinement
    if time.time() < deadline:
        print(f"[{time.time():.1f}] pairwise refinement")
        overall_best_seq, overall_best_score = _pairwise_swap_local(
            overall_best_seq.copy(),
            deadline,
            best_seq=overall_best_seq,
            best_score=overall_best_score,
            max_trials=300_000,
        )

    # ------------------------------------------------------------------
    print(f"Final best score: {overall_best_score:.6f} (length={len(overall_best_seq)})")
    # Return a plain Python list – evaluate_sequence will handle clipping, etc.
    return overall_best_seq.tolist()
