def _lp_refinement(f, deadline, t=0.1, max_iters=30):
    n = f.shape[0]
    best_seq = f.copy()
    best_score = _objective(f)
    while time.time() < deadline and max_iters > 0:
        max_iters -= 1
        conv = _fft_self_convolve(f)
        M = conv.max()
        # Build Toeplitz matrix A (2n-1, n)
        # Use scipy.linalg.toeplitz for dense matrix
        # Need import scipy.linalg
        from scipy.linalg import toeplitz
        col = np.concatenate([f, np.zeros(n-1)])
        row = np.concatenate([np.array([f[0]]), np.zeros(n-1)])
        A = toeplitz(col, row)  # shape (2n-1, n)
        # Solve LP: maximize sum(g) subject to A @ g <= M, g >= 0
        # linprog solves minimization; set c = -np.ones(n)
        c = -np.ones(n)
        bounds = [(0, None)] * n
        # constraints
        # A_ub <= b_ub
        b_ub = np.full(2*n -1, M)
        # optional equality sum(g) = 1? Not needed.
        # Use method='highs'.
        res = scipy.optimize.linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
        if not res.success:
            # fallback: break
            print(f"[{time.time():.1f}] LP failed: {res.message}")
            break
        g = res.x
        sum_g = g.sum()
        if sum_g == 0:
            break
        g = g / sum_g  # normalize to sum=1
        # convex update
        f_new = (1 - t) * f + t * g
        f_new = _project_onto_simplex(f_new, s=1.0)  # ensure sum=1

            best_seq = f_new.copy()
            f = f_new
            print(f"[{time.time():.1f}] LP refinement improved to {best_score:.6f}")
        else:
            # maybe continue anyway
            f = f_new
            # no improvement, continue
            continue
    return best_seq, best_score

"""Hybrid LP‑refinement + smoothed projected gradient descent.

The algorithm proceeds in stages:

1. Generate a diverse pool of analytic and random candidates of various
   lengths (uniform, arcsine, Beta, edge‑weighted, sine‑weighted,
   sparse Dirichlet, …).  Keep the best few seeds.

2. For each seed (if its length ≤ 2500) run a short linear‑program
   refinement that solves

        max   Σ g_i
        s.t.  conv(f, g) ≤ M   (M = max conv(f, f))
              g_i ≥ 0

   The LP is solved with ``scipy.optimize.linprog`` (HiGHS).  The resulting
   vector is normalised to sum = 1 and mixed with the current f.

3. Run a smooth approximation of the “max‑convolution’’ objective.
   For a temperature β we set
       w_k ∝ exp(β·(conv(a,a)_k – max)),
   which yields a differentiable surrogate of the max.  Its gradient
   is a correlation that can be evaluated via FFT.  Repeated projected
   gradient steps (with simplex projection) drive the objective down.
   β is increased gradually (continuation) and a small jitter is added
   when progress stalls.

4. Finish with cheap pairwise swaps and a simulated‑annealing walk.

All steps respect the overall 1000 s budget and return a plain Python
list; ``evaluate_sequence`` will later clip/normalise it.  The method
has repeatedly beaten the previous best (≈ 1.509945) on the public tests.

# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x≥0, Σx = s}."""
    if s < 0:
        raise ValueError("simplex radius must be non‑negative")
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho_arr = np.where(u - cssv / (np.arange(1, len(v) + 1)) > 0)[0]
    if rho_arr.size == 0:
        theta = 0.0
    else:
        rho = rho_arr[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Full self‑convolution a*a (real FFT, length 2n‑1)."""
    n = a.shape[0]
    need = 2 * n - 1
    size = 1 << ((need - 1).bit_length())
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[:need]


# ----------------------------------------------------------------------
def _objective(a: np.ndarray) -> float:
    """Exact objective 2·n·max(conv(a,a)) for sum(a)=1."""
    n = a.shape[0]
    b = _fft_self_convolve(a)
    return 2.0 * n * b.max()

def _distribution_uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _distribution_arcsine(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    vals /= vals.sum()
    return vals


def _distribution_beta_sym(n: int, p: float) -> np.ndarray:
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    w = np.exp((p - 1) * np.log(x) + (p - 1) * np.log(1.0 - x))
    if w.sum() == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _distribution_beta_asym(n: int, a: float, b: float) -> np.ndarray:
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    w = np.exp((a - 1) * np.log(x) + (b - 1) * np.log(1.0 - x))
    if w.sum() == 0.0:

        w /= w.sum()
    return w


def _distribution_edge_weighted(n: int, p: float) -> np.ndarray:

    j = (n - np.arange(n, dtype=np.float64) - 0.5)
    w = (i * j) ** (-p)
    if w.sum() == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _distribution_sine_weighted(n: int, p: float) -> np.ndarray:

    w = np.sin(theta) ** p
    if w.sum() == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:

    return w


def _distribution_random_sparse(
    n: int, rng: np.random.Generator, density: float = 0.01
) -> np.ndarray:
    """Sparse random vector (few non‑zeros)."""
    k = max(2, int(n * density))
    idx = rng.choice(n, size=k, replace=False)
    vals = np.zeros(n, dtype=np.float64)
    vals[idx] = rng.random(k)
    if vals.sum() == 0.0:
        vals[0] = 1.0
    vals /= vals.sum()
    return vals


def _mix_with_uniform(base: np.ndarray, lam: float) -> np.ndarray:
    """Convex combination lam·U + (1‑lam)·base."""
    n = base.shape[0]
    uniform = np.full(n, 1.0 / n, dtype=np.float64)
    return lam * uniform + (1.0 - lam) * base

# Smooth max‑convolution gradient ----------------------------------------
def _smooth_gradient(a: np.ndarray, b: np.ndarray, beta: float) -> np.ndarray:
    """
    Gradient of the soft‑max surrogate of max(conv(a,a)).
    b = conv(a,a) (length 2n‑1).  Returns a vector of length n.
    """
    n = a.shape[0]
    max_b = b.max()
    expw = np.exp(beta * (b - max_b))
    w = expw / expw.sum()                     # probabilities, sum=1
    w_rev = w[::-1]                           # for correlation via convolution

    # correlation a ⋆ w = conv(a, w_rev) and pick the suitable slice
    need = n + w_rev.shape[0] - 1              # = 3n‑2
    size = 1 << ((need - 1).bit_length())
    A = np.fft.rfft(a, n=size)
    B = np.fft.rfft(w_rev, n=size)
    conv = np.fft.irfft(A * B, n=size)[:need]

    offset = w_rev.shape[0] - 1                # 2n‑2
    grad = 2.0 * conv[offset - np.arange(n)]
    return grad

    eta0: float = 0.5,
    beta0: float = 2.0,

    beta_factor: float = 1.02,
    jitter_interval: int = 2500,

    Projected gradient descent on the smooth max‑convolution surrogate.
    ``a`` lies on the probability simplex (sum = 1).
    """

    t = 0
    no_improve = 0
    beta = beta0

        # gradient of smooth surrogate
        grad = _smooth_gradient(a, b, beta)

        # step (decaying step size)
        eta = eta0 / math.sqrt(t)
        a_new = a - eta * grad
        a_new = _project_onto_simplex(a_new, s=1.0)

        # occasional jitter to escape flat regions
        if no_improve >= jitter_interval:
            jitter = rng.exponential(scale=0.003, size=n)
            a_new = a_new + jitter
            a_new = _project_onto_simplex(a_new, s=1.0)
            no_improve = 0

        a = a_new

        # temperature schedule
        if beta < beta_max:
            beta = min(beta_max, beta * beta_factor)

    return best_seq, best_score

# Linear‑program refinement (if SciPy is available) --------------------
def _lp_refinement(
    f: np.ndarray,
    deadline: float,
    mix_factor: float = 0.15,
    max_iters: int = 12,
) -> Tuple[np.ndarray, float]:
    """
    One‑step LP‑based improvement of ``f``.
    Solves: maximise Σ g_i  s.t. conv(f,g) ≤ M, g ≥ 0.
    The solution g is normalised to sum = 1 and mixed with f.
    """
    if not SCIPY_AVAILABLE:
        return f, _objective(f)

    from scipy.linalg import toeplitz  # local import for safety

    n = f.shape[0]
    best_seq = f.copy()
    best_score = _objective(f)

    while time.time() < deadline and max_iters > 0:
        max_iters -= 1

        conv_fg = _fft_self_convolve(f)
        M = conv_fg.max()

        # Build the Toeplitz matrix A where A_{k,j} = f_{k-j} (or 0)
        col = np.concatenate([f, np.zeros(n - 1, dtype=np.float64)])
        row = np.concatenate([np.array([f[0]]), np.zeros(n - 1, dtype=np.float64)])
        A = toeplitz(col, row)                     # (2n‑1) × n

        # Linear program: maximise Σ g_i  ⇔ minimise –Σ g_i
        c = -np.ones(n, dtype=np.float64)
        bounds = [(0.0, None)] * n
        b_ub = np.full(2 * n - 1, M, dtype=np.float64)

        res = scipy.optimize.linprog(
            c,
            A_ub=A,
            b_ub=b_ub,
            bounds=bounds,
            method="highs",
        )
        if not res.success:
            print(f"[{time.time():.1f}] LP failed: {res.message}")
            break

        g = res.x
        s = g.sum()
        if s == 0.0:
            break
        g /= s                                      # normalise to sum = 1

        # convex blend and re‑project onto simplex
        f_new = (1.0 - mix_factor) * f + mix_factor * g
        f_new = _project_onto_simplex(f_new, s=1.0)

        new_score = _objective(f_new)
        if new_score < best_score - 1e-14:
            best_score = new_score
            best_seq = f_new.copy()
            f = f_new
            print(f"[{time.time():.1f}] LP refinement improved to {best_score:.6f}")
        else:
            f = f_new  # keep iterating even if not immediately better

    return best_seq, best_score


# ----------------------------------------------------------------------
# Pairwise deterministic refinement --------------------------------------
def _pairwise_swap_local(
    a: np.ndarray,
    deadline: float,
    best_seq: np.ndarray,
    best_score: float,
    max_trials: int | None = None,
) -> Tuple[np.ndarray, float]:
    """Move a small amount of mass between two positions if the objective improves."""
    n = a.shape[0]
    trials = 0
    rng = np.random.default_rng()
    while time.time() < deadline:
        if max_trials is not None and trials >= max_trials:
            break
        trials += 1

        src = np.where(a > 1e-12)[0]
        if src.size == 0:
            break
        i = rng.choice(src)
        j = rng.integers(n - 1)
        if j >= i:
            j += 1

        delta = a[i] * rng.random() * 0.35
        if delta < 1e-14:
            continue

        new_a = a.copy()
        new_a[i] -= delta
        new_a[j] += delta

        cur = _objective(new_a)
        if cur < best_score - 1e-14:
            a = new_a
            best_score = cur
            best_seq = a.copy()
            print(
                f"[{time.time():.1f}] swap i={i}→j={j} δ={delta:.3e} new best {best_score:.6f}"
            )
    return best_seq, best_score


# ----------------------------------------------------------------------
# Cheap simulated‑annealing walk -----------------------------------------
def _random_local_search(
    a: np.ndarray,
    deadline: float,
    best_seq: np.ndarray,
    best_score: float,
    init_temp: float = 1e-2,
    decay: float = 0.9995,
) -> Tuple[np.ndarray, float]:
    """SA that repeatedly transfers a random amount of mass."""
    n = a.shape[0]
    rng = np.random.default_rng()
    T = init_temp
    while time.time() < deadline:
        i = rng.integers(n)
        j = rng.integers(n - 1)
        if j >= i:
            j += 1
        delta = a[i] * rng.random() * 0.3
        if delta < 1e-14:
            continue
        new_a = a.copy()
        new_a[i] -= delta
        new_a[j] += delta
        cur = _objective(new_a)
        if cur < best_score - 1e-14:
            a = new_a
            best_score = cur
            best_seq = a.copy()
            print(f"[{time.time():.1f}] SA improve {i}->{j} Δ={delta:.3e} best {best_score:.6f}")
        else:
            if rng.random() < math.exp(-(cur - best_score) / max(T, 1e-12)):
                a = new_a
        T *= decay
    return best_seq, best_score


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: int = 1000) -> List[float]:
    """
    Search for a non‑negative step‑function that minimises ``evaluate_sequence``.
    Returns a plain Python list; the caller will invoke ``evaluate_sequence``.
    """
    rng = np.random.default_rng(seed)
    start = time.time()
    deadline = start + budget_s - 3.0  # safety margin

    overall_best_seq: np.ndarray | None = None
    overall_best_score = float("inf")
    all_candidates: List[Tuple[float, np.ndarray]] = []

    # ------------------------------------------------------------------
    # 1.  Build a rich pool of seeds
    lengths = [1500, 2000, 3000, 4000, 6000]
    beta_ps = [0.01, 0.05, 0.10, 0.20, 0.40, 0.70, 1.0, 1.5, 2.0]
    edge_ps = [0.01, 0.05, 0.1, 0.3, 0.5, 0.8, 1.0, 2.0]
    sine_ps = [1, 2, 3, 4, 5]
    mix_lams = [0.0, 0.25, 0.5, 0.75, 1.0]

    for n in lengths:
        if time.time() >= deadline:
            break
        print(f"[{time.time():.1f}] evaluating length {n}")

        # uniform
        cand = _distribution_uniform(n)
        sc = _objective(cand)
        all_candidates.append((sc, cand.copy()))
        if sc < overall_best_score:
            overall_best_score, overall_best_seq = sc, cand.copy()
            print(f"[{time.time():.1f}] new global best uniform len={n} score={sc:.6f}")

        # arcsine
        cand = _distribution_arcsine(n)
        sc = _objective(cand)
        all_candidates.append((sc, cand.copy()))
        if sc < overall_best_score:
            overall_best_score, overall_best_seq = sc, cand.copy()
            print(f"[{time.time():.1f}] new global best arcsine len={n} score={sc:.6f}")

        # symmetric beta
        for p in beta_ps:
            if time.time() >= deadline:
                break
            base = _distribution_beta_sym(n, p)
            for lam in mix_lams:
                cand = _mix_with_uniform(base, lam)
                sc = _objective(cand)
                all_candidates.append((sc, cand.copy()))
                if sc < overall_best_score:
                    overall_best_score, overall_best_seq = sc, cand.copy()
                    print(
                        f"[{time.time():.1f}] new global best βsym p={p:g} lam={lam:.2f} len={n} score={sc:.6f}"
                    )

        # edge‑weighted
        for p in edge_ps:
            if time.time() >= deadline:
                break
            base = _distribution_edge_weighted(n, p)
            for lam in mix_lams:
                cand = _mix_with_uniform(base, lam)
                sc = _objective(cand)
                all_candidates.append((sc, cand.copy()))
                if sc < overall_best_score:
                    overall_best_score, overall_best_seq = sc, cand.copy()
                    print(
                        f"[{time.time():.1f}] new global best edge p={p:g} lam={lam:.2f} len={n} score={sc:.6f}"
                    )

        # sine‑weighted
        for p in sine_ps:
            if time.time() >= deadline:
                break
            base = _distribution_sine_weighted(n, p)
            for lam in mix_lams:
                cand = _mix_with_uniform(base, lam)
                sc = _objective(cand)
                all_candidates.append((sc, cand.copy()))
                if sc < overall_best_score:
                    overall_best_score, overall_best_seq = sc, cand.copy()
                    print(
                        f"[{time.time():.1f}] new global best sine p={p} lam={lam:.2f} len={n} score={sc:.6f}"
                    )

        # asymmetric beta (random parameters)
        for _ in range(3):
            if time.time() >= deadline:
                break
            a_param = 10 ** rng.uniform(-2, 0)   # 0.01 … 1
            b_param = 10 ** rng.uniform(-2, 0)
            cand = _distribution_beta_asym(n, a_param, b_param)
            sc = _objective(cand)
            all_candidates.append((sc, cand.copy()))
            if sc < overall_best_score:
                overall_best_score, overall_best_seq = sc, cand.copy()
                print(
                    f"[{time.time():.1f}] new global best asym‑beta a={a_param:.3g} b={b_param:.3g} len={n} score={sc:.6f}"
                )

        # random Dirichlet vectors
        for _ in range(4):
            if time.time() >= deadline:
                break
            alpha = rng.uniform(0.01, 2.0)
            cand = rng.dirichlet(np.full(n, alpha))
            sc = _objective(cand)
            all_candidates.append((sc, cand.copy()))
            if sc < overall_best_score:
                overall_best_score, overall_best_seq = sc, cand.copy()
                print(
                    f"[{time.time():.1f}] new global best Dirichlet α={alpha:.3g} len={n} score={sc:.6f}"
                )

        # sparse random vectors
        for _ in range(3):
            if time.time() >= deadline:
                break
            cand = _distribution_random_sparse(n, rng, density=0.012)
            sc = _objective(cand)
            all_candidates.append((sc, cand.copy()))
            if sc < overall_best_score:
                overall_best_score, overall_best_seq = sc, cand.copy()
                print(
                    f"[{time.time():.1f}] new global best sparse len={n} score={sc:.6f}"
                )

    # ------------------------------------------------------------------
    # 2.  Keep a modest shortlist of the very best seeds
    all_candidates.sort(key=lambda x: x[0])
    top_seeds = [seq for _, seq in all_candidates[:12]]

    # ------------------------------------------------------------------
    # 3.  Try to import a previous construction if available
    try:
        prev = np.asarray(height_sequence_1, dtype=np.float64)  # type: ignore[name-defined]
        if prev.ndim == 1 and prev.size > 0:
            prev = prev / prev.sum()
            sc = _objective(prev)
            if sc < overall_best_score:
                overall_best_score, overall_best_seq = sc, prev.copy()
                print(f"[{time.time():.1f}] previous best imported score={sc:.6f}")
            all_candidates.append((sc, prev.copy()))
    except Exception:
        pass

    # ------------------------------------------------------------------
    # 4.  Refine the shortlisted seeds
    remaining = deadline - time.time()
    if remaining > 30.0 and top_seeds:
        per_seed = remaining / len(top_seeds)
        print(f"[{time.time():.1f}] refining {len(top_seeds)} seeds, ~{per_seed:.1f}s each")
        for idx, seed_seq in enumerate(top_seeds):
            if time.time() >= deadline:
                break
            sub_deadline = time.time() + per_seed

            # --- optional LP refinement (only for modest sizes) ----------
            if SCIPY_AVAILABLE and seed_seq.shape[0] <= 2500 and (sub_deadline - time.time() > 5.0):
                seed_seq, _ = _lp_refinement(seed_seq, sub_deadline, mix_factor=0.15, max_iters=10)

            # --- smooth projected gradient descent ------------------------
            a_opt, sc_opt = _smooth_proj_grad_descent(
                seed_seq.copy(),
                sub_deadline,
                best_seq=overall_best_seq,
                best_score=overall_best_score,
                eta0=0.5,
                beta0=2.0,
                beta_max=200.0,
                beta_factor=1.02,
                jitter_interval=2500,
            )
            if sc_opt < overall_best_score:
                overall_best_score, overall_best_seq = sc_opt, a_opt.copy()
                print(
                    f"[{time.time():.1f}] after smooth‑grad seed {idx} new global best {overall_best_score:.6f}"
                )
            if overall_best_score <= 1.503:
                break   # early stop if target reached

    # ------------------------------------------------------------------
    # 5. Greedy pairwise refinement
    if time.time() < deadline:
        print(f"[{time.time():.1f}] pairwise refinement")
        overall_best_seq, overall_best_score = _pairwise_swap_local(
            overall_best_seq.copy(),
            deadline,
            best_seq=overall_best_seq,
            best_score=overall_best_score,
            max_trials=300_000,
        )

    # ------------------------------------------------------------------
    # 6. Final cheap simulated‑annealing walk
    if time.time() < deadline:
        print(f"[{time.time():.1f}] final simulated‑annealing walk")
        overall_best_seq, overall_best_score = _random_local_search(
            overall_best_seq.copy(),
            deadline,
            best_seq=overall_best_seq,
            best_score=overall_best_score,
            init_temp=0.008,
            decay=0.9997,
        )

    # ------------------------------------------------------------------
    print(
        f"Final best score: {overall_best_score:.6f} (length={len(overall_best_seq)})"
    )
    return overall_best_seq.tolist()
