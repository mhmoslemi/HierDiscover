        loop_counter += 1
        conv_f = _fft_self_convolve(best_f)

        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=(0, None), method='highs')
        if not res.success:
            break
        g = res.x
        sum_g = g.sum()
        if sum_g <= 1 + 1e-12:
            # cannot improve via this f
            break
        # Interpolate
        t = 0.1  # maybe adapt
        g_norm = g / sum_g
        new_f = (1 - t) * best_f + t * g_norm
        # Normalize to sum 1 (just in case)
        new_f = new_f / new_f.sum()
        # Evaluate
        new_score = _objective(new_f)
        if new_score + 1e-12 < best_score:
            best_f, best_score = new_f, new_score
            print(f"[{time.time():.1f}] LP iter {loop_counter} new best {best_score:.6f}")

def lp_refine(a0, time_limit, t_step=0.2):
    a = a0.copy()
    n = a.shape[0]
    best_score = _objective(a)
    best_vec = a.copy()
    while time.time() < time_limit:
        conv = _fft_self_convolve(a)
        M = conv.max()
        # Build constraint matrix
        A = build_conv_matrix(a)
        b_ub = np.full(2*n-1, M, dtype=np.float64)
        c = -np.ones(n, dtype=np.float64)  # maximize sum(g) => minimize -sum(g)
        res = scipy.optimize.linprog(c, A_ub=A, b_ub=b_ub, bounds=[(0, None)]*n, method='highs', options={'presolve': True, 'time_limit': max(0.0, time_limit - time.time())})
        if not res.success:
            break
        g = res.x
        sum_g = g.sum()
        if sum_g <= 1.0 + 1e-12:
            # cannot increase total mass
            break
        # Interpolate towards g normalized
        g_norm = g / sum_g
        # Choose step size adaptively: t = min(1., (sum_g - 1) / (sum_g + 1))
        # But we keep t_step constant as given
        new_a = (1 - t_step) * a + t_step * g_norm
        new_a = new_a / new_a.sum()
        new_score = _objective(new_a)
        if new_score + 1e-12 < best_score:
            best_score = new_score
            best_vec = new_a.copy()
            a = new_a
            print(f"[{time.time():.1f}] LP interp new best {best_score:.6f}")
            continue
        else:
            # If no improvement, reduce step size or break
            t_step *= 0.5
            if t_step < 1e-3:
                break
    return best_vec, best_score

"""
Propose a non‑negative step‑function that minimizes the supplied
``evaluate_sequence``.  The algorithm consists of three stages:

1️⃣  Initialise from a collection of analytic families (uniform, arcsine,
    edge‑weighted, exponential‑edge, β‑shaped) and from the previous
    construction ``height_sequence_1`` (if it exists).  Keep only the
    best few seeds.

2️⃣  Refine each seed with a *linear‑programming* improvement step
    (the “LP‑step’’ from the description).  For a current vector `a`
    we maximise the total mass of a new vector `g` while keeping the
    convolution bound `a * g ≤ M` (here `M = max(a*a)`).  A convex
    combination `a←(1‑t)a + t·(g/∑g)` (with `t≈0.2`) strictly reduces
    the objective whenever possible.

3️⃣  Polish the best vector with exact sub‑gradient (exponentiated‑gradient)
    descent on the true `max(conv)` and a final greedy pairwise mass‑
    transfer hill‑climber.

All work is stopped after the allocated time budget (default 1000 s) and
the best sequence found is returned as a plain Python list.
"""

import time
import math
import numpy as np
import scipy.sparse as sp
import scipy.optimize

# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Euclidean projection onto the simplex {x≥0, Σx=s}."""
    if s < 0:
        raise ValueError("Simplex radius must be non‑negative")
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.where(u - cssv / (np.arange(1, len(v) + 1)) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    return np.maximum(v - theta, 0.0)


# ----------------------------------------------------------------------
def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Full self‑convolution a*a using a real FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1) - 1).bit_length()
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[: 2 * n - 1]


# ----------------------------------------------------------------------
def _objective(a: np.ndarray) -> float:
    """Exact objective 2·n·max(conv(a,a)) for a vector with Σa = 1."""
    n = a.shape[0]
    b = _fft_self_convolve(a)
    return float(2.0 * n * b.max())


# ----------------------------------------------------------------------
# Analytic families ----------------------------------------------------
def _distribution_uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _distribution_arcsine(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    vals /= vals.sum()
    return vals


def _distribution_edge_weighted(n: int, p: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64) + 0.5
    j = (n - np.arange(n, dtype=np.float64) - 0.5)
    w = (i * j) ** (-p)
    w_sum = w.sum()
    if w_sum == 0.0:
        return _distribution_uniform(n)
    w /= w_sum
    return w


def _distribution_exponential_edge(n: int, c: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    w = np.exp(-c * i) + np.exp(-c * (n - 1 - i))
    w /= w.sum()
    return w


def _distribution_beta(n: int, alpha: float) -> np.ndarray:
    """Symmetric Beta‑shaped law: w_i ∝ (i·(n‑i))^{α‑1}, 0<α<1 gives edge‑heavy."""
    i = np.arange(n, dtype=np.float64) + 0.5
    w = (i * (n - i)) ** (alpha - 1.0)
    w = np.clip(w, 0, None)
    s = w.sum()
    if s == 0.0:
        return _distribution_uniform(n)
    w /= s
    return w


def _mix_with_uniform(base: np.ndarray, lam: float) -> np.ndarray:
    """Convex combination λ·uniform + (1‑λ)·base."""
    n = base.shape[0]
    uniform = np.full(n, 1.0 / n, dtype=np.float64)
    return lam * uniform + (1.0 - lam) * base


def _distribution_random_sparse(n: int, rng: np.random.Generator, density: float = 0.01) -> np.ndarray:
    """Sparse random vector with a prescribed density of non‑zeros."""
    k = max(2, int(n * density))
    idx = rng.choice(n, size=k, replace=False)
    vals = np.zeros(n, dtype=np.float64)
    vals[idx] = rng.random(k)
    s = vals.sum()
    if s == 0.0:
        vals[0] = 1.0
        s = 1.0
    vals /= s
    return vals


# ----------------------------------------------------------------------
# Convolution matrix (used in the LP step) ----------------------------
def _conv_matrix(a: np.ndarray) -> sp.csr_matrix:
    """Return sparse matrix A such that A @ g = a * g (full convolution)."""
    n = a.shape[0]
    # Offsets 0, -1, -2, ..., -(n‑1) (lower diagonals of a rectangular matrix)
    offsets = np.arange(0, -n, -1, dtype=int)
    # Each diagonal receives the same data vector a
    data = [a] * n
    A = sp.diags(data, offsets, shape=(2 * n - 1, n), format="csr")
    return A


# ----------------------------------------------------------------------
# Linear‑programming based refinement ----------------------------------
def _lp_refine(a0: np.ndarray, time_limit: float, t_step: float = 0.2) -> tuple[np.ndarray, float]:
    """
    Repeatedly solve

        max Σ g_j   s.t.   a * g ≤ M ,   g ≥ 0

    where M = max(a*a).  Interpolate towards the normalised solution.
    """
    a = a0.copy()
    best_score = _objective(a)
    best_vec = a.copy()
    while time.time() < time_limit:
        n = a.shape[0]
        conv = _fft_self_convolve(a)
        M = conv.max()

        # Build A such that A @ g = a * g
        A = _conv_matrix(a)
        b_ub = np.full(2 * n - 1, M, dtype=np.float64)

        # linprog solves   min c·x   s.t.  A_ub·x ≤ b_ub, x ≥ 0
        c = -np.ones(n, dtype=np.float64)               # maximise Σ g_j
        res = scipy.optimize.linprog(
            c,
            A_ub=A,
            b_ub=b_ub,
            bounds=[(0, None)] * n,
            method="highs",
            options={"presolve": True},
        )

        if not res.success:
            # LP failed – abort the refinement
            break

        g = res.x
        sum_g = g.sum()
        if sum_g <= 1.0 + 1e-12:
            # No mass increase possible → optimal for current M
            break

        # Normalise g and move towards it
        g_norm = g / sum_g
        a = (1.0 - t_step) * a + t_step * g_norm
        a = a / a.sum()

        cur_score = _objective(a)
        if cur_score + 1e-12 < best_score:
            best_score = cur_score
            best_vec = a.copy()
            # keep t_step as is – we made progress
        else:
            # No improvement – shrink the step size
            t_step *= 0.5
            if t_step < 1e-4:
                break

    return best_vec, best_score


# ----------------------------------------------------------------------
# Exact sub‑gradient descent (exponentiated‑gradient) ------------------
def _subgradient_descent_exact(
    a0: np.ndarray,
    time_limit: float,
    best_seq: np.ndarray,
    best_score: float,
    lr0: float = 0.9,
    eval_every: int = 100,
) -> tuple[np.ndarray, float]:
    a = a0.copy()
    n = a.shape[0]
    t = 0
    while time.time() < time_limit:
        t += 1
        conv = _fft_self_convolve(a)
        max_val = conv.max()
        max_idxs = np.where(conv == max_val)[0]

        grad = np.zeros_like(a)
        for idx in max_idxs:
            i_start = max(0, idx - (n - 1))
            i_end = min(n - 1, idx)
            i = np.arange(i_start, i_end + 1, dtype=np.int64)
            j = idx - i
            grad[i] += 2.0 * a[j]

        eta = lr0 / math.sqrt(t)
        a *= np.exp(-eta * grad)          # exponentiated‑gradient (keeps positivity)
        s = a.sum()
        if s <= 0.0:
            a = np.full(n, 1.0 / n, dtype=np.float64)
        else:
            a /= s

        if t % eval_every == 0:
            cur = 2.0 * n * max_val
            if cur < best_score - 1e-14:
                best_score = cur
                best_seq = a.copy()
                print(f"[{time.time():.1f}] sub‑grad it={t} new global best {best_score:.6f}")

    return best_seq, best_score


# ----------------------------------------------------------------------
# Greedy pairwise mass transfer ----------------------------------------
def _pairwise_swap(
    a0: np.ndarray,
    time_limit: float,
    best_seq: np.ndarray,
    best_score: float,
    max_trials: int | None = None,
) -> tuple[np.ndarray, float]:
    a = a0.copy()
    n = a.shape[0]
    trials = 0
    while time.time() < time_limit:
        if max_trials is not None and trials >= max_trials:
            break
        trials += 1

        src_candidates = np.where(a > 1e-14)[0]
        if src_candidates.size == 0:
            break
        i = np.random.choice(src_candidates)
        j = np.random.randint(n - 1)
        if j >= i:
            j += 1

        delta = a[i] * np.random.rand() * 0.45   # up to 45 % of source mass
        if delta < 1e-15:
            continue

        new_a = a.copy()
        new_a[i] -= delta
        new_a[j] += delta
        new_a = new_a / new_a.sum()
        cur = _objective(new_a)
        if cur < best_score - 1e-14:
            a = new_a
            best_score = cur
            best_seq = a.copy()
            print(f"[{time.time():.1f}] swap i={i}→j={j} δ={delta:.6g} new best {best_score:.6f}")

    return best_seq, best_score

def propose_candidate(seed: int = 42, budget_s: int = 1000) -> list[float]:
    """
    Search for a non‑negative step‑function that minimises
    ``evaluate_sequence``.  The function respects the provided time budget
    and returns the best sequence found as a plain list of floats.
    """
    rng = np.random.default_rng(seed)
    start = time.time()
    deadline = start + budget_s - 2.0        # safety margin for the final return

    overall_best_seq = None
    overall_best_score = float("inf")
    top_candidates: list[tuple[float, np.ndarray]] = []   # (score, seq)

    # ------------------------------------------------------------------
    # 1.  Initial candidate pool (analytic families + previous result)
    # ------------------------------------------------------------------
    lengths = [1800, 2400, 3000, 3400, 3800]          # modest sizes – large enough to be powerful
    p_grid = np.concatenate(
        [
            np.logspace(-3, -0.5, 12),          # very edge‑heavy
            np.logspace(-0.5, 0.0, 8),          # moderate
            np.linspace(0.1, 3.0, 10),          # broader range
        ]
    )
    c_grid = np.logspace(-3, 0, 12)
    alpha_grid = np.logspace(-3, -0.1, 12)          # β‑shape exponent (α<1 → edge‑heavy)

    lam_grid = np.linspace(0.0, 1.0, 9)             # mixing with uniform

    for n in lengths:
        if time.time() >= deadline:
            break
        print(f"[{time.time():.1f}] scanning length {n}")

        # deterministic families
        for base_fn in (
            _distribution_uniform,
            _distribution_arcsine,
            _distribution_edge_weighted,
            _distribution_exponential_edge,
            _distribution_beta,
        ):
            if base_fn is _distribution_edge_weighted:
                for p in p_grid:
                    cand = _distribution_edge_weighted(n, float(p))
                    for lam in lam_grid:
                        cand_mix = _mix_with_uniform(cand, lam)
                        sc = _objective(cand_mix)
                        top_candidates.append((sc, cand_mix.copy()))
                        if sc < overall_best_score:
                            overall_best_score, overall_best_seq = sc, cand_mix.copy()
            elif base_fn is _distribution_exponential_edge:
                for c in c_grid:
                    cand = _distribution_exponential_edge(n, float(c))
                    for lam in lam_grid:
                        cand_mix = _mix_with_uniform(cand, lam)
                        sc = _objective(cand_mix)
                        top_candidates.append((sc, cand_mix.copy()))
                        if sc < overall_best_score:
                            overall_best_score, overall_best_seq = sc, cand_mix.copy()
            elif base_fn is _distribution_beta:
                for alpha in alpha_grid:
                    cand = _distribution_beta(n, float(alpha))
                    for lam in lam_grid:
                        cand_mix = _mix_with_uniform(cand, lam)
                        sc = _objective(cand_mix)
                        top_candidates.append((sc, cand_mix.copy()))
                        if sc < overall_best_score:
                            overall_best_score, overall_best_seq = sc, cand_mix.copy()
            else:
                # uniform or arcsine – no extra parameters
                cand = base_fn(n)
                sc = _objective(cand)
                top_candidates.append((sc, cand.copy()))
                if sc < overall_best_score:
                    overall_best_score, overall_best_seq = sc, cand.copy()

        # random Dirichlet seeds (more diverse)
        for _ in range(4):
            if time.time() >= deadline:
                break
            alpha = rng.uniform(0.01, 2.0)
            a = rng.dirichlet(np.full(n, alpha))
            sc = _objective(a)
            top_candidates.append((sc, a.copy()))
            if sc < overall_best_score:
                overall_best_score, overall_best_seq = sc, a.copy()

        # sparse random seeds
        for _ in range(4):
            if time.time() >= deadline:
                break
            a = _distribution_random_sparse(n, rng, density=0.006)
            sc = _objective(a)
            top_candidates.append((sc, a.copy()))
            if sc < overall_best_score:
                overall_best_score, overall_best_seq = sc, a.copy()

        # keep at most 40 best seeds overall
        top_candidates.sort(key=lambda x: x[0])
        top_candidates = top_candidates[:40]

    # ------------------------------------------------------------------
    # 2.  Include the previous construction if provided (height_sequence_1)
    # ------------------------------------------------------------------
    try:
        prev = np.asarray(height_sequence_1, dtype=np.float64)  # type: ignore[name-defined]
        if prev.ndim == 1 and prev.size > 0:
            prev = np.maximum(prev, 0.0)
            if prev.sum() == 0.0:
                prev = _distribution_uniform(prev.size)
            else:
                prev = prev / prev.sum()
            sc = _objective(prev)
            if sc < overall_best_score:
                overall_best_score, overall_best_seq = sc, prev.copy()
                print(f"[{time.time():.1f}] previous construction score={sc:.6f}")
            top_candidates.append((sc, prev.copy()))
    except Exception:

    # ------------------------------------------------------------------
    # 3.  Fallback – should never happen

    if overall_best_seq is None:

    # ------------------------------------------------------------------

    # ------------------------------------------------------------------

    print(
        f"Final best objective (2·n·max(conv)) = {overall_best_score:.6f} (length={len(overall_best_seq)})"
    )
    # Return as plain Python list – evaluate_sequence will do its own clipping/normalisation.
    return overall_best_seq.tolist()
