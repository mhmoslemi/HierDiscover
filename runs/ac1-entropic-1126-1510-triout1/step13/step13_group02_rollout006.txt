def _binary_search_refine(a, deadline):
    n = len(a)
    conv = _fft_self_convolve(a)
    M = conv.max()
    A_ub = _feasibility_lp_matrix(a)
    b_ub_base = np.full(2*n-1, M)
    # set up equality constraint: sum(b)=1
    A_eq = np.ones((1, n))
    b_eq = np.array([1.0])
    # Use scipy.optimize.linprog with objective = zeros
    c = np.zeros(n)
    best_a = a.copy()
    best_M = M
    low = 0.0
    high = M
    for _ in range(30):
        if time.time() >= deadline:
            break
        mid = (low + high) / 2
        # Solve feasibility with upper bound mid
        # Slight slack to handle numerical tolerance
        b_ub = np.full(2*n-1, mid + 1e-12)
        res = linprog(c, A_ub=A_ub, b_ub=b_ub,
                      A_eq=A_eq, b_eq=b_eq,
                      bounds=[(0, None)] * n,
                      method='highs', options={'presolve': True})
        if res.success:
            # feasible -> improve high
            candidate = np.maximum(res.x, 0.0)
            # re-normalize sum if needed
            s = candidate.sum()
            if s > 0:
                candidate /= s
            best_a = candidate
            best_M = mid
            high = mid
        else:
            low = mid
    return best_a, best_M

"""
step_function_search_v2.py

A 1000‑second optimiser for the “max‑convolution” objective used by
``evaluate_sequence``.  The method builds on the previous EG‑+‑annealing
pipeline but adds a *feasibility LP refinement* that directly lowers the
peak of the self‑convolution.  The workflow is:

1. **Diverse seeds** (uniform, arcsine, Hann, edge‑weighted,
   generalized‑normal, Dirichlet, sparse) of moderate length (n≈2000).

2. **Short exponentiated‑gradient polishing** of every seed.

3. **Long EG run** on the current global best.

4. **LP binary‑search refinement**:
   – Build the linear convolution matrix for the current vector.
   – Repeatedly solve a feasibility LP (via SciPy highs) that
     enforces ``a*a ≤ M`` with a lower bound on ``M``.
   – The smallest feasible ``M`` yields a new vector with a strictly
     smaller maximal convolution entry.

5. **Upsample** the refined vector (up to ~8000 entries) and finish
   with another EG phase.

6. **Final simulated‑annealing** to escape any remaining local optimum.

All steps respect a hard 1000 s deadline; the routine returns a plain
``list[float]`` ready for ``evaluate_sequence``.
"""

import time
import math
import numpy as np
from scipy.optimize import linprog  # LP solver (HiGHS)

# ----------------------------------------------------------------------
# Simplex utilities
# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Euclidean projection onto the simplex {x≥0, Σx=s}."""
    if s < 0:
        raise ValueError("simplex radius must be non‑negative")
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.where(u - cssv / (np.arange(1, len(v) + 1)) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# FFT‑based self‑convolution (full linear convolution)
# ----------------------------------------------------------------------
def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    n = a.shape[0]
    size = 1 << ((2 * n - 1) - 1).bit_length()
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[: 2 * n - 1]


# ----------------------------------------------------------------------
# Exact objective (sum fixed to 1 ⇒ denominator = 1)
# ----------------------------------------------------------------------
def _objective(a: np.ndarray) -> float:
    """Return the value of ``evaluate_sequence`` for a probability vector."""
    n = a.shape[0]
    max_b = _fft_self_convolve(a).max()
    if max_b < 0:
        max_b = 0.0
    return float(2.0 * n * max_b)


# ----------------------------------------------------------------------
# Analytic families (all return a probability vector, Σ=1)
# ----------------------------------------------------------------------
def _distribution_uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _distribution_arcsine(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    vals /= vals.sum()
    return vals


def _distribution_hann(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    w = 0.5 - 0.5 * np.cos(2.0 * np.pi * i / (n - 1))
    w_sum = w.sum()
    if w_sum == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w_sum
    return w


def _distribution_edge_weighted(n: int, p: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64) + 0.5
    j = (n - np.arange(n, dtype=np.float64) - 0.5)
    w = (i * j) ** (-p)
    w_sum = w.sum()
    if w_sum == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w_sum
    return w


def _distribution_generalized_normal(n: int, q: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64) + 0.5
    x = i / n - 0.5
    w = np.exp(- (np.abs(x) / 0.5) ** q)
    w_sum = w.sum()
    if w_sum == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w_sum
    return w


def _distribution_random_sparse(n: int, rng: np.random.Generator, density: float = 0.01) -> np.ndarray:
    k = max(2, int(n * density))
    idx = rng.choice(n, size=k, replace=False)
    vals = np.zeros(n, dtype=np.float64)
    vals[idx] = rng.random(k)
    if vals.sum() == 0.0:
        vals[0] = 1.0
    vals /= vals.sum()
    return vals


def _mix_with_uniform(base: np.ndarray, lam: float) -> np.ndarray:
    n = base.shape[0]
    uniform = np.full(n, 1.0 / n, dtype=np.float64)
    return lam * uniform + (1.0 - lam) * base


# ----------------------------------------------------------------------
# Gradient of the active convolution entry (sub‑gradient)
# ----------------------------------------------------------------------
def _grad_of_max_conv(a: np.ndarray) -> tuple[np.ndarray, int]:
    """
    Gradient of the currently maximal convolution entry.
    Returns (grad, k) where k is the index attaining the maximum.
    """
    n = a.shape[0]
    conv = _fft_self_convolve(a)
    k = int(np.argmax(conv))

    grad = np.zeros_like(a)
    if k < n:
        grad[: k + 1] = 2.0 * a[k::-1]
    else:
        start = k - (n - 1)
        grad[start:] = 2.0 * a[start:][::-1]
    return grad, k


# ----------------------------------------------------------------------
# Exponentiated‑gradient descent on the exact objective
# ----------------------------------------------------------------------
def _gradient_descent(
    a: np.ndarray,
    deadline: float,
    best_seq: np.ndarray,
    best_score: float,
    eta0: float = 0.9,
    eval_every: int = 60,
) -> tuple[np.ndarray, float]:
    """
    Mirror‑descent (exponentiated gradient) on the exact objective.
    Update: a ← a·exp(‑η·grad) then ℓ₁‑normalise.
    """
    n = a.shape[0]
    t = 0
    while time.time() < deadline:
        t += 1
        grad, _ = _grad_of_max_conv(a)
        eta = eta0 / math.sqrt(t)
        a = a * np.exp(-eta * grad)            # stays non‑negative
        s = a.sum()
        if s == 0.0:
            a = np.full(n, 1.0 / n, dtype=np.float64)
        else:
            a = a / s

        if t % eval_every == 0:
            cur = _objective(a)
            if cur < best_score - 1e-14:
                best_score = cur
                best_seq = a.copy()
                print(f"[{time.time():.1f}] GD it={t} new global best {best_score:.6f}")
    return best_seq, best_score


# ----------------------------------------------------------------------
# Simulated‑annealing style pairwise mass transfers
# ----------------------------------------------------------------------
def _pairwise_annealing(
    a: np.ndarray,
    deadline: float,
    best_seq: np.ndarray,
    best_score: float,
    rng: np.random.Generator,
) -> tuple[np.ndarray, float]:
    """
    Randomly transfer a fraction of mass between two bins.
    Moves that improve the objective are always accepted; early on
    deteriorating moves can be accepted with a temperature‑dependent
    probability.
    """
    n = a.shape[0]
    start_time = time.time()
    total_time = deadline - start_time
    while time.time() < deadline:
        src_candidates = np.where(a > 1e-12)[0]
        if src_candidates.size == 0:
            break
        i = rng.choice(src_candidates)

        j = rng.integers(n - 1)
        if j >= i:
            j += 1

        delta = a[i] * rng.random() * 0.4
        if delta < 1e-16:
            continue

        new_a = a.copy()
        new_a[i] -= delta
        new_a[j] += delta

        cur = _objective(new_a)

        elapsed = time.time() - start_time
        temperature = max(0.0, 1.0 - elapsed / total_time) * best_score * 0.01

        accept = False
        if cur < best_score - 1e-14:
            accept = True
        elif temperature > 0.0:
            prob = math.exp(-(cur - best_score) / temperature)
            if rng.random() < prob:
                accept = True

        if accept:
            a = new_a
            if cur < best_score - 1e-14:
                best_score = cur
                best_seq = a.copy()
                print(f"[{time.time():.1f}] anneal i={i}→j={j} δ={delta:.3e} new best {best_score:.6f}")
    return best_seq, best_score


# ----------------------------------------------------------------------
# Upsampling (linear interpolation while preserving the simplex)
# ----------------------------------------------------------------------
def _upsample(a: np.ndarray, new_n: int) -> np.ndarray:
    """Interpolate a probability vector to a larger length."""
    old_n = a.shape[0]
    if new_n == old_n:
        return a.copy()
    x_old = (np.arange(old_n, dtype=np.float64) + 0.5) / old_n
    x_new = (np.arange(new_n, dtype=np.float64) + 0.5) / new_n
    a_new = np.interp(x_new, x_old, a, left=0.0, right=0.0)
    a_new = np.maximum(a_new, 0.0)
    s = a_new.sum()
    if s == 0.0:
        a_new[:] = 1.0 / new_n
    else:
        a_new /= s
    return a_new


# ----------------------------------------------------------------------
# Convolution constraint matrix (full linear constraints a * b ≤ M)
# ----------------------------------------------------------------------
def _conv_constraint_matrix(a: np.ndarray) -> np.ndarray:
    """
    Build the (2·n‑1) × n matrix C such that (C @ b) = a * b (full linear convolution).
    """
    n = a.shape[0]
    rows = 2 * n - 1
    C = np.zeros((rows, n), dtype=np.float64)
    for i in range(n):
        C[i:i + n, i] = a
    return C


# ----------------------------------------------------------------------
# Feasibility LP refinement (binary search on the maximal convolution)
# ----------------------------------------------------------------------
def _lp_binary_refine(
    a: np.ndarray,
    deadline: float,
    rng: np.random.Generator,
) -> tuple[np.ndarray, float]:
    """
    Binary‑search the smallest *M* for which a feasible vector b (Σb=1, b≥0,
    a*b ≤ M) exists.  Returns the feasible b attaining the smallest found M
    and that M value.
    """
    n = a.shape[0]
    conv = _fft_self_convolve(a)
    M_current = conv.max()
    # Build constant parts of the LP
    C = _conv_constraint_matrix(a)               # shape (2n‑1, n)
    A_eq = np.ones((1, n), dtype=np.float64)
    b_eq = np.array([1.0], dtype=np.float64)
    c = np.zeros(n, dtype=np.float64)           # dummy objective (feasibility)

    best_b = a.copy()
    best_M = M_current

    low = 0.0
    high = M_current
    # up to 30 iterations or until time runs out
    for it in range(30):
        if time.time() >= deadline:
            break
        mid = (low + high) * 0.5
        # Slight slack to avoid numerical infeasibility
        b_ub = np.full(2 * n - 1, mid + 1e-12, dtype=np.float64)
        # Solve feasibility LP
        res = linprog(
            c,
            A_ub=C,
            b_ub=b_ub,
            A_eq=A_eq,
            b_eq=b_eq,
            bounds=[(0.0, None)] * n,
            method="highs",
            options={"presolve": True},
        )
        if res.success:
            # feasible → we can lower the upper bound
            candidate = np.maximum(res.x, 0.0)
            s = candidate.sum()
            if s > 0:
                candidate /= s
            best_b = candidate
            best_M = mid
            high = mid
            # tiny improvement message
            print(f"[{time.time():.1f}] LP feasible at M={mid:.6e} (it={it})")
        else:
            low = mid
    return best_b, best_M
