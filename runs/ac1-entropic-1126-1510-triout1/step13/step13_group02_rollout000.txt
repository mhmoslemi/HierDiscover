def _lp_optimize(a, M, max_solve_time=None):
    n = len(a)
    # Build matrix Mmat shape (2n-1, n) with a placed in each column.
    Mmat = np.zeros((2*n-1, n), dtype=np.float64)
    for j in range(n):
        Mmat[j:j+n, j] = a
    # Solve LP
    b = cp.Variable(n)
    constraints = [b >= 0, Mmat @ b <= M]
    objective = cp.Maximize(cp.sum(b))
    prob = cp.Problem(objective, constraints)
    prob.solve(solver='ECOS', warm_start=True, max_iters=5000)
    if b.value is None:
        # fallback: return a (previous)
        return a
    b_opt = np.maximum(b.value, 0.0)
    total = b_opt.sum()
    if total == 0:
        return a
    b_norm = b_opt / total
    return b_norm
```

"""
step_function_search_v2.py

Search for a non‑negative step function that minimises
    evaluate_sequence(seq) = 2·n·max(conv(seq,seq)) / (∑seq)²

The algorithm proceeds in three stages:

1.  Small‑scale exploration (n ≈ 1024).  A large collection of analytic
    families (uniform, arcsine, Hann, edge‑weighted, geometric, beta,
    generalized‑normal, Dirichlet, sparse) produces many seeds.
    The best seeds are refined by a *linear‑program* that maximises the
    total mass under the current peak‑convolution bound, followed by a
    short exponentiated‑gradient descent.

2.  Upsampling.  The best small‑scale vector is linearly interpolated to a
    larger length (≈ 4096–8192) and polished again with a longer gradient
    descent.

3.  Final polishing.  A pairwise mass‑transfer simulated annealing
    explores the neighbourhood of the current best solution.

All work respects the 1000 s budget (a 2 s safety margin).  The returned
value is a plain ``list[float]`` ready for ``evaluate_sequence``.
"""

import math
import numpy as np

# ----------------------------------------------------------------------
# Simplex projection (used by gradient descent if needed)
# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x ≥ 0, Σx = s}."""
    if s < 0:
        raise ValueError("simplex radius must be non‑negative")
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.where(u - cssv / (np.arange(1, len(v) + 1)) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Convolution helpers (FFT based, full linear convolution)
# ----------------------------------------------------------------------
def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Full linear self‑convolution a*a using a real FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1) - 1).bit_length()   # next power of two
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[: 2 * n - 1]


def _fft_cross_convolve(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full linear convolution a*b using FFT (both real)."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1) - 1).bit_length()
    A = np.fft.rfft(a, n=size)
    B = np.fft.rfft(b, n=size)
    conv = np.fft.irfft(A * B, n=size)
    return conv[: 2 * n - 1]


# ----------------------------------------------------------------------
# Exact objective for a probability vector (sum = 1)
# ----------------------------------------------------------------------
def _objective_norm1(a: np.ndarray) -> float:
    """Return 2·n·max(conv(a,a)) for a with Σa = 1."""
    n = a.shape[0]
    max_b = _fft_self_convolve(a).max()
    return float(2.0 * n * max_b)


# ----------------------------------------------------------------------
# Analytic families (all return a probability vector, Σ = 1)
# ----------------------------------------------------------------------
def _distribution_uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _distribution_arcsine(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    vals /= vals.sum()
    return vals


def _distribution_hann(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    w = 0.5 - 0.5 * np.cos(2.0 * np.pi * i / (n - 1))
    w_sum = w.sum()
    if w_sum == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w_sum
    return w


def _distribution_edge_weighted(n: int, p: float) -> np.ndarray:
    """Weight ∝ (i·j)^{‑p} where i,j are distances to the two edges."""
    i = np.arange(n, dtype=np.float64) + 0.5
    j = (n - np.arange(n, dtype=np.float64) - 0.5)
    w = (i * j) ** (-p)
    w_sum = w.sum()
    if w_sum == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w_sum
    return w


def _distribution_geometric(n: int, rho: float) -> np.ndarray:
    """Symmetric geometric decay a_i ∝ ρ**i + ρ**(n‑1‑i) (ρ ∈ (0,1))."""
    i = np.arange(n, dtype=np.float64)
    w = rho ** i + rho ** (n - 1 - i)
    w_sum = w.sum()
    if w_sum == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w_sum
    return w


def _distribution_beta(n: int, alpha: float, beta: float) -> np.ndarray:
    """Beta‑like shape a_i ∝ x^{α} (1‑x)^{β},  x∈(0,1)."""
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    w = np.power(x, max(alpha, -0.999)) * np.power(1.0 - x, max(beta, -0.999))
    w_sum = w.sum()
    if w_sum == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w_sum
    return w


def _distribution_generalized_normal(n: int, q: float) -> np.ndarray:
    """Density ∝ exp(‑|x‑½|^{q}) on [0,1]."""
    i = np.arange(n, dtype=np.float64) + 0.5
    x = i / n - 0.5
    w = np.exp(- (np.abs(x) / 0.5) ** q)
    w_sum = w.sum()
    if w_sum == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w_sum
    return w


def _distribution_random_sparse(n: int, rng: np.random.Generator,
                               density: float = 0.01) -> np.ndarray:
    """Sparse random vector with a given density."""
    k = max(2, int(n * density))
    idx = rng.choice(n, size=k, replace=False)
    vals = np.zeros(n, dtype=np.float64)
    vals[idx] = rng.random(k)
    if vals.sum() == 0.0:
        vals[0] = 1.0
    vals /= vals.sum()
    return vals


def _mix_with_uniform(base: np.ndarray, lam: float) -> np.ndarray:
    """Convex combination with the uniform distribution."""
    n = base.shape[0]
    uniform = np.full(n, 1.0 / n, dtype=np.float64)
    return lam * uniform + (1.0 - lam) * base


# ----------------------------------------------------------------------
# Sub‑gradient of the maximal convolution entry
# ----------------------------------------------------------------------
def _grad_of_max_conv(a: np.ndarray) -> np.ndarray:
    """
    Sub‑gradient of the currently maximal entry of a*a.
    For maximizing a*a entry k, the gradient entry i is:
        2 * a_{k-i}  (if 0≤k-i<n) else 0.
    """
    n = a.shape[0]
    conv = _fft_self_convolve(a)
    k = int(np.argmax(conv))

    grad = np.zeros_like(a)
    if k < n:
        # overlap length = k+1
        grad[: k + 1] = 2.0 * a[k::-1]
    else:
        start = k - (n - 1)
        grad[start:] = 2.0 * a[start:][::-1]
    return grad


# ----------------------------------------------------------------------
# Exponentiated‑gradient (mirror descent) on the exact objective
# ----------------------------------------------------------------------
def _exp_gradient_descent(a: np.ndarray,
                         deadline: float,
                         best_seq: np.ndarray,
                         best_score: float,
                         eta0: float = 0.9,
                         eval_every: int = 80) -> tuple[np.ndarray, float]:
    """
    Mirror‑descent a ← a·exp(‑η·grad) followed by ℓ₁‑normalisation.
    Returns the best sequence encountered during the run.
    """
    n = a.shape[0]
    t = 0
    while time.time() < deadline:
        t += 1
        grad = _grad_of_max_conv(a)
        eta = eta0 / math.sqrt(t)
        a = a * np.exp(-eta * grad)  # stays non‑negative
        s = a.sum()
        if s == 0.0:
            a = np.full(n, 1.0 / n, dtype=np.float64)
        else:
            a = a / s
        if t % eval_every == 0:
            cur = _objective_norm1(a)
            if cur < best_score - 1e-14:
                best_score = cur
                best_seq = a.copy()
                print(f"[{time.time():.1f}] GD it={t} new global best {best_score:.6f}")
    return best_seq, best_score


# ----------------------------------------------------------------------
# Linear programming step (maximize total mass under current peak bound)
# ----------------------------------------------------------------------
def _lp_optimize(a: np.ndarray, M: float, max_solve_time: float = 15.0) -> np.ndarray:
    """
    Solve
        max Σ b_j
        s.t. a * b ≤ M (pointwise),  b ≥ 0
    Return the normalized solution b / Σ b (or a unchanged if LP fails).
    """
    if not _CVXPY_AVAILABLE:
        return a
    n = a.shape[0]
    # Build convolution matrix: conv = C @ b,  C_{k,j} = a_{k-j}  (zero if out‑of‑range)
    C = np.zeros((2 * n - 1, n), dtype=np.float64)
    for j in range(n):
        C[j:j + n, j] = a
    # Small epsilon to avoid division‑by‑zero when a contains exact zeros
    a_safe = np.maximum(a, 1e-12)
    # Re‑build matrix with safe a (helps the solver)
    C[:] = 0.0
    for j in range(n):
        C[j:j + n, j] = a_safe

    # CVXPY model
    b = cp.Variable(n, nonneg=True)
    constraints = [C @ b <= M + 1e-12]   # tiny slack for numerical stability
    objective = cp.Maximize(cp.sum(b))
    prob = cp.Problem(objective, constraints)

    try:
        prob.solve(solver='ECOS', warm_start=True,
                   max_iters=5000, # ECOS respects max_iters
                   abstol=1e-8, reltol=1e-8, feastol=1e-8)
    except Exception:
        # any solver failure → fall back to current a
        return a

    if b.value is None:
        return a
    b_opt = np.maximum(b.value, 0.0)
    total = b_opt.sum()
    if total == 0.0:
        return a
    return b_opt / total


# ----------------------------------------------------------------------
# Upsampling (linear interpolation while preserving the simplex)
# ----------------------------------------------------------------------
def _upsample(a: np.ndarray, new_n: int) -> np.ndarray:
    """Interpolate a probability vector to a larger length."""
    old_n = a.shape[0]
    if new_n == old_n:
        return a.copy()
    x_old = (np.arange(old_n, dtype=np.float64) + 0.5) / old_n
    x_new = (np.arange(new_n, dtype=np.float64) + 0.5) / new_n
    a_new = np.interp(x_new, x_old, a, left=0.0, right=0.0)
    a_new = np.maximum(a_new, 0.0)
    s = a_new.sum()
    if s == 0.0:
        a_new[:] = 1.0 / new_n
    else:
        a_new /= s
    return a_new


# ----------------------------------------------------------------------
# Simulated‑annealing style pairwise mass transfers
# ----------------------------------------------------------------------
def _pairwise_annealing(a: np.ndarray,
                        deadline: float,
                        best_seq: np.ndarray,
                        best_score: float,
                        rng: np.random.Generator) -> tuple[np.ndarray, float]:
    """
    Randomly transfer a fraction of mass between two bins.
    Moves that improve the objective are always accepted; deteriorating
    moves may be accepted with a temperature‑dependent probability.
    """
    n = a.shape[0]
    start_time = time.time()
    total_time = deadline - start_time
    while time.time() < deadline:
        # pick source with non‑negligible mass
        src_candidates = np.where(a > 1e-12)[0]
        if src_candidates.size == 0:
            break
        i = rng.choice(src_candidates)

        # pick a distinct target
        j = rng.integers(n - 1)
        if j >= i:
            j += 1

        # transfer a random fraction of source mass
        delta = a[i] * rng.random() * 0.4
        if delta < 1e-16:
            continue

        new_a = a.copy()
        new_a[i] -= delta
        new_a[j] += delta

        cur = _objective_norm1(new_a)

        elapsed = time.time() - start_time
        temperature = max(0.0, 1.0 - elapsed / total_time) * best_score * 0.01

        accept = False
        if cur < best_score - 1e-14:
            accept = True
        elif temperature > 0.0:
            prob = math.exp(-(cur - best_score) / temperature)
            if rng.random() < prob:
                accept = True

        if accept:
            a = new_a
            if cur < best_score - 1e-14:
                best_score = cur
                best_seq = a.copy()
                print(
                    f"[{time.time():.1f}] anneal i={i}→j={j} δ={delta:.3e} new best {best_score:.6f}"
                )
    return best_seq, best_score

def propose_candidate(seed: int = 42, budget_s: int = 1000) -> list[float]:

    deadline = start + budget_s - 2.0  # safety margin

    # ------------------------------------------------------------------
    # 1.  Small‑scale (n≈1024) seeding
    # ------------------------------------------------------------------
    n_small = 1024
    candidates: list[tuple[float, np.ndarray]] = []

    def _add_candidate(v: np.ndarray, label: str = "") -> None:
        """Normalise v to sum = 1, compute the objective and store."""
        if v.ndim != 1:
            return

        s = v.sum()
        if s == 0.0:
            v = _distribution_uniform(v.shape[0])
        else:
            v = v / s
        score = _objective_norm1(v)
        candidates.append((score, v))
        # keep a tiny log for debugging
        # print(f"seed {label} score={score:.6f}")

    # deterministic families
    _add_candidate(_distribution_uniform(n_small), "uniform")
    _add_candidate(_distribution_arcsine(n_small), "arcsine")
    _add_candidate(_distribution_hann(n_small), "hann")

    # edge‑weighted grid
    p_vals = np.concatenate((
        np.logspace(-3.0, -0.5, 12, base=10),
        np.logspace(-0.5, 0.0, 12, base=10),
        np.linspace(0.1, 2.0, 12),
        np.logspace(0.0, 0.7, 8, base=10)

    lam_vals = np.linspace(0.0, 1.0, 9)
    for p in p_vals:
        base = _distribution_edge_weighted(n_small, float(p))
        for lam in lam_vals:
            cand = _mix_with_uniform(base, float(lam))
            _add_candidate(cand, f"edge p={p:.3g} λ={lam:.2f}")

    # geometric decay family
    for rho in np.linspace(0.05, 0.95, 13):
        cand = _distribution_geometric(n_small, float(rho))
        _add_candidate(cand, f"geo rho={rho:.2f}")

    for a in alphas:
        for b in betas:

            _add_candidate(cand, f"beta a={a:.2f} b={b:.2f}")

    # generalized normal (q)
    for q in np.linspace(0.3, 4.0, 14):
        cand = _distribution_generalized_normal(n_small, float(q))

    for _ in range(20):

        alpha = 10 ** log_alpha
        cand = rng.dirichlet(np.full(n_small, alpha))
        _add_candidate(cand, f"dirichlet α={alpha:.3g}")

    # Sparse random seeds

    except Exception:

    overall_best_seq = top_seeds[0].copy()

            print(f"[{time.time():.1f}] upsampleed new best {overall_best_score:.6f}")

            overall_best_seq.copy(),
            sub_dead,

            best_score=overall_best_score,

            eval_every=60,
        )
        if sc_opt < overall_best_score - 1e-12:
            overall_best_seq = a_opt.copy()
            overall_best_score = sc_opt
            print(f"[{time.time():.1f}] post‑upsample GD improved best to {overall_best_score:.6f}")

    # ------------------------------------------------------------------
    # 4.  Final simulated‑annealing polishing
    # ------------------------------------------------------------------
    if time.time() < deadline:
        print(f"[{time.time():.1f}] final annealing phase")
        overall_best_seq, overall_best_score = _pairwise_annealing(
            overall_best_seq.copy(),
            deadline,

            best_score=overall_best_score,
            rng=rng,
        )

    # ------------------------------------------------------------------
    # 5.  Return the result
    # ------------------------------------------------------------------

        f"Final best score (objective=2·n·max(conv)) = {overall_best_score:.6f} "
        f"(length={len(overall_best_seq)})"
    )
    return overall_best_seq.tolist()
