def _smooth_gradient_descent_momentum(a_init, deadline, best_seq, best_score,
                                      lr0=0.5, beta=0.9, gamma=30.0,
                                      eval_interval=200, patience=2000):
    n = a_init.shape[0]
    # initialize x log probabilities
    eps = 1e-12
    x = np.log(a_init + eps)
    # zero-mean adjust
    x = x - np.max(x)
    momentum = np.zeros_like(x)
    t = 0
    no_improve = 0
    rng = np.random.default_rng()
    while time.time() < deadline and t < 200_000:
        t += 1
        # compute a via softmax
        # for numerical stability, subtract max of x
        max_x = np.max(x)
        ex = np.exp(x - max_x)
        a = ex / ex.sum()
        # compute full convolution a*a via FFT
        conv = _fft_self_convolve(a)  # length 2n-1
        cmax = conv.max()
        # compute softmax weights w
        # use gamma
        # compute exp(gamma*(conv - cmax)) stable
        exp_term = np.exp(gamma * (conv - cmax))
        w = exp_term / exp_term.sum()
        # compute gradient G wrt a
        L = len(w)  # 2n - 1
        w_rev = w[::-1]
        # compute FFT size for convolution a * w_rev
        size = 1 << ((n + L - 1 - 1).bit_length())
        A = np.fft.rfft(a, n=size)
        W = np.fft.rfft(w_rev, n=size)
        conv_aw = np.fft.irfft(A * W, n=size)[:n + L - 1]  # length 3n-2
        # G_i index
        idx = (L - 1) - np.arange(n)  # array of indices
        G = 2.0 * conv_aw[idx]
        # gradient wrt x
        mu = np.dot(a, G)
        grad_x = a * (G - mu)
        # momentum update
        momentum = beta * momentum + (1 - beta) * grad_x
        # decaying learning rate
        eta = lr0 / math.sqrt(t)
        # update x (gradient descent)
        x = x - eta * momentum
        # maintain stability: subtract max
        if np.max(x) > 100:
            x = x - np.max(x)
        # periodic exact evaluation
        if t % eval_interval == 0:
            cur_obj = _objective(a)
            if cur_obj < best_score - 1e-14:
                best_score = cur_obj
                best_seq = a.copy()
                no_improve = 0
                print(f"[{time.time():.1f}] it={t:,} new best exact {best_score:.6f}")
            else:
                no_improve += 1
        # jitter for stagnation
        if no_improve >= patience:
            # add small Gaussian jitter to x
            jitter = rng.normal(scale=0.01, size=n)
            x = x + jitter
            no_improve = 0
    return best_seq, best_score

"""
Propose a step‑function (non‑negative heights) with a low value of
``evaluate_sequence``.  

The algorithm proceeds in three phases

  1️⃣  *Seed generation* – a modest collection of analytic families,
      Dirichlet samples and a possible previous best is created for a
      single length (≈3500).  The 5 best seeds (by the exact objective)
      are kept.

  2️⃣  *Smooth‑max gradient descent* – the exact max in the objective
      is replaced by a soft‑max (parameter γ).  Using the log‑softmax
      parametrisation ``a = softmax(x)`` we obtain a smooth gradient that
      can be optimised with exponentiated‑gradient / momentum.  Each
      iteration uses two FFTs (O(n log n)).  The best exact value seen
      is stored.

  3️⃣  *Local polishing* – a cheap greedy pairwise swap followed by a
      very cheap simulated‑annealing style search (the same routines
      used in the reference code) clean up the final solution.

All numbers are non‑negative and sum to one; the returned list can be
fed directly into ``evaluate_sequence``.  The routine respects the
given time budget (default 1000 s) and prints progress.
"""

import time
import math
import numpy as np
from numpy.fft import rfft, irfft

# ----------------------------------------------------------------------
# Simple utilities ------------------------------------------------------
def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Full self‑convolution a*a (length 2·n‑1) using a real FFT."""
    n = a.shape[0]
    need = 2 * n - 1
    size = 1 << ((need - 1).bit_length())
    A = rfft(a, n=size)
    conv = irfft(A * A, n=size)
    return conv[:need]


def _objective(a: np.ndarray) -> float:
    """Exact objective 2·n·max(a*a) under the normalisation Σa = 1."""
    n = a.shape[0]
    b = _fft_self_convolve(a)
    return 2.0 * n * b.max()


# ----------------------------------------------------------------------
# Analytic families ----------------------------------------------------
def _distribution_uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _distribution_arcsine(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    vals /= vals.sum()
    return vals


def _distribution_beta_sym(n: int, p: float) -> np.ndarray:
    """Symmetric Beta(p,p) on the discrete grid."""
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    w = np.exp((p - 1) * np.log(x) + (p - 1) * np.log(1.0 - x))
    s = w.sum()
    if s == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= s
    return w


def _distribution_edge_weighted(n: int, p: float) -> np.ndarray:
    """U‑shaped: w_i ∝ ((i+0.5)*(n‑i‑0.5))^{−p}."""
    i = np.arange(n, dtype=np.float64) + 0.5
    j = (n - np.arange(n, dtype=np.float64) - 0.5)
    w = (i * j) ** (-p)
    s = w.sum()
    if s == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= s
    return w


def _distribution_edge_inversepow(n: int, p: float) -> np.ndarray:
    """Second edge family: w_i ∝ (i+0.5)^{−p} + (n−i−0.5)^{−p}."""
    i = np.arange(n, dtype=np.float64) + 0.5
    j = (n - np.arange(n, dtype=np.float64) - 0.5)
    w = i ** (-p) + j ** (-p)
    s = w.sum()
    if s == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= s
    return w


def _distribution_sine_weighted(n: int, p: float) -> np.ndarray:
    """w_i ∝ sin(π(i+½)/n)^p, vanishes at the ends."""
    theta = np.pi * (np.arange(n, dtype=np.float64) + 0.5) / n
    w = np.sin(theta) ** p
    s = w.sum()
    if s == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= s
    return w


def _distribution_tent(n: int) -> np.ndarray:
    """Symmetric triangular shape."""
    i = np.arange(n, dtype=np.float64)
    half = (n - 1) / 2.0
    w = np.where(i <= half, i + 1, n - i)
    w /= w.sum()
    return w


def _distribution_quadratic(n: int) -> np.ndarray:
    """Quadratic bell (∝ 1/(distance+1)^2)."""
    i = np.arange(n, dtype=np.float64)
    half = (n - 1) / 2.0
    d = np.abs(i - half) + 1.0
    w = 1.0 / (d ** 2)
    w /= w.sum()
    return w


def _distribution_dirichlet(n: int, rng: np.random.Generator) -> np.ndarray:
    """Random Dirichlet with a random concentration in (0.01,2)."""
    alpha = rng.uniform(0.01, 2.0)
    a = rng.dirichlet(np.full(n, alpha))
    return a.astype(np.float64)


# ----------------------------------------------------------------------
# Pairwise greedy transfer ---------------------------------------------
def _pairwise_swap_local(a: np.ndarray,
                         deadline: float,
                         best_seq: np.ndarray,
                         best_score: float,
                         max_trials: int | None = None) -> tuple[np.ndarray, float]:
    """Randomly move a fraction of mass from i→j if the exact objective improves."""
    n = a.shape[0]
    rng = np.random.default_rng()
    trials = 0
    while time.time() < deadline:
        if max_trials is not None and trials >= max_trials:
            break
        trials += 1
        src = np.where(a > 1e-12)[0]
        if src.size == 0:
            break
        i = rng.choice(src)
        j = rng.integers(n - 1)
        if j >= i:
            j += 1
        delta = a[i] * rng.random() * 0.35
        if delta < 1e-14:
            continue
        new_a = a.copy()
        new_a[i] -= delta
        new_a[j] += delta
        cur = _objective(new_a)
        if cur < best_score - 1e-14:
            a = new_a
            best_score = cur
            best_seq = a.copy()
            print(f"[{time.time():.1f}] swap {i}->{j} Δ={delta:.3e} new best {best_score:.6f}")
    return best_seq, best_score


# ----------------------------------------------------------------------
# Very cheap simulated‑annealing style local search --------------------
def _random_local_search(a: np.ndarray,
                         deadline: float,
                         best_seq: np.ndarray,
                         best_score: float,
                         init_temp: float = 0.008,
                         decay: float = 0.9997) -> tuple[np.ndarray, float]:
    """Repeat random mass transfers; accept worsening moves with a tiny temperature."""
    n = a.shape[0]
    rng = np.random.default_rng()
    T = init_temp
    while time.time() < deadline:
        i = rng.integers(n)
        j = rng.integers(n - 1)
        if j >= i:
            j += 1
        delta = a[i] * rng.random() * 0.3
        if delta < 1e-14:
            continue
        new_a = a.copy()
        new_a[i] -= delta
        new_a[j] += delta
        cur = _objective(new_a)
        if cur < best_score - 1e-14:
            a = new_a
            best_score = cur
            best_seq = a.copy()
            print(f"[{time.time():.1f}] SA improve {i}->{j} Δ={delta:.3e} best {best_score:.6f}")
        else:
            if rng.random() < math.exp(-(cur - best_score) / max(T, 1e-12)):
                a = new_a
        T *= decay
    return best_seq, best_score


# ----------------------------------------------------------------------
# Smooth‑max gradient descent (soft‑max replacement of max) ---------
def _smooth_gradient_descent_momentum(a_init: np.ndarray,
                                      deadline: float,
                                      best_seq: np.ndarray,
                                      best_score: float,
                                      lr0: float = 0.5,
                                      beta: float = 0.9,
                                      gamma: float = 30.0,
                                      eval_interval: int = 200,
                                      patience: int = 2000) -> tuple[np.ndarray, float]:
    """
    Optimises the *soft* max of the self‑convolution using exponentiated‑gradient
    with momentum on the log‑softmax parameters.
    """
    n = a_init.shape[0]
    eps = 1e-12
    x = np.log(a_init + eps)          # log‑softmax parameters
    # keep the parametrisation shift‑invariant
    x = x - np.max(x)

    momentum = np.zeros_like(x)
    t = 0
    no_improve = 0
    rng = np.random.default_rng()

    while time.time() < deadline and t < 150_000:
        t += 1

        # --- a = softmax(x) -------------------------------------------------
        max_x = np.max(x)
        ex = np.exp(x - max_x)
        a = ex / ex.sum()

        # --- FFT size (enough for both convolutions) -----------------------
        L = 2 * n - 1
        size = 1 << ((3 * n - 2).bit_length())   # ≥ 3n‑2
        A = rfft(a, n=size)

        # a*a (full) ---------------------------------------------------------
        conv_full = irfft(A * A, n=size)[:L]      # length 2n‑1
        c_max = conv_full.max()

        # soft‑max over convolution entries -----------------------------------
        # w_k = exp(gamma*(c_k - c_max)) / Σ exp(...)
        exp_term = np.exp(gamma * (conv_full - c_max))
        w = exp_term / exp_term.sum()

        # gradient wrt a : G_i = 2 * Σ_j a_j w_{i+j}
        w_rev = w[::-1]                          # length L
        W = rfft(w_rev, n=size)
        conv_aw = irfft(A * W, n=size)[:n + L - 1]   # length 3n‑2
        idx = (L - 1) - np.arange(n)                # indices 2n‑2‑i
        G = 2.0 * conv_aw[idx]

        # gradient wrt x (log‑softmax) --------------------------------------
        mu = np.dot(a, G)
        grad_x = a * (G - mu)

        # momentum update ----------------------------------------------------
        momentum = beta * momentum + (1.0 - beta) * grad_x

        # decaying learning rate --------------------------------------------
        eta = lr0 / math.sqrt(t)

        # gradient step (we minimise)
        x = x - eta * momentum

        # keep numbers bounded (softmax invariant to additive constant)
        if np.max(x) > 100.0:
            x = x - np.max(x)

        # --------------------------------------------------------------------
        # occasional exact evaluation
        if t % eval_interval == 0:
            cur = _objective(a)
            if cur < best_score - 1e-14:
                best_score = cur
                best_seq = a.copy()
                no_improve = 0
                print(f"[{time.time():.1f}] it={t:,} new exact best {best_score:.6f}")
            else:
                no_improve += 1

        # jitter if stuck
        if no_improve >= patience:
            jitter = rng.normal(scale=0.005, size=n)
            x = x + jitter
            no_improve = 0

    return best_seq, best_score

# Main entry point ------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: int = 1000) -> list[float]:
    """
    Search for a non‑negative step‑function that minimises ``evaluate_sequence``.
    The routine respects the given time budget and returns a plain Python list.
    """
    rng = np.random.default_rng(seed)

    # keep a small safety margin for the final polishing steps
    deadline = start + budget_s - 5.0

    n = 3500
    seed_list = []

    # deterministic families
    seed_list.append(_distribution_uniform(n))
    seed_list.append(_distribution_arcsine(n))
    seed_list.append(_distribution_tent(n))
    seed_list.append(_distribution_quadratic(n))

    # parametrised edge families

        seed_list.append(_distribution_edge_weighted(n, p))
        seed_list.append(_distribution_edge_inversepow(n, p))
        seed_list.append(_distribution_beta_sym(n, p))
        seed_list.append(_distribution_sine_weighted(n, p))

    # random Dirichlet seeds
    for _ in range(12):
        seed_list.append(_distribution_dirichlet(n, rng))

    # possible previously discovered construction
    try:
        prev = np.asarray(height_sequence_1, dtype=np.float64)  # type: ignore[name-defined]
        if prev.ndim == 1 and prev.size > 0:
            # adapt length – truncate or pad with zeros
            if prev.size > n:
                prev = prev[:n]
            elif prev.size < n:
                tmp = np.zeros(n, dtype=np.float64)
                tmp[:prev.size] = prev
                prev = tmp
            prev = prev / prev.sum()
            seed_list.append(prev)
            print(f"[{time.time():.1f}] loaded previous construction")
    except Exception:
        pass

    # ------------------------------------------------------------------
    # 2. keep the best few seeds (exact objective)
    top_k = 7
    scored = []
    for a in seed_list:
        obj = _objective(a)
        scored.append((obj, a.copy()))
        if obj < 1.0e30:
            print(f"[{time.time():.1f}] seed score {obj:.6f}")

    scored.sort(key=lambda x: x[0])
    best_seq, best_score = None, float("inf")
    for obj, a in scored[:top_k]:
        if obj < best_score:
            best_score = obj
            best_seq = a.copy()
            print(f"[{time.time():.1f}] initial best {best_score:.6f}")

    if best_seq is None:                     # safeguard
        best_seq = _distribution_uniform(n)
        best_score = _objective(best_seq)

    # ------------------------------------------------------------------
    # 3. smooth‑max gradient descent (majority of the budget)
    if time.time() < deadline:
        remaining = deadline - time.time()
        sub_deadline = time.time() + remaining - 30.0   # leave 30 s for polishing
        print(f"[{time.time():.1f}] start smooth‑max gradient descent "
              f"(≈{remaining - 30:.1f}s left)")

        best_seq, best_score = _smooth_gradient_descent_momentum(
            best_seq.copy(),
            sub_deadline,
            best_seq,
            best_score,
            lr0=0.6,
            beta=0.9,
            gamma=30.0,
            eval_interval=150,
            patience=2500,
        )
        print(f"[{time.time():.1f}] after descent best ≈ {best_score:.6f}")

    # ------------------------------------------------------------------
    # 4. greedy pairwise refinement
    if time.time() < deadline:
        print(f"[{time.time():.1f}] greedy pairwise refinement")
        best_seq, best_score = _pairwise_swap_local(
            best_seq.copy(),
            deadline,
            best_seq,
            best_score,
            max_trials=300_000,
        )

    # 5. final cheap simulated‑annealing search
    if time.time() < deadline:

        best_seq, best_score = _random_local_search(
            best_seq.copy(),
            deadline,
            best_seq,
            best_score,
            init_temp=0.01,
            decay=0.9995,
        )

    # ------------------------------------------------------------------
    print(f"Final best exact score ≈ {best_score:.6f} (length={len(best_seq)})")
    return best_seq.tolist()
