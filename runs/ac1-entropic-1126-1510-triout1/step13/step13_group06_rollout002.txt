def exponential_descent_with_momentum(a, deadline, best_seq, best_score):
    n = len(a)
    grad_mom = np.zeros_like(a)
    beta = 0.9
    t = 0
    tau = tau0
    while time.time() < deadline and t < max_iter:
        t+=1
        conv = _fft_self_convolve(a)
        w = _softmax_weights(conv, tau)
        grad = _weighted_gradient(a, w)
        grad_mom = beta * grad_mom + (1-beta) * grad
        eta = eta0 / math.sqrt(t)
        a = a * np.exp(-eta * grad_mom)

'''
Optimiser for the non‑negative step‑function problem.

The algorithm proceeds in three stages:

1. **Broad candidate generation** – for a range of lengths we evaluate
   a handful of analytic families (uniform, arcsine, symmetric Beta,
   edge‑weighted, sine‐weighted) on a fine grid of the shape parameter.
   The 30 best candidates (lowest exact objective) are kept.

2. **Soft‑max exponential descent with momentum** – each of the top‑10
   seeds is refined by a multiplicative mirror‑descent on the exact
   soft‑max approximation of the max‑convolution.  A momentum term
   smooths the gradients, a simple learning‑rate schedule ensures
   stability, and occasional jitter restarts the optimisation when it
   stalls.

3. **Local refinement** – a cheap pairwise‑swap hill‑climb followed by a
   short simulated‑annealing walk polish the best vector found.

All work respects the supplied time budget (default 1000 s).  The
returned list is a plain Python list; ``evaluate_sequence`` will
normalise it internally.
'''

import time
import math
import numpy as np

# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x ≥ 0, Σx = s}."""
    if s < 0:
        raise ValueError("simplex radius must be non‑negative")
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho_arr = np.where(u - cssv / (np.arange(1, len(v) + 1)) > 0)[0]
    if rho_arr.size == 0:
        theta = 0.0
    else:
        rho = rho_arr[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Full self‑convolution of a (real FFT, length 2n‑1)."""
    n = a.shape[0]
    need = 2 * n - 1
    size = 1 << ((need - 1).bit_length())
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[:need]


# ----------------------------------------------------------------------
def _objective(a: np.ndarray) -> float:
    """Exact objective 2·n·max(conv(a,a)) for a vector with Σa = 1."""
    n = a.shape[0]
    b = _fft_self_convolve(a)
    return float(2.0 * n * b.max())


# ----------------------------------------------------------------------
# Analytic families ----------------------------------------------------
def _distribution_uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _distribution_arcsine(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    vals /= vals.sum()
    return vals


def _distribution_beta_sym(n: int, p: float) -> np.ndarray:
    """Symmetric beta (Beta(p,p))."""
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    w = np.exp((p - 1) * np.log(x) + (p - 1) * np.log(1.0 - x))
    if w.sum() == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _distribution_beta_asym(n: int, a: float, b: float) -> np.ndarray:
    """Asymmetric beta (Beta(a,b))."""
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    w = np.exp((a - 1) * np.log(x) + (b - 1) * np.log(1.0 - x))
    if w.sum() == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _distribution_edge_weighted(n: int, p: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64) + 0.5
    j = (n - np.arange(n, dtype=np.float64) - 0.5)
    w = (i * j) ** (-p)
    if w.sum() == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _distribution_sine_weighted(n: int, p: float) -> np.ndarray:
    theta = np.pi * (np.arange(n, dtype=np.float64) + 0.5) / n
    w = np.sin(theta) ** p
    if w.sum() == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _distribution_random_sparse(
    n: int, rng: np.random.Generator, density: float = 0.01
) -> np.ndarray:
    """Sparse random vector with fixed density of non‑zeros."""
    k = max(2, int(n * density))
    idx = rng.choice(n, size=k, replace=False)
    vals = np.zeros(n, dtype=np.float64)
    vals[idx] = rng.random(k)
    if vals.sum() == 0.0:
        vals[0] = 1.0
    vals /= vals.sum()
    return vals


def _mix_with_uniform(base: np.ndarray, lam: float) -> np.ndarray:
    """Convex combination lam·U + (1‑lam)·base."""
    n = base.shape[0]
    uniform = np.full(n, 1.0 / n, dtype=np.float64)
    return lam * uniform + (1.0 - lam) * base


# ----------------------------------------------------------------------
# Soft‑max utilities ----------------------------------------------------
def _softmax_weights(conv: np.ndarray, tau: float) -> np.ndarray:
    """soft‑max of τ·conv (stable) – returns a probability vector."""
    max_c = conv.max()
    w_raw = np.exp(tau * (conv - max_c))
    total = w_raw.sum()
    if total == 0.0:
        return np.full_like(conv, 1.0 / conv.size)
    return w_raw / total


def _weighted_gradient(a: np.ndarray, w: np.ndarray) -> np.ndarray:
    """Gradient of Σ_i w_i·(a*a)[i]; closed form: 2·(a_rev ⊛ w)."""
    n = a.shape[0]
    a_rev = a[::-1]
    need = len(w) + n - 1          # = 3·n‑2 for full length
    size = 1 << ((need - 1).bit_length())
    A = np.fft.rfft(a_rev, n=size)
    W = np.fft.rfft(w, n=size)
    conv = np.fft.irfft(A * W, n=size)
    conv = conv[:need]
    grad = 2.0 * conv[n - 1 : n - 1 + n]
    return grad


# ----------------------------------------------------------------------
# Exponential descent with momentum --------------------------------------
def _exp_descent_momentum(
    a0: np.ndarray,
    deadline: float,
    best_seq: np.ndarray,
    best_score: float,
    eta0: float = 0.6,
    tau0: float = 0.4,
    tau_mult: float = 1.003,
    eval_interval: int = 200,
    patience: int = 2000,
) -> tuple[np.ndarray, float]:
    """
    Multiplicative mirror descent on the soft‑max surrogate.
    A simple momentum (exponential moving average) smooths the gradient.
    """
    a = a0.copy()
    n = a.shape[0]
    rng = np.random.default_rng()
    t = 0
    no_improve = 0
    beta = 0.9                         # momentum coefficient
    g_mom = np.zeros_like(a)
    tau = tau0

    while time.time() < deadline and t < 300_000:
        t += 1
        conv = _fft_self_convolve(a)
        w = _softmax_weights(conv, tau)
        grad = _weighted_gradient(a, w)

        # momentum update
        g_mom = beta * g_mom + (1.0 - beta) * grad

        # learning‑rate schedule (diminishing)
        eta = eta0 / math.sqrt(t)

        # multiplicative update on the simplex
        a = a * np.exp(-eta * g_mom)
        s = a.sum()
        if s == 0.0:
            a = np.full(n, 1.0 / n, dtype=np.float64)
        else:
            a = a / s

        # temperature schedule – focus more on the true max
        tau *= tau_mult

        # occasional exact evaluation
        if t % eval_interval == 0:
            cur = _objective(a)
            if cur < best_score - 1e-14:
                best_score, best_seq = cur, a.copy()
                no_improve = 0
                print(f"[{time.time():.1f}] descent it={t} new best {best_score:.6f}")
            else:
                no_improve += 1

        # jitter if stuck
        if no_improve >= patience:
            jitter = rng.exponential(scale=0.004, size=n)
            a = a + jitter
            a = np.maximum(a, 0.0)
            a = a / a.sum()
            no_improve = 0

    return best_seq, best_score

# Pairwise swap local refinement -----------------------------------------
def _pairwise_swap_local(
    a0: np.ndarray,
    deadline: float,
    best_seq: np.ndarray,
    best_score: float,
    max_trials: int | None = None,
) -> tuple[np.ndarray, float]:
    """Transfer a small chunk of mass between two positions if it improves."""
    a = a0.copy()
    n = a.shape[0]
    rng = np.random.default_rng()
    trials = 0
    while time.time() < deadline:
        if max_trials is not None and trials >= max_trials:
            break
        trials += 1

        # choose source with non‑negligible mass
        src_candidates = np.where(a > 1e-14)[0]
        if src_candidates.size == 0:
            break
        i = rng.choice(src_candidates)

        # choose a distinct destination
        j = rng.integers(n - 1)
        if j >= i:
            j += 1

        delta = a[i] * rng.random() * 0.33
        if delta < 1e-15:
            continue

        new_a = a.copy()
        new_a[i] -= delta
        new_a[j] += delta

        cur = _objective(new_a)
        if cur < best_score - 1e-14:
            a = new_a
            best_score, best_seq = cur, new_a.copy()
            print(
                f"[{time.time():.1f}] swap i={i}→j={j} δ={delta:.3e} new best {best_score:.6f}"
            )
    return best_seq, best_score

# Simulated‑annealing style walk -----------------------------------------
def _random_local_search(
    a0: np.ndarray,
    deadline: float,
    best_seq: np.ndarray,
    best_score: float,
    init_temp: float = 0.008,
    decay: float = 0.9997,
) -> tuple[np.ndarray, float]:
    """Very cheap SA that repeatedly transfers a random amount of mass."""
    a = a0.copy()
    n = a.shape[0]
    rng = np.random.default_rng()
    T = init_temp
    while time.time() < deadline:
        i = rng.integers(n)
        j = rng.integers(n - 1)
        if j >= i:
            j += 1
        delta = a[i] * rng.random() * 0.3
        if delta < 1e-15:
            continue
        new_a = a.copy()
        new_a[i] -= delta
        new_a[j] += delta
        cur = _objective(new_a)
        if cur < best_score - 1e-14:
            a = new_a
            best_score, best_seq = cur, new_a.copy()
            print(f"[{time.time():.1f}] SA improve {i}->{j} Δ={delta:.3e} best {best_score:.6f}")
        else:
            if rng.random() < math.exp(-(cur - best_score) / max(T, 1e-12)):
                a = new_a
        T *= decay
    return best_seq, best_score

def propose_candidate(seed: int = 42, budget_s: int = 1000) -> list[float]:
    """
    Search for a non‑negative step‑function minimising ``evaluate_sequence``.
    Returns a plain Python list; the caller will apply the final normalisation.
    """
    rng = np.random.default_rng(seed)
    start = time.time()
    deadline = start + budget_s - 3.0          # a small safety margin

    # ------------------------------------------------------------------
    # 1. Broad generation of analytic / random seeds
    # ------------------------------------------------------------------
    overall_best_seq = None
    overall_best_score = float("inf")
    top_candidates: list[tuple[float, np.ndarray]] = []

    # a set of lengths (moderate – FFT is cheap)
    lengths = [
        2000, 2500, 3000, 3500, 4000, 5000,
        6000, 7500, 9000, 11000, 13000, 15000,
    ]

    # shape parameter grid for symmetric beta / edge / sine families
    p_grid = np.concatenate(
        [
            np.logspace(-2, -1, 8),   # 0.01 … 0.1
            np.logspace(-1, 0, 10),   # 0.1 … 1
            np.logspace(0, 0.7, 12),  # 1 … 5
        ]
    )
    lam_grid = np.linspace(0.0, 1.0, 5)  # mixing with uniform

    for n in lengths:
        if time.time() >= deadline:
            break
        print(f"[{time.time():.1f}] exploring length {n}")

        # a) deterministic families
        for base_func in (_distribution_uniform, _distribution_arcsine):
            cand = base_func(n)
            sc = _objective(cand)
            top_candidates.append((sc, cand.copy()))
            if sc < overall_best_score:
                overall_best_score, overall_best_seq = sc, cand.copy()
                print(f"[{time.time():.1f}] new global best len={n} score={sc:.6f}")

        # b) parametrised symmetric families
        for p in p_grid:
            if time.time() >= deadline:
                break
            beta = _distribution_beta_sym(n, p)
            edge = _distribution_edge_weighted(n, p)
            sine = _distribution_sine_weighted(n, p)
            for base in (beta, edge, sine):
                # plain base
                sc = _objective(base)
                top_candidates.append((sc, base.copy()))
                if sc < overall_best_score:
                    overall_best_score, overall_best_seq = sc, base.copy()
                    print(f"[{time.time():.1f}] len={n} p={p:.3g} base score={sc:.6f}")

                # mixing with uniform
                for lam in lam_grid:
                    if time.time() >= deadline:
                        break
                    mixed = _mix_with_uniform(base, lam)
                    sc = _objective(mixed)
                    top_candidates.append((sc, mixed.copy()))
                    if sc < overall_best_score:
                        overall_best_score, overall_best_seq = sc, mixed.copy()
                        print(
                            f"[{time.time():.1f}] len={n} p={p:.3g} lam={lam:.2f} score={sc:.6f}"
                        )

        # c) random asymmetric beta (few samples)
        for _ in range(4):
            if time.time() >= deadline:
                break
            a_param = 10 ** rng.uniform(-2, 0.7)   # 0.01 … ~5
            b_param = 10 ** rng.uniform(-2, 0.7)
            cand = _distribution_beta_asym(n, a_param, b_param)
            sc = _objective(cand)
            top_candidates.append((sc, cand.copy()))
            if sc < overall_best_score:
                overall_best_score, overall_best_seq = sc, cand.copy()
                print(
                    f"[{time.time():.1f}] asym‑beta len={n} a={a_param:.3g} b={b_param:.3g} score={sc:.6f}"
                )

        # d) Dirichlet random vectors (different concentration)
        for alpha in (0.05, 0.2, 0.8, 1.5):
            if time.time() >= deadline:
                break
            cand = rng.dirichlet(np.full(n, alpha))
            sc = _objective(cand)
            top_candidates.append((sc, cand.copy()))
            if sc < overall_best_score:
                overall_best_score, overall_best_seq = sc, cand.copy()
                print(f"[{time.time():.1f}] Dirichlet len={n} α={alpha:.3f} score={sc:.6f}")

        # e) Sparse random seeds
        for _ in range(3):
            if time.time() >= deadline:
                break
            cand = _distribution_random_sparse(n, rng, density=0.006)
            sc = _objective(cand)
            top_candidates.append((sc, cand.copy()))
            if sc < overall_best_score:
                overall_best_score, overall_best_seq = sc, cand.copy()
                print(f"[{time.time():.1f}] sparse len={n} score={sc:.6f}")

        # keep only the most promising candidates globally
        top_candidates.sort(key=lambda x: x[0])
        top_candidates = top_candidates[:40]

    # ------------------------------------------------------------------
    # 2. Incorporate previous best if available
    # ------------------------------------------------------------------
    try:
        prev = np.asarray(height_sequence_1, dtype=np.float64)  # type: ignore[name-defined]
        if prev.ndim == 1 and prev.size > 0:
            prev = prev / prev.sum()
            sc = _objective(prev)
            if sc < overall_best_score:
                overall_best_score, overall_best_seq = sc, prev.copy()
                print(f"[{time.time():.1f}] previous best improved score={sc:.6f}")
            top_candidates.append((sc, prev.copy()))
    except Exception:
        pass

    # ------------------------------------------------------------------
    # 3. Fallback (should never fire)
    # ------------------------------------------------------------------
    if overall_best_seq is None:
        n = 3000
        overall_best_seq = _distribution_uniform(n)
        overall_best_score = _objective(overall_best_seq)
        print(f"[{time.time():.1f}] fallback uniform score={overall_best_score:.6f}")

    # ------------------------------------------------------------------
    # 4. Intensive optimisation on the very best seeds
    # ------------------------------------------------------------------
    # select a handful of top seeds for long‑run descent
    top_candidates.sort(key=lambda x: x[0])
    seeds_for_descent = [seq for _, seq in top_candidates[:8]]

    remaining = deadline - time.time()
    if remaining > 30.0 and seeds_for_descent:
        per_seed = remaining / max(1, len(seeds_for_descent))
        print(f"[{time.time():.1f}] starting momentum descent on {len(seeds_for_descent)} seeds")
        for idx, seed_seq in enumerate(seeds_for_descent):
            if time.time() >= deadline:
                break
            sub_deadline = time.time() + per_seed
            a_opt, sc_opt = _exp_descent_momentum(
                seed_seq.copy(),
                sub_deadline,
                best_seq=overall_best_seq,
                best_score=overall_best_score,
                eta0=0.6,
                tau0=0.4,
                tau_mult=1.003,
                eval_interval=200,
                patience=2000,
            )
            if sc_opt < overall_best_score:
                overall_best_score, overall_best_seq = sc_opt, a_opt.copy()
                print(
                    f"[{time.time():.1f}] after descent seed {idx} new global best {overall_best_score:.6f}"
                )

    # ------------------------------------------------------------------
    # 5. Pairwise swap refinement (fast hill‑climb)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        print(f"[{time.time():.1f}] pairwise refinement")
        overall_best_seq, overall_best_score = _pairwise_swap_local(
            overall_best_seq.copy(),
            deadline,
            best_seq=overall_best_seq,
            best_score=overall_best_score,
            max_trials=400_000,
        )

    # ------------------------------------------------------------------
    # 6. Final cheap simulated‑annealing walk
    # ------------------------------------------------------------------
    if time.time() < deadline:
        print(f"[{time.time():.1f}] final stochastic local search")
        overall_best_seq, overall_best_score = _random_local_search(
            overall_best_seq.copy(),
            deadline,
            best_seq=overall_best_seq,
            best_score=overall_best_score,
            init_temp=0.008,
            decay=0.9997,
        )

    # ------------------------------------------------------------------
    print(
        f"Final best score: {overall_best_score:.6f} (length={len(overall_best_seq)})"
    )
    # ``evaluate_sequence`` will perform its own normalisation, clipping, etc.
    return overall_best_seq.tolist()
