"""
Propose a non‑negative step function that (approximately) minimizes
``evaluate_sequence``.  The algorithm proceeds in three phases:

1️⃣  **Diverse seed pool** –  Analytic families (uniform, edge‑weighted,
    edge‑inverse, sine‑weighted, beta, exponential, Gaussian, cosine),
    random Dirichlet draws and sparse random vectors are sampled for a few
    lengths.  The ~10 best candidates (lowest exact objective) are kept.

2️⃣  **Multi‑start mirror descent** –  Each of the retained seeds is
    refined independently by exponentiated‑gradient descent with momentum.
    The active set of convolution entries that are close to the current
    maximum is used to build a sub‑gradient.  Hyper‑parameters are
    deliberately aggressive (small active‑fraction, larger step‑size) to
    escape shallow local minima.  The best result over all starts is kept.

3️⃣  **Local polishing** –  The current overall best sequence undergoes a
    greedy pairwise mass transfer and a cheap simulated‑annealing search.
    The final list of heights is returned; it can be fed directly to
    ``evaluate_sequence`` (the evaluator will normalise it).

The implementation respects the 1000 s time budget, prints occasional
progress messages and uses only NumPy (no file or network I/O).
"""

import time
import math
import numpy as np
from numpy.fft import rfft, irfft

# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x ≥ 0, Σx = s}."""
    if s < 0:
        raise ValueError("simplex radius must be non‑negative")
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho_arr = np.where(u - cssv / (np.arange(1, len(v) + 1)) > 0)[0]
    if rho_arr.size == 0:
        theta = 0.0
    else:
        rho = rho_arr[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Full convolution a*a using a real FFT (returns length 2n‑1)."""
    n = a.shape[0]
    need = 2 * n - 1
    size = 1 << ((need - 1).bit_length())
    A = rfft(a, n=size)
    conv = irfft(A * A, n=size)
    return conv[:need]


# ----------------------------------------------------------------------
def _objective(a: np.ndarray) -> float:
    """Exact objective 2·n·max(conv(a,a)) when Σa = 1."""
    n = a.shape[0]
    b = _fft_self_convolve(a)
    return 2.0 * n * b.max()

# Analytic families ----------------------------------------------------
def _distribution_uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _distribution_tent(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    half = (n - 1) / 2.0
    w = np.where(i <= half, i + 1, n - i)
    w /= w.sum()
    return w


def _distribution_quadratic(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    half = (n - 1) / 2.0
    d = np.abs(i - half) + 1.0
    w = 1.0 / (d ** 2)
    w /= w.sum()
    return w


def _distribution_edge_weighted(n: int, p: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64) + 0.5
    j = (n - np.arange(n, dtype=np.float64) - 0.5)
    w = (i * j) ** (-p)
    s = w.sum()
    if s == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= s
    return w


def _distribution_edge_inversepow(n: int, p: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64) + 0.5
    j = (n - np.arange(n, dtype=np.float64) - 0.5)
    w = i ** (-p) + j ** (-p)
    s = w.sum()
    if s == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= s
    return w


def _distribution_sine_weighted(n: int, p: float) -> np.ndarray:
    theta = np.pi * (np.arange(n, dtype=np.float64) + 0.5) / n
    w = np.sin(theta) ** p
    s = w.sum()
    if s == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= s
    return w


def _distribution_beta_sym(n: int, p: float) -> np.ndarray:
    """Symmetric Beta(p,p) – p>0."""
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    w = np.exp((p - 1) * np.log(x) + (p - 1) * np.log(1.0 - x))
    s = w.sum()
    if s == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= s
    return w


def _distribution_exponential_decay(n: int, rate: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    half = (n - 1) / 2.0
    w = np.exp(-rate * np.abs(i - half))
    w /= w.sum()
    return w


def _distribution_gaussian(n: int, sigma: float) -> np.ndarray:
    """Gaussian bell centred in the middle, σ taken in absolute index units."""
    i = np.arange(n, dtype=np.float64)
    half = (n - 1) / 2.0
    w = np.exp(-0.5 * ((i - half) / sigma) ** 2)
    s = w.sum()
    if s == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= s
    return w


def _distribution_cosine_weighted(n: int) -> np.ndarray:
    """U‑shaped cosine: w_i ∝ 1 – cos(2π i/(n‑1))."""
    i = np.arange(n, dtype=np.float64)
    w = 1.0 - np.cos(2.0 * np.pi * i / (n - 1))
    w /= w.sum()
    return w


def _distribution_dirichlet(n: int, rng: np.random.Generator) -> np.ndarray:
    """Random Dirichlet sample with a random concentration."""
    alpha = rng.uniform(0.01, 2.0)
    a = rng.dirichlet(np.full(n, alpha))
    return a.astype(np.float64)


def _distribution_random_sparse(
    n: int, rng: np.random.Generator, density: float = 0.01
) -> np.ndarray:
    """Sparse random vector (few non‑zeros)."""
    k = max(2, int(n * density))
    idx = rng.choice(n, size=k, replace=False)
    vals = np.zeros(n, dtype=np.float64)
    vals[idx] = rng.random(k)
    if vals.sum() == 0.0:
        vals[0] = 1.0
    vals /= vals.sum()
    return vals


def _mix_with_uniform(base: np.ndarray, lam: float) -> np.ndarray:
    """Convex combination lam·U + (1‑lam)·base, where U is uniform."""
    n = base.shape[0]
    uniform = np.full(n, 1.0 / n, dtype=np.float64)
    return lam * uniform + (1.0 - lam) * base

# Mirror descent (exponentiated gradient) with momentum -----------------
def _subgradient_descent_momentum(
    a: np.ndarray,
    deadline: float,
    best_seq: np.ndarray,
    best_score: float,
    lr0: float = 0.9,
    active_frac: float = 0.01,
    max_active: int = 20,
    beta: float = 0.85,
    eval_interval: int = 150,
    patience: int = 2500,
) -> tuple[np.ndarray, float]:
    """
    Exponentiated‑gradient descent on the exact objective.
    ``a`` is assumed to sum to one.
    Returns the best sequence and its objective value encountered.
    """
    n = a.shape[0]
    t = 0
    no_improve = 0
    momentum = np.zeros_like(a)
    rng = np.random.default_rng()
    ind_grid = np.arange(n, dtype=np.int32)

    while time.time() < deadline and t < 300_000:
        t += 1

        conv = _fft_self_convolve(a)
        max_c = conv.max()
        thresh = max_c * (1.0 - active_frac)

        act = np.where(conv >= thresh)[0]
        if act.size > max_active:
            top_idx = np.argpartition(conv[act], -max_active)[-max_active:]
            act = act[top_idx]

        grad = np.zeros_like(a)
        for idx in act:
            offset = idx - ind_grid
            mask = (offset >= 0) & (offset < n)
            grad[mask] += 2.0 * a[offset[mask]]

        momentum = beta * momentum + (1.0 - beta) * grad
        eta = lr0 / math.sqrt(t)

        a = a * np.exp(-eta * momentum)
        a = _project_onto_simplex(a, s=1.0)

        # occasional exact evaluation
        if t % eval_interval == 0:
            cur = _objective(a)
            if cur < best_score - 1e-14:
                best_score = cur
                best_seq = a.copy()
                no_improve = 0
                print(f"[{time.time():.1f}] GD it={t:,} new best {best_score:.6f}")
            else:
                no_improve += 1

        # injected jitter if stuck
        if no_improve >= patience:
            jitter = rng.exponential(scale=0.008, size=n)
            a = a + jitter
            a = np.maximum(a, 0.0)
            a = _project_onto_simplex(a, s=1.0)
            no_improve = 0

    return best_seq, best_score

# Greedy pairwise refinement -------------------------------------------
def _pairwise_swap_local(
    a: np.ndarray,
    deadline: float,
    best_seq: np.ndarray,
    best_score: float,
    max_trials: int | None = None,
) -> tuple[np.ndarray, float]:
    """Transfer a random fraction of mass from i to j if it improves the objective."""
    n = a.shape[0]
    trials = 0
    rng = np.random.default_rng()

    while time.time() < deadline:
        if max_trials is not None and trials >= max_trials:
            break
        trials += 1

        src = np.where(a > 1e-12)[0]
        if src.size == 0:
            break
        i = rng.choice(src)
        j = rng.integers(n - 1)
        if j >= i:
            j += 1

        delta = a[i] * rng.random() * 0.33
        if delta < 1e-14:
            continue

        new_a = a.copy()
        new_a[i] -= delta
        new_a[j] += delta
        cur = _objective(new_a)

        if cur < best_score - 1e-14:
            a = new_a
            best_score = cur
            best_seq = a.copy()
            print(f"[{time.time():.1f}] swap {i}->{j} Δ={delta:.3e} new best {best_score:.6f}")

    return best_seq, best_score

# Simulated‑annealing style local search -------------------------------
def _random_local_search(
    a: np.ndarray,
    deadline: float,
    best_seq: np.ndarray,
    best_score: float,
    init_temp: float = 0.009,
    decay: float = 0.9996,
) -> tuple[np.ndarray, float]:
    """Very cheap SA that repeatedly transfers a random amount of mass."""
    n = a.shape[0]
    rng = np.random.default_rng()
    T = init_temp

    while time.time() < deadline:
        i = rng.integers(n)
        j = rng.integers(n - 1)
        if j >= i:
            j += 1
        delta = a[i] * rng.random() * 0.3
        if delta < 1e-14:
            continue
        new_a = a.copy()
        new_a[i] -= delta
        new_a[j] += delta
        cur = _objective(new_a)

        if cur < best_score - 1e-14:
            a = new_a
            best_score = cur
            best_seq = a.copy()
            print(f"[{time.time():.1f}] SA improve {i}->{j} Δ={delta:.3e} best {best_score:.6f}")
        else:
            if rng.random() < math.exp(-(cur - best_score) / max(T, 1e-12)):
                a = new_a
        T *= decay

    return best_seq, best_score

        [
            np.logspace(-3, 0.0, 8),          # 0.001 … 1
            np.logspace(0.0, 0.5, 6),         # 1 … 3.16
        ]
    )
    lam_grid = np.linspace(0.0, 0.5, 6)
    rate_grid = np.linspace(0.001, 0.025, 5)
    sigma_frac_grid = np.linspace(0.05, 0.35, 6)   # sigma as fraction of n

    for n in lengths:
        if time.time() >= deadline:
            break
        print(f"[{time.time():.1f}] scanning length {n}")

        # deterministic families
        for cand in (
            _distribution_uniform(n),
            _distribution_tent(n),
            _distribution_quadratic(n),
            _distribution_cosine_weighted(n),
        ):
            sc = _objective(cand)
            all_candidates.append((sc, cand.copy()))
            if sc < overall_best_score:
                overall_best_score, overall_best_seq = sc, cand.copy()
                print(f"[{time.time():.1f}] new global best (det) len={n} score={sc:.6f}")

        # parametrised families
        for p in p_grid:
            if time.time() >= deadline:
                break
            edge = _distribution_edge_weighted(n, p)
            edge2 = _distribution_edge_inversepow(n, p)
            sine = _distribution_sine_weighted(n, p)
            beta = _distribution_beta_sym(n, p)

            for base in (edge, edge2, sine, beta):
                for lam in lam_grid:
                    if time.time() >= deadline:
                        break
                    cand = _mix_with_uniform(base, lam)
                    sc = _objective(cand)
                    all_candidates.append((sc, cand.copy()))
                    if sc < overall_best_score:
                        overall_best_score, overall_best_seq = sc, cand.copy()
                        print(f"[{time.time():.1f}] new best mix len={n} p={p:.3g} lam={lam:.2f} score={sc:.6f}")

        # exponential‑decay family
        for rate in rate_grid:
            if time.time() >= deadline:
                break
            cand = _distribution_exponential_decay(n, rate)
            sc = _objective(cand)
            all_candidates.append((sc, cand.copy()))
            if sc < overall_best_score:
                overall_best_score, overall_best_seq = sc, cand.copy()
                print(f"[{time.time():.1f}] new best exp‑decay len={n} rate={rate:.4f} score={sc:.6f}")

        # Gaussian family (σ proportional to n)
        for sf in sigma_frac_grid:
            if time.time() >= deadline:
                break
            sigma = sf * n
            cand = _distribution_gaussian(n, sigma)
            sc = _objective(cand)
            all_candidates.append((sc, cand.copy()))
            if sc < overall_best_score:
                overall_best_score, overall_best_seq = sc, cand.copy()
                print(f"[{time.time():.1f}] new best gaussian len={n} σ={sigma:.1f} score={sc:.6f}")

        # random Dirichlet seeds
        for _ in range(6):
            if time.time() >= deadline:
                break
            a = _distribution_dirichlet(n, rng)
            sc = _objective(a)
            all_candidates.append((sc, a.copy()))
            if sc < overall_best_score:
                overall_best_score, overall_best_seq = sc, a.copy()
                print(f"[{time.time():.1f}] new best Dirichlet len={n} score={sc:.6f}")

        # sparse random seeds
        for _ in range(6):
            if time.time() >= deadline:
                break
            a = _distribution_random_sparse(n, rng, density=0.008)
            sc = _objective(a)
            all_candidates.append((sc, a.copy()))
            if sc < overall_best_score:
                overall_best_score, overall_best_seq = sc, a.copy()
                print(f"[{time.time():.1f}] new best sparse len={n} score={sc:.6f}")

        # keep only a modest number of top candidates so far
        all_candidates.sort(key=lambda x: x[0])
        all_candidates = all_candidates[:30]

    # ------------------------------------------------------------------
    # 2️⃣  Incorporate a previous construction if available
    try:
        prev = np.asarray(height_sequence_1, dtype=np.float64)  # type: ignore[name-defined]
        if prev.ndim == 1 and prev.size > 0:
            prev = prev / prev.sum()
            sc = _objective(prev)
            all_candidates.append((sc, prev.copy()))
            if sc < overall_best_score:
                overall_best_score, overall_best_seq = sc, prev.copy()
                print(f"[{time.time():.1f}] previous construction improved score={sc:.6f}")
    except Exception:
        pass

    # ------------------------------------------------------------------
    # Guard against no seed (should never happen)
    if overall_best_seq is None:
        overall_best_seq = _distribution_uniform(3000)
        overall_best_score = _objective(overall_best_seq)
        print(f"[{time.time():.1f}] fallback uniform score={overall_best_score:.6f}")

    # ------------------------------------------------------------------
    # 3️⃣  Multi‑start mirror‑descent on the best few seeds
    top_seeds = all_candidates[:10]   # up to ten promising starts
    remaining = deadline - time.time()
    if remaining > 30 and top_seeds:
        per_start = max(8.0, (remaining - 30.0) / len(top_seeds))
        print(f"[{time.time():.1f}] multi‑start GD: {len(top_seeds)} seeds, ≈{per_start:.1f}s each")
        for idx, (seed_score, seed_vec) in enumerate(top_seeds):
            if time.time() >= deadline:
                break
            sub_deadline = time.time() + per_start
            print(f"[{time.time():.1f}]  start {idx + 1}/{len(top_seeds)} (seed score {seed_score:.6f})")
            best_seq, best_score = _subgradient_descent_momentum(
                seed_vec.copy(),
                sub_deadline,
                best_seq=seed_vec.copy(),
                best_score=seed_score,
                lr0=1.0,
                active_frac=0.01,
                max_active=20,
                beta=0.85,
                eval_interval=150,
                patience=2000,
            )
            # brief pairwise polishing inside the allocated window
            if time.time() + 5.0 < sub_deadline:
                best_seq, best_score = _pairwise_swap_local(
                    best_seq.copy(),
                    sub_deadline - 5.0,
                    best_seq.copy(),
                    best_score,
                    max_trials=120_000,
                )
            if best_score < overall_best_score:
                overall_best_score = best_score
                overall_best_seq = best_seq.copy()
                print(f"[{time.time():.1f}] NEW GLOBAL BEST after start {idx+1}: {overall_best_score:.6f}")

    # ------------------------------------------------------------------
    # 4️⃣  Final polishing on the overall best (if time permits)
    if time.time() < deadline - 10.0:
        print(f"[{time.time():.1f}] final polishing – pairwise + SA")
        overall_best_seq, overall_best_score = _pairwise_swap_local(
            overall_best_seq.copy(),
            deadline - 6.0,
            best_seq=overall_best_seq,
            best_score=overall_best_score,
            max_trials=300_000,
        )
        if time.time() < deadline - 2.0:
            overall_best_seq, overall_best_score = _random_local_search(
                overall_best_seq.copy(),
                deadline - 2.0,
                best_seq=overall_best_seq,
                best_score=overall_best_score,
                init_temp=0.01,
                decay=0.9995,
            )

    # ------------------------------------------------------------------
    print(f"Final best approx. objective ≈ {overall_best_score:.6f} (length={len(overall_best_seq)})")
    return overall_best_seq.tolist()
