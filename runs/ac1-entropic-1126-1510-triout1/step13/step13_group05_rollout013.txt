def _fft_convolve(a,b):
    n = len(a) + len(b) - 1
    size = 1 << ((n - 1).bit_length())
    A = np.fft.rfft(a, size)
    B = np.fft.rfft(b, size)
    conv = np.fft.irfft(A * B, size)[:n]
    return conv

"""
Hybrid optimisation for the step‑function problem.

The routine consists of four phases:

1. **Broad scan** – generate a diverse set of candidate vectors from
   analytic families (uniform, arcsine, sine, cosine, quadratic,
   beta, edge‑weighted, power‑sum, exponential) on a few moderate
   lengths, plus random Dirichlet / sparse seeds.  The ≈15 best
   candidates (according to the exact objective
   ``2·n·max(conv)`` with the vector normalised to sum=1) are retained.

2. **Smoothed gradient descent** – each retained seed is refined by
   minimising a smooth surrogate of the ``max`` (log‑sum‑exp) using a
   projected gradient step on the simplex.  The gradient of the
   surrogate is computed with two FFT‑based convolutions, giving an
   O(n log n) per‑iteration cost.  The temperature of the
   log‑sum‑exp is annealed, and the step‑size decays as η₀/√t.

3. **Exact L‑BFGS fine‑tuning** – starting from the best vector found
   by the previous stage we run a limited L‑BFGS optimisation on the
   true objective (using a sub‑gradient of the max‑convolution).  The
   optimisation is performed in the unconstrained “soft‑max”
   parametrisation ``a_i = exp(u_i)/∑exp(u)``; the gradient w.r.t. the
   parameters is obtained analytically.

4. **Greedy pairwise polishing** – a fast hill‑climbing pass moves a
   fraction of mass from one index to another while strictly decreasing
   the exact objective.

The whole procedure respects the supplied time budget (default
1000 s).  The final vector is returned as a plain Python list of
non‑negative floats (already normalised – the evaluation function is
scale‑invariant).

def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x≥0, Σx=s}."""
    if s < 0:
        raise ValueError("Simplex radius must be non‑negative")
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.where(u - cssv / (np.arange(1, len(v) + 1)) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    return np.maximum(v - theta, 0.0)


# ----------------------------------------------------------------------
def _fft_convolve(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """
    Full convolution of two real vectors using a real FFT.
    Returns a vector of length len(a)+len(b)-1.
    """
    n = a.shape[0] + b.shape[0] - 1
    size = 1 << ((n - 1).bit_length())   # next power of two
    A = np.fft.rfft(a, n=size)
    B = np.fft.rfft(b, n=size)
    conv = np.fft.irfft(A * B, n=size)[:n]
    return conv


def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Full self‑convolution a*a using a real FFT (length 2·n−1)."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1 - 1).bit_length())
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[: 2 * n - 1]

def _objective(a: np.ndarray) -> float:
    """
    Exact objective: 2·n·max(conv(a,a)) for a vector a whose entries sum to 1.
    """
    n = a.shape[0]
    b = _fft_self_convolve(a)
    return float(2.0 * n * b.max())

def _distribution_uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _distribution_arcsine(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    vals /= vals.sum()
    return vals


def _distribution_sine(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64) + 0.5
    w = np.sin(np.pi * i / n)
    w /= w.sum()
    return w


def _distribution_cosine(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64) + 0.5
    w = 1.0 - np.cos(np.pi * i / n)
    w /= w.sum()
    return w


def _distribution_quadratic(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    x = (2.0 * i) / (n - 1) - 1.0
    w = 1.0 - x * x
    w = np.clip(w, 0.0, None)
    w /= w.sum()
    return w


def _distribution_beta_sym(n: int, p: float) -> np.ndarray:
    """Symmetric Beta(p,p) on a regular grid."""
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    w = np.exp((p - 1) * np.log(x) + (p - 1) * np.log(1.0 - x))
    if w.sum() == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _distribution_edge_weighted(n: int, p: float) -> np.ndarray:
    """
    Edge‑weighted family: w_i ∝ ((i+½)*(n‑i‑½))^{‑p}.
    """
    i = np.arange(n, dtype=np.float64) + 0.5
    j = (n - np.arange(n, dtype=np.float64) - 0.5)
    w = (i * j) ** (-p)
    total = w.sum()
    if total > 0:
        w /= total
    else:
        w = np.full(n, 1.0 / n)
    return w


def _distribution_power_sum(n: int, p: float) -> np.ndarray:
    """w_i ∝ (i+1)^{‑p} + (n‑i)^{‑p} – another edge‑heavy family."""
    i = np.arange(1, n + 1, dtype=np.float64)
    w = i ** (-p) + (n + 1 - i) ** (-p)
    w /= w.sum()
    return w


def _distribution_exponential(n: int, lam: float) -> np.ndarray:
    """Exponential decay w_i ∝ exp(‑lam·i).  Normalised to sum = 1."""
    i = np.arange(n, dtype=np.float64)
    w = np.exp(-lam * i)
    w /= w.sum()
    return w


def _distribution_random_sparse(
    n: int, rng: np.random.Generator, density: float = 0.01
) -> np.ndarray:
    """Sparse random vector with a prescribed density of non‑zeros."""
    k = max(2, int(n * density))
    idx = rng.choice(n, size=k, replace=False)
    vals = np.zeros(n, dtype=np.float64)
    vals[idx] = rng.random(k)
    if vals.sum() == 0.0:
        vals[0] = 1.0
    vals /= vals.sum()
    return vals


def _mix_with_uniform(base: np.ndarray, lam: float) -> np.ndarray:
    """Convex combination λ·uniform + (1‑λ)·base."""
    n = base.shape[0]
    uniform = np.full(n, 1.0 / n, dtype=np.float64)
    return lam * uniform + (1.0 - lam) * base

def _max_convolution_subgradient(a: np.ndarray) -> np.ndarray:
    """
    Sub‑gradient of max_k Σ_i a_i a_{k‑i} .
    Returns a vector of the same shape as a.
    """
    conv = _fft_self_convolve(a)
    max_val = conv.max()
    tol = 1e-12 * max_val
    active = np.where(conv >= max_val - tol)[0]
    n = a.shape[0]
    grad = np.zeros_like(a)
    for k in active:
        lo = max(0, k - (n - 1))
        hi = min(n - 1, k)
        idx = np.arange(lo, hi + 1)
        grad[idx] += 2.0 * a[k - idx]
    if len(active) > 0:
        grad /= len(active)
    return grad


# ----------------------------------------------------------------------
# Gradient of a soft‑max (log‑sum‑exp) surrogate ------------------------
def _weighted_gradient(a: np.ndarray, w: np.ndarray) -> np.ndarray:
    """
    Gradient of the weighted sum Σ_k w_k·conv_k(a) .
    w must have length 2·n‑1 and be non‑negative (e.g. soft‑max weights).
    Returns a vector of length n.
    """
    n = a.shape[0]
    # term1: conv of a with w (full length 3·n‑2)
    conv1 = _fft_convolve(a, w)                     # length 3n‑2
    # term2: conv of a with reversed w
    conv2 = _fft_convolve(a, w[::-1])               # length 3n‑2
    # extract needed slices
    term1 = conv1[:n]                               # Σ_i a_i w_{l‑i}
    term2 = conv2[2 * n - 2 - np.arange(n)]         # Σ_i a_i w_{i+l}
    grad = term1 + term2
    return grad


# ----------------------------------------------------------------------
# Smoothed (log‑sum‑exp) gradient descent -------------------------------
def _smooth_gradient_descent(
    a0: np.ndarray,
    time_limit: float,
    best_seq: np.ndarray,
    best_score: float,
    eta0: float = 0.5,
    tau0: float = 0.1,
) -> tuple[np.ndarray, float]:
    """
    Projected gradient descent on the log‑sum‑exp surrogate.
    Uses a simple Euclidean projection onto the simplex.
    """
    a = a0.copy()
    n = a.shape[0]
    t = 0
    tau = tau0
    while time.time() < time_limit:
        t += 1
        conv = _fft_self_convolve(a)                # length 2n‑1
        max_val = conv.max()
        # anneal temperature
        if t % 250 == 0:
            tau = max(tau * 0.85, 1e-7)
        # soft‑max weights
        exps = np.exp((conv - max_val) / tau)
        w = exps / exps.sum()
        # gradient of the surrogate
        grad = _weighted_gradient(a, w)
        # step size
        eta = eta0 / math.sqrt(t)
        a = a - eta * grad
        a = _project_onto_simplex(a, 1.0)            # keep Σa = 1, a ≥ 0
        if t % 50 == 0:
            cur = _objective(a)
            if cur < best_score - 1e-14:
                best_score = cur
                best_seq = a.copy()
                print(
                    f"[{time.time():.1f}] smooth‑gd it={t} new best {best_score:.6f}"
                )
    return best_seq, best_score


# ----------------------------------------------------------------------
# L‑BFGS fine‑tuning (exact max‑convolution) ---------------------------
def _lbfgs_optimize(
    a0: np.ndarray,
    time_limit: float,
    best_seq: np.ndarray,
    best_score: float,
    max_iter: int = 200,
) -> tuple[np.ndarray, float]:
    """
    Run a limited L‑BFGS optimisation on the true objective.
    Optimisation variable is the unconstrained vector u with
    a = softmax(u).  The gradient w.r.t. u is obtained analytically.
    """
    try:
        import scipy.optimize
    except Exception:
        print("SciPy not available – skipping L‑BFGS stage.")
        return best_seq, best_score

    n = a0.shape[0]
    eps = 1e-12
    u0 = np.log(np.maximum(a0, eps))

    def func(u: np.ndarray):
        a = np.exp(u)
        s = a.sum()
        if s == 0:
            a = np.full(n, 1.0 / n)
        else:
            a = a / s
        conv = _fft_self_convolve(a)
        max_conv = conv.max()
        val = 2.0 * n * max_conv
        # gradient of max‑conv w.r.t. a (sub‑gradient)
        grad_a = _max_convolution_subgradient(a)
        grad_a = 2.0 * n * grad_a
        # chain rule through soft‑max
        dot = np.dot(grad_a, a)
        grad_u = a * (grad_a - dot)
        return float(val), grad_u.astype(float)

    opts = {"maxiter": max_iter, "disp": False}
    res = scipy.optimize.minimize(
        func, u0, method="L-BFGS-B", jac=True, options=opts
    )
    if not res.success:
        # even if it failed we still extract the point
        pass
    u_opt = res.x
    a_opt = np.exp(u_opt)
    a_opt = a_opt / a_opt.sum()
    cur = _objective(a_opt)
    if cur < best_score - 1e-14:
        best_score = cur
        best_seq = a_opt.copy()
        print(f"[{time.time():.1f}] L‑BFGS new best {best_score:.6f}")
    return best_seq, best_score


# ----------------------------------------------------------------------
# Greedy pairwise polishing -------------------------------------------
def _pairwise_swap(
    a0: np.ndarray,
    time_limit: float,
    best_seq: np.ndarray,
    best_score: float,
    max_trials: int | None = None,
) -> tuple[np.ndarray, float]:
    """
    Randomly move a fraction of mass from one index to another.
    Only moves that strictly improve the exact objective are kept.
    """
    a = a0.copy()
    n = a.shape[0]
    trials = 0
    while time.time() < time_limit:
        if max_trials is not None and trials >= max_trials:
            break
        trials += 1

        src = np.where(a > 1e-14)[0]
        if src.size == 0:
            break
        i = np.random.choice(src)
        j = np.random.randint(n - 1)
        if j >= i:
            j += 1

        delta = a[i] * np.random.rand() * 0.45  # up to 45 % of source mass
        if delta < 1e-15:
            continue

        new_a = a.copy()
        new_a[i] -= delta
        new_a[j] += delta

        cur = _objective(new_a)
        if cur < best_score - 1e-14:
            a = new_a
            best_score = cur
            best_seq = a.copy()
            print(
                f"[{time.time():.1f}] swap i={i}→j={j} δ={delta:.2e} new best {best_score:.6f}"
            )
    return best_seq, best_score


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: int = 1000) -> list[float]:
    """
    Search for a non‑negative step‑function that minimises ``evaluate_sequence``.
    The routine follows the four‑phase plan described in the module docstring.
    """
    rng = np.random.default_rng(seed)
    start = time.time()
    deadline = start + budget_s - 2.0  # safety margin

    overall_best_seq = None
    overall_best_score = float("inf")
    top_candidates: list[tuple[float, np.ndarray]] = []  # (score, seq)

    # ------------------------------------------------------------------
    # 1.  Scan a diverse collection of analytic families
    # ------------------------------------------------------------------
    lengths = [1024, 2048, 4096]   # moderate sizes – enough expressive power
    p_grid = np.concatenate(
        [
            np.logspace(-2, -0.3, 8),   # 0.01 … 0.5
            np.linspace(0.5, 2.0, 10),
            np.linspace(2.0, 5.0, 6),
        ]
    )
    lam_grid = np.linspace(0.0, 1.0, 5)   # 0, .25, .5, .75, 1

    for n in lengths:
        if time.time() >= deadline:
            break
        print(f"[{time.time():.1f}] scanning length {n}")

        # deterministic families
        for base_fn in (
            _distribution_uniform,
            _distribution_arcsine,
            _distribution_sine,
            _distribution_cosine,
            _distribution_quadratic,
        ):
            cand = base_fn(n)
            sc = _objective(cand)
            top_candidates.append((sc, cand.copy()))
            if sc < overall_best_score:
                overall_best_score = sc
                overall_best_seq = cand.copy()
                print(f"[{time.time():.1f}] new global best len={n} score={sc:.6f}")

        # parametric families
        for p in p_grid:
            if time.time() >= deadline:
                break
            edge_seq = _distribution_edge_weighted(n, p)
            beta_seq = _distribution_beta_sym(n, p)
            pow_seq = _distribution_power_sum(n, p)
            exp_seq = _distribution_exponential(n, p)  # treat p as λ
            for base in (edge_seq, beta_seq, pow_seq, exp_seq):
                for lam in lam_grid:
                    if time.time() >= deadline:
                        break
                    cand = _mix_with_uniform(base, lam)
                    sc = _objective(cand)
                    top_candidates.append((sc, cand.copy()))
                    if sc < overall_best_score:
                        overall_best_score = sc
                        overall_best_seq = cand.copy()
                        print(
                            f"[{time.time():.1f}] new best len={n} p={p:.3g} lam={lam:.2f} score={sc:.6f}"
                        )

        # random Dirichlet seeds
        for _ in range(4):
            if time.time() >= deadline:
                break
            alpha = rng.uniform(0.01, 2.0)
            a = rng.dirichlet(np.full(n, alpha))
            sc = _objective(a)
            top_candidates.append((sc, a.copy()))
            if sc < overall_best_score:
                overall_best_score = sc
                overall_best_seq = a.copy()
                print(f"[{time.time():.1f}] Dirichlet len={n} score={sc:.6f}")

        # sparse random seeds
        for _ in range(4):
            if time.time() >= deadline:
                break
            a = _distribution_random_sparse(n, rng, density=0.008)
            sc = _objective(a)
            top_candidates.append((sc, a.copy()))
            if sc < overall_best_score:
                overall_best_score = sc
                overall_best_seq = a.copy()
                print(f"[{time.time():.1f}] sparse len={n} score={sc:.6f}")

        # retain only the most promising ~20 candidates
        top_candidates.sort(key=lambda x: x[0])
        top_candidates = top_candidates[:20]

    # ------------------------------------------------------------------
    # 2.  Incorporate any previously discovered construction (if any)
    # ------------------------------------------------------------------
    try:
        prev = np.asarray(height_sequence_1, dtype=np.float64)  # type: ignore[name-defined]
        if prev.ndim == 1 and prev.size > 0:
            prev = np.maximum(prev, 0.0)
            total = prev.sum()
            if total > 0:
                prev = prev / total
            else:
                prev = _distribution_uniform(prev.size)
            sc = _objective(prev)
            if sc < overall_best_score:
                overall_best_score = sc
                overall_best_seq = prev.copy()
                print(f"[{time.time():.1f}] previous construction score={sc:.6f}")
            top_candidates.append((sc, prev.copy()))
    except Exception:
        pass

    # ------------------------------------------------------------------
    # 3.  Fallback – should never happen.
    # ------------------------------------------------------------------
    if overall_best_seq is None:
        n = 3000
        overall_best_seq = _distribution_uniform(n)
        overall_best_score = _objective(overall_best_seq)
        print(
            f"[{time.time():.1f}] fallback uniform length={n} score={overall_best_score:.6f}"
        )

    # ------------------------------------------------------------------
    # 4.  Smoothed gradient descent on a handful of top seeds
    # ------------------------------------------------------------------
    remaining = deadline - time.time()
    if remaining > 30.0 and top_candidates:
        seeds = [seq for _, seq in sorted(top_candidates, key=lambda x: x[0])[:6]]
        per_seed = remaining / max(1, len(seeds))
        print(f"[{time.time():.1f}] start smoothed‑GD on {len(seeds)} seeds")
        for idx, seed_seq in enumerate(seeds):
            if time.time() >= deadline:
                break
            sg_deadline = time.time() + per_seed
            a_opt, sc_opt = _smooth_gradient_descent(
                seed_seq,
                sg_deadline,
                best_seq=overall_best_seq,
                best_score=overall_best_score,
                eta0=0.7,
                tau0=0.15,
            )
            if sc_opt < overall_best_score - 1e-12:
                overall_best_score = sc_opt
                overall_best_seq = a_opt.copy()
                print(
                    f"[{time.time():.1f}] after smooth‑GD seed {idx} new best {overall_best_score:.6f}"
                )

    # ------------------------------------------------------------------
    # 5.  Exact L‑BFGS fine‑tuning (if time permits)
    # ------------------------------------------------------------------
    if time.time() < deadline - 5.0:
        print(f"[{time.time():.1f}] L‑BFGS fine‑tuning")
        a_opt, sc_opt = _lbfgs_optimize(
            overall_best_seq.copy(),
            deadline,
            best_seq=overall_best_seq,
            best_score=overall_best_score,
            max_iter=250,
        )
        if sc_opt < overall_best_score - 1e-12:
            overall_best_score = sc_opt
            overall_best_seq = a_opt.copy()
            print(f"[{time.time():.1f}] after L‑BFGS new best {overall_best_score:.6f}")

    # ------------------------------------------------------------------
    # 6.  Final greedy polishing
    # ------------------------------------------------------------------
    if time.time() < deadline:
        print(f"[{time.time():.1f}] final greedy polishing")
        a_opt, sc_opt = _pairwise_swap(
            overall_best_seq.copy(),
            deadline,
            best_seq=overall_best_seq,
            best_score=overall_best_score,
        )
        if sc_opt < overall_best_score - 1e-12:
            overall_best_score = sc_opt
            overall_best_seq = a_opt.copy()
            print(f"[{time.time():.1f}] after polishing new best {overall_best_score:.6f}")

    print(
        f"Final best score (2·n·max(conv)): {overall_best_score:.6f} (length={len(overall_best_seq)})"
    )
    # Return a plain Python list; the evaluation function will scale it internally.
    return overall_best_seq.tolist()
