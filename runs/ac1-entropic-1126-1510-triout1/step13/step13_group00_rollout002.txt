"""
Propose a non‑negative step function minimizing ``evaluate_sequence``.

Algorithm
---------

* **Diverse seeding** – generate many analytic families (uniform,
  arcsine, symmetric and asymmetric Beta, edge‑weighted, edge‑inverse‑power,
  sine‑weighted) for several moderate lengths (≈ 3– 6 k).  p‑parameters are
  explored on a wide logarithmic grid up to 15.  Random Dirichlet and
  sparse vectors are added.  The 30 best seeds (lowest exact objective)
  are kept.

* **Smooth max mirror descent** – minimise the exact objective
  ``2·n·max(a∗a)`` on the simplex using a *smooth* approximation of the
  max (log‑sum‑exp).  The gradient of the smooth objective is obtained
  analytically via FFT‑based convolution, yielding a fast (≈ O(n log n))
  update.  An exponentiated‑gradient (entropy mirror) step keeps the
  iterates on the probability simplex.  The smoothing parameter α is
  increased slowly from 5 to 50, providing a very accurate surrogate for
  the true max while keeping gradients stable.

* **Upsampling** – the best distribution is linearly interpolated to a
  larger length (≈ 2×, capped at 18000) and refined again with the same
  smooth descent.  Larger n lowers the theoretical lower bound
  ``2·n·max`` and usually improves the score.

* **Deterministic cleanup** – a greedy pairwise mass‑transfer sweep
  (exact objective) followed by a cheap simulated‑annealing style
  local search removes any remaining spikes.

The whole procedure respects the 1000 s budget and returns the best
sequence found as a plain Python list of floats.
"""

# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x≥0, sum(x)=s}."""
    if s < 0:
        raise ValueError("simplex radius must be non‑negative")
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho_arr = np.where(u - cssv / (np.arange(1, len(v) + 1)) > 0)[0]
    if rho_arr.size == 0:
        theta = 0.0
    else:
        rho = rho_arr[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _fft_convolve(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Linear convolution of two real vectors using FFT (full length)."""
    n = a.shape[0] + b.shape[0] - 1
    size = 1 << ((n - 1).bit_length())
    A = rfft(a, n=size)
    B = rfft(b, n=size)
    conv = irfft(A * B, n=size)
    return conv[:n]


def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a (real FFT, length 2·n‑1)."""
    n = a.shape[0]
    need = 2 * n - 1
    size = 1 << ((need - 1).bit_length())
    A = rfft(a, n=size)
    conv = irfft(A * A, n=size)
    return conv[:need]


# ----------------------------------------------------------------------
def _objective(a: np.ndarray) -> float:
    """Exact objective 2·n·max(a*a) for a with sum(a)=1."""
    n = a.shape[0]
    b = _fft_self_convolve(a)
    return 2.0 * n * b.max()

# Analytic families ----------------------------------------------------
def _distribution_uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _distribution_arcsine(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    vals /= vals.sum()
    return vals


def _distribution_beta_sym(n: int, p: float) -> np.ndarray:
    """Symmetric Beta(p,p)."""
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    w = np.exp((p - 1) * np.log(x) + (p - 1) * np.log(1.0 - x))
    if w.sum() == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _distribution_beta_asym(n: int, a: float, b: float) -> np.ndarray:
    """Asymmetric Beta(a,b)."""
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    logw = (a - 1) * np.log(x) + (b - 1) * np.log(1.0 - x)
    w = np.exp(logw - np.max(logw))
    if w.sum() == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _distribution_edge_weighted(n: int, p: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64) + 0.5
    j = (n - np.arange(n, dtype=np.float64) - 0.5)
    w = (i * j) ** (-p)
    if w.sum() == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _distribution_edge_inversepow(n: int, p: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64) + 0.5
    j = (n - np.arange(n, dtype=np.float64) - 0.5)
    w = i ** (-p) + j ** (-p)
    if w.sum() == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _distribution_sine_weighted(n: int, p: float) -> np.ndarray:
    theta = np.pi * (np.arange(n, dtype=np.float64) + 0.5) / n
    w = np.sin(theta) ** p
    if w.sum() == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _distribution_random_sparse(
    n: int, rng: np.random.Generator, density: float = 0.006
) -> np.ndarray:
    k = max(2, int(n * density))
    idx = rng.choice(n, size=k, replace=False)
    vals = np.zeros(n, dtype=np.float64)
    vals[idx] = rng.random(k)
    if vals.sum() == 0.0:
        vals[0] = 1.0
    vals /= vals.sum()
    return vals


def _distribution_dirichlet(n: int, rng: np.random.Generator) -> np.ndarray:
    alpha = rng.uniform(0.01, 2.0)
    a = rng.dirichlet(np.full(n, alpha))
    return a.astype(np.float64)


def _mix_with_uniform(base: np.ndarray, lam: float) -> np.ndarray:
    """lam·U + (1‑lam)·base, where U is the uniform distribution."""
    n = base.shape[0]
    uniform = np.full(n, 1.0 / n, dtype=np.float64)
    return lam * uniform + (1.0 - lam) * base

def _smooth_max_gradient(a: np.ndarray, alpha: float) -> tuple[float, np.ndarray]:
    """
    Returns a smooth approximation of max(a*a) and its gradient w.r.t a.
    Uses the log‑sum‑exp smoothed max with parameter alpha.
    Gradient is computed via FFT (cross‑correlation).
    """
    n = a.shape[0]
    conv = _fft_self_convolve(a)                     # length 2n‑1
    # smoothed max
    scaled = alpha * conv
    max_scaled = np.max(scaled)
    exp_scaled = np.exp(scaled - max_scaled)        # shift for stability
    w = exp_scaled / exp_scaled.sum()               # softmax weights, sum to 1
    smooth_max = (max_scaled + np.log(exp_scaled.sum())) / alpha

    # gradient of smooth_max (scaled by 2·n later)
    a_rev = a[::-1]
    # cross‑correlation a_rev * w (full)
    cross = _fft_convolve(a_rev, w)                 # length 3n‑2
    # slice the part that corresponds to gradient entries
    grad_core = cross[n - 1 : n - 1 + n]            # shape (n,)
    # gradient of objective 2·n·smooth_max:
    grad = 4.0 * n * grad_core
    return smooth_max, grad


def _smooth_descent_mirror(
    a0: np.ndarray,
    deadline: float,
    best_seq: np.ndarray,
    best_score: float,
    lr0: float = 0.8,
    alpha_start: float = 5.0,
    alpha_end: float = 50.0,
    eval_interval: int = 80,
    patience: int = 1500,
) -> tuple[np.ndarray, float]:
    """
    Entropy mirror descent on the smooth approximation of the objective.
    The smoothing parameter alpha is increased gradually from alpha_start
    to alpha_end.
    """
    a = a0.copy()
    n = a.shape[0]
    t = 0
    no_improve = 0
    rng = np.random.default_rng()
    alpha = alpha_start
    while time.time() < deadline and t < 500_000:
        t += 1

        # update smoothing parameter (geometric schedule)
        if alpha < alpha_end:
            alpha = min(alpha_end, alpha * 1.001)

        smax, grad = _smooth_max_gradient(a, alpha)

        # exponentiated‑gradient step
        eta = lr0 / math.sqrt(t)
        # clip exponent to avoid under/overflow
        step = -eta * grad
        step = np.clip(step, -50, 50)
        a = a * np.exp(step)

        # project onto simplex
        s = a.sum()
        if s == 0.0:
            a = np.full(n, 1.0 / n, dtype=np.float64)
        else:
            a /= s

        if t % eval_interval == 0:
            cur = _objective(a)
            if cur + 1e-14 < best_score:
                best_score, best_seq = cur, a.copy()
                no_improve = 0
                print(f"[{time.time():.1f}] it={t:,} new global best {best_score:.6f}")
            else:
                no_improve += 1

        if no_improve >= patience:
            # shake the simplex a little
            jitter = rng.exponential(scale=0.002, size=n)
            a = a + jitter
            a = np.maximum(a, 0.0)
            a /= a.sum()
            no_improve = 0

    return best_seq, best_score

def _pairwise_swap_local(
    a: np.ndarray,
    deadline: float,
    best_seq: np.ndarray,
    best_score: float,
    max_trials: int | None = None,
) -> tuple[np.ndarray, float]:
    """Greedy transfer of a random fraction of mass between two coordinates."""
    n = a.shape[0]
    rng = np.random.default_rng()
    trials = 0
    while time.time() < deadline:
        if max_trials is not None and trials >= max_trials:
            break
        trials += 1
        src = np.where(a > 1e-12)[0]
        if src.size == 0:
            break
        i = rng.choice(src)
        j = rng.integers(n - 1)
        if j >= i:
            j += 1
        delta = a[i] * rng.random() * 0.33
        if delta < 1e-14:
            continue
        new_a = a.copy()
        new_a[i] -= delta
        new_a[j] += delta
        cur = _objective(new_a)
        if cur + 1e-14 < best_score:
            a = new_a
            best_score = cur
            best_seq = a.copy()
            print(f"[{time.time():.1f}] swap {i}->{j} δ={delta:.3e} new best {best_score:.6f}")
    return best_seq, best_score


def _random_local_search(
    a: np.ndarray,
    deadline: float,
    best_seq: np.ndarray,
    best_score: float,
    init_temp: float = 0.008,
    decay: float = 0.9995,
) -> tuple[np.ndarray, float]:
    """Cheap simulated‑annealing style mass‑transfer search."""
    n = a.shape[0]
    rng = np.random.default_rng()
    T = init_temp
    while time.time() < deadline:
        i = rng.integers(n)
        j = rng.integers(n - 1)
        if j >= i:
            j += 1
        delta = a[i] * rng.random() * 0.28
        if delta < 1e-14:
            continue
        new_a = a.copy()
        new_a[i] -= delta
        new_a[j] += delta
        cur = _objective(new_a)
        if cur + 1e-14 < best_score:
            a = new_a
            best_score = cur
            best_seq = a.copy()
            print(f"[{time.time():.1f}] SA improve {i}->{j} Δ={delta:.3e} best {best_score:.6f}")
        else:
            if rng.random() < math.exp(-(cur - best_score) / max(T, 1e-12)):
                a = new_a
        T *= decay
    return best_seq, best_score

def _upsample(seq: np.ndarray, new_len: int) -> np.ndarray:
    """Linear interpolation of a simplex‑vector to a larger length."""
    old_len = seq.shape[0]
    if new_len == old_len:
        return seq.copy()
    x_old = np.linspace(0.0, 1.0, old_len)
    x_new = np.linspace(0.0, 1.0, new_len)
    new_seq = np.interp(x_new, x_old, seq)
    new_seq = np.maximum(new_seq, 0.0)
    new_seq /= new_seq.sum()
    return new_seq

        # asymmetric beta seeds
        for _ in range(8):
            if time.time() >= deadline:
                break
            a = rng.uniform(0.001, 5.0)
            b = rng.uniform(0.001, 5.0)
            cand = _distribution_beta_asym(n, a, b)
            sc = _objective(cand)
            top_candidates.append((sc, cand.copy()))
            if sc < overall_best_score:
                overall_best_score, overall_best_seq = sc, cand.copy()
                print(
                    f"[{time.time():.1f}] new best asym‑beta len={n} a={a:.3g} b={b:.3g} score={sc:.6f}"
                )

        # random Dirichlet seeds
        for _ in range(8):
            if time.time() >= deadline:
                break
            cand = _distribution_dirichlet(n, rng)
            sc = _objective(cand)
            top_candidates.append((sc, cand.copy()))
            if sc < overall_best_score:
                overall_best_score, overall_best_seq = sc, cand.copy()
                print(
                    f"[{time.time():.1f}] new best Dirichlet len={n} score={sc:.6f}"
                )

        # sparse random seeds
        for _ in range(8):
            if time.time() >= deadline:
                break
            cand = _distribution_random_sparse(n, rng, density=0.005)
            sc = _objective(cand)
            top_candidates.append((sc, cand.copy()))
            if sc < overall_best_score:
                overall_best_score, overall_best_seq = sc, cand.copy()
                print(
                    f"[{time.time():.1f}] new best sparse len={n} score={sc:.6f}"
                )

        # keep only the best few seeds overall
        top_candidates.sort(key=lambda x: x[0])
        top_candidates = top_candidates[:30]

    # ------------------------------------------------------------------
    # 2. Include previously discovered construction if present
    try:
        prev = np.asarray(height_sequence_1, dtype=np.float64)  # type: ignore[name-defined]
        if prev.ndim == 1 and prev.size > 0:
            prev = np.maximum(prev, 0.0)
            prev = prev / prev.sum()
            sc = _objective(prev)
            if sc < overall_best_score:
                overall_best_score, overall_best_seq = sc, prev.copy()
                print(f"[{time.time():.1f}] previous best score={sc:.6f}")
            top_candidates.append((sc, prev.copy()))
    except Exception:
        pass

    # ------------------------------------------------------------------
    # 3. Fallback (should never happen)
    if overall_best_seq is None:
        n = 3500
        overall_best_seq = _distribution_uniform(n)
        overall_best_score = _objective(overall_best_seq)
        print(f"[{time.time():.1f}] fallback uniform score={overall_best_score:.6f}")

    # ------------------------------------------------------------------
    # 4. Heavy smooth‑max mirror descent on a handful of top seeds
    remaining = deadline - time.time()
    if remaining > 40.0 and top_candidates:
        seeds = [seq for _, seq in sorted(top_candidates, key=lambda x: x[0])[:7]]
        per_seed = remaining * 0.55 / max(1, len(seeds))
        print(
            f"[{time.time():.1f}] smooth descent on {len(seeds)} seeds, ~{per_seed:.1f}s each"
        )
        for idx, seed_seq in enumerate(seeds):
            if time.time() >= deadline:
                break
            sub_deadline = time.time() + per_seed
            a_opt, sc_opt = _smooth_descent_mirror(
                seed_seq.copy(),
                sub_deadline,
                best_seq=overall_best_seq,
                best_score=overall_best_score,
                lr0=0.85,
                alpha_start=5.0,
                alpha_end=40.0,
                eval_interval=100,
                patience=1500,
            )
            if sc_opt < overall_best_score:
                overall_best_score, overall_best_seq = sc_opt, a_opt.copy()
                print(
                    f"[{time.time():.1f}] after smooth descent seed {idx} new best {overall_best_score:.6f}"
                )

    # ------------------------------------------------------------------
    # 5. Upsample to a larger length if time permits
    remaining = deadline - time.time()
    if remaining > 150.0:
        large_n = min(18000, len(overall_best_seq) * 2)
        print(f"[{time.time():.1f}] up‑sampling to {large_n} points")
        up_seq = _upsample(overall_best_seq, large_n)

        sub_deadline = time.time() + remaining * 0.65
        up_seq_opt, up_score = _smooth_descent_mirror(
            up_seq,
            sub_deadline,
            best_seq=overall_best_seq,
            best_score=overall_best_score,
            lr0=0.6,
            alpha_start=8.0,
            alpha_end=45.0,
            eval_interval=80,
            patience=1800,
        )
        if up_score < overall_best_score:
            overall_best_score, overall_best_seq = up_score, up_seq_opt.copy()
            print(
                f"[{time.time():.1f}] up‑sampled descent yielded new best {overall_best_score:.6f}"
            )

    # ------------------------------------------------------------------
    # 6. Deterministic pairwise refinement
    if time.time() < deadline:
        print(f"[{time.time():.1f}] deterministic pairwise refinement")
        overall_best_seq, overall_best_score = _pairwise_swap_local(
            overall_best_seq.copy(),
            deadline,
            best_seq=overall_best_seq,
            best_score=overall_best_score,
            max_trials=500_000,
        )

    # ------------------------------------------------------------------
    # 7. Final cheap simulated‑annealing clean‑up
    if time.time() < deadline:
        print(f"[{time.time():.1f}] final simulated‑annealing clean‑up")
        overall_best_seq, overall_best_score = _random_local_search(
            overall_best_seq.copy(),
            deadline,
            best_seq=overall_best_seq,
            best_score=overall_best_score,
            init_temp=0.008,
            decay=0.9995,
        )

    print(
        f"Final best score: {overall_best_score:.6f} (length={len(overall_best_seq)})"
    )
    # Return a Python list; let ``evaluate_sequence`` perform its clipping
    return overall_best_seq.tolist()
