"""
Search for a non‑negative step function that minimizes
``evaluate_sequence`` (i.e. 2·n·max(a∗a)/(sum a)²).

The algorithm proceeds in several stages:

1. **Broad seeding** – many analytic families (uniform, Beta,
   edge‑inverse‑power, two‑spike, etc.) and random Dirichlet vectors are
   generated for several moderate lengths.  The 20‑30 best seeds (by the
   exact objective with sum=1) are retained.

2. **Active‑set mirror descent** – an exponentiated‑gradient (entropy)
   descent on the exact non‑smooth objective.  At each iteration only the
   convolution entries that are within a small fraction of the current
   maximum are kept (the active set) and the sub‑gradient is computed
   from them.  Momentum (Adam‑style) and a decreasing stepsize
   ``η≈lr0/√t`` are used.  This phase is run on the top seeds and later
   on an up‑sampled version of the best seed.

3. **Progressive up‑sampling** – the current best distribution is linearly
   interpolated to a larger length (≈2×) and refined again by mirror
   descent.  Larger discretisations allow a finer approximation of the
   continuous optimum.

4. **Deterministic pair‑wise swaps** – a greedy sweep that moves a random
   fraction of mass from one coordinate to another if it improves the
   objective.

5. **Final cheap simulated annealing** – a Metropolis walk that makes
   small random transfers to polish the result.

The whole search respects a user‑provided time budget (default 1000 s)
and always returns a plain Python list of non‑negative heights.
"""

from __future__ import annotations
import time
import math
import numpy as np

# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x≥0, sum(x)=s}."""
    if s < 0:
        raise ValueError("simplex radius must be non‑negative")
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho_arr = np.where(u - cssv / (np.arange(1, len(v) + 1)) > 0)[0]
    if rho_arr.size == 0:
        theta = 0.0
    else:
        rho = rho_arr[-1]
        theta = cssv[rho] / (rho + 1)
    return np.maximum(v - theta, 0.0)


# ----------------------------------------------------------------------
def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a using real FFT (length 2·n‑1)."""
    n = a.shape[0]
    need = 2 * n - 1
    size = 1 << ((need - 1).bit_length())      # next power of two
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[:need]


# ----------------------------------------------------------------------
def _objective(a: np.ndarray) -> float:
    """Exact objective 2·n·max(conv(a,a)) for a with sum(a)=1."""
    n = a.shape[0]
    b = _fft_self_convolve(a)
    return 2.0 * n * b.max()


# ----------------------------------------------------------------------
# Analytic families ----------------------------------------------------
def _distribution_uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _distribution_beta_sym(n: int, p: float) -> np.ndarray:
    """Symmetric Beta(p,p) on a regular grid."""
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    logw = (p - 1) * np.log(x) + (p - 1) * np.log(1.0 - x)
    # shift to avoid overflow
    w = np.exp(logw - np.max(logw))
    if w.sum() == 0.0:
        return np.full(n, 1.0 / n, dtype=np.float64)
    w /= w.sum()
    return w


def _distribution_edge_inversepow(n: int, p: float) -> np.ndarray:
    """Weight proportional to i⁻ᵖ + (n‑i‑1)⁻ᵖ."""
    i = np.arange(n, dtype=np.float64) + 0.5
    j = (n - np.arange(n, dtype=np.float64) - 0.5)
    w = i ** (-p) + j ** (-p)
    if w.sum() == 0.0:
        return np.full(n, 1.0 / n, dtype=np.float64)
    w /= w.sum()
    return w


def _distribution_spike_edges(n: int, span: int) -> np.ndarray:
    """Uniform mass on the first and last *span* coordinates."""
    w = np.zeros(n, dtype=np.float64)
    if span <= 0:
        w[0] = 1.0
    else:
        w[:span] = 1.0
        w[-span:] = 1.0
    w /= w.sum()
    return w


def _mix_with_uniform(base: np.ndarray, lam: float) -> np.ndarray:
    """lam·U + (1‑lam)·base, where U is the uniform distribution."""
    n = base.shape[0]
    uniform = np.full(n, 1.0 / n, dtype=np.float64)
    return lam * uniform + (1.0 - lam) * base

# Mirror‑descent (exponentiated‑gradient) -------------------------------
def _subgradient_descent_momentum(
    a: np.ndarray,
    deadline: float,
    best_seq: np.ndarray,
    best_score: float,
    lr0: float = 0.8,
    active_frac: float = 0.025,
    max_active: int = 60,
    beta: float = 0.9,
    eval_interval: int = 150,
    patience: int = 2000,
) -> tuple[np.ndarray, float]:
    """
    Entropy mirror descent on the exact non‑smooth objective.
    The active set consists of all convolution entries whose value is
    within ``active_frac`` of the current maximum.
    """
    n = a.shape[0]
    t = 0
    no_improve = 0
    momentum = np.zeros_like(a)
    rng = np.random.default_rng()
    ind_grid = np.arange(n, dtype=np.int32)

    while time.time() < deadline and t < 800_000:
        t += 1

        conv = _fft_self_convolve(a)
        max_c = conv.max()
        thresh = max_c * (1.0 - active_frac)

        act = np.where(conv >= thresh)[0]
        if act.size > max_active:
            # keep the largest ones only
            top_idx = np.argpartition(conv[act], -max_active)[-max_active:]
            act = act[top_idx]

        # sub‑gradient of the max‑convolution (sum over active shifts)
        grad = np.zeros_like(a)
        for k in act:
            shift = k - ind_grid
            mask = (shift >= 0) & (shift < n)
            grad[mask] += 2.0 * a[shift[mask]]

        momentum = beta * momentum + (1.0 - beta) * grad
        eta = lr0 / math.sqrt(t)
        a = a * np.exp(-eta * momentum)

        # project back onto the simplex (guarantees sum=1, non‑negative)
        s = a.sum()
        if s == 0.0:
            a = np.full(n, 1.0 / n, dtype=np.float64)
        else:
            a /= s

        if t % eval_interval == 0:
            cur = _objective(a)
            if cur + 1e-14 < best_score:
                best_score, best_seq = cur, a.copy()
                no_improve = 0
                print(f"[{time.time():.1f}] it={t:,} new global best {best_score:.6f}")
            else:
                no_improve += 1

        if no_improve >= patience:
            # random jitter to escape plateaus
            jitter = rng.exponential(scale=0.004, size=n)
            a = a + jitter
            a = np.maximum(a, 0.0)
            a /= a.sum()
            no_improve = 0

    return best_seq, best_score


# ----------------------------------------------------------------------
# Pairwise deterministic refinement --------------------------------------
def _pairwise_swap_local(
    a: np.ndarray,
    deadline: float,
    best_seq: np.ndarray,
    best_score: float,
    max_trials: int | None = None,
) -> tuple[np.ndarray, float]:
    """
    Transfer a random fraction of mass from one coordinate to another.
    Keep the change only if the objective improves.
    """
    n = a.shape[0]
    trials = 0
    rng = np.random.default_rng()
    while time.time() < deadline:
        if max_trials is not None and trials >= max_trials:
            break
        trials += 1

        src = np.where(a > 1e-12)[0]
        if src.size == 0:
            break
        i = rng.choice(src)

        j = rng.integers(n - 1)
        if j >= i:
            j += 1

        delta = a[i] * rng.random() * 0.33
        if delta < 1e-14:
            continue

        new_a = a.copy()
        new_a[i] -= delta
        new_a[j] += delta

        cur = _objective(new_a)
        if cur + 1e-14 < best_score:
            a = new_a
            best_score = cur
            best_seq = a.copy()
            print(
                f"[{time.time():.1f}] swap {i}->{j} δ={delta:.3e} new best {best_score:.6f}"
            )

    return best_seq, best_score


# ----------------------------------------------------------------------
# Cheap simulated annealing ---------------------------------------------
def _random_local_search(
    a: np.ndarray,
    deadline: float,
    best_seq: np.ndarray,
    best_score: float,
    init_temp: float = 0.009,
    decay: float = 0.9994,
) -> tuple[np.ndarray, float]:
    """Transfer mass between random positions; Metropolis acceptance."""
    n = a.shape[0]
    rng = np.random.default_rng()
    T = init_temp
    while time.time() < deadline:
        i = rng.integers(n)
        j = rng.integers(n - 1)
        if j >= i:
            j += 1
        delta = a[i] * rng.random() * 0.28
        if delta < 1e-14:
            continue
        new_a = a.copy()
        new_a[i] -= delta
        new_a[j] += delta

        cur = _objective(new_a)
        if cur + 1e-14 < best_score:
            a = new_a
            best_score = cur
            best_seq = a.copy()
            print(
                f"[{time.time():.1f}] SA improve {i}->{j} Δ={delta:.3e} best {best_score:.6f}"
            )
        else:
            if rng.random() < math.exp(-(cur - best_score) / max(T, 1e-12)):
                a = new_a
        T *= decay
    return best_seq, best_score


# ----------------------------------------------------------------------
# Upsampling helper -------------------------------------------------------
def _upsample(seq: np.ndarray, new_len: int) -> np.ndarray:
    """Linear interpolation of a simplex‑vector to a larger length."""
    old_len = seq.shape[0]
    if new_len == old_len:
        return seq.copy()
    x_old = np.linspace(0.0, 1.0, old_len)
    x_new = np.linspace(0.0, 1.0, new_len)
    new_seq = np.interp(x_new, x_old, seq)
    new_seq = np.maximum(new_seq, 0.0)
    new_seq /= new_seq.sum()
    return new_seq


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: int = 1000) -> list[float]:
    """
    Search for a non‑negative step function (list of heights) that
    minimises ``evaluate_sequence``.  The search respects the supplied
    time budget (default 1000 s) and returns a plain Python list.
    """
    rng = np.random.default_rng(seed)
    start = time.time()
    deadline = start + budget_s - 2.0          # a tiny safety margin

    overall_best_seq: np.ndarray | None = None
    overall_best_score = float("inf")
    top_candidates: list[tuple[float, np.ndarray]] = []

    # ------------------------------------------------------------------
    # 1. Diverse seed pool (moderate lengths)
    lengths = [1800, 2500, 3000, 3500, 4000, 5000, 6000]
    # a very wide p‑grid, extending into the extreme U‑shaped regime
    p_grid = np.concatenate(
        [
            np.logspace(-6, -3, 6),   # 1e-6 … 1e-3
            np.logspace(-3, -1, 8),   # 1e-3 … 1e-1
            np.logspace(-1, 0, 8),    # 1e-1 … 1
            np.logspace(0, 0.5, 6),   # 1 … 3.16
        ]
    )
    lam_grid = np.linspace(0.0, 0.95, 12)   # mixing parameter for uniform

    for n in lengths:
        if time.time() >= deadline:
            break
        print(f"[{time.time():.1f}] length {n} – seed generation")

        # deterministic bases
        for cand in (_distribution_uniform(n), _distribution_beta_sym(n, 0.5)):
            sc = _objective(cand)
            top_candidates.append((sc, cand.copy()))
            if sc < overall_best_score:
                overall_best_score, overall_best_seq = sc, cand.copy()
                print(f"[{time.time():.1f}] new global best (uniform/β=0.5) {sc:.6f}")

        # parameterised families
        for p in p_grid:
            if time.time() >= deadline:
                break
            beta_sym = _distribution_beta_sym(n, p)
            edge_inv = _distribution_edge_inversepow(n, p)

            for base in (beta_sym, edge_inv):
                for lam in lam_grid:
                    if time.time() >= deadline:
                        break
                    cand = _mix_with_uniform(base, lam)
                    sc = _objective(cand)
                    top_candidates.append((sc, cand.copy()))
                    if sc < overall_best_score:
                        overall_best_score, overall_best_seq = sc, cand.copy()
                        print(
                            f"[{time.time():.1f}] new best p={p:.2e} lam={lam:.2f} score={sc:.6f}"
                        )

        # two‑spike edge family
        for span_frac in [0.001, 0.002, 0.003, 0.005, 0.01]:
            span = max(1, int(span_frac * n))
            cand = _distribution_spike_edges(n, span)
            sc = _objective(cand)
            top_candidates.append((sc, cand.copy()))
            if sc < overall_best_score:
                overall_best_score, overall_best_seq = sc, cand.copy()
                print(f"[{time.time():.1f}] new best spikes span={span} score={sc:.6f}")

        # random Dirichlet seeds (wide range for concentration)
        for _ in range(6):
            if time.time() >= deadline:
                break
            # concentration α ∈ [1e‑4, 1] (log‑uniform)
            alpha = 10 ** rng.uniform(-4, 0)
            cand = rng.dirichlet(np.full(n, alpha)).astype(np.float64)
            sc = _objective(cand)
            top_candidates.append((sc, cand.copy()))
            if sc < overall_best_score:
                overall_best_score, overall_best_seq = sc, cand.copy()
                print(f"[{time.time():.1f}] new best Dirichlet α={alpha:.4g} score={sc:.6f}")

        # keep only a compact set of the best seeds
        top_candidates.sort(key=lambda x: x[0])
        top_candidates = top_candidates[:30]

    # ------------------------------------------------------------------
    # 2. Use a previously discovered construction if present
    try:
        prev = np.asarray(height_sequence_1, dtype=np.float64)  # type: ignore[name-defined]
        if prev.ndim == 1 and prev.size > 0:
            prev = np.maximum(prev, 0.0)
            prev /= prev.sum()
            sc = _objective(prev)
            if sc < overall_best_score:
                overall_best_score, overall_best_seq = sc, prev.copy()
                print(f"[{time.time():.1f}] previous best score={sc:.6f}")
            top_candidates.append((sc, prev.copy()))
    except Exception:
        pass

    # ------------------------------------------------------------------
    # 3. Fallback guard
    if overall_best_seq is None:
        n = 3000
        overall_best_seq = _distribution_uniform(n)
        overall_best_score = _objective(overall_best_seq)
        print(f"[{time.time():.1f}] fallback uniform score={overall_best_score:.6f}")

    # ------------------------------------------------------------------
    # 4. Heavy mirror‑descent refinement on a handful of top seeds
    remaining = deadline - time.time()
    if remaining > 30.0 and top_candidates:
        seeds = [seq for _, seq in sorted(top_candidates, key=lambda x: x[0])[:10]]
        per_seed_time = remaining * 0.55 / max(1, len(seeds))
        print(
            f"[{time.time():.1f}] heavy descent on {len(seeds)} seeds (~{per_seed_time:.1f}s each)"
        )
        for idx, seed_seq in enumerate(seeds):
            if time.time() >= deadline:
                break
            sub_deadline = time.time() + per_seed_time
            a_opt, sc_opt = _subgradient_descent_momentum(
                seed_seq.copy(),
                sub_deadline,
                best_seq=overall_best_seq,
                best_score=overall_best_score,
                lr0=0.9,
                active_frac=0.02,
                max_active=80,
                beta=0.9,
                eval_interval=150,
                patience=2000,
            )
            if sc_opt < overall_best_score:
                overall_best_score, overall_best_seq = sc_opt, a_opt.copy()
                print(
                    f"[{time.time():.1f}] after descent seed {idx} new global best {overall_best_score:.6f}"
                )

    # ------------------------------------------------------------------
    # 5. Upsample to a larger length if time permits
    remaining = deadline - time.time()
    if remaining > 150.0:
        target_len = min(25000, int(len(overall_best_seq) * 2))
        print(f"[{time.time():.1f}] up‑sampling to {target_len} points")
        up_seq = _upsample(overall_best_seq, target_len)

        sub_deadline = time.time() + remaining * 0.60
        up_seq_opt, up_score = _subgradient_descent_momentum(
            up_seq,
            sub_deadline,
            best_seq=overall_best_seq,
            best_score=overall_best_score,
            lr0=0.7,
            active_frac=0.03,
            max_active=70,
            beta=0.92,
            eval_interval=120,
            patience=2500,
        )
        if up_score < overall_best_score:
            overall_best_score, overall_best_seq = up_score, up_seq_opt.copy()
            print(
                f"[{time.time():.1f}] up‑sampled descent improved to {overall_best_score:.6f}"
            )

    # ------------------------------------------------------------------
    # 6. Deterministic pairwise refinement (greedy mass transfers)
    if time.time() < deadline:
        print(f"[{time.time():.1f}] deterministic pairwise refinement")
        overall_best_seq, overall_best_score = _pairwise_swap_local(
            overall_best_seq.copy(),
            deadline,
            best_seq=overall_best_seq,
            best_score=overall_best_score,
            max_trials=300_000,
        )

    # ------------------------------------------------------------------
    # 7. Final cheap simulated‑annealing clean‑up
    if time.time() < deadline:
        print(f"[{time.time():.1f}] final simulated‑annealing clean‑up")
        overall_best_seq, overall_best_score = _random_local_search(
            overall_best_seq.copy(),
            deadline,
            best_seq=overall_best_seq,
            best_score=overall_best_score,
            init_temp=0.009,
            decay=0.9994,
        )

    # ------------------------------------------------------------------
    print(
        f"Final best score: {overall_best_score:.6f} (length={len(overall_best_seq)})"
    )
    # Return a plain Python list – let ``evaluate_sequence`` perform its own clipping
    return overall_best_seq.tolist()
