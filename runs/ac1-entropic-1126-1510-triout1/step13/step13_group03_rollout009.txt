def _lbfgs_refine(a0, time_limit, best_seq, best_score):
    from scipy.optimize import minimize
    n = len(a0)
    def obj(x):
        # ensure sum to 1, but we enforce via constraints, so no scaling
        conv = _fft_self_convolve(x)
        return 2.0 * n * conv.max()
    # constraints
    cons = ({'type': 'eq', 'fun': lambda x: np.sum(x) - 1.0},)
    bounds = [(0.0, None)] * n
    # Use starting point best_seq
    res = minimize(obj, best_seq, method='SLSQP', bounds=bounds, constraints=cons,
                   options={'maxiter': 200, 'ftol': 1e-12, 'disp': False})
    if res.success:
        x_opt = res.x
        # enforce simplex
        x_opt = _project_onto_simplex(x_opt)
        score = obj(x_opt)
        if score < best_score - 1e-12:
            return x_opt, score
    return best_seq, best_score

"""
Propose a step‑function (non‑negative vector) that minimises 

    evaluate_sequence(seq) = 2·n·max(conv(seq,seq)) / (sum(seq)²)

The implementation follows three phases:

1️⃣  A fast sweep of many analytic families (uniform, arcsine, sine, cosine,
    quadratic, edge‑weighted, exponential‑edge, symmetric Beta) on several
    lengths.  The best few candidates are kept.

2️⃣  For the most promising seeds we run **smooth‑max mirror descent**:
    the L∞ objective is approximated by a soft‑max (parameter γ) and an
    exponentiated‑gradient update is performed on the simplex.  γ is
    increased from a gentle value (γ≈5) to a sharp one (γ≈200), which
    quickly reduces the maximal self‑convolution.

3️⃣  A final greedy pairwise mass‑transfer hill‑climber polishes the result.

All work respects the supplied time budget (default 1000 s).  The returned
list is a plain Python list; the evaluator will normalise it internally.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x ≥ 0, Σx = s}."""
    if s < 0:
        raise ValueError("Simplex radius must be non‑negative")
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.where(u - cssv / (np.arange(1, len(v) + 1)) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    return np.maximum(v - theta, 0.0)


# ----------------------------------------------------------------------
def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Full linear self‑convolution a*a using a real FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1) - 1).bit_length()   # next power of two
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[: 2 * n - 1]


def _fft_convolve(u: np.ndarray, v: np.ndarray) -> np.ndarray:
    """Full linear convolution of two real vectors using a real FFT."""
    m = len(u) + len(v) - 1
    size = 1 << (m - 1).bit_length()
    U = np.fft.rfft(u, n=size)
    V = np.fft.rfft(v, n=size)
    conv = np.fft.irfft(U * V, n=size)
    return conv[:m]


# ----------------------------------------------------------------------
def _objective(a: np.ndarray) -> float:
    """Exact objective 2·n·max(conv(a,a)) assuming Σa = 1."""
    n = a.shape[0]
    conv = _fft_self_convolve(a)
    return float(2.0 * n * conv.max())


# ----------------------------------------------------------------------
# Analytic/discrete families (all return a vector that already sums to 1)
def _distribution_uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _distribution_arcsine(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    vals /= vals.sum()
    return vals


def _distribution_sine(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64) + 0.5
    w = np.sin(np.pi * i / n)
    w /= w.sum()
    return w


def _distribution_cosine(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64) + 0.5
    w = 1.0 - np.cos(np.pi * i / n)
    w /= w.sum()
    return w


def _distribution_quadratic(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    x = (2.0 * i) / (n - 1) - 1.0
    w = 1.0 - x * x
    w = np.clip(w, 0.0, None)
    w /= w.sum()
    return w


def _distribution_edge_weighted(n: int, p: float) -> np.ndarray:
    """Symmetric edge‑heavy weights: w_i ∝ (i+½)^(‑p)·(n-i‑½)^(‑p)."""
    i = np.arange(n, dtype=np.float64) + 0.5
    j = (n - np.arange(n, dtype=np.float64) - 0.5)
    w = (i * j) ** (-p)
    s = w.sum()
    if s == 0.0:
        return _distribution_uniform(n)
    w /= s
    return w


def _distribution_exponential_edge(n: int, c: float) -> np.ndarray:
    """w_i ∝ exp(‑c·i) + exp(‑c·(n‑1‑i)).  Symmetric."""
    i = np.arange(n, dtype=np.float64)
    w = np.exp(-c * i) + np.exp(-c * (n - 1 - i))
    w /= w.sum()
    return w


def _distribution_beta(n: int, alpha: float) -> np.ndarray:
    """Symmetric Beta(α,α) density discretised on the centre of each cell."""
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    # avoid zero or one when alpha<1
    eps = np.finfo(float).eps
    x = np.clip(x, eps, 1 - eps)
    w = (x ** (alpha - 1)) * ((1 - x) ** (alpha - 1))
    w /= w.sum()
    return w


def _distribution_random_sparse(
    n: int, rng: np.random.Generator, density: float = 0.01
) -> np.ndarray:
    """Sparse random vector (symmetric) with a prescribed density of non‑zeros."""
    k = max(2, int(n * density))
    idx = rng.choice(n, size=k, replace=False)
    vals = np.zeros(n, dtype=np.float64)
    vals[idx] = rng.random(k)
    # symmetrise
    vals = (vals + vals[::-1]) / 2.0
    s = vals.sum()
    if s == 0.0:
        vals[0] = 1.0
        s = 1.0
    vals /= s
    return vals


def _mix_with_uniform(base: np.ndarray, lam: float) -> np.ndarray:
    """Convex combination λ·uniform + (1‑λ)·base (both already normalised)."""
    n = base.shape[0]
    uniform = np.full(n, 1.0 / n, dtype=np.float64)
    return lam * uniform + (1.0 - lam) * base


# ----------------------------------------------------------------------
# 2️⃣  Smooth‑max mirror descent (exponentiated‑gradient)
def _smooth_descent(
    a0: np.ndarray,
    time_limit: float,
    best_seq: np.ndarray,
    best_score: float,
    gamma_start: float = 5.0,
    gamma_end: float = 200.0,
    eta0: float = 0.6,
    eval_every: int = 50,
) -> tuple[np.ndarray, float]:
    """
    Mirror descent on the soft‑max approximation of max(conv).
    The update is a multiplicative (exponentiated‑gradient) step followed
    by Euclidean projection onto the simplex.  Symmetry is enforced after
    each projection.
    """
    a = a0.copy()
    n = a.shape[0]
    t = 0
    while time.time() < time_limit:
        t += 1
        # 1. convolution and soft‑max weights
        conv = _fft_self_convolve(a)
        max_conv = conv.max()
        # linear schedule for γ
        gamma = gamma_start + (gamma_end - gamma_start) * min(1.0, t / 2000.0)
        # shift by max to keep exponent non‑positive (numerical safety)
        w_raw = np.exp(gamma * (conv - max_conv))
        w = w_raw / w_raw.sum()          # ∑ w_k = 1

        # 2. gradient: ∂f/∂a_i = 2 * Σ_k w_k a_{k‑i}
        a_rev = a[::-1]
        conv_w_rev = _fft_convolve(w, a_rev)          # length 3n‑2
        grad = 2.0 * conv_w_rev[(n - 1) : (2 * n - 1)]

        # 3. EG update + simplex projection
        eta = eta0 / math.sqrt(t)
        a *= np.exp(-eta * grad)
        a = _project_onto_simplex(a, s=1.0)

        # 4. enforce symmetry (helps to keep convolution flat)
        a = (a + a[::-1]) * 0.5
        a = _project_onto_simplex(a, s=1.0)

        # 5. evaluate the exact objective occasionally
        if t % eval_every == 0:
            cur_score = 2.0 * n * max_conv
            if cur_score < best_score - 1e-14:
                best_score = cur_score
                best_seq = a.copy()
                print(
                    f"[{time.time():.1f}] smooth descent it={t} new global best {best_score:.6f}"
                )
    return best_seq, best_score


# ----------------------------------------------------------------------
# 3️⃣  Greedy pairwise mass‑transfer hill‑climber
def _pairwise_swap(
    a0: np.ndarray,
    time_limit: float,
    best_seq: np.ndarray,
    best_score: float,
    max_trials: int | None = None,
) -> tuple[np.ndarray, float]:
    """
    Randomly move a fraction of mass from one index to another.
    Accept only strictly improving moves.
    """
    a = a0.copy()
    n = a.shape[0]
    trials = 0
    while time.time() < time_limit:
        if max_trials is not None and trials >= max_trials:
            break
        trials += 1

        src = np.where(a > 1e-14)[0]
        if src.size == 0:
            break
        i = np.random.choice(src)
        # pick a different target index uniformly
        j = np.random.randint(n - 1)
        if j >= i:
            j += 1

        delta = a[i] * np.random.rand() * 0.45
        if delta < 1e-15:
            continue

        new_a = a.copy()
        new_a[i] -= delta
        new_a[j] += delta

        cur = _objective(new_a)
        if cur < best_score - 1e-14:
            a = new_a
            best_score = cur
            best_seq = a.copy()
            print(
                f"[{time.time():.1f}] swap i={i}→j={j} δ={delta:.6g} new best {best_score:.6f}"
            )
    return best_seq, best_score


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: int = 1000) -> list[float]:
    """
    Return a step‑function (plain Python list) that minimises
    evaluate_sequence within the given time budget.
    """
    rng = np.random.default_rng(seed)
    start = time.time()
    deadline = start + budget_s - 1.0   # safety margin

    overall_best_seq = None
    overall_best_score = float("inf")
    top_candidates: list[tuple[float, np.ndarray]] = []   # (score, seq)

    # ------------------------------------------------------------------
    # 1️⃣  Scan analytic families
    # ------------------------------------------------------------------
    lengths = [2000, 4000, 6000, 8000, 12000, 16000]
    # parameter grids
    p_grid = np.concatenate(
        [
            np.logspace(-3, -0.2, 20),   # extreme edge‑heavy
            np.logspace(-0.2, 0.5, 15),  # moderate
            np.linspace(0.5, 5.0, 12),   # heavier tails
        ]
    )
    lam_grid = np.linspace(0.0, 1.0, 11)       # mixing weight with uniform
    c_grid = np.logspace(-3, 0.5, 14)         # exponential‑edge decay
    beta_grid = np.linspace(0.05, 0.95, 19)   # symmetric Beta(α,α)

    for n in lengths:
        if time.time() >= deadline:
            break
        print(f"[{time.time():.1f}] scanning length {n}")

        # deterministic symmetric families
        for base_fn in (
            _distribution_uniform,
            _distribution_arcsine,
            _distribution_sine,
            _distribution_cosine,
            _distribution_quadratic,
        ):
            a = base_fn(n)
            sc = _objective(a)
            top_candidates.append((sc, a.copy()))
            if sc < overall_best_score:
                overall_best_score, overall_best_seq = sc, a.copy()
                print(f"[{time.time():.1f}] new global best (det) len={n} score={sc:.6f}")

        # edge‑weighted + uniform mixes
        for p in p_grid:
            if time.time() >= deadline:
                break
            edge = _distribution_edge_weighted(n, float(p))
            for lam in lam_grid:
                if time.time() >= deadline:
                    break
                a = _mix_with_uniform(edge, lam)
                sc = _objective(a)
                top_candidates.append((sc, a.copy()))
                if sc < overall_best_score:
                    overall_best_score, overall_best_seq = sc, a.copy()
                    print(
                        f"[{time.time():.1f}] len={n} edge p={p:.4g} lam={lam:.2f} score={sc:.6f}"
                    )

        # exponential‑edge family
        for c in c_grid:
            if time.time() >= deadline:
                break
            exp_edge = _distribution_exponential_edge(n, float(c))
            for lam in lam_grid:
                if time.time() >= deadline:
                    break
                a = _mix_with_uniform(exp_edge, lam)
                sc = _objective(a)
                top_candidates.append((sc, a.copy()))
                if sc < overall_best_score:
                    overall_best_score, overall_best_seq = sc, a.copy()
                    print(
                        f"[{time.time():.1f}] len={n} exp c={c:.4g} lam={lam:.2f} score={sc:.6f}"
                    )

        # symmetric beta family
        for alpha in beta_grid:
            if time.time() >= deadline:
                break
            a = _distribution_beta(n, float(alpha))
            sc = _objective(a)
            top_candidates.append((sc, a.copy()))
            if sc < overall_best_score:
                overall_best_score, overall_best_seq = sc, a.copy()
                print(
                    f"[{time.time():.1f}] len={n} beta α={alpha:.3f} score={sc:.6f}"
                )

        # random Dirichlet seeds (symmetric)
        for _ in range(6):
            if time.time() >= deadline:
                break
            alpha = rng.uniform(0.01, 2.0)
            a = rng.dirichlet(np.full(n, alpha))
            a = (a + a[::-1]) * 0.5
            a /= a.sum()
            sc = _objective(a)
            top_candidates.append((sc, a.copy()))
            if sc < overall_best_score:
                overall_best_score, overall_best_seq = sc, a.copy()
                print(f"[{time.time():.1f}] Dirichlet len={n} score={sc:.6f}")

        # sparse random seeds
        for _ in range(4):
            if time.time() >= deadline:
                break
            a = _distribution_random_sparse(n, rng, density=0.006)
            sc = _objective(a)
            top_candidates.append((sc, a.copy()))
            if sc < overall_best_score:
                overall_best_score, overall_best_seq = sc, a.copy()
                print(f"[{time.time():.1f}] sparse len={n} score={sc:.6f}")

        # keep a bounded pool of the best candidates seen so far
        top_candidates.sort(key=lambda x: x[0])
        top_candidates = top_candidates[:150]

    # ------------------------------------------------------------------
    # 2️⃣  Inject previous construction if available
    # ------------------------------------------------------------------
    try:
        prev = np.asarray(height_sequence_1, dtype=np.float64)  # type: ignore[name-defined]
        if prev.ndim == 1 and prev.size > 0:
            prev = np.maximum(prev, 0.0)
            if prev.sum() == 0.0:
                prev = _distribution_uniform(prev.size)
            else:
                prev = prev / prev.sum()
            # enforce symmetry (optional but often beneficial)
            prev = (prev + prev[::-1]) * 0.5
            prev /= prev.sum()
            sc = _objective(prev)
            if sc < overall_best_score:
                overall_best_score, overall_best_seq = sc, prev.copy()
                print(f"[{time.time():.1f}] previous construction score={sc:.6f}")
            top_candidates.append((sc, prev.copy()))
    except Exception:
        pass

    # ------------------------------------------------------------------
    # 3️⃣  Fallback (should never be needed)
    # ------------------------------------------------------------------
    if overall_best_seq is None:
        n = 4000
        overall_best_seq = _distribution_uniform(n)
        overall_best_score = _objective(overall_best_seq)
        print(
            f"[{time.time():.1f}] fallback uniform length={n} score={overall_best_score:.6f}"
        )

    # ------------------------------------------------------------------
    # 4️⃣  Smooth‑max mirror‑descent on the most promising seeds
    # ------------------------------------------------------------------
    remaining = deadline - time.time()
    if remaining > 30.0 and top_candidates:
        # take the 8 best seeds for deeper optimisation
        seeds = [seq for _, seq in sorted(top_candidates, key=lambda x: x[0])[:8]]
        per_seed = remaining / max(1, len(seeds))
        print(f"[{time.time():.1f}] start smooth descent on {len(seeds)} seeds")
        for idx, seed_seq in enumerate(seeds):
            if time.time() >= deadline:
                break
            sg_deadline = time.time() + per_seed
            a_opt, sc_opt = _smooth_descent(
                seed_seq,
                sg_deadline,
                best_seq=overall_best_seq,
                best_score=overall_best_score,
                eta0=0.6,
                eval_every=100,
            )
            if sc_opt < overall_best_score - 1e-12:
                overall_best_score, overall_best_seq = sc_opt, a_opt.copy()
                print(
                    f"[{time.time():.1f}] after smooth seed {idx} new global best {overall_best_score:.6f}"
                )

    # ------------------------------------------------------------------
    # 5️⃣  Final greedy pairwise polishing
    # ------------------------------------------------------------------
    if time.time() < deadline:
        print(f"[{time.time():.1f}] final greedy pairwise polishing")
        a_opt, sc_opt = _pairwise_swap(
            overall_best_seq.copy(),
            deadline,
            best_seq=overall_best_seq,
            best_score=overall_best_score,
        )
        if sc_opt < overall_best_score - 1e-12:
            overall_best_score, overall_best_seq = sc_opt, a_opt.copy()
            print(f"[{time.time():.1f}] after final swaps new best {overall_best_score:.6f}")

    # ------------------------------------------------------------------
    print(
        f"Final best objective (2·n·max(conv)): {overall_best_score:.6f} (length={len(overall_best_seq)})"
    )
    # The evaluator will renormalise internally; returning a plain Python list is enough.
    return overall_best_seq.tolist()
