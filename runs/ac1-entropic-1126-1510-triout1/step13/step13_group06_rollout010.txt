def _adam_softmax_descent(a, deadline, best_seq, best_score, ...):
    n = a.shape[0]; t = 0
    m = np.zeros_like(a); v = np.zeros_like(a)
    beta1, beta2 = 0.9, 0.999
    epsilon = 1e-8
    eta_base = 0.5 (learning rate)
    tau = tau0
    no_improve = 0
    while time < deadline:
        t += 1
        conv = _fft_self_convolve(a)
        w = _softmax_weights(conv, tau)
        grad = _weighted_gradient(a, w)  # shape n
        # Adam updates
        m = beta1*m + (1-beta1)*grad
        v = beta2*v + (1-beta2)*(grad**2)
        m_hat = m/(1-beta1**t)
        v_hat = v/(1-beta2**t)
        step = eta_base/np.sqrt(t)  # maybe decreasing
        a = a - step * m_hat/(np.sqrt(v_hat)+epsilon)
        # clip negative
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s > 0:
            a = a / s
        else:
            a = np.full(n, 1.0/n)
        # increase tau

"""Search for a low‑score step function using Adam‑based soft‑max descent.

The algorithm:
1. Generate a diverse pool of candidate distributions (uniform, arcsine,
   Beta, edge‑weighted, sine‑weighted, sparse, Dirichlet).  All candidates are
   normalised to sum = 1 and evaluated with the exact objective.
2. Keep the best few candidates (across several sequence lengths) and
   refine each of them with a projected Adam optimiser that minimises a
   temperature‑controlled soft‑max surrogate of the max‑convolution.
   The temperature is increased gradually so the surrogate converges to the
   true max.
3. After the continuous optimisation, perform cheap deterministic pairwise
   mass swaps followed by a final simulated‑annealing walk.
4. Return the best sequence found within the time budget.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x≥0, Σx = s}."""
    if s < 0:
        raise ValueError("simplex radius must be non‑negative")
    # Fast special case
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho_arr = np.where(u - cssv / (np.arange(1, len(v) + 1)) > 0)[0]
    if rho_arr.size == 0:
        theta = 0.0
    else:
        rho = rho_arr[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Full self‑convolution a*a using a real FFT (returns length 2n‑1)."""
    n = a.shape[0]
    need = 2 * n - 1
    size = 1 << ((need - 1).bit_length())
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[:need]


# ----------------------------------------------------------------------
def _softmax_weights(conv: np.ndarray, tau: float) -> np.ndarray:
    """Soft‑max weights of τ·conv (stable). Returns a probability vector."""
    max_c = conv.max()
    # To keep numbers in a safe range we shift by the max before exponentiation
    w_raw = np.exp(tau * (conv - max_c))
    total = w_raw.sum()
    if total == 0.0 or np.isnan(total):
        # fallback to uniform
        return np.full_like(conv, 1.0 / conv.size)
    return w_raw / total


# ----------------------------------------------------------------------
def _weighted_gradient(a: np.ndarray, w: np.ndarray) -> np.ndarray:
    """
    Gradient of   L(a)= Σ_i w_i·(a*a)[i]   w.r.t a.
    Closed‑form:   ∇L = 2 * (a_rev ⊛ w)   (restricted to the first n entries).
    """
    n = a.shape[0]
    a_rev = a[::-1]
    need = len(w) + n - 1               # = 3·n‑2
    size = 1 << ((need - 1).bit_length())
    A_rev = np.fft.rfft(a_rev, n=size)
    W = np.fft.rfft(w, n=size)
    conv = np.fft.irfft(A_rev * W, n=size)
    conv = conv[:need]
    grad = 2.0 * conv[n - 1 : n - 1 + n]
    return grad


# ----------------------------------------------------------------------
def _objective(a: np.ndarray) -> float:
    """Exact objective 2·n·max(conv(a,a)) for a vector normalised to sum = 1."""
    n = a.shape[0]
    b = _fft_self_convolve(a)
    return float(2.0 * n * b.max())


# ----------------------------------------------------------------------
# Analytic families ----------------------------------------------------
def _distribution_uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _distribution_arcsine(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    vals /= vals.sum()
    return vals


def _distribution_beta_sym(n: int, p: float) -> np.ndarray:
    """Symmetric beta (Beta(p,p)) on the discrete grid."""
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    # use log‑space to avoid overflow for extreme p
    logw = (p - 1) * np.log(x) + (p - 1) * np.log(1.0 - x)
    w = np.exp(logw - np.max(logw))  # shift for stability
    w_sum = w.sum()
    if w_sum == 0.0:
        return np.full_like(w, 1.0 / n)
    return w / w_sum


def _distribution_beta_asym(n: int, a: float, b: float) -> np.ndarray:
    """Asymmetric beta (Beta(a,b)) on the discrete grid."""
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    logw = (a - 1) * np.log(x) + (b - 1) * np.log(1.0 - x)
    w = np.exp(logw - np.max(logw))
    w_sum = w.sum()
    if w_sum == 0.0:
        return np.full_like(w, 1.0 / n)
    return w / w_sum


def _distribution_edge_weighted(n: int, p: float) -> np.ndarray:
    """U‑shaped family: w_i ∝ ((i+0.5)*(n‑i‑0.5))^(−p)."""
    i = np.arange(n, dtype=np.float64) + 0.5
    j = (n - np.arange(n, dtype=np.float64) - 0.5)
    w = (i * j) ** (-p)
    w_sum = w.sum()
    if w_sum == 0.0:
        return np.full_like(w, 1.0 / n)
    return w / w_sum


def _distribution_sine_weighted(n: int, p: float) -> np.ndarray:
    """Weight proportional to sin(π (i+½)/n)^p."""
    theta = np.pi * (np.arange(n, dtype=np.float64) + 0.5) / n
    w = np.sin(theta) ** p
    w_sum = w.sum()
    if w_sum == 0.0:
        return np.full_like(w, 1.0 / n)
    return w / w_sum


def _distribution_random_sparse(
    n: int, rng: np.random.Generator, density: float = 0.01
) -> np.ndarray:
    """Sparse random vector with a given density of non‑zeros."""
    k = max(2, int(n * density))
    idx = rng.choice(n, size=k, replace=False)
    vals = np.zeros(n, dtype=np.float64)
    vals[idx] = rng.random(k)
    if vals.sum() == 0.0:
        vals[0] = 1.0
    vals /= vals.sum()
    return vals


def _mix_with_uniform(base: np.ndarray, lam: float) -> np.ndarray:
    """Convex combination lam·U + (1‑lam)·base, where U is uniform."""
    n = base.shape[0]
    uniform = np.full(n, 1.0 / n, dtype=np.float64)
    return lam * uniform + (1.0 - lam) * base


# ----------------------------------------------------------------------
def _adam_softmax_descent(
    a: np.ndarray,
    deadline: float,
    best_seq: np.ndarray,
    best_score: float,
    eta0: float = 0.2,
    tau0: float = 0.3,
    tau_mult: float = 1.012,
    eval_interval: int = 150,
    patience: int = 2500,
) -> tuple[np.ndarray, float]:
    """
    Projected Adam optimiser for the soft‑max surrogate of max‑convolution.
    The temperature ``tau`` starts low (smooth) and is multiplied by
    ``tau_mult`` each iteration, gradually turning the surrogate into the true
    max.  ``eta0`` controls the base learning rate; it decays as 1/√t.
    """
    n = a.shape[0]
    t = 0
    m = np.zeros_like(a)      # first moment
    v = np.zeros_like(a)      # second moment
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-12
    tau = tau0
    no_improve = 0
    rng = np.random.default_rng()

    while time.time() < deadline and t < 250_000:
        t += 1

        # convolution and soft‑max weights
        conv = _fft_self_convolve(a)
        w = _softmax_weights(conv, tau)

        # gradient of the surrogate loss
        grad = _weighted_gradient(a, w)

        # Adam moment updates
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)

        # bias‑correction
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)

        # step size schedule
        eta = eta0 / math.sqrt(t)

        # additive Adam update followed by projection onto the simplex
        a = a - eta * m_hat / (np.sqrt(v_hat) + eps)
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s > 0.0:
            a = a / s
        else:
            a = np.full(n, 1.0 / n, dtype=np.float64)

        # increase temperature → focus more on the true max
        tau = min(tau * tau_mult, 50.0)   # cap to avoid overflow

        # intermittent exact evaluation
        if t % eval_interval == 0:
            cur = _objective(a)
            if cur < best_score - 1e-14:
                best_score = cur
                best_seq = a.copy()
                no_improve = 0
                print(f"[{time.time():.1f}] Adam it={t} new best {best_score:.6f}")
            else:
                no_improve += 1

        # jitter if stagnating
        if no_improve >= patience:
            jitter = rng.exponential(scale=0.0025, size=n)
            a = a + jitter
            a = np.maximum(a, 0.0)
            a = a / a.sum()
            no_improve = 0

    return best_seq, best_score


# ----------------------------------------------------------------------
def _pairwise_swap_local(
    a: np.ndarray,
    deadline: float,
    best_seq: np.ndarray,
    best_score: float,
    max_trials: int | None = None,
) -> tuple[np.ndarray, float]:
    """Randomly transfer a small chunk of mass; keep change if objective improves."""
    n = a.shape[0]
    trials = 0
    rng = np.random.default_rng()
    while time.time() < deadline:
        if max_trials is not None and trials >= max_trials:
            break
        trials += 1

        src = np.where(a > 1e-12)[0]
        if src.size == 0:
            break
        i = rng.choice(src)

        j = rng.integers(n - 1)
        if j >= i:
            j += 1

        delta = a[i] * rng.random() * 0.35
        if delta < 1e-14:
            continue

        new_a = a.copy()
        new_a[i] -= delta
        new_a[j] += delta

        cur = _objective(new_a)
        if cur < best_score - 1e-14:
            a = new_a
            best_score = cur
            best_seq = a.copy()
            print(
                f"[{time.time():.1f}] swap i={i}→j={j} δ={delta:.3e} new best {best_score:.6f}"
            )
    return best_seq, best_score


# ----------------------------------------------------------------------
def _random_local_search(
    a: np.ndarray,
    deadline: float,
    best_seq: np.ndarray,
    best_score: float,
    init_temp: float = 1e-2,
    decay: float = 0.9996,
) -> tuple[np.ndarray, float]:
    """Cheap simulated‑annealing style walk that transfers random mass."""
    n = a.shape[0]
    rng = np.random.default_rng()
    T = init_temp
    while time.time() < deadline:
        i = rng.integers(n)
        j = rng.integers(n - 1)
        if j >= i:
            j += 1
        delta = a[i] * rng.random() * 0.3
        if delta < 1e-14:
            continue
        new_a = a.copy()
        new_a[i] -= delta
        new_a[j] += delta
        cur = _objective(new_a)
        if cur < best_score - 1e-14:
            a = new_a
            best_score = cur
            best_seq = a.copy()
            print(f"[{time.time():.1f}] SA improve {i}->{j} Δ={delta:.3e} best {best_score:.6f}")
        else:
            if rng.random() < math.exp(-(cur - best_score) / max(T, 1e-12)):
                a = new_a
        T *= decay
    return best_seq, best_score

                    if time.time() >= deadline:

                    cand = _mix_with_uniform(base, lam)
                    sc = _objective(cand)
                    top_candidates.append((sc, cand.copy()))
                    if sc < overall_best_score:

        for _ in range(4):
            if time.time() >= deadline:

            cand = _distribution_beta_asym(n, a_param, b_param)
            sc = _objective(cand)
            top_candidates.append((sc, cand.copy()))
            if sc < overall_best_score:
                overall_best_score, overall_best_seq = sc, cand.copy()
                print(
                    f"[{time.time():.1f}] asym‑beta len={n} a={a_param:.3g} b={b_param:.3g} score={sc:.6f}"
                )

        # random Dirichlet seeds
        for _ in range(6):
            if time.time() >= deadline:
                break
            alpha = rng.uniform(0.02, 1.5)
            cand = rng.dirichlet(np.full(n, alpha))
            sc = _objective(cand)
            top_candidates.append((sc, cand.copy()))
            if sc < overall_best_score:
                overall_best_score, overall_best_seq = sc, cand.copy()

"""Search for a low‑score step function using Adam‑based soft‑max descent.

The algorithm:
1. Generate a large, diverse pool of candidate distributions (uniform,
   arcsine, Beta, edge‑weighted, sine‑weighted, sparse, Dirichlet) for several
   sequence lengths.  All candidates are normalised to sum = 1 and evaluated
   with the exact objective.
2. Keep the best few candidates and refine each of them with a projected
   Adam optimiser that minimises a temperature‑controlled soft‑max surrogate of
   the max‑convolution.  The temperature starts low (smooth) and is multiplied
   by a factor each iteration, gradually turning the surrogate into the true
   max.  The Adam step size decays as 1/√t.
3. Apply a cheap deterministic pairwise‑swap local search and a final
   simulated‑annealing walk to fine‑tune the best solution.
4. Return the best sequence found within the time budget.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Euclidean projection of ``v`` onto the simplex {x ≥ 0, Σx = s}."""
    if s < 0:
        raise ValueError("simplex radius must be non‑negative")
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho_arr = np.where(u - cssv / (np.arange(1, len(v) + 1)) > 0)[0]
    if rho_arr.size == 0:
        theta = 0.0
    else:
        rho = rho_arr[-1]
        theta = cssv[rho] / (rho + 1)
    return np.maximum(v - theta, 0.0)


# ----------------------------------------------------------------------
def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Full self‑convolution ``a*a`` using a real FFT (returns length 2n‑1)."""
    n = a.shape[0]
    need = 2 * n - 1
    size = 1 << ((need - 1).bit_length())
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[:need]


# ----------------------------------------------------------------------
def _softmax_weights(conv: np.ndarray, tau: float) -> np.ndarray:
    """Soft‑max weights of τ·conv (stable). Returns a probability vector."""
    max_c = conv.max()
    w_raw = np.exp(tau * (conv - max_c))
    total = w_raw.sum()
    if total == 0.0 or np.isnan(total):
        return np.full_like(conv, 1.0 / conv.size)
    return w_raw / total


# ----------------------------------------------------------------------
def _weighted_gradient(a: np.ndarray, w: np.ndarray) -> np.ndarray:
    """
    Gradient of L(a) = Σ_i w_i·(a*a)[i] w.r.t. ``a``.
    Closed‑form: ∇L = 2·(a_rev ⊛ w) restricted to the first n entries.
    """
    n = a.shape[0]
    a_rev = a[::-1]
    need = len(w) + n - 1               # = 3·n‑2
    size = 1 << ((need - 1).bit_length())
    A_rev = np.fft.rfft(a_rev, n=size)
    W = np.fft.rfft(w, n=size)
    conv = np.fft.irfft(A_rev * W, n=size)
    conv = conv[:need]
    grad = 2.0 * conv[n - 1 : n - 1 + n]
    return grad


# ----------------------------------------------------------------------
def _objective(a: np.ndarray) -> float:
    """Exact objective 2·n·max(conv(a,a)) for a vector normalised to sum = 1."""
    n = a.shape[0]
    b = _fft_self_convolve(a)
    return float(2.0 * n * b.max())


# ----------------------------------------------------------------------
# Analytic families ----------------------------------------------------
def _distribution_uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _distribution_arcsine(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    vals /= vals.sum()
    return vals


def _distribution_beta_sym(n: int, p: float) -> np.ndarray:
    """Symmetric beta (Beta(p,p)) on the discrete grid."""
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    logw = (p - 1) * np.log(x) + (p - 1) * np.log(1.0 - x)
    w = np.exp(logw - np.max(logw))
    s = w.sum()
    if s == 0.0:
        return np.full_like(w, 1.0 / n)
    return w / s


def _distribution_beta_asym(n: int, a: float, b: float) -> np.ndarray:
    """Asymmetric beta (Beta(a,b)) on the discrete grid."""
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    logw = (a - 1) * np.log(x) + (b - 1) * np.log(1.0 - x)
    w = np.exp(logw - np.max(logw))
    s = w.sum()
    if s == 0.0:
        return np.full_like(w, 1.0 / n)
    return w / s


def _distribution_edge_weighted(n: int, p: float) -> np.ndarray:
    """U‑shaped family: w_i ∝ ((i+0.5)*(n‑i‑0.5))^(−p)."""
    i = np.arange(n, dtype=np.float64) + 0.5
    j = (n - np.arange(n, dtype=np.float64) - 0.5)
    w = (i * j) ** (-p)
    s = w.sum()
    if s == 0.0:
        return np.full_like(w, 1.0 / n)
    return w / s


def _distribution_sine_weighted(n: int, p: float) -> np.ndarray:
    """Weight proportional to sin(π (i+½)/n)^p."""
    theta = np.pi * (np.arange(n, dtype=np.float64) + 0.5) / n
    w = np.sin(theta) ** p
    s = w.sum()
    if s == 0.0:
        return np.full_like(w, 1.0 / n)
    return w / s


def _distribution_random_sparse(
    n: int, rng: np.random.Generator, density: float = 0.01
) -> np.ndarray:
    """Sparse random vector with a given density of non‑zeros."""
    k = max(2, int(n * density))
    idx = rng.choice(n, size=k, replace=False)
    vals = np.zeros(n, dtype=np.float64)
    vals[idx] = rng.random(k)
    if vals.sum() == 0.0:
        vals[0] = 1.0
    vals /= vals.sum()
    return vals


def _mix_with_uniform(base: np.ndarray, lam: float) -> np.ndarray:
    """Convex combination lam·U + (1‑lam)·base, where U is uniform."""
    n = base.shape[0]
    uniform = np.full(n, 1.0 / n, dtype=np.float64)
    return lam * uniform + (1.0 - lam) * base


# ----------------------------------------------------------------------
def _adam_softmax_descent(
    a: np.ndarray,
    deadline: float,
    best_seq: np.ndarray,
    best_score: float,
    eta0: float = 0.2,
    tau0: float = 0.3,
    tau_mult: float = 1.012,
    eval_interval: int = 150,
    patience: int = 2500,
) -> tuple[np.ndarray, float]:
    """
    Projected Adam optimiser for the soft‑max surrogate of ``max(conv(a,a))``.
    ``tau`` starts low (smooth) and is multiplied by ``tau_mult`` each
    iteration, gradually focusing the surrogate onto the true max.
    """
    n = a.shape[0]
    t = 0
    m = np.zeros_like(a)          # first moment
    v = np.zeros_like(a)          # second moment
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-12
    tau = tau0
    no_improve = 0
    rng = np.random.default_rng()

    while time.time() < deadline and t < 250_000:
        t += 1

        conv = _fft_self_convolve(a)
        w = _softmax_weights(conv, tau)
        grad = _weighted_gradient(a, w)

        # Adam moment updates
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)

        # bias correction
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)

        # learning‑rate schedule
        eta = eta0 / math.sqrt(t)

        # additive Adam step followed by projection onto the simplex
        a = a - eta * m_hat / (np.sqrt(v_hat) + eps)
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s > 0.0:
            a = a / s
        else:
            a = np.full(n, 1.0 / n, dtype=np.float64)

        # increase temperature → surrogate becomes sharper
        tau = min(tau * tau_mult, 50.0)

        # occasional exact evaluation
        if t % eval_interval == 0:
            cur = _objective(a)
            if cur < best_score - 1e-14:
                best_score = cur
                best_seq = a.copy()
                no_improve = 0
                print(f"[{time.time():.1f}] Adam it={t} new best {best_score:.6f}")
            else:
                no_improve += 1

        # jitter if we seem stuck
        if no_improve >= patience:
            jitter = rng.exponential(scale=0.0025, size=n)
            a = a + jitter
            a = np.maximum(a, 0.0)
            a = a / a.sum()
            no_improve = 0

    return best_seq, best_score


# ----------------------------------------------------------------------
def _pairwise_swap_local(
    a: np.ndarray,
    deadline: float,
    best_seq: np.ndarray,
    best_score: float,
    max_trials: int | None = None,
) -> tuple[np.ndarray, float]:
    """Randomly transfer a small chunk of mass; keep the change if it improves."""
    n = a.shape[0]
    trials = 0
    rng = np.random.default_rng()
    while time.time() < deadline:
        if max_trials is not None and trials >= max_trials:
            break
        trials += 1

        src = np.where(a > 1e-12)[0]
        if src.size == 0:
            break
        i = rng.choice(src)

        j = rng.integers(n - 1)
        if j >= i:
            j += 1

        delta = a[i] * rng.random() * 0.35
        if delta < 1e-14:
            continue

        new_a = a.copy()
        new_a[i] -= delta
        new_a[j] += delta

        cur = _objective(new_a)
        if cur < best_score - 1e-14:
            a = new_a
            best_score = cur
            best_seq = a.copy()
            print(
                f"[{time.time():.1f}] swap i={i}→j={j} δ={delta:.3e} new best {best_score:.6f}"
            )
    return best_seq, best_score


# ----------------------------------------------------------------------
def _random_local_search(
    a: np.ndarray,
    deadline: float,
    best_seq: np.ndarray,
    best_score: float,
    init_temp: float = 1e-2,
    decay: float = 0.9996,
) -> tuple[np.ndarray, float]:
    """Very cheap simulated‑annealing walk that transfers random mass."""
    n = a.shape[0]
    rng = np.random.default_rng()
    T = init_temp
    while time.time() < deadline:
        i = rng.integers(n)
        j = rng.integers(n - 1)
        if j >= i:
            j += 1
        delta = a[i] * rng.random() * 0.3
        if delta < 1e-14:
            continue
        new_a = a.copy()
        new_a[i] -= delta
        new_a[j] += delta
        cur = _objective(new_a)
        if cur < best_score - 1e-14:
            a = new_a
            best_score = cur
            best_seq = a.copy()
            print(f"[{time.time():.1f}] SA improve {i}->{j} Δ={delta:.3e} best {best_score:.6f}")
        else:
            if rng.random() < math.exp(-(cur - best_score) / max(T, 1e-12)):
                a = new_a
        T *= decay
    return best_seq, best_score
