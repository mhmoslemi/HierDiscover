"""
Propose a step‑function (non‑negative list) that minimizes

    2 * n * max(conv(a, a)) / (sum(a)^2)

using a multi‑stage search:

1.  Generate a diverse small pool of analytic and random seeds
    (uniform, arcsine, edge‑weighted, Dirichlet, sparse).

2.  Refine each seed with a *smooth* exponential‑gradient descent.
    The objective max‑convolution is approximated by the
    log‑sum‑exp of all convolution entries.  The exact gradient of the
    smoothed objective is obtained via FFT in O(n log n) time.
    The smoothing parameter α is increased gradually (homotopy) so the
    iterates first see a smooth landscape and finally focus on the true
    max‑convolution.

3.  A short greedy pairwise polishing (mass transfer) step is applied
    to the best solution.

All updates keep the vector non‑negative and renormalised to sum = 1,
so the final call to ``evaluate_sequence`` is stable.  The routine
respects the supplied time budget (default 1000 s) and prints progress.
"""

import time
import math
import numpy as np


# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x≥0, Σx=s}."""
    if s < 0:
        raise ValueError("Simplex radius must be non‑negative")
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.where(u - cssv / (np.arange(1, len(v) + 1)) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    return np.maximum(v - theta, 0.0)


# ----------------------------------------------------------------------
def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Full self‑convolution a*a using a real FFT (length 2·n‑1)."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1) - 1).bit_length()  # next power of two
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[: 2 * n - 1]


# ----------------------------------------------------------------------
def _objective(a: np.ndarray) -> float:
    """Exact objective 2·n·max(conv(a,a)) assuming Σa = 1."""
    n = a.shape[0]
    b = _fft_self_convolve(a)
    return float(2.0 * n * b.max())


# ----------------------------------------------------------------------
# Analytic families (all return a probability vector, i.e. sum=1)
def _distribution_uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _distribution_arcsine(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    vals /= vals.sum()
    return vals


def _distribution_edge_weighted(n: int, p: float) -> np.ndarray:
    """Edge‑weighted family: w_i ∝ ((i+½)*(n‑i‑½))^{‑p}."""
    i = np.arange(n, dtype=np.float64) + 0.5
    j = (n - np.arange(n, dtype=np.float64) - 0.5)
    w = (i * j) ** (-p)
    w = np.clip(w, 0.0, None)
    total = w.sum()
    if total > 0:
        w /= total
    else:
        w = np.full(n, 1.0 / n)
    return w


def _distribution_random_sparse(
    n: int, rng: np.random.Generator, density: float = 0.01
) -> np.ndarray:
    """Sparse random vector with a prescribed density of non‑zeros."""
    k = max(2, int(n * density))
    idx = rng.choice(n, size=k, replace=False)
    vals = np.zeros(n, dtype=np.float64)
    vals[idx] = rng.random(k)
    if vals.sum() == 0.0:
        vals[0] = 1.0
    vals /= vals.sum()
    return vals


# ----------------------------------------------------------------------
def _smooth_gradient(a: np.ndarray, alpha: float) -> np.ndarray:
    """
    Gradient of the smoothed objective
        f_α(a) = (1/α) log Σ_k exp(α·conv_k)
    where conv = a * a (full convolution).
    Returns a vector of the same size as a.
    """
    n = a.shape[0]
    conv = _fft_self_convolve(a)                     # length 2n‑1
    max_c = conv.max()
    # stable soft‑max weights
    w = np.exp(alpha * (conv - max_c))
    w_sum = w.sum()
    if w_sum == 0.0:
        w_norm = np.full_like(w, 1.0 / w.size)
    else:
        w_norm = w / w_sum

    # gradient_i = 2 * Σ_j a_j * w_{i+j}
    # compute via convolution of a with reversed weight vector
    w_rev = w_norm[::-1]                              # length 2n‑1
    size = 1 << ((n + w_rev.size - 1) - 1).bit_length()
    A = np.fft.rfft(a, n=size)
    W = np.fft.rfft(w_rev, n=size)
    conv_rev_full = np.fft.irfft(A * W, n=size)       # length = size
    # The needed slice is indices n‑1 … 2n‑2 (inclusive)
    start = n - 1
    end = start + n
    needed = conv_rev_full[start:end]                 # length n
    grad = 2.0 * needed[::-1]                         # reverse to get i‑ordering
    return grad


# ----------------------------------------------------------------------
def _smooth_exp_descent(
    a0: np.ndarray,
    time_limit: float,
    best_seq: np.ndarray,
    best_score: float,
) -> tuple[np.ndarray, float]:
    """
    Exponentiated‑gradient descent on the *smoothed* max‑convolution.
    The smoothing parameter α starts small and is multiplied by 1.5 every
    200 iterations (capped at 20).  The step‑size decays as η₀/√t.
    """
    a = a0.copy()
    t = 0
    alpha = 0.5
    eta0 = 0.12
    while time.time() < time_limit:
        if t % 200 == 0 and alpha < 20.0:
            alpha = min(alpha * 1.5, 20.0)

        grad = _smooth_gradient(a, alpha)
        eta = eta0 / math.sqrt(t + 1)
        # multiplicative update keeps positivity
        a = a * np.exp(-eta * grad)
        s = a.sum()
        if s > 0.0:
            a = a / s
        else:
            a = _distribution_uniform(len(a))

        cur = _objective(a)
        if cur < best_score - 1e-14:
            best_score = cur
            best_seq = a.copy()
            print(
                f"[{time.time():.1f}] smooth‑exp it={t} α={alpha:.2f} new best {best_score:.6f}"
            )
        t += 1
    return best_seq, best_score


# ----------------------------------------------------------------------
def _pairwise_swap(
    a0: np.ndarray,
    time_limit: float,
    best_seq: np.ndarray,
    best_score: float,
    max_trials: int | None = None,
) -> tuple[np.ndarray, float]:
    """
    Randomly move a fraction of mass from one index to another.
    Only strictly improving moves are kept.  Simple hill‑climbing.
    """
    a = a0.copy()
    n = a.shape[0]
    trials = 0
    while time.time() < time_limit:
        if max_trials is not None and trials >= max_trials:
            break
        trials += 1

        src = np.where(a > 1e-14)[0]
        if src.size == 0:
            break
        i = np.random.choice(src)
        j = np.random.randint(n - 1)
        if j >= i:
            j += 1

        delta = a[i] * np.random.rand() * 0.45
        if delta < 1e-15:
            continue

        new_a = a.copy()
        new_a[i] -= delta
        new_a[j] += delta

        cur = _objective(new_a)
        if cur < best_score - 1e-14:
            a = new_a
            best_score = cur
            best_seq = a.copy()
            print(
                f"[{time.time():.1f}] swap i={i}→j={j} δ={delta:.6g} new best {best_score:.6f}"
            )
    return best_seq, best_score


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: int = 1000) -> list[float]:
    """
    Search for a non‑negative step‑function (list of floats) that
    minimises ``evaluate_sequence``.  The routine respects the supplied
    time budget and returns the best sequence found.
    """
    rng = np.random.default_rng(seed)
    start = time.time()
    deadline = start + budget_s - 2.0  # safety margin

    # ---- 1. collect a modest pool of seeds ---------------------------
    seed_vectors = []

    # a) previous construction if available
    try:
        prev = np.asarray(height_sequence_1, dtype=np.float64)  # type: ignore[name-defined]
        prev = np.maximum(prev, 0.0)
        if prev.sum() > 0.0:
            prev = prev / prev.sum()
        else:
            prev = _distribution_uniform(len(prev))
        seed_vectors.append(prev)
    except Exception:
        pass

    # b) analytic families (few lengths, few parameters)
    lengths = [2500, 3000, 3500, 4000]
    for n in lengths:
        seed_vectors.append(_distribution_uniform(n))
        seed_vectors.append(_distribution_arcsine(n))
        for p in (0.5, 0.8, 1.2, 2.0):
            seed_vectors.append(_distribution_edge_weighted(n, p))

    # c) random Dirichlet seeds
    for n in [3000, 3500]:
        for _ in range(3):
            alpha = rng.uniform(0.05, 2.0)
            a = rng.dirichlet(np.full(n, alpha))
            seed_vectors.append(a)

    # d) sparse random seeds
    for n in [3000, 3500]:
        for _ in range(2):
            a = _distribution_random_sparse(n, rng, density=0.008)
            seed_vectors.append(a)

    # ---- evaluate seeds and keep the most promising ------------------
    scored = []
    for vec in seed_vectors:
        if time.time() >= deadline:
            break
        # ensure proper normalisation
        total = vec.sum()
        if total > 0.0:
            vec = vec / total
        else:
            vec = _distribution_uniform(len(vec))
        sc = _objective(vec)
        scored.append((sc, vec.copy()))
        if sc < 1e9:  # just to silence any unforeseen issues
            pass

    if not scored:
        # fallback to uniform of a modest size
        n = 3000
        best_seq = _distribution_uniform(n)
        best_score = _objective(best_seq)
        print(f"[{time.time():.1f}] fallback uniform, score={best_score:.6f}")
    else:
        scored.sort(key=lambda x: x[0])
        best_score, best_seq = scored[0][0], scored[0][1].copy()
        print(f"[{time.time():.1f}] initial best score={best_score:.6f}")

    # keep top few seeds for intensive refinement
    top_k = 6
    top_candidates = [vec for _, vec in scored[:top_k]]

    # ---- 2. smooth exponential‑gradient refinement --------------------
    remaining = deadline - time.time()
    if remaining > 30.0 and top_candidates:
        per_seed = remaining / max(1, len(top_candidates))
        print(f"[{time.time():.1f}] smooth‑exp refinement on {len(top_candidates)} seeds")
        for idx, seed_vec in enumerate(top_candidates):
            if time.time() >= deadline:
                break
            seed_deadline = time.time() + per_seed
            a_opt, sc_opt = _smooth_exp_descent(
                seed_vec, seed_deadline, best_seq, best_score
            )
            if sc_opt < best_score - 1e-12:
                best_score, best_seq = sc_opt, a_opt.copy()
                print(
                    f"[{time.time():.1f}] after smooth‑exp seed {idx} new global best {best_score:.6f}"
                )

    # ---- 3. final greedy polishing -----------------------------------
    if time.time() < deadline:
        print(f"[{time.time():.1f}] final greedy polishing")
        a_opt, sc_opt = _pairwise_swap(
            best_seq.copy(), deadline, best_seq, best_score
        )
        if sc_opt < best_score - 1e-12:
            best_score, best_seq = sc_opt, a_opt.copy()
            print(f"[{time.time():.1f}] after final swaps new best {best_score:.6f}")

    print(
        f"Final best score (2·n·max(conv)): {best_score:.6f} (length={len(best_seq)})"
    )
    # Return a plain Python list (evaluate_sequence expects a list)
    return best_seq.tolist()
