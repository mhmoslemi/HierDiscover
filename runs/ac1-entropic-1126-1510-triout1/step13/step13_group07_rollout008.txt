def _fft_convolve(u: np.ndarray, v: np.ndarray) -> np.ndarray:
    need = len(u) + len(v) - 1
    size = 1 << ((need - 1).bit_length())
    U = np.fft.rfft(u, n=size)
    V = np.fft.rfft(v, n=size)
    conv = np.fft.irfft(U * V, n=size)
    return conv[:need]

def _smooth_gradient(a: np.ndarray, beta: float) -> np.ndarray:

"""
Propose a non‑negative step‑function (list of heights) that minimizes
``evaluate_sequence``.

Algorithm
---------
1. **Broad seeding** – generate a few hundred analytic / random
   candidates (uniform, beta, edge‑weighted, sine‑weighted, Dirichlet,
   sparse).  All candidates are normalised to sum = 1 and scored with the
   exact objective 2·n·max(conv(a,a)).  The best ~15 seeds are kept.

2. **Smooth exponential descent** – for each seed we minimise a
   *smoothed* version of the max–convolution using an exponentiated‑gradient
   (mirror) update on the simplex.
   *   Compute the self‑convolution `c = a*a`.
   *   Form a soft‑max weight vector  
       `w_k ∝ exp(β·(c_k–max(c)))`  (β grows each iteration).
   *   The gradient of the smoothed objective is  
       `g = 2·(w * a_rev)`, i.e. a convolution of `w` with the reversed
       sequence.  It is obtained by a real FFT (full convolution, length
       3 n−2) and a simple slice.
   *   Update with an exponentiated‑gradient step  
       `a_i ← a_i·exp(−η·g_i)` and renormalise to the simplex.
   *   Occasionally inject a tiny exponential jitter to escape stagnation.

   The method starts with a large smoothing (small β) and gradually
   focuses on the true max (large β), thereby escaping local minima that
   trap a pure sub‑gradient method.

3. **Refinement** – on the globally best sequence perform a deterministic
   pairwise mass transfer (as in the original code) and a tiny smoothing
   by a 3‑point kernel.  If the symmetric version `(a+a[::-1])/2` is better,
   keep it.

The whole routine respects the 1000 s budget (the caller supplies the
budget) and returns a plain Python list; ``evaluate_sequence`` will later
apply the final clipping, etc.
"""

# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Euclidean projection onto the simplex {x≥0, Σx = s}."""
    if s < 0:
        raise ValueError("simplex radius must be non‑negative")
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho_arr = np.where(u - cssv / (np.arange(1, len(v) + 1)) > 0)[0]
    rho = rho_arr[-1] if rho_arr.size else 0
    theta = cssv[rho] / (rho + 1) if rho_arr.size else 0.0
    return np.maximum(v - theta, 0.0)


# ----------------------------------------------------------------------
def _fft_convolve(u: np.ndarray, v: np.ndarray) -> np.ndarray:
    """Full linear convolution of two real vectors using a real FFT."""
    need = len(u) + len(v) - 1
    size = 1 << ((need - 1).bit_length())
    U = np.fft.rfft(u, n=size)
    V = np.fft.rfft(v, n=size)
    conv = np.fft.irfft(U * V, n=size)
    return conv[:need]


def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Full self‑convolution a*a (length 2 n‑1)."""
    return _fft_convolve(a, a)


def _objective(a: np.ndarray) -> float:
    """Exact objective 2·n·max(conv(a,a)) for a vector with Σa = 1."""
    n = a.shape[0]
    b = _fft_self_convolve(a)
    return 2.0 * n * b.max()

# Analytic families ----------------------------------------------------
def _distribution_uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _distribution_arcsine(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    vals /= vals.sum()
    return vals


def _distribution_beta_sym(n: int, p: float) -> np.ndarray:
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    w = np.exp((p - 1) * np.log(x) + (p - 1) * np.log(1.0 - x))
    if w.sum() == 0.0:
        w[:] = 1.0 / n
    else:
        w /= w.sum()
    return w


def _distribution_beta_asym(n: int, a: float, b: float) -> np.ndarray:
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    w = np.exp((a - 1) * np.log(x) + (b - 1) * np.log(1.0 - x))
    if w.sum() == 0.0:
        w[:] = 1.0 / n
    else:
        w /= w.sum()
    return w


def _distribution_edge_weighted(n: int, p: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64) + 0.5
    j = (n - np.arange(n, dtype=np.float64) - 0.5)
    w = (i * j) ** (-p)
    if w.sum() == 0.0:
        w[:] = 1.0 / n
    else:
        w /= w.sum()
    return w


def _distribution_sine_weighted(n: int, p: float) -> np.ndarray:
    theta = np.pi * (np.arange(n, dtype=np.float64) + 0.5) / n
    w = np.sin(theta) ** p
    if w.sum() == 0.0:
        w[:] = 1.0 / n
    else:
        w /= w.sum()
    return w


def _distribution_random_sparse(
    n: int, rng: np.random.Generator, density: float = 0.01
) -> np.ndarray:
    k = max(2, int(n * density))
    idx = rng.choice(n, size=k, replace=False)
    vals = np.zeros(n, dtype=np.float64)
    vals[idx] = rng.random(k)
    if vals.sum() == 0.0:
        vals[0] = 1.0
    vals /= vals.sum()
    return vals


def _mix_with_uniform(base: np.ndarray, lam: float) -> np.ndarray:
    n = base.shape[0]
    uniform = np.full(n, 1.0 / n, dtype=np.float64)
    return lam * uniform + (1.0 - lam) * base


# ----------------------------------------------------------------------
def _symmetrize(a: np.ndarray) -> np.ndarray:
    """Return (a + a[::-1]) / 2 (still sums to 1 after renormalisation)."""
    b = (a + a[::-1]) * 0.5
    s = b.sum()
    return b / s if s > 0 else a.copy()

def _smooth_exponential_descent(
    a: np.ndarray,
    deadline: float,
    best_seq: np.ndarray,
    best_score: float,
    eta0: float = 0.5,
    beta0: float = 5.0,
    beta_growth: float = 1.02,
    max_iter: int = 200_000,
    jitter_interval: int = 3000,
) -> tuple[np.ndarray, float]:
    """
    Minimise 2·n·max(conv(a,a)) using a soft‑max approximation of the max.
    The update is an exponentiated‑gradient (mirror descent) on the simplex.
    """
    n = a.shape[0]
    t = 0
    no_improve = 0
    rng = np.random.default_rng()
    beta = beta0

    while time.time() < deadline and t < max_iter:
        t += 1

        # 1) self‑convolution and current true objective
        conv = _fft_self_convolve(a)
        cur_score = 2.0 * n * conv.max()

        if cur_score < best_score - 1e-14:
            best_score = cur_score
            best_seq = a.copy()
            no_improve = 0
            print(f"[{time.time():.1f}] smooth‑EG it={t} new best {best_score:.6f}")
        else:
            no_improve += 1

        # 2) soft‑max weight vector (more focus on large conv entries)
        maxc = conv.max()
        w = np.exp(beta * (conv - maxc))          # stable: max(c) gives 1 at peak
        w_sum = w.sum()
        if w_sum == 0.0:
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        # 3) gradient of the smoothed objective:
        #    g = 2 * (w * a_rev)   – a full convolution, then slice.
        a_rev = a[::-1]
        conv_wr = _fft_convolve(w, a_rev)        # length 3n‑2
        # extract n entries reversed around centre (indices n‑1,…,0)
        grad = 2.0 * conv_wr[n - 1 : n - 1 - n : -1]

        # 4) exponentiated‑gradient step and re‑normalisation
        eta = eta0 / math.sqrt(t)
        a = a * np.exp(-eta * grad)
        s = a.sum()
        if s == 0.0:
            a = np.full(n, 1.0 / n, dtype=np.float64)
        else:
            a = a / s

        # 5) increase focus on the true max
        beta *= beta_growth

        # 6) occasional jitter to avoid stagnation
        if no_improve >= jitter_interval:
            jitter = rng.exponential(scale=0.001, size=n)
            a = a + jitter
            a = a / a.sum()
            no_improve = 0

    return best_seq, best_score


# ----------------------------------------------------------------------
def _pairwise_swap_local(
    a: np.ndarray,
    deadline: float,
    best_seq: np.ndarray,
    best_score: float,
    max_trials: int | None = None,
) -> tuple[np.ndarray, float]:
    """Randomly move a small amount of mass; keep move only if it improves."""
    n = a.shape[0]
    trials = 0
    rng = np.random.default_rng()
    while time.time() < deadline:
        if max_trials is not None and trials >= max_trials:
            break
        trials += 1

        src = np.where(a > 1e-12)[0]
        if src.size == 0:
            break
        i = rng.choice(src)
        j = rng.integers(n - 1)
        if j >= i:
            j += 1

        delta = a[i] * rng.random() * 0.35
        if delta < 1e-14:
            continue

        new_a = a.copy()
        new_a[i] -= delta
        new_a[j] += delta

        cur = _objective(new_a)
        if cur < best_score - 1e-14:
            a = new_a
            best_score = cur
            best_seq = a.copy()
            print(
                f"[{time.time():.1f}] swap i={i}→j={j} δ={delta:.3e} new best {best_score:.6f}"
            )
    return best_seq, best_score

                )

    # keep a limited set of the very best seeds
    all_candidates.sort(key=lambda x: x[0])
    top_seeds = [seq for _, seq in all_candidates[:15]]

    # --------------------------------------------------------------
    # 2. Smooth exponential descent on the most promising seeds
    remaining = deadline - time.time()
    if remaining > 30.0 and top_seeds:
        per_seed = remaining / len(top_seeds)
        print(f"[{time.time():.1f}] starting smooth‑EG on {len(top_seeds)} seeds")
        for idx, seed_seq in enumerate(top_seeds):
            if time.time() >= deadline:
                break
            sub_deadline = time.time() + per_seed
            a_opt, sc_opt = _smooth_exponential_descent(
                seed_seq.copy(),
                sub_deadline,
                best_seq=overall_best_seq,
                best_score=overall_best_score,
                eta0=0.5,
                beta0=5.0,
                beta_growth=1.02,
                jitter_interval=3000,
            )
            if sc_opt < overall_best_score:
                overall_best_score, overall_best_seq = sc_opt, a_opt.copy()
                print(
                    f"[{time.time():.1f}] after smooth‑EG seed {idx} new global best {overall_best_score:.6f}"
                )
            # early stop if we already beat the target
            if overall_best_score <= 1.503:
                break

    # --------------------------------------------------------------
    # 3. Deterministic pairwise refinement
    if overall_best_seq is not None and time.time() < deadline:
        print(f"[{time.time():.1f}] pairwise refinement")
        overall_best_seq, overall_best_score = _pairwise_swap_local(
            overall_best_seq.copy(),
            deadline,
            best_seq=overall_best_seq,
            best_score=overall_best_score,
            max_trials=250_000,
        )

    # --------------------------------------------------------------
    # 4. Optional symmetrisation & tiny smoothing
    if overall_best_seq is not None and time.time() < deadline:
        # try enforcing symmetry (often helps the convolution shape)
        sym = _symmetrize(overall_best_seq)
        sym_score = _objective(sym)
        if sym_score < overall_best_score:
            overall_best_seq, overall_best_score = sym, sym_score
            print(f"[{time.time():.1f}] symmetrised version improves to {sym_score:.6f}")

        # very cheap 3‑point smoothing
        kernel = np.array([0.25, 0.5, 0.25], dtype=np.float64)
        smooth = np.convolve(overall_best_seq, kernel, mode="same")
        smooth = smooth / smooth.sum()
        smooth_score = _objective(smooth)
        if smooth_score < overall_best_score:
            overall_best_seq, overall_best_score = smooth, smooth_score
            print(f"[{time.time():.1f}] smoothed version improves to {smooth_score:.6f}")

    # --------------------------------------------------------------
    if overall_best_seq is None:
        # fallback – uniform distribution of a moderate length
        overall_best_seq = _distribution_uniform(3000)

    print(
        f"Final best score: {overall_best_score:.6f} (length={len(overall_best_seq)})"
    )
    # Return a plain Python list; ``evaluate_sequence`` will apply final clipping.
    return overall_best_seq.tolist()
