def greedy_sidons(n, max_count=2, time_limit=some):
    best_S = None
    best_m = 0
    while time.time() < time_limit:
        S = set()
        count = np.zeros(2*n-1, dtype=np.int16)
        # random permutation of indices
        perm = np.random.permutation(n)
        for i in perm:
            # compute contributions (vectorized)
            # existing indices in S as array
            if not S:
                # adding first element always ok
                S.add(i)
                count[i+i] += 1
                continue
            js = np.array(list(S), dtype=np.int32)
            sum_idxs = i + js
            # potential new counts after adding i
            new_counts = count[sum_idxs] + 2  # each pair i,j adds 2
            # diag count at 2*i will add 1 (self)
            if count[2*i]+1 > max_count:
                continue
            if np.any(new_counts > max_count):
                continue
            # all good, add
            S.add(i)
            count[sum_idxs] += 2
            count[2*i] += 1
        if len(S) > best_m:
            best_m = len(S)
            best_S = S.copy()
    return best_S
```

def _greedy_subset(n, max_count, rng, within_seconds):
    deadline = time.time() + within_seconds
    bestS = set()
    bestsize = 0
    while time.time() < deadline:
        S = []
        counts = np.zeros(2*n-1, dtype=np.int16)
        # random ordering of indices
        order = rng.permutation(n)
        for i in order:
            if not S:
                # first element always ok
                S.append(i)
                counts[2*i] += 1
                continue
            js = np.array(S, dtype=np.int32)
            sum_idxs = i + js
            # check if any count + 2 exceeds max_count
            # we need to add self count separately
            if counts[2*i] + 1 > max_count:
                continue
            if np.any(counts[sum_idxs] + 2 > max_count):
                continue
            # add
            S.append(i)
            counts[sum_idxs] += 2
            counts[2*i] += 1
        if len(S) > bestsize:
            bestsize = len(S)
            bestS = set(S)
            # optional early break if we hit theoretical limit
            # bound m <= floor( ( math.sqrt(8*n+1)-1)/2 )
            theoretical_max = int((math.isqrt(8*n+1)-1)//2)
            if bestsize >= theoretical_max:
                break
    return bestS

def _reduce_max_conv(a, deadline):
    n = len(a)
    while time.time() < deadline:
        conv = _fft_self_convolve(a)
        maxk = conv.argmax()
        # find indices i that contribute to this sum: i in [maxk - (n-1), n-1] etc.
        # All i such that j = maxk - i in[0,n-1]
        contrib = np.arange(max(0, maxk - (n-1)), min(n-1, maxk) + 1)
        # shift contributions: weights a[i] * a[maxk-i]
        # We'll compute contribution values: arr = a[contrib] * a[maxk - contrib]
        # picks i with highest product a[i]*a[maxk-i] (i.e., large).
        # Choose i_max = argmax product.
        prods = a[contrib] * a[maxk - contrib]
        i_max = contrib[prods.argmax()]
        j_max = maxk - i_max
        # move small delta mass from i_max to a random other position not j_max maybe
        delta = a[i_max] * 0.1  # or small factor
        if delta < 1e-12:
            break
        a[i_max] -= delta
        # pick random target not i_max
        tgt = np.random.randint(n)
        a[tgt] += delta
        # project onto simplex
        a = a / a.sum()

"""
Search for a step–function (list of non‑negative heights) that minimises
``evaluate_sequence``.

Strategy
--------
1. **Broad random seeding** – many Dirichlet draws, sparse random vectors,
   a few analytic families, all on several moderate lengths.
2. **Greedy low‑collision (Sidon‑like) construction** – builds a sparse support
   whose pairwise sums collide rarely (max count per sum bounded by a small
   constant).  Uniform mass on the support is a strong candidate.
3. **Heavy mirror‑descent** – exponentiated‑gradient (entropy mirror) descent
   on the exact objective starting from the best few seeds.
4. **Final polishing** – deterministic pairwise swaps and a cheap simulated‑
   annealing sweep.

The whole routine respects the supplied time budget (default 1000 s) and
returns a plain Python list of heights.
"""

import time
import math
import numpy as np
from numpy.fft import rfft, irfft

# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x≥0, sum(x)=s}."""
    if s < 0:
        raise ValueError("simplex radius must be non‑negative")
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.where(u - cssv / (np.arange(1, len(v) + 1)) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a (real FFT, length 2·n‑1)."""
    n = a.shape[0]
    need = 2 * n - 1
    size = 1 << ((need - 1).bit_length())   # next power of two
    A = rfft(a, n=size)
    conv = irfft(A * A, n=size)
    return conv[:need]


# ----------------------------------------------------------------------
def _objective(a: np.ndarray) -> float:
    """Exact objective 2·n·max(conv(a,a)) for a with sum(a)=1."""
    n = a.shape[0]
    b = _fft_self_convolve(a)
    return 2.0 * n * b.max()

# Analytic families ----------------------------------------------------
def _distribution_uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _distribution_arcsine(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    vals /= vals.sum()
    return vals


def _distribution_dirichlet(n: int, rng: np.random.Generator) -> np.ndarray:
    """Dirichlet draw with a random concentration (0.01 … 2.0)."""
    alpha = rng.uniform(0.01, 2.0)

    if x.sum() == 0.0:
        x[0] = 1.0
    return x / x.sum()

    """lam·U + (1‑lam)·base, where U is the uniform distribution."""
    n = base.shape[0]
    uniform = np.full(n, 1.0 / n, dtype=np.float64)
    return lam * uniform + (1.0 - lam) * base

# Greedy low‑collision (Sidon‑like) construction -----------------------
def _greedy_low_collision_set(
    n: int,
    max_count: int,
    rng: np.random.Generator,
    time_limit: float,
) -> tuple[list[int] | None, int]:
    """
    Build a set S⊂{0,…,n‑1} such that for every k the number of ordered
    pairs (i,j)∈S×S with i+j=k does not exceed ``max_count``.
    Returns the best set found within ``time_limit`` seconds and its size.
    """
    deadline = time.time() + time_limit
    best_set: list[int] = []
    best_m = 0

    # theoretical upper bound from  m(m+1)/2 ≤ max_count·(2n‑1)
    # (a loose bound, only used to stop early)
    #   ⇒ m ≈ sqrt(2·max_count·n)
    theoretical = int(math.sqrt(2.0 * max_count * n))

    while time.time() < deadline:
        S: list[int] = []
        counts = np.zeros(2 * n - 1, dtype=np.int16)
        order = rng.permutation(n)
        for i in order:
            if not S:
                S.append(i)
                counts[2 * i] = 1
                continue

            js = np.array(S, dtype=np.int32)
            sums = i + js                     # vector of new pair sums
            # self‑pair contribution
            if counts[2 * i] + 1 > max_count:
                continue
            # contributions from i with all existing elements
            if np.any(counts[sums] + 2 > max_count):
                continue

            # accept i
            S.append(i)
            counts[sums] += 2                 # ordered pairs (i,j) and (j,i)
            counts[2 * i] += 1                # (i,i)

        m = len(S)
        if m > best_m:
            best_m = m
            best_set = S.copy()
            if best_m >= theoretical:
                break

    return (best_set if best_m > 0 else None, best_m)

# Momentum mirror descent (same spirit as the reference implementation) ----------
def _subgradient_descent_momentum(
    a: np.ndarray,
    deadline: float,
    best_seq: np.ndarray,
    best_score: float,
    lr0: float = 0.8,
    active_frac: float = 0.035,
    max_active: int = 30,
    beta: float = 0.9,
    eval_interval: int = 120,
    patience: int = 2000,
) -> tuple[np.ndarray, float]:
    """
    Exponentiated‑gradient (entropy mirror) descent on the exact objective.
    ``active_frac`` determines which convolution entries belong to the
    active set (those within that fraction of the current maximum).
    """
    n = a.shape[0]
    t = 0
    no_improve = 0
    momentum = np.zeros_like(a)
    rng = np.random.default_rng()
    ind_grid = np.arange(n, dtype=np.int32)

    while time.time() < deadline and t < 500_000:
        t += 1

        conv = _fft_self_convolve(a)
        max_c = conv.max()
        thresh = max_c * (1.0 - active_frac)

        act = np.where(conv >= thresh)[0]
        if act.size > max_active:
            top = np.argpartition(conv[act], -max_active)[-max_active:]
            act = act[top]

        grad = np.zeros_like(a)
        for k in act:
            shift = k - ind_grid
            mask = (shift >= 0) & (shift < n)
            grad[mask] += 2.0 * a[shift[mask]]

        # momentum update
        momentum = beta * momentum + (1.0 - beta) * grad
        eta = lr0 / math.sqrt(t)
        a = a * np.exp(-eta * momentum)

        # projection onto the simplex (preserves sum=1, non‑negative)
        s = a.sum()
        if s == 0.0:
            a = np.full(n, 1.0 / n, dtype=np.float64)
        else:
            a /= s

        if t % eval_interval == 0:
            cur = _objective(a)
            if cur + 1e-14 < best_score:
                best_score, best_seq = cur, a.copy()
                no_improve = 0
                print(f"[{time.time():.1f}] it={t:,}   new global best {best_score:.6f}")
            else:
                no_improve += 1

        if no_improve >= patience:
            # shake out of a plateau
            jitter = rng.exponential(scale=0.004, size=n)
            a = a + jitter
            a = np.maximum(a, 0.0)
            a /= a.sum()
            no_improve = 0

    return best_seq, best_score

# Pairwise deterministic refinement --------------------------------------
def _pairwise_swap_local(
    a: np.ndarray,
    deadline: float,
    best_seq: np.ndarray,
    best_score: float,
    max_trials: int | None = None,
) -> tuple[np.ndarray, float]:
    """
    Transfer a random fraction of mass from one coordinate to another.
    Keep the change only if the objective improves.
    """
    n = a.shape[0]
    trials = 0
    rng = np.random.default_rng()
    while time.time() < deadline:
        if max_trials is not None and trials >= max_trials:
            break
        trials += 1

        src = np.where(a > 1e-12)[0]
        if src.size == 0:
            break
        i = rng.choice(src)

        j = rng.integers(n - 1)
        if j >= i:
            j += 1

        delta = a[i] * rng.random() * 0.33
        if delta < 1e-14:
            continue

        new_a = a.copy()
        new_a[i] -= delta
        new_a[j] += delta

        cur = _objective(new_a)
        if cur + 1e-14 < best_score:
            a = new_a
            best_score = cur
            best_seq = a.copy()
            print(f"[{time.time():.1f}] swap {i}->{j} δ={delta:.3e}   new best {best_score:.6f}")

    return best_seq, best_score

def _random_local_search(
    a: np.ndarray,
    deadline: float,
    best_seq: np.ndarray,
    best_score: float,
    init_temp: float = 0.008,
    decay: float = 0.9995,
) -> tuple[np.ndarray, float]:
    """Transfer mass between random positions; Metropolis acceptance."""
    n = a.shape[0]
    rng = np.random.default_rng()
    T = init_temp
    while time.time() < deadline:
        i = rng.integers(n)
        j = rng.integers(n - 1)
        if j >= i:
            j += 1
        delta = a[i] * rng.random() * 0.28

            continue

        new_a[j] += delta

            best_score = cur

        else:
            if rng.random() < math.exp(-(cur - best_score) / max(T, 1e-12)):

    return best_seq, best_score

                overall_best_score, overall_best_seq = sc_opt, a_opt.copy()
                print(f"[{time.time():.1f}] after descent seed {idx} new best {overall_best_score:.6f}")

    # ------------------------------------------------------------------
    # 4. Deterministic pairwise cleanup
    if time.time() < deadline:
        print(f"[{time.time():.1f}] pairwise cleanup")
        overall_best_seq, overall_best_score = _pairwise_swap_local(
            overall_best_seq.copy(),
            deadline,
            best_seq=overall_best_seq,
            best_score=overall_best_score,
            max_trials=200_000,
        )

    # ------------------------------------------------------------------
    # 5. Final cheap simulated‑annealing clean‑up
    if time.time() < deadline:
        print(f"[{time.time():.1f}] final simulated‑annealing")
        overall_best_seq, overall_best_score = _random_local_search(
            overall_best_seq.copy(),
            deadline,
            best_seq=overall_best_seq,
            best_score=overall_best_score,
            init_temp=0.01,
            decay=0.9994,
        )

    # ------------------------------------------------------------------
    print(
        f"Final best score: {overall_best_score:.6f} (length={len(overall_best_seq)})"
    )
    # Return a plain Python list; evaluate_sequence will clip as required.
    return overall_best_seq.tolist()
