def _fft_convolve(x, y):
    size = 1 << ((len(x)+len(y)-1)-1).bit_length()
    X = np.fft.rfft(x, n=size)
    Y = np.fft.rfft(y, n=size)
    conv = np.fft.irfft(X * Y, n=size)
    return conv[:len(x)+len(y)-1]

Implementation of beta distribution:

def _distribution_beta(n, α, β):
    i = np.arange(n, dtype=np.float64) + 0.5
    x = i / n  # in (0,1)
    w = (x ** (α-1)) * ((1 - x) ** (β-1))
    if np.isnan(w).any():
        w = np.nan_to_num(w, nan=0.0)
    w_sum = w.sum()
    if w_sum == 0:
        return np.full(n, 1.0/n)
    w /= w_sum
    return w

"""
Hybrid search for a non‑negative step‑function that minimises
``evaluate_sequence`` (i.e. 2·n·max(conv(a,a)) / (sum a)²).

The algorithm consists of four stages:

1️⃣  **Analytic families** – a fast sweep over a wide collection of
    parameterised families (uniform, arcsine, sine/cosine, beta,
    edge‑weighted, exponential‑edge, …) on several lengths.  The
    best few seeds are kept.

2️⃣  **Smoothed‑max mirror descent** – the non‑smooth objective
    `max(conv)` is replaced by a log‑sum‑exp (smooth) surrogate.
    Gradient steps are performed on the simplex using an Adam‑style
    update and a continuation on the smoothing parameter β
    (β grows from 0.5 to 200).  This lets the optimiser escape local
    plateaus that trap plain sub‑gradient descent.

3️⃣  **Exact sub‑gradient refinement** – from the best smoothed
    solutions an exponentiated‑gradient descent on the true
    L∞‑objective (as in the previous state‑of‑the‑art) is run for a
    short while.

4️⃣  **Greedy pairwise mass‑transfer** – a final hill‑climber that
    moves a random fraction of mass between two positions whenever
    it strictly improves the exact objective.

All work is bounded by the supplied time budget (default 1000 s);
the function returns the best sequence found as a plain list.
"""

import time
import math
import numpy as np

# ------------------------------------------------------------
#  Utility: projection onto the probability simplex
# ------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Euclidean projection of `v` onto {x ≥ 0, Σx = s}."""
    if s < 0:
        raise ValueError("Simplex radius must be non‑negative")
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.where(u - cssv / (np.arange(1, len(v) + 1)) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    return np.maximum(v - theta, 0.0)


# ------------------------------------------------------------
#  FFT based convolutions (real inputs)
# ------------------------------------------------------------
def _fft_convolve(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full convolution of two real vectors using a real FFT."""
    n = len(x) + len(y) - 1
    size = 1 << ((n - 1).bit_length())
    X = np.fft.rfft(x, n=size)
    Y = np.fft.rfft(y, n=size)
    conv = np.fft.irfft(X * Y, n=size)[:n]
    return conv


def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Full self‑convolution a * a using a real FFT."""
    return _fft_convolve(a, a)


# ------------------------------------------------------------
#  Exact objective (sum(a) = 1)
# ------------------------------------------------------------
def _objective(a: np.ndarray) -> float:
    """Exact value of 2·n·max(conv(a,a)) for a with Σa = 1."""
    n = a.shape[0]
    b = _fft_self_convolve(a)
    return float(2.0 * n * b.max())


# ------------------------------------------------------------
#  Analytic families
# ------------------------------------------------------------
def _distribution_uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _distribution_arcsine(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    vals /= vals.sum()
    return vals


def _distribution_sine(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64) + 0.5
    w = np.sin(np.pi * i / n)
    w /= w.sum()
    return w


def _distribution_cosine(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64) + 0.5
    w = 1.0 - np.cos(np.pi * i / n)
    w /= w.sum()
    return w


def _distribution_quadratic(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    x = (2.0 * i) / (n - 1) - 1.0
    w = 1.0 - x * x
    w = np.clip(w, 0.0, None)
    w /= w.sum()
    return w


def _distribution_edge_weighted(n: int, p: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64) + 0.5
    j = (n - np.arange(n, dtype=np.float64) - 0.5)
    w = (i * j) ** (-p)
    w_sum = w.sum()
    if w_sum == 0.0:
        return _distribution_uniform(n)
    w /= w_sum
    return w


def _distribution_exponential_edge(n: int, c: float) -> np.ndarray:
    """w_i ∝ exp(-c·i) + exp(-c·(n‑1‑i))   (c>0)."""
    i = np.arange(n, dtype=np.float64)
    w = np.exp(-c * i) + np.exp(-c * (n - 1 - i))
    w /= w.sum()
    return w


def _distribution_beta(n: int, alpha: float, beta: float) -> np.ndarray:
    """Symmetric beta‑type distribution on the discrete grid."""
    i = np.arange(n, dtype=np.float64) + 0.5
    x = i / n  # in (0,1)
    # avoid NaNs when alpha or beta is <1
    w = (x ** (alpha - 1.0)) * ((1.0 - x) ** (beta - 1.0))
    w = np.nan_to_num(w, nan=0.0, posinf=0.0, neginf=0.0)
    total = w.sum()
    if total == 0.0:
        return _distribution_uniform(n)
    w /= total
    return w


def _mix_with_uniform(base: np.ndarray, lam: float) -> np.ndarray:
    """Convex combination λ·uniform + (1‑λ)·base."""
    n = base.shape[0]
    uniform = np.full(n, 1.0 / n, dtype=np.float64)
    return lam * uniform + (1.0 - lam) * base


# ------------------------------------------------------------
#  Smoothed‑max mirror descent (Adam + β‑continuation)
# ------------------------------------------------------------
def _smooth_max_descent(
    a0: np.ndarray,
    time_limit: float,
    best_seq: np.ndarray,
    best_score: float,
) -> tuple[np.ndarray, float]:
    """
    Minimises the exact objective using a smooth surrogate
    (log‑sum‑exp) with an Adam optimiser, gradually increasing
    the smoothing parameter β.
    """
    a = a0.copy()
    n = a.shape[0]
    # Adam state
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8
    t = 0
    # β schedule (log‑space)
    beta_schedule = [0.5, 1.0, 2.0, 5.0, 10.0, 20.0, 50.0, 100.0, 200.0]

    for beta in beta_schedule:
        # a few thousand iterations per β (stop early if time runs out)
        while time.time() < time_limit:
            t += 1
            # full self‑convolution
            conv = _fft_self_convolve(a)               # length 2n‑1
            max_val = conv.max()
            # softmax weights (stable)
            shifted = conv - max_val                   # ≤ 0
            exp_beta = np.exp(beta * shifted)
            w = exp_beta / exp_beta.sum()              # Σ w = 1, w ≥ 0
            # gradient of the smoothed max:
            #   grad = 2 * (w * a_rev) convolved
            a_rev = a[::-1]
            grad_conv = _fft_convolve(w, a_rev)        # length 3n‑2
            grad = 2.0 * grad_conv[n - 1 : n - 1 + n]   # size n

            # gradient of the (smooth) objective J = 2·n·smooth_max
            grad_J = 2.0 * n * grad

            # Adam update (projected onto the simplex afterwards)
            m = beta1 * m + (1.0 - beta1) * grad_J
            v = beta2 * v + (1.0 - beta2) * (grad_J ** 2)
            m_hat = m / (1.0 - beta1 ** t)
            v_hat = v / (1.0 - beta2 ** t)
            step = 0.15 / (np.sqrt(v_hat) + eps)       # base step size tuned experimentally
            a -= step * m_hat
            a = _project_onto_simplex(a, s=1.0)

            # Periodic exact evaluation
            if t % 120 == 0:
                cur = _objective(a)
                if cur < best_score - 1e-14:
                    best_score, best_seq = cur, a.copy()
                    print(
                        f"[{time.time():.1f}] β={beta:.1f} it={t} new best {best_score:.6f}"
                    )
            # give up on current β after ~2500 iterations to keep the whole
            # schedule within the allocated budget
            if t % 2500 == 0:
                break
        if time.time() >= time_limit:
            break
    return best_seq, best_score


# ------------------------------------------------------------
#  Exact sub‑gradient descent (exponentiated gradient)
# ------------------------------------------------------------
def _subgradient_descent_exact(
    a0: np.ndarray,
    time_limit: float,
    best_seq: np.ndarray,
    best_score: float,
    lr0: float = 0.9,
    eval_every: int = 100,
) -> tuple[np.ndarray, float]:
    """Exponentiated‑gradient descent on the true L∞‑objective."""
    a = a0.copy()
    n = a.shape[0]
    t = 0
    while time.time() < time_limit:
        t += 1
        conv = _fft_self_convolve(a)
        max_val = conv.max()
        max_idxs = np.where(conv == max_val)[0]

        # exact sub‑gradient (sum of a shifted copies)
        grad = np.zeros_like(a)
        for idx in max_idxs:
            i_start = max(0, idx - (n - 1))
            i_end = min(n - 1, idx)
            i = np.arange(i_start, i_end + 1, dtype=np.int64)
            j = idx - i
            grad[i] += 2.0 * a[j]

        eta = lr0 / math.sqrt(t)
        a *= np.exp(-eta * grad)          # EG keeps positivity
        s = a.sum()
        if s <= 0.0:
            a = np.full(n, 1.0 / n, dtype=np.float64)
        else:
            a /= s

        if t % eval_every == 0:
            cur = 2.0 * n * max_val
            if cur < best_score - 1e-14:
                best_score, best_seq = cur, a.copy()
                print(
                    f"[{time.time():.1f}] exact subgrad it={t} new global best {best_score:.6f}"
                )
    return best_seq, best_score


# ------------------------------------------------------------
#  Greedy pairwise mass‑transfer hill‑climber
# ------------------------------------------------------------
def _pairwise_swap(
    a0: np.ndarray,
    time_limit: float,
    best_seq: np.ndarray,
    best_score: float,
    max_trials: int | None = None,
) -> tuple[np.ndarray, float]:
    """Randomly relocate a fraction of mass; accept only strict improvements."""
    a = a0.copy()
    n = a.shape[0]
    trials = 0
    while time.time() < time_limit:
        if max_trials is not None and trials >= max_trials:
            break
        trials += 1

        src = np.where(a > 1e-14)[0]
        if src.size == 0:
            break
        i = np.random.choice(src)
        j = np.random.randint(n - 1)
        if j >= i:
            j += 1

        delta = a[i] * np.random.rand() * 0.45
        if delta < 1e-15:
            continue

        new_a = a.copy()
        new_a[i] -= delta
        new_a[j] += delta

        cur = _objective(new_a)
        if cur < best_score - 1e-14:
            a = new_a
            best_score, best_seq = cur, a.copy()
            print(
                f"[{time.time():.1f}] swap i={i}→j={j} δ={delta:.6g} new best {best_score:.6f}"
            )
    return best_seq, best_score

def propose_candidate(seed: int = 42, budget_s: int = 1000) -> list[float]:
    """
    Return a step‑function (plain Python list) that minimises
    ``evaluate_sequence`` within the given time budget.

    deadline = start + budget_s - 1.5

    overall_best_score = float("inf")
    top_candidates: list[tuple[float, np.ndarray]] = []   # (score, vector)

    lengths = [1200, 2000, 3000, 4000, 6000, 8000, 12000]

        ]
    )
    lam_grid = np.linspace(0.0, 1.0, 11)       # mixing weight with uniform
    c_grid = np.logspace(-3, 0, 10)           # exponential‑edge decay
    alpha_grid = np.logspace(-0.5, 0.7, 12)   # beta‑distribution parameters (α=β)

    for n in lengths:
        if time.time() >= deadline:
            break
        print(f"[{time.time():.1f}] scanning length {n}")

        # deterministic families -------------------------------------------------
        for base_fn in (
            _distribution_uniform,
            _distribution_arcsine,
            _distribution_sine,
            _distribution_cosine,
            _distribution_quadratic,
        ):
            cand = base_fn(n)
            sc = _objective(cand)
            top_candidates.append((sc, cand.copy()))
            if sc < overall_best_score:
                overall_best_score, overall_best_seq = sc, cand.copy()
                print(f"[{time.time():.1f}] new global best len={n} score={sc:.6f}")

        # beta‑type families ----------------------------------------------------
        for alpha in alpha_grid:
            if time.time() >= deadline:
                break
            cand = _distribution_beta(n, float(alpha), float(alpha))
            sc = _objective(cand)
            top_candidates.append((sc, cand.copy()))
            if sc < overall_best_score:
                overall_best_score, overall_best_seq = sc, cand.copy()
                print(f"[{time.time():.1f}] beta α={alpha:.4g} len={n} score={sc:.6f}")

        for p in p_grid:
            if time.time() >= deadline:
                break
            edge = _distribution_edge_weighted(n, float(p))
            for lam in lam_grid:
                if time.time() >= deadline:
                    break
                cand = _mix_with_uniform(edge, lam)
                sc = _objective(cand)
                top_candidates.append((sc, cand.copy()))
                if sc < overall_best_score:
                    overall_best_score, overall_best_seq = sc, cand.copy()
                    print(
                        f"[{time.time():.1f}] edge p={p:.4g} lam={lam:.2f} len={n} score={sc:.6f}"
                    )

        # exponential‑edge family ------------------------------------------------
        for c in c_grid:
            if time.time() >= deadline:
                break
            exp_edge = _distribution_exponential_edge(n, float(c))
            for lam in lam_grid:
                if time.time() >= deadline:
                    break
                cand = _mix_with_uniform(exp_edge, lam)
                sc = _objective(cand)
                top_candidates.append((sc, cand.copy()))
                if sc < overall_best_score:
                    overall_best_score, overall_best_seq = sc, cand.copy()
                    print(
                        f"[{time.time():.1f}] exp c={c:.4g} lam={lam:.2f} len={n} score={sc:.6f}"
                    )

        # Dirichlet random seeds -------------------------------------------------
        for _ in range(4):
            if time.time() >= deadline:
                break
            alpha = rng.uniform(0.01, 2.0)
            a = rng.dirichlet(np.full(n, alpha))
            sc = _objective(a)
            top_candidates.append((sc, a.copy()))
            if sc < overall_best_score:
                overall_best_score, overall_best_seq = sc, a.copy()
                print(f"[{time.time():.1f}] Dirichlet len={n} score={sc:.6f}")

            if time.time() >= deadline:

            a = _distribution_random_sparse(n, rng, density=0.006)
            sc = _objective(a)
            top_candidates.append((sc, a.copy()))
            if sc < overall_best_score:
                overall_best_score, overall_best_seq = sc, a.copy()
                print(f"[{time.time():.1f}] sparse len={n} score={sc:.6f}")

        # keep a bounded number of top seeds (memory & time)
        top_candidates.sort(key=lambda x: x[0])
        top_candidates = top_candidates[:40]

    # ------------------------------------------------------------------

    try:
        prev = np.asarray(height_sequence_1, dtype=np.float64)  # type: ignore[name-defined]
        if prev.ndim == 1 and prev.size > 0:

            if prev.sum() == 0.0:
                prev = _distribution_uniform(prev.size)
            else:
                prev = prev / prev.sum()
            sc = _objective(prev)
            if sc < overall_best_score:
                overall_best_score, overall_best_seq = sc, prev.copy()
                print(f"[{time.time():.1f}] previous best score={sc:.6f}")
            top_candidates.append((sc, prev.copy()))
    except Exception:
        pass

    # ------------------------------------------------------------------
    # 3️⃣  Fallback (should never be needed)
    # ------------------------------------------------------------------
    if overall_best_seq is None:
        n = 2000
        overall_best_seq = _distribution_uniform(n)
        overall_best_score = _objective(overall_best_seq)
        print(
            f"[{time.time():.1f}] fallback uniform length={n} score={overall_best_score:.6f}"
        )

    # ------------------------------------------------------------------
    # 4️⃣  Smoothed‑max mirror descent on a few promising seeds
    # ------------------------------------------------------------------
    remaining = deadline - time.time()
    if remaining > 40.0 and top_candidates:

            sub_deadline = time.time() + per_seed
            a_opt, sc_opt = _smooth_max_descent(
                seed_seq,
                sub_deadline,
                best_seq=overall_best_seq,
                best_score=overall_best_score,
            )
            if sc_opt < overall_best_score - 1e-12:
                overall_best_score, overall_best_seq = sc_opt, a_opt.copy()
                print(
                    f"[{time.time():.1f}] after smooth seed {idx} new global best {overall_best_score:.6f}"
                )

    # ------------------------------------------------------------------
    # 5️⃣  Exact sub‑gradient refinement (exponentiated‑gradient)
    # ------------------------------------------------------------------
    remaining = deadline - time.time()
    if remaining > 30.0:
        print(f"[{time.time():.1f}] exact sub‑gradient refinement")
        a_opt, sc_opt = _subgradient_descent_exact(
            overall_best_seq.copy(),
            deadline,
            best_seq=overall_best_seq,
            best_score=overall_best_score,
            lr0=0.9,
            eval_every=120,
        )
        if sc_opt < overall_best_score - 1e-12:

    # ------------------------------------------------------------------
    # 6️⃣  Final greedy pairwise polishing
    # ------------------------------------------------------------------
    if time.time() < deadline:
        print(f"[{time.time():.1f}] final greedy swap polishing")
        a_opt, sc_opt = _pairwise_swap(
            overall_best_seq.copy(),
            deadline,
            best_seq=overall_best_seq,
            best_score=overall_best_score,
        )
        if sc_opt < overall_best_score - 1e-12:
            overall_best_score, overall_best_seq = sc_opt, a_opt.copy()
            print(
                f"[{time.time():.1f}] after final swaps new best {overall_best_score:.6f}"
            )

    # ------------------------------------------------------------------
    print(
        f"Final best objective (2·n·max(conv)) = {overall_best_score:.6f} (length={len(overall_best_seq)})"
    )
    # ``evaluate_sequence`` will re‑scale internally, so we can return the
    # raw simplex‑normalised vector.
    return overall_best_seq.tolist()


# ------------------------------------------------------------
#  Helper for sparse random seeds (used in the scan)
# ------------------------------------------------------------
def _distribution_random_sparse(
    n: int, rng: np.random.Generator, density: float = 0.01
) -> np.ndarray:
    """Sparse random vector with a prescribed density of non‑zeros."""
    k = max(2, int(n * density))
    idx = rng.choice(n, size=k, replace=False)
    vals = np.zeros(n, dtype=np.float64)
    vals[idx] = rng.random(k)
    s = vals.sum()
    if s == 0.0:
        vals[0] = 1.0
        s = 1.0
    vals /= s
    return vals
