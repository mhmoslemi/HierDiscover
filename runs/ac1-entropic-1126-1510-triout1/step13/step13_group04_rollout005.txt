"""
Propose a non‑negative step function that (almost) minimizes
``evaluate_sequence``.

Algorithm
---------

1.  Build a diverse pool of seeds from many analytic families
    (uniform, arcsine, Hann, symmetric beta, edge‑weighted,
    generalized‑normal, power‑law, Gaussian, triangle, cosine,
    random Dirichlet, sparse).  Keep the best ~30 candidates.

2.  For each top seed run *smooth gradient descent* on the exact
    objective.  The nonsmooth ``max`` in the objective is replaced by a
    soft‑max (log‑sum‑exp) with an annealing temperature λ that grows
    from 1 to ≈80 during the run.  The gradient of the smoothed
    objective is computed analytically using FFT‑based convolutions and
    an Euclidean projection onto the probability simplex after each
    step.  This gives a fast, stable descent that penetrates the local
    minima that multiplicative updates got stuck in.

3.  If time permits, up‑sample the current best distribution (linear
    interpolation while staying on the simplex) and run a short burst of
    the same smooth descent on the larger vector.

4.  Finish with a long‑run pairwise‑mass‑transfer annealing (as in the
    previous solution) to polish the final answer.

All stages respect the total 1000 s budget (2 s safety margin).  The
routine returns a plain ``list[float]`` ready for ``evaluate_sequence``.
"""

# ----------------------------------------------------------------------
# Simplex projection (Euclidean)
# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Projection of v onto the simplex {x≥0, Σx=s}."""
    if s < 0:
        raise ValueError("simplex radius must be non‑negative")
    n = v.shape[0]
    # fast path
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.where(u - cssv / (np.arange(1, n + 1)) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# FFT based convolutions
# ----------------------------------------------------------------------
def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Full linear self‑convolution a*a using a real FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1) - 1).bit_length()
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[: 2 * n - 1]


def _fft_convolve(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full linear convolution a*b using a real FFT."""
    n = a.shape[0]
    m = b.shape[0]
    size = 1 << ((n + m - 1) - 1).bit_length()
    A = np.fft.rfft(a, n=size)
    B = np.fft.rfft(b, n=size)
    conv = np.fft.irfft(A * B, n=size)
    return conv[: n + m - 1]


# ----------------------------------------------------------------------
# Exact objective (scale‑invariant)
# ----------------------------------------------------------------------
def _objective(a: np.ndarray) -> float:
    """Exact value of ``evaluate_sequence`` for an (unnormalised) vector."""
    n = a.shape[0]
    b = _fft_self_convolve(a)
    max_b = b.max()
    sum_a = a.sum()
    if sum_a < 0.01:
        return float("inf")
    return float(2.0 * n * max_b / (sum_a * sum_a))


# ----------------------------------------------------------------------
# Analytic families – always return a *probability* vector (sum = 1)
# ----------------------------------------------------------------------
def _distribution_uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _distribution_arcsine(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    vals /= vals.sum()
    return vals


def _distribution_hann(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    w = 0.5 - 0.5 * np.cos(2.0 * np.pi * i / (n - 1))
    w /= w.sum()
    return w


def _distribution_beta_sym(n: int, p: float) -> np.ndarray:
    """Symmetric Beta(p,p) on a uniform grid."""
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    w = np.exp((p - 1) * np.log(x) + (p - 1) * np.log(1.0 - x))
    if w.sum() == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _distribution_edge_weighted(n: int, p: float) -> np.ndarray:
    """Weight ∝ (i·j)^{-p} where i,j are distances to the two edges."""
    i = np.arange(n, dtype=np.float64) + 0.5
    j = (n - np.arange(n, dtype=np.float64) - 0.5)
    w = (i * j) ** (-p)
    if w.sum() == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _distribution_generalized_normal(n: int, q: float) -> np.ndarray:
    """Density ∝ exp(-|x-1/2|^{q}) on [0,1]."""
    i = np.arange(n, dtype=np.float64) + 0.5
    x = i / n - 0.5
    w = np.exp(- (np.abs(x) / 0.5) ** q)
    w /= w.sum()
    return w


def _distribution_power_law(n: int, p: float) -> np.ndarray:
    """Monotone decreasing power‑law (p>0)."""
    i = np.arange(1, n + 1, dtype=np.float64)
    w = i ** (-p)
    w /= w.sum()
    return w


def _distribution_gaussian(n: int, sigma_frac: float) -> np.ndarray:
    """Discrete Gaussian centred at the middle, σ = sigma_frac·n."""
    i = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    sigma = sigma_frac * n
    w = np.exp(-0.5 * ((i - centre) / sigma) ** 2)
    w /= w.sum()
    return w


def _distribution_triangle(n: int) -> np.ndarray:
    """Tent‐shaped piecewise linear distribution."""
    i = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    w = 1.0 - np.abs(i - centre) / (centre + 1.0)
    w = np.maximum(w, 0.0)
    w /= w.sum()
    return w


def _distribution_cosine(n: int) -> np.ndarray:
    """Squared cosine window."""
    i = np.arange(n, dtype=np.float64)
    w = np.cos(np.pi * (i / (n - 1) - 0.5)) ** 2
    w /= w.sum()
    return w


def _distribution_random_sparse(n: int, rng: np.random.Generator, density: float = 0.01) -> np.ndarray:
    """Sparse random vector with a given density."""
    k = max(2, int(n * density))
    idx = rng.choice(n, size=k, replace=False)
    vals = np.zeros(n, dtype=np.float64)
    vals[idx] = rng.random(k)
    if vals.sum() == 0.0:
        vals[0] = 1.0
    vals /= vals.sum()
    return vals


def _mix_with_uniform(base: np.ndarray, lam: float) -> np.ndarray:
    """Convex combination with the uniform distribution."""
    n = base.shape[0]
    uniform = np.full(n, 1.0 / n, dtype=np.float64)
    return lam * uniform + (1.0 - lam) * base


# ----------------------------------------------------------------------
# Smooth gradient descent (soft‑max relaxation of max‑convolution)
# ----------------------------------------------------------------------
def _smooth_gradient_descent(
    a: np.ndarray,
    deadline: float,
    best_seq: np.ndarray,
    best_score: float,
    lam_start: float = 1.0,
    lam_end: float = 80.0,
    eta0: float = 0.2,
    eval_every: int = 30,
) -> tuple[np.ndarray, float]:
    """
    Projected gradient descent on the smoothed objective

        f(a) = 2·n·M_λ(a),   M_λ = log‑sum‑exp(λ·(a*a))/λ

    The temperature λ is slowly increased from ``lam_start`` to
    ``lam_end`` (continuation).  After each step we project onto the
    probability simplex.  The best exact objective seen so far is
    returned.
    """
    n = a.shape[0]
    lam = lam_start
    t = 0
    while time.time() < deadline:
        t += 1
        # ----- convolution -------------------------------------------------
        conv = _fft_self_convolve(a)                     # length 2n‑1

        # ----- soft‑max of λ·conv -----------------------------------------
        max_z = lam * conv.max()
        # subtract max for numerical stability
        expz = np.exp(lam * conv - max_z)
        sumexp = expz.sum()
        soft = expz / sumexp                           # soft‑max weights (sum 1)
        M = (np.log(sumexp) + max_z) / lam             # smoothed max

        # ----- gradient of M ------------------------------------------------
        # ∂M/∂a = Σ_k soft_k ∂conv_k/∂a  ,   ∂conv_k/∂a_j = 2·a_{k‑j}
        w_rev = soft[::-1]
        conv2 = _fft_convolve(a, w_rev)                # length 3n‑2
        gradM = 2.0 * conv2[n - 1 : 2 * n - 1][::-1]    # shape (n,)

        # ----- gradient of the exact objective (ignoring the sum‑constraint)----
        grad_f = 2.0 * n * gradM                        # because f = 2·n·M

        # ----- step size (decreasing) --------------------------------------
        eta = eta0 / math.sqrt(t)
        a = a - eta * grad_f

        # ----- projection onto the simplex -----------------------------------
        a = _project_onto_simplex(a)

        # ----- increase λ gradually -----------------------------------------
        if lam < lam_end:
            lam = min(lam_end, lam * 1.001)   # very slow increase

        # ----- occasional exact evaluation -----------------------------------
        if t % eval_every == 0:
            cur = _objective(a)
            if cur < best_score - 1e-14:
                best_score = cur
                best_seq = a.copy()
                print(f"[{time.time():.1f}] smoothGD it={t:5d} λ={lam:5.2f} score={best_score:.6f}")

    return best_seq, best_score

# Frank‑Wolfe refinement (unchanged, kept for a final polish)
# ----------------------------------------------------------------------
def _frank_wolfe_descent(
    a: np.ndarray,
    deadline: float,
    best_seq: np.ndarray,
    best_score: float,
    eval_every: int = 100,
) -> tuple[np.ndarray, float]:
    """Simple FW iteration that moves mass towards the coordinate with the smallest gradient."""
    n = a.shape[0]
    t = 0
    while time.time() < deadline:
        t += 1
        # compute sub‑gradient of the *max* convolution entry
        conv = _fft_self_convolve(a)
        k = int(np.argmax(conv))
        grad = np.zeros_like(a)
        if k < n:
            grad[: k + 1] = 2.0 * a[k::-1]
        else:
            start = k - (n - 1)
            grad[start:] = 2.0 * a[start:][::-1]
        i_min = int(np.argmin(grad))
        gamma = 2.0 / (t + 2.0)
        a = a + gamma * ((np.eye(1, n, i_min)[0]) - a)
        if t % eval_every == 0:
            cur = _objective(a)
            if cur < best_score - 1e-14:
                best_score = cur
                best_seq = a.copy()
                print(f"[{time.time():.1f}] FW it={t} new best {best_score:.6f}")
    return best_seq, best_score

# Simulated‑annealing style pairwise mass transfers (final polishing)
# ----------------------------------------------------------------------
def _pairwise_annealing(
    a: np.ndarray,
    deadline: float,
    best_seq: np.ndarray,
    best_score: float,
    rng: np.random.Generator,
    max_trials: int | None = None,
) -> tuple[np.ndarray, float]:
    """Randomly move a fraction of mass between two bins; accept improvements
    always and occasionally accept worsening moves early on."""
    n = a.shape[0]
    start_time = time.time()
    total_time = deadline - start_time
    trials = 0
    while time.time() < deadline:
        if max_trials is not None and trials >= max_trials:
            break
        trials += 1

        src_candidates = np.where(a > 1e-12)[0]
        if src_candidates.size == 0:
            break
        i = rng.choice(src_candidates)

        j = rng.integers(n - 1)
        if j >= i:
            j += 1

        delta = a[i] * rng.random() * 0.4
        if delta < 1e-16:
            continue

        new_a = a.copy()
        new_a[i] -= delta
        new_a[j] += delta

        cur = _objective(new_a)
        elapsed = time.time() - start_time
        temperature = max(0.0, 1.0 - elapsed / total_time) * best_score * 0.01

        if cur < best_score - 1e-14:
            accept = True
        else:
            accept = temperature > 0.0 and rng.random() < math.exp(-(cur - best_score) / temperature)

        if accept:
            a = new_a
            if cur < best_score - 1e-14:
                best_score = cur
                best_seq = a.copy()
                print(f"[{time.time():.1f}] anneal i={i}->j={j} δ={delta:.3e} new best {best_score:.6f}")
    return best_seq, best_score


# ----------------------------------------------------------------------
# Linear up‑sampling while preserving the simplex
# ----------------------------------------------------------------------
def _upsample(a: np.ndarray, new_n: int) -> np.ndarray:
    """Interpolate a probability vector to a larger length."""
    old_n = a.shape[0]
    if new_n == old_n:
        return a.copy()
    x_old = (np.arange(old_n, dtype=np.float64) + 0.5) / old_n
    x_new = (np.arange(new_n, dtype=np.float64) + 0.5) / new_n
    a_new = np.interp(x_new, x_old, a, left=0.0, right=0.0)
    a_new = np.maximum(a_new, 0.0)
    if a_new.sum() == 0.0:
        a_new[:] = 1.0 / new_n
    else:
        a_new /= a_new.sum()
    return a_new

            _distribution_cosine(n),
        ):
            sc = _objective(cand)
            top_candidates.append((sc, cand.copy()))
            if sc < overall_best_score:
                overall_best_score = sc
                overall_best_seq = cand.copy()
                print(f"[{time.time():.1f}] new global best len={n} score={sc:.6f}")

        # beta and edge‑weighted families
        for p in p_grid:
            if time.time() >= deadline:
                break
            beta = _distribution_beta_sym(n, p)
            edge = _distribution_edge_weighted(n, p)
            for base in (beta, edge):
                for lam in lam_grid:
                    if time.time() >= deadline:
                        break
                    cand = _mix_with_uniform(base, lam)
                    sc = _objective(cand)
                    top_candidates.append((sc, cand.copy()))
                    if sc < overall_best_score:
                        overall_best_score = sc
                        overall_best_seq = cand.copy()
                        print(f"[{time.time():.1f}] new best len={n} p={p:.3g} lam={lam:.2f} score={sc:.6f}")

        # generalized normal shapes
        for q in q_grid:
            if time.time() >= deadline:
                break
            cand = _distribution_generalized_normal(n, q)
            sc = _objective(cand)
            top_candidates.append((sc, cand.copy()))
            if sc < overall_best_score:
                overall_best_score = sc
                overall_best_seq = cand.copy()
                print(f"[{time.time():.1f}] new best gn q={q:.2f} len={n} score={sc:.6f}")

        # power‑law families
        for pwr in power_grid:
            if time.time() >= deadline:
                break
            cand = _distribution_power_law(n, pwr)
            sc = _objective(cand)
            top_candidates.append((sc, cand.copy()))
            if sc < overall_best_score:
                overall_best_score = sc
                overall_best_seq = cand.copy()
                print(f"[{time.time():.1f}] new best powerlaw p={pwr:.2f} len={n} score={sc:.6f}")

        # Gaussian families
        for sigma_frac in sigma_grid:
            if time.time() >= deadline:
                break
            cand = _distribution_gaussian(n, sigma_frac)
            sc = _objective(cand)
            top_candidates.append((sc, cand.copy()))
            if sc < overall_best_score:
                overall_best_score = sc
                overall_best_seq = cand.copy()
                print(f"[{time.time():.1f}] new best gaussian σ={sigma_frac:.3f} len={n} score={sc:.6f}")

        # random Dirichlet seeds
        for _ in range(6):
            if time.time() >= deadline:
                break
            alpha = rng.uniform(0.01, 2.0)
            cand = rng.dirichlet(np.full(n, alpha))
            sc = _objective(cand)
            top_candidates.append((sc, cand.copy()))
            if sc < overall_best_score:
                overall_best_score = sc
                overall_best_seq = cand.copy()
                print(f"[{time.time():.1f}] new best Dirichlet len={n} score={sc:.6f}")

        # sparse random seeds
        for _ in range(6):
            if time.time() >= deadline:
                break
            cand = _distribution_random_sparse(n, rng, density=0.01)
            sc = _objective(cand)
            top_candidates.append((sc, cand.copy()))
            if sc < overall_best_score:
                overall_best_score = sc
                overall_best_seq = cand.copy()
                print(f"[{time.time():.1f}] new best sparse len={n} score={sc:.6f}")

        # keep only the best 30 across all lengths examined so far
        top_candidates.sort(key=lambda x: x[0])
        top_candidates = top_candidates[:30]

    # ------------------------------------------------------------------
    # 2.  Include a previously discovered construction if available.
    # ------------------------------------------------------------------
    try:
        prev = np.asarray(height_sequence_1, dtype=np.float64)  # type: ignore[name-defined]
        if prev.ndim == 1:
            prev = prev / prev.sum()
            sc = _objective(prev)
            if sc < overall_best_score:
                overall_best_score = sc
                overall_best_seq = prev.copy()
                print(f"[{time.time():.1f}] previous best score={sc:.6f}")
            top_candidates.append((sc, prev.copy()))
    except Exception:
        pass

    # ------------------------------------------------------------------
    # 3.  Guarantee a non‑empty result.
    # ------------------------------------------------------------------
    if overall_best_seq is None:
        n = 4096
        overall_best_seq = _distribution_uniform(n)
        overall_best_score = _objective(overall_best_seq)
        print(f"[{time.time():.1f}] fallback uniform score={overall_best_score:.6f}")

    # ------------------------------------------------------------------
    # 4.  Smooth gradient descent on a few top seeds.
    # ------------------------------------------------------------------
    remaining = deadline - time.time()
    if remaining > 30.0 and top_candidates:
        seed_seqs = [seq for _, seq in sorted(top_candidates, key=lambda x: x[0])[:5]]
        per_seed = remaining * 0.30 / max(1, len(seed_seqs))
        print(f"[{time.time():.1f}] smooth GD on {len(seed_seqs)} seeds")
        for idx, seed_seq in enumerate(seed_seqs):
            if time.time() >= deadline:
                break
            sub_deadline = time.time() + per_seed
            a_opt, sc_opt = _smooth_gradient_descent(
                seed_seq.copy(),
                sub_deadline,
                best_seq=overall_best_seq,
                best_score=overall_best_score,
                lam_start=1.0,
                lam_end=80.0,
                eta0=0.25,
                eval_every=40,
            )
            if sc_opt < overall_best_score - 1e-12:
                overall_best_score = sc_opt
                overall_best_seq = a_opt.copy()
                print(f"[{time.time():.1f}] after smooth GD seed {idx} new global best {overall_best_score:.6f}")

    # ------------------------------------------------------------------
    # 5.  Frank‑Wolfe refinement (optional, time permitting)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        fw_time = (deadline - time.time()) * 0.15
        print(f"[{time.time():.1f}] Frank‑Wolfe refinement (budget {fw_time:.1f}s)")
        sub_deadline = time.time() + fw_time
        overall_best_seq, overall_best_score = _frank_wolfe_descent(
            overall_best_seq.copy(),
            sub_deadline,
            best_seq=overall_best_seq,
            best_score=overall_best_score,
            eval_every=80,
        )
        print(f"[{time.time():.1f}] after FW best score {overall_best_score:.6f}")

    # ------------------------------------------------------------------
    # 6.  Upsample to larger lengths and short smooth‑GD bursts.
    # ------------------------------------------------------------------
    while time.time() < deadline:
        cur_n = overall_best_seq.shape[0]
        if cur_n >= 64000:
            break
        target_n = min(cur_n * 2, 64000)
        print(f"[{time.time():.1f}] upsampling from {cur_n} → {target_n}")
        up_seq = _upsample(overall_best_seq, target_n)
        up_score = _objective(up_seq)
        if up_score < overall_best_score:
            overall_best_score = up_score
            overall_best_seq = up_seq
            print(f"[{time.time():.1f}] upsampled new best {overall_best_score:.6f}")

        # short smooth‑GD on the upsampled vector
        sub_deadline = time.time() + min(30.0, (deadline - time.time()) * 0.20)
        a_opt, sc_opt = _smooth_gradient_descent(
            overall_best_seq.copy(),
            sub_deadline,
            best_seq=overall_best_seq,
            best_score=overall_best_score,
            lam_start=2.0,
            lam_end=80.0,
            eta0=0.2,
            eval_every=30,
        )
        if sc_opt < overall_best_score - 1e-12:
            overall_best_score = sc_opt
            overall_best_seq = a_opt.copy()
            print(f"[{time.time():.1f}] after upsample GD new best {overall_best_score:.6f}")

    # ------------------------------------------------------------------
    # 7.  Final long‑run annealing / polishing.
    # ------------------------------------------------------------------
    if time.time() < deadline:
        print(f"[{time.time():.1f}] final annealing polish")
        a_opt, sc_opt = _pairwise_annealing(
            overall_best_seq.copy(),
            deadline,
            best_seq=overall_best_seq,
            best_score=overall_best_score,
            rng=rng,
            max_trials=None,
        )
        if sc_opt < overall_best_score - 1e-12:
            overall_best_score = sc_opt
            overall_best_seq = a_opt.copy()
            print(f"[{time.time():.1f}] after annealing new best {overall_best_score:.6f}")

    # ------------------------------------------------------------------
    print(
        f"Final best score (objective=2·n·max(conv)/sum²): {overall_best_score:.6f} "
        f"(length={len(overall_best_seq)})"
    )
    # ``evaluate_sequence`` truncates values >1000, but our vectors are safely
    # normalised (sum≈1) and all entries are ≤1, so we can return them directly.
    return overall_best_seq.tolist()
