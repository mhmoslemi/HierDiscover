def _smooth_gradient(a, alpha):
    conv = _fft_self_convolve(a)
    # compute softmax weights
    max_c = conv.max()
    # For numerical stability, shift: w = np.exp(alpha*(conv - max_c))

"""
Propose a step‑function (non‑negative vector whose sum is 1) that
minimises ``evaluate_sequence``.

The search consists of three stages:

1️⃣  **Diversified seeding** –  Analytic families (uniform, arcsine,
    symmetric beta, edge‑weighted, edge‑inverse‑power, sine‑weighted,
    quadratic, exponential‑decay) are instantiated for several lengths
    and a dense grid of parameters.  Random Dirichlet and sparse
    vectors are also added.  The ~10 best seeds (lowest exact objective)
    are retained.

2️⃣  **Smooth mirror‑descent** –  From each retained seed a mirror‑descent
    on the *smooth* approximation of the objective is run.  The smooth
    max is  Lα(a)=1/α·log∑exp(α·(a∗a)), whose gradient can be written as
    a correlation between the current probability vector and a softmax
    weight on the convolution entries.  The gradient is obtained by FFT
    (O(n log n)).  An exponentiated‑gradient update with momentum is
    performed, the iterate is projected onto the simplex and the exact
    objective is evaluated periodically.  The smoothing parameter α is
    increased slowly, turning the smooth objective into the true max.

3️⃣  **Local polishing** –  A greedy pairwise mass‑transfer step followed
    by a cheap simulated‑annealing style search removes any remaining
    irregularities.

The routine respects the time budget (default 1000 s) and returns a plain
Python list that can be fed directly to ``evaluate_sequence``.

# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x ≥ 0, sum(x) = s}."""
    if s < 0:
        raise ValueError("simplex radius must be non‑negative")
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho_arr = np.where(u - cssv / (np.arange(1, len(v) + 1)) > 0)[0]
    if rho_arr.size == 0:
        theta = 0.0
    else:
        rho = rho_arr[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Full self‑convolution a∗a using a real FFT (length 2n‑1)."""
    n = a.shape[0]
    need = 2 * n - 1
    size = 1 << ((need - 1).bit_length())
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[:need]


# ----------------------------------------------------------------------
def _objective(a: np.ndarray) -> float:
    """Exact objective 2·n·max(a∗a) for a probability vector (sum=1)."""
    n = a.shape[0]
    b = _fft_self_convolve(a)
    return 2.0 * n * b.max()

# Analytic families ----------------------------------------------------
def _distribution_uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _distribution_arcsine(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    vals /= vals.sum()
    return vals


def _distribution_beta_sym(n: int, p: float) -> np.ndarray:
    """Symmetric beta density  (Beta(p,p))."""
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    w = np.exp((p - 1) * np.log(x) + (p - 1) * np.log(1.0 - x))
    s = w.sum()
    if s == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= s
    return w


def _distribution_beta_general(n: int, alpha: float, beta: float) -> np.ndarray:
    """Asymmetric beta density (Beta(alpha,beta))."""
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    eps = np.finfo(float).tiny
    x = np.clip(x, eps, 1 - eps)
    w = np.exp((alpha - 1) * np.log(x) + (beta - 1) * np.log(1.0 - x))
    s = w.sum()
    if s == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= s
    return w


def _distribution_edge_weighted(n: int, p: float) -> np.ndarray:
    """U‑shaped family w_i ∝ ((i+½)(n‑i‑½))^{−p}."""
    i = np.arange(n, dtype=np.float64) + 0.5
    j = (n - np.arange(n, dtype=np.float64) - 0.5)
    w = (i * j) ** (-p)
    s = w.sum()
    if s == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= s
    return w


def _distribution_edge_inversepow(n: int, p: float) -> np.ndarray:
    """Second edge family w_i ∝ (i+½)^{−p} + (n−i−½)^{−p}."""
    i = np.arange(n, dtype=np.float64) + 0.5
    j = (n - np.arange(n, dtype=np.float64) - 0.5)
    w = i ** (-p) + j ** (-p)
    s = w.sum()
    if s == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= s
    return w


def _distribution_sine_weighted(n: int, p: float) -> np.ndarray:
    """Weight ∝ sin(π (i+½)/n)^p, vanishing at the ends."""
    theta = np.pi * (np.arange(n, dtype=np.float64) + 0.5) / n
    w = np.sin(theta) ** p
    s = w.sum()
    if s == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= s
    return w


def _distribution_quadratic(n: int) -> np.ndarray:
    """Symmetric quadratic bell."""
    i = np.arange(n, dtype=np.float64)
    half = (n - 1) / 2.0
    d = np.abs(i - half) + 1.0
    w = 1.0 / (d ** 2)
    w /= w.sum()
    return w


def _distribution_exponential_decay(n: int, rate: float) -> np.ndarray:
    """Exponential decay from the centre."""
    i = np.arange(n, dtype=np.float64)
    half = (n - 1) / 2.0
    w = np.exp(-rate * np.abs(i - half))
    w /= w.sum()
    return w


def _distribution_random_sparse(
    n: int, rng: np.random.Generator, density: float = 0.01
) -> np.ndarray:
    """Sparse random vector (density controls number of non‑zeros)."""
    k = max(2, int(n * density))
    idx = rng.choice(n, size=k, replace=False)
    vals = np.zeros(n, dtype=np.float64)
    vals[idx] = rng.random(k)
    if vals.sum() == 0.0:
        vals[0] = 1.0
    vals /= vals.sum()
    return vals


def _distribution_dirichlet(n: int, rng: np.random.Generator) -> np.ndarray:
    """Random Dirichlet sample with a random concentration."""
    alpha = rng.uniform(0.01, 2.0)
    a = rng.dirichlet(np.full(n, alpha))
    return a.astype(np.float64)


def _mix_with_uniform(base: np.ndarray, lam: float) -> np.ndarray:
    """Convex combination lam·U + (1‑lam)·base."""
    n = base.shape[0]
    uniform = np.full(n, 1.0 / n, dtype=np.float64)
    return lam * uniform + (1.0 - lam) * base

# Smooth mirror‑descent (exponentiated gradient) -------------------------
def _smooth_descent(
    a: np.ndarray,
    deadline: float,
    best_seq: np.ndarray,
    best_score: float,
    lr0: float = 0.6,
    alpha0: float = 5.0,
    alpha_max: float = 80.0,
    beta: float = 0.9,
    eval_interval: int = 150,
    patience: int = 3000,
) -> tuple[np.ndarray, float]:
    """
    Exponentiated‑gradient descent on the smoothed max objective.
    ``a`` must sum to 1.  Returns the best sequence seen and its exact
    objective value.
    """
    n = a.shape[0]
    t = 0
    no_improve = 0
    momentum = np.zeros_like(a)
    rng = np.random.default_rng()
    # pre‑compute the FFT size used for the correlation step:
    need_corr = 3 * n - 2
    size_corr = 1 << ((need_corr - 1).bit_length())

    while time.time() < deadline and t < 400_000:
        t += 1

        # ----- convolution and smooth weights --------------------------------
        conv = _fft_self_convolve(a)                      # length 2n‑1
        max_c = conv.max()

        # schedule for smoothing parameter α
        alpha = alpha0 + (alpha_max - alpha0) * min(1.0, t / 15000.0)

        # softmax weights (numerically stable)
        w = np.exp(alpha * (conv - max_c))
        w_sum = w.sum()
        if w_sum == 0.0:
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        # ----- gradient via correlation (FFT) -------------------------------
        # correlation r[k] = Σ_i a[i] * w[i + k]  for k ≥ 0 .
        A = np.fft.rfft(a, n=size_corr)
        W = np.fft.rfft(w, n=size_corr)
        corr = np.fft.irfft(np.conj(A) * W, n=size_corr)   # real values
        grad = 2.0 * corr[:n]                              # gradient of smooth max

        # ----- momentum and exponential update -----------------------------
        momentum = beta * momentum + (1.0 - beta) * grad
        eta = lr0 / math.sqrt(t)
        a = a * np.exp(-eta * momentum)                     # stay positive
        a = _project_onto_simplex(a, s=1.0)                 # simplex projection

        # ----- periodic exact evaluation ------------------------------------
        if t % eval_interval == 0:
            cur = 2.0 * n * max_c          # exact objective (sum(a)=1)
            if cur < best_score - 1e-12:
                best_score = cur
                best_seq = a.copy()
                no_improve = 0
                print(f"[{time.time():.1f}] it={t:,} new global best {best_score:.6f}")
            else:
                no_improve += 1

        # ----- jitter when stuck --------------------------------------------
        if no_improve >= patience:
            jitter = rng.exponential(scale=0.008, size=n)
            a = a + jitter
            a = np.maximum(a, 0.0)
            a = _project_onto_simplex(a, s=1.0)
            no_improve = 0

    return best_seq, best_score

# Greedy pairwise refinement -------------------------------------------
def _pairwise_swap_local(
    a: np.ndarray,
    deadline: float,
    best_seq: np.ndarray,
    best_score: float,
    max_trials: int | None = None,
) -> tuple[np.ndarray, float]:
    """Transfer a random amount of mass between two positions if it improves."""
    n = a.shape[0]
    trials = 0
    rng = np.random.default_rng()
    while time.time() < deadline:
        if max_trials is not None and trials >= max_trials:
            break
        trials += 1

        src = np.where(a > 1e-12)[0]
        if src.size == 0:
            break
        i = rng.choice(src)
        j = rng.integers(n - 1)
        if j >= i:
            j += 1

        delta = a[i] * rng.random() * 0.35
        if delta < 1e-14:
            continue

        new_a = a.copy()
        new_a[i] -= delta
        new_a[j] += delta
        cur = _objective(new_a)

        if cur < best_score - 1e-12:
            a = new_a
            best_score = cur
            best_seq = a.copy()
            print(f"[{time.time():.1f}] swap {i}->{j} Δ={delta:.3e} new best {best_score:.6f}")

    return best_seq, best_score

# Simulated‑annealing style local search -------------------------------
def _random_local_search(
    a: np.ndarray,
    deadline: float,
    best_seq: np.ndarray,
    best_score: float,
    init_temp: float = 0.01,
    decay: float = 0.9995,
) -> tuple[np.ndarray, float]:
    """Cheap SA that repeatedly moves a random mass chunk."""
    n = a.shape[0]
    rng = np.random.default_rng()
    T = init_temp
    while time.time() < deadline:
        i = rng.integers(n)
        j = rng.integers(n - 1)
        if j >= i:
            j += 1
        delta = a[i] * rng.random() * 0.3
        if delta < 1e-14:
            continue
        new_a = a.copy()
        new_a[i] -= delta
        new_a[j] += delta
        cur = _objective(new_a)
        if cur < best_score - 1e-12:
            a = new_a
            best_score = cur
            best_seq = a.copy()
            print(f"[{time.time():.1f}] SA improve {i}->{j} Δ={delta:.3e} best {best_score:.6f}")
        else:
            if rng.random() < math.exp(-(cur - best_score) / max(T, 1e-12)):
                a = new_a
        T *= decay
    return best_seq, best_score

        if time.time() >= deadline:
            break
        print(f"[{time.time():.1f}] generating seeds for length {n}")

        # deterministic families
        for cand in (
            _distribution_uniform(n),
            _distribution_arcsine(n),
            _distribution_quadratic(n),
            _distribution_tent(n) if False else _distribution_quadratic(n),  # placeholder
        ):
            sc = _objective(cand)
            top_candidates.append((sc, cand.copy()))
            if sc < overall_best_score:
                overall_best_score, overall_best_seq = sc, cand.copy()
                print(f"[{time.time():.1f}] new best (det) len={n} score={sc:.6f}")

        # parametrised families with mixing
        for p in p_grid:
            if time.time() >= deadline:
                break
            edge = _distribution_edge_weighted(n, p)
            edge2 = _distribution_edge_inversepow(n, p)
            sine = _distribution_sine_weighted(n, p)
            beta = _distribution_beta_sym(n, p)

            for base in (edge, edge2, sine, beta):
                for lam in lam_grid:
                    if time.time() >= deadline:
                        break
                    cand = _mix_with_uniform(base, lam)
                    sc = _objective(cand)
                    top_candidates.append((sc, cand.copy()))
                    if sc < overall_best_score:
                        overall_best_score, overall_best_seq = sc, cand.copy()
                        print(f"[{time.time():.1f}] new best param len={n} p={p:.3g} lam={lam:.2f} score={sc:.6f}")

        # exponential decay family
        for rate in rate_grid:
            if time.time() >= deadline:
                break
            cand = _distribution_exponential_decay(n, rate)
            sc = _objective(cand)
            top_candidates.append((sc, cand.copy()))
            if sc < overall_best_score:
                overall_best_score, overall_best_seq = sc, cand.copy()
                print(f"[{time.time():.1f}] new best exp len={n} rate={rate:.4f} score={sc:.6f}")

        # random Dirichlet seeds
        for _ in range(5):
            if time.time() >= deadline:
                break
            a = _distribution_dirichlet(n, rng)
            sc = _objective(a)
            top_candidates.append((sc, a.copy()))
            if sc < overall_best_score:
                overall_best_score, overall_best_seq = sc, a.copy()
                print(f"[{time.time():.1f}] new best Dirichlet len={n} score={sc:.6f}")

        # sparse random seeds
        for _ in range(5):
            if time.time() >= deadline:
                break
            a = _distribution_random_sparse(n, rng, density=0.008)
            sc = _objective(a)
            top_candidates.append((sc, a.copy()))
            if sc < overall_best_score:
                overall_best_score, overall_best_seq = sc, a.copy()
                print(f"[{time.time():.1f}] new best sparse len={n} score={sc:.6f}")

        # asymmetric beta samples
        for _ in range(4):
            if time.time() >= deadline:
                break
            alpha = 10 ** rng.uniform(-2, 1)   # 0.01 … 10
            beta_ = 10 ** rng.uniform(-2, 1)
            a = _distribution_beta_general(n, alpha, beta_)
            sc = _objective(a)
            top_candidates.append((sc, a.copy()))
            if sc < overall_best_score:
                overall_best_score, overall_best_seq = sc, a.copy()
                print(f"[{time.time():.1f}] new best beta(α={alpha:.3g},β={beta_:.3g}) len={n} score={sc:.6f}")

        # keep only the strongest seeds (limiting memory)
        top_candidates.sort(key=lambda x: x[0])
        top_candidates = top_candidates[:30]

    # ------------------------------------------------------------------
    # 2. Incorporate any previously discovered construction
    try:
        prev = np.asarray(height_sequence_1, dtype=np.float64)  # type: ignore[name-defined]
        if prev.ndim == 1 and prev.size > 0:
            prev = prev / prev.sum()
            sc = _objective(prev)
            if sc < overall_best_score:
                overall_best_score, overall_best_seq = sc, prev.copy()
                print(f"[{time.time():.1f}] previous best score={sc:.6f}")
            top_candidates.append((sc, prev.copy()))
    except Exception:
        pass

    # ------------------------------------------------------------------
    # Guard against not having a seed (unlikely)
    if overall_best_seq is None:
        n = 3000
        overall_best_seq = _distribution_uniform(n)
        overall_best_score = _objective(overall_best_seq)
        print(f"[{time.time():.1f}] fallback uniform score={overall_best_score:.6f}")

    # ------------------------------------------------------------------
    # 3. Smooth mirror‑descent from the top seeds
    top_candidates.sort(key=lambda x: x[0])
    selected = [cand for (_, cand) in top_candidates[:8]]   # up to 8 starts

    remaining = deadline - time.time()
    for idx, cand in enumerate(selected):
        if time.time() >= deadline:
            break
        # allocate roughly equal remaining budget per start, keep a small buffer
        per_start_deadline = time.time() + max(1.0, remaining / (len(selected) - idx) - 5.0)
        print(f"[{time.time():.1f}] smooth descent start {idx+1}/{len(selected)} (len={cand.shape[0]})")
        overall_best_seq, overall_best_score = _smooth_descent(
            cand.copy(),
            per_start_deadline,
            best_seq=overall_best_seq,
            best_score=overall_best_score,
        )
        remaining = deadline - time.time()
        print(f"[{time.time():.1f}] after start {idx+1} best score {overall_best_score:.6f}")

    # ------------------------------------------------------------------
    # 4. Greedy pairwise refinement
    if time.time() < deadline:
        print(f"[{time.time():.1f}] greedy pairwise refinement")
        overall_best_seq, overall_best_score = _pairwise_swap_local(
            overall_best_seq.copy(),
            deadline,
            best_seq=overall_best_seq,
            best_score=overall_best_score,
            max_trials=250_000,
        )

    # ------------------------------------------------------------------
    # 5. Final simulated‑annealing style search
    if time.time() < deadline:
        print(f"[{time.time():.1f}] final simulated‑annealing search")
        overall_best_seq, overall_best_score = _random_local_search(
            overall_best_seq.copy(),
            deadline,
            best_seq=overall_best_seq,
            best_score=overall_best_score,
            init_temp=0.012,
            decay=0.9994,
        )

    # ------------------------------------------------------------------
    print(
        f"Final best exact score ≈ {overall_best_score:.6f} (length={len(overall_best_seq)})"
    )
    return overall_best_seq.tolist()
