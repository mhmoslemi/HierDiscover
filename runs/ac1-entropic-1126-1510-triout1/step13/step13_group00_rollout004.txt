"""
Propose a non‑negative step function that minimizes `evaluate_sequence`.

Algorithm
---------
1. **Seed generation** – Sample a handful of random Dirichlet vectors and a
   few analytic families (uniform, arcsine, symmetric β) for a few moderate
   lengths. Keep the best one (lowest exact objective).

2. **Smooth‑max exponential‑gradient descent** – Parameterise the step
   function as a soft‑max of free variables `x` (so that the vector is always
   non‑negative and sums to 1).  Replace the nondifferentiable `max(conv)`
   by a temperature‑smoothed max  `Mτ = τ·log Σ exp(conv/τ)`.  Compute the exact
   gradient of the smoothed objective using FFT‑based convolutions and
   perform Adam updates.  The temperature `τ` is initially a fraction of the
   current maximum convolution value and is slowly decreased.

3. **Upsampling** – When the current best sequence is optimised, linearly
   interpolate it to a larger length (≈ 2×) and repeat the smooth‑max
   optimisation.  Larger lengths have a strictly smaller theoretical lower
   bound, so this frequently yields a better score.

4. **Local deterministic cleanup** – Finally perform a cheap pairwise mass
   transfer sweep to eliminate any remaining spikes.

All stages respect the global time budget (default 1000 s).  The returned
value is a plain Python list of floats; `evaluate_sequence` will perform its
own clipping.
"""

# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x≥0, sum(x)=s}."""
    if s < 0:
        raise ValueError("simplex radius must be non‑negative")
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho_arr = np.where(u - cssv / (np.arange(1, len(v) + 1)) > 0)[0]
    if rho_arr.size == 0:
        theta = 0.0
    else:
        rho = rho_arr[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _fft_convolve(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full linear convolution a*b using real FFT."""
    n = a.shape[0]
    m = b.shape[0]
    need = n + m - 1
    size = 1 << ((need - 1).bit_length())
    A = rfft(a, n=size)
    B = rfft(b, n=size)
    conv = irfft(A * B, n=size)
    return conv[:need]


def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Convenience wrapper for a*a."""
    return _fft_convolve(a, a)


# ----------------------------------------------------------------------
def _objective(a: np.ndarray) -> float:
    """Exact objective 2·n·max(conv(a,a)) when sum(a)=1."""
    n = a.shape[0]
    b = _fft_self_convolve(a)
    return 2.0 * n * b.max()

# Analytic families ----------------------------------------------------
def _distribution_uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _distribution_arcsine(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    vals /= vals.sum()
    return vals


def _distribution_beta_sym(n: int, p: float) -> np.ndarray:
    """Symmetric Beta(p,p) on the discrete grid."""
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    w = np.exp((p - 1) * np.log(x) + (p - 1) * np.log(1.0 - x))
    if w.sum() == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _distribution_dirichlet(n: int, rng: np.random.Generator) -> np.ndarray:
    alpha = rng.uniform(0.01, 2.0)
    a = rng.dirichlet(np.full(n, alpha))
    return a.astype(np.float64)


def _mix_with_uniform(base: np.ndarray, lam: float) -> np.ndarray:
    """lam·U + (1‑lam)·base, where U is the uniform distribution."""
    n = base.shape[0]
    uniform = np.full(n, 1.0 / n, dtype=np.float64)
    return lam * uniform + (1.0 - lam) * base


# ----------------------------------------------------------------------
def _upsample(seq: np.ndarray, new_len: int) -> np.ndarray:
    """Linear interpolation of a simplex‑vector to a larger length."""
    old_len = seq.shape[0]
    if new_len == old_len:
        return seq.copy()
    x_old = np.linspace(0.0, 1.0, old_len)
    x_new = np.linspace(0.0, 1.0, new_len)
    new_seq = np.interp(x_new, x_old, seq)
    new_seq = np.maximum(new_seq, 0.0)
    new_seq /= new_seq.sum()
    return new_seq

def _smooth_max_optimize(
    init_seq: np.ndarray,
    deadline: float,
    global_best_seq: np.ndarray,
    global_best_score: float,
) -> tuple[np.ndarray, float]:

    Adam optimisation of the smoothed max objective.
    The distribution is represented as softmax(x) so that the simplex constraints
    are always satisfied.
    """
    n = init_seq.shape[0]
    # initialise x (log‑its). add epsilon to avoid log(0)
    eps = 1e-12
    x = np.log(np.clip(init_seq, eps, None))

    m = np.zeros_like(x)
    v = np.zeros_like(x)
    beta1, beta2, eps_adam = 0.9, 0.999, 1e-8
    lr0 = 0.25
    t = 0
    eval_interval = 150
    # temperature schedule
    tau_factor = 0.2
    min_tau_factor = 0.001

    best_seq = global_best_seq.copy()
    best_score = global_best_score

    while time.time() < deadline and t < 600_000:
        t += 1

        # ----- soft‑max -----
        max_x = np.max(x)
        exp_x = np.exp(x - max_x)
        a = exp_x / exp_x.sum()

        # ----- convolution and smooth max -----
        conv = _fft_self_convolve(a)
        max_c = conv.max()
        # temperature proportional to current max
        tau = max(max_c * max(tau_factor, 1e-12), 1e-12)

        # log‑sum‑exp trick
        shift = conv - max_c          # <= 0
        exp_shift = np.exp(shift / tau)
        sum_exp = exp_shift.sum()
        p = exp_shift / sum_exp       # softmax over convolution entries

        # ----- gradient wrt a -----
        a_rev = a[::-1]
        conv_full = _fft_convolve(p, a_rev)           # length = 3n-2
        start = n - 1
        grad_a = 4.0 * n * conv_full[start:start + n]

        # ----- gradient wrt x (softmax chain rule) -----
        dot = np.dot(a, grad_a)
        grad_x = a * (grad_a - dot)

        # ----- Adam update -----
        m = beta1 * m + (1.0 - beta1) * grad_x
        v = beta2 * v + (1.0 - beta2) * (grad_x ** 2)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        lr = lr0 / math.sqrt(t)   # decaying learning rate
        x = x - lr * m_hat / (np.sqrt(v_hat) + eps_adam)

        # ----- periodic evaluation with the *exact* max -----
        if t % eval_interval == 0:
            cur_score = _objective(a)
            if cur_score + 1e-14 < best_score:
                best_score = cur_score
                best_seq = a.copy()
                print(f"[{time.time():.1f}] iter={t:,} new best {best_score:.6f}")
                # reset temperature factor to be a bit larger after an improvement
                tau_factor = max(tau_factor, 0.2)
            else:
                # slowly cool down
                tau_factor = max(min_tau_factor, tau_factor * 0.96)

    return best_seq, best_score


# ----------------------------------------------------------------------
def _pairwise_swap_local(
    a: np.ndarray,
    deadline: float,
    best_seq: np.ndarray,
    best_score: float,
    max_trials: int | None = None,
) -> tuple[np.ndarray, float]:
    """
    Simple deterministic cleanup: move a random fraction of mass from one index
    to another if it improves the exact objective.
    """
    n = a.shape[0]
    trials = 0
    rng = np.random.default_rng()
    while time.time() < deadline:
        if max_trials is not None and trials >= max_trials:
            break
        trials += 1

        src_candidates = np.where(a > 1e-12)[0]
        if src_candidates.size == 0:
            break
        i = rng.choice(src_candidates)
        j = rng.integers(n - 1)
        if j >= i:
            j += 1

        delta = a[i] * rng.random() * 0.33
        if delta < 1e-14:
            continue

        new_a = a.copy()
        new_a[i] -= delta
        new_a[j] += delta

        cur = _objective(new_a)
        if cur + 1e-14 < best_score:
            a = new_a
            best_score = cur
            best_seq = a.copy()
            print(f"[{time.time():.1f}] swap {i}->{j} δ={delta:.3e} new best {best_score:.6f}")

    return best_seq, best_score


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: int = 1000) -> list[float]:
    """
    Search for a non‑negative step function that minimises ``evaluate_sequence``.
    Returns a plain Python list of heights for the final submission.
    """
    rng = np.random.default_rng(seed)
    start = time.time()
    # safety margin so we always return something before the hard timeout
    deadline = start + budget_s - 2.0

    # --------------------------------------------------------------
    # 0. initial best – try to use the previously discovered construction
    try:
        prev = np.asarray(height_sequence_1, dtype=np.float64)  # type: ignore[name-defined]
        if prev.ndim == 1 and prev.size > 0:
            prev = np.maximum(prev, 0.0)
            if prev.sum() == 0.0:
                raise ValueError
            prev = prev / prev.sum()
            overall_best_seq = prev.copy()
        else:
            raise ValueError
    except Exception:
        overall_best_seq = _distribution_uniform(2000)

    overall_best_score = _objective(overall_best_seq)
    print(f"[{time.time():.1f}] initial best score {overall_best_score:.6f}")

    # --------------------------------------------------------------
    # 1. Quick seed generation for a few lengths
    seed_lengths = [2000, 3000, 4000]
    for n in seed_lengths:
        if time.time() >= deadline:
            break
        # try a few random Dirichlet vectors
        for _ in range(8):
            cand = _distribution_dirichlet(n, rng)
            sc = _objective(cand)
            if sc < overall_best_score:
                overall_best_score, overall_best_seq = sc, cand.copy()
                print(f"[{time.time():.1f}] seed length={n} random improved to {sc:.6f}")

        # try a couple of analytic families
        for base in (_distribution_uniform(n), _distribution_arcsine(n), _distribution_beta_sym(n, 0.5)):
            sc = _objective(base)
            if sc < overall_best_score:
                overall_best_score, overall_best_seq = sc, base.copy()
                print(f"[{time.time():.1f}] seed length={n} analytic improved to {sc:.6f}")

    # --------------------------------------------------------------
    # 2. Smooth‑max Adam optimisation on the current best,
    #    upsamping progressively to larger sizes
    progressive_sizes = [len(overall_best_seq), 6000, 12000, 18000]
    for target_n in progressive_sizes:
        if time.time() >= deadline:
            break
        # upsample if needed
        seq = overall_best_seq
        if seq.shape[0] != target_n:
            seq = _upsample(seq, target_n)

        remaining = deadline - time.time()
        # reserve a fraction for the optimisation; the rest will be used for later stages
        allocate = min(remaining * 0.55, 240.0)   # at most ~4 min per size
        sub_deadline = time.time() + allocate

        seq_opt, sc_opt = _smooth_max_optimize(
            seq.copy(),
            sub_deadline,
            best_seq=overall_best_seq,
            best_score=overall_best_score,
        )
        if sc_opt < overall_best_score:
            overall_best_score, overall_best_seq = sc_opt, seq_opt.copy()
            print(f"[{time.time():.1f}] after size {target_n} new best {overall_best_score:.6f}")

        # occasional random restart at the current length
        if time.time() < deadline:
            rand_seq = _distribution_dirichlet(target_n, rng)
            sub_deadline2 = time.time() + min(30.0, deadline - time.time())
            seq_opt2, sc_opt2 = _smooth_max_optimize(
                rand_seq,
                sub_deadline2,
                best_seq=overall_best_seq,
                best_score=overall_best_score,
            )
            if sc_opt2 < overall_best_score:
                overall_best_score, overall_best_seq = sc_opt2, seq_opt2.copy()
                print(f"[{time.time():.1f}] random restart size {target_n} improved to {sc_opt2:.6f}")

    # --------------------------------------------------------------
    # 3. Deterministic pairwise refinement (cheap but often squeezes a bit more)
    if time.time() < deadline:
        print(f"[{time.time():.1f}] final pairwise refinement")
        overall_best_seq, overall_best_score = _pairwise_swap_local(
            overall_best_seq.copy(),
            deadline,
            best_seq=overall_best_seq,
            best_score=overall_best_score,
            max_trials=250_000,
        )

    # --------------------------------------------------------------
    print(f"Final best score: {overall_best_score:.6f} (length={len(overall_best_seq)})")
    # Return a plain Python list; evaluate_sequence will do its own clipping.
    return overall_best_seq.tolist()
