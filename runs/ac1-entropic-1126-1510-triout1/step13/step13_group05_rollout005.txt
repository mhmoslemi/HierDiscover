def _interpolate_sequence(a, new_n):
    n = a.shape[0]
    if n == new_n:
        return a.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(new_n) + 0.5) / new_n
    a_new = np.interp(x_new, x_old, a, left=0.0, right=0.0)
    # to ensure sum=1
    total = a_new.sum()
    if total == 0:
        a_new = np.full(new_n, 1.0 / new_n)
    else:
        a_new = a_new / total
    return a_new

# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x ≥ 0, Σx = s}."""
    if s < 0:
        raise ValueError("Simplex radius must be non‑negative")
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.where(u - cssv / (np.arange(1, len(v) + 1)) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    return np.maximum(v - theta, 0.0)


# ----------------------------------------------------------------------
def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Full auto‑convolution a∗a (length 2·n‑1) using a real FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1) - 1).bit_length()
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[: 2 * n - 1]


# ----------------------------------------------------------------------
def _objective(a: np.ndarray) -> float:
    """Exact objective 2·n·max(conv) for a probability vector (Σa = 1)."""
    n = a.shape[0]
    return float(2.0 * n * _fft_self_convolve(a).max())

# Analytic families ----------------------------------------------------
def _distribution_uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _distribution_arcsine(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    vals /= vals.sum()
    return vals


def _distribution_sine(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64) + 0.5
    w = np.sin(np.pi * i / n)
    w /= w.sum()
    return w


def _distribution_cosine(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64) + 0.5
    w = 1.0 - np.cos(np.pi * i / n)
    w /= w.sum()
    return w


def _distribution_quadratic(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    x = (2.0 * i) / (n - 1) - 1.0
    w = 1.0 - x * x
    w = np.clip(w, 0.0, None)
    w /= w.sum()
    return w


def _distribution_beta_sym(n: int, p: float) -> np.ndarray:
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    w = np.exp((p - 1) * np.log(x) + (p - 1) * np.log(1.0 - x))
    if w.sum() == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _distribution_edge_weighted(n: int, p: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64) + 0.5
    j = (n - np.arange(n, dtype=np.float64) - 0.5)
    w = (i * j) ** (-p)
    w = np.clip(w, 0.0, None)
    total = w.sum()
    if total > 0:
        w /= total
    else:
        w = np.full(n, 1.0 / n)
    return w


def _distribution_power_sum(n: int, p: float) -> np.ndarray:
    i = np.arange(1, n + 1, dtype=np.float64)
    w = i ** (-p) + (n + 1 - i) ** (-p)
    w = np.clip(w, 0.0, None)
    w /= w.sum()
    return w


def _distribution_random_sparse(n: int, rng: np.random.Generator,
                               density: float = 0.01) -> np.ndarray:
    k = max(2, int(n * density))
    idx = rng.choice(n, size=k, replace=False)
    vals = np.zeros(n, dtype=np.float64)
    vals[idx] = rng.random(k)
    if vals.sum() == 0.0:
        vals[0] = 1.0
    vals /= vals.sum()
    return vals


def _mix_with_uniform(base: np.ndarray, lam: float) -> np.ndarray:
    """Convex combination λ·uniform + (1‑λ)·base."""
    n = base.shape[0]
    uniform = np.full(n, 1.0 / n, dtype=np.float64)
    return lam * uniform + (1.0 - lam) * base


# ----------------------------------------------------------------------
# Exact sub‑gradient (max‑convolution) ----------------------------------
def _max_convolution_subgradient(a: np.ndarray) -> np.ndarray:
    """Sub‑gradient of max_k Σ_i a_i a_{k‑i} (active set based)."""
    conv = _fft_self_convolve(a)
    max_val = conv.max()
    tol = 1e-12 * max_val
    active = np.where(conv >= max_val - tol)[0]
    n = a.shape[0]
    grad = np.zeros_like(a)
    for k in active:
        lo = max(0, k - (n - 1))
        hi = min(n - 1, k)
        idx = np.arange(lo, hi + 1)
        grad[idx] += 2.0 * a[k - idx]
    if active.size:
        grad /= active.size
    return grad


# ----------------------------------------------------------------------
# Smoothed gradient (log‑sum‑exp of convolution) -----------------------
def _smooth_max_subgradient(a: np.ndarray, tau: float) -> np.ndarray:
    """
    Gradient of   fτ(a) = (1/τ)·log Σ_k exp(τ·(a∗a)_k) .
    Returns a vector of shape a.  Uses FFT‑based correlation.
    """
    n = a.shape[0]
    conv = _fft_self_convolve(a)                     # length 2n‑1
    # soft‑max weights
    exp_term = np.exp(tau * conv - np.max(tau * conv))
    w = exp_term / exp_term.sum()                    # length 2n‑1

    # gradient = 2 * (a ⋆ w)   (correlation of a with w)
    m = w.shape[0]
    size = 1 << ((m + n - 1) - 1).bit_length()       # enough for linear correlation
    Fa = np.fft.rfft(a, n=size)
    Fw = np.fft.rfft(w, n=size)
    corr = np.fft.irfft(np.conj(Fw) * Fa, n=size)    # linear correlation a⋆w
    grad = 2.0 * corr[:n]                            # keep first n lags
    return grad


# ----------------------------------------------------------------------
# Exponentiated‑gradient descent (exact max) ---------------------------
def _exp_gradient_descent_exact(a0: np.ndarray,
                               time_limit: float,
                               best_seq: np.ndarray,
                               best_score: float,
                               eta0: float = 0.6,
                               report_every: int = 500) -> tuple[np.ndarray, float]:
    """Multiplicative mirror‑descent using the exact sub‑gradient."""
    a = a0.copy()
    n = a.shape[0]
    t = 0
    while time.time() < time_limit:
        t += 1
        grad = _max_convolution_subgradient(a)
        eta = eta0 / math.sqrt(t)
        a *= np.exp(-eta * grad)
        a = _project_onto_simplex(a)          # keep Σa = 1
        if t % report_every == 0:
            cur = _objective(a)
            if cur < best_score - 1e-14:
                best_score = cur
                best_seq = a.copy()
                print(f"[{time.time():.1f}] exact‑gd it={t} new best {best_score:.6f}")
    return best_seq, best_score


# ----------------------------------------------------------------------
# Exponentiated‑gradient descent (smooth surrogate) --------------------
def _exp_gradient_descent_smooth(a0: np.ndarray,
                                tau_schedule: np.ndarray,
                                time_limit: float,
                                best_seq: np.ndarray,
                                best_score: float,
                                eta0: float = 0.6,
                                report_every: int = 200) -> tuple[np.ndarray, float]:
    """
    Optimises the smooth surrogate fτ with a growing temperature τ.
    The update rule is the same multiplicative scheme as the exact
    method, but the gradient comes from the log‑sum‑exp surrogate.
    """
    a = a0.copy()
    n = a.shape[0]
    total_iters = 0
    for tau in tau_schedule:
        # give each temperature a modest quota (the outer while respects the global deadline)
        while time.time() < time_limit:
            total_iters += 1
            grad = _smooth_max_subgradient(a, tau)
            eta = eta0 / math.sqrt(total_iters)
            a *= np.exp(-eta * grad)
            a = _project_onto_simplex(a)

            if total_iters % report_every == 0:
                cur = _objective(a)          # exact value for monitoring
                if cur < best_score - 1e-14:
                    best_score = cur
                    best_seq = a.copy()
                    print(f"[{time.time():.1f}] smooth τ={tau:.2f} it={total_iters} new best {best_score:.6f}")
        # move to next tau
    return best_seq, best_score


# ----------------------------------------------------------------------
# Simple interpolation (piecewise‑linear) -----------------------------
def _interpolate_sequence(a: np.ndarray, new_n: int) -> np.ndarray:
    """Interpolate a probability vector a (sum = 1) to length new_n."""
    n = a.shape[0]
    if new_n == n:
        return a.copy()
    # mid‑points of the original and target grids
    x_old = (np.arange(n, dtype=np.float64) + 0.5) / n
    x_new = (np.arange(new_n, dtype=np.float64) + 0.5) / new_n
    a_new = np.interp(x_new, x_old, a, left=0.0, right=0.0)
    total = a_new.sum()
    if total == 0.0:
        a_new = np.full(new_n, 1.0 / new_n, dtype=np.float64)
    else:
        a_new /= total
    return a_new


# ----------------------------------------------------------------------
# Pairwise mass‑transfer hill‑climbing ---------------------------------
def _pairwise_swap(a0: np.ndarray,
                   time_limit: float,
                   best_seq: np.ndarray,
                   best_score: float,
                   max_trials: int | None = None) -> tuple[np.ndarray, float]:
    """Randomly move a fraction of mass from i to j; keep strictly improving moves."""
    a = a0.copy()
    n = a.shape[0]
    trials = 0
    while time.time() < time_limit:
        if max_trials is not None and trials >= max_trials:
            break
        trials += 1
        src = np.where(a > 1e-14)[0]
        if src.size == 0:
            break
        i = int(np.random.choice(src))
        j = np.random.randint(n - 1)
        if j >= i:
            j += 1
        delta = a[i] * np.random.rand() * 0.45
        if delta < 1e-15:
            continue
        new_a = a.copy()
        new_a[i] -= delta
        new_a[j] += delta
        cur = _objective(new_a)
        if cur < best_score - 1e-14:
            a = new_a
            best_score = cur
            best_seq = a.copy()
            print(f"[{time.time():.1f}] swap i={i}→j={j} δ={delta:.2e} new best {best_score:.6f}")
    return best_seq, best_score


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: int = 1000) -> list[float]:
    """
    Search for a step‑function that minimises ``evaluate_sequence``.
    The routine follows three stages (scan → refine → upsample → polish)
    while staying within the supplied time budget.
    """
    rng = np.random.default_rng(seed)
    start = time.time()
    deadline = start + budget_s - 2.0          # safety margin

    overall_best_seq = None
    overall_best_score = float("inf")
    top_candidates: list[tuple[float, np.ndarray]] = []

    # ------------------------------------------------------------------
    # 0.  If a previous construction exists, seed the search with it.
    # ------------------------------------------------------------------
    try:
        prev = np.asarray(height_sequence_1, dtype=np.float64)  # type: ignore[name-defined]
        if prev.ndim == 1 and prev.size > 0:
            prev = np.maximum(prev, 0.0)
            s = prev.sum()
            if s == 0.0:
                prev = _distribution_uniform(prev.size)
            else:
                prev = prev / s
            sc = _objective(prev)
            overall_best_seq = prev.copy()
            overall_best_score = sc
            top_candidates.append((sc, prev.copy()))
            print(f"[{time.time():.1f}] loaded previous construction, score={sc:.6f}")
    except Exception:
        pass

    # ------------------------------------------------------------------
    # 1.  Scan a modest set of analytic families on a *single* moderate length.
    # ------------------------------------------------------------------
    n0 = 4096
    print(f"[{time.time():.1f}] scanning analytic families, n={n0}")

    families = [
        _distribution_uniform,
        _distribution_arcsine,
        _distribution_sine,
        _distribution_cosine,
        _distribution_quadratic,
    ]

    for fn in families:
        cand = fn(n0)
        sc = _objective(cand)
        top_candidates.append((sc, cand))
        if sc < overall_best_score:
            overall_best_score = sc
            overall_best_seq = cand.copy()
            print(f"[{time.time():.1f}] new global best (family {fn.__name__}) score={sc:.6f}")

    # parametric families
    p_grid = np.concatenate([
        np.logspace(-2, -0.3, 12),           # 0.01 … 0.5
        np.linspace(0.5, 2.5, 15),           # 0.5 … 2.5
        np.linspace(2.5, 6.0, 10),           # 2.5 … 6
    ])
    lam_grid = np.linspace(0.0, 1.0, 5)  # 0, .25, .5, .75, 1

    for p in p_grid:
        if time.time() >= deadline:
            break
        ed = _distribution_edge_weighted(n0, p)
        bet = _distribution_beta_sym(n0, p)
        pw = _distribution_power_sum(n0, p)
        for base in (ed, bet, pw):
            for lam in lam_grid:
                if time.time() >= deadline:
                    break
                cand = _mix_with_uniform(base, lam)
                sc = _objective(cand)
                top_candidates.append((sc, cand))
                if sc < overall_best_score:
                    overall_best_score = sc
                    overall_best_seq = cand.copy()
                    print(f"[{time.time():.1f}] new best p={p:.3g} lam={lam:.2f} score={sc:.6f}")

    # random Dirichlet seeds
    for _ in range(8):
        if time.time() >= deadline:
            break
        alpha = rng.uniform(0.01, 2.0)
        a = rng.dirichlet(np.full(n0, alpha))
        sc = _objective(a)
        top_candidates.append((sc, a))
        if sc < overall_best_score:
            overall_best_score = sc
            overall_best_seq = a.copy()
            print(f"[{time.time():.1f}] Dirichlet seed score={sc:.6f}")

    # sparse random seeds
    for _ in range(6):
        if time.time() >= deadline:
            break
        a = _distribution_random_sparse(n0, rng, density=0.010)
        sc = _objective(a)
        top_candidates.append((sc, a))
        if sc < overall_best_score:
            overall_best_score = sc
            overall_best_seq = a.copy()
            print(f"[{time.time():.1f}] sparse seed score={sc:.6f}")

    # keep only the most promising few candidates
    top_candidates.sort(key=lambda x: x[0])
    top_candidates = top_candidates[:12]      # a dozen seeds are enough for the next stage

    # ------------------------------------------------------------------
    # 2.  Refine those seeds with exact exponentiated‑gradient descent.
    # ------------------------------------------------------------------
    if time.time() < deadline:
        remaining = deadline - time.time()
        per_seed = remaining / max(1, len(top_candidates))
        print(f"[{time.time():.1f}] exact mirror‑descent on {len(top_candidates)} seeds")
        for idx, (score, seed_seq) in enumerate(top_candidates):
            if time.time() >= deadline:
                break
            sg_deadline = time.time() + per_seed
            best_seq, best_score = _exp_gradient_descent_exact(
                seed_seq,
                sg_deadline,
                best_seq=overall_best_seq,
                best_score=overall_best_score,
                eta0=0.7,
                report_every=200,
            )
            if best_score < overall_best_score - 1e-12:
                overall_best_score = best_score
                overall_best_seq = best_seq.copy()
                print(f"[{time.time():.1f}] after exact‑gd seed {idx} new global best {overall_best_score:.6f}")

    # ------------------------------------------------------------------
    # 3.  Upsample the current best shape to larger lengths and polish.
    # ------------------------------------------------------------------
    upsample_lengths = [8192, 16384, 32768]   # larger problems (if time permits)
    for L in upsample_lengths:
        if time.time() >= deadline:
            break
        print(f"[{time.time():.1f}] upsampling to length {L}")
        cand = _interpolate_sequence(overall_best_seq, L)
        sc = _objective(cand)
        if sc < overall_best_score:
            overall_best_score = sc
            overall_best_seq = cand.copy()
            print(f"[{time.time():.1f}] upsample gave immediate improvement score={sc:.6f}")

        # smooth exponential‑gradient descent on the upsampled vector
        tau_sched = np.geomspace(0.5, 16.0, num=6)   # 0.5 → 1 → 2 → 4 → 8 → 16
        sg_deadline = deadline - 1.0               # keep a tiny slack for final polishing
        best_seq, best_score = _exp_gradient_descent_smooth(
            cand,
            tau_schedule=tau_sched,
            time_limit=sg_deadline,
            best_seq=overall_best_seq,
            best_score=overall_best_score,
            eta0=0.5,
            report_every=300,
        )
        if best_score < overall_best_score - 1e-12:
            overall_best_score = best_score
            overall_best_seq = best_seq.copy()
            print(f"[{time.time():.1f}] after smooth‑gd length {L} new best {overall_best_score:.6f}")

    # ------------------------------------------------------------------
    # 4.  Final greedy polishing (pairwise swaps).
    # ------------------------------------------------------------------
    if time.time() < deadline:
        print(f"[{time.time():.1f}] final greedy polishing")
        best_seq, best_score = _pairwise_swap(
            overall_best_seq.copy(),
            deadline,
            best_seq=overall_best_seq,
            best_score=overall_best_score,
            max_trials=None,
        )
        if best_score < overall_best_score - 1e-12:
            overall_best_score = best_score
            overall_best_seq = best_seq.copy()
            print(f"[{time.time():.1f}] after polishing new best {overall_best_score:.6f}")

    # ------------------------------------------------------------------
    # 5.  Fallback (should never happen).
    # ------------------------------------------------------------------
    if overall_best_seq is None:
        overall_best_seq = _distribution_uniform(5000)
        overall_best_score = _objective(overall_best_seq)
        print(f"[{time.time():.1f}] fallback uniform, score={overall_best_score:.6f}")

    print(f"[{time.time():.1f}] FINAL best score (2·n·max(conv)) = {overall_best_score:.6f} "
          f"length = {len(overall_best_seq)}")
    return overall_best_seq.tolist()
