"""
Propose a non‑negative step function that (almost) minimises the
``evaluate_sequence`` objective.

The algorithm performs the following stages (all respecting a 1000 s
budget):

1. **Diverse seed generation** – a large pool of analytic families
   (uniform, arcsine, Hann, symmetric/asymmetric Beta, edge‑weighted,
   generalized‑normal, power‑law, random Dirichlet and sparse vectors)
   for several lengths.  The best 50 are kept.

2. **Exponentiated‑gradient descent** – a mirror‑descent on the simplex
   using the sub‑gradient of the active convolution entry.  The best
   few seeds are refined for a sizable fraction of the remaining time.

3. **Frank‑Wolfe refinement** – a classic FW step that moves mass toward
   the coordinate with the smallest gradient.

4. **Targeted pairwise annealing** – repeatedly transfers mass from the
   index that contributes most to the current maximal convolution entry
   to the index that contributes least, accepting improving moves and
   occasionally worsening ones with a linear cooling schedule.

5. **Upsampling & final polishing** – the current best vector is
   interpolated to twice its length, a short burst of EG descent is run,
   and a long‑run generic pairwise annealing finishes the search.

All optimisation steps keep the vector on the probability simplex
(`sum = 1`, `a_i ≥ 0`).  The returned list can be fed directly to
``evaluate_sequence``.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Simplex projection (Euclidean)
# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Project v onto the simplex {x ≥ 0, Σx = s}."""
    if s < 0:
        raise ValueError("simplex radius must be non‑negative")
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.where(u - cssv / (np.arange(1, len(v) + 1)) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# FFT based linear self‑convolution
# ----------------------------------------------------------------------
def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Full linear self‑convolution a * a using a real FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1) - 1).bit_length()   # next power of two
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[: 2 * n - 1]


# ----------------------------------------------------------------------
# Exact objective used by ``evaluate_sequence`` (scale‑invariant)
# ----------------------------------------------------------------------
def _objective(a: np.ndarray) -> float:
    """Return the true value of ``evaluate_sequence`` for a (unnormalised)."""
    n = a.shape[0]
    b = _fft_self_convolve(a)
    max_b = b.max()
    sum_a = a.sum()
    if sum_a < 0.01:
        return float("inf")
    return float(2.0 * n * max_b / (sum_a * sum_a))


# ----------------------------------------------------------------------
# Analytic families – all return a probability vector (sum = 1)
# ----------------------------------------------------------------------
def _distribution_uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _distribution_arcsine(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    vals /= vals.sum()
    return vals


def _distribution_hann(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    w = 0.5 - 0.5 * np.cos(2.0 * np.pi * i / (n - 1))
    w_sum = w.sum()
    if w_sum == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w_sum
    return w


def _distribution_beta_sym(n: int, p: float) -> np.ndarray:
    """Symmetric Beta(p,p) on a uniform grid."""
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    w = np.exp((p - 1) * np.log(x) + (p - 1) * np.log(1.0 - x))
    if w.sum() == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _distribution_beta_asym(n: int, alpha: float, beta: float) -> np.ndarray:
    """Asymmetric Beta(alpha,beta) on a uniform grid."""
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    # avoid overflow/underflow for extreme parameters
    w = np.exp((alpha - 1) * np.log(x) + (beta - 1) * np.log(1.0 - x))
    if w.sum() == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _distribution_edge_weighted(n: int, p: float) -> np.ndarray:
    """Weight ∝ (i·j)^{-p} where i,j are distances to the two edges."""
    i = np.arange(n, dtype=np.float64) + 0.5
    j = (n - np.arange(n, dtype=np.float64) - 0.5)
    w = (i * j) ** (-p)
    if w.sum() == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _distribution_generalized_normal(n: int, q: float) -> np.ndarray:
    """Density ∝ exp(-|x-1/2|^{q}) on [0,1]."""
    i = np.arange(n, dtype=np.float64) + 0.5
    x = i / n - 0.5
    w = np.exp(- (np.abs(x) / 0.5) ** q)
    w_sum = w.sum()
    if w_sum == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w_sum
    return w


def _distribution_power_law(n: int, p: float) -> np.ndarray:
    """Monotone decreasing power‑law (p>0)."""
    i = np.arange(1, n + 1, dtype=np.float64)
    w = i ** (-p)
    w /= w.sum()
    return w


def _distribution_random_sparse(
    n: int, rng: np.random.Generator, density: float = 0.01
) -> np.ndarray:
    """Sparse random vector with a given density."""
    k = max(2, int(n * density))
    idx = rng.choice(n, size=k, replace=False)
    vals = np.zeros(n, dtype=np.float64)
    vals[idx] = rng.random(k)
    if vals.sum() == 0.0:
        vals[0] = 1.0
    vals /= vals.sum()
    return vals


def _mix_with_uniform(base: np.ndarray, lam: float) -> np.ndarray:
    """Convex combination with the uniform distribution."""
    n = base.shape[0]
    uniform = np.full(n, 1.0 / n, dtype=np.float64)
    return lam * uniform + (1.0 - lam) * base


# ----------------------------------------------------------------------
# Sub‑gradient of the currently maximal convolution entry
# ----------------------------------------------------------------------
def _grad_of_max_conv(a: np.ndarray) -> tuple[np.ndarray, int]:
    """
    Compute the sub‑gradient of the maximal self‑convolution entry.
    Returns (grad, active_index).
    """
    n = a.shape[0]
    conv = _fft_self_convolve(a)
    k = int(np.argmax(conv))

    grad = np.zeros_like(a)
    if k < n:
        grad[: k + 1] = 2.0 * a[k::-1]   # overlap length = k+1
    else:
        start = k - (n - 1)
        grad[start:] = 2.0 * a[start:][::-1]   # overlap length = 2·n-1-k
    return grad, k


# ----------------------------------------------------------------------
# Exponentiated‑gradient descent on the true objective (sub‑gradient)
# ----------------------------------------------------------------------
def _exp_gradient_descent(
    a: np.ndarray,
    deadline: float,
    best_seq: np.ndarray,
    best_score: float,
    eta0: float = 0.8,
    eval_every: int = 80,
) -> tuple[np.ndarray, float]:
    """Perform exponentiated‑gradient descent (mirror descent on the simplex)."""
    n = a.shape[0]
    t = 0
    while time.time() < deadline:
        t += 1
        grad, _ = _grad_of_max_conv(a)
        eta = eta0 / math.sqrt(t)
        a = a * np.exp(-eta * grad)
        s = a.sum()
        if s == 0.0:
            a = np.full(n, 1.0 / n, dtype=np.float64)
        else:
            a = a / s

        if t % eval_every == 0:
            cur = _objective(a)
            if cur < best_score - 1e-14:
                best_score = cur
                best_seq = a.copy()
                print(f"[{time.time():.1f}] EG it={t} new global best {best_score:.6f}")
    return best_seq, best_score


# ----------------------------------------------------------------------
# Frank‑Wolfe style refinement (pairwise mass transfer)
# ----------------------------------------------------------------------
def _frank_wolfe_descent(
    a: np.ndarray,
    deadline: float,
    best_seq: np.ndarray,
    best_score: float,
    eval_every: int = 100,
) -> tuple[np.ndarray, float]:
    """Simple Frank‑Wolfe iteration that moves towards the vertex with the
    smallest gradient entry."""
    n = a.shape[0]
    t = 0
    while time.time() < deadline:
        t += 1
        grad, _ = _grad_of_max_conv(a)
        i_min = int(np.argmin(grad))
        gamma = 2.0 / (t + 2.0)               # diminishing step size
        direction = -a
        direction[i_min] = 1.0 - a[i_min]    # add e_i_min, remove a
        a = a + gamma * direction

        if t % eval_every == 0:
            cur = _objective(a)
            if cur < best_score - 1e-14:
                best_score = cur
                best_seq = a.copy()
                print(f"[{time.time():.1f}] FW it={t} new global best {best_score:.6f}")
    return best_seq, best_score


# ----------------------------------------------------------------------
# Targeted pairwise annealing (mass from high‑gradient to low‑gradient)
# ----------------------------------------------------------------------
def _targeted_pairwise_annealing(
    a: np.ndarray,
    deadline: float,
    best_seq: np.ndarray,
    best_score: float,
    rng: np.random.Generator,
) -> tuple[np.ndarray, float]:
    """Move mass from the index that contributes most to max‑conv
    to the one that contributes least, with a linear cooling schedule."""
    n = a.shape[0]
    start = time.time()
    total = deadline - start

    while time.time() < deadline:
        grad, _ = _grad_of_max_conv(a)
        i_src = int(np.argmax(grad))
        i_dst = int(np.argmin(grad))
        if i_src == i_dst:
            break
        # transfer a random fraction of the source mass
        delta = a[i_src] * rng.random() * 0.4  # up to 40 % of source
        if delta < 1e-16:
            continue
        new_a = a.copy()
        new_a[i_src] -= delta
        new_a[i_dst] += delta

        cur = _objective(new_a)
        if cur < best_score - 1e-14:
            a = new_a
            best_seq = new_a.copy()
            best_score = cur
            print(f"[{time.time():.1f}] target anneal {i_src}->{i_dst} δ={delta:.3e} new best {best_score:.6f}")
        else:
            # occasional uphill moves (simulated annealing)
            elapsed = time.time() - start
            temperature = max(0.0, 1.0 - elapsed / total) * best_score * 0.01
            if temperature > 0.0 and rng.random() < math.exp(-(cur - best_score) / temperature):
                a = new_a
    return best_seq, best_score


# ----------------------------------------------------------------------
# Generic pairwise annealing (used as final polishing)
# ----------------------------------------------------------------------
def _pairwise_annealing(
    a: np.ndarray,
    deadline: float,
    best_seq: np.ndarray,
    best_score: float,
    rng: np.random.Generator,
    max_trials: int | None = None,
) -> tuple[np.ndarray, float]:
    """Randomly transfer mass between two bins; accept improving moves
    or uphill moves with a decreasing temperature."""
    n = a.shape[0]
    start_time = time.time()
    total_time = deadline - start_time
    trials = 0
    while time.time() < deadline:
        if max_trials is not None and trials >= max_trials:
            break
        trials += 1

        src_candidates = np.where(a > 1e-12)[0]
        if src_candidates.size == 0:
            break
        i = rng.choice(src_candidates)

        j = rng.integers(n - 1)
        if j >= i:
            j += 1

        delta = a[i] * rng.random() * 0.4
        if delta < 1e-16:
            continue

        new_a = a.copy()
        new_a[i] -= delta
        new_a[j] += delta

        cur = _objective(new_a)
        elapsed = time.time() - start_time
        temperature = max(0.0, 1.0 - elapsed / total_time) * best_score * 0.01

        if cur < best_score - 1e-14:
            accept = True
        else:
            accept = temperature > 0.0 and rng.random() < math.exp(-(cur - best_score) / temperature)

        if accept:
            a = new_a
            if cur < best_score - 1e-14:
                best_score = cur
                best_seq = a.copy()
                print(f"[{time.time():.1f}] anneal i={i}->j={j} δ={delta:.3e} new best {best_score:.6f}")
    return best_seq, best_score


# ----------------------------------------------------------------------
# Linear up‑sampling while staying on the simplex
# ----------------------------------------------------------------------
def _upsample(a: np.ndarray, new_n: int) -> np.ndarray:
    """Interpolate a probability vector to a larger length."""
    old_n = a.shape[0]
    if new_n == old_n:
        return a.copy()
    x_old = (np.arange(old_n, dtype=np.float64) + 0.5) / old_n
    x_new = (np.arange(new_n, dtype=np.float64) + 0.5) / new_n
    a_new = np.interp(x_new, x_old, a, left=0.0, right=0.0)
    a_new = np.maximum(a_new, 0.0)
    if a_new.sum() == 0.0:
        a_new[:] = 1.0 / new_n
    else:
        a_new /= a_new.sum()
    return a_new


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: int = 1000) -> list[float]:
    """
    Search for a non‑negative step function that minimises ``evaluate_sequence``.
    The routine respects the total 1000 s budget (2 s safety margin) and
    returns a plain ``list[float]`` ready for ``evaluate_sequence``.
    """
    rng = np.random.default_rng(seed)
    start = time.time()
    deadline = start + budget_s - 2.0   # safety margin

    overall_best_seq: np.ndarray | None = None
    overall_best_score = float("inf")
    top_candidates: list[tuple[float, np.ndarray]] = []

    # ------------------------------------------------------------------
    # 1.  Populate a diverse pool of seeds.
    # ------------------------------------------------------------------
    lengths = [2000, 4000, 8000, 12000, 16000]   # moderate‑to‑large sizes
    # p for edge‑weighted / beta families – extended range
    p_grid = np.concatenate(
        [
            np.logspace(-2.0, -0.5, 12, base=10),
            np.logspace(-0.5, 0.0, 12, base=10),
            np.logspace(0.0, 0.7, 10, base=10),
            np.logspace(0.7, 1.2, 6, base=10),   # extra tail
        ]
    )
    lam_grid = np.linspace(0.0, 0.6, 13)        # finer mixing
    q_grid = np.linspace(0.5, 3.0, 9)
    power_grid = np.linspace(0.2, 2.5, 9)
    beta_alpha_grid = np.linspace(0.2, 2.0, 9)   # for asymmetric beta

    def _register(cand: np.ndarray):
        nonlocal overall_best_seq, overall_best_score, top_candidates
        sc = _objective(cand)
        top_candidates.append((sc, cand.copy()))
        if sc < overall_best_score:
            overall_best_score = sc
            overall_best_seq = cand.copy()
            print(f"[{time.time():.1f}] new global best len={len(cand)} score={sc:.6f}")

    for n in lengths:
        if time.time() >= deadline:
            break
        print(f"[{time.time():.1f}] scanning length {n}")

        # deterministic families
        for cand in (_distribution_uniform(n), _distribution_arcsine(n), _distribution_hann(n)):
            _register(cand)

        # edge‑weighted + uniform mix
        for p in p_grid:
            if time.time() >= deadline:
                break
            base = _distribution_edge_weighted(n, p)
            for lam in lam_grid:
                cand = _mix_with_uniform(base, lam)
                _register(cand)

        # symmetric Beta(p,p) + uniform mix
        for p in p_grid:
            if time.time() >= deadline:
                break
            base = _distribution_beta_sym(n, p)
            for lam in lam_grid:
                cand = _mix_with_uniform(base, lam)
                _register(cand)

        # asymmetric Beta(alpha,beta) + uniform mix
        for alpha in beta_alpha_grid:
            for beta in beta_alpha_grid:
                if time.time() >= deadline:
                    break
                base = _distribution_beta_asym(n, alpha, beta)
                for lam in lam_grid:
                    cand = _mix_with_uniform(base, lam)
                    _register(cand)

        # generalized normal shapes
        for q in q_grid:
            if time.time() >= deadline:
                break
            cand = _distribution_generalized_normal(n, q)
            _register(cand)

        # power‑law families
        for pwr in power_grid:
            if time.time() >= deadline:
                break
            cand = _distribution_power_law(n, pwr)
            _register(cand)

        # random Dirichlet seeds
        for _ in range(8):
            if time.time() >= deadline:
                break
            alpha = rng.uniform(0.01, 2.0)
            cand = rng.dirichlet(np.full(n, alpha))
            _register(cand)

        # sparse random seeds
        for _ in range(8):
            if time.time() >= deadline:
                break
            cand = _distribution_random_sparse(n, rng, density=0.008)
            _register(cand)

        # keep only the best few (50)
        top_candidates.sort(key=lambda x: x[0])
        top_candidates = top_candidates[:50]

    # ------------------------------------------------------------------
    # 2.  Include a previously discovered construction if available.
    # ------------------------------------------------------------------
    try:
        prev = np.asarray(height_sequence_1, dtype=np.float64)  # type: ignore[name-defined]
        if prev.ndim == 1:
            prev = prev / prev.sum()
            sc = _objective(prev)
            if sc < overall_best_score:
                overall_best_score = sc
                overall_best_seq = prev.copy()
                print(f"[{time.time():.1f}] previous best score={sc:.6f}")
            top_candidates.append((sc, prev.copy()))
    except Exception:
        pass

    # ------------------------------------------------------------------
    # 3.  Guarantee a non‑empty result.
    # ------------------------------------------------------------------
    if overall_best_seq is None:
        n = 4000
        overall_best_seq = _distribution_uniform(n)
        overall_best_score = _objective(overall_best_seq)
        print(f"[{time.time():.1f}] fallback uniform score={overall_best_score:.6f}")

    # ------------------------------------------------------------------
    # 4.  Exponentiated‑gradient descent on a few top seeds.
    # ------------------------------------------------------------------
    remaining = deadline - time.time()
    if remaining > 30.0 and top_candidates:
        seed_seqs = [seq for _, seq in sorted(top_candidates, key=lambda x: x[0])[:8]]
        per_seed = remaining * 0.35 / max(1, len(seed_seqs))
        print(f"[{time.time():.1f}] EG descent on {len(seed_seqs)} seeds")
        for idx, seed_seq in enumerate(seed_seqs):
            if time.time() >= deadline:
                break
            sub_deadline = time.time() + per_seed
            a_opt, sc_opt = _exp_gradient_descent(
                seed_seq.copy(),
                sub_deadline,
                best_seq=overall_best_seq,
                best_score=overall_best_score,
                eta0=0.9,
                eval_every=80,
            )
            if sc_opt < overall_best_score - 1e-12:
                overall_best_score = sc_opt
                overall_best_seq = a_opt.copy()
                print(f"[{time.time():.1f}] after EG seed {idx} new global best {overall_best_score:.6f}")

    # ------------------------------------------------------------------
    # 5.  Frank‑Wolfe refinement (if time permits)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        fw_budget = (deadline - time.time()) * 0.15
        print(f"[{time.time():.1f}] Frank‑Wolfe refinement (budget {fw_budget:.1f}s)")
        sub_deadline = time.time() + fw_budget
        overall_best_seq, overall_best_score = _frank_wolfe_descent(
            overall_best_seq.copy(),
            sub_deadline,
            best_seq=overall_best_seq,
            best_score=overall_best_score,
            eval_every=100,
        )
        print(f"[{time.time():.1f}] after FW best score {overall_best_score:.6f}")

    # ------------------------------------------------------------------
    # 6.  Targeted pairwise annealing (mass from high‑grad → low‑grad)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        print(f"[{time.time():.1f}] targeted annealing")
        overall_best_seq, overall_best_score = _targeted_pairwise_annealing(
            overall_best_seq.copy(),
            deadline,
            best_seq=overall_best_seq,
            best_score=overall_best_score,
            rng=rng,
        )
        print(f"[{time.time():.1f}] after targeted anneal best score {overall_best_score:.6f}")

    # ------------------------------------------------------------------
    # 7.  Upsample to a larger length and short EG burst.
    # ------------------------------------------------------------------
    if time.time() < deadline:
        cur_n = overall_best_seq.shape[0]
        target_n = cur_n * 2
        if target_n <= 120000:
            print(f"[{time.time():.1f}] upsampling {cur_n} → {target_n}")
            up_seq = _upsample(overall_best_seq, target_n)
            up_score = _objective(up_seq)
            if up_score < overall_best_score:
                overall_best_seq = up_seq
                overall_best_score = up_score
                print(f"[{time.time():.1f}] upsampled new best {overall_best_score:.6f}")

            sub_deadline = time.time() + (deadline - time.time()) * 0.15
            a_opt, sc_opt = _exp_gradient_descent(
                overall_best_seq.copy(),
                sub_deadline,
                best_seq=overall_best_seq,
                best_score=overall_best_score,
                eta0=0.7,
                eval_every=60,
            )
            if sc_opt < overall_best_score - 1e-12:
                overall_best_seq = a_opt.copy()
                overall_best_score = sc_opt
                print(f"[{time.time():.1f}] EG after upsample new best {overall_best_score:.6f}")

    # ------------------------------------------------------------------
    # 8.  Final long‑run generic annealing polish.
    # ------------------------------------------------------------------
    if time.time() < deadline:
        print(f"[{time.time():.1f}] final annealing polish")
        overall_best_seq, overall_best_score = _pairwise_annealing(
            overall_best_seq.copy(),
            deadline,
            best_seq=overall_best_seq,
            best_score=overall_best_score,
            rng=rng,
            max_trials=None,
        )
        print(f"[{time.time():.1f}] after final anneal best score {overall_best_score:.6f}")

    # ------------------------------------------------------------------
    print(
        f"Final best score (objective = 2·n·max(conv)/sum²): "
        f"{overall_best_score:.6f} (length={len(overall_best_seq)})"
    )
    return overall_best_seq.tolist()
