"""
Propose a step‑function (non‑negative sequence) that minimizes
``evaluate_sequence``.  The algorithm consists of three stages:

1️⃣  **Broad analytic scan** – generate many candidate distributions
    (uniform, arcsine, sine, cosine, quadratic, edge‑weighted,
    exponential‑edge, and a flexible Beta family) on several lengths.
    Keep the few best seeds.

2️⃣  **Smooth‑gradient descent with annealing** – starting from each seed we
    minimise the exact objective 2·n·max(conv(a,a)) using a
    log‑sum‑exp (soft‑max) surrogate for the max.  The gradient is
    computed by a single FFT‑based convolution, and the update is an
    exponentiated‑gradient step followed by simplex projection.

3️⃣  **Greedy pairwise mass‑transfer polishing** – random moves of a
    fraction of mass from one index to another are accepted only if the
    true objective improves.

All work respects the supplied time budget (default 1000 s).  The
returned list is a plain Python list; ``evaluate_sequence`` will
re‑normalise it internally.
"""

# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x ≥ 0, Σx = s}."""
    if s < 0:
        raise ValueError("Simplex radius must be non‑negative")
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.where(u - cssv / (np.arange(1, len(v) + 1)) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    return np.maximum(v - theta, 0.0)


# ----------------------------------------------------------------------
def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Full self‑convolution a*a using a real FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1) - 1).bit_length()   # next power of two
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[: 2 * n - 1]


# ----------------------------------------------------------------------
def _fft_correlation(w: np.ndarray, a: np.ndarray) -> np.ndarray:
    """
    Correlation ``corr[i] = Σ_j w[i+j]·a[j]`` for i=0…n‑1 (valid mode).
    Implemented with a single FFT convolution of w with the reversed a.
    """
    n = a.shape[0]
    L = w.shape[0]
    size = 1 << ((L + n - 1) - 1).bit_length()
    W = np.fft.rfft(w, n=size)
    A_rev = np.fft.rfft(a[::-1], n=size)
    full = np.fft.irfft(W * A_rev, n=size)
    start = n - 1
    end = start + n
    return full[start:end]


# ----------------------------------------------------------------------
def _objective(a: np.ndarray) -> float:
    """
    Exact objective 2·n·max(conv(a,a)) for a probability vector (Σa=1).
    The clipping performed in ``evaluate_sequence`` is irrelevant here
    because we keep a non‑negative and normalised.
    """
    n = a.shape[0]
    b = _fft_self_convolve(a)
    return float(2.0 * n * b.max())

# Analytic families ----------------------------------------------------
def _distribution_uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _distribution_arcsine(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    vals /= vals.sum()
    return vals


def _distribution_sine(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64) + 0.5
    w = np.sin(np.pi * i / n)
    w /= w.sum()
    return w


def _distribution_cosine(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64) + 0.5
    w = 1.0 - np.cos(np.pi * i / n)
    w /= w.sum()
    return w


def _distribution_quadratic(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    x = (2.0 * i) / (n - 1) - 1.0
    w = 1.0 - x * x
    w = np.clip(w, 0.0, None)
    w /= w.sum()
    return w


def _distribution_edge_weighted(n: int, p: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64) + 0.5
    j = (n - np.arange(n, dtype=np.float64) - 0.5)
    w = (i * j) ** (-p)
    if w.sum() == 0.0:
        return _distribution_uniform(n)
    w /= w.sum()
    return w


def _distribution_exponential_edge(n: int, c: float) -> np.ndarray:
    """w_i ∝ exp(-c·i) + exp(-c·(n‑1‑i))   (c>0)."""
    i = np.arange(n, dtype=np.float64)
    w = np.exp(-c * i) + np.exp(-c * (n - 1 - i))
    w /= w.sum()
    return w


def _distribution_beta(n: int, alpha: float, beta: float) -> np.ndarray:
    """
    General Beta‑type shape on {0,…,n‑1} (continuous analogue).
    w_i ∝ ( (i+0.5)/n )^{α‑1} · (1‑(i+0.5)/n)^{β‑1}.
    """
    i = np.arange(n, dtype=np.float64) + 0.5
    x = i / n
    # protect against zero powers
    eps = 1e-14
    x = np.clip(x, eps, 1.0 - eps)
    w = (x ** (alpha - 1.0)) * ((1.0 - x) ** (beta - 1.0))
    if w.sum() == 0.0 or np.any(np.isnan(w)):
        # fall back to uniform
        return _distribution_uniform(n)
    w /= w.sum()
    return w


def _mix_with_uniform(base: np.ndarray, lam: float) -> np.ndarray:
    """Convex combination λ·uniform + (1‑λ)·base."""
    n = base.shape[0]
    uniform = np.full(n, 1.0 / n, dtype=np.float64)
    return lam * uniform + (1.0 - lam) * base

# Smooth (soft‑max) gradient descent ------------------------------------
def _smooth_gradient_descent(
    a0: np.ndarray,
    time_limit: float,
    best_seq: np.ndarray,
    best_score: float,
    lam_schedule=None,
    steps_per_lambda: int = 200,
) -> tuple[np.ndarray, float]:
    """
    Minimise the exact L∞‑objective using a soft‑max surrogate.
    ``lam_schedule`` is a list of increasing smoothing parameters.
    """
    if lam_schedule is None:
        lam_schedule = [0.1, 0.5, 1.0, 2.0, 5.0, 10.0, 30.0, 100.0]

    a = a0.copy()
    n = a.shape[0]
    t = 0
    # base learning rate – ensures that the largest exponent change is ≈0.1
    eta_base = 0.1 / (4.0 * n)

    for lam in lam_schedule:
        if time.time() >= time_limit:
            break
        for _ in range(steps_per_lambda):
            if time.time() >= time_limit:
                break

            # 1. current convolution & soft‑max weights
            conv = _fft_self_convolve(a)
            max_val = conv.max()
            # shift by the maximum for stability
            shifted = lam * (conv - max_val)
            w = np.exp(shifted)
            w_sum = w.sum()
            if w_sum == 0.0:
                w = np.full_like(w, 1.0 / w.size)
            else:
                w /= w_sum

            # 2. gradient via correlation
            corr = _fft_correlation(w, a)          # length n
            grad = 4.0 * n * corr                  # ∂/∂a of the smooth surrogate

            # 3. EG update + simplex normalisation
            eta = eta_base / math.sqrt(t + 1)
            a *= np.exp(-eta * grad)
            s = a.sum()
            if s <= 0.0:
                a = np.full(n, 1.0 / n, dtype=np.float64)
            else:
                a /= s

            t += 1

            # occasional exact evaluation
            if t % 20 == 0:
                cur = _objective(a)
                if cur < best_score - 1e-14:
                    best_score = cur
                    best_seq = a.copy()
                    print(
                        f"[{time.time():.1f}] smooth lam={lam:.2g} step={t} new best {best_score:.6f}"
                    )
    return best_seq, best_score


# ----------------------------------------------------------------------
# Greedy pairwise mass transfer ----------------------------------------
def _pairwise_swap(
    a0: np.ndarray,
    time_limit: float,
    best_seq: np.ndarray,
    best_score: float,
    max_trials: int | None = None,
) -> tuple[np.ndarray, float]:
    """
    Randomly move a fraction of mass from one index to another.
    The move is accepted only if it strictly improves the exact objective.
    """
    a = a0.copy()
    n = a.shape[0]
    trials = 0
    while time.time() < time_limit:
        if max_trials is not None and trials >= max_trials:
            break
        trials += 1

        src = np.where(a > 1e-14)[0]
        if src.size == 0:
            break
        i = np.random.choice(src)
        j = np.random.randint(n - 1)
        if j >= i:
            j += 1

        delta = a[i] * np.random.rand() * 0.45   # up to 45 % of source mass
        if delta < 1e-15:
            continue

        new_a = a.copy()
        new_a[i] -= delta
        new_a[j] += delta

        cur = _objective(new_a)
        if cur < best_score - 1e-14:
            a = new_a
            best_score = cur
            best_seq = a.copy()
            print(
                f"[{time.time():.1f}] swap i={i}→j={j} δ={delta:.6g} new best {best_score:.6f}"
            )
    return best_seq, best_score


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: int = 1000) -> list[float]:
    """
    Return a non‑negative step‑function (plain Python list) that minimises
    ``evaluate_sequence`` within the provided time budget.
    """
    rng = np.random.default_rng(seed)
    start = time.time()
    # keep a small safety margin for finalisation
    deadline = start + budget_s - 2.0

    overall_best_seq = None
    overall_best_score = float("inf")
    top_candidates: list[tuple[float, np.ndarray]] = []   # (score, seq)

    # ------------------------------------------------------------------
    # 1️⃣  Analytic families scan
    # ------------------------------------------------------------------
    lengths = [1000, 2000, 4000, 8000, 16000]

    # grids for parameters
    p_grid = np.concatenate(
        [
            np.logspace(-3, -0.5, 12),       # very edge‑heavy
            np.logspace(-0.5, 0.0, 12),      # moderate
            np.linspace(0.1, 3.0, 10),       # broader
        ]
    )
    lam_grid = np.linspace(0.0, 1.0, 9)      # mix with uniform
    c_grid = np.logspace(-3, 0, 12)        # exponential‑edge decay
    # Beta exponents (α = β) – symmetric shapes
    alpha_grid = np.concatenate(
        [
            np.logspace(-2, -0.2, 12),
            np.linspace(0.3, 2.5, 12),
            np.logspace(0.3, 1.0, 8),
        ]
    )
    # asymmetric Beta – sparse selection to keep budget modest
    alpha_grid_asym = np.array([0.2, 0.5, 1.0, 2.0])
    beta_grid_asym = np.array([0.2, 0.5, 1.0, 2.0])

    for n in lengths:
        if time.time() >= deadline:
            break
        print(f"[{time.time():.1f}] scanning length {n}")

        # ----- deterministic families -------------------------------------------------
        for base_fn in (
            _distribution_uniform,
            _distribution_arcsine,
            _distribution_sine,
            _distribution_cosine,
            _distribution_quadratic,
        ):
            cand = base_fn(n)
            sc = _objective(cand)
            top_candidates.append((sc, cand.copy()))
            if sc < overall_best_score:
                overall_best_score, overall_best_seq = sc, cand.copy()
                print(f"[{time.time():.1f}] new global best len={n} score={sc:.6f}")

        # ----- Beta families -----------------------------------------------------------
        for α in alpha_grid:
            cand = _distribution_beta(n, α, α)
            sc = _objective(cand)
            top_candidates.append((sc, cand.copy()))
            if sc < overall_best_score:
                overall_best_score, overall_best_seq = sc, cand.copy()
                print(f"[{time.time():.1f}] beta α={α:.3g} len={n} score={sc:.6f}")

        # a few skewed Beta candidates
        for α in alpha_grid_asym:
            for β in beta_grid_asym:
                if α == β:
                    continue
                cand = _distribution_beta(n, α, β)
                sc = _objective(cand)
                top_candidates.append((sc, cand.copy()))
                if sc < overall_best_score:
                    overall_best_score, overall_best_seq = sc, cand.copy()
                    print(
                        f"[{time.time():.1f}] beta α={α:.2g},β={β:.2g} len={n} score={sc:.6f}"
                    )

        # ----- edge‑weighted + uniform mixes -------------------------------------------
        for p in p_grid:
            if time.time() >= deadline:
                break
            edge = _distribution_edge_weighted(n, float(p))
            for lam in lam_grid:
                if time.time() >= deadline:
                    break
                cand = _mix_with_uniform(edge, lam)
                sc = _objective(cand)
                top_candidates.append((sc, cand.copy()))
                if sc < overall_best_score:
                    overall_best_score, overall_best_seq = sc, cand.copy()
                    print(
                        f"[{time.time():.1f}] edge p={p:.4g} lam={lam:.2f} len={n} score={sc:.6f}"
                    )

        # ----- exponential‑edge + uniform mixes ----------------------------------------
        for c in c_grid:
            if time.time() >= deadline:
                break
            exp_edge = _distribution_exponential_edge(n, float(c))
            for lam in lam_grid:
                if time.time() >= deadline:
                    break
                cand = _mix_with_uniform(exp_edge, lam)
                sc = _objective(cand)
                top_candidates.append((sc, cand.copy()))
                if sc < overall_best_score:
                    overall_best_score, overall_best_seq = sc, cand.copy()
                    print(
                        f"[{time.time():.1f}] exp‑edge c={c:.4g} lam={lam:.2f} len={n} score={sc:.6f}"
                    )

        # ----- random Dirichlet seeds --------------------------------------------------
        for _ in range(4):
            if time.time() >= deadline:
                break
            alpha = rng.uniform(0.01, 2.0)
            a = rng.dirichlet(np.full(n, alpha))
            sc = _objective(a)
            top_candidates.append((sc, a.copy()))
            if sc < overall_best_score:
                overall_best_score, overall_best_seq = sc, a.copy()
                print(f"[{time.time():.1f}] Dirichlet len={n} score={sc:.6f}")

        # ----- sparse random seeds -----------------------------------------------------
        for _ in range(4):
            if time.time() >= deadline:
                break
            a = _distribution_random_sparse(n, rng, density=0.005)
            sc = _objective(a)
            top_candidates.append((sc, a.copy()))
            if sc < overall_best_score:
                overall_best_score, overall_best_seq = sc, a.copy()
                print(f"[{time.time():.1f}] sparse len={n} score={sc:.6f}")

        # keep only a limited set of the best candidates in memory
        top_candidates.sort(key=lambda x: x[0])
        top_candidates = top_candidates[:40]

    # ------------------------------------------------------------------
    # 2️⃣  Include previous construction if available
    # ------------------------------------------------------------------
    try:
        prev = np.asarray(height_sequence_1, dtype=np.float64)  # type: ignore[name-defined]
        if prev.ndim == 1 and prev.size > 0:
            prev = np.maximum(prev, 0.0)
            if prev.sum() == 0.0:
                prev = _distribution_uniform(prev.size)
            else:
                prev = prev / prev.sum()
            sc = _objective(prev)
            if sc < overall_best_score:
                overall_best_score, overall_best_seq = sc, prev.copy()
                print(f"[{time.time():.1f}] previous construction score={sc:.6f}")
            top_candidates.append((sc, prev.copy()))
    except Exception:
        pass

    # ------------------------------------------------------------------
    # 3️⃣  Fallback – should never happen
    # ------------------------------------------------------------------
    if overall_best_seq is None:
        n = 2000
        overall_best_seq = _distribution_uniform(n)
        overall_best_score = _objective(overall_best_seq)
        print(
            f"[{time.time():.1f}] fallback uniform length={n} score={overall_best_score:.6f}"
        )

    # ------------------------------------------------------------------
    # 4️⃣  Smooth‑gradient refinement on the most promising seeds
    # ------------------------------------------------------------------
    remaining = deadline - time.time()
    if remaining > 30.0 and top_candidates:
        # keep the six best seeds for deeper optimisation
        seeds = [seq for _, seq in sorted(top_candidates, key=lambda x: x[0])[:6]]
        per_seed = remaining / max(1, len(seeds))
        print(f"[{time.time():.1f}] start smooth descent on {len(seeds)} seeds")
        for idx, seed_seq in enumerate(seeds):
            if time.time() >= deadline:
                break
            sg_deadline = time.time() + per_seed
            a_opt, sc_opt = _smooth_gradient_descent(
                seed_seq,
                sg_deadline,
                best_seq=overall_best_seq,
                best_score=overall_best_score,
                lam_schedule=[0.1, 0.5, 1.0, 2.0, 5.0, 10.0, 30.0, 100.0],
                steps_per_lambda=250,
            )
            if sc_opt < overall_best_score - 1e-12:
                overall_best_score, overall_best_seq = sc_opt, a_opt.copy()
                print(
                    f"[{time.time():.1f}] after smooth seed {idx} new global best {overall_best_score:.6f}"
                )

    # ------------------------------------------------------------------
    # 5️⃣  Final greedy pairwise polishing
    # ------------------------------------------------------------------
    if time.time() < deadline:
        print(f"[{time.time():.1f}] final greedy pairwise polishing")
        a_opt, sc_opt = _pairwise_swap(
            overall_best_seq.copy(),
            deadline,
            best_seq=overall_best_seq,
            best_score=overall_best_score,
            max_trials=200_000,
        )
        if sc_opt < overall_best_score - 1e-12:
            overall_best_score, overall_best_seq = sc_opt, a_opt.copy()
            print(f"[{time.time():.1f}] after polishing new best {overall_best_score:.6f}")

    # ------------------------------------------------------------------
    print(
        f"Final best objective (2·n·max(conv)): {overall_best_score:.6f} (length={len(overall_best_seq)})"
    )
    # ``evaluate_sequence`` will internally re‑scale, so we simply return the list.
    return overall_best_seq.tolist()


# ----------------------------------------------------------------------
def _distribution_random_sparse(
    n: int, rng: np.random.Generator, density: float = 0.01
) -> np.ndarray:
    """Sparse random vector with a prescribed density of non‑zeros."""
    k = max(2, int(n * density))
    idx = rng.choice(n, size=k, replace=False)
    vals = np.zeros(n, dtype=np.float64)
    vals[idx] = rng.random(k)
    s = vals.sum()
    if s == 0.0:
        vals[0] = 1.0
        s = 1.0
    vals /= s
    return vals
