    (uniform, arcsine, edge‑weighted, edge‑inverse‑power, sine‑weighted,
    symmetric beta, exponential, Dirichlet, sparse).  All candidates are
    normalised to sum = 1 and the ~15 best (lowest exact objective) are kept.

2️⃣  **Smoothed‑max descent** – for the current best seed we run a
    mirror‑descent (exponentiated‑gradient) on a smooth surrogate of the
    max‑convolution:
        * the “max’’ is approximated by the L^q‑norm  (∑b_i^q)^{1/q},
          i.e.  w_i ∝ b_i^{q‑1}.  The exponent q is increased from a small
          value (≈2) up to a larger one (≈60) so the optimisation first sees
          a smooth landscape and later focuses on the true maximum.
        * the gradient of the surrogate is obtained analytically:
          ∂b_k/∂a_j = 2·a_{k‑j} (when the index is valid).  This yields a
          convolution which is evaluated by a real FFT (O(n log n)).
        * after the exponential update the iterate is projected onto the
          probability simplex (preserves non‑negativity and sum = 1).

    The descent runs for the bulk of the time budget, periodically evaluates
    the exact objective and injects a small random jitter when no improvement
    is seen for a while.

3️⃣  **Local polishing** – a greedy pairwise mass‑transfer step followed by a
    cheap simulated‑annealing style search cleans up any remaining irregularities.

The routine respects the 1000 s budget, prints timing / progress information,
and returns a plain Python list of heights that can be fed directly to
``evaluate_sequence``.
"""

import time
import math
import numpy as np
from numpy.fft import rfft, irfft

# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x ≥ 0, sum(x)=s}."""
    if s < 0:
        raise ValueError("simplex radius must be non‑negative")
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho_arr = np.where(u - cssv / (np.arange(1, len(v) + 1)) > 0)[0]
    if rho_arr.size == 0:
        theta = 0.0
    else:
        rho = rho_arr[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _fft_convolve(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full convolution a*b using a real FFT."""
    n = a.shape[0]
    m = b.shape[0]
    need = n + m - 1
    size = 1 << ((need - 1).bit_length())
    A = rfft(a, n=size)
    B = rfft(b, n=size)
    conv = irfft(A * B, n=size)
    return conv[:need]


def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Full self‑convolution a*a (length 2n‑1)."""
    n = a.shape[0]
    need = 2 * n - 1
    size = 1 << ((need - 1).bit_length())
    A = rfft(a, n=size)
    conv = irfft(A * A, n=size)
    return conv[:need]


# ----------------------------------------------------------------------
def _objective(a: np.ndarray) -> float:
    """Exact objective 2·n·max(conv(a,a)) assuming sum(a)=1."""
    n = a.shape[0]
    b = _fft_self_convolve(a)
    return 2.0 * n * b.max()


# ----------------------------------------------------------------------
# Analytic families ----------------------------------------------------
def _distribution_uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _distribution_arcsine(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    vals /= vals.sum()
    return vals


def _distribution_beta_sym(n: int, p: float) -> np.ndarray:
    """Symmetric beta (Beta(p,p))."""
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    w = np.exp((p - 1) * np.log(x) + (p - 1) * np.log(1.0 - x))
    s = w.sum()
    if s == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= s
    return w


def _distribution_beta_general(n: int, alpha: float, beta: float) -> np.ndarray:
    """Asymmetric beta (Beta(alpha,beta))."""
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    eps = np.finfo(float).tiny
    x = np.clip(x, eps, 1 - eps)
    w = np.exp((alpha - 1) * np.log(x) + (beta - 1) * np.log(1.0 - x))
    s = w.sum()
    if s == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= s
    return w


def _distribution_edge_weighted(n: int, p: float) -> np.ndarray:
    """U‑shaped family w_i ∝ ((i+0.5)*(n‑i‑0.5))^{−p}."""
    i = np.arange(n, dtype=np.float64) + 0.5
    j = (n - np.arange(n, dtype=np.float64) - 0.5)
    w = (i * j) ** (-p)
    s = w.sum()
    if s == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= s
    return w


def _distribution_edge_inversepow(n: int, p: float) -> np.ndarray:
    """Second edge family w_i ∝ (i+0.5)^{−p} + (n−i−0.5)^{−p}."""
    i = np.arange(n, dtype=np.float64) + 0.5
    j = (n - np.arange(n, dtype=np.float64) - 0.5)
    w = i ** (-p) + j ** (-p)
    s = w.sum()
    if s == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= s
    return w


def _distribution_sine_weighted(n: int, p: float) -> np.ndarray:
    """Weight ∝ sin(π (i+½)/n)^p, vanishing at the ends."""
    theta = np.pi * (np.arange(n, dtype=np.float64) + 0.5) / n
    w = np.sin(theta) ** p
    s = w.sum()
    if s == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= s
    return w


def _distribution_tent(n: int) -> np.ndarray:
    """Triangular (tent) shape."""
    i = np.arange(n, dtype=np.float64)
    half = (n - 1) / 2.0
    w = np.where(i <= half, i + 1, n - i)
    w /= w.sum()
    return w


def _distribution_quadratic(n: int) -> np.ndarray:
    """Quadratic bell shape, symmetric."""
    i = np.arange(n, dtype=np.float64)
    half = (n - 1) / 2.0
    d = np.abs(i - half) + 1.0
    w = 1.0 / (d ** 2)
    w /= w.sum()
    return w


def _distribution_exponential_decay(n: int, rate: float) -> np.ndarray:
    """Exponential decay from the centre."""
    i = np.arange(n, dtype=np.float64)
    half = (n - 1) / 2.0
    w = np.exp(-rate * np.abs(i - half))
    w /= w.sum()
    return w


def _distribution_random_sparse(
    n: int, rng: np.random.Generator, density: float = 0.01
) -> np.ndarray:
    """Sparse random vector with a given density."""
    k = max(2, int(n * density))
    idx = rng.choice(n, size=k, replace=False)
    vals = np.zeros(n, dtype=np.float64)
    vals[idx] = rng.random(k)
    if vals.sum() == 0.0:
        vals[0] = 1.0
    vals /= vals.sum()
    return vals


def _distribution_dirichlet(n: int, rng: np.random.Generator) -> np.ndarray:
    """Random Dirichlet sample with a random concentration."""
    alpha = rng.uniform(0.01, 2.0)
    a = rng.dirichlet(np.full(n, alpha))
    return a.astype(np.float64)


def _mix_with_uniform(base: np.ndarray, lam: float) -> np.ndarray:
    """Convex combination λ·U + (1‑λ)·base, where U is uniform."""
    n = base.shape[0]
    uniform = np.full(n, 1.0 / n, dtype=np.float64)
    return lam * uniform + (1.0 - lam) * base


# ----------------------------------------------------------------------
def _gradient_descent_power(
    a: np.ndarray,
    deadline: float,
    best_seq: np.ndarray,
    best_score: float,
    q_start: float = 2.0,
    q_end: float = 60.0,
    eta0: float = 0.8,
    eval_interval: int = 200,
    patience: int = 3000,
    jitter_scale: float = 0.006,
) -> tuple[np.ndarray, float]:
    """
    Mirror‑descent on the L^q surrogate of max(conv(a,a)).
    ``a`` must sum to one.  Returns the best sequence encountered and its exact objective.
    """
    n = a.shape[0]
    rng = np.random.default_rng()
    t = 0
    no_improve = 0
    start = time.time()
    max_iters = 500_000

    while time.time() < deadline and t < max_iters:
        t += 1

        # self‑convolution
        b = _fft_self_convolve(a)

        # current exponent q (linear schedule)
        q = q_start + (q_end - q_start) * min(1.0, t / max_iters)

        # smooth weight proportional to b^{q‑1}
        eps = 1e-20
        w = np.maximum(b, eps) ** (q - 1.0)
        w_sum = w.sum()
        if w_sum == 0.0:  # should not happen, but guard
            continue
        w /= w_sum

        # gradient: g_j = 2 * sum_i w_i * a_{i‑j}
        w_rev = w[::-1]
        conv_rev = _fft_convolve(a, w_rev)                 # length 3n‑2
        offset = len(w_rev) - 1                           # =2n‑2
        grad = 2.0 * conv_rev[offset - np.arange(n)]

        # Clip extreme values to avoid overflow in the exponential update
        grad = np.clip(grad, -100.0, 100.0)

        # learning rate decays like 1/√t
        eta = eta0 / math.sqrt(t)

        # exponentiated‑gradient (mirror descent on simplex)
        a = a * np.exp(-eta * grad)
        a = _project_onto_simplex(a, s=1.0)

        # periodic exact evaluation
        if t % eval_interval == 0:
            cur = _objective(a)
            if cur < best_score - 1e-14:
                best_score = cur
                best_seq = a.copy()
                no_improve = 0
                print(f"[{time.time():.1f}] it={t:,} q={q:.2f} new global best {best_score:.6f}")
            else:
                no_improve += 1

        # random jitter when stagnating
        if no_improve >= patience:
            jitter = rng.exponential(scale=jitter_scale, size=n)
            a = a + jitter
            a = np.maximum(a, 0.0)
            a = _project_onto_simplex(a, s=1.0)
            no_improve = 0
            print(f"[{time.time():.1f}] jitter injected at it={t:,}")

    return best_seq, best_score


# ----------------------------------------------------------------------
def _pairwise_swap_local(
    a: np.ndarray,
    deadline: float,
    best_seq: np.ndarray,
    best_score: float,
    max_trials: int | None = None,
) -> tuple[np.ndarray, float]:
    """Greedy transfer of a random amount of mass between two positions."""
    n = a.shape[0]
    trials = 0
    rng = np.random.default_rng()
    while time.time() < deadline:
        if max_trials is not None and trials >= max_trials:
            break
        trials += 1

        src = np.where(a > 1e-12)[0]
        if src.size == 0:
            break
        i = rng.choice(src)
        j = rng.integers(n - 1)
        if j >= i:
            j += 1

        delta = a[i] * rng.random() * 0.35
        if delta < 1e-14:
            continue

        new_a = a.copy()
        new_a[i] -= delta
        new_a[j] += delta
        cur = _objective(new_a)

        if cur < best_score - 1e-14:
            a = new_a
            best_score = cur
            best_seq = a.copy()
            print(f"[{time.time():.1f}] swap {i}->{j} Δ={delta:.3e} new best {best_score:.6f}")

    return best_seq, best_score


# ----------------------------------------------------------------------
def _random_local_search(
    a: np.ndarray,
    deadline: float,
    best_seq: np.ndarray,
    best_score: float,
    init_temp: float = 0.008,
    decay: float = 0.9997,
) -> tuple[np.ndarray, float]:
    """Very cheap simulated‑annealing style search."""
    n = a.shape[0]
    rng = np.random.default_rng()
    T = init_temp
    while time.time() < deadline:
        i = rng.integers(n)
        j = rng.integers(n - 1)
        if j >= i:
            j += 1
        delta = a[i] * rng.random() * 0.3
        if delta < 1e-14:
            continue
        new_a = a.copy()
        new_a[i] -= delta
        new_a[j] += delta
        cur = _objective(new_a)
        if cur < best_score - 1e-14:
            a = new_a
            best_score = cur
            best_seq = a.copy()
            print(f"[{time.time():.1f}] SA improve {i}->{j} Δ={delta:.3e} best {best_score:.6f}")
        else:
            if rng.random() < math.exp(-(cur - best_score) / max(T, 1e-12)):
                a = new_a
        T *= decay
    return best_seq, best_score


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: int = 1000) -> list[float]:
    """
    Search for a non‑negative step‑function that minimises
    ``evaluate_sequence``.  The routine respects the given time budget
    (default 1000 s) and returns a plain Python list of heights.
    """
    rng = np.random.default_rng(seed)
    start = time.time()
    deadline = start + budget_s - 2.0          # safety margin

    overall_best_seq: np.ndarray | None = None
    overall_best_score = float("inf")
    top_candidates: list[tuple[float, np.ndarray]] = []

    # ------------------------------------------------------------------
    # 1. Build a diversified pool of seeds
    lengths = [2000, 3000, 4000, 6000, 8000, 12000, 16000]
    p_grid = np.concatenate(
        [
            np.logspace(-3, -0.5, 8, base=10),    # 0.001 … 0.316
            np.logspace(-0.5, 0.0, 8, base=10),   # 0.316 … 1
            np.logspace(0.0, 0.5, 6, base=10),    # 1 … 3.16
        ]
    )
    lam_grid = np.linspace(0.0, 0.6, 7)
    rate_grid = np.linspace(0.001, 0.02, 6)

    for n in lengths:
        if time.time() >= deadline:
            break
        print(f"[{time.time():.1f}] scanning length {n}")

        # deterministic families (no mixing)
        for cand in (
            _distribution_uniform(n),
            _distribution_arcsine(n),
            _distribution_tent(n),
            _distribution_quadratic(n),
        ):
            sc = _objective(cand)
            top_candidates.append((sc, cand.copy()))
            if sc < overall_best_score:
                overall_best_score, overall_best_seq = sc, cand.copy()
                print(f"[{time.time():.1f}] new global best len={n} score={sc:.6f}")

        # parametrised families + mixing with uniform
        for p in p_grid:
            if time.time() >= deadline:
                break
            edge = _distribution_edge_weighted(n, p)
            edge2 = _distribution_edge_inversepow(n, p)
            sine = _distribution_sine_weighted(n, p)
            beta = _distribution_beta_sym(n, p)

            for base in (edge, edge2, sine, beta):
                for lam in lam_grid:
                    if time.time() >= deadline:
                        break
                    cand = _mix_with_uniform(base, lam)
                    sc = _objective(cand)
                    top_candidates.append((sc, cand.copy()))
                    if sc < overall_best_score:
                        overall_best_score, overall_best_seq = sc, cand.copy()
                        print(
                            f"[{time.time():.1f}] new global best len={n} p={p:.3g} lam={lam:.2f} score={sc:.6f}"
                        )

        # exponential‑decay family
        for rate in rate_grid:
            if time.time() >= deadline:
                break
            cand = _distribution_exponential_decay(n, rate)
            sc = _objective(cand)
            top_candidates.append((sc, cand.copy()))
            if sc < overall_best_score:
                overall_best_score, overall_best_seq = sc, cand.copy()
                print(
                    f"[{time.time():.1f}] new global best exp‑decay n={n} rate={rate:.4f} score={sc:.6f}"
                )

        # random Dirichlet seeds (including very low concentration)
        for _ in range(4):
            if time.time() >= deadline:
                break
            a = _distribution_dirichlet(n, rng)
            sc = _objective(a)
            top_candidates.append((sc, a.copy()))
            if sc < overall_best_score:
                overall_best_score, overall_best_seq = sc, a.copy()
                print(
                    f"[{time.time():.1f}] new global best Dirichlet len={n} score={sc:.6f}"
                )

        # sparse random seeds
        for _ in range(4):
            if time.time() >= deadline:
                break
            a = _distribution_random_sparse(n, rng, density=0.006)
            sc = _objective(a)
            top_candidates.append((sc, a.copy()))
            if sc < overall_best_score:
                overall_best_score, overall_best_seq = sc, a.copy()
                print(
                    f"[{time.time():.1f}] new global best sparse len={n} score={sc:.6f}"
                )

        # asymmetric beta samples
        for _ in range(3):
            if time.time() >= deadline:
                break
            alpha = 10 ** rng.uniform(-2, 1)   # 0.01 … 10
            beta = 10 ** rng.uniform(-2, 1)
            a = _distribution_beta_general(n, alpha, beta)
            sc = _objective(a)
            top_candidates.append((sc, a.copy()))
            if sc < overall_best_score:
                overall_best_score, overall_best_seq = sc, a.copy()
                print(
                    f"[{time.time():.1f}] new global best beta(α={alpha:.3g},β={beta:.3g}) len={n} score={sc:.6f}"
                )

        # keep only the strongest seeds overall (size limited)
        top_candidates.sort(key=lambda x: x[0])
        top_candidates = top_candidates[:30]

    # ------------------------------------------------------------------
    # 2. Incorporate a previously discovered construction if present
    try:
        prev = np.asarray(height_sequence_1, dtype=np.float64)  # type: ignore[name-defined]
        if prev.ndim == 1 and prev.size > 0:
            prev = np.maximum(prev, 0.0)
            if prev.sum() == 0.0:
                prev = _distribution_uniform(len(prev))
            else:
                prev = prev / prev.sum()
            sc = _objective(prev)
            if sc < overall_best_score:
                overall_best_score, overall_best_seq = sc, prev.copy()
                print(f"[{time.time():.1f}] previous best score={sc:.6f}")
            top_candidates.append((sc, prev.copy()))
    except Exception:
        pass

    # ------------------------------------------------------------------
    # Guard against empty seed pool
    if overall_best_seq is None:
        n = 3000
        overall_best_seq = _distribution_uniform(n)
        overall_best_score = _objective(overall_best_seq)
        print(f"[{time.time():.1f}] fallback uniform score={overall_best_score:.6f}")

    # Keep the top few distinct seeds for extra exploration
    top_candidates.sort(key=lambda x: x[0])
    top_seeds = [seq for _, seq in top_candidates[:15]]

    # ------------------------------------------------------------------
    # 3. Long smoothed‑max descent (majority of the remaining budget)
    remaining = deadline - time.time()
    if remaining > 30.0:
        sub_deadline = time.time() + remaining - 30.0
        print(f"[{time.time():.1f}] start smoothed‑max descent – about {remaining-30:.1f}s left")
        overall_best_seq, overall_best_score = _gradient_descent_power(
            overall_best_seq.copy(),
            sub_deadline,
            best_seq=overall_best_seq,
            best_score=overall_best_score,
            q_start=2.0,
            q_end=60.0,
            eta0=0.8,
            eval_interval=200,
            patience=3000,
        )
        print(f"[{time.time():.1f}] after descent best score {overall_best_score:.6f}")

    # ------------------------------------------------------------------
    # 4. Brief exploration from a few other seeds (short runs)
    for seed_seq in top_seeds[1:5]:
        if time.time() >= deadline - 20.0:
            break
        short_deadline = time.time() + min(15.0, (deadline - time.time()) / 2)
        seq, sc = _gradient_descent_power(
            seed_seq.copy(),
            short_deadline,
            best_seq=overall_best_seq,
            best_score=overall_best_score,
            q_start=2.0,
            q_end=40.0,
            eta0=0.6,
            eval_interval=150,
            patience=2000,
        )
        if sc < overall_best_score:
            overall_best_seq, overall_best_score = seq, sc
            print(f"[{time.time():.1f}] improved from alternate seed – new best {sc:.6f}")

    # ------------------------------------------------------------------
    # 5. Greedy pairwise refinement
    if time.time() < deadline:
        print(f"[{time.time():.1f}] greedy pairwise refinement")
        overall_best_seq, overall_best_score = _pairwise_swap_local(
            overall_best_seq.copy(),
            deadline,
            best_seq=overall_best_seq,
            best_score=overall_best_score,
            max_trials=300_000,
        )

    # ------------------------------------------------------------------
    # 6. Final simulated‑annealing polishing
    if time.time() < deadline:
        print(f"[{time.time():.1f}] final simulated‑annealing search")
        overall_best_seq, overall_best_score = _random_local_search(
            overall_best_seq.copy(),
            deadline,
            best_seq=overall_best_seq,
            best_score=overall_best_score,
            init_temp=0.01,
            decay=0.9995,
        )

    # ------------------------------------------------------------------
    print(f"Final best score (exact) ≈ {overall_best_score:.6f} (length={len(overall_best_seq)})")
    return overall_best_seq.tolist()
