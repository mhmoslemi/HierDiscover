"""
Propose a non‑negative step function that minimises ``evaluate_sequence``.

Algorithm
---------

1. **Initial diverse seeds** – a handful of analytic shapes (uniform,
   arcsine, symmetric beta, edge‑weighted, sine‑weighted) and a random
   Dirichlet sample are generated for a moderate length (≈ 2500).  
   Each seed is refined by a short entropic‑mirror‑descent run.
   The best seed becomes the starting point.

2. **Multi‑scale refinement** – the current best sequence is repeatedly
   *upsampled* (linear interpolation) by a factor two, symmetrised,
   and further refined by mirror‑descent.  This is repeated while time
   permits, growing the length up to a user‑defined maximum (≈ 15000).

3. **Deterministic pairwise cleanup** – a greedy transfer of mass from
   one coordinate to another is performed for the remaining time.

4. **Cheap simulated‑annealing clean‑up** – final stochastic moves
   (mass transfers) are accepted with a Metropolis rule to escape tiny
   local plateaus.

All updates keep the vector on the probability simplex and enforce
symmetry around the centre (a_i = a_{n‑1‑i}), which is known to be
optimal for the convolution objective.  The whole search respects the
supplied time budget (default 1000 s).

The objective used internally is
    2·n·max(conv(a,a))   (a is normalised to sum = 1)
which is exactly the value returned by ``evaluate_sequence`` after its
clipping.
"""

import time
import math
import numpy as np
from numpy.fft import rfft, irfft

# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x≥0, sum(x)=s}."""
    if s < 0:
        raise ValueError("simplex radius must be non‑negative")
    # Fast path
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho_arr = np.where(u - cssv / (np.arange(1, len(v) + 1)) > 0)[0]
    if rho_arr.size == 0:
        theta = 0.0
    else:
        rho = rho_arr[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a using a real FFT."""
    n = a.shape[0]
    need = 2 * n - 1
    size = 1 << ((need - 1).bit_length())  # next power of two
    A = rfft(a, n=size)
    conv = irfft(A * A, n=size)
    return conv[:need]


# ----------------------------------------------------------------------
def _objective(a: np.ndarray) -> float:
    """Exact objective 2·n·max(conv(a,a)) for a with sum(a)=1."""
    n = a.shape[0]
    b = _fft_self_convolve(a)
    return 2.0 * n * b.max()

# Analytic families ----------------------------------------------------
def _distribution_uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _distribution_arcsine(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    vals /= vals.sum()
    return vals


def _distribution_beta_sym(n: int, p: float) -> np.ndarray:
    """Symmetric Beta(p,p) on the regular grid."""
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    w = np.exp((p - 1) * np.log(x) + (p - 1) * np.log(1.0 - x))
    if w.sum() == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _distribution_edge_weighted(n: int, p: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64) + 0.5
    j = (n - np.arange(n, dtype=np.float64) - 0.5)
    w = (i * j) ** (-p)
    if w.sum() == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _distribution_sine_weighted(n: int, p: float) -> np.ndarray:
    theta = np.pi * (np.arange(n, dtype=np.float64) + 0.5) / n
    w = np.sin(theta) ** p
    if w.sum() == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _distribution_dirichlet(n: int, rng: np.random.Generator) -> np.ndarray:
    alpha = rng.uniform(0.01, 2.0)
    a = rng.dirichlet(np.full(n, alpha))
    return a.astype(np.float64)


# ----------------------------------------------------------------------
def _symmetrize(a: np.ndarray) -> np.ndarray:
    """Force a_i = a_{n-1-i}."""
    return (a + a[::-1]) * 0.5


# ----------------------------------------------------------------------
def _mirror_descent(
    a0: np.ndarray,
    deadline: float,
    best_seq: np.ndarray,
    best_score: float,
    lr0: float = 0.8,
    active_frac: float = 0.025,
    max_active: int = 40,
    beta: float = 0.9,
    eval_interval: int = 100,
    patience: int = 1500,
) -> tuple[np.ndarray, float]:
    """
    Entropic (multiplicative) mirror descent on the exact non‑smooth objective.
    ``a0`` is assumed to be on the simplex.
    """
    a = a0.copy()
    n = a.shape[0]
    ind_grid = np.arange(n, dtype=np.int32)

    t = 0
    no_improve = 0
    momentum = np.zeros_like(a)
    rng = np.random.default_rng()

    while time.time() < deadline and t < 600_000:
        t += 1

        conv = _fft_self_convolve(a)
        max_c = conv.max()
        thresh = max_c * (1.0 - active_frac)

        act = np.where(conv >= thresh)[0]
        if act.size > max_active:
            # keep only the largest contributions
            top = np.argpartition(conv[act], -max_active)[-max_active:]
            act = act[top]

        # gradient of max(conv) w.r.t. a (subgradient)
        grad = np.zeros_like(a)
        for k in act:
            shift = k - ind_grid
            mask = (shift >= 0) & (shift < n)
            grad[mask] += 2.0 * a[shift[mask]]

        # exponential‑gradient update (entropy mirror)
        momentum = beta * momentum + (1.0 - beta) * grad
        eta = lr0 / math.sqrt(t)
        a = a * np.exp(-eta * momentum)

        # project back onto simplex and enforce symmetry
        a = _project_onto_simplex(a, 1.0)
        a = _symmetrize(a)

        # occasional evaluation
        if t % eval_interval == 0:
            cur = _objective(a)
            if cur + 1e-14 < best_score:
                best_score, best_seq = cur, a.copy()
                no_improve = 0
                print(f"[{time.time():.1f}] it={t:,} new global best {best_score:.6f}")
            else:
                no_improve += 1

        # stagnation handling
        if no_improve >= patience:
            jitter = rng.exponential(scale=0.004, size=n)
            a = a + jitter
            a = np.maximum(a, 0.0)
            a = _project_onto_simplex(a, 1.0)
            a = _symmetrize(a)
            no_improve = 0

    return best_seq, best_score


# ----------------------------------------------------------------------
def _pairwise_swap_local(
    a0: np.ndarray,
    deadline: float,
    best_seq: np.ndarray,
    best_score: float,
    max_trials: int | None = None,
) -> tuple[np.ndarray, float]:
    """
    Deterministic greedy mass‑transfer between two coordinates.
    """
    a = a0.copy()
    n = a.shape[0]
    trials = 0
    rng = np.random.default_rng()
    while time.time() < deadline:
        if max_trials is not None and trials >= max_trials:
            break
        trials += 1

        src = np.where(a > 1e-12)[0]
        if src.size == 0:
            break
        i = rng.choice(src)
        j = rng.integers(n - 1)
        if j >= i:
            j += 1

        delta = a[i] * rng.random() * 0.33
        if delta < 1e-14:
            continue
        new_a = a.copy()
        new_a[i] -= delta
        new_a[j] += delta
        new_a = _project_onto_simplex(new_a, 1.0)

        cur = _objective(new_a)
        if cur + 1e-14 < best_score:
            a = new_a
            best_score = cur
            best_seq = a.copy()
            print(f"[{time.time():.1f}] swap {i}->{j} δ={delta:.3e} new best {best_score:.6f}")

    return best_seq, best_score


# ----------------------------------------------------------------------
def _random_local_search(
    a0: np.ndarray,
    deadline: float,
    best_seq: np.ndarray,
    best_score: float,
    init_temp: float = 0.01,
    decay: float = 0.9996,
) -> tuple[np.ndarray, float]:
    """
    Cheap simulated‑annealing style local search (mass transfers).
    """
    a = a0.copy()
    n = a.shape[0]
    rng = np.random.default_rng()
    T = init_temp
    while time.time() < deadline:
        i = rng.integers(n)
        j = rng.integers(n - 1)
        if j >= i:
            j += 1
        delta = a[i] * rng.random() * 0.28
        if delta < 1e-14:
            continue
        new_a = a.copy()
        new_a[i] -= delta
        new_a[j] += delta
        new_a = _project_onto_simplex(new_a, 1.0)

        cur = _objective(new_a)
        if cur + 1e-14 < best_score:
            a = new_a
            best_score = cur
            best_seq = a.copy()
            print(f"[{time.time():.1f}] SA improve {i}->{j} Δ={delta:.3e} best {best_score:.6f}")
        else:
            if rng.random() < math.exp(-(cur - best_score) / max(T, 1e-12)):
                a = new_a
        T *= decay
    return best_seq, best_score


# ----------------------------------------------------------------------
def _upsample(seq: np.ndarray, new_len: int) -> np.ndarray:
    """Linear interpolation of a simplex vector to a larger length."""
    old_len = seq.shape[0]
    if new_len == old_len:
        return seq.copy()
    x_old = np.linspace(0.0, 1.0, old_len)
    x_new = np.linspace(0.0, 1.0, new_len)
    new_seq = np.interp(x_new, x_old, seq)
    new_seq = np.maximum(new_seq, 0.0)
    new_seq = _project_onto_simplex(new_seq, 1.0)
    new_seq = _symmetrize(new_seq)
    return new_seq


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: int = 1000) -> list[float]:
    """
    Search for a non‑negative step function (list of heights) that minimises
    ``evaluate_sequence``.  The function respects the supplied time budget.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 1.0   # tiny safety margin

    overall_best_seq: np.ndarray | None = None
    overall_best_score = float("inf")

    # ------------------------------------------------------------------
    # 1.  Generate a few diverse seeds of moderate size
    n0 = 2500
    seeds: list[np.ndarray] = []

    seeds.append(_distribution_uniform(n0))
    seeds.append(_distribution_arcsine(n0))
    seeds.append(_distribution_beta_sym(n0, p=0.5))
    seeds.append(_distribution_edge_weighted(n0, p=1.0))
    seeds.append(_distribution_sine_weighted(n0, p=1.0))
    seeds.append(_distribution_dirichlet(n0, rng))

    # optionally incorporate previous construction if present
    try:
        prev = np.asarray(height_sequence_1, dtype=np.float64)   # type: ignore[name-defined]
        if prev.ndim == 1 and prev.size > 0:
            prev = np.maximum(prev, 0.0)
            prev = _project_onto_simplex(prev, 1.0)
            prev = _symmetrize(prev)
            seeds.append(prev)
    except Exception:
        pass

    # ------------------------------------------------------------------
    # 2.  Quick descent on each seed (fair share of the budget)
    remaining = deadline - time.time()
    per_seed = max(10.0, remaining * 0.4 / max(1, len(seeds)))  # at least 10 s each
    print(f"[{time.time():.1f}] Stage 1 – seed refinement (≈ {per_seed:.1f}s per seed)")

    for idx, s in enumerate(seeds):
        if time.time() >= deadline:
            break
        sub_deadline = time.time() + per_seed
        cur_score = _objective(s)
        if cur_score < overall_best_score:
            overall_best_score, overall_best_seq = cur_score, s.copy()
            print(f"[{time.time():.1f}] initial seed {idx} best {overall_best_score:.6f}")

        # mirror descent from this seed
        best_seq, best_score = _mirror_descent(
            s,
            sub_deadline,
            best_seq=overall_best_seq,
            best_score=overall_best_score,
            lr0=0.8,
            active_frac=0.025,
            max_active=40,
            beta=0.9,
            eval_interval=80,
            patience=1200,
        )
        if best_score < overall_best_score:
            overall_best_score, overall_best_seq = best_score, best_seq.copy()
            print(f"[{time.time():.1f}] new global best after seed {idx} → {overall_best_score:.6f}")

    # ------------------------------------------------------------------
    # 3.  Multi‑scale upsampling & refinement
    max_len = 15000
    scale_stage = 0
    while time.time() < deadline and overall_best_seq is not None and len(overall_best_seq) < max_len:
        scale_stage += 1
        new_len = min(max_len, len(overall_best_seq) * 2)
        print(f"[{time.time():.1f}] Stage 2.{scale_stage} – upsample to {new_len}")

        up_seq = _upsample(overall_best_seq, new_len)

        # allocate a slice of the remaining budget; aim for ~20 % of what is left
        remaining = deadline - time.time()
        sub_budget = max(30.0, remaining * 0.2)
        sub_deadline = time.time() + sub_budget

        best_seq, best_score = _mirror_descent(
            up_seq,
            sub_deadline,
            best_seq=overall_best_seq,
            best_score=overall_best_score,
            lr0=0.6,
            active_frac=0.025,
            max_active=30,
            beta=0.92,
            eval_interval=80,
            patience=1200,
        )
        if best_score < overall_best_score:
            overall_best_score, overall_best_seq = best_score, best_seq.copy()
            print(f"[{time.time():.1f}] new global best after upsample {scale_stage} → {overall_best_score:.6f}")

    # ------------------------------------------------------------------
    # 4.  Deterministic pairwise cleanup (as long as time permits)
    if overall_best_seq is not None and time.time() < deadline:
        print(f"[{time.time():.1f}] Stage 3 – deterministic pairwise refinement")
        overall_best_seq, overall_best_score = _pairwise_swap_local(
            overall_best_seq,
            deadline,
            best_seq=overall_best_seq,
            best_score=overall_best_score,
            max_trials=500_000,
        )

    # ------------------------------------------------------------------
    # 5.  Final cheap simulated‑annealing style local search
    if overall_best_seq is not None and time.time() < deadline:
        print(f"[{time.time():.1f}] Stage 4 – final simulated‑annealing clean‑up")
        overall_best_seq, overall_best_score = _random_local_search(
            overall_best_seq,
            deadline,
            best_seq=overall_best_seq,
            best_score=overall_best_score,
            init_temp=0.012,
            decay=0.9995,
        )

    # ------------------------------------------------------------------
    if overall_best_seq is None:
        # Fallback – uniform distribution of a modest size
        overall_best_seq = _distribution_uniform(2000)
        overall_best_score = _objective(overall_best_seq)
        print(f"[{time.time():.1f}] fallback uniform (score={overall_best_score:.6f})")

    print(f"[{time.time():.1f}] Final best score: {overall_best_score:.6f} (length={len(overall_best_seq)})")
    # Return a plain Python list – evaluate_sequence will apply its own clipping
    return overall_best_seq.tolist()
