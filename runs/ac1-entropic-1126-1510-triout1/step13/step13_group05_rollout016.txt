def _softmax_descent(a0, time_limit, best_seq, best_score,
                     init_tau=None, tau_decay=0.8, min_tau=1e-6,
                     max_iter_per_tau=2000):
    a = a0.copy()
    n = a.shape[0]
    best_local = a.copy()
    best_local_score = _objective(a)
    conv = _fft_self_convolve(a)
    if init_tau is None:
        # set initial temperature around magnitude of conv differences
        init_tau = max(conv) * 0.2  # 20% of max conv
        if init_tau < 1e-6:
            init_tau = 1e-6
    tau = init_tau
    # Outer loop on temperatures
    while tau > min_tau and time.time() < time_limit:
        # Perform iterations at current tau
        for t in range(max_iter_per_tau):
            if time.time() >= time_limit:
                break
            # compute convolution
            conv = _fft_self_convolve(a)
            maxc = conv.max()
            # compute softmax weights
            # subtract maxc for numeric stability
            w = np.exp((conv - maxc) / tau)
            w_sum = w.sum()
            if w_sum == 0:
                # fallback: uniform
                w = np.full_like(w, 1.0 / w.size)
            else:
                w /= w_sum
            # gradient via convolution with reversed a
            a_rev = a[::-1]
            # compute full convolution via FFT
            size = 1 << ((len(w) + len(a_rev) - 1) - 1).bit_length()
            W = np.fft.rfft(w, n=size)
            A = np.fft.rfft(a_rev, n=size)
            grad_full = np.fft.irfft(W * A, n=size)
            # extract slice
            grad = 2.0 * grad_full[len(w)-1 : len(w)-1 + n]
            # step size
            lr = tau  # base step size
            # try multiple step sizes adaptively
            # try until improvement or give up
            succeeded = False
            for factor in (1.0, 0.5, 0.25):
                step = lr * factor
                a_new = a - step * grad
                # project onto simplex
                a_new = _project_onto_simplex(a_new, s=1.0)
                new_score = _objective(a_new)
                if new_score < best_local_score - 1e-14:
                    best_local_score = new_score
                    best_local = a_new.copy()
                if new_score < _objective(a):
                    a = a_new
                    succeeded = True
                    # optionally update best global
                    if new_score < best_score - 1e-14:
                        best_score = new_score
                        best_seq = a_new.copy()
                        print(f"[{time.time():.1f}] softmax descent tau={tau:.2e} iter={t} new best {best_score:.6f}")
                    break
                # else try smaller step
            if not succeeded:
                # if no improvement, maybe reduce step size gradually
                # we could also break early
                pass
        tau *= tau_decay
    return best_local, best_local_score

def _annealing_polish(a0, time_limit, best_seq, best_score,
                      init_temp=1e-3, temp_decay=0.995, min_temp=1e-7):
    a = a0.copy()
    n = a.shape[0]
    best_local = a.copy()
    best_local_score = best_score
    temp = init_temp
    while time.time() < time_limit and temp > min_temp:
        # propose random move
        i = np.random.randint(n)
        # ensure source has >0 mass
        if a[i] < 1e-12:
            continue
        j = np.random.randint(n-1)
        if j >= i:
            j += 1
        delta = a[i] * np.random.rand() * 0.5  # move up to 50% of source mass
        a_new = a.copy()
        a_new[i] -= delta
        a_new[j] += delta
        # project onto simplex? Actually sum unchanged; zeros remain fine.
        # compute objective
        new_score = _objective(a_new)
        diff = new_score - best_local_score
        if diff < 0 or np.random.rand() < np.exp(-diff / temp):
            a = a_new
            if new_score < best_local_score:
                best_local_score = new_score
                best_local = a_new.copy()
                # update global best
                if new_score < best_score:
                    best_score = new_score
                    best_seq = a_new.copy()
                    print(f"[{time.time():.1f}] anneal new best {best_score:.6f}")

    return best_local, best_local_score

"""
Optimisation of a non‑negative step‑function (a probability vector) to 
minimise 2·n·max(conv(a,a)) / (sum a)^2, the exact expression used by
``evaluate_sequence``.

The search proceeds in three phases:

1. **Diverse seeds** – a quick sweep over several analytic families
   (uniform, arcsine, sine, cosine, quadratic, symmetric beta,
   edge‑weighted, power‑sum) for a handful of lengths, together with a
   few random Dirichlet / sparse vectors.  The 10 best candidates are
   kept.

2. **Smoothed‑max gradient descent** – for each retained seed we minimise
   the smooth approximation  

        M_τ(a) = τ·log∑_k exp(conv_k(a)/τ)

   (τ → 0 limits to the true max).  The gradient
   ∇M_τ = 2·corr(w,a) with w_i = softmax(conv_i/τ) is evaluated via FFT.
   An annealing schedule reduces τ while a simple projected gradient step
   (Euclidean projection onto the simplex) is performed.  This often
   escapes the plateaux where a pure sub‑gradient method stalls.

3. **Stochastic polishing** – a simulated‑annealing style hill‑climber
   that moves a random fraction of mass between two coordinates;
   uphill moves are occasionally accepted, allowing the algorithm to
   explore beyond the basin of attraction of the gradient descent.

The whole routine respects the supplied time budget (default 1000 s) and
finally returns a plain list of non‑negative floats (already normalised
to sum = 1).  The returned sequence is ready for ``evaluate_sequence``. 
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Euclidean projection of ``v`` onto the simplex {x≥0, Σx=s}."""
    if s < 0:
        raise ValueError("Simplex radius must be non‑negative")
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    # sort v in descending order
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.where(u - cssv / (np.arange(1, len(v) + 1)) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    return np.maximum(v - theta, 0.0)


# ----------------------------------------------------------------------
def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Full self‑convolution ``a*a`` using real FFT (length 2·n‑1)."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1) - 1).bit_length()   # next power of two
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[: 2 * n - 1]


# ----------------------------------------------------------------------
def _objective(a: np.ndarray) -> float:
    """Exact objective 2·n·max(conv) for a probability vector (sum=1)."""
    n = a.shape[0]
    b = _fft_self_convolve(a)
    return float(2.0 * n * b.max())

# Analytic families ----------------------------------------------------
def _distribution_uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _distribution_arcsine(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    vals /= vals.sum()
    return vals


def _distribution_sine(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64) + 0.5
    w = np.sin(np.pi * i / n)
    w /= w.sum()
    return w


def _distribution_cosine(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64) + 0.5
    w = 1.0 - np.cos(np.pi * i / n)
    w /= w.sum()
    return w


def _distribution_quadratic(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    x = (2.0 * i) / (n - 1) - 1.0
    w = 1.0 - x * x
    w = np.clip(w, 0.0, None)
    w /= w.sum()
    return w


def _distribution_beta_sym(n: int, p: float) -> np.ndarray:
    """Symmetric Beta(p,p) on the regular grid."""
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    w = np.exp((p - 1) * np.log(x) + (p - 1) * np.log(1.0 - x))
    if w.sum() == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _distribution_edge_weighted(n: int, p: float) -> np.ndarray:
    """Edge‑weighted family w_i ∝ ((i+½)*(n‑i‑½))^{‑p}."""
    i = np.arange(n, dtype=np.float64) + 0.5
    j = (n - np.arange(n, dtype=np.float64) - 0.5)
    w = (i * j) ** (-p)
    w = np.clip(w, 0.0, None)
    total = w.sum()
    if total > 0:
        w /= total
    else:
        w = np.full(n, 1.0 / n)
    return w


def _distribution_power_sum(n: int, p: float) -> np.ndarray:
    """w_i ∝ (i+1)^{‑p} + (n‑i)^{‑p}."""
    i = np.arange(1, n + 1, dtype=np.float64)
    w = i ** (-p) + (n + 1 - i) ** (-p)
    w = np.clip(w, 0.0, None)
    w /= w.sum()
    return w


def _distribution_random_sparse(
    n: int, rng: np.random.Generator, density: float = 0.01
) -> np.ndarray:
    """Sparse random vector with a prescribed density."""
    k = max(2, int(n * density))
    idx = rng.choice(n, size=k, replace=False)
    vals = np.zeros(n, dtype=np.float64)
    vals[idx] = rng.random(k)
    if vals.sum() == 0.0:
        vals[0] = 1.0
    vals /= vals.sum()
    return vals


def _mix_with_uniform(base: np.ndarray, lam: float) -> np.ndarray:
    """Convex combination λ·uniform + (1‑λ)·base."""
    n = base.shape[0]
    uniform = np.full(n, 1.0 / n, dtype=np.float64)
    return lam * uniform + (1.0 - lam) * base

# Smoothed‑max (log‑sum‑exp) gradient descent ---------------------------
def _smooth_max_descent(
    a0: np.ndarray,
    time_limit: float,
    best_seq: np.ndarray,
    best_score: float,
) -> tuple[np.ndarray, float]:
    """
    Annealed gradient descent on   τ·log∑exp(conv/τ)   with projection onto
    the simplex.  Returns the best vector seen during the run.
    """
    a = a0.copy()
    n = a.shape[0]

    # initialise temperature as a modest fraction of the current max conv
    cur_conv = _fft_self_convolve(a)
    tau = max(cur_conv.max() * 0.2, 1e-6)

    # parameters (tuned experimentally)
    tau_decay = 0.78            # geometric reduction each outer loop
    min_tau = 5e-7
    iters_per_tau = 250        # inner iterations for each temperature

    best_local = a.copy()
    best_local_score = _objective(a)

    while time.time() < time_limit and tau > min_tau:
        for _ in range(iters_per_tau):
            if time.time() >= time_limit:
                break

            # 1) self‑convolution and softmax weights
            conv = _fft_self_convolve(a)
            maxc = conv.max()
            # softmax – subtract max to avoid overflow
            w = np.exp((conv - maxc) / tau)
            w_sum = w.sum()
            if w_sum == 0.0:
                w = np.full_like(w, 1.0 / w.size)
            else:
                w /= w_sum

            # 2) gradient = 2 * corr(w, a)
            a_rev = a[::-1]
            size = 1 << ((len(w) + len(a_rev) - 1) - 1).bit_length()
            W = np.fft.rfft(w, n=size)
            A = np.fft.rfft(a_rev, n=size)
            grad_full = np.fft.irfft(W * A, n=size)
            grad = 2.0 * grad_full[n - 1 : n - 1 + n]   # slice of length n

            # 3) simple line search (geometric back‑off)
            step = tau                       # base step proportional to temperature
            improved = False
            for factor in (1.0, 0.5, 0.25):
                a_candidate = a - step * factor * grad
                a_candidate = _project_onto_simplex(a_candidate, s=1.0)
                cand_score = _objective(a_candidate)

                # record any global improvement
                if cand_score < best_score - 1e-14:
                    best_score = cand_score
                    best_seq = a_candidate.copy()
                    print(f"[{time.time():.1f}] smooth‑gd τ={tau:.2e} new global best {best_score:.6f}")

                if cand_score < best_local_score - 1e-14:
                    best_local_score = cand_score
                    best_local = a_candidate.copy()

                if cand_score < _objective(a) - 1e-14:
                    a = a_candidate
                    improved = True
                    break
            if not improved:
                # no decrease – reduce temperature faster
                break
        tau *= tau_decay
    return best_local, best_local_score

# Simulated‑annealing style polishing ----------------------------------
def _annealing_polish(
    a0: np.ndarray,
    time_limit: float,
    best_seq: np.ndarray,
    best_score: float,
) -> tuple[np.ndarray, float]:
    """
    Random pairwise mass moves with a Metropolis acceptance rule.
    Allows occasional uphill steps, helping to escape shallow local minima.
    """
    a = a0.copy()
    n = a.shape[0]

    # schedule of the temperature
    temp = 1e-3
    temp_min = 1e-7
    decay = 0.997

    while time.time() < time_limit and temp > temp_min:
        # pick source with non‑zero mass
        src_idx = np.where(a > 1e-14)[0]
        if src_idx.size == 0:
            break
        i = int(np.random.choice(src_idx))
        # pick a different destination
        j = np.random.randint(n - 1)
        if j >= i:
            j += 1

        # move a random fraction (up to 45 %) of the source mass
        delta = a[i] * np.random.rand() * 0.45
        if delta < 1e-15:
            continue

        a_new = a.copy()
        a_new[i] -= delta
        a_new[j] += delta

        new_score = _objective(a_new)
        diff = new_score - best_score

        if diff < 0 or np.random.rand() < math.exp(-diff / temp):
            a = a_new
            if new_score < best_score - 1e-14:
                best_score = new_score
                best_seq = a_new.copy()
                print(f"[{time.time():.1f}] anneal swap i={i}→j={j} δ={delta:.3g} new best {best_score:.6f}")

        temp *= decay

    return best_seq, best_score
