"""
Improved search for a non‑negative step function that minimizes
``evaluate_sequence``.

Algorithm Overview
------------------
1️⃣  **Rich seed pool** –  A large collection of analytic families
    (uniform, arcsine, edge‑weighted, edge‑inverse‑power,
    sine‑weighted, beta‑sym, beta‑general, exponential‑decay) is
    sampled for many lengths (up to 30 000) and many parameter values.
    Random Dirichlet vectors (widely varying concentration) and sparse
    vectors are also added.  The ~40 best seeds (lowest *exact* objective)
    are retained.

2️⃣  **Smooth‑max exponentiated gradient descent** –  Starting from each
    retained seed we run a short (few thousand‑iteration) descent on a
    *smoothed* version of the maximum convolution:
        Sα(a) = (1/α)·log Σ exp(α·(a*a))
    whose gradient is dense and well behaved:
        ∇Sα = 2 · conv_rev(weights, a)
    where the weights are the soft‑max of the convolution entries.
    An exponentiated‑gradient update keeps the iterate on the simplex.
    The smoothing parameter α is increased gradually, turning the
    objective from a gentle average towards the true max.

3️⃣  **Long refined descent** –  The best sequence from the short runs
    is further optimised for the bulk of the remaining budget using a
    momentum version of the same smooth‑max descent.

4️⃣  **Local polishing** –  A greedy pairwise mass‑transfer step followed
    by a cheap simulated‑annealing style search finishes the optimisation.

All steps respect the 1000 s time budget; the routine prints progress
and finally returns a plain Python list that can be fed directly to
``evaluate_sequence``.
"""

import time
import math
import numpy as np
from numpy.fft import rfft, irfft

# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x ≥ 0, sum(x)=s}."""
    if s < 0:
        raise ValueError("simplex radius must be non‑negative")
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho_arr = np.where(u - cssv / (np.arange(1, len(v) + 1)) > 0)[0]
    if rho_arr.size == 0:
        theta = 0.0
    else:
        rho = rho_arr[-1]
        theta = cssv[rho] / (rho + 1)
    return np.maximum(v - theta, 0.0)


# ----------------------------------------------------------------------
def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Full self‑convolution a*a using a real FFT (returns length 2n‑1)."""
    n = a.shape[0]
    need = 2 * n - 1
    size = 1 << ((need - 1).bit_length())
    A = rfft(a, n=size)
    conv = irfft(A * A, n=size)
    return conv[:need]


# ----------------------------------------------------------------------
def _objective(a: np.ndarray) -> float:
    """Exact objective 2·n·max(conv(a,a)) assuming sum(a)=1."""
    n = a.shape[0]
    b = _fft_self_convolve(a)
    return 2.0 * n * b.max()


# ----------------------------------------------------------------------
# Analytic families ----------------------------------------------------
def _distribution_uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _distribution_arcsine(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    vals /= vals.sum()
    return vals


def _distribution_edge_weighted(n: int, p: float) -> np.ndarray:
    """U‑shaped family w_i ∝ ((i+0.5)*(n‑i‑0.5))^{−p}."""
    i = np.arange(n, dtype=np.float64) + 0.5
    j = (n - np.arange(n, dtype=np.float64) - 0.5)
    w = (i * j) ** (-p)
    w /= w.sum()
    return w


def _distribution_edge_inversepow(n: int, p: float) -> np.ndarray:
    """Edge family w_i ∝ (i+0.5)^{−p} + (n−i−0.5)^{−p}."""
    i = np.arange(n, dtype=np.float64) + 0.5
    j = (n - np.arange(n, dtype=np.float64) - 0.5)
    w = i ** (-p) + j ** (-p)
    w /= w.sum()
    return w


def _distribution_sine_weighted(n: int, p: float) -> np.ndarray:
    """Weight ∝ sin⁽ᵖ⁾(π (i+½)/n)."""
    theta = np.pi * (np.arange(n, dtype=np.float64) + 0.5) / n
    w = np.sin(theta) ** p
    w /= w.sum()
    return w


def _distribution_beta_sym(n: int, p: float) -> np.ndarray:
    """Symmetric beta distribution Beta(p,p)."""
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    w = np.exp((p - 1) * (np.log(x) + np.log(1.0 - x)))
    w_sum = w.sum()
    if w_sum == 0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w_sum
    return w


def _distribution_beta_general(n: int, alpha: float, beta: float) -> np.ndarray:
    """Asymmetric beta distribution Beta(alpha,beta)."""
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    eps = np.finfo(float).tiny
    x = np.clip(x, eps, 1 - eps)
    w = np.exp((alpha - 1) * np.log(x) + (beta - 1) * np.log(1.0 - x))
    w_sum = w.sum()
    if w_sum == 0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w_sum
    return w


def _distribution_exponential_decay(n: int, rate: float) -> np.ndarray:
    """Exponential decay from the centre."""
    i = np.arange(n, dtype=np.float64)
    half = (n - 1) / 2.0
    w = np.exp(-rate * np.abs(i - half))
    w /= w.sum()
    return w


def _distribution_dirichlet(n: int, rng: np.random.Generator) -> np.ndarray:
    """Random Dirichlet sample with random concentration."""
    alpha = rng.uniform(0.01, 5.0)
    a = rng.dirichlet(np.full(n, alpha))
    return a.astype(np.float64)


def _distribution_random_sparse(
    n: int, rng: np.random.Generator, density: float = 0.006
) -> np.ndarray:
    """Sparse random vector with given density."""
    k = max(2, int(n * density))
    idx = rng.choice(n, size=k, replace=False)
    vals = np.zeros(n, dtype=np.float64)
    vals[idx] = rng.random(k)
    if vals.sum() == 0.0:
        vals[0] = 1.0
    vals /= vals.sum()
    return vals


def _mix_with_uniform(base: np.ndarray, lam: float) -> np.ndarray:
    """Convex combination lam·U + (1‑lam)·base."""
    n = base.shape[0]
    uniform = np.full(n, 1.0 / n, dtype=np.float64)
    return lam * uniform + (1.0 - lam) * base


# ----------------------------------------------------------------------
# Smooth‑max based exponentiated gradient descent -----------------------
def _smooth_weights(conv: np.ndarray, alpha: float) -> np.ndarray:
    """Soft‑max weights for a given smoothing parameter."""
    mx = conv.max()
    # subtract max for numerical stability
    exp_vals = np.exp(alpha * (conv - mx))
    s = exp_vals.sum()
    if s == 0.0:
        return np.full_like(conv, 1.0 / conv.size)
    return exp_vals / s


def _smooth_gradient(a: np.ndarray, alpha: float) -> np.ndarray:
    """
    Gradient of the smoothed max:  Sα(a) = (1/α)·log Σ exp(α·conv).
    Returns a vector of size n.
    """
    n = a.shape[0]
    conv = _fft_self_convolve(a)
    w = _smooth_weights(conv, alpha)            # length 2n‑1
    # reversed weights
    w_rev = w[::-1]
    need = len(w_rev) + n - 1                    # = 3n‑2
    size = 1 << ((need - 1).bit_length())
    A = rfft(w_rev, n=size)
    B = rfft(a, n=size)
    full = irfft(A * B, n=size)[:need]
    # grad_i = 2 * full[(len(w_rev)-1) - i]
    grad = 2.0 * full[(len(w_rev) - 1) - np.arange(n)]
    return grad


def _smooth_exponential_descent(
    a0: np.ndarray,
    deadline: float,
    best_seq: np.ndarray,
    best_score: float,
    lr0: float = 0.9,
    beta: float = 0.9,
    init_alpha: float = 5.0,
    alpha_inc: float = 0.001,
    eval_interval: int = 250,
) -> tuple[np.ndarray, float]:
    """
    Exponentiated‑gradient descent on the *smooth* objective.
    ``a0`` must already sum to one.
    """
    a = a0.copy()
    n = a.shape[0]
    t = 0
    momentum = np.zeros_like(a)
    alpha = init_alpha
    while time.time() < deadline and t < 300_000:
        t += 1
        grad = _smooth_gradient(a, alpha)
        # momentum
        momentum = beta * momentum + (1.0 - beta) * grad
        # decaying learning rate
        eta = lr0 / math.sqrt(t)
        # exponentiated update
        a = a * np.exp(-eta * momentum)
        a = _project_onto_simplex(a)

        # slowly tighten the smoothing
        alpha += alpha_inc

        if t % eval_interval == 0:
            cur = _objective(a)               # exact objective
            if cur < best_score - 1e-14:
                best_score = cur
                best_seq = a.copy()
                print(f"[{time.time():.1f}] it={t:,} new best {best_score:.6f}")
    return best_seq, best_score


# ----------------------------------------------------------------------
def _pairwise_swap_local(
    a: np.ndarray,
    deadline: float,
    best_seq: np.ndarray,
    best_score: float,
    max_trials: int | None = None,
) -> tuple[np.ndarray, float]:
    """Greedy random mass transfer between two positions."""
    n = a.shape[0]
    rng = np.random.default_rng()
    trials = 0
    while time.time() < deadline:
        if max_trials is not None and trials >= max_trials:
            break
        trials += 1

        src = np.where(a > 1e-12)[0]
        if src.size == 0:
            break
        i = rng.choice(src)
        j = rng.integers(n - 1)
        if j >= i:
            j += 1

        delta = a[i] * rng.random() * 0.35
        if delta < 1e-14:
            continue

        new_a = a.copy()
        new_a[i] -= delta
        new_a[j] += delta
        cur = _objective(new_a)

        if cur < best_score - 1e-14:
            a = new_a
            best_score = cur
            best_seq = a.copy()
            print(
                f"[{time.time():.1f}] swap {i}->{j} Δ={delta:.3e} new best {best_score:.6f}"
            )
    return best_seq, best_score


def _random_local_search(
    a: np.ndarray,
    deadline: float,
    best_seq: np.ndarray,
    best_score: float,
    init_temp: float = 0.01,
    decay: float = 0.9995,
) -> tuple[np.ndarray, float]:
    """Cheap SA‑style local search."""
    n = a.shape[0]
    rng = np.random.default_rng()
    T = init_temp
    while time.time() < deadline:
        i = rng.integers(n)
        j = rng.integers(n - 1)
        if j >= i:
            j += 1
        delta = a[i] * rng.random() * 0.3
        if delta < 1e-14:
            continue
        new_a = a.copy()
        new_a[i] -= delta
        new_a[j] += delta
        cur = _objective(new_a)
        if cur < best_score - 1e-14:
            a = new_a
            best_score = cur
            best_seq = a.copy()
            print(
                f"[{time.time():.1f}] SA improve {i}->{j} Δ={delta:.3e} best {best_score:.6f}"
            )
        else:
            if rng.random() < math.exp(-(cur - best_score) / max(T, 1e-12)):
                a = new_a
        T *= decay
    return best_seq, best_score

        print(f"[{time.time():.1f}] after SA best score {overall_best_score:.6f}")

    # ------------------------------------------------------------------
    print(
        f"Final best score (exact) ≈ {overall_best_score:.6f} (length={len(overall_best_seq)})"
    )
    # Return plain Python list – ``evaluate_sequence`` will normalise it.
    return overall_best_seq.tolist()
