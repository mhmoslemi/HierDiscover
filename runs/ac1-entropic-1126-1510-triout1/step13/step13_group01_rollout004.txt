"""
Propose a step‑function (non‑negative vector with sum 1) that minimises
``evaluate_sequence``.

Algorithm
---------

1️⃣ Diversified seeding – several analytic families (uniform, arcsine,
   beta, edge‑weighted, edge‑inverse‑power, sine‑weighted, tent,
   quadratic, exponential) are sampled for a few lengths.  Random
   Dirichlet and sparse vectors are added.  The ∼10 best seeds (lowest
   exact objective) are retained.

2️⃣ Smooth‐gradient projected‐gradient descent – from each retained seed we
   run a momentum descent that minimises the *maximum* of the self‑convolution.
   The non‑smooth max is approximated by a soft‑max with temperature ε,
   giving a dense gradient that can be computed by two FFT‑based
   convolutions per iteration.  After each step we project back onto the
   simplex (non‑negative, sum = 1).  Learning‑rate decays as 1/√t; a jitter
   is injected if progress stalls.

3️⃣ Local polishing – a greedy pairwise mass‑transfer followed by a cheap
   simulated‑annealing style search cleans up any remaining irregularities.

All work respects the 1000 s budget; intermediate progress and best scores
are printed.  The returned plain Python list can be fed directly to the
evaluator.
"""

import time

# ----------------------------------------------------------------------
# Simplex projection (Euclidean)
def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Project vector v onto the simplex {x ≥ 0, sum(x)=s}."""
    if s < 0:
        raise ValueError("simplex radius must be non‑negative")
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho_arr = np.where(u - cssv / (np.arange(1, len(v) + 1)) > 0)[0]
    if rho_arr.size == 0:
        theta = 0.0
    else:
        rho = rho_arr[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# FFT‑based convolutions
def _fft_convolve(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full convolution of real vectors a and b using a real FFT."""
    n = a.shape[0] + b.shape[0] - 1
    size = 1 << ((n - 1).bit_length())
    A = np.fft.rfft(a, n=size)
    B = np.fft.rfft(b, n=size)
    C = A * B
    conv = np.fft.irfft(C, n=size)
    return conv[:n]


def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Self‑convolution a*a (full length 2·len(a)‑1) via FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1 - 1).bit_length())
    A = np.fft.rfft(a, n=size)
    C = A * A
    conv = np.fft.irfft(C, n=size)
    return conv[: 2 * n - 1]


# ----------------------------------------------------------------------
# Exact objective (identical to evaluate_sequence up to the normalisation)
def _objective(a: np.ndarray) -> float:
    """Return 2·n·max(conv(a,a)) under the constraint sum(a)=1."""
    n = a.shape[0]
    conv = _fft_self_convolve(a)
    return 2.0 * n * conv.max()

# Analytic families (all return a normalized vector, i.e. sum == 1)
def _distribution_uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _distribution_arcsine(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    vals /= vals.sum()
    return vals


def _distribution_beta_sym(n: int, p: float) -> np.ndarray:
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    w = np.exp((p - 1) * np.log(x) + (p - 1) * np.log(1.0 - x))
    s = w.sum()
    if s == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= s
    return w


def _distribution_beta_general(n: int, alpha: float, beta: float) -> np.ndarray:
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    eps = np.finfo(float).tiny
    x = np.clip(x, eps, 1 - eps)
    w = np.exp((alpha - 1) * np.log(x) + (beta - 1) * np.log(1.0 - x))
    s = w.sum()
    if s == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= s
    return w


def _distribution_edge_weighted(n: int, p: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64) + 0.5
    j = (n - np.arange(n, dtype=np.float64) - 0.5)
    w = (i * j) ** (-p)
    s = w.sum()
    if s == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= s
    return w


def _distribution_edge_inversepow(n: int, p: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64) + 0.5
    j = (n - np.arange(n, dtype=np.float64) - 0.5)
    w = i ** (-p) + j ** (-p)
    s = w.sum()
    if s == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= s
    return w


def _distribution_sine_weighted(n: int, p: float) -> np.ndarray:
    theta = np.pi * (np.arange(n, dtype=np.float64) + 0.5) / n
    w = np.sin(theta) ** p
    s = w.sum()
    if s == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= s
    return w


def _distribution_tent(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    half = (n - 1) / 2.0
    w = np.where(i <= half, i + 1, n - i)
    w /= w.sum()
    return w


def _distribution_quadratic(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    half = (n - 1) / 2.0
    d = np.abs(i - half) + 1.0
    w = 1.0 / (d ** 2)
    w /= w.sum()
    return w


def _distribution_exponential_decay(n: int, rate: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    half = (n - 1) / 2.0
    w = np.exp(-rate * np.abs(i - half))
    w /= w.sum()
    return w


def _distribution_random_sparse(
    n: int, rng: np.random.Generator, density: float = 0.01
) -> np.ndarray:
    k = max(2, int(n * density))
    idx = rng.choice(n, size=k, replace=False)
    vals = np.zeros(n, dtype=np.float64)
    vals[idx] = rng.random(k)
    if vals.sum() == 0.0:
        vals[0] = 1.0
    vals /= vals.sum()
    return vals


def _distribution_dirichlet(n: int, rng: np.random.Generator) -> np.ndarray:
    alpha = rng.uniform(0.01, 2.0)
    a = rng.dirichlet(np.full(n, alpha))
    return a.astype(np.float64)


def _mix_with_uniform(base: np.ndarray, lam: float) -> np.ndarray:
    """lam * uniform + (1‑lam) * base."""
    n = base.shape[0]
    uniform = np.full(n, 1.0 / n, dtype=np.float64)
    return lam * uniform + (1.0 - lam) * base


# ----------------------------------------------------------------------
# Smooth‑max gradient (soft‑max over convolution entries)
def _smooth_max_gradient(a: np.ndarray, eps: float = 20.0) -> np.ndarray:
    """
    Returns a (sub‑)gradient of the smooth max:
        g = ∇ ( (1/eps) log Σ_k exp(eps * (a*a)[k]) )
    The gradient can be written as
        g_i = 2 * Σ_k w_k * a_{k-i}
    where w_k = softmax(eps * conv_k).  It is computed by FFT
    in O(n log n) time.
    """
    n = a.shape[0]
    conv = _fft_self_convolve(a)                # length 2n‑1
    max_c = conv.max()
    # soft‑max weights (numerically stable)
    diff = conv - max_c                         # ≤ 0
    # avoid overflow/underflow
    exp = np.exp(eps * diff)
    sum_exp = exp.sum()
    if sum_exp == 0.0 or not np.isfinite(sum_exp):
        # fallback to uniform weights on the active set
        w = np.zeros_like(conv)
        w[conv == max_c] = 1.0 / (conv == max_c).sum()
    else:
        w = exp / sum_exp

    # gradient = 2 * (w * a_rev)  (cross‑correlation)
    grad_full = _fft_convolve(w, a[::-1])      # length 3n‑2
    grad = 2.0 * grad_full[n - 1 : n - 1 + n]   # slice i=0..n‑1
    return grad

# Projected gradient descent with momentum (smooth‑max)
def _smooth_gradient_descent(
    a0: np.ndarray,
    deadline: float,
    best_seq: np.ndarray,
    best_score: float,
    lr0: float = 0.6,
    beta: float = 0.9,
    eps: float = 20.0,
    eval_interval: int = 250,
    patience: int = 3500,
) -> tuple[np.ndarray, float]:
    """
    Momentum projected gradient descent on the smooth–max objective.
    Returns the best sequence encountered and its exact objective.
    """
    a = a0.copy()
    n = a.shape[0]
    t = 0
    no_improve = 0
    momentum = np.zeros_like(a)
    rng = np.random.default_rng()

    while time.time() < deadline and t < 1_000_000:
        t += 1

        grad = _smooth_max_gradient(a, eps=eps)

        # momentum (exponential moving average of gradients)
        momentum = beta * momentum + (1.0 - beta) * grad

        # learning‑rate schedule  η_t = lr0 / sqrt(t)
        eta = lr0 / math.sqrt(t)

        # projected gradient step
        a = a - eta * momentum
        a = _project_onto_simplex(a, s=1.0)

        # periodic exact evaluation
        if t % eval_interval == 0:
            cur = _objective(a)
            if cur < best_score - 1e-14:
                best_score = cur
                best_seq = a.copy()
                no_improve = 0
                print(f"[{time.time():.1f}] it={t:,} new global best {best_score:.6f}")
            else:
                no_improve += 1

        # jitter if stuck
        if no_improve >= patience:
            jitter = rng.exponential(scale=0.008, size=n)
            a = a + jitter
            a = np.maximum(a, 0.0)
            a = _project_onto_simplex(a, s=1.0)
            no_improve = 0

    return best_seq, best_score

# Greedy pairwise mass‑transfer refinement
def _pairwise_swap_local(
    a: np.ndarray,
    deadline: float,
    best_seq: np.ndarray,
    best_score: float,
    max_trials: int | None = None,
) -> tuple[np.ndarray, float]:
    """Move a random amount of mass from i to j if the exact objective improves."""
    n = a.shape[0]
    trials = 0
    rng = np.random.default_rng()
    while time.time() < deadline:
        if max_trials is not None and trials >= max_trials:
            break
        trials += 1

        src = np.where(a > 1e-12)[0]
        if src.size == 0:
            break
        i = rng.choice(src)
        j = rng.integers(n - 1)
        if j >= i:
            j += 1

        delta = a[i] * rng.random() * 0.35
        if delta < 1e-14:
            continue

        new_a = a.copy()
        new_a[i] -= delta
        new_a[j] += delta
        cur = _objective(new_a)

        if cur < best_score - 1e-14:
            a = new_a
            best_score = cur
            best_seq = a.copy()
            print(f"[{time.time():.1f}] swap {i}->{j} Δ={delta:.3e} new best {best_score:.6f}")

    return best_seq, best_score

# Very cheap simulated‑annealing style local search
def _random_local_search(
    a: np.ndarray,
    deadline: float,
    best_seq: np.ndarray,
    best_score: float,
    init_temp: float = 0.01,
    decay: float = 0.9995,
) -> tuple[np.ndarray, float]:
    """Random mass moves with Metropolis acceptance."""
    n = a.shape[0]
    rng = np.random.default_rng()
    T = init_temp
    while time.time() < deadline:
        i = rng.integers(n)
        j = rng.integers(n - 1)
        if j >= i:
            j += 1
        delta = a[i] * rng.random() * 0.30
        if delta < 1e-14:
            continue
        new_a = a.copy()
        new_a[i] -= delta
        new_a[j] += delta
        cur = _objective(new_a)
        if cur < best_score - 1e-14:
            a = new_a
            best_score = cur
            best_seq = a.copy()
            print(f"[{time.time():.1f}] SA improve {i}->{j} Δ={delta:.3e} best {best_score:.6f}")
        else:
            if rng.random() < math.exp(-(cur - best_score) / max(T, 1e-12)):
                a = new_a
        T *= decay
    return best_seq, best_score

        for _ in range(4):
            if time.time() >= deadline:
                break
            a = _distribution_dirichlet(n, rng)
            sc = _objective(a)
            top_candidates.append((sc, a.copy()))
            if sc < overall_best_score:
                overall_best_score, overall_best_seq = sc, a.copy()
                print(f"[{time.time():.1f}] new best Dirichlet len={n} score={sc:.6f}")

        # sparse random seeds
        for _ in range(4):
            if time.time() >= deadline:
                break
            a = _distribution_random_sparse(n, rng, density=0.006)
            sc = _objective(a)
            top_candidates.append((sc, a.copy()))
            if sc < overall_best_score:
                overall_best_score, overall_best_seq = sc, a.copy()
                print(f"[{time.time():.1f}] new best sparse len={n} score={sc:.6f}")

        # asymmetric beta seeds
        for _ in range(3):
            if time.time() >= deadline:
                break
            alpha = 10 ** rng.uniform(-2, 1)   # 0.01 … 10
            beta = 10 ** rng.uniform(-2, 1)
            a = _distribution_beta_general(n, alpha, beta)
            sc = _objective(a)
            top_candidates.append((sc, a.copy()))
            if sc < overall_best_score:
                overall_best_score, overall_best_seq = sc, a.copy()
                print(
                    f"[{time.time():.1f}] new best beta(α={alpha:.3g},β={beta:.3g}) len={n} score={sc:.6f}"
                )

        # keep only the strongest seeds overall
        top_candidates.sort(key=lambda x: x[0])
        top_candidates = top_candidates[:12]

    # ------------------------------------------------------------------
    # 2. Include a previously discovered construction if present
    try:
        prev = np.asarray(height_sequence_1, dtype=np.float64)  # type: ignore[name-defined]
        if prev.ndim == 1 and prev.size > 0:
            prev = np.maximum(prev, 0.0)
            prev = prev / prev.sum()
            sc = _objective(prev)
            if sc < overall_best_score:
                overall_best_score, overall_best_seq = sc, prev.copy()
                print(f"[{time.time():.1f}] previous best score={sc:.6f}")
            top_candidates.append((sc, prev.copy()))
    except Exception:
        pass

    # ------------------------------------------------------------------
    # 3. Fallback in the unlikely case we have no seed
    if overall_best_seq is None:
        n = 3000
        overall_best_seq = _distribution_uniform(n)
        overall_best_score = _objective(overall_best_seq)
        print(f"[{time.time():.1f}] fallback uniform score={overall_best_score:.6f}")

    # ------------------------------------------------------------------
    # 4. Smooth‑gradient momentum descent on the best few seeds
    remaining_time = deadline - time.time()
    if remaining_time > 30.0:
        # allocate the bulk of the remaining budget equally to ≤3 top seeds
        num_descent = min(3, len(top_candidates))
        per_budget = remaining_time / num_descent
        for idx in range(num_descent):
            seed_score, seed_vec = top_candidates[idx]
            sub_deadline = time.time() + per_budget - 2.0
            print(
                f"[{time.time():.1f}] smooth‑gradient descent start (seed {idx}, len={len(seed_vec)})"
            )
            overall_best_seq, overall_best_score = _smooth_gradient_descent(
                seed_vec.copy(),
                sub_deadline,
                best_seq=overall_best_seq,
                best_score=overall_best_score,
                lr0=0.7,
                beta=0.9,
                eps=20.0,
                eval_interval=200,
                patience=3000,
            )
            print(
                f"[{time.time():.1f}] after descent {idx} best score {overall_best_score:.6f}"
            )

    # ------------------------------------------------------------------
    # 5. Greedy pairwise refinement
    if time.time() < deadline:
        print(f"[{time.time():.1f}] greedy pairwise refinement")
        overall_best_seq, overall_best_score = _pairwise_swap_local(
            overall_best_seq.copy(),
            deadline,
            best_seq=overall_best_seq,
            best_score=overall_best_score,
            max_trials=400_000,
        )

    # ------------------------------------------------------------------
    # 6. Final stochastic local search (simulated‑annealing style)
    if time.time() < deadline:
        print(f"[{time.time():.1f}] final simulated‑annealing search")
        overall_best_seq, overall_best_score = _random_local_search(
            overall_best_seq.copy(),
            deadline,
            best_seq=overall_best_seq,
            best_score=overall_best_score,
            init_temp=0.012,
            decay=0.9996,
        )

    # ------------------------------------------------------------------
    print(
        f"Final best exact score ≈ {overall_best_score:.6f} (length={len(overall_best_seq)})"
    )
    # Return a plain Python list – the evaluator will normalise it again.
    return overall_best_seq.tolist()
