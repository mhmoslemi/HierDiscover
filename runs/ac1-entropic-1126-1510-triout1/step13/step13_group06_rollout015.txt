"""
Propose a low‑score step function.

Algorithm Overview
-------------------
1. **Broad seed generation** – create many analytic families
   (uniform, arcsine, symmetric/asymmetric Beta, edge‑weighted,
   sine‑weighted) for a range of lengths (up to 18 000).  Random
   Dirichlet and sparse vectors are also sampled.  All candidates are
   normalised to a simplex (∑a_i = 1) and the best ~20 are kept.

2. **Two‑stage continuous optimisation**
   * **Soft‑max exponential descent** – minimise a smooth
     log‑sum‑exp surrogate of the max‑convolution.  A single FFT
     supplies the exact gradient of the surrogate; an exponentiated‑
     gradient (mirror descent on the simplex) updates the vector.
   * **Exact max‑subgradient descent** – after the surrogate has
     converged, switch to the true subgradient of the max entry.
     The subgradient is obtained by a cheap slice of the vector
     (no FFT needed).  The same multiplicative update is used and
     the temperature is set to ∞ (i.e. τ → ∞).

3. **Local refinement**
   * **Deterministic pairwise swaps** – move a random chunk of mass
     from a “high‑gradient’’ index to a “low‑gradient’’ one whenever it
     reduces the exact objective.
   * **Lightweight simulated‑annealing walk** – occasional uphill
     moves allow the search to escape shallow local minima.

All steps respect the global 1000 s budget; intermediate progress and
current best scores are printed.  The routine finally returns a plain
Python list (the `evaluate_sequence` wrapper will perform the final
clipping/normalisation itself).
"""

def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x≥0, Σx = s}."""
    if s < 0:
        raise ValueError("simplex radius must be non‑negative")
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho_arr = np.where(u - cssv / (np.arange(1, len(v) + 1)) > 0)[0]
    if rho_arr.size == 0:
        theta = 0.0
    else:
        rho = rho_arr[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Full linear self‑convolution a∗a using a real FFT."""
    n = a.shape[0]
    need = 2 * n - 1
    size = 1 << ((need - 1).bit_length())
    A = rfft(a, n=size)
    conv = irfft(A * A, n=size)
    return conv[:need]


def _objective(a: np.ndarray) -> float:
    """Exact objective 2·n·max(conv(a,a)) for a simplex‑normalised vector."""
    n = a.shape[0]
    b = _fft_self_convolve(a)
    return float(2.0 * n * b.max())

# Analytic families

def _distribution_uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _distribution_arcsine(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    vals /= vals.sum()
    return vals


def _distribution_beta_sym(n: int, p: float) -> np.ndarray:
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    logw = (p - 1) * np.log(x) + (p - 1) * np.log(1.0 - x)
    w = np.exp(logw - np.max(logw))      # stable exponentiation
    w[w < 0] = 0.0
    total = w.sum()
    if total == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= total
    return w


def _distribution_beta_asym(n: int, a: float, b: float) -> np.ndarray:
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    logw = (a - 1) * np.log(x) + (b - 1) * np.log(1.0 - x)
    w = np.exp(logw - np.max(logw))
    w[w < 0] = 0.0
    total = w.sum()
    if total == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= total
    return w


def _distribution_edge_weighted(n: int, p: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64) + 0.5
    j = (n - np.arange(n, dtype=np.float64) - 0.5)
    w = (i * j) ** (-p)
    w[w < 0] = 0.0
    total = w.sum()
    if total == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= total
    return w


def _distribution_sine_weighted(n: int, p: float) -> np.ndarray:
    theta = np.pi * (np.arange(n, dtype=np.float64) + 0.5) / n
    w = np.sin(theta) ** p
    w[w < 0] = 0.0
    total = w.sum()
    if total == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= total
    return w


def _distribution_random_sparse(
    n: int, rng: np.random.Generator, density: float = 0.007
) -> np.ndarray:
    k = max(2, int(n * density))
    idx = rng.choice(n, size=k, replace=False)
    vals = np.zeros(n, dtype=np.float64)
    vals[idx] = rng.random(k)
    if vals.sum() == 0.0:
        vals[0] = 1.0
    vals /= vals.sum()
    return vals


def _mix_with_uniform(base: np.ndarray, lam: float) -> np.ndarray:
    """Convex combination lam·U + (1‑lam)·base."""
    n = base.shape[0]
    uniform = np.full(n, 1.0 / n, dtype=np.float64)
    return lam * uniform + (1.0 - lam) * base


# ----------------------------------------------------------------------
# Gradient utilities for the soft‑max surrogate
# ----------------------------------------------------------------------
def _softmax_weights(conv: np.ndarray, tau: float) -> np.ndarray:
    """stable soft‑max of τ·conv."""
    max_c = conv.max()
    w_raw = np.exp(tau * (conv - max_c))
    total = w_raw.sum()
    if total == 0.0:
        return np.full_like(conv, 1.0 / conv.size)
    return w_raw / total


def _weighted_gradient(a: np.ndarray, w: np.ndarray) -> np.ndarray:
    """
    Gradient of L(a)= Σ w_k·(a*a)[k].
    Closed form: ∇L = 2·(a_rev ⊛ w)  (first n entries).
    """
    n = a.shape[0]
    a_rev = a[::-1]
    need = len(w) + n - 1               # = 3·n‑2
    size = 1 << ((need - 1).bit_length())
    A = rfft(a_rev, n=size)
    W = rfft(w, n=size)
    conv = irfft(A * W, n=size)[:need]
    grad = 2.0 * conv[n - 1 : n - 1 + n]
    return grad

# Soft‑max exponential descent (first stage)
# ----------------------------------------------------------------------
def _softmax_exponential_descent(
    a: np.ndarray,
    deadline: float,
    best_seq: np.ndarray,
    best_score: float,
    eta0: float = 0.6,
    tau0: float = 0.5,
    tau_mult: float = 1.002,
    eval_interval: int = 120,
    patience: int = 2500,
) -> tuple[np.ndarray, float]:
    """Exponentiated‑gradient descent on the log‑sum‑exp surrogate."""
    n = a.shape[0]
    t = 0
    no_improve = 0
    rng = np.random.default_rng()
    tau = tau0

    while time.time() < deadline and t < 300_000:
        t += 1

        conv = _fft_self_convolve(a)
        w = _softmax_weights(conv, tau)
        grad = _weighted_gradient(a, w)

        eta = eta0 / math.sqrt(t)
        a = a * np.exp(-eta * grad)

        # project onto simplex
        s = a.sum()
        a = a / s if s > 0 else np.full(n, 1.0 / n)

        tau *= tau_mult

        if t % eval_interval == 0:
            cur = _objective(a)
            if cur < best_score - 1e-14:
                best_score, best_seq = cur, a.copy()
                no_improve = 0
                print(f"[{time.time():.1f}] soft‑max it={t} new best {best_score:.6f}")
            else:
                no_improve += 1

        if no_improve >= patience:
            # tiny jitter to escape plateaus
            jitter = rng.exponential(scale=0.003, size=n)
            a = a + jitter
            a = np.maximum(a, 0.0)
            a = a / a.sum()
            no_improve = 0

    return best_seq, best_score


# ----------------------------------------------------------------------
# Exact max‑subgradient descent (second stage)
# ----------------------------------------------------------------------
def _max_subgrad_descent(
    a: np.ndarray,
    deadline: float,
    best_seq: np.ndarray,
    best_score: float,
    eta0: float = 0.55,
    eval_interval: int = 80,
    patience: int = 1500,
) -> tuple[np.ndarray, float]:
    """Projected multiplicative descent using the true max‑subgradient."""
    n = a.shape[0]
    t = 0
    no_improve = 0
    rng = np.random.default_rng()

    while time.time() < deadline and t < 500_000:
        t += 1

        b = _fft_self_convolve(a)
        max_val = b.max()
        # pick a random index among the maximisers
        ks = np.where(b == max_val)[0]
        k = ks[rng.integers(len(ks))]

        # subgradient for the chosen max entry
        w = np.zeros(2 * n - 1, dtype=np.float64)
        w[k] = 1.0
        grad = _weighted_gradient(a, w)          # = 2·a_rev ⊛ e_k

        eta = eta0 / math.sqrt(t)
        a = a * np.exp(-eta * grad)

        s = a.sum()
        a = a / s if s > 0 else np.full(n, 1.0 / n)

        if t % eval_interval == 0:
            cur = _objective(a)
            if cur < best_score - 1e-14:
                best_score, best_seq = cur, a.copy()
                no_improve = 0
                print(f"[{time.time():.1f}] max‑subgrad it={t} new best {best_score:.6f}")
            else:
                no_improve += 1

        if no_improve >= patience:
            jitter = rng.exponential(scale=0.004, size=n)
            a = a + jitter
            a = np.maximum(a, 0.0)
            a = a / a.sum()
            no_improve = 0

    return best_seq, best_score


# ----------------------------------------------------------------------
# Pairwise swap local refinement
# ----------------------------------------------------------------------
def _pairwise_swap_local(
    a: np.ndarray,
    deadline: float,
    best_seq: np.ndarray,
    best_score: float,
    max_trials: int | None = None,
) -> tuple[np.ndarray, float]:
    """Random mass transfers between two positions, keeping improvements."""
    n = a.shape[0]
    rng = np.random.default_rng()
    trials = 0
    while time.time() < deadline:
        if max_trials is not None and trials >= max_trials:
            break
        trials += 1
        nonzero = np.where(a > 1e-12)[0]
        if nonzero.size == 0:
            break
        i = rng.choice(nonzero)
        j = rng.integers(n - 1)
        if j >= i:
            j += 1
        delta = a[i] * rng.random() * 0.35
        if delta < 1e-14:
            continue
        new_a = a.copy()
        new_a[i] -= delta
        new_a[j] += delta
        cur = _objective(new_a)
        if cur < best_score - 1e-14:
            a = new_a
            best_score = cur
            best_seq = a.copy()
            print(f"[{time.time():.1f}] swap i={i}→j={j} δ={delta:.3e} best {best_score:.6f}")
    return best_seq, best_score


# ----------------------------------------------------------------------
# Cheap simulated‑annealing walk
# ----------------------------------------------------------------------
def _random_local_search(
    a: np.ndarray,
    deadline: float,
    best_seq: np.ndarray,
    best_score: float,
    init_temp: float = 1e-2,
    decay: float = 0.9996,
) -> tuple[np.ndarray, float]:
    """Very cheap SA that occasionally accepts worsening moves."""
    n = a.shape[0]
    rng = np.random.default_rng()
    T = init_temp
    while time.time() < deadline:
        i = rng.integers(n)
        j = rng.integers(n - 1)
        if j >= i:
            j += 1
        delta = a[i] * rng.random() * 0.3
        if delta < 1e-14:
            continue
        new_a = a.copy()
        new_a[i] -= delta
        new_a[j] += delta
        cur = _objective(new_a)
        if cur < best_score - 1e-14:
            a = new_a
            best_score = cur
            best_seq = a.copy()
            print(f"[{time.time():.1f}] SA improve {i}->{j} Δ={delta:.3e} best {best_score:.6f}")
        else:
            if rng.random() < math.exp(-(cur - best_score) / max(T, 1e-12)):
                a = new_a
        T *= decay
    return best_seq, best_score

# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: int = 1000) -> list[float]:
    """
    Search for a non‑negative step‑function that minimises ``evaluate_sequence``.
    Returns a plain Python list; the routine respects the supplied time budget.
    """
    rng = np.random.default_rng(seed)
    start = time.time()
    deadline = start + budget_s - 3.0      # small safety margin

    overall_best_seq = None
    overall_best_score = float("inf")
    top_candidates: list[tuple[float, np.ndarray]] = []

    # ------------------------------------------------------------------
    # 1. Seed generation (analytic + random)
    # ------------------------------------------------------------------
    lengths = [2500, 3500, 5000, 7500, 11000, 15000, 18000]
    p_grid = np.concatenate([
        np.logspace(-2, -1, 12, base=10),   # 0.01 … 0.1
        np.logspace(-1, 0, 15, base=10),    # 0.1 … 1
        np.logspace(0, 0.7, 12, base=10),   # 1 … 5
    ])
    lam_grid = np.linspace(0.0, 1.0, 9)      # mixing with uniform

    base_families = []
    for n in lengths:
        if time.time() >= deadline:
            break
        base_families.append((_distribution_uniform(n), "uniform"))
        base_families.append((_distribution_arcsine(n), "arcsine"))

        # parametrised families
        for p in p_grid:
            beta = _distribution_beta_sym(n, p)
            edge = _distribution_edge_weighted(n, p)
            sine = _distribution_sine_weighted(n, p)
            base_families.extend([
                (beta, f"beta_sym p={p:.3g}"),
                (edge, f"edge p={p:.3g}"),
                (sine, f"sine p={p:.3g}"),
            ])

        # asymmetric beta few random instances
        for _ in range(5):
            a_param = 10 ** rng.uniform(-2, 0.7)
            b_param = 10 ** rng.uniform(-2, 0.7)
            base_families.append((_distribution_beta_asym(n, a_param, b_param),
                                 f"beta_asym a={a_param:.3g} b={b_param:.3g}"))

        # random Dirichlet seeds
        for _ in range(6):
            alpha = rng.uniform(0.01, 2.0)
            cand = rng.dirichlet(np.full(n, alpha))
            base_families.append((cand, "dirichlet"))

        # sparse random
        for _ in range(6):
            cand = _distribution_random_sparse(n, rng, density=0.006)
            base_families.append((cand, "sparse"))

    # evaluate all base candidates, keep a short ranked list
    for vec, tag in base_families:
        if time.time() >= deadline:
            break
        sc = _objective(vec)
        top_candidates.append((sc, vec.copy()))
        if sc < overall_best_score:
            overall_best_score, overall_best_seq = sc, vec.copy()
            print(f"[{time.time():.1f}] new global best ({tag}) score={sc:.6f}")

    # mix each base with uniform (different λ) – cheap extra candidates
    mixed: list[tuple[float, np.ndarray]] = []
    for sc, vec in top_candidates[:30]:   # restrict to top few for mixing
        for lam in lam_grid:
            cand = _mix_with_uniform(vec, lam)
            mixed.append((cand, lam))
    for cand, lam in mixed:
        if time.time() >= deadline:
            break
        sc = _objective(cand)
        top_candidates.append((sc, cand.copy()))
        if sc < overall_best_score:
            overall_best_score, overall_best_seq = sc, cand.copy()
            print(f"[{time.time():.1f}] mixed lam={lam:.2f} new best score={sc:.6f}")

    # keep only the most promising seeds (global ranking)
    top_candidates.sort(key=lambda x: x[0])
    top_candidates = top_candidates[:20]          # keep ~20 best seeds

    # ------------------------------------------------------------------
    # 2. Continuous optimisation on the elite seeds
    # ------------------------------------------------------------------
    remaining = deadline - time.time()
    if remaining > 60.0 and top_candidates:
        per_seed = remaining / max(1, len(top_candidates))
        print(f"[{time.time():.1f}] start optimisation on {len(top_candidates)} seeds")
        for idx, (_, seed_vec) in enumerate(top_candidates):
            if time.time() >= deadline:
                break
            sub_deadline = time.time() + per_seed * 0.6   # 60 % for soft‑max
            # soft‑max phase
            a_opt, sc_opt = _softmax_exponential_descent(
                seed_vec.copy(),
                sub_deadline,
                best_seq=overall_best_seq,
                best_score=overall_best_score,
                eta0=0.6,
                tau0=0.5,
                tau_mult=1.003,
                eval_interval=150,
                patience=3000,
            )
            if sc_opt < overall_best_score:
                overall_best_score, overall_best_seq = sc_opt, a_opt.copy()
                print(f"[{time.time():.1f}] after soft‑max seed {idx} best {overall_best_score:.6f}")

            # exact max‑subgradient phase (use remaining slice)
            sub_deadline2 = time.time() + per_seed * 0.35
            a_opt2, sc_opt2 = _max_subgrad_descent(
                a_opt,
                sub_deadline2,
                best_seq=overall_best_seq,
                best_score=overall_best_score,
                eta0=0.55,
                eval_interval=100,
                patience=2000,
            )
            if sc_opt2 < overall_best_score:
                overall_best_score, overall_best_seq = sc_opt2, a_opt2.copy()
                print(f"[{time.time():.1f}] after max‑subgrad seed {idx} best {overall_best_score:.6f}")

            # reserve a tiny portion for pairwise refinement later
            # (continue loop)

    # ------------------------------------------------------------------
    # 3. Deterministic pairwise refinement
    # ------------------------------------------------------------------
    if time.time() < deadline:
        print(f"[{time.time():.1f}] pairwise refinement")
        overall_best_seq, overall_best_score = _pairwise_swap_local(
            overall_best_seq.copy(),
            deadline,
            best_seq=overall_best_seq,
            best_score=overall_best_score,
            max_trials=350_000,
        )

    # ------------------------------------------------------------------
    # 4. Final cheap simulated‑annealing walk
    # ------------------------------------------------------------------
    if time.time() < deadline:
        print(f"[{time.time():.1f}] final stochastic local search")
        overall_best_seq, overall_best_score = _random_local_search(
            overall_best_seq.copy(),
            deadline,
            best_seq=overall_best_seq,
            best_score=overall_best_score,
            init_temp=0.008,
            decay=0.9997,
        )

    # ------------------------------------------------------------------
    # 5. Return result
    # ------------------------------------------------------------------
    print(
        f"Final best score: {overall_best_score:.6f} (length={len(overall_best_seq)})"
    )
    return overall_best_seq.tolist()
