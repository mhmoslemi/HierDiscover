def _lp_refine(a, max_iter=10, t=0.5):
    n = len(a)
    a = _ensure_positive(a)
    best_seq = a.copy()
    best_obj = _objective(a)
    for it in range(max_iter):
        # compute max conv
        conv = _fft_self_convolve(a)
        M = conv.max()
        # build constraint matrix
        c = -np.ones(n)  # maximize sum
        # Build Toeplitz matrix
        ccol = np.r_[a, np.zeros(n-1)]
        crow = np.r_[a[0], np.zeros(n-1)]
        A = scipy.linalg.toeplitz(ccol, crow)
        b = np.full(2*n-1, M)
        # Solve LP
        res = scipy.optimize.linprog(c, A_ub=A, b_ub=b, bounds=(0, None), method='highs')
        if not res.success:
            # LP failed, break
            break
        g = res.x
        # normalize g
        g = g / g.sum()
        # convex combination
        a = (1 - t) * a + t * g
        # ensure positivity, normalise
        a = np.maximum(a, 0)
        a /= a.sum()
        # evaluate
        obj = _objective(a)
        if obj < best_obj - 1e-14:
            best_obj = obj
            best_seq = a.copy()
        # decrease t
        t *= 0.9
    return best_seq, best_obj
```

"""
Propose a low‑score step‑function.
---------------------------------

The search proceeds in three stages.

1. **Broad seed generation** – analytic families (uniform,
   arcsine, symmetric / asymmetric beta, edge‑weighted, sine‑weighted),
   Dirichlet draws and very sparse random vectors are created for a few
   moderate lengths (2000‑4000).  All seeds are normalised to a unit sum
   and the ∼30 best (lowest exact objective) are retained.

2. **Linear‑programming refinement** – for each retained seed we solve
   a linear programme

        maximise   Σ g_i
        s.t.       (f * g)_k ≤ M   for all k,
                   g_i ≥ 0,

   where * denotes the (full) convolution, *f* is the current candidate
   and *M* = max(f * f).  The LP pushes the sum of *g* as far as the
   convolution bound allows; after normalising *g* we take a convex
   combination `f ← (1‑t)·f + t·g` (t≈0.5, decaying).  This step is
   guaranteed to never increase the objective and often yields a
   substantial drop.

3. **Fine‑grained exponential descent** – the best LP‑refined candidate
   is polished with the multiplicative (exponentiated) gradient descent
   that was used in the previous state‑of‑the‑art code.  A soft‑max
   surrogate of the max‑convolution is employed, the gradient is
   obtained by a single FFT‑based correlation and the iterate is
   projected onto the simplex after each step.

The routine respects the supplied time budget (default 1000 s) and
returns a plain Python list; ``evaluate_sequence`` will perform the
final checks and normalisation.
"""
import time
import math
import numpy as np
import scipy.linalg
import scipy.optimize

# ----------------------------------------------------------------------
# Utility functions ----------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x≥0, Σx = s}."""
    if s < 0:
        raise ValueError("simplex radius must be non‑negative")
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.where(u - cssv / (np.arange(1, len(v) + 1)) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Full self‑convolution a*a using a real FFT (returns length 2n‑1)."""
    n = a.shape[0]
    need = 2 * n - 1
    size = 1 << ((need - 1).bit_length())
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[:need]


def _objective(a: np.ndarray) -> float:
    """Exact objective 2·n·max(conv(a,a)) for a vector with sum = 1."""
    n = a.shape[0]
    b = _fft_self_convolve(a)
    return float(2.0 * n * b.max())


# ----------------------------------------------------------------------
# Analytic families ----------------------------------------------------
def _distribution_uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _distribution_arcsine(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    vals /= vals.sum()
    return vals


def _distribution_beta_sym(n: int, p: float) -> np.ndarray:
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    w = np.exp((p - 1) * np.log(x) + (p - 1) * np.log(1.0 - x))
    if w.sum() == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _distribution_beta_asym(n: int, a: float, b: float) -> np.ndarray:
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    w = np.exp((a - 1) * np.log(x) + (b - 1) * np.log(1.0 - x))
    if w.sum() == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _distribution_edge_weighted(n: int, p: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64) + 0.5
    j = (n - np.arange(n, dtype=np.float64) - 0.5)
    w = (i * j) ** (-p)
    w /= w.sum()
    return w


def _distribution_sine_weighted(n: int, p: float) -> np.ndarray:
    theta = np.pi * (np.arange(n, dtype=np.float64) + 0.5) / n
    w = np.sin(theta) ** p
    w /= w.sum()
    return w


def _distribution_exponential_edge(n: int, alpha: float) -> np.ndarray:
    """Symmetric exponential decay from the centre."""
    i = np.arange(n, dtype=np.float64)
    d = np.minimum(i, n - 1 - i)
    w = np.exp(-alpha * d)
    w += 1e-12                      # avoid exact zeros
    w /= w.sum()
    return w


def _distribution_random_sparse(
    n: int, rng: np.random.Generator, density: float = 0.01
) -> np.ndarray:
    """Sparse random vector with a given density of non‑zeros."""
    k = max(2, int(n * density))
    idx = rng.choice(n, size=k, replace=False)
    vals = np.zeros(n, dtype=np.float64)
    vals[idx] = rng.random(k)
    if vals.sum() == 0.0:
        vals[0] = 1.0
    vals /= vals.sum()
    return vals


def _mix_with_uniform(base: np.ndarray, lam: float) -> np.ndarray:
    """Convex combination lam·U + (1‑lam)·base, where U is uniform."""
    n = base.shape[0]
    uniform = np.full(n, 1.0 / n, dtype=np.float64)
    return lam * uniform + (1.0 - lam) * base


# ----------------------------------------------------------------------
# Linear‑programming refinement -----------------------------------------
def _lp_refine(
    a: np.ndarray,
    max_iter: int = 8,
    t0: float = 0.5,
    eps: float = 1e-12,
) -> tuple[np.ndarray, float]:
    """
    Perform the LP‑based descent described in the problem statement.
    Returns the best vector found and its objective.
    """
    n = a.shape[0]
    # make all entries strictly positive (required for bounded LP)
    a = np.maximum(a, eps)
    a = a / a.sum()

    best_seq = a.copy()
    best_obj = _objective(a)
    t = t0

    for it in range(max_iter):
        # current peak of the self‑convolution
        conv = _fft_self_convolve(a)
        M = conv.max()

        # build the (2n‑1)×n Toeplitz matrix for (a * g) ≤ M
        col = np.r_[a, np.zeros(n - 1)]
        row = np.r_[a[0], np.zeros(n - 1)]
        A = scipy.linalg.toeplitz(col, row)   # dense (2n‑1)×n matrix

        # LP: maximise Σ g_i  ⇔  minimise −Σ g_i
        c = -np.ones(n, dtype=np.float64)      # objective direction
        b = np.full(2 * n - 1, M, dtype=np.float64)

        res = scipy.optimize.linprog(
            c,
            A_ub=A,
            b_ub=b,
            bounds=[(0, None)] * n,
            method="highs",
            options={"presolve": True},
        )
        if not res.success:
            # If the LP fails we abort further refinement.
            break

        g = res.x
        # normalise g to a probability vector
        g = np.maximum(g, 0)
        if g.sum() == 0.0:
            g = np.full_like(g, 1.0 / n)
        else:
            g = g / g.sum()

        # convex combination and projection back onto the simplex
        a = (1.0 - t) * a + t * g
        a = _project_onto_simplex(a, s=1.0)

        # evaluate
        obj = _objective(a)
        if obj < best_obj - 1e-14:
            best_obj = obj
            best_seq = a.copy()
            print(f"[{time.time():.1f}] LP iter={it+1} new best {best_obj:.6f}")

        # gradually reduce the mixing factor
        t *= 0.9

    return best_seq, best_obj


# ----------------------------------------------------------------------
# Exponential descent (soft‑max) ----------------------------------------
def _softmax_weights(conv: np.ndarray, tau: float) -> np.ndarray:
    """soft‑max of τ·conv (stable). Returns a probability vector."""
    max_c = conv.max()
    w_raw = np.exp(tau * (conv - max_c))
    total = w_raw.sum()
    if total == 0.0:
        return np.full_like(conv, 1.0 / conv.size)
    return w_raw / total


def _weighted_gradient(a: np.ndarray, w: np.ndarray) -> np.ndarray:
    """
    Gradient of   L(a)= Σ_i w_i·(a*a)[i]   w.r.t a.
    Closed‑form:   ∇L = 2 * (a_rev ⊛ w)   (first n entries).
    """
    n = a.shape[0]
    a_rev = a[::-1]
    need = len(w) + n - 1          # = 3·n‑2
    size = 1 << ((need - 1).bit_length())
    A = np.fft.rfft(a_rev, n=size)
    W = np.fft.rfft(w, n=size)
    conv = np.fft.irfft(A * W, n=size)
    conv = conv[:need]
    grad = 2.0 * conv[n - 1 : n - 1 + n]   # length n
    return grad


def _full_exponential_descent(
    a: np.ndarray,
    deadline: float,
    best_seq: np.ndarray,
    best_score: float,
    eta0: float = 0.55,
    tau0: float = 0.5,
    tau_mult: float = 1.002,
    eval_interval: int = 120,
    patience: int = 2600,
) -> tuple[np.ndarray, float]:
    """
    Multiplicative (exponentiated) gradient descent on the *soft‑max*
    surrogate of the max‑convolution.
    """
    n = a.shape[0]
    t = 0
    no_improve = 0
    rng = np.random.default_rng()
    tau = tau0

    while time.time() < deadline and t < 300_000:
        t += 1

        conv = _fft_self_convolve(a)
        w = _softmax_weights(conv, tau)
        grad = _weighted_gradient(a, w)

        eta = eta0 / math.sqrt(t)
        a = a * np.exp(-eta * grad)

        # project onto simplex (keeps sum = 1)
        a = _project_onto_simplex(a, s=1.0)

        tau *= tau_mult

        if t % eval_interval == 0:
            cur = _objective(a)
            if cur < best_score - 1e-14:
                best_score = cur
                best_seq = a.copy()
                no_improve = 0
                print(f"[{time.time():.1f}] descent it={t} new best {best_score:.6f}")
            else:
                no_improve += 1

        if no_improve >= patience:
            # small random jitter to escape plateaus
            jitter = rng.exponential(scale=0.003, size=n)
            a = a + jitter
            a = np.maximum(a, 0.0)
            a = a / a.sum()
            no_improve = 0

    return best_seq, best_score

def propose_candidate(seed: int = 42, budget_s: int = 1000) -> list[float]:
    """

            cand = _distribution_random_sparse(n, rng, density=0.006)
