def _adam_optimize(a0, deadline, best_seq, best_score, start_lr=0.1, seed_lr=0.5, lambda_start=1.0, lambda_end=20.0, eval_every=100):
    u = np.log(a0 + 1e-15)  # avoid log 0
    m = np.zeros_like(u)
    v = np.zeros_like(u)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    t = 0
    total_iters_est = 0.0 # not used
    n = a0.shape[0]
    while time.time() < deadline:
        t += 1
        a = np.exp(u)
        a = a / a.sum()
        # compute lambda scheduling
        # compute ratio based on progress: progress = (time.time() - start) / (deadline - start)
        # But start not passed; compute progress using proportion of iterations? We'll approximate using t.
        # We'll just use fixed lambda = lambda_end.
        lam = lambda_end  # we can later adjust
        c = _fft_self_convolve(a)
        max_c = c.max()
        # compute softmax weights:
        exp_vals = np.exp(lam * (c - max_c))
        sum_exp = exp_vals.sum()
        if sum_exp == 0.0:
            # fallback to argmax
            grad_a = np.zeros_like(a)
        else:
            p = exp_vals / sum_exp
            p_rev = p[::-1]
            conv_full = _fft_convolve_full(a, p_rev)
            idx = (2 * n - 2) - np.arange(n)
            grad_a = 2.0 * conv_full[idx]
        # compute gradient in u
        grad_u = a * (grad_a - np.dot(grad_a, a))
        # Adam update
        m = beta1 * m + (1 - beta1) * grad_u
        v = beta2 * v + (1 - beta2) * (grad_u ** 2)
        m_hat = m / (1 - beta1 ** t)
        v_hat = v / (1 - beta2 ** t)
        # step size: lr schedule decay (maybe 1/sqrt(t))
        lr = start_lr / np.sqrt(t)  # decaying learning rate
        u = u - lr * m_hat / (np.sqrt(v_hat) + eps)

        # evaluate every few iterations
        if t % eval_every == 0:
            # compute exact objective
            a = np.exp(u)
            a = a / a.sum()
            cur = _objective(a)
            if cur < best_score:
                best_score = cur
                best_seq = a.copy()
                print(f"[{time.time():.1f}] Adam it={t} new best {cur:.6f}")
    return best_seq, best_score

"""
step_function_search_v2.py

A 1000‑second heuristic for the non‑negative step‑function problem.

The search proceeds in three stages:

1. **Seed generation** – A diverse collection of probability vectors is
   created (uniform, arcsine, Hann, edge‑weighted, Generalised‑Normal,
   Dirichlet, sparse).  All vectors are normalised to sum = 1 and the
   best few (≈20) are retained.

2. **Adam‑based optimisation** – Each retained seed is polished
   with Adam (exponentiated‑gradient on the soft‑max of the convolution
   entries).  The gradient is the exact sub‑gradient of the maximal
   convolution entry and is transformed to the log‑softmax
   parametrisation, guaranteeing the simplex constraint.  A short run
   is performed on many seeds, then a longer run on the current global
   best.

3. **Upsampling & final polishing** – The best vector is linearly
   up‑sampled to double length (if time permits) and refined again with
   Adam, followed by a lightweight pair‑wise mass‑transfer simulated
   annealing to escape any remaining local optimum.

All work respects the 1000 s budget (2 s safety margin).  The function
``propose_candidate`` returns a plain ``list[float]`` ready for
``evaluate_sequence``.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Simplex utilities
# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Euclidean projection onto the simplex {x>=0, Σx=s}."""
    if s < 0:
        raise ValueError("simplex radius must be non‑negative")
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.where(u - cssv / (np.arange(1, len(v) + 1)) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Convolution helpers (FFT based, linear convolution)
# ----------------------------------------------------------------------
def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Full linear self‑convolution a*a using a real FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1) - 1).bit_length()
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[: 2 * n - 1]


def _fft_convolve_full(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution of two real vectors using FFT."""
    n = len(x) + len(y) - 1
    size = 1 << (n - 1).bit_length()
    X = np.fft.rfft(x, n=size)
    Y = np.fft.rfft(y, n=size)
    conv = np.fft.irfft(X * Y, n=size)
    return conv[:n]


# ----------------------------------------------------------------------
# Exact objective (scale‑invariant, sum=1 ⇒ denominator = 1)
# ----------------------------------------------------------------------
def _objective(a: np.ndarray) -> float:
    """Exact value of ``evaluate_sequence`` for a probability vector a."""
    n = a.shape[0]
    b = _fft_self_convolve(a)
    max_b = b.max()
    if max_b < 0:
        max_b = 0.0
    return float(2.0 * n * max_b)


# ----------------------------------------------------------------------
# Analytic families (all return a probability vector, sum=1)
# ----------------------------------------------------------------------
def _distribution_uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _distribution_arcsine(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    vals /= vals.sum()
    return vals


def _distribution_hann(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    w = 0.5 - 0.5 * np.cos(2.0 * np.pi * i / (n - 1))
    w_sum = w.sum()
    if w_sum == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w_sum
    return w


def _distribution_edge_weighted(n: int, p: float) -> np.ndarray:
    """Weight ∝ (i·j)^{‑p} where i,j are distances to the two edges."""
    i = np.arange(n, dtype=np.float64) + 0.5
    j = (n - np.arange(n, dtype=np.float64) - 0.5)
    w = (i * j) ** (-p)
    w_sum = w.sum()
    if w_sum == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w_sum
    return w


def _distribution_generalized_normal(n: int, q: float) -> np.ndarray:
    """Density ∝ exp(‑|x‑½|^{q}) on [0,1]."""
    i = np.arange(n, dtype=np.float64) + 0.5
    x = i / n - 0.5
    w = np.exp(- (np.abs(x) / 0.5) ** q)
    w_sum = w.sum()
    if w_sum == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w_sum
    return w


def _distribution_random_sparse(
    n: int, rng: np.random.Generator, density: float = 0.01
) -> np.ndarray:
    """Sparse random vector with a given density."""
    k = max(2, int(n * density))
    idx = rng.choice(n, size=k, replace=False)
    vals = np.zeros(n, dtype=np.float64)
    vals[idx] = rng.random(k)
    if vals.sum() == 0.0:
        vals[0] = 1.0
    vals /= vals.sum()
    return vals


def _mix_with_uniform(base: np.ndarray, lam: float) -> np.ndarray:
    """Convex combination with the uniform distribution."""
    n = base.shape[0]
    uniform = np.full(n, 1.0 / n, dtype=np.float64)
    return lam * uniform + (1.0 - lam) * base


# ----------------------------------------------------------------------
# Gradient helpers
# ----------------------------------------------------------------------
def _grad_of_max_conv(a: np.ndarray) -> tuple[np.ndarray, int]:
    """
    Sub‑gradient of the currently maximal convolution entry.
    Returns (grad, k) where k is the index attaining the maximum.
    """
    n = a.shape[0]
    conv = _fft_self_convolve(a)
    k = int(np.argmax(conv))

    grad = np.zeros_like(a)
    if k < n:
        grad[: k + 1] = 2.0 * a[k::-1]
    else:
        start = k - (n - 1)
        grad[start:] = 2.0 * a[start:][::-1]
    return grad, k


def _softmax_max_gradient(a: np.ndarray, lam: float) -> np.ndarray:
    """
    Gradient of the smooth (soft‑max) approximation of max(conv(a,a)).
    Uses the identity ∇S = Σ_k p_k ∇c_k with p_k = softmax(λ·c_k).
    """
    n = a.shape[0]
    c = _fft_self_convolve(a)
    c_max = c.max()
    # stable soft‑max
    exp_vals = np.exp(lam * (c - c_max))
    sum_exp = exp_vals.sum()
    if sum_exp == 0.0:
        # fallback to hard max gradient
        grad, _ = _grad_of_max_conv(a)
        return grad
    p = exp_vals / sum_exp               # shape (2n‑1,)
    # correlation a * p (as derived in the analysis)
    p_rev = p[::-1]
    conv_full = _fft_convolve_full(a, p_rev)   # length 3n‑2
    idx = (2 * n - 2) - np.arange(n)            # pick the right slice
    grad = 2.0 * conv_full[idx]
    return grad


# ----------------------------------------------------------------------
# Adam optimiser on the simplex (log‑softmax parametrisation)
# ----------------------------------------------------------------------
def _adam_optimize(
    a0: np.ndarray,
    deadline: float,
    best_seq: np.ndarray,
    best_score: float,
    lr0: float = 0.5,
    lam: float = 10.0,
    use_softmax: bool = False,
    eval_every: int = 50,
) -> tuple[np.ndarray, float]:
    """
    Adam optimisation on the exact objective.
    The variable is the log‑softmax log‑probabilities (u), guaranteeing
    non‑negativity and Σa=1 at every step.
    """
    eps = 1e-15
    u = np.log(a0 + eps)          # initialise logits
    m = np.zeros_like(u)
    v = np.zeros_like(u)
    beta1 = 0.9
    beta2 = 0.999
    t = 0
    n = a0.shape[0]

    while time.time() < deadline:
        t += 1
        # reconstruct a from logits
        a = np.exp(u)
        a_sum = a.sum()
        if a_sum == 0.0:
            a = np.full(n, 1.0 / n, dtype=np.float64)
        else:
            a /= a_sum

        # gradient of the objective w.r.t. a
        if use_softmax:
            grad_a = _softmax_max_gradient(a, lam)
        else:
            grad_a, _ = _grad_of_max_conv(a)

        # transform to gradient w.r.t. logits (softmax Jacobian)
        dot = np.dot(grad_a, a)
        grad_u = a * (grad_a - dot)

        # Adam moment updates
        m = beta1 * m + (1.0 - beta1) * grad_u
        v = beta2 * v + (1.0 - beta2) * (grad_u ** 2)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)

        # learning‑rate decay
        lr = lr0 / math.sqrt(t)

        # parameter update
        u = u - lr * m_hat / (np.sqrt(v_hat) + eps)

        # periodic evaluation
        if t % eval_every == 0:
            a = np.exp(u)
            a = a / a.sum()
            cur = _objective(a)
            if cur < best_score - 1e-14:
                best_score = cur
                best_seq = a.copy()
                print(f"[{time.time():.1f}] Adam it={t} new best {best_score:.6f}")

    return best_seq, best_score


# ----------------------------------------------------------------------
# Simulated‑annealing style pairwise mass transfers
# ----------------------------------------------------------------------
def _pairwise_annealing(
    a: np.ndarray,
    deadline: float,
    best_seq: np.ndarray,
    best_score: float,
    rng: np.random.Generator,
) -> tuple[np.ndarray, float]:
    """
    Randomly transfer a fraction of mass between two bins.
    Moves improving the objective are always accepted; early on
    deteriorating moves may be accepted with a temperature‑dependent probability.
    """
    n = a.shape[0]
    start_time = time.time()
    total_time = deadline - start_time
    while time.time() < deadline:
        src_candidates = np.where(a > 1e-12)[0]
        if src_candidates.size == 0:
            break
        i = rng.choice(src_candidates)
        j = rng.integers(n - 1)
        if j >= i:
            j += 1
        delta = a[i] * rng.random() * 0.4
        if delta < 1e-16:
            continue
        new_a = a.copy()
        new_a[i] -= delta
        new_a[j] += delta
        cur = _objective(new_a)

        elapsed = time.time() - start_time
        temperature = max(0.0, 1.0 - elapsed / total_time) * best_score * 0.01

        accept = False
        if cur < best_score - 1e-14:
            accept = True
        elif temperature > 0.0:
            prob = math.exp(-(cur - best_score) / temperature)
            if rng.random() < prob:
                accept = True

        if accept:
            a = new_a
            if cur < best_score - 1e-14:
                best_score = cur
                best_seq = a.copy()
                print(
                    f"[{time.time():.1f}] anneal i={i}→j={j} δ={delta:.3e} new best {best_score:.6f}"
                )
    return best_seq, best_score


# ----------------------------------------------------------------------
# Upsampling (linear interpolation while preserving the simplex)
# ----------------------------------------------------------------------
def _upsample(a: np.ndarray, new_n: int) -> np.ndarray:
    """Interpolate a probability vector to a larger length."""
    old_n = a.shape[0]
    if new_n == old_n:
        return a.copy()
    x_old = (np.arange(old_n, dtype=np.float64) + 0.5) / old_n
    x_new = (np.arange(new_n, dtype=np.float64) + 0.5) / new_n
    a_new = np.interp(x_new, x_old, a, left=0.0, right=0.0)
    a_new = np.maximum(a_new, 0.0)
    s = a_new.sum()
    if s == 0.0:
        a_new[:] = 1.0 / new_n
    else:
        a_new /= s
    return a_new
