"""
step_function_search_v2.py

A fresh 1000‑second optimiser for ``evaluate_sequence``.  The core ideas
are:

* **Rich seed pool** – several analytic families (uniform, arcsine,
  Hann, edge‑weighted, beta, generalized‑normal, sparse, Dirichlet)
  for a handful of lengths (≤ 16384).

* **Smoothed‑max gradient descent** – instead of the sub‑gradient of the
  max convolution entry we minimise the smooth approximation
  ``F_λ(a) = (1/λ)·log Σ_k exp(λ·(a*a)_k)``.  The soft‑max weight vector
  is obtained from the current autocorrelation and the gradient is a
  weighted correlation (computed with a single FFT).  The temperature
  λ is increased gradually so the algorithm first moves on a smooth
  landscape and later concentrates on the true maximum.

* **Exponential‑mirror updates** – the iterate stays in the simplex
  (non‑negative, unit sum) by an element‑wise exponential update followed
  by ℓ₁‑normalisation.

* **Multi‑phase refinement** – short smoothed runs on the best few
  seeds, a long run on the global best, an optional up‑sampling step,
  a final projected‑gradient polishing (sub‑gradient of the true max)
  and a light simulated‑annealing mass‑transfer finish.

The routine respects the 1000 s budget, prints timing / improvement
messages and finally returns a plain ``list[float]`` ready for
``evaluate_sequence``.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Simplex utilities
# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x≥0, Σx=s}."""
    if s < 0:
        raise ValueError("simplex radius must be non‑negative")
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.where(u - cssv / (np.arange(1, len(v) + 1)) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# FFT‑based convolution helpers
# ----------------------------------------------------------------------
def _fft_convolve(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full linear convolution of a and b (real inputs)."""
    n = a.shape[0] + b.shape[0] - 1
    size = 1 << ((n - 1).bit_length())
    A = np.fft.rfft(a, n=size)
    B = np.fft.rfft(b, n=size)
    conv = np.fft.irfft(A * B, n=size)
    return conv[:n]


def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Full linear self‑convolution a*a using a real FFT."""
    return _fft_convolve(a, a)


# ----------------------------------------------------------------------
# Exact objective (scale‑invariant – sum(a)=1 ⇒ denominator = 1)
# ----------------------------------------------------------------------
def _objective(a: np.ndarray) -> float:
    """Exact value of ``evaluate_sequence`` for a probability vector."""
    n = a.shape[0]
    b = _fft_self_convolve(a)
    max_b = b.max()
    return float(2.0 * n * max_b)


# ----------------------------------------------------------------------
# Analytic families (all return a probability vector, Σ=1)
# ----------------------------------------------------------------------
def _distribution_uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _distribution_arcsine(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    vals /= vals.sum()
    return vals


def _distribution_hann(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    w = 0.5 - 0.5 * np.cos(2.0 * np.pi * i / (n - 1))
    w_sum = w.sum()
    if w_sum == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w_sum
    return w


def _distribution_edge_weighted(n: int, p: float) -> np.ndarray:
    """Weight ∝ (i·j)^(-p) where i,j are distances to the two edges."""
    i = np.arange(n, dtype=np.float64) + 0.5
    j = (n - np.arange(n, dtype=np.float64) - 0.5)
    w = (i * j) ** (-p)
    w_sum = w.sum()
    if w_sum == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w_sum
    return w


def _distribution_generalized_normal(n: int, q: float) -> np.ndarray:
    """Density ∝ exp(-|x-½|^q) on [0,1]."""
    i = np.arange(n, dtype=np.float64) + 0.5
    x = i / n - 0.5
    w = np.exp(- (np.abs(x) / 0.5) ** q)
    w_sum = w.sum()
    if w_sum == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w_sum
    return w


def _distribution_beta(n: int, alpha: float, beta: float) -> np.ndarray:
    """Beta‑type density ∝ i^(α‑1)·(n‑i)^(β‑1) on the index grid."""
    i = np.arange(n, dtype=np.float64) + 0.5
    w = (i ** (alpha - 1.0)) * ((n - i) ** (beta - 1.0))
    w_sum = w.sum()
    if w_sum == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w_sum
    return w


def _distribution_random_sparse(
    n: int, rng: np.random.Generator, density: float = 0.01
) -> np.ndarray:
    """Sparse random vector with a given density."""
    k = max(2, int(n * density))
    idx = rng.choice(n, size=k, replace=False)
    vals = np.zeros(n, dtype=np.float64)
    vals[idx] = rng.random(k)
    if vals.sum() == 0.0:
        vals[0] = 1.0
    vals /= vals.sum()
    return vals


def _mix_with_uniform(base: np.ndarray, lam: float) -> np.ndarray:
    """Convex combination with the uniform distribution."""
    n = base.shape[0]
    uniform = np.full(n, 1.0 / n, dtype=np.float64)
    return lam * uniform + (1.0 - lam) * base


# ----------------------------------------------------------------------
# Smoothed‑max gradient descent
# ----------------------------------------------------------------------
def _smooth_gradient_descent(
    a0: np.ndarray,
    deadline: float,
    best_seq: np.ndarray,
    best_score: float,
    lambda_max: float = 3000.0,
    eta0: float = 0.6,
    eval_every: int = 80,
) -> tuple[np.ndarray, float]:
    """
    Mirror‑descent on the smooth approximation
        F_λ(a) = (1/λ)·log Σ_k exp(λ·(a*a)_k).

    ``lambda_max`` is the asymptotic temperature; during the run the
    actual λ(t) = λ_max·(1 - exp(-t/τ)) with τ≈2000.
    """
    a = a0.copy()
    n = a.shape[0]
    t = 0
    while time.time() < deadline:
        t += 1
        # 1) autocorrelation
        b = _fft_self_convolve(a)                # length 2n‑1
        # 2) soft‑max weights
        b_max = b.max()
        # gradually increase λ
        lam = lambda_max * (1.0 - math.exp(-t / 2000.0))
        # use b‑max shift for stability: all arguments ≤ 0
        exp_arg = lam * (b - b_max)
        w = np.exp(exp_arg)
        w_sum = w.sum()
        if w_sum == 0.0:
            # fall back to one‑hot at the maximal index
            w = np.zeros_like(b)
            w[np.argmax(b)] = 1.0
            w_sum = 1.0
        p = w / w_sum                           # probabilities, sum=1

        # 3) gradient of the smoothed objective:
        #    ∇F = 2·(p ⋆ a)[i] where ⋆ denotes cross‑correlation.
        a_rev = a[::-1]                         # reversed a
        conv_full = _fft_convolve(p, a_rev)     # length 3n‑2
        grad = 2.0 * conv_full[n - 1 : n - 1 + n]  # slice of length n

        # 4) mirror‑descent (exponentiated‑gradient) update
        eta = eta0 / math.sqrt(t)
        a = a * np.exp(-eta * grad)
        s = a.sum()
        if s == 0.0:
            a = np.full(n, 1.0 / n, dtype=np.float64)
        else:
            a = a / s

        # 5) occasional evaluation
        if t % eval_every == 0:
            cur = _objective(a)
            if cur < best_score - 1e-14:
                best_score = cur
                best_seq = a.copy()
                print(
                    f"[{time.time():.1f}] smooth GD it={t} new best {best_score:.6f}"
                )
    return best_seq, best_score


# ----------------------------------------------------------------------
# Sub‑gradient (max‑only) descent – used for final polishing
# ----------------------------------------------------------------------
def _grad_of_max_conv(a: np.ndarray) -> tuple[np.ndarray, int]:
    """Gradient of the currently maximal convolution entry."""
    n = a.shape[0]
    conv = _fft_self_convolve(a)
    k = int(np.argmax(conv))

    grad = np.zeros_like(a)
    if k < n:
        grad[: k + 1] = 2.0 * a[k::-1]
    else:
        start = k - (n - 1)
        grad[start:] = 2.0 * a[start:][::-1]
    return grad, k


def _gradient_descent(
    a: np.ndarray,
    deadline: float,
    best_seq: np.ndarray,
    best_score: float,
    eta0: float = 0.9,
    eval_every: int = 60,
) -> tuple[np.ndarray, float]:
    """Mirror‑descent on the true max‑objective (sub‑gradient)."""
    n = a.shape[0]
    t = 0
    while time.time() < deadline:
        t += 1
        grad, _ = _grad_of_max_conv(a)
        eta = eta0 / math.sqrt(t)
        a = a * np.exp(-eta * grad)
        s = a.sum()
        if s == 0.0:
            a = np.full(n, 1.0 / n, dtype=np.float64)
        else:
            a = a / s

        if t % eval_every == 0:
            cur = _objective(a)
            if cur < best_score - 1e-14:
                best_score = cur
                best_seq = a.copy()
                print(f"[{time.time():.1f}] max‑GD it={t} new best {best_score:.6f}")
    return best_seq, best_score


# ----------------------------------------------------------------------
# Simulated‑annealing style pairwise mass transfers
# ----------------------------------------------------------------------
def _pairwise_annealing(
    a: np.ndarray,
    deadline: float,
    best_seq: np.ndarray,
    best_score: float,
    rng: np.random.Generator,
) -> tuple[np.ndarray, float]:
    """Randomly shift a fraction of mass between two bins."""
    n = a.shape[0]
    start_time = time.time()
    total_time = deadline - start_time
    while time.time() < deadline:
        src_candidates = np.where(a > 1e-12)[0]
        if src_candidates.size == 0:
            break
        i = rng.choice(src_candidates)

        j = rng.integers(n - 1)
        if j >= i:
            j += 1

        delta = a[i] * rng.random() * 0.4
        if delta < 1e-16:
            continue

        new_a = a.copy()
        new_a[i] -= delta
        new_a[j] += delta

        cur = _objective(new_a)

        elapsed = time.time() - start_time
        temperature = max(0.0, 1.0 - elapsed / total_time) * best_score * 0.01

        if cur < best_score - 1e-14:
            accept = True
        else:
            accept = False
            if temperature > 0.0:
                prob = math.exp(-(cur - best_score) / temperature)
                if rng.random() < prob:
                    accept = True

        if accept:
            a = new_a
            if cur < best_score - 1e-14:
                best_score = cur
                best_seq = a.copy()
                print(
                    f"[{time.time():.1f}] anneal i={i}→j={j} δ={delta:.3e} new best {best_score:.6f}"
                )
    return best_seq, best_score


# ----------------------------------------------------------------------
# Upsampling (linear interpolation while preserving the simplex)
# ----------------------------------------------------------------------
def _upsample(a: np.ndarray, new_n: int) -> np.ndarray:
    """Interpolate a probability vector to a larger length."""
    old_n = a.shape[0]
    if new_n == old_n:
        return a.copy()
    x_old = (np.arange(old_n, dtype=np.float64) + 0.5) / old_n
    x_new = (np.arange(new_n, dtype=np.float64) + 0.5) / new_n
    a_new = np.interp(x_new, x_old, a, left=0.0, right=0.0)
    a_new = np.maximum(a_new, 0.0)
    s = a_new.sum()
    if s == 0.0:
        a_new[:] = 1.0 / new_n
    else:
        a_new /= s
    return a_new


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: int = 1000) -> list[float]:
    """
    Search for a non‑negative step function minimising the metric used by
    ``evaluate_sequence``.  The routine respects a 1000 s wall‑clock budget.
    """
    rng = np.random.default_rng(seed)
    start = time.time()
    deadline = start + budget_s - 2.0  # safety margin

    overall_best_seq: np.ndarray | None = None
    overall_best_score = float("inf")
    cand_pool: list[tuple[float, np.ndarray]] = []

    # ------------------------------------------------------------------
    # 1.  Seed generation (moderate lengths)
    # ------------------------------------------------------------------
    lengths = [2048, 4096, 8192, 16384]  # increasing flexibility
    # parameters for the parametric families
    p_vals = np.concatenate((
        np.logspace(-3.0, -0.5, 12, base=10),   # 0.001 → 0.316
        np.logspace(-0.5, 0.0, 12, base=10),    # 0.316 → 1
        np.linspace(0.1, 2.0, 12),               # linear region
        np.logspace(0.0, 0.7, 8, base=10)       # 1 → 5
    ))
    lam_vals = np.linspace(0.0, 1.0, 9)          # mixing with uniform
    beta_alphas = np.concatenate((
        np.logspace(-1.5, -0.2, 7),
        np.linspace(0.3, 3.0, 7)
    ))
    beta_betas = beta_alphas.copy()            # symmetric grid
    q_vals = np.linspace(0.3, 4.0, 12)
    dirichlet_alphas_log = np.linspace(-2.0, 1.0, 10)  # α ∈ [0.01,10]

    for n in lengths:
        if time.time() >= deadline:
            break
        print(f"[{time.time():.1f}] Seeding length = {n}")

        # --- deterministic families -------------------------------------------------
        for base in (
            _distribution_uniform(n),
            _distribution_arcsine(n),
            _distribution_hann(n),
        ):
            sc = _objective(base)
            cand_pool.append((sc, base.copy()))
            if sc < overall_best_score:
                overall_best_score = sc
                overall_best_seq = base.copy()
                print(f"[{time.time():.1f}] new global best (det) {sc:.6f}")

        # --- edge‑weighted + mixing -------------------------------------------------
        for p in p_vals:
            if time.time() >= deadline:
                break
            base = _distribution_edge_weighted(n, float(p))
            for lam in lam_vals:
                if time.time() >= deadline:
                    break
                cand = _mix_with_uniform(base, float(lam))
                sc = _objective(cand)
                cand_pool.append((sc, cand.copy()))
                if sc < overall_best_score:
                    overall_best_score = sc
                    overall_best_seq = cand.copy()
                    print(f"[{time.time():.1f}] new best edge p={p:.3g} λ={lam:.2f} {sc:.6f}")

        # --- beta family ------------------------------------------------------------
        for alpha in beta_alphas:
            for beta in beta_betas:
                if time.time() >= deadline:
                    break
                cand = _distribution_beta(n, float(alpha), float(beta))
                sc = _objective(cand)
                cand_pool.append((sc, cand.copy()))
                if sc < overall_best_score:
                    overall_best_score = sc
                    overall_best_seq = cand.copy()
                    print(f"[{time.time():.1f}] new best beta α={alpha:.2g} β={beta:.2g} {sc:.6f}")

        # --- generalized normal ------------------------------------------------------
        for q in q_vals:
            if time.time() >= deadline:
                break
            cand = _distribution_generalized_normal(n, float(q))
            sc = _objective(cand)
            cand_pool.append((sc, cand.copy()))
            if sc < overall_best_score:
                overall_best_score = sc
                overall_best_seq = cand.copy()
                print(f"[{time.time():.1f}] new best gn q={q:.2f} {sc:.6f}")

        # --- random Dirichlet --------------------------------------------------------
        for _ in range(12):
            if time.time() >= deadline:
                break
            log_alpha = rng.uniform(-2.0, 1.0)
            alpha = 10 ** log_alpha
            cand = rng.dirichlet(np.full(n, alpha))
            sc = _objective(cand)
            cand_pool.append((sc, cand.copy()))
            if sc < overall_best_score:
                overall_best_score = sc
                overall_best_seq = cand.copy()
                print(f"[{time.time():.1f}] new best Dirichlet α={alpha:.3g} {sc:.6f}")

        # --- sparse random -----------------------------------------------------------
        for _ in range(6):
            if time.time() >= deadline:
                break
            cand = _distribution_random_sparse(n, rng, density=0.02)
            sc = _objective(cand)
            cand_pool.append((sc, cand.copy()))
            if sc < overall_best_score:
                overall_best_score = sc
                overall_best_seq = cand.copy()
                print(f"[{time.time():.1f}] new best sparse {sc:.6f}")

        # --- retain only the top seeds (30) -----------------------------------------
        cand_pool.sort(key=lambda x: x[0])
        cand_pool = cand_pool[:30]

    # ------------------------------------------------------------------
    # 2.  Incorporate a previously found construction if present
    # ------------------------------------------------------------------
    try:
        prev = np.asarray(height_sequence_1, dtype=np.float64)  # type: ignore[name-defined]
        if prev.ndim == 1:
            prev = prev / prev.sum()
            sc = _objective(prev)
            if sc < overall_best_score:
                overall_best_score = sc
                overall_best_seq = prev.copy()
                print(f"[{time.time():.1f}] previous best score={sc:.6f}")
            cand_pool.append((sc, prev.copy()))
    except Exception:
        pass

    # ------------------------------------------------------------------
    # 3.  Ensure we have a non‑empty result
    # ------------------------------------------------------------------
    if overall_best_seq is None:
        n = 4096
        overall_best_seq = _distribution_uniform(n)
        overall_best_score = _objective(overall_best_seq)
        print(f"[{time.time():.1f}] fallback uniform {overall_best_score:.6f}")

    # ------------------------------------------------------------------
    # 4.  Short smoothed‑gradient polishing on all top seeds
    # ------------------------------------------------------------------
    remaining = deadline - time.time()
    if remaining > 25.0 and cand_pool:
        per_candidate = max(0.3, remaining * 0.35 / max(len(cand_pool), 1))
        print(f"[{time.time():.1f}] short smooth GD on {len(cand_pool)} seeds (~{per_candidate:.2f}s each)")
        for idx, (score, a0) in enumerate(cand_pool):
            if time.time() >= deadline:
                break
            sub_dead = time.time() + per_candidate
            a_opt, sc_opt = _smooth_gradient_descent(
                a0.copy(),
                sub_dead,
                best_seq=overall_best_seq,
                best_score=overall_best_score,
                lambda_max=2500.0,
                eta0=0.8,
                eval_every=40,
            )
            if sc_opt < overall_best_score - 1e-12:
                overall_best_score = sc_opt
                overall_best_seq = a_opt.copy()
                print(f"[{time.time():.1f}] seed {idx} improved best to {overall_best_score:.6f}")

    # ------------------------------------------------------------------
    # 5.  Longer smoothed‑gradient descent on the current global best
    # ------------------------------------------------------------------
    if time.time() < deadline:
        sub_dead = time.time() + (deadline - time.time()) * 0.35
        a_opt, sc_opt = _smooth_gradient_descent(
            overall_best_seq.copy(),
            sub_dead,
            best_seq=overall_best_seq,
            best_score=overall_best_score,
            lambda_max=3500.0,
            eta0=0.5,
            eval_every=60,
        )
        if sc_opt < overall_best_score - 1e-12:
            overall_best_score = sc_opt
            overall_best_seq = a_opt.copy()
            print(f"[{time.time():.1f}] long smooth GD improved best to {overall_best_score:.6f}")

    # ------------------------------------------------------------------
    # 6.  Optional up‑sampling & further polishing
    # ------------------------------------------------------------------
    if time.time() < deadline:
        cur_n = overall_best_seq.shape[0]
        target_n = cur_n * 2
        if target_n <= 64000:
            print(f"[{time.time():.1f}] upsampling {cur_n} → {target_n}")
            up = _upsample(overall_best_seq, target_n)
            up_score = _objective(up)
            if up_score < overall_best_score:
                overall_best_seq = up
                overall_best_score = up_score
                print(f"[{time.time():.1f}] upsampled new best {overall_best_score:.6f}")

            # short smooth GD on upsampled vector
            sub_dead = time.time() + (deadline - time.time()) * 0.2
            a_opt, sc_opt = _smooth_gradient_descent(
                overall_best_seq.copy(),
                sub_dead,
                best_seq=overall_best_seq,
                best_score=overall_best_score,
                lambda_max=3000.0,
                eta0=0.6,
                eval_every=50,
            )
            if sc_opt < overall_best_score - 1e-12:
                overall_best_seq = a_opt.copy()
                overall_best_score = sc_opt
                print(f"[{time.time():.1f}] post‑upsample smooth GD new best {overall_best_score:.6f}")

    # ------------------------------------------------------------------
    # 7.  Final max‑gradient polishing (sub‑gradient) for a few seconds
    # ------------------------------------------------------------------
    if time.time() < deadline:
        sub_dead = time.time() + min(8.0, deadline - time.time())
        a_opt, sc_opt = _gradient_descent(
            overall_best_seq.copy(),
            sub_dead,
            best_seq=overall_best_seq,
            best_score=overall_best_score,
            eta0=0.7,
            eval_every=30,
        )
        if sc_opt < overall_best_score - 1e-12:
            overall_best_seq = a_opt.copy()
            overall_best_score = sc_opt
            print(f"[{time.time():.1f}] final max‑GD new best {overall_best_score:.6f}")

    # ------------------------------------------------------------------
    # 8.  Final simulated‑annealing polishing
    # ------------------------------------------------------------------
    if time.time() < deadline:
        print(f"[{time.time():.1f}] final annealing phase")
        a_opt, sc_opt = _pairwise_annealing(
            overall_best_seq.copy(),
            deadline,
            best_seq=overall_best_seq,
            best_score=overall_best_score,
            rng=rng,
        )
        if sc_opt < overall_best_score - 1e-12:
            overall_best_seq = a_opt.copy()
            overall_best_score = sc_opt
            print(f"[{time.time():.1f}] after annealing new best {overall_best_score:.6f}")

    # ------------------------------------------------------------------
    print(
        f"Final best score (objective=2·n·max(conv)): {overall_best_score:.6f} "
        f"(length={len(overall_best_seq)})"
    )
    return overall_best_seq.tolist()
