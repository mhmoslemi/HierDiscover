"""
Optimization of a non‑negative step function for the evaluate_sequence() metric.

The algorithm consists of four stages:

1️⃣  **Rich analytic family sweep** – a wide collection of parametrised
    distributions (uniform, arcsine, edge‑weighted, exponential‑edge,
    sine, cosine, quadratic, Beta, random sparse) is generated for a
    handful of lengths.  The best ~20 seeds (lowest exact objective)
    are kept.

2️⃣  **Accelerated mirror‑descent (Adam‑style EG)** – each seed is refined
    with an exponentiated‑gradient update that uses an Adam momentum term.
    The sub‑gradient is assembled from the *top‑k* convolution entries
    (default k=10) instead of only the absolute maxima; this gives a
    smoother descent direction and frequently pushes the objective below
    the previous record.

3️⃣  **Simulated‑annealing mass‑transfer** – starting from the current
    global best, random pairwise‑mass moves are accepted with a
    Metropolis probability that slowly cools.  This helps escape shallow
    plateaus that the deterministic mirror‑descent can miss.

4️⃣  **Greedy polishing** – a final hill‑climber swaps a fraction of mass
    between two indices whenever it strictly improves the exact objective.

All work is bounded by the supplied time budget (default 1000 s); the
function returns the best sequence found as a plain Python list.
"""

import time

# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Euclidean projection onto {x ≥ 0, Σx = s}."""
    if s < 0:
        raise ValueError("Simplex radius must be non‑negative")
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.where(u - cssv / (np.arange(1, len(v) + 1)) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    return np.maximum(v - theta, 0.0)


# ----------------------------------------------------------------------
def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Full self‑convolution a*a using a real FFT (fast)."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1) - 1).bit_length()   # next power of two
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[: 2 * n - 1]


# ----------------------------------------------------------------------
def _objective(a: np.ndarray) -> float:
    """
    Exact objective  2·n·max(conv(a,a))  for a vector with Σa = 1.
    """
    n = a.shape[0]
    b = _fft_self_convolve(a)
    return float(2.0 * n * b.max())

# Analytic families ----------------------------------------------------
def _distribution_uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _distribution_arcsine(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    vals /= vals.sum()
    return vals


def _distribution_edge_weighted(n: int, p: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64) + 0.5
    j = (n - np.arange(n, dtype=np.float64) - 0.5)
    w = (i * j) ** (-p)
    w_sum = w.sum()
    if w_sum == 0.0:
        return np.full(n, 1.0 / n, dtype=np.float64)
    w /= w_sum
    return w


def _distribution_exponential_edge(n: int, c: float) -> np.ndarray:
    """w_i ∝ exp(-c·i) + exp(-c·(n‑1‑i))   (c>0)."""
    i = np.arange(n, dtype=np.float64)
    w = np.exp(-c * i) + np.exp(-c * (n - 1 - i))
    w /= w.sum()
    return w


def _distribution_sine(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64) + 0.5
    w = np.sin(np.pi * i / n)
    w /= w.sum()
    return w


def _distribution_cosine(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64) + 0.5
    w = 1.0 - np.cos(np.pi * i / n)
    w /= w.sum()
    return w


def _distribution_quadratic(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    x = (2.0 * i) / (n - 1) - 1.0
    w = 1.0 - x * x
    w = np.clip(w, 0.0, None)
    w /= w.sum()
    return w


def _distribution_beta(n: int, alpha: float, beta: float) -> np.ndarray:
    """Beta‑like density on the grid i∈{0,…,n‑1}."""
    i = np.arange(n, dtype=np.float64) + 0.5
    x = i / n
    # avoid overflow for extreme parameters
    w = np.exp((alpha - 1) * np.log(x) + (beta - 1) * np.log(1 - x))
    w[np.isnan(w)] = 0.0
    s = w.sum()
    if s == 0.0:
        return np.full(n, 1.0 / n, dtype=np.float64)
    w /= s
    return w


def _distribution_random_sparse(
    n: int, rng: np.random.Generator, density: float = 0.01
) -> np.ndarray:
    """Sparse random vector with a prescribed density of non‑zeros."""
    k = max(2, int(n * density))
    idx = rng.choice(n, size=k, replace=False)
    vals = np.zeros(n, dtype=np.float64)
    vals[idx] = rng.random(k)
    s = vals.sum()
    if s == 0.0:
        vals[0] = 1.0
        s = 1.0
    vals /= s
    return vals


def _mix_with_uniform(base: np.ndarray, lam: float) -> np.ndarray:
    """Convex combination λ·uniform + (1‑λ)·base."""
    n = base.shape[0]
    uniform = np.full(n, 1.0 / n, dtype=np.float64)
    return lam * uniform + (1.0 - lam) * base


# ----------------------------------------------------------------------
def _gradient_topk(a: np.ndarray, conv: np.ndarray, topk: int) -> np.ndarray:
    """
    Build a (sub)gradient from the `topk` largest convolution entries.
    For each chosen index k we add the contribution 2·a_{k‑i} to grad_i.
    """
    n = a.shape[0]
    # indices of the `topk` largest entries (ties are fine)
    idxs = np.argpartition(-conv, topk - 1)[:topk]
    grad = np.zeros_like(a)
    for idx in idxs:
        i_start = max(0, idx - (n - 1))
        i_end = min(n - 1, idx)
        i = np.arange(i_start, i_end + 1, dtype=np.int64)
        j = idx - i
        # contribution 2 * a_j to each i
        grad[i] += 2.0 * a[j]
    return grad

def _refine_adam_exponential(
    a0: np.ndarray,
    deadline: float,
    best_seq: np.ndarray,
    best_score: float,
    topk: int = 10,
    lr0: float = 0.6,
    beta1: float = 0.9,
    beta2: float = 0.999,
    eval_every: int = 80,
) -> tuple[np.ndarray, float]:
    """
    Exponentiated‑gradient descent with Adam‑style momentum.
    Uses the top‑k convolution entries to form a smoother sub‑gradient.
    """
    a = a0.copy()
    n = a.shape[0]
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    eps = 1e-16
    t = 0
    while time.time() < deadline:
        t += 1
        conv = _fft_self_convolve(a)
        max_val = conv.max()

        # gradient from top‑k entries (including the true maximum)
        grad = _gradient_topk(a, conv, topk)

        # Adam moments
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)

        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)

        # exponentiated update in log‑space
        step = lr0 / math.sqrt(t)
        log_a = np.log(a + eps) - step * (m_hat / (np.sqrt(v_hat) + eps))
        a = np.exp(log_a)

        # renorm to simplex
        s = a.sum()
        if s <= 0.0:
            a = np.full(n, 1.0 / n, dtype=np.float64)
        else:
            a /= s

        if t % eval_every == 0:
            cur = 2.0 * n * max_val
            if cur < best_score - 1e-14:
                best_score = cur
                best_seq = a.copy()
                print(
                    f"[{time.time():.1f}] Adam‑EG it={t} new global best {best_score:.6f}"
                )
    return best_seq, best_score

def _anneal_mass_transfer(
    a0: np.ndarray,
    deadline: float,
    best_seq: np.ndarray,
    best_score: float,
    start_temp: float = 0.02,
    end_temp: float = 0.0005,
    max_frac: float = 0.45,
) -> tuple[np.ndarray, float]:
    """
    Simulated‑annealing style random mass transfers.
    A move i → j transfers a random fraction (≤max_frac) of a[i].
    Acceptance follows a Metropolis rule with a temperature that decays
    from `start_temp` to `end_temp`.
    """
    a = a0.copy()
    n = a.shape[0]
    start = time.time()
    while time.time() < deadline:
        elapsed = time.time() - start
        frac = elapsed / (deadline - start)
        T = start_temp * ((end_temp / start_temp) ** frac)  # exponential cooling

        # choose source (must have some mass) and distinct target
        src_candidates = np.where(a > 1e-14)[0]
        if src_candidates.size == 0:
            break
        i = np.random.choice(src_candidates)
        j = np.random.randint(n - 1)
        if j >= i:
            j += 1

        delta = a[i] * np.random.rand() * max_frac
        if delta < 1e-15:
            continue

        new_a = a.copy()
        new_a[i] -= delta
        new_a[j] += delta
        cur = _objective(new_a)

        if cur < best_score - 1e-14:
            # always accept strict improvement
            a = new_a
            best_score = cur
            best_seq = new_a.copy()
            print(
                f"[{time.time():.1f}] anneal improve i={i}→j={j} δ={delta:.3g} new best {best_score:.6f}"
            )
        else:
            # Metropolis acceptance
            if np.random.rand() < math.exp(-(cur - best_score) / (T + 1e-12)):
                a = new_a
    return best_seq, best_score

def _pairwise_swap(
    a0: np.ndarray,
    deadline: float,
    best_seq: np.ndarray,
    best_score: float,
    max_trials: int | None = None,
) -> tuple[np.ndarray, float]:
    """
    Deterministic hill‑climber: move a random fraction of mass from one index
    to another, keeping the move only if it strictly improves the exact
    objective.
    """
    a = a0.copy()
    n = a.shape[0]
    trials = 0
    while time.time() < deadline:
        if max_trials is not None and trials >= max_trials:
            break
        trials += 1

        src_candidates = np.where(a > 1e-14)[0]
        if src_candidates.size == 0:
            break
        i = np.random.choice(src_candidates)
        j = np.random.randint(n - 1)
        if j >= i:
            j += 1

        delta = a[i] * np.random.rand() * 0.45   # up to 45 % of source mass
        if delta < 1e-15:
            continue

        new_a = a.copy()
        new_a[i] -= delta
        new_a[j] += delta

        cur = _objective(new_a)
        if cur < best_score - 1e-14:
            a = new_a
            best_score = cur
            best_seq = a.copy()
            print(
                f"[{time.time():.1f}] swap i={i}→j={j} δ={delta:.6g} new best {best_score:.6f}"
            )
    return best_seq, best_score

        # take the best 6 seeds for deeper optimisation
        seeds = [seq for _, seq in sorted(top_candidates, key=lambda x: x[0])[:6]]
        per_seed = remaining / max(1, len(seeds))
        print(f"[{time.time():.1f}] start Adam‑EG refinement on {len(seeds)} seeds")
        for idx, seed_seq in enumerate(seeds):
            if time.time() >= deadline:
                break
            sg_deadline = time.time() + per_seed
            a_opt, sc_opt = _refine_adam_exponential(
                seed_seq,
                sg_deadline,
                best_seq=overall_best_seq,
                best_score=overall_best_score,
                topk=10,
                lr0=0.6,
                eval_every=80,
            )
            if sc_opt < overall_best_score - 1e-12:
                overall_best_score, overall_best_seq = sc_opt, a_opt.copy()
                print(
                    f"[{time.time():.1f}] after Adam‑EG seed {idx} new global best {overall_best_score:.6f}"
                )

    # ------------------------------------------------------------------
    # 5.  Simulated‑annealing mass‑transfer polishing
    # ------------------------------------------------------------------
    if time.time() < deadline:
        print(f"[{time.time():.1f}] annealing phase")
        a_opt, sc_opt = _anneal_mass_transfer(
            overall_best_seq.copy(),
            deadline,
            best_seq=overall_best_seq,
            best_score=overall_best_score,
            start_temp=0.02,
            end_temp=0.0005,
        )
        if sc_opt < overall_best_score - 1e-12:
            overall_best_score, overall_best_seq = sc_opt, a_opt.copy()
            print(
                f"[{time.time():.1f}] after annealing new global best {overall_best_score:.6f}"
            )

    # ------------------------------------------------------------------
    # 6.  Final greedy pairwise polishing (hill‑climbing)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        print(f"[{time.time():.1f}] final greedy swap polishing")
        a_opt, sc_opt = _pairwise_swap(
            overall_best_seq.copy(),
            deadline,
            best_seq=overall_best_seq,
            best_score=overall_best_score,
        )
        if sc_opt < overall_best_score - 1e-12:
            overall_best_score, overall_best_seq = sc_opt, a_opt.copy()
            print(f"[{time.time():.1f}] after final swaps new best {overall_best_score:.6f}")

    # ------------------------------------------------------------------
    print(
        f"Final best objective (2·n·max(conv)) = {overall_best_score:.6f} (length={len(overall_best_seq)})"
    )
    # Return a plain Python list – evaluate_sequence will re‑normalise internally.
    return overall_best_seq.tolist()
