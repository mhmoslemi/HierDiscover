def _solve_lp_max_sum(f, M):
    n = len(f)
    m = 2 * n - 1
    # Build A matrix for constraints A @ g <= M
    # Efficient building:
    A = np.zeros((m, n), dtype=np.float64)
    # we can fill columnwise
    for i in range(n):
        A[i:i+n, i] = f
    # Use linprog
    # Objective: maximize sum(g) => minimize -sum(g)
    c = -np.ones(n, dtype=np.float64)
    bounds = [(0, None) for _ in range(n)]
    # Use method='highs' (fast)
    result = scipy.optimize.linprog(c, A_ub=A, b_ub=np.full(m, M), bounds=bounds, method='highs')
    if result.success:
        # Clip tiny negative values due to numeric tolerance
        g = np.maximum(result.x, 0.0)
        return g
    else:
        # return None
        return None

"""
Linear‑programming based alternating optimisation for the max‑convolution
objective.

- Generate a handful of analytic seeds (edge‑weighted, arcsine, beta,
  random Dirichlet) for several lengths (≈300‑800).
- For each seed run a few iterations of the classic two‑step
  optimisation:
    1. Compute M = max (a * a).
    2. Solve a linear programme: maximise Σ g_i  subject to
       (a * g)_k ≤ M  for all k,   g_i ≥ 0.
    3. Rescale g to the same ℓ₁‑norm as a and mix  a←(1‑t)a+tg  with a
       moderate step‑size t (back‑tracking guarantees that the new
       max‑convolution does not increase).
- Keep the globally best vector (measured by 2·n·max(conv)/∥a∥₁², which is
  exactly the value returned by ``evaluate_sequence`` up to the final
  normalisation).
- The whole search respects the 1000 s budget and returns a plain
  Python list.
"""

import math
import numpy as np

# Optional: a fast linear‑programming solver (SciPy).  If SciPy is not
# available we fall back to the previous sub‑gradient routine.

# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x≥0, Σx = s}."""
    if s < 0:
        raise ValueError("simplex radius must be non‑negative")
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho_arr = np.where(u - cssv / (np.arange(1, len(v) + 1)) > 0)[0]
    if rho_arr.size == 0:
        theta = 0.0
    else:
        rho = rho_arr[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a via a real FFT (length 2·n‑1)."""
    n = a.shape[0]
    need = 2 * n - 1
    size = 1 << ((need - 1).bit_length())
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[:need]


# ----------------------------------------------------------------------
def _objective(a: np.ndarray) -> float:
    """Exact value 2·n·max(conv(a,a)) for a with Σa = 1."""
    n = a.shape[0]
    max_b = _fft_self_convolve(a).max()
    return 2.0 * n * max_b

# Analytic families (same as in the reference code) --------------------
def _distribution_uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _distribution_arcsine(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    vals /= vals.sum()
    return vals


def _distribution_beta_sym(n: int, p: float) -> np.ndarray:
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    w = np.exp((p - 1) * np.log(x) + (p - 1) * np.log(1.0 - x))
    if w.sum() == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _distribution_beta_asym(n: int, a: float, b: float) -> np.ndarray:
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    w = np.exp((a - 1) * np.log(x) + (b - 1) * np.log(1.0 - x))
    if w.sum() == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _distribution_edge_weighted(n: int, p: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64) + 0.5
    j = (n - np.arange(n, dtype=np.float64) - 0.5)
    w = (i * j) ** (-p)
    if w.sum() == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _distribution_sine_weighted(n: int, p: float) -> np.ndarray:
    theta = np.pi * (np.arange(n, dtype=np.float64) + 0.5) / n
    w = np.sin(theta) ** p
    if w.sum() == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _mix_with_uniform(base: np.ndarray, lam: float) -> np.ndarray:
    n = base.shape[0]
    return lam * np.full(n, 1.0 / n, dtype=np.float64) + (1.0 - lam) * base


def _distribution_random_sparse(
    n: int, rng: np.random.Generator, density: float = 0.01
) -> np.ndarray:
    k = max(2, int(n * density))
    idx = rng.choice(n, size=k, replace=False)
    vals = np.zeros(n, dtype=np.float64)
    vals[idx] = rng.random(k)
    if vals.sum() == 0.0:
        vals[0] = 1.0
    vals /= vals.sum()
    return vals


# ----------------------------------------------------------------------
# Linear‑programming utilities -----------------------------------------
def _conv_matrix(f: np.ndarray) -> np.ndarray:
    """
    Build the matrix A (size (2·n‑1) × n) such that A @ g = f * g (full
    linear convolution).  Row k corresponds to index k of the convolution.
    """
    n = f.shape[0]
    m = 2 * n - 1
    A = np.zeros((m, n), dtype=np.float64)
    for i in range(n):
        A[i : i + n, i] = f
    return A


def _solve_lp_max_sum(f: np.ndarray, M: float) -> np.ndarray | None:
    """
    Solve  max Σ g_i   s.t.  (f * g)_k ≤ M  (for every k)  and  g_i ≥ 0.
    Returns the optimal vector g or None on failure.
    """
    if linprog is None:  # pragma: no cover
        return None
    n = f.shape[0]
    A = _conv_matrix(f)
    b = np.full(2 * n - 1, M, dtype=np.float64)
    c = -np.ones(n, dtype=np.float64)          # maximise Σg ⇔ minimise -Σg
    bounds = [(0, None) for _ in range(n)]
    # HiGHS is fast for medium‑sized LPs
    result = linprog(
        c,
        A_ub=A,
        b_ub=b,
        bounds=bounds,
        method="highs",
        options={"presolve": True},
    )
    if not result.success:
        return None
    g = np.maximum(result.x, 0.0)  # clip tiny negatives caused by rounding
    return g


# ----------------------------------------------------------------------
# Alternating LP refinement ---------------------------------------------
def _lp_refine(
    a0: np.ndarray,
    max_iter: int,
    deadline: float,
    best_seq: np.ndarray,
    best_score: float,
) -> tuple[np.ndarray, float]:
    """
    Perform the alternating optimisation described in the statement:
    minimise max‑convolution while keeping the ℓ₁‑norm fixed.
    """
    a = a0.copy()
    n = a.shape[0]

    for it in range(max_iter):
        if time.time() >= deadline:
            break

        # 1) current bound
        conv_a = _fft_self_convolve(a)
        M = conv_a.max()

        # 2) LP – maximise ℓ₁ norm of a partner g
        g = _solve_lp_max_sum(a, M)
        if g is None:
            break

        sum_g = g.sum()
        if sum_g <= 0.0:
            break

        # 3) rescale g to have the same ℓ₁‑norm as a (here Σa = 1)
        g = a.sum() * g / sum_g

        # 4) Mix a and g.  Use back‑tracking on the mixing factor to keep
        #    the maximal convolution non‑increasing.
        t = 0.5
        improved = False
        while t > 1e-3:
            a_candidate = (1.0 - t) * a + t * g
            a_candidate = a_candidate / a_candidate.sum()   # enforce Σ=1
            M_new = _fft_self_convolve(a_candidate).max()
            if M_new <= M * (1.0 + 1e-12):      # accept non‑increase
                a = a_candidate
                improved = True
                break
            t *= 0.5

        if not improved:
            # mixing failed – stop the refinement for this seed
            break

        cur_score = _objective(a)
        if cur_score < best_score - 1e-12:
            best_score = cur_score
            best_seq = a.copy()
            print(
                f"[{time.time():.1f}] LP‑refine it={it} t={t:.3f} score={best_score:.6f}"
            )
        # early exit if objective already beats the target
        if best_score <= 1.503:
            break

    return best_seq, best_score

def propose_candidate(seed: int = 42, budget_s: int = 1000) -> list[float]:
    """
    Search for a non‑negative step‑function that minimises ``evaluate_sequence``.
    The routine builds a small pool of analytic seeds, then applies the
    LP‑based alternating optimisation described above.  The best vector found
    within the allotted time is returned as a plain Python list.
    """
    rng = np.random.default_rng(seed)
    start = time.time()
    deadline = start + budget_s - 5.0  # small safety margin

    overall_best_seq = None
    overall_best_score = float("inf")

    # ------------------------------------------------------------------
    # 1) generate a few diverse seeds (lengths from 300 to 800)
    lengths = [300, 500, 700, 800]
    for n in lengths:
        if time.time() >= deadline:
            break
        print(f"[{time.time():.1f}] exploring length {n}")

        # a few deterministic families
        seed_vectors = []

        seed_vectors.append(_distribution_uniform(n))
        seed_vectors.append(_distribution_arcsine(n))
        seed_vectors.append(_distribution_edge_weighted(n, p=0.4))
        seed_vectors.append(_distribution_edge_weighted(n, p=0.8))
        seed_vectors.append(_distribution_beta_sym(n, p=0.3))
        seed_vectors.append(_distribution_beta_sym(n, p=1.2))
        seed_vectors.append(_distribution_sine_weighted(n, p=2.0))

        # random Dirichlet seeds
        for _ in range(3):
            alpha = rng.uniform(0.02, 1.0)
            seed_vectors.append(rng.dirichlet(np.full(n, alpha)))

        # random sparse seeds
        for _ in range(2):
            seed_vectors.append(_distribution_random_sparse(n, rng, density=0.02))

        # mixture of a base shape with the uniform distribution
        for base in list(seed_vectors):
            for lam in (0.25, 0.5, 0.75):
                seed_vectors.append(_mix_with_uniform(base, lam))

        # evaluate seeds and keep the best few
        evaluated = []
        for vec in seed_vectors:
            vec = np.maximum(vec, 0.0)
            if vec.sum() == 0.0:
                continue
            vec = vec / vec.sum()                 # enforce Σ=1
            sc = _objective(vec)
            evaluated.append((sc, vec.copy()))
            if sc < overall_best_score:
                overall_best_score = sc
                overall_best_seq = vec.copy()
                print(
                    f"[{time.time():.1f}] new global best (seed) len={n} score={sc:.6f}"
                )

        # keep the top 5 seeds for the LP refinement stage
        evaluated.sort(key=lambda x: x[0])
        top_seeds = [seq for _, seq in evaluated[:5]]

        # ------------------------------------------------------------------
        # 2) LP‑based alternating refinement on the selected seeds
        per_seed_time = max(0.0, (deadline - time.time())) / max(1, len(top_seeds))
        for idx, seed_vec in enumerate(top_seeds):
            if time.time() >= deadline:
                break
            sub_deadline = time.time() + per_seed_time
            # number of LP steps – we keep it modest to stay within time
            max_iters = 30
            a_opt, sc_opt = _lp_refine(
                seed_vec,
                max_iter=max_iters,
                deadline=sub_deadline,
                best_seq=overall_best_seq,
                best_score=overall_best_score,
            )
            if sc_opt < overall_best_score:
                overall_best_score = sc_opt
                overall_best_seq = a_opt.copy()
                print(
                    f"[{time.time():.1f}] post‑LP seed {idx} new best {overall_best_score:.6f}"
                )
            if overall_best_score <= 1.503:
                break

        if overall_best_score <= 1.503:
            break

    # ------------------------------------------------------------------
    # 3) optional fallback: if SciPy is missing we fall back to the older
    #    projected‑gradient routine (the code from the reference).  This
    #    block is reached only when ``linprog`` could not be imported.
    if overall_best_seq is None:
        # Very simple uniform fallback (never happens on the reference platform)
        overall_best_seq = _distribution_uniform(500)

    print(
        f"Final best score: {overall_best_score:.6f} (length={len(overall_best_seq)})"
    )
    # Return a plain list – the caller will invoke ``evaluate_sequence``.
    return overall_best_seq.tolist()
