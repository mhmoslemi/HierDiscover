    # conv = a*a
    conv = _fft_self_convolve(a)
    max_val = conv.max()

"""
Search for a low‑value step‑function for ``evaluate_sequence``.

Strategy
--------
1.  Scan a variety of analytic families (uniform, arcsine, sine,
    cosine, quadratic, beta, edge‑weighted, power‑sum) on a few
    moderate lengths and keep the best few candidates.

2.  Refine each candidate with *soft‑max exponential gradient descent*.
    Instead of the sub‑gradient that only sees the currently maximal
    convolution entry we use a smooth approximation of the max:
        Mτ(a) = τ·log(∑k exp(conv[k]/τ)).
    Its gradient is a weighted sum of all convolution entries, which
    yields a smoother landscape and helps escape local minima.
    The update is performed with exponentiated‑gradient steps
    (automatic positivity) and a Euclidean projection onto the simplex
    (normalisation).

3.  Finish with a cheap greedy pairwise polishing phase.

All work respects the supplied time budget (default 1000 s).  The
returned vector is non‑negative, normalised to sum = 1 and is already a
plain ``list`` suitable for ``evaluate_sequence``.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Euclidean projection of ``v`` onto the simplex {x≥0, Σx=s}."""
    if s < 0:
        raise ValueError("Simplex radius must be non‑negative")
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.where(u - cssv / (np.arange(1, len(v) + 1)) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    return np.maximum(v - theta, 0.0)


# ----------------------------------------------------------------------
def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Full self‑convolution ``a*a`` using a real FFT (length 2·n‑1)."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1) - 1).bit_length()  # next power of two
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[: 2 * n - 1]


# ----------------------------------------------------------------------
def _objective(a: np.ndarray) -> float:
    """Exact objective ``2·n·max(conv(a,a))`` for a vector with Σa = 1."""
    n = a.shape[0]
    b = _fft_self_convolve(a)
    return float(2.0 * n * b.max())

# Analytic families ----------------------------------------------------
def _distribution_uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _distribution_arcsine(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    vals /= vals.sum()
    return vals


def _distribution_sine(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64) + 0.5
    w = np.sin(np.pi * i / n)
    w /= w.sum()
    return w


def _distribution_cosine(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64) + 0.5
    w = 1.0 - np.cos(np.pi * i / n)
    w /= w.sum()
    return w


def _distribution_quadratic(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    x = (2.0 * i) / (n - 1) - 1.0
    w = 1.0 - x * x
    w = np.clip(w, 0.0, None)
    w /= w.sum()
    return w


def _distribution_beta_sym(n: int, p: float) -> np.ndarray:
    """Symmetric Beta(p,p) on a regular grid."""
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    w = np.exp((p - 1) * np.log(x) + (p - 1) * np.log(1.0 - x))
    if w.sum() == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _distribution_edge_weighted(n: int, p: float) -> np.ndarray:
    """
    Edge‑weighted family: w_i ∝ ((i+½)*(n‑i‑½))^{‑p}.
    """
    i = np.arange(n, dtype=np.float64) + 0.5
    j = (n - np.arange(n, dtype=np.float64) - 0.5)
    w = (i * j) ** (-p)
    w = np.clip(w, 0.0, None)
    total = w.sum()
    if total > 0:
        w /= total
    else:
        w = np.full(n, 1.0 / n)
    return w


def _distribution_power_sum(n: int, p: float) -> np.ndarray:
    """w_i ∝ (i+1)^{‑p} + (n‑i)^{‑p}."""
    i = np.arange(1, n + 1, dtype=np.float64)
    w = i ** (-p) + (n + 1 - i) ** (-p)
    w = np.clip(w, 0.0, None)
    w /= w.sum()
    return w


def _distribution_random_sparse(
    n: int, rng: np.random.Generator, density: float = 0.01
) -> np.ndarray:
    """Sparse random vector with a prescribed density of non‑zeros."""
    k = max(2, int(n * density))
    idx = rng.choice(n, size=k, replace=False)
    vals = np.zeros(n, dtype=np.float64)
    vals[idx] = rng.random(k)
    if vals.sum() == 0.0:
        vals[0] = 1.0
    vals /= vals.sum()
    return vals


def _mix_with_uniform(base: np.ndarray, lam: float) -> np.ndarray:
    """Convex combination λ·uniform + (1‑λ)·base."""
    n = base.shape[0]
    uniform = np.full(n, 1.0 / n, dtype=np.float64)
    return lam * uniform + (1.0 - lam) * base


# ----------------------------------------------------------------------
# Soft‑max (smooth max) gradient utilities -----------------------------
def _softmax_gradient(a: np.ndarray, tau: float) -> np.ndarray:
    """
    Gradient of the smooth max  τ·log∑ exp(conv/τ) .
    Returns a vector of shape ``a.shape``.
    """
    n = a.shape[0]
    conv = _fft_self_convolve(a)          # length 2n‑1
    max_val = conv.max()
    # weights w_k ∝ exp((conv_k - max)/τ)
    exp_vals = np.exp((conv - max_val) / tau)
    w = exp_vals / exp_vals.sum()         # shape (2n‑1,)

    # gradient_i = 2·∑_k w_k a_{k‑i}
    a_rev = a[::-1]                       # length n
    conv_len = len(w) + len(a_rev) - 1    # = 3n‑2
    size = 1 << (conv_len - 1).bit_length()
    W = np.fft.rfft(w, n=size)
    A_rev = np.fft.rfft(a_rev, n=size)
    conv_wr = np.fft.irfft(W * A_rev, n=size)   # length = size
    # Relevant slice: indices n‑1 … n‑1+n‑1
    grad = 2.0 * conv_wr[n - 1 : n - 1 + n]
    return grad


def _softmax_exponential_descent(
    a0: np.ndarray,
    deadline: float,
    best_seq: np.ndarray,
    best_score: float,
    eta0: float = 0.2,
) -> tuple[np.ndarray, float]:
    """
    Smooth‑max (soft‑max) exponential‑gradient descent.
    ``a0`` is assumed already normalised to sum = 1.
    """
    a = a0.copy()
    n = a.shape[0]
    t = 0
    # keep a running record of the best seen within this run
    local_best_seq = a.copy()
    local_best_score = _objective(a)

    while time.time() < deadline:
        t += 1
        # exact objective (few extra FLOPs – cheap)
        cur_obj = _objective(a)
        if cur_obj < best_score - 1e-14:
            best_score = cur_obj
            best_seq = a.copy()
            print(f"[{time.time():.1f}] softmax it={t} new global best {best_score:.6f}")

        # temperature τ – a decreasing fraction of the current max‑conv
        conv = _fft_self_convolve(a)
        max_conv = conv.max()
        # τ = max_conv * max(0.01, 0.6 / √t)  – starts fairly large then shrinks
        tau_factor = max(0.01, 0.6 / math.sqrt(t))
        tau = max_conv * tau_factor

        # gradient of the smooth max
        grad = _softmax_gradient(a, tau)

        # step size schedule (classic 1/√t)
        eta = eta0 / math.sqrt(t)

        # exponentiated‑gradient update (keeps positivity)
        a = a * np.exp(-eta * grad)

        # renormalise to the simplex
        total = a.sum()
        if total > 0:
            a = a / total
        else:
            a = np.full(n, 1.0 / n, dtype=np.float64)

        # occasional random jitter to escape plateaus
        if t % 2000 == 0:
            jitter = np.random.default_rng().normal(scale=1e-4, size=n)
            a = np.maximum(a + jitter, 0.0)
            a = a / a.sum()

        # stop early if we see no improvement for a long stretch
        if t % 5000 == 0:
            recent = _objective(a)
            if recent >= local_best_score * (1.0 + 1e-8):
                # stagnation – break
                break
            else:
                local_best_score = recent
                local_best_seq = a.copy()

    # return the best we have found globally
    return best_seq, best_score

# Greedy pairwise polishing (identical to the one from the previous run)
def _pairwise_swap(
    a0: np.ndarray,
    time_limit: float,
    best_seq: np.ndarray,
    best_score: float,
    max_trials: int | None = None,
) -> tuple[np.ndarray, float]:
    """
    Randomly move a fraction of mass from one coordinate to another,
    keeping only strictly improving moves.
    """
    a = a0.copy()
    n = a.shape[0]
    trials = 0
    while time.time() < time_limit:
        if max_trials is not None and trials >= max_trials:
            break
        trials += 1

        src = np.where(a > 1e-14)[0]
        if src.size == 0:
            break
        i = np.random.choice(src)
        j = np.random.randint(n - 1)
        if j >= i:
            j += 1

        delta = a[i] * np.random.rand() * 0.45
        if delta < 1e-15:
            continue

        new_a = a.copy()
        new_a[i] -= delta
        new_a[j] += delta

        cur = _objective(new_a)
        if cur < best_score - 1e-14:
            a = new_a
            best_score = cur
            best_seq = a.copy()
            print(
                f"[{time.time():.1f}] swap i={i}→j={j} δ={delta:.6g} new best {best_score:.6f}"
            )
    return best_seq, best_score

def propose_candidate(seed: int = 0, budget_s: int = 1000) -> list[float]:
    """
    Main entry‑point used by the evaluator.
    Returns a list of non‑negative numbers (a step‑function) that
    strives to minimise ``evaluate_sequence``.
    """
    rng = np.random.default_rng(seed)
    start = time.time()
    deadline = start + budget_s - 2.0  # safety margin

    overall_best_seq = None
    overall_best_score = float("inf")
    top_candidates: list[tuple[float, np.ndarray]] = []

    # ---------------------------------------------------------------
    # 1.  Scan a rich set of analytic families on a few lengths.
    # ---------------------------------------------------------------
    lengths = [4096, 8192, 16384]          # moderate – enough resolution, still fast
    # parameters for the edge‑weighted / power‑sum families
    p_grid = np.concatenate(
        [
            np.logspace(-2, -0.2, 12),   # 0.01 … 0.63
            np.linspace(0.6, 3.0, 15),   # 0.6 … 3.0
            np.linspace(3.0, 6.0, 10),   # 3 … 6
        ]
    )
    lam_grid = np.linspace(0.0, 1.0, 5)   # mixing with uniform

    for n in lengths:
        if time.time() >= deadline:
            break
        print(f"[{time.time():.1f}] scanning length {n}")

        # deterministic families -------------------------------------------------
        for fn in (
            _distribution_uniform,
            _distribution_arcsine,
            _distribution_sine,
            _distribution_cosine,
            _distribution_quadratic,
        ):
            cand = fn(n)
            sc = _objective(cand)
            top_candidates.append((sc, cand.copy()))
            if sc < overall_best_score:
                overall_best_score = sc
                overall_best_seq = cand.copy()
                print(f"[{time.time():.1f}] new global best len={n} score={sc:.6f}")

        # parametric families ----------------------------------------------------
        for p in p_grid:
            if time.time() >= deadline:
                break

            edge_seq = _distribution_edge_weighted(n, p)
            pow_seq = _distribution_power_sum(n, p)
            beta_seq = _distribution_beta_sym(n, p)

            for base in (edge_seq, pow_seq, beta_seq):
                for lam in lam_grid:
                    if time.time() >= deadline:
                        break
                    cand = _mix_with_uniform(base, lam)
                    sc = _objective(cand)
                    top_candidates.append((sc, cand.copy()))
                    if sc < overall_best_score:
                        overall_best_score = sc
                        overall_best_seq = cand.copy()
                        print(
                            f"[{time.time():.1f}] new best len={n} p={p:.3g} lam={lam:.2f} score={sc:.6f}"
                        )

        # random Dirichlet seeds -------------------------------------------------
        for _ in range(3):
            if time.time() >= deadline:
                break
            alpha = rng.uniform(0.05, 2.0)
            cand = rng.dirichlet(np.full(n, alpha))
            sc = _objective(cand)
            top_candidates.append((sc, cand.copy()))
            if sc < overall_best_score:
                overall_best_score = sc
                overall_best_seq = cand.copy()
                print(f"[{time.time():.1f}] Dirichlet len={n} α={alpha:.3g} score={sc:.6f}")

        # sparse random seeds ----------------------------------------------------
        for _ in range(3):
            if time.time() >= deadline:
                break
            cand = _distribution_random_sparse(n, rng, density=0.006)
            sc = _objective(cand)
            top_candidates.append((sc, cand.copy()))
            if sc < overall_best_score:
                overall_best_score = sc
                overall_best_seq = cand.copy()
                print(f"[{time.time():.1f}] sparse len={n} score={sc:.6f}")

        # keep only the best ~30 seeds across all lengths
        top_candidates.sort(key=lambda x: x[0])
        top_candidates = top_candidates[:30]

    # ---------------------------------------------------------------
    # 2.  Incorporate a previously discovered construction if present.
    # ---------------------------------------------------------------
    try:
        prev = np.asarray(height_sequence_1, dtype=np.float64)  # type: ignore[name-defined]
        if prev.ndim == 1 and prev.size > 0:
            prev = np.maximum(prev, 0.0)
            s = prev.sum()
            if s == 0.0:
                prev = _distribution_uniform(prev.size)
            else:
                prev = prev / s
            sc = _objective(prev)
            if sc < overall_best_score:
                overall_best_score = sc
                overall_best_seq = prev.copy()
                print(f"[{time.time():.1f}] previous construction improved score={sc:.6f}")
            top_candidates.append((sc, prev.copy()))
    except Exception:
        pass

    # ---------------------------------------------------------------
    # 3.  Fallback – should never trigger.
    # ---------------------------------------------------------------
    if overall_best_seq is None:
        n = 5000
        overall_best_seq = _distribution_uniform(n)
        overall_best_score = _objective(overall_best_seq)
        print(f"[{time.time():.1f}] fallback uniform len={n} score={overall_best_score:.6f}")

    # ---------------------------------------------------------------
    # 4.  Refine the top few seeds with smooth‑max exponentiated GD.
    # ---------------------------------------------------------------
    remaining = deadline - time.time()
    if remaining > 30.0 and top_candidates:
        # pick the 6 most promising seeds for deeper work
        seeds = [seq for _, seq in sorted(top_candidates, key=lambda x: x[0])[:6]]
        per_seed = remaining / max(1, len(seeds))
        print(f"[{time.time():.1f}] Soft‑max GD on {len(seeds)} seeds")
        for idx, seed_seq in enumerate(seeds):
            if time.time() >= deadline:
                break
            sg_deadline = time.time() + per_seed
            overall_best_seq, overall_best_score = _softmax_exponential_descent(
                seed_seq,
                sg_deadline,
                overall_best_seq,
                overall_best_score,
                eta0=0.25,
            )
            print(
                f"[{time.time():.1f}] after softmax GD #{idx} best={overall_best_score:.6f}"
            )

    # ---------------------------------------------------------------
    # 5.  Final greedy polishing.
    # ---------------------------------------------------------------
    if time.time() < deadline:
        print(f"[{time.time():.1f}] final greedy polishing")
        overall_best_seq, overall_best_score = _pairwise_swap(
            overall_best_seq.copy(),
            deadline,
            overall_best_seq,
            overall_best_score,
        )
        print(
            f"[{time.time():.1f}] after polishing best score={overall_best_score:.6f}"
        )

    # ---------------------------------------------------------------
    # 6.  Enforce exact symmetry (helps the objective) and renormalise.
    # ---------------------------------------------------------------
    if overall_best_seq is not None:
        a = overall_best_seq
        a = (a + a[::-1]) * 0.5
        if a.sum() > 0:
            a = a / a.sum()
        else:
            a = _distribution_uniform(a.shape[0])
        overall_best_seq = a

    print(
        f"Final best score (2·n·max(conv)) = {overall_best_score:.6f} (length={len(overall_best_seq)})"
    )
    return overall_best_seq.tolist()
