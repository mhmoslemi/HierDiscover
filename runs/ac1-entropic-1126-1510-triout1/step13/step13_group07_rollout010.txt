"""
search for a non‑negative step function (list of heights) that minimizes
evaluate_sequence (i.e. 2·n·max(conv(a,a)) with ∑a = 1).

Algorithm
---------
1. Build a rich pool of analytic families (uniform, arcsine, symmetric / 
   asymmetric Beta, edge‑weighted, exponential‐edge, sine‑weighted) for
   several lengths (2000 … 15000).  For each family sweep a coarse grid of
   parameters, keep the ~15 best candidates (lowest objective).

2. Refine every promising seed with a *projected momentum sub‑gradient*
   descent.  The sub‑gradient of max(conv) is cheap (O(n)) and we keep a
   momentum term (β≈0.9) to accelerate convergence.  Learning rate decays
   as η₀/√t, and occasional exponential jitter helps escaping plateaus.

3. Run a deterministic pairwise‑mass‑transfer hill‑climb (many random
   moves, keep a move only if the objective improves).

4. Finish with a cheap simulated‑annealing walk that occasionally
   accepts worsening moves, again using pairwise transfers.

The whole routine respects the 1000 s budget and returns a plain Python
list; ``evaluate_sequence`` will apply the final clipping / normalisation.
"""

import time
import math
import heapq
import numpy as np

# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Euclidean projection of v onto {x ≥ 0, Σx = s}."""
    if s < 0:
        raise ValueError("simplex radius must be non‑negative")
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho_arr = np.where(u - cssv / (np.arange(1, len(v) + 1)) > 0)[0]
    if rho_arr.size == 0:
        theta = 0.0
    else:
        rho = rho_arr[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Full self‑convolution a*a (length 2·n‑1) using a real FFT."""
    n = a.shape[0]
    need = 2 * n - 1
    size = 1 << ((need - 1).bit_length())
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[:need]


# ----------------------------------------------------------------------
def _objective(a: np.ndarray) -> float:
    """Exact objective 2·n·max(conv(a,a)) for a vector with ∑a = 1."""
    n = a.shape[0]
    b = _fft_self_convolve(a)
    return 2.0 * n * b.max()

# Analytic families ----------------------------------------------------
def _distribution_uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _distribution_arcsine(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    vals /= vals.sum()
    return vals


def _distribution_beta_sym(n: int, p: float) -> np.ndarray:
    """Symmetric beta (Beta(p,p)) discretised on {1,…,n}."""
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    w = np.exp((p - 1) * np.log(x) + (p - 1) * np.log(1.0 - x))
    if w.sum() == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _distribution_beta_asym(n: int, a: float, b: float) -> np.ndarray:
    """Asymmetric beta (Beta(a,b))."""
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    w = np.exp((a - 1) * np.log(x) + (b - 1) * np.log(1.0 - x))
    if w.sum() == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _distribution_edge_weighted(n: int, p: float) -> np.ndarray:
    """U‑shaped family w_i ∝ ((i+½)*(n‑i‑½))^(−p)."""
    i = np.arange(n, dtype=np.float64) + 0.5
    j = (n - np.arange(n, dtype=np.float64) - 0.5)
    w = (i * j) ** (-p)
    if w.sum() == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _distribution_exponential_edge(n: int, lam: float) -> np.ndarray:
    """Symmetric exponential edge: w_i ∝ exp(-lam·i) + exp(-lam·(n-1‑i))."""
    i = np.arange(n, dtype=np.float64)
    w = np.exp(-lam * i) + np.exp(-lam * (n - 1 - i))
    if w.sum() == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _distribution_sine_weighted(n: int, p: float) -> np.ndarray:
    """Weight ∝ sin(π (i+½)/n)^p."""
    theta = np.pi * (np.arange(n, dtype=np.float64) + 0.5) / n
    w = np.sin(theta) ** p
    if w.sum() == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _distribution_random_sparse(
    n: int, rng: np.random.Generator, density: float = 0.01
) -> np.ndarray:
    """Sparse random vector (few non‑zeros)."""
    k = max(2, int(n * density))
    idx = rng.choice(n, size=k, replace=False)
    vals = np.zeros(n, dtype=np.float64)
    vals[idx] = rng.random(k)
    if vals.sum() == 0.0:
        vals[0] = 1.0
    vals /= vals.sum()
    return vals


def _mix_with_uniform(base: np.ndarray, lam: float) -> np.ndarray:
    """Convex mixture lam·U + (1‑lam)·base, U uniform."""
    n = base.shape[0]
    uniform = np.full(n, 1.0 / n, dtype=np.float64)
    return lam * uniform + (1.0 - lam) * base

# Projected sub‑gradient descent with momentum -------------------------
def _proj_grad_descent_momentum(
    a: np.ndarray,
    deadline: float,
    best_seq: np.ndarray,
    best_score: float,
    eta0: float = 0.6,
    beta: float = 0.9,
    jitter_interval: int = 2000,
    max_iter: int = 300_000,
) -> tuple[np.ndarray, float]:
    """
    Minimise 2·n·max(conv(a,a)) using a sub‑gradient with Nesterov‑style
    momentum, projection onto the probability simplex and occasional jitter.
    """
    n = a.shape[0]
    t = 0
    no_improve = 0
    rng = np.random.default_rng()
    v = np.zeros_like(a)                # momentum buffer
    cur_score = _objective(a)

    while time.time() < deadline and t < max_iter:
        t += 1

        # 1) convolution & max value
        conv = _fft_self_convolve(a)
        max_val = conv.max()
        cur_score = 2.0 * n * max_val

        # update global best
        if cur_score < best_score - 1e-14:
            best_score = cur_score
            best_seq = a.copy()
            no_improve = 0
            print(f"[{time.time():.1f}] it={t} new best {best_score:.6f}")
        else:
            no_improve += 1

        # 2) sub‑gradient for all maximal indices
        ks = np.where(conv == max_val)[0]
        grad = np.zeros_like(a)
        rev_a = a[::-1]

        for k in ks:
            offset = n - 1 - k
            if offset >= 0:
                i_start = 0
                i_end = n - offset
                grad[i_start:i_end] += 2.0 * rev_a[offset:offset + i_end - i_start]
            else:
                off = -offset
                i_start = off
                i_end = n
                grad[i_start:i_end] += 2.0 * rev_a[0:i_end - i_start]

        # 3) momentum update and step
        v = beta * v + grad
        eta = eta0 / math.sqrt(t)
        a_new = a - eta * v
        a_new = _project_onto_simplex(a_new, s=1.0)

        # 4) jitter if stuck
        if no_improve >= jitter_interval:
            jitter = rng.exponential(scale=0.003, size=n)
            a_new = a_new + jitter
            a_new = _project_onto_simplex(a_new, s=1.0)
            no_improve = 0

        a = a_new

    return best_seq, best_score

# Deterministic pairwise refinement ------------------------------------
def _pairwise_swap_local(
    a: np.ndarray,
    deadline: float,
    best_seq: np.ndarray,
    best_score: float,
    max_trials: int | None = None,
) -> tuple[np.ndarray, float]:
    """Move a small amount of mass between two positions; keep the move if it improves."""
    n = a.shape[0]
    rng = np.random.default_rng()
    trials = 0
    while time.time() < deadline:
        if max_trials is not None and trials >= max_trials:
            break
        trials += 1

        src = np.where(a > 1e-12)[0]
        if src.size == 0:
            break
        i = rng.choice(src)

        j = rng.integers(n - 1)
        if j >= i:
            j += 1

        delta = a[i] * rng.random() * 0.35
        if delta < 1e-14:
            continue

        new_a = a.copy()
        new_a[i] -= delta
        new_a[j] += delta

        cur = _objective(new_a)
        if cur < best_score - 1e-14:
            a = new_a
            best_score = cur
            best_seq = a.copy()
            print(
                f"[{time.time():.1f}] swap i={i}→j={j} δ={delta:.3e} new best {best_score:.6f}"
            )
    return best_seq, best_score

# Cheap simulated annealing -------------------------------------------
def _random_local_search(
    a: np.ndarray,
    deadline: float,
    best_seq: np.ndarray,
    best_score: float,
    init_temp: float = 1e-2,
    decay: float = 0.9995,
) -> tuple[np.ndarray, float]:
    """Very cheap SA that repeatedly transfers a random amount of mass."""
    n = a.shape[0]
    rng = np.random.default_rng()
    T = init_temp
    while time.time() < deadline:
        i = rng.integers(n)
        j = rng.integers(n - 1)
        if j >= i:
            j += 1
        delta = a[i] * rng.random() * 0.3
        if delta < 1e-14:
            continue
        new_a = a.copy()
        new_a[i] -= delta
        new_a[j] += delta
        cur = _objective(new_a)
        if cur < best_score - 1e-14:
            a = new_a
            best_score = cur
            best_seq = a.copy()
            print(f"[{time.time():.1f}] SA improve {i}->{j} Δ={delta:.3e} best {best_score:.6f}")
        else:
            if rng.random() < math.exp(-(cur - best_score) / max(T, 1e-12)):
                a = new_a
        T *= decay
    return best_seq, best_score

        for _ in range(3):
            if time.time() >= deadline:
                break
            a_param = 10 ** rng.uniform(-2, 0)   # 0.01 … 1
            b_param = 10 ** rng.uniform(-2, 0)
            cand = _distribution_beta_asym(n, a_param, b_param)
            sc = _objective(cand)
            _push_candidate(sc, cand.copy())
            if sc < overall_best_score:
                overall_best_score, overall_best_seq = sc, cand.copy()
                print(
                    f"[{time.time():.1f}] new global best asym‑beta a={a_param:.3g} b={b_param:.3g} n={n} score={sc:.6f}"
                )

        # Dirichlet random vectors
        for _ in range(5):
            if time.time() >= deadline:
                break
            alpha = rng.uniform(0.01, 2.0)
            cand = rng.dirichlet(np.full(n, alpha))
            sc = _objective(cand)
            _push_candidate(sc, cand.copy())
            if sc < overall_best_score:
                overall_best_score, overall_best_seq = sc, cand.copy()
                print(
                    f"[{time.time():.1f}] new global best Dirichlet α={alpha:.3g} n={n} score={sc:.6f}"
                )

        # Sparse random vectors
        for _ in range(3):
            if time.time() >= deadline:
                break
            cand = _distribution_random_sparse(n, rng, density=0.008)
            sc = _objective(cand)
            _push_candidate(sc, cand.copy())
            if sc < overall_best_score:
                overall_best_score, overall_best_seq = sc, cand.copy()
                print(
                    f"[{time.time():.1f}] new global best sparse n={n} score={sc:.6f}"
                )

    # Extract the best seeds for refinement (sorted by score asc)
    heap.sort(key=lambda x: -x[0])            # smallest scores last
    top_seeds = [seq for _, seq in heap[:max_kept]]

    # ------------------------------------------------------------------
    # 2. Refine each promising seed with momentum sub‑gradient descent
    remaining_time = deadline - time.time()
    if remaining_time > 30.0 and top_seeds:
        per_seed = remaining_time / len(top_seeds)
        print(f"[{time.time():.1f}] start momentum sub‑gradient on {len(top_seeds)} seeds")
        for idx, seed_seq in enumerate(top_seeds):
            if time.time() >= deadline:
                break
            sub_deadline = time.time() + per_seed
            a_opt, sc_opt = _proj_grad_descent_momentum(
                seed_seq.copy(),
                sub_deadline,
                best_seq=overall_best_seq,
                best_score=overall_best_score,
                eta0=0.7,
                beta=0.9,
                jitter_interval=1500,
            )
            if sc_opt < overall_best_score:
                overall_best_score, overall_best_seq = sc_opt, a_opt.copy()
                print(
                    f"[{time.time():.1f}] after momentum seed {idx} new global best {overall_best_score:.6f}"
                )
            # early exit if we already beat the target threshold
            if overall_best_score <= 1.503:
                break

    # ------------------------------------------------------------------
    # 3. Greedy pairwise refinement (deterministic hill‑climb)
    if time.time() < deadline:
        print(f"[{time.time():.1f}] pairwise hill‑climb")
        overall_best_seq, overall_best_score = _pairwise_swap_local(
            overall_best_seq.copy(),
            deadline,
            best_seq=overall_best_seq,
            best_score=overall_best_score,
            max_trials=400_000,
        )

    # ------------------------------------------------------------------
    # 4. Final cheap simulated‑annealing walk
    if time.time() < deadline:
        print(f"[{time.time():.1f}] final simulated‑annealing")
        overall_best_seq, overall_best_score = _random_local_search(
            overall_best_seq.copy(),
            deadline,
            best_seq=overall_best_seq,
            best_score=overall_best_score,
            init_temp=0.01,
            decay=0.9997,
        )

    # ------------------------------------------------------------------
    print(
        f"Final best score: {overall_best_score:.6f} (length={len(overall_best_seq)})"
    )
    return overall_best_seq.tolist()
