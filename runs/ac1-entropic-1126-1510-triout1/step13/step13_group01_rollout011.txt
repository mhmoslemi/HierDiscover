"""
Optimisation of a non‑negative step function for the provided
evaluate_sequence() objective.

The search proceeds in three phases:

1️⃣  **Rich seed generation** – a large variety of analytic families
    (uniform, arcsine, beta, edge‑power, edge‑general, cauchy,
    exponential, sparse / Dirichlet samples) are generated for several
    lengths.  Two‑seed convex mixtures are also tried.  The ≈30 best
    candidates (lowest exact objective) are kept.

2️⃣  **Multi‑restart exponentiated‑gradient (mirror) descent** – each of
    the retained seeds is refined by an EG step with momentum.  The
    update uses the sub‑gradient formed from all convolution entries
    that are within a tiny fraction of the current maximum.  Restarts
    are performed until the time budget is almost exhausted.

3️⃣  **Local polishing** – a greedy pairwise mass‑transfer followed by a
    cheap simulated‑annealing style search cleans up remaining
    irregularities.

All numbers are non‑negative, summed to 1 (the evaluator re‑normalises
any scale) and the function returns a plain Python list ready for
evaluate_sequence().
"""

import time
import math
import numpy as np
from numpy.fft import rfft, irfft

# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x ≥ 0, sum(x)=s}."""
    if s < 0:
        raise ValueError("simplex radius must be non‑negative")
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho_arr = np.where(u - cssv / (np.arange(1, len(v) + 1)) > 0)[0]
    if rho_arr.size == 0:
        theta = 0.0
    else:
        rho = rho_arr[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Full self‑convolution a*a using a real FFT (returns length 2n‑1)."""
    n = a.shape[0]
    need = 2 * n - 1
    size = 1 << ((need - 1).bit_length())
    A = rfft(a, n=size)
    conv = irfft(A * A, n=size)
    return conv[:need]


# ----------------------------------------------------------------------
def _objective(a: np.ndarray) -> float:
    """Exact objective 2·n·max(conv(a,a)) when sum(a)=1."""
    n = a.shape[0]
    b = _fft_self_convolve(a)
    return 2.0 * n * b.max()

def _distribution_uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _distribution_arcsine(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    vals /= vals.sum()
    return vals


def _distribution_beta_sym(n: int, p: float) -> np.ndarray:
    """Symmetric beta (Beta(p,p))."""
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    w = np.exp((p - 1) * np.log(x) + (p - 1) * np.log(1.0 - x))
    s = w.sum()
    if s == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= s
    return w


def _distribution_beta_general(n: int, alpha: float, beta: float) -> np.ndarray:
    """Asymmetric beta (Beta(alpha,beta))."""
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    eps = np.finfo(float).tiny
    x = np.clip(x, eps, 1 - eps)
    w = np.exp((alpha - 1) * np.log(x) + (beta - 1) * np.log(1.0 - x))
    s = w.sum()
    if s == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= s
    return w


def _distribution_edge_weighted(n: int, p: float) -> np.ndarray:
    """U‑shaped family w_i ∝ ((i+0.5)*(n‑i‑0.5))^{−p}."""
    i = np.arange(n, dtype=np.float64) + 0.5
    j = (n - np.arange(n, dtype=np.float64) - 0.5)
    w = (i * j) ** (-p)
    s = w.sum()
    if s == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= s
    return w


def _distribution_edge_general(n: int, p: float, gamma: float) -> np.ndarray:
    """
    General edge‑power family:
    w_i ∝ ((i+γ)*(n‑1‑i+γ))^{−p};  γ>0 avoids singularities.
    """
    i = np.arange(n, dtype=np.float64) + gamma
    j = (n - 1 - np.arange(n, dtype=np.float64)) + gamma
    w = (i * j) ** (-p)
    # replace infinities / NaNs by zero
    w = np.where(np.isfinite(w), w, 0.0)
    s = w.sum()
    if s == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= s
    return w


def _distribution_edge_inversepow(n: int, p: float) -> np.ndarray:
    """Second edge family w_i ∝ (i+0.5)^{−p} + (n−i−0.5)^{−p}."""
    i = np.arange(n, dtype=np.float64) + 0.5
    j = (n - np.arange(n, dtype=np.float64) - 0.5)
    w = i ** (-p) + j ** (-p)
    s = w.sum()
    if s == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= s
    return w


def _distribution_sine_weighted(n: int, p: float) -> np.ndarray:
    """Weight ∝ sin(π (i+½)/n)^p, vanishing at the ends."""
    theta = np.pi * (np.arange(n, dtype=np.float64) + 0.5) / n
    w = np.sin(theta) ** p
    s = w.sum()
    if s == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= s
    return w


def _distribution_tent(n: int) -> np.ndarray:
    """Triangular (tent) shape, peak in the centre."""
    i = np.arange(n, dtype=np.float64)
    half = (n - 1) / 2.0
    w = np.where(i <= half, i + 1, n - i)
    w /= w.sum()
    return w


def _distribution_quadratic(n: int) -> np.ndarray:
    """Quadratic bell shape, symmetric."""
    i = np.arange(n, dtype=np.float64)
    half = (n - 1) / 2.0
    d = np.abs(i - half) + 1.0
    w = 1.0 / (d ** 2)
    w /= w.sum()
    return w


def _distribution_exponential_decay(n: int, rate: float) -> np.ndarray:
    """Exponential decay from the centre."""
    i = np.arange(n, dtype=np.float64)
    half = (n - 1) / 2.0
    w = np.exp(-rate * np.abs(i - half))
    w /= w.sum()
    return w


def _distribution_cauchy_weighted(n: int, scale: float, p: float) -> np.ndarray:
    """
    Heavy‑tailed Cauchy‑like shape:
    w_i ∝ (1 + ((i-centre)/scale)^2)^{‑p}.
    """
    centre = (n - 1) / 2.0
    i = np.arange(n, dtype=np.float64)
    d2 = ((i - centre) / scale) ** 2
    w = (1.0 + d2) ** (-p)
    s = w.sum()
    if s == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= s
    return w


def _distribution_random_sparse(
    n: int, rng: np.random.Generator, density: float = 0.01
) -> np.ndarray:
    """Sparse random vector with a given density of non‑zeros."""
    k = max(2, int(n * density))
    idx = rng.choice(n, size=k, replace=False)
    vals = np.zeros(n, dtype=np.float64)
    vals[idx] = rng.random(k)
    if vals.sum() == 0.0:
        vals[0] = 1.0
    vals /= vals.sum()
    return vals


def _distribution_dirichlet(n: int, rng: np.random.Generator) -> np.ndarray:
    """Random Dirichlet sample with a random concentration."""
    alpha = rng.uniform(0.01, 2.0)
    a = rng.dirichlet(np.full(n, alpha))
    return a.astype(np.float64)


def _mix_with_uniform(base: np.ndarray, lam: float) -> np.ndarray:
    """Convex combination lam·U + (1‑lam)·base, where U is uniform."""
    n = base.shape[0]
    uniform = np.full(n, 1.0 / n, dtype=np.float64)
    return lam * uniform + (1.0 - lam) * base


def _mix_two_vectors(v1: np.ndarray, v2: np.ndarray, lam: float) -> np.ndarray:
    """Return lam·v1 + (1‑lam)·v2, re‑normalised to sum‑one."""
    mixed = lam * v1 + (1.0 - lam) * v2
    s = mixed.sum()
    if s == 0.0:
        return np.full_like(mixed, 1.0 / len(mixed))
    return mixed / s

# Mirror‑descent with momentum on the exact objective -------------------
def _subgradient_descent_momentum(
    a: np.ndarray,
    deadline: float,
    best_seq: np.ndarray,
    best_score: float,
    lr0: float = 0.7,
    active_frac: float = 0.04,
    max_active: int = 20,
    beta: float = 0.9,
    eval_interval: int = 200,
    patience: int = 3000,
) -> tuple[np.ndarray, float]:
    """
    Exponentiated‑gradient descent with momentum.
    ``a`` is assumed to sum to one.
    Returns the best sequence and its exact objective encountered.
    """
    n = a.shape[0]
    t = 0
    no_improve = 0
    momentum = np.zeros_like(a)
    rng = np.random.default_rng()
    ind_grid = np.arange(n, dtype=np.int32)

    while time.time() < deadline and t < 500_000:
        t += 1

        conv = _fft_self_convolve(a)
        max_c = conv.max()
        thresh = max_c * (1.0 - active_frac)

        # active set (tight constraints)
        act = np.where(conv >= thresh)[0]
        if act.size > max_active:
            # keep only the largest few
            top_idx = np.argpartition(conv[act], -max_active)[-max_active:]
            act = act[top_idx]

        # sub‑gradient accumulation
        grad = np.zeros_like(a)
        for idx in act:
            offset = idx - ind_grid
            mask = (offset >= 0) & (offset < n)
            grad[mask] += 2.0 * a[offset[mask]]

        # momentum update
        momentum = beta * momentum + (1.0 - beta) * grad

        # decaying learning rate
        eta = lr0 / math.sqrt(t)

        # mirror (exponential) update
        a = a * np.exp(-eta * momentum)

        # projection onto simplex
        a = _project_onto_simplex(a, s=1.0)

        # periodic evaluation
        if t % eval_interval == 0:
            cur = _objective(a)
            if cur < best_score - 1e-14:
                best_score = cur
                best_seq = a.copy()
                no_improve = 0
                print(f"[{time.time():.1f}] it={t:,} new global best {best_score:.6f}")
            else:
                no_improve += 1

        # occasional jitter if stuck
        if no_improve >= patience:
            jitter = rng.exponential(scale=0.006, size=n)
            a = a + jitter
            a = np.maximum(a, 0.0)
            a = _project_onto_simplex(a, s=1.0)
            no_improve = 0

    return best_seq, best_score


# ----------------------------------------------------------------------
# Greedy pairwise refinement -------------------------------------------
def _pairwise_swap_local(
    a: np.ndarray,
    deadline: float,
    best_seq: np.ndarray,
    best_score: float,
    max_trials: int | None = None,
) -> tuple[np.ndarray, float]:
    """Move a random amount of mass between two positions if the objective improves."""
    n = a.shape[0]
    trials = 0
    rng = np.random.default_rng()
    while time.time() < deadline:
        if max_trials is not None and trials >= max_trials:
            break
        trials += 1

        src = np.where(a > 1e-12)[0]
        if src.size == 0:
            break
        i = rng.choice(src)
        j = rng.integers(n - 1)
        if j >= i:
            j += 1

        delta = a[i] * rng.random() * 0.35
        if delta < 1e-14:
            continue

        new_a = a.copy()
        new_a[i] -= delta
        new_a[j] += delta
        cur = _objective(new_a)

        if cur < best_score - 1e-14:
            a = new_a
            best_score = cur
            best_seq = a.copy()
            print(
                f"[{time.time():.1f}] swap {i}->{j} Δ={delta:.3e} new best {best_score:.6f}"
            )
    return best_seq, best_score


# ----------------------------------------------------------------------
# Simulated‑annealing style local search -------------------------------
def _random_local_search(
    a: np.ndarray,
    deadline: float,
    best_seq: np.ndarray,
    best_score: float,
    init_temp: float = 0.008,
    decay: float = 0.9997,
) -> tuple[np.ndarray, float]:
    """Very cheap SA that repeatedly transfers a random amount of mass."""
    n = a.shape[0]
    rng = np.random.default_rng()
    T = init_temp
    while time.time() < deadline:
        i = rng.integers(n)
        j = rng.integers(n - 1)
        if j >= i:
            j += 1
        delta = a[i] * rng.random() * 0.3
        if delta < 1e-14:
            continue
        new_a = a.copy()
        new_a[i] -= delta
        new_a[j] += delta
        cur = _objective(new_a)
        if cur < best_score - 1e-14:
            a = new_a
            best_score = cur
            best_seq = a.copy()
            print(
                f"[{time.time():.1f}] SA improve {i}->{j} Δ={delta:.3e} best {best_score:.6f}"
            )
        else:
            if rng.random() < math.exp(-(cur - best_score) / max(T, 1e-12)):
                a = new_a
        T *= decay
    return best_seq, best_score


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: int = 1000) -> list[float]:
    """
    Search for a non‑negative step‑function that minimises ``evaluate_sequence``.
    The routine respects the given time budget (default 1000 s) and returns
    a plain Python list of heights.
    """
    rng = np.random.default_rng(seed)
    start = time.time()
    deadline = start + budget_s - 2.0   # tiny safety margin

    overall_best_seq: np.ndarray | None = None
    overall_best_score = float("inf")
    top_candidates: list[tuple[float, np.ndarray]] = []

    # ------------------------------------------------------------------
    # 1.  Diverse seed creation (analytic families + random)
    lengths = [2000, 3000, 4000, 6000, 8000, 12000, 16000]

    # parameter grids
    p_grid = np.concatenate(
        [
            np.logspace(-3, -0.5, 8, base=10),      # 0.001 … 0.316
            np.logspace(-0.5, 0.0, 8, base=10),    # 0.316 … 1
            np.logspace(0.0, 0.5, 6, base=10),     # 1 … 3.16
        ]
    )
    gamma_grid = np.array([0.3, 0.5, 1.0, 2.0])
    lam_grid = np.linspace(0.0, 0.6, 7)          # mixing with uniform
    rate_grid = np.linspace(0.001, 0.02, 6)     # exponential decay
    scale_frac_grid = np.array([0.1, 0.2, 0.33]) # for cauchy (fraction of n)
    cauchy_p_grid = np.array([0.5, 1.0, 2.0])   # exponent on tail

    for n in lengths:
        if time.time() >= deadline:
            break
        print(f"[{time.time():.1f}] scanning length {n}")

        # deterministic basic families
        for cand in (
            _distribution_uniform(n),
            _distribution_arcsine(n),
            _distribution_tent(n),
            _distribution_quadratic(n),
        ):
            sc = _objective(cand)
            top_candidates.append((sc, cand.copy()))
            if sc < overall_best_score:
                overall_best_score, overall_best_seq = sc, cand.copy()
                print(f"[{time.time():.1f}] new global best len={n} score={sc:.6f}")

        # edge‑weighted (original) + edge‑general + sine + beta
        for p in p_grid:
            if time.time() >= deadline:
                break
            edge = _distribution_edge_weighted(n, p)
            edge2 = _distribution_edge_inversepow(n, p)
            sine = _distribution_sine_weighted(n, p)
            beta = _distribution_beta_sym(n, p)

            for base in (edge, edge2, sine, beta):
                for lam in lam_grid:
                    if time.time() >= deadline:
                        break
                    cand = _mix_with_uniform(base, lam)
                    sc = _objective(cand)
                    top_candidates.append((sc, cand.copy()))
                    if sc < overall_best_score:
                        overall_best_score, overall_best_seq = sc, cand.copy()
                        print(
                            f"[{time.time():.1f}] new global best len={n} p={p:.3g} lam={lam:.2f} score={sc:.6f}"
                        )

        # new family: edge_general (extra offset gamma)
        for p in p_grid:
            if time.time() >= deadline:
                break
            for gamma in gamma_grid:
                if time.time() >= deadline:
                    break
                base = _distribution_edge_general(n, p, gamma)
                for lam in lam_grid:
                    if time.time() >= deadline:
                        break
                    cand = _mix_with_uniform(base, lam)
                    sc = _objective(cand)
                    top_candidates.append((sc, cand.copy()))
                    if sc < overall_best_score:
                        overall_best_score, overall_best_seq = sc, cand.copy()
                        print(
                            f"[{time.time():.1f}] new global best edge_general n={n} p={p:.3g} γ={gamma:.2f} lam={lam:.2f} score={sc:.6f}"
                        )

        # Cauchy‑like family
        for frac in scale_frac_grid:
            if time.time() >= deadline:
                break
            scale = max(1.0, int(frac * n))
            for p in cauchy_p_grid:
                base = _distribution_cauchy_weighted(n, scale, p)
                for lam in lam_grid:
                    if time.time() >= deadline:
                        break
                    cand = _mix_with_uniform(base, lam)
                    sc = _objective(cand)
                    top_candidates.append((sc, cand.copy()))
                    if sc < overall_best_score:
                        overall_best_score, overall_best_seq = sc, cand.copy()
                        print(
                            f"[{time.time():.1f}] new global best cauchy n={n} scale={scale} p={p:.2f} lam={lam:.2f} score={sc:.6f}"
                        )

        # exponential decay family
        for rate in rate_grid:
            if time.time() >= deadline:
                break
            cand = _distribution_exponential_decay(n, rate)
            sc = _objective(cand)
            top_candidates.append((sc, cand.copy()))
            if sc < overall_best_score:
                overall_best_score, overall_best_seq = sc, cand.copy()
                print(
                    f"[{time.time():.1f}] new global best exp‑decay n={n} rate={rate:.4f} score={sc:.6f}"
                )

        # random Dirichlet seeds (including low concentration)
        for _ in range(6):
            if time.time() >= deadline:
                break
            a = _distribution_dirichlet(n, rng)
            sc = _objective(a)
            top_candidates.append((sc, a.copy()))
            if sc < overall_best_score:
                overall_best_score, overall_best_seq = sc, a.copy()
                print(
                    f"[{time.time():.1f}] new global best Dirichlet len={n} score={sc:.6f}"
                )

        # sparse random seeds
        for _ in range(6):
            if time.time() >= deadline:
                break
            a = _distribution_random_sparse(n, rng, density=0.006)
            sc = _objective(a)
            top_candidates.append((sc, a.copy()))
            if sc < overall_best_score:
                overall_best_score, overall_best_seq = sc, a.copy()
                print(
                    f"[{time.time():.1f}] new global best sparse len={n} score={sc:.6f}"
                )

        # asymmetric beta samples
        for _ in range(5):
            if time.time() >= deadline:
                break
            alpha = 10 ** rng.uniform(-2, 1)   # 0.01 … 10
            beta = 10 ** rng.uniform(-2, 1)
            a = _distribution_beta_general(n, alpha, beta)
            sc = _objective(a)
            top_candidates.append((sc, a.copy()))
            if sc < overall_best_score:
                overall_best_score, overall_best_seq = sc, a.copy()
                print(
                    f"[{time.time():.1f}] new global best beta(α={alpha:.3g},β={beta:.3g}) len={n} score={sc:.6f}"
                )

        # keep the top‑30 across all lengths examined so far
        top_candidates.sort(key=lambda x: x[0])
        top_candidates = top_candidates[:30]

    # ------------------------------------------------------------------
    # 2. Mix top candidates pairwise (convex combinations) – new idea
    mixed_added = 0
    for i in range(min(10, len(top_candidates))):
        for j in range(i + 1, min(10, len(top_candidates))):
            if time.time() >= deadline:
                break
            v1 = top_candidates[i][1]
            v2 = top_candidates[j][1]
            for lam in np.linspace(0.1, 0.9, 9):
                cand = _mix_two_vectors(v1, v2, lam)
                sc = _objective(cand)
                top_candidates.append((sc, cand.copy()))
                if sc < overall_best_score:
                    overall_best_score, overall_best_seq = sc, cand.copy()
                    print(
                        f"[{time.time():.1f}] new global best mix ({i},{j}) lam={lam:.2f} score={sc:.6f}"
                    )
                mixed_added += 1
    # re‑trim after mixing
    top_candidates.sort(key=lambda x: x[0])
    top_candidates = top_candidates[:30]

    # ------------------------------------------------------------------
    # 3. Incorporate a previously discovered construction if present
    try:
        prev = np.asarray(height_sequence_1, dtype=np.float64)  # type: ignore[name-defined]
        if prev.ndim == 1 and prev.size > 0:
            prev = prev / prev.sum()
            sc = _objective(prev)
            if sc < overall_best_score:
                overall_best_score, overall_best_seq = sc, prev.copy()
                print(f"[{time.time():.1f}] previous best score={sc:.6f}")
            top_candidates.append((sc, prev.copy()))
    except Exception:
        pass

    # ------------------------------------------------------------------
    # 4. Guard against the unlikely case that no seed exists
    if overall_best_seq is None:
        n = 3000
        overall_best_seq = _distribution_uniform(n)
        overall_best_score = _objective(overall_best_seq)
        print(
            f"[{time.time():.1f}] fallback uniform score={overall_best_score:.6f}"
        )

    # ------------------------------------------------------------------
    # 5. Multi‑restart mirror‑descent (bulk of remaining budget)
    # We reserve ~30 s for the final polishing stages.
    while time.time() < deadline - 30.0:
        # pick a random top seed (biased towards better ones)
        weights = np.exp(
            -np.array([s for s, _ in top_candidates])
        )  # larger weight for lower score
        probs = weights / weights.sum()
        idx = rng.choice(len(top_candidates), p=probs)
        start_seq = top_candidates[idx][1].copy()

        sub_deadline = time.time() + min(25.0, (deadline - time.time() - 30.0) / 2.0)

        seq, sc = _subgradient_descent_momentum(
            start_seq,
            sub_deadline,
            best_seq=overall_best_seq,
            best_score=overall_best_score,
            lr0=rng.uniform(0.5, 1.0),
            active_frac=rng.uniform(0.015, 0.035),
            max_active=25,
            beta=0.85,
            eval_interval=150,
            patience=2500,
        )
        if sc < overall_best_score:
            overall_best_score, overall_best_seq = sc, seq.copy()
            print(
                f"[{time.time():.1f}] *** NEW GLOBAL BEST after restart {overall_best_score:.6f}"
            )
        # keep the refined candidate in pool for possible later mixing
        top_candidates.append((sc, seq.copy()))
        top_candidates.sort(key=lambda x: x[0])
        top_candidates = top_candidates[:30]

    # ------------------------------------------------------------------
    # 6. Greedy pairwise refinement
    if time.time() < deadline:
        print(f"[{time.time():.1f}] greedy pairwise refinement")
        overall_best_seq, overall_best_score = _pairwise_swap_local(
            overall_best_seq.copy(),
            deadline,
            best_seq=overall_best_seq,
            best_score=overall_best_score,
            max_trials=250_000,
        )

    # ------------------------------------------------------------------
    # 7. Final simulated‑annealing style search
    if time.time() < deadline:
        print(f"[{time.time():.1f}] final simulated‑annealing search")
        overall_best_seq, overall_best_score = _random_local_search(
            overall_best_seq.copy(),
            deadline,
            best_seq=overall_best_seq,
            best_score=overall_best_score,
            init_temp=0.012,
            decay=0.9996,
        )

    # ------------------------------------------------------------------
    print(
        f"Final best exact score ≈ {overall_best_score:.6f} (length={len(overall_best_seq)})"
    )
    # Return a plain Python list – evaluate_sequence will normalise it.
    return overall_best_seq.tolist()
