    n = len(a)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    lr = 0.1
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    t = 0
    # schedule for alpha: start at 1, multiply by factor each epoch
    alpha = 1.0
    alpha_inc = 1.05
    # optionally limit alpha maximum 200.

    while time.time() < deadline and t < max_iters:
        t += 1

        # compute conv and max
        conv = _fft_self_convolve(a)
        max_val = conv.max()
        # compute smoothed max
        # compute weights
        # using log-sum-exp for numerical stability
        # shift = conv - max_val
        shift = conv - max_val
        exp_shift = np.exp(alpha * shift)
        sum_exp = exp_shift.sum()
        w = exp_shift / sum_exp

        # compute gradient using w
        w_rev = w[::-1]
        # convolve a with w_rev via FFT for speed
        # we need convolution of length n+L-1 = 3n-2
        # compute using FFT.
        grad = _weighted_grad(a, w)   # implement separate function.

        # Adam update
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad ** 2)
        m_hat = m / (1 - beta1 ** t)
        v_hat = v / (1 - beta2 ** t)

        a_new = _project_onto_simplex(a_new, s=1.0)

        a = a_new

    return best_seq, best_score

Implementation:

def _weighted_grad(a, w):
    n = a.shape[0]
    L = 2*n - 1
    # compute convolution of a and w_rev
    # using real FFT for speed
    need = n + L - 1
    size = 1 << ((need - 1).bit_length())
    A = np.fft.rfft(a, n=size)
    B = np.fft.rfft(w[::-1], n=size)
    conv = np.fft.irfft(A * B, n=size)[:need]
    # take slice indices L-1 - i
    idx = L - 1 - np.arange(n)
    grad = 2.0 * conv[idx]
    return grad

"""
Propose a non‑negative step‑function (list of heights) that minimizes
`evaluate_sequence`.  The search consists of three stages:

1. **Seed generation** – a collection of analytic families (uniform,
   arcsine, symmetric β, edge‑weighted, offset‑β, sparse vectors, random
   Dirichlet) for several lengths.  The best few seeds are kept.

2. **Smooth Adam optimisation** – we minimise the true objective
   `2·n·max(conv(a,a))` on the probability simplex using a smooth
   approximation of the max:
   Mα(a)= (1/α)·log∑ exp(α·conv(a,a)) .
   The gradient of Mα is a weighted sum of the individual
   convolution gradients and can be computed efficiently by a single
   FFT‑based convolution.  Adam with projection onto the simplex
   (and a slowly increasing smoothing parameter α) drives the
   distribution to a near‑flat autocorrelation.

3. **Deterministic pairwise refinement + cheap Simulated Annealing**
   – final local moves that polish the solution.

All helper functions are top‑level; no file or network I/O is used.
The returned list is a plain Python list – `evaluate_sequence` will apply
the final clipping and normalisation.
"""

import time

# ----------------------------------------------------------------------
# Simplex projection ----------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Euclidean projection of v onto {x≥0, Σx = s}."""
    if s < 0:
        raise ValueError("simplex radius must be non‑negative")
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho_arr = np.where(u - cssv / (np.arange(1, len(v) + 1)) > 0)[0]
    if rho_arr.size == 0:
        theta = 0.0
    else:
        rho = rho_arr[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# FFT‑based convolutions -----------------------------------------------
def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Full self‑convolution a*a (length 2·n‑1) using a real FFT."""
    n = a.shape[0]
    need = 2 * n - 1
    size = 1 << ((need - 1).bit_length())
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[:need]


def _weighted_gradient(a: np.ndarray, w: np.ndarray) -> np.ndarray:
    """
    Gradient of  Σ_k w_k·(a*a)[k]  w.r.t. a.
    Uses the identity  ∂(a*a)[k]/∂a_i = 2·a_{k-i}  (out‑of‑range → 0).

    Parameters
    ----------
    a : ndarray, shape (n,)
        Current probability vector (sum = 1, a_i ≥ 0).
    w : ndarray, shape (2·n‑1,)
        Non‑negative weights that sum to 1 (softmax of the convolution).

    Returns
    -------
    grad : ndarray, shape (n,)
        Exact gradient of the weighted sum.
    """
    n = a.shape[0]
    L = 2 * n - 1

    # convolution a * w_rev  (full length n+L‑1 = 3·n‑2)
    need = n + L - 1
    size = 1 << ((need - 1).bit_length())
    A = np.fft.rfft(a, n=size)
    B = np.fft.rfft(w[::-1], n=size)
    conv = np.fft.irfft(A * B, n=size)[:need]

    # g_i = 2 * conv[L‑1‑i]   for i = 0 … n‑1
    idx = L - 1 - np.arange(n)
    grad = 2.0 * conv[idx]
    return grad


# ----------------------------------------------------------------------
# Exact objective (used for bookkeeping) -------------------------------
def _objective(a: np.ndarray) -> float:
    """Exact objective 2·n·max(conv(a,a))  for a probability vector."""
    n = a.shape[0]
    b = _fft_self_convolve(a)
    return 2.0 * n * b.max()


# ----------------------------------------------------------------------
# Analytic families -----------------------------------------------------
def _distribution_uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _distribution_arcsine(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    vals /= vals.sum()
    return vals


def _distribution_beta_sym(n: int, p: float, delta: float = 0.0) -> np.ndarray:
    """Symmetric Beta(p,p) with optional offset δ (≥0)."""
    i = np.arange(n, dtype=np.float64) + 0.5 + delta
    total = n + 2 * delta
    x = i / total
    with np.errstate(divide="ignore", invalid="ignore"):
        w = np.exp((p - 1) * np.log(x) + (p - 1) * np.log(1.0 - x))
    w[np.isnan(w)] = 0.0
    if w.sum() == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _distribution_edge_weighted(n: int, p: float, delta: float = 0.0) -> np.ndarray:
    """U‑shaped family w_i ∝ ((i+½+δ)*(n‑i‑½+δ))^(−p)."""
    i = np.arange(n, dtype=np.float64) + 0.5 + delta
    j = (n - np.arange(n, dtype=np.float64) - 0.5) + delta
    w = (i * j) ** (-p)
    if w.sum() == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _distribution_random_sparse(
    n: int, rng: np.random.Generator, density: float = 0.01
) -> np.ndarray:
    """Sparse random vector with given density."""
    k = max(2, int(n * density))
    idx = rng.choice(n, size=k, replace=False)
    vals = np.zeros(n, dtype=np.float64)
    vals[idx] = rng.random(k)
    if vals.sum() == 0.0:
        vals[0] = 1.0
    vals /= vals.sum()
    return vals


def _mix_with_uniform(base: np.ndarray, lam: float) -> np.ndarray:
    """Convex combination lam·U + (1‑lam)·base (U = uniform)."""
    n = base.shape[0]
    uniform = np.full(n, 1.0 / n, dtype=np.float64)
    return lam * uniform + (1.0 - lam) * base


# ----------------------------------------------------------------------
# Smooth Adam optimisation -----------------------------------------------
def _smooth_adam_optimize(
    a: np.ndarray,
    deadline: float,
    best_seq: np.ndarray,
    best_score: float,
    lr0: float = 0.12,
    max_iters: int = 20000,
    alpha_start: float = 6.0,
    alpha_factor: float = 1.02,
    alpha_max: float = 300.0,
    beta1: float = 0.9,
    beta2: float = 0.999,
    eps: float = 1e-8,
) -> tuple[np.ndarray, float]:
    """
    Minimise the true objective using a smooth approximation of the max.
    ``a`` must lie on the probability simplex (sum = 1).
    """
    n = a.shape[0]
    m = np.zeros_like(a)          # first moment
    v = np.zeros_like(a)          # second moment
    alpha = alpha_start
    t = 0

    while time.time() < deadline and t < max_iters:
        t += 1

        # 1) convolution and true max (for bookkeeping)
        conv = _fft_self_convolve(a)
        max_val = conv.max()
        cur_score = 2.0 * n * max_val
        if cur_score < best_score - 1e-12:
            best_score = cur_score
            best_seq = a.copy()
            # early exit if we already beat the target
            if best_score <= 1.503:
                break

        # 2) smooth weights (soft‑max over the convolution)
        shift = conv - max_val               # ≤ 0
        exp_shift = np.exp(alpha * shift)    # stable because max shift = 0
        w = exp_shift / exp_shift.sum()      # Σ w = 1

        # 3) gradient of the smoothed objective
        grad = _weighted_gradient(a, w)

        # 4) Adam update
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        step = lr0 * m_hat / (np.sqrt(v_hat) + eps)

        a_new = a - step
        a_new = _project_onto_simplex(a_new, s=1.0)

        # 5) slowly tighten the smoothing
        if t % 500 == 0:
            alpha = min(alpha * alpha_factor, alpha_max)

        a = a_new

    return best_seq, best_score

# Deterministic pairwise refinement ---------------------------------------
def _pairwise_swap_local(
    a: np.ndarray,
    deadline: float,
    best_seq: np.ndarray,
    best_score: float,
    max_trials: int | None = None,
) -> tuple[np.ndarray, float]:
    """Move a small amount of mass between two coordinates if it improves the score."""
    n = a.shape[0]
    trials = 0
    rng = np.random.default_rng()
    while time.time() < deadline:
        if max_trials is not None and trials >= max_trials:
            break
        trials += 1

        src = np.where(a > 1e-12)[0]
        if src.size == 0:
            break
        i = rng.choice(src)

        j = rng.integers(n - 1)
        if j >= i:
            j += 1

        delta = a[i] * rng.random() * 0.35
        if delta < 1e-14:
            continue

        new_a = a.copy()
        new_a[i] -= delta
        new_a[j] += delta

        cur = _objective(new_a)
        if cur < best_score - 1e-12:
            a = new_a
            best_score = cur
            best_seq = a.copy()
            print(f"[{time.time():.1f}] swap i={i}→j={j} δ={delta:.3e} new best {best_score:.6f}")

    return best_seq, best_score

# Very cheap simulated‑annealing walk ------------------------------------
def _random_local_search(
    a: np.ndarray,
    deadline: float,
    best_seq: np.ndarray,
    best_score: float,
    init_temp: float = 0.01,
    decay: float = 0.9996,
) -> tuple[np.ndarray, float]:
    """Cheap SA that repeatedly transfers a random amount of mass."""
    n = a.shape[0]
    rng = np.random.default_rng()
    T = init_temp
    while time.time() < deadline:
        i = rng.integers(n)
        j = rng.integers(n - 1)
        if j >= i:
            j += 1
        delta = a[i] * rng.random() * 0.3
        if delta < 1e-14:
            continue
        new_a = a.copy()
        new_a[i] -= delta
        new_a[j] += delta

        cur = _objective(new_a)
        if cur < best_score - 1e-12:
            a = new_a
            best_score = cur
            best_seq = a.copy()
            print(f"[{time.time():.1f}] SA improve {i}->{j} Δ={delta:.3e} best {best_score:.6f}")
        else:
            # Metropolis acceptance
            if rng.random() < math.exp(-(cur - best_score) / max(T, 1e-12)):
                a = new_a
        T *= decay
    return best_seq, best_score

                break
            alpha = rng.uniform(0.01, 2.0)
            cand = rng.dirichlet(np.full(n, alpha))
            sc = _objective(cand)
            all_candidates.append((sc, cand.copy()))
            if sc < overall_best_score:
                overall_best_score, overall_best_seq = sc, cand.copy()
                print(f"[{time.time():.1f}] new global best Dirichlet α={alpha:.3g} len={n} score={sc:.6f}")

        # Sparse random vectors
        for _ in range(2):
            if time.time() >= deadline:
                break
            cand = _distribution_random_sparse(n, rng, density=0.008)
            sc = _objective(cand)
            all_candidates.append((sc, cand.copy()))
            if sc < overall_best_score:
                overall_best_score, overall_best_seq = sc, cand.copy()
                print(f"[{time.time():.1f}] new global best sparse len={n} score={sc:.6f}")

    # Keep only the very best seeds (typically 8‑10)
    all_candidates.sort(key=lambda x: x[0])
    top_seeds = [seq for _, seq in all_candidates[:8]]

    # ------------------------------------------------------------------
    # 2. Bring in a previously discovered construction if available
    if "height_sequence_1" in globals():
        try:
            prev = np.asarray(height_sequence_1, dtype=np.float64)
            if prev.ndim == 1 and prev.size > 0:
                prev = prev / prev.sum()
                sc = _objective(prev)
                if sc < overall_best_score:
                    overall_best_score, overall_best_seq = sc, prev.copy()
                    print(f"[{time.time():.1f}] imported previous best score={sc:.6f}")
                all_candidates.append((sc, prev.copy()))
        except Exception:
            pass

    # ------------------------------------------------------------------
    # 3. Smooth Adam optimisation on the most promising seeds
    remaining = deadline - time.time()
    if remaining > 30.0 and top_seeds:
        per_seed = remaining / len(top_seeds)
        print(f"[{time.time():.1f}] starting smooth Adam on {len(top_seeds)} seeds")
        for idx, seed_vec in enumerate(top_seeds):
            if time.time() >= deadline:
                break
            sub_deadline = time.time() + per_seed
            seq_opt, sc_opt = _smooth_adam_optimize(
                seed_vec.copy(),
                sub_deadline,
                best_seq=overall_best_seq,
                best_score=overall_best_score,
                lr0=0.12,
                max_iters=25000,
            )
            if sc_opt < overall_best_score:
                overall_best_score, overall_best_seq = sc_opt, seq_opt.copy()
                print(f"[{time.time():.1f}] Adam seed {idx} new best {overall_best_score:.6f}")
            if overall_best_score <= 1.503:
                break

    # ------------------------------------------------------------------
    # 4. Deterministic pairwise refinement
    if time.time() < deadline:
        print(f"[{time.time():.1f}] pairwise refinement")
        overall_best_seq, overall_best_score = _pairwise_swap_local(
            overall_best_seq.copy(),
            deadline,
            best_seq=overall_best_seq,
            best_score=overall_best_score,
            max_trials=300_000,
        )

    # ------------------------------------------------------------------
    # 5. Final cheap simulated‑annealing walk
    if time.time() < deadline:
        print(f"[{time.time():.1f}] final simulated‑annealing")
        overall_best_seq, overall_best_score = _random_local_search(
            overall_best_seq.copy(),
            deadline,
            best_seq=overall_best_seq,
            best_score=overall_best_score,
            init_temp=0.015,
            decay=0.9995,
        )

    # ------------------------------------------------------------------
    print(f"Final best score: {overall_best_score:.6f} (length={len(overall_best_seq)})")
    return overall_best_seq.tolist()
