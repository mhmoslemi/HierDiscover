"""
search for a step‑function (list of non‑negative heights) that minimizes
evaluate_sequence().

Approach
--------
1. Build a rich collection of seeds:
   * Uniform, arcsine, linear ramp, exponential (decay) and the classic
     edge‑weighted family w_i ∝ ((i+½)(n‑i‑½))^(‑p) for a dense grid of p.
   * For each analytic shape also try a convex mixture with the uniform
     distribution:   λ·U + (1‑λ)·base   (λ from 0 to 1).
   * Random Dirichlet draws (α in [0.01, 2]) and sparse random vectors.
   * Random convex combinations of a few previously generated seeds.

   All vectors are normalised to have sum = 1 (the objective
   2·n·max(conv) is invariant to scaling).

2. Keep the best ~20 seeds (lowest objective) and launch a longer
   projected sub‑gradient descent on each of them.  The descent
   (exact sub‑gradient of the maximal convolution entry) uses a
   decreasing stepsize ηₜ = η₀ / sqrt(t) and projects onto the simplex.
   If progress stalls we inject a tiny exponential jitter.

3. Refine the best candidate with a cheap greedy pairwise mass transfer
   and a short simulated‑annealing walk (both operate on the simplex).

The whole routine respects a 1000 s budget; the final python list is
returned – ``evaluate_sequence`` will later apply the mandatory clipping
and normalisation.  The algorithm is deliberately different from the
previous version by using (i) a much finer sweep over the edge‑weighted
parameter p, (ii) an exponential family, (iii) systematic mixing with
the uniform distribution and (iv) a larger pool of random seeds.
"""

# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex {x≥0, Σx=s}."""
    if s < 0:
        raise ValueError("simplex radius must be non‑negative")
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho_arr = np.where(u - cssv / (np.arange(1, len(v) + 1)) > 0)[0]
    if rho_arr.size == 0:
        theta = 0.0
    else:
        rho = rho_arr[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Full self‑convolution a*a using a real FFT (returns length 2n‑1)."""
    n = a.shape[0]
    need = 2 * n - 1
    size = 1 << ((need - 1).bit_length())
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[:need]


# ----------------------------------------------------------------------
def _objective(a: np.ndarray) -> float:
    """Exact objective 2·n·max(conv(a,a)) for a vector with sum = 1."""
    n = a.shape[0]
    b = _fft_self_convolve(a)
    return 2.0 * n * b.max()


# ----------------------------------------------------------------------
# Analytic families ----------------------------------------------------
def _distribution_uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _distribution_arcsine(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    vals /= vals.sum()
    return vals


def _distribution_linear_ramp(n: int) -> np.ndarray:
    """Increasing ramp a_i ∝ i+1 (normalised)."""
    vals = np.arange(1, n + 1, dtype=np.float64)
    vals /= vals.sum()
    return vals


def _distribution_exponential(n: int, alpha: float) -> np.ndarray:
    """Geometric‑decay: a_i ∝ exp(-alpha·i/(n‑1))."""
    i = np.arange(n, dtype=np.float64)
    vals = np.exp(-alpha * i / max(1, n - 1))
    vals /= vals.sum()
    return vals


def _distribution_edge_weighted(n: int, p: float) -> np.ndarray:
    """U‑shaped family w_i ∝ ((i+½)*(n‑i‑½))^(−p)."""
    i = np.arange(n, dtype=np.float64) + 0.5
    j = (n - np.arange(n, dtype=np.float64) - 0.5)
    w = (i * j) ** (-p)
    w /= w.sum()
    return w


def _distribution_random_sparse(
    n: int, rng: np.random.Generator, density: float = 0.015
) -> np.ndarray:
    """Sparse random vector (few non‑zeros)."""
    k = max(2, int(n * density))
    idx = rng.choice(n, size=k, replace=False)
    vals = np.zeros(n, dtype=np.float64)
    vals[idx] = rng.random(k)
    vals /= vals.sum()
    return vals


def _mix_with_uniform(base: np.ndarray, lam: float) -> np.ndarray:
    """Convex combination lam·U + (1‑lam)·base, U uniform."""
    n = base.shape[0]
    uniform = np.full(n, 1.0 / n, dtype=np.float64)
    return lam * uniform + (1.0 - lam) * base

# Projected sub‑gradient descent -----------------------------------------
def _proj_grad_descent(
    a: np.ndarray,
    deadline: float,
    best_seq: np.ndarray,
    best_score: float,
    eta0: float = 0.6,
    jitter_interval: int = 2000,
    max_iter: int = 300_000,
) -> tuple[np.ndarray, float]:
    """
    Minimise 2·n·max(conv(a,a)) with exact sub‑gradient.
    `a` must lie on the simplex (sum = 1).  The routine keeps the global
    best (across all seeds) in the provided containers.
    """
    n = a.shape[0]
    t = 0
    no_improve = 0
    rng = np.random.default_rng()
    cur_score = _objective(a)

    while time.time() < deadline and t < max_iter:
        t += 1

        # convolution and current maximal index set
        conv = _fft_self_convolve(a)
        max_val = conv.max()
        cur_score = 2.0 * n * max_val

        # track global best
        if cur_score < best_score - 1e-13:
            best_score = cur_score
            best_seq = a.copy()
            no_improve = 0
            print(f"[{time.time():.1f}] proj‑grad it={t} new best {best_score:.6f}")
        else:
            no_improve += 1

        # sub‑gradient of the (possibly multiple) maximal entry
        ks = np.where(conv == max_val)[0]
        grad = np.zeros_like(a)
        rev_a = a[::-1]
        for k in ks:
            offset = n - 1 - k
            if offset >= 0:
                i_start = 0
                i_end = n - offset
                grad[i_start:i_end] += 2.0 * rev_a[offset:offset + i_end - i_start]
            else:
                off = -offset
                i_start = off
                i_end = n
                grad[i_start:i_end] += 2.0 * rev_a[0:i_end - i_start]

        # step with decaying learning rate and simplex projection
        eta = eta0 / math.sqrt(t)
        a_new = a - eta * grad
        a_new = _project_onto_simplex(a_new, s=1.0)

        # inject jitter if we are stuck for a while
        if no_improve >= jitter_interval:
            jitter = rng.exponential(scale=0.003, size=n)
            a_new = a_new + jitter
            a_new = _project_onto_simplex(a_new, s=1.0)
            no_improve = 0

        a = a_new

    return best_seq, best_score


# ----------------------------------------------------------------------
# Pairwise deterministic refinement --------------------------------------
def _pairwise_swap_local(
    a: np.ndarray,
    deadline: float,
    best_seq: np.ndarray,
    best_score: float,
    max_trials: int | None = None,
) -> tuple[np.ndarray, float]:
    """Randomly move a small amount of mass; keep move only if objective improves."""
    n = a.shape[0]
    trials = 0
    rng = np.random.default_rng()
    while time.time() < deadline:
        if max_trials is not None and trials >= max_trials:
            break
        trials += 1

        src = np.where(a > 1e-12)[0]
        if src.size == 0:
            break
        i = rng.choice(src)

        j = rng.integers(n - 1)
        if j >= i:
            j += 1

        delta = a[i] * rng.random() * 0.35
        if delta < 1e-14:
            continue

        new_a = a.copy()
        new_a[i] -= delta
        new_a[j] += delta

        cur = _objective(new_a)
        if cur < best_score - 1e-13:
            a = new_a
            best_score = cur
            best_seq = a.copy()
            print(
                f"[{time.time():.1f}] swap i={i}→j={j} δ={delta:.3e} new best {best_score:.6f}"
            )
    return best_seq, best_score


# ----------------------------------------------------------------------
# Cheap simulated‑annealing walk -----------------------------------------
def _random_local_search(
    a: np.ndarray,
    deadline: float,
    best_seq: np.ndarray,
    best_score: float,
    init_temp: float = 1e-2,
    decay: float = 0.9995,
) -> tuple[np.ndarray, float]:
    """Cheap SA that repeatedly transfers a random amount of mass."""
    n = a.shape[0]
    rng = np.random.default_rng()
    T = init_temp
    while time.time() < deadline:
        i = rng.integers(n)
        j = rng.integers(n - 1)
        if j >= i:
            j += 1
        delta = a[i] * rng.random() * 0.3
        if delta < 1e-14:
            continue
        new_a = a.copy()
        new_a[i] -= delta
        new_a[j] += delta
        cur = _objective(new_a)
        if cur < best_score - 1e-13:
            a = new_a
            best_score = cur
            best_seq = a.copy()
            print(f"[{time.time():.1f}] SA improve {i}->{j} Δ={delta:.3e} best {best_score:.6f}")
        else:
            if rng.random() < math.exp(-(cur - best_score) / max(T, 1e-12)):
                a = new_a
        T *= decay
    return best_seq, best_score


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 123, budget_s: int = 1000) -> list[float]:
    """
    Search for a non‑negative step‑function that minimises evaluate_sequence().
    Returns a plain Python list (the caller will invoke evaluate_sequence).
    """
    rng = np.random.default_rng(seed)
    start = time.time()
    deadline = start + budget_s - 4.0  # safety margin

    # ------------------------------------------------------------------
    # 1. generate a large pool of candidate seeds --------------------
    n = 4000                     # a comfortable length; objective ≈ 2·f_max
    all_candidates: list[tuple[float, np.ndarray]] = []

    # ----- deterministic analytic families ---------------------------
    print(f"[{time.time():.1f}] building deterministic candidates (n={n})")

    # uniform
    cand = _distribution_uniform(n)
    sc = _objective(cand)
    all_candidates.append((sc, cand.copy()))

    # arcsine
    cand = _distribution_arcsine(n)
    sc = _objective(cand)
    all_candidates.append((sc, cand.copy()))

    # linear ramp (increasing)
    cand = _distribution_linear_ramp(n)
    sc = _objective(cand)
    all_candidates.append((sc, cand.copy()))

    # edge‑weighted with a fine grid on p
    edge_ps = np.linspace(0.05, 1.40, 28)    # 0.05,0.10,…,1.40
    mix_lams = np.linspace(0.0, 1.0, 11)    # 0,0.1,…,1
    for p in edge_ps:
        base = _distribution_edge_weighted(n, p)
        for lam in mix_lams:
            cand = _mix_with_uniform(base, lam)
            sc = _objective(cand)
            all_candidates.append((sc, cand.copy()))

    # exponential decays (α ≈ 0.5 … 2.5)
    alphas = np.linspace(0.5, 2.5, 21)
    for a in alphas:
        base = _distribution_exponential(n, a)
        for lam in mix_lams:
            cand = _mix_with_uniform(base, lam)
            sc = _objective(cand)
            all_candidates.append((sc, cand.copy()))

    # ----- random constructions ---------------------------------------
    # Dirichlet draws
    for _ in range(30):
        alpha = rng.uniform(0.01, 2.0)
        cand = rng.dirichlet(np.full(n, alpha))
        sc = _objective(cand)
        all_candidates.append((sc, cand.copy()))

    # Sparse random vectors
    for _ in range(15):
        cand = _distribution_random_sparse(n, rng, density=0.012)
        sc = _objective(cand)
        all_candidates.append((sc, cand.copy()))

    # Random convex combos of a few existing seeds (to explore the convex hull)
    existing = [c for _, c in all_candidates[:30]]  # take 30 diverse seeds
    for _ in range(40):
        k = rng.integers(2, 5)                # combine 2–4 seeds
        choices = rng.choice(existing, size=k, replace=False)
        weights = rng.random(k)
        weights /= weights.sum()
        combo = np.zeros(n, dtype=np.float64)
        for w, seq in zip(weights, choices):
            combo += w * seq
        combo /= combo.sum()
        sc = _objective(combo)
        all_candidates.append((sc, combo.copy()))

    # ------------------------------------------------------------------
    # 2. keep the most promising seeds (lowest objective)
    all_candidates.sort(key=lambda x: x[0])
    top_k = 20
    top_seeds = [seq for _, seq in all_candidates[:top_k]]
    overall_best_score, overall_best_seq = all_candidates[0]

    print(f"[{time.time():.1f}] best seed after pool: score={overall_best_score:.6f}")

    # ------------------------------------------------------------------
    # 3. refined projected sub‑gradient descent on the best seeds
    remaining = deadline - time.time()
    if remaining > 30.0 and top_seeds:
        per_seed = remaining / len(top_seeds)
        print(f"[{time.time():.1f}] starting sub‑gradient descent on {len(top_seeds)} seeds")
        for idx, seed_seq in enumerate(top_seeds):
            if time.time() >= deadline:
                break
            sub_deadline = time.time() + per_seed
            a_opt, sc_opt = _proj_grad_descent(
                seed_seq.copy(),
                sub_deadline,
                best_seq=overall_best_seq,
                best_score=overall_best_score,
                eta0=0.6,
                jitter_interval=2500,
                max_iter=250_000,
            )
            if sc_opt < overall_best_score:
                overall_best_score, overall_best_seq = sc_opt, a_opt.copy()
                print(
                    f"[{time.time():.1f}] after descent seed {idx} new global best {overall_best_score:.6f}"
                )
            # early stop if we already beat the target
            if overall_best_score <= 1.503:
                break

    # ------------------------------------------------------------------
    # 4. greedy pairwise refinement
    if time.time() < deadline:
        print(f"[{time.time():.1f}] pairwise deterministic refinement")
        overall_best_seq, overall_best_score = _pairwise_swap_local(
            overall_best_seq.copy(),
            deadline,
            best_seq=overall_best_seq,
            best_score=overall_best_score,
            max_trials=300_000,
        )

    # ------------------------------------------------------------------
    # 5. final cheap simulated‑annealing walk
    if time.time() < deadline:
        print(f"[{time.time():.1f}] final simulated‑annealing walk")
        overall_best_seq, overall_best_score = _random_local_search(
            overall_best_seq.copy(),
            deadline,
            best_seq=overall_best_seq,
            best_score=overall_best_score,
            init_temp=0.008,
            decay=0.9997,
        )

    # ------------------------------------------------------------------
    print(
        f"Final best score: {overall_best_score:.6f} (length={len(overall_best_seq)})"
    )
    # Return a plain Python list – evaluate_sequence will take care of clipping.
    return overall_best_seq.tolist()
