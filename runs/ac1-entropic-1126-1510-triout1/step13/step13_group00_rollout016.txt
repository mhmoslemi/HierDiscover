    offsets = np.arange(n)

    # Use diags([f]*n, offsets, shape=(2*n-1, n), format='csc')

    return res.x

def _lp_refine(a, deadline, max_iter=20):
    """Iteratively apply LP step to improve a's max convolution.
    Returns (best_seq, best_score)."""
    n = a.shape[0]
    best_seq = a.copy()
    best_score = _objective(a)
    f = a.copy()
    iter_count = 0
    while time.time() < deadline and iter_count < max_iter:
        iter_count += 1
        # compute current M
        M = _fft_self_convolve(f).max()
        # solve LP
        b = _solve_lp_max_sum(f, M)
        sum_b = b.sum()
        if sum_b <= 1.0 + 1e-12:
            # cannot improve
            break
        g = b / sum_b
        # mixing weight
        lam = min(0.5, 0.2 * (sum_b - 1) / sum_b)  # heuristics
        new_f = (1 - lam) * f + lam * g
        # reproject onto simplex (sum=1)
        new_f = np.clip(new_f, 0, None)
        new_f_sum = new_f.sum()
        if new_f_sum == 0:
            new_f = np.full(n, 1.0 / n)
        else:
            new_f /= new_f_sum
        new_score = _objective(new_f)
        if new_score + 1e-12 < best_score:
            best_score, best_seq = new_score, new_f.copy()
            print(f"[{time.time():.1f}] LP iter {iter_count} new best {best_score:.6f}")
        f = new_f
    return best_seq, best_score

Now for _solve_lp_max_sum:

def _solve_lp_max_sum(f, M):
    n = f.shape[0]
    from scipy.sparse import diags
    # Build convolution matrix A (2n-1) x n
    offsets = np.arange(n)
    # Using diags with list of f repeated n times
    A = diags([f] * n, offsets, shape=(2 * n - 1, n), format='csc')
    # Build right-hand side vector
    b_ub = np.full(2 * n - 1, M, dtype=np.float64)

    # objective: maximize sum(b) => min -1*b
    c = -np.ones(n, dtype=np.float64)

    # call linprog
    import scipy.optimize as opt
    res = opt.linprog(c, A_ub=A, b_ub=b_ub, bounds=(0, None), method='highs')
    if not res.success:
        # fallback: return f (feasible)
        return f.copy()
    return res.x

"""
Propose a non‑negative step function that minimises ``evaluate_sequence``.

Algorithm
---------

1. **Initial seed** – use the previously discovered construction
   ``height_sequence_1`` if it exists, otherwise a 2500‑point
   arcsine distribution (the arcsine law concentrates mass at the
   edges and gives a low initial self‑convolution peak).

2. **Iterative LP refinement** – for a fixed length (≈2500) repeatedly
   solve a linear program

        maximise   Σ b_j
        subject to   (f * b)_k ≤ M   for all k,
                     b_j ≥ 0,

   where ``M`` is the current maximum of the self‑convolution
   ``f*f``.  The LP gives a vector ``b`` that can carry more total
   mass without violating the convolution bound.  Normalise
   ``g = b / Σb`` and update the current density by a convex step

        f ← (1‑λ)·f + λ·g,    λ = min(0.5, (Σb‑1)/(Σb+1)).

   Each iteration never increases the maximum convolution; the
   objective therefore never rises.  The process stops after a
   time budget or when no improvement is possible.

3. **Upsampling & fine tuning** – if time permits the refined density
   is linearly interpolated to a larger length (up to 18000 points) and
   a short mirror‑descent / simulated‑annealing phase (the same code
   used in the original solution) removes any remaining spikes.

All helper routines are top‑level; no closures or lambdas are used.
The implementation relies on NumPy, SciPy (``linprog`` and sparse
matrices) and pure Python.
"""

import time
import math
import numpy as np
from numpy.fft import rfft, irfft
from scipy.sparse import diags

# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x≥0, Σx = s}."""
    if s < 0:
        raise ValueError("simplex radius must be non‑negative")
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.where(u - cssv / (np.arange(1, len(v) + 1)) > 0)[0]
    theta = 0.0 if rho.size == 0 else cssv[rho[-1]] / (rho[-1] + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a using real FFT (length 2·n‑1)."""
    n = a.shape[0]
    need = 2 * n - 1
    size = 1 << ((need - 1).bit_length())   # next power of two
    A = rfft(a, n=size)
    conv = irfft(A * A, n=size)
    return conv[:need]


# ----------------------------------------------------------------------
def _objective(a: np.ndarray) -> float:
    """Exact objective 2·n·max(conv(a,a)) for a with Σa = 1."""
    n = a.shape[0]
    b = _fft_self_convolve(a)
    return 2.0 * n * b.max()


# ----------------------------------------------------------------------
# Analytic families (only a subset is required for the seed)
def _distribution_arcsine(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    vals /= vals.sum()
    return vals


def _distribution_uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _distribution_dirichlet(n: int, rng: np.random.Generator) -> np.ndarray:
    alpha = rng.uniform(0.01, 2.0)
    a = rng.dirichlet(np.full(n, alpha))
    return a.astype(np.float64)


# ----------------------------------------------------------------------
def _downsample(seq: np.ndarray, new_len: int) -> np.ndarray:
    """Linear interpolation from a longer sequence to a shorter one."""
    old_len = seq.shape[0]
    if new_len == old_len:
        return seq.copy()
    x_old = np.linspace(0.0, 1.0, old_len)
    x_new = np.linspace(0.0, 1.0, new_len)
    new_seq = np.interp(x_new, x_old, seq)
    new_seq = np.maximum(new_seq, 0.0)
    new_seq /= new_seq.sum()
    return new_seq


def _upsample(seq: np.ndarray, new_len: int) -> np.ndarray:
    """Linear interpolation to a longer length (preserves simplex)."""
    return _downsample(seq, new_len)   # same routine works both ways

def _solve_lp_max_sum(f: np.ndarray, M: float) -> np.ndarray:
    """
    Solve  max Σ b_j   s.t.   (f * b)_k ≤ M   for all k,
    b_j ≥ 0  .  Returns the optimal b (may be equal to f if LP fails).
    """
    n = f.shape[0]
    # Build the (2n‑1) × n convolution matrix A where A @ b = f * b.
    # Each diagonal offset d = -(0…n‑1) has constant value f[d].
    diagonals = [np.full(n, f[d], dtype=np.float64) for d in range(n)]
    offsets = -np.arange(n)                      # column – row = offset
    A = diags(diagonals, offsets, shape=(2 * n - 1, n), format='csc')

    c = -np.ones(n, dtype=np.float64)            # maximise Σb   ⇒ minimise –Σb
    ub = np.full(2 * n - 1, M, dtype=np.float64)

    # Use HiGHS (fast interior‑point/simplex) – tolerates dense constraints.
    res = linprog(c, A_ub=A, b_ub=ub, bounds=(0, None), method='highs')
    if not res.success:
        # A safe fallback – keep the current f (which is always feasible).
        return f.copy()
    return res.x


# ----------------------------------------------------------------------
def _lp_refine(a: np.ndarray, deadline: float, max_iter: int = 30):
    """
    Repeatedly apply the LP step to improve the current density.
    Returns the best density found and its objective value.
    """
    n = a.shape[0]
    best_seq = a.copy()
    best_score = _objective(a)
    f = a.copy()
    it = 0

    while time.time() < deadline and it < max_iter:
        it += 1
        # current bound
        M = _fft_self_convolve(f).max()
        # LP maximising the total mass that fits under M
        b = _solve_lp_max_sum(f, M)
        sum_b = b.sum()

        # No improvement possible – the LP cannot increase the mass.
        if sum_b <= 1.0 + 1e-12:
            break

        # Normalise the LP solution
        g = b / sum_b
        # Convex mixing factor (theoretical guarantee: non‑increasing M)
        lam = min(0.5, (sum_b - 1.0) / (sum_b + 1.0))
        new_f = (1.0 - lam) * f + lam * g
        # Project back onto the probability simplex (tiny numerical drift)
        new_f = _project_onto_simplex(new_f, s=1.0)

        cur_score = _objective(new_f)
        if cur_score + 1e-14 < best_score:
            best_score, best_seq = cur_score, new_f.copy()
            print(f"[{time.time():.1f}] LP iter {it} new best {best_score:.6f}")

        f = new_f

    return best_seq, best_score


# ----------------------------------------------------------------------
def _subgradient_descent_momentum(
    a: np.ndarray,
    deadline: float,
    best_seq: np.ndarray,
    best_score: float,
    lr0: float = 0.7,
    active_frac: float = 0.04,
    max_active: int = 30,
    beta: float = 0.9,
    eval_interval: int = 80,
    patience: int = 1500,
) -> tuple[np.ndarray, float]:
    """
    Exponentiated‑gradient (entropy mirror) descent on the exact
    non‑smooth objective.  The active set consists of the convolution
    entries whose value is within ``active_frac`` of the current maximum.
    """
    n = a.shape[0]
    t = 0
    no_improve = 0
    momentum = np.zeros_like(a)
    ind_grid = np.arange(n, dtype=np.int32)

    while time.time() < deadline and t < 400_000:
        t += 1

        conv = _fft_self_convolve(a)
        max_c = conv.max()
        thresh = max_c * (1.0 - active_frac)

        act = np.where(conv >= thresh)[0]
        if act.size > max_active:
            top_idx = np.argpartition(conv[act], -max_active)[-max_active:]
            act = act[top_idx]

        grad = np.zeros_like(a)
        for k in act:
            shift = k - ind_grid
            mask = (shift >= 0) & (shift < n)
            grad[mask] += 2.0 * a[shift[mask]]

        momentum = beta * momentum + (1.0 - beta) * grad
        eta = lr0 / math.sqrt(t)
        a = a * np.exp(-eta * momentum)

        # project onto simplex
        a = _project_onto_simplex(a, s=1.0)

        if t % eval_interval == 0:
            cur = _objective(a)
            if cur + 1e-14 < best_score:
                best_score, best_seq = cur, a.copy()
                no_improve = 0
                print(f"[{time.time():.1f}] it={t:,} new global best {best_score:.6f}")
            else:
                no_improve += 1

        if no_improve >= patience:
            rng = np.random.default_rng()
            jitter = rng.exponential(scale=0.004, size=n)
            a = a + jitter
            a = _project_onto_simplex(a, s=1.0)
            no_improve = 0

    return best_seq, best_score


# ----------------------------------------------------------------------
def _random_local_search(
    a: np.ndarray,
    deadline: float,
    best_seq: np.ndarray,
    best_score: float,
    init_temp: float = 0.006,
    decay: float = 0.999,
) -> tuple[np.ndarray, float]:
    """Cheap simulated‑annealing style local search on the simplex."""
    n = a.shape[0]
    rng = np.random.default_rng()
    T = init_temp
    while time.time() < deadline:
        i = rng.integers(n)
        j = rng.integers(n - 1)
        if j >= i:
            j += 1
        delta = a[i] * rng.random() * 0.28
        if delta < 1e-14:
            continue
        new_a = a.copy()
        new_a[i] -= delta
        new_a[j] += delta
        cur = _objective(new_a)
        if cur + 1e-14 < best_score:
            a = new_a
            best_score = cur
            best_seq = a.copy()
            print(f"[{time.time():.1f}] SA improve {i}->{j} Δ={delta:.3e} best {best_score:.6f}")
        else:
            if rng.random() < math.exp(-(cur - best_score) / max(T, 1e-12)):
                a = new_a
        T *= decay
    return best_seq, best_score

def propose_candidate(seed: int = 42, budget_s: int = 1000) -> list[float]:
    """
    Search for a non‑negative step function that minimises ``evaluate_sequence``.
    The whole search respects the given time budget (default 1000 s).
    Returns a plain Python list of heights.
    """
    rng = np.random.default_rng(seed)
    start = time.time()
    deadline = start + budget_s - 2.0          # small safety margin

    # ------------------------------------------------------------------
    # 0. Choose initial seed (prefer the previously discovered one)
    max_len_lp = 2500                         # size convenient for the LP stage
    prev = globals().get("height_sequence_1", None)

    if prev is not None:
        try:
            a0 = np.asarray(prev, dtype=np.float64)
            if a0.ndim == 1 and a0.size > 0:
                a0 = np.maximum(a0, 0.0)
                a0 = a0 / max(a0.sum(), 1e-12)
                if a0.size > max_len_lp:
                    a0 = _downsample(a0, max_len_lp)
                print(f"[{time.time():.1f}] using previous height_sequence_1 (len={a0.size})")
            else:
                raise ValueError
        except Exception:
            a0 = None
    else:
        a0 = None

    if a0 is None:
        a0 = _distribution_arcsine(max_len_lp)
        print(f"[{time.time():.1f}] using arcsine seed (len={max_len_lp})")

    # ------------------------------------------------------------------
    # 1. LP‑based refinement (core of the new idea)
    best_seq, best_score = _lp_refine(a0, deadline, max_iter=40)
    print(f"[{time.time():.1f}] after LP refinement – score {best_score:.6f}")

    # ------------------------------------------------------------------
    # 2. Upsample to a larger length and perform a short mirror‑descent
    if time.time() < deadline - 30:
        large_len = min(18000, best_seq.size * 2)
        up_seq = _upsample(best_seq, large_len)
        sub_deadline = deadline - 5
        up_seq_opt, up_score = _subgradient_descent_momentum(
            up_seq.copy(),
            sub_deadline,
            best_seq=best_seq,
            best_score=best_score,
            lr0=0.6,
            active_frac=0.04,
            max_active=30,
            beta=0.92,
            eval_interval=100,
            patience=1200,
        )
        if up_score < best_score:
            best_seq, best_score = up_seq_opt, up_score
            print(f"[{time.time():.1f}] up‑sample + descent new best {best_score:.6f}")

    # ------------------------------------------------------------------
    # 3. Final cheap simulated‑annealing clean‑up
    if time.time() < deadline:
        best_seq, best_score = _random_local_search(
            best_seq.copy(),
            deadline,
            best_seq=best_seq,
            best_score=best_score,
            init_temp=0.006,
            decay=0.9995,
        )
        print(f"[{time.time():.1f}] final SA cleanup – best score {best_score:.6f}")

    print(f"Final best score: {best_score:.6f} (length={len(best_seq)})")
    # Return plain Python list – evaluate_sequence will do the final clipping.
    return best_seq.tolist()
