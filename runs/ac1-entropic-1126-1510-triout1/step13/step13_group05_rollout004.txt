def propose_candidate(seed=0, budget_s=1000):
    rng = np.random.default_rng(seed)
    start = time.time()
    deadline = start + budget_s - 5.0

    best_seq = None
    best_score = float('inf')
    candidates = []

    # Stage 1: scan families
    lengths = [1024, 2048, 4096, 8192, 12000]

    for n in lengths:

        # deterministic families
        for fn in [_distribution_uniform, _distribution_arcsine, _distribution_sine,
                   _distribution_cosine, _distribution_quadratic]:
            a = fn(n)
            sc = _objective(a)
            candidates.append((sc, a.copy()))
            if sc < best_score:
                best_score, best_seq = sc, a.copy()
                print(f"new best {best_score:.6f} len={n}")
        # parametric families
        for p in p_vals:
            if time.time()>deadline: break
            edge = _distribution_edge_weighted(n, p)
            beta = _distribution_beta_sym(n, p)
            psum = _distribution_power_sum(n, p)
            for base in (edge, beta, psum):
                for lam in lam_vals:
                    if time.time()>deadline: break
                    a = _mix_with_uniform(base, lam)
                    sc = _objective(a)
                    candidates.append((sc, a.copy()))
                    if sc < best_score:
                        best_score, best_seq = sc, a.copy()
                        print(...)
        # random Dirichlet seeds
        for _ in range(3):
            if time.time()>deadline: break
            alpha = rng.uniform(0.01, 2.0)
            a = rng.dirichlet(np.full(n, alpha))
            sc = _objective(a)
            candidates.append((sc, a.copy()))
            if sc < best_score:
                best_score, best_seq = sc, a.copy()
                print(...)
        # sparse random seeds
        for _ in range(3):
            if time.time()>deadline: break
            a = _distribution_random_sparse(n, rng, density=0.01)
            sc = _objective(a)
            candidates.append((sc, a.copy()))
            if sc < best_score:
                best_score, best_seq = sc, a.copy()
                print(...)

        # keep only top 20 candidates overall
        candidates.sort(key=lambda x: x[0])
        candidates = candidates[:20]

    # Stage 2: upsample top candidates to longer lengths
    max_n = 20000
    upsample_scales = [1.0, 1.5, 2.0, 3.0]
    up_candidates = []
    for score, a in candidates[:10]: # top 10
        n0 = a.shape[0]
        for scale in upsample_scales:
            n_new = int(round(n0 * scale))
            if n_new > max_n or n_new <= n0:

            a_up = _interpolate_distribution(a, n_new)
            sc_up = _objective(a_up)
            up_candidates.append((sc_up, a_up))

                best_score, best_seq = sc_up, a_up.copy()
                print(f"new best upscaled len={n_new} score={sc_up:.6f}")

    # Merge candidates and keep top ~12
    all_candidates = candidates + up_candidates
    all_candidates.sort(key=lambda x: x[0])
    top_seeds = [seq for _, seq in all_candidates[:6]]

    # Stage 3: refined gradient descent
    remaining = deadline - time.time()
    if remaining > 30.0:
        per_seed = remaining / max(1, len(top_seeds))
        for idx, seed_seq in enumerate(top_seeds):
            if time.time() >= deadline: break
            seed_deadline = time.time() + per_seed
            a_opt, sc_opt = _exp_gradient_descent(seed_seq, seed_deadline,

    print(f"Finished best score {best_score:.6f} length {len(best_seq)}")

"""
Search for a non‑negative step function (list of heights) that minimizes
the supplied ``evaluate_sequence`` objective.

The algorithm consists of four stages, all respecting the given time
budget (default 1000 s).

1️⃣ **Broad family scan** – deterministic analytic families
   (uniform, arcsine, sine, cosine, quadratic), several parametric
   families (edge‑weighted, symmetric beta, power‑sum) on several
   lengths, plus random Dirichlet / sparse seeds.  The 20 best
   candidates (according to the *exact* objective
   ``2·n·max(conv)`` for a unit‑sum vector) are retained.

2️⃣ **Exact exponentiated‑gradient refinement** – each retained seed is
   refined with a mirror‑descent (exponentiated‑gradient) method that
   uses the *exact* sub‑gradient of the max‑convolution entry.
   An AdaGrad‑style per‑coordinate step size is employed
   (η_i = η₀ / √(∑_τ g_i(τ)²)).  The iterate is always projected onto
   the probability simplex.  The best improvement is recorded.

3️⃣ **Local uniform‑mix sweep** – after refinement we try a few convex
   mixtures with the uniform distribution (λ∈[0,0.2]) to exploit the
   convexity of the objective.

4️⃣ **Greedy pairwise polishing** – a cheap hill‑climbing phase moves a
   random fraction of mass from one index to another, keeping only
   strictly improving moves.

If a variable ``height_sequence_1`` is present in the global scope the
previous construction is also injected as a seed.

The routine returns a plain ``list[float]`` (already normalised to sum = 1)
which can be fed directly to ``evaluate_sequence``.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Utility: projection onto the probability simplex Σx = s, x ≥ 0
def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Euclidean projection of vector v onto {x≥0, Σx=s}."""
    if s < 0:
        raise ValueError("Simplex radius must be non‑negative")
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.where(u - cssv / (np.arange(1, len(v) + 1)) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    return np.maximum(v - theta, 0.0)


# ----------------------------------------------------------------------
# Fast full self‑convolution via real FFT
def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Return full convolution a*a (length 2·n‑1) using a real FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1) - 1).bit_length()   # next power of two
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[: 2 * n - 1]


# ----------------------------------------------------------------------
# Exact objective for a unit‑sum vector: 2·n·max(conv)
def _objective(a: np.ndarray) -> float:
    """Exact objective (2·n·max self‑convolution) for Σa = 1."""
    n = a.shape[0]
    b = _fft_self_convolve(a)
    return float(2.0 * n * b.max())


# ----------------------------------------------------------------------
# Analytic families (all return a unit‑sum vector)
def _distribution_uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _distribution_arcsine(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    vals /= vals.sum()
    return vals


def _distribution_sine(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64) + 0.5
    w = np.sin(np.pi * i / n)
    w /= w.sum()
    return w


def _distribution_cosine(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64) + 0.5
    w = 1.0 - np.cos(np.pi * i / n)
    w /= w.sum()
    return w


def _distribution_quadratic(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    x = (2.0 * i) / (n - 1) - 1.0
    w = 1.0 - x * x
    w = np.clip(w, 0.0, None)
    w /= w.sum()
    return w


def _distribution_edge_weighted(n: int, p: float) -> np.ndarray:
    """
    Edge‑weighted family: w_i ∝ ((i+½)*(n‑i‑½))^{‑p}.
    """
    i = np.arange(n, dtype=np.float64) + 0.5
    j = (n - np.arange(n, dtype=np.float64) - 0.5)
    w = (i * j) ** (-p)
    w = np.clip(w, 0.0, None)
    total = w.sum()
    if total > 0:
        w /= total
    else:
        w = np.full(n, 1.0 / n)
    return w


def _distribution_beta_sym(n: int, p: float) -> np.ndarray:
    """Symmetric Beta(p,p) on a regular grid."""
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    # avoid log(0) for extreme p
    with np.errstate(divide='ignore'):
        w = np.exp((p - 1) * np.log(x) + (p - 1) * np.log(1.0 - x))
    if w.sum() == 0.0 or np.isnan(w).any():
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _distribution_beta(n: int, p: float, q: float) -> np.ndarray:
    """General (asymmetric) Beta(p,q) on a regular grid."""
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    with np.errstate(divide='ignore'):
        w = np.exp((p - 1) * np.log(x) + (q - 1) * np.log(1.0 - x))
    if w.sum() == 0.0 or np.isnan(w).any():
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _distribution_power_sum(n: int, p: float) -> np.ndarray:
    """w_i ∝ (i+1)^{‑p} + (n‑i)^{‑p}."""
    i = np.arange(1, n + 1, dtype=np.float64)
    w = i ** (-p) + (n + 1 - i) ** (-p)
    w = np.clip(w, 0.0, None)
    w /= w.sum()
    return w


def _distribution_random_sparse(
    n: int, rng: np.random.Generator, density: float = 0.01
) -> np.ndarray:
    """Sparse random vector with a prescribed density of non‑zeros."""
    k = max(2, int(n * density))
    idx = rng.choice(n, size=k, replace=False)
    vals = np.zeros(n, dtype=np.float64)
    vals[idx] = rng.random(k)
    if vals.sum() == 0.0:
        vals[0] = 1.0
    vals /= vals.sum()
    return vals


def _mix_with_uniform(base: np.ndarray, lam: float) -> np.ndarray:
    """Convex mixture λ·uniform + (1‑λ)·base."""
    n = base.shape[0]
    uniform = np.full(n, 1.0 / n, dtype=np.float64)
    return lam * uniform + (1.0 - lam) * base


# ----------------------------------------------------------------------
# Interpolation (used for up‑sampling a candidate)
def _interpolate_distribution(a: np.ndarray, target_n: int) -> np.ndarray:
    """Linear interpolation of a probability vector to length target_n."""
    if target_n == a.shape[0]:
        return a.copy()
    src_x = np.linspace(0, 1, a.shape[0], endpoint=False) + 0.5 / a.shape[0]
    tgt_x = np.linspace(0, 1, target_n, endpoint=False) + 0.5 / target_n
    values = np.interp(tgt_x, src_x, a, left=0.0, right=0.0)
    # clip potential tiny negatives caused by interpolation noise
    values = np.clip(values, 0.0, None)
    total = values.sum()
    if total == 0.0:
        values = np.full(target_n, 1.0 / target_n, dtype=np.float64)
    else:
        values = values / total
    return values


# ----------------------------------------------------------------------
# Sub‑gradient of the max‑convolution entry
def _max_convolution_subgradient(a: np.ndarray) -> np.ndarray:
    """
    Sub‑gradient of  max_k Σ_i a_i a_{k-i}.
    Returns a vector of the same shape as ``a``.
    """
    conv = _fft_self_convolve(a)
    max_val = conv.max()
    tol = 1e-12 * max_val
    active = np.where(conv >= max_val - tol)[0]
    n = a.shape[0]
    grad = np.zeros_like(a)
    for k in active:
        lo = max(0, k - (n - 1))
        hi = min(n - 1, k)
        idx = np.arange(lo, hi + 1)
        grad[idx] += 2.0 * a[k - idx]
    if active.size:
        grad /= active.size
    return grad


# ----------------------------------------------------------------------
# Exponentiated‑gradient descent with AdaGrad‑style step sizes
def _exp_gradient_descent(
    a0: np.ndarray,
    time_limit: float,
    best_seq: np.ndarray,
    best_score: float,
    eta0: float = 0.6,
    report_every: int = 300,
) -> tuple[np.ndarray, float]:
    """
    Mirror‑descent on the exact max‑convolution objective.
    Uses AdaGrad‑style per‑coordinate steps.
    """
    a = a0.copy()
    n = a.shape[0]
    G = np.zeros_like(a)          # accumulate squared grads
    it = 0
    eps = 1e-12

    while time.time() < time_limit:
        it += 1
        grad = _max_convolution_subgradient(a)
        G += grad * grad
        # AdaGrad step (element‑wise)
        step = eta0 / (np.sqrt(G) + eps)
        a = a * np.exp(-step * grad)
        # project onto the simplex (here just renormalise)
        s = a.sum()
        if s > 0:
            a = a / s
        else:   # numerical safeguard
            a = np.full(n, 1.0 / n, dtype=np.float64)

        if it % report_every == 0:
            cur = _objective(a)
            if cur < best_score - 1e-14:
                best_score = cur
                best_seq = a.copy()
                print(f"[{time.time():.1f}] exp‑gd it={it} new best {best_score:.6f}")

    return best_seq, best_score


# ----------------------------------------------------------------------
# Greedy pairwise polishing (simple hill‑climber)
def _pairwise_swap(
    a0: np.ndarray,
    time_limit: float,
    best_seq: np.ndarray,
    best_score: float,
    max_trials: int | None = None,
) -> tuple[np.ndarray, float]:
    """
    Randomly move a fraction of mass from one index to another.
    Only strictly improving moves are kept.
    """
    a = a0.copy()
    n = a.shape[0]
    trials = 0
    while time.time() < time_limit:
        if max_trials is not None and trials >= max_trials:
            break
        trials += 1

        src = np.where(a > 1e-14)[0]
        if src.size == 0:
            break
        i = np.random.choice(src)
        j = np.random.randint(n - 1)
        if j >= i:
            j += 1

        delta = a[i] * np.random.rand() * 0.45   # up to 45 % of source mass
        if delta < 1e-15:
            continue

        new_a = a.copy()
        new_a[i] -= delta
        new_a[j] += delta

        cur = _objective(new_a)
        if cur < best_score - 1e-14:
            a = new_a
            best_score = cur
            best_seq = a.copy()
            print(
                f"[{time.time():.1f}] swap i={i}→j={j} δ={delta:.6g} new best {best_score:.6f}"
            )
    return best_seq, best_score


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: int = 1000) -> list[float]:
    """
    Main entry point required by the competition harness.
    Returns a list of non‑negative floats (already normalised to sum = 1).
    """
    rng = np.random.default_rng(seed)
    start = time.time()
    # keep a small safety margin for final polishing
    deadline = start + budget_s - 5.0

    overall_best_seq = None
    overall_best_score = float("inf")
    top_candidates: list[tuple[float, np.ndarray]] = []   # (score, seq)

    # ------------------------------------------------------------------
    # 1. Scan a rich collection of analytic families on several lengths
    # ------------------------------------------------------------------
    lengths = [2048, 4096, 8192, 12000]          # moderate sizes
    # finer p‑grid around the region that performed well in earlier runs
    p_grid = np.concatenate(
        [
            np.logspace(-2, -0.2, 12),          # 0.01 … 0.63
            np.linspace(0.6, 3.0, 12),          # 0.6 … 3.0
            np.linspace(3.0, 8.0, 6),           # 3 … 8  (edge‑heavy tail)
        ]
    )
    lam_grid = np.linspace(0.0, 1.0, 7)          # 0, .166, …, 1

    for n in lengths:
        if time.time() >= deadline:
            break
        print(f"[{time.time():.1f}] scanning length {n}")

        # ---------- deterministic families ----------
        for base_fn in (
            _distribution_uniform,
            _distribution_arcsine,
            _distribution_sine,
            _distribution_cosine,
            _distribution_quadratic,
        ):
            cand = base_fn(n)
            sc = _objective(cand)
            top_candidates.append((sc, cand.copy()))
            if sc < overall_best_score:
                overall_best_score, overall_best_seq = sc, cand.copy()
                print(f"[{time.time():.1f}] new global best (det) len={n} score={sc:.6f}")

        # ---------- parametric families ----------
        for p in p_grid:
            if time.time() >= deadline:
                break

            edge_seq = _distribution_edge_weighted(n, p)
            beta_seq = _distribution_beta_sym(n, p)
            pow_seq = _distribution_power_sum(n, p)

            for base in (edge_seq, beta_seq, pow_seq):
                for lam in lam_grid:
                    if time.time() >= deadline:
                        break
                    cand = _mix_with_uniform(base, lam)
                    sc = _objective(cand)
                    top_candidates.append((sc, cand.copy()))
                    if sc < overall_best_score:
                        overall_best_score, overall_best_seq = sc, cand.copy()
                        print(
                            f"[{time.time():.1f}] new global best (param) len={n} p={p:.3g} lam={lam:.2f} score={sc:.6f}"
                        )

        # ---------- random Dirichlet seeds ----------
        for _ in range(5):
            if time.time() >= deadline:
                break
            alpha = rng.uniform(0.01, 2.0)
            cand = rng.dirichlet(np.full(n, alpha))
            sc = _objective(cand)
            top_candidates.append((sc, cand.copy()))
            if sc < overall_best_score:
                overall_best_score, overall_best_seq = sc, cand.copy()
                print(f"[{time.time():.1f}] new global best (dirichlet) len={n} score={sc:.6f}")

        # ---------- sparse random seeds ----------
        for _ in range(5):
            if time.time() >= deadline:
                break
            cand = _distribution_random_sparse(n, rng, density=0.008)
            sc = _objective(cand)
            top_candidates.append((sc, cand.copy()))
            if sc < overall_best_score:
                overall_best_score, overall_best_seq = sc, cand.copy()
                print(f"[{time.time():.1f}] new global best (sparse) len={n} score={sc:.6f}")

        # keep only the most promising ~30 across all lengths processed so far
        top_candidates.sort(key=lambda x: x[0])
        top_candidates = top_candidates[:30]

    # ------------------------------------------------------------------
    # 2. Incorporate any previously discovered construction (if present)
    # ------------------------------------------------------------------
    try:
        prev = np.asarray(height_sequence_1, dtype=np.float64)  # type: ignore[name-defined]
        if prev.ndim == 1 and prev.size > 0:
            prev = np.maximum(prev, 0.0)
            total = prev.sum()
            if total == 0.0:
                prev = _distribution_uniform(prev.size)
            else:
                prev = prev / total
            sc = _objective(prev)
            if sc < overall_best_score:
                overall_best_score, overall_best_seq = sc, prev.copy()
                print(f"[{time.time():.1f}] previous construction improves to {sc:.6f}")
            top_candidates.append((sc, prev.copy()))
    except Exception:
        pass

    # ------------------------------------------------------------------
    # 3. Fallback (very unlikely)
    # ------------------------------------------------------------------
    if overall_best_seq is None:
        n = 5000
        overall_best_seq = _distribution_uniform(n)
        overall_best_score = _objective(overall_best_seq)
        print(
            f"[{time.time():.1f}] fallback uniform length={n} score={overall_best_score:.6f}"
        )

    # ------------------------------------------------------------------
    # 4. Optional up‑sampling of the very best seeds (helps discretisation)
    # ------------------------------------------------------------------
    max_n = 20000
    upscaled: list[tuple[float, np.ndarray]] = []
    for score, seq in top_candidates[:8]:   # only the top few for up‑sampling
        src_n = seq.shape[0]
        for factor in (1.5, 2.0, 3.0):
            tgt_n = int(round(src_n * factor))
            if tgt_n <= src_n or tgt_n > max_n:
                continue
            cand = _interpolate_distribution(seq, tgt_n)
            sc = _objective(cand)
            upscaled.append((sc, cand))
            if sc < overall_best_score:
                overall_best_score, overall_best_seq = sc, cand.copy()
                print(f"[{time.time():.1f}] up‑scaled len={tgt_n} score={sc:.6f}")

    # Merge up‑scaled candidates and keep a concise list for refinement
    all_candidates = top_candidates + upscaled
    all_candidates.sort(key=lambda x: x[0])
    seed_vectors = [seq for _, seq in all_candidates[:6]]   # six best seeds

    # ------------------------------------------------------------------
    # 5. Exact exponentiated‑gradient refinement (AdaGrad)
    # ------------------------------------------------------------------
    remaining = deadline - time.time()
    if remaining > 30.0 and seed_vectors:
        # allocate ~70 % of the remaining time to refinement
        refine_budget = remaining * 0.70
        per_seed = refine_budget / max(1, len(seed_vectors))
        print(f"[{time.time():.1f}] starting refinement on {len(seed_vectors)} seeds")
        for idx, seed_seq in enumerate(seed_vectors):
            if time.time() >= deadline:
                break
            seed_deadline = time.time() + per_seed
            a_opt, sc_opt = _exp_gradient_descent(
                seed_seq,
                seed_deadline,
                best_seq=overall_best_seq,
                best_score=overall_best_score,
                eta0=0.7,
                report_every=400,
            )
            if sc_opt < overall_best_score - 1e-12:
                overall_best_score, overall_best_seq = sc_opt, a_opt.copy()
                print(
                    f"[{time.time():.1f}] after refinement seed {idx} new best {overall_best_score:.6f}"
                )

    # ------------------------------------------------------------------
    # 6. Local uniform‑mix sweep (tiny λ’s) – cheap convex perturbations
    # ------------------------------------------------------------------
    for lam in np.linspace(0.0, 0.2, 9):
        cand = _mix_with_uniform(overall_best_seq, lam)
        sc = _objective(cand)
        if sc < overall_best_score - 1e-13:
            overall_best_score, overall_best_seq = sc, cand.copy()
            print(
                f"[{time.time():.1f}] uniform‑mix λ={lam:.3f} improves to {overall_best_score:.6f}"
            )

    # ------------------------------------------------------------------
    # 7. Final greedy polishing (pairwise mass transfers)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        print(f"[{time.time():.1f}] final greedy polishing")
        a_opt, sc_opt = _pairwise_swap(
            overall_best_seq.copy(),
            deadline,
            best_seq=overall_best_seq,
            best_score=overall_best_score,
        )
        if sc_opt < overall_best_score - 1e-12:
            overall_best_score, overall_best_seq = sc_opt, a_opt.copy()
            print(
                f"[{time.time():.1f}] after polishing new best {overall_best_score:.6f}"
            )

    # ------------------------------------------------------------------
    # 8. Report and return
    # ------------------------------------------------------------------
    print(
        f"Finished: best score (2·n·max(conv)) = {overall_best_score:.6f}, length = {len(overall_best_seq)}"
    )
    # The list returned is plain Python floats (already on the simplex)
    return overall_best_seq.tolist()
