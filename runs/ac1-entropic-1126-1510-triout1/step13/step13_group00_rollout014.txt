def _gradient_active(a, active_idx):
    n = len(a)
    active = np.zeros(2*n-1, dtype=np.float64)
    active[active_idx] = 1.0
    a_rev = a[::-1]
    size = 1 << ((len(active)+len(a_rev)-1 - 1).bit_length())
    A = np.fft.rfft(active, n=size)
    B = np.fft.rfft(a_rev, n=size)
    conv = np.fft.irfft(A * B, n=size)
    # conv length = size (since irfft returns size)
    # we need slice conv[n-1 : n-1+n]
    start = n-1
    grad = 2.0 * conv[start:start+n]
    return grad

def _softmax_gradient(a, epsilon):
    n = len(a)
    conv = _fft_self_convolve(a) # shape 2n-1
    max_c = conv.max()
    # compute weights
    # scaling for numerical stability:
    exp_vals = np.exp((conv - max_c) / epsilon)
    sum_exp = exp_vals.sum()
    if sum_exp == 0 or np.isnan(sum_exp):
        w = np.zeros_like(conv)
        # fallback to uniform (active set)
        # set w for indices with max conv maybe
        w[conv == max_c] = 1.0 / np.sum(conv == max_c)
    else:
        w = exp_vals / sum_exp

    # gradient = 2 * convolution of a_rev and w
    # compute via FFT
    a_rev = a[::-1]
    # length needed = len(a_rev)+len(w)-1 = n + (2n-1) -1 = 3n-2
    size = 1 << ((len(a_rev)+len(w)-1 - 1).bit_length())
    A = np.fft.rfft(w, n=size)
    B = np.fft.rfft(a_rev, n=size)
    conv_full = np.fft.irfft(A * B, n=size)
    start = n-1
    grad = 2.0 * conv_full[start:start+n]
    return grad

def _smooth_descent(a, deadline, best_seq, best_score, start_eta=0.5, epsilon0=0.1, epsilon_decay=0.9995, eval_interval=100, patience=2000):
    t=0
    no_improve=0
    eta0=start_eta
    eps=epsilon0
    while time.time()<deadline and t<300000:
        t+=1
        grad = _softmax_gradient(a, eps)
        # learning rate schedule
        eta = eta0 / math.sqrt(t)
        a = a * np.exp(-eta * grad)  # exponentiated gradient
        # projection onto simplex
        sum_a = a.sum()
        if sum_a == 0:
            a = np.full_like(a, 1.0/a.shape[0])
        else:
            a = a / sum_a
        # optionally clip small numbers
        if t%eval_interval == 0:
            cur = _objective(a) # our internal objective using sum=1
            if cur + 1e-14 < best_score:
                best_score = cur
                best_seq = a.copy()
                no_improve = 0
                print(f"[{time.time():.1f}] smooth descent iter={t} new best {best_score:.6f}")
            else:
                no_improve += 1
        # epsilon decay
        eps = max(eps * epsilon_decay, 1e-6)
        if no_improve >= patience:
            # jitter reinject
            jitter = np.random.default_rng().exponential(scale=0.001, size=a.shape)
            a = a + jitter
            a = np.maximum(a,0)
            a/= a.sum()
            no_improve=0
    return best_seq, best_score

def _weighted_gradient(a, w):
    # w length 2n-1
    a_rev = a[::-1]
    size = 1 << ((len(a_rev)+len(w)-1 -1).bit_length())
    # compute rfft of both
    Aw = np.fft.rfft(w, n=size)
    Aa = np.fft.rfft(a_rev, n=size)
    conv = np.fft.irfft(Aw * Aa, n=size)
    start = len(a)-1
    grad = 2.0 * conv[start:start+len(a)]
    return grad

        # project onto simplex
        s = a.sum()

            a /= s

"""
Propose a non‑negative step‑function (list of heights) that minimises
``evaluate_sequence``.

Algorithm
---------

1. **Seed generation** – sample a diverse pool of analytic families
   (uniform, arcsine, Beta, edge‑weighted, sine‑weighted, tent, random
   Dirichlet / sparse vectors) for several moderate lengths (≈2k–5k).
   Keep the best few candidates (low exact objective
   ``2·n·max(conv(a,a))``).

2. **Active‑set exponentiated‑gradient descent** – for each kept seed run a
   mirror‑descent style optimisation.  In every iteration the full
   self‑convolution is computed by FFT; a binary mask selects all
   convolution entries whose value is within a fixed fraction of the
   current maximum.  The gradient of the max‑norm (sub‑gradient) is then
   obtained by a second FFT that convolves the mask with the reversed
   distribution.  The update ``a ← a·exp(‑η·grad)`` preserves positivity,
   after which the vector is projected onto the probability simplex.
   A small jitter is added when no improvement is seen for a while.

3. **Upsampling** – linearly interpolate the current best distribution to
   a larger length (up to 50 000 points) and repeat the same descent; a
   larger discretisation gives the algorithm more freedom to lower the
   peak of the convolution.

4. **Local cleanup** – a greedy pairwise mass‑transfer sweep followed by a
   cheap simulated‑annealing style local search removes any remaining
   spikes.

All stages respect a total wall‑clock budget (default 1000 s) and print
progress information.  The returned list contains plain Python floats;
``evaluate_sequence`` will apply the required clipping.
"""

import time
import math
import numpy as np
from numpy.fft import rfft, irfft

# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x≥0, sum(x)=s}."""
    if s < 0:
        raise ValueError("Simplex radius must be non‑negative")
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho_arr = np.where(u - cssv / (np.arange(1, len(v) + 1)) > 0)[0]
    if rho_arr.size == 0:
        theta = 0.0
    else:
        rho = rho_arr[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a (real FFT, length 2·n‑1)."""
    n = a.shape[0]
    need = 2 * n - 1
    size = 1 << ((need - 1).bit_length())   # next power of two
    A = rfft(a, n=size)
    conv = irfft(A * A, n=size)
    return conv[:need]


# ----------------------------------------------------------------------
def _objective(a: np.ndarray) -> float:
    """Exact objective 2·n·max(conv(a,a)) for a with sum(a)=1."""
    n = a.shape[0]
    b = _fft_self_convolve(a)
    return 2.0 * n * b.max()

# Analytic families ----------------------------------------------------
def _distribution_uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _distribution_arcsine(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    vals /= vals.sum()
    return vals


def _distribution_beta_sym(n: int, p: float) -> np.ndarray:
    """Symmetric Beta(p,p) on a uniform grid."""
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    w = np.exp((p - 1) * np.log(x) + (p - 1) * np.log(1.0 - x))
    if w.sum() == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _distribution_beta_asym(n: int, a: float, b: float) -> np.ndarray:
    """Asymmetric Beta(a,b) on a uniform grid."""
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    logw = (a - 1) * np.log(x) + (b - 1) * np.log(1.0 - x)
    w = np.exp(logw - np.max(logw))      # shift for stability
    if w.sum() == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _distribution_edge_weighted(n: int, p: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64) + 0.5
    j = (n - np.arange(n, dtype=np.float64) - 0.5)
    w = (i * j) ** (-p)
    if w.sum() == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _distribution_edge_inversepow(n: int, p: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64) + 0.5
    j = (n - np.arange(n, dtype=np.float64) - 0.5)
    w = i ** (-p) + j ** (-p)
    if w.sum() == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _distribution_sine_weighted(n: int, p: float) -> np.ndarray:
    theta = np.pi * (np.arange(n, dtype=np.float64) + 0.5) / n
    w = np.sin(theta) ** p
    if w.sum() == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _distribution_tent(n: int) -> np.ndarray:
    """Symmetric tent (triangular) distribution."""
    w = np.minimum(np.arange(1, n + 1), np.arange(n, 0, -1)).astype(np.float64)
    w /= w.sum()
    return w


def _distribution_random_sparse(
    n: int, rng: np.random.Generator, density: float = 0.006
) -> np.ndarray:
    """Very sparse random vector (few non‑zeros)."""
    k = max(2, int(n * density))
    idx = rng.choice(n, size=k, replace=False)
    vals = np.zeros(n, dtype=np.float64)
    vals[idx] = rng.random(k)
    if vals.sum() == 0.0:
        vals[0] = 1.0
    vals /= vals.sum()
    return vals


def _distribution_dirichlet(n: int, rng: np.random.Generator) -> np.ndarray:
    """Draw from a symmetric Dirichlet with a random concentration."""
    alpha = rng.uniform(0.01, 2.0)
    a = rng.dirichlet(np.full(n, alpha))
    return a.astype(np.float64)


def _mix_with_uniform(base: np.ndarray, lam: float) -> np.ndarray:
    """lam·U + (1‑lam)·base, where U is the uniform distribution."""
    n = base.shape[0]
    uniform = np.full(n, 1.0 / n, dtype=np.float64)
    return lam * uniform + (1.0 - lam) * base


# ----------------------------------------------------------------------
def _weighted_gradient(a: np.ndarray, w: np.ndarray) -> np.ndarray:
    """
    Gradient of the max‑norm (or a generic weighted sum) w.r.t. ``a``.
    w : weight vector of length (2·n‑1).  The gradient is
        2·(a_rev * w)   (linear convolution of the reversed a with w).
    """
    n = a.shape[0]
    a_rev = a[::-1]
    need = n + w.shape[0] - 1
    size = 1 << ((need - 1).bit_length())
    A = rfft(a_rev, n=size)
    W = rfft(w, n=size)
    conv = irfft(A * W, n=size)
    start = n - 1
    grad = 2.0 * conv[start:start + n]
    return grad


def _active_set_descent(
    a: np.ndarray,
    deadline: float,
    best_seq: np.ndarray,
    best_score: float,
    lr0: float = 0.6,
    active_frac: float = 0.20,
    max_active: int | None = None,
    eval_interval: int = 80,
    patience: int = 1500,
) -> tuple[np.ndarray, float]:
    """
    Exponentiated‑gradient descent where the gradient is built from an
    active set of convolution entries whose value is within ``active_frac``
    of the current maximum.  The gradient is computed by a single FFT per
    iteration, making the method fast even for large ``n``.
    """
    n = a.shape[0]
    t = 0
    no_improve = 0
    rng = np.random.default_rng()
    while time.time() < deadline and t < 500_000:
        t += 1

        conv = _fft_self_convolve(a)
        max_c = conv.max()
        thresh = max_c * (1.0 - active_frac)

        # binary mask of the active indices
        w = (conv >= thresh).astype(np.float64)
        if max_active is not None and w.sum() > max_active:
            # keep only the *max_active* largest entries
            idx = np.argpartition(conv, -max_active)[-max_active:]
            w[:] = 0.0
            w[idx] = 1.0

        grad = _weighted_gradient(a, w)

        eta = lr0 / math.sqrt(t)
        a = a * np.exp(-eta * grad)          # exponentiated‑gradient step
        a = _project_onto_simplex(a)         # ensure a ≥ 0 and sum(a)=1

        if t % eval_interval == 0:
            cur = _objective(a)
            if cur + 1e-14 < best_score:
                best_score, best_seq = cur, a.copy()
                no_improve = 0
                print(
                    f"[{time.time():.1f}] active descent it={t:,} new best {best_score:.6f}"
                )
            else:
                no_improve += 1

        if no_improve >= patience:
            # shake out of a plateau with a tiny random jitter
            jitter = rng.exponential(scale=0.003, size=n)
            a = a + jitter
            a = _project_onto_simplex(a)
            no_improve = 0

    return best_seq, best_score


# ----------------------------------------------------------------------
def _pairwise_swap_local(
    a: np.ndarray,
    deadline: float,
    best_seq: np.ndarray,
    best_score: float,
    max_trials: int | None = None,
) -> tuple[np.ndarray, float]:
    """
    Greedy transfer of a random fraction of mass from one coordinate to
    another.  The change is kept only if the exact objective improves.
    """
    n = a.shape[0]
    trials = 0
    rng = np.random.default_rng()
    while time.time() < deadline:
        if max_trials is not None and trials >= max_trials:
            break
        trials += 1

        src = np.where(a > 1e-12)[0]
        if src.size == 0:
            break
        i = rng.choice(src)

        j = rng.integers(n - 1)
        if j >= i:
            j += 1

        delta = a[i] * rng.random() * 0.33
        if delta < 1e-14:
            continue

        new_a = a.copy()
        new_a[i] -= delta
        new_a[j] += delta

        cur = _objective(new_a)
        if cur + 1e-14 < best_score:
            a = new_a
            best_score = cur
            best_seq = a.copy()
            print(
                f"[{time.time():.1f}] swap {i}->{j} δ={delta:.3e} new best {best_score:.6f}"
            )
    return best_seq, best_score


# ----------------------------------------------------------------------
def _random_local_search(
    a: np.ndarray,
    deadline: float,
    best_seq: np.ndarray,
    best_score: float,
    init_temp: float = 0.008,
    decay: float = 0.9995,
) -> tuple[np.ndarray, float]:
    """Cheap simulated‑annealing style local search."""
    n = a.shape[0]
    rng = np.random.default_rng()
    T = init_temp
    while time.time() < deadline:
        i = rng.integers(n)
        j = rng.integers(n - 1)
        if j >= i:
            j += 1
        delta = a[i] * rng.random() * 0.28
        if delta < 1e-14:
            continue
        new_a = a.copy()
        new_a[i] -= delta
        new_a[j] += delta
        cur = _objective(new_a)
        if cur + 1e-14 < best_score:
            a = new_a
            best_score = cur
            best_seq = a.copy()
            print(
                f"[{time.time():.1f}] SA improve {i}->{j} Δ={delta:.3e} best {best_score:.6f}"
            )
        else:
            if rng.random() < math.exp(-(cur - best_score) / max(T, 1e-12)):
                a = new_a
        T *= decay
    return best_seq, best_score


# ----------------------------------------------------------------------
def _upsample(seq: np.ndarray, new_len: int) -> np.ndarray:
    """Linear interpolation of a simplex‑vector to a larger length."""
    old_len = seq.shape[0]
    if new_len == old_len:
        return seq.copy()
    x_old = np.linspace(0.0, 1.0, old_len)
    x_new = np.linspace(0.0, 1.0, new_len)
    new_seq = np.interp(x_new, x_old, seq)
    new_seq = np.maximum(new_seq, 0.0)
    new_seq /= new_seq.sum()
    return new_seq

            sc = _objective(base)

            if sc < overall_best_score:
                overall_best_score, overall_best_seq = sc, base.copy()
                print(f"[{time.time():.1f}] new global best (deterministic) {sc:.6f}")

        # parameterised families
        for p in p_grid:

            edge = _distribution_edge_weighted(n, p)

            sc = _objective(cand)

            if sc < overall_best_score:
