"""
propose_candidate.py

Search for a non‑negative step function (probability vector) that
minimises the evaluation function used in the competition:

    2·n·max(conv(a,a)) / (∑a)²

The algorithm works in three stages under a 1000 s budget:

1. **Diversified seeding** – generate a rich collection of candidates
   from several analytic families (uniform, arcsine, Hann, edge‑weighted,
   generalized‑normal, Dirichlet, sparse) for several lengths
   (2000 → 4000 → 8000 → 16000).  All vectors are normalised to sum = 1
   and the best ~30 are retained.

2. **Mirror‑descent polishing** – apply an exponentiated‑gradient (mirror
   descent) on the *exact* objective.  The sub‑gradient is taken from a
   randomly chosen *active* convolution entry (any entry whose value is
   within a tiny tolerance of the current maximum).  The step size follows
   ηₜ = η₀/√t with η₀≈0.8.  Short runs are performed on every retained
   seed; then a longer run refines the current global best.

3. **Upsampling + final simulated annealing** – if time permits the
   best vector is linearly up‑sampled (doubling its length) and polished
   again with mirror descent.  Finally a pairwise‑mass‑transfer annealing
   phase explores the neighbourhood aggressively (higher initial

All operations respect the 1000 s budget (2 s safety margin).  The

# ----------------------------------------------------------------------
# Simplex utilities (kept for possible future use)
# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """ Euclidean projection of a vector onto the simplex {x≥0, Σx=s}. """
    if s < 0:
        raise ValueError("simplex radius must be non‑negative")
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.where(u - cssv / (np.arange(1, len(v) + 1)) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Convolution helpers (real FFT, linear convolution a*a)
# ----------------------------------------------------------------------
def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """ Full linear self‑convolution a*a using a real FFT. """
    n = a.shape[0]
    size = 1 << ((2 * n - 1) - 1).bit_length()   # next power‑of‑two
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[: 2 * n - 1]


# ----------------------------------------------------------------------
# Exact objective (works on probability vectors, i.e. Σa = 1)
# ----------------------------------------------------------------------
def _objective(a: np.ndarray) -> float:
    """ Return 2·n·max(conv(a,a)) for a probability vector a. """
    n = a.shape[0]
    b = _fft_self_convolve(a)
    max_b = b.max()
    if max_b < 0:
        max_b = 0.0
    return float(2.0 * n * max_b)


# ----------------------------------------------------------------------
# Analytic families (all return a probability vector, sum=1)
# ----------------------------------------------------------------------
def _distribution_uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _distribution_arcsine(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    vals /= vals.sum()
    return vals


def _distribution_hann(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    w = 0.5 - 0.5 * np.cos(2.0 * np.pi * i / (n - 1))
    w_sum = w.sum()
    if w_sum == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w_sum
    return w


def _distribution_edge_weighted(n: int, p: float) -> np.ndarray:
    """ Weight ∝ (i·j)^{‑p}, i,j distance to the edges (1‑based centres). """
    i = np.arange(n, dtype=np.float64) + 0.5
    j = (n - np.arange(n, dtype=np.float64) - 0.5)
    w = (i * j) ** (-p)
    w_sum = w.sum()
    if w_sum == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w_sum
    return w


def _distribution_generalized_normal(n: int, q: float) -> np.ndarray:
    """ Density ∝ exp(‑|x‑½|^{q}) on [0,1]. """
    i = np.arange(n, dtype=np.float64) + 0.5
    x = i / n - 0.5
    w = np.exp(- (np.abs(x) / 0.5) ** q)
    w_sum = w.sum()
    if w_sum == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w_sum
    return w


def _distribution_random_sparse(
    n: int, rng: np.random.Generator, density: float = 0.01
) -> np.ndarray:
    """ Sparse random vector with a given density. """
    k = max(2, int(n * density))
    idx = rng.choice(n, size=k, replace=False)
    vals = np.zeros(n, dtype=np.float64)
    vals[idx] = rng.random(k)
    if vals.sum() == 0.0:
        vals[0] = 1.0
    vals /= vals.sum()
    return vals


def _mix_with_uniform(base: np.ndarray, lam: float) -> np.ndarray:
    """ Convex combination with the uniform distribution. """
    n = base.shape[0]
    uniform = np.full(n, 1.0 / n, dtype=np.float64)
    return lam * uniform + (1.0 - lam) * base

# Gradient of a specific convolution entry
# ----------------------------------------------------------------------
def _grad_of_conv_at_k(a: np.ndarray, k: int) -> np.ndarray:
    """ Gradient of (a*a)_k w.r.t. a (exact, not a sub‑gradient). """
    n = a.shape[0]
    grad = np.zeros_like(a)
    if k < n:
        # overlap of length k+1
        grad[: k + 1] = 2.0 * a[k::-1]
    else:
        start = k - (n - 1)
        grad[start:] = 2.0 * a[start:][::-1]
    return grad


# ----------------------------------------------------------------------
# Mirror‑descent (exponentiated gradient) on the exact objective
# ----------------------------------------------------------------------
def _mirror_descent(
    a: np.ndarray,
    deadline: float,
    best_seq: np.ndarray,
    best_score: float,
    rng: np.random.Generator,
    eta0: float = 0.8,
    eval_every: int = 60,
) -> tuple[np.ndarray, float]:
    """
    Exponentiated‑gradient descent with a *multi‑active* sub‑gradient.
    At each iteration a random index among those whose convolution value
    is within a tiny tolerance of the current maximum is selected.
    """
    n = a.shape[0]
    t = 0
    while time.time() < deadline:
        t += 1

        conv = _fft_self_convolve(a)
        max_val = conv.max()
        # Relative tolerance for the active set (≈ 1e‑6 of the max)
        tol = max_val * 1e-6 + 1e-12
        active = np.where(conv >= max_val - tol)[0]
        if active.size == 0:
            idx = int(np.argmax(conv))
        else:
            idx = rng.choice(active)

        grad = _grad_of_conv_at_k(a, int(idx))

        eta = eta0 / math.sqrt(t)
        a = a * np.exp(-eta * grad)          # stays non‑negative
        s = a.sum()
        if s == 0.0:
            a = np.full(n, 1.0 / n, dtype=np.float64)
        else:
            a /= s

        if t % eval_every == 0:
            cur = _objective(a)
            if cur < best_score - 1e-14:
                best_score = cur
                best_seq = a.copy()
                print(f"[{time.time():.1f}] MD it={t} new global best {best_score:.6f}")

    return best_seq, best_score


# ----------------------------------------------------------------------
# Simulated‑annealing style pairwise mass transfers (more aggressive)
# ----------------------------------------------------------------------
def _pairwise_annealing(
    a: np.ndarray,
    deadline: float,
    best_seq: np.ndarray,
    best_score: float,
    rng: np.random.Generator,
) -> tuple[np.ndarray, float]:
    """
    Randomly move a fraction of mass between two bins.
    Early in the run the temperature is 10 % of the current best score,
    linearly cooling to 0.
    """
    n = a.shape[0]
    start_time = time.time()
    total_time = deadline - start_time
    while time.time() < deadline:
        # pick a source with non‑negligible mass
        src_candidates = np.where(a > 1e-12)[0]
        if src_candidates.size == 0:
            break
        i = rng.choice(src_candidates)

        # pick a distinct destination uniformly
        j = rng.integers(n - 1)
        if j >= i:
            j += 1

        # transfer up to 40 % of the source mass
        delta = a[i] * rng.random() * 0.4
        if delta < 1e-16:
            continue

        new_a = a.copy()
        new_a[i] -= delta
        new_a[j] += delta

        cur = _objective(new_a)

        elapsed = time.time() - start_time
        temperature = max(0.0, 1.0 - elapsed / total_time) * best_score * 0.10

        accept = False
        if cur < best_score - 1e-14:
            accept = True
        elif temperature > 0.0:
            prob = math.exp(-(cur - best_score) / temperature)
            if rng.random() < prob:
                accept = True

        if accept:
            a = new_a
            if cur < best_score - 1e-14:
                best_score = cur
                best_seq = a.copy()
                print(
                    f"[{time.time():.1f}] anneal i={i}→j={j} δ={delta:.3e} new best {best_score:.6f}"
                )
    return best_seq, best_score


# ----------------------------------------------------------------------
# Upsampling (linear interpolation while preserving the simplex)
# ----------------------------------------------------------------------
def _upsample(a: np.ndarray, new_n: int) -> np.ndarray:
    """ Interpolate a probability vector to a larger length. """
    old_n = a.shape[0]
    if new_n == old_n:
        return a.copy()
    x_old = (np.arange(old_n, dtype=np.float64) + 0.5) / old_n
    x_new = (np.arange(new_n, dtype=np.float64) + 0.5) / new_n
    a_new = np.interp(x_new, x_old, a, left=0.0, right=0.0)
    a_new = np.maximum(a_new, 0.0)
    s = a_new.sum()
    if s == 0.0:
        a_new[:] = 1.0 / new_n
    else:
        a_new /= s
    return a_new

        # edge‑weighted grid
        p_vals = np.concatenate((
            np.logspace(-3.0, -0.5, 10, base=10),   # 0.001 … 0.316
            np.logspace(-0.5, 0.0, 10, base=10),    # 0.316 … 1
            np.linspace(0.1, 2.0, 12),               # linear region
            np.logspace(0.0, 0.7, 6, base=10)       # 1 … 5
        ))
        lam_vals = np.linspace(0.0, 1.0, 7)
        for p in p_vals:
            if time.time() >= deadline:
                break
            base = _distribution_edge_weighted(n, float(p))
            for lam in lam_vals:
                if time.time() >= deadline:
                    break
                cand = _mix_with_uniform(base, float(lam))
                sc = _objective(cand)
                cand_pool.append((sc, cand.copy()))
                if sc < best_score:
                    best_score = sc
                    best_seq = cand.copy()
                    print(f"[{time.time():.1f}] new best edge p={p:.3g} λ={lam:.2f} {sc:.6f}")

        # generalized normal family
        for q in np.linspace(0.2, 6.0, 15):
            if time.time() >= deadline:
                break
            cand = _distribution_generalized_normal(n, float(q))
            sc = _objective(cand)
            cand_pool.append((sc, cand.copy()))
            if sc < best_score:
                best_score = sc
                best_seq = cand.copy()
                print(f"[{time.time():.1f}] new best gn q={q:.2f} {sc:.6f}")

        # random Dirichlet seeds (log‑uniform concentration)
        for _ in range(8):
            if time.time() >= deadline:
                break
            log_alpha = rng.uniform(-2.0, 1.0)   # α ∈ [0.01, 10]
            alpha = 10 ** log_alpha
            cand = rng.dirichlet(np.full(n, alpha))
            sc = _objective(cand)
            cand_pool.append((sc, cand.copy()))
            if sc < best_score:
                best_score = sc
                best_seq = cand.copy()
                print(f"[{time.time():.1f}] new best Dirichlet α={alpha:.3g} {sc:.6f}")

        # sparse random seeds
        for _ in range(5):
            if time.time() >= deadline:
                break
            cand = _distribution_random_sparse(n, rng, density=0.02)
            sc = _objective(cand)
            cand_pool.append((sc, cand.copy()))
            if sc < best_score:
                best_score = sc
                best_seq = cand.copy()
                print(f"[{time.time():.1f}] new best sparse {sc:.6f}")

        # keep only the very best seeds overall (30)
        cand_pool.sort(key=lambda x: x[0])
        cand_pool = cand_pool[:30]

    # ------------------------------------------------------------------
    # 2.  Incorporate a previously found construction if present
    # ------------------------------------------------------------------
    try:
        prev = np.asarray(height_sequence_1, dtype=np.float64)  # type: ignore[name-defined]
        if prev.ndim == 1:
            prev = prev / prev.sum()
            sc = _objective(prev)
            if sc < best_score:
                best_score = sc
                best_seq = prev.copy()
                print(f"[{time.time():.1f}] previous best {sc:.6f}")
            cand_pool.append((sc, prev.copy()))
    except Exception:
        pass

    # ------------------------------------------------------------------
    # Ensure a non‑empty result
    # ------------------------------------------------------------------
    if best_seq is None:
        best_seq = _distribution_uniform(4000)
        best_score = _objective(best_seq)
        print(f"[{time.time():.1f}] fallback uniform {best_score:.6f}")

    # ------------------------------------------------------------------
    # 3.  Quick mirror‑descent polishing on all retained seeds
    # ------------------------------------------------------------------
    remaining = deadline - time.time()
    if remaining > 30.0 and cand_pool:
        per_candidate = max(0.2, remaining * 0.25 / max(len(cand_pool), 1))
        print(f"[{time.time():.1f}] quick MD on {len(cand_pool)} seeds (~{per_candidate:.2f}s each)")
        for idx, (score, a0) in enumerate(cand_pool):
            if time.time() >= deadline:
                break
            sub_dead = time.time() + per_candidate
            a_opt, sc_opt = _mirror_descent(
                a0.copy(),
                sub_dead,
                best_seq=best_seq,
                best_score=best_score,
                rng=rng,
                eta0=0.9,
                eval_every=40,
            )
            if sc_opt < best_score - 1e-12:
                best_score = sc_opt
                best_seq = a_opt.copy()
                print(f"[{time.time():.1f}] after quick MD seed {idx} new best {best_score:.6f}")

    # ------------------------------------------------------------------
    # 4.  Longer MD on the current global best
    # ------------------------------------------------------------------
    if time.time() < deadline:
        sub_dead = time.time() + (deadline - time.time()) * 0.30
        a_opt, sc_opt = _mirror_descent(
            best_seq.copy(),
            sub_dead,
            best_seq=best_seq,
            best_score=best_score,
            rng=rng,
            eta0=0.8,
            eval_every=60,
        )
        if sc_opt < best_score - 1e-12:
            best_score = sc_opt
            best_seq = a_opt.copy()
            print(f"[{time.time():.1f}] long MD improved best to {best_score:.6f}")

    # ------------------------------------------------------------------
    # 5.  Upsample & polish (if enough time remains)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        cur_n = best_seq.shape[0]
        target_n = cur_n * 2
        if target_n <= 64000:                     # keep FFT size modest
            print(f"[{time.time():.1f}] upsampling {cur_n} → {target_n}")
            up = _upsample(best_seq, target_n)
            up_sc = _objective(up)
            if up_sc < best_score:
                best_seq = up
                best_score = up_sc
                print(f"[{time.time():.1f}] upsample new best {best_score:.6f}")

            # short MD on the upsampled vector
            sub_dead = time.time() + (deadline - time.time()) * 0.20
            a_opt, sc_opt = _mirror_descent(
                best_seq.copy(),
                sub_dead,
                best_seq=best_seq,
                best_score=best_score,
                rng=rng,
                eta0=0.7,
                eval_every=50,
            )
            if sc_opt < best_score - 1e-12:
                best_seq = a_opt.copy()
                best_score = sc_opt
                print(f"[{time.time():.1f}] MD after upsample new best {best_score:.6f}")

    # ------------------------------------------------------------------
    # 6.  Final simulated‑annealing polishing
    # ------------------------------------------------------------------
    if time.time() < deadline:
        print(f"[{time.time():.1f}] final annealing phase")
        a_opt, sc_opt = _pairwise_annealing(
            best_seq.copy(),
            deadline,
            best_seq=best_seq,
            best_score=best_score,
            rng=rng,
        )
        if sc_opt < best_score - 1e-12:
            best_seq = a_opt.copy()
            best_score = sc_opt
            print(f"[{time.time():.1f}] after annealing new best {best_score:.6f}")

    # ------------------------------------------------------------------
    # 7.  Report and return
    # ------------------------------------------------------------------
    print(
        f"Final best score (objective=2·n·max(conv)): {best_score:.6f} "
        f"(length={len(best_seq)})"
    )
    return best_seq.tolist()
