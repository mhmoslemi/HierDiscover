"""
Propose a non‑negative step‑function (list of heights) that minimises

    evaluate_sequence(seq) = 2·n·max(conv(seq,seq)) / (sum(seq)**2)

The implementation follows three stages:

1️⃣  **Rich family scan** – generate many analytic or random candidates
    (uniform, arcsine, sine, cosine, quadratic, beta, edge‑weighted,
    exponential‑edge, sparse Dirichlet, …) for several lengths and keep
    the best few.

2️⃣  **Annealed smoothed mirror‑descent** – for each promising seed run an
    exponentiated‑gradient descent on a smooth approximation of the
    ``max`` (log‑sum‑exp).  The temperature is slowly decreased, which
    gradually turns the smooth sub‑gradient into the exact sub‑gradient.
    After every update the vector is projected back onto the simplex and
    forced to be mirror‑symmetric (a_i = a_{n‑1‑i}), halving the number of
    free variables and greatly improving convergence.

3️⃣  **Polishing** – a cheap greedy pairwise mass‑transfer hill‑climber
    (and, if time permits, a few exact sub‑gradient steps) finishes the
    optimisation.

The whole search respects the supplied time budget (default 1000 s) and
returns the best sequence found as a plain Python list.
"""

def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x ≥ 0, Σx = s}."""
    if s < 0:
        raise ValueError("Simplex radius must be non‑negative")
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.where(u - cssv / (np.arange(1, len(v) + 1)) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    return np.maximum(v - theta, 0.0)


def _fft_convolve(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full convolution of real vectors x and y using a power‑of‑two FFT."""
    n = len(x) + len(y) - 1
    size = 1 << ((n - 1).bit_length())            # next power of two
    X = np.fft.rfft(x, n=size)
    Y = np.fft.rfft(y, n=size)
    conv = np.fft.irfft(X * Y, n=size)
    return conv[:n]


def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Full self‑convolution a*a using a real FFT."""
    return _fft_convolve(a, a)


def _objective(a: np.ndarray) -> float:
    """
    Exact objective 2·n·max(conv(a,a)) when Σa = 1.
    """
    n = a.shape[0]
    b = _fft_self_convolve(a)
    return float(2.0 * n * b.max())


def _make_symmetric(a: np.ndarray) -> np.ndarray:
    """Force mirror symmetry a_i = a_{n‑1‑i}."""
    return (a + a[::-1]) / 2.0

def _distribution_uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _distribution_arcsine(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    vals /= vals.sum()
    return vals


def _distribution_sine(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64) + 0.5
    w = np.sin(np.pi * i / n)
    w /= w.sum()
    return w


def _distribution_cosine(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64) + 0.5
    w = 1.0 - np.cos(np.pi * i / n)
    w /= w.sum()
    return w


def _distribution_quadratic(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    x = (2.0 * i) / (n - 1) - 1.0
    w = 1.0 - x * x
    w = np.clip(w, 0.0, None)
    w /= w.sum()
    return w


def _distribution_beta(n: int, alpha: float, beta: float) -> np.ndarray:
    """Discretised Beta(α,β) on (0,1) with centre‑point rule."""
    i = np.arange(n, dtype=np.float64) + 0.5
    w = (i / n) ** (alpha - 1) * (1 - i / n) ** (beta - 1)
    w = np.clip(w, 0.0, None)
    total = w.sum()
    if total == 0.0:
        return _distribution_uniform(n)
    w /= total
    return w


def _distribution_edge_weighted(n: int, p: float) -> np.ndarray:
    """w_i ∝ ( (i+0.5)(n-i-0.5) )^{-p}."""
    i = np.arange(n, dtype=np.float64) + 0.5
    j = (n - np.arange(n, dtype=np.float64) - 0.5)
    w = (i * j) ** (-p)
    w_sum = w.sum()
    if w_sum == 0.0:
        return _distribution_uniform(n)
    w /= w_sum
    return w


def _distribution_exponential_edge(n: int, c: float) -> np.ndarray:
    """w_i ∝ exp(-c·i) + exp(-c·(n‑1‑i))."""
    i = np.arange(n, dtype=np.float64)
    w = np.exp(-c * i) + np.exp(-c * (n - 1 - i))
    w /= w.sum()
    return w


def _distribution_random_sparse(
    n: int, rng: np.random.Generator, density: float = 0.01
) -> np.ndarray:
    """Sparse random vector with prescribed density."""
    k = max(2, int(n * density))
    idx = rng.choice(n, size=k, replace=False)
    vals = np.zeros(n, dtype=np.float64)
    vals[idx] = rng.random(k)
    s = vals.sum()
    if s == 0.0:
        vals[0] = 1.0
        s = 1.0
    vals /= s
    return vals


def _mix_with_uniform(base: np.ndarray, lam: float) -> np.ndarray:
    """Convex combination λ·uniform + (1‑λ)·base."""
    n = base.shape[0]
    uniform = np.full(n, 1.0 / n, dtype=np.float64)
    return lam * uniform + (1.0 - lam) * base

def _smoothed_gradient_descent(
    a0: np.ndarray,
    time_limit: float,
    best_seq: np.ndarray,
    best_score: float,
    init_lr: float = 0.8,
    tau0: float = 0.002,
    tau_mult: float = 0.92,
    tau_iter: int = 400,
    eval_every: int = 100,
) -> tuple[np.ndarray, float]:
    """
    Exponentiated‑gradient descent on the smoothed objective

        f_τ(a) = τ·log Σ_k exp( (a∗a)_k / τ ),

    whose gradient tends to the exact sub‑gradient as τ → 0.
    The vector is kept symmetric and projected onto the simplex after each update.
    """
    a = a0.copy()
    a = _make_symmetric(a)
    n = a.shape[0]
    t = 0
    tau = tau0

    while time.time() < time_limit:
        t += 1

        # ----- convolution and exact objective -----
        conv = _fft_self_convolve(a)                # length 2n‑1
        max_val = conv.max()
        cur_score = 2.0 * n * max_val
        if cur_score < best_score - 1e-14:
            best_score = cur_score
            best_seq = a.copy()
            print(f"[{time.time():.1f}] smoothed it={t} new global best {best_score:.6f}")

        # ----- smoothed gradient (log‑sum‑exp) -----
        # shift to avoid overflow
        shifted = conv - max_val
        expo = np.exp(shifted / tau)
        p = expo / expo.sum()                       # probability distribution on convolution indices

        # gradient = 2 * (p * a_rev)_{reversed first n entries}
        conv_grad = _fft_convolve(p, a[::-1])       # length 3n‑2
        grad = 2.0 * conv_grad[(n - 1)::-1][:n]

        # ----- EG update, symmetry, simplex projection -----
        eta = init_lr / math.sqrt(t)
        a *= np.exp(-eta * grad)                    # positivity is kept
        a = _make_symmetric(a)                      # mirror symmetry
        s = a.sum()
        if s <= 0.0:
            a = np.full(n, 1.0 / n, dtype=np.float64)
        else:
            a /= s

        # ----- anneal temperature -----
        if t % tau_iter == 0:
            tau = max(tau * tau_mult, 1e-12)

        # optional less‑frequent reporting
        if eval_every and t % eval_every == 0:
            pass

    return best_seq, best_score


# ----------------------------------------------------------------------
# Exact sub‑gradient descent (fallback / final polishing)
# ----------------------------------------------------------------------


def _subgradient_descent_exact(
    a0: np.ndarray,
    time_limit: float,
    best_seq: np.ndarray,
    best_score: float,
    lr0: float = 0.9,
    eval_every: int = 100,
) -> tuple[np.ndarray, float]:
    """
    Exponentiated‑gradient descent on the true L∞ objective.
    The sub‑gradient is built from *all* maximal convolution entries.
    """
    a = a0.copy()
    a = _make_symmetric(a)
    n = a.shape[0]
    t = 0
    while time.time() < time_limit:
        t += 1
        conv = _fft_self_convolve(a)
        max_val = conv.max()
        max_idxs = np.where(conv == max_val)[0]

        # exact sub‑gradient
        grad = np.zeros_like(a)
        for idx in max_idxs:
            i_start = max(0, idx - (n - 1))
            i_end = min(n - 1, idx)
            i = np.arange(i_start, i_end + 1, dtype=np.int64)
            j = idx - i
            grad[i] += 2.0 * a[j]

        eta = lr0 / math.sqrt(t)
        a *= np.exp(-eta * grad)          # EG step
        a = _make_symmetric(a)
        s = a.sum()
        if s <= 0.0:
            a = np.full(n, 1.0 / n, dtype=np.float64)
        else:
            a /= s

        if t % eval_every == 0:
            cur_score = 2.0 * n * max_val
            if cur_score < best_score - 1e-14:
                best_score = cur_score
                best_seq = a.copy()
                print(f"[{time.time():.1f}] exact subgrad it={t} new global best {best_score:.6f}")
    return best_seq, best_score


# ----------------------------------------------------------------------
# Simple greedy hill‑climber
# ----------------------------------------------------------------------


def _pairwise_swap(
    a0: np.ndarray,
    time_limit: float,
    best_seq: np.ndarray,
    best_score: float,
    max_trials: int | None = None,
) -> tuple[np.ndarray, float]:
    """
    Randomly move a fraction of mass from one index to another.
    Accept only strictly improving moves.
    """
    a = a0.copy()
    n = a.shape[0]
    trials = 0
    while time.time() < time_limit:
        if max_trials is not None and trials >= max_trials:
            break
        trials += 1

        src_candidates = np.where(a > 1e-14)[0]
        if src_candidates.size == 0:
            break
        i = np.random.choice(src_candidates)
        j = np.random.randint(n - 1)
        if j >= i:
            j += 1

        delta = a[i] * np.random.rand() * 0.45   # up to 45 % of source mass
        if delta < 1e-15:
            continue

        new_a = a.copy()
        new_a[i] -= delta
        new_a[j] += delta
        new_a = _make_symmetric(new_a)
        new_a /= new_a.sum()

        cur = _objective(new_a)
        if cur < best_score - 1e-14:
            a = new_a
            best_score = cur
            best_seq = a.copy()
            print(
                f"[{time.time():.1f}] swap i={i}→j={j} δ={delta:.6g} new best {best_score:.6f}"
            )
    return best_seq, best_score


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: int = 1000) -> list[float]:
    """
    Search for a non‑negative step‑function that (approximately) minimises
    ``evaluate_sequence``.  The routine respects the time budget and returns
    the best sequence as a plain Python list.
    """
    rng = np.random.default_rng(seed)
    start = time.time()
    deadline = start + budget_s - 2.0          # safety margin

    overall_best_seq: np.ndarray | None = None
    overall_best_score = float("inf")
    top_candidates: list[tuple[float, np.ndarray]] = []   # (score, seq)

    # ------------------------------------------------------------------
    # 1. Analytic / random family scan
    # ------------------------------------------------------------------
    lengths = [1000, 2000, 3000, 4000, 6000, 8000, 12000]

    # grids for edge‑weighted families
    p_grid = np.concatenate(
        [
            np.logspace(-3, -0.5, 15),
            np.logspace(-0.5, 0.0, 10),
            np.linspace(0.1, 3.0, 12),
        ]
    )
    lam_grid = np.linspace(0.0, 1.0, 11)       # mixing weight with uniform
    c_grid = np.logspace(-3, 0, 12)           # exponential‑edge decay
    alpha_grid = np.logspace(-1, 1, 8)        # Beta‑distribution alphas
    beta_grid = np.logspace(-1, 1, 8)         # Beta‑distribution betas

    for n in lengths:
        if time.time() >= deadline:
            break
        print(f"[{time.time():.1f}] scanning length {n}")

        # deterministic families -------------------------------------------------
        for base_fn in (
            _distribution_uniform,
            _distribution_arcsine,
            _distribution_sine,
            _distribution_cosine,
            _distribution_quadratic,
        ):
            cand = base_fn(n)
            cand = _make_symmetric(cand)
            sc = _objective(cand)
            top_candidates.append((sc, cand.copy()))
            if sc < overall_best_score:
                overall_best_score, overall_best_seq = sc, cand.copy()
                print(f"[{time.time():.1f}] new global best len={n} score={sc:.6f}")

        # edge‑weighted + uniform mixes -----------------------------------------
        for p in p_grid:
            if time.time() >= deadline:
                break
            edge = _distribution_edge_weighted(n, float(p))
            for lam in lam_grid:
                if time.time() >= deadline:
                    break
                cand = _mix_with_uniform(edge, lam)
                cand = _make_symmetric(cand)
                sc = _objective(cand)
                top_candidates.append((sc, cand.copy()))
                if sc < overall_best_score:
                    overall_best_score, overall_best_seq = sc, cand.copy()
                    print(
                        f"[{time.time():.1f}] len={n} edge p={p:.4g} lam={lam:.2f} score={sc:.6f}"
                    )

        # exponential‑edge family ------------------------------------------------
        for c in c_grid:
            if time.time() >= deadline:
                break
            exp_edge = _distribution_exponential_edge(n, float(c))
            for lam in lam_grid:
                if time.time() >= deadline:
                    break
                cand = _mix_with_uniform(exp_edge, lam)
                cand = _make_symmetric(cand)
                sc = _objective(cand)
                top_candidates.append((sc, cand.copy()))
                if sc < overall_best_score:
                    overall_best_score, overall_best_seq = sc, cand.copy()
                    print(
                        f"[{time.time():.1f}] len={n} exp c={c:.4g} lam={lam:.2f} score={sc:.6f}"
                    )

        # beta‑distribution family ------------------------------------------------
        for a_par in alpha_grid:
            if time.time() >= deadline:
                break
            for b_par in beta_grid:
                if time.time() >= deadline:
                    break
                cand = _distribution_beta(n, float(a_par), float(b_par))
                cand = _make_symmetric(cand)
                sc = _objective(cand)
                top_candidates.append((sc, cand.copy()))
                if sc < overall_best_score:
                    overall_best_score, overall_best_seq = sc, cand.copy()
                    print(
                        f"[{time.time():.1f}] len={n} beta α={a_par:.3g} β={b_par:.3g} score={sc:.6f}"
                    )

        # random Dirichlet seeds -------------------------------------------------
        for _ in range(6):
            if time.time() >= deadline:
                break
            alpha = rng.uniform(0.01, 2.0)
            a = rng.dirichlet(np.full(n, alpha))
            a = _make_symmetric(a)
            sc = _objective(a)
            top_candidates.append((sc, a.copy()))
            if sc < overall_best_score:
                overall_best_score, overall_best_seq = sc, a.copy()
                print(f"[{time.time():.1f}] Dirichlet len={n} score={sc:.6f}")

        # sparse random seeds ----------------------------------------------------
        for _ in range(6):
            if time.time() >= deadline:
                break
            a = _distribution_random_sparse(n, rng, density=0.007)
            a = _make_symmetric(a)
            sc = _objective(a)
            top_candidates.append((sc, a.copy()))
            if sc < overall_best_score:
                overall_best_score, overall_best_seq = sc, a.copy()
                print(f"[{time.time():.1f}] sparse len={n} score={sc:.6f}")

        # keep only a bounded set of best seeds
        top_candidates.sort(key=lambda x: x[0])
        top_candidates = top_candidates[:40]

    # ------------------------------------------------------------------
    # 2.  Previous construction (if any)
    # ------------------------------------------------------------------
    try:
        prev = np.asarray(height_sequence_1, dtype=np.float64)  # type: ignore[name-defined]
        if prev.ndim == 1 and prev.size > 0:
            prev = np.maximum(prev, 0.0)
            if prev.sum() == 0.0:
                prev = _distribution_uniform(prev.size)
            else:
                prev = prev / prev.sum()
            prev = _make_symmetric(prev)
            sc = _objective(prev)
            top_candidates.append((sc, prev.copy()))
            if sc < overall_best_score:
                overall_best_score, overall_best_seq = sc, prev.copy()
                print(f"[{time.time():.1f}] previous construction score={sc:.6f}")
    except Exception:
        pass

    # ------------------------------------------------------------------
    # 3.  Fallback (should never happen)
    # ------------------------------------------------------------------
    if overall_best_seq is None:
        n = 2000
        overall_best_seq = _distribution_uniform(n)
        overall_best_score = _objective(overall_best_seq)
        print(
            f"[{time.time():.1f}] fallback uniform length={n} score={overall_best_score:.6f}"
        )

    # ------------------------------------------------------------------
    # 4.  Smoothed mirror‑descent refinement on a few best seeds
    # ------------------------------------------------------------------
    remaining = deadline - time.time()
    if remaining > 30.0 and top_candidates:
        # pick the six currently best seeds
        seeds = [seq for _, seq in sorted(top_candidates, key=lambda x: x[0])[:6]]
        per_seed = remaining / max(1, len(seeds))
        print(f"[{time.time():.1f}] start smoothed descent on {len(seeds)} seeds")
        for idx, seed_seq in enumerate(seeds):
            if time.time() >= deadline:
                break
            sg_deadline = time.time() + per_seed
            a_opt, sc_opt = _smoothed_gradient_descent(
                seed_seq,
                sg_deadline,
                best_seq=overall_best_seq,
                best_score=overall_best_score,
                init_lr=0.8,
                tau0=0.002,
                tau_mult=0.92,
                tau_iter=400,
                eval_every=100,
            )
            if sc_opt < overall_best_score - 1e-12:
                overall_best_score, overall_best_seq = sc_opt, a_opt.copy()
                print(
                    f"[{time.time():.1f}] after smoothed seed {idx} new global best {overall_best_score:.6f}"
                )

    # ------------------------------------------------------------------
    # 5.  Optional exact sub‑gradient polishing (if we still have time)
    # ------------------------------------------------------------------
    if time.time() < deadline - 10.0:
        print(f"[{time.time():.1f}] exact sub‑gradient polishing")
        a_opt, sc_opt = _subgradient_descent_exact(
            overall_best_seq.copy(),
            deadline,
            best_seq=overall_best_seq,
            best_score=overall_best_score,
            lr0=0.9,
            eval_every=200,
        )
        if sc_opt < overall_best_score - 1e-12:
            overall_best_score, overall_best_seq = sc_opt, a_opt.copy()
            print(
                f"[{time.time():.1f}] after exact polishing new global best {overall_best_score:.6f}"
            )

    # ------------------------------------------------------------------
    # 6.  Final greedy pairwise polishing
    # ------------------------------------------------------------------
    if time.time() < deadline:
        print(f"[{time.time():.1f}] final greedy swap polishing")
        a_opt, sc_opt = _pairwise_swap(
            overall_best_seq.copy(),
            deadline,
            best_seq=overall_best_seq,
            best_score=overall_best_score,
        )
        if sc_opt < overall_best_score - 1e-12:
            overall_best_score, overall_best_seq = sc_opt, a_opt.copy()
            print(f"[{time.time():.1f}] after final swaps new best {overall_best_score:.6f}")

    # ------------------------------------------------------------------
    print(
        f"Final best objective (2·n·max(conv)): {overall_best_score:.6f} (length={len(overall_best_seq)})"
    )
    # Return a plain Python list – evaluate_sequence will re‑normalise internally.
    return overall_best_seq.tolist()
