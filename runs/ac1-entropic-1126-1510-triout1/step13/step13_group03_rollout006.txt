The method consists of three stages:

    arcsine, sine, cosine, quadratic, Dirichlet, sparse) on a few

    allowing the optimiser to first move on a smooth landscape and then
    focus on the true peak.  Gradients are obtained analytically via an
    FFT‑based cross‑correlation, and the simplex constraint is enforced
    by working in log‑probability (softmax) space.

3️⃣  A final cheap pairwise‑mass‑transfer hill‑climber (the same routine
    used in the reference implementation) polishes the solution.

All work is bounded by the supplied time budget (default 1000 s).  The

import time

# ----------------------------------------------------------------------
# Helper: FFT‑based full self‑convolution (real signal)
def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    n = a.shape[0]
    size = 1 << ((2 * n - 1) - 1).bit_length()   # next power of two
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[: 2 * n - 1]


# ----------------------------------------------------------------------
# Exact objective for a vector normalised to sum = 1
def _objective(a: np.ndarray) -> float:
    n = a.shape[0]
    b = _fft_self_convolve(a)
    return float(2.0 * n * b.max())


# ----------------------------------------------------------------------
# Analytic families (same as the previous code, kept for reuse)
def _distribution_uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _distribution_arcsine(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    vals /= vals.sum()
    return vals


def _distribution_edge_weighted(n: int, p: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64) + 0.5
    j = (n - np.arange(n, dtype=np.float64) - 0.5)
    w = (i * j) ** (-p)
    w_sum = w.sum()
    if w_sum == 0.0:
        return np.full(n, 1.0 / n, dtype=np.float64)
    w /= w_sum
    return w


def _distribution_exponential_edge(n: int, c: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    w = np.exp(-c * i) + np.exp(-c * (n - 1 - i))
    w /= w.sum()
    return w


def _distribution_sine(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64) + 0.5
    w = np.sin(np.pi * i / n)
    w /= w.sum()
    return w


def _distribution_cosine(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64) + 0.5
    w = 1.0 - np.cos(np.pi * i / n)
    w /= w.sum()
    return w


def _distribution_quadratic(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    x = (2.0 * i) / (n - 1) - 1.0
    w = 1.0 - x * x
    w = np.clip(w, 0.0, None)
    w /= w.sum()
    return w


def _distribution_random_sparse(
    n: int, rng: np.random.Generator, density: float = 0.01
) -> np.ndarray:
    k = max(2, int(n * density))
    idx = rng.choice(n, size=k, replace=False)
    vals = np.zeros(n, dtype=np.float64)
    vals[idx] = rng.random(k)
    s = vals.sum()
    if s == 0.0:
        vals[0] = 1.0
        s = 1.0
    vals /= s
    return vals


def _mix_with_uniform(base: np.ndarray, lam: float) -> np.ndarray:
    n = base.shape[0]
    uniform = np.full(n, 1.0 / n, dtype=np.float64)
    return lam * uniform + (1.0 - lam) * base

# Smooth‑max (log‑sum‑exp) and its exact gradient w.r.t. a
def _smooth_max_and_grad(a: np.ndarray, lam: float):
    """
    Returns
        M        – the (exact) max of the self‑convolution (for reporting)
        grad_a   – ∂(smooth_max)/∂a  (vector of same length as a)
    The smooth approximation of max is
        S_λ(a) = (1/λ)·log∑_k exp(λ·conv_k)
    whose gradient can be written in closed form and evaluated by a single
    FFT‑based cross‑correlation.
    """
    n = a.shape[0]

    # 1) Full self‑convolution
    conv = _fft_self_convolve(a)                     # length 2n‑1
    max_val = conv.max()

    # 2) Soft‑max weights (stable)
    scaled = lam * (conv - max_val)                  # shift to avoid overflow
    wp = np.exp(scaled)
    w = wp / wp.sum()                               # ∑ w_k = 1

    # 3) Gradient: g_i = 2·∑_k w_k·a_{k‑i}
    # This is a cross‑correlation of w with a.
    rev_w = w[::-1]                                 # reversed weights
    size = 1 << ((len(rev_w) + n - 1) - 1).bit_length()
    W = np.fft.rfft(rev_w, n=size)
    A = np.fft.rfft(a, n=size)
    corr = np.fft.irfft(W * A, n=size)              # linear conv rev_w * a
    L = len(w)                                      # = 2n‑1
    start = L - 1
    # Extract the segment that corresponds to indices 0 … n‑1 and reverse back
    grad = 2.0 * corr[start : start + n][::-1]

    return max_val, grad


# ----------------------------------------------------------------------
# Adam optimiser on the smooth‑max objective (operates on logits)
def _adam_optimize(
    a0: np.ndarray,
    time_limit: float,
    overall_best_seq: np.ndarray,
    overall_best_score: float,
    seed_idx: int = 0,
) -> tuple[np.ndarray, float]:
    """
    Mirror‑descent on the smoothed max using Adam.
    The optimisation runs until ``time_limit`` is reached.
    """
    a = a0.copy()
    n = a.shape[0]

    # logits for a softmax parametrisation (guarantees positivity & sum = 1)
    logits = np.log(a + 1e-16)

    # Adam state
    m = np.zeros_like(logits)
    v = np.zeros_like(logits)
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8
    lr = 0.05                         # base learning rate
    it = 0

    # linear schedule for the smoothing parameter λ
    lam_start, lam_end = 5.0, 40.0
    lam_steps = 2000                 # after this many iters λ is at its final value

    while time.time() < time_limit:
        it += 1

        # current smoothing strength
        lam = lam_start + (lam_end - lam_start) * min(1.0, it / lam_steps)

        # smooth max value (only needed for reporting) and gradient w.r.t. a
        _, grad_a = _smooth_max_and_grad(np.exp(logits) / np.exp(logits).sum(), lam)

        # convert gradient to logits space (∇_θ L = a·(∇_a L – ⟨∇_a L, a⟩))
        avg = np.dot(grad_a, np.exp(logits) / np.exp(logits).sum())
        grad_logits = np.exp(logits) / np.exp(logits).sum() * (grad_a - avg)

        # Adam update
        m = beta1 * m + (1 - beta1) * grad_logits
        v = beta2 * v + (1 - beta2) * (grad_logits ** 2)
        m_hat = m / (1 - beta1 ** it)
        v_hat = v / (1 - beta2 ** it)
        logits -= lr * m_hat / (np.sqrt(v_hat) + eps)

        # re‑normalise to a probability vector (softmax)
        logits -= np.max(logits)                     # numerical stability
        a = np.exp(logits)
        a_sum = a.sum()
        if a_sum == 0.0:
            a = np.full(n, 1.0 / n, dtype=np.float64)
            logits = np.log(a)
        else:
            a /= a_sum

        # occasional exact evaluation (cheap compared to gradient computation)
        if it % 30 == 0:
            cur_score = _objective(a)
            if cur_score < overall_best_score - 1e-14:
                overall_best_score = cur_score
                overall_best_seq = a.copy()
                print(
                    f"[{time.time():.1f}] seed {seed_idx} it={it} new best {overall_best_score:.6f}"
                )
    return overall_best_seq, overall_best_score


# ----------------------------------------------------------------------
# Greedy pairwise mass‑transfer hill‑climber (identical to the reference)
def _pairwise_swap(
    a0: np.ndarray,
    time_limit: float,
    best_seq: np.ndarray,
    best_score: float,
    max_trials: int | None = None,
) -> tuple[np.ndarray, float]:
    a = a0.copy()
    n = a.shape[0]
    trials = 0
    rng = np.random.default_rng()
    while time.time() < time_limit:
        if max_trials is not None and trials >= max_trials:
            break
        trials += 1

        src_candidates = np.where(a > 1e-14)[0]
        if src_candidates.size == 0:
            break
        i = rng.choice(src_candidates)
        j = rng.integers(n - 1)
        if j >= i:
            j += 1

        delta = a[i] * rng.random() * 0.45   # move up to 45 % of source mass
        if delta < 1e-15:
            continue

        new_a = a.copy()
        new_a[i] -= delta
        new_a[j] += delta

        cur = _objective(new_a)
        if cur < best_score - 1e-14:
            a = new_a
            best_score = cur
            best_seq = a.copy()
            print(
                f"[{time.time():.1f}] swap i={i}→j={j} δ={delta:.6g} new best {best_score:.6f}"
            )
    return best_seq, best_score


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: int = 1000) -> list[float]:
    """
    Search for a non‑negative step‑function that minimises ``evaluate_sequence``.
    The routine respects the given time budget and returns the best sequence found.
    """
    rng = np.random.default_rng(seed)
    start = time.time()
    deadline = start + budget_s - 2.0      # tiny safety margin

    overall_best_seq = None
    overall_best_score = float("inf")
    top_candidates: list[tuple[float, np.ndarray]] = []   # (score, sequence)

    # ------------------------------------------------------------------
    # 1.  Scan a wide variety of analytic families on a few lengths
    # ------------------------------------------------------------------
    lengths = [4000, 6000, 8000]           # larger lengths give a better asymptotic bound
    for n in lengths:
        if time.time() >= deadline:
            break
        print(f"[{time.time():.1f}] scanning length {n}")

        # deterministic families -------------------------------------------------
        for base_fn in (
            _distribution_uniform,
            _distribution_arcsine,
            _distribution_sine,
            _distribution_cosine,
            _distribution_quadratic,
        ):
            cand = base_fn(n)
            sc = _objective(cand)
            top_candidates.append((sc, cand.copy()))
            if sc < overall_best_score:
                overall_best_score, overall_best_seq = sc, cand.copy()
                print(f"[{time.time():.1f}] new global best (det) len={n} score={sc:.6f}")

        # edge‑weighted families with random parameters -------------------------
        for _ in range(30):
            if time.time() >= deadline:
                break
            p = rng.uniform(0.01, 5.0)
            edge = _distribution_edge_weighted(n, p)
            lam = rng.random()
            cand = _mix_with_uniform(edge, lam)
            sc = _objective(cand)
            top_candidates.append((sc, cand.copy()))
            if sc < overall_best_score:
                overall_best_score, overall_best_seq = sc, cand.copy()
                print(f"[{time.time():.1f}] len={n} edge p={p:.4g} lam={lam:.2f} score={sc:.6f}")

        # exponential‑edge families ------------------------------------------------
        for _ in range(20):
            if time.time() >= deadline:
                break
            c = rng.uniform(1e-3, 0.5)
            exp_edge = _distribution_exponential_edge(n, c)
            lam = rng.random()
            cand = _mix_with_uniform(exp_edge, lam)
            sc = _objective(cand)
            top_candidates.append((sc, cand.copy()))
            if sc < overall_best_score:
                overall_best_score, overall_best_seq = sc, cand.copy()
                print(f"[{time.time():.1f}] len={n} exp c={c:.4g} lam={lam:.2f} score={sc:.6f}")

        # random Dirichlet seeds -------------------------------------------------
        for _ in range(10):
            if time.time() >= deadline:
                break
            alpha = rng.uniform(0.05, 2.0)
            a = rng.dirichlet(np.full(n, alpha))
            sc = _objective(a)
            top_candidates.append((sc, a.copy()))
            if sc < overall_best_score:
                overall_best_score, overall_best_seq = sc, a.copy()
                print(f"[{time.time():.1f}] Dirichlet len={n} α={alpha:.2g} score={sc:.6f}")

        # sparse random seeds ----------------------------------------------------
        for _ in range(10):
            if time.time() >= deadline:
                break
            a = _distribution_random_sparse(n, rng, density=0.008)
            sc = _objective(a)
            top_candidates.append((sc, a.copy()))
            if sc < overall_best_score:
                overall_best_score, overall_best_seq = sc, a.copy()
                print(f"[{time.time():.1f}] sparse len={n} score={sc:.6f}")

        # keep only a modest number of promising seeds
        top_candidates.sort(key=lambda x: x[0])
        top_candidates = top_candidates[:40]

    # ------------------------------------------------------------------
    # 2.  Incorporate the previous construction if it exists
    # ------------------------------------------------------------------
    try:
        prev = np.asarray(height_sequence_1, dtype=np.float64)  # type: ignore[name-defined]
        if prev.ndim == 1 and prev.size > 0:
            prev = np.maximum(prev, 0.0)
            if prev.sum() == 0.0:
                prev = _distribution_uniform(prev.size)
            else:
                prev = prev / prev.sum()
            sc = _objective(prev)
            if sc < overall_best_score:
                overall_best_score, overall_best_seq = sc, prev.copy()
                print(f"[{time.time():.1f}] previous construction score={sc:.6f}")
            top_candidates.append((sc, prev.copy()))
    except Exception:
        pass

    # ------------------------------------------------------------------
    # 3.  Fallback (should never happen)
    # ------------------------------------------------------------------
    if overall_best_seq is None:
        n = 4000
        overall_best_seq = _distribution_uniform(n)
        overall_best_score = _objective(overall_best_seq)
        print(f"[{time.time():.1f}] fallback uniform length={n} score={overall_best_score:.6f}")

    # ------------------------------------------------------------------
    # 4.  Adam‑based smooth‑max refinement on the best seeds
    # ------------------------------------------------------------------
    remaining = deadline - time.time()
    if remaining > 30.0 and top_candidates:
        # take the six most promising seeds
        seeds = [seq for _, seq in sorted(top_candidates, key=lambda x: x[0])[:6]]
        per_seed = remaining / max(1, len(seeds))
        print(f"[{time.time():.1f}] starting Adam refinement on {len(seeds)} seeds")
        for idx, seed_seq in enumerate(seeds):
            if time.time() >= deadline:
                break
            seed_deadline = time.time() + per_seed
            overall_best_seq, overall_best_score = _adam_optimize(
                seed_seq,
                seed_deadline,
                overall_best_seq,
                overall_best_score,
                seed_idx=idx,
            )
    # ------------------------------------------------------------------
    # 5.  Final greedy pairwise polishing
    # ------------------------------------------------------------------
    if time.time() < deadline:
        print(f"[{time.time():.1f}] final greedy swap polishing")
        overall_best_seq, overall_best_score = _pairwise_swap(
            overall_best_seq.copy(),
            deadline,
            best_seq=overall_best_seq,
            best_score=overall_best_score,
        )

    # ------------------------------------------------------------------
    print(
        f"Final best objective (2·n·max(conv)) = {overall_best_score:.6f} "
        f"(length={len(overall_best_seq)})"
    )
    # Return a plain Python list; evaluate_sequence will handle final normalisation.
    return overall_best_seq.tolist()
