   entries.  A decreasing step size and occasional exponential jitter
   are used; the iterate is projected onto the probability simplex.

3. **Greedy pairwise mass transfer** – While time permits move a small
   amount of mass between two positions if the objective improves.

4. **Cheap simulated annealing walk** – A final SA with a slowly
   decaying temperature explores the neighbourhood of the current best.

The whole routine respects the 1000 s budget and returns a plain Python
list; ``evaluate_sequence`` will apply the final clipping and normalisation.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x≥0, Σx = s}."""
    if s < 0:
        raise ValueError("simplex radius must be non‑negative")
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho_arr = np.where(u - cssv / (np.arange(1, len(v) + 1)) > 0)[0]
    if rho_arr.size == 0:
        theta = 0.0
    else:
        rho = rho_arr[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Full linear self‑convolution a*a using a real FFT (output length 2n‑1)."""
    n = a.shape[0]
    need = 2 * n - 1
    size = 1 << ((need - 1).bit_length())
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[:need]


# ----------------------------------------------------------------------
def _objective(a: np.ndarray) -> float:
    """Exact objective 2·n·max(conv(a,a)) for a vector with Σa = 1."""
    n = a.shape[0]
    b = _fft_self_convolve(a)
    return 2.0 * n * b.max()


# ----------------------------------------------------------------------
# Analytic families ----------------------------------------------------
def _distribution_uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _distribution_arcsine(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    vals /= vals.sum()
    return vals


def _distribution_beta_sym(n: int, p: float) -> np.ndarray:
    """Symmetric Beta(p,p) on the discrete grid."""
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    # use logs to avoid under/overflow
    logw = (p - 1) * (np.log(x) + np.log(1.0 - x))
    w = np.exp(logw)
    if w.sum() == 0.0 or np.isnan(w).any():
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _distribution_edge_weighted(n: int, p: float) -> np.ndarray:
    """U‑shaped family w_i ∝ ((i+½)*(n‑i‑½))^(−p)."""
    i = np.arange(n, dtype=np.float64) + 0.5
    j = (n - np.arange(n, dtype=np.float64) - 0.5)
    w = (i * j) ** (-p)
    if w.sum() == 0.0 or np.isnan(w).any():
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _distribution_beta_asym(n: int, a: float, b: float) -> np.ndarray:
    """Asymmetric Beta(a,b)."""
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    logw = (a - 1) * np.log(x) + (b - 1) * np.log(1.0 - x)
    w = np.exp(logw)
    if w.sum() == 0.0 or np.isnan(w).any():
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _distribution_random_sparse(
    n: int, rng: np.random.Generator, density: float = 0.01
) -> np.ndarray:
    """Sparse random vector (few non‑zeros)."""
    k = max(2, int(n * density))
    idx = rng.choice(n, size=k, replace=False)
    vals = np.zeros(n, dtype=np.float64)
    vals[idx] = rng.random(k)
    if vals.sum() == 0.0:
        vals[0] = 1.0
    vals /= vals.sum()
    return vals


def _distribution_random_dirichlet(
    n: int, rng: np.random.Generator, alpha: float
) -> np.ndarray:
    """Dirichlet sample with all concentration parameters equal to alpha."""
    cand = rng.dirichlet(np.full(n, alpha))
    # already sums to 1
    return cand


def _mix_with_uniform(base: np.ndarray, lam: float) -> np.ndarray:
    """Convex combination lam·U + (1‑lam)·base, where U is uniform."""
    n = base.shape[0]
    uniform = np.full(n, 1.0 / n, dtype=np.float64)
    return lam * uniform + (1.0 - lam) * base


# ----------------------------------------------------------------------
# Projected sub‑gradient descent on max‑convolution --------------------
def _proj_grad_descent(
    a: np.ndarray,
    deadline: float,
    best_seq: np.ndarray,
    best_score: float,
    eta0: float = 0.8,
    jitter_interval: int = 1500,
) -> tuple[np.ndarray, float]:
    """
    Minimise 2·n·max(conv(a,a)) with exact sub‑gradient.
    ``a`` lives on the probability simplex (Σa = 1, a≥0).
    """
    n = a.shape[0]
    t = 0
    no_improve = 0
    rng = np.random.default_rng()
    while time.time() < deadline and t < 500_000:
        t += 1

        conv = _fft_self_convolve(a)
        max_val = conv.max()
        cur_score = 2.0 * n * max_val

        # keep global best
        if cur_score < best_score - 1e-14:
            best_score = cur_score
            best_seq = a.copy()
            no_improve = 0
            print(f"[{time.time():.1f}] proj‑grad it={t} new best {best_score:.6f}")
        else:
            no_improve += 1

        # sub‑gradient of the maximal entry (sum if several)
        ks = np.where(conv == max_val)[0]
        grad = np.zeros_like(a)
        rev_a = a[::-1]                     # reversed a for indexing

        for k in ks:
            offset = n - 1 - k
            if offset >= 0:
                i_start = 0
                i_end = n - offset
                grad[i_start:i_end] += 2.0 * rev_a[offset : offset + i_end - i_start]
            else:
                off = -offset
                i_start = off
                i_end = n
                grad[i_start:i_end] += 2.0 * rev_a[0 : i_end - i_start]

        # step size (decreasing) and projection
        eta = eta0 / math.sqrt(t)
        a_new = a - eta * grad
        a_new = _project_onto_simplex(a_new, s=1.0)

        # optional jitter if stagnating
        if no_improve >= jitter_interval:
            jitter = rng.exponential(scale=0.004, size=n)
            a_new = a_new + jitter
            a_new = _project_onto_simplex(a_new, s=1.0)
            no_improve = 0

        a = a_new

    return best_seq, best_score


# ----------------------------------------------------------------------
# Pairwise deterministic refinement ------------------------------------
def _pairwise_swap_local(
    a: np.ndarray,
    deadline: float,
    best_seq: np.ndarray,
    best_score: float,
    max_trials: int | None = None,
) -> tuple[np.ndarray, float]:
    """Move a small amount of mass between two positions if it improves the objective."""
    n = a.shape[0]
    trials = 0
    rng = np.random.default_rng()
    while time.time() < deadline:
        if max_trials is not None and trials >= max_trials:
            break
        trials += 1

        src = np.where(a > 1e-12)[0]
        if src.size == 0:
            break
        i = rng.choice(src)
        j = rng.integers(n - 1)
        if j >= i:
            j += 1

        delta = a[i] * rng.random() * 0.34
        if delta < 1e-14:
            continue

        new_a = a.copy()
        new_a[i] -= delta
        new_a[j] += delta

        cur = _objective(new_a)
        if cur < best_score - 1e-14:
            a = new_a
            best_score = cur
            best_seq = a.copy()
            print(
                f"[{time.time():.1f}] swap i={i}→j={j} δ={delta:.3e} new best {best_score:.6f}"
            )
    return best_seq, best_score


# ----------------------------------------------------------------------
# Cheap simulated‑annealing walk --------------------------------------
def _random_local_search(
    a: np.ndarray,
    deadline: float,
    best_seq: np.ndarray,
    best_score: float,
    init_temp: float = 1e-2,
    decay: float = 0.9996,
) -> tuple[np.ndarray, float]:
    """Cheap SA that repeatedly transfers a random amount of mass."""
    n = a.shape[0]
    rng = np.random.default_rng()
    T = init_temp
    while time.time() < deadline:
        i = rng.integers(n)
        j = rng.integers(n - 1)
        if j >= i:
            j += 1
        delta = a[i] * rng.random() * 0.3
        if delta < 1e-14:
            continue
        new_a = a.copy()
        new_a[i] -= delta
        new_a[j] += delta
        cur = _objective(new_a)
        if cur < best_score - 1e-14:
            a = new_a
            best_score = cur
            best_seq = a.copy()
            print(f"[{time.time():.1f}] SA improve {i}->{j} Δ={delta:.3e} best {best_score:.6f}")
        else:
            if rng.random() < math.exp(-(cur - best_score) / max(T, 1e-12)):
                a = new_a
        T *= decay
    return best_seq, best_score


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: int = 1000) -> list[float]:
    """
    Search for a non‑negative step‑function that minimises ``evaluate_sequence``.
    Returns a plain Python list (the caller will later invoke ``evaluate_sequence``).
    """
    rng = np.random.default_rng(seed)
    start = time.time()
    deadline = start + budget_s - 4.0   # safety margin

    overall_best_seq: np.ndarray | None = None
    overall_best_score = float("inf")
    all_candidates: list[tuple[float, np.ndarray]] = []

    # --------------------------------------------------------------
    # 1. Parameter sweep over analytic families
    # --------------------------------------------------------------
    lengths = [2000, 3000, 4000, 5000, 6000, 8000, 10000]
    lam_grid = np.linspace(0.0, 1.0, 11)          # 0,0.1,…,1
    # p‑grids (denser around the region that usually works best)
    p_edge = np.concatenate(
        [
            np.linspace(0.02, 0.3, 12),
            np.linspace(0.3, 1.0, 20),
            np.linspace(1.0, 3.0, 15),
            np.linspace(3.0, 6.0, 6),
        ]
    )
    p_beta = np.concatenate(
        [
            np.linspace(0.05, 0.25, 12),
            np.linspace(0.25, 1.0, 20),
            np.linspace(1.0, 3.0, 15),
            np.linspace(3.0, 6.0, 6),
        ]
    )

    for n in lengths:
        if time.time() >= deadline:
            break
        print(f"[{time.time():.1f}] scanning length {n}")

        uniform = _distribution_uniform(n)
        # uniform itself
        sc = _objective(uniform)
        all_candidates.append((sc, uniform.copy()))
        if sc < overall_best_score:
            overall_best_score, overall_best_seq = sc, uniform.copy()
            print(f"[{time.time():.1f}] NEW GLOBAL BEST (uniform) len={n} score={sc:.6f}")

        # arcsine (including mixing)
        arcs = _distribution_arcsine(n)
        for lam in lam_grid:
            cand = _mix_with_uniform(arcs, lam)
            sc = _objective(cand)
            all_candidates.append((sc, cand.copy()))
            if sc < overall_best_score:
                overall_best_score, overall_best_seq = sc, cand.copy()
                print(
                    f"[{time.time():.1f}] NEW GLOBAL BEST (arcsine) lam={lam:.2f} len={n} score={sc:.6f}"
                )

        # symmetric beta
        for p in p_beta:
            if time.time() >= deadline:
                break
            base = _distribution_beta_sym(n, p)
            for lam in lam_grid:
                cand = _mix_with_uniform(base, lam)
                sc = _objective(cand)
                all_candidates.append((sc, cand.copy()))
                if sc < overall_best_score:
                    overall_best_score, overall_best_seq = sc, cand.copy()
                    print(
                        f"[{time.time():.1f}] NEW GLOBAL BEST (βsym) p={p:.3f} lam={lam:.2f} len={n} score={sc:.6f}"
                    )

        # edge‑weighted
        for p in p_edge:
            if time.time() >= deadline:
                break
            base = _distribution_edge_weighted(n, p)
            for lam in lam_grid:
                cand = _mix_with_uniform(base, lam)
                sc = _objective(cand)
                all_candidates.append((sc, cand.copy()))
                if sc < overall_best_score:
                    overall_best_score, overall_best_seq = sc, cand.copy()
                    print(
                        f"[{time.time():.1f}] NEW GLOBAL BEST (edge) p={p:.3f} lam={lam:.2f} len={n} score={sc:.6f}"
                    )

        # random Dirichlet seeds
        for _ in range(3):
            if time.time() >= deadline:
                break
            alpha = rng.uniform(0.01, 2.0)
            cand = _distribution_random_dirichlet(n, rng, alpha)
            sc = _objective(cand)
            all_candidates.append((sc, cand.copy()))
            if sc < overall_best_score:
                overall_best_score, overall_best_seq = sc, cand.copy()
                print(
                    f"[{time.time():.1f}] NEW GLOBAL BEST (Dirichlet) α={alpha:.3g} len={n} score={sc:.6f}"
                )

        # random sparse seeds
        for _ in range(2):
            if time.time() >= deadline:
                break
            cand = _distribution_random_sparse(n, rng, density=0.008)
            sc = _objective(cand)
            all_candidates.append((sc, cand.copy()))
            if sc < overall_best_score:
                overall_best_score, overall_best_seq = sc, cand.copy()
                print(
                    f"[{time.time():.1f}] NEW GLOBAL BEST (sparse) len={n} score={sc:.6f}"
                )

    # --------------------------------------------------------------
    # 2. Keep a limited set of very promising seeds
    # --------------------------------------------------------------
    all_candidates.sort(key=lambda x: x[0])
    top_seeds = [seq for _, seq in all_candidates[:12]]
    print(f"[{time.time():.1f}] retained {len(top_seeds)} top seeds (best score {overall_best_score:.6f})")

    # --------------------------------------------------------------
    # 3. Optional: include the previously discovered construction
    # --------------------------------------------------------------
    try:
        prev = np.asarray(height_sequence_1, dtype=np.float64)  # type: ignore[name-defined]
        if prev.ndim == 1 and prev.size > 0:
            prev = prev / prev.sum()
            sc = _objective(prev)
            if sc < overall_best_score:
                overall_best_score, overall_best_seq = sc, prev.copy()
                print(f"[{time.time():.1f}] imported previous best score={sc:.6f}")
            all_candidates.append((sc, prev.copy()))
    except Exception:
        pass

    # --------------------------------------------------------------
    # 4. Long sub‑gradient descent on the most promising seeds
    # --------------------------------------------------------------
    remaining = deadline - time.time()
    if remaining > 30.0 and top_seeds:
        # give more time to the very best seeds, less to the tail
        per_seed = remaining / len(top_seeds)
        print(f"[{time.time():.1f}] starting projected gradient descent (≈{per_seed:.1f}s per seed)")
        for idx, seed_seq in enumerate(top_seeds):
            if time.time() >= deadline:
                break
            sub_deadline = time.time() + per_seed
            a_opt, sc_opt = _proj_grad_descent(
                seed_seq.copy(),
                sub_deadline,
                best_seq=overall_best_seq,
                best_score=overall_best_score,
                eta0=0.8,
                jitter_interval=1800,
            )
            if sc_opt < overall_best_score:
                overall_best_score, overall_best_seq = sc_opt, a_opt.copy()
                print(
                    f"[{time.time():.1f}] after proj‑grad seed {idx} new global best {overall_best_score:.6f}"
                )
            # early exit if we already have a very good bound
            if overall_best_score <= 1.503:
                break

    # --------------------------------------------------------------
    # 5. Greedy pairwise refinement
    # --------------------------------------------------------------
    if time.time() < deadline and overall_best_seq is not None:
        print(f"[{time.time():.1f}] greedy pairwise refinement")
        overall_best_seq, overall_best_score = _pairwise_swap_local(
            overall_best_seq.copy(),
            deadline,
            best_seq=overall_best_seq,
            best_score=overall_best_score,
            max_trials=300_000,
        )

    # --------------------------------------------------------------
    # 6. Final cheap simulated‑annealing walk
    # --------------------------------------------------------------
    if time.time() < deadline and overall_best_seq is not None:
        print(f"[{time.time():.1f}] final simulated‑annealing walk")
        overall_best_seq, overall_best_score = _random_local_search(
            overall_best_seq.copy(),
            deadline,
            best_seq=overall_best_seq,
            best_score=overall_best_score,
            init_temp=0.009,
            decay=0.9997,
        )

    # --------------------------------------------------------------
    # 7. Return the best sequence found
    # --------------------------------------------------------------
    print(
        f"FINAL best score: {overall_best_score:.6f} (length = {len(overall_best_seq)})"
    )
    # Return a plain Python list – the external ``evaluate_sequence`` will take care of clipping etc.
    return overall_best_seq.tolist()
