"""
Improved search for a step‑function that minimizes ``evaluate_sequence``.

The core ideas
----------------
1. **Very diverse initial pool** – analytic families (uniform, arcsine,
   sine, cosine, quadratic, triangular, edge‑weighted, power‑sum,
   symmetric/asymmetric Beta) on several lengths, plus many random
   Dirichlet / sparse seeds.  The 30 best candidates (by the exact
   objective `2·n·max(conv)`, assuming Σa = 1) are retained.

2. **Smooth exponential‑gradient descent** – a continuation method.
   The nonsmooth “max’’ in the objective is replaced by a
   temperature‑controlled soft‑max:
        w_k ∝ exp( (a*a)_k / τ )
   The gradient of the smoothed objective is obtained analytically via
   a single FFT per iteration.  τ is decreased slowly (τ ∝ 1/t) so the
   algorithm first moves in a well‑behaved landscape and later refines
   the true max‑objective.

3. **Exact mirror‑descent** – a final pass of the original
   sub‑gradient exponential descent (the method used previously) to
   polish the solution.

4. **Greedy pairwise polishing** – random mass transfers that strictly
   improve the exact objective.

All steps respect the supplied time budget (default 1000 s) and the
result is a plain list of non‑negative numbers (already normalised to
sum = 1, which makes ``evaluate_sequence`` scale‑invariant).

The code is self‑contained and uses only NumPy and the standard library.
"""

# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Euclidean projection of ``v`` onto the simplex {x≥0, Σx=s}."""
    if s < 0:
        raise ValueError("Simplex radius must be non‑negative")
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.where(u - cssv / (np.arange(1, len(v) + 1)) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    return np.maximum(v - theta, 0.0)


# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Return the smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


# ----------------------------------------------------------------------
def _fft_convolve(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """
    Full convolution (real FFT) of two 1‑D real arrays.
    """
    m = x.shape[0] + y.shape[0] - 1
    size = _next_pow2(m)
    X = np.fft.rfft(x, n=size)
    Y = np.fft.rfft(y, n=size)
    conv = np.fft.irfft(X * Y, n=size)
    return conv[:m]


# ----------------------------------------------------------------------
def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Full self‑convolution ``a*a`` using a real FFT."""
    return _fft_convolve(a, a)


# ----------------------------------------------------------------------
def _objective(a: np.ndarray) -> float:
    """
    Exact objective ``2·n·max(conv(a,a))`` for a vector that
    is already normalised to Σa = 1.
    """
    n = a.shape[0]
    b = _fft_self_convolve(a)
    return float(2.0 * n * b.max())

# Analytic families ----------------------------------------------------
def _distribution_uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _distribution_arcsine(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    vals /= vals.sum()
    return vals


def _distribution_sine(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64) + 0.5
    w = np.sin(np.pi * i / n)
    w /= w.sum()
    return w


def _distribution_cosine(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64) + 0.5
    w = 1.0 - np.cos(np.pi * i / n)
    w /= w.sum()
    return w


def _distribution_quadratic(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    x = (2.0 * i) / (n - 1) - 1.0
    w = 1.0 - x * x
    w = np.clip(w, 0.0, None)
    w /= w.sum()
    return w


def _distribution_triangle(n: int) -> np.ndarray:
    """Triangular shape – linear rise to the centre, then fall."""
    i = np.arange(n, dtype=np.float64) + 1.0
    rise = i
    fall = n + 1 - i
    w = np.minimum(rise, fall)
    w /= w.sum()
    return w


def _distribution_edge_weighted(n: int, p: float) -> np.ndarray:
    """Edge‑heavy: w_i ∝ ((i+½)*(n‑i‑½))^{‑p}."""
    i = np.arange(n, dtype=np.float64) + 0.5
    j = (n - np.arange(n, dtype=np.float64) - 0.5)
    w = (i * j) ** (-p)
    w = np.clip(w, 0.0, None)
    total = w.sum()
    if total > 0:
        w /= total
    else:
        w = np.full(n, 1.0 / n)
    return w


def _distribution_power_sum(n: int, p: float) -> np.ndarray:
    """w_i ∝ (i+1)^{‑p} + (n‑i)^{‑p}."""
    i = np.arange(1, n + 1, dtype=np.float64)
    w = i ** (-p) + (n + 1 - i) ** (-p)
    w = np.clip(w, 0.0, None)
    w /= w.sum()
    return w


def _distribution_beta(n: int, a: float, b: float) -> np.ndarray:
    """Beta(a,b) density on a regular grid."""
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    # use log‑space to avoid underflow for tiny a,b
    log_w = (a - 1) * np.log(x) + (b - 1) * np.log(1.0 - x)
    w = np.exp(log_w - np.max(log_w))   # shift for stability
    total = w.sum()
    if total == 0.0:
        w = np.full(n, 1.0 / n)
    else:
        w /= total
    return w


def _distribution_random_sparse(
    n: int, rng: np.random.Generator, density: float = 0.01
) -> np.ndarray:
    """Sparse random vector with a prescribed density of non‑zeros."""
    k = max(2, int(n * density))
    idx = rng.choice(n, size=k, replace=False)
    vals = np.zeros(n, dtype=np.float64)
    vals[idx] = rng.random(k)
    if vals.sum() == 0.0:
        vals[0] = 1.0
    vals /= vals.sum()
    return vals


def _mix_two(a: np.ndarray, b: np.ndarray, lam: float) -> np.ndarray:
    """Convex combination λ·a + (1‑λ)·b."""
    return lam * a + (1.0 - lam) * b

# Exact sub‑gradient of max‑convolution --------------------------------
def _max_convolution_subgradient(a: np.ndarray) -> np.ndarray:
    """Sub‑gradient of  max_k Σ_i a_i a_{k-i}."""
    conv = _fft_self_convolve(a)
    max_val = conv.max()
    tol = 1e-12 * max_val
    active = np.where(conv >= max_val - tol)[0]
    n = a.shape[0]
    grad = np.zeros_like(a)
    for k in active:
        lo = max(0, k - (n - 1))
        hi = min(n - 1, k)
        idx = np.arange(lo, hi + 1)
        grad[idx] += 2.0 * a[k - idx]
    if len(active):
        grad /= len(active)
    return grad

# Smoothed gradient (soft‑max) -----------------------------------------
def _smooth_gradient_from_conv(a: np.ndarray, conv: np.ndarray, tau: float) -> np.ndarray:
    """
    Gradient of the soft‑max approximation of max(conv).
    Parameters
    ----------
    a   : probability vector (Σa = 1)
    conv: full self‑convolution of ``a`` (length 2·n‑1)
    tau : temperature (>0). Small τ → true max.
    Returns
    -------
    grad: array of shape ``a``.
    """
    n = a.shape[0]
    max_c = conv.max()
    # soft‑max weights (stable: subtract max_c)
    diff = (conv - max_c) / tau
    w = np.exp(diff)
    w /= w.sum()

    # cross‑correlation w ⋆ a (i.e. Σ_k w_k a_{k-i})
    a_rev = a[::-1]
    conv_wa = _fft_convolve(w, a_rev)                # length 3·n‑2
    grad = 2.0 * conv_wa[n - 1 : n - 1 + n]          # slice of length n
    return grad

# Mirror (exponentiated) gradient descent -------------------------------
def _exp_gradient_descent(
    a0: np.ndarray,
    deadline: float,
    best_seq: np.ndarray,
    best_score: float,
    eta0: float = 0.6,
    report_every: int = 200,
) -> tuple[np.ndarray, float]:
    """
    Exponentiated‑gradient descent on the exact max‑convolution objective.
    Uses the true sub‑gradient of the active convolution index.
    """
    a = a0.copy()
    n = a.shape[0]
    t = 0
    while time.time() < deadline:
        t += 1
        grad = _max_convolution_subgradient(a)
        eta = eta0 / math.sqrt(t)
        a = a * np.exp(-eta * grad)
        s = a.sum()
        if s > 0:
            a = a / s
        else:
            a = np.full(n, 1.0 / n, dtype=np.float64)

        if t % report_every == 0:
            cur = _objective(a)
            if cur < best_score - 1e-14:
                best_score = cur
                best_seq = a.copy()
                print(f"[{time.time():.1f}] exp‑gd it={t} new global best {best_score:.6f}")
    return best_seq, best_score

def _smooth_exp_gradient_descent(
    a0: np.ndarray,
    deadline: float,
    best_seq: np.ndarray,
    best_score: float,
    eta0: float = 0.8,
    tau_start_factor: float = 0.5,
    tau_min_factor: float = 1e-6,
    report_every: int = 200,
) -> tuple[np.ndarray, float]:
    """
    Smoothed exponential gradient descent.
    The temperature τ is decreased like τ = max_c * max(tau_start/(1+t), τ_min).
    The update uses a KL‑mirror step (multiplicative weights) followed by
    normalisation onto the simplex.
    """
    a = a0.copy()
    n = a.shape[0]
    t = 0
    while time.time() < deadline:
        t += 1
        conv = _fft_self_convolve(a)
        max_c = conv.max()
        tau = max_c * max(tau_start_factor / (1.0 + t), tau_min_factor)
        grad = _smooth_gradient_from_conv(a, conv, tau)

        eta = eta0 / math.sqrt(t)
        a = a * np.exp(-eta * grad)
        s = a.sum()
        if s > 0:
            a = a / s
        else:
            a = np.full(n, 1.0 / n, dtype=np.float64)

        if t % report_every == 0:
            cur = _objective(a)
            if cur < best_score - 1e-14:
                best_score = cur
                best_seq = a.copy()
                print(f"[{time.time():.1f}] smooth‑gd it={t} τ={tau:.2e} new best {best_score:.6f}")
    return best_seq, best_score


# ----------------------------------------------------------------------
def _pairwise_swap(
    a0: np.ndarray,
    deadline: float,
    best_seq: np.ndarray,
    best_score: float,
    max_trials: int | None = None,
) -> tuple[np.ndarray, float]:
    """
    Randomly move a fraction of mass from one index to another.
    Only moves that strictly improve the exact objective are kept.
    """
    a = a0.copy()
    n = a.shape[0]
    trials = 0
    while time.time() < deadline:
        if max_trials is not None and trials >= max_trials:
            break
        trials += 1

        src = np.where(a > 1e-14)[0]
        if src.size == 0:
            break
        i = np.random.choice(src)
        j = np.random.randint(n - 1)
        if j >= i:
            j += 1

        delta = a[i] * np.random.rand() * 0.45   # up to 45 % of source mass
        if delta < 1e-15:
            continue

        new_a = a.copy()
        new_a[i] -= delta
        new_a[j] += delta

        cur = _objective(new_a)
        if cur < best_score - 1e-14:
            a = new_a
            best_score = cur
            best_seq = a.copy()
            print(
                f"[{time.time():.1f}] swap i={i}→j={j} δ={delta:.6g} new best {best_score:.6f}"
            )
    return best_seq, best_score

    rng = np.random.default_rng(seed)
    start = time.time()
    deadline = start + budget_s - 2.0          # safety margin

    overall_best_seq: np.ndarray | None = None
    overall_best_score = float("inf")
    top_candidates: list[tuple[float, np.ndarray]] = []  # (score, seq)

    # ------------------------------------------------------------------
    # 1.  Scan a wide variety of analytic families on several lengths.
    # ------------------------------------------------------------------
    lengths = [2048, 4096, 8192, 12000]      # moderate sizes for the scan
    # parameter grids
    p_grid = np.concatenate(
        [
            np.logspace(-3, -0.5, 12),   # 0.001 … 0.3
            np.linspace(0.3, 5.0, 14),
            np.linspace(5.0, 15.0, 6),
        ]
    )
    beta_params = np.array([0.05, 0.1, 0.2, 0.5, 1.0, 2.0, 5.0])
    lam_grid = np.linspace(0.0, 1.0, 5)    # 0, .25, .5, .75, 1

    for n in lengths:
        if time.time() >= deadline:
            break
        print(f"[{time.time():.1f}] scanning length {n}")

        # deterministic families -------------------------------------------------
        for base_fn in (
            _distribution_uniform,
            _distribution_arcsine,
            _distribution_sine,
            _distribution_cosine,
            _distribution_quadratic,
            _distribution_triangle,
        ):
            cand = base_fn(n)
            sc = _objective(cand)
            top_candidates.append((sc, cand.copy()))
            if sc < overall_best_score:
                overall_best_score = sc
                overall_best_seq = cand.copy()
                print(f"[{time.time():.1f}] new global best len={n} score={sc:.6f}")

        # parametric families ----------------------------------------------------
        for p in p_grid:
            if time.time() >= deadline:
                break
            edge = _distribution_edge_weighted(n, float(p))
            beta = _distribution_beta(n, float(p), float(p))
            power = _distribution_power_sum(n, float(p))
            for base in (edge, beta, power):
                for lam in lam_grid:
                    if time.time() >= deadline:
                        break
                    cand = _mix_two(base, _distribution_uniform(n), lam)
                    sc = _objective(cand)
                    top_candidates.append((sc, cand.copy()))
                    if sc < overall_best_score:
                        overall_best_score = sc
                        overall_best_seq = cand.copy()
                        print(
                            f"[{time.time():.1f}] new global best len={n} p={p:.4g} lam={lam:.2f} score={sc:.6f}"
                        )

        # asymmetric Beta --------------------------------------------------------
        for a in beta_params:
            for b in beta_params:
                if a == b:
                    continue
                if time.time() >= deadline:
                    break
                cand = _distribution_beta(n, float(a), float(b))
                sc = _objective(cand)
                top_candidates.append((sc, cand.copy()))
                if sc < overall_best_score:
                    overall_best_score = sc
                    overall_best_seq = cand.copy()
                    print(
                        f"[{time.time():.1f}] new global best (asym) len={n} a={a:.2f} b={b:.2f} score={sc:.6f}"
                    )

        # random Dirichlet seeds -------------------------------------------------
        for _ in range(8):
            if time.time() >= deadline:
                break
            alpha = rng.uniform(0.01, 2.0)
            cand = rng.dirichlet(np.full(n, alpha))
            sc = _objective(cand)
            top_candidates.append((sc, cand.copy()))
            if sc < overall_best_score:
                overall_best_score = sc
                overall_best_seq = cand.copy()
                print(f"[{time.time():.1f}] Dirichlet len={n} α={alpha:.3f} score={sc:.6f}")

        # sparse random seeds ----------------------------------------------------
        for _ in range(8):
            if time.time() >= deadline:
                break
            cand = _distribution_random_sparse(n, rng, density=0.004)
            sc = _objective(cand)
            top_candidates.append((sc, cand.copy()))
            if sc < overall_best_score:
                overall_best_score = sc
                overall_best_seq = cand.copy()
                print(f"[{time.time():.1f}] sparse len={n} score={sc:.6f}")

        # keep only the most promising ~30 candidates overall
        top_candidates.sort(key=lambda x: x[0])
        top_candidates = top_candidates[:30]

    # ------------------------------------------------------------------
    # 2.  Incorporate any previously discovered construction (if present).
    # ------------------------------------------------------------------
    try:
        prev = np.asarray(height_sequence_1, dtype=np.float64)  # type: ignore[name-defined]
        if prev.ndim == 1 and prev.size > 0:
            prev = np.maximum(prev, 0.0)
            total = prev.sum()
            if total == 0.0:
                prev = _distribution_uniform(prev.size)
            else:
                prev = prev / total
            sc = _objective(prev)
            if sc < overall_best_score:
                overall_best_score = sc
                overall_best_seq = prev.copy()
                print(f"[{time.time():.1f}] previous construction score={sc:.6f}")
            top_candidates.append((sc, prev.copy()))
    except Exception:
        pass

    # ------------------------------------------------------------------
    # 3.  Fallback (should never be needed).
    # ------------------------------------------------------------------
    if overall_best_seq is None:
        n = 5000
        overall_best_seq = _distribution_uniform(n)
        overall_best_score = _objective(overall_best_seq)
        print(
            f"[{time.time():.1f}] fallback uniform length={n} score={overall_best_score:.6f}"
        )

    # ------------------------------------------------------------------
    # 4.  Refine the most promising seeds.
    # ------------------------------------------------------------------
    remaining = deadline - time.time()
    if remaining > 30.0 and top_candidates:
        # Take the best few seeds for a deep optimisation pass
        seeds = [seq for _, seq in sorted(top_candidates, key=lambda x: x[0])[:5]]
        per_seed = remaining / max(1, len(seeds))
        print(f"[{time.time():.1f}] refining {len(seeds)} seeds, {per_seed:.1f}s each")
        for idx, seed_seq in enumerate(seeds):
            if time.time() >= deadline:
                break
            sg_deadline = time.time() + per_seed

            # --- smooth exponential descent (continuation) ---
            a_opt, sc_opt = _smooth_exp_gradient_descent(
                seed_seq,
                sg_deadline,
                overall_best_seq,
                overall_best_score,
                eta0=0.9,
                report_every=250,
            )
            if sc_opt < overall_best_score - 1e-12:
                overall_best_score = sc_opt
                overall_best_seq = a_opt.copy()
                print(
                    f"[{time.time():.1f}] after smooth‑gd seed {idx} new global best {overall_best_score:.6f}"
                )

            # --- exact sub‑gradient descent for final polishing ---
            a_opt2, sc_opt2 = _exp_gradient_descent(
                a_opt,
                sg_deadline,
                overall_best_seq,
                overall_best_score,
                eta0=0.7,
                report_every=200,
            )
            if sc_opt2 < overall_best_score - 1e-12:
                overall_best_score = sc_opt2
                overall_best_seq = a_opt2.copy()
                print(
                    f"[{time.time():.1f}] after exact‑gd seed {idx} new global best {overall_best_score:.6f}"
                )

    # ------------------------------------------------------------------
    # 5.  Final greedy polishing.
    # ------------------------------------------------------------------
    if time.time() < deadline:
        print(f"[{time.time():.1f}] final greedy polishing")
        a_opt, sc_opt = _pairwise_swap(
            overall_best_seq.copy(),
            deadline,
            overall_best_seq,
            overall_best_score,
        )
        if sc_opt < overall_best_score - 1e-12:
            overall_best_score = sc_opt
            overall_best_seq = a_opt.copy()
            print(f"[{time.time():.1f}] after final swaps new best {overall_best_score:.6f}")

    print(
        f"Final best score (2·n·max(conv)): {overall_best_score:.6f} (length={len(overall_best_seq)})"
    )
    # Return a plain Python list – evaluate_sequence will re‑normalise internally.
    return overall_best_seq.tolist()
