"""
Non‑negative step‑function optimizer.

The algorithm proceeds in four stages:

1️⃣  **Rich analytic scan** – generate a small collection of candidate
    distributions (uniform, arcsine, beta, Gaussian, edge‑weighted,
    exponential‑edge, sine / cosine, quadratic) on several lengths and
    keep the best one.

2️⃣  **Differential Evolution (DE)** – start a small population (the
    best candidate + random Dirichlet seeds) and evolve it on the
    simplex using classic DE mutation, crossover and selection,
    always re‑projecting onto the probability simplex.

3️⃣  **Exact sub‑gradient descent** – from the best DE solution run a
    mirror‑descent / exponentiated‑gradient optimiser that uses the
    *exact* sub‑gradient of the L∞‐convolution objective
    (all indices attaining the maximal convolution value).

4️⃣  **Greedy pairwise mass transfer** – a final hill‑climber that moves
    a random fraction of mass from one coordinate to another whenever
    the objective improves.

All work is limited by the supplied time budget (default 1000 s).  The
returned list is a plain Python list of non‑negative numbers; the
evaluation routine will normalise it internally.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x ≥ 0, Σx = s}."""
    if s < 0:
        raise ValueError("Simplex radius must be non‑negative")
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.where(u - cssv / (np.arange(1, len(v) + 1)) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    return np.maximum(v - theta, 0.0)


# ----------------------------------------------------------------------
def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Full self‑convolution a*a using a real FFT."""
    n = a.shape[0]
    # next power of two ≥ 2*n‑1
    size = 1 << ((2 * n - 1) - 1).bit_length()
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[: 2 * n - 1]


# ----------------------------------------------------------------------
def _objective(a: np.ndarray) -> float:
    """
    Exact objective 2·n·max(conv(a,a)) for a vector with Σa = 1.
    """
    n = a.shape[0]
    b = _fft_self_convolve(a)
    return float(2.0 * n * b.max())


# ----------------------------------------------------------------------
# Analytic families ----------------------------------------------------
def _distribution_uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _distribution_arcsine(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    vals /= vals.sum()
    return vals


def _distribution_sine(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64) + 0.5
    w = np.sin(np.pi * i / n)
    w /= w.sum()
    return w


def _distribution_cosine(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64) + 0.5
    w = 1.0 - np.cos(np.pi * i / n)
    w /= w.sum()
    return w


def _distribution_quadratic(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    x = (2.0 * i) / (n - 1) - 1.0
    w = 1.0 - x * x
    w = np.clip(w, 0.0, None)
    w /= w.sum()
    return w


def _distribution_edge_weighted(n: int, p: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64) + 0.5
    j = (n - np.arange(n, dtype=np.float64) - 0.5)
    w = (i * j) ** (-p)
    w_sum = w.sum()
    if w_sum == 0.0:
        return np.full(n, 1.0 / n, dtype=np.float64)
    w /= w_sum
    return w


def _distribution_exponential_edge(n: int, c: float) -> np.ndarray:
    """w_i ∝ exp(-c·i) + exp(-c·(n‑1‑i))   (c>0)."""
    i = np.arange(n, dtype=np.float64)
    w = np.exp(-c * i) + np.exp(-c * (n - 1 - i))
    w /= w.sum()
    return w


def _distribution_beta(n: int, alpha: float = 0.5) -> np.ndarray:
    """Symmetric Beta(α,α) density on the discrete grid."""
    i = np.arange(n, dtype=np.float64) + 0.5
    x = i / n
    # avoid zero^negative by clipping
    with np.errstate(divide='ignore', invalid='ignore'):
        w = np.power(x, alpha - 1.0) * np.power(1.0 - x, alpha - 1.0)
    w = np.nan_to_num(w, nan=0.0, posinf=0.0, neginf=0.0)
    if w.sum() == 0.0:
        return np.full(n, 1.0 / n, dtype=np.float64)
    w /= w.sum()
    return w


def _distribution_gaussian(n: int, sigma_factor: float = 0.2) -> np.ndarray:
    """Discrete Gaussian centred at the middle, σ = sigma_factor·n."""
    i = np.arange(n, dtype=np.float64)
    mu = (n - 1) / 2.0
    sigma = max(1e-12, sigma_factor * n)
    w = np.exp(-0.5 * ((i - mu) / sigma) ** 2)
    if w.sum() == 0.0:
        return np.full(n, 1.0 / n, dtype=np.float64)
    w /= w.sum()
    return w


def _distribution_random_sparse(
    n: int, rng: np.random.Generator, density: float = 0.01
) -> np.ndarray:
    """Sparse random vector with a prescribed density of non‑zeros."""
    k = max(2, int(n * density))
    idx = rng.choice(n, size=k, replace=False)
    vals = np.zeros(n, dtype=np.float64)
    vals[idx] = rng.random(k)
    s = vals.sum()
    if s == 0.0:
        vals[0] = 1.0
        s = 1.0
    vals /= s
    return vals

# Exact sub‑gradient / exponentiated‑gradient descent -------------------
def _subgradient_descent_exact(
    a0: np.ndarray,
    time_limit: float,
    best_seq: np.ndarray,
    best_score: float,
    lr0: float = 0.9,
    eval_every: int = 100,
) -> tuple[np.ndarray, float]:
    """
    Exponentiated‑gradient descent on the exact L∞ objective.
    Sub‑gradient is built from *all* indices attaining the maximal
    convolution value.
    """
    a = a0.copy()
    n = a.shape[0]
    t = 0
    while time.time() < time_limit:
        t += 1
        conv = _fft_self_convolve(a)
        max_val = conv.max()
        max_idxs = np.where(conv == max_val)[0]

        # exact sub‑gradient (see derivation in the original code)
        grad = np.zeros_like(a)
        for idx in max_idxs:
            i_start = max(0, idx - (n - 1))
            i_end = min(n - 1, idx)
            i = np.arange(i_start, i_end + 1, dtype=np.int64)
            j = idx - i
            grad[i] += 2.0 * a[j]

        eta = lr0 / math.sqrt(t)
        a *= np.exp(-eta * grad)                # EG update, keeps positivity
        s = a.sum()
        if s <= 0.0:
            a = np.full(n, 1.0 / n, dtype=np.float64)
        else:
            a /= s                                 # simplex normalisation

        if t % eval_every == 0:
            cur = 2.0 * n * max_val
            if cur < best_score - 1e-14:
                best_score = cur
                best_seq = a.copy()
                print(
                    f"[{time.time():.1f}] exact subgrad it={t} new global best {best_score:.6f}"
                )
    return best_seq, best_score

# Greedy pairwise mass transfer ----------------------------------------
def _pairwise_swap(
    a0: np.ndarray,
    time_limit: float,
    best_seq: np.ndarray,
    best_score: float,
    max_trials: int | None = None,
) -> tuple[np.ndarray, float]:
    """
    Randomly move a fraction of mass from one index to another.
    The move is accepted only if it strictly improves the exact objective.
    """
    a = a0.copy()
    n = a.shape[0]
    trials = 0
    while time.time() < time_limit:
        if max_trials is not None and trials >= max_trials:
            break
        trials += 1

        src_candidates = np.where(a > 1e-14)[0]
        if src_candidates.size == 0:
            break
        i = np.random.choice(src_candidates)
        j = np.random.randint(n - 1)
        if j >= i:
            j += 1

        delta = a[i] * np.random.rand() * 0.45   # up to 45 % of source mass
        if delta < 1e-15:
            continue

        new_a = a.copy()
        new_a[i] -= delta
        new_a[j] += delta

        cur = _objective(new_a)
        if cur < best_score - 1e-14:
            a = new_a
            best_score = cur
            best_seq = a.copy()
            print(
                f"[{time.time():.1f}] swap i={i}→j={j} δ={delta:.6g} new best {best_score:.6f}"
            )
    return best_seq, best_score

            sc = _objective(a)
            if sc < overall_best_score:
                overall_best_score, overall_best_seq, best_length = sc, a.copy(), n
                print(
                    f"[{time.time():.1f}] len={n} gauss σ={sigma_factor:.3g} score={sc:.6f}"
                )

        # random Dirichlet seeds ------------------------------------------------
        for _ in range(5):
            if time.time() >= deadline:
                break
            alpha = rng.uniform(0.01, 2.0)
            a = rng.dirichlet(np.full(n, alpha))
            a = np.maximum(a, 0.0)
            a /= a.sum()
            sc = _objective(a)
            if sc < overall_best_score:
                overall_best_score, overall_best_seq, best_length = sc, a.copy(), n
                print(f"[{time.time():.1f}] Dirichlet len={n} score={sc:.6f}")

        # sparse random seeds ----------------------------------------------------
        for _ in range(4):
            if time.time() >= deadline:
                break
            a = _distribution_random_sparse(n, rng, density=0.007)
            a = np.maximum(a, 0.0)
            a /= a.sum()
            sc = _objective(a)
            if sc < overall_best_score:
                overall_best_score, overall_best_seq, best_length = sc, a.copy(), n
                print(f"[{time.time():.1f}] sparse len={n} score={sc:.6f}")

    # ------------------------------------------------------------------
    # 2️⃣  Incorporate previous construction if available
    # ------------------------------------------------------------------
    try:
        prev = np.asarray(height_sequence_1, dtype=np.float64)  # type: ignore[name-defined]
        if prev.ndim == 1 and prev.size > 0:
            prev = np.maximum(prev, 0.0)
            if prev.sum() == 0.0:
                prev = _distribution_uniform(prev.size)
            else:
                prev = prev / prev.sum()
            sc = _objective(prev)
            if sc < overall_best_score:
                overall_best_score, overall_best_seq, best_length = sc, prev.copy(), prev.size
                print(f"[{time.time():.1f}] previous best len={best_length} score={sc:.6f}")
    except Exception:
        pass

    # ------------------------------------------------------------------
    # 3️⃣  Fallback if nothing was found (should never happen)
    # ------------------------------------------------------------------
    if overall_best_seq is None:
        n = 2000
        overall_best_seq = _distribution_uniform(n)
        overall_best_score = _objective(overall_best_seq)
        best_length = n
        print(
            f"[{time.time():.1f}] fallback uniform len={n} score={overall_best_score:.6f}"
        )

    # ------------------------------------------------------------------
    # 4️⃣  Differential Evolution on the best length
    # ------------------------------------------------------------------
    n = best_length
    pop_size = 12
    population: list[np.ndarray] = []
    pop_scores: list[float] = []

    # seed with the current best
    population.append(overall_best_seq.copy())
    pop_scores.append(overall_best_score)

    # fill the rest of the population with random Dirichlet vectors
    for _ in range(pop_size - 1):
        a = rng.dirichlet(np.full(n, 1.0))
        a = np.maximum(a, 0.0)
        a /= a.sum()
        population.append(a)
        pop_scores.append(_objective(a))

    best_seq = overall_best_seq.copy()
    best_score = overall_best_score

    F = 0.8          # mutation scaling factor
    Cr = 0.9         # crossover probability

    while time.time() < deadline:
        for i in range(pop_size):
            # pick three distinct individuals different from i
            idxs = list(range(pop_size))
            idxs.remove(i)
            a_idx, b_idx, c_idx = rng.choice(idxs, size=3, replace=False)

            donor = population[a_idx] + F * (population[b_idx] - population[c_idx])
            donor = np.maximum(donor, 0.0)
            donor /= donor.sum() if donor.sum() > 0 else 1.0

            # binomial crossover
            mask = rng.random(n) < Cr
            trial = np.where(mask, donor, population[i])
            trial = np.maximum(trial, 0.0)
            trial_sum = trial.sum()
            if trial_sum == 0.0:
                trial = _distribution_uniform(n)
                trial_sum = trial.sum()
            trial /= trial_sum

            trial_score = _objective(trial)
            if trial_score < pop_scores[i]:
                population[i] = trial
                pop_scores[i] = trial_score
                if trial_score < best_score:
                    best_seq = trial.copy()
                    best_score = trial_score
                    print(
                        f"[{time.time():.1f}] DE new global best score={best_score:.6f}"
                    )
        # quick exit if only a couple of seconds remain
        if time.time() + 5.0 >= deadline:
            break

    # ------------------------------------------------------------------
    # 5️⃣  Exact sub‑gradient (mirror) descent refinement
    # ------------------------------------------------------------------
    remaining = deadline - time.time()
    if remaining > 25.0:
        sg_deadline = time.time() + remaining * 0.55
        best_seq, best_score = _subgradient_descent_exact(
            best_seq,
            sg_deadline,
            best_seq,
            best_score,
            lr0=0.7,
            eval_every=80,
        )

    # ------------------------------------------------------------------
    # 6️⃣  Final greedy pairwise polishing
    # ------------------------------------------------------------------
    if time.time() < deadline:
        best_seq, best_score = _pairwise_swap(
            best_seq,
            deadline,
            best_seq,
            best_score,
        )

    print(f"Final best objective (2·n·max(conv)): {best_score:.6f} (length={len(best_seq)})")
    # Return a plain Python list – evaluate_sequence will re‑normalise.
    return best_seq.tolist()
