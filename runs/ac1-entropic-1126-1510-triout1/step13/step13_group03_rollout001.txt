```python
def _distribution_kaiser(n, beta):
    i = np.arange(n, dtype=np.float64)
    # normalized to [0, n-1]
    arg = beta * np.sqrt(1.0 - ((2.0*i)/(n-1) - 1.0)**2)
    w = i0(arg) / i0(beta)
    w = np.maximum(w, 0.0)
    w /= w.sum()
    return w
```

"""
Search for a non‑negative step‑function that minimizes ``evaluate_sequence``.

The algorithm is a 4‑phase pipeline:

1️⃣  **Rich analytic scan** – many families of symmetric “window” shapes
    (uniform, arcsine, sine, cosine, quadratic, edge‑weighted, exponential‑edge,
    Kaiser, Blackman, Nuttall, Tukey) are instantiated for a variety of
    lengths and parameters.  Each candidate is normalised (Σa = 1) and its exact
    objective 2·n·max(conv(a,a)) is recorded.  The 30 best candidates are kept.

2️⃣  **Accelerated mirror‑descent** – the best seed is refined using an
    Adam‑style exponentiated‑gradient update that works on the true L∞
    objective (sub‑gradient built from *all* maximal convolution entries).
    The step‑size adapts automatically; the loop runs until the time budget
    expires (leaving a short margin for polishing).

3️⃣  **Greedy pairwise mass transfers** – a fast hill‑climber swaps a random
    fraction of mass between two positions and accepts only strict improvements.
    This quickly removes remaining spikes.

4️⃣  **Return** – the best sequence found (as a plain Python list) is returned.
    The external ``evaluate_sequence`` will re‑normalise it, but we keep the
    vector already normalised to Σa = 1 for efficiency.
"""

import time

import numpy as np
from scipy.special import i0  # Kaiser window

# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Euclidean projection onto the simplex {x ≥ 0, Σx = s}."""
    if s < 0:
        raise ValueError("Simplex radius must be non‑negative")
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.where(u - cssv / (np.arange(1, len(v) + 1)) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    return np.maximum(v - theta, 0.0)


# ----------------------------------------------------------------------
def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Full self‑convolution using a real FFT (length 2·n‑1)."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1) - 1).bit_length()  # next power of two
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[: 2 * n - 1]


# ----------------------------------------------------------------------
def _objective(a: np.ndarray) -> float:
    """Exact objective 2·n·max(conv(a,a)) for a vector with Σa = 1."""
    n = a.shape[0]
    b = _fft_self_convolve(a)
    return float(2.0 * n * b.max())

# Analytic families ----------------------------------------------------
def _distribution_uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _distribution_arcsine(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    vals /= vals.sum()
    return vals


def _distribution_sine(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64) + 0.5
    w = np.sin(np.pi * i / n)
    w /= w.sum()
    return w


def _distribution_cosine(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64) + 0.5
    w = 1.0 - np.cos(np.pi * i / n)
    w /= w.sum()
    return w


def _distribution_quadratic(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    x = (2.0 * i) / (n - 1) - 1.0
    w = 1.0 - x * x
    w = np.clip(w, 0.0, None)
    w /= w.sum()
    return w


def _distribution_edge_weighted(n: int, p: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64) + 0.5
    j = (n - np.arange(n, dtype=np.float64) - 0.5)
    w = (i * j) ** (-p)
    w_sum = w.sum()
    if w_sum == 0.0:
        return np.full(n, 1.0 / n, dtype=np.float64)
    w /= w_sum
    return w


def _distribution_exponential_edge(n: int, c: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    w = np.exp(-c * i) + np.exp(-c * (n - 1 - i))
    w /= w.sum()
    return w


def _distribution_kaiser(n: int, beta: float) -> np.ndarray:
    """Kaiser window (beta≥0).  beta=0 yields the rectangular window."""
    if n == 1:
        return np.array([1.0])
    i = np.arange(n, dtype=np.float64)
    arg = beta * np.sqrt(1.0 - ((2.0 * i) / (n - 1) - 1.0) ** 2)
    w = i0(arg) / i0(beta) if beta > 0 else np.ones_like(i)
    w = np.maximum(w, 0.0)
    w /= w.sum()
    return w


def _distribution_blackman(n: int) -> np.ndarray:
    """Blackman window."""
    if n == 1:
        return np.array([1.0])
    i = np.arange(n, dtype=np.float64)
    a0 = 0.42
    a1 = 0.5
    a2 = 0.08
    w = a0 - a1 * np.cos(2.0 * np.pi * i / (n - 1)) + a2 * np.cos(4.0 * np.pi * i / (n - 1))
    w = np.clip(w, 0.0, None)
    w /= w.sum()
    return w


def _distribution_nuttall(n: int) -> np.ndarray:
    """Nuttall window."""
    if n == 1:
        return np.array([1.0])
    i = np.arange(n, dtype=np.float64)
    a0, a1, a2, a3 = 0.355768, 0.487396, 0.144232, 0.012604
    w = (a0
         - a1 * np.cos(2.0 * np.pi * i / (n - 1))
         + a2 * np.cos(4.0 * np.pi * i / (n - 1))
         - a3 * np.cos(6.0 * np.pi * i / (n - 1)))
    w = np.clip(w, 0.0, None)
    w /= w.sum()
    return w


def _distribution_tukey(n: int, alpha: float) -> np.ndarray:
    """Tukey (tapered cosine) window, 0≤alpha≤1."""
    if n == 1:
        return np.array([1.0])
    i = np.arange(n, dtype=np.float64)
    period = n - 1
    taper = alpha / 2.0
    w = np.ones_like(i)
    # left taper
    left = i < taper * period
    w[left] = 0.5 * (1 + np.cos(np.pi * (2 * i[left] / (alpha * period) - 1)))
    # right taper
    right = i > (1 - taper) * period
    w[right] = 0.5 * (1 + np.cos(np.pi * (2 * i[right] / (alpha * period) - 2 / alpha + 1)))
    w = np.clip(w, 0.0, None)
    w /= w.sum()
    return w


def _distribution_random_sparse(
    n: int, rng: np.random.Generator, density: float = 0.01
) -> np.ndarray:
    """Sparse random vector with prescribed density of non‑zeros."""
    k = max(2, int(n * density))
    idx = rng.choice(n, size=k, replace=False)
    vals = np.zeros(n, dtype=np.float64)
    vals[idx] = rng.random(k)
    s = vals.sum()
    if s == 0.0:
        vals[0] = 1.0
        s = 1.0
    vals /= s
    return vals


def _mix_with_uniform(base: np.ndarray, lam: float) -> np.ndarray:
    """Convex combination λ·uniform + (1‑λ)·base."""
    n = base.shape[0]
    uniform = np.full(n, 1.0 / n, dtype=np.float64)
    return lam * uniform + (1.0 - lam) * base

# Adam‑style exponentiated‑gradient descent (exact L∞ sub‑gradient)
# ----------------------------------------------------------------------
def _subgradient_descent_adam(
    a0: np.ndarray,
    time_limit: float,
    best_seq: np.ndarray,
    best_score: float,
    eta: float = 0.5,
    beta1: float = 0.9,
    beta2: float = 0.999,
    epsilon: float = 1e-8,
    eval_every: int = 200,
) -> tuple[np.ndarray, float]:
    """
    Adam‑like exponentiated‑gradient descent on the exact L∞ objective.
    Sub‑gradient is built from *all* indices attaining the maximal convolution value.
    """
    a = a0.copy()
    n = a.shape[0]
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    iter_cnt = 0

    while time.time() < time_limit:
        iter_cnt += 1

        conv = _fft_self_convolve(a)
        max_val = conv.max()
        max_idxs = np.where(conv == max_val)[0]

        # exact sub‑gradient (sum of contributions from each maximal index)
        grad = np.zeros_like(a)
        for idx in max_idxs:
            i_start = max(0, idx - (n - 1))
            i_end = min(n - 1, idx)
            i = np.arange(i_start, i_end + 1, dtype=np.int64)
            j = idx - i
            grad[i] += 2.0 * a[j]

        # optional clipping to keep numbers moderate (prevents exp overflow)
        grad = np.clip(grad, -1e5, 1e5)

        # Adam moment updates
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)

        # bias‑corrected estimates
        m_hat = m / (1.0 - beta1 ** iter_cnt)
        v_hat = v / (1.0 - beta2 ** iter_cnt)

        # exponentiated‑gradient step
        step = -eta * m_hat / (np.sqrt(v_hat) + epsilon)
        step = np.clip(step, -50.0, 50.0)          # avoid overflow in exp
        a = a * np.exp(step)

        # simplex projection (maintains non‑negativity and Σa = 1)
        s = a.sum()
        if s <= 0.0 or not np.isfinite(s):
            a = np.full(n, 1.0 / n, dtype=np.float64)
        else:
            a = a / s

        # occasional evaluation
        if iter_cnt % eval_every == 0:
            cur = 2.0 * n * max_val
            if cur < best_score - 1e-14:
                best_score = cur
                best_seq = a.copy()
                print(f"[{time.time():.1f}] adam it={iter_cnt} new global best {best_score:.6f}")

    return best_seq, best_score


# ----------------------------------------------------------------------
# Greedy pairwise mass transfer hill‑climber
# ----------------------------------------------------------------------
def _pairwise_swap(
    a0: np.ndarray,
    time_limit: float,
    best_seq: np.ndarray,
    best_score: float,
    max_trials: int | None = None,
) -> tuple[np.ndarray, float]:
    """
    Randomly move a fraction of mass from one index to another.
    The move is accepted only if it strictly improves the exact objective.
    """
    a = a0.copy()
    n = a.shape[0]
    trials = 0
    while time.time() < time_limit:
        if max_trials is not None and trials >= max_trials:
            break
        trials += 1

        src_candidates = np.where(a > 1e-14)[0]
        if src_candidates.size == 0:
            break
        i = np.random.choice(src_candidates)
        j = np.random.randint(n - 1)
        if j >= i:
            j += 1

        delta = a[i] * np.random.rand() * 0.45   # up to 45 % of source mass
        if delta < 1e-15:
            continue

        new_a = a.copy()
        new_a[i] -= delta
        new_a[j] += delta

        cur = _objective(new_a)
        if cur < best_score - 1e-14:
            a = new_a
            best_score = cur
            best_seq = a.copy()
            print(
                f"[{time.time():.1f}] swap i={i}→j={j} δ={delta:.6g} new best {best_score:.6f}"
            )
    return best_seq, best_score
