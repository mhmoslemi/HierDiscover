"""
propose_candidate

A three‑stage optimizer for the “step‑function” problem.

1. **Diverse seed generation** – a rich family of analytic shapes
   (uniform, arcsine, sine, cosine, quadratic, symmetric Beta,
   edge‑weighted, power‑sum) on several lengths, together with many
   random Dirichlet and sparse seeds.  The parameter grids are
   intentionally fine‑grained (including p≈10⁻⁴) so that the search can
   discover extremely edge‑heavy distributions.

2. **Exact mirror (exponentiated) descent** – each of the top few seeds
   is refined using the true sub‑gradient of the max‑convolution.
   The step‑size decays as η₀/√t and the iterate is projected onto the
   probability simplex after every update.  The run time is split
   proportionally amongst the candidates.

3. **Simulated annealing + greedy polishing** – starting from the best
   refined vector we run a temperature‑controlled random walk that moves
   a fraction of mass between two coordinates; uphill moves are accepted
   with probability exp(−Δ/𝑇).  Finally a fast greedy pairwise swap
   polishing is performed.

The routine respects the supplied time budget (default 1000 s) and
returns a plain list of non‑negative floats ready for ``evaluate_sequence``.
"""

# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Euclidean projection of ``v`` onto the simplex {x≥0, Σx=s}."""
    if s < 0:
        raise ValueError("Simplex radius must be non‑negative")
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.where(u - cssv / (np.arange(1, len(v) + 1)) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    return np.maximum(v - theta, 0.0)


# ----------------------------------------------------------------------
def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Full self‑convolution a*a using a real FFT (length 2·n−1)."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1) - 1).bit_length()   # next power of two
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[: 2 * n - 1]


# ----------------------------------------------------------------------
def _objective(a: np.ndarray) -> float:
    """
    Exact objective ``2·n·max(conv(a,a))`` for a vector whose entries sum to 1.
    """
    n = a.shape[0]
    b = _fft_self_convolve(a)
    return float(2.0 * n * b.max())

# Analytic families ----------------------------------------------------
def _distribution_uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _distribution_arcsine(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    vals /= vals.sum()
    return vals


def _distribution_sine(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64) + 0.5
    w = np.sin(np.pi * i / n)
    w /= w.sum()
    return w


def _distribution_cosine(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64) + 0.5
    w = 1.0 - np.cos(np.pi * i / n)
    w /= w.sum()
    return w


def _distribution_quadratic(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    x = (2.0 * i) / (n - 1) - 1.0
    w = 1.0 - x * x
    w = np.clip(w, 0.0, None)
    w /= w.sum()
    return w


def _distribution_beta_sym(n: int, p: float) -> np.ndarray:
    """Symmetric Beta(p,p) on a regular grid."""
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    # avoid log(0) when p<1
    eps = np.finfo(float).eps
    x = np.clip(x, eps, 1 - eps)
    w = np.exp((p - 1) * np.log(x) + (p - 1) * np.log(1.0 - x))
    if w.sum() == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _distribution_edge_weighted(n: int, p: float) -> np.ndarray:
    """
    Edge‑weighted family: w_i ∝ ((i+½)*(n‑i‑½))^{‑p}.
    """
    i = np.arange(n, dtype=np.float64) + 0.5
    j = (n - np.arange(n, dtype=np.float64) - 0.5)
    w = (i * j) ** (-p)
    w = np.clip(w, 0.0, None)
    total = w.sum()
    if total > 0:
        w /= total
    else:
        w = np.full(n, 1.0 / n)
    return w


def _distribution_power_sum(n: int, p: float) -> np.ndarray:
    """w_i ∝ (i+1)^{‑p} + (n‑i)^{‑p} – another edge‑heavy family."""
    i = np.arange(1, n + 1, dtype=np.float64)
    w = i ** (-p) + (n + 1 - i) ** (-p)
    w = np.clip(w, 0.0, None)
    w /= w.sum()
    return w


def _distribution_random_sparse(
    n: int, rng: np.random.Generator, density: float = 0.01
) -> np.ndarray:
    """Sparse random vector with a prescribed density of non‑zeros."""
    k = max(2, int(n * density))
    idx = rng.choice(n, size=k, replace=False)
    vals = np.zeros(n, dtype=np.float64)
    vals[idx] = rng.random(k)
    if vals.sum() == 0.0:
        vals[0] = 1.0
    vals /= vals.sum()
    return vals


def _mix_with_uniform(base: np.ndarray, lam: float) -> np.ndarray:
    """Convex combination λ·uniform + (1‑λ)·base."""
    n = base.shape[0]
    uniform = np.full(n, 1.0 / n, dtype=np.float64)
    return lam * uniform + (1.0 - lam) * base


# ----------------------------------------------------------------------
def _max_convolution_subgradient(a: np.ndarray) -> np.ndarray:
    """
    Sub‑gradient of the max‑convolution functional.
    Returns a vector of the same shape as ``a``.
    """
    conv = _fft_self_convolve(a)
    max_val = conv.max()
    # treat all numerically maximal indices as active
    tol = 1e-12 * max_val
    active = np.where(conv >= max_val - tol)[0]
    n = a.shape[0]
    grad = np.zeros_like(a)
    for k in active:
        lo = max(0, k - (n - 1))
        hi = min(n - 1, k)
        idx = np.arange(lo, hi + 1)
        # contribution from each (i, k-i) pair appears twice except when i=k-i
        grad[idx] += 2.0 * a[k - idx]
    if len(active) > 0:
        grad /= len(active)          # convex combination of active gradients
    return grad

def _exp_gradient_descent(
    a0: np.ndarray,
    time_limit: float,
    best_seq: np.ndarray,
    best_score: float,
    eta0: float = 0.6,
    report_every: int = 400,
    max_iter: int = 300_000,
) -> tuple[np.ndarray, float]:
    """
    Exponentiated‑gradient descent on the exact max‑convolution objective.
    Uses the true sub‑gradient of the maximum convolution entry.
    """
    a = a0.copy()
    n = a.shape[0]
    t = 0
    # main loop
    while time.time() < time_limit and t < max_iter:
        t += 1
        grad = _max_convolution_subgradient(a)
        eta = eta0 / math.sqrt(t)
        a = a * np.exp(-eta * grad)
        sum_a = a.sum()
        if sum_a > 0.0:
            a = a / sum_a
        else:
            a = np.full(n, 1.0 / n, dtype=np.float64)

        if t % report_every == 0:
            cur = _objective(a)
            if cur < best_score - 1e-14:
                best_score = cur
                best_seq = a.copy()
                print(
                    f"[{time.time():.1f}] exp‑gd it={t} new global best {best_score:.6f}"
                )
    return best_seq, best_score

def _pairwise_swap(
    a0: np.ndarray,
    time_limit: float,
    best_seq: np.ndarray,
    best_score: float,
    max_trials: int | None = None,
) -> tuple[np.ndarray, float]:
    """
    Randomly move a fraction of mass from one index to another.
    Only moves that strictly improve the exact objective are kept.
    """
    a = a0.copy()
    n = a.shape[0]
    trials = 0
    while time.time() < time_limit:
        if max_trials is not None and trials >= max_trials:
            break
        trials += 1

        src = np.where(a > 1e-14)[0]
        if src.size == 0:
            break
        i = np.random.choice(src)
        j = np.random.randint(n - 1)
        if j >= i:
            j += 1

        delta = a[i] * np.random.rand() * 0.45   # up to 45 % of source mass
        if delta < 1e-15:
            continue

        new_a = a.copy()
        new_a[i] -= delta
        new_a[j] += delta

        cur = _objective(new_a)
        if cur < best_score - 1e-14:
            a = new_a
            best_score = cur
            best_seq = a.copy()
            print(
                f"[{time.time():.1f}] swap i={i}→j={j} δ={delta:.6g} new best {best_score:.6f}"
            )
    return best_seq, best_score

def _simulated_annealing(
    a0: np.ndarray,
    time_limit: float,
    rng: np.random.Generator,
    best_seq: np.ndarray,
    best_score: float,
    init_temp: float = 1e-2,
    final_temp: float = 1e-6,
    cooling: float = 0.9997,
    max_move_frac: float = 0.35,
) -> tuple[np.ndarray, float]:
    """
    Temperature‑controlled random walk: move a fraction of mass from a
    source coordinate to a destination coordinate.  Moves that improve
    the objective are always accepted; worsening moves are accepted
    probabilistically according to the current temperature.
    """
    a = a0.copy()
    n = a.shape[0]
    cur_score = _objective(a)
    best_local_seq = a.copy()
    best_local_score = cur_score
    temp = init_temp
    it = 0

    while time.time() < time_limit:
        it += 1

        # choose source proportionally to its mass (more mass → more moves)
        src_candidates = np.where(a > 1e-14)[0]
        if src_candidates.size == 0:
            break
        probs = a[src_candidates] / a[src_candidates].sum()
        i = rng.choice(src_candidates, p=probs)

        # choose a destination distinct from source uniformly
        j = rng.integers(0, n - 1)
        if j >= i:
            j += 1

        # mass to move
        delta = a[i] * rng.random() * max_move_frac
        if delta < 1e-15:
            continue

        new_a = a.copy()
        new_a[i] -= delta
        new_a[j] += delta

        new_score = _objective(new_a)
        accept = False
        if new_score < cur_score:
            accept = True
        else:
            # Metropolis criterion
            if rng.random() < math.exp(-(new_score - cur_score) / max(temp, 1e-15)):
                accept = True

        if accept:
            a = new_a
            cur_score = new_score
            if cur_score < best_local_score:
                best_local_score = cur_score
                best_local_seq = a.copy()
                # propagate to global best if better
                if cur_score < best_score:
                    best_score = cur_score
                    best_seq = a.copy()
                    print(
                        f"[{time.time():.1f}] SA new global best {best_score:.6f}"
                    )

        # temperature schedule
        if it % 50 == 0:
            temp = max(temp * cooling, final_temp)

    return best_local_seq, best_local_score

    # ------------------------------------------------------------------
    if time.time() < deadline and overall_best_seq is not None:
        print(f"[{time.time():.1f}] simulated annealing phase")
        sa_deadline = deadline
        a_opt, sc_opt = _simulated_annealing(
            overall_best_seq.copy(),
            sa_deadline,
            rng,
            best_seq=overall_best_seq,
            best_score=overall_best_score,
            init_temp=2e-2,
            final_temp=1e-6,
            cooling=0.9995,
            max_move_frac=0.4,
        )
        if sc_opt < overall_best_score - 1e-12:
            overall_best_score = sc_opt
            overall_best_seq = a_opt.copy()
            print(f"[{time.time():.1f}] after SA new global best {overall_best_score:.6f}")

    # ------------------------------------------------------------------
    # 6.  Final greedy polishing (pairwise swaps)
    # ------------------------------------------------------------------
    if time.time() < deadline and overall_best_seq is not None:
        print(f"[{time.time():.1f}] final greedy polishing")
        a_opt, sc_opt = _pairwise_swap(
            overall_best_seq.copy(),
            deadline,
            best_seq=overall_best_seq,
            best_score=overall_best_score,
        )
        if sc_opt < overall_best_score - 1e-12:
            overall_best_score = sc_opt
            overall_best_seq = a_opt.copy()
            print(f"[{time.time():.1f}] after final swaps new best {overall_best_score:.6f}")

    print(
        f"Final best score (2·n·max(conv)) = {overall_best_score:.6f} (length={len(overall_best_seq)})"
    )
    # Return a plain Python list for the external evaluator
    return overall_best_seq.tolist()
