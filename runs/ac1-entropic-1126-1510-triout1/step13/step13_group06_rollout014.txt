def generate_seed_distributions(n, rng):
    seeds = []
    # uniform
    seeds.append(_distribution_uniform(n))
    # arcsine
    seeds.append(_distribution_arcsine(n))
    # triangular: linear ramp up then down
    tri = np.arange(1, n+1, dtype=float)
    tri = tri / tri.sum()
    # Actually we want symmetrical linear shape: values increase linearly from start to middle then decrease.
    # We'll define shape to peak at middle.
    idx = np.arange(n)
    center = (n-1)/2
    tri = 1 - np.abs(idx - center)/center  # values from 0 to 1
    tri[tri<0] = 0
    tri = tri / tri.sum()
    seeds.append(tri)

    # Beta symmetric for p in p_grid
    p_grid = np.concatenate([... as earlier]) but we can restrict.
    for p in p_grid:
        seeds.append(_distribution_beta_sym(n, p))
        seeds.append(_distribution_edge_weighted(n, p))
        seeds.append(_distribution_sine_weighted(n, p))

    # Asymmetric beta random parameters, few random.

        a_param = 10 ** rng.uniform(-2, 0.8)

"""
Propose a low‑score step‑function.

Algorithm
---------
1. **Analytic seeds** – generate a diverse set of probability vectors:
   uniform, arcsine, symmetric/asymmetric Beta, edge‑weighted and sine‑weighted.
   For each family also consider a convex mix with the uniform distribution.
   Keep the ∼30 best (lowest exact objective).

2. **Exponentiated Adam descent** – for each seed run a multiplicative
   Adam‑style gradient descent on the *soft‑max* surrogate of the max‑convolution.
   The gradient of the weighted sum of convolution entries is obtained by a single
   FFT‑based correlation (O(n log n)).  The temperature τ is increased so the
   soft‑max converges to the true max.  After each step the vector is renormalised
   onto the probability simplex.

3. **Greedy pairwise swaps** – repeatedly move a small chunk of mass between two
   positions whenever this reduces the exact objective.

4. **Cheap simulated‑annealing walk** – final stochastic refinement that accepts
   occasional worsening moves.

All steps respect the supplied time budget (default 1000 s).  The returned list
is a plain Python list; ``evaluate_sequence`` will perform the final checks.
"""

# Utility functions

def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x≥0, Σx = s}."""
    if s < 0:
        raise ValueError("simplex radius must be non‑negative")
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho_arr = np.where(u - cssv / (np.arange(1, len(v) + 1)) > 0)[0]
    if rho_arr.size == 0:
        theta = 0.0
    else:
        rho = rho_arr[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Full linear self‑convolution a*a using a real FFT (returns length 2n‑1)."""
    n = a.shape[0]
    need = 2 * n - 1
    size = 1 << ((need - 1).bit_length())
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[:need]


def _objective(a: np.ndarray) -> float:
    """Exact objective 2·n·max(conv(a,a)) for a vector with sum = 1."""
    n = a.shape[0]
    b = _fft_self_convolve(a)
    return float(2.0 * n * b.max())


# ----------------------------------------------------------------------
# Analytic families (all return a probability vector, i.e. sum=1)
# ----------------------------------------------------------------------
def _distribution_uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _distribution_arcsine(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    vals /= vals.sum()
    return vals


def _distribution_beta_sym(n: int, p: float) -> np.ndarray:
    """Symmetric beta (Beta(p,p)) on the discrete grid."""
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    w = np.exp((p - 1) * np.log(x) + (p - 1) * np.log(1.0 - x))
    if w.sum() == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _distribution_beta_asym(n: int, a: float, b: float) -> np.ndarray:
    """Asymmetric beta (Beta(a,b)) on the discrete grid."""
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    w = np.exp((a - 1) * np.log(x) + (b - 1) * np.log(1.0 - x))
    if w.sum() == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _distribution_edge_weighted(n: int, p: float) -> np.ndarray:
    """U‑shaped family: w_i ∝ ((i+0.5)*(n‑i‑0.5))^(−p)."""
    i = np.arange(n, dtype=np.float64) + 0.5
    j = (n - np.arange(n, dtype=np.float64) - 0.5)
    w = (i * j) ** (-p)
    if w.sum() == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _distribution_sine_weighted(n: int, p: float) -> np.ndarray:
    """Weight proportional to sin(π (i+½)/n)^p."""
    theta = np.pi * (np.arange(n, dtype=np.float64) + 0.5) / n
    w = np.sin(theta) ** p
    if w.sum() == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _distribution_triangular(n: int) -> np.ndarray:
    """Symmetric triangular (linear) shape."""
    idx = np.arange(n, dtype=np.float64)
    center = (n - 1) / 2.0
    w = 1.0 - np.abs(idx - center) / center
    w[w < 0] = 0.0
    w /= w.sum()
    return w


def _distribution_random_sparse(
    n: int, rng: np.random.Generator, density: float = 0.006
) -> np.ndarray:
    """Sparse random vector with a given density of non‑zeros."""
    k = max(2, int(n * density))
    idx = rng.choice(n, size=k, replace=False)
    vals = np.zeros(n, dtype=np.float64)
    vals[idx] = rng.random(k)
    if vals.sum() == 0.0:
        vals[0] = 1.0
    vals /= vals.sum()
    return vals


def _mix_with_uniform(base: np.ndarray, lam: float) -> np.ndarray:
    """Convex combination lam·U + (1‑lam)·base, where U is uniform."""
    n = base.shape[0]
    uniform = np.full(n, 1.0 / n, dtype=np.float64)
    return lam * uniform + (1.0 - lam) * base


# ----------------------------------------------------------------------
# Gradient utilities (soft‑max surrogate)
# ----------------------------------------------------------------------
def _softmax_weights(conv: np.ndarray, tau: float) -> np.ndarray:
    """Soft‑max of τ·conv (stable). Returns a probability vector."""
    max_c = conv.max()
    w_raw = np.exp(tau * (conv - max_c))
    total = w_raw.sum()
    if total == 0.0:
        return np.full_like(conv, 1.0 / conv.size)
    return w_raw / total


def _weighted_gradient(a: np.ndarray, w: np.ndarray) -> np.ndarray:
    """
    Gradient of   L(a)= Σ_i w_i·(a*a)[i]   w.r.t a.
    Closed‑form:   ∇L = 2 * (a_rev ⊛ w)   (restricted to the first n entries).
    """
    n = a.shape[0]
    a_rev = a[::-1]                     # length n
    need = len(w) + n - 1                # = 3·n‑2
    size = 1 << ((need - 1).bit_length())
    A = np.fft.rfft(a_rev, n=size)
    W = np.fft.rfft(w, n=size)
    conv = np.fft.irfft(A * W, n=size)
    conv = conv[:need]                  # length need
    grad = 2.0 * conv[n - 1 : n - 1 + n]   # slice of length n
    return grad


# ----------------------------------------------------------------------
# Exponentiated Adam descent on the soft‑max surrogate
# ----------------------------------------------------------------------
def _exponential_adam_descent(
    a: np.ndarray,
    deadline: float,
    best_seq: np.ndarray,
    best_score: float,
    alpha: float = 0.20,
    beta1: float = 0.90,
    beta2: float = 0.999,
    epsilon: float = 1e-8,
    tau0: float = 0.5,
    tau_mult: float = 1.003,
    eval_interval: int = 100,
    patience: int = 2000,
) -> tuple[np.ndarray, float]:
    """
    Multiplicative Adam‑style descent on the *soft‑max* approximation of the
    max‑convolution.  The entire convolution vector is used, gradient obtained
    via a single FFT‑based correlation.
    """
    n = a.shape[0]
    rng = np.random.default_rng()
    t = 0
    no_improve = 0
    tau = tau0
    m = np.zeros_like(a)
    v = np.zeros_like(a)

    while time.time() < deadline and t < 500_000:
        t += 1

        # convolution and soft‑max weights
        conv = _fft_self_convolve(a)
        w = _softmax_weights(conv, tau)

        # gradient of weighted sum
        grad = _weighted_gradient(a, w)

        # Adam moment updates
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad ** 2)

        # bias‑corrected estimates
        m_hat = m / (1 - beta1 ** t)
        v_hat = v / (1 - beta2 ** t)

        step = alpha * m_hat / (np.sqrt(v_hat) + epsilon)

        # multiplicative update, always stays positive
        a = a * np.exp(-step)

        # renormalise onto simplex (preserves Σa = 1)
        s = a.sum()
        if s == 0.0:
            a = np.full(n, 1.0 / n, dtype=np.float64)
        else:
            a = a / s

        # increase temperature → focus on the true max
        tau *= tau_mult

        # occasional exact evaluation
        if t % eval_interval == 0:
            cur = _objective(a)
            if cur < best_score - 1e-14:
                best_score = cur
                best_seq = a.copy()
                no_improve = 0
                print(f"[{time.time():.1f}] Adam it={t} new best {best_score:.6f}")
            else:
                no_improve += 1

        # jitter if stuck
        if no_improve >= patience:
            jitter = rng.exponential(scale=0.003, size=n)
            a = a + jitter
            a = np.maximum(a, 0.0)
            a = a / a.sum()
            no_improve = 0

    return best_seq, best_score


# ----------------------------------------------------------------------
# Pairwise swap refinement
# ----------------------------------------------------------------------
def _pairwise_swap_local(
    a: np.ndarray,
    deadline: float,
    best_seq: np.ndarray,
    best_score: float,
    max_trials: int = 300_000,
) -> tuple[np.ndarray, float]:
    """Randomly transfer a small chunk of mass; keep change if objective improves."""
    n = a.shape[0]
    trials = 0
    rng = np.random.default_rng()
    while time.time() < deadline and trials < max_trials:
        trials += 1

        src = np.where(a > 1e-12)[0]
        if src.size == 0:
            break
        i = rng.choice(src)

        j = rng.integers(n - 1)
        if j >= i:
            j += 1

        delta = a[i] * rng.random() * 0.35
        if delta < 1e-14:
            continue

        new_a = a.copy()
        new_a[i] -= delta
        new_a[j] += delta

        cur = _objective(new_a)
        if cur < best_score - 1e-14:
            a = new_a
            best_score = cur
            best_seq = a.copy()
            print(
                f"[{time.time():.1f}] swap i={i}→j={j} δ={delta:.3e} new best {best_score:.6f}"
            )
    return best_seq, best_score

# Cheap simulated‑annealing walk
# ----------------------------------------------------------------------
def _random_local_search(
    a: np.ndarray,
    deadline: float,
    best_seq: np.ndarray,
    best_score: float,
    init_temp: float = 0.008,
    decay: float = 0.9997,
) -> tuple[np.ndarray, float]:
    """Very cheap SA that repeatedly transfers a random amount of mass."""
    n = a.shape[0]
    rng = np.random.default_rng()
    T = init_temp
    while time.time() < deadline:
        i = rng.integers(n)
        j = rng.integers(n - 1)
        if j >= i:
            j += 1
        delta = a[i] * rng.random() * 0.3
        if delta < 1e-14:
            continue
        new_a = a.copy()
        new_a[i] -= delta
        new_a[j] += delta
        cur = _objective(new_a)
        if cur < best_score - 1e-14:
            a = new_a
            best_score = cur
            best_seq = a.copy()
            print(f"[{time.time():.1f}] SA improve {i}->{j} Δ={delta:.3e} best {best_score:.6f}")
        else:
            if rng.random() < math.exp(-(cur - best_score) / max(T, 1e-12)):
                a = new_a
        T *= decay
    return best_seq, best_score

def _generate_candidates_for_n(n: int, rng: np.random.Generator):
    """Yield a stream of probability vectors for a given length."""
    # deterministic families
    yield _distribution_uniform(n)
    yield _distribution_arcsine(n)
    yield _distribution_triangular(n)

    # parametric families
    p_grid = np.concatenate(
        [
            np.logspace(-2, -0.5, 6, base=10),   # 0.01 … 0.316
            np.logspace(-0.5, 0, 6, base=10),    # 0.316 … 1
            np.logspace(0, 0.5, 6, base=10),     # 1 … 3.16
        ]
    )
    lam_grid = np.linspace(0.0, 1.0, 5)          # 0 … 1

    for p in p_grid:
        beta = _distribution_beta_sym(n, p)
        edge = _distribution_edge_weighted(n, p)
        sine = _distribution_sine_weighted(n, p)

        for base in (beta, edge, sine):
            for lam in lam_grid:
                yield _mix_with_uniform(base, lam)

    # asymmetric beta (few random draws)
    for _ in range(4):
        a_param = 10 ** rng.uniform(-2, 0.6)   # 0.01 … ≈4
        b_param = 10 ** rng.uniform(-2, 0.6)
        yield _distribution_beta_asym(n, a_param, b_param)

    # random Dirichlet vectors
    for _ in range(4):
        alpha = rng.uniform(0.01, 2.0)
        yield rng.dirichlet(np.full(n, alpha))

    # sparse random vectors
    for _ in range(4):
        yield _distribution_random_sparse(n, rng, density=0.005)

            )

            deadline,
            best_seq=overall_best_seq,
            best_score=overall_best_score,
            max_trials=600_000,
        )

    # ------------------------------------------------------------------
    # 6. Final simulated‑annealing style walk
    if time.time() < deadline:
        print(f"[{time.time():.1f}] final stochastic local search")
        overall_best_seq, overall_best_score = _random_local_search(
            overall_best_seq.copy(),
            deadline,
            best_seq=overall_best_seq,
            best_score=overall_best_score,
            init_temp=0.008,
            decay=0.9997,
        )

    # ------------------------------------------------------------------
    print(
        f"Final best score: {overall_best_score:.6f} (length={len(overall_best_seq)})"
    )
    # Return a plain Python list; evaluate_sequence will perform final checks.
    return overall_best_seq.tolist()
