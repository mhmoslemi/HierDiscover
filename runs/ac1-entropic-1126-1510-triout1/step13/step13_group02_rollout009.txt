"""
step_function_search_v2.py

A 1000 s time‑bounded search for a non‑negative step function that
minimises the evaluation metric used by ``evaluate_sequence``:

    2·n·max(conv(a,a)) / (∑a)²

The algorithm works in four phases:

1. **Broad seeding** – generate a large, diverse pool of probability
   vectors from many analytic families (uniform, arcsine, Hann,
   edge‑weighted, generalized‑normal, Beta, random Dirichlet,
   sparse).  All vectors are normalised to Σa = 1 and the best few are
   retained.

2. **Accelerated mirror‑descent** – each retained seed is polished with
   an exponentiated‑gradient / Adam‑style update on the *exact*
   objective.  The update uses a *soft* sub‑gradient that averages the
   gradients of all convolution entries whose value is within a tiny
   tolerance of the current maximum (this helps escape plateaus).

3. **Upsampling & final polish** – the best vector is linearly
   up‑sampled (if time permits) and refined again with the accelerated
   descent.  Afterwards a pairwise‑mass‑transfer simulated‑annealing
   phase makes a last push.

All steps respect the 1000 s budget (2 s safety margin).  The routine
returns a plain ``list[float]`` ready for ``evaluate_sequence``.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Simplex utilities
# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Euclidean projection of ``v`` onto the simplex {x≥0, Σx=s}."""
    if s < 0:
        raise ValueError("simplex radius must be non‑negative")
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.where(u - cssv / (np.arange(1, len(v) + 1)) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# FFT‑based self‑convolution (full linear convolution)
# ----------------------------------------------------------------------
def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Return the full linear self‑convolution a*a using a real FFT."""
    n = a.shape[0]
    # zero‑pad to the next power of two for speed
    size = 1 << ((2 * n - 1) - 1).bit_length()
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[: 2 * n - 1]


# ----------------------------------------------------------------------
# Exact objective (Σa = 1 ⇒ denominator = 1)
# ----------------------------------------------------------------------
def _objective(a: np.ndarray) -> float:
    """Exact value of ``evaluate_sequence`` for a probability vector."""
    n = a.shape[0]
    b = _fft_self_convolve(a)
    max_b = b.max()
    return float(2.0 * n * max_b)


# ----------------------------------------------------------------------
# Analytic families (return probability vectors, i.e. Σa = 1)
# ----------------------------------------------------------------------
def _distribution_uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _distribution_arcsine(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    vals /= vals.sum()
    return vals


def _distribution_hann(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    w = 0.5 - 0.5 * np.cos(2.0 * np.pi * i / (n - 1))
    w /= w.sum()
    return w


def _distribution_edge_weighted(n: int, p: float) -> np.ndarray:
    """Weight ∝ (i·j)^{-p} where i,j are distances to the two edges."""
    i = np.arange(n, dtype=np.float64) + 0.5
    j = (n - np.arange(n, dtype=np.float64) - 0.5)
    w = (i * j) ** (-p)
    w /= w.sum()
    return w


def _distribution_generalized_normal(n: int, q: float) -> np.ndarray:
    """Density ∝ exp(‑|x‑½|^{q}) on [0,1]."""
    i = np.arange(n, dtype=np.float64) + 0.5
    x = i / n - 0.5
    w = np.exp(- (np.abs(x) / 0.5) ** q)
    w /= w.sum()
    return w


def _distribution_beta(n: int, a: float, b: float) -> np.ndarray:
    """Beta(α,β) density on [0,1] (α,β > 0)."""
    i = np.arange(n, dtype=np.float64) + 0.5
    x = i / n
    # avoid exact 0/1 to keep finite values
    eps = 1e-16
    x = np.clip(x, eps, 1 - eps)
    w = (x ** (a - 1.0)) * ((1.0 - x) ** (b - 1.0))
    w_sum = w.sum()
    if w_sum == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w_sum
    return w


def _distribution_random_sparse(
    n: int, rng: np.random.Generator, density: float = 0.01
) -> np.ndarray:
    """Sparse random vector with a given density."""
    k = max(2, int(n * density))
    idx = rng.choice(n, size=k, replace=False)
    vals = np.zeros(n, dtype=np.float64)
    vals[idx] = rng.random(k)
    if vals.sum() == 0.0:
        vals[0] = 1.0
    vals /= vals.sum()
    return vals


def _mix_with_uniform(base: np.ndarray, lam: float) -> np.ndarray:
    """Convex combination with the uniform distribution."""
    n = base.shape[0]
    uniform = np.full(n, 1.0 / n, dtype=np.float64)
    return lam * uniform + (1.0 - lam) * base


# ----------------------------------------------------------------------
# Gradient utilities
# ----------------------------------------------------------------------
def _grad_of_conv_at_k(a: np.ndarray, k: int) -> np.ndarray:
    """
    Gradient of the convolution entry (a*a)[k] w.r.t. ``a``.
    Returns a vector of length n.  The entry is 2*a_{k-i} whenever the
    index ``k-i`` lies inside the array.
    """
    n = a.shape[0]
    grad = np.zeros_like(a)
    if k < n:
        # overlap region: i=0..k  (length k+1)
        grad[: k + 1] = 2.0 * a[k::-1]
    else:
        # overlap region: i=k-(n-1)..n-1  (length 2n-1-k)
        start = k - (n - 1)
        grad[start:] = 2.0 * a[start:][::-1]
    return grad


def _soft_active_gradient(a: np.ndarray, eps_rel: float = 1e-12) -> np.ndarray:
    """
    Compute a “soft” sub‑gradient: average the gradients of all
    convolution entries whose value is within ``eps_rel`` relative of
    the current maximum.
    """
    conv = _fft_self_convolve(a)
    max_val = conv.max()
    # treat all entries whose value is within the tolerance as active
    tol = max(eps_rel * max_val, 1e-16)
    active = np.where(conv >= max_val - tol)[0]

    # average the gradients of the active entries
    grad = np.zeros_like(a)
    for k in active:
        grad += _grad_of_conv_at_k(a, int(k))
    if len(active) > 0:
        grad /= len(active)
    return grad


# ----------------------------------------------------------------------
# Accelerated mirror‑descent (Adam‑style) on the exact objective
# ----------------------------------------------------------------------
def _adam_descent(
    a0: np.ndarray,
    deadline: float,
    best_seq: np.ndarray,
    best_score: float,
    eta0: float = 0.5,
    beta1: float = 0.9,
    beta2: float = 0.999,
    eval_every: int = 80,
) -> tuple[np.ndarray, float]:
    """
    Adam‑style optimisation on the *exact* objective while staying on
    the probability simplex (exponentiated‑gradient update).  The
    algorithm works in the log‑domain: we keep ``log_a = log(a)``,
    perform Adam on the gradients in that domain and finally map back
    with a softmax (ensuring non‑negativity and Σ=1).
    """
    n = a0.shape[0]
    # initialise in log‑space (add epsilon to avoid log(0))
    eps = 1e-16
    log_a = np.log(a0 + eps)

    m = np.zeros_like(log_a)
    v = np.zeros_like(log_a)

    t = 0
    while time.time() < deadline:
        t += 1

        # recover a from log_a (softmax)
        a = np.exp(log_a - np.max(log_a))  # for numerical stability
        a /= a.sum()

        # compute a sub‑gradient of the objective w.r.t. a
        grad_a = _soft_active_gradient(a)

        # gradient w.r.t. log_a:  ∂obj/∂log_a_i = a_i * (grad_a_i - <a,grad_a>)
        # where <a,grad_a> is the Lagrange multiplier enforcing Σa=1.
        mean = np.dot(a, grad_a)          # scalar
        grad_log = a * (grad_a - mean)

        # Adam moments
        m = beta1 * m + (1.0 - beta1) * grad_log
        v = beta2 * v + (1.0 - beta2) * (grad_log ** 2)

        # bias correction
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)

        # learning‑rate schedule
        eta = eta0 / math.sqrt(t)

        # update log_a
        log_a -= eta * m_hat / (np.sqrt(v_hat) + eps)

        # Evaluate periodically
        if t % eval_every == 0:
            # recompute a after the update
            a = np.exp(log_a - np.max(log_a))
            a /= a.sum()
            cur = _objective(a)
            if cur < best_score - 1e-14:
                best_score = cur
                best_seq = a.copy()
                print(f"[{time.time():.1f}] Adam it={t} new best {best_score:.6f}")

    # final projection (just to be safe)
    a = np.exp(log_a - np.max(log_a))
    a /= a.sum()
    return a, best_score


# ----------------------------------------------------------------------
# Simulated‑annealing style pairwise mass transfers
# ----------------------------------------------------------------------
def _pairwise_annealing(
    a: np.ndarray,
    deadline: float,
    best_seq: np.ndarray,
    best_score: float,
    rng: np.random.Generator,
) -> tuple[np.ndarray, float]:
    """
    Randomly transfer a fraction of mass between two bins.
    Improvements are always accepted; deteriorations are accepted with
    a temperature‑dependent probability that linearly cools to zero.
    """
    n = a.shape[0]
    start_time = time.time()
    total_time = deadline - start_time
    while time.time() < deadline:
        # pick a source bin (must have some mass)
        src_candidates = np.where(a > 1e-12)[0]
        if src_candidates.size == 0:
            break
        i = rng.choice(src_candidates)

        # pick a distinct target bin uniformly
        j = rng.integers(n - 1)
        if j >= i:
            j += 1

        # transfer a random fraction of the source mass (up to 40 %)
        delta = a[i] * rng.random() * 0.40
        if delta < 1e-16:
            continue

        new_a = a.copy()
        new_a[i] -= delta
        new_a[j] += delta

        cur = _objective(new_a)

        # temperature decays linearly from 1% of the current best score
        elapsed = time.time() - start_time
        temperature = max(0.0, 1.0 - elapsed / total_time) * best_score * 0.01

        accept = False
        if cur < best_score - 1e-14:
            accept = True
        elif temperature > 0.0:
            prob = math.exp(-(cur - best_score) / temperature)
            if rng.random() < prob:
                accept = True

        if accept:
            a = new_a
            if cur < best_score - 1e-14:
                best_score = cur
                best_seq = a.copy()
                print(
                    f"[{time.time():.1f}] anneal i={i}→j={j} δ={delta:.3e} new best {best_score:.6f}"
                )
    return best_seq, best_score


# ----------------------------------------------------------------------
# Upsampling (linear interpolation while preserving the simplex)
# ----------------------------------------------------------------------
def _upsample(a: np.ndarray, new_n: int) -> np.ndarray:
    """Interpolate a probability vector to a larger length."""
    old_n = a.shape[0]
    if new_n == old_n:
        return a.copy()
    x_old = (np.arange(old_n, dtype=np.float64) + 0.5) / old_n
    x_new = (np.arange(new_n, dtype=np.float64) + 0.5) / new_n
    a_new = np.interp(x_new, x_old, a, left=0.0, right=0.0)
    a_new = np.maximum(a_new, 0.0)
    s = a_new.sum()
    if s == 0.0:
        a_new[:] = 1.0 / new_n
    else:
        a_new /= s
    return a_new
