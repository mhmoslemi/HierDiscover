def _smooth_gradient_descent(a0, time_limit, best_seq, best_score):
    a = a0.copy()
    n = len(a)
    t = 0
    alpha = 10.0
    while time.time() < time_limit:
        # maybe schedule: increase alpha after some iterations.
        conv = _fft_self_convolve(a)
        max_val = conv.max()
        cur_obj = 2.0 * n * max_val
        if cur_obj < best_score - 1e-12:
            best_score = cur_obj
            best_seq = a.copy()
            print(f"[{time.time():.1f}] new best after smooth grad {best_score:.6f}")
        # compute smoothed weights
        # use vector shift using alpha and conv: exp(alpha*(conv - max_val)) stable
        exps = np.exp(alpha * (conv - max_val))
        w = exps / exps.sum()
        # gradient
        grad = _grad_from_weights(a, w)
        # update step size
        t += 1
        eta = 0.1 / math.sqrt(t)  # decays
        # additive gradient step
        a = a - eta * grad
        # ensure non-negative
        a = np.maximum(a, 0.0)
        # project onto simplex
        s = a.sum()
        if s <= 0:
            a = np.full(n, 1.0 / n)
        else:
            a /= s
        # Periodically increase alpha
        if t % 200 == 0:
            alpha = min(500.0, alpha * 1.05)  # slowly increase
        if t % 500 == 0 and time.time() >= time_limit - 5:
            break
    return best_seq, best_score

Search for a non‑negative step function that minimises ``evaluate_sequence``.

The algorithm comprises three stages:

1️⃣  **Exploratory scan** – a wide collection of analytic families
    (uniform, Beta / edge‑weighted, exponential‑edge, sine, cosine,
    quadratic, arcsine, random Dirichlet / sparse) is evaluated on several

2️⃣  **Smooth‑max mirror descent** – each retained seed is refined with a
    differentiable surrogate of the L∞ objective:
    *soft‑max* of the convolution entries.  The gradient is obtained
    efficiently via FFT‑based correlation, the update is an
    exponentiated‑gradient step and the simplex constraint is enforced by

    candidate from the previous stage is sharpened with the exact

# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x ≥ 0, Σx = s}."""
    if s < 0:
        raise ValueError("Simplex radius must be non‑negative")
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.where(u - cssv / (np.arange(1, len(v) + 1)) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    return np.maximum(v - theta, 0.0)


# ----------------------------------------------------------------------
def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Full linear self‑convolution a*a using a real FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1) - 1).bit_length()
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[: 2 * n - 1]


# ----------------------------------------------------------------------
def _objective(a: np.ndarray) -> float:
    """Exact objective 2·n·max(conv(a,a)) for a vector with Σa = 1."""
    n = a.shape[0]
    b = _fft_self_convolve(a)
    return float(2.0 * n * b.max())

# Analytic families ----------------------------------------------------
def _distribution_uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _distribution_arcsine(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    vals /= vals.sum()
    return vals


def _distribution_quadratic(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    x = (2.0 * i) / (n - 1) - 1.0
    w = 1.0 - x * x
    w = np.clip(w, 0.0, None)
    w /= w.sum()
    return w


def _distribution_sine(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64) + 0.5
    w = np.sin(np.pi * i / n)
    w /= w.sum()
    return w


def _distribution_cosine(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64) + 0.5
    w = 1.0 - np.cos(np.pi * i / n)
    w /= w.sum()
    return w


def _distribution_edge_weighted(n: int, p: float) -> np.ndarray:
    """w_i ∝ (i+½)^{‑p}·(n‑i‑½)^{‑p} , i=0…n‑1."""
    i = np.arange(n, dtype=np.float64) + 0.5
    j = (n - np.arange(n, dtype=np.float64) - 0.5)
    w = (i * j) ** (-p)
    w_sum = w.sum()
    if w_sum == 0.0:
        return np.full(n, 1.0 / n, dtype=np.float64)
    w /= w_sum
    return w


def _distribution_exponential_edge(n: int, c: float) -> np.ndarray:
    """w_i ∝ exp(‑c·i) + exp(‑c·(n‑1‑i)),   c>0."""
    i = np.arange(n, dtype=np.float64)
    w = np.exp(-c * i) + np.exp(-c * (n - 1 - i))
    w /= w.sum()
    return w


def _distribution_beta_asymmetric(n: int, alpha: float, beta: float) -> np.ndarray:
    """Beta density on the discrete grid x_i = (i+½)/n."""
    i = np.arange(n, dtype=np.float64) + 0.5
    x = i / n
    # avoid zero powers
    w = np.power(x, alpha - 1.0) * np.power(1.0 - x, beta - 1.0)
    w = np.nan_to_num(w, nan=0.0, posinf=0.0, neginf=0.0)
    s = w.sum()
    if s == 0.0:
        return np.full(n, 1.0 / n, dtype=np.float64)
    w /= s
    return w


def _distribution_random_sparse(
    n: int, rng: np.random.Generator, density: float = 0.01
) -> np.ndarray:
    """Sparse random vector with a prescribed density of non‑zeros."""
    k = max(2, int(n * density))
    idx = rng.choice(n, size=k, replace=False)
    vals = np.zeros(n, dtype=np.float64)
    vals[idx] = rng.random(k)
    s = vals.sum()
    if s == 0.0:
        vals[0] = 1.0
        s = 1.0
    vals /= s
    return vals


def _mix_with_uniform(base: np.ndarray, lam: float) -> np.ndarray:
    """Convex combination λ·uniform + (1‑λ)·base."""
    n = base.shape[0]
    uniform = np.full(n, 1.0 / n, dtype=np.float64)
    return lam * uniform + (1.0 - lam) * base

# Gradient utilities ---------------------------------------------------
def _grad_from_weights(a: np.ndarray, w: np.ndarray) -> np.ndarray:
    """
    Gradient of  Σ_k w_k·conv_k (soft‑max surrogate) w.r.t. a.
    Implements   grad = 2·B·a   where B_{ij}=w_{i+j}.
    Computed via FFT‑based correlation for speed.
    """
    n = a.shape[0]
    # reverse w for correlation
    w_rev = w[::-1]
    size = 1 << ((3 * n - 2) - 1).bit_length()   # >= len(w_rev)+len(a)-1
    W = np.fft.rfft(w_rev, n=size)
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(W * A, n=size)
    # slice the relevant n entries: indices n‑1 … 2n‑2 (inclusive)
    t = conv[n - 1 : 2 * n - 1]
    return 2.0 * t

# Smooth‑max mirror descent --------------------------------------------
def _smooth_max_descent(
    a0: np.ndarray,
    time_limit: float,
    best_seq: np.ndarray,
    best_score: float,
    init_alpha: float = 12.0,
) -> tuple[np.ndarray, float]:
    """Exponentiated‑gradient descent on a soft‑max approximation of max(conv)."""
    a = a0.copy()
    n = a.shape[0]
    t = 0
    alpha = init_alpha
    while time.time() < time_limit:
        t += 1
        conv = _fft_self_convolve(a)
        max_val = conv.max()
        cur_obj = 2.0 * n * max_val
        if cur_obj < best_score - 1e-12:
            best_score = cur_obj
            best_seq = a.copy()
            print(
                f"[{time.time():.1f}] smooth it={t} new best {best_score:.6f}"
            )

        # soft‑max weights (stabilised)
        exp_terms = np.exp(alpha * (conv - max_val))
        w = exp_terms / exp_terms.sum()

        # gradient of the surrogate
        grad = _grad_from_weights(a, w)

        # step size – decays like 1/√t
        eta = 0.6 / math.sqrt(t)

        # multiplicative (exponentiated) update + simplex normalisation
        a = a * np.exp(-eta * grad)
        s = a.sum()
        if s <= 0.0:
            a = np.full(n, 1.0 / n, dtype=np.float64)
        else:
            a /= s

        # slowly tighten the soft‑max (increase temperature)
        if t % 200 == 0:
            alpha = min(500.0, alpha * 1.06)

    return best_seq, best_score

def _subgradient_descent_exact(
    a0: np.ndarray,
    time_limit: float,
    best_seq: np.ndarray,
    best_score: float,
    lr0: float = 0.9,
    eval_every: int = 80,
) -> tuple[np.ndarray, float]:
    """EG descent on the true max‑convolution objective."""
    a = a0.copy()
    n = a.shape[0]
    t = 0
    while time.time() < time_limit:
        t += 1
        conv = _fft_self_convolve(a)
        max_val = conv.max()
        max_idxs = np.where(conv == max_val)[0]

        # exact sub‑gradient (sum over all active indices)
        grad = np.zeros_like(a)
        for idx in max_idxs:
            i_start = max(0, idx - (n - 1))
            i_end = min(n - 1, idx)
            i = np.arange(i_start, i_end + 1, dtype=np.int64)
            j = idx - i
            grad[i] += 2.0 * a[j]

        eta = lr0 / math.sqrt(t)
        a *= np.exp(-eta * grad)
        s = a.sum()
        if s <= 0.0:
            a = np.full(n, 1.0 / n, dtype=np.float64)
        else:
            a /= s

        if t % eval_every == 0:
            cur_obj = 2.0 * n * max_val
            if cur_obj < best_score - 1e-12:
                best_score = cur_obj
                best_seq = a.copy()
                print(
                    f"[{time.time():.1f}] exact it={t} new best {best_score:.6f}"
                )
    return best_seq, best_score

# Greedy pairwise mass transfer ----------------------------------------
def _pairwise_swap(
    a0: np.ndarray,
    time_limit: float,
    best_seq: np.ndarray,
    best_score: float,
    max_trials: int | None = None,
) -> tuple[np.ndarray, float]:
    """Random local moves – move a fraction of mass from i to j if it improves."""
    a = a0.copy()
    n = a.shape[0]
    trials = 0
    while time.time() < time_limit:
        if max_trials is not None and trials >= max_trials:
            break
        trials += 1

        src = np.where(a > 1e-14)[0]
        if src.size == 0:
            break
        i = np.random.choice(src)
        j = np.random.randint(n - 1)
        if j >= i:
            j += 1

        delta = a[i] * np.random.rand() * 0.45
        if delta < 1e-15:
            continue

        new_a = a.copy()
        new_a[i] -= delta
        new_a[j] += delta

        cur = _objective(new_a)
        if cur < best_score - 1e-12:
            a = new_a
            best_score = cur
            best_seq = a.copy()
            print(
                f"[{time.time():.1f}] swap i={i}→j={j} δ={delta:.6g} new best {best_score:.6f}"
            )
    return best_seq, best_score

            if time.time() >= deadline:
                break
            edge = _distribution_edge_weighted(n, float(p))
            for lam in lam_grid:
                if time.time() >= deadline:
                    break
                cand = _mix_with_uniform(edge, lam)
                sc = _objective(cand)
                top_candidates.append((sc, cand.copy()))
                if sc < overall_best_score:
                    overall_best_score, overall_best_seq = sc, cand.copy()
                    print(
                        f"[{time.time():.1f}] len={n} edge p={p:.4g} lam={lam:.2f} score={sc:.6f}"
                    )

        # exponential‑edge family ------------------------------------------------
        for c in c_grid:
            if time.time() >= deadline:
                break
            exp_edge = _distribution_exponential_edge(n, float(c))
            for lam in lam_grid:
                if time.time() >= deadline:
                    break
                cand = _mix_with_uniform(exp_edge, lam)
                sc = _objective(cand)
                top_candidates.append((sc, cand.copy()))
                if sc < overall_best_score:
                    overall_best_score, overall_best_seq = sc, cand.copy()
                    print(
                        f"[{time.time():.1f}] len={n} exp c={c:.4g} lam={lam:.2f} score={sc:.6f}"
                    )

        # Beta families -----------------------------------------------------------
        for alpha, beta in beta_pairs:
            if time.time() >= deadline:
                break
            beta_seq = _distribution_beta_asymmetric(n, alpha, beta)
            sc = _objective(beta_seq)
            top_candidates.append((sc, beta_seq.copy()))
            if sc < overall_best_score:
                overall_best_score, overall_best_seq = sc, beta_seq.copy()
                print(
                    f"[{time.time():.1f}] len={n} Beta(α={alpha:.2g},β={beta:.2g}) score={sc:.6f}"
                )

        # random Dirichlet seeds -------------------------------------------------
        for _ in range(5):
            if time.time() >= deadline:
                break
            alpha = rng.uniform(0.02, 2.5)
            a = rng.dirichlet(np.full(n, alpha))
            sc = _objective(a)
            top_candidates.append((sc, a.copy()))
            if sc < overall_best_score:
                overall_best_score, overall_best_seq = sc, a.copy()
                print(f"[{time.time():.1f}] Dirichlet len={n} score={sc:.6f}")

        # sparse random seeds ----------------------------------------------------
        for _ in range(5):
            if time.time() >= deadline:
                break
            a = _distribution_random_sparse(n, rng, density=0.008)
            sc = _objective(a)
            top_candidates.append((sc, a.copy()))
            if sc < overall_best_score:
                overall_best_score, overall_best_seq = sc, a.copy()
                print(f"[{time.time():.1f}] sparse len={n} score={sc:.6f}")

        # keep only a limited number of best seeds (memory & time bound)
        top_candidates.sort(key=lambda x: x[0])
        top_candidates = top_candidates[:40]

    # ------------------------------------------------------------------
    # 2. Randomised fine‑grained search (exploit remaining budget)
    # ------------------------------------------------------------------
    while time.time() < deadline - 30.0:
        n = int(rng.choice([1000, 2000, 3000, 5000, 8000, 12000, 16000]))
        # pick a random family and random parameters
        choice = rng.integers(0, 5)
        if choice == 0:
            cand = _distribution_uniform(n)
        elif choice == 1:
            p = 10 ** rng.uniform(-2, 1)          # p∈[0.01,10]
            edge = _distribution_edge_weighted(n, p)
            lam = rng.random()
            cand = _mix_with_uniform(edge, lam)
        elif choice == 2:
            c = 10 ** rng.uniform(-3, 0)          # c∈[0.001,1]
            exp_edge = _distribution_exponential_edge(n, c)
            lam = rng.random()
            cand = _mix_with_uniform(exp_edge, lam)
        elif choice == 3:
            alpha = rng.uniform(0.2, 2.0)
            beta = rng.uniform(0.2, 2.0)
            cand = _distribution_beta_asymmetric(n, alpha, beta)
        elif choice == 4:
            a = rng.dirichlet(np.full(n, rng.uniform(0.01, 2.0)))
            cand = a
        else:
            cand = _distribution_random_sparse(n, rng, density=0.01)

        sc = _objective(cand)
        top_candidates.append((sc, cand.copy()))
        if sc < overall_best_score:
            overall_best_score, overall_best_seq = sc, cand.copy()
            print(f"[{time.time():.1f}] random fine‑search len={n} score={sc:.6f}")

        # prune
        top_candidates.sort(key=lambda x: x[0])
        top_candidates = top_candidates[:40]

    # ------------------------------------------------------------------
    # 3. Include previous construction if present (height_sequence_1)
    # ------------------------------------------------------------------
    try:
        prev = np.asarray(height_sequence_1, dtype=np.float64)  # type: ignore[name-defined]
        if prev.ndim == 1 and prev.size > 0:
            prev = np.maximum(prev, 0.0)
            if prev.sum() == 0.0:
                prev = _distribution_uniform(prev.size)
            else:
                prev = prev / prev.sum()
            sc = _objective(prev)
            if sc < overall_best_score:
                overall_best_score, overall_best_seq = sc, prev.copy()
                print(f"[{time.time():.1f}] previous construction score={sc:.6f}")
            top_candidates.append((sc, prev.copy()))
    except Exception:
        pass

    # ------------------------------------------------------------------
    # 4. Fallback (should never happen)
    # ------------------------------------------------------------------
    if overall_best_seq is None:
        n = 4096
        overall_best_seq = _distribution_uniform(n)
        overall_best_score = _objective(overall_best_seq)
        print(
            f"[{time.time():.1f}] fallback uniform length={n} score={overall_best_score:.6f}"
        )

    # ------------------------------------------------------------------
    # 5. Smooth‑max mirror‑descent on the most promising seeds
    # ------------------------------------------------------------------
    remaining = deadline - time.time()
    if remaining > 40.0 and top_candidates:
        # choose up to six best seeds for deeper optimisation
        seeds = [seq for _, seq in sorted(top_candidates, key=lambda x: x[0])[:6]]
        per_seed = remaining * 0.5 / max(1, len(seeds))
        print(f"[{time.time():.1f}] start smooth‑max descent on {len(seeds)} seeds")
        for idx, seed_seq in enumerate(seeds):
            if time.time() >= deadline:
                break
            sg_deadline = time.time() + per_seed
            a_opt, sc_opt = _smooth_max_descent(
                seed_seq,
                sg_deadline,
                best_seq=overall_best_seq,
                best_score=overall_best_score,
                init_alpha=15.0,
            )
            if sc_opt < overall_best_score - 1e-12:
                overall_best_score, overall_best_seq = sc_opt, a_opt.copy()
                print(
                    f"[{time.time():.1f}] after smooth seed {idx} new global best {overall_best_score:.6f}"
                )

    # ------------------------------------------------------------------
    # 6. Exact sub‑gradient refinement (shorter budget)
    # ------------------------------------------------------------------
    remaining = deadline - time.time()
    if remaining > 30.0:
        sg_deadline = time.time() + remaining * 0.4
        a_opt, sc_opt = _subgradient_descent_exact(
            overall_best_seq.copy(),
            sg_deadline,
            best_seq=overall_best_seq,
            best_score=overall_best_score,
            lr0=0.9,
            eval_every=120,
        )
        if sc_opt < overall_best_score - 1e-12:
            overall_best_score, overall_best_seq = sc_opt, a_opt.copy()
            print(
                f"[{time.time():.1f}] after exact sub‑grad new best {overall_best_score:.6f}"
            )

    # ------------------------------------------------------------------
    # 7. Final greedy pairwise polishing
    # ------------------------------------------------------------------
    if time.time() < deadline:
        print(f"[{time.time():.1f}] final greedy pairwise polishing")
        a_opt, sc_opt = _pairwise_swap(
            overall_best_seq.copy(),
            deadline,
            best_seq=overall_best_seq,
            best_score=overall_best_score,
        )
        if sc_opt < overall_best_score - 1e-12:
            overall_best_score, overall_best_seq = sc_opt, a_opt.copy()
            print(f"[{time.time():.1f}] after final swaps new best {overall_best_score:.6f}")

    # ------------------------------------------------------------------
    print(
        f"Final best objective (2·n·max(conv)) = {overall_best_score:.6f} (length={len(overall_best_seq)})"
    )
    # Return plain Python list – evaluate_sequence will re‑normalise internally.
    return overall_best_seq.tolist()
