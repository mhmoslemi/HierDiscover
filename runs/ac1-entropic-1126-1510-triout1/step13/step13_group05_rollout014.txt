"""
Propose a step‑function (non‑negative vector) that minimises
``evaluate_sequence``.  The algorithm is a three‑phase search:

1. **Seed generation** – a small collection of analytic families
   (uniform, edge‑weighted, symmetric beta) together with random
   Dirichlet / sparse vectors.  All seeds are normalised to have unit
   ℓ₁‑norm.

2. **Smoothed exponential‑gradient descent** – for each seed we minimise
   a smooth approximation of the max‑convolution
   `f(a)=max_k (a*a)_k`.  The smooth objective is the log‑sum‑exp
   `log∑exp(β·conv)` with a temperature parameter β that is increased
   during optimisation.  The exact gradient of this surrogate is
   obtained via FFT‐based correlation, and a mirror‑descent update
   `a ← a·exp(−η·∇)` is applied, followed by projection onto the simplex.
   The best vector found (according to the true objective
   `2·n·max(conv)`) is retained.

3. **Greedy polishing** – a cheap hill‑climbing phase that moves a
   random fraction of mass from one coordinate to another only when the
   exact objective strictly improves.

The routine respects the supplied time budget (default 1000 s) and
returns a plain list of non‑negative floats (already normalised).
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """ Euclidean projection onto the simplex {x≥0, Σx=s}. """
    if s < 0:
        raise ValueError("Simplex radius must be non‑negative")
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.where(u - cssv / (np.arange(1, len(v) + 1)) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    return np.maximum(v - theta, 0.0)


# ----------------------------------------------------------------------
def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """ Full self‑convolution a*a using a real FFT. """
    n = a.shape[0]
    size = 1 << ((2 * n - 1) - 1).bit_length()   # next power of two
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[: 2 * n - 1]


# ----------------------------------------------------------------------
def _objective(a: np.ndarray) -> float:
    """ Exact objective 2·n·max(conv) for a vector with Σa = 1. """
    n = a.shape[0]
    conv = _fft_self_convolve(a)
    return float(2.0 * n * conv.max())


# ----------------------------------------------------------------------
# Analytic families ----------------------------------------------------
def _distribution_uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _distribution_edge_weighted(n: int, p: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64) + 0.5
    j = (n - np.arange(n, dtype=np.float64) - 0.5)
    w = (i * j) ** (-p)
    w = np.clip(w, 0.0, None)
    total = w.sum()
    if total > 0.0:
        w /= total
    else:
        w = np.full(n, 1.0 / n)
    return w


def _distribution_beta_sym(n: int, p: float) -> np.ndarray:
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    w = np.exp((p - 1) * np.log(x) + (p - 1) * np.log(1.0 - x))
    if w.sum() == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _distribution_random_sparse(
    n: int, rng: np.random.Generator, density: float = 0.01
) -> np.ndarray:
    """Sparse random vector with a prescribed density of non‑zeros."""
    k = max(2, int(n * density))
    idx = rng.choice(n, size=k, replace=False)
    vals = np.zeros(n, dtype=np.float64)
    vals[idx] = rng.random(k)
    if vals.sum() == 0.0:
        vals[0] = 1.0
    vals /= vals.sum()
    return vals


# ----------------------------------------------------------------------
# Gradient of the smooth max–convolution ---------------------------------
def _gradient_given_weights(a: np.ndarray, w: np.ndarray) -> np.ndarray:
    """
    Gradient of the smoothed max‑convolution w.r.t. a.
    w must be a probability vector of length 2·n‑1 (softmax weights).
    Returns a vector of length n.
    """
    n = a.shape[0]
    # linear convolution of a with reversed w
    size = 1 << ((3 * n - 2) - 1).bit_length()   # >= 3n‑2
    A = np.fft.rfft(a, n=size)
    W_rev = np.fft.rfft(w[::-1], n=size)
    conv_full = np.fft.irfft(A * W_rev, n=size)
    # slice indices n‑1 … 2n‑2 (inclusive) → length n
    slice_vals = conv_full[n - 1 : 2 * n - 1]
    # gradient = 2 * reverse(slice_vals)
    return 2.0 * slice_vals[::-1]


# ----------------------------------------------------------------------
# Smoothed exponential‑gradient descent ----------------------------------
def _smooth_exp_gradient_descent(
    a0: np.ndarray,
    time_limit: float,
    best_seq: np.ndarray,
    best_score: float,
    eta0: float = 0.6,
    beta0: float = 1.0,
    report_every: int = 400,
) -> tuple[np.ndarray, float]:
    """
    Mirror‑descent on the log‑sum‑exp surrogate for the max‑convolution.
    The temperature β is increased geometrically; the update is
    multiplicative (exponentiated gradient) followed by ℓ₁‑normalisation.
    """
    a = a0.copy()
    n = a.shape[0]
    size = 1 << ((3 * n - 2) - 1).bit_length()   # static FFT size
    t = 0
    beta = beta0
    max_iter = 500_000

    while time.time() < time_limit and t < max_iter:
        t += 1

        # ----- self‑convolution (for the softmax) -----
        A = np.fft.rfft(a, n=size)
        conv = np.fft.irfft(A * A, n=size)[: 2 * n - 1]
        max_val = conv.max()

        # ----- softmax weights -----
        # numeric stability: subtract max_val before exponentiation
        exp_arg = beta * (conv - max_val)
        w = np.exp(exp_arg)
        w_sum = w.sum()
        if w_sum == 0.0:
            w = np.full_like(w, 1.0 / w.shape[0])
        else:
            w /= w_sum

        # ----- gradient of the smooth objective -----
        grad = _gradient_given_weights(a, w)

        # ----- mirror‑descent update (exponentiated) -----
        eta = eta0 / math.sqrt(t)
        a = a * np.exp(-eta * grad)

        # projection onto the simplex (ℓ₁‑normalisation)
        s = a.sum()
        if s > 0.0:
            a = a / s
        else:
            a = np.full(n, 1.0 / n, dtype=np.float64)

        # ----- temperature schedule -----
        if t % 2000 == 0:
            beta = min(beta * 1.2, 200.0)   # avoid overflow

        # ----- occasional exact evaluation -----
        if t % report_every == 0:
            cur = _objective(a)
            if cur < best_score - 1e-14:
                best_score = cur
                best_seq = a.copy()
                print(
                    f"[{time.time():.1f}] smooth‑gd it={t} β={beta:.2f} new best {best_score:.6f}"
                )
    return best_seq, best_score


# ----------------------------------------------------------------------
# Greedy pairwise polishing ---------------------------------------------
def _pairwise_swap(
    a0: np.ndarray,
    time_limit: float,
    best_seq: np.ndarray,
    best_score: float,
    max_trials: int | None = None,
) -> tuple[np.ndarray, float]:
    """
    Randomly move a fraction of mass from one index to another.
    Accept only strict improvements of the exact objective.
    """
    a = a0.copy()
    n = a.shape[0]
    trials = 0
    while time.time() < time_limit:
        if max_trials is not None and trials >= max_trials:
            break
        trials += 1

        src = np.where(a > 1e-14)[0]
        if src.size == 0:
            break
        i = np.random.choice(src)
        j = np.random.randint(n - 1)
        if j >= i:
            j += 1

        delta = a[i] * np.random.rand() * 0.45   # up to 45 % of source mass
        if delta < 1e-15:
            continue

        new_a = a.copy()
        new_a[i] -= delta
        new_a[j] += delta

        cur = _objective(new_a)
        if cur < best_score - 1e-14:
            a = new_a
            best_score = cur
            best_seq = a.copy()
            print(
                f"[{time.time():.1f}] swap i={i}→j={j} δ={delta:.6g} new best {best_score:.6f}"
            )
    return best_seq, best_score
