def _subgradient_descent_adagrad(a, deadline, best_seq, best_score, lr0=0.8, max_active=50, eps=1e-12, eval_interval=200, patience=2000):

   a = a.copy()
   n = a.shape[0]
   G = np.zeros_like(a) + eps
   no_improve = 0
   t = 0
   rng = np.random.default_rng()
   ind_grid = np.arange(n)

   while time.time() < deadline and t < 500_000:
       t += 1

       conv = _fft_self_convolve(a)
       # Determine active set via top k
       if max_active < conv.size:
           top_idx = np.argpartition(conv, -max_active)[-max_active:]
       else:
           top_idx = np.arange(conv.size)

       grad = np.zeros_like(a)
       for k in top_idx:
           shift = k - ind_grid
           mask = (shift >= 0) & (shift < n)
           grad[mask] += 2.0 * a[shift[mask]]

       G += grad**2
       step = lr0 / np.sqrt(G)

       # exponentiated update
       a = a * np.exp(- step * grad)

       # projection onto simplex
       s = a.sum()
       if s <= 0:
           a = np.full(n, 1.0/n)
       else:
           a /= s

       if t % eval_interval == 0:
           cur = _objective(a)
           if cur + 1e-14 < best_score:
               best_score = cur
               best_seq = a.copy()
               no_improve = 0
               # print message
               print(f"[{time.time():.1f}] descent iteration {t} new best {best_score:.6f}")
           else:
               no_improve += 1

       if no_improve >= patience:
           # jitter
           jitter = rng.exponential(scale=0.005, size=n)
           a = a + jitter
           a = np.maximum(a, 0.0)
           a /= a.sum()
           no_improve = 0

   return best_seq, best_score

"""
Optimise a non‑negative step function for the benchmark ``evaluate_sequence``.

Algorithm
---------

1. **Seed pool** – generate a diverse set of candidates of moderate length
   (≈ 2000–4000).  Families include uniform, arcsine, symmetric / asymmetric
   Beta, edge‑weighted power laws, sine‑weighted, triangular and cosine‑shaped
   distributions together with random Dirichlet / sparse vectors.  Each candidate
   is projected onto the simplex (sum = 1) and ranked by the exact objective
   ``2·n·max(a*a)``.

2. **AdaGrad mirror descent** – for the best few seeds we run a projected
   exponentiated‑gradient (entropy mirror) descent with an AdaGrad‑style
   per‑coordinate step‑size.  At every iteration only the *k* largest
   convolution entries are used (``max_active``).  The update is
   ``a ← a·exp(‑η·grad/√G)`` where ``G`` accumulates squared sub‑gradients.
   Simplex projection guarantees feasibility.  An occasional jitter
   (exponential noise) helps escape plateaus.

3. **Hierarchical upsampling** – the current global best is linearly
   interpolated to a larger length (≈ 2×) and refined again with the same
   AdaGrad descent.  Larger vectors have a strictly lower theoretical lower
   bound, so this step often yields a noticeable drop.

4. **Local cleanup** – a cheap greedy pairwise mass‑transfer sweep followed
   by a simulated‑annealing style Metropolis search removes any remaining
   spikes.

The whole routine respects the supplied time budget (default 1000 s) and
returns a plain ``list[float]`` ready for ``evaluate_sequence``.
"""

import time
import math
import numpy as np
from numpy.fft import rfft, irfft

# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x≥0, sum(x)=s}."""
    if s < 0:
        raise ValueError("simplex radius must be non‑negative")
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho_arr = np.where(u - cssv / (np.arange(1, len(v) + 1)) > 0)[0]
    if rho_arr.size == 0:
        theta = 0.0
    else:
        rho = rho_arr[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a (real FFT, length 2·n‑1)."""
    n = a.shape[0]
    need = 2 * n - 1
    size = 1 << ((need - 1).bit_length())   # next power of two
    A = rfft(a, n=size)
    conv = irfft(A * A, n=size)
    return conv[:need]


# ----------------------------------------------------------------------
def _objective(a: np.ndarray) -> float:
    """Exact objective 2·n·max(conv(a,a)) for a with sum(a)=1."""
    n = a.shape[0]
    b = _fft_self_convolve(a)
    return 2.0 * n * b.max()

# Analytic families ----------------------------------------------------
def _distribution_uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _distribution_arcsine(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    vals /= vals.sum()
    return vals


def _distribution_beta_sym(n: int, p: float) -> np.ndarray:
    """Symmetric Beta(p,p) on the regular grid."""
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    w = np.exp((p - 1) * np.log(x) + (p - 1) * np.log(1.0 - x))
    if w.sum() == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _distribution_beta_asym(n: int, a: float, b: float) -> np.ndarray:
    """Asymmetric Beta(a,b) on the regular grid."""
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    logw = (a - 1) * np.log(x) + (b - 1) * np.log(1.0 - x)
    w = np.exp(logw - np.max(logw))
    if w.sum() == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _distribution_edge_weighted(n: int, p: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64) + 0.5
    j = (n - np.arange(n, dtype=np.float64) - 0.5)
    w = (i * j) ** (-p)
    if w.sum() == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _distribution_edge_inversepow(n: int, p: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64) + 0.5
    j = (n - np.arange(n, dtype=np.float64) - 0.5)
    w = i ** (-p) + j ** (-p)
    if w.sum() == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _distribution_sine_weighted(n: int, p: float) -> np.ndarray:
    theta = np.pi * (np.arange(n, dtype=np.float64) + 0.5) / n
    w = np.sin(theta) ** p
    if w.sum() == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _distribution_triangular(n: int) -> np.ndarray:
    """Symmetric triangle – linearly rising to the centre then falling."""
    i = np.arange(n, dtype=np.float64) + 1.0
    w = np.minimum(i, n - i + 1.0)
    w /= w.sum()
    return w


def _distribution_cosine_weighted(n: int, p: float) -> np.ndarray:
    """Raised cosine: (1‑cos(2πx))^p on the grid."""
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    w = (1.0 - np.cos(2.0 * np.pi * x)) ** p
    if w.sum() == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _distribution_random_sparse(
    n: int, rng: np.random.Generator, density: float = 0.006
) -> np.ndarray:
    """Very few non‑zero entries, normalised to a simplex."""
    k = max(2, int(n * density))
    idx = rng.choice(n, size=k, replace=False)
    vals = np.zeros(n, dtype=np.float64)
    vals[idx] = rng.random(k)
    if vals.sum() == 0.0:
        vals[0] = 1.0
    vals /= vals.sum()
    return vals


def _distribution_dirichlet(n: int, rng: np.random.Generator) -> np.ndarray:
    """Dirichlet with a random concentration."""
    alpha = rng.uniform(0.01, 2.0)
    a = rng.dirichlet(np.full(n, alpha))
    return a.astype(np.float64)


def _mix_with_uniform(base: np.ndarray, lam: float) -> np.ndarray:
    """lam·U + (1‑lam)·base, U uniform."""
    n = base.shape[0]
    uniform = np.full(n, 1.0 / n, dtype=np.float64)
    return lam * uniform + (1.0 - lam) * base

def _subgradient_descent_adagrad(
    a: np.ndarray,
    deadline: float,
    best_seq: np.ndarray,
    best_score: float,
    lr0: float = 0.8,
    max_active: int = 50,
    eps: float = 1e-12,
    eval_interval: int = 200,
    patience: int = 1500,
) -> tuple[np.ndarray, float]:
    """
    Projected exponentiated‑gradient descent with AdaGrad scaling.
    Only the ``max_active`` largest convolution entries contribute to the
    sub‑gradient.  The routine periodically projects onto the simplex and
    keeps the globally best point.
    """
    a = a.copy()
    n = a.shape[0]
    G = np.full_like(a, eps)               # accumulated squared gradients
    no_improve = 0
    t = 0
    rng = np.random.default_rng()
    ind_grid = np.arange(n, dtype=np.int32)

    while time.time() < deadline and t < 500_000:
        t += 1

        conv = _fft_self_convolve(a)
        # pick the largest entries – always at least one (the centre)
        if max_active < conv.size:
            top_idx = np.argpartition(conv, -max_active)[-max_active:]
        else:
            top_idx = np.arange(conv.size)

        grad = np.zeros_like(a)
        for k in top_idx:
            shift = k - ind_grid
            mask = (shift >= 0) & (shift < n)
            grad[mask] += 2.0 * a[shift[mask]]

        G += grad * grad
        step = lr0 / np.sqrt(G)                # per‑coordinate AdaGrad step

        # exponentiated‑gradient update (keeps positivity)
        a = a * np.exp(-step * grad)

        # projection onto the simplex
        s = a.sum()
        if s <= 0.0:
            a = np.full(n, 1.0 / n, dtype=np.float64)
        else:
            a /= s

        if t % eval_interval == 0:
            cur = _objective(a)
            if cur + 1e-14 < best_score:
                best_score = cur
                best_seq = a.copy()
                no_improve = 0
                print(f"[{time.time():.1f}] it={t:,} new best {best_score:.6f}")
            else:
                no_improve += 1

        if no_improve >= patience:
            # small random jitter to escape flat regions
            jitter = rng.exponential(scale=0.005, size=n)
            a = a + jitter
            a = np.maximum(a, 0.0)
            a /= a.sum()
            no_improve = 0

    return best_seq, best_score


# ----------------------------------------------------------------------
def _pairwise_swap_local(
    a: np.ndarray,
    deadline: float,
    best_seq: np.ndarray,
    best_score: float,
    max_trials: int | None = None,
) -> tuple[np.ndarray, float]:
    """
    Deterministic greedy mass transfer: move a random fraction of mass from
    one coordinate to another if it improves the objective.
    """
    n = a.shape[0]
    trials = 0
    rng = np.random.default_rng()
    while time.time() < deadline:
        if max_trials is not None and trials >= max_trials:
            break
        trials += 1
        src = np.where(a > 1e-12)[0]
        if src.size == 0:
            break
        i = rng.choice(src)
        j = rng.integers(n - 1)
        if j >= i:
            j += 1
        delta = a[i] * rng.random() * 0.33
        if delta < 1e-14:
            continue
        new_a = a.copy()
        new_a[i] -= delta
        new_a[j] += delta
        cur = _objective(new_a)
        if cur + 1e-14 < best_score:
            a = new_a
            best_score = cur
            best_seq = a.copy()
            print(f"[{time.time():.1f}] swap {i}->{j} δ={delta:.3e} new best {best_score:.6f}")
    return best_seq, best_score


# ----------------------------------------------------------------------
def _random_local_search(
    a: np.ndarray,
    deadline: float,
    best_seq: np.ndarray,
    best_score: float,
    init_temp: float = 0.008,
    decay: float = 0.9995,
) -> tuple[np.ndarray, float]:
    """
    Simple Metropolis–Hastings style annealing that transfers a random amount
    of mass between two positions and accepts worsening moves with a
    temperature that decays exponentially.
    """
    n = a.shape[0]
    rng = np.random.default_rng()
    T = init_temp
    while time.time() < deadline:
        i = rng.integers(n)
        j = rng.integers(n - 1)
        if j >= i:
            j += 1
        delta = a[i] * rng.random() * 0.28
        if delta < 1e-14:
            continue
        new_a = a.copy()
        new_a[i] -= delta
        new_a[j] += delta
        cur = _objective(new_a)
        if cur + 1e-14 < best_score:
            a = new_a
            best_score = cur
            best_seq = a.copy()
            print(f"[{time.time():.1f}] SA improve {i}->{j} Δ={delta:.3e} best {best_score:.6f}")
        else:
            if rng.random() < math.exp(-(cur - best_score) / max(T, 1e-12)):
                a = new_a
        T *= decay
    return best_seq, best_score


# ----------------------------------------------------------------------
def _upsample(seq: np.ndarray, new_len: int) -> np.ndarray:
    """Linear interpolation of a simplex‑vector to a larger length."""
    old_len = seq.shape[0]
    if new_len == old_len:
        return seq.copy()
    x_old = np.linspace(0.0, 1.0, old_len)
    x_new = np.linspace(0.0, 1.0, new_len)
    new_seq = np.interp(x_new, x_old, seq)
    new_seq = np.maximum(new_seq, 0.0)
    new_seq /= new_seq.sum()
    return new_seq


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: int = 1000) -> list[float]:
    """
    Return a list of non‑negative heights (sum≈1) that (ideally) beats the
    current benchmark.  The whole search respects the time budget.
    """
    rng = np.random.default_rng(seed)
    start = time.time()
    deadline = start + budget_s - 2.0  # safety margin

    overall_best_seq: np.ndarray | None = None
    overall_best_score = float("inf")
    top_candidates: list[tuple[float, np.ndarray]] = []

    # ------------------------------------------------------------------
    # 1. Seed generation (moderate lengths)
    lengths = [2000, 3000, 4000]
    p_grid = np.concatenate(
        [
            np.logspace(-3, -1, 7),   # 0.001 … 0.1
            np.logspace(-1, 0, 7),   # 0.1 … 1
            np.logspace(0, 0.5, 5),  # 1 … 3.16
        ]
    )
    lam_grid = np.linspace(0.0, 0.9, 6)

    for n in lengths:
        if time.time() >= deadline:
            break
        print(f"[{time.time():.1f}] sampling length {n}")

        # deterministic families (uniform & arcsine)
        for base in (_distribution_uniform(n), _distribution_arcsine(n)):
            sc = _objective(base)
            top_candidates.append((sc, base.copy()))
            if sc < overall_best_score:
                overall_best_score, overall_best_seq = sc, base.copy()
                print(f"[{time.time():.1f}] new global best (uniform/arcsine) {sc:.6f}")

        # parameterised families
        for p in p_grid:
            if time.time() >= deadline:
                break
            beta = _distribution_beta_sym(n, p)
            edge = _distribution_edge_weighted(n, p)
            edge2 = _distribution_edge_inversepow(n, p)
            sine = _distribution_sine_weighted(n, p)
            tri = _distribution_triangular(n)
            cos = _distribution_cosine_weighted(n, p)

            for base in (beta, edge, edge2, sine, tri, cos):
                for lam in lam_grid:
                    if time.time() >= deadline:
                        break
                    cand = _mix_with_uniform(base, lam)
                    sc = _objective(cand)
                    top_candidates.append((sc, cand.copy()))
                    if sc < overall_best_score:
                        overall_best_score, overall_best_seq = sc, cand.copy()
                        print(
                            f"[{time.time():.1f}] new global best len={n} p={p:.3g} lam={lam:.2f} score={sc:.6f}"
                        )
        # asymmetric beta (two‑parameter)
        for _ in range(8):
            if time.time() >= deadline:
                break
            a = rng.uniform(0.001, 3.0)
            b = rng.uniform(0.001, 3.0)
            cand = _distribution_beta_asym(n, a, b)
            sc = _objective(cand)
            top_candidates.append((sc, cand.copy()))
            if sc < overall_best_score:
                overall_best_score, overall_best_seq = sc, cand.copy()
                print(
                    f"[{time.time():.1f}] new global best asym‑beta len={n} a={a:.3g} b={b:.3g} score={sc:.6f}"
                )
        # random Dirichlet seeds
        for _ in range(8):
            if time.time() >= deadline:
                break
            cand = _distribution_dirichlet(n, rng)
            sc = _objective(cand)
            top_candidates.append((sc, cand.copy()))
            if sc < overall_best_score:
                overall_best_score, overall_best_seq = sc, cand.copy()
                print(
                    f"[{time.time():.1f}] new global best Dirichlet len={n} score={sc:.6f}"
                )
        # sparse random seeds
        for _ in range(8):
            if time.time() >= deadline:
                break
            cand = _distribution_random_sparse(n, rng, density=0.006)
            sc = _objective(cand)
            top_candidates.append((sc, cand.copy()))
            if sc < overall_best_score:
                overall_best_score, overall_best_seq = sc, cand.copy()
                print(
                    f"[{time.time():.1f}] new global best sparse len={n} score={sc:.6f}"
                )

        # keep only the very best seeds so far
        top_candidates.sort(key=lambda x: x[0])
        top_candidates = top_candidates[:30]

    # ------------------------------------------------------------------
    # 2. Add a previously discovered construction if present
    try:
        prev = np.asarray(height_sequence_1, dtype=np.float64)  # type: ignore[name-defined]
        if prev.ndim == 1 and prev.size > 0:
            prev = np.maximum(prev, 0.0)
            prev = prev / prev.sum()
            sc = _objective(prev)
            if sc < overall_best_score:
                overall_best_score, overall_best_seq = sc, prev.copy()
                print(f"[{time.time():.1f}] previous best score={sc:.6f}")
            top_candidates.append((sc, prev.copy()))
    except Exception:
        pass

    # ------------------------------------------------------------------
    # 3. Fallback guard
    if overall_best_seq is None:
        n = 3000
        overall_best_seq = _distribution_uniform(n)
        overall_best_score = _objective(overall_best_seq)
        print(f"[{time.time():.1f}] fallback uniform score={overall_best_score:.6f}")

    # ------------------------------------------------------------------
    # 4. AdaGrad mirror‑descent on the most promising seeds
    remaining = deadline - time.time()
    if remaining > 30.0 and top_candidates:
        # take the 10 best seeds; allocate geometrically decreasing time
        seeds = [seq for _, seq in sorted(top_candidates, key=lambda x: x[0])[:10]]
        total_factor = sum(0.5 ** i for i in range(len(seeds)))
        base_time = remaining * 0.6 / total_factor   # keep a fraction for later stages
        for i, seed_seq in enumerate(seeds):
            if time.time() >= deadline:
                break
            sub_time = base_time * (0.5 ** i)
            sub_deadline = time.time() + sub_time
            a_opt, sc_opt = _subgradient_descent_adagrad(
                seed_seq.copy(),
                sub_deadline,
                best_seq=overall_best_seq,
                best_score=overall_best_score,
                lr0=0.9,
                max_active=60,
                eval_interval=150,
                patience=1800,
            )
            if sc_opt < overall_best_score:
                overall_best_score, overall_best_seq = sc_opt, a_opt.copy()
                print(
                    f"[{time.time():.1f}] after descent seed {i} new global best {overall_best_score:.6f}"
                )

    # ------------------------------------------------------------------
    # 5. Upsample to a larger size (if time permits) and refine again
    remaining = deadline - time.time()
    if remaining > 150.0:
        large_n = min(18000, len(overall_best_seq) * 2)
        print(f"[{time.time():.1f}] up‑sampling to {large_n} points for final refinement")
        up_seq = _upsample(overall_best_seq, large_n)
        sub_deadline = time.time() + remaining * 0.65
        up_seq_opt, up_score = _subgradient_descent_adagrad(
            up_seq,
            sub_deadline,
            best_seq=overall_best_seq,
            best_score=overall_best_score,
            lr0=0.6,
            max_active=80,
            eval_interval=120,
            patience=2000,
        )
        if up_score < overall_best_score:
            overall_best_score, overall_best_seq = up_score, up_seq_opt.copy()
            print(
                f"[{time.time():.1f}] up‑sampled descent yielded new best {overall_best_score:.6f}"
            )

    # ------------------------------------------------------------------
    # 6. Deterministic pairwise refinement
    if time.time() < deadline:
        print(f"[{time.time():.1f}] deterministic pairwise refinement")
        overall_best_seq, overall_best_score = _pairwise_swap_local(
            overall_best_seq.copy(),
            deadline,
            best_seq=overall_best_seq,
            best_score=overall_best_score,
            max_trials=300_000,
        )

    # ------------------------------------------------------------------
    # 7. Final cheap simulated‑annealing clean‑up
    if time.time() < deadline:
        print(f"[{time.time():.1f}] final simulated‑annealing clean‑up")
        overall_best_seq, overall_best_score = _random_local_search(
            overall_best_seq.copy(),
            deadline,
            best_seq=overall_best_seq,
            best_score=overall_best_score,
            init_temp=0.008,
            decay=0.9995,
        )

    # ------------------------------------------------------------------
    print(
        f"Final best score: {overall_best_score:.6f} (length={len(overall_best_seq)})"
    )
    # Return a plain Python list; evaluation will clip/normalise as required.
    return overall_best_seq.tolist()
