def _correlation_valid(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    # Computes correlation of x and y: (x ⋆ y)[i] = sum_j x[i+j] * y[j]; returns length len(y)

    n = y.shape[0]
    m = x.shape[0]

    X = np.fft.rfft(x, n=size)

"""
Multi‑stage optimizer for a non‑negative step‑function minimizing
``evaluate_sequence`` (i.e. 2·n·max(self‑convolution) with the sum of the
coefficients normalised to 1).

Stage 1 – generate a diverse set of candidate shapes from several analytic
families (uniform, arcsine, sine, cosine, quadratic, symmetric‑Beta,
edge‑weighted, power‑sum) on a few modest lengths, plus random Dirichlet
and sparse seeds.  The 30 best are kept.

Stage 2 – apply a *smooth* exponential‑gradient descent.  The non‑smooth
objective ``max(conv)`` is replaced by a soft‑max
``S_τ(a)= (1/τ)·log Σ_k exp(τ·(a*a)_k)``.
For a given temperature τ the gradient can be written as a
correlation, which is evaluated efficiently by FFT.  τ is increased
gradually (τ ← τ·growth) so that optimisation first sees a smoothed
landscape and then converges to the true objective.

Stage 3 – exact sub‑gradient (mirror) descent on the original
non‑smooth objective (as in the reference implementation) starting from
the best smoothed results.

Stage 4 – a cheap greedy pairwise “mass‑transfer’’ polishing that only
keeps strictly improving moves.

All steps respect the supplied time budget (default 1000 s) and the
final vector is returned as a plain list of non‑negative floats.
"""

# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Euclidean projection onto the simplex {x≥0, Σx=s}."""
    if s < 0:
        raise ValueError("Simplex radius must be non‑negative")
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.where(u - cssv / (np.arange(1, len(v) + 1)) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    return np.maximum(v - theta, 0.0)


# ----------------------------------------------------------------------
def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Full self‑convolution ``a*a`` using a real FFT (length 2·n−1)."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1) - 1).bit_length()
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[: 2 * n - 1]


# ----------------------------------------------------------------------
def _objective(a: np.ndarray) -> float:
    """
    Exact objective ``2·n·max(conv(a,a))`` for a vector that sums to 1.
    """
    n = a.shape[0]
    b = _fft_self_convolve(a)
    return float(2.0 * n * b.max())

# Analytic families ----------------------------------------------------
def _distribution_uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _distribution_arcsine(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    vals /= vals.sum()
    return vals


def _distribution_sine(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64) + 0.5
    w = np.sin(np.pi * i / n)
    w /= w.sum()
    return w


def _distribution_cosine(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64) + 0.5
    w = 1.0 - np.cos(np.pi * i / n)
    w /= w.sum()
    return w


def _distribution_quadratic(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    x = (2.0 * i) / (n - 1) - 1.0
    w = 1.0 - x * x
    w = np.clip(w, 0.0, None)
    w /= w.sum()
    return w


def _distribution_beta_sym(n: int, p: float) -> np.ndarray:
    """Symmetric Beta(p,p) on a regular grid."""
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    w = np.exp((p - 1) * np.log(x) + (p - 1) * np.log(1.0 - x))
    if w.sum() == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _distribution_edge_weighted(n: int, p: float) -> np.ndarray:
    """
    Edge‑weighted family: w_i ∝ ((i+½)*(n‑i‑½))^{‑p}.
    """
    i = np.arange(n, dtype=np.float64) + 0.5
    j = n - np.arange(n, dtype=np.float64) - 0.5
    w = (i * j) ** (-p)
    w = np.clip(w, 0.0, None)
    total = w.sum()
    if total > 0:
        w /= total
    else:
        w = np.full(n, 1.0 / n)
    return w


def _distribution_power_sum(n: int, p: float) -> np.ndarray:
    """w_i ∝ (i+1)^{‑p} + (n‑i)^{‑p}."""
    i = np.arange(1, n + 1, dtype=np.float64)
    w = i ** (-p) + (n + 1 - i) ** (-p)
    w = np.clip(w, 0.0, None)
    w /= w.sum()
    return w


def _distribution_random_sparse(
    n: int, rng: np.random.Generator, density: float = 0.01
) -> np.ndarray:
    """Sparse random vector with a prescribed density of non‑zeros."""
    k = max(2, int(n * density))
    idx = rng.choice(n, size=k, replace=False)
    vals = np.zeros(n, dtype=np.float64)
    vals[idx] = rng.random(k)
    if vals.sum() == 0.0:
        vals[0] = 1.0
    vals /= vals.sum()
    return vals


def _mix_with_uniform(base: np.ndarray, lam: float) -> np.ndarray:
    """Convex combination λ·uniform + (1‑λ)·base."""
    n = base.shape[0]
    uniform = np.full(n, 1.0 / n, dtype=np.float64)
    return lam * uniform + (1.0 - lam) * base


# ----------------------------------------------------------------------
# Correlation (valid part) via FFT ---------------------------------------
def _correlation_valid(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """
    Compute the valid correlation (x ⋆ y)[i] = Σ_j x[i+j]·y[j]  for i=0…len(y)-1.
    Implemented with FFT for O(N log N) speed.
    """
    m = x.shape[0]
    n = y.shape[0]
    size = 1 << ((m + n - 1) - 1).bit_length()
    X = np.fft.rfft(x, n=size)
    Y = np.fft.rfft(y[::-1], n=size)          # reverse y for correlation
    corr_full = np.fft.irfft(X * Y, n=size)[: m + n - 1]
    start = n - 1
    return corr_full[start:start + n]

# Exact sub‑gradient (used later for refinement) ------------------------
def _max_convolution_subgradient(a: np.ndarray) -> np.ndarray:
    """
    Sub‑gradient of  max_k Σ_i a_i a_{k-i}.
    Returns a vector of the same shape as ``a``.
    """
    conv = _fft_self_convolve(a)
    max_val = conv.max()
    tol = 1e-12 * max_val
    active = np.where(conv >= max_val - tol)[0]
    n = a.shape[0]
    grad = np.zeros_like(a)
    for k in active:
        lo = max(0, k - (n - 1))
        hi = min(n - 1, k)
        idx = np.arange(lo, hi + 1)
        grad[idx] += 2.0 * a[k - idx]
    if len(active) > 0:
        grad /= len(active)
    return grad


def _exp_gradient_descent(
    a0: np.ndarray,
    time_limit: float,
    best_seq: np.ndarray,
    best_score: float,
    eta0: float = 0.6,
    report_every: int = 400,
    max_iter: int = 300_000,
) -> tuple[np.ndarray, float]:
    """
    Exponentiated‑gradient descent on the exact max‑convolution objective.
    Uses the true sub‑gradient of the active convolution index.
    """
    a = a0.copy()
    n = a.shape[0]
    t = 0
    while time.time() < time_limit and t < max_iter:
        t += 1
        grad = _max_convolution_subgradient(a)
        eta = eta0 / math.sqrt(t)
        a = a * np.exp(-eta * grad)
        a = _project_onto_simplex(a)
        if t % report_every == 0:
            cur = _objective(a)
            if cur < best_score - 1e-14:
                best_score = cur
                best_seq = a.copy()
                print(f"[{time.time():.1f}] exp‑gd it={t} new best {best_score:.6f}")
    return best_seq, best_score

# Greedy pairwise polishing ---------------------------------------------
def _pairwise_swap(
    a0: np.ndarray,
    time_limit: float,
    best_seq: np.ndarray,
    best_score: float,
    max_trials: int | None = None,
) -> tuple[np.ndarray, float]:
    """
    Randomly move a fraction of mass from one index to another.
    Only moves that strictly improve the exact objective are kept.
    """
    a = a0.copy()
    n = a.shape[0]
    trials = 0
    while time.time() < time_limit:
        if max_trials is not None and trials >= max_trials:
            break
        trials += 1

        src = np.where(a > 1e-14)[0]
        if src.size == 0:
            break
        i = np.random.choice(src)
        j = np.random.randint(n - 1)
        if j >= i:
            j += 1

        delta = a[i] * np.random.rand() * 0.45   # up to 45 % of source mass
        if delta < 1e-15:
            continue

        new_a = a.copy()
        new_a[i] -= delta
        new_a[j] += delta

        cur = _objective(new_a)
        if cur < best_score - 1e-14:
            a = new_a
            best_score = cur
            best_seq = a.copy()
            print(f"[{time.time():.1f}] swap i={i}→j={j} δ={delta:.6g} new best {best_score:.6f}")
    return best_seq, best_score

# Smooth (soft‑max) exponential‑gradient descent -------------------------
def _smooth_gradient_descent(
    a0: np.ndarray,
    time_limit: float,
    best_seq: np.ndarray,
    best_score: float,
    eta0: float = 0.02,
    tau_initial: float = 1.0,
    tau_max: float = 30.0,
    tau_growth: float = 1.2,
    report_every: int = 300,
) -> tuple[np.ndarray, float]:
    """
    Gradient descent on the *soft‑max* surrogate
        S_τ(a) = (1/τ)·log Σ_k exp(τ·(a*a)_k)

    The exact gradient is
        ∇S_τ = 2·corr(w, a) - w_diag·a,
    where w_k = exp(τ·conv_k) / Σ exp(τ·conv).
    The true objective gradient is 2·n·∇S_τ.
    A simple diminishing step‑size η_t = η₀/√t is used, followed by
    projection onto the simplex.
    """
    a = a0.copy()
    n = a.shape[0]
    t = 0
    tau = tau_initial

    while time.time() < time_limit:
        t += 1
        conv = _fft_self_convolve(a)
        max_conv = conv.max()
        # soft‑max probabilities (numerically stable)
        w = np.exp(tau * (conv - max_conv))
        w_sum = w.sum()
        if w_sum == 0.0:
            w = np.full_like(conv, 1.0 / conv.size)
        else:
            w /= w_sum

        # correlation w ⋆ a (valid part)
        corr = _correlation_valid(w, a)                # shape (n,)

        # diagonal weights w_{2i}
        diag_idx = 2 * np.arange(n)
        w_diag = w[diag_idx]                           # always in range because len(conv)=2n‑1

        # gradient of the *actual* objective (2·n·S_τ)
        grad = 4.0 * n * corr - 2.0 * n * w_diag * a

        eta = eta0 / math.sqrt(t)
        a = a - eta * grad
        a = _project_onto_simplex(a)

        # increase temperature gradually
        if t % 500 == 0:
            tau = min(tau_max, tau * tau_growth)

        if t % report_every == 0:
            cur = _objective(a)
            if cur < best_score - 1e-14:
                best_score = cur
                best_seq = a.copy()
                print(f"[{time.time():.1f}] smooth it={t} τ={tau:.2f} new best {best_score:.6f}")

    return best_seq, best_score

            best_seq=overall_best_seq,
            best_score=overall_best_score,
        )
        if sc_opt < overall_best_score - 1e-12:
            overall_best_score = sc_opt
            overall_best_seq = a_opt.copy()
            print(f"after final swaps new best {overall_best_score:.6f}")

    print(
        f"Final best score (2·n·max(conv)): {overall_best_score:.6f} "
        f"(length={len(overall_best_seq)})"
    )
    return overall_best_seq.tolist()
