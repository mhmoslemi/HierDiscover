    if s < 0:
        raise ValueError("simplex radius must be non‑negative")
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho_arr = np.where(u - cssv / (np.arange(1, len(v) + 1)) > 0)[0]
    if rho_arr.size == 0:
        theta = 0.0
    else:
        rho = rho_arr[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Full self‑convolution of a (length 2·n‑1) using a real FFT."""
    n = a.shape[0]
    need = 2 * n - 1
    size = 1 << ((need - 1).bit_length())          # next power of two
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[:need]


# ----------------------------------------------------------------------
def _objective(a: np.ndarray) -> float:
    """Exact objective 2·n·max(conv(a,a)) for a vector with Σa = 1."""
    n = a.shape[0]
    b = _fft_self_convolve(a)
    return 2.0 * n * b.max()


# ----------------------------------------------------------------------
# Analytic families ----------------------------------------------------
def _distribution_uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _distribution_arcsine(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    vals /= vals.sum()
    return vals


def _distribution_beta_sym(n: int, p: float) -> np.ndarray:
    """Discrete Beta(p,p)."""
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    w = np.exp((p - 1) * np.log(x) + (p - 1) * np.log(1.0 - x))
    if w.sum() == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _distribution_edge_weighted(n: int, p: float) -> np.ndarray:
    """U‑shaped family w_i ∝ ((i+½)*(n‑i‑½))^(−p)."""
    i = np.arange(n, dtype=np.float64) + 0.5
    j = (n - np.arange(n, dtype=np.float64) - 0.5)
    w = (i * j) ** (-p)
    if w.sum() == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _distribution_sine_weighted(n: int, p: float) -> np.ndarray:
    """Weight ∝ sin(π (i+½)/n)^p."""
    theta = np.pi * (np.arange(n, dtype=np.float64) + 0.5) / n
    w = np.sin(theta) ** p
    if w.sum() == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _distribution_random_dirichlet(n: int, rng: np.random.Generator,
                                   alpha: float = 0.2) -> np.ndarray:
    """Random Dirichlet vector (α controls sparsity)."""
    vec = rng.dirichlet(np.full(n, alpha))
    if vec.sum() == 0.0:
        vec[0] = 1.0
    return vec / vec.sum()


# ----------------------------------------------------------------------
def _smooth_adam_descent(a: np.ndarray,
                         deadline: float,
                         best_seq: np.ndarray,
                         best_score: float,
                         lr: float = 0.15,
                         beta1: float = 0.9,
                         beta2: float = 0.999,
                         epsilon: float = 1e-8,
                         alpha_start: float = 2.0,
                         alpha_end: float = 60.0,
                         ramp_iters: int = 200_000,
                         jitter_interval: int = 20_000) -> Tuple[np.ndarray, float]:
    """
    Adam optimiser on the smooth approximation of max‑convolution.
    The schedule for the smoothing parameter α starts at ``alpha_start``,
    linearly increases to ``alpha_end`` over ``ramp_iters`` iterations.
    """
    n = a.shape[0]
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    t_iter = 0
    no_improve = 0
    rng = np.random.default_rng()
    cur = a.copy()

    while time.time() < deadline:
        t_iter += 1

        # ----- smoothing parameter -----
        ramp = min(1.0, t_iter / ramp_iters)
        alpha = alpha_start + ramp * (alpha_end - alpha_start)

        # ----- convolution and softmax -----
        conv = _fft_self_convolve(cur)
        max_c = conv.max()
        # softmax on (conv - max) to avoid overflow
        exp_vals = np.exp(alpha * (conv - max_c))
        w = exp_vals / exp_vals.sum()

        # ----- gradient (cross‑correlation) -----
        # np.correlate(w, cur, mode='valid') gives length n, exactly Σ w_{k+i}·a_i
        grad = 2.0 * np.correlate(w, cur, mode='valid')

        # ----- Adam update -----
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)

        m_hat = m / (1.0 - beta1 ** t_iter)
        v_hat = v / (1.0 - beta2 ** t_iter)

        step = lr * m_hat / (np.sqrt(v_hat) + epsilon)

        nxt = cur - step
        nxt = _project_onto_simplex(nxt, s=1.0)

        cur_score = _objective(nxt)

        if cur_score < best_score - 1e-13:
            best_score = cur_score
            best_seq = nxt.copy()
            no_improve = 0
            print(f"[{time.time():.1f}] Adam it={t_iter} new best {best_score:.6f}")
        else:
            no_improve += 1

        # ----- occasional jitter to escape stagnation -----
        if no_improve >= jitter_interval:
            jitter = rng.exponential(scale=0.002, size=n)
            nxt = _project_onto_simplex(nxt + jitter, s=1.0)
            no_improve = 0
            # recompute score (may be a little worse – that's fine)
            cur_score = _objective(nxt)

        cur = nxt

    return best_seq, best_score


# ----------------------------------------------------------------------
def _pairwise_swap_local(a: np.ndarray,
                         deadline: float,
                         best_seq: np.ndarray,
                         best_score: float,
                         max_trials: int | None = None) -> Tuple[np.ndarray, float]:
    """Greedy deterministic move of a small amount of mass."""
    n = a.shape[0]
    rng = np.random.default_rng()
    trials = 0
    while time.time() < deadline:
        if max_trials is not None and trials >= max_trials:
            break
        trials += 1

        # pick source with non‑trivial mass
        src = np.where(a > 1e-12)[0]
        if src.size == 0:
            break
        i = rng.choice(src)

        # pick a distinct destination
        j = rng.integers(n - 1)
        if j >= i:
            j += 1

        delta = a[i] * rng.random() * 0.35
        if delta < 1e-14:
            continue

        new_a = a.copy()
        new_a[i] -= delta
        new_a[j] += delta

        cur = _objective(new_a)
        if cur < best_score - 1e-14:
            a = new_a
            best_score = cur
            best_seq = a.copy()
            print(
                f"[{time.time():.1f}] swap i={i}→j={j} δ={delta:.3e} new best {best_score:.6f}"
            )
    return best_seq, best_score


# ----------------------------------------------------------------------
def _random_local_search(a: np.ndarray,
                         deadline: float,
                         best_seq: np.ndarray,
                         best_score: float,
                         init_temp: float = 0.008,
                         decay: float = 0.9997) -> Tuple[np.ndarray, float]:
    """Very cheap simulated‑annealing walk."""
    n = a.shape[0]
    rng = np.random.default_rng()
    T = init_temp
    while time.time() < deadline:
        i = rng.integers(n)
        j = rng.integers(n - 1)
        if j >= i:
            j += 1
        delta = a[i] * rng.random() * 0.3
        if delta < 1e-14:
            continue
        new_a = a.copy()
        new_a[i] -= delta
        new_a[j] += delta
        cur = _objective(new_a)
        if cur < best_score - 1e-14:
            a = new_a
            best_score = cur
            best_seq = a.copy()
            print(f"[{time.time():.1f}] SA improve {i}->{j} Δ={delta:.3e} best {best_score:.6f}")
        else:
            if rng.random() < math.exp(-(cur - best_score) / max(T, 1e-12)):
                a = new_a
        T *= decay
    return best_seq, best_score

def propose_candidate(seed: int = 42, budget_s: int = 1000) -> List[float]:
    """Return a list of heights (non‑negative, sum ≈ 1) that minimises
    ``evaluate_sequence``.  The function respects the given time budget."""
    rng = np.random.default_rng(seed)
    start = time.time()
    deadline = start + budget_s - 3.0               # safety margin

    overall_best_seq = None
    overall_best_score = float("inf")

    # ------------------------------------------------------------------
    # 1. generate a modest pool of analytic / random candidates
    lengths = [2000, 3000, 4000, 5000, 6000, 8000]
    beta_ps = [0.2, 0.5, 1.0, 2.0]                  # a few symmetric‑Beta exponents
    edge_ps = [0.3, 0.7, 1.0]
    sine_ps = [2, 4, 6]

    for n in lengths:
        if time.time() >= deadline:
            break
        print(f"[{time.time():.1f}] testing length {n}")

        # uniform
        cand = _distribution_uniform(n)
        sc = _objective(cand)
        if sc < overall_best_score:
            overall_best_score, overall_best_seq = sc, cand.copy()
            print(f"  new global best (uniform) score={sc:.6f}")

        # arcsine
        cand = _distribution_arcsine(n)
        sc = _objective(cand)
        if sc < overall_best_score:
            overall_best_score, overall_best_seq = sc, cand.copy()
            print(f"  new global best (arcsine) score={sc:.6f}")

        # symmetric beta family
        for p in beta_ps:
            cand = _distribution_beta_sym(n, p)
            sc = _objective(cand)
            if sc < overall_best_score:
                overall_best_score, overall_best_seq = sc, cand.copy()
                print(f"  new global best (βsym p={p}) score={sc:.6f}")

        # edge‑weighted
        for p in edge_ps:
            cand = _distribution_edge_weighted(n, p)
            sc = _objective(cand)
            if sc < overall_best_score:
                overall_best_score, overall_best_seq = sc, cand.copy()
                print(f"  new global best (edge p={p}) score={sc:.6f}")

        # sine‑weighted
        for p in sine_ps:
            cand = _distribution_sine_weighted(n, p)
            sc = _objective(cand)
            if sc < overall_best_score:
                overall_best_score, overall_best_seq = sc, cand.copy()
                print(f"  new global best (sine p={p}) score={sc:.6f}")

        # a few random Dirichlet vectors
        for _ in range(3):
            alpha = rng.uniform(0.05, 0.5)
            cand = _distribution_random_dirichlet(n, rng, alpha=alpha)
            sc = _objective(cand)
            if sc < overall_best_score:
                overall_best_score, overall_best_seq = sc, cand.copy()
                print(f"  new global best (Dirichlet α={alpha:.3g}) score={sc:.6f}")

    # ------------------------------------------------------------------
    # 2. try to import any previously discovered construction
    try:
        prev = np.asarray(height_sequence_1, dtype=np.float64)  # type: ignore[name-defined]
        if prev.ndim == 1 and prev.size > 0:
            prev = prev / prev.sum()
            sc = _objective(prev)
            if sc < overall_best_score:
                overall_best_score, overall_best_seq = sc, prev.copy()
                print(f"[{time.time():.1f}] imported previous best score={sc:.6f}")
    except Exception:
        pass

    # ------------------------------------------------------------------
    # 3. Smooth Adam descent on the current best seed
    if overall_best_seq is None:
        # fallback – create a uniform vector
        n = 4000
        overall_best_seq = _distribution_uniform(n)
        overall_best_score = _objective(overall_best_seq)

    remaining = deadline - time.time()
    if remaining > 30.0:
        print(f"[{time.time():.1f}] starting smooth Adam descent (n={len(overall_best_seq)})")
        overall_best_seq, overall_best_score = _smooth_adam_descent(
            overall_best_seq.copy(),
            deadline,
            best_seq=overall_best_seq,
            best_score=overall_best_score,
            lr=0.15,
            jitter_interval=15_000,
        )
        print(f"[{time.time():.1f}] after Adam best score={overall_best_score:.6f}")

    # ------------------------------------------------------------------
    # 4. Greedy pairwise refinement (deterministic)
    if time.time() < deadline:
        print(f"[{time.time():.1f}] greedy pairwise refinement")
        overall_best_seq, overall_best_score = _pairwise_swap_local(
            overall_best_seq.copy(),
            deadline,
            best_seq=overall_best_seq,
            best_score=overall_best_score,
            max_trials=300_000,
        )

    # ------------------------------------------------------------------
    # 5. Final cheap simulated‑annealing walk
    if time.time() < deadline:
        print(f"[{time.time():.1f}] final simulated‑annealing walk")
        overall_best_seq, overall_best_score = _random_local_search(
            overall_best_seq.copy(),
            deadline,
            best_seq=overall_best_seq,
            best_score=overall_best_score,
            init_temp=0.008,
            decay=0.9997,
        )

    print(
        f"Final best score: {overall_best_score:.6f} (length={len(overall_best_seq)})"
    )
    # ``evaluate_sequence`` will later apply clipping etc.; we just return the raw list.
    return overall_best_seq.tolist()
