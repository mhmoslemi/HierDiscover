def lp_improve(f, sub_deadline):
    # guarantee f sum = 1
    while time.time() < sub_deadline:
        # compute M = max conv(f, f)
        conv = _fft_self_convolve(f)
        M = conv.max()
        # build constraint matrix A_ub
        n = f.shape[0]
        rows = 2*n - 1
        # using dense? choose n small enough; else skip
        A = np.zeros((rows, n))
        for j in range(n):
            A[j:j+n, j] = f
        # solve LP
        c = -np.ones(n)
        bounds = [(0, None)]*n
        res = linprog(c, A_ub=A, b_ub=np.full(rows, M), bounds=bounds, method='highs', options={'presolve': True})
        if not res.success:
            break
        g = res.x
        s = np.sum(g)
        if s <= 0:
            break
        g_norm = g / s
        # line search for best mixing alpha
        # sample alphas
        alphas = np.linspace(0, 1, 11)[1:]  # exclude 0
        best_local_score = _objective(f)
        best_local_f = f
        for alpha in alphas:
            new_f = (1-alpha)*f + alpha*g_norm
            # ensure sum to 1 (should already)
            score = _objective(new_f)
            if score < best_local_score:
                best_local_score = score
                best_local_f = new_f.copy()
        # if improvement, update f else break or jitter
        if best_local_score < best_score - 1e-12:
            f = best_local_f
            best_score = best_local_score
            best_seq = f.copy()
        else:
            # jitter to escape
            jitter = rng.exponential(scale=0.001, size=n)
            f = f + jitter
            f = np.maximum(f, 0)
            f = _project_onto_simplex(f)
    return best_seq, best_score
```

"""
Hybrid LP + mirror‑descent optimizer for the step‑function minimisation problem.

The routine builds a diverse pool of analytic and random seeds, keeps the
best few, then (if SciPy is available) improves the current best using a
linear‑programming “direction‑finding’’ step: maximise the L1‑mass of a
candidate g under the constraint ‖f∗g‖∞ ≤ M = ‖f∗f‖∞.  The optimiser mixes
the obtained g with f (line‑search over the mixing weight) and repeats this
procedure while respecting the overall time budget.

The refined sequence is finally polished with an exponentiated‑gradient
(momentum) mirror‑descent, a greedy pairwise mass‑transfer, and a cheap
simulated‑annealing local search.  All numbers are non‑negative; the
returned list can be fed directly to ``evaluate_sequence``.
"""

import time
import math
import numpy as np
from numpy.fft import rfft, irfft

# Optional SciPy imports – the LP phase is skipped if SciPy is unavailable.

def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Euclidean projection onto the simplex {x≥0, sum(x)=s}."""
    if s < 0:
        raise ValueError("simplex radius must be non‑negative")
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho_arr = np.where(u - cssv / (np.arange(1, len(v) + 1)) > 0)[0]
    if rho_arr.size == 0:
        theta = 0.0
    else:
        rho = rho_arr[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Full self‑convolution a*a using a real FFT (length 2n‑1)."""
    n = a.shape[0]
    need = 2 * n - 1
    size = 1 << ((need - 1).bit_length())
    A = rfft(a, n=size)
    conv = irfft(A * A, n=size)
    return conv[:need]


def _objective(a: np.ndarray) -> float:
    """Exact objective 2·n·max(conv(a,a)) for sum(a)=1."""
    n = a.shape[0]
    b = _fft_self_convolve(a)
    return 2.0 * n * b.max()

# Analytic families ----------------------------------------------------
def _distribution_uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _distribution_arcsine(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    vals /= vals.sum()
    return vals


def _distribution_beta_sym(n: int, p: float) -> np.ndarray:
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    w = np.exp((p - 1) * np.log(x) + (p - 1) * np.log(1.0 - x))
    s = w.sum()
    if s == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= s
    return w


def _distribution_beta_general(n: int, alpha: float, beta: float) -> np.ndarray:
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    eps = np.finfo(float).tiny
    x = np.clip(x, eps, 1 - eps)
    w = np.exp((alpha - 1) * np.log(x) + (beta - 1) * np.log(1.0 - x))
    s = w.sum()
    if s == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= s
    return w


def _distribution_edge_weighted(n: int, p: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64) + 0.5
    j = (n - np.arange(n, dtype=np.float64) - 0.5)
    w = (i * j) ** (-p)
    s = w.sum()
    if s == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= s
    return w


def _distribution_edge_inversepow(n: int, p: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64) + 0.5
    j = (n - np.arange(n, dtype=np.float64) - 0.5)
    w = i ** (-p) + j ** (-p)
    s = w.sum()
    if s == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= s
    return w


def _distribution_sine_weighted(n: int, p: float) -> np.ndarray:
    theta = np.pi * (np.arange(n, dtype=np.float64) + 0.5) / n
    w = np.sin(theta) ** p
    s = w.sum()
    if s == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= s
    return w


def _distribution_tent(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    half = (n - 1) / 2.0
    w = np.where(i <= half, i + 1, n - i)
    w /= w.sum()
    return w


def _distribution_quadratic(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    half = (n - 1) / 2.0
    d = np.abs(i - half) + 1.0
    w = 1.0 / (d ** 2)
    w /= w.sum()
    return w


def _distribution_exponential_decay(n: int, rate: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    half = (n - 1) / 2.0
    w = np.exp(-rate * np.abs(i - half))
    w /= w.sum()
    return w


def _distribution_random_sparse(n: int, rng: np.random.Generator,
                               density: float = 0.01) -> np.ndarray:
    k = max(2, int(n * density))
    idx = rng.choice(n, size=k, replace=False)
    vals = np.zeros(n, dtype=np.float64)
    vals[idx] = rng.random(k)
    if vals.sum() == 0.0:
        vals[0] = 1.0
    vals /= vals.sum()
    return vals


def _distribution_dirichlet(n: int, rng: np.random.Generator) -> np.ndarray:
    alpha = rng.uniform(0.01, 2.0)
    a = rng.dirichlet(np.full(n, alpha))
    return a.astype(np.float64)


def _mix_with_uniform(base: np.ndarray, lam: float) -> np.ndarray:
    n = base.shape[0]
    uniform = np.full(n, 1.0 / n, dtype=np.float64)
    return lam * uniform + (1.0 - lam) * base

# Momentum mirror‑descent (same as before, only tiny parameter tweaks)
def _subgradient_descent_momentum(
    a: np.ndarray,
    deadline: float,
    best_seq: np.ndarray,
    best_score: float,
    lr0: float = 0.8,
    active_frac: float = 0.03,
    max_active: int = 25,
    beta: float = 0.9,
    eval_interval: int = 250,
    patience: int = 4000,
) -> tuple[np.ndarray, float]:
    """Exponentiated‑gradient descent with momentum, projected onto simplex."""
    n = a.shape[0]
    t = 0
    no_improve = 0
    momentum = np.zeros_like(a)
    rng = np.random.default_rng()
    ind_grid = np.arange(n, dtype=np.int32)

    while time.time() < deadline and t < 500_000:
        t += 1

        conv = _fft_self_convolve(a)
        max_c = conv.max()
        thresh = max_c * (1.0 - active_frac)

        act = np.where(conv >= thresh)[0]
        if act.size > max_active:
            top_idx = np.argpartition(conv[act], -max_active)[-max_active:]
            act = act[top_idx]

        grad = np.zeros_like(a)
        for idx in act:
            offset = idx - ind_grid
            mask = (offset >= 0) & (offset < n)
            grad[mask] += 2.0 * a[offset[mask]]

        momentum = beta * momentum + (1.0 - beta) * grad
        eta = lr0 / math.sqrt(t)

        a = a * np.exp(-eta * momentum)
        a = _project_onto_simplex(a, s=1.0)

        if t % eval_interval == 0:
            cur = _objective(a)
            if cur < best_score - 1e-14:
                best_score = cur
                best_seq = a.copy()
                no_improve = 0
                print(f"[{time.time():.1f}] it={t:,} new global best {best_score:.6f}")
            else:
                no_improve += 1

        if no_improve >= patience:
            jitter = rng.exponential(scale=0.006, size=n)
            a = a + jitter
            a = np.maximum(a, 0.0)
            a = _project_onto_simplex(a, s=1.0)
            no_improve = 0

    return best_seq, best_score

# Greedy pairwise refinement -------------------------------------------
def _pairwise_swap_local(
    a: np.ndarray,
    deadline: float,
    best_seq: np.ndarray,
    best_score: float,
    max_trials: int | None = None,
) -> tuple[np.ndarray, float]:
    """Move a random amount of mass between two positions if the objective improves."""
    n = a.shape[0]
    trials = 0
    rng = np.random.default_rng()
    while time.time() < deadline:
        if max_trials is not None and trials >= max_trials:
            break
        trials += 1

        src = np.where(a > 1e-12)[0]
        if src.size == 0:
            break
        i = rng.choice(src)
        j = rng.integers(n - 1)
        if j >= i:
            j += 1

        delta = a[i] * rng.random() * 0.35
        if delta < 1e-14:
            continue

        new_a = a.copy()
        new_a[i] -= delta
        new_a[j] += delta
        cur = _objective(new_a)

        if cur < best_score - 1e-14:
            a = new_a
            best_score = cur
            best_seq = a.copy()
            print(f"[{time.time():.1f}] swap {i}->{j} Δ={delta:.3e} new best {best_score:.6f}")

    return best_seq, best_score

# Simulated‑annealing style local search -------------------------------
def _random_local_search(
    a: np.ndarray,
    deadline: float,
    best_seq: np.ndarray,
    best_score: float,
    init_temp: float = 0.01,
    decay: float = 0.9995,
) -> tuple[np.ndarray, float]:
    """Very cheap SA that repeatedly transfers a random amount of mass."""
    n = a.shape[0]
    rng = np.random.default_rng()
    T = init_temp
    while time.time() < deadline:
        i = rng.integers(n)
        j = rng.integers(n - 1)
        if j >= i:
            j += 1
        delta = a[i] * rng.random() * 0.3
        if delta < 1e-14:
            continue
        new_a = a.copy()
        new_a[i] -= delta
        new_a[j] += delta
        cur = _objective(new_a)
        if cur < best_score - 1e-14:
            a = new_a
            best_score = cur
            best_seq = a.copy()
            print(f"[{time.time():.1f}] SA improve {i}->{j} Δ={delta:.3e} best {best_score:.6f}")
        else:
            if rng.random() < math.exp(-(cur - best_score) / max(T, 1e-12)):
                a = new_a
        T *= decay
    return best_seq, best_score

def _lp_iterative_improve(

    best_seq: np.ndarray,

    Repeatedly solve the LP:
        maximise 1ᵀg   s.t.   conv(a,g) ≤ M,   g ≥ 0

    to obtain a new, improved a.
    """

        print("[LP] SciPy not available – skipping LP improvement.")
        return best_seq, best_score

    n = a.shape[0]
    a = _project_onto_simplex(a, s=1.0)
    rng = np.random.default_rng()
    while time.time() < deadline:
        # current bound
        conv = _fft_self_convolve(a)
        M = conv.max()
        # build Toeplitz convolution matrix (rows = 2n‑1, cols = n)
        col = np.concatenate([a, np.zeros(n - 1, dtype=np.float64)])
        row = np.concatenate([a[:1], np.zeros(n - 1, dtype=np.float64)])
        A = toeplitz(col, row)     # dense, but modest size for n ≤ 1500

        c = -np.ones(n, dtype=np.float64)           # maximise Σg → minimise −Σg
        bounds = [(0, None)] * n
        b_ub = np.full(2 * n - 1, M, dtype=np.float64)

        # time‑limit for the LP solver (to respect the outer deadline)
        time_left = max(0.0, deadline - time.time())
        opts = {"presolve": True}
        if time_left > 0:
            opts["time_limit"] = time_left

        try:
            res = linprog(c, A_ub=A, b_ub=b_ub,
                          bounds=bounds, method='highs', options=opts)
        except Exception as e:                      # pragma: no cover
            print(f"[LP] Solver exception: {e}")
            break

        if not res.success:                         # pragma: no cover
            print(f"[LP] Solver failed: {res.message}")
            break

        g = res.x
        sum_g = g.sum()
        if sum_g <= 0:
            break
        g_norm = g / sum_g    # enforce unit L1‑norm

        # line‑search over mixing weight α ∈ (0,1]

            score = _objective(new_f)
            if score < best_local_score:

                best_local_f = new_f.copy()

        if best_local_score < best_score - 1e-12:
            a = best_local_f
            best_score = best_local_score

            a = a + jitter

    return best_seq, best_score

def propose_candidate(seed: int = 42, budget_s: int = 1000) -> list[float]:
    """
    Search for a non‑negative step‑function that minimises ``evaluate_sequence``.
    The algorithm consists of:
      1) diversified seed generation,
      2) a linear‑programming directional improvement (if SciPy is present),
      3) long momentum mirror‑descent,
      4) greedy pairwise polishing,
      5) cheap simulated‑annealing local search.
    Returns a plain Python list of heights.
    """
    rng = np.random.default_rng(seed)
    start = time.time()
    deadline = start + budget_s - 2.0      # safety margin

    overall_best_seq: np.ndarray | None = None
    overall_best_score = float("inf")
    top_candidates: list[tuple[float, np.ndarray]] = []

    # ------------------------------------------------------------------
    # 1. Diverse seed generation (fast, many analytic families)
    lengths = [800, 1200, 1500, 2000, 2500, 3000]   # modest sizes for LP stage
    p_grid = np.concatenate([
        np.logspace(-3, -0.5, 8),      # 0.001 … 0.316
        np.logspace(-0.5, 0.0, 8),    # 0.316 … 1
        np.logspace(0.0, 0.5, 6),     # 1 … 3.16
    ])
    lam_grid = np.linspace(0.0, 0.9, 10)    # more extensive mixing
    rate_grid = np.linspace(0.001, 0.02, 6)

    for n in lengths:
        if time.time() >= deadline:
            break
        print(f"[{time.time():.1f}] generating seeds, length={n}")

        # deterministic families
        for cand in (_distribution_uniform(n),
                     _distribution_arcsine(n),
                     _distribution_tent(n),
                     _distribution_quadratic(n)):
            sc = _objective(cand)
            top_candidates.append((sc, cand.copy()))
            if sc < overall_best_score:
                overall_best_score, overall_best_seq = sc, cand.copy()
                print(f"[{time.time():.1f}] new global best (det) len={n} score={sc:.6f}")

        # parametrised families + uniform mixing
        for p in p_grid:
            if time.time() >= deadline:
                break
            edge = _distribution_edge_weighted(n, p)
            edge2 = _distribution_edge_inversepow(n, p)
            sine = _distribution_sine_weighted(n, p)
            beta = _distribution_beta_sym(n, p)
            for base in (edge, edge2, sine, beta):
                for lam in lam_grid:
                    if time.time() >= deadline:
                        break
                    cand = _mix_with_uniform(base, lam)
                    sc = _objective(cand)
                    top_candidates.append((sc, cand.copy()))
                    if sc < overall_best_score:
                        overall_best_score, overall_best_seq = sc, cand.copy()
                        print(f"[{time.time():.1f}] new global best p={p:.3g} lam={lam:.2f} len={n} score={sc:.6f}")

        # exponential decay family
        for rate in rate_grid:
            if time.time() >= deadline:
                break
            cand = _distribution_exponential_decay(n, rate)
            sc = _objective(cand)
            top_candidates.append((sc, cand.copy()))
            if sc < overall_best_score:
                overall_best_score, overall_best_seq = sc, cand.copy()
                print(f"[{time.time():.1f}] new global best exp‑decay rate={rate:.4f} len={n} score={sc:.6f}")

        # random Dirichlet seeds
        for _ in range(8):
            if time.time() >= deadline:
                break
            a = _distribution_dirichlet(n, rng)
            sc = _objective(a)
            top_candidates.append((sc, a.copy()))
            if sc < overall_best_score:
                overall_best_score, overall_best_seq = sc, a.copy()
                print(f"[{time.time():.1f}] new global best Dirichlet len={n} score={sc:.6f}")

        # sparse random seeds
        for _ in range(8):
            if time.time() >= deadline:
                break
            a = _distribution_random_sparse(n, rng, density=0.008)
            sc = _objective(a)
            top_candidates.append((sc, a.copy()))
            if sc < overall_best_score:
                overall_best_score, overall_best_seq = sc, a.copy()
                print(f"[{time.time():.1f}] new global best sparse len={n} score={sc:.6f}")

        # asymmetric beta samples
        for _ in range(5):
            if time.time() >= deadline:
                break
            alpha = 10 ** rng.uniform(-2, 1)   # 0.01 … 10
            beta = 10 ** rng.uniform(-2, 1)
            a = _distribution_beta_general(n, alpha, beta)
            sc = _objective(a)
            top_candidates.append((sc, a.copy()))
            if sc < overall_best_score:
                overall_best_score, overall_best_seq = sc, a.copy()
                print(f"[{time.time():.1f}] new global best beta(α={alpha:.3g},β={beta:.3g}) len={n} score={sc:.6f}")

        # keep only the top ~30 seeds overall
        top_candidates.sort(key=lambda x: x[0])
        top_candidates = top_candidates[:30]

    # ------------------------------------------------------------------
    # 2. Incorporate a previously discovered construction if present
    try:
        prev = np.asarray(height_sequence_1, dtype=np.float64)   # type: ignore[name-defined]
        if prev.ndim == 1 and prev.size > 0:
            prev = prev / prev.sum()
            sc = _objective(prev)
            if sc < overall_best_score:
                overall_best_score, overall_best_seq = sc, prev.copy()
                print(f"[{time.time():.1f}] previous best score={sc:.6f}")
            top_candidates.append((sc, prev.copy()))
    except Exception:
        pass

    # ------------------------------------------------------------------
    # 3. Guard against the unlikely case that we have no seed
    if overall_best_seq is None:
        n = 1500
        overall_best_seq = _distribution_uniform(n)
        overall_best_score = _objective(overall_best_seq)
        print(f"[{time.time():.1f}] fallback uniform score={overall_best_score:.6f}")

    # ------------------------------------------------------------------
    # 4. LP directional improvement (if possible) on a modest‑size seed
    # Pick the best candidate whose length does not exceed 1500 (to keep LP cheap)
    max_n_lp = 1500
    lp_start = time.time()
    lp_candidates = [cand for (sc, cand) in top_candidates if cand.shape[0] <= max_n_lp]
    if lp_candidates:
        # take the one with the smallest objective among them
        lp_base = min(lp_candidates, key=lambda a: _objective(a))
    else:
        # fall back to the overall best (even if large) – we will skip LP
        lp_base = overall_best_seq

    # allocate roughly 30 % of remaining budget to LP stage
    remaining = deadline - time.time()
    lp_deadline = time.time() + min(remaining * 0.30, 300.0)  # cap at 5 min
    if _SCIPY and lp_base.shape[0] <= max_n_lp:
        print(f"[{time.time():.1f}] starting LP improvement (len={lp_base.shape[0]})")
        overall_best_seq, overall_best_score = _lp_iterative_improve(
            lp_base.copy(),
            lp_deadline,
            best_seq=overall_best_seq,
            best_score=overall_best_score,
        )
        print(f"[{time.time():.1f}] LP stage finished, best score {overall_best_score:.6f}")
    else:
        print(f"[{time.time():.1f}] skipping LP stage (size {lp_base.shape[0]})")

    # ------------------------------------------------------------------
    # 5. Long mirror‑descent on the current overall best (bulk of the budget)
    if time.time() < deadline:
        remaining = deadline - time.time()
        # keep ~30 s buffer for final polishing
        sub_deadline = time.time() + max(0.0, remaining - 30.0)
        print(f"[{time.time():.1f}] start mirror‑descent, approx {remaining-30:.1f}s left")
        overall_best_seq, overall_best_score = _subgradient_descent_momentum(
            overall_best_seq.copy(),
            sub_deadline,
            best_seq=overall_best_seq,
            best_score=overall_best_score,
            lr0=0.8,
            active_frac=0.025,
            max_active=30,
            beta=0.9,
            eval_interval=200,
            patience=3500,
        )
        print(f"[{time.time():.1f}] after mirror‑descent best score {overall_best_score:.6f}")

    # ------------------------------------------------------------------
    # 6. Greedy pairwise refinement
    if time.time() < deadline:
        print(f"[{time.time():.1f}] greedy pairwise refinement")
        overall_best_seq, overall_best_score = _pairwise_swap_local(
            overall_best_seq.copy(),
            deadline,
            best_seq=overall_best_seq,
            best_score=overall_best_score,
            max_trials=300_000,
        )

    # ------------------------------------------------------------------
    # 7. Final simulated‑annealing style local search
    if time.time() < deadline:
        print(f"[{time.time():.1f}] final simulated‑annealing search")
        overall_best_seq, overall_best_score = _random_local_search(
            overall_best_seq.copy(),
            deadline,
            best_seq=overall_best_seq,
            best_score=overall_best_score,
            init_temp=0.01,
            decay=0.9995,
        )

    # ------------------------------------------------------------------
    print(f"Final best score (exact) ≈ {overall_best_score:.6f} (length={len(overall_best_seq)})")
    return overall_best_seq.tolist()
