def _distribution_exponential_edge(n: int, decay: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)

    w_sum = w.sum()
    if w_sum > 0:

    else:

#!/usr/bin/env python
"""
Propose a step‑function (list of non‑negative heights) that minimises the
objective used in ``evaluate_sequence``.

The algorithm consists of three phases within the allotted time budget:

1. **Extensive shape scan** – generate a large variety of analytic
   families (uniform, arcsine, sine, cosine, quadratic,
   edge‑weighted, symmetric and asymmetric Beta, power‑sum,
   exponential‑edge, random Dirichlet and sparse seeds).  All candidates
   are normalised to sum = 1 and evaluated with a fast FFT‑based
   convolution.  The best ≈ 30 candidates are retained.

2. **Intensive mirror‑descent refinement** – the retained seeds are
   refined with exponentiated‑gradient descent on the exact
   ``max(conv)`` objective.  The exact sub‑gradient of the active
   convolution entries is used, the iterates are projected onto the
   probability simplex, and the step‑size decays as η₀/√t.  This phase
   consumes the bulk of the time budget and aggressively lowers the
   upper bound.

3. **Final greedy polishing** – a hill‑climbing routine repeatedly
   moves a random fraction of mass from one index to another, only
   keeping strictly improving moves.  It runs until time expires.

All work respects the 1000 s limit; the returned list is ready for the
external ``evaluate_sequence`` call (no further normalisation needed).
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Euclidean projection of ``v`` onto the simplex {x≥0, Σx=s}."""
    if s < 0:
        raise ValueError("Simplex radius must be non‑negative")
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.where(u - cssv / (np.arange(1, len(v) + 1)) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    return np.maximum(v - theta, 0.0)


# ----------------------------------------------------------------------
def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Full linear self‑convolution ``a*a`` using a real FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1) - 1).bit_length()     # next power of two
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[: 2 * n - 1]


# ----------------------------------------------------------------------
def _objective(a: np.ndarray) -> float:
    """Exact objective ``2·n·max(conv(a,a))`` for a vector that sums to 1."""
    n = a.shape[0]
    b = _fft_self_convolve(a)
    return float(2.0 * n * b.max())

# Analytic families ----------------------------------------------------
def _distribution_uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _distribution_arcsine(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    vals /= vals.sum()
    return vals


def _distribution_sine(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64) + 0.5
    w = np.sin(np.pi * i / n)
    w /= w.sum()
    return w


def _distribution_cosine(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64) + 0.5
    w = 1.0 - np.cos(np.pi * i / n)
    w /= w.sum()
    return w


def _distribution_quadratic(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    x = (2.0 * i) / (n - 1) - 1.0
    w = 1.0 - x * x
    w = np.clip(w, 0.0, None)
    w /= w.sum()
    return w


def _distribution_edge_weighted(n: int, p: float) -> np.ndarray:
    """
    Edge‑weighted family: w_i ∝ ((i+½)*(n−i−½))^{−p}.
    """
    i = np.arange(n, dtype=np.float64) + 0.5
    j = (n - np.arange(n, dtype=np.float64) - 0.5)
    w = (i * j) ** (-p)
    w = np.clip(w, 0.0, None)
    total = w.sum()
    if total > 0:
        w /= total
    else:
        w = np.full(n, 1.0 / n)
    return w


def _distribution_beta_sym(n: int, p: float) -> np.ndarray:
    """Symmetric Beta(p,p) on a regular grid."""
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    w = np.exp((p - 1) * np.log(x) + (p - 1) * np.log(1.0 - x))
    if w.sum() == 0.0 or np.isnan(w).any():
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _distribution_beta_general(n: int, p: float, q: float) -> np.ndarray:
    """Beta(p,q) on a regular grid (both p,q > 0)."""
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    # use logs to avoid overflow
    logw = (p - 1) * np.log(x) + (q - 1) * np.log(1.0 - x)
    w = np.exp(logw)
    if w.sum() == 0.0 or np.isnan(w).any():
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _distribution_power_sum(n: int, p: float) -> np.ndarray:
    """w_i ∝ (i+1)^{−p} + (n−i)^{−p}."""
    i = np.arange(1, n + 1, dtype=np.float64)
    w = i ** (-p) + (n + 1 - i) ** (-p)
    w = np.clip(w, 0.0, None)
    w /= w.sum()
    return w


def _distribution_exponential_edge(n: int, decay: float) -> np.ndarray:
    """
    Edge‑heavy exponential family:
        w_i ∝ exp(decay * |i - centre|)
    where centre = (n-1)/2.
    """
    i = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    dist = np.abs(i - centre)
    w = np.exp(decay * dist)
    w = np.clip(w, 0.0, None)
    total = w.sum()
    if total > 0:
        w /= total
    else:
        w = np.full(n, 1.0 / n)
    return w


def _distribution_random_sparse(
    n: int, rng: np.random.Generator, density: float = 0.01
) -> np.ndarray:
    """Sparse random vector with a prescribed density of non‑zeros."""
    k = max(2, int(n * density))
    idx = rng.choice(n, size=k, replace=False)
    vals = np.zeros(n, dtype=np.float64)
    vals[idx] = rng.random(k)
    if vals.sum() == 0.0:
        vals[0] = 1.0
    vals /= vals.sum()
    return vals


def _mix_with_uniform(base: np.ndarray, lam: float) -> np.ndarray:
    """Convex combination λ·uniform + (1‑λ)·base."""
    n = base.shape[0]
    uniform = np.full(n, 1.0 / n, dtype=np.float64)
    return lam * uniform + (1.0 - lam) * base


# ----------------------------------------------------------------------
# Sub‑gradient of the max‑convolution ----------------------------------
def _max_convolution_subgradient(a: np.ndarray) -> np.ndarray:
    """
    Sub‑gradient of  max_k Σ_i a_i a_{k-i}.
    Returns a vector of the same shape as ``a``.
    """
    conv = _fft_self_convolve(a)
    max_val = conv.max()
    tol = 1e-12 * max_val
    active = np.where(conv >= max_val - tol)[0]
    n = a.shape[0]
    grad = np.zeros_like(a)
    for k in active:
        lo = max(0, k - (n - 1))
        hi = min(n - 1, k)
        idx = np.arange(lo, hi + 1)
        grad[idx] += 2.0 * a[k - idx]
    if len(active) > 0:
        grad /= len(active)
    return grad

# Mirror (exponentiated) gradient descent -----------------------------
def _exp_gradient_descent(
    a0: np.ndarray,
    time_limit: float,
    best_seq: np.ndarray,
    best_score: float,
    eta0: float = 0.8,
    report_every: int = 500,
    max_iter: int = 0,
) -> tuple[np.ndarray, float]:
    """
    Exponentiated‑gradient descent on the exact max‑convolution objective.
    Operates until ``time_limit`` is reached (or ``max_iter`` if non‑zero).
    """
    a = a0.copy()
    n = a.shape[0]
    t = 0
    while time.time() < time_limit:
        if max_iter and t >= max_iter:
            break
        t += 1
        grad = _max_convolution_subgradient(a)
        eta = eta0 / math.sqrt(t)
        a = a * np.exp(-eta * grad)
        a_sum = a.sum()
        if a_sum > 0:
            a = a / a_sum
        else:
            a = np.full(n, 1.0 / n, dtype=np.float64)

        if t % report_every == 0:
            cur = _objective(a)
            if cur < best_score - 1e-14:
                best_score = cur
                best_seq = a.copy()
                print(f"[{time.time():.1f}] exp‑gd it={t} new best {best_score:.6f}")
    return best_seq, best_score

def _pairwise_swap(
    a0: np.ndarray,
    time_limit: float,
    best_seq: np.ndarray,
    best_score: float,
    max_trials: int | None = None,
) -> tuple[np.ndarray, float]:
    """
    Randomly move a fraction of mass from one index to another.
    Only moves that strictly improve the exact objective are kept.
    """
    a = a0.copy()
    n = a.shape[0]
    trials = 0
    while time.time() < time_limit:
        if max_trials is not None and trials >= max_trials:
            break
        trials += 1

        src = np.where(a > 1e-14)[0]
        if src.size == 0:
            break
        i = np.random.choice(src)
        j = np.random.randint(n - 1)
        if j >= i:
            j += 1

        delta = a[i] * np.random.rand() * 0.45   # up to 45 % of source mass
        if delta < 1e-15:
            continue

        new_a = a.copy()
        new_a[i] -= delta
        new_a[j] += delta

        cur = _objective(new_a)
        if cur < best_score - 1e-14:
            a = new_a
            best_score = cur
            best_seq = a.copy()
            print(
                f"[{time.time():.1f}] swap i={i}→j={j} δ={delta:.6g} new best {best_score:.6f}"
            )
    return best_seq, best_score
