    conv = _fft_self_convolve(a)  # length 2n-1
    maxc = conv.max()
    # subtract maxc for numerical stability
    exp_vals = np.exp(alpha * (conv - maxc))
    w = exp_vals / exp_vals.sum()
    # compute gradient: 2 * (a cross-correlated with w)
    # Use FFT: correlate a with w:
    size = 1 << ((len(a) + len(w) - 1) - 1).bit_length()
    A = np.fft.rfft(a, n=size)
    W = np.fft.rfft(w, n=size)

    offset = len(w) - 1

"""
Propose a near‑optimal step‑function (list of non‑negative heights) for
``evaluate_sequence``.

Algorithm
---------

1. **Broad scan** – generate many analytic candidates on several
   lengths (2048 … 16384).  Families: uniform, quadratic, sine‑power,
   edge‑weighted (w_i ∝ ((i+½)*(n‑i‑½))^{-p}), and mixtures with the
   uniform vector.  The parameter *p* is searched on a dense grid
   (≈150 values) and the best ≈10 seeds are kept.

2. **Local one‑dimensional refinement** – for each retained seed the
   shape parameter *p* (or the mixing weight λ) is refined by a tiny
   coordinate‑descent loop that evaluates the exact objective
   ``2·n·max(conv)`` (the sum is normalized to 1, so this matches
   ``evaluate_sequence`` up to the clipping performed inside the
   evaluator).

3. **Smoothed Adam optimisation** – starting from the best refined
   seed we minimise a smooth approximation of the max‑convolution:

       φ(a) = 2·n· (1/α)·log Σ_k exp(α·(a*a)[k])

   with α increased from 5 to 80 while the optimisation proceeds.
   The gradient of φ is computed via FFT‑based cross‑correlation.
   Adam (additive updates) is used together with a Euclidean projection
   onto the simplex {x≥0, Σx = 1}.  The optimiser runs until the time
   budget is exhausted.

4. **Greedy pairwise polishing** – random moves that transfer a
   fraction of mass from one index to another are accepted only when
   they strictly improve the exact objective.  This hill‑climbing
   phase finishes the search.

The routine is careful with the 1000 s budget, prints progress, and
finally returns a plain ``list[float]`` suitable for ``evaluate_sequence``.
"""

# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Euclidean projection of ``v`` onto the simplex {x≥0, Σx=s}."""
    if s < 0:
        raise ValueError("simplex radius must be non‑negative")
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.where(u - cssv / (np.arange(1, len(v) + 1)) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    return np.maximum(v - theta, 0.0)


# ----------------------------------------------------------------------
def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Full self‑convolution a*a using a real FFT (length 2·n−1)."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1) - 1).bit_length()   # next power of two
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[: 2 * n - 1]


# ----------------------------------------------------------------------
def _objective(a: np.ndarray) -> float:
    """
    Exact objective 2·n·max(conv(a,a)) for a vector whose entries sum to 1.
    """
    n = a.shape[0]
    b = _fft_self_convolve(a)
    return float(2.0 * n * b.max())

# Analytic families ----------------------------------------------------
def _distribution_uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _distribution_quadratic(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    x = (2.0 * i) / (n - 1) - 1.0
    w = 1.0 - x * x
    w = np.clip(w, 0.0, None)
    w /= w.sum()
    return w


def _distribution_sine_power(n: int, p: float) -> np.ndarray:
    """w_i ∝ sin(π·(i+½)/n)^{p}."""
    i = np.arange(n, dtype=np.float64) + 0.5
    w = np.sin(np.pi * i / n) ** p
    w = np.clip(w, 0.0, None)
    w /= w.sum()
    return w


def _distribution_edge_weighted(n: int, p: float) -> np.ndarray:
    """
    Edge‑weighted family: w_i ∝ ((i+½)*(n‑i‑½))^{‑p}.
    """
    i = np.arange(n, dtype=np.float64) + 0.5
    j = (n - np.arange(n, dtype=np.float64) - 0.5)
    w = (i * j) ** (-p)
    w = np.clip(w, 0.0, None)
    total = w.sum()
    if total > 0.0:
        w /= total
    else:
        w = np.full(n, 1.0 / n)
    return w


def _mix_with_uniform(base: np.ndarray, lam: float) -> np.ndarray:
    """Convex combination λ·uniform + (1‑λ)·base."""
    n = base.shape[0]
    uniform = np.full(n, 1.0 / n, dtype=np.float64)
    return lam * uniform + (1.0 - lam) * base

# Smoothed objective utilities ------------------------------------------
def _gradient_logsumexp(a: np.ndarray, alpha: float) -> np.ndarray:
    """
    Gradient of   2·n·(1/α)·log Σ_k exp(α·(a*a)[k])   with respect to a.
    Returns a vector of shape a.shape.
    """
    n = a.shape[0]
    conv = _fft_self_convolve(a)                     # shape 2n‑1
    maxc = conv.max()
    # softmax weights (stable)
    exp_vals = np.exp(alpha * (conv - maxc))
    w = exp_vals / exp_vals.sum()                    # shape 2n‑1, Σ w = 1

    # correlation a ⋆ w  (lag ℓ = i) gives Σ_j a_j w_{i+j}
    L = w.shape[0]                                   # 2·n‑1
    size = 1 << ((n + L - 1) - 1).bit_length()
    A = np.fft.rfft(a, n=size)
    W = np.fft.rfft(w, n=size)
    corr_full = np.fft.irfft(A * np.conj(W), n=size)[: n + L - 1]
    offset = L - 1
    corr_i = corr_full[offset: offset + n]           # length n

    # d/d a of 2·n·log‑sum‑exp = 4·n·corr_i
    grad = 4.0 * n * corr_i
    return grad

# Adam optimiser --------------------------------------------------------
def _adam_optimize(
    a0: np.ndarray,
    start_time: float,
    deadline: float,
    best_seq: np.ndarray,
    best_score: float,
    lr: float = 0.12,
    alpha_start: float = 5.0,
    alpha_end: float = 80.0,
    beta1: float = 0.9,
    beta2: float = 0.999,
    eps: float = 1e-8,
    report_every: int = 4000,
) -> tuple[np.ndarray, float]:
    """
    Adam optimisation of the smoothed max‑convolution objective.
    The iterate is projected onto the probability simplex after each step.
    """
    a = a0.copy()
    n = a.shape[0]
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    it = 0
    while time.time() < deadline and it < 300_000:
        it += 1
        elapsed = time.time() - start_time
        total = deadline - start_time
        t_frac = elapsed / total if total > 0 else 0.0
        alpha = alpha_start + t_frac * (alpha_end - alpha_start)

        grad = _gradient_logsumexp(a, alpha)

        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)

        # bias‑correction (optional – improves early steps)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        step = lr * m_hat / (np.sqrt(v_hat) + eps)
        a = a - step

        # projection onto the simplex
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s > 0.0:
            a = a / s
        else:
            a = np.full(n, 1.0 / n, dtype=np.float64)

        if it % report_every == 0:
            cur = _objective(a)
            if cur < best_score - 1e-14:
                best_score = cur
                best_seq = a.copy()
                print(f"[{time.time():.1f}] Adam it={it} new global best {best_score:.6f}")

    return best_seq, best_score

# Greedy pairwise polishing ---------------------------------------------
def _pairwise_swap(
    a0: np.ndarray,
    deadline: float,
    best_seq: np.ndarray,
    best_score: float,
    max_trials: int | None = None,
) -> tuple[np.ndarray, float]:
    """
    Randomly moves a fraction of mass between two coordinates.
    Only moves that strictly improve the exact objective are kept.
    """
    a = a0.copy()
    n = a.shape[0]
    trials = 0
    while time.time() < deadline:
        if max_trials is not None and trials >= max_trials:
            break
        trials += 1
        src = np.where(a > 1e-14)[0]
        if src.size == 0:
            break
        i = int(np.random.choice(src))
        j = np.random.randint(n - 1)
        if j >= i:
            j += 1
        delta = a[i] * np.random.rand() * 0.45
        if delta < 1e-15:
            continue
        new_a = a.copy()
        new_a[i] -= delta
        new_a[j] += delta
        cur = _objective(new_a)
        if cur < best_score - 1e-14:
            a = new_a
            best_score = cur
            best_seq = a.copy()
            print(f"[{time.time():.1f}] swap i={i}→j={j} δ={delta:.6g} new best {best_score:.6f}")
    return best_seq, best_score

        # uniform baseline
        cand = _distribution_uniform(n)
        sc = _objective(cand)
        top_candidates.append((sc, cand.copy()))
        if sc < overall_best_score:
            overall_best_score = sc
            overall_best_seq = cand.copy()
            print(f"[{time.time():.1f}] new global best (uniform) len={n} score={sc:.6f}")

        # quadratic baseline
        cand = _distribution_quadratic(n)
        sc = _objective(cand)
        top_candidates.append((sc, cand.copy()))
        if sc < overall_best_score:
            overall_best_score = sc
            overall_best_seq = cand.copy()
            print(f"[{time.time():.1f}] new global best (quadratic) len={n} score={sc:.6f}")

        # sine‑power family (p as exponent)
        for p in (0.5, 1.0, 2.0, 3.0):
            cand = _distribution_sine_power(n, p)
            sc = _objective(cand)
            top_candidates.append((sc, cand.copy()))
            if sc < overall_best_score:
                overall_best_score = sc
                overall_best_seq = cand.copy()
                print(f"[{time.time():.1f}] new global best (sine^p) len={n} p={p:.2f} score={sc:.6f}")

        # edge‑weighted family with dense p grid
        for p in p_grid:
            if time.time() >= deadline:
                break
            base = _distribution_edge_weighted(n, p)

            # raw edge‑weighted
            sc = _objective(base)
            top_candidates.append((sc, base.copy()))
            if sc < overall_best_score:
                overall_best_score = sc
                overall_best_seq = base.copy()
                print(f"[{time.time():.1f}] new global best (edge) len={n} p={p:.4g} score={sc:.6f}")

            # mixtures with uniform
            for lam in lam_grid:
                if time.time() >= deadline:
                    break
                cand = _mix_with_uniform(base, lam)
                sc = _objective(cand)
                top_candidates.append((sc, cand.copy()))
                if sc < overall_best_score:
                    overall_best_score = sc
                    overall_best_seq = cand.copy()
                    print(f"[{time.time():.1f}] new global best (mix) len={n} p={p:.4g} lam={lam:.2f} score={sc:.6f}")

        # random Dirichlet seeds (few)
        for _ in range(3):
            if time.time() >= deadline:
                break
            alpha = rng.uniform(0.01, 2.0)
            a = rng.dirichlet(np.full(n, alpha))
            sc = _objective(a)
            top_candidates.append((sc, a.copy()))
            if sc < overall_best_score:
                overall_best_score = sc
                overall_best_seq = a.copy()
                print(f"[{time.time():.1f}] Dirichlet seed len={n} score={sc:.6f}")

        # sparse random seeds
        for _ in range(3):
            if time.time() >= deadline:
                break
            a = np.zeros(n, dtype=np.float64)
            k = max(2, int(n * 0.008))
            idx = rng.choice(n, size=k, replace=False)
            a[idx] = rng.random(k)
            if a.sum() == 0.0:
                a[0] = 1.0
            a /= a.sum()
            sc = _objective(a)
            top_candidates.append((sc, a.copy()))
            if sc < overall_best_score:
                overall_best_score = sc
                overall_best_seq = a.copy()
                print(f"[{time.time():.1f}] sparse seed len={n} score={sc:.6f}")

        # keep only the most promising ~30 candidates overall
        top_candidates.sort(key=lambda x: x[0])
        top_candidates = top_candidates[:30]

    # ------------------------------------------------------------------
    # 2 – One‑dimensional refinements (p and λ)
    # ------------------------------------------------------------------
    refined_candidates: list[tuple[float, np.ndarray]] = []
    for score, seq in top_candidates:
        if time.time() >= deadline:
            break
        n = seq.shape[0]
        # If the candidate is exactly an edge‑weighted base, try to refine p
        # Detect by checking if it is a mixture with uniform (lam≈0 or 1)
        # Simplify: always try a local p‑refinement on the edge family.
        # First, recover the p that generated the candidate (if possible) – we
        # will just run a few p refinements starting from the original p.
        # For robustness we just perform a generic p‑refinement using the
        # edge‑weighted family (even if the candidate came from elsewhere).
        # Choose an initial guess: we can estimate p by fitting the shape:
        # Instead we just start from p=0.5 and let the refinement improve.
        # --------------------------------------------------------------
        # Edge‑weighted refinement
        p_guess = 0.5
        p_opt, p_score = _refine_edge_param(n, p_guess)
        refined = _distribution_edge_weighted(n, p_opt)
        refined_score = _objective(refined)
        if refined_score < overall_best_score:
            overall_best_score = refined_score
            overall_best_seq = refined.copy()
            print(f"[{time.time():.1f}] refined edge p={p_opt:.6f} len={n} score={refined_score:.6f}")
        refined_candidates.append((refined_score, refined))

        # Mixture refinement – try mixing the refined edge with uniform
        best_lam = 0.0
        best_mix_score = refined_score
        # Rough scan of λ; then a tiny local search
        for lam in lam_grid:
            mixed = _mix_with_uniform(refined, lam)
            sc = _objective(mixed)
            if sc < best_mix_score:
                best_mix_score = sc
                best_lam = lam
        # further local search around best_lam
        lam_opt, lam_score = _refine_lambda(n, refined, best_lam)
        mixed_opt = _mix_with_uniform(refined, lam_opt)
        mixed_opt_score = _objective(mixed_opt)
        if mixed_opt_score < overall_best_score:
            overall_best_score = mixed_opt_score
            overall_best_seq = mixed_opt.copy()
            print(f"[{time.time():.1f}] refined mix λ={lam_opt:.4f} len={n} score={mixed_opt_score:.6f}")
        refined_candidates.append((mixed_opt_score, mixed_opt))

    # keep the top few for deep optimisation
    refined_candidates.sort(key=lambda x: x[0])
    seed_vectors = [seq for _, seq in refined_candidates[:5]]

    # ------------------------------------------------------------------
    # 3 – Deep Adam optimisation on the best seeds
    # ------------------------------------------------------------------
    remaining = deadline - time.time()
    if remaining > 30.0 and seed_vectors:
        per_seed = remaining / max(1, len(seed_vectors))
        print(f"[{time.time():.1f}] start Adam optimisation on {len(seed_vectors)} seeds")
        for idx, seed in enumerate(seed_vectors):
            if time.time() >= deadline:
                break
            sg_deadline = time.time() + per_seed
            a_opt, sc_opt = _adam_optimize(
                seed,
                start_time=time.time(),
                deadline=sg_deadline,
                best_seq=overall_best_seq,
                best_score=overall_best_score,
                lr=0.13,
                alpha_start=5.0,
                alpha_end=80.0,
                report_every=2000,
            )
            if sc_opt < overall_best_score - 1e-12:
                overall_best_score = sc_opt
                overall_best_seq = a_opt.copy()
                print(
                    f"[{time.time():.1f}] after Adam seed {idx} new global best {overall_best_score:.6f}"
                )

    # ------------------------------------------------------------------
    # 4 – Final greedy pairwise polishing
    # ------------------------------------------------------------------
    if time.time() < deadline:
        print(f"[{time.time():.1f}] final greedy polishing")
        a_opt, sc_opt = _pairwise_swap(
            overall_best_seq.copy(),
            deadline,
            best_seq=overall_best_seq,
            best_score=overall_best_score,
            max_trials=200_000,
        )
        if sc_opt < overall_best_score - 1e-12:
            overall_best_score = sc_opt
            overall_best_seq = a_opt.copy()
            print(f"[{time.time():.1f}] after polishing new best {overall_best_score:.6f}")

    print(
        f"Final best objective (2·n·max(conv)) = {overall_best_score:.6f}, length = {len(overall_best_seq)}"
    )
    # ``evaluate_sequence`` rescales internally, so returning the raw vector is fine.
    return overall_best_seq.tolist()
