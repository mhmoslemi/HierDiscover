"""
Propose a non‑negative step function (list of heights) that minimises
``evaluate_sequence``.

Algorithm
---------

1. **Seed generation** – sample a diverse collection of distributions
   (uniform, arcsine, symmetric Beta, edge‑weighted, sine‑weighted,
   random Dirichlet and random convex mixtures of these families) for
   moderate lengths (≈ 3 k– 5 k).  Keep the best few candidates.

2. **Heavy mirror‑descent** – for each kept seed run an exponentiated‑
   gradient (entropy mirror) descent on the exact non‑smooth objective.
   Only convolution entries whose value is within a small fraction of the
   current maximum are kept in the active set (sub‑gradient of ``max``).
   The learning‑rate decays as 1/√t and a momentum term (β≈0.85‑0.9) smooths
   the updates.  The best result is retained.

3. **Progressive up‑sampling** – linearly interpolate the current best
   sequence to a larger length (up to 25 000 points) and repeat the same
   mirror‑descent.  Larger lengths give a strictly smaller theoretical
   lower bound, often yielding a noticeable improvement.

4. **Local deterministic cleanup** – a greedy pairwise mass‑transfer
   sweep followed by a cheap simulated‑annealing style search removes
   any remaining spikes.

All vectors are projected onto the probability simplex after every
update, guaranteeing non‑negativity and unit sum.  The implementation
uses only NumPy and runs completely self‑contained under the 1000 s
budget.
"""

import time
import math
import numpy as np
from numpy.fft import rfft, irfft


# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Euclidean projection onto the simplex {x≥0, sum(x)=s}."""
    if s < 0:
        raise ValueError("simplex radius must be non‑negative")
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho_arr = np.where(u - cssv / (np.arange(1, len(v) + 1)) > 0)[0]
    if rho_arr.size == 0:
        theta = 0.0
    else:
        rho = rho_arr[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a using real FFT (length 2·n‑1)."""
    n = a.shape[0]
    need = 2 * n - 1
    size = 1 << ((need - 1).bit_length())   # next power of two
    A = rfft(a, n=size)
    conv = irfft(A * A, n=size)
    return conv[:need]


# ----------------------------------------------------------------------
def _objective(a: np.ndarray) -> float:
    """Exact objective 2·n·max(conv(a,a)) for a with sum(a)=1."""
    n = a.shape[0]
    b = _fft_self_convolve(a)
    return 2.0 * n * b.max()

# Analytic families ----------------------------------------------------
def _distribution_uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _distribution_arcsine(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    vals /= vals.sum()
    return vals


def _distribution_beta_sym(n: int, p: float) -> np.ndarray:
    """Symmetric Beta(p,p) on the grid."""
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    logw = (p - 1) * np.log(x) + (p - 1) * np.log(1.0 - x)
    w = np.exp(logw - np.max(logw))
    if w.sum() == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _distribution_beta_asym(n: int, a: float, b: float) -> np.ndarray:
    """Asymmetric Beta(a,b) on the grid."""
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    logw = (a - 1) * np.log(x) + (b - 1) * np.log(1.0 - x)
    w = np.exp(logw - np.max(logw))
    if w.sum() == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _distribution_edge_weighted(n: int, p: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64) + 0.5
    j = (n - np.arange(n, dtype=np.float64) - 0.5)
    w = (i * j) ** (-p)
    if w.sum() == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _distribution_edge_inversepow(n: int, p: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64) + 0.5
    j = (n - np.arange(n, dtype=np.float64) - 0.5)
    w = i ** (-p) + j ** (-p)
    if w.sum() == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _distribution_sine_weighted(n: int, p: float) -> np.ndarray:
    theta = np.pi * (np.arange(n, dtype=np.float64) + 0.5) / n
    w = np.sin(theta) ** p
    if w.sum() == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _distribution_dirichlet(n: int, rng: np.random.Generator) -> np.ndarray:
    alpha = rng.uniform(0.01, 2.0)
    a = rng.dirichlet(np.full(n, alpha))
    return a.astype(np.float64)


def _distribution_random_sparse(
    n: int,
    rng: np.random.Generator,
    density: float = 0.006,
) -> np.ndarray:
    k = max(2, int(n * density))
    idx = rng.choice(n, size=k, replace=False)
    vals = np.zeros(n, dtype=np.float64)
    vals[idx] = rng.random(k)
    if vals.sum() == 0.0:
        vals[0] = 1.0
    vals /= vals.sum()
    return vals


def _mix_with_uniform(base: np.ndarray, lam: float) -> np.ndarray:
    """lam·U + (1‑lam)·base, U uniform."""
    n = base.shape[0]
    uniform = np.full(n, 1.0 / n, dtype=np.float64)
    return lam * uniform + (1.0 - lam) * base


# ----------------------------------------------------------------------
def _subgradient_descent_momentum(
    a: np.ndarray,
    deadline: float,
    best_seq: np.ndarray,
    best_score: float,
    lr0: float = 0.7,
    active_frac: float = 0.035,
    max_active: int = 30,
    beta: float = 0.9,
    eval_interval: int = 120,
    patience: int = 2000,
) -> tuple[np.ndarray, float]:
    """
    Exponentiated‑gradient (entropy mirror) descent on the exact objective.
    The active set consists of all convolution entries whose value lies
    within ``active_frac`` of the current maximum.
    """
    n = a.shape[0]
    t = 0
    no_improve = 0
    momentum = np.zeros_like(a)
    ind_grid = np.arange(n, dtype=np.int32)

    while time.time() < deadline and t < 500_000:
        t += 1

        conv = _fft_self_convolve(a)
        max_c = conv.max()
        thresh = max_c * (1.0 - active_frac)

        act = np.where(conv >= thresh)[0]
        if act.size > max_active:
            # keep the max_active largest values
            top_idx = np.argpartition(conv[act], -max_active)[-max_active:]
            act = act[top_idx]

        # gradient of the max‑convolution (sub‑gradient)
        grad = np.zeros_like(a)
        for k in act:
            shift = k - ind_grid
            mask = (shift >= 0) & (shift < n)
            grad[mask] += 2.0 * a[shift[mask]]

        # momentum and multiplicative update
        momentum = beta * momentum + (1.0 - beta) * grad
        eta = lr0 / math.sqrt(t)
        # avoid overflow in the exponent
        a = a * np.exp(-eta * momentum)

        # project onto simplex (preserves non‑negativity)
        s = a.sum()
        if s == 0.0:
            a = np.full(n, 1.0 / n, dtype=np.float64)
        else:
            a /= s

        if t % eval_interval == 0:
            cur = _objective(a)
            if cur + 1e-14 < best_score:
                best_score, best_seq = cur, a.copy()
                no_improve = 0
                print(f"[{time.time():.1f}] it={t:,} new global best {best_score:.6f}")
            else:
                no_improve += 1

        if no_improve >= patience:
            # small stochastic perturbation to escape plateaux
            jitter = np.exponential(scale=0.004, size=n)
            a = a + jitter
            a = np.maximum(a, 0.0)
            a /= a.sum()
            no_improve = 0

    return best_seq, best_score


# ----------------------------------------------------------------------
def _pairwise_swap_local(
    a: np.ndarray,
    deadline: float,
    best_seq: np.ndarray,
    best_score: float,
    max_trials: int | None = None,
) -> tuple[np.ndarray, float]:
    """
    Transfer a random fraction of mass from one coordinate to another.
    Keep the change only if the objective improves.
    """
    n = a.shape[0]
    trials = 0
    rng = np.random.default_rng()
    while time.time() < deadline:
        if max_trials is not None and trials >= max_trials:
            break
        trials += 1

        src = np.where(a > 1e-12)[0]
        if src.size == 0:
            break
        i = rng.choice(src)

        j = rng.integers(n - 1)
        if j >= i:
            j += 1

        delta = a[i] * rng.random() * 0.33
        if delta < 1e-14:
            continue

        new_a = a.copy()
        new_a[i] -= delta
        new_a[j] += delta

        cur = _objective(new_a)
        if cur + 1e-14 < best_score:
            a = new_a
            best_score = cur
            best_seq = a.copy()
            print(f"[{time.time():.1f}] swap {i}->{j} δ={delta:.3e} new best {best_score:.6f}")

    return best_seq, best_score


# ----------------------------------------------------------------------
def _random_local_search(
    a: np.ndarray,
    deadline: float,
    best_seq: np.ndarray,
    best_score: float,
    init_temp: float = 0.008,
    decay: float = 0.9995,
) -> tuple[np.ndarray, float]:
    """Transfer mass between random positions; Metropolis acceptance."""
    n = a.shape[0]
    rng = np.random.default_rng()
    T = init_temp
    while time.time() < deadline:
        i = rng.integers(n)
        j = rng.integers(n - 1)
        if j >= i:
            j += 1
        delta = a[i] * rng.random() * 0.28
        if delta < 1e-14:
            continue
        new_a = a.copy()
        new_a[i] -= delta
        new_a[j] += delta
        cur = _objective(new_a)
        if cur + 1e-14 < best_score:
            a = new_a
            best_score = cur
            best_seq = a.copy()
            print(f"[{time.time():.1f}] SA improve {i}->{j} Δ={delta:.3e} best {best_score:.6f}")
        else:
            # Metropolis acceptance
            if rng.random() < math.exp(-(cur - best_score) / max(T, 1e-12)):
                a = new_a
        T *= decay
    return best_seq, best_score


# ----------------------------------------------------------------------
def _upsample(seq: np.ndarray, new_len: int) -> np.ndarray:
    """Linear interpolation of a simplex‑vector to a larger length."""
    old_len = seq.shape[0]
    if new_len == old_len:
        return seq.copy()
    x_old = np.linspace(0.0, 1.0, old_len)
    x_new = np.linspace(0.0, 1.0, new_len)
    new_seq = np.interp(x_new, x_old, seq)
    new_seq = np.maximum(new_seq, 0.0)
    new_seq /= new_seq.sum()
    return new_seq
