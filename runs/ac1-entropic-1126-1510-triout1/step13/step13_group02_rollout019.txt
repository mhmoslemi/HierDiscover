def _softmax_gradient(a, conv, alpha):
    # compute softmax weights
    exp_alpha_conv = np.exp(alpha * conv - np.max(alpha * conv)) # subtract max for stability
    w = exp_alpha_conv / exp_alpha_conv.sum()
    # compute grad_local = sum_k w_k * grad_of_conv_k
    grad_local = np.zeros_like(a)
    n = len(a)
    # compute contributions
    for k, wk in enumerate(w):
        if wk == 0:
            continue
        if k < n:
            grad_local[:k+1] += 2 * wk * a[k::-1]
        else:
            start = k - (n - 1)
            grad_local[start:] += 2 * wk * a[start:][::-1]
    return grad_local

"""
propose_candidate – improved step‑function search

The algorithm consists of four stages that complement each other
and stay well inside the 1000 s budget.

1. **Linear‑programming “capacity” step (tiny length).**
   For n≈512 we solve the LP  
       maximise Σ g_i  subject to (f∗g)_k ≤ M, g≥0,
   where M = max(f∗f).  The solution spreads the mass more evenly.
   The LP is solved with ``scipy.optimize.linprog`` (high‑s) and the
   result is blended into the current f (τ≈0.5).  A handful of iterations
   quickly moves the distribution towards the optimal shape.

2. **Upsampling.**
   The obtained small‑n vector is linearly interpolated to a medium size
   (≈4000–8000) which gives the algorithm enough resolution while keeping
   convolutions cheap.

3. **Bundle exponentiated‑gradient descent.**
   Instead of a sub‑gradient of the single maximal convolution entry we
   use the sum of gradients of the *K* largest entries (K≈8).  This
   smoothes the descent direction and reduces several peaks of the
   autocorrelation simultaneously.  The update is
       a ← a·exp(−η·grad) → simplex,
   with a step size η = η₀ / √t (η₀≈0.9).  The best vector found so far
   is kept globally.

4. **Final simulated‑annealing polish.**
   A cheap pair‑wise mass‑transfer annealing (as in the reference code)
   escapes any remaining local optimum.

All phases respect the overall time limit; intermediate best scores
are printed for progress monitoring.  The function returns a plain
list of floats ready for ``evaluate_sequence``.
"""

import time
import math
import numpy as np
from scipy import sparse
from scipy.optimize import linprog

# ----------------------------------------------------------------------
# Simplex utilities
# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Euclidean projection onto the simplex {x≥0, Σx=s}."""
    if s < 0:
        raise ValueError("simplex radius must be non‑negative")
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.where(u - cssv / (np.arange(1, len(v) + 1)) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Convolution helpers (FFT based, linear convolution)
# ----------------------------------------------------------------------
def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Full linear self‑convolution a*a using a real FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1) - 1).bit_length()
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[: 2 * n - 1]


# ----------------------------------------------------------------------
# Exact objective (scale‑invariant, Σa=1 ⇒ denominator = 1)
# ----------------------------------------------------------------------
def _objective(a: np.ndarray) -> float:
    """Exact value of ``evaluate_sequence`` for a probability vector."""
    n = a.shape[0]
    b = _fft_self_convolve(a)
    return float(2.0 * n * b.max())


def _max_conv(a: np.ndarray) -> float:
    """Maximum entry of the self‑convolution (used in the LP)."""
    return float(_fft_self_convolve(a).max())


# ----------------------------------------------------------------------
# Analytic families (all return a probability vector, Σ=1)
# ----------------------------------------------------------------------
def _distribution_uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _distribution_arcsine(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    vals /= vals.sum()
    return vals


def _distribution_edge_weighted(n: int, p: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64) + 0.5
    j = (n - np.arange(n, dtype=np.float64) - 0.5)
    w = (i * j) ** (-p)
    w /= w.sum()
    return w


def _distribution_hann(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    w = 0.5 - 0.5 * np.cos(2.0 * np.pi * i / (n - 1))
    w /= w.sum()
    return w


def _distribution_generalized_normal(n: int, q: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64) + 0.5
    x = i / n - 0.5
    w = np.exp(- (np.abs(x) / 0.5) ** q)
    w /= w.sum()
    return w


def _mix_with_uniform(base: np.ndarray, lam: float) -> np.ndarray:
    n = base.shape[0]
    uniform = np.full(n, 1.0 / n, dtype=np.float64)
    return lam * uniform + (1.0 - lam) * base


# ----------------------------------------------------------------------
# Gradient helpers
# ----------------------------------------------------------------------
def _grad_of_max_conv(a: np.ndarray) -> np.ndarray:
    """Gradient of the *single* maximal convolution entry."""
    n = a.shape[0]
    conv = _fft_self_convolve(a)
    k = int(conv.argmax())
    grad = np.zeros_like(a)
    if k < n:
        grad[: k + 1] = 2.0 * a[k::-1]
    else:
        start = k - (n - 1)
        grad[start:] = 2.0 * a[start:][::-1]
    return grad


def _grad_of_top_conv(a: np.ndarray, top_k: int = 8) -> np.ndarray:
    """
    Gradient obtained by summing the contributions of the *top_k*
    convolution entries (ties are broken arbitrarily).
    """
    n = a.shape[0]
    conv = _fft_self_convolve(a)
    if top_k >= conv.size:
        idx = np.arange(conv.size)
    else:
        idx = np.argpartition(conv, -top_k)[-top_k:]
    grad = np.zeros_like(a)
    for k in idx:
        if k < n:
            grad[: k + 1] += 2.0 * a[k::-1]
        else:
            start = k - (n - 1)
            grad[start:] += 2.0 * a[start:][::-1]
    return grad


# ----------------------------------------------------------------------
# Exponentiated‑gradient descent (bundle variant)
# ----------------------------------------------------------------------
def _bundle_gradient_descent(
    a: np.ndarray,
    deadline: float,
    best_seq: np.ndarray,
    best_score: float,
    eta0: float = 0.9,
    top_k: int = 8,
    eval_every: int = 40,
) -> tuple[np.ndarray, float]:
    """Mirror‑descent with the gradient of the top‑K convolution entries."""
    t = 0
    while time.time() < deadline:
        t += 1
        grad = _grad_of_top_conv(a, top_k=top_k)
        eta = eta0 / math.sqrt(t)
        a = a * np.exp(-eta * grad)
        a = a / a.sum()
        if t % eval_every == 0:
            cur = _objective(a)
            if cur < best_score - 1e-12:
                best_score = cur
                best_seq = a.copy()
                print(f"[{time.time():.1f}] bundle GD it={t} new best {best_score:.6f}")
    return best_seq, best_score


# ----------------------------------------------------------------------
# Simulated‑annealing style pairwise mass transfers
# ----------------------------------------------------------------------
def _pairwise_annealing(
    a: np.ndarray,
    deadline: float,
    best_seq: np.ndarray,
    best_score: float,
    rng: np.random.Generator,
) -> tuple[np.ndarray, float]:
    """Random pairwise mass transfers with a temperature schedule."""
    n = a.shape[0]
    start_time = time.time()
    total_time = deadline - start_time
    while time.time() < deadline:
        src = rng.choice(np.where(a > 1e-12)[0])
        dst = rng.integers(n - 1)
        if dst >= src:
            dst += 1
        delta = a[src] * rng.random() * 0.4
        if delta <= 1e-16:
            continue
        new_a = a.copy()
        new_a[src] -= delta
        new_a[dst] += delta
        cur = _objective(new_a)

        elapsed = time.time() - start_time
        temperature = max(0.0, 1.0 - elapsed / total_time) * best_score * 0.01

        accept = False
        if cur < best_score - 1e-14:
            accept = True
        elif temperature > 0.0:
            prob = math.exp(-(cur - best_score) / temperature)
            if rng.random() < prob:
                accept = True
        if accept:
            a = new_a
            if cur < best_score - 1e-14:
                best_score = cur
                best_seq = a.copy()
                print(
                    f"[{time.time():.1f}] anneal {src}->{dst} δ={delta:.3e} new best {best_score:.6f}"
                )
    return best_seq, best_score


# ----------------------------------------------------------------------
# Upsampling (linear interpolation while preserving the simplex)
# ----------------------------------------------------------------------
def _upsample(a: np.ndarray, new_n: int) -> np.ndarray:
    """Interpolate a probability vector to a larger length."""
    old_n = a.shape[0]
    if new_n == old_n:
        return a.copy()
    x_old = (np.arange(old_n, dtype=np.float64) + 0.5) / old_n
    x_new = (np.arange(new_n, dtype=np.float64) + 0.5) / new_n
    a_new = np.interp(x_new, x_old, a, left=0.0, right=0.0)
    a_new = np.maximum(a_new, 0.0)
    s = a_new.sum()
    if s == 0.0:
        a_new[:] = 1.0 / new_n
    else:
        a_new /= s
    return a_new


# ----------------------------------------------------------------------
# Linear‑programming “capacity” step (tiny length)
# ----------------------------------------------------------------------
def _build_conv_matrix(f: np.ndarray) -> sparse.csr_matrix:
    """
    Returns the (2n‑1)×n matrix A with A[k,i] = f_{k‑i} whenever
    0 ≤ k‑i < n, otherwise 0.  Used for the LP  (f * g)_k ≤ M.
    """
    n = f.shape[0]
    rows = np.arange(2 * n - 1)[:, None]           # shape (2n‑1,1)
    cols = np.arange(n)                           # shape (n,)
    diff = rows - cols                            # (2n‑1, n)
    mask = (diff >= 0) & (diff < n)
    data = f[diff[mask]]
    row_idx = rows[mask]
    col_idx = cols[mask]
    return sparse.csr_matrix((data, (row_idx, col_idx)), shape=(2 * n - 1, n))


def _lp_improve(f: np.ndarray, tau: float) -> np.ndarray | None:
    """
    One LP–based improvement step.
    Returns the blended vector (1‑τ)·f + τ·g   with Σg = 1,
    or ``None`` if the LP fails.
    """
    n = f.shape[0]
    M = _max_conv(f)                     # the current bound
    A = _build_conv_matrix(f)           # (2n‑1)×n matrix
    b = np.full(2 * n - 1, M, dtype=np.float64)

    # maximise Σ g   ⇔ minimise −Σ g
    c = -np.ones(n, dtype=np.float64)
    bounds = [(0.0, None)] * n

    # use HiGHS (fast, handles dense matrices reasonably for n≈512)
    res = linprog(
        c,
        A_ub=A,
        b_ub=b,
        bounds=bounds,
        method="highs",
        options={"presolve": True},
    )
    if not res.success:
        return None
    g = res.x
    total = g.sum()
    if total <= 0.0:
        return None
    g /= total               # enforce Σg = 1
    new_f = (1.0 - tau) * f + tau * g
    return new_f / new_f.sum()

    if target_n <= cur_n * 2:
        target_n = cur_n * 2
    print(f"[{time.time():.1f}] upsampling {cur_n} → {target_n}")
    up = _upsample(overall_best_seq, target_n)
    up_sc = _objective(up)
    if up_sc < overall_best_score:
        overall_best_score = up_sc
        overall_best_seq = up.copy()
        print(f"[{time.time():.1f}] upsample improved best {up_sc:.6f}")

    # ------------------------------------------------------------------
    # 3.  Bundle exponentiated‑gradient descent (medium length)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        sub_dead = time.time() + (deadline - time.time()) * 0.45
        best_seq, best_score = _bundle_gradient_descent(
            overall_best_seq.copy(),
            sub_dead,
            best_seq=overall_best_seq,
            best_score=overall_best_score,
            eta0=0.9,
            top_k=8,
            eval_every=30,
        )
        if best_score < overall_best_score - 1e-12:
            overall_best_score = best_score
            overall_best_seq = best_seq.copy()
            print(f"[{time.time():.1f}] bundle GD finished best {best_score:.6f}")

    # ------------------------------------------------------------------
    # 4.  Short standard GD passes (polishing)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        sub_dead = time.time() + (deadline - time.time()) * 0.25
        best_seq, best_score = _bundle_gradient_descent(
            overall_best_seq.copy(),
            sub_dead,
            best_seq=overall_best_seq,
            best_score=overall_best_score,
            eta0=0.7,
            top_k=6,
            eval_every=20,
        )
        if best_score < overall_best_score - 1e-12:
            overall_best_score = best_score
            overall_best_seq = best_seq.copy()
            print(f"[{time.time():.1f}] second GD polishing best {best_score:.6f}")

    # ------------------------------------------------------------------
    # 5.  Final annealing phase
    # ------------------------------------------------------------------
    if time.time() < deadline:
        print(f"[{time.time():.1f}] final annealing phase")
        best_seq, best_score = _pairwise_annealing(
            overall_best_seq.copy(),
            deadline,
            best_seq=overall_best_seq,
            best_score=overall_best_score,
            rng=rng,
        )
        if best_score < overall_best_score - 1e-12:
            overall_best_seq = best_seq.copy()
            overall_best_score = best_score
            print(f"[{time.time():.1f}] annealing improved best {best_score:.6f}")

    # ------------------------------------------------------------------
    # 6.  Incorporate any externally supplied construction (if present)
    # ------------------------------------------------------------------
    try:
        prev = np.asarray(height_sequence_1, dtype=np.float64)  # type: ignore[name-defined]
        if prev.ndim == 1:
            prev = prev / prev.sum()
            sc = _objective(prev)
            if sc < overall_best_score:
                overall_best_score = sc
                overall_best_seq = prev.copy()
                print(f"[{time.time():.1f}] previous construction improved best {sc:.6f}")
    except Exception:
        pass

    # ------------------------------------------------------------------
    # 7.  Safety fallback (should never happen)
    # ------------------------------------------------------------------
    if overall_best_seq is None:
        overall_best_seq = _distribution_uniform(2000)
        overall_best_score = _objective(overall_best_seq)
        print(f"[{time.time():.1f}] fallback uniform seq score={overall_best_score:.6f}")

    print(
        f"Final best score (2·n·max(conv)) = {overall_best_score:.6f} "
        f"with length {len(overall_best_seq)}"
    )
    return overall_best_seq.tolist()
