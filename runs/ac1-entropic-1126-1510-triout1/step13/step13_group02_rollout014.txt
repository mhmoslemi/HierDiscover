def _grad_multi(a, top_k=5):
    n = a.shape[0]
    conv = _fft_self_convolve(a)
    # Get indices of top_k values
    if top_k >= conv.shape[0]:
        idxs = np.arange(conv.shape[0])
    else:
        idxs = np.argpartition(conv, -top_k)[-top_k:]
    grad = np.zeros_like(a)
    for k in idxs:
        if k < n:
            grad[:k+1] += 2 * a[k::-1]
        else:
            start = k - (n - 1)
            grad[start:] += 2 * a[start:][::-1]
    # average
    grad /= len(idxs)
    return grad
```

def _exp_gradient_multi(a, deadline, best_seq, best_score,
                      eta0=0.5, top_k=5, eval_every=100):
    n = a.shape[0]
    t = 0
    while time.time() < deadline:
        t += 1
        grad = _grad_multi(a, top_k)
        eta = eta0 / math.sqrt(t)
        a = a * np.exp(-eta * grad)
        s = a.sum()
        if s == 0:
            a = np.full(n, 1.0 / n)
        else:
            a = a / s
        if t % eval_every == 0:
            cur = _objective(a)
            if cur < best_score - 1e-14:
                best_score = cur
                best_seq = a.copy()
                print(f"[{time.time():.1f}] GD(it={t}) new best {best_score:.6f}")
    return best_seq, best_score
```

"""
Improved search for a non‑negative step function minimizing
2·n·max(conv(a,a))/(∑a)².

Approach:
1. Produce a diverse pool of probability vectors from several
   analytic families (uniform, arcsine, beta, edge‑weighted,
   generalized normal, sparse, Dirichlet) for a few sizes.
2. For each seed run a short exponentiated‑gradient optimisation.
   The gradient is the average of the sub‑gradients of the
   *top‑k* largest convolution entries (instead of a single
   active one), which smooths the descent.
3. Run a long optimisation on the current global best.
4. If time permits, up‑sample the best vector, polish it again,
   and finish with a simple pairwise‑mass‑transfer simulated
   annealing.
All work respects the 1000 s budget (2 s safety margin) and
always returns a list of non‑negative floats.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Simplex projection (Euclidean)
# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x≥0, Σx=s}."""
    if s < 0:
        raise ValueError("simplex radius must be non‑negative")
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.where(u - cssv / (np.arange(1, n + 1)) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# FFT based self‑convolution (full linear)
# ----------------------------------------------------------------------
def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Full linear self‑convolution a*a using a real FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1) - 1).bit_length()   # next power of two
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[: 2 * n - 1]


# ----------------------------------------------------------------------
# Exact objective for a probability vector (∑a = 1)
# ----------------------------------------------------------------------
def _objective(a: np.ndarray) -> float:
    """Return 2·n·max(conv(a,a)) for a with sum=1."""
    n = a.shape[0]
    b = _fft_self_convolve(a)
    max_b = b.max()
    return float(2.0 * n * max_b)


# ----------------------------------------------------------------------
# Analytic families (all return a probability vector, sum=1)
# ----------------------------------------------------------------------
def _distribution_uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _distribution_arcsine(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    vals /= vals.sum()
    return vals


def _distribution_hann(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    w = 0.5 - 0.5 * np.cos(2.0 * np.pi * i / (n - 1))
    w_sum = w.sum()
    if w_sum == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w_sum
    return w


def _distribution_edge_weighted(n: int, p: float) -> np.ndarray:
    """Weight ∝ (i·j)^{‑p} where i,j are distances to the two edges."""
    i = np.arange(n, dtype=np.float64) + 0.5
    j = (n - np.arange(n, dtype=np.float64) - 0.5)
    w = (i * j) ** (-p)
    w_sum = w.sum()
    if w_sum == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w_sum
    return w


def _distribution_generalized_normal(n: int, q: float) -> np.ndarray:
    """Density ∝ exp(‑|x‑½|^{q}) on [0,1]."""
    i = np.arange(n, dtype=np.float64) + 0.5
    x = i / n - 0.5
    w = np.exp(- (np.abs(x) / 0.5) ** q)
    w_sum = w.sum()
    if w_sum == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w_sum
    return w


def _distribution_beta(n: int, alpha: float) -> np.ndarray:
    """Symmetric Beta(α,α) density on [0,1] discretised on bins."""
    i = np.arange(n, dtype=np.float64) + 0.5
    x = i / n
    # Beta pdf ∝ x^{α-1} (1-x)^{α-1}
    w = (x ** (alpha - 1.0)) * ((1.0 - x) ** (alpha - 1.0))
    w_sum = w.sum()
    if w_sum == 0.0 or np.isnan(w_sum):
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w_sum
    return w


def _distribution_random_sparse(
    n: int, rng: np.random.Generator, density: float = 0.01
) -> np.ndarray:
    """Sparse random vector with a given density."""
    k = max(2, int(n * density))
    idx = rng.choice(n, size=k, replace=False)
    vals = np.zeros(n, dtype=np.float64)
    vals[idx] = rng.random(k)
    if vals.sum() == 0.0:
        vals[0] = 1.0
    vals /= vals.sum()
    return vals


def _mix_with_uniform(base: np.ndarray, lam: float) -> np.ndarray:
    """Convex combination lam·uniform + (1‑lam)·base."""
    n = base.shape[0]
    uniform = np.full(n, 1.0 / n, dtype=np.float64)
    return lam * uniform + (1.0 - lam) * base


# ----------------------------------------------------------------------
# Sub‑gradient of the *top‑k* active convolution entries
# ----------------------------------------------------------------------
def _grad_multi(a: np.ndarray, top_k: int = 3) -> np.ndarray:
    """
    Compute the average of sub‑gradients belonging to the `top_k`
    largest convolution entries.
    """
    n = a.shape[0]
    conv = _fft_self_convolve(a)
    if top_k >= conv.shape[0]:
        idxs = np.arange(conv.shape[0])
    else:
        idxs = np.argpartition(conv, -top_k)[-top_k:]
    grad = np.zeros_like(a)
    for k in idxs:
        # contribution of entry k
        if k < n:
            # overlap length = k+1
            grad[: k + 1] += 2.0 * a[k::-1]
        else:
            start = k - (n - 1)
            grad[start:] += 2.0 * a[start:][::-1]
    grad /= float(len(idxs))
    return grad


# ----------------------------------------------------------------------
# Exponentiated‑gradient optimiser (uses _grad_multi)
# ----------------------------------------------------------------------
def _exp_gradient_multi(
    a: np.ndarray,
    deadline: float,
    best_seq: np.ndarray,
    best_score: float,
    eta0: float = 0.6,
    top_k: int = 3,
    eval_every: int = 80,
) -> tuple[np.ndarray, float]:
    """
    Mirror‑descent on the exact objective using exponentiated updates.
    The gradient is the average of the `top_k` active sub‑gradients.
    """
    n = a.shape[0]
    t = 0
    while time.time() < deadline:
        t += 1
        grad = _grad_multi(a, top_k=top_k)
        eta = eta0 / math.sqrt(t)
        # multiplicative update stays non‑negative
        a = a * np.exp(-eta * grad)
        s = a.sum()
        if s == 0.0:
            a = np.full(n, 1.0 / n, dtype=np.float64)
        else:
            a /= s

        if t % eval_every == 0:
            cur = _objective(a)
            if cur < best_score - 1e-14:
                best_score = cur
                best_seq = a.copy()
                print(f"[{time.time():.1f}] GD it={t} new global best {best_score:.6f}")
    return best_seq, best_score


# ----------------------------------------------------------------------
# Simple pairwise simulated‑annealing polishing
# ----------------------------------------------------------------------
def _pairwise_annealing(
    a: np.ndarray,
    deadline: float,
    best_seq: np.ndarray,
    best_score: float,
    rng: np.random.Generator,
) -> tuple[np.ndarray, float]:
    """
    Random transfer of a fraction of mass between two bins.
    Accept improvements always; occasional uphill moves by a temperature schedule.
    """
    n = a.shape[0]
    start_time = time.time()
    total_time = max(deadline - start_time, 1e-6)
    while time.time() < deadline:
        # pick a source with non‑trivial mass
        src_candidates = np.where(a > 1e-12)[0]
        if src_candidates.size == 0:
            break
        i = rng.choice(src_candidates)
        # pick a distinct target uniformly
        j = rng.integers(n - 1)
        if j >= i:
            j += 1
        # transfer a random fraction of the source mass
        delta = a[i] * rng.random() * 0.4
        if delta < 1e-16:
            continue
        new_a = a.copy()
        new_a[i] -= delta
        new_a[j] += delta
        cur = _objective(new_a)

        elapsed = time.time() - start_time
        # linear cooling schedule
        temperature = max(0.0, 1.0 - elapsed / total_time) * best_score * 0.02

        accept = False
        if cur < best_score - 1e-14:
            accept = True
        elif temperature > 0.0:
            prob = math.exp(-(cur - best_score) / temperature)
            if rng.random() < prob:
                accept = True
        if accept:
            a = new_a
            if cur < best_score - 1e-14:
                best_score = cur
                best_seq = a.copy()
                print(
                    f"[{time.time():.1f}] anneal i={i}→j={j} δ={delta:.3e} new best {best_score:.6f}"
                )
    return best_seq, best_score


# ----------------------------------------------------------------------
# Upsampling (linear interpolation while preserving the simplex)
# ----------------------------------------------------------------------
def _upsample(a: np.ndarray, new_n: int) -> np.ndarray:
    """Interpolate a probability vector to a larger length."""
    old_n = a.shape[0]
    if new_n == old_n:
        return a.copy()
    x_old = (np.arange(old_n, dtype=np.float64) + 0.5) / old_n
    x_new = (np.arange(new_n, dtype=np.float64) + 0.5) / new_n
    a_new = np.interp(x_new, x_old, a, left=0.0, right=0.0)
    a_new = np.maximum(a_new, 0.0)
    s = a_new.sum()
    if s == 0.0:
        a_new[:] = 1.0 / new_n
    else:
        a_new /= s
    return a_new

    # ------------------------------------------------------------------

        print(f"[{time.time():.1f}] short GD on {len(cand_pool)} seeds (≈{per_candidate:.2f}s each)")

        f"(length={len(overall_best_seq)})"
    )
    return overall_best_seq.tolist()

"""
Improved search for a non‑negative step function minimizing

    2·n·max(conv(a,a)) / (∑a)²

Key ideas
---------

1. **Rich seed pool** – Uniform, arcsine, Hann, edge‑weighted,
   generalized‑normal, symmetric‑Beta, Dirichlet and sparse vectors
   for a few lengths (≈2k–8k).  All are normalised to sum = 1.
2. **Exponentiated‑gradient (mirror descent)** – the gradient is the
   *average* of the sub‑gradients belonging to the `top_k` largest
   convolution entries (instead of a single active entry).  This
   smooths the descent and removes plateaus.
3. **Long polishing** – a longer run on the current global best,
   followed by linear up‑sampling (if time permits) and another short
   polishing.
4. **Final annealing** – simple pairwise mass‑transfer simulated
   annealing to escape any remaining local optimum.

The whole routine respects the 1000 s budget (2 s safety margin) and
always returns a plain ``list[float]`` ready for ``evaluate_sequence``.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Simplex projection (Euclidean)
# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Projection of `v` onto the simplex {x ≥ 0, Σx = s}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.where(u - cssv / (np.arange(1, n + 1)) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# FFT‑based full linear self‑convolution
# ----------------------------------------------------------------------
def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    n = a.shape[0]
    size = 1 << ((2 * n - 1) - 1).bit_length()   # next power of two
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[: 2 * n - 1]


# ----------------------------------------------------------------------
# Exact objective for a probability vector (∑a = 1)
# ----------------------------------------------------------------------
def _objective(a: np.ndarray) -> float:
    n = a.shape[0]
    b = _fft_self_convolve(a)
    return float(2.0 * n * b.max())


# ----------------------------------------------------------------------
# Analytic families (all return a probability vector, sum = 1)
# ----------------------------------------------------------------------
def _distribution_uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _distribution_arcsine(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    vals /= vals.sum()
    return vals


def _distribution_hann(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    w = 0.5 - 0.5 * np.cos(2.0 * np.pi * i / (n - 1))
    w_sum = w.sum()
    if w_sum == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w_sum
    return w


def _distribution_edge_weighted(n: int, p: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64) + 0.5
    j = (n - np.arange(n, dtype=np.float64) - 0.5)
    w = (i * j) ** (-p)
    w_sum = w.sum()
    if w_sum == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w_sum
    return w


def _distribution_generalized_normal(n: int, q: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64) + 0.5
    x = i / n - 0.5
    w = np.exp(- (np.abs(x) / 0.5) ** q)
    w_sum = w.sum()
    if w_sum == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w_sum
    return w


def _distribution_beta(n: int, alpha: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64) + 0.5
    x = i / n
    w = (x ** (alpha - 1.0)) * ((1.0 - x) ** (alpha - 1.0))
    w_sum = w.sum()
    if w_sum == 0.0 or np.isnan(w_sum):
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w_sum
    return w


def _distribution_random_sparse(
    n: int, rng: np.random.Generator, density: float = 0.01
) -> np.ndarray:
    k = max(2, int(n * density))
    idx = rng.choice(n, size=k, replace=False)
    vals = np.zeros(n, dtype=np.float64)
    vals[idx] = rng.random(k)
    if vals.sum() == 0.0:
        vals[0] = 1.0
    vals /= vals.sum()
    return vals


def _mix_with_uniform(base: np.ndarray, lam: float) -> np.ndarray:
    n = base.shape[0]
    uniform = np.full(n, 1.0 / n, dtype=np.float64)
    return lam * uniform + (1.0 - lam) * base


# ----------------------------------------------------------------------
# Gradient built from the `top_k` dominant convolution entries
# ----------------------------------------------------------------------
def _grad_multi(a: np.ndarray, top_k: int = 3) -> np.ndarray:
    """Average sub‑gradient of the `top_k` largest convolution values."""
    n = a.shape[0]
    conv = _fft_self_convolve(a)
    if top_k >= conv.shape[0]:
        idxs = np.arange(conv.shape[0])
    else:
        idxs = np.argpartition(conv, -top_k)[-top_k:]
    grad = np.zeros_like(a)
    for k in idxs:
        if k < n:
            grad[: k + 1] += 2.0 * a[k::-1]
        else:
            start = k - (n - 1)
            grad[start:] += 2.0 * a[start:][::-1]
    grad /= float(len(idxs))
    return grad


# ----------------------------------------------------------------------
# Exponentiated‑gradient optimiser (uses the multi‑gradient)
# ----------------------------------------------------------------------
def _exp_gradient_multi(
    a: np.ndarray,
    deadline: float,
    best_seq: np.ndarray,
    best_score: float,
    eta0: float = 0.6,
    top_k: int = 3,
    eval_every: int = 80,
) -> tuple[np.ndarray, float]:
    """
    Mirror‑descent with multiplicative updates.
    The gradient is the average of the `top_k` active sub‑gradients.
    """
    n = a.shape[0]
    t = 0
    while time.time() < deadline:
        t += 1
        grad = _grad_multi(a, top_k=top_k)
        eta = eta0 / math.sqrt(t)
        a = a * np.exp(-eta * grad)
        s = a.sum()
        if s == 0.0:
            a = np.full(n, 1.0 / n, dtype=np.float64)
        else:
            a /= s

        if t % eval_every == 0:
            cur = _objective(a)
            if cur < best_score - 1e-14:
                best_score = cur
                best_seq = a.copy()
                print(f"[{time.time():.1f}] GD it={t} new global best {best_score:.6f}")
    return best_seq, best_score


# ----------------------------------------------------------------------
# Pairwise mass‑transfer simulated annealing (final polishing)
# ----------------------------------------------------------------------
def _pairwise_annealing(
    a: np.ndarray,
    deadline: float,
    best_seq: np.ndarray,
    best_score: float,
    rng: np.random.Generator,
) -> tuple[np.ndarray, float]:
    n = a.shape[0]
    start_time = time.time()
    total_time = max(deadline - start_time, 1e-6)
    while time.time() < deadline:
        src = np.where(a > 1e-12)[0]
        if src.size == 0:
            break
        i = rng.choice(src)
        j = rng.integers(n - 1)
        if j >= i:
            j += 1
        delta = a[i] * rng.random() * 0.4
        if delta < 1e-16:
            continue
        new_a = a.copy()
        new_a[i] -= delta
        new_a[j] += delta
        cur = _objective(new_a)

        elapsed = time.time() - start_time
        temperature = max(0.0, 1.0 - elapsed / total_time) * best_score * 0.02

        accept = False
        if cur < best_score - 1e-14:
            accept = True
        elif temperature > 0.0:
            prob = math.exp(-(cur - best_score) / temperature)
            if rng.random() < prob:
                accept = True

        if accept:
            a = new_a
            if cur < best_score - 1e-14:
                best_score = cur
                best_seq = a.copy()
                print(
                    f"[{time.time():.1f}] anneal i={i}→j={j} δ={delta:.3e} new best {best_score:.6f}"
                )
    return best_seq, best_score


# ----------------------------------------------------------------------
# Upsample a probability vector while preserving the simplex
# ----------------------------------------------------------------------
def _upsample(a: np.ndarray, new_n: int) -> np.ndarray:
    old_n = a.shape[0]
    if new_n == old_n:
        return a.copy()
    x_old = (np.arange(old_n, dtype=np.float64) + 0.5) / old_n
    x_new = (np.arange(new_n, dtype=np.float64) + 0.5) / new_n
    a_new = np.interp(x_new, x_old, a, left=0.0, right=0.0)
    a_new = np.maximum(a_new, 0.0)
    s = a_new.sum()
    if s == 0.0:
        a_new[:] = 1.0 / new_n
    else:
        a_new /= s
    return a_new

    # 1. Build a diversified pool of seeds (moderate lengths)

                     _distribution_arcsine(n),
                     _distribution_hann(n)):
            sc = _objective(base)
            cand_pool.append((sc, base.copy()))
            if sc < overall_best_score:
                overall_best_score = sc
                overall_best_seq = base.copy()
                print(f"[{time.time():.1f}] new best (det) score={sc:.6f}")

        # edge‑weighted (fine grid on p) + mixing with uniform
        p_vals = np.concatenate((
            np.logspace(-3.0, -0.2, 15, base=10),
            np.linspace(0.2, 2.5, 15),
            np.logspace(0.0, 0.7, 10, base=10)
        ))
        lam_vals = np.linspace(0.0, 1.0, 9)
        for p in p_vals:
            if time.time() >= deadline:
                break
            base = _distribution_edge_weighted(n, float(p))
            for lam in lam_vals:
                if time.time() >= deadline:
                    break
                cand = _mix_with_uniform(base, float(lam))
                sc = _objective(cand)
                cand_pool.append((sc, cand.copy()))
                if sc < overall_best_score:
                    overall_best_score = sc
                    overall_best_seq = cand.copy()
                    print(f"[{time.time():.1f}] new best edge p={p:.3g} λ={lam:.2f} score={sc:.6f}")

        # generalized normal family
        for q in np.linspace(0.2, 5.0, 18):
            if time.time() >= deadline:
                break
            cand = _distribution_generalized_normal(n, float(q))
            sc = _objective(cand)
            cand_pool.append((sc, cand.copy()))
            if sc < overall_best_score:
                overall_best_score = sc
                overall_best_seq = cand.copy()
                print(f"[{time.time():.1f}] new best gn q={q:.2f} score={sc:.6f}")

        # symmetric Beta family
        for alpha in np.logspace(-0.7, 0.7, 14):
            if time.time() >= deadline:
                break
            cand = _distribution_beta(n, float(alpha))
            sc = _objective(cand)
            cand_pool.append((sc, cand.copy()))
            if sc < overall_best_score:
                overall_best_score = sc
                overall_best_seq = cand.copy()
                print(f"[{time.time():.1f}] new best beta α={alpha:.3g} score={sc:.6f}")

        # random Dirichlet seeds
        for _ in range(12):
            if time.time() >= deadline:
                break
            log_alpha = rng.uniform(-2.0, 1.0)
            alpha = 10 ** log_alpha
            cand = rng.dirichlet(np.full(n, alpha))
            sc = _objective(cand)
            cand_pool.append((sc, cand.copy()))
            if sc < overall_best_score:
                overall_best_score = sc
                overall_best_seq = cand.copy()
                print(f"[{time.time():.1f}] new best Dirichlet α={alpha:.3g} score={sc:.6f}")

        # sparse random seeds
        for _ in range(8):
            if time.time() >= deadline:
                break
            cand = _distribution_random_sparse(n, rng, density=0.02)
            sc = _objective(cand)
            cand_pool.append((sc, cand.copy()))
            if sc < overall_best_score:
                overall_best_score = sc
                overall_best_seq = cand.copy()
                print(f"[{time.time():.1f}] new best sparse score={sc:.6f}")

        # keep only the top candidates (memory bound)
        cand_pool.sort(key=lambda x: x[0])
        cand_pool = cand_pool[:40]

    # 2. Incorporate a previously found construction if present

            prev = prev / prev.sum()
            sc = _objective(prev)
            if sc < overall_best_score:
                overall_best_score = sc
                overall_best_seq = prev.copy()
                print(f"[{time.time():.1f}] previous best score={sc:.6f}")
            cand_pool.append((sc, prev.copy()))
    except Exception:
        pass

    # ------------------------------------------------------------------
    # 3. Ensure we have a non‑empty result
    # ------------------------------------------------------------------
    if overall_best_seq is None:
        n = 4000
        overall_best_seq = _distribution_uniform(n)
        overall_best_score = _objective(overall_best_seq)
        print(f"[{time.time():.1f}] fallback uniform score={overall_best_score:.6f}")

    # ------------------------------------------------------------------
    # 4. Short gradient polishing on all seeds
    # ------------------------------------------------------------------
    remaining = deadline - time.time()
    if remaining > 25.0 and cand_pool:
        per_candidate = max(0.2, remaining * 0.25 / max(len(cand_pool), 1))
        print(f"[{time.time():.1f}] short GD on {len(cand_pool)} seeds (≈{per_candidate:.2f}s each)")

            if time.time() >= deadline:

                sub_dead,
                best_seq=overall_best_seq,
                best_score=overall_best_score,

            )

                overall_best_score = sc_opt
                overall_best_seq = a_opt.copy()
                print(f"[{time.time():.1f}] after short GD seed {idx} new best {overall_best_score:.6f}")

    # 5. Long optimisation on the current global best
    # ------------------------------------------------------------------

        print(f"[{time.time():.1f}] long GD on best vector (len={overall_best_seq.shape[0]})")
        a_opt, sc_opt = _exp_gradient_multi(
            overall_best_seq.copy(),
            sub_dead,
            best_seq=overall_best_seq,
            best_score=overall_best_score,
            eta0=0.6,
            top_k=5,
            eval_every=80,
        )
        if sc_opt < overall_best_score - 1e-12:
            overall_best_score = sc_opt
            overall_best_seq = a_opt.copy()
            print(f"[{time.time():.1f}] long GD improved best to {overall_best_score:.6f}")

    # ------------------------------------------------------------------
    # 6. Upsample & polish (if time permits)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        cur_n = overall_best_seq.shape[0]
        target_n = cur_n * 2
        if target_n <= 64000:
            print(f"[{time.time():.1f}] upsampling {cur_n} → {target_n}")
            up = _upsample(overall_best_seq, target_n)
            up_sc = _objective(up)
            if up_sc < overall_best_score:
                overall_best_seq = up
                overall_best_score = up_sc
                print(f"[{time.time():.1f}] upsampled new best {overall_best_score:.6f}")

            # short GD on the upsampled vector
            sub_dead = time.time() + (deadline - time.time()) * 0.25
            a_opt, sc_opt = _exp_gradient_multi(
                overall_best_seq.copy(),
                sub_dead,
                best_seq=overall_best_seq,
                best_score=overall_best_score,
                eta0=0.5,
                top_k=5,
                eval_every=60,
            )
            if sc_opt < overall_best_score - 1e-12:
                overall_best_seq = a_opt.copy()
                overall_best_score = sc_opt
                print(f"[{time.time():.1f}] GD after upsample new best {overall_best_score:.6f}")

    # ------------------------------------------------------------------
    # 7. Final annealing phase
    # ------------------------------------------------------------------
    if time.time() < deadline:
        print(f"[{time.time():.1f}] final annealing phase")
        a_opt, sc_opt = _pairwise_annealing(
            overall_best_seq.copy(),
            deadline,
            best_seq=overall_best_seq,
            best_score=overall_best_score,
            rng=rng,
        )
        if sc_opt < overall_best_score - 1e-12:
            overall_best_seq = a_opt.copy()
            overall_best_score = sc_opt
            print(f"[{time.time():.1f}] after annealing new best {overall_best_score:.6f}")

    # ------------------------------------------------------------------
    print(
        f"Final best score (objective=2·n·max(conv)): {overall_best_score:.6f} "
        f"(length={len(overall_best_seq)})"
    )
    return overall_best_seq.tolist()
