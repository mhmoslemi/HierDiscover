def _fft_convolve(u, v):
    n = len(u) + len(v) - 1
    size = 1 << ((n) - 1).bit_length()
    Uf = np.fft.rfft(u, n=size)
    Vf = np.fft.rfft(v, n=size)
    conv = np.fft.irfft(Uf * Vf, n=size)
    return conv[:n]
```

"""
Soft‑max Adam optimizer for the step‑function problem.

The evaluation to minimise is

    F(a) = 2·n·max_k (a * a)[k] / (∑ a_i)² .

Since the objective is homogeneous, we may normalise the sequence to
∑ a_i = 1 and simply minimise 2·n·max(conv(a,a)).  The code below
searches over a handful of analytic families to obtain promising
initial seeds, then refines every seed with a smooth approximation of
the “max’’ (a soft‑max) and a full‑gradient Adam optimiser (the
gradient is obtained analytically via a convolution).  The optimiser
works directly on the logits y with a = softmax(y); this guarantees
non‑negativity and the simplex constraint.  After the Adam phase a
short greedy pairwise hill‑climb is performed for final polishing.

The routine respects the supplied time budget (default 1000 s) and
returns a plain python list of non‑negative floats.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Euclidean projection of ``v`` onto the simplex {x≥0, Σx=s}."""
    if s < 0:
        raise ValueError("Simplex radius must be non‑negative")
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.where(u - cssv / (np.arange(1, len(v) + 1)) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    return np.maximum(v - theta, 0.0)


# ----------------------------------------------------------------------
def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Full self‑convolution a*a using a real FFT (length 2·n‑1)."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1) - 1).bit_length()     # next power of two
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[: 2 * n - 1]


def _fft_convolve(u: np.ndarray, v: np.ndarray) -> np.ndarray:
    """Linear convolution of two real vectors using FFT."""
    n = len(u) + len(v) - 1
    size = 1 << ((n) - 1).bit_length()
    Uf = np.fft.rfft(u, n=size)
    Vf = np.fft.rfft(v, n=size)
    conv = np.fft.irfft(Uf * Vf, n=size)
    return conv[:n]


# ----------------------------------------------------------------------
def _objective(a: np.ndarray) -> float:
    """Exact objective 2·n·max(conv(a,a)) for a with Σa = 1."""
    n = a.shape[0]
    b = _fft_self_convolve(a)
    return float(2.0 * n * b.max())


# ----------------------------------------------------------------------
# Analytic families (same as in the previous solution, kept for diversity)
def _distribution_uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _distribution_arcsine(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    vals /= vals.sum()
    return vals


def _distribution_sine(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64) + 0.5
    w = np.sin(np.pi * i / n)
    w /= w.sum()
    return w


def _distribution_cosine(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64) + 0.5
    w = 1.0 - np.cos(np.pi * i / n)
    w /= w.sum()
    return w


def _distribution_quadratic(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    x = (2.0 * i) / (n - 1) - 1.0
    w = 1.0 - x * x
    w = np.clip(w, 0.0, None)
    w /= w.sum()
    return w


def _distribution_beta_sym(n: int, p: float) -> np.ndarray:
    """Symmetric Beta(p,p) on a regular grid."""
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    w = np.exp((p - 1) * np.log(x) + (p - 1) * np.log(1.0 - x))
    if w.sum() == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _distribution_edge_weighted(n: int, p: float) -> np.ndarray:
    """Edge‑weighted family: w_i ∝ ((i+½)*(n‑i‑½))^{‑p}."""
    i = np.arange(n, dtype=np.float64) + 0.5
    j = (n - np.arange(n, dtype=np.float64) - 0.5)
    w = (i * j) ** (-p)
    w = np.clip(w, 0.0, None)
    total = w.sum()
    if total > 0:
        w /= total
    else:
        w = np.full(n, 1.0 / n)
    return w


def _distribution_power_sum(n: int, p: float) -> np.ndarray:
    """w_i ∝ (i+1)^{‑p} + (n‑i)^{‑p}."""
    i = np.arange(1, n + 1, dtype=np.float64)
    w = i ** (-p) + (n + 1 - i) ** (-p)
    w = np.clip(w, 0.0, None)
    w /= w.sum()
    return w


def _distribution_random_sparse(
    n: int, rng: np.random.Generator, density: float = 0.01
) -> np.ndarray:
    """Sparse random vector with prescribed density."""
    k = max(2, int(n * density))
    idx = rng.choice(n, size=k, replace=False)
    vals = np.zeros(n, dtype=np.float64)
    vals[idx] = rng.random(k)
    if vals.sum() == 0.0:
        vals[0] = 1.0
    vals /= vals.sum()
    return vals


def _mix_with_uniform(base: np.ndarray, lam: float) -> np.ndarray:
    """Convex combination λ·uniform + (1‑λ)·base."""
    n = base.shape[0]
    uniform = np.full(n, 1.0 / n, dtype=np.float64)
    return lam * uniform + (1.0 - lam) * base


# ----------------------------------------------------------------------
def _softmax_adam_optimize(
    a0: np.ndarray,
    deadline: float,
    initial_alpha: float = 6.0,
    final_alpha: float = 80.0,
    lr: float = 0.07,
    beta1: float = 0.9,
    beta2: float = 0.999,
    eps: float = 1e-8,
) -> tuple[np.ndarray, float]:
    """
    Adam optimisation on the *smooth* objective

        G(y) = 2·n· (1/α)·log Σ_k exp(α·conv_k)

    where a = softmax(y) and conv = a*a.
    The method returns the best exact (max‑based) objective seen.
    """
    n = a0.shape[0]
    # initialise logits (log‑softmax) – avoid exact zeros
    y = np.log(np.maximum(a0, 1e-15))
    m = np.zeros_like(y)
    v = np.zeros_like(y)

    best_a = a0.copy()
    best_score = _objective(best_a)

    t = 0
    # linearly increase the smoothing parameter α
    while time.time() < deadline:
        t += 1
        # ------------------------------------------------------------------
        # 1) build the probability vector a from logits
        a = np.exp(y)
        total = a.sum()
        if total == 0.0:
            a = _distribution_uniform(n)
        else:
            a /= total

        # ------------------------------------------------------------------
        # 2) exact convolution and exact objective (for tracking)
        conv = _fft_self_convolve(a)
        cur_exact = 2.0 * n * conv.max()
        if cur_exact < best_score - 1e-14:
            best_score = cur_exact
            best_a = a.copy()
            print(f"[{time.time():.1f}]  Adam iter={t}  new best {best_score:.6f}")

        # ------------------------------------------------------------------
        # 3) soft‑max weights – α grows from initial_alpha to final_alpha
        alpha = initial_alpha + (final_alpha - initial_alpha) * min(1.0, t / 2000.0)
        # numeric stabilisation by subtracting the current maximum
        mx = conv.max()
        exp_alpha = np.exp(alpha * (conv - mx))
        w = exp_alpha / exp_alpha.sum()                     # shape (2n‑1,)

        # ------------------------------------------------------------------
        # 4) gradient w.r.t. a:  ∂G/∂a = 4·n·corr(w,a)   (corr = cross‑correlation)
        #    corr(w,a)[p] = Σ_j w_{p+j} a_j = (w * a_rev)[n‑1‑p]
        c = _fft_convolve(w, a[::-1])                      # length 3n‑2
        grad_a = 4.0 * n * c[n - 1 :: -1]                   # length n

        # ------------------------------------------------------------------
        # 5) gradient w.r.t. logits y (softmax chain rule)
        dot = np.dot(a, grad_a)
        grad_y = a * (grad_a - dot)                        # shape (n,)

        # ------------------------------------------------------------------
        # 6) Adam update on y
        m = beta1 * m + (1.0 - beta1) * grad_y
        v = beta2 * v + (1.0 - beta2) * (grad_y * grad_y)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        y -= lr * m_hat / (np.sqrt(v_hat) + eps)

        # keep y numerically stable (optional – prevents overflow)
        if t % 500 == 0:
            max_y = np.max(np.abs(y))
            if max_y > 50.0:
                y = y / max_y * 40.0

    return best_a, best_score


# ----------------------------------------------------------------------
def _pairwise_greedy(a0: np.ndarray, deadline: float, best_score: float) -> tuple[np.ndarray, float]:
    """
    Simple hill‑climbing: move a random fraction of mass from one index
    to another, accepting only strict improvements of the exact objective.
    """
    a = a0.copy()
    n = a.shape[0]
    cur_best = best_score
    best_seq = a.copy()
    attempts = 0
    while time.time() < deadline:
        attempts += 1
        # pick a source with some mass
        src_idx = np.where(a > 1e-12)[0]
        if src_idx.size == 0:
            break
        i = np.random.choice(src_idx)
        # pick a distinct target
        j = np.random.randint(n - 1)
        if j >= i:
            j += 1
        # move up to 45 % of the source mass
        delta = a[i] * np.random.rand() * 0.45
        if delta < 1e-15:
            continue
        new_a = a.copy()
        new_a[i] -= delta
        new_a[j] += delta
        val = _objective(new_a)
        if val < cur_best - 1e-14:
            a = new_a
            cur_best = val
            best_seq = new_a.copy()
            print(f"[{time.time():.1f}]  swap i={i}→j={j} δ={delta:.3e}  new best {cur_best:.6f}")
    return best_seq, cur_best
