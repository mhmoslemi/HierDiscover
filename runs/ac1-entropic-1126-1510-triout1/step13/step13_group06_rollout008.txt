def _adam_descent(a, deadline, best_seq, best_score):
    n = len(a)
    m = np.zeros_like(a)  # first moment
    v = np.zeros_like(a)  # second moment
    beta1 = 0.9; beta2=0.999; eps =1e-8
    t=0
    while time.time() < deadline:
        t+=1
        conv = _fft_self_convolve(a)
        max_cur = conv.max()
        # compute exact objective = 2*n*max_cur
        cur_score = 2.0 * n * max_cur
        if cur_score < best_score - 1e-12:
            best_score = cur_score; best_seq = a.copy()
        # softmax weights for gradient
        w = _softmax_weights(conv, tau)
        grad = _weighted_gradient(a, w)
        # Adam update
        m = beta1 * m + (1-beta1) * grad
        v = beta2 * v + (1-beta2) * (grad**2)
        m_hat = m / (1 - beta1**t)
        v_hat = v / (1 - beta2**t)
        step = eta * m_hat / (np.sqrt(v_hat) + eps)
        a = a - step
        # Ensure non-negativity and project onto simplex
        a = np.maximum(a, 0)
        sum_a = a.sum()
        if sum_a == 0:
            a = np.full(n, 1.0/n)
        else:
            a = a / sum_a
        # increase tau gradually

def _max_subgradient_descent(a, deadline, best_seq, best_score, eta0=0.2):
    n = a.shape[0]
    t=0
    while time.time() < deadline:
        t+=1
        conv = _fft_self_convolve(a)
        max_val = conv.max()
        cur_score = 2.0*n*max_val
        if cur_score < best_score - 1e-14:
            best_score = cur_score
            best_seq = a.copy()
            print(f"[{time.time():.1f}] subgrad it={t} new best {best_score:.6f}")
        # compute gradient g for argmax index
        k = int(conv.argmax())
        # build gradient
        lo = max(0, k - (n-1))
        hi = min(n-1, k)
        # use slice
        idx = np.arange(lo, hi+1, dtype=np.int64)
        g = np.zeros_like(a)
        g[idx] = 2 * a[k - idx]
        # step size
        eta = eta0 / np.sqrt(t)
        a = a - eta * g
        # enforce nonnegative
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s == 0:
            a = np.full_like(a, 1.0/n)
        else:
            a = a / s
    return best_seq, best_score

"""
Propose a low‑score step function.

The algorithm proceeds in three stages:

1. **Seed selection** – a small pool of analytically‑motivated distributions
   (edge‑weighted, symmetric Beta, arcsine) is sampled for several lengths.
   The best (lowest exact objective) seed is kept.

2. **Continuous optimisation** – the chosen seed is refined with a
   *mirrored Adam* optimiser that minimises a soft‑max surrogate of the
   max‑convolution.  Temperature (τ) is increased slowly so the surrogate
   converges to the true maximum.  After each Adam step the iterate is
   projected onto the probability simplex.

3. **Hard‑max subgradient + local moves** – once τ is large the exact
   max index is used to perform sub‑gradient descent (projected on the
   simplex).  Finally deterministic pairwise swaps and a cheap simulated‑
   annealing walk are applied to polish the result.

All steps respect the supplied time budget (default 1000 s).  The returned
list is a plain Python list; `evaluate_sequence` will perform the final
normalisation, clipping, etc.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x≥0, Σx = s}."""
    if s < 0:
        raise ValueError("simplex radius must be non‑negative")
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho_arr = np.where(u - cssv / (np.arange(1, len(v) + 1)) > 0)[0]
    if rho_arr.size == 0:
        theta = 0.0
    else:
        rho = rho_arr[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Full self‑convolution a*a using a real FFT (returns length 2n‑1)."""
    n = a.shape[0]
    need = 2 * n - 1
    size = 1 << ((need - 1).bit_length())
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[:need]


# ----------------------------------------------------------------------
def _objective(a: np.ndarray) -> float:
    """Exact objective 2·n·max(conv(a,a)) for a vector normalised to sum = 1."""
    n = a.shape[0]
    b = _fft_self_convolve(a)
    return float(2.0 * n * b.max())


# ----------------------------------------------------------------------
# Analytic families ----------------------------------------------------
def _distribution_uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _distribution_arcsine(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    vals /= vals.sum()
    return vals


def _distribution_beta_sym(n: int, p: float) -> np.ndarray:
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    w = np.exp((p - 1) * np.log(x) + (p - 1) * np.log(1.0 - x))
    if w.sum() == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _distribution_edge_weighted(n: int, p: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64) + 0.5
    j = (n - np.arange(n, dtype=np.float64) - 0.5)
    w = (i * j) ** (-p)
    if w.sum() == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _mix_with_uniform(base: np.ndarray, lam: float) -> np.ndarray:
    """Convex combination lam·U + (1‑lam)·base, where U is uniform."""
    n = base.shape[0]
    uniform = np.full(n, 1.0 / n, dtype=np.float64)
    return lam * uniform + (1.0 - lam) * base


# ----------------------------------------------------------------------
# Soft‑max utilities ----------------------------------------------------
def _softmax_weights(conv: np.ndarray, tau: float) -> np.ndarray:
    """Soft‑max of τ·conv (stable). Returns a probability vector."""
    max_c = conv.max()
    w_raw = np.exp(tau * (conv - max_c))
    total = w_raw.sum()
    if total == 0.0:
        return np.full_like(conv, 1.0 / conv.size)
    return w_raw / total


def _weighted_gradient(a: np.ndarray, w: np.ndarray) -> np.ndarray:
    """
    Gradient of L(a)=∑_i w_i·(a*a)[i] w.r.t a.
    Closed‑form: ∇L = 2·(a_rev ⊛ w)  (restricted to the first n entries).
    """
    n = a.shape[0]
    a_rev = a[::-1]
    need = len(w) + n - 1               # = 3·n‑2
    size = 1 << ((need - 1).bit_length())
    A = np.fft.rfft(a_rev, n=size)
    W = np.fft.rfft(w, n=size)
    conv = np.fft.irfft(A * W, n=size)
    conv = conv[:need]
    grad = 2.0 * conv[n - 1 : n - 1 + n]
    return grad


# ----------------------------------------------------------------------
# Adam optimiser (mirror descent with soft‑max surrogate) -------------
def _adam_optimize(
    a: np.ndarray,
    deadline: float,
    best_seq: np.ndarray,
    best_score: float,
    tau0: float = 0.5,
    tau_mult: float = 1.001,
    eta0: float = 0.20,
) -> tuple[np.ndarray, float]:
    """Mirrored Adam on the soft‑max approximation of the max‑convolution."""
    n = a.shape[0]
    m = np.zeros_like(a)          # first moment
    v = np.zeros_like(a)          # second moment
    beta1, beta2 = 0.9, 0.999
    bc1, bc2 = 1.0, 1.0           # bias‑correction accumulators
    tau = tau0
    it = 0
    while time.time() < deadline:
        it += 1
        # current convolution + exact score
        conv = _fft_self_convolve(a)
        cur = 2.0 * n * conv.max()
        if cur < best_score - 1e-12:
            best_score, best_seq = cur, a.copy()
            print(f"[{time.time():.1f}] Adam it={it} new best {best_score:.6f}")

        # soft‑max weighted gradient
        w = _softmax_weights(conv, tau)
        g = _weighted_gradient(a, w)

        # Adam updates
        m = beta1 * m + (1.0 - beta1) * g
        v = beta2 * v + (1.0 - beta2) * (g * g)
        bc1 *= beta1
        bc2 *= beta2
        m_hat = m / (1.0 - bc1)
        v_hat = v / (1.0 - bc2)

        # step size decays like 1/√t
        eta = eta0 / math.sqrt(it)
        a = a - eta * m_hat / (np.sqrt(v_hat) + 1e-8)

        # simplex projection (keeps sum = 1)
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s == 0.0:
            a = np.full_like(a, 1.0 / n)
        else:
            a = a / s

        # increase temperature – focus more on the true max
        tau *= tau_mult
    return best_seq, best_score


# ----------------------------------------------------------------------
# Sub‑gradient descent using the true max index -----------------------
def _hard_max_descent(
    a: np.ndarray,
    deadline: float,
    best_seq: np.ndarray,
    best_score: float,
    eta0: float = 0.15,
) -> tuple[np.ndarray, float]:
    """Projected sub‑gradient descent on the exact max‑convolution."""
    n = a.shape[0]
    it = 0
    while time.time() < deadline:
        it += 1
        conv = _fft_self_convolve(a)
        max_val = conv.max()
        cur = 2.0 * n * max_val
        if cur < best_score - 1e-12:
            best_score, best_seq = cur, a.copy()
            print(f"[{time.time():.1f}] Hard‑max it={it} new best {best_score:.6f}")

        # index where the maximum is attained
        k = int(conv.argmax())
        # gradient of max(conv) w.r.t a (sub‑gradient)
        lo = max(0, k - (n - 1))
        hi = min(n - 1, k)
        idx = np.arange(lo, hi + 1, dtype=np.int64)
        g = np.zeros_like(a)
        g[idx] = 2.0 * a[k - idx]

        # step size schedule
        eta = eta0 / math.sqrt(it)
        a = a - eta * g
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s == 0.0:
            a = np.full_like(a, 1.0 / n)
        else:
            a = a / s
    return best_seq, best_score


# ----------------------------------------------------------------------
# Pairwise swap local refinement ---------------------------------------
def _pairwise_swap_local(
    a: np.ndarray,
    deadline: float,
    best_seq: np.ndarray,
    best_score: float,
    max_trials: int | None = None,
) -> tuple[np.ndarray, float]:
    """Randomly transfer a small chunk of mass; keep change if objective improves."""
    n = a.shape[0]
    trials = 0
    rng = np.random.default_rng()
    while time.time() < deadline:
        if max_trials is not None and trials >= max_trials:
            break
        trials += 1

        src = np.where(a > 1e-12)[0]
        if src.size == 0:
            break
        i = rng.choice(src)

        j = rng.integers(n - 1)
        if j >= i:
            j += 1

        delta = a[i] * rng.random() * 0.35
        if delta < 1e-14:
            continue

        new_a = a.copy()
        new_a[i] -= delta
        new_a[j] += delta

        cur = _objective(new_a)
        if cur < best_score - 1e-14:
            a = new_a
            best_score = cur
            best_seq = a.copy()
            print(f"[{time.time():.1f}] swap i={i}→j={j} δ={delta:.3e} new best {best_score:.6f}")
    return best_seq, best_score


# ----------------------------------------------------------------------
# Simulated‑annealing style local search ---------------------------------
def _random_local_search(
    a: np.ndarray,
    deadline: float,
    best_seq: np.ndarray,
    best_score: float,
    init_temp: float = 8e-3,
    decay: float = 0.9997,
) -> tuple[np.ndarray, float]:
    """Very cheap SA that repeatedly transfers a random amount of mass."""
    n = a.shape[0]
    rng = np.random.default_rng()
    T = init_temp
    while time.time() < deadline:
        i = rng.integers(n)
        j = rng.integers(n - 1)
        if j >= i:
            j += 1
        delta = a[i] * rng.random() * 0.30
        if delta < 1e-14:
            continue
        new_a = a.copy()
        new_a[i] -= delta
        new_a[j] += delta
        cur = _objective(new_a)
        if cur < best_score - 1e-14:
            a = new_a
            best_score = cur
            best_seq = a.copy()
            print(f"[{time.time():.1f}] SA improve {i}->{j} Δ={delta:.3e} best {best_score:.6f}")
        else:
            if rng.random() < math.exp(-(cur - best_score) / max(T, 1e-12)):
                a = new_a
        T *= decay
    return best_seq, best_score

def propose_candidate(seed: int = 42, budget_s: int = 1000) -> list[float]:
    """
    Search for a non‑negative step‑function that minimises ``evaluate_sequence``.
    Returns a plain Python list (the sum is irrelevant – ``evaluate_sequence`` will
    normalise internally).  The routine respects the supplied time budget.
    """
    rng = np.random.default_rng(seed)
    start = time.time()
    deadline = start + budget_s - 4.0          # safety margin
    overall_best_seq: np.ndarray | None = None
    overall_best_score = float("inf")

    # ------------------------------------------------------------------
    # 1. quick scan over a handful of lengths & analytic families
    lengths = [3000, 5000, 8000, 12000, 16000, 20000]
    for n in lengths:
        if time.time() >= deadline:
            break
        print(f"[{time.time():.1f}] scanning length {n}")

        # uniform baseline (useful for sanity)
        cand = _distribution_uniform(n)
        sc = _objective(cand)
        if sc < overall_best_score:
            overall_best_score, overall_best_seq = sc, cand.copy()
            print(f"  uniform score {sc:.6f}")

        # edge‑weighted + uniform mixture (random p, λ)
        p = 10 ** rng.uniform(-2, 0.6)               # 0.01 … ~4
        edge = _distribution_edge_weighted(n, p)
        lam = rng.uniform(0.0, 0.6)
        cand = _mix_with_uniform(edge, lam)
        sc = _objective(cand)
        if sc < overall_best_score:
            overall_best_score, overall_best_seq = sc, cand.copy()
            print(f"  edge‑weighted p={p:.3g} λ={lam:.2f} → {sc:.6f}")

        # symmetric Beta + uniform mixture
        p = 10 ** rng.uniform(-2, 0.6)
        beta = _distribution_beta_sym(n, p)
        lam = rng.uniform(0.0, 0.6)
        cand = _mix_with_uniform(beta, lam)
        sc = _objective(cand)
        if sc < overall_best_score:
            overall_best_score, overall_best_seq = sc, cand.copy()
            print(f"  beta sym p={p:.3g} λ={lam:.2f} → {sc:.6f}")

        # arcsine (no mixing) – often a good starter
        cand = _distribution_arcsine(n)
        sc = _objective(cand)
        if sc < overall_best_score:
            overall_best_score, overall_best_seq = sc, cand.copy()
            print(f"  arcsine score {sc:.6f}")

    # ------------------------------------------------------------------
    # 2. include a previously discovered construction if present
    try:
        prev = np.asarray(height_sequence_1, dtype=np.float64)  # type: ignore[name-defined]

            sc = _objective(prev)
            if sc < overall_best_score:
                overall_best_score, overall_best_seq = sc, prev.copy()
                print(f"[{time.time():.1f}] previous best score={sc:.6f}")
    except Exception:
        pass

    # ------------------------------------------------------------------
    # 3. fallback – should never happen
    if overall_best_seq is None:
        n = 4000
        overall_best_seq = _distribution_uniform(n)
        overall_best_score = _objective(overall_best_seq)
        print(f"[{time.time():.1f}] fallback uniform score={overall_best_score:.6f}")

    # ------------------------------------------------------------------
    # 4. intensive continuous optimisation on the best seed
    if time.time() < deadline:
        print(f"[{time.time():.1f}] starting Adam optimisation (len={len(overall_best_seq)})")
        sub_deadline = deadline - 2.0    # keep a few seconds for polishing steps
        overall_best_seq, overall_best_score = _adam_optimize(
            overall_best_seq.copy(),
            sub_deadline,
            best_seq=overall_best_seq,
            best_score=overall_best_score,
            tau0=0.5,
            tau_mult=1.001,
            eta0=0.20,
        )

    # ------------------------------------------------------------------
    # 5. hard‑max sub‑gradient phase (focus on the true max)
    if time.time() < deadline:
        print(f"[{time.time():.1f}] hard‑max sub‑gradient descent")
        sub_deadline = deadline - 1.5
        overall_best_seq, overall_best_score = _hard_max_descent(
            overall_best_seq.copy(),
            sub_deadline,
            best_seq=overall_best_seq,
            best_score=overall_best_score,
            eta0=0.15,
        )

    # ------------------------------------------------------------------
    # 6. deterministic pairwise refinement
    if time.time() < deadline:
        print(f"[{time.time():.1f}] pairwise refinement")
        overall_best_seq, overall_best_score = _pairwise_swap_local(
            overall_best_seq.copy(),
            deadline,

            deadline,
