"""
step_function_search_v2.py

A 1000‑second search for a non‑negative step function that minimises

    2·n·max(conv(a,a)) / (∑a)²

The algorithm is a mix of:

*   a diverse pool of analytically‑generated seeds,
*   fast Adam (adaptive moment) optimisation on the exact non‑smooth
    objective (gradient = ∂/∂a_k max (conv(a,a))),
*   an optional up‑sampling stage (linear interpolation) followed by
    a second Adam run,
*   a cheap symmetrisation step,
*   a final pair‑wise mass‑transfer simulated‑annealing.

All work respects the 1000 s budget (2 s safety margin).  The function
``propose_candidate`` returns a plain ``list[float]`` ready for
``evaluate_sequence``.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Simplex utilities
# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Euclidean projection of a vector onto the simplex {x≥0, Σx=s}."""
    if s < 0:
        raise ValueError("simplex radius must be non‑negative")
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.where(u - cssv / (np.arange(1, len(v) + 1)) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Convolution helpers (FFT based, linear convolution)
# ----------------------------------------------------------------------
def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Full linear self‑convolution a*a using a real FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1) - 1).bit_length()      # next power‑of‑two
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[: 2 * n - 1]


# ----------------------------------------------------------------------
# Exact objective (scale‑invariant, sum=1 ⇒ denominator = 1)
# ----------------------------------------------------------------------
def _objective(a: np.ndarray) -> float:
    """Return 2·n·max(conv(a,a)) for a probability vector (∑a=1)."""
    n = a.shape[0]
    b = _fft_self_convolve(a)
    max_b = b.max()
    return float(2.0 * n * max_b)


# ----------------------------------------------------------------------
# Analytic seed families (all return a probability vector, sum=1)
# ----------------------------------------------------------------------
def _distribution_uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _distribution_arcsine(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    vals /= vals.sum()
    return vals


def _distribution_hann(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    w = 0.5 - 0.5 * np.cos(2.0 * np.pi * i / (n - 1))
    w_sum = w.sum()
    if w_sum == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w_sum
    return w


def _distribution_edge_weighted(n: int, p: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64) + 0.5
    j = (n - np.arange(n, dtype=np.float64) - 0.5)
    w = (i * j) ** (-p)
    w_sum = w.sum()
    if w_sum == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w_sum
    return w


def _distribution_generalized_normal(n: int, q: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64) + 0.5
    x = i / n - 0.5
    w = np.exp(- (np.abs(x) / 0.5) ** q)
    w_sum = w.sum()
    if w_sum == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w_sum
    return w


def _distribution_random_sparse(
    n: int, rng: np.random.Generator, density: float = 0.01
) -> np.ndarray:
    """Sparse random vector with a given density."""
    k = max(2, int(n * density))
    idx = rng.choice(n, size=k, replace=False)
    vals = np.zeros(n, dtype=np.float64)
    vals[idx] = rng.random(k)
    if vals.sum() == 0.0:
        vals[0] = 1.0
    vals /= vals.sum()
    return vals


def _mix_with_uniform(base: np.ndarray, lam: float) -> np.ndarray:
    """Convex combination with the uniform distribution."""
    n = base.shape[0]
    uniform = np.full(n, 1.0 / n, dtype=np.float64)
    return lam * uniform + (1.0 - lam) * base


# ----------------------------------------------------------------------
# Gradient of the active convolution entry (sub‑gradient)
# ----------------------------------------------------------------------
def _grad_of_max_conv(a: np.ndarray) -> tuple[np.ndarray, int]:
    """
    Compute a sub‑gradient of max(conv(a,a)).
    Returns (grad, k) where k is the index attaining the maximum.
    """
    n = a.shape[0]
    conv = _fft_self_convolve(a)
    k = int(np.argmax(conv))

    grad = np.zeros_like(a)
    if k < n:
        # overlap length = k+1
        grad[: k + 1] = 2.0 * a[k::-1]
    else:
        start = k - (n - 1)
        grad[start:] = 2.0 * a[start:][::-1]
    return grad, k


# ----------------------------------------------------------------------
# Adam optimiser on the exact (non‑smooth) objective
# ----------------------------------------------------------------------
def _adam_optimize(
    a: np.ndarray,
    deadline: float,
    best_seq: np.ndarray,
    best_score: float,
    alpha: float = 0.12,
    beta1: float = 0.90,
    beta2: float = 0.999,
    eps: float = 1e-8,
    eval_every: int = 200,
) -> tuple[np.ndarray, float]:
    """
    Adaptive‑moment (Adam) descent on 2·n·max(conv(a,a)).
    The update is a ← a − α·m̂/(√v̂+ε)  followed by projection onto the
    simplex {x≥0, Σx=1}.
    """
    n = a.shape[0]
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    t = 0
    while time.time() < deadline:
        t += 1
        grad, _ = _grad_of_max_conv(a)

        # Adam moments
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)

        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)

        step = alpha * m_hat / (np.sqrt(v_hat) + eps)

        a = a - step
        a = np.maximum(a, 0.0)                # enforce non‑negativity

        # project onto simplex (simple renormalisation works because all entries ≥0)
        s = a.sum()
        if s <= 0.0:
            a = np.full_like(a, 1.0 / n)
        else:
            a = a / s

        if t % eval_every == 0:
            cur = _objective(a)
            if cur < best_score - 1e-14:
                best_score = cur
                best_seq = a.copy()
                print(f"[{time.time():.1f}] Adam it={t} new best {best_score:.6f}")
    return best_seq, best_score


# ----------------------------------------------------------------------
# Simulated‑annealing style pairwise mass transfers
# ----------------------------------------------------------------------
def _pairwise_annealing(
    a: np.ndarray,
    deadline: float,
    best_seq: np.ndarray,
    best_score: float,
    rng: np.random.Generator,
) -> tuple[np.ndarray, float]:
    """
    Randomly transfer a fraction of mass between two bins.
    Moves that improve the objective are always kept; uphill moves are
    occasionally accepted according to a simple linear cooling schedule.
    """
    n = a.shape[0]
    start_time = time.time()
    total_time = deadline - start_time
    while time.time() < deadline:
        src_candidates = np.where(a > 1e-12)[0]
        if src_candidates.size == 0:
            break
        i = rng.choice(src_candidates)

        # pick a different destination
        j = rng.integers(n - 1)
        if j >= i:
            j += 1

        # move up to 40 % of the source mass (larger steps than the original code)
        delta = a[i] * rng.random() * 0.4
        if delta < 1e-16:
            continue

        new_a = a.copy()
        new_a[i] -= delta
        new_a[j] += delta

        cur = _objective(new_a)

        elapsed = time.time() - start_time
        # temperature is 5 % of the current best in the early phase, linearly cooling
        temperature = max(0.0, 1.0 - elapsed / total_time) * best_score * 0.05

        accept = False
        if cur < best_score - 1e-14:
            accept = True
        elif temperature > 0.0:
            prob = math.exp(-(cur - best_score) / temperature)
            if rng.random() < prob:
                accept = True

        if accept:
            a = new_a
            if cur < best_score - 1e-14:
                best_score = cur
                best_seq = a.copy()
                print(
                    f"[{time.time():.1f}] anneal i={i}→j={j} δ={delta:.3e} new best {best_score:.6f}"
                )
    return best_seq, best_score


# ----------------------------------------------------------------------
# Upsampling (linear interpolation while preserving the simplex)
# ----------------------------------------------------------------------
def _upsample(a: np.ndarray, new_n: int) -> np.ndarray:
    """Interpolate a probability vector to a larger length."""
    old_n = a.shape[0]
    if new_n == old_n:
        return a.copy()
    x_old = (np.arange(old_n, dtype=np.float64) + 0.5) / old_n
    x_new = (np.arange(new_n, dtype=np.float64) + 0.5) / new_n
    a_new = np.interp(x_new, x_old, a, left=0.0, right=0.0)
    a_new = np.maximum(a_new, 0.0)
    s = a_new.sum()
    if s == 0.0:
        a_new[:] = 1.0 / new_n
    else:
        a_new /= s
    return a_new


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: int = 1000) -> list[float]:
    """
    Search for a non‑negative step function minimising ``evaluate_sequence``.
    Returns the best sequence found (as a plain Python list).
    """
    rng = np.random.default_rng(seed)
    start = time.time()
    deadline = start + budget_s - 2.0                # safety margin

    overall_best_seq: np.ndarray | None = None
    overall_best_score = float("inf")
    cand_pool: list[tuple[float, np.ndarray]] = []

    # ------------------------------------------------------------------
    # 1.  Determine base length (moderate size to allow many iterations)
    # ------------------------------------------------------------------
    n = 4096
    print(f"[{time.time():.1f}] base length = {n}")

    # ------------------------------------------------------------------
    # 2.  Assemble a diverse pool of seed vectors
    # ------------------------------------------------------------------
    seeds = [
        _distribution_uniform(n),
        _distribution_arcsine(n),
        _distribution_hann(n),
        _distribution_edge_weighted(n, p=0.5),
        _distribution_generalized_normal(n, q=2.0),
    ]

    # Dirichlet families (different concentrations)
    for alpha in (0.2, 0.8, 2.0):
        seeds.append(rng.dirichlet(np.full(n, alpha)))

    # Sparse random seeds
    for _ in range(3):
        seeds.append(_distribution_random_sparse(n, rng, density=0.015))

    # Mix a few of the edge‑weighted families with uniform
    for lam in np.linspace(0.0, 0.7, 5):
        base = _distribution_edge_weighted(n, p=0.7)
        seeds.append(_mix_with_uniform(base, lam))

    # ------------------------------------------------------------------
    # 3.  Include the previously found construction if it exists
    # ------------------------------------------------------------------
    try:
        prev = np.asarray(height_sequence_1, dtype=np.float64)  # type: ignore[name-defined]
        if prev.ndim == 1:
            prev = prev / prev.sum()
            seeds.append(prev)
    except Exception:
        pass

    # ------------------------------------------------------------------
    # 4.  Evaluate all seeds and keep the very best ones
    # ------------------------------------------------------------------
    for s in seeds:
        sc = _objective(s)
        cand_pool.append((sc, s.copy()))
        if sc < overall_best_score:
            overall_best_score = sc
            overall_best_seq = s.copy()
            print(f"[{time.time():.1f}] new global best (seed) score={sc:.6f}")

    # Keep a small pool for short‑run polish
    cand_pool.sort(key=lambda x: x[0])
    cand_pool = cand_pool[:20]

    # ------------------------------------------------------------------
    # 5.  Quick Adam polishing on the seed pool (uses ~30 % of time)
    # ------------------------------------------------------------------
    remaining = deadline - time.time()
    if remaining > 30.0 and cand_pool:
        per_candidate = max(0.2, remaining * 0.3 / max(len(cand_pool), 1))
        print(f"[{time.time():.1f}] quick Adam on {len(cand_pool)} seeds (≈{per_candidate:.2f}s each)")
        for idx, (score, a0) in enumerate(cand_pool):
            if time.time() >= deadline:
                break
            sub_dead = time.time() + per_candidate
            a_opt, sc_opt = _adam_optimize(
                a0.copy(),
                sub_dead,
                best_seq=overall_best_seq,
                best_score=overall_best_score,
                alpha=0.12,
                eval_every=150,
            )
            if sc_opt < overall_best_score - 1e-12:
                overall_best_score = sc_opt
                overall_best_seq = a_opt.copy()
                print(f"[{time.time():.1f}] after quick Adam seed {idx} new best {overall_best_score:.6f}")

    # ------------------------------------------------------------------
    # 6.  Long Adam run on the current global best (≈35 % of budget)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        sub_dead = time.time() + (deadline - time.time()) * 0.35
        a_opt, sc_opt = _adam_optimize(
            overall_best_seq.copy(),
            sub_dead,
            best_seq=overall_best_seq,
            best_score=overall_best_score,
            alpha=0.09,
            eval_every=200,
        )
        if sc_opt < overall_best_score - 1e-12:
            overall_best_score = sc_opt
            overall_best_seq = a_opt.copy()
            print(f"[{time.time():.1f}] long Adam improved best to {overall_best_score:.6f}")

    # ------------------------------------------------------------------
    # 7.  Upsample and polish again (if time permits)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        cur_n = overall_best_seq.shape[0]
        target_n = cur_n * 2
        if target_n <= 64000:
            print(f"[{time.time():.1f}] upsampling {cur_n} → {target_n}")
            up = _upsample(overall_best_seq, target_n)
            up_sc = _objective(up)
            if up_sc < overall_best_score:
                overall_best_seq = up
                overall_best_score = up_sc
                print(f"[{time.time():.1f}] upsampled new best {overall_best_score:.6f}")

            # short Adam on the upsampled vector
            sub_dead = time.time() + (deadline - time.time()) * 0.25
            a_opt, sc_opt = _adam_optimize(
                overall_best_seq.copy(),
                sub_dead,
                best_seq=overall_best_seq,
                best_score=overall_best_score,
                alpha=0.08,
                eval_every=250,
            )
            if sc_opt < overall_best_score - 1e-12:
                overall_best_seq = a_opt.copy()
                overall_best_score = sc_opt
                print(f"[{time.time():.1f}] Adam after upsample new best {overall_best_score:.6f}")

    # ------------------------------------------------------------------
    # 8.  (Optional) Symmetrisation – many near‑optimal solutions are
    #     close to symmetric.
    # ------------------------------------------------------------------
    if time.time() < deadline:
        sym = (overall_best_seq + overall_best_seq[::-1]) * 0.5
        sym_sc = _objective(sym)
        if sym_sc < overall_best_score - 1e-12:
            overall_best_seq = sym
            overall_best_score = sym_sc
            print(f"[{time.time():.1f}] symmetry step improved best to {overall_best_score:.6f}")

    # ------------------------------------------------------------------
    # 9.  Final simulated‑annealing polishing (uses the remaining budget)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        print(f"[{time.time():.1f}] final annealing phase")
        overall_best_seq, overall_best_score = _pairwise_annealing(
            overall_best_seq.copy(),
            deadline,
            best_seq=overall_best_seq,
            best_score=overall_best_score,
            rng=rng,
        )
        print(f"[{time.time():.1f}] after annealing best score={overall_best_score:.6f}")

    # ------------------------------------------------------------------
    # 10.  Report and return
    # ------------------------------------------------------------------
    print(
        f"Final best score (objective=2·n·max(conv)): {overall_best_score:.6f} "
        f"(length={len(overall_best_seq)})"
    )
    return overall_best_seq.tolist()
