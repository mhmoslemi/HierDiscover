def _annealing_polish(a0, time_limit, best_seq, best_score):
    a = a0.copy()
    n = len(a)
    cur_score = best_score
    T0 = 0.001  # initial temperature maybe 1e-3
    # adapt T with exponential cooling
    steps = 0
    while time.time() < time_limit:
        steps += 1
        # Choose source and destination
        src = np.where(a > 1e-14)[0]
        i = rng.choice(src)
        j = rng.randint(n-1)
        if j >= i: j += 1
        # Choose delta as fraction of a[i] up to e.g., 0.5
        delta = a[i] * rng.random() * 0.5
        if delta < 1e-15: continue
        new_a = a.copy()
        new_a[i] -= delta
        new_a[j] += delta
        # Normalize (should sum to 1). Sum stays same as delta moved.
        # Compute objective
        new_score = _objective(new_a)
        d = new_score - cur_score
        # schedule temperature
        T = T0 * (0.995 ** steps)  # decaying
        accept = False
        if d < -1e-14:
            accept = True
        else:
            if T > 0:
                accept = rng.random() < math.exp(-d / T)
        if accept:
            a = new_a
            cur_score = new_score
            if cur_score < best_score - 1e-14:
                best_score = cur_score
                best_seq = a.copy()
                print(f"[{time.time():.1f}] anneal step {steps} new best {best_score:.6f}")
    return best_seq, best_score
```

"""
Search for a non‑negative step function that minimizes the
``evaluate_sequence`` objective.

Algorithm outline
-----------------
1. **Large‑scale analytic scan** – generate many candidate shapes on a
   reasonably large length (default 65536).  Families include uniform,
   arcsine (edge‑weighted with p=0.5), a two‑parameter edge‑weighted
   family w_i ∝ ((i+δ)*(n‑i‑δ))^{-p} and the symmetric Beta(p,p) family.
   Each base shape is also mixed with the uniform distribution:
   λ·uniform + (1‑λ)·base, exploring λ∈[0,1].

2. **Exact mirror‑descent refinement** – the best ~6 seeds are refined
   with exponentiated‑gradient descent on the exact non‑smooth objective.
   The update uses the true sub‑gradient of the maximum convolution
   entry and projects back onto the simplex.

3. **Simulated‑annealing polishing** – a hill‑climbing phase that
   occasionally accepts uphill moves (temperature decreasing geometrically)
   to escape local minima.  Randomly moves a fraction of mass from one
   coordinate to another and keeps any strict improvement.

The routine respects the provided time budget (default 1000 s) and finally
returns the best sequence found (already normalised to sum = 1).  The
sequence can safely be passed to ``evaluate_sequence``.
"""

# ----------------------------------------------------------------------
def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Full self‑convolution a*a using a real FFT (output length 2·n‑1)."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1) - 1).bit_length()   # next power of two
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[: 2 * n - 1]


def _objective(a: np.ndarray) -> float:
    """
    Exact objective 2·n·max(conv) for a vector whose entries sum to 1.
    The function does *not* renormalise – callers must ensure Σa = 1.
    """
    n = a.shape[0]
    b = _fft_self_convolve(a)
    return float(2.0 * n * b.max())

# Analytic families ----------------------------------------------------
def _distribution_uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _distribution_arcsine(n: int) -> np.ndarray:
    """Special case of the edge‑weighted family with p=0.5, offset=0.5."""
    i = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    vals /= vals.sum()
    return vals


def _distribution_edge_weighted(n: int, p: float, offset: float = 0.5) -> np.ndarray:
    """
    w_i ∝ ((i+offset)*(n‑i‑offset))^{‑p}.
    The usual arcsine case uses offset=0.5, p=0.5.
    """
    i = np.arange(n, dtype=np.float64) + offset
    j = (n - np.arange(n, dtype=np.float64) - offset)
    w = (i * j) ** (-p)
    w = np.clip(w, 0.0, None)
    total = w.sum()
    if total > 0:
        w /= total
    else:
        w = np.full(n, 1.0 / n, dtype=np.float64)
    return w


def _distribution_beta_sym(n: int, p: float) -> np.ndarray:
    """Symmetric Beta(p,p) on a regular grid."""
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    w = np.exp((p - 1) * np.log(x) + (p - 1) * np.log(1.0 - x))
    if w.sum() == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _distribution_power_sum(n: int, p: float) -> np.ndarray:
    """w_i ∝ (i+1)^{‑p} + (n‑i)^{‑p} – another edge‑heavy family."""
    i = np.arange(1, n + 1, dtype=np.float64)
    w = i ** (-p) + (n + 1 - i) ** (-p)
    w = np.clip(w, 0.0, None)
    w /= w.sum()
    return w


def _mix_with_uniform(base: np.ndarray, lam: float) -> np.ndarray:
    """Convex combination λ·uniform + (1‑λ)·base."""
    n = base.shape[0]
    uniform = np.full(n, 1.0 / n, dtype=np.float64)
    return lam * uniform + (1.0 - lam) * base

# Sub‑gradient of the max‑convolution -----------------------------------
def _max_convolution_subgradient(a: np.ndarray) -> np.ndarray:
    """
    Sub‑gradient of max_k Σ_i a_i a_{k-i}.
    Returns a vector of the same shape as ``a``.
    """
    conv = _fft_self_convolve(a)
    max_val = conv.max()
    # treat numerically maximal indices as active
    tol = 1e-12 * max_val
    active = np.where(conv >= max_val - tol)[0]
    n = a.shape[0]
    grad = np.zeros_like(a)
    for k in active:
        lo = max(0, k - (n - 1))
        hi = min(n - 1, k)
        idx = np.arange(lo, hi + 1)
        grad[idx] += 2.0 * a[k - idx]
    if active.size > 0:
        grad /= active.size
    return grad

def _exp_gradient_descent(
    a0: np.ndarray,
    time_limit: float,
    best_seq: np.ndarray,
    best_score: float,
    eta0: float = 1.0,
    report_every: int = 500,
    max_iter: int = 200_000,
) -> tuple[np.ndarray, float]:
    """
    Exponentiated‑gradient descent (mirror descent) on the exact max‑convolution.
    Updates a ← a·exp(−η·grad) and re‑normalises to the simplex.
    """
    a = a0.copy()
    n = a.shape[0]
    t = 0
    while time.time() < time_limit and t < max_iter:
        t += 1
        grad = _max_convolution_subgradient(a)
        eta = eta0 / math.sqrt(t)
        a = a * np.exp(-eta * grad)
        s = a.sum()
        if s > 0:
            a = a / s
        else:
            a = np.full(n, 1.0 / n, dtype=np.float64)

        if t % report_every == 0:
            cur = _objective(a)
            if cur < best_score - 1e-14:
                best_score = cur
                best_seq = a.copy()
                print(f"[{time.time():.1f}] exp‑gd it={t} new global best {best_score:.6f}")
    return best_seq, best_score

def _annealing_polish(
    a0: np.ndarray,
    time_limit: float,
    best_seq: np.ndarray,
    best_score: float,
    rng: np.random.Generator,
) -> tuple[np.ndarray, float]:
    """
    Simulated‑annealing hill climbing: move a random fraction of mass from one
    coordinate to another.  Accepts all improving moves and occasionally
    accepts worsening moves with a decaying temperature.
    """
    a = a0.copy()
    n = a.shape[0]
    cur_score = best_score
    steps = 0
    T0 = 5e-4                     # initial temperature (tuned experimentally)
    while time.time() < time_limit:
        steps += 1
        # choose source (must have some mass)
        src = np.where(a > 1e-14)[0]
        if src.size == 0:
            break
        i = rng.choice(src)
        # choose distinct destination
        j = rng.integers(n - 1)
        if j >= i:
            j += 1

        # transfer up to 40 % of the source mass
        delta = a[i] * rng.random() * 0.40
        if delta < 1e-15:
            continue

        new_a = a.copy()
        new_a[i] -= delta
        new_a[j] += delta

        new_score = _objective(new_a)
        d = new_score - cur_score
        # geometric cooling
        T = T0 * (0.995 ** steps)

        accept = False
        if d < -1e-14:
            accept = True
        elif T > 0 and rng.random() < math.exp(-d / T):
            accept = True

        if accept:
            a = new_a
            cur_score = new_score
            if cur_score < best_score - 1e-14:
                best_score = cur_score
                best_seq = a.copy()
                print(f"[{time.time():.1f}] anneal step {steps} new best {best_score:.6f}")
    return best_seq, best_score
