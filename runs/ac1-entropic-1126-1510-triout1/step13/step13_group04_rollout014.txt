def _distribution_triangular(n):
    i = np.arange(n, dtype=np.float64) + 0.5
    x = i / n
    # triangular peaked at center: w = 1 - 2*abs(x-0.5)
    w = 1 - 2*np.abs(x-0.5)
    w = np.maximum(w, 0)
    w /= w.sum()
    return w

    x = i / n

def _grad_lse(a, tau):

    maxc = conv.max()

"""
Propose a non‑negative step function that (almost) minimizes
``evaluate_sequence``.

Algorithm
---------
1.  Build a diverse pool of symmetric probability vectors from many
    analytic families (uniform, arcsine, Hann, triangular, V‑shape,
    beta, edge‑weighted, generalized‑normal, power‑law, random Dirichlet,
    sparse).  Keep the best 20‑30 candidates.

2.  Refine the top candidates with **smooth projected gradient descent**.
    Instead of the sub‑gradient of the max‑convolution we minimise a
    smooth surrogate:
        Lτ(a) = τ·log Σ_k exp(( (a*a)[k] - max(a*a) ) / τ)
    whose gradient is a weighted sum of the usual max‑convolution gradients.
    Starting with a relatively large τ (0.2) we run a few thousand
    iterations, then halve τ repeatedly down to 0.01.  After each step
    we enforce symmetry (a = (a+a[::-1])/2) and project onto the simplex.
    The best vector found over all seeds is kept.

3.  Upsample the current best vector (linear interpolation) to double the
    length and run a short burst of the same smooth descent – this
    reduces discretisation error.

4.  Finish with a long pair‑wise annealing (mass transfers) as in the
    previous state‑of‑the‑art, to polish any remaining slack.

All steps respect the global 1000 s budget (2 s safety margin).  The
routine returns a plain ``list[float]`` ready for ``evaluate_sequence``.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Simplex projection (Euclidean)
# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x≥0, Σx=s}."""
    if s < 0:
        raise ValueError("simplex radius must be non‑negative")
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.where(u - cssv / (np.arange(1, len(v) + 1)) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# FFT‑based self‑convolution (full linear)
# ----------------------------------------------------------------------
def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    n = a.shape[0]
    size = 1 << ((2 * n - 1) - 1).bit_length()
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[: 2 * n - 1]


# ----------------------------------------------------------------------
# Exact objective used by ``evaluate_sequence`` (scale‑invariant)
# ----------------------------------------------------------------------
def _objective(a: np.ndarray) -> float:
    n = a.shape[0]
    b = _fft_self_convolve(a)
    max_b = b.max()
    sum_a = a.sum()
    if sum_a < 0.01:
        return float("inf")
    return float(2.0 * n * max_b / (sum_a * sum_a))


# ----------------------------------------------------------------------
# Analytic families – always return a probability vector (sum = 1)
# ----------------------------------------------------------------------
def _distribution_uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _distribution_arcsine(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    vals /= vals.sum()
    return vals


def _distribution_hann(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    w = 0.5 - 0.5 * np.cos(2.0 * np.pi * i / (n - 1))
    w /= w.sum()
    return w


def _distribution_triangular(n: int) -> np.ndarray:
    """Tent shape, peak in the centre, zero at the edges."""
    i = np.arange(n, dtype=np.float64) + 0.5
    x = i / n
    w = 1.0 - 2.0 * np.abs(x - 0.5)
    w[w < 0] = 0.0
    w /= w.sum()
    return w


def _distribution_vshape(n: int) -> np.ndarray:
    """V‑shape, minimum in the centre, maximum at the edges."""
    i = np.arange(n, dtype=np.float64) + 0.5
    x = i / n
    w = np.abs(2.0 * x - 1.0)
    w /= w.sum()
    return w


def _distribution_beta_sym(n: int, p: float) -> np.ndarray:
    """Symmetric Beta(p,p) on a uniform grid."""
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    w = np.exp((p - 1) * np.log(x) + (p - 1) * np.log(1.0 - x))
    if w.sum() == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _distribution_edge_weighted(n: int, p: float) -> np.ndarray:
    """Weight ∝ (i·j)^{-p} where i,j are distances to the two edges."""
    i = np.arange(n, dtype=np.float64) + 0.5
    j = (n - np.arange(n, dtype=np.float64) - 0.5)
    w = (i * j) ** (-p)
    if w.sum() == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _distribution_generalized_normal(n: int, q: float) -> np.ndarray:
    """Density ∝ exp(-|x-½|^{q}) on [0,1]."""
    i = np.arange(n, dtype=np.float64) + 0.5
    x = i / n - 0.5
    w = np.exp(- (np.abs(x) / 0.5) ** q)
    w /= w.sum()
    return w


def _distribution_power_law(n: int, p: float) -> np.ndarray:
    """Monotone decreasing power‑law (p>0)."""
    i = np.arange(1, n + 1, dtype=np.float64)
    w = i ** (-p)
    w /= w.sum()
    return w


def _distribution_random_sparse(
    n: int, rng: np.random.Generator, density: float = 0.01
) -> np.ndarray:
    """Sparse random vector with a given density."""
    k = max(2, int(n * density))
    idx = rng.choice(n, size=k, replace=False)
    vals = np.zeros(n, dtype=np.float64)
    vals[idx] = rng.random(k)
    if vals.sum() == 0.0:
        vals[0] = 1.0
    vals /= vals.sum()
    return vals


def _mix_with_uniform(base: np.ndarray, lam: float) -> np.ndarray:
    """Convex combination with the uniform distribution."""
    n = base.shape[0]
    uniform = np.full(n, 1.0 / n, dtype=np.float64)
    return lam * uniform + (1.0 - lam) * base


# ----------------------------------------------------------------------
# Gradient of the smoothed max‑convolution (log‑sum‑exp surrogate)
# ----------------------------------------------------------------------
def _grad_lse(a: np.ndarray, tau: float) -> np.ndarray:
    """
    Gradient of τ·log Σ_k exp((conv_k‑max_conv)/τ) w.r.t. a.
    Returns a vector of length n.
    """
    n = a.shape[0]
    conv = _fft_self_convolve(a)               # length 2n‑1
    maxc = conv.max()
    # safe exponentiation
    w = np.exp((conv - maxc) / tau)            # w_k ≥ 0, max entry gives 1
    w_sum = w.sum()
    if w_sum == 0.0:
        # degenerate case – fall back to hard max gradient
        grad = np.zeros_like(a)
        k = int(np.argmax(conv))
        if k < n:
            grad[: k + 1] = 2.0 * a[k::-1]
        else:
            start = k - (n - 1)
            grad[start:] = 2.0 * a[start:][::-1]
        return grad

    rev_a = a[::-1]
    size = 1 << ((len(w) + n - 1 - 1).bit_length())
    W = np.fft.rfft(w, n=size)
    R = np.fft.rfft(rev_a, n=size)
    conv_wr = np.fft.irfft(W * R, n=size)[: len(w) + n - 1]
    start = n - 1
    grad = 2.0 * conv_wr[start:start + n] / w_sum
    return grad


# ----------------------------------------------------------------------
# Smooth projected gradient descent with decreasing temperature
# ----------------------------------------------------------------------
def _smooth_pg_descent(
    a: np.ndarray,
    deadline: float,
    best_seq: np.ndarray,
    best_score: float,
    tau_schedule: list[float],
    eta0: float = 0.5,
    eval_every: int = 50,
) -> tuple[np.ndarray, float]:
    """
    Minimise a smooth surrogate of the max‑convolution.
    The vector is kept on the probability simplex and forced to stay symmetric.
    """
    n = a.shape[0]
    for tau in tau_schedule:
        t = 0
        while time.time() < deadline:
            t += 1
            grad = _grad_lse(a, tau)
            eta = eta0 / math.sqrt(t)
            a = a - eta * grad
            # enforce symmetry
            a = (a + a[::-1]) / 2.0
            # project onto simplex (keeps sum = 1, non‑negative)
            a = _project_onto_simplex(a)
            if t % eval_every == 0:
                cur = _objective(a)
                if cur < best_score - 1e-14:
                    best_score = cur
                    best_seq = a.copy()
                    print(
                        f"[{time.time():.1f}] smooth‑PGD τ={tau:.3g} it={t} new best {best_score:.6f}"
                    )
    return best_seq, best_score


# ----------------------------------------------------------------------
# Upsampling while preserving the simplex (linear interpolation)
# ----------------------------------------------------------------------
def _upsample(a: np.ndarray, new_n: int) -> np.ndarray:
    """Interpolate a probability vector to a larger length."""
    old_n = a.shape[0]
    if new_n == old_n:
        return a.copy()
    x_old = (np.arange(old_n, dtype=np.float64) + 0.5) / old_n
    x_new = (np.arange(new_n, dtype=np.float64) + 0.5) / new_n
    a_new = np.interp(x_new, x_old, a, left=0.0, right=0.0)
    a_new = np.maximum(a_new, 0.0)
    if a_new.sum() == 0.0:
        a_new[:] = 1.0 / new_n
    else:
        a_new /= a_new.sum()
    # re‑symmetrise just in case interpolation introduced tiny asymmetry
    a_new = (a_new + a_new[::-1]) / 2.0
    a_new = _project_onto_simplex(a_new)
    return a_new


# ----------------------------------------------------------------------
# Pairwise mass‑transfer annealing (final polishing)
# ----------------------------------------------------------------------
def _pairwise_annealing(
    a: np.ndarray,
    deadline: float,
    best_seq: np.ndarray,
    best_score: float,
    rng: np.random.Generator,
) -> tuple[np.ndarray, float]:
    """Randomly move a fraction of mass between two bins; accept if improves."""
    n = a.shape[0]
    start_time = time.time()
    total_time = deadline - start_time
    trials = 0
    while time.time() < deadline:
        trials += 1
        src_candidates = np.where(a > 1e-12)[0]
        if src_candidates.size == 0:
            break
        i = rng.choice(src_candidates)
        # pick a distinct destination
        j = rng.integers(n - 1)
        if j >= i:
            j += 1
        delta = a[i] * rng.random() * 0.4
        if delta < 1e-16:
            continue
        new_a = a.copy()
        new_a[i] -= delta
        new_a[j] += delta
        # re‑symmetrise and project to stay on the simplex
        new_a = (new_a + new_a[::-1]) / 2.0
        new_a = _project_onto_simplex(new_a)

        cur = _objective(new_a)
        elapsed = time.time() - start_time
        temperature = max(
            0.0, 1.0 - elapsed / total_time
        ) * best_score * 0.01  # simple linear schedule

        if cur < best_score - 1e-14:
            accept = True
        else:
            accept = temperature > 0.0 and rng.random() < math.exp(-(cur - best_score) / temperature)

        if accept:
            a = new_a
            if cur < best_score - 1e-14:
                best_score = cur
                best_seq = a.copy()
                print(
                    f"[{time.time():.1f}] anneal i={i}->j={j} δ={delta:.3e} new best {best_score:.6f}"
                )
    return best_seq, best_score
