"""
Hybrid search for a low‑peak self‑convolution step function.

The algorithm proceeds in three stages:

1. **Seed generation** – a small collection of analytic families
   (uniform, arcsine, symmetric Beta, edge‑weighted, sine‑weighted) and
   a few random Dirichlet vectors are built for several lengths.
   The best few seeds (by the exact objective 2·n·max(conv)) are kept.

2. **Linear‑programming refinement** – for each promising seed we solve a
   linear program that maximises the total mass of a new vector `g`
   subject to the pointwise bound `conv(f, g) ≤ M`, where `M` is the
   current peak of `conv(f, f)`.  After the LP we rescale `g` to unit
   sum and mix it with the current `f` (step size `t`).  This is the
   classic “LP‑based improvement” described in the statement and
   strictly decreases the objective when a progress step is accepted.
   A few iterations are performed while time permits.

3. **Projected sub‑gradient descent** – the refined vector is finally
   polished with a simple sub‑gradient method on the nonsmooth
   `max(conv)` functional, keeping the vector on the probability
   simplex.

All operations respect a global deadline (`budget_s` seconds).  The
function returns a plain Python list; the external `evaluate_sequence`
will apply the required clipping and normalisation.
"""

import time
import math
import numpy as np

# Optional import – the LP stage is used only if CVXPY is available.
try:
    import cvxpy as cp
    _HAS_CVXPY = True
except Exception:  # pragma: no cover
    _HAS_CVXPY = False

# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Euclidean projection of `v` onto the simplex {x ≥ 0, Σx = s}."""
    if s < 0:
        raise ValueError("simplex radius must be non‑negative")
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho_arr = np.where(u - cssv / (np.arange(1, len(v) + 1)) > 0)[0]
    if rho_arr.size == 0:
        theta = 0.0
    else:
        rho = rho_arr[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Full linear self‑convolution a*a using a real FFT."""
    n = a.shape[0]
    need = 2 * n - 1
    size = 1 << ((need - 1).bit_length())
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[:need]


# ----------------------------------------------------------------------
def _objective(a: np.ndarray) -> float:
    """Exact objective: 2·n·max(conv(a, a)) with Σa = 1."""
    n = a.shape[0]
    b = _fft_self_convolve(a)
    return 2.0 * n * b.max()

# Analytic families ----------------------------------------------------
def _distribution_uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _distribution_arcsine(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    vals /= vals.sum()
    return vals


def _distribution_beta_sym(n: int, p: float) -> np.ndarray:
    """Symmetric beta (Beta(p,p)) on the discrete grid."""
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    w = np.exp((p - 1) * np.log(x) + (p - 1) * np.log(1.0 - x))
    if w.sum() == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _distribution_beta_asym(n: int, a: float, b: float) -> np.ndarray:
    """Asymmetric beta (Beta(a,b))."""
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    w = np.exp((a - 1) * np.log(x) + (b - 1) * np.log(1.0 - x))
    if w.sum() == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _distribution_edge_weighted(n: int, p: float) -> np.ndarray:
    """U‑shaped family w_i ∝ ((i+½)*(n‑i‑½))^(−p)."""
    i = np.arange(n, dtype=np.float64) + 0.5
    j = (n - np.arange(n, dtype=np.float64) - 0.5)
    w = (i * j) ** (-p)
    if w.sum() == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _distribution_sine_weighted(n: int, p: float) -> np.ndarray:
    """Weight ∝ sin(π (i+½)/n)^p."""
    theta = np.pi * (np.arange(n, dtype=np.float64) + 0.5) / n
    w = np.sin(theta) ** p
    if w.sum() == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _distribution_random_sparse(
    n: int, rng: np.random.Generator, density: float = 0.01
) -> np.ndarray:
    """Sparse random vector (few non‑zeros)."""
    k = max(2, int(n * density))
    idx = rng.choice(n, size=k, replace=False)
    vals = np.zeros(n, dtype=np.float64)
    vals[idx] = rng.random(k)
    if vals.sum() == 0.0:
        vals[0] = 1.0
    vals /= vals.sum()
    return vals


def _mix_with_uniform(base: np.ndarray, lam: float) -> np.ndarray:
    """Convex combination lam·U + (1‑lam)·base, where U is uniform."""
    n = base.shape[0]
    uniform = np.full(n, 1.0 / n, dtype=np.float64)
    return lam * uniform + (1.0 - lam) * base

# Projected sub‑gradient descent on max‑convolution --------------------
def _proj_grad_descent(
    a: np.ndarray,
    deadline: float,
    best_seq: np.ndarray,
    best_score: float,
    eta0: float = 0.6,
    jitter_interval: int = 2500,
) -> tuple[np.ndarray, float]:
    """
    Minimise 2·n·max(conv(a,a)) with exact sub‑gradient.
    `a` is assumed to lie on the probability simplex (Σa = 1).
    """
    n = a.shape[0]
    t = 0
    no_improve = 0
    rng = np.random.default_rng()
    cur_score = _objective(a)

    while time.time() < deadline and t < 200_000:
        t += 1

        conv = _fft_self_convolve(a)
        max_val = conv.max()
        cur_score = 2.0 * n * max_val

        if cur_score < best_score - 1e-14:
            best_score = cur_score
            best_seq = a.copy()
            no_improve = 0
            print(f"[{time.time():.1f}] proj‑grad iter={t} new best {best_score:.6f}")
        else:
            no_improve += 1

        # sub‑gradient of the maximal entry (average over ties)
        ks = np.where(conv == max_val)[0]
        grad = np.zeros_like(a)
        rev_a = a[::-1]
        for k in ks:
            offset = n - 1 - k
            if offset >= 0:
                i_start = 0
                i_end = n - offset
                grad[i_start:i_end] += 2.0 * rev_a[offset:offset + i_end]
            else:
                off = -offset
                i_start = off
                i_end = n
                grad[i_start:i_end] += 2.0 * rev_a[0:i_end - i_start]

        eta = eta0 / math.sqrt(t)
        a_new = a - eta * grad
        a_new = _project_onto_simplex(a_new, s=1.0)

        if no_improve >= jitter_interval:
            jitter = rng.exponential(scale=0.003, size=n)
            a_new = a_new + jitter
            a_new = _project_onto_simplex(a_new, s=1.0)
            no_improve = 0

        a = a_new

    return best_seq, best_score

# Linear‑programming refinement (LP‑based improvement) ---------------
def _conv_matrix(f: np.ndarray) -> np.ndarray:
    """
    Build the (2n‑1) × n matrix A such that (A @ g) == conv(f, g)
    for any vector g of length n.
    """
    n = f.shape[0]
    conv_len = 2 * n - 1
    A = np.zeros((conv_len, n), dtype=np.float64)
    # each column j contributes f shifted by j
    for j in range(n):
        A[j:j + n, j] = f
    return A


def _lp_refine(
    f: np.ndarray,
    deadline: float,
    max_iters: int = 20,
    mix_factor: float = 0.5,
) -> tuple[np.ndarray, float]:
    """
    Perform a few LP‑based improvement steps.

    Returns the refined vector (still a probability distribution) and its
    objective value 2·n·max(conv(f,f)).
    """
    if not _HAS_CVXPY:
        return f, _objective(f)

    n = f.shape[0]
    best = f.copy()
    best_M = _fft_self_convolve(best).max()
    best_score = 2.0 * n * best_M

    it = 0
    while time.time() < deadline and it < max_iters:
        it += 1
        M = best_M
        A = _conv_matrix(best)

        # Linear program: maximise Σ g   subject to g ≥ 0 , A·g ≤ M·1
        g = cp.Variable(n, nonneg=True)
        constraints = [A @ g <= M * np.ones(2 * n - 1)]
        prob = cp.Problem(cp.Maximize(cp.sum(g)), constraints)

        try:
            prob.solve(solver=cp.ECOS, warm_start=True, max_iters=5000, verbose=False)
        except Exception as e:  # pragma: no cover
            print(f"LP solve failed: {e}")
            break

        if prob.status not in ("optimal", "optimal_inaccurate"):
            print(f"LP not solved to optimality (status {prob.status}); aborting LP phase.")
            break

        g_val = np.asarray(g.value).flatten()
        if g_val is None or g_val.size == 0:
            break
        S = g_val.sum()
        if S <= 1e-12:
            break

        # Rescale to unit sum
        g_scaled = g_val / S

        # Try a small convex combination with the current best.
        improved = False
        for t in (mix_factor, mix_factor * 0.5, mix_factor * 0.25):
            candidate = (1.0 - t) * best + t * g_scaled
            M_new = _fft_self_convolve(candidate).max()
            if M_new < best_M - 1e-14:
                best = candidate
                best_M = M_new
                best_score = 2.0 * n * best_M
                improved = True
                print(f"[{time.time():.1f}] LP iter={it} t={t:.3f} M={best_M:.8f} score={best_score:.6f}")
                break
        if not improved:
            # No improvement with any mixing factor → stop early.
            break

    return best, best_score

# Greedy pairwise refinement (deterministic) -------------------------
def _pairwise_swap_local(
    a: np.ndarray,
    deadline: float,
    best_seq: np.ndarray,
    best_score: float,
    max_trials: int | None = None,
) -> tuple[np.ndarray, float]:
    """Move a small amount of mass between two indices if it improves the objective."""
    n = a.shape[0]
    trials = 0
    rng = np.random.default_rng()
    while time.time() < deadline:
        if max_trials is not None and trials >= max_trials:
            break
        trials += 1

        src = np.where(a > 1e-12)[0]
        if src.size == 0:
            break
        i = rng.choice(src)

        j = rng.integers(n - 1)
        if j >= i:
            j += 1

        delta = a[i] * rng.random() * 0.35
        if delta < 1e-14:
            continue

        new_a = a.copy()
        new_a[i] -= delta
        new_a[j] += delta

        cur = _objective(new_a)
        if cur < best_score - 1e-14:
            a = new_a
            best_score = cur
            best_seq = a.copy()
            print(f"[{time.time():.1f}] swap i={i}→j={j} δ={delta:.3e} new best {best_score:.6f}")

    return best_seq, best_score

# Cheap simulated‑annealing walk ---------------------------------------
def _random_local_search(
    a: np.ndarray,
    deadline: float,
    best_seq: np.ndarray,
    best_score: float,
    init_temp: float = 1e-2,
    decay: float = 0.9995,
) -> tuple[np.ndarray, float]:
    """Very cheap SA that repeatedly transfers a random amount of mass."""
    n = a.shape[0]
    rng = np.random.default_rng()
    T = init_temp
    while time.time() < deadline:
        i = rng.integers(n)
        j = rng.integers(n - 1)
        if j >= i:
            j += 1
        delta = a[i] * rng.random() * 0.3
        if delta < 1e-14:
            continue
        new_a = a.copy()
        new_a[i] -= delta
        new_a[j] += delta
        cur = _objective(new_a)
        if cur < best_score - 1e-14:
            a = new_a
            best_score = cur
            best_seq = a.copy()
            print(f"[{time.time():.1f}] SA improve {i}->{j} Δ={delta:.3e} best {best_score:.6f}")
        else:
            if rng.random() < math.exp(-(cur - best_score) / max(T, 1e-12)):
                a = new_a
        T *= decay
    return best_seq, best_score

def propose_candidate(seed: int = 42, budget_s: int = 1000) -> list[float]:
    """
    Search for a non‑negative step‑function (list of heights) that minimises
    ``evaluate_sequence``.  The routine builds diverse seeds, improves them with
    an LP‑based step, then finishes with sub‑gradient and local search.
    """
    rng = np.random.default_rng(seed)
    start = time.time()
    deadline = start + budget_s - 3.0          # small safety margin

    overall_best_seq = None
    overall_best_score = float("inf")
    candidate_pool: list[tuple[float, np.ndarray]] = []

    # ------------------------------------------------------------------
    # 1. Seed generation (few lengths, a handful of families)
    lengths = [1800, 2400, 3000, 3600]          # modest sizes for fast LP
    beta_ps = [0.05, 0.1, 0.2, 0.4, 0.7, 1.0, 1.5, 2.0]
    edge_ps = [0.05, 0.1, 0.2, 0.3, 0.5, 0.8, 1.2, 2.0]
    sine_ps = [1, 2, 3, 4, 5, 6]
    mix_lams = [0.0, 0.25, 0.5, 0.75, 1.0]

    for n in lengths:
        if time.time() >= deadline:
            break
        print(f"[{time.time():.1f}] Generating seeds for length {n}")

        # uniform
        cand = _distribution_uniform(n)
        sc = _objective(cand)
        candidate_pool.append((sc, cand.copy()))
        if sc < overall_best_score:
            overall_best_score, overall_best_seq = sc, cand.copy()
            print(f"[{time.time():.1f}] new global best (uniform) n={n} score={sc:.6f}")

        # arcsine
        cand = _distribution_arcsine(n)
        sc = _objective(cand)
        candidate_pool.append((sc, cand.copy()))
        if sc < overall_best_score:
            overall_best_score, overall_best_seq = sc, cand.copy()
            print(f"[{time.time():.1f}] new global best (arcsine) n={n} score={sc:.6f}")

        # symmetric beta
        for p in beta_ps:
            base = _distribution_beta_sym(n, p)
            for lam in mix_lams:
                cand = _mix_with_uniform(base, lam)
                sc = _objective(cand)
                candidate_pool.append((sc, cand.copy()))
                if sc < overall_best_score:
                    overall_best_score, overall_best_seq = sc, cand.copy()
                    print(f"[{time.time():.1f}] new best βsym p={p} lam={lam} n={n} score={sc:.6f}")

        # edge‑weighted
        for p in edge_ps:
            base = _distribution_edge_weighted(n, p)
            for lam in mix_lams:
                cand = _mix_with_uniform(base, lam)
                sc = _objective(cand)
                candidate_pool.append((sc, cand.copy()))
                if sc < overall_best_score:
                    overall_best_score, overall_best_seq = sc, cand.copy()
                    print(f"[{time.time():.1f}] new best edge p={p} lam={lam} n={n} score={sc:.6f}")

        # sine‑weighted
        for p in sine_ps:
            base = _distribution_sine_weighted(n, p)
            for lam in mix_lams:
                cand = _mix_with_uniform(base, lam)
                sc = _objective(cand)
                candidate_pool.append((sc, cand.copy()))
                if sc < overall_best_score:
                    overall_best_score, overall_best_seq = sc, cand.copy()
                    print(f"[{time.time():.1f}] new best sine p={p} lam={lam} n={n} score={sc:.6f}")

        # random Dirichlet
        for _ in range(3):
            alpha = rng.uniform(0.01, 2.0)
            cand = rng.dirichlet(np.full(n, alpha))
            sc = _objective(cand)
            candidate_pool.append((sc, cand.copy()))
            if sc < overall_best_score:
                overall_best_score, overall_best_seq = sc, cand.copy()
                print(f"[{time.time():.1f}] new best Dirichlet α={alpha:.3g} n={n} score={sc:.6f}")

        # sparse random vectors
        for _ in range(2):
            cand = _distribution_random_sparse(n, rng, density=0.01)
            sc = _objective(cand)
            candidate_pool.append((sc, cand.copy()))
            if sc < overall_best_score:
                overall_best_score, overall_best_seq = sc, cand.copy()
                print(f"[{time.time():.1f}] new best sparse n={n} score={sc:.6f}")

    # Keep a modest set of the very best seeds for refinement.
    candidate_pool.sort(key=lambda x: x[0])
    top_seeds = [seq for _, seq in candidate_pool[:6]]

    # ------------------------------------------------------------------
    # 2. Optional import of a previously discovered construction.
    try:
        prev = np.asarray(height_sequence_1, dtype=np.float64)  # type: ignore[name-defined]
        if prev.ndim == 1 and prev.size > 0:
            prev = np.maximum(prev, 0.0)
            prev = prev / prev.sum()
            sc = _objective(prev)
            if sc < overall_best_score:
                overall_best_score, overall_best_seq = sc, prev.copy()
                print(f"[{time.time():.1f}] imported previous best score={sc:.6f}")
            candidate_pool.append((sc, prev.copy()))
            top_seeds.append(prev.copy())
    except Exception:
        pass

    # ------------------------------------------------------------------
    # 3. LP‑based refinement on the most promising seeds.
    if _HAS_CVXPY and time.time() < deadline - 20.0:
        remaining = deadline - time.time()
        per_seed_time = remaining / max(1, len(top_seeds))
        print(f"[{time.time():.1f}] Starting LP refinement on {len(top_seeds)} seeds")
        for idx, seed_vec in enumerate(top_seeds):
            if time.time() >= deadline:
                break
            sub_deadline = time.time() + per_seed_time
            refined, refined_score = _lp_refine(
                seed_vec.copy(),
                sub_deadline,
                max_iters=12,            # few LP iterations per seed
                mix_factor=0.5,
            )
            if refined_score < overall_best_score:
                overall_best_score, overall_best_seq = refined_score, refined.copy()
                print(f"[{time.time():.1f}] LP seed {idx} new global best {overall_best_score:.6f}")

            # early stop if we already beat the target
            if overall_best_score <= 1.503:
                break

    # ------------------------------------------------------------------
    # 4. Projected sub‑gradient descent polishing (on the current best)
    if time.time() < deadline - 10.0:
        print(f"[{time.time():.1f}] Polishing with projected sub‑gradient descent")
        overall_best_seq, overall_best_score = _proj_grad_descent(
            overall_best_seq.copy(),
            deadline,
            best_seq=overall_best_seq,
            best_score=overall_best_score,
            eta0=0.8,
            jitter_interval=3000,
        )

    # ------------------------------------------------------------------
    # 5. Greedy pairwise swaps
    if time.time() < deadline:
        print(f"[{time.time():.1f}] Greedy pairwise refinement")
        overall_best_seq, overall_best_score = _pairwise_swap_local(
            overall_best_seq.copy(),
            deadline,
            best_seq=overall_best_seq,
            best_score=overall_best_score,
            max_trials=200_000,
        )

    # ------------------------------------------------------------------
    # 6. Final cheap simulated‑annealing walk
    if time.time() < deadline:
        print(f"[{time.time():.1f}] Final simulated‑annealing walk")
        overall_best_seq, overall_best_score = _random_local_search(
            overall_best_seq.copy(),
            deadline,
            best_seq=overall_best_seq,
            best_score=overall_best_score,
            init_temp=0.008,
            decay=0.9997,
        )

    # ------------------------------------------------------------------
    print(
        f"Final best score (raw objective): {overall_best_score:.6f} (length={len(overall_best_seq)})"
    )
    # Return a plain Python list; evaluate_sequence will apply final clipping etc.
    return overall_best_seq.tolist()
