import scipy.optimize as opt
import scipy.sparse as sp

def solve_lp_dir(f):
    n = f.shape[0]
    M = np.max(_fft_self_convolve(f))
    # Build constraint matrix: (2n-1) rows, n cols.
    # coefficients: for each row k, entries for i where 0 <= i < n and 0 <= k-i < n => i in [max(0, k-(n-1)), min(k, n-1)]
    # For each k, row slice columns i in that range have coefficient f[i].
    rows = []
    cols = []
    data = []
    for k in range(2*n - 1):
        i_start = max(0, k - (n - 1))
        i_end = min(k, n - 1)
        # number of columns = i_end - i_start + 1
        # for i in [i_start, i_end]:
        # coefficient f[i]
        length = i_end - i_start + 1
        rows.extend([k] * length)
        cols.extend(range(i_start, i_end + 1))
        data.extend([f[i] for i in range(i_start, i_end + 1)])
    A = sp.csr_matrix((data, (rows, cols)), shape=(2*n-1, n))
    b = np.full(2*n-1, M)
    c = -np.ones(n)  # maximize sum(g) -> minimize -sum(g)
    bounds = [(0, None)] * n
    res = opt.linprog(c, A_ub=A, b_ub=b, bounds=bounds, method='highs')
    if not res.success:
        # fallback maybe return current f
        return None
    g = res.x
    S = g.sum()
    if S <= 1e-12:
        return None
    g_norm = g / S
    return g_norm

"""
Improved search for a non‑negative step function that minimizes
``evaluate_sequence``.

The algorithm consists of three phases:

1️⃣  **Diverse seed pool** – Analytic families (uniform, arcsine,
    beta, edge‑weighted, sine‑powered, exponential, etc.) together with
    random Dirichlet and sparse vectors are generated for several lengths.
    The ~15 best seeds (lowest exact objective) are retained.

2️⃣  **Active‑set linear‑programming improvement** –  For each retained seed
    we repeatedly solve a tiny LP (≤ 30 constraints) that maximises the total
    mass of a candidate ``g`` under the current bound
    ``(f * g)[k] ≤ M`` where ``M`` is the present maximum of ``f*f``.
    The active‑set consists of the convolution entries that are within a
    small fraction of ``M``.  The solution ``g`` is rescaled to a probability
    vector and blended with the current iterate ``f`` using a line‑search over
    a few step sizes.  Each successful blend strictly reduces the objective.

3️⃣  **Polishing** –  A short mirror‑descent (exponentiated‑gradient) run,
    followed by the greedy pairwise‐swap and cheap simulated‑annealing
    refinements from the original code.

All operations respect the 1000 s budget and the returned list can be fed
directly to ``evaluate_sequence`` (the evaluator will normalise the sum).
"""

def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x ≥ 0, sum(x) = s}."""
    if s < 0:
        raise ValueError("simplex radius must be non‑negative")
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho_arr = np.where(u - cssv / (np.arange(1, len(v) + 1)) > 0)[0]
    if rho_arr.size == 0:
        theta = 0.0
    else:
        rho = rho_arr[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Full linear self‑convolution a*a using real FFT (returns length 2n‑1)."""
    n = a.shape[0]
    need = 2 * n - 1
    size = 1 << ((need - 1).bit_length())
    A = rfft(a, n=size)
    conv = irfft(A * A, n=size)
    return conv[:need]


def _objective(a: np.ndarray) -> float:
    """Exact objective 2·n·max(conv(a,a)) when sum(a)=1."""
    n = a.shape[0]
    b = _fft_self_convolve(a)
    return 2.0 * n * b.max()

# Analytic families ----------------------------------------------------
def _distribution_uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _distribution_arcsine(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    vals /= vals.sum()
    return vals


def _distribution_beta_sym(n: int, p: float) -> np.ndarray:
    """Symmetric beta (Beta(p, p))."""
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    w = np.exp((p - 1) * np.log(x) + (p - 1) * np.log(1.0 - x))
    s = w.sum()
    if s == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= s
    return w


def _distribution_beta_general(n: int, alpha: float, beta: float) -> np.ndarray:
    """Asymmetric beta distribution."""
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    eps = np.finfo(float).tiny
    x = np.clip(x, eps, 1 - eps)
    w = np.exp((alpha - 1) * np.log(x) + (beta - 1) * np.log(1.0 - x))
    s = w.sum()
    if s == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= s
    return w


def _distribution_edge_weighted(n: int, p: float) -> np.ndarray:
    """U‑shaped family w_i ∝ ((i+0.5)*(n‑i‑0.5))^{−p}."""
    i = np.arange(n, dtype=np.float64) + 0.5
    j = (n - np.arange(n, dtype=np.float64) - 0.5)
    w = (i * j) ** (-p)
    s = w.sum()
    if s == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= s
    return w


def _distribution_edge_inversepow(n: int, p: float) -> np.ndarray:
    """Second edge family w_i ∝ (i+0.5)^{−p} + (n−i−0.5)^{−p}."""
    i = np.arange(n, dtype=np.float64) + 0.5
    j = (n - np.arange(n, dtype=np.float64) - 0.5)
    w = i ** (-p) + j ** (-p)
    s = w.sum()
    if s == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= s
    return w


def _distribution_sine_weighted(n: int, p: float) -> np.ndarray:
    """Weight ∝ sin(π (i+½)/n)^p, vanishing at the ends."""
    theta = np.pi * (np.arange(n, dtype=np.float64) + 0.5) / n
    w = np.sin(theta) ** p
    s = w.sum()
    if s == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= s
    return w


def _distribution_tent(n: int) -> np.ndarray:
    """Triangular (tent) shape."""
    i = np.arange(n, dtype=np.float64)
    half = (n - 1) / 2.0
    w = np.where(i <= half, i + 1, n - i)
    w /= w.sum()
    return w


def _distribution_quadratic(n: int) -> np.ndarray:
    """Quadratic bell shape, symmetric."""
    i = np.arange(n, dtype=np.float64)
    half = (n - 1) / 2.0
    d = np.abs(i - half) + 1.0
    w = 1.0 / (d ** 2)
    w /= w.sum()
    return w


def _distribution_exponential_decay(n: int, rate: float) -> np.ndarray:
    """Exponential decay from the centre."""
    i = np.arange(n, dtype=np.float64)
    half = (n - 1) / 2.0
    w = np.exp(-rate * np.abs(i - half))
    w /= w.sum()
    return w


def _distribution_random_sparse(
    n: int, rng: np.random.Generator, density: float = 0.01
) -> np.ndarray:
    """Sparse random vector (given density)."""
    k = max(2, int(n * density))
    idx = rng.choice(n, size=k, replace=False)
    vals = np.zeros(n, dtype=np.float64)
    vals[idx] = rng.random(k)
    if vals.sum() == 0.0:
        vals[0] = 1.0
    vals /= vals.sum()
    return vals


def _distribution_dirichlet(n: int, rng: np.random.Generator) -> np.ndarray:
    """Random Dirichlet sample with random concentration."""
    alpha = rng.uniform(0.01, 2.0)
    a = rng.dirichlet(np.full(n, alpha))
    return a.astype(np.float64)


def _mix_with_uniform(base: np.ndarray, lam: float) -> np.ndarray:
    """Convex mix lam·U + (1‑lam)·base."""
    n = base.shape[0]
    uniform = np.full(n, 1.0 / n, dtype=np.float64)
    return lam * uniform + (1.0 - lam) * base

# Active‑set linear‑programming step ------------------------------------
def _lp_step(
    a: np.ndarray,
    active_frac: float = 0.04,
    max_active: int = 30,
) -> np.ndarray | None:
    """
    Solve a tiny LP that maximises the total mass of a candidate g
    under the current bound (a * g)[k] ≤ M for the active convolution entries.
    Returns a normalised vector g (sum = 1) or None if the LP fails.
    """
    n = a.shape[0]
    conv = _fft_self_convolve(a)
    M = conv.max()
    thresh = M * (1.0 - active_frac)

    act = np.where(conv >= thresh)[0]
    if act.size > max_active:
        # keep those with largest conv values
        top = np.argpartition(conv[act], -max_active)[-max_active:]
        act = act[top]

    # Build sparse constraint matrix: rows = active indices, cols = variables g_i
    rows = []
    cols = []
    data = []
    for row, k in enumerate(act):
        i_start = max(0, k - (n - 1))
        i_end = min(k, n - 1)
        # coeff for g_i is a_i (when i contributes to (a*g)[k])
        length = i_end - i_start + 1
        rows.extend([row] * length)
        cols.extend(range(i_start, i_end + 1))
        data.extend([a[i] for i in range(i_start, i_end + 1)])

    A = sp.csr_matrix((data, (rows, cols)), shape=(len(act), n))
    b = np.full(len(act), M, dtype=np.float64)

    # maximise sum(g)  <=> minimise -sum(g)
    c = -np.ones(n, dtype=np.float64)
    bounds = [(0.0, None)] * n

    res = opt.linprog(c, A_ub=A, b_ub=b, bounds=bounds, method="highs")
    if not res.success:
        return None

    g = res.x
    S = g.sum()
    if S <= 1e-12:
        return None
    return g / S

# Mirror‑descent (exponentiated‑gradient) – unchanged from the previous code
def _subgradient_descent_momentum(
    a: np.ndarray,
    deadline: float,
    best_seq: np.ndarray,
    best_score: float,
    lr0: float = 0.7,
    active_frac: float = 0.04,
    max_active: int = 20,
    beta: float = 0.9,
    eval_interval: int = 200,
    patience: int = 3000,
) -> tuple[np.ndarray, float]:
    n = a.shape[0]
    t = 0
    no_improve = 0
    momentum = np.zeros_like(a)
    rng = np.random.default_rng()
    ind_grid = np.arange(n, dtype=np.int32)

    while time.time() < deadline and t < 500_000:
        t += 1
        conv = _fft_self_convolve(a)
        max_c = conv.max()
        thresh = max_c * (1.0 - active_frac)

        act = np.where(conv >= thresh)[0]
        if act.size > max_active:
            top_idx = np.argpartition(conv[act], -max_active)[-max_active:]
            act = act[top_idx]

        grad = np.zeros_like(a)
        for idx in act:
            offset = idx - ind_grid
            mask = (offset >= 0) & (offset < n)
            grad[mask] += 2.0 * a[offset[mask]]

        momentum = beta * momentum + (1.0 - beta) * grad
        eta = lr0 / math.sqrt(t)
        a = a * np.exp(-eta * momentum)
        a = _project_onto_simplex(a, s=1.0)

        if t % eval_interval == 0:
            cur = _objective(a)
            if cur < best_score - 1e-14:
                best_score = cur
                best_seq = a.copy()
                no_improve = 0
                print(f"[{time.time():.1f}] it={t:,} new global best {best_score:.6f}")
            else:
                no_improve += 1

        if no_improve >= patience:
            jitter = rng.exponential(scale=0.006, size=n)
            a = a + jitter
            a = np.maximum(a, 0.0)
            a = _project_onto_simplex(a, s=1.0)
            no_improve = 0

    return best_seq, best_score

# Greedy pairwise refinement
def _pairwise_swap_local(
    a: np.ndarray,
    deadline: float,
    best_seq: np.ndarray,
    best_score: float,
    max_trials: int | None = None,
) -> tuple[np.ndarray, float]:
    n = a.shape[0]
    trials = 0
    rng = np.random.default_rng()
    while time.time() < deadline:
        if max_trials is not None and trials >= max_trials:
            break
        trials += 1
        src = np.where(a > 1e-12)[0]
        if src.size == 0:
            break
        i = rng.choice(src)
        j = rng.integers(n - 1)
        if j >= i:
            j += 1
        delta = a[i] * rng.random() * 0.35
        if delta < 1e-14:
            continue
        new_a = a.copy()
        new_a[i] -= delta
        new_a[j] += delta
        cur = _objective(new_a)
        if cur < best_score - 1e-14:
            a = new_a
            best_score = cur
            best_seq = a.copy()
            print(f"[{time.time():.1f}] swap {i}->{j} Δ={delta:.3e} new best {best_score:.6f}")
    return best_seq, best_score

# Simulated‑annealing style local search
def _random_local_search(
    a: np.ndarray,
    deadline: float,
    best_seq: np.ndarray,
    best_score: float,
    init_temp: float = 0.008,
    decay: float = 0.9997,
) -> tuple[np.ndarray, float]:
    n = a.shape[0]
    rng = np.random.default_rng()
    T = init_temp
    while time.time() < deadline:
        i = rng.integers(n)
        j = rng.integers(n - 1)
        if j >= i:
            j += 1
        delta = a[i] * rng.random() * 0.3
        if delta < 1e-14:
            continue
        new_a = a.copy()
        new_a[i] -= delta
        new_a[j] += delta
        cur = _objective(new_a)
        if cur < best_score - 1e-14:
            a = new_a
            best_score = cur
            best_seq = a.copy()
            print(f"[{time.time():.1f}] SA improve {i}->{j} Δ={delta:.3e} best {best_score:.6f}")
        else:
            if rng.random() < math.exp(-(cur - best_score) / max(T, 1e-12)):
                a = new_a
        T *= decay
    return best_seq, best_score

            np.logspace(-0.5, 0.0, 8, base=10),    # 0.316 … 1
            np.logspace(0.0, 0.5, 6, base=10),     # 1 … 3.16
        ]
    )
    lam_grid = np.linspace(0.0, 0.6, 7)
    rate_grid = np.linspace(0.001, 0.02, 6)

    for n in lengths:
        if time.time() >= deadline:
            break
        print(f"[{time.time():.1f}] scanning length {n}")

        # deterministic families
        for cand in (_distribution_uniform(n),
                     _distribution_arcsine(n),
                     _distribution_tent(n),
                     _distribution_quadratic(n)):
            sc = _objective(cand)
            top_candidates.append((sc, cand.copy()))
            if sc < overall_best_score:
                overall_best_score, overall_best_seq = sc, cand.copy()
                print(f"[{time.time():.1f}] new global best len={n} score={sc:.6f}")

        # parametrised families + mixing
        for p in p_grid:
            if time.time() >= deadline:
                break
            edge = _distribution_edge_weighted(n, p)
            edge2 = _distribution_edge_inversepow(n, p)
            sine = _distribution_sine_weighted(n, p)
            beta = _distribution_beta_sym(n, p)
            for base in (edge, edge2, sine, beta):
                for lam in lam_grid:
                    if time.time() >= deadline:
                        break
                    cand = _mix_with_uniform(base, lam)
                    sc = _objective(cand)
                    top_candidates.append((sc, cand.copy()))
                    if sc < overall_best_score:
                        overall_best_score, overall_best_seq = sc, cand.copy()
                        print(f"[{time.time():.1f}] new global best len={n} p={p:.3g} lam={lam:.2f} score={sc:.6f}")

        # exponential‑decay family
        for rate in rate_grid:
            if time.time() >= deadline:
                break
            cand = _distribution_exponential_decay(n, rate)
            sc = _objective(cand)
            top_candidates.append((sc, cand.copy()))
            if sc < overall_best_score:
                overall_best_score, overall_best_seq = sc, cand.copy()
                print(f"[{time.time():.1f}] new global best exp‑decay n={n} rate={rate:.4f} score={sc:.6f}")

        # random Dirichlet seeds
        for _ in range(6):
            if time.time() >= deadline:
                break
            a = _distribution_dirichlet(n, rng)
            sc = _objective(a)
            top_candidates.append((sc, a.copy()))
            if sc < overall_best_score:
                overall_best_score, overall_best_seq = sc, a.copy()
                print(f"[{time.time():.1f}] new global best Dirichlet len={n} score={sc:.6f}")

        # sparse random seeds
        for _ in range(6):
            if time.time() >= deadline:
                break
            a = _distribution_random_sparse(n, rng, density=0.006)
            sc = _objective(a)
            top_candidates.append((sc, a.copy()))
            if sc < overall_best_score:
                overall_best_score, overall_best_seq = sc, a.copy()
                print(f"[{time.time():.1f}] new global best sparse len={n} score={sc:.6f}")

        # asymmetric beta samples
        for _ in range(5):
            if time.time() >= deadline:
                break
            alpha = 10 ** rng.uniform(-2, 1)   # 0.01 … 10
            beta = 10 ** rng.uniform(-2, 1)
            a = _distribution_beta_general(n, alpha, beta)
            sc = _objective(a)
            top_candidates.append((sc, a.copy()))
            if sc < overall_best_score:
                overall_best_score, overall_best_seq = sc, a.copy()
                print(f"[{time.time():.1f}] new global best beta(α={alpha:.3g},β={beta:.3g}) len={n} score={sc:.6f}")

        # keep only a modest number of strong seeds
        top_candidates.sort(key=lambda x: x[0])
        top_candidates = top_candidates[:30]

    # ------------------------------------------------------------------
    # 2.  Inject a previously discovered construction if present
    try:
        prev = np.asarray(height_sequence_1, dtype=np.float64)  # type: ignore[name-defined]
        if prev.ndim == 1 and prev.size > 0:
            prev = prev / prev.sum()
            sc = _objective(prev)
            if sc < overall_best_score:
                overall_best_score, overall_best_seq = sc, prev.copy()
                print(f"[{time.time():.1f}] previous best score={sc:.6f}")
            top_candidates.append((sc, prev.copy()))
    except Exception:
        pass

    # ------------------------------------------------------------------
    # Guard against empty seed pool
    if overall_best_seq is None:
        n = 3000
        overall_best_seq = _distribution_uniform(n)
        overall_best_score = _objective(overall_best_seq)
        print(f"[{time.time():.1f}] fallback uniform score={overall_best_score:.6f}")

    # ------------------------------------------------------------------
    # 3.  Active‑set LP improvement on a handful of the strongest seeds
    # keep only the very best seeds for this expensive stage
    top_candidates.sort(key=lambda x: x[0])
    elite = [cand for _, cand in top_candidates[:8]]

    for idx, seed_vec in enumerate(elite):
        if time.time() >= deadline:
            break
        print(f"[{time.time():.1f}] LP refinement start for elite #{idx+1} (len={seed_vec.shape[0]})")
        a = seed_vec.copy()
        best_local_seq = a.copy()
        best_local_score = _objective(a)

        # iterate LP / blending steps
        lp_iter = 0
        while time.time() < deadline and lp_iter < 40:
            lp_iter += 1
            g = _lp_step(a, active_frac=0.04, max_active=20)
            if g is None:
                break

            # try a small set of mixing coefficients
            improved = False
            for t in (0.2, 0.4, 0.6, 0.8):
                new_a = (1.0 - t) * a + t * g
                # convex combination already respects simplex, only tiny numerical drift possible
                new_a = _project_onto_simplex(new_a, s=1.0)
                cur = _objective(new_a)
                if cur < best_local_score - 1e-14:
                    best_local_score = cur
                    best_local_seq = new_a.copy()
                    a = new_a
                    improved = True
                    print(f"[{time.time():.1f}] elite {idx+1} LP‑step {lp_iter} t={t:.2f} new best {cur:.6f}")
                    break  # accept first improving t
            if not improved:
                # cannot improve with current direction – stop early for this seed
                break

        # after LP phase, polish with a short mirror‑descent run
        if time.time() < deadline:
            sub_dead = time.time() + min(30.0, deadline - time.time() - 10.0)
            best_local_seq, best_local_score = _subgradient_descent_momentum(
                best_local_seq,
                sub_dead,
                best_seq=best_local_seq,
                best_score=best_local_score,
                lr0=0.6,
                active_frac=0.03,
                max_active=25,
                beta=0.9,
                eval_interval=150,
                patience=2500,
            )
        if best_local_score < overall_best_score - 1e-12:
            overall_best_score, overall_best_seq = best_local_score, best_local_seq.copy()
            print(f"[{time.time():.1f}] *** new overall best after LP+MD {overall_best_score:.6f}")

    # ------------------------------------------------------------------
    # 4.  Final polishing on the current overall best
    if time.time() < deadline:
        remaining = deadline - time.time()
        sub_dead = time.time() + remaining - 25.0
        print(f"[{time.time():.1f}] final long mirror‑descent (≈{remaining-25:.1f}s left)")
        overall_best_seq, overall_best_score = _subgradient_descent_momentum(
            overall_best_seq.copy(),
            sub_dead,
            best_seq=overall_best_seq,
            best_score=overall_best_score,
            lr0=0.5,
            active_frac=0.025,
            max_active=25,
            beta=0.9,
            eval_interval=200,
            patience=3000,
        )
        print(f"[{time.time():.1f}] after final descent best {overall_best_score:.6f}")

    if time.time() < deadline:
        print(f"[{time.time():.1f}] greedy pairwise refinement")
        overall_best_seq, overall_best_score = _pairwise_swap_local(
            overall_best_seq.copy(),
            deadline,
            best_seq=overall_best_seq,
            best_score=overall_best_score,
            max_trials=400_000,
        )

    if time.time() < deadline:
        print(f"[{time.time():.1f}] final simulated‑annealing search")
        overall_best_seq, overall_best_score = _random_local_search(
            overall_best_seq.copy(),
            deadline,
            best_seq=overall_best_seq,
            best_score=overall_best_score,
            init_temp=0.01,
            decay=0.9995,
        )

    print(f"Final best score (exact) ≈ {overall_best_score:.6f} (length={len(overall_best_seq)})")
    return overall_best_seq.tolist()
