    f = f0.copy()
    n = len(f)

    cur_score = _objective(f)

"""
Propose a step‐function that minimizes the provided ``evaluate_sequence``.

Algorithm
---------
1️⃣  Scan a few analytic families (uniform, arcsine, edge‑weighted,
    exponential‑edge, sparse random) on several modest lengths and keep
    the best few seeds.

2️⃣  For each seed run an *alternating linear‑program* improvement:
   * Let M = max(conv(f,f)).
   * Solve   max Σ g_j   s.t.   f * g ≤ M   and   g ≥ 0   (LP).
   * Normalise g to a probability vector and mix  f←(1‑t)f + t g .
   * Repeat while M decreases (t is reduced by a factor ½ if a trial
     does not improve).

   The LP constraints are the full convolution A·g ≤ M·1 with
   A[i,j]=f_i if i‑j∈[0,n‑1] else 0.  A is built once per length and
   reused via a sparse CSR matrix (≈ n² non‑zeros).

3️⃣  Polishing – run a short exact sub‑gradient (exponentiated‑gradient)
   descent on the true L∞ objective and finish with a greedy pairwise
   mass‑transfer hill‑climber.

All work respects the given time budget (default 1000 s).  The returned
list is a plain Python list; ``evaluate_sequence`` will internally
renormalise it.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Optional SciPy imports (used for the LP step)
try:
    from scipy.optimize import linprog
    from scipy.sparse import csr_matrix
    _SCIPY_AVAILABLE = True
except Exception:  # pragma: no cover
    _SCIPY_AVAILABLE = False

# ----------------------------------------------------------------------
# Helper caches for the LP matrix structure (row/col indices)
_lp_struct_cache = {}   # n -> (row_idx, col_idx)


def _get_lp_structure(n: int):
    """Return pre‑computed (row_idx, col_idx) for the convolution matrix
    of size (2n‑1) × n.  The matrix entry (k, j) equals f_{k‑j} if
    0≤k‑j<n, else 0.  Only the index pattern is cached; the actual data
    (the f‑values) are supplied during each solve.
    """
    if n not in _lp_struct_cache:
        I = np.repeat(np.arange(n, dtype=np.int32), n)      # i – source index of f
        J = np.tile(np.arange(n, dtype=np.int32), n)       # j – index of g
        row_idx = I + J                                    # convolution index
        col_idx = J                                        # column = g_j
        _lp_struct_cache[n] = (row_idx, col_idx)
    return _lp_struct_cache[n]


# ----------------------------------------------------------------------
def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Full self‑convolution a*a using a real FFT (fast, O(n log n))."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1) - 1).bit_length()   # next power of two
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[: 2 * n - 1]


def _objective(a: np.ndarray) -> float:
    """Exact objective 2·n·max(conv(a,a)) for a vector with Σa = 1."""
    n = a.shape[0]
    b = _fft_self_convolve(a)
    return float(2.0 * n * b.max())


# ----------------------------------------------------------------------
# Analytic families ----------------------------------------------------
def _distribution_uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _distribution_arcsine(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    vals /= vals.sum()
    return vals


def _distribution_edge_weighted(n: int, p: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64) + 0.5
    j = (n - np.arange(n, dtype=np.float64) - 0.5)
    w = (i * j) ** (-p)
    if w.sum() == 0.0:
        return np.full(n, 1.0 / n, dtype=np.float64)
    w /= w.sum()
    return w


def _distribution_exponential_edge(n: int, c: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    w = np.exp(-c * i) + np.exp(-c * (n - 1 - i))
    w /= w.sum()
    return w


def _distribution_random_sparse(
    n: int, rng: np.random.Generator, density: float = 0.01
) -> np.ndarray:
    """Sparse random vector with a prescribed density of non‑zeros."""
    k = max(2, int(n * density))
    idx = rng.choice(n, size=k, replace=False)
    vals = np.zeros(n, dtype=np.float64)
    vals[idx] = rng.random(k)
    s = vals.sum()
    if s == 0.0:
        vals[0] = 1.0
        s = 1.0
    vals /= s
    return vals


# ----------------------------------------------------------------------
# Linear‑program (LP) step ---------------------------------------------
def _solve_lp_max_sum(f: np.ndarray, M: float) -> np.ndarray | None:
    """Solve   max Σ g_j   s.t.   f * g ≤ M   and   g ≥ 0.
    Returns the optimal g (as a NumPy array) or ``None`` on failure.
    """
    if not _SCIPY_AVAILABLE:                     # pragma: no cover
        return None

    n = f.shape[0]
    row_idx, col_idx = _get_lp_structure(n)
    # data for the sparse matrix: coefficient at (row, col) is f_i where i = row‑col
    data = f[row_idx - col_idx]
    A = csr_matrix((data, (row_idx, col_idx)), shape=(2 * n - 1, n))

    c = -np.ones(n, dtype=np.float64)           # linprog minimizes ⇒ maximize Σg
    b = np.full(2 * n - 1, M, dtype=np.float64)

    bounds = [(0.0, None)] * n
    try:
        res = linprog(
            c,
            A_ub=A,
            b_ub=b,
            bounds=bounds,
            method="highs",
            options={"presolve": True, "disp": False},
        )
    except Exception:                           # pragma: no cover
        return None

    if not res.success or res.x is None:
        return None
    return res.x


def _lp_refine(f0: np.ndarray, time_limit: float,
               best_seq: np.ndarray, best_score: float) -> tuple[np.ndarray, float]:
    """
    Repeated LP‑based refinement of a probability vector ``f0``.
    Stops when no improvement is possible or the time budget expires.
    Returns the best vector found and its objective value.
    """
    eps = 1e-12
    f = np.maximum(f0, eps)
    f /= f.sum()
    n = f.shape[0]
    cur_score = _objective(f)
    if cur_score < best_score:
        best_score, best_seq = cur_score, f.copy()

    while time.time() < time_limit:
        # 1) current maximal self‑convolution
        conv = _fft_self_convolve(f)
        M = conv.max()

        # 2) LP to maximise total mass under the same peak M
        g = _solve_lp_max_sum(f, M)
        if g is None or g.sum() == 0.0:
            break

        # 3) Normalise g to a probability vector
        g = np.maximum(g, eps)
        g /= g.sum()

        # 4) Try a convex blend; reduce the step size if no improvement
        improved = False
        t = 0.5
        while t > 1e-3:
            new_f = (1.0 - t) * f + t * g
            new_f = np.maximum(new_f, eps)
            new_f /= new_f.sum()
            new_M = _fft_self_convolve(new_f).max()
            if new_M < M - 1e-14:      # strict improvement
                f = new_f
                M = new_M
                improved = True
                break
            t *= 0.5

        if not improved:
            break   # cannot improve further

        # 5) Record global best if applicable
        cur_score = 2.0 * n * M   # objective when Σf = 1
        if cur_score < best_score - 1e-12:
            best_score, best_seq = cur_score, f.copy()
            print(f"[{time.time():.1f}] LP refine new global best {best_score:.6f}")

    return best_seq, best_score

# Exact mirror‑descent (sub‑gradient) ----------------------------------
def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x≥0, Σx=s}."""
    if s < 0:
        raise ValueError("Simplex radius must be non‑negative")
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.where(u - cssv / (np.arange(1, len(v) + 1)) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    return np.maximum(v - theta, 0.0)


def _subgradient_descent_exact(
    a0: np.ndarray,
    time_limit: float,
    best_seq: np.ndarray,
    best_score: float,
    lr0: float = 0.9,
    eval_every: int = 200,
) -> tuple[np.ndarray, float]:
    """Exponentiated‑gradient (mirror) descent on the exact L∞ objective."""
    a = a0.copy()
    n = a.shape[0]
    t = 0
    while time.time() < time_limit:
        t += 1
        conv = _fft_self_convolve(a)
        max_val = conv.max()
        max_idxs = np.where(conv == max_val)[0]

        # exact sub‑gradient (sum over all maximal indices)
        grad = np.zeros_like(a)
        for idx in max_idxs:
            i_start = max(0, idx - (n - 1))
            i_end = min(n - 1, idx)
            i = np.arange(i_start, i_end + 1, dtype=np.int64)
            j = idx - i
            grad[i] += 2.0 * a[j]

        eta = lr0 / math.sqrt(t)
        a *= np.exp(-eta * grad)          # positivity kept
        s = a.sum()
        if s <= 0.0:
            a = np.full(n, 1.0 / n, dtype=np.float64)
        else:
            a /= s                          # simplex normalisation

        if t % eval_every == 0:
            cur = 2.0 * n * max_val        # Σa = 1
            if cur < best_score - 1e-14:
                best_score = cur
                best_seq = a.copy()
                print(
                    f"[{time.time():.1f}] subgrad it={t} new global best {best_score:.6f}"
                )
    return best_seq, best_score

# Greedy pairwise polishing -------------------------------------------
def _pairwise_swap(
    a0: np.ndarray,
    time_limit: float,
    best_seq: np.ndarray,
    best_score: float,
    max_trials: int | None = None,
) -> tuple[np.ndarray, float]:
    """Random pairwise mass transfers that accept only strictly improving moves."""
    a = a0.copy()
    n = a.shape[0]
    trials = 0
    while time.time() < time_limit:
        if max_trials is not None and trials >= max_trials:
            break
        trials += 1

        src_candidates = np.where(a > 1e-14)[0]
        if src_candidates.size == 0:
            break
        i = np.random.choice(src_candidates)
        j = np.random.randint(n - 1)
        if j >= i:
            j += 1

        delta = a[i] * np.random.rand() * 0.45   # move up to 45 % of source mass
        if delta < 1e-15:
            continue

        new_a = a.copy()
        new_a[i] -= delta
        new_a[j] += delta

        cur = _objective(new_a)
        if cur < best_score - 1e-14:
            a = new_a
            best_score = cur
            best_seq = a.copy()
            print(
                f"[{time.time():.1f}] swap i={i}→j={j} δ={delta:.6g} new best {best_score:.6f}"
            )
    return best_seq, best_score

def propose_candidate(seed: int = 42, budget_s: int = 1000) -> list[float]:
    """
    Return a step‑function (plain Python list) that minimises
    ``evaluate_sequence`` within the given time budget.
    """
    rng = np.random.default_rng(seed)
    start = time.time()
    deadline = start + budget_s - 1.0   # small safety margin

    overall_best_seq = None
    overall_best_score = float("inf")
    candidate_pool = []   # list of (score, np.ndarray)

    # ------------------------------------------------------------------
    # 1. Analytic families on a few modest lengths
    # ------------------------------------------------------------------
    lengths = [1200, 1500, 2000]

    # edge‑weight exponent grid (few representative values)
    p_grid = [0.1, 0.3, 0.5, 0.8, 1.2, 1.8]
    # exponential‑edge decay grid
    c_grid = [0.01, 0.04, 0.1, 0.2, 0.5]

    for n in lengths:
        if time.time() >= deadline:
            break
        print(f"[{time.time():.1f}] scanning length {n}")

        # deterministic families
        for base_fn in (
            _distribution_uniform,
            _distribution_arcsine,
        ):
            cand = base_fn(n)
            cand = np.maximum(cand, 1e-12)
            cand /= cand.sum()
            sc = _objective(cand)
            candidate_pool.append((sc, cand.copy()))
            if sc < overall_best_score:
                overall_best_score, overall_best_seq = sc, cand.copy()
                print(f"[{time.time():.1f}] new global best len={n} score={sc:.6f}")

        # edge‑weighted families
        for p in p_grid:
            if time.time() >= deadline:
                break
            edge = _distribution_edge_weighted(n, float(p))
            edge = np.maximum(edge, 1e-12)
            edge /= edge.sum()
            sc = _objective(edge)
            candidate_pool.append((sc, edge.copy()))
            if sc < overall_best_score:
                overall_best_score, overall_best_seq = sc, edge.copy()
                print(f"[{time.time():.1f}] len={n} edge p={p:.3g} score={sc:.6f}")

        # exponential‑edge families
        for c in c_grid:
            if time.time() >= deadline:
                break
            exp_edge = _distribution_exponential_edge(n, float(c))
            exp_edge = np.maximum(exp_edge, 1e-12)
            exp_edge /= exp_edge.sum()
            sc = _objective(exp_edge)
            candidate_pool.append((sc, exp_edge.copy()))
            if sc < overall_best_score:
                overall_best_score, overall_best_seq = sc, exp_edge.copy()
                print(f"[{time.time():.1f}] len={n} exp c={c:.3g} score={sc:.6f}")

        # a few random Dirichlet draws
        for _ in range(4):
            if time.time() >= deadline:
                break
            alpha = rng.uniform(0.05, 1.5)
            a = rng.dirichlet(np.full(n, alpha))
            a = np.maximum(a, 1e-12)
            a /= a.sum()
            sc = _objective(a)
            candidate_pool.append((sc, a.copy()))
            if sc < overall_best_score:
                overall_best_score, overall_best_seq = sc, a.copy()
                print(f"[{time.time():.1f}] Dirichlet len={n} score={sc:.6f}")

        # sparse random seeds
        for _ in range(3):
            if time.time() >= deadline:
                break
            a = _distribution_random_sparse(n, rng, density=0.008)
            a = np.maximum(a, 1e-12)
            a /= a.sum()
            sc = _objective(a)
            candidate_pool.append((sc, a.copy()))
            if sc < overall_best_score:
                overall_best_score, overall_best_seq = sc, a.copy()
                print(f"[{time.time():.1f}] sparse len={n} score={sc:.6f}")

    # keep the best few seeds for deeper work
    candidate_pool.sort(key=lambda x: x[0])
    top_seeds = [seq for _, seq in candidate_pool[:6]]

    # ------------------------------------------------------------------
    # 2. Optional: use a previous construction if available
    # ------------------------------------------------------------------
    try:
        prev = np.asarray(height_sequence_1, dtype=np.float64)  # type: ignore[name-defined]
        if prev.ndim == 1 and prev.size > 0:
            prev = np.maximum(prev, 1e-12)
            if prev.sum() == 0.0:
                prev = _distribution_uniform(prev.size)
            else:
                prev = prev / prev.sum()
            sc = _objective(prev)
            if sc < overall_best_score:
                overall_best_score, overall_best_seq = sc, prev.copy()
                print(f"[{time.time():.1f}] previous construction score={sc:.6f}")
            top_seeds.append(prev.copy())
    except Exception:
        pass

    # ------------------------------------------------------------------
    # 3. LP‑based alternating refinement on top seeds
    # ------------------------------------------------------------------
    remaining = deadline - time.time()
    if remaining > 30.0 and top_seeds and _SCIPY_AVAILABLE:
        per_seed = remaining / max(1, len(top_seeds))
        print(f"[{time.time():.1f}] start LP alternating refinement on {len(top_seeds)} seeds")
        for idx, seed_seq in enumerate(top_seeds):
            if time.time() >= deadline:
                break
            seed_deadline = time.time() + per_seed
            f_opt, sc_opt = _lp_refine(
                seed_seq,
                seed_deadline,
                overall_best_seq,
                overall_best_score,
            )
            if sc_opt < overall_best_score - 1e-12:
                overall_best_score, overall_best_seq = sc_opt, f_opt.copy()
                print(
                    f"[{time.time():.1f}] after LP seed {idx} new global best {overall_best_score:.6f}"
                )

    # ------------------------------------------------------------------
    # 4. Sub‑gradient descent (short) for polishing
    # ------------------------------------------------------------------
    if time.time() < deadline:
        subgrad_time = deadline - time.time()
        print(f"[{time.time():.1f}] final sub‑gradient polishing (≈{subgrad_time:.1f}s left)")
        overall_best_seq, overall_best_score = _subgradient_descent_exact(
            overall_best_seq.copy(),
            subgrad_time,
            best_seq=overall_best_seq,
            best_score=overall_best_score,
            lr0=0.9,
            eval_every=150,
        )

    # ------------------------------------------------------------------
    # 5. Greedy pairwise hill‑climber (final polish)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        print(f"[{time.time():.1f}] final greedy pairwise polishing")
        overall_best_seq, overall_best_score = _pairwise_swap(
            overall_best_seq.copy(),
            deadline,
            best_seq=overall_best_seq,
            best_score=overall_best_score,
            max_trials=None,
        )

    # ------------------------------------------------------------------
    # 6. Report and return
    # ------------------------------------------------------------------
    print(
        f"Final best objective (2·n·max(conv)) = {overall_best_score:.6f} (length={len(overall_best_seq)})"
    )
    # Return a plain Python list – ``evaluate_sequence`` will handle its own normalisation.
    return overall_best_seq.tolist()
