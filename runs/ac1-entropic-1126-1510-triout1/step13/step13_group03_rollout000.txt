def _fft_convolve(x,y):
    size = 1 << ((len(x)+len(y)-1)-1).bit_length()
    X = np.fft.rfft(x, n=size)
    Y = np.fft.rfft(y, n=size)
    conv = np.fft.irfft(X * Y, n=size)
    return conv[:len(x)+len(y)-1]

"""
Propose a step‑function (non‑negative list) that minimizes
``evaluate_sequence``.

Algorithm
---------
1. **Exploratory scan** – generate a modest collection of analytic
   distributions (uniform, arcsine, edge‑weighted, exponential‑edge,
   beta‑type and a few random Dirichlet vectors) for several lengths
   (up to 30 000).  Keep the best ~10 candidates (lowest exact objective).

2. **Smooth L∞ descent** – for each top candidate run a gradient
   descent on a smooth approximation of the L∞‑norm:
   Gε(a)=ε·log∑exp(conv(a,a)/ε).  The gradient is obtained by a
   weighted convolution, computed with FFTs.  The iterate is updated
   with an exponentiated‑gradient step and projected back onto the
   simplex (normalisation).  ε is decreased geometrically during the
   run.

3. **Pairwise mass‑transfer polishing** – a cheap hill‑climber that
   moves a random fraction of mass from one index to another whenever
   it strictly improves the exact objective.

All work respects the overall time budget (default 1000 s).  The
returned list is normalised to sum = 1, which is safe because the
objective is scale‑invariant.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x≥0, Σx = s}."""
    if s < 0:
        raise ValueError("Simplex radius must be non‑negative")
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.where(u - cssv / (np.arange(1, len(v) + 1)) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    return np.maximum(v - theta, 0.0)


# ----------------------------------------------------------------------
def _fft_convolve(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full convolution of two real vectors via FFT."""
    m = len(x) + len(y) - 1
    size = 1 << ((m - 1).bit_length())          # next power of two
    X = np.fft.rfft(x, n=size)
    Y = np.fft.rfft(y, n=size)
    conv = np.fft.irfft(X * Y, n=size)
    return conv[:m]


def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Full self‑convolution a*a using a real FFT."""
    return _fft_convolve(a, a)


# ----------------------------------------------------------------------
def _objective(a: np.ndarray) -> float:
    """Exact objective 2·n·max(conv(a,a)) for a vector with Σa = 1."""
    n = a.shape[0]
    conv = _fft_self_convolve(a)
    return float(2.0 * n * conv.max())


# ----------------------------------------------------------------------
# Analytic families ----------------------------------------------------
def _distribution_uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _distribution_arcsine(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    vals /= vals.sum()
    return vals


def _distribution_edge_weighted(n: int, p: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64) + 0.5
    j = (n - np.arange(n, dtype=np.float64) - 0.5)
    w = (i * j) ** (-p)
    w_sum = w.sum()
    if w_sum == 0.0:
        return np.full(n, 1.0 / n, dtype=np.float64)
    w /= w_sum
    return w


def _distribution_exponential_edge(n: int, c: float) -> np.ndarray:
    """w_i ∝ exp(-c·i) + exp(-c·(n‑1‑i))   (c>0)."""
    i = np.arange(n, dtype=np.float64)
    w = np.exp(-c * i) + np.exp(-c * (n - 1 - i))
    w /= w.sum()
    return w


def _distribution_beta(n: int, alpha: float, beta: float) -> np.ndarray:
    """Beta‑type density on the discrete grid."""
    i = np.arange(n, dtype=np.float64) + 0.5
    x = i / n
    # avoid overflow/underflow by working in log‑space
    log_w = (alpha - 1) * np.log(x) + (beta - 1) * np.log(1 - x)
    # set any -inf (when x==0 or 1) to a very large negative number
    log_w = np.where(np.isfinite(log_w), log_w, -1e300)
    w = np.exp(log_w - np.max(log_w))
    w /= w.sum()
    return w


def _mix_with_uniform(base: np.ndarray, lam: float) -> np.ndarray:
    """Convex combination λ·uniform + (1‑λ)·base."""
    n = base.shape[0]
    uniform = np.full(n, 1.0 / n, dtype=np.float64)
    return lam * uniform + (1.0 - lam) * base


# ----------------------------------------------------------------------
# Smooth L∞ approximation ------------------------------------------------
def _softmax_weights(conv: np.ndarray, eps: float) -> np.ndarray:
    """Compute w_i = exp(conv_i/eps) / Σ exp(...).  Stable version."""
    # subtract max for numerical stability
    max_c = conv.max()
    scaled = (conv - max_c) / eps
    exp_scaled = np.exp(scaled)
    w = exp_scaled / exp_scaled.sum()
    return w


def _grad_from_weights(a: np.ndarray, w: np.ndarray) -> np.ndarray:
    """
    Gradient of Gε(a) = ε·log Σ exp(conv/ε) with respect to a,
    assuming Σ a = 1.  Result: 2 * convolution(reversed(a), w)
    restricted to the middle n entries.
    """
    rev = a[::-1]
    conv_rev = _fft_convolve(rev, w)           # length = len(a)+len(w)-1 = 3n-2
    start = len(a) - 1                         # n‑1
    grad = 2.0 * conv_rev[start:start + len(a)]
    return grad


# ----------------------------------------------------------------------
def _smooth_descent(
    a0: np.ndarray,
    deadline: float,
    best_seq: np.ndarray,
    best_score: float,
    eps0: float = 0.1,
    eps_decay: float = 0.98,
    lr0: float = 0.5,
    eval_every: int = 50,
) -> tuple[np.ndarray, float]:
    """
    Gradient descent on the smooth L∞ approximation.
    Updates are performed with an exponentiated‑gradient step and simplex
    normalisation.  ``eps`` (softness) is reduced geometrically.
    """
    a = a0.copy()
    n = a.shape[0]
    t = 0
    eps = eps0
    while time.time() < deadline:
        t += 1

        # compute convolution and smooth weights
        conv = _fft_self_convolve(a)
        max_conv = conv.max()
        w = _softmax_weights(conv, eps)

        # gradient (exact for the smooth objective)
        grad = _grad_from_weights(a, w)

        # step size schedule
        eta = lr0 / math.sqrt(t)

        # EG update (clip exponent to avoid overflow)
        exponent = -eta * grad
        exponent = np.clip(exponent, -50.0, 50.0)
        a *= np.exp(exponent)

        # simplex projection (just normalise, non‑negativity already holds)
        s = a.sum()
        if s > 0.0:
            a /= s
        else:
            a = np.full(n, 1.0 / n, dtype=np.float64)

        # occasional reporting
        if t % eval_every == 0:
            cur_obj = 2.0 * n * max_conv          # Σa = 1
            if cur_obj < best_score - 1e-14:
                best_score = cur_obj
                best_seq = a.copy()
                print(
                    f"[{time.time():.1f}] smooth descent it={t} eps={eps:.3e} new best {best_score:.6f}"
                )

        # decay epsilon slowly (but never let it vanish)
        eps = max(eps * eps_decay, 1e-6)

    return best_seq, best_score


# ----------------------------------------------------------------------
def _pairwise_swap(
    a0: np.ndarray,
    deadline: float,
    best_seq: np.ndarray,
    best_score: float,
    max_trials: int | None = None,
) -> tuple[np.ndarray, float]:
    """
    Simple hill‑climber that transfers a random fraction of mass from one
    index to another whenever the exact objective improves.
    """
    a = a0.copy()
    n = a.shape[0]
    trials = 0
    while time.time() < deadline:
        if max_trials is not None and trials >= max_trials:
            break
        trials += 1

        # choose a source with non‑trivial mass
        src_candidates = np.where(a > 1e-14)[0]
        if src_candidates.size == 0:
            break
        i = np.random.choice(src_candidates)
        # choose a different target
        j = np.random.randint(n - 1)
        if j >= i:
            j += 1

        # mass to move (up to 45 % of source)
        delta = a[i] * np.random.rand() * 0.45
        if delta < 1e-15:
            continue

        new_a = a.copy()
        new_a[i] -= delta
        new_a[j] += delta

        cur = _objective(new_a)
        if cur < best_score - 1e-14:
            a = new_a
            best_score = cur
            best_seq = a.copy()
            print(
                f"[{time.time():.1f}] swap i={i}→j={j} δ={delta:.6g} new best {best_score:.6f}"
            )
    return best_seq, best_score
