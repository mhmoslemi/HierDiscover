"""
Non‑negative step‑function optimizer.

Main idea
---------
The true objective   J(a)=2·n·max_k (a*a)[k] / (∑a)^2
is scale‑invariant, so we restrict to probability vectors (∑a=1).
The inner‐max is a non‑smooth convex function.
We replace the hard max by a smooth log‑sum‑exp approximation

    J_β(a)=2·n· (1/β)·log Σ_k exp(β·(a*a)[k]),      β≫1

Its gradient can be written in closed form using one additional
convolution, which allows cheap O(n log n) updates via FFT.
A multiplicative (exponentiated) mirror‑descent step on this smooth
objective followed by a projection onto the simplex yields a fast
way to push the mass away from the currently largest convolution
entries while keeping the iterate feasible.

Algorithm (≈1000 s budget)
---------------------------
1. Build a diverse pool of seed probability vectors from several
   analytic families (uniform, arcsine, beta, edge‑weighted, Hann,
   generalized‑normal, power‑law, Dirichlet, sparse).  Keep the
   best 15 seeds.

2. For each seed run *smooth* exponentiated‑gradient descent.
   β is increased gradually (β∈[0.5,200]) so the method first
   smooths the landscape and later focuses on the true maximum.
   The best sequence over all runs is retained.

3. If time permits, up‑sample the current best distribution
   (linear interpolation) to roughly double the length and repeat
   a short burst of smooth descent – the larger dimension offers
   a finer discretisation of the optimal shape.

4. Final polishing by a simple pairwise mass‑transfer annealing
   (accept‑if‑improve + occasional uphill moves).

All steps respect the overall time budget; intermediate progress
(and any new global best) is printed.  The routine returns a plain
list of floats ready for ``evaluate_sequence``.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Simplex utilities
# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x≥0, Σx=s}."""
    if s < 0:
        raise ValueError("simplex radius must be non‑negative")
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.where(u - cssv / (np.arange(1, len(v) + 1)) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Convolution helpers (FFT based)
# ----------------------------------------------------------------------
def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Full linear self‑convolution a*a using a real FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1) - 1).bit_length()
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[: 2 * n - 1]


def _fft_convolve(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Linear convolution of x and y via FFT."""
    m = len(x) + len(y) - 1
    size = 1 << (m - 1).bit_length()
    X = np.fft.rfft(x, n=size)
    Y = np.fft.rfft(y, n=size)
    conv = np.fft.irfft(X * Y, n=size)
    return conv[:m]


# ----------------------------------------------------------------------
# Exact objective used for reporting (scale‑invariant)
# ----------------------------------------------------------------------
def _objective(a: np.ndarray) -> float:
    """Return the exact value of ``evaluate_sequence`` for a (unnormalised)."""
    n = a.shape[0]
    b = _fft_self_convolve(a)
    max_b = b.max()
    sum_a = a.sum()
    if sum_a < 0.01:
        return float("inf")
    return float(2.0 * n * max_b / (sum_a * sum_a))


# ----------------------------------------------------------------------
# Analytic families – always return a probability vector (sum = 1)
# ----------------------------------------------------------------------
def _distribution_uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _distribution_arcsine(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    vals /= vals.sum()
    return vals


def _distribution_beta_sym(n: int, p: float) -> np.ndarray:
    """Symmetric Beta(p,p) on a uniform grid."""
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    w = np.exp((p - 1) * np.log(x) + (p - 1) * np.log(1.0 - x))
    if w.sum() == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _distribution_edge_weighted(n: int, p: float) -> np.ndarray:
    """Weight ∝ (i·j)^{-p} where i,j are distances to the two edges."""
    i = np.arange(n, dtype=np.float64) + 0.5
    j = (n - np.arange(n, dtype=np.float64) - 0.5)
    w = (i * j) ** (-p)
    if w.sum() == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _distribution_hann(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    w = 0.5 - 0.5 * np.cos(2.0 * np.pi * i / (n - 1))
    w_sum = w.sum()
    if w_sum == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w_sum
    return w


def _distribution_generalized_normal(n: int, q: float) -> np.ndarray:
    """Density ∝ exp(-|x-½|^{q}) on [0,1]."""
    i = np.arange(n, dtype=np.float64) + 0.5
    x = i / n - 0.5
    w = np.exp(- (np.abs(x) / 0.5) ** q)
    w_sum = w.sum()
    if w_sum == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w_sum
    return w


def _distribution_power_law(n: int, p: float) -> np.ndarray:
    """Monotone decreasing power‑law (p>0)."""
    i = np.arange(1, n + 1, dtype=np.float64)
    w = i ** (-p)
    w /= w.sum()
    return w


def _distribution_random_sparse(n: int, rng: np.random.Generator, density: float = 0.01) -> np.ndarray:
    """Sparse random vector with a given density."""
    k = max(2, int(n * density))
    idx = rng.choice(n, size=k, replace=False)
    vals = np.zeros(n, dtype=np.float64)
    vals[idx] = rng.random(k)
    if vals.sum() == 0.0:
        vals[0] = 1.0
    vals /= vals.sum()
    return vals


def _mix_with_uniform(base: np.ndarray, lam: float) -> np.ndarray:
    """Convex combination with the uniform distribution."""
    n = base.shape[0]
    uniform = np.full(n, 1.0 / n, dtype=np.float64)
    return lam * uniform + (1.0 - lam) * base


# ----------------------------------------------------------------------
# Gradient of the smoothed max‑convolution (log‑sum‑exp)
# ----------------------------------------------------------------------
def _smooth_gradient(a: np.ndarray, beta: float) -> np.ndarray:
    """
    Compute ∇_a  (1/β)·log Σ_k exp(β·(a*a)[k]).
    Returns a vector of length n.
    """
    n = a.shape[0]
    # 1) self‑convolution (a*a)
    S = _fft_self_convolve(a)                     # shape 2n‑1
    maxS = np.max(S)
    # 2) softmax weights over all convolution entries
    exp_term = np.exp(beta * (S - maxS))
    w = exp_term / np.sum(exp_term)               # Σ w = 1, shape 2n‑1

    # 3) gradient = 2 * (a_rev * w) restricted to n entries
    a_rev = a[::-1]
    conv_rev = _fft_convolve(a_rev, w)            # length 3n‑2
    # slice indices n‑1 … n‑1+n‑1   (inclusive of start, exclusive of end)
    grad = 2.0 * conv_rev[n - 1 : n - 1 + n]
    return grad


# ----------------------------------------------------------------------
# Smooth exponentiated‑gradient descent (mirror descent on the simplex)
# ----------------------------------------------------------------------
def _smooth_exp_gradient_descent(
    a: np.ndarray,
    deadline: float,
    best_seq: np.ndarray,
    best_score: float,
    eta0: float = 1.0,
    beta0: float = 0.5,
    beta_max: float = 200.0,
    eval_every: int = 80,
) -> tuple[np.ndarray, float]:
    """
    Multiplicative update a←a·exp(-η·∇J_β) with a β schedule that
    gradually sharpens the log‑sum‑exp approximation.
    """
    n = a.shape[0]
    t = 0
    while time.time() < deadline:
        t += 1
        # linearly increase β (capped)
        beta = beta0 + (beta_max - beta0) * min(1.0, t / 2000.0)
        grad = _smooth_gradient(a, beta)
        eta = eta0 / math.sqrt(t)
        # multiplicative mirror‑descent step
        a = a * np.exp(-eta * grad)
        # renormalise to the simplex
        sum_a = a.sum()
        if sum_a == 0.0:
            a = np.full(n, 1.0 / n, dtype=np.float64)
        else:
            a = a / sum_a

        if t % eval_every == 0:
            cur = _objective(a)
            if cur < best_score - 1e-14:
                best_score = cur
                best_seq = a.copy()
                print(
                    f"[{time.time():.1f}] SmoothEG it={t:6d} β={beta:6.2f} best={best_score:.6f}"
                )
    return best_seq, best_score


# ----------------------------------------------------------------------
# Pairwise annealing (final polishing)
# ----------------------------------------------------------------------
def _pairwise_annealing(
    a: np.ndarray,
    deadline: float,
    best_seq: np.ndarray,
    best_score: float,
    rng: np.random.Generator,
    max_trials: int | None = None,
) -> tuple[np.ndarray, float]:
    """Randomly move a fraction of mass between two bins; accept if improves."""
    n = a.shape[0]
    start_time = time.time()
    total_time = deadline - start_time
    trials = 0
    while time.time() < deadline:
        if max_trials is not None and trials >= max_trials:
            break
        trials += 1

        src = rng.choice(np.where(a > 1e-12)[0])
        # pick a different target uniformly
        tgt = rng.integers(n - 1)
        if tgt >= src:
            tgt += 1

        delta = a[src] * rng.random() * 0.4
        if delta < 1e-16:
            continue

        new_a = a.copy()
        new_a[src] -= delta
        new_a[tgt] += delta

        cur = _objective(new_a)
        elapsed = time.time() - start_time
        temperature = max(0.0, 1.0 - elapsed / total_time) * best_score * 0.01

        if cur < best_score - 1e-14:
            accept = True
        else:
            accept = temperature > 0.0 and rng.random() < math.exp(-(cur - best_score) / temperature)

        if accept:
            a = new_a
            if cur < best_score - 1e-14:
                best_score = cur
                best_seq = a.copy()
                print(
                    f"[{time.time():.1f}] anneal src={src}->tgt={tgt} δ={delta:.3e} new best {best_score:.6f}"
                )
    return best_seq, best_score


# ----------------------------------------------------------------------
# Linear up‑sampling while preserving the simplex
# ----------------------------------------------------------------------
def _upsample(a: np.ndarray, new_n: int) -> np.ndarray:
    """Interpolate a probability vector to a larger length."""
    old_n = a.shape[0]
    if new_n == old_n:
        return a.copy()
    x_old = (np.arange(old_n, dtype=np.float64) + 0.5) / old_n
    x_new = (np.arange(new_n, dtype=np.float64) + 0.5) / new_n
    a_new = np.interp(x_new, x_old, a, left=0.0, right=0.0)
    a_new = np.maximum(a_new, 0.0)
    if a_new.sum() == 0.0:
        a_new[:] = 1.0 / new_n
    else:
        a_new /= a_new.sum()
    return a_new


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: int = 1000) -> list[float]:
    """
    Search for a non‑negative step function that minimises ``evaluate_sequence``.

    Returns a plain list of floats (a probability vector).  The routine
    respects the 1000 s total budget; intermediate progress is printed.
    """
    rng = np.random.default_rng(seed)
    start = time.time()
    deadline = start + budget_s - 2.0  # 2 s safety margin

    overall_best_seq: np.ndarray | None = None
    overall_best_score = float("inf")
    top_candidates: list[tuple[float, np.ndarray]] = []

    # ------------------------------------------------------------------
    # 1.  Build a diverse set of seed distributions.
    # ------------------------------------------------------------------
    lengths = [2000, 4000, 8000]         # we keep them moderate for speed
    p_grid = np.concatenate(
        [
            np.logspace(-2.0, -0.5, 8, base=10),
            np.logspace(-0.5, 0.0, 8, base=10),
            np.logspace(0.0, 0.7, 6, base=10),
        ]
    )
    lam_grid = np.linspace(0.0, 1.0, 4)
    q_grid = np.linspace(0.5, 3.0, 5)
    power_grid = np.linspace(0.2, 2.5, 6)

    for n in lengths:
        if time.time() >= deadline:
            break
        print(f"[{time.time():.1f}] generating seeds for length {n}")

        # deterministic families
        for cand in (_distribution_uniform(n), _distribution_arcsine(n), _distribution_hann(n)):
            sc = _objective(cand)
            top_candidates.append((sc, cand.copy()))
            if sc < overall_best_score:
                overall_best_score = sc
                overall_best_seq = cand.copy()
                print(f"[{time.time():.1f}] new global best (det) len={n} score={sc:.6f}")

        # beta & edge‑weighted families, mixed with uniform
        for p in p_grid:
            if time.time() >= deadline:
                break
            beta = _distribution_beta_sym(n, p)
            edge = _distribution_edge_weighted(n, p)
            for base in (beta, edge):
                for lam in lam_grid:
                    if time.time() >= deadline:
                        break
                    cand = _mix_with_uniform(base, lam)
                    sc = _objective(cand)
                    top_candidates.append((sc, cand.copy()))
                    if sc < overall_best_score:
                        overall_best_score = sc
                        overall_best_seq = cand.copy()
                        print(
                            f"[{time.time():.1f}] new best (beta/edge) n={n} p={p:.3g} lam={lam:.2f} score={sc:.6f}"
                        )

        # generalized normal
        for q in q_grid:
            if time.time() >= deadline:
                break
            cand = _distribution_generalized_normal(n, q)
            sc = _objective(cand)
            top_candidates.append((sc, cand.copy()))
            if sc < overall_best_score:
                overall_best_score = sc
                overall_best_seq = cand.copy()
                print(f"[{time.time():.1f}] new best (gn) n={n} q={q:.2f} score={sc:.6f}")

        # power‑law
        for pwr in power_grid:
            if time.time() >= deadline:
                break
            cand = _distribution_power_law(n, pwr)
            sc = _objective(cand)
            top_candidates.append((sc, cand.copy()))
            if sc < overall_best_score:
                overall_best_score = sc
                overall_best_seq = cand.copy()
                print(f"[{time.time():.1f}] new best (pl) n={n} p={pwr:.2f} score={sc:.6f}")

        # random Dirichlet seeds
        for _ in range(4):
            if time.time() >= deadline:
                break
            alpha = rng.uniform(0.01, 2.0)
            cand = rng.dirichlet(np.full(n, alpha))
            sc = _objective(cand)
            top_candidates.append((sc, cand.copy()))
            if sc < overall_best_score:
                overall_best_score = sc
                overall_best_seq = cand.copy()
                print(f"[{time.time():.1f}] new best (dir) n={n} score={sc:.6f}")

        # sparse random seeds
        for _ in range(4):
            if time.time() >= deadline:
                break
            cand = _distribution_random_sparse(n, rng, density=0.01)
            sc = _objective(cand)
            top_candidates.append((sc, cand.copy()))
            if sc < overall_best_score:
                overall_best_score = sc
                overall_best_seq = cand.copy()
                print(f"[{time.time():.1f}] new best (sparse) n={n} score={sc:.6f}")

        # keep only the best few (15)
        top_candidates.sort(key=lambda x: x[0])
        top_candidates = top_candidates[:15]

    # ------------------------------------------------------------------
    # 2.  Incorporate any previously known construction (if present).
    # ------------------------------------------------------------------
    try:
        prev = np.asarray(height_sequence_1, dtype=np.float64)  # type: ignore[name-defined]
        if prev.ndim == 1:
            prev = prev / prev.sum()
            sc = _objective(prev)
            if sc < overall_best_score:
                overall_best_score = sc
                overall_best_seq = prev.copy()
                print(f"[{time.time():.1f}] previous-best score={sc:.6f}")
            top_candidates.append((sc, prev.copy()))
    except Exception:
        pass

    # ------------------------------------------------------------------
    # 3.  Fallback in the improbable case of an empty candidate set.
    # ------------------------------------------------------------------
    if overall_best_seq is None:
        n = 4000
        overall_best_seq = _distribution_uniform(n)
        overall_best_score = _objective(overall_best_seq)
        print(f"[{time.time():.1f}] fallback uniform score={overall_best_score:.6f}")

    # ------------------------------------------------------------------
    # 4.  Smooth exponentiated‑gradient descent on the top seeds.
    # ------------------------------------------------------------------
    now = time.time()
    remaining = deadline - now
    if remaining > 30.0 and top_candidates:
        # allocate roughly 30‑40 % of the remaining budget to this phase
        budget_phase = remaining * 0.35
        # split equally among the best K seeds (K≈5)
        seed_list = [vec for _, vec in sorted(top_candidates, key=lambda x: x[0])[:5]]
        per_seed = budget_phase / max(1, len(seed_list))
        print(f"[{time.time():.1f}] smooth‑EG on {len(seed_list)} seeds, {per_seed:.1f}s each")
        for idx, seed_vec in enumerate(seed_list):
            if time.time() >= deadline:
                break
            sub_deadline = time.time() + per_seed
            a_opt, sc_opt = _smooth_exp_gradient_descent(
                seed_vec.copy(),
                sub_deadline,
                best_seq=overall_best_seq,
                best_score=overall_best_score,
                eta0=1.0,
                beta0=0.5,
                beta_max=200.0,
                eval_every=80,
            )
            if sc_opt < overall_best_score - 1e-12:
                overall_best_score = sc_opt
                overall_best_seq = a_opt.copy()
                print(
                    f"[{time.time():.1f}] after smooth‑EG seed {idx} new global best {overall_best_score:.6f}"
                )

    # ------------------------------------------------------------------
    # 5.  Upsample to a larger length and run a short smooth‑EG burst.
    # ------------------------------------------------------------------
    if time.time() < deadline:
        cur_n = overall_best_seq.shape[0]
        target_n = cur_n * 2
        if target_n <= 64000:
            print(f"[{time.time():.1f}] upsampling {cur_n} → {target_n}")
            up_seq = _upsample(overall_best_seq, target_n)
            up_sc = _objective(up_seq)
            if up_sc < overall_best_score:
                overall_best_score = up_sc
                overall_best_seq = up_seq
                print(f"[{time.time():.1f}] upsampled new best {overall_best_score:.6f}")

            # short smooth‑EG on the larger vector
            sub_deadline = time.time() + (deadline - time.time()) * 0.15
            a_opt, sc_opt = _smooth_exp_gradient_descent(
                overall_best_seq.copy(),
                sub_deadline,
                best_seq=overall_best_seq,
                best_score=overall_best_score,
                eta0=0.7,
                beta0=1.0,
                beta_max=300.0,
                eval_every=60,
            )
            if sc_opt < overall_best_score - 1e-12:
                overall_best_score = sc_opt
                overall_best_seq = a_opt.copy()
                print(f"[{time.time():.1f}] post‑upsample smooth‑EG best {overall_best_score:.6f}")

    # ------------------------------------------------------------------
    # 6.  Final polishing via pairwise annealing.
    # ------------------------------------------------------------------
    if time.time() < deadline:
        print(f"[{time.time():.1f}] final annealing phase")
        a_opt, sc_opt = _pairwise_annealing(
            overall_best_seq.copy(),
            deadline,
            best_seq=overall_best_seq,
            best_score=overall_best_score,
            rng=rng,
            max_trials=None,
        )
        if sc_opt < overall_best_score - 1e-12:
            overall_best_score = sc_opt
            overall_best_seq = a_opt.copy()
            print(f"[{time.time():.1f}] after annealing new best {overall_best_score:.6f}")

    # ------------------------------------------------------------------
    print(
        f"Final best objective = {overall_best_score:.6f}  (length={len(overall_best_seq)})"
    )
    return overall_best_seq.tolist()
