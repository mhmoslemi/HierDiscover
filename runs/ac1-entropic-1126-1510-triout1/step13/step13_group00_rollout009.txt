def _subgrad_descent(a, deadline, best_seq, best_score):
    n = len(a)
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    t = 0
    tolerance = 1e-14
    while time.time() < deadline and t < 500_000:
        t += 1
        conv = _fft_self_convolve(a)
        max_c = conv.max()
        # active indices within fractional tolerance
        thresh = max_c * (1 - 0.015)   # active_frac
        act = np.where(conv >= thresh)[0]
        if act.size > 80:
            top_idx = np.argpartition(conv[act], -80)[-80:]
            act = act[top_idx]
        grad = np.zeros_like(a)
        ind = np.arange(n, dtype=np.int32)
        for k in act:
            shift = k - ind
            mask = (shift >= 0) & (shift < n)
            grad[mask] += 2.0 * a[shift[mask]]
        # Adam update
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad * grad)
        m_hat = m / (1 - beta1 ** t)
        v_hat = v / (1 - beta2 ** t)
        step = 0.8 / (np.sqrt(t) + 1.0)   # learning rate scaling
        # The direction is negative gradient, we use exponentiated gradient:
        a = a * np.exp(-step * (m_hat / (np.sqrt(v_hat) + eps)))
        # project onto simplex
        a_sum = a.sum()
        if a_sum > 0:
            a /= a_sum
        else:
            a = np.full(n, 1.0 / n)
        # evaluation every 200 iters
        if t % 200 == 0:
            cur = _objective(a)
            if cur + tolerance < best_score:
                best_score, best_seq = cur, a.copy()
                # maybe print
        # stagnation detection: if no improvement after some iters, jitter
    return best_seq, best_score

"""
Propose a non‑negative step function that (hopefully) beats the previous
best bound 1.509942 for ``evaluate_sequence``.

Algorithm
---------

1. **Broad seed generation** – a large pool of candidate distributions
   (uniform, arcsine, symmetric / asymmetric Beta, edge‑weighted,
   edge‑inverse‑power, sine‑weighted, random Dirichlet and sparse vectors)
   is created for a few moderate lengths (≈ 3000– 5000).  The best 30
   candidates (lowest exact objective) are kept.

2. **Adaptive exponentiated‑gradient descent** – each kept seed is refined
   by an Adam‑style mirror‑descent on the exact non‑smooth objective.
   The sub‑gradient is built from all convolution entries whose value is
   within a small fraction (``active_frac``) of the current maximum.
   Momentum (first and second moments) together with a decaying learning
   rate give fast convergence.  After every update the vector is projected
   onto the probability simplex (non‑negativity + unit sum).

3. **Iterative up‑sampling** – the current best distribution is linearly
   interpolated to a larger length (≈ ×2) and the adaptive descent is
   re‑run.  This step is repeated while time permits, up to a length of
   about 30 000, because the continuous limit of the problem has a lower
   bound that does not depend on the discretisation size.

4. **Deterministic pairwise sweep & cheap simulated‑annealing** – after the
   final up‑sampling a greedy mass‑transfer sweep followed by a short SA
   phase cleans up any remaining spikes.

All operations stay on the simplex, therefore the objective is invariant
to scaling and the final list can be returned directly.  The routine respects
the supplied time budget (default 1000 s) and prints intermediate progress.
"""

import time
import math
import numpy as np
from numpy.fft import rfft, irfft

# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x≥0, sum(x)=s}."""
    if s < 0:
        raise ValueError("simplex radius must be non‑negative")
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.where(u - cssv / (np.arange(1, len(v) + 1)) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a (real FFT, length 2·n‑1)."""
    n = a.shape[0]
    need = 2 * n - 1
    size = 1 << ((need - 1).bit_length())
    A = rfft(a, n=size)
    conv = irfft(A * A, n=size)
    return conv[:need]


# ----------------------------------------------------------------------
def _objective(a: np.ndarray) -> float:
    """Exact objective 2·n·max(conv(a,a)) for a with sum(a)=1."""
    n = a.shape[0]
    conv = _fft_self_convolve(a)
    return 2.0 * n * conv.max()


# ----------------------------------------------------------------------
# Analytic families ----------------------------------------------------
def _distribution_uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _distribution_arcsine(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    vals /= vals.sum()
    return vals


def _distribution_beta_sym(n: int, p: float) -> np.ndarray:
    """Symmetric Beta(p,p) on the grid."""
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    w = np.exp((p - 1) * np.log(x) + (p - 1) * np.log(1.0 - x))
    s = w.sum()
    if s == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= s
    return w


def _distribution_beta_asym(n: int, a: float, b: float) -> np.ndarray:
    """Asymmetric Beta(a,b) on the grid."""
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    logw = (a - 1) * np.log(x) + (b - 1) * np.log(1.0 - x)
    # shift to avoid overflow/underflow
    logw -= np.max(logw)
    w = np.exp(logw)
    s = w.sum()
    if s == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= s
    return w


def _distribution_edge_weighted(n: int, p: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64) + 0.5
    j = (n - np.arange(n, dtype=np.float64) - 0.5)
    w = (i * j) ** (-p)
    s = w.sum()
    if s == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= s
    return w


def _distribution_edge_inversepow(n: int, p: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64) + 0.5
    j = (n - np.arange(n, dtype=np.float64) - 0.5)
    w = i ** (-p) + j ** (-p)
    s = w.sum()
    if s == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= s
    return w


def _distribution_sine_weighted(n: int, p: float) -> np.ndarray:
    theta = np.pi * (np.arange(n, dtype=np.float64) + 0.5) / n
    w = np.sin(theta) ** p
    s = w.sum()
    if s == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= s
    return w


def _distribution_random_sparse(
    n: int, rng: np.random.Generator, density: float = 0.006
) -> np.ndarray:
    k = max(2, int(n * density))
    idx = rng.choice(n, size=k, replace=False)
    vals = np.zeros(n, dtype=np.float64)
    vals[idx] = rng.random(k)
    if vals.sum() == 0.0:
        vals[0] = 1.0
    vals /= vals.sum()
    return vals


def _distribution_dirichlet(n: int, rng: np.random.Generator) -> np.ndarray:
    alpha = rng.uniform(0.01, 2.0)
    a = rng.dirichlet(np.full(n, alpha))
    return a.astype(np.float64)


def _mix_with_uniform(base: np.ndarray, lam: float) -> np.ndarray:
    """lam·U + (1‑lam)·base, where U is the uniform distribution."""
    n = base.shape[0]
    uniform = np.full(n, 1.0 / n, dtype=np.float64)
    return lam * uniform + (1.0 - lam) * base


# ----------------------------------------------------------------------
def _upsample(seq: np.ndarray, new_len: int) -> np.ndarray:
    """Linear interpolation of a simplex‑vector to a larger length."""
    old_len = seq.shape[0]
    if new_len == old_len:
        return seq.copy()
    x_old = np.linspace(0.0, 1.0, old_len)
    x_new = np.linspace(0.0, 1.0, new_len)
    new_seq = np.interp(x_new, x_old, seq)
    new_seq = np.maximum(new_seq, 0.0)
    new_seq /= new_seq.sum()
    return new_seq

def _subgrad_descent_adam(
    a: np.ndarray,
    deadline: float,
    best_seq: np.ndarray,
    best_score: float,
    lr0: float = 0.9,
    active_frac: float = 0.015,
    max_active: int = 80,
    beta1: float = 0.9,
    beta2: float = 0.999,
    eps: float = 1e-8,
    eval_interval: int = 150,
    patience: int = 2500,
) -> tuple[np.ndarray, float]:
    """
    Adam‑style exponentiated‑gradient descent on the exact objective.
    The sub‑gradient is built from all convolution entries whose value is
    within ``active_frac`` of the current maximum.
    """
    n = a.shape[0]
    m = np.zeros_like(a)          # first moment
    v = np.zeros_like(a)          # second moment
    t = 0
    no_improve = 0
    rng = np.random.default_rng()
    ind_grid = np.arange(n, dtype=np.int32)

    while time.time() < deadline and t < 500_000:
        t += 1

        # convolution and active set
        conv = _fft_self_convolve(a)
        max_c = conv.max()
        thresh = max_c * (1.0 - active_frac)
        act = np.where(conv >= thresh)[0]
        if act.size > max_active:
            # keep only the largest ones
            top_idx = np.argpartition(conv[act], -max_active)[-max_active:]
            act = act[top_idx]

        # sub‑gradient of the max‑convolution
        grad = np.zeros_like(a)
        for k in act:
            shift = k - ind_grid
            mask = (shift >= 0) & (shift < n)
            grad[mask] += 2.0 * a[shift[mask]]

        # Adam updates (first and second moments)
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)

        # bias correction
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)

        # learning rate that decays as 1/√t
        eta = lr0 / (math.sqrt(t) + 1.0)

        # exponentiated‑gradient step (negative direction)
        a = a * np.exp(-eta * (m_hat / (np.sqrt(v_hat) + eps)))

        # simplex projection (guarantees non‑negativity and unit sum)
        s = a.sum()
        if s > 0.0:
            a /= s
        else:
            a = np.full(n, 1.0 / n, dtype=np.float64)

        # occasional evaluation
        if t % eval_interval == 0:
            cur = _objective(a)
            if cur + 1e-14 < best_score:
                best_score, best_seq = cur, a.copy()
                no_improve = 0
                print(f"[{time.time():.1f}] it={t:,} new global best {best_score:.6f}")
            else:
                no_improve += 1

        # if we see no improvement for a while, inject a tiny jitter
        if no_improve >= patience:
            jitter = rng.exponential(scale=0.003, size=n)
            a = a + jitter
            a = np.maximum(a, 0.0)
            a /= a.sum()
            no_improve = 0

    return best_seq, best_score


# ----------------------------------------------------------------------
def _pairwise_swap_local(
    a: np.ndarray,
    deadline: float,
    best_seq: np.ndarray,
    best_score: float,
    max_trials: int | None = None,
) -> tuple[np.ndarray, float]:
    """
    Greedy mass‑transfer between two coordinates.
    Keep a change only if the objective improves.
    """
    n = a.shape[0]
    rng = np.random.default_rng()
    trials = 0

    while time.time() < deadline:
        if max_trials is not None and trials >= max_trials:
            break
        trials += 1

        src_candidates = np.where(a > 1e-12)[0]
        if src_candidates.size == 0:
            break
        i = rng.choice(src_candidates)

        j = rng.integers(n - 1)
        if j >= i:
            j += 1

        delta = a[i] * rng.random() * 0.33
        if delta < 1e-14:
            continue

        new_a = a.copy()
        new_a[i] -= delta
        new_a[j] += delta

        cur = _objective(new_a)
        if cur + 1e-14 < best_score:
            a = new_a
            best_score = cur
            best_seq = a.copy()
            print(
                f"[{time.time():.1f}] swap {i}->{j} δ={delta:.3e} new best {best_score:.6f}"
            )

    return best_seq, best_score


# ----------------------------------------------------------------------
def _random_local_search(
    a: np.ndarray,
    deadline: float,
    best_seq: np.ndarray,
    best_score: float,
    init_temp: float = 0.008,
    decay: float = 0.9995,
) -> tuple[np.ndarray, float]:
    """Metropolis‑style simulated annealing on the exact objective."""
    n = a.shape[0]
    rng = np.random.default_rng()
    T = init_temp

    while time.time() < deadline:
        i = rng.integers(n)
        j = rng.integers(n - 1)
        if j >= i:
            j += 1
        delta = a[i] * rng.random() * 0.28
        if delta < 1e-14:
            continue

        new_a = a.copy()
        new_a[i] -= delta
        new_a[j] += delta

        cur = _objective(new_a)
        if cur + 1e-14 < best_score:
            a = new_a
            best_score = cur
            best_seq = a.copy()
            print(
                f"[{time.time():.1f}] SA improve {i}->{j} Δ={delta:.3e} best {best_score:.6f}"
            )
        else:
            # accept occasionally to escape plateaus
            prob = math.exp(-(cur - best_score) / max(T, 1e-12))
            if rng.random() < prob:
                a = new_a

        T *= decay

    return best_seq, best_score

def propose_candidate(seed: int = 42, budget_s: int = 1000) -> list[float]:
    """
    Search for a non‑negative step function that minimises ``evaluate_sequence``.
    The whole search respects the given time budget (default 1000 s).
    Returns a plain Python list of heights.
    """
    rng = np.random.default_rng(seed)
    start = time.time()
    deadline = start + budget_s - 2.0       # safety margin

    # ------------------------------------------------------------------
    # 1. Generate a diverse pool of seeds (moderate lengths)
    lengths = [3000, 3500, 4000, 4500, 5000]
    p_grid = np.concatenate(
        [
            np.logspace(-3, -1, 11),   # 0.001 … 0.1
            np.logspace(-1, 0, 11),   # 0.1 … 1
            np.logspace(0, 0.5, 9),   # 1 … 3.16
        ]
    )
    lam_grid = np.linspace(0.0, 0.9, 10)

    seed_pool: list[np.ndarray] = []

    for n in lengths:
        if time.time() >= deadline:
            break

        # deterministic families
        for cand in (_distribution_uniform(n), _distribution_arcsine(n)):
            seed_pool.append(cand)

        # symmetric families
        for p in p_grid:
            if time.time() >= deadline:
                break
            beta = _distribution_beta_sym(n, p)
            edge = _distribution_edge_weighted(n, p)
            edge2 = _distribution_edge_inversepow(n, p)
            sine = _distribution_sine_weighted(n, p)
            for base in (beta, edge, edge2, sine):
                for lam in lam_grid:
                    if time.time() >= deadline:
                        break
                    seed_pool.append(_mix_with_uniform(base, lam))

        # asymmetric beta
        for _ in range(6):
            if time.time() >= deadline:
                break
            a_param = rng.uniform(0.001, 3.0)
            b_param = rng.uniform(0.001, 3.0)
            seed_pool.append(_distribution_beta_asym(n, a_param, b_param))

        # random Dirichlet
        for _ in range(8):
            if time.time() >= deadline:
                break
            seed_pool.append(_distribution_dirichlet(n, rng))

        # sparse random
        for _ in range(6):
            if time.time() >= deadline:
                break
            seed_pool.append(_distribution_random_sparse(n, rng, density=0.006))

    # ------------------------------------------------------------------
    # 2. Incorporate a previously discovered construction if available
    try:
        prev = np.asarray(height_sequence_1, dtype=np.float64)   # type: ignore[name-defined]
        if prev.ndim == 1 and prev.size > 0:
            prev = np.maximum(prev, 0.0)
            prev = prev / prev.sum()
            seed_pool.append(prev)
    except Exception:
        pass

    # ------------------------------------------------------------------
    # 3. Keep only the most promising seeds (by exact objective)
    scored = [( _objective(s), s) for s in seed_pool]
    scored.sort(key=lambda x: x[0])
    top_seeds = [seq for _, seq in scored[:30]]
    overall_best_seq = top_seeds[0].copy()
    overall_best_score = _objective(overall_best_seq)
    print(f"[{time.time():.1f}] initial best score {overall_best_score:.6f}")

    # ------------------------------------------------------------------
    # 4. Heavy descent on the best few seeds
    remaining = deadline - time.time()
    if remaining > 120.0 and top_seeds:
        # take the 7 best distinct seeds
        seeds_to_refine = [s for _, s in scored[:7]]
        per_seed = max(5.0, remaining * 0.5 / len(seeds_to_refine))
        for idx, s in enumerate(seeds_to_refine):
            if time.time() >= deadline:
                break
            sub_dead = time.time() + per_seed
            a_opt, sc_opt = _subgrad_descent_adam(
                s.copy(),
                sub_dead,
                best_seq=overall_best_seq,
                best_score=overall_best_score,
                lr0=0.9,
                active_frac=0.018,
                max_active=80,
                beta1=0.9,
                beta2=0.999,
                eval_interval=120,
                patience=2000,
            )
            if sc_opt < overall_best_score:
                overall_best_score, overall_best_seq = sc_opt, a_opt.copy()
                print(
                    f"[{time.time():.1f}] after descent seed {idx} new best {overall_best_score:.6f}"
                )

    # ------------------------------------------------------------------
    # 5. Iterative up‑sampling & refinement while time permits
    while True:
        remaining = deadline - time.time()
        if remaining < 80.0:
            break
        cur_len = len(overall_best_seq)
        new_len = min(cur_len * 2, 30000)
        if new_len == cur_len:
            break
        print(
            f"[{time.time():.1f}] up‑sampling from {cur_len} to {new_len} (remaining {remaining:.1f}s)"
        )
        up_seq = _upsample(overall_best_seq, new_len)

        sub_dead = time.time() + remaining * 0.45
        up_seq_opt, up_score = _subgrad_descent_adam(
            up_seq,
            sub_dead,
            best_seq=overall_best_seq,
            best_score=overall_best_score,
            lr0=0.6,
            active_frac=0.02,
            max_active=80,
            beta1=0.94,
            beta2=0.999,
            eval_interval=100,
            patience=1800,
        )
        if up_score < overall_best_score:
            overall_best_score, overall_best_seq = up_score, up_seq_opt.copy()
            print(
                f"[{time.time():.1f}] up‑sample descent yielded new best {overall_best_score:.6f}"
            )
        else:
            overall_best_seq = up_seq_opt  # keep the refined up‑sampled version

        # quick deterministic sweep
        if time.time() < deadline:
            overall_best_seq, overall_best_score = _pairwise_swap_local(
                overall_best_seq.copy(),
                deadline,
                best_seq=overall_best_seq,
                best_score=overall_best_score,
                max_trials=120_000,
            )
        # short SA clean‑up
        if time.time() < deadline:
            overall_best_seq, overall_best_score = _random_local_search(
                overall_best_seq.copy(),
                deadline,
                best_seq=overall_best_seq,
                best_score=overall_best_score,
                init_temp=0.006,
                decay=0.9996,
            )

    # ------------------------------------------------------------------
    # 6. Final fine‑grained local optimisation
    if time.time() < deadline:
        print(f"[{time.time():.1f}] final deterministic pairwise refinement")
        overall_best_seq, overall_best_score = _pairwise_swap_local(
            overall_best_seq.copy(),
            deadline,
            best_seq=overall_best_seq,
            best_score=overall_best_score,
            max_trials=500_000,
        )

    if time.time() < deadline:
        print(f"[{time.time():.1f}] final simulated‑annealing clean‑up")
        overall_best_seq, overall_best_score = _random_local_search(
            overall_best_seq.copy(),
            deadline,
            best_seq=overall_best_seq,
            best_score=overall_best_score,
            init_temp=0.004,
            decay=0.9997,
        )

    # ------------------------------------------------------------------
    print(
        f"Final best score: {overall_best_score:.6f} (length={len(overall_best_seq)})"
    )
    # Convert to plain Python list; evaluate_sequence will perform its own clipping.
    return overall_best_seq.tolist()
