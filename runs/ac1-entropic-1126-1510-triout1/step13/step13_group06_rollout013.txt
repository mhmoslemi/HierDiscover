"""
Optimise a non‑negative step function for the objective
    2·n·max(conv(a,a)) / (sum(a)**2)

The algorithm consists of three phases:
1.  Broad random/analytic seeding (including extreme edge‑weighted
    distributions, asymmetric Beta, Dirichlet, very sparse vectors) on
    several lengths up to 20000.
2.  For the most promising seeds a short Adam‑style mirror descent on the
    smooth soft‑max surrogate of the max‑convolution, with a temperature that
    is increased until the true max is approximated.
3.  Deterministic pairwise mass swaps and a cheap simulated‑annealing walk
    to polish the final result.

All updates keep the vector on the probability simplex (non‑negative,
unit sum).  The routine respects a user supplied time budget (default
1000 s) and returns a plain Python list; ``evaluate_sequence`` performs the
final clipping/normalisation.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x≥0, Σx = s}."""
    if s < 0:
        raise ValueError("simplex radius must be non‑negative")
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho_arr = np.where(u - cssv / (np.arange(1, len(v) + 1)) > 0)[0]
    if rho_arr.size == 0:
        theta = 0.0
    else:
        rho = rho_arr[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Full self‑convolution a*a using a real FFT (returns length 2n‑1)."""
    n = a.shape[0]
    need = 2 * n - 1
    size = 1 << ((need - 1).bit_length())
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[:need]


# ----------------------------------------------------------------------
def _objective(a: np.ndarray) -> float:
    """Exact objective for a vector with sum = 1."""
    n = a.shape[0]
    b = _fft_self_convolve(a)
    return float(2.0 * n * b.max())


# ----------------------------------------------------------------------
# Analytic families ----------------------------------------------------
def _distribution_uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _distribution_arcsine(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64) + 0.5
    w = 1.0 / np.sqrt(i * (n - i))
    w /= w.sum()
    return w


def _distribution_edge_weighted(n: int, p: float) -> np.ndarray:
    """U‑shaped family w_i ∝ ((i+½)(n‑i‑½))^(−p)."""
    i = np.arange(n, dtype=np.float64) + 0.5
    j = n - np.arange(n, dtype=np.float64) - 0.5
    w = (i * j) ** (-p)
    w /= w.sum()
    return w


def _distribution_beta_sym(n: int, p: float) -> np.ndarray:
    """Symmetric Beta(p,p) on the discrete grid."""
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    # work in log‑space to avoid under‑/overflow
    logw = (p - 1) * (np.log(x) + np.log(1.0 - x))
    w = np.exp(logw - np.max(logw))          # normalise numerically
    if w.sum() == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _distribution_beta_asym(n: int, a: float, b: float) -> np.ndarray:
    """Asymmetric Beta(a,b) on the discrete grid."""
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    logw = (a - 1) * np.log(x) + (b - 1) * np.log(1.0 - x)
    w = np.exp(logw - np.max(logw))
    if w.sum() == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _distribution_random_sparse(
    n: int, rng: np.random.Generator, density: float = 0.006
) -> np.ndarray:
    """Sparse random vector with a given density of non‑zeros."""
    k = max(2, int(n * density))
    idx = rng.choice(n, size=k, replace=False)
    vals = np.zeros(n, dtype=np.float64)
    vals[idx] = rng.random(k)
    vals /= vals.sum()
    return vals


def _mix_with_uniform(base: np.ndarray, lam: float) -> np.ndarray:
    """Convex combination lam·U + (1‑lam)·base, U is uniform."""
    n = base.shape[0]
    uniform = np.full(n, 1.0 / n, dtype=np.float64)
    return lam * uniform + (1.0 - lam) * base


# ----------------------------------------------------------------------
# Soft‑max surrogate utilities -----------------------------------------
def _softmax_weights(conv: np.ndarray, tau: float) -> np.ndarray:
    """stable soft‑max of τ·conv (returns a probability vector)."""
    max_c = conv.max()
    w_raw = np.exp(tau * (conv - max_c))
    total = w_raw.sum()
    if total == 0.0:
        return np.full_like(conv, 1.0 / conv.size)
    return w_raw / total


def _weighted_gradient(a: np.ndarray, w: np.ndarray) -> np.ndarray:
    """
    Gradient of Σ_i w_i·(a*a)[i] w.r.t. a.
    The result is 2·(rev(a) ⊛ w) restricted to the first n entries.
    """
    n = a.shape[0]
    a_rev = a[::-1]
    need = len(w) + n - 1  # = 3·n‑2
    size = 1 << ((need - 1).bit_length())
    A = np.fft.rfft(a_rev, n=size)
    W = np.fft.rfft(w, n=size)
    conv = np.fft.irfft(A * W, n=size)[:need]
    grad = 2.0 * conv[n - 1 : n - 1 + n]
    return grad


# ----------------------------------------------------------------------
# Adam‑style mirror descent on the soft‑max surrogate ------------------
def _adam_descent(
    a: np.ndarray,
    deadline: float,
    best_seq: np.ndarray,
    best_score: float,
    lr0: float = 0.12,
    beta1: float = 0.9,
    beta2: float = 0.999,
    eps: float = 1e-8,
    tau0: float = 0.5,
    tau_mult: float = 1.015,
    eval_interval: int = 80,
) -> tuple[np.ndarray, float]:
    """
    Adam on the smooth max‑convolution.  After each step the iterate is
    projected onto the simplex (unit‑sum, non‑negative).
    """
    n = a.shape[0]
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    tau = tau0
    t = 0

    while time.time() < deadline and t < 300_000:
        t += 1

        conv = _fft_self_convolve(a)
        w = _softmax_weights(conv, tau)
        grad = _weighted_gradient(a, w)

        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)

        # bias‑correction
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)

        # learning rate schedule (inverse‑sqrt)
        lr = lr0 / math.sqrt(t)

        a = a - lr * m_hat / (np.sqrt(v_hat) + eps)

        # project back onto the simplex (keeps sum = 1)
        a = _project_onto_simplex(a, s=1.0)

        tau *= tau_mult

        if t % eval_interval == 0:
            cur = _objective(a)
            if cur < best_score - 1e-14:
                best_score = cur
                best_seq = a.copy()
                print(f"[{time.time():.1f}] Adam it={t} new best {best_score:.6f}")

    return best_seq, best_score

# Greedy pairwise refinement -------------------------------------------
def _pairwise_swap_local(
    a: np.ndarray,
    deadline: float,
    best_seq: np.ndarray,
    best_score: float,
    max_trials: int | None = None,
) -> tuple[np.ndarray, float]:
    """Randomly transfer a small chunk of mass; keep if objective improves."""
    n = a.shape[0]
    trials = 0
    rng = np.random.default_rng()
    while time.time() < deadline:
        if max_trials is not None and trials >= max_trials:
            break
        trials += 1

        # choose a source index with non‑trivial mass
        src = np.where(a > 1e-12)[0]
        if src.size == 0:
            break
        i = rng.choice(src)

        # choose a distinct destination
        j = rng.integers(n - 1)
        if j >= i:
            j += 1

        # amount to move (up to 35 % of the source mass)
        delta = a[i] * rng.random() * 0.35
        if delta < 1e-14:
            continue

        new_a = a.copy()
        new_a[i] -= delta
        new_a[j] += delta

        cur = _objective(new_a)
        if cur < best_score - 1e-14:
            a = new_a
            best_score = cur
            best_seq = a.copy()
            print(
                f"[{time.time():.1f}] swap i={i}→j={j} δ={delta:.3e} new best {best_score:.6f}"
            )
    return best_seq, best_score

# Cheap simulated‑annealing walk ----------------------------------------
def _random_local_search(
    a: np.ndarray,
    deadline: float,
    best_seq: np.ndarray,
    best_score: float,
    init_temp: float = 0.01,
    decay: float = 0.9995,
) -> tuple[np.ndarray, float]:
    """Very cheap SA that repeatedly transfers a random amount of mass."""
    n = a.shape[0]
    rng = np.random.default_rng()
    T = init_temp
    while time.time() < deadline:
        i = rng.integers(n)
        j = rng.integers(n - 1)
        if j >= i:
            j += 1
        delta = a[i] * rng.random() * 0.3
        if delta < 1e-14:
            continue
        new_a = a.copy()
        new_a[i] -= delta
        new_a[j] += delta
        cur = _objective(new_a)
        if cur < best_score - 1e-14:
            a = new_a
            best_score = cur
            best_seq = a.copy()
            print(f"[{time.time():.1f}] SA improve {i}->{j} Δ={delta:.3e} best {best_score:.6f}")
        else:
            if rng.random() < math.exp(-(cur - best_score) / max(T, 1e-12)):
                a = new_a
        T *= decay
    return best_seq, best_score

            if time.time() >= deadline:
                break
            sub_deadline = time.time() + per_seed
            a_opt, sc_opt = _adam_descent(
                seed_seq.copy(),
                sub_deadline,
                best_seq=overall_best_seq,
                best_score=overall_best_score,
                lr0=0.12,
                tau0=0.5,
                tau_mult=1.015,
                eval_interval=80,
            )
            if sc_opt < overall_best_score:
                overall_best_score, overall_best_seq = sc_opt, a_opt.copy()
                print(
                    f"[{time.time():.1f}] after Adam seed {idx} new global best {overall_best_score:.6f}"
                )

    # ------------------------------------------------------------------
    # 4. Greedy pairwise refinement on the global best
    if time.time() < deadline:
        print(f"[{time.time():.1f}] pairwise refinement")
        overall_best_seq, overall_best_score = _pairwise_swap_local(
            overall_best_seq.copy(),
            deadline,
            best_seq=overall_best_seq,
            best_score=overall_best_score,
            max_trials=200_000,
        )

    # ------------------------------------------------------------------
    # 5. Final cheap simulated‑annealing walk
    if time.time() < deadline:
        print(f"[{time.time():.1f}] final stochastic local search")
        overall_best_seq, overall_best_score = _random_local_search(
            overall_best_seq.copy(),
            deadline,
            best_seq=overall_best_seq,
            best_score=overall_best_score,
            init_temp=0.01,
            decay=0.9995,
        )

    # ------------------------------------------------------------------
    print(
        f"Final best score: {overall_best_score:.6f} (length = {len(overall_best_seq)})"
    )
    # ``evaluate_sequence`` will perform the final clipping / normalisation.
    return overall_best_seq.tolist()
