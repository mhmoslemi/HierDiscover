import time, math, numpy as np

def propose_candidate(seed=42, budget_s=1000):
    """
    Search a non‑negative step function (list of floats) that minimizes
    ``evaluate_sequence``.  The strategy mixes analytic seed families,
    exponentiated gradient descent, a soft‑max “smoothed” EG stage, an optional
    linear‑program refinement (if SciPy is present), up‑sampling and a final
    pairwise‑annealing polish.  All work respects the 1000 s budget.
    """
    # --- helper utilities -------------------------------------------------
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 2.0   # safety margin

    # Simplex projection (Euclidean)
    def _project_onto_simplex(v, s=1.0):
        if s < 0:
            raise ValueError("simplex radius must be non‑negative")
        if np.all(v >= 0) and np.isclose(v.sum(), s):
            return v.copy()
        u = np.sort(v)[::-1]
        cssv = np.cumsum(u) - s
        rho = np.where(u - cssv / (np.arange(1, len(v)+1)) > 0)[0]
        if rho.size == 0:
            theta = 0.0
        else:
            rho = rho[-1]
            theta = cssv[rho] / (rho+1)
        w = np.maximum(v - theta, 0.0)
        return w

    # FFT based self‑convolution
    def _fft_self_convolve(a):
        n = a.shape[0]
        size = 1 << ((2*n-1)-1).bit_length()
        A = np.fft.rfft(a, n=size)
        conv = np.fft.irfft(A * A, n=size)
        return conv[:2*n-1]

    # Exact objective (scale‑invariant)
    def _objective(a):
        n = a.shape[0]
        b = _fft_self_convolve(a)
        max_b = b.max()
        sum_a = a.sum()
        if sum_a < 0.01:
            return float("inf")
        return float(2.0 * n * max_b / (sum_a * sum_a))

    # Gradient of the (single) maximal convolution entry
    def _grad_of_max_conv(a):
        n = a.shape[0]
        conv = _fft_self_convolve(a)
        k = int(np.argmax(conv))
        grad = np.zeros_like(a)
        if k < n:
            grad[:k+1] = 2.0 * a[k::-1]
        else:
            start = k - (n-1)
            grad[start:] = 2.0 * a[start:][::-1]
        return grad, k

    # Gradient of the soft‑max weighted sum of all convolution entries
    def _grad_softmax(a, beta):
        """Return ∇ Σ_k w_k (a*a)_k where w ∝ exp(beta·conv)."""
        n = a.shape[0]
        conv = _fft_self_convolve(a)                     # length 2n‑1
        maxc = conv.max()
        w = np.exp(beta * (conv - maxc))
        w = w / w.sum()                                 # weights, length L=2n‑1
        L = 2 * n - 1
        # linear correlation a ⋆ w  (lag i = Σ_j a_j w_{j+i})
        size = 1 << ((n + L - 1) - 1).bit_length()
        A = np.fft.rfft(a, n=size)
        W_rev = np.fft.rfft(w[::-1], n=size)            # w reversed
        corr = np.fft.irfft(A * W_rev, n=size)
        corr = corr[: n + L - 1]                         # length 3n‑2
        # block = corr[n‑1 : 2n‑1] has length n; grad = 2·block[::-1]
        block = corr[n-1:2*n-1]
        grad = 2.0 * block[::-1]
        return grad

    # Exponentiated‑gradient descent on the true objective (sub‑gradient of max entry)
    def _exp_gradient_descent(a, deadline, best_seq, best_score, eta0=0.8, eval_every=80):
        t = 0
        while time.time() < deadline:
            t += 1
            grad, _ = _grad_of_max_conv(a)
            eta = eta0 / math.sqrt(t)
            a = a * np.exp(-eta * grad)
            s = a.sum()
            if s == 0.0:
                a = np.full_like(a, 1.0 / a.shape[0])
            else:
                a = a / s
            if t % eval_every == 0:
                cur = _objective(a)
                if cur < best_score - 1e-14:
                    best_score = cur
                    best_seq = a.copy()
                    print(f"[{time.time():.1f}] EG it={t} new best {best_score:.6f}")
        return best_seq, best_score

    # Soft‑max EG stage (smooth gradient)
    def _softmax_exp_gradient_descent(a, deadline, best_seq, best_score,
                                     beta0=0.5, beta_factor=1.02,
                                     eta0=0.6, eval_every=80):
        t = 0
        beta = beta0
        while time.time() < deadline:
            t += 1
            grad = _grad_softmax(a, beta)
            eta = eta0 / math.sqrt(t)
            a = a * np.exp(-eta * grad)
            s = a.sum()
            if s == 0.0:
                a = np.full_like(a, 1.0 / a.shape[0])
            else:
                a = a / s
            beta *= beta_factor   # slowly sharpen the soft‑max
            if t % eval_every == 0:
                cur = _objective(a)
                if cur < best_score - 1e-14:
                    best_score = cur
                    best_seq = a.copy()
                    print(f"[{time.time():.1f}] SoftEG it={t} β={beta:.2f} new best {best_score:.6f}")
        return best_seq, best_score

    # Frank‑Wolfe style refinement (same as previous version)
    def _frank_wolfe_descent(a, deadline, best_seq, best_score, eval_every=100):
        t = 0

        return best_seq, best_score

    def _pairwise_annealing(a, deadline, best_seq, best_score, rng):

            new_a[j] += delta
            cur = _objective(new_a)

            if cur < best_score - 1e-14:

            else:

        return best_seq, best_score

    # Linear up‑sampling (preserves sum)
    def _upsample(a, new_n):
        old_n = a.shape[0]
        if new_n == old_n:
            return a.copy()
        x_old = (np.arange(old_n, dtype=np.float64) + 0.5) / old_n
        x_new = (np.arange(new_n, dtype=np.float64) + 0.5) / new_n
        a_new = np.interp(x_new, x_old, a, left=0.0, right=0.0)
        a_new = np.maximum(a_new, 0.0)
        s = a_new.sum()
        if s == 0.0:
            a_new[:] = 1.0 / new_n
        else:
            a_new /= s
        return a_new

    # Analytic families (return probability vectors)
    def _distribution_uniform(n):
        return np.full(n, 1.0 / n, dtype=np.float64)

    def _distribution_arcsine(n):
        i = np.arange(n, dtype=np.float64)
        vals = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
        vals /= vals.sum()
        return vals

    def _distribution_beta_sym(n, p):
        x = (np.arange(n, dtype=np.float64) + 0.5) / n
        w = np.exp((p - 1) * np.log(x) + (p - 1) * np.log(1.0 - x))
        if w.sum() == 0.0:
            return np.full_like(w, 1.0 / n)
        return w / w.sum()

    def _distribution_edge_weighted(n, p):
        i = np.arange(n, dtype=np.float64) + 0.5
        j = (n - np.arange(n, dtype=np.float64) - 0.5)
        w = (i * j) ** (-p)
        if w.sum() == 0.0:
            return np.full_like(w, 1.0 / n)
        return w / w.sum()

    def _distribution_hann(n):
        i = np.arange(n, dtype=np.float64)
        w = 0.5 - 0.5 * np.cos(2.0 * np.pi * i / (n - 1))
        w_sum = w.sum()
        if w_sum == 0.0:
            w = np.full_like(w, 1.0 / n)
        else:
            w /= w_sum
        return w

    def _distribution_generalized_normal(n, q):
        i = np.arange(n, dtype=np.float64) + 0.5
        x = i / n - 0.5
        w = np.exp(- (np.abs(x) / 0.5) ** q)
        w_sum = w.sum()
        if w_sum == 0.0:
            w = np.full_like(w, 1.0 / n)
        else:
            w /= w_sum
        return w

    def _distribution_power_law(n, p):
        i = np.arange(1, n + 1, dtype=np.float64)
        w = i ** (-p)
        w /= w.sum()
        return w

    def _distribution_random_sparse(n, density=0.01):
        k = max(2, int(n * density))
        idx = rng.choice(n, size=k, replace=False)
        vals = np.zeros(n, dtype=np.float64)
        vals[idx] = rng.random(k)
        if vals.sum() == 0.0:
            vals[0] = 1.0
        vals /= vals.sum()
        return vals

    # Mix a distribution with uniform
    def _mix_with_uniform(base, lam):
        n = base.shape[0]
        uniform = np.full(n, 1.0 / n, dtype=np.float64)
        return lam * uniform + (1.0 - lam) * base

    # -----------------------------------------------------------------

    # -----------------------------------------------------------------

    ])
    lam_grid = np.linspace(0.0, 1.0, 5)

import time
import math
import numpy as np

def propose_candidate(seed: int = 42, budget_s: int = 1000) -> list[float]:
    """
    Search a non‑negative step function (list of floats) that minimises the
    provided ``evaluate_sequence``.  The strategy builds a diverse pool of
    analytic seeds, refines the best ones with exponentiated‑gradient descent,
    adds a *soft‑max* EG stage (smooth approximation to the max‑convolution),
    optionally performs a linear‑program refinement (if SciPy is available),
    up‑samples the current best vector, does a short gradient burst and finally
    finishes with a pairwise‑annealing polish.  All work respects the 1000 s
    time budget.
    """
    # -----------------------------------------------------------------
    # 0.  Basic utilities
    # -----------------------------------------------------------------
    rng = np.random.default_rng(seed)
    start = time.time()
    deadline = start + budget_s - 2.0          # safety margin

    def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
        """Euclidean projection onto the simplex {x≥0, Σx=s}."""
        if s < 0:
            raise ValueError("simplex radius must be non‑negative")
        if np.all(v >= 0) and np.isclose(v.sum(), s):
            return v.copy()
        u = np.sort(v)[::-1]
        cssv = np.cumsum(u) - s
        rho = np.where(u - cssv / (np.arange(1, len(v) + 1)) > 0)[0]
        if rho.size == 0:
            theta = 0.0
        else:
            rho = rho[-1]
            theta = cssv[rho] / (rho + 1)
        w = np.maximum(v - theta, 0.0)
        return w

    def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
        """Full linear self‑convolution a*a using a real FFT."""
        n = a.shape[0]
        size = 1 << ((2 * n - 1) - 1).bit_length()
        A = np.fft.rfft(a, n=size)
        conv = np.fft.irfft(A * A, n=size)
        return conv[: 2 * n - 1]

    def _objective(a: np.ndarray) -> float:
        """Exact value of ``evaluate_sequence`` (scale‑invariant)."""
        n = a.shape[0]
        b = _fft_self_convolve(a)
        max_b = b.max()
        sum_a = a.sum()
        if sum_a < 0.01:
            return float("inf")
        return float(2.0 * n * max_b / (sum_a * sum_a))

    def _grad_of_max_conv(a: np.ndarray) -> tuple[np.ndarray, int]:
        """Sub‑gradient of the currently maximal convolution entry."""
        n = a.shape[0]
        conv = _fft_self_convolve(a)
        k = int(np.argmax(conv))
        grad = np.zeros_like(a)
        if k < n:
            grad[: k + 1] = 2.0 * a[k::-1]
        else:
            start = k - (n - 1)
            grad[start:] = 2.0 * a[start:][::-1]
        return grad, k

    def _grad_softmax(a: np.ndarray, beta: float) -> np.ndarray:
        """
        Gradient of Σ_k w_k·(a*a)_k where w ∝ exp(beta·conv).
        Returns a vector of length n.
        """
        n = a.shape[0]
        conv = _fft_self_convolve(a)                # length 2n‑1
        maxc = conv.max()
        w = np.exp(beta * (conv - maxc))
        w /= w.sum()                                # length L = 2n‑1

        L = 2 * n - 1
        size = 1 << ((n + L - 1) - 1).bit_length()
        A = np.fft.rfft(a, n=size)
        W_rev = np.fft.rfft(w[::-1], n=size)        # reversed weights
        corr = np.fft.irfft(A * W_rev, n=size)
        corr = corr[: n + L - 1]                     # length 3n‑2

        # block = corr[n‑1 : 2n‑1] has length n; grad = 2·block[::-1]
        block = corr[n - 1 : 2 * n - 1]
        grad = 2.0 * block[::-1]
        return grad

    # -----------------------------------------------------------------
    # 1. Analytic seed families (all return probability vectors, Σ=1)
    # -----------------------------------------------------------------
    def _distribution_uniform(n: int) -> np.ndarray:
        return np.full(n, 1.0 / n, dtype=np.float64)

    def _distribution_arcsine(n: int) -> np.ndarray:
        i = np.arange(n, dtype=np.float64)
        vals = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
        vals /= vals.sum()
        return vals

    def _distribution_beta_sym(n: int, p: float) -> np.ndarray:
        x = (np.arange(n, dtype=np.float64) + 0.5) / n
        w = np.exp((p - 1) * np.log(x) + (p - 1) * np.log(1.0 - x))
        if w.sum() == 0.0:
            return np.full_like(w, 1.0 / n)
        return w / w.sum()

    def _distribution_edge_weighted(n: int, p: float) -> np.ndarray:
        i = np.arange(n, dtype=np.float64) + 0.5
        j = (n - np.arange(n, dtype=np.float64) - 0.5)
        w = (i * j) ** (-p)
        if w.sum() == 0.0:
            return np.full_like(w, 1.0 / n)
        return w / w.sum()

    def _distribution_hann(n: int) -> np.ndarray:
        i = np.arange(n, dtype=np.float64)
        w = 0.5 - 0.5 * np.cos(2.0 * np.pi * i / (n - 1))
        w_sum = w.sum()
        if w_sum == 0.0:
            w = np.full_like(w, 1.0 / n)
        else:
            w /= w_sum
        return w

    def _distribution_generalized_normal(n: int, q: float) -> np.ndarray:
        i = np.arange(n, dtype=np.float64) + 0.5
        x = i / n - 0.5
        w = np.exp(- (np.abs(x) / 0.5) ** q)
        w_sum = w.sum()
        if w_sum == 0.0:
            w = np.full_like(w, 1.0 / n)
        else:
            w /= w_sum
        return w

    def _distribution_power_law(n: int, p: float) -> np.ndarray:
        i = np.arange(1, n + 1, dtype=np.float64)
        w = i ** (-p)
        w /= w.sum()
        return w

    def _distribution_random_sparse(n: int, density: float = 0.01) -> np.ndarray:
        k = max(2, int(n * density))
        idx = rng.choice(n, size=k, replace=False)
        vals = np.zeros(n, dtype=np.float64)
        vals[idx] = rng.random(k)
        if vals.sum() == 0.0:
            vals[0] = 1.0
        vals /= vals.sum()
        return vals

    def _mix_with_uniform(base: np.ndarray, lam: float) -> np.ndarray:
        n = base.shape[0]
        uniform = np.full(n, 1.0 / n, dtype=np.float64)
        return lam * uniform + (1.0 - lam) * base

    # -----------------------------------------------------------------
    # 2. Optimisation primitives
    # -----------------------------------------------------------------
    def _exp_gradient_descent(
        a: np.ndarray,
        deadline: float,
        best_seq: np.ndarray,
        best_score: float,
        eta0: float = 0.8,
        eval_every: int = 80,
    ) -> tuple[np.ndarray, float]:
        """Exponentiated‑gradient descent on the true max‑conv objective."""
        t = 0
        while time.time() < deadline:
            t += 1
            grad, _ = _grad_of_max_conv(a)
            eta = eta0 / math.sqrt(t)
            a = a * np.exp(-eta * grad)
            s = a.sum()
            if s == 0.0:
                a = np.full_like(a, 1.0 / a.shape[0])
            else:
                a = a / s
            if t % eval_every == 0:
                cur = _objective(a)
                if cur < best_score - 1e-14:
                    best_score, best_seq = cur, a.copy()
                    print(f"[{time.time():.1f}] EG it={t} new best {best_score:.6f}")
        return best_seq, best_score

    def _softmax_exp_gradient_descent(
        a: np.ndarray,
        deadline: float,
        best_seq: np.ndarray,
        best_score: float,
        beta0: float = 0.5,
        beta_factor: float = 1.02,
        eta0: float = 0.6,
        eval_every: int = 80,
    ) -> tuple[np.ndarray, float]:
        """EG using the smooth (soft‑max) gradient of the convolution max."""
        t = 0
        beta = beta0
        while time.time() < deadline:
            t += 1
            grad = _grad_softmax(a, beta)
            eta = eta0 / math.sqrt(t)
            a = a * np.exp(-eta * grad)
            s = a.sum()
            if s == 0.0:
                a = np.full_like(a, 1.0 / a.shape[0])
            else:
                a = a / s
            beta *= beta_factor
            if t % eval_every == 0:
                cur = _objective(a)
                if cur < best_score - 1e-14:
                    best_score, best_seq = cur, a.copy()
                    print(f"[{time.time():.1f}] SoftEG it={t} β={beta:.2f} new best {best_score:.6f}")
        return best_seq, best_score

    def _frank_wolfe_descent(
        a: np.ndarray,
        deadline: float,
        best_seq: np.ndarray,
        best_score: float,
        eval_every: int = 100,
    ) -> tuple[np.ndarray, float]:
        """Simple Frank‑Wolfe iteration moving mass to the weakest gradient."""
        n = a.shape[0]
        t = 0
        while time.time() < deadline:
            t += 1
            grad, _ = _grad_of_max_conv(a)
            i_min = int(np.argmin(grad))
            gamma = 2.0 / (t + 2.0)
            a = a + gamma * ((np.eye(1, n, i_min)[0]) - a)
            if t % eval_every == 0:
                cur = _objective(a)
                if cur < best_score - 1e-14:
                    best_score, best_seq = cur, a.copy()
                    print(f"[{time.time():.1f}] FW it={t} new best {best_score:.6f}")
        return best_seq, best_score

    def _pairwise_annealing(
        a: np.ndarray,
        deadline: float,
        best_seq: np.ndarray,
        best_score: float,
        rng: np.random.Generator,
    ) -> tuple[np.ndarray, float]:
        """Random pairwise mass transfer with simple simulated‑annealing acceptance."""
        n = a.shape[0]
        start = time.time()
        total = deadline - start
        while time.time() < deadline:
            src = np.where(a > 1e-12)[0]
            if src.size == 0:
                break
            i = rng.choice(src)
            j = rng.integers(n - 1)
            if j >= i:
                j += 1
            delta = a[i] * rng.random() * 0.4
            if delta < 1e-16:
                continue
            new_a = a.copy()
            new_a[i] -= delta
            new_a[j] += delta
            cur = _objective(new_a)
            elapsed = time.time() - start
            temperature = max(0.0, 1.0 - elapsed / total) * best_score * 0.01
            if cur < best_score - 1e-14:
                accept = True
            else:
                accept = temperature > 0.0 and rng.random() < math.exp(-(cur - best_score) / temperature)
            if accept:
                a = new_a
                if cur < best_score - 1e-14:
                    best_score, best_seq = cur, a.copy()
                    print(f"[{time.time():.1f}] anneal i={i}->j={j} δ={delta:.3e} new best {best_score:.6f}")
        return best_seq, best_score

    def _upsample(a: np.ndarray, new_n: int) -> np.ndarray:
        """Linear interpolation to a larger length while staying on the simplex."""
        old_n = a.shape[0]
        if new_n == old_n:
            return a.copy()
        x_old = (np.arange(old_n, dtype=np.float64) + 0.5) / old_n
        x_new = (np.arange(new_n, dtype=np.float64) + 0.5) / new_n
        a_new = np.interp(x_new, x_old, a, left=0.0, right=0.0)
        a_new = np.maximum(a_new, 0.0)
        s = a_new.sum()
        if s == 0.0:
            a_new[:] = 1.0 / new_n
        else:
            a_new /= s
        return a_new

    # -----------------------------------------------------------------
    # 3. Build a diverse pool of seeds
    # -----------------------------------------------------------------
    top_candidates: list[tuple[float, np.ndarray]] = []
    overall_best_seq: np.ndarray | None = None
    overall_best_score = float("inf")

    lengths = [1500, 3000, 6000, 12000]      # moderate sizes – longer later
    p_grid = np.concatenate(
        [
            np.logspace(-2.0, -0.5, 12, base=10),
            np.logspace(-0.5, 0.0, 12, base=10),
            np.logspace(0.0, 0.7, 10, base=10),
        ]
    )
    lam_grid = np.linspace(0.0, 1.0, 5)
    q_grid = np.linspace(0.5, 3.0, 6)
    power_grid = np.linspace(0.2, 2.5, 7)

    for n in lengths:
        if time.time() >= deadline:
            break
        print(f"[{time.time():.1f}] generating seeds, length={n}")

        # deterministic families
        for cand in (_distribution_uniform(n), _distribution_arcsine(n), _distribution_hann(n)):
            sc = _objective(cand)
            top_candidates.append((sc, cand.copy()))
            if sc < overall_best_score:
                overall_best_score, overall_best_seq = sc, cand.copy()
                print(f"   new global best (det) len={n} score={sc:.6f}")

        # beta & edge‑weighted families
        for p in p_grid:
            if time.time() >= deadline:
                break
            beta = _distribution_beta_sym(n, p)
            edge = _distribution_edge_weighted(n, p)
            for base in (beta, edge):
                for lam in lam_grid:
                    if time.time() >= deadline:
                        break
                    cand = _mix_with_uniform(base, lam)
                    sc = _objective(cand)
                    top_candidates.append((sc, cand.copy()))
                    if sc < overall_best_score:
                        overall_best_score, overall_best_seq = sc, cand.copy()
                        print(f"   new best (beta/edge) p={p:.3g} lam={lam:.2f} score={sc:.6f}")

        # generalized normal shapes
        for q in q_grid:
            if time.time() >= deadline:
                break
            cand = _distribution_generalized_normal(n, q)
            sc = _objective(cand)
            top_candidates.append((sc, cand.copy()))
            if sc < overall_best_score:
                overall_best_score, overall_best_seq = sc, cand.copy()
                print(f"   new best (gn) q={q:.2f} score={sc:.6f}")

        # power‑law families
        for pw in power_grid:
            if time.time() >= deadline:
                break
            cand = _distribution_power_law(n, pw)
            sc = _objective(cand)
            top_candidates.append((sc, cand.copy()))
            if sc < overall_best_score:
                overall_best_score, overall_best_seq = sc, cand.copy()
                print(f"   new best (pl) p={pw:.2f} score={sc:.6f}")

        # random Dirichlet seeds
        for _ in range(4):
            if time.time() >= deadline:
                break
            alpha = rng.uniform(0.01, 2.0)
            cand = rng.dirichlet(np.full(n, alpha))
            sc = _objective(cand)
            top_candidates.append((sc, cand.copy()))
            if sc < overall_best_score:
                overall_best_score, overall_best_seq = sc, cand.copy()
                print(f"   new best (dir) score={sc:.6f}")

        # sparse random seeds
        for _ in range(4):
            if time.time() >= deadline:
                break
            cand = _distribution_random_sparse(n, density=0.01)
            sc = _objective(cand)
            top_candidates.append((sc, cand.copy()))
            if sc < overall_best_score:
                overall_best_score, overall_best_seq = sc, cand.copy()
                print(f"   new best (sparse) score={sc:.6f}")

        # keep the 25 best overall
        top_candidates.sort(key=lambda x: x[0])
        top_candidates = top_candidates[:25]

    # -----------------------------------------------------------------
    # 4. Warm‑start from a previously discovered construction if available
    # -----------------------------------------------------------------
    try:
        prev = np.asarray(height_sequence_1, dtype=np.float64)  # type: ignore[name-defined]
        if prev.ndim == 1:
            prev = prev / prev.sum()
            sc = _objective(prev)
            if sc < overall_best_score:
                overall_best_score, overall_best_seq = sc, prev.copy()
                print(f"   previous best score={sc:.6f}")
            top_candidates.append((sc, prev.copy()))
    except Exception:
        pass

    # -----------------------------------------------------------------
    # 5. Fallback safety
    # -----------------------------------------------------------------
    if overall_best_seq is None:
        n = 4000
        overall_best_seq = _distribution_uniform(n)
        overall_best_score = _objective(overall_best_seq)
        print(f"fallback uniform score={overall_best_score:.6f}")

    # -----------------------------------------------------------------
    # 6. Exponentiated‑gradient descent on top seeds
    # -----------------------------------------------------------------
    remaining = deadline - time.time()
    if remaining > 30.0 and top_candidates:
        seed_seqs = [seq for _, seq in sorted(top_candidates, key=lambda x: x[0])[:5]]
        per_seed = remaining * 0.30 / max(1, len(seed_seqs))
        print(f"[{time.time():.1f}] EG descent on {len(seed_seqs)} seeds, {per_seed:.1f}s each")
        for idx, seq in enumerate(seed_seqs):
            if time.time() >= deadline:
                break
            sub_dead = time.time() + per_seed
            best_seq, best_score = _exp_gradient_descent(
                seq.copy(),
                sub_dead,
                overall_best_seq,
                overall_best_score,
                eta0=0.9,
                eval_every=80,
            )
            if best_score < overall_best_score - 1e-12:
                overall_best_score, overall_best_seq = best_score, best_seq.copy()
                print(f"[{time.time():.1f}] after EG seed {idx} new best {overall_best_score:.6f}")

    # -----------------------------------------------------------------
    # 7. Optional linear‑program refinement (only for modest n)
    # -----------------------------------------------------------------
    can_use_lp = False
    try:
        from scipy.optimize import linprog
        can_use_lp = True
    except Exception:
        can_use_lp = False

    if can_use_lp and overall_best_seq.shape[0] <= 2000 and time.time() < deadline:
        print(f"[{time.time():.1f}] LP refinement (n={overall_best_seq.shape[0]})")
        n = overall_best_seq.shape[0]
        f = overall_best_seq.copy()
        M = _fft_self_convolve(f).max()
        L = 2 * n - 1
        A = np.zeros((L, n), dtype=np.float64)
        for k in range(L):
            j_min = max(0, k - (n - 1))
            j_max = min(k, n - 1)
            idx = np.arange(j_min, j_max + 1, dtype=int)
            A[k, idx] = f[k - idx]
        b_ub = np.full(L, M, dtype=np.float64)
        c = -np.ones(n, dtype=np.float64)          # maximise sum(b)
        bounds = [(0, None)] * n
        lp_dead = time.time() + 15.0
        while time.time() < lp_dead:
            res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds,
                          method='highs',
                          options={'presolve': True,
                                   'time_limit': max(0.5, lp_dead - time.time())})
            if res.success:
                b_opt = res.x
                sum_b = b_opt.sum()
                if sum_b > 0:
                    b_norm = b_opt / sum_b
                    t = 0.5
                    f_new = (1 - t) * f + t * b_norm
                    f_new = f_new / f_new.sum()
                    new_score = _objective(f_new)
                    if new_score < overall_best_score:
                        overall_best_score, overall_best_seq = new_score, f_new.copy()
                        print(f"[{time.time():.1f}] LP improve new best {overall_best_score:.6f}")
                break
            else:
                break

    # -----------------------------------------------------------------
    # 8. Soft‑max exponential‑gradient stage (smoothes the max‑conv)
    # -----------------------------------------------------------------
    if time.time() < deadline:
        soft_dead = time.time() + (deadline - time.time()) * 0.25
        print(f"[{time.time():.1f}] soft‑max EG stage (budget {soft_dead - time.time():.1f}s)")
        overall_best_seq, overall_best_score = _softmax_exp_gradient_descent(
            overall_best_seq.copy(),
            soft_dead,
            best_seq=overall_best_seq,
            best_score=overall_best_score,
            beta0=0.5,
            beta_factor=1.05,
            eta0=0.6,
            eval_every=80,
        )
        print(f"[{time.time():.1f}] after soft‑max stage best {overall_best_score:.6f}")

    # -----------------------------------------------------------------
    # 9. Upsample + brief EG refinement (if time permits)
    # -----------------------------------------------------------------
    if time.time() < deadline:
        cur_n = overall_best_seq.shape[0]
        target_n = cur_n * 2
        if target_n <= 64000:
            print(f"[{time.time():.1f}] upsampling {cur_n} → {target_n}")
            up_seq = _upsample(overall_best_seq, target_n)
            up_score = _objective(up_seq)
            if up_score < overall_best_score:
                overall_best_score, overall_best_seq = up_score, up_seq.copy()
                print(f"   upsample improved to {overall_best_score:.6f}")

            # short EG burst on the larger vector
            sub_dead = time.time() + (deadline - time.time()) * 0.15
            overall_best_seq, overall_best_score = _exp_gradient_descent(
                overall_best_seq.copy(),
                sub_dead,
                best_seq=overall_best_seq,
                best_score=overall_best_score,
                eta0=0.7,
                eval_every=60,
            )
            print(f"   post‑upsample EG best {overall_best_score:.6f}")

    # -----------------------------------------------------------------
    # 10. Final annealing polishing
    # -----------------------------------------------------------------
    if time.time() < deadline:
        print(f"[{time.time():.1f}] final annealing")
        overall_best_seq, overall_best_score = _pairwise_annealing(
            overall_best_seq.copy(),
            deadline,
            best_seq=overall_best_seq,
            best_score=overall_best_score,
            rng=rng,
        )
        print(f"   after annealing best {overall_best_score:.6f}")

    # -----------------------------------------------------------------
    print(f"Final best objective {overall_best_score:.6f} (length={len(overall_best_seq)})")
    return overall_best_seq.tolist()
