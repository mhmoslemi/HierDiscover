    n = len(f)

"""
Propose a low‑objective step‑function.

The algorithm works in three phases:

1️⃣ **Diversified seed generation** – many analytical families (uniform,
    arcsine, edge‑weighted, sinusoidal, beta, exponential, random
    Dirichlet, random sparse) are sampled at several lengths.  The ~20
    best seeds (lowest exact objective) are kept.

2️⃣ **Alternating linear‑programming + momentum descent** – From each of the
    top seeds we run a short “LP‑mix’’ phase: given the current sequence
    `a` we compute `M = max(conv(a,a))` and solve the linear program

        maximise    ∑ g_i
        subject to  conv(a,g) ≤ M   (2n‑1 linear inequalities)
                    g_i ≥ 0 .

    The solution `g` is normalised and mixed with `a`.
    After a few such mixes we finish with a long exponentiated‑gradient
    (mirror) descent on the exact objective.

3️⃣ **Local polishing** – a greedy pairwise mass‑transfer step followed by a
    cheap simulated‑annealing style search removes any remaining irregularities.

All vectors are projected onto the simplex {x≥0,∑x=1} after each update.
The routine respects the 1000 s budget and returns a plain Python list that
can be fed directly to ``evaluate_sequence``.
"""

import time
import math
import numpy as np
from numpy.fft import rfft, irfft
from scipy.optimize import linprog


# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x ≥ 0, sum(x) = s}."""
    if s < 0:
        raise ValueError("simplex radius must be non‑negative")
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho_arr = np.where(u - cssv / (np.arange(1, len(v) + 1)) > 0)[0]
    if rho_arr.size == 0:
        theta = 0.0
    else:
        rho = rho_arr[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Full self‑convolution a*a using a real FFT (returns length 2n‑1)."""
    n = a.shape[0]
    need = 2 * n - 1
    size = 1 << ((need - 1).bit_length())
    A = rfft(a, n=size)
    conv = irfft(A * A, n=size)
    return conv[:need]


# ----------------------------------------------------------------------
def _objective(a: np.ndarray) -> float:
    """Exact objective 2·n·max(conv(a,a)) for a sequence with sum = 1."""
    n = a.shape[0]
    b = _fft_self_convolve(a)
    return 2.0 * n * b.max()

# Analytic families ----------------------------------------------------
def _distribution_uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _distribution_arcsine(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    vals /= vals.sum()
    return vals


def _distribution_tent(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    half = (n - 1) / 2.0
    w = np.where(i <= half, i + 1, n - i)
    w /= w.sum()
    return w


def _distribution_quadratic(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    half = (n - 1) / 2.0
    d = np.abs(i - half) + 1.0
    w = 1.0 / (d ** 2)
    w /= w.sum()
    return w


def _distribution_edge_weighted(n: int, p: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64) + 0.5
    j = (n - np.arange(n, dtype=np.float64) - 0.5)
    w = (i * j) ** (-p)
    w /= w.sum()
    return w


def _distribution_edge_inversepow(n: int, p: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64) + 0.5
    j = (n - np.arange(n, dtype=np.float64) - 0.5)
    w = i ** (-p) + j ** (-p)
    w /= w.sum()
    return w


def _distribution_sine_weighted(n: int, p: float) -> np.ndarray:
    theta = np.pi * (np.arange(n, dtype=np.float64) + 0.5) / n
    w = np.sin(theta) ** p
    w /= w.sum()
    return w


def _distribution_beta_sym(n: int, p: float) -> np.ndarray:
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    w = np.exp((p - 1) * np.log(x) + (p - 1) * np.log(1.0 - x))
    w /= w.sum()
    return w


def _distribution_exponential_decay(n: int, rate: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    half = (n - 1) / 2.0
    w = np.exp(-rate * np.abs(i - half))
    w /= w.sum()
    return w


def _distribution_random_sparse(
    n: int, rng: np.random.Generator, density: float = 0.01
) -> np.ndarray:
    k = max(2, int(n * density))
    idx = rng.choice(n, size=k, replace=False)
    vals = np.zeros(n, dtype=np.float64)
    vals[idx] = rng.random(k)
    if vals.sum() == 0.0:
        vals[0] = 1.0
    vals /= vals.sum()
    return vals


def _distribution_dirichlet(n: int, rng: np.random.Generator) -> np.ndarray:
    alpha = rng.uniform(0.01, 10.0)          # broadened concentration range
    a = rng.dirichlet(np.full(n, alpha))
    return a.astype(np.float64)


def _mix_with_uniform(base: np.ndarray, lam: float) -> np.ndarray:
    """Convex combination lam·U + (1‑lam)·base, where U is uniform."""
    n = base.shape[0]
    uniform = np.full(n, 1.0 / n, dtype=np.float64)
    return lam * uniform + (1.0 - lam) * base


# ----------------------------------------------------------------------
# Mirror‑descent (exponentiated gradient) -------------------------------
def _subgradient_descent_momentum(
    a: np.ndarray,
    deadline: float,
    best_seq: np.ndarray,
    best_score: float,
    lr0: float = 0.7,
    active_frac: float = 0.04,
    max_active: int = 30,
    beta: float = 0.9,
    eval_interval: int = 200,
    patience: int = 3000,
) -> tuple[np.ndarray, float]:
    """
    Exponentiated‑gradient descent with momentum on the exact objective.
    ``a`` is assumed to sum to one.
    """
    n = a.shape[0]
    t = 0
    no_improve = 0
    momentum = np.zeros_like(a)
    rng = np.random.default_rng()
    ind_grid = np.arange(n, dtype=np.int32)

    while time.time() < deadline and t < 500_000:
        t += 1
        conv = _fft_self_convolve(a)
        max_c = conv.max()
        thresh = max_c * (1.0 - active_frac)

        act = np.where(conv >= thresh)[0]
        if act.size > max_active:
            top_idx = np.argpartition(conv[act], -max_active)[-max_active:]
            act = act[top_idx]

        # sub‑gradient accumulation
        grad = np.zeros_like(a)
        for idx in act:
            offset = idx - ind_grid
            mask = (offset >= 0) & (offset < n)
            grad[mask] += 2.0 * a[offset[mask]]

        # momentum update
        momentum = beta * momentum + (1.0 - beta) * grad

        # decaying learning rate
        eta = lr0 / math.sqrt(t)

        # exponential (mirror) update
        a = a * np.exp(-eta * momentum)

        # project onto simplex
        a = _project_onto_simplex(a, s=1.0)

        # periodic evaluation
        if t % eval_interval == 0:
            cur = _objective(a)
            if cur < best_score - 1e-14:
                best_score = cur
                best_seq = a.copy()
                no_improve = 0
                print(f"[{time.time():.1f}] it={t:,} new global best {best_score:.6f}")
            else:
                no_improve += 1

        # jitter when stuck
        if no_improve >= patience:
            jitter = rng.exponential(scale=0.006, size=n)
            a = a + jitter
            a = np.maximum(a, 0.0)
            a = _project_onto_simplex(a, s=1.0)
            no_improve = 0

    return best_seq, best_score

# Greedy pairwise refinement -------------------------------------------
def _pairwise_swap_local(
    a: np.ndarray,
    deadline: float,
    best_seq: np.ndarray,
    best_score: float,
    max_trials: int | None = None,
) -> tuple[np.ndarray, float]:
    """Move a random amount of mass between two positions if the objective improves."""
    n = a.shape[0]
    trials = 0
    rng = np.random.default_rng()
    while time.time() < deadline:
        if max_trials is not None and trials >= max_trials:
            break
        trials += 1

        src = np.where(a > 1e-12)[0]
        if src.size == 0:
            break
        i = rng.choice(src)
        j = rng.integers(n - 1)
        if j >= i:
            j += 1

        delta = a[i] * rng.random() * 0.35
        if delta < 1e-14:
            continue

        new_a = a.copy()
        new_a[i] -= delta
        new_a[j] += delta
        cur = _objective(new_a)

        if cur < best_score - 1e-14:
            a = new_a
            best_score = cur
            best_seq = a.copy()
            print(f"[{time.time():.1f}] swap {i}->{j} Δ={delta:.3e} new best {best_score:.6f}")

    return best_seq, best_score

# Simulated‑annealing style local search -------------------------------
def _random_local_search(
    a: np.ndarray,
    deadline: float,
    best_seq: np.ndarray,
    best_score: float,
    init_temp: float = 0.008,
    decay: float = 0.9997,
) -> tuple[np.ndarray, float]:
    """Very cheap SA that repeatedly transfers a random amount of mass."""
    n = a.shape[0]
    rng = np.random.default_rng()
    T = init_temp
    while time.time() < deadline:
        i = rng.integers(n)
        j = rng.integers(n - 1)
        if j >= i:
            j += 1
        delta = a[i] * rng.random() * 0.3
        if delta < 1e-14:
            continue
        new_a = a.copy()
        new_a[i] -= delta
        new_a[j] += delta
        cur = _objective(new_a)
        if cur < best_score - 1e-14:
            a = new_a
            best_score = cur
            best_seq = a.copy()
            print(f"[{time.time():.1f}] SA improve {i}->{j} Δ={delta:.3e} best {best_score:.6f}")
        else:
            if rng.random() < math.exp(-(cur - best_score) / max(T, 1e-12)):
                a = new_a
        T *= decay
    return best_seq, best_score

# LP helper ------------------------------------------------------------
def _build_conv_matrix(f: np.ndarray) -> np.ndarray:
    """Return the (2n‑1)×n matrix A such that A @ g = conv(f,g)."""
    n = f.shape[0]
    rows = 2 * n - 1
    A = np.empty((rows, n), dtype=np.float64)
    A[:] = 0.0
    for i in range(n):
        A[i:i + n, i] = f[i]
    return A


def _alternating_lp_mix(
    a: np.ndarray,
    deadline: float,
    max_iter: int = 10,
    t: float = 0.5,
) -> tuple[np.ndarray, float]:
    """
    One round of “LP‑mix’’: maximise ∑g subject to conv(a,g)≤M, then mix
    g with a.  Repeated a few times (max_iter) gives a rapid flattening of
    the convolution.
    """
    best_seq = a.copy()
    best_score = _objective(a)
    n = a.shape[0]

    while time.time() < deadline and max_iter > 0:
        max_iter -= 1
        conv = _fft_self_convolve(a)
        M = conv.max()

        # build linear constraints conv(a,g) ≤ M
        A = _build_conv_matrix(a)
        b = np.full(2 * n - 1, M, dtype=np.float64)
        c = -np.ones(n, dtype=np.float64)               # maximise ∑g  ⇔ minimise -∑g
        bounds = [(0.0, None)] * n

        res = linprog(c, A_ub=A, b_ub=b, bounds=bounds, method="highs")
        if not res.success:
            # LP failed – fall back to plain gradient step
            break

        g = res.x
        total = g.sum()
        if total <= 0.0:
            break
        g_norm = g / total

        # mix and re‑project
        a = (1.0 - t) * a + t * g_norm
        a = _project_onto_simplex(a, s=1.0)

        cur = _objective(a)
        if cur < best_score - 1e-14:
            best_score = cur
            best_seq = a.copy()
            print(f"[{time.time():.1f}] LP‑mix improve to {best_score:.6f}")

        # slowly shrink the mixing weight
        t = max(0.1, t * 0.9)

    return best_seq, best_score

                    sc = _objective(cand)
                    top_candidates.append((sc, cand.copy()))
                    if sc < (top_candidates[0][0] if top_candidates else float("inf")):
                        print(f"[{time.time():.1f}] new seed p={p:.3g} lam={lam:.2f} score={sc:.6f}")

        # random Dirichlet seeds
        for _ in range(6):
            if time.time() >= deadline:
                break
            a = _distribution_dirichlet(n, rng)
            sc = _objective(a)
            top_candidates.append((sc, a.copy()))
            if sc < (top_candidates[0][0] if top_candidates else float("inf")):
                print(f"[{time.time():.1f}] new Dirichlet seed score={sc:.6f}")

        # sparse random seeds
        for _ in range(4):
            if time.time() >= deadline:
                break
            a = _distribution_random_sparse(n, rng, density=0.004)
            sc = _objective(a)
            top_candidates.append((sc, a.copy()))
            if sc < (top_candidates[0][0] if top_candidates else float("inf")):
                print(f"[{time.time():.1f}] new sparse seed score={sc:.6f}")

        # keep only the strongest seeds overall
        top_candidates.sort(key=lambda x: x[0])
        top_candidates = top_candidates[:keep]

    # ------------------------------------------------------------------
    # potential previous construction
    try:
        prev = np.asarray(height_sequence_1, dtype=np.float64)  # type: ignore[name-defined]
        if prev.ndim == 1 and prev.size > 0:
            prev = prev / prev.sum()
            sc = _objective(prev)
            top_candidates.append((sc, prev.copy()))
            print(f"[{time.time():.1f}] previous construction score={sc:.6f}")
    except Exception:
        pass

    if not top_candidates:
        # fallback – uniform of size 3000
        a = _distribution_uniform(3000)
        sc = _objective(a)
        top_candidates = [(sc, a)]
        print(f"[{time.time():.1f}] fallback uniform score={sc:.6f}")

    # sort final pool
    top_candidates.sort(key=lambda x: x[0])
    best_seq, best_score = top_candidates[0][1].copy(), top_candidates[0][0]

    # ------------------------------------------------------------------
    # 2.  Apply a few rounds of LP‑mix on the very best few seeds
    #    (limit to modest lengths so that the LP stays tractable)
    lp_candidates = [seq for (_, seq) in top_candidates[:5] if seq.shape[0] <= 2500]

    for idx, seed_seq in enumerate(lp_candidates):
        if time.time() >= deadline:
            break
        sub_deadline = time.time() + min(25.0, deadline - time.time())
        seq, score = _alternating_lp_mix(seed_seq.copy(), sub_deadline,
                                         max_iter=8, t=0.6)
        if score < best_score:
            best_seq, best_score = seq.copy(), score
            print(f"[{time.time():.1f}] LP‑mix seed {idx} improved best to {best_score:.6f}")

    # ------------------------------------------------------------------
    # 3.  Long momentum mirror‑descent on the current overall best
    if time.time() < deadline - 30.0:
        sub_deadline = time.time() + (deadline - time.time() - 30.0)
        print(f"[{time.time():.1f}] start long mirror descent")
        best_seq, best_score = _subgradient_descent_momentum(
            best_seq.copy(),
            sub_deadline,
            best_seq.copy(),
            best_score,
            lr0=0.8,
            active_frac=0.04,
            max_active=30,
            beta=0.9,
            eval_interval=250,
            patience=3000,
        )
        print(f"[{time.time():.1f}] after mirror descent best {best_score:.6f}")

    # ------------------------------------------------------------------
    # 4.  Greedy pairwise refinement
    if time.time() < deadline - 10.0:
        print(f"[{time.time():.1f}] greedy pairwise refinement")
        best_seq, best_score = _pairwise_swap_local(
            best_seq.copy(),
            deadline,
            best_seq.copy(),
            best_score,
            max_trials=300_000,
        )
        print(f"[{time.time():.1f}] after pairwise best {best_score:.6f}")

    # ------------------------------------------------------------------
    # 5.  Final simulated‑annealing style search
    if time.time() < deadline:
        print(f"[{time.time():.1f}] final simulated‑annealing search")
        best_seq, best_score = _random_local_search(
            best_seq.copy(),
            deadline,
            best_seq.copy(),
            best_score,
            init_temp=0.01,
            decay=0.9995,
        )
        print(f"[{time.time():.1f}] final best {best_score:.6f}")

    # ------------------------------------------------------------------
    print(f"Final best score (exact) ≈ {best_score:.6f} (length={len(best_seq)})")
    # Return a plain Python list – ``evaluate_sequence`` will normalise it.
    return best_seq.tolist()
