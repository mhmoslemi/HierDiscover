This solver builds a step‑function (a non‑negative vector that sums to 1) that minimises  

\[
\text{evaluate\_sequence}(a)=\frac{2\,n\;\max(a*a)}{(\sum a)^2}
\]

For a normalised vector (`sum(a)=1`) the objective simplifies to  

\[
F(a)=2\,n\;\max(a*a).
\]

The algorithm works in three stages, all bounded by the given time budget.

1. **Rich analytic initialisation** –   a handful of well‑known families  
   (uniform, arcsine, β‑distribution, edge‑weighted, exponential‑edge, …) are
   sampled on several lengths and mixed with the uniform distribution.
   The best few candidates are kept.

2. **Exact mirror‑descent (exponentiated‑gradient) optimisation** –  
   Starting from the selected seeds we run a sub‑gradient method that
   uses *all* indices attaining the current maximal convolution value.
   The update is an exponentiated‑gradient step followed by a projection
   onto the probability simplex.

3. **Stochastic polishing** –  
   A simulated‑annealing phase performs random mass transfers between
   coordinates; worsening moves are accepted with a temperature that
   decays geometrically.  
   Finally a greedy pair‑wise hill‑climber removes any remaining
   tiny imperfections.

All helper routines are written at module level (no closures) and only
standard libraries (`numpy`, `math`, `time`) are used.  
The function `propose_candidate` returns a plain Python list ready for
`evaluate_sequence`.

```python
"""
Proposes a step‑function that minimises ``evaluate_sequence``.

The method consists of

1. Scanning several analytic families (uniform, arcsine, β‑distribution,
   edge‑weighted, exponential‑edge, etc.) on a range of lengths and
   mixing them with the uniform distribution.  The best few seeds are
   retained.

2. Exact mirror‑descent (exponentiated‑gradient) on the true L∞ objective.
   The sub‑gradient is built from *all* indices attaining the maximal
   convolution value, and after each exponential update the vector is
   projected onto the probability simplex.

3. A short simulated‑annealing phase that randomly moves a fraction of
   mass from one coordinate to another; worsening moves are accepted
   with a temperature that decays geometrically.  A final greedy
   pair‑wise swap polishing finishes the search.

The whole pipeline respects a 1000 s wall‑time limit and returns the best
sequence found as a plain ``list[float]``.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x ≥ 0, Σx = s}."""
    if s < 0:
        raise ValueError("Simplex radius must be non‑negative")
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.where(u - cssv / (np.arange(1, len(v) + 1)) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    return np.maximum(v - theta, 0.0)


# ----------------------------------------------------------------------
def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Full self‑convolution a*a using a real FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1) - 1).bit_length()   # next power of two
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[: 2 * n - 1]


# ----------------------------------------------------------------------
def _objective(a: np.ndarray) -> float:
    """
    Exact objective 2·n·max(conv(a,a)) for a vector with Σa = 1.
    (The clipping performed in ``evaluate_sequence`` is omitted because we
    keep the vector non‑negative and normalised to sum 1.)
    """
    n = a.shape[0]
    b = _fft_self_convolve(a)
    return float(2.0 * n * b.max())


# ----------------------------------------------------------------------
# Analytic families ----------------------------------------------------
def _distribution_uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _distribution_arcsine(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    vals /= vals.sum()
    return vals


def _distribution_edge_weighted(n: int, p: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64) + 0.5
    j = (n - np.arange(n, dtype=np.float64) - 0.5)
    w = (i * j) ** (-p)
    w_sum = w.sum()
    if w_sum == 0.0:
        return np.full(n, 1.0 / n, dtype=np.float64)
    w /= w_sum
    return w


def _distribution_exponential_edge(n: int, c: float) -> np.ndarray:
    """w_i ∝ exp(-c·i) + exp(-c·(n‑1‑i))   (c>0)."""
    i = np.arange(n, dtype=np.float64)
    w = np.exp(-c * i) + np.exp(-c * (n - 1 - i))
    w /= w.sum()
    return w


def _distribution_beta(n: int, alpha: float, beta: float) -> np.ndarray:
    """Discretised Beta(α,β) on {0,…,n‑1}."""
    x = np.linspace(0.0, 1.0, n, dtype=np.float64)
    # add a tiny epsilon to avoid zeroes when α or β < 1
    eps = 1e-12
    vals = (x + eps) ** (alpha - 1) * (1 - x + eps) ** (beta - 1)
    vals /= vals.sum()
    return vals


def _distribution_random_sparse(
    n: int, rng: np.random.Generator, density: float = 0.01
) -> np.ndarray:
    """Sparse random vector with a prescribed density of non‑zeros."""
    k = max(2, int(n * density))
    idx = rng.choice(n, size=k, replace=False)
    vals = np.zeros(n, dtype=np.float64)
    vals[idx] = rng.random(k)
    s = vals.sum()
    if s == 0.0:
        vals[0] = 1.0
        s = 1.0
    vals /= s
    return vals


def _mix_with_uniform(base: np.ndarray, lam: float) -> np.ndarray:
    """Convex combination λ·uniform + (1‑λ)·base."""
    n = base.shape[0]
    uniform = np.full(n, 1.0 / n, dtype=np.float64)
    return lam * uniform + (1.0 - lam) * base


# ----------------------------------------------------------------------
# Mirror‑descent with exact sub‑gradient -------------------------------
def _subgradient_descent_exact(
    a0: np.ndarray,
    time_limit: float,
    best_seq: np.ndarray,
    best_score: float,
    lr0: float = 0.9,
    eval_every: int = 200,
) -> tuple[np.ndarray, float]:
    """
    Exponentiated‑gradient descent on the exact L∞ objective.
    The sub‑gradient is built from *all* indices attaining the maximal
    convolution value.
    """
    a = a0.copy()
    n = a.shape[0]
    t = 0
    while time.time() < time_limit:
        t += 1
        conv = _fft_self_convolve(a)
        max_val = conv.max()
        max_idxs = np.where(conv == max_val)[0]

        # exact sub‑gradient
        grad = np.zeros_like(a)
        for idx in max_idxs:
            i_start = max(0, idx - (n - 1))
            i_end = min(n - 1, idx)
            i = np.arange(i_start, i_end + 1, dtype=np.int64)
            j = idx - i
            grad[i] += 2.0 * a[j]

        eta = lr0 / math.sqrt(t)
        a *= np.exp(-eta * grad)          # EG step, keeps positivity
        s = a.sum()
        if s <= 0.0:
            a = np.full(n, 1.0 / n, dtype=np.float64)
        else:
            a /= s

        if t % eval_every == 0:
            cur = 2.0 * n * max_val
            if cur < best_score - 1e-14:
                best_score = cur
                best_seq = a.copy()
                print(
                    f"[{time.time():.1f}] subgrad it={t} new global best {best_score:.6f}"
                )
    return best_seq, best_score

# Simulated annealing --------------------------------------------------
def _simulated_annealing(
    a0: np.ndarray,
    time_limit: float,
    best_seq: np.ndarray,
    best_score: float,
    init_temp: float = 0.02,
    cooling: float = 0.9997,
) -> tuple[np.ndarray, float]:
    """
    Random mass‑transfer annealing.
    Moves that worsen the objective are accepted with probability
    exp(-Δ/temperature); the temperature decays geometrically.
    """
    a = a0.copy()
    n = a.shape[0]
    cur_score = _objective(a)
    best_local = a.copy()
    best_local_score = cur_score
    temperature = init_temp
    iteration = 0

    while time.time() < time_limit:
        iteration += 1

        # pick a source with non‑negligible mass
        src_candidates = np.where(a > 1e-14)[0]
        if src_candidates.size == 0:
            break
        i = np.random.choice(src_candidates)
        j = np.random.randint(n - 1)
        if j >= i:
            j += 1

        # move up to 45 % of source mass
        delta = a[i] * np.random.rand() * 0.45
        if delta < 1e-15:
            continue

        new_a = a.copy()
        new_a[i] -= delta
        new_a[j] += delta

        new_score = _objective(new_a)
        diff = new_score - cur_score

        if diff < 0 or np.random.rand() < math.exp(-diff / temperature):
            a = new_a
            cur_score = new_score
            if cur_score < best_local_score - 1e-14:
                best_local_score = cur_score
                best_local = a.copy()
                print(
                    f"[{time.time():.1f}] SA iter={iteration} new local best {best_local_score:.6f}"
                )
        temperature = max(temperature * cooling, 1e-12)

    if best_local_score < best_score - 1e-14:
        best_score = best_local_score
        best_seq = best_local.copy()
        print(f"[{time.time():.1f}] SA final best {best_score:.6f}")
    return best_seq, best_score

# Greedy pairwise mass transfer ----------------------------------------
def _pairwise_swap(
    a0: np.ndarray,
    time_limit: float,
    best_seq: np.ndarray,
    best_score: float,
    max_trials: int | None = None,
) -> tuple[np.ndarray, float]:
    """
    Deterministic hill‑climber that moves a random fraction of mass
    from one coordinate to another, accepting only strictly improving
    moves.
    """
    a = a0.copy()
    n = a.shape[0]
    trials = 0
    while time.time() < time_limit:
        if max_trials is not None and trials >= max_trials:
            break
        trials += 1

        src_candidates = np.where(a > 1e-14)[0]
        if src_candidates.size == 0:
            break
        i = np.random.choice(src_candidates)
        j = np.random.randint(n - 1)
        if j >= i:
            j += 1

        delta = a[i] * np.random.rand() * 0.45
        if delta < 1e-15:
            continue

        new_a = a.copy()
        new_a[i] -= delta
        new_a[j] += delta

        cur = _objective(new_a)
        if cur < best_score - 1e-14:
            a = new_a
            best_score = cur
            best_seq = a.copy()
            print(
                f"[{time.time():.1f}] swap i={i}→j={j} δ={delta:.6g} new best {best_score:.6f}"
            )
    return best_seq, best_score

            top_candidates.append((sc, a.copy()))
            if sc < overall_best_score:
                overall_best_score, overall_best_seq = sc, a.copy()
                print(f"[{time.time():.1f}] Dirichlet len={n} α={alpha:.3f} score={sc:.6f}")

        # sparse random seeds ----------------------------------------------------
        for _ in range(4):
            if time.time() >= deadline:
                break
            a = _distribution_random_sparse(n, rng, density=0.006)
            sc = _objective(a)
            top_candidates.append((sc, a.copy()))
            if sc < overall_best_score:
                overall_best_score, overall_best_seq = sc, a.copy()
                print(f"[{time.time():.1f}] sparse len={n} score={sc:.6f}")

        # keep only the best 40 seeds (memory & time bound)
        top_candidates.sort(key=lambda x: x[0])
        top_candidates = top_candidates[:40]

    # ------------------------------------------------------------------
    # 2. Include previous construction if it exists (height_sequence_1)
    # ------------------------------------------------------------------
    try:
        prev = np.asarray(height_sequence_1, dtype=np.float64)  # type: ignore[name-defined]
        if prev.ndim == 1 and prev.size > 0:
            prev = np.maximum(prev, 0.0)
            if prev.sum() == 0.0:
                prev = _distribution_uniform(prev.size)
            else:
                prev = prev / prev.sum()
            sc = _objective(prev)
            if sc < overall_best_score:
                overall_best_score, overall_best_seq = sc, prev.copy()
                print(f"[{time.time():.1f}] previous construction score={sc:.6f}")
            top_candidates.append((sc, prev.copy()))
    except Exception:
        pass

    # ------------------------------------------------------------------
    # 3. Fallback – should never happen
    # ------------------------------------------------------------------
    if overall_best_seq is None:
        n = 8000
        overall_best_seq = _distribution_uniform(n)
        overall_best_score = _objective(overall_best_seq)
        print(
            f"[{time.time():.1f}] fallback uniform length={n} score={overall_best_score:.6f}"
        )

    # ------------------------------------------------------------------
    # 4. Exact mirror‑descent refinement on the most promising seeds
    # ------------------------------------------------------------------
    remaining = deadline - time.time()
    if remaining > 30.0 and top_candidates:
        # keep the five best seeds for deeper optimisation
        seeds = [seq for _, seq in sorted(top_candidates, key=lambda x: x[0])[:5]]
        per_seed = remaining / max(1, len(seeds))
        print(f"[{time.time():.1f}] start exact subgradient on {len(seeds)} seeds")
        for idx, seed_seq in enumerate(seeds):
            if time.time() >= deadline:
                break
            sg_deadline = time.time() + per_seed
            a_opt, sc_opt = _subgradient_descent_exact(
                seed_seq,
                sg_deadline,
                best_seq=overall_best_seq,
                best_score=overall_best_score,
                lr0=0.9,
                eval_every=200,
            )
            if sc_opt < overall_best_score - 1e-12:
                overall_best_score, overall_best_seq = sc_opt, a_opt.copy()
                print(
                    f"[{time.time():.1f}] after subgrad seed {idx} new global best {overall_best_score:.6f}"
                )

    # ------------------------------------------------------------------
    # 5. Simulated annealing polishing
    # ------------------------------------------------------------------
    remaining = deadline - time.time()
    if remaining > 10.0:
        print(f"[{time.time():.1f}] simulated annealing polishing")
        a_opt, sc_opt = _simulated_annealing(
            overall_best_seq.copy(),
            deadline,
            best_seq=overall_best_seq,
            best_score=overall_best_score,
            init_temp=0.02,
            cooling=0.9998,
        )
        if sc_opt < overall_best_score - 1e-12:
            overall_best_score, overall_best_seq = sc_opt, a_opt.copy()
            print(f"[{time.time():.1f}] after SA new best {overall_best_score:.6f}")

    # ------------------------------------------------------------------
    # 6. Final greedy pairwise polishing (hill‑climbing)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        print(f"[{time.time():.1f}] final greedy swap polishing")
        a_opt, sc_opt = _pairwise_swap(
            overall_best_seq.copy(),
            deadline,
            best_seq=overall_best_seq,
            best_score=overall_best_score,
            max_trials=None,
        )
        if sc_opt < overall_best_score - 1e-12:
            overall_best_score, overall_best_seq = sc_opt, a_opt.copy()
            print(f"[{time.time():.1f}] after final swaps new best {overall_best_score:.6f}")

    # ------------------------------------------------------------------
    print(
        f"Final best objective (2·n·max(conv)) = {overall_best_score:.6f} (length={len(overall_best_seq)})"
    )
    # Return a plain Python list – ``evaluate_sequence`` will re‑normalise internally.
    return overall_best_seq.tolist()
```
