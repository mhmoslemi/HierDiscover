"""
Propose a non‑negative step function that (almost) minimises
``evaluate_sequence``.

The search has three ideas that differ from the previous implementation:

1. **Soft‑max (smooth max) gradient descent** – instead of using the
   sub‑gradient of the single maximal convolution entry we minimise a
   smooth approximation  τ·log Σ exp(conv/τ).  The gradient is a weighted
   sum of the gradients of many entries and therefore provides a
   stronger descent direction.  The temperature τ is annealed as
   τ ∝ 1/√t.

2. **Symmetrisation** – the optimal shape is symmetric around the centre,
   so after each update we replace a by (a + a[::-1]) / 2.  This reduces
   the number of effective variables and keeps the search in the
   promising symmetric sub‑space.

3. **Hybrid schedule** – we start from a diverse pool of analytic
   families (uniform, arcsine, beta, edge‑weighted, Hann,
   generalised‑normal, power‑law, Dirichlet and sparse).  The best few
   seeds are refined with the smooth‑max descent, then a short exact
   max‑sub‑gradient phase, possible up‑sampling and a final
   pair‑wise annealing polish.

All steps respect a global 1000 s budget (2 s safety margin) and
work only with NumPy, no external solvers.

# ----------------------------------------------------------------------
# Simplex projection (Euclidean, used only for safety)
# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x ≥ 0, Σx = s}."""
    if s < 0:
        raise ValueError("simplex radius must be non‑negative")
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.where(u - cssv / (np.arange(1, len(v) + 1)) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# FFT based convolutions (full linear convolution)
# ----------------------------------------------------------------------
def _fft_convolve(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full linear convolution a * b using a real FFT."""
    n = a.shape[0]
    m = b.shape[0]
    size = 1 << ((n + m - 1) - 1).bit_length()
    A = np.fft.rfft(a, n=size)
    B = np.fft.rfft(b, n=size)
    conv = np.fft.irfft(A * B, n=size)
    return conv[: n + m - 1]


def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Full linear self‑convolution a*a using a real FFT."""
    return _fft_convolve(a, a)


# ----------------------------------------------------------------------
# Exact objective used by ``evaluate_sequence`` (scale‑invariant)
# ----------------------------------------------------------------------
def _objective(a: np.ndarray) -> float:
    """Return the exact value of ``evaluate_sequence`` for a (unnormalised)."""
    n = a.shape[0]
    b = _fft_self_convolve(a)
    max_b = b.max()
    sum_a = a.sum()
    if sum_a < 0.01:
        return float("inf")
    return float(2.0 * n * max_b / (sum_a * sum_a))

# ----------------------------------------------------------------------
def _distribution_uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _distribution_arcsine(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    vals /= vals.sum()
    return vals


def _distribution_beta(n: int, a: float, b: float) -> np.ndarray:
    """Beta(a,b) on the uniform grid (a,b > 0)."""
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    w = np.exp((a - 1) * np.log(x) + (b - 1) * np.log(1.0 - x))
    if w.sum() == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w

    j = (n - np.arange(n, dtype=np.float64) - 0.5)
    w = (i * j) ** (-p)
    if w.sum() == 0.0:

    else:

    return w

    w = 0.5 - 0.5 * np.cos(2.0 * np.pi * i / (n - 1))
    w_sum = w.sum()

    n = base.shape[0]
    uniform = np.full(n, 1.0 / n, dtype=np.float64)
    return lam * uniform + (1.0 - lam) * base


# ----------------------------------------------------------------------
# Gradient of the *soft‑max* of the convolution (smooth approximation)
# ----------------------------------------------------------------------
def _softmax_gradient(a: np.ndarray, temperature: float) -> np.ndarray:
    """
    Compute ∇_{a} Σ_k w_k (a*a)[k] where
    w_k ∝ exp(conv_k / temperature) and Σ w_k = 1.
    """
    n = a.shape[0]
    conv = _fft_self_convolve(a)          # length 2n‑1
    max_conv = conv.max()
    # avoid overflow – shift by max_conv
    w = np.exp((conv - max_conv) / max(temperature, 1e-12))
    w_sum = w.sum()
    if w_sum == 0.0:
        w = np.full_like(w, 1.0 / w.shape[0])
    else:
        w /= w_sum

    # gradient = 2 * (a ★ w_rev)[offset‑j]  (★ = convolution)
    w_rev = w[::-1]
    full = _fft_convolve(a, w_rev)               # length n + (2n‑1) – 1 = 3n‑2
    offset = w.shape[0] - 1                       # index where a and w align
    grad = 2.0 * full[offset - np.arange(n)]
    return grad


# ----------------------------------------------------------------------
# Exponentiated‑gradient descent on the *exact* max‑convolution (sub‑gradient)
# ----------------------------------------------------------------------
def _exp_gradient_descent(a: np.ndarray,
                         deadline: float,
                         best_seq: np.ndarray,
                         best_score: float,
                         eta0: float = 0.8,
                         eval_every: int = 80) -> tuple[np.ndarray, float]:
    """Multiplicative mirror descent using the sub‑gradient of the *max* entry."""
    n = a.shape[0]
    t = 0
    while time.time() < deadline:
        t += 1
        # active index gradient
        conv = _fft_self_convolve(a)
        k = int(np.argmax(conv))
        grad = np.zeros_like(a)
        if k < n:
            grad[: k + 1] = 2.0 * a[k::-1]
        else:
            start = k - (n - 1)
            grad[start:] = 2.0 * a[start:][::-1]

        eta = eta0 / math.sqrt(t)
        a = a * np.exp(-eta * grad)
        s = a.sum()
        if s == 0.0:
            a = np.full(n, 1.0 / n, dtype=np.float64)
        else:
            a = a / s

        if t % eval_every == 0:
            cur = _objective(a)
            if cur < best_score - 1e-14:
                best_score = cur
                best_seq = a.copy()
                print(f"[{time.time():.1f}] EG it={t} new best {best_score:.6f}")
    return best_seq, best_score


# ----------------------------------------------------------------------
# Smooth‑max (softmax) exponentiated‑gradient descent
# ----------------------------------------------------------------------
def _softmax_exp_descent(a: np.ndarray,
                         deadline: float,
                         best_seq: np.ndarray,
                         best_score: float,
                         eta0: float = 0.6,
                         temp0_factor: float = 0.2,
                         eval_every: int = 100) -> tuple[np.ndarray, float]:
    """
    Multiplicative update using the gradient of a temperature‑scaled
    soft‑max of the convolution.  The temperature decays as 1/√t.
    """
    n = a.shape[0]
    t = 0
    while time.time() < deadline:
        t += 1
        conv = _fft_self_convolve(a)
        max_conv = conv.max()
        # temperature schedule
        temperature = max(1e-12, temp0_factor * max_conv / math.sqrt(t))
        grad = _softmax_gradient(a, temperature)

        eta = eta0 / math.sqrt(t)
        a = a * np.exp(-eta * grad)
        # enforce simplex
        s = a.sum()
        if s == 0.0:
            a = np.full(n, 1.0 / n, dtype=np.float64)
        else:
            a = a / s

        # enforce symmetry (helps convergence)
        a = 0.5 * (a + a[::-1])

        if t % eval_every == 0:
            cur = _objective(a)
            if cur < best_score - 1e-14:
                best_score = cur
                best_seq = a.copy()
                print(f"[{time.time():.1f}] SGD it={t} new best {best_score:.6f}")
    return best_seq, best_score


# ----------------------------------------------------------------------
# Frank‑Wolfe style refinement (same as before – kept for safety)
# ----------------------------------------------------------------------
def _frank_wolfe_descent(a: np.ndarray,
                         deadline: float,
                         best_seq: np.ndarray,
                         best_score: float,
                         eval_every: int = 100) -> tuple[np.ndarray, float]:
    n = a.shape[0]
    t = 0
    while time.time() < deadline:
        t += 1
        conv = _fft_self_convolve(a)
        grad, _ = _grad_of_max_conv(a)  # re‑use exact max gradient
        i_min = int(np.argmin(grad))
        gamma = 2.0 / (t + 2.0)
        a = a + gamma * (np.eye(1, n, i_min)[0] - a)
        if t % eval_every == 0:
            cur = _objective(a)
            if cur < best_score - 1e-14:
                best_score = cur
                best_seq = a.copy()
                print(f"[{time.time():.1f}] FW it={t} new best {best_score:.6f}")
    return best_seq, best_score


# ----------------------------------------------------------------------
# Helper to compute exact max‑gradient (used by FW above)
# ----------------------------------------------------------------------
def _grad_of_max_conv(a: np.ndarray) -> tuple[np.ndarray, int]:
    n = a.shape[0]
    conv = _fft_self_convolve(a)
    k = int(np.argmax(conv))
    grad = np.zeros_like(a)
    if k < n:
        grad[: k + 1] = 2.0 * a[k::-1]
    else:
        start = k - (n - 1)
        grad[start:] = 2.0 * a[start:][::-1]
    return grad, k


# ----------------------------------------------------------------------
# Simulated‑annealing style pairwise mass transfers (final polishing)
# ----------------------------------------------------------------------
def _pairwise_annealing(a: np.ndarray,
                        deadline: float,
                        best_seq: np.ndarray,
                        best_score: float,
                        rng: np.random.Generator,
                        max_trials: int | None = None) -> tuple[np.ndarray, float]:
    n = a.shape[0]
    start_time = time.time()
    total_time = deadline - start_time
    trials = 0
    while time.time() < deadline:
        if max_trials is not None and trials >= max_trials:
            break
        trials += 1

        src_candidates = np.where(a > 1e-12)[0]
        if src_candidates.size == 0:
            break
        i = rng.choice(src_candidates)

        j = rng.integers(n - 1)
        if j >= i:
            j += 1

        delta = a[i] * rng.random() * 0.4
        if delta < 1e-16:
            continue

        new_a = a.copy()
        new_a[i] -= delta
        new_a[j] += delta

        cur = _objective(new_a)
        elapsed = time.time() - start_time
        temperature = max(0.0, 1.0 - elapsed / total_time) * best_score * 0.01

        if cur < best_score - 1e-14:
            accept = True
        else:
            accept = temperature > 0.0 and rng.random() < math.exp(-(cur - best_score) / temperature)

        if accept:
            a = new_a
            if cur < best_score - 1e-14:
                best_score = cur
                best_seq = a.copy()
                print(f"[{time.time():.1f}] anneal i={i}->j={j} δ={delta:.3e} new best {best_score:.6f}")
    return best_seq, best_score


# ----------------------------------------------------------------------
# Linear up‑sampling while preserving the simplex (used after a good candidate)
# ----------------------------------------------------------------------
def _upsample(a: np.ndarray, new_n: int) -> np.ndarray:
    """Interpolate a probability vector to a larger length."""
    old_n = a.shape[0]
    if new_n == old_n:
        return a.copy()
    x_old = (np.arange(old_n, dtype=np.float64) + 0.5) / old_n
    x_new = (np.arange(new_n, dtype=np.float64) + 0.5) / new_n
    a_new = np.interp(x_new, x_old, a, left=0.0, right=0.0)
    a_new = np.maximum(a_new, 0.0)
    if a_new.sum() == 0.0:
        a_new[:] = 1.0 / new_n
    else:
        a_new /= a_new.sum()
    return a_new


# ----------------------------------------------------------------------
# Main entry point – the function that will be called by the evaluator
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: int = 1000) -> list[float]:
    """
    Search for a non‑negative step function that minimises
    ``evaluate_sequence``.  The routine respects a 1000 s budget and
    returns a plain ``list[float]`` ready for evaluation.
    """
    rng = np.random.default_rng(seed)
    start = time.time()
    deadline = start + budget_s - 2.0   # safety margin

    overall_best_seq: np.ndarray | None = None
    overall_best_score = float("inf")
    top_candidates: list[tuple[float, np.ndarray]] = []

    # ------------------------------------------------------------------
    # 1.  Populate a diverse pool of seeds (length 4096 gives a good
    #     trade‑off between resolution and speed).
    # ------------------------------------------------------------------
    n0 = 4096
    p_grid = np.concatenate(
        [np.logspace(-1.5, -0.2, 12), np.logspace(-0.2, 0.5, 12)]
    )
    lam_grid = np.linspace(0.0, 1.0, 5)
    q_grid = np.linspace(0.4, 3.0, 9)
    power_grid = np.linspace(0.2, 2.0, 9)

    print(f"[{time.time():.1f}] generating seeds (length={n0})")

    # deterministic families
    for cand in (_distribution_uniform(n0),
                 _distribution_arcsine(n0),
                 _distribution_hann(n0)):
        sc = _objective(cand)
        top_candidates.append((sc, cand.copy()))
        if sc < overall_best_score:
            overall_best_score = sc
            overall_best_seq = cand.copy()
            print(f"[{time.time():.1f}] new global best (deterministic) {sc:.6f}")

    # beta families – both symmetric and asymmetric
    for a in p_grid:
        if time.time() >= deadline:
            break
        for b in p_grid:
            beta = _distribution_beta(n0, a, b)
            sc = _objective(beta)
            top_candidates.append((sc, beta.copy()))
            if sc < overall_best_score:
                overall_best_score = sc
                overall_best_seq = beta.copy()
                print(f"[{time.time():.1f}] new best Beta({a:.3g},{b:.3g}) {sc:.6f}")

    # edge‑weighted power‑law
    for p in p_grid:
        if time.time() >= deadline:
            break
        edge = _distribution_edge_weighted(n0, p)
        sc = _objective(edge)
        top_candidates.append((sc, edge.copy()))
        if sc < overall_best_score:
            overall_best_score = sc
            overall_best_seq = edge.copy()
            print(f"[{time.time():.1f}] new best edge weighted p={p:.3g} {sc:.6f}")

    # generalized‑normal
    for q in q_grid:
        if time.time() >= deadline:
            break
        gn = _distribution_generalized_normal(n0, q)
        sc = _objective(gn)
        top_candidates.append((sc, gn.copy()))
        if sc < overall_best_score:
            overall_best_score = sc
            overall_best_seq = gn.copy()
            print(f"[{time.time():.1f}] new best G‑Normal q={q:.2f} {sc:.6f}")

    # power‑law families
    for pwr in power_grid:
        if time.time() >= deadline:
            break
        pl = _distribution_power_law(n0, pwr)
        sc = _objective(pl)
        top_candidates.append((sc, pl.copy()))
        if sc < overall_best_score:
            overall_best_score = sc
            overall_best_seq = pl.copy()
            print(f"[{time.time():.1f}] new best power‑law p={pwr:.2f} {sc:.6f}")

    # random Dirichlet seeds
    for _ in range(12):
        if time.time() >= deadline:
            break
        alpha = rng.uniform(0.01, 2.0)
        cand = rng.dirichlet(np.full(n0, alpha))
        sc = _objective(cand)
        top_candidates.append((sc, cand.copy()))
        if sc < overall_best_score:
            overall_best_score = sc
            overall_best_seq = cand.copy()
            print(f"[{time.time():.1f}] new best Dirichlet α={alpha:.3f} {sc:.6f}")

    # sparse random seeds
    for _ in range(8):
        if time.time() >= deadline:
            break
        cand = _distribution_random_sparse(n0, rng, density=0.01)
        sc = _objective(cand)
        top_candidates.append((sc, cand.copy()))
        if sc < overall_best_score:
            overall_best_score = sc
            overall_best_seq = cand.copy()
            print(f"[{time.time():.1f}] new best sparse {sc:.6f}")

    # keep the best 30 candidates
    top_candidates.sort(key=lambda x: x[0])
    top_candidates = top_candidates[:30]

    # ------------------------------------------------------------------
    # 2.  Warm‑start from a previously discovered construction if present.
    # ------------------------------------------------------------------
    try:
        prev = np.asarray(height_sequence_1, dtype=np.float64)  # type: ignore[name-defined]
        if prev.ndim == 1:
            prev = prev / prev.sum()
            sc = _objective(prev)
            if sc < overall_best_score:
                overall_best_score = sc
                overall_best_seq = prev.copy()
                print(f"[{time.time():.1f}] imported previous best {sc:.6f}")
            top_candidates.append((sc, prev.copy()))
    except Exception:
        pass

    # ------------------------------------------------------------------
    # 3.  Safety fallback (should never trigger).
    # ------------------------------------------------------------------
    if overall_best_seq is None:
        overall_best_seq = _distribution_uniform(n0)
        overall_best_score = _objective(overall_best_seq)
        print(f"[{time.time():.1f}] fallback uniform {overall_best_score:.6f}")

    # ------------------------------------------------------------------
    # 4.  Refine the top seeds with *soft‑max* exponentiated gradient descent.
    # ------------------------------------------------------------------
    remaining = deadline - time.time()
    if remaining > 20.0 and top_candidates:
        seed_vectors = [seq for _, seq in sorted(top_candidates, key=lambda x: x[0])[:5]]
        per_seed = remaining * 0.4 / max(1, len(seed_vectors))
        print(f"[{time.time():.1f}] soft‑max GD on {len(seed_vectors)} seeds, {per_seed:.1f}s each")
        for idx, seq in enumerate(seed_vectors):
            if time.time() >= deadline:
                break
            sub_deadline = time.time() + per_seed
            a_opt, sc_opt = _softmax_exp_descent(
                seq.copy(),
                sub_deadline,
                best_seq=overall_best_seq,
                best_score=overall_best_score,
                eta0=0.6,
                temp0_factor=0.25,
                eval_every=80,
            )
            if sc_opt < overall_best_score - 1e-12:
                overall_best_score = sc_opt
                overall_best_seq = a_opt.copy()
                print(f"[{time.time():.1f}] after SGD seed {idx} new best {overall_best_score:.6f}")

    # ------------------------------------------------------------------
    # 5.  A short exact‑max EG refinement (helps escape soft‑max plateaus).
    # ------------------------------------------------------------------
    if time.time() < deadline:
        sub_deadline = time.time() + min(remaining * 0.15, 30.0)
        print(f"[{time.time():.1f}] exact‑max EG refinement")
        overall_best_seq, overall_best_score = _exp_gradient_descent(
            overall_best_seq.copy(),
            sub_deadline,
            best_seq=overall_best_seq,
            best_score=overall_best_score,
            eta0=0.9,
            eval_every=60,
        )
        print(f"[{time.time():.1f}] after exact EG best score {overall_best_score:.6f}")

    # ------------------------------------------------------------------
    # 6.  Optional up‑sampling and a second soft‑max burst on a larger vector.
    # ------------------------------------------------------------------
    if time.time() < deadline:
        cur_n = overall_best_seq.shape[0]
        target_n = cur_n * 2
        if target_n <= 64000:
            print(f"[{time.time():.1f}] upsampling {cur_n} → {target_n}")
            up_seq = _upsample(overall_best_seq, target_n)
            up_score = _objective(up_seq)
            if up_score < overall_best_score:
                overall_best_score = up_score
                overall_best_seq = up_seq
                print(f"[{time.time():.1f}] upsampled new best {overall_best_score:.6f}")

            # brief soft‑max optimisation on the larger vector
            sub_deadline = time.time() + (deadline - time.time()) * 0.15
            a_opt, sc_opt = _softmax_exp_descent(
                overall_best_seq.copy(),
                sub_deadline,
                best_seq=overall_best_seq,
                best_score=overall_best_score,
                eta0=0.5,
                temp0_factor=0.2,
                eval_every=70,
            )
            if sc_opt < overall_best_score - 1e-12:
                overall_best_score = sc_opt
                overall_best_seq = a_opt.copy()
                print(f"[{time.time():.1f}] SGD after upsample new best {overall_best_score:.6f}")

    # ------------------------------------------------------------------
    # 7.  Final long‑run pairwise annealing polish.
    # ------------------------------------------------------------------
    if time.time() < deadline:
        print(f"[{time.time():.1f}] final annealing polish")
        overall_best_seq, overall_best_score = _pairwise_annealing(
            overall_best_seq.copy(),
            deadline,
            best_seq=overall_best_seq,
            best_score=overall_best_score,
            rng=rng,
            max_trials=None,
        )
        print(f"[{time.time():.1f}] after annealing best score {overall_best_score:.6f}")

    # ------------------------------------------------------------------
    # 8.  Report and return
    # ------------------------------------------------------------------
    print(
        f"Final best score (2·n·max(conv)/sum²) = {overall_best_score:.6f} "
        f"with length = {len(overall_best_seq)}"
    )
    # ``evaluate_sequence`` will clamp values to [0,1000] and enforce non‑negativity,
    # but we keep the vector clean already.
    return overall_best_seq.tolist()
