def _fft_convolve(x, y):
    n = len(x)
    m = len(y)
    size = 1 << ((n + m - 1) - 1).bit_length()
    X = np.fft.rfft(x, n=size)
    Y = np.fft.rfft(y, n=size)
    conv = np.fft.irfft(X * Y, n=size)
    return conv[: n + m - 1]

"""
Search for a non‑negative step‑function that minimizes ``evaluate_sequence``.

Algorithm
---------
1️⃣  Scan a large collection of analytic families (uniform, arcsine, edge‑weighted,
    exponential‑edge, Gaussian, super‑Gaussian, Kaiser, Tukey, Blackman, …) for
    several lengths.  Random convex mixtures of two families are also generated.
    The best few candidates are kept.

2️⃣  Each kept candidate is refined by a *smooth‑max* mirror‑descent (soft‑max
    approximation of the L∞ convolution norm).  After a number of iterations the
    schedule switches to the exact sub‑gradient descent used earlier to polish
    the solution.

3️⃣  Finally a greedy pairwise mass‑transfer hill‑climber swaps small fractions
    of mass between positions and accepts strictly improving moves.

All work is bounded by the supplied budget (default 1000 s).  The function
``propose_candidate`` returns a plain Python list that can be fed directly to
``evaluate_sequence``.

def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x ≥ 0, Σx = s}."""
    if s < 0:
        raise ValueError("Simplex radius must be non‑negative")
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.where(u - cssv / (np.arange(1, len(v) + 1)) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    return np.maximum(v - theta, 0.0)


# ----------------------------------------------------------------------
def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Full self‑convolution a*a using a real FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1) - 1).bit_length()   # next power of two
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[: 2 * n - 1]


def _fft_convolve(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution of two real vectors via FFT."""
    n = len(x)
    m = len(y)
    size = 1 << ((n + m - 1) - 1).bit_length()
    X = np.fft.rfft(x, n=size)
    Y = np.fft.rfft(y, n=size)
    conv = np.fft.irfft(X * Y, n=size)
    return conv[: n + m - 1]


# ----------------------------------------------------------------------
def _objective(a: np.ndarray) -> float:
    """
    Exact objective 2·n·max(conv(a,a)) for a vector with Σa = 1.
    (The clipping performed in ``evaluate_sequence`` is omitted,
    because we keep the vector non‑negative and normalised to sum 1.)
    """
    n = a.shape[0]
    b = _fft_self_convolve(a)
    return float(2.0 * n * b.max())

# Analytic families ----------------------------------------------------
def _distribution_uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _distribution_arcsine(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    vals /= vals.sum()
    return vals


def _distribution_sine(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64) + 0.5
    w = np.sin(np.pi * i / n)
    w /= w.sum()
    return w


def _distribution_cosine(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64) + 0.5
    w = 1.0 - np.cos(np.pi * i / n)
    w /= w.sum()
    return w


def _distribution_quadratic(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    x = (2.0 * i) / (n - 1) - 1.0
    w = 1.0 - x * x
    w = np.clip(w, 0.0, None)
    w /= w.sum()
    return w


def _distribution_edge_weighted(n: int, p: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64) + 0.5
    j = (n - np.arange(n, dtype=np.float64) - 0.5)
    w = (i * j) ** (-p)
    w_sum = w.sum()
    if w_sum == 0.0:
        return _distribution_uniform(n)
    w /= w_sum
    return w


def _distribution_exponential_edge(n: int, c: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    w = np.exp(-c * i) + np.exp(-c * (n - 1 - i))
    w /= w.sum()
    return w


def _distribution_gaussian(n: int, sigma: float) -> np.ndarray:
    """Symmetric Gaussian centred in the middle."""
    centre = (n - 1) / 2.0
    i = np.arange(n, dtype=np.float64)
    w = np.exp(-0.5 * ((i - centre) / sigma) ** 2)
    w_sum = w.sum()
    if w_sum == 0.0:
        return _distribution_uniform(n)
    w /= w_sum
    return w


def _distribution_supergaussian(n: int, sigma: float, power: float) -> np.ndarray:
    centre = (n - 1) / 2.0
    i = np.arange(n, dtype=np.float64)
    w = np.exp(-((np.abs(i - centre) / sigma) ** power))
    w_sum = w.sum()
    if w_sum == 0.0:
        return _distribution_uniform(n)
    w /= w_sum
    return w


def _distribution_kaiser(n: int, beta: float) -> np.ndarray:
    if not _has_signal:
        return _distribution_uniform(n)
    w = signal.kaiser(n, beta)
    w = np.maximum(w, 0.0)
    w /= w.sum()
    return w


def _distribution_tukey(n: int, alpha: float) -> np.ndarray:
    if not _has_signal:
        return _distribution_uniform(n)
    w = signal.tukey(n, alpha)
    w = np.maximum(w, 0.0)
    w /= w.sum()
    return w


def _distribution_blackman(n: int) -> np.ndarray:
    if not _has_signal:
        return _distribution_uniform(n)
    w = signal.blackman(n)
    w = np.maximum(w, 0.0)
    w /= w.sum()
    return w


def _distribution_random_sparse(
    n: int, rng: np.random.Generator, density: float = 0.01
) -> np.ndarray:
    """Sparse random vector with a prescribed density of non‑zeros."""
    k = max(2, int(n * density))
    idx = rng.choice(n, size=k, replace=False)
    vals = np.zeros(n, dtype=np.float64)
    vals[idx] = rng.random(k)
    s = vals.sum()
    if s == 0.0:
        vals[0] = 1.0
        s = 1.0
    vals /= s
    return vals


def _mix_with_uniform(base: np.ndarray, lam: float) -> np.ndarray:
    """Convex combination λ·uniform + (1‑λ)·base."""
    n = base.shape[0]
    uniform = np.full(n, 1.0 / n, dtype=np.float64)
    return lam * uniform + (1.0 - lam) * base

# Exact sub‑gradient descent (same as in the reference implementation)
def _subgradient_descent_exact(
    a0: np.ndarray,
    time_limit: float,
    best_seq: np.ndarray,
    best_score: float,
    lr0: float = 0.9,
    eval_every: int = 100,
) -> tuple[np.ndarray, float]:
    """
    Sub‑gradient descent on the exact L∞ objective (max‑convolution).
    Uses exponentiated‑gradient updates and projects onto the simplex.
    """
    a = a0.copy()
    n = a.shape[0]
    t = 0
    while time.time() < time_limit:
        t += 1
        conv = _fft_self_convolve(a)
        max_val = conv.max()
        max_idxs = np.where(conv == max_val)[0]

        # exact sub‑gradient (sum over all active max indices)
        grad = np.zeros_like(a)
        for idx in max_idxs:
            i_start = max(0, idx - (n - 1))
            i_end = min(n - 1, idx)
            i = np.arange(i_start, i_end + 1, dtype=np.int64)
            j = idx - i
            grad[i] += 2.0 * a[j]

        eta = lr0 / math.sqrt(t)
        a *= np.exp(-eta * grad)          # exponentiated‑gradient step
        s = a.sum()
        if s <= 0.0:
            a = np.full(n, 1.0 / n, dtype=np.float64)
        else:
            a /= s

        if t % eval_every == 0:
            cur = 2.0 * n * max_val        # Σa = 1 ⇒ objective = 2·n·maxconv
            if cur < best_score - 1e-14:
                best_score = cur
                best_seq = a.copy()
                print(
                    f"[{time.time():.1f}] exact subgrad it={t} new global best {best_score:.6f}"
                )
    return best_seq, best_score

# Smooth‑max gradient descent (soft‑max approximation of the max)
def _smooth_gradient_descent(
    a0: np.ndarray,
    time_limit: float,
    best_seq: np.ndarray,
    best_score: float,
    beta_start: float = 0.5,
    beta_target: float = 150.0,
    lr0: float = 0.5,
    eval_every: int = 200,
) -> tuple[np.ndarray, float]:
    """
    Mirror‑descent on the smooth approximation
        φ_β(a) = (1/β)·log Σ_k exp(β·conv_k(a))

    The gradient of φ_β is a weighted sum of the gradients of each convolution
    entry.  An exponentiated‑gradient update keeps the iterate on the simplex.
    The smoothing parameter β is increased geometrically during the run.
    """
    a = a0.copy()
    n = a.shape[0]
    t = 0
    beta = beta_start
    beta_inc = (beta_target / beta_start) ** (1 / max(1, int((time_limit - time.time()) / 0.05)))
    while time.time() < time_limit:
        t += 1
        conv = _fft_self_convolve(a)                 # length 2n‑1
        # soft‑max weights, numerically stable
        shifted = conv - conv.max()
        w = np.exp(beta * shifted)
        w_sum = w.sum()
        if w_sum == 0.0:
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        # gradient = 2 * (a ⋆ w_rev) restricted to n components
        w_rev = w[::-1]
        grad_full = _fft_convolve(a, w_rev)          # length 3n‑2
        grad = 2.0 * grad_full[2 * n - 2 - np.arange(n)]

        eta = lr0 / math.sqrt(t)
        a *= np.exp(-eta * grad)
        a_sum = a.sum()
        if a_sum <= 0.0:
            a = np.full(n, 1.0 / n, dtype=np.float64)
        else:
            a /= a_sum

        # increase smoothing
        beta = min(beta_target, beta * beta_inc)

        if t % eval_every == 0:
            cur = _objective(a)
            if cur < best_score - 1e-14:
                best_score = cur
                best_seq = a.copy()
                print(
                    f"[{time.time():.1f}] smooth descent it={t} β={beta:.2f} new best {best_score:.6f}"
                )
    return best_seq, best_score

# Greedy pairwise mass transfer ----------------------------------------
def _pairwise_swap(
    a0: np.ndarray,
    time_limit: float,
    best_seq: np.ndarray,
    best_score: float,
    max_trials: int | None = None,
) -> tuple[np.ndarray, float]:
    """
    Randomly move a fraction of mass from one index to another.
    Only strictly improving moves are kept.
    """
    a = a0.copy()
    n = a.shape[0]
    trials = 0
    while time.time() < time_limit:
        if max_trials is not None and trials >= max_trials:
            break
        trials += 1

        src = np.where(a > 1e-14)[0]
        if src.size == 0:
            break
        i = int(np.random.choice(src))
        j = np.random.randint(n - 1)
        if j >= i:
            j += 1

        delta = float(a[i] * np.random.rand() * 0.45)   # up to 45 % of source mass
        if delta < 1e-15:
            continue

        new_a = a.copy()
        new_a[i] -= delta
        new_a[j] += delta

        cur = _objective(new_a)
        if cur < best_score - 1e-14:
            a = new_a
            best_score = cur
            best_seq = a.copy()
            print(
                f"[{time.time():.1f}] swap i={i}→j={j} δ={delta:.6g} new best {best_score:.6f}"
            )
    return best_seq, best_score

        # ---- Sparse random seeds ---------------------------------------------------
        for _ in range(4):
            if time.time() >= deadline:
                break
            a = _distribution_random_sparse(n, rng, density=0.008)
            sc = _objective(a)
            top_candidates.append((sc, a.copy()))
            if sc < overall_best_score:
                overall_best_score, overall_best_seq = sc, a.copy()
                print(f"[{time.time():.1f}] sparse len={n} score={sc:.6f}")

        # ---- Random convex mixtures of two families --------------------------------
        mix_fns = [
            _distribution_edge_weighted,
            _distribution_exponential_edge,
            _distribution_gaussian,
            _distribution_supergaussian,
            _distribution_kaiser,
            _distribution_tukey,
            _distribution_arcsine,
            _distribution_sine,
        ]
        for _ in range(6):
            if time.time() >= deadline:
                break
            fn1 = rng.choice(mix_fns)
            fn2 = rng.choice(mix_fns)
            # generate parameters for each
            if fn1 in (_distribution_edge_weighted,):
                p1 = float(rng.uniform(0.0, 3.0))
                base1 = fn1(n, p1)
            elif fn1 in (_distribution_exponential_edge,):
                c1 = float(rng.uniform(1e-3, 1.0))
                base1 = fn1(n, c1)
            elif fn1 in (_distribution_gaussian,):
                sigma1 = float(rng.uniform(0.05, 0.4) * n)
                base1 = fn1(n, sigma1)
            elif fn1 in (_distribution_supergaussian,):
                sigma1 = float(rng.uniform(0.05, 0.4) * n)
                pwr1 = float(rng.choice(power_grid))
                base1 = fn1(n, sigma1, pwr1)
            elif fn1 in (_distribution_kaiser,):
                beta1 = float(rng.uniform(0.0, 12.0))
                base1 = fn1(n, beta1)
            elif fn1 in (_distribution_tukey,):
                alp1 = float(rng.uniform(0.0, 1.0))
                base1 = fn1(n, alp1)
            else:  # arcsine or sine
                base1 = fn1(n)

            # second base
            if fn2 in (_distribution_edge_weighted,):
                p2 = float(rng.uniform(0.0, 3.0))
                base2 = fn2(n, p2)
            elif fn2 in (_distribution_exponential_edge,):
                c2 = float(rng.uniform(1e-3, 1.0))
                base2 = fn2(n, c2)
            elif fn2 in (_distribution_gaussian,):
                sigma2 = float(rng.uniform(0.05, 0.4) * n)
                base2 = fn2(n, sigma2)
            elif fn2 in (_distribution_supergaussian,):
                sigma2 = float(rng.uniform(0.05, 0.4) * n)
                pwr2 = float(rng.choice(power_grid))
                base2 = fn2(n, sigma2, pwr2)
            elif fn2 in (_distribution_kaiser,):
                beta2 = float(rng.uniform(0.0, 12.0))
                base2 = fn2(n, beta2)
            elif fn2 in (_distribution_tukey,):
                alp2 = float(rng.uniform(0.0, 1.0))
                base2 = fn2(n, alp2)
            else:
                base2 = fn2(n)

            lam = float(rng.random())
            cand = lam * base1 + (1.0 - lam) * base2
            cand /= cand.sum()
            sc = _objective(cand)
            top_candidates.append((sc, cand.copy()))
            if sc < overall_best_score:
                overall_best_score, overall_best_seq = sc, cand.copy()
                print(
                    f"[{time.time():.1f}] random mix len={n} lam={lam:.3f} score={sc:.6f}"
                )

        # keep only a limited number of best seeds
        top_candidates.sort(key=lambda x: x[0])
        top_candidates = top_candidates[:80]

    # ------------------------------------------------------------------
    # 2.  Include previous construction if it exists (height_sequence_1)
    # ------------------------------------------------------------------
    try:
        prev = np.asarray(height_sequence_1, dtype=np.float64)  # type: ignore[name-defined]
        if prev.ndim == 1 and prev.size > 0:
            prev = np.maximum(prev, 0.0)
            if prev.sum() == 0.0:
                prev = _distribution_uniform(prev.size)
            else:
                prev = prev / prev.sum()
            sc = _objective(prev)
            if sc < overall_best_score:
                overall_best_score, overall_best_seq = sc, prev.copy()
                print(f"[{time.time():.1f}] previous construction score={sc:.6f}")
            top_candidates.append((sc, prev.copy()))
    except Exception:  # pragma: no cover
        pass

    # ------------------------------------------------------------------
    # 3.  Fallback – should never happen
    # ------------------------------------------------------------------
    if overall_best_seq is None:
        n = 2000
        overall_best_seq = _distribution_uniform(n)
        overall_best_score = _objective(overall_best_seq)
        print(
            f"[{time.time():.1f}] fallback uniform length={n} score={overall_best_score:.6f}"
        )

    # ------------------------------------------------------------------
    # 4.  Refinement: smooth‑max descent + exact sub‑gradient
    # ------------------------------------------------------------------
    remaining = deadline - time.time()
    if remaining > 40.0 and top_candidates:
        # keep a few best seeds for deeper optimisation
        seeds = [seq for _, seq in sorted(top_candidates, key=lambda x: x[0])[:6]]
        per_seed = remaining / max(1, len(seeds))
        print(f"[{time.time():.1f}] start refinement on {len(seeds)} seeds")
        for idx, seed_seq in enumerate(seeds):
            if time.time() >= deadline:
                break
            # split the allotted time roughly half/half
            half = per_seed * 0.5
            # smooth‑max phase
            a_mid, sc_mid = _smooth_gradient_descent(
                seed_seq,
                time.time() + half,
                best_seq=overall_best_seq,
                best_score=overall_best_score,
                beta_start=0.3,
                beta_target=120.0,
                lr0=0.8,
                eval_every=150,
            )
            if sc_mid < overall_best_score - 1e-12:
                overall_best_score, overall_best_seq = sc_mid, a_mid.copy()
                print(
                    f"[{time.time():.1f}] after smooth seed {idx} new global best {overall_best_score:.6f}"
                )
            # exact sub‑gradient phase (use remaining half)
            if time.time() >= deadline:
                break
            a_opt, sc_opt = _subgradient_descent_exact(
                a_mid,
                time.time() + (per_seed - half),
                best_seq=overall_best_seq,
                best_score=overall_best_score,
                lr0=0.9,
                eval_every=150,
            )
            if sc_opt < overall_best_score - 1e-12:
                overall_best_score, overall_best_seq = sc_opt, a_opt.copy()
                print(
                    f"[{time.time():.1f}] after exact subgrad seed {idx} new global best {overall_best_score:.6f}"
                )

    # ------------------------------------------------------------------
    # 5.  Final greedy pairwise polishing (hill‑climbing)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        print(f"[{time.time():.1f}] final greedy swap polishing")
        a_opt, sc_opt = _pairwise_swap(
            overall_best_seq.copy(),
            deadline,
            best_seq=overall_best_seq,
            best_score=overall_best_score,
            max_trials=None,
        )
        if sc_opt < overall_best_score - 1e-12:
            overall_best_score, overall_best_seq = sc_opt, a_opt.copy()
            print(f"[{time.time():.1f}] after final swaps new best {overall_best_score:.6f}")

    # ------------------------------------------------------------------
    print(
        f"Final best objective (2·n·max(conv)): {overall_best_score:.6f} (length={len(overall_best_seq)})"
    )
    # Return a plain Python list – evaluate_sequence will internally re‑normalise.
    return overall_best_seq.tolist()
