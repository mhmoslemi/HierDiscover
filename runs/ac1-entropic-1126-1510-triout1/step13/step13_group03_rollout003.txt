def _optimize_edge_params(n, p0, lam0, max_iter=50):
    import scipy.optimize as opt
    def f(x):
        p, lam = x
        if p <= 0 or lam < 0 or lam > 1:
            return np.inf
        arr = _mix_with_uniform(_distribution_edge_weighted(n, p), lam)
        return _objective(arr)
    bounds = [(1e-6, 20.0), (0.0, 1.0)]
    res = opt.minimize(f, x0=np.array([p0, lam0]), method='L-BFGS-B', bounds=bounds, options={'maxiter': max_iter})

"""
Search for a non‑negative step‑function that minimizes ``evaluate_sequence``.

The algorithm proceeds in several stages:

1️⃣  Scan a variety of analytic families (uniform, arcsine, triangular,
    edge‑weighted, exponential‑edge, …) for several lengths.  The
    scan explores a richer parameter grid than before (including larger
    edge‑weights) and keeps the best few candidates.

2️⃣  For the best edge‑weighted / exponential‑edge candidates run a
    small continuous optimisation (L‑BFGS‑B) on the mixing parameter λ
    and on the shape parameters (p or c).  This refines the grid
    search without a costly full‑grid sweep.

3️⃣  Perform a short exact mirror‑descent (exponentiated‑gradient) on
    the most promising seeds, allocating the remaining time budget per
    seed.

4️⃣  Finish with a greedy pairwise‑mass‑transfer hill‑climber to polish
    the final solution.

All steps respect the global time limit (default 1000 s).  The returned
list is a plain Python list – ``evaluate_sequence`` will handle the
normalisation internally.
"""

import time

# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x ≥ 0, Σx = s}."""
    if s < 0:
        raise ValueError("Simplex radius must be non‑negative")
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.where(u - cssv / (np.arange(1, len(v) + 1)) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    return np.maximum(v - theta, 0.0)


# ----------------------------------------------------------------------
def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Full self‑convolution a*a using a real FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1) - 1).bit_length()
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[: 2 * n - 1]


# ----------------------------------------------------------------------
def _objective(a: np.ndarray) -> float:
    """Exact objective 2·n·max(conv(a,a)) for a vector with Σa = 1."""
    n = a.shape[0]
    b = _fft_self_convolve(a)
    return float(2.0 * n * b.max())

# Analytic families ----------------------------------------------------
def _distribution_uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _distribution_arcsine(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    vals /= vals.sum()
    return vals


def _distribution_sine(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64) + 0.5
    w = np.sin(np.pi * i / n)
    w /= w.sum()
    return w


def _distribution_cosine(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64) + 0.5
    w = 1.0 - np.cos(np.pi * i / n)
    w /= w.sum()
    return w


def _distribution_quadratic(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    x = (2.0 * i) / (n - 1) - 1.0
    w = 1.0 - x * x
    w = np.clip(w, 0.0, None)
    w /= w.sum()
    return w


def _distribution_triangular(n: int) -> np.ndarray:
    """Symmetric triangular – linearly increasing to the centre then decreasing."""
    i = np.arange(n, dtype=np.float64)
    half = (n - 1) / 2.0
    w = np.where(i <= half, i + 1, n - i)
    w = w.astype(np.float64)
    w /= w.sum()
    return w


def _distribution_edge_weighted(n: int, p: float) -> np.ndarray:
    """w_i ∝ ((i+0.5)*(n‑i‑0.5))^(−p).  Symmetric, edge‑heavy for larger p."""
    i = np.arange(n, dtype=np.float64) + 0.5
    j = (n - np.arange(n, dtype=np.float64) - 0.5)
    w = (i * j) ** (-p)
    w_sum = w.sum()
    if w_sum == 0.0:
        return np.full(n, 1.0 / n, dtype=np.float64)
    w /= w_sum
    return w


def _distribution_exponential_edge(n: int, c: float) -> np.ndarray:
    """w_i ∝ exp(−c·i) + exp(−c·(n‑1‑i))."""
    i = np.arange(n, dtype=np.float64)
    w = np.exp(-c * i) + np.exp(-c * (n - 1 - i))
    w /= w.sum()
    return w


def _mix_with_uniform(base: np.ndarray, lam: float) -> np.ndarray:
    """Convex combination λ·uniform + (1‑λ)·base."""
    n = base.shape[0]
    uniform = np.full(n, 1.0 / n, dtype=np.float64)
    return lam * uniform + (1.0 - lam) * base


def _distribution_random_sparse(
    n: int, rng: np.random.Generator, density: float = 0.01
) -> np.ndarray:
    """Sparse random vector with a prescribed density of non‑zeros."""
    k = max(2, int(n * density))
    idx = rng.choice(n, size=k, replace=False)
    vals = np.zeros(n, dtype=np.float64)
    vals[idx] = rng.random(k)
    s = vals.sum()
    if s == 0.0:
        vals[0] = 1.0
        s = 1.0
    vals /= s
    return vals

# Sub‑gradient / mirror descent (exact max‑index gradient) ----------
def _subgradient_descent_exact(
    a0: np.ndarray,
    time_limit: float,
    best_seq: np.ndarray,
    best_score: float,
    lr0: float = 0.9,
    eval_every: int = 100,
) -> tuple[np.ndarray, float]:
    """
    Exponentiated‑gradient descent on the exact L∞ objective.
    The sub‑gradient is built from *all* indices attaining the maximal
    convolution value.
    """
    a = a0.copy()
    n = a.shape[0]
    t = 0
    while time.time() < time_limit:
        t += 1
        conv = _fft_self_convolve(a)
        max_val = conv.max()
        max_idxs = np.where(conv == max_val)[0]

        # exact sub‑gradient
        grad = np.zeros_like(a)
        for idx in max_idxs:
            i_start = max(0, idx - (n - 1))
            i_end = min(n - 1, idx)
            i = np.arange(i_start, i_end + 1, dtype=np.int64)
            j = idx - i
            grad[i] += 2.0 * a[j]

        eta = lr0 / math.sqrt(t)
        a *= np.exp(-eta * grad)          # EG update, keeps positivity
        s = a.sum()
        if s <= 0.0:
            a = np.full(n, 1.0 / n, dtype=np.float64)
        else:
            a /= s

        if t % eval_every == 0:
            cur = 2.0 * n * max_val
            if cur < best_score - 1e-14:
                best_score = cur
                best_seq = a.copy()
                print(
                    f"[{time.time():.1f}] subgrad it={t} new global best {best_score:.6f}"
                )
    return best_seq, best_score

# Greedy pairwise mass transfer ----------------------------------------
def _pairwise_swap(
    a0: np.ndarray,
    time_limit: float,
    best_seq: np.ndarray,
    best_score: float,
    max_trials: int | None = None,
) -> tuple[np.ndarray, float]:
    """
    Randomly move a fraction of mass from one index to another.
    The move is accepted only if it strictly improves the exact objective.
    """
    a = a0.copy()
    n = a.shape[0]
    trials = 0
    while time.time() < time_limit:
        if max_trials is not None and trials >= max_trials:
            break
        trials += 1

        src_candidates = np.where(a > 1e-14)[0]
        if src_candidates.size == 0:
            break
        i = np.random.choice(src_candidates)
        j = np.random.randint(n - 1)
        if j >= i:
            j += 1

        delta = a[i] * np.random.rand() * 0.45
        if delta < 1e-15:
            continue

        new_a = a.copy()
        new_a[i] -= delta
        new_a[j] += delta

        cur = _objective(new_a)
        if cur < best_score - 1e-14:
            a = new_a
            best_score = cur
            best_seq = a.copy()
            print(
                f"[{time.time():.1f}] swap i={i}→j={j} δ={delta:.6g} new best {best_score:.6f}"
            )
    return best_seq, best_score

# Continuous parameter optimisation for edge / exponential families --
def _optimize_edge_params(
    n: int, p0: float, lam0: float, time_limit: float
) -> tuple[float, float, float]:
    """
    Refine (p, λ) for the mixed edge‑weighted family using L‑BFGS‑B.
    Returns (p_opt, lam_opt, score_opt).  Stops early if time runs out.
    """
    def fun(x):
        p, lam = x
        # enforce bounds manually – the optimiser respects them, but guard against NaNs
        if p <= 0 or lam < 0 or lam > 1:
            return np.inf
        cand = _mix_with_uniform(_distribution_edge_weighted(n, float(p)), float(lam))
        return _objective(cand)

    bounds = [(1e-6, 20.0), (0.0, 1.0)]
    x0 = np.array([p0, lam0])
    # limit total function evaluations to keep within the global time budget
    opts = {"maxiter": 80, "ftol": 1e-12}
    res = opt.minimize(fun, x0, method="L-BFGS-B", bounds=bounds, options=opts)
    if not res.success:
        # fallback to the starting point
        p_opt, lam_opt = p0, lam0
        score_opt = fun([p_opt, lam_opt])
    else:
        p_opt, lam_opt = res.x
        score_opt = res.fun
    # safety net against time overrun – the caller must check the deadline
    return float(p_opt), float(lam_opt), float(score_opt)


def _optimize_exp_params(
    n: int, c0: float, lam0: float, time_limit: float
) -> tuple[float, float, float]:
    """
    Refine (c, λ) for the mixed exponential‑edge family.
    Returns (c_opt, lam_opt, score_opt).
    """
    def fun(x):
        c, lam = x
        if c <= 0 or lam < 0 or lam > 1:
            return np.inf
        cand = _mix_with_uniform(_distribution_exponential_edge(n, float(c)), float(lam))
        return _objective(cand)

    bounds = [(1e-6, 5.0), (0.0, 1.0)]
    x0 = np.array([c0, lam0])
    opts = {"maxiter": 80, "ftol": 1e-12}
    res = opt.minimize(fun, x0, method="L-BFGS-B", bounds=bounds, options=opts)
    if not res.success:
        c_opt, lam_opt = c0, lam0
        score_opt = fun([c_opt, lam_opt])

    return float(c_opt), float(lam_opt), float(score_opt)

            if time.time() >= deadline:
                break
            sg_deadline = time.time() + per_seed
            a_opt, sc_opt = _subgradient_descent_exact(
                seed_seq,
                sg_deadline,
                best_seq=overall_best_seq,
                best_score=overall_best_score,
                lr0=0.9,
                eval_every=200,
            )
            if sc_opt < overall_best_score - 1e-12:
                overall_best_score, overall_best_seq = sc_opt, a_opt.copy()
                print(
                    f"[{time.time():.1f}] after subgrad seed {idx} new global best {overall_best_score:.6f}"
                )

    # ------------------------------------------------------------------
    # 5️⃣ Final greedy pairwise polishing
    # ------------------------------------------------------------------
    if time.time() < deadline:
        print(f"[{time.time():.1f}] final greedy polishing")
        a_opt, sc_opt = _pairwise_swap(
            overall_best_seq.copy(),
            deadline,
            best_seq=overall_best_seq,
            best_score=overall_best_score,
        )
        if sc_opt < overall_best_score - 1e-12:
            overall_best_score, overall_best_seq = sc_opt, a_opt.copy()
            print(f"[{time.time():.1f}] after final swaps new best {overall_best_score:.6f}")

    # ------------------------------------------------------------------
    print(
        f"Final best objective (2·n·max(conv)): {overall_best_score:.6f} (length={len(overall_best_seq)})"
    )
    # Return as plain Python list – ``evaluate_sequence`` will re‑normalise internally.
    return overall_best_seq.tolist()
