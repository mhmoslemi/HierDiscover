def _corr(a, w):
    # compute cross-correlation a ⋆ w for i=0..n-1 (i offset)
    # Use FFT: conj(fft(a)) * fft(w)
    size = 1 << ((len(a) + len(w) - 1).bit_length())
    Fa = np.fft.rfft(a, size)
    Fw = np.fft.rfft(w, size)
    corr = np.fft.irfft(np.conj(Fa) * Fw, size)
    # correlation result has length size; first (len(a)+len(w)-1) elements are valid
    # For i from 0..len(a)-1, correlation index i+len(w)-1
    start = len(w) - 1
    return corr[start:start+len(a)]

Now grad = 2 * corr.

"""
search for a non‑negative step function that minimises
evaluate_sequence (i.e. 2·n·max(convolution) / (sum)^2).

The algorithm works in three stages:

1️⃣  **Diverse seeding** – generate a couple of analytic families
    (uniform, arcsine, symmetric beta, edge‑weighted, sine‑weighted)
    and a few random Dirichlet vectors for a moderate length.
    The best few seeds (by the exact objective) are kept.

2️⃣  **Smoothed max‑gradient descent** – minimise the non‑smooth
    objective by a mirror‑descent on a *smoothed* max:
        smooth_maxα(x)= (1/α)·log Σ exp(α·x_i) .
    Its gradient is a simple correlation of the current distribution
    with the soft‑max weights, which we evaluate with an FFT.
    An exponentiated‑gradient update keeps the iterates on the simplex.
    Momentum and occasional random jitter accelerate convergence.

3️⃣  **Upsampling + fine local search** – if time permits the best
    distribution is linearly interpolated to a longer length (≈2×)
    and the same smoothed descent is run again.
    Finally a cheap pairwise mass‑transfer sweep and a short simulated
    annealing phase clean remaining spikes.

All operations are vectorised, use FFT‑based convolutions and respect the

def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x≥0, sum(x)=s}."""
    if s < 0:
        raise ValueError("simplex radius must be non‑negative")
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho_arr = np.where(u - cssv / (np.arange(1, len(v) + 1)) > 0)[0]
    if rho_arr.size == 0:
        theta = 0.0
    else:
        rho = rho_arr[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w

def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a (real FFT, length 2·n‑1)."""
    n = a.shape[0]
    need = 2 * n - 1
    size = 1 << ((need - 1).bit_length())
    A = rfft(a, n=size)
    conv = irfft(A * A, n=size)
    return conv[:need]


# ----------------------------------------------------------------------
def _corr(a: np.ndarray, w: np.ndarray) -> np.ndarray:
    """
    Linear cross‑correlation (a ⋆ w)[i] = Σ_j a_j · w_{i+j}
    for i = 0 … len(a)-1.
    Implemented with FFT; both inputs are real.
    """
    n = a.shape[0]
    m = w.shape[0]                # usually 2·n‑1
    need = n + m - 1
    size = 1 << ((need - 1).bit_length())
    Fa = rfft(a, n=size)
    Fw = rfft(w, n=size)
    corr_full = irfft(np.conj(Fa) * Fw, n=size)
    return corr_full[:n]          # only the non‑negative lags we need

def _objective(a: np.ndarray) -> float:
    """Exact objective 2·n·max(conv(a,a)) for a with sum(a)=1."""
    n = a.shape[0]
    b = _fft_self_convolve(a)
    return 2.0 * n * b.max()

def _distribution_uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _distribution_arcsine(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    vals /= vals.sum()
    return vals


def _distribution_beta_sym(n: int, p: float) -> np.ndarray:
    """Symmetric Beta(p,p) on the regular grid."""
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    w = np.exp((p - 1) * np.log(x) + (p - 1) * np.log(1.0 - x))
    if w.sum() == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _distribution_edge_weighted(n: int, p: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64) + 0.5
    j = (n - np.arange(n, dtype=np.float64) - 0.5)
    w = (i * j) ** (-p)
    if w.sum() == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _distribution_sine_weighted(n: int, p: float) -> np.ndarray:
    theta = np.pi * (np.arange(n, dtype=np.float64) + 0.5) / n
    w = np.sin(theta) ** p
    if w.sum() == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w

    alpha = rng.uniform(0.01, 2.0)
    a = rng.dirichlet(np.full(n, alpha))

    n = base.shape[0]
    uniform = np.full(n, 1.0 / n, dtype=np.float64)

def _smooth_descent(
    a: np.ndarray,
    deadline: float,
    best_seq: np.ndarray,
    best_score: float,
    *,
    alpha: float = 20.0,
    lr0: float = 0.8,
    beta: float = 0.9,
    eval_interval: int = 120,
    patience: int = 1800,
) -> tuple[np.ndarray, float]:
    """
    Mirror‑descent on the smoothed ``max`` of the self‑convolution.
    The update is an exponentiated gradient (entropy mirror) with
    optional momentum.  An active‑set of all entries is not needed:
    the smoothed gradient already concentrates on the large values.
    """
    n = a.shape[0]
    t = 0
    no_improve = 0
    momentum = np.zeros_like(a)
    rng = np.random.default_rng()

    while time.time() < deadline and t < 500_000:
        t += 1

        # ----- convolution and smoothed max weights -----
        conv = _fft_self_convolve(a)                # length 2·n‑1
        max_c = conv.max()
        # soft‑max weights; subtract max to avoid overflow
        w = np.exp(alpha * (conv - max_c))
        w_sum = w.sum()
        if w_sum == 0.0:
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        # ----- gradient of the smoothed max (correlation) -----
        grad = 2.0 * _corr(a, w)                    # shape (n,)

        # ----- momentum + exponentiated update -----
        momentum = beta * momentum + (1.0 - beta) * grad
        eta = lr0 / math.sqrt(t)
        a = a * np.exp(-eta * momentum)

        # ----- projection onto the simplex -----
        s = a.sum()
        if s == 0.0:
            a = np.full_like(a, 1.0 / n)
        else:
            a /= s

        # ----- periodic evaluation -----
        if t % eval_interval == 0:
            cur = _objective(a)
            if cur + 1e-14 < best_score:
                best_score, best_seq = cur, a.copy()
                no_improve = 0
                print(f"[{time.time():.1f}] it={t:,} new global best {best_score:.6f}")
            else:
                no_improve += 1

        # ----- escape plateaus -----
        if no_improve >= patience:
            jitter = rng.exponential(scale=0.004, size=n)
            a = a + jitter
            a = np.maximum(a, 0.0)
            a /= a.sum()
            no_improve = 0

    return best_seq, best_score

def _pairwise_swap_local(
    a: np.ndarray,
    deadline: float,
    best_seq: np.ndarray,
    best_score: float,
    max_trials: int | None = None,
) -> tuple[np.ndarray, float]:
    """
    Deterministic local search: move a random fraction of mass from one
    coordinate to another if the objective improves.
    """
    n = a.shape[0]
    trials = 0
    rng = np.random.default_rng()
    while time.time() < deadline:
        if max_trials is not None and trials >= max_trials:
            break
        trials += 1

        src = np.where(a > 1e-12)[0]
        if src.size == 0:
            break
        i = rng.choice(src)

        j = rng.integers(n - 1)
        if j >= i:
            j += 1

        delta = a[i] * rng.random() * 0.33
        if delta < 1e-14:
            continue

        new_a = a.copy()
        new_a[i] -= delta
        new_a[j] += delta

        cur = _objective(new_a)
        if cur + 1e-14 < best_score:
            a = new_a
            best_score = cur
            best_seq = a.copy()
            print(f"[{time.time():.1f}] swap {i}->{j} δ={delta:.3e} new best {best_score:.6f}")

    return best_seq, best_score

def _random_local_search(
    a: np.ndarray,
    deadline: float,
    best_seq: np.ndarray,
    best_score: float,
    init_temp: float = 0.008,
    decay: float = 0.9995,
) -> tuple[np.ndarray, float]:
    """Metropolis‑style simulated annealing on the exact objective."""
    n = a.shape[0]
    rng = np.random.default_rng()
    T = init_temp
    while time.time() < deadline:
        i = rng.integers(n)
        j = rng.integers(n - 1)
        if j >= i:
            j += 1
        delta = a[i] * rng.random() * 0.28
        if delta < 1e-14:
            continue
        new_a = a.copy()
        new_a[i] -= delta
        new_a[j] += delta
        cur = _objective(new_a)
        if cur + 1e-14 < best_score:
            a = new_a
            best_score = cur
            best_seq = a.copy()
            print(f"[{time.time():.1f}] SA improve {i}->{j} Δ={delta:.3e} best {best_score:.6f}")
        else:
            if rng.random() < math.exp(-(cur - best_score) / max(T, 1e-12)):
                a = new_a
        T *= decay
    return best_seq, best_score

def _upsample(seq: np.ndarray, new_len: int) -> np.ndarray:
    """Linear interpolation of a simplex‑vector to a larger length."""
    old_len = seq.shape[0]
    if new_len == old_len:
        return seq.copy()
    x_old = np.linspace(0.0, 1.0, old_len)
    x_new = np.linspace(0.0, 1.0, new_len)
    new_seq = np.interp(x_new, x_old, seq)
    new_seq = np.maximum(new_seq, 0.0)
    new_seq /= new_seq.sum()
    return new_seq

        print(f"[{time.time():.1f}] generating seeds for length {n}")

        # deterministic families
        for base in (_distribution_uniform(n), _distribution_arcsine(n)):
            sc = _objective(base)
            candidate_pool.append((sc, base.copy()))
            if sc < overall_best_score:
                overall_best_score, overall_best_seq = sc, base.copy()
                print(f"[{time.time():.1f}] new global best (basic) {sc:.6f}")

        # parameterised families
        for p in p_grid:
            if time.time() >= deadline:
                break
            beta = _distribution_beta_sym(n, p)
            edge = _distribution_edge_weighted(n, p)
            sine = _distribution_sine_weighted(n, p)

            for base in (beta, edge, sine):
                for lam in lam_grid:
                    if time.time() >= deadline:
                        break
                    cand = _mix_with_uniform(base, lam)
                    sc = _objective(cand)
                    candidate_pool.append((sc, cand.copy()))
                    if sc < overall_best_score:
                        overall_best_score, overall_best_seq = sc, cand.copy()
                        print(
                            f"[{time.time():.1f}] new global best len={n} p={p:.3g} lam={lam:.2f} score={sc:.6f}"
                        )

        # random Dirichlet seeds
        for _ in range(4):
            if time.time() >= deadline:
                break
            cand = _distribution_dirichlet(n, rng)
            sc = _objective(cand)
            candidate_pool.append((sc, cand.copy()))
            if sc < overall_best_score:
                overall_best_score, overall_best_seq = sc, cand.copy()
                print(f"[{time.time():.1f}] new global best Dirichlet len={n} score={sc:.6f}")

    # keep only the most promising seeds
    candidate_pool.sort(key=lambda x: x[0])
    top_seeds = [seq for _, seq in candidate_pool[:7]]

    # ------------------------------------------------------------------
    # 2️⃣  Smoothed‑max mirror descent on the selected seeds
    remaining = deadline - time.time()
    if remaining > 30.0 and top_seeds:
        per_seed = remaining * 0.55 / max(1, len(top_seeds))
        print(f"[{time.time():.1f}] descent on {len(top_seeds)} seeds, ~{per_seed:.1f}s each")
        for idx, seed_seq in enumerate(top_seeds):
            if time.time() >= deadline:
                break
            sub_dead = time.time() + per_seed
            a_opt, sc_opt = _smooth_descent(
                seed_seq.copy(),
                sub_dead,
                best_seq=overall_best_seq,
                best_score=overall_best_score,
                alpha=22.0,
                lr0=0.9,
                beta=0.88,
                eval_interval=100,
                patience=1500,
            )
            if sc_opt < overall_best_score:
                overall_best_score, overall_best_seq = sc_opt, a_opt.copy()
                print(
                    f"[{time.time():.1f}] after descent seed {idx} new global best {overall_best_score:.6f}"
                )

    # ------------------------------------------------------------------
    # 3️⃣  Upsample if we still have ample time
    remaining = deadline - time.time()
    if remaining > 120.0 and overall_best_seq is not None:
        large_n = min(18000, len(overall_best_seq) * 2)
        print(f"[{time.time():.1f}] up‑sampling to {large_n} points")
        up_seq = _upsample(overall_best_seq, large_n)

        sub_dead = time.time() + remaining * 0.65
        up_opt, up_score = _smooth_descent(
            up_seq,
            sub_dead,
            best_seq=overall_best_seq,
            best_score=overall_best_score,
            alpha=25.0,
            lr0=0.6,
            beta=0.92,
            eval_interval=80,
            patience=2000,
        )
        if up_score < overall_best_score:
            overall_best_score, overall_best_seq = up_score, up_opt.copy()
            print(
                f"[{time.time():.1f}] up‑sample descent yielded new best {overall_best_score:.6f}"
            )

    # ------------------------------------------------------------------
    # 4️⃣  Fine deterministic/combinatorial cleanup
    if time.time() < deadline:
        print(f"[{time.time():.1f}] deterministic pairwise refinement")
        overall_best_seq, overall_best_score = _pairwise_swap_local(
            overall_best_seq.copy(),
            deadline,
            best_seq=overall_best_seq,
            best_score=overall_best_score,
            max_trials=300_000,
        )

    if time.time() < deadline:
        print(f"[{time.time():.1f}] final simulated‑annealing clean‑up")
        overall_best_seq, overall_best_score = _random_local_search(
            overall_best_seq.copy(),
            deadline,
            best_seq=overall_best_seq,
            best_score=overall_best_score,
            init_temp=0.008,
            decay=0.9995,
        )

    # ------------------------------------------------------------------
    print(
        f"Final best score: {overall_best_score:.6f} (length={len(overall_best_seq)})"
    )
    # Convert to plain Python list – let evaluate_sequence perform its safety clipping
    return overall_best_seq.tolist()
