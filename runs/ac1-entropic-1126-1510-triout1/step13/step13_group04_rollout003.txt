"""
Propose a non‑negative step function that (almost) minimizes ``evaluate_sequence``.

Algorithm outline
-----------------
1. **Diverse seed generation** – Build a pool of probability vectors from a
   large collection of analytic families (uniform, arcsine, beta, edge‑product,
   edge‑sum, two‑block, edge‑central block, generalized‑normal, power‑law,
   Dirichlet, sparse).  The best few (≈50) are kept.

2. **Exponentiated‑gradient refinement** – For the top seeds run a mirror‑descent
   (multiplicative) update on the exact objective using the sub‑gradient of the
   active convolution entry.  This quickly drives the candidates toward a local
   optimum.

3. **Mass‑transfer descent** – A greedy pairwise move that shifts a fraction of
   mass from the coordinate with the largest gradient to the one with the
   smallest gradient.  This directly reduces the maximal convolution entry.

4. **Upsampling & fine‑tuning** – The best sequence is interpolated to a larger
   length (≈2×) and refined again with a short EG burst followed by another
   mass‑transfer phase.

5. **Final annealing** – A simple simulated‑annealing style pairwise transfer
   runs until the time budget expires, polishing the solution.

All steps respect the 1000 s budget (2 s safety margin) and maintain the
simplex constraint Σa=1, a_i≥0.  The routine returns a plain ``list[float]``.
"""

# ----------------------------------------------------------------------
# Simplex projection (euclidean)
# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Project vector v onto the simplex {x≥0, Σx=s}."""
    if s < 0:
        raise ValueError("simplex radius must be non‑negative")
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.where(u - cssv / np.arange(1, len(v) + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# FFT based full linear self‑convolution
# ----------------------------------------------------------------------
def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Return the full linear convolution a * a using a real FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1) - 1).bit_length()  # next power of two
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[: 2 * n - 1]


# ----------------------------------------------------------------------
# Exact objective (identical to ``evaluate_sequence`` after normalisation)
# ----------------------------------------------------------------------
def _objective(a: np.ndarray) -> float:
    """Compute the objective for a (non‑normalised) probability vector."""
    n = a.shape[0]
    b = _fft_self_convolve(a)
    max_b = b.max()
    sum_a = a.sum()
    if sum_a < 0.01:
        return float("inf")
    return float(2.0 * n * max_b / (sum_a * sum_a))


# ----------------------------------------------------------------------
# Sub‑gradient of the currently maximal convolution entry
# ----------------------------------------------------------------------
def _grad_of_max_conv(a: np.ndarray) -> tuple[np.ndarray, int]:
    """Return (grad, active_index) where grad is the sub‑gradient of max(b)."""
    n = a.shape[0]
    conv = _fft_self_convolve(a)
    k = int(np.argmax(conv))

    grad = np.zeros_like(a)
    if k < n:
        # overlapping region length = k+1
        grad[: k + 1] = 2.0 * a[k::-1]
    else:
        # region length = 2·n-1-k
        start = k - (n - 1)
        grad[start:] = 2.0 * a[start:][::-1]
    return grad, k


# ----------------------------------------------------------------------
# Exponentiated‑gradient descent (multiplicative mirror descent)
# ----------------------------------------------------------------------
def _exp_gradient_descent(
    a: np.ndarray,
    deadline: float,
    best_seq: np.ndarray,
    best_score: float,
    eta0: float = 1.0,
    eval_every: int = 40,
) -> tuple[np.ndarray, float]:
    """Run EG descent on the exact objective until the deadline."""
    n = a.shape[0]
    t = 0
    while time.time() < deadline:
        t += 1
        grad, _ = _grad_of_max_conv(a)
        eta = eta0 / math.sqrt(t)
        a = a * np.exp(-eta * grad)
        s = a.sum()
        if s == 0.0:
            a = np.full(n, 1.0 / n, dtype=np.float64)
        else:
            a = a / s
        if t % eval_every == 0:
            cur = _objective(a)
            if cur < best_score - 1e-14:
                best_score = cur
                best_seq = a.copy()
                print(f"[{time.time():.1f}] EG it={t} new best {best_score:.6f}")
    return best_seq, best_score


# ----------------------------------------------------------------------
# Greedy mass‑transfer descent
# ----------------------------------------------------------------------
def _mass_transfer_descent(
    a: np.ndarray,
    deadline: float,
    best_seq: np.ndarray,
    best_score: float,
    step_factor: float = 0.25,
    eval_every: int = 200,
) -> tuple[np.ndarray, float]:
    """Move mass from large‑gradient coordinates to small‑gradient ones."""
    n = a.shape[0]
    it = 0
    while time.time() < deadline:
        it += 1
        grad, _ = _grad_of_max_conv(a)
        i_max = int(np.argmax(grad))
        i_min = int(np.argmin(grad))
        if grad[i_max] - grad[i_min] < 1e-12:
            break
        delta = a[i_max] * step_factor
        if delta <= 0.0:
            break
        new_a = a.copy()
        new_a[i_max] -= delta
        new_a[i_min] += delta
        cur = _objective(new_a)
        if cur < best_score - 1e-14:
            a = new_a
            best_score = cur
            best_seq = a.copy()
            print(f"[{time.time():.1f}] MT it={it} new best {best_score:.6f}")
        if it % eval_every == 0:
            # occasional re‑evaluation (already done above)
            pass
    return best_seq, best_score


# ----------------------------------------------------------------------
# Simple simulated‑annealing pairwise transfer (final polish)
# ----------------------------------------------------------------------
def _pairwise_annealing(
    a: np.ndarray,
    deadline: float,
    best_seq: np.ndarray,
    best_score: float,
    rng: np.random.Generator,
) -> tuple[np.ndarray, float]:
    """Random pairwise moves with a temperature schedule."""
    n = a.shape[0]
    start = time.time()
    total = deadline - start
    while time.time() < deadline:
        src = np.where(a > 1e-12)[0]
        if src.size == 0:
            break
        i = rng.choice(src)
        j = rng.integers(n - 1)
        if j >= i:
            j += 1
        delta = a[i] * rng.random() * 0.4
        if delta < 1e-16:
            continue
        new_a = a.copy()
        new_a[i] -= delta
        new_a[j] += delta
        cur = _objective(new_a)
        elapsed = time.time() - start
        temperature = max(0.0, 1.0 - elapsed / total) * best_score * 0.01
        accept = cur < best_score - 1e-14 or (temperature > 0.0 and rng.random() < math.exp(-(cur - best_score) / temperature))
        if accept:
            a = new_a
            if cur < best_score - 1e-14:
                best_score = cur
                best_seq = a.copy()
                print(f"[{time.time():.1f}] anneal i={i}->j={j} δ={delta:.3e} new best {best_score:.6f}")
    return best_seq, best_score


# ----------------------------------------------------------------------
# Linear up‑sampling while preserving the simplex
# ----------------------------------------------------------------------
def _upsample(a: np.ndarray, new_n: int) -> np.ndarray:
    """Interpolate a probability vector to a larger length."""
    old_n = a.shape[0]
    if new_n == old_n:
        return a.copy()
    x_old = (np.arange(old_n, dtype=np.float64) + 0.5) / old_n
    x_new = (np.arange(new_n, dtype=np.float64) + 0.5) / new_n
    a_new = np.interp(x_new, x_old, a, left=0.0, right=0.0)
    a_new = np.maximum(a_new, 0.0)
    s = a_new.sum()
    if s == 0.0:
        a_new[:] = 1.0 / new_n
    else:
        a_new /= s
    return a_new


# ----------------------------------------------------------------------
# Analytic families (all return a probability vector, sum = 1)
# ----------------------------------------------------------------------
def _distribution_uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _distribution_arcsine(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    vals /= vals.sum()
    return vals


def _distribution_beta_sym(n: int, p: float) -> np.ndarray:
    """Symmetric Beta(p,p) on a uniform grid."""
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    w = np.exp((p - 1) * np.log(x) + (p - 1) * np.log(1.0 - x))
    if w.sum() == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _distribution_edge_product(n: int, p: float) -> np.ndarray:
    """Weight ∝ (i·j)^{-p} where i,j are distances to the two edges."""
    i = np.arange(n, dtype=np.float64) + 0.5
    j = (n - np.arange(n, dtype=np.float64) - 0.5)
    w = (i * j) ** (-p)
    if w.sum() == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _distribution_edge_sum(n: int, p: float) -> np.ndarray:
    """Weight ∝ (i)^{-p} + (j)^{-p}."""
    i = np.arange(n, dtype=np.float64) + 0.5
    j = (n - np.arange(n, dtype=np.float64) - 0.5)
    w = (i ** (-p)) + (j ** (-p))
    if w.sum() == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _distribution_hann(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    w = 0.5 - 0.5 * np.cos(2.0 * np.pi * i / (n - 1))
    if w.sum() == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _distribution_generalized_normal(n: int, q: float) -> np.ndarray:
    """Density ∝ exp(-|x-1/2|^{q}) on [0,1]."""
    i = np.arange(n, dtype=np.float64) + 0.5
    x = i / n - 0.5
    w = np.exp(- (np.abs(x) / 0.5) ** q)
    if w.sum() == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _distribution_power_law(n: int, p: float) -> np.ndarray:
    """Monotone decreasing power‑law (p>0)."""
    i = np.arange(1, n + 1, dtype=np.float64)
    w = i ** (-p)
    w /= w.sum()
    return w


def _distribution_random_sparse(n: int, rng: np.random.Generator, density: float = 0.01) -> np.ndarray:
    """Sparse random vector with a given density."""
    k = max(2, int(n * density))
    idx = rng.choice(n, size=k, replace=False)
    vals = np.zeros(n, dtype=np.float64)
    vals[idx] = rng.random(k)
    if vals.sum() == 0.0:
        vals[0] = 1.0
    vals /= vals.sum()
    return vals


def _distribution_two_blocks(n: int, block_frac: float) -> np.ndarray:
    """Two symmetric edge blocks, equal mass (each block gets half total mass)."""
    block_len = max(1, int(round(block_frac * n)))
    a = np.zeros(n, dtype=np.float64)
    mass_per_elem = 0.5 / block_len
    a[:block_len] = mass_per_elem
    a[-block_len:] = mass_per_elem
    return a


def _distribution_edge_central_block(n: int, edge_frac: float, ratio: float) -> np.ndarray:
    """Edges and a middle part. Edge weight per element = ratio * center weight."""
    edge_len = max(1, int(round(edge_frac * n / 2.0)))
    central_len = n - 2 * edge_len
    if central_len <= 0:
        # fallback to two‑block
        return _distribution_two_blocks(n, edge_frac)
    # Let w_c be centre weight per element, w_e = ratio * w_c
    w_c = 1.0 / (2 * edge_len * ratio + central_len)
    w_e = ratio * w_c
    a = np.full(n, w_c, dtype=np.float64)
    a[:edge_len] = w_e
    a[-edge_len:] = w_e
    a /= a.sum()
    return a


def _mix_with_uniform(base: np.ndarray, lam: float) -> np.ndarray:
    """Convex combination with the uniform distribution."""
    n = base.shape[0]
    uniform = np.full(n, 1.0 / n, dtype=np.float64)
    return lam * uniform + (1.0 - lam) * base

        print(f"[{time.time():.1f}] Fallback uniform score={overall_best_score:.6f}")

    # ------------------------------------------------------------------
    # 4. Exponentiated‑gradient descent on top seeds
    # ------------------------------------------------------------------
    remaining = deadline - time.time()
    if remaining > 30.0 and top_candidates:
        per_seed_time = remaining * 0.30 / max(1, len(top_candidates))
        print(f"[{time.time():.1f}] EG descent on {len(top_candidates)} seeds (≈{per_seed_time:.1f}s each)")
        for idx, (_, seed_seq) in enumerate(top_candidates):
            if time.time() >= deadline:
                break
            sub_deadline = time.time() + per_seed_time
            a_opt, sc_opt = _exp_gradient_descent(
                seed_seq.copy(),
                sub_deadline,
                best_seq=overall_best_seq,
                best_score=overall_best_score,
                eta0=1.0,
                eval_every=40,
            )
            if sc_opt < overall_best_score - 1e-12:
                overall_best_score = sc_opt
                overall_best_seq = a_opt.copy()
                print(f"[{time.time():.1f}] EG seed {idx} improved global best {overall_best_score:.6f}")

    # ------------------------------------------------------------------
    # 5. Greedy mass‑transfer refinement on the current best
    # ------------------------------------------------------------------
    if time.time() < deadline:
        mt_deadline = time.time() + (deadline - time.time()) * 0.20
        print(f"[{time.time():.1f}] Mass‑transfer refinement (budget {mt_deadline - time.time():.1f}s)")
        overall_best_seq, overall_best_score = _mass_transfer_descent(
            overall_best_seq.copy(),
            mt_deadline,
            best_seq=overall_best_seq,
            best_score=overall_best_score,
            step_factor=0.25,
            eval_every=250,
        )
        print(f"[{time.time():.1f}] After MT best score {overall_best_score:.6f}")

    # ------------------------------------------------------------------
    # 6. Upsample to larger length and fine‑tune
    # ------------------------------------------------------------------
    if time.time() < deadline:
        cur_n = overall_best_seq.shape[0]
        target_n = cur_n * 2
        if target_n <= 64000:
            print(f"[{time.time():.1f}] Upsampling {cur_n} → {target_n}")
            up_seq = _upsample(overall_best_seq, target_n)
            up_score = _objective(up_seq)
            if up_score < overall_best_score:
                overall_best_score = up_score
                overall_best_seq = up_seq
                print(f"[{time.time():.1f}] Upsampled new best {overall_best_score:.6f}")

            # short EG on upsampled vector
            sub_deadline = time.time() + (deadline - time.time()) * 0.12
            a_opt, sc_opt = _exp_gradient_descent(
                overall_best_seq.copy(),
                sub_deadline,
                best_seq=overall_best_seq,
                best_score=overall_best_score,
                eta0=0.8,
                eval_every=30,
            )
            if sc_opt < overall_best_score - 1e-12:
                overall_best_score = sc_opt
                overall_best_seq = a_opt.copy()
                print(f"[{time.time():.1f}] EG after upsample new best {overall_best_score:.6f}")

            # another mass‑transfer phase
            sub_deadline = time.time() + (deadline - time.time()) * 0.12
            overall_best_seq, overall_best_score = _mass_transfer_descent(
                overall_best_seq.copy(),
                sub_deadline,
                best_seq=overall_best_seq,
                best_score=overall_best_score,
                step_factor=0.2,
                eval_every=200,
            )
            print(f"[{time.time():.1f}] Post‑upsample MT best {overall_best_score:.6f}")

    # ------------------------------------------------------------------
    # 7. Final annealing polish
    # ------------------------------------------------------------------
    if time.time() < deadline:
        print(f"[{time.time():.1f}] Final annealing polish")
        overall_best_seq, overall_best_score = _pairwise_annealing(
            overall_best_seq.copy(),
            deadline,
            best_seq=overall_best_seq,
            best_score=overall_best_score,
            rng=rng,
        )
        print(f"[{time.time():.1f}] After annealing best {overall_best_score:.6f}")

    # ------------------------------------------------------------------
    print(
        f"Final best score (objective=2·n·max(conv)/sum²): {overall_best_score:.6f} "
        f"(length={len(overall_best_seq)})"
    )
    return overall_best_seq.tolist()
