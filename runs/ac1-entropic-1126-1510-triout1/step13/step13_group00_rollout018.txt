"""
Propose a non‑negative step function (list of heights) that minimises
``evaluate_sequence``.

Algorithm
---------

* **Diverse seeding** – sample many analytic families (uniform, arcsine,
  symmetric / asymmetric Beta, edge‑weighted, edge‑inverse‑power,
  sine‑weighted, power‑law, exponential, triangular, Gaussian) together
  with Dirichlet and sparse random vectors for several moderate lengths.
  Keep the ~30 best (lowest exact objective 2·n·max(conv)).

* **Heavy mirror‑descent** – run a long exponentiated‑gradient (mirror)
  descent on each seed.  The active set now contains all convolution
  entries whose value lies within ``active_frac`` of the current maximum
  (or within an absolute epsilon).  Momentum (Adam‑style) and a
  decreasing learning rate ``η≈1/√t`` are used.  The routine runs as long
  as time permits.

* **Progressive up‑sampling** – repeatedly double the current best
  sequence (linear interpolation) up to a user‑defined maximum length
  (≈ 30 k).  After each up‑sampling a short mirror‑descent refinement is
  performed.  Larger length yields a strictly smaller theoretical lower
  bound, so this often yields a noticeable improvement.

* **Fine‑grained deterministic refinement** – a greedy pairwise mass‑
  transfer sweep (up to the time limit) removes any remaining spikes.

* **Cheap simulated‑annealing clean‑up** – final Metropolis‑style local
  search to polish the solution.

All vectors are projected onto the simplex after each update,
guaranteeing non‑negativity and a fixed sum (the scaling does not affect
the objective).  The implementation uses only NumPy (FFT for fast
convolutions) and respects the 1000 s time budget.
"""

# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x≥0, sum(x)=s}."""
    if s < 0:
        raise ValueError("simplex radius must be non‑negative")
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho_arr = np.where(u - cssv / (np.arange(1, len(v) + 1)) > 0)[0]
    if rho_arr.size == 0:
        theta = 0.0
    else:
        rho = rho_arr[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a (real FFT, length 2·n‑1)."""
    n = a.shape[0]
    need = 2 * n - 1
    size = 1 << ((need - 1).bit_length())   # next power of two
    A = rfft(a, n=size)
    conv = irfft(A * A, n=size)
    return conv[:need]


# ----------------------------------------------------------------------
def _objective(a: np.ndarray) -> float:
    """Exact objective 2·n·max(conv(a,a)) assuming sum(a)=1."""
    n = a.shape[0]
    b = _fft_self_convolve(a)
    return 2.0 * n * b.max()

# Analytic families ----------------------------------------------------
def _distribution_uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _distribution_arcsine(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    vals /= vals.sum()
    return vals


def _distribution_beta_sym(n: int, p: float) -> np.ndarray:
    """Symmetric Beta(p,p)."""
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    w = np.exp((p - 1) * np.log(x) + (p - 1) * np.log(1.0 - x))
    w_sum = w.sum()
    if w_sum == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w_sum
    return w


def _distribution_beta_asym(n: int, a: float, b: float) -> np.ndarray:
    """Asymmetric Beta(a,b)."""
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    logw = (a - 1) * np.log(x) + (b - 1) * np.log(1.0 - x)
    # shift to avoid overflow
    w = np.exp(logw - np.max(logw))
    w_sum = w.sum()
    if w_sum == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w_sum
    return w


def _distribution_edge_weighted(n: int, p: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64) + 0.5
    j = (n - np.arange(n, dtype=np.float64) - 0.5)
    w = (i * j) ** (-p)
    w_sum = w.sum()
    if w_sum == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w_sum
    return w


def _distribution_edge_inversepow(n: int, p: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64) + 0.5
    j = (n - np.arange(n, dtype=np.float64) - 0.5)
    w = i ** (-p) + j ** (-p)
    w_sum = w.sum()
    if w_sum == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w_sum
    return w


def _distribution_sine_weighted(n: int, p: float) -> np.ndarray:
    theta = np.pi * (np.arange(n, dtype=np.float64) + 0.5) / n
    w = np.sin(theta) ** p
    w_sum = w.sum()
    if w_sum == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w_sum
    return w


def _distribution_power_law(n: int, p: float) -> np.ndarray:
    i = np.arange(1, n + 1, dtype=np.float64)
    w = i ** (-p)
    w_sum = w.sum()
    if w_sum == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w_sum
    return w


def _distribution_exponential(n: int, lam: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    w = np.exp(-lam * i)
    w_sum = w.sum()
    if w_sum == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w_sum
    return w


def _distribution_triangle(n: int) -> np.ndarray:
    """Increasing linear shape (0 … n‑1) normalised."""
    w = np.arange(1, n + 1, dtype=np.float64)
    w /= w.sum()
    return w


def _distribution_inverted_triangle(n: int) -> np.ndarray:
    w = np.arange(n, 0, -1, dtype=np.float64)
    w /= w.sum()
    return w


def _distribution_gaussian(n: int, sigma: float) -> np.ndarray:
    """Discrete Gaussian centred at the middle."""
    xs = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    w = np.exp(-0.5 * ((xs - centre) / sigma) ** 2)
    w_sum = w.sum()
    if w_sum == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w_sum
    return w


def _distribution_random_sparse(
    n: int, rng: np.random.Generator, density: float = 0.006
) -> np.ndarray:
    k = max(2, int(n * density))
    idx = rng.choice(n, size=k, replace=False)
    vals = np.zeros(n, dtype=np.float64)
    vals[idx] = rng.random(k)
    if vals.sum() == 0.0:
        vals[0] = 1.0
    vals /= vals.sum()
    return vals


def _distribution_dirichlet(n: int, rng: np.random.Generator) -> np.ndarray:
    alpha = rng.uniform(0.01, 2.0)
    a = rng.dirichlet(np.full(n, alpha))
    return a.astype(np.float64)


def _mix_with_uniform(base: np.ndarray, lam: float) -> np.ndarray:
    """lam·U + (1‑lam)·base, where U is the uniform distribution."""
    n = base.shape[0]
    uniform = np.full(n, 1.0 / n, dtype=np.float64)
    return lam * uniform + (1.0 - lam) * base

# Momentum mirror descent (Adam‑like) -----------------------------------
def _subgradient_descent_momentum(
    a: np.ndarray,
    deadline: float,
    best_seq: np.ndarray,
    best_score: float,
    lr0: float = 0.7,
    active_frac: float = 0.07,

    beta1: float = 0.9,
    beta2: float = 0.999,

) -> tuple[np.ndarray, float]:

    The active set consists of all convolution entries whose value lies
    within ``active_frac`` of the current maximum (or within ``eps`` of it).
    """
    n = a.shape[0]
    t = 0
    no_improve = 0
    m = np.zeros_like(a)          # 1st moment
    v = np.zeros_like(a)          # 2nd moment
    rng = np.random.default_rng()
    ind_grid = np.arange(n, dtype=np.int32)

        t += 1
        conv = _fft_self_convolve(a)
        max_c = conv.max()
        # absolute tolerance grows with max_c, ensures we capture the
        # true “active” constraints even when max_c is tiny.
        thresh = max(max_c * (1.0 - active_frac), max_c - eps)

        act = np.where(conv >= thresh)[0]
        if act.size > max_active:
            # keep the most critical ones only
            top = np.argpartition(conv[act], -max_active)[-max_active:]
            act = act[top]

        # sub‑gradient of the max‑convolution (sum of contributions)
        grad = np.zeros_like(a)
        for k in act:
            shift = k - ind_grid
            mask = (shift >= 0) & (shift < n)
            grad[mask] += 2.0 * a[shift[mask]]

        # Adam updates on the *log* of a (exponentiated gradient)
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)

        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)

        eta = lr0 / math.sqrt(t)
        # log‑space update (mirrored on the simplex)
        a = a * np.exp(-eta * m_hat / (np.sqrt(v_hat) + eps))

        # simplex projection (guarantees sum(a)=1)
        s = a.sum()
        if s == 0.0:
            a = np.full(n, 1.0 / n, dtype=np.float64)
        else:
            a /= s

        if t % eval_interval == 0:
            cur = _objective(a)
            if cur + 1e-14 < best_score:
                best_score, best_seq = cur, a.copy()
                no_improve = 0
                print(f"[{time.time():.1f}] it={t:,} new global best {best_score:.6f}")
            else:
                no_improve += 1

        if no_improve >= patience:
            # shake the basin with a tiny random jitter
            jitter = rng.exponential(scale=0.002, size=n)
            a = a + jitter
            a = np.maximum(a, 0.0)
            a /= a.sum()
            no_improve = 0

    return best_seq, best_score


# ----------------------------------------------------------------------
# Pairwise deterministic refinement --------------------------------------
def _pairwise_swap_local(
    a: np.ndarray,
    deadline: float,
    best_seq: np.ndarray,
    best_score: float,
    max_trials: int | None = None,
) -> tuple[np.ndarray, float]:
    """
    Transfer a random fraction of mass from one coordinate to another.
    Keep the change only if the objective improves.
    """
    n = a.shape[0]
    trials = 0
    rng = np.random.default_rng()
    while time.time() < deadline:
        if max_trials is not None and trials >= max_trials:
            break
        trials += 1

        src = np.where(a > 1e-12)[0]
        if src.size == 0:
            break
        i = rng.choice(src)

        # pick a distinct destination
        j = rng.integers(n - 1)
        if j >= i:
            j += 1

        delta = a[i] * rng.random() * 0.33
        if delta < 1e-14:
            continue

        new_a = a.copy()
        new_a[i] -= delta
        new_a[j] += delta

        cur = _objective(new_a)
        if cur + 1e-14 < best_score:
            a = new_a
            best_score = cur
            best_seq = a.copy()
            print(
                f"[{time.time():.1f}] swap {i}->{j} δ={delta:.3e} new best {best_score:.6f}"
            )
    return best_seq, best_score


# ----------------------------------------------------------------------
# Cheap simulated annealing ---------------------------------------------
def _random_local_search(
    a: np.ndarray,
    deadline: float,
    best_seq: np.ndarray,
    best_score: float,
    init_temp: float = 0.008,
    decay: float = 0.9995,
) -> tuple[np.ndarray, float]:
    """Transfer mass between random positions; Metropolis acceptance."""
    n = a.shape[0]
    rng = np.random.default_rng()
    T = init_temp
    while time.time() < deadline:
        i = rng.integers(n)
        j = rng.integers(n - 1)
        if j >= i:
            j += 1
        delta = a[i] * rng.random() * 0.28
        if delta < 1e-14:
            continue
        new_a = a.copy()
        new_a[i] -= delta
        new_a[j] += delta
        cur = _objective(new_a)
        if cur + 1e-14 < best_score:
            a = new_a
            best_score = cur
            best_seq = a.copy()
            print(
                f"[{time.time():.1f}] SA improve {i}->{j} Δ={delta:.3e} best {best_score:.6f}"
            )
        else:
            if rng.random() < math.exp(-(cur - best_score) / max(T, 1e-12)):
                a = new_a
        T *= decay
    return best_seq, best_score


# ----------------------------------------------------------------------
# Upsampling helper -------------------------------------------------------
def _upsample(seq: np.ndarray, new_len: int) -> np.ndarray:
    """Linear interpolation of a simplex‑vector to a larger length."""
    old_len = seq.shape[0]
    if new_len == old_len:
        return seq.copy()
    x_old = np.linspace(0.0, 1.0, old_len)
    x_new = np.linspace(0.0, 1.0, new_len)
    new_seq = np.interp(x_new, x_old, seq)
    new_seq = np.maximum(new_seq, 0.0)
    new_seq /= new_seq.sum()
    return new_seq


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: int = 1000) -> list[float]:
    """
    Search for a non‑negative step function that minimises ``evaluate_sequence``.
    The whole search respects the given time budget (default 1000 s).
    Returns a plain Python list of heights.
    """
    rng = np.random.default_rng(seed)
    start = time.time()
    deadline = start + budget_s - 3.0  # a safety margin

    overall_best_seq: np.ndarray | None = None
    overall_best_score = float("inf")
    top_candidates: list[tuple[float, np.ndarray]] = []

    # ------------------------------------------------------------------
    # 1. Seed generation (moderate lengths)
    lengths = [2000, 3000, 4000, 5000]  # keep lengths reasonable for quick convs
    # parameter grids
    p_grid = np.concatenate(
        [
            np.logspace(-2, -0.2, 7),  # 0.01 … 0.63
            np.logspace(-0.2, 0.5, 8),  # 0.63 … 3.16
        ]
    )
    lam_grid = np.linspace(0.0, 0.8, 9)

    for n in lengths:
        if time.time() >= deadline:
            break
        print(f"[{time.time():.1f}] sampling length {n}")

        # deterministic families (uniform, arcsine)
        for cand in (_distribution_uniform(n), _distribution_arcsine(n)):
            sc = _objective(cand)
            top_candidates.append((sc, cand.copy()))
            if sc < overall_best_score:
                overall_best_score, overall_best_seq = sc, cand.copy()
                print(f"[{time.time():.1f}] new global best (uniform/arcsine) {sc:.6f}")

        # parameterised families
        for p in p_grid:
            if time.time() >= deadline:
                break
            beta = _distribution_beta_sym(n, p)
            edge = _distribution_edge_weighted(n, p)
            edge2 = _distribution_edge_inversepow(n, p)
            sine = _distribution_sine_weighted(n, p)
            power = _distribution_power_law(n, p)
            expo = _distribution_exponential(n, p)

            for base in (beta, edge, edge2, sine, power, expo):
                for lam in lam_grid:
                    if time.time() >= deadline:
                        break
                    cand = _mix_with_uniform(base, lam)
                    sc = _objective(cand)
                    top_candidates.append((sc, cand.copy()))
                    if sc < overall_best_score:
                        overall_best_score, overall_best_seq = sc, cand.copy()
                        print(
                            f"[{time.time():.1f}] new global best len={n} p={p:.3g} lam={lam:.2f} score={sc:.6f}"
                        )

        # asymmetric beta
        for _ in range(8):
            if time.time() >= deadline:
                break
            a = rng.uniform(0.001, 3.0)
            b = rng.uniform(0.001, 3.0)
            cand = _distribution_beta_asym(n, a, b)
            sc = _objective(cand)
            top_candidates.append((sc, cand.copy()))
            if sc < overall_best_score:
                overall_best_score, overall_best_seq = sc, cand.copy()
                print(
                    f"[{time.time():.1f}] new global best asym‑beta len={n} a={a:.3g} b={b:.3g} score={sc:.6f}"
                )

        # triangular shapes
        tri = _distribution_triangle(n)
        inv_tri = _distribution_inverted_triangle(n)
        for base in (tri, inv_tri):
            cand = _mix_with_uniform(base, 0.5)
            sc = _objective(cand)
            top_candidates.append((sc, cand.copy()))
            if sc < overall_best_score:
                overall_best_score, overall_best_seq = sc, cand.copy()
                print(f"[{time.time():.1f}] new global best triangle len={n} score={sc:.6f}")

        # Gaussian family (several sigmas)
        for sigma in (n / 6.0, n / 4.0, n / 3.0):
            cand = _distribution_gaussian(n, sigma)
            sc = _objective(cand)
            top_candidates.append((sc, cand.copy()))
            if sc < overall_best_score:
                overall_best_score, overall_best_seq = sc, cand.copy()
                print(f"[{time.time():.1f}] new global best gaussian len={n} σ={sigma:.1f} score={sc:.6f}")

        # random Dirichlet seeds
        for _ in range(6):
            if time.time() >= deadline:
                break
            cand = _distribution_dirichlet(n, rng)
            sc = _objective(cand)
            top_candidates.append((sc, cand.copy()))
            if sc < overall_best_score:
                overall_best_score, overall_best_seq = sc, cand.copy()
                print(f"[{time.time():.1f}] new global best Dirichlet len={n} score={sc:.6f}")

        # sparse random seeds
        for _ in range(6):
            if time.time() >= deadline:
                break
            cand = _distribution_random_sparse(n, rng, density=0.006)
            sc = _objective(cand)
            top_candidates.append((sc, cand.copy()))
            if sc < overall_best_score:
                overall_best_score, overall_best_seq = sc, cand.copy()
                print(f"[{time.time():.1f}] new global best sparse len={n} score={sc:.6f}")

        # keep only the very best seeds so far
        top_candidates.sort(key=lambda x: x[0])
        top_candidates = top_candidates[:30]

    # ------------------------------------------------------------------
    # 2. Incorporate any previously discovered construction
    try:
        prev = np.asarray(height_sequence_1, dtype=np.float64)  # type: ignore[name-defined]
        if prev.ndim == 1 and prev.size > 0:
            prev = np.maximum(prev, 0.0)
            prev = prev / prev.sum()
            sc = _objective(prev)
            if sc < overall_best_score:
                overall_best_score, overall_best_seq = sc, prev.copy()
                print(f"[{time.time():.1f}] previous best score={sc:.6f}")
            top_candidates.append((sc, prev.copy()))
    except Exception:
        pass

    # ------------------------------------------------------------------
    # 3. Fallback guard
    if overall_best_seq is None:
        n = 3000
        overall_best_seq = _distribution_uniform(n)
        overall_best_score = _objective(overall_best_seq)
        print(f"[{time.time():.1f}] fallback uniform score={overall_best_score:.6f}")

    # ------------------------------------------------------------------
    # 4. Heavy mirror‑descent on a handful of promising seeds
    remaining = deadline - time.time()
    if remaining > 40.0 and top_candidates:
        seeds = [seq for _, seq in sorted(top_candidates, key=lambda x: x[0])[:8]]
        per_seed_time = remaining * 0.55 / max(1, len(seeds))
        print(
            f"[{time.time():.1f}] heavy descent on {len(seeds)} seeds (~{per_seed_time:.1f}s each)"
        )
        for idx, seed_seq in enumerate(seeds):
            if time.time() >= deadline:
                break
            sub_deadline = time.time() + per_seed_time
            a_opt, sc_opt = _subgradient_descent_momentum(
                seed_seq.copy(),
                sub_deadline,
                best_seq=overall_best_seq,
                best_score=overall_best_score,
                lr0=0.85,
                active_frac=0.07,
                max_active=70,
                beta1=0.9,
                beta2=0.999,
                eval_interval=150,
                patience=2500,
            )
            if sc_opt < overall_best_score:
                overall_best_score, overall_best_seq = sc_opt, a_opt.copy()
                print(
                    f"[{time.time():.1f}] after descent seed {idx} new global best {overall_best_score:.6f}"
                )

    # ------------------------------------------------------------------
    # 5. Progressive up‑sampling (as long as we have time)
    max_len = 35000  # upper bound for the final length
    while True:
        remaining = deadline - time.time()
        if remaining < 30.0:
            break
        cur_len = len(overall_best_seq)
        if cur_len >= max_len:
            break
        new_len = min(max_len, cur_len * 2)
        print(f"[{time.time():.1f}] up‑sampling from {cur_len} → {new_len}")
        up_seq = _upsample(overall_best_seq, new_len)

        sub_deadline = time.time() + remaining * 0.45
        up_seq_opt, up_score = _subgradient_descent_momentum(
            up_seq,
            sub_deadline,
            best_seq=overall_best_seq,
            best_score=overall_best_score,
            lr0=0.6,
            active_frac=0.06,
            max_active=60,
            beta1=0.9,
            beta2=0.999,
            eval_interval=120,
            patience=2000,
        )
        if up_score < overall_best_score:
            overall_best_score, overall_best_seq = up_score, up_seq_opt.copy()
            print(
                f"[{time.time():.1f}] up‑sampled descent yielded new best {overall_best_score:.6f}"
            )
        else:
            # even if not improving, keep the larger sequence for a final sweep
            overall_best_seq = up_seq_opt.copy()
            overall_best_score = up_score

    # ------------------------------------------------------------------
    # 6. Fine‑grained deterministic pairwise refinement
    if time.time() < deadline:
        print(f"[{time.time():.1f}] deterministic pairwise refinement")
        overall_best_seq, overall_best_score = _pairwise_swap_local(
            overall_best_seq.copy(),
            deadline,
            best_seq=overall_best_seq,
            best_score=overall_best_score,
            max_trials=500_000,
        )

    # ------------------------------------------------------------------
    # 7. Final cheap simulated‑annealing clean‑up
    if time.time() < deadline:
        print(f"[{time.time():.1f}] final simulated‑annealing clean‑up")
        overall_best_seq, overall_best_score = _random_local_search(
            overall_best_seq.copy(),
            deadline,
            best_seq=overall_best_seq,
            best_score=overall_best_score,
            init_temp=0.008,
            decay=0.9995,
        )

    # ------------------------------------------------------------------
    print(
        f"Final best score: {overall_best_score:.6f} (length={len(overall_best_seq)})"
    )
    return overall_best_seq.tolist()
