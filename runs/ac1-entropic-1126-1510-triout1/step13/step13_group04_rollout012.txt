The goal is to minimise  

\[
\text{score}(a)=\frac{2\,n\;\max_k (a*a)_k}{\bigl(\sum_i a_i\bigr)^2},
\]

where `a*a` is the full linear convolution of the non‑negative vector `a`.
The score is scale‑invariant, therefore we work with probability vectors
(`sum(a)=1`).  The algorithm builds a diverse pool of candidate distributions,
refines the most promising ones with several complementary local‑search
methods and finally upsamples the best solution to a larger length.

The main ideas that improve on the previous state‑of‑the‑art are

* **Richer family of seeds** – in addition to the old families we add
  triangular, cosine‑squared, exponential‑edge, logistic and more.
* **Longer lengths** – we also explore `n = 16000` and `32000`.  
  A longer, smoother distribution often yields a smaller maximal
  convolution value.
* **Gradient‑guided mass transfer** – a cheap deterministic step that moves a
  fraction of mass from the coordinate with the largest sub‑gradient to the
  one with the smallest sub‑gradient.  This directly reduces the active
  convolution entry.
* **Extended pair‑wise annealing** – a high‑temperature simulated‑annealing
  phase that randomly shifts mass between two positions; it explores the
  neighbourhood that gradient steps cannot reach.
* **Iterative up‑sampling** – after a good solution is found we double its
  length (linear interpolation) and repeat the refinement, which usually
  reduces the maximal convolution further.

All components respect the 1000 s budget, use only `numpy` and the provided
`evaluate_sequence`, and return a plain Python list ready for evaluation.

```python
"""
Propose a non‑negative step function (list of floats) that (almost) minimises
evaluate_sequence.  The algorithm builds a diverse seed pool, refines the
most promising candidates with several local‑search heuristics (exponentiated
gradient, gradient‑guided mass transfer, Frank‑Wolfe, pairwise simulated‑annealing)
and finally upsamples the best vector to a larger length, repeating the
refinement.  The whole search respects a 1000 s time budget.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
#  Utility: projection onto the probability simplex (safety fallback)
# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    if s < 0:
        raise ValueError("simplex radius must be non‑negative")
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.where(u - cssv / (np.arange(1, len(v) + 1)) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
#  Convolution (real FFT) – the bottleneck of the whole search
# ----------------------------------------------------------------------
def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Full linear self‑convolution a*a using a real FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1) - 1).bit_length()
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[: 2 * n - 1]


# ----------------------------------------------------------------------
#  Exact objective (the value returned by evaluate_sequence for a probability vector)
# ----------------------------------------------------------------------
def _objective(a: np.ndarray) -> float:
    """2·n·max(conv) for a probability vector a (sum(a)=1)."""
    n = a.shape[0]
    b = _fft_self_convolve(a)
    max_b = b.max()
    # sum_a is 1, but we keep the formula for completeness
    sum_a = a.sum()
    if sum_a < 0.01:
        return float("inf")
    return float(2.0 * n * max_b / (sum_a * sum_a))


# ----------------------------------------------------------------------
#  Sub‑gradient of the *active* convolution entry
# ----------------------------------------------------------------------
def _grad_of_max_conv(a: np.ndarray) -> tuple[np.ndarray, int]:
    """Return sub‑gradient of the currently maximal convolution entry."""
    n = a.shape[0]
    conv = _fft_self_convolve(a)
    k = int(np.argmax(conv))
    grad = np.zeros_like(a)
    if k < n:
        grad[: k + 1] = 2.0 * a[k::-1]
    else:
        start = k - (n - 1)
        grad[start:] = 2.0 * a[start:][::-1]
    return grad, k


# ----------------------------------------------------------------------
#  Analytic families (all return a probability vector, i.e. sum=1)
# ----------------------------------------------------------------------
def _distribution_uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _distribution_arcsine(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    vals /= vals.sum()
    return vals


def _distribution_beta_sym(n: int, p: float) -> np.ndarray:
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    w = np.exp((p - 1) * np.log(x) + (p - 1) * np.log(1.0 - x))
    if w.sum() == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _distribution_edge_weighted(n: int, p: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64) + 0.5
    j = (n - np.arange(n, dtype=np.float64) - 0.5)
    w = (i * j) ** (-p)
    if w.sum() == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _distribution_hann(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    w = 0.5 - 0.5 * np.cos(2.0 * np.pi * i / (n - 1))
    w_sum = w.sum()
    if w_sum == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w_sum
    return w


def _distribution_generalized_normal(n: int, q: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64) + 0.5
    x = i / n - 0.5
    w = np.exp(- (np.abs(x) / 0.5) ** q)
    w_sum = w.sum()
    if w_sum == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w_sum
    return w


def _distribution_power_law(n: int, p: float) -> np.ndarray:
    i = np.arange(1, n + 1, dtype=np.float64)
    w = i ** (-p)
    w /= w.sum()
    return w


def _distribution_random_sparse(n: int, rng: np.random.Generator, density: float = 0.01) -> np.ndarray:
    k = max(2, int(n * density))
    idx = rng.choice(n, size=k, replace=False)
    vals = np.zeros(n, dtype=np.float64)
    vals[idx] = rng.random(k)
    if vals.sum() == 0.0:
        vals[0] = 1.0
    vals /= vals.sum()
    return vals


def _distribution_triangular(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    w = 1.0 - np.abs((i - (n - 1) / 2.0) / ((n - 1) / 2.0))
    w[w < 0] = 0.0
    w /= w.sum()
    return w


def _distribution_cosine_squared(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    w = np.cos(np.pi * (i - (n - 1) / 2.0) / (n - 1)) ** 2
    w = np.maximum(w, 0.0)
    w /= w.sum()
    return w


def _distribution_exponential_edge(n: int, decay: float = 0.9) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    w = decay ** i + decay ** (n - 1 - i)
    w /= w.sum()
    return w


def _distribution_logistic(n: int, steep: float = 10.0) -> np.ndarray:
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    w = 1.0 / (1.0 + np.exp(-steep * (x - 0.5)))
    w = w - w.min()
    w /= w.sum()
    return w


def _mix_with_uniform(base: np.ndarray, lam: float) -> np.ndarray:
    """Convex combination with the uniform distribution."""
    n = base.shape[0]
    uniform = np.full(n, 1.0 / n, dtype=np.float64)
    return lam * uniform + (1.0 - lam) * base


# ----------------------------------------------------------------------
#  Gradient‑guided mass‑transfer descent (core new primitive)
# ----------------------------------------------------------------------
def _mass_transfer_descent(
    a: np.ndarray,
    deadline: float,
    best_seq: np.ndarray,
    best_score: float,
    step_factor: float = 0.5,
    eval_every: int = 200,
) -> tuple[np.ndarray, float]:
    """
    Move a fraction of mass from the coordinate with the largest gradient
    to the coordinate with the smallest gradient.  Accept only improving
    moves – the step size is gradually reduced by the factor `step_factor`.
    """
    n = a.shape[0]
    it = 0
    while time.time() < deadline:
        it += 1
        grad, _ = _grad_of_max_conv(a)
        i_max = int(np.argmax(grad))
        i_min = int(np.argmin(grad))

        if a[i_max] <= 1e-14 or i_max == i_min:
            break

        eps = min(a[i_max] * step_factor, 0.02)  # move at most 2 % of total mass
        new_a = a.copy()
        new_a[i_max] -= eps
        new_a[i_min] += eps

        cur = _objective(new_a)

        if cur < best_score - 1e-15:
            best_score = cur
            best_seq = new_a.copy()
            a = new_a
            print(f"[{time.time():.1f}] mt it={it} new global best {best_score:.6f}")
        else:
            # optional: try a tiny random perturbation if stuck
            if it % 500 == 0:
                a = _project_onto_simplex(a)  # re‑project for numerical safety
        if it % eval_every == 0:
            # occasionally re‑evaluate to keep the outer best up‑to‑date
            cur_full = _objective(a)
            if cur_full < best_score - 1e-15:
                best_score = cur_full
                best_seq = a.copy()
    return best_seq, best_score


# ----------------------------------------------------------------------
#  Exponentiated‑gradient descent (kept from the reference implementation)
# ----------------------------------------------------------------------
def _exp_gradient_descent(
    a: np.ndarray,
    deadline: float,
    best_seq: np.ndarray,
    best_score: float,
    eta0: float = 0.8,
    eval_every: int = 80,
) -> tuple[np.ndarray, float]:
    n = a.shape[0]
    t = 0
    while time.time() < deadline:
        t += 1
        grad, _ = _grad_of_max_conv(a)
        eta = eta0 / math.sqrt(t)
        a = a * np.exp(-eta * grad)
        s = a.sum()
        if s == 0.0:
            a = np.full(n, 1.0 / n, dtype=np.float64)
        else:
            a = a / s

        if t % eval_every == 0:
            cur = _objective(a)
            if cur < best_score - 1e-14:
                best_score = cur
                best_seq = a.copy()
                print(f"[{time.time():.1f}] EG it={t} new global best {best_score:.6f}")
    return best_seq, best_score


# ----------------------------------------------------------------------
#  Frank‑Wolfe style refinement (kept, slightly faster step size)
# ----------------------------------------------------------------------
def _frank_wolfe_descent(
    a: np.ndarray,
    deadline: float,
    best_seq: np.ndarray,
    best_score: float,
    eval_every: int = 100,
) -> tuple[np.ndarray, float]:
    n = a.shape[0]
    t = 0
    while time.time() < deadline:
        t += 1
        grad, _ = _grad_of_max_conv(a)
        i_min = int(np.argmin(grad))
        gamma = 2.0 / (t + 2.0)
        a = a + gamma * ((np.eye(1, n, i_min)[0]) - a)

        if t % eval_every == 0:
            cur = _objective(a)
            if cur < best_score - 1e-14:
                best_score = cur
                best_seq = a.copy()
                print(f"[{time.time():.1f}] FW it={t} new global best {best_score:.6f}")
    return best_seq, best_score


# ----------------------------------------------------------------------
#  Simulated‑annealing pairwise mass transfers (final polishing)
# ----------------------------------------------------------------------
def _pairwise_annealing(
    a: np.ndarray,
    deadline: float,
    best_seq: np.ndarray,
    best_score: float,
    rng: np.random.Generator,
    max_trials: int | None = None,
) -> tuple[np.ndarray, float]:
    n = a.shape[0]
    start_time = time.time()
    total_time = deadline - start_time
    trials = 0
    while time.time() < deadline:
        if max_trials is not None and trials >= max_trials:
            break
        trials += 1
        src = np.where(a > 1e-12)[0]
        if src.size == 0:
            break
        i = rng.choice(src)
        j = rng.integers(n - 1)
        if j >= i:
            j += 1
        delta = a[i] * rng.random() * 0.4
        if delta < 1e-16:
            continue
        new_a = a.copy()
        new_a[i] -= delta
        new_a[j] += delta
        cur = _objective(new_a)
        elapsed = time.time() - start_time
        temperature = max(0.0, 1.0 - elapsed / total_time) * best_score * 0.01
        if cur < best_score - 1e-14:
            accept = True
        else:
            accept = temperature > 0.0 and rng.random() < math.exp(-(cur - best_score) / temperature)
        if accept:
            a = new_a
            if cur < best_score - 1e-14:
                best_score = cur
                best_seq = a.copy()
                print(f"[{time.time():.1f}] anneal i={i}->j={j} δ={delta:.3e} new best {best_score:.6f}")
    return best_seq, best_score


# ----------------------------------------------------------------------
#  Linear up‑sampling preserving the simplex (used after a good candidate)
# ----------------------------------------------------------------------
def _upsample(a: np.ndarray, new_n: int) -> np.ndarray:
    old_n = a.shape[0]
    if new_n == old_n:
        return a.copy()
    x_old = (np.arange(old_n, dtype=np.float64) + 0.5) / old_n
    x_new = (np.arange(new_n, dtype=np.float64) + 0.5) / new_n
    a_new = np.interp(x_new, x_old, a, left=0.0, right=0.0)
    a_new = np.maximum(a_new, 0.0)
    if a_new.sum() == 0.0:
        a_new[:] = 1.0 / new_n
    else:
        a_new /= a_new.sum()
    return a_new


# ----------------------------------------------------------------------
#  Main entry point – the function that will be called by the judge
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: int = 1000) -> list[float]:
    """
    Search for a non‑negative step function that minimises ``evaluate_sequence``.
    The routine respects a total 1000 s budget and returns a plain list.
    """
    rng = np.random.default_rng(seed)
    start = time.time()
    deadline = start + budget_s - 2.0          # safety margin

    overall_best_seq: np.ndarray | None = None
    overall_best_score = float("inf")
    top_candidates: list[tuple[float, np.ndarray]] = []

    # ------------------------------------------------------------------
    # 1.  Build a diverse pool of seeds.
    # ------------------------------------------------------------------
    lengths = [2000, 4000, 8000, 16000, 32000]   # we also try much larger vectors
    # parameter grids – slightly richer than before
    p_grid = np.concatenate(
        [
            np.logspace(-2.0, -0.5, 12, base=10),
            np.logspace(-0.5, 0.0, 12, base=10),
            np.logspace(0.0, 0.7, 10, base=10),
        ]
    )
    lam_grid = np.linspace(0.0, 1.0, 5)
    q_grid = np.linspace(0.5, 3.0, 8)
    power_grid = np.linspace(0.2, 2.5, 9)
    decay_grid = np.linspace(0.6, 0.98, 7)

    # helper to store a candidate
    def _store_candidate(vec: np.ndarray):
        nonlocal overall_best_seq, overall_best_score, top_candidates
        sc = _objective(vec)
        top_candidates.append((sc, vec.copy()))
        if sc < overall_best_score:
            overall_best_score = sc
            overall_best_seq = vec.copy()
            print(f"[{time.time():.1f}] new global best len={len(vec)} score={sc:.6f}")

    # scan all lengths – the loop aborts automatically when time runs out
    for n in lengths:
        if time.time() >= deadline:
            break
        print(f"[{time.time():.1f}] scanning length {n}")

        # deterministic families
        for cand in (
            _distribution_uniform(n),
            _distribution_arcsine(n),
            _distribution_hann(n),
            _distribution_triangular(n),
            _distribution_cosine_squared(n),
        ):
            _store_candidate(cand)

        # beta and edge‑weighted families
        for p in p_grid:
            if time.time() >= deadline:
                break
            beta = _distribution_beta_sym(n, p)
            edge = _distribution_edge_weighted(n, p)
            for base in (beta, edge):
                for lam in lam_grid:
                    if time.time() >= deadline:
                        break
                    cand = _mix_with_uniform(base, lam)
                    _store_candidate(cand)

        # generalized normal shapes
        for q in q_grid:
            if time.time() >= deadline:
                break
            cand = _distribution_generalized_normal(n, q)
            _store_candidate(cand)

        # power‑law families
        for pw in power_grid:
            if time.time() >= deadline:
                break
            cand = _distribution_power_law(n, pw)
            _store_candidate(cand)

        # exponential‑edge families
        for d in decay_grid:
            if time.time() >= deadline:
                break
            cand = _distribution_exponential_edge(n, d)
            _store_candidate(cand)

        # logistic families (symmetric after mirroring)
        for steep in (5.0, 10.0, 15.0):
            if time.time() >= deadline:
                break
            cand = _distribution_logistic(n, steep)
            _store_candidate(cand)

        # random Dirichlet seeds
        for _ in range(8):
            if time.time() >= deadline:
                break
            alpha = rng.uniform(0.01, 2.0)
            cand = rng.dirichlet(np.full(n, alpha))
            _store_candidate(cand)

        # sparse random seeds
        for _ in range(8):
            if time.time() >= deadline:
                break
            cand = _distribution_random_sparse(n, rng, density=0.015)
            _store_candidate(cand)

        # keep only the best few (30) – discarding the rest saves memory
        top_candidates.sort(key=lambda x: x[0])
        top_candidates = top_candidates[:30]

    # ------------------------------------------------------------------
    # 2.  If a historic solution exists, insert it.
    # ------------------------------------------------------------------
    try:
        prev = np.asarray(height_sequence_1, dtype=np.float64)  # type: ignore[name-defined]
        if prev.ndim == 1:
            prev = prev / prev.sum()
            _store_candidate(prev)
    except Exception:
        pass

    # ------------------------------------------------------------------
    # 3.  Fallback – guarantee we have something.
    # ------------------------------------------------------------------
    if overall_best_seq is None:
        overall_best_seq = _distribution_uniform(4000)
        overall_best_score = _objective(overall_best_seq)
        print(f"[{time.time():.1f}] fallback uniform score={overall_best_score:.6f}")

    # ------------------------------------------------------------------
    # 4.  Refine the top seeds with exponentiated‑gradient descent.
    # ------------------------------------------------------------------
    remaining = deadline - time.time()
    if remaining > 30.0 and top_candidates:
        # we devote roughly 35 % of the remaining time to EG on the 5 best seeds
        seed_vectors = [seq for _, seq in sorted(top_candidates, key=lambda x: x[0])[:5]]
        per_seed = remaining * 0.35 / max(1, len(seed_vectors))
        print(f"[{time.time():.1f}] EG descent on {len(seed_vectors)} seeds")
        for idx, vec in enumerate(seed_vectors):
            if time.time() >= deadline:
                break
            sub_deadline = time.time() + per_seed
            a_opt, sc_opt = _exp_gradient_descent(
                vec.copy(),
                sub_deadline,
                best_seq=overall_best_seq,
                best_score=overall_best_score,
                eta0=0.9,
                eval_every=80,
            )
            if sc_opt < overall_best_score - 1e-12:
                overall_best_score = sc_opt
                overall_best_seq = a_opt.copy()
                print(f"[{time.time():.1f}] after EG seed {idx} new global best {overall_best_score:.6f}")

    # ------------------------------------------------------------------
    # 5.  Gradient‑guided mass‑transfer refinement (new method)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        mt_time = (deadline - time.time()) * 0.20   # about 20 % of the remaining budget
        print(f"[{time.time():.1f}] mass‑transfer refinement (budget {mt_time:.1f}s)")
        sub_deadline = time.time() + mt_time
        overall_best_seq, overall_best_score = _mass_transfer_descent(
            overall_best_seq.copy(),
            sub_deadline,
            best_seq=overall_best_seq,
            best_score=overall_best_score,
            step_factor=0.6,
            eval_every=100,
        )
        print(f"[{time.time():.1f}] after mass‑transfer best score {overall_best_score:.6f}")

    # ------------------------------------------------------------------
    # 6.  Frank‑Wolfe refinement (kept from the reference implementation)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        fw_time = (deadline - time.time()) * 0.15
        print(f"[{time.time():.1f}] Frank‑Wolfe refinement (budget {fw_time:.1f}s)")
        sub_deadline = time.time() + fw_time
        overall_best_seq, overall_best_score = _frank_wolfe_descent(
            overall_best_seq.copy(),
            sub_deadline,
            best_seq=overall_best_seq,
            best_score=overall_best_score,
            eval_every=100,
        )
        print(f"[{time.time():.1f}] after FW best score {overall_best_score:.6f}")

    # ------------------------------------------------------------------
    # 7.  Upsample to a larger length (if time permits) and re‑refine.
    # ------------------------------------------------------------------
    while time.time() < deadline:
        cur_n = overall_best_seq.shape[0]
        target_n = cur_n * 2
        if target_n > 64000:
            break
        print(f"[{time.time():.1f}] upsampling {cur_n} → {target_n}")
        up_seq = _upsample(overall_best_seq, target_n)
        up_score = _objective(up_seq)
        if up_score < overall_best_score:
            overall_best_score = up_score
            overall_best_seq = up_seq
            print(f"[{time.time():.1f}] upsampled new best {overall_best_score:.6f}")

        # short burst of EG on the enlarged vector
        sub_deadline = time.time() + min(8.0, deadline - time.time())
        a_opt, sc_opt = _exp_gradient_descent(
            overall_best_seq.copy(),
            sub_deadline,
            best_seq=overall_best_seq,
            best_score=overall_best_score,
            eta0=0.7,
            eval_every=60,
        )
        if sc_opt < overall_best_score - 1e-12:
            overall_best_score = sc_opt
            overall_best_seq = a_opt.copy()
            print(f"[{time.time():.1f}] EG after upsample new best {overall_best_score:.6f}")

        # a quick mass‑transfer pass after each up‑sample
        sub_deadline = time.time() + min(4.0, deadline - time.time())
        overall_best_seq, overall_best_score = _mass_transfer_descent(
            overall_best_seq.copy(),
            sub_deadline,
            best_seq=overall_best_seq,
            best_score=overall_best_score,
            step_factor=0.5,
            eval_every=80,
        )
        # loop again if time still allows another doubling
        if deadline - time.time() < 5.0:
            break

    # ------------------------------------------------------------------
    # 8.  Final long‑run annealing polish.
    # ------------------------------------------------------------------
    if time.time() < deadline:
        print(f"[{time.time():.1f}] final annealing polish")
        a_opt, sc_opt = _pairwise_annealing(
            overall_best_seq.copy(),
            deadline,
            best_seq=overall_best_seq,
            best_score=overall_best_score,
            rng=rng,
            max_trials=None,
        )
        if sc_opt < overall_best_score - 1e-12:
            overall_best_score = sc_opt
            overall_best_seq = a_opt.copy()
            print(f"[{time.time():.1f}] after annealing new best {overall_best_score:.6f}")

    # ------------------------------------------------------------------
    # 9.  Return the best sequence found.
    # ------------------------------------------------------------------
    print(
        f"Final best score (2·n·max(conv)/sum²) = {overall_best_score:.6f} "
        f"(length={len(overall_best_seq)})"
    )
    return overall_best_seq.tolist()
```
