def _cross_corr(a, v):
    # returns correlation v ⋆ a for non-negative lags up to len(a)-1
    n = a.shape[0]
    m = v.shape[0]  # = 2n-1
    size = 1 << ((n+m-2).bit_length())

    corr = np.fft.irfft(V * np.conj(A), n=size)
    # The linear cross-correlation length is n+m-1; we need first n entries
    return corr[:n]

def _smooth_descent(a, deadline, best_seq, best_score,
                    lr0=0.5, momentum_beta=0.9,
                    alpha_start=5.0, alpha_factor=1.001, alpha_max=2000.0,
                    eval_interval=200, patience=2000):
    # Mirror descent with momentum on smoothed objective
    n = a.shape[0]
    t = 0
    momentum = np.zeros_like(a)
    no_improve = 0
    rng = np.random.default_rng()
    alpha = alpha_start
    while time.time() < deadline and t < 500_000:
        t += 1
        # compute S and grad
        # S is smooth objective; we only need grad for update, not S for evaluation.
        S, grad = _smooth_obj_and_grad(a, alpha)
        # momentum update
        momentum = momentum_beta * momentum + (1 - momentum_beta) * grad
        # learning rate
        eta = lr0 / math.sqrt(t)
        delta = -eta * momentum
        # clip delta to avoid overflow
        delta = np.clip(delta, -30, 30)
        a = a * np.exp(delta)
        # project onto simplex
        a = _project_onto_simplex(a, s=1.0)
        # increase alpha gradually
        alpha = min(alpha_max, alpha * alpha_factor)
        if t % eval_interval == 0:
            cur = _objective(a)
            if cur < best_score - 1e-12:
                best_score = cur
                best_seq = a.copy()
                no_improve = 0
                print(f"[{time.time():.1f}] it={t:,} new global best {best_score:.6f} alpha={alpha:.2f}")
            else:
                no_improve += 1
        if no_improve >= patience:
            # jitter
            jitter = rng.exponential(scale=0.005, size=n)
            a = a + jitter
            a = np.maximum(a, 0)
            a = _project_onto_simplex(a, s=1.0)
            no_improve = 0
    return best_seq, best_score

"""
Propose a step–function (non‑negative sequence) that minimizes
``evaluate_sequence``.  

The algorithm consists of three stages:

1️⃣  **Diverse seeding** – a handful of analytic families (edge‑weighted,
    edge‑inverse‑power, sine‑weighted, symmetric beta, arcsine, uniform)
    are sampled for a single length (≈8000).  The ~10 best seeds (lowest
    exact objective) are retained.

2️⃣  **Smoothed mirror descent** – we minimise a smooth approximation of
    the max‑convolution using an exponentiated‑gradient (mirror) descent
    with momentum.  The smoothness is controlled by a temperature
    parameter ``α`` that is increased exponentially during the run,
    turning the smoothed objective into the true ``max`` when the time
    budget ends.  Gradients are obtained via FFT‑based cross‑correlation,
    which is O(n log n) per iteration.

3️⃣  **Local polishing** – a greedy pairwise mass‑transfer step followed by
    a cheap simulated‑annealing style search removes any remaining
    irregularities.

The routine respects the 1000 s budget, prints progress and finally
returns a plain Python list that can be fed directly to
``evaluate_sequence``.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Helper: projection onto the probability simplex
def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Euclidean projection of v onto {x ≥ 0, Σx = s}."""
    if s < 0:
        raise ValueError("simplex radius must be non‑negative")
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho_arr = np.where(u - cssv / (np.arange(1, len(v) + 1)) > 0)[0]
    rho = rho_arr[-1] if rho_arr.size else -1
    theta = cssv[rho] / (rho + 1) if rho >= 0 else 0.0
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Fast self‑convolution via real FFT (returns length 2n‑1)
def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    n = a.shape[0]
    need = 2 * n - 1
    size = 1 << ((need - 1).bit_length())
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[:need]


# ----------------------------------------------------------------------
# Exact objective (used for evaluation only)
def _objective(a: np.ndarray) -> float:
    """Exact value 2·n·max(conv) when Σa = 1."""
    n = a.shape[0]
    b = _fft_self_convolve(a)
    return 2.0 * n * b.max()


# ----------------------------------------------------------------------
# Analytic families ----------------------------------------------------
def _distribution_uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _distribution_arcsine(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    vals /= vals.sum()
    return vals


def _distribution_beta_sym(n: int, p: float) -> np.ndarray:
    """Symmetric beta (Beta(p,p))."""
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    w = np.exp((p - 1) * np.log(x) + (p - 1) * np.log(1.0 - x))
    s = w.sum()
    if s == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= s
    return w


def _distribution_edge_weighted(n: int, p: float) -> np.ndarray:
    """U‑shaped family w_i ∝ ((i+0.5)*(n‑i‑0.5))^{−p}."""
    i = np.arange(n, dtype=np.float64) + 0.5
    j = (n - np.arange(n, dtype=np.float64) - 0.5)
    w = (i * j) ** (-p)
    s = w.sum()
    if s == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= s
    return w


def _distribution_edge_inversepow(n: int, p: float) -> np.ndarray:
    """Second edge family w_i ∝ (i+0.5)^{−p} + (n−i−0.5)^{−p}."""
    i = np.arange(n, dtype=np.float64) + 0.5
    j = (n - np.arange(n, dtype=np.float64) - 0.5)
    w = i ** (-p) + j ** (-p)
    s = w.sum()
    if s == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= s
    return w


def _distribution_sine_weighted(n: int, p: float) -> np.ndarray:
    """Weight ∝ sin(π (i+½)/n)^p."""
    theta = np.pi * (np.arange(n, dtype=np.float64) + 0.5) / n
    w = np.sin(theta) ** p
    s = w.sum()
    if s == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= s
    return w


def _distribution_random_sparse(
    n: int, rng: np.random.Generator, density: float = 0.01
) -> np.ndarray:
    """Sparse random vector with a given density of non‑zeros."""
    k = max(2, int(n * density))
    idx = rng.choice(n, size=k, replace=False)
    vals = np.zeros(n, dtype=np.float64)
    vals[idx] = rng.random(k)
    if vals.sum() == 0.0:
        vals[0] = 1.0
    vals /= vals.sum()
    return vals


def _distribution_dirichlet(n: int, rng: np.random.Generator) -> np.ndarray:
    """Random Dirichlet sample (random concentration)."""
    alpha = rng.uniform(0.01, 2.0)
    a = rng.dirichlet(np.full(n, alpha))
    return a.astype(np.float64)


def _mix_with_uniform(base: np.ndarray, lam: float) -> np.ndarray:
    """lam·U + (1‑lam)·base."""
    n = base.shape[0]
    uniform = np.full(n, 1.0 / n, dtype=np.float64)
    return lam * uniform + (1.0 - lam) * base

# Greedy pairwise refinement -------------------------------------------
def _pairwise_swap_local(
    a: np.ndarray,
    deadline: float,
    best_seq: np.ndarray,
    best_score: float,
    max_trials: int | None = None,
) -> tuple[np.ndarray, float]:
    n = a.shape[0]
    trials = 0
    rng = np.random.default_rng()
    while time.time() < deadline:
        if max_trials is not None and trials >= max_trials:
            break
        trials += 1

        src = np.where(a > 1e-12)[0]
        if src.size == 0:
            break
        i = rng.choice(src)
        j = rng.integers(n - 1)
        if j >= i:
            j += 1

        delta = a[i] * rng.random() * 0.35
        if delta < 1e-14:
            continue

        new_a = a.copy()
        new_a[i] -= delta
        new_a[j] += delta
        cur = _objective(new_a)

        if cur < best_score - 1e-14:
            a = new_a
            best_score = cur
            best_seq = a.copy()
            print(f"[{time.time():.1f}] swap {i}->{j} Δ={delta:.3e} new best {best_score:.6f}")
    return best_seq, best_score

# Simulated‑annealing style local search -------------------------------
def _random_local_search(
    a: np.ndarray,
    deadline: float,
    best_seq: np.ndarray,
    best_score: float,
    init_temp: float = 0.008,
    decay: float = 0.9997,
) -> tuple[np.ndarray, float]:
    n = a.shape[0]
    rng = np.random.default_rng()
    T = init_temp
    while time.time() < deadline:
        i = rng.integers(n)
        j = rng.integers(n - 1)
        if j >= i:
            j += 1
        delta = a[i] * rng.random() * 0.3
        if delta < 1e-14:
            continue
        new_a = a.copy()
        new_a[i] -= delta
        new_a[j] += delta
        cur = _objective(new_a)
        if cur < best_score - 1e-14:
            a = new_a
            best_score = cur
            best_seq = a.copy()
            print(f"[{time.time():.1f}] SA improve {i}->{j} Δ={delta:.3e} best {best_score:.6f}")
        else:
            if rng.random() < math.exp(-(cur - best_score) / max(T, 1e-12)):
                a = new_a
        T *= decay
    return best_seq, best_score


# ----------------------------------------------------------------------
# Smoothed mirror descent (main optimisation engine)
def _smooth_descent(
    a: np.ndarray,
    deadline: float,
    best_seq: np.ndarray,
    best_score: float,
    lr0: float = 0.5,
    momentum_beta: float = 0.9,
    alpha_start: float = 5.0,
    alpha_factor: float = 1.001,
    alpha_max: float = 2000.0,
    eval_interval: int = 250,
    patience: int = 3000,
) -> tuple[np.ndarray, float]:
    """
    Exponentiated‑gradient descent on a smoothed version of the max‑convolution.
    The smoothing parameter ``α`` is slowly increased, turning the objective
    into the true max at the end of the run.
    """
    n = a.shape[0]
    t = 0
    no_improve = 0
    momentum = np.zeros_like(a)
    rng = np.random.default_rng()
    # FFT size that can hold a (length n) and its self‑convolution (length 2n‑1)
    size = 1 << ((3 * n - 2 - 1).bit_length())
    alpha = alpha_start

    while time.time() < deadline and t < 500_000:
        t += 1

        # ---------- FFT of the current vector ----------
        A = np.fft.rfft(a, n=size)

        # ---------- self‑convolution (full) ----------
        conv_full = np.fft.irfft(A * A, n=size)[: 2 * n - 1]

        # ---------- smoothed max ----------
        max_c = conv_full.max()
        # stable log‑sum‑exp
        exp_terms = np.exp(alpha * (conv_full - max_c))
        sum_exp = exp_terms.sum()
        # weights sum to 1 and focus on large entries as α grows
        weights = exp_terms / sum_exp

        # ---------- gradient = 2·(weights ⋆ a) ----------
        W = np.fft.rfft(weights, n=size)
        # cross‑correlation (weights ⋆ a) via FFT
        corr = np.fft.irfft(W * np.conj(A), n=size)[:n]
        grad = 2.0 * corr

        # ---------- momentum ----------
        momentum = momentum_beta * momentum + (1.0 - momentum_beta) * grad

        # ---------- learning rate ----------
        eta = lr0 / math.sqrt(t)

        # ---------- mirror‑descent update ----------
        delta = -eta * momentum
        delta = np.clip(delta, -30.0, 30.0)          # avoid overflow in exp
        a = a * np.exp(delta)

        # ---------- simplex projection ----------
        a = _project_onto_simplex(a, s=1.0)

        # ---------- increase smoothing ----------
        alpha = min(alpha_max, alpha * alpha_factor)

        # ---------- periodic exact evaluation ----------
        if t % eval_interval == 0:
            cur = _objective(a)
            if cur < best_score - 1e-14:
                best_score = cur
                best_seq = a.copy()
                no_improve = 0
                print(
                    f"[{time.time():.1f}] it={t:,} new best {best_score:.6f} α={alpha:.2f}"
                )
            else:
                no_improve += 1

        # ---------- jitter when stuck ----------
        if no_improve >= patience:
            jitter = rng.exponential(scale=0.006, size=n)
            a = a + jitter
            a = np.maximum(a, 0.0)
            a = _project_onto_simplex(a, s=1.0)
            no_improve = 0

    return best_seq, best_score
