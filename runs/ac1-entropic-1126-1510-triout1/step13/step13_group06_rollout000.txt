"""
Propose a low‑score step‑function.

Algorithm
---------
1. **Exploratory scan** – For a wide range of lengths (≈400 … 4000) we
   evaluate a handful of analytically motivated families:
   * arcsine  (the continuous optimal shape for self‑convolution)
   * symmetric beta  (Beta(p,p) with p from 0.2 to 5)
   * edge‑weighted  (∝((i+½)(n‑i‑½))⁻ᵖ)
   * a uniform mix of the above (convex combination with the uniform vector)

   All candidates are normalised to unit‐sum, because the objective
   ``evaluate_sequence`` is scale‑invariant.  The best candidate seen so far
   (lowest exact objective) is kept.

2. **Local refinement** – The best few seeds (≤ 4) undergo a short
   multiplicative‑gradient descent on a soft‑max surrogate of the max‑convolution
   (the same “exponential descent’’ as in the previous state‑of‑the‑art).
   This removes tiny, systematic artefacts caused by discretisation.

3. **Optional pairwise/SA tweaks** – a brief deterministic swap and a cheap
   simulated‑annealing walk finish the optimisation if time permits.

The whole routine respects the supplied time budget (default 1000 s) and
returns a plain Python list suitable for ``evaluate_sequence``.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x≥0, Σx = s}."""
    if s < 0:
        raise ValueError("simplex radius must be non‑negative")
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho_arr = np.where(u - cssv / (np.arange(1, len(v) + 1)) > 0)[0]
    if rho_arr.size == 0:
        theta = 0.0
    else:
        rho = rho_arr[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Full self‑convolution a*a using a real FFT (returns length 2n‑1)."""
    n = a.shape[0]
    need = 2 * n - 1
    size = 1 << ((need - 1).bit_length())
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[:need]


# ----------------------------------------------------------------------
def _objective(a: np.ndarray) -> float:
    """
    Exact objective used by ``evaluate_sequence`` for a unit‑sum vector.
    Returns 2·n·max(conv(a,a)).
    """
    a = a.astype(np.float64, copy=False)
    s = a.sum()
    if s == 0.0:
        return float("inf")
    a = a / s                     # normalise to sum = 1
    n = a.shape[0]
    b = _fft_self_convolve(a)
    return float(2.0 * n * b.max())


# ----------------------------------------------------------------------
# Analytic families ----------------------------------------------------
def _distribution_uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _distribution_arcsine(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    vals /= vals.sum()
    return vals


def _distribution_beta_sym(n: int, p: float) -> np.ndarray:
    """Symmetric beta (Beta(p,p)) on the discrete grid."""
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    w = np.exp((p - 1) * np.log(x) + (p - 1) * np.log(1.0 - x))
    if w.sum() == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _distribution_edge_weighted(n: int, p: float) -> np.ndarray:
    """U‑shaped family: w_i ∝ ((i+0.5)*(n‑i‑0.5))^(−p)."""
    i = np.arange(n, dtype=np.float64) + 0.5
    j = (n - np.arange(n, dtype=np.float64) - 0.5)
    w = (i * j) ** (-p)
    if w.sum() == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _mix_with_uniform(base: np.ndarray, lam: float) -> np.ndarray:
    """Convex combination lam·U + (1‑lam)·base, where U is uniform."""
    n = base.shape[0]
    uniform = np.full(n, 1.0 / n, dtype=np.float64)
    return lam * uniform + (1.0 - lam) * base


# ----------------------------------------------------------------------
# Soft‑max exponential descent (same flavour as the previous code) ---
def _softmax_weights(conv: np.ndarray, tau: float) -> np.ndarray:
    """Soft‑max (stable) of τ·conv."""
    max_c = conv.max()
    w_raw = np.exp(tau * (conv - max_c))
    total = w_raw.sum()
    if total == 0.0:
        return np.full_like(conv, 1.0 / conv.size)
    return w_raw / total


def _weighted_gradient(a: np.ndarray, w: np.ndarray) -> np.ndarray:
    """
    Gradient of L(a)= Σ_i w_i·(a*a)[i].
    Closed‑form: ∇L = 2 * (a_rev ⊛ w) restricted to the first n entries.
    """
    n = a.shape[0]
    a_rev = a[::-1]
    need = len(w) + n - 1
    size = 1 << ((need - 1).bit_length())
    A = np.fft.rfft(a_rev, n=size)
    W = np.fft.rfft(w, n=size)
    conv = np.fft.irfft(A * W, n=size)[:need]
    grad = 2.0 * conv[n - 1 : n - 1 + n]
    return grad


def _full_exponential_descent(
    a: np.ndarray,
    deadline: float,
    best_seq: np.ndarray,
    best_score: float,
    eta0: float = 0.55,
    tau0: float = 0.5,
    tau_mult: float = 1.002,
    eval_interval: int = 120,
    patience: int = 2600,
) -> tuple[np.ndarray, float]:
    """
    Multiplicative (mirror) gradient descent on the soft‑max surrogate of
    the max‑convolution.  The entire con⁠volution vector is used, and the
    gradient is obtained by a single FFT‑based correlation.
    """
    n = a.shape[0]
    step = 0
    no_improve = 0
    rng = np.random.default_rng()
    tau = tau0

    while time.time() < deadline and step < 300_000:
        step += 1

        conv = _fft_self_convolve(a)
        w = _softmax_weights(conv, tau)
        grad = _weighted_gradient(a, w)

        eta = eta0 / math.sqrt(step)
        a = a * np.exp(-eta * grad)

        # project back onto the probability simplex
        s = a.sum()
        if s == 0.0:
            a = np.full(n, 1.0 / n, dtype=np.float64)
        else:
            a = a / s

        tau *= tau_mult

        if step % eval_interval == 0:
            cur = _objective(a)
            if cur < best_score - 1e-14:
                best_score = cur
                best_seq = a.copy()
                no_improve = 0
                print(f"[{time.time():.1f}] descent it={step} new best {best_score:.6f}")
            else:
                no_improve += 1

        if no_improve >= patience:
            jitter = rng.exponential(scale=0.003, size=n)
            a = a + jitter
            a = np.maximum(a, 0.0)
            a = a / a.sum()
            no_improve = 0

    return best_seq, best_score


# ----------------------------------------------------------------------
def _pairwise_swap_local(
    a: np.ndarray,
    deadline: float,
    best_seq: np.ndarray,
    best_score: float,
    max_trials: int | None = None,
) -> tuple[np.ndarray, float]:
    """Randomly transfer a small chunk of mass; keep change if we improve."""
    n = a.shape[0]
    trials = 0
    rng = np.random.default_rng()
    while time.time() < deadline:
        if max_trials is not None and trials >= max_trials:
            break
        trials += 1

        src = np.where(a > 1e-12)[0]
        if src.size == 0:
            break
        i = rng.choice(src)

        j = rng.integers(n - 1)
        if j >= i:
            j += 1

        delta = a[i] * rng.random() * 0.35
        if delta < 1e-14:
            continue

        new_a = a.copy()
        new_a[i] -= delta
        new_a[j] += delta

        cur = _objective(new_a)
        if cur < best_score - 1e-14:
            a = new_a
            best_score = cur
            best_seq = a.copy()
            print(
                f"[{time.time():.1f}] swap i={i}→j={j} δ={delta:.3e} new best {best_score:.6f}"
            )
    return best_seq, best_score


def _random_local_search(
    a: np.ndarray,
    deadline: float,
    best_seq: np.ndarray,
    best_score: float,
    init_temp: float = 1e-2,
    decay: float = 0.9996,
) -> tuple[np.ndarray, float]:
    """Very cheap SA that repeatedly transfers a random amount of mass."""
    n = a.shape[0]
    rng = np.random.default_rng()
    T = init_temp
    while time.time() < deadline:
        i = rng.integers(n)
        j = rng.integers(n - 1)
        if j >= i:
            j += 1
        delta = a[i] * rng.random() * 0.3
        if delta < 1e-14:
            continue
        new_a = a.copy()
        new_a[i] -= delta
        new_a[j] += delta
        cur = _objective(new_a)
        if cur < best_score - 1e-14:
            a = new_a
            best_score = cur
            best_seq = a.copy()
            print(f"[{time.time():.1f}] SA improve {i}->{j} Δ={delta:.3e} best {best_score:.6f}")
        else:
            if rng.random() < math.exp(-(cur - best_score) / max(T, 1e-12)):
                a = new_a
        T *= decay
    return best_seq, best_score


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: int = 1000) -> list[float]:
    """
    Search for a non‑negative step‑function that minimises ``evaluate_sequence``.
    The routine scans many analytic families over a range of lengths,
    refines the best few with gradient descent and finishes with tiny local
    moves.  Returns a plain Python list.
    """
    rng = np.random.default_rng(seed)
    start = time.time()
    deadline = start + budget_s - 3.0          # safety margin

    best_seq = None
    best_score = float("inf")
    top_candidates = []                       # (score, vector)

    # ---------------------------------------------------------------
    # 1.  Exhaustive scan over lengths and families
    # ---------------------------------------------------------------
    # length grid: we concentrate around the region where arcsine is known to be good.
    lengths = list(range(400, 4010, 50))       # 400, 450, …, 4000
    # parameters for beta / edge‑weighted families
    p_grid = np.concatenate(
        [
            np.logspace(-0.5, 0.0, 8, base=10),   # 0.316 … 1
            np.logspace(0.0, 0.7, 10, base=10),   # 1 … 5
        ]
    )
    lam_grid = np.linspace(0.0, 0.6, 7)       # mixing weight with uniform

    for n in lengths:
        if time.time() >= deadline:
            break
        # ----- arcsine -----
        cand = _distribution_arcsine(n)
        sc = _objective(cand)
        top_candidates.append((sc, cand.copy()))
        if sc < best_score:
            best_score, best_seq = sc, cand.copy()
            print(f"[{time.time():.1f}] new global best arcsine n={n} score={sc:.6f}")

        # mix arcsine with uniform
        for lam in lam_grid:
            mixed = _mix_with_uniform(cand, lam)
            sc = _objective(mixed)
            top_candidates.append((sc, mixed.copy()))
            if sc < best_score:
                best_score, best_seq = sc, mixed.copy()
                print(f"[{time.time():.1f}] mix arc λ={lam:.2f} n={n} score={sc:.6f}")

        # ----- symmetric beta -----
        for p in p_grid:
            beta = _distribution_beta_sym(n, p)
            sc = _objective(beta)
            top_candidates.append((sc, beta.copy()))
            if sc < best_score:
                best_score, best_seq = sc, beta.copy()
                print(f"[{time.time():.1f}] beta p={p:.3g} n={n} score={sc:.6f}")

            # mix with uniform
            for lam in lam_grid:
                mixed = _mix_with_uniform(beta, lam)
                sc = _objective(mixed)
                top_candidates.append((sc, mixed.copy()))
                if sc < best_score:
                    best_score, best_seq = sc, mixed.copy()
                    print(f"[{time.time():.1f}] mix beta λ={lam:.2f} p={p:.3g} n={n} score={sc:.6f}")

        # ----- edge‑weighted -----
        for p in p_grid:
            edge = _distribution_edge_weighted(n, p)
            sc = _objective(edge)
            top_candidates.append((sc, edge.copy()))
            if sc < best_score:
                best_score, best_seq = sc, edge.copy()
                print(f"[{time.time():.1f}] edge p={p:.3g} n={n} score={sc:.6f}")

            for lam in lam_grid:
                mixed = _mix_with_uniform(edge, lam)
                sc = _objective(mixed)
                top_candidates.append((sc, mixed.copy()))
                if sc < best_score:
                    best_score, best_seq = sc, mixed.copy()
                    print(f"[{time.time():.1f}] mix edge λ={lam:.2f} p={p:.3g} n={n} score={sc:.6f}")

        # keep a short list of the most promising seeds
        top_candidates.sort(key=lambda x: x[0])
        top_candidates = top_candidates[:40]

    # ---------------------------------------------------------------
    # 2.  Refine the best few seeds with exponential descent
    # ---------------------------------------------------------------
    remaining = deadline - time.time()
    if remaining > 30.0 and top_candidates:
        # take the 5 best distinct seeds
        seeds = [seq for _, seq in sorted(top_candidates, key=lambda x: x[0])[:5]]
        per_seed = remaining / max(1, len(seeds))
        print(f"[{time.time():.1f}] start descent on {len(seeds)} seeds")
        for idx, seed_vec in enumerate(seeds):
            if time.time() >= deadline:
                break
            sub_dead = time.time() + per_seed
            a_opt, sc_opt = _full_exponential_descent(
                seed_vec.copy(),
                sub_dead,
                best_seq=best_seq,
                best_score=best_score,
                eta0=0.55,
                tau0=0.5,
                tau_mult=1.002,
                eval_interval=100,
                patience=2000,
            )
            if sc_opt < best_score:
                best_score, best_seq = sc_opt, a_opt.copy()
                print(f"[{time.time():.1f}] descent seed {idx} new best {best_score:.6f}")

    # ---------------------------------------------------------------
    # 3.  Small deterministic swap refinement
    # ---------------------------------------------------------------
    if time.time() < deadline:
        print(f"[{time.time():.1f}] pairwise refinement")
        best_seq, best_score = _pairwise_swap_local(
            best_seq.copy(),
            deadline,
            best_seq=best_seq,
            best_score=best_score,
            max_trials=200_000,
        )

    # ---------------------------------------------------------------
    # 4.  Final cheap simulated‑annealing walk
    # ---------------------------------------------------------------
    if time.time() < deadline:
        print(f"[{time.time():.1f}] final SA walk")
        best_seq, best_score = _random_local_search(
            best_seq.copy(),
            deadline,
            best_seq=best_seq,
            best_score=best_score,
            init_temp=0.008,
            decay=0.9997,
        )

    print(f"Finished – best score {best_score:.6f} (length={len(best_seq)})")
    return best_seq.tolist()
