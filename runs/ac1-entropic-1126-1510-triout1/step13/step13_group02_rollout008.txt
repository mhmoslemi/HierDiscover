def _distribution_beta(n, alpha, beta):
    i = np.arange(n, dtype=np.float64) + 0.5
    x = i / n
    vals = (x**(alpha-1)) * ((1-x)**(beta-1))
    vals[vals < 0] = 0  # ensure non-negative
    vals_sum = vals.sum()
    if vals_sum == 0:
        vals = np.full(n, 1.0/n)
    else:
        vals = vals / vals_sum
    return vals

"""
step_function_search_v2.py

A 1000‑second search for a non‑negative step function that minimises

    2·n·max(conv(a,a)) / (∑a)²

The algorithm is built around three ideas that differ from the previous
implementation:

1. **Smooth max‑approximation** – instead of steering the update only
   with the single active convolution entry we minimise the
   log‑sum‑exp of the whole convolution (a smooth upper bound of the
   max).  The gradient of this surrogate can be obtained by a single
   extra FFT‑based convolution, which makes each iteration cheap.

2. **Continuation on the smoothness parameter** – we start with a very
   smooth surrogate (β≈1) to explore the landscape, then increase the
   temperature β (β≈100) so that the algorithm finally behaves like the
   original max‑optimiser but from a much better starting point.

3. **More diverse seeds** – besides uniform, arcsine, Hann, edge‑weighted
   and generalized‑normal families we also try symmetric Beta‑shapes
   and a finer sweep of the edge‑weight exponent (including negative
   values).  A tiny pool of the best seeds (≈15) is kept and polished
   with the smooth‑gradient descent.  Afterwards the current best vector
   is up‑sampled (linear interpolation) and refined again.

All work respects the 1000 s budget (a 2 s safety margin).  The function
`propose_candidate` returns a plain ``list[float]`` ready for
``evaluate_sequence``.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Simplex utilities
# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Euclidean projection of a vector onto the simplex {x≥0, Σx=s}."""
    if s < 0:
        raise ValueError("simplex radius must be non‑negative")
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.where(u - cssv / (np.arange(1, len(v) + 1)) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# FFT‑based convolution helpers
# ----------------------------------------------------------------------
def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Full linear self‑convolution a*a using a real FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1) - 1).bit_length()
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[: 2 * n - 1]


def _fft_cross_convolve(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution x*y using a real FFT."""
    m = x.shape[0] + y.shape[0] - 1
    size = 1 << ((m) - 1).bit_length()
    X = np.fft.rfft(x, n=size)
    Y = np.fft.rfft(y, n=size)
    conv = np.fft.irfft(X * Y, n=size)
    return conv[:m]


# ----------------------------------------------------------------------
# Exact objective (probability vectors, i.e. Σa = 1)
# ----------------------------------------------------------------------
def _objective(a: np.ndarray) -> float:
    """Exact value of ``evaluate_sequence`` for a probability vector."""
    n = a.shape[0]
    b = _fft_self_convolve(a)
    max_b = b.max()
    return float(2.0 * n * max_b)


# ----------------------------------------------------------------------
# Analytic families (all return a probability vector, sum=1)
# ----------------------------------------------------------------------
def _distribution_uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _distribution_arcsine(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    vals /= vals.sum()
    return vals


def _distribution_hann(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    w = 0.5 - 0.5 * np.cos(2.0 * np.pi * i / (n - 1))
    w_sum = w.sum()
    if w_sum == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w_sum
    return w


def _distribution_edge_weighted(n: int, p: float) -> np.ndarray:
    """Weight ∝ (i·j)^{‑p} where i,j are distances to the two edges."""
    i = np.arange(n, dtype=np.float64) + 0.5
    j = (n - np.arange(n, dtype=np.float64) - 0.5)
    w = (i * j) ** (-p)
    w_sum = w.sum()
    if w_sum == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w_sum
    return w


def _distribution_generalized_normal(n: int, q: float) -> np.ndarray:
    """Density ∝ exp(-|x‑½|^{q}) on [0,1]."""
    i = np.arange(n, dtype=np.float64) + 0.5
    x = i / n - 0.5
    w = np.exp(- (np.abs(x) / 0.5) ** q)
    w_sum = w.sum()
    if w_sum == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w_sum
    return w


def _distribution_beta(n: int, alpha: float, beta: float) -> np.ndarray:
    """Symmetric Beta‑type density on [0,1] (α,β > 0)."""
    i = np.arange(n, dtype=np.float64) + 0.5
    x = i / n
    vals = (x ** (alpha - 1)) * ((1 - x) ** (beta - 1))
    vals[vals < 0] = 0.0
    total = vals.sum()
    if total == 0.0:
        vals = np.full_like(vals, 1.0 / n)
    else:
        vals /= total
    return vals


def _distribution_random_sparse(
    n: int, rng: np.random.Generator, density: float = 0.01
) -> np.ndarray:
    """Sparse random vector with a given density."""
    k = max(2, int(n * density))
    idx = rng.choice(n, size=k, replace=False)
    vals = np.zeros(n, dtype=np.float64)
    vals[idx] = rng.random(k)
    if vals.sum() == 0.0:
        vals[0] = 1.0
    vals /= vals.sum()
    return vals


def _mix_with_uniform(base: np.ndarray, lam: float) -> np.ndarray:
    """Convex combination with the uniform distribution."""
    n = base.shape[0]
    uniform = np.full(n, 1.0 / n, dtype=np.float64)
    return lam * uniform + (1.0 - lam) * base


# ----------------------------------------------------------------------
# Upsample (linear interpolation while preserving the simplex)
# ----------------------------------------------------------------------
def _upsample(a: np.ndarray, new_n: int) -> np.ndarray:
    """Interpolate a probability vector to a larger length."""
    old_n = a.shape[0]
    if new_n == old_n:
        return a.copy()
    x_old = (np.arange(old_n, dtype=np.float64) + 0.5) / old_n
    x_new = (np.arange(new_n, dtype=np.float64) + 0.5) / new_n
    a_new = np.interp(x_new, x_old, a, left=0.0, right=0.0)
    a_new = np.maximum(a_new, 0.0)
    s = a_new.sum()
    if s == 0.0:
        a_new[:] = 1.0 / new_n
    else:
        a_new /= s
    return a_new


# ----------------------------------------------------------------------
# Smooth‑gradient descent (log‑sum‑exp surrogate)
# ----------------------------------------------------------------------
def _smooth_gradient_descent(
    a: np.ndarray,
    deadline: float,
    best_seq: np.ndarray,
    best_score: float,
    beta_schedule: list[float],
    eta0: float = 0.5,
    eval_every: int = 50,
) -> tuple[np.ndarray, float]:
    """
    Mirror‑descent on the smooth upper bound

        Lβ(a) = max(conv) + (1/β)·log Σ exp(β·(conv – max(conv)))

    The gradient is obtained via a second FFT‑based convolution.
    """
    n = a.shape[0]
    it = 0
    for beta in beta_schedule:
        # allocate a modest slice of remaining time to each β‑phase
        phase_dead = time.time() + min(5.0, (deadline - time.time()) * 0.4)
        while time.time() < phase_dead and time.time() < deadline:
            it += 1
            # self‑convolution
            conv = _fft_self_convolve(a)                     # length 2n‑1
            shift = conv.max()                               # for numerical stability
            # softmax weights over convolution entries
            scaled = beta * (conv - shift)
            exp_part = np.exp(scaled)
            w = exp_part / exp_part.sum()                    # Σ w = 1
            # gradient = 2 * (w * a)_{slice}
            conv_wa = _fft_cross_convolve(w, a)              # length 3n‑2
            grad = 2.0 * conv_wa[n - 1 : 2 * n - 1]

            # exponentiated‑gradient step
            eta = eta0 / math.sqrt(it)
            a = a * np.exp(-eta * grad)
            a = _project_onto_simplex(a)

            if it % eval_every == 0:
                cur = _objective(a)
                if cur < best_score - 1e-14:
                    best_score = cur
                    best_seq = a.copy()
                    print(
                        f"[{time.time():.1f}] smooth GD it={it} β={beta:.1f}"
                        f"  new best {best_score:.6f}"
                    )
    return best_seq, best_score


# ----------------------------------------------------------------------
# Simple pairwise annealing (used as a final polishing step)
# ----------------------------------------------------------------------
def _pairwise_annealing(
    a: np.ndarray,
    deadline: float,
    best_seq: np.ndarray,
    best_score: float,
    rng: np.random.Generator,
) -> tuple[np.ndarray, float]:
    """
    Random mass transfers between two bins.  Moves that improve the exact
    objective are always accepted; deteriorating moves are accepted with a
    decreasing temperature.
    """
    n = a.shape[0]
    start = time.time()
    total = deadline - start
    while time.time() < deadline:
        # pick a source proportional to its contribution to the current max
        conv = _fft_self_convolve(a)
        k = int(np.argmax(conv))
        contrib = np.zeros_like(a)
        # contribution of a_i to conv_k
        start_i = max(0, k - (n - 1))
        contrib[start_i : start_i + n - (k - start_i)] = a[
            start_i : start_i + n - (k - start_i)
        ] * a[k - np.arange(start_i, start_i + n - (k - start_i))]
        if contrib.sum() == 0.0:
            src = rng.integers(n)
        else:
            prob = contrib / contrib.sum()
            src = rng.choice(n, p=prob)

        # choose a distinct destination uniformly
        dst = rng.integers(n - 1)
        if dst >= src:
            dst += 1

        delta = a[src] * rng.random() * 0.35
        if delta < 1e-16:
            continue

        new_a = a.copy()
        new_a[src] -= delta
        new_a[dst] += delta

        cur = _objective(new_a)

        elapsed = time.time() - start
        temperature = max(0.0, 1.0 - elapsed / total) * best_score * 0.01

        accept = False
        if cur < best_score - 1e-14:
            accept = True
        elif temperature > 0.0:
            prob = math.exp(-(cur - best_score) / temperature)
            if rng.random() < prob:
                accept = True

        if accept:
            a = new_a
            if cur < best_score - 1e-14:
                best_score = cur
                best_seq = a.copy()
                print(
                    f"[{time.time():.1f}] anneal src={src}→dst={dst}"
                    f"  δ={delta:.3e}  new best {best_score:.6f}"
                )
    return best_seq, best_score
