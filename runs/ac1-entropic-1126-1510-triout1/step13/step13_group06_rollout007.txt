"""
propose_candidate
-----------------

Search for a non‑negative step‑function that minimizes
```
2·n·max(convolve(a,a)) / (sum(a)**2)
```
as used by the supplied ``evaluate_sequence``.  The search proceeds in
stages:

1. **Wide seed pool** – analytic families (uniform, arcsine, symmetric
   and asymmetric beta, edge‑weighted, sine‑weighted, Gaussian,
   double‑Gaussian), random Dirichlet draws and sparse vectors.  The best
   ~30 seeds are kept.

2. **Soft‑max exponential descent** – a multiplicative mirror–descent
   on a smooth soft‑max surrogate of the max‑convolution (same idea as
   the previous state‑of‑the‑art but with a richer seed set).

3. **Exact‑max exponential descent** – a focused exponentiated‑gradient
   step where the weight vector is the unit vector at the current
   maximising convolution index.  This drives the true max directly.

4. **Directed pairwise swaps** – using the gradient of the max entry to
   pick a donor (large gradient) and a receiver (small gradient) and
   move a small chunk of mass.

5. **Cheap simulated‑annealing walk** – stochastic refinement that
   accepts occasional worsening moves.

All phases respect a 1000 s wall‑clock budget and the best sequence found
is returned as a plain Python list.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x≥0, Σx = s}."""
    if s < 0:
        raise ValueError("simplex radius must be non‑negative")
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho_arr = np.where(u - cssv / (np.arange(1, len(v) + 1)) > 0)[0]
    if rho_arr.size == 0:
        theta = 0.0
    else:
        rho = rho_arr[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Full self‑convolution a*a using a real FFT (returns length 2n‑1)."""
    n = a.shape[0]
    need = 2 * n - 1
    size = 1 << ((need - 1).bit_length())
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[:need]


# ----------------------------------------------------------------------
def _objective(a: np.ndarray) -> float:
    """Exact objective 2·n·max(conv(a,a)) for a vector with sum = 1."""
    n = a.shape[0]
    b = _fft_self_convolve(a)
    return float(2.0 * n * b.max())


# ----------------------------------------------------------------------
# Analytic families ----------------------------------------------------
def _distribution_uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _distribution_arcsine(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    vals /= vals.sum()
    return vals


def _distribution_beta_sym(n: int, p: float) -> np.ndarray:
    """Symmetric beta (Beta(p,p))."""
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    w = np.exp((p - 1) * np.log(x) + (p - 1) * np.log(1.0 - x))
    if w.sum() == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _distribution_beta_asym(n: int, a: float, b: float) -> np.ndarray:
    """Asymmetric beta (Beta(a,b))."""
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    w = np.exp((a - 1) * np.log(x) + (b - 1) * np.log(1.0 - x))
    if w.sum() == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _distribution_edge_weighted(n: int, p: float) -> np.ndarray:
    """U‑shaped family: w_i ∝ ((i+0.5)*(n‑i‑0.5))^(−p)."""
    i = np.arange(n, dtype=np.float64) + 0.5
    j = (n - np.arange(n, dtype=np.float64) - 0.5)
    w = (i * j) ** (-p)
    if w.sum() == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _distribution_sine_weighted(n: int, p: float) -> np.ndarray:
    """Weight ∝ sin(π (i+½)/n)^p."""
    theta = np.pi * (np.arange(n, dtype=np.float64) + 0.5) / n
    w = np.sin(theta) ** p
    if w.sum() == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _distribution_gaussian(n: int, sigma: float, mu: float = None) -> np.ndarray:
    """Discrete Gaussian (or truncated) with given sigma."""
    if mu is None:
        mu = (n - 1) / 2.0
    i = np.arange(n, dtype=np.float64)
    w = np.exp(-0.5 * ((i - mu) / sigma) ** 2)
    if w.sum() == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _distribution_double_gaussian(
    n: int, sigma: float, mu1: float, mu2: float, weight: float = 0.5
) -> np.ndarray:
    """Mixture of two equally‑weighted Gaussians."""
    g1 = _distribution_gaussian(n, sigma, mu1)
    g2 = _distribution_gaussian(n, sigma, mu2)
    w = weight * g1 + (1 - weight) * g2
    w /= w.sum()
    return w


def _distribution_random_sparse(
    n: int, rng: np.random.Generator, density: float = 0.01
) -> np.ndarray:
    """Sparse random vector with a given density of non‑zeros."""
    k = max(2, int(n * density))
    idx = rng.choice(n, size=k, replace=False)
    vals = np.zeros(n, dtype=np.float64)
    vals[idx] = rng.random(k)
    if vals.sum() == 0.0:
        vals[0] = 1.0
    vals /= vals.sum()
    return vals


def _mix_with_uniform(base: np.ndarray, lam: float) -> np.ndarray:
    """Convex combination lam·U + (1‑lam)·base, where U is uniform."""
    n = base.shape[0]
    uniform = np.full(n, 1.0 / n, dtype=np.float64)
    return lam * uniform + (1.0 - lam) * base


# ----------------------------------------------------------------------
# Gradient utilities ----------------------------------------------------
def _softmax_weights(conv: np.ndarray, tau: float) -> np.ndarray:
    """Soft‑max of τ·conv (stable). Returns a probability vector."""
    max_c = conv.max()
    w_raw = np.exp(tau * (conv - max_c))
    total = w_raw.sum()
    if total == 0.0:
        return np.full_like(conv, 1.0 / conv.size)
    return w_raw / total


def _weighted_gradient(a: np.ndarray, w: np.ndarray) -> np.ndarray:
    """
    Gradient of   L(a)= Σ_i w_i·(a*a)[i]   w.r.t a.
    Closed‑form:   ∇L = 2 * (a_rev ⊛ w)   (restricted to the first n entries).
    """
    n = a.shape[0]
    a_rev = a[::-1]                     # length n
    need = len(w) + n - 1                # = 3·n‑2
    size = 1 << ((need - 1).bit_length())
    A = np.fft.rfft(a_rev, n=size)
    W = np.fft.rfft(w, n=size)
    conv = np.fft.irfft(A * W, n=size)
    conv = conv[:need]                  # length need
    grad = 2.0 * conv[n - 1 : n - 1 + n]   # slice of length n
    return grad

# Full‑softmax exponential descent (as in the previous state‑of‑the‑art)
def _full_exponential_descent(
    a: np.ndarray,
    deadline: float,
    best_seq: np.ndarray,
    best_score: float,
    eta0: float = 0.55,
    tau0: float = 0.5,
    tau_mult: float = 1.002,
    eval_interval: int = 120,
    patience: int = 2600,
) -> tuple[np.ndarray, float]:
    n = a.shape[0]
    t = 0
    no_improve = 0
    rng = np.random.default_rng()
    tau = tau0

    while time.time() < deadline and t < 250_000:
        t += 1

        conv = _fft_self_convolve(a)
        w = _softmax_weights(conv, tau)
        grad = _weighted_gradient(a, w)

        eta = eta0 / math.sqrt(t)
        a = a * np.exp(-eta * grad)

        s = a.sum()
        if s == 0.0:
            a = np.full(n, 1.0 / n, dtype=np.float64)
        else:
            a = a / s

        tau *= tau_mult

        if t % eval_interval == 0:
            cur = _objective(a)
            if cur < best_score - 1e-14:
                best_score = cur
                best_seq = a.copy()
                no_improve = 0
                print(f"[{time.time():.1f}] soft‑max it={t} new best {best_score:.6f}")
            else:
                no_improve += 1

        if no_improve >= patience:
            jitter = rng.exponential(scale=0.003, size=n)
            a = a + jitter
            a = np.maximum(a, 0.0)
            a = a / a.sum()
            no_improve = 0

    return best_seq, best_score


# ----------------------------------------------------------------------
# Exact‑max exponential descent (focus on the true max entry)
def _max_exponential_descent(
    a: np.ndarray,
    deadline: float,
    best_seq: np.ndarray,
    best_score: float,
    eta0: float = 0.6,
    eval_interval: int = 80,
    patience: int = 1500,
) -> tuple[np.ndarray, float]:
    n = a.shape[0]
    t = 0
    no_improve = 0
    rng = np.random.default_rng()
    while time.time() < deadline and t < 200_000:
        t += 1
        conv = _fft_self_convolve(a)
        k = int(conv.argmax())
        w = np.zeros_like(conv)
        w[k] = 1.0
        grad = _weighted_gradient(a, w)

        eta = eta0 / math.sqrt(t)
        a = a * np.exp(-eta * grad)

        s = a.sum()
        if s == 0.0:
            a = np.full(n, 1.0 / n, dtype=np.float64)
        else:
            a = a / s

        if t % eval_interval == 0:
            cur = _objective(a)
            if cur < best_score - 1e-14:
                best_score = cur
                best_seq = a.copy()
                no_improve = 0
                print(f"[{time.time():.1f}] max‑exp it={t} new best {best_score:.6f}")
            else:
                no_improve += 1

        if no_improve >= patience:
            jitter = rng.exponential(scale=0.002, size=n)
            a = a + jitter
            a = np.maximum(a, 0.0)
            a = a / a.sum()
            no_improve = 0

    return best_seq, best_score

# Directed pairwise swap local refinement
def _directed_pairwise_swap_local(
    a: np.ndarray,
    deadline: float,
    best_seq: np.ndarray,
    best_score: float,
    max_trials: int | None = None,
) -> tuple[np.ndarray, float]:
    """Move mass from a high‑gradient index to a low‑gradient one."""
    n = a.shape[0]
    trials = 0
    rng = np.random.default_rng()
    while time.time() < deadline:
        if max_trials is not None and trials >= max_trials:
            break
        trials += 1

        conv = _fft_self_convolve(a)
        k = int(conv.argmax())

        # gradient g_i = a_{k-i} (or 0 if out of range)
        g = np.zeros(n, dtype=np.float64)
        lo = max(0, k - (n - 1))
        hi = min(n - 1, k)
        idx = np.arange(lo, hi + 1)
        g[idx] = a[k - idx]

        # donor = index with large a_i * g_i
        donor_scores = a * g
        if donor_scores.max() == 0.0:
            continue
        i = int(np.argmax(donor_scores))

        # receiver = index with small gradient (preferably also small a)
        low_mask = g < np.quantile(g, 0.2)
        cand = np.where(low_mask)[0]
        cand = cand[cand != i]
        if cand.size == 0:
            j = rng.integers(n - 1)
            if j >= i:
                j += 1
        else:
            j = int(rng.choice(cand))

        delta = a[i] * rng.random() * 0.3
        if delta < 1e-14:
            continue

        new_a = a.copy()
        new_a[i] -= delta
        new_a[j] += delta

        cur = _objective(new_a)
        if cur < best_score - 1e-14:
            a = new_a
            best_seq = a.copy()
            best_score = cur
            print(
                f"[{time.time():.1f}] directed swap i={i}->{j} Δ={delta:.3e} new best {best_score:.6f}"
            )
    return best_seq, best_score


# ----------------------------------------------------------------------
# Cheap simulated‑annealing walk (same as before)
def _random_local_search(
    a: np.ndarray,
    deadline: float,
    best_seq: np.ndarray,
    best_score: float,
    init_temp: float = 1e-2,
    decay: float = 0.9997,
) -> tuple[np.ndarray, float]:
    n = a.shape[0]
    rng = np.random.default_rng()
    T = init_temp
    while time.time() < deadline:
        i = rng.integers(n)
        j = rng.integers(n - 1)
        if j >= i:
            j += 1
        delta = a[i] * rng.random() * 0.3
        if delta < 1e-14:
            continue
        new_a = a.copy()
        new_a[i] -= delta
        new_a[j] += delta
        cur = _objective(new_a)
        if cur < best_score - 1e-14:
            a = new_a
            best_score = cur
            best_seq = a.copy()
            print(f"[{time.time():.1f}] SA improve {i}->{j} Δ={delta:.3e} best {best_score:.6f}")
        else:
            if rng.random() < math.exp(-(cur - best_score) / max(T, 1e-12)):
                a = new_a
        T *= decay
    return best_seq, best_score

def propose_candidate(seed: int = 42, budget_s: int = 1000) -> list[float]:
    """
    Main entry point.  Returns a list of non‑negative heights (sum is irrelevant
    – ``evaluate_sequence`` will normalise it).  The routine observes the given
    time budget (seconds) and makes best use of it.
    """
    rng = np.random.default_rng(seed)
    start = time.time()
    deadline = start + budget_s - 3.0  # safety margin

    overall_best_seq = None
    overall_best_score = float("inf")
    top_candidates: list[tuple[float, np.ndarray]] = []

    # ------------------------------------------------------------------
    # 1. Generate a diverse pool of analytic / random seeds
    lengths = [2000, 3000, 4000, 5000, 7500]  # moderate sizes
    # parameter grids
    p_vals = np.array([0.2, 0.5, 0.8, 1.0, 2.0, 3.0, 5.0])
    lam_vals = np.linspace(0.0, 1.0, 7)  # mixing with uniform
    sigma_factors = [0.15, 0.22, 0.30]   # fractions of n for Gaussian width
    for n in lengths:
        if time.time() >= deadline:
            break
        print(f"[{time.time():.1f}] generating seeds for length {n}")

        # deterministic families
        for base in (_distribution_uniform(n), _distribution_arcsine(n)):
            sc = _objective(base)
            top_candidates.append((sc, base.copy()))
            if sc < overall_best_score:
                overall_best_score, overall_best_seq = sc, base.copy()
                print(f"[{time.time():.1f}] new global best (uniform/arcsine) {sc:.6f}")

        # parametrised families + mixing
        for p in p_vals:
            if time.time() >= deadline:
                break
            beta = _distribution_beta_sym(n, p)
            edge = _distribution_edge_weighted(n, p)
            sine = _distribution_sine_weighted(n, p)
            for base in (beta, edge, sine):
                for lam in lam_vals:
                    if time.time() >= deadline:
                        break
                    cand = _mix_with_uniform(base, lam)
                    sc = _objective(cand)
                    top_candidates.append((sc, cand.copy()))
                    if sc < overall_best_score:
                        overall_best_score, overall_best_seq = sc, cand.copy()
                        print(
                            f"[{time.time():.1f}] best (p={p:.2g},lam={lam:.2f}) {sc:.6f}"
                        )

        # asymmetric beta draws
        for _ in range(6):
            if time.time() >= deadline:
                break
            a_param = 10 ** rng.uniform(-2, 0.7)  # 0.01 … ~5
            b_param = 10 ** rng.uniform(-2, 0.7)
            cand = _distribution_beta_asym(n, a_param, b_param)
            sc = _objective(cand)
            top_candidates.append((sc, cand.copy()))
            if sc < overall_best_score:
                overall_best_score, overall_best_seq = sc, cand.copy()
                print(
                    f"[{time.time():.1f}] asym‑beta a={a_param:.3g} b={b_param:.3g} {sc:.6f}"
                )

        # Gaussian / double‑Gaussian families
        for fac in sigma_factors:
            sigma = fac * n
            cand = _distribution_gaussian(n, sigma)
            sc = _objective(cand)
            top_candidates.append((sc, cand.copy()))
            if sc < overall_best_score:
                overall_best_score, overall_best_seq = sc, cand.copy()
                print(f"[{time.time():.1f}] Gaussian sigma={sigma:.1f} {sc:.6f}")

            # double peak
            mu1 = 0.25 * n
            mu2 = 0.75 * n
            cand2 = _distribution_double_gaussian(n, sigma, mu1, mu2, weight=0.5)
            sc2 = _objective(cand2)
            top_candidates.append((sc2, cand2.copy()))
            if sc2 < overall_best_score:
                overall_best_score, overall_best_seq = sc2, cand2.copy()
                print(f"[{time.time():.1f}] double‑gauss sigma={sigma:.1f} {sc2:.6f}")

        # random Dirichlet seeds
        for alpha in [0.05, 0.2, 0.5, 1.0, 2.0]:
            if time.time() >= deadline:
                break
            cand = rng.dirichlet(np.full(n, alpha))
            sc = _objective(cand)
            top_candidates.append((sc, cand.copy()))
            if sc < overall_best_score:
                overall_best_score, overall_best_seq = sc, cand.copy()
                print(f"[{time.time():.1f}] Dirichlet α={alpha:.2f} {sc:.6f}")

        # sparse random seeds
        for density in [0.006, 0.012]:
            for _ in range(4):
                if time.time() >= deadline:
                    break
                cand = _distribution_random_sparse(n, rng, density=density)
                sc = _objective(cand)
                top_candidates.append((sc, cand.copy()))
                if sc < overall_best_score:
                    overall_best_score, overall_best_seq = sc, cand.copy()
                    print(f"[{time.time():.1f}] sparse density={density:.3f} {sc:.6f}")

        # keep only the most promising seeds globally
        top_candidates.sort(key=lambda x: x[0])
        top_candidates = top_candidates[:50]

    # ------------------------------------------------------------------
    # 2. Include any previously discovered construction, if present
    try:
        prev = np.asarray(height_sequence_1, dtype=np.float64)  # type: ignore[name-defined]
        if prev.ndim == 1 and prev.size > 0:
            prev = np.maximum(prev, 0.0)
            if prev.sum() < 1e-12:
                prev = np.full_like(prev, 1.0 / prev.size)
            else:
                prev = prev / prev.sum()
            sc = _objective(prev)
            if sc < overall_best_score:
                overall_best_score, overall_best_seq = sc, prev.copy()
                print(f"[{time.time():.1f}] previous best {sc:.6f}")
            top_candidates.append((sc, prev.copy()))
    except Exception:
        pass

    # ------------------------------------------------------------------
    # 3. Fallback (should never be needed)
    if overall_best_seq is None:
        n_fallback = 3000
        overall_best_seq = _distribution_uniform(n_fallback)
        overall_best_score = _objective(overall_best_seq)
        print(f"[{time.time():.1f}] fallback uniform {overall_best_score:.6f}")

    # ------------------------------------------------------------------
    # 4. Full‑softmax exponential descent on the top few seeds
    remaining = deadline - time.time()
    if remaining > 60.0 and top_candidates:
        seed_vectors = [seq for _, seq in sorted(top_candidates, key=lambda x: x[0])[:6]]
        per_seed = remaining * 0.55 / max(1, len(seed_vectors))
        print(f"[{time.time():.1f}] soft‑max descent on {len(seed_vectors)} seeds")
        for idx, sv in enumerate(seed_vectors):
            if time.time() >= deadline:
                break
            sub_deadline = time.time() + per_seed
            a_opt, sc_opt = _full_exponential_descent(
                sv.copy(),
                sub_deadline,
                best_seq=overall_best_seq,
                best_score=overall_best_score,
                eta0=0.55,
                tau0=0.7,
                tau_mult=1.003,
                eval_interval=150,
                patience=2400,
            )
            if sc_opt < overall_best_score:
                overall_best_score, overall_best_seq = sc_opt, a_opt.copy()
                print(
                    f"[{time.time():.1f}] after soft‑max seed {idx} new best {overall_best_score:.6f}"
                )

    # ------------------------------------------------------------------
    # 5. Exact‑max exponential descent on the current best
    remaining = deadline - time.time()
    if remaining > 40.0:
        sub_deadline = time.time() + remaining * 0.35
        print(f"[{time.time():.1f}] exact‑max descent")
        overall_best_seq, overall_best_score = _max_exponential_descent(
            overall_best_seq.copy(),
            sub_deadline,
            best_seq=overall_best_seq,
            best_score=overall_best_score,
            eta0=0.6,
            eval_interval=80,
            patience=1200,
        )

    # ------------------------------------------------------------------
    # 6. Directed pairwise swap refinement
    if time.time() < deadline:
        print(f"[{time.time():.1f}] directed pairwise refinement")
        overall_best_seq, overall_best_score = _directed_pairwise_swap_local(
            overall_best_seq.copy(),
            deadline,
            best_seq=overall_best_seq,
            best_score=overall_best_score,
            max_trials=400_000,
        )

    # ------------------------------------------------------------------
    # 7. Final cheap simulated‑annealing walk
    if time.time() < deadline:
        print(f"[{time.time():.1f}] final simulated‑annealing walk")
        overall_best_seq, overall_best_score = _random_local_search(
            overall_best_seq.copy(),
            deadline,
            best_seq=overall_best_seq,
            best_score=overall_best_score,
            init_temp=0.008,
            decay=0.9997,
        )

    # ------------------------------------------------------------------
    print(
        f"Final best score: {overall_best_score:.6f} (length={len(overall_best_seq)})"
    )
    # Return a plain Python list; ``evaluate_sequence`` will handle final checks.
    return overall_best_seq.tolist()
