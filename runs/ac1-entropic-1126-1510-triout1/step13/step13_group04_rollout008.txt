def _lp_step(f, M):
    n = len(f)
    g = cp.Variable(n, nonneg=True)
    constraints = []
    # Build convolution results linear expression for each offset
    for k in range(2*n-1):
        # sum_i f_i * g_{k-i}
        # define expression as sum_{i} f_i * g_{k-i}
        # This can be expressed as sum_{i = max(0, k-(n-1))}^{min(k,n-1)} f_i * g_{k-i}

    # optional: also set upper bound for sum(g) maybe not needed

"""
Propose a non‑negative step function that (almost) minimizes
``evaluate_sequence``.

Algorithm
---------
1. **Seed generation** – Build a compact set of probability vectors from
   several analytic families (uniform, arcsine, Hann, triangular, cosine‑squared,
   symmetric beta, edge‑weighted, generalized normal, power‑law, random Dirichlet
   and a sparse random vector).  Keep the best few seeds.

2. **Linear‑program (LP) refinement** – For a seed `f` compute the current
   maximal self‑convolution value `M = max(f*f)`.  Solve the linear program

        maximize   Σ g_j
        subject to  f * g ≤ M   (convolution is linear in g)
                    g_j ≥ 0

   This yields a direction `g` that can increase the ℓ₁‑norm while still
   respecting the current worst‐convolution bound.  Move a fraction of the
   mass toward `g` (convex combination) and repeat a few times.
   The LP is solved with SciPy’s *highs* solver on a sparse Toeplitz‑like
   constraint matrix.

3. **Exponentiated‑gradient (EG) descent** – Apply multiplicative mirror
   descent on the exact objective (sub‑gradient of the active convolution
   entry) to polish the current best vector.

4. **Upsampling & short EG bursts** – Double the length of the best vector,
   linearly interpolate while staying on the simplex, and perform a brief
   EG optimisation.  Repeat while time permits.

5. **Final annealing** – Perform a long‑run simulated‑annealing style
   pairwise mass‑transfer that accepts improving moves always and
   occasionally accepts worsening moves early on.

All stages respect a total wall‑clock budget of 1000 s (2 s safety margin).
The routine returns a plain ``list[float]`` ready for ``evaluate_sequence``.
"""

# ----------------------------------------------------------------------
# Simplex projection (fallback)
# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x≥0, Σx=s}."""
    if s < 0:
        raise ValueError("simplex radius must be non‑negative")
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.where(u - cssv / (np.arange(1, len(v) + 1)) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# FFT based full linear self‑convolution
# ----------------------------------------------------------------------
def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Full linear self‑convolution a*a using a real FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1) - 1).bit_length()
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[: 2 * n - 1]


# ----------------------------------------------------------------------
# Exact objective used by ``evaluate_sequence`` (scale‑invariant)
# ----------------------------------------------------------------------
def _objective(a: np.ndarray) -> float:
    """Return the exact value of ``evaluate_sequence`` for a (unnormalised)."""
    n = a.shape[0]
    b = _fft_self_convolve(a)
    max_b = b.max()
    sum_a = a.sum()
    if sum_a < 0.01:
        return float("inf")
    return float(2.0 * n * max_b / (sum_a * sum_a))


# ----------------------------------------------------------------------
# Analytic families – always return a probability vector (sum = 1)
# ----------------------------------------------------------------------
def _distribution_uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _distribution_arcsine(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    vals /= vals.sum()
    return vals


def _distribution_hann(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    w = 0.5 - 0.5 * np.cos(2.0 * np.pi * i / (n - 1))
    w_sum = w.sum()
    if w_sum == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w_sum
    return w


def _distribution_beta_sym(n: int, p: float) -> np.ndarray:
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    w = np.exp((p - 1) * np.log(x) + (p - 1) * np.log(1.0 - x))
    if w.sum() == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _distribution_edge_weighted(n: int, p: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64) + 0.5
    j = (n - np.arange(n, dtype=np.float64) - 0.5)
    w = (i * j) ** (-p)
    if w.sum() == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _distribution_generalized_normal(n: int, q: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64) + 0.5
    x = i / n - 0.5
    w = np.exp(- (np.abs(x) / 0.5) ** q)
    w_sum = w.sum()
    if w_sum == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w_sum
    return w


def _distribution_power_law(n: int, p: float) -> np.ndarray:
    i = np.arange(1, n + 1, dtype=np.float64)
    w = i ** (-p)
    w /= w.sum()
    return w


def _distribution_triangle(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    w = 1.0 - np.abs((i - (n - 1) / 2.0) / ((n - 1) / 2.0))
    w = np.maximum(w, 0.0)
    w /= w.sum()
    return w


def _distribution_cosine_squared(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    w = np.cos(np.pi * (i - (n - 1) / 2.0) / (n - 1))
    w = np.maximum(w, 0.0) ** 2
    w /= w.sum()
    return w


def _distribution_random_sparse(n: int, rng: np.random.Generator, density: float = 0.01) -> np.ndarray:
    k = max(2, int(n * density))
    idx = rng.choice(n, size=k, replace=False)
    vals = np.zeros(n, dtype=np.float64)
    vals[idx] = rng.random(k)
    if vals.sum() == 0.0:
        vals[0] = 1.0
    vals /= vals.sum()
    return vals


def _mix_with_uniform(base: np.ndarray, lam: float) -> np.ndarray:
    n = base.shape[0]
    uniform = np.full(n, 1.0 / n, dtype=np.float64)
    return lam * uniform + (1.0 - lam) * base


# ----------------------------------------------------------------------
# Gradient of the active convolution entry (sub‑gradient of max‑conv)
# ----------------------------------------------------------------------
def _grad_of_max_conv(a: np.ndarray) -> tuple[np.ndarray, int]:
    """
    Compute the sub‑gradient of the currently maximal convolution entry.
    Returns (grad, active_index).
    """
    n = a.shape[0]
    conv = _fft_self_convolve(a)
    k = int(np.argmax(conv))

    grad = np.zeros_like(a)
    if k < n:
        grad[: k + 1] = 2.0 * a[k::-1]
    else:
        start = k - (n - 1)
        grad[start:] = 2.0 * a[start:][::-1]
    return grad, k

def _exp_gradient_descent(
    a: np.ndarray,
    deadline: float,
    best_seq: np.ndarray,
    best_score: float,
    eta0: float = 0.8,
    eval_every: int = 80,
) -> tuple[np.ndarray, float]:
    """Perform exponentiated‑gradient descent on the exact objective."""
    n = a.shape[0]
    t = 0
    while time.time() < deadline:
        t += 1
        grad, _ = _grad_of_max_conv(a)
        eta = eta0 / math.sqrt(t)
        a = a * np.exp(-eta * grad)
        s = a.sum()
        if s == 0.0:
            a = np.full(n, 1.0 / n, dtype=np.float64)
        else:
            a = a / s

        if t % eval_every == 0:
            cur = _objective(a)
            if cur < best_score - 1e-14:
                best_score = cur
                best_seq = a.copy()
                print(f"[{time.time():.1f}] EG it={t} new global best {best_score:.6f}")
    return best_seq, best_score


# ----------------------------------------------------------------------
# Frank‑Wolfe style refinement (pairwise mass transfer guided by the gradient)
# ----------------------------------------------------------------------
def _frank_wolfe_descent(
    a: np.ndarray,
    deadline: float,
    best_seq: np.ndarray,
    best_score: float,
    eval_every: int = 100,
) -> tuple[np.ndarray, float]:
    """Frank‑Wolfe style descent moving towards the coordinate with smallest gradient."""
    n = a.shape[0]
    t = 0
    while time.time() < deadline:
        t += 1
        grad, _ = _grad_of_max_conv(a)
        i_min = int(np.argmin(grad))
        gamma = 2.0 / (t + 2.0)
        e_i = np.zeros_like(a)
        e_i[i_min] = 1.0
        a = a + gamma * (e_i - a)
        a = np.maximum(a, 0.0)
        a /= a.sum()
        if t % eval_every == 0:
            cur = _objective(a)
            if cur < best_score - 1e-14:
                best_score = cur
                best_seq = a.copy()
                print(f"[{time.time():.1f}] FW it={t} new global best {best_score:.6f}")
    return best_seq, best_score


# ----------------------------------------------------------------------
# Pairwise annealing (final polishing)
# ----------------------------------------------------------------------
def _pairwise_annealing(
    a: np.ndarray,
    deadline: float,
    best_seq: np.ndarray,
    best_score: float,
    rng: np.random.Generator,
    max_trials: int | None = None,
) -> tuple[np.ndarray, float]:
    """Random pairwise mass transfer with simulated‑annealing acceptance."""
    n = a.shape[0]
    start_time = time.time()
    total_time = deadline - start_time
    trials = 0
    while time.time() < deadline:
        if max_trials is not None and trials >= max_trials:
            break
        trials += 1

        src = np.where(a > 1e-12)[0]
        if src.size == 0:
            break
        i = rng.choice(src)

        j = rng.integers(n - 1)
        if j >= i:
            j += 1

        delta = a[i] * rng.random() * 0.4
        if delta < 1e-16:
            continue

        new_a = a.copy()
        new_a[i] -= delta
        new_a[j] += delta

        cur = _objective(new_a)
        elapsed = time.time() - start_time
        temperature = max(0.0, 1.0 - elapsed / total_time) * best_score * 0.01

        if cur < best_score - 1e-14:
            accept = True
        else:
            accept = temperature > 0.0 and rng.random() < math.exp(-(cur - best_score) / temperature)

        if accept:
            a = new_a
            if cur < best_score - 1e-14:
                best_score = cur
                best_seq = a.copy()
                print(f"[{time.time():.1f}] anneal i={i}->j={j} δ={delta:.3e} new best {best_score:.6f}")
    return best_seq, best_score


# ----------------------------------------------------------------------
# Linear up‑sampling while preserving the simplex
# ----------------------------------------------------------------------
def _upsample(a: np.ndarray, new_n: int) -> np.ndarray:
    """Interpolate a probability vector to a larger length."""
    old_n = a.shape[0]
    if new_n == old_n:
        return a.copy()
    x_old = (np.arange(old_n, dtype=np.float64) + 0.5) / old_n
    x_new = (np.arange(new_n, dtype=np.float64) + 0.5) / new_n
    a_new = np.interp(x_new, x_old, a, left=0.0, right=0.0)
    a_new = np.maximum(a_new, 0.0)
    if a_new.sum() == 0.0:
        a_new[:] = 1.0 / new_n
    else:
        a_new /= a_new.sum()
    return a_new


# ----------------------------------------------------------------------
# Linear‑program helper (solve max Σg  subject to f*g ≤ M)
# ----------------------------------------------------------------------
def _lp_optimize_step(f: np.ndarray, M: float) -> np.ndarray:
    """
    One LP step: maximise Σ g_j subject to the linear constraints
    (f * g)_k ≤ M   for all k,  g_j ≥ 0.
    Returns the optimal g (may be the zero vector if solver fails).
    """
    n = f.shape[0]
    rows = []
    cols = []
    data = []

    # Build the (2n‑1) × n constraint matrix.
    # entry (k, j) = f_i where i = k - j  (provided 0 ≤ i < n)
    for k in range(2 * n - 1):
        i_start = max(0, k - (n - 1))
        i_end = min(k, n - 1)
        length = i_end - i_start + 1
        idx_i = np.arange(i_start, i_end + 1)
        idx_j = k - idx_i
        rows.extend([k] * length)
        cols.extend(idx_j.tolist())
        data.extend(f[i_start:i_end + 1].tolist())

    A = csr_matrix((data, (rows, cols)), shape=(2 * n - 1, n))
    b = np.full(2 * n - 1, M, dtype=np.float64)
    c = -np.ones(n, dtype=np.float64)          # minimise -Σg   ⇒   maximise Σg
    bounds = [(0.0, None)] * n

    res = linprog(c, A_ub=A, b_ub=b, bounds=bounds, method='highs', options={'presolve': True})
    if not res.success:
        # fallback – return a zero vector (the LP is always feasible with g=0)
        return np.zeros_like(f)
    return res.x


def _lp_refine(a: np.ndarray,
               deadline: float,
               best_seq: np.ndarray,
               best_score: float,
               max_iter: int = 10) -> tuple[np.ndarray, float]:
    """
    Repeated LP‑based refinement of a probability vector.
    At each iteration we move towards the LP solution with a diminishing step size.
    """
    cur = a.copy()
    for it in range(max_iter):
        if time.time() >= deadline:
            break
        conv = _fft_self_convolve(cur)
        M = conv.max()
        g0 = _lp_optimize_step(cur, M)
        if g0.sum() <= 1e-12:
            break
        # Normalise to keep a probability vector
        g = g0 / g0.sum()
        step = 0.6 / (it + 2.0)          # decreasing step size
        cur = (1.0 - step) * cur + step * g
        cur = np.maximum(cur, 0.0)
        cur /= cur.sum()
        cur_score = _objective(cur)
        if cur_score < best_score - 1e-14:
            best_score = cur_score
            best_seq = cur.copy()
            print(f"[{time.time():.1f}] LP it={it} new global best {best_score:.6f}")
    return best_seq, best_score


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: int = 1000) -> list[float]:
    """
    Search for a non‑negative step function that minimises ``evaluate_sequence``.
    Returns a plain list of floats (the heights of the step function).
    """
    rng = np.random.default_rng(seed)
    start = time.time()
    deadline = start + budget_s - 2.0   # safety margin

    # ------------------------------------------------------------------
    # 0.  Basic parameters
    # ------------------------------------------------------------------
    base_n = 1024                     # size for the intensive LP stage
    overall_best_seq: np.ndarray | None = None
    overall_best_score = float("inf")

    # ------------------------------------------------------------------
    # 1.  Build a small diverse pool of seeds (probability vectors)
    # ------------------------------------------------------------------
    seed_vectors = []

    # deterministic families
    seed_vectors.append(_distribution_uniform(base_n))
    seed_vectors.append(_distribution_arcsine(base_n))
    seed_vectors.append(_distribution_hann(base_n))
    seed_vectors.append(_distribution_triangle(base_n))
    seed_vectors.append(_distribution_cosine_squared(base_n))

    # beta, edge‑weighted, generalized‑normal, power‑law
    for p in (0.3, 2.0):
        seed_vectors.append(_distribution_beta_sym(base_n, p))
        seed_vectors.append(_distribution_edge_weighted(base_n, p))
    for q in (0.8, 2.0):
        seed_vectors.append(_distribution_generalized_normal(base_n, q))
    for pw in (0.5, 1.5):
        seed_vectors.append(_distribution_power_law(base_n, pw))

    # random Dirichlet
    for _ in range(4):
        alpha = rng.uniform(0.05, 2.0)
        seed_vectors.append(rng.dirichlet(np.full(base_n, alpha)))

    # sparse random
    for _ in range(2):
        seed_vectors.append(_distribution_random_sparse(base_n, rng, density=0.015))

    # mixture with uniform (adds some robustness)
    mixed = []
    for base in list(seed_vectors):
        for lam in (0.0, 0.3, 0.6):
            mixed.append(_mix_with_uniform(base, lam))
    seed_vectors.extend(mixed)

    # previous construction, if available
    try:
        prev = np.asarray(height_sequence_1, dtype=np.float64)   # type: ignore[name-defined]
        if prev.ndim == 1:
            prev = np.maximum(prev, 0.0)
            prev /= prev.sum()
            seed_vectors.append(prev)
    except Exception:
        pass

    # Normalise and evaluate everything, keep the best 10 as starting points
    evaluated = []
    for vec in seed_vectors:
        vec = np.maximum(vec, 0.0)
        if vec.sum() == 0.0:
            continue
        vec = vec / vec.sum()
        sc = _objective(vec)
        evaluated.append((sc, vec))
        if sc < overall_best_score:
            overall_best_score = sc
            overall_best_seq = vec.copy()
            print(f"[{time.time():.1f}] initial global best score {sc:.6f}")

    evaluated.sort(key=lambda x: x[0])
    top_start = [seq for _, seq in evaluated[:10]]

    # ------------------------------------------------------------------
    # 2.  LP‑based refinement on a few elite seeds
    # ------------------------------------------------------------------
    remaining = deadline - time.time()
    if remaining > 30.0 and top_start:
        per_seed = remaining * 0.35 / max(1, len(top_start))
        for idx, svec in enumerate(top_start):
            if time.time() >= deadline:
                break
            sub_deadline = time.time() + per_seed
            best_seq, best_score = _lp_refine(
                svec.copy(),
                sub_deadline,
                overall_best_seq,
                overall_best_score,
                max_iter=12,
            )
            if best_score < overall_best_score - 1e-12:
                overall_best_score = best_score
                overall_best_seq = best_seq.copy()
                print(f"[{time.time():.1f}] after LP seed {idx} new global best {overall_best_score:.6f}")

    # ------------------------------------------------------------------
    # 3.  Exponentiated‑gradient descent for fine polishing
    # ------------------------------------------------------------------
    if time.time() < deadline:
        eg_time = (deadline - time.time()) * 0.25
        print(f"[{time.time():.1f}] EG polishing (budget {eg_time:.1f}s)")
        overall_best_seq, overall_best_score = _exp_gradient_descent(
            overall_best_seq.copy(),
            time.time() + eg_time,
            overall_best_seq,
            overall_best_score,
            eta0=0.7,
            eval_every=60,
        )
        print(f"[{time.time():.1f}] after EG best score {overall_best_score:.6f}")

    # ------------------------------------------------------------------
    # 4.  Progressive up‑sampling with short EG bursts
    # ------------------------------------------------------------------
    while time.time() < deadline:
        cur_n = overall_best_seq.shape[0]
        target_n = cur_n * 2
        if target_n > 20000:
            break
        print(f"[{time.time():.1f}] Upsampling {cur_n} → {target_n}")
        up_seq = _upsample(overall_best_seq, target_n)
        up_score = _objective(up_seq)
        if up_score < overall_best_score:
            overall_best_seq = up_seq
            overall_best_score = up_score
            print(f"[{time.time():.1f}] upsampled new best {overall_best_score:.6f}")

        # a brief EG run on the larger vector
        burst_deadline = time.time() + min(8.0, (deadline - time.time()) * 0.1)
        overall_best_seq, overall_best_score = _exp_gradient_descent(
            overall_best_seq.copy(),
            burst_deadline,
            overall_best_seq,
            overall_best_score,
            eta0=0.5,
            eval_every=40,
        )
        print(f"[{time.time():.1f}] after EG burst best score {overall_best_score:.6f}")

    # ------------------------------------------------------------------
    # 5.  Final long‑run annealing polish
    # ------------------------------------------------------------------
    if time.time() < deadline:
        print(f"[{time.time():.1f}] Final annealing polish")
        overall_best_seq, overall_best_score = _pairwise_annealing(
            overall_best_seq.copy(),
            deadline,
            overall_best_seq,
            overall_best_score,
            rng,
            max_trials=None,
        )
        print(f"[{time.time():.1f}] after annealing best score {overall_best_score:.6f}")

    # ------------------------------------------------------------------
    # 6.  Return the result
    # ------------------------------------------------------------------
    print(
        f"Final best score (objective=2·n·max(conv)/sum²): {overall_best_score:.6f} "
        f"(length={len(overall_best_seq)})"
    )
    return overall_best_seq.tolist()
