"""
improved_step_function_search.py

A faster, more aggressive optimiser for the objective used by
``evaluate_sequence``::

    O(a) = 2·n·max(conv(a,a)) / (∑a)²   (scale‑invariant, we keep ∑a = 1)

Main ideas
---------

1. **Richer seed pool** – families: uniform, arcsine, Hann, edge‑weighted,
   generalized‑normal, symmetric power‑law, random Dirichlet (wide range of
   concentration) and sparse random vectors.  Parameter grids are much
   finer than in the previous version (p up to 12, q up to 5, etc.).

2. **Dual optimiser** – for each seed we run two complementary first‑order
   methods:

   * *Exponentiated gradient* (mirror‑descent) – multiplicative updates
   * *Additive projected gradient* – standard gradient step followed by an
     Euclidean projection onto the probability simplex.

   The sub‑gradient is the gradient of the currently maximal convolution
   entry, obtained in O(n log n) by an FFT based self‑convolution.

3. **Iterative up‑sampling** – after the best candidate is found we
   repeatedly double its length (linear interpolation, renormalisation) and
   polish it again.  Each up‑sampling step typically lowers the constant
   factor *c* in ``max(conv)=c/n`` and therefore reduces the final score.

4. **Final annealing** – a pairwise mass‑transfer simulated annealing with
   a temperature schedule that slowly cools to zero, allowing occasional
   uphill moves to escape shallow local minima.

The whole routine respects the 1000 s budget (2 s safety margin) and
returns a plain ``list[float]`` ready for ``evaluate_sequence``.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Simplex utilities
# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Euclidean projection of *v* onto the simplex {x ≥ 0, Σx = s}."""
    if s < 0:
        raise ValueError("simplex radius must be non‑negative")
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.where(u - cssv / (np.arange(1, len(v) + 1)) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# FFT based self‑convolution (full linear convolution)
# ----------------------------------------------------------------------
def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Return a * a (full linear convolution) using a real FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1) - 1).bit_length()   # next power of two
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[: 2 * n - 1]


# ----------------------------------------------------------------------
# Exact objective for probability vectors (∑a = 1)
# ----------------------------------------------------------------------
def _objective(a: np.ndarray) -> float:
    """2·n·max(conv(a,a))  (assumes ∑a = 1)."""
    n = a.shape[0]
    b = _fft_self_convolve(a)
    return float(2.0 * n * b.max())


# ----------------------------------------------------------------------
# Analytic families (all return a probability vector, sum = 1)
# ----------------------------------------------------------------------
def _distribution_uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _distribution_arcsine(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    vals /= vals.sum()
    return vals


def _distribution_hann(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    w = 0.5 - 0.5 * np.cos(2.0 * np.pi * i / (n - 1))
    w_sum = w.sum()
    if w_sum == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w_sum
    return w


def _distribution_edge_weighted(n: int, p: float) -> np.ndarray:
    """Weight ∝ (i·j)^‑p where i,j are distances to the two edges."""
    i = np.arange(n, dtype=np.float64) + 0.5
    j = (n - np.arange(n, dtype=np.float64) - 0.5)
    w = (i * j) ** (-p)
    w_sum = w.sum()
    if w_sum == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w_sum
    return w


def _distribution_generalized_normal(n: int, q: float) -> np.ndarray:
    """Density ∝ exp(‑|x‑½|^q) on [0,1]."""
    i = np.arange(n, dtype=np.float64) + 0.5
    x = i / n - 0.5
    w = np.exp(- (np.abs(x) / 0.5) ** q)
    w_sum = w.sum()
    if w_sum == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w_sum
    return w


def _distribution_power_law(n: int, p: float) -> np.ndarray:
    """Symmetric power‑law: (i+½)^‑p + (n‑i‑½)^‑p."""
    i = np.arange(n, dtype=np.float64) + 0.5
    w = (i ** (-p) + (n + 0.5 - i) ** (-p))
    w_sum = w.sum()
    if w_sum == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w_sum
    return w


def _distribution_random_sparse(
    n: int, rng: np.random.Generator, density: float = 0.015
) -> np.ndarray:
    """Sparse random vector with the given density."""
    k = max(2, int(n * density))
    idx = rng.choice(n, size=k, replace=False)
    vals = np.zeros(n, dtype=np.float64)
    vals[idx] = rng.random(k)
    if vals.sum() == 0.0:
        vals[0] = 1.0
    vals /= vals.sum()
    return vals


def _mix_with_uniform(base: np.ndarray, lam: float) -> np.ndarray:
    """Convex combination λ·uniform + (1‑λ)·base."""
    n = base.shape[0]
    uniform = np.full(n, 1.0 / n, dtype=np.float64)
    return lam * uniform + (1.0 - lam) * base


# ----------------------------------------------------------------------
# Sub‑gradient of the active convolution entry
# ----------------------------------------------------------------------
def _grad_of_max_conv(a: np.ndarray) -> tuple[np.ndarray, int]:
    """
    Compute the sub‑gradient of the currently maximal convolution entry.
    Returns (grad, k) where k is the index of the maximal entry.
    """
    n = a.shape[0]
    conv = _fft_self_convolve(a)
    k = int(np.argmax(conv))

    grad = np.zeros_like(a)
    if k < n:
        # overlap length = k+1
        grad[: k + 1] = 2.0 * a[k::-1]
    else:
        start = k - (n - 1)
        grad[start:] = 2.0 * a[start:][::-1]
    return grad, k


# ----------------------------------------------------------------------
# Exponentiated‑gradient (mirror‑descent) optimiser
# ----------------------------------------------------------------------
def _exp_gradient_descent(
    a0: np.ndarray,
    deadline: float,
    best_seq: np.ndarray,
    best_score: float,
    eta0: float = 0.9,
    eval_every: int = 40,
) -> tuple[np.ndarray, float]:
    """Multiplicative update a ← a·exp(‑η·grad) followed by ℓ₁‑normalisation."""
    a = a0.copy()
    n = a.shape[0]
    t = 0
    while time.time() < deadline:
        t += 1
        grad, _ = _grad_of_max_conv(a)
        eta = eta0 / math.sqrt(t)
        a = a * np.exp(-eta * grad)
        s = a.sum()
        if s == 0.0:
            a = np.full(n, 1.0 / n, dtype=np.float64)
        else:
            a /= s
        if t % eval_every == 0:
            cur = _objective(a)
            if cur < best_score:
                best_score = cur
                best_seq = a.copy()
                print(f"[{time.time():.1f}] exp‑GD it={t} new best {best_score:.6f}")
    return best_seq, best_score


# ----------------------------------------------------------------------
# Additive projected‑gradient optimiser
# ----------------------------------------------------------------------
def _additive_gradient_descent(
    a0: np.ndarray,
    deadline: float,
    best_seq: np.ndarray,
    best_score: float,
    eta0: float = 0.2,
    decay: float = 0.5,
    eval_every: int = 40,
) -> tuple[np.ndarray, float]:
    """Standard gradient step a ← Π_{simplex}(a − η·grad)."""
    a = a0.copy()
    n = a.shape[0]
    t = 0
    while time.time() < deadline:
        t += 1
        grad, _ = _grad_of_max_conv(a)
        eta = eta0 / (t ** decay)
        a = _project_onto_simplex(a - eta * grad)
        if t % eval_every == 0:
            cur = _objective(a)
            if cur < best_score:
                best_score = cur
                best_seq = a.copy()
                print(f"[{time.time():.1f}] add‑GD it={t} new best {best_score:.6f}")
    return best_seq, best_score


# ----------------------------------------------------------------------
# Pairwise mass‑transfer simulated annealing
# ----------------------------------------------------------------------
def _pairwise_annealing(
    a0: np.ndarray,
    deadline: float,
    best_seq: np.ndarray,
    best_score: float,
    rng: np.random.Generator,
) -> tuple[np.ndarray, float]:
    """Randomly move a fraction of mass from i → j; accept with temperature."""
    a = a0.copy()
    n = a.shape[0]
    start = time.time()
    total_time = deadline - start
    while time.time() < deadline:
        # choose source with non‑negligible mass
        src = rng.choice(np.where(a > 1e-14)[0])
        # choose distinct target
        tgt = rng.integers(n - 1)
        if tgt >= src:
            tgt += 1
        delta = a[src] * rng.random() * 0.4   # up to 40 % of source mass
        if delta < 1e-16:
            continue
        new_a = a.copy()
        new_a[src] -= delta
        new_a[tgt] += delta
        cur = _objective(new_a)

        # temperature decays linearly to zero
        elapsed = time.time() - start
        temperature = max(0.0, 1.0 - elapsed / total_time) * best_score * 0.02

        accept = cur < best_score
        if not accept and temperature > 0.0:
            prob = math.exp(-(cur - best_score) / temperature)
            accept = rng.random() < prob

        if accept:
            a = new_a
            if cur < best_score:
                best_score = cur
                best_seq = a.copy()
                print(
                    f"[{time.time():.1f}] anneal i={src}→j={tgt} δ={delta:.3e} new best {best_score:.6f}"
                )
    return best_seq, best_score


# ----------------------------------------------------------------------
# Linear up‑sampling (interpolation) while preserving probability mass
# ----------------------------------------------------------------------
def _upsample(a: np.ndarray, new_n: int) -> np.ndarray:
    """Interpolate a probability vector to a larger length."""
    old_n = a.shape[0]
    if new_n == old_n:
        return a.copy()
    x_old = (np.arange(old_n, dtype=np.float64) + 0.5) / old_n
    x_new = (np.arange(new_n, dtype=np.float64) + 0.5) / new_n
    a_new = np.interp(x_new, x_old, a, left=0.0, right=0.0)
    a_new = np.maximum(a_new, 0.0)
    s = a_new.sum()
    if s == 0.0:
        a_new[:] = 1.0 / new_n
    else:
        a_new /= s
    return a_new

            overall_best_score = score
            overall_best_seq = seq.copy()
            print(
                f"[{time.time():.1f}] *** NEW GLOBAL BEST {overall_best_score:.6f} (len={len(seq)})"
            )

    # ------------------------------------------------------------------
    # 1.  Diverse seed generation for several lengths
    # ------------------------------------------------------------------
    lengths = [4096, 8192, 16384]   # modest start, will be up‑sampled later
    p_grid = np.concatenate((
        np.logspace(-3.0, -0.2, 14, base=10),   # 0.001 → 0.63
        np.logspace(-0.2, 0.3, 14, base=10),    # 0.63 → 2
        np.linspace(0.1, 3.0, 14),               # linear region
        np.logspace(0.3, 1.0, 10, base=10)      # 2 → 10
    ))
    q_grid = np.linspace(0.3, 5.0, 22)          # generalized normal
    lam_grid = np.linspace(0.0, 1.0, 9)        # mixing with uniform
    power_grid = [0.5, 1, 1.5, 2, 3, 4, 5]     # power‑law exponent

    for n in lengths:
        if time.time() >= deadline:
            break
        print(f"[{time.time():.1f}] Generating seeds for length = {n}")

        seed_vecs: list[np.ndarray] = []

        # deterministic families
        seed_vecs.append(_distribution_uniform(n))
        seed_vecs.append(_distribution_arcsine(n))
        seed_vecs.append(_distribution_hann(n))

        # edge‑weighted family with mixing
        for p in p_grid:
            base = _distribution_edge_weighted(n, float(p))
            for lam in lam_grid:
                seed_vecs.append(_mix_with_uniform(base, float(lam)))

        # power‑law family (symmetric)
        for p in power_grid:
            seed_vecs.append(_distribution_power_law(n, float(p)))

        # generalized normal family
        for q in q_grid:
            seed_vecs.append(_distribution_generalized_normal(n, float(q)))

        # random Dirichlet seeds – wide concentration range
        for _ in range(10):
            log_alpha = rng.uniform(-3.0, 1.0)      # α ∈ [0.001, 10]
            alpha = 10 ** log_alpha
            seed_vecs.append(rng.dirichlet(np.full(n, alpha)))

        # sparse random seeds
        for _ in range(6):
            seed_vecs.append(_distribution_random_sparse(n, rng, density=0.015))

        # keep a modest pool of the best seeds (by objective)
        scored = [(_objective(v), v) for v in seed_vecs]
        scored.sort(key=lambda x: x[0])
        top_candidates = [v for _, v in scored[:30]]
        for v in top_candidates:
            _maybe_update(v, _objective(v))

    # ------------------------------------------------------------------
    # 2.  Incorporate a previously found construction if available
    # ------------------------------------------------------------------
    try:
        prev = np.asarray(height_sequence_1, dtype=np.float64)  # type: ignore[name-defined]
        if prev.ndim == 1 and prev.size > 0:
            prev = prev / prev.sum()
            _maybe_update(prev, _objective(prev))
            # also treat it as a seed for the next phase
            top_candidates.append(prev)
    except Exception:
        pass

    # ------------------------------------------------------------------
    # 3.  Guard against an empty result
    # ------------------------------------------------------------------
    if overall_best_seq is None:
        overall_best_seq = _distribution_uniform(4096)
        overall_best_score = _objective(overall_best_seq)
        print(f"[{time.time():.1f}] fallback uniform score={overall_best_score:.6f}")

    # ------------------------------------------------------------------
    # 4.  Short polishing on all retained seeds (additive + exponential)
    # ------------------------------------------------------------------
    remaining = deadline - time.time()
    if remaining > 20.0 and top_candidates:
        per_seed = max(0.2, remaining * 0.25 / max(len(top_candidates), 1))
        print(f"[{time.time():.1f}] Quick polishing on {len(top_candidates)} seeds (≈{per_seed:.2f}s each)")
        for idx, seed_vec in enumerate(top_candidates):
            if time.time() >= deadline:
                break
            sub_dead = time.time() + per_seed
            # exponential GD
            seq_exp, sc_exp = _exp_gradient_descent(
                seed_vec.copy(),
                sub_dead,
                overall_best_seq,
                overall_best_score,
                eta0=0.8,
                eval_every=30,
            )
            _maybe_update(seq_exp, sc_exp)

            # additive GD
            if time.time() >= deadline:
                break
            seq_add, sc_add = _additive_gradient_descent(
                seed_vec.copy(),
                sub_dead,
                overall_best_seq,
                overall_best_score,
                eta0=0.1,
                eval_every=30,
            )
            _maybe_update(seq_add, sc_add)

    # ------------------------------------------------------------------
    # 5.  Longer polishing of the current global best (exponential GD)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        sub_dead = time.time() + (deadline - time.time()) * 0.30
        seq, sc = _exp_gradient_descent(
            overall_best_seq.copy(),
            sub_dead,
            overall_best_seq,
            overall_best_score,
            eta0=0.6,
            eval_every=50,
        )
        _maybe_update(seq, sc)

    # ------------------------------------------------------------------
    # 6.  Repeated up‑sampling + polishing loop (as long as time permits)
    # ------------------------------------------------------------------
    max_len = 200_000               # keep FFT size reasonable
    while time.time() < deadline and overall_best_seq.shape[0] < max_len:
        cur_n = overall_best_seq.shape[0]
        new_n = min(2 * cur_n, max_len)
        if new_n == cur_n:
            break
        print(f"[{time.time():.1f}] Upsampling {cur_n} → {new_n}")
        up = _upsample(overall_best_seq, new_n)
        up_score = _objective(up)
        _maybe_update(up, up_score)

        # polish the upsampled vector – allocate half of remaining time
        sub_dead = time.time() + (deadline - time.time()) * 0.45
        up_seq, up_sc = _exp_gradient_descent(
            up.copy(),
            sub_dead,
            overall_best_seq,
            overall_best_score,
            eta0=0.7,
            eval_every=40,
        )
        _maybe_update(up_seq, up_sc)

        # another short additive pass (helps escaping plateaus)
        if time.time() < deadline:
            sub_dead2 = time.time() + (deadline - time.time()) * 0.30
            up_seq2, up_sc2 = _additive_gradient_descent(
                overall_best_seq.copy(),
                sub_dead2,
                overall_best_seq,
                overall_best_score,
                eta0=0.12,
                eval_every=40,
            )
            _maybe_update(up_seq2, up_sc2)

    # ------------------------------------------------------------------
    # 7.  Final simulated‑annealing polishing
    # ------------------------------------------------------------------
    if time.time() < deadline:
        print(f"[{time.time():.1f}] Final annealing phase")
        seq, sc = _pairwise_annealing(
            overall_best_seq.copy(),
            deadline,
            overall_best_seq,
            overall_best_score,
            rng,
        )
        _maybe_update(seq, sc)

    # ------------------------------------------------------------------
    print(
        f"Final best score (2·n·max(conv)) = {overall_best_score:.6f} "
        f"(length = {len(overall_best_seq)})"
    )
    # Return a plain Python list – evaluate_sequence expects a list
    return overall_best_seq.tolist()
