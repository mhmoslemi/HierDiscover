def _smooth_adam_optimize(a0, deadline, best_seq, best_score, eta0=0.5):
    n = len(a0)
    # initialize logits x
    epsilon_small = 1e-12
    x = np.log(a0 + epsilon_small)  # shape n
    m = np.zeros_like(x)
    v = np.zeros_like(x)
    beta1 = 0.9
    beta2 = 0.999
    t = 0
    while time.time() < deadline:
        t += 1
        # compute a = softmax(x)
        # stable: subtract max
        mx = np.max(x)
        exp_x = np.exp(x - mx)
        a = exp_x / exp_x.sum()
        # compute full conv
        conv = _fft_self_convolve(a)
        # compute alpha based on schedule
        if t < 2000:
            alpha = 8.0
        elif t < 8000:
            alpha = 30.0
        else:
            alpha = 80.0
        # compute smoothed weights
        maxc = conv.max()
        exp_terms = np.exp(alpha * (conv - maxc))
        sum_exp = exp_terms.sum()
        w = exp_terms / sum_exp
        # compute gradient w.r.t a
        a_rev = a[::-1]
        grad_temp = _fft_convolve_general(w, a_rev)  # length 3n-2
        grad_a = grad_temp[n-1:n-1+n] * (4.0 * n)
        # gradient w.r.t x
        dot = np.dot(a, grad_a)
        grad_x = a * (grad_a - dot)
        # Adam update
        m = beta1 * m + (1 - beta1) * grad_x
        v = beta2 * v + (1 - beta2) * (grad_x * grad_x)
        m_hat = m / (1 - beta1**t)
        v_hat = v / (1 - beta2**t)
        # step size schedule: could decay
        lr = eta0 / (1 + 1e-5 * t)  # or something
        # update x
        x -= lr * m_hat / (np.sqrt(v_hat) + 1e-8)

        if t % 50 == 0:

                best_seq = a.copy()
                print(f"[{time.time():.1f}] Adam iter={t} new best {best_score:.6f}")

    la = len(a)
    lb = len(b)
    n = la + lb - 1
    size = 1 << ((n - 1).bit_length())
    A = np.fft.rfft(a, n=size)

    conv = np.fft.irfft(A * B, n=size)

2. **Smooth “log‑sum‑exp’’ Adam optimisation** – the exact objective
   = 2·n·max(a∗a) is non‑smooth.  We replace the max by a smooth
   approximation
        Mα(a) = (1/α)·log Σi exp(α·(a∗a)i)
   (α grows from a small value to a large one).  The gradient of Mα
   w.r.t. the probabilities a is obtained analytically via a
   convolution, and we optimise the *logits* x (a = softmax(x)) with an
   Adam‑style update.  This gives much more informative directions than
   the simple active‑set subgradient used previously.

3. **Upsampling & final combinatorial cleanup** – after the Adam phase the
   best distribution is linearly interpolated to a larger length (up to
   20 k points) and the Adam optimiser is run again, exploiting the fact
   that the continuous limit of the problem has a strictly lower bound.
   Finally a cheap pairwise‐swap sweep and simulated annealing clean up any
   remaining spikes.

The whole procedure respects the 1000 s budget, printing progress and
returning a plain Python list of heights.
"""

import time
import math
import numpy as np
from numpy.fft import rfft, irfft

# --------------------------------------------------------------
# Helper – projection (kept for compatibility, not used by the main optimiser)
def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x≥0, sum(x)=s}."""
    if s < 0:
        raise ValueError("simplex radius must be non‑negative")
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho_arr = np.where(u - cssv / (np.arange(1, len(v) + 1)) > 0)[0]
    if rho_arr.size == 0:
        theta = 0.0
    else:
        rho = rho_arr[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# --------------------------------------------------------------
# FFT‑based convolutions -------------------------------------------------
def _next_pow2(x: int) -> int:
    """Return the next power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a (real FFT, length 2·n‑1)."""
    n = a.shape[0]
    need = 2 * n - 1
    size = _next_pow2(need)
    A = rfft(a, n=size)
    conv = irfft(A * A, n=size)
    return conv[:need]


def _fft_convolve_general(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full linear convolution of two arbitrary real vectors."""
    la, lb = a.shape[0], b.shape[0]
    need = la + lb - 1
    size = _next_pow2(need)
    A = rfft(a, n=size)
    B = rfft(b, n=size)
    conv = irfft(A * B, n=size)
    return conv[:need]


# --------------------------------------------------------------
# Exact objective (sum of a = 1)
def _objective(a: np.ndarray) -> float:
    """Exact objective 2·n·max(conv(a,a)) assuming sum(a)=1."""
    n = a.shape[0]
    conv = _fft_self_convolve(a)
    return 2.0 * n * conv.max()


# --------------------------------------------------------------
# Analytic seed families -------------------------------------------------
def _distribution_uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _distribution_arcsine(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    vals /= vals.sum()
    return vals


def _distribution_beta_sym(n: int, p: float) -> np.ndarray:
    """Symmetric Beta(p,p) on the grid."""
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    w = np.exp((p - 1) * np.log(x) + (p - 1) * np.log(1.0 - x))
    if w.sum() == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _distribution_beta_asym(n: int, a: float, b: float) -> np.ndarray:
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    logw = (a - 1) * np.log(x) + (b - 1) * np.log(1.0 - x)
    w = np.exp(logw - np.max(logw))
    if w.sum() == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _distribution_random_sparse(
    n: int, rng: np.random.Generator, density: float = 0.006
) -> np.ndarray:
    k = max(2, int(n * density))
    idx = rng.choice(n, size=k, replace=False)
    vals = np.zeros(n, dtype=np.float64)
    vals[idx] = rng.random(k)
    if vals.sum() == 0.0:
        vals[0] = 1.0
    vals /= vals.sum()
    return vals


def _distribution_dirichlet(n: int, rng: np.random.Generator) -> np.ndarray:
    alpha = rng.uniform(0.01, 2.0)
    a = rng.dirichlet(np.full(n, alpha))
    return a.astype(np.float64)


def _mix_with_uniform(base: np.ndarray, lam: float) -> np.ndarray:
    """lam·U + (1‑lam)·base, where U is the uniform distribution."""
    n = base.shape[0]
    uniform = np.full(n, 1.0 / n, dtype=np.float64)
    return lam * uniform + (1.0 - lam) * base


# --------------------------------------------------------------
# Upsampling helper ---------------------------------------------------
def _upsample(seq: np.ndarray, new_len: int) -> np.ndarray:
    """Linear interpolation of a probability vector to a larger length."""
    old_len = seq.shape[0]
    if new_len == old_len:
        return seq.copy()
    x_old = np.linspace(0.0, 1.0, old_len)
    x_new = np.linspace(0.0, 1.0, new_len)
    new_seq = np.interp(x_new, x_old, seq)
    new_seq = np.maximum(new_seq, 0.0)
    new_seq /= new_seq.sum()
    return new_seq


# --------------------------------------------------------------
# Smooth Adam optimiser (log‑softmax parametrisation)
def _adam_optimize(
    a0: np.ndarray,
    deadline: float,
    best_seq: np.ndarray,
    best_score: float,
    init_lr: float = 0.5,
) -> tuple[np.ndarray, float]:
    """
    Adam optimiser on the *smooth* surrogate
          fα(a) = 2·n·Mα(a),
    where Mα is the log‑sum‑exp approximation of max(a∗a).
    a is represented as softmax(logits) to stay on the simplex.
    """
    n = a0.shape[0]
    eps = 1e-12
    # initialise logits from the seed (add eps to avoid log(0))
    x = np.log(a0 + eps)
    m = np.zeros_like(x)
    v = np.zeros_like(x)
    beta1 = 0.9
    beta2 = 0.999
    t = 0
    no_improve = 0
    rng = np.random.default_rng()
    # schedule of smoothing parameter α (tuples: (step_limit, α))
    schedule = [(2000, 8.0), (8000, 30.0), (float("inf"), 80.0)]

    while time.time() < deadline and t < 800_000:
        t += 1

        # softmax -> a
        mx = np.max(x)
        exp_x = np.exp(x - mx)
        a = exp_x / exp_x.sum()

        # full self‑convolution
        conv = _fft_self_convolve(a)
        maxc = conv.max()                      # exact max for fast objective
        cur_score = 2.0 * n * maxc              # exact objective value

        # keep best (exact) solution
        if cur_score + 1e-14 < best_score:
            best_score = cur_score
            best_seq = a.copy()
            no_improve = 0
            print(f"[{time.time():.1f}] Adam it={t:,} new global best {best_score:.6f}")
        else:
            no_improve += 1

        # pick α according to schedule
        for limit, alpha in schedule:
            if t <= limit:
                α = alpha
                break

        # smooth weights w_i = exp(α·(conv_i−maxc)) / Σ …
        exp_terms = np.exp(α * (conv - maxc))
        sum_exp = exp_terms.sum()
        w = exp_terms / sum_exp                     # shape 2·n‑1, sums to 1

        # gradient wrt a: 4·n·cross‑correlation(w, a)
        a_rev = a[::-1]
        corr = _fft_convolve_general(w, a_rev)      # length 3·n‑2
        grad_a = corr[n - 1 : n - 1 + n] * (4.0 * n)  # keep middle n entries

        # gradient wrt logits x (Entropy mirror)
        dot = np.dot(a, grad_a)
        grad_x = a * (grad_a - dot)

        # Adam update (bias‑corrected)
        m = beta1 * m + (1.0 - beta1) * grad_x
        v = beta2 * v + (1.0 - beta2) * (grad_x * grad_x)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)

        lr = init_lr / (1.0 + 1e-5 * t)  # slowly decaying learning rate
        x -= lr * m_hat / (np.sqrt(v_hat) + 1e-8)

        # occasional random jitter to escape plateaus
        if no_improve >= 2500:
            jitter = rng.normal(scale=0.004, size=n)
            x += jitter
            no_improve = 0

    return best_seq, best_score


# --------------------------------------------------------------
# Pairwise deterministic refinement (same as before)
def _pairwise_swap_local(
    a: np.ndarray,
    deadline: float,
    best_seq: np.ndarray,
    best_score: float,
    max_trials: int | None = None,
) -> tuple[np.ndarray, float]:
    n = a.shape[0]
    trials = 0
    rng = np.random.default_rng()
    while time.time() < deadline:
        if max_trials is not None and trials >= max_trials:
            break
        trials += 1

        src = np.where(a > 1e-12)[0]
        if src.size == 0:
            break
        i = rng.choice(src)

        j = rng.integers(n - 1)
        if j >= i:
            j += 1

        delta = a[i] * rng.random() * 0.33
        if delta < 1e-14:
            continue

        new_a = a.copy()
        new_a[i] -= delta
        new_a[j] += delta

        cur = _objective(new_a)
        if cur + 1e-14 < best_score:
            a = new_a
            best_score = cur
            best_seq = a.copy()
            print(f"[{time.time():.1f}] swap {i}->{j} δ={delta:.3e} new best {best_score:.6f}")

    return best_seq, best_score


# --------------------------------------------------------------
# Cheap simulated annealing (same as before)
def _random_local_search(
    a: np.ndarray,
    deadline: float,
    best_seq: np.ndarray,
    best_score: float,
    init_temp: float = 0.008,
    decay: float = 0.9995,
) -> tuple[np.ndarray, float]:
    n = a.shape[0]
    rng = np.random.default_rng()
    T = init_temp
    while time.time() < deadline:
        i = rng.integers(n)
        j = rng.integers(n - 1)
        if j >= i:
            j += 1
        delta = a[i] * rng.random() * 0.28
        if delta < 1e-14:
            continue
        new_a = a.copy()
        new_a[i] -= delta
        new_a[j] += delta
        cur = _objective(new_a)
        if cur + 1e-14 < best_score:
            a = new_a
            best_score = cur
            best_seq = a.copy()
            print(f"[{time.time():.1f}] SA improve {i}->{j} Δ={delta:.3e} best {best_score:.6f}")
        else:
            if rng.random() < math.exp(-(cur - best_score) / max(T, 1e-12)):
                a = new_a
        T *= decay
    return best_seq, best_score

def propose_candidate(seed: int = 42, budget_s: int = 1000) -> list[float]:
    """
    Search for a non‑negative step function that minimises ``evaluate_sequence``.
    The whole search respects the given time budget (default 1000 s).
    Returns a plain Python list of heights.
    """
    rng = np.random.default_rng(seed)
    start = time.time()
    deadline = start + budget_s - 2.0   # safety margin

    overall_best_seq: np.ndarray | None = None
    overall_best_score = float("inf")
    seed_pool: list[np.ndarray] = []

    # ------------------------------------------------------------------
    # 1️⃣  Generate a diverse set of analytic / random seeds (moderate n)
    lengths = [2000, 2500, 3000, 3500, 4000, 5000]
    for n in lengths:
        if time.time() >= deadline:
            break
        print(f"[{time.time():.1f}] sampling seeds of length {n}")

        # deterministic families
        for cand in (_distribution_uniform(n), _distribution_arcsine(n)):
            sc = _objective(cand)
            seed_pool.append(cand.copy())
            if sc < overall_best_score:
                overall_best_score, overall_best_seq = sc, cand.copy()
                print(f"[{time.time():.1f}] new global best (deterministic) {sc:.6f}")

        # parameterised families
        for p in np.concatenate(
            (np.logspace(-2, -0.2, 12), np.logspace(-0.2, 0.5, 12))
        ):
            if time.time() >= deadline:
                break
            beta_sym = _distribution_beta_sym(n, p)
            seed_pool.append(beta_sym)
            sc = _objective(beta_sym)
            if sc < overall_best_score:
                overall_best_score, overall_best_seq = sc, beta_sym.copy()
                print(f"[{time.time():.1f}] new best symmetric‑Beta n={n} p={p:.3g} {sc:.6f}")

            # mix with uniform (lam in [0,0.8] coarse)
            for lam in np.linspace(0.0, 0.8, 5):
                cand = _mix_with_uniform(beta_sym, lam)
                sc = _objective(cand)
                seed_pool.append(cand)
                if sc < overall_best_score:
                    overall_best_score, overall_best_seq = sc, cand.copy()
                    print(
                        f"[{time.time():.1f}] mix β p={p:.3g} lam={lam:.2f} score={sc:.6f}"
                    )

        # asymmetric beta (two‑parameter random search)
        for _ in range(8):
            a = rng.uniform(0.001, 3.0)
            b = rng.uniform(0.001, 3.0)
            cand = _distribution_beta_asym(n, a, b)
            sc = _objective(cand)
            seed_pool.append(cand)
            if sc < overall_best_score:
                overall_best_score, overall_best_seq = sc, cand.copy()
                print(
                    f"[{time.time():.1f}] asym‑beta a={a:.3g} b={b:.3g} score={sc:.6f}"
                )

        # random Dirichlet
        for _ in range(6):
            cand = _distribution_dirichlet(n, rng)
            sc = _objective(cand)
            seed_pool.append(cand)
            if sc < overall_best_score:
                overall_best_score, overall_best_seq = sc, cand.copy()
                print(f"[{time.time():.1f}] Dirichlet seed score={sc:.6f}")

        # sparse random
        for _ in range(6):
            cand = _distribution_random_sparse(n, rng, density=0.006)
            sc = _objective(cand)
            seed_pool.append(cand)
            if sc < overall_best_score:
                overall_best_score, overall_best_seq = sc, cand.copy()
                print(f"[{time.time():.1f}] sparse seed score={sc:.6f}")

        # Keep only the top‑15 seeds to bound memory
        seed_pool.sort(key=lambda arr: _objective(arr))
        seed_pool = seed_pool[:15]

    # ------------------------------------------------------------------
    # 2️⃣  If a previous construction exists, inject it
    try:
        prev = np.asarray(height_sequence_1, dtype=np.float64)  # type: ignore[name-defined]
        if prev.ndim == 1 and prev.size > 0:
            prev = np.maximum(prev, 0.0)
            prev = prev / prev.sum()
            sc = _objective(prev)
            if sc < overall_best_score:
                overall_best_score, overall_best_seq = sc, prev.copy()
                print(f"[{time.time():.1f}] previous best improved to {sc:.6f}")
            seed_pool.append(prev)
    except Exception:
        pass

    # ------------------------------------------------------------------
    # Guard – shouldn't happen, but just in case
    if overall_best_seq is None:
        n = 3000
        overall_best_seq = _distribution_uniform(n)
        overall_best_score = _objective(overall_best_seq)
        print(f"[{time.time():.1f}] fallback uniform score={overall_best_score:.6f}")

    # ------------------------------------------------------------------
    # 3️⃣  Smooth‑Adam refinement on a few strongest seeds
    remaining = deadline - time.time()
    if remaining > 60.0 and seed_pool:
        # pick up to 5 best seeds for intensive optimisation
        seed_pool.sort(key=lambda arr: _objective(arr))
        seeds = [seq for seq in seed_pool[:5]]
        per_seed_time = remaining * 0.5 / max(1, len(seeds))
        print(
            f"[{time.time():.1f}] starting Adam on {len(seeds)} seeds, ~{per_seed_time:.1f}s each"
        )
        for idx, s in enumerate(seeds):
            if time.time() >= deadline:
                break
            sub_deadline = time.time() + per_seed_time
            a_opt, sc_opt = _adam_optimize(
                s.copy(),
                sub_deadline,
                best_seq=overall_best_seq,
                best_score=overall_best_score,
                init_lr=0.6,
            )
            if sc_opt < overall_best_score:
                overall_best_score, overall_best_seq = sc_opt, a_opt.copy()
                print(
                    f"[{time.time():.1f}] after Adam seed {idx} new global best {overall_best_score:.6f}"
                )

    # ------------------------------------------------------------------
    # 4️⃣  Upsample to a larger length and continue refining
    remaining = deadline - time.time()
    if remaining > 120.0:
        target_len = min(18000, len(overall_best_seq) * 2)
        print(f"[{time.time():.1f}] up‑sampling to {target_len} points")
        up_seq = _upsample(overall_best_seq, target_len)

        sub_deadline = time.time() + remaining * 0.6
        up_seq_opt, up_score = _adam_optimize(
            up_seq,
            sub_deadline,
            best_seq=overall_best_seq,
            best_score=overall_best_score,
            init_lr=0.4,
        )
        if up_score < overall_best_score:
            overall_best_score, overall_best_seq = up_score, up_seq_opt.copy()
            print(
                f"[{time.time():.1f}] up‑sampled Adam gave new best {overall_best_score:.6f}"
            )

    # ------------------------------------------------------------------
    # 5️⃣  Deterministic pairwise cleanup (fast)
    if time.time() < deadline:
        print(f"[{time.time():.1f}] deterministic pairwise refinement")
        overall_best_seq, overall_best_score = _pairwise_swap_local(
            overall_best_seq.copy(),
            deadline,
            best_seq=overall_best_seq,
            best_score=overall_best_score,
            max_trials=250_000,
        )

    # ------------------------------------------------------------------
    # 6️⃣  Final cheap simulated‑annealing clean‑up
    if time.time() < deadline:
        print(f"[{time.time():.1f}] final simulated‑annealing clean‑up")
        overall_best_seq, overall_best_score = _random_local_search(
            overall_best_seq.copy(),
            deadline,
            best_seq=overall_best_seq,
            best_score=overall_best_score,
            init_temp=0.009,
            decay=0.9994,
        )

    # ------------------------------------------------------------------
    print(
        f"Final best score: {overall_best_score:.6f} (length={len(overall_best_seq)})"
    )
    # Convert to plain Python list – evaluate_sequence will perform its own clipping
    return overall_best_seq.tolist()
