def _softmax_descent(a0, deadline, best_seq, best_score, init_lr=0.4, init_alpha=5.0, max_alpha=200.0, alpha_growth=1.02, eval_interval=200, patience=5000):
    a = a0.copy()
    n = a.shape[0]
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    t=0
    last_improve_it = 0
    while time.time() < deadline and t < 1_000_000:
        t+=1
        conv = _fft_self_convolve(a)
        max_c = conv.max()
        # update alpha gradually
        alpha = min(max_alpha, init_alpha * (alpha_growth ** (t // 500)))
        w_raw = np.exp(alpha * (conv - max_c))  # stable because max at 0
        w_sum = w_raw.sum()
        if w_sum == 0:
            w = np.full_like(w_raw, 1.0 / w_raw.size)
        else:
            w = w_raw / w_sum

        # gradient
        grad_full = _fft_convolve(a, w)   # length = n + (2n-1) -1 = 3n-2
        offset = len(w) - n  # = n-1
        grad = grad_full[offset : offset + n]

        # Adam updates
        m = 0.9 * m + 0.1 * grad
        v = 0.999 * v + 0.001 * (grad * grad)
        bias_correction1 = 1 - 0.9 ** t
        bias_correction2 = 1 - 0.999 ** t
        m_hat = m / bias_correction1
        v_hat = v / bias_correction2

        # learning rate schedules
        lr = init_lr / np.sqrt(t)
        # Exponentiated gradient step: a = a * exp(-lr * grad_eff)
        # Use gradient direction opposite to ascent: We want to reduce max, gradient points direction of increase of smoothed max; we compute grad of smoothed max (increase). So to minimize we need step opposite sign: -lr * m_hat / (sqrt(v_hat)+eps)
        update = -lr * m_hat / (np.sqrt(v_hat) + 1e-12)
        a = a * np.exp(update)

        # projection
        a = _project_onto_simplex(a)

        if t % eval_interval == 0:
            cur = _objective(a)
            if cur + 1e-14 < best_score:
                best_score = cur
                best_seq = a.copy()
                last_improve_it = t
                print(f"[{time.time():.1f}] descent it={t:,} new best {best_score:.6f}")
            else:
                # if no improvement for a while, inject jitter
                if t - last_improve_it > patience:
                    jitter = np.random.default_rng().exponential(scale=0.005, size=n)
                    a = a + jitter
                    a = np.maximum(a, 0.0)
                    a = a / a.sum()
                    last_improve_it = t
                    print(f"[{time.time():.1f}] descent jitter injected at it={t:,}")

    return best_seq, best_score

   and random Dirichlet / sparse vectors for a moderate size (≈ 4 k).

2. **Smooth max‑gradient descent** – minimise a *soft‑max* surrogate
   of the ∞‑norm of the self‑convolution:
         f(a) = (1/α)·log Σₖ exp(α·(a∗a)ₖ) .
   Its gradient is a correlation of the current distribution with the
   soft‑max weights.  An Adam‑style exponentiated‑gradient (mirror)
   update is applied and the iterate is projected onto the probability
   simplex after every step.

3. **Upsample** – the best distribution is linearly interpolated to a
   larger length (up to ≈ 18 k) and the same smooth‑max descent is
   continued – larger dimensions give a strictly lower theoretical
   bound.

4. **Final cleanup** – a deterministic pairwise mass‑transfer sweep
   followed by a cheap simulated‑annealing pass removes any remaining
   spikes.

All heavy work uses FFT‑based convolutions (O(n log n)) and fits well
inside the 1000 s budget.
"""

import time
import math
import numpy as np
from numpy.fft import rfft, irfft

# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Euclidean projection of v onto {x≥0, Σx = s}."""
    if s < 0:
        raise ValueError("Simplex radius must be non‑negative")
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho_arr = np.where(u - cssv / (np.arange(1, len(v) + 1)) > 0)[0]
    if rho_arr.size == 0:
        theta = 0.0
    else:
        rho = rho_arr[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Linear convolution a*a (real FFT, length 2·n‑1)."""
    n = a.shape[0]
    need = 2 * n - 1
    size = 1 << ((need - 1).bit_length())
    A = rfft(a, n=size)
    conv = irfft(A * A, n=size)
    return conv[:need]


# ----------------------------------------------------------------------
def _fft_convolve(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Linear convolution of two real vectors via FFT."""
    n, m = a.shape[0], b.shape[0]
    need = n + m - 1
    size = 1 << ((need - 1).bit_length())
    A = rfft(a, n=size)
    B = rfft(b, n=size)
    conv = irfft(A * B, n=size)
    return conv[:need]


# ----------------------------------------------------------------------
def _objective(a: np.ndarray) -> float:
    """Exact objective 2·n·max(conv(a,a)) for a with Σa = 1."""
    n = a.shape[0]
    b = _fft_self_convolve(a)
    return 2.0 * n * b.max()


# ----------------------------------------------------------------------
# Analytic families ----------------------------------------------------
def _distribution_uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _distribution_arcsine(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    vals /= vals.sum()
    return vals


def _distribution_beta_sym(n: int, p: float) -> np.ndarray:
    """Symmetric Beta(p,p)."""
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    w = np.exp((p - 1) * np.log(x) + (p - 1) * np.log(1.0 - x))
    if w.sum() == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _distribution_beta_asym(n: int, a: float, b: float) -> np.ndarray:
    """Asymmetric Beta(a,b)."""
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    logw = (a - 1) * np.log(x) + (b - 1) * np.log(1.0 - x)
    w = np.exp(logw - np.max(logw))
    if w.sum() == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _distribution_edge_weighted(n: int, p: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64) + 0.5
    j = (n - np.arange(n, dtype=np.float64) - 0.5)
    w = (i * j) ** (-p)
    if w.sum() == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _distribution_edge_inversepow(n: int, p: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64) + 0.5
    j = (n - np.arange(n, dtype=np.float64) - 0.5)
    w = i ** (-p) + j ** (-p)
    if w.sum() == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _distribution_sine_weighted(n: int, p: float) -> np.ndarray:
    theta = np.pi * (np.arange(n, dtype=np.float64) + 0.5) / n
    w = np.sin(theta) ** p
    if w.sum() == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _distribution_random_sparse(
    n: int, rng: np.random.Generator, density: float = 0.006
) -> np.ndarray:
    k = max(2, int(n * density))
    idx = rng.choice(n, size=k, replace=False)
    vals = np.zeros(n, dtype=np.float64)
    vals[idx] = rng.random(k)
    if vals.sum() == 0.0:
        vals[0] = 1.0
    vals /= vals.sum()
    return vals


def _distribution_dirichlet(n: int, rng: np.random.Generator) -> np.ndarray:
    alpha = rng.uniform(0.01, 2.0)
    a = rng.dirichlet(np.full(n, alpha))
    return a.astype(np.float64)


def _mix_with_uniform(base: np.ndarray, lam: float) -> np.ndarray:
    """lam·U + (1‑lam)·base, U is the uniform distribution."""
    n = base.shape[0]
    uniform = np.full(n, 1.0 / n, dtype=np.float64)
    return lam * uniform + (1.0 - lam) * base


# ----------------------------------------------------------------------
def _smooth_max_descent(
    a0: np.ndarray,
    deadline: float,
    best_seq: np.ndarray,
    best_score: float,
    init_lr: float = 0.6,
    init_alpha: float = 5.0,
    max_alpha: float = 200.0,
    alpha_growth: float = 1.02,
    eval_interval: int = 200,
    patience: int = 3000,
) -> tuple[np.ndarray, float]:
    """
    Adam‑style exponentiated‑gradient descent on the soft‑max surrogate
    of max(conv(a,a)).  Returns the best sequence seen.
    """
    a = a0.copy()
    n = a.shape[0]
    m = np.zeros_like(a)          # first‑moment accumulator
    v = np.zeros_like(a)          # second‑moment accumulator
    t = 0
    last_improve_it = 0
    rng = np.random.default_rng()
    eps = 1e-12

    while time.time() < deadline and t < 800_000:
        t += 1

        # 1) self‑convolution
        conv = _fft_self_convolve(a)
        max_c = conv.max()

        # 2) soft‑max weights (stable)
        alpha = min(max_alpha, init_alpha * (alpha_growth ** (t // 500)))
        w_raw = np.exp(alpha * (conv - max_c))          # max entry becomes 1
        w_sum = w_raw.sum()
        if w_sum == 0.0:
            w = np.full_like(w_raw, 1.0 / w_raw.size)
        else:
            w = w_raw / w_sum

        # 3) gradient = correlation a ⋆ w (linear slice of a∗w)
        grad_full = _fft_convolve(a, w)                 # length 3n‑2
        offset = len(w) - n                            # = n‑1
        grad = grad_full[offset : offset + n]

        # 4) Adam update (exponentiated‑gradient)
        m = 0.9 * m + 0.1 * grad
        v = 0.999 * v + 0.001 * (grad * grad)

        bias_correction1 = 1 - 0.9 ** t
        bias_correction2 = 1 - 0.999 ** t
        m_hat = m / bias_correction1
        v_hat = v / bias_correction2

        lr = init_lr / math.sqrt(t)                    # decaying learning rate
        update = -lr * m_hat / (np.sqrt(v_hat) + eps)   # move opposite to gradient
        a = a * np.exp(update)                         # mirror (entropy) step

        # 5) projection onto the simplex Σa = 1 , a ≥ 0
        a = _project_onto_simplex(a)

        # 6) occasional evaluation
        if t % eval_interval == 0:
            cur = _objective(a)
            if cur + 1e-14 < best_score:
                best_score, best_seq = cur, a.copy()
                last_improve_it = t
                print(f"[{time.time():.1f}] it={t:,} new best {best_score:.6f}")
            elif t - last_improve_it > patience:
                # inject a small random jitter to escape possible plateaus
                jitter = rng.exponential(scale=0.004, size=n)
                a = a + jitter
                a = np.maximum(a, 0.0)
                a = _project_onto_simplex(a)
                last_improve_it = t
                print(f"[{time.time():.1f}] jitter injected at it={t:,}")

    return best_seq, best_score


# ----------------------------------------------------------------------
def _pairwise_swap_local(
    a: np.ndarray,
    deadline: float,
    best_seq: np.ndarray,
    best_score: float,
    max_trials: int | None = None,
) -> tuple[np.ndarray, float]:
    """Greedy random transfer of a fraction of mass between two coordinates."""
    n = a.shape[0]
    trials = 0
    rng = np.random.default_rng()
    while time.time() < deadline:
        if max_trials is not None and trials >= max_trials:
            break
        trials += 1
        src = np.where(a > 1e-12)[0]
        if src.size == 0:
            break
        i = rng.choice(src)
        j = rng.integers(n - 1)
        if j >= i:
            j += 1
        delta = a[i] * rng.random() * 0.33
        if delta < 1e-14:
            continue
        new_a = a.copy()
        new_a[i] -= delta
        new_a[j] += delta
        cur = _objective(new_a)
        if cur + 1e-14 < best_score:
            a = new_a
            best_score = cur
            best_seq = a.copy()
            print(f"[{time.time():.1f}] swap {i}->{j} Δ={delta:.3e} new best {best_score:.6f}")
    return best_seq, best_score


# ----------------------------------------------------------------------
def _random_local_search(
    a: np.ndarray,
    deadline: float,
    best_seq: np.ndarray,
    best_score: float,
    init_temp: float = 0.008,
    decay: float = 0.9995,
) -> tuple[np.ndarray, float]:
    """Metropolis‑style simulated annealing using random mass transfers."""
    n = a.shape[0]
    rng = np.random.default_rng()
    T = init_temp
    while time.time() < deadline:
        i = rng.integers(n)
        j = rng.integers(n - 1)
        if j >= i:
            j += 1
        delta = a[i] * rng.random() * 0.28
        if delta < 1e-14:
            continue
        new_a = a.copy()
        new_a[i] -= delta
        new_a[j] += delta
        cur = _objective(new_a)
        if cur + 1e-14 < best_score:
            a = new_a
            best_score = cur
            best_seq = a.copy()
            print(f"[{time.time():.1f}] SA improve {i}->{j} Δ={delta:.3e} best {best_score:.6f}")
        else:
            if rng.random() < math.exp(-(cur - best_score) / max(T, 1e-12)):
                a = new_a
        T *= decay
    return best_seq, best_score


# ----------------------------------------------------------------------
def _upsample(seq: np.ndarray, new_len: int) -> np.ndarray:
    """Linear interpolation of a simplex‑vector to a larger length."""
    old_len = seq.shape[0]
    if new_len == old_len:
        return seq.copy()
    x_old = np.linspace(0.0, 1.0, old_len)
    x_new = np.linspace(0.0, 1.0, new_len)
    new_seq = np.interp(x_new, x_old, seq)
    new_seq = np.maximum(new_seq, 0.0)
    new_seq /= new_seq.sum()
    return new_seq


# ----------------------------------------------------------------------
def _initial_seeds(n: int, rng: np.random.Generator) -> list[np.ndarray]:
    """Generate a diverse pool of candidate distributions."""
    seeds: list[np.ndarray] = []

    # deterministic families
    seeds.append(_distribution_uniform(n))
    seeds.append(_distribution_arcsine(n))

    # symmetric beta (p varies)
    for p in [0.2, 0.3, 0.5, 0.7, 1.0, 1.5, 2.0]:
        seeds.append(_distribution_beta_sym(n, p))

    # edge‑weighted / edge‑inverse‑power
    for p in [0.5, 1.0, 1.5, 2.0]:
        seeds.append(_distribution_edge_weighted(n, p))
        seeds.append(_distribution_edge_inversepow(n, p))

    # sine‑weighted
    for p in [1.0, 2.0, 3.0]:
        seeds.append(_distribution_sine_weighted(n, p))

    # asymmetric beta (few random draws)
    for _ in range(6):
        a = rng.uniform(0.001, 3.0)
        b = rng.uniform(0.001, 3.0)
        seeds.append(_distribution_beta_asym(n, a, b))

    # random Dirichlet
    for _ in range(6):
        seeds.append(_distribution_dirichlet(n, rng))

    # sparse random patterns
    for _ in range(6):
        seeds.append(_distribution_random_sparse(n, rng, density=0.012))

    # mixtures with uniform
    for _ in range(6):
        base = _distribution_arcsine(n)
        lam = rng.uniform(0.0, 0.9)
        seeds.append(_mix_with_uniform(base, lam))

    return seeds


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: int = 1000) -> list[float]:
    """
    Search for a non‑negative step function that minimises ``evaluate_sequence``.
    The whole search respects the given time budget (default 1000 s).
    Returns a plain Python list of heights.
    """
    rng = np.random.default_rng(seed)
    start = time.time()
    deadline = start + budget_s - 2.0    # safety margin

    overall_best_seq: np.ndarray | None = None
    overall_best_score = float("inf")
    top_candidates: list[tuple[float, np.ndarray]] = []

    # ------------------------------------------------------------------
    # 1. Seed generation (moderate length)
    n0 = 4000 if budget_s >= 500 else 2500
    print(f"[{time.time():.1f}] Generating seeds of length {n0}")

    seeds = _initial_seeds(n0, rng)
    for seq in seeds:
        sc = _objective(seq)
        top_candidates.append((sc, seq.copy()))
        if sc < overall_best_score:
            overall_best_score, overall_best_seq = sc, seq.copy()
            print(f"[{time.time():.1f}] New global best (seed) {sc:.6f}")

    # keep a compact set of the very best seeds
    top_candidates.sort(key=lambda x: x[0])
    top_candidates = top_candidates[:25]

    # ------------------------------------------------------------------
    # 2. If a previously discovered construction exists, include it
    try:
        prev = np.asarray(height_sequence_1, dtype=np.float64)  # type: ignore[name-defined]
        if prev.ndim == 1 and prev.size > 0:
            prev = np.maximum(prev, 0.0)
            prev = prev / prev.sum()
            sc = _objective(prev)
            if sc < overall_best_score:
                overall_best_score, overall_best_seq = sc, prev.copy()
                print(f"[{time.time():.1f}] Previous best incorporated, score={sc:.6f}")
            top_candidates.append((sc, prev.copy()))
    except Exception:
        pass

    # ------------------------------------------------------------------
    # 3. Heavy smooth‑max descent on a handful of strongest seeds
    remaining = deadline - time.time()
    if remaining > 40.0 and top_candidates:
        seeds_to_refine = [seq for _, seq in sorted(top_candidates, key=lambda x: x[0])[:7]]
        per_seed_time = remaining * 0.55 / max(1, len(seeds_to_refine))
        print(f"[{time.time():.1f}] Refinement stage – {len(seeds_to_refine)} seeds, ~{per_seed_time:.1f}s each")
        for idx, s in enumerate(seeds_to_refine):
            if time.time() >= deadline:
                break
            sub_deadline = time.time() + per_seed_time
            best_seq, best_score = _smooth_max_descent(
                s,
                sub_deadline,
                best_seq=overall_best_seq,
                best_score=overall_best_score,
                init_lr=0.7,
                init_alpha=5.0,
                max_alpha=150.0,
                eval_interval=180,
                patience=2500,
            )
            if best_score < overall_best_score:
                overall_best_score, overall_best_seq = best_score, best_seq.copy()
                print(f"[{time.time():.1f}] after seed {idx} new global best {overall_best_score:.6f}")

    # ------------------------------------------------------------------
    # 4. Upsample to a larger length if time permits
    remaining = deadline - time.time()
    if remaining > 150.0:
        large_n = min(18000, len(overall_best_seq) * 2)
        print(f"[{time.time():.1f}] Upsampling to {large_n} points")
        up_seq = _upsample(overall_best_seq, large_n)
        sub_deadline = time.time() + remaining * 0.65
        best_seq, best_score = _smooth_max_descent(
            up_seq,
            sub_deadline,
            best_seq=overall_best_seq,
            best_score=overall_best_score,
            init_lr=0.5,
            init_alpha=8.0,
            max_alpha=200.0,
            eval_interval=150,
            patience=3000,
        )
        if best_score < overall_best_score:
            overall_best_score, overall_best_seq = best_score, best_seq.copy()
            print(f"[{time.time():.1f}] Upsampled descent yielded new best {overall_best_score:.6f}")

    # ------------------------------------------------------------------
    # 5. Deterministic pairwise refinement
    if time.time() < deadline:
        print(f"[{time.time():.1f}] Pairwise deterministic refinement")
        overall_best_seq, overall_best_score = _pairwise_swap_local(
            overall_best_seq.copy(),
            deadline,
            best_seq=overall_best_seq,
            best_score=overall_best_score,
            max_trials=300_000,
        )

    # ------------------------------------------------------------------
    # 6. Final cheap simulated‑annealing clean‑up
    if time.time() < deadline:
        print(f"[{time.time():.1f}] Final simulated‑annealing clean‑up")
        overall_best_seq, overall_best_score = _random_local_search(
            overall_best_seq.copy(),
            deadline,
            best_seq=overall_best_seq,
            best_score=overall_best_score,
            init_temp=0.008,
            decay=0.9995,
        )

    # ------------------------------------------------------------------
    print(
        f"Final best score: {overall_best_score:.6f} (length={len(overall_best_seq)})"
    )
    # Return a plain Python list; evaluate_sequence will perform its own clipping.
    return overall_best_seq.tolist()
